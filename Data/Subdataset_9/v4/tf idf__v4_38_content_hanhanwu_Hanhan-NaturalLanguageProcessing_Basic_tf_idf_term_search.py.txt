import nltk
import json

def main():
    # Specify the path to the JSON file containing Google+ posts
    json_file_path = '[change to your googleplus_posts.json location]'
    
    # Load the data from the JSON file
    data = json.loads(open(json_file_path).read())
    
    # Define the query terms to search for
    query_terms = ['mobile']
    
    # Extract the content of each activity and convert it to lowercase
    activities = [activity['object']['content'].lower().split() for activity in data if activity['object']['content'] != '']
    
    # Create a TextCollection object from the activities
    text_collection = nltk.TextCollection(activities)
    
    # Initialize an empty list to store relevant activities
    relevant_activities = []
    
    # Iterate through each activity
    for i in range(len(activities)):
        # Initialize the score for the current activity
        score = 0
        
        # Calculate the TF-IDF score for each query term in the current activity
        for term in query_terms:
            score += text_collection.tf_idf(term.lower(), activities[i])
        
        # If the score is greater than 0, consider the activity relevant
        if score > 0:
            relevant_activities.append({'score': score, 'title': data[i]['title'], 'url': data[i]['url']})
    
    # Sort the relevant activities based on their scores in descending order
    relevant_activities = sorted(relevant_activities, key=lambda a: a['score'], reverse=True)
    
    # Print the relevant activities along with their titles, URLs, and scores
    for ra in relevant_activities:
        print('Title:', ra['title'])
        print('URL:', ra['url'])
        print('Score:', ra['score'])

if __name__ == '__main__':
    main()
