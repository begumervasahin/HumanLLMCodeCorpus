# Initialize variables
n = 2
rest = []
primes = [1]  # 1 is not considered prime
count = 0

# Loop through numbers up to 3001 to find primes
while n <= 3001:
    i = 0
    m = primes[i]
    
    # Check divisibility with previously found primes
    while m < n:
        p = n % m
        rest.append(p)
        count = rest.count(0)
        i += 1
        
        # Check if all primes have been checked or if n is composite
        if i >= len(primes) or count > 1:
            break
        m = primes[i]
        
    # If n is prime, add it to the list of primes and print
    if count == 1:
        print(n, "is prime")
        primes.append(n)
    
    # Increment n by 2 if n is odd, else increment by 1
    if n == 2:
        n += 1
    else:
        n += 2
    
    # Reset the list of remainders for the next iteration
    rest = []
