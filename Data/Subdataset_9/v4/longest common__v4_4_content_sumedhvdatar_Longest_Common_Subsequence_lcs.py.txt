# Convert strings to arrays
row = len(s2) + 2
column = len(s1) + 2

# Initialize table
table = [['' for _ in range(column)] for _ in range(row)]
table[0][0] = "L"
table[0][1] = "P"
table[1][0] = "P"

# Populate table with s1 and s2
for i, letter in enumerate(s1, start=2):
    table[0][i] = letter
    table[1][i] = 0

for i, letter in enumerate(s2, start=2):
    table[i][0] = letter
    table[i][1] = 0

# Fill the table with LCS values
for i in range(2, row):
    for j in range(2, column):
        if table[i][0] == table[0][j]:
            element_in_the_preceding_diagonal = table[i - 1][j - 1]
            table[i][j] = element_in_the_preceding_diagonal + 1
        else:
            table[i][j] = max(table[i][j - 1], table[i - 1][j])
