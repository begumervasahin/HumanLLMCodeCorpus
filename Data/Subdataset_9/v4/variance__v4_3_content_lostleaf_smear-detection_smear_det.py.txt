import numpy as np
import glob
import matplotlib.pyplot as plt
from collections import deque

# Define thresholds for variance
LOW_THRESHOLD = 200
HIGH_THRESHOLD = 600

# Function to perform breadth-first search
def bfs(variance_map, initial_positions):
    # Define directions to explore
    DIRECTIONS_X = [1, 0, -1, 0]
    DIRECTIONS_Y = [0, 1, 0, -1]
    
    # Initialize a set to keep track of visited positions
    visited_positions = set(initial_positions)
    
    # Initialize a queue for BFS
    queue = deque(initial_positions)
    
    # Initialize a mask with the same shape as the variance map
    mask = np.zeros_like(variance_map, dtype=np.uint8)
    
    # Mark positions with variance below the low threshold as foreground
    mask[variance_map < LOW_THRESHOLD] = 255
    
    # Perform BFS
    while queue:
        x, y = queue.popleft()
        # Explore neighboring positions
        for dx, dy in zip(DIRECTIONS_X, DIRECTIONS_Y):
            nx, ny = x + dx, y + dy
            # Check if the neighboring position satisfies conditions
            if (0 <= nx < variance_map.shape[0] and 0 <= ny < variance_map.shape[1] and
                    variance_map[nx, ny] < HIGH_THRESHOLD and (nx, ny) not in visited_positions):
                # Mark the position in the mask and add it to the visited set and queue
                visited_positions.add((nx, ny))
                mask[nx, ny] = 255
                queue.append((nx, ny))
    
    return mask

# Main function
def main():
    # Get file names of variance maps
    variance_files = glob.glob("*.npy")
    
    # Process each variance map
    for idx, filename in enumerate(variance_files):
        # Load variance map
        variance_map = np.load(filename)
        
        # Get positions with variance below the low threshold
        initial_positions = np.argwhere(variance_map < LOW_THRESHOLD).tolist()
        
        # Perform BFS to generate mask
        mask = bfs(variance_map, initial_positions)
        
        # Plot the mask
        plt.subplot(231 + idx)
        plt.imshow(mask, cmap="gray")
        plt.title(filename)
    
    # Show the plots
    plt.show()

# Entry point of the program
if __name__ == "__main__":
    main()
