import heapq

# Function to find the shortest path recursively
def shortest(v, path):
    if v.previous:
        path.append(v.previous.getId())
        shortest(v.previous, path)

# Dijkstra's algorithm implementation
def dijkstra(graph, start):
    # Setting the distance of the starting vertex to 0
    start.setDistance(0)
    
    # Creating a priority queue with tuples of (distance, vertex)
    unvisitedQueue = [(vertex.getDistance(), vertex) for vertex in graph]
    heapq.heapify(unvisitedQueue)
    
    # Main loop
    while len(unvisitedQueue):
        # Pop the vertex with the smallest distance from the priority queue
        aVertex = heapq.heappop(unvisitedQueue)
        current = aVertex[1]
        current.setVisited()
        
        # Iterating over adjacent vertices
        for next_vertex in current.adjacent:
            # If the next vertex is already visited, skip it
            if next_vertex.visited:
                continue
            
            # Calculate the new distance
            newDistance = current.getDistance() + current.getWeight(next_vertex)
            
            # Update distance and previous vertex if the new distance is shorter
            if newDistance < next_vertex.getDistance():
                next_vertex.setDistance(newDistance)
                next_vertex.setPrevious(current)
        
        # Clear the priority queue
        while len(unvisitedQueue):
            heapq.heappop(unvisitedQueue)
        
        # Rebuild the priority queue with unvisited vertices
        unvisitedQueue = [(vertex.getDistance(), vertex) for vertex in graph if not vertex.visited]
        heapq.heapify(unvisitedQueue)
