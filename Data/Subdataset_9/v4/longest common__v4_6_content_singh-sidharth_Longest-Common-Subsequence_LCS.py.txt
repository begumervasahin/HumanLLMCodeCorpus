def longest_common_subsequence(str1, str2):
    # Length of the first string
    len_str1 = len(str1)
    # Length of the second string
    len_str2 = len(str2)
    
    # Create a matrix to store the length of the longest common subsequence
    # Initialize the matrix with zeros
    lcs_matrix = [[0] * (len_str2 + 1) for _ in range(len_str1 + 1)]
    
    # Iterate over the characters of the strings and populate the matrix
    for i in range(1, len_str1 + 1):
        for j in range(1, len_str2 + 1):
            if str1[i - 1] == str2[j - 1]:
                # If characters match, increment the length of the LCS by 1
                lcs_matrix[i][j] = lcs_matrix[i - 1][j - 1] + 1
            else:
                # If characters don't match, take the maximum length from either
                # excluding the current character of str1 or str2
                lcs_matrix[i][j] = max(lcs_matrix[i - 1][j], lcs_matrix[i][j - 1])
    
    # Return the length of the longest common subsequence
    return lcs_matrix[len_str1][len_str2]

# Input strings from the user
string1 = input("Enter the first string: ")
string2 = input("Enter the second string: ")

# Calculate and print the length of the longest common subsequence
print("The length of the longest common subsequence is:", longest_common_subsequence(string1, string2))
