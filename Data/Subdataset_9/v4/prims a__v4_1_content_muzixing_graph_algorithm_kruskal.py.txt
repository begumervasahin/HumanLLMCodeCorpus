def kruskal(graph):
    # Check if the input graph is a dictionary
    assert type(graph) == dict
    
    # Get the nodes from the graph
    nodes = graph.keys()
    
    # Set to keep track of visited nodes
    visited = set()
    
    # List to store the minimum spanning tree path
    path = []
    
    # Variable to hold the next node
    next_node = None
    
    # Loop until all nodes are visited
    while len(visited) < len(nodes):
        # Set initial distance to infinity
        distance = float('inf')
        
        # Loop through all possible edges
        for source in nodes:
            for destination in nodes:
                # Skip if both nodes are visited or are the same
                if source in visited and destination in visited or source == destination:
                    continue
                
                # Check if the edge distance is smaller than the current minimum distance
                if graph[source][destination] < distance:
                    distance = graph[source][destination]
                    previous_node = source
                    next_node = destination
        
        # Add the edge with minimum distance to the path
        path.append((previous_node, next_node))
        
        # Mark both nodes as visited
        visited.add(previous_node)
        visited.add(next_node)
    
    return path
