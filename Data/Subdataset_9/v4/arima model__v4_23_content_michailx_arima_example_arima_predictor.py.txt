# Importing the ARIMA model from the statsmodels library
from statsmodels.tsa.arima_model import ARIMA

# Defining the time series data
time_series = [16, 20, 32, 40, 20, 18, 11, 21, 4, 6, 31, 48, 43, 49, 37]

# Creating an ARIMA model with specified order (p=4, d=1, q=1)
arima_model = ARIMA(time_series, order=(4, 1, 1))

# Fitting the model to the time series data
model_fit = arima_model.fit(disp=0)

# Generating predictions for time steps 16 to 19
predictions = model_fit.predict(start=16, end=19, typ='levels')

# Printing the predicted values
print(predictions)
