import common as c  # Assuming the 'common' module contains the 'minimum' and 'maximum' functions

def radix_sort(arr, verbose=0, desc=0):
    # Function to calculate the number of passes needed for Radix Sort
    def passes(number, base):
        count = 0
        while number:
            count += 1
            number = number // base
        return count

    # Function to extract a positional digit from a number
    def positional_digit(number, pos, base=10):
        count, remainder, is_valid = 0, 0, True
        while is_valid:
            count += 1
            remainder = number % base
            number = number // base
            if count > pos:
                is_valid = False
        return remainder

    # Function to distribute numbers into buckets based on a positional digit
    def bucketize(pos, base):
        n = len(arr)
        buckets = [[] for _ in range(n)]
        for i in range(n):
            digit = positional_digit(arr[i], pos, base)
            buckets[digit].append(arr[i])
            if verbose == 2:
                print("    Bucket:", i, " :: ", buckets)
        return [buckets[x][y] for x in range(len(buckets)) for y in range(len(buckets[x]))]

    base = 10  # Base for numbers (e.g., 10 for decimal)

    # Adjust array to make it non-negative
    smallest = c.minimum(arr)[0]
    if smallest:
        arr = [x - smallest for x in arr]

    # Find the largest number and calculate the number of passes needed
    largest = c.maximum(arr)[0]
    iter_count = passes(largest, base)

    # Perform Radix Sort
    for i in range(iter_count):
        arr = bucketize(i, base)
        if verbose:
            print("Iteration", i + 1, ":", arr)

    # Adjust array back to its original form
    if smallest:
        arr = [x + smallest for x in arr]

    # Optionally reverse the sorted array
    if desc:
        arr = arr[::-1]

    return arr
