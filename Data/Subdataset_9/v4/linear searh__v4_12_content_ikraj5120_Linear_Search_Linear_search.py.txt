import time
from matplotlib import pyplot as plt

test_case = [y for y in range(1000)]
result = []

# Function to perform linear search
def search(L, e):
    for i in range(len(L)):
        if L[i] == e:
            return True
        if L[i] > e:
            return False
    return False

# Function to test linear search on a list
def test(L):
    for j in range(0, len(L)):
        start_time = time.time()  # Record start time
        flag = search(L, j)  # Perform linear search
        end_time = time.time()  # Record end time
        if flag:
            result.append(end_time - start_time)  # Store execution time if found
    return result

# Function to display the results
def show_results(L, res):
    plt.plot(L, res, color='green', marker='o', linestyle='solid')
    plt.title("Linear Search Performance")
    plt.xlabel('List Elements')
    plt.ylabel('Execution Time (s)')
    plt.show()

# Test linear search on the test case
test(test_case)

# Display the results
show_results(test_case, result)
