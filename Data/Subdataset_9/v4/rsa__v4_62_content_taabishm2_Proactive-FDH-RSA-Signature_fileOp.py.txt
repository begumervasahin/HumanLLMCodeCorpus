import os

# Get the current working directory
save_path = os.getcwd() + "\RSAfiles\\"

# Function to write a list into a file
def write_list(filename, l):
    complete_name = save_path + filename + '.txt'
    with open(complete_name, 'w') as file:
        file.write(','.join(map(str, l)))

# Function to read a list from a file (reads only integers)
def read_list(filename):
    complete_name = save_path + filename + '.txt'
    with open(complete_name, 'r') as file:
        arr = file.read().split(',')
        return list(map(int, arr))

# Function to read a list from a file (reads non-integers)
def read_list_noint(filename):
    complete_name = save_path + filename + '.txt'
    with open(complete_name, 'r') as file:
        arr = file.read().split(',')
        return arr

# Function to read basic data from a file
def read_large_data(filename):
    msg = ""
    with open(save_path + filename + ".txt") as infile:
        for line in infile:
            msg += line
    return msg

# Function to read a binary file
def read_binary_file(filename):
    msg = ""
    with open(save_path + filename, 'rb') as infile:
        for line in infile:
            msg += str(line)
    return msg
