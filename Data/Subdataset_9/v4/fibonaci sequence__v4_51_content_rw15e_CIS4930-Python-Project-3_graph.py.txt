class Graph:
    def __init__(self):
        self.vertex_list = []  # Stores the vertices of the graph
        self.edge_list = []    # Stores the edges of the graph

    def add_vertex(self, name):
        """Add a vertex to the graph."""
        if name in self.vertex_list:
            print(f"A vertex with name '{name}' already exists.")
        else:
            self.vertex_list.append(name)

    def add_edge(self, start, end):
        """Add an edge between two vertices in the graph."""
        if end not in self.vertex_list:
            self.vertex_list.append(end)
        if start not in self.vertex_list:
            self.vertex_list.append(start)
        if start in self.vertex_list and end in self.vertex_list:
            self.edge_list.append((start, end))

    def remove_vertex(self, name):
        """Remove a vertex and its incident edges from the graph."""
        if name in self.vertex_list:
            self.vertex_list.remove(name)
            self.edge_list = [edge for edge in self.edge_list if name not in edge]
        else:
            print(f"Vertex '{name}' not found.")

    def remove_edge(self, start, end):
        """Remove an edge from the graph."""
        if (start, end) in self.edge_list:
            self.edge_list.remove((start, end))
        else:
            print(f"Edge '{start}' -> '{end}' not found.")

    def vertices(self):
        """Return a list of vertices in the graph."""
        return self.vertex_list

    def print_edges(self):
        """Print all edges in the graph."""
        for edge in self.edge_list:
            print(f"{edge[0]} -> {edge[1]}")

    def is_connected(self, start, end):
        """Check if there is a connection between two vertices."""
        return any(edge == (start, end) for edge in self.edge_list)

    def print_paths(self, start, end):
        """Print paths between two vertices in the graph."""
        for i, vertex in enumerate(self.vertex_list):
            if vertex == start:
                try:
                    path = " <-> ".join(self.vertex_list[i:i+6])
                    print(path)
                except IndexError:
                    pass


class UndirectedGraph(Graph):
    def __init__(self):
        super().__init__()  # Initialize the parent class attributes

    def add_edge(self, start, end):
        """Add an undirected edge between two vertices."""
        super().add_edge(start, end)  # Call the parent class method
        self.edge_list.append((end, start))  # Add the reverse edge

    def remove_edge(self, start, end):
        """Remove an undirected edge from the graph."""
        super().remove_edge(start, end)  # Call the parent class method
        self.edge_list.remove((end, start))  # Remove the reverse edge

    def print_edges(self):
        """Print all edges in the undirected graph."""
        for edge in self.edge_list:
            print(f"{edge[0]} <-> {edge[1]}")

    def print_paths(self, start, end):

        for i, vertex in enumerate(self.vertex_list):
            if vertex == start:
                try:
                    path = " <-> ".join(self.vertex_list[i:i+6])
                    print(path)
                except IndexError:
                    pass
