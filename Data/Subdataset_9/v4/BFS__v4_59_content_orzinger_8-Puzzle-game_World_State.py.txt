class World_State:
    """
    Class describing a world state for the 8-puzzle game.
    
    Attributes:
        state: Current state of the puzzle.
        operator: Operator which led to this state.
        parent: Parent state.
        depth: Depth from the root of the state.
        f_value: The sum of g(x) + h(x) for A* algorithm.
    """
    
    def __init__(self, _state, _operator, _parent, _depth, _f_value):
        """
        Initializes a world state object.
        
        Args:
            _state: Current state of the puzzle.
            _operator: Operator which led to this state.
            _parent: Parent state.
            _depth: Depth from the root of the state.
            _f_value: The sum of g(x) + h(x) for A* algorithm.
        """
        self.state = _state
        self.operator = _operator
        self.parent = _parent
        self.depth = _depth
        self.f_value = _f_value

    def __eq__(self, other):
        """
        Compares two world states for equality based on their states.
        
        Args:
            other: Another world state object.
        
        Returns:
            bool: True if the states are equal, False otherwise.
        """
        return self.state == other.state

    def __lt__(self, other):
        """
        Compares two world states based on their f_values for A* algorithm.
        
        Args:
            other: Another world state object.
        
        Returns:
            bool: True if the f_value of self is less than other, False otherwise.
        """
        return self.f_value < other.f_value
