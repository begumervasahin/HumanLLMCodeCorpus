import emmorphpy

# Initialize the EmMorphPy object
emmorph = emmorphpy.EmMorphPy()

# Open the file containing test words
with open('test/test_words.txt') as file:
    # Iterate over each line in the file
    for line in file:
        # Remove leading and trailing whitespace from the line
        word = line.strip()
        
        # Tokenize the word and obtain its morphological analysis
        analyses = emmorph.dstem(word, out_mode=list)
        
        # Iterate over each analysis
        for analysis in analyses:
            # Check if the analysis has all expected components
            if len(analysis) == 5:
                # Extract relevant components from the analysis
                base_form = analysis[0]
                pos_tag = analysis[1]
                lemma = analysis[2]
                
                # Print the original word, its lemma, base form, and POS tag
                print(word, lemma, base_form, pos_tag, sep='\t')
            else:
                # Print '<unknown>' if the analysis is incomplete
                print(word, '<unknown>', sep='\t')
        
        # Print an empty line for readability
        print()
