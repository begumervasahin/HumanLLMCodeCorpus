def xor_hex(hex1, hex2):
    bin1 = hex_to_binary(hex1)
    bin2 = hex_to_binary(hex2)
    xord = int(bin1, 2) ^ int(bin2, 2)
    hexed = hex(xord)[2:].zfill(8)  # Ensure 8-character hex representation
    return hexed

def hex_to_binary(hex):
    return bin(int(hex, 16))

def rotate_word(word):
    return word[1:] + word[:1]

def substitute_word(word):
    substituted_word = ()
    for i in range(4):
        row, col = get_row_col_indices(word[i])
        sbox_index = (row * 16) + col
        piece = format(sbox[sbox_index], '02x')
        substituted_word += (piece,)
    return substituted_word

def xor_with_rcon(word, rcon_value):
    return xor_hex(word, format(rcon_value, '02x'))

def get_row_col_indices(hex_val):
    if hex_val[0].isdigit():
        row = int(hex_val[0]) + 1
    else:
        row = ord(hex_val[0]) - ord('a') + 10
    if hex_val[1].isdigit():
        col = int(hex_val[1]) + 1
    else:
        col = ord(hex_val[1]) - ord('a') + 10
    return row, col

def pretty_print(key_schedule):
    print("\n\nKey Schedule: \n")
    for i, word in enumerate(key_schedule):
        print(f"w{i} = {' '.join(word)}")
