def problem1():
    # Ask the user to input two sequences without spaces between characters
    x = input("Enter the first sequence with no spaces between characters: ")
    y = input("Enter the second sequence with no spaces between characters: ")
    
    # Initialize a matrix to store computed values
    matrix = []
    for i in range(len(x)):
        row = []
        for j in range(len(y)):
            row.append(None)
        matrix.append(row)
    
    # Calculate the length of the longest common subsequence
    score = calculate_LCS_length(x, y, matrix, len(x), len(y))
    
    # Print the length of the longest common subsequence
    print("The length of the longest common sequence of:")
    print(x)
    print("and")
    print(y)
    print("is " + str(score))
    print()
    
    # Find and print one of the longest common subsequences
    print("A longest common subsequence of X and Y is:")
    sequence = find_LCS(x, y, matrix, len(x), len(y))
    print(sequence)

def find_LCS(sequenceOne, sequenceTwo, array, i, j):
    if i == 0 or j == 0:
        return ""
    
    if sequenceOne[i - 1] == sequenceTwo[j - 1]:
        return find_LCS(sequenceOne, sequenceTwo, array, i - 1, j - 1) + sequenceOne[i - 1]
    else:
        topScore = array[i - 2][j - 1] if i >= 2 else None
        leftScore = array[i - 1][j - 2] if j >= 2 else None
        
        if topScore is not None and leftScore is not None and topScore >= leftScore:
            return find_LCS(sequenceOne, sequenceTwo, array, i - 1, j)
        else:
            return find_LCS(sequenceOne, sequenceTwo, array, i, j - 1)

def calculate_LCS_length(sequenceOne, sequenceTwo, array, i, j):
    if i == 0 or j == 0:
        return 0
    elif array[i - 1][j - 1] is not None:
        return array[i - 1][j - 1]
    
    if sequenceOne[i - 1] == sequenceTwo[j - 1]:
        if array[i - 1][j - 1] is not None:
            return 1 + array[i - 1][j - 1]
        else:
            score = calculate_LCS_length(sequenceOne, sequenceTwo, array, i - 1, j - 1) + 1
            array[i - 1][j - 1] = score
            return score
    else:
        score = max(calculate_LCS_length(sequenceOne, sequenceTwo, array, i - 1, j),
                    calculate_LCS_length(sequenceOne, sequenceTwo, array, i, j - 1))
        array[i - 1][j - 1] = score
        return score

# Call the main function
problem1()
