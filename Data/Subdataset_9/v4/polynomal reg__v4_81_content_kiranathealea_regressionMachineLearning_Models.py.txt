from sklearn.metrics import mean_squared_error, r2_score

class LinearModel:
    """
    Trains a Linear Model and computes metrics.
    
    Parameters:
    model : Linear model to be trained.
    
    Attributes:
    intercept_ : Intercept of the linear model.
    coeffs_ : Coefficients of the model.
    rmse_ : Root Mean Squared Error (RMSE) of the model.
    r2_ : R-squared error of the model.
    """
    
    def __init__(self, model):
        self.model = model
    
    def compute_metrics(self, x, y):
        """
        Trains the model and computes metrics.
        
        Parameters:
        x : array-like, shape = [n_samples, n_features]
            Training samples.
        y : array-like, shape = [n_samples, n_target_values]
            Target values.
        
        Returns:
        self : Instance of self.
        """
        # Fit the model to the data
        self.model.fit(x, y)
        
        # Predict the target values
        y_predicted = self.model.predict(x)
        
        # Set attributes
        self.intercept_ = self.model.intercept_
        self.coef_ = self.model.coef_
        self.rmse_ = mean_squared_error(y, y_predicted)
        self.r2_ = r2_score(y, y_predicted)
        
        return self
