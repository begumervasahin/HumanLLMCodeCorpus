import zlib
import base64
from Crypto.PublicKey import RSA
from Crypto.Cipher import PKCS1_OAEP

def encrypt_blob(blob, public_key):
    # Import the RSA public key
    rsa_key = RSA.importKey(public_key)
    # Create a cipher object using PKCS1_OAEP padding
    rsa_key = PKCS1_OAEP.new(rsa_key)
    
    # Compress the blob using zlib
    blob = zlib.compress(blob)
    
    # Define chunk size for encryption
    chunk_size = 470
    offset = 0
    end_loop = False
    encrypted = b''  # Initialize an empty byte string for encrypted data
    
    # Encrypt data in chunks
    while not end_loop:
        chunk = blob[offset:offset + chunk_size]
        
        # Pad the chunk if its length is not a multiple of chunk_size
        if len(chunk) % chunk_size != 0:
            end_loop = True
            chunk += b" " * (chunk_size - len(chunk))
        
        # Encrypt the chunk and append to the encrypted data
        encrypted += rsa_key.encrypt(chunk)
        offset += chunk_size
    
    # Encode the encrypted data in base64
    encoded_encrypted = base64.b64encode(encrypted)
    return encoded_encrypted

# Read the public key from file
with open("TA_public_key.pem", "rb") as fd:
    public_key = fd.read()

# Read the unencrypted image blob from file
with open("rootbeer.jpg", "rb") as fd:
    unencrypted_blob = fd.read()

# Encrypt the image blob using the public key
encrypted_blob = encrypt_blob(unencrypted_blob, public_key)

# Write the encrypted image to file
with open("encrypted_img.jpg", "wb") as fd:
    fd.write(encrypted_blob)
    print('Stored encrypted image to', fd.name)
