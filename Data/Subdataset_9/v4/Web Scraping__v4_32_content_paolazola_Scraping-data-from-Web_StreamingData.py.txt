# Import necessary libraries
import datetime
import time
import pandas as pd
import requests
from lxml import html
from random import uniform

# Function to fetch real-time data from Yahoo Finance
def Yahoo_realTime(ticker, finish_time, delta):
    # Yahoo Finance URL for the given ticker
    url = 'https://it.finance.yahoo.com/quote/' + str(ticker) + '?p=' + str(ticker)
    
    # Headers for the request
    headers = {'User-Agent': 'Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/42.0.2311.90 Safari/537.36'}
    
    # Lists to store scraped data
    list_values = []
    list_timeNow = []
    list_timeYahoo = []
    
    # Continuously fetch data until finish_time is reached
    while datetime.datetime.now() < finish_time:
        # Make request to the Yahoo Finance page
        page = requests.get(url, headers=headers)
        
        # Random delay to simulate human-like behavior
        time.sleep(uniform(2, 3))
        
        # Parse the HTML content of the page
        parser = html.fromstring(page.content)
        
        # Extract the stock value
        value = parser.xpath('//span[@class="Trsdu(0.3s) Fw(b) Fz(36px) Mb(-4px) D(ib)"]//text()')[0]
        
        # Extract the time from Yahoo Finance
        yahooTime = parser.xpath('//span[@data-reactid="37"]//text()')[0]
        
        # Append fetched data to respective lists
        list_values.append(value)
        list_timeNow.append(datetime.datetime.now())
        list_timeYahoo.append(yahooTime)
        
        # Delay before fetching data again
        time.sleep(delta * 60)
    
    # Create a DataFrame to store the scraped data
    dataframe = pd.DataFrame({'scraperTime': list_timeNow, 'quotes': list_values, 'YahooPriceTime': list_timeYahoo})
    
    # Return the DataFrame
    return dataframe
