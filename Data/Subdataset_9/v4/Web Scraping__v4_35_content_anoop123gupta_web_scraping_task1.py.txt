import requests
import json
import os
from bs4 import BeautifulSoup

# Function to scrape the top-rated Indian movies from IMDb
def top_scrape_list():
    # Check if cache file exists
    if os.path.isfile("cache_file_for_task_1.json"):
        # If cache file exists, load data from it
        with open("cache_file_for_task_1.json", "r+") as data_file:
            data = json.load(data_file)
            return data
    else:
        # If cache file doesn't exist, scrape data from IMDb website
        endpoint = 'https://www.imdb.com/india/top-rated-indian-movies/'
        req = requests.get(endpoint)
        result = req.text
        soup = BeautifulSoup(result, 'html.parser')
        
        # Find main content area
        main = soup.find('div', class_='article')
        sub_main = main.find('div', class_='lister')
        tbody = sub_main.find('tbody', class_='lister-list')
        trs = tbody.find_all('tr')
        
        # Initialize variables
        movie_list = []  # List to store movie details
        rank = 0  # Variable to keep track of movie rank
        unique_years = []  # List to store unique years
        
        # Iterate through each movie row
        for tr in trs:
            rank += 1
            # Extract movie title and year
            td = tr.find('td', class_='titleColumn')
            title = td.find('a').text
            year = int(td.find('span').text[1:5])  # Extract year and convert to integer
            
            # Check if the year is not already added to the list of unique years
            if year not in unique_years:
                unique_years.append(year)
                
            # Extract movie URL
            movie_url = "https://www.imdb.com" + td.find('a').get('href')
            
            # Extract movie rating
            rating = tr.find('td', class_='ratingColumn imdbRating').text.strip()
            
            # Create dictionary for movie details and append it to the movie list
            movie_details = {'Title': title, 'Rank': rank, 'Year': year, 'URL': movie_url, 'Rating': rating}
            movie_list.append(movie_details)
        
        # Save the scraped data to cache file
        with open("cache_file_for_task_1.json", "w") as cache_file:
            json.dump(movie_list, cache_file)
        
        # Return the scraped movie data
        return movie_list

# Call the function to scrape top-rated Indian movies
movies = top_scrape_list()
