def calculate_yearly_balance(balance, annual_interest):
    """
    This function calculates the minimum monthly payment needed to pay off a debt 
    given the initial balance and annual interest rate, using bisection method.
    
    Args:
    balance (float): The initial debt amount.
    annual_interest (float): The annual interest rate.

    Returns:
    float: The minimum monthly payment required to pay off the debt.
    """
    # Initialize variables
    unpaid_balance = balance
    monthly_interest = annual_interest / 12
    monthly_lower_bound = unpaid_balance / 12
    upper_monthly_bound = (unpaid_balance * ((1 + monthly_interest) ** 12)) / 12
    
    # Loop until the unpaid balance is close to zero
    while abs(unpaid_balance) > 0.01:
        # Calculate the midpoint as the potential monthly payment
        result = (monthly_lower_bound + upper_monthly_bound) / 2
        unpaid_balance = balance
        
        # Calculate the unpaid balance after 12 months with the current monthly payment
        for _ in range(12):
            unpaid_balance -= result
            interest_amount = (annual_interest / 12) * unpaid_balance
            unpaid_balance += interest_amount
        
        # Adjust the bounds based on the result
        if unpaid_balance > 0.01:
            monthly_lower_bound = result
        elif unpaid_balance < -0.01:
            upper_monthly_bound = result
        else:
            # If the balance is within acceptable range, exit the loop
            break
    
    # Round the result to two decimal places and return
    return round(result, 2)
