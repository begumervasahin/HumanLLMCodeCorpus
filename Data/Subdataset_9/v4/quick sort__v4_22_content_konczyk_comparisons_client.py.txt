import argparse
import random
import sys
from pivots import first_index, last_index, median_of_three_index
from comparisons import Comparisons

# Dictionary to map pivot choices to corresponding functions
pivots_choice = {
    'first': first_index,
    'last': last_index,
    'median': median_of_three_index
}

# Create ArgumentParser object
parser = argparse.ArgumentParser(
    description='Comparisons client.',
    formatter_class=argparse.RawTextHelpFormatter
)

# Define mutually exclusive group for input options
group = parser.add_mutually_exclusive_group()
group.add_argument('-b', '--bound', type=int, help='Upper bound number')
group.add_argument('infile', nargs='?', type=argparse.FileType('r'),
                   default=sys.stdin,
                   help='Input file with numbers (default stdin)')

# Add argument for pivot choice
parser.add_argument('-p', '--pivot', choices=pivots_choice.keys(),
                    metavar='PIVOT', default='first',
                    help='first: use first element as a pivot (default)\n'
                         'last: use last element as a pivot\n'
                         'median: use median of three elements as a pivot')

# Parse command line arguments
args = parser.parse_args()

# Generate numbers based on input or from a file
if args.bound is not None:
    nums = list(range(1, args.bound + 1))
    random.shuffle(nums)
else:
    nums = [int(num) for num in args.infile]

# Initialize Comparisons object with numbers and chosen pivot function
comp = Comparisons(nums, pivots_choice[args.pivot])

# Print the number of comparisons
print(comp.count())
