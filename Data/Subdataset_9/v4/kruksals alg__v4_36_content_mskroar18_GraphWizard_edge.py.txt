class Edge:
    def __init__(self, vertex1, vertex2, weight):
        self.vertex1 = vertex1
        self.vertex2 = vertex2
        self.weight = weight
        self.extra = None
        self.selected = False

    def __repr__(self):
        return f"V1: {self.vertex1}  V2: {self.vertex2}  W: {self.weight}  SEL: {self.selected}"
