class Puzzle:
    """
    Represents a snapshot of a full-information puzzle, which may be solved, unsolved, or unsolvable.
    """

    def fail_fast(self):
        """
        Check if Puzzle self can never be extended to a solution.

        Returns:
            bool: True if the puzzle can't be solved, False otherwise.
        """
        return False

    def is_solved(self):
        """
        Check if Puzzle self is solved.

        Raises:
            NotImplementedError: This is an abstract method that must be implemented in a subclass.

        Returns:
            bool: True if the puzzle is solved, False otherwise.
        """
        raise NotImplementedError

    def extensions(self):
        """
        Generate legal extensions of Puzzle self.

        Raises:
            NotImplementedError: This is an abstract method that must be implemented in a subclass.

        Yields:
            Puzzle: A legal extension of the puzzle.
        """
        raise NotImplementedError
