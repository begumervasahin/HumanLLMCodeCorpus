# Importing necessary libraries
from optparse import OptionParser    # Library for parsing command-line parameters
from apriori import Apriori           # Custom Apriori algorithm implementation

if __name__ == '__main__':
    # Parsing command-line parameters
    optParser = OptionParser()
    optParser.add_option('-f', '--file',
                         dest='filePath',
                         help='Input a CSV file path',
                         type='string',
                         default=None)  # Default: no file path provided

    optParser.add_option('-s', '--minSupp',
                         dest='minSupp',
                         help='Minimum support threshold',
                         type='float',
                         default=0.10)  # Default minimum support threshold

    optParser.add_option('-c', '--minConf', dest='minConf',
                         help='Minimum confidence threshold',
                         type='float',
                         default=0.40)  # Default minimum confidence threshold

    optParser.add_option('-r', '--rhs', dest='rhs',
                         help='Right destination',
                         type='string',
                         default=None)  # Default: no right-hand side item provided

    # Parsing command-line arguments
    (options, args) = optParser.parse_args()

    # Extracting parameters from the command line
    filePath = options.filePath
    minSupp = options.minSupp
    minConf = options.minConf
    rhs = frozenset([options.rhs]) if options.rhs else None

    # Printing out the parameters for reference
    print("Parameters:")
    print(f" - File Path: {filePath}")
    print(f" - Minimum Support: {minSupp}")
    print(f" - Minimum Confidence: {minConf}")
    print(f" - Right-hand Side Item: {rhs}\n")

    # Running the Apriori algorithm with the specified parameters
    objApriori = Apriori(minSupp, minConf)
    itemCountDict, freqSet = objApriori.fit(filePath)

    # Printing frequent item sets found by the Apriori algorithm
    for key, value in freqSet.items():
        print(f'Frequent {key}-term set:')
        print('-' * 20)
        for itemset in value:
            print(list(itemset))
        print()

    # Generating association rules based on the specified right-hand side item
    if rhs:
        rules = objApriori.getSpecRules(rhs)
        print('-' * 20)
        print('Rules referring to {}'.format(list(rhs)))
        for key, value in rules.items():
            print('{} -> {}: {}'.format(list(key), list(rhs), value))
