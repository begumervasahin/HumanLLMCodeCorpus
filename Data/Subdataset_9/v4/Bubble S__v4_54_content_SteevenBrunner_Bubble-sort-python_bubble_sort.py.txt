import random
from time import time

# Generate a list of random numbers
max_range = 10001
unsorted_list = random.sample(range(1, max_range), max_range - 1)

# Bubble sort implementation
def bubble_sort(input_list):
    is_sorted = False
    while not is_sorted:
        is_sorted = True
        for i in range(len(input_list) - 1):
            if input_list[i] > input_list[i + 1]:
                input_list[i], input_list[i + 1] = input_list[i + 1], input_list[i]
                is_sorted = False
    return input_list

# Copy the unsorted list for sorting comparison
unsorted_list_cpy = unsorted_list[:]

# Measure time taken for bubble sort
start = time()
bubble_sort(unsorted_list_cpy)
print(f"bubble sort algorithm ends in {time() - start} seconds to sort a list of {max_range - 1} items.")

# Copy the unsorted list for system sort comparison
unsorted_list_cpy = unsorted_list[:]

# Measure time taken for system sort (Timsort)
start = time()
unsorted_list_cpy.sort()
print(f"system python sort() Timsort algorithm ends in {time() - start} seconds to sort a list of {max_range - 1} items.")
