from comparable import Comparable

class HuffElement(Comparable):
    """
    Represents an element used in the Huffman coding algorithm.
    Each instance stores information about a character, its frequency,
    and its Huffman code.
    """

    def __init__(self, char):
        """
        Initializes a HuffElement object with the given character.
        Initializes character frequency to zero and the code to an empty string.

        Args:
            char (str): The character associated with this HuffElement.
        """
        self._ch = char
        self._ch_freq = 0
        self._code = ""

    def inc_freq(self):
        """
        Increments the character frequency count.
        """
        self._ch_freq += 1

    def get_freq(self):
        """
        Returns the character frequency count.
        """
        return self._ch_freq

    def set_freq(self, count):
        """
        Sets the character frequency count.

        Args:
            count (int): The frequency count to set.
        """
        self._ch_freq = count

    def get_code(self):
        """
        Returns the Huffman code for the character.
        """
        return self._code

    def set_code(self, code):
        """
        Sets the Huffman code for the character.

        Args:
            code (str): The Huffman code to set.
        """
        self._code = code

    def get_char(self):
        """
        Returns the character associated with this HuffElement.
        """
        return self._ch

    def set_char(self, char):
        """
        Sets the character associated with this HuffElement.

        Args:
            char (str): The character to set.
        """
        self._ch = char

    def compare(self, other_huff_elem):
        """
        Compares two HuffElement objects based on their character frequency count.

        Args:
            other_huff_elem (HuffElement): The other HuffElement to compare with.

        Returns:
            int: 1 if this element has higher frequency, -1 if lower, 0 if equal.
        """
        if self._ch_freq > other_huff_elem.get_freq():
            return 1
        elif self._ch_freq < other_huff_elem.get_freq():
            return -1
        else:
            return 0

    def __str__(self):
        """
        Returns a string representation of this HuffElement.
        """
        return "Char: " + self._ch + " Code: " + self._code + " Count: " + str(self._ch_freq)
