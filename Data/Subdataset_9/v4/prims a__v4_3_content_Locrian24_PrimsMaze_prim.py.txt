import argparse
import random
import time
from PIL import Image

from mazes import Maze  # Assuming the mazes module contains the Maze class

def generate_image_maze(filename, size):
    start_time = time.time()
    print("Computing Maze and Saving...")

    # Adjust size if it's even to ensure odd-sized maze
    if size % 2 == 0:
        size += 1

    # Create a maze object
    maze = Maze(size)

    # Generate maze image data
    data = []
    for cell in maze.cells:
        if cell.IN or not cell.Wall:
            data.append((255, 255, 255))  # White color for open paths
        else:
            data.append((0, 0, 0))  # Black color for walls

    # Choose random starting and ending points
    start = random.randrange(1, size, 2)
    end = random.randrange(1, size, 2)
    data[start] = (255, 255, 255)  # Mark start point
    data[len(data) - 1 - end] = (255, 255, 255)  # Mark end point

    # Create image and save it
    img = Image.new('RGB', (size, size), color='black')
    img.putdata(data)
    img.save(filename)

    # Calculate and print elapsed time
    end_time = time.time()
    total_time = end_time - start_time
    print("Time elapsed:", total_time)

def main():
    # Parse command-line arguments
    parser = argparse.ArgumentParser()
    parser.add_argument("filename")
    parser.add_argument("size")
    args = parser.parse_args()
    
    size = int(args.size)

    # Check if size exceeds a certain limit
    if size > 500:
        print("Maze size is too large. Please choose a smaller size.")
        exit()

    # Generate maze image
    generate_image_maze(args.filename, size)

if __name__ == "__main__":
    main()
