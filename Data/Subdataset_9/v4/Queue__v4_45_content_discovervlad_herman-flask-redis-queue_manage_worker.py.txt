import unittest
import redis
from rq import Connection, Worker
from flask.cli import FlaskGroup
from project.worker import create_app

# Create Flask app instance
app = create_app()

# Create Flask CLI group
cli = FlaskGroup(create_app=create_app)

# Define CLI command to run worker
@cli.command('run_worker')
def run_worker():
    # Get Redis URL from Flask app config
    redis_url = app.config['REDIS_URL']
    # Create Redis connection
    redis_connection = redis.from_url(redis_url)
    # Establish connection context
    with Connection(redis_connection):
        # Create RQ worker with specified queues
        worker = Worker(app.config['QUEUES'])
        # Start worker
        worker.work()

# Entry point to execute CLI commands
if __name__ == '__main__':
    cli()
