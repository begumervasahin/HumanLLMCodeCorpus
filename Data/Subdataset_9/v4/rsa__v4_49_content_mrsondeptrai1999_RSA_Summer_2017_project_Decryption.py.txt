# Prompt the user to input the encoded message in numbers
cipher_text = input('Please enter the encoded message here in numbers form: ')

# Split the input into a list of integers
input_list = cipher_text.split()
input_list = [int(a) for a in input_list]

# Prompt the user to input passwords, public key, and private key
p_1 = input('Please enter your first password: ')
p_2 = input('Please enter your second password: ')
N = input('Please enter your Public Key number: ')
e = input('Please enter your Private Key number: ')

# Import necessary functions and modules
from fractions import gcd
from random import randint

# Define a function to check if a given number is a private key
def is_private_key(d):
    if gcd(e * d - 1, (p_1 - 1) * (p_2 - 1)) == (p_1 - 1) * (p_2 - 1):
        return True
    else:
        return False

# Loop to generate a valid private key
invalid = True
while invalid:
    key = randint(0, N)
    if is_private_key(key):
        invalid = False
    else:
        pass

# Function to decrypt the encoded message
def decrypt_1(input1):
    M = []
    for C in input1:
        M.append((C ** key) % N)
    return M

# Decrypt the message and convert it to characters
list_number = decrypt_1(input_list)
def decrypt_2(input2):
    decrypted_text = []
    for char_code in input2:
        decrypted_text.append(chr(char_code))
    return decrypted_text

# Join the characters to form the final decrypted message
final = ''.join(decrypt_2(list_number))

# Print the decrypted message and a thank-you message
print('\nDecrypted message:', final)
print('\nThank you for trusting our service!')
