# Importing SizerPy module
import SizerPy as sp

# Main function
if __name__ == '__main__':
    # Demonstration of resizing raster data set (using GDAL) by calling SizerPy
    args = sp.RasterParser()  # Parsing arguments for raster processing
    inRasterObj = sp.SizeRaster(args.inputRaster)  # Creating an object for input raster
    # Resizing the input raster and saving the result to an output raster
    inRasterObj.SizeBands(args.outRaster, factor=args.factor, d1Interpolate=args.d1Interpolate)
    print('done')  # Indicating completion of the resizing process
