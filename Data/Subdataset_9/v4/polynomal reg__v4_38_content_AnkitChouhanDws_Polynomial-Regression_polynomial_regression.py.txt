# Let's start by importing the necessary libraries
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd

# Now, we'll load our dataset
dataset = pd.read_csv('Path To Dataset')

# Extracting the features (X) and the target variable (y) from the dataset
X = dataset.iloc[:, 1:2].values
y = dataset.iloc[:, 2].values

# Importing the Linear Regression model from sklearn
from sklearn.linear_model import LinearRegression

# Creating a Linear Regression model and fitting it to our data
lin_reg = LinearRegression()
lin_reg.fit(X, y)

# Importing Polynomial Features and setting degree to 5
from sklearn.preprocessing import PolynomialFeatures
poly_reg = PolynomialFeatures(degree=5)

# Transforming our features to polynomial features
X_poly = poly_reg.fit_transform(X)

# Creating another Linear Regression model for polynomial regression and fitting it
lin_reg2 = LinearRegression()
lin_reg2.fit(X_poly, y)

# Visualizing the linear regression model
plt.scatter(X, y, color='red')
plt.plot(X, lin_reg.predict(X), color='blue')
plt.title('Truth or Bluff (Linear Regression)')
plt.xlabel('Position level')
plt.ylabel('Salary')
plt.show()

# Visualizing the polynomial regression model
X_grid = np.arange(min(X), max(X), 0.1)  # Creating a smoother curve
X_grid = X_grid.reshape(len(X_grid), 1)
plt.scatter(X, y, color='red')
plt.plot(X_grid, lin_reg2.predict(poly_reg.fit_transform(X_grid)), color='blue')
plt.title('Truth or Bluff (Polynomial Regression)')
plt.xlabel('Position level')
plt.ylabel('Salary')
plt.show()

# Asking user for input to predict salary using linear regression
level = float(input('Enter your Experience level: '))
print("Estimated Salary (Linear Regression):", lin_reg.predict([[level]])[0])

# Asking user for input to predict salary using polynomial regression
level = float(input('Enter your Experience level: '))
print("Estimated Salary (Polynomial Regression):", lin_reg2.predict(poly_reg.fit_transform([[level]]))[0])
