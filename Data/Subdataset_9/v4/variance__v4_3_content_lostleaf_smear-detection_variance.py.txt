import cv2
import numpy as np
import matplotlib.pyplot as plt
import glob
import sys
import os

# Function to calculate the variance of a set of images
def calculate_variance(file_names):
    # Reading the first image to determine dimensions
    img = cv2.imread(file_names[0], cv2.IMREAD_GRAYSCALE)
    img_sum = np.zeros_like(img, dtype=np.float64)
    img_sqr_sum = np.zeros_like(img, dtype=np.float64)
    
    # Calculating sum and squared sum of pixel values
    for name in file_names:
        img = cv2.imread(name, cv2.IMREAD_GRAYSCALE).astype(np.float64)        
        img_sum += img
        img_sqr_sum += img ** 2
    
    # Number of images
    n = len(file_names)
    
    # Calculating variance
    variance = img_sqr_sum / n - ((img_sum / n) ** 2)
    
    return variance

# Main function
def main(argv):
    # Checking for the correct number of arguments
    if len(argv) != 2:
        print("Usage: python %s [folder of images]" % argv[0])
        sys.exit(-1)
    
    # Extracting path from arguments
    path = argv[1]
    
    # Calculating variance
    variance = calculate_variance(glob.glob(os.path.join(path, "*.jpg")))
    
    # Displaying variance as an image
    plt.imshow(variance)
    plt.colorbar()
    plt.show()
    
    # Removing trailing slash if present
    if path[-1] == '/':
        path = path[:-1]
    
    # Extracting folder name
    folder_name = os.path.basename(path)
    
    # Saving variance as a .npy file
    np.save("%s_variance.npy" % folder_name, variance)

if __name__ == '__main__':
    main(sys.argv)
