import sys

# Function to compare actual and expected outputs
def verify(actual, expected):
    failed = 0
    for i in range(len(actual)):
        if actual[i] != expected[i]:
            print("Test case", i+1, "failed")
            failed += 1
    print("---------------------------------")
    if failed == 0:
        print("Passed all test cases!")
    else:
        print("Failed", failed, "/", len(actual), "test cases.")   
    print("---------------------------------")

# Main function
def main():
    # Check if the correct number of arguments is provided
    if len(sys.argv) != 2:
        sys.exit("Usage: `python judge.py EXPECTED < ACTUAL")        

    # Read actual output from stdin
    actual = [int(l) for l in sys.stdin]

    # Read expected output from file
    expected = None
    with open(sys.argv[1], 'r') as f:
        expected = [int(l) for l in f]

    # Check if actual and expected outputs have the same length
    if len(actual) != len(expected):
        sys.exit("Error: The expected and actual outputs have different lengths")

    # Verify the outputs
    verify(actual, expected)

# Entry point of the program
if __name__ == '__main__':
    main()
