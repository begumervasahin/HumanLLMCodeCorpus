import csv
from selenium import webdriver
import time

# Setting up proxy for the webdriver
PROXY = "5.58.85.32:43214"
chrome_options = webdriver.ChromeOptions()
chrome_options.add_argument('--proxy-server=%s' % PROXY)
chrome_options.add_argument('--ignore_certificate-errors')
chrome_options.add_argument('--ignore-ssl-errors')

# Constants for pagination and CSV file
MAX_PAGE_NUM = 16
MAX_PAGE_DIG = 3

# Opening CSV file to write data
with open("contacts.csv", "w") as csv_file:
    # Initializing webdriver
    driver = webdriver.Chrome(options=chrome_options)

    # Iterating through page numbers
    for i in range(1, MAX_PAGE_NUM + 1):
        # Formatting page number with leading zeros
        page_num = (MAX_PAGE_DIG - len(str(i))) * "0" + str(i)
        
        # Constructing URL for each page
        url = "https://www.pagesjaunes.fr/annuaire/chercherlespros?quoiqui=avocat%20droit%20du%20travail&ou=paris-75&proximite=0&quoiQuiInterprete=avocat%20droit%20du%20travail&contexte=V0RcB2CgpYFodNUZBe%2BLCQ%3D%3D&idOu=L07505600&page=" + page_num

        # Opening the webpage
        driver.get(url)
        
        # Adding a pause for page load
        time.sleep(10)
        
        # Extracting contact elements from the webpage
        contacts = driver.find_elements_by_xpath('//*[@id="bi-contact-06253114000031C0001"]')
        
        # Getting the number of contact items on the page
        num_page_items = len(contacts)
        
        # Writing contact information to CSV file
        for j in range(num_page_items):
            csv_file.write(contacts[j].text + "," + "\n")
        
    # Closing the webdriver
    driver.close()

# Printing completion message
print("Finished writing contacts to CSV.")
