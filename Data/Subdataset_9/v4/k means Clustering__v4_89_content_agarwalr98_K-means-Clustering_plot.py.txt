import numpy as np
import matplotlib.pyplot as plt
from matplotlib.lines import Line2D

# Display a message indicating K Means Clustering
print("========== K MEANS CLUSTERING ===========\n")

# Define colors for different classes
colors = ['red', 'blue', 'green']

# Count total data points
TotalData = 0
with open("Data/NonLinear/group03.txt") as myfile:
    next(myfile)
    for i in myfile:
        TotalData += 1

# Initialize an array to store data
index = 0
Data = np.zeros(shape=(int(TotalData), int(2)), dtype=float)

# Read data from file and store it in the array
with open("Data/NonLinear/group03.txt") as myfile:
    next(myfile)
    for i in myfile:
        i = i.split()
        Data[index][0] = float(i[0])
        Data[index][1] = float(i[1])
        index += 1

# Plot the data points with different colors based on class
index = 0
for i in Data:
    if index < 300:
        plt.plot(float(i[0]), float(i[1]), "o", color=colors[0])
    elif 300 <= index < 800:
        plt.plot(float(i[0]), float(i[1]), "o", color=colors[1])
    else:
        plt.plot(float(i[0]), float(i[1]), "o", color=colors[2])
    index += 1

# Create legend for the plot
points = [Line2D([0], [0], color=c) for c in colors]
labels = ['Class 1', 'Class 2', 'Class 3']
plt.legend(points, labels)

# Show the plot
plt.show()
