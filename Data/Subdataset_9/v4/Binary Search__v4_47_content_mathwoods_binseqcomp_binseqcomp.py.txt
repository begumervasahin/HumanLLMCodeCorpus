from __future__ import print_function
import random
import timeit
import matplotlib.pyplot as plt

def sequential_search(arr=None, x=None):
    """
    Perform sequential search to find the index of an element in the array.

    Args:
        arr (list): The input array.
        x (int): The element to search for.

    Returns:
        int: The index of the element if found, otherwise -1.
    """
    i = 0
    while i < len(arr):
        if arr[i] == x:
            return i
        i = i + 1
    return -1

def binary_search(arr=None, x=None):
    """
    Perform binary search to find the index of an element in the array.

    Args:
        arr (list): The input array.
        x (int): The element to search for.

    Returns:
        int: The index of the element if found, otherwise -1.
    """
    l, r = 0, len(arr)-1
    while l <= r:
        m = int((l+r)/2)
        if x > arr[m]:
            l = m+1
        if x < arr[m]:
            r = m-1
        if x == arr[m]:
            return m
    return -1

def wrapper(func, *args, **kwargs):
    """
    Function to create a wrapped function for timeit.

    Args:
        func (function): The function to be wrapped.
        *args: Variable length argument list for the function.
        **kwargs: Arbitrary keyword arguments for the function.

    Returns:
        function: The wrapped function.
    """
    def wrapped():
        return func(*args, **kwargs)
    return wrapped

print("\n" + '\x1B[1;7m'+ "SEQUENTIAL SEARCH VS BINARY SEARCH" + '\x1B[0m')
i, trial = 2, 1
list_results = []
timeit_loops = 10

while i <= 6:
    n = 10 ** i
    tup_num = tuple(range(0, n))
    print("\n" + '\x1B[4m'+ "ARRAY SIZE = %s (10^i where i = %s)" % (n, i) + '\x1B[0m')
    
    while trial <= 3:
        r = random.randint(0, n-1)
        print("TRIAL %s" % trial)
        print("Random generated value: %s" % r)
        
        print("Running SEQ SEARCH...", end="")
        wrapped = wrapper(sequential_search, tup_num, r)
        seq_time = timeit.timeit(wrapped, number=timeit_loops)
        print("DONE!")
        print("Total time for %s searches: %s seconds." % (timeit_loops, seq_time))
        
        print("Running BIN SEARCH...", end="")
        wrapped = wrapper(binary_search, tup_num, r)
        bin_time = timeit.timeit(wrapped, number=timeit_loops)
        print("DONE!")
        print("Total time for %s searches: %s seconds." % (timeit_loops, bin_time))
        
        list_results.append({'power': i, 'trial': trial, 'tup_size': n, 'rand': r,
                             'seq_time': seq_time, 'bin_time': bin_time, 'timeit_loops': timeit_loops})
        trial = trial + 1
    
    i = i + 1
    trial = 1

print("\n" + '\x1B[1;7m' + "To end program close graph window." + '\x1B[0m' + "\n")

lst_xcoordseq = []
lst_ycoordseq = []
lst_xcoordbin = []
lst_ycoordbin = []

for entry in list_results:
    lst_xcoordseq.append(entry['power'])
    lst_xcoordbin.append(entry['power'])
    lst_ycoordseq.append(entry['seq_time'])
    lst_ycoordbin.append(entry['bin_time'])

plt.plot(lst_xcoordseq, lst_ycoordseq, 'rD', label='Seq Search')
plt.plot(lst_xcoordbin, lst_ycoordbin, 'bo', label='Bin Search')

ymax = max(lst_ycoordseq)
plt.axis([0, 7, -0.1, ymax + 0.1])
plt.xlabel('Size of array (powers of 10)')
plt.ylabel('Search time (sec)')
fig = plt.gcf()
fig.canvas.set_window_title('Math Woods - Sequential vs. Binary Search')
plt.grid(True)
plt.legend(loc='upper left', numpoints=1)
plt.show()
