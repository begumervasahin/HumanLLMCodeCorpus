# Define the graph structure
graph = {
    "inicio": {"a": 6, "b": 2},
    "a": {"fim": 1},
    "b": {"a": 3, "fim": 5},
    "fim": {}
}

# Initialize constants and data structures
infinity = float("inf")
costs = {"a": 6, "b": 2, "fim": infinity}
parents = {"a": "inicio", "b": "inicio", "fim": None}
processed_nodes = []

# Function to find the node with the lowest cost among unprocessed nodes
def find_lowest_cost_node(costs, processed_nodes):
    lowest_cost = infinity
    lowest_cost_node = None
    for node in costs:
        cost = costs[node]
        if cost < lowest_cost and node not in processed_nodes:
            lowest_cost = cost
            lowest_cost_node = node
    return lowest_cost_node

# Find the lowest cost node and update costs and parents accordingly
node = find_lowest_cost_node(costs, processed_nodes)
while node is not None:
    cost = costs[node]
    neighbors = graph[node]
    for neighbor in neighbors:
        new_cost = cost + neighbors[neighbor]
        if costs[neighbor] > new_cost:
            costs[neighbor] = new_cost
            parents[neighbor] = node
    processed_nodes.append(node)
    node = find_lowest_cost_node(costs, processed_nodes)

# Print the resulting costs after processing the graph
print(costs)
