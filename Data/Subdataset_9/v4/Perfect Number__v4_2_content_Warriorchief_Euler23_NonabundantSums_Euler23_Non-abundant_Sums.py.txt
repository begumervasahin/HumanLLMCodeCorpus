import math

def find_sum_of_proper_factors(x):
    factors = []
    for i in range(1, math.ceil(x / 2) + 1):
        if x % i == 0:
            factors.append(i)
    return sum(factors)

abundants = []
for k in range(11, 28123):
    if k % 1000 == 0:
        print("Trying", k, "...")
    if find_sum_of_proper_factors(k) > k:
        abundants.append(k)

sum_of_2_abundants = set()
for p in range(len(abundants)):
    if p % 100 == 0:
        print("Passing through p being", p, "out of", len(abundants))
    for q in range(len(abundants)):
        if abundants[p] + abundants[q] < 28123:
            sum_of_2_abundants.add(abundants[p] + abundants[q])

output = 0
for z in range(28123):
    if z % 100 == 0:
        print("Passing through z being", z, "out of 28123")
    if z not in sum_of_2_abundants:
        output += z

print("The answer to this question should be", output)
