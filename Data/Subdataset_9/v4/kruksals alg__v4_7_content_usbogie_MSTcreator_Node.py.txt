class Node:
    """
    Represents a node in a graph.

    Attributes:
        xloc (int): The x-coordinate location of the node.
        yloc (int): The y-coordinate location of the node.
        kval (int): The value associated with the node.
        adjList (list): The list of adjacent nodes.
    """

    def __init__(self, xloc, yloc, kval, adjList):
        """
        Initializes a node with the given attributes.

        Parameters:
            xloc (int): The x-coordinate location of the node.
            yloc (int): The y-coordinate location of the node.
            kval (int): The value associated with the node.
            adjList (list): The list of adjacent nodes.
        """
        self.xloc = xloc
        self.yloc = yloc
        self.kval = kval
        self.adjList = adjList

    def __eq__(self, other):
        """
        Checks if this node is equal to another node.

        Parameters:
            other (Node): The other node to compare with.

        Returns:
            bool: True if the nodes are equal, False otherwise.
        """
        return (
            self.xloc == other.xloc
            and self.yloc == other.yloc
            and self.kval == other.kval
        )
