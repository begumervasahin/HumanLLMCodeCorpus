# Importing the Dictionary module
import Dictionary as dict

class Blocking:
    # Method to create a hash table from given data
    def CreateHashTable(self, data):
        # Initializing an empty hash table
        hash_table = dict.my_dictionary()
        # Variable to keep track of entries
        count = -1
        # Iterating through the data
        for d in data:
            # Creating a hash key for the current data entry
            key = self.CreateHashKey(d)
            # Incrementing the count
            count = count + 1
            # Appending the count to the data entry
            d = d + ',' + str(count)
            # Checking if the key already exists in the hash table
            if key in hash_table:
                # If exists, updating the entry
                hash_table.update(key, d)
            else:
                # If not exists, adding a new entry
                hash_table.add(key, d)
        # Returning the created hash table
        return hash_table
    
    # Method to create a hash key from a given name
    def CreateHashKey(self, name):
        # Initializing an empty string for the key
        key = ''
        # Iterating through the words in the name
        for w in name.split():
            # Appending the first character of each word to the key
            key = key + w[0]
        # Sorting the key
        key = ''.join(sorted(key))
        # Returning the created key
        return key
    
    # Method to find keys based on a given key
    def FindKeys(self, key):
        # Calculating the size of the key
        size = 1 << len(key)
        # Getting the length of the key
        num = len(key)
        # List to store found keys
        lstKeys = []
        # Iterating through all possible combinations of the key
        for i in range(1, size-1, 1):
            # Converting the current index to binary and padding it
            val = bin(i)[2:].zfill(num)
            # Initializing a total string
            tot = ''
            # Iterating through the characters of the binary representation
            for j in range(0, len(val)):
                if j < num:
                    # Checking if the character is '1' and not a digit in the original key
                    if val[j] == '1' and not key[j].isdigit():
                        # Appending the character to the total string
                        tot = tot + key[j]
            # Appending the total string to the list of keys
            lstKeys.append(tot)
        # Returning the list of found keys after removing duplicates
        return list(set(lstKeys))
