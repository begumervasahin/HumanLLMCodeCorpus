import numpy
from numpy.random import randint as rand

def generate_maze(width=81, height=51, complexity=.75, density=.75):
    # Define the shape of the maze
    shape = ((height // 2) * 2 + 1, (width // 2) * 2 + 1)
    
    # Calculate the complexity and density of the maze
    complexity = int(complexity * (5 * (shape[0] + shape[1])))
    density = int(density * ((shape[0] // 2) * (shape[1] // 2)))
    
    # Initialize the maze grid
    maze_grid = numpy.zeros(shape, dtype=bool)
    
    # Set the outer boundaries of the maze to be walls
    maze_grid[0, :] = maze_grid[-1, :] = 1
    maze_grid[:, 0] = maze_grid[:, -1] = 1
    
    # Generate the maze paths
    for i in range(density):
        # Choose a random starting point for the path
        x, y = rand(0, shape[1] // 2) * 2, rand(0, shape[0] // 2) * 2
        
        # Carve out the path
        maze_grid[y, x] = 1
        
        # Expand the path randomly to create complexity
        for j in range(complexity):
            # Find neighboring cells
            neighbours = []
            if x > 1:
                neighbours.append((y, x - 2))
            if x < shape[1] - 2:
                neighbours.append((y, x + 2))
            if y > 1:
                neighbours.append((y - 2, x))
            if y < shape[0] - 2:
                neighbours.append((y + 2, x))
                
            # Choose a random neighboring cell
            if len(neighbours):
                y_, x_ = neighbours[rand(0, len(neighbours) - 1)]
                
                # Carve out the path to the chosen neighboring cell
                if maze_grid[y_, x_] == 0:
                    maze_grid[y_, x_] = 1
                    maze_grid[y_ + (y - y_) // 2, x_ + (x - x_) // 2] = 1
                    x, y = x_, y_
    
    return maze_grid
