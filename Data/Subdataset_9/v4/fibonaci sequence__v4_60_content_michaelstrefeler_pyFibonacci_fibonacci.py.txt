from turtle import *

# Get user input for the number of Fibonacci sequence elements to generate
amount = input('Choose up to which number of the Fibonacci sequence you want to go up to: ')

# Validate user input to ensure it is a number greater than or equal to 5
while not amount.isdecimal() or int(amount) < 5:
    amount = input('Please write a number greater than or equal to 5: ')

# Convert the user input to an integer
amount = int(amount)

# Generate the Fibonacci sequence
fibonacci_sequence = [0, 1]
while len(fibonacci_sequence) < amount:
    next_number = fibonacci_sequence[-1] + fibonacci_sequence[-2]
    fibonacci_sequence.append(next_number)

# Print the Fibonacci sequence
for index, number in enumerate(fibonacci_sequence):
    print(f'Fibonacci {index}: {number}')

# Initialize Turtle graphics
penup()
goto(135.00, -20.00)
pendown()

# Draw Fibonacci sequence using Turtle
first_one = True
for fib in fibonacci_sequence:
    if fib == 0:
        pass
    elif fib == 1:
        if first_one:
            forward(12.5)
            write(fib)
            forward(12.5)
            left(90)
            forward(25)
            left(90)
            forward(25)
            left(90)
            forward(25)
            left(90)
            first_one = False
        else:
            forward(25)
            forward(12.5)
            write(fib)
            forward(12.5)
            left(90)
            forward(25)
            left(90)
            forward(25)
            left(90)
            forward(25)
            left(90)
            forward(25)
    else:
        forward(12.5 * fib)
        left(90)
        forward(12.5 * fib)
        left(90)
        forward(12.5)
        write(fib)
        forward(12.5)
        left(90)

# Configure Turtle for drawing circles
penup()
goto(135.00, 5.00)
pendown()
color('red')
shape('turtle')
pensize(2)
setheading(180)

# Draw circles based on Fibonacci numbers
for i in range(len(fibonacci_sequence)):
    if i < 9:
        circle(12.5 * fibonacci_sequence[i], 90)
    else:
        break

done()
