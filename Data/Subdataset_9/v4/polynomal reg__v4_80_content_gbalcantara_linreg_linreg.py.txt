import numpy as np
import matplotlib.pyplot as plt
import sys

def gradientDescent(x, y, th, eps, m, iteration):
    xTrans = x.transpose()
    previous_cost = None
    while True:
        h = np.dot(x, th)
        loss = h - y
        cost = np.sum(loss ** 2) / (2 * m)
        if previous_cost is not None and round(cost, 2) == previous_cost:
            break
        previous_cost = round(cost, 2)
        gradient = np.dot(xTrans, loss) / m
        th = th - eps * gradient
        eps += 0.00001
    return th, eps

# Extracting command line arguments or using default values
var1 = sys.argv[1] if len(sys.argv) > 1 else "0"
var2 = sys.argv[2] if len(sys.argv) > 2 else "0"
var3 = sys.argv[3] if len(sys.argv) > 3 else "0"
x2 = int(var1)
x1 = int(var2)
x0 = int(var3)

# Generating data points
x = np.zeros(shape=(20, 3))
y = np.zeros(shape=20)
for i, num in enumerate(range(-10, 10)):
    x[i][0] = 1
    x[i][1] = num
    x[i][2] = num ** 2
    y[i] = ((x2) * num ** 2) + ((x1) * num) + x0

# Adding noise to the data
noise = np.random.uniform(-0.1, 0.1, y.shape)
y = y + noise

# Setting up gradient descent parameters
m, n = np.shape(x)
iteration = 5000
eps = 0.000001
th = np.ones(n)

# Applying gradient descent to find optimal parameters
th, eps = gradientDescent(x, y, th, eps, m, iteration)

print('Learning rate =', eps)
print('x2 =', th[2])
print('x1 =', th[1])
print('x0 =', th[0])

# Plotting the data and the regression line
xx = np.linspace(-10, 10, 20)
yy = ((th[2]) * xx ** 2) + ((th[1]) * xx) + th[0]
plt.plot(x[:, 1], y, 'kx', xx, yy)
plt.show()
