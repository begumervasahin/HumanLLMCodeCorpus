def calculate_yearly_balance(balance, annual_interest, monthly_rate):
    """
    This function calculates the remaining debt at the end of the year.
    
    Parameters:
    - balance: The initial balance of the debt.
    - annual_interest: The annual interest rate.
    - monthly_rate: The monthly payment rate.

    Returns:
    The remaining debt at the end of the year.
    """
    unpaid_balance = balance

    for _ in range(12):
        unpaid_balance -= monthly_rate * unpaid_balance
        interest_amount = (annual_interest / 12) * unpaid_balance
        unpaid_balance += interest_amount

    return unpaid_balance
