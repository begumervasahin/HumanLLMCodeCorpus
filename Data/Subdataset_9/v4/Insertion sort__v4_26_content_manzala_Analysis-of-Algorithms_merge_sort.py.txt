import timeit
import random

# Function to copy elements from one array to another
def copy_array(src, src_pos, dest, dest_pos, length):
    for i in range(length):
        dest[i + dest_pos] = src[i + src_pos]

# Function to perform merge sort
def split_sort(array):
    if len(array) <= 1:
        return array
    
    mid = len(array) // 2
    left = array[:mid]
    right = array[mid:]

    split_sort(left)
    split_sort(right)
    merge_sort(left, right, array)
    
    return array

# Function to merge two sorted arrays
def merge_sort(left, right, array):
    rcount = 0
    lcount = 0
    merge_count = 0

    while lcount < len(left) and rcount < len(right):
        if left[lcount] > right[rcount]:
            array[merge_count] = right[rcount]
            rcount += 1
        else:
            array[merge_count] = left[lcount]
            lcount += 1
        merge_count += 1
    
    copy_array(left, lcount, array, merge_count, len(left) - lcount)
    copy_array(right, rcount, array, merge_count, len(right) - rcount)

# Generate a list of random numbers
numbers = [7, 4, 8, 3, 1, 12, 9, 14]
for _ in range(5000):
    rand = random.randint(1, 99999)
    numbers.append(rand)

# Measure the time taken to perform merge sort
start_time = timeit.default_timer()
split_sort(numbers)
elapsed_time = timeit.default_timer() - start_time

# Print the elapsed time and the sorted list
print("Merge Sort: " + str(elapsed_time))
print(numbers)
