from pyspark import SparkConf, SparkContext
from operator import add
import re
import sys

class WordCountApp:
    def processFile(self, documentAndContents):
        # Extracting document name from the path
        document = documentAndContents[0].split('/')[-1]
        # Finding all words using regular expression
        wordList = re.compile('\w+').findall(documentAndContents[1])
        # Mapping each word to a tuple of ((document_name, word), 1)
        return map(lambda word: ((str(document), str(word.lower())), 1), wordList)

    def main(self, sc, input_file, output_file):
        # Creating a Spark context
        wholeTextFilesRDD = sc.wholeTextFiles(input_file)
        # Caching RDD for efficient reuse
        wholeTextFilesRDD.cache()
        # Applying flatMap to process each document and count words
        word_count = wholeTextFilesRDD.flatMap(self.processFile).reduceByKey(add, numPartitions=1).sortByKey()
        # Saving the word count result to the output file
        word_count.saveAsTextFile(output_file)

if __name__ == '__main__':
    # Configuring Spark application
    conf = SparkConf().setAppName('Part 1: Word Count Per Book').set('spark.executor.memory', '2g').set('spark.ui.port', 6200)
    # Creating Spark context
    sc = SparkContext(conf=conf)
    # Getting input and output file paths from command line arguments
    input_file = sys.argv[1]
    output_file = sys.argv[2]
    # Creating an instance of WordCountApp
    wordCountApp = WordCountApp()
    # Running the main function with Spark context and file paths
    wordCountApp.main(sc, input_file, output_file)
