import numpy as np
from itertools import combinations
from math import acos, sqrt, pow, pi, cos, sin
import Matrix

# Function to compute projection of matrix A
def compute_projection(A):
    interim = np.matmul(A.transpose(), A)
    return np.reshape(interim, -1)

# Function to decompose q
def decompose_q(q, d):
    xxT = Matrix.getClosestSymMatrix(np.reshape(q, (d, d)))
    x = -xxT[d - 1]
    return x

# Function to compute q component
def compute_q_component(x):
    interim = np.outer(x, x)
    return np.reshape(interim, -1)

# Convert Cartesian coordinates to Spherical coordinates
def cartesian_to_spherical(x):
    d = len(x)
    phi = np.zeros((d-1))
    d_ = len(phi)
    sum_squares = pow(x[d_], 2) + pow(x[d_-1], 2)
    last_angle = acos(x[d_-1] / (sqrt(sum_squares)))
    
    if x[d_] >= 0:
        phi[d_-1] = last_angle
    else:
        phi[d_-1] = 2 * pi - last_angle
    
    for k in range(d_-2, -1, -1):
        sum_squares = sum_squares + pow(x[k], 2)
        phi[k] = acos(x[k] / sqrt(sum_squares))
    
    return phi

# Convert Spherical coordinates to Cartesian coordinates
def spherical_to_cartesian(phi):
    dim = len(phi)
    x = np.zeros(dim+1)
    sins = 1
    
    for i in range(0, dim):
        x[i] = sins * cos(phi[i])
        sins *= sin(phi[i])
    
    x[dim] = sins
    return x

# Alternative code: Appendix method

# Function to compute alternative projection of matrix A
def compute_alt_projection(A, j):
    d = len(A[0, :])
    p = np.array([])
    p_1 = np.reshape(np.multiply(A, A), (1, A.shape[0] * A.shape[1]))
    p = np.append(p, p_1)
    index = range(d + 1)
    comb = list(combinations(index, 2))
    
    for i in range(d - j):
        p_2 = 2 * np.multiply(A[i, [x[0] for x in comb]], A[i, [x[1] for x in comb]])
        p = np.append(p, p_2)
    
    return p

# Function to decompose q (alternative)
def decompose_alt_q(q, d, j):
    x_ = np.sqrt(np.abs(q[0:d]))
    signs = q[d * (d - j):d * (d - j) + d - 1]
    x = np.zeros(d)
    x[0] = np.absolute(x_[0])
    
    for i in range(1, len(x_)):
        if signs[i - 1] < 0:
            x[i] = -np.absolute(x_[i])
        else:
            x[i] = np.absolute(x_[i])
    
    return x
