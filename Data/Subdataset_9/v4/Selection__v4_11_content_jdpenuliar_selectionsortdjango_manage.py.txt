import os
import sys

if __name__ == "__main__":
    # Set the DJANGO_SETTINGS_MODULE environment variable to the specified Django settings module
    os.environ.setdefault("DJANGO_SETTINGS_MODULE", "selection_sort.settings")
    
    try:
        # Try to import the execute_from_command_line function from Django's management module
        from django.core.management import execute_from_command_line
    except ImportError:
        try:
            # If Django is not imported, try to import the Django module itself
            import django
        except ImportError:
            # If Django is still not imported, raise an ImportError with a detailed message
            raise ImportError(
                "Couldn't import Django. Are you sure it's installed and "
                "available on your PYTHONPATH environment variable? Did you "
                "forget to activate a virtual environment?"
            )
        raise
    
    # Execute the Django management command specified in sys.argv
    execute_from_command_line(sys.argv)
