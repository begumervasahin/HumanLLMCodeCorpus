import sys

# Open and read the Lithuanian lemma dictionary
with open('dazninis.utf8.txt', 'r', encoding='utf-8') as dic:
    lt_lemmas = []
    word_0 = ''
    for line in dic:
        line = line.split('\t')
        if line[0] != word_0:
            lt_lemmas.append(line[0])
        word_0 = line[0]

# Get the file paths from command-line arguments
file_path = sys.argv[1]
doc_path = sys.argv[2]

# Open the input file for reading and the output file for writing
with open(file_path, 'r', encoding='utf-8') as file:
    with open(doc_path, 'w', encoding='utf-8') as doc:
        # Iterate through each line in the input file
        for line in file:
            line = line.split('\t')
            # Check if the line has 10 fields (indicating it's a word entry)
            if len(line) == 10:
                lemma = line[2]
                # Check if the word is a verb
                if line[3] == 'VERB':
                    # Handle cases where lemma starts with 'nebe'
                    if lemma.startswith('nebe'):
                        if lemma.startswith('nebesi'):
                            # Handle reflexive verbs with 'nebesi'
                            if 'Reflex=Yes' in line[5]:
                                if lemma[6:] + 's' in lt_lemmas:
                                    line[2] = lemma[6:] + 's'
                        elif lemma[4:] in lt_lemmas:
                            line[2] = lemma[4:]
                        elif lemma[2:] in lt_lemmas:
                            line[2] = lemma[2:]
                    # Handle cases where lemma starts with 'ne'
                    elif lemma.startswith('ne'):
                        if 'Reflex=Yes' in line[5]:
                            line[2] = lemma[4:] + 's'
                        elif lemma[2:] in lt_lemmas:
                            line[2] = lemma[2:]
                    # Handle cases where lemma starts with 'tebe'
                    elif lemma.startswith('tebe'):
                        if 'Reflex=Yes' in line[5]:
                            line[2] = lemma[6:] + 's'
                        elif lemma[2:] in lt_lemmas:
                            line[2] = lemma[2:]
                        else:
                            line[2] = lemma[4:]
                    # Handle cases where lemma starts with 'be'
                    elif lemma.startswith('be'):
                        if 'Reflex=Yes' in line[5]:
                            line[2] = lemma[4:] + 's'
                        elif lemma[2:] in lt_lemmas:
                            line[2] = lemma[2:]
                # Join the modified line and write it to the output file
                line1 = '\t'.join(line)
                doc.write(line1)
            else:
                # Write lines with other than 10 fields directly to the output file
                line1 = '\t'.join(line)
                doc.write(line1)
