import os

def create_domain_directory(directory):
    """
    Creates a directory if it does not already exist.

    Args:
        directory (str): The directory to be created.
    """
    if not os.path.exists(directory):
        print('Creating directory ' + directory)
        os.makedirs(directory)

def create_data_files(project_name, base_url):
    """
    Creates data files for the project if they do not already exist.

    Args:
        project_name (str): Name of the project.
        base_url (str): Base URL for the project.
    """
    queue = os.path.join(project_name, 'queue.txt')
    crawled = os.path.join(project_name, "crawled.txt")
    
    if not os.path.isfile(queue):
        write_file(queue, base_url)
    
    if not os.path.isfile(crawled):
        write_file(crawled, '')

def write_file(path, data):
    """
    Writes data to a file.

    Args:
        path (str): Path to the file.
        data (str): Data to be written to the file.
    """
    with open(path, 'w') as f:
        f.write(data)

def append_to_file(path, data):
    """
    Appends data to a file.

    Args:
        path (str): Path to the file.
        data (str): Data to be appended to the file.
    """
    with open(path, 'a') as file:
        file.write(data + '\n')

def delete_file_contents(path):
    """
    Deletes the contents of a file.

    Args:
        path (str): Path to the file.
    """
    open(path, 'w').close()

def get_set_from_file(file_name):
    """
    Retrieves a set of data from a file.

    Args:
        file_name (str): Name of the file.

    Returns:
        set: Set of data retrieved from the file.
    """
    results = set()
    with open(file_name, 'rt') as f:
        for line in f:
            results.add(line.replace('\n', ''))
    return results

def write_set_to_file(links, file_name):
    """
    Writes a set of data to a file.

    Args:
        links (set): Set of data to be written to the file.
        file_name (str): Name of the file.
    """
    with open(file_name, "w") as f:
        for l in sorted(links):
            f.write(l + "\n")
