import time
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.neighbors import KNeighborsClassifier

class TextColors:
    HEADER = '\033[95m'
    OKBLUE = '\033[94m'
    OKGREEN = '\033[92m'
    WARNING = '\033[93m'
    FAIL = '\033[91m'
    ENDC = '\033[0m'
    BOLD = '\033[1m'
    UNDERLINE = '\033[4m'

def get_features(data):
    """
    Get the features from the dataset.
    """
    features = list(data.columns)
    features.remove("label")
    return features

def feature_selection(data):
    """
    Perform feature selection using SelectKBest with chi-squared score.
    """
    from sklearn.feature_selection import SelectKBest, chi2
    features = get_features(data)
    X = data[features]
    y = data["label"]
    selector = SelectKBest(score_func=chi2, k=5)
    selector.fit(X, y)
    indexes_selected = selector.get_support(indices=True)
    selected_features = [features[i] for i in indexes_selected]
    print("[+] Selected features -> " + TextColors.OKBLUE + str(selected_features) + TextColors.ENDC)
    return data[selected_features]

def with_feature_selection(data):
    """
    Train and evaluate KNN classifier with selected features.
    """
    print(TextColors.HEADER + TextColors.UNDERLINE + "Testing with selected features" + TextColors.ENDC)
    X = feature_selection(data)
    y = data["label"]
    X_train, X_test, y_train, y_test = train_test_split(X, y, random_state=0)
    knn = KNeighborsClassifier(n_neighbors=5)
    start = time.time()
    knn.fit(X_train, y_train)
    print("[+] Classifier trained in " + TextColors.OKGREEN + str(time.time() - start) + TextColors.ENDC)
    start = time.time()
    score = knn.score(X_test, y_test)
    print("[+] Model Evaluated in " + TextColors.OKGREEN + str(time.time() - start) + TextColors.ENDC)
    print("[!] Test score is " + TextColors.OKBLUE + str(score) + TextColors.ENDC)
    print(TextColors.BOLD + TextColors.WARNING + "-------------------------------------------------" + TextColors.ENDC)

def with_full_features(data):
    """
    Train and evaluate KNN classifier with full features.
    """
    print(TextColors.HEADER + TextColors.UNDERLINE + "Testing with full data" + TextColors.ENDC)
    X = data[get_features(data)]
    y = data["label"]
    X_train, X_test, y_train, y_test = train_test_split(X, y, random_state=0)
    knn = KNeighborsClassifier(n_neighbors=5)
    start = time.time()
    knn.fit(X_train, y_train)
    print("[+] Classifier trained in " + TextColors.OKGREEN + str(time.time() - start) + TextColors.ENDC)
    start = time.time()
    score = knn.score(X_test, y_test)
    print("[+] Model Evaluated in " + TextColors.OKGREEN + str(time.time() - start) + TextColors.ENDC)
    print("[!] Test score is " + TextColors.OKBLUE + str(score) + TextColors.ENDC)
    print(TextColors.BOLD + TextColors.WARNING + "-------------------------------------------------" + TextColors.ENDC)

def test_with_real(data, test_filename):
    """
    Test the KNN classifier with real data from a file.
    """
    X = data[get_features(data)]
    y = data["label"]
    knn = KNeighborsClassifier(n_neighbors=5)
    knn.fit(X, y)
    with open(test_filename, "r") as test_file:
        lines = [line.strip().split(",") for line in test_file]
    print(len(lines))
    test_results = []
    for i, line in enumerate(lines):
        print(i)
        print(line)
        test_results.append(knn.predict([line]))
    count_dict = {}
    for result in test_results:
        try:
            count_dict[result[0]] += 1
        except KeyError:
            count_dict[result[0]] = 1
    print(count_dict)

def main():
    filename = "preprocessed_kddcup_data"
    data = pd.read_csv(filename)
    with_full_features(data)
    with_feature_selection(data)

if __name__ == "__main__":
    main()
