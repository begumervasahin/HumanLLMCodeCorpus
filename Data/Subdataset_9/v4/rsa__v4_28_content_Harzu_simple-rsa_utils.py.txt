# Function to generate a prime number with specified bits length
def generatePrime(bits=1024):
    while True:
        # Generate a random number with the specified number of bits
        p = random.getrandbits(bits)
        # Check if the generated number is prime
        if testMillerRabin(p):
            return p

# Function to test primality using Miller-Rabin algorithm
def testMillerRabin(prime, rounds=50):
    # Base cases for small primes
    if prime == 2 or prime == 3:
        return True
    # Check for even numbers and numbers less than 2
    if prime < 2 or prime % 2 == 0:
        return False
    
    # Write prime - 1 as 2^s * t, where t is odd
    t = prime - 1
    s = 0
    while t % 2 == 0:
        s += 1
        t //= 2
    
    # Perform Miller-Rabin probabilistic primality test
    for _ in range(rounds):
        a = random.randint(1, prime - 1)
        x = pow(a, t, prime)
        if x == 1 or x == prime - 1:
            continue
        for _ in range(s):
            x = pow(x, 2, prime)
            if x == 1:
                return False
            if x == prime - 1:
                break
        if x != prime - 1:
            return False
    return True

# Function to compute the greatest common divisor
def gcd(a, b):
    while a != 0:
        a, b = b % a, a
    return b

# Function to find the modular multiplicative inverse
def findModInverse(a, m):
    # Check if 'a' and 'm' are coprime
    if gcd(a, m) != 1:
        return None
    # Initialize variables for extended Euclidean algorithm
    u1, u2, u3 = 1, 0, a
    v1, v2, v3 = 0, 1, m
    # Perform extended Euclidean algorithm
    while v3 != 0:
        q = u3 // v3
        v1, v2, v3, u1, u2, u3 = (u1 - q * v1), (u2 - q * v2), (u3 - q * v3), v1, v2, v3
    # Return the modular multiplicative inverse
    return u1 % m
