# Create an empty dictionary to store word frequencies
dictionary = {}

# Open the file containing word frequencies
with open("dosyalar/derlemtr2016-10000.txt", 'r', encoding='utf-8') as fdict:
    # Read each line in the file
    for line in fdict:
        # Check if the line does not start with '0' or '1'
        if line[0] not in ['0', '1']:
            continue
        # Split the line into frequency and word
        freq, word = line.strip().split()
        # Store the word frequency in the dictionary
        dictionary[word] = int(freq)

# Function to check if a word is a stop word
def is_stop_word(word):
    """
    Checks if a word is in the dictionary.
    
    Parameters:
        word (str): The word to check.
    
    Returns:
        bool: True if the word is a stop word, False otherwise.
    """
    return word in dictionary

# Function to get the frequency of a word
def get_word_freq(word):
    """
    Returns the frequency of a word.
    
    Parameters:
        word (str): The word to get the frequency of.
    
    Returns:
        int: The frequency of the word, or 0 if the word is not in the dictionary.
    """
    if is_stop_word(word):
        return dictionary[word]
    else:
        return 0

if __name__ == '__main__':
    # Open the file containing stop words
    with open("dosyalar/turkce-stop-words", encoding="utf-8") as ftest:
        # Iterate over each word in the file
        for word in ftest:
            word = word.strip()
            # Check if the word is not in the dictionary
            if word not in dictionary:
                print('{} is not in the dictionary'.format(word))
