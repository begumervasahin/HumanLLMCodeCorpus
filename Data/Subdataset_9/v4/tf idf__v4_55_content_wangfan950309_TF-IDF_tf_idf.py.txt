import pandas as pd
import math
import numpy as np
from web_scraping.scraping import main_path

# Function to perform a search using TF-IDF algorithm
def search_algorithm(keywords):
    # Splitting the search keywords
    search_key = keywords
    keys = search_key.split()

    # Creating a dictionary to store IDF values for each keyword
    key_dic = dict()

    # Reading the job data from CSV file
    df = pd.read_csv("%s/data/Job_search.csv" % main_path(), index_col=None)
    search_field = df['Job_Name'].tolist()

    # Calculating IDF values for each keyword
    for key in keys:
        idf_counter = 0
        for field in search_field:
            if key in field:
                idf_counter += 1
        key_dic[key] = math.log(len(df.index) / (1 + idf_counter))

    # Calculating TF-IDF scores and updating dataframe
    df['TF-IDF'] = 0
    for key in keys:
        tf = []
        for field in search_field:
            counter = 0
            job = field.split()
            for word in job:
                if key.lower() == word.lower():
                    counter += 1
            tf.append(counter / len(job))
        
        # Updating dataframe with TF-IDF scores for each keyword
        df[key] = np.array(tf)
        df[key] = df[key] * key_dic[key]
        df['TF-IDF'] = df['TF-IDF'] + df[key]

    # Sorting dataframe by TF-IDF scores in descending order
    df = df.sort_values(by=['TF-IDF'], ascending=False)

    # Filtering out rows with zero TF-IDF scores
    df = df.loc[df['TF-IDF'] != 0]

    # Keeping only relevant columns and returning the resulting dataframe
    df = df.iloc[:, :4]
    return df
