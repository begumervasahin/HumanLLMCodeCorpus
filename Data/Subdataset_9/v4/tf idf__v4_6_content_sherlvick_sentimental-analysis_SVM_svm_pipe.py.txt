# Importing necessary libraries
import pandas as pd
import numpy as np
import pickle
import os
from sklearn.feature_extraction.text import CountVectorizer, TfidfTransformer
from sklearn.pipeline import Pipeline
from sklearn import svm

# Function to train the SVM model
def training():
    # Setting seed for reproducibility
    np.random.seed(100)
    
    # Loading data from CSV file
    Corpus = pd.read_csv("Test.csv", encoding='utf-8-sig')
    
    # Extracting text and labels from the dataset
    my_data = Corpus['text']
    my_data1 = Corpus['label']
    
    # Creating a pipeline for text classification
    text_clf = Pipeline([
        ('vect', CountVectorizer()),  # Converting text to a matrix of token counts
        ('tfidf', TfidfTransformer()),  # Converting token counts to TF-IDF representation
        ('clf', svm.SVC(C=5, kernel='rbf', degree=3, gamma=0.5)),  # Support Vector Classifier
    ])
    
    # Fitting the model
    my_clf = text_clf.fit(my_data, my_data1)
    
    # Saving the trained model to a file
    filename = 'Amz_5000_.sav'
    pickle.dump(my_clf, open(filename, 'wb'))
    
    # Checking if the model file exists
    model_exists = os.path.exists('Amz_5000_.sav')
    
    # Generating messages based on whether the model was saved successfully
    messages = ["Successful : Training model Saved", "Oh Snap! something's wrong", str(len(Corpus))]
    if model_exists:  # If model file exists
        del messages[1]  # Remove the "Oh Snap!" message
        return messages
    else:  # If model file doesn't exist
        del messages[0]  # Remove the "Successful" message
        return messages

# Calling the training function
training()
