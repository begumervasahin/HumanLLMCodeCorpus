import asyncio
import json
import logging
import asyncpg
from nats.aio.client import Client as NATS

# Constants
NATS_HOST = 'localhost'
NATS_PORT = '4222'
CHANNEL = 'fixtures.parameters'

async def bridge(loop):
    """
    Connects to PostgreSQL database and NATS server, adds a listener to PostgreSQL for a specific channel,
    and publishes messages to NATS when updates are received.
    """
    async def connect_listener():
        # Connect to PostgreSQL database
        pgconn = await asyncpg.connect(user='postgres', password='password', database='postgres', host='127.0.0.1', loop=loop)
        print("Connected to PostgreSQL")
        
        # Add a listener to PostgreSQL for the specified channel
        await pgconn.add_listener(CHANNEL, publish_update)
        
        # Connect to NATS server
        await nc.connect(f"{NATS_HOST}:{NATS_PORT}", loop=loop)
        print("Listener added")
        
    def publish_update(_con, _pid, _channel, payload):
        # Publish update to NATS
        asyncio.run_coroutine_threadsafe(nc.publish(CHANNEL, json.dumps(payload).encode('utf-8')), loop)
        logger.info(f"Update received: {payload}")
    
    # Initialize NATS client
    nc = NATS()
    
    # Connect listeners
    await connect_listener()

if __name__ == '__main__':
    # Configure logging
    logging.basicConfig(format='%(asctime)s, %(message)s', level=logging.INFO)
    logger = logging.getLogger(__name__)
    
    # Get the event loop
    event_loop = asyncio.get_event_loop()
    
    # Run the bridge coroutine
    event_loop.run_until_complete(bridge(event_loop))
    
    try:
        print('Run forever')
        event_loop.run_forever()
    finally:
        print('Closing')
