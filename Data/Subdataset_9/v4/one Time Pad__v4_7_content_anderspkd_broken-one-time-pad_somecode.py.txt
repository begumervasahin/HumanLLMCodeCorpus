from base64 import b64encode
from time import time

PAD_LEN = 1024 * 100

def is_valid_character(c):
    """
    Checks if the given character is a valid ASCII character.
    """
    return ((64 < c < 91) or          # Uppercase letters
            (96 < c < 123) or         # Lowercase letters
            (47 < c < 58) or          # Digits
            (c == 43) or              # '+'
            (c == 47) or              # '/'
            (c == 61))                # '='

def split_ciphertext():
    """
    Splits the ciphertext into a list of lists, each containing PAD_LEN characters.
    """
    chunks = [[] for _ in range(PAD_LEN)]
    with open('out', 'rb') as f:
        i = 0
        done = False
        while not done:
            c = f.read(1)
            if c != b'':
                chunks[i % PAD_LEN].append(ord(c))
                i += 1
            else:
                done = True
    return chunks

def find_pad():
    """
    Finds the potential key values for each position in the ciphertext.
    """
    ciphertext_chunks = split_ciphertext()
    potential_key = [[] for _ in range(PAD_LEN)]
    t = time()
    for j in range(PAD_LEN):
        chk = ciphertext_chunks[j]
        potential_key_values = []
        for i in range(2**8):
            is_valid = True
            for c in chk:
                if not is_valid_character(c ^ i):
                    is_valid = False
                    break
            if is_valid:
                potential_key_values.append(i)
        if len(potential_key_values) > 1:
            print('Potential key values for ka[{}]: {}'.format(j, potential_key_values))
        potential_key[j] = potential_key_values
        if time() - t > 5:
            t = time()
            print('Progress: {:.4f}%'.format(100 * (j / PAD_LEN)))
    write_key(potential_key)

def write_key(key):
    """
    Writes the potential key values to a file.
    """
    with open('key', 'w') as f:
        f.write(str(key))

def read_key():
    """
    Reads the potential key values from a file.
    """
    with open('key', 'r') as f:
        key = eval(f.read())
    return key

if __name__ == '__main__':
