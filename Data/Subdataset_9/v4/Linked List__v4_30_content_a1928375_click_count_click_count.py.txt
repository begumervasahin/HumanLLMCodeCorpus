def record_user_click(index, keyword, url):
    # Check if the keyword exists in the index
    urls = lookup_keyword(index, keyword)
    
    # If the keyword exists, iterate over the URLs associated with it
    if urls:
        for entry in urls:
            # If the URL matches, increment the click count
            if entry[0] == url:
                entry[1] += 1

def lookup_keyword(index, keyword):
    # Search for the keyword in the index
    for entry in index:
        if entry[0] == keyword:
            return entry[1]  # Return the URLs associated with the keyword
    return None  # If keyword not found, return None

def add_to_index(index, keyword, url):
    # Iterate over the index entries
    for entry in index:
        # If the keyword already exists in the index
        if entry[0] == keyword:
            # Check if the URL is already associated with the keyword
            for url_entry in entry[1]:
                if url_entry[0] == url:
                    return  # URL already exists, no need to add again
            # If URL not found, add it along with initial click count of 0
            entry[1].append([url, 0])
            return
    # If keyword not found, add a new entry with the keyword and associated URL
    index.append([keyword, [[url, 0]]])

def union(a, b):
    # Function to merge two lists without duplicates
    for e in b:
        if e not in a:
            a.append(e)

def get_next_target(page):
    # Function to extract the next URL link from HTML content
    # This function searches for the '<a href=' tag and extracts the URL
    start_link = page.find('<a href=')
    if start_link == -1:
        return None, 0
    start_quote = page.find('"', start_link)
    end_quote = page.find('"', start_quote + 1)
    url = page[start_quote + 1:end_quote]
    return url, end_quote

def get_all_links(page):
    # Function to extract all URLs from HTML content
    links = []
    while True:
        url, endpos = get_next_target(page)
        if url:
            links.append(url)
            page = page[endpos:]
        else:
            break
    return links

def crawl_web(seed):
    # Function to crawl the web starting from a given seed URL
    tocrawl = [seed]  # List of URLs to crawl
    crawled = []  # List of URLs already crawled
    index = []  # Index to store keywords and associated URLs with click counts
    
    # Loop until there are URLs to crawl
    while tocrawl:
        page = tocrawl.pop()  # Get the next URL to crawl
        if page not in crawled:
            content = get_page(page)  # Fetch the content of the page
            add_page_to_index(index, page, content)  # Add page content to the index
            union(tocrawl, get_all_links(content))  # Extract links from the page and add to tocrawl
            crawled.append(page)  # Add the crawled page to the list of crawled URLs
    return index  # Return the index containing keywords and associated URLs

def add_page_to_index(index, url, content):
    # Function to add the content of a page to the index
    words = content.split()  # Split the content into words
    for word in words:
        add_to_index(index, word, url)  # Add each word and its associated URL to the index
