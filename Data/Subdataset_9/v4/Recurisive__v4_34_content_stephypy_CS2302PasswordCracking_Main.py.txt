import hashlib
import time

def hash_with_sha256(string):
    hash_object = hashlib.sha256(string.encode('utf-8'))
    hex_digest = hash_object.hexdigest()
    return hex_digest

def find_password(password):
    with open("password_file.txt") as file:
        for line in file:
            elements = line.split(",")
            hashline = elements[2].replace('\n', '')
            new_hashline = hash_with_sha256(password + str(elements[1]))
            if new_hashline == hashline:
                print(elements[0] + ': ' + password)

def get_combination(size, combination):
    if size == 0:
        find_password(combination)
    else:
        for digit in range(0, 10):
            new_combination = combination + str(digit)
            get_combination(size - 1, new_combination)

def hack_file(start, end):
    if not isinstance(start, int) or not isinstance(end, int):
        print('Invalid input. Please try again')
        return
    if start < 3 or end > 7:
        print('Invalid input. Please try again')
        return
    for size in range(start, end + 1):
        get_combination(size, '')

def test_cases():
    start = time.time()
    hack_file(3.0, 7.0)
    end = time.time()
    print('Running time for invalid data type:', end - start, 'seconds\n')
    
    start = time.time()
    hack_file(2, 7)
    end = time.time()
    print('Running time for invalid range:', end - start, 'seconds\n')
    
    start = time.time()
    hack_file(3, 8)
    end = time.time()
    print('Running time for invalid range:', end - start, 'seconds\n')

def main():
    start = time.time()
    hack_file(3, 7)
    end = time.time()
    print('Running time for normal run: ', end - start, 'seconds\n')
    test_cases()

if __name__ == "__main__":
    main()
