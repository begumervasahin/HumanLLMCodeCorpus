class Graph:
    def __init__(self):
        self.adjacency_list = {}  # Dictionary to store the adjacency list representation of the graph
        self.size = 0  # Number of vertices in the graph

    def get_vertices(self):
        """
        Returns a list of all vertices in the graph.
        """
        return list(self.adjacency_list.keys())

    def get_neighbours(self, v):
        """
        Returns a list of neighbours of a given vertex v.
        """
        return self.adjacency_list[v] if v in self.adjacency_list else []

    def print_vertices(self):
        """
        Prints all vertices in the graph.
        """
        print("Vertices:")
        for vertex in self.adjacency_list:
            print(vertex)
        print('\n')

    def print_neighbours(self, v):
        """
        Prints the neighbours of a given vertex v.
        """
        print("Neighbours of " + v)
        for neighbour in self.adjacency_list.get(v, []):
            print(neighbour[0] + ", weight of edge: " + str(neighbour[1]))

    def add_vertex(self, v):
        """
        Adds a vertex v to the graph.
        """
        self.adjacency_list[v] = []

    def add_undirected_edge(self, u, v):
        """
        Adds an undirected edge between vertices u and v.
        """
        if u not in self.adjacency_list:
            self.add_vertex(u)
        if v not in self.adjacency_list:
            self.add_vertex(v)
        self.adjacency_list[u].append(v)

    def add_directed_edge(self, u, v):
        """
        Adds a directed edge from vertex u to vertex v.
        """
        self.add_undirected_edge(u, v)
        self.add_undirected_edge(v, u)
