import os
import xattr
from datetime import datetime

# Function to get the last used date of a file
def get_last_used_date(file_path):
    try:
        # Attempt to fetch the last used date attribute
        last_used_date_bytes = xattr.getxattr(file_path, 'com.apple.lastuseddate')
        
        # Convert the retrieved bytes to a timestamp
        last_used_timestamp = int.from_bytes(last_used_date_bytes, byteorder='big')
        
        # Convert the timestamp to a human-readable datetime object
        last_used_date = datetime.fromtimestamp(last_used_timestamp)
        
        return last_used_date  # Return the last used date if successful
    except OSError:
        return None  # Return None if the file doesn't exist or the attribute is not found

# Main block
if __name__ == "__main__":
    file_path = "/path/to/your/file"  # Specify the path to the file
    last_used_date = get_last_used_date(file_path)  # Retrieve the last used date
    if last_used_date:
        # If the last used date is retrieved, print it
        print(f"The last used date of the file '{file_path}' is: {last_used_date}")
    else:
        # If unable to retrieve the last used date, print a message
        print(f"Unable to retrieve the last used date for the file '{file_path}'")
