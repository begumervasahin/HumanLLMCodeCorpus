def is_prime(n):
    # Check if n is an integer
    assert type(n) == int
    
    # 0 and 1 are not prime numbers
    if n < 2:
        return False
    else:
        # Check for factors from 2 to n-1
        for i in range(2, n):
            if n % i == 0:
                return False
        return True

def divisor(n):
    # Find all divisors of n
    lst = []
    for i in range(1, n + 1):
        if n % i == 0:
            lst.append(i)
    return lst

def prime_factor(x):
    fact_lst = []
    
    # If x is a prime number, return it as a factor
    if is_prime(x):
        fact_lst.append(x)
    
    # Check for prime factors of x
    for i in divisor(x):
        if i == 1:
            pass
        elif is_prime(i):
            fact_lst.append(i)
    return fact_lst

def gcd(a, b):
    # Calculate the Greatest Common Divisor (GCD) of a and b
    if b > a:
        return gcd(b, a)
    elif a % b == 0:
        return b
    return gcd(b, a % b)

def relatively_prime(a, b):
    # Check if a and b are relatively prime
    while a > 0 and b > 0:
        if gcd(a, b) == 1:
            return True
        else:
            return False

def common_factor(a, b):
    # Find common factors of a and b
    a_fact = divisor(a)
    b_fact = divisor(b)
    common_fact = []
    for elem in a_fact:
        if elem in b_fact:
            common_fact.append(elem)
    return common_fact
