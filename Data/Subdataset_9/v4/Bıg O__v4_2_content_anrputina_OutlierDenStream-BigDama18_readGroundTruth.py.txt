# Importing required library
import pandas as pd

# Defining the GroundTruth class
class GroundTruth:
    def __init__(self, filename, file_type='txt'):
        # Initializing the class attributes
        self.filename = filename
        self.events = []
        self.clears = []
        
        # Handling different file types
        if file_type == 'txt':
            self.read_file()
        elif file_type == 'csv':
            self.read_file_csv()

    # Method to add an event to the events list
    def add_event(self, event):
        self.events.append(event)

    # Method to add a clear to the clears list
    def add_clear(self, clear):
        self.clears.append(clear)

    # Method to build the clear records based on events
    def build_clear(self):
        for idx in range(len(self.events)):
            # Check if the event is single type
            if self.events[idx]['type'] == 'single':
                # Handling if the event is not the last event
                if idx < len(self.events) - 1:
                    idx2 = idx + 1
                    # Find the next single type event
                    if idx2 <= len(self.events) - 1:
                        while self.events[idx2]['type'] == 'multiple' and idx2 < len(self.events) - 1:
                            idx2 += 1
                        # Creating clear record
                        clear_record = {
                            'name': 'clear' + str(idx),
                            'startTime': self.events[idx]['endTime'] + 1,
                            'endTime': self.events[idx2]['startTime'] - 1
                        }
                        self.clears.append(clear_record)
                # Handling if the event is the last event
                if idx == len(self.events) - 1:
                    clear_record = {
                        'name': 'clear' + str(idx),
                        'startTime': self.events[idx]['endTime'] + 1,
                        'endTime': self.events[idx]['endTime'] + 1 + 300
                    }
                    self.clears.append(clear_record)
        # Changing all events to single type
        for event in self.events:
            event['type'] = 'single'

    # Method to read data from a text file
    def read_file(self):
        # Reading data from the text file
        df = pd.read_csv(self.filename, sep='\t', header=None)
        df.columns = ["node", "ip", "startTime", "event"]
        df['endTime'] = df['startTime'] + 300
        df['clearStart'] = df['endTime'] + 1
        df['clearEnd'] = df['startTime'].shift(-1) - 1 - 5
        df['clearEnd'].iloc[-1] = 2000000000
        # Processing each row of the dataframe
        for row in df.iterrows():
            event_record = {
                'name': row[0],
                'startTime': row[1]['startTime'],
                'endTime': row[1]['endTime'],
                'node': row[1]['node'],
                'type': 'single',
                'ONLINE': False,
                'endSent': False
            }
            # Adding event record to events list
            self.add_event(event_record)
            clear_record = {
                'name': row[0],
                'startTime': row[1]['clearStart'],
                'endTime': row[1]['clearEnd']
            }
            # Adding clear record to clears list
            self.add_clear(clear_record)

    # Method to read data from a csv file
    def read_file_csv(self):
        # Reading data from the csv file
        self.df = pd.read_csv(self.filename, sep=',')
        # Processing each row of the dataframe
        for row in self.df.iterrows():
            event_record = {
                'name': row[0],
                'startTime': row[1]['Start'],
                'endTime': row[1]['End'],
                'node': row[1]['Node'],
                'type': row[1]['Type'],
                'ONLINE': False,
                'endSent': False
            }
            # Adding event record to events list
            self.add_event(event_record)
        # Building clear records based on events
        self.build_clear()
