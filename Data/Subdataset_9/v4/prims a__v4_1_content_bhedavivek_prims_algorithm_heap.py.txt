class Heap:
    def __init__(self):
        self.array = [0]  # Using the first element to store the size of the heap

    def insert(self, key_value):
        """
        Inserts a new value into the heap, maintaining the heap property.
        """
        self.array.append(key_value)
        # Update the heap size
        self.array[0] = len(self.array) - 1
        # Ensure the new element maintains the heap property
        self._heapify_up(self.array[0])

    def _heapify_up(self, index):
        """
        Moves the element at the specified index up to maintain the heap property.
        """
        while index > 1:
            parent_index = index // 2
            if self.array[parent_index].distance > self.array[index].distance:
                # Swap if the parent's distance is greater than the child's
                self.array[parent_index], self.array[index] = self.array[index], self.array[parent_index]
                index = parent_index
            else:
                break

    def extract_min(self):
        """
        Removes and returns the element with the smallest distance.
        """
        if self.array[0] < 1:
            return None  # Heap is empty

        min_element = self.array[1]
        last_element = self.array[self.array[0]]
        # Move the last element to the root and reduce the heap size
        self.array[1] = last_element
        self.array[0] -= 1
        if self.array[0] > 1:
            self._heapify_down(1)
        # Remove the last element (now redundant)
        self.array.pop()
        return min_element

    def _heapify_down(self, index):
        """
        Moves the element at the specified index down to maintain the heap property.
        """
        while 2 * index <= self.array[0]:
            left_child_index = 2 * index
            right_child_index = left_child_index + 1

            # Determine which child is smaller
            if right_child_index > self.array[0] or self.array[left_child_index].distance < self.array[right_child_index].distance:
                smaller_child_index = left_child_index
            else:
                smaller_child_index = right_child_index

            # If the current element is greater than the smaller child, swap them
            if self.array[index].distance > self.array[smaller_child_index].distance:
                self.array[index], self.array[smaller_child_index] = self.array[smaller_child_index], self.array[index]
                index = smaller_child_index
            else:
                break
