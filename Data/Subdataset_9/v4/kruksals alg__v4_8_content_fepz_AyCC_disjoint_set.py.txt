class DisjointSets:
    """
    Represents a collection of disjoint sets.
    """

    def __init__(self, elements):
        """
        Initializes the DisjointSets with a list of elements.

        Parameters:
            elements (iterable): The elements to initialize the sets with.
        """
        self.elements = list(elements)
        self.sets = list(range(len(elements)))
        self.ranks = [1] * len(elements)

    def find(self, x):
        """
        Finds the label of the set containing object x.

        Parameters:
            x: The element whose set label needs to be found.

        Returns:
            int: The label of the set containing x.
        """
        i = r = self.elements.index(x)
        while self.sets[r] != r:
            r = self.sets[r]
        while i != r:
            j = self.sets[i]
            self.sets[i] = r
            i = j
        return r

    def merge(self, a, b):
        """
        Merges the sets labelled a and b; we assume a != b.

        Parameters:
            a: The label of the first set to merge.
            b: The label of the second set to merge.
        """
        if self.ranks[a] == self.ranks[b]:
            self.ranks[a] = self.ranks[a] + 1
            self.sets[b] = a
        else:
            if self.ranks[a] > self.ranks[b]:
                self.sets[b] = a
            else:
                self.sets[a] = b
