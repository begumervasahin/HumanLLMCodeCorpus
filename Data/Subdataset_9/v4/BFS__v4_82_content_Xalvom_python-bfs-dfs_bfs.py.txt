# Define the graph
graph = {
    'A': set(['B']),
    'B': set(['A', 'C']),
    'C': set(['B', 'D', 'H', 'I']),
    'D': set(['C', 'E', 'F', 'H', 'M']),
    'E': set(['D']),
    'F': set(['D', 'G', 'M']),
    'G': set(['F', 'H']),
    'H': set(['C', 'D', 'G', 'L']),
    'I': set(['C', 'J', 'K']),
    'J': set(['I']),
    'K': set(['I', 'L']),
    'L': set(['H', 'K']),
    'M': set(['D', 'F'])
}

print("*********BFS***********")

# Input start and end nodes
start_node = input("Input Start Node: ")
end_node = input("Input Target Node: ")

print("***********************")
print()

def bfs(graph, start, target):
    queue = [[start]]  # Initialize the queue with the start node
    visited = set()    # Set to keep track of visited nodes

    while queue:
        path = queue.pop(0)   # Get the first path from the queue
        current_node = path[-1]  # Get the current node from the path

        # Check if the current node is the target node
        if current_node == target:
            return path

        # Check if the current node has been visited
        if current_node not in visited:
            # Explore all neighbors of the current node
            for neighbor in graph.get(current_node, []):
                new_path = list(path)  # Create a new path
                new_path.append(neighbor)  # Add the neighbor to the new path
                queue.append(new_path)   # Add the new path to the queue

            visited.add(current_node)  # Mark the current node as visited

    return "Path not found"

print(bfs(graph, start_node, end_node))  # Perform BFS search

print()
