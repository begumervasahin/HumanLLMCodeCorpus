import doctest

def fizzbuzz(n):
    """
    Returns 'fizz' if n is divisible by three.
    Returns 'buzz' if n is divisible by five.
    Returns 'fizzbuzz' if n is divisible by both three and five.
    Otherwise, returns n as a string.

    >>> fizzbuzz(1)
    '1'
    >>> fizzbuzz(2)
    '2'
    >>> fizzbuzz(3)
    'fizz'
    >>> fizzbuzz(4)
    '4'
    >>> fizzbuzz(5)
    'buzz'
    >>> fizzbuzz(6)
    'fizz'
    >>> fizzbuzz(14)
    '14'
    >>> fizzbuzz(15)
    'fizzbuzz'
    >>> fizzbuzz(16)
    '16'
    """
    return str(n)

if __name__ == '__main__':
    # Run doctests
    doctest.testmod()

    # Iterate through numbers from 0 to 99 and print their fizzbuzz results
    for n in range(100):
        print(fizzbuzz(n))
