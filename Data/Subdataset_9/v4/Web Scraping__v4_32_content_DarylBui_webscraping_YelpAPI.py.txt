def do_search(term='Food', location='San Francisco'):
    # Set up base URL and format search parameters
    base_url = 'https://api.yelp.com/v2/search'
    term = term.replace(' ', '+')
    location = location.replace(' ', '+')
    
    # Construct the full URL with search parameters
    url = "{base_url}?term={term}&location={location}".format(base_url=base_url,
                        term=term,
                        location=location)
    
    # Set up OAuth1 authentication
    auth = OAuth1(api_key, consumer_secret, token, token_secret)
    
    # Make a GET request to Yelp API
    response = requests.get(url, auth=auth)
    
    # Extract JSON data and text data from the response
    json_data = response.json()
    text_data = response.text
    
    return json_data, text_data
