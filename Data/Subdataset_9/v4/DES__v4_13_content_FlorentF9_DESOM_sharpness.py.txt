import numpy as np

def prototype_sharpness_ratio(input_samples, som_prototypes):
    """
    Calculate the prototype sharpness ratio for image datasets.

    Parameters
    ----------
    input_samples : array-like, shape = [n_samples, input_dim]
        Input samples.
    som_prototypes : array-like
        SOM prototype vectors.
    """

    # Calculate the size of the input samples and reshape them
    input_size = int(np.sqrt(input_samples.shape[-1]))
    input_samples_reshaped = input_samples.reshape(-1, input_size, input_size, 1)

    # Calculate the size of the SOM prototypes and reshape them
    som_size = int(np.sqrt(som_prototypes.shape[-1]))
    som_prototypes_reshaped = som_prototypes.reshape(-1, som_size, som_size, 1)

    # Calculate sharpness for input samples
    data_sharpness = calculate_sharpness(input_samples_reshaped)

    # Calculate sharpness for SOM prototypes
    som_sharpness = calculate_sharpness(som_prototypes_reshaped)

    # Calculate and return the sharpness ratio
    return som_sharpness / data_sharpness

def calculate_sharpness(data):
    """
    Calculate sharpness of data using gradient.

    Parameters
    ----------
    data : array-like
        Input data to calculate sharpness.

    Returns
    -------
    float
        Sharpness value.
    """
    total_sharpness = 0.0
    for sample in data:
        gx, gy = np.gradient(sample[:, :, 0])
        gradient_norm = np.sqrt(gx ** 2 + gy ** 2)
        total_sharpness += np.mean(gradient_norm)
    return total_sharpness / data.shape[0]

    input_samples = np.random.rand(100, 64)  # 100 samples of 64-dimensional data
    som_prototypes = np.random.rand(10, 64)  # 10 SOM prototypes of 64 dimensions each  

    # Calculate sharpness ratio
    sharpness_ratio = prototype_sharpness_ratio(input_samples, som_prototypes)
    print("Sharpness Ratio:", sharpness_ratio)
