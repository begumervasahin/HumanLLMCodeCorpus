# Import necessary libraries
from bs4 import BeautifulSoup as BS
import requests
import pandas as pd

# Define user-agent headers for making requests
headers = {
    'user-agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/77.0.3865.120 Safari/537.36',
    'referrer': 'https://google.com',
    'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,image/webp,image/apng,*/*;q=0.8',
    'Accept-Encoding': 'gzip, deflate, br',
    'Accept-Language': 'en-US,en;q=0.9',
    'Pragma': 'no-cache',
    'Cache-Control': 'no-cache'
}

# URLs to scrape
urls = [
    "https://www.subaruofmiami.com/promotions/service/index.htm",
    "https://www.lehmansubaru.com/promotions/service/index.htm",
]

# Function to scrape data from the first website
def scrape_subaru_of_miami(url):
    # Initialize a dictionary to store scraped data
    scraped_data = {"miami": {}}
    
    # Make a request to the URL
    response = requests.get(url, timeout=15, headers=headers)
    
    # Parse the HTML content
    soup = BS(response.content, "html.parser")
    
    # Find promo title and price HTML elements
    promo_titles = soup.find_all("div", class_="promo-title text-center h1")
    promo_price = soup.find_all("div", class_="promo-discountValue text-center h1 has-image")[2]
    
    # Extract relevant data
    for title in promo_titles:
        if "OIL" in title.text.strip():  # Check if the promo is for oil change
            scraped_data["miami"]["url"] = url[0:29]
            scraped_data["miami"]["product"] = title.text.strip()
            scraped_data["miami"]["price"] = promo_price.text[1:8]
    
    # Convert scraped data to DataFrame and display
    print(pd.DataFrame.from_dict(scraped_data, orient='index'))
