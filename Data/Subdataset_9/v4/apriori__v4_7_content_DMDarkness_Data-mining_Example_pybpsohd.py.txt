import pybpsohd as pybp  # Import the package for BPSO-HD
import re  # Import the regular expression module for string splitting and extracting items from transactions

def read_dataset(file_name):
    """
    Read the transactional dataset from the given file and return a list of transactions.
    Each transaction is represented as a list of integers.
    """
    transactions = []
    with open(file_name, 'r') as file:
        for line in file:
            # Split the dataset by space or newline and extract integers
            items = re.findall(r'\d+', line)
            if items:
                transactions.append([int(item) for item in items])
    return transactions

if __name__ == '__main__':
    # Read the transactional dataset from the file
    dataset = read_dataset("kosarak.dat")

    # Mine long patterns from the dataset
    # Set the minimum support to 0.001% with a pattern length of 10
    # Use a population size of 30 and run for 30 generations
    # Set the inertia weight to 0.5 and both accelerations to 1
    frequent_patterns = pybp.getFP(dataset, 0.00001, 30, 30, 0.5, 1, 1, 10)

    # Display the frequent patterns
    for pattern in frequent_patterns:
        print(pattern)
