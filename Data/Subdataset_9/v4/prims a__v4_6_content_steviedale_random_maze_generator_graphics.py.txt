import pygame
from time import sleep

# Define colors
RED = (255, 0, 0)
BLUE = (0, 0, 255)
GREEN = (0, 255, 0)
BLACK = (0, 0, 0)
WHITE = (255, 255, 255)

class BoardDisplay:
    def __init__(self, num_rows, num_columns, width) -> None:
        # Initialize pygame
        pygame.init()
        
        # Set display dimensions
        self.display = pygame.display.set_mode((num_columns * width, num_rows * width))
        
        # Store board dimensions and square width
        self.num_rows = num_rows
        self.num_columns = num_columns
        self.width = width
        
        # Fill the display with black color
        self.display.fill(BLACK)

    def draw_position(self, row, col, color):
        # Ensure row and column indices are integers
        assert(type(row) is int)
        assert(type(col) is int)
        
        # Draw a colored rectangle at the specified position
        pygame.draw.rect(self.display, color, pygame.Rect(row*self.width, col*self.width, self.width, self.width))

    def start(self, trace):
