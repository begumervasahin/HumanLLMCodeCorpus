# Import required modules
from kruskal import Algoritimo_kruskal as kr
import networkx as nx
import random
import time
import matplotlib.pyplot as plt
import prim_graph as prim

def generate_complete_graphs():
    kruskal = kr()
    vertices_list = []
    kruskal_avg_times = []
    prim_avg_times = []
    total_vertices = 2011
    total_time = time.time()

    for num_vertices in range(10, total_vertices, 100):
        print('Number of Vertices:', num_vertices)
        G = nx.complete_graph(num_vertices)
        edges = G.edges()
        matrix = [[0] * num_vertices for _ in range(num_vertices)]

        for v1, v2 in edges:
            weight = random.randint(10, 100)
            matrix[v1][v2] = weight
            matrix[v2][v1] = weight
            G.edges[v1, v2]['weight'] = weight

        prim_graph = prim.Graph(num_vertices)
        prim_graph.graph = matrix

        kruskal_time_sum = 0
        prim_time_sum = 0

        for _ in range(10):
            start_time = time.time()
            kruskal_mst = kruskal.kruskal(G)
            kruskal_time = time.time() - start_time
            kruskal_time_sum += kruskal_time

            start_time = time.time()
            prim_mst = prim_graph.primMST()
            prim_time = time.time() - start_time
            prim_time_sum += prim_time

        print("Average Execution Time (Kruskal):", kruskal_time_sum / 10, "seconds")
        print("Average Execution Time (Prim):", prim_time_sum / 10, "seconds")

        vertices_list.append(num_vertices)
        kruskal_avg_times.append(kruskal_time_sum / 10)
        prim_avg_times.append(prim_time_sum / 10)

    print("Total Time:", time.time() - total_time)

    plt.plot(vertices_list, prim_avg_times, label="Prim's Algorithm")
    plt.plot(vertices_list, kruskal_avg_times, label="Kruskal's Algorithm")
    plt.legend()
    plt.axis([0, total_vertices + 10, 0, max(kruskal_avg_times) * 2])
    plt.ylabel('Execution Time (Seconds)')
    plt.xlabel('Number of Vertices')
    plt.title("Performance Comparison: Kruskal vs Prim (Complete Graphs)")
    plt.savefig('Kruskal_vs_Prim_Complete_Graphs.png')
    plt.show()

    print("Vertices List:")
    print(vertices_list)
    print("Average Kruskal Times:")
    print(kruskal_avg_times)
    print("Average Prim Times:")
    print(prim_avg_times)

def generate_graph(num_vertices):
    G = nx.connected_watts_strogatz_graph(num_vertices, 25, 0.5)
    edges = G.edges()
    matrix = [[0] * num_vertices for _ in range(num_vertices)]

    for v1, v2 in edges:
        weight = random.randint(10, 100)
        matrix[v1][v2] = weight
        matrix[v2][v1] = weight
        G.edges[v1, v2]['weight'] = weight

    kruskal = kr()
    prim_graph = prim.Graph(num_vertices)
    prim_graph.graph = matrix

    kruskal_mst = kruskal.kruskal(G)
    prim_mst = prim_graph.primMST()

    plt.show()
    nx.draw(G, node_size=5, alpha=1, node_color="black", edge_color="blue")
    plt.axis('equal')
    plt.savefig('Graph.png')
    plt.show()

    plt.figure(figsize=(8, 8))
    nx.draw(prim_mst, node_size=5, alpha=1, node_color="black", edge_color="red")
    plt.axis('equal')
    plt.savefig('Prim_MST.png')
    plt.show()

    plt.figure(figsize=(8, 8))
    nx.draw(kruskal_mst, node_size=5, alpha=1, node_color="black", edge_color="green")
    plt.axis('equal')
    plt.savefig('Kruskal_MST.png')
    plt.show()

generate_graph(500)
