"""
This script provides functions for reading and extracting text data from PDF files.
Author: Mubashar Hussain
"""

import fitz
from storage import enumrateFilenames

def read_all_pdfs():
    """
    Read all PDF files from the storage and return the text along with the file name.

    Returns:
        list: A list of tuples containing PDF name and text data from all the PDFs.
    """
    pages = []
    for pdf in enumrateFilenames():
        with fitz.open(pdf) as infile:
            for page in infile:
                pages.append((pdf, page.get_text()))
    return pages

def read_pdf(pdfname):
    """
    Read a PDF file from the storage and return the text along with the file name.

    Args:
        pdfname (str): Path to the PDF file.

    Returns:
        list: A list containing a tuple with PDF name and text data from the PDF.
    """
    pages = []
    with fitz.open(pdfname) as infile:
        for page in infile:
            pages.append((pdfname, page.get_text()))
    return pages
