import struct
import os

def long_to_bytes(n, blocksize=0):
    """Convert an integer to a byte string.
    
    Args:
        n (int): The integer to convert.
        blocksize (int, optional): The block size for padding (default: 0).
    
    Returns:
        bytes: The byte string representation of the integer.
    """
    s = b''
    n = int(n)
    pack = struct.pack
    while n > 0:
        s = pack('>I', n & 0xffffffff) + s
        n = n >> 32
    for i in range(len(s)):
        if s[i] != 0:
            break
    else:
        s = b'\x00'
        i = 0
    s = s[i:]
    if blocksize > 0 and len(s) % blocksize:
        s = (blocksize - len(s) % blocksize) * b'\x00' + s
    return s

def strxor(var1, var2):
    """Perform a bitwise XOR operation on two byte strings."""
    return bytes([a ^ b for (a, b) in zip(var1, var2)])

def get_random_bytes(n):
    """Generate n random bytes."""
