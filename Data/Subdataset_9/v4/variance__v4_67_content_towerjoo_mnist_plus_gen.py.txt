# Import necessary libraries
import random
import os
import uuid
from keras.datasets import mnist
import matplotlib.pyplot as plt
from PIL import Image

# Function to generate synthetic dataset
def generate_dataset(train_samples=5000, test_samples=1000):
    # Load MNIST dataset
    (x_train, y_train), (x_test, y_test) = mnist.load_data()
    
    # Define directories for train and test data
    train_dir = "datasets/train/"
    test_dir = "datasets/test/"
    
    # Create directories if they don't exist
    if not os.path.exists(train_dir):
        os.makedirs(train_dir)
    if not os.path.exists(test_dir):
        os.makedirs(test_dir)
    
    # Loop to generate train and test data
    for i in range(train_samples + test_samples):
        # Randomly select an image from the training set
        index = random.choice(range(x_train.shape[0]))
        image = Image.fromarray(x_train[index])  # Convert array to PIL Image
        label = str(y_train[index])  # Get corresponding label
        
        # Apply random transformations
        resize_ratio = random.random() * 0.5 + 0.5  # Random resize ratio between 0.5 and 1
        new_size = (int(image.size[0] * resize_ratio), int(image.size[1] * resize_ratio))
        image = image.resize(new_size)  # Resize image
        
        rotate_degree = random.randint(0, 90)  # Random rotation degree between 0 and 90
        image = image.rotate(rotate_degree)  # Rotate image
        
        image = image.resize((28, 28))  # Resize image to MNIST standard size
        
        # Define path for saving image
        if i < train_samples:
            path = "{}{}-{}.jpg".format(train_dir, label, uuid.uuid4().hex)
        else:
            path = "{}{}-{}.jpg".format(test_dir, label, uuid.uuid4().hex)
        
        # Save image
        image.save(path)

# Entry point of the program
if __name__ == "__main__":
    generate_dataset()  # Call the function to generate dataset
