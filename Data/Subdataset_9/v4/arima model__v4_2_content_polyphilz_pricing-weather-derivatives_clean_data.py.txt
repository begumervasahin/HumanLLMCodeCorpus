import sys
import pandas as pd

# Function to check the legitimacy of temperature data
def check_legitimacy(grouped_data):
    """
    Check if the maximum temperature exceeds 100˚F for any given day.
    Some specific dates are known to have recorded high temperatures.
    """
    for date, row in grouped_data.iterrows():
        if row["max"] > 100:
            print(f"Date: {date}, Min Temp: {row['min']}, Max Temp: {row['max']}, Mean Temp: {row['mean']}")

# Function to fill NaN values using forward filling method
def fill_missing_values(grouped_data):
    """
    Fill missing temperature values using forward fill method.
    """
    grouped_data.fillna(method="ffill", inplace=True)
    return grouped_data

# Function to replace negative values with NaN
def replace_negatives_with_nan(grouped_data):
    """
    Replace negative temperature values with NaN.
    """
    grouped_data["min"] = grouped_data["min"].apply(lambda x: float("nan") if x < 0 else x)
    grouped_data["max"] = grouped_data["max"].apply(lambda x: float("nan") if x < 0 else x)
    grouped_data["mean"] = grouped_data["mean"].apply(lambda x: float("nan") if x < 12.46 else x)
    return grouped_data

# Function to group temperature data by day and calculate min, max, and mean temperatures
def group_temperature_data(df):
    """
    Group temperature data by day and calculate min, max, and mean temperatures.
    """
    grouped_data = df.groupby(pd.Grouper(freq="D"))["TemperatureF"].agg(
        ["min", "max", "mean"]
    )
    return grouped_data

# Function to process the raw temperature data
def process_temperature_data(df):
    """
    Process raw temperature data by grouping, filling missing values, and replacing negatives with NaNs.
    """
    grouped_data = group_temperature_data(df)
    grouped_data = replace_negatives_with_nan(grouped_data)
    grouped_data = fill_missing_values(grouped_data)
    return grouped_data

# Main function to execute the script
def main():
    try:
        # Read raw temperature data from CSV file
        df = pd.read_csv("temp_data_raw.csv", index_col="Time", parse_dates=True)
    except FileNotFoundError:
        print("Please run `scrape_data.py` to generate the raw data first.")
        sys.exit(0)
    
    # Drop unnamed columns if any
    df.drop(df.columns[df.columns.str.contains("Unnamed")], axis=1, inplace=True)
    
    # Process the temperature data
    grouped_data = process_temperature_data(df)
    
    # Check the legitimacy of temperature data
    check_legitimacy(grouped_data)
    
    # Save cleaned temperature data to CSV file
    grouped_data.to_csv("temp_data_cleaned.csv")

# Entry point of the script
if __name__ == "__main__":
    main()
