import sys

# Function to check if a number is prime
def is_prime(number):
    number *= 1.0
    prime = True
    for divisor in range(2, int(number ** 0.5 + 1)):
        if number % divisor == 0:
            prime = False
            break
    return prime

# Function to check if a number has not been visited
def not_visited(number, visited):
    for item in visited:
        if item == number:
            return False
    return True

# Function to get possible actions by changing digits
def get_possible_actions(number, visited):
    possible_primes = []
    str_number = str(number)
    for i in range(1, len(str_number) + 1):
        for digit in range(10):
            new_number = str_number[:i - 1] + str(digit) + str_number[i:]
            if new_number[0] != '0' and new_number != str_number \
                    and not_visited(new_number, visited) and is_prime(int(new_number)):
                possible_primes.append(new_number)
    return possible_primes

# Function to find a path from startP to endP
def get_path(start, end):
    visited_start = []
    visited_end = []
    queue_start = [[str(start)]]
    queue_end = [[str(end)]]
    path_start = []
    path_end = []

    while queue_start or queue_end:
        if queue_start:
            path_start = queue_start.pop(0)
            node_start = path_start[-1]
            visited_start.append(node_start)

            if node_start == end or not not_visited(node_start, visited_end):
                return path_start

            adjacent_start = get_possible_actions(node_start, visited_start)
            for adj_start in adjacent_start:
                new_path_start = list(path_start)
                new_path_start.append(adj_start)
                queue_start.append(new_path_start)

        if queue_end:
            path_end = queue_end.pop(0)
            node_end = path_end[-1]
            visited_end.append(node_end)

            if node_end == start or not not_visited(node_end, visited_start):
                return path_end

            adjacent_end = get_possible_actions(node_end, visited_end)
            for adj_end in adjacent_end:
                new_path_end = list(path_end)
                new_path_end.append(adj_end)
                queue_end.append(new_path_end)

    return []

def main():
    for line in sys.stdin.readlines():
        primes = str(line).split()
        paths = get_path(primes[0], primes[1])
        if len(paths) == 0 or paths[-1] != primes[1]:
            print("UNSOLVABLE")
        else:
            print(' '.join(paths))

if __name__ == '__main__':
    main()
