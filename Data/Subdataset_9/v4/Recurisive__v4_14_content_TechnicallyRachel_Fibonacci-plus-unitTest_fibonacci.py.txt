def fib(n):
    """Generate Fibonacci series up to n."""
    result = []
    a, b = 0, 1
    while b < n:
        result.append(b)
        a, b = b, a + b
    return result

def main():
    """Main function to demonstrate Fibonacci series generation."""
    print(fib(10))

if __name__ == '__main__':
    main()
