from flask import Flask, render_template, request
import util

# Initialize Flask application
app = Flask(__name__)

# Route for the home page
@app.route("/")
def root():
    return render_template("index.html")

# Route for the results page
@app.route("/results", methods=["GET", "POST"])
def results():
    if request.method == "GET":
        # Get the query and mode from the request
        query = request.args['question']
        mode = request.args['type']
        
        # Get the list of URLs based on the query
        urls = util.get_list_of_urls(query, 10)
        
        # Initialize dictionary to store current results
        current_results = {}
        
        # Process each URL
        for url in urls:
            # Get the text from the URL
            text = util.get_text_from_url(url)
            
            # Determine the mode and update results accordingly
            if mode == 'Who':
                current_results = util.find_name(text, current_results)
            else:
                current_results = util.find_date(text, current_results)
        
        # Sort the results by value and limit to top 10
        sorted_results = util.sort_dict_by_value(current_results, 10)
        
        # Render the results template with query and sorted results
        return render_template('results.html', question=query, results=sorted_results)
    else:
        # If the request method is POST, redirect to the home page
        return render_template('index.html')

if __name__ == "__main__":
    # Run the Flask application
    app.run(host="0.0.0.0", port=8000, debug=True)
