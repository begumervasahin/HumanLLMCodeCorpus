class DisjointSet:
    def __init__(self, size):
        if size < 0:
            raise ValueError("Size must be >= 0")
        self.size = size
        # Initialize the disjoint set with each element as a root (-1 indicates a root)
        self.set = [-1] * (size + 1)

    def union(self, root1, root2):
        # Check if the given roots are valid
        if not (0 <= root1 <= self.size) or not (0 <= root2 <= self.size):
            raise ValueError("Illegal value")
        # Make root2 join the set of root1
        self.set[root2] = root1

    def find(self, root):
        # Check if the given root is valid
        if not (0 <= root <= self.size):
            raise ValueError("Illegal value")
        # If the element is a root, return its value
        if self.set[root] < 0:
            return root
        # Otherwise, recursively find the root of the element
        else:
            return self.find(self.set[root])
