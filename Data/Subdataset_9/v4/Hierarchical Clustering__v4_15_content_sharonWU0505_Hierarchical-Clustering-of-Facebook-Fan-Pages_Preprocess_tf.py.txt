import os
import sys
import csv
import re
import jieba
from tqdm import tqdm

# Get the path to the directory containing posts from command-line arguments
posts_path = sys.argv[1]

# List all directories in the posts directory
dirs = os.listdir(posts_path)
dir_count = len(dirs)

# Iterate over each directory
for idx, d in enumerate(dirs):
    print(f'Progress: {idx + 1}/{dir_count}')
    pages_path = os.path.join(posts_path, d)

    # List all files in the current directory
    pages = os.listdir(pages_path)

    # Create a directory for storing TF files if it doesn't exist
    if not os.path.isdir('tfs'):
        os.makedirs('tfs')

    # Iterate over each file in the directory
    for i in tqdm(pages):
        tfs = dict()  # Dictionary to store term frequencies

        # Read the contents of the file
        with open(os.path.join(pages_path, i), 'r') as file:
            # Read the CSV content and remove null characters
            csv_content = list(csv.reader((line.replace('\0', '') for line in file)))

            # Check if there are at least two rows in the CSV
            if len(csv_content) >= 2:
                page_name = csv_content[1][0]  # Get the page name from the second row
                csv_content = csv_content[1:]  # Exclude the first row (header)

                # Tokenize and preprocess text using Jieba and regular expressions
                tokens = [list(jieba.cut(re.sub(r'^https?:\/\/.*[\r\n]*', '', i[1], flags=re.MULTILINE)))
                          for i in csv_content if len(i) >= 2]
                tokens = [token for sublist in tokens for token in sublist]

                # Calculate term frequencies
                for term in tokens:
                    tfs[term] = tfs.get(term, 0) + 1

                # Write term frequencies to a CSV file
                with open(f'tfs/{page_name.replace("/", "")}.csv', 'w') as w:
                    writer = csv.writer(w)
                    for term, frequency in tfs.items():
                        writer.writerow([term, frequency])
