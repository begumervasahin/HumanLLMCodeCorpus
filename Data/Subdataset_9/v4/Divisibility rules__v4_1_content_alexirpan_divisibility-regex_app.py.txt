from flask import Flask, render_template, request
import divisibility_dfa

app = Flask(__name__)

@app.route('/')
def homepage():
    """Render the homepage."""
    return render_template('index.html')

@app.route('/faq')
def faq():
    """Render the FAQ page."""
    return render_template('faq.html')

@app.route('/regex', methods=['POST'])
def generate_regex():
    """Generate regex based on user input."""
    form = request.form

    # Check if 'base' and 'divisor' fields are present in the form
    if 'base' not in form or 'divisor' not in form:
        return 'Missing form fields'

    # Extract base and divisor from the form
    base = form['base']
    div = form['divisor']

    # Validate the base input
    valid_bases = ('binary', 'decimal', 'hexadecimal')
    if base not in valid_bases:
        return 'Invalid base'

    # Validate the divisor input
    valid_divisors = [str(i) for i in range(1, 10)]
    if div not in valid_divisors:
        return 'Invalid divisor'

    # Convert divisor to integer
    div = int(div)

    # Build DFA and set start state
    dfa = divisibility_dfa.build_dfa(base, div)
    dfa.set_start(div)

    # Render the regex template with generated regex
    return render_template('regex.html', regex=divisibility_dfa.dfa_to_regex(dfa, 0))
