def locate_the_middle_node(linked_list):
    """
    Locates the middle node of a linked list.

    Args:
        linked_list (LinkedList): The linked list.

    Returns:
        Node: The middle node of the linked list.
    """
    cursor = linked_list.head
    ds_cursor = linked_list.head
    while ds_cursor and ds_cursor.next_node:
        prev_cursor = cursor
        cursor = cursor.next_node
        ds_cursor = ds_cursor.next_node.next_node
    prev_cursor.next_node = None
    return cursor

def reverse_second_half(middle_node):
    """
    Reverses the second half of a linked list starting from the middle node.

    Args:
        middle_node (Node): The middle node of the linked list.

    Returns:
        Node: The new head of the second half of the linked list.
    """
    cursor = middle_node
    next_cursor = cursor.next_node
    cursor.next_node = None
    prev_cursor = cursor
    cursor = next_cursor
    while cursor.next_node:
        next_cursor = cursor.next_node
        cursor.next_node = prev_cursor
        prev_cursor = cursor
        cursor = next_cursor
    cursor.next_node = prev_cursor
    return cursor
