import pyAesCrypt

class AES256:
    """ 
    AES256-CBC encryption/decryption implementation. 
    """

    def __init__(self, bufferSize=512*1024):
        """ 
        Initialize AES256 object with a given buffer size.

        Args:
            bufferSize (int): Buffer memory size in bytes (default: 512*1024).
        """
        self.bufferSize = bufferSize

    def crypt(self, filename):
        """ 
        Encrypts a file with AES256-CBC encryption.

        Args:
            filename (str): The name of the file to be encrypted.
        """
        print('-----------------------------------')
        password = input('Enter key: ')
        try:
            pyAesCrypt.encryptFile(filename, filename + '.aes', password, self.bufferSize)       
            print('[Crypted] ' + filename + '.aes')
        except Exception as e:
            print('Error:', e)

    def decrypt(self, filename):
        """ 
        Decrypts a file encrypted with AES256-CBC encryption.

        Args:
            filename (str): The name of the file to be decrypted.
        """
        try:
            print('-----------------------------------')        
            password = input('Enter key: ')
            pyAesCrypt.decryptFile(filename + '.aes', 'decrypted.' + filename, password, self.bufferSize)
            print('[Decrypted] decrypted.' + filename)
        except Exception as e:
            print('Error:', e)
