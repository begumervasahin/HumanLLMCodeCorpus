# Importing necessary libraries
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from sklearn import decomposition
from sklearn import datasets
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import train_test_split
from sklearn.neighbors import KNeighborsClassifier
from sklearn import metrics

# Load the dataset
df = pd.read_csv("D:\Python_programs\ML\iris.csv")

# Extracting features and target variable
features = ['sepal length', 'sepal width', 'petal length', 'petal width']       
X = df.loc[:, features].values
Y = df.loc[:,['target']].values

# Splitting the dataset into training and testing sets
X_train, X_test, y_train, y_test = train_test_split(
    X, Y, test_size=0.3, random_state=100)

# Flatten the target arrays
y_train = y_train.ravel()
y_test = y_test.ravel()

# Initialize and train a K-Nearest Neighbors model
model = KNeighborsClassifier()
model.fit(X_train, y_train)

# Make predictions on the test set
Yhat = model.predict(X_test)

# Evaluate the model's accuracy before applying PCA
print('*'*11, 'Accuracy of IRIS Dataset Before PCA', '*'*11, '\n')
for K in range(25):
    K_value = K + 1
    # Initialize KNN classifier with current K value
    neigh = KNeighborsClassifier(n_neighbors=K_value, weights='uniform', algorithm='auto')
    # Train the model
    neigh.fit(X_train, y_train)
    # Make predictions
    y_pred = neigh.predict(X_test)
    # Calculate and print accuracy
    print("Accuracy is ", metrics.accuracy_score(y_test, y_pred) * 100, "% for K-Value:", K_value)
