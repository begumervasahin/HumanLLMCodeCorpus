# Importing necessary libraries
import pandas as pd
import numpy as np
import math
import matplotlib.pyplot as plt
from sklearn import svm
from sklearn.metrics import mean_squared_error, mean_absolute_error

# Defining a function to calculate Mean Absolute Percentage Error
def mean_absolute_percentage_error(y_true, y_pred):
    sum_error = 0
    for i in range(len(y_true)):
        sum_error += abs(y_pred[i] - y_true[i]) / y_true[i]
    return (sum_error / len(y_true)) * 100

# Custom function to parse date strings
def parse_datetime(x):
    return pd.to_datetime(x, format='%Y-%m-%d %H:%M:%S')

# Reading the dataset
data = pd.read_csv('/Users/daihanru/Desktop/arima-lstm/DataSet/FEB15-2.csv', usecols=[2, 3], header=0,
                   parse_dates=[0], index_col=0, squeeze=True, date_parser=parse_datetime)

# Extracting values and converting to float32
values = data.values.astype('float32')

# Defining train and test data ranges
train_start_index, train_end_index = 4500, 6000
test_start_index, test_end_index = 6409, 6600

# Splitting the data into train and test sets
train_data, test_data = values[train_start_index:train_end_index], values[test_start_index - 1:test_end_index - 1]
real_test_data = np.array(values[test_start_index:test_end_index]).reshape(-1, 1)
train_data = np.array(train_data).reshape(-1, 1)
test_data = np.array(test_data).reshape(-1, 1)

# Creating and training SVR model
model = svm.SVR()
model.fit(train_data[0:train_end_index - train_start_index - 1], train_data[1:train_end_index - train_start_index])

# Predicting with SVR model
predicted_values = model.predict(test_data)

# Calculating evaluation metrics
mse = mean_squared_error(real_test_data, test_data)
rmse = math.sqrt(mse)
mae = mean_absolute_error(real_test_data, test_data)
mape = mean_absolute_percentage_error(real_test_data, test_data)

# Printing evaluation metrics
print('SVR Test MAE: %.3f MSE: %.3f RMSE: %.3f MAPE: %.3f%%' % (mae, mse, rmse, mape))

# Plotting real vs predicted values
plt.plot(real_test_data, '-', label="Real Flow")
plt.plot(predicted_values, '--', color='red', label="SVR Predictions")
plt.legend(loc='upper right')
plt.xlabel("Period (15-minute intervals)")
plt.ylabel("Volume (vehicles/period)")
plt.ylim(0, 800)
plt.show()
