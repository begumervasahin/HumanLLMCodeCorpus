import numpy as np
import math
from ROOT import *

# Load training data
train_file = "../Higgs13TeV_train_118_130_ggh.csv"
reader = np.loadtxt(train_file, delimiter=",")      
X_train = reader[:, 0:21].astype(float)
Y_train = reader[:, 21]
nevents_train = len(Y_train)
dimensions = len(X_train[0])

# Calculate number of covariance matrices
n_covariances = math.factorial(dimensions) / (math.factorial(dimensions - 2) * 2)

print(f"You have {dimensions} dimensions resulting in {n_covariances} covariance matrices.")

# Separate signal and background events
ns = sum(1 for label in Y_train if label == 1)
nb = nevents_train - ns
limit = min(ns, nb)

sX_train = X_train[Y_train == 1][:limit]
bX_train = X_train[Y_train == 0][:limit]

# Define dimension names
dim_names = ["l1 p_{T}", "l2 p_{T}", "l3 p_{T}", "l4 p_{T}", 
             "j1 p_{T}", "j2 p_{T}", "Njets"]

# Initialize covariance matrix and histogram
fconv_matrix = np.zeros((dimensions, dimensions))
fconv_hist = TH2D("fconv_hist", "", 21, 0, 21, 21, 0, 21)

# Calculate covariance matrix
svals_sum = np.sum(sX_train, axis=0)
bvals_sum = np.sum(bX_train, axis=0)

for st_d in range(dimensions):
    for nd_d in range(dimensions):
        fcovariance = np.dot((sX_train[:, st_d] - svals_sum[st_d]), 
                             (bX_train[:, nd_d] - bvals_sum[nd_d])) / (limit - 1)
        fconv_matrix[st_d][nd_d] = fcovariance
        fconv_hist.SetBinContent(nd_d + 1, dimensions - st_d, fcovariance)

# Write covariance matrix to file
with open('full_covariance_matrix.txt', 'w') as full_conv_matrix:
    for row in fconv_matrix:
        full_conv_matrix.write(','.join(map(str, row)) + '\n')

# Plot covariance matrix
gROOT.SetBatch()
gStyle.SetOptStat(0)
gStyle.SetPadLeftMargin(0.14)
gStyle.SetTitleOffset(1.4, "y")
gStyle.SetPaintTextFormat("0.2e")
cv = TCanvas("cv", "", 10, 10, 2300, 1000)
fconv_hist.Draw("text")
fconv_hist.SetTitle("Full Covariance Matrix")
gPad.Update()
cv.Print("plots/Full_covariance_matrix.png")

# Perform PCA
feigenvalues, feigenvectors = np.linalg.eig(fconv_matrix)

# Plot eigenvectors
feig_vec_hist = TH2D("feig_vec_hist", "", 21, 0, 21, 22, 0, 22)

for i, (eigenvalue, eigenvector) in enumerate(zip(feigenvalues, feigenvectors)):
    for j, value in enumerate(eigenvector):
        feig_vec_hist.SetBinContent(j + 1, dimensions - i, value)
    feig_vec_hist.SetBinContent(i + 1, dimensions, eigenvalue)
    feig_vec_hist.GetXaxis().SetBinLabel(i + 1, dim_names[i])
    feig_vec_hist.GetYaxis().SetBinLabel(dimensions - i, dim_names[i])

feig_vec_hist.GetYaxis().SetBinLabel(dimensions, "eigs")
feig_vec_hist.Draw("text")
feig_vec_hist.SetTitle("Full Eigenvectors")
cv.Print("plots/Full_eigenvectors.png")
cv.Close()

# Transpose dataset and eigenvectors
transp_dataset = np.transpose(X_train)
transp_eigenvectors = np.transpose(feigenvectors)

# Sort eigenvectors by eigenvalues
sorted_indexes = np.argsort(feigenvalues)[::-1]
sorted_eigenvectors = [transp_eigenvectors[index] for index in sorted_indexes]
transp_eigenvectors = sorted_eigenvectors

# Write sorted eigenvectors to file
with open('full_eigenvectors.txt', 'w') as eigenvectors_file:
    for eigenvector in transp_eigenvectors:
        eigenvectors_file.write(','.join(map(str, eigenvector)) + '\n')

# Project dataset through eigenvectors space
projected_dataset = np.dot(X_train - np.mean(X_train, axis=0), transp_eigenvectors)
