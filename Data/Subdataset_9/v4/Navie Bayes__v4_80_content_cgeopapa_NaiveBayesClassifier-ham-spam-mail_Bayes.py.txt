from os import listdir
from os.path import isfile, join
import csv
import string
from nltk.corpus import stopwords

# Function to save dictionary to a CSV file
def save(filename, dictionary):
    with open(filename, "w") as file:
        writer = csv.writer(file)
        for key, val in dictionary.items():
            writer.writerow([key, val])

# Function to load dictionary from a CSV file
def load(filename):
    dictionary = {}
    with open(filename, mode='r') as file:
        reader = csv.reader(file)
        for row in reader:
            if len(row) == 2:
                dictionary[row[0]] = float(row[1])
    return dictionary

# Function to clean and preprocess text data
def cleaner(text):
    # Convert text to lowercase
    text = text.lower()
    # Remove punctuation and special characters
    translation_table = dict.fromkeys(map(ord, string.punctuation + '\0'), None)
    text = text.translate(translation_table)
    # Remove stopwords
    stop_words = set(stopwords.words('english'))
    stop_words.add("subject")  # Add "subject" to stopwords
    text = text.replace('\n', ' ')
    mail = text.split(' ')
    mail = [word for word in mail if word not in stop_words]
    return mail

# Function to train and build a dictionary of word frequencies
def train(filename):
    files = [f for f in listdir(filename) if isfile(join(filename, f))]
    dictionary = {}
    total_words = 0
    for file in files:
        with open(join(filename, file), errors="ignore") as f:
            text = f.read()
            mail = cleaner(text)
            total_words += len(mail)
            for word in mail:
                if word in dictionary:
                    dictionary[word] += 1
                else:
                    dictionary[word] = 1
    file_count = len(files)
    for word in dictionary:
        dictionary[word] = float(dictionary[word]) / total_words
    dictionary["fileCount"] = file_count
    return dictionary

# Function to classify input text as spam or ham
def classify(input_text, dictionary, priori):
    input_text = cleaner(input_text)
    p = 1
    for word in input_text:
        if word in dictionary:
            p *= dictionary[word]
        else:
            p *= 1 / (dictionary["fileCount"] + len(dictionary) + 1)
    p *= priori
    return p

# Function to train and save dictionaries for spam and ham
def train_and_save():
    spam_dict = train("spam")
    ham_dict = train("ham")
    save("spam_train_results.csv", spam_dict)
    save("ham_train_results.csv", ham_dict)

# Function to test the classifier
def test():
    ham_dict = load("ham_train_results.csv")
    spam_dict = load("spam_train_results.csv")
    files = [f for f in listdir("input") if isfile(join("input", f))]
    spam_count = ham_count = 0
    for file in files:
        with open(join("input", file), errors="ignore") as f:
            text = f.read()
            ham_prob = classify(text, ham_dict, 0.67)
            spam_prob = classify(text, spam_dict, 0.33)
            if ham_prob <= spam_prob:
                spam_count += 1
            else:
                ham_count += 1
    print("Spam:\t" + str(ham_count))
    print("Ham:\t" + str(spam_count))

# Train and save dictionaries, then test the classifier
train_and_save()
test()
