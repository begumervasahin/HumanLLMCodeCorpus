# Define a function to calculate the factorial of a number recursively
def factorial(n):
    # Base case: factorial of 0 or negative number is 1
    if n < 1:
        return 1
    else:
        # Recursively calculate factorial
        returnNumber = n * factorial(n - 1)
        # Print factorial calculation for each step
        print(str(n) + '! = ' + str(returnNumber))
        return returnNumber

# Calculate and print factorial of 4
factorial(4)
