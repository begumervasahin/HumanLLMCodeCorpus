import urllib.request
import os
import zipfile

# Define directories and URLs for downloading
data_dir = 'data/'
model_dir = 'models/'
bert_model_url = 'https://s3.amazonaws.com/models.huggingface.co/bert/bert-base-uncased-pytorch_model.bin'
bert_config_url = 'https://s3.amazonaws.com/models.huggingface.co/bert/bert-base-uncased-config.json'
bert_vocab_url = 'https://s3.amazonaws.com/models.huggingface.co/bert/bert-base-uncased-vocab.txt'
stsb_dataset_url = 'https://firebasestorage.googleapis.com/v0/b/mtl-sentence-representations.appspot.com/o/data%2FSTS-B.zip?alt=media&token=bddb94a7-8706-4e0d-a694-1109e12273b5'

# Create directories if they don't exist
if not os.path.isdir(data_dir):
    os.mkdir(data_dir)
if not os.path.isdir(model_dir):
    os.mkdir(model_dir)

def download_models():
    print("Starting download process...")

    # Download BERT model files
    print("Downloading bert-base-uncased model...")
    urllib.request.urlretrieve(bert_model_url, os.path.join(model_dir, "pytorch_model.bin"))
    print("BERT model downloaded and saved.")

    print("Downloading bert-base-uncased config file...")
    urllib.request.urlretrieve(bert_config_url, os.path.join(model_dir, "bert_config.json"))
    print("BERT config file downloaded and saved.")

    print("Downloading bert-base-uncased vocabulary file...")
    urllib.request.urlretrieve(bert_vocab_url, os.path.join(model_dir, 'vocab.txt'))
    print("BERT vocabulary file downloaded and saved.")

    # Download and extract STS-B dataset
    print("Downloading STS-B dataset...")
    stsb_zip = os.path.join(data_dir, 'sts_b.zip')
    urllib.request.urlretrieve(stsb_dataset_url, stsb_zip)
    
    with zipfile.ZipFile(stsb_zip) as zip_ref:
        zip_ref.extractall(data_dir)
    
    os.remove(stsb_zip)
    print("STS-B dataset downloaded and extracted.")

    print("Download process completed.")

# Call the download_models function to start the download
download_models()
