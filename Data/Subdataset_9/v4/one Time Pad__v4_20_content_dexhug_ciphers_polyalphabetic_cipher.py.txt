# Polyalphabetic cipher that encrypts a message by shifting the content letters an
# amount given by a word.
# Ex: Shift by DEXTER
# D = 4, E = 5, X = 24, T = 20, E = 5, R = 18
# So the cipher would shift the first letter by 4, second by 5, etc.
# After the word is finished, you simply repeat the process.
# Note: Whitespace is skipped over.
# The settings for the shift word and message are in the Main function.
# Created by Dexter.
# Python 3.6

import string

def encrypt(message, shift):
    """
    Encrypts a message using a polyalphabetic cipher with the given shift values.

    Parameters:
        message (str): The message to be encrypted.
        shift (list): A list of shift values corresponding to each letter in the shift word.

    Returns:
        str: The encrypted message.
    """
    encryption = ""
    counter = 0
    
    for char in message:
        if char in string.ascii_uppercase:
            index = string.ascii_uppercase.find(char)
            shifted_index = (index + shift[counter]) % 26
            encryption += string.ascii_uppercase[shifted_index]
            counter = (counter + 1) % len(shift)
        elif char in string.digits:
            encryption += char
        elif char in string.whitespace:
            encryption += char
            
    return encryption

def decrypt(encryption, shift):
    """
    Decrypts an encrypted message using the inverse shift values.

    Parameters:
        encryption (str): The encrypted message.
        shift (list): A list of shift values corresponding to each letter in the shift word.

    Returns:
        str: The decrypted message.
    """
    decryption = ""
    dec_shift = [(26 - s) % 26 for s in shift]
    
    for char in encryption:
        if char in string.ascii_uppercase:
            index = string.ascii_uppercase.find(char)
            shifted_index = (index + dec_shift[0]) % 26
            decryption += string.ascii_uppercase[shifted_index]
            dec_shift.append(dec_shift.pop(0))
        else:
            decryption += char
            
    return decryption

def Main():
    """
    Main function to demonstrate the use of the polyalphabetic cipher.
    """
    print("Welcome to the Polyalphabetic Cipher.\n")
    
    shift_word = "encrypt"
    message = "Thanks for taking a look at my polyalphabetic cipher!"
    
    # Remove punctuation and whitespace from the shift word
    shift_word = shift_word.translate(str.maketrans('', '', string.punctuation + string.whitespace))
    
    # Calculate shift values based on the letters of the shift word
    shift = [(string.ascii_uppercase.find(char) + 1) for char in shift_word.upper()]
    
    # Remove punctuation from the message
    message = message.translate(str.maketrans('', '', string.punctuation))
    
    print("Shifting the input by this list:", shift)
    print("Message to be encrypted:\n", message, sep="")
    
    # Encrypt the message
    encryption = encrypt(message.upper(), shift)
    print("\nYour encrypted message is:\n", encryption, sep="")
    
    # Decrypt the message
    decryption = decrypt(encryption, shift)
    print("Your decrypted message is:\n", decryption, sep="")

# Run the main function
Main()
