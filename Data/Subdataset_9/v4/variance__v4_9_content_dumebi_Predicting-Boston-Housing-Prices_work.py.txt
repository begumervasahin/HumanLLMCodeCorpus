import sys
import numpy as np
import pandas as pd
from sklearn.tree import DecisionTreeRegressor
from sklearn.model_selection import train_test_split

# Initialize a DecisionTreeRegressor with a fixed random state for reproducibility
regressor = DecisionTreeRegressor(random_state=0)

# Function to load the data from a CSV file
def load_data():
    # Read data from CSV file named 'AUDJPY_2.csv'
    work_data = pd.read_csv('AUDJPY_2.csv')
    # Print the shape of the dataset
    print("The work dataset has {} data points with {} variables each.".format(*work_data.shape))
    return work_data

# Function to extract features from the data
def get_features(data):
    # Drop unnecessary columns from the dataset
    features = data.drop(['Date', 'Time', 'Close', 'High', 'Low', 'Total Ticks'], axis=1)
    return features

# Function to extract the 'Close' column from the data
def get_close(data):
    # Drop other unnecessary columns and retain only 'Close' column
    close = data.drop(['Date', 'Time', 'Open', 'High', 'Low', 'Total Ticks'], axis=1)
    return close

# Function to make predictions based on the input 'open' value
def predict(open_value):
    # Load data
    data = load_data()
    # Extract features and close prices from the data
    features = get_features(data)
    close = get_close(data)
    
    # Split the data into training and testing sets
    X_train, X_test, y_train, y_test = train_test_split(features, close, test_size=0.4, shuffle=True, random_state=101)
    
    # Fit the regression model on the training data
    regressor.fit(X_train, y_train)
    
    # Create a numpy array for the input 'open' value
    test = np.array([[open_value]])
    
    # Make predictions
    pred = regressor.predict(test)
    
    # Determine if the predicted value is higher or lower than the input 'open' value
    if pred > open_value:
        return "Up"
    else:
        return "Down"

# Perform prediction for an input value of 'open'
pred = predict(85.226)
print("Prediction:", pred)
