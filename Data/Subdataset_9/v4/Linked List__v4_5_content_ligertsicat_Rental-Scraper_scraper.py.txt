import requests
from requests.exceptions import RequestException
from contextlib import closing
from bs4 import BeautifulSoup

# Open file for writing CSV output
output = open("rentpad_list.csv", "w+")

def scrape_rentpad(city):
    # Payload for POST request
    payload = {
        "a": "31",
        "cityName": city,
        "propertyTypeIDs": [2, 3, 5],
        "furnishTypeIDs": [1, 2, 3],
        "placeIDs": [],
        "statusTypeIDs": [],
        "amenityIDs": [],
        "longMonthRateLow": "0",
        "longMonthRateHigh": "30,000",
        "numBedroomsLow": "0",
        "numBedroomsHigh": "0",
        "itemsPerPage": "1000",
        "pageNumber": "1",
        "lengthOfStay": "",
        "ham": "ham"
    }
    
    try:
        # Send POST request to Rentpad
        r = requests.post(
            url="https://rentpad.com.ph/long-term-rentals/mandaluyong/apartment",
            data=payload,
        )
        
        # Parse HTML response
        html = BeautifulSoup(r.text, 'html.parser')
        
        # Extract apartment details
        prices = html.find_all(itemprop="price")
        apartments = html.find_all(itemprop="name")
        links = html.find_all(itemprop="offers")
        
        i = 0
        # Write apartment details to CSV file
        for apartment in apartments:
            if apartment.string is None:
                continue
            price = prices[i].string[1:].replace(",", "")
            link = links[i].find('a').get('href')
            output.write(apartment.string + "," + price + "," + link + "\n")
            i += 1
    except RequestException as e:
        print("Error during request:", e)

# Call function to scrape Rentpad for Mandaluyong city
scrape_rentpad("Mandaluyong")

# Close output file
output.close()
