import re
import requests
from bs4 import BeautifulSoup
from threading import Thread
from proxy_list import get_proxies
from parse_and_write import parse_and_write
from clean_db import db_cleaner

# Target URL
url = 'https://coinmarketcap.com/'

# Fetching the page content
page = requests.get(url)

# Parsing the page content using BeautifulSoup
soup = BeautifulSoup(page.content, "html.parser")

# List to store coin names
coins = []

# Iterating through all 'a' tags in the page
for link in soup.find_all('a'):
    if (link.get('href') is not None and link.get('href') != '/' and link.get('href') != ''):
        # Checking if the link matches the pattern '/currencies'
        if (re.match('^/currencies' , link.get('href'))):
            # Extracting coin name from the link
            coin = re.search('/currencies/(.+?)/', link.get('href')).group(1)
            # Adding the coin to the list if it's not already present
            if(coin not in coins and coin != 'volume'):
                coins.append(coin)

# Getting proxies
proxies = get_proxies()

# List to store threads
threads = []

# Creating threads to parse and write data for coins in batches
t1 = Thread(target=parse_and_write, args=(coins[0:25], proxies))
t2 = Thread(target=parse_and_write, args=(coins[25:50], proxies))
t3 = Thread(target=parse_and_write, args=(coins[50:75], proxies))
t4 = Thread(target=parse_and_write, args=(coins[75:100], proxies))

# Starting threads
t1.start()
t2.start()
t3.start()
t4.start()

# Waiting for threads to complete
t1.join()
t2.join()
t3.join()
t4.join()

# Appending threads to the list
threads.append(t1)
threads.append(t2)
threads.append(t3)
threads.append(t4)

# Waiting for all threads to complete
for t in threads:
    t.join()

# Printing message indicating the main thread is exiting
print("Exiting Main Thread")
