# Importing necessary libraries
from bs4 import BeautifulSoup
import requests
import sqlite3

# Establishing connection with SQLite database
conn = sqlite3.connect('movie.db')
c = conn.cursor()

# Creating a table named 'movie' in the database
c.execute("""CREATE TABLE IF NOT EXISTS movie (
                Rank TEXT,
                MovieTitle TEXT,
                Total TEXT,
                Opening TEXT,
                Open TEXT
            )""")

# Looping through years from 2010 to 2019
for year in range(2010, 2020):
    # Generating URL for each year
    url = 'https://www.boxofficemojo.com/yearly/chart/?yr=' + str(year) + '&p=.htm'
    
    # Sending request to fetch the webpage
    source = requests.get(url)
    
    # Parsing the HTML content
    soup = BeautifulSoup(source.text, 'lxml')
    
    # Extracting the table containing movie data
    movie_table = soup.find_all('table')[6]
    
    # Extracting column headers from the table
    headers_row = movie_table.find_all('tr')[0]
    headers = headers_row.find_all('td')
    header_row_list = [headers[0].text, headers[1].text, headers[4].text, headers[5].text, headers[6].text]
    page_data_list = [header_row_list]
    
    # Looping through each movie row in the table
    for movie_row in range(2, len(movie_table.find_all('tr')) - 4):
        movie_data_row = movie_table.find_all('tr')[movie_row]
        movie_data_items = movie_data_row.find_all('td')
        
        # Extracting movie details from each row
        movie_details = [movie_data_items[0].text, movie_data_items[1].text, 
                         movie_data_items[4].text, movie_data_items[5].text, 
                         movie_data_items[6].text]
        
        # Appending movie details to the list of data for the page
        page_data_list.append(movie_details)
        
        # Inserting movie details into the SQLite database
        c.execute("INSERT INTO movie VALUES (?,?,?,?,?)", (movie_data_items[0].text, movie_data_items[1].text,
