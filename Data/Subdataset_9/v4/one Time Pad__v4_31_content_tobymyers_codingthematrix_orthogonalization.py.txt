from vec import Vec

def project_along(b, v, eps=1E-20):
    '''
    Project vector b along vector v.
    
    Inputs:
        - b: a Vec representing the vector to be projected
        - v: a Vec representing the direction of projection
        - eps (default: 1E-20): threshold below which squared norms are considered zero
        
    Output:
        - a Vec representing the projection of b onto v
    '''
    sigma = ((b * v) / (v * v)) if v * v > eps else 0
    return sigma * v

def project_orthogonal(b, vlist):
    '''
    Project vector b orthogonal to a list of vectors vlist.
    
    Inputs:
        - b: a Vec representing the vector to be projected
        - vlist: a list of Vecs representing the vectors forming the basis for projection
        
    Output:
        - a Vec representing the projection of b orthogonal to the vectors in vlist
    '''
    for v in vlist:
        b = b - project_along(b, v)
    return b

def aug_project_orthogonal(b, vlist, eps=1E-20):
    '''
    Augmented version of projecting vector b orthogonal to a list of vectors vlist.
    
    Inputs:
        - b: a Vec representing the vector to be projected
        - vlist: a list of Vecs representing the vectors forming the basis for projection
        - eps (default: 1E-20): threshold below which squared norms are considered zero
        
    Output:
        - a tuple (result, alphadict) where result is a Vec representing the projection of b orthogonal to vlist,
          and alphadict is a dictionary containing the coefficients used in the projection
    '''
    alphadict = {len(vlist): 1}
    for i, v in enumerate(vlist):
        sigma = (b * v) / (v * v) if v * v > eps else 0
        alphadict[i] = sigma
        b = b - sigma * v
    return (b, alphadict)

def orthogonalize(vlist):
    '''
    Orthogonalize a list of vectors preserving order.
    The ith vector in the output list is the projection of vlist[i] orthogonal to
    the space spanned by all the previous vectors in the output list.
    
    Input:
        - vlist: a list of Vecs representing the vectors to be orthogonalized
        
    Output:
        - a list of mutually orthogonal Vecs spanning the same space as the input Vecs
    '''
    assert isinstance(vlist, list)
    vstarlist = []
    for v in vlist:
        vstarlist.append(project_orthogonal(v, vstarlist))
    return vstarlist

def aug_orthogonalize(vlist):
    '''
    Augmented version of orthogonalizing a list of vectors.
    
    Input:
        - vlist: a list of Vecs representing the vectors to be orthogonalized
        
    Output:
        - a tuple (vstarlist, sigma_vecs) where vstarlist is a list of orthonormal Vecs spanning the same space as the input Vecs,
          and sigma_vecs is a list of Vecs containing the coefficients used in the projection for each vector
    '''
    assert isinstance(vlist, list)
    vstarlist = []
    sigma_vecs = []
    D = set(range(len(vlist)))
    for v in vlist:
        (vstar, sigmadict) = aug_project_orthogonal(v, vstarlist)
        vstarlist.append(vstar)
        sigma_vecs.append(Vec(D, sigmadict))
    return vstarlist, sigma_vecs
