import requests
from bs4 import BeautifulSoup
from selenium import webdriver

# Function to start web scraping
def start():
    # Initializing a Chrome webdriver
    browser = webdriver.Chrome('chromedriver.exe')
    
    # URL of the website to scrape
    url = 'http://www.footballsquads.co.uk/ger/2017-2018/'
    
    # Specific tag for German Bundesliga
    tag = 'gerbun.htm'
    
    # Combining URL and tag
    browser.get(url + tag)
    
    # Getting the HTML content of the page
    html = browser.page_source
    
    # Parsing the HTML using BeautifulSoup
    soup = BeautifulSoup(html, "html.parser")
    
    # Finding the main div tag
    divTag = soup.find_all("div", {"id": "main"})
    
    # Finding all the h5 tags within the main div
    h5Tag = divTag[0].find_all('h5')
    
    # List to store player details
    p_list = []
    
    # Iterating through each team
    for i in h5Tag:
        # Extracting team name
        team = i.text
        
        # Visiting the team's page
        browser.get(url + i.find("a").get("href"))
        
        # Getting the HTML content of the team's page
        html = browser.page_source
        
        # Parsing the HTML using BeautifulSoup
        soup = BeautifulSoup(html, "html.parser")
        
        # Finding the main div tag of the team's page
        divTag = soup.find_all("div", {"id": "main"})
        
        # Finding all the tables within the main div
        tableTag = divTag[0].find_all('table')
        
        # Finding all the table rows within the first table
        trTag = tableTag[0].find('tbody').find_all('tr')
        
        # Iterating through each row
        for i in range(1, len(trTag)):
            temp = []
            # Finding all the table cells within the row
            tdTag = trTag[i].find_all('td')
            try:
                # Checking if the second cell contains text (to avoid empty rows)
                if tdTag[1].text:
                    # Appending player details to temp list
                    temp.append(tdTag[1].text)
                    temp.append(tdTag[2].text)
                    temp.append(tdTag[3].text)
                    temp.append(team)
                    # Appending temp list to player list
                    p_list.append(temp)
            except:
                break  # Exiting loop if an exception occurs (e.g., empty row)
    
    # Returning the list of player details
    return p_list
