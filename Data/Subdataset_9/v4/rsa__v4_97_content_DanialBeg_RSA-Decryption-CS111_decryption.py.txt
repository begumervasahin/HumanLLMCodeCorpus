import string

# Function to calculate the inverse modulo
def invModulo(a, b):
    for num in range(1, b):
        if ((a * num) % b) == 1:
            return num
    return 1

# Function to find factors of a number
def factors(r):
    listFactors = []
    i = 1
    while i <= r:
        if r % i == 0:
            listFactors.append(i)
        i += 1
    return listFactors

# Main function
def main():
    # Input public key values
    e = int(input("Please enter the exponent value in the public key: "))
    n = int(input("Please enter the n value in the public key: "))
    print("Public key values: ", e, n)

    # Calculate factors of n
    factorsList = factors(n)
    p = factorsList[1]
    q = factorsList[2]

    # Calculate the totient value
    nval = (p + 1) * (q + 1)

    # Calculate the decryption exponent
    decrypt_e = invModulo(e, nval)

    # Input array containing encrypted message
    inputArray = [51, 12, 51, 39, 31, 21, 14, 10, 20, 17, 7, 25, 14, 26, 33, 52, 15, 7,
                  27, 51, 7, 49, 8, 15, 51, 7, 8, 25, 7, 25, 10, 49, 18, 52, 51, 7, 8,
                  25, 7, 18, 26, 25, 25, 10, 27, 52, 51, 7, 27, 33, 21, 7, 20, 26, 21,
                  7, 25, 10, 49, 18, 52, 51, 39]

    # List of alphabets including space
    alphabet = list(string.ascii_lowercase)
    alphabet.append(' ')

    # Decrypt and print the message
    for i in inputArray:
        charval = (i ** decrypt_e) % n
        print(str(alphabet[charval - 2]), end="")

# Entry point of the program
if __name__ == "__main__":
    main()
