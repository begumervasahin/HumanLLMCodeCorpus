def quicksort(arr):
    """
    Sorts the given list using the quicksort algorithm.

    Args:
        arr (list): The list to be sorted.
    """
    quicksort_helper(arr, 0, len(arr) - 1)

def quicksort_helper(arr, first, last):
    """
    Helper function for quicksort.

    Args:
        arr (list): The list to be sorted.
        first (int): The index of the first element in the sublist.
        last (int): The index of the last element in the sublist.
    """
    if first < last:
        split_point = partition(arr, first, last)
        quicksort_helper(arr, first, split_point - 1)
        quicksort_helper(arr, split_point + 1, last)

def partition(arr, first, last):
    """
    Partitions the list for quicksort.

    Args:
        arr (list): The list to be partitioned.
        first (int): The index of the first element in the sublist.
        last (int): The index of the last element in the sublist.

    Returns:
        int: The split point.
    """
    pivot_value = arr[first]
    left_mark = first + 1
    right_mark = last
    done = False
    while not done:
        while left_mark <= right_mark and arr[left_mark] <= pivot_value:
            left_mark += 1
        while arr[right_mark] >= pivot_value and right_mark >= left_mark:
            right_mark -= 1
        if right_mark < left_mark:
            done = True
        else:
            arr[left_mark], arr[right_mark] = arr[right_mark], arr[left_mark]
    arr[first], arr[right_mark] = arr[right_mark], arr[first]
    return right_mark
