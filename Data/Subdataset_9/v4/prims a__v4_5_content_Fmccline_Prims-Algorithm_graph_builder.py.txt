import json
import networkx as nx

class GraphBuilder:
    """ 
    Helper class for creating and working with networkx graphs
    """
    
    @staticmethod
    def build_graph_from_file(filename):
        """ 
        Creates a graph from a given JSON file
        
        :param filename: The JSON source file of the graph
        :return: A networkx graph created from the given file
        """
        graph = nx.Graph()  # Create an empty graph object
        
        # Open the JSON file and load its content
        with open(filename, 'r') as file:
            raw_data = json.load(file)
        
        # Extract nodes, neighbors, and weights from the raw data
        nodes = raw_data["nodes"]
        
        # Iterate through each node in the data
        for node in nodes:
            name = node["name"]  # Get the name of the node
            neighbors = node["neighbors"]  # Get the list of neighbors
            weights = node["weights"]  # Get the corresponding weights
            
            # Iterate through each neighbor of the node
            for edge in range(0, len(weights)):
                neighbor = neighbors[edge]  # Get the name of the neighbor
                weight = weights[edge]  # Get the weight of the edge
                
                # Add the node to the graph if it's not already present
                graph.add_node(name)
                
                # Add the edge between the current node and its neighbor with the specified weight
                graph.add_edge(name, neighbor, weight=weight)
        
        return graph  # Return the constructed graph
