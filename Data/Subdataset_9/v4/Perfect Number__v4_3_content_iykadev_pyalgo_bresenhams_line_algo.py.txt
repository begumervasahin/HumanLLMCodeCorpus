def draw_line(self, x0, y0, x1, y1):
    """
    Draws a line using Bresenham's line algorithm.
    
    Parameters:
    - x0, y0: The starting coordinates of the line.
    - x1, y1: The ending coordinates of the line.
    """
    dx = abs(x1 - x0)
    dy = abs(y1 - y0)
    x, y = x0, y0
    sx = -1 if x0 > x1 else 1
    sy = -1 if y0 > y1 else 1
    
    if dx > dy:
        err = dx / 2.0
        while x != x1:
            self.set_pixel(x, y)
            err -= dy
            if err < 0:
                y += sy
                err += dx
            x += sx
    else:
        err = dy / 2.0
        while y != y1:
            self.set_pixel(x, y)
            err -= dx
            if err < 0:
                x += sx
                err += dy
            y += sy
    self.set_pixel(x, y)

# Attach the draw_line method to the Bitmap class
Bitmap.draw_line = draw_line

# Example usage
bitmap = Bitmap(17, 17)
# Draw lines between specified points
for points in ((1, 8, 8, 16), (8, 16, 16, 8), (16, 8, 8, 1), (8, 1, 1, 8)):
    bitmap.draw_line(*points)

# Display the bitmap
bitmap.display()
