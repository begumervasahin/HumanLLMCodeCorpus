import os
import re
import time
from hash import HashTable

def read_directory(directory):
    paths = [os.path.join(directory, name) for name in os.listdir(directory)]
    files = [file for file in paths if os.path.isfile(file)]
    return files

def generate_index(file_path, index, num):
    with open(file_path, 'r') as file:
        words = file.read().lower()
    
    words = re.sub(r'[,.!?\r\t\n]', '', words)
    words = words.split(" ")
    
    for word in words:
        count = words.count(word)
        flag = True
        
        if index.get_value(word):
            for entry in index.get_value(word):
                if entry[0] == count and entry[1] == num + 1:
                    flag = False
            
            if flag:
                index.insert_value(word, [count, num + 1])
        else:
            index.insert_value(word, [count, num + 1])

hashing_multiplication_linear = HashTable(1000000, 3, 'multiplication')
hashing_multiplication_quadratic = HashTable(1000000, 3, 'multiplication', 'quadratic')
hashing_division_linear = HashTable(1000000, 3)
hashing_division_quadratic = HashTable(1000000, 3, 'division', 'linear')

start_time = time.time()
files = read_directory('base')

for i in range(len(files)):
    generate_index(files[i], hashing_multiplication_linear, i)

for key in sorted(hashing_multiplication_linear.get_keys()):
    print(key, hashing_multiplication_linear.get_value(key)[0][0], files[(hashing_multiplication_linear.get_value(key)[0][1]) - 1])

end_time = time.time()
print('Time taken for hashing using multiplication method and linear collision:', end_time - start_time)
