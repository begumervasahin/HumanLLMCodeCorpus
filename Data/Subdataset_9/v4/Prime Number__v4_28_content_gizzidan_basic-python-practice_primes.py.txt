# Get user input for the bounds
bound_1 = int(input("Enter the first number: "))
bound_2 = int(input("Enter the second number: "))

# Ensure bounds are in ascending order
bounds = [bound_1, bound_2]
bounds.sort()

# Initialize a list to store prime numbers
primes = []

# Find prime numbers between the bounds
for x in range(bounds[0] + 1, bounds[1]):
    if all(x % i != 0 for i in range(2, x)):
        primes.append(x)

# Modify prime numbers for formatting
if len(primes) > 1:
    for i in range(0, len(primes) - 1, 3):
        primes[i] = str(primes[i]) + ":"
    for i in range(1, len(primes) - 1, 3):
        primes[i] = str(primes[i]) + "!"
    for i in range(2, len(primes) - 1, 3):
        primes[i] = str(primes[i]) + ","
    print("".join(map(str, primes)))
elif len(primes) == 1:
    print("".join(map(str, primes)))
else:
    print("No prime numbers found")
