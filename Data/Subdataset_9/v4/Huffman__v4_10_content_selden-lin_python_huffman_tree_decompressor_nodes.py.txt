class HuffmanNode:
    """
    A node in a Huffman tree.
    Symbols occur only at leaves.
    
    Attributes:
    - symbol (int): symbol located in this node, if any
    - left (HuffmanNode): left subtree
    - right (HuffmanNode): right subtree
    - number (int): node number
    """
    def __init__(self, symbol=None, left=None, right=None):
        """
        Create a new HuffmanNode with the given parameters.
        
        Args:
        - symbol (int|Node): symbol to be stored in this node, if any
        - left (HuffmanNode|Node): a tree rooted at 'left', if any
        - right (HuffmanNode|Node): a tree rooted at 'right', if any
        """
        self.symbol = symbol
        self.left = left
        self.right = right
        self.number = None

    def __eq__(self, other):
        """
        Return True iff self is equivalent to other.
        """
        return (type(self) == type(other) and self.symbol == other.symbol and
                self.left == other.left and self.right == other.right)

    def __lt__(self, other):
        """
        Return True iff self is less than other.
        """
        return False

    def __repr__(self):
        """
        Return constructor-style string representation.
        """
        return 'HuffmanNode({}, {}, {})'.format(self.symbol,
                                                self.left, self.right)

    def is_leaf(self):
        """
        Return True iff self is a leaf.
        """
        return not self.left and not self.right


class ReadNode:
    """
    A node as read from a compressed file.
    Each node consists of type and data information as described in the handout.
    
    Attributes:
    - l_type (int): 0/1 (if the corresponding HuffmanNode's left is a leaf)
    - l_data (int): a symbol or the node number of a HuffmanNode's left
    - r_type (int): 0/1 (if the corresponding HuffmanNode's right is a leaf)
    - r_data (int): a symbol or the node number of a HuffmanNode's right
    """
    def __init__(self, l_type, l_data, r_type, r_data):
        """
        Create a new ReadNode with the given parameters.
        
        Args:
        - l_type (int): used to initialize self.l_type
        - l_data (int): used to initialize self.l_data
        - r_type (int): used to initialize self.r_type
        - r_data (int): used to initialize self.r_data
        """
        self.l_type = l_type
        self.l_data = l_data
        self.r_type = r_type
        self.r_data = r_data

    def __repr__(self):
        """
        Return constructor-style string representation.
        """
        return 'ReadNode({}, {}, {}, {})'.format(
            self.l_type, self.l_data, self.r_type, self.r_data)
