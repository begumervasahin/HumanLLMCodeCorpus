import requests
import os
import re
from bs4 import BeautifulSoup

class PaperSpider:
    def __init__(self, name, url="https://dlnext.acm.org/toc/tompecs/2016/1/1", filename="ACM_papers_list.txt"):
        self.name = name
        self.url = url
        self.filename = filename
        self.papers_order = 1

    def get_html(self, url):
        try:
            headers = {"User-Agent": "Safari/12.1.2"}
            response = requests.get(url, headers=headers)
            response.raise_for_status()
            response.encoding = response.apparent_encoding
            html = os.linesep.join([s for s in response.text.splitlines() if s])
            return html
        except Exception as e:
            print("Error fetching HTML:", e)
            return ""

    def parse_page(self, html):
        try:
            current_issue_papers = []
            soup = BeautifulSoup(html, 'html.parser')
            title_boxes = soup.find_all('h5', class_='issue-item__title')
            for title_box in title_boxes:
                title = title_box.text.strip().replace('\n', ' ').strip()
                if title == "List of Reviewers":
                    continue
                doi_num = re.search(r'/doi/abs(.*?)">', str(title_box)).group(1)
                doi_link = "https://doi.org" + doi_num
                pair = [doi_link, title]
                current_issue_papers.append(pair)
            print(current_issue_papers)
            self.store_paper_info(current_issue_papers)
            next_box = soup.find('a', class_='content-navigation__btn--next')
            if next_box and next_box['href'] != "javascript:void(0)":
                next_link = "https://dlnext.acm.org" + next_box['href']
                next_html = self.get_html(next_link)
                self.parse_page(next_html)
        except Exception as e:
            print("Error parsing page:", e)

    def store_paper_info(self, current_issue_papers):
        try:
            with open(self.filename, "a") as f:
                for paper_order, pair in enumerate(current_issue_papers, start=self.papers_order):
                    paper_title, doi_link = pair
                    f.write(f"paper{paper_order}\nTitle: {paper_title}\nDoi_link: {doi_link}\n")
                    f.write("==========================\n")
                self.papers_order += len(current_issue_papers)
        except Exception as e:
            print("Error storing paper info:", e)

if __name__ == '__main__':
    run = True
    while run:
        print("Welcome to use the ACM Transactions journal papers Spider!")
        start_url = input("Enter a start url or quit: ")
        if start_url == 'quit':
            print("Goodbye, human!")
            break
        file_name = input("Enter a filename (e.g., ACM_papers_list.txt) or quit: ")
        if file_name == 'quit':
            print("Goodbye, human!")
            break
        acm_perf_model = PaperSpider("acmPerfModel", start_url, file_name)
        html_result = acm_perf_model.get_html(start_url)
        acm_perf_model.parse_page(html_result)
