# Import necessary libraries
from skimage import io
import math
import numpy as np
import os

# Define a function to compute cluster center and radius
def cluster_center_radius(path):
    # Get list of file names in the given path
    name_list = os.listdir(path)
    
    # Initialize lists to store mean and variance values
    mean_list = []
    variance_list = []
    
    # Initialize variables for cluster center coordinates and sum of Euclidean distances
    cluster_center_x = 0
    cluster_center_y = 0
    eu_dist_sum = 0
    
    # Iterate through each image file in the directory
    for i in range(len(name_list)):
        # Construct full path to the image file
        img_path = os.path.join(path, name_list[i])
        
        # Read the image as grayscale using skimage
        img = io.imread(img_path, as_grey=True)
        
        # Get dimensions of the image
        row = len(img)
        column = len(img[0])
        
        # Initialize variables to calculate mean and variance
        summary = 0
        diff_sum = 0
        
        # Calculate sum of pixel values for mean calculation
        for c in range(row):
            for n in range(column):
                summary += img[c][n]
        
        # Compute mean pixel value
        mean = summary / (row * column)
        mean_list.append(mean)
        
        # Calculate sum of squared differences for variance calculation
        for a in range(row):
            for b in range(column):
                diff_sum += (img[a][b] - mean) ** 2
        
        # Compute standard deviation (square root of variance)
        variance = math.sqrt(diff_sum / (row * column))
        variance_list.append(variance)
    
    # Calculate cluster center coordinates
    for k in range(len(name_list)):
        cluster_center_x += mean_list[k]
        cluster_center_y += variance_list[k]
    x_mean = cluster_center_x / len(name_list)
    y_mean = cluster_center_y / len(name_list)
    cluster_center = np.array([x_mean, y_mean])
    
    # Calculate average Euclidean distance from cluster center
    for p in range(len(name_list)):
        eu_dist = math.sqrt((mean_list[p] - x_mean) ** 2 + (variance_list[p] - y_mean) ** 2)
        eu_dist_sum += eu_dist
    cluster_r = eu_dist_sum / len(name_list)
    
    # Return cluster center coordinates and radius
    return cluster_center, cluster_r
