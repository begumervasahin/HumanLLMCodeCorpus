def recursive_fibonacci(num):
    '''Calculate the Fibonacci value recursively.'''
    if num <= 1:
        return num
    return recursive_fibonacci(num - 2) + recursive_fibonacci(num - 1)

def run_fibonacci_program():
    '''Run the Fibonacci program.'''
    while True:
        try:
            num_terms = int(input('How many terms of the Fibonacci sequence would you like to generate?: '))
        except ValueError:
            print('\nPlease enter an integer.')
        else:
            if num_terms <= 0:
                print("\nPlease enter a positive integer.")
            else:
                break

    print("\nGenerated Fibonacci sequence:\n")
    for num in range(num_terms + 1):
        print(recursive_fibonacci(num))

if __name__ == "__main__":
    run_fibonacci_program()
