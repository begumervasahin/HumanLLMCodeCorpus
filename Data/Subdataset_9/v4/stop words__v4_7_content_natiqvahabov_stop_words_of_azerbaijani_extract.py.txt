import nltk
import urllib.request as urllib2
from nltk.tokenize import RegexpTokenizer
from bs4 import BeautifulSoup
from collections import Counter

# URL of the Wikipedia page containing selected articles
wiki_page_url = "https://az.wikipedia.org/wiki/Vikipediya:Se%C3%A7ilmi%C5%9F_m%C9%99qal%C9%99l%C9%99r"

# Open the Wikipedia page
page = urllib2.urlopen(wiki_page_url)
soup = BeautifulSoup(page, 'html.parser')

# Find the section containing the selected articles
articles_section = soup.find('div', attrs={'class': 'mw-parser-output'}).find('tr').find_next('tr')

# Extract URLs and titles of the selected articles
wiki_list_urls = []
wiki_list_titles = []

for td in articles_section.find_all('td'):
    for link in td.find_all('a'):
        if link.get('title'):
            wiki_list_urls.append('https://az.wikipedia.org' + link['href'])
            wiki_list_titles.append(link.get('title'))

# Test list of Wikipedia articles (for testing purposes)
wiki_list_test = ['https://az.wikipedia.org/wiki/GRB_970508', 'https://az.wikipedia.org/wiki/G%C3%BCn%C9%99%C5%9F_sistemi']

# Tokenize and process the content of each Wikipedia article
tokenized_final = []

for idx, wiki_url in enumerate(wiki_list_urls, start=1):
    print("Processing article", idx)
    page_single = urllib2.urlopen(wiki_url)
    soup_single = BeautifulSoup(page_single, 'html.parser')
    content = soup_single.find('div', attrs={'class': 'mw-parser-output'}).get_text()

    # Tokenize the content using regular expressions
    tokenizer = RegexpTokenizer(r'\w+')
    tokenized_text = tokenizer.tokenize(content)
    
    # Add tokens to the final token list
    tokenized_final += tokenized_text

# Count the occurrences of each token
token_counter = Counter(tokenized_final)

# Write the token counts to a file
with open('stopwordsAZE.txt', 'a') as file:
    file.write(str(token_counter))
