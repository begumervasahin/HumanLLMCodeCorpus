class Functions:
    # Function to calculate the median of a given dataset
    def Median(self, data):
        # Calculate the length of the data
        ln = len(data)
        # Sort the data in ascending order
        data = sorted(data)
        # Check if the length of data is even
        if ln % 2 == 0:
            # If even, calculate the median by taking the average of two middle values
            return (data[ln // 2] + data[(ln // 2) + 1]) / 2
        else:
            # If odd, return the middle value as median
            return data[(ln // 2) + 1]
        # Note: The following line won't be reached as it's after the return statement
        print('Hello')

    # Function to calculate the mode of a given dataset
    def Mode(self, data):
        # Create an empty dictionary to store frequency counts
        dic = {}
        # Count occurrences of each unique element in the data
        for i in set(data):
            dic[i] = data.count(i)
        # Find the maximum frequency
        x = dic[max(dic, key=dic.get)]
        # If all elements occur only once, return a list containing 0
        if x <= 1:
            return [0]
        else:
            # Otherwise, return a list of modes (elements with maximum frequency)
            return [key for key, val in dic.items() if val == x]

    # Function to calculate the mean of a given dataset
    def Mean(self, data):
        # Calculate the mean by summing up all elements and dividing by the total count
        return sum(data) / len(data)

    # Function to calculate the variance of a given dataset
    def Variance(self, data):
        # Calculate the mean of the dataset
        mean = self.Mean(data)
        # Calculate the variance using the formula
        return sum([(i - mean) ** 2 for i in data]) / len(data)

    # Function to calculate the standard deviation of a given dataset
    def StaDev(self, data):
        # Calculate the standard deviation using the square root of the variance
        return (self.Variance(data)) ** 0.5

# Main function to execute the program
def main():
    # Create an instance of the Functions class
    func = Functions()
    # Read data from a file named "data.txt" into a list
    data = [float(i) for i in open("data.txt", 'r')]
    # Print the results
    print('Median is %.2f' % (func.Mean(data)))
    print('Mean is %.2f' % (func.Median(data)))
    print('Mode is ' + (', '.join(map(str, [i for i in func.Mode(data)]))))
    print('Variance is %.2f' % (func.Variance(data)))
    print('Standard Deviation is %.2f' % (func.StaDev(data)))

# Check if the script is being run directly
if __name__ == "__main__":
    # Call the main function
    main()
