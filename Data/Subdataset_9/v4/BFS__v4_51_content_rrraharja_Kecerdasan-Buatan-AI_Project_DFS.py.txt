# Define the map graph
maps = {
    'ALASKA': {'CALIFORNIA'},
    'CALIFORNIA': {'CAROLINA', 'ALASKA'},
    'CAROLINA': {'CALIFORNIA', 'INDIANA', 'HAWAII', 'DAREWALE'},
    'DAREWALE': {'CAROLINA', 'EDINBURGH', 'HAWAII', 'FLORIDA'},
    'EDINBURGH': {'DAREWALE'},
    'FLORIDA': {'DAREWALE', 'GEORGIA'},
    'GEORGIA': {'FLORIDA', 'HAWAII'},
    'HAWAII': {'DARAWALE', 'CAROLINA', 'GEORGIA', 'LOUSIANA'},
    'INDIANA': {'CAROLINA', 'IDAKO', 'KENTUCKY'},
    'IDAKO': {'INDIANA'},
    'KENTUCKY': {'LOUSIANA', 'INDIANA'},
    'LOUSIANA': {'KENTUCKY', 'HAWAII'}
}


def dfs(graph, start, finish):
    stack = [[start]]  # Initialize stack with start node as a path
    visited = set()  # Initialize set to keep track of visited nodes

    while stack:
        # Get the last path in the stack
        path = stack.pop()

        # Get the last node in the path
        state = path[-1]

        # Check if the last node is the finish node
        if state == finish:
            return path  # Return the path if it reaches the finish node

        # Check if the last node has been visited
        elif state not in visited:
            # Explore all branches from the current node
            for branch in graph.get(state, []):
                # Create a new path by appending the branch to the current path
                new_path = list(path)
                new_path.append(branch)
                # Add the new path to the stack
                stack.append(new_path)

            # Mark the current node as visited
            visited.add(state)

    # If the stack is empty and finish node is not found, print "NOT FOUND 404"
    print("NOT FOUND 404")


# Test the DFS function
print(dfs(maps, 'ALASKA', 'INDIANA'))
