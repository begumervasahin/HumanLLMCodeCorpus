def print_string_properties(input_string):
    """Print properties of the input string."""
    char_list = list(input_string)  # Convert the string to a list of characters
    even_char_list = [char for index, char in enumerate(char_list) if index % 2 == 1]  # Select characters with even indexes
    even_string = "".join(even_char_list)  # Join the even-indexed characters to form a string

    # Print the input string, characters with even indexes, and the string in reverse order
    print("The input string is:", input_string)
    print("The characters that have even indexes are:", even_string)
    print("The string in reverse order is:", input_string[::-1])

    # Check if the string is a palindrome and print the result
    if input_string == input_string[::-1]:
        print("The string is a palindrome: True")
    else:
        print("The string is a palindrome: False")

def print_set(lower_limit, upper_limit):
    """Print a set of numbers divisible by 7 but not by 5 within the given range."""
    num_list = []
    for i in range(lower_limit + 1, upper_limit):  # Generate numbers between lower_limit and upper_limit-1
        if i % 7 == 0 and i % 5 != 0:  # Check if the number is divisible by 7 but not by 5
            num_list.append(i)  # Add the number to the list

    result_set = set(num_list)  # Convert the list to a set
    print(result_set)  # Print the resulting set

# Test the functions
print_string_properties("11411")
print_set(11, 21)
