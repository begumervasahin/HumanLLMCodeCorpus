import itertools

def floyd_warshall(graph, num_vertices):
    """
    Applies the Floyd-Warshall algorithm to find the shortest paths between all pairs of vertices in a graph.

    Args:
        graph (list of lists): The adjacency matrix representing the graph.
        num_vertices (int): The number of vertices in the graph.

    Returns:
        None. Updates the input graph with shortest path distances.
    """
    for k in range(num_vertices):
        for i, j in itertools.product(range(num_vertices), range(num_vertices)):
            if graph[i][j] > graph[i][k] + graph[k][j]:
                graph[i][j] = graph[i][k] + graph[k][j]

# Initialize the graph with maximum distances
graph = [[1000 for _ in range(4)] for _ in range(4)]

# Set diagonal elements to zero
for i in range(4):
    graph[i][i] = 0

# Assign specific distances between vertices
graph[0][2] = -2
graph[1][0] = 4
graph[1][2] = 3
graph[2][3] = 2
graph[3][1] = -1

# Print the original graph
print("Original Graph:")
for row in graph:
    print(row)

# Apply Floyd-Warshall algorithm
print('\nAfter Floyd-Warshall:')
floyd_warshall(graph, 4)

# Print the graph after applying Floyd-Warshall
for row in graph:
    print(row)
