def list_to_string(char_list):
    """Converts a list of characters to a string"""
    return ''.join(char_list)

def print_numbers_up_to(n):
    """Prints all numbers from 1 to n in increasing order"""
    if n >= 1:
        print_numbers_up_to(n - 1)
        print(str(n))
    else:
        return

def print_numbers_down_to(n):
    """Prints all numbers from n to 1 in decreasing order"""
    if n >= 1:
        print(str(n))
        print_numbers_down_to(n - 1)

def has_divisor_smaller_than(n, i):
    """Checks if a number has a divisor smaller than a given number"""
    if i == 1 or i == 0:
        return False
    elif n % i == 0:
        return True
    return has_divisor_smaller_than(n, i - 1)

def is_prime(n):
    """Checks if a number is prime"""
    if n < 2:
        return False
    i = int(n ** 0.5)
    if not has_divisor_smaller_than(n, i):
        return True
    return False

def factorial(n):
    """Calculates the factorial of a number"""
    if n == 1:
        return 1
    return n * factorial(n - 1)

def exponential_limit(n, x):
    """Calculates the limit of (1 + x/n)^n as n approaches infinity"""
    if n == 0:
        return 1
    return (x ** n) / factorial(n) + exponential_limit(n - 1, x)

def solve_hanoi_towers(hanoi, n, src, dest, temp):
    """Solves the Tower of Hanoi puzzle with graphical representation"""
    if n < 1:
        return
    elif n == 1:
        hanoi.move(src, dest)
        return
    solve_hanoi_towers(hanoi, n - 1, src, temp, dest)
    hanoi.move(src, dest)
    solve_hanoi_towers(hanoi, n - 1, temp, dest, src)

def print_all_sequences(char_list, n):
    """Prints all possible words of length n using characters from a given list"""
    if n == 0:
        return
    k = n
    n = len(set(char_list))
    print_sequences_recursive(char_list, "", n, k)

def print_sequences_recursive(char_list, prefix, n, k):
    """Recursively prints all possible words with characters from a given list"""
    if k == 0:
        print(prefix)
        return
    for i in range(n):
        new_prefix = prefix + char_list[i]
        print_sequences_recursive(char_list, new_prefix, n, k - 1)

def print_unique_sequences(char_list, n):
    """Prints all possible strings of length n with characters from a list without repetitions"""
    if n == 0:
        return
    print_no_repetition_sequences_recursive(char_list, "", n)

def print_no_repetition_sequences_recursive(char_list, prefix, n):
    """Recursively prints all possible strings of length n with characters from a list without repetitions"""
    if n == 0:
        print(prefix)
        return
    for i, char in enumerate(char_list):
        new_prefix = prefix + char
        print_no_repetition_sequences_recursive(char_list[:i] + char_list[i+1:], new_prefix, n - 1)

def generate_parentheses(n):
    """Generates all properly opened and closed combinations of n pairs of parentheses"""
    str_lst = [""] * 2 * n
    results_lst = []
    if n > 0:
        generate_parentheses_recursive(str_lst, 0, n, 0, 0, results_lst)
    return results_lst

def generate_parentheses_recursive(str_lst, pos, n, open, close, results_lst):
    """Recursively generates all properly opened and closed combinations of n pairs of parentheses"""
    if close == n:
        result_str = "".join(str_lst)
        results_lst.append(result_str)
        return
    else:
        if open > close:
            str_lst[pos] = ')'
            generate_parentheses_recursive(str_lst, pos + 1, n, open, close + 1, results_lst)
        if open < n:
            str_lst[pos] = '('
            generate_parentheses_recursive(str_lst, pos + 1, n, open + 1, close, results_lst)

def generate_paths_up_right(n, k):
    """Generates all possible paths from (0,0) to (n,k) using only 'u' (up) and 'r' (right) movements"""
    if k == 0 and n == 0:
        return None
    if n < 0 or k < 0:
        return None
    generate_paths_up_right_recursive(n, k, "")

def generate_paths_up_right_recursive(n, k, prefix=''):
    """Recursively generates all possible paths from (0,0) to (n,k) using only 'u' (up) and 'r' (right) movements"""
    if n == 0 and k == 0:
        print(prefix)
        return
    elif n == 0:
        generate_paths_up_right_recursive(n, k - 1, prefix + 'u')
        return
    elif k == 0:
        generate_paths_up_right_recursive(n - 1, k, prefix + 'r')
        return
    generate_paths_up_right_recursive(n, k - 1, prefix + 'u')
    generate_paths_up_right_recursive(n - 1, k, prefix + 'r')

def fill_surrounding(image, start):
    """Flood fills areas in the image"""
    x, y = start
    flood_fill_recursive(image, x, y, '.', '*')

def flood_fill_recursive(image, x, y, empty_char, full_char):
    """Recursively performs flood fill starting at given point (x,y)"""
    image_width = len(image)
    image_height = len(image[0])
    if image[x][y] != empty_char:
        return
    image[x][y] = full_char
    flood_fill_recursive(image, x - 1, y, empty_char, full_char)
    flood_fill_recursive(image, x, y - 1, empty_char, full_char)
    flood_fill_recursive(image, x + 1, y, empty_char, full_char)
    flood_fill_recursive(image, x, y + 1, empty_char, full_char)
