import sys
import numpy as np

# Initialize a 2D array of zeros to store intermediate results
arr = np.zeros((2048, 2048), dtype=int)

# Function to calculate the length of the longest common subsequence of two strings A and B
def LCS(A, B):
    m = len(A)
    n = len(B)
    
    # Loop through each character of both strings
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if A[i - 1] == B[j - 1]:
                arr[i][j] = arr[i - 1][j - 1] + 1
            else:
                arr[i][j] = max(arr[i - 1][j], arr[i][j - 1])
    
    # Return the length of the longest common subsequence
    return arr[m][n]

# Main function to read input from standard input and compute LCS
def main():
    # Check if command line arguments are provided
    if len(sys.argv) != 1:
        sys.exit('Usage: `python LCS.py < input`')
    
    # Read input from standard input
    for line in sys.stdin:
        A, B = line.split()
        # Compute and print the length of the longest common subsequence of A and B
        print(LCS(A, B))

# Check if the script is being run as the main program
if __name__ == '__main__':
    main()
