import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
import sympy as sp
import argparse
import os

# Setting up command-line arguments
parser = argparse.ArgumentParser()
parser.add_argument('--input', help='Path to the input data file (csv) relative to the working directory.')
parser.add_argument('--poly_degree', default=2, type=int, help='Degree of the polynomial regression. Defaults to 1.')
args = parser.parse_args()

# Reading the dataset
dataset = pd.read_csv(args.input)
X = dataset.iloc[:, 0].values
y = dataset.iloc[:, 1].values

# Extracting dataset information
dataset_name = os.path.splitext(args.input)[0].replace('_', ' ').title()
dataset_columns = dataset.columns.tolist()

# Scatter plot of the dataset
plt.scatter(X, y, color='green')

# Function to plot polynomial regression
def plot_polynomial_regression():
    # Calculating polynomial regression weights
    weights = np.polyfit(X, y, args.poly_degree)
    model = np.poly1d(weights)
    # Plotting the regression line
    plt.plot(X, model(X), color='orange')
    # Initializing Sympy for symbolic expression
    sp.init_printing()
    # Printing the polynomial regression equation
    print(sp.latex(sp.Poly(model.coef, spx).as_expr()))

# Function to display the plot
def show_plot():
    plt.title(dataset_name)
    plt.xlabel(dataset_columns[0])
    plt.ylabel(dataset_columns[1])
    plt.show()

# Plotting polynomial regression and displaying the plot
plot_polynomial_regression()
show_plot()
