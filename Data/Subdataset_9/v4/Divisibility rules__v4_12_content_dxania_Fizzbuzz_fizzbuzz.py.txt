def get_input():
    # Get input for the first list from the user
    list1 = input("Enter the first list (separate elements with commas): ").split(",")
    print("List1:", list1)
    print("List1 contains", len(list1), "elements")

    # Get input for the second list from the user
    list2 = input("Enter the second list (separate elements with commas): ").split(",")
    print("List2:", list2)
    print("List2 contains", len(list2), "elements")

    # Calculate and print Fizzbuzz result
    print(f"Fizzbuzz result: {Fizzbuzz(list1, list2)}")

def Fizzbuzz(a, b):
    # Calculate the total length of both lists
    total_length = len(a) + len(b)

    # Check for Fizz, Buzz, or FizzBuzz conditions
    if total_length % 3 == 0 and total_length % 5 == 0:
        return 'FizzBuzz'
    elif total_length % 3 == 0:
        return 'Fizz'
    elif total_length % 5 == 0:
        return 'Buzz'
    else:
        return total_length
