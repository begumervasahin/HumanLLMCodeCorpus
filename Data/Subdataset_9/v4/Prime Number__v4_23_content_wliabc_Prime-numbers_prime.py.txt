# Prompt the user to enter an integer for checking prime
mynum = int(input("Please enter an integer you want to check: "))

# Initialize a variable to store the prime status
isprime = True

# Initialize a variable for iterating through possible divisors
i = 2

# Iterate through possible divisors to check for primality
while i < mynum:
    if (mynum % i) == 0:
        # If the number is divisible by any divisor other than 1 and itself, it is not prime
        isprime = False
        break
    i += 1

# Check the prime status and print the result
if isprime:
    print(str(mynum) + " is a prime number")
else:
    print(str(mynum) + " is not a prime number")
