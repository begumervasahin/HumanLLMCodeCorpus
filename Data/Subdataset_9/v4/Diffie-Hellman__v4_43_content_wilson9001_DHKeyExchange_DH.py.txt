def mod_exp(x, y, N):
    """
    Computes modular exponentiation: x^y mod N.
    
    Args:
    x (int): Base.
    y (int): Exponent.
    N (int): Modulus.
    
    Returns:
    int: Result of modular exponentiation.
    """
    if y == 0:
        return 1
    z = mod_exp(x, y // 2, N)
    if y % 2 == 0:
        return z ** 2 % N
    else:
        return (x * z ** 2) % N

def truncate_int(integer, digits_to_truncate):
    """
    Truncates an integer to the specified number of digits.
    
    Args:
    integer (int): The integer to be truncated.
    digits_to_truncate (int): Number of digits to truncate.
    
    Returns:
    int: Truncated integer.
    """
    if digits_to_truncate > 0:
        truncating_divisor = 10 ** digits_to_truncate
        integer = integer // truncating_divisor
    return integer

def find_int_truncate_length(smaller_size_int, larger_size_int):
    """
    Finds the number of digits to truncate for two integers.
    
    Args:
    smaller_size_int (int): Smaller integer.
    larger_size_int (int): Larger integer.
    
    Returns:
    int: Number of digits to truncate.
    """
    return len(str(larger_size_int)) - len(str(smaller_size_int))
