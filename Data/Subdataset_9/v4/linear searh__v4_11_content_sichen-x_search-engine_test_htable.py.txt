from htable import *

def test_empty():
    # Create an empty hash table with size 5
    table = htable(5)
    
    # Ensure that the string representation of the hash table is "{}"
    assert htable_str(table) == "{}"
    
    # Ensure that the string representation of hash table buckets is empty
    assert htable_buckets_str(table) == """0000->
0001->
0002->
0003->
0004->
"""

def test_single():
    # Create an empty hash table with size 5
    table = htable(5)
    
    # Put a key-value pair into the hash table
    htable_put(table, "parrt", 99)
    
    # Ensure that the string representation of the hash table is "{parrt:99}"
    assert htable_str(table) == "{parrt:99}"
    
    # Ensure that the string representation of hash table buckets reflects the inserted key-value pair
    assert htable_buckets_str(table) == """0000->
0001->
0002->
0003->parrt:99
0004->
"""

def test_singleon():
    # Create an empty hash table with size 5
    table = htable(5)
    
    # Put a key-value pair where value is a set
    htable_put(table, "parrt", set([99]))
    
    # Ensure that the string representation of the hash table reflects the inserted set
    assert htable_str(table) == "{parrt:set([99])}"
    
    # Ensure that the string representation of hash table buckets reflects the inserted key-value pair
    assert htable_buckets_str(table) == """0000->
0001->
0002->
0003->parrt:set([99])
0004->
"""

def test_int_to_int():
    # Create an empty hash table with size 5
    table = htable(5)
    
    # Insert key-value pairs where both key and value are integers
    for i in range(1, 11):
        htable_put(table, i, i)
    
    # Ensure that the string representation of the hash table reflects the inserted key-value pairs
    assert htable_str(table) == "{5:5, 10:10, 1:1, 6:6, 2:2, 7:7, 3:3, 8:8, 4:4, 9:9}"
    
    # Ensure that the string representation of hash table buckets reflects the inserted key-value pairs
    assert htable_buckets_str(table) == """0000->5:5, 10:10
0001->1:1, 6:6
0002->2:2, 7:7
0003->3:3, 8:8
0004->4:4, 9:9
"""

def test_str_to_str():
    # Create an empty hash table with size 5
    table = htable(5)
    
    # Insert key-value pairs where both key and value are strings
    htable_put(table, "a", "x")
    htable_put(table, "b", "y")
    htable_put(table, "c", "z")
    htable_put(table, "f", "i")
    htable_put(table, "g", "j")
    htable_put(table, "k", "k")
    
    # Ensure that the string representation of the hash table reflects the inserted key-value pairs
    assert htable_str(table) == '{a:x, f:i, k:k, b:y, g:j, c:z}'
    
    # Ensure that the string representation of hash table buckets reflects the inserted key-value pairs
    assert htable_buckets_str(table) == """0000->
0001->
0002->a:x, f:i, k:k
0003->b:y, g:j
0004->c:z
"""

def test_str_to_set():
    # Create an empty hash table with size 5
    table = htable(5)
    
    # Insert key-value pairs where value is a list
    htable_put(table, "parrt", [2, 99, 3942])
    htable_put(table, "tombu", [6, 3, 1024, 99, 102342])
    
    # Ensure that the string representation of the hash table reflects the inserted key-value pairs
    assert htable_str(table) == "{tombu:[6, 3, 1024, 99, 102342], parrt:[2, 99, 3942]}"
    
    # Ensure that the string representation of hash table buckets reflects the inserted key-value pairs
    assert htable_buckets_str(table) == """0000->
0001->tombu:[6, 3, 1024, 99, 102342]
0002->
0003->parrt:[2, 99, 3942]
0004->
"""
