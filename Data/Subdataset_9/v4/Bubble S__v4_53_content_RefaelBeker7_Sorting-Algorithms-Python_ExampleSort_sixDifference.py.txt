# Function to calculate the percentage difference between two numbers
def check_difference(num1, num2):
    if num1 > num2:
        return ((num2 / num1) * 100)
    else:
        return ((num1 / num2) * 100)

# Function to print an array
def print_array(arr, name):
    if name == 'Bubble':
        print('Sorted array from Bubble Sort:')
    elif name == 'Insertion':
        print('Sorted array from Insertion Sort:')
    elif name == 'Selection':
        print('Sorted array from Selection Sort:')
    elif name == 'Quick':
        print('Sorted array from Quick Sort:')
    elif name == 'Merge':
        print('Sorted array from Merge Sort:')
    elif name == 'Heap':
        print('Sorted array from Heap Sort:')
    
    for i in range(len(arr)):

# Function to measure the time taken by a sorting algorithm
def check_time(name_func, start_time):
    elapsed_time = time.time() - start_time
    print('Function [{}] finished in {:.3f} seconds'.format(name_func.__name__, float(elapsed_time)))
    return elapsed_time

# Heap Sort functions
def heapify(arr, n, i):
    largest = i
    l = 2 * i + 1
    r = 2 * i + 2
    count = 0

    if l < n and arr[i] < arr[l]:
        largest = l
        count += 1

    if r < n and arr[largest] < arr[r]:
        largest = r
        count += 1

    if largest != i:
        arr[i], arr[largest] = arr[largest], arr[i]
        count += 1
        heapify(arr, n, largest)
    
    return count

def heap_sort(arr):
    n = len(arr)
    count = 0

    for i in range(n, -1, -1):
        count += 1
        count += heapify(arr, n, i)
    
    for i in range(n - 1, 0, -1):
        count += 1
        arr[i], arr[0] = arr[0], arr[i]
        count += heapify(arr, i, 0)

# Merge Sort function
def merge_sort(arr):
    if len(arr) > 1:
        mid = len(arr) // 2
        L = arr[:mid]
        R = arr[mid:]

        merge_sort(L)
        merge_sort(R)

        i = j = k = 0

        while i < len(L) and j < len(R):
            if L[i] < R[j]:
                arr[k] = L[i]
                i += 1
            else:
                arr[k] = R[j]
                j += 1
            k += 1

        while i < len(L):
            arr[k] = L[i]
            i += 1
            k += 1

        while j < len(R):
            arr[k] = R[j]
            j += 1
            k += 1

# Quick Sort functions
def partition(arr, low, high):
    i = (low - 1)
    pivot = arr[high]

    for j in range(low, high):
        if arr[j] <= pivot:
            i = i + 1
            arr[i], arr[j] = arr[j], arr[i]

    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    return (i + 1)

def quick_sort_helper(arr, low, high):
    if low < high:
        pi = partition(arr, low, high)
        quick_sort_helper(arr, low, pi - 1)
        quick_sort_helper(arr, pi + 1, high)

def quick_sort(arr):
    quick_sort_helper(arr, 0, len(arr) - 1)

# Selection Sort function
def selection_sort(arr):
    count = 0

    for i in range(len(arr)):
        count += 1
        min_index = i

        for j in range(i + 1, len(arr)):
            count += 1
            if arr[min_index] > arr[j]:
                min_index = j

        arr[i], arr[min_index] = arr[min_index], arr[i]

# Insertion Sort function
def insertion_sort(arr):
    count = 0

    for i in range(1, len(arr)):
        key = arr[i]
        j = i - 1

        while j >= 0 and key < arr[j]:
            count += 1
            arr[j + 1] = arr[j]
            j -= 1
        
        arr[j + 1] = key
        count += 1

# Bubble Sort function
def bubble_sort(arr):
    n = len(arr)
    count = 0

    for i in range(n):
        for j in range(0, n - i - 1):
            count += 1
            if arr[j] > arr[j + 1]:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]
        
        count += 1
