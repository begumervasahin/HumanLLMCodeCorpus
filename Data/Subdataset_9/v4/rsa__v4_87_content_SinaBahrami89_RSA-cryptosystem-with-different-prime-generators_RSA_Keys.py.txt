import prime_generator
from random import randrange
from math import gcd
from datetime import datetime

def generate_keys(bits):
    # Generate two large prime numbers
    p = prime_generator.gen_mr(bits)
    q = prime_generator.gen_mr(bits)
    
    # Calculate n and phi(n)
    n = p * q
    phi = (p - 1) * (q - 1)
    
    # Choose a random integer e such that 1 < e < phi and gcd(e, phi) = 1
    while True:
        e = randrange(3, phi, 2)
        if gcd(e, phi) == 1:
            break
    
    # Calculate the private key using the extended Euclidean algorithm
    a, b = phi, e
    t, t1 = 0, 1
    while b != 0:
        d = a // b
        t, t1 = t1, t - d * t1
        a, b = b, a - d * b
    if t < 0:
        t = t + phi
    
    # Return public and private keys
    public_key = (e, n)
    private_key = (t, n)
    return 'Public key: {}\nPrivate key: {}'.format(public_key, private_key)
