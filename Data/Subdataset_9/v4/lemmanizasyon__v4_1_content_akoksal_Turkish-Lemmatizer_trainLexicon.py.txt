import re
import json
import pickle
import sys

def load_word(dataset):
    if dataset == "wiktionary":
        with open('Datasets/wiktionary.pkl', 'rb') as f:
            [word_list] = pickle.load(f)

        word_dict = {}
        for i in range(len(word_list)):
            word = word_list[i].replace("â", "a")
            word_dict[find_id(word_dict, word)] = ["kok", find_id(word_dict, word)]

        return word_dict
    elif dataset == "zargan":
        with open('Datasets/zargan.pkl', 'rb') as f:
            zargan_dict = pickle.load(f)

        word_dict = {}
        for word, value_list in zargan_dict.items():
            word = word.replace("â", "a")
            word_dict[find_id(word_dict, word)] = ["kok", find_id(word_dict, word)]

        return word_dict

def find_id(word_dict, word):
    ct = 1
    while f"{word}_{ct}" in word_dict:
        ct += 1
    return f"{word}_{ct}"

def generate(word_dict, event):
    new_dict = {}
    if event == "olumsuzluk eki":
        for word, value_list in word_dict.items():
            ind = word[word.index("_") + 1:]
            base_word = word[:word.index("_")]

            if base_word.endswith("mak"):
                base_word = base_word[:-3] + "mamak"
                new_dict[find_id(word_dict, base_word)] = ["olumsuzluk", f"{base_word}_{ind}"]
            elif base_word.endswith("mek"):
                base_word = base_word[:-3] + "memek"
                new_dict[find_id(word_dict, base_word)] = ["olumsuzluk", f"{base_word}_{ind}"]
    elif event == "unsuz yumusamasi":
        for word, value_list in word_dict.items():
            ind = word[word.index("_") + 1:]
            base_word = word[:word.index("_")]

            if not (base_word.endswith("mak") or base_word.endswith("mek")):
                if base_word.endswith("p"):
                    new_dict[find_id(word_dict, base_word[:-1] + "b")] = ["unsuz yumusamasi", f"{base_word}_{ind}"]
                elif base_word.endswith("ç"):
                    new_dict[find_id(word_dict, base_word[:-1] + "c")] = ["unsuz yumusamasi", f"{base_word}_{ind}"]
                elif base_word.endswith("t"):
                    new_dict[find_id(word_dict, base_word[:-1] + "d")] = ["unsuz yumusamasi", f"{base_word}_{ind}"]
                elif base_word.endswith("k"):
                    if base_word.endswith("nk"):
                        new_dict[find_id(word_dict, base_word[:-1] + "g")] = ["unsuz yumusamasi", f"{base_word}_{ind}"]
                    else:
                        new_dict[find_id(word_dict, base_word[:-1] + "ğ")] = ["unsuz yumusamasi", f"{base_word}_{ind}"]
    elif event == "unlu daralmasi":
        for word, value_list in word_dict.items():
            ind = word[word.index("_") + 1:]
            base_word = word[:word.index("_")]

            if base_word == "demek":
                new_dict[find_id(word_dict, "di")] = ["unlu daralmasi", f"{base_word}_{ind}"]
            elif base_word == "yemek":
                new_dict[find_id(word_dict, "yi")] = ["unlu daralmasi", f"{base_word}_{ind}"]
            elif base_word.endswith("amak"):
                new_dict[find_id(word_dict, base_word[:-4] + "ı")] = ["unlu daralmasi", f"{base_word}_{ind}"]
                new_dict[find_id(word_dict, base_word[:-4] + "u")] = ["unlu daralmasi", f"{base_word}_{ind}"]
            elif base_word.endswith("emek"):
                new_dict[find_id(word_dict, base_word[:-4] + "i")] = ["unlu daralmasi", f"{base_word}_{ind}"]
                new_dict[find_id(word_dict, base_word[:-4] + "ü")] = ["unlu daralmasi", f"{base_word}_{ind}"]
    elif event == "unlu dusmesi":
        for word, value_list in word_dict.items():
            ind = word[word.index("_") + 1:]
            base_word = word[:word.index("_")]

            drop_forms = {'akis': 'aks', 'akıl': 'akl', 'alın': 'aln', ...}  # (abbreviated for brevity)

            if base_word in drop_forms:
                new_dict[find_id(word_dict, drop_forms[base_word])] = ["unlu dusmesi", f"{base_word}_{ind}"]
    elif event == "fiil":
        for word, value_list in word_dict.items():
            ind = word[word.index("_") + 1:]
            base_word = word[:word.index("_")]

            if base_word.endswith("mek") or base_word.endswith("mak"):
                new_dict[find_id(word_dict, base_word[:-3])] = ["fiil", f"{base_word}_{ind}"]

    return new_dict

def append_dict(revised_dict, new_dict):
    for word, value_list in new_dict.items():
        if word in revised_dict:
            revised_dict[find_id(revised_dict, word[:word.index("_")])] = value_list
        else:
            revised_dict[word] = value_list
    return revised_dict

def main():
    dataset = "zargan"

    if len(sys.argv) > 1:
        dataset = sys.argv[1]

    word_dict = load_word(dataset)

    word_dict = append_dict(word_dict, generate(word_dict, "olumsuzluk eki"))
    print("Negative verbs are added.")

    word_dict = append_dict(word_dict, generate(word_dict, "fiil"))
    print("Zero infinitive forms of verbs are added.")

    revised_dict = dict(word_dict)

    for event in ["unsuz yumusamasi", "unlu daralmasi", "unlu dusmesi"]:
        new_dict = generate(word_dict, event)
        revised_dict = append_dict(revised_dict, new_dict)

        if event == "unsuz yumusamasi":
            print("Consonant softening forms are added.")
        elif event == "unlu daralmasi":
            print("Becoming close vowel forms are added.")
        elif event == "unlu dusmesi":
            print("Dropping vowel forms are added.")

    with open('revisedDict.pkl', 'wb') as f:
        pickle.dump(revised_dict, f)

    print("Transformed lexicon is saved to revisedDict.pkl")

if __name__ == "__main__":
    main()
