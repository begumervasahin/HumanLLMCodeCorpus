"""
This Telegram bot uses the Updater class to handle the bot.
First, a few handler functions are defined. Then, those functions are passed to
the Dispatcher and registered at their respective places.
Then, the bot is started and runs until we press Ctrl-C on the command line.
Usage:
Basic Echobot example, repeats messages.
Press Ctrl-C on the command line or send a signal to the process to stop the
bot.
"""

import logging
import binascii
import argparse
import itertools
from telegram.ext import Updater, CommandHandler, MessageHandler, Filters

# Define the encryption key
key = '%^*%'

def start(bot, update):
    """
    Handler function for the /start command.
    Replies to the user with a welcome message.
    """
    # Extract the user's message
    user_message = update.message.text
    
    # Check if the message starts with a '~' indicating it's encrypted
    if user_message.startswith('~'):
        # Decrypt the message
        decrypted_message = decrypt(user_message, key)
    else:
        # Encrypt the message
        encrypted_message = encrypt(user_message, key)
    
    # Send the encrypted or decrypted message back to the user
    bot.sendMessage(chat_id=update.message.chat.id, text=decrypted_message)

def encrypt(msg, key):
    """
    Encrypts the given message using the XOR encryption algorithm with the provided key.
    Returns the encrypted message.
    """
    # Convert the message to hexadecimal
    hex_msg = (binascii.hexlify(msg.encode('utf-8'))).decode('utf-8')
    
    # XOR the message with the key
    encrypted_msg = xor_str(hex_msg, key)
    
    # Convert the encrypted message to hexadecimal
    cipher = (binascii.hexlify(encrypted_msg.encode())).decode()
    
    # Prepend '~' to indicate that the message is encrypted
    cipher = "~" + cipher
    
    return cipher

def decrypt(cipher, key):
    """
    Decrypts the given ciphertext using the XOR encryption algorithm with the provided key.
    Returns the decrypted message.
    """
    # Remove the leading '~' from the ciphertext
    cipher = cipher[1:]
    
    # Convert the ciphertext from hexadecimal
    cipher = (binascii.unhexlify(cipher.encode())).decode()
    
    # XOR the ciphertext with the key to get the original message
    msg = xor_str(cipher, key)
    
    # Convert the decrypted message from hexadecimal
    msg = (binascii.unhexlify(msg.encode('utf-8'))).decode('utf-8')
    
    return msg

def xor_str(a, b):
    """
    Performs XOR operation on two strings a and b.
    Returns the result of the XOR operation.
    """
    # XOR each character of the two strings
    xor_result = ''.join([chr(ord(x) ^ ord(y)) for x, y in zip(a, itertools.cycle(b))])
    return xor_result
