import networkx as nx
import matplotlib.pyplot as plt
from prim_algorithm import prim

# Function to build a Fibonacci heap
def build_heap(V, k, heap):
    nodes_heap = dict()
    for v in V:
        nodes_heap[v] = heap.insert(k[v], v)
    return nodes_heap

# Creating an empty graph
G = nx.Graph()

# Defining the set of nodes
node_set = ('a', 'b', 'c', 'd')

# Adding nodes to the graph with positions and initial keys
G.add_node('a', pos=(0, 1), key=0)
G.add_node('b', pos=(2, 1), key=0)
G.add_node('c', pos=(1, 0), key=0)
G.add_node('d', pos=(1, 2), key=0)

# Adding edges to the graph with weights
G.add_edge('a', 'b', weight=2)
G.add_edge('a', 'c', weight=30)
G.add_edge('b', 'c', weight=90)
G.add_edge('d', 'b', weight=1)
G.add_edge('d', 'a', weight=3)

# Getting attributes of the edges and nodes
weight = nx.get_edge_attributes(G, 'weight')
pos = nx.get_node_attributes(G, 'pos')
edges = G.edges
start_node = 'a'

# Finding the Minimum Spanning Tree (MST) using Prim's algorithm
MST = prim(G, weight, start_node, node_set)
edges_MST = MST.edges()
weight_MST = nx.get_edge_attributes(MST, 'weight')

# Plotting the original graph and its MST
plt.subplot(121)
nx.draw(G, pos, with_labels=True)
nx.draw_networkx_edge_labels(G, pos, edge_labels=weight)
nx.draw(G, pos, edges=edges)

plt.subplot(122)
nx.draw(MST, pos, with_labels=True)
nx.draw_networkx_edge_labels(MST, pos, edge_labels=weight_MST)
nx.draw(MST, pos, edges=edges_MST)

plt.show()
