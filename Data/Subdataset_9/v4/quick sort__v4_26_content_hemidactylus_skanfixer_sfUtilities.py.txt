import os

# List of supported image file extensions
PIC_SUFFIXES = ['jpg', 'jpeg', 'bmp', 'tiff', 'gif', 'png']

def safeBuildFileName(dir, body, extension):
    '''
    Combines a directory, a filename body, and an extension
    Adds a numeric suffix to the body if the file already exists.
    '''
    new_body = body
    iter_count = 0
    
    while True:
        fname = os.path.join(dir, '%s.%s' % (new_body, extension))
        
        if not os.path.isfile(fname):
            return fname
        else:
            iter_count += 1
            new_body = '%s_r%03i' % (body, iter_count)
            continue

def ensureDirectoryExists(dirName):
    '''
    Creates the directory if it does not exist.
    '''
    if not os.path.isdir(dirName):
        os.mkdir(dirName)

def rightClipText(text, length):
    '''
    Clips the text from the right if it exceeds the specified length.
    '''
    if len(text) > length:
        return '...' + text[-length:]
    else:
        return text

def popItem(qList, qItem):
    '''
    Removes an item from the list.
    '''
    qList.pop(qList.index(qItem))
    return

def listImageFiles(nDir):
    '''
    Builds a sorted list of image files in the given directory.
    '''
    return sorted([fN for fN in os.listdir(nDir) if isPicture(fN)])

def isPicture(filename):
    '''
    Checks if the file has a supported image file extension.
    '''
    return any(filename[-len(suffix):] == suffix for suffix in PIC_SUFFIXES)

def findRescaleFactor(imgSize, allowedSize, allowZoom=True):
    '''
    Calculates the rescale factor given an image size and the size of the container.
    '''
    mFactor = max(float(imgDim) / float(alwDim) for imgDim, alwDim in zip(imgSize, allowedSize))
    
    if not allowZoom and mFactor < 1:
        mFactor = 1
    
    print('mFactor=%.3f' % mFactor)
    return mFactor

# Allowed characters for filename
fileNameAllowedChars = 'qwertyuiopasdfghjklzxcvbnmQWERTYUIOPASDFGHJKLZXCVBNM' + \
                        '1234567890-_~.,'

def normalizeString(inText):
    '''
    Strictly removes characters not in the allowed set.
    Used for storing labels as filename parts.
    '''
    return ''.join([c for c in inText if c in fileNameAllowedChars])
