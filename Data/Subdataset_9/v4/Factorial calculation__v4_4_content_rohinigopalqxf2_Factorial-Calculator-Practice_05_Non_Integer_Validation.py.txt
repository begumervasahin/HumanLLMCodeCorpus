# 2) Navigate to https://qainterview.pythonanywhere.com/ page
# 3) Enter an integer value in the textbox
# 4) Click on the calculate! button and check the validation message
# 5) Verify that the factorial value is printed
# 6) Close the browser

import time
from selenium import webdriver

# Launch Chrome Driver
driver = webdriver.Chrome()
driver.maximize_window()

try:
    # Navigate to the page
    driver.get("https://qainterview.pythonanywhere.com/")

    # Enter an invalid value (non-integer) in the textbox
    input_value = 'aa'
    textbox = driver.find_element_by_id("number")
    textbox.send_keys(input_value)

    # Click on the calculate! button
    calculate_button = driver.find_element_by_xpath("//*[@id='getFactorial']")
    calculate_button.click()

    # Check the validation message
    validation_message = driver.find_element_by_xpath("//*[@id='resultDiv']").text      

    # Verify that the factorial calculation failed for the non-integer input
    if validation_message == "Please enter an integer":
        print("Success: Factorial calculation failed for non-integer input")        
    else:
        print("Failure: Factorial is calculated for non-integer input")     

finally:
    # Close the browser
    driver.close()
