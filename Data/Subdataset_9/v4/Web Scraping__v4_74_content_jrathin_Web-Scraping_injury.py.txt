import requests
from bs4 import BeautifulSoup
from selenium import webdriver

def scrape_injuries():
    # Define the URL to scrape
    url = 'https://www.online-betting.me.uk/germany-bundesliga-injuries-and-suspensions.php'
    
    # Initialize a Chrome webdriver
    browser = webdriver.Chrome('chromedriver.exe')
    
    # Open the webpage using the webdriver
    browser.get(url)
    
    # Get the HTML content of the webpage
    html = browser.page_source
    
    # Parse the HTML content using BeautifulSoup
    soup = BeautifulSoup(html, 'html.parser')
    
    # Find the relevant section containing injury information
    injury_section = soup.find('div', {'class': 'injuries-table-container'})
    
    # Extract the headings of the injury tables
    injury_headings = injury_section.find_all('h3')
    
    # Extract the injury tables
    injury_tables = injury_section.find_all('table', {'class': 'injurytable'})
    
    # Initialize a list to store the injury data
    injuries = []
    
    # Loop through each injury table
    for table in injury_tables:
        # Extract the rows of the table
        rows = table.find('tbody').find_all('tr')
        
        # Loop through each row (excluding the header row)
        for row in rows[1:]:
            # Extract the columns of the row
            columns = row.find_all('td')
            
            # Extract relevant information from the columns
            injury_type = columns[0].get('class')[0]
            player_name = columns[1].text
            
            # Append the injury data to the list
            injuries.append({'injury_type': injury_type, 'player_name': player_name})
    
    # Close the webdriver
    browser.quit()
    
    # Return the list of injuries
    return injuries

# Call the function to scrape injuries and print the results
injuries_data = scrape_injuries()
for injury in injuries_data:
    print(injury)
    print('----------------------------------------------')
