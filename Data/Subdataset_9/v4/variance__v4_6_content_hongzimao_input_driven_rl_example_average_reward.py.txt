from collections import deque

class AveragePerStepReward:
    def __init__(self, size):
        # Initialize deque objects to store rewards and times
        self.reward_record = deque(maxlen=size)
        self.time_record = deque(maxlen=size)
        # Initialize variables to store the sum of rewards and times
        self.reward_sum = 0
        self.time_sum = 0
    
    def add(self, reward, time):
        # Check if the deque is at maximum capacity
        if len(self.reward_record) >= self.reward_record.maxlen:
            # Remove the oldest reward and time
            stale_reward = self.reward_record.popleft()
            stale_time = self.time_record.popleft()
            # Update the sum of rewards and times
            self.reward_sum -= stale_reward
            self.time_sum -= stale_time
        # Add the new reward and time to the deque
        self.reward_record.append(reward)
        self.time_record.append(time)
        # Update the sum of rewards and times
        self.reward_sum += reward
        self.time_sum += time
    
    def add_list(self, list_reward, list_time):
        # Ensure that the lengths of the lists are equal
        assert len(list_reward) == len(list_time)
        # Iterate through the lists and add each reward and time
        for i in range(len(list_reward)):
            self.add(list_reward[i], list_time[i])
    
    def add_list_filter_zero(self, list_reward, list_time):
        # Ensure that the lengths of the lists are equal
        assert len(list_reward) == len(list_time)
        # Iterate through the lists and add each non-zero reward and time
        for i in range(len(list_reward)):
            if list_time[i] != 0:
                self.add(list_reward[i], list_time[i])
            else:
                # Ensure that zero time corresponds to zero reward
                assert list_reward[i] == 0
    
    def get_avg_per_step_reward(self):
        # Calculate the average reward per step
        if self.time_sum == 0:
            return 0
        else:
            return float(self.reward_sum) / float(self.time_sum)

# Example usage:
if __name__ == "__main__":
    # Create an instance of AveragePerStepReward with a deque size of 5
    average_reward = AveragePerStepReward(5)
    # Define lists of rewards and times
    rewards = [10, 20, 30, 40, 50]
    times = [1, 1, 1, 1, 1]
    # Add the lists of rewards and times to calculate the average reward per step
    average_reward.add_list(rewards, times)
    # Print the average reward per step
    print("Average Reward per Step:", average_reward.get_avg_per_step_reward())
