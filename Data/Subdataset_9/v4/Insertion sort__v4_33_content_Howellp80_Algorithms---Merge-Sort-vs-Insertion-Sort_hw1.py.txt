import time
import random

# Function for merge sort algorithm
def merge_sort(arr):
    if len(arr) > 1:
        mid = len(arr) // 2
        left_half = arr[:mid]
        right_half = arr[mid:]
        merge_sort(left_half)
        merge_sort(right_half)

        i = j = k = 0
        while i < len(left_half) and j < len(right_half):
            if left_half[i] < right_half[j]:
                arr[k] = left_half[i]
                i += 1
            else:
                arr[k] = right_half[j]
                j += 1
            k += 1

        while i < len(left_half):
            arr[k] = left_half[i]
            i += 1
            k += 1

        while j < len(right_half):
            arr[k] = right_half[j]
            j += 1
            k += 1

# Function for insertion sort algorithm
def insertion_sort(arr):
    for i in range(1, len(arr)):
        cur_val = arr[i]
        pos = i
        while pos > 0 and arr[pos - 1] > cur_val:
            arr[pos] = arr[pos - 1]
            pos -= 1
        arr[pos] = cur_val

# List of array sizes to test
arr_sizes = [0, 2000, 8000, 32000, 128000, 512000, 1024000, 4096000]

# Iterate over array sizes
for size in arr_sizes:
    arr = [random.randint(1, 1000) for _ in range(size)]

    # Measure time taken for merge sort
    mt1 = time.time()
    merge_sort(arr)
    mt2 = time.time()
    merge_time = mt2 - mt1

    # Measure time taken for insertion sort
    it1 = time.time()
    insertion_sort(arr.copy())
    it2 = time.time()
    insertion_time = it2 - it1

    # Print results
    print("Size:", size)
    print("Merge Sort Time:", merge_time)
    print("Insertion Sort Time:", insertion_time)
    print()
