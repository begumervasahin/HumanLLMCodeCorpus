import numpy as np
import wave
import struct

# Define a dictionary to map sample widths to struct format characters
widthStruct = {
    1: 'b',  # 1 byte
    2: 'h'   # 2 bytes
}

# Function to open a WAV file and read its contents
def open_wav(filename):
    # Open the WAV file in read mode
    file = wave.open(filename, 'r')
    
    # Get various properties of the WAV file
    nchannels = file.getnchannels()
    sampwidth = file.getsampwidth()
    framerate = file.getframerate()
    nframes = file.getnframes()
    
    # Store the headers as a tuple
    headers = (nchannels, sampwidth, framerate, nframes)
    
    # Define the struct format string based on the sample width
    structFormat = "<{0}{1}".format(nframes, widthStruct[sampwidth])
    
    # Read frames from the WAV file
    frames = file.readframes(nframes)
    
    # Unpack the frames into samples using the struct format
    samples = np.array(struct.unpack(structFormat, frames))
    
    # Close the WAV file
    file.close()
    
    # Return the headers and samples
    return headers, samples

# Function to save data as a WAV file
def save_wav(filename, headers, samples):
    # Open a WAV file in write mode
    file = wave.open(filename, 'w')
    
    # Unpack the headers
    nchannels, sampwidth, framerate, nframes = headers
    
    # Set various properties of the WAV file
    file.setnchannels(nchannels)
    file.setsampwidth(sampwidth)
    file.setframerate(framerate)
    file.setnframes(nframes)
    
    # Clip the samples to fit within the specified sample width
    samples = np.clip(samples[:nframes], -2**(sampwidth*8-1), 2**(sampwidth*8-1)-1)
    
    # Define the struct format string based on the sample width
    structFormat = "<{0}{1}".format(nframes, widthStruct[sampwidth])
    
    # Pack the samples into frames using the struct format
    frames = struct.pack(structFormat, *samples)
    
    # Write the frames to the WAV file
    file.writeframes(frames)
    
    # Close the WAV file
    file.close()
