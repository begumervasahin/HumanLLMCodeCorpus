# Importing necessary libraries
import codecs
import win_unicode_console
from nltk.tokenize import sent_tokenize, word_tokenize

# Enabling Unicode for Windows console
win_unicode_console.enable()

# Opening the file in utf-8 encoding
archivo = codecs.open('texto.txt', 'r', encoding='utf-8')

# Initializing an empty string to store the text
texto = ""

# Reading the file line by line and appending to the text string
for linea in archivo:
    linea = linea.strip()  # Removing leading and trailing whitespaces
    texto = texto + " " + linea  # Concatenating lines to form the complete text

# Tokenizing the text into sentences
tokens = sent_tokenize(texto)

# Displaying information about the sentences
print("\nThe text contains {} sentences.".format(len(tokens)))
print("Numbered from 0 to {}\n".format(len(tokens) - 1))

# Asking user to input a sentence number
numero_sentencia = int(input("Enter the sentence number: "))

# Validating the input sentence number
if 0 < numero_sentencia < len(tokens):
    print("\n[Sentence {}] -> {}\n".format(numero_sentencia, tokens[numero_sentencia]))
    
    # Tokenizing the selected sentence into words
    word_tokens = word_tokenize(tokens[numero_sentencia])

    # Displaying each word token along with its index
    for index, token in enumerate(word_tokens):
        print("Token {} -> {}".format(index, token))
    
    # Displaying total word tokens in the selected sentence
    print("\nTotal word tokens in sentence {}: {}".format(numero_sentencia, len(word_tokens)))
    
    # Displaying total number of sentences
    print("Total sentences: {}\n".format(len(tokens) - 1))
else:
    print("\nInvalid sentence number. Please enter a valid number.")
