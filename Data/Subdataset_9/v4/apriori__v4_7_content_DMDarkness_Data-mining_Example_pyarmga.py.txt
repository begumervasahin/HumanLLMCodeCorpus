# This script demonstrates the usage of the pyarmga package to mine association rules from a transactional dataset.

# Import the necessary modules
import pyarmga as ga
import re

# Define a function to read the transactional dataset from a file
def read_dataset(file_name):
    transactions = []
    with open(file_name, 'r') as file:
        for line in file:
            # Split the line by space or newline and extract integers
            items = re.findall(r'\d+', line)
            if items:
                transactions.append([int(item) for item in items])
    return transactions

if __name__ == '__main__':
    # Specify the file containing the transactional dataset
    dataset_file = "kosarak.dat"

    # Read the transactional dataset from the file
    dataset = read_dataset(dataset_file)

    # Mine association rules from the dataset
    # Set minimum support to 0.7, minimum confidence to 1
    # Use a population size of 30 and run for 30 generations
    # Set mutation probability to 0.25, selection probability to 1, and crossover probability to 1
    # Set the length of rules to 10
    association_rules = ga.getAR(dataset, 0.7, 1, 30, 30, 0.25, 1, 1, 10)

    # Display the mined association rules
    for rule in association_rules:
        print(rule)
