# Import necessary libraries
from dotenv import load_dotenv
import os
import psycopg2
import pandas as pd
import matplotlib.pyplot as plt

# Initialize variables
conn = None
df = None

# Function to create variables from environment variables
def create_variables():
    global hostname, port, database, username, password
    # Load environment variables
    hostname = os.getenv('hostname')
    port = os.getenv('port')
    database = os.getenv('database')
    username = os.getenv('username')
    password = os.getenv('password')

# Function to compute results from database
def compute_results():
    print('Computing results...')
    # Establish database connection
    conn = psycopg2.connect(host=hostname, database=database, user=username, password=password)
    # Execute SQL query and read results into a DataFrame
    df = pd.read_sql('SELECT name FROM studies RIGHT JOIN sponsors ON studies.nct_id = sponsors.nct_id', con=conn)
    print('Done.')
    return df

# Function to extract unique sponsor names from DataFrame
def get_unique_sponsor_names(df):
    names = []
    # Iterate through DataFrame column to find unique names
    for name in df['name'].tolist():
        if name not in names:
            names.append(name)
    return names

# Function to write list of names to a file
def write_list_to_file(names_list):
    with open('unique_sponsor_names.txt', 'w') as f:
        # Write each name to the file
        for item in names_list:
            f.write("%s\n" % item)
    print("New list created.")

if __name__ == '__main__':
    # Load environment variables from .env file
    load_dotenv()
    # Create variables from environment variables
    create_variables()
    # Compute results from the database
    df = compute_results()
    
    # Find sponsor names containing "Merck" or "MSD"
    merck_names = {}
    for name in df['name'].tolist():
        if "Merck" in name or "MSD" in name:
            if name not in merck_names:
                merck_names[name] = 1
            else:
                merck_names[name] += 1
    
    # Plot a pie chart of Merck-related sponsor names
    plt.pie(merck_names.values(), labels=merck_names.keys())
    plt.show()
