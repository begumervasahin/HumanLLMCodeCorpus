from words import get_text, words

def linear_search(files, terms):
    """
    Search for files containing all terms in their content.
    
    Args:
        files (list): A list of fully-qualified filenames.
        terms (list): A list of strings representing search terms.
        
    Returns:
        list: A list of filenames whose content contains all search terms.
    """
    # Check if input parameters are valid
    if files is None or terms is None or len(files) == 0 or len(terms) == 0:
        return None
    
    # Initialize a list to store matching documents
    matching_docs = []
    
    # Iterate over each file
    for file in files:
        # Get the content of the file
        file_content = get_text(file)
        
        # Flag to track if all terms are found in the file
        all_terms_found = True
        
        # Get the words from the file content
        words_in_file = words(file_content)
        
        # Check if each term is present in the file content
        for term in terms:
            if term not in words_in_file:
                # If any term is not found, set the flag to False and break the loop
                all_terms_found = False
                break
        
        # If all terms are found, add the file to the list of matching documents
        if all_terms_found:
            matching_docs.append(file)
    
    # Return the list of matching documents
    return matching_docs
