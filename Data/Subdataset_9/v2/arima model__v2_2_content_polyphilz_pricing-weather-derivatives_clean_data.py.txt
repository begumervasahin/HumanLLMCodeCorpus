import sys
import pandas as pd

# Function to check if maximum temperature exceeds 100°F for any given day
def test_legitimacy(grouped):
    """
    Check if the maximum temperature exceeds 100°F for any given day.
    """
    for index, row in grouped.iterrows():
        if row["max"] > 100:
            print(f"Date: {index}, Min Temp: {row['min']}, Max Temp: {row['max']}, Mean Temp: {row['mean']}")

# Function to fill NaN values in the grouped data
def fill_nans(grouped):
    grouped.fillna(method="ffill", inplace=True)
    return grouped

# Function to replace negative values with NaN in the grouped data
def make_negatives_nans(grouped):
    grouped["min"] = grouped["min"].apply(lambda x: float("nan") if x < 0 else x)
    grouped["max"] = grouped["max"].apply(lambda x: float("nan") if x < 0 else x)
    grouped["mean"] = grouped["mean"].apply(lambda x: float("nan") if x < 12.46 else x)
    return grouped

# Function to group the temperature data by day and compute min, max, and mean temperatures
def group_data(df):
    grouped = df.groupby(pd.Grouper(freq="D"))["TemperatureF"].agg(
        ["min", "max", "mean"]
    )
    return grouped

# Function to process the temperature data
def process_data(df):
    grouped = group_data(df)
    grouped = make_negatives_nans(grouped)
    grouped = fill_nans(grouped)
    return grouped

# Main function to read the raw temperature data, clean it, test its legitimacy, and save the cleaned data
def main():
    try:
        # Read the raw temperature data from a CSV file
        df = pd.read_csv("temp_data_raw.csv", index_col="Time", parse_dates=True)
    except FileNotFoundError:
        # If the file is not found, prompt the user to run a script to generate the raw data
        print("Please run `scrape_data.py` to generate the raw data first.")
        sys.exit(0)

    # Drop any unnamed columns in the DataFrame
    df.drop(df.columns[df.columns.str.contains("Unnamed")], axis=1, inplace=True)
    
    # Process the temperature data
    grouped = process_data(df)
    
    # Test the legitimacy of the data
    test_legitimacy(grouped)
    
    # Save the cleaned data to a CSV file
    grouped.to_csv("temp_data_cleaned.csv")

# Entry point of the script
if __name__ == "__main__":
    main()
