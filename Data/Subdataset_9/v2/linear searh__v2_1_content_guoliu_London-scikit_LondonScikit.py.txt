import numpy as np
from sklearn.svm import LinearSVC
from sklearn.model_selection import train_test_split, GridSearchCV

# Load the training and test data
x_train = np.loadtxt(open("train.csv", "rb"), delimiter=",", skiprows=0)
y_train = np.loadtxt(open("trainLabels.csv", "rb"), delimiter=",", skiprows=0)
x_test = np.loadtxt(open("test.csv", "rb"), delimiter=",", skiprows=0)

# Define the range of values for the hyperparameter C
tuned_parameters = {'C': 10.**np.arange(-5, 5)}

# Split the training data into training and validation sets
x_train_split, x_val, y_train_split, y_val = train_test_split(x_train, y_train, test_size=0.2, random_state=42)

# Create a Linear Support Vector Classification (LinearSVC) model
classifier = LinearSVC()

# Use GridSearchCV for hyperparameter tuning
grid_search = GridSearchCV(classifier, tuned_parameters, cv=5, verbose=3)
grid_search.fit(x_train_split, y_train_split)

# Display the best hyperparameters found during tuning
print("Best parameters found:")
print(grid_search.best_params_)

# Display the cross-validation scores for different hyperparameter values
print("\nGrid scores:")
for params, mean_score, scores in grid_search.grid_scores_:
    print("%0.3f (+/-%0.03f) for %r" % (mean_score, scores.std() * 2, params))

# Train the model with the best hyperparameters on the full training set
best_classifier = LinearSVC(C=grid_search.best_params_['C'])
best_classifier.fit(x_train, y_train)

# Make predictions on the test set
predictions = best_classifier.predict(x_test)

# Save predictions to a CSV file
output = np.vstack((np.arange(len(predictions)).astype(int) + 1, predictions.astype(int))).T
np.savetxt("LinearSVMresult.csv", output, header="Id,Solution", fmt="%d", comments="", delimiter=",")
