def counting_sort(A, verbose=0, desc=0):
    """
    Sorts a list of integers using the counting sort algorithm.

    Args:
        A (list): The unsorted list of integers.
        verbose (int, optional): Controls verbosity level. Defaults to 0.
        desc (int, optional): Controls whether the output is sorted in descending order. Defaults to 0.

    Returns:
        list: The sorted list.
    """
    # Find the largest and smallest elements in the list
    largest = max(A)
    smallest = min(A)

    # Normalize the data by shifting the smallest number to 0
    if smallest:
        A = [x - smallest for x in A]
    if verbose:
        print("Normalized list:", A)

    # Create a count list based on the frequency of occurrences of the numbers
    count = [0] * (largest - smallest + 1)
    n = len(A)
    for i in range(n):
        count[A[i]] += 1

    # Update the count list to be cumulative
    for i in range(1, len(count)):
        count[i] += count[i - 1]
        if verbose == 2:
            print("Sub:", i, "::", count[i], count[i - 1])

    # Initialize the output list
    B = [0] * n

    # Scan the original list right to left, update output list with key at the index provided by count
    for i in range(n - 1, -1, -1):
        count[A[i]] -= 1
        B[count[A[i]]] = A[i]
        if verbose:
            print("Iteration", i, ":", B)

    # Restore the original values by adding the smallest value back
    if smallest:
        B = [x + smallest for x in B]

    # Reverse the list if descending order is requested
    if desc:
        B = B[::-1]

    return B
