# Initial unsorted array
my_arr = [8, 5, 2, 6, 9, 3, 1, 4, 0, 7]

# Function to perform selection sort
def selection_sort(my_arr):
    # Traverse through all elements of the array
    for i in range(len(my_arr)):
        # Find the index of the minimum element in the unsorted portion of the array
        min_idx = i
        for j in range(i + 1, len(my_arr)):
            if my_arr[min_idx] > my_arr[j]:
                min_idx = j
        # Swap the found minimum element with the first element of the unsorted portion
        temp = my_arr[i]
        my_arr[i] = my_arr[min_idx]
        my_arr[min_idx] = temp
    return my_arr

# Call the selection_sort function with the initial array and print the sorted array
print(selection_sort(my_arr))
