import time
import glob
import numpy as np
import librosa
from sklearn.naive_bayes import GaussianNB
from sklearn.metrics import confusion_matrix, accuracy_score

start_time = time.time()

def extract_audio_features(file_path):
    """Extract MFCC features from audio file."""
    y, _ = librosa.load(file_path)
    mfcc = librosa.feature.mfcc(y)
    mfcc /= np.amax(np.absolute(mfcc))
    return np.ndarray.flatten(mfcc)[:300]

def generate_features_and_labels():
    """Generate features and labels from audio files."""
    all_features = []
    all_labels = []
    ganjils = ['satu', 'tiga', 'lima', 'tujuh', 'sembilan']
    for ganjil in ganjils:
        sound_files = glob.glob(f'dataset/{ganjil}/*.wav')
        print(f'Processing {len(sound_files)} sound files in {ganjil} ...')
        for file in sound_files:
            features = extract_audio_features(file)
            all_features.append(features)
            all_labels.append(ganjil)
    return np.stack(all_features), all_labels

def softmax(x):
    """Compute softmax values for each sets of scores in x."""
    e_x = np.exp(x - np.max(x))
    return e_x / e_x.sum(axis=0)

# Generate features and labels
features, labels = generate_features_and_labels()
features = softmax(features)

# Combine features and labels
alldata = np.column_stack((features, labels))
np.random.seed(200)
np.random.shuffle(alldata)

# Split data into training and testing sets
training_split = 0.9
split_idx = int(len(alldata) * training_split)
train_data, test_data = alldata[:split_idx, :], alldata[split_idx:, :]

train_input = train_data[:, :-256].astype(np.float)
train_labels = train_data[:, -1:]

test_input = test_data[:, :-256].astype(np.float)
test_labels = test_data[:, -1:]

def predict_all():
    """Predict labels using Naive Bayes classifier and evaluate the performance."""
    nb = GaussianNB()
    nb.fit(train_input, train_labels)
    pred = nb.predict(test_input)
    print("Predicted Labels:", pred)
    print("Actual Labels:", test_labels)
    print('\nEvaluation Results for Naive Bayes:')
    print(confusion_matrix(test_labels, pred))
    print("\nAccuracy:", accuracy_score(test_labels, pred))

# Perform prediction and evaluation
predict_all()

end_time = time.time()
print("Time taken:", end_time - start_time)
