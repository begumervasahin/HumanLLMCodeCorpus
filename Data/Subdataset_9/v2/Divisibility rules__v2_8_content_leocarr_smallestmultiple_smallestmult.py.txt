def find_smallest_evenly_divisible(max_mult):
    # Initialize variables
    found = False
    smallest = 0
    current = max_mult

    # Loop until the smallest evenly divisible number is found
    while not found:
        # Count the number of divisors for the current number
        div_count = 0
        for i in range(1, max_mult + 1):
            if current % i == 0:
                div_count += 1

        # If the current number has divisors for all numbers up to max_mult, it is the smallest
        if div_count == max_mult:
            found = True
            smallest = current
        else:
            # Increment the current number by max_mult to check the next potential smallest number
            current += max_mult

    return smallest

if __name__ == "__main__":
    # Get user input for the max multiple
    max_multiple = int(input("What is the max multiple? "))

    # Call the function to find the smallest evenly divisible number
    result = find_smallest_evenly_divisible(max_multiple)

    # Print the result
    print(f"The smallest number that is evenly divisible by all numbers from 1 to {max_multiple} is {result}")
