import math

def kalman_filter():
    # Initialize the count variable
    count = 0

    # Print the Prediction Steps header
    print("\nPrediction Steps")
    print("----------------------------------------")

    # Prompt the user to input values for A, B, prior mean, and current mean
    At = float(input("A: "))
    Bt = float(input("B: "))
    ut_ = float(input("Enter Prior Mean: "))
    ut = float(input("Enter Current Mean: "))

    # Calculate the predicted mean
    pMean = At * ut_ + Bt * ut
    print("Predicted Mean: ", pMean)

    # Prompt the user to input values for E(t-1), A(T/t), and Q(t)
    Et_ = float(input("\nE(t-1): "))
    Att = float(input("A(T/t): "))
    Qt = float(input("Q(t): "))

    # Calculate the predicted covariance
    pCo = At * Et_ * Att + Qt
    print("Predicted Convariance: ", pCo)

    # Start the Kalman filter loop
    while count == 0:
        # Print the Update Steps header
        print("\nUpdate Steps")
        print("----------------------------------------")

        # Prompt the user to input values for C(T/t), C/(t), and measurement covariance
        Ctt = float(input("C(T/t): "))
        Ct = float(input("C/(t): "))
        Rt = float(input("Measurement Covariance: "))

        # Calculate the Kalman gain
        kGain = pCo * Ctt * math.pow((Ct * pCo * Ctt + Rt), (-1))
        print("Kalman Gain: ", kGain)

        # Prompt the user to input the actual measurement
        zt = float(input("\nActual Measurement: "))

        # Update the mean
        uMean = pMean + kGain * (zt - Ct * pMean)
        print("Updated Mean: ", uMean)

        # Update the covariance
        uCo = pCo - kGain * Ct * pCo
        print("\nUpdated Covariance: ", uCo)

        # Print the Prediction Steps header for the next iteration
        print("\nPrediction Steps")
        print("----------------------------------------")

        # Prompt the user to input values for A, B, and the current mean
        At = float(input("A: "))
        Bt = float(input("B: "))
        ut = float(input("Enter Current Mean: "))

        # Calculate the predicted mean for the next iteration
        pMean = At * uMean + Bt * ut
        print("Predicted Mean: ", pMean)

        # Prompt the user to input values for A(T/t) and Q(t)
        Att = float(input("\nA(T/t): "))
        Qt = float(input("Q(t): "))

        # Calculate the predicted covariance for the next iteration
        pCo = At * uCo * Att + Qt
        print("Predicted Convariance: ", pCo)

# Call the Kalman filter function to start the process
kalman_filter()
