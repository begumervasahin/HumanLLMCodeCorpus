import datetime
import sys
import primeSearchServer
import isPrimeSearch

def main():
    # Establish connection to Prime Server
    prime_server_connection = primeSearchServer.primeServer()

    # Default search interval and flags
    search_interval = 1000
    force_use_of_incomplete_intervals = False

    # Check command-line arguments
    if len(sys.argv) > 1:
        if sys.argv[1].isnumeric() and int(sys.argv[1]) > 0:
            search_interval = int(sys.argv[1])
        elif sys.argv[1] == "status":
            # Display status information
            display_server_status(prime_server_connection)
            return
        elif sys.argv[1] == "-?":
            # Display help information
            display_help()
            return
        elif sys.argv[1] == "useIntervals":
            force_use_of_incomplete_intervals = True

    # Get search interval from Prime Server
    start_search_at, end_search_at = prime_server_connection.returnSearchInterval(search_interval, force_use_of_incomplete_intervals)

    # Initialize prime counter and start time
    prime_counter = 0
    start_time = datetime.datetime.now()

    # Display search information
    display_search_info(start_time, start_search_at, end_search_at)

    # Perform prime number search
    prime_counter = search_prime_numbers(start_search_at, end_search_at, prime_server_connection)

    # Display completion and final results
    display_completion_and_results(start_time, start_search_at, end_search_at, prime_counter)

    # Clean up and exit
    prime_server_connection.returnSearchedInterval(start_search_at, end_search_at)
    del prime_server_connection

def display_server_status(connection):
    """
    Display status information from the server.
    """
    print("Number of primes found:", f"{connection.returnTotalNumberOfPrimesFound():,d}".replace(",", "."))
    print("Biggest prime found:", f"{connection.returnHighestPrimeFound():,d}".replace(",", "."))
    print("Number of unfinished intervals:", f"{connection.returnUnfinishedIntervals():,d}".replace(",", "."))

def display_help():
    """
    Display help information.
    """
    print("useIntervals forces to use any open intervals on the server")
    print("status gives a status from the server")
    print("follow command with a number, and that is the interval of primes being searched")

def display_search_info(start_time, start_search_at, end_search_at):
    """
    Display search information.
    """
    print("Search for prime numbers\nStarting at:", start_time.strftime("%Y-%m-%d %H:%M:%S"))
    print("Searching from", f"{start_search_at:,d}".replace(",", "."), "to", f"{end_search_at:,d}".replace(",", "."), "(interval:", f"{(end_search_at - start_search_at):,d}".replace(",", "."), ")")
    print("\n0% completed", end="", flush=True)

def search_prime_numbers(start_search_at, end_search_at, connection):
    """
    Search for prime numbers within the given range.
    """
    prime_counter = 0
    completion_rate = 0
    current_number = start_search_at

    while current_number <= end_search_at:
        if isPrimeSearch.checkForPrime6(current_number):
            prime_counter += 1
            connection.returnPrimeFound(current_number)
            current_number += 2
        else:
            current_number += 1

        new_completion_rate = int(round(((current_number - start_search_at) / (end_search_at - start_search_at)) * 100, 0))
        if new_completion_rate > completion_rate:
            completion_rate = new_completion_rate
            print("\r" + str(completion_rate) + "% completed", end="", flush=True)

    return prime_counter

def display_completion_and_results(start_time, start_search_at, end_search_at, prime_counter):
    """
    Display completion and final results.
    """
    end_time = datetime.datetime.now()
    timedelta = end_time - start_time

    print("\r100% completed")
    print("\nStarting at:", start_time.strftime("%Y-%m-%d %H:%M:%S"))
    print("Ending at:", end_time.strftime("%Y-%m-%d %H:%M:%S"))
    print("Running time:", int(timedelta.seconds / 60), "minutes", timedelta.seconds % 60, "seconds")
    print("Searched from", f"{start_search_at:,d}".replace(",", "."), "to", f"{end_search_at:,d}".replace(",", "."), "(interval:", f"{(end_search_at - start_search_at):,d}".replace(",", "."), ")")
    print("Number of primes:", f"{prime_counter:,d}".replace(",", "."), "(", round(prime_counter / (end_search_at - start_search_at), 5), "%)")
    if prime_counter > 0:
        print("Seconds per prime:", round(timedelta.seconds / prime_counter, 6))

if __name__ == "__main__":
    main()
