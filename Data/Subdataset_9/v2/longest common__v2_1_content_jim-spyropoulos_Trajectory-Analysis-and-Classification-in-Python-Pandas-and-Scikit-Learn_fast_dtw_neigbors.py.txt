import csv
import os
import re
import numpy as np
import pandas as pd
from fastdtw import fastdtw
from auxiliaryfunctions import haversine_np, print_results

# Read the cleaned DataFrame
df = pd.read_pickle('./final_cleaned.df')

# Read the test set CSV
with open('./test_set_a1.csv', 'r') as f:
    reader = csv.reader(f)
    # Process each row in the CSV
    data = list(reader)
    trajectories_read = []
    for row in data[1:]:  # Skip the header
        trajectory = []
        for j in range(0, len(row), 3):
            # Extract longitude, latitude, and timestamp from the row
            longitude = float(re.sub('[[]', '', row[j]))
            latitude = float(row[j + 1])
            timestamp = float(re.sub('[]]', '', row[j + 2]))
            trajectory.append([longitude, latitude, timestamp])
        trajectories_read.append(trajectory)
print("Dataset read successfully.")

# Create a directory to store results
os.mkdir("DTWresults")

# Perform DTW calculation for each trajectory in the test set
for j, traj in enumerate(trajectories_read):
    distances = []
    # Calculate DTW distance between each trajectory in the DataFrame and the test trajectory
    for elem in df['timestamp_longitude_latitude']:
        distance, _ = fastdtw(elem, traj, dist=haversine_np)
        distances.append(distance)
    distancesnp = np.array(distances)
    # Find the indices of the top 5 closest trajectories
    sorted_ind = np.argsort(distancesnp)[:5]
    # Print and save the results
    for k, elem in enumerate(sorted_ind):
        print_results(df, distancesnp, elem, f"./DTWresults/trajectory{j}neighbor{k}")
