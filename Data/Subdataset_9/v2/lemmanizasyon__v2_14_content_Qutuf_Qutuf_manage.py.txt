import os
import sys

def main():
    # Set the Django settings module to 'webservice.settings'
    os.environ.setdefault("DJANGO_SETTINGS_MODULE", "webservice.settings")
    
    try:
        # Try to import execute_from_command_line from django.core.management
        from django.core.management import execute_from_command_line
    except ImportError:
        try:
            # If Django cannot be imported, try importing Django itself
            import django
        except ImportError:
            # If Django is still not importable, raise an ImportError with a descriptive message
            raise ImportError(
                "Couldn't import Django. Are you sure it's installed and "
                "available on your PYTHONPATH environment variable? Did you "
                "forget to activate a virtual environment?"
            )
        # If Django is imported but execute_from_command_line cannot be imported, re-raise the ImportError
        raise
    
    # If everything is imported successfully, execute the Django management command line
    execute_from_command_line(sys.argv)

if __name__ == "__main__":
    # Call the main function if the script is executed directly
    main()
