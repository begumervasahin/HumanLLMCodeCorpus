# Import necessary libraries
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from sklearn.svm import SVR

# Load the dataset
dataset = pd.read_csv('Position_Salaries.csv')

# Extract features (Position Level) and target variable (Salary)
X = dataset.iloc[:, 1:2].values
Y = dataset.iloc[:, 2].values

# Create SVR model
regressor = SVR(kernel='rbf')  # Using Radial Basis Function kernel

# Fit the SVR model to the data
regressor.fit(X, Y)

# Predict salary for a new position level (6.5)
predicted_salary = regressor.predict([[6.5]])
print("Predicted Salary for Position Level 6.5:", predicted_salary)

# Visualize the SVR results
plt.scatter(X, Y, color='red', label='Actual Data')  # Plot actual data points
plt.plot(X, regressor.predict(X), color='blue', label='SVR Model')  # Plot SVR predictions
plt.title('Truth or Bluff (SVR)')
plt.xlabel('Position Level')
plt.ylabel('Salary')
plt.legend()
plt.show()

# Visualize with higher resolution and smoother curve
X_grid = np.arange(min(X), max(X), 0.01)  # Generate a range of values with finer resolution
X_grid = X_grid.reshape((len(X_grid), 1))
plt.scatter(X, Y, color='red', label='Actual Data')  # Plot actual data points
plt.plot(X_grid, regressor.predict(X_grid), color='blue', label='SVR Model')  # Plot SVR predictions
plt.title('Truth or Bluff (SVR)')
plt.xlabel('Position Level')
plt.ylabel('Salary')
plt.legend()
plt.show()
