import math
import matplotlib.pyplot as plt
from pandas import read_csv, datetime

def parse_datetime(x):
    return datetime.strptime(x, '%Y-%m-%d %H:%M:%S')

def mean_absolute_percentage_error(y_true, y_pred):
    total_error = sum(math.fabs((y_p - y_t) / y_t) for y_t, y_p in zip(y_true, y_pred))
    return (total_error / len(y_true)) * 100

# Replace the file path with the actual CSV file path
csv_file_path = '/Users/daihanru/Desktop/arima-lstm/DataSet/FEB15.csv'

# Read CSV file and parse dates using the defined parse_datetime function
series = read_csv(csv_file_path, usecols=[2, 3], header=0, parse_dates=[0],
                  index_col=0, squeeze=True, date_parser=parse_datetime)

# Extract the first 1000 values for plotting
X = series.values[:1000]

# Plot the time series
plt.plot(X, '-', label="Real Flow")
plt.legend(loc='upper right')
plt.xlabel("Period (15-minute intervals)")
plt.ylabel("Volume (vehicle/period)")
plt.show()

# Example of using the mean_absolute_percentage_error function
# Replace y_true and y_pred with your actual data
y_true = [1, 2, 3, 4, 5]
y_pred = [0.9, 2.1, 3.2, 4.3, 5.1]

mape = mean_absolute_percentage_error(y_true, y_pred)
print(f"Mean Absolute Percentage Error (MAPE): {mape:.2f}%")
