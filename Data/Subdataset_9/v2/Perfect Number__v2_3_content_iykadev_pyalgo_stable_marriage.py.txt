import copy

def check_stability(engaged, guyprefers, galprefers):
    # Create a dictionary mapping engaged values to keys
    inverse_engaged = {v: k for k, v in engaged.items()}
    
    # Iterate through engagements
    for she, he in engaged.items():
        shelikes = galprefers[she]
        shelikesbetter = shelikes[:shelikes.index(he)]
        helikes = guyprefers[he]
        helikesbetter = helikes[:helikes.index(she)]
        
        # Check if there are better matches for the current engagement
        for guy in shelikesbetter:
            guysgirl = inverse_engaged[guy]
            guylikes = guyprefers[guy]
            if guylikes.index(guysgirl) > guylikes.index(she):
                print("%s and %s like each other better than "
                      "their present partners: %s and %s, respectively"
                      % (she, guy, he, guysgirl))
                return False
        for gal in helikesbetter:
            girlsguy = engaged[gal]
            gallikes = galprefers[gal]
            if gallikes.index(girlsguy) > gallikes.index(he):
                print("%s and %s like each other better than "
                      "their present partners: %s and %s, respectively"
                      % (he, gal, she, girlsguy))
                return False
    return True

def matchmaker(guyprefers, galprefers):
    # Initialize list of free guys and dictionary of engagements
    guysfree = list(guyprefers.keys())
    engaged = {}
    
    # Create deep copies of preference dictionaries
    guyprefers_copy = copy.deepcopy(guyprefers)
    galprefers_copy = copy.deepcopy(galprefers)

    # Iterate through free guys until all are engaged
    while guysfree:
        guy = guysfree.pop(0)
        guyslist = guyprefers_copy[guy]
        gal = guyslist.pop(0)
        fiance = engaged.get(gal)
        
        # If the girl is not engaged, engage her with the current guy
        if not fiance:
            engaged[gal] = guy
            print("  %s and %s" % (guy, gal))
        else:
            galslist = galprefers_copy[gal]
            # If the girl prefers the current guy to her fiance, engage her with the current guy
            if galslist.index(fiance) > galslist.index(guy):
                engaged[gal] = guy
                print("  %s dumped %s for %s" % (gal, fiance, guy))
                # If the dumped guy has other choices, add him back to the list of free guys
                if guyprefers_copy[fiance]:
                    guysfree.append(fiance)
            else:
                # If the girl prefers her current fiance, move to the next guy in her preference list
                if guyslist:
                    guysfree.append(guy)
    return engaged
