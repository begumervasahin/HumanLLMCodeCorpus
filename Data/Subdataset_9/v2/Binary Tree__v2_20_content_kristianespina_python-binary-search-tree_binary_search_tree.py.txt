from random import randint

class Node:
    """
    Represents a node in a binary tree.
    """

    def __init__(self, value):
        """
        Initializes a node with a given value.
        """
        self.value = value
        self.left = None
        self.right = None

    def insert(self, value):
        """
        Inserts a new value into the binary tree.
        """
        if value < self.value:
            if self.left is None:
                self.left = Node(value)
            else:
                self.left.insert(value)
        elif value > self.value:
            if self.right is None:
                self.right = Node(value)
            else:
                self.right.insert(value)
        else:
            self.value = value

    def convert_to_array(self, node, array):
        """
        Converts the binary tree to an array using Depth-First Search (DFS).
        """
        if node:
            array.append(node.value)
            self.convert_to_array(node.left, array)
            self.convert_to_array(node.right, array)
        return array

class BinaryTree:
    """
    Represents a binary tree data structure.
    """

    def __init__(self):
        """
        Initializes an empty binary tree.
        """
        self.root = None

    def insert(self, data):
        """
        Inserts data into the binary tree.
        """
        if self.root is None:
            self.root = Node(data)
        else:
            self.root.insert(data)

    def find(self, data):
        """
        Finds if a given data value exists in the binary tree.
        """
        return self.find_node(self.root, data)

    def find_node(self, node, data):
        """
        Helper method to recursively find a node with a given data value.
        """
        if node is None:
            return False
        elif data == node.value:
            return True
        elif data < node.value:
            return self.find_node(node.left, data)
        else:
            return self.find_node(node.right, data)

    def convert_to_array(self):
        """
        Converts the binary tree to an array.
        """
        return self.root.convert_to_array(self.root, [])

    def rearrange(self):
        """
        Re-arranges the binary tree such that the median becomes the root.
        """
        nodes = self.convert_to_array()
        nodes.sort()
        middle = len(nodes) // 2
        self.root = Node(nodes.pop(middle))
        for element in nodes:
            self.root.insert(element)
        return self.convert_to_array()

# Example usage:
tree = BinaryTree()
for i in range(50):
    tree.insert(randint(0, 100))

print("Original Tree:", tree.convert_to_array())
print("Rearranged Tree with Median as Root:", tree.rearrange())
print("Is 55 in the tree?", tree.find(55))
