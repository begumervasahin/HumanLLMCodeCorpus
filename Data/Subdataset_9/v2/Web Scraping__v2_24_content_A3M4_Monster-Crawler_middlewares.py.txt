from scrapy import signals, Spider, Request
from scrapy.crawler import CrawlerProcess

# Define a custom spider
class CustomSpider(Spider):
    name = 'custom_spider'
    start_urls = ['http://example.com']

    def parse(self, response):
        self.logger.info('Parsing response from: %s' % response.url)

# Middleware for spider
class MonsterSpiderMiddleware(object):
    @classmethod
    def from_crawler(cls, crawler):
        # Connect the spider_opened signal to the spider_opened method
        s = cls()
        crawler.signals.connect(s.spider_opened, signal=signals.spider_opened)
        return s

    # Process spider input
    def process_spider_input(self, response, spider):
        return None

    # Process spider output
    def process_spider_output(self, response, result, spider):
        for i in result:
            yield i

    # Process spider exception
    def process_spider_exception(self, response, exception, spider):
        pass

    # Process start requests
    def process_start_requests(self, start_requests, spider):
        for r in start_requests:
            yield r

    # Spider opened method
    def spider_opened(self, spider):
        spider.logger.info('Spider opened: %s' % spider.name)

# Middleware for downloader
class MonsterDownloaderMiddleware(object):
    @classmethod
    def from_crawler(cls, crawler):
        # Connect the spider_opened signal to the spider_opened method
        s = cls()
        crawler.signals.connect(s.spider_opened, signal=signals.spider_opened)
        return s

    # Process request
    def process_request(self, request, spider):
        return None

    # Process response
    def process_response(self, request, response, spider):
        return response

    # Process exception
    def process_exception(self, request, exception, spider):
        pass

    # Spider opened method
    def spider_opened(self, spider):
        spider.logger.info('Spider opened: %s' % spider.name)

# Main entry point of the script
if __name__ == "__main__":
    # Configure crawler process
    process = CrawlerProcess({
        'USER_AGENT': 'Mozilla/5.0',
        'LOG_LEVEL': 'INFO',
        'ITEM_PIPELINES': {},
        'DOWNLOADER_MIDDLEWARES': {
            __name__ + '.MonsterDownloaderMiddleware': 543,
        },
        'SPIDER_MIDDLEWARES': {
            __name__ + '.MonsterSpiderMiddleware': 543,
        },
    })

    # Start crawling the custom spider
    process.crawl(CustomSpider)
    process.start()  # the script will block here until the crawling is finished
