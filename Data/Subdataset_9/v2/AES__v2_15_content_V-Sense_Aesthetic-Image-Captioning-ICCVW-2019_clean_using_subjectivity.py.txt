import json
from langdetect import detect
import re
from collections import Counter
from nltk.util import ngrams
from nltk.tokenize import word_tokenize

# File paths for the input and output data.
input_file_path = 'CLEAN_AVA_FULL_COMMENTS.json'
output_file_path = 'processed_comments.json'

def is_comment_in_english(comment_text):
    """
    Determines if a given comment is written in English.
    
    Args:
        comment_text (str): The text of the comment to be checked.
        
    Returns:
        bool: True if the comment is in English, False otherwise.
    """
    try:
        return detect(comment_text) == 'en'
    except:
        return False

def normalize_comment_text(text):
    """
    Normalizes the comment text by reducing repeated characters and removing unwanted punctuation.
    
    Args:
        text (str): The original comment text.
        
    Returns:
        str: The normalized comment text.
    """
    # Reduce excessive repetition of characters to a maximum of two.
    text = re.sub(r'(.)\1+', r'\1\1', text)
    # Remove characters that are not word characters, whitespaces, or selected punctuation.
    text = re.sub(r'[^\w\s,!.?]', '', text)
    return text

def filter_tokens_by_bad_words(tokens, bad_words):
    """
    Filters out tokens that are present in a list of bad words.
    
    Args:
        tokens (list of str): The original list of word tokens.
        bad_words (list of str): The list of words to be filtered out.
        
    Returns:
        list of str: The filtered list of tokens.
    """
    return [token for token in tokens if token.lower() not in bad_words]

def process_comments(input_file_path, output_file_path):
    """
    Processes comments from an input file and writes the processed comments to an output file.
    
    Args:
        input_file_path (str): The path to the input JSON file containing comments.
        output_file_path (str): The path to the output JSON file to write processed comments.
    """
    processed_comments = []
    unigram_counts = Counter()
    bigram_counts = Counter()

    with open(input_file_path, 'r') as file:
        comments_data = json.load(file)

    for item in comments_data:
        comment_text = item.get('comment', '')
        if is_comment_in_english(comment_text):
            normalized_text = normalize_comment_text(comment_text)
            tokens = word_tokenize(normalized_text)
            filtered_tokens = filter_tokens_by_bad_words(tokens, ['badword1', 'badword2'])  # Example bad words list

            # Update unigram and bigram counts with the filtered tokens
            unigram_counts.update(filtered_tokens)
            bigram_counts.update(ngrams(filtered_tokens, 2))

            # Append the processed comment to the list
            processed_comments.append({'comment': ' '.join(filtered_tokens)})

    # Optionally: Implement further filtering based on unigram/bigram frequency here.

    # Write the processed comments to the output file.
    with open(output_file_path, 'w') as file:
        json.dump(processed_comments, file, indent=4)

# Process the comments in the specified input file and save the results to the output file.
process_comments(input_file_path, output_file_path)
