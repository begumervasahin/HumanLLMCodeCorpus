# Import necessary classes
from graph import Vertex as Vertex
from graph import Graph as Graph
from pythonds.graphs import PriorityQueue
import sys

# Define Prim's algorithm
def prim(aGraph, start):
    # Initialize priority queue
    pq = PriorityQueue()

    # Initialize distances and previous vertices for all vertices
    for vertex in aGraph:
        # Set distance to infinity
        vertex.setDistance(sys.maxsize)
        # Set previous vertex to None
        vertex.setPrevious(None)

    # Set distance of the start vertex to 0
    start.setDistance(0)

    # Build heap for priority queue
    # with tuples of (distance, vertex)
    pq.buildHeap([(vertex.getDistance(), vertex) for vertex in aGraph])

    # Main loop
    while not pq.isEmpty():
        # Get the vertex with minimum distance
        currentVertex = pq.delMin()

        # Explore neighbors of the current vertex
        for nextVertex in currentVertex.getConnections():
            # Calculate the new cost to reach the neighbor
            newCost = currentVertex.getWeight(nextVertex)

            # Update distance and previous vertex
            # if a shorter path is found
            if nextVertex in pq and newCost < nextVertex.getDistance():
                nextVertex.setPrevious(currentVertex)
                nextVertex.setDistance(newCost)
                # Update priority queue with the new cost
                pq.decreaseKey(nextVertex, newCost)

# Example usage
if __name__ == "__main__":
    # Create a sample graph
    g = Graph()
    for i in range(6):
        g.addVertex(i)
    # Add edges to the graph with their weights
    g.addEdge(0, 1, 5)
    g.addEdge(0, 5, 2)
    g.addEdge(1, 2, 4)
    g.addEdge(2, 3, 9)
    g.addEdge(3, 4, 7)
    g.addEdge(3, 5, 3)
    g.addEdge(4, 0, 1)
    g.addEdge(5, 4, 8)
    g.addEdge(5, 2, 1)

    # Run Prim's algorithm starting from vertex 0
    prim(g, g.getVertex(0))

    # Output the minimum spanning tree
    for v in g:
        if v.getPrevious():
            print("(%s, %s) -> %s" % (v.getPrevious().getId(), v.getId(), v.getDistance()))
