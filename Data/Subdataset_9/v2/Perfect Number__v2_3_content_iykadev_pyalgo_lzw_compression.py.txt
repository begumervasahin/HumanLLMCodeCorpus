def compress_string(uncompressed):
    """Compress a string using LZW compression."""
    # Initialize the dictionary with ASCII characters
    dictionary_size = 256
    dictionary = {chr(i): chr(i) for i in range(dictionary_size)}
    
    # Initialize variables
    current_string = ""
    compressed_result = []
    
    # Loop through each character in the uncompressed string
    for char in uncompressed:
        # Append the current character to the current string
        new_string = current_string + char
        
        # Check if the new string is in the dictionary
        if new_string in dictionary:
            current_string = new_string
        else:
            # Add the code for the current string to the result
            compressed_result.append(dictionary[current_string])
            
            # Add the new string to the dictionary
            dictionary[new_string] = dictionary_size
            dictionary_size += 1
            
            # Reset the current string
            current_string = char
    
    # Add the code for the last string to the result
    if current_string:
        compressed_result.append(dictionary[current_string])
    
    return compressed_result

def decompress_string(compressed):
    """Decompress a list of compressed codes to a string."""
    # Initialize the dictionary with ASCII characters
    dictionary_size = 256
    dictionary = {chr(i): chr(i) for i in range(dictionary_size)}
    
    # Initialize variables
    decompressed_result = []
    current_string = compressed.pop(0)
    decompressed_result.append(current_string)
    
    # Loop through each code in the compressed list
    for code in compressed:
        if code in dictionary:
            entry = dictionary[code]
        elif code == dictionary_size:
            entry = current_string + current_string[0]
        else:
            raise ValueError('Bad compressed code: %s' % code)
        
        # Add the entry to the decompressed result
        decompressed_result.append(entry)
        
        # Add the new entry to the dictionary
        dictionary[dictionary_size] = current_string + entry[0]
        dictionary_size += 1
        
        # Update the current string
        current_string = entry
    
    return ''.join(decompressed_result)

# Test the compression and decompression functions
compressed = compress_string('ALIALIALIALIALI')
print("Compressed:", compressed)
decompressed = decompress_string(compressed)
print("Decompressed:", decompressed)
