class Heap:
    def __init__(self):
        # Initialize the heap with a list containing a placeholder zero
        self.array = [0]

    def insert(self, key_value):
        # Append the new key value to the heap array
        self.array.append(key_value)
        # Update the first element of the heap array to reflect the current size
        self.array[0] = len(self.array) - 1
        # Fix the heap property by moving the newly inserted element up
        self.heapify_up(self.array[0])

    def heapify_up(self, index):
        # Move the element up the heap until its parent is smaller
        while index > 1:
            parent_index = index // 2
            if int(self.array[parent_index].distance) > int(self.array[index].distance):
                # Swap the parent and the child if the parent is greater
                self.array[index], self.array[parent_index] = self.array[parent_index], self.array[index]
                index = parent_index
            else:
                # Break the loop if the parent is smaller or equal
                break

    def extract_min(self):
        # Get the last index of the heap
        last_index = self.array[0]
        # Retrieve the minimum element from the top of the heap
        ret = self.array[1]
        # Replace the root of the heap with the last element
        self.array[1] = self.array[last_index]
        # Decrease the size of the heap
        self.array[0] -= 1
        # Restore the heap property by moving the root down
        if self.array[0] > 1:
            self.heapify_down(1)
        # Remove the last element from the heap array
        del self.array[-1]
        return ret

    def heapify_down(self, index):
        # Move the element down the heap until its children are greater
        while 2 * index <= self.array[0]:
            left_child_index = 2 * index
            right_child_index = 2 * index + 1 if (2 * index < self.array[0]) else left_child_index
            if self.array[left_child_index].distance < self.array[right_child_index].distance:
                # Choose the smaller child
                smaller_child_index = left_child_index
            else:
                smaller_child_index = right_child_index
            if self.array[smaller_child_index].distance < self.array[index].distance:
                # Swap with the smaller child if it is smaller than the parent
                self.array[index], self.array[smaller_child_index] = self.array[smaller_child_index], self.array[index]
                index = smaller_child_index
            else:
                # Break the loop if the parent is smaller or equal to its children
                break
