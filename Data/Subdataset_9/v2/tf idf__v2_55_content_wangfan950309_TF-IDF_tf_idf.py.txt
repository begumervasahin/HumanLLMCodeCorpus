import pandas as pd
import math
import numpy as np

def search_algorithm(keywords, csv_file_path):
    # Split the search keywords
    search_terms = keywords.split()
    
    # Create a dictionary to store IDF values for each search term
    idf_values = {}
    
    # Read the CSV file containing job data
    job_data = pd.read_csv(csv_file_path, index_col=None)
    
    # Extract the job names from the DataFrame
    job_names = job_data['Job_Name'].tolist()
    
    # Calculate IDF values for each search term
    for term in search_terms:
        term_count = sum(1 for name in job_names if term in name)
        idf_values[term] = math.log(len(job_data) / (1 + term_count))
    
    # Initialize TF-IDF column in DataFrame
    job_data['TF-IDF'] = 0
    
    # Calculate TF-IDF scores for each search term
    for term in search_terms:
        term_tf_values = []
        for name in job_names:
            word_count = len(name.split())
            term_count = sum(1 for word in name.split() if word.lower() == term.lower())
            term_tf_values.append(term_count / word_count)
        job_data[term] = np.array(term_tf_values) * idf_values[term]
        job_data['TF-IDF'] += job_data[term]
    
    # Sort the DataFrame by TF-IDF scores in descending order
    job_data = job_data.sort_values(by='TF-IDF', ascending=False)
    
    # Remove rows with TF-IDF scores equal to 0
    job_data = job_data.loc[job_data['TF-IDF'] != 0]
    
    # Retain only necessary columns
    job_data = job_data.iloc[:, :4]
    
    return job_data

# Example usage
csv_file_path = "path_to_your_data/Job_search.csv"  # Replace with the actual file path
keywords = "data science"  # Example search keywords
result = search_algorithm(keywords, csv_file_path)
print(result)
