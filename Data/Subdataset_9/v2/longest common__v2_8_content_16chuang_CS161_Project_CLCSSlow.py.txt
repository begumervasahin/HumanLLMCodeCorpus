import sys
import numpy as np

# Initialize a 2D array to store results of subproblems
arr = np.zeros((2048, 2048), dtype=int)

# Function to find the length of the longest common subsequence
def LCS(A, B):
    m = len(A)
    n = len(B)
    # Iterate through each pair of characters in A and B
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            # If characters match, extend the LCS
            if A[i - 1] == B[j - 1]:
                arr[i][j] = arr[i - 1][j - 1] + 1
            # If characters don't match, take the maximum length of LCS
            else:
                arr[i][j] = max(arr[i - 1][j], arr[i][j - 1])
    # Return the length of the LCS
    return arr[m][n]

# Function to find the length of the longest common rotated subsequence
def CLCS(A, B):
    # Initialize longest with the length of LCS of A and B
    longest = LCS(A, B)
    # Rotate A and check for longer common subsequences
    for i in range(0, len(A)):
        rotated = A[i:] + A[:i]
        clcs = LCS(rotated, B)
        if clcs > longest:
            longest = clcs
    # Return the length of the longest common rotated subsequence
    return longest

def main():
    # Ensure correct usage of the script
    if len(sys.argv) != 1:
        sys.exit('Usage: `python LCS.py < input`')
    
    # Read input strings line by line from standard input
    for l in sys.stdin:
        A, B = l.split()
        # Compute and print the length of the longest common rotated subsequence
        print(CLCS(A, B))

if __name__ == '__main__':
    main()
