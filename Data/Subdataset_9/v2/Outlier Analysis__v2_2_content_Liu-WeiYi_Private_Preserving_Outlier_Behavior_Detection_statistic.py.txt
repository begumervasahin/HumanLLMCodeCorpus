import sys
import json
from Process_TianChi_Main import analysis_Time  # Importing a custom function from external module
from utils import *  # Importing utilities

def main():
    # Read user IDs from a file
    users = []
    with open('all_user_id.txt', 'r+') as f:
        users = [line.strip() for line in f.readlines()]

    # Get the choice from the command line argument
    choose = sys.argv[1]

    # Option 0: Analyze devices group
    if choose == '0':
        print('Analyzing devices group...')
        with open('all_user_info.json') as f:
            all_user_info = json.load(f)
            devices = set()
            count = 0
            # Iterate through each user's information
            for user in all_user_info.keys():
                count += 1
                # Calculate and display processing progress
                percentage = 100 * count / len(all_user_info.keys())
                sys.stdout.write('\r>> Processing Users............ %.2f %%' % percentage)
                sys.stdout.flush()
                info = all_user_info[user]["Reach_Time"]
                # Extract devices from each user's reach time data
                for time in info.keys():
                    current_devices = info[time].keys()
                    for d in current_devices:
                        devices.add(d)
        num_devices = len(devices)
        print('\nTotal number of devices:', num_devices)

    # Option 1: Analyze time group
    elif choose == '1':
        print('Analyzing time group...')
        with open('user_time_info.json') as f:
            user_time = json.load(f)
            all_time = 0
            count = 0
            # Iterate through each user's time information
            for user in user_time.keys():
                count += 1
                # Calculate and display processing progress
                percentage = 100 * count / len(user_time.keys())
                sys.stdout.write('\r>> Processing Users............ %.2f %%' % percentage)
                sys.stdout.flush()
                per_user_info = user_time[user]
                _, Dates_Interval_Dict = analysis_Time(user, per_user_info)
                all_time += len(Dates_Interval_Dict.keys())
        print('\nTotal time groups:', all_time)

    # Option 2: Analyze features
    elif choose == '2':
        print('Analyzing features...')
        with open('all_user_info.json') as f:
            all_user_info = json.load(f)
            all_keyword = 0
            avg_keyword = 0
            all_IP = 0
            avg_IP = 0
            count = 0
            # Iterate through each user's information
            for user in all_user_info.keys():
                count += 1
                # Calculate and display processing progress
                percentage = 100 * count / len(all_user_info.keys())
                sys.stdout.write('\r>> Processing Users............ %.2f %%' % percentage)
                sys.stdout.flush()
                info = all_user_info[user]["Reach_Time"]
                # Extract keywords and IP addresses from each user's reach time data
                for time in info.keys():
                    for user in info[time].keys():
                        all_keyword += len(info[time][user]["Keyword"])
                        all_IP += len(info[time][user]["Device_IP"])
            avg_keyword = all_keyword / len(all_user_info.keys())
            avg_IP = all_IP / len(all_user_info.keys())
        print('\nAverage number of keywords:', avg_keyword)
        print('Average number of IP addresses:', avg_IP)

if __name__ == "__main__":
    main()
