# Initialize a dictionary to store Fibonacci calculations for caching
fibonacci_cache = {}

# Define a function to calculate Fibonacci numbers with caching
def fibonacci(n):
    # Check if the input is an integer
    if not isinstance(n, int):
        return False
    
    # Check if the Fibonacci number is already calculated and cached
    if n in fibonacci_cache:
        return fibonacci_cache[n]
    
    # Base cases for Fibonacci sequence
    if n == 0 or n == 1:
        return 1
    
    # Calculate Fibonacci number recursively and store it in cache
    else:
        result = fibonacci(n - 1) + fibonacci(n - 2)
        fibonacci_cache[n] = result
        return result

# Print the first 1000 Fibonacci numbers
for n in range(1000):
    print(f"{n}: {fibonacci(n)}")
