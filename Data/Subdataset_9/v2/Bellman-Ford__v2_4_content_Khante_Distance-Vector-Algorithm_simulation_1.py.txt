import network_1
import link_1
import threading
from time import sleep

router_queue_size = 0
simulation_time = 2

if __name__ == '__main__':
    # Create a list to hold all network objects
    object_list = []

    # Create two hosts
    host_1 = network_1.Host('H1')
    object_list.append(host_1)

    host_2 = network_1.Host('H2')
    object_list.append(host_2)

    # Configure routers and add them to the object list
    cost_d_ra = {'H1': {0: 1}, 'RB': {1: 1}}
    router_a = network_1.Router(name='RA', cost_D=cost_d_ra, max_queue_size=router_queue_size)
    object_list.append(router_a)

    cost_d_rb = {'H2': {1: 3}, 'RA': {0: 1}}
    router_b = network_1.Router(name='RB', cost_D=cost_d_rb, max_queue_size=router_queue_size)
    object_list.append(router_b)

    # Create a link layer
    link_layer = link_1.LinkLayer()
    object_list.append(link_layer)

    # Add links to connect hosts and routers
    link_layer.add_link(link_1.Link(host_1, 0, router_a, 0))
    link_layer.add_link(link_1.Link(router_a, 1, router_b, 0))
    link_layer.add_link(link_1.Link(router_b, 1, host_2, 0))

    # Create threads for each network object and start them
    thread_list = []
    for obj in object_list:
        thread_list.append(threading.Thread(name=obj.__str__(), target=obj.run))

    for thread in thread_list:
        thread.start()

    # Send routes from router A and wait for convergence
    router_a.send_routes(1)
    sleep(simulation_time)

    # Print converged routing tables
    print("Converged routing tables")
    for obj in object_list:
        if isinstance(obj, network_1.Router):
            obj.print_routes()

    # Send a message from host 1 to host 2 and wait
    host_1.udt_send('H2', 'MESSAGE_FROM_H1')
    sleep(simulation_time)

    # Stop all threads
    for obj in object_list:
        obj.stop = True

    for thread in thread_list:
        thread.join()

    print("All simulation threads joined")
