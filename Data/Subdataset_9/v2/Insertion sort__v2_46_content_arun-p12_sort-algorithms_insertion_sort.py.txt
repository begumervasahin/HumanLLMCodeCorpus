def insertion_sort(A, verbose=0, desc=0):
    """
    Sorts a list of integers using the insertion sort algorithm.

    Args:
        A (list): The unsorted list of integers.
        verbose (int, optional): Controls verbosity level. Defaults to 0.
        desc (int, optional): Controls whether the output is sorted in descending order. Defaults to 0.

    Returns:
        list: The sorted list.
    """
    for i in range(1, len(A)):
        # Select the current element as the key
        key = A[i]
        # Initialize a pointer to the element just before the key
        j = i - 1
        # Iterate backwards through the sorted portion of the list
        while j >= 0 and A[j] > key:
            # Shift elements to the right until finding the correct position for the key
            A[j + 1] = A[j]
            j -= 1
            if verbose == 2:
                print("Shifting:", j, "::", A)
        # Place the key at its correct position in the sorted portion of the list
        A[j + 1] = key
        if verbose:
            print("Iteration", i, ":", A)
    # Optionally reverse the list if descending order is requested
    if desc:
        A = A[::-1]
    return A
