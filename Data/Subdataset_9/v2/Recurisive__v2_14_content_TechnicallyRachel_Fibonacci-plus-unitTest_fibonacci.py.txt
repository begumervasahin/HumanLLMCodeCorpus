def fibonacci_sequence(n):
    """Generate the Fibonacci sequence up to a given limit n."""
    result = []
    a, b = 0, 1
    while b < n:
        result.append(b)
        a, b = b, a + b
    return result

def main():
    """Main function to demonstrate the Fibonacci sequence generation."""
    print(fibonacci_sequence(10))

if __name__ == '__main__':
    main()
