def longest_common_subsequence(a, b):
    # Initialize matrix and maximum length
    matrix = []
    max_length = 0
    
    # Initialize the matrix
    for i in range(len(a)+1):
        row = []
        for j in range(len(b)+1):
            if i == 0 or j == 0:
                # Initialize the first row and column
                row.append('0H')  # '0H' represents zero value with a 'horizontal' direction
            else:
                if a[i-1] != b[j-1]:
                    # If characters don't match, choose the maximum length from either left or above cell
                    top = matrix[i-1][j]
                    left = row[j-1]
                    if int(top[:-1]) >= int(left[:-1]):
                        row.append('{0}U'.format(top[:-1]))  # 'U' represents 'up'
                    else:
                        row.append('{0}S'.format(left[:-1]))  # 'S' represents 'side'
                else:
                    # If characters match, increment the diagonal cell value by 1
                    diagonal = matrix[i - 1][j - 1]
                    row.append('{0}D'.format(str(int(diagonal[:-1]) + 1)))  # 'D' represents 'diagonal'
                    if max_length < int(diagonal[:-1]) + 1:
                        max_length = int(diagonal[:-1]) + 1
        matrix.append(row)
    
    return matrix, max_length

def print_lcs(a, c, i, j):
    lcs_list = []
    x = c[i][j]
    
    if i == 0 or j == 0:
        # Base case: reached end of one of the strings
        return "".join(lcs_list[::-1])  # Return the LCS sequence in reverse order
    
    if x[-1] == 'D':
        # If the direction is diagonal, add the character to LCS list and move diagonally up
        lcs_list.append(a[i-1])
        return print_lcs(a, c, i-1, j-1)
    elif x[-1] == 'U':
        # If the direction is up, move up
        return print_lcs(a, c, i-1, j)
    else:
        # If the direction is side, move left
        return print_lcs(a, c, i, j-1)

# Input strings
a = input("Enter 1st string:")
b = input("Enter 2nd string:")

# Get LCS matrix and length
matrix, length = longest_common_subsequence(a, b)

# Print the LCS matrix
for i in range(len(a)+1):
    for j in range(len(b)+1):
        print(matrix[i][j]+" ", end="")
    print()

# Print length of the LCS
print("Length of the longest common subsequence is:", length)

# Get and print the LCS sequence
lcs_sequence = print_lcs(a, matrix, len(a), len(b))
print("Longest common subsequence is:", lcs_sequence)

# Calculate percent difference
max_length = max(len(a), len(b))
percent_difference = ((max_length - length) / max_length) * 100

# Check if signature matches or not based on percent difference
print("Percent difference is: %.2f%%" % percent_difference)
if percent_difference > 7:
    print("Signature mismatch")
else:
