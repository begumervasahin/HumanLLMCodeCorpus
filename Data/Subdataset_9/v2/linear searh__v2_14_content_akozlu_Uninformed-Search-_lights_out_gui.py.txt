import sys
import tkinter as tk
import random

# Custom tkinter Canvas class representing a light
class Light(tk.Canvas):
    BACKGROUND_ON = "white"
    BACKGROUND_OFF = "gray50"
    BORDER_ON = "red"
    BORDER_OFF = "black"

    def __init__(self, master, size=60):
        tk.Canvas.__init__(self, master, height=size, width=size, background=Light.BACKGROUND_OFF,
                           highlightthickness=2, highlightbackground=Light.BORDER_OFF)

    # Set the state of the light (on or off)
    def set_state(self, state):
        color = Light.BACKGROUND_ON if state else Light.BACKGROUND_OFF
        self.configure(background=color)

    # Set the selected state of the light
    def set_selected(self, selected):
        color = Light.BORDER_ON if selected else Light.BORDER_OFF
        self.configure(highlightbackground=color)

# Custom tkinter Frame class representing the game board
class Board(tk.Frame):
    def __init__(self, master, puzzle, rows, cols):
        tk.Frame.__init__(self, master)
        self.puzzle = puzzle
        self.rows = rows
        self.cols = cols
        self.lights = []

        # Create lights on the board
        for row in range(self.rows):
            row_lights = []
            for col in range(self.cols):
                light = Light(self)
                light.grid(row=row, column=col, padx=1, pady=1)
                light.bind("<Button-1>", lambda event, row=row, col=col: self.click(row, col))
                row_lights.append(light)
            self.lights.append(row_lights)

    # Handle click event on a light
    def click(self, row, col):
        self.puzzle.perform_move(row, col)
        self.update_lights()

    # Update the lights on the board based on the current state of the puzzle
    def update_lights(self):
        puzzle_board = self.puzzle.get_board()
        for row in range(self.rows):
            for col in range(self.cols):
                self.lights[row][col].set_state(puzzle_board[row][col])

    # Animate moves on the board
    def animate_moves(self, moves, delay=500):
        if moves:
            row, col = moves[0]

            # Define stages of animation
            def stage_1():
                self.lights[row][col].set_selected(True)
                self.after(delay, stage_2)

            def stage_2():
                self.lights[row][col].set_selected(False)
                self.puzzle.perform_move(row, col)
                self.update_lights()
                self.after(delay, stage_3)

            def stage_3():
                self.animate_moves(moves[1:], delay=delay)

            stage_1()

# Custom tkinter Frame class representing the Lights Out game GUI
class LightsOutGUI(tk.Frame):
    def __init__(self, master, rows, cols):
        tk.Frame.__init__(self, master)
        self.puzzle = LightsOutPuzzle(rows, cols)
        self.board = Board(self, self.puzzle, rows, cols)
        self.board.pack(side=tk.LEFT, padx=1, pady=1)

        # Create menu buttons
        menu = tk.Frame(self)
        tk.Button(menu, text="Scramble", command=self.scramble_click).pack(fill=tk.X, padx=1, pady=1)
        tk.Button(menu, text="Solve", command=self.solve_click).pack(fill=tk.X, padx=1, pady=1)
        menu.pack(side=tk.RIGHT)

    # Handle click event on the scramble button
    def scramble_click(self):
        self.puzzle.scramble()
        self.board.update_lights()

    # Handle click event on the solve button
    def solve_click(self):
        self.board.animate_moves(self.puzzle.find_solution())

# Class representing the Lights Out puzzle
class LightsOutPuzzle:
    def __init__(self, rows, cols):
        self.rows = rows
        self.cols = cols
        self.board = [[False] * cols for _ in range(rows)]

    # Perform a move on the puzzle
    def perform_move(self, row, col):
        self.toggle(row, col)
        if row > 0:
            self.toggle(row - 1, col)
        if row < self.rows - 1:
            self.toggle(row + 1, col)
        if col > 0:
            self.toggle(row, col - 1)
        if col < self.cols - 1:
            self.toggle(row, col + 1)

    # Toggle the state of a light
    def toggle(self, row, col):
        self.board[row][col] = not self.board[row][col]

    # Get the current state of the board
    def get_board(self):
        return self.board

    # Scramble the board
