from heapq import heappush, heappop, heapify
from collections import defaultdict

def prims_minimum_spanning_tree(graph):
    num_vertices = len(graph)
    visited = set()  # Set to keep track of visited vertices
    source_vertex = next(iter(graph.keys()))  # Pick the first vertex as the source
    visited.add(source_vertex)
    source_connections = list(graph[source_vertex])  # Get the connections of the source vertex
    edge_costs = source_connections
    heapify(edge_costs)  # Convert the list of connections into a heap
    selected_nodes = [source_vertex]  # List to keep track of selected vertices in MST
    total_cost = 0  # Variable to store the total cost of the MST

    while len(selected_nodes) != num_vertices:
        # Get the next edge with the minimum cost
        next_edge = heappop(edge_costs)
        if next_edge[1] in visited:
            continue
        total_cost += next_edge[0]  # Add the cost of the edge to the total cost
        added_vertex = next_edge[1]
        selected_nodes.append(added_vertex)
        visited.add(added_vertex)
        # Explore the connections of the added vertex
        for edge in graph[added_vertex]:
            if edge[1] not in visited:
                heappush(edge_costs, edge)  # Add the unvisited edges to the heap

    return selected_nodes, total_cost

def read_graph_from_file(filename):
    graph = defaultdict(set)
    with open(filename, 'r') as data_file:
        next(data_file)  # Skip the first line
        for line in data_file:
            vals = line.split()
            # Add the edge and its cost to the graph
            graph[vals[0]].add((int(vals[2]), vals[1]))
            graph[vals[1]].add((int(vals[2]), vals[0]))
    return graph

if __name__ == "__main__":
    graph = read_graph_from_file('edges.txt')
    mst_nodes, total_cost = prims_minimum_spanning_tree(graph)
    print("Minimum Spanning Tree Nodes:", mst_nodes)
    print("Total Cost:", total_cost)
