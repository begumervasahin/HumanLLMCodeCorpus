import random

def get_odd_number(start, end):
    rand_num = random.randrange(start, end)
    if rand_num % 2 == 0:
        rand_num += 1
    return rand_num

def is_composite(n, a, r, s):
    res = pow(a, r, n)
    if res == 1 or res == n - 1:
        return False
    for _ in range(int(s)):
        res = pow(res, 2, n)
        if res == n - 1:
            return False
    return True

def is_prime(n):
    r = n - 1
    s = 0
    while r % 2 == 0:
        r //= 2
        s += 1
    limit = (n - 1) // 4
    for _ in range(limit):
        a = random.randint(2, n - 1)
        if is_composite(n, a, r, s):
            return False
    return True

def generate_prime_number(start, end):
    while True:
        num = get_odd_number(start, end)
        if is_prime(num):
            return num

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def multiplicative_inverse(e, phi):
    temp_phi = phi
    y, x = 0, 1
    while e > 1:
        quotient = e // temp_phi
        temp_phi, e = e % temp_phi, temp_phi
        y, x = x - quotient * y, y
    return x % phi

# Example usage:
lower_bound = 10000
upper_bound = 100000

p = generate_prime_number(lower_bound, upper_bound)
q = generate_prime_number(lower_bound, upper_bound)
n = p * q
phi = (p - 1) * (q - 1)

e = random.randint(2, phi - 1)
while gcd(e, phi) != 1:
    e = random.randint(2, phi - 1)

d = multiplicative_inverse(e, phi)

print("p:", p)
print("q:", q)
print("n:", n)
print("phi:", phi)
print("e:", e)
print("d:", d)
