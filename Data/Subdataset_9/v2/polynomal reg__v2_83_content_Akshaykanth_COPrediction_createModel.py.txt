import pandas as pd
from sklearn.preprocessing import PolynomialFeatures
from sklearn.linear_model import LinearRegression
from sklearn.model_selection import train_test_split

# Data Preparation
print("Step 1: Data Preprocessing")
data = pd.read_csv("coEmission.csv")  # Load the data from the CSV file
x = data.iloc[:, 0:3].values  # Extract features (Year, Temp, Fuel)
y = data.iloc[:, 3].values  # Extract target variable (CO Emission)
x_train, x_test, y_train, y_test = train_test_split(x, y, test_size=0.2, random_state=0)  # Split the data into training and testing sets

# Model Building
print("Step 2: Model Creation")
poly_reg = PolynomialFeatures(degree=4)  # Create a polynomial feature transformer
x_poly = poly_reg.fit_transform(x_train)  # Transform the features into polynomial features
pol_reg = LinearRegression()  # Create a linear regression model

# Model Training
print("Step 3: Training Started")
pol_reg.fit(x_poly, y_train)  # Train the linear regression model using polynomial features
print("Training Ended")

# User Interaction
print("\nPredict CO Emission (Enter Year, Temp, Fuel) or type 'quit' to exit")
while True:
    user_input = input("Year: ")

    if user_input.lower() == 'quit':
        print("Exiting...")
        break

    year = int(user_input) - 2005
    temp = int(input("Temp: "))
    fuel = int(input("Fuel: "))

    # Make a prediction based on user input
    params = [temp, fuel, year]
    co_emission = pol_reg.predict(poly_reg.fit_transform([params]))[0]

    # Display the predicted CO emission
    print("Predicted CO emission:", co_emission, "grams")
