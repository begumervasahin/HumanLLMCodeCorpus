from feature_type import Satisfaction

class NaiveBayesClassifier:
    def __init__(self, sensor_name, feature_weights, satisfaction_threshold):
        self.sensor_name = sensor_name
        self.feature_weights = feature_weights
        self.satisfaction_threshold = satisfaction_threshold
        self.satisfactions = []
        self.interaction_count = 0
        self.successful_interactions = 0
        self.reputation_history = {}
        self.simulation_steps = 0

    def add_satisfaction(self, satisfaction):
        self.satisfactions.append(satisfaction)
        self.reputation_history[satisfaction.feature_type] = []

    def compute(self, data):
        self.interaction_count += 1
        satisfaction_score = sum(self.feature_weights[satisfaction.feature_type] * satisfaction.get_satisfaction(data) for satisfaction in self.satisfactions)
        
        if satisfaction_score >= self.satisfaction_threshold:
            self.successful_interactions += 1
            for satisfaction in self.satisfactions:
                if satisfaction.get_satisfaction(data) >= self.feature_weights[satisfaction.feature_type] * self.satisfaction_threshold:
                    self.reputation_history[satisfaction.feature_type].append(self.reputation(satisfaction.feature_type))

    def reputation(self, feature):
        if self.interaction_count == 0 or self.successful_interactions == 0:
            return (0, 0)
        trust_probability = self.successful_interactions / self.interaction_count
        feature_probability = sum(1 for satisfaction in self.satisfactions if satisfaction.feature_type == feature and satisfaction.get_satisfaction(data) >= self.feature_weights[satisfaction.feature_type] * self.satisfaction_threshold) / self.interaction_count
        return (feature_probability / trust_probability, trust_probability)

    def save_reputation(self):
        self.simulation_steps += 1

    def get_reputation_history(self, features):
        reputation_history = []
        for i in range(self.simulation_steps):
            reputation_score = self.reputation_history[features[0]][i][1]
            for feature in features:
                reputation_score *= self.reputation_history[feature][i][0]
            reputation_history.append(reputation_score)
        return reputation_history

# Example usage
if __name__ == "__main__":
    sensor_name = "example_sensor"
    feature_weights = {"temperature": 0.5, "humidity": 0.3, "pressure": 0.2}
    satisfaction_threshold = 0.7

    nb_classifier = NaiveBayesClassifier(sensor_name, feature_weights, satisfaction_threshold)

    # Assuming you have some satisfaction objects created elsewhere
    satisfaction1 = Satisfaction("temperature", ...)
    satisfaction2 = Satisfaction("humidity", ...)
    satisfaction3 = Satisfaction("pressure", ...)

    nb_classifier.add_satisfaction(satisfaction1)
    nb_classifier.add_satisfaction(satisfaction2)
    nb_classifier.add_satisfaction(satisfaction3)

    # Assuming you have some data to compute with
    data = {"temperature": ..., "humidity": ..., "pressure": ...}

    nb_classifier.compute(data)

    # Save reputation history after some iterations
    nb_classifier.save_reputation()

    # Get reputation history for specific features
    reputation_history = nb_classifier.get_reputation_history(["temperature", "humidity"])
    print(reputation_history)
