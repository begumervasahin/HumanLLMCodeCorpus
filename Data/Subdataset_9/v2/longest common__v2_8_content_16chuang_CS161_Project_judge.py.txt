import sys

def verify(actual, expected):
    # Initialize a counter for failed test cases
    failed = 0
    
    # Iterate through each index in the lists of actual and expected outputs
    for i in range(len(actual)):
        # Compare the actual output with the expected output at the same index
        if actual[i] != expected[i]:
            # If they are different, print a message indicating the failure
            print(f"Test case {i+1} failed: Expected {expected[i]}, but got {actual[i]}")
            failed += 1  # Increment the counter for failed test cases
    
    # Print a separator line
    print("---------------------------------")
    
    # Print a message indicating whether all test cases passed or some failed
    if failed == 0:
        print("Passed all test cases!")
    else:
        print(f"Failed {failed}/{len(actual)} test cases.")
    
    # Print another separator line
    print("---------------------------------")

def main():
    # Check if the correct number of command-line arguments is provided
    if len(sys.argv) != 2:
        sys.exit("Usage: `python judge.py EXPECTED < ACTUAL")
    
    # Read the actual output from standard input
    actual = [int(l) for l in sys.stdin.readlines()]
    
    # Initialize a variable to store the expected output
    expected = None
    
    # Open the file containing the expected output specified in the command-line argument
    with open(sys.argv[1], 'r') as f:
        # Read the expected output from the file
        expected = [int(l) for l in f]
    
    # Check if the lengths of the actual and expected outputs are the same
    if len(actual) != len(expected):
        sys.exit("Error: The expected and actual outputs have different lengths")
    
    # Verify the actual output against the expected output
    verify(actual, expected)

# Check if the script is being run as the main program
if __name__ == '__main__':
    # Call the main function
    main()
