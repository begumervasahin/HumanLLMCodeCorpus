from typing import Tuple

def modulo_multiply(a: int, b: int, mod: int) -> int:
    """
    Evaluates (a * b) % mod.
    
    Args:
        a (int): The first operand.
        b (int): The second operand.
        mod (int): The modulo value.
    
    Returns:
        int: The result of (a * b) % mod.
    
    Raises:
        Exception: If mod is 0.
    """
    if mod == 0:
        raise Exception("Divide by zero error")
    return ((a % mod) * (b % mod)) % mod

def modulo_pow(a: int, b: int, mod: int) -> int:
    """
    Evaluates (a^b) % mod.
    
    Args:
        a (int): The base.
        b (int): The exponent.
        mod (int): The modulo value.
    
    Returns:
        int: The result of (a^b) % mod.
    """
    result = 1
    while b:
        result = modulo_multiply(result, a, mod)
        b -= 1
    return result % mod

def egcd(a: int, b: int) -> Tuple[int, int, int]:
    """
    Extended Euclidean algorithm to compute the gcd.
    
    Args:
        a (int): The first number.
        b (int): The second number.
    
    Returns:
        Tuple[int, int, int]: A tuple (g, x, y) where a*x + b*y = gcd(x, y).
    """
    if a == 0:
        return (b, 0, 1)
    g, x, y = egcd(b % a, a)
    return (g, y - (b // a) * x, x)

def modulo_div(a: int, b: int, mod: int) -> int:
    """
    Evaluates (a / b) % mod.
    
    Args:
        a (int): The dividend.
        b (int): The divisor.
        mod (int): The modulo value.
    
    Returns:
        int: The result of (a / b) % mod.
    
    Raises:
        Exception: If the modular inverse does not exist.
    """
    return modulo_multiply(a, mulinv(b, mod), mod)

def mulinv(b: int, n: int) -> int:
    """
    Multiplicative inverse of b modulo n.
    
    Args:
        b (int): The number.
        n (int): The modulo value.
    
    Returns:
        int: The multiplicative inverse of b modulo n.
    
    Raises:
        Exception: If the modular inverse does not exist.
    """
    g, x, _ = egcd(b, n)
    if g == 1:
        return x % n
    raise Exception("Modular Inverse does not exist")
