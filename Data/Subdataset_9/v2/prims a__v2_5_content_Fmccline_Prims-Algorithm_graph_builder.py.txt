import json
import networkx as nx

class GraphBuilder:
    """ Helper class for working with networkx graphs
    """

    @staticmethod
    def build_graph_from_file(filename):
        """ Creates a graph from a given JSON file
        :param filename: The JSON source file of the graph
        :return: A networkx graph made from the given file
        """

        # Create an empty graph
        graph = nx.Graph()

        # Open the JSON file and load data
        with open(filename, 'r') as file:
            raw_data = json.load(file)

        # Extract node information from the JSON data
        nodes = raw_data["nodes"]

        # Iterate over each node in the JSON data
        for node in nodes:
            # Extract node name, neighbors, and weights
            name = node["name"]
            neighbors = node["neighbors"]
            weights = node["weights"]

            # Iterate over each neighbor and its corresponding weight
            for edge in range(len(neighbors)):
                # Extract neighbor and weight for the current edge
                neighbor = neighbors[edge]
                weight = weights[edge]

                # Add the node and edge to the graph
                graph.add_node(name)  # Adding the current node
                graph.add_edge(name, neighbor, weight=weight)  # Adding the edge

        # Return the constructed graph
        return graph

# Example usage:
if __name__ == "__main__":
    filename = "graph_data.json"  # Replace with your JSON file path  
    graph = GraphBuilder.build_graph_from_file(filename)
    print("Nodes:", graph.nodes)
    print("Edges:", graph.edges)
