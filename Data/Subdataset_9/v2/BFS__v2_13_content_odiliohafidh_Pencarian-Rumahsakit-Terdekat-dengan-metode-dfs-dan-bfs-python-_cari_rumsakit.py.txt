def bfs(graph, start):
    """
    Performs Breadth-First Search (BFS) traversal on the given graph starting from the specified start node.
    Returns the shortest path to any of the target nodes (RS1, RS2, RS3).
    """
    queue = [[start]]  # Initialize queue with the start node
    visited = set()  # Set to keep track of visited nodes
    targets = {'RS1', 'RS2', 'RS3'}  # Set of target nodes

    while queue:
        path = queue.pop(0)  # Dequeue the first path from the queue
        node = path[-1]  # Get the last node from the path

        if node in targets:
            return path  # Return the path if the node is a target

        if node not in visited:
            for neighbor in graph.get(node, []):  # Iterate through neighbors of the current node
                new_path = list(path)  # Create a new path by copying the current path
                new_path.append(neighbor)  # Append the neighbor to the new path
                queue.append(new_path)  # Enqueue the new path

            visited.add(node)  # Mark the current node as visited

    return None  # Return None if no path to targets is found


def dfs(graph, start):
    """
    Performs Depth-First Search (DFS) traversal on the given graph starting from the specified start node.
    Returns the path to any of the target nodes (RS1, RS2, RS3).
    """
    stack = [[start]]  # Initialize stack with the start node
    visited = set()  # Set to keep track of visited nodes
    targets = {'RS1', 'RS2', 'RS3'}  # Set of target nodes

    while stack:
        path = stack.pop()  # Pop the top path from the stack
        node = path[-1]  # Get the last node from the path

        if node in targets:
            return path  # Return the path if the node is a target

        if node not in visited:
            for neighbor in graph.get(node, []):  # Iterate through neighbors of the current node
                new_path = list(path)  # Create a new path by copying the current path
                new_path.append(neighbor)  # Append the neighbor to the new path
                stack.append(new_path)  # Push the new path onto the stack

            visited.add(node)  # Mark the current node as visited

    return None  # Return None if no path to targets is found
