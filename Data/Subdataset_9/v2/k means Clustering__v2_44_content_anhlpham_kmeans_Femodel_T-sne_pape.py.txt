# Importing necessary libraries
import numpy as np
import matplotlib.pyplot as plt
from sklearn import manifold
from scipy.io import loadmat
import seaborn as sns

# Load data from the file 'Parekh_all2.mat'
data = loadmat('Parekh_all2.mat')
Xn = data['Xn']
IDX_new = data['IDX_new']
IDX = np.reshape(IDX_new, (35885))

# Define target classes
target = np.array([1, 2, 3, 4])

# Define perplexities for t-SNE
perplexities = [5, 30, 50, 100]

# Initialize t-SNE with specified parameters
tsne = manifold.TSNE(n_components=2, init='random', random_state=0)

# Perform t-SNE embedding
Y = tsne.fit_transform(Xn)

# Set up seaborn for better visual aesthetics
sns.set(style='white', context='notebook', rc={'figure.figsize':(14,10)})

# Plot t-SNE embedding
plt.scatter(Y[:, 0], Y[:, 1], c=[sns.color_palette()[x] for x in IDX])
plt.gca().set_aspect('equal', 'datalim')  # Fix aspect ratio
plt.title('t-SNE Projection of sFe-all Model', fontsize=24)
plt.savefig('tsne-sFe-all.pdf')  # Save the plot as a PDF file
plt.show()  # Display the plot
