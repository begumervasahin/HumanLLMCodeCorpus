import scrapy
import json
from bs4 import BeautifulSoup
from bs4.element import Comment
import urllib.request

# Function to determine if a tag's text is visible
def tag_visible(element):
    # Exclude certain elements and comments
    if element.parent.name in ['style', 'script', 'head', 'title', 'meta', '[document]']:
        return False
    if isinstance(element, Comment):
        return False
    return True

# Function to extract visible text from HTML body
def text_from_html(body):
    # Parse HTML using BeautifulSoup
    soup = BeautifulSoup(body, 'html.parser')
    # Find all text elements
    texts = soup.findAll(text=True)
    # Filter visible text elements
    visible_texts = filter(tag_visible, texts)
    # Join visible text elements into a single string
    return " ".join(t.strip() for t in visible_texts)

# Spider class for scraping Monster.com job listings
class MonsterSpiderSpider(scrapy.Spider):
    # Spider name
    name = 'monster-spider'
    # Allowed domains to crawl
    allowed_domains = ['monster.com']
    # Start URLs for crawling (2 pages)
    start_urls = ['https://www.monster.com/jobs/search/pagination/?q=Product-Manager&where=USA&isDynamicPage=true&isMKPagination=true&page={}'.format(i + 1) for i in range(2)]
    
    # Parse method for handling response
    def parse(self, response):
        # Parse JSON response
        results = json.loads(response.body)
        # Iterate over job data
        for result in results['data']:
            try:
                # Extract job ID
                job_id = result['MusangKingId']
                # Construct URL for detailed job view
                next_url = 'https://job-openings.monster.com/v2/job/pure-json-view?jobid={}'.format(job_id)
                # Follow link to detailed job view and call parse_detail method
                yield response.follow(next_url, callback=self.parse_detail)
            except Exception as e:
                # Log error if encountered
                self.logger.error('Error parsing job details: %s', e)
    
    # Method to parse detailed job view
    def parse_detail(self, response):
        # Parse JSON response
        result = json.loads(response.body)
        # Initialize detail dictionary for storing job details
        detail = {
            "JobID": result.get('jobId', ''),
            "Title": result['companyInfo'].get('companyHeader', ''),
            "Location": result['companyInfo'].get('jobLocation', ''),
            "Company": result['companyInfo'].get('name', ''),
            "Job Category": result.get("jobCategory", ''),
            "Description": text_from_html(result.get("jobDescription", '')),
            "Company Size": result['companyInfo'].get("companySizeName", '')
        }
        # Extract additional info
        info = result['summary'].get('info', [])
        for i in info:
            detail[i['title']] = i['text']
        # Yield job details
        yield detail
