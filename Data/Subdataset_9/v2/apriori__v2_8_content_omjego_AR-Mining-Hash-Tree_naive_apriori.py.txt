import itertools
import csv

# Parameters class for support and confidence

# Load data from a CSV file
def load_data(filename, max_attr=100):
    """
    Loads data from a CSV file.
    """
    with open(filename, 'r') as f:
        data = list(csv.reader(f, delimiter=','))
    
    result = []
    for i in range(len(data)):
        attrbs = []
        for val in data[i][1:]:
            val = val.strip()
            if int(val) <= max_attr:
                attrbs.append(val)
        if len(attrbs) > 0:
            result.append(attrbs)
    
    return result

# Find all subsets of a set with size m
def find_subsets(S, m):
    """
    Finds all subsets of size m from set S.
    """
    return set(itertools.combinations(S, m))

# Check if a dataset has any infrequent subset
def has_infrequent_subset(dataset, prev_l, k):
    """
    Checks if a dataset has any infrequent subset.
    """
    subsets = find_subsets(dataset, k)
    for item in subsets:
        s = []
        for l in item:
            s.append(l)
        s.sort()
        if s not in prev_l:
            return True
    return False

# Generate candidate itemsets of size k using the Apriori algorithm
def apriori(prev_l, k):
    """
    Generates candidate itemsets of size k using the Apriori algorithm.
    """
    length = k
    result = []
    for list1 in prev_l:
        for list2 in prev_l:
            count = 0
            c = []
            if list1 != list2:
                while count < length - 1:
                    if list1[count] != list2[count]:
                        break
                    else:
                        count += 1
                else:
                    if list1[length - 1] < list2[length - 1]:
                        for item in list1:
                            c.append(item)
                        c.append(list2[length - 1])
                        if not has_infrequent_subset(c, prev_l, k):
                            result.append(c)
    return result
