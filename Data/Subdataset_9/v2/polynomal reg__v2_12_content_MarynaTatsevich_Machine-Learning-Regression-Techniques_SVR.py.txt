# Importing necessary libraries
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from sklearn.preprocessing import StandardScaler
from sklearn.svm import SVR

# Step 1: Load the dataset
dataset = pd.read_csv('Position_Salaries.csv')

# Extracting features (X) and target variable (y) from the dataset
X = dataset.iloc[:, 1:2].values
y = dataset.iloc[:, 2].values

# Step 2: Feature Scaling
# Standardize features by removing the mean and scaling to unit variance
sc_X = StandardScaler()
sc_y = StandardScaler()
X = sc_X.fit_transform(X)
y = sc_y.fit_transform(y.reshape(-1, 1)).reshape(-1)

# Step 3: Building the SVR model
# Using Support Vector Regression with Radial Basis Function (RBF) kernel
regressor = SVR(kernel='rbf')
regressor.fit(X, y)

# Step 4: Making a prediction for a new data point
# Inverse transforming the scaled prediction to get the original scale
scaled_prediction = regressor.predict(sc_X.transform(np.array([[6.5]])))
prediction = sc_y.inverse_transform(scaled_prediction.reshape(-1, 1))[0][0]
print("Predicted salary for level 6.5:", prediction)

# Step 5: Visualizing the SVR results
# Scatter plot of the original data points
plt.scatter(X, y, color='red')
# Plotting the SVR regression line
plt.plot(X, regressor.predict(X), color='blue')
plt.title('Truth or Bluff (SVR)')
plt.xlabel('Position level')
plt.ylabel('Salary')
plt.show()

# Step 6: Visualizing the SVR results with higher resolution
# Generating a smoother curve by using a smaller interval
X_grid = np.arange(min(X), max(X), 0.01)
X_grid = X_grid.reshape((len(X_grid), 1))
plt.scatter(X, y, color='red')
plt.plot(X_grid, regressor.predict(X_grid), color='blue')
plt.title('Truth or Bluff (SVR)')
plt.xlabel('Position level')
plt.ylabel('Salary')
plt.show()
