def binary_search(input_array, value):
    # Initialize upper and lower bounds for the search
    upper_bound = len(input_array)
    lower_bound = 0
    
    # Initialize the middle index
    mid = upper_bound // 2
    
    # Initialize a counter to track iterations
    count = upper_bound
    
    # Continue searching until the value is found or the counter reaches zero
    while (value != input_array[mid]) and (count >= 0):
        # Update upper and lower bounds based on the comparison with the middle element
        if value < input_array[mid]:
            upper_bound = mid
            mid = mid - (upper_bound - lower_bound) // 2
        else:
            lower_bound = mid
            mid = mid + (upper_bound - lower_bound) // 2
        
        # Decrement the counter
        count = count - 1
    
    # If the counter reaches zero, the value was not found
    if count < 0:
        return -1
    # Otherwise, return the index of the value
    else:
        return mid

# Test cases
test_list = [1, 3, 9, 11, 15, 19, 29]
test_val1 = 25
test_val2 = 15
print(binary_search(test_list, test_val1))  # Output: -1 (not found)
print(binary_search(test_list, test_val2))  # Output: 4 (found at index 4)
