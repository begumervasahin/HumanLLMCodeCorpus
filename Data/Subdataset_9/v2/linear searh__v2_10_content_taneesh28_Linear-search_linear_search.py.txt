def linear_search(arr, val):
    found = False
    for i, element in enumerate(arr):
        if element == val:
            print(f"The value {val} is found at index {i}")
            found = True
            # Uncomment the following line if you want to stop after finding the first occurrence
            # break
    if not found:
        print(f"The value {val} is not found in the array")

def main():
    # Get the size of the array from the user
    array_size = int(input("Enter the size of the array: "))

    # Initialize an empty array
    arr = []

    # Input elements for the array from the user
    for i in range(array_size):
        element = int(input(f"Enter element {i + 1}: "))
        arr.append(element)

    # Display the input array
    print("Array:", arr)

    # Get the value to search for from the user
    search_value = int(input("Enter the value you want to search for: "))

    # Perform linear search
    linear_search(arr, search_value)

if __name__ == "__main__":
    main()
