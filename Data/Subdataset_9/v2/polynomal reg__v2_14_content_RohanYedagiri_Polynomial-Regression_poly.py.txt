# Importing necessary libraries
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import PolynomialFeatures

# Load the dataset
dataset = pd.read_csv('Position_Salaries.csv')

# Separate features (X) and target variable (y)
X = dataset.drop(['Position', 'Salary'], axis=1)
y = dataset['Salary']

# Linear Regression Model
linreg = LinearRegression()  # Initialize linear regression model
linreg.fit(X, y)  # Fit the model to the data

# Polynomial Regression Model
polyreg = PolynomialFeatures(degree=4)  # Initialize polynomial features
X_poly = polyreg.fit_transform(X)  # Transform features to polynomial features
linreg2 = LinearRegression()  # Initialize linear regression model for polynomial features
linreg2.fit(X_poly, y)  # Fit the model to the polynomial features

# Plotting Linear Regression
plt.scatter(X, y, color='red')  # Scatter plot of the data
plt.plot(X, linreg.predict(X), color='blue')  # Plotting linear regression line
plt.title('Linear Regression Predictions')  # Title of the plot
plt.xlabel('Position level')  # X-axis label
plt.ylabel('Salaries')  # Y-axis label
plt.show()  # Show the plot

# Plotting Polynomial Regression
plt.scatter(X, y, color='red')  # Scatter plot of the data
plt.plot(X, linreg2.predict(X_poly), color='blue')  # Plotting polynomial regression line
plt.title('Polynomial Regression Predictions')  # Title of the plot
plt.xlabel('Position level')  # X-axis label
plt.ylabel('Salaries')  # Y-axis label
plt.show()  # Show the plot

# Predictions
position_level = [[6.5]]  # Position level for prediction
linear_prediction = linreg.predict(position_level)[0]  # Predict using linear regression
polynomial_prediction = linreg2.predict(polyreg.fit_transform(position_level))[0]  # Predict using polynomial regression

# Output the predictions
print("Linear Regression Prediction for position level 6.5:", linear_prediction)
print("Polynomial Regression Prediction for position level 6.5:", polynomial_prediction)
