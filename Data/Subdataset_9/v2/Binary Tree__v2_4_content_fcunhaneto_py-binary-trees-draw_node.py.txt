class TreeNode:
    def __init__(self, key):
        self.key = key
        self.left_child = None
        self.right_child = None
        self.parent_node = None
        self.height = 0
        self.real_height = 0
        self.color = 1  # Default color is set to 1 (e.g., red)

# Example usage:
# Create tree nodes and set their attributes
node1 = TreeNode(10)
node1.height = 3
node1.real_height = 4
node1.color = 1  # Set color to 1 (red)

node2 = TreeNode(20)
node2.height = 2
node2.real_height = 3
node2.color = 0  # Set color to 0 (black)

# Print node attributes for demonstration
print("Node 1:")
print("Key:", node1.key)
print("Left Child:", node1.left_child)
print("Right Child:", node1.right_child)
print("Parent Node:", node1.parent_node)
print("Height:", node1.height)
print("Real Height:", node1.real_height)
print("Color:", node1.color)

print("\nNode 2:")
print("Key:", node2.key)
print("Left Child:", node2.left_child)
print("Right Child:", node2.right_child)
print("Parent Node:", node2.parent_node)
print("Height:", node2.height)
print("Real Height:", node2.real_height)
print("Color:", node2.color)
