# Check if Matplotlib library is installed, raise ImportError if not
try:
    import matplotlib.pyplot as plt
except ImportError:
    raise ImportError("Matplotlib library is required. Install it using 'pip install matplotlib'")

# Check if NetworkX library is installed, raise ImportError if not
try:
    import networkx as nx
except ImportError:
    raise ImportError("NetworkX library is required. Install it using 'pip install networkx'")

# Import the Prim's Algorithm implementation from prim.py
from prim import prim

# Read the graph from the edgelist file
G = nx.read_edgelist("flights.edgelist")

# Find the Minimal Spanning Tree (MST) using Prim's Algorithm starting from 'Madrid'
mst = prim(G, 'Madrid')

# Find other edges that are not in the MST
othr = [n for n in G.edges() if n not in mst]

# Print the Minimal Spanning Tree
print("Minimal Spanning Tree: ", mst)

# Layout for visualization
pos = nx.spring_layout(G)

# Draw the graph
nx.draw_networkx_nodes(G, pos, node_size=500)  # Draw nodes
nx.draw_networkx_edges(G, pos, edgelist=mst, width=6)  # Draw MST edges
nx.draw_networkx_edges(G, pos, edgelist=othr, width=6, alpha=0.5, edge_color='b', style='dashed')  # Draw other edges
nx.draw_networkx_labels(G, pos, font_size=20, font_family='sans-serif')  # Draw labels
plt.axis('off')  # Turn off axis
plt.show()  # Show the plot
