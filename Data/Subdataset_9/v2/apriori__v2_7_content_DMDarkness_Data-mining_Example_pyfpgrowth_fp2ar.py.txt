import pyfpgrowth as pyfp  # Import the FP-Growth algorithm package
import fp2ar  # Import the package for forming association rules from frequent patterns
import re  # Import the regular expression module for string operations

def read_transactional_dataset(file_name):
    """
    Read the transactional dataset from the specified file and return a list of transactions.
    Each transaction is represented as a list of integers.
    """
    transactions = []
    with open(file_name, 'r') as file:
        for line in file:
            # Split the line by space or newline and extract integers
            items = re.findall(r'\d+', line)
            if items:
                transactions.append([int(item) for item in items])
    return transactions

if __name__ == '__main__':
    # Specify the file containing the transactional dataset
    dataset_file = "kosarak.dat"

    # Read the transactional dataset from the file
    dataset = read_transactional_dataset(dataset_file)

    # Mine the dataset with a minimum support of 1%
    # After mining, there should be 383 frequent patterns
    frequent_patterns = pyfp.getFP(dataset, 0.01)

    # Form association rules from the frequent patterns
    # The minimum confidence is set to 75%
    # The minimum lift is set to 1
    # After execution, there should be 431 association rules
    association_rules = fp2ar.getAR(frequent_patterns, len(dataset), 0.75, 1)
