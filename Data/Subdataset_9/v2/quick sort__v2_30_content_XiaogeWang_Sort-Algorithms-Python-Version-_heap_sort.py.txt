from collections import deque

class MinHeap:
    def __init__(self):
        self.heap = deque()

    def push(self, value):
        self.heap.append(value)
        self._heapify_up()

    def pop(self):
        if not self.heap:
            raise IndexError("Heap is empty")
        root = self.heap[0]
        if len(self.heap) > 1:
            self.heap[0] = self.heap.pop()
            self._heapify_down()
        else:
            self.heap.pop()
        return root

    def _heapify_up(self):
        index = len(self.heap) - 1
        while index > 0 and self.heap[index] < self.heap[(index - 1) // 2]:
            self.heap[index], self.heap[(index - 1) // 2] = self.heap[(index - 1) // 2], self.heap[index]
            index = (index - 1) // 2

    def _heapify_down(self):
        index = 0
        while (2 * index + 1) < len(self.heap):
            child_index = 2 * index + 1
            if (child_index + 1) < len(self.heap) and self.heap[child_index + 1] < self.heap[child_index]:
                child_index += 1
            if self.heap[index] <= self.heap[child_index]:
                break
            self.heap[index], self.heap[child_index] = self.heap[child_index], self.heap[index]
            index = child_index

a = [8, 1, 3, 4, 2, 5, 0, 9, 6, 7]
heap = MinHeap()
for value in a:
    heap.push(value)

sorted_a = []
for _ in range(len(a)):
    sorted_a.append(heap.pop())

print(sorted_a)
