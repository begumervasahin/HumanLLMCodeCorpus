def binary_search(sorted_list, target):
    # Set initial pointers for the binary search
    first = 0
    last = len(sorted_list) - 1
    found = False

    # Perform binary search until the target is found or the search space is exhausted
    while first <= last and not found:
        midpoint = (first + last) // 2  # Calculate the midpoint of the search space
        if sorted_list[midpoint] == target:
            found = True  # Target found
        else:
            if target < sorted_list[midpoint]:
                last = midpoint - 1  # Update the last pointer
            else:
                first = midpoint + 1  # Update the first pointer

    return found

# Test the binary search function
test_list = [0, 1, 2, 8, 13, 17, 19, 32, 42]
print(binary_search(test_list, 3))   # Output: False
print(binary_search(test_list, 13))  # Output: True
