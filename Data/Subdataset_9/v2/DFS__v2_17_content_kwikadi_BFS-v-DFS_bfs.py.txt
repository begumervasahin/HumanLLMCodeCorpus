from collections import defaultdict

class Graph:
    def __init__(self, count):
        # Initialize a graph with a defaultdict for adjacency lists and total count of vertices
        self.graph = defaultdict(list)
        self.count = count

    def addEdge(self, u, v):
        # Add an edge from vertex u to vertex v in the graph
        self.graph[u].append(v)

    def BFS(self, start, goal):
        # Perform Breadth-First Search (BFS) traversal in the graph from start to goal
        visited = [False] * self.count  # Initialize a list to track visited vertices
        queue = [start]  # Initialize a queue for BFS with the start vertex
        visited[start] = True  # Mark the start vertex as visited
        while queue:
            current_vertex = queue.pop(0)  # Get the next vertex from the queue
            print(current_vertex, end=" ")  # Print the current vertex
            if current_vertex == goal:
                print("Goal found.")  # Print a message if the goal vertex is found
                return
            for neighbor in self.graph[current_vertex]:
                # Traverse all neighbors of the current vertex
                if not visited[neighbor]:
                    queue.append(neighbor)  # Add unvisited neighbors to the queue
                    visited[neighbor] = True  # Mark neighbors as visited
        print("Search failed.")  # Print a message if the goal vertex is not found

# Example usage
def main():
    with open('input.txt', 'r') as inputFile:
        count = int(inputFile.readline())  # Read the total number of vertices
        start_vertex = int(inputFile.readline())  # Read the start vertex for BFS
        goal_vertex = int(inputFile.readline())  # Read the goal vertex for BFS
        graph = Graph(count)  # Create an instance of the Graph class
        for line in inputFile:
            u, v = map(int, line.split())  # Read edges from input file and add them to the graph
            graph.addEdge(u, v)
    graph.BFS(start_vertex, goal_vertex)  # Perform BFS from start to goal

if __name__ == "__main__":
    main()
