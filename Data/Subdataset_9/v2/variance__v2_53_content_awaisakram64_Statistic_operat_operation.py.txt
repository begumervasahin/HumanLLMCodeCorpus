class StatisticsCalculator:
    # Method to calculate the median of a dataset
    def calculate_median(self, data):
        # Sort the data in ascending order
        sorted_data = sorted(data)
        # Calculate the length of the dataset
        data_length = len(sorted_data)
        # Check if the length of the dataset is even
        if data_length % 2 == 0:
            # If even, return the average of the middle two numbers
            middle_index = data_length // 2
            return (sorted_data[middle_index - 1] + sorted_data[middle_index]) / 2
        else:
            # If odd, return the middle number
            return sorted_data[data_length // 2]

    # Method to calculate the mode of a dataset
    def calculate_mode(self, data):
        # Create a dictionary to store each unique element and its frequency
        frequency_dict = {}
        for item in set(data):
            frequency_dict[item] = data.count(item)
        # Find the maximum frequency
        max_frequency = max(frequency_dict.values())
        # If all frequencies are 1, return 0 as there is no mode
        if max_frequency <= 1:
            return [0]
        else:
            # Find all elements with maximum frequency
            mode_elements = [key for key, val in frequency_dict.items() if val == max_frequency]
            return mode_elements

    # Method to calculate the mean of a dataset
    def calculate_mean(self, data):
        # Sum of all elements divided by the number of elements
        return sum(data) / len(data)

    # Method to calculate the variance of a dataset
    def calculate_variance(self, data):
        # Mean of the dataset
        mean = self.calculate_mean(data)
        # Sum of squared differences between each element and the mean, divided by the number of elements
        return sum((item - mean) ** 2 for item in data) / len(data)

    # Method to calculate the standard deviation of a dataset
    def calculate_standard_deviation(self, data):
        # Square root of the variance
        return (self.calculate_variance(data)) ** 0.5


def main():
    # Create an instance of the StatisticsCalculator class
    calculator = StatisticsCalculator()
    # Read data from the file
    with open("data.txt", 'r') as file:
        data = [float(line.strip()) for line in file]
    # Print calculated statistics
    print('Median is %.2f' % (calculator.calculate_median(data)))
    print('Mean is %.2f' % (calculator.calculate_mean(data)))
    print('Mode is ' + (', '.join(map(str, [i for i in calculator.calculate_mode(data)]))))
    print('Variance is %.2f' % (calculator.calculate_variance(data)))
    print('Standard Deviation is %.2f' % (calculator.calculate_standard_deviation(data)))


if __name__ == "__main__":
    main()
