import sys
import numpy as np

# Initialize a 2D array for storing LCS lengths
arr = np.zeros((2048, 2048), dtype=int)

# Function to find the length of the Longest Common Subsequence (LCS) of two strings
def LCS(A, B):
    m = len(A)
    n = len(B)
    
    # Iterate through each character of both strings
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            # If the characters match, increment the LCS length by 1
            if A[i - 1] == B[j - 1]:
                arr[i][j] = arr[i - 1][j - 1] + 1
            # If the characters don't match, take the maximum LCS length from the adjacent cells
            else:
                arr[i][j] = max(arr[i - 1][j], arr[i][j - 1])
    
    # Return the length of LCS
    return arr[m][n]

# Main function to handle input and output
def main():
    # Check if the script is run with correct usage
    if len(sys.argv) != 1:
        sys.exit('Usage: `python LCS.py < input`')
    
    # Read input strings from stdin and find LCS for each pair
    for line in sys.stdin:
        A, B = line.split()
        print(LCS(A, B))

# Execute the main function if the script is run directly
if __name__ == '__main__':
    main()
