import requests
import webbrowser
import random
from bs4 import BeautifulSoup

# Function to fetch HTML content of a webpage and parse it into BeautifulSoup object
def get_soup(url):
    response = requests.get(url)
    if response.status_code == 200:
        return BeautifulSoup(response.text, 'html.parser')
    else:
        print("Failed to fetch page:", url)
        return None

# Function to display categories to the user
def display_categories(categories):
    print("Categories:")
    for idx, category in enumerate(categories, start=1):
        print(f"{idx}. {category.text.strip()}")

# Function to extract product links from the BeautifulSoup object
def get_product_links(soup):
    products = soup.find_all("div", class_="_2i1r")
    return ["https://paytmmall.com/" + product.find("a")["href"] for product in products]

# Main function to orchestrate the process
def main():
    # Base URL of the webpage
    base_url = "https://paytmmall.com/sports-and-health-clpid-83?src=store&use_sf=1"
    # Fetching and parsing the HTML content of the base URL
    soup = get_soup(base_url)
    # Checking if the webpage is successfully fetched
    if not soup:
        return

    # Displaying a welcome message with the title of the webpage
    print("***********Welcome to", soup.title.text.strip(), "*************")

    # Finding the main division containing category links
    main_div = soup.find("div", class_="_1_Nn")
    # Extracting category links
    categories = main_div.find_all("a")

    # Displaying available categories to the user
    display_categories(categories)
    # Prompting the user to choose a category
    choice = int(input("Which category do you want? (Enter the number): "))

    # Constructing the URL of the selected category
    selected_category_url = "https://paytmmall.com/" + categories[choice - 1]["href"]
    # Fetching and parsing the HTML content of the selected category page
    soup_category = get_soup(selected_category_url)
    # Checking if the webpage is successfully fetched
    if not soup_category:
        return

    # Extracting product links from the selected category page
    product_links = get_product_links(soup_category)

    # Selecting a random product link from the extracted links
    random_product_url = random.choice(product_links)
    # Informing the user about opening a random product from the selected category
    print("Opening a random product from the selected category...")
    # Opening the random product link in a new web browser tab
    webbrowser.open_new_tab(random_product_url)

# Ensuring that the main function is executed when the script is run
if __name__ == "__main__":
    main()
