# Import necessary libraries
from selenium import webdriver
from selenium.webdriver.common.keys import Keys
import time
from bs4 import BeautifulSoup
import pandas as pd

# Prompt user for LinkedIn credentials and company ID
COMPANY = input("Enter Company ID: ")
USERNAME = input("Enter username: ")
PASSWORD = input("Enter password: ")

# Specify the desired number of employees to scrape
EMPLOYEE = 1000

# LinkedIn URL
linkedin_url = 'https://www.linkedin.com'

# Initialize the Firefox web driver
browser = webdriver.Firefox()
browser.get(linkedin_url)
time.sleep(3)  # Pause for page loading

# Locate email and password fields and input user credentials
email_field = browser.find_element_by_name('session_key')
password_field = browser.find_element_by_name('session_password')
email_field.send_keys(USERNAME + Keys.RETURN)
password_field.send_keys(PASSWORD + Keys.RETURN)
time.sleep(3)  # Pause for login process

# Construct the search URL for employees of the specified company
search_url = f"https://www.linkedin.com/search/results/people/?facetCurrentCompany=%5B%22{COMPANY}%22%5D"
browser.get(search_url)
time.sleep(3)  # Pause for page loading

# Scroll to the bottom of the page to load more search results
browser.execute_script("window.scrollTo(0, document.body.scrollHeight);")

# Initialize an empty DataFrame to store scraped data
df = pd.DataFrame(columns=['name', 'title', 'location', 'profile'])

# Loop through search result pages
current_url = 'url_placeholder'
while True:
    # Check if reached page 100 or unable to load new results
    if current_url.find('page=100') != -1:
        break
    previous_url = current_url
    current_url = browser.current_url
    if current_url == previous_url:
        break
    
    # Parse the current page using BeautifulSoup
    page = BeautifulSoup(browser.page_source, 'lxml')
    page_names = page.find_all('span', class_='actor-name')
    page_titles = page.find_all('p', class_='subline-level-1')
    page_locations = page.find_all('p', class_='subline-level-2')
    page_profiles = page.find_all('a', class_='search-result__result-link')
    
    # Extract information from the parsed page
    names = [name.text for name in page_names]
    titles = [title.text.replace('\n', '') for title in page_titles]
    locations = [location.text.replace('\n', '') for location in page_locations]
    profiles = [linkedin_url + profile['href'] for profile in page_profiles][::2]
    
    # Create a DataFrame from the extracted information
    temp_df = pd.DataFrame({'name': names, 'title': titles, 'location': locations, 'profile': profiles})
    temp_df = temp_df[temp_df['name'] != 'LinkedIn Member']  # Exclude LinkedIn Member entries
    df = df.append(temp_df)
    
    # Break loop if the desired number of employees is reached
    if df.shape[0] >= EMPLOYEE:
        break
    
    # Navigate to the next page of search results
    next_button = browser.find_element_by_class_name('next')
    next_button.click()
    time.sleep(5)  # Pause for page loading

# Reset the DataFrame index
df.reset_index(drop=True, inplace=True)

# Save the DataFrame to a CSV file
df.to_csv("output_search.csv", index=False)

# Quit the browser
browser.quit()
