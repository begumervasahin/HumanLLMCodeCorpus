# Importing necessary libraries
import pandas as pd
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import PolynomialFeatures
import matplotlib.pyplot as plt

# Step 1: Read the dataset
dataset = pd.read_csv("Position_Salaries.csv")

# Step 2: Extract features (X) and target variable (y)
X = dataset.iloc[:, 1:2].values  # Extracting the "Level" feature
y = dataset.iloc[:, 2].values     # Extracting the "Salary" target variable

# Step 3: Fitting a linear regression model
linearRegr = LinearRegression()
linearRegr.fit(X, y)

# Step 4: Fitting a polynomial regression model
polynomFeat = PolynomialFeatures(degree=4)
X_poly = polynomFeat.fit_transform(X)

linearRegrPoly.fit(X_poly, y)

# Step 5: Predictions using linear regression
predictions_linear = linearRegr.predict(X)

# Step 6: Predictions using polynomial regression
predictions_poly = linearRegrPoly.predict(polynomFeat.fit_transform(X))

# Step 7: Plotting results for linear regression
plt.scatter(X, y, color="blue")
plt.plot(X, predictions_linear, color="green", label="Linear Regression")
plt.xlabel("Level")
plt.ylabel("Salary")
plt.title("Linear Regression")
plt.legend()
plt.show()

# Step 8: Plotting results for polynomial regression
plt.scatter(X, y, color="blue")
plt.plot(X, predictions_poly, color="green", label="Polynomial Regression")
plt.xlabel("Level")
