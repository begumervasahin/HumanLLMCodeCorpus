import numpy as np

def primegen():
    # Ask the user for the number of bits
    bits = int(input('How many bits? '))

    # Calculate the maximum number based on the number of bits
    final = 2 ** bits

    # Generate a list of integers from 2 to the calculated maximum
    ints_to_final = np.array(list(range(2, final + 1)))

    # Initialize index for looping through the list of integers
    index = 0

    # Loop until the index reaches the length of the list
    while index < len(ints_to_final):
        # Create an array to mark numbers to delete
        del_vect = np.zeros_like(ints_to_final)

        # Iterate through the list of integers
        for i in range(len(ints_to_final)):
            # Check if the number is the same as the current index
            if ints_to_final[i] == ints_to_final[index]:
                del_vect[i] = 0
            else:
                # Check if the number is divisible by the number at the current index
                if ints_to_final[i] % ints_to_final[index] == 0:
                    del_vect[i] = ints_to_final[i]
                else:
                    del_vect[i] = 0

        # Remove marked numbers from the list of integers
        ints_to_final = ints_to_final - del_vect
        ints_to_final = ints_to_final[ints_to_final != 0]

        # Increment the index
        index = index + 1

    # Select two random prime numbers from the list of remaining integers
    random1 = int(np.random.choice(ints_to_final, 1))
    random2 = int(np.random.choice(ints_to_final, 1))

    # Print the hidden P and Q values
    print(f'(Hidden) P Value: {int(random1)}')
    print(f'(Hidden) Q Value: {int(random2)}')

    # Return the selected prime numbers
    return random1, random2

if __name__ == "__main__":
    # Call the primegen function if the script is executed directly
    primegen()
