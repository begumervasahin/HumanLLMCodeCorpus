import time
import random
import matplotlib.pyplot as plt
import networkx as nx
import prim_graph as prim
from kruskal import Algoritimo_kruskal as kr

def generate_complete_graph_plots():
    kruskal_algo = kr()
    vertices_list = []
    kruskal_avg_time_list = []
    prim_avg_time_list = []
    total_vertices = 2011
    total_time = time.time()

    for num_vertices in range(10, total_vertices, 100):
        print('Number of Vertices:', num_vertices)
        G = nx.complete_graph(num_vertices)
        edges = G.edges()
        matrix = [[0] * num_vertices for _ in range(num_vertices)]

        for v1, v2 in edges:
            weight = random.randint(10, 100)
            matrix[v1][v2] = weight
            matrix[v2][v1] = weight
            G.edges[v1, v2]['weight'] = weight

        prim_graph = prim.Graph(num_vertices)
        prim_graph.graph = matrix
        kruskal_time_sum = 0.
        prim_time_sum = 0.

        for _ in range(1, 11):
            start_time = time.time()
            mst_kruskal = kruskal_algo.kruskal(G)
            kruskal_time = time.time() - start_time
            kruskal_time_sum += kruskal_time

            start_time = time.time()
            parent = prim_graph.primMST()
            prim_time = time.time() - start_time
            prim_time_sum += prim_time

        print("Average Execution Time Kruskal:", kruskal_time_sum / 10, "seconds")
        print("Average Execution Time Prim:", prim_time_sum / 10, "seconds")

        vertices_list.append(num_vertices)
        kruskal_avg_time_list.append(kruskal_time_sum / 10)
        prim_avg_time_list.append(prim_time_sum / 10)

    print("Total Time:", time.time() - total_time)

    plt.plot(vertices_list, prim_avg_time_list, label="Prim's algorithm")
    plt.plot(vertices_list, kruskal_avg_time_list, label="Kruskal's algorithm")
    plt.legend()
    plt.axis([0, total_vertices + 10, 0, max(kruskal_avg_time_list) * 2])
    plt.ylabel('Execution Time (Seconds)')
    plt.xlabel('Number of Vertices')
    plt.title("Performance Comparison: Kruskal vs Prim (Complete Graph)")
    plt.savefig('Kruskal_vs_Prim_Complete_Graph.png')
    plt.show()

    print("Vertices List:", vertices_list)
    print("Average Time Kruskal:", kruskal_avg_time_list)
    print("Average Time Prim:", prim_avg_time_list)

def generate_graph(num_vertices):
    G = nx.connected_watts_strogatz_graph(num_vertices, 25, 0.5)
    edges = G.edges()
    matrix = [[0] * num_vertices for _ in range(num_vertices)]

    for v1, v2 in edges:
        weight = random.randint(10, 100)
        matrix[v1][v2] = weight
        matrix[v2][v1] = weight
        G.edges[v1, v2]['weight'] = weight

    kruskal_algo = kr()
    prim_graph = prim.Graph(num_vertices)
    prim_graph.graph = matrix
    mst_kruskal = kruskal_algo.kruskal(G)
    mst_prim = prim_graph.primMST()

    plt.show()
    nx.draw(G, node_size=5, alpha=1, node_color="black", edge_color="red")
    plt.axis('equal')
    plt.savefig('Graph.png')
    plt.show()

    plt.figure(figsize=(8, 8))
    nx.draw(mst_prim, node_size=5, alpha=1, node_color="black", edge_color="blue")
    plt.axis('equal')
    plt.savefig('mst_prim.png')
    plt.show()

    nx.draw(mst_kruskal, node_size=5, alpha=1, node_color="black", edge_color="green")
    plt.axis('equal')
    plt.savefig('mst_kruskal.png')
    plt.show()

generate_graph(500)
generate_non_complete_graph_plots()
