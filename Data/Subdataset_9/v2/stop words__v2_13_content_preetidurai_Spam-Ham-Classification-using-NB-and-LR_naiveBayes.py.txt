from collections import Counter
import math as m
import sys
from NBHelper import *

class NaiveBayesClassifier:
    def __init__(self, num_classes, class_names):
        self.num_classes = num_classes
        self.class_mapping = {i: class_names[i] for i in range(num_classes)}
        self.vocabularies = {}
        self.prior_probabilities = {}
        self.conditional_probabilities = {}
        self.total_vocab_list = []
        self.vocab_set = set()

    def build_vocabulary(self, path_array):
        num_of_mails = {}
        total_mails = 0
        for key, value in self.class_mapping.items():
            mails = getMailDictionary(path_array[key])
            num_of_mails[value] = len(mails)
            total_mails += len(mails)
            self.vocabularies[value] = getVocabulary(mails)
        for key, value in num_of_mails.items():
            self.prior_probabilities[key] = num_of_mails[key] / total_mails

    def build_vocabulary_without_stopwords(self, path_array, stop_word_path):
        num_of_mails = {}
        total_mails = 0
        stop_words = readStopWords(stop_word_path)
        for key, value in self.class_mapping.items():
            mails = getMailDictionaryWOStopWords(path_array[key], stop_words)
            num_of_mails[value] = len(mails)
            total_mails += len(mails)
            self.vocabularies[value] = getVocabulary(mails)
        for key, value in num_of_mails.items():
            self.prior_probabilities[key] = num_of_mails[key] / total_mails

    def build_vocab_set(self):
        for value in self.vocabularies.values():
            self.total_vocab_list.extend(value)
        self.vocab_set = set(self.total_vocab_list)

    def train(self):
        for value in self.class_mapping.values():
            word_freq_dict = Counter(self.vocabularies[value])
            self.conditional_probabilities[value] = {}
            for term in self.vocab_set:
                if term in word_freq_dict:
                    term_count = word_freq_dict[term]
                    self.conditional_probabilities[value][term] = (1 + term_count) / (len(self.vocab_set) + len(self.vocabularies[value]))
                else:
                    self.conditional_probabilities[value][term] = 1 / (len(self.vocab_set) + len(self.vocabularies[value]))

    def classify(self, words):
        posterior_probabilities = {val: m.log(self.prior_probabilities[val]) for val in self.class_mapping.values()}
        for term in words:
            for val in self.class_mapping.values():
                if term in self.conditional_probabilities[val]:
                    posterior_probabilities[val] += m.log(self.conditional_probabilities[val][term])
        return max(posterior_probabilities, key=posterior_probabilities.get)

    def validate(self, test_paths):
        spam_test_mails = getMailDictionary(test_paths[0])
        ham_test_mails = getMailDictionary(test_paths[1])
        spam_correct = sum(1 for value in spam_test_mails.values() if self.classify(value) == "spam")
        ham_correct = sum(1 for value in ham_test_mails.values() if self.classify(value) == "ham")
        total_spam_mails = len(spam_test_mails)
        total_ham_mails = len(ham_test_mails)
        total_correct = spam_correct + ham_correct
        total_mails = total_spam_mails + total_ham_mails
        accuracy = (total_correct / total_mails) * 100
        print("Accuracy without removing stopwords:", accuracy)

    def validate_without_stopwords(self, test_paths, stop_path):
        stop_words = readStopWords(stop_path)
        spam_test_mails = getMailDictionaryWOStopWords(test_paths[0], stop_words)
        ham_test_mails = getMailDictionaryWOStopWords(test_paths[1], stop_words)
        spam_correct = sum(1 for value in spam_test_mails.values() if self.classify(value) == "spam")
        ham_correct = sum(1 for value in ham_test_mails.values() if self.classify(value) == "ham")
        total_spam_mails = len(spam_test_mails)
        total_ham_mails = len(ham_test_mails)
        total_correct = spam_correct + ham_correct
        total_mails = total_spam_mails + total_ham_mails
        accuracy = (total_correct / total_mails) * 100
        print("Accuracy after removing stopwords:", accuracy)

def main():
    classifier = NaiveBayesClassifier(2, ["spam", "ham"])
    argv_list = sys.argv
    train_paths = [argv_list[1], argv_list[2]]
    test_paths = [argv_list[3], argv_list[4]]
    stop_path = argv_list[5]

    classifier.build_vocabulary(train_paths)
    classifier.build_vocab_set()
    classifier.train()
    classifier.validate(test_paths)

    classifier_without_stopwords = NaiveBayesClassifier(2, ["spam", "ham"])
    classifier_without_stopwords.build_vocabulary_without_stopwords(train_paths, stop_path)
    classifier_without_stopwords.build_vocab_set()
    classifier_without_stopwords.train()
    classifier_without_stopwords.validate_without_stopwords(test_paths, stop_path)

if __name__ == "__main__":
    main()
