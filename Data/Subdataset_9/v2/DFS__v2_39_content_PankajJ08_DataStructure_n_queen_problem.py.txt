# Unicode symbol for queen chess piece
QUEEN = "\u265B"
# Size of the chessboard
N = 8
# Counter for solution number
solution_count = 1

def is_safe(board, row, col):
    """Check if a queen can be placed at board[row][col]."""
    # Check if there's a queen in the same row to the left
    for i in range(col):
        if board[row][i] == QUEEN:
            return False
    # Check upper-left diagonal
    for i, j in zip(range(row, -1, -1), range(col, -1, -1)):
        if board[i][j] == QUEEN:
            return False
    # Check lower-left diagonal
    for i, j in zip(range(row, N, 1), range(col, -1, -1)):
        if board[i][j] == QUEEN:
            return False
    return True

def n_queen_solution(board, col):
    """Find all solutions for the N-Queen problem."""
    global solution_count
    # If all queens are placed (base case)
    if col >= N:
        # Print the solution
        print("Solution", solution_count, ": ")
        for i in range(N):
            for j in range(N):
                print(board[i][j], end="  ")
            print()
        print()
        solution_count += 1
        return True
    # Try placing a queen in each row of the current column
    for i in range(N):
        if is_safe(board, i, col):
            board[i][col] = QUEEN
            # Recursively search for solutions for the next column
            if n_queen_solution(board, col + 1):
                return True
            # If no solution found, backtrack and try next row
            board[i][col] = '_'
    return False

def n_queen():
    """Find and print all solutions for the N-Queen problem."""
    # Initialize the chessboard
    board = [['_' for _ in range(N)] for _ in range(N)]
    # Find solutions starting from the first column
    if not n_queen_solution(board, 0):
        print("No solution exists!")

if __name__ == '__main__':
    # Start the program
    n_queen()
