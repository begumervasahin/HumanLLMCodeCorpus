# Open the file containing the results
with open('cpp_res.txt') as file:
    # Read all lines from the file
    lines = file.readlines()

# Initialize an empty list to store the time values
times = []

# Iterate through each line in the file
for line in lines:
    # Check if the line contains the word "real"
    if "real" in line:
        # Split the line and extract the time value
        time_value = float(line.split()[1])
        # Append the time value to the list
        times.append(time_value)

# Calculate the average time
average_time = sum(times) / len(times)

# Print the average time
print("The average time was:")
print(average_time)
print("seconds")
