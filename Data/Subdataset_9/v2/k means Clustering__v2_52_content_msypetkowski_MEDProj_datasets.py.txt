import pandas as pd

# Function to load the Iris dataset
def load_iris():
    # Read the Iris dataset from the given URL
    return pd.read_csv('https://raw.githubusercontent.com/mwaskom/seaborn-data/master/iris.csv')

# Function to load the Adult dataset
def load_adult():
    # Read the Adult dataset from the given URL with specific column names and handling missing values
    return pd.read_csv('https://archive.ics.uci.edu/ml/machine-learning-databases/adult/adult.data',
                       names=["Age", "Workclass", "fnlwgt", "Education", "Education-Num", "Martial Status",
                              "Occupation", "Relationship", "Race", "Sex", "Capital Gain", "Capital Loss",
                              "Hours per week", "Country", "Target"],
                       sep=r'\s*,\s*', engine='python', na_values="?")

# Function to load the Cardiotocography dataset
def load_ctg():
    # Read the Cardiotocography dataset from the given URL and drop the 'Unnamed: 0' column
    ret = pd.read_csv('http://staff.ii.pw.edu.pl/~gprotazi/dydaktyka/dane/cardioto_all_corr.csv')
    ret = ret.drop('Unnamed: 0', axis=1)
    return ret

# Function to load the Car Evaluation dataset
def load_cars():
    # Read the Car Evaluation dataset from the given URL with specific column names
    return pd.read_csv('https://archive.ics.uci.edu/ml/machine-learning-databases/car/car.data',
                       names=["class", "buying", "maint", "doors", "persons", "lug_boot", "safety"])

# Main block to test the dataset loading functions
if __name__ == "__main__":
    # Load the Iris dataset and print the first few rows
    iris_df = load_iris()
    print("IRIS dataset loaded successfully.")
    print(iris_df.head())

    # Load the Adult dataset and print the first few rows
    adult_df = load_adult()
    print("\nADULT dataset loaded successfully.")
    print(adult_df.head())

    # Load the Cardiotocography dataset and print the first few rows
    ctg_df = load_ctg()
    print("\nCTG dataset loaded successfully.")
    print(ctg_df.head())

    # Load the Car Evaluation dataset and print the first few rows
    cars_df = load_cars()
    print("\nCARS dataset loaded successfully.")
    print(cars_df.head())
