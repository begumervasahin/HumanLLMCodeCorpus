import os
import imageio
import numpy as np
import matplotlib.pyplot as plt

class SortVisualization:
    def __init__(self):
        self.img = []
        self.fig = plt.figure()
        plt.ion()
        self.count = 0
        self.path = 'sort_process/'
        if not os.path.exists(self.path):
            os.mkdir(self.path)

    def merge_sort(self, s, start, end):
        if end - start < 2:
            return
        mid = (end - start) // 2
        self.merge_sort(s, start, start + mid)
        self.merge_sort(s, start + mid, end)
        self.merge(start, mid, end, s)

    def merge(self, start, mid, end, s):
        i, j = start, start + mid
        while i < start + mid and j < end:
            if j == end or (i < start + mid and s[i] < s[j]):
                i += 1
            else:
                s.insert(i, s[j])
                j += 1
                s.pop(j)
                self.draw(s, 'MERGE SORT')

    def quick_sort(self, s, left, right):
        if left >= right:
            return
        pivot = s[right]
        l, r = left, right - 1
        while l <= r:
            while l <= r and s[l] < pivot:
                l += 1
            while l <= r and s[r] > pivot:
                r -= 1
            if l <= r:
                s[l], s[r] = s[r], s[l]
                l += 1
                r -= 1
        s[l], s[right] = s[right], s[l]
        if l != right:
            self.draw(s, 'QUICK SORT')
        self.quick_sort(s, left, l - 1)
        self.quick_sort(s, l + 1, right)

    def select_sort(self, s):
        self.draw(s, 'SELECTION SORT')
        for i in range(len(s)):
            if self.selection_step(s, i):
                self.draw(s, 'SELECTION SORT')

    def selection_step(self, s, i):
        min_index = i
        for j in range(i, len(s)):
            if s[j] < s[min_index]:
                min_index = j
        if min_index != i:
            s[i], s[min_index] = s[min_index], s[i]
            return True
        return False

    def heap_adjust(self, s, root, length):
        child = 2 * root + 1
        if child == length - 1:
            if s[root] < s[child]:
                s[root], s[child] = s[child], s[root]
        while child < length - 1:
            if s[child] < s[child + 1]:
                child += 1
            if s[root] > s[child]:
                break
            s[root], s[child] = s[child], s[root]
            root, child = child, 2 * child + 1

    def heap_sort(self, s):
        length = len(s)
        for i in range(length // 2 - 1, -1, -1):
            self.heap_adjust(s, i, length)
        for i in range(length - 1, 0, -1):
            s[0], s[i] = s[i], s[0]
            self.draw(s, 'HEAP SORT')
            self.heap_adjust(s, 0, i)

    def insert_sort(self, s):
        for i in range(1, len(s)):
            if self.insert_step(s, i):
                self.draw(s, 'INSERT SORT')

    def insert_step(self, s, k):
        index, value = k, s[k]
        for i in range(k, len(s)):
            if s[i] < value:
                index, value = i, s[i]
        s[k], s[index] = s[index], s[k]
        return index != k

    def shell_sort(self, s):
        t = int(np.log2(len(s) + 1))
        for count in range(1, t + 1):
            delta = 2 ** (t - count + 1) - 1
            for i in range(len(s) - delta):
                if self.shell_step(s, i, i + delta):
                    self.draw(s, 'SHELL SORT')

    def shell_step(self, s, start, end):
        index, value = start, s[start]
        for i in range(start, end + 1):
            if s[i] < value:
                index, value = i, s[i]
        s[start], s[index] = s[index], s[start]
        return index != start

    def bubble_sort(self, s):
        for i in range(len(s) - 1):
            for j in range(len(s) - i - 1):
                if s[j] > s[j + 1]:
                    s[j], s[j + 1] = s[j + 1], s[j]
                    self.draw(s, 'BUBBLE SORT')

    def draw(self, s, sort_name):
        if len(s):
            self.count += 1
            plt.clf()
            plt.bar(range(len(s)), s, width=0.3, color='purple')
            plt.title(sort_name)
            plt.savefig(self.path + str(self.count))
            plt.pause(0.01)

    def get_gif(self):
        names = os.listdir(self.path)
        names.sort(key=lambda x: int(x[:-4]))
        for name in names:
            path = self.path + name
            self.img.append(imageio.imread(path))
        imageio.mimsave('sort.gif', self.img, fps=8)

if __name__ == '__main__':
    s = [100 - k for k in range(100)]
    sort_obj = SortVisualization()
    sort_obj.select_sort(list(s))
    sort_obj.heap_sort(list(s))
    sort_obj.quick_sort(list(s), 0, len(s) - 1)
    sort_obj.merge_sort(list(s), 0, len(s))
    sort_obj.insert_sort(list(s))
    sort_obj.shell_sort(list(s))
    sort_obj.get_gif()
