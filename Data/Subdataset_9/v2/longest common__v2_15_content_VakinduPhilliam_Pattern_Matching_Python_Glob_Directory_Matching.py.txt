# Import the glob module to search for files
import glob

# Find all GIF files in the current directory
gif_files = glob.glob('*.gif')
print("Here are the GIF files I found in this folder:")
print(gif_files)

# Find all files that start with 'c' in the current directory
c_files = glob.glob('.c*')
print("\nAnd here are the files that start with 'c':")
print(c_files)
