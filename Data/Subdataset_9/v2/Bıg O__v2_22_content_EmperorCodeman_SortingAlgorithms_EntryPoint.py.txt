from BubbleSort import Bubble_Sort
from ShakerSort import Shaker_Sort
from SelectionSort import SelectionSort
from QuickSort import QuickSort, ModifiedQuickSort
from MergeSort import MergeSort
from HashSort import HashSort
from random import shuffle, randrange, seed

# Function to test sorting algorithms
def test(iterations=1000, debug=False):
    # Set random seed for reproducibility if debugging
    if debug:
        seed(10)
    
    # Generate a random test case
    case = [randrange(0, 7) for x in range(7)]
    
    # Sort the test case for comparison
    testCase = case[:]
    testCase.sort()
    
    # List to store failure flags
    failure = []
    
    # Set recursion limit for sorting algorithms
    import sys
    sys.setrecursionlimit(12000)
    
    # Perform sorting algorithm tests
    for test in range(iterations):
        # Test Bubble Sort
        case_copy = case[:]  
        Bubble_Sort(case_copy, tabulations)
        failure.append(case_copy == testCase)

        # Test Shaker Sort
        case_copy = case[:]
        Shaker_Sort(case_copy, tabulations)
        failure.append(case_copy == testCase)

        # Test Selection Sort
        case_copy = case[:]
        SelectionSort(case_copy, tabulations)
        failure.append(case_copy == testCase)

        # Test Quick Sort
        case_copy = case[:]
        QuickSort(case_copy, tabulations)
        failure.append(case_copy == testCase)

        # Test Modified Quick Sort
        case_copy = case[:]
        ModifiedQuickSort(case_copy, tabulations)
        failure.append(case_copy == testCase)

        # Test Merge Sort
        case_copy = case[:]
        MergeSort(case_copy, tabulations)
        failure.append(case_copy == testCase)

        # Test Hash Sort
        case_copy = case[:]
        HashSort(case_copy, tabulations)
        failure.append(case_copy == testCase)

        shuffle(case)  # Shuffle for the next iteration

    # Print test result
    if False in failure:
        print(False)
    else:
        print(True)

# Metrics dictionary for tracking comparisons and swaps
metrics = {"comparisons": 0, "swaps": 0}

# Dictionary to store metrics for each sorting algorithm
tabulations = {
    "bubbleSort": metrics,
    "shakerSort": metrics,
    "selectionSort": metrics,
    "quickSort": metrics,
    "modifiedQuickSort": metrics,
    "mergeSort": metrics,
    "hashSort": metrics
}

# Run the test function
test()
