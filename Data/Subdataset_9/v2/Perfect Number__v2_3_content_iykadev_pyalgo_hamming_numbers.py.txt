import sys

def is_hamming_number(x):
    """Check if a number is a Hamming number."""
    if x == 1:
        return True
    if x % 2 == 0:
        return is_hamming_number(x // 2)
    if x % 3 == 0:
        return is_hamming_number(x // 3)
    if x % 5 == 0:
        return is_hamming_number(x // 5)
    return False

def print_hamming_numbers(x):
    """Print Hamming numbers up to a given limit."""
    if x == 1:
        return
    if is_hamming_number(x):
        print(x, end=' ')
    print_hamming_numbers(x - 1)

def hamming_numbers_main():
    """Main function to print Hamming numbers."""
    sys.stdout.write("Hamming Numbers: ")
    print_hamming_numbers(9830)

if __name__ == "__main__":
    # Increase recursion limit to handle large numbers
    sys.setrecursionlimit(10000)
    hamming_numbers_main()
