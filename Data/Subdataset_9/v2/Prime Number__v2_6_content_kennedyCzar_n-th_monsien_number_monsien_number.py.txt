import random

def nth_prime(n):
    """
    Finds the nth prime number.
    """
    number = 2
    primes_found = 0
    while primes_found < n:
        if is_prime(number):
            primes_found += 1
        number += 1
    return number - 1

def is_prime(num):
    """
    Checks if a number is prime.
    """
    if num <= 1:
        return False
    for i in range(2, int(num**0.5) + 1):
        if num % i == 0:
            return False
    return True

def find_monisen(n):
    """
    Finds the n-th Monisen number.
    """
    prime_count = 0
    number = 2
    while True:
        if is_prime(number):
            m = 2 ** number - 1
            if is_prime(m):
                prime_count += 1
                if prime_count == n:
                    return m
        number += 1

def generate_monisen_file(n):
    """
    Generates a text file containing the n-th Monisen number.
    """
    monisen = find_monisen(n)
    with open(f'monisen_{n}.txt', 'w') as file:
        file.write(str(monisen))

def convert_to_base(num, base):
    """
    Converts a number to a different base recursively.
    """
    if num >= base:
        convert_to_base(num // base, base)
    print(num % base, end=' ')
