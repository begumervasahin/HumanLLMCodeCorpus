from queue import Queue

class TreeNode:
    def __init__(self, data):
        self.data = data
        self.left = None
        self.right = None

def display_tree(root, full_depth):
    """Display nodes at a specific depth in the tree"""
    if root is None:
        return
    if full_depth == 0:
        print(root.data, end=' ')
    display_tree(root.left, full_depth - 1)
    display_tree(root.right, full_depth - 1)

def display_level_order(root):
    """Display nodes in level order (Breadth First Traversal)"""
    if root is None:
        return
    nodes = Queue()
    nodes.put(root)
    while not nodes.empty():
        node_levels = nodes.qsize()
        while node_levels > 0:
            node = nodes.get()
            print(node.data, end=' ')
            if node.left is not None:
                nodes.put(node.left)
            if node.right is not None:
                nodes.put(node.right)
            node_levels -= 1
        print("")

# Binary Tree creation
root = TreeNode(1)
root.left = TreeNode(4)
root.right = TreeNode(5)
root.left.left = TreeNode(2)
root.left.right = TreeNode(8)
root.right.left = TreeNode(3)
root.right.right = TreeNode(7)
root.left.left.left = TreeNode(0)
root.left.left.right = TreeNode(1)
root.left.right.left = TreeNode(3)
root.left.right.right = TreeNode(9)
root.right.right.left = TreeNode(1)
root.right.right.right = TreeNode(10)

# Displaying the binary tree in level order
display_level_order(root)
