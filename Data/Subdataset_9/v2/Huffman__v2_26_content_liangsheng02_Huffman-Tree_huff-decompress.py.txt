import os
import pickle
import argparse
import time

class Node:
    """
    Represents a node in the Huffman tree.
    """

    def __init__(self, char=None):
        """
        Initializes a node with the given character.
        """
        self.char = char
        self.left = None
        self.right = None

class HuffmanDecoder:
    """
    Class for decompressing Huffman-coded files.
    """

    def __init__(self, file):
        """
        Initializes the HuffmanDecoder with the file to decompress.
        """
        self.file = os.path.splitext(file)[0]
        self.text_code, self.text_string = self.__readfile()
        self.root = self.__to_tree()
        self.final_text = self.__decompress()
        print('Decompression completed:', len(self.final_text), 'characters')

    def __readfile(self):
        """
        Load pickle and binary files, and convert the binary array into a long string.
        Returns the word-frequency dictionary and the string of compressed code.
        """
        # Load word-frequency dictionary from pickle file
        text_code = pickle.load(open(self.file + '-symbol-model.pkl', 'rb'))
        # Load compressed code from binary file
        padding = text_code.pop('padding_length')
        text_bin = open(self.file + '.bin', 'rb').read()
        text_string = (''.join([bin(i)[2:].zfill(8) for i in text_bin]))[:-padding]
        return text_code, text_string

    def __to_tree(self):
        """
        Rebuilds the Huffman tree from the word-frequency dictionary.
        Returns the root of the Huffman tree.
        """
        root = Node(None)
        for char in self.text_code.keys():
            node = root
            for code in self.text_code[char]:
                if code == '0':
                    node.left = node.left if node.left else Node(None)
                    node = node.left
                else:
                    node.right = node.right if node.right else Node(None)
                    node = node.right
            node.char = char
        return root

    def __decompress(self):
        """
        Decompresses the text by traversing the Huffman tree.
        Returns the decompressed text.
        """
        final_text = ''
        node = self.root
        for i in self.text_string:
            node = node.left if i == "0" else node.right
            if node.char:
                final_text += node.char
                node = self.root
        # Save decompressed text to a file
        output_file = self.file + "-decompressed.txt"
        with open(output_file, 'w', encoding='utf-8', newline='\n') as f:
            f.write(final_text)
        return final_text

if __name__ == '__main__':
    # Parse command-line arguments
    parser = argparse.ArgumentParser()
    parser.add_argument('infile', type=str, help="Input file to decompress (with or without extension)")
    args = parser.parse_args()

    # Decompress the input file
    start_time = time.time()
    decompress = HuffmanDecoder(args.infile)
    end_time = time.time()
    print("Decoding time:", end_time - start_time)
