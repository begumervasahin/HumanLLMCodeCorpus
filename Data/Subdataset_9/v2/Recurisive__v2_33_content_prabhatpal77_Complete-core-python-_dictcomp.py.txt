# Create a dictionary 'x' where keys are p and values are p squared for p in the range from 0 to 9
x = {p: p*p for p in range(10)}
print(x)

# Create a dictionary 'y' where keys are q and values are q squared for q in the range from 10 to 19 if q is even
y = {q: q**2 for q in range(10, 20) if q % 2 == 0}
print(y)

# Create a dictionary 'z' where keys are r and values are r doubled for r in the range from 20 to 29 if r is odd
z = {r: r*2 for r in range(20, 30) if r % 2 != 0}
print(z)
