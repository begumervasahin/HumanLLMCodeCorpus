import csv
from selenium import webdriver
import time

# Proxy configuration
PROXY = "5.58.85.32:43214"
chrome_options = webdriver.ChromeOptions()
chrome_options.add_argument('--proxy-server=%s' % PROXY)
chrome_options.add_argument('--ignore_certificate-errors')
chrome_options.add_argument('--ignore-ssl-errors')

# Constants for page scraping
MAX_PAGE_NUM = 16
MAX_PAGE_DIG = 3

# Open CSV file for writing
with open("contacts.csv", "w", newline='', encoding='utf-8') as csv_file:
    # Create CSV writer object
    csv_writer = csv.writer(csv_file)
    # Write header row
    csv_writer.writerow(["Name", "Address", "Phone"])

    # Initialize Chrome WebDriver
    driver = webdriver.Chrome(options=chrome_options)

    # Iterate over page numbers
    for page_num in range(1, MAX_PAGE_NUM + 1):
        # Format page number with leading zeros
        formatted_page_num = str(page_num).zfill(MAX_PAGE_DIG)
        # Construct URL for current page
        url = f"https://www.pagesjaunes.fr/annuaire/chercherlespros?quoiqui=avocat%20droit%20du%20travail&ou=paris-75&proximite=0&quoiQuiInterprete=avocat%20droit%20du%20travail&contexte=V0RcB2CgpYFodNUZBe%2BLCQ%3D%3D&idOu=L07505600&page={formatted_page_num}"
        # Load the page
        driver.get(url)
        # Wait for page to load
        time.sleep(5)

        # Find all contact elements on the page
        contacts = driver.find_elements_by_xpath('//li[@class="bi-bloc blocs "]')
        
        # Iterate over each contact element
        for contact in contacts:
            # Extract name, address, and phone information
            name = contact.find_element_by_xpath('.//div[@class="denomination-links pj-lb pj-link"]').text
            address = contact.find_element_by_xpath('.//div[@class="adresse pj-lb pj-link"]').text
            phone = contact.find_element_by_xpath('.//span[@class="num"]').text
            # Write the extracted information to the CSV file
            csv_writer.writerow([name, address, phone])

    # Close the WebDriver
    driver.quit()

# Print a message indicating successful completion
print("Scraping finished. Data saved to contacts.csv")
