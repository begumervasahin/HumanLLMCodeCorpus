import sys
import numpy as np

# Global variable to count recursive calls
num_recursive_calls = 0

def memoization(x, y):
    """
    Computes the length of the longest common subsequence between strings x and y
    using a memoized algorithm.
    
    Args:
        x (str): The first string sequence.
        y (str): The second string sequence.
    
    Returns:
        int: The length of the longest common subsequence.
    """
    # Initialize a memoization table
    m = len(x)
    n = len(y)
    c = np.ones(shape=(m, n), dtype='int64') * -1
    return memoized_LCS(x, y, c)

def memoized_LCS(x, y, c):
    """
    Helper function for memoization.
    Recursively computes the length of the longest common subsequence between strings x and y.
    
    Args:
        x (str): The first string sequence.
        y (str): The second string sequence.
        c (numpy.ndarray): A memoization table.
    
    Returns:
        int: The length of the longest common subsequence.
    """
    global num_recursive_calls
    num_recursive_calls += 1
    
    # Base case: if either sequence is empty, return 0
    if len(x) == 0 or len(y) == 0:
        return 0
    
    # Indices for accessing the memoization table
    i = len(x) - 1
    j = len(y) - 1
    
    # If the subproblem is already solved, return the value from the memoization table
    if c[i, j] > -1:
        return c[i, j]
    else:
        # If the last characters match, recursively compute LCS for the remaining substrings
        if x[-1] == y[-1]:
            q = 1 + memoized_LCS(x[:-1], y[:-1], c)
        # If the last characters don't match, explore both possibilities and take the maximum
        else:
            q = max(memoized_LCS(x, y[:-1], c), memoized_LCS(x[:-1], y, c))  
        
        # Memoize the result and return
        c[i, j] = q
        return q

def test(x, y):
    """
    Computes the length of the longest common subsequence between strings x and y.
    
    Args:
        x (str): The first string pattern.
        y (str): The second string pattern.
    
    Returns:
        int: The length of the longest common subsequence.
    """
    return memoization(x, y)

if __name__ == "__main__":
    # Check if command-line arguments are provided
    if len(sys.argv) < 2:
        print("Usage: python3 " + sys.argv[0] + " acgt/bits")
        print("acgt - generate sequences of acgt")
        print("bits - generate sequences of binary digits")
    else:
        # Read input sequences from file
        file = sys.argv[1] + ".txt"
        with open(file, 'r') as f:
            x = f.readline().strip()
            y = f.readline().strip()
            
            # Compute LCS and print results
            lcs_length = test(x, y)
            print("Length of LCS: " + str(lcs_length))
            print("Number of recursive calls: " + str(num_recursive_calls))
