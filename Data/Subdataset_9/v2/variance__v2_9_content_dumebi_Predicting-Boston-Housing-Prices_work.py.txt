import numpy as np
import pandas as pd
from sklearn.tree import DecisionTreeRegressor
from sklearn.model_selection import train_test_split

# Function to load data from CSV file
def load_data(file_path):
    # Read data from CSV file
    work_data = pd.read_csv(file_path)
    # Print dataset information
    print("The dataset contains {} data points with {} variables each.".format(*work_data.shape))
    return work_data

# Function to extract features from data
def get_features(data):
    # Exclude non-feature columns from the data
    features = data.drop(['Date', 'Time', 'Close', 'High', 'Low', 'Total Ticks'], axis=1)
    return features

# Function to extract target variable (Close price) from data
def get_close(data):
    # Extract the 'Close' column from the data
    close = data['Close']
    return close

# Function to predict price movement (up or down) based on opening value
def predict(open_value, file_path):
    # Load data from CSV file
    data = load_data(file_path)
    # Extract features and target variable
    features = get_features(data)
    close = get_close(data)
    
    # Split the data into training and testing sets
    X_train, X_test, y_train, y_test = train_test_split(features, close, test_size=0.4, shuffle=True, random_state=101)
    
    # Initialize Decision Tree Regressor model
    regressor = DecisionTreeRegressor(random_state=0)
    # Fit the model on training data
    regressor.fit(X_train, y_train)
    
    # Prepare test data (opening value)
    test = np.array([[open_value]])
    # Make prediction
    pred = regressor.predict(test)
    
    # Determine price movement based on prediction
    if pred > open_value:
        return "Predicted movement: up"
    else:
        return "Predicted movement: down"
