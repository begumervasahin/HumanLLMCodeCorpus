# Importing the sys module to access system-specific parameters and functions
import sys

# Defining a class named Node
class Node(object):
    # Constructor method to initialize the object
    def __init__(self, x, y, i):
        # Assigning values to instance variables
        self.x = x  # x-coordinate
        self.y = y  # y-coordinate
        self.idx = i  # index
        # Initializing the key attribute with the maximum size of an integer
        self.key = sys.maxsize  
        # Initializing the parent attribute to None
        self.parent = None

    # Method to represent the object as a string
    def __str__(self):
        # Returning a string representation of the node
        return str(self.idx) + ': (' + str(self.x) + ',' + str(self.y) + ')'

# Example usage:
# Creating a Node object with coordinates (1, 2) and index 0
node1 = Node(1, 2, 0)
# Printing the string representation of the node (index and coordinates)
print(node1)  # Output: 0: (1,2)
# Printing the value of the key attribute of the node
print("Key:", node1.key)  # Output: Key: 9223372036854775807 (maximum size of an integer)
