import time

# Function to add corresponding elements of two lists
def add(list1, list2):
    result = []
    for i in range(len(list1)):
        result.append(list1[i] + list2[i])
    return result

# Generating two lists of consecutive numbers
nums1 = range(0, 9999999)
nums2 = range(100, 10000100)

# Measuring the start time
start_time = time.time()

# Adding the two lists
result = add(nums1, nums2)

# Calculating the elapsed time
elapsed_time = time.time() - start_time

# Printing the elapsed time and the length of the result list
print("Time elapsed:", elapsed_time)
print("Result list length:", len(result))

# Printing numbers in the result list that are divisible by 777777
for item in result:
    if item % 777777 == 0:
        print(item)
