import unittest
from fizzbuzz import Fizzbuzz

class FizzbuzzTest(unittest.TestCase):
    def test_invalid_input_both_non_list(self):
        # Test Fizzbuzz function with non-list inputs
        result = Fizzbuzz(5, 8)
        self.assertEqual(result, "Invalid input")

    def test_invalid_input_one_non_list(self):
        # Test Fizzbuzz function with one non-list input
        result = Fizzbuzz([2, 3, 4], 7)
        self.assertEqual(result, "Invalid input")

    def test_fizz(self):
        # Test Fizzbuzz function for 'fizz' output
        result = Fizzbuzz([1, 3, 4], ['a', 'b', 'c'])
        self.assertEqual(result, "fizz")

    def test_buzz(self):
        # Test Fizzbuzz function for 'buzz' output
        result = Fizzbuzz([4, 5, 4], [3, 4])
        self.assertEqual(result, "buzz")

    def test_fizz_buzz(self):
        # Test Fizzbuzz function for 'Fizzbuzz' output
        result = Fizzbuzz([1, 2, 3, 3, 4, 5, 5, 5, 5, 6], [7, 7, 4, 5, 9])
        self.assertEqual(result, "Fizzbuzz")

    def test_other_total(self):
        # Test Fizzbuzz function for other total length
        result = Fizzbuzz([4, 5, 6], [3])
        self.assertEqual(result, 4)

    def test_both_empty(self):
        # Test Fizzbuzz function when both lists are empty
        result = Fizzbuzz([1, 3, 5], [])
        self.assertEqual(result, "fizz")

if __name__ == '__main__':
    unittest.main()
