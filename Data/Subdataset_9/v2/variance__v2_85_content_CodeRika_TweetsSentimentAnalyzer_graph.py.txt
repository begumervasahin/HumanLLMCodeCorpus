# Import the necessary modules from Plotly for plotting and handling CSV files.
import plotly.graph_objs as go
import plotly.offline as py_offline
import csv

# Set your Plotly account credentials (replace with your actual username and API key).
plotly.tools.set_credentials_file(username='YOUR_USERNAME', api_key='YOUR_API_KEY')

# Initialize lists to store the data for cities and their sentiment scores.
city_list = []
positive_list = []
negative_list = []
neutral_list = []

# Open and read the CSV file containing the sentiment values.
with open('sentimentValueList.csv', 'r') as cityData:
    csv_reader = csv.reader(cityData, delimiter=',')
    next(csv_reader)  # Skip the header row.
    for row in csv_reader:
        city_list.append(row[0])  # City names.
        positive_list.append(float(row[1]))  # Positive sentiment scores.
        negative_list.append(float(row[2]))  # Negative sentiment scores.
        neutral_list.append(float(row[3]))  # Neutral sentiment scores.

# Define the data for the positive sentiment plot.
trace_positive = go.Scatter(
    x=city_list,
    y=positive_list,
    name='Positive Sentiment Values',
    line=dict(color='rgb(205, 12, 24)', width=2)
)

# Define the data for the negative sentiment plot.
trace_negative = go.Scatter(
    x=city_list,
    y=negative_list,
    name='Negative Sentiment Values',
    line=dict(color='rgb(22, 96, 167)', width=2)
)

# Define the data for the neutral sentiment plot.
trace_neutral = go.Scatter(
    x=city_list,
    y=neutral_list,
    name='Neutral Sentiment Values',
    line=dict(color='rgb(0, 0, 0)', width=2, dash='dash')
)

# Group the traces into a list for plotting.
data = [trace_positive, trace_negative, trace_neutral]

# Define the layout of the plot.
layout = dict(
    title='Tweet Sentiment Analysis',
    xaxis=dict(title='City'),
    yaxis=dict(title='Sentiment Score (%)')
)

# Combine the data and layout into a figure.
fig = dict(data=data, layout=layout)

# Plot the figure and save it as an HTML file.
py_offline.plot(fig, filename='tweet_sentiment_analysis.html')
