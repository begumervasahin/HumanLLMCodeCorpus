def calculate_yearly_balance(balance, annual_interest):
    """
    Calculates the minimum monthly payment needed to pay off a yearly debt with annual interest.
    """
    unpaid_balance = balance
    monthly_payment = 0
    
    while unpaid_balance >= 0:
        unpaid_balance = balance
        
        for _ in range(12):
            unpaid_balance -= monthly_payment
            interest_amount = (annual_interest / 12) * unpaid_balance
            unpaid_balance += interest_amount
        
        monthly_payment += 10
    
    return monthly_payment - 10

# Example usage:
balance_input = float(input("Enter the total debt for the year: "))
interest_input = float(input("Enter the annual interest rate (as a decimal): "))

result = calculate_yearly_balance(balance_input, interest_input)
print(f"The minimum monthly payment required to pay off the debt in a year is: ${result}")
