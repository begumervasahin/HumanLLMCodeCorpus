import sys

def read_matrix(file_path):
    with open(file_path, 'r') as file:
        content = file.readlines()
    
    matrix = []
    for line in content:
        row = [int(cell) for cell in line.strip().split()]
        matrix.append(row)
    
    return matrix

def find_adjacent_nodes(matrix):
    adjacent_nodes = []
    for i in range(len(matrix)):
        adjacent_nodes.append([(j, matrix[i][j]) for j in range(len(matrix)) if matrix[i][j] != 1024])
    return adjacent_nodes

def dijkstra(adjacent_nodes, start, end):
    visited_nodes = [0] * len(adjacent_nodes)
    solution = []
    local = [0] * len(adjacent_nodes)

    current_node = int(start)
    current_distance = 0
    previous_node = ""
    previous_distance = 0
    next_node = ""
    next_distance = ""

    solution.append(f'- {start} 0')
    visited_nodes[int(start)] = 1

    while visited_nodes[int(end)] != 1:
        next_node = adjacent_nodes[current_node][0][0]
        next_distance = int(adjacent_nodes[current_node][0][1]) + current_distance

        for i in range(1, len(adjacent_nodes[current_node])):
            if visited_nodes[int(next_node)] != 1:
                if int(next_node) == int(end):
                    previous_node = current_node
                else:
                    adj = adjacent_nodes[current_node][i][0]
                    if visited_nodes[int(adj)] != 1:
                        next_distance_adj = int(adjacent_nodes[current_node][i][1]) + current_distance
                        if next_distance_adj < next_distance:
                            next_node = adj
                            next_distance = next_distance_adj
                            previous_node = current_node
            elif previous_node == int(start) or visited_nodes[int(next_node)] == 1:
                next_node = adjacent_nodes[current_node][i][0]
                next_distance = int(adjacent_nodes[current_node][i][1]) + current_distance

        if previous_node == "":
            previous_node = current_node

        previous_distance = int(solution[local[int(previous_node)]][0].split(" ")[2])

        if len(solution) > 2 and previous_node != int(start) and int(next_node) != int(end):
            for i in range(len(adjacent_nodes[int(previous_node)])):
                adj = adjacent_nodes[int(previous_node)][i][0]
                if visited_nodes[int(adj)] != 1:
                    next_distance_adj = int(adjacent_nodes[int(previous_node)][i][1]) + previous_distance
                    if next_distance_adj < next_distance:
                        next_node = adjacent_nodes[int(previous_node)][i][0]
                        next_distance = next_distance_adj

        solution.append(f'{previous_node} {next_node} {next_distance}')
        visited_nodes[int(next_node)] = 1
        local[int(next_node)] = len(solution) - 1
        current_node = int(next_node)
        current_distance = int(next_distance)

    return solution

def display_solution(solution, start, end):
    if start == end:
        print(start)
    else:
        path = ""
        path += solution[-1][0].split(" ")[1] + " "
        after = solution[-1][0].split(" ")[0]

        while int(after) != int(start):
            for i in range(1, len(solution) - 1):
                if solution[i][0].split(" ")[1] == after:
                    path += solution[i][0].split(" ")[1] + " "
                    after = solution[i][0].split(" ")[0]

        path += solution[0][0].split(" ")[1]
        print(path[::-1])

if __name__ == "__main__":
    file_path, start_node, end_node = sys.argv[1:]
    adjacent_nodes = find_adjacent_nodes(read_matrix(file_path))
    solution = dijkstra(adjacent_nodes, start_node, end_node)
    display_solution(solution, start_node, end_node)
