import time

def merge_sort(arr):
    """
    Sorts a list of integers using the merge sort algorithm.

    Args:
        arr (list): The list of integers to be sorted.
    """
    if len(arr) > 1:
        mid = len(arr) // 2
        L = arr[:mid]
        R = arr[mid:]
        merge_sort(L)
        merge_sort(R)

        # Merge the sorted sublists
        i = j = k = 0
        while i < len(L) and j < len(R):
            if L[i] < R[j]:
                arr[k] = L[i]
                i += 1
            else:
                arr[k] = R[j]
                j += 1
            k += 1

        # Add remaining elements from left and right sublists
        while i < len(L):
            arr[k] = L[i]
            i += 1
            k += 1
        while j < len(R):
            arr[k] = R[j]
            j += 1
            k += 1

def print_list(arr):
    """
    Prints the elements of a list.

    Args:
        arr (list): The list to be printed.
    """
    for num in arr:
        print(num, end=" ")
    print()

if __name__ == '__main__':
    start_time = time.time()
    count = 0

    # Read input from a file
    with open("case", "r") as fp:
        arr = fp.read().split()
        list2 = [int(x) for x in arr]

    # Sort the list using merge sort
    merge_sort(list2)

    # Print the sorted array and total comparisons
    print("Sorted array:")
    print_list(list2)
    print("Total comparisons:", count)

    end_time = time.time()
