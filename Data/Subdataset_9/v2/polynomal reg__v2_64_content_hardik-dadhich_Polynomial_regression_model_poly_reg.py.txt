# Importing necessary libraries
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import PolynomialFeatures

# Load the dataset
# The dataset contains information about positions and corresponding salaries
dataset = pd.read_csv('Position_Salaries.csv')

# Extracting the features (Position Level) and target variable (Salary)
X = dataset.iloc[:, 1:2].values  # Feature: Position Level
y = dataset.iloc[:, 2].values    # Target: Salary

# Fitting Linear Regression to the dataset
# Linear Regression models the relationship between the position level and salary linearly
lin_reg = LinearRegression()
lin_reg.fit(X, y)

# Fitting Polynomial Regression to the dataset
# Polynomial Regression models the relationship between position level and salary with polynomial terms
poly_reg = PolynomialFeatures(degree=2)  # Using polynomial features up to degree 2
X_poly = poly_reg.fit_transform(X)       # Transforming features to include polynomial terms
lin_reg_2 = LinearRegression()
lin_reg_2.fit(X_poly, y)

# Visualizing the Linear Regression results
# Plotting the original data points and the linear regression line
plt.scatter(X, y, color='red')               # Plotting the original data points
plt.plot(X, lin_reg.predict(X), color='blue')# Plotting the linear regression line
plt.title('Truth or Bluff (Linear Regression)')  # Title of the plot
plt.xlabel('Position level')                  # X-axis label
plt.ylabel('Salary')                          # Y-axis label
plt.show()                                    # Displaying the plot

# Visualizing the Polynomial Regression results
# Plotting the original data points and the polynomial regression curve
plt.scatter(X, y, color='red')                         # Plotting the original data points
plt.plot(X, lin_reg_2.predict(poly_reg.fit_transform(X)), color='blue') # Plotting the polynomial regression curve
plt.title('Truth or Bluff (Polynomial Regression)')    # Title of the plot
plt.xlabel('Position level')                            # X-axis label
plt.ylabel('Salary')                                    # Y-axis label
plt.show()                                              # Displaying the plot

# Predicting a new result with Linear Regression
# Predicting the salary for a new position level using linear regression model
predicted_salary_linear = lin_reg.predict([[6.5]])[0]  
print("Predicted salary with Linear Regression for position level 6.5:", predicted_salary_linear)

# Predicting a new result with Polynomial Regression
# Predicting the salary for a new position level using polynomial regression model
predicted_salary_poly = lin_reg_2.predict(poly_reg.fit_transform([[6.5]]))[0]
print("Predicted salary with Polynomial Regression for position level 6.5:", predicted_salary_poly)
