import sys

class Node:
    def __init__(self, key, idx):
        self.key = key  # The value associated with the node
        self.idx = idx  # The index of the node

class MinQueue:
    def __init__(self, nodes):
        # Initialize the MinQueue with a list of nodes
        self.nodes = nodes[:]

    def __len__(self):
        # Returns the number of nodes in the MinQueue
        return len(self.nodes)

    def contains(self, key):
        # Checks if the MinQueue contains a node with a given key
        for node in self.nodes:
            if node.key == key:
                return True
        return False

    def extractMin(self):
        # Extracts the node with the minimum key value from the MinQueue
        min_key = sys.maxsize
        min_node = None

        # Find the node with the minimum key value
        for node in self.nodes:
            if node.key < min_key:
                min_key = node.key
                min_node = node

        # Remove the minimum node from the MinQueue
        self.nodes.remove(min_node)

        # Return the index of the minimum node
        return min_node.idx

# Example usage:
if __name__ == "__main__":
    # Create some nodes
    nodes = [Node(5, 0), Node(3, 1), Node(7, 2), Node(1, 3)]

    # Create a MinQueue instance
    min_queue = MinQueue(nodes)

    # Test functionality
    print("Initial queue length:", len(min_queue))
    print("Does queue contain node with key 7?", min_queue.contains(7))
    print("Extracted minimum node index:", min_queue.extractMin())    
    print("Queue length after extraction:", len(min_queue))
