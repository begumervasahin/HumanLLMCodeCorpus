class TOperation:
    """
    Represents a single operation performed in the program.

    Attributes:
        o_type (str): The type of operation.
        shape (object): The shape on which the operation was performed.        
        num (int): The index of the shape in the shape list.
    """

    def __init__(self, o_type=None, shape=None, num=None):
        """
        Initializes the TOperation object.

        Args:
            o_type (str): The type of operation.
            shape (object): The shape on which the operation was performed.
            num (int): The index of the shape in the shape list.
        """
        self.o_type = o_type
        self.shape = shape
        self.num = num

# Example usage:
# Create a TOperation object
operation = TOperation(o_type="resize", shape="rectangle", num=1)

# Print the attributes of the TOperation object
print("Operation Type:", operation.o_type)
print("Shape:", operation.shape)
print("Number:", operation.num)
