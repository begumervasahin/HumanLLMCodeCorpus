# Importing necessary functions from the module "myFunctions"
from myFunctions import get_three_random_numbers, random_case_algorithm
import random

# Number of iterations for the experiment
iterations = 10000

# Initializing variables to keep track of attempts
attempts = 0
highest_attempt = 0
lowest_attempt = 1000  # Initializing with a high value

# Generating three random numbers
three_random_numbers = get_three_random_numbers()

# Looping through the specified number of iterations
for x in range(iterations):
    # Running the random case algorithm with the generated numbers
    iteration = random_case_algorithm(three_random_numbers)
    
    # Adding the attempts from this iteration to the total attempts
    attempts += iteration
    
    # Updating the highest and lowest attempt counts
    if iteration > highest_attempt:
        highest_attempt = iteration
    if iteration < lowest_attempt:
        lowest_attempt = iteration

# Printing the results
print("Number of Tries:", iterations)
print("Highest number of guesses in a try:", highest_attempt)
print("Lowest tries:", lowest_attempt)
print("Number of Correct tries:", iterations)
print("Average number of tries:", f"{attempts}/{iterations}: {attempts/iterations}")
