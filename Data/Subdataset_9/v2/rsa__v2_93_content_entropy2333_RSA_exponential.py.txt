import time

def montgomery_multiplication(a, b, n):
    # Initialize reverse
    reverse = 1
    
    # Calculate r = 2^(n-1)
    r = pow(2, n - 1)
    
    # Find the reverse value using Montgomery's method
    for i in range(1, n + 1):
        if (reverse * r) % n == 1:
            pass
        else:
            reverse += 1
    
    # Calculate the result using Montgomery multiplication
    result = (a * b * reverse) % n
    return result

def modular_exponentiation(base, exponent, n):
    # Initialize i to exponent - 1
    i = exponent - 1
    
    # Calculate x using Montgomery's method for initial value
    x = montgomery_multiplication(base, 1, n)
    
    # Perform modular exponentiation
    while i > 0:
        x = montgomery_multiplication(x, 1, n)
        i -= 1
    
    return x
