class Node:
    def __init__(self, value):
        """
        Initialize a Node object with its properties.

        Parameters:
        value: The value of the node.
        """
        self.value = value
        self.parent = self  # Initially, the node is its own parent
        self.rank = 0  # Initially, the node's rank is 0

    def __repr__(self):
        """
        Represent the Node object as a string.

        Returns:
        str: A string representation of the Node object.
        """
        return f"Node(value={self.value}, rank={self.rank}, parent={self.parent.value})"

# Example usage
node1 = Node('A')
node2 = Node('B')
node3 = Node('C')

print(node1)  # Output: Node(value=A, rank=0, parent=A)
print(node2)  # Output: Node(value=B, rank=0, parent=B)
print(node3)  # Output: Node(value=C, rank=0, parent=C)
