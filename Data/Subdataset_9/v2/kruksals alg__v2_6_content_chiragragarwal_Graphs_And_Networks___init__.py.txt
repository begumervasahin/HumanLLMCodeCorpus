# Import the PlotGraph and ReadGraph classes
from PlotGraph import PlotGraph
from ReadGraph import ReadGraph

# Create an instance of ReadGraph to read the graph data from 'graph_data.txt'
graph_reader = ReadGraph('graph_data.txt')

# Read the graph data from the file
graph_data = graph_reader.read()

# Create an instance of PlotGraph with the graph data
graph_plotter = PlotGraph(graph_data)

# Plot the graph using the provided data
graph_plotter.plot()
