import os
import pandas as pd
import csv
import re

# Directory where data is stored
DATA_DIR = "data"
# Path to the CSV file containing MBTI data
MBTI_CLEAN_CSV_PATH = os.path.join(DATA_DIR, "mbti_clean.csv")
# MBTI dimensions
DIMENSIONS = ("IE", "NS", "TF", "PJ")

# Read the MBTI data CSV file into a DataFrame
df = pd.read_csv(MBTI_CLEAN_CSV_PATH)

# Function to filter and write posts to a CSV file
def write_posts_to_csv(posts, letter):
    # Path to the CSV file to write posts for a specific letter
    test_csv_path = os.path.join(DATA_DIR, f"test_{letter}.csv")
    # Write posts to the CSV file
    with open(test_csv_path, "w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        for post in posts:
            writer.writerow([post])

# Iterate over each MBTI dimension
for dimension in DIMENSIONS:
    # Split the dimension into two letters
    letter_1, letter_2 = dimension
    # Iterate over each letter in the dimension
    for letter in [letter_1, letter_2]:
        posts = []  # List to store posts for the current letter
        # Iterate over each row (profile) in the MBTI data
        for index, row in df.iterrows():
            # Check if the current profile contains the letter
            if letter in row["type"]:
                # Split the posts into individual posts
                hundred_posts = row["posts"].split("|||")
                # Filter and append posts
                for post in hundred_posts:
                    # Exclude posts containing URLs, empty posts, or non-alphabetic posts
                    if ("http" in post) or (post == "") or (post is None) or (not re.search("[a-zA-Z]", post)):
                        continue
                    posts.append(post)
        # Write filtered posts to a CSV file for the current letter
        write_posts_to_csv(posts, letter)

print("CSV files generated successfully.")
