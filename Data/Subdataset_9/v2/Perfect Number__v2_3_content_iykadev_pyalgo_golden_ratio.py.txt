import cProfile

def compute_golden_ratio_sequence(n):
    """Compute and print the golden ratio sequence up to the nth term."""
    s = 1
    t = 1
    for i in range(2, n):
        c = s + t
        s = t
        t = c
        ratio = t / s
        print(ratio)

def main():
    """Main function to compute and print the golden ratio sequence."""
    n = 1476  # Number of terms in the sequence
    compute_golden_ratio_sequence(n)

if __name__ == "__main__":
    # Profile the main function and save results to a file
    cProfile.run("main()", filename="profile_results.txt")
