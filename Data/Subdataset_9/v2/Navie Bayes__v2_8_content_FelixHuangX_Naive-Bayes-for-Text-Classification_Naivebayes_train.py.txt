import numpy as np
from participle import fenci
from wordcloud import WordCloud
import matplotlib.pyplot as plt

def load_data_set(raw_path, stop_path):
    result, data_target = fenci(raw_path, stop_path)
    return result, data_target

def create_unique_set(dataset):
    unique_set = {}
    for data in dataset:
        for word in data:
            if word in unique_set:
                unique_set[word] += 1
            else:
                unique_set[word] = 1
    unique_set = {k: v for k, v in unique_set.items() if v > 50}
    unique_words = list(unique_set.keys())
    unique_words.sort(key=lambda x: unique_set[x], reverse=True)
    return unique_words

def words_to_vec(unique_set_list, input_doc):
    return_vec = [0] * len(unique_set_list)
    for word in input_doc:
        if word in unique_set_list:
            return_vec[unique_set_list.index(word)] += 1
    return return_vec

def train_naive_b(train_mat, train_class):
    num_train_docs = len(train_mat)
    num_words = len(train_mat[0])
    data_class = set(train_class)
    p_target = []
    p_vec = []
    for cls in data_class:
        pos_sample_rate = train_class.count(cls) / float(num_train_docs)
        p_target.append(pos_sample_rate)
        p_num = np.ones(num_words)
        p_sum = 2.0
        for j in range(num_train_docs):
            if train_class[j] == cls:
                p_num += train_mat[j]
                p_sum += sum(train_mat[j])
        p_vec.append(np.log(p_num / p_sum))
    return p_vec, p_target

if __name__ == "__main__":
    labels = ['Sport', 'Entertainment', 'Household', 'House Property', 'Education', 'Fashion', 'Current Politics', 'Game', 'Science and Technology', 'Finance']
    stop_path = r'./cnews/cnews.vocab.txt'
    train_path = r'./cnews/cnews.train.txt'
    test_path = r'./cnews/cnews.test.txt'
    raw_features, raw_classes = load_data_set(train_path, stop_path)
    unique_set_list = create_unique_set(raw_features)
    train_mat = [words_to_vec(unique_set_list, single_features) for single_features in raw_features]
    p_vec, pos = train_naive_b(np.array(train_mat), raw_classes)
    np.save('models/pos.npy', np.array(pos))
    with open('models/unique_set_list.txt', 'w') as file:
        for word in unique_set_list:
            file.write(word + '\n')
    np.save('models/pV.npy', np.array(p_vec))
