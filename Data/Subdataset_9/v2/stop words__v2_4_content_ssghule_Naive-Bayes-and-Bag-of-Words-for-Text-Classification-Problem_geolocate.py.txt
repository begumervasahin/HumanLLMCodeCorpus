import pandas as pd
import string
import nltk
from nltk.tokenize import word_tokenize
from nltk.stem.snowball import SnowballStemmer
from nltk.corpus import stopwords
import sys

# Download NLTK resources
nltk.download('stopwords')
nltk.download('punkt')

def parse_file(file_path, city_list):
    with open(file_path, 'r') as file:
        content = file.readlines()
    
    words = [word_tokenize(line) for line in content]
    stop = stopwords.words('english')
    punct_stopwords = list(string.punctuation) + ['i', '\x89', '_CA', '_TX', '_IL', '_NY', '_PA', '_GA', '_Ontario', '_MA', '_FL', '_DC', '__', '___'] + stop
    stemmer = SnowballStemmer("english")
    
    filtered_words = [[stemmer.stem(w) for w in item if w not in punct_stopwords] for item in words if len(item) > 1]
    return [word for word in filtered_words if len(word) > 1 and word[0] in city_list]

def create_training_dataframe(parsed_data, city_list):
    word_list = set([w for word in parsed_data for w in word])
    data = pd.DataFrame(0, index=word_list, columns=city_list)
    
    for word in parsed_data:
        current_city = word[0]
        for w in word[1:]:
            data.at[w, current_city] += 1
            
    return data

def create_test_dictionary(file_path):
    with open(file_path, 'r') as file:
        content = file.readlines()
    
    test_dict = {}
    stop = stopwords.words('english')
    punct_stopwords = list(string.punctuation) + ['i', '\x89', '_CA', '_TX', '_IL', '_NY', '_PA', '_GA', '_Ontario', '_MA', '_FL', '_DC', '__', '___'] + stop
    stemmer = SnowballStemmer("english")
    
    for line in content:
        words = word_tokenize(line)
        stemmed_words = [stemmer.stem(w) for w in words if w not in punct_stopwords]
        if stemmed_words:
            stemmed_words.pop(0)
        test_dict[line] = stemmed_words
    
    return test_dict

def calculate_city_probability(parsed_data, city_prob):
    for word in parsed_data:
        city_prob[word[0]] += 1
    
    total = sum(city_prob.values())
    for city in city_prob:
        city_prob[city] = float(city_prob[city] / total)

def apply_bayes_classification(tdf, test_data, output_file, city_probabilities, city_names, city_codes):
    with open(output_file, 'w') as file:
        for tweet, words in test_data.items():
            if not words:
                continue
            else:
                word_posteriors = [[(tdf.at[word, city] / sum(tdf.loc[:, city]) * city_probabilities[city]) if word in tdf.index else 0.00001 * city_probabilities[city] for city in city_codes] for word in words]
                posterior_probabilities = word_posteriors[0]
                for i in range(1, len(word_posteriors)):
                    posterior_probabilities = [x * y for x, y in zip(posterior_probabilities, word_posteriors[i])]
                
                predicted_city = city_names[posterior_probabilities.index(max(posterior_probabilities))]
                file.write(predicted_city + " " + tweet)
    
    print("Output written to ", output_file)

if __name__ == "__main__":
    # Command line arguments: input_file1.txt input_file2.txt output_file.txt
    input_file = str(sys.argv[1])
    test_file = str(sys.argv[2])
    output_file = str(sys.argv[3])
    
    # City information
    city_names = ['Los_Angeles,_CA', 'San_Francisco,_CA', 'San_Diego,_CA', 'Houston,_TX', 'Chicago,_IL', 'Philadelphia,_PA', 'Toronto,_Ontario', 'Atlanta,_GA', 'Boston,_MA', 'Orlando,_FL', 'Washington,_DC', 'Manhattan,_NY']
    city_codes = ["los_angel", "san_francisco", "san_diego", "houston", "chicago", "philadelphia", "toronto", "atlanta", "boston", "orlando", "washington", "manhattan"]
    city_probabilities = dict.fromkeys(city_codes, 0)
    
    # Parse training data
    parsed_training_data = parse_file(input_file, city_codes)
    
    # Create training DataFrame
    training_dataframe = create_training_dataframe(parsed_training_data, city_codes)
    
    # Parse test data
    test_data = create_test_dictionary(test_file)
    
    # Calculate city probabilities
    calculate_city_probability(parsed_training_data, city_probabilities)
    
    # Apply Naive Bayes classification
    apply_bayes_classification(training_dataframe, test_data, output_file, city_probabilities, city_names, city_codes)
    
    print('Top five words for each city: ')
    for city in city_codes:
        print(city_names[city_codes.index(city)], training_dataframe.nlargest(5, city).index.tolist())
