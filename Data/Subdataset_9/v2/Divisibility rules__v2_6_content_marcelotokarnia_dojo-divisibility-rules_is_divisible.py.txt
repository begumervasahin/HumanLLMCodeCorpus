# Define a generator function to generate the alternating pattern for the divisibility rule
def div_rule(m):
    current_index = 0
    while True:
        yield (-1) ** current_index  # Yield alternating 1 and -1
        current_index += 1

# Define a function to check if a number is divisible by another number using a specific divisibility rule
def is_divisible(n, m):
    # Create an instance of the divisibility rule generator
    divisibility_rule = div_rule(m)
    
    # Initialize the total sum to track the intermediate result
    total_sum = 0
    
    # Convert the number to a string and reverse it for processing digit by digit
    n = str(n)[::-1]
    
    # Iterate through each digit of the reversed number
    for digit in n:
        # Get the next multiplier from the divisibility rule generator
        multiplier = next(divisibility_rule)
        
        # Multiply the digit by the multiplier and add it to the total sum
        total_sum += int(digit) * multiplier
        
        # If the absolute value of the total sum exceeds 3 times the divisor, reduce it to a manageable range
        if abs(total_sum) > 3 * m:
            total_sum %= 3 * m
    
    # Check if the final total sum is divisible by the divisor and return True or False accordingly
    return total_sum % m == 0

# Example usage:
number = 123456  # The number to check for divisibility
divisor = 7      # The divisor to check against
result = is_divisible(number, divisor)  # Check if number is divisible by divisor using the defined rule
print(result)  # Output: True or False depending on whether number is divisible by divisor
