from Memoria import Cerebro

class Memorization:
    def __init__(self):
        # Set up memory with a limit and cleaning threshold
        self.memoria = Cerebro(10000000, 0.7)

    def calc(self, n):
        # Base cases for Fibonacci sequence
        if n == 0:
            return 0
        if n == 1:
            return 1

        # Convert n-1 and n-2 to strings for memoization
        str_n1 = str(n - 1)
        str_n2 = str(n - 2)

        # Retrieve values from memory if available
        n_1 = self.memoria.get_lembranca(str_n1)
        n_2 = self.memoria.get_lembranca(str_n2)

        # Calculate Fibonacci numbers recursively if not in memory
        if n_1 == -1:
            n_1 = self.calc(n - 1)
            self.memoria.nova_memoria(str_n1, n_1)

        if n_2 == -1:
            n_2 = self.calc(n - 2)
            self.memoria.nova_memoria(str_n2, n_2)

        # Return the sum of Fibonacci numbers
        return n_1 + n_2

    def get_funcao_calc(self):
        # Return a lambda function for calculating Fibonacci numbers
        return lambda n: self.calc(n)

    def __str__(self):
        return "Recursivo + Memoization"

if __name__ == "__main__":
    # Create an instance of Memorization class
    memorization = Memorization()
    
    # Input the number for which Fibonacci number is to be calculated
    n = int(input("Enter a number: "))
    
    # Calculate and print the Fibonacci number
    result = memorization.calc(n)
    print("Result:", result)
