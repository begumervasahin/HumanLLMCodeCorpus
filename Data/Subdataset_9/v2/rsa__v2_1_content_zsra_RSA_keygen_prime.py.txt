import random
import xgcd

# Function to get a random prime number within a given range
def get_random_prime(min_value, max_value):
    # Generate a list of prime numbers within the specified range
    primes = [num for num in range(min_value, max_value) if is_prime(num)]
    # Choose a random prime number from the list
    return random.choice(primes)

# Function to check if a number is prime using the Miller-Rabin primality test
def is_prime(number):
    # Special cases: 0, 1, 4, 6, 8, 9 are not prime
    if number in {0, 1, 4, 6, 8, 9}:
        return False
    # Special cases: 2, 3, 5, 7 are prime
    if number in {2, 3, 5, 7}:
        return True
    
    # Decompose n-1 into the form 2^s * d
    d = number - 1
    s = 0
    while d % 2 == 0:
        d //= 2
        s += 1

    # Miller-Rabin primality test
    def is_composite(a):
        if pow(a, d, number) == 1:
            return False
        for i in range(s):
            if pow(a, 2 ** i * d, number) == number - 1:
                return False
        return True

    # Repeat the test multiple times for accuracy
    for _ in range(8):
        a = random.randrange(2, number)
        if is_composite(a):
            return False
    return True

# Function to check if two numbers are coprime using the Extended Euclidean Algorithm
def are_coprime(a, b):
    # If the greatest common divisor is 1, the numbers are coprime
    if xgcd.GCD(a, b) == 1:
        return True
    else:
        return False
