import numpy
from numpy.random import randint as rand

def generate_maze(width=81, height=51, complexity=.75, density=.75):
    # Ensure odd dimensions
    shape = ((height // 2) * 2 + 1, (width // 2) * 2 + 1)
    
    # Adjust complexity and density based on user input
    complexity = int(complexity * (5 * (shape[0] + shape[1])))
    density = int(density * ((shape[0] // 2) * (shape[1] // 2)))
    
    # Initialize maze grid
    maze_grid = numpy.zeros(shape, dtype=bool)
    
    # Set outer walls
    maze_grid[0, :] = maze_grid[-1, :] = 1
    maze_grid[:, 0] = maze_grid[:, -1] = 1
    
    # Carve out the maze
    for _ in range(density):
        x, y = rand(0, shape[1] // 2) * 2, rand(0, shape[0] // 2) * 2
        maze_grid[y, x] = 1
        
        for _ in range(complexity):
            neighbours = []
            if x > 1:             neighbours.append((y, x - 2))
            if x < shape[1] - 2:  neighbours.append((y, x + 2))
            if y > 1:             neighbours.append((y - 2, x))
            if y < shape[0] - 2:  neighbours.append((y + 2, x))
            
            if len(neighbours):
                y_, x_ = neighbours[rand(0, len(neighbours) - 1)]
                
                if maze_grid[y_, x_] == 0:
                    maze_grid[y_, x_] = 1
                    maze_grid[y_ + (y - y_) // 2, x_ + (x - x_) // 2] = 1
                    x, y = x_, y_
                    
    return maze_grid

# Example usage
maze = generate_maze()
print(maze)
