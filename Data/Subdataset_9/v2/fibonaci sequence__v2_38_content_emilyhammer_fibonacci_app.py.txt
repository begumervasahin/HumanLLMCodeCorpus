from flask import Flask, jsonify

app = Flask(__name__)

# Route for generating Fibonacci sequence
@app.route('/fibonacci/<int:endNumber>', methods=['GET'])
def generate_fibonacci(endNumber):
    if endNumber == 0:
        return ""
    elif endNumber > 0:
        return jsonify(calculate_fibonacci(endNumber))
    else:
        return handle_invalid_input()

# Function to calculate Fibonacci sequence
def calculate_fibonacci(n):
    result = []
    a, b = 0, 1
    for _ in range(n):
        result.append(a)
        a, b = b, a + b
    return result

# Error handler for invalid input
@app.errorhandler(400)
def handle_invalid_input(error=None):
    message = "Please enter a positive integer."
    response = jsonify(message)
    response.status_code = 400
    return response

# Run the Flask app
if __name__ == '__main__':
    app.run(host='0.0.0.0', debug=True)
