import unittest
from SentimentPredictor import SentimentPredictor
from driver_3 import imdb_data_preprocess

# Define base classes for reporting
class Reporter:
    def write(self, msg):
        pass

class NullReporter(Reporter):
    def write(self, msg):
        print()

# Define a function to generate test corpus
def generate_corpus():
    return []

# Test case class for testing SentimentPredictor builder
class TestPredictorBuilderTests(unittest.TestCase):   
    def test_can_create_from_default(self):
        # Test if SentimentPredictor can be created with default settings
        sut = SentimentPredictor().build()
        self.assertIsNotNone(sut)

    def test_can_inject_console_reporter(self):       
        # Test if SentimentPredictor can be created with a console reporter injected
        sut = SentimentPredictor().with_reporter(NullReporter()).build()
        self.assertIsNotNone(sut)

    def test_can_run_on_default_test_data(self):      
        # Test if SentimentPredictor can run on default test data
        sut = SentimentPredictor().with_reporter(NullReporter()).build()
        corpus = generate_corpus()
        sut.fit()
        results = [sut.predict(x) for x in corpus]    
        self.assertIsNotNone(results)

# Test case class for testing data preprocessing function
class XformerTests(unittest.TestCase):
    def test_run(self):
        # Test running data preprocessing without mixing
        imdb_data_preprocess('./')

    def test_run_mixed(self):
        # Test running data preprocessing with mixing
        imdb_data_preprocess('./', mix=True)

if __name__ == "__main__":
    # Run all the defined unit tests
    unittest.main()
