import sys

def convert_to_excel_column(n):
    colID = ""
    while n > 0:
        n -= 1
        colID = chr(ord('A') + n % 26) + colID
        n //= 26
    return colID

if len(sys.argv) < 2:
    print("Usage: python script.py <input_file>")
    sys.exit(1)

input_file = sys.argv[1]
with open(input_file, 'r') as file:
    for line in file:
        n = int(line.strip())
        colID = convert_to_excel_column(n)
        print(colID)
