class Node:
    def __init__(self, data=None, next=None):
        """
        Initialize a node with data and a reference to the next node.

        Parameters:
        data: Data to be stored in the node.
        next: Reference to the next node.
        """
        self.data = data
        self.next = None

class LinkedList:
    def __init__(self):
        """
        Initialize a linked list with a head node.
        """
        self.head = None

    def printList(self):
        """
        Print the elements of the linked list.
        """
        temp = self.head
        while temp:
            print(temp.data, end=" ")
            temp = temp.next
        print()

    def insertTailNode(self, data):
        """
        Insert a node with given data at the end of the linked list.

        Parameters:
        data: Data to be stored in the new node.
        """
        if self.head is None:
            self.head = Node(data)
        else:
            current = self.head
            while current.next:
                current = current.next
            current.next = Node(data)

    def insertHeadNode(self, data):
        """
        Insert a node with given data at the beginning of the linked list.

        Parameters:
        data: Data to be stored in the new node.
        """
        current = Node(data)
        current.next = self.head
        self.head = current

    def insertSpecificPosition(self, data, position):
        """
        Insert a node with given data at a specific position in the linked list.

        Parameters:
        data: Data to be stored in the new node.
        position: Position at which the new node should be inserted.
        """
        if position != 0:
            current = self.head
            current_position = 1
            while position - current_position > 0:
                current = current.next
                current_position += 1
            if current.next is None:
                current.next = Node(data, None)
            else:
                prev = current.next
                current.next = Node(data, prev)
        else:
            self.head = Node(data, self.head)

    def delete(self, position):
        """
        Delete the node at the specified position in the linked list.

        Parameters:
        position: Position of the node to be deleted.
        """
        temp = self.head
        if position == 0:
            return temp.next
        while position - 1 > 0:
            self.head = self.head.next
            position -= 1
        self.head.next = self.head.next.next
        return temp

    def printReverseIterative(self):
        """
        Print the elements of the linked list in reverse order using iteration.
        """
        if self.head:
            stack = [self.head]
            while stack[-1].next:
                node = stack[-1]
                stack.append(node.next)
            while stack:
                node = stack.pop()
                print(node.data, end=" ")
        print()

    def reverse(self):
        """
        Reverse the linked list.
        """
        current = self.head
        previous = None
        next_node = None
        while current is not None:
            next_node = current.next
            current.next = previous
            previous = current
            current = next_node
        self.head = previous

if __name__ == '__main__':
    llist = LinkedList()
    llist.insertHeadNode(1)
    llist.insertSpecificPosition(2, 1)
    llist.insertSpecificPosition(3, 2)
    llist.insertTailNode(4)
    print("Linked list:")
    llist.printList()
    print("\nReverse linked list:")
    llist.reverse()
    llist.printList()
