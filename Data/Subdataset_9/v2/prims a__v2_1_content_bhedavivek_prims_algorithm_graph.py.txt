# Define the Vertex class to represent vertices in the graph
class Vertex:
    def __init__(self, id, parentId, distance, position=0):
        self.id = id            # Unique identifier of the vertex
        self.parentId = parentId   # Parent vertex id (if any)
        self.distance = distance   # Distance from the parent vertex
        self.position = position   # Position attribute (default is 0)

# Define the Graph class to represent the graph structure
class Graph:
    def __init__(self, n):
        # Initialize the adjacency list with empty lists for each vertex
        self.adjacency_list = [[] for _ in range(n)]

    # Method to insert an edge between vertices with weights
    def insert(self, u, v, w):
        # Create a new vertex object for vertex v with parent u and distance w
        vert = Vertex(v, u, w)
        # Add the vertex to the adjacency list of vertex u
        self.adjacency_list[u - 1].append(vert)

# Example usage:
if __name__ == "__main__":
    # Create a new graph with 5 vertices
    g = Graph(5)
    # Insert edges between vertices with corresponding weights
    g.insert(1, 2, 10)
    g.insert(1, 3, 15)
    g.insert(2, 4, 20)
    g.insert(3, 5, 25)
    # Print the adjacency list of the graph
    print(g.adjacency_list)
