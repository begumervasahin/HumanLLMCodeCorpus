import numpy as np

class Dataset:
    def __init__(self, data):
        # Initialize variables to keep track of dataset state
        self._index_in_epoch = 0
        self._epochs_completed = 0
        self._data = data
        self.num_examples = data.shape[0]  # Number of examples in the dataset

    @property
    def data(self):
        # Getter method for the dataset
        return self._data

    def next_batch(self, batch_size, shuffle=True):
        start = self._index_in_epoch  # Start index for the current batch
        if start == 0 and self._epochs_completed == 0:
            # Shuffle the dataset at the start of the first epoch
            idx = np.arange(0, self.num_examples)
            np.random.shuffle(idx)
            self._data = self.data[idx]

        if start + batch_size > self.num_examples:
            # If the current batch exceeds the dataset size, start a new epoch
            self._epochs_completed += 1
            rest_num_examples = self.num_examples - start
            data_rest_part = self.data[start:self.num_examples]
            idx0 = np.arange(0, self.num_examples)
            np.random.shuffle(idx0)
            self._data = self.data[idx0]
            start = 0
            self._index_in_epoch = batch_size - rest_num_examples
            end = self._index_in_epoch
            data_new_part = self._data[start:end]
            return np.concatenate((data_rest_part, data_new_part), axis=0)
        else:
            # Move to the next batch within the current epoch
            self._index_in_epoch += batch_size
            end = self._index_in_epoch
            return self._data[start:end]

def preprocess(dataset):
    # Preprocess the dataset by creating a Dataset object
    return Dataset(dataset)

# Example usage:
if __name__ == "__main__":
    # Example dataset
    data = np.random.rand(100, 10)  # 100 samples, 10 features
    processed_dataset = preprocess(data)
    batch = processed_dataset.next_batch(32)
    print(batch.shape)  # Example output shape for a batch
