import chainer

# Define a function to move Chainer variables between CPU and GPU
def move_between_cpu_gpu(var):
    var_on_gpu = chainer.cuda.to_gpu(var)  # Move to GPU
    var_on_cpu = chainer.cuda.to_cpu(var_on_gpu)  # Move back to CPU
    return var_on_cpu

# Example usage:
var = chainer.Variable(chainer.cuda.to_gpu(some_data))
result = move_between_cpu_gpu(var)
