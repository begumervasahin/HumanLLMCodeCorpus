from PersistentBST import PersistentBST

print("\n\tARCHEIO")
tree = PersistentBST()  # Initialize the Persistent Binary Search Tree
current_branch = "master"
previous_branch = None
branches = ["master"]

while True:
    print("\nCurrent Branch:", current_branch)
    print("\n1. List Files\n2. View File\n3. New File\n4. Delete File\n5. Edit File\n6. List Branches\n7. New Branch\n8. Switch Branch\n9. Exit\n")
    choice = input("$ ")
    print()

    if choice == "1":
        print("Files: ")
        # Use the `tree.inFix()` method to list files in the current branch
        tree.inFix(current_branch)
    elif choice == "2":
        name = input("Enter name of file to be opened: ")
        # Use the `tree.search()` method to search for the file in the current branch
        file = tree.search(name, current_branch)
        if file is None:
            print("File does not exist!")
        else:
            print(file.getData())
    elif choice == "3":
        name = input("Enter name of new file: ")
        # Use the `tree.insert()` method to insert a new file in the current branch
        tree.insert(name, current_branch)
    elif choice == "4":
        name = input("Enter name of file to be deleted: ")
        # Use the `tree.delete()` method to delete a file from the current branch
        tree.delete(name, current_branch)
    elif choice == "5":
        name = input("Enter name of file to be edited: ")
        # Use the `tree.edit()` method to edit a file in the current branch
        tree.edit(name, current_branch)
    elif choice == "6":
        print("Branches:")
        for branch in branches:
            print(branch)
    elif choice == "7":
        branch_name = input("Enter name of new branch: ")
        if branch_name in branches:
            print("Branch already exists")
            continue
        branches.append(branch_name)
        previous_branch, current_branch = current_branch, branch_name
        # Use the `tree.newBranch()` method to create a new branch
        tree.newBranch(current_branch, previous_branch)
    elif choice == "8":
        branch_name = input("Enter name of branch to switch to: ")
        if branch_name not in branches:
            print("Branch doesn't exist")
            continue
        previous_branch, current_branch = current_branch, branch_name
    elif choice == "exit" or choice == "9":
        print("Exiting...")
        break
    elif choice == "debug":
        print()
    else:
        print("Invalid Input")
