import networkx as nx
import matplotlib.pyplot as plt

def visualize_graph(graph):
    """
    Visualizes the given graph using NetworkX and Matplotlib.

    Args:
        graph: The graph to visualize.

    Returns:
        None
    """
    # Draw the graph with labels, specifying node color, size, and font size
    nx.draw(graph, with_labels=True, node_color='skyblue', node_size=1000, font_size=12)
    # Set the title of the plot
    plt.title("Graph Visualization")
    # Show the plot
    plt.show()

if __name__ == "__main__":
    # Example usage:
    # Create a new empty undirected graph
    graph = nx.Graph()
    # Add nodes to the graph
    graph.add_nodes_from([1, 2, 3, 4, 5])
    # Add edges to the graph
    graph.add_edges_from([(1, 2), (1, 3), (2, 3), (3, 4), (4, 5)])
    # Visualize the graph
    visualize_graph(graph)
