class Topology:
    def __init__(self, filename):
        self.Nodes = []  # List to store node IDs
        self.Links = []  # List to store link information
        with open(filename, 'r') as topology_file:
            # Skip the first three lines (headers)
            for _ in range(3):
                next(topology_file)
            # Read node data
            for line in topology_file:
                data = line.split()
                if not data:  # Check for empty lines
                    break
                node_id = data[2]
                self.Nodes.append(node_id)
            # Skip the next three lines (headers)
            for _ in range(3):
                next(topology_file)
            # Read link data
            for line in topology_file:
                data = line.split()
                if not data:  # Check for empty lines
                    break
                origin, destination, length = data[2:5]
                self.Links.append([origin, destination, length])

# Usage example:
topology = Topology('data/topologies/PacificBell')
print("Nodes:", topology.Nodes)
print("Links:", topology.Links)
