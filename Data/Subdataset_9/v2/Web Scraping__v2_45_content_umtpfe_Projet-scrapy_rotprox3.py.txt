import requests
from bs4 import BeautifulSoup
from fake_useragent import UserAgent
import random
import time

# Initialize UserAgent for generating random user agents
user_agent_generator = UserAgent()

def fetch_proxies():
    """
    Fetches a list of proxies from 'https://www.sslproxies.org/'.
    Returns a list of dictionaries containing IP addresses and ports of proxies.
    """
    proxies = []
    url = 'https://www.sslproxies.org/'
    headers = {'User-Agent': user_agent_generator.random}
    
    # Fetching the proxy list
    response = requests.get(url, headers=headers)
    soup = BeautifulSoup(response.text, 'html.parser')
    rows = soup.find(id='proxylisttable').find_all('tr')
    
    # Extracting IP addresses and ports from the table rows
    for row in rows:
        columns = row.find_all('td')
        proxy = {
            'ip': columns[0].text,
            'port': columns[1].text
        }
        proxies.append(proxy)
        
    return proxies

def main():
    # Fetch a list of proxies
    proxies_list = fetch_proxies()
    
    # Select a random proxy from the list
    random_proxy_index = random.randint(0, len(proxies_list) - 1)
    proxy = proxies_list[random_proxy_index]
    print("Using proxy:", proxy['ip'] + ':' + proxy['port'])

    # Loop for sending requests using proxies
    for request_number in range(1, 101):
        # Change proxy every 10 requests
        if request_number % 10 == 0:
            random_proxy_index = random.randint(0, len(proxies_list) - 1)
            proxy = proxies_list[random_proxy_index]
            print("Switching to proxy:", proxy['ip'] + ':' + proxy['port'])

        try:
            # Send request to Google using the selected proxy
            response = requests.get('https://www.google.com/', proxies={"http": proxy['ip'] + ':' + proxy['port']})
            print('Request', request_number, '- Status Code:', response.status_code)
        except Exception as e:
            # Print error if request fails
            print("Request", request_number, "- Error:", e)

        # Delay between requests to avoid overwhelming the server
        time.sleep(1)

if __name__ == '__main__':
    main()
