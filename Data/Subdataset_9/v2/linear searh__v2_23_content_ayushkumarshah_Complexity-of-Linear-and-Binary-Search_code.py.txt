import random
from time import time

# Function to perform linear search
def linear_search(the_values, target):
    for value in the_values:
        if value == target:
            return True
    return False

# Function to perform linear search on a sorted list
def sorted_linear_search(the_values, target):
    for value in the_values:
        if value == target:
            return True
        elif value > target:
            return False
    return False

# Function to find the smallest element in a list
def find_smallest(the_values):
    smallest = the_values[0]
    for value in the_values[1:]:
        if value < smallest:
            smallest = value
    return smallest

# Function to perform binary search
def binary_search(the_values, target):
    low = 0
    high = len(the_values) - 1
    while low <= high:
        mid = (high + low) // 2
        if the_values[mid] == target:
            return True
        elif target < the_values[mid]:
            high = mid - 1
        else:
            low = mid + 1
    return False

# Set seed for random number generation
random.seed(42)

# Generate a random list of numbers
my_random_number = random.sample(range(1000000), 100000)
# Sort the random list
my_sorted_random_number = sorted(my_random_number)
# Choose a random number from the sorted list
random_number_from_the_list = random.choice(my_sorted_random_number)

# Lists to store elapsed time for each operation
elapsed_unsorted = []
elapsed_sorted = []
elapsed_smallest = []
elapsed_binary = []

# Loop over different sizes of the list
for i in range(10000, 100001, 10000):
    # Measure time for unsorted linear search
    start_time = time()
    linear_search(my_random_number[:i], random_number_from_the_list)
    end_time = time()
    elapsed_unsorted.append(end_time - start_time)

print("\nUnsorted Linear Search times:")
for i in range(10):
    print(elapsed_unsorted[i])

# Loop over different sizes of the list
for i in range(10000, 100001, 10000):
    # Measure time for sorted linear search
    start_time = time()
    sorted_linear_search(my_sorted_random_number[:i], random_number_from_the_list)
    end_time = time()
    elapsed_sorted.append(end_time - start_time)

print("\nSorted Linear Search times:")
for i in range(10):
    print(elapsed_sorted[i])

# Loop over different sizes of the list
for i in range(10000, 100001, 10000):
    # Measure time for finding the smallest element
    start_time = time()
    find_smallest(my_random_number[:i])
    end_time = time()
    elapsed_smallest.append(end_time - start_time)

print("\nFinding Smallest element times:")
for i in range(10):
    print(elapsed_smallest[i])

# Loop over different sizes of the list
for i in range(10000, 100001, 10000):
    # Measure time for binary search
    start_time = time()
    binary_search(my_random_number[:i], random_number_from_the_list)
    end_time = time()
    elapsed_binary.append(end_time - start_time)

print("\nBinary Search times:")
for i in range(10):
    print(elapsed_binary[i])
