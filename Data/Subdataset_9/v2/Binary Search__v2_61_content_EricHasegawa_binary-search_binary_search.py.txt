def binary_search():
    # Input validation loop to ensure the user enters an integer
    while True:
        try:
            target = int(input("Enter an integer: "))
            break  # Exit the loop if the input is valid
        except ValueError:
            print("You did not enter an integer, try again.")

    # Read and sort the numbers from the file 'numberlist.txt'
    with open('numberlist.txt', 'r') as numberlist_file:
        numberlist = numberlist_file.read().split(", ")
        sorted_results = sorted(map(int, numberlist))

    # Initialize variables for binary search
    high = len(sorted_results) - 1
    low = 0

    # Binary search algorithm
    while high >= low:
        middle = (high + low) // 2
        if target == sorted_results[middle]:
            print("Target found")
            break
        elif middle == low:
            print("Target not found")
            break
        elif target > sorted_results[middle]:
            low = middle + 1
        elif target < sorted_results[middle]:
            high = middle - 1

    # Prompt the user to test another number or exit the program
    while True:
        yn = input("Would you like to test another number? (y/n): ")
        if yn == 'y':
            binary_search()  # Recursively call binary_search for another number
            break
        elif yn == 'n':
            print("Exiting the program")
            break
        else:
            print("You did not enter y or n, try again.")


if __name__ == "__main__":
    binary_search()  # Start the binary search program
