# Importing necessary functions from the words module
from words import get_text, words

# Function to create an index using a hash table
def myhtable_create_index(files):
    """
    Build an index from word to set of document indexes
    This function creates an index similar to create_index() but uses a hash table.
    The hash table has 4011 buckets.
    """
    # Checking if there are any files
    if len(files) <= 0:
        return None
    
    # Creating a hash table with 4011 buckets
    table = htable(4011)
    
    # Iterating through the files
    for i in range(len(files)):
        # Getting the text content of the file
        file_content = get_text(files[i])
        
        # Extracting words from the text content
        key_words = words(file_content)
        
        # Adding words and their corresponding document indexes to the hash table
        for word in key_words:
            htable_put(table, word, set([i]))
    
    return table

# Function to search for terms in the index
def myhtable_index_search(files, index, terms):
    """
    Search for terms in the index
    This function performs a search similar to index_search() but using a hash table.
    """
    # Checking if any argument is missing or empty
    if files is None or index is None or terms is None or \
            len(files) == 0 or len(index) == 0 or len(terms) == 0:    
        return None

    # Initializing an empty list to store the resulting files
    ret_file_list = []
    
    # Initializing an empty list to store sets of document indexes corresponding to terms
    list_of_sets = []
    
    # Iterating through the search terms
    for term in terms:
        # Getting the document indexes set for the term from the index hash table
        value = htable_get(index, term)
        
        # If the term is not found in the index, return None
        if value is None:
            return None
        
        # If the term is found, add its document indexes set to the list
        list_of_sets.append(value)

    # If there are sets of document indexes for terms
    if len(list_of_sets) > 0:
        # Find the intersection of all sets of document indexes
        intersection_ids = set.intersection(*list_of_sets)
        
        # Append corresponding files to the result list
        for id in intersection_ids:
            ret_file_list.append(files[id])
    else:
        # If no intersection is found, return None
        return None

    return ret_file_list

# Example usage:

# Assuming you have some files in a list
files = ["file1.txt", "file2.txt", "file3.txt"]

# Creating the index
index = myhtable_create_index(files)

# Searching for terms
search_terms = ["term1", "term2", "term3"]
result = myhtable_index_search(files, index, search_terms)

# Printing the result
if result:
    print("Search Result:")
    for file in result:
        print(file)
else:
    print("No results found.")
