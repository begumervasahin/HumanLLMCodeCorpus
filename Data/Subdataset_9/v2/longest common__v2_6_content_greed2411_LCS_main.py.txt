import re

def verify_sequence(sequence):
    # Checks if the sequence contains only valid DNA characters or spaces
    return bool(re.match("^[ATCG ]*$", sequence))

def find_longest_common_subsequence(seq1, seq2, len_seq1, len_seq2):
    # Finds the longest common subsequence between two sequences
    table = [[0 for _ in range(len_seq2+1)] for _ in range(len_seq1+1)]

    for i in range(len_seq1+1):
        for j in range(len_seq2+1):
            if i == 0 or j == 0:
                table[i][j] = 0
            elif seq1[i-1] == seq2[j-1]:
                table[i][j] = table[i-1][j-1] + 1
            else:
                table[i][j] = max(table[i-1][j], table[i][j-1])

    index = table[len_seq1][len_seq2]
    lcs_chars = [""] * (index+1)
    lcs_chars[index] = "\0"

    i = len_seq1
    j = len_seq2
    while i > 0 and j > 0:
        if seq1[i-1] == seq2[j-1]:
            lcs_chars[index-1] = seq1[i-1]
            i -= 1
            j -= 1
            index -= 1
        elif table[i-1][j] > table[i][j-1]:
            i -= 1
        else:
            j -= 1

    lcs_sequence = "".join(lcs_chars)
    print("\n\nThe Longest Common Subsequence of " + seq1 + " and " + seq2 + " is " + lcs_sequence + 
          " \nand its length is: " + str(len(lcs_sequence)-1))

# Input sequences
sequence1 = input("Enter the First Sequence\n").upper()
sequence2 = input("\nEnter the Querying Sequence\n").upper()

# Validation and processing
if (verify_sequence(sequence1) and verify_sequence(sequence2)) and (len(sequence1) >= len(sequence2)):
    print(sequence1, sequence2)
    len_seq1 = len(sequence1)
    len_seq2 = len(sequence2)
    find_longest_common_subsequence(sequence1, sequence2, len_seq1, len_seq2)
elif not (len(sequence1) >= len(sequence2)):
    print("The Querying Sequence should be shorter than the First Sequence")
else:
