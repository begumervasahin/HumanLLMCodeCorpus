# Function to check if a number is prime
def is_prime(n):
    assert type(n) == int  # Ensuring input is an integer
    if n < 2:
        return False  # Numbers less than 2 are not prime
    # Checking divisibility up to the square root of n
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False  # If divisible by any number, it's not prime
    return True  # If not divisible by any number, it's prime

# Function to find all divisors of a number
def divisor(n):
    lst = []
    for i in range(1, n + 1):
        if n % i == 0:
            lst.append(i)  # Adding divisor to the list
    return lst

# Function to find prime factors of a number
def prime_factor(x):
    fact_lst = []
    if is_prime(x):
        fact_lst.append(x)  # If x itself is prime, it's its only prime factor
    # Finding prime factors using divisor function
    for i in divisor(x):
        if i == 1:
            pass  # Skipping 1 as it's not a prime factor
        elif is_prime(i):
            fact_lst.append(i)  # Adding prime factor to the list
    return fact_lst

# Function to find the greatest common divisor (GCD) of two numbers
def gcd(a, b):
    if b > a:
        return gcd(b, a)  # Ensuring a is greater than b
    elif a % b == 0:
        return b  # If b divides a, b is the GCD
    return gcd(b, a % b)  # Recursive call with remainder

# Function to check if two numbers are relatively prime (have no common factors other than 1)
def relatively_prime(a, b):
    return gcd(a, b) == 1  # If GCD is 1, numbers are relatively prime

# Function to find common factors of two numbers
def common_factor(a, b):
    a_fact = divisor(a)  # Finding divisors of first number
    b_fact = divisor(b)  # Finding divisors of second number
    common_fact = []
    # Finding common factors
    for elem in a_fact:
        if elem in b_fact:
            common_fact.append(elem)  # Adding common factor to the list
    return common_fact

# Testing the functions
print(is_prime(7))  # True
print(divisor(12))  # [1, 2, 3, 4, 6, 12]
print(prime_factor(24))  # [2, 3]
print(gcd(24, 36))  # 12
print(relatively_prime(24, 35))  # True
print(common_factor(24, 36))  # [1, 2, 3, 4, 6, 12]
