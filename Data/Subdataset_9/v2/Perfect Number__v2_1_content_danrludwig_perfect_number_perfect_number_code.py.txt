import time

def find_perfect_numbers_up_to(limit):
    """Find all perfect numbers up to the given limit.
    
    A perfect number is a positive integer that is equal to the sum of its
    proper divisors, excluding itself.
    
    Args:
    - limit: The upper bound for searching perfect numbers.
    
    Returns:
    - A list of all perfect numbers below the given limit.
    """
    perfect_numbers = []  # Initialize an empty list to store perfect numbers
    for number in range(1, limit):
        # Calculate the sum of divisors of the number
        divisor_sum = 0
        for potential_divisor in range(1, number // 2 + 1):  # Optimized to only go up to half the number
            if number % potential_divisor == 0:
                divisor_sum += potential_divisor
        
        # Check if the current number is perfect
        if divisor_sum == number:
            perfect_numbers.append(number)
    
    return perfect_numbers

# Set the upper limit for finding perfect numbers
search_limit = 10000

# Record the start time
start_time = time.perf_counter()

# Find perfect numbers within the limit
found_perfect_numbers = find_perfect_numbers_up_to(search_limit)

# Record the end time
end_time = time.perf_counter()

# Display the results
print(f"Perfect numbers below {search_limit}:")
for perfect_number in found_perfect_numbers:
    print(perfect_number)

# Display how long the calculation took
print(f"Calculation took {end_time - start_time:.6f} seconds.")
