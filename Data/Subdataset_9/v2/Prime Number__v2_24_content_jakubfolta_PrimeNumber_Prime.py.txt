def is_prime(digit):
    """Check if a number is prime."""
    if digit < 2:
        return False
    elif digit == 2:
        return True
    else:
        for number in range(2, digit):
            if digit % number == 0:
                return False
        return True

print(is_prime(21))

def prime(number):
    """Check if a number is prime."""
    if number < 2:
        return False
    elif number == 2:
        return False
    else:
        for x in range(2, number):
            if number % x == 0:
                return False
        return True

print(prime(18))

def check_if_prime_number(number):
    """Check if a number is prime."""
    if number < 2:
        return False
    else:
        for x in range(2, number - 1):
            if number % x == 0:
                return False
        return True

print(check_if_prime_number(15))
