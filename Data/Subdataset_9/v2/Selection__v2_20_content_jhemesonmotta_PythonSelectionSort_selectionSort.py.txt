def selection_sort(arr):
    # Iterate over the array in reverse order
    for last_pos in range(len(arr) - 1, 0, -1):
        # Assume the first element is the largest
        largest_pos = 0
        # Iterate over the array up to the last unsorted position
        for pos in range(1, last_pos + 1):
            # Check if the current element is larger than the current largest
            if arr[pos] > arr[largest_pos]:
                largest_pos = pos
        # Swap the largest element with the last unsorted element
        arr[last_pos], arr[largest_pos] = arr[largest_pos], arr[last_pos]

# Example usage
arr = [54, 26, 93, 17, 77, 31, 44, 55, 20]
selection_sort(arr)
print("Sorted array:", arr)
