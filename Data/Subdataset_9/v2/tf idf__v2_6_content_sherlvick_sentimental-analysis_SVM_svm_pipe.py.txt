import pandas as pd
import pickle
import os
from sklearn.feature_extraction.text import CountVectorizer, TfidfTransformer
from sklearn.pipeline import Pipeline
from sklearn.svm import SVC
import numpy as np

def train_and_save_model():
    # Set a seed for reproducibility
    np.random.seed(100)
    
    # Read data from CSV file
    dataset = pd.read_csv("Test.csv", encoding='utf-8-sig')
    
    # Extract text data and corresponding labels
    texts = dataset['text']
    labels = dataset['label']
    
    # Define a text classification pipeline
    text_clf = Pipeline([
        ('vectorizer', CountVectorizer()),  # Convert text to token counts
        ('tfidf_transformer', TfidfTransformer()),  # Convert token counts to TF-IDF features
        ('classifier', SVC(C=5, kernel='rbf', degree=3, gamma=0.5)),  # SVM classifier
    ])
    
    # Train the model on the dataset
    trained_model = text_clf.fit(texts, labels)
    
    # Save the trained model to a file
    model_filename = 'text_classification_model.sav'
    with open(model_filename, 'wb') as model_file:
        pickle.dump(trained_model, model_file)
    
    # Check if the model file was successfully saved
    model_file_exists = os.path.exists(model_filename)
    
    # Prepare a list to return the result of training
    training_result = ["Success: Model trained and saved.", "Error: Model could not be saved.", f"Number of records: {len(dataset)}"]
    
    # Adjust the return list based on whether the model file exists
    if model_file_exists:
        del training_result[1]
    else:
        del training_result[0]
    
    return training_result

if __name__ == "__main__":
    # Call the training function and print the result
    training_result = train_and_save_model()
    print(training_result)
