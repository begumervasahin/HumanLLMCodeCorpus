import discord
import requests
import bs4

# Initialize Discord client
client = discord.Client()

# Define behavior when a message is received
@client.event
async def on_message(message):
    # Ignore messages from the bot itself
    if message.author == client.user:
        return
    
    # Define list of available commands
    commands_list = ['!help', '!commands']
    
    # Check if message contains any of the available commands
    if any(command in message.content.lower() for command in commands_list): 
        # Send a message explaining the available commands
        msg = f'Hello {message.author.mention}, currently the only commands that are active are !price "Insert Ticker Here" which will display the current price of a stock.'
        await message.channel.send(msg)
    elif message.content.startswith('!price'): # Check if message is a price query
        # Extract stock ticker from the message
        stock = message.content.split(' ')[1]
        # Construct the URL to fetch the stock price
        url = "https://finance.yahoo.com/quote/"
        full_url = url + stock
        # Fetch the webpage content
        response = requests.get(full_url).content
        # Parse the webpage content using BeautifulSoup
        soup = bs4.BeautifulSoup(response, 'html.parser')
        # Find the element containing the stock price
        stock_price = soup.find(class_="Trsdu(0.3s) Trsdu(0.3s) Fw(b) Fz(36px) Mb(-4px) D(b)").text
        # Send the stock price as a message
        await message.channel.send(f"The stock price for {stock.upper()} is ${stock_price} currently.")

# Define behavior when the bot is ready
@client.event
async def on_ready():
    print('Logged in as') # Log the bot's name
    print(client.user.name)
    print(client.user.id) # Log the bot's ID
    print('------')
