import random
import os
import uuid
from keras.datasets import mnist
from PIL import Image

def generate_data(train_samples=5000, test_samples=1000):
    # Load the MNIST dataset
    (x_train, y_train), (x_test, y_test) = mnist.load_data()
    
    # Define directories for train and test data
    train_dir = "datasets/train/"
    test_dir = "datasets/test/"
    
    # Create directories if they don't exist
    if not os.path.exists(train_dir):
        os.makedirs(train_dir)
    if not os.path.exists(test_dir):
        os.makedirs(test_dir)
    
    # Generate images for training and testing
    for i in range(train_samples + test_samples):
        # Determine whether to use training or testing data
        if i < train_samples:
            data = x_train
            labels = y_train
            directory = train_dir
        else:
            data = x_test
            labels = y_test
            directory = test_dir
        
        # Select a random image from the dataset
        index = random.choice(range(data.shape[0]))
        image = Image.fromarray(data[index])  # Convert numpy array to PIL Image
        label = str(labels[index])  # Get the corresponding label
        
        # Apply random transformations (resizing and rotation)
        resize_ratio = random.random() * 0.5 + 0.5  # Random resize ratio between 0.5 and 1.0
        new_size = (int(image.size[0] * resize_ratio), int(image.size[1] * resize_ratio))
        image = image.resize(new_size)
        
        rotate_degree = random.randint(0, 90)  # Random rotation angle between 0 and 90 degrees
        image = image.rotate(rotate_degree)
        
        # Resize the image to 28x28 (MNIST image size)
        image = image.resize((28, 28))
        
        # Generate a unique filename using UUID for each image
        filename = "{}{}-{}.jpg".format(directory, label, uuid.uuid4().hex)
        
        # Save the image
        image.save(filename)

if __name__ == "__main__":
    generate_data()
