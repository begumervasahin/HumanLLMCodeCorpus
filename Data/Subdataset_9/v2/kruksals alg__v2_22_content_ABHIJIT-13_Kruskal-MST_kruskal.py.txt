def find_root(vertex):
    """
    Find the root of a vertex in the disjoint set.
    """
    if root_ids[vertex] == vertex:
        return vertex
    else:
        # Path compression: update the root of each vertex along the path
        while root_ids[vertex] != vertex:
            root_ids[vertex] = root_ids[root_ids[vertex]]
            vertex = root_ids[vertex]
        return vertex

def union(u, v):
    """
    Perform union operation on two vertices.
    """
    u_root = find_root(u)
    v_root = find_root(v)
    # Union by rank: attach the root of the smaller tree to the root of the larger tree
    root_ids[v_root] = root_ids[u_root]

def kruskal(graph_edges):
    """
    Implement Kruskal's algorithm to find the minimum spanning tree weight.
    """
    global root_ids
    min_weight = 0
    # Sort edges by weight
    graph_edges = sorted(graph_edges, key=lambda x: x[2])
    for edge in graph_edges:
        u, v, weight = edge
        if find_root(u) != find_root(v):
            # If the vertices are not in the same set, add the edge to the minimum spanning tree
            min_weight += weight
            union(u, v)
    return min_weight

# Input
V, E = map(int, input().split())
graph_edges = []
for i in range(E):
    graph_edges.append(list(map(int, input().split())))

# Initialize disjoint set with each vertex as its own set
root_ids = list(range(V + 1))

# Output the weight of the minimum spanning tree
print(kruskal(graph_edges)))
