import tkinter as tk
from math import pi

def check_prime_number():
    """
    GUI function to check if a number is prime.
    """
    cp = tk.Tk()
    cp.title("Check Prime Number")
    cp.geometry("500x500")
    cp.configure(background='yellow')

    entry_var = tk.StringVar()
    entry_box = tk.Entry(cp, textvariable=entry_var, font=('Arial', 20, 'bold'))
    entry_box.pack()

    def prime_number():
        num = int(entry_var.get())
        if num <= 1:
            result_label["text"] = f'{num} is not prime'
        else:
            for i in range(2, num):
                if num % i == 0:
                    result_label["text"] = f'{num} is not prime'
                    break
            else:
                result_label["text"] = f'{num} is prime'

    result_label = tk.Label(cp, font=('Arial', 20, 'bold'))
    result_label.pack()

    btn = tk.Button(cp, text='Calculate', command=prime_number, font=('Arial', 20, 'bold'))
    btn.pack()

def radian_to_degree():
    """
    GUI function to convert radians to degrees.
    """
    rd = tk.Tk()
    rd.title('Radian to Degree')
    rd.geometry("500x500")
    rd.configure(background='yellow')

    r_var = tk.StringVar()
    entry_box = tk.Entry(rd, textvariable=r_var, font=('Arial', 20, 'bold'))
    entry_box.pack()

    def convert():
        rad = float(r_var.get())
        degree = rad * 180 / pi
        result_label["text"] = f'Degrees = {degree}'

    result_label = tk.Label(rd, font=('Arial', 20, 'bold'))
    result_label.pack()

    btn = tk.Button(rd, text='Calculate', command=convert, font=('Arial', 20, 'bold'))
    btn.pack()

def compute_arc_length():
    """
    GUI function to compute arc length of an angle.
    """
    cal = tk.Tk()
    cal.title('Compute Arc Length of an Angle')
    cal.geometry("400x400")
    cal.configure(background="yellow")

    d_var = tk.StringVar()
    a_var = tk.StringVar()

    tk.Label(cal, text="Please Enter Values To find Arc Length", bg='yellow', font=('Arial', 15, 'bold')).pack()
    tk.Label(cal, text="Please Enter Diameter here: ", bg='yellow', font=('Arial', 12, 'bold')).pack()
    entry1 = tk.Entry(cal, textvariable=d_var, font=('Arial', 20, 'bold'))
    entry1.pack()
    tk.Label(cal, text="Please Enter Angle here: ", bg='yellow', font=('Arial', 12, 'bold')).pack()
    entry2 = tk.Entry(cal, textvariable=a_var, font=('Arial', 20, 'bold'))
    entry2.pack()

    def arc_length():
        diameter = float(entry1.get())
        angle = float(entry2.get())
        arc_len = (pi * diameter) * (angle / 360)
        result_label["text"] = f"Arc Length is: {arc_len}"

    result_label = tk.Label(cal, font=('Arial', 15, 'bold'))
    result_label.pack()

    btn = tk.Button(cal, text="Calculate", command=arc_length, font=('Arial', 20, 'bold'))
    btn.pack()

def area_of_sector():
    """
    GUI function to compute area of sector.
    """
    aos = tk.Tk()
    aos.title('Compute Area Of Sector')
    aos.geometry("400x400")
    aos.configure(background="yellow")

    radius_var = tk.StringVar()
    angle_var = tk.StringVar()

    tk.Label(aos, text="Please Enter Radius Here", bg='yellow', font=('Arial', 20, 'bold')).pack()
    entry1 = tk.Entry(aos, textvariable=radius_var, font=('Arial', 20, 'bold'))
    entry1.pack()
    tk.Label(aos, text="Please Enter Angle Here", bg='yellow', font=('Arial', 20, 'bold')).pack()
    entry2 = tk.Entry(aos, textvariable=angle_var, font=('Arial', 20, 'bold'))
    entry2.pack()

    def calculate_area():
        radius = float(entry1.get())
        angle = float(entry2.get())
        area = (1 / 2) * radius ** 2 * angle
        result_label["text"] = f"Area of Sector: {area}"

    result_label = tk.Label(aos, font=('Arial', 20, 'bold'))
    result_label.pack()

    btn = tk.Button(aos, text="Calculate", command=calculate_area, font=('Arial', 20, 'bold'))
    btn.pack()
