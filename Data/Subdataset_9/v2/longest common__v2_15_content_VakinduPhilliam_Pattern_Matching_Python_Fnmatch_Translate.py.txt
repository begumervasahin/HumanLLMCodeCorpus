import fnmatch
import re

# Function to check if a string matches a given fnmatch pattern
def does_match_fnmatch(pattern, string):
    """
    Checks if the string matches the provided fnmatch pattern.
    """
    return fnmatch.fnmatch(string, pattern)

# Function to check if a string matches a given regular expression pattern
def does_match_regex(pattern, string):
    """
    Checks if the string matches the provided regular expression pattern.
    """
    translated_regex = fnmatch.translate(pattern)
    compiled_regex = re.compile(translated_regex)
    return compiled_regex.match(string)

# Example usage:
pattern = '*.txt'  # The pattern we want to match
string = 'foobar.txt'  # The string we want to check for a match

# Check if the string matches the pattern using fnmatch
fnmatch_result = does_match_fnmatch(pattern, string)

# Check if the string matches the pattern using regular expressions
regex_result = does_match_regex(pattern, string)

# Output the results
print(f"Using fnmatch, the string matches the pattern: {fnmatch_result}")
print(f"Using regular expressions, the string matches the pattern: {bool(regex_result)}")
