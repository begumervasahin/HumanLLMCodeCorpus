import os
import sys
import pygame.mixer as mixer

# Define paths for input, save, and alternative directories
input_directory = "/home/error-404/Desktop/songs"
save_directory = "/home/error-404/Desktop/dance"
alternative_directory = "/home/error-404/Desktop/others"

# Get list of audio files in the input directory
audio_files = [file for file in os.listdir(input_directory) if os.path.isfile(os.path.join(input_directory, file))]

# Initialize Pygame mixer
mixer.init()

# Flag to track pause/unpause state
paused = False

# Loop through each audio file in the input directory
for file in audio_files:
    print("Now playing:", file)
    mixer.music.load(os.path.join(input_directory, file))  # Load audio file
    mixer.music.play(0, 40)  # Start playing from 40 seconds

    # Loop to handle user input while audio is playing
    while True:
        user_input = input()

        # Seek to a specific position (multiplied by 10 seconds)
        if user_input.isdigit():
            mixer.music.rewind()
            mixer.music.play(0, int(user_input) * 10)

        # Stop playback and move the file to the alternative directory
        elif user_input == '':
            mixer.music.stop()
            os.rename(os.path.join(input_directory, file), os.path.join(alternative_directory, file))
            break

        # Stop playback and move the file to the save directory
        elif user_input == '/':
            mixer.music.stop()
            os.rename(os.path.join(input_directory, file), os.path.join(save_directory, file))
            break

        # Quit the program
        elif user_input == 'q':
            sys.exit(0)

        # Pause/unpause the playback
        elif user_input == 'p':
            if paused:
                mixer.music.unpause()
                paused = False
            else:
                mixer.music.pause()
                paused = True
