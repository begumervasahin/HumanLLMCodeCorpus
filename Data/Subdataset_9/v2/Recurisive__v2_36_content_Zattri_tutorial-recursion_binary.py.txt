class Node:
    def __init__(self, value):
        # Initialize the Node with a value and set pointers to None
        self.value = value
        self.left = None
        self.right = None
        self.parent = None

    def add_left_child(self, child_node):
        # Add a left child to the current node
        self.left = child_node
        child_node.parent = self

    def add_right_child(self, child_node):
        # Add a right child to the current node
        self.right = child_node
        child_node.parent = self

    def print_left_child(self):
        # Print the value of the left child if it exists
        if self.left:
            print("Left Child:", self.left.value)
        else:
            print("No left child")

    def print_right_child(self):
        # Print the value of the right child if it exists
        if self.right:
            print("Right Child:", self.right.value)
        else:
            print("No right child")

    def print_value(self):
        # Print the value of the current node
        print("Node Value:", self.value)

    def print_parent(self):
        # Print the value of the parent node if it exists
        if self.parent:
            print("Parent Node:", self.parent.value)
        else:
            print("No parent node")
