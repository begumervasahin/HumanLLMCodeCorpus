def gcd(a, b):
    # Euclidean algorithm to find the greatest common divisor
    while b > 0:
        a, b = b, a % b
    return a

def next_prime(p):
    # Find the next prime number greater than p
    while p % 12 != 11:
        p += 1
    return next_prime_step(p)

def next_prime_step(p):
    # Helper function for next_prime
    # Uses a probabilistic prime test and various conditions to find the next prime
    m_ = 5 * 7 * 11 * 13 * 17 * 19 * 23 * 29 * 31
    while True:
        while gcd(p, m_) != 1 or gcd((p - 1) // 2, m_) != 1:
            p += 12
        if pow(7, p - 1, p) != 1:
            p += 12
            continue
        if pow(7, (p - 3) // 2, (p - 1) // 2) != 1:
            p += 12
            continue
        return p
