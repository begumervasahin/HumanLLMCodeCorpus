import unittest
from flask_testing import TestCase
from flask import abort, url_for
from app import create_app, db
from app.models import Employee, Department, Role

class TestBase(TestCase):
    # Set up the Flask application for testing
    def create_app(self):
        config_name = 'testing'
        app = create_app(config_name)
        app.config.update(SQLALCHEMY_DATABASE_URI='sqlite:///:memory:')
        return app

    # Prepare the test environment before each test
    def setUp(self):
        db.create_all()
        # Create sample data
        admin = Employee(username="admin", password="admin2016", is_admin=True)
        employee = Employee(username="test_user", password="test2016")
        db.session.add(admin)
        db.session.add(employee)
        db.session.commit()

    # Clean up after each test
    def tearDown(self):
        db.session.remove()
        db.drop_all()

class TestModels(TestBase):
    # Test the User model
    def test_user_model(self):
        self.assertEqual(Employee.query.count(), 2)

    # Test the Department model
    def test_department_model(self):
        department = Department(name="IT", description="the IT department")
        db.session.add(department)
        db.session.commit()
        self.assertEqual(Department.query.count(), 1)

    # Test the Role model
    def test_role_model(self):
        role = Role(name="CEO", description="Run the whole company")
        db.session.add(role)
        db.session.commit()
        self.assertEqual(Role.query.count(), 1)

class TestViews(TestBase):
    # Test accessing the homepage
    def test_homepage_view(self):
        response = self.client.get(url_for('home.homepage'))
        self.assertEqual(response.status_code, 200)

    # Test accessing the login page
    def test_login_view(self):
        response = self.client.get(url_for('auth.login'))
        self.assertEqual(response.status_code, 200)

    # More view tests would follow...

class TestErrorPages(TestBase):
    # Test accessing a forbidden page
    def test_403_forbidden(self):
        response = self.client.get('/403')
        self.assertEqual(response.status_code, 403)
        self.assertTrue("403 Error" in response.data)

    # Test accessing a non-existent page
    def test_404_not_found(self):
        response = self.client.get('/nothinghere')
        self.assertEqual(response.status_code, 404)
        self.assertTrue("404 Error" in response.data)

    # Test simulating an internal server error
    def test_500_internal_server_error(self):
        response = self.client.get('/500')
        self.assertEqual(response.status_code, 500)
        self.assertTrue("500 Error" in response.data)

if __name__ == '__main__':
    unittest.main()
