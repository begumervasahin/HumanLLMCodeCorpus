class Map:
    """
    Represents a map object that fetches data from a .txt file and renders it into a list of lists.
    """

    def __init__(self, filename):
        """
        Initializes the Map object with the provided filename.
        Constructs a 2D list (board) from the .txt file and stores its dimensions and cell size.
        :param filename: Path to the map file (string).
        """
        self.filename = filename
        self.board = self._boardify_file(self._get_file(filename))
        self.height = len(self.board)
        self.width = len(self.board[0])
        self.cell_size = 30

    @staticmethod
    def _get_file(filename):
        """
        Opens the map .txt file and returns the file object.
        :param filename: Path to the map file (string).
        :return: Opened file object.
        """
        file = open(filename)
        return file

    @staticmethod
    def _boardify_file(file):
        """
        Reads the file object, splits its lines, and returns a list of lists (2D array).
        :param file: Map file object.
        :return: 2D list representing the map.
        """
        lines = file.read().splitlines()
        file.close()
        return lines

# Example usage
filename = "map.txt"  # Replace this with the actual path to your map file
map_obj = Map(filename)
print("Map Dimensions:", map_obj.height, "x", map_obj.width)
print("Cell Size:", map_obj.cell_size)
print("Map Content:")
for row in map_obj.board:
    print(row)
