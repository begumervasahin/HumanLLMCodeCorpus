import textwrap
import conversion
import permutations
import misc

def encrypt(text, key):
    # Convert the input text to binary representation
    binary_text = misc.text_to_bits(text)
    
    # Wrap the binary text into 64-bit blocks
    binary_text = textwrap.wrap(binary_text, 64)
    
    # Ensure the last block is filled with zeros if needed
    last_text_part = binary_text[-1]
    binary_text[-1] = last_text_part + '0' * (64 - len(last_text_part))
    
    # Generate subkeys from the primary key
    subkeys = conversion.primary_key_to_subkeys(key)

    # Encrypt each 64-bit block
    for part_num, part in enumerate(binary_text):
        part = permutations.permutation(part, "initial")
        lpt, rpt = part[:32], part[32:]
        
        # Perform 16 rounds of Feistel encryption
        for i in range(16):
            lpt, rpt = rpt, misc.binary_addition(lpt, function(rpt, subkeys[i]), 32)
        
        # Permute the final output and update the encrypted block
        binary_text[part_num] = permutations.permutation(rpt + lpt, "inverse") 
    
    # Concatenate the encrypted blocks to form the encrypted text
    return ''.join(binary_text)

def function(rpt, subkey):
    # Apply expansion permutation
    rpt = permutations.permutation(rpt, "expansion")
    
    # Perform XOR operation with the subkey
    rpt = misc.XOR(rpt, subkey)
    
    # Apply S-box permutation
    rpt = permutations.sBox_permutation(rpt)
    
    # Apply permutation
    rpt = permutations.permutation(rpt, "pBox")
    
    return rpt

# Example usage
text = "Hello, World!"
key = "123456789ABCDEF0"  # Example key
encrypted_text = encrypt(text, key)
print("Encrypted text:", encrypted_text)
