import cv2

# Read the image
img = cv2.imread('newa4.jpg')

# Get image dimensions
height, width, channels = img.shape

# Define the font for displaying text
font = cv2.FONT_HERSHEY_SIMPLEX

# Iterate over grid cells
for i in range(0, height, 100):
    for j in range(0, width, 100):
        # Get the color of the current grid cell
        color = img[j, i]

        # Display color information as text on the image
        cv2.putText(img, str(color), (i, j), font, 0.4, (255, 255, 255), 2)

        # Draw a circle at the top-left corner of the grid cell
        cv2.circle(img, (i, j), 3, (255, 255, 255), 1)

# Display the image with color information and circles
cv2.imshow('Image', img)

# Wait for a key press and then close all windows
cv2.waitKey(0)
cv2.destroyAllWindows()
