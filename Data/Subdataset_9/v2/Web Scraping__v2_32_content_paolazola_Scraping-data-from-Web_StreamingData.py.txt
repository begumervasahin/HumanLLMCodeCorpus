import datetime
import time
import pandas as pd
import requests
from lxml import html
from random import uniform

def fetch_real_time_stock_data(ticker, finish_time, delta_minutes):
    # Construct the URL for Yahoo Finance
    yahoo_finance_url = 'https://finance.yahoo.com/quote/' + str(ticker)
    
    # Define user-agent for the HTTP request headers
    headers = {
        'User-Agent': 'Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/42.0.2311.90 Safari/537.36'
    }
    
    # Initialize lists to store data
    stock_values = []
    scraper_timestamps = []
    yahoo_price_timestamps = []
    
    # Run the loop until the specified finish time
    while datetime.datetime.now() < finish_time:
        # Send GET request to Yahoo Finance
        response = requests.get(yahoo_finance_url, headers=headers)
        
        # Pause for a random time interval
        time.sleep(uniform(2, 3))
        
        # Parse the HTML content
        parsed_content = html.fromstring(response.content)
        
        # Extract stock value and Yahoo's timestamp
        try:
            stock_value = parsed_content.xpath('//span[@data-reactid="50"]//text()')[0]
            yahoo_timestamp = parsed_content.xpath('//div[@id="quote-market-notice"]//text()')[0]
        except IndexError:
            print("Error: Failed to extract data from Yahoo Finance.")
            continue
        
        # Append data to respective lists
        stock_values.append(stock_value)
        scraper_timestamps.append(datetime.datetime.now())
        yahoo_price_timestamps.append(yahoo_timestamp)
        
        # Wait for the specified delta before checking again
        time.sleep(delta_minutes * 60)
    
    # Create a DataFrame to organize the collected data
    data = {
        'scraper_timestamp': scraper_timestamps,
        'stock_value': stock_values,
        'yahoo_price_timestamp': yahoo_price_timestamps
    }
    df = pd.DataFrame(data)
    
    return df

# Example usage:
finish_time = datetime.datetime.now() + datetime.timedelta(minutes=5)  # Run for 5 minutes
delta_minutes = 1  # Check every 1 minute
ticker_symbol = 'AAPL'  # Example ticker symbol (Apple Inc.)
result_dataframe = fetch_real_time_stock_data(ticker_symbol, finish_time, delta_minutes)
print(result_dataframe)
