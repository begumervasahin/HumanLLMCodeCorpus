import rsa
import gmpy2

class RSA(object):
    # Function to generate two prime numbers, p and q, each with 'bits' number of bits
    def get_p_q(self, bits):
        p = rsa.prime.getprime(bits)
        q = rsa.prime.getprime(bits)
        return p, q

    # Function to compute the value of e given phi
    def get_e(self, phi):
        return phi - 1

    # Function to generate RSA keys (d, (n, e)) given the number of bits
    def get_keys(self, bits):
        p, q = self.get_p_q(bits)
        n = p * q
        phi = (p - 1) * (q - 1)
        e = self.get_e(phi)
        d = int(gmpy2.invert(e, phi).digits())
        return d, (n, e)

    # Function to encrypt a message using RSA public key (e, n)
    def crypt(self, e, n, message):
        # Convert message to integer using little-endian byte order
        E = int.from_bytes(bytes(message.encode()), byteorder='little')
        # Encrypt message using RSA encryption formula
        return pow(E, e, n)

    # Function to decrypt an encrypted message using RSA private key (d, n)
    def decrypt(self, d, n, message):
        # Decrypt message using RSA decryption formula
        msg = pow(message, d, n)
        # Convert decrypted message from integer to bytes, strip padding, and decode to string
        msg = msg.to_bytes((msg.bit_length() // 8) + 1, byteorder='little').strip(b'\x00').decode("utf-8")
        return msg
