import collections

class Dijkstra:
    def __init__(self, *nodes):
        # Initialize the Dijkstra algorithm with a list of nodes
        self.nodes = list(nodes)
        self.result = {node: float('inf') for node in self.nodes}  # Initialize result with infinite distance for all nodes
        self.edges = collections.defaultdict(dict)  # Store edges as a defaultdict of dictionaries

    def add_path(self, src, dest, dist):
        # Add a path between source and destination with a given distance
        if src not in self.edges:
            self.edges[src] = {}  # Initialize the destination dictionary if it doesn't exist
        self.edges[src][dist] = dest  # Add the destination and distance to the source's dictionary of destinations

    def traverse(self, index, item):
        # Traverse through the destinations of a node and update distances if shorter paths are found
        min_dist = float('inf')
        for key, value in item.items():
            min_dist = min(min_dist, key)
            if self.result[value] == float('inf') or self.result[value] > key + self.result[index]:
                self.result[value] = key + self.result[index]
        self.source = item[min_dist]

    def calculate(self, source):
        # Calculate shortest paths from a source node to all other nodes
        self.source = source
        if self.source in self.edges:
            self.result[self.source] = 0  # Initialize distance to the source node as 0
            i = 0
            while i <= len(self.edges):  # Ensure the loop doesn't run indefinitely
                for key, value in self.edges.items():
                    if key == self.source or self.result[key] != float('inf'):
                        self.traverse(key, value)
                i += 1
        print(self.result)  # Print the calculated distances

# Example usage:
if __name__ == '__main__':
    dijkstra = Dijkstra('A', 'B', 'C', 'D', 'E')
    dijkstra.add_path('A', 'B', 4)
    dijkstra.add_path('A', 'C', 2)
    dijkstra.add_path('B', 'C', 5)
    dijkstra.add_path('B', 'D', 10)
    dijkstra.add_path('C', 'D', 3)
    dijkstra.add_path('C', 'E', 7)
    dijkstra.add_path('D', 'E', 8)
    dijkstra.calculate('A')  # Calculate shortest paths from node 'A'
