from json import dump
import spotipy
import spotipy.util as util
from time import perf_counter
from typing import Any, List
from sys import stderr
import config

# Try to get a Spotify token
try:
    token = util.prompt_for_user_token(config.username, config.scope,
                                       config.client_id, config.client_secret,
                                       config.redirect_uri)
    spotify_client = spotipy.Spotify(token)
except:
    # If there's an issue obtaining the token, print an error message
    print(f"Failed to obtain a token for user '{config.username}'", file=stderr)
    exit(1)

# Function to retrieve followed artists
def retrieve_artists() -> List[Any]:
    artists = []  # List to store artists
    last_artist_id = None  # To keep track of the last artist ID processed
    while True:
        # Get the next batch of followed artists
        followed_artists = spotify_client.current_user_followed_artists(50, last_artist_id)
        artist_items = followed_artists["artists"]["items"]
        # If no more artists are returned, break the loop
        if len(artist_items) == 0:
            break
        # Update the last artist ID for the next iteration
        last_artist_id = artist_items[-1]["id"]
        # Extend the list of artists with the newly retrieved ones
        artists.extend(artist_items)
    return artists

# Main function to execute the retrieval process
def main():
    print("Starting to retrieve followed artists...")
    start_time = perf_counter()  # Start time to measure execution time
    artists = retrieve_artists()  # Retrieve followed artists
    elapsed_time = perf_counter() - start_time  # Calculate elapsed time
    print(f"Retrieved {len(artists)} artists in {elapsed_time:.2f} seconds.")

    # Write the retrieved artist data to a JSON file
    with open("json/artists.json", "w") as outfile:
        dump(artists, outfile, indent=4)  # Indent for better readability

# Entry point of the script
if __name__ == "__main__":
    main()  # Execute the main function
