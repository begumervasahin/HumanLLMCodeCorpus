# Import necessary classes and functions
from Weighted_Graph import Weighted_Graph
from functions import update, total_sum

# Define the function to find Minimum Spanning Tree using Prim's algorithm
def Prims(edge_file_name, start_vertex=0, draw=False):
    # Initialize the set of vertices and edges of the Minimum Spanning Tree
    MST = ({start_vertex}, [])
    
    # Create a weighted graph object from the provided edge file
    graph = Weighted_Graph(edge_file_name)
    
    # Optionally draw the original graph
    if draw:
        graph.draw_graph()
    
    # Continue expanding the Minimum Spanning Tree until it covers all vertices
    while MST[0] != graph.vertex_set():
        # Update the Minimum Spanning Tree using Prim's algorithm
        MST = update(MST, graph)
        
        # Optionally draw the current subgraph of the Minimum Spanning Tree
        if draw:
            graph.draw_subgraph(MST)
    
    # Calculate and print the total cost of the Minimum Spanning Tree
    print("The minimum spanning tree costs: " + str(total_sum(MST, graph)))
    
    # Return the Minimum Spanning Tree
    return MST

# Example usage when running the script
if __name__ == "__main__":
    # Replace "your_edge_file_name.txt" with the actual edge file name
    edge_file_name = "your_edge_file_name.txt"
    
    # Call the Prims function with appropriate arguments
    Prims(edge_file_name, start_vertex=0, draw=True)
