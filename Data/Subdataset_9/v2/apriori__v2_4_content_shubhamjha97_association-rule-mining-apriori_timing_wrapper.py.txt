# Import the necessary module
from time import time

# Define a decorator function to measure the execution time of a function
def timeit(fn):
    # Define the wrapper function to calculate the execution time
    def wrapper(*args, **kwargs):
        # Record the start time
        start = time()
        # Call the original function with its arguments
        res = fn(*args, **kwargs)
        # Calculate the elapsed time
        elapsed_time = time() - start
        # Print the function name and its execution time
        print(f"Function {fn.__name__} took {elapsed_time:.6f} seconds.")
        # Return the result of the original function
        return res
    # Return the wrapper function
    return wrapper

# Define a function to demonstrate the timing functionality
@timeit
def some_function():
    # Iterate through a range to simulate some processing
    for i in range(1000000):
        pass

# Entry point of the script
if __name__ == "__main__":
    # Call the decorated function to test the timing functionality
    some_function()
