from dotenv import load_dotenv
import os
import psycopg2
import pandas as pd
import time

# Initialize the database connection variable
conn = None

def create_env_vars():
    """
    Load environment variables from a .env file.
    """
    global hostname, database, username, password
    load_dotenv()
    hostname = os.getenv('hostname')
    port = os.getenv('port')  # Unused variable
    database = os.getenv('database')
    username = os.getenv('username')
    password = os.getenv('password')

def connect_to_database():
    """
    Establish a connection to the PostgreSQL database.
    """
    global conn
    conn = psycopg2.connect(host=hostname, database=database, user=username, password=password)

def get_studies():
    """
    Retrieve data from the 'studies' table.
    """
    return pd.read_sql('SELECT * FROM ctgov.studies', con=conn)

def get_keywords():
    """
    Retrieve data from the 'keywords' table.
    """
    return pd.read_sql('SELECT * FROM ctgov.keywords', con=conn)

def join_tables(first_table_name, second_table_name):
    """
    Perform a right join between two tables based on the 'nct_id' column.
    """
    return pd.read_sql(f'''
        SELECT * 
        FROM {first_table_name} 
        RIGHT JOIN {second_table_name} 
        ON {first_table_name}.nct_id = {second_table_name}.nct_id
    ''', con=conn)

if __name__ == '__main__':
    create_env_vars()  # Load environment variables
    connect_to_database()  # Connect to the database

    # Measure the time taken for joining tables using the custom function
    start_time = time.time()
    results = join_tables('ctgov.studies', 'ctgov.keywords')
    end_time = time.time()
    print("Time taken for joining tables:", end_time - start_time)

    # Measure the time taken for joining tables using a direct SQL query
    start_time = time.time()
    pd.read_sql('SELECT * FROM ctgov.studies RIGHT JOIN ctgov.keywords ON ctgov.studies.nct_id = ctgov.keywords.nct_id', con=conn)
    end_time = time.time()
    print("Time taken for joining tables (direct query):", end_time - start_time)
