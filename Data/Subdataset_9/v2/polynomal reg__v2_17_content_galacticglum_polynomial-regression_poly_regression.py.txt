# Import necessary libraries
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
import sympy as sp
from sympy.abc import x as spx
import argparse
import os

# Define command-line arguments for better interaction
parser = argparse.ArgumentParser()
parser.add_argument('--input', help='Specify the path to your data file (CSV format), relative to the current directory.')
parser.add_argument('--poly_degree', default=2, type=int, help='Degree of the polynomial regression (default is 2).')
args = parser.parse_args()

# Load the dataset
dataset = pd.read_csv(args.input)
X = dataset.iloc[:, 0].values
y = dataset.iloc[:, 1].values

# Get information about the dataset
dataset_name = os.path.splitext(args.input)[0].replace('_', ' ').title()
dataset_columns = dataset.columns.tolist()

# Scatter plot to visualize the dataset
plt.scatter(X, y, color='green')

# Define a function to plot polynomial regression
def plot_polynomial_regression():
    # Calculate the regression coefficients
    weights = np.polyfit(X, y, args.poly_degree)
    # Create a polynomial model
    model = np.poly1d(weights)
    # Plot the regression line
    plt.plot(X, model(X), color='orange')
    # Print the polynomial equation in LaTeX format
    sp.init_printing()
    print(sp.latex(sp.Poly(model.coef, spx).as_expr()))

# Define a function to display the plot
def show_plot():
    plt.title(dataset_name)
    plt.xlabel(dataset_columns[0])
    plt.ylabel(dataset_columns[1])
    plt.show()

# Call the functions to plot and display the regression
plot_polynomial_regression()
show_plot()
