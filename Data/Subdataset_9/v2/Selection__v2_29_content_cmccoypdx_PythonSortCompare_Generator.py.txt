import random

def generate_array(size):
    """
    Generate a list of random integers within the range [-size, size].

    Args:
        size (int): The size of the array.

    Returns:
        list: A list of random integers.
    """
    return [random.randint(-size, size) for _ in range(size)]

def generate_payload():
    """
    Generate a payload containing lists of random integers with varying sizes.

    Returns:
        list: A list containing lists of random integers.
    """
    sizes = [10, 100, 1000, 10000, 100000, 1000000, 10000000]
    payload = [generate_array(size) for size in sizes]
    return payload

if __name__ == "__main__":
    # Example usage:
    payload = generate_payload()
    print(payload)
