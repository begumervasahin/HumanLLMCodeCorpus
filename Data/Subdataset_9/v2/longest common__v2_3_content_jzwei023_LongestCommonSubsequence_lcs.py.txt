# Define a function to calculate the longest common subsequence (LCS) of two strings
def calculate_LCS(A, B):
    # Get the lengths of the input strings
    n = len(A)
    m = len(B)
    
    # Initialize matrices to store the length of LCS and parent array
    opt = [[0 for _ in range(m + 1)] for _ in range(n + 1)]
    pi = [[0 for _ in range(m + 1)] for _ in range(n + 1)]
    
    # Populate the matrices to compute the length of LCS and the parent array
    for i in range(1, n + 1):
        for j in range(1, m + 1):
            if A[i - 1] == B[j - 1]:
                opt[i][j] = opt[i - 1][j - 1] + 1
                pi[i][j] = 0  # Diagonal direction
            elif opt[i][j - 1] >= opt[i - 1][j]:
                opt[i][j] = opt[i][j - 1]
                pi[i][j] = 1  # Left direction
            else:
                opt[i][j] = opt[i - 1][j]
                pi[i][j] = 2  # Up direction

    # Reconstruct the LCS from the parent array
    i, j = n, m
    lcs = ''
    while i > 0 and j > 0:
        if pi[i][j] == 0:  # Diagonal direction
            lcs = A[i - 1] + lcs
            i -= 1
            j -= 1
        elif pi[i][j] == 2:  # Up direction
            i -= 1
        else:
            j -= 1  # Left direction

    # Return the length of LCS and the LCS itself
    return str(opt[n][m]), lcs

# Define file paths for input and output
input_file = "input.txt"
output_file = "output.txt"

# Read input strings from file
with open(input_file, "r") as file_in:
    lines = file_in.readlines()
    A = lines[0].strip()  # First string
    B = lines[1].strip()  # Second string

# Calculate LCS length and sequence
length, sequence = calculate_LCS(A, B)

# Write LCS length and sequence to output file
with open(output_file, "w") as file_out:
    file_out.write(length + "\n")  # Write LCS length
    file_out.write(sequence)  # Write LCS sequence
