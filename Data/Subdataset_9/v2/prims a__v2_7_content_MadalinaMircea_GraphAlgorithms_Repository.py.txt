class Graph:
    def __init__(self, num_vertices):
        # Initialize the graph with the given number of vertices
        self.num_vertices = num_vertices
        # Create an empty adjacency list for each vertex
        self.adjacency_list = [[] for _ in range(num_vertices)]

    def addEdge(self, v1, v2, c=1):
        # Add an edge between vertices v1 and v2 with an optional cost c
        self.adjacency_list[v1].append((v2, c))

    def getNrVertices(self):
        # Return the total number of vertices in the graph
        return self.num_vertices

    def getNrEdges(self):
        # Return the total number of edges in the graph
        total_edges = sum(len(edges) for edges in self.adjacency_list)
        return total_edges // 2  # Divide by 2 because each edge is counted twice (for undirected graphs)

    def isEdge(self, v1, v2):
        # Check if there exists an edge between vertices v1 and v2
        return any(v2 in neighbors for neighbors in self.adjacency_list[v1])

    def getDegree(self, v):
        # Return the degree of the given vertex v (number of edges incident to v)
        return len(self.adjacency_list[v])

    def parseNeighbours(self, v):
        # Return a list of neighbors of the given vertex v
        return [neighbor[0] for neighbor in self.adjacency_list[v]]

    def parseAll(self):
        # Return a list of all edges in the graph
        all_edges = []
        for v in range(self.num_vertices):
            for neighbor, _ in self.adjacency_list[v]:
                all_edges.append((v, neighbor))
        return all_edges


class Repository:
    def __init__(self, filename="graph1.txt"):
        # Initialize the repository by loading data from the given file
        self._loadFromFile(filename)

    def _loadFromFile(self, filename):
        try:
            with open(filename, 'r') as file:
                # Read the number of vertices from the first line of the file
                text = file.readline()
                num_vertices = int(text.strip())
                # Create a new graph instance with the read number of vertices
                self.graph = Graph(num_vertices)
                # Read each line of the file representing an edge and add it to the graph
                for line in file:
                    v1, v2 = map(int, line.split())
                    self.graph.addEdge(v1, v2)
                    # Assuming undirected graph, add the reverse edge as well
                    self.graph.addEdge(v2, v1)
        except FileNotFoundError:
            print("File not found.")

    def getNrVertices(self):
        # Delegate the call to getNrVertices method of the underlying graph
        return self.graph.getNrVertices()

    def getNrEdges(self):
        # Delegate the call to getNrEdges method of the underlying graph
        return self.graph.getNrEdges()

    def isEdge(self, v1, v2):
        # Delegate the call to isEdge method of the underlying graph
        return self.graph.isEdge(v1, v2)

    def getDegree(self, v):
        # Delegate the call to getDegree method of the underlying graph
        return self.graph.getDegree(v)

    def getNeighbours(self, v):
        # Delegate the call to parseNeighbours method of the underlying graph
        return self.graph.parseNeighbours(v)

    def showAll(self):
        # Delegate the call to parseAll method of the underlying graph
        return self.graph.parseAll()

    def addEdge(self, v1, v2, c=1):
        # Delegate the call to addEdge method of the underlying graph
        return self.graph.addEdge(v1, v2, c)


# Example usage
repo = Repository("graph1.txt")
print("Number of vertices:", repo.getNrVertices())
print("Number of edges:", repo.getNrEdges())
print("Is there an edge between vertices 1 and 2?", repo.isEdge(1, 2))
print("Degree of vertex 1:", repo.getDegree(1))
print("Neighbours of vertex 1:", repo.getNeighbours(1))
print("All edges in the graph:", repo.showAll())
