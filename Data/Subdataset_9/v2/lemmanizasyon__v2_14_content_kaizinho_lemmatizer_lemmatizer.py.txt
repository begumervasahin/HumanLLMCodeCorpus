import re
import json

def lemmatize(sentence, word_list):
    """
    Lemmatize a sentence using a given word list.
    
    Args:
        sentence (str): The sentence to lemmatize.
        word_list (dict): A dictionary containing words and their lemmas.
        
    Returns:
        str: The lemmatized sentence.
    """
    lemm_sentence = ""
    for word in sentence.split():
        lemm_sentence += word_list.get(word, word) + " "
    return lemm_sentence.strip()

def extract_words(raw_data_file, filename="data/swedish_words.json"):
    """
    Extract words and their lemmas from raw data and store them in a JSON file.
    
    Args:
        raw_data_file (str): The path to the raw data file.
        filename (str): The name of the JSON file to store the extracted data.
    """
    # Regular expression patterns for extracting data
    pattern1 = r"^(===(Verb|Substantiv|Adjektiv|Adverb|Konjuktion|Pronomen|Räkneord|Interjektion|Konjugation)===)$"
    pattern2 = r"^'''([a-zA-ZåäöÅÄÖ]+)'''$"
    pattern3 = r"^'''([a-zA-ZåäöÅÄÖ]+)''' \(([^)]+)\)$"
    
    data = {}  # Dictionary to store extracted data
    current_type = None  # Variable to keep track of the current type
    
    # Open the raw data file and process each line
    with open(raw_data_file, "r", encoding='utf-8') as input_file:
        for line in input_file:
            match1 = re.match(pattern1, line)
            match2 = re.match(pattern2, line)
            match3 = re.match(pattern3, line)
            
            if match1:  # If line matches the type pattern
                current_type = match1.group(2)  # Set the current type
            elif match2 and current_type:  # If line matches the word pattern and type is set
                word = match2.group(1)  # Get the word
                if current_type not in data:
                    data[current_type] = {}  # Initialize dictionary for the current type if not present
                data[current_type][word] = ""  # Add the word to the dictionary
            elif match3 and current_type:  # If line matches the word with lemma pattern and type is set
                word = match3.group(1)  # Get the word
                lemma = match3.group(2)  # Get the lemma
                if current_type not in data:
                    data[current_type] = {}  # Initialize dictionary for the current type if not present
                data[current_type][word] = lemma  # Add the word and lemma to the dictionary
    
    # Write the extracted data to a JSON file
    with open(filename, "w", encoding='utf-8') as output_file:
        json.dump(data, output_file, ensure_ascii=False, indent=4)

# Example usage:
extract_words("raw_data_file.txt", "swedish_words.json")
