import cv2
import numpy as np
import matplotlib.pyplot as plt
import glob
import sys
import os

# Function to calculate variance of a set of grayscale images
def calculate_variance(file_names):
    # Read the first image to initialize variables
    img = cv2.imread(file_names[0], cv2.IMREAD_GRAYSCALE)
    img_sum = np.zeros_like(img, dtype=np.float64)
    img_sqr_sum = np.zeros_like(img, dtype=np.float64)
    
    # Iterate over all image files
    for name in file_names:
        # Read the image and convert it to float64 for accurate calculations
        img = cv2.imread(name, cv2.IMREAD_GRAYSCALE).astype(np.float64)        
        img_sum += img  # Accumulate pixel values
        img_sqr_sum += img ** 2  # Accumulate squared pixel values
    
    # Calculate the mean and variance
    n = len(file_names)
    mean = img_sum / n
    variance = img_sqr_sum / n - (mean ** 2)
    
    return variance

# Main function
def main(argv):
    # Check if the correct number of arguments is provided
    if len(argv) != 2:
        print("Usage: python %s [folder of images]" % argv[0])
        sys.exit(-1)
    
    # Get the path to the folder containing images
    path = argv[1]
    
    # Calculate the variance of images in the specified folder
    variance = calculate_variance(glob.glob(os.path.join(path, "*.jpg")))
    
    # Display the variance image
    plt.imshow(variance, cmap='gray')
    plt.colorbar()
    plt.show()
    
    # Save the variance matrix as a NumPy array
    if path.endswith('/'):
        path = path[:-1]  # Remove trailing slash if present
    folder_name = os.path.basename(path)
    np.save("%s_variance.npy" % folder_name, variance)

if __name__ == '__main__':
    main(sys.argv)
