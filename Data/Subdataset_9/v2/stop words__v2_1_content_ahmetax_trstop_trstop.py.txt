# Create an empty dictionary to store Turkish words and their frequencies
turkish_dictionary = {}

# Load dictionary from file
with open("dosyalar/derlemtr2016-10000.txt", 'r', encoding='utf-8') as dictionary_file:
    # Iterate through each line in the file
    for line in dictionary_file:
        # Check if the line represents a word and its frequency
        if line[0] not in ['0', '1']:
            continue
        # Extract frequency and word from the line
        frequency, word = line.strip().split()
        # Add the word and its frequency to the dictionary
        turkish_dictionary[word] = int(frequency)

# Function to check if a word is a stop word
def is_stop_word(word):
    """
    Checks if the given word is in the Turkish dictionary.
    
    Parameters:
        word (str): The word to check.
        
    Returns:
        bool: True if the word is a stop word, False otherwise.
    """
    return word in turkish_dictionary

# Function to get the frequency of a word
def get_word_frequency(word):
    """
    Returns the frequency of the given word in the Turkish dictionary.
    
    Parameters:
        word (str): The word to get the frequency for.
        
    Returns:
        int: The frequency of the word, or 0 if the word is not in the dictionary.
    """
    if is_stop_word(word):
        return turkish_dictionary[word]
    else:
        return 0

if __name__ == '__main__':
    # Open file containing stop words
    with open("dosyalar/turkce-stop-words", encoding="utf-8") as stop_words_file:
        # Iterate through each stop word in the file
        for stop_word in stop_words_file:
            # Remove any leading or trailing whitespace
            stop_word = stop_word.strip()
            # Check if the stop word is not in the dictionary
            if stop_word not in turkish_dictionary:
                # Print a message indicating that the stop word is not in the dictionary
                print('{} is not in the dictionary'.format(stop_word))
