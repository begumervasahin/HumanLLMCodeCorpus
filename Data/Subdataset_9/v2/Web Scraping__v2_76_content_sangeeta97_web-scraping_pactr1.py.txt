import requests
from bs4 import BeautifulSoup
import pandas as pd
import time
from multiprocessing.dummy import Pool as ThreadPool
import multiprocessing as mp

# Generate a list of trial IDs and their corresponding URLs
start_id = 86
end_id = 10000  # Note: Adjust this end_id as needed
index = list(range(start_id, end_id))
urls = [f"https://pactr.samrc.ac.za/TrialDisplay.aspx?TrialID={i}" for i in index]

# Prepare a DataFrame to hold the trial IDs and URLs
df = pd.DataFrame({'number': index, 'url': urls})

# Define a function to fetch and parse the data from each trial URL
def extract_text(url):
    try:
        # Fetch the page content
        response = requests.get(url, verify=False)  # 'verify=False' skips SSL certificate verification
        soup = BeautifulSoup(response.text, 'lxml')
        
        # Extract specific pieces of data from the structured HTML
        primary_topic = soup.find_all('table')[2].find_all('td')[2].get_text().strip()
        participant_type = soup.find_all('table')[2].find_all('td')[9].get_text().strip()
        study_type = soup.find_all('table')[2].find_all('td')[11].get_text().strip()
        notes = ' '.join([td.text.strip() for td in soup.find_all('table')[4].find_all('td')])
        
        # Artificial delay to avoid overwhelming the server
        time.sleep(2)
        
        return primary_topic, participant_type, study_type, notes
    except Exception as e:
        # If an error occurs, print it out and return a placeholder
        print(f"Error processing URL: {url} | Error: {str(e)}")
        return ('no_value',) * 4  # Return a tuple with 'no_value' for each column

# Use multiprocessing to fetch and parse data from URLs concurrently
pool = ThreadPool(mp.cpu_count())
results = pool.map(extract_text, urls)
pool.close()
pool.join()

# Convert the results into a DataFrame
df2 = pd.DataFrame(results, columns=['Primary Topic', 'Participant Type', 'Study Type', 'Notes'])

# Merge the original DataFrame with the new data
df_final = pd.concat([df, df2], axis=1)

# Save the combined data to an Excel file
df_final.to_excel('pac_trials_info.xlsx', index=False)

print("Data extraction and saving completed.")
