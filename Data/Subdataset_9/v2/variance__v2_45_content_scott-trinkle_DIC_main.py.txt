# Importing necessary modules
from dic.experiment import Experiment
from dic.plots import generate_plots

# Define the main function
def main():
    # Setting up the experiment configuration
    experiment = Experiment(
        lens=40,
        weak_grad=False,
        lamda=546,
        approaches=['A: 2x2', 'A: 2x3', 'B: 2x3', 'B: 2x4'],
        k=1e2,
        fromZero=True,
        save=False,
        filepath=None
    )

    # Generating data for the experiment
    sigma_g, sigma_t = experiment.generate_data(sample_size=100)

    # Generating plots based on the experiment data
    generate_plots(
        sigma_g, sigma_t, experiment,
        polar=True,
        areaplot=True,
        SNR=False,
        report=False,
        show=True
    )

# Check if the script is being run directly
if __name__ == "__main__":
    # Call the main function
    main()
