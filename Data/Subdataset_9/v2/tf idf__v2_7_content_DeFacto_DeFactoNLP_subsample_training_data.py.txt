import os
import jsonlines
from random import shuffle

# Define file paths
train_file_path = "data/train.jsonl"
subsampled_file_path = "data/subsample_train.jsonl"

# Define the number of samples required for each label
num_required_support = 3000
num_required_refute = 3000
num_required_notenoughinfo = 4000

# Open and read the training file
with jsonlines.open(train_file_path) as train_file:
    # Read the lines from the file into a list
    train_set = [line for line in train_file]

# Shuffle the training set
shuffle(train_set)

# Initialize lists to store samples for each label
support = []
refute = []
not_enough_info = []

# Iterate through each line in the shuffled training set
for line in train_set:
    label = line['label']  # Extract the label from the line
    # Check the label and the number of required samples for each label
    if label == "SUPPORTS" and num_required_support > 0:
        support.append(line)  # Append the line to the support list
        num_required_support -= 1  # Decrement the count of required support samples
    elif label == "REFUTES" and num_required_refute > 0:
        refute.append(line)  # Append the line to the refute list
        num_required_refute -= 1  # Decrement the count of required refute samples
    elif label == "NOT ENOUGH INFO" and num_required_notenoughinfo > 0:
        not_enough_info.append(line)  # Append the line to the not_enough_info list
        num_required_notenoughinfo -= 1  # Decrement the count of required not_enough_info samples

# Concatenate the lists for different labels
subsampled = support + refute + not_enough_info

# Shuffle the subsampled list
shuffle(subsampled)

# Write subsampled data to a new file
with jsonlines.open(subsampled_file_path, mode='w') as writer:
    for sample in subsampled:
        writer.write(sample)  # Write each sample to the subsampled file

# Print the total lines in the original and subsampled datasets
print(f"Total lines in the original dataset: {len(train_set)}")
print(f"Total lines in the subsampled dataset: {len(subsampled)}")
print("Subsampling and writing completed successfully.")
