import numpy as np
from matplotlib import pyplot
from collections import Counter
import warnings

# Function to perform k-nearest neighbors classification
def k_nearest_neighbors(data, predict, k=3):
    # Check if the dataset has enough data points for the specified k value
    if len(data) >= k:
        warnings.warn("k is too small")
    
    # Calculate Euclidean distances between the new point and all existing data points
    distances = []
    for group in data:
        for features in data[group]:
            euclidean_distance = np.linalg.norm(np.array(features) - np.array(predict))
            distances.append([euclidean_distance, group])
    
    # Sort the distances and get the top k nearest neighbors
    sorted_distances = [i[1] for i in sorted(distances)]
    top_nearest = sorted_distances[:k]
    
    # Determine the most common group among the nearest neighbors
    group_res = Counter(top_nearest).most_common(1)[0][0]
    
    # Calculate confidence as the proportion of the most common group among the nearest neighbors
    confidence = Counter(top_nearest).most_common(1)[0][1] * 1.0 / k
    
    return group_res, confidence

if __name__ == '__main__':
    # Example dataset with two groups of data points
    dataset = {'black': [[1, 2], [2, 3], [3, 1]], 'red': [[6, 5], [7, 7], [8, 6]]}
    
    # New features to be classified
    new_features = [3.5, 5.2]
    
    # Plot existing data points
    for group, points in dataset.items():
        for point in points:
            pyplot.scatter(point[0], point[1], s=50, color=group)
    
    # Predict the group for new features using k-nearest neighbors
    which_group, confidence = k_nearest_neighbors(dataset, new_features, k=3)
    print("Predicted Group:", which_group)
    print("Confidence:", confidence)
    
    # Plot the new feature point with the predicted group color
    pyplot.scatter(new_features[0], new_features[1], s=300, color=which_group)
    
    pyplot.show()
