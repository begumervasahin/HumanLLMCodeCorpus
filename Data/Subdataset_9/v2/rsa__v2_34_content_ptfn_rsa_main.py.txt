import random
import sys

def main():
    # Generate prime numbers within the range [1, 2^8]
    prime_numbers = generate_primes(1, 2**8)

    # Choose random prime numbers p and q
    p = prime_numbers[random.randint(0, len(prime_numbers)-1)]
    q = prime_numbers[random.randint(0, len(prime_numbers)-1)]

    # Calculate n and Euler's totient function phi
    n = p * q
    phi = (p - 1) * (q - 1)

    # Choose random integer e coprime with phi
    e = prime_numbers[random.randint(0, len(prime_numbers)-1)]

    # Calculate d, the modular multiplicative inverse of e modulo phi
    d = modular_inverse(e, phi)

    # Print the generated key pair
    print(f"p = {p}\nq = {q}\nn = {n}\nphi = {phi}\ne = {e}\nd = {d}")

    # Check for command-line arguments for encryption/decryption
    if len(sys.argv) >= 3:
        choice = sys.argv[1]
        message = int(sys.argv[2])
        if choice in ("enc", "e"):
            print(f"Message = {message}\nEncrypted = {encrypt(message, e, n)}")
        elif choice in ("dec", "d"):
            print(f"Ciphertext = {message}\nDecrypted = {decrypt(message, d, n)}")
    else:
        print("Please provide arguments for encryption/decryption.")

def generate_primes(start, end):
    primes = []
    for num in range(start, end):
        if is_prime(num):
            primes.append(num)
    return primes

def is_prime(num):
    if num <= 1:
        return False
    for i in range(2, int(num**0.5) + 1):
        if num % i == 0:
            return False
    return True

def modular_inverse(e, phi):
    for d in range(0, phi):
        if (e * d) % phi == 1:
            return d

def encrypt(message, e, n):
    return (message ** e) % n

def decrypt(ciphertext, d, n):
    return (ciphertext ** d) % n

if __name__ == "__main__":
    main()
