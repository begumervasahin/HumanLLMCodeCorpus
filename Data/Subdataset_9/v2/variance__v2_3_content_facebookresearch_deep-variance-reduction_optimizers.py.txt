import torch.optim as optim
import torch_svrg
import recompute_svrg
import scsg

def select_optimizer(model, args):
    print("Using", args.method)

    if args.method == "sgd":
        optimizer = optim.SGD(model.parameters(), lr=args.lr, weight_decay=args.decay, momentum=args.momentum)
    elif args.method == "svrg":
        optimizer = torch_svrg.SVRG(model.parameters(), args=args, lr=args.lr,
                                    nbatches=args.nbatches, momentum=args.momentum, weight_decay=args.decay)
    elif args.method == "recompute_svrg":
        optimizer = recompute_svrg.RecomputeSVRG(model.parameters(), lr=args.lr,
                                                nbatches=args.nbatches, model=model,
                                                vr_bn_at_recalibration=args.vr_bn_at_recalibration,
                                                vr_from_epoch=args.vr_from_epoch,
                                                momentum=args.momentum, weight_decay=args.decay)
    elif args.method == "scsg":
        optimizer = scsg.SCSG(model.parameters(), args=args, lr=args.lr,
                              nbatches=args.nbatches, model=model,
                              vr_bn_at_recalibration=args.vr_bn_at_recalibration,
                              vr_from_epoch=args.vr_from_epoch,
                              momentum=args.momentum, weight_decay=args.decay)
    else:
        raise ValueError("Unknown optimizer method:", args.method)

    return optimizer

class Arguments:
    def __init__(self, method, lr, decay, momentum, nbatches, vr_bn_at_recalibration, vr_from_epoch):
        self.method = method
        self.lr = lr
        self.decay = decay
        self.momentum = momentum
        self.nbatches = nbatches
        self.vr_bn_at_recalibration = vr_bn_at_recalibration
        self.vr_from_epoch = vr_from_epoch

class ExampleModel:
    def __init__(self):
        self.parameters = lambda: "Parameters"

# Example usage:
args = Arguments(method="svrg", lr=0.01, decay=0.0001, momentum=0.9, nbatches=100, vr_bn_at_recalibration=False, vr_from_epoch=1)
model = ExampleModel()
optimizer = select_optimizer(model, args)
