# This script reads data from a CSV file named "distance_matrix_1698.csv".
# It prints the length of the fifth line of the file plus 1698,
# and then prints the third element of that line.

# Author: 's'

# Open the file for reading
with open("distance_matrix_1698.csv", "r+") as file:
    # Initialize an empty list to store data
    data = []
    line_counter = 0

    # Iterate through each line in the file
    for line in file:
        # Check if we have read 5 lines already
        if line_counter == 5:
            break
        
        # Increment the line counter
        line_counter += 1
        
        # Split the line into a list of values
        data = line.split(",")
    
    # Print the length of the list plus 1698
    print("Total elements in the fifth line plus 1698:", len(data) + 1698)
    
    # Print the third element of the list
    print("Third element of the fifth line:", data[2])
