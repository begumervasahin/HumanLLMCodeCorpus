import sys
import getopt
import scripts.cryptage as cryptage
import scripts.decryptage as decryptage
import scripts.cryptImg as cryptImg
import scripts.decryptImg as decryptImg

def main():
    try:
        opts, args = getopt.getopt(sys.argv[1:], "cdpi:o:k:", ["crypt", "decrypt", "picture", "input", "output", "key"])
    except getopt.GetoptError as e:
        print(f"Error: {e}")
        sys.exit(2)

    decryption = False
    encryption = False
    picture = False
    input_file = ""
    output_file = ""
    key_file = ""

    for opt, arg in opts:
        if opt in ("-c", "--crypt"):
            encryption = True
        elif opt in ("-d", "--decrypt"):
            decryption = True
        elif opt in ("-p", "--picture"):
            picture = True
        elif opt in ("-i", "--input"):
            input_file = arg
        elif opt in ("-o", "--output"):
            output_file = arg
        elif opt in ("-k", "--key"):
            key_file = arg

    if not input_file:
        print("Error: You must specify an input file.")
        sys.exit(2)

    if not key_file and decryption:
        key_file = input_file + "key"

    if not output_file:
        output_file = "out"

    if picture:
        if encryption:
            cryptImg.cryptImg(input_file, output_file)
        if decryption:
            if encryption:
                decryptImg.decryptImg(output_file, "Decrypt_" + input_file, key_file)
            else:
                decryptImg.decryptImg(input_file, output_file, key_file)
    else:
        if encryption:
            cryptage.crypt(input_file, output_file)
        if decryption:
            if encryption:
                decryptage.decrypt(output_file, "Decrypt_" + input_file, key_file)
            else:
                decryptage.decrypt(input_file, output_file, key_file)

if __name__ == "__main__":
    main()
