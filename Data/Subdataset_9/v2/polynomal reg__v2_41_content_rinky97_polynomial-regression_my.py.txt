# Import necessary libraries
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import PolynomialFeatures

# Load the dataset
dataset = pd.read_csv('Position_Salaries.csv')

# Extract features (X) and target variable (y)
X = dataset.iloc[:, 1:2].values  # Independent variable (Position Level)
y = dataset.iloc[:, 2].values    # Dependent variable (Salary)

# Fit Linear Regression
linear_regressor = LinearRegression()
linear_regressor.fit(X, y)

# Fit Polynomial Regression
polynomial_features = PolynomialFeatures(degree=2)
X_poly = polynomial_features.fit_transform(X)
polynomial_regressor = LinearRegression()
polynomial_regressor.fit(X_poly, y)

# Visualize Linear Regression results
plt.scatter(X, y, color='red')                           # Plot original data points
plt.plot(X, linear_regressor.predict(X), color='blue')   # Plot linear regression line
plt.title('Linear Regression')
plt.xlabel('Position Level')
plt.ylabel('Salary')
plt.show()

# Visualize Polynomial Regression results
plt.scatter(X, y, color='red')                                    # Plot original data points
plt.plot(X, polynomial_regressor.predict(X_poly), color='blue')   # Plot polynomial regression curve
plt.title('Polynomial Regression')
plt.xlabel('Position Level')
plt.ylabel('Salary')
plt.show()
