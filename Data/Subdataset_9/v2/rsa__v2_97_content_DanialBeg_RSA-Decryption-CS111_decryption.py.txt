def calculate_factors(number):
    # Function to calculate factors of a number
    factors = []
    for i in range(1, number + 1):
        if number % i == 0:
            factors.append(i)
    return factors[1], factors[2]  # Returning the second and third factors (first factor is always 1)

def modular_inverse(a, b):
    # Function to compute the modular multiplicative inverse using brute force
    for num in range(1, b):
        if (a * num) % b == 1:
            return num
    return 1  # If inverse doesn't exist, return 1 as a default value
