from scrapy import signals

class CraigslistSpiderMiddleware:
    # Constructor method to create an instance of the middleware
    @classmethod
    def from_crawler(cls, crawler):
        middleware_instance = cls()
        # Connect to the spider_opened signal
        crawler.signals.connect(middleware_instance.spider_opened, signal=signals.spider_opened)
        return middleware_instance

    # Process input received by the spider
    def process_spider_input(self, response, spider):
        return None

    # Process spider output
    def process_spider_output(self, response, result, spider):
        for item in result:
            yield item

    # Handle spider exceptions
    def process_spider_exception(self, response, exception, spider):
        pass

    # Process start requests for the spider
    def process_start_requests(self, start_requests, spider):
        for request in start_requests:
            yield request

    # Log when spider is opened
    def spider_opened(self, spider):
        spider.logger.info('Spider opened: %s' % spider.name)


class CraigslistDownloaderMiddleware:
    # Constructor method to create an instance of the downloader middleware
    @classmethod
    def from_crawler(cls, crawler):
        middleware_instance = cls()
        # Connect to the spider_opened signal
        crawler.signals.connect(middleware_instance.spider_opened, signal=signals.spider_opened)
        return middleware_instance

    # Process requests made by the downloader middleware
    def process_request(self, request, spider):
        return None

    # Process responses received by the downloader middleware
    def process_response(self, request, response, spider):
        return response

    # Handle exceptions raised during processing
    def process_exception(self, request, exception, spider):
        pass

    # Log when spider is opened
    def spider_opened(self, spider):
        spider.logger.info('Spider opened: %s' % spider.name)
