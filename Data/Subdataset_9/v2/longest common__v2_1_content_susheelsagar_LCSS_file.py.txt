# Open the file "distance2.csv" in write mode
with open("distance2.csv", "w+") as file:
    # Initialize a list called 's'
    s = [0, ",", 1]
    
    # Write the list 's' to the file
    file.write(str(s))
    
    # Write the string "hello" to the file followed by a newline character
    file.write("hello\n")
    
    # Write the list 's' to the file again
    file.write(str(s))
    
    # Create a 2x2 numpy array filled with zeros
    similarity = np.zeros((2, 2))
    
    # Modify the elements of the 'similarity' array
    similarity[1][0] = 2

    similarity[1][0] -= 1
    
    # Print the modified elements of the 'similarity' array
