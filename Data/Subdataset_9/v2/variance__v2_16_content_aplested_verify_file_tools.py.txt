import os

# Function to read lines from a file and strip them
def read_lines_from_file(file_path):
    print('Reading file: ' + file_path)
    with open(file_path, 'r') as file:
        lines = file.readlines()
    # Strip first and last lines and remove leading/trailing whitespace
    stripped_lines = [line.strip() for line in lines[1:-1]]
    print("First line of content: " + stripped_lines[0])
    return stripped_lines

# Function to write lines of data to a file
def write_lines_to_file(file_path, lines):
    with open(file_path, "w") as file:
        for line in lines:
            # Join elements with tab delimiter and write to file
            joined_line = '\t'.join(line)
            file.write(joined_line + '\n')

# Function to add a prefix to a filename
def add_prefix_to_filename(file_path, prefix=""):
    directory, filename = os.path.split(file_path)
    prefixed_filename = prefix + filename
    return os.path.join(directory, prefixed_filename)

# Function to convert lines of ASCII text into traces (a list of lists of floats)
def lines_to_traces(lines):
    split_lines = [line.split('\t') for line in lines]
    num_of_traces = len(split_lines[0])
    traces = [[] for _ in range(num_of_traces)]
    for line in split_lines:
        for i in range(num_of_traces):
            traces[i].append(float(line[i]))
    return traces

# Function to convert traces (a list of lists of floats) into lines of ASCII text
def traces_to_lines(traces):
    num_of_points = len(traces[0])
    lines = [[] for _ in range(num_of_points)]
    for point in range(num_of_points):
        for trace in traces:
            lines[point].append(str(trace[point]))
    return lines

# Example usage
if __name__ == "__main__":
    # Reading data from a file
    lines = read_lines_from_file("example.txt")

    # Converting lines to traces
    traces = lines_to_traces(lines)

    # Modifying traces if needed
    # For example, adding 1 to each value
    for trace in traces:
        for i in range(len(trace)):
            trace[i] += 1

    # Converting traces back to lines
    modified_lines = traces_to_lines(traces)

    # Writing modified lines back to file
    write_lines_to_file("modified_example.txt", modified_lines)
