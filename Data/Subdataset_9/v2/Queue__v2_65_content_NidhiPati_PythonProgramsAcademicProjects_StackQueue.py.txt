import sys

class MyStack:
    """Implementation of a stack data structure."""

    def __init__(self):
        """Initialize an empty stack."""
        self._data = []

    def push(self, item):
        """Push an item onto the stack."""
        self._data.append(item)

    def peek(self):
        """Return the top item of the stack without removing it."""
        if len(self._data) == 0:
            raise IndexError("Stack is empty")
        return self._data[-1]

    def pop(self):
        """Remove and return the top item of the stack."""
        if len(self._data) == 0:
            raise IndexError("Stack is empty")
        return self._data.pop()

    def __len__(self):
        """Return the number of items in the stack."""
        return len(self._data)

    def get_size(self):
        """Return the size of the stack in bytes."""
        return sys.getsizeof(self._data)

class CircularQueue:
    """Implementation of a circular queue data structure."""

    DEFAULT_CAPACITY = 10

    def __init__(self):
        """Initialize an empty circular queue."""
        self._data = [None] * CircularQueue.DEFAULT_CAPACITY
        self._number_of_items = 0
        self._front_index = 0

    def __len__(self):
        """Return the number of items in the queue."""
        return self._number_of_items

    def is_empty(self):
        """Check if the queue is empty."""
        return self._number_of_items == 0

    def first(self):
        """Return the first item in the queue without removing it."""
        if self.is_empty():
            raise IndexError("Empty Queue")
        return self._data[self._front_index]

    def dequeue(self):
        """Remove and return the first item in the queue."""
        if self.is_empty():
            raise IndexError("Empty Queue")
        value = self._data[self._front_index]
        self._data[self._front_index] = None
        self._front_index = (self._front_index + 1) % len(self._data)
        self._number_of_items -= 1
        return value

    def enqueue(self, item):
        """Add an item to the end of the queue."""
        if self._number_of_items == len(self._data):
            self._resize(len(self._data) * 2)
        next_available_index = (self._front_index + self._number_of_items) % len(self._data)
        self._data[next_available_index] = item
        self._number_of_items += 1

    def _resize(self, capacity):
        """Resize the internal array of the queue."""
        old_data = self._data
        self._data = [None] * capacity
        current_index = self._front_index
        for index in range(0, len(old_data)):
            self._data[index] = old_data[current_index]
            current_index = (current_index + 1) % len(old_data)
        self._front_index = 0

    def get_size(self):
        """Return the size of the queue in bytes."""
        return sys.getsizeof(self._data)

if __name__ == "__main__":
    # Example usage of MyStack
    stack = MyStack()
    stack.push(1)
    stack.push(2)
    stack.push(3)
    print(stack.pop())  # Output: 3
    print(stack.peek())  # Output: 2

    queue = CircularQueue()
    queue.enqueue(1)
    queue.enqueue(2)
    queue.enqueue(3)
    print(queue.dequeue())  # Output: 1
    print(queue.first())    # Output: 2
