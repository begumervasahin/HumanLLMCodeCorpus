# Dijkstra's algorithm implementation
print("[+] Dijkstra's algorithm running!")

# Initialize constants and data structures
infinity = float("inf")
processed = []
graph = {
    "a": {"fin": 1},
    "b": {"a": 3, "fin": 5},
    "fin": {}
}
costs = {
    "a": 6,
    "b": 2,
    "fin": infinity
}
parents = {
    "a": "start",
    "b": "start",
    "fin": None
}

# Function to find the node with the lowest cost
def find_lowest_cost_node(costs, processed):
    lowest_cost = infinity
    lowest_cost_node = None
    for node, cost in costs.items():
        if cost < lowest_cost and node not in processed:
            lowest_cost = cost
            lowest_cost_node = node
    return lowest_cost_node

# Main loop of Dijkstra's algorithm
node = find_lowest_cost_node(costs, processed)
while node is not None:
    # Update costs and parents for neighbors of the current node
    cost = costs[node]
    neighbours = graph[node]
    for n, edge_cost in neighbours.items():
        new_cost = cost + edge_cost
        if costs[n] > new_cost:
            costs[n] = new_cost
            parents[n] = node
    # Mark the current node as processed
    processed.append(node)
    # Find the next node with the lowest cost
    node = find_lowest_cost_node(costs, processed)

# Output the results
print("Costs:", costs)
print("Parents:", parents)
print("Processed:", processed)
