def fizzbuzz_program():
    # Prompt the user to enter the value of 'x'
    value = input("This program will evaluate numbers from 1 to x. Please enter the value of x (numerical value only):\n")
    print("\n")

    try:
        # Convert the input value to an integer and add 1 to include the upper limit
        max_value = int(value) + 1

        # Iterate through numbers from 1 to 'max_value'
        for num in range(1, max_value):
            # Check if the number is divisible by both 3 and 5
            if num % 3 == 0 and num % 5 == 0:
                print("FizzBuzz!")
            # Check if the number is divisible by 3
            elif num % 3 == 0:
                print("Fizz")
            # Check if the number is divisible by 5
            elif num % 5 == 0:
                print("Buzz")
            # Print the number if it doesn't meet any of the above conditions
            else:
                print(num)
    except ValueError:
        # Handle the case where the user enters a non-numeric value
        print("Error: Please enter a valid numerical value.")

if __name__ == "__main__":
    # Call the FizzBuzz program function when the script is executed
    fizzbuzz_program()
