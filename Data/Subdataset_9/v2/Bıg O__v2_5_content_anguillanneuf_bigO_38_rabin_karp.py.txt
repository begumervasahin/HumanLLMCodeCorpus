def rabin_karp_search(text, substring):
    found_indices = []  # Initialize a list to store indices where substring is found
    text_length = len(text)  # Length of the text
    substring_length = len(substring)  # Length of the substring
    prime = 101  # Prime number used for hashing
    d = 128  # Number of characters in the ASCII character set
    h = pow(d, substring_length - 1) % prime  # Precompute h

    # Define a helper function to compute the hash value of a string
    def hash_function(s):
        hash_val = 0
        for char in s:
            hash_val = (hash_val * d + ord(char)) % prime
        return hash_val

    # Calculate the hash value of the substring and the initial substring of the text
    substring_hash = hash_function(substring)
    text_hash = hash_function(text[:substring_length])

    # Iterate through the text to find matches using Rabin-Karp algorithm
    for i in range(text_length - substring_length + 1):
        # Check if the hash values of the substring and the current text window match
        if substring_hash == text_hash:
            # If the hash values match, compare the actual substrings to confirm the match
            if text[i:i+substring_length] == substring:
                found_indices.append(i)  # Store the index where the match is found
                print("Substring found at index:", i)  # Print the index where substring is found
        # Update the text hash for the next window
        if i < text_length - substring_length:
            text_hash = (d * (text_hash - h * ord(text[i])) + ord(text[i + substring_length])) % prime
            if text_hash < 0:
                text_hash += prime  # Ensure the hash value is non-negative

    return found_indices  # Return the list of indices where substring is found


# Example usage
text = 'Get bit; set bit; clear bit; update bit'
substring = 'bit'
print("Indices of substring found using Rabin-Karp algorithm:", rabin_karp_search(text, substring))
