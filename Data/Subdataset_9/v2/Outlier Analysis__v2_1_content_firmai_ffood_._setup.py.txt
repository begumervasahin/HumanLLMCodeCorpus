import os
import xattr
from datetime import datetime

# Function to retrieve the last used date of a file
def get_last_used_date(file_path):
    try:
        # Attempt to retrieve the last used date attribute
        last_used_date = xattr.getxattr(file_path, 'com.apple.lastuseddate')
        
        # Convert the retrieved value to a datetime object
        last_used_date = datetime.fromtimestamp(int.from_bytes(last_used_date, byteorder='big'))
        
        return last_used_date  # Return the last used date if successful
    except OSError:
        return None  # Return None if file doesn't exist or attribute not found

# Main block
if __name__ == "__main__":
    file_path = "/path/to/your/file"  # Specify the path to the file
    last_used_date = get_last_used_date(file_path)  # Retrieve the last used date
    if last_used_date:
        # If last used date is retrieved, print it
        print(f"Last used date of {file_path}: {last_used_date}")
    else:
        # If unable to retrieve last used date, print a message
        print(f"Unable to retrieve last used date for {file_path}")
