class Node:
    def __init__(self, value):
        # Initialize a node with a value and a reference to the next node
        self.value = value
        self.next = None

    def getValue(self):
        # Retrieve the value of the node
        return self.value

    def getNext(self):
        # Retrieve the reference to the next node
        return self.next

    def setValue(self, new_value):
        # Set a new value for the node
        self.value = new_value

    def setNext(self, new_next):
        # Set a new reference to the next node
        self.next = new_next

    def __str__(self):
        # Return the string representation of the node's value
        return str(self.value)

    __repr__ = __str__


class OrderedLinkedList:
    def __init__(self):
        # Initialize an ordered linked list with head, tail, and count
        self.head = None
        self.tail = None
        self.count = 0

    def add(self, value):
        # Add a value to the ordered linked list
        new_node = Node(value)
        if self.head is None:
            # If the list is empty, set the new node as both head and tail
            self.head = new_node
            self.tail = new_node
        elif self.head.value > new_node.value:
            # If the new node's value is smaller than the head's value,
            # set the new node as the new head
            new_node.next = self.head
            self.head = new_node
        elif self.tail.value < new_node.value:
            # If the new node's value is larger than the tail's value,
            # set the new node as the new tail
            self.tail.next = new_node
            self.tail = new_node
        else:
            # Otherwise, find the appropriate position to insert the new node
            temp = self.head
            while temp.value < new_node.value:
                prev = temp
                temp = temp.next
            prev.next = new_node
            new_node.next = temp

    def delete(self, value):
        # Delete a value from the ordered linked list
        if self.head.value == value:
            # If the value to delete is the head's value, set the next node as the new head
            self.head = self.head.next
        elif self.tail.value == value:
            # If the value to delete is the tail's value, find the new tail
            temp = self.head
            while temp.next != self.tail:
                temp = temp.next
            self.tail = temp
            self.tail.next = None
        else:
            # Otherwise, find the value to delete and adjust the references
            temp = self.head
            while temp.value != value:
                prev = temp
                temp = temp.next
            prev.next = temp.next
            temp.next = None

    def search(self, value):
        # Search for a value in the ordered linked list
        temp = self.head
        while temp is not None:
            if temp.value == value:
                return True
            temp = temp.next
        return False

    def pop(self):
        # Pop the last value from the ordered linked list
        if self.head is None:
            return None
        val = self.tail.value
        temp = self.head
        while temp.next != self.tail:
            temp = temp.next
        self.tail = temp
        self.tail.next = None
        return val

    def isEmpty(self):
        # Check if the ordered linked list is empty
        return self.head is None

    def size(self):
        # Get the size of the ordered linked list
        count = 0
        temp = self.head
        while temp is not None:
            count += 1
            temp = temp.next
        return count

    def printList(self):
        # Print the elements of the ordered linked list
        temp = self.head
        while temp:
            print(temp.getValue(), end=' ')
            temp = temp.getNext()
