import nltk
import json

def main():
    # Path to the JSON file containing Google+ posts data
    file_path = '[change to your googleplus_posts.json location]'  
    
    # Load data from the JSON file
    with open(file_path, 'r') as file:
        data = json.load(file)
    
    # Define query terms to search for
    QUERY_TERMS = ['mobile']
    
    # Extract content of each activity and convert to lowercase
    activities = [activity['object']['content'].lower().split() for activity in data if activity['object']['content'] != '']
    
    # Create a TextCollection object for calculating TF-IDF scores
    tc = nltk.TextCollection(activities)
    
    # List to store relevant activities along with their scores, titles, and URLs
    relevant_activities = []
    
    # Calculate TF-IDF scores for each activity
    for i in range(len(activities)):
        score = 0
        for term in QUERY_TERMS:
            score += tc.tf_idf(term.lower(), activities[i])
        
        # If the score is greater than 0, the activity is relevant, so add it to the list
        if score > 0:
            relevant_activities.append({
                'score': score, 
                'title': data[i]['title'], 
                'url': data[i]['url']
            })
    
    # Sort relevant activities based on their scores in descending order
    relevant_activities = sorted(relevant_activities, key=lambda a: a['score'], reverse=True)
    
    # Print out relevant activities with their scores, titles, and URLs
    for ra in relevant_activities:
        print('Title:', ra['title'])
        print('URL:', ra['url'])
        print('Score:', ra['score'])
        print()  # Add an empty line for better readability

if __name__ == '__main__':
    main()
