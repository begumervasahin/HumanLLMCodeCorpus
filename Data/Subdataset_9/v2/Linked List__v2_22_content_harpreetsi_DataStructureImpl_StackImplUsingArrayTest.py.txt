class Stack:
    def __init__(self, capacity):
        self.capacity = capacity
        self.stack = []

    def push(self, item):
        """Pushes an item onto the stack."""
        if len(self.stack) < self.capacity:
            self.stack.append(item)
            return "success"
        else:
            return "Stack full"

    def pop(self):
        """Pops an item from the stack."""
        if self.is_empty():
            return "Stack empty"
        else:
            return self.stack.pop()

    def top_element(self):
        """Returns the top element of the stack."""
        if self.is_empty():
            return "Stack empty"
        else:
            return self.stack[-1]

    def is_empty(self):
        """Checks if the stack is empty."""
        return len(self.stack) == 0

    def print_stack(self):
        """Prints the elements of the stack."""
        if self.is_empty():
            print("Stack empty")
        else:
            print("Stack:", self.stack)


import unittest

class TestStackImplUsingArray(unittest.TestCase):
    def test_push(self):
        """Test push operation."""
        my_stack = Stack(10)
        self.assertEqual(my_stack.push(5), "success")

    def test_stack_overflow(self):
        """Test stack overflow."""
        my_stack = Stack(3)
        my_stack.push(5)
        my_stack.push(5)
        my_stack.push(5)
        self.assertEqual(my_stack.push(5), "Stack full")

    def test_top_element(self):
        """Test top element."""
        my_stack = Stack(10)
        my_stack.push(10)
        self.assertEqual(my_stack.top_element(), 10)

    def test_top_element_stack_empty(self):
        """Test top element when stack is empty."""
        my_stack = Stack(10)
        self.assertEqual(my_stack.top_element(), "Stack empty")

    def test_pop(self):
        """Test pop operation."""
        my_stack = Stack(10)
        my_stack.push(10)
        self.assertEqual(my_stack.pop(), 10)

    def test_stack_underflow(self):
        """Test stack underflow."""
        my_stack = Stack(10)
        self.assertEqual(my_stack.pop(), "Stack empty")

    def test_is_empty(self):
        """Test is_empty method."""
        my_stack = Stack(10)
        self.assertTrue(my_stack.is_empty())

    def test_is_not_empty(self):
        """Test is_empty method when stack is not empty."""
        my_stack = Stack(10)
        my_stack.push(10)
        self.assertFalse(my_stack.is_empty())

    def test_print_empty_stack(self):
        """Test printing an empty stack."""
        my_stack = Stack(10)
        my_stack.print_stack()

    def test_print(self):
        """Test printing a non-empty stack."""
        my_stack = Stack(10)
        my_stack.push(10)
        my_stack.push(20)
        my_stack.print_stack()

if __name__ == '__main__':
    unittest.main()
