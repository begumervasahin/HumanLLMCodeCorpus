def longest_common_subsequence(str1, str2):
    """
    Calculate the length of the longest common subsequence between two strings.

    :param str1: First string
    :param str2: Second string
    :return: Length of the longest common subsequence
    """
    # Get the lengths of both strings
    len_str1 = len(str1)
    len_str2 = len(str2)

    # Initialize a 2D list (matrix) to store the lengths of longest common subsequences
    # for substrings of str1 and str2. The size of the matrix is (len_str1+1) x (len_str2+1)
    # because we also consider empty prefixes of the strings.
    dp_matrix = [[0] * (len_str2 + 1) for _ in range(len_str1 + 1)]

    # Fill dp_matrix
    for i in range(1, len_str1 + 1):
        for j in range(1, len_str2 + 1):
            if str1[i - 1] == str2[j - 1]:
                # If the current characters in both strings match,
                # the length of the LCS is the length found by removing the current character
                # in both strings plus one.
                dp_matrix[i][j] = dp_matrix[i - 1][j - 1] + 1
            else:
                # If the current characters in both strings do not match,
                # the length of the LCS is the maximum length found by removing the current character
                # from one of the strings.
                dp_matrix[i][j] = max(dp_matrix[i - 1][j], dp_matrix[i][j - 1])

    # The bottom-right corner of the matrix contains the length of the LCS of the entire strings.
    return dp_matrix[len_str1][len_str2]

# Ask the user for input
first_string = input("First string: ")
second_string = input("Second string: ")

# Calculate and print the length of the LCS
print("The length of the LCS is:", longest_common_subsequence(first_string, second_string))
