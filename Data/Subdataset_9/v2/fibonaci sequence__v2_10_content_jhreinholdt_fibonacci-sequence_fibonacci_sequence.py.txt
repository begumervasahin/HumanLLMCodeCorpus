# Function to calculate the Fibonacci number at a given position
def fibonacci_number(position):
    # Base cases: Fibonacci numbers at positions 0 and 1
    if position == 0:
        return 0
    elif position == 1:
        return 1
    else:
        # Recursive case: Sum of two previous Fibonacci numbers
        return fibonacci_number(position - 2) + fibonacci_number(position - 1)

# Function to generate and print the Fibonacci sequence up to a given number of elements
def fibonacci_sequence(num_elements):
    # Check if the input is valid
    if num_elements <= 0:
        print("Please enter a positive integer")
    else:
        print("The Fibonacci Sequence with", num_elements, "elements is:")
        # Generate and print each Fibonacci number up to the given number of elements
        for position in range(num_elements):
            print(fibonacci_number(position), "", end="")

# Main function to interact with the user and control the program flow
def main():
    while True:
        try:
            # Prompt the user to enter the number of elements for the Fibonacci sequence
            num_elements = int(input('\nEnter a positive number to generate the Fibonacci Sequence with that amount of elements: '))
            # Generate and print the Fibonacci sequence
            fibonacci_sequence(num_elements)
            break  # Exit the loop if a valid input is provided
        except ValueError:
            print("Please enter a valid positive integer.")

# Execute the main function when the script is run directly
if __name__ == '__main__':
    main()
