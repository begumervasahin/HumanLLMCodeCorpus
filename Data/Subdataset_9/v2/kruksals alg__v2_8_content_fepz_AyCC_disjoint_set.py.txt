class DisjointSets:
    def __init__(self, elements):
        """
        Initializes the DisjointSets data structure with given elements.

        Parameters:
        elements (list): List of elements to initialize the sets with.
        """
        self.elements = list(elements)
        self.set_labels = list(range(len(elements)))
        self.ranks = [1] * len(elements)

    def find(self, x):
        """
        Finds the label of the set containing object x.

        Parameters:
        x: The element whose set label needs to be found.

        Returns:
        int: The label of the set containing x.
        """
        i = r = self.elements.index(x)
        while self.set_labels[r] != r:
            r = self.set_labels[r]
        while i != r:
            j = self.set_labels[i]
            self.set_labels[i] = r
            i = j
        return r

    def merge(self, a, b):
        """
        Merges the sets labeled a and b.

        Parameters:
        a: Label of the first set to merge.
        b: Label of the second set to merge.
        """
        if self.ranks[a] == self.ranks[b]:
            self.ranks[a] += 1
            self.set_labels[b] = a
        else:
            if self.ranks[a] > self.ranks[b]:
                self.set_labels[b] = a
            else:
                self.set_labels[a] = b

# Example usage
elements = ['A', 'B', 'C', 'D', 'E']
disjoint_sets = DisjointSets(elements)

print("Initial sets:", disjoint_sets.set_labels)
print("Initial ranks:", disjoint_sets.ranks)

# Merge sets containing 'A' and 'B'
disjoint_sets.merge(disjoint_sets.find('A'), disjoint_sets.find('B'))

print("\nAfter merging sets containing 'A' and 'B':")
print("Sets:", disjoint_sets.set_labels)
print("Ranks:", disjoint_sets.ranks)

# Merge sets containing 'C' and 'D'
disjoint_sets.merge(disjoint_sets.find('C'), disjoint_sets.find('D'))

print("\nAfter merging sets containing 'C' and 'D':")
print("Sets:", disjoint_sets.set_labels)
print("Ranks:", disjoint_sets.ranks)
