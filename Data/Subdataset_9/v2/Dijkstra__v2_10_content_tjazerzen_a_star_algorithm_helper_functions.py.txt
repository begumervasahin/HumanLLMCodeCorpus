from math import sqrt

# Function to calculate the Euclidean distance between two points
def calculate_distance(x1, y1, x2, y2):
    # Squared differences in x and y coordinates
    d1 = (x1 - x2)**2
    d2 = (y1 - y2)**2
    # Square root of the sum of squared differences
    distance = sqrt(d1 + d2)
    return distance

# Function to calculate heuristic values for each point based on the end point
def heuristic_value(end_point, coordinates):
    # Extract x and y coordinates of the end point
    x1 = coordinates[end_point][0]
    y1 = coordinates[end_point][1]
    lst = []  # List to store heuristic values
    # Iterate over all points
    for i in range(len(coordinates)):
        # Extract x and y coordinates of the current point
        x2 = coordinates[i][0]
        y2 = coordinates[i][1]
        # Calculate Euclidean distance between the end point and current point
        distance = calculate_distance(x1, y1, x2, y2)
        lst.append(distance)  # Append the distance to the list
    return lst

# Function to calculate distances between connected points
def calculate_connections(coordinates, connection_list):
    lst = [[0]*len(connection_list[i]) for i in range(len(connection_list))]  # Initialize list for distances
    # Iterate over all points
    for i in range(len(connection_list)):
        # Iterate over connections of each point
        for j in range(len(connection_list[i])):
            # Extract the indices of connected points
            point1 = i
            point2 = connection_list[i][j]
            # Extract coordinates of connected points
            x1 = coordinates[point1][0]
            y1 = coordinates[point1][1]
            x2 = coordinates[point2][0]
            y2 = coordinates[point2][1]
            # Calculate Euclidean distance between connected points
            distance = calculate_distance(x1, y1, x2, y2)
            # Store the distance in the list
            lst[i][j] = distance
    return lst

# Example usage:
coordinates = [(0, 0), (3, 4), (6, 8)]
connection_list = [[1], [0, 2], [1]]
end_point = 0

# Print heuristic values and connection distances
print("Heuristic values:", heuristic_value(end_point, coordinates))
print("Connection distances:", calculate_connections(coordinates, connection_list))
