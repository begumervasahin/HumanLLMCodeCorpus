import random

# Function to perform linear search
def linear_search(data, target):
    for number in data:
        if number == target:
            return True  # Target value found in the list
    return False  # Target value not found in the list

# Sample data and a random target value
data = [1, 2, 3, 4, 5, 6]
target = random.randrange(1, 10)

# Displaying the list and target value
print("List:", data)
print("Target:", target)

# Perform linear search and check if the target value is present
result = linear_search(data, target)

# Display the result
if result:
    print("The target value is present in the list.")
else:
    print("The target value is not present in the list.")
