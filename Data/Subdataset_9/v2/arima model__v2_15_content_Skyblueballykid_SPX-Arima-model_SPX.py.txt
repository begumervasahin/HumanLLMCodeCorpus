# Import necessary libraries
import numpy as np
import pandas as pd
from statsmodels.tsa.arima.model import ARIMA

# Load the S&P 500 (SPX) weekly dataset
SPX = pd.read_csv("SPX Weekly 2000-2017.csv")

# Display the first 5 rows of the dataset
print("First 5 rows of the SPX dataset:")
print(SPX.head(n=5))

# Fit an ARIMA model with a differencing order of 1
order = (0, 1, 0)  # ARIMA order: (p, d, q)
model = ARIMA(SPX['Close'], order=order)
model_fit = model.fit()

# Forecast the next value using the fitted model
forecasted_value = model_fit.forecast()[0]
print("Forecasted value:", forecasted_value)

# Display the model summary (optional)
print("\nARIMA Model Summary:")
print(model_fit.summary())
