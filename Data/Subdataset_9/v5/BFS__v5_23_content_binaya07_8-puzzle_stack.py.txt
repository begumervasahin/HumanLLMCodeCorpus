class Stack:
    """
    Stack data structure implementation.
    """

    def __init__(self):
        """
        Initializes an empty stack.
        """
        self.stack = []  # List to store stack elements
        self.unique_states = set()  # Set to keep track of unique states

    def push(self, item):
        """
        Pushes an item onto the top of the stack.
        """
        self.stack.append(item)  # Add item to the stack
        self.unique_states.add(tuple(item.stack))  # Add state to the set

    def pop(self):
        """
        Removes and returns the item at the top of the stack in LIFO order.
        """
        if not self.is_empty():
            top_of_stack = self.stack.pop()  # Get and remove the top item from the stack
            self.unique_states.remove(tuple(top_of_stack.stack))  # Remove state from set
            return top_of_stack
        else:
            return None

    def is_empty(self):
        """
        Checks if the stack is empty.
        """
        return len(self.stack) == 0

    def clear_stack(self):
        """
        Clears the stack.
        """
        self.stack.clear()  # Clear the list
        self.unique_states.clear()  # Clear the set
