class Node:
    def __init__(self, data):
        self.data = data
        self.left = None
        self.right = None

def k_sum_paths(root, k):
    stack = []
    path_stack = []
    temp = root

    while True:
        # Traverse left subtree
        while temp:
            stack.append(temp)
            if len(path_stack) == 0:
                path_stack.append(temp)
                top_sum = 0
            else:
                top_sum = path_stack.pop()
                path_stack.append(temp)
            path_stack.append(temp.data + top_sum)
            temp = temp.left

        # Traverse right subtree or backtrack
        while temp == None and len(stack):
            temp = stack.pop()
            top_sum = path_stack.pop()
            # Backtrack to find sum of path
            while path_stack[-1] != temp:
                top_sum -= path_stack.pop().data
            temp_sum = top_sum
            i = 0
            # Check for k-sum paths
            while i < len(path_stack):
                if temp_sum == k:
                    print_path(path_stack, i)
                temp_sum -= path_stack[i].data
                i += 1
            path_stack.append(top_sum)
            temp = temp.right

        # Break if traversal is complete
        if temp == None and len(stack) == 0:
            break

def print_path(path_stack, start_index):
    """Print the k-sum path."""
    for j in range(start_index, len(path_stack)):
        print(path_stack[j].data, end=' ')
    print()

root = Node(1)

root.right.left.left = Node(1)
root.right.left.right = Node(2)
root.right.right = Node(5)
root.right.right.right = Node(6)

k_sum_paths(root, 5)
