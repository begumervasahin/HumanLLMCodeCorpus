import networkx as nx

# Solve the problem
def solve(client):
    # End the current communication session and start a new one
    client.end()
    client.start()
    
    # Get the graph from the client
    graph = client.G
    
    # Find the minimum spanning tree of the graph
    mst = nx.minimum_spanning_tree(graph)
    
    # Find the minimum spanning tree remotely
    remote_spanning_tree(mst, client)
    
    # Print the number of bots needed to be rescued
    print("Number of bots needed to be rescued:")
    print(client.l)
    
    # Print the number of final rescued bots
    print("Number of final rescued bots:")
    print(client.bot_count[client.home])
    
    # End the communication session
    client.end()

# Find the minimum spanning tree remotely
def remote_spanning_tree(mst, client):
    # Get the degrees of the nodes in the minimum spanning tree   
    degrees = list(mst.degree)
    
    # Loop until there's only one node left in the minimum spanning tree
    while len(degrees) > 1:
        # Find the next node with degree 1 that is not the home node
        u = next((node for node, degree in degrees if degree == 1 and node != client.h), None)
        
        if u is None:
            break
        
        # Get the neighbor of the current node
        v = next(iter(mst[u]))
        
        # Remote the current node and its edge
        client.remote(u, v)
        
        # Remove the current node from the minimum spanning tree
        mst.remove_node(u)
        
        # Update the degrees list
        degrees = list(mst.degree)
