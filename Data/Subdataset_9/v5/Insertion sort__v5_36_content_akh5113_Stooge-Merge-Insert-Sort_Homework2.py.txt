def read_arrays_from_file(file_name):
    """
    Read arrays from a file.

    Args:
    file_name (str): The name of the file to read.

    Returns:
    list: A list of arrays read from the file.
    """
    arrays = []
    with open(file_name, "r") as file:
        for line in file:
            # Split each line by spaces and convert elements to integers
            values = list(map(int, line.strip().split()[1:]))
            arrays.append(values)
    return arrays

def stooge_sort(array):
    """
    Sort an array using Stooge Sort algorithm.

    Args:
    array (list): The array to be sorted.
    """
    def recursive_sort(arr, start, end):
        num_elms = end - start + 1
        if num_elms == 2:
            # Swap elements if necessary
            if arr[start] > arr[end]:
                arr[start], arr[end] = arr[end], arr[start]
        elif num_elms > 2:
            # Divide the array into three parts
            third = num_elms // 3
            # Recursively sort the first 2/3 elements
            recursive_sort(arr, start, end - third)
            # Recursively sort the last 2/3 elements
            recursive_sort(arr, start + third, end)
            # Recursively sort the first 2/3 elements again
            recursive_sort(arr, start, end - third)

    recursive_sort(array, 0, len(array) - 1)

def write_arrays_to_file(arrays, file_name):
    """
    Write arrays to a file.

    Args:
    arrays (list): A list of arrays to write to the file.
    file_name (str): The name of the file to write.
    """
    with open(file_name, "w") as file:
        for array in arrays:
            # Write each array to a line in the file
            file.write(" ".join(map(str, array)) + "\n")

def main():
    file_name = "data.txt"

    # Read arrays from file
    arrays = read_arrays_from_file(file_name)

    print("Stooge Sort")
    
    # Print unsorted arrays
    print("The unsorted arrays are:")
    for array in arrays:
        print(array)

    # Sort arrays using Stooge Sort
    for array in arrays:
        stooge_sort(array)

    # Print sorted arrays
    print("The sorted arrays are:")
    for array in arrays:
        print(array)

    # Write sorted arrays to file
    write_arrays_to_file(arrays, "stooge.out")

if __name__ == "__main__":
    main()
