import pandas as pd
import sys
from multiprocessing import Pool
from readyscrape import RequestsBS4, DataProcessing

class Scraper:
    def __init__(self, site="https://www.macys.com"):
        self.site = site

    def get_url_categories(self):
        """Extracts category URLs from Macy's homepage."""
        request = RequestsBS4(self.site)
        soup = request.get_url()
        page = request.page
        print(page.status_code)
        print("\nGETTING CATEGORIES URLs...")
        categories = set()
        for tag in soup.find_all("a", href=True):
            path = tag["href"]
            if "http" not in path and "COL" in path and "/shop/" in path:
                categories.add(self.site + path)
        return list(categories)

    def get_url_products_test(self):
        """Gets product URLs for testing purposes."""
        category_urls = self.get_url_categories()
        test_category_url = category_urls[-1]
        soup = RequestsBS4(test_category_url).get_url()
        product_urls = [self.site + tag.get("href") for tag in soup.find_all("a", {"class": "productDescLink"})]
        print('\n{} URLs are now fetched from {}'.format(len(product_urls), test_category_url))
        add_data = {'url': product_urls}
        colnames = ['url']
        fname = "product-url.csv"
        DataProcessing(add_data, colnames, fname).add_to_csv()
        return fname

    def get_url_products(self, url):
        """Gets product URLs from a given category URL."""
        soup = RequestsBS4(url).get_url()
        product_urls = [self.site + tag.get("href") for tag in soup.find_all("a", {"class": "productDescLink"})]
        print('\n{} URLs are now fetched from {}'.format(len(product_urls), url))
        add_data = {'url': product_urls}
        colnames = ['url']
        fname = "product-url.csv"
        DataProcessing(add_data, colnames, fname).add_to_csv()
        return fname

    def scrape_and_save(self, url):
        """Scrapes product information from a given URL and saves it to a CSV file."""
        request = RequestsBS4(url)
        soup = request.get_url()
        page = request.page
        if page.status_code == 200:
            print('\nProcessing... {}'.format(url))
        try:
            name = soup.find_all("h1", {"class": "p-name h3"})[0].text.strip().upper()
            price = soup.find_all("div", {"class": "price"})[0].text.strip()
            description = soup.find_all("p", {"data-auto": "product-description"})[0].text.strip()
            if name and price and description:
                DataProcessing({'name': [name], 'price': [price], 'description': [description]},
                                ['name', 'price', 'description'], "macys-products.csv").add_to_csv()
        except IndexError:
            pass

    def get_product_info(self):
        """Searches for product information by name in the CSV file."""
        while True:
            product_name = input("\nEnter product name\nOr type 'q' to quit the program: ").upper()
            if product_name == 'Q':
                break
            df = pd.read_csv('macys-products.csv')
            matching_products = df[df['name'].str.upper() == product_name]
            if not matching_products.empty:
                print(matching_products.to_string(index=False))
                print('\nNext search...')
            else:
                print('\nNo product with such name. Please try again!')


if __name__ == "__main__":
    scraper = Scraper()
    while True:
        scrape_choice = input("\nEnter 't' to scrape a small sample,\n'f' to scrape full web,\n"
                              "'s' to search for product information,\n"
                              "or press 'q' to quit the program: ").upper()
        if scrape_choice == 'Q':
            sys.exit()
        elif scrape_choice in ['T', 'F', 'S']:
            break

    if scrape_choice == 'T':
        url = scraper.get_url_products_test()
    elif scrape_choice == 'F':
        category_urls = scraper.get_url_categories()
        with Pool(processes=4) as p:
            product_url_files = p.map(scraper.get_url_products, category_urls)
        product_urls = pd.concat([pd.read_csv(file) for file in product_url_files], ignore_index=True)
        url = product_urls["url"].values.tolist()
    else:
        scraper.get_product_info()

    if scrape_choice in ['T', 'F']:
        with Pool(processes=2) as p:
            p.map(scraper.scrape_and_save, url)
