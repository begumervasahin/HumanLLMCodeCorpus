import os
import string
import numpy as np
import pandas as pd
from time import time
from nltk.corpus import stopwords
from nltk.stem.wordnet import WordNetLemmatizer

def read_words_list(file_path):
    df = pd.read_csv(file_path, header=0)
    return df['word']

def preprocess_word(word, lemmatizer):
    word = lemmatizer.lemmatize(word.lower())
    return word

def is_valid_word(word):
    return word not in stopwords.words('english') and word not in string.punctuation and len(word) > 2 and not word.isdigit()

def process_file(file_path, words, lemmatizer):
    words_list_array = np.zeros(words.size)

    with open(file_path, "r", encoding='utf-8', errors='ignore') as file_reading:
        for word in file_reading.read().split():
            word = preprocess_word(word, lemmatizer)

            if is_valid_word(word):
                for i, target_word in enumerate(words):
                    if target_word == word:
                        words_list_array[i] += 1
                        break

    return words_list_array

def write_output_row(output_file, words_list_array, label):
    output_file.write(','.join(map(str, words_list_array.astype(int))))
    output_file.write(f",{label}\n")

def main():
    words = read_words_list('wordslist.csv')
    lemmatizer = WordNetLemmatizer()

    input_directory = "emails/"
    output_file_path = "frequency.csv"

    with open(output_file_path, "w+") as output_file:
        output_file.write(','.join(map(str, words)) + ',output\n')

    start_time = time()
    file_count = 0

    for file_name in os.listdir(input_directory):
        file_name = file_name.decode("utf-8")
        full_file_path = os.path.join(os.getcwd(), input_directory, file_name)
        
        words_list_array = process_file(full_file_path, words, lemmatizer)

        with open(output_file_path, "a") as output_file:
            label = -1 if len(file_name) == 68 else 1
            write_output_row(output_file, words_list_array, label)

        file_count += 1

        if file_count % 100 == 0:
            print("Processed {} files".format(file_count))

    print("Time taken to process {} files: {:.2f} seconds".format(file_count, time() - start_time))

if __name__ == "__main__":
    main()
