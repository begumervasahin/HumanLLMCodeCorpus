import random

def get_odd_number(a, b):
    """Generate a random odd number between a and b."""
    rand = random.randrange(a, b)
    if rand % 2 == 0:
        rand += 1
    return rand

def is_composite(n, a, r, s):
    """Check if a number is composite."""
    res = pow(a, r, n)
    if res == 1 or res == n - 1:
        return False
    for _ in range(int(s)):
        res = pow(res, 2, n)
        if res == n - 1:
            return False
    return True

def is_prime(n):
    """Check if a number is prime."""
    r = n - 1
    s = 0
    while r % 2 == 0:
        r //= 2
        s += 1
    limit = (n - 1) // 4  # corrected division to floor division
    for _ in range(limit):
        a = random.randint(2, 100)
        if is_composite(n, a, r, s):
            return False
    return True

def generate_prime_number(a, b):
    """Generate a prime number between a and b."""
    while True:
        number = get_odd_number(a, b)
        if is_prime(number):
            return number

def gcd(a, b):
    """Calculate the greatest common divisor (GCD)."""
    while b != 0:
        a, b = b, a % b
    return a

def multiplicative_inverse(e, phi):
    """Calculate the multiplicative inverse."""
    x, y = 1, 0
    temp_phi = phi
    while e > 1:
        quotient = e // temp_phi
        e, temp_phi = temp_phi, e % temp_phi
        x, y = y, x - quotient * y
    return x % phi
