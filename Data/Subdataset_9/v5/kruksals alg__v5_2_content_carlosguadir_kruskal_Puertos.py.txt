def port_list():
    """
    Creates a list of all ports from the provided country port lists.
    """
    ports = ['-Empty-']
    for port_list in COUNTRIES.values():
        ports.extend(port_list)
    return ports

def country_for_port(port):
    """
    Returns the country name for a given port.
    """
    for country, ports in COUNTRIES.items():
        if port in ports:
            return country
