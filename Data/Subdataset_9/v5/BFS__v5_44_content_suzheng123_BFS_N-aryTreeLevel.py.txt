from collections import deque

class Node:
    def __init__(self, val, children):
        self.val = val  # Value of the node
        self.children = children  # List of children nodes


class Solution:
    def levelOrder(self, root):
        """
        Return the level order traversal of an N-ary tree.

        Args:
        root (Node): The root of the N-ary tree.

        Returns:
        List[List[int]]: The level order traversal.
        """
        # Check if the root is None
        if not root:
            return []

        # Initialize a queue to store nodes
        queue = deque([root])
        # Initialize a list to store the tree in level order
        tree = []

        # Loop until the queue is empty
        while queue:
            level = []  # List to store nodes at the current level
            # Traverse through nodes in the current level
            for _ in range(len(queue)):
                node = queue.popleft()
                level.append(node.val)  # Append value of the node to the current level list
                # Traverse through children of the current node
                for child in node.children:
                    queue.append(child)  # Add child to the queue
            tree.append(level)  # Add the current level list to the tree

        return tree  # Return the tree in level order
