# Importing necessary function from a helper module
from data_helper import store_data

class Dictionary:
    # Constants for dictionary indices
    DOCUMENT_FREQUENCY = 0
    IDF = 0
    TERM_ID = 1
    TERM_OFFSET = 1

    def __init__(self, file_path):
        """Initialize the Dictionary."""
        self.terms = {}  # Dictionary to store terms
        self.file_path = file_path  # File to store data
        self.total_num_documents = 0  # Total number of documents

    def has_term(self, term):
        """Check if the term exists in the dictionary."""
        return term in self.terms

    def get_term_id(self, term):
        """Get the term ID."""
        return self.terms[term][Dictionary.TERM_ID]

    def add_term(self, term, term_id):
        """Add a new term to the dictionary."""
        # Initialize term with document frequency and term ID
        self.terms[term] = [1, term_id]

    def get_terms(self):
        """Retrieve all terms in the dictionary."""
        return self.terms

    def add_document_frequency(self, term):
        """Increment the document frequency of a term."""
        self.terms[term][Dictionary.DOCUMENT_FREQUENCY] += 1

    def set_idf(self, term, idf):
        """Set the IDF (Inverse Document Frequency) of a term."""
        self.terms[term][Dictionary.IDF] = idf

    def set_offset(self, term, offset):
        """Set the offset of a term."""
        self.terms[term][Dictionary.TERM_OFFSET] = offset

    def save_to_disk(self):
        """Save the dictionary to disk using a helper function."""
        store_data(self.file_path, self.terms)
