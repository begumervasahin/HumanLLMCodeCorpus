import random
from collections import defaultdict

DEAD_CELL, LIVE_CELL = '-', '*'
MAX_GENERATIONS = 3
CELL_COUNT = (3, 3)
CELL_TABLE = defaultdict(int, {
    (1, 2): 1,
    (1, 3): 1,
    (0, 3): 1,
})

def print_universe(generation, universe):
    """
    Prints the universe for the given generation.
    """
    print("\nGeneration %3i:" % generation)
    for row in range(CELL_COUNT[1]):
        print("  ", ''.join(str(universe[(row, col)])
                            for col in range(CELL_COUNT[0])).replace(
            '0', DEAD_CELL).replace('1', LIVE_CELL)
        )

def update_universe(universe):
    """
    Updates the universe based on the rules of the Game of Life.
    """
    next_generation = defaultdict(int)
    for row in range(CELL_COUNT[1]):
        for col in range(CELL_COUNT[0]):
            sum_neighbors = sum(universe[(r, c)]
                                for r in range(row - 1, row + 2)
                                for c in range(col - 1, col + 2)
                                if (r, c) != (row, col))
            next_generation[(row, col)] = CELL_TABLE[
                (universe[(row, col)], sum_neighbors)
            ]
    return next_generation

def initialize_universe():
    """
    Initializes the universe with random live cells.
    """
    universe = defaultdict(int)
    for row in range(CELL_COUNT[1]):
        for col in range(CELL_COUNT[0]):
            universe[(row, col)] = random.randint(0, 1)
    return universe

# Main function
def main():
    universe = initialize_universe()
    for generation in range(MAX_GENERATIONS):
        print_universe(generation, universe)
        universe = update_universe(universe)

if __name__ == "__main__":
    main()
