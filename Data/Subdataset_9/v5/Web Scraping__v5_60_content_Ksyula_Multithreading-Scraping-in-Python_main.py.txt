import re
import requests
from bs4 import BeautifulSoup
from threading import Thread
from proxy_list import get_proxies
from parse_and_write import parse_and_write
from clean_db import db_cleaner

# Target URL
url = 'https://coinmarketcap.com/'

# Fetching the page content
page = requests.get(url)

# Parsing the page content using BeautifulSoup
soup = BeautifulSoup(page.content, "html.parser")

# List to store coin names
coins = []

# Extracting coin names from links
for link in soup.find_all('a'):
    href = link.get('href')
    if href and href != '/' and re.match('^/currencies', href):
        coin = re.search('/currencies/(.+?)/', href)
        if coin:
            coin_name = coin.group(1)
            if coin_name != 'volume' and coin_name not in coins:
                coins.append(coin_name)

# Getting proxies
proxies = get_proxies()

# Splitting coins into batches for processing
batch_size = 25
coin_batches = [coins[i:i+batch_size] for i in range(0, len(coins), batch_size)]

# List to store threads
threads = []

# Creating threads to parse and write data for coin batches
for coin_batch in coin_batches:
    t = Thread(target=parse_and_write, args=(coin_batch, proxies))
    threads.append(t)
    t.start()

# Waiting for all threads to complete
for t in threads:
    t.join()

# Printing message indicating the main thread is exiting
print("Exiting Main Thread")
