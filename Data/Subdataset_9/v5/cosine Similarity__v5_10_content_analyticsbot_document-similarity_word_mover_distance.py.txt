from pyemd import emd
from sklearn.feature_extraction.text import CountVectorizer
from sklearn.metrics import euclidean_distances
from gensim.models.word2vec import Word2Vec
from text_unidecode import unidecode
import os
import numpy as np

# Sample documents
document1 = "Obama speaks to the media in Illinois"
document2 = "The President addresses the press in Chicago"

# File paths
embeddings_file = "data/embed.dat"
vocab_file = "data/embed.vocab"
word2vec_model_path = "data/GoogleNews-vectors-negative300.bin.gz"

# Function to cache word embeddings and vocabulary
def cache_embeddings(word2vec_model, embeddings_file, vocab_file):
    if not os.path.exists(embeddings_file):
        print("Caching word embeddings in memmapped format...")
        word2vec_model.init_sims(replace=True)
        with np.memmap(embeddings_file, dtype=np.double, mode='w+', shape=word2vec_model.syn0norm.shape) as fp:
            fp[:] = word2vec_model.syn0norm[:]
        with open(vocab_file, "w") as f:
            for _, word in sorted((voc.index, word) for word, voc in word2vec_model.vocab.items()):
                print(unidecode(word), file=f)

# Load word embeddings and vocabulary
def load_embeddings(embeddings_file, vocab_file):
    word2vec_model = Word2Vec.load_word2vec_format(word2vec_model_path, binary=True)
    embeddings = np.memmap(embeddings_file, dtype=np.double, mode="r", shape=word2vec_model.syn0.shape)
    with open(vocab_file) as f:
        vocab_list = map(str.strip, f.readlines())
    vocab_dict = {word: idx for idx, word in enumerate(vocab_list)}
    return word2vec_model, embeddings, vocab_dict

# Calculate similarity using Earth Mover's Distance (EMD)
def calculate_similarity(doc1, doc2, embeddings, vocab_dict):
    vectorizer = CountVectorizer(stop_words="english").fit([doc1, doc2])
    word_indices = [vocab_dict[word] for word in vectorizer.get_feature_names()]        
    embeddings_subset = embeddings[word_indices]

    distances = euclidean_distances(embeddings_subset)
    doc1_vector, doc2_vector = vectorizer.transform([doc1, doc2]).toarray()
    doc1_vector = doc1_vector.astype(np.double) / doc1_vector.sum()
    doc2_vector = doc2_vector.astype(np.double) / doc2_vector.sum()

    distances = distances.astype(np.double) / distances.max()

    return emd(doc1_vector, doc2_vector, distances)

# Main execution
cache_embeddings(Word2Vec.load_word2vec_format(word2vec_model_path, binary=True), embeddings_file, vocab_file)
word2vec_model, embeddings, vocab_dict = load_embeddings(embeddings_file, vocab_file)
similarity_score = calculate_similarity(document1, document2, embeddings, vocab_dict)
print("Similarity score between document1 and document2:", similarity_score)
