import sys
import os

def file_read(file_path):
    """
    Read lines from a file, stripping leading and trailing whitespaces.
    """
    print('Reading ' + file_path + '...')
    with open(file_path, 'r') as file:
        lines = file.readlines()
    lines = [line.strip() for line in lines]  # Strip leading and trailing whitespaces
    del lines[0]  # Remove the first line
    del lines[-1]  # Remove the last line
    print(lines[0])
    return lines

def file_write(file_path, data):
    """
    Write lines to a file, joining elements with tabs.
    """
    with open(file_path, "w") as file:
        for line in data:
            joined_line = '\t'.join(line)
            file.write(joined_line + '\n')

def addFilenamePrefix(path, prefix=""):
    """
    Add a prefix to a filename.
    """
    directory, filename = os.path.split(path)
    filename = prefix + filename
    return os.path.join(directory, filename)

def lines_into_traces(lines):
    """
    Convert lines of ASCII text into traces (lists of floats).
    """
    split_lines = [line.split('\t') for line in lines]
    traces = [[] for _ in range(len(split_lines[0]))]
    for line in split_lines:
        for i, value in enumerate(line):
            traces[i].append(float(value))
    return traces

def traces_into_lines(traces):
    """
    Convert traces (lists of floats) into lines of ASCII text.
    """
    lines_of_output = len(traces[0])
    lines = [[] for _ in range(lines_of_output)]
    for point in range(lines_of_output):
        for trace in traces:
            lines[point].append(str(trace[point]))
    return lines
