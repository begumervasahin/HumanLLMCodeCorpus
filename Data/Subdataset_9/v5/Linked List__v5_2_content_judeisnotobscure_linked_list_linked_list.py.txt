class Node:
    def __init__(self, val=None):
        self.val = val
        self.nextval = None

class LinkedList:
    def __init__(self):
        self.head = None

    def add_end(self, data):
        new_node = Node(data)
        if self.head is None:
            self.head = new_node
            return
        last_node = self.head
        while last_node.nextval:
            last_node = last_node.nextval
        last_node.nextval = new_node

    def remove_node(self, key):
        current_node = self.head
        if current_node and current_node.val == key:
            self.head = current_node.nextval
            current_node = None
            return
        prev = None
        while current_node and current_node.val != key:
            prev = current_node
            current_node = current_node.nextval
        if current_node is None:
            return
        prev.nextval = current_node.nextval
        current_node = None

    def print_list(self):
        current_node = self.head
        while current_node:
            print(current_node.val)
            current_node = current_node.nextval

def main():
    days_of_week = ["Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday", "Sunday"]
    day_list = LinkedList()
    for day in days_of_week:
        day_list.add_end(day)
    day_list.print_list()
    print("*" * 40)
    day_list.add_end("frargsday")
    day_list.print_list()
    print("*" * 40)
    day_list.remove_node("frargsday")
    day_list.print_list()

if __name__ == "__main__":
    main()
