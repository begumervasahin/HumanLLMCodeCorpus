import networkx as nx
import operator
import matplotlib.pyplot as plt
from disjoint_set import DisjointSet

# Create an empty graph and disjoint set
G = nx.Graph()
disjoint_set = DisjointSet(['a', 'b', 'c', 'd'])

# Define nodes with positions and keys
nodes_data = {
    'a': {'pos': (0, 1), 'key': 0},
    'b': {'pos': (2, 1), 'key': 0},
    'c': {'pos': (1, 0), 'key': 0},
    'd': {'pos': (1, 2), 'key': 0}
}

# Add nodes to the graph
G.add_nodes_from(nodes_data.items())

# Define edges with weights
edges_data = {
    ('a', 'b'): {'weight': 2},
    ('a', 'c'): {'weight': 10},
    ('b', 'c'): {'weight': 90},
    ('b', 'd'): {'weight': 1},
    ('d', 'a'): {'weight': 3}
}

# Add edges to the graph
G.add_edges_from(edges_data.items())

# Get edge and node attributes
edges = nx.get_edge_attributes(G, 'weight')
pos = nx.get_node_attributes(G, 'pos')

# Initialize an empty graph for Kruskal's algorithm
KruskalG = nx.Graph()

# Sort edges by weight
sorted_edges = sorted(edges.items(), key=operator.itemgetter(1))

# Iterate through sorted edges
for uv, weight in sorted_edges:
    u, v = uv
    
    # Check if adding edge (u, v) forms a cycle
    if disjoint_set.find(u) != disjoint_set.find(v):
        disjoint_set.union(u, v)
        KruskalG.add_edge(u, v, weight=weight, color='b')  # Add edge to Kruskal's graph with blue color
    else:
