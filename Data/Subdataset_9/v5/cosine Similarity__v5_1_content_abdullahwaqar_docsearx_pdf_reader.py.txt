"""
This script provides functions for reading and extracting text data from PDF files.
Author: Mubashar Hussain
"""

import fitz
from storage import enumerate_filenames

def read_all_pdfs():
    """
    Read all PDF files from the storage and return the text along with the file name.

    Returns:
        list: A list of tuples containing the PDF name and text data from all the PDFs.
    """
    all_pages = []
    for pdf_file in enumerate_filenames():
        with fitz.open(pdf_file) as pdf_document:
            for page in pdf_document:
                all_pages.append((pdf_file, page.get_text()))
    return all_pages

def read_pdf(pdf_path):
    """
    Read a PDF file from the storage and return the text along with the file name.

    Args:
        pdf_path (str): Path to the PDF file.

    Returns:
        list: A list containing a tuple with the PDF name and text data from the PDF.
    """
    pages = []
    with fitz.open(pdf_path) as pdf_file:
        for page in pdf_file:
            pages.append((pdf_path, page.get_text()))
    return pages
