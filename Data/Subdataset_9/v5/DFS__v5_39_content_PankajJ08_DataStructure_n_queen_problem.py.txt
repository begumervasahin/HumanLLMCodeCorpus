# Unicode symbol representing the queen chess piece
QUEEN = "\u265B"
# Size of the chessboard
BOARD_SIZE = 8
# Counter for solution numbering
solution_count = 1

def is_safe(board, row, col):
    """Check if a queen can be placed at board[row][col]."""
    # Check if there's a queen in the same row to the left
    for i in range(col):
        if board[row][i] == QUEEN:
            return False
    # Check upper-left diagonal
    for i, j in zip(range(row, -1, -1), range(col, -1, -1)):
        if board[i][j] == QUEEN:
            return False
    # Check lower-left diagonal
    for i, j in zip(range(row, BOARD_SIZE), range(col, -1, -1)):
        if board[i][j] == QUEEN:
            return False
    return True

def find_all_solutions(board, col):
    """Find and print all solutions for the N-Queen problem."""
    global solution_count
    # Base case: all queens are placed
    if col >= BOARD_SIZE:
        # Print the solution
        print("Solution", solution_count, ": ")
        for i in range(BOARD_SIZE):
            for j in range(BOARD_SIZE):
                print(board[i][j], end="  ")
            print()
        print()
        # Increment solution count
        solution_count += 1
        return True
    # Try placing a queen in each row of the current column
    for i in range(BOARD_SIZE):
        if is_safe(board, i, col):
            # Place the queen
            board[i][col] = QUEEN
            # Recursively search for solutions for the next column
            if find_all_solutions(board, col + 1):
                return True
            # If no solution found, backtrack and try next row
            board[i][col] = '_'
    return False

def n_queen():
    """Start the program to find and print all N-Queen solutions."""
    # Initialize the chessboard
    board = [['_'] * BOARD_SIZE for _ in range(BOARD_SIZE)]
    # Find solutions starting from the first column
    if not find_all_solutions(board, 0):
        print("No solution exists!")

if __name__ == '__main__':
    # Start the program
    n_queen()
