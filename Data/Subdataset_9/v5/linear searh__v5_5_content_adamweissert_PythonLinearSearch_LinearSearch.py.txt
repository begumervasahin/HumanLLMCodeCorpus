import time

def linear_search(item_to_find, my_list):
    found = False
    position = 0
    
    while position < len(my_list) and not found:
        if my_list[position] == item_to_find:
            found = True
        position += 1
    
    return found

def calculate_runtime(search_function, item, my_list):
    start_time = time.perf_counter()
    result = search_function(item, my_list)
    end_time = time.perf_counter()
    
    elapsed_time = (end_time - start_time) * 100000
    return elapsed_time

def main():
    numbers = [-125, -50, -1, 0, 1, 5, 7, 10, 12, 20, 26, 30, 50, 127, 500, 833, 900, 1000, 12345, 123456]
    
    number_to_find = int(input("What number do you want to find? "))
    is_found = linear_search(number_to_find, numbers)
    time_to_find = calculate_runtime(linear_search, number_to_find, numbers)
    
    if is_found:
        print("Your item is in the list.")
        print("Time taken to find: {:.6f} microseconds".format(time_to_find))
    else:
        print("Your item is not in the list.")
        print("Time taken to search: {:.6f} microseconds".format(time_to_find))

if __name__ == "__main__":
    main()
