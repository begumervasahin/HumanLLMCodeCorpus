import numpy as np
import random
from copy import deepcopy

class Tree(list):
    """
    Represents a tree structure for genetic programming.
    """

    def __init__(self, content):
        super().__init__(content)
        self.fitness = None
        self.errors = None

    @property
    def height(self):
        """
        Computes the height of the tree.
        """
        stack = [0]
        max_depth = 0
        for node in self:
            depth = stack.pop()
            max_depth = max(max_depth, depth)
            stack.extend([depth + 1] * node[2])
        return max_depth

    def subtree(self, begin):
        """
        Returns the end index of the subtree rooted at the given index.
        """
        end = begin + 1
        total = self[begin][2]
        while total > 0:
            total += self[end][2] - 1
            end += 1
        return end

    def __str__(self):
        """
        Returns a string representation of the tree.
        """
        stack = []
        for node in self[::-1]:
            name, func, arity, const = node
            if arity != 0:
                stack.append(name + "(" + ",".join([stack.pop() for _ in range(arity)]) + ")")
            else:
                stack.append(name)
        return stack[0]

    def run(self, X):
        """
        Executes the tree on the given input data.
        """
        ones = np.ones(len(X))
        x = X.flatten()
        stack = []
        for node in self[::-1]:
            name, func, arity, const = node
            if name == "x":
                func_ret = x
            elif arity == 0:
                func_ret = ones * const
            else:
                func_ret = func(*[stack.pop() for _ in range(arity)])
            stack.append(func_ret)
        return stack[0]

    def to_tex_text(self):
        """
        Converts the tree to a LaTeX expression.
        """
        trans_dict = {
            "add": lambda a, b: "({}+{})".format(a, b),
            "sub": lambda a, b: "({}-{})".format(a, b),
            "mul": lambda a, b: "{}{}".format(a, b),
            "div": lambda a, b: "\\frac{{{0}}}{{{1}}}".format(a, b),
            "sin": lambda a: "\\sin({})".format(a),
            "cos": lambda a: "\\cos({})".format(a),
            "tan": lambda a: "\\tan({})".format(a),
            "log": lambda a: "\\log({})".format(a),
            "exp": lambda a: "\\exp({})".format(a),
            "sqrt": lambda a: "\\sqrt{{{0}}}".format(a),
            "-1": lambda: "-1",
            "0": lambda: "0",
            "1": lambda: "1",
            "0.5": lambda: "0.5",
            "pi": lambda: "\\pi",
            "e": lambda: "e",
            "x": lambda: "x",
        }
        stack = []
        for node in self[::-1]:
            name, func, arity, const = node
            trans_func = trans_dict[name]
            stack.append(trans_func(*[stack.pop() for _ in range(arity)]))
        return stack[0]

    def to_sympy_input_text(self):
        """
        Converts the tree to a SymPy input expression.
        """
        trans_dict = {
            "add": lambda a, b: "({}+{})".format(a, b),
            "sub": lambda a, b: "({}-{})".format(a, b),
            "mul": lambda a, b: "{}*{}".format(a, b),
            "div": lambda a, b: "{}/{}".format(a, b),
            "sin": lambda a: "sin({})".format(a),
            "cos": lambda a: "cos({})".format(a),
            "tan": lambda a: "tan({})".format(a),
            "log": lambda a: "log({})".format(a),
            "exp": lambda a: "exp({})".format(a),
            "sqrt": lambda a: "{}**(1/2)".format(a),
            "-1": lambda: "-1",
            "0": lambda: "0",
            "1": lambda: "1",
            "0.5": lambda: "0.5",
            "pi": lambda: "pi",
            "e": lambda: "e",
            "x": lambda: "x",
        }
        stack = []
        for node in self[::-1]:
            name, func, arity, const = node
            trans_func = trans_dict[name]
            stack.append(trans_func(*[stack.pop() for _ in range(arity)]))
        return stack[0]

def create_tree(node_set, leaf_set):
    """
    Creates a tree with random nodes and leaves.
    """
    height = random.randint(5, 5)
    items = []
    stack = [0]
    while stack:
        depth = stack.pop()
        if depth == height:
            node = random.choice(leaf_set)
        else:
            node = random.choice(node_set)
            stack.extend([depth + 1] * node[2])
        items.append(node)
    return Tree(items)

def swap_tree(tree_a, tree_b, limit):
    """
    Swaps subtrees between two trees.
    """
    if len(tree_a) == 1 or len(tree_b) == 1:
        return tree_a, tree_b
    for _ in range(100):
        s_a = random.randrange(1, len(tree_a))
        s_b = random.randrange(1, len(tree_b))
        e_a = tree_a.subtree(s_a)
        e_b = tree_b.subtree(s_b)
        tree_a[s_a:e_a], tree_b[s_b:e_b] = tree_b[s_b:e_b], tree_a[s_a:e_a]
        if tree_a.height <= limit and tree_b.height <= limit:
            break
        tree_a[s_a:s_a + (e_b - s_b)], tree_b[s_b:s_b + (e_a - s_a)] = tree_b[s_b:s_b + (e_a - s_a)], tree_a[s_a:s_a + (e_b - s_b)]
    return tree_a, tree_b

def point_change(tree, node_set, leaf_set, mutpb):
    """
    Performs point mutation on the tree.
    """
    for i, node in enumerate(tree):
        if random.random() < mutpb:
            name, func, arity, const = node
            if arity == 0:
                tree[i] = random.choice(leaf_set)
            else:
                tree[i] = random.choice([_node for _node in node_set if _node[2] == arity])
    return tree

def make_node_set():
    """
    Creates sets of nodes and leaves.
    """
    node_set = []
    leaf_set = []
    node_set.append(("add", np.add, 2, None))
    node_set.append(("sub", np.subtract, 2, None))
    node_set.append(("mul", np.multiply, 2, None))
    node_set.append(("div", np.divide, 2, None))
    node_set.append(("sin", np.sin, 1, None))
    node_set.append(("cos", np.cos, 1, None))
    node_set.append(("tan", np.tan, 1, None))
    node_set.append(("log", np.log, 1, None))
    node_set.append(("exp", np.exp, 1, None))
    node_set.append(("sqrt", np.sqrt, 1, None))
    leaf_set.append(("pi", None, 0, np.pi))
    leaf_set.append(("e", None, 0, np.e))
    leaf_set.append(("x", None, 0, None))
    return node_set, leaf_set

def initial_population(n, node_set, leaf_set):
    """
    Generates an initial population of trees.
    """
    return [create_tree(node_set, leaf_set) for _ in range(n)]

def evaluation(pop, X, y):
    """
    Evaluates the fitness of each tree in the population.
    """
    for tree in pop:
        if tree.fitness is None:
            _y = tree.run(X)
            errors = np.abs(y - _y)
            fitness = np.sum(errors) / len(y)
            if np.isfinite(fitness) and fitness < 9999999999:
                tree.fitness = fitness
                tree.errors = errors
            else:
                tree.fitness = None
    return pop

def selection(pop, elite, batch_size=8, tour_size=64):
    """
    Selects individuals for the next generation using tournament selection.
    """
    pop_size = len(pop)
    pop = [tree for tree in pop if tree.fitness is not None]
    best = min(pop, key=lambda tree: tree.fitness)
    if elite is None or best.fitness < elite.fitness:
        elite = deepcopy(best)
    T = np.arange(len(pop[0].errors))
    np.random.shuffle(T)
    t_start = 0
    next_pop = []
    for _ in range(pop_size - 1):
        best = None
        best_ave = 0
        subpop = random.sample(pop, tour_size) if len(pop) > tour_size else pop
        t_end = t_start + batch_size if (t_start + batch_size) < len(T) else len(T)
        for tree in subpop:
            ave = np.sum([tree.errors[t] for t in T[t_start:t_end]]) / batch_size
            if best is None or ave < best_ave:
                best = tree
                best_ave = ave
        next_pop.append(deepcopy(best))
        t_start = t_end
        if t_start == len(T):
            t_start = 0
    next_pop.append(deepcopy(elite))
    return next_pop, elite

def crossover(pop, cxpb, limit):
    """
    Applies crossover to the population.
    """
    for i in range(0, len(pop), 2):
        if random.random() < cxpb:
            pop[i], pop[i + 1] = swap_tree(pop[i], pop[i + 1], limit)
            pop[i].fitness = None
            pop[i + 1].fitness = None
    return pop

def mutation(pop, mutpb, node_set, leaf_set):
    """
    Applies mutation to the population.
    """
    for i in range(len(pop)):
        pop[i] = point_change(pop[i], node_set, leaf_set, mutpb)
        pop[i].fitness = None
    return pop
