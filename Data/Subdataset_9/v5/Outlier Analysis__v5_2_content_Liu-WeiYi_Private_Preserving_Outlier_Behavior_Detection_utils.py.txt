def convert_to_seconds(time_str):
    """
    Convert time string to seconds since epoch
    """
    format_time_str = "%s-%s-%s %s:%s:%s" % (
        time_str[0:4],
        time_str[4:6],
        time_str[6:8],
        time_str[8:10],
        time_str[10:12],
        time_str[12:]
    )
    seconds = datetime.strptime(format_time_str, "%Y-%m-%d %H:%M:%S")
    return time.mktime(seconds.timetuple())

def calculate_median_delta_T(all_time):
    """
    Calculate the median time difference between consecutive events
    """
    delta_times = []
    for idx in range(len(all_time) - 1):
        t1 = convert_to_seconds(all_time[idx])
        t2 = convert_to_seconds(all_time[idx + 1])
        delta_times.append(abs(t2 - t1))

    delta_times = [delta for delta in delta_times if delta != 0]

    if not delta_times:
        delta_times = [0]

    return mean(delta_times)

def generate_time_groups(all_time, deltaT):
    """
    Generate time groups based on a maximum time difference
    """
    interval_count = 1
    interval_dict = {}
    t1_idx = 0
    last_one_hit = False

    while t1_idx < len(all_time):
        t2_idx = t1_idx + 1
        if t2_idx >= len(all_time):
            break

        t1 = all_time[t1_idx]
        t2 = all_time[t2_idx]
        group = set()

        while abs(convert_to_seconds(t1) - convert_to_seconds(t2)) <= deltaT:       
            group.add(t1)
            group.add(t2)
            t1_idx += 1

            if all_time[t1_idx] == all_time[-1]:
                last_one_hit = True

            t2_idx += 1

            if t2_idx >= len(all_time):
                interval_dict[interval_count] = list(group)       
                interval_count += 1
                t1_idx = t2_idx
                break

            t1 = all_time[t1_idx]
            t2 = all_time[t2_idx]

        interval_dict[interval_count] = list(group)
        interval_count += 1
        t1_idx = t2_idx

    if not last_one_hit:
        interval_dict[interval_count] = [all_time[-1]]

    return interval_dict
