import numpy as np
from k_means import KMeans

class BisectingKMeans:
    def fit(self, X, n_clusters):
        '''
        Parameters
        ----------
        X : shape (n_samples, n_features)
            Training data
        n_clusters : The number of clusters
        Returns
        -------
        y : shape (n_samples,)
            Predicted cluster label per sample.
        '''
        n_samples = X.shape[0]
        data = X
        clusters = []
        
        while True:
            # Initialize a KMeans model
            kmeans_model = KMeans()
            # Fit the KMeans model to the data with 2 clusters and maximum 100 iterations
            cluster_labels = kmeans_model.fit(data, 2, 100)
            
            # Append indices of data points for each cluster
            clusters.append(np.flatnonzero(cluster_labels == 0))
            clusters.append(np.flatnonzero(cluster_labels == 1))
            
            # Check if desired number of clusters is reached
            if len(clusters) == n_clusters:
                break
            
            # Calculate sum of squared errors for each cluster
            cluster_errors = [np.var(data[cluster]) for cluster in clusters]
            # Choose cluster with highest SSE
            data = data[clusters[np.argmax(cluster_errors)]]
            # Remove chosen cluster from list
            del clusters[np.argmax(cluster_errors)]
        
        # Assign cluster labels to each data point
        y = np.zeros(n_samples)
        for i in range(len(clusters)):
            y[clusters[i]] = i
        
        return y
