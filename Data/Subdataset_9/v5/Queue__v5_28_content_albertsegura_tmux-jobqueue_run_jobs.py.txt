import sys
import argparse
import os
import subprocess
import datetime

# Sets of possible yes and no answers
YES_ANSWERS = {'yes', 'y', '', 'si', 's'}
NO_ANSWERS = {'no', 'n'}

def parse_command_line_arguments():
    """
    Parse command-line arguments.
    """
    parser = argparse.ArgumentParser(description='Mem trace analysis run folder')
    parser.add_argument('session_name', action="store", help='Session name')
    parser.add_argument('run_file', action="store", help='Path to the run file')
    return parser.parse_args()

def generate_session_name(session_name):
    """
    Generate session name based on current timestamp.
    """
    timestamp = datetime.datetime.now().strftime("%Y-%m-%d_%H-%M-%S")
    return f"job_{os.path.relpath(session_name)}_{timestamp}"

def display_run_file_info(run_file):
    """
    Display runfile content.
    """
    print("Runfile content:")
    print("========================================")
    with open(run_file) as file:
        for idx, line in enumerate(file):
            linecmd = line.rstrip('\n')
            if linecmd != "":
                print(f"{idx}: {linecmd}")
    print("========================================")
    return idx

def confirm_runfile_content():
    """
    Prompt user to confirm runfile content.
    """
    is_correct = input("Is runfile content correct? [Y/n] ").lower()
    return is_correct in YES_ANSWERS

def check_inside_tmux():
    """
    Check if running inside TMUX and prompt user to proceed.
    """
    if os.environ.copy().get("TMUX") is not None:
        inside_tmux = input("Running inside TMUX, proceed? [Y/n] ").lower()
        if inside_tmux not in YES_ANSWERS:
            exit(-1)
        del os.environ['TMUX']

def launch_tmux_session(session_name, runfile):
    """
    Launch a new session using TMUX.
    """
    tmux_cmds = f"python3 ~/scripts/run_jobs_launcher.py {session_name} {runfile} ; read"
    cmd = ["tmux", "new-session", "-d", "-s", session_name, tmux_cmds]
    try:
        subprocess.check_output(cmd)
        print("Session launched")
    except subprocess.CalledProcessError:
        print("Error: failed to launch TMUX session")
        exit(-1)

def prompt_to_switch_to_tmux_session(session_name):
    """
    Prompt user to switch to the TMUX job session.
    """
    want_to_attach = input("Switch to tmux job session? [Y/n] ").lower()
    if want_to_attach in YES_ANSWERS:
        cmd = ["tmux", "switch-client", "-t", session_name]
        try:
            subprocess.check_output(cmd)
        except subprocess.CalledProcessError:
            print("Error: failed to switch to TMUX job session")

def main():
    """
    Main function to execute the script.
    """
    args = parse_command_line_arguments()
    session_name = generate_session_name(args.session_name)

    print("Session name:", session_name)
    print("Runfile:", args.run_file, "\n")

    run_file_info = display_run_file_info(args.run_file)
    if not confirm_runfile_content():
        exit(-1)

    check_inside_tmux()

    if os.path.exists(session_name):
        print("Error: session", session_name, "already exists")
        exit(-1)

    launch_tmux_session(session_name, args.run_file)
    prompt_to_switch_to_tmux_session(session_name)

if __name__ == '__main__':
    main()
