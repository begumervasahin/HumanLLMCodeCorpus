def quick_sort(arr, left, right):
    """
    Sorts the given list using the quicksort algorithm.

    Args:
        arr (list): The list to be sorted.
        left (int): The index of the leftmost element in the sublist to be sorted.
        right (int): The index of the rightmost element in the sublist to be sorted.
    """
    if left < right:
        # Choose the pivot element (first element in the sublist)
        pivot = arr[left]
        # Initialize the partition index
        partition_index = left
        # Iterate through the sublist from left+1 to right
        for i in range(left + 1, right):
            # If the current element is less than the pivot, swap it with the element at partition index
            if arr[i] < pivot:
                partition_index += 1
                arr[partition_index], arr[i] = arr[i], arr[partition_index]
        # Swap the pivot element with the element at the partition index
        arr[left], arr[partition_index] = arr[partition_index], arr[left]
        # Recursively sort the sublists to the left and right of the pivot
        quick_sort(arr, left, partition_index - 1)
        quick_sort(arr, partition_index + 1, right)
