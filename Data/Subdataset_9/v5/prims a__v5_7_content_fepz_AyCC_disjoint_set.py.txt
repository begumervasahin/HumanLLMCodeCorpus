class DisjointSets:
    def __init__(self, elements):
        # Initialize the disjoint sets with given elements
        self.elements = list(elements)
        # Initialize sets: each element is initially in its own set
        self.parent_sets = list(range(len(elements)))
        # Initialize ranks of each set to 1
        self.set_ranks = [1] * len(elements)

    def find(self, element):
        """Find the representative of the set containing the given element."""
        index = root_index = self.elements.index(element)
        # Find the root of the set containing the element
        while self.parent_sets[root_index] != root_index:
            root_index = self.parent_sets[root_index]
        # Perform path compression
        while index != root_index:
            next_index = self.parent_sets[index]
            self.parent_sets[index] = root_index
            index = next_index
        return root_index

    def merge(self, set_a, set_b):
        """Merge two sets specified by their representatives."""
        if self.set_ranks[set_a] == self.set_ranks[set_b]:
            # If ranks are equal, increment the rank of the resulting set
            self.set_ranks[set_a] += 1
            # Attach set_b to set_a
            self.parent_sets[set_b] = set_a
        else:
            # Attach the smaller ranked set to the larger ranked set
            if self.set_ranks[set_a] > self.set_ranks[set_b]:
                self.parent_sets[set_b] = set_a
            else:
                self.parent_sets[set_a] = set_b
