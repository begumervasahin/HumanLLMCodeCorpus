]

SEEDS = [62, 43, 97, 104, 123, 1003, 554, 221, 11, 992]

# Generate unique run IDs based on combinations of parameters
run_ids = []
for seed in SEEDS:

                run_ids.append((seed, game, gamma, name, rp, noise, use_s, use_s_a, use_s_a_sprime))
