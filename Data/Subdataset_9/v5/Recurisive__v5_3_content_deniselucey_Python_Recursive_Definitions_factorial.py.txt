def factorial(n):
    # Base case: factorial of 0 or negative number is 1
    if n <= 1:
        return 1
    else:
        # Recursively calculate factorial
        result = n * factorial(n - 1)
        # Print each step of the factorial calculation
        print(f"Factorial of {n}: {result}")
        return result

# Calculate and print factorial of 4
factorial(4)
