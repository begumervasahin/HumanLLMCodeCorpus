def is_anagram(s1, s2):
    """
    Check if two strings are anagrams of each other.

    Parameters:
    - s1: str, first string
    - s2: str, second string

    Returns:
    - bool, True if s1 and s2 are anagrams, False otherwise
    """
    # Convert strings to lists and sort them
    s1_sorted = sorted(s1)
    s2_sorted = sorted(s2)

    # Check if sorted lists are equal
    return s1_sorted == s2_sorted

# Test the function
print(is_anagram('ey edip', 'pide ye'))
