from bs4 import BeautifulSoup
import requests
import csv
import os

def construct_search_query(product):
    return f"/search?q={product}&otracker=search&otracker1=search&marketplace=FLIPKART&as-show=off&as=off"

def get_next_page_link(soup):
    return soup.find_all(class_="_3fVaIS")[-1].get("href")

def extract_product_info(product):
    title = product.find(class_='_3wU53n').text.strip()
    price = product.find(class_="_1vC4OE _2rQ-NK").text.strip()[1:]
    rating = product.find(class_="hGSR34")
    rating = "No Rating" if rating is None else rating.text.strip()
    return title, price, rating

def search_flipkart(product, pages):
    search_query = construct_search_query(product)
    file_name = f"Flipkart Scraping on {product}.csv"
    
    if os.path.exists(file_name):
        print("You have already searched for this...\nDeleting your previous results...")
        os.remove(file_name)

    for page_num in range(pages):
        try:
            with open(file_name, "a", newline='', encoding='utf-8') as file:
                csv_file = csv.writer(file)
                
                if page_num == 0:
                    csv_file.writerow(["TITLE", "Price", "Rating"])
                
                url = "https://www.flipkart.com" + search_query
                print("Working on Page No :", page_num + 1)
                
                response = requests.get(url)
                soup = BeautifulSoup(response.text, 'html.parser')
                
                products = soup.find_all(class_="_1UoZlX")
                for product in products:
                    title, price, rating = extract_product_info(product)
                    csv_file.writerow([title, price, rating])

                search_query = get_next_page_link(soup)

        except IndexError:
            print("Out Of Pages for your search!!!")
            break
        except Exception as e:
            print("An error occurred:", e)
            print("One Item Skipped Due to an internal error")

if __name__ == "__main__":
    product = input("Enter product you want to search: ")
    pages = int(input("Enter number of pages: "))

    search_flipkart(product, pages)
