import os
import pandas as pd

# Function to get a list of files in a directory
def get_file_list(directory):
    path = directory
    files = os.listdir(path)
    return files

# Get file lists from two different directories
all_files_original = get_file_list('/TF-IDF')
all_files_modified = get_file_list('/TF-IDF1')

# Change directory
os.chdir('/TF-IDF')
destination_path = '/TF-IDF1'

# Initialize a counter for exceptions
exception_count = 0

# Process each file in the original directory
for file_name in all_files_original:
    # Check if the file is not present in the modified directory
    if file_name not in all_files_modified:
        try:
            # Read the CSV file with specific settings
            data = pd.read_csv(file_name, sep='    ', header=None, engine='python')
            # Filter rows where the second column is greater than 0
            filtered_data = data[data[1] > 0]
            # Sort the filtered data by the second column in descending order
            sorted_data = filtered_data.sort_values(by=[1], axis=0, ascending=False)
            
            # Check the length of sorted data
            if len(sorted_data) <= 50:
                # Write sorted data to a new file
                new_file_name = file_name.replace('.txt', '')
                new_file_path = f"{destination_path}/{new_file_name}.txt"
                sorted_data.to_csv(new_file_path, sep='\t', header=None, index=False)
            else:
                # Select the top 50 rows if the length exceeds 50
                top_50 = sorted_data.iloc[0:50, :]
                new_file_name = file_name.replace('.txt', '')
                new_file_path = f"{destination_path}/{new_file_name}.txt"
                top_50.to_csv(new_file_path, sep='\t', header=None, index=False)
        except Exception as e:
            exception_count += 1

# Print a message after execution
print('Execution complete!')
