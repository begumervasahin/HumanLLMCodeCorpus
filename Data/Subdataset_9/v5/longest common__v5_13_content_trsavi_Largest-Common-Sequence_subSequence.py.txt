def longest_common_subsequence(string1, string2):
    """
    Finds the longest common subsequence in order between two strings.
    
    Args:
        string1 (str): The first input string.
        string2 (str): The second input string.
    
    Returns:
        str: The longest common subsequence.
    """
    # Convert strings to lists for easier manipulation
    string1_chars = list(string1)
    string2_chars = list(string2)
    
    # Initialize empty lists to store results
    current_subsequence = []
    longest_subsequence = []
    
    # Iterate through each character in the shorter string
    for i in range(len(string1_chars)):
        for j in range(len(string2_chars)):
            if string1_chars[i] == string2_chars[j]:
                # If characters match, add to the current subsequence
                current_subsequence.append(string1_chars[i])
                i += 1
        
        # Update the longest subsequence if the current one is longer
        if len(current_subsequence) > len(longest_subsequence):
            longest_subsequence = current_subsequence[:]
        
        # Clear the current subsequence for the next iteration
        current_subsequence.clear()
    
    # Convert the longest subsequence list to a string
    longest_subsequence_str = ''.join(longest_subsequence)
    
    return longest_subsequence_str
