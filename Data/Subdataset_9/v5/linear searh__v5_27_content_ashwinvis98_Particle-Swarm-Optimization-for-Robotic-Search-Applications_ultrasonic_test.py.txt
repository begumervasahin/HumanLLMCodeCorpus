import os
import time
import threading
import nxt
from nxt.sensor import *

# Connect to the NXT brick
brick_address = '00:16:53:0C:1D:29'
brick = nxt.locator.find_one_brick(brick_address)

# Function to continuously read ultrasonic sensor data
def read_ultrasonic_sensor():
    while True:
        distance = Ultrasonic(brick, PORT_2, check_compatible=False).get_sample()
        print("Distance to obstacle:", distance)

# Function to continuously increment a variable and print its value   
def increment_counter(start_value):
    counter = start_value
    while True:
        counter += 1
        print("Counter:", counter)

# Create threads for ultrasonic sensor and counter functions
ultrasonic_thread = threading.Thread(target=read_ultrasonic_sensor)        
counter_thread = threading.Thread(target=increment_counter, args=(1,))

# Start the threads
ultrasonic_thread.start()
counter_thread.start()

# Wait for the threads to finish
ultrasonic_thread.join()
counter_thread.join()
