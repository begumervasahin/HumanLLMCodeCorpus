import nltk
import warnings
import numpy as np
import random
import string
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

# Download NLTK resources
nltk.download('punkt')
nltk.download('wordnet')

warnings.filterwarnings("ignore")

# Read the text files
with open('nlp_python_answer_finals.txt', 'r', errors='ignore') as f:
    python_text = f.read().lower()

with open('modules_pythons.txt', 'r', errors='ignore') as m:
    module_text = m.read().lower()

# Tokenization
def tokenize(text):
    return nltk.sent_tokenize(text), nltk.word_tokenize(text)

sent_tokens, word_tokens = tokenize(python_text)
sent_tokens_module, word_tokens_module = tokenize(module_text)

# Lemmatization
lemmatizer = nltk.stem.WordNetLemmatizer()

def lemmatize(tokens):
    return [lemmatizer.lemmatize(token) for token in tokens]

# Remove punctuation and lemmatize
def normalize(text):
    remove_punct_dict = dict((ord(punct), None) for punct in string.punctuation)
    return lemmatize(nltk.word_tokenize(text.lower().translate(remove_punct_dict)))

# Responses
greeting_inputs = ("hello", "hi", "hey", "greetings", "sup", "what's up")
greeting_responses = ["Hi", "Hey", "Hello", "I am glad you are talking to me"]

introduce_responses = [
    "My name is PyBot.",
    "My name is PyBot you can call me Pi.",
    "I'm PyBot :)",
    "My name is PyBot and my nickname is Pi. I am happy to solve your queries :)"
]

basic_qa_pairs = {
    "what is python ?": "Python is a high-level, interpreted, interactive and object-oriented scripting programming language. Python is designed to be highly readable. It uses English keywords frequently whereas other languages use punctuation, and it has fewer syntactical constructions than other languages."
}

module_qa_pairs = [
    ("what is module", "what is module?"),
    ("what is module?", "what is module?")
]

module_responses = [
    "Consider a module to be the same as a code library.",
    "A file containing a set of functions you want to include in your application.",
    "A module can define functions, classes and variables. A module can also include runnable code. Grouping related code into a module makes the code easier to understand and use."
]

def greeting(sentence):
    for word in sentence.split():
        if word.lower() in greeting_inputs:
            return random.choice(greeting_responses)

def basic_response(sentence):
    for question, answer in basic_qa_pairs.items():
        if sentence.lower() == question:
            return answer

def module_response(sentence):
    for question_pair in module_qa_pairs:
        if sentence.lower() == question_pair[0] or sentence.lower() == question_pair[1]:
            return random.choice(module_responses)

def introduce_response(sentence):
    return random.choice(introduce_responses)

def get_response(user_response, sent_tokens):
    robo_response = ''
    sent_tokens.append(user_response)

    tfidf_vectorizer = TfidfVectorizer(tokenizer=normalize, stop_words='english')
    tfidf_matrix = tfidf_vectorizer.fit_transform(sent_tokens)

    cosine_similarities = cosine_similarity(tfidf_matrix[-1], tfidf_matrix)
    idx = cosine_similarities.argsort()[0][-2]

    req_tfidf = cosine_similarities.flatten()[idx]

    if req_tfidf == 0:
        robo_response = "I am sorry! I don't understand you"
    else:
        robo_response = sent_tokens[idx]

    return robo_response

def chat(user_response):
    user_response = user_response.lower()
    keyword = "module"

    if user_response != 'bye':
        if user_response in ['thanks', 'thank you']:
            return "You are welcome.."
        elif module_response(user_response) is not None:
            return module_response(user_response)
        else:
            if keyword in user_response:
                return get_response(user_response, sent_tokens_module)
            elif greeting(user_response) is not None:
                return greeting(user_response)
            elif "your name" in user_response:
                return introduce_response(user_response)
            elif basic_response(user_response) is not None:
                return basic_response(user_response)
            else:
                return get_response(user_response, sent_tokens)
    else:
        return "Bye! take care.."

# Main
if __name__ == "__main__":
    while True:
        user_input = input("You: ")
        bot_response = chat(user_input)
        print("PyBot:", bot_response)
        if user_input.lower() == 'bye':
            break
