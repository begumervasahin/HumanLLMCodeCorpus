import itertools

def floyd_warshall(adjacency_matrix, num_vertices):
    """
    Applies the Floyd-Warshall algorithm to find the shortest paths between all pairs of vertices in a graph.

    Args:
        adjacency_matrix (list of lists): The adjacency matrix representing the graph.
        num_vertices (int): The number of vertices in the graph.

    Returns:
        None. Updates the input adjacency_matrix with shortest path distances.
    """
    for k in range(num_vertices):
        for i, j in itertools.product(range(num_vertices), range(num_vertices)):
            if adjacency_matrix[i][j] > adjacency_matrix[i][k] + adjacency_matrix[k][j]:
                adjacency_matrix[i][j] = adjacency_matrix[i][k] + adjacency_matrix[k][j]

# Initialize the adjacency matrix with maximum distances
adjacency_matrix = [[1000 for _ in range(4)] for _ in range(4)]

# Set diagonal elements to zero
for i in range(4):
    adjacency_matrix[i][i] = 0

# Assign specific distances between vertices
adjacency_matrix[0][2] = -2
adjacency_matrix[1][0] = 4
adjacency_matrix[1][2] = 3
adjacency_matrix[2][3] = 2
adjacency_matrix[3][1] = -1

# Print the original graph
print("Original Graph:")
for row in adjacency_matrix:
    print(row)

# Apply Floyd-Warshall algorithm
print('\nAfter Floyd-Warshall:')
floyd_warshall(adjacency_matrix, 4)

# Print the graph after applying Floyd-Warshall
for row in adjacency_matrix:
    print(row)
