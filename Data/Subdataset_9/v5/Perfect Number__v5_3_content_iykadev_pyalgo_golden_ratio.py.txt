def calculate_golden_ratio(n):
    """
    Calculate the golden ratio up to the nth term.

    Args:
        n (int): The number of terms to calculate.

    Returns:
        list: A list containing the golden ratio values up to the nth term.
    """
    golden_ratios = []
    s, t = 1, 1

    for _ in range(2, n):
        c = s + t
        s, t = t, c
        ratio = t / s
        golden_ratios.append(ratio)

    return golden_ratios

def print_golden_ratio(n):
    """
    Print the golden ratio up to the nth term.

    Args:
        n (int): The number of terms to print.
    """
    golden_ratios = calculate_golden_ratio(n)
    for ratio in golden_ratios:
        print(ratio)
