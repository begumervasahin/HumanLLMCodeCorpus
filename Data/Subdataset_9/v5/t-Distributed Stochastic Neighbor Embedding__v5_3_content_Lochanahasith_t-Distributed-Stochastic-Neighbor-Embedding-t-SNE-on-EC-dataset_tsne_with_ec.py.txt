import pandas as pd
import numpy as np
from sklearn.decomposition import PCA
from sklearn.manifold import TSNE
from sklearn import preprocessing
import matplotlib.pyplot as plt

# Load the data from the CSV file
data = pd.read_csv('endometrial.csv').dropna()

# Remove the first column (assuming it's an index column)
data = data.iloc[:, 1:]

# Scale the data
scaled_data = preprocessing.scale(data.T)  # Transpose because scale function expects samples as rows

# Apply t-SNE
tsne = TSNE(n_components=2, n_iter=2500, random_state=0)
tsne_data = tsne.fit_transform(scaled_data)

# Visualize t-SNE result
plt.scatter(tsne_data[:, 0], tsne_data[:, 1])
plt.show()
