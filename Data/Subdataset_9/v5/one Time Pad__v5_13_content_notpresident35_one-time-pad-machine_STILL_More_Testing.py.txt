from random import randint
from time import strftime

Alphabet = 'abcdefghijklmnopqrstuvwxyz'

def generate_one_time_pads(sheets, length):
    """
    Generate one-time pads.

    Args:
    - sheets (int): Number of sheets to generate.
    - length (int): Maximum message length.

    Each sheet is saved to a separate file.
    """
    for sheet in range(sheets):
        with open(f"otp{sheet}{strftime('%Y%m%d%H%M%S')}.txt", "w") as f:
            for _ in range(length):
                f.write(str(randint(0, 26)) + "\n")

def load_sheet(filename):
    """
    Load a one-time pad sheet from a file.

    Args:
    - filename (str): Name of the file containing the one-time pad sheet.

    Returns:
    - list: List of integers representing the one-time pad sheet.
    """
    with open(filename, "r") as f:
        contents = f.read().splitlines()
    return contents

def get_plaintext():
    """
    Get plaintext input from the user.

    Returns:
    - str: The plaintext message entered by the user.
    """
    plain_text = input("Enter the message to be encrypted: ")
    return plain_text.lower()

def load_file(filename):
    """
    Load data from a file.

    Args:
    - filename (str): Name of the file to load.

    Returns:
    - str: Contents of the file.
    """
    with open(filename, "r") as f:
        contents = f.read()
    return contents

def save_file(filename, data):
    """
    Save data to a file.

    Args:
    - filename (str): Name of the file to save to.
    - data (str): Data to save to the file.
    """
    with open(filename, 'w') as f:
        f.write(data)

def encrypt(plaintext, sheet):
    """
    Encrypt plaintext using a one-time pad sheet.

    Args:
    - plaintext (str): Plaintext to encrypt.
    - sheet (list): One-time pad sheet.

    Returns:
    - str: Encrypted ciphertext.
    """
    ciphertext = ''
    for position, character in enumerate(plaintext):
        if character not in Alphabet:
            ciphertext += character
        else:
            encrypted = (Alphabet.index(character) + int(sheet[position])) % 26
            ciphertext += Alphabet[encrypted]
    return ciphertext

def decrypt(ciphertext, sheet):
    """
    Decrypt ciphertext using a one-time pad sheet.

    Args:
    - ciphertext (str): Ciphertext to decrypt.
    - sheet (list): One-time pad sheet.

    Returns:
    - str: Decrypted plaintext.
    """
    plaintext = ''
    for position, character in enumerate(ciphertext):
        if character not in Alphabet:
            plaintext += character
        else:
            decrypted = (Alphabet.index(character) - int(sheet[position])) % 26
            plaintext += Alphabet[decrypted]
    return plaintext

def menu():
    """
    Display the main menu and handle user input.
    """
    while True:
        print('What would you like to do?')
        print('1. Generate one-time pads')
        print('2. Encrypt a message')
        print('3. Decrypt a message')
        print('4. Quit the program')
        choice = input('Please type 1, 2, 3, or 4 and press Enter: ')

        if choice == "1":
            sheets = int(input('How many one-time pads would you like to generate? '))
            length = int(input('What will be your maximum message length? '))
            generate_one_time_pads(sheets, length)

        elif choice == '2':
            filename = input('Type in the filename of the OTP you want to use: ')
            sheet = load_sheet(filename)
            plaintext = get_plaintext()
            ciphertext = encrypt(plaintext, sheet)
            filename = input('What will be the name of the encrypted file? ')
            save_file(filename, ciphertext)

        elif choice == '3':
            filename = input('Type in the filename of the OTP you want to use: ')
            sheet = load_sheet(filename)
            filename = input('Type in the name of the file to be decrypted: ')
            ciphertext = load_file(filename)
            plaintext = decrypt(ciphertext, sheet)
            print('The message reads:')
            print('')
            print(plaintext)

        elif choice == '4':
            print("Exiting the program...")
            break

if __name__ == "__main__":
    menu()
