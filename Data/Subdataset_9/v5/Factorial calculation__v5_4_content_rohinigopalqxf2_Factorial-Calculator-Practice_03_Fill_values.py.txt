# Import necessary modules
import time
from selenium import webdriver

# Define the main test function
def test_factorial_validation_message():
    # Launch Firefox Driver
    driver = webdriver.Firefox()
    
    try:
        # Navigate to the page
        driver.get('https://qainterview.pythonanywhere.com/')
        
        # Enter an integer value in the textbox
        textbox = driver.find_element_by_id("number")
        textbox.send_keys('12')
        
        # Click on the calculate! button
        calculate_button = driver.find_element_by_xpath("//*[@id='getFactorial']").click()  
        
        # Check the validation message
        validation_result = driver.find_element_by_xpath("//p[@id='resultDiv']")
        time.sleep(3)
        
        # Verify that no error message is displayed
        if validation_result.text != "Please enter an integer":
            print("Success: Message displayed successfully")
        else:
            print("Failed: Validation message is failing")
    finally:
        # Close the browser
        driver.quit()

# Run the test
if __name__ == "__main__":
    test_factorial_validation_message()
