def exponential_smoothing(data, column, alpha=0.5, smooth_high=True, smooth_low=True):
    """
    Apply exponential smoothing to a column in the DataFrame.

    Parameters:
        data (DataFrame): Input DataFrame.
        column (str): Name of the column to apply smoothing.
        alpha (float): Smoothing factor.
        smooth_high (bool): Flag to indicate smoothing for high values.
        smooth_low (bool): Flag to indicate smoothing for low values.

    Returns:
        None
    """
    # Calculate percentiles for the column
    percentiles = data[column].quantile([0.01, 0.05, 0.95, 0.99])
    p1, p5, p95, p99 = percentiles[0.01], percentiles[0.05], percentiles[0.95], percentiles[0.99]

    # Check if alpha is within the valid range
    if not 0 < alpha < 1:
        raise ValueError('Alpha should be between 0 and 1')

    # Sort the data by the column values
    data.sort_values(column, ascending=True, inplace=True)

    # Apply exponential smoothing for high values if requested
    if smooth_high:
        st = data.loc[(p95 >= data[column]) & (data[column] <= p99) & (data[column].notnull()), column].mean()
        for i in data[(p99 < data[column]) & (data[column].notnull())].index:
            data.loc[i, column] = alpha * data.loc[i, column] + (1 - alpha) * st
            st = data.loc[i, column]

    # Apply exponential smoothing for low values if requested
    if smooth_low:
        st = data.loc[(p1 <= data[column]) & (data[column] <= p5) & (data[column].notnull()), column].mean()
        for i in data[(p1 > data[column]) & (data[column].notnull())].index[::-1]:
            data.loc[i, column] = alpha * data.loc[i, column] + (1 - alpha) * st
            st = data.loc[i, column]

def capping_and_flooring(data, column, upper_bound, lower_bound, cap_high=True, floor_low=True):
    """
    Apply capping and flooring to a column in the DataFrame.

    Parameters:
        data (DataFrame): Input DataFrame.
        column (str): Name of the column to apply capping and flooring.
        upper_bound (float): Upper bound for capping.
        lower_bound (float): Lower bound for flooring.
        cap_high (bool): Flag to indicate capping for high values.
        floor_low (bool): Flag to indicate flooring for low values.

    Returns:
        None
    """
    if cap_high:
        data.loc[(data[column] > upper_bound) & (data[column].notnull()), column] = upper_bound
    if floor_low:
        data.loc[(data[column] < lower_bound) & (data[column].notnull()), column] = lower_bound

def create_dummies(data, column):
    """
    Create dummy variables for a categorical column.

    Parameters:
        data (DataFrame): Input DataFrame.
        column (str): Name of the categorical column.

    Returns:
        list: List of categories converted to dummies.
    """
    conversion_list = []
    value_counts = data[column].value_counts(dropna=False).sort_values(ascending=False)
    for category in value_counts.index[:-1]:
        conversion_list.append(category)
        data[column + '_dum_' + str(category)] = (data[column] == category).astype(int)
    return conversion_list

def create_binary_dummies(data, column):
    """
    Create binary dummy variables for a categorical column.

    Parameters:
        data (DataFrame): Input DataFrame.
        column (str): Name of the categorical column.

    Returns:
        list: List of categories converted to binary dummies.
    """
    value_counts = data[column].value_counts(dropna=False).sort_index()
    categories = value_counts.index.tolist()
    total_categories = len(categories)
    num_digits = len(str(total_categories))
    bin_conversion = [int(bin(i)[2:]) for i in range(total_categories)]
    conv_list = [(cat, bin_val) for cat, bin_val in zip(categories, bin_conversion)]
    for j in range(num_digits):
        data[column + '_dum_' + str(j)] = 0
        for i, (_, bin_val) in enumerate(conv_list):
            data.loc[data[column] == categories[i], column + '_dum_' + str(j)] = bin_val % 10
            bin_conversion[i] = bin_val // 10
    return conv_list
