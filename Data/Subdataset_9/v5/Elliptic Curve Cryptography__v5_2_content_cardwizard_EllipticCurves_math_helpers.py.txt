from typing import Tuple

def modulo_multiply(a: int, b: int, mod: int) -> int:
    """
    Computes (a * b) % mod.

    Args:
        a (int): The first operand.
        b (int): The second operand.
        mod (int): The modulo value.

    Returns:
        int: The result of (a * b) % mod.

    Raises:
        Exception: If mod is 0.
    """
    if mod == 0:
        raise ValueError("Modulo cannot be zero")
    return (a * b) % mod

def modulo_pow(a: int, b: int, mod: int) -> int:
    """
    Computes (a^b) % mod.

    Args:
        a (int): The base.
        b (int): The exponent.
        mod (int): The modulo value.

    Returns:
        int: The result of (a^b) % mod.
    """
    result = 1
    for _ in range(b):
        result = modulo_multiply(result, a, mod)
    return result

def egcd(a: int, b: int) -> Tuple[int, int, int]:
    """
    Extended Euclidean algorithm to compute the greatest common divisor.

    Args:
        a (int): The first integer.
        b (int): The second integer.

    Returns:
        Tuple[int, int, int]: A tuple (g, x, y) where a*x + b*y = gcd(x, y).
    """
    if a == 0:
        return b, 0, 1
    g, x, y = egcd(b % a, a)
    return g, y - (b // a) * x, x

def modulo_div(a: int, b: int, mod: int) -> int:
    """
    Computes (a / b) % mod.

    Args:
        a (int): The dividend.
        b (int): The divisor.
        mod (int): The modulo value.

    Returns:
        int: The result of (a / b) % mod.

    Raises:
        ValueError: If the modular inverse does not exist.
    """
    return modulo_multiply(a, mulinv(b, mod), mod)

def mulinv(b: int, n: int) -> int:
    """
    Computes the multiplicative inverse of b modulo n.

    Args:
        b (int): The number.
        n (int): The modulo value.

    Returns:
        int: The multiplicative inverse of b modulo n.

    Raises:
        ValueError: If the modular inverse does not exist.
    """
    g, x, _ = egcd(b, n)
    if g != 1:
        raise ValueError("Modular Inverse does not exist")
    return x % n
