import pickle
import sys

def load_word(dataset):
    file_path = f'Datasets/{dataset}.pkl'
    with open(file_path, 'rb') as file:
        word_list = pickle.load(file)

    word_dict = {}
    for i, word_entry in enumerate(word_list):
        word = word_entry.replace("â", "a")
        word_id = find_id(word_dict, word)
        word_dict[word_id] = ["kok", word_id]

    return word_dict

def find_id(word_dict, word):
    count = 1
    while f"{word}_{count}" in word_dict:
        count += 1
    return f"{word}_{count}"

def generate(word_dict, event):
    new_dict = {}

    def add_entry(base_word, suffix, category):
        new_word = base_word[:-len(suffix)] + suffix
        new_id = find_id(word_dict, new_word)
        new_dict[new_id] = [category, f"{new_word}_{ind}"]

    for word, value_list in word_dict.items():
        ind = word[word.index("_") + 1:]
        base_word = word[:word.index("_")]

        if event == "olumsuzluk eki" and base_word.endswith(("mak", "mek")):
            add_entry(base_word, "mamak" if base_word.endswith("mak") else "memek", "olumsuzluk")

        elif event == "unsuz yumusamasi" and not base_word.endswith(("mak", "mek")):
            endings = {"p": "b", "ç": "c", "t": "d", "k": "g" if base_word.endswith("nk") else "ğ"}
            for end, replacement in endings.items():
                if base_word.endswith(end):
                    add_entry(base_word, replacement, "unsuz yumusamasi")

        elif event == "unlu daralmasi":
            replacements = {"demek": "di", "yemek": "yi", "amak": "ı", "emek": "i"}
            if base_word in replacements:
                add_entry(base_word, replacements[base_word], "unlu daralmasi")

        elif event == "unlu dusmesi":
            drop_forms = {'akis': 'aks', 'akıl': 'akl', 'alın': 'aln', ...}
            if base_word in drop_forms:
                add_entry(base_word, drop_forms[base_word], "unlu dusmesi")

        elif event == "fiil" and base_word.endswith(("mak", "mek")):
            add_entry(base_word, "", "fiil")

    return new_dict

def append_dict(revised_dict, new_dict):
    for word, value_list in new_dict.items():
        if word in revised_dict:
            revised_dict[find_id(revised_dict, word[:word.index("_")])] = value_list
        else:
            revised_dict[word] = value_list
    return revised_dict

def main():
    dataset = "zargan"

    if len(sys.argv) > 1:
        dataset = sys.argv[1]

    word_dict = load_word(dataset)

    word_dict = append_dict(word_dict, generate(word_dict, "olumsuzluk eki"))
    print("Negative verbs are added.")

    word_dict = append_dict(word_dict, generate(word_dict, "fiil"))
    print("Zero infinitive forms of verbs are added.")

    revised_dict = dict(word_dict)

    for event in ["unsuz yumusamasi", "unlu daralmasi", "unlu dusmesi"]:
        new_dict = generate(word_dict, event)
        revised_dict = append_dict(revised_dict, new_dict)

        if event == "unsuz yumusamasi":
            print("Consonant softening forms are added.")
        elif event == "unlu daralmasi":
            print("Becoming close vowel forms are added.")
        elif event == "unlu dusmesi":
            print("Dropping vowel forms are added.")

    with open('revisedDict.pkl', 'wb') as file:
        pickle.dump(revised_dict, file)

    print("Transformed lexicon is saved to revisedDict.pkl")

if __name__ == "__main__":
    main()
