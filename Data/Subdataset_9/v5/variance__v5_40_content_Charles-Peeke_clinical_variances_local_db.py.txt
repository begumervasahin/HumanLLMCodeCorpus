import os
import time
import pandas as pd
import psycopg2
from dotenv import load_dotenv

# Global variables
conn = None
df = None
df_keywords = None
df_nct = None

# Load environment variables
def load_env_variables():
    global hostname, port, database, username, password
    hostname = os.getenv('hostname')
    port = os.getenv('port')
    database = os.getenv('database')
    username = os.getenv('username')
    password = os.getenv('password')

# Function to connect to the database
def connect_to_database():
    global conn
    conn = psycopg2.connect(host=hostname, database=database, user=username, password=password)

# Function to execute SQL query and return results
def execute_query(query):
    return pd.read_sql(query, con=conn)

# Function to get studies from the database
def get_studies():
    query = 'SELECT * FROM ctgov.studies'
    return execute_query(query)

# Function to get keywords from the database
def get_keywords():
    query = 'SELECT * FROM ctgov.keywords'
    return execute_query(query)

# Function to get sponsors from the database
def get_sponsors():
    query = 'SELECT * FROM ctgov.sponsors'
    return execute_query(query)

# Function to join two tables in the database
def join_tables(first_table_name, second_table_name):
    query = f"""
        SELECT * FROM {first_table_name}
        RIGHT JOIN {second_table_name}
        ON {first_table_name}.nct_id = {second_table_name}.nct_id
    """
    return execute_query(query)

if __name__ == '__main__':
    # Load environment variables
    load_dotenv()
    load_env_variables()
    
    # Connect to the database
    connect_to_database()
    
    # Measure time for joining tables
    start_time = time.time()
    results = join_tables('ctgov.studies', 'ctgov.keywords')
    end_time = time.time()
    print("Time taken for joining tables:", end_time - start_time)
    
    # Measure time for another query
    start_time = time.time()
    query = """
        SELECT * FROM ctgov.studies
        RIGHT JOIN ctgov.keywords
        ON ctgov.studies.nct_id = ctgov.keywords.nct_id
    """
    execute_query(query)
    end_time = time.time()
    print("Time taken for another query:", end_time - start_time)
