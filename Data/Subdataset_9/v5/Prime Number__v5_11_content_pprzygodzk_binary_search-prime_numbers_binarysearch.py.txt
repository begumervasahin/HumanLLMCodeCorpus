def binary_search_iterative(array, value):
    """
    Searches for the given value in the sorted array using an iterative approach.

    Parameters:
        array (list): The sorted array to search in.
        value: The value to search for in the array.

    Returns:
        str: A message indicating whether the value was found and its position in the array.
    """
    left = 0
    right = len(array) - 1
    while left <= right:
        mid = (left + right) // 2
        if array[mid] == value:
            return f"Value {value} is at position {mid} in the array"
        elif array[mid] > value:
            right = mid - 1
        else:
            left = mid + 1
    return f"Value {value} is not in the array"

def binary_search_recursive(array, value, left, right):
    """
    Searches for the given value in the sorted array using a recursive approach.

    Parameters:
        array (list): The sorted array to search in.
        value: The value to search for in the array.
        left (int): The left boundary of the search range.
        right (int): The right boundary of the search range.

    Returns:
        str: A message indicating whether the value was found and its position in the array.
    """
    if left > right:
        return f"Value {value} is not in the array"
    mid = (left + right) // 2
    if array[mid] == value:
        return f"Value {value} is at position {mid} in the array"
    elif array[mid] > value:
        return binary_search_recursive(array, value, left, mid - 1)
    else:
        return binary_search_recursive(array, value, mid + 1, right)

    prime_numbers = [
