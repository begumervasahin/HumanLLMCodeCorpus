from __future__ import division
import numpy as np
import matplotlib.pyplot as plt
from lcs import rlcs as rlcs

# Set plot directory
plot_dir = './'

# Load sample data
X = np.load('sample-data/X.npy')
Y = np.load('sample-data/Y.npy')

# Parameters
tau_dist = 0.005
score_thres = 1e-4
len_thres = 10

# Run RLCS algorithm
score, diag, cost = rlcs.rlcs(X, Y, tau_dist=tau_dist, delta=0.5)
segment = rlcs.backtrack(X, Y, score, diag, cost)

# Get soft segments
x_segments, y_segments = rlcs.getSoftSegments(segment, X, Y, score_thres=score_thres, len_thres=len_thres)

# Plot matched signals
fig, ax = plt.subplots()

# Find the longest segment
segment_lengths = [seg.shape[0] for seg in x_segments]
longest_segment_index = segment_lengths.index(max(segment_lengths))
x_longest_segment, y_longest_segment = x_segments[longest_segment_index], y_segments[longest_segment_index]

# Plot the longest segment
ax.plot(range(x_longest_segment.size), x_longest_segment)
ax.plot(range(y_longest_segment.size), y_longest_segment)

# Set labels and title
ax.set_xlabel('x')
ax.set_ylabel('y')
ax.set_title(f'Matched signals for RLCS with distance threshold {tau_dist}')

# Save the plot
fig.savefig(plot_dir + 'rlcsMain_getSegs.eps')

# Plot LCS after backtrack
fig, ax = rlcs.plotLCS(segment, X, Y)

# Set labels and title
ax.set_xlabel('query')
ax.set_ylabel('reference')
ax.set_title(f'Match of signals after backtrack with distance threshold {tau_dist}')

# Save and display the plot
fig.savefig(plot_dir + 'rlcsMain_backtrack.eps')
plt.show()
