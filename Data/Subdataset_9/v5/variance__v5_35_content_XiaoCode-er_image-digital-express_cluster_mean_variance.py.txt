import os
import math
import numpy as np
from skimage import io

def compute_mean_variance(img):
    # Calculate mean pixel value
    mean = np.mean(img)
    
    # Calculate standard deviation (square root of variance)
    variance = np.std(img)
    
    return mean, variance

def compute_cluster_center_radius(path):
    # Get list of file names in the given path
    file_names = os.listdir(path)
    
    # Initialize lists to store mean and variance values
    mean_list = []
    variance_list = []
    
    # Iterate through each image file in the directory
    for file_name in file_names:
        # Construct full path to the image file
        img_path = os.path.join(path, file_name)
        
        # Read the image as grayscale using skimage
        img = io.imread(img_path, as_gray=True)
        
        # Compute mean and variance of the image
        mean, variance = compute_mean_variance(img)
        
        # Append mean and variance to the lists
        mean_list.append(mean)
        variance_list.append(variance)
    
    # Calculate cluster center coordinates
    cluster_center = np.mean(mean_list), np.mean(variance_list)
    
    # Calculate average Euclidean distance from cluster center
    euclidean_distances = [
        math.sqrt((mean - cluster_center[0]) ** 2 + (variance - cluster_center[1]) ** 2)
        for mean, variance in zip(mean_list, variance_list)
    ]
    cluster_radius = np.mean(euclidean_distances)
    
    return cluster_center, cluster_radius
