# Define prime numbers
p = 43
q = 59

# Calculate n which is the product of p and q
n = p * q

# Define the encryption exponent
e = 13

# Prompt user for input
user_message = input("\nEnter your message: ").upper()

# Convert the user's message to a list of characters
user_characters = list(user_message)
print("User message as list of characters:", user_characters)

# Convert each character to its corresponding integer value based on its position in the alphabet
msg_integers = []
for char in user_characters:
    # Subtracting the ASCII value of 'A' to map A to 0, B to 1, and so on
    for num in str(ord(char) - ord('A')):
        msg_integers.append(num)
print("Message integers:", msg_integers)

# Determine the block size based on the length of n
block_size = len(str(n))
print("Block Size:", block_size)

# Break the message integers into blocks of size equal to the block size
blocked_msg_integers = [msg_integers[i * block_size:(i + 1) * block_size] for i in range((len(msg_integers) + block_size - 1) // block_size)]
print("Blocked message integers:", blocked_msg_integers)

# Encrypt each block of integers
encrypted_messages = []
for block in blocked_msg_integers:
    # Pad the block if its length is less than the block size
    if len(block) != block_size:
        for i in range(1, (block_size - len(block)) + 1):
            block.append(0)
    integers_string = ''.join(block)
    # Encrypt the block using RSA encryption formula
    encrypted_messages.append((int(integers_string) ** e) % n)
print("Encrypted messages:", encrypted_messages)

# Convert encrypted messages to a string
encrypted_string = ''.join(map(str, encrypted_messages))
print("\nEncrypted Messages:", encrypted_string)
