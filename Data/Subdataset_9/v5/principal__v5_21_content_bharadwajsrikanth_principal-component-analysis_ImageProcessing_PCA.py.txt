import cv2
import numpy as np
from sklearn.feature_extraction import image
from matplotlib import pyplot as plt
from PIL import Image

# Load the image
img = cv2.imread('clockwork-angels.jpg').astype(np.float64)

# Extract Red channel
R_channel = img[:, :, 2]

# Extract patches from the Red channel
patches = image.extract_patches_2d(R_channel, patch_size=(16, 16))

# Initialize correlation matrix
correlation_matrix = np.zeros((256, 256))

# Compute correlation matrix from patches
for patch in patches:
    flattened_patch = patch.flatten().reshape(-1, 1)
    correlation_matrix += np.dot(flattened_patch, flattened_patch.T)

# Compute eigenvalues and eigenvectors of the correlation matrix
eigenvalues, eigenvectors = np.linalg.eig(correlation_matrix)

# Sort eigenvalues in descending order and get corresponding indices
sorted_indices = np.argsort(eigenvalues)[::-1]

# Save and display the top 64 eigenvectors as patches
for i in range(64):
    # Select eigenvector corresponding to current index
    eigenvector = eigenvectors[:, sorted_indices[i]]
    # Reshape eigenvector to patch size
    patch = eigenvector.reshape(16, 16)
    # Save patch as an image
    plt.imsave(f"Result{i}.png", patch.real, cmap='Greys')

# Create a new blank image
back_img = Image.new('RGB', (142, 142), "white")

# Paste patches onto the blank image
x, y = 0, 0
for i in range(64):
    # Open the saved patch image
    patch_img = Image.open(f"Result{i}.png")
    # Paste the patch onto the blank image
    back_img.paste(patch_img, (x, y))
    x += 18
    if (i + 1) % 8 == 0:
        x = 0
        y += 18

# Display the resulting image
back_img.show()
