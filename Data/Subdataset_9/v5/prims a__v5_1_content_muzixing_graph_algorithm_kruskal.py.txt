def kruskal(graph):
    # Check if the input graph is a dictionary
    assert isinstance(graph, dict), "Input graph must be a dictionary"
    
    # Get the nodes from the graph
    nodes = list(graph.keys())
    
    # Set to keep track of visited nodes
    visited = set()
    
    # List to store the minimum spanning tree path
    minimum_spanning_tree = []
    
    # Loop until all nodes are visited
    while len(visited) < len(nodes):
        # Set initial distance to infinity
        min_distance = float('inf')
        
        # Variables to store the current edge
        current_source = None
        current_destination = None
        
        # Loop through all possible edges
        for source in nodes:
            for destination in nodes:
                # Skip if both nodes are visited or are the same
                if source in visited and destination in visited or source == destination:
                    continue
                
                # Check if the edge distance is smaller than the current minimum distance
                if graph[source][destination] < min_distance:
                    min_distance = graph[source][destination]
                    current_source = source
                    current_destination = destination
        
        # Add the edge with minimum distance to the minimum spanning tree
        minimum_spanning_tree.append((current_source, current_destination))
        
        # Mark both nodes as visited
        visited.add(current_source)
        visited.add(current_destination)
    
    return minimum_spanning_tree
