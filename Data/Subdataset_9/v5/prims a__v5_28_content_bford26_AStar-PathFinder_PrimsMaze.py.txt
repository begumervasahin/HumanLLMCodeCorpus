import numpy as np
from numpy.random import randint

def initialize_maze(width, height):
    """Initialize the maze grid with walls around the edges."""
    shape = (height, width)
    maze = np.zeros(shape, dtype=bool)
    maze[0, :] = maze[-1, :] = 1
    maze[:, 0] = maze[:, -1] = 1
    return maze

def add_walls(maze, start_x, start_y):
    """Mark the initial position as part of the maze."""
    maze[start_y, start_x] = 1

def get_neighbours(x, y, width, height):
    """Return a list of valid neighbouring cells."""
    neighbours = []
    if x > 1:
        neighbours.append((y, x - 2))
    if x < width - 2:
        neighbours.append((y, x + 2))
    if y > 1:
        neighbours.append((y - 2, x))
    if y < height - 2:
        neighbours.append((y + 2, x))
    return neighbours

def carve_path(maze, x, y, neighbours):
    """Carve a path from the current cell to a randomly chosen neighbouring cell."""
    y_, x_ = neighbours[randint(0, len(neighbours) - 1)]
    if maze[y_, x_] == 0:  # Check if the neighbour cell is a wall
        maze[y_, x_] = 1
        maze[y_ + (y - y_) // 2, x_ + (x - x_) // 2] = 1
        return x_, y_
    return x, y

def generate_maze(width=81, height=51, complexity=.75, density=.75):
    """Generate a maze using randomized Prim's algorithm."""
    width = (width // 2) * 2 + 1
    height = (height // 2) * 2 + 1
    maze = initialize_maze(width, height)

    complexity = int(complexity * (5 * (width + height)))
    density = int(density * ((width // 2) * (height // 2)))

    for _ in range(density):
        x, y = randint(0, width // 2) * 2, randint(0, height // 2) * 2
        add_walls(maze, x, y)

        for _ in range(complexity):
            neighbours = get_neighbours(x, y, width, height)
            if neighbours:
                x, y = carve_path(maze, x, y, neighbours)
    
    return maze

# Example of how to use the function
maze = generate_maze()
print(maze)
