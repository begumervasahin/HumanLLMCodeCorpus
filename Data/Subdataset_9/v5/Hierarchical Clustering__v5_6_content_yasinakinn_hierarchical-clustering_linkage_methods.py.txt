import pandas as pd
import matplotlib.pyplot as plt
from sklearn.cluster import AgglomerativeClustering
import scipy.cluster.hierarchy as sch

# Load dataset
dataset = pd.read_csv('customers_dataset.csv')
X = dataset.iloc[:, 2:8].values

# Perform hierarchical clustering with different linkage methods
linkage_methods = ['ward', 'complete', 'average', 'single']
n_clusters = 4
clustering_algorithms = [
    AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage=method) 
    for method in linkage_methods
]

# Fit and predict clusters
cluster_predictions = [algorithm.fit_predict(X) for algorithm in clustering_algorithms]

# Define a function to plot clusters
def plot_clusters(X, predict, title):
    colors = ['red', 'blue', 'green', 'yellow']
    for i, color in enumerate(colors):
        plt.scatter(X[predict == i, 0], X[predict == i, 1], s=100, c=color)
    plt.title(title)
    plt.legend(['Cluster 1', 'Cluster 2', 'Cluster 3', 'Cluster 4'])
    plt.show()

# Plot clusters for different linkage methods
for method, predict in zip(linkage_methods, cluster_predictions):
    plot_clusters(X, predict, method.capitalize())

# Define a function to plot dendrograms
def plot_dendrogram(X, method, title):
    dendrogram = sch.dendrogram(sch.linkage(X, method=method))
    plt.title(title)
    plt.xlabel("Data")
    plt.ylabel("Euclidean")
    plt.legend(['Cluster 1', 'Cluster 2', 'Cluster 3', 'Cluster 4'])
    plt.show()

# Plot dendrograms for different linkage methods
for method in linkage_methods:
    plot_dendrogram(X, method, method.capitalize())
