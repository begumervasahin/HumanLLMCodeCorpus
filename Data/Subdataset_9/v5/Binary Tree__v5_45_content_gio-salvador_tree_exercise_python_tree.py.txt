class TreeNode:
    def __init__(self, data):
        self.left = None
        self.right = None
        self.data = data

def create_tree1():
    tree = TreeNode(2)
    log.debug('TREE: %s' % tree.data)
    tree.left = TreeNode(8)
    log.debug('TREE.LEFT: %s' % tree.left.data)
    tree.left.left = TreeNode(1)
    log.debug('TREE.LEFT.LEFT: %s' % tree.left.left.data)
    tree.left.right = TreeNode(3)
    log.debug('TREE.LEFT.RIGHT: %s' % tree.left.right.data)
    tree.right = TreeNode(9)
    log.debug('TREE.RIGHT: %s' %  tree.right.data)
    tree.right.left = TreeNode(4)
    log.debug('TREE.RIGHT.LEFT: %s' % tree.right.left.data)
    tree.right.right = TreeNode(5)
    log.debug('TREE.RIGHT.RIGHT: %s' % tree.right.right.data)
    return tree

def create_tree2():
    tree = TreeNode(1)
    log.debug('TREE: %s' % tree.data)
    tree.left = TreeNode(8)
    log.debug('TREE.LEFT: %s' % tree.left.data)
    tree.left.right = TreeNode(3)
    log.debug('TREE.LEFT.RIGHT: %s' % tree.left.right.data)
    tree.right = TreeNode(4)
    log.debug('TREE.RIGHT: %s' % tree.right.data)
    tree.right.right = TreeNode(5)
    log.debug('TREE.RIGHT.RIGHT: %s' % tree.right.right.data)
    tree.right.right.right = TreeNode(7)
    log.debug('TREE.RIGHT.RIGHT.RIGHT: %s' % tree.right.right.right.data)
    return tree

def max_depth(node):
    if node is None:
        return 0
    else:
        left_depth = max_depth(node.left)
        right_depth = max_depth(node.right)
        return max(left_depth, right_depth) + 1

def get_node_depth(node, level=0, result=[]):
    if node is None:
        return
    else:
        get_node_depth(node.left, level + 1, result)
        get_node_depth(node.right, level + 1, result)
        result.append([level, node.data])
        return result

def sort_by_level(node):
    result = []
    depth = max_depth(node)
    tuples = get_node_depth(node, result=[])
    for d in range(depth):
        for r in range(len(tuples)):
            if tuples[r][0] == d:
                result.append(tuples[r][1])
    return result
