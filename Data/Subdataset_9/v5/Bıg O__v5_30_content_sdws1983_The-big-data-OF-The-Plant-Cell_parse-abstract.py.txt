from bs4 import BeautifulSoup
import urllib2
import re
import sys

def get_html(url):
    """
    Retrieve HTML content from the specified URL.
    """
    headers = {
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; WOW64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/47.0.2526.80 Safari/537.36',
        'Accept': '*/*',
        'Connection': 'keep-alive',
        'Host': 'www.plantcell.org'
    }
    req = urllib2.Request(url, headers=headers)
    response = urllib2.urlopen(req)
    html = response.read().decode('utf-8')
    return html

def analyse(html):
    """
    Extract content, authors, and addresses from the HTML.
    """
    soup = BeautifulSoup(html, 'lxml')
    contents = []
    author_list = []
    address_list = []

    # Extract content from paragraphs
    for paragraph in soup.find_all('p'):
        try:
            if u"p-" in str(paragraph['id']):
                content = re.sub(r'<.*?>', '', str(paragraph)[(str(paragraph).find(">") + 1):(str(paragraph).find("</p>", str(paragraph).find(">") + 1))])
                content = re.sub(r'\n', ' ', str(content))
                content = re.sub(r' +', ' ', str(content))
                if len(content) > 250:
                    contents.append(content)
        except:
            pass

    count = 1
    # Extract authors and addresses
    for item in soup.find_all('li'):
        try:
            if 'last' in item['class'] and u'name' in str(item):
                author = item.find_all('a')[0].string
                author_list.append(author)
            elif 'aff' in item['class']:
                address = str(item.find_all('address')[0])
                address = re.sub(r'<.*?>', '', str(address))
                address = re.sub(r' +', ' ', str(address))
                address = re.sub(r'\n', '', str(address))
                if re.findall('[a-z]', address[0]):
                    address = address[1:]
                address = (str(count) + "\t" + address + "\n")
                address_list.append(address)
                count += 1
        except:
            pass

    if len(contents) > 2:
        print("Content error")
        contents = []
        address_list = []
        author_list = []

    return ''.join(contents), ''.join(author_list), ''.join(address_list)

def main(vol, page):
    """
    Main function to fetch HTML, analyse it, and write results to CSV files.
    """
    url = f"http://www.plantcell.org/content/28/{vol}/{page}.abstract"
    html = get_html(url)
    content, author, address = analyse(html)

    with open(f"{vol}-content.csv", 'a') as f1, \
         open(f"{vol}-author.csv", 'a') as f2, \
         open(f"{vol}-address.csv", 'a') as f3:
        if content and content[-1] != "\n":
            content += "\n"
        if author and author[-1] != "\n":
            author += "\n"
        if address and address[-1] != "\n":
            address += "\n"
        f1.write(f">{page}\n{content}")
        f2.write(f">{page}\n{author}")
        f3.write(f">{page}\n{address}")
