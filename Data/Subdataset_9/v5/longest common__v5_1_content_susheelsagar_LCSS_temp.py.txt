# Open the file for reading and writing
with open("distance_matrix_1698.csv", "r+") as file:
    # Initialize an empty list to store data
    data = []
    
    # Counter for limiting the number of lines to read
    line_count = 0
    
    # Loop through each line in the file
    for line in file:
        # Check if we've read 5 lines already
        if line_count == 5:
            break
        line_count += 1
        
        # Split the line into a list using commas as separators
        row_data = line.split(",")
        
        # Update the data list with the row data
        data = row_data

# Calculate the length of the data list and add 1698 to it
total_length = len(data) + 1698

# Print the total length
print("Total length:", total_length)

# Print the value at index 2 in the data list
print("Value at index 2:", data[2])
