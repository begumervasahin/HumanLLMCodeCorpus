import pandas as pd
from sklearn.tree import DecisionTreeRegressor
from sklearn.model_selection import train_test_split

# Initialize a DecisionTreeRegressor with a fixed random state for reproducibility
regressor = DecisionTreeRegressor(random_state=0)

def load_dataset():
    """Load the dataset from a CSV file and print its shape."""
    dataset = pd.read_csv('AUDJPY_2.csv')
    print(f"The dataset contains {dataset.shape[0]} data points with {dataset.shape[1]} variables each.")
    return dataset

def extract_features(data):
    """Extract features from the dataset."""
    features = data.drop(['Date', 'Time', 'Close', 'High', 'Low', 'Total Ticks'], axis=1)
    return features

def extract_close_prices(data):
    """Extract close prices from the dataset."""
    close_prices = data['Close']
    return close_prices

def predict_price(open_price):
    """
    Predict whether the future price will be higher or lower than the given open price.
    
    Args:
    - open_price: The current open price.
    
    Returns:
    - Prediction: 'Up' if the predicted price is higher, 'Down' otherwise.
    """
    # Load dataset
    dataset = load_dataset()
    
    # Extract features and close prices
    features = extract_features(dataset)
    close_prices = extract_close_prices(dataset)
    
    # Split the dataset into training and testing sets
    X_train, X_test, y_train, y_test = train_test_split(features, close_prices, test_size=0.4, shuffle=True, random_state=101)
    
    # Fit the regression model on the training data
    regressor.fit(X_train, y_train)
    
    # Make a prediction for the given open price
    open_price_array = [[open_price]]  # Convert to 2D array
    predicted_price = regressor.predict(open_price_array)
    
    # Determine the prediction result
    if predicted_price > open_price:
        return "Up"
    else:
        return "Down"

# Perform prediction for a given open price
predicted_direction = predict_price(85.226)
print("Predicted Direction:", predicted_direction)
