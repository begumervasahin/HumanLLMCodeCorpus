class Node:
    def __init__(self, value):
        self.left = None     # Initialize left child node
        self.right = None    # Initialize right child node
        self.parent = None   # Initialize parent node
        self.value = value   # Value of the node

    def add_left(self, child_node):
        """Add a left child node."""
        self.left = child_node   # Set the left child node
        child_node.parent = self  # Set the parent of the left child

    def add_right(self, child_node):
        """Add a right child node."""
        self.right = child_node  # Set the right child node
        child_node.parent = self  # Set the parent of the right child

    def print_left(self):
        """Print the left child node."""
        print("Left child:", self.left)

    def print_right(self):
        """Print the right child node."""
        print("Right child:", self.right)

    def print_value(self):
        """Print the value of the node."""
        print("Node value:", self.value)

    def print_parent(self):
        """Print the parent node."""
        print("Parent node:", self.parent)
