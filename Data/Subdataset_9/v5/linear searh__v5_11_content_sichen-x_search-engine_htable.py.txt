# Function to create a hashtable with a specified number of buckets   
def create_hashtable(nbuckets):
    """Return a list of nbuckets lists representing a hashtable"""    
    return [[] for _ in range(nbuckets)]

# Function to generate a hashcode for strings and integers
def generate_hashcode(o):
    """
    Return a hashcode for strings and integers; return None for other types
    For integers, return the integer value.
    For strings, perform operation h = h*31 + ord(c) for all characters in the string
    """
    if isinstance(o, int):
        return o
    elif isinstance(o, str):
        h = 0
        for char in o:
            h = h * 31 + ord(char)
        return h
    else:
        return None

# Function to insert a key-value pair into the hashtable
def insert_entry(table, key, value):
    """
    Insert a (key,value) pair into the hashtable.
    If the key already exists, update its value.
    """
    if table is None or len(table) == 0:
        return

    bucket_index = generate_hashcode(key) % len(table)
    bucket = table[bucket_index]

    for i, (existing_key, existing_value) in enumerate(bucket):
        if existing_key == key:
            bucket[i] = (key, existing_value | value)
            return

    bucket.append((key, value))

# Function to retrieve the value associated with a given key from the hashtable
def get_value(table, key):
    """
    Retrieve the value associated with the given key from the hashtable.
    Return None if the key is not found.
    """
    if table is None or len(table) == 0:
        return None

    bucket_index = generate_hashcode(key) % len(table)
    bucket = table[bucket_index]

    for entry_key, entry_value in bucket:
        if entry_key == key:
            return entry_value

    return None

# Function to represent the buckets of the hashtable as a string      
def hashtable_buckets_str(table):
    """
    Return a string representation of the buckets of the hashtable.   
    Each line corresponds to a bucket and its contents.
    """
    if table is None:
        return None

    output = ""

    for i, bucket in enumerate(table):
        output += str(i).zfill(4) + '->'
        for entry_key, entry_value in bucket:
            output += str(entry_key) + ':' + str(entry_value) + ', '        
        output = output.rstrip(', ') + '\n'

    return output

# Function to represent the hashtable as a string
def hashtable_str(table):
    """
    Return a string representation of the hashtable similar to a Python dict.
    """
    if table is None:
        return None

    output = '{'

    for bucket in table:
        for entry_key, entry_value in bucket:
            output += str(entry_key) + ':' + str(entry_value) + ', '        

    output = output.rstrip(', ') + '}'

    return output
