import numpy as np
import math
from scipy.stats import norm

# Function to calculate the minimum value in a given array
def get_min(arr):
    return min(arr)

# Function to calculate the maximum value in a given array
def get_max(arr):
    return max(arr)

# The function f(x) = x^3 + 3x^2 + 3x + 1
def f(x):
    return x**3 + 3*x**2 + 3*x + 1

# Function to calculate the mean of f(x) values for a given array
def calculate_mean_f(arr):
    return np.mean([f(x) for x in arr])

# Function to calculate the variance of f(x) values for a given array
def calculate_variance_f(arr, f_mean):
    return np.var([f(x) for x in arr], ddof=1)

# Loading data from CSV files
data_uniform = np.genfromtxt('RandUnif.csv', delimiter='')
data_normal = np.genfromtxt('RandDist.csv', delimiter='')

# Constants
bottom_uniform = 2
top_uniform = 4
N = len(data_uniform)
confidence_probability = 0.95
k_value = norm.ppf(confidence_probability)

# Calculating mean and variance for uniform distribution data
mean_uniform = calculate_mean_f(data_uniform)
variance_uniform = calculate_variance_f(data_uniform, mean_uniform)
scaled_variance_uniform = variance_uniform / N

# Calculating confidence interval for uniform distribution data
lower_uniform = (top_uniform - bottom_uniform) * (mean_uniform - (k_value * scaled_variance_uniform))
upper_uniform = (top_uniform - bottom_uniform) * (mean_uniform + (k_value * scaled_variance_uniform))
print(f'Uniform Distribution Mean-Variance Result (Section A):\n[{lower_uniform}, {upper_uniform}]')

# Constants for normal distribution data
bottom_normal = get_min(data_normal)
top_normal = get_max(data_normal)

# Calculating mean and variance for normal distribution data
mean_normal = calculate_mean_f(data_normal)
variance_normal = calculate_variance_f(data_normal, mean_normal)
scaled_variance_normal = variance_normal / N

# Calculating confidence interval for normal distribution data
lower_normal = (top_normal - bottom_normal) * (mean_normal - (k_value * scaled_variance_normal))
upper_normal = (top_normal - bottom_normal) * (mean_normal + (k_value * scaled_variance_normal))
print(f'Normal Distribution Mean-Variance Result (Section B):\n[{lower_normal}, {upper_normal}]')
