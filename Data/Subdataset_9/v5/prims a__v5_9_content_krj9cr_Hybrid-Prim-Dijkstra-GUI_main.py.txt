import sys

# Importing Tkinter module with appropriate version handling
if sys.version_info[0] < 3:
    import Tkinter as tk
else:
    import tkinter as tk

# Importing the App module
from App import App

# Function to initialize and run the application
def run_application():
    # Creating a Tkinter root window
    root = tk.Tk()
    root.wm_title("Prim & Dijkstra's Algorithms")  # Setting window title

    # Creating an instance of the App class
    app_instance = App(root)

    # Starting the Tkinter event loop
    root.mainloop()

# Checking if the script is being run directly
if __name__ == "__main__":
    # Calling the run_application function to start the program
    run_application()
