import time

def find_perfect_numbers(limit):
    """
    Finds all perfect numbers below a given limit.
    
    Args:
    - limit: The upper limit for searching perfect numbers (exclusive).
    
    Returns:
    - A list of all perfect numbers below the specified limit.
    """
    return [number for number in range(1, limit) if is_perfect(number)]

def is_perfect(number):
    """
    Checks if a number is perfect. A number is perfect if it is equal to the sum of its proper divisors.
    
    Args:
    - number: The number to check.
    
    Returns:
    - True if the number is perfect, False otherwise.
    """
    if number < 2:
        return False  # Early return for 1, which is not a perfect number
    divisor_sum = 1  # Start with 1 because it is a proper divisor of all numbers
    for divisor in range(2, int(number**0.5) + 1):  # Only go up to the square root
        if number % divisor == 0:
            divisor_sum += divisor
            if divisor != number // divisor:  # Add the complement if it's not the square root
                divisor_sum += number // divisor
    return divisor_sum == number

if __name__ == "__main__":
    start_time = time.perf_counter()
    
    # Find and print perfect numbers below 10000
    perfect_numbers_below_10000 = find_perfect_numbers(10000)
    for number in perfect_numbers_below_10000:
        print(number)
    
    # Calculate and print the computation time
    end_time = time.perf_counter()
    print(f"Calculation took {end_time - start_time:.2f} seconds.")
