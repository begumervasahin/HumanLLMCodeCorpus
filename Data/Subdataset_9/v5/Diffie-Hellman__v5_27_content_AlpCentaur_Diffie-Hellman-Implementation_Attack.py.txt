# Function to generate shared key
def generate_shared_key(exponent, public_key_victim):
    shared_secret_victim = '{:x}'.format(pow(public_key_victim, exponent, PRIME))
    hash_victim = hashlib.sha512(shared_secret_victim.encode())
    key_victim = hash_victim.hexdigest()
    return key_victim

# Class for Diffie-Hellman Attack
class DiffieHellmanAttack:
    def __init__(self):
        self.prime = PRIME
        self.generator = GENERATOR

    # Method for performing the birthday attack
    def birthday_attack(self, public_key_victim1, public_key_victim2, num_cpus, collision_number):      
        iteration_number = num_cpus * 2

        # Function to generate shared key for public_key_victim1
        def generate_shared_key_victim1(x):
            return generate_shared_key(x, public_key_victim1)

        # Function to generate shared key for public_key_victim2
        def generate_shared_key_victim2(x):
            return generate_shared_key(x, public_key_victim2)

        # Lists to store generated keys
        compare_table_victim1 = []
        compare_table_victim2 = []
        i = 1

        # Loop for performing the attack
        while True:
            i += iteration_number
            compare_table_victim1.append(generate_shared_key_victim1(i))
            compare_table_victim2.append(generate_shared_key_victim2(i + 1))
            l = len(compare_table_victim2)

            # Check for collision
            for n in range(l - 1):
                if compare_table_victim1[n][:collision_number] == compare_table_victim2[l - 1][:collision_number]:
                    out11 = compare_table_victim1[n]
                    out12 = compare_table_victim2[l - 1]
                    outnum1 = (n + 1) * 2 + 1
                    outnum2 = (l) * 2 + 2
                    return outnum1, outnum2

            # Check for collision in the reverse order
            for m in range(l - 1):
                if compare_table_victim2[m][:collision_number] == compare_table_victim1[l - 1][:collision_number]:
                    out21 = compare_table_victim1[l - 1]
                    out22 = compare_table_victim2[m]
                    outnum2 = (m + 1) * 2 + 2
                    outnum1 = (l) * 2 + 1
                    return outnum1, outnum2
