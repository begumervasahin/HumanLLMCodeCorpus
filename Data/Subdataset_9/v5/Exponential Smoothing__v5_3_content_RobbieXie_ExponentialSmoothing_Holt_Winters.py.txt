def exponential_smoothing(alpha, s):
    """
    Apply exponential smoothing to the given time series.

    Parameters:
        alpha (float): Smoothing factor.
        s (numpy.ndarray): Time series data.

    Returns:
        numpy.ndarray: Exponentially smoothed time series.
    """
    smoothed_series = np.zeros(s.shape)
    smoothed_series[0] = s[0]

    for i in range(1, len(s)):
        smoothed_series[i] = alpha * s[i] + (1 - alpha) * smoothed_series[i - 1]

    return smoothed_series

def predict_double_exponential(s_single, s_double, alpha):
    """
    Predict values using double exponential smoothing.

    Parameters:
        s_single (numpy.ndarray): Single exponentially smoothed series.
        s_double (numpy.ndarray): Double exponentially smoothed series.
        alpha (float): Smoothing factor.

    Returns:
        numpy.ndarray: Predicted values using double exponential smoothing.
    """
    a_double = 2 * s_single - s_double
    b_double = (alpha / (1 - alpha)) * (s_single - s_double)
    return a_double + b_double

def predict_triple_exponential(s_single, s_double, s_triple, alpha):
    """
    Predict values using triple exponential smoothing.

    Parameters:
        s_single (numpy.ndarray): Single exponentially smoothed series.
        s_double (numpy.ndarray): Double exponentially smoothed series.
        s_triple (numpy.ndarray): Triple exponentially smoothed series.
        alpha (float): Smoothing factor.

    Returns:
        numpy.ndarray: Predicted values using triple exponential smoothing.
    """
    a_triple = 3 * s_single - 3 * s_double + s_triple
    b_triple = (alpha / (2 * ((1 - alpha) ** 2))) * (
                (6 - 5 * alpha) * s_single - 2 * ((5 - 4 * alpha) * s_double) + (4 - 3 * alpha) * s_triple)
    c_triple = ((alpha ** 2) / (2 * ((1 - alpha) ** 2))) * (s_single - 2 * s_double + s_triple)
    return a_triple + b_triple + c_triple

def extend_year_array(year, pre_year):
    """
    Extend the year array to include prediction years.

    Parameters:
        year (numpy.ndarray): Original years.
        pre_year (numpy.ndarray): Years for which predictions are made.

    Returns:
        numpy.ndarray: Extended year array.
    """
    return np.insert(year, len(year), values=pre_year, axis=0)

def extend_predicted_array(s_pre, last_values, alpha, prediction_years):
    """
    Extend the predicted array to include values for prediction years.

    Parameters:
        s_pre (numpy.ndarray): Predicted values array.
        last_values (numpy.ndarray): Last values of the smoothed series.
        alpha (float): Smoothing factor.
        prediction_years (int): Number of years to predict.

    Returns:
        numpy.ndarray: Extended array with predicted values.
    """
    extension = np.array([last_values[-1] + alpha * i for i in range(1, prediction_years + 1)])
    return np.insert(s_pre, len(s_pre), values=extension, axis=0)
