def selection_sort(num_list):
    """
    Sorts the input list of numbers using the selection sort algorithm.

    Parameters:
        num_list (list): List of numbers to be sorted.
    """
    sorted_list = []
    while num_list:
        # Find the minimum value in the remaining unsorted part of the list
        min_value = min(num_list)
        # Append the minimum value to the sorted list
        sorted_list.append(min_value)
        # Remove the minimum value from the original list
        num_list.remove(min_value)
        # Print the current step
        print('\nStep', len(sorted_list))
        print('Unsorted list:', num_list)
        print('Sorted list:', sorted_list)
    # Print the final sorted list and any remaining elements in the original list
    print('\nFinish:', num_list)
    print('Sorted list:', sorted_list)

numbers = [2, 1, 5, 3, 4, 7, 6, 9, 8, 10]
selection_sort(numbers)
