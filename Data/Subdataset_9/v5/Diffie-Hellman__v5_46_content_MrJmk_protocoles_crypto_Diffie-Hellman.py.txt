import random

def is_prime(number):
    """Check if a number is prime."""
    if number <= 1:
        return False
    if number <= 3:
        return True
    if number % 2 == 0 or number % 3 == 0:
        return False
    for i in range(5, int(number ** 0.5) + 1, 6):
        if number % i == 0 or number % (i + 2) == 0:
            return False
    return True

def generate_prime(lower_bound, upper_bound):
    """Generate a prime number between lower_bound and upper_bound."""
    prime = 0
    while not is_prime(prime):
        prime = random.randint(lower_bound, upper_bound)
    return prime

def main():
    """Main function to demonstrate the Rabin-Miller Algorithm."""
    print("---------------------------------")
    print("----------ALGORITHME R-H---------")
    print("---------------------------------")

    # Generate a prime number p
    p = generate_prime(1000, 99999)
    print("Generated prime number P:", p)

    # Generate a random number a between 1 and p-1
    a = random.randint(1, p - 1)
    print("Chosen number A (1 < A < P-1):", a)

    # Generate private keys x1 and x2 for Alice and Bob
    x1 = random.randint(0, 99999)
    x2 = random.randint(0, 99999)
    print("\nPrivate key x1 (Alice):", x1)
    print("Private key x2 (Bob):", x2)

    # Calculate public keys y1 and y2
    y1 = pow(a, x1, p)
    y2 = pow(a, x2, p)

    # Calculate shared secrets k1 and k2
    k1 = pow(y2, x1, p)
    k2 = pow(y1, x2, p)

    print("\nShared secret key K1 (Alice):", k1)
    print("Shared secret key K2 (Bob):", k2)

if __name__ == "__main__":
    main()
