import graphviz as gv

class Edge:
    """Class representing an edge in a graph."""
    def __init__(self, start_node, end_node, capacity):
        self.start_node = start_node
        self.end_node = end_node
        self.capacity = capacity
        self.flow = 0

class Graph:
    """Class representing a graph."""
    def __init__(self):
        self.edges = []

    def add_edge(self, start, end, capacity):
        """Add an edge to the graph."""
        self.edges.append(Edge(start, end, capacity))
        return self

    def list_nodes(self):
        """List all nodes in the graph."""
        nodes = set()
        for edge in self.edges:
            nodes.add(edge.start_node)
            nodes.add(edge.end_node)
        return list(nodes)

    def get_neighbors(self, node):
        """Get neighbors of a node."""
        neighbors = []
        for edge in self.edges:
            if edge.start_node == node:
                neighbors.append(edge)
        return neighbors

    def print_edges(self):
        """Print edge details."""
        edge_list = []
        for edge in self.edges:
            edge_list.append([edge.start_node, edge.end_node, edge.flow, edge.capacity])
        return edge_list

    def find_path(self, start_node, end_node):
        """Find a path between two nodes."""
        predecessors = {node: 0 for node in self.list_nodes()}
        visited_nodes = [start_node]

        while visited_nodes:
            current_node = visited_nodes.pop(0)
            for edge in self.get_neighbors(current_node):
                if edge.capacity != edge.flow:
                    predecessors[edge.end_node] = edge.start_node
                    if edge.end_node == end_node:
                        path = []
                        node = end_node
                        while node != start_node:
                            for e in self.edges:
                                if e.end_node == node and e.start_node == predecessors[node]:
                                    path.insert(0, e)
                                    node = e.start_node
                                    break
                        max_flow = min(edge.capacity - edge.flow for edge in path)
                        for e in path:
                            e.flow += max_flow
                        return [path, max_flow]
                    visited_nodes.append(edge.end_node)
        return None

def main():
    graph = Graph()
    graph.add_edge('Start', 'B', 5)\
         .add_edge('Start', 'C', 4)\
         .add_edge('C', 'B', 6)\
         .add_edge('B', 'E', 4)\
         .add_edge('C', 'E', 4)\
         .add_edge('E', 'Sink', 7)\
         .add_edge('C', 'Sink', 4)

    paths = []
    total_max_flow = 0

    for _ in range(3):
        path = graph.find_path('Start', 'Sink')
        if path:
            paths.append(path)
            total_max_flow += path[1]

    drawing = gv.Digraph(format='png')
    for edge in graph.print_edges():
        drawing.edge(str(edge[0]), str(edge[1]), '0/%s' % str(edge[3]), color='black')

    drawing.render(view=True)

if __name__ == "__main__":
    main()
