def longest_common_subsequence(selector, i, j):
    if i < 0 or j < 0:
        if selector == 'check':
            return answer[::-1], len(answer)
        return 0

    if selector == 'check':
        while i >= 0 and second.find(first[i], 0, j + 1) == -1:
            i -= 1
        while j >= 0 and first.find(second[j], 0, i + 1) == -1:
            j -= 1
        if not is_match(i, j):
            inear = longest_common_subsequence('row', i - 1, j)
            jnear = longest_common_subsequence('col', i, j - 1)
            inear = inear if inear != 'nf' else i
            jnear = jnear if jnear != 'nf' else j
            if inear == i and jnear == j:
                longest_common_subsequence('check', i - 1, j - 1)
            elif inear == i and jnear != j:
                longest_common_subsequence('check', i, jnear)
            elif inear != i and jnear == j:
                longest_common_subsequence('check', inear, j)
            else:
                longest_common_subsequence('check', inear, j) if (inear + 1) * (j + 1) > (jnear + 1) * (i + 1) else \
                    longest_common_subsequence('check', i, jnear)
        else:
            answer.append(first[i])
            longest_common_subsequence('check', i - 1, j - 1)
    elif selector == 'row':
        if i == 0 and not is_match(i, j):
            return 'nf' if not is_match(i, j) else 0
        return i if is_match(i, j) else longest_common_subsequence('row', i - 1, j)
    elif selector == 'col':
        if j == 0:
            return 'nf' if not is_match(i, j) else 0
        return j if is_match(i, j) else longest_common_subsequence('col', i, j - 1)

    return first[i] == second[j]
