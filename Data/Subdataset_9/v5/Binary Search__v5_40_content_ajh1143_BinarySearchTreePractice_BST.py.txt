class Node:
    def __init__(self, value):
        self.value = value
        self.left_child = None
        self.right_child = None

    def insert(self, data):
        """
        Insert a new node with the given data value.

        :param data: The value to insert.
        :return: True if insertion is successful, False otherwise.
        """
        if self.value == data:
            return False
        elif self.value > data:
            if self.left_child:
                return self.left_child.insert(data)
            else:
                self.left_child = Node(data)
                return True
        else:
            if self.right_child:
                return self.right_child.insert(data)
            else:
                self.right_child = Node(data)
                return True

    def find(self, data):
        """
        Check if the given data value exists in the tree.

        :param data: The value to search for.
        :return: True if the value exists, False otherwise.
        """
        if self.value == data:
            return True
        elif self.value > data and self.left_child:
            return self.left_child.find(data)
        elif self.right_child:
            return self.right_child.find(data)
        return False

    def preorder(self):
        """
        Perform a preorder traversal starting from this node.
        """
        if self:
            print(str(self.value))
            if self.left_child:
                self.left_child.preorder()
            if self.right_child:
                self.right_child.preorder()

    def inorder(self):
        """
        Perform an inorder traversal starting from this node.
        """
        if self:
            if self.left_child:
                self.left_child.inorder()
            print(str(self.value))
            if self.right_child:
                self.right_child.inorder()

    def postorder(self):
        """
        Perform a postorder traversal starting from this node.
        """
        if self:
            if self.left_child:
                self.left_child.postorder()
            if self.right_child:
                self.right_child.postorder()
            print(str(self.value))


class Tree:
    def __init__(self):
        self.root = None

    def insert(self, data):
        """
        Insert a new node with the given data value into the tree.

        :param data: The value to insert.
        """
        if self.root:
            return self.root.insert(data)
        else:
            self.root = Node(data)

    def find(self, data):
        """
        Check if the given data value exists in the tree.

        :param data: The value to search for.
        :return: True if the value exists, False otherwise.
        """
        if self.root:
            return self.root.find(data)
        return False

    def preorder(self):
        """
        Perform a preorder traversal of the tree.
        """
        print("Preorder")
        self.root.preorder()

    def inorder(self):
        """
        Perform an inorder traversal of the tree.
        """
        print("Inorder")
        self.root.inorder()

    def postorder(self):
        """
        Perform a postorder traversal of the tree.
        """
        print("Postorder")
        self.root.postorder()


bst = Tree()
for value in [1, 2, 3]:
    bst.insert(value)

bst.preorder()
bst.postorder()
