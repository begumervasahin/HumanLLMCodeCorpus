import time
import numpy as np
import matplotlib.pyplot as plt
from random import randrange
from Selection_Algorithm import *
from Linkedlist import Linkedlist

def draw_comparison_graph(timings, sizes):
    plt.figure(figsize=(13, 6))
    plt.gcf().canvas.set_window_title('Comparison')
    
    plt.subplot(2, 1, 1)
    plt.grid(True)
    plt.title("Comparison")
    plt.xlabel("Size of Inputs")
    plt.ylabel("Time (seconds)")
    colors = ['r', 'b', 'c']
    methods = ['List', 'Linked List (Swap Nodes)', 'Linked List (Swap Data)']
    for i in range(len(timings)):
        plt.plot(sizes, timings[i], colors[i], label=methods[i])
    plt.legend()
    plt.axis([0, 1500, 0, 6])
    
    plt.subplot(2, 1, 2)
    plt.grid(True)
    plt.xlabel("Size of Inputs")
    plt.ylabel("Time (seconds)")
    for i in range(len(timings)):
        plt.bar(sizes, timings[i], 50, color=colors[i], align='center', label=methods[i])
    plt.legend()
    plt.show()

def generate_random_array(size):
    return [randrange(0, 1000000) for _ in range(size)]

def generate_linked_list(arr):
    linked_list = Linkedlist()
    for item in arr:
        linked_list.add(item)
    return linked_list

def measure_time(func, *args):
    start_time = time.time()
    func(*args)
    end_time = time.time()
    return end_time - start_time

def measure_and_store_time(method, arr, linked_list, sizes):
    timings = []
    for size in sizes:
        arr_slice = arr[:size]
        linked_list_copy = linked_list.copy()
        if method == 'List':
            timings.append(measure_time(Selection_Sort_Array, arr_slice))
        elif method == 'Linked List (Swap Nodes)':
            timings.append(measure_time(Selection_Sort_Linked_List, linked_list_copy))
        elif method == 'Linked List (Swap Data)':
            timings.append(measure_time(selectionsort_linked, linked_list_copy))
    return timings

def automatic_comparison():
    sizes = np.logspace(1, 4, num=4, dtype=int)
    arr = generate_random_array(max(sizes))
    linked_list = generate_linked_list(arr)
    timings = []
    methods = ['List', 'Linked List (Swap Nodes)', 'Linked List (Swap Data)']
    for method in methods:
        timings.append(measure_and_store_time(method, arr, linked_list, sizes))
    draw_comparison_graph(timings, sizes)

def help_manual_comparison():
    size = 10000
    arr = generate_random_array(size)
    linked_list = generate_linked_list(arr)
    timings = []
    methods = ['List', 'Linked List (Swap Nodes)', 'Linked List (Swap Data)']
    for method in methods:
        timings.append(measure_and_store_time(method, arr, linked_list, [size])[0])
    draw_comparison_graph([timings], [size])

if __name__ == '__main__':
    automatic_comparison()
