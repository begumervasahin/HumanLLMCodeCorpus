# Function to calculate the greatest common divisor
def gcd(a, b):
    while b > 0:
        a, b = b, a % b
    return a

# Function to find the next prime number
def next_prime(p):
    while p % 12 != 11:
        p += 1
    return next_prime_step(p)

# Helper function for next_prime
def next_prime_step(p):
    m_ = 5 * 7 * 11 * 13 * 17 * 19 * 23 * 29 * 31
    while True:
        if gcd(p, m_) == 1 and gcd((p - 1) // 2, m_) == 1:
            if pow(7, p - 1, p) == 1 and pow(7, (p - 3) // 2, (p - 1) // 2) == 1:
                return p
        p += 12

# Calculate next prime number
p = next_prime(p)

px = (p - 1) // 2
