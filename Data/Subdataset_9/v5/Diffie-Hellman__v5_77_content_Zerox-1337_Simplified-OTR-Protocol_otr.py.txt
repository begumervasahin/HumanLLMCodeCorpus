import hashlib

def extEuclideanAlg(a, b):
    """Extended Euclidean Algorithm for finding modular inverse."""
    if b == 0:
        return 1, 0, a
    else:
        x, y, gcd = extEuclideanAlg(b, a % b)
        return y, x - y * (a // b), gcd

def modInvEuclid(a, m):
    """Modular Inverse using Extended Euclidean Algorithm."""
    x, y, gcd = extEuclideanAlg(a, m)
    if gcd == 1:
        return x % m
    else:
        return None

def convert_to_utf(x):
    """Convert integer to UTF-8 bytes."""
    return x.to_bytes((x.bit_length() + 7) // 8, 'big')

def h(g):
    """Hash function using SHA-1."""
    passphrase = b'eitn41 <3'
    g_bytes = convert_to_utf(g)
    return hashlib.sha1(g_bytes + passphrase).hexdigest()
