# Define the lookup table keys and their corresponding ranges
lookup_table_keys = [
    (0, 12), (13, 37), (38, 62), (63, 87), (88, 112),
    (113, 137), (138, 162), (163, 187), (188, 212),
    (213, 237), (238, 9999)
]

# Define the lookup table with ratings and expected results
lookup_table = {
    'upset': {
        key_range: value for key_range, value in zip(lookup_table_keys, [8, 10, 13, 16, 20, 25, 30, 35, 40, 45, 50])
    },
    'expected_result': {
        key_range: value for key_range, value in zip(lookup_table_keys, [8, 7, 6, 5, 4, 3, 2, 2, 1, 1, 0])
    }
}

def get_adjustment(winner_rating, loser_rating):
    """
    Calculate the adjustment based on winner and loser ratings.

    Args:
        winner_rating (int): The rating of the winner.
        loser_rating (int): The rating of the loser.

    Returns:
        int: The adjustment value based on the score differential.
    """
    # Determine which lookup table to use based on the winner and loser ratings
    table_key = 'upset' if winner_rating < loser_rating else 'expected_result'
    
    # Calculate the score differential between winner and loser ratings
    score_differential = abs(winner_rating - loser_rating)
    
    # Iterate through the lookup table to find the adjustment value based on the score differential
    for key_range, value in lookup_table[table_key].items():
        if score_differential in range(key_range[0], key_range[1] + 1):
            return value
