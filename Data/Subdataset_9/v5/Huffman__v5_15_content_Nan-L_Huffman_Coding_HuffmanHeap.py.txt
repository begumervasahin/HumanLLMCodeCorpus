class HuffmanHeap:
    """
    A priority queue implementation for Huffman trees.
    """

    def __init__(self, old=None, new=None):
        """
        Initialize the HuffmanHeap.

        Args:
            old (list): The old list of Huffman trees, sorted. Defaults to an empty list.
            new (list): The new list of Huffman trees, sorted. Defaults to an empty list.
        """
        self.old = old or []
        self.new = new or []

    def enqueue(self, item):
        """
        Add a Huffman tree to the new list.

        Args:
            item: A Huffman tree to be added.
        """
        self.new.append(item)

    def dequeue(self):
        """
        Remove and return the smallest item from the heap.

        Returns:
            The smallest item. If the heap is empty, returns None.
        """
        if not self.old and not self.new:
            print("Both old and new lists are empty!")
            return None

        if not self.old:
            return self.new.pop(0)

        if not self.new:
            return self.old.pop(0)

        if self.old[0].get_freq() >= self.new[0].get_freq():
            return self.new.pop(0)
        else:
            return self.old.pop(0)
