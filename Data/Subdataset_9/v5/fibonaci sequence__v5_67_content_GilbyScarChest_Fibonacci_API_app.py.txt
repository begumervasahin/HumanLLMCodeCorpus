from flask import Flask

app = Flask(__name__)

# Global variable to keep track of the current Fibonacci number index
current_index = 0

# Function to calculate the nth Fibonacci number recursively
def fibonacci(n: int) -> int:
    if n < 1:
        return 0
    elif n == 1:
        return 1
    else:
        return fibonacci(n - 1) + fibonacci(n - 2)

# Route for the home page
@app.route("/")
def home():
    return (
        "Available Routes:<br/>"
        "/previous<br/>"
        "/current<br/>"
        "/next<br/>"
    )

# Route for the current Fibonacci number
@app.route("/current")
def current_fib():
    global current_index
    return str(fibonacci(current_index))

# Route for the next Fibonacci number
@app.route("/next")
def next_fib():
    global current_index
    current_index += 1
    return str(fibonacci(current_index))

# Route for the previous Fibonacci number
@app.route("/previous")
def prev_fib():
    global current_index
    if current_index > 0:
        current_index -= 1
    return str(fibonacci(current_index))

if __name__ == '__main__':
    app.run()
