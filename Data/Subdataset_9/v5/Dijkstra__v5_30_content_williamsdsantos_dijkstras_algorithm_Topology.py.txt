class Topology:
    def __init__(self, filename):
        self.nodes = []  # List to store node IDs
        self.links = []  # List to store link information

        # Open the topology file
        with open(filename, 'r') as topology_file:
            # Skip the header lines
            self._skip_header(topology_file, 3)

            # Read node data
            self._read_nodes(topology_file)

            # Skip the next set of header lines
            self._skip_header(topology_file, 3)

            # Read link data
            self._read_links(topology_file)

    def _skip_header(self, file_obj, num_lines):
        """Skip a specified number of header lines in the file."""
        for _ in range(num_lines):
            next(file_obj)

    def _read_nodes(self, file_obj):
        """Read node data from the file."""
        for line in file_obj:
            data = line.split()
            if not data:  # Check for empty lines
                break
            try:
                node_id = data[2]
                self.nodes.append(node_id)
            except IndexError:
                break  # Stop reading if unexpected data format is encountered

    def _read_links(self, file_obj):
        """Read link data from the file."""
        for line in file_obj:
            data = line.split()
            if not data:  # Check for empty lines
                break
            try:
                origin, destination, length = data[2:5]
                elem_link = [origin, destination, length]
                self.links.append(elem_link)
            except IndexError:
                break  # Stop reading if unexpected data format is encountered
