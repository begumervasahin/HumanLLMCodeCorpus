def find_encryption_key(phi, n):
    """
    Finds the encryption key (e) given Euler's totient function (phi) and n.
    """
    e = 2
    while e < phi:
        if gcd(e, phi) == 1:
            return e
        e += 1
    return None

def gcd(a, b):
    """
    Finds the greatest common divisor (gcd) of two numbers a and b.
    """
    while b != 0:
        a, b = b, a % b
    return a

def find_decryption_key(e, phi):
    """
    Finds the decryption key (d) given the encryption key (e) and Euler's totient function (phi).
    """
    d = 1
    while True:
        if (d * e) % phi == 1:
            return d
        d += 1

def encrypt_message(message, e, n, letters):
    """
    Encrypts the given message using the encryption key (e) and n.
    """
    encrypted_msg = []
    for char in message:
        encrypted_msg.append((letters.index(char)**e) % n)
    return encrypted_msg

def decrypt_message(encrypted_msg, d, n, letters):
    """
    Decrypts the given message using the decryption key (d) and n.
    """
    decrypted_msg = []
    for char in encrypted_msg:
        decrypted_msg.append(letters[(char**d) % n])
    return decrypted_msg

# Prompting the user to enter two prime numbers
print("Enter 2 PRIME numbers for 'p' and 'q'.")
p = int(input("p: "))
q = int(input("q: "))

# Calculating the product of the prime numbers
n = p * q
print("\nn = " + str(n))

# Calculating Euler's totient function (phi)
phi = (p - 1) * (q - 1)
print("phi = " + str(phi))

# Finding the encryption key (e)
e = find_encryption_key(phi, n)
if e is None:
    print("Failed to find encryption key.")
    exit()
print("e = " + str(e))
print("ENCRYPTION KEY= (" + str(e) + ", " + str(n) + ")")

# Finding the decryption key (d)
d = find_decryption_key(e, phi)
print("d = " + str(d))
print("DECRYPTION KEY= (" + str(d) + ", " + str(n) + ")")

# Setting up the alphabet for encryption and decryption
alphabet = ".abcdefghijklmnopqrstuvwxyz ABCDEFGHIJKLMNOPQRSTUVWXYZ"   
letters = list(alphabet)

# Encrypting the message
msg = input("\nEnter simple message (letters only): ")
encrypted_msg = encrypt_message(msg, e, n, letters)
print("\nEncrypted message: ")
print(encrypted_msg)

# Decrypting the message
decrypted_msg = decrypt_message(encrypted_msg, d, n, letters)
print("\nDecrypted Message: ")
print(''.join(decrypted_msg))
