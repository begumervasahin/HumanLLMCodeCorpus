import struct
import sys
from HUFFMAN.components import huffnode, lists, huffmantree, encodehufftree

def main():
    # Input and output file handling
    try:
        input_file = sys.argv[1]
        output_file = sys.argv[2] if len(sys.argv) > 2 else "compressed.huff"
    except IndexError:
        print("Usage: python compress.py <input_file> [<output_file>]")
        return

    # Read input file
    try:
        with open(input_file, 'rb') as file:
            inp = file.read()
            original_size = len(inp)
    except FileNotFoundError:
        print("Input file not found.")
        return

    # Initialize array to store frequency of each byte
    arr = [huffnode(i, 0) for i in range(0, 256)]
    for ch in inp:
        arr[ch].freq += 1

    # Sort array based on frequency
    arr.sort(key=lambda x: x.freq)

    # Build Huffman tree
    link = lists(arr[0])
    for a in arr[:0:-1]:
        if a.freq != 0:
            new = lists(a)
            new.next = link
            link = new
        else:
            break

    tree = build_huffman_tree(link)

    # Open output file for writing compressed data
    with open(output_file, 'wb') as comp:
        try:
            generate_encoded_data(tree, inp, comp)
            print("Compression completed successfully!")
        except Exception as e:
            print(f"An error occurred: {e}")

def build_huffman_tree(link):
    tree = 0
    while True:
        a = link.top()
        link = link.next
        b = link.top()
        link = link.next
        n = huffnode(257, a.freq + b.freq)
        n.right = a if a.freq > b.freq else b
        n.left = b if a.freq > b.freq else a
        new = lists(n)
        if link != 0:
            link = link.insert(new)
        else:
            tree = n
            break
    return tree

def generate_encoded_data(tree, inp, comp):
    # Generate address of characters corresponding to Huffman tree
    dictionary = {}
    huffmantree(tree, dictionary, "")
    encodehufftree(tree, 16)
    codedtree = combine(pt)
    size = len(codedtree)

    # Write size of the coded tree to the output file
    comp.write(struct.pack('BB', size // 256, size % 256))

    chunk = ""
    for ch in codedtree:
        chunk += ch
        if len(chunk) > 8:
            comp.write(struct.pack('B', int(chunk[:8], 2)))
            chunk = chunk[8:]

    df = ""
    for ch in inp:
        chunk += dictionary[ch]
        df += dictionary[ch]
        if len(chunk) > 8:
            comp.write(struct.pack('B', int(chunk[:8], 2)))
            chunk = chunk[8:]

    chunk += "1"
    while len(chunk) % 8 == 0:
        chunk += "0"

    while chunk:
        comp.write(struct.pack('B', int("0b" + chunk[:8], 2)))
        chunk = chunk[8:]

def combine(st):
    return ''.join(st)

if __name__ == "__main__":
    main()
