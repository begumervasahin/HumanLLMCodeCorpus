def bubble_sort(arr):
    # Flag to track if any swaps were made during a pass
    swapped = True

    # Continue looping until no swaps are made in a pass
    while swapped:
        swapped = False  # Reset swapped flag for each pass
        for i in range(len(arr) - 1):
            if arr[i] > arr[i + 1]:  # If the current element is greater than the next one
                # Swap the elements
                arr[i], arr[i + 1] = arr[i + 1], arr[i]
                swapped = True  # Set swapped flag to True as a swap was made

# Example usage:
if __name__ == "__main__":
    arr = [9, 8, 6, 6, 5, 4, 3, 2, 1, 0]
    bubble_sort(arr)
    print(arr)
