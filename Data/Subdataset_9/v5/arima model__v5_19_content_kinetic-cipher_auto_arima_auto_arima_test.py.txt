import numpy as np
import matplotlib.pyplot as plt
import auto_arima_model as am
import argparse

def get_cmd_line_args():
    parser = argparse.ArgumentParser()
    parser.add_argument("-f", "--forecast", help="perform forecasts and plot results", action="store_true")
    return parser.parse_args()

def generate_time_series():
    A = 1
    a = 0.35
    t_start = 0
    t_end = 60
    t_delta = 0.1
    noise_mean = 0
    noise_std = 1

    time = np.arange(t_start, t_end, t_delta)
    y = A * np.sin(time)
    noise = np.random.normal(noise_mean, noise_std, len(y))
    return time, np.add(y, a * noise)

def fit_and_forecast(arima_model, data, num_forecast_steps):
    arima_model.fit(data)
    return arima_model.forecast(num_steps=num_forecast_steps, conf_int=True)

def plot_forecast(ax, time, data, forecast, conf_int, color):
    ax.plot(time[:len(data)], data, 'b')
    ax.plot(time[len(data):], forecast, 'r')
    ax.fill_between(time[len(data):], conf_int[:, 0], conf_int[:, 1], alpha=0.3, color='b')
    ax.grid()
    ax.set(xlabel='Time Sample', ylabel='Amplitude')

def main():
    args = get_cmd_line_args()
    num_forecast_steps = 100

    print("Analyzing Noisy Sine...")
    time, yn = generate_time_series()
    arima_model = am.AutoArimaModel(start_p=1, max_p=3, start_q=1, max_q=3)
    model_args = arima_model.get_params()
    print(model_args)

    if args.forecast:
        Ypred1, conf_int1 = fit_and_forecast(arima_model, yn, num_forecast_steps)
        time_axis = np.arange(len(yn) + len(Ypred1))
    
    print("=====================")

    for title, data_func in [("Up-Ramp", lambda t: np.add(t, yn)),
                              ("Down-Ramp", lambda t: np.add(t_end - t, yn)),
                              ("Quadratic-Ramp", lambda t: np.add(0.01 * np.power(t, 2), yn)),
                              ("Cubic-Ramp", lambda t: np.add((1 / 5000) * np.power(t, 3), yn)),
                              ("Fourth-power Ramp", lambda t: np.add((1 / 200000) * np.power(t, 4), yn))]:

        print(f"Analyzing {title}...")
        y = data_func(time)
        arima_model.fit(y)
        model_args = arima_model.get_params()
        print(model_args)

        if args.forecast:
            forecast_data, conf_int = fit_and_forecast(arima_model, y, num_forecast_steps)

            fig, ax = plt.subplots()
            plot_forecast(ax, time_axis, y, forecast_data, conf_int, 'b')
            ax.set_title(title)
            plt.show()

if __name__ == "__main__":
    main()
