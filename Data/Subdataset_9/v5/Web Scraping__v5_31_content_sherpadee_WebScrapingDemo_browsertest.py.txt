from selenium import webdriver
import time
import os

def get_driver_path(browser_name):
    """Construct and return the driver path for the given browser."""
    extension = '.exe' if os.name == 'nt' else ''
    return f"{os.getcwd()}/{browser_name}driver{extension}"

def launch_browser(browser, url):
    """Launch the specified browser and navigate to the given URL."""
    driver_path = get_driver_path(browser)
    driver = webdriver.Chrome(executable_path=driver_path)
    driver.get(url)
    return driver

def take_full_page_screenshot(driver, filename):
    """Take a screenshot of the entire page and save it as the specified filename."""
    total_width = driver.execute_script("return document.body.offsetWidth")
    total_height = driver.execute_script("return document.body.scrollHeight")
    driver.set_window_size(total_width, total_height)
    driver.save_screenshot(filename)

def accept_cookies(driver):
    """Accept cookies by clicking on the OK button."""
    cookie_ok_button = driver.find_element_by_class_name('CallToAction.CallToAction--primary')
    cookie_ok_button.click()

# URLs
google_url = 'https://www.google.com'
job_board_url = 'https://nuernberg.digital/festival/jobboerse'

# Launch Firefox browser for Google
firefox_browser = launch_browser('gecko', google_url)
time.sleep(5)  # Wait for 5 seconds
firefox_browser.quit()  # Close the Firefox browser

# Launch Chrome browser for Google
chrome_browser = launch_browser('chrome', google_url)
time.sleep(5)  # Wait for 5 seconds
chrome_browser.quit()  # Close the Chrome browser

# Launch Chrome browser for Nürnberg Digital Festival job board
chrome_browser = launch_browser('chrome', job_board_url)
time.sleep(1)  # Wait for 1 second

# Take a screenshot of the entire page and save it
take_full_page_screenshot(chrome_browser, "nuedigital.png")

# Accept cookies
accept_cookies(chrome_browser)

# Take a screenshot of the job listings section and save it
job_listings_section = chrome_browser.find_element_by_class_name('JobList')
job_listings_section.screenshot('nuedigitaljobs.png')

# Close the Chrome browser
chrome_browser.quit()
