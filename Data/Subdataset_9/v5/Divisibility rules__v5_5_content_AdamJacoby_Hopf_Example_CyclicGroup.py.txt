from ExampleGroup_Functions import *
from HopfClass import HopfAlgebra
import numpy as np
import scipy.sparse as sps
from itertools import product

def create_cyclic_group_mult_matrix(dim):
    # Create a matrix representing the multiplication operation for a cyclic group
    mult = np.zeros((dim, dim**2), dtype=complex)
    for i, j in product(range(dim), repeat=2):
        mult[(i + j) % dim, i * dim + j] = 1
    # Convert the matrix to a sparse format for efficiency
    mult_sparse = sps.csr_matrix(mult, dtype=complex)
    return mult_sparse

def create_cyclic_group_antipode(dim):
    # Create the antipode matrix for a cyclic group
    antipode = np.zeros((dim, dim), dtype=complex)
    for i in range(dim):
        antipode[i, (-i) % dim] = 1
    # Convert the matrix to a sparse format for efficiency
    antipode_sparse = sps.csr_matrix(antipode, dtype=complex)
    return antipode_sparse

def generate_cyclic_group_element_names(dim, element_name):
    # Generate element names for a cyclic group
    element_names = [f"{element_name}^{i}" for i in range(dim)]
    return element_names

def create_cyclic_group(dim, element_name):
    # Construct a Hopf algebra for a cyclic group
    mult = create_cyclic_group_mult_matrix(dim)
    comult = Group_Comult_Matrix(dim)  # Assuming Group_Comult_Matrix is defined elsewhere
    counit = Group_Counit(dim)        # Assuming Group_Counit is defined elsewhere
    intg = Group_Integral(dim)        # Assuming Group_Integral is defined elsewhere
    antipode = create_cyclic_group_antipode(dim)
    name = f"C_{dim}"
    element_names = generate_cyclic_group_element_names(dim, element_name)
    # Create the Hopf algebra object
    cyclic_group = HopfAlgebra(name, element_names, mult, comult, counit, antipode)
    cyclic_group.Input_Integral(intg)
    return cyclic_group
