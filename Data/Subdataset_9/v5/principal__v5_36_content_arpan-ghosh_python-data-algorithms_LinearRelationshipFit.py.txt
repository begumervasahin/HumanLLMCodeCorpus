import pandas as pd
import matplotlib.pyplot as plt
import numpy as np
from scipy.stats import linregress

# Function to read data from CSV files
def read_data(file_path):
    colnames = ['x', 'y']
    return pd.read_csv(file_path, names=colnames, header=None)

# Function to plot scatter plots for each dataset
def plot_scatter(datasets, titles):
    fig, axs = plt.subplots(1, len(datasets), figsize=(16, 8), sharey=True)
    for i, data in enumerate(datasets):
        data.plot(kind='scatter', x='x', y='y', ax=axs[i])
        axs[i].set_title(titles[i])
        axs[i].set_xlabel("x")
        axs[i].set_ylabel("y")
    plt.show()

# Function to fit polynomial and calculate R-squared
def fit_polynomial(data, degree):
    coefficients = np.polyfit(data['x'], data['y'], degree)
    yfit = np.polyval(coefficients, data['x'])
    yresid = data['y'] - yfit
    SSresid = np.sum(np.power(yresid, 2))
    SStotal = len(data['y']) * np.var(data['y'])
    rsq = 1 - SSresid / SStotal
    return coefficients, rsq

# Function to print results
def print_results(data_name, degree, coefficients, rsquared, r_value, p_value):
    print(f"Dataset {data_name}.csv:")
    print(f"Polynomial degree {degree} coefficients:", coefficients)
    print("R-squared value (from polynomial fitting):", rsquared)
    print("R-squared value (scipy method):", r_value ** 2)
    print("P-value:", p_value)
    print("--------------------------------------------------")
