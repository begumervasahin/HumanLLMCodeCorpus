from Shape import Shape
import newick

class PhyloTree(Shape):
    """
    PhyloTree class represents a phylogenetic tree, extending the Shape class.
    It overrides methods from Shape class to account for the distinguishability of leaves.
    """
    def __init__(self, leaf=None, children=None):
        """
        Initializes a new PhyloTree object.
        """
        super().__init__(children)
        self.leaf = leaf
        assert not self.is_leaf or (leaf is not None)

    def shape(self):
        """
        Returns the Shape associated with self, forgetting the labels of the leaves.
        """
        if self.is_leaf:
            return Shape(None)
        else:
            return Shape([x.shape() for x in self.children])

    def to_newick_tuple(self):
        """
        Returns a tuple representing the simplified Newick code of self.
        """
        if self.is_leaf:
            return self.leaf
        else:
            return tuple(x.to_newick_tuple() for x in self.children)

    def leaves(self):
        """
        Yields the labels of the leaves of self.
        """
        if self.is_leaf:
            yield self.leaf
        else:
            for x in self.children:
                for l in x.leaves():
                    yield l

    def labels(self):
        """
        Returns a sorted list of labels appearing in self.
        """
        return sorted(list(self.leaves()))

    def is_phylo(self):
        """
        Returns True if self is a phylogenetic tree (no repeated leaves), False otherwise.
        """
        labels = self.labels()
        return len(labels) == len(set(labels))

def from_newick(X):
    """
    Creates a PhyloTree object from a Newick code string.
    """
    return newick_node_to_tree(newick.loads(X)[0])

def from_newick_list(X):
    """
    Creates a list of PhyloTree objects from a list of Newick code strings.
    """
    return [newick_node_to_tree(n) for n in newick.loads(X)]

def newick_node_to_tree(node):
    """
    Converts a Node object to a PhyloTree object.
    """
    if not node.descendants:
        return PhyloTree(node.name)
    return PhyloTree(None, sorted([newick_node_to_tree(x) for x in node.descendants]))

def trees_from_file(fname, encoding='utf8', strip_comments=False, **kwargs):
    """
    Loads a list of trees from a Newick formatted file.
    """
    nodes = newick.read(fname, encoding, strip_comments, **kwargs)
    return [newick_node_to_tree(node) for node in nodes]
