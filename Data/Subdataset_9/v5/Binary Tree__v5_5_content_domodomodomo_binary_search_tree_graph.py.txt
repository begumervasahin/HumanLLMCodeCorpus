import graphviz
import random
import string
import log

def generate_source(binary_search_tree, max_depth=0):
    """
    Generates the source code of a graph visualization of a binary search tree.

    Args:
        binary_search_tree (BinarySearchTree): The binary search tree to visualize.
        max_depth (int, optional): Maximum depth to visualize. Defaults to 0, which visualizes the entire tree.

    Returns:
        str: The source code of the graph.
    """
    diagraph = _create_diagraph(binary_search_tree, max_depth)
    return diagraph.source

def render_tree(binary_search_tree, max_depth=0, file_name='sample'):
    """
    Renders a graph visualization of a binary search tree and saves it as an image file.

    Args:
        binary_search_tree (BinarySearchTree): The binary search tree to visualize.
        max_depth (int, optional): Maximum depth to visualize. Defaults to 0, which visualizes the entire tree.
        file_name (str, optional): Name of the image file to save. Defaults to 'sample'.
    """
    diagraph = _create_diagraph(binary_search_tree, max_depth)
    diagraph.render('img/' + file_name, cleanup=True)

def _create_diagraph(binary_search_tree, max_depth=0):
    """
    Creates a graphviz Diagraph object representing a binary search tree.

    Args:
        binary_search_tree (BinarySearchTree): The binary search tree to visualize.
        max_depth (int, optional): Maximum depth to visualize. Defaults to 0, which visualizes the entire tree.

    Returns:
        graphviz.Digraph: The graphviz Diagraph object representing the binary search tree.
    """
    root = binary_search_tree.root
    node_list = [node for node in log.generate_node(root)]
    route_list = [list(log.generate_route(root, node)) for node in node_list]
    depth_list = [len(route) - 1 for route in route_list]
    max_depth = max_depth if max_depth else max(depth_list)
    height_list = [max_depth - depth for depth in depth_list]

    diagraph = graphviz.Digraph('BST', format='png')
    for node, height in zip(node_list, height_list):
        diagraph.node(str(id(node)), str(node.value))
        if node.left:
            diagraph.edge(str(id(node)), str(id(node.left)))
        elif height:
            _add_blank_edge(diagraph, str(id(node)), height)
        if node.right:
            diagraph.edge(str(id(node)), str(id(node.right)))
        elif height:
            _add_blank_edge(diagraph, str(id(node)), height)
    return diagraph

def _add_blank_edge(diagraph, identity, height):
    """
    Adds a blank edge to the graph.

    Args:
        diagraph (graphviz.Digraph): The graphviz Diagraph object.
        identity (str): ID of the node to add the blank edge from.
        height (int): Height of the blank edge.
    """
    diagraph.edge(identity, _add_blank_node(diagraph, height - 1), color='gray91')

def _add_blank_node(diagraph, height):
    """
    Creates a blank node in the graph.

    Args:
        diagraph (graphviz.Digraph): The graphviz Diagraph object.
        height (int): Height of the blank node.

    Returns:
        str: ID of the created blank node.
    """
    lowercase = string.ascii_lowercase
    identity = ''.join([random.choice(lowercase) for _ in range(10)])
    diagraph.node(identity, '', color='gray91')
    if height:
        _add_blank_edge(diagraph, identity, height)
        _add_blank_edge(diagraph, identity, height)
    return identity
