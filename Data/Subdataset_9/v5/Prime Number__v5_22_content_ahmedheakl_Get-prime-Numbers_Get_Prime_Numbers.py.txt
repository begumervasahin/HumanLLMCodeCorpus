def get_prime_numbers(end_range):
    """
    Generates prime numbers up to the specified end of the range.

    Args:
        end_range (int): The end of the range to search for prime numbers.

    Returns:
        list: A list of prime numbers found within the specified range.
    """
    # Initialize an empty list to store prime numbers
    prime_numbers = []

    # Iterate through a range of numbers to find potential prime numbers
    for i in range(1, end_range):
        # Calculate the square root of (24 * i + 1)
        sqrt_value = (24 * i + 1) ** 0.5

        # Check if the square root is an integer and less than the specified end of the range
        if sqrt_value % 1 == 0 and int(sqrt_value) < end_range:
            prime_numbers.append(int(sqrt_value))
        # Stop the loop if the square root exceeds the specified end of the range
        elif int(sqrt_value) > end_range:
            break

    # Remove elements from the list if they are divisible by numbers from 2 to 9
    prime_numbers = [num for num in prime_numbers if all(num % i != 0 for i in range(2, 10))]

    # Square the remaining elements in the list
    prime_squares = [num ** 2 for num in prime_numbers]

    # Remove elements from the list if they match the squares
    prime_numbers = [num for num in prime_numbers if num not in prime_squares]

    # Add prime numbers 2, 3, 5, and 7 to the list
    prime_numbers.extend([2, 3, 5, 7])

    # Sort the list of prime numbers
    prime_numbers.sort()

    return prime_numbers

if __name__ == "__main__":
    # Prompt user to enter the end of the range
    end_range = int(input('Enter the end of the range: '))
    
    # Generate prime numbers up to the specified end of the range
    primes = get_prime_numbers(end_range)
    
    # Print the sorted list of prime numbers
    print(primes)
