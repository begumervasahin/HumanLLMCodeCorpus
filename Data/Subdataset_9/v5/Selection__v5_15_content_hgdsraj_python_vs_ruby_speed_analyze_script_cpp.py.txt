def extract_average_time_from_file(file_path):
    """
    Extracts the average time from a file containing time values.

    Args:
    - file_path: The path to the file containing time values.

    Returns:
    - The average time extracted from the file.
    """
    # Open the file and read its lines
    with open(file_path) as f:
        lines = f.readlines()

    # Extract the time values
    times = []
    for line in lines:
        if "real" in line:
            time_str = line[9:12]
            times.append(float(time_str))

    # Calculate the average time
    total_time = sum(times)
    average_time = total_time / len(times)
    return average_time
