import numpy as np
import matplotlib.pyplot as plt
from mpl_toolkits import mplot3d
from scipy.sparse import rand
from scipy.sparse.linalg import svds

# Function to generate sparse random data
def generate_sparse_data(rows, cols):
    X = rand(rows, cols, density=0.1)
    k = min(X.shape) - 1
    return X, k

# Function to perform Singular Value Decomposition (SVD)
def perform_svd(X, k):
    U, S, V = svds(X, k=k)
    singular_values = np.dot(U, np.diag(S))
    return singular_values

# Function to plot the singular values in 3D space
def plot_svd_graph(singular_values):
    fig = plt.figure(figsize=(12, 6))

    # Scatter plot of singular values
    ax1 = fig.add_subplot(121, projection='3d')
    ax1.scatter(singular_values[:, 0], singular_values[:, 1], singular_values[:, 2])
    ax1.set_title('Singular Values Scatter Plot')

    # Line plot of singular values
    ax2 = fig.add_subplot(122, projection='3d')
    ax2.plot3D(singular_values[:, 0], singular_values[:, 1], singular_values[:, 2])
    ax2.set_title('Singular Values 3D Line Plot')

    plt.show()

def main():
    # Input for number of rows in the sparse random matrix
    rows = int(input('Enter the number of rows for your sparse random matrix:\n')) 
    print('Currently, the program supports 3-dimensional plots. More dimensions will be supported in the future.')
    
    # Generate data and perform SVD
    X, k = generate_sparse_data(rows, 3)
    singular_values = perform_svd(X, k)

    # Plot the singular values
    print('Displaying a graph with your singular vectors plotted along with the singular value matrix...')
    plot_svd_graph(singular_values)

if __name__ == '__main__':
    main()
