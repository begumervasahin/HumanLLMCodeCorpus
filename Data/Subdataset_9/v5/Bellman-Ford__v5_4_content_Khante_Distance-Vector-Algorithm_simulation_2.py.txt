import network_2
import link_2
import threading
from time import sleep

router_queue_size = 0
simulation_time = 2

def create_network():
    object_list = []

    host_1 = network_2.Host('H1')
    object_list.append(host_1)

    host_2 = network_2.Host('H2')
    object_list.append(host_2)

    cost_dict_a = {'H1': {0: 1}, 'RB': {1: 1}}
    router_a = network_2.Router(name='RA', cost_D=cost_dict_a, max_queue_size=router_queue_size)
    object_list.append(router_a)

    cost_dict_b = {'H2': {1: 3}, 'RA': {0: 1}}
    router_b = network_2.Router(name='RB', cost_D=cost_dict_b, max_queue_size=router_queue_size)
    object_list.append(router_b)

    link_layer = link_2.LinkLayer()
    object_list.append(link_layer)

    link_layer.add_link(link_2.Link(host_1, 0, router_a, 0))
    link_layer.add_link(link_2.Link(router_a, 1, router_b, 0))
    link_layer.add_link(link_2.Link(router_b, 1, host_2, 0))

    return object_list

def start_threads(object_list):
    thread_list = [threading.Thread(name=obj.__str__(), target=obj.run) for obj in object_list]
    for thread in thread_list:
        thread.start()
    return thread_list

def stop_threads(object_list, thread_list):
    for obj in object_list:
        obj.stop = True
    for thread in thread_list:
        thread.join()
    print("All simulation threads joined")

def simulate_network():
    object_list = create_network()
    thread_list = start_threads(object_list)

    router_a = next(obj for obj in object_list if isinstance(obj, network_2.Router) and obj.name == 'RA')
    router_a.send_routes(1)
    sleep(simulation_time)

    print("Converged routing tables")
    for obj in object_list:
        if isinstance(obj, network_2.Router):
            obj.print_routes()

    object_list[0].udt_send('H2', 'MESSAGE_FROM_H1')
    object_list[1].udt_send('H1', 'reply for MESSAGE_FROM_H1')

    stop_threads(object_list, thread_list)

if __name__ == '__main__':
    simulate_network()
