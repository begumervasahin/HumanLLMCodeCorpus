import math
from scipy import spatial

def inverse_document_frequency(term, all_documents):
    """
    Calculate the inverse document frequency of a term.
    
    Args:
        term (str): The term whose IDF is to be calculated.
        all_documents (list of str): List of all documents.
        
    Returns:
        float: The inverse document frequency of the term.
    """
    num_documents_with_term = sum(1 for doc in all_documents if term.lower() in doc.lower().split())
    if num_documents_with_term > 0:
        idf = 1.0 + math.log(len(all_documents) / num_documents_with_term)
    else:
        idf = 1.0
    return idf

def term_frequency(term, document):
    """
    Calculate the term frequency of a term in a document.
    
    Args:
        term (str): The term whose TF is to be calculated.
        document (str): The document containing the term.
        
    Returns:
        float: The term frequency of the term in the document.
    """
    normalized_document = document.lower().split()
    return normalized_document.count(term.lower()) / float(len(normalized_document))    

def get_tfidf(document, all_documents):
    """
    Calculate TF-IDF scores for terms in a document.
    
    Args:
        document (str): The document for which TF-IDF scores are to be calculated.
        all_documents (list of str): List of all documents.
        
    Returns:
        list of float: List of TF-IDF scores for each term in the document.
    """
    terms = document.lower().split()
    tfidfs = []
    for term in terms:
        tf = term_frequency(term, document)
        idf = inverse_document_frequency(term, all_documents)
        tfidf = tf * idf
        tfidfs.append(tfidf)
    return tfidfs

def cosine_similarity(query, document):
    """
    Calculate cosine similarity between a query and a document.
    
    Args:
        query (list): List of TF-IDF scores for the query.
        document (list): List of TF-IDF scores for the document.
        
    Returns:
        float: The cosine similarity between the query and the document.
    """
    padding_length = abs(len(query) - len(document))
    padding = [0] * padding_length
    if len(query) > len(document):
        document.extend(padding)
    elif len(document) > len(query):
        query.extend(padding)
    return 1 - spatial.distance.cosine(query, document)
