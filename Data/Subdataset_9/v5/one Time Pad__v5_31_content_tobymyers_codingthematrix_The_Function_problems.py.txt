def tuple_list_sum(list_a, list_b):
    """
    Sums up corresponding elements of two lists of tuples.

    Args:
        list_a: A list of tuples.
        list_b: A list of tuples.

    Returns:
        A list of pairs (x, y) where each pair is the sum of corresponding elements from tuples in list_a and list_b.
    """
    return [(a[0] + b[0], a[1] + b[1]) for a, b in zip(list_a, list_b)]

def inverse_dictionary(d):
    """
    Computes the inverse of a dictionary representing an invertible function.

    Args:
        d: A dictionary representing an invertible function f.

    Returns:
        A dictionary representing the inverse of f, where keys are the values of d and values are the keys of d.
    """
    return {value: key for key, value in d.items()}

def generate_sequence(start, length):
    """
    Generates a list of 'length' elements where each element is 'start' + index.

    Args:
        start: A number indicating the starting value of the sequence.
        length: A number indicating the length of the sequence.

    Returns:
        A list of 'length' elements where each element is 'start' + index.
    """
    return [start + i for i in range(length)]
