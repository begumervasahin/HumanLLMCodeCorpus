class ListNode:
    def __init__(self, value, next_node=None):
        self.value = value
        self.next = next_node

def length_iterative(ll):
    """Returns the length of the linked list iteratively."""
    count = 0
    while ll is not None:
        count += 1
        ll = ll.next
    return count

def length_recursive(ll):
    """Returns the length of the linked list recursively."""
    if ll is None:
        return 0
    return 1 + length_recursive(ll.next)

def sum_iterative(ll):
    """Returns the sum of values in the linked list iteratively."""
    total = 0
    while ll is not None:
        total += ll.value
        ll = ll.next
    return total
