import socket
import os

# Constants
PORT = 8998
HOST = "0.0.0.0"

def generate_key(secret_key_size=10):
    """Generate a secret key."""
    return int.from_bytes(os.urandom(secret_key_size), "big")

def steve():
    """Function for Steve to establish a shared key."""
    # Generate a secret key for Steve
    secret_key = generate_key()

    # Create a server socket and bind it to the specified host and port
    server_socket = socket.socket()
    server_socket.bind((HOST, PORT))
    server_socket.listen(5)

    print("Steve's server is running...")

    while True:
        # Accept incoming connections
        client_socket, client_address = server_socket.accept()
        print(f"Connection established with {client_address}")

        # Receive the common key from the other party
        common_key = int(client_socket.recv(1000))

        # Compute the shared key
        shared_key = common_key + secret_key

        # Send the shared key back to the other party
        client_socket.send(bytes(str(shared_key), 'utf-8'))

        # Receive the state from the other party
        state1 = int(client_socket.recv(100))

        # Close the connection
        client_socket.close()

        # Compute the shared key again
        shared_key_with_state = state1 + secret_key

        # Print the secret key and shared key
        print("Secret Key:", secret_key)
        print("Shared Key with state:", shared_key_with_state)

def do_diffie_hellman(steve_address, steves_port=PORT, secret_key_size=10):
    """Function for another party to perform Diffie-Hellman key exchange with Steve."""
    # Generate a secret key for the other party
    secret_key = generate_key()

    # Generate a common key
    common_key = generate_key()

    # Create a client socket and connect to Steve's server
    client_socket = socket.socket()
    client_socket.connect((steve_address, steves_port))

    # Send the common key to Steve
    client_socket.send(bytes(str(common_key), 'utf-8'))

    # Compute the shared key
    shared_key = common_key + secret_key

    # Receive the state from Steve
    state1 = int(client_socket.recv(100))

    # Send the state to Steve
    client_socket.send(bytes(str(shared_key), 'utf-8'))

    # Close the connection
    client_socket.close()

    # Compute the shared key with state
    shared_key_with_state = state1 + secret_key

    # Print the secret key and shared key
    print("Secret Key:", secret_key)
    print("Shared Key with state:", shared_key_with_state)

# Example usage
if __name__ == "__main__":
    steve()
