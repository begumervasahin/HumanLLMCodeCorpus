import numpy as np

def sigmoid(w):
    """
    Sigmoid function.

    Args:
        w (float): Input value.

    Returns:
        float: Output value of the sigmoid function.
    """
    return 1 / (1 + np.exp(-w))

def sigmoid_derivative(w):
    """
    Derivative of the sigmoid function.

    Args:
        w (float): Input value.

    Returns:
        float: Output value of the derivative of the sigmoid function.
    """
    exp_val = np.exp(-w)
    if exp_val < 10 ** 16:
        if exp_val > 10 ** -16:
            f = exp_val / ((1 + exp_val) ** 2)
        else:
            f = exp_val
    else:
        f = np.exp(w)
    return f

def sigmoid_second_derivative(w):
    """
    Second derivative of the sigmoid function.

    Args:
        w (float): Input value.

    Returns:
        float: Output value of the second derivative of the sigmoid function.
    """
    exp_val = np.exp(-w)
    if exp_val < 10 ** 16:
        if exp_val > 10 ** -16:
            f = -exp_val * (np.exp(w) - 1) / ((1 + np.exp(w)) ** 3)
        else:
            f = -np.exp(-w)
    else:
        f = np.exp(w)
    return f

def loss(x, S, A, y, arg=1, penalization=False):
    '''
    Input parameters :
    x: array, point for which the loss must be computed
    S: list, subset of indices
    A: matrix of the datapoints
    y: vector of labels
    arg: int, number of output parameters
    penalization: bool, use of a penalized empirical loss function or not
    Returns:
    value of the loss function at point x if arg==1
    value of the loss function and its gradient at point x if arg==2
    value of the loss function, its gradient, and its Hessian at point x if arg==3
    '''
    d = np.size(A, 1)
    card_S = len(S)
    f = 0
    if arg == 2 or arg == 3:
        g = np.zeros(d)
    if arg == 3:
        H = np.zeros((d, d))
    for i in range(card_S):
        y_S_i = y[S[i]]
        A_S_i = A[S[i]]
        scal_prod = np.dot(x, A_S_i)
        f += (y_S_i - sigmoid(scal_prod)) ** 2
        if arg == 2 or arg == 3:
            g += 2 * A_S_i * sigmoid_derivative(scal_prod) * (sigmoid(scal_prod) - y_S_i)
        if arg == 3:
            H += 2 * ((sigmoid(scal_prod) - y_S_i) * sigmoid_second_derivative(scal_prod) +
                      sigmoid_derivative(scal_prod) ** 2) * np.dot(np.transpose([A_S_i]), [A_S_i])
    if penalization:
        f += 0.5 * np.linalg.norm(x) ** 2
    f = f / card_S
    if arg == 3:
        H = 0.5 * (H + np.transpose(H))
        if penalization:
            g += x
            H += np.eye(d)
        g = g / card_S
        H = H / card_S
        return f, g, H
    elif arg == 2:
        if penalization:
            g += x
        g = g / card_S
        return f, g
    else:
        return f
