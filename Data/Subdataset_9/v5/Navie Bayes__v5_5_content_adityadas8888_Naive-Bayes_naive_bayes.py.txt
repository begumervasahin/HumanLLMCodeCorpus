import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
import math
from scipy.stats import norm

def generate_data(tsize0, tsize1):
    mean1 = [1, 0]
    mean2 = [0, 1]
    Sigma1 = Sigma2 = [[1, 0.75], [0.75, 1]]

    x1, y1 = np.random.multivariate_normal(mean1, Sigma1, tsize0).T
    x2, y2 = np.random.multivariate_normal(mean2, Sigma2, tsize1).T

    return x1, y1, x2, y2

def train_naive_bayes(x1, y1, x2, y2):
    dtrain = pd.DataFrame({'x': np.concatenate([x1, x2]), 'y': np.concatenate([y1, y2]), 'label': np.concatenate([np.zeros(len(x1)), np.ones(len(x2))])})
    dtest = pd.DataFrame({'x': np.concatenate([x1, x2]), 'y': np.concatenate([y1, y2]), 'actual': np.concatenate([np.zeros(len(x1)), np.ones(len(x2))])})

    prob0 = len(x1) / (len(x1) + len(x2))
    prob1 = len(x2) / (len(x1) + len(x2))

    meanx0, meany0 = np.mean(x1), np.mean(y1)
    meanx1, meany1 = np.mean(x2), np.mean(y2)
    stdx0, stdy0 = np.std(x1), np.std(y1)
    stdx1, stdy1 = np.std(x2), np.std(y2)

    label0 = norm.pdf(dtest['x'], loc=meanx0, scale=stdx0) * norm.pdf(dtest['y'], loc=meany0, scale=stdy0)
    label1 = norm.pdf(dtest['x'], loc=meanx1, scale=stdx1) * norm.pdf(dtest['y'], loc=meany1, scale=stdy1)

    post0, post1 = label0 * prob0, label1 * prob1

    dtest['post0'], dtest['post1'] = post0, post1
    dtest['pred'] = dtest[['post0', 'post1']].max(axis=1)
    dtest['predicted'] = np.where(dtest['pred'] == post1, 0, 1)
    dtest['color'] = np.where(dtest['pred'] == post1, 'r', 'b')

    return dtest

def evaluate_model(dtest):
    tp = np.sum((dtest['predicted'] == 1) & (dtest['actual'] == 1))
    tn = np.sum((dtest['predicted'] == 0) & (dtest['actual'] == 0))
    fp = np.sum((dtest['predicted'] == 1) & (dtest['actual'] == 0))
    fn = np.sum((dtest['predicted'] == 0) & (dtest['actual'] == 1))

    accuracy = (tp + tn) / (tp + fp + fn + tn)
    error_rate = 1 - accuracy
    recall = tp / (tp + fn)
    precision = tp / (tp + fp)

    print("Accuracy: {:.2f}, Error Rate: {:.2f}, Recall: {:.2f}, Precision: {:.2f}".format(accuracy, error_rate, recall, precision))
    print("True Positive: {}, True Negative: {}, False Positive: {}, False Negative: {}".format(tp, tn, fp, fn))

    return accuracy

def plot_data(dtest):
    plt.scatter(dtest['x'], dtest['y'], color=dtest['color'], alpha=0.5, edgecolor='k', marker='x')
    plt.show()

def main():
    x1, y1, x2, y2 = generate_data(500, 500)
    dtest = train_naive_bayes(x1, y1, x2, y2)
    accuracy = evaluate_model(dtest)
    plot_data(dtest)

if __name__ == "__main__":
    main()
