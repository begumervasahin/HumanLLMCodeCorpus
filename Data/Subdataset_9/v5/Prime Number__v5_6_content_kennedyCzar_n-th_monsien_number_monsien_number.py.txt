import random

def find_monisen(n):
    """
    Find the n-th Monisen number.

    A number M is a Monisen number if M = 2**P - 1
    and both M and P are prime numbers.

    Args:
    - n: The index of the Monisen number to find.

    Returns:
    - The n-th Monisen number and its corresponding prime exponent.
    """
    number = 10
    p = 2
    while p < n:
        if all(number % i != 0 for i in range(2, number)):
            p += 1
        number += 1
        m = 2**p - 1
    return number - 1, m

def print_even_numbers():
    """Prints even numbers from 1 to 10."""
    even_numbers = [i + 1 for i in range(10) if i % 2 == 0]
    print(even_numbers)

def calculate_sum():
    """Calculates the sum of numbers until it exceeds 10."""
    sumA = 0
    i = 1
    while True:
        sumA += i
        i += 1
        if sumA > 10:
            break
    print('i = {}, sum = {}'.format(i, sumA))

def print_numbers_divisible_by_3():
    """Prints numbers until reaching 10 or a number divisible by 3."""
    i = 1
    while(i % 3):
        print(i, end=' ')
        if (i >= 10):
            break
        i += 1

def convert_base(num, base):
    """Converts a number to the given base."""
    if(num >= base):
        convert_base(num // base, base)
    print(num % base, end=' ')
