# Import necessary libraries
import numpy as np
import pandas as pd
from statsmodels.tsa.arima.model import ARIMA

# Read the CSV file into a pandas DataFrame
SPX = pd.read_csv("SPX Weekly 2000-2017.csv")

# Display the first 5 rows of the DataFrame
print("First 5 rows of the SPX DataFrame:")
print(SPX.head(n=5))

# Create an ARIMA model with order (0, 1, 0)
arima_order = (0, 1, 0)
model = ARIMA(SPX, order=arima_order)

# Fit the ARIMA model to the data
model_fit = model.fit()

# Forecast the next data point using the fitted model
forecasted_value = model_fit.forecast()[0]

# Display the summary of the ARIMA model
print("Summary of the ARIMA model:")
model_summary = model_fit.summary().tables[1]
print(model_summary)
