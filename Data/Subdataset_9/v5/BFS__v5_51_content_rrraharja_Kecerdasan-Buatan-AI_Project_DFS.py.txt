def dfs(graph, start, finish):
    """
    Perform depth-first search (DFS) on a graph to find a path from the start node to the finish node.

    Args:
        graph (dict): The graph represented as a dictionary.
        start: The starting node for DFS.
        finish: The target node to reach.

    Returns:
        list or None: A path from the start node to the finish node, or None if no path is found.
    """
    stack = [[start]]  # Initialize stack with start node as a path
    visited = set()  # Initialize set to keep track of visited nodes

    while stack:
        path = stack.pop()  # Get the last path in the stack
        state = path[-1]  # Get the last node in the path

        if state == finish:
            return path  # Return the path if it reaches the finish node

        if state not in visited:
            # Explore all branches from the current node
            for branch in graph.get(state, []):
                new_path = path + [branch]  # Create a new path by appending the branch to the current path
                stack.append(new_path)  # Add the new path to the stack

            visited.add(state)  # Mark the current node as visited

    print("NOT FOUND 404")  # If the stack is empty and finish node is not found, print "NOT FOUND 404"
    return None
