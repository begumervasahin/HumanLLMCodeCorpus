import matplotlib.pyplot as plt
import numpy as np

# Function to read data from file
def read_file(file_path):
    data = np.loadtxt(file_path, delimiter=',')
    tau = data[:, 0]
    variance = data[:, 1]
    error = data[:, 2]
    return tau, variance, error

# Read data from files
tau_adj, variance_adj, error_adj = read_file('allanVar_adj.dat')
tau_sep, variance_sep, error_sep = read_file('allanVar_sep.dat')

# Calculate expected values
expected_adj = variance_adj[0] * (tau_adj / tau_adj[0]) ** (-1)
expected_sep = variance_sep[0] * (tau_sep / tau_sep[0]) ** (-1)

# Plotting
plt.errorbar(tau_adj, variance_adj, yerr=2 * error_adj, color='b', ecolor='r', marker='.', label='Mid')
plt.errorbar(tau_sep, variance_sep, yerr=2 * error_sep, color='black', ecolor='r', marker='x', label='Edge')
plt.plot(tau_adj, expected_adj, c='g', label='Expected (Mid)')
plt.plot(tau_sep, expected_sep, c='orange', label='Expected (Edge)')

# Scaling
plt.yscale('log')
plt.xscale('log')

# Labels and title
plt.title('Allan Variance Plot with (t^-1) Comparison Line')
plt.xlabel('Integration Time (s)')
plt.ylabel('Allan Variance (arbitrary units)')
plt.legend(loc='upper right')

# Save figure
plt.savefig('allan_variance_plot.png')

# Display the plot
plt.show()
