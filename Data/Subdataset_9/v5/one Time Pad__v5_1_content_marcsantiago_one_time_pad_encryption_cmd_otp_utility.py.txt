import argparse
from OTP import OneTimePadEncryption

def main():
    # Create an instance of the OneTimePadEncryption class
    otp = OneTimePadEncryption.OneTimePadEncryption()

    # Create an ArgumentParser object to handle command-line arguments
    parser = argparse.ArgumentParser(description='One Time Pad Encryption Software')

    # Define command-line arguments
    parser.add_argument('-e', '--encrypt_string', help='Encrypts a string entered by the user')
    parser.add_argument('-d', '--decrypt_string', help='Decrypts a string. Requires a key provided by the user')
    parser.add_argument('-k', '--key', help='The key provided by the user as a string')
    parser.add_argument('-kf', '--key_file', help='The file path containing the key provided by the user')
    parser.add_argument('-ef', '--encrypt_file', help='Encrypts data from a file')
    parser.add_argument('-df', '--decrypt_file', help='Decrypts data from a file. The key can be provided as a file or a string.')

    # Parse the command-line arguments
    args = vars(parser.parse_args())

    # Perform encryption or decryption based on the provided arguments
    if args['encrypt_string']:
        otp.encrypt_string_or_file(args['encrypt_string'], string_file_mode=False)
    elif args['encrypt_file']:
        otp.encrypt_string_or_file(args['encrypt_file'], string_file_mode=True)
    elif args['decrypt_string'] and args['key_file']:
        otp.decrypt_string_or_file(args['key_file'], args['decrypt_string'], key_file_mode=True, encrypted_string_file_mode=False)
    elif args['decrypt_string'] and args['key']:
        otp.decrypt_string_or_file(args['key'], args['decrypt_string'], key_file_mode=False, encrypted_string_file_mode=False)
    elif args['decrypt_file'] and args['key_file']:
        otp.decrypt_string_or_file(args['key_file'], args['decrypt_file'], key_file_mode=True, encrypted_string_file_mode=True)
    elif args['decrypt_file'] and args['key']:
        otp.decrypt_string_or_file(args['key'], args['decrypt_file'], key_file_mode=False, encrypted_string_file_mode=True)
    else:
        print_usage(parser)

def print_usage(parser):
    print("""
    Usage:
    -e ENCRYPT_STRING, --encrypt_string ENCRYPT_STRING
        Encrypts a string entered by the user
    -d DECRYPT_STRING, --decrypt_string DECRYPT_STRING
        Decrypts a string. Requires a key provided by the user
    -k KEY, --key KEY
        The key provided by the user as a string
    -kf KEY_FILE, --key_file KEY_FILE
        The file path containing the key provided by the user
    -ef ENCRYPT_FILE, --encrypt_file ENCRYPT_FILE
        Encrypts data from a file
    -df DECRYPT_FILE, --decrypt_file DECRYPT_FILE
        Decrypts data from a file. The key can be provided as a
        file or a string.
    """)

if __name__ == "__main__":
    main()
