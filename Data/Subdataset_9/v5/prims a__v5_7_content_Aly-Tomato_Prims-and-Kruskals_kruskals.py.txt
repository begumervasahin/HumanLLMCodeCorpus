import heapq
import global_variables as glb  # Assuming 'global_variables' module contains necessary global variables

# Read the graph from a file
def read_graph(file_path, delimiter):
    with open(file_path) as file:
        for line in file:
            # Split the line based on the delimiter
            vertex1, vertex2, weight = [x.strip() for x in line.split(delimiter)]
            
            # Skip if the edge already exists
            if (int(weight), vertex2, vertex1) in glb.EDGES:
                continue
            
            # Add the edge to the global list of edges
            heapq.heappush(glb.EDGES, (int(weight), vertex1, vertex2))
            
            # Add vertices to the global set of vertices
            glb.VERTICES.add(vertex1)
            glb.VERTICES.add(vertex2)
    
    return glb.EDGES

# Create a set for a vertex
def make_set(vertex):
    glb.PARENT[vertex] = vertex
    glb.RANK[vertex] = 0

# Find the representative of a set (using path compression)
def find(vertex):
    if glb.PARENT[vertex] != vertex:
        glb.PARENT[vertex] = find(glb.PARENT[vertex])
    return glb.PARENT[vertex]

# Perform union of two sets (using union by rank)
def union(vertex1, vertex2):
    root1 = find(vertex1)
    root2 = find(vertex2)
    if root1 != root2:
        if glb.RANK[root1] > glb.RANK[root2]:
            glb.PARENT[root2] = root1
        else:
            glb.PARENT[root1] = root2
            glb.RANK[root2] += 1

# Perform Kruskal's algorithm
def kruskals():
    total_weight = 0
    
    # Create sets for each vertex
    for vertex in glb.VERTICES:
        make_set(vertex)
    
    # Process edges in ascending order of weights
    while glb.EDGES:
        weight, vertex1, vertex2 = heapq.heappop(glb.EDGES)
        # Check for cycle using disjoint set union
        if find(vertex1) != find(vertex2):
            # Merge sets and update total weight
            union(vertex1, vertex2)
            total_weight += weight
            # Add edge to the minimum spanning tree
            glb.MST.append((vertex1, vertex2, str(weight), str(total_weight)))
    
    return (glb.MST, total_weight)
