import os
import fnmatch
import logging

def delete_matching_files(directory, patterns):
    """
    Recursively searches for files matching deletion patterns in the specified directory and deletes them.

    Args:
        directory (str): The directory path to search for files.
        patterns (list): List of file patterns to match for deletion.

    Returns:
        list: List of deleted files.
    """
    deleted_files = []
    for root, _, filenames in os.walk(directory):
        for pattern in patterns:
            for name in fnmatch.filter(filenames, pattern):
                file_path = os.path.join(root, name)
                os.remove(file_path)
                deleted_files.append(file_path)
    return deleted_files

def setup_logging(log_file):
    """
    Configures logging to write messages to a file.

    Args:
        log_file (str): The path to the log file.
    """
    logging.basicConfig(format='%(asctime)s %(message)s',
                        filename=log_file,
                        level=logging.INFO)

def main():
    # Define the path to the Wowza recording directory
    recording_directory = "/path/to/wowza/recording/dir"

    # Define patterns of files to be deleted
    deletion_patterns = ["*_240p*.mp4",
                         "*_360p*.mp4",
                         "*_480p*.mp4",
                         "*_720p*.mp4",
                         "*_1080p*.mp4"]

    # Configure logging
    log_file = '/var/log/wowza_delete_recordings.log'
    setup_logging(log_file)

    # Log script start
    logging.info("Script started")

    # Delete matching files
    deleted_files = delete_matching_files(recording_directory, deletion_patterns)

    # Check if there are files to delete
    if not deleted_files:
        logging.info("Nothing to delete")
    else:
        # Log deleted files
        for deleted_file in deleted_files:
            logging.info("Deleted file: %s", deleted_file)

    # Log script end
    logging.info("Script ended")

if __name__ == "__main__":
    main()
