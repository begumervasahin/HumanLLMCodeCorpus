def linear_search(lst, target):
    """
    Perform linear search to find the target in the list.

    Args:
    - lst: The list to search in.
    - target: The target number to search for.

    Returns:
    - True if the target is found, False otherwise.
    """
    for index, number in enumerate(lst):
        if number == int(target):
            global position
            position = index
            return True
    return False

# Initialize the list and position variable
position = -1
number_list = [21, 33, 45, 9, 24, 9, 34, 7, 88, 96, 54, 35, 66, 77]

# Ask user for the number to search
target_number = input("Enter the number you want to search for: ")

# Perform the search and print the result
if linear_search(number_list, target_number):
    print("The number is found at position:", position + 1)
else:
    print("Number not found.")
