import datetime

# Dictionary to cache Fibonacci values
fib_cache = {}

def fibonacci(n):
    """Calculate the nth Fibonacci number recursively with memoization."""
    # Base cases
    if n == 1 or n == 2:
        return 1
    
    # Check if value is already cached
    if n in fib_cache:
        return fib_cache[n]
    
    # Calculate Fibonacci value recursively
    fib_value = fibonacci(n - 1) + fibonacci(n - 2)
    
    # Cache the Fibonacci value
    fib_cache[n] = fib_value
    
    return fib_value

def main():
    # Get user input for the number of Fibonacci terms
    n_terms = int(input("Enter the number of Fibonacci terms: "))
    
    # Calculate and print Fibonacci numbers
    for i in range(1, n_terms + 1):
        print(f"{i}: {fibonacci(i)}")
    
    # Calculate and print execution time
    start_time = datetime.datetime.now()
    stop_time = datetime.datetime.now()
    execution_time = stop_time - start_time
    print("Execution time:", execution_time)

if __name__ == "__main__":
    main()
