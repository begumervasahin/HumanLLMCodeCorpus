# Importing necessary modules and functions
from Weighted_Graph import Weighted_Graph
from functions import update, total_sum

# Function to find the minimum spanning tree using Prim's algorithm
def find_minimum_spanning_tree(edge_file_name, start_vertex=0, draw=False):
    # Initialize the minimum spanning tree T with start_vertex as the root and an empty list for edges
    minimum_spanning_tree = ({start_vertex}, [])

    # Create a weighted graph G from the given edge file
    graph = Weighted_Graph(edge_file_name)

    # If draw parameter is True, draw the original graph
    if draw:
        graph.draw_graph()

    # Loop until all vertices are included in the minimum spanning tree T
    while minimum_spanning_tree[0] != graph.vertex_set():
        # Update the minimum spanning tree T by adding the next edge
        minimum_spanning_tree = update(minimum_spanning_tree, graph)

        # If draw parameter is True, draw the updated subgraph    
        if draw:
            graph.draw_subgraph(minimum_spanning_tree)

    # Print the total cost of the minimum spanning tree
    print("The minimum spanning tree costs: " + str(total_sum(minimum_spanning_tree, graph)))

    # Return the minimum spanning tree T
    return minimum_spanning_tree
