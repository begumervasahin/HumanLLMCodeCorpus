def get_positive_integer():
    """Get a positive integer from user input."""
    n_max = -1
    while n_max <= 0:
        s = input('Enter a positive integer: ')
        try:
            n_max = int(s)
        except ValueError:
            print('Invalid input')
        print()
    return n_max

def generate_fibonacci_sequence(n_max):
    """Generate Fibonacci sequence up to n_max terms."""
    f1 = 1
    f2 = 2
    n = 1
    while n <= n_max:
        print('%4d: %d' % (n, f1))
        f1, f2 = f2, f1 + f2
        n += 1

def main():
    """Main function."""
    n_max = get_positive_integer()
    generate_fibonacci_sequence(n_max)

if __name__ == "__main__":
    main()
