import math

def find_sum_of_proper_factors(x):
    # Function to find the sum of proper factors of a number
    factors = []
    for i in range(1, math.ceil(x / 2) + 1):
        if x % i == 0:
            factors.append(i)
    return sum(factors)

def find_abundant_numbers(limit):
    # Function to find abundant numbers up to a given limit
    abundants = []
    for k in range(11, limit):
        if find_sum_of_proper_factors(k) > k:
            abundants.append(k)
    return abundants

def find_sum_of_2_abundants(abundants):
    # Function to find the sum of all possible pairs of abundant numbers
    sum_of_2_abundants = set()
    for i in range(len(abundants)):
        for j in range(i, len(abundants)):  # Avoid duplicate pairs and symmetry
            if abundants[i] + abundants[j] < 28123:
                sum_of_2_abundants.add(abundants[i] + abundants[j])
    return sum_of_2_abundants

def find_non_abundant_sum(limit):
    # Function to find the sum of all positive integers that cannot be written as the sum of two abundant numbers
    abundants = find_abundant_numbers(limit)
    sum_of_2_abundants = find_sum_of_2_abundants(abundants)
    non_abundant_sum = sum(i for i in range(limit) if i not in sum_of_2_abundants)
    return non_abundant_sum

def main():
    # Main function to calculate and print the answer
    limit = 28123
    answer = find_non_abundant_sum(limit)
    print("The answer to this question should be", answer)

if __name__ == "__main__":
    main()
