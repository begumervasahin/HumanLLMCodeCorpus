from curve import Curve
from point import Point

class Public_Key:
    def __init__(self, modulus: int, factor: int):
        """
        Initialize a Public_Key object with the given modulus and factor.

        :param modulus: modulus for arithmetic operations
        :param factor: factor for generating the public key
        """
        self.p = modulus
        self.curve = Curve.make_curve(self.p)
        self.P = self.curve.random_point()
        self.Q = self.curve.multiply(factor, self.P)

    def __repr__(self) -> str:
        """
        Return a string representation of the Public_Key object.

        :return: string representation of the Public_Key object
        """
        return f"Curve: {self.curve}\nPublic Key (P): {self.P}\nDerived Public Key (Q): {self.Q}"

    @staticmethod
    def make_public_key(modulus: int, factor: int) -> 'Public_Key':
        """
        Static method to create a new Public_Key object.

        :param modulus: modulus for arithmetic operations
        :param factor: factor for generating the public key
        :return: new Public_Key object
        """
        return Public_Key(modulus, factor)
