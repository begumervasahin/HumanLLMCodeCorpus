import pandas as pd
import numpy as np
import pickle
import os
from sklearn.feature_extraction.text import CountVectorizer, TfidfTransformer
from sklearn.pipeline import Pipeline
from sklearn import svm

# Function to train the SVM model and save it
def train_and_save_model():
    # Set random seed for reproducibility
    np.random.seed(100)
    
    # Load data from CSV file
    corpus = pd.read_csv("Test.csv", encoding='utf-8-sig')
    
    # Extract text and labels from the dataset
    texts = corpus['text']
    labels = corpus['label']
    
    # Create a text classification pipeline
    text_classifier = Pipeline([
        ('vect', CountVectorizer()),    # Convert text to a matrix of token counts
        ('tfidf', TfidfTransformer()),  # Convert token counts to TF-IDF representation
        ('clf', svm.SVC(C=5, kernel='rbf', degree=3, gamma=0.5)),  # Support Vector Classifier
    ])
    
    # Fit the model
    trained_classifier = text_classifier.fit(texts, labels)
    
    # Save the trained model to a file
    model_filename = 'Amz_5000_.sav'
    with open(model_filename, 'wb') as model_file:
        pickle.dump(trained_classifier, model_file)
    
    # Check if the model file exists
    model_exists = os.path.exists(model_filename)
    
    # Generate messages based on whether the model was saved successfully
    success_message = "Successful: Training model saved."
    error_message = "Error: Something went wrong while saving the model."
    corpus_length = f"Number of data points: {len(corpus)}"
    
    if model_exists:  
        return [success_message, corpus_length]
    else:  
        return [error_message, corpus_length]

# Call the training function
train_and_save_model()
