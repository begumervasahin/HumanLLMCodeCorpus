from node import Node

class TreeSet:
    """
    Represents a set of d-heaps using a tree structure.
    """

    def __init__(self):
        """
        Initializes an empty tree set.
        """
        self.elements = {}

    def make_set(self, value):
        """
        Creates a new set with a single element.

        Parameters:
            value: The value of the element to be added to the set.
        """
        self.elements[value] = Node(value)

    def find(self, value):
        """
        Finds the representative of the set containing the given element.

        Parameters:
            value: The element whose representative needs to be found.

        Returns:
            int: The representative of the set containing the element.
        """
        node = self.elements[value]
        while node.parent != node.parent.parent:
            node.parent = node.parent.parent
            node = node.parent
        return node.parent.value

    def union(self, u, v):
        """
        Combines the sets that contain the elements u and v.

        Parameters:
            u: The first element.
            v: The second element.

        Returns:
            Node: The representative of the combined set.
        """
        node_u = self.elements[u]
        node_v = self.elements[v]

        if node_u.rank > node_v.rank:
            node_v.parent = node_u
            return node_u
        elif node_u.rank < node_v.rank:
            node_u.parent = node_v
            return node_v
        else:
            node_u.parent = node_v
            node_v.rank += 1
            return node_v

    def print_data(self):
        """
        Prints the elements and their corresponding nodes in the tree set.
        """
        print(self.elements)
