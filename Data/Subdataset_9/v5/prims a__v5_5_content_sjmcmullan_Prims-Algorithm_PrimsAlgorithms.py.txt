import sys
from GraphStuff import PriorityQueue

def prim(G, start):
    # Initializing priority queue
    pq = PriorityQueue()

    # Setting initial distances and predecessors for vertices
    for v in G:
        v.setDistance(sys.maxsize)  # Setting distance to infinity    
        v.setPred(None)  # Setting predecessor to None

    start.setDistance(0)  # Setting start vertex distance to 0        

    # Building the priority queue with vertices and their distances   
    pq.buildHeap([(v.getDistance(), v) for v in G])

    # Main loop to process vertices
    while not pq.isEmpty():
        # Getting the vertex with the minimum distance
        current_vert = pq.delMin()

        # Iterating over the adjacent vertices of the current vertex  
        for next_vert in current_vert.getConnections():
            # Getting the cost of the edge between current and adjacent vertex
            new_cost = current_vert.getWeight(next_vert)

            # Checking if the adjacent vertex is in the priority queue
            if next_vert in pq and new_cost < next_vert.getDistance():   
                # Updating predecessor and distance if the new cost is smaller
                next_vert.setPred(current_vert)
                next_vert.setDistance(new_cost)
                # Decreasing the key (updating priority) in the priority queue
                pq.decreaseKey(next_vert, new_cost)
