import json
import logging

def dict_to_embedded_json(dict_data: dict) -> str:
    """Converts a dictionary to a JSON string, removing outer curly braces."""
    return json.dumps(dict_data)[1: -1]

def convert_log_method(log_method):
    """Wrapper function to handle logging with JSON messages."""
    def inner_convert_log_method(level, msg, args, exc_info=None, extra=None, stack_info=False):
        if isinstance(msg, dict):
            json_msg = dict_to_embedded_json(msg)
        elif isinstance(msg, str):
            json_msg = dict_to_embedded_json({'msg': msg})
        else:
            json_msg = dict_to_embedded_json({'msg': str(msg)})
        return log_method(level, json_msg, args, exc_info, extra, stack_info)
    return inner_convert_log_method

class JsonLogger(logging.Logger):
    """Custom logger class that wraps logging methods to handle JSON messages."""
    def __getattribute__(self, item):
        if item == '_log':
            return convert_log_method(super(JsonLogger, self).__getattribute__(item))
        else:
            return super(JsonLogger, self).__getattribute__(item)
