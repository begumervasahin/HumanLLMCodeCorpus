import requests
from bs4 import BeautifulSoup
import re
import random
import logging
import time

# Configure logging
logging.basicConfig(filename='log.log',
                    filemode='w',
                    format='%(asctime)s %(message)s',
                    level=logging.DEBUG)
logging.info('Script started')

# Constants
BASE_URL = 'http://forum.pcgames.de/'
DELAY_RANGE = (0.8, 1.2)
PAGE_REGEX = re.compile(r'Seite\s\d+\svon\s(\d+)', re.I)

# Function to extract titles from page elements
def extract_titles(soup, tag, class_name):
    titles = []
    elements = soup.find_all(tag, class_=class_name)
    for element in elements:
        title_link = element.find('a')
        if title_link:
            titles.append(title_link.get('href'))
    return titles

# Function to retrieve page content and handle exceptions
def get_page_content(url):
    try:
        response = requests.get(url)
        response.raise_for_status()
        return BeautifulSoup(response.text, 'html.parser')
    except Exception as e:
        logging.error(f'ERROR: Failed to fetch {url} because of {e}')
        return None

# Function to simulate human-like delays
def simulate_human_delay():
    delay = random.uniform(*DELAY_RANGE)
    time.sleep(delay)

# Function to scrape pages for additional pages
def scrape_for_additional_pages(urls):
    updated_urls = []
    for url in urls:
        simulate_human_delay()
        page_soup = get_page_content(url)
        if page_soup:
            pagination_form = page_soup.find('form', class_='pagination')
            if pagination_form:
                num_pages = int(PAGE_REGEX.match(pagination_form.text).group(1))
                updated_urls.extend([f'{url}/index{page}.html' for page in range(1, num_pages + 1)])
            else:
                updated_urls.append(url)
    return updated_urls

# Retrieve main page content
main_page_soup = get_page_content(BASE_URL)
if main_page_soup:
    # Extract forum and subforum titles
    forum_titles = extract_titles(main_page_soup, 'h2', 'forumtitle')
    subforum_titles = extract_titles(main_page_soup, 'li', 'subforum')

    # Write forum and subforum titles to files
    with open('forumtitles.txt', 'w') as file:
        file.write('\n'.join(forum_titles))
    with open('subforumtitles.txt', 'w') as file:
        file.write('\n'.join(subforum_titles))

    # Scrape for additional pages of forum and subforum titles
    updated_forum_titles = scrape_for_additional_pages(forum_titles)
    updated_subforum_titles = scrape_for_additional_pages(subforum_titles)

    # Write updated forum and subforum titles to files
    with open('forumtitles_new.txt', 'w') as file:
        file.write('\n'.join(updated_forum_titles))
    with open('subforumtitles_new.txt', 'w') as file:
        file.write('\n'.join(updated_subforum_titles))

    # Combine all titles
    all_titles = updated_forum_titles + updated_subforum_titles

    # Write all titles to a file
    with open('all_forumtitles_new.txt', 'w') as file:
        file.write('\n'.join(all_titles))

    # Scrape for thread URLs
    thread_urls = scrape_for_additional_pages(all_titles)

    # Write thread URLs to a file
    with open('threadurls.txt', 'w') as file:
        file.write('\n'.join(thread_urls))

    # Scrape for additional pages of thread URLs
    updated_thread_urls = scrape_for_additional_pages(thread_urls)

    # Write updated thread URLs to a file
    with open('threadurls_new.txt', 'w') as file:
        file.write('\n'.join(updated_thread_urls))

logging.info('Script ended')
