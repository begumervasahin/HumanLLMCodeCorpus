# Importing necessary libraries
import pandas as pd
import numpy as np
from sklearn.decomposition import PCA
from xgboost import XGBClassifier

# Loading training and testing data
train_data = pd.read_csv('train.dat', header=None, sep=' ', quoting=3)
labels = pd.read_csv('train.labels', header=None, sep=' ', quoting=3)
test_data = pd.read_csv('test.dat', header=None, sep=' ', quoting=3)

# Splitting features and labels
X_train = train_data.iloc[:, :887]
y_train = labels.iloc[:, 0]
X_test = test_data.iloc[:, :887]

# Centering the data by subtracting mean
X_train_mean = np.mean(X_train, axis=0)
X_train -= X_train_mean
X_test_mean = np.mean(X_test, axis=0)
X_test -= X_test_mean

# Performing Principal Component Analysis (PCA) for dimensionality reduction
pca = PCA(n_components=28)
X_train = pca.fit_transform(X_train)
X_test = pca.transform(X_test)

# Training XGBoost classifier
xgb = XGBClassifier(learning_rate=0.3, n_estimators=2000, max_depth=7,
                    min_child_weight=1, gamma=0, subsample=0.8, colsample_bytree=0.8,
                    objective='binary:logistic', nthread=4, scale_pos_weight=1, seed=27)
xgb.fit(X_train, y_train)

# Making predictions on test data
y_pred = xgb.predict(X_test)

# Saving predictions to file
output = pd.DataFrame(data=y_pred)
output.to_csv("output.dat", index=False, quoting=3, header=None)
