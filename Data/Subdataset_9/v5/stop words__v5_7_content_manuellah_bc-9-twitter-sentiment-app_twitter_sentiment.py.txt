import tweepy
import progressbar
from colorama import init, Fore
from config import *

# Initialize colorama for colored output
init()

class TwitterSentiment:
    '''
    This class fetches frequent tweets of a particular Twitter user, analyzes the tweets,
    and returns the most frequently used words with their frequencies.
    '''
    auth = tweepy.OAuthHandler(API_KEYS, API_SECRET)
    auth.secure = True
    auth.set_access_token(ACCESS_TOKEN, ACCESS_TOKEN_SECRET)
    api = tweepy.API(auth)

    def __init__(self, username, max_tweets=3200):
        '''
        Initializes the TwitterSentiment object with the username and maximum number of tweets to fetch.
        '''
        self.username = username
        self.max_tweets = max_tweets

    def fetch_data(self):
        '''
        Fetches data from Twitter for the specified user and stores it in a dictionary.
        '''
        try:
            tweet_data = {}
            all_tweets = []
            print()

            while len(all_tweets) < self.max_tweets:
                count = min(self.max_tweets - len(all_tweets), 200)
                new_tweets = TwitterSentiment.api.user_timeline(self.username, count=count)

                if not new_tweets:
                    break

                all_tweets.extend(new_tweets)
                oldest = all_tweets[-1].id - 1

                for tweet in new_tweets:
                    tweet_data[tweet.id] = (tweet.text, tweet.created_at)

            return tweet_data
        except tweepy.error.TweepError as e:
            print(Fore.RED + "Error fetching tweets. Please double-check the Twitter handle and try again.")

    def display_tweets(self):
        '''
        Displays fetched tweets.
        '''
        tweet_dict = self.fetch_data()
        header = f'{Fore.BLUE}\nTWITTER ID'.ljust(30) + f'{Fore.BLUE}DATE POSTED'.ljust(30) + f'{Fore.BLUE}THE TWEET\n\n' + "==" * 70
        print(header)

        try:
            for key, (tweet_text, created_at) in tweet_dict.items():
                print(f'{Fore.YELLOW}{str(key).ljust(30)}{Fore.GREEN}{str(created_at).ljust(30)}{Fore.RED}{tweet_text}')
        except TypeError:
            print(Fore.RED + "No tweets available for this Twitter handle.")

    def analyse_data(self):
        '''
        Analyses the fetched tweets and returns the most frequent words (excluding stop words).
        '''
        tweet_dict = self.fetch_data()
        freq_dict = {}

        for key in tweet_dict:
            tweet_words = str(tweet_dict[key][0]).lower().split()
            for word in tweet_words:
                if word in stop_words:
                    continue
                freq_dict[word] = freq_dict.get(word, 0) + 1

        sorted_freq = sorted(freq_dict.items(), key=lambda x: x[1], reverse=True)[:10]

        print(f'{Fore.BLUE}\n\nCOMMON TWEETED WORD'.ljust(30) + f'{Fore.BLUE}THE FREQUENCY\n'.ljust(30) + '\n\n' + "==" * 70 + '\n\n')

        for word, freq in sorted_freq:
            print(f' {Fore.YELLOW}{word.ljust(30)}{Fore.GREEN}{str(freq).ljust(30)}')
