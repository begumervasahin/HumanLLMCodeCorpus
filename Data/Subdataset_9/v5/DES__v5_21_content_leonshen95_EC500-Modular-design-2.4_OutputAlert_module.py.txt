def receive_basic_input_data(signal_loss, shock_alert, oxygen_supply, fever, hypotension, hypertension):
    """
    Receive basic input data and format it into a dictionary.

    Args:
        signal_loss (bool): Indicates if there is a signal loss.
        shock_alert (bool): Indicates if there is a shock alert.
        oxygen_supply (bool): Indicates if there is an oxygen supply.
        fever (bool): Indicates if there is fever.
        hypotension (bool): Indicates if there is hypotension.
        hypertension (bool): Indicates if there is hypertension.

    Returns:
        dict: A dictionary containing the basic input data.
    """
    basic_result = {
        'Signal_Loss': signal_loss,
        'Shock_Alert': shock_alert,
        'Oxygen_Supply': oxygen_supply,
        'Fever': fever,
        'Hypotension': hypotension,
        'Hypertension': hypertension
    }
    return basic_result

def display_AI_input_data(blood_oxygen, blood_pressure, pulses):
    """
    Display AI input data and predictions.

    Args:
        blood_oxygen: Value representing blood oxygen level.
        blood_pressure: Value representing blood pressure.
        pulses: Value representing pulses.
    """
    pressure_predict_result, oxygen_predict_result, pulse_predict_result = AI_module.AI_Module(blood_oxygen, blood_pressure, pulses)

    print('Blood pressure prediction:')
    print(pressure_predict_result)
    print('Blood oxygen prediction:')
    print(oxygen_predict_result)
    print('Pulse prediction:')
    print(pulse_predict_result)
