import re
import os
import csv
from datetime import datetime
import math

# Define custom exceptions
class TransactionError(Exception):
    pass

class DateError(Exception):
    pass

# Initialize dictionaries to hold stock data, portfolio data, and transactions
stocks = {}
portfolio = {}
transactions = []

# Function to normalize date strings
def normalize_date(date_str):
    formats = ['%d.%m.%Y', '%Y-%m-%d', '%Y/%m/%d']
    for fmt in formats:
        try:
            return datetime.strptime(date_str, fmt).strftime("%Y-%m-%d")
        except ValueError:
            continue
    raise DateError("Invalid date format: {0}".format(date_str))

# Function to load stock data from CSV into dictionary
def load_stock(symbol):
    symbol_dict = {}
    symbol = symbol.upper()
    symbol_file = '{}.csv'.format(symbol)
    path = os.path.join('stockdata', symbol_file)
    if not os.path.exists(path):
        raise FileNotFoundError("Stock data file not found in stockdata directory")
    
    with open(path, mode="rt", encoding="utf8") as ifile:
        ifile.readline()
        data_reader = csv.reader(ifile, delimiter=",")
        for row in data_reader:
            date = normalize_date(row[0])
            try:
                row[1:5] = [float(val) for val in row[1:5]]
            except ValueError:
                raise ValueError("Invalid format in CSV file")
            symbol_dict[date] = row[1:5]
    stocks[symbol] = symbol_dict

# Function to load portfolio data from CSV file
def load_portfolio(fname='portfolio.csv'):
    portfolio.clear()
    transactions.clear()
    if not os.path.exists(fname):
        raise FileNotFoundError("Portfolio file not found")
    
    with open(fname, mode="rt", encoding="utf8") as f:
        f_reader = csv.reader(f, delimiter=",")
        for i, line in enumerate(f_reader):
            try:
                if i == 0:
                    portfolio["date"] = normalize_date(line[0])
                elif i == 1:
                    cash = float(line[0])
                    if cash < 0:
                        raise ValueError("Cash cannot be negative")
                    portfolio["cash"] = cash
                else:
                    symbol, volume = line
                    volume = int(volume)
                    portfolio[symbol] = volume
                    load_stock(symbol)
            except ValueError:
                raise ValueError("Invalid format in portfolio file")

# Function to evaluate portfolio value on a given date
def evaluate_portfolio(date=None, verbose=False):
    if date is None:
        date = portfolio.get('date')
    date = normalize_date(date)
    total_value = portfolio.get('cash')
    total_stock = [{'Capital type': 'Cash', 'Volume': 1, 'Val/Unit*': total_value, 'Value in £*': total_value}]
    
    if date < portfolio.get('date'):
        raise DateError("Date cannot be earlier than portfolio date")
    
    for stock, volume in portfolio.items():
        if stock in stocks:
            if date not in stocks[stock]:
                raise DateError("Date is not a trading day")
            
            unit_value = stocks[stock][date][2]
            value = volume * unit_value
            total_value += value
            total_stock.append({'Capital type': f'Shares of {stock}', 'Volume': volume, 'Val/Unit*': unit_value, 'Value in £*': value})
    
    if verbose:
        print(f"Portfolio on {date}:")
        print("[* share values based on the lowest price on {}]\n".format(date))
        print("{:<22} | {:<6} | {:<9} | {:^8}".format("Capital type", "Volume", "Val/Unit*", "Value in £*"))
        print("-" * 23 + "+" + "-" * 8 + "+" + "-" * 11 + "+" + "-" * 13)
        for item in total_stock:
            print("{Capital type:<22} | {Volume:>6} | {Val/Unit*:9.2f} | {Value in £*: 11.2f}".format(**item))
        print("-" * 23 + "+" + "-" * 8 + "+" + "-" * 11 + "+" + "-" * 13)
        print("TOTAL VALUE{:>46.2f}".format(total_value))
    
    return total_value

# Function to add a transaction to the portfolio
def add_transaction(trans, verbose=False):
    date = normalize_date(trans.get('date'))
    symbol = trans.get('symbol')
    volume = trans.get('volume')
    
    if symbol not in stocks:
        raise ValueError("Symbol not found in stock data")
    if date < portfolio['date']:
        raise DateError("Transaction date cannot be earlier than portfolio date")
    
    price = stocks[symbol][date][2 if volume < 0 else 1]
    total_price = price * volume
    cash = portfolio.get('cash') - total_price
    
    if cash < 0 or (symbol not in portfolio and volume < 0) or (symbol in portfolio and portfolio[symbol] + volume < 0):
        raise TransactionError("Not enough cash or shares to perform transaction")
    
    portfolio['date'] = date
    portfolio['cash'] = cash
    
    if symbol in portfolio:
        if portfolio[symbol] + volume != 0:
            portfolio[symbol] += volume
        else:
            del portfolio[symbol]
    else:
        portfolio[symbol] = volume
    
    transactions.append(trans)
    
    if verbose:
        action = 'Sold' if volume < 0 else 'Bought'
        remaining_cash = cash
        if volume < 0:
            remaining_cash = cash
        else:
            remaining_cash = cash
        print(f"{date}: {action} {abs(volume)} shares of {symbol} for a total of {abs(total_price)}")
        print(f"{'Available' if volume < 0 else 'Remaining'} cash: £ {remaining_cash:.2f}")

# Function to save the updated portfolio to a CSV file
def save_portfolio(fname="portfolio.csv"):
    with open(fname, mode="wt", encoding="utf8") as csv_file:
        f_writer = csv.writer(csv_file)
        for key, value in portfolio.items():
            f_writer.writerow([key, value])

# Function to sell all shares in portfolio on a particular date
def sell_all(date=None, verbose=False):
    if date is None:
        date = portfolio.get('date')
    date = normalize_date(date)
    
    for key in portfolio.copy():
        if key in stocks:
            key_trans = {'date': date, 'symbol': key, 'volume': -portfolio[key]}
            add_transaction(key_trans, verbose)

# Function to load historic stock data from subdirectory into dictionary
def load_all_stocks():
    all_files = [file for file in os.listdir('stockdata') if os.path.isfile(os.path.join('stockdata', file)) and re.search('\.csv', file)]
    
    for file in all_files:
        try:
            symbol = re.sub('\.csv$', '', file)
            load_stock(symbol)
        except ValueError:
            pass
