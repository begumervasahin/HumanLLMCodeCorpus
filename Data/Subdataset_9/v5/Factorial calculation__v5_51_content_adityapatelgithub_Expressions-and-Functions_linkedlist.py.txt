class LinkedList:
    class Node:
        def __init__(self, value, next_node=None):
            self.value = value
            self.next = next_node

    def __init__(self):
        self.head = None
        self.tail = None
        self.length = 0

    def __iter__(self):
        current = self.head
        while current:
            yield current.value
            current = current.next

    def __len__(self):
        return self.length

    def __str__(self):
        return " -> ".join(str(item) for item in self)

    def prepend(self, value):
        """Adds a new element at the beginning of the linked list."""
        new_node = self.Node(value, self.head)
        if not self.head:
            self.head = self.tail = new_node
        else:
            self.head = new_node
        self.length += 1

    def pop(self):
        """Removes and returns the first element of the linked list."""
        if not self.head:
            raise IndexError("pop from empty list")
        value = self.head.value
        self.head = self.head.next
        if not self.head:
            self.tail = None
        self.length -= 1
        return value

    def last(self):
        """Returns the value of the last element in the linked list."""
        if not self.tail:
            raise IndexError("list is empty")
        return self.tail.value

    def first(self):
        """Returns the value of the first element in the linked list without removing it."""
        if not self.head:
            raise IndexError("list is empty")
        return self.head.value

    def is_empty(self):
        """Checks whether the linked list is empty."""
        return self.length == 0
