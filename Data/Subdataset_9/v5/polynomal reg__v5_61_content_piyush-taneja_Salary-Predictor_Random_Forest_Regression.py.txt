# Importing necessary libraries
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd

# Load dataset
dataset = pd.read_csv('Position_Salaries.csv')

# Extract features and target variable
X = dataset.iloc[:, 1:2].values
y = dataset.iloc[:, 2].values

# Import RandomForestRegressor
from sklearn.ensemble import RandomForestRegressor

# Create regressor with specified parameters
regressor = RandomForestRegressor(n_estimators=300, random_state=0)

# Fit the regressor to the data
regressor.fit(X, y)

# Predict salary for a specific position level
predicted_salary = regressor.predict([[6.5]])

# Generate a smoother curve for visualization
X_grid = np.arange(min(X), max(X), 0.01).reshape(-1, 1)

# Visualize the results
plt.scatter(X, y, color='red', label='Actual Data')
plt.plot(X_grid, regressor.predict(X_grid), color='blue', label='Random Forest Regression')
plt.title('Truth or Bluff (Random Forest Regression)')
plt.xlabel('Position level')
plt.ylabel('Salary')
plt.legend()
plt.show()
