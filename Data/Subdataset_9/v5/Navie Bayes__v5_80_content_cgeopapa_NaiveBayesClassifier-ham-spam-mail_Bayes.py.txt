import csv
import string
from os import listdir
from os.path import isfile, join
from nltk.corpus import stopwords

# Function to save a dictionary to a CSV file
def save_dict_to_csv(filename, dictionary):
    with open(filename, "w") as file:
        writer = csv.writer(file)
        for key, val in dictionary.items():
            writer.writerow([key, val])

# Function to load a dictionary from a CSV file
def load_dict_from_csv(filename):
    dictionary = {}
    with open(filename, mode='r') as file:
        reader = csv.reader(file)
        for row in reader:
            if len(row) == 2:
                dictionary[row[0]] = float(row[1])
    return dictionary

# Function to clean and preprocess text data
def preprocess_text(text):
    text = text.lower()  # Convert text to lowercase
    translation_table = dict.fromkeys(map(ord, string.punctuation + '\0'), None)
    text = text.translate(translation_table)  # Remove punctuation and special characters
    stop_words = set(stopwords.words('english'))
    stop_words.add("subject")  # Add "subject" to stopwords
    text = text.replace('\n', ' ')
    words = text.split(' ')
    words = [word for word in words if word not in stop_words]  # Remove stopwords
    return words

# Function to train and build a dictionary of word frequencies
def train_word_frequencies(directory):
    files = [f for f in listdir(directory) if isfile(join(directory, f))]
    word_freq_dict = {}
    total_words = 0
    for file in files:
        with open(join(directory, file), errors="ignore") as f:
            text = f.read()
            words = preprocess_text(text)
            total_words += len(words)
            for word in words:
                word_freq_dict[word] = word_freq_dict.get(word, 0) + 1
    word_freq_dict = {word: freq / total_words for word, freq in word_freq_dict.items()}
    word_freq_dict["fileCount"] = len(files)
    return word_freq_dict

# Function to classify input text as spam or ham
def classify_text(input_text, word_freq_dict, priori):
    words = preprocess_text(input_text)
    probability = 1
    for word in words:
        probability *= word_freq_dict.get(word, 1 / (word_freq_dict["fileCount"] + len(word_freq_dict) + 1))
    probability *= priori
    return probability

# Function to train, save, and test the classifier
def train_save_and_test():
    # Train and save dictionaries for spam and ham
    spam_dict = train_word_frequencies("spam")
    ham_dict = train_word_frequencies("ham")
    save_dict_to_csv("spam_train_results.csv", spam_dict)
    save_dict_to_csv("ham_train_results.csv", ham_dict)

    # Test the classifier
    ham_dict = load_dict_from_csv("ham_train_results.csv")
    spam_dict = load_dict_from_csv("spam_train_results.csv")
    spam_count = ham_count = 0
    files = [f for f in listdir("input") if isfile(join("input", f))]
    for file in files:
        with open(join("input", file), errors="ignore") as f:
            text = f.read()
            ham_prob = classify_text(text, ham_dict, 0.67)
            spam_prob = classify_text(text, spam_dict, 0.33)
            if ham_prob <= spam_prob:
                spam_count += 1
            else:
                ham_count += 1
    print("Spam:\t" + str(ham_count))
    print("Ham:\t" + str(spam_count))

# Train, save, and test the classifier
train_save_and_test()
