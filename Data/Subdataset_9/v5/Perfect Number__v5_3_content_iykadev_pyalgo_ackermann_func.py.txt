import sys

# Set recursion limit to avoid maximum recursion depth error
sys.setrecursionlimit(3000)

def ackermann_function(m, n):
    """
    Compute the value of the Ackermann function for non-negative integers m and n.
    
    Parameters:
    - m: int, non-negative integer
    - n: int, non-negative integer
    
    Returns:
    - int, result of the Ackermann function for given m and n
    """
    if m == 0:
        # Base case: if m is 0, return n + 1
        return n + 1
    elif n == 0:
        # Recursive case: if n is 0, call ackermann_function with (m - 1, 1)
        return ackermann_function(m - 1, 1)
    else:
        # Recursive case: call ackermann_function with (m - 1, ackermann_function(m, n - 1))
        return ackermann_function(m - 1, ackermann_function(m, n - 1))

# Example usage
result = ackermann_function(3, 4)
print(result)
