import numpy as np
from itertools import combinations
from math import acos, sqrt, pow, pi, cos, sin
import Matrix

def compute_projection(A):
    """
    Compute the projection of matrix A.

    Parameters:
        A (numpy.ndarray): Input matrix.

    Returns:
        numpy.ndarray: Flattened projection of A.
    """
    interim = np.matmul(A.transpose(), A)
    return interim.flatten()

def decompose_q(q, d):
    """
    Decompose q.

    Parameters:
        q (numpy.ndarray): Input vector.
        d (int): Dimension.

    Returns:
        numpy.ndarray: Decomposed vector.
    """
    xxT = Matrix.getClosestSymMatrix(np.reshape(q, (d, d)))
    x = -xxT[d - 1]
    return x

def compute_q_component(x):
    """
    Compute the q component.

    Parameters:
        x (numpy.ndarray): Input vector.

    Returns:
        numpy.ndarray: Flattened q component.
    """
    interim = np.outer(x, x)
    return interim.flatten()

def cartesian_to_spherical(x):
    """
    Convert Cartesian coordinates to Spherical coordinates.

    Parameters:
        x (numpy.ndarray): Cartesian coordinates.

    Returns:
        numpy.ndarray: Spherical coordinates.
    """
    d = len(x)
    phi = np.zeros((d - 1))
    sum_squares = pow(x[-1], 2) + pow(x[-2], 2)
    last_angle = acos(x[-2] / sqrt(sum_squares))

    if x[-1] >= 0:
        phi[-1] = last_angle
    else:
        phi[-1] = 2 * pi - last_angle

    for k in range(d - 3, -1, -1):
        sum_squares += pow(x[k], 2)
        phi[k] = acos(x[k] / sqrt(sum_squares))

    return phi

def spherical_to_cartesian(phi):
    """
    Convert Spherical coordinates to Cartesian coordinates.

    Parameters:
        phi (numpy.ndarray): Spherical coordinates.

    Returns:
        numpy.ndarray: Cartesian coordinates.
    """
    dim = len(phi)
    x = np.zeros(dim + 1)
    sins = 1

    for i in range(dim):
        x[i] = sins * cos(phi[i])
        sins *= sin(phi[i])

    x[dim] = sins
    return x

def compute_alt_projection(A, j):
    """
    Compute alternative projection of matrix A.

    Parameters:
        A (numpy.ndarray): Input matrix.
        j (int): Dimension.

    Returns:
        numpy.ndarray: Flattened alternative projection of A.
    """
    d = len(A[0])
    p = np.array([])
    p_1 = np.reshape(np.multiply(A, A), (1, A.shape[0] * A.shape[1]))
    p = np.append(p, p_1)
    index = range(d + 1)
    comb = list(combinations(index, 2))

    for i in range(d - j):
        p_2 = 2 * np.multiply(A[i, [x[0] for x in comb]], A[i, [x[1] for x in comb]])
        p = np.append(p, p_2)

    return p

def decompose_alt_q(q, d, j):
    """
    Decompose alternative q.

    Parameters:
        q (numpy.ndarray): Input vector.
        d (int): Dimension.
        j (int): Dimension.

    Returns:
        numpy.ndarray: Decomposed vector.
    """
    x_ = np.sqrt(np.abs(q[:d]))
    signs = q[d * (d - j):d * (d - j) + d - 1]
    x = np.zeros(d)
    x[0] = np.absolute(x_[0])

    for i in range(1, len(x_)):
        x[i] = -np.absolute(x_[i]) if signs[i - 1] < 0 else np.absolute(x_[i])

    return x
