import requests
from bs4 import BeautifulSoup
import pandas as pd

# Define user-agent headers for making requests
headers = {
    'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/77.0.3865.120 Safari/537.36',
    'Referer': 'https://google.com',
    'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,image/webp,image/apng,*/*;q=0.8',
    'Accept-Encoding': 'gzip, deflate, br',
    'Accept-Language': 'en-US,en;q=0.9',
    'Pragma': 'no-cache',
    'Cache-Control': 'no-cache'
}

# URLs to scrape
urls = [
    "https://www.subaruofmiami.com/promotions/service/index.htm",
    "https://www.lehmansubaru.com/promotions/service/index.htm",
]

def scrape_promotions(url):
    # Initialize a dictionary to store scraped data
    scraped_data = {}

    # Make a request to the URL
    response = requests.get(url, timeout=15, headers=headers)

    # Parse the HTML content
    soup = BeautifulSoup(response.content, "html.parser")

    # Find promo title and price HTML elements
    promo_titles = soup.find_all("div", class_="promo-title text-center h1")
    promo_prices = soup.find_all("div", class_="promo-discountValue text-center h1 has-image")

    # Extract relevant data
    for title, price in zip(promo_titles, promo_prices):
        title_text = title.text.strip().lower()
        if "oil" in title_text:  # Check if the promo is for oil change
            scraped_data["url"] = url
            scraped_data["product"] = title_text
            scraped_data["price"] = price.text[1:8]  # Assuming the price is formatted consistently

    # Convert scraped data to DataFrame and display
    return pd.DataFrame(scraped_data, index=[0])

# Call the function to scrape data from respective websites
for url in urls:
    scraped_data = scrape_promotions(url)
    dealer_name = "Subaru of Miami" if "subaruofmiami" in url else "Lehman Subaru"
    print(f"Scraped data from {dealer_name}:\n{scraped_data}\n")
