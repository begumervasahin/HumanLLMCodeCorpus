import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
import datetime

def cluster_analysis(data_array, k):
    """
    Perform K-Means clustering on input data array and visualize the clusters.

    Parameters:
        data_array : array-like, shape (n_samples, n_features)
            Input data array containing images.
        k : int
            Number of clusters to form.

    Returns:
        labels : array, shape (n_samples,)
            Cluster labels for each sample.
    """
    # Initialize KMeans model
    model = KMeans(n_clusters=int(k))
    
    # Fit the model to the data and predict cluster labels
    labels = model.fit_predict(data_array)
    
    # Extract x and y coordinates from data_array
    xs = data_array[:, 0]
    ys = data_array[:, 1]
    
    # Get cluster centroids
    centroids = model.cluster_centers_
    centroids_x = centroids[:, 0]
    centroids_y = centroids[:, 1]
    
    # Plot data points with cluster labels and centroids
    plt.scatter(xs, ys, c=labels, alpha=0.5, label='Data Points')
    plt.scatter(centroids_x, centroids_y, marker='D', s=50, c='red', label='Centroids')
    
    # Set plot title with current date and time
    plt.title(datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S"))
    
    # Set axis labels
    plt.xlabel('X')
    plt.ylabel('Y')
    
    # Show legend
    plt.legend()
    
    # Display plot
    print("\nInspect the cluster plot shown.\nWhen ready, close the plot and the images will be sorted\n"
          "and copied into subdirectories.")
    plt.show()

    return labels
