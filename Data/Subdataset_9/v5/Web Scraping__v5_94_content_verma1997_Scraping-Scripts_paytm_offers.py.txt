def get_random_user_agent():
    """Selects a random user agent header."""
    return {'User-Agent': choice(USER_AGENTS)}

def scrape_offers(url):
    """Scrapes flight offers from the given URL."""
    response = requests.get(url, headers=get_random_user_agent())
    soup = BeautifulSoup(response.text, 'html.parser')
    
    offers = soup.find_all("li", class_="slider-slide")
    
    collected_offers = []
    for offer in offers:
        link_element = offer.find("a")
        link = link_element['href']
        image_src = link_element.img['src']
        promo_code_element = offer.find("p", class_="PromoCode")
        promo_code = promo_code_element.text.replace("Use promocode : ", "")
        
        collected_offers.append((link, image_src, promo_code))
    
    return collected_offers

def save_offers_to_csv(offers, filename="paytm_offers.csv"):
    """Saves the list of offers to a CSV file."""
    with open(filename, "a") as file:
        file.write("Link,Image Link,Promocode\n")
        for link, image, promocode in offers:
            file.write(f"{link},{image},{promocode}\n")

if __name__ == "__main__":
    url = "https://paytm.com/flights"
    offers = scrape_offers(url)
    save_offers_to_csv(offers)
