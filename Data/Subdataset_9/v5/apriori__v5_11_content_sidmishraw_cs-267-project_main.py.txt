import sys
import os
import numpy as np
import pandas as pd
from re import sub
from json import dumps, loads
from os.path import sep, exists
from nltk import download
from apriori import determine_word_positions, determine_doc_frequency, read_input_files
from simplicial_complex import SimplicialComplex

# Global variables
corpus = None
tokens = None
token_list = None
docs = None
df_output = None
props = None

# Constants
INPUT_FILEPATH = 'input_jsons_filepath'
SHARED_OBJ_FILEPATH = 'shared_obj_filepath'
OUTPUT_FILEPATH = 'output_filepath'

# Setting up NLTK stopwords
def nltk_stopwords_setup():
    if not exists('mystopwords/corpora/stopwords'):
        download(download_dir='mystopwords')

# Reading data and initializing the pipeline
def read_data_init():
    global corpus, token_list, docs, props

    with open('properties.json', 'r') as props_read:
        props = loads(props_read.read())

    corpus, tokens = read_input_files(props[INPUT_FILEPATH])
    token_list = list(tokens.keys())
    docs = sorted(corpus.keys())

# Computing term frequency
def compute_term_frequency():
    global docs, props

    term_frequency_with_positions = {}

    for doc in docs:
        term_frequency_with_positions[doc] = determine_word_positions(corpus[doc])

    with open(f'{props[OUTPUT_FILEPATH]}{sep}output_term_freq.json', 'w') as tf_file_output:
        tf_file_output.write(dumps(term_frequency_with_positions, sort_keys=True, indent=2))

# Computing document frequency
def compute_document_frequency():
    global docs, token_list, corpus, props, df_output

    df_output = determine_doc_frequency(docs, token_list, corpus)

    df_output.to_csv(f'{props[OUTPUT_FILEPATH]}{sep}output_doc_freq.csv', header=token_list, sep=',')
    np.savetxt(f'{props[OUTPUT_FILEPATH]}{sep}output_doc_freq.txt', df_output.values, fmt='%d')

# Processing simplicial complex
def process_simplical_complex():
    global df_output, props, token_list

    string_vector_all = ''

    for index, series in df_output.iterrows():
        whitespace_regex = r'\s+'
        string_vector = np.array_str(series.values)
        string_vector = sub(whitespace_regex, '', string_vector)
        string_vector_all += string_vector[1:len(token_list)]
        bit_vector_as_string = string_vector[1:(len(token_list))].encode('utf-8')

    bit_vector_as_string = string_vector_all.encode('utf-8')
    sc2 = SimplicialComplex(shared_obj_path=props[SHARED_OBJ_FILEPATH])
    sc2.directProcess(1, 0.05, len(token_list) - 1, len(docs), bit_vector_as_string)

# Main function
if __name__ == '__main__':
    # Set up NLTK stopwords
    nltk_stopwords_setup()
    # Read data and initialize
    read_data_init()
    # Compute term frequency
    compute_term_frequency()
    # Compute document frequency
    compute_document_frequency()
    # Process simplicial complex
    process_simplical_complex()
