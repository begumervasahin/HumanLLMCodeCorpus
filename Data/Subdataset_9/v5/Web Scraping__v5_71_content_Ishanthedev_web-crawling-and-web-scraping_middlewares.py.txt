from scrapy import signals

class QuotetutorialSpiderMiddleware:
    @classmethod
    def from_crawler(cls, crawler):
        instance = cls()
        instance.connect_signals(crawler)
        return instance
    
    def connect_signals(self, crawler):
        crawler.signals.connect(self.spider_opened, signal=signals.spider_opened)
    
    def process_spider_input(self, response, spider):
        # No processing needed, return None
        return None
    
    def process_spider_output(self, response, result, spider):
        # Yield each result item
        for item in result:
            yield item
    
    def process_spider_exception(self, response, exception, spider):
        # No processing needed, pass
        pass
    
    def process_start_requests(self, start_requests, spider):
        # Yield each start request
        for request in start_requests:
            yield request
    
    def spider_opened(self, spider):
        # Log spider opening
        spider.logger.info(f'Spider opened: {spider.name}')
