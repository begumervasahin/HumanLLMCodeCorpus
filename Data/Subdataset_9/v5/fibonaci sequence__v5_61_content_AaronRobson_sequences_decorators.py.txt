class Memoised:
    '''A decorator that caches a function's return value each time it is called.

    If called later with the same arguments, the cached value is returned, and
    not re-evaluated.
    
    See: http://en.wikipedia.org/wiki/Memoization
    '''
    def __init__(self, func):
        # Initialize the Memoised object with the provided function and an empty cache.
        self.func = func
        self.cache = {}

    def __call__(self, *args):
        try:
            # Check if the result for the given arguments is already in the cache.
            return self.cache[args]
        except KeyError:
            # If not, call the original function with the provided arguments.
            value = self.func(*args)
            # Cache the result for future use.
            self.cache[args] = value
            return value
        except TypeError:
            # If the arguments are not hashable, fall back to calling the original function.
            return self.func(*args)

    def __repr__(self):
        '''Return the function's docstring.'''
        # Return the docstring of the original function.
        return self.func.__doc__

# Alias for the Memoised class
# This allows using either memoized or memoised to refer to the Memoised decorator.
memoized = memoised = Memoised
