def bubble_sort(nums):
    """
    Sorts a list of numbers using the bubble sort algorithm.

    Args:
    - nums (list): The list of numbers to be sorted.

    Returns:
    - None: The sorting is done in-place, and the input list is updated directly.
    """
    n = len(nums)
    for i in range(n - 1, 0, -1):
        for j in range(i):
            if nums[j] > nums[j + 1]:
                # Swap elements if they are in the wrong order
                nums[j], nums[j + 1] = nums[j + 1], nums[j]
        # Print the list after each pass (optional)
        print(nums)

# Example usage:
nums = [5, 3, 8, 6, 7, 2]
print("Unsorted List = ", nums)
bubble_sort(nums)
print("Sorted List = ", nums)
