from random import randint
import os

# Define the character list for encryption
CHARACTER_LIST = 'aAbBcCdDeEfFgGhHiIjJkKlLmMnNoOpPqQrRsStTuUvVwWxXyYzZ,./\;:"[]{}()_-!@'

def generate_one_time_pads(sheets, length):
    """Generate one-time pads and save them to files."""
    for sheet in range(sheets):
        with open(f"otp{sheet}.txt", "w") as f:
            for _ in range(length):
                f.write(str(randint(0, len(CHARACTER_LIST)-1)) + "\n")

def load_sheet(filename):
    """Load a one-time pad from a file."""
    with open(filename, "r") as f:
        contents = f.read().splitlines()
    return contents

def get_plaintext():
    """Get the plaintext from user input."""
    return input('Enter your message: ')

def load_file(filename):
    """Load data from a file."""
    with open(filename, 'r') as f:
        contents = f.read()
    return contents

def save_file(filename, data):
    """Save data to a file."""
    with open(filename, 'w') as f:
        f.write(data)

def encrypt(plaintext, sheet):
    """Encrypt plaintext using a one-time pad."""
    ciphertext = ''
    for position, character in enumerate(plaintext):
        if character not in CHARACTER_LIST:
            ciphertext += character
        else:
            encrypted = (CHARACTER_LIST.index(character) + int(sheet[position])) % len(CHARACTER_LIST)
            ciphertext += CHARACTER_LIST[encrypted]
    return ciphertext

def decrypt(ciphertext, sheet):
    """Decrypt ciphertext using a one-time pad."""
    plaintext = ''
    for position, character in enumerate(ciphertext):
        if character not in CHARACTER_LIST:
            plaintext += character
        else:
            decrypted = (CHARACTER_LIST.index(character) - int(sheet[position])) % len(CHARACTER_LIST)
            plaintext += CHARACTER_LIST[decrypted]
    return plaintext

def menu():
    """Display the menu and handle user choices."""
    while True:
        print('1. Generate one-time pads')
        print('2. Encrypt a message')
        print('3. Decrypt a message')
        print('4. Quit program')
        choice = input('Enter number: ')
        if choice == '1':
            sheets = int(input('How many OTPs should be generated? '))
            length = int(input('What will be the maximum message length? '))
            generate_one_time_pads(sheets, length)
        elif choice == '2':
            filename = input('Enter filename of the OTP you want to use: ')
            sheet = load_sheet(filename)
            plaintext = get_plaintext()
            ciphertext = encrypt(plaintext, sheet)
            filename = input('Enter name of encrypted file: ')
            save_file(filename, ciphertext)
        elif choice == '3':
            filename = input('Enter filename of the OTP you want to use: ')
            sheet = load_sheet(filename)
            filename = input('Type the name of the file to be decrypted: ')
            ciphertext = load_file(filename)
            plaintext = decrypt(ciphertext, sheet)
            print('Decrypted Message: \n' + plaintext)
        elif choice == '4':
            break

if __name__ == '__main__':
    menu()
