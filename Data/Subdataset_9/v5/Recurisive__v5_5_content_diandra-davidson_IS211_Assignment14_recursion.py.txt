def fibonacci(num):
    """Calculate the Fibonacci sequence up to the given number."""
    if num <= 1:
        return num
    else:
        return fibonacci(num - 1) + fibonacci(num - 2)

def gcd(a, b):
    """Calculate the Greatest Common Divisor (GCD) of two numbers."""
    if b > a:
        return gcd(b, a)
    r = a % b
    if r == 0:
        return b
    else:
        return gcd(b, r)

def compare_strings(s1, s2):
    """
    Compare two strings lexicographically.
    Returns:
        -1 if s1 < s2
         0 if s1 == s2
         1 if s1 > s2
    """
    if len(s1) < len(s2):
        return -1
    elif len(s1) == len(s2):
        return 0
    elif len(s1) > len(s2):
        return 1
