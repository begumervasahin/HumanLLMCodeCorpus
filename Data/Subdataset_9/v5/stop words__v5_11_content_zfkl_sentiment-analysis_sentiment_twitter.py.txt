# Function to clean tweet text
def clean_tweet(tweet):
    try:
        tweet = re.sub('[\s]+', ' ', tweet)
        tweet = re.sub('((www\.[^\s]+)|(https://[^\s]+))', '', tweet)
        tweet = re.sub(r'<.*?>', '', tweet)
        tweet = re.sub("[!,?,\",\']", " ", tweet)
        tweet = re.sub(r"<br /><br />", " ", tweet)
    except:
        pass
    return tweet
