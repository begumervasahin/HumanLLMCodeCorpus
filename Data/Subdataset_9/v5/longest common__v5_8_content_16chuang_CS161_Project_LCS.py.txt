import sys
import numpy as np

# Initialize a 2D array of zeros to store intermediate results
memoization_table = np.zeros((2048, 2048), dtype=int)

# Function to compute the length of the longest common subsequence (LCS) of two strings A and B
def longest_common_subsequence(A, B):
    len_A = len(A)
    len_B = len(B)
    
    # Iterate through each character of both strings
    for i in range(1, len_A + 1):
        for j in range(1, len_B + 1):
            if A[i - 1] == B[j - 1]:
                # If characters match, increment the LCS length
                memoization_table[i][j] = memoization_table[i - 1][j - 1] + 1
            else:
                # If characters don't match, take the maximum LCS length from the previous positions
                memoization_table[i][j] = max(memoization_table[i - 1][j], memoization_table[i][j - 1])
    
    # Return the length of the LCS of strings A and B
    return memoization_table[len_A][len_B]

# Main function to read input from standard input and compute LCS
def main():
    # Check if command line arguments are provided
    if len(sys.argv) != 1:
        sys.exit('Usage: `python LCS.py < input`')
    
    # Read input from standard input
    for line in sys.stdin:
        A, B = line.split()
        # Compute and print the length of the longest common subsequence of strings A and B
        print(longest_common_subsequence(A, B))

# Check if the script is being run as the main program
if __name__ == '__main__':
    main()
