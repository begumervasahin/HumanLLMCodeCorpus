import os
import sys

def set_django_settings_module():
    """
    Ensure that the Django settings module is properly set.
    """
    os.environ.setdefault("DJANGO_SETTINGS_MODULE", "webservice.settings")

def execute_django_management_commands():
    """
    Try to execute Django management commands from the command line arguments.
    """
    try:
        from django.core.management import execute_from_command_line
        execute_from_command_line(sys.argv)
    except ImportError as e:
        # Handle the case where Django is not installed
        handle_django_import_error(e)

def handle_django_import_error(error):
    """
    Handle ImportError related to Django import failure.
    """
    try:
        import django
    except ImportError:
        # Raise error if Django is not installed or not found in PYTHONPATH
        raise ImportError(
            "Couldn't import Django. Please ensure it's installed "
            "and available on your PYTHONPATH environment variable. "
            "Did you forget to activate a virtual environment?"
        ) from error

if __name__ == "__main__":
    set_django_settings_module()
    execute_django_management_commands()
