def longest_common_subsequence(s1, s2):
    # Initialize variables
    num_rows = len(s2) + 1
    num_cols = len(s1) + 1
    costs = create_table(num_rows, num_cols, 0)
    directions = create_table(num_rows, num_cols, "A")
    
    # Initialize the first row and column of costs and directions
    initialize_first_row_and_column(costs, directions)
    
    # Find the longest common subsequence
    max_row, max_col, max_val = find_longest_subsequence(s1, s2, costs, directions)
    
    # Print tables and alignment
    print_tables_and_alignment(costs, directions, s1, s2, max_row, max_col)
    
    # Return the LCS score
    return costs[max_row][max_col]

def create_table(num_rows, num_cols, value):
    # Create a 2D table with specified dimensions and initial value
    return [[value] * num_cols for _ in range(num_rows)]

def initialize_first_row_and_column(costs, directions):
    # Initialize the first row and column of costs and directions
    num_rows, num_cols = len(costs), len(costs[0])
    for i in range(num_cols):
        costs[0][i] = 0
        directions[0][i] = "F"
    for j in range(1, num_rows):
        costs[j][0] = 0
        directions[j][0] = "F"

def find_longest_subsequence(s1, s2, costs, directions):
    # Find the longest common subsequence
    max_row, max_col, max_val = 0, 0, 0
    for y in range(1, len(costs)):
        for x in range(1, len(costs[0])):
            val_top = costs[y - 1][x]
            val_left = costs[y][x - 1]
            val_diag = costs[y - 1][x - 1] + (s1[x - 1] == s2[y - 1])
            val = max(val_top, val_left, val_diag)
            costs[y][x] = val
            if val > max_val:
                max_val = val
                max_row, max_col = y, x
            directions[y][x] = "D" if val == val_diag else ("L" if val == val_left else "T")
    return max_row, max_col, max_val

def print_tables_and_alignment(costs, directions, s1, s2, max_row, max_col):
    # Print tables and alignment
    print_table(costs, "costs.txt")
    print_table(directions, "directions.txt")
    align_and_write(directions, s1, s2, max_row, max_col, "alignment.txt")

def print_table(table, filename):
    # Print the table to a file
    with open(filename, 'w') as file:
        for row in table:
            file.write('\t'.join(map(str, row)) + '\n')

def align_and_write(directions, s1, s2, row, col, filename):
    # Align the sequences and write to a file
    with open(filename, 'w') as file:
        x, y = col, row
        lcs = ""
        while directions[y][x] != "F":
            if directions[y][x] == "T":
                y -= 1
            elif directions[y][x] == "L":
                x -= 1
            elif directions[y][x] == "D":
                lcs = s1[x - 1] + lcs
                x -= 1
                y -= 1
        # Write aligned sequences to file, with 50 characters per line
        for i in range(0, len(lcs), 50):
            file.write(lcs[i:i + 50] + "\n")

optimal_score = longest_common_subsequence(s, t)
print(s)
