from releash import *

# Create a GitPush release target
git_push = ReleaseTargetGitPush()

def define_package(package_name, path_prefix='vaex-', distribution_name=None):
    name = package_name[5:]
    path = f"packages/{path_prefix}{name}"
    distribution_name = distribution_name if distribution_name else package_name
    package = add_package(path, package_name, f"vaex.{name}", distribution_name=distribution_name)

    version_path = '{path}/vaex_' + name + '/_version.py' if name == 'arrow' else '{path}/vaex/' + name + '/_version.py'
    version = VersionSource(package, version_path)
    package.version_source = version

    version_target_path = version_path if name == 'arrow' else '{path}/vaex/' + name + '/_version.py'
    package.version_targets.append(VersionTarget(package, version_target_path))

    gittag = ReleaseTargetGitTagVersion(version_source=version, prefix=name + '-v', msg=f'Release {{version}} of {distribution_name}')
    package.release_targets.extend([gittag, ReleaseTargetSourceDist(package), git_push])

    if name == 'meta':
        feedstock_name = 'vaex' + '-feedstock'
    else:
        feedstock_name = f'vaex-{name}-feedstock'

    package.release_targets.append(ReleaseTargetCondaForge(package, f'../feedstocks/{feedstock_name}'))

    return package

# Define the 'vaex-core' package
core = define_package('vaex-core', 'vaex-core')

# Define other packages
package_names = ['vaex-meta', 'vaex-viz', 'vaex-hdf5', 'vaex-server', 'vaex-astro', 'vaex-ui', 'vaex-jupyter', 'vaex-distributed', 'vaex-arrow', 'vaex-ml']

for package_name in package_names:
    define_package(package_name)
