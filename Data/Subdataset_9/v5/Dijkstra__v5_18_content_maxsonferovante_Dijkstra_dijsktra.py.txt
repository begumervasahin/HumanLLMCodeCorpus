import matplotlib.pyplot as plt
import networkx as nx
import numpy as np
from heapq import heappush, heappop
import timeit

def draw_graph(graph, export=False, name='graph'):
    """Draws the graph with node labels and optionally exports it."""
    temp_graph = graph.copy()
    for node in temp_graph.nodes():
        temp_graph.node[node]['label'] = node

    pos = nx.spring_layout(temp_graph)
    nx.draw(temp_graph, pos)
    node_labels = nx.get_node_attributes(temp_graph, 'label')
    nx.draw_networkx_labels(temp_graph, pos, labels=node_labels)

    if export:
        plt.savefig(name + '.png')
    else:
        plt.show()
    plt.close()

def dijkstra(graph, sources):
    """Runs Dijkstra's algorithm on the graph from the given sources."""
    temp_graph = graph.copy()
    push, pop = heappush, heappop
    nodes = temp_graph.nodes()

    # Initialize node attributes for Dijkstra's algorithm
    for node in nodes:
        temp_graph.node[node]['lambda'] = np.inf
        temp_graph.node[node]['pi'] = None

    # Set initial lambda values for source nodes
    for source in sources:
        temp_graph.node[source]['lambda'] = 0

    queue, visited = [], []
    for node in nodes:
        push(queue, (temp_graph.node[node]['lambda'], node))

    while queue:
        current_node = pop(queue)
        current_node = current_node[1]
        visited.append(current_node)

        # Update lambda values of neighbors if necessary
        for neighbor in temp_graph.neighbors(current_node):
            if neighbor not in visited and temp_graph.node[neighbor]['lambda'] > (
                temp_graph.node[current_node]['lambda'] + graph[current_node][neighbor]['_weight']
            ):
                queue.remove((temp_graph.node[neighbor]['lambda'], neighbor))
                temp_graph.node[neighbor]['lambda'] = temp_graph.node[current_node]['lambda'] + graph[current_node][neighbor]['_weight']
                push(queue, (temp_graph.node[neighbor]['lambda'], neighbor))
                temp_graph.node[neighbor]['pi'] = current_node

    # Construct and return the resulting shortest path tree
    shortest_path_tree = nx.Graph()
    for node in temp_graph.nodes():
        shortest_path_tree.add_node(node)
        if temp_graph.node[node]['pi'] is not None:
            shortest_path_tree.add_edge(node, temp_graph.node[node]['pi'])
            shortest_path_tree[node][temp_graph.node[node]['pi']]['_weight'] = graph[node][temp_graph.node[node]['pi']]['_weight']

    return shortest_path_tree

def read_weighted_adjacency_matrix(path):
    """Reads a weighted adjacency matrix from a file and creates a graph."""
    data = np.loadtxt(path)
    rows, cols = np.where(data > 0)
    edges = zip(rows, cols)
    graph = nx.Graph(edges)
    for u, v in zip(rows, cols):
        graph[u][v]['weight'] = data[u][v]
        graph[u][v]['_weight'] = data[u][v]
    return graph

input_file = "adjacentes1.txt"
graph = read_weighted_adjacency_matrix(input_file)
sources = []  # Add source nodes here if needed
execution_times = []

# Run Dijkstra's algorithm multiple times and calculate average execution time
for _ in range(10):
    start_time = timeit.default_timer()
    dijkstra(graph, sources)
    end_time = timeit.default_timer()
