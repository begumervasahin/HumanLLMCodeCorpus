from flask import Flask, render_template, request
from graph.weighted_graph import weighted_graph
import time

# List of supported currencies
SUPPORTED_CURRS = ['EUR', 'USD', 'JPY', 'CHF', 'AUD', 'CAD', 'NZD', 'GBP', 'SEK', 'NOK', 'MXN', 'TRY', 'ZAR', 'CNH', 'XAU', 'XAG']
SUPPORTED_CURRS_STRING = ', '.join(SUPPORTED_CURRS)

app = Flask(__name__)

@app.route('/', methods=['GET', 'POST'])
def currencies():
    currencies_string = ""
    opportunity_string = ""
    temp_list = []
    opportunities = []

    if request.method == 'POST':
        currencies_string = request.form['currs'].upper().replace(" ", "")
        temp_list = currencies_string.split(',')
        curr_list = [currency for currency in temp_list if currency in SUPPORTED_CURRS]
        curr_list = list(set(curr_list))
        currencies_string = ', '.join(curr_list)

        if len(curr_list) > 1:
            graph = weighted_graph(curr_list)
            opportunities = graph.show_arbitrage_opportunities()
            time_epoch = graph.get_time()
            time_string = time.strftime("%Z - %Y/%m/%d, %H:%M:%S", time.localtime(time_epoch))

            if len(opportunities) == 0:
                opportunity_string = f"No arbitrage opportunity ({time_string})"
            else:
                opp_string = ' -> '.join(opportunities[0])
                opportunity_string = f"{opp_string} at {time_string}"
        else:
            opportunity_string = "No arbitrage opportunity available with less than two currencies..."
    else:
        opportunity_string = "No arbitrage opportunity available with less than two currencies..."

    return render_template('home.html', currencies=currencies_string, support_currs=SUPPORTED_CURRS_STRING, opportunity=opportunity_string)

if __name__ == '__main__':
    app.run(debug=True)
