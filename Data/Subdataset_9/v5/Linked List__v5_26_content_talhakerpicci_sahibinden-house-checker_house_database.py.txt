import sqlite3

class House:
    def __init__(self, house_link, title, price, m2, date, neighborhood, room, img):
        """Initialize a House object with attributes."""
        self.house_link = house_link
        self.title = title
        self.price = price
        self.m2 = m2
        self.date = date
        self.neighborhood = neighborhood
        self.room = room
        self.img = img

class DatabasePost:
    def __init__(self):
        """Initialize a DatabasePost object and connect to the database."""
        self.connect_database()

    def connect_database(self):
        """Connect to the SQLite database and create necessary tables if they do not exist."""
        self.connection = sqlite3.connect("Sahibinden.db")
        self.cursor = self.connection.cursor()
        query = """
                CREATE TABLE IF NOT EXISTS Tbl_Posts (
                    Link TEXT,
                    Title TEXT,
                    Price TEXT,
                    M2 TEXT,
                    Date TEXT,
                    Neighborhood TEXT,
                    Room TEXT,
                    Img TEXT
                );
                CREATE TABLE IF NOT EXISTS Tbl_Posts_Backup (
                    Link TEXT,
                    Title TEXT,
                    Price TEXT,
                    M2 TEXT,
                    Date TEXT,
                    Neighborhood TEXT,
                    Room TEXT,
                    Img TEXT
                );
                """
        self.cursor.executescript(query)
        self.connection.commit()

    def check_if_house_exists(self, link):
        """Check if a house with the given link exists in the database."""
        query = "SELECT * FROM Tbl_Posts WHERE Link = ?"
        self.cursor.execute(query, (link,))
        posts = self.cursor.fetchall()
        return len(posts) > 0

    def add_house(self, house, backup=False):
        """Add a house to the database."""
        table_name = "Tbl_Posts_Backup" if backup else "Tbl_Posts"
        query = f"INSERT INTO {table_name} VALUES (?, ?, ?, ?, ?, ?, ?, ?)"
        self.cursor.execute(query, (house.house_link, house.title, house.price, house.m2, house.date,
                                    house.neighborhood, house.room, house.img))
        self.connection.commit()

    def clear_posts(self, day):
        """Delete posts from the database based on a given date."""
        query = f"DELETE FROM Tbl_Posts WHERE Date LIKE '%{day}%'"
        self.cursor.execute(query)
        self.connection.commit()
