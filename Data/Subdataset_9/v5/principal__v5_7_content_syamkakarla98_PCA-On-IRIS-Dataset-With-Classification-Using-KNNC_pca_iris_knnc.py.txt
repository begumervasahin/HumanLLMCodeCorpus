# Import necessary libraries
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.neighbors import KNeighborsClassifier
from sklearn import metrics

# Load the Iris dataset
df = pd.read_csv("D:\Python_programs\ML\iris.csv")

# Define feature columns and target variable
feature_columns = ['sepal length', 'sepal width', 'petal length', 'petal width']
X = df[feature_columns].values
y = df['target'].values

# Split the dataset into training and testing sets
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=100)

# Flatten the target arrays
y_train = y_train.ravel()
y_test = y_test.ravel()

# Initialize and train a K-Nearest Neighbors model
knn_model = KNeighborsClassifier()
knn_model.fit(X_train, y_train)

# Make predictions on the test set
y_pred = knn_model.predict(X_test)

# Evaluate the model's accuracy before applying PCA
print('*' * 11, 'Accuracy of IRIS Dataset Before PCA', '*' * 11, '\n')
for k_value in range(1, 26):
    # Initialize KNN classifier with current K value
    knn_classifier = KNeighborsClassifier(n_neighbors=k_value, weights='uniform', algorithm='auto')
    # Train the model
    knn_classifier.fit(X_train, y_train)
    # Make predictions
    y_pred = knn_classifier.predict(X_test)
    # Calculate and print accuracy
    accuracy = metrics.accuracy_score(y_test, y_pred) * 100
    print(f"Accuracy is {accuracy:.2f}% for K-Value: {k_value}")
