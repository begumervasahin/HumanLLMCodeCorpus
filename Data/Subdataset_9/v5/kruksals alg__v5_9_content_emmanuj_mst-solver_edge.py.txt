class Edge:
    """
    Represents an edge in a graph.

    Attributes:
        u (any): The starting vertex of the edge.
        v (any): The ending vertex of the edge.
        weight (float): The weight of the edge.
    """

    def __init__(self, start_vertex, end_vertex, weight):
        """
        Initializes an edge with the given attributes.

        Parameters:
            start_vertex (any): The starting vertex of the edge.
            end_vertex (any): The ending vertex of the edge.
            weight (float): The weight of the edge.
        """
        self.u = start_vertex
        self.v = end_vertex
        self.weight = weight

    def __repr__(self):
        """
        Returns a string representation of the edge.

        Returns:
            str: A string representation of the edge.
        """
        return f"Edge({self.u}, {self.v}, {self.weight})"
