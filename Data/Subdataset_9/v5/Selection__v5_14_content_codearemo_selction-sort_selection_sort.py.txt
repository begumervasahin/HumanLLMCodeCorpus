def selection_sort(items):
    """
    Sorts a list of items using the selection sort algorithm.

    Args:
    - items: The list of items to be sorted.

    Returns:
    - None (the list is sorted in place).
    """
    # Iterate over each index in the list
    for step in range(len(items)):
        # Assume the current index has the smallest value
        location_of_smallest = step
        # Find the index of the smallest value in the unsorted portion
        for location in range(step, len(items)):
            if items[location] < items[location_of_smallest]:
                location_of_smallest = location
        # Swap the current index with the smallest value found
        items[step], items[location_of_smallest] = items[location_of_smallest], items[step]

    # Print the sorted list and its length
    print("Sorted items:", items)
    print("Total number of items:", len(items))

if __name__ == "__main__":
    # Example usage
    items = [64, 25, 12, 22, 11]
    selection_sort(items.copy())  # Use copy to keep the original list unchanged
