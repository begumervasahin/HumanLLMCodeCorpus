import os

# Ignore warnings
if not sys.warnoptions:
    warnings.simplefilter("ignore")

# Function to get user input
def get_user_input(prompt):
    return input(prompt).upper()

# Function to determine if it's a diploma course
def is_diploma(low):
    return 'Y' if low >= 400 else 'N'

# Function to set semester specific settings
def set_semester_settings(semc, dip):
    subcode = 52
    iloop = 8
    if semc in ['3', '4']:
        iloop = 9
        subcode = 58
        if dip == 'Y':
            iloop = 10
            subcode = 64
    return subcode, iloop

# Set up Tesseract OCR
pytesseract.pytesseract.tesseract_cmd = r'C:\Program Files (x86)\Tesseract-OCR\tesseract'

# Get user inputs
college = get_user_input("Enter the college code\n")
year = get_user_input('Enter the year\n')
branch = get_user_input('Please enter the branch\n')
low = int(input('Enter starting USN\n'))
high = int(input('Enter last USN\n')) + 1
semc = get_user_input('Enter the Semester\n')

# Determine if it's a diploma course
dip = is_diploma(low)

# Set semester specific settings
subcode, iloop = set_semester_settings(semc, dip)

# Open a file to write marks
with open('marks.txt', 'w+') as f:
    # Set up webdriver
    driver = webdriver.Chrome('C:\Program Files (x86)\chromedriver_win32\chromedriver.exe')
    
    # Loop through each USN
    for u in range(low, high):
        usn_suffix = '00' + str(u) if u < 10 else ('0' + str(u) if u < 100 else str(u))
        usn = '1' + college + year + branch + usn_suffix

        cv2.imwrite('cap.png', crop_img)
        tex = pytesseract.image_to_string(Image.open('cap.png'))
        tex = tex.strip(',')
        tex = tex.strip(' ')
        
        # Input captcha
        try:
            captcha = int(tex)
        except:
            print("Invalid CAPTCHA Detected.")
            continue
        us = driver.find_element_by_name("lns")

        us.send_keys(usn)
        cap.send_keys(captcha)
        driver.find_element_by_id("submit").click()
        
        # Parse result page
        try:
            soup = BeautifulSoup(driver.page_source)
        except:
            alert = driver.switch_to.alert
            if alert.text == "University Seat Number is not available or Invalid..!":
                print("No results for : " + usn + "\n")
                alert.accept()
                continue
            elif alert.text == "Invalid captcha code !!!":
                print("Invalid CAPTCHA Detected for USN : " + usn + "\n") 
                alert.accept()
                continue
        
        # Extract data from parsed page
        tds = soup.find_all('td')
        ths = soup.find_all('th')
        divs = soup.find_all('div', attrs={'class': 'col-md-12'})
        divCell = soup.find_all('div', attrs={'class': 'divTableCell'})   
        
        # Process extracted data
        try:
            sem = divs[5].div.text.strip('Semester : ')
        except AttributeError:
            print("INVALID USN/ INCOMPATIBLE DATA : " + usn + "\n")       
            continue
        if tds[0].text != 'University Seat Number ' or sem != semc:       
            print("INVALID USN/ INCOMPATIBLE DATA : " + usn + "\n")       
            continue
        record = ''
        record += re.sub('[!@
        record += ','
        record += re.sub('[!@
        record += ','
        sortList1 = [divCell[i].text[-3:] if divCell[i].text[-3:].isdigit() else divCell[i].text[-2:] for i in range(6, subcode, 6)]
        sortList1.sort()
        ilist = [j for i in range(0, iloop) for j in range(6, subcode, 6) if (divCell[j].text[-3:] if divCell[j].text[-3:].isdigit() else divCell[j].text[-2:]) == sortList1[i]]
        
        print(record, end='\t')
        try:
            for l in ilist:
                for j in range(l, l + 6):
                    if j == l + 1:
                        continue
                    else:
                        char = divCell[j].text
                        if char.isdigit():
                            record += str(int(char)) + ','        
                        else:
                            record += char + ','
                        print(divCell[j].text, end='\t\t')
                        if j == l + 5:
                            pf += divCell[j].text + ','
            f.write(record + '\n')
            print('\n')
        except IndexError:
            pass
    driver.quit()

# Export marks to Excel
import xlwt
book = xlwt.Workbook()
ws = book.add_sheet('Sheet1')
f = open('marks.txt', 'r+')
alignment = xlwt.Alignment()
font = xlwt.Font()
alignment.horz = xlwt.Alignment.HORZ_CENTER
font.bold = True
style = xlwt.XFStyle()
data = f.readlines()
for i in range(len(data)):
    row = data[i].split(',')
    for j in range(len(row)):
        if row[j].isdigit():
            ws.write(i, j, int(row[j]), style)
        else:
            ws.write(i, j, row[j], style)
pth = 'ExcelFiles/'
if not os.path.exists(pth):
    os.makedirs(pth)
filename = pth + '1' + college + year + branch + str(low) + '-' + str(high - 1) + ('.xls' if dip == 'N' else 'DIP.xls')
book.save(filename)
f.close()
