def check_condition(a, b, k, m):
    """Check if a[k-1] is less than b[-m]."""
    if a[k - 1] < b[-m]:
        return "YES"
    else:
        return "NO"

def find_min_a_max_b_index(a, b):
    """Find the index where minimum value of a and maximum value of b occur simultaneously."""
    min_value, max_value = min(a), max(b)
    for i, (x, y) in enumerate(zip(a, b)):
        if x == min_value and y == max_value:
            return i + 1
    return -1

def calculate_final_lists(a, b):
    """Determine the final lists."""
    n = len(a)
    final_1 = [False] * n
    final_2 = [False] * n
    i = j = 0
    chosen = 0
    while chosen < n:
        if a[i] < b[j]:
            final_1[i] = True
            i += 1
        else:
            final_2[j] = True
            j += 1
        chosen += 1
    return final_1, final_2

def print_final_lists(final_1, final_2, n):
    """Print the final lists."""
    for i in range(n):
        print(1 if final_1[i] or i < n // 2 else 0, end='')
    print()
    for i in range(n):
        print(1 if final_2[i] or i < n // 2 else 0, end='')

def calculate_ways(n, m, a):
    """Calculate the number of ways."""
    count = [0] * (m + 1)
    for genre in a:
        count[genre] += 1

    ways = 0
    for i in range(1, m):
        for j in range(i + 1, m + 1):
            ways += count[i] * count[j]
    return ways
