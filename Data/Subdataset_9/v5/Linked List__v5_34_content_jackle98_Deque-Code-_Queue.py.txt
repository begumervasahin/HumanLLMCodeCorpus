from Deque_Generator import get_deque

class Queue:
    """
    Implementation of a queue using a deque.
    """

    def __init__(self):
        """
        Initialize the queue.
        """
        self._deque = get_deque(1)

    def __str__(self):
        """
        Return a string representation of the queue.
        """
        return str(self._deque)

    def __len__(self):
        """
        Return the length of the queue.
        """
        return len(self._deque)

    def enqueue(self, value):
        """
        Add an element to the back of the queue.

        Args:
            value: The value to be added to the queue.
        """
        self._deque.push_back(value)

    def dequeue(self):
        """
        Remove and return the element from the front of the queue.

        Returns:
            The element removed from the front of the queue, or None if the queue is empty.
        """
        if not self._deque:
            return None
        return self._deque.pop_front()
