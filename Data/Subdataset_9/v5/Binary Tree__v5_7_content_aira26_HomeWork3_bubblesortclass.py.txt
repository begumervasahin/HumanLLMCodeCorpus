import random
import time
import matplotlib.pyplot as plt

class BubbleSort:
    def __init__(self):
        self.content = []

    def bubble_sort(self):
        """Sort the list using bubble sort algorithm."""
        for passnum in range(len(self.content)-1, 0, -1):
            for i in range(passnum):
                if self.content[i] > self.content[i+1]:
                    # Swap elements if they are in the wrong order
                    self.content[i], self.content[i + 1] = self.content[i + 1], self.content[i]

    def add(self, value):
        """Add value to the list and perform bubble sort."""
        self.content.append(value)
        self.bubble_sort()

    def get_min(self):
        """Return the minimum value in the list."""
        return self.content[0]

    def get_max(self):
        """Return the maximum value in the list."""
        return self.content[-1]

def measure_time(a, this_list):
    """Measure the time taken for adding, getting min, and getting max operations."""
    tot_time_add, tot_time_min, tot_time_max = 0, 0, 0
    for num in this_list:
        start = time.time()
        a.add(num)
        tot_time_add += (time.time() - start)

        start = time.time()
        a.get_min()
        tot_time_min += (time.time() - start)

        start = time.time()
        a.get_max()
        tot_time_max += (time.time() - start)

    return tot_time_add, tot_time_min, tot_time_max

    # Performance testing
    repetitions = 3
    max_operations = 500
    step = 100
    values_bubble, values_bubble_min, values_bubble_max = [], [], []

    for rounds in range(step, max_operations, step):
        this_list = [random.randint(0, 500) for _ in range(rounds)]
        tot_time_add, tot_time_min, tot_time_max = 0, 0, 0

        for repetition in range(repetitions):
            a = BubbleSort()
            my_add, my_min, my_max = measure_time(a, this_list)
            tot_time_add += my_add
            tot_time_min += my_min
            tot_time_max += my_max

        tot_time_add /= repetitions
        tot_time_min /= repetitions
        tot_time_max /= repetitions

        values_bubble.append(tot_time_add * 1000)  # Convert to milliseconds
        values_bubble_min.append(tot_time_min * 1000)
        values_bubble_max.append(tot_time_max * 1000)

    xlabels = range(step, max_operations, step)
    plt.plot(xlabels, values_bubble, label='Add')
    plt.plot(xlabels, values_bubble_min, label='Get Min')
    plt.plot(xlabels, values_bubble_max, label='Get Max')
    plt.legend()
    plt.xlabel("Number of Operations")
    plt.ylabel("Execution time (msec)")
    plt.title("Performance of Bubble Sort Solution")
    plt.show()
