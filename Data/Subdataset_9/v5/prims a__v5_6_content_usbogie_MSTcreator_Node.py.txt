class Node:
    def __init__(self, x_location, y_location, k_value, adjacent_nodes):
        # Initialize the attributes of the Node class
        self.x_location = x_location  # x-coordinate location
        self.y_location = y_location  # y-coordinate location
        self.k_value = k_value  # k-value
        self.adjacent_nodes = adjacent_nodes  # List of adjacent nodes

    def __eq__(self, other_node):
        # Define equality between nodes based on their x, y, and k values
        return (
            self.x_location == other_node.x_location 
            and self.y_location == other_node.y_location 
            and self.k_value == other_node.k_value
        )
