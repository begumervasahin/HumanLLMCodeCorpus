import network_1
import link_1
import threading
from time import sleep

router_queue_size = 0
simulation_time = 2

if __name__ == '__main__':
    object_L = []
    
    # Creating hosts
    host_1 = network_1.Host('H1')
    object_L.append(host_1)
    
    host_2 = network_1.Host('H2')
    object_L.append(host_2)
    
    # Configuring routers
    cost_D_RA = {'H1': {0: 1}, 'RB': {1: 1}}
    router_a = network_1.Router(name='RA',
                                 cost_D=cost_D_RA,
                                 max_queue_size=router_queue_size)
    object_L.append(router_a)
    
    cost_D_RB = {'H2': {1: 3}, 'RA': {0: 1}}
    router_b = network_1.Router(name='RB',
                                 cost_D=cost_D_RB,
                                 max_queue_size=router_queue_size)
    object_L.append(router_b)
    
    # Creating link layer
    link_layer = link_1.LinkLayer()
    object_L.append(link_layer)
    
    # Adding links
    link_layer.add_link(link_1.Link(host_1, 0, router_a, 0))
    link_layer.add_link(link_1.Link(router_a, 1, router_b, 0))
    link_layer.add_link(link_1.Link(router_b, 1, host_2, 0))
    
    # Creating threads
    thread_L = []
    for obj in object_L:
        thread_L.append(threading.Thread(name=obj.__str__(), target=obj.run))
    
    # Starting threads
    for t in thread_L:
        t.start()
    
    # Sending routes from router A
    router_a.send_routes(1)
    sleep(simulation_time)
    
    # Printing converged routing tables
    print("Converged routing tables")
    for obj in object_L:
        if isinstance(obj, network_1.Router):
            obj.print_routes()
    
    # Sending message from host 1 to host 2
    host_1.udt_send('H2', 'MESSAGE_FROM_H1')
    sleep(simulation_time)
    
    # Stopping all threads
    for o in object_L:
        o.stop = True
    for t in thread_L:
        t.join()
    
    print("All simulation threads joined")
