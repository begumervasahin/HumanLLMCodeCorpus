# Function to solve the second problem
def solve_second_problem():
    a = []
    b = []
    n = int(input())
    for i in range(n):
        x, y = map(int, input().split())
        a.append(x)
        b.append(y)
    answer = -1
    min_value, max_value = min(a), max(b)
    for i in range(n):
        if a[i] == min_value and b[i] == max_value:
            answer = i + 1
            break
    print(answer)

# Function to solve the fourth problem
def solve_fourth_problem():
    n, m = map(int, input().split())
    a = list(map(int, input().split()))
    count = [0] * (m + 1)
    for genre in a:
        count[genre] += 1
    ways = 0
    for i in range(1, m):
        for j in range(i + 1, m + 1):
            ways += count[i] * count[j]
    print(ways)
