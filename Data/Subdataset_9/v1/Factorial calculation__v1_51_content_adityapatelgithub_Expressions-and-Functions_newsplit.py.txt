def new_split_iter(expr):
    """
    Splits a given expression into tokens without needing spaces between them.
    The function yields tokens one by one, functioning as an iterator.
    Operators, identifiers, numbers, and specific character combinations are treated as separate tokens.
    """
    # Define the list of single character operators.
    operators = ['+', '-', '*', '/', '%', '(', ')', '=', '<', '>', '!', '?', ':', ';', ',']
    
    # Append a semicolon to the expression to mark the end.
    expr += ";"
    
    pos = 0  # Start position for parsing.
    
    while expr[pos] != ";":
        if expr[pos] == ' ':
            # Skip spaces.
            pos += 1
        elif expr[pos].isalpha():
            # Handle alphabetic identifiers.
            start = pos
            while expr[pos] not in operators and expr[pos] != ' ':
                pos += 1
            yield expr[start:pos]
        elif expr[pos].isdigit():
            # Handle numeric literals.
            start = pos
            while expr[pos].isdigit():
                pos += 1
            yield expr[start:pos]
        elif expr[pos] in ['=', '!', '<', '>'] and expr[pos + 1] == '=':
            # Handle two-character operators ('==', '!=', '<=', '>=').
            yield expr[pos:pos + 2]
            pos += 2
        else:
            # Yield single character tokens, including operators.
            yield expr[pos]
            pos += 1
    
    # Yield the final semicolon marking the end of the expression.
    yield ";"

# Example usage
expression = "a = 5;"
for token in new_split_iter(expression):
    print(token)
