def game_state(tab):
    win = [(0, 1, 2), (3, 4, 5), (6, 7, 8), (0, 3, 6), (1, 4, 7), (2, 5, 8), (0, 4, 8), (2, 4, 6)]
    for i in win:
        if tab[i[0]] == 1 and tab[i[1]] == 1 and tab[i[2]] == 1:
            return 1
        if tab[i[0]] == -1 and tab[i[1]] == -1 and tab[i[2]] == -1:
            return -1
    if 0 in tab:
        return -2
    return 0

def search(tab, player, t):
    win = [(0, 1, 2), (3, 4, 5), (6, 7, 8), (0, 3, 6), (1, 4, 7), (2, 5, 8), (0, 4, 8), (2, 4, 6)]
    state = game_state(tab)
    if state != -2:
        return state * player, -1
    list_best = [-2]
    for cell in range(len(tab)):
        if tab[cell] == 0:
            tab[cell] = player
            val = -search(tab, -player, t + 1)[0]
            if val > list_best[0]:
                list_best = [val, cell]
            elif val == list_best[0]:
                list_best.append(cell)
            tab[cell] = 0
    return list_best

def print_board(tab):
    for i in range(0, 9, 3):
        print(tab[i:i+3])

tab = [0, 0, 0, 0, 0, 0, 0, 0, 0]
player = 1
res = search(tab, player, 0)
winner = res[0] * player
if player == 1:
    print("Player 1 starts.")

    print("It's a draw!")
else:
    winner_number = int(winner * -0.5 + 1.5)
