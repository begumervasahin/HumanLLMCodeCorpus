import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from sklearn.preprocessing import StandardScaler
from sklearn.svm import SVR

# Load the dataset
dataset = pd.read_csv('Position_Salaries.csv')

# Extracting features and target variable
X = dataset.iloc[:, 1:2].values
Y = dataset.iloc[:, 2].values

# Feature scaling
sc_X = StandardScaler()
sc_Y = StandardScaler()
X = sc_X.fit_transform(X)
Y = sc_Y.fit_transform(Y.reshape(-1,1))

# SVR model fitting
regressor = SVR(kernel='rbf')
regressor.fit(X, Y)

# Predicting a new result
scaled_prediction = regressor.predict(sc_X.transform(np.array([[6.5]])))
prediction = sc_Y.inverse_transform(scaled_prediction)

# Visualizing the SVR results
plt.scatter(X, Y, color='red')
plt.plot(X, regressor.predict(X), color='blue')
plt.title('Truth or Bluff (SVR)')
plt.xlabel('Position level')
plt.ylabel('Salary')
plt.show()

# Visualizing the SVR results with higher resolution and smoother curve
X_grid = np.arange(min(X), max(X), 0.01)  # smoother curve
X_grid = X_grid.reshape((len(X_grid), 1))
plt.scatter(X, Y, color='red')
plt.plot(X_grid, regressor.predict(X_grid), color='blue')
plt.title('Truth or Bluff (SVR with smoother curve)')
plt.xlabel('Position level')
plt.ylabel('Salary')
plt.show()

print("Predicted salary for position level 6.5:", prediction)
