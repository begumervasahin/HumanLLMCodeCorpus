class MaxHeap(object):
    """A class for representing a max heap."""
    def __init__(self):
        self.H = [-1]
        self.D = [-1]
        self.n = 0

    def maximum(self):
        """Return the maximum value in the heap."""
        return self.H[1]

    def insert(self, name, value):
        """Insert a new element into the heap."""
        self.n += 1
        self.H.append(name)
        self.D.append(value)
        self.heapify(self.n)

    def update(self, name, value):
        """Update the value of an existing element in the heap."""
        for i in range(1, self.n + 1):
            if self.H[i] == name:
                self.D[i] = value
                self.heapify(i)
                break

    def delete(self, h):
        """Delete an element from the heap."""
        if self.n == 1 or h == self.n:
            self.D.pop()
            self.H.pop()
            self.n -= 1
        else:
            self.D[h] = self.D.pop()
            self.H[h] = self.H.pop()
            self.n -= 1
            self.heapify(h)

    def heapify(self, k):
        """Heapify the heap."""
        if k > 1 and self.D[k] > self.D[k // 2]:
            h = k
            while h > 1 and self.D[h] > self.D[h // 2]:
                self.swap(h, h // 2)
                h //= 2
        else:
            while k <= self.n // 2 and ((2 * k + 1 <= self.n and self.D[k] < max(self.D[2 * k], self.D[2 * k + 1])) or (2 * k + 1 > self.n and self.D[k] < self.D[2 * k])):
                d = 2 * k
                if 2 * k + 1 <= self.n and self.D[2 * k] < self.D[2 * k + 1]:
                    d = 2 * k + 1
                self.swap(k, d)
                k = d

    def swap(self, h, d):
        """Swap two elements in the heap."""
        self.D[h], self.D[d] = self.D[d], self.D[h]
        self.H[h], self.H[d] = self.H[d], self.H[h]

    def heap_sort(self):
        """Perform heap sort on the heap."""
        sorted_edges = []
        sorted_weights = []
        while self.n >= 1:
            sorted_edges.append(self.H[1])
            sorted_weights.append(self.D[1])
            self.delete(1)
        return sorted_weights, sorted_edges
