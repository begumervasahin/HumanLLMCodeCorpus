import os
import csv
import collections
import pandas as pd
from matplotlib import pyplot as plt
from matplotlib.dates import datestr2num
from scipy.cluster.hierarchy import dendrogram, linkage

class Data:
    """
    Class to process data from CSV files in a given directory and perform hierarchical clustering.
    """

    def __init__(self, pca_csv_path):
        self.pca_csv_path = pca_csv_path
        self.directories = []
        self.names = []
        self.data = []
        self.text = None
        self.countries = []
        self.imports = {}
        self.timeseries = None
        self.exports = {}
        self.time = []
        self.importlist = []
        self.exportlist = []
        self.sortcountrylist = []
        self.values = []
        self.Z = None

    def get_data(self):
        """
        Get the PCA CSV path.
        """
        return self.pca_csv_path

    def get_path(self):
        """
        Change directory to PCA CSV path.
        """
        os.chdir(self.pca_csv_path)

    def read_csv(self):
        """
        Read CSV files in the directory and append data.
        """
        self.names = os.listdir(self.pca_csv_path)
        for item in self.names:
            self.directories.append(item)
        for item in self.directories:
            with open(item, "r+") as f:
                self.text = csv.reader(f)
                for row in self.text:
                    self.data.append(row)
                f.close()
        return self.data

    def get_countries(self):
        """
        Get the list of countries.
        """
        for item in self.directories:
            self.countries.append(item)
        return self.countries

    def imports(self):
        """
        Extract imports data from CSV files.
        """
        for i in self.directories:
            with open(i, "r+") as f:
                dat = csv.reader(f)
                for row in dat:
                    try:
                        self.imports[i].append(float(row[1]))
                    except:
                        print("Error")
        self.timeseries = pd.DataFrame.from_dict(self.imports)
        print("Timeseries:", str(self.timeseries))

    def exports(self):
        """
        Extract exports data from CSV files.
        """
        for i in self.directories:
            with open(i, "r+") as f:
                dat = csv.reader(f)
                for row in dat:
                    try:
                        self.exports[i].append(row[0])
                    except:
                        print("Error")
        for key in self.exports:
            self.time.append(key)

    def axes(self):
        """
        Prepare axes for plotting.
        """
        SortImports = collections.OrderedDict(sorted(self.exports.items(), key=lambda t: t[1]))
        SortExports = collections.OrderedDict(sorted(self.imports.items(), key=lambda t: t[1]))
        for key, value in SortImports.items():
            self.importlist.append(value)
            self.sortcountrylist.append(key)
        for key,value in SortExports.items():
            self.exportlist.append(value)
        self.importlist = datestr2num(self.importlist)
        self.values = zip(self.importlist, self.exportlist)
        return self.values

    def cluster(self):
        """
        Perform hierarchical clustering.
        """
        correlation_matrix = self.timeseries.corr(method='pearson')
        self.Z = linkage(self.timeseries, 'average')

    def plot_dendrogram(self):
        """
        Plot the hierarchical clustering dendrogram.
        """
        plt.figure()
        plt.title('Hierarchical Clustering Dendrogram')
        plt.xlabel('Sample Index')
        plt.ylabel('Distance')
        print(len(self.sortcountrylist))
        dendrogram(self.Z, leaf_rotation=90., leaf_font_size=8., labels=self.sortcountrylist)
        plt.show()
