def sbox_substitution(input_string):
    """Performs S-box substitution on the input string."""
    # Convert each character in the input string to its hexadecimal representation
    hex_values = [hex(ord(char)) for char in input_string]
    
    # Extract the significant hexadecimal digits (after '0x') and perform S-box substitution
    substituted_values = [hex(Sbox[int(hex_val[2:], 16)]) for hex_val in hex_values]
    
    # Convert the substituted hexadecimal values back to characters
    substituted_string = ''.join([chr(int(hex_val, 16)) for hex_val in substituted_values])
    
    return substituted_string
