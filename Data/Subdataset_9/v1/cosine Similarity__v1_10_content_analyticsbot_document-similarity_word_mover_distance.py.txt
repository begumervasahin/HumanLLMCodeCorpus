from pyemd import emd
from sklearn.feature_extraction.text import CountVectorizer
import numpy as np
from sklearn.metrics import euclidean_distances
import os
from gensim.models import KeyedVectors
from text_unidecode import unidecode

# Sample documents
d1 = "Obama speaks to the media in Illinois"
d2 = "The President addresses the press in Chicago"

# Check if word embeddings file exists, if not, create it
if not os.path.exists("data/embed.dat"):
    print("Caching word embeddings in memmapped format...")
    # Load pre-trained Word2Vec model
    wv = KeyedVectors.load_word2vec_format(
        "data/GoogleNews-vectors-negative300.bin.gz",
        binary=True)
    wv.init_sims(replace=True)
    # Create a memmap file for word embeddings
    fp = np.memmap("data/embed.dat", dtype=np.double, mode='w+', shape=wv.vectors_norm.shape)
    fp[:] = wv.vectors_norm[:]
    with open("data/embed.vocab", "w") as f:
        for word in sorted(wv.vocab.keys()):
            print(unidecode(word), file=f)
    del fp

# Load word embeddings and vocabulary
wv = KeyedVectors.load_word2vec_format(
        "data/GoogleNews-vectors-negative300.bin.gz",
        binary=True)
W = np.memmap("data/embed.dat", dtype=np.double, mode="r", shape=wv.vectors.shape)
with open("data/embed.vocab") as f:
    vocab_list = [line.strip() for line in f.readlines()]
vocab_dict = {w: k for k, w in enumerate(vocab_list)}

# Function to calculate similarity using Earth Mover's Distance (EMD)
def similarity(d1, d2):
    vect = CountVectorizer(stop_words="english").fit([d1, d2])
    W_ = W[[vocab_dict[w] for w in vect.get_feature_names()]]
    D_ = euclidean_distances(W_)
    v_1, v_2 = vect.transform([d1, d2])
    v_1 = v_1.toarray().ravel().astype(np.double)
    v_2 = v_2.toarray().ravel().astype(np.double)
    v_1 /= v_1.sum()
    v_2 /= v_2.sum()
    D_ = D_.astype(np.double)
    D_ /= D_.max()
    return emd(v_1, v_2, D_)

# Calculate similarity between d1 and d2
similarity_score = similarity(d1, d2)
print("Similarity score between d1 and d2:", similarity_score)
