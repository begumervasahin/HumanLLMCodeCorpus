class HashTable:
    def __init__(self, size=31):
        self.nbuckets = size
        self.the_table = []
        for i in range(self.nbuckets):
            self.the_table.append([])

    def hashcode(self, o):
        if isinstance(o, int):
            return o
        elif isinstance(o, str):
            h = 0
            for char in o:
                h = h * 31 + ord(char)
            return h
        else:
            return None

    def buckets_str(self):
        if self.the_table is None:
            return None
        output = ""
        for i in range(len(self.the_table)):
            output += str(i).zfill(4)
            output += '->'
            for node in self.the_table[i]:
                output += str(node[0]) + ':' + str(node[1])
                output += ', '
            output = output.rstrip(', ')
            output += '\n'
        return output

    def __str__(self):
        if self.the_table is None:
            return None
        output = '{'
        for i in range(len(self.the_table)):
            for node in self.the_table[i]:
                output += str(node[0]) + ':' + str(node[1])
                output += ', '
        output = output.rstrip(', ')
        output += '}'
        return output

    def put(self, key, value):
        if self.the_table is None or len(self.the_table) == 0:
            return
        bucket_index = self.hashcode(key) % len(self.the_table)
        bucket = self.the_table[bucket_index]
        new_item_flag = True
        for i in range(len(bucket)):
            if bucket[i][0] == key:
                new_value = bucket[i][1] | value
                bucket[i] = (key, new_value)
                new_item_flag = False
                break
        if new_item_flag:
            bucket.append((key, value))

    def get(self, key):
        if self.the_table is None or len(self.the_table) == 0:
            return None
        bucket_index = self.hashcode(key) % len(self.the_table)
        bucket = self.the_table[bucket_index]
        for node in bucket:
            if node[0] == key:
                return node[1]
        return None

    def bucket_indexof(self, key):
        if self.the_table is None or len(self.the_table) == 0:
            return None
        output = '{'
        for node in self.the_table[key]:
            output += str(node[0]) + ':' + str(node[1])
            output += ', '
        output = output.rstrip(', ')
        output += '}'
        return output

    def __getitem__(self, key):
        return self.get(key)

    def __setitem__(self, key, value):
        self.put(key, value)
