class Edge:
    def __init__(self, Vertex1, Vertex2, weight, extra):
        self.Vertex1 = Vertex1
        self.Vertex2 = Vertex2
        self.weight = weight
        self.extra = extra
        self.selected = False

    def __repr__(self):
        return "V1: {}  V2: {}  W: {}  SEL: {}".format(self.Vertex1, self.Vertex2, self.weight, self.selected)

# Example usage
if __name__ == "__main__":
    edge = Edge(1, 2, 10, 0)
    print(edge)
