import click
import aes

@click.command()
@click.option(
    '--encrypt/--decrypt',
    '-e/-d',
    default=True,
    help="Specify whether to encrypt or decrypt the input file."
)
@click.option(
    '--input_file',
    '-i',
    required=True,
    help="Input file to be encrypted or decrypted."
)
@click.option(
    '--output_file',
    '-o',
    required=True,
    help="Output file to store the encrypted or decrypted data."
)
@click.option(
    '--block-cipher-mode',
    '-m',
    type=click.Choice(["ECB", "CBC", "CTR"]),
    default="CTR",
    help="Specify the block cipher mode (ECB, CBC, CTR) for encryption."
)
@click.option(
    '--key-length',
    '-l',
    type=click.Choice(["128", "192", "256"]),
    default="128",
    help="Specify the length of the encryption key (128, 192, 256)."
)
def main(encrypt, input_file, output_file, block_cipher_mode, key_length):
    """
    Encrypt or decrypt a file using AES encryption algorithm.
    """
    if encrypt:
        # Generate a random key
        key = aes.random_key_generator(int(key_length))

        # Create AES object based on key length
        AES = aes.AES(key, int(key_length))

        # Create block cipher mode object
        if block_cipher_mode == "ECB":
            bcm = aes.ECB(AES)
        elif block_cipher_mode == "CBC":
            bcm = aes.CBC(AES, 16)
        elif block_cipher_mode == "CTR":
            bcm = aes.CTR(AES)

        # Encrypt the input file and write to output file
        bcm.cipher(input_file, output_file)
        print("Cipher Key:", key)
        write_key(key)
    else:
        # Read key from file
        key = read_key()

        # Check if key exists
        if key == 1:
            print("File key.txt doesn't exist! Can't decrypt without key")
            exit(1)

        # Determine key length
        key_length = len(key) * 4

        # Create AES object based on key length
        AES = aes.AES(key, key_length)

        # Create block cipher mode object
        if block_cipher_mode == "ECB":
            bcm = aes.ECB(AES)
        elif block_cipher_mode == "CBC":
            bcm = aes.CBC(AES, 16)
        elif block_cipher_mode == "CTR":
            bcm = aes.CTR(AES)

        # Decrypt the input file and write to output file
        bcm.decipher(input_file, output_file)

def read_key():
    """
    Read key from key.txt file.
    """
    try:
        with open("key.txt", "r") as f:
            key = f.read()
        return key
    except FileNotFoundError:
        return 1

def write_key(key):
    """
    Write key to key.txt file.
    """
    with open("key.txt", "w") as f:
        f.write(key)

if __name__ == "__main__":
    main()
