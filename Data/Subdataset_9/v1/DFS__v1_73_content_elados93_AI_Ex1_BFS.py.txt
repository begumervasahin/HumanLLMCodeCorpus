from TilePuzzleLogic import backtracking

class BFS_Search:
    """
    BFS_Search is used to search level by level in a given graph.
    The search method uses the logic provided in the constructor.
    """
    def __init__(self, logic):
        self._logic = logic

    def search(self, init_state):
        """
        Search the given state with the current logic in order to find a path for the goal.
        :param init_state: The state of the puzzle.
        :return: Tuple containing the path of operators, the number of states visited, and 0.
        """
        open_list = []
        count = 0
        open_list.append(init_state)
        while len(open_list):
            current = open_list.pop(0)
            count += 1
            if self._logic.is_goal_state(current):
                return backtracking(current), str(count), '0'
            children = self._logic.next_states(current)
            for child in children:
                open_list.append(child)
        raise Exception('Puzzle is not solvable!')
