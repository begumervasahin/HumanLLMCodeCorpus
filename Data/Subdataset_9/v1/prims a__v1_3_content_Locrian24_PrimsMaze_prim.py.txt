import argparse
from PIL import Image, ImageDraw
from mazes import Maze  # Assuming mazes.py is your maze generation module
import time
import random

def imageMaze(filename, size):
    t0 = time.time()
    print("Computing Maze and Saving...")

    # Ensure size is odd
    if size % 2 == 0:
        size += 1

    # Generate the maze
    mz = Maze(size)

    # Prepare data for image
    data = []
    for c in mz.cells:
        if c.IN or not c.Wall:
            data.append((255, 255, 255))  # White for passages
        else:
            data.append((0, 0, 0))  # Black for walls

    # Randomly select start and end points
    start = random.randrange(1, size, 2)
    end = random.randrange(1, size, 2)
    data[start] = (255, 255, 255)  # Start point
    data[len(data) - 1 - end] = (255, 255, 255)  # End point

    # Create image and save
    img = Image.new('RGB', (size, size), color='black')
    img.putdata(data)
    img.save(filename)

    t1 = time.time()
    total = t1 - t0
    print("Time elapsed:", total)

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("filename", help="Output filename for the maze image")
    parser.add_argument("size", type=int, help="Size of the maze (an odd integer)")
    args = parser.parse_args()

    size = args.size
    if size > 500:
        print("Please choose a size less than or equal to 500.")
        exit()

    imageMaze(args.filename, size)

if __name__ == "__main__":
    main()
