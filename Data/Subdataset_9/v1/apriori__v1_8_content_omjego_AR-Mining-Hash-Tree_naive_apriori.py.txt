import itertools
import csv

# Parameters module
class Parameters:
    SUPPORT = 30
    CONFIDENCE = 70

def load_data(filename, max_attr=100):
    """
    Loads data from the given CSV file.
    :param filename: Name of the CSV file
    :param max_attr: Maximum number of attributes to consider
    :return: List of transactions with no more than max_attr attributes
    """
    with open(filename, 'r') as f:
        data = list(csv.reader(f, delimiter=','))
    result = []
    for i in range(len(data)):
        attrbs = []
        for val in data[i][1:]:
            val = val.strip()
            if int(val) <= max_attr:
                attrbs.append(val)
        if len(attrbs) > 0:
            result.append(attrbs)
    return result

def solve(data, support, confidence):
    """
    Applies Apriori Algorithm to find association rules.
    :param data: List of transactions
    :param support: Minimum support threshold
    :param confidence: Minimum confidence threshold
    :return: Number of association rules generated
    """
    transactions = len(data)
    hashmap = {}
    for row in data:
        for word in row:
            if word not in hashmap:
                hashmap[word] = 1
            else:
                hashmap[word] += 1

    level = []
    for key in hashmap:
        if (100 * hashmap[key] / transactions) >= support:
            level.append([key])

    return find_association_rules(data, level, support, confidence)

def find_subsets(S, m):
    """
    Finds all subsets of size m from set S.
    :param S: Set of elements
    :param m: Size of subsets
    :return: Set of subsets
    """
    return set(itertools.combinations(S, m))

def has_infrequent_subset(dataset, prev_l, k):
    """
    Checks if a dataset has any infrequent subset.
    :param dataset: Dataset
    :param prev_l: Previous level of itemsets
    :param k: Size of subsets
    :return: True if an infrequent subset is found, False otherwise
    """
    subsets = find_subsets(dataset, k)
    for item in subsets:
        s = []
        for l in item:
            s.append(l)
        s.sort()
        if s not in prev_l:
            return True
    return False

def apriori(prev_l, k):
    """
    Generates candidate itemsets of size k using the Apriori algorithm.
    :param prev_l: Previous level of itemsets
    :param k: Size of itemsets to generate
    :return: List of candidate itemsets
    """
    length = k
    result = []
    for list1 in prev_l:
        for list2 in prev_l:
            count = 0
            c = []
            if list1 != list2:
                while count < length - 1:
                    if list1[count] != list2[count]:
                        break
                    else:
                        count += 1
                else:
                    if list1[length - 1] < list2[length - 1]:
                        for item in list1:
                            c.append(item)
                        c.append(list2[length - 1])
                        if not has_infrequent_subset(c, prev_l, k):
                            result.append(c)
    return result

def apply_support(data, singles, support):
    """
    Applies support threshold to generate frequent itemsets.
    :param data: List of transactions
    :param singles: List of single-item frequent sets
    :param support: Minimum support threshold
    :return: List of frequent itemsets
    """
    k = 2
    prev_l = []
    L = []
    for item in singles:
        prev_l.append(item)
    while prev_l:
        current = []
        sets = apriori(prev_l, k - 1)
        for c in sets:
            cnt = 0
            trans = len(data)
            s = set(c)
            for T in data:
                t = set(T)
                if s.issubset(t):
                    cnt += 1
            if (100 * cnt / trans) >= support:
                c.sort()
                current.append(c)
        prev_l = []
        for l in current:
            prev_l.append(l)
        k += 1
        if current:
            L.append(current)
    return L

def find_association_rules(data, singles, support, confidence):
    """
    Finds association rules with given support and confidence thresholds.
    :param data: List of transactions
    :param singles: List of single-item frequent sets
    :param support: Minimum support threshold
    :param confidence: Minimum confidence threshold
    :return: Number of association rules generated
    """
    num = 1
    L = apply_support(data, singles, support)
    result = 0
    for itemsets in L:
        for itemset in itemsets:
            length = len(itemset)
            count = 1
            while count < length:
                subsets = find_subsets(itemset, count)
                count += 1
                for item in subsets:
                    inc1 = 0
                    inc2 = 0
                    s = []
                    m = []
                    for i in item:
                        s.append(i)
                    for T in data:
                        if set(s).issubset(set(T)):
                            inc1 += 1
                        if set(itemset).issubset(set(T)):
                            inc2 += 1
                    if 100 * inc2 / inc1 >= confidence:
                        for index in itemset:
                            if index not in s:
                                m.append(index)
                        left = ','.join(s)
                        right = ','.join(set(itemset) - set(s))
                        print(' ==> '.join([left, right]))
                        result += 1
                        num += 1

    print('Total rules generated:', result)
    return result

# Main function
def main():
    """
    Main function.
    """
    # Load data
    data = load_data('1000-out1.csv')

    # Solve Association Rules Mining
    solve(data, Parameters.SUPPORT, Parameters.CONFIDENCE)

if __name__ == "__main__":
    main()
