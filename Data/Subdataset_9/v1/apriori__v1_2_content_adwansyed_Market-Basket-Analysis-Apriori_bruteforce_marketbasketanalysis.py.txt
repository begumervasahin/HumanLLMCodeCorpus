import time

def frequent_triples(dataset, tHold):
    """
    Find frequent triples with a minimum support threshold.
    
    Args:
        dataset (str): Path to the dataset file.
        tHold (float): Minimum support threshold.
        
    Returns:
        None
    """
    with open(dataset, 'r') as myFile:
        lines = myFile.readlines()
        triples_table = {}
        
        for line in lines:
            items = sorted(line.split())
            for i in range(len(items)):
                for j in range(i + 1, len(items)):
                    for k in range(j + 1, len(items)):
                        key = ','.join([items[i], items[j], items[k]])
                        triples_table[key] = triples_table.get(key, 0) + 1
        
        ordered = [(value, key) for key, value in triples_table.items() if value > len(lines) * tHold]
        ordered.sort(reverse=True)
        
        for idx, (count, items) in enumerate(ordered, 1):
            print(idx, ')', items, ':', count)

def frequent_doubles(dataset, tHold):
    """
    Find frequent doubles with a minimum support threshold.
    
    Args:
        dataset (str): Path to the dataset file.
        tHold (float): Minimum support threshold.
        
    Returns:
        None
    """
    with open(dataset, 'r') as myFile:
        lines = myFile.readlines()
        pairs_table = {}
        
        for line in lines:
            items = sorted(line.split())
            for i in range(len(items)):
                for j in range(i + 1, len(items)):
                    key = ','.join([items[i], items[j]])
                    pairs_table[key] = pairs_table.get(key, 0) + 1
        
        ordered = [(value, key) for key, value in pairs_table.items() if value > len(lines) * tHold]
        ordered.sort(reverse=True)
        
        for idx, (count, items) in enumerate(ordered, 1):
            print(idx, ')', items, ':', count)

if __name__ == "__main__":
    print('------------------------------------------------------------------------')
    print('Running...')
    print('Done!')

    threshold = 0.03  # minimum support threshold

    # Frequent Doubles
    print('------------------------------------------------------------------------')
    print('Running Frequent Doubles...')
    start = time.time()
    frequent_doubles('movies.dat', threshold)
    end = time.time()

    print('Done!')

    # Frequent Triples

    start = time.time()

    print('Done!')
