import RPi.GPIO as GPIO
from time import sleep

class Vibrators:
    def __init__(self):
        # Define GPIO pins for left and right vibrators
        self.leftPinGPIO = 12
        self.rightPinGPIO = 13
        
        # Set GPIO mode and setup pins as output
        GPIO.setmode(GPIO.BCM)
        GPIO.setup(self.leftPinGPIO, GPIO.OUT)
        GPIO.setup(self.rightPinGPIO, GPIO.OUT)
        
        # Initialize PWM objects for left and right vibrators
        self.leftPwm = GPIO.PWM(self.leftPinGPIO, 100)
        self.rightPwm = GPIO.PWM(self.rightPinGPIO, 100)
        
        # Start PWM with initial duty cycle of 0 (OFF)
        self.leftPwm.start(0)
        self.rightPwm.start(0)
        
        # Define PWM duty cycle levels for different profiles
        self.HIGH = 100
        self.MID = 50
        self.OFF = 0
        
        # Define profiles with corresponding left and right duty cycles
        self.profiles = {
            "NO CHANGE": [self.OFF, self.OFF],
            "LEFT": [self.MID, self.OFF],
            "MORE LEFT": [self.HIGH, self.OFF],
            "RIGHT": [self.OFF, self.MID],
            "MORE RIGHT": [self.OFF, self.HIGH]
        }
    
    # Method to set left vibrator magnitude
    def set_left(self, magnitude):
        self.leftPwm.ChangeDutyCycle(magnitude)
    
    # Method to set right vibrator magnitude
    def set_right(self, magnitude):
        self.rightPwm.ChangeDutyCycle(magnitude)
    
    # Method to set vibration profile
    def set_profile(self, profile):
        if profile in self.profiles:
            self.set_left(self.profiles[profile][0])
            self.set_right(self.profiles[profile][1])
        else:
            print("Unsupported Profile")

if __name__ == "__main__":
    # Create instance of Vibrators class
    v = Vibrators()
    
    # Set profile to LEFT for 3 seconds
    v.set_profile("LEFT")
    sleep(3)
    
    # Set profile back to NO CHANGE
    v.set_profile("NO CHANGE")

# Clean up GPIO resources
GPIO.cleanup()
