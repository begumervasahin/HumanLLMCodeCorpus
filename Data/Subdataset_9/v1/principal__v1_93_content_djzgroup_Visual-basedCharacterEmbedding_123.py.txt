# Function to display matrix as image
def show(matrix):
    a = np.reshape(matrix, (16, 16))
    plt.imshow(a)

# Function to convert image to matrix
def ImageToMatrix(filename):
    im = Image.open(filename)
    width, height = im.size
    im = im.convert("L")
    data = im.getdata()
    data = np.matrix(data, dtype='float') / 255.0
    new_data = np.reshape(data, (height, width))
    return new_data

# Load images and convert them to matrices
img = []

    img.append(ImageToMatrix('G:\\Yeah!!!\\img2vec/img/{}.jpg'.format(i + 1)))    
img = np.array(img)

# Perform PCA with 10 components
