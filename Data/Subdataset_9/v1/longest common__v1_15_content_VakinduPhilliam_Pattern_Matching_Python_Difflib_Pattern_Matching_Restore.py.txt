from difflib import ndiff, restore

# Compute differences between two sets of lines
diff = ndiff('one\ntwo\nthree\n'.splitlines(keepends=True),
             'ore\ntree\nemu\n'.splitlines(keepends=True))
diff = list(diff)

# Restore the original lines based on the differences
restored_1 = ''.join(restore(diff, 1))
restored_2 = ''.join(restore(diff, 2))

# Print the restored lines
print(restored_1, end="")
print(restored_2, end="")
