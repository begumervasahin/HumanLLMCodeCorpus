class Vertice:
    def __init__(self, id):
        self.id = id
        self.vertices_adjacentes = {}  # Dictionary to store adjacent vertices and their weights

    def inserir_vertice_adjacente(self, vertice_adjacente, peso=0):
        """Adds an adjacent vertex with the given weight."""
        self.vertices_adjacentes[vertice_adjacente] = peso

class Grafo:
    def __init__(self, direcionado=False):
        self.vertices = {}  # Dictionary to store vertices
        self.direcionado = direcionado

    def inserir_vertice(self, id):
        """Adds a vertex with the given ID to the graph."""
        v = Vertice(id)
        self.vertices[id] = v
        return v

    def inserir_aresta(self, de, para, peso=0):
        """Adds an edge from vertex 'de' to vertex 'para' with the given weight."""
        if de not in self.vertices:
            self.inserir_vertice(de)
        if para not in self.vertices:
            self.inserir_vertice(para)
        self.vertices[de].inserir_vertice_adjacente(self.vertices[para], peso)
        if not self.direcionado:
            self.vertices[para].inserir_vertice_adjacente(self.vertices[de], peso)

    def get_vertices(self):
        """Returns all vertices in the graph."""
        return list(self.vertices.values())

    def get_vertice(self, id):
        """Returns the vertex with the given ID."""
        return self.vertices.get(id)

    def get_arestas(self):
        """Returns all edges in the graph."""
        arestas = set()
        for vertice in self.vertices.values():
            for adjacente, peso in vertice.vertices_adjacentes.items():
                arestas.add((vertice, adjacente, peso))
        return arestas

    def __iter__(self):
        """Allows iteration over the vertices in the graph."""
        return iter(self.vertices.values())
