import json
import math
import re
from urllib.request import urlopen

def download_exchange_rates():
    """
    Download exchange rates from the given URL.
    """
    graph = {}
    url = "http://fx.priceonomics.com/v1/rates/?q=1"
    with urlopen(url) as response:
        jsrates = json.loads(response.read())

    pattern = re.compile("([A-Z]{3})_([A-Z]{3})")
    for key, value in jsrates.items():
        matches = pattern.match(key)
        if matches:
            from_rate = matches.group(1)
            to_rate = matches.group(2)
            if from_rate != to_rate:
                if from_rate not in graph:
                    graph[from_rate] = {}
                graph[from_rate][to_rate] = -math.log(float(value))
    return graph

def initialize_distances_and_paths(graph, source):
    """
    Initialize distances and predecessors.
    """
    distances = {}
    predecessors = {}
    for node in graph:
        distances[node] = float('inf')
        predecessors[node] = None
    distances[source] = 0
    return distances, predecessors

def relax_edge(node, neighbor, graph, distances, predecessors):
    """
    Relax the edge between node and neighbor.
    """
    if distances[neighbor] > distances[node] + graph[node][neighbor]:
        distances[neighbor] = distances[node] + graph[node][neighbor]
        predecessors[neighbor] = node

def retrace_negative_loop(predecessors, start):
    """
    Retrace the negative loop in the graph.
    """
    arbitrage_loop = [start]
    next_node = start
    while True:
        next_node = predecessors[next_node]
        if next_node not in arbitrage_loop:
            arbitrage_loop.append(next_node)
        else:
            arbitrage_loop.append(next_node)
            arbitrage_loop = arbitrage_loop[arbitrage_loop.index(next_node):]
            return arbitrage_loop

def bellman_ford(graph, source):
    """
    Bellman-Ford algorithm to find arbitrage opportunities.
    """
    distances, predecessors = initialize_distances_and_paths(graph, source)

    for _ in range(len(graph) - 1):
        for node in graph:
            for neighbor in graph[node]:
                relax_edge(node, neighbor, graph, distances, predecessors)

    for node in graph:
        for neighbor in graph[node]:
            if distances[neighbor] < distances[node] + graph[node][neighbor]:
                return retrace_negative_loop(predecessors, source)

    return None

def find_arbitrage_opportunities():
    """
    Find arbitrage opportunities in the exchange rates.
    """
    paths = []
    graph = download_exchange_rates()

    for key in graph:
        path = bellman_ford(graph, key)
        if path and path not in paths:
            paths.append(path)

    for path in paths:
        if not path:
            print("No arbitrage opportunity found.")
        else:
            money = 100
            print(f"Starting with {money} in {path[0]}")
            for i, value in enumerate(path):
                if i + 1 < len(path):
                    start = path[i]
                    end = path[i + 1]
                    rate = math.exp(-graph[start][end])
                    money *= rate
                    print(f"{start} to {end} at {rate} = {money}")
        print()

if __name__ == "__main__":
    find_arbitrage_opportunities()
