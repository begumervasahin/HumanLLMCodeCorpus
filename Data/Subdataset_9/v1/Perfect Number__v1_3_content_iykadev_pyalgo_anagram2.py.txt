def anagram(s1, s2):
    alist1 = list(s1)  # Convert s1 to a list
    alist2 = list(s2)  # Convert s2 to a list
    alist1.sort()      # Sort the list of characters in s1
    alist2.sort()      # Sort the list of characters in s2
    pos = 0
    matches = True
    while pos < len(s1) and matches:
        if alist1[pos] == alist2[pos]:
            pos += 1
        else:
            matches = False
    return matches

# Test the function with example inputs
print(anagram('ey edip', 'pide ye'))  # Output: True
