def fizzbuzz(a, b):
    if isinstance(a, list) and isinstance(b, list):
        len_sum = len(a) + len(b)
        if len_sum % 3 == 0 and len_sum % 5 == 0:
            return 'fizzbuzz'
        if len_sum % 3 == 0:
            return 'fizz'
        if len_sum % 5 == 0:
            return 'buzz'
        else:
            return len_sum
    else:
        return 'Invalid input'

if __name__ == "__main__":
    try:
        a = eval(input("Enter a list for a: "))
        b = eval(input("Enter a list for b: "))
        result = fizzbuzz(a, b)
        print(result)
    except Exception as e:
        print("Invalid input. Error:", e)
