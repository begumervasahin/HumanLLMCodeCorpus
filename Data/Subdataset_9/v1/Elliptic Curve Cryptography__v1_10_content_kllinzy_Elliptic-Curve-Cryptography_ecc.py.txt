# Function to encode a string using elliptic curve
def ECencode(s, curve):
    # Extract parameters from the curve
    a = curve.a
    b = curve.b
    p = curve.p

    # Convert string to integer representation
    s = str(s)
    x0 = 0
    for i in range(0, len(s)):
        x0 = x0 + ord(s[i]) * 256 ** i

    # Iterate to find a suitable x value
    for i in range(1000):
        x = 1000 * x0 + i % p
        t = x ^ 3 + a * x + b % p
        res = prime_mod_sqrt(t, p)
        if len(res) != 0:
            return Point.make_point(x, res[0], 1, p)

# Function to encode a string into points on the elliptic curve
def encode_string(s, curve, length):
    arr = []
    l = list(chunkstring(s, length))
    for i in range(0, len(l)):
        arr.append(ECencode(l[i], curve))
    return arr

# Function to encrypt a string using elliptic curve cryptography
def encrypt(s, key):
    curve = key.curve
    p = curve.p
    return encrypt_long(encode_string(s, curve, int(math.log(p // (1000 - 1), 256))), key)

# Function to decode a string from points on the elliptic curve
def decode_string(arr):
    v = []
    for i in range(0, len(arr)):
        v.append(ECdecode(arr[i]))
    return ''.join(v)

# Function to decrypt a string using elliptic curve cryptography
def decrypt(arr, key, k):
    return decode_string(decrypt_long(arr, key, k))

# Main function to generate a random prime number using Miller-Rabin test
def random_prime(low, high):
    result = 0
    while True:
        result = randint(low, high)
        if miller_rabin(result, 45):
            break
    return result
