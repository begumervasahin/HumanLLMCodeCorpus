import gensim
import glob
import os
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from rake_nltk import Rake

# Initialize Rake for keyword extraction
r = Rake()

# Get a list of file paths from a directory
file_list = glob.glob(os.path.join(os.getcwd(), "data/", "*.txt"))

# Read raw documents from files
raw_documents = []
for file_path in file_list:
    with open(file_path) as f_input:
        raw_documents.append(f_input.read())  

# Tokenize the raw documents
gen_docs = [[w.lower() for w in word_tokenize(text)] for text in raw_documents]

# Create a dictionary of unique words from the documents
dictionary = gensim.corpora.Dictionary(gen_docs)

# Convert documents into Bag of Words (BoW) representation
corpus = [dictionary.doc2bow(gen_doc) for gen_doc in gen_docs]

# Create TF-IDF model
tf_idf = gensim.models.TfidfModel(corpus)

# Calculate total number of documents
total_docs = sum(len(doc) for doc in corpus)

# Calculate similarity matrix
sims = gensim.similarities.Similarity('/var/www/html/Auto-tagging-of-Theses-and-Dissertations-of-Bicol-University-Searching-and-Matching-/result/result',
                                       tf_idf[corpus], num_features=len(dictionary))

# Tokenize query document
query_doc = [w.lower() for w in word_tokenize("I'm taking the show on the road.")]

# Convert query document into BoW representation
query_doc_bow = dictionary.doc2bow(query_doc)

# Calculate TF-IDF representation of the query document
query_doc_tf_idf = tf_idf[query_doc_bow]

# Calculate similarity between query document and corpus documents
similarities = sims[query_doc_tf_idf]

print(similarities)
