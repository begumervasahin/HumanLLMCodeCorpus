import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from sklearn.svm import SVR

# Load the dataset
dataset = pd.read_csv('Position_Salaries.csv')

# Extracting features and target variable
X = dataset.iloc[:, 1:2].values
Y = dataset.iloc[:, 2].values

# Feature Scaling (not used in this example)
"""
from sklearn.preprocessing import StandardScaler
sc_X = StandardScaler()
sc_Y = StandardScaler()
X = sc_X.fit_transform(X)
Y = sc_Y.fit_transform(Y.reshape(-1,1)).flatten()
"""

# SVR Model
regressor = SVR(kernel='rbf')
regressor.fit(X, Y)

# Predicting a new result
Y_pred = regressor.predict([[6.5]])
print("Predicted Salary for Position Level 6.5:", Y_pred)

# Visualizing the SVR results
plt.scatter(X, Y, color='red', label='Actual Data')
plt.plot(X, regressor.predict(X), color='blue', label='SVR Model')
plt.title('Truth or Bluff (SVR)')
plt.xlabel('Position Level')
plt.ylabel('Salary')
plt.legend()
plt.show()

# Visualizing with higher resolution and smoother curve
X_grid = np.arange(min(X), max(X), 0.01) # finer resolution
X_grid = X_grid.reshape((len(X_grid), 1))
plt.scatter(X, Y, color='red', label='Actual Data')
plt.plot(X_grid, regressor.predict(X_grid), color='blue', label='SVR Model')
plt.title('Truth or Bluff (SVR)')
plt.xlabel('Position Level')
plt.ylabel('Salary')
plt.legend()
plt.show()
