import numpy as np

def KNNClassifier(unLabel, dataSet, labels, k):
    numSamples = dataSet.shape[0]
    diff = np.tile(unLabel, (numSamples, 1)) - dataSet
    sqDiff = diff ** 2
    sqDist = np.sum(sqDiff, axis=1)
    distance = sqDist ** 0.5
    sortedDistIndex = np.argsort(distance)
    classCount = {}
    for i in range(k):
        voteILabel = labels[sortedDistIndex[i]]
        classCount[voteILabel] = classCount.get(voteILabel, 0) + 1
    maxCount = 0
    for key, value in classCount.items():
        if value > maxCount:
            maxCount = value
            maxIndex = key
    return maxIndex

def file2matrix(filename, dim2):
    data = np.loadtxt(filename)
    return data[:, :dim2], data[:, -1].astype(int)

def autoNorm(dataSet):
    minValue = dataSet.min(0)
    maxValue = dataSet.max(0)
    ranges = maxValue - minValue
    normDataSet = (dataSet - minValue) / ranges
    return normDataSet, ranges, minValue

def datingClassTest(filename, dim2, rate=0.1, k=3):
    datingDataMat, datingLabels = file2matrix(filename, dim2)
    normMat, _, _ = autoNorm(datingDataMat)
    testVectorCount = int(normMat.shape[0] * rate)
    errorCount = 0.0
    for i in range(testVectorCount):
        classifiedResult = KNNClassifier(normMat[i, :], normMat[testVectorCount:, :], datingLabels[testVectorCount:], k)
        print(f"Result from KNN Classifier is: {classifiedResult}, the actual class is: {datingLabels[i]}")
        if classifiedResult != datingLabels[i]:
            errorCount += 1
    print(f"The total error rate is: {errorCount / (testVectorCount * 1.0)}")

def classifyPeople(filename, dim2):
    results = ['not at all', 'in small doses', 'in large doses']
    percentTimeGame = float(input('Percentage of time spent playing video games? '))
    freFlyMilesEarn = float(input('Frequent flier miles earned per year? '))
    iceConsumed = float(input('Liters of ice cream consumed per year? '))
    datingDataMat, datingLabels = file2matrix(filename, dim2)
    normMat, ranges, minValue = autoNorm(datingDataMat)
    inputArray = np.array([percentTimeGame, freFlyMilesEarn, iceConsumed])
    classifiedResult = KNNClassifier((inputArray - minValue) / ranges, normMat, datingLabels, 3)
    print(f'You will probably like this person: {results[classifiedResult - 1]}')

if __name__ == '__main__':
    # Test the KNN classifier with the dating dataset
    datingClassTest('datingTestSet2.txt', dim2=3)

    # Classify people based on their input attributes
    classifyPeople('datingTestSet2.txt', dim2=3)
