class PrefixMatcher(object):
    def __init__(self, options):
        self._opts = options

    def matchunique(self, value):
        """
        Matches a unique option based on the provided value. Raises exceptions for ambiguous or unmatched values.

        Args:
            value (str): The value to match.

        Returns:
            str: The unique matched option.
        
        Raises:
            Exception: If the value does not match any options or is ambiguous.
        """
        result = self.match(value)
        if len(result) != 1:
            if len(result) == 0:
                raise Exception("'%s' did not match any options." % (value))
            else:
                raise Exception("'%s' is ambiguous. Please clarify further. Available: %s" % (value, ", ".join(sorted(list(result)))))
        return result[0]

    def match(self, value):
        """
        Matches options based on the provided value.

        Args:
            value (str): The value to match.

        Returns:
            list: A list of matched options.
        """
        return [option for option in self._opts if option.startswith(value)]
