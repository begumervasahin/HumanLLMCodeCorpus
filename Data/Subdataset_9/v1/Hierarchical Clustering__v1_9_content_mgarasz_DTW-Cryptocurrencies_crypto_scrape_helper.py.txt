def coin_lookup(coin):
    btc_url = 'https://poloniex.com/public?command=returnChartData&currencyPair=USDT_BTC&start=1439020800&end=1511202699&period=14400'
    cc = 'BTC_' + coin
    url = f'https://poloniex.com/public?command=returnChartData&currencyPair={cc}&start=1439020800&end=1511202699&period=14400'
    ico = pd.read_json(url)
    return ico
