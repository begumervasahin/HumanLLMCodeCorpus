import numpy as np
import math
from scipy.stats import norm

def get_min(tab):
    return np.min(tab)

def get_max(tab):
    return np.max(tab)

def f(x):
    return math.pow(x, 3) + (3 * math.pow(x, 2)) + (3 * x) + 1

def fa(tab, n):
    return np.mean([f(x) for x in tab])

def calculate_vf(tab, f_, n):
    return np.sum(np.power(np.array([f(x) for x in tab]) - f_, 2)) / (n - 1)

data_uniform = np.genfromtxt('RandUnif.csv', delimiter='')
btm_uniform = 2
top_uniform = 4
N_uniform = len(data_uniform)
prob = 0.95
k = norm.ppf(prob)
f_aksen_uniform = fa(data_uniform, N_uniform)
Vf_uniform = calculate_vf(data_uniform, f_aksen_uniform, N_uniform)
Vf_aksen_uniform = Vf_uniform / N_uniform
lower_uniform = (top_uniform - btm_uniform) * (f_aksen_uniform - (k * Vf_aksen_uniform))
upper_uniform = (top_uniform - btm_uniform) * (f_aksen_uniform + (k * Vf_aksen_uniform))
print(f'Hasil mean-variance (uniform) bag. A\n[{lower_uniform}, {upper_uniform}]')

data_normal = np.genfromtxt('RandDist.csv', delimiter='')
btm_normal = get_min(data_normal)
top_normal = get_max(data_normal)
N_normal = len(data_normal)
f_aksen_normal = fa(data_normal, N_normal)
Vf_normal = calculate_vf(data_normal, f_aksen_normal, N_normal)
Vf_aksen_normal = Vf_normal / N_normal
lower_normal = (top_normal - btm_normal) * (f_aksen_normal - (k * Vf_aksen_normal))
upper_normal = (top_normal - btm_normal) * (f_aksen_normal + (k * Vf_aksen_normal))
print(f'Hasil mean-variance (normal) bag. B\n[{lower_normal}, {upper_normal}]')
