import timeit
import random

def quickSort(a, l, r):
    if l < r:
        m = partition(a, l, r)
        quickSort(a, l, m - 1)
        quickSort(a, m + 1, r)

def partition(a, l, r):
    x = a[r]
    i = l - 1
    for j in range(l, r):
        if a[j] <= x:
            i += 1
            a[i], a[j] = a[j], a[i]
    a[i + 1], a[r] = a[r], a[i + 1]
    return i + 1

def randomQuickSort(a, l, r):
    if l < r:
        m = randomPartition(a, l, r)
        randomQuickSort(a, l, m - 1)
        randomQuickSort(a, m + 1, r)

def randomPartition(a, l, r):
    i = random.randint(l, r)
    a[r], a[i] = a[i], a[r]
    return partition(a, l, r)

def medianRandomQuickSort(a, l, r):
    if l < r:
        m = medianRandomPartition(a, l, r)
        medianRandomQuickSort(a, l, m - 1)
        medianRandomQuickSort(a, m + 1, r)

def medianRandomPartition(a, l, r):
    medianIndex = getMedianIndex(a, l, r)
    a[r], a[medianIndex] = a[medianIndex], a[r]
    return partition(a, l, r)

def getMedianIndex(a, l, r):
    mid = (l + r) // 2
    if a[l] <= a[mid] <= a[r] or a[r] <= a[mid] <= a[l]:
        return mid
    elif a[mid] <= a[l] <= a[r] or a[r] <= a[l] <= a[mid]:
        return l
    else:
        return r

if __name__ == "__main__":
    print("Quick Sort")
    for size in [1000, 10000, 100000]:
        s = [random.random() for _ in range(size)]
        print(f"{size} elements:")
        print(min(timeit.Timer('a=s[:]; quickSort(a, 0, len(a) - 1)', globals=globals()).repeat(10, 100)))

    print("Random Quick Sort")
    for size in [1000, 10000, 100000]:
        s = [random.random() for _ in range(size)]
        print(f"{size} elements:")
        print(min(timeit.Timer('a=s[:]; randomQuickSort(a, 0, len(a) - 1)', globals=globals()).repeat(10, 100)))

    print("Median Random Quick Sort")
    for size in [1000, 10000, 100000]:
        s = [random.random() for _ in range(size)]
        print(f"{size} elements:")
        print(min(timeit.Timer('a=s[:]; medianRandomQuickSort(a, 0, len(a) - 1)', globals=globals()).repeat(10, 100)))
