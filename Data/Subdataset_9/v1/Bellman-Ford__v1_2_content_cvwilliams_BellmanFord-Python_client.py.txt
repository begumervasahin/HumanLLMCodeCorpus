import socket
import pickle
import sys

MAXINT = sys.maxsize
table0 = [[0, -1, 0], [1, 0, 1], [2, 1, 3], [3, 2, 7]]

def printTable(table):
    print("DESTINATION-------INTERFACE------LINKCOST")
    for i in table:
        if i[1] == -1:
            print("    ", i[0], "            ", i[1], "           ", i[2])
        elif i[1] == MAXINT:
            print("    ", i[0], "            inf            inf")
        else:
            print("    ", i[0], "            ", i[1], "            ", i[2])

def main():
    print("Initial Table:")
    printTable(table0)

    s = socket.socket()
    host = "afsconnect2.njit.edu"
    port = 57171

    try:
        s.connect((host, port))
        print("Connected to server")

        s.send(pickle.dumps(table0))
        print("Sent initial table to server")

        tab_update = s.recv(1024)
        tab_update = pickle.loads(tab_update)

        print("Updated Table:")
        printTable(tab_update)
    except socket.error as e:
        print("Error:", e)
    finally:
        s.close()
        print('Client: Connection closed')

if __name__ == '__main__':
    main()
