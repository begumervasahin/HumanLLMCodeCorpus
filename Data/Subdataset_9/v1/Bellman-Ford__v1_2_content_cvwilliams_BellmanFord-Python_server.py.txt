import pickle
import socket
import sys

MAXINT = sys.maxsize

def printTable(table):
    """
    Print the table in a formatted manner.
    """
    print("DESTINATION-------INTERFACE------LINKCOST")
    for i in table:
        if i[1] == -1:
            print("    ",i[0],"            ",i[1],"           ",i[2])
        elif i[1] == MAXINT:
            print("    ",i[0],"            inf            inf") 
        else:
            print("    ",i[0],"            ",i[1],"            ",i[2])

def main():
    table1 = [[0, 0, 1], [1, -1, 0], [2, 1, 1], [3, MAXINT, MAXINT]]
    print("Initial table: ")
    printTable(table1)

    s = socket.socket()
    port = 57171
    host = ''
    
    try:
        s.bind((host, port))
    except socket.error as msg:
        print('Bind failed. Error Code : ' + str(msg[0]) + ' Message ' + msg[1])
        sys.exit()

    print('Socket bind complete')
    s.listen(1)
    client, addr = s.accept()
    print('Got connection from', addr)

    table0 = client.recv(1024)
    table0 = pickle.loads(table0)
    print("Current table: ")
    printTable(table1)
    print("Received table: ")
    printTable(table0)

    num = 0
    len1 = len(table1)
    len2 = len(table0)
    
    for i in range(len1):
        for j in range(len2):
            num = table1[i][2] + int(table0[j][2])
            if num < table0[i][2]:
                table0[i][2] = num + 1

    print("Updated table: ")
    printTable(table0)
    client.send(pickle.dumps(table0))
    s.close()

if __name__ == "__main__":
    main()
