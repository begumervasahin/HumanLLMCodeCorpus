import sequences

def test1():
    """
    Test if the return value of the Fibonacci function matches the expected result.
    """
    expected_list = [1, 1, 2, 3, 5]
    ret_list = sequences.fibonacci(5)

    assert isinstance(ret_list, list)
    assert len(ret_list) == len(expected_list)
    assert expected_list == ret_list
    print("Test 1 success")

def test2():
    """
    Output the Fibonacci list for a certain valid number and verify the last number of the list.
    """
    ret_list = sequences.fibonacci(8)
    print(ret_list[7])
    assert ret_list[7] == 21
    print("Test 2 success")

def test3():
    """
    Test if the Fibonacci function can validate the input.
    """
    invalid_values = [-1, 2.2, 0]
    for value in invalid_values:
        assert not sequences.fibonacci(value)
    print("Input validation success")

if __name__ == "__main__":
    print("Test 1 start")
    test1()
    print("Test 2 start")
    test2()
    print("Test 3 start")
    test3()
