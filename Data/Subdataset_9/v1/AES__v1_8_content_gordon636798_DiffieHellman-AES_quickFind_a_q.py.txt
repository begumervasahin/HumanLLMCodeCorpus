from gmpy2 import *
import gmpy2

# Generate a random number of 256 bits
rand = mpz_rrandomb(random_state(), 256)
print("Random number:", rand)

# Find the next prime number after the random number
q = next_prime(rand)
print("Prime number:", q)

# Function to find a value satisfying certain conditions
def find_a():
    for i in range(2, q):
        for j in range(2, f_div(q - 1, 2) + 1):
            if f_mod(q - 1, j) == 0 and is_prime(j) and powmod(i, f_div(q - 1, j), q) == 1:
                break
            elif j == f_div(q - 1, 2):
                return i

# Find and print the value of 'a'
a = find_a()
print("Value of 'a':", a)
