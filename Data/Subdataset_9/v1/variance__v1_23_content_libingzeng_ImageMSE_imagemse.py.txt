import cv2

def clmap(v, k, upBound):
    val = v * k
    if val > upBound:
        return upBound
    else:
        return val

inImage_1 = './tests/living-room_pfmlt_m310_6629s.png'
inImage_2 = './tests/reference/living-room-reference.png'
colourbar = './colourbar.jpg'
dif_img = 'dif_' + inImage_1

img_1 = cv2.imread(inImage_1)
img_2 = cv2.imread(inImage_2)
img_colourbar = cv2.imread(colourbar)

dif = img_1.copy()
show_dif = dif.copy()

width = img_1.shape[0]
height = img_1.shape[1]
cwidth = img_colourbar.shape[0]
cheight = img_colourbar.shape[1]

sum_squared_diff = 0

for i in range(width):
    for j in range(height):
        # Convert to Y channel (YUV color space)
        y1 = img_1[i, j][2] * 0.212671 + img_1[i, j][1] * 0.715160 + img_1[i, j][0] * 0.072169
        y2 = img_2[i, j][2] * 0.212671 + img_2[i, j][1] * 0.715160 + img_2[i, j][0] * 0.072169
        
        # Compute pixel-wise difference
        diff = y1 - y2
        sum_squared_diff += abs(diff) ** 2
        
        # Clamp and map the difference value
        diff0 = clmap(abs(diff), 5, 255)
        pos = int((diff0 / 255.0) * (cheight - 10))
        
        # Use colour bar for visualization
        show_dif[i, j] = img_colourbar[0, pos]
        dif[i, j] = abs(diff0)

mse = sum_squared_diff / (width * height)
print('MSE:', mse)
print(dif_img)
print('different data:')
print('max:', dif.max())
print('min:', dif.min())
print('mean:', dif.mean())

# Save the visualization
cv2.imwrite(dif_img, show_dif)

# Display the difference image
cv2.imshow('_dif', show_dif)
cv2.waitKey(0)
cv2.destroyAllWindows()
