import argparse

class BellmanFord:
    def __init__(self, start_node, graph):
        self.graph = graph
        self.distances = {}
        self.distances[start_node] = 0
        self.start_node = start_node
        self.nodes = list(graph.keys())
        self.nodes.remove(start_node)
        self.nodes.insert(0, start_node)
        self.relax()
        self.detect_negative_cycle()

    def relax(self):
        """
        Relaxation step: Update distances using Bellman-Ford algorithm.
        """
        for _ in range(len(self.graph) - 1):
            for node in self.nodes:
                for neighbor, weight in self.graph[node].items():
                    if self.distances[node] == float('inf'):
                        break
                    elif self.distances[neighbor] == float('inf'):
                        self.distances[neighbor] = weight + self.distances[node]
                    elif weight + self.distances[node] < self.distances[neighbor]:
                        self.distances[neighbor] = weight + self.distances[node]

        print(self.distances)

    def detect_negative_cycle(self):
        """
        Check for negative-weight cycles in the graph.
        """
        for node in self.nodes:
            for neighbor, weight in self.graph[node].items():
                if weight + self.distances[node] < self.distances[neighbor]:
                    print('The Graph has a negative-weight cycle')
