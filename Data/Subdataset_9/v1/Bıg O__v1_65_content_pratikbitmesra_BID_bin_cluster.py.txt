import csv
import itertools
import sys

def read_extracted_csv(input_file):
    """
    Reads the contents of an extracted CSV file, processes the data, and writes the results to a new CSV file.

    Args:
        input_file (str): Path to the input CSV file.

    Returns:
        None
    """
    size_of_disk = 250
    total_no_bins = (size_of_disk * 1024 / 128)
    lba_list = []
    xfrlen_list = []
    timestamp_list = []
    operation_list = []
    bin_list = []
    clustered_bins = []
    flag = 0

    with open(input_file, 'r') as extracted_csv:
        for line in extracted_csv:
            l = line.strip()
            store_line = l.split(',')
            try:
                if flag == 1:
                    bin_list.append(int(store_line[0]))
                    lba_list.append(int(store_line[1]))
                    xfrlen_list.append(int(store_line[2]))
                    operation_list.append(store_line[3])
                    timestamp_list.append(float(store_line[4]))
                flag = 1
            except IndexError:
                continue

    clustered_bins = [list(v) for k, v in itertools.groupby(bin_list)]

    bin_number = []
    bin_count = []
    end_time = []
    xfrlen_bin_sum = []
    operation_bin = []
    counter = 0
    flag_counter = 0

    for y in clustered_bins:
        bin_number.append(y[-1])
        bin_count.append(len(y))
        counter += len(y)
        end_time.append(timestamp_list[counter - 1])
        operation_bin.append(operation_list[counter - 1])
        xfrlen_bin_sum.append(sum(xfrlen_list[flag_counter:counter]) / (2 * 1024.0))
        flag_counter += len(y)

    single_bin = zip(bin_number, bin_count, xfrlen_bin_sum, operation_bin, end_time)

    with open(input_file + '_bin_info.csv', 'w+') as bin_info_file:
        bin_info_file.write("bin_number,bin_count,xfrlen_bin_sum(MB),operation_bin,end_time\n")
        writer = csv.writer(bin_info_file, delimiter=',')
        writer.writerows(single_bin)

def main():
    if len(sys.argv) != 2:
        print("Usage: python script.py <input_file>")
        sys.exit(1)

    input_file = sys.argv[1]
    read_extracted_csv(input_file)

if __name__ == "__main__":
    main()
