import glob

# Match files/directories starting with a single digit and any extension
print(glob.glob('./[0-9].*'))

# Match files with a .gif extension in the current directory
print(glob.glob('*.gif'))

# Match files with a single character filename and .gif extension
print(glob.glob('?.gif'))

# Match .txt files recursively in all subdirectories
print(glob.glob('**/*.txt', recursive=True))

# Match all directories and subdirectories
print(glob.glob('./**/', recursive=True))
