class Node:
    def __init__(self, val):
        self.value = val
        self.parent = self
        self.rank = 0
    
    def __repr__(self):
        return "n " + str(self.value) + " r " + str(self.rank) + " p " + str(self.parent.value)

# Example usage
node1 = Node('A')
node2 = Node('B')
node3 = Node('C')

print(node1)  # Output: n A r 0 p A
print(node2)  # Output: n B r 0 p B
print(node3)  # Output: n C r 0 p C
