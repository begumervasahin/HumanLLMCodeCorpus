class DisjointSet:
    def __init__(self, size):
        if size < 0:
            raise ValueError("size must be >= 0")
        self.size = size
        # Initialize parent pointers to themselves and rank to 0
        self.parent = [i for i in range(size + 1)]
        self.rank = [0] * (size + 1)

    def find(self, x):
        """
        Find the representative item of the set that x is a part of.
        Uses path compression to flatten the structure of the tree,
        which leads to amortized constant time complexity.
        """
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])  # Path compression
        return self.parent[x]

    def union(self, x, y):
        """
        Perform the union of two sets: one containing x and the other containing y.
        Uses union by rank to attach the smaller tree to the root of the larger tree.
        """
        rootX = self.find(x)
        rootY = self.find(y)

        if rootX != rootY:
            if self.rank[rootX] > self.rank[rootY]:
                self.parent[rootY] = rootX
            elif self.rank[rootX] < self.rank[rootY]:
                self.parent[rootX] = rootY
            else:
                self.parent[rootY] = rootX
                self.rank[rootX] += 1
