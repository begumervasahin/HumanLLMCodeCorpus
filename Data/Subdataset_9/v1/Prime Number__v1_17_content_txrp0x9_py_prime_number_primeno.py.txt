print("Welcome random tester, this program will print all the prime numbers from 2 to n")
n = int(input("Enter n:\n"))

primes = [True] * (n + 1)  # Initialize a list to store whether each number is prime or not

p = 2
while p * p <= n:
    if primes[p] == True:
        for i in range(p * 2, n + 1, p):
            primes[i] = False  # Mark multiples of prime numbers as non-prime
    p += 1

# Print all prime numbers from 2 to n
for x in range(2, n):
    if primes[x]:
        print(x)
