import matplotlib.pyplot as plt
from ktsne import Ktsne
from sklearn import datasets
from sklearn.preprocessing import MinMaxScaler
from sklearn.decomposition import PCA
from sklearn.utils import shuffle

# Load dataset
digits = datasets.load_digits()
X = digits.data
y = digits.target

# Shuffle and select a subset of data
X, y = shuffle(X, y)
X = X[:500]
y = y[:500]

# Scale the data
scaler = MinMaxScaler(feature_range=(-1, 1))
X_scaled = scaler.fit_transform(X)

# Define t-SNE parameters
f_opts = {'p_degree': 2.0, 'p_dims': 12, 'eta': 25.0,
          'perplexity': 50.0, 'n_dims': 2, 'ker': 'pca', 'gamma': 1.0}

# Create subplot for original PCA plot
plt.subplot(2, 1, 1)

# Perform PCA and plot
X_pca = PCA(n_components=2).fit_transform(X_scaled)
plt.scatter(X_pca[:, 0], X_pca[:, 1], c=y, cmap=plt.cm.Set1, edgecolor='k')
plt.xlabel('PCA Component 1')
plt.ylabel('PCA Component 2')
plt.title('PCA without ktsne')

# Initialize and run t-SNE with ktsne
k_tsne = Ktsne(X_scaled, f_opts=f_opts)
X_reduced = k_tsne.get_solution(3000)
X_reduced = scaler.fit_transform(X_reduced)

# Create subplot for t-SNE plot
plt.subplot(2, 1, 2)
plt.scatter(X_reduced[:, 0], X_reduced[:, 1], c=y, cmap=plt.cm.Set1, edgecolor='k')
plt.xlabel('t-SNE Component 1')
plt.ylabel('t-SNE Component 2')
plt.title('t-SNE with %s kernel' % f_opts["ker"])

# Adjust subplot layout and display plot
plt.subplots_adjust(hspace=0.5)
plt.show()
