import base64
from Crypto.PublicKey import RSA
from Crypto.Cipher import PKCS1_v1_5
from Crypto import Random

# Load encrypted data from file
file = "secret.txt"
with open(file, "rb") as f:
    encrypted_data = f.read()

# Decode base64-encoded data
decoded_data = base64.b64decode(encrypted_data)

# Load RSA private key from file
private_key_file = "rsa_private_key.pem"
with open(private_key_file, "rb") as f:
    private_key_data = f.read()
    rsa_key = RSA.importKey(private_key_data)

# Decrypt using PKCS1_v1_5
decryptor = PKCS1_v1_5.new(rsa_key)
decrypted_data = decryptor.decrypt(decoded_data, Random.new().read)
print("Decrypted using Crypto module PKCS1_v1_5:", decrypted_data)

# Alternatively, using the rsa module
import rsa
priv_key = rsa.PrivateKey.load_pkcs1(private_key_data)
rsa_decrypted_data = rsa.decrypt(decoded_data, priv_key)
print("Decrypted using rsa module:", rsa_decrypted_data)
