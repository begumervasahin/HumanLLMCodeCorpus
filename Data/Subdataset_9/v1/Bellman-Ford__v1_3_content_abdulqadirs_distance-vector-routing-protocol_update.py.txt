# Read router configurations and store port information
def read_routers_from_file(filename):
    with open(filename, 'r') as file:
        num_routers = int(file.readline())
        ports_info = {}
        for _ in range(num_routers):
            router_line = file.readline()
            router_data = router_line.split()
            router_data[1] = float(router_data[1])
            ports_info[router_data[0]] = int(router_data[2])
        return ports_info
