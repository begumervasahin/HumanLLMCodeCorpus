from curve import Curve
from point import Point

class Public_Key:
    curve = None
    p = 0
    P = None
    Q = None

    def __init__(self, p, k):
        """
        Initialize a Public_Key object with the modulus p and a factor k.

        :param p: modulus for arithmetic operations
        :param k: factor for generating the public key
        """
        self.p = p
        self.curve = Curve.make_curve(self.p)
        self.P = self.curve.random_point()
        self.Q = self.curve.multiply(k, self.P)

    def __repr__(self):
        """
        Return a string representation of the Public_Key object.

        :return: string representation of the Public_Key object
        """
        return f"Curve: {self.curve}\nPublic Key (P): {self.P}\nDerived Public Key (Q): {self.Q}"

    @staticmethod
    def make_public_key(p, k):
        """
        Static method to create a new Public_Key object.

        :param p: modulus for arithmetic operations
        :param k: factor for generating the public key
        :return: new Public_Key object
        """
        return Public_Key(p, k)

# Example usage:
public_key = Public_Key(23, 5)
print(public_key)  # Output: Curve: y^2 = x^3 + 0x + 7 (mod 23)
                   #         Public Key (P): (12, 10, 1)
                   #         Derived Public Key (Q): (18, 20, 1)

# Another example:
another_public_key = Public_Key.make_public_key(47, 9)
print(another_public_key)  # Output: Curve: y^2 = x^3 + 0x + 2 (mod 47)
                           #         Public Key (P): (19, 15, 1)
                           #         Derived Public Key (Q): (18, 32, 1)
