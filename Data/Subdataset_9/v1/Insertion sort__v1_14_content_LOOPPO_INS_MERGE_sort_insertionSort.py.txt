def insertion_sort(A):
    """
    Sorts a list using the Insertion Sort algorithm.

    Args:
        A (list): The list to be sorted.

    Returns:
        list: The sorted list.
    """
    for i in range(1, len(A)):
        key = A[i]
        j = i - 1
        while j >= 0 and key < A[j]:
            A[j + 1] = A[j]
            j -= 1
        A[j + 1] = key
    return A
