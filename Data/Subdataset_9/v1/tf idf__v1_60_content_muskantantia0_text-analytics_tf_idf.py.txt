import re
import string
import pandas as pd
from math import log

# Read the CSV file
data = pd.read_csv('tf_idf.csv')

# Get reviewer IDs and document titles
reviewer = data['id'].tolist()
reviewer.insert(0, "reviewerID")
doc = data['title'].tolist()

# Initialize punctuation removal regex
punc = re.compile('[%s]' % re.escape(string.punctuation))

# Function to flatten nested lists
def flatten(lists):
    wordlist = []
    for numbers in lists:
        for x in numbers:
            wordlist.append(x)
    return wordlist

# Lowercase, remove punctuation, and split documents into terms
term_vec = []
for d in doc:
    d = str(d).lower()
    d = punc.sub('', d)
    d = d.split(' ')
    term_vec.append(d)

# Flatten the list of terms
wordlist = flatten(term_vec)

# Function to convert word list to frequency dictionary
def wordListToFreqDict(wordlist):
    wordfreq = [wordlist.count(p) for p in wordlist]
    d = dict(zip(wordlist, wordfreq))
    items = [(v, k) for k, v in d.items()]
    items.sort()
    items.reverse()
    finals = [(k, v, round(log(round((len(wordlist) / v), 4), 10), 4), (v * (log((len(wordlist) / v), 10)))) for v, k in items]
    return finals

# Calculate word frequencies
finals = wordListToFreqDict(wordlist)

# Print the results
print(finals)
