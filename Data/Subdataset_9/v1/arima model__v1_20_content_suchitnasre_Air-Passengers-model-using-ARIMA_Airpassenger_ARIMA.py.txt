import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import warnings
import itertools
from statsmodels.tsa.seasonal import seasonal_decompose
import statsmodels.api as sm
from pyramid.arima import auto_arima
from statsmodels.tsa.stattools import adfuller

# Read the data
data = pd.read_csv("AirPassengers.csv")
data["Month"] = pd.to_datetime(data["Month"])
data.set_index("Month", inplace=True)
data = pd.Series(data["#Passengers"])  # Assuming "#Passengers" is the column containing passenger data

# Function to perform stationary test
def stationary_test(timeseries):
    rolling_mean = timeseries.rolling(12).mean()
    rolling_std = timeseries.rolling(12).std()
    orig = plt.plot(timeseries, color="blue", label="original")
    mean = plt.plot(rolling_mean, color="red", label="Mean")
    std = plt.plot(rolling_std, color="black", label="Std")
    print("Result of Dickey-Fuller test")
    DF_test = adfuller(timeseries, autolag='AIC')
    print("ADF statistics:", DF_test[0])
    print("p-value:", DF_test[1])
    print("No. of Lags used:", DF_test[2])
    print("No. of observation used:", DF_test[3])
    print("Critical values:")
    for key, value in DF_test[4].items():
        print((key, value))
    if DF_test[1] < 0.05:
        print("We fail to accept the null hypothesis")
        print("Data is stationary")
    else:
        print("We rejected the alternate hypothesis")
        print("Data is non-stationary")

# Perform stationary test on original data
stationary_test(data)

# Take log transformation of data
data_log = np.log(data)

# Perform stationary test on log transformed data
stationary_test(data_log)

# Calculate moving average
moving_avg = data_log.rolling(12).mean()

# Calculate difference between log transformed data and moving average
data_diff_log = data_log - moving_avg
data_diff_log.dropna(inplace=True)

# Perform stationary test on differenced data
stationary_test(data_diff_log)

# Exponentially Weighted Moving Average
expweighted_avg = data_log.ewm(span=12).mean()
diff_data_log_ewma = data_log - expweighted_avg

# Perform stationary test on exponentially weighted differenced data
stationary_test(diff_data_log_ewma)

# Differencing
data_log_diff = data_log - data_log.shift()
data_log_diff.dropna(inplace=True)

# Perform stationary test on differenced data
stationary_test(data_log_diff)

# Second differencing
data_log_diff_1 = data_log_diff - data_log_diff.shift()
data_log_diff_1.dropna(inplace=True)

# Perform stationary test on second differenced data
stationary_test(data_log_diff_1)

# Seasonal decomposition
decomposition = seasonal_decompose(data_log)
decomposition.plot()

# Auto ARIMA
result = auto_arima(data, start_p=2, d=1, start_q=2, max_p=5, max_d=5, max_q=5, start_P=0,
                    D=1, start_Q=1, max_P=5, max_D=5, max_Q=5, seasonal=True, stationary=False, m=12,
                    trace=True, suppress_warnings=True, stepwise=True, information_criterion="aic",
                    error_action="ignore")

result.aic()
result.summary()

# Split data into train and test sets
train = data.loc["1949-01-01":"1958-12-31"]
test = data.loc["1959-01-01":]

# Fit the model
result.fit()

# Make predictions
prediction = result.predict(n_periods=24)

# Calculate Mean Squared Error
mse = ((prediction - test) ** 2).mean()
print('The Mean Squared Error of our forecasts is {}'.format(round(mse, 2)))

# Plot predictions
prediction = pd.DataFrame(prediction, index=test.index, columns=["Pred"])
prediction.head()
prediction.tail()
pd.concat([test, prediction], axis=1).plot()

# SARIMAX Parameter Selection
p = d = q = range(0, 3)
Non_seasonal_pdq = list(itertools.product(p, d, q))
Seasonal_PDQ = [(x[0], x[1], x[2], 12) for x in list(itertools.product(p, d, q))]
warnings.filterwarnings("ignore")

best_result = [0, 0, 10000000]
for param in Non_seasonal_pdq:
    for param_seasonal in Seasonal_PDQ:
        try:
            mod = sm.tsa.statespace.SARIMAX(data, order=param, seasonal_order=param_seasonal,
                                            enforce_stationarity=False, enforce_invertibility=False)
            results = mod.fit()
            print('ARIMA{} x {} - AIC: {}'.format(param, param_seasonal, results.aic))
            if results.aic < best_result[2]:
                best_result = [param, param_seasonal, results.aic]
        except:
            continue

print('\nBest Result:', best_result)

# Fit the model with best parameters
mod = sm.tsa.statespace.SARIMAX(data, order=(best_result[0][0], best_result[0][1], best_result[0][1]),
                                seasonal_order=(best_result[1][0], best_result[1][1], best_result[1][2],
                                                best_result[1][3]), enforce_stationarity=False,
                                enforce_invertibility=False)
results = mod.fit()

# Make predictions
prediction = results.predict(n_periods=24)

# Calculate Mean Squared Error
mse = ((prediction - test) ** 2).mean()
print('The Mean Squared Error of our forecasts is {}'.format(round(mse, 2)))

# Plot predictions
prediction = pd.DataFrame(prediction, index=test.index, columns=["Pred"])
prediction.head()
prediction.tail()
pd.concat([test, prediction], axis=1).plot()

plt.show()
