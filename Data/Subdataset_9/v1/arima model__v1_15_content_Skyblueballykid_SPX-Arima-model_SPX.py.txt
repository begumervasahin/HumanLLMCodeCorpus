import numpy as np
import pandas as pd
from statsmodels.tsa.arima.model import ARIMA

# Load the SPX weekly dataset
SPX = pd.read_csv("SPX Weekly 2000-2017.csv")

# Display the first 5 rows of the dataset
print(SPX.head(n=5))

# Fit an ARIMA model with order (0, 1, 0)
model = ARIMA(SPX['Close'], order=(0, 1, 0))
model_fit = model.fit()

# Forecast the next value
outcome = model_fit.forecast()[0]
print("Forecasted value:", outcome)

# Display the model summary (optional)
print(model_fit.summary())
