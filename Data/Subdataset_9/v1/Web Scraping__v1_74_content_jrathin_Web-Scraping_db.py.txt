# Function to compare strings and find the most similar one
def compare(s, l):
    max_v = -1
    element = ''
    for i in l:
        ratio = SequenceMatcher(None, s, i).ratio()
        if ratio > max_v:
            max_v = ratio
            element = i
    return element

# Insert player data into the 'players' table
