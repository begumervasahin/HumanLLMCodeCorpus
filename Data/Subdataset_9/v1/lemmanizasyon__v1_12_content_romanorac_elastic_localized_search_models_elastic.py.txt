from elasticsearch_dsl import Document, analyzer, String

# Define custom analyzer
lemmagen_sl = analyzer('lemmagen_sl', type='custom',
                       tokenizer="uax_url_email",
                       filter=["lowercase"]
                       )

# Define Document class inheriting from elasticsearch_dsl's DocType
class MyDocument(Document):
    """
    The :class:`Document` class defines a Type in ElasticSearch
    """
    title = String(analyzer=lemmagen_sl)

# Example usage
if __name__ == "__main__":
    # Instantiate a document object
    doc = MyDocument()

    # Set values
    doc.title = "Example Document Title"

    # Access values
    print("Title:", doc.title)

    # Indexing in ElasticSearch
    # To index the document in Elasticsearch, you would typically use the `save` method.
    # For example:
    # doc.save()
