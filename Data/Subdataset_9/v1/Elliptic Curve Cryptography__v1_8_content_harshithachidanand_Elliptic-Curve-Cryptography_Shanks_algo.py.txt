print("Shank's algorithm!\n")
print("\n\nS table\n")
for x in range(46):
    val = (5 ** (45 * x)) % 2017
    print(x, " ", val, "\n")

print("\n\nT table\n")
for y in range(46):
    val = (1736 * (5 ** y)) % 2017
    print(y, "  ", val, "\n")

print("\n\nPairs (x, y) where S[x] == T[y]\n")
for x in range(46):
    x_val = (5 ** (45 * x)) % 2017
    for y in range(46):
        y_val = (1736 * (5 ** y)) % 2017
        if x_val == y_val:
            print(x, " ", y, "\n")
