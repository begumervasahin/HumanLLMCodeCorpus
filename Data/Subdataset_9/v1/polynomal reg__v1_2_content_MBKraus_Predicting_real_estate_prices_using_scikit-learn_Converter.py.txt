import pandas as pd
from geopy.geocoders import GoogleV3
from geopy.exc import GeocoderTimedOut

# Read CSV files
a = pd.read_csv('AmsterdamPage1to50.csv', sep=';')
b = pd.read_csv('AmsterdamPage51to100.csv', sep=';')
c = pd.read_csv('AmsterdamPage101to155.csv', sep=';')

# Concatenate dataframes
data = pd.concat([a, b, c], ignore_index=True)

# Extract last two characters from 'rooms' column
data['rooms_new'] = data['rooms'].str[-2:]

# Extract first four characters from 'zipcode' column
data['zipcode_new'] = data['zipcode'].str[:4]

# Remove ' von' from 'price' column
data['price_new'] = data['price'].str.replace(' von', '')

# Initialize Google Geocoding API
g = GoogleV3(api_key='your API-key')

latcoordinates = {}
loncoordinates = {}

# Geocode unique zipcodes
for x in data['zipcode_new'].unique():
    try:
        y = x + " Amsterdam"
        location = g.geocode(y, timeout=60)
        lat = location.latitude
        latcoordinates[x] = lat
        lon = location.longitude
        loncoordinates[x] = lon
        print(y)
    except GeocoderTimedOut as e:
        print("Error: geocode failed on input %s with message %s" % (y, e))

# Map latitude and longitude to dataframe
data["latitude"] = data["zipcode_new"].map(lambda x: latcoordinates.get(x))
data["longitude"] = data["zipcode_new"].map(lambda x: loncoordinates.get(x))

# Save dataframe to CSV
data.to_csv("Overview_new.csv", sep=";", encoding='utf-8')
