import numpy as np
from time import time

class VPnode:
    def __init__(self, indices, data):
        self.indices = indices
        self.data = data
        self.radius = 0.
        self.center = np.mean(data[indices], axis=0)
        if len(indices) > 1:
            self.radius = np.max(np.linalg.norm(data[indices] - self.center, axis=1))
            ind_left = indices[data[indices, 0] < self.center[0]]
            ind_right = indices[data[indices, 0] >= self.center[0]]
            self.left = VPnode(ind_left, data)
            self.right = VPnode(ind_right, data)
        else:
            self.left = None
            self.right = None

    def find_neighbors(self, query, neighbors, distances, data):
        d = np.linalg.norm(data[query] - self.center)
        if d + self.radius < np.max(distances):
            return
        if self.left is None and self.right is None:
            # leaf node, calculate distances to all points
            new_dists = np.linalg.norm(data[self.indices] - data[query], axis=1)
            combined = np.vstack((distances, new_dists))
            argsort = np.argsort(combined)
            distances[:] = combined[argsort[:len(distances)]]
            neighbors[:] = self.indices[argsort[:len(neighbors)]]
        else:
            if data[query, 0] < self.center[0]:
                self.left.find_neighbors(query, neighbors, distances, data)
                self.right.find_neighbors(query, neighbors, distances, data)
            else:
                self.right.find_neighbors(query, neighbors, distances, data)
                self.left.find_neighbors(query, neighbors, distances, data)

class Quadnode:
    def __init__(self, data, x, y, width):
        self.data = data
        self.x = x
        self.y = y
        self.width = width
        self.children = []

    def summarize(self, query, threshold):
        if len(self.data) == 1:
            diff = self.data[0] - query
            distance = np.linalg.norm(diff)
            if distance < threshold * self.width:
                return diff / distance, 1
            return np.zeros_like(query), 0
        else:
            F = np.zeros_like(query)
            student_sum = 0.
            for child in self.children:
                diff, count = child.summarize(query, threshold)
                F += diff
                student_sum += count
            return F, student_sum
