import numpy as np

def prototype_sharpness_ratio(X, som):
    """Prototype sharpness ratio for image datasets
    Parameters
    ----------
    X : array, shape = [n_samples, input_dim]
        input samples
    som : array
        SOM prototype vectors
    """
    size = int(np.sqrt(X.shape[-1]))
    X = X.reshape(-1, size, size, 1)
    som = som.reshape(-1, size, size, 1)
    data_sharpness = 0.0

    # Calculate sharpness for input samples
    for x in X:
        gx, gy = np.gradient(x[:, :, 0])
        gnorm = np.sqrt(gx ** 2 + gy ** 2)
        data_sharpness += np.mean(gnorm)
    data_sharpness /= X.shape[0]

    som_sharpness = 0.0

    # Calculate sharpness for SOM prototypes
    for prototype in som:
        gx, gy = np.gradient(prototype[:, :, 0])
        gnorm = np.sqrt(gx ** 2 + gy ** 2)
        som_sharpness += np.mean(gnorm)
    som_sharpness /= som.shape[0]

    # Calculate and return the sharpness ratio
    return som_sharpness / data_sharpness

# Example usage
if __name__ == "__main__":
    # Example input data and SOM prototypes
    X = np.random.rand(100, 64)  # 100 samples of 64-dimensional data
    som_prototypes = np.random.rand(10, 64)  # 10 SOM prototypes of 64 dimensions each

    # Calculate sharpness ratio
    sharpness_ratio = prototype_sharpness_ratio(X, som_prototypes)
    print("Sharpness Ratio:", sharpness_ratio)
