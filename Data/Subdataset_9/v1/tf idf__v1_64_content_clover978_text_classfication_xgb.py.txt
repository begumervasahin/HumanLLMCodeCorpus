import xgboost as xgb
from sklearn.metrics import classification_report, confusion_matrix

# Load data
dtrain = xgb.DMatrix('/home/dxx/Backup/STUDY/syd/data/train.txt')
dtest = xgb.DMatrix('/home/dxx/Backup/STUDY/syd/data/test.txt')

# XGBoost parameters
param = {'silent': 0,
         'max_depth': 10,
         'eta': 0.1,
         'subsample': 1.0,
         'min_child_weight': 5,
         'colsample_bytree': 0.2,
         'objective': 'multi:softmax',
         'num_class': 6}
num_round = 100
watchlist = [(dtrain, 'train')]

def train():
    # Train the model
    bst = xgb.train(param, dtrain, num_round, evals=watchlist)
    # Save the trained model
    bst.save_model('xtrain.model')

def test():
    # Load the trained model
    bst = xgb.Booster()
    bst.load_model('xtrain.model')
    # Predict on the test data
    preds = bst.predict(dtest)
    labels = dtest.get_label()
    # Write predictions to a file
    with open('result.txt', 'w') as f:
        for y, y_ in zip(labels, preds):
            f.write('label: {:d}\tprediction: {:d}\n'.format(int(y), int(y_)))
    # Print classification report and confusion matrix
    print(classification_report(labels, preds))
    print(confusion_matrix(labels, preds))

def main():
    train()
    test()

if __name__ == '__main__':
    main()
