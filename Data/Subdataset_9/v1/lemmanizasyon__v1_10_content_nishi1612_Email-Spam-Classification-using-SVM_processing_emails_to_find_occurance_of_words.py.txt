import os
import string
import numpy as np
import pandas as pd
from time import time
from nltk.corpus import stopwords
from nltk.stem.wordnet import WordNetLemmatizer

start_time = time()

# Read the wordslist.csv file
df = pd.read_csv('wordslist.csv', header=0)
words = df['word']
lmtzr = WordNetLemmatizer()

directory_in_str = "emails/"
directory = os.fsencode(directory_in_str)

# Create and write headers to the frequency.csv file
with open("frequency.csv", "w+") as f:
    for i in words:
        f.write(str(i) + ',')
    f.write('output\n')

# Process each email file in the directory
k = 0
for file in os.listdir(directory):
    file = file.decode("utf-8")
    file_name = str(os.getcwd()) + '/emails/' + ''.join(c for c in file if c not in {'b', "'"})
    k += 1

    # Read the email file
    with open(file_name, "r", encoding='utf-8', errors='ignore') as file_reading:
        words_list_array = np.zeros(words.size)

        # Process each word in the file
        for word in file_reading.read().split():
            word = lmtzr.lemmatize(word.lower())
            if (word in stopwords.words('english') or word in string.punctuation or len(word) <= 2 or word.isdigit()):
                continue
            for i, w in enumerate(words):
                if w == word:
                    words_list_array[i] += 1
                    break

        # Write the word frequencies to the frequency.csv file
        with open("frequency.csv", "a") as f:
            for i in range(words.size):
                f.write(str(int(words_list_array[i])) + ',')
            if len(file_name) == 68:
                f.write("-1")
            elif len(file_name) == 71:
                f.write("1")
            f.write('\n')

    if k % 100 == 0:
        print("Done " + str(k))

print("Time (in seconds) to segregate the entire dataset to form the input vector: " + str(round(time() - start_time, 2)))
