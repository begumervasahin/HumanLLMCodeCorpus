def quick_sort(A, verbose=0, desc=0):
    """
    Sorts a list of integers using the quick sort algorithm.

    Args:
        A (list): The unsorted list of integers.
        verbose (int, optional): Controls verbosity level. Defaults to 0.
        desc (int, optional): Controls whether the output is sorted in descending order. Defaults to 0.

    Returns:
        list: The sorted list.
    """
    def partition(lb, ub):
        pivot, start, end = A[lb], lb, ub
        while start < end:
            while A[start] <= pivot and start < ub:
                start += 1
            while A[end] > pivot and end > lb:
                end -= 1
            if start < end:
                A[start], A[end] = A[end], A[start]
            if verbose == 2:
                print("Sub:", pivot, start, end, "::", A)
        A[lb], A[end] = A[end], A[lb]
        return end

    def quick_sort(x, y):
        if x < y:
            loc = partition(x, y)
            if verbose:
                print("Iteration:", loc, x, y, "::", A)
            quick_sort(x, loc - 1)
            quick_sort(loc + 1, y)

    quick_sort(0, len(A) - 1)
    if desc:
        A = A[::-1]
    return A
