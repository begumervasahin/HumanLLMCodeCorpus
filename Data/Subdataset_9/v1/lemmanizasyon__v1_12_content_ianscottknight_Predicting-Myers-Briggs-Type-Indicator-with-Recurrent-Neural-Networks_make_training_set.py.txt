import os
import collections
import pandas as pd
import csv
import re

DATA_DIR = "data"
MBTI_UNCLEAN_CSV_PATH = os.path.join(DATA_DIR, "mbti_unclean.csv")
DIMENSIONS = ("IE", "NS", "TF", "PJ")

# Read CSV file into a pandas DataFrame
df = pd.read_csv(MBTI_UNCLEAN_CSV_PATH)

# Initialize counts dictionary using defaultdict
counts = collections.defaultdict(int)

# Iterate over dimensions
for dimension in DIMENSIONS:
    letter_1, letter_2 = dimension
    
    # Iterate over rows in the DataFrame
    for index, row in df.iterrows():
        mbti = row["type"]
        hundred_posts = row["posts"].split("|||")
        
        # Iterate over posts
        for post in hundred_posts:
            # Skip posts with URLs, empty posts, None posts, or posts without alphabetic characters
            if ("http" in post) or (post == "") or (post is None) or (not re.search("[a-zA-Z]", post)):
                continue
            
            # Update counts based on MBTI type
            if letter_1 in mbti:
                counts[letter_1] += 1
            if letter_2 in mbti:
                counts[letter_2] += 1

# Iterate over dimensions for post extraction
for dimension in DIMENSIONS:
    letter_1, letter_2 = dimension
    
    # Determine limit based on counts
    limit = min(counts[letter_1], counts[letter_2])
    
    # Initialize lists to store posts and counter
    posts = []
    i = 0
    
    # Iterate over rows in the DataFrame
    for index, row in df.iterrows():
        # Check if the current row's MBTI type contains the current letter
        if letter_1 in row["type"] or letter_2 in row["type"]:
            hundred_posts = row["posts"].split("|||")
            
            # Iterate over posts
            for post in hundred_posts:
                # Skip posts with URLs, empty posts, None posts, or posts without alphabetic characters
                if (
                    ("http" in post)
                    or (post == "")
                    or (post is None)
                    or (not re.search("[a-zA-Z]", post))
                ):
                    continue
                
                # Add post to the list and increment the counter
                posts.append(post)
                i += 1
                
                # Break if the limit is reached
                if i == limit:
                    break
    
    # Create a CSV file for the extracted posts
    train_csv_path = os.path.join(DATA_DIR, f"train_{letter_1}.csv")
    with open(train_csv_path, "w") as f:
        writer = csv.writer(f)
        for post in posts:
            writer.writerow([post])
