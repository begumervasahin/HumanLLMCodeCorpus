import os
import torchvision.models as models
import sys
import time
import math
from torch import nn
import torch.nn.init as init
from torchvision import datasets, transforms
import numpy as np
import torch.utils.data as data
from torch.utils.data.dataset import *
from torchvision.datasets.vision import *

DATA_PATH = "train/"
BATCH_SIZE = 64

# Define transformations for input images
TRANSFORM_IMG = transforms.Compose([
    transforms.Resize(224),
    transforms.ToTensor(),
    transforms.Normalize([0.485, 0.456, 0.406], [0.229, 0.224, 0.225])
])

# Load test data
test_data = datasets.ImageFolder(root=DATA_PATH, transform=TRANSFORM_IMG)

# Create data loader
data_loader = data.DataLoader(test_data, batch_size=BATCH_SIZE, shuffle=True)

# Load pre-trained Inception V3 model
model_pre = models.inception_v3(pretrained=True).cuda()

# Modify the fully connected layer of the model
num_ftrs = model_pre.fc.in_features
model_pre.fc = nn.Sequential()

# Freeze parameters so we don't backprop through them
for param in model_pre.parameters():
    param.requires_grad = False

# Set input size
input_size = 299

# Set the model to evaluation mode
model_pre.eval()

# Initialize counters and lists
cnt = 0
data_x = []
data_y = []

# Iterate through the data loader
for x, y in data_loader:
    cnt += x.shape[0]
    xx = model_pre(x.cuda())  # Forward pass through the model
    data_x.extend(list(xx.cpu()))  # Append output to data_x
    data_y.extend(y.tolist())  # Append labels to data_y

# Convert data lists to numpy arrays
train_data_x = [x.numpy() for x in data_x]
train_data_x = np.array(train_data_x)
train_data_y = np.array(data_y)

# Print shape of the data
print(train_data_x.shape)

# Save the data as numpy arrays
np.save("tiny_imagenet_train_x", train_data_x)
np.save("tiny_imagenet_train_y", train_data_y)
