from difflib import SequenceMatcher

# Define custom lambda function
custom_lambda = lambda x: x in " \t"

# Initialize SequenceMatcher instances with different parameters
s1 = SequenceMatcher(None, " abcd", "abcd abcd")
s2 = SequenceMatcher(custom_lambda, " abcd", "abcd abcd")
s3 = SequenceMatcher(None, "abxcd", "abcd")
s4 = SequenceMatcher(None, "qabxcd", "abycdf")
s5 = SequenceMatcher(None, "abcd", "bcde")

# Demonstrate find_longest_match method
print("Longest match (default lambda):", s1.find_longest_match(0, 5, 0, 9))
print("Longest match (custom lambda):", s2.find_longest_match(0, 5, 0, 9))

# Demonstrate get_matching_blocks method
print("Matching blocks:", s3.get_matching_blocks())
