import pyclt  # Import the package for Central Limit Theorem based algorithm
import fp2ar  # Import the package to form association rules from frequent patterns
import re  # Import the regular expression module for string splitting and extracting items from transactions

def read_dataset(file_name):
    """
    Read the transactional dataset from the given file and return a list of transactions.
    Each transaction is represented as a list of integers.
    """
    transactions = []
    with open(file_name, 'r') as f:
        for line in f:
            # Split the dataset by space or newline and extract integers
            items = re.findall(r'\d+', line)
            if items:
                transactions.append([int(item) for item in items])
    return transactions

if __name__ == '__main__':
    # Read the transactional dataset from the file
    dataset_file = "kosarak.dat"
    dataset = read_dataset(dataset_file)

    # Mine dataset with the minimum support of 3%
    # The range of deviation of any support is limited in [-0.5%, 0.5%] with
    # probability at least 99%
    # fi is the formed frequent pattern
    # sN is the size of the sample
    fi, sN = pyclt.getFP(dataset, 0.03, 0.005, 0.01)

    # Form the association rules from fi
    # The minimum confidence is set to 75%
    # The minimum lift is set to 1
    ar = fp2ar.getAR(fi, sN, 0.75, 1)
