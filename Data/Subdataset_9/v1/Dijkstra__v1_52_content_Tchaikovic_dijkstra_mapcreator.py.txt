import os
import csv

def parse_and_create_map(coordinates_directory):
    points = []
    
    # Iterate over files in the directory
    for filename in os.listdir(coordinates_directory):
        with open(os.path.join(coordinates_directory, filename), 'r') as f:
            reader = csv.reader(f, delimiter=',')
            for line in reader:
                # Remove leading and trailing whitespaces
                line = [x.strip() for x in line]
                points.extend(line)

    # Filter out empty strings
    filtered = filter(None, points)
    
    # Initialize points matrix
    points_matrix = [[0 for _ in range(len(filtered))] for _ in range(2)]
    
    # Populate points matrix
    for i in range(len(filtered)):
        x, y = map(int, filtered[i][1:].split("y"))
        points_matrix[0][i] = x
        points_matrix[1][i] = y
    
    # Initialize map
    max_x = max(points_matrix[0]) + 1
    max_y = max(points_matrix[1]) + 1
    map_output = [['.' for _ in range(max_x)] for _ in range(max_y)]
    
    # Populate map with points
    for i in range(len(points_matrix[0])):
        map_output[points_matrix[1][i]][points_matrix[0][i]] = 'x'
    
    # Write map to file
    with open('map.txt', 'w') as thefile:
        for row in map_output:
            thefile.write(''.join(row) + '\n')
    
    return map_output

# Example usage
parse_and_create_map("CoordinateSystem")
