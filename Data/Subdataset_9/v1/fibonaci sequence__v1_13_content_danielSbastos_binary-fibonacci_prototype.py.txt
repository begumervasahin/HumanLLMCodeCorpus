import numpy as np
import matplotlib.pyplot as plt

def fib_iterative(n):
    """Generates the nth Fibonacci number iteratively."""
    a, b = 0, 1
    for _ in range(n):
        a, b = b, a + b
    return a

def format_bin(number, width=80):
    """Formats a number into its binary representation with a fixed width."""
    return format(number, f'0{width}b')

def generate_fibonacci_sequence(n):
    """Generates a list of the first n Fibonacci numbers."""
    return [fib_iterative(i) for i in range(n + 1)]

def convert_to_binary(fib_list):
    """Converts a list of numbers into their binary representations."""
    return [format_bin(num) for num in fib_list]

def split_binary_list(fib_list_bin):
    """Splits the binary strings into lists of integers."""
    return [[int(bit) for bit in num] for num in fib_list_bin]

def visualize_binary_matrix(binary_matrix):
    """Visualizes a binary matrix using matplotlib."""
    plt.figure(figsize=(10, 10))
    plt.imshow(binary_matrix, interpolation='nearest', cmap='gray')
    plt.title("Binary Representation of Fibonacci Sequence")
    plt.xlabel("Bit Position")
    plt.ylabel("Fibonacci Index")
    plt.show()

def visualize_bit_distribution(binary_matrix):
    """Visualizes the distribution of bits in the binary matrix."""
    hist_one = [sum(row) for row in binary_matrix]
    hist_zero = [len(binary_matrix[0]) - sum(row) for row in binary_matrix]
    indices = list(range(len(hist_one)))

    plt.bar(indices, hist_one, 0.6, label='Ones', color='r')
    plt.bar(indices, hist_zero, 0.6, label='Zeros', color='y', bottom=hist_one)

    plt.title("Bit Distribution in Fibonacci Binary Representation")
    plt.xlabel("Fibonacci Index")
    plt.ylabel("Count")
    plt.legend(loc="upper left")
    plt.show()

if __name__ == "__main__":
    n = 100
    fib_list = generate_fibonacci_sequence(n)
    fib_list_bin = convert_to_binary(fib_list)
    fib_list_bin_splited = split_binary_list(fib_list_bin)
    splited_binary_table = np.array(fib_list_bin_splited)

    visualize_binary_matrix(splited_binary_table)
    visualize_bit_distribution(splited_binary_table)
