import csv
import nltk
from nltk.stem import PorterStemmer
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from nltk.probability import FreqDist
from nltk.collocations import BigramCollocationFinder
from nltk.metrics import BigramAssocMeasures
from string import punctuation
from collections import Counter

# Download NLTK resources if not already downloaded
nltk.download('punkt')
nltk.download('stopwords')
nltk.download('averaged_perceptron_tagger')

# Read data from CSV
with open('rev.csv', 'r') as csv_file:
    csv_reader = csv.reader(csv_file)
    next(csv_reader)
    reviewsList = [line[3] for line in csv_reader]

# Initialize Porter Stemmer and stopwords
ps = PorterStemmer()
stop_words = set(stopwords.words('english'))

# Tokenize, stem, and filter stopwords
tokenizedList = [word_tokenize(review) for review in reviewsList]
stemsList = [[ps.stem(word) for word in tokens] for tokens in tokenizedList]
filtered_reviews = [[word for word in stems if word not in stop_words] for stems in stemsList]

# Function to preprocess text
def preprocess(text):
    tokenizedText = word_tokenize(text)
    filtered_words = [word for word in tokenizedText if word not in stop_words and word not in punctuation]
    return filtered_words

# Concatenate all reviews
reviewsText = ' '.join(reviewsList)

# Get most common words
most_common_words = Counter(preprocess(reviewsText)).most_common(10)

# Function to extract n-grams
def extract_ngrams(text, n):
    words = preprocess(text)
    return list(zip(*[words[i:] for i in range(n)]))

# Get most common bigrams
bigrams = extract_ngrams(reviewsText, 2)
most_common_bigrams = Counter(bigrams).most_common(10)

# Function to score bigrams
def score_bigrams(bigrams):
    bgm = BigramAssocMeasures()
    finder = BigramCollocationFinder.from_documents([bigrams])
    scored = finder.score_ngrams(bgm.likelihood_ratio)
    return scored

# Get scored bigrams
scored_bigrams = score_bigrams(bigrams)
sorted_scored_bigrams = [bigram for bigram, score in scored_bigrams]

# Function to count POS tags
def count_pos_tags(text):
    tagged_text = nltk.pos_tag(word_tokenize(text))
    tag_fd = nltk.FreqDist(tag for (word, tag) in tagged_text)
    return tag_fd.most_common()

# Get most common POS tags
most_common_tags = count_pos_tags(reviewsText)

# Function to find words with specific POS tag
def find_words_with_tag(text, pos_tag):
    tagged_text = nltk.pos_tag(word_tokenize(text))
    return [word for word, tag in tagged_text if tag == pos_tag]

# Get words with specific POS tag
pronouns = find_words_with_tag(reviewsText, 'PRP')

# Printing results
print("Most Common Words:", most_common_words)
print("Most Common Bigrams:", most_common_bigrams)
print("Scored Bigrams:", sorted_scored_bigrams)
print("Most Common POS Tags:", most_common_tags)
print("Pronouns:", pronouns)
