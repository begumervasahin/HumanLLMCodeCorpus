import csv
import time
from selenium import webdriver

# Setting up Firefox profile with proxy
profile = webdriver.FirefoxProfile()
profile.set_preference("network.proxy.type", 1)
profile.set_preference("network.proxy.http", "200.37.54.10")
profile.set_preference("network.proxy.http_port", 57040)
profile.update_preferences()

# Maximum page number and digits
MAX_PAGE_NUM = 50
MAX_PAGE_DIG = 5

# Initializing Firefox WebDriver with the profile
driver = webdriver.Firefox(firefox_profile=profile)

# Open a CSV file to write the results
with open("idmail.csv", "w", newline='', encoding='utf-8') as csvfile:
    csvwriter = csv.writer(csvfile)
    
    # Loop through each page
    for i in range(1, MAX_PAGE_NUM + 1):
        page_num = (MAX_PAGE_DIG - len(str(i))) * "0" + str(i)
        url = "http://jecherchemonexpertcomptable.oecmaroc.com/users/index/" + page_num
        driver.get(url)
        
        # Wait for some time to load the page
        time.sleep(3)
        
        # Find all elements with id="page-content"
        ids = driver.find_elements_by_xpath('//*[@id="page-content"]')
        
        # Write data to CSV
        for element in ids:
            csvwriter.writerow([element.text])
        
        # Log progress
        print(f"Page {page_num} processed")
    
# Close the WebDriver
driver.quit()

print("Finished scraping all pages.")
