class BinaryHeap:
    def __init__(self):
        self.heap_list = [0]  # Initial dummy element
        self.current_size = 0

    def insert(self, key):
        """
        Insert a new key into the heap.
        """
        self.heap_list.append(key)
        self.current_size += 1
        self._percolate_up(self.current_size)

    def delete_min(self):
        """
        Delete the minimum element from the heap.
        """
        if self.current_size == 0:
            return None
        min_value = self.heap_list[1]
        self.heap_list[1] = self.heap_list[self.current_size]
        self.current_size -= 1
        self.heap_list.pop()
        self._percolate_down(1)
        return min_value

    def build_heap(self, alist):
        """
        Build a heap from the given list.
        """
        self.current_size = len(alist)
        self.heap_list = [0] + alist[:]
        mid_point = len(alist) // 2
        for i in range(mid_point, 0, -1):
            self._percolate_down(i)

    def _percolate_up(self, i):
        """
        Restore heap order from a child node to the root.
        """
        while i // 2 > 0:
            if self.heap_list[i] < self.heap_list[i // 2]:
                self._swap(i, i // 2)
            i //= 2

    def _percolate_down(self, i):
        """
        Restore heap order from the root to a leaf node.
        """
        while i * 2 <= self.current_size:
            mc = self._min_child(i)
            if self.heap_list[i] > self.heap_list[mc]:
                self._swap(i, mc)
            i = mc

    def _min_child(self, i):
        """
        Return the index of the smallest child.
        """
        left_child = i * 2
        right_child = left_child + 1
        if right_child > self.current_size:
            return left_child
        else:
            return left_child if self.heap_list[left_child] < self.heap_list[right_child] else right_child

    def _swap(self, i, j):
        """
        Swap elements at indices i and j in the heap.
        """
        self.heap_list[i], self.heap_list[j] = self.heap_list[j], self.heap_list[i]


# Example usage
bh = BinaryHeap()
bh.build_heap([9, 5, 6, 2, 3])
print(bh.delete_min())
print(bh.delete_min())
