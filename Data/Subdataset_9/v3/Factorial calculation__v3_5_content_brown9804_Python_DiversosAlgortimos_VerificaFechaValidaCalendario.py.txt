def is_valid_day(day):
    """
    Validate the given day.

    Args:
        day (int): The day to validate.

    Returns:
        bool: True if the day is valid, False otherwise.
    """
    return 1 <= day <= 31

def is_valid_month(month):
    """
    Validate the given month.

    Args:
        month (int): The month to validate.

    Returns:
        bool: True if the month is valid, False otherwise.
    """
    return 1 <= month <= 12

def is_valid_year(year):
    """
    Validate the given year.

    Args:
        year (int): The year to validate.

    Returns:
        bool: True if the year is valid, False otherwise.
    """
    return year >= 1582

def is_leap_year(year):
    """
    Check if the given year is a leap year.

    Args:
        year (int): The year to check.

    Returns:
        bool: True if the year is a leap year, False otherwise.
    """
    return (year % 400 == 0) or ((year % 4 == 0) and (year % 100 != 0))

def is_valid_date(day, month, year):
    """
    Validate the given date.

    Args:
        day (int): The day of the date.
        month (int): The month of the date.
        year (int): The year of the date.

    Returns:
        None: Prints whether the date is valid or not.
    """
    if is_valid_day(day) and is_valid_month(month) and is_valid_year(year):
        if (month in [4, 6, 9, 11]) and day == 31:
            print("The date is invalid.")
        elif month == 2 and (day == 30 or day == 31):
            print("The date is invalid.")
        elif month == 2 and day == 29 and not is_leap_year(year):
            print("The date is invalid.")
        else:
            print("The date is valid.")
    else:
        print("The date is invalid.")

def get_user_input():
    """
    Get user input for day, month, and year.

    Returns:
        tuple: A tuple containing day, month, and year.
    """
    day = int(input("Enter the day: "))
    month = int(input("Enter the month: "))
    year = int(input("Enter the year: "))
    return day, month, year

def main():
    """
    Main function to run the program.
    """
    day, month, year = get_user_input()
    is_valid_date(day, month, year)

if __name__ == "__main__":
    main()
