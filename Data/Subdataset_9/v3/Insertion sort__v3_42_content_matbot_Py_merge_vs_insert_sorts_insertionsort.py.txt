def insertion_sort(array):
    """
    Sorts an array of integers using the insertion sort algorithm.

    Args:
        array (list): The unsorted array of integers.

    Returns:
        list: The sorted array.
    """
    for i in range(1, len(array)):
        current_value = array[i]
        j = i - 1
        # Move elements of array[0..i-1], that are greater than current_value,
        # to one position ahead of their current position
        while j >= 0 and array[j] > current_value:
            array[j + 1] = array[j]
            j -= 1
        array[j + 1] = current_value
    return array

if __name__ == "__main__":
    # Read unsorted lists of integers from the file
    number_lists = []
    with open("data.txt", "r") as input_file:
        for line in input_file:
            # Parse each line and extract the integers
            data_list = list(map(int, line.strip().split()[1:]))
            number_lists.append(data_list)

    # Sort each list using insertion sort
    for number_list in number_lists:
        insertion_sort(number_list)

    # Write sorted lists to a new file
    with open("insertionsort.txt", "w") as output_file:
        for number_list in number_lists:
            # Write each sorted list to the file
            output_file.write(" ".join(map(str, number_list)) + "\n")
