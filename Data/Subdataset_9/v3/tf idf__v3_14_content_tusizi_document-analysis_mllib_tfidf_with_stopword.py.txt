import json
import jieba
from sklearn.feature_extraction.text import CountVectorizer, TfidfTransformer

def compute_tfidf(corpus):
    # Initialize CountVectorizer and TfidfTransformer 
    vectorizer = CountVectorizer()
    transformer = TfidfTransformer()

    # Compute TF-IDF
    tfidf_matrix = transformer.fit_transform(vectorizer.fit_transform(corpus))

    # Get feature names (words) and their corresponding weights
    words = vectorizer.get_feature_names()
    weights = tfidf_matrix.toarray()

    # Define the output filename
    output_file = "/vagrant/vocabulary/mllib.txt"     

    # Write TF-IDF values to the output file
    with open(output_file, "a+") as file_out:
        for doc_index, doc_weights in enumerate(weights):
            for word_index, weight in enumerate(doc_weights):
                # Retrieve word and its weight
                word = words[word_index]
                
                # Convert word and its weight to JSON format
                data = json.dumps([word, weight], ensure_ascii=False)

                # Write JSON data to the file
                file_out.write(data)
                file_out.write("\n")

if __name__ == "__main__":
    # Sample corpus
    corpus = [
        "This is the first document.",
        "This document is the second document.",      
        "And this is the third one.",
        "Is this the first document?",
    ]

    # Tokenize each document in the corpus
    def tokenize_document(document):
        return " ".join(jieba.cut(document, cut_all=True))

    # Tokenize the corpus
    corpus_tokenized = list(map(tokenize_document, corpus))

    # Compute TF-IDF values and write them to file    
    compute_tfidf(corpus_tokenized)
