from collections import namedtuple

# Define namedtuple for Operator Information
OperatorInfo = namedtuple('OperatorInfo', 'precedence associativity')
LEFT, RIGHT = 'Left Right'.split()

# Define operators with their precedence and associativity
OPERATORS = {
    '^': OperatorInfo(precedence=4, associativity=RIGHT),
    '*': OperatorInfo(precedence=3, associativity=LEFT),
    '/': OperatorInfo(precedence=3, associativity=LEFT),
    '+': OperatorInfo(precedence=2, associativity=LEFT),
    '-': OperatorInfo(precedence=2, associativity=LEFT),
    '(': OperatorInfo(precedence=9, associativity=LEFT),
    ')': OperatorInfo(precedence=0, associativity=LEFT),
}

# Token types
NUMBER, LPAREN, RPAREN = 'NUMBER ( )'.split()


def tokenize_input(expression):
    """
    Tokenizes the input expression.
    """
    tokens = expression.strip().split()
    token_values = []
    for token in tokens:
        if token in OPERATORS:
            token_values.append((token, OPERATORS[token]))
        else:
            token_values.append((NUMBER, token))
    return token_values


def shunting_yard(token_values):
    """
    Converts infix expression to reverse polish notation (RPN) using the Shunting Yard Algorithm.
    """
    output_queue, operator_stack = [], []
    table = [['TOKEN', 'ACTION', 'RPN OUTPUT', 'OP STACK', 'NOTES']]

    for token, value in token_values:
        if token is NUMBER:
            output_queue.append(value)
            table.append((value, 'Add number to output', ' '.join(output_queue), ' '.join(s[0] for s in operator_stack), ''))
        elif token in OPERATORS:
            token_type, (precedence, associativity) = token, value
            v = token_type

            while operator_stack:
                top_token, (top_precedence, top_associativity) = operator_stack[-1]

                if (associativity == LEFT and precedence <= top_precedence) or \
                        (associativity == RIGHT and precedence < top_precedence):
                    if token_type != RPAREN:
                        if top_token != LPAREN:
                            operator_stack.pop()
                            output_queue.append(top_token)
                        else:
                            break
                    else:
                        if top_token != LPAREN:
                            operator_stack.pop()
                            output_queue.append(top_token)
                        else:
                            operator_stack.pop()
                            table.append((v, '(Pop & discard "(")', ' '.join(output_queue), ' '.join(s[0] for s in operator_stack), ''))
                            break
                else:
                    break

            if token_type != RPAREN:
                operator_stack.append((token, value))
            else:
                pass  # Discard ")"

            table.append((v, 'Push op token to stack' if token_type != RPAREN else 'Discard ")"', ' '.join(output_queue), ' '.join(s[0] for s in operator_stack), ''))

    while operator_stack:
        top_token, (top_precedence, top_associativity) = operator_stack.pop()
        output_queue.append(top_token)
        table.append(('', '(Pop op)', ' '.join(output_queue), ' '.join(s[0] for s in operator_stack), 'Drain stack to output'))

    return table


if __name__ == '__main__':
    infix_expression = '3 + 4 * 2 / (1 - 5) ^ 2 ^ 3'
    print('Infix expression: %r\n' % infix_expression)

    rpn_output = shunting_yard(tokenize_input(infix_expression))

    max_column_widths = [len(max(x, key=len)) for x in zip(*rpn_output)]
    header_row = rpn_output[0]
    print(' '.join('{cell:^{width}}'.format(width=width, cell=cell) for (width, cell) in zip(max_column_widths, header_row)))

    for row in rpn_output[1:]:
        print(' '.join('{cell:<{width}}'.format(width=width, cell=cell) for (width, cell) in zip(max_column_widths, row)))

    print('\n The final output RPN is: %r' % rpn_output[-1][2])
