import numpy as np
import sys

# Initialize variables
output_matrix = []
sum_for_centroid = []
current_cluster = -2
len_matrix = -1

# Iterate over lines from stdin
for line in sys.stdin:
    line = line.strip()
    cluster, index, matrix_string = line.split('\t')
    matrix = np.array([float(x) for x in matrix_string.split(',')])
    len_matrix = len(matrix)
    cluster = int(cluster)
    index = int(index)

    # Check if a new cluster has been encountered
    if current_cluster != cluster:
        # If not the first cluster, calculate and print the centroid of the previous cluster
        if current_cluster != -2:
            new_centroid = sum_for_centroid / len(output_matrix)
            print('%s\t%s\t%s' % (current_cluster, ','.join(map(str, output_matrix)), ','.join(map(str, new_centroid))))
        
        # Reset output matrix and sum for centroid for the new cluster
        output_matrix = []
        sum_for_centroid = np.zeros(len_matrix)
        current_cluster = cluster

    # Append index to output matrix and update sum for centroid
    output_matrix.append(index)
    sum_for_centroid += matrix

# Calculate and print the centroid of the last cluster
new_centroid = sum_for_centroid / len(output_matrix)
print('%s\t%s\t%s' % (current_cluster, ','.join(map(str, output_matrix)), ','.join(map(str, new_centroid))))
