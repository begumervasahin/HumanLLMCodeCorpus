class LinkedListNode:
    """A class representing a node in a linked list."""

    def __init__(self, data, next_node=None):
        """Initialize the node with data and a reference to the next node."""
        self.data = data
        self.next = next_node


class LinkedList:
    """A class representing a linked list."""

    def __init__(self, data=None):
        """Initialize an empty linked list or a linked list with one node."""
        self.head = LinkedListNode(data) if data is not None else None

    def remove(self, data):
        """Remove a node with the given data from the linked list."""
        if self.head is None:
            return -1  # List is empty

        current = self.head
        prev = None

        while current:
            if current.data == data:
                if prev is None:
                    # Removing head node
                    self.head = current.next
                else:
                    prev.next = current.next
                return 0  # Node removed successfully
            prev = current
            current = current.next

        return -1  # Node not found

    def insert(self, data):
        """Insert a node with the given data into the linked list."""
        new_node = LinkedListNode(data)
        if self.head is None:
            self.head = new_node
        elif data < self.head.data:
            # Insert before head
            new_node.next = self.head
            self.head = new_node
        else:
            current = self.head
            while current.next and current.next.data < data:
                current = current.next
            new_node.next = current.next
            current.next = new_node

    def contains(self):
        """Print the contents of the linked list."""
        if self.head is None:
            print("Empty.")
        else:
            current = self.head
            while current:
                print(current.data, end=", ")
                current = current.next
            print()
