import unittest

def is_prime(number):
    """
    Check if a number is prime.

    Args:
        number (int): The number to be checked.

    Returns:
        bool: True if the number is prime, False otherwise.
    """
    if number <= 1:
        return False
    for divisor in range(2, int(number**0.5) + 1):
        if number % divisor == 0:
            return False
    return True

def generate_prime(number):
    """
    Generate prime numbers up to a given number.

    Args:
        number (int): The upper limit for generating prime numbers.

    Returns:
        str: A message indicating that prime generation is not implemented yet.
    """
    if not isinstance(number, int):
        return 'Cannot input anything besides a whole number'
    if number <= 1:
        return "Retry with a positive integer greater than 1"
    return "Prime generation not implemented yet"

class TestIsPrime(unittest.TestCase):
    """Test cases for is_prime function."""

    def test_zero(self):
        """Test if 0 is not prime."""
        self.assertFalse(is_prime(0))

    def test_negative_number(self):
        """Test if a negative number is not prime."""
        self.assertFalse(is_prime(-1))

    def test_one(self):
        """Test if 1 is not prime."""
        self.assertFalse(is_prime(1))

    def test_prime_number(self):
        """Test if 13 is prime."""
        self.assertTrue(is_prime(13))

    def test_big_prime_number(self):
        """Test if a large prime number is prime."""
        self.assertTrue(is_prime(373))

class TestGeneratePrime(unittest.TestCase):
    """Test cases for generate_prime function."""

    def test_string_input(self):
        """Test if string input is rejected."""
        self.assertEqual(generate_prime('i'), 'Cannot input anything besides a whole number')

    def test_float_input(self):
        """Test if float input is rejected."""
        self.assertEqual(generate_prime(2.58), 'Cannot input anything besides a whole number')

    def test_negative_input(self):
        """Test if negative input is rejected."""
        self.assertEqual(generate_prime(-2), "Retry with a positive integer greater than 1")

    def test_zero_input(self):
        """Test if zero input is rejected."""
        self.assertEqual(generate_prime(0), "Retry with a positive integer greater than 1")

    def test_input_one(self):
        """Test if input of 1 is rejected."""
        self.assertEqual(generate_prime(1), "Retry with a positive integer greater than 1")

if __name__ == '__main__':
    unittest.main()
