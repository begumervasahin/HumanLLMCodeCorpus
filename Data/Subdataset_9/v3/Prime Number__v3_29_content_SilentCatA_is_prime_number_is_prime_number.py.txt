def is_prime(n):
    """
    Check if a number is prime.

    Parameters:
        n (int): The number to check for primality.

    Returns:
        bool: True if n is prime, False otherwise.
    """
    if n < 2:
        return False
    for i in range(2, int(n ** 0.5) + 1):
        if n % i == 0:
            return False
    return True

def main():
    while True:
        n = input('> ')  # Prompt the user for input
        if n.lower() == 'q':  # Check if the user wants to quit
            break
        else:
            try:
                n = int(n)  # Convert input to an integer
                prime_check = is_prime(n)
                print(f'{n} is prime: {prime_check}')  # Print whether the number is prime or not
            except ValueError:
                print("Invalid input. Please enter a valid integer or 'q' to quit.")

if __name__ == "__main__":
    main()
