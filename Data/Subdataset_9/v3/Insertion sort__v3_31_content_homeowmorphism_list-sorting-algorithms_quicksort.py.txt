import random

def quicksort(arr):
    # Base case: If the list has one element or is empty, return it as it is already sorted
    if len(arr) <= 1:
        return arr

    # Choose a random pivot element
    pivot_index = random.randint(0, len(arr) - 1)
    pivot = arr[pivot_index]
    print("Pivot element:", pivot, "in list:", arr)

    # Initialize lists to hold elements less than, greater than, and equal to the pivot
    left = []
    right = []
    pivot_clones = [pivot]

    # Partition the list around the pivot
    for i, num in enumerate(arr):
        if i == pivot_index:
            continue
        elif num < pivot:
            left.append(num)
        elif num > pivot:
            right.append(num)
        else:
            pivot_clones.append(num)

    print("Left sublist:", left, "; Right sublist:", right)

    # Recursively sort the left and right sublists
    sorted_left = quicksort(left)
    sorted_right = quicksort(right)

    # Concatenate the sorted left sublist, pivot clones, and sorted right sublist       
    return sorted_left + pivot_clones + sorted_right

# Example usage
original_list = [3, 6, 8, 10, 1, 2, 1]
print("Original list:", original_list)
print("Sorted list:", quicksort(original_list))
