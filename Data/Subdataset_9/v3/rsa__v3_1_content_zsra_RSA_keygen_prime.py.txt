import random
import xgcd

def get_random_prime(min_value, max_value):
    """
    Get a random prime number within a given range.
    
    Args:
        min_value (int): Minimum value of the range.
        max_value (int): Maximum value of the range.
    
    Returns:
        int: Random prime number within the specified range.
    """
    primes_in_range = [num for num in range(min_value, max_value) if is_prime(num)]
    return random.choice(primes_in_range)

def is_prime(number):
    """
    Check if a number is prime using the Miller-Rabin primality test.
    
    Args:
        number (int): Number to check for primality.
    
    Returns:
        bool: True if the number is prime, False otherwise.
    """
    if number in {0, 1, 4, 6, 8, 9}:
        return False
    if number in {2, 3, 5, 7}:
        return True
    
    d = number - 1
    s = 0
    while d % 2 == 0:
        d //= 2
        s += 1

    def is_composite(a):
        if pow(a, d, number) == 1:
            return False
        for i in range(s):
            if pow(a, 2 ** i * d, number) == number - 1:
                return False
        return True

    for _ in range(8):
        witness = random.randrange(2, number)
        if is_composite(witness):
            return False
    return True

def are_coprime(a, b):
    """
    Check if two numbers are coprime using the Extended Euclidean Algorithm.
    
    Args:
        a (int): First number.
        b (int): Second number.
    
    Returns:
        bool: True if the numbers are coprime, False otherwise.
    """
    return xgcd.GCD(a, b) == 1
