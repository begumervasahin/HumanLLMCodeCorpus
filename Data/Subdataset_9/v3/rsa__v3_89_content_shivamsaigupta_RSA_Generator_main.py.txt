import tkinter as tk
from tkinter import *

class GUIGen:
    """Class for RSA key generation functionality."""
    def __init__(self):
        pass

class GUIE:
    """Class for RSA encryption functionality."""
    def __init__(self):
        pass

class GUID:
    """Class for RSA decryption functionality."""
    def __init__(self):
        pass

class GUI:
    """Main GUI class for RSA Key Generator."""
    def __init__(self):
        self.root = tk.Tk()
        self.initialize_window()
        self.create_components()
        self.pack_components()
        self.set_title()
        self.run_event_loop()

    def initialize_window(self):
        """Initialize the GUI window."""
        self.frame = Frame(self.root)

    def create_components(self):
        """Create GUI components: label and buttons."""
        self.label = Label(self.root, text="RSA Program by Shivam Sai Gupta for Discrete Maths, Prof. Mahavir Jhawar")
        self.button1 = Button(self.root, text="Generate", command=self.generate_rsa_keys)
        self.button2 = Button(self.root, text="Encrypt", command=self.encrypt_message)
        self.button3 = Button(self.root, text="Decrypt", command=self.decrypt_message)

    def pack_components(self):
        """Pack GUI components."""
        self.label.pack()
        self.button1.pack()
        self.button2.pack()
        self.button3.pack()

    def set_title(self):
        """Set title for the GUI window."""
        self.root.title("RSA Key Generator by Shivam Sai Gupta")      

    def run_event_loop(self):
        """Run the GUI event loop."""
        self.root.mainloop()

    def generate_rsa_keys(self):
        """Function to handle RSA key generation."""
        self.rsa_key_generator = GUIGen()

    def encrypt_message(self):
        """Function to handle message encryption using RSA."""
        self.message_encrypter = GUIE()

    def decrypt_message(self):
        """Function to handle message decryption using RSA."""
        self.message_decrypter = GUID()

if __name__ == "__main__":

    import sys

    my_gui = GUI()
