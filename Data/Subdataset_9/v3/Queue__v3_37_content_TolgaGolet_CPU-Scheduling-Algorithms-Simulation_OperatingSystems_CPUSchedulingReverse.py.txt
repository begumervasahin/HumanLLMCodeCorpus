import numpy as np
import matplotlib.pyplot as plt

def create_process_pool(number_of_processes, burst_time_lambda, arrival_time_scale, sjf_priority):
    # Initialize process pool
    process_pool = initialize_process_pool(number_of_processes)

    # Set arrival times and burst times using exponential and Poisson distributions
    max_arrival_time = set_arrival_and_burst_times(process_pool, number_of_processes, arrival_time_scale, burst_time_lambda)

    # Sort the process pool based on arrival times
    sort_process_pool(process_pool)

    # Print process pool details
    print_process_pool_info(number_of_processes, burst_time_lambda, arrival_time_scale, sjf_priority, process_pool)

    # Plot histograms for burst times and arrival times
    plot_histograms(process_pool)

def initialize_process_pool(number_of_processes):
    # Create an empty process pool
    process_pool = []
    for i in range(number_of_processes):
        process_pool.append([i+1, 0, 0, np.random.randint(1, 3)])  # [processID, arrivalTime, burstTime, priority]
    return process_pool

def set_arrival_and_burst_times(process_pool, number_of_processes, arrival_time_scale, burst_time_lambda):
    max_arrival_time = 0
    for process in process_pool:
        # Set arrival times using exponential distribution
        exp = np.random.exponential(arrival_time_scale)
        arrival_time = max(1, int(exp))
        process[1] = arrival_time
        max_arrival_time = max(max_arrival_time, exp)
        
        # Set burst times using Poisson distribution
        burst_time = max(1, np.random.poisson(burst_time_lambda))
        process[2] = burst_time
    return max_arrival_time

def sort_process_pool(process_pool):
    # Sort the process pool based on arrival times
    process_pool.sort(key=lambda x: x[1])

def print_process_pool_info(number_of_processes, burst_time_lambda, arrival_time_scale, sjf_priority, process_pool):
    # Print process pool details
    print("Creating the process pool...")
    print("Number Of Processes:", number_of_processes)
    print("Burst Time Lambda:", burst_time_lambda)
    print("Arrival Time Scale:", arrival_time_scale)
    print("SJF Priority:", sjf_priority * 100, "%")
    print("FCFS Priority:", (1 - sjf_priority) * 100, "%")
    print("[processID, arrivalTime, burstTime, priority]   priority=1=foreground(SJF), priority=2=batch(FCFS)")
    print("Process Pool:", process_pool)

def plot_histograms(process_pool):
    # Plot histograms for burst times and arrival times
    plt.figure(figsize=(10, 8))
    
    # Burst times histogram
    plt.subplot(2, 1, 1)
    plt.hist([process[2] for process in process_pool])
    plt.title('Poisson Burst Times')
    plt.ylabel('Frequency')
    plt.xlabel('Burst Times')
    
    # Arrival times histogram
    plt.subplot(2, 1, 2)
    plt.hist([process[1] for process in process_pool])
    plt.title('Exponential Arrival Times')
    plt.ylabel('Frequency')
    plt.xlabel('Arrival Times')
    
    plt.tight_layout()
    plt.show()

if __name__ == "__main__":
    number_of_processes = 15
    burst_time_lambda = 3
    arrival_time_scale = 15.0
    sjf_priority = 0.6
    create_process_pool(number_of_processes, burst_time_lambda, arrival_time_scale, sjf_priority)
