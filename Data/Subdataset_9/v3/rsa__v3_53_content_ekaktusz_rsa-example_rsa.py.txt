import random
import decimal_string
import prime

def greatest_common_divisor(a, b):
    """Find the greatest common divisor of two numbers."""
    while b != 0:
        a, b = b, a % b
    return a

def are_coprime(a, b):
    """Check if two numbers are coprime."""
    return greatest_common_divisor(a, b) == 1

def extended_euclidean_algorithm(a, b):
    """
    Find the modular multiplicative inverse using the extended Euclidean algorithm.
    """
    x, y = a, b
    t0, t1 = 0, 1
    r, t = b, 1
    while (x % y) != 0:
        q = x // y
        r = x % y
        t = (t0 - q * t1) % a
        x = y
        y = r
        t0 = t1
        t1 = t
    return t

def generate_rsa_keys():
    """
    Generate public and private keys for RSA encryption.
    """
    # Generate two random prime numbers
    p, q = prime.generate_two_random_prime(512)
    # Calculate n and Euler's totient function
    n = p * q
    phi = (p - 1) * (q - 1)
    # Choose a random integer e such that 1 < e < phi and e is coprime with phi
    e = random.randrange(1, phi)
    while not are_coprime(e, phi):
        e = random.randrange(1, phi)
    # Calculate the modular multiplicative inverse of e (i.e., d)
    d = extended_euclidean_algorithm(phi, e)
    return ((e, n), (d, n))

def encrypt_message(public_key, message):
    """
    Encrypt a message using RSA.
    """
    e, n = public_key
    message_number = decimal_string.text_to_integer(message)
    # Apply RSA encryption: ciphertext = plaintext^e mod n
    encrypted_message = pow(message_number, e, n)
    return encrypted_message

def decrypt_message(private_key, encrypted_message):
    """
    Decrypt an encrypted message using RSA.
    """
    d, n = private_key
    # Apply RSA decryption: plaintext = ciphertext^d mod n
    decrypted_message_number = pow(encrypted_message, d, n)
    decrypted_message = decimal_string.integer_to_text(decrypted_message_number)
    return decrypted_message

# Example usage:
# Generate public and private keys
public_key, private_key = generate_rsa_keys()

# Encrypt a message
message = "Hello, World!"
encrypted_message = encrypt_message(public_key, message)
print("Encrypted message:", encrypted_message)

# Decrypt the message
decrypted_message = decrypt_message(private_key, encrypted_message)
print("Decrypted message:", decrypted_message)
