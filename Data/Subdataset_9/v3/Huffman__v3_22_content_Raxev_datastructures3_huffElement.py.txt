class Comparable:
    """
    A base class for creating comparable objects.
    """

    def compare(self, other):
        """
        Compare method to be overridden by subclasses.
        """
        raise NotImplementedError("Subclasses must implement compare method.")


class HuffElement(Comparable):
    """
    Represents an element used in Huffman encoding.
    """

    def __init__(self, char):
        """
        Initializes a HuffElement object with the given character.
        """
        self.character = char  # Character
        self.frequency = 0  # Frequency count of the character
        self.code = ""  # Huffman code for the character

    def increment_frequency(self):
        """
        Increments the frequency count of the character.
        """
        self.frequency += 1

    def get_frequency(self):
        """
        Returns the frequency count of the character.
        """
        return self.frequency

    def set_frequency(self, count):
        """
        Sets the frequency count of the character.
        """
        self.frequency = count

    def get_code(self):
        """
        Returns the Huffman code for the character.
        """
        return self.code

    def set_code(self, code):
        """
        Sets the Huffman code for the character.
        """
        self.code = code

    def get_character(self):
        """
        Returns the character.
        """
        return self.character

    def set_character(self, char):
        """
        Sets the character.
        """
        self.character = char

    def compare(self, other_huff_elem):
        """
        Compares two HuffElement objects based on their frequency counts.
        Returns 1 if the frequency count of self is greater,
        -1 if it is less, and 0 if they are equal.
        """
        if self.frequency > other_huff_elem.get_frequency():
            return 1
        elif self.frequency < other_huff_elem.get_frequency():
            return -1
        else:
            return 0

    def __str__(self):
        """
        Returns a string representation of this HuffElement object.
        """
        return f"Character: {self.character}, Code: {self.code}, Frequency Count: {self.frequency}"


# Example usage
if __name__ == "__main__":
    # Create a HuffElement object for character 'a'
    huff_elem = HuffElement('a')
    print(huff_elem)  # Output the string representation of huff_elem
