import discord
from discord.ext import commands
import requests
from bs4 import BeautifulSoup

# Initialize Discord bot with a command prefix
bot = commands.Bot(command_prefix='!')

def get_stock_price(ticker):
    """Fetches the current price of a stock ticker from Yahoo Finance."""
    base_url = "https://finance.yahoo.com/quote/"
    response = requests.get(f"{base_url}{ticker}").content
    soup = BeautifulSoup(response, 'html.parser')
    # This class name might change in the future, check Yahoo Finance if it fails
    price_container = soup.find(class_="Trsdu(0.3s) Trsdu(0.3s) Fw(b) Fz(36px) Mb(-4px) D(b)")
    if price_container:
        return price_container.text
    else:
        return None

@bot.command(name='price')
async def fetch_stock_price(ctx, *, ticker: str):
    """Responds with the current price of the stock ticker."""
    stock_price = get_stock_price(ticker)
    if stock_price:
        await ctx.send(f"The stock price for {ticker.upper()} is ${stock_price} currently.")
    else:
        await ctx.send(f"Could not find the stock price for {ticker.upper()}.")

@bot.command(name='help', hidden=True)
async def custom_help(ctx):
    """Shows this message."""
    help_text = "Hello! Here are the commands you can use:\n"
    help_text += "`!price <TICKER>`: Gets the current price of the specified stock ticker.\n"
    # Add additional commands here
    await ctx.send(help_text)

@bot.event
async def on_ready():
    print(f'Logged in as {bot.user.name}')
    print(f'User ID: {bot.user.id}')
    print('------ Ready to go!')
