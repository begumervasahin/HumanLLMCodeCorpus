class Heap:
    def __init__(self):
        # Initialize the heap with a list containing a placeholder zero
        self.heap_list = [0]

    def insert(self, key_value):
        # Append the new key value to the heap array
        self.heap_list.append(key_value)
        # Update the first element of the heap array to reflect the current size
        self.update_heap_size()
        # Fix the heap property by moving the newly inserted element up
        self.heapify_up(self.size())

    def update_heap_size(self):
        # Update the first element of the heap list to reflect the current size
        self.heap_list[0] = self.size()

    def size(self):
        # Return the size of the heap
        return len(self.heap_list) - 1

    def heapify_up(self, index):
        # Move the element up the heap until its parent is smaller
        while index > 1:
            parent_index = index // 2
            if self.is_parent_greater(parent_index, index):
                # Swap the parent and the child if the parent is greater
                self.swap_nodes(parent_index, index)
                index = parent_index
            else:
                # Break the loop if the parent is smaller or equal
                break

    def is_parent_greater(self, parent_index, index):
        # Check if the distance of the parent node is greater than the distance of the current node
        return int(self.heap_list[parent_index].distance) > int(self.heap_list[index].distance)

    def swap_nodes(self, index1, index2):
        # Swap the nodes at the given indices
        self.heap_list[index1], self.heap_list[index2] = self.heap_list[index2], self.heap_list[index1]

    def extract_min(self):
        # Get the last index of the heap
        last_index = self.size()
        # Retrieve the minimum element from the top of the heap
        min_element = self.heap_list[1]
        # Replace the root of the heap with the last element
        self.heap_list[1] = self.heap_list[last_index]
        # Decrease the size of the heap
        self.update_heap_size()
        # Restore the heap property by moving the root down
        if self.size() > 1:
            self.heapify_down(1)
        # Remove the last element from the heap array
        self.heap_list.pop()
        return min_element

    def heapify_down(self, index):
        # Move the element down the heap until its children are greater
        while 2 * index <= self.size():
            left_child_index = 2 * index
            right_child_index = left_child_index + 1
            smaller_child_index = self.find_smaller_child(left_child_index, right_child_index)
            if self.is_child_smaller(smaller_child_index, index):
                # Swap with the smaller child if it is smaller than the parent
                self.swap_nodes(index, smaller_child_index)
                index = smaller_child_index
            else:
                # Break the loop if the parent is smaller or equal to its children
                break

    def find_smaller_child(self, left_child_index, right_child_index):
        # Return the index of the smaller child
        if right_child_index > self.size() or \
                int(self.heap_list[left_child_index].distance) < int(self.heap_list[right_child_index].distance):
            return left_child_index
        else:
            return right_child_index

    def is_child_smaller(self, child_index, parent_index):
        # Check if the distance of the child node is smaller than the distance of the parent node
        return int(self.heap_list[child_index].distance) < int(self.heap_list[parent_index].distance)

if __name__ == "__main__":
    # Creating a heap object
    h = Heap()

    # Inserting elements into the heap
    h.insert(5)
    h.insert(3)
    h.insert(8)
    h.insert(1)
    h.insert(10)

    # Extracting the minimum element from the heap
    print(h.extract_min())  # Output: 1
    print(h.extract_min())  # Output: 3
