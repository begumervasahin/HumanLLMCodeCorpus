import pandas as pd
import os
import nltk.data
import logging
import numpy as np
import gensim
from sklearn.metrics.pairwise import cosine_similarity
from KaggleWord2VecUtility import KaggleWord2VecUtility

def make_feature_vector(words, model, num_features):
    """Create a feature vector for a list of words."""
    feature_vector = np.zeros((num_features,), dtype="float32")
    num_words = 0
    index2word_set = set(model.wv.index2word)
    for word in words:
        if word in index2word_set:
            num_words += 1
            feature_vector += model.wv[word]
    return feature_vector / max(num_words, 1)

def get_avg_feature_vectors(skucollection, model, num_features):
    """Get average feature vectors for a collection of SKUs."""
    review_feature_vectors = np.zeros((len(skucollection), num_features), dtype="float32")
    for i, sku in enumerate(skucollection):
        if i % 1000 == 0:
            print(f"Processing SKU {i+1} of {len(skucollection)}")
        review_feature_vectors[i] = make_feature_vector(sku, model, num_features)
    return review_feature_vectors

def get_clean_skucollection(skucollection, column_name):
    """Get clean SKUs from a DataFrame column."""
    clean_skucollection = []
    for sku in skucollection[column_name]:
        clean_skucollection.append(KaggleWord2VecUtility.sku_to_wordlist(sku, remove_stopwords=False))
    return clean_skucollection

if __name__ == '__main__':
    # Read train and test data
    train = pd.read_csv(os.path.join(os.path.dirname(__file__), 'data_crowflower', 'train.csv'), header=0, delimiter=",", quoting=6)
    test = pd.read_csv(os.path.join(os.path.dirname(__file__), 'data_crowflower', 'test.csv'), header=0, delimiter=",", quoting=6)

    # Print the number of labeled train and test SKUs
    print(f"Read {len(train)} labeled train SKUs, {len(test)} labeled test SKUs")

    # Load tokenizer and set up logging
    tokenizer = nltk.data.load('tokenizers/punkt/english.pickle')
    logging.basicConfig(format='%(asctime)s : %(levelname)s : %(message)s', level=logging.INFO)

    # Set the number of features and load the Word2Vec model
    num_features = 300
    model = gensim.models.Word2Vec.load('300features_40minwords_10_SKU')

    # Create average feature vectors for train SKUs
    print("Creating average feature vectors for training SKUs")
    train_data_vectors = get_avg_feature_vectors(get_clean_skucollection(train, "product_title"), model, num_features)

    # Create average feature vectors for test SKUs
    print("Creating average feature vectors for test SKUs")
    test_data_vectors = get_avg_feature_vectors(get_clean_skucollection(test, "query"), model, num_features)

    # Compute cosine similarity between test and train SKUs
    print("Query ID, SKU ID, Cosine")
    for query, test_vector in enumerate(test_data_vectors):
        for isku, train_vector in enumerate(train_data_vectors):
            try:
                cos = cosine_similarity(test_vector.reshape(1, -1), train_vector.reshape(1, -1))[0][0]
                print(f"{query}, {isku}, {cos}")
            except Exception as e:
                print(f"Error processing query {query} and SKU {isku}: {e}")
