import os
import time
import shutil
import argparse

def sort_files(directory, separator):
    """
    Sort files based on modification date and move them to corresponding directories.

    Args:
        directory (str): The path to the directory containing files.
        separator (str): Separator for directory structure.
    """
    for (dirpath, _, filenames) in os.walk(directory):
        for filename in filenames:
            if filename.lower().endswith((".jpg", ".mp4", ".mts", ".mov")):
                parse_file(dirpath, filename, separator)

def parse_file(directory_path, filename, separator):
    """
    Parse each file and move it to a directory based on its modification date.

    Args:
        directory_path (str): The path to the directory containing the file.
        filename (str): The name of the file.
        separator (str): Separator for directory structure.
    """
    full_filepath = os.path.join(directory_path, filename)
    print("Processing file:", full_filepath)
    destination_path = get_destination_path(full_filepath, separator)
    if not os.path.exists(destination_path):
        os.makedirs(destination_path)
    new_filepath = os.path.join(destination_path, filename)
    print("Moving to:", new_filepath)
    os.rename(full_filepath, new_filepath)

def get_destination_path(full_filepath, separator):
    """
    Get the destination directory path based on the modification date of the file.

    Args:
        full_filepath (str): The full path to the file.
        separator (str): Separator for directory structure.

    Returns:
        str: The destination directory path.
    """
    modification_time = os.path.getmtime(full_filepath)
    formatted_date = time.strftime('%Y' + separator + '%Y %m %d', time.localtime(modification_time))
    return os.path.join(os.path.dirname(full_filepath), formatted_date)

def main():
    parser = argparse.ArgumentParser(description="Sort files based on modification date")
    parser.add_argument("--separator", default="\\", help="Separator for directory structure")
    parser.add_argument("--path", required=True, help="Path to the directory containing files")
    args = parser.parse_args()

    print("Separator:", args.separator)
    print("Path:", args.path)
    sort_files(args.path, args.separator)

if __name__ == "__main__":
    main()
