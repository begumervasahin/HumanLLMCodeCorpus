from Crypto.Cipher import AES
import hashlib
import random
import socket
from ecc import getcurvebyname

# Server and client configuration
host = "127.168.2.75"
port = 4446

def setup_server():
    # Server setup
    server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server_socket.bind((host, port))
    server_socket.listen(1)
    print("Listening for connections.. ")
    return server_socket.accept()

def setup_client():
    # Client setup
    client_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    client_socket.connect((host, port))
    return client_socket

def send_receive_data():
    # Sending and receiving data
    client_socket, _ = setup_server()
    data = input("Enter data to be sent: ")
    client_socket.send(data.encode())
    client_socket.close()

    client_socket = setup_client()
    msg = client_socket.recv(1024)
    print("Message from server: " + msg.strip().decode('ascii'))
    client_socket.close()

def setup_ecc():
    # ECC setup
    curve = getcurvebyname("ed25519")
    g = curve.G
    a = random.randint(1, 2**255 - 19)
    Alice = g * a
    Bob = g * random.randint(1, 2**255 - 19)
    return curve, g, a, Alice, Bob

def generate_keys(curve, g, a, Alice, Bob):
    # ECC communication and key generation
    k = [hashlib.blake2s() for _ in range(10)]

    for i in range(10):
        k[i].update(str((Bob + Alice * (-i) * a) * a).encode())
        k[i] = k[i].digest()

    return k

def encrypt_decrypt_messages(k, var):
    # Encryption and decryption
    cipher = [AES.new(key.digest(), AES.MODE_ECB) for key in k]
    en = [cipher[i].encrypt(var[i].encode()) for i in range(10)]

    m = hashlib.blake2s()
    m.update(str(Alice * random.randint(1, 2**255 - 19)).encode())
    Bob_key = m.digest()

    cipher1 = AES.new(Bob_key, AES.MODE_ECB)
    message = [cipher1.decrypt(en[i]).decode() for i in range(10)]

    return message

def main():
    send_receive_data()

    curve, g, a, Alice, Bob = setup_ecc()
    k = generate_keys(curve, g, a, Alice, Bob)

    # User input for variables
    var = [''] * 10
    for i in range(10):
        var[i] = input("Please enter something: ").ljust(32)
        print("You entered: " + str(var[i]))

    # Encrypting and decrypting messages
    message = encrypt_decrypt_messages(k, var)

    # Displaying decrypted messages
    print('\nBob decrypts the messages:')
    for i in range(10):
        print("message:", message[i])

if __name__ == "__main__":
    main()
