import math

def move_disk(source, target):
    """
    Print a message to move a disk from the source peg to the target peg.
    """
    print(f"Move disk from {source} to {target}")

def hanoi_tower_recursive(n, source, auxiliary, target):
    """
    Recursive implementation of the Tower of Hanoi algorithm.
    """
    if n == 1:
        move_disk(source, target)
    else:
        hanoi_tower_recursive(n - 1, source, target, auxiliary)
        move_disk(source, target)
        hanoi_tower_recursive(n - 1, auxiliary, source, target)

def execute_hanoi_tower(ndiscs):
    """
    Execute the Tower of Hanoi algorithm with the specified number of discs.
    """
    print(f"Executing Tower of Hanoi Algorithm with {ndiscs} discs:")
    hanoi_tower_recursive(ndiscs, "A", "B", "C")
