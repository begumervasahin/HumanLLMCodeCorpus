import fnmatch
import re

def does_match_fnmatch(pattern, string):
    """
    Checks if the given string matches the provided fnmatch pattern.
    """
    return fnmatch.fnmatch(string, pattern)

def does_match_regex(pattern, string):
    """
    Checks if the given string matches the provided regular expression pattern.
    """
    translated_regex = fnmatch.translate(pattern)
    compiled_regex = re.compile(translated_regex)
    return compiled_regex.match(string) is not None

# Example usage:
pattern = '*.txt'
string = 'foobar.txt'

fnmatch_result = does_match_fnmatch(pattern, string)
regex_result = does_match_regex(pattern, string)

# Output the results
print(f"Using fnmatch, the string matches the pattern: {fnmatch_result}")
print(f"Using regular expressions, the string matches the pattern: {regex_result}")
