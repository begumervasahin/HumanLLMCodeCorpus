# linkedlist.py
class LinkedList:
    class Node:
        def __init__(self, data):
            self.data = data
            self.next = None

    def __init__(self):
        self.head = None

    def push(self, data):
        """
        Pushes a new node with the given data onto the top of the stack.
        """
        new_node = self.Node(data)
        new_node.next = self.head
        self.head = new_node

    def pop(self):
        """
        Pops and returns the data from the top node of the stack.
        """
        if self.head is None:
            raise IndexError("Pop from empty stack")
        data = self.head.data
        self.head = self.head.next
        return data

    def top(self):
        """
        Returns the data from the top node of the stack without removing it.
        """
        if self.head is None:
            raise IndexError("Top from empty stack")
        return self.head.data

# eval_postfix.py
from linkedlist import LinkedList
from vartree import BinarySearchTree
from infixtotree import to_expr_tree

var_tree = BinarySearchTree()

def eval_postfix(postfix_expr):
    """
    Evaluates a postfix expression and returns the result.

    Args:
        postfix_expr (list): The postfix expression to evaluate.

    Returns:
        float: The result of the evaluation.
    """
    stack = LinkedList()
    for token in postfix_expr:
        if token[0].isdigit() or token[0].isalpha():
            stack.push(token)
        elif token == '=':
            top = stack.pop()
            bottom = stack.pop()
            if top[0].isalpha():
                top = var_tree.lookup(top)
            var_tree.assign(bottom, top)
            stack.push(bottom)
        else:
            top = stack.pop()
            bottom = stack.pop()
            if top[0].isalpha():
                top = var_tree.lookup(top)
            if bottom[0].isalpha():
                bottom = var_tree.lookup(bottom)
            temp = eval(str(bottom) + token + str(top))
            stack.push(str(temp))
    if stack.top()[0].isdigit():
        return float(stack.top())
    return var_tree.lookup(stack.top())

# Example usage
postfix_expression = ['2', '3', '+', '5', '*']
result = eval_postfix(postfix_expression)
