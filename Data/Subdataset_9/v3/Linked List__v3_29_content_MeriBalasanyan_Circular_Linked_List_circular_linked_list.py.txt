class Node:
    """A node in a circular linked list."""
    def __init__(self, data=None):
        self.data = data
        self.next = None

class Student:
    """Represents a student with a name, ID, and GPA."""
    def __init__(self, name: str, ID: str, GPA: str):
        self.name = name
        self.ID = ID
        self.GPA = GPA

class CircularLinkedList:
    """A circular linked list of students."""
    def __init__(self):
        self.head = None

    def find(self, name: str, ID: str) -> Node or None:
        """Finds a student node by name and ID."""
        current = self.head
        while current and (current.data.name != name or current.data.ID != ID):
            current = current.next
            if current == self.head:
                return None
        return current

    def display(self) -> None:
        """Displays all students in the list."""
        if not self.head:
            print("List is empty")
            return
        current = self.head
        while True:
            print(f"Name: {current.data.name}, ID: {current.data.ID}, GPA: {current.data.GPA}")
            current = current.next
            if current == self.head:
                break
        print("----------")

    def append(self, new_student: Student) -> None:
        """Appends a new student to the list."""
        new_node = Node(new_student)
        if not self.head:
            new_node.next = new_node
            self.head = new_node
        else:
            current = self.head
            while current.next != self.head:
                current = current.next
            current.next = new_node
            new_node.next = self.head

    def reverse(self) -> None:
        """Reverses the order of students in the list."""
        if not self.head:
            print("List is empty")
            return
        prev = None
        current = self.head
        while current.next != self.head:
            next_node = current.next
            current.next = prev
            prev = current
            current = next_node
        current.next = prev
        self.head.next = current
        self.head = current

def main():
    student_list = CircularLinkedList()
    st1 = Student("Meri", "111", "3.0")
    student_list.append(st1)
    student_list.display()
    student_list.reverse()
    student_list.display()

if __name__ == "__main__":
    main()
