import time
from sys import exit

# Function to compute the factorial of a nonnegative integer
def factorial(n):
    result = 1
    for i in range(1, n + 1):
        result *= i
    return result

# Function to convert a list of digits to an integer
def list_to_integer(lst):
    result = 0
    power = len(lst) - 1
    for digit in lst:
        if not str(digit).isdigit():  # Check if the digit is a valid number
            return None
        result += digit * 10 ** power
        power -= 1
    return result

def get_nonnegative_integer_input():
    """Prompt the user to input a nonnegative integer and validate it."""
    number = input('Input a nonnegative integer: ')
    if not number.isdigit() or number.startswith('0'):
        print('Incorrect input, giving up...')
        exit()
    return int(number)

def compute_max_factorial(number):
    """Compute the maximum factorial required for the given number."""
    Max = 1
    while factorial(Max) <= number:
        Max += 1
    return Max - 1

def convert_to_factorial_base(number, Max):
    """Convert the decimal number into factorial base representation."""
    middle_result = 0
    factorial_results = []
    while number:
        for i in range(Max, 0, -1):
            while middle_result * factorial(i) < number:
                middle_result += 1
            if middle_result * factorial(i) > number:
                middle_result -= 1
            factorial_results.append(middle_result)
            number -= middle_result * factorial(i)
            middle_result = 0
    return factorial_results

# Main function
def main():
    number = get_nonnegative_integer_input()
    Max = compute_max_factorial(number)
    factorial_results = convert_to_factorial_base(number, Max)
    print('Decimal {} reads as {} in factorial base.'.format(number, list_to_integer(factorial_results)))

if __name__ == "__main__":
    main()
