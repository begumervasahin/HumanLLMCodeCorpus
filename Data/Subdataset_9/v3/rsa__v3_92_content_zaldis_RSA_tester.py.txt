# Importing necessary classes from RSA-related modules
from rsa.crypt.crypt import Crypt
from rsa.decrypt.decrypt import Decrypt
from rsa.key_generator.key_generator import KeyGenerator

# Function to read bytes from a file
def read_file_bytes(file_path):
    with open(file_path, 'rb') as file:
        return file.read()

# Function to demonstrate encryption and decryption
def encrypt_decrypt_demo(data_bytes):
    # Generate RSA key pair
    private_key, public_key = KeyGenerator().generate()
    
    # Encrypt data using RSA public key
    encrypted_bytes = Crypt(public_key).crypt(list(data_bytes))
    
    # Decrypt data using RSA private key
    decrypted_bytes = Decrypt(private_key).decrypt(encrypted_bytes)
    
    # Display the encrypted and decrypted data
    print("Encrypted Bytes:", encrypted_bytes)
    print("Decrypted Bytes:", bytes(decrypted_bytes))

# Function to find two large prime numbers
def find_two_large_primes():
    import math
    
    # Initialize an empty list to store prime numbers
    primes = []
    
    # Iterate over a range to find prime numbers
    for num in range(10_000, 2, -1):  # Start from 10,000 and go backwards
        is_prime = True
        
        # Check if the number is prime
        for i in range(2, int(math.sqrt(num)) + 1):  # Check divisibility up to the square root of the number
            if num % i == 0:  # If divisible by any other number, it's not prime
                is_prime = False
                break
        
        # If the number is prime, add it to the list
        if is_prime:
            primes.append(num)
        
        # Exit the loop if two prime numbers are found
        if len(primes) == 2:
            break
    
    # Display the two prime numbers found
    print("Two Large Prime Numbers:", primes)

# Main execution
if __name__ == "__main__":
    # Read bytes from a file
    input_bytes = read_file_bytes("files/input.txt")
    print("Input Bytes:", input_bytes)
    
    # Demonstrate encryption and decryption
    encrypt_decrypt_demo(input_bytes)
    
    # Find two large prime numbers
    find_two_large_primes()
