import pandas as pd
from apyori import apriori

# Function to load the dataset
def load_dataset(file_path):
    """
    Load the dataset from the specified file path.

    Parameters:
    - file_path (str): Path to the dataset file.

    Returns:
    - transactions (list): List of transaction records.
    """
    dataset = pd.read_csv(file_path, header=None)
    transactions = [[str(dataset.values[i, j]) for j in range(len(dataset.columns))] for i in range(len(dataset))]
    return transactions

# Main function to run the Apriori algorithm
def main():
    # File path for the dataset
    file_path = 'Market_Basket_Optimisation.csv'

    # Parameters for Apriori algorithm
    min_support = 0.003
    min_confidence = 0.2
    min_lift = 3
    min_length = 2

    # Load dataset
    transactions = load_dataset(file_path)

    # Run Apriori algorithm on the dataset
    rules = apriori(transactions,
                    min_support=min_support,
                    min_confidence=min_confidence,
                    min_lift=min_lift,
                    min_length=min_length)

    # Print the association rules
    print_association_rules(rules)

# Function to print association rules
def print_association_rules(rules):
    """
    Print the association rules generated by the Apriori algorithm.

    Parameters:
    - rules (generator): Generator object containing association rules.
    """
    for result in rules:
        print("Association Rule:", result.items)
        print("Support:", result.support)
        print("Confidence:", result.ordered_statistics[0].confidence)
        print("Lift:", result.ordered_statistics[0].lift)
        print("===================================")

if __name__ == "__main__":
    main()
