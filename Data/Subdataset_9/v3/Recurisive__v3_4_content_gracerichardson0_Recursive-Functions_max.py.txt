def get_positive_integer_input(prompt):
    """Prompt the user to enter a positive integer until valid input is received."""
    while True:
        user_input = input(prompt)
        try:
            n = int(user_input)
            if n > 0:
                return n
            else:
                print('Please enter a positive integer.')
        except ValueError:
            print('Invalid input. Please enter a positive integer.')

# Prompt the user to enter a positive integer
nMax = get_positive_integer_input('Enter a positive integer: ')

# Initialize Fibonacci sequence variables
first_term = 1
second_term = 2
n = 1  # Counter for tracking the current term

# Generate and print Fibonacci sequence up to the specified maximum term
while n <= nMax:
    print(f'{n:4d}: {first_term}')  # Print current term
    # Calculate next term in the sequence
    next_term = first_term + second_term
    first_term = second_term
    second_term = next_term
    n += 1  # Increment counter for next term
