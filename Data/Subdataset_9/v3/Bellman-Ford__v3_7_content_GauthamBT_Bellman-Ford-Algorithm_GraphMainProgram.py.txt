    directed_graph.add_edge(3, 2, -3)

    directed_graph.add_edge(4, 2, 7)

def bellman_ford_algorithm(directed_graph):
    num_vertices = directed_graph.num_vertices()
    distances = [float('inf')] * num_vertices
    predecessors = [-1] * num_vertices
    distances[0] = 0  # Assuming source vertex is always 0

    for _ in range(num_vertices - 1):
        for u, v, weight in directed_graph.get_edges():
            if distances[u] != float('inf') and distances[u] + weight < distances[v]:
                distances[v] = distances[u] + weight
                predecessors[v] = u

    # Check for negative cycles
    for u, v, weight in directed_graph.get_edges():
        if distances[u] != float('inf') and distances[u] + weight < distances[v]:
            print("Graph contains negative cycle")
            return

    directed_graph.set_shortest_path(distances, predecessors)

    return directed_graph

def main():
    num_vertices = 5
    directed_graph = DirectedGraph(num_vertices)
    create_directed_graph(directed_graph, source_vertex=0)
    directed_graph = bellman_ford_algorithm(directed_graph)
    if directed_graph:
        directed_graph.print_shortest_path()

if __name__ == "__main__":
    main()
