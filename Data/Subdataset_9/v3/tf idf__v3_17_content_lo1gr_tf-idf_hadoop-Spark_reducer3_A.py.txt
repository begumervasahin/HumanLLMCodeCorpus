import sys
from math import log10

# Initialize variables
prev_word = None
count = 1
word = None
doc_counts = {}  # Stores document counts for each word
temp_arr = []    # Temporary array to store word-document pairs
D = 10.0         # Total number of documents

# Process input from stdin
for line in sys.stdin:
    # Parse input line
    line = line.strip()
    word, rest = line.split('\t', 1)
    filename, n, N, c = rest.split(' ', 3)
    
    # Update counts if it's the same word
    if prev_word == word:
        count += int(c)
    else:
        # Store counts for the previous word if it's a new word
        if prev_word is not None:
            doc_counts[prev_word] = f"{n} {N} {count}"
            temp_arr.append(f"{prev_word} {filename}")
        count = 1
        prev_word = word

# Store counts for the last word
if prev_word is not None:
    doc_counts[prev_word] = f"{n} {N} {count}"
    temp_arr.append(f"{prev_word} {filename}")

# Calculate TF-IDF for each word-document pair
for pair in temp_arr:
    word, filename = pair.split(' ', 1)
    n, N, m = doc_counts[word].split(' ', 2)
    n, N, m = float(n), float(N), float(m)
    tfidf = (n / N) * log10(D / m)
    print(f"{pair}\t{tfidf}")
