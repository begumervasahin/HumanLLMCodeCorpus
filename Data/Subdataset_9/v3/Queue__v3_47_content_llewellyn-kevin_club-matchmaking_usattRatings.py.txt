# Define the lookup table
lookup_table = {
    'upset': {
        (0, 12): 8, (13, 37): 10, (38, 62): 13, (63, 87): 16, (88, 112): 20,
        (113, 137): 25, (138, 162): 30, (163, 187): 35, (188, 212): 40,
        (213, 237): 45, (238, 9999): 50
    },
    'expected_result': {
        (0, 12): 8, (13, 37): 7, (38, 62): 6, (63, 87): 5, (88, 112): 4,
        (113, 137): 3, (138, 162): 2, (163, 187): 2, (188, 212): 1,
        (213, 237): 1, (238, 9999): 0
    }
}

def get_adjustment(winner_rating, loser_rating):
    """
    Get the adjustment score based on the rating difference between winner and loser.

    Args:
        winner_rating (int): The rating of the winner.
        loser_rating (int): The rating of the loser.

    Returns:
        int: The adjustment score.
    """
    # Determine the key to use in the lookup table based on the rating difference
    key = 'upset' if winner_rating < loser_rating else 'expected_result'
    score_differential = abs(winner_rating - loser_rating)

    # Find the adjustment score based on the rating difference
    for score_range, adjustment in lookup_table[key].items():
        if score_range[0] <= score_differential <= score_range[1]:
            return adjustment

# Test the function
winner_rating = 100
loser_rating = 110
adjustment = get_adjustment(winner_rating, loser_rating)
print(f"Adjustment for winner rating {winner_rating} and loser rating {loser_rating}:", adjustment)
