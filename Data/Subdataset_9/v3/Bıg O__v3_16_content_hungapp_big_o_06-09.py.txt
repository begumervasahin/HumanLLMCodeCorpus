import queue

INF = int(1e9)

class Node:
    def __init__(self, city, dist):
        self.city = city
        self.dist = dist

    def __lt__(self, other):
        return self.dist <= other.dist

def dijkstra(source, n, graph):
    pq = queue.PriorityQueue()
    pq.put(Node(source, 0))
    distance = [INF] * (n + 1)
    distance[source] = 0
    while not pq.empty():
        top = pq.get()
        u = top.city
        w = top.dist
        for neighbor in graph[u]:
            if w + neighbor.dist < distance[neighbor.city]:
                distance[neighbor.city] = w + neighbor.dist
                pq.put(Node(neighbor.city, distance[neighbor.city]))
    return distance

def solve_problem_two():
    def read_input():
        n = int(input())
        graph = [[] for _ in range(n + 1)]
        cities = []
        for i in range(n):
            city = input()
            cities.append(city)
            p = int(input())
            for j in range(p):
                nr, c = map(int, input().split())
                graph[i + 1].append(Node(nr, c))
        return n, graph, cities

    tc = int(input())
    for _ in range(tc):
        n, graph, cities = read_input()
        r = int(input())
        for _ in range(r):
            source, destination = input().split()
            start = cities.index(source) + 1
            end = cities.index(destination) + 1
            cost = dijkstra(start, n, graph)
            print(cost[end])
        input()

def solve_problem_three():
    n, m, k, x = map(int, input().split())
    ks = list(map(int, input().split()))
    graph = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v, d = map(int, input().split())
        graph[u].append(Node(v, d))
        graph[v].append(Node(u, d))
    a, b = map(int, input().split())
    distA = dijkstra(a, n, graph)
    distB = dijkstra(b, n, graph)
    min_time = INF
    for i in range(k):
        if distB[ks[i]] <= x:
            min_time = min(min_time, distA[ks[i]] + distB[ks[i]])
    if min_time < INF:
        print(min_time)
    else:
        print(-1)

def solve_problem_four():
    def bfs(s, n, graph):
        pq = queue.Queue()
        pq.put(s)
        distance = [INF] * n
        distance[s] = 0
        while not pq.empty():
            u = pq.get()
            for neighbor in graph[u]:
                if distance[neighbor] == INF:
                    distance[neighbor] = distance[u] + 1
                    pq.put(neighbor)
        return distance

    t = int(input())
    for c in range(t):
        n = int(input())
        graph = [[] for _ in range(n)]
        r = int(input())
        for _ in range(r):
            u, v = map(int, input().split())
            graph[u].append(v)
            graph[v].append(u)
        s, d = map(int, input().split())
        distS = bfs(s, n, graph)
        distD = bfs(d, n, graph)
        max_time = max(distS[i] + distD[i] for i in range(n) if distS[i] != INF and distD[i] != INF)
        print('Case {}: {}'.format(c + 1, max_time))
