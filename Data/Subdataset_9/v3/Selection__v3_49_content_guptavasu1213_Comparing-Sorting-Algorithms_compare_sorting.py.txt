import random
import time

class SelectionSort:
    def __init__(self, array):
        self.array = array
        self.array_accesses = 0
        self.extra_memory = 0
        self.recursive_calls = 0
        self.selection_sort()

    def selection_sort(self):
        for i in range(len(self.array)-1, 0, -1):
            max_position = 0
            for location in range(1, i+1):
                self.array_accesses += 2
                if self.array[location] > self.array[max_position]:
                    max_position = location
            self.array[max_position], self.array[i] = self.array[i], self.array[max_position]
            self.array_accesses += 4

class QuickSort:
    def __init__(self, array, start, end):
        self.array = array
        self.start = start
        self.end = end
        self.extra_memory = 0
        self.array_accesses = 0
        self.recursive_calls = -1
        self.quick_sort(self.array, self.start, self.end)

    def quick_sort(self, array, start, end):
        self.recursive_calls += 1
        if start < end:
            pivot = self.partition(array, start, end)
            self.quick_sort(array, start, pivot-1)
            self.quick_sort(array, pivot+1, end)

    def partition(self, array, first, last):
        big = first + 1
        small = last
        pivot = array[first]
        self.array_accesses += 1
        while big <= small:
            while big <= last and array[big] <= pivot:
                self.array_accesses += 1
                big += 1
            while array[small] > pivot:
                self.array_accesses += 1
                small -= 1
            if big < small:
                array[big], array[small] = array[small], array[big]
                self.array_accesses += 4
        array[first], array[small] = array[small], array[first]
        self.array_accesses += 4
        return small

class MergeSort:
    def __init__(self, array):
        self.array = array
        self.extra_memory = 0
        self.array_accesses = 0
        self.recursive_calls = -1
        self.array = self.merge_sort(self.array)

    def merge_sort(self, array):
        self.recursive_calls += 1
        n = len(array)
        if n <= 1:
            return array
        left_array = array[:n//2]
        right_array = array[n//2:]
        list1 = self.merge_sort(left_array)
        list2 = self.merge_sort(right_array)
        return self.merge(list1, list2)

    def merge(self, l1, l2):
        final_array = []
        i = j = 0
        while i < len(l1) and j < len(l2):
            if l1[i] < l2[j]:
                final_array.append(l1[i])
                i += 1
            else:
                final_array.append(l2[j])
                j += 1
        final_array.extend(l1[i:])
        final_array.extend(l2[j:])
        return final_array

class HeapSort:
    def __init__(self, array):
        self.array = array
        self.extra_memory = 0
        self.recursive_calls = 0
        self.array_accesses = 0
        self.size = len(self.array)
        i = (self.size-2)//2
        while i >= 0:
            self.max_heap(i)
            i -= 1
        self.heap_sort()

    def max_heap(self, index):
        largest = index
        left = 2*index+1
        right = 2*index+2
        if left < self.size and self.array[left] > self.array[largest]:
            largest = left
        if right < self.size and self.array[right] > self.array[largest]:
            largest = right
        if not largest == index:
            self.array[largest], self.array[index] = self.array[index], self.array[largest]
            self.recursive_calls += 1
            self.max_heap(largest)

    def heap_sort(self):
        while self.size > 1:
            self.array[0], self.array[self.size-1] = self.array[self.size-1], self.array[0]
            self.size -= 1
            self.max_heap(0)

def create_array(given_array_size):
    return random.sample(range(1, given_array_size+1), given_array_size)

        except ValueError:
            print("\nInvalid input. Try again!\n")
            continue

        unsorted_array = create_array(array_size)

        print("Array Before Sorting:")
        print(unsorted_array)

        starting_time = time.time()

        if user_input == 1:
            print("*** Merge Sort ***")
            sorted_array = MergeSort(unsorted_array).array
        elif user_input == 2:
            print("*** Heap Sort ***")
            sorted_array = HeapSort(unsorted_array).array
        elif user_input == 3:
            print("*** Quick Sort ***")
            sorted_array = QuickSort(unsorted_array, 0, len(unsorted_array)-1).array
        elif user_input == 4:
            print("*** Selection Sort ***")
            sorted_array = SelectionSort(unsorted_array).array
        elif user_input == 5:
            print("\nExiting program...\n")
            break

        elapsed_time = time.time() - starting_time

        print("Array After Sorting:")

if __name__ == "__main__":
