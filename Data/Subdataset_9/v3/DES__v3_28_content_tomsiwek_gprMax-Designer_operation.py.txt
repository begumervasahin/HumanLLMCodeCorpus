class TOperation:
    """
    Represents a single operation performed in the program.

    Attributes:
        o_type (str): The type of operation.
        shape (object): The shape on which the operation was performed.        
        num (int): The index of the shape in the shape list.
    """

    def __init__(self, operation_type=None, shape=None, index=None):
        """
        Initializes a TOperation object.

        Args:
            operation_type (str): The type of operation.
            shape (object): The shape on which the operation was performed.    
            index (int): The index of the shape in the shape list.
        """
        self.operation_type = operation_type
        self.shape = shape
        self.index = index

# Example usage:
# Create a TOperation object
operation = TOperation(operation_type="resize", shape="rectangle", index=1)

# Print the attributes of the TOperation object
print("Operation Type:", operation.operation_type)
print("Shape:", operation.shape)
print("Index:", operation.index)
