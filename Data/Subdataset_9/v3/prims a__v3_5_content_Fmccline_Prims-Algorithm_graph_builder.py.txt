import json
import networkx as nx

class GraphBuilder:
    """ Helper class for working with networkx graphs
    """

    @staticmethod
    def build_graph_from_file(filename):
        """Creates a graph from a given JSON file.
        
        Args:
            filename (str): The JSON source file of the graph.
        
        Returns:
            networkx.Graph: A graph constructed from the given file.
        """

        # Create an empty graph
        graph = nx.Graph()

        # Open the JSON file and load data
        with open(filename, 'r') as file:
            raw_data = json.load(file)

        # Extract node information from the JSON data
        nodes = raw_data["nodes"]

        # Iterate over each node in the JSON data
        for node_data in nodes:
            # Extract node attributes
            name = node_data["name"]
            neighbors = node_data["neighbors"]
            weights = node_data["weights"]

            # Add the current node to the graph
            graph.add_node(name)

            # Iterate over each neighbor and its corresponding weight 
            for neighbor, weight in zip(neighbors, weights):
                # Add the edge to the graph
                graph.add_edge(name, neighbor, weight=weight)

        # Return the constructed graph
        return graph

# Example usage:
if __name__ == "__main__":
    filename = "graph_data.json"  # Replace with your JSON file path  
    graph = GraphBuilder.build_graph_from_file(filename)
    print("Nodes:", graph.nodes)
    print("Edges:", graph.edges)
