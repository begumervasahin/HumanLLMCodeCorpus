import collections

class SimpleGraph:
    def __init__(self):
        self.edges = {}

    def get_neighbors(self, node):
        """Return the neighbors of a given node."""
        return self.edges.get(node, [])

    def print_graph(self):
        """Print the graph's nodes and their corresponding neighbors."""
        for node, neighbors in self.edges.items():
            print(f"{node}: {neighbors}")

class Queue:
    def __init__(self):
        self.elements = collections.deque()

    def is_empty(self):
        """Check if the queue is empty."""
        return len(self.elements) == 0

    def enqueue(self, item):
        """Add an item to the end of the queue."""
        self.elements.append(item)

    def dequeue(self):
        """Remove and return the item at the front of the queue."""
        return self.elements.popleft()

def breadth_first_search(graph, start_node):
    """Perform breadth-first search starting from a given node."""
    frontier = Queue()
    frontier.enqueue(start_node)
    visited = {start_node: True}

    while not frontier.is_empty():
        current_node = frontier.dequeue()
        print("Visiting node:", current_node)
        for neighbor_node in graph.get_neighbors(current_node):
            if neighbor_node not in visited:
                frontier.enqueue(neighbor_node)
                visited[neighbor_node] = True

        'D': ['E', 'A'],

    # Perform breadth-first search starting from node 'A'
    breadth_first_search(example_graph, 'A')
