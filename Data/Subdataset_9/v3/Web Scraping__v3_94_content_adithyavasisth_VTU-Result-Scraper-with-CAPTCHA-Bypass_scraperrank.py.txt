import re
import sys
import warnings
import cv2
import pytesseract
from PIL import Image
from bs4 import BeautifulSoup
from selenium import webdriver

# Ignore warnings
if not sys.warnoptions:
    warnings.simplefilter("ignore")

# Set Tesseract path
pytesseract.pytesseract.tesseract_cmd = r'C:\Program Files (x86)\Tesseract-OCR\tesseract'

def generate_usn(college_code, year, branch, usn_number):
    usn_number_str = str(usn_number).zfill(3)
    return f"{college_code}{year}{branch}{usn_number_str}"

def scrape_captcha():
    driver = webdriver.Chrome('C:\Program Files (x86)\chromedriver_win32\chromedriver.exe')
    driver.get('http://results.vtu.ac.in/resultsvitavicbcs_19/index.php')
    driver.save_screenshot('python_org.png')
    img = cv2.imread("python_org.png")
    crop_img = img[467:508, 667:885]
    cv2.imwrite('cap.png', crop_img)
    captcha_text = pytesseract.image_to_string(Image.open('cap.png')).strip().replace(',', '').strip()
    driver.quit()
    return int(captcha_text)

def scrape_results(usn):
    driver = webdriver.Chrome('C:\Program Files (x86)\chromedriver_win32\chromedriver.exe')
    driver.get('http://results.vtu.ac.in/resultsvitavicbcs_19/index.php')
    driver.find_element_by_name("lns").send_keys(usn)
    driver.find_element_by_name("captchacode").send_keys(scrape_captcha())
    driver.find_element_by_id("submit").click()
    try:
        soup = BeautifulSoup(driver.page_source, 'html.parser')
    except:
        alert = driver.switch_to.alert
        alert.dismiss()
        driver.quit()
        return None
    finally:
        driver.quit()
    return soup

def parse_results(soup, subject_code, loop_count):
    ths = soup.find_all('th')
    divCell = soup.find_all('div', attrs={'class': 'divTableCell'})
    headers = ','.join(re.sub('[!@#$]', '', str(ths[i].text)) for i in range(2))
    sort_list = [divCell[i].text[-3:] if divCell[i].text[-3:].isdigit() else divCell[i].text[-2:] for i in range(6, subject_code, 6)]
    sort_list.sort()
    index_list = []
    for i in range(loop_count):
        for j in range(6, subject_code, 6):
            last_three = divCell[j].text[-3:]
            if last_three.isdigit():
                if last_three == sort_list[i] and j not in index_list:
                    index_list.append(j)
            elif divCell[j].text[-2:] == sort_list[i] and j not in index_list:
                index_list.append(j)
    records = []
    for l in index_list:
        record = ','.join(divCell[j].text if j != l + 1 else '' for j in range(l, l + 6))
        records.append(record)
    return headers, records

# Input data
college_codes = ["4AI", "1BG", "1CR", "1AM", "1BI"]
year = input('Enter the year: ')
branch = input('Please enter the branch: ').upper()
starting_usn = int(input('Enter starting USN: '))
ending_usn = int(input('Enter last USN: ')) + 1
semester = input('Enter the Semester: ')
cycle = 'N'
dip = 'Y' if starting_usn >= 400 else 'N'
subject_code = 52
loop_count = 8

# Adjust subject code and loop count based on semester and cycle
if semester in ['1', '2']:
    cycle = input('Enter the Cycle: ').upper()
    if cycle == 'P':
        loop_count = 7
        subject_code = 46
elif semester in ['3', '4'] and dip == 'Y':
    loop_count = 9
    subject_code = 58

# Writing data to file
with open('test2.txt', 'w+') as output_file:
    for college_code in college_codes:
        for usn_number in range(starting_usn, ending_usn):
            usn = generate_usn(college_code, year, branch, usn_number)
            soup = scrape_results(usn)
            if soup:
                headers, records = parse_results(soup, subject_code, loop_count)
                output_file.write(f"{headers}\n")
                for record in records:
                    output_file.write(f"{record}\n")

# Calling gpa2 if dip is not 'Y'
if dip != 'Y':
    from sgparank import gpa2
    gpa2(year, branch, starting_usn, ending_usn, semester, cycle)
