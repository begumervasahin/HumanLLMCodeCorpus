import numpy as np

# Function to display image matrix
def display_image(matrix):
    image = np.reshape(matrix, (16, 16))
    plt.imshow(image)

# Convert image to matrix
def image_to_matrix(filename):
    image = Image.open(filename)
    grayscale_image = image.convert("L")
    pixel_data = np.array(grayscale_image, dtype='float') / 255.0
    return pixel_data

# Load images and convert to matrices
def load_images(num_images):
    image_matrices = []
    for i in range(1, num_images + 1):
        filename = 'G:\\Yeah!!!\\img2vec/img/{}.jpg'.format(i)
        image_matrix = image_to_matrix(filename)
        image_matrices.append(image_matrix)
    return np.array(image_matrices)

# Perform PCA with 10 components
def apply_pca(data, n_components=10):
    pca_model = PCA(n_components=n_components)
    transformed_data = pca_model.fit_transform(data)
    return transformed_data

# Main function
def main():
    # Load images
    image_matrices = load_images(8984)

    # Flatten image matrices
    flattened_matrices = image_matrices.reshape(len(image_matrices), -1)

    # Apply PCA
    pca_transformed_data = apply_pca(flattened_matrices)

    # Set precision for printing
    np.set_printoptions(precision=9)

    # Print the PCA transformation of a specific image (index 178)
    print("PCA transformation of image at index 178:")
    print(pca_transformed_data[178])

if __name__ == "__main__":
    main()
