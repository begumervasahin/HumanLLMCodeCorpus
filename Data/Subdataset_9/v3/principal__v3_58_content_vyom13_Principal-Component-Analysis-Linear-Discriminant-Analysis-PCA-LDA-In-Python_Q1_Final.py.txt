# Principal Component Analysis (PCA)
def perform_pca(x):
    column_mean = np.mean(x, axis=0)
    x_mean_centered = x - column_mean
    covariance_matrix = np.cov(x_mean_centered, rowvar=False)
    eigen_values, eigen_vectors = np.linalg.eig(covariance_matrix)
    sorted_indices = np.argsort(eigen_values)[::-1]
    eigen_values = eigen_values[sorted_indices]
    eigen_vectors = eigen_vectors[:, sorted_indices]
    pca_scores = np.dot(x_mean_centered, eigen_vectors)
    return {'PC_variance': eigen_values, 'loadings': eigen_vectors, 'scores': pca_scores}

pca_results = perform_pca(data)

# Plot regression using PCA loadings
plt.title('Regression Plot Using PCA Loadings')
plt.scatter(data_V1[:30], data_V2[:30], marker='o', color='purple', alpha=0.5, label='Class 1')
plt.scatter(data_V1[30:60], data_V2[30:60], marker='o', color='orange', alpha=0.5, label='Class 2')
plt.plot([0, 100 * pca_results['loadings'][0, 0]], [0, 100 * pca_results['loadings'][1, 0]],
         color='blue', linewidth=3, label='Principal Component 1')
plt.xlabel('V1')
plt.ylabel('V2')
plt.xlim(0, 40)

plt.legend()

# Linear Discriminant Analysis (LDA)
df = pd.read_csv("C:/Users/vyoms/Desktop/dataset_1.csv", header=None).iloc[1:, :].astype(float)
df1, df2 = df.iloc[:30], df.iloc[30:]
mean_all, mean_1, mean_2 = df.mean(), df1.mean(), df2.mean()
within_class_scatter = ((df1 - mean_1).T @ (df1 - mean_1)) + ((df2 - mean_2).T @ (df2 - mean_2))
between_class_scatter = (len(df1) * (mean_1 - mean_all).reshape(-1, 1) @ (mean_1 - mean_all).reshape(1, -1)) + \
                        (len(df2) * (mean_2 - mean_all).reshape(-1, 1) @ (mean_2 - mean_all).reshape(1, -1))
eig_vals, eig_vecs = np.linalg.eig(np.linalg.inv(within_class_scatter) @ between_class_scatter)
W = eig_vecs[:, np.argmax(eig_vals)]
lda_project = df @ W

# Plot LDA projection
plt.figure()
plt.title('LDA Projection')
plt.scatter(lda_project[:30], np.zeros(30), marker='o', color='purple', alpha=0.5, label='Class 1')
plt.scatter(lda_project[30:], np.zeros(30), marker='o', color='orange', alpha=0.5, label='Class 2')
plt.xlabel('LD1')
plt.ylabel('')
plt.legend()
plt.show()

# Calculate PCA variance
pca_variance = np.var(pca_results['scores'], axis=0)
total_pca_variance = np.sum(pca_variance)
percent_variance_explained = (pca_variance / total_pca_variance) * 100

# Print PCA variance explained by each component
for i, pc_variance in enumerate(percent_variance_explained, start=1):
    print(f"PC{i} explains: {pc_variance:.2f}% variance")

# Calculate covariance between PCA scores
covariance_pc1_pc2 = np.cov(pca_results['scores'][:, 0], pca_results['scores'][:, 1])
PC1_variance, PC2_variance = np.var(pca_results['scores'][:, 0]), np.var(pca_results['scores'][:, 1])

print("Covariance between PC1 and PC2:", covariance_pc1_pc2)
print("Variance of PC1:", PC1_variance)
print("Variance of PC2:", PC2_variance)
