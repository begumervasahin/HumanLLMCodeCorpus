# Function to check if a number is prime
def is_prime(n):
    assert type(n) == int, "Input must be an integer"
    if n < 2:
        return False
    # Check divisibility up to the square root of n
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

# Function to find all divisors of a number
def divisor(n):
    return [i for i in range(1, n + 1) if n % i == 0]

# Function to find prime factors of a number
def prime_factor(x):
    factors = []
    if is_prime(x):
        factors.append(x)
    # Find prime factors using divisor function
    for i in divisor(x):
        if i == 1:
            continue
        if is_prime(i):
            factors.append(i)
    return factors

# Function to find the greatest common divisor (GCD) of two numbers
def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

# Function to check if two numbers are relatively prime (have no common factors other than 1)
def relatively_prime(a, b):
    return gcd(a, b) == 1

# Function to find common factors of two numbers
def common_factor(a, b):
    return [elem for elem in divisor(a) if elem in divisor(b)]

# Testing the functions
print("Is 7 prime?", is_prime(7))  # True
print("Divisors of 12:", divisor(12))  # [1, 2, 3, 4, 6, 12]
print("Prime factors of 24:", prime_factor(24))  # [2, 3]
print("GCD of 24 and 36:", gcd(24, 36))  # 12
print("Are 24 and 35 relatively prime?", relatively_prime(24, 35))  # True
print("Common factors of 24 and 36:", common_factor(24, 36))  # [1, 2, 3, 4, 6, 12]
