def longest_common_subsequence(str1, str2):
    """
    Find the longest common subsequence between two strings.
    """

    # Convert strings to lists of characters
    str1_chars = list(str1)
    str2_chars = list(str2)

    # Initialize variables to store the current and longest subsequences
    current_subsequence = []
    longest_subsequence = []

    # Iterate through possible starting positions in the shorter string
    for start_index in range(len(str1_chars) + 1):
        # Reset current subsequence for each starting position
        current_subsequence = []

        # Iterate through characters in the shorter string starting from the current position
        for char in str1_chars[start_index:]:
            # If the character is found in the longer string after the current index
            if char in str2_chars:
                # Add the character to the current subsequence
                current_subsequence.append(char)
                # Update the index to search from next
                index = str2_chars.index(char) + 1
                # Update str2_chars to search from the next index
                str2_chars = str2_chars[index:]

        # If the current subsequence is longer than the longest one found so far, update it
        if len(current_subsequence) > len(longest_subsequence):
            longest_subsequence = current_subsequence

        # Reset str2_chars to its original value for the next iteration
        str2_chars = list(str2)

    # Convert the longest subsequence list back to a string
    result = ''.join(longest_subsequence)
    return result


# Test cases
print(longest_common_subsequence('ABCABA', 'ABBA'))  # Output: 'ABBA'
print(longest_common_subsequence('aaaaa', 'aa'))  # Output: 'aa'
print(longest_common_subsequence('AGGTAB', 'GXTXAYB'))  # Output: 'GTAB'
print(longest_common_subsequence('ABAZDC', 'BACBAD'))  # Output: 'ABAD'
