import urllib.request
from bs4 import BeautifulSoup
import ssl

def get_user_input():
    """Prompt the user for URL, count, and position."""
    url = input('Enter URL: ')
    count = int(input('Enter count: '))
    position = int(input('Enter position: '))
    return url, count, position

def ignore_ssl_errors():
    """Ignore SSL certificate errors."""
    ssl_context = ssl.create_default_context()
    ssl_context.check_hostname = False
    ssl_context.verify_mode = ssl.CERT_NONE
    return ssl_context

def retrieve_url_at_position(url, position, ssl_context):
    """Retrieve the URL at the specified position."""
    html = urllib.request.urlopen(url, context=ssl_context).read()
    soup = BeautifulSoup(html, 'html.parser')
    anchor_tags = soup.find_all('a')
    next_url = anchor_tags[position - 1].get('href', None)
    return next_url

def main():
    url, count, position = get_user_input()
    ssl_context = ignore_ssl_errors()

    for _ in range(count):
        print('Retrieving:', url)
        url = retrieve_url_at_position(url, position, ssl_context)

if __name__ == '__main__':
    main()
