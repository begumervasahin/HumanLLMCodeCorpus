def prim(graph, root):
    assert isinstance(graph, dict)

    unvisited_nodes = list(graph.keys())
    unvisited_nodes.remove(root)
    
    visited_nodes = [root]
    path = []
    next_node = None

    while unvisited_nodes:
        shortest_distance = float('inf')

        for source_node in visited_nodes:
            for destination_node, distance in graph[source_node].items():
                if destination_node in visited_nodes or source_node == destination_node:
                    continue

                if distance < shortest_distance:
                    shortest_distance = distance
                    previous_node = source_node
                    next_node = destination_node

        path.append((previous_node, next_node))
        visited_nodes.append(next_node)
        unvisited_nodes.remove(next_node)

    return path
