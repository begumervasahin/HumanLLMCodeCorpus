import shelve

class QuestionNode:
    def __init__(self, data):
        self.data = data
        self.left = None
        self.right = None

class QuestionTree:
    def __init__(self):
        self.root = QuestionNode("hippo")

    def play_game(self):
        print("Welcome to 20 Questions.")
        self.root = self._play_game_helper(self.root)

    def _play_game_helper(self, node):
        if node.left is None and node.right is None:
            return self._play_endgame(node)
        else:
            print(node.data)
            response = input("Is it what you were thinking of? (y/n) ")
            if response == "y":
                node = self._play_game_helper(node.right)
            elif response == "n":
                node = self._play_game_helper(node.left)
            else:
                node = self._play_game_helper(node)
            return node

    def _play_endgame(self, node):
        print("You're thinking of:", node.data)
        response = input("Is it what you were thinking of? (y/n) ")
        if response == "y":
            print("I win!!")
            return node
        else:
            return self._add_answer(node)

    def _add_answer(self, old_node):
        your_thing = input("Enter the thing you were thinking of: ")
        question = input("Enter your disambiguating question: ")
        yes_or_no = input("Is the answer 'yes' or 'no' for your thing? (y/n) ")
        your_thing = QuestionNode(your_thing)
        new_question = QuestionNode(question)
        if yes_or_no == "y":
            new_question.right = your_thing
            new_question.left = old_node
        elif yes_or_no == "n":
            new_question.right = old_node
            new_question.left = your_thing
        else:
            print("Invalid input. Please enter 'y' or 'n'.")
            return self._add_answer(old_node)
        return new_question

def main():
    savefile = shelve.open("twenty_questions.save")
    twenty_questions = savefile.get("tree", QuestionTree())
    try:
        twenty_questions.play_game()
        keep_playing = True
        while keep_playing:
            savefile["tree"] = twenty_questions
            response = input("Do you want to play again? (y/n) ")
            if response == "y":
                twenty_questions.play_game()
            elif response == "n":
                print("Thanks for playing! Goodbye.")
                keep_playing = False
            else:
                print("I'm sorry, I didn't understand you. Please enter 'y' or 'n'.")
    finally:
        savefile.close()

if __name__ == "__main__":
    main()
