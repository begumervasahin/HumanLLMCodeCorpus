from turtle import *

def generate_fibonacci_sequence(limit):
    """Generate Fibonacci sequence up to the specified limit."""
    fibonacci = [0, 1]
    while fibonacci[-1] + fibonacci[-2] <= limit:
        fibonacci.append(fibonacci[-1] + fibonacci[-2])
    return fibonacci

def get_valid_limit(prompt):
    """Get a valid integer input for the Fibonacci sequence limit."""
    while True:
        try:
            limit = int(input(prompt))
            if limit < 5:
                print("Please enter a number greater than or equal to 5.")
            else:
                return limit
        except ValueError:
            print("Invalid input. Please enter an integer value.")

def draw_fibonacci_turtle(fibonacci_sequence):
    """Draw Fibonacci sequence using Turtle graphics."""
    penup()
    goto(135.00, -20.00)
    pendown()
    first_one = True

    for fib in fibonacci_sequence:
        if fib == 0:
            pass
        elif fib == 1:
            if first_one:
                forward(25 * fib / 2)
                write(fib)
                forward(25 * fib / 2)
                left(90)
                forward(25 * fib)
                left(90)
                forward(25 * fib)
                left(90)
                forward(25 * fib)
                left(90)
                first_one = False
            else:
                forward(25 * fib)
                forward(25 * fib / 2)
                write(fib)
                forward(25 * fib / 2)
                left(90)
                forward(25 * fib)
                left(90)
                forward(25 * fib)
                left(90)
                forward(25 * fib)
                left(90)
                forward(25 * fib)
                left(90)
                forward(25 * fib)
        elif fib == 21:
            forward(25 * fib)
            left(90)
            forward(25 * fib)
            left(90)
            forward(25 * fib / 2)
            write(fib)
            forward(25 * fib / 2)
            left(90)
            break
        else:
            if fib == 2 or fib == 13:
                forward(25 * fib)
                left(90)
                forward(25 * fib)
                left(90)
                forward(25 * fib)
                left(90)
                forward(25 * fib / 2)
                write(fib)
                forward(25 * fib / 2)
                left(90)
            elif fib == 3:
                forward(25 * fib)
                left(90)
                forward(25 * fib)
                left(90)
                forward(25 * fib / 2)
                write(fib)
                forward(25 * fib / 2)
                left(90)
                forward(25 * fib)
                left(90)
            elif fib == 5:
                forward(25 * fib)
                left(90)
                forward(25 * fib / 2)
                write(fib)
                forward(25 * fib / 2)
                left(90)
                forward(25 * fib)
                left(90)
                forward(25 * fib)
                left(90)
            else:
                forward(25 * fib / 2)
                write(fib)
                forward(25 * fib / 2)
                left(90)
                forward(25 * fib)
                left(90)
                forward(25 * fib)
                left(90)
                forward(25 * fib)
                left(90)
            forward(25 * fib)
            left(90)
            forward(25 * fib)

def main():
    amount = get_valid_limit('Choose up to which number of the Fibonacci sequence you want to go up to: ')
    fibonacci_sequence = generate_fibonacci_sequence(amount)

    # Display Fibonacci sequence
    index = 0
    for number in fibonacci_sequence:
        print(f'F° {index}: {number}')
        index += 1

    # Draw Fibonacci sequence using Turtle graphics
    draw_fibonacci_turtle(fibonacci_sequence)

    done()

if __name__ == "__main__":
    main()
