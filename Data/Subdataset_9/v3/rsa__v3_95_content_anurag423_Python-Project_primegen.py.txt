import numpy as np

def primegen():
    # Ask the user for the number of bits
    bits = int(input('How many bits? '))

    # Calculate the maximum number based on the number of bits
    max_number = 2 ** bits

    # Generate a list of integers from 2 to the calculated maximum
    candidate_primes = np.arange(2, max_number + 1)

    # Remove non-prime numbers from the list
    primes = sieve_of_eratosthenes(candidate_primes)

    # Select two random prime numbers from the list of remaining integers
    p, q = np.random.choice(primes, 2)

    # Print the hidden P and Q values
    print(f'(Hidden) P Value: {p}')
    print(f'(Hidden) Q Value: {q}')

    # Return the selected prime numbers
    return p, q

def sieve_of_eratosthenes(candidate_primes):
    primes = []
    is_prime = np.ones_like(candidate_primes, dtype=bool)

    for num in candidate_primes:
        if is_prime[num - 2]:
            primes.append(num)
            is_prime[num * num - 2::num] = False

    return primes

if __name__ == "__main__":
    # Call the primegen function if the script is executed directly
    primegen()
