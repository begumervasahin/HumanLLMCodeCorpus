def depth_first_search(graph, start, goal):
    """
    Perform Depth-First Search (DFS) on the given graph to find a path from start to goal.

    Parameters:
    graph (dict): The graph in the form of adjacency list.
    start (str): The starting node.
    goal (str): The goal node.

    Returns:
    list or None: A list representing the path from start to goal if found, else None.
    """
    stack = [[start]]  # Initialize stack with start state
    visited = set()    # Keep track of visited states

    while stack:
        path = stack.pop()  # Retrieve the last path in the stack

        state = path[-1]    # Get the current state

        if state == goal:
            return path     # If goal state reached, return the path

        elif state not in visited:
            for neighbor in graph.get(state, []):
                new_path = path + [neighbor]   # Create a new path
                stack.append(new_path)         # Add new path to stack

            visited.add(state)  # Mark current state as visited

    print("Goal state not found.")
    return None
