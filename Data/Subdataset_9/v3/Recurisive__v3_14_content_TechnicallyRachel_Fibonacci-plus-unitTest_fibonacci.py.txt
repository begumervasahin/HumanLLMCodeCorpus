def generate_fibonacci_sequence(limit):
    """Generate the Fibonacci sequence up to a given limit."""
    result = []
    a, b = 0, 1
    while b < limit:
        result.append(b)
        a, b = b, a + b
    return result

def main():
    """Demonstrate the generation of the Fibonacci sequence."""
    print(generate_fibonacci_sequence(10))

if __name__ == '__main__':
    main()
