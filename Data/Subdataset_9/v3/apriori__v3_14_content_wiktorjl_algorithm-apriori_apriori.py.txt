from collections import Counter
import argparse
import itertools


def apriori(source_file, minsup):
    transactions = read_transactions(source_file)
    minsup = float(minsup)
    n = len(transactions)
    frequent_items = find_frequent_items(transactions, minsup)
    candidates = generate_candidates(transactions)


def read_transactions(source_file):
    with open(source_file) as file:
        return [line.strip().split() for line in file.readlines()]


def find_frequent_items(transactions, minsup):
    flattened_transactions = list(itertools.chain.from_iterable(transactions))
    item_counts = Counter(flattened_transactions)
    total_transactions = len(transactions)
    frequent_items = [item for item, count in item_counts.items() if count / total_transactions > minsup]
    return sorted(frequent_items)


def generate_candidates(transactions):
    candidates = []
    transactions = sorted(ensure_list_of_lists(transactions))
    prefixes = sorted([list(x) for x in set(frozenset(i) for i in (set(i[:-1]) for i in transactions))])

    for prefix in prefixes:
        postfixes = [p[-1:] for p in transactions if p[:-1] == prefix]
        postfixes = list(itertools.chain.from_iterable(postfixes))
        comb = list(itertools.combinations(postfixes, 2))

        for c in comb:
            candidates.append(prefix + list(c))

    return candidates


def restricted_float(infloat):
    infloat = float(infloat)
    if infloat < 0.0 or infloat > 1.0:
        raise argparse.ArgumentTypeError("%r not in range [0.0, 1.0]" % (infloat,))
    return infloat


def ensure_list_of_lists(F):
    if not isinstance(F[0], list):
        return [[x] for x in F]
    else:
        return F


def main():
    parser = argparse.ArgumentParser(description='Apriori algorithm')
    parser.add_argument("source", help="Source transaction file location")
    parser.add_argument("minsup", help="Minimum support", type=restricted_float)
    args = parser.parse_args()
    apriori(args.source, args.minsup)


if __name__ == "__main__":
    main()
