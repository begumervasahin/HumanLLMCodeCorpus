import tkinter as tk
import tkinter.simpledialog as tkSimpleDialog

def binary_search(nums, target):
    """Perform binary search on a sorted list."""
    left, right = 0, len(nums) - 1
    while left <= right:
        mid = (left + right) // 2
        if nums[mid] == target:
            return mid
        elif nums[mid] < target:
            left = mid + 1
        else:
            right = mid - 1
    return -1

def search():
    """Search for a target number in a list of numbers."""
    num_count = int(entry_count.get())
    numbers = []
    target_number = tkSimpleDialog.askinteger("", "Enter the number to search for")
    for _ in range(num_count):
        dialog = tkSimpleDialog.askinteger("", "Enter the numbers in order")
        numbers.append(dialog)
    numbers_str = str(numbers)
    label_numbers.config(text=numbers_str)

    result_index = binary_search(numbers, target_number)
    result_text = f"The number is at index: {result_index}"
    label_result.config(text=result_text)

app = tk.Tk()
app.title("Binary Search")
app.geometry('250x200')
app.configure(bg='SkyBlue2')

label_count = tk.Label(app, text="How many numbers will you enter?", font='Helvetica 12', bg='SkyBlue2')
label_count.pack()

entry_count = tk.Entry(app, width=8)
entry_count.pack()

button_ok = tk.Button(app, text="Ok!", command=search)
button_ok.pack(pady=(10, 0))

label_numbers = tk.Label(app, text="", bg='SkyBlue2')
label_numbers.pack(pady=(10, 0))

label_result = tk.Label(app, text="", bg='SkyBlue2')
label_result.pack(pady=(10, 0))

app.mainloop()
