import sys
import re

DI_ONLY = True
ANSWERS_ONLY = False


def parse_fields(line):
    """Parse and extract fields from the input line."""
    match = re.match('^(\S+)\s+(\S+)\s+(\S+)(?:\s+(\S+))?', line)
    if match:
        return match.groups()
    else:
        print('Error: Unable to parse line:', line)
        sys.exit(1)


def build_key(fields):
    """Construct a key from the given fields."""
    return ' '.join(fields[:3])


def process_input(file_obj):
    """Process input lines and build a dictionary."""
    state = 0
    buffer = ""
    data_dict = {}
    key = ""
    fields = ()

    for line in file_obj:
        if line.strip():  # Skip empty lines
            text = line.rstrip()
            if text.endswith('\\'):
                if state != 1:
                    if key:
                        data_dict[key] = (fields, buffer)
                    key = ""
                    buffer = ""
                    fields = ()
                buffer += line
                state = 1
            else:
                if text[0].isalpha():
                    if state == 2:
                        if key:
                            data_dict[key] = (fields, buffer)
                        key = ""
                        buffer = ""
                        fields = ()
                    fields = parse_fields(text)
                    key = build_key(fields)
                    buffer += line
                    state = 2
                else:
                    if text[0] == ' ' or text[0] == '\t':
                        buffer += line
                        state = 3
                    else:
                        state = 4

    if key:
        data_dict[key] = (fields, buffer)

    return data_dict


def print_sorted_output(data_dict):
    """Print the sorted output."""
    sorted_data = {}
    for key, value in data_dict.items():
        new_key = value[0][1]
        if new_key in sorted_data:
            sorted_data[new_key].append(value)
        else:
            sorted_data[new_key] = [value]

    sorted_keys = sorted(sorted_data.keys())

    for a_key in sorted_keys:
        for item in sorted_data[a_key]:
            if DI_ONLY and item[0][0] == 'd-i':
                print(item[1])
            elif not DI_ONLY:
                print(item[1])


if __name__ == '__main__':
    file_path = sys.argv[1] if len(sys.argv) > 1 else None
    input_file = open(file_path, 'r') if file_path else sys.stdin
    input_data = process_input(input_file)
    print_sorted_output(input_data)
