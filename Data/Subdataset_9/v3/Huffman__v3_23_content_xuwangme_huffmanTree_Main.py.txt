from FileOperation import FileOperation

def main():
    """
    Main function to unzip a file using FileOperation class.
    """
    # Create an instance of the FileOperation class
    file_operation = FileOperation()

    # Specify the paths for the zip file and the file to be unzipped
    zip_file_path = "./data/Aesop_Fables_zip.txt"
    unzip_file_path = "./data/Aesop_Fables_unzipped.txt"

    # Unzip the specified file
    file_operation.unzipFile(zipFileName=zip_file_path, unzipFileName=unzip_file_path)

if __name__ == '__main__':
    main()
