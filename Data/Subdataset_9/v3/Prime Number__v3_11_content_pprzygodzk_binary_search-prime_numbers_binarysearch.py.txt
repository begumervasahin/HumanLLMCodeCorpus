def binary_search_iterative(array, value):
    """Searches for the given value in the sorted array (iterative approach)."""
    low = 0
    high = len(array) - 1

    while low <= high:
        mid = (low + high) // 2
        if array[mid] == value:
            return f"Value {value} found at index {mid} in the array."
        elif array[mid] < value:
            low = mid + 1
        else:
            high = mid - 1

    return f"Value {value} is not present in the array."

def binary_search_recursive(array, value, low, high):
    """Searches for the given value in the sorted array (recursive approach)."""
    if low > high:
        return f"Value {value} is not present in the array."

    mid = (low + high) // 2
    if array[mid] == value:
        return f"Value {value} found at index {mid} in the array."
    elif array[mid] < value:
        return binary_search_recursive(array, value, mid + 1, high)
    else:
        return binary_search_recursive(array, value, low, mid - 1)
