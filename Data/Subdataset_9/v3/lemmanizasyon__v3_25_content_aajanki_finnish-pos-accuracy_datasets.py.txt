from pathlib import Path
from preprocess_data import split_into_sentences

class Token:
    def __init__(self, text, lemma, pos, space_after=True):
        self.text = text
        self.lemma = lemma
        self.pos = pos
        self.space_after = space_after

class TestSentence:
    def __init__(self, tokens):
        self.tokens = tokens

    def text(self):
        return ''.join(t.text + (' ' if t.space_after else '') for t in self.tokens).rstrip()
    
    def lemmas(self):
        return [t.lemma for t in self.tokens]
    
    def pos(self):
        return [t.pos for t in self.tokens]
    
    def count_tokens(self):
        return len(self.tokens)

class Dataset:
    def __init__(self, name, datapath):
        self.name = name
        self.datapath = datapath
        self.sentences = self._parse_conllu()

    def _parse_conllu(self):
        with open(self.datapath, 'r', encoding='utf-8') as file:
            sentences = []
            for sentence in split_into_sentences(file.readlines()):
                tokens = []
                for line in sentence:
                    if line.startswith('#'):
                        continue
                    fields = line.strip().split('\t')
                    if len(fields) != 10:
                        raise ValueError("Expected 10 fields per line")
                    text = fields[1].replace(' ', '')
                    lemma = fields[2].replace(' ', '')
                    pos = fields[3]
                    space_after = fields[9] != 'SpaceAfter=No'
                    tokens.append(Token(text, lemma, pos, space_after))
                sentences.append(TestSentence(tokens))
            return sentences
    
    def count_tokens(self):
        return sum(sentence.count_tokens() for sentence in self.sentences)

def gold_path(testset_name):
    paths = {
        'UD_Finnish_TDT': 'data/preprocessed/UD_Finnish-TDT/fi_tdt-ud-test.conllu',
        'ftb1u': 'data/preprocessed/ftb1/ftb1u_sample.tsv',
        'ftb2-news': 'data/preprocessed/ftb2/FinnTreeBank_2/news-samples_tab.txt',
        'ftb2-sofie': 'data/preprocessed/ftb2/FinnTreeBank_2/sofie12_tab.txt',
        'ftb2-wikipedia': 'data/preprocessed/ftb2/FinnTreeBank_2/wikipedia-samples_tab.txt',
    }
    return Path(paths.get(testset_name, ''))

def initialize_datasets():
    testset_names = ['UD_Finnish_TDT', 'ftb1u', 'ftb2-news', 'ftb2-sofie', 'ftb2-wikipedia']
    return [Dataset(name, gold_path(name)) for name in testset_names]

def main():
    all_testsets = initialize_datasets()
    for dataset in all_testsets:
        print(f"Dataset: {dataset.name}")
        print(f"Number of sentences: {len(dataset.sentences)}")
        print(f"Number of tokens: {dataset.count_tokens()}\n")

if __name__ == "__main__":
    main()
