from Prims import Prims

def display_menu():
    print("\nMST Theory:")
    print("1. Prims Algorithm")
    print("2. Kruskal's Algorithm (Currently unavailable)")
    print("9. Exit")

def get_choice():
    return input("\nPlease enter your choice: ")

def handle_prims():
    file_name = input("\nEnter the filename of your graph: ")
    start_vertex = input("Choose a starting vertex: ")
    draw_graph = input("Would you like to draw the graph? (Y/N): ").strip().lower()
    
    if draw_graph == "y":
        Prims(file_name, int(start_vertex), True)
    elif draw_graph == "n":
        Prims(file_name, int(start_vertex), False)
    else:
        print("Invalid choice. Returning to the main menu...")

def main():
    while True:
        display_menu()
        choice = get_choice()
        
        if choice == "1":
            handle_prims()
        elif choice == "2":
            print("Kruskal's Algorithm is currently unavailable.")
        elif choice == "9":
            print("Thank you. Goodbye!")
            break
        else:
            print("Invalid choice. Please try again.")

if __name__ == "__main__":
    main()
