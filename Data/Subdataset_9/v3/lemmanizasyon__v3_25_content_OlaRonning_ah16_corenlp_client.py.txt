import json
from jsonrpc import ServerProxy, JsonRpc20, TransportTcpIp
from pprint import pprint

class StanfordNLP:
    def __init__(self):
        # Initialize connection to the Stanford CoreNLP server
        self.server = ServerProxy(
            jsonrpc_version=JsonRpc20(),
            transport=TransportTcpIp(addr=("127.0.0.1", 8080))
        )
    
    def parse(self, text):
        # Send the text to the CoreNLP server for parsing
        response = self.server.parse(text)
        # Parse the JSON response from the server
        parsed_result = json.loads(response)
        return parsed_result

def main():
    # Initialize the StanfordNLP object
    nlp = StanfordNLP()
    
    # Sample text to be parsed
    text_to_parse = "He couldn't fully explain why, but ventured that ``people have a summer rally in July and August and then it's just profit-taking.''"
    
    # Parse the text
    parsed_result = nlp.parse(text_to_parse)
    
    # Print the parsed result
    pprint(parsed_result)

if __name__ == "__main__":
    # Entry point of the script
    main()
