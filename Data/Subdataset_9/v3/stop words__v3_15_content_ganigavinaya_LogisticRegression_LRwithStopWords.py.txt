import os
import io
import re
import numpy as np

class LogisticRegression:
    def __init__(self, train_ham_folder, train_spam_folder, test_ham_folder, test_spam_folder):
        self.train_ham_folder = train_ham_folder
        self.train_spam_folder = train_spam_folder
        self.test_ham_folder = test_ham_folder
        self.test_spam_folder = test_spam_folder
        self.weights = {}
        self.vocabulary = []
        self.learning_rate = 0.0001
        self.regularization_parameter = 5
        self.file_data = []
        self.stop_words = self.load_stop_words()

    def load_stop_words(self):
        stop_words = set()
        with open("stop_words.txt", "r") as file:
            stop_words.update(file.read().splitlines())
        return stop_words

    def run(self):
        self.build_vocabulary()
        self.train_model()

    def count_words(self, path, word_count):
        with io.open(path, 'r', encoding='iso-8859-1') as file:
            lines = file.readlines()
            for line in lines:
                words_only = re.sub("[^a-zA-Z0-9\s]", "", line).lower().split()
                for word in words_only:
                    if word not in self.stop_words:
                        word_count[word] = word_count.get(word, 0) + 1

    def build_vocabulary(self):
        ham_word_count = self.get_word_count(self.train_ham_folder)
        spam_word_count = self.get_word_count(self.train_spam_folder)
        self.vocabulary = set(ham_word_count.keys()) | set(spam_word_count.keys())
        self.weights = {word: 0.0 for word in self.vocabulary}

    def get_word_count(self, folder):
        word_count = {}
        for filename in os.listdir(folder):
            self.count_words(os.path.join(folder, filename), word_count)
        return word_count

    def process_file(self, file, classification):
        word_count = {}
        self.count_words(file, word_count)
        self.file_data.append({'filename': file, 'tokens': word_count, 'class': classification})

    def train_model(self):
        for _ in range(500):
            self.update_error()
            self.update_weights()

    def sigmoid(self, x):
        denom = 1 + np.exp(-x)
        return 1 / denom

    def update_error(self):
        for data in self.file_data:
            tokens = data["tokens"]
            value = 1
            for token, count in tokens.items():
                value += count * self.weights[token]
            data["error"] = self.sigmoid(value)

    def update_weights(self):
        for token in self.weights.keys():
            val = 0
            for data in self.file_data:
                tokens = data["tokens"]
                true_value = data["class"]
                if token in tokens:
                    temp = true_value - data["error"]
                    val += tokens[token] * temp
            self.weights[token] += ((val * self.learning_rate) - (self.learning_rate * self.regularization_parameter * self.weights[token]))

    def test(self):
        ham_accuracy = self.calculate_accuracy(self.test_ham_folder, ">")
        spam_accuracy = self.calculate_accuracy(self.test_spam_folder, "<")
        total_accuracy = (ham_accuracy + spam_accuracy) / 2
        print("Ham accuracy is ", ham_accuracy)
        print("Spam accuracy is ", spam_accuracy)
        print("Total accuracy is ", total_accuracy)

    def calculate_accuracy(self, folder, operator):
        correct = 0
        total_files = len(os.listdir(folder))
        for filename in os.listdir(folder):
            word_count = self.get_word_count(os.path.join(folder, filename))
            value = sum(self.weights.get(token, 0) * count for token, count in word_count.items())
            result = self.sigmoid(value)
            if eval(f"result {operator} 0.5"):
                correct += 1
        accuracy = (correct / total_files) * 100
        return accuracy

if __name__ == "__main__":
    train_ham_folder = "train/ham"
    train_spam_folder = "train/spam"
    test_ham_folder = "test/ham"
    test_spam_folder = "test/spam"

    model = LogisticRegression(train_ham_folder, train_spam_folder, test_ham_folder, test_spam_folder)
    model.run()
    model.test()
