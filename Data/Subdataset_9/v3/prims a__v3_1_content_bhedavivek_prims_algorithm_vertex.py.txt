class Vertex:
    def __init__(self, id):
        # Initialize vertex attributes
        self.position = 0          # Position of the vertex
        self.parent_id = 0         # Parent ID of the vertex
        self.distance = 9999999    # Distance of the vertex, initialized to a large value
        self.id = id               # Unique ID of the vertex

    def get_distance(self):
        # Method to get the distance of the vertex
        return self.distance

# Example usage:
# Create a vertex with ID 1
vertex1 = Vertex(1)
# Print the distance of vertex1
print("Distance of Vertex 1:", vertex1.get_distance())  # Output: 9999999
