import subprocess

import urllib.request
import json
import sqlite3
import pandas as pd

import plotly.graph_objs as go
import plotly.io as pio

# Define constants
DATA_DIR = os.path.join(os.path.dirname(os.path.abspath(__file__)), "data")
DATA_PATH = os.path.join(DATA_DIR, "temperature.db")
API_KEY = '0f0bb5973a4d0927'

# Connect to SQLite database
conn = sqlite3.connect(DATA_PATH)
cursor = conn.cursor()

# Create temperature table if not exists
cursor.execute('''CREATE TABLE IF NOT EXISTS temperature
             (date text, inside real, outside real)''')

# Function to read raw temperature data

    output, _ = cat_process.communicate()

# Function to read temperature
def read_temp():
    lines = read_temp_raw()
    while lines[0].strip()[-3:] != 'YES':
        time.sleep(0.2)
        lines = read_temp_raw()
    equals_pos = lines[1].find('t=')
    if equals_pos != -1:
        temp_string = lines[1][equals_pos+2:]
        temp_celsius = float(temp_string) / 1000.0
        temp_fahrenheit = temp_celsius * 9.0 / 5.0 + 32.0
        print('Time:', datetime.datetime.now())
        print('Indoor temperature:', temp_fahrenheit)
        return temp_celsius, temp_fahrenheit

# Function to get outside temperature from API
def get_outside_temp():
    url = f'http://api.wunderground.com/api/{API_KEY}/conditions/q/CA/Palm_Springs.json'
    with urllib.request.urlopen(url) as response:
        data = json.loads(response.read().decode())
        outside_temp_fahrenheit = data['current_observation']['temp_f']
        print('Outside temperature:', outside_temp_fahrenheit)
        return outside_temp_fahrenheit

# Main loop for data collection and forecasting
while True:
    # Read sensor data
    indoor_temp = round(read_temp()[1], 2)
    outdoor_temp = round(get_outside_temp(), 1)
    current_time = datetime.datetime.now().strftime('%Y-%m-%d %H:%M')

    # Insert data into SQLite database
    new_row = [(current_time, indoor_temp, outdoor_temp)]
    cursor.executemany("INSERT INTO temperature ('date', 'inside', 'outside') VALUES (?,?,?)", new_row)
    conn.commit()

    # Perform ARIMA forecasting
    df = pd.read_sql_query("SELECT * FROM temperature ORDER BY date DESC LIMIT 24*7", con=conn)
    model = ARIMA(df['outside'], order=(5, 1, 0))
    model_fit = model.fit(disp=0)
    forecast = model_fit.forecast(5)

    # Plotting using Plotly
    fig = go.Figure()
    fig.add_trace(go.Scatter(x=df['date'], y=df['inside'], mode='lines', name='Indoor Temperature'))
    fig.add_trace(go.Scatter(x=df['date'], y=df['outside'], mode='lines', name='Outdoor Temperature'))

    forecast_dates = [df['date'].iloc[-1] + datetime.timedelta(minutes=60*i) for i in range(1, 6)]
    forecast_dates_str = [date.strftime('%Y-%m-%d %H:%M') for date in forecast_dates]

    fig.add_trace(go.Scatter(x=forecast_dates_str, y=forecast[0], mode='lines', name='ARIMA Forecast'))

    fig.update_layout(title='Temperature', yaxis=dict(title='Temp [deg F]'))
    pio.write_html(fig, 'temperature_plot.html', auto_open=False)

    # Wait for one hour before the next iteration
    time.sleep(60*60)
