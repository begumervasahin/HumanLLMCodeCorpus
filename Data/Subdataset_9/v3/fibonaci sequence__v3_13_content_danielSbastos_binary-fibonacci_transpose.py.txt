import numpy as np
import matplotlib.pyplot as plt

# Function to generate Fibonacci numbers iteratively
def fibonacci_iterative(n):
    a, b = 0, 1
    for _ in range(n):
        a, b = b, a + b
    return a

# Function to format a number into an 80-bit binary representation
def format_binary(number):
    return format(number, '080b')

# Define the number of Fibonacci numbers to generate
num_fibonacci = 100

# Generate Fibonacci numbers and convert them to binary
fibonacci_sequence = [fibonacci_iterative(i) for i in range(num_fibonacci + 1)]
binary_sequence = [format_binary(num) for num in fibonacci_sequence]

# Convert binary sequence to numpy array
binary_array = np.array([[int(bit) for bit in binary] for binary in binary_sequence])

# Transpose the binary array
transposed_binary_array = np.transpose(binary_array)

# Plot the transposed binary array
plt.imshow(transposed_binary_array, interpolation='nearest')
plt.show()
