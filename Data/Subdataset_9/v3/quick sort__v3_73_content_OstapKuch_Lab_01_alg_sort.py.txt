class Student:
    def __init__(self, rating, growth):
        self.rating = rating
        self.growth = growth

class SortingAlgorithms:
    def __init__(self):
        self.comparisons = 0
        self.swaps = 0

    def bubble_sort(self, arr):
        self.reset_counts()
        swapped = True
        while swapped:
            swapped = False
            for i in range(len(arr) - 1):
                self.comparisons += 1
                if arr[i].rating < arr[i + 1].rating:
                    arr[i], arr[i + 1] = arr[i + 1], arr[i]
                    self.swaps += 1
                    swapped = True
        print("Bubble Sort:\nComparison times:", self.comparisons, "\nSwap times:", self.swaps)

    def partition(self, pivot_index, start, end, arr):
        pivot_value = arr[pivot_index].growth
        arr[pivot_index], arr[end - 1] = arr[end - 1], arr[pivot_index]
        left_ptr = start
        for right_ptr in range(start, end - 1):
            self.comparisons += 1
            if arr[right_ptr].growth < pivot_value:
                arr[left_ptr], arr[right_ptr] = arr[right_ptr], arr[left_ptr]
                self.swaps += 1
                left_ptr += 1
        arr[left_ptr], arr[end - 1] = arr[end - 1], arr[left_ptr]
        self.swaps += 1
        return left_ptr

    def quick_sort(self, start, end, arr):
        if start < end:
            pivot_index = (start + end) // 2
            new_pivot_index = self.partition(pivot_index, start, end, arr)
            self.quick_sort(start, new_pivot_index, arr)
            self.quick_sort(new_pivot_index + 1, end, arr)

    def reset_counts(self):
        self.comparisons = 0
        self.swaps = 0

# Testing the sorting algorithms
if __name__ == "__main__":
    sorting_algo = SortingAlgorithms()

    # Creating student objects
    students = [
        Student(80, 170),
        Student(75, 165),
        Student(85, 175),
        Student(70, 160),
        Student(90, 180)
    ]

    # Bubble sort
    print("Bubble Sort:")
    sorting_algo.bubble_sort(students[:])
    print()

    # Quick sort
    print("Quick Sort:")
    sorting_algo.quick_sort(0, len(students), students)
    print("Comparison times:", sorting_algo.comparisons)
    print("Swap times:", sorting_algo.swaps)
