import numpy as np
import re
import random
import csv

POS_DATASET_PATH = 'twitter-sentiment-dataset/tw-data.pos'
NEG_DATASET_PATH = 'twitter-sentiment-dataset/tw-data.neg'
VOC_PATH = 'twitter-sentiment-dataset/vocab.csv'
VOC_INV_PATH = 'twitter-sentiment-dataset/vocab_inv.csv'


def clean_string(string):
    """
    Clean and tokenize the input string.
    """
    string = re.sub(r"[^A-Za-z0-9(),!?\'\`]", " ", string)
    string = re.sub(r'(.)\1+', r'\1\1', string)
    string = re.sub(r"\'s", " 's", string)
    string = re.sub(r"\'ve", " 've", string)
    string = re.sub(r"n\'t", " n't", string)
    string = re.sub(r"\'re", " 're", string)
    string = re.sub(r"\'d", " 'd", string)
    string = re.sub(r"\'ll", " 'll", string)
    string = re.sub(r",", " , ", string)
    string = re.sub(r"!", " ! ", string)
    string = re.sub(r"\(", " ( ", string)
    string = re.sub(r"\)", " ) ", string)
    string = re.sub(r"\?", " ? ", string)
    string = re.sub(r"\s{2,}", " ", string)
    return string.strip().lower()


def sample_fraction(lst, fraction):
    """
    Randomly sample a fraction of the input list.
    """
    return random.sample(lst, int(len(lst) * fraction))


def load_data_and_labels(dataset_fraction):
    """
    Load data from files, process it, and generate labels.
    """
    print("Loading and processing data...")
    positive_examples = [line.strip() for line in open(POS_DATASET_PATH).readlines()]
    negative_examples = [line.strip() for line in open(NEG_DATASET_PATH).readlines()]

    positive_examples = sample_fraction(positive_examples, dataset_fraction)
    negative_examples = sample_fraction(negative_examples, dataset_fraction)

    x_text = positive_examples + negative_examples

    print("Cleaning strings...")
    x_text = [clean_string(sent) for sent in x_text]
    x_text = [sent.split(" ") for sent in x_text]

    print("Generating labels...")
    positive_labels = [[0, 1] for _ in positive_examples]
    negative_labels = [[1, 0] for _ in negative_examples]

    y = np.concatenate([positive_labels, negative_labels], axis=0)
    return x_text, y


def pad_sentences(sentences, padding_word="<PAD/>"):
    """
    Pad sentences to the same length.
    """
    max_len = max(len(sentence) for sentence in sentences)
    padded_sentences = [sentence + [padding_word] * (max_len - len(sentence)) for sentence in sentences]
    return padded_sentences


def build_vocab():
    """
    Build vocabulary and its inverse mapping.
    """
    with open(VOC_PATH, 'r') as f:
        voc = csv.reader(f)
        vocabulary = {word: index for word, index in voc}

    with open(VOC_INV_PATH, 'r') as f:
        voc_inv = csv.reader(f)
        vocabulary_inv = list(voc_inv)

    return vocabulary, vocabulary_inv


def build_input_data(sentences, labels, vocabulary):
    """
    Map sentences and labels to vectors using vocabulary.
    """
    x = np.array([[vocabulary[word] for word in sentence] for sentence in sentences])
    y = np.array(labels)
    return x, y


def load_data(dataset_fraction):
    """
    Load and preprocess data.
    """
    x_text, labels = load_data_and_labels(dataset_fraction)

    print("Padding strings...")
    sentences_padded = pad_sentences(x_text)

    print("Building vocabulary...")
    vocabulary, vocabulary_inv = build_vocab()

    print("Building processed datasets...")
    x, y = build_input_data(sentences_padded, labels, vocabulary)
    return x, y, vocabulary, vocabulary_inv


def batch_iterator(data, batch_size, num_epochs):
    """
    Generate batches of data for training.
    """
    data = np.array(data)
    data_size = len(data)
    num_batches_per_epoch = (len(data) + batch_size - 1) // batch_size

    for epoch in range(num_epochs):
        shuffle_indices = np.random.permutation(np.arange(data_size))
        shuffled_data = data[shuffle_indices]

        for batch_num in range(num_batches_per_epoch):
            start_index = batch_num * batch_size
            end_index = min((batch_num + 1) * batch_size, data_size)
            yield shuffled_data[start_index:end_index]
