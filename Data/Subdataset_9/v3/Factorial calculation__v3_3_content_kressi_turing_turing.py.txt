from turing_machine import TuringMachine
import ast
import argparse

def execute_turing_machine(config_name, step=False, verbose=False, tapes=None):
    """
    Execute a Turing machine based on the provided configuration.

    Args:
    - config_name (str): Name of the Turing configuration file (without the extension). 
    - step (bool): Flag indicating whether to step through with Enter.
    - verbose (bool): Flag indicating whether to print every step.
    - tapes (str): Content of tapes separated by comma.

    Returns:
    - None
    """
    # Parse command-line arguments
    args = parse_command_line_args()

    # Initialize Turing machine configuration
    config = load_turing_config(args.config)

    # Set tapes input
    tapes_input = args.tapes.split(',') if args.tapes else input_tapes()

    # Create Turing machine instance
    turing_machine = create_turing_machine(config, tapes_input)

    # Execute Turing machine
    execute_steps(turing_machine, args.step, args.verbose)

    # Print result
    print_result(turing_machine)

def parse_command_line_args():
    """Parse command-line arguments."""
    parser = argparse.ArgumentParser(description='Execute a Turing machine.')
    parser.add_argument('config', help='name of Turing configuration')
    parser.add_argument('-s', '--step', action='store_true', help='step through with Enter')
    parser.add_argument('-v', '--verbose', action='store_true', help='print every step')
    parser.add_argument('-t', '--tapes', help='content of tapes separated by comma (e.g. 00000,10000)')
    return parser.parse_args()

def load_turing_config(config_name):
    """Load Turing machine configuration."""
    config_file_name = config_name + ".conf"
    with open(config_file_name, 'r') as config_file:
        return ast.literal_eval(config_file.read())

def input_tapes():
    """Prompt user for content of tapes."""
    return str(input("Content of tapes? ")).split()

def create_turing_machine(config, tapes_input):
    """Create Turing machine instance."""
    return TuringMachine(tapesInput=tapes_input, **config)

def execute_steps(turing_machine, step, verbose):
    """Execute Turing machine steps."""
    while turing_machine.step():
        if step:
            input("Press Enter to continue...")
        if verbose:
            print(turing_machine)

def print_result(turing_machine):
    """Print the result of Turing machine execution."""
    char = '0'
    print("{0} occurs {1} times".format(char, turing_machine.getCount(char)))
    print("Steps: {0}".format(turing_machine.getStepCount()))

if __name__ == "__main__":
    execute_turing_machine()
