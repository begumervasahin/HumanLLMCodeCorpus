# Import necessary modules
import os
import sys

# Check if the script is being run as the main program
if __name__ == "__main__":
    # Set the Django settings module to "root.settings" if not already set   
    os.environ.setdefault("DJANGO_SETTINGS_MODULE", "root.settings")

    try:
        # Attempt to import the execute_from_command_line function from Django
        from django.core.management import execute_from_command_line
    except ImportError as exc:
        # If Django is not installed, raise an ImportError with an informative message
        raise ImportError(
            "Failed to import Django. Please make sure it's installed and "
            "accessible on your PYTHONPATH environment variable. Have you "
            "activated a virtual environment?"
        ) from exc

    # Execute Django management commands passed through the command line     
    execute_from_command_line(sys.argv)
