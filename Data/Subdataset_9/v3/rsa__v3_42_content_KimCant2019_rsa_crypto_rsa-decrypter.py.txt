import os
from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives import hashes, padding
from cryptography.hazmat.primitives.asymmetric import rsa, serialization

def load_private_key(key_file_path):
    with open(key_file_path, 'rb') as key_file:
        return serialization.load_pem_private_key(
            key_file.read(),
            password=None,
            backend=default_backend()
        )

def load_encrypted_message(message_file_path):
    with open(message_file_path, 'rb') as message_file:
        return message_file.read()

def decrypt_message(private_key, encrypted_message):
    return private_key.decrypt(
        encrypted_message,
        padding.OAEP(
            mgf=padding.MGF1(algorithm=hashes.SHA256()),
            algorithm=hashes.SHA256(),
            label=None
        )
    ).decode('utf-8')

def main():
    # Get the directory path where the script resides
    dir_path = os.path.dirname(os.path.realpath(__file__))

    # Load the private key
    private_key_path = os.path.join(dir_path, 'private.pem')
    private_key = load_private_key(private_key_path)

    # Load the encrypted message
    encrypted_message_path = os.path.join(dir_path, 'encrypted-message.enc')
    encrypted_message = load_encrypted_message(encrypted_message_path)

    # Decrypt the message
    plaintext = decrypt_message(private_key, encrypted_message)

    # Print the decrypted plaintext
    print("Decrypted message:\n", plaintext)

if __name__ == "__main__":
    main()
