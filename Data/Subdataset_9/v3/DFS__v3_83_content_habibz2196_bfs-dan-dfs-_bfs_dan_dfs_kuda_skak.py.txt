def breadth_first_search(graph, start, end):
    """
    Perform Breadth-First Search (BFS) on the given graph to find a path from the start node to the end node.
    Returns the path if found, otherwise prints "Tidak ditemukan".
    """
    queue = [[start]]  # Initialize a queue with the start node
    visited = set()  # Initialize a set to keep track of visited nodes

    while queue:
        path = queue.pop(0)  # Dequeue the first path from the queue
        state = path[-1]  # Get the current node from the path

        if state == end:
            return path  # Return the path if the end node is reached
        elif state not in visited:
            for neighbor in graph.get(state, []):
                new_path = list(path)  # Create a new path by appending the neighbor to the current path
                new_path.append(neighbor)
                queue.append(new_path)  # Enqueue the new path
            visited.add(state)  # Mark the current node as visited

    print("Tidak ditemukan")

def depth_first_search(graph, start, end):
    """
    Perform Depth-First Search (DFS) on the given graph to find a path from the start node to the end node.
    Returns the path if found, otherwise prints "Tidak ditemukan".
    """
    stack = [[start]]  # Initialize a stack with the start node
    visited = set()  # Initialize a set to keep track of visited nodes

    while stack:
        path = stack.pop()  # Pop the top path from the stack
        state = path[-1]  # Get the current node from the path

        if state == end:
            return path  # Return the path if the end node is reached
        elif state not in visited:
            for neighbor in graph.get(state, []):
                new_path = list(path)  # Create a new path by appending the neighbor to the current path
                new_path.append(neighbor)
                stack.append(new_path)  # Push the new path onto the stack
            visited.add(state)  # Mark the current node as visited

    print("Tidak ditemukan")
