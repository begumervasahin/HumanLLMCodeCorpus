import time
from mpi4py import MPI

start_time = time.time()

def longest_common_subsequence(s1, s2):
    # Function to find the longest common subsequence between two strings
    matrix = [["" for _ in range(len(s2))] for _ in range(len(s1))]
    
    # Populate the matrix
    for i in range(len(s1)):
        for j in range(len(s2)):
            if s1[i] == s2[j]:
                if i == 0 or j == 0:
                    matrix[i][j] = s1[i]
                else:
                    matrix[i][j] = matrix[i - 1][j - 1] + s1[i]
            else:
                matrix[i][j] = max(matrix[i - 1][j], matrix[i][j - 1], key=len)
    
    # Get the longest common subsequence and its length
    longest_subsequence = matrix[-1][-1]
    return len(longest_subsequence), longest_subsequence

# MPI setup
comm = MPI.COMM_WORLD

    num_procs = comm.Get_size()
    
    # Read input strings from files
    str1 = open('a.txt').read()

    # Broadcast sub_x to all processes
    for b in range(num_procs):
        if b <= x_size:
            sub_x[:] = str1
    comm.bcast(sub_x, root=0)
    
    # Send sub_y to all processes
    for k in range(1, num_procs):
        comm.send(sub_y, dest=k)
    
    comm.barrier()
    
    # Compute LCS
    longest_common_subsequence(open('a.txt').read(), open('b.txt').read())
    
    if num_procs > 1:
        comm.Isend(f_tab[r_size], dest=0)
else:
    # Receive sub_x from root process

    # Compute LCS

        comm.Isend(f_tab[r_size], dest=0)
