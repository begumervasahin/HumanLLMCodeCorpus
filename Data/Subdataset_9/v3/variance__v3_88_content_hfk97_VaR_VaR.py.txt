import subprocess
import importlib
import sys

def ensure_package(package_name):
    """
    Ensures that a package is imported, installing it if necessary.
    """
    try:
        return importlib.import_module(package_name)
    except ImportError:
        subprocess.call([sys.executable, "-m", "pip", "install", package_name], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        return importlib.import_module(package_name)

# Ensure necessary packages are available
np = ensure_package("numpy")
from scipy.stats import norm

def historical_var(returns, confidence_level, days):
    """
    Calculates the historical Value at Risk (VaR) for a series of returns.
    """
    alpha = 1 - (confidence_level / 100)
    var = -np.quantile(returns, alpha) * np.sqrt(days)
    return var

def variance_covariance_var(returns, confidence_level, days):
    """
    Calculates the VaR using the variance-covariance method.
    """
    alpha = 1 - (confidence_level / 100)
    std_dev = np.std(returns)
    mean = np.mean(returns)
    var = (norm.ppf(1 - alpha) * std_dev - mean) * np.sqrt(days)
    return var

def historical_cvar(returns, var, days):
    """
    Calculates the historical Conditional Value at Risk (CVaR).
    """
    es = -returns[returns < -var / np.sqrt(days)].mean() * np.sqrt(days)
    return es

def variance_covariance_cvar(returns, confidence_level, days):
    """
    Calculates the CVaR using the variance-covariance method.
    """
    alpha = 1 - (confidence_level / 100)
    std_dev = np.std(returns)
    mean = np.mean(returns)
    cvar = (alpha**-1 * norm.pdf(norm.ppf(alpha)) * std_dev - mean) * np.sqrt(days)
    return cvar

if __name__ == "__main__":
    # Example usage
    returns = np.random.randn(1000)  # Sample returns
    confidence_level = 95  # Confidence level in percentage
    days = 1  # Number of days
    
    # Calculate VaR
    var_hist_result = historical_var(returns, confidence_level, days)
    var_vcov_result = variance_covariance_var(returns, confidence_level, days)
    print("Historical VaR:", var_hist_result)
    print("Variance-Covariance VaR:", var_vcov_result)

    # Calculate CVaR
    cvar_hist_result = historical_cvar(returns, var_hist_result, days)
    cvar_vcov_result = variance_covariance_cvar(returns, confidence_level, days)
    print("Historical CVaR:", cvar_hist_result)
    print("Variance-Covariance CVaR:", cvar_vcov_result)
