import sys

# Import the appropriate Tkinter module based on Python version
if sys.version_info[0] < 3:
    import Tkinter as Tk
else:
    import tkinter as Tk

# Import the App class from the App module
from App import App

# Define the main function to run the application
def main():
    # Create the main application window
    root = Tk.Tk()
    # Set the title of the window
    root.title("Prim & Dijkstra's Algorithms")
    # Initialize the application with the main window
    app = App(root)
    # Start the Tkinter event loop
    root.mainloop()

# Ensure that the main function is called when the script is executed
if __name__ == "__main__":
    main()
