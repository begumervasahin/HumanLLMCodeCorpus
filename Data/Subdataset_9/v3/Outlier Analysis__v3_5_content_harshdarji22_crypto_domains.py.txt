# Load domain names from a CSV file
def load_domains(file_path):
    with open(file_path, 'r') as csvfile:
        csv_reader = csv.reader(csvfile, delimiter=' ', quotechar='|')
        return [row[0] for row in csv_reader]

# Function to fetch and save data for a given cryptocurrency
def fetch_and_save_crypto_data(crypto_name, domains):
    crypto_counts = []
    for domain in domains:
        # Query to count occurrences of a cryptocurrency in the content matching the domain
        query = f"""
            SELECT COUNT(*) 
            FROM CryptoNews.cryptonews 
            WHERE MATCH (content) AGAINST ('{crypto_name}' IN NATURAL LANGUAGE MODE) 
            AND link LIKE '%{domain}%'
        """
        result = pd.read_sql(query, connection)
        crypto_counts.append([domain, result.iloc[0, 0]])

    # Save the data to a CSV file
    file_name = f"dom_{crypto_name}.csv"
    with open(file_name, 'w', newline='') as csvfile:
        csv_writer = csv.writer(csvfile, quoting=csv.QUOTE_ALL)
        csv_writer.writerows(crypto_counts)
    print(f"Data for {crypto_name} saved to {file_name}.")

# Cryptocurrencies to fetch data for
cryptocurrencies = ['bitcoin', 'ethereum', 'ripple', 'bitcoin cash', 'eos', 'litecoin', 'monero']

# Load domains
domains = load_domains('domains.csv')

# Fetch and save data for each cryptocurrency
for crypto in cryptocurrencies:
    fetch_and_save_crypto_data(crypto, domains)
