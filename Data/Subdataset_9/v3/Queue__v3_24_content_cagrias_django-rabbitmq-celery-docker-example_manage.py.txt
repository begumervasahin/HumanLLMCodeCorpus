import os
import sys

def set_django_settings_module():
    """Set the DJANGO_SETTINGS_MODULE environment variable to your Django project's settings module."""
    os.environ.setdefault('DJANGO_SETTINGS_MODULE', 'mypubsub.settings')

def execute_django_command():
    """Execute Django commands from the command line arguments."""
    try:
        # Import the execute_from_command_line function from Django's management module
        from django.core.management import execute_from_command_line
    except ImportError as exc:
        # If Django is not installed or import fails, raise ImportError with a helpful message
        raise ImportError(
            "Couldn't import Django. Make sure it's installed and "
            "available on your PYTHONPATH environment variable. Also, "
            "ensure that you've activated a virtual environment if needed."
        ) from exc

    # Execute Django commands from the command line arguments
    execute_from_command_line(sys.argv)

def main():
    """Main function to set up Django environment and execute Django commands."""
    set_django_settings_module()
    execute_django_command()

if __name__ == '__main__':
    # Call the main function when the script is executed
    main()
