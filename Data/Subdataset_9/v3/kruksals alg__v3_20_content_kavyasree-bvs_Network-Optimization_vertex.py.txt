class Vertex:
    def __init__(self, name):
        """
        Initialize a vertex with a given name and an empty list of neighbors.
        """
        self.name = name
        self.neighbors = []

    def add_neighbor(self, neighbor):
        """
        Add a neighbor to the vertex.
        """
        if isinstance(neighbor, Vertex):
            if neighbor.name not in self.neighbors:
                self.neighbors.append(neighbor.name)
                neighbor.neighbors.append(self.name)
                # Sort the list of neighbors for consistency
                self._sort_neighbors()
                neighbor._sort_neighbors()

    def _sort_neighbors(self):
        """
        Sort the list of neighbors.
        """
        self.neighbors.sort()

    def __repr__(self):
        """
        Return a string representation of the vertex's neighbors.
        """
        return str(self.neighbors)

# Example usage:
if __name__ == "__main__":
    # Create vertices
    v1 = Vertex("A")
    v2 = Vertex("B")
    v3 = Vertex("C")

    # Add neighbors
    v1.add_neighbor(v2)
    v1.add_neighbor(v3)
    v2.add_neighbor(v3)

    # Print neighbors for each vertex
    for vertex in [v1, v2, v3]:
        print("Neighbors of vertex", vertex.name, ":", vertex)
