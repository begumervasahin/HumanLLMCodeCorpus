class ListNode:
    def __init__(self, value, next_node=None):
        self.value = value
        self.next = next_node

# Functions to operate on linked lists
def length_linked_list(ll):
    """
    Iteratively calculates the length of a linked list.
    """
    count = 0
    while ll:
        count += 1
        ll = ll.next
    return count

def length_linked_list_recursive(ll):
    """
    Recursively calculates the length of a linked list.
    """
    if not ll:
        return 0
    return 1 + length_linked_list_recursive(ll.next)

def sum_linked_list(ll):
    """
    Iteratively calculates the sum of values in a linked list.
    """
    total = 0
    while ll:
        total += ll.value
        ll = ll.next
    return total

def sum_linked_list_recursive(ll):
    """
    Recursively calculates the sum of values in a linked list.
    """
    if not ll:
        return 0
    return ll.value + sum_linked_list_recursive(ll.next)

def stringify_linked_list(ll):
    """
    Converts a linked list into a string representation.
    """
    answer = ''
    while ll:
        answer += str(ll.value) + '->'
        ll = ll.next
    return answer + 'None'

def stringify_linked_list_recursive(ll):
    """
    Recursively converts a linked list into a string representation.
    """
    if not ll:
        return 'None'
    return str(ll.value) + '->' + stringify_linked_list_recursive(ll.next)

def list_to_linked_list(lst):
    """
    Converts a list to a linked list.
    """
    if not lst:
        return None
    head = current = ListNode(lst[0])
    for value in lst[1:]:
        current.next = ListNode(value)
        current = current.next
    return head

def list_to_linked_list_recursive(lst):
    """
    Recursively converts a list to a linked list.
    """
    if not lst:
        return None
    return ListNode(lst[0], list_to_linked_list_recursive(lst[1:]))
