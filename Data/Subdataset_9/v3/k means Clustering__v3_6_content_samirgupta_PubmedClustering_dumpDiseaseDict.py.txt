import sys
from time import time
import pickle

def process_pubtator_file(input_file_name):
    """
    Process the PubTator file and extract PMID-disease mappings.
    
    Args:
    input_file_name (str): Name of the input file.
    
    Returns:
    dict: Dictionary containing PMID-disease mappings.
    """
    pmid_disease_dict = {}
    
    # Open and read the input file
    with open(input_file_name, "r") as lines_fh:
        lines = lines_fh.readlines()
    
    # Record the starting time for performance measurement
    start_time = time()
    
    # Iterate through each line in the file
    for line in lines:
        line = line.strip()  # Remove leading/trailing whitespaces
        tokens = line.split("\t")  # Split the line into tokens using tab as delimiter
        
        # Check if the line contains exactly 4 tokens
        if len(tokens) == 4:
            pmid, mesh_id, dis_mens, dis_source = tokens[:4]
            diseases = dis_mens.split("|")  # Split Disease Mentions into individual diseases
            
            # Update the dictionary with PMID-disease mappings
            pmid_disease_dict.setdefault(pmid, []).extend(diseases)

    # Print the time taken to process the file
    print("Processing done in {:.2f} seconds".format(time() - start_time))
    print()

    return pmid_disease_dict

def run():
    """
    Main function to run the script.
    """
    # Check if the correct number of command-line arguments is provided
    if len(sys.argv) != 2:
        print("Usage: python script.py <input_file_name>")
        return

    input_file_name = sys.argv[1]  # Get the input file name from command-line arguments
    pmid_disease_dict = process_pubtator_file(input_file_name)  # Process the PubTator file
    file_name = './pubtator_dumps/pmid_disease_dict.pickle'  # Define the path for saving the pickle file

    # Save the PMID-disease dictionary as a pickle file
    with open(file_name, 'wb') as handle:
        pickle.dump(pmid_disease_dict, handle, protocol=pickle.HIGHEST_PROTOCOL)

if __name__ == '__main__':
    run()  # Execute the run function if the script is executed directly
