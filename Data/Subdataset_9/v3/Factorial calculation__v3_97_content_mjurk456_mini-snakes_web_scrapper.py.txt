import requests
import os
import bs4

def download_images_from_url(url, download_path):
    """
    Downloads images from a specified URL and saves them to a directory.

    Parameters:
    url (str): The URL of the webpage containing images.
    download_path (str): The directory path to save the downloaded images.
    """
    # Create the download directory if it doesn't exist
    os.makedirs(download_path, exist_ok=True)

    # Loop through pages until there are no more next links
    while True:
        # Download the webpage content
        print("Downloading page:", url)
        response = requests.get(url)
        response.raise_for_status()
        soup = bs4.BeautifulSoup(response.text, "lxml")

        # Find image elements on the webpage
        image_elements = soup.select(".image-size-full")
        for element in image_elements:
            img_url = element.get('src')
            print("Downloading image:", img_url)

            # Download the image content
            image_response = requests.get(img_url)
            image_response.raise_for_status()

            # Save the image to the specified directory
            with open(os.path.join(download_path, os.path.basename(img_url)), 'wb') as image_file:
                for chunk in image_response.iter_content(100000):
                    image_file.write(chunk)

        # Find the link to the next page
        next_link = soup.find("a", class_="pagination")
        if next_link:
            url = next_link.get('href')
        else:
            break

    print("Download completed.")

if __name__ == "__main__":
    # URL and download directory
    url = 'https://www.boredpanda.com/funny-pet-jerks/'
    download_directory = "/home/maria/Pictures/downloaded/"

    # Call the function to download images
    download_images_from_url(url, download_directory)
