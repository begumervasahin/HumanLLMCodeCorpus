class TreeNode:
    def __init__(self, key):
        self.key = key
        self.left_child = None
        self.right_child = None
        self.parent_node = None
        self.height = 0
        self.real_height = 0
        self.color = 1  # Default color is set to 1 (red)

# Helper function to print node attributes
def print_node_info(node):
    print(f"Key: {node.key}")
    print(f"Left Child: {node.left_child}")
    print(f"Right Child: {node.right_child}")
    print(f"Parent Node: {node.parent_node}")
    print(f"Height: {node.height}")
    print(f"Real Height: {node.real_height}")
    print(f"Color: {'Red' if node.color == 1 else 'Black'}")  # Print color as Red or Black

# Example usage:
if __name__ == "__main__":
    # Create tree nodes and set their attributes
    node1 = TreeNode(10)
    node1.height = 3
    node1.real_height = 4
    node1.color = 1  # Set color to 1 (red)

    node2 = TreeNode(20)
    node2.height = 2
    node2.real_height = 3
    node2.color = 0  # Set color to 0 (black)

    # Print node attributes for demonstration
    print("Node 1:")
    print_node_info(node1)

    print("\nNode 2:")
    print_node_info(node2)
