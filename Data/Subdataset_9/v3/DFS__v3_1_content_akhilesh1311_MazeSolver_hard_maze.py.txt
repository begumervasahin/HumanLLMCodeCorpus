import random
import copy

def breadth_first_search(maze, is_path=False):
    path = []
    queue = [(0, 0)]
    visited = set()
    visited.add((0, 0))
    parent_node = {}

    while queue:
        cur_i, cur_j = queue.pop(0)

        if maze[cur_i][cur_j] == 'G':
            ans = 0
            while cur_i + cur_j:
                path.append((cur_i, cur_j))
                cur_i, cur_j = parent_node[(cur_i, cur_j)]
                ans += 1
            if is_path:
                for p, q in path:
                    if maze[p][q] in ['O', 'X']:
                        maze[p][q] = ' '
                display_maze(maze)
                return
            else:
                return ans

        for i, j in [(cur_i - 1, cur_j), (cur_i + 1, cur_j), (cur_i, cur_j - 1), (cur_i, cur_j + 1)]:
            if 0 <= i < len(maze) and 0 <= j < len(maze) and maze[i][j] != 'X' and (i, j) not in visited:
                queue.append((i, j))
                parent_node[(i, j)] = (cur_i, cur_j)
                visited.add((i, j))

    return -1

def find_best_maze(maze_list):
    evaluation = []

    for index, maze in enumerate(maze_list):
        nodes_expanded = breadth_first_search(maze)
        evaluation.append((nodes_expanded, random.random(), index))

    evaluation.sort(reverse=True)
    return evaluation[0][-1]

def display_maze(maze):
    for row in maze:
        print(*row, sep=' ')
    print()

if __name__ == '__main__':
    # Parameters
    dimension = 15
    obstacle_probability = 0.3
    iterations = 100

    # Generate initial maze
    maze = [['X' if random.random() < obstacle_probability else 'O' for _ in range(dimension)] for _ in range(dimension)]
    maze[0][0] = 'S'
    maze[dimension - 1][dimension - 1] = 'G'

    # Display initial maze
    print("Displaying initial maze:")
    display_maze(maze)

    # Find and display shortest path in initial maze
    print("Displaying initial maze with the shortest path:")
    breadth_first_search(maze, True)

    # Convert explored cells in shortest path to open cells 'O'
    for i in range(dimension):
        for j in range(dimension):
            if maze[i][j] == ' ':
                maze[i][j] = 'O'

    # Randomly select starting point for hill climbing
    x, y = random.randint(0, dimension - 1), random.randint(0, dimension - 1)

    # Hill climbing iterations
    for _ in range(iterations):
        maze_list = [maze]

        # Generate neighboring mazes by flipping obstacles
        for i, j in [(x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)]:
            if 0 <= i < dimension and 0 <= j < dimension and maze[i][j] in ['O', 'X']:
                new_maze = copy.deepcopy(maze)
                new_maze[i][j] = 'O' if new_maze[i][j] == 'X' else 'X'
                maze_list.append(new_maze)

        # Find the best maze from generated neighbors
        index = find_best_maze(maze_list)
        maze = maze_list[index]

        # Update starting point if a better maze is found
        if index:
            x, y = [(x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)][index - 1]
        else:
            x, y = random.randint(0, dimension - 1), random.randint(0, dimension - 1)

    # Display final maze after hill climbing
    print("Displaying maze after hill climbing:")
    display_maze(maze)

    # Display final maze with the shortest path after hill climbing
    print("Displaying maze with the shortest path after hill climbing:")
    breadth_first_search(maze, True)
