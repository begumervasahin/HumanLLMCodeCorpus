def update_cell(row, col, board):
    """Update possible choices for a cell."""
    choices = []
    for k in [str(v) for v in range(1, 10)]:
        if is_valid_choice(row, col, k, board):
            choices.append(k)
    return choices

def is_valid_choice(row, col, value, board):
    """Check if a value is a valid choice for a cell."""
    for y in range(9):
        if board[row][y] == value or \
           board[y][col] == value or \
           board[row - row % 3 + y // 3][col - col % 3 + y % 3] == value:
            return False
    return True

def print_board(board):
    """Print the Sudoku board."""
    for row in board:
        print(" ".join(row))

def sudoku_solve(board):
    """Solve the Sudoku puzzle recursively."""
    row, col = find_empty_cell(board)
    if row == -1 and col == -1:
        print_board(board)
        return True
    choices = update_cell(row, col, board)
    for value in choices:
        board[row][col] = value
        if sudoku_solve(board):
            return True
        board[row][col] = '.'
    return False

def find_empty_cell(board):
    """Find the first empty cell in the Sudoku board."""
    for i in range(9):
        for j in range(9):
            if board[i][j] == '.':
                return i, j
    return -1, -1
