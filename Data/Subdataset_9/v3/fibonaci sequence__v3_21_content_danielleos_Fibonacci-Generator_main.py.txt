def fibonacci_generator(limit):
    fibonacci_sequence = [0, 1]  # Initialize with first two Fibonacci numbers
    while len(fibonacci_sequence) < limit:
        fibonacci_sequence.append(fibonacci_sequence[-1] + fibonacci_sequence[-2])
    return fibonacci_sequence[:limit]

while True:
    try:
        num_fibonacci = int(input("How many Fibonacci numbers do you want to generate? "))
    except ValueError:
        print("Please enter a valid whole number.")
        continue

    if num_fibonacci <= 0:
        print("Please enter a positive integer.")
        continue
    else:
        print(f"Generating {num_fibonacci} Fibonacci numbers...")
        fib_numbers = fibonacci_generator(num_fibonacci)
        print(f"Fibonacci sequence: {fib_numbers}")
        break
