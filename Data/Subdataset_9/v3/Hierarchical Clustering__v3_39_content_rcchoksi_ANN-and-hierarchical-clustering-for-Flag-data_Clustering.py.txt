import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from scipy.cluster.hierarchy import dendrogram, linkage, cophenet, maxdists
from scipy.spatial.distance import pdist

# Load the dataset
file_path = 'Flagdata.csv'
df = pd.read_csv(file_path)

# Standardize the data
scaler = StandardScaler()
scaled_data = scaler.fit_transform(df)

# Perform hierarchical clustering
linkage_matrix = linkage(scaled_data, method="average")

# Calculate the cophenetic correlation coefficient
coph_corr, coph_distances = cophenet(linkage_matrix, pdist(scaled_data))
print('Cophenet Correlation coefficient:', coph_corr)
print('Cophenet pairwise distances:', coph_distances)
print("First Cluster:", linkage_matrix[0])

# Calculate the maximum pairwise distances
max_dist_array = maxdists(linkage_matrix)
print("Distance Array:", max_dist_array)

# Plot the dendrogram
plt.figure(figsize=(30, 15))
plt.title('Dendrogram for Flag Data')
plt.xlabel('Sample Index')
plt.ylabel('Distance')
dendrogram(
    linkage_matrix,
    leaf_rotation=90.,
    leaf_font_size=10.,
)
plt.show()

# Plot the truncated dendrogram
plt.figure(figsize=(20, 10))
dendrogram(
    linkage_matrix,
    truncate_mode='lastp',
    p=12,
    leaf_rotation=90.,
    leaf_font_size=12.,
    show_contracted=True,
)
plt.show()
