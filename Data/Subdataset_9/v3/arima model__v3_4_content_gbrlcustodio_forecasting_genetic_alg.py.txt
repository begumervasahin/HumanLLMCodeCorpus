import random
import copy
import numpy as np
import train_rbfn
from itertools import repeat, combinations
from cromossome import Cromossome

class GeneticAlgorithm:
    def __init__(self, training_data, test_data):
        self.training_data = training_data
        self.test_data = test_data
        self.population = []
        self.min_individual_size = 2
        self.max_individual_size = len(training_data["INPUT"]) - 1
        self.training_input = np.asarray([[x.tolist()] for x in training_data["INPUT"].values])
        self.training_output = np.asarray([[x.tolist()] for x in training_data["OUTPUT"].values])

    def initialize_population(self, size):
        for _ in repeat(None, size):
            individual = Cromossome()
            individual.string = random.sample(range(len(self.training_data["INPUT"])), random.randint(self.min_individual_size, self.max_individual_size))
            self.population.append(individual)

    def decode_individual(self, chromosome):
        centers = []
        for center_idx in chromosome:
            centers.append([self.training_data["INPUT"].values[center_idx].tolist()])
        return np.asarray(centers)

    def evaluate_individual(self, chromosome):
        network = train_rbfn.train(self.training_input, self.training_output, self.decode_individual(chromosome.string))
        chromosome.fitness = [network.aic(self.training_data), network.aic(self.test_data)]

    def evolve(self, generations):
        for generation in range(1, generations + 1):
            print("Generation ", generation)
            self.evaluate_population()
            self.rank_population()
            new_population = self.generate_new_population()
            self.population = new_population

        best_individual = self.get_best_individual()
        print(self.decode_individual(best_individual.string))
        return train_rbfn.train(self.training_input, self.training_output, self.decode_individual(best_individual.string))

    def evaluate_population(self):
        for individual in self.population:
            self.evaluate_individual(individual)

    def rank_population(self):
        ranked_population = sorted(self.population, key=lambda x: x.rank, reverse=True)
        for i, individual in enumerate(ranked_population):
            individual.rank = i + 1

    def select_individual(self):
        total_rank = sum(individual.rank for individual in self.population)
        pick = random.uniform(0, total_rank)
        current = 0
        for individual in self.population:
            current += individual.rank
            if current > pick:
                return individual

    def perform_crossover(self, parents):
        smaller, bigger = sorted(parents, key=lambda x: len(x.string))
        common_genes = set(smaller.string).intersection(bigger.string)
        different_genes_smaller = list(set(smaller.string) - common_genes)
        different_genes_bigger = list(set(bigger.string) - common_genes)

        if different_genes_smaller:
            quantity = random.randint(1, len(different_genes_smaller))
            first_sample = random.sample(different_genes_smaller, quantity)
            if different_genes_bigger:
                second_sample = random.sample(different_genes_bigger, quantity)
                for i, j in zip(first_sample, second_sample):
                    smaller.string[smaller.string.index(i)], bigger.string[bigger.string.index(j)] = bigger.string[bigger.string.index(j)], smaller.string[smaller.string.index(i)]

        return [smaller, bigger]

    def perform_mutation(self, chromosome):
        for idx, gene in enumerate(chromosome.string):
            if random.random() <= 0.01:
                chromosome.string[idx] = random.sample(set(range(len(self.training_data["INPUT"]))).difference(chromosome.string), 1)[0]

    def perform_deletion(self, chromosome):
        if random.random() <= 0.01:
            start_idx = random.randint(0, len(chromosome.string) - 1)
            end_idx = start_idx + random.randint(1, len(chromosome.string) - start_idx - 1)
            chromosome.string = [gene for idx, gene in enumerate(chromosome.string) if idx < start_idx or idx >= end_idx]

    def perform_addition(self, chromosome):
        if random.random() <= 0.01 and len(chromosome.string) < self.max_individual_size:
            quantity = random.randint(1, self.max_individual_size - len(chromosome.string))
            chromosome.string.extend(random.sample(set(range(len(self.training_data["INPUT"]))).difference(chromosome.string), quantity))

    def generate_new_population(self):
        new_population = []
        while len(new_population) < len(self.population):
            parents = [self.select_individual() for _ in range(2)]
            offspring = self.perform_crossover(parents) if random.random() <= 0.90 else [copy.deepcopy(parent) for parent in parents]
            for child in offspring:
                self.perform_mutation(child)
                self.perform_deletion(child)
                self.perform_addition(child)
                self.evaluate_individual(child)
                new_population.append(child)
        return new_population

    def get_best_individual(self):
        return max(self.population, key=lambda x: x.rank)
