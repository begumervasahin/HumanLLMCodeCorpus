import sys
import kruskals
import prims
import glb

def print_usage():
    """
    Print usage instructions for the script.
    """
    print("USAGE: main.py [\"file.txt\"] [algo] where p = prims, k = kruskals")
    sys.exit()

def execute_algorithm(file, delimiter, algorithm):
    """
    Execute the specified algorithm on the provided graph data.
    """
    if algorithm == 'p':
        # Execute Prim's algorithm
        prims.read_graph(file, delimiter)
        graph, total_distance = prims.prims()
    elif algorithm == 'k':
        # Execute Kruskal's algorithm
        kruskals.read_graph(file, delimiter)
        graph, total_distance = kruskals.kruskals()
    else:
        # Invalid algorithm specified, display usage
        print_usage()

    # Print the result
    glb.print_pretty()
    print(f"Total Distance: {total_distance}")

def main():
    """
    Main function of the script.
    """
    if len(sys.argv) < 3:
        print_usage()

    file_path = sys.argv[1]
    delimiter = ' '
    algorithm = sys.argv[2]

    execute_algorithm(file_path, delimiter, algorithm)

if __name__ == "__main__":
    main()
