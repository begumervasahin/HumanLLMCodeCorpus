from queue import Queue

class NQueens:
    """Class to solve the N-Queens problem using Depth-First Search (DFS) and Breadth-First Search (BFS)."""

    def __init__(self, size):
        """Initialize the NQueens object with the board size."""
        self.size = size

    def solve_dfs(self):
        """Solve the N-Queens problem using Depth-First Search."""
        if self.size < 1:
            return []
        solutions = []
        stack = [[]]  # Initialize stack with an empty solution
        while stack:
            solution = stack.pop()
            if self.has_conflict(solution):
                continue  # Skip this solution if it has conflicts
            row = len(solution)
            if row == self.size:
                solutions.append(solution)  # Solution found
                continue
            for col in range(self.size):
                queen = (row, col)
                queens = solution.copy()
                queens.append(queen)
                stack.append(queens)
        return solutions

    def solve_bfs(self):
        """Solve the N-Queens problem using Breadth-First Search."""
        if self.size < 1:
            return []
        solutions = []
        queue = Queue()  # Initialize queue
        queue.put([])  # Enqueue the initial empty solution
        while not queue.empty():
            solution = queue.get()
            if self.has_conflict(solution):
                continue  # Skip this solution if it has conflicts
            row = len(solution)
            if row == self.size:
                solutions.append(solution)  # Solution found
                continue
            for col in range(self.size):
                queen = (row, col)
                queens = solution.copy()
                queens.append(queen)
                queue.put(queens)  # Enqueue the new solution
        return solutions

    def has_conflict(self, queens):
        """Check if there is any conflict among the queens."""
        for i in range(1, len(queens)):
            for j in range(i):
                a, b = queens[i]
                c, d = queens[j]
                if a == c or b == d or abs(a - c) == abs(b - d):
                    return True  # Conflict found
        return False  # No conflict found

    def print_board(self, queens):
        """Print the board with queens placed."""
        for i in range(self.size):
            print(' ---' * self.size)
            for j in range(self.size):
                p = 'Q' if (i, j) in queens else ' '
                print('| %s ' % p, end='')
            print('|')
        print(' ---' * self.size)
