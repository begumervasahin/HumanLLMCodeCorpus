from selenium import webdriver
from selenium.common.exceptions import NoSuchElementException, TimeoutException, WebDriverException
import time

def switch_proxy(driver):
    # Create a new Firefox profile with proxy settings
    profile = configure_proxy()
    # Quit the current driver instance
    driver.quit()
    # Initialize a new driver instance with the updated Firefox profile
    driver = webdriver.Firefox(firefox_profile=profile)
    return driver

def configure_proxy():
    # Create a new Firefox profile
    profile = webdriver.FirefoxProfile()
    # Access the SSL proxies website
    proxy_url = "https://www.sslproxies.org/"
    driver = webdriver.Firefox()
    driver.get(proxy_url)
    # Extract proxy details from the website and configure Firefox profile
    tbody = driver.find_element_by_tag_name("tbody")
    for row in tbody.find_elements_by_tag_name("tr"):
        host, port = row.text.split(" ")
        profile.set_preference("network.proxy.type", 1)
        profile.set_preference("network.proxy.http", host)
        profile.set_preference("network.proxy.http_port", int(port))
        profile.set_preference("network.proxy.ssl", host)
        profile.set_preference("network.proxy.ssl_port", int(port))
    # Quit the temporary driver instance
    driver.quit()
    return profile

def main():
    while True:
        try:
            # Initialize a new Firefox driver instance
            driver = webdriver.Firefox()
            # Define the search query URL
            search_query_url = "http://www.google.com/search?q=chanel"
            # Load the search query URL
            driver.get(search_query_url)
            # Continuously loop to interact with search results
            while True:
                try:
                    # Find all search results on the page
                    results = driver.find_elements_by_xpath('//*[@id="rso"]')
                    # Print each search result's text
                    for result in results:
                        print(result.text)
                    # Find and click the 'Next' button to move to the next page of results
                    next_button = driver.find_element_by_xpath('//*[@id="pnnext"]')
                    next_button.click()
                except NoSuchElementException:
                    try:
                        # Check if a reCAPTCHA challenge is present
                        frame = driver.find_element_by_xpath('//iframe[contains(@src, "recaptcha")]')
                        driver.switch_to.frame(frame)
                        driver.find_element_by_xpath("//*[@id='recaptcha-anchor']")
                        # If reCAPTCHA is detected, switch to a new proxy
                        driver = switch_proxy(driver)
                        # Wait for a while before proceeding
                        time.sleep(30)
                        # Switch back to the main content
                        driver.switch_to.default_content()
                        continue
                    except NoSuchElementException:
                        # If no reCAPTCHA is detected, break out of the inner loop
                        break
        except (WebDriverException, TimeoutException) as e:
            # Handle any WebDriver exceptions or timeouts
            print(f"Error occurred: {e}")
        finally:
            # Quit the current driver instance in any case
            driver.quit()

if __name__ == "__main__":
    # Start the main function
    main()
