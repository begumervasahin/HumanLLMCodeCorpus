import sys

# Set recursion limit to avoid "RecursionError: maximum recursion depth exceeded"
sys.setrecursionlimit(3000)

def ackermann(m, n):
    """
    Computes the value of the Ackermann function for given non-negative integers m and n.
    
    Parameters:
        m (int): Non-negative integer.
        n (int): Non-negative integer.
    
    Returns:
        int: Result of the Ackermann function for the given inputs.
    """
    if m == 0:
        # Base case: if m is 0, return n + 1
        return n + 1
    elif n == 0:
        # Recursive case: if n is 0, call the Ackermann function with m - 1 and 1
        return ackermann(m - 1, 1)
    else:
        # Recursive case: call the Ackermann function with m - 1 and the result of the Ackermann function with m and n - 1      
        return ackermann(m - 1, ackermann(m, n - 1))

def main():
    """
    Main function to demonstrate the Ackermann function with an example.
    """
    m = 3
    n = 4
    result = ackermann(m, n)
    print(f"Ackermann({m}, {n}) = {result}")

# Call the main function
if __name__ == "__main__":
    main()
