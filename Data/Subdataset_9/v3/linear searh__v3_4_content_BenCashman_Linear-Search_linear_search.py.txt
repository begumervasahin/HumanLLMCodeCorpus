# Define a function to find a target value in the list
def findValue(args, target):
    for index, value in enumerate(args):
        if value == target:
            print(f"The target element {target} was found at index {index}")
            return True

    # If the target value is not found, print a message
    print("Value Not Found")
    return False

# Demonstrate the functionality by searching for specific values

print("We know 9999999 isn't in the list")
