import os
import pickle
import argparse
import time

class Node:
    """
    Represents a node in the Huffman tree.
    """

    def __init__(self, char=None):
        """
        Initializes a node with the given character.
        """
        self.char = char
        self.left = None
        self.right = None

class HuffmanDecoder:
    """
    Class for decompressing Huffman-coded files.
    """

    def __init__(self, file):
        """
        Initializes the HuffmanDecoder with the file to decompress.
        """
        self.file_name = os.path.splitext(file)[0]
        self.text_code, self.text_string = self._read_compressed_file()
        self.root = self._build_huffman_tree()
        self.final_text = self._decompress_text()
        print('Decompression completed:', len(self.final_text), 'characters')

    def _read_compressed_file(self):
        """
        Load the pickle and binary files and convert the binary array into a string.
        Returns the word-frequency dictionary and the compressed code string.
        """
        # Load word-frequency dictionary from a pickle file
        with open(self.file_name + '-symbol-model.pkl', 'rb') as f:
            text_code = pickle.load(f)

        # Load compressed code from a binary file
        with open(self.file_name + '.bin', 'rb') as f:
            padding = text_code.pop('padding_length')
            text_bin = f.read()

        # Convert binary array into a string
        text_string = ''.join(format(byte, '08b') for byte in text_bin)[:-padding]
        return text_code, text_string

    def _build_huffman_tree(self):
        """
        Rebuilds the Huffman tree from the word-frequency dictionary.
        Returns the root of the Huffman tree.
        """
        root = Node()
        for char, code in self.text_code.items():
            node = root
            for bit in code:
                if bit == '0':
                    if node.left is None:
                        node.left = Node()
                    node = node.left
                else:
                    if node.right is None:
                        node.right = Node()
                    node = node.right
            node.char = char
        return root

    def _decompress_text(self):
        """
        Decompresses the text by traversing the Huffman tree.
        Returns the decompressed text.
        """
        final_text = ''
        current_node = self.root
        for bit in self.text_string:
            current_node = current_node.left if bit == '0' else current_node.right
            if current_node.char:
                final_text += current_node.char
                current_node = self.root

        # Save decompressed text to a file
        output_file = self.file_name + "-decompressed.txt"
        with open(output_file, 'w', encoding='utf-8', newline='\n') as f:
            f.write(final_text)
        return final_text

if __name__ == '__main__':
    # Parse command-line arguments
    parser = argparse.ArgumentParser()
    parser.add_argument('infile', type=str, help="Input file to decompress (with or without extension)")
    args = parser.parse_args()

    # Decompress the input file
    start_time = time.time()
    decompress = HuffmanDecoder(args.infile)
    end_time = time.time()
    print("Decoding time:", end_time - start_time)
