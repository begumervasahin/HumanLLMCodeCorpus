import pandas as pd

# Read the Excel files into DataFrames
altname_df = pd.read_excel("C:/Users/IMART/Downloads/Altname_data.xlsx", sheet_name="Export Worksheet")
subcat_df = pd.read_excel("C:/Users/IMART/Downloads/Subcat_super_PMCAT.xlsx", sheet_name="Export Worksheet")
wordfreq_df = pd.read_excel("C:/Users/IMART/Downloads/wordfrequencyfinal.xlsx", sheet_name="wordfrequencyfinal")

# Convert strings to lowercase for comparison
wordfreq_df["HIGH FREQUENCY WORDS"] = wordfreq_df["HIGH FREQUENCY WORDS"].str.lower()
subcat_df["GLCAT_MCAT_NAME"] = subcat_df["GLCAT_MCAT_NAME"].str.lower()
altname_df["GLCAT_ALTMCAT_NAME"] = altname_df["GLCAT_ALTMCAT_NAME"].str.lower()

# Create dictionaries for efficient lookup
wordfreq_dict = wordfreq_df.set_index('SUBCAT ID')['HIGH FREQUENCY WORDS'].to_dict()
merged_dict = subcat_df.merge(altname_df, left_on='SUBCAT_ID', right_on='FK_GLCAT_CAT_ID', how='inner')
merged_dict = merged_dict[['SUBCAT_ID', 'GLCAT_MCAT_NAME', 'GLCAT_ALTMCAT_NAME']].set_index('SUBCAT_ID').to_dict('index')

# Function to process alternative names and remove stopwords
def process_alt_names(row):
    subcat_id = row['FK_GLCAT_CAT_ID']
    alt_name = row['GLCAT_ALTMCAT_NAME']
    
    if subcat_id in merged_dict:
        stopwords = wordfreq_dict.get(subcat_id, merged_dict[subcat_id]['GLCAT_MCAT_NAME'])
        alt_name_split = alt_name.split(" ")
        attribute = " ".join(word for word in alt_name_split if word not in stopwords.split(" "))
        return attribute.strip()
    else:
        return alt_name.strip()

# Apply the function to process alternative names
altname_df['Attribute'] = altname_df.apply(process_alt_names, axis=1)

# Save the processed data to a CSV file
altname_df.to_csv("stopwordsremovedaltnames.csv", index=False)
