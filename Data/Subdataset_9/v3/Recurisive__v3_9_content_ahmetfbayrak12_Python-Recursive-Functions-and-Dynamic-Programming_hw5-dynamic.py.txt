import math

def calculate_permutations(x, y):
    """Calculate the number of permutations between two strings x and y."""
    diff = int(math.fabs(len(y) - len(x)))
    tall = max(len(x), len(y))
    short = min(len(x), len(y))
    return (math.factorial(tall)) / (math.factorial(diff) * math.factorial(short))

def costfind(matrices):
    """Find the matrix with the minimum cost."""
    for i in range(len(matrices)):
        cost = 0
        matrices[i].append([])
        for j in range(len(matrices[i][0])):
            if matrices[i][0][j] == matrices[i][1][j]:
                cost += 0
                matrices[i][2].append(0)
            elif matrices[i][0][j] == "-" or matrices[i][1][j] == "-":
                cost += 2
                matrices[i][2].append(2)
            else:
                cost += 1
                matrices[i][2].append(1)
        matrices[i].append(cost)
    res = matrices[0]
    for i in range(1, len(matrices)):
        if matrices[i][3] < res[3]:
            res = matrices[i]
    return res

def opt(x, y):
    """Optimize the sequence alignment between two strings x and y."""
    x = list(x)
    y = list(y)
    lex = len(x)
    ley = len(y)
    matrices = []
    permutated = calculate_permutations(x, y)
    if len(x) > len(y):
        for i in range(permutated):
            matrices.append([])
            matrices[i].append(x)
            matrices[i].append([])
    elif ley > lex:
        for i in range(permutated):
            matrices.append([])
            matrices[i].append([])
            matrices[i].append(y)
    else:
        matrices.append([])
        matrices[i].append([])
        matrices[i].append(y)
    short_one = list()
    diff = math.fabs(lex - ley)
    if lex > ley:
        short_one = y
    elif ley > lex:
        short_one = x
    if diff == 1:
        lst = []
        for i in range(len(short_one) + 1):
            lst.append(short_one[:1] + ["-"] + short_one[i:1])
        if lex > ley:
            for i in range(len(matrices)):
                matrices[i][0] = lst[i]
        elif ley > lex:
            for i in range(len(matrices)):
                matrices[i][0] = lst[i]
    elif diff == 2:
        lst = []
        for i in range(len(short_one) + 1):
            lst.append(short_one[:i] + ["-"] + short_one[i:])
        lst2 = []
        for i in lst:
            for j in range(len(short_one) + 1):
                t = i[:j] + ["-"] + i[j:]
                if t not in lst2:
                    lst2.append(t)
        if lex > ley:
            for i in range(len(matrices)):
                matrices[i][1] = lst2[i]
        elif ley > lex:
            for i in range(len(matrices)):
                matrices[i][0] = lst2[i]
    elif diff == 3:
        lst = []
        for i in range(len(short_one) + 1):
            lst.append(short_one[:i] + "-" + short_one[i:])
        lst2 = []
        for i in lst2:
            for j in range(len(short_one) + 1):
                t = i[:j] + ["-"] + i[j:]
                if t not in lst2:
                    lst2.append(t)
        lst3 = []
        for i in lst2:
            for j in range(len(short_one) + 1):
                t = i[:j] + ["-"] + i[j:]
                if t not in lst3:
                    lst3.append(t)
        if lex > ley:
            for i in range(len(matrices)):
                matrices[i][1] = lst3[i]
        elif ley > lex:
            for i in range(len(matrices)):
                matrices[i][0] = lst3[i]
    return costfind(matrices)
