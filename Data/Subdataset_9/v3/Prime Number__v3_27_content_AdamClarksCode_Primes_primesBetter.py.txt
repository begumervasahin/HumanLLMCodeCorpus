def sieve_of_eratosthenes(n):
    """
    Generate prime numbers up to 'n' using the Sieve of Eratosthenes algorithm.

    Parameters:
        n (int): The upper limit for generating prime numbers.

    Returns:
        list: List of prime numbers up to 'n'.
    """
    # Initialize a boolean list representing whether each number is prime
    sieve = [True] * n

    # Loop through odd numbers starting from 3 up to the square root of 'n'
    for i in range(3, int(n ** 0.5) + 1, 2):
        if sieve[i]:
            # Mark multiples of 'i' as not prime
            sieve[i * i::2 * i] = [False] * ((n - i * i - 1) // (2 * i) + 1)

    # Return the list of prime numbers
    return [2] + [i for i in range(3, n, 2) if sieve[i]]

def optimized_sieve_of_eratosthenes(n):
    """
    Generate prime numbers up to 'n' using an optimized version of the Sieve of Eratosthenes algorithm.

    Parameters:
        n (int): The upper limit for generating prime numbers.

    Returns:
        list: List of prime numbers up to 'n'.
    """
    # Initialize a boolean list representing whether each odd number is prime
    sieve = [True] * (n // 2)

    # Loop through odd numbers starting from 3 up to the square root of 'n'
    for i in range(3, int(n ** 0.5) + 1, 2):
        if sieve[i // 2]:
            # Mark multiples of 'i' as not prime
            sieve[i * i // 2::i] = [False] * ((n - i * i - 1) // (2 * i) + 1)

    # Return the list of prime numbers
    return [2] + [2 * i + 1 for i in range(1, n // 2) if sieve[i]]

    list_of_primes = optimized_sieve_of_eratosthenes(limit)
