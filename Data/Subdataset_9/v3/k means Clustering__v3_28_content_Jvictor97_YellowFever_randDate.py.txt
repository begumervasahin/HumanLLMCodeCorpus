import random
import time
from datetime import timedelta, datetime

def generate_random_date(start_date, end_date, proportion):
    """
    Generate a random date within the specified range.

    Args:
        start_date (str): The start date of the range in '%Y-%m-%d' format.
        end_date (str): The end date of the range in '%Y-%m-%d' format.
        proportion (float): A value between 0 and 1 representing the position within the range.

    Returns:
        str: A random date within the specified range.
    """
    start_time = time.mktime(time.strptime(start_date, '%Y-%m-%d'))
    end_time = time.mktime(time.strptime(end_date, '%Y-%m-%d'))   
    random_time = start_time + proportion * (end_time - start_time)
    return time.strftime('%Y-%m-%d', time.localtime(random_time)) 

def generate_random_hot_date():
    """
    Generate a random date within the hot season (January to June).

    Returns:
        str: A random date within the hot season.
    """
    return generate_random_date("2018-01-01", "2018-06-30", random.random())

def generate_random_cold_date():
    """
    Generate a random date within the cold season (July to December).

    Returns:
        str: A random date within the cold season.
    """
    return generate_random_date("2018-07-01", "2018-12-31", random.random())

def generate_random_date_within_delta(start_date, delta_days):    
    """
    Generate a random date within a custom range specified by a start date and a delta in days.

    Args:
        start_date (str): The start date of the range in '%Y-%m-%d' format.
        delta_days (int): The number of days to span the range.   

    Returns:
        str: A random date within the custom range.
    """
    start_date_obj = datetime.strptime(start_date, '%Y-%m-%d').date()
    end_date = str(start_date_obj + timedelta(days=delta_days))   
    return generate_random_date(start_date, end_date, random.random())

# Example usage:
print("Random hot date:", generate_random_hot_date())
print("Random cold date:", generate_random_cold_date())
print("Random date with custom range and delta:", generate_random_date_within_delta("2023-01-01", 365))
