def str_xor(a, b):
    """Perform XOR operation on two strings and return the result."""
    return "".join([chr(ord(x) ^ ord(y)) for x, y in zip(a, b)])

def read_user_input(prompt="Enter input: "):
    """Read user input and return it."""
    return input(prompt)

def print_intercepted_data(data):
    """Print the intercepted data."""
    print("Intercepted data:", data)
