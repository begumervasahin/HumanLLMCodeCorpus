def is_prime(n, primes):
    """
    Check if a number is prime by verifying its divisibility with primes less than itself.

    Parameters:
        n (int): The number to check for primality.
        primes (list): List of prime numbers found so far.

    Returns:
        bool: True if n is prime, False otherwise.
    """
    for prime in primes:
        if prime * prime > n:
            break
        if n % prime == 0:
            return False
    return True

def find_primes(limit):
    """
    Find prime numbers up to a specified limit.

    Parameters:
        limit (int): The upper limit to search for prime numbers.

    Returns:
        list: List of prime numbers found within the limit.
    """
    primes = [2]  # Initialize with the first prime number
    for n in range(3, limit + 1, 2):  # Start from 3 and increment by 2 to skip even numbers
        if is_prime(n, primes):
            primes.append(n)
    return primes

def main():
    limit = 3001
    primes = find_primes(limit)
    for prime in primes:
        print(prime, "is prime")

if __name__ == "__main__":
    main()
