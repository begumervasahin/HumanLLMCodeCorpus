import _pickle as cPickle
import string
import secrets

class OneTimePad:
    DEFAULT_ALPHABET = string.digits + string.ascii_letters + string.punctuation + " "
    DEFAULT_MESSAGE_LENGTH = 64
    DEFAULT_KEY_NUMBER = 0xffff
    DEFAULT_FILE = "key.dict"

    def __init__(self, alphabet=None, message_length=None, key_number=None, file=None, outqueue=None):
        self.alphabet = alphabet or self.DEFAULT_ALPHABET
        self.message_length = message_length or self.DEFAULT_MESSAGE_LENGTH
        self.max_keys = key_number or self.DEFAULT_KEY_NUMBER
        self.file = file or self.DEFAULT_FILE
        if outqueue:
            outqueue.put("Starting")
        self.secure_random = secrets.SystemRandom()
        try:
            with open(self.file, 'rb') as f:
                self.load_key_settings(f)
        except FileNotFoundError:
            self.generate_keys(outqueue)
        print("Keys remaining: {}".format(len(self.key_dict) - 1))
        if outqueue:
            outqueue.put("Stopped")

    def load_key_settings(self, f):
        self.key_dict = cPickle.load(f)
        print("Loaded settings from file")
        settings = self.key_dict[0]
        self.alphabet, self.message_length, self.max_keys = settings

    def generate_keys(self, outqueue):
        print("Generating new key dictionary with {} keys".format(self.max_keys))
        print("This may take a few minutes")
        keys = {0: (self.alphabet, self.message_length, self.max_keys)}
        for i in range(self.max_keys):
            keys[i + 1] = "".join(self.secure_random.choice(self.alphabet) for _ in range(self.message_length))
            if i % int(self.max_keys / 100) == 0 and outqueue:
                outqueue.put("Step")
        with open(self.file, 'wb') as f:
            cPickle.dump(keys, f, -1)
        self.key_dict = keys
        print("Finished")

    def get_key_len(self, x):
        return len(hex(x)) - 2

    def encode(self, message):
        try:
            prefix, key = self.pop_random_key()
            prefix = self.format_key_prefix(prefix)
            message = self.pad_message(message)
            encoded_message = self.encode_message(message, key)
            return prefix + encoded_message, True
        except (KeyError, ValueError) as err:
            return f"Unable to encode message: {err}", False

    def pop_random_key(self):
        prefix, key = self.secure_random.choice(list(self.key_dict.items())[1:])
        self.key_dict.pop(prefix)
        with open(self.file, 'wb') as f:
            cPickle.dump(self.key_dict, f, -1)
        return prefix, key

    def format_key_prefix(self, prefix):
        keyprefix_len = self.get_key_len(self.max_keys)
        return "{0:0{1}x}".format(prefix, keyprefix_len)

    def pad_message(self, message):
        return message.ljust(self.message_length)

    def encode_message(self, message, key):
        return ''.join(self.alphabet[(self.alphabet.index(message[i]) + self.alphabet.index(key[i])) % len(self.alphabet)] for i in range(self.message_length))

    def decode(self, encoded_data):
        try:
            key_prefix, encoded_message = encoded_data[:self.get_key_len(self.max_keys)], encoded_data[self.get_key_len(self.max_keys):]
            key = self.pop_key_by_prefix(int(key_prefix, 16))
            return self.decode_message(encoded_message, key)
        except KeyError:
            return "Unable to decode data"

    def pop_key_by_prefix(self, prefix):
        key = self.key_dict[prefix]
        self.key_dict.pop(prefix)
        with open(self.file, 'wb') as f:
            cPickle.dump(self.key_dict, f, -1)
        return key

    def decode_message(self, encoded_data, key):
        return ''.join(self.alphabet[(self.alphabet.index(encoded_data[i]) - self.alphabet.index(key[i])) % len(self.alphabet)] for i in range(self.message_length))

# Example usage:
if __name__ == "__main__":
    otp = OneTimePad()
    message = "Hello, World!"
    encoded_message, _ = otp.encode(message)
    print("Encoded message:", encoded_message)
    decoded_message = otp.decode(encoded_message)
    print("Decoded message:", decoded_message)
