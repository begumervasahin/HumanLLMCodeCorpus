def make_env(env_id, seed, rank, log_dir):
    """
    Create a Gym environment with appropriate wrappers.

    Args:
        env_id (str): The environment ID.
        seed (int): The seed for environment randomness.
        rank (int): The rank of the environment (for multiprocessing).
        log_dir (str): The directory for logging environment data.

    Returns:
        function: A function that creates the environment.
    """
    def _thunk():
        # Create Gym environment
        env = gym.make(env_id)
        is_atari = hasattr(gym.envs, 'atari') and isinstance(env.unwrapped, gym.envs.atari.atari_env.AtariEnv)

        # Apply Atari specific wrappers if applicable
        if is_atari:
            env = make_atari(env_id)

        env.seed(seed + rank)  # Set environment seed
        if log_dir is not None:
            # Monitor environment if log directory provided
            env = bench.Monitor(env, os.path.join(log_dir, str(rank)))

        # Apply DeepMind wrapper if Atari environment
        if is_atari:
            env = wrap_deepmind(env)

        obs_shape = env.observation_space.shape

        # Transpose observation shape for PyTorch compatibility if necessary
        if len(obs_shape) == 3 and obs_shape[2] in [1, 3]:
            env = WrapPyTorch(env)

        return env

    return _thunk

class WrapPyTorch(gym.ObservationWrapper):
    """
    Wrapper class to transpose observations for PyTorch compatibility.
    """
    def __init__(self, env=None):
        super(WrapPyTorch, self).__init__(env)
        obs_shape = self.observation_space.shape
        self.observation_space = Box(
            self.observation_space.low[0,0,0],
            self.observation_space.high[0,0,0],
            [obs_shape[2], obs_shape[1], obs_shape[0]]
        )

    def observation(self, observation):
        return observation.transpose(2, 0, 1)
