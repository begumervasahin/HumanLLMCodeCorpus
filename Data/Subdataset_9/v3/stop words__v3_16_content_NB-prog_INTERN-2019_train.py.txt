import os
import pickle
import torch
from torch.optim import Adam
from torch.utils.data import DataLoader
from tqdm import tqdm
from DataHandler import TextLoader, SkipGramData, write_embedding
from Model import SkipGram
import config

# Ensure necessary directories exist
def ensure_directories_exist():
    for directory in ['model', 'embeddings', 'data']:
        if not os.path.isdir(directory):
            os.mkdir(directory)

# Load or create corpus and vocabulary based on configuration
def load_or_create_corpus():
    if config.cont_traning:
        with open(os.path.join('data', 'corpus.dat'), 'rb') as f:
            corpus = pickle.load(f)
        with open(os.path.join('data', 'word_count.dat'), 'rb') as f:
            word_count = pickle.load(f)
        with open(os.path.join('data', 'word_freq.dat'), 'rb') as f:
            word_freq = pickle.load(f)
        with open(os.path.join('data', 'vocab.dat'), 'rb') as f:
            vocab = pickle.load(f)
        with open(os.path.join('data', 'idx2word.dat'), 'rb') as f:
            idx2word = pickle.load(f)
        with open(os.path.join('data', 'word2idx.dat'), 'rb') as f:
            word2idx = pickle.load(f)
        print("\nContinuing training from previous state for {} epochs ...".format(config.epochs))
    else:
        text_loader = TextLoader(corpus_file=config.corpus_file, max_vocab_size=int(config.max_vocab_size),
                                 max_corpus_size=int(config.max_corpus_size), fw_subsampling_th=config.fw_subsampling_th)
        corpus, idx2word, word2idx, vocab, word_count, word_freq = text_loader.load_corpus()
        # Save loaded data to files
        with open(os.path.join('data', 'corpus.dat'), 'wb') as f:
            pickle.dump(corpus, f, protocol=pickle.HIGHEST_PROTOCOL)
        with open(os.path.join('data', 'word_count.dat'), 'wb') as f:
            pickle.dump(word_count, f, protocol=pickle.HIGHEST_PROTOCOL)
        with open(os.path.join('data', 'word_freq.dat'), 'wb') as f:
            pickle.dump(word_freq, f, protocol=pickle.HIGHEST_PROTOCOL)
        with open(os.path.join('data', 'vocab.dat'), 'wb') as f:
            pickle.dump(vocab, f, protocol=pickle.HIGHEST_PROTOCOL)
        with open(os.path.join('data', 'idx2word.dat'), 'wb') as f:
            pickle.dump(idx2word, f, protocol=pickle.HIGHEST_PROTOCOL)
        with open(os.path.join('data', 'word2idx.dat'), 'wb') as f:
            pickle.dump(word2idx, f, protocol=pickle.HIGHEST_PROTOCOL)
        print("\nTraining for {} epochs ...".format(config.epochs))
    return corpus, idx2word, word2idx, vocab, word_count, word_freq

# Initialize SkipGram model
def initialize_skipgram_model(idx2word, word2idx, word_freq):
    skipgram_model = SkipGram(vocab_size=len(idx2word), embedding_dims=config.embedding_dims,
                              neg_samples=config.neg_samples, word_freq=word_freq, padding_idx=word2idx[pad_word])
    device = 'cuda' if torch.cuda.is_available() else 'cpu'
    skipgram_model = skipgram_model.to(device=device)
    optimizer = Adam(skipgram_model.parameters())
    return skipgram_model, optimizer

# Main training loop
def train_skipgram_model(skipgram_model, optimizer, dataset, idx2word, word2idx):
    dataloader = DataLoader(dataset, batch_size=config.mini_batch_size, shuffle=True)
    for epoch in range(1, config.epochs + 1):
        progress_bar = tqdm(dataloader)
        progress_bar.set_description("[Epoch {}]".format(epoch))
        for targets, contexts in progress_bar:
            loss = skipgram_model(targets.to(dtype=torch.long, device=device),
                                   contexts.to(dtype=torch.long, device=device))
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()
            progress_bar.set_postfix(loss=loss.item())

        # Save model and optimizer state after each epoch
        torch.save(skipgram_model.state_dict(), os.path.join('model', 'model.pt'))
        torch.save(optimizer.state_dict(), os.path.join('model', 'optimizer.pt'))

        # Write embeddings to files after each epoch
        input_embeddings = skipgram_model.input_embedding.weight.data.cpu().numpy()
        output_embeddings = skipgram_model.output_embedding.weight.data.cpu().numpy()
        write_embedding(os.path.join('embeddings', 'input_embeddings.txt'), idx2word, word2idx, input_embeddings)
        write_embedding(os.path.join('embeddings', 'output_embeddings.txt'), idx2word, word2idx, output_embeddings)

if __name__ == "__main__":
    ensure_directories_exist()
    corpus, idx2word, word2idx, vocab, word_count, word_freq = load_or_create_corpus()
    skipgram_model, optimizer = initialize_skipgram_model(idx2word, word2idx, word_freq)
    dataset = SkipGramData(corpus, word2idx, window=config.window_size)
    train_skipgram_model(skipgram_model, optimizer, dataset, idx2word, word2idx)
