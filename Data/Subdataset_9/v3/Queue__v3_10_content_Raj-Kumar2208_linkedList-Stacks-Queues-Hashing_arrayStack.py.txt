class StackArray:
    def __init__(self, limit=10):
        """
        Initialize a stack with a specified limit.

        Args:
            limit (int, optional): The maximum capacity of the stack. Defaults to 10.
        """
        self.limit = limit
        self.stack = []

    def push(self, data):
        """
        Pushes an element onto the stack.

        Args:
            data: The data to be pushed onto the stack.

        Raises:
            ValueError: If the stack is already full (overflow).
        """
        if len(self.stack) >= self.limit:
            raise ValueError("Stack Overflow")
        else:
            self.stack.append(data)

    def is_empty(self):
        """
        Checks if the stack is empty.

        Returns:
            bool: True if the stack is empty, False otherwise.
        """
        return len(self.stack) == 0

    def pop(self):
        """
        Removes and returns the top element from the stack.

        Returns:
            The top element of the stack.

        Raises:
            ValueError: If the stack is empty (underflow).
        """
        if self.is_empty():
            raise ValueError("Stack Underflow")
        else:
            return self.stack.pop()

    def peek(self):
        """
        Returns the top element of the stack without removing it.

        Returns:
            The top element of the stack.

        Raises:
            ValueError: If the stack is empty (underflow).
        """
        if self.is_empty():
            raise ValueError("Stack Underflow")
        else:
            return self.stack[-1]

    def size(self):
        """
        Returns the number of elements currently in the stack.

        Returns:
            int: The number of elements in the stack.
        """
        return len(self.stack)

    def print_stack(self):
        """
        Prints the elements of the stack.
        """
        print(self.stack)
