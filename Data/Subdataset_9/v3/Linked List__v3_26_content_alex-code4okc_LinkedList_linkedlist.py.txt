class LinkedList:
    class Node:
        def __init__(self, item, prev_node=None, next_node=None):
            self.item = item
            self.prev_node = prev_node
            self.next_node = next_node

    def __init__(self):
        self.head = self.Node(None)
        self.tail = self.Node(None)
        self.size = 0

    def insert_last(self, item):
        node = self.Node(item)
        if self.is_empty():
            self.head.next_node = node
            self.tail.prev_node = node
        else:
            node.prev_node = self.tail.prev_node
            self.tail.prev_node.next_node = node
            self.tail.prev_node = node
        self.size += 1

    def insert_first(self, item):
        node = self.Node(item)
        if self.is_empty():
            self.head.next_node = node
            self.tail.prev_node = node
        else:
            node.next_node = self.head.next_node
            self.head.next_node.prev_node = node
            self.head.next_node = node
        self.size += 1

    def get_first(self):
        if self.is_empty():
            return "Linked List is empty!"
        return self.head.next_node.item

    def get_last(self):
        if self.is_empty():
            return "Linked List is empty!"
        return self.tail.prev_node.item

    def __len__(self):
        return self.size

    def __str__(self):
        if self.is_empty():
            return "Linked List is empty!"
        else:
            current = self.head.next_node
            items = []
            while current:
                items.append(str(current.item))
                current = current.next_node
            return ", ".join(items)

    def insert_into(self, item, index):
        if index < 0:
            index = 0
        if index >= self.size:
            self.insert_last(item)
        elif index == 0:
            self.insert_first(item)
        else:
            node = self.Node(item)
            current = self.head.next_node
            for _ in range(index):
                current = current.next_node
            node.prev_node = current.prev_node
            node.next_node = current
            current.prev_node.next_node = node
            current.prev_node = node
            self.size += 1

    def remove_first(self):
        if self.is_empty():
            print("List is empty! No elements to remove.")
        else:
            self.head.next_node = self.head.next_node.next_node
            if self.head.next_node:
                self.head.next_node.prev_node = None
            else:
                self.tail.prev_node = None
            self.size -= 1

    def remove_last(self):
        if self.is_empty():
            print("List is empty! No elements to remove.")
        else:
            self.tail.prev_node = self.tail.prev_node.prev_node
            if self.tail.prev_node:
                self.tail.prev_node.next_node = None
            else:
                self.head.next_node = None
            self.size -= 1

    def is_empty(self):
        return self.size == 0
