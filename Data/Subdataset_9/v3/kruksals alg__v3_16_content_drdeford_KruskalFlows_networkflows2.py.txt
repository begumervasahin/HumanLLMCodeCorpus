import numpy as np
import matplotlib.pyplot as plt
import math
import time
from operator import itemgetter

def main():
    # Example graph
    a = [[0, 1, 3, 4, 0], [1, 0, 0, 2, 0], [3, 0, 0, 0, 5], [4, 2, 0, 0, 0], [0, 0, 5, 0, 0]]
    MST_visualization(a)

def MST_visualization(adjacency_matrix):
    # Calculate and visualize MST using Kruskal's, Prim's, and Sollin's algorithms
    graph_plot(adjacency_matrix)
    kruskal_time, kruskal_cost = kruskal_algorithm(adjacency_matrix)
    prim_time, prim_cost = prim_algorithm(adjacency_matrix)
    sollin_time, sollin_cost = sollin_algorithm(adjacency_matrix)
    print('Kruskal Time:', kruskal_time, '\nKruskal Cost:', kruskal_cost,
          '\nPrim Time:', prim_time, '\nPrim Cost:', prim_cost,
          '\nSollin Time:', sollin_time, '\nSollin Cost:', sollin_cost)
    plt.show()

def graph_plot(adjacency_matrix):
    # Plot the original graph
    size = len(adjacency_matrix)
    x = []
    y = []
    theta = 2 * math.pi / size
    plt.figure(1)
    plt.title('Original Graph')
    for i in range(size):
        x.append(size * math.cos(i * theta))
        y.append(size * math.sin(i * theta))
        plt.text(x[i], y[i], i + 1, fontsize=15, color='green')
    for i in range(size):
        for j in range(i, size):
            if adjacency_matrix[i][j] != 0:
                plt.plot([x[i], x[j]], [y[i], y[j]], 'b-', linewidth=3)
                plt.text((x[i] + x[j]) / 2, (y[i] + y[j]) / 2, adjacency_matrix[i][j], fontsize=15, color='red')
    plt.scatter(x, y, s=50)

def kruskal_algorithm(adjacency_matrix):
    # Kruskal's algorithm for finding the MST
    start_time = time.time()
    edges = get_sorted_edges(adjacency_matrix)
    min_spanning_tree = kruskal_mst(adjacency_matrix, edges)
    k_time = time.time() - start_time
    return k_time, calculate_cost(adjacency_matrix, min_spanning_tree)

def get_sorted_edges(adjacency_matrix):
    edges = []
    for i in range(len(adjacency_matrix)):
        for j in range(i, len(adjacency_matrix)):
            if adjacency_matrix[i][j] != 0:
                edges.append((i, j, adjacency_matrix[i][j]))
    return sorted(edges, key=itemgetter(2))

def kruskal_mst(adjacency_matrix, edges):
    min_spanning_tree = []
    forest = [{i} for i in range(len(adjacency_matrix))]
    for edge in edges:
        u, v, weight = edge
        if u != v and find_set(u, forest) != find_set(v, forest):
            min_spanning_tree.append(edge)
            union(u, v, forest)
    return min_spanning_tree

def find_set(vertex, forest):
    for i, subset in enumerate(forest):
        if vertex in subset:
            return i

def union(u, v, forest):
    u_set = find_set(u, forest)
    v_set = find_set(v, forest)
    if u_set != v_set:
        forest[u_set] |= forest[v_set]
        del forest[v_set]

def calculate_cost(adjacency_matrix, edges):
    return sum(weight for _, _, weight in edges)

def prim_algorithm(adjacency_matrix):
    # Prim's algorithm for finding the MST
    start_time = time.time()
    min_spanning_tree = prim_mst(adjacency_matrix)
    p_time = time.time() - start_time
    return p_time, calculate_cost(adjacency_matrix, min_spanning_tree)

def prim_mst(adjacency_matrix):
    min_spanning_tree = []
    num_vertices = len(adjacency_matrix)
    in_tree = [False] * num_vertices
    in_tree[0] = True
    while len(min_spanning_tree) < num_vertices - 1:
        min_edge = find_min_edge(adjacency_matrix, in_tree)
        min_spanning_tree.append(min_edge)
        in_tree[min_edge[1]] = True
    return min_spanning_tree

def find_min_edge(adjacency_matrix, in_tree):
    min_weight = float('inf')
    min_edge = None
    num_vertices = len(adjacency_matrix)
    for u in range(num_vertices):
        if in_tree[u]:
            for v in range(num_vertices):
                if not in_tree[v] and adjacency_matrix[u][v] != 0 and adjacency_matrix[u][v] < min_weight:
                    min_weight = adjacency_matrix[u][v]
                    min_edge = (u, v, min_weight)
    return min_edge

def sollin_algorithm(adjacency_matrix):
    # Sollin's algorithm for finding the MST
    start_time = time.time()
    min_spanning_tree = sollin_mst(adjacency_matrix)
    s_time = time.time() - start_time
    return s_time, calculate_cost(adjacency_matrix, min_spanning_tree)

def sollin_mst(adjacency_matrix):
    min_spanning_tree = []
    num_vertices = len(adjacency_matrix)
    forest = [{i} for i in range(num_vertices)]
    while len(min_spanning_tree) < num_vertices - 1:
        for i in range(num_vertices):
            min_edge = find_min_edge_sollin(adjacency_matrix, forest[i])
            if min_edge is not None and min_edge not in min_spanning_tree:
                min_spanning_tree.append(min_edge)
                u, v, _ = min_edge
                forest[u] |= forest[v]
                del forest[v]
    return min_spanning_tree

def find_min_edge_sollin(adjacency_matrix, subset):
    min_weight = float('inf')
    min_edge = None
    for u in subset:
        for v in range(len(adjacency_matrix)):
            if v not in subset and adjacency_matrix[u][v] != 0 and adjacency_matrix[u][v] < min_weight:
                min_weight = adjacency_matrix[u][v]
                min_edge = (u, v, min_weight)
    return min_edge
