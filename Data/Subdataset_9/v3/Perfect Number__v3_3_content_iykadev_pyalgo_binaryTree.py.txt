class BinaryTree:
    def __init__(self, value):
        self.value = value
        self.left = None
        self.right = None

    def insert_left(self, value):
        if self.left is None:
            self.left = BinaryTree(value)
        else:
            new_node = BinaryTree(value)
            new_node.left = self.left
            self.left = new_node

    def insert_right(self, value):
        if self.right is None:
            self.right = BinaryTree(value)
        else:
            new_node = BinaryTree(value)
            new_node.right = self.right
            self.right = new_node

    def get_root_value(self):
        return self.value

    def set_root_value(self, value):
        self.value = value

    def get_left_child(self):
        return self.left

    def get_right_child(self):
        return self.right


# Example usage
tree = BinaryTree(3)
tree.insert_left(4)
tree.insert_left(5)
tree.insert_right(6)
tree.insert_right(7)

left_child = tree.get_left_child()
print("Left child of root:", left_child.get_root_value())
left_child.set_root_value(9)
print("Updated tree:", tree.get_root_value())

left_child.insert_left(11)
print("Updated tree after inserting 11 as left child of 9:", tree.get_root_value())

right_right_child = tree.get_right_child().get_right_child()
print("Right child of right child of root:", right_right_child.get_root_value())
