import sys
import time
from datetime import date
from itertools import count, islice
from math import sqrt
from functools import reduce
from gephistreamer import graph, streamer
import Ut

class Utx:
    """
    Represents an object that generates the Phi tree and constructs it in Gephi.
    """

    def __init__(self, seed=None):
        """
        Initializes the Utx object with optional seed.
        """
        self.ut = Ut.Ut()
        self.added_elements = []
        try:
            self.stream = streamer.Streamer(streamer.GephiWS())
            print("Connected to Gephistreamer. Add a reference node in Gephi before adding points.")
            print("Keep this reference node in range before adding points. Connect in master mode.")
        except:
            print('Failed to connect to Gephistreamer.')

        if isinstance(seed, Ut.Ut):
            self.sow(seed.manifold)

    def __repr__(self):
        """
        Returns a string representation of the Utx object.
        """
        return 'Utx : {} => {}'.format(self.tree, self.manifold.keys())

    def out(self, elements=None):
        """
        Streams out the added element (node, edge) or unpacks the list of added elements.
        """
        if not elements:
            return 'Nothing to add'
        
        if isinstance(elements, (graph.Node, graph.Edge)):
            self.stream.add_node(elements) if isinstance(elements, graph.Node) else self.stream.add_edge(elements)
        elif isinstance(elements, (list, tuple)):
            for elem in elements:
                self.out(elem)
        elif isinstance(elements, dict):
            for val in elements.values():
                self.out(val)
        elif isinstance(elements, (int, str)):
            return 'Object cannot be immediately graphed'
        else:
            return elements

    def clear_nodes(self):
        """
        Clears all nodes and edges from the graph.
        """
        for values in self.manifold.values():
            for item in values:
                self.stream.delete_node(item) if isinstance(item, graph.Node) else self.stream.delete_edge(item)
            self.manifold.clear()

    def connect(self, num):
        """
        Connects a number to its factors.
        """
        elements = []
        if not self.ut.tree:
            self.ut.tree_me(num)
            
        if not self.ut.check_tree(num):
            node = graph.Node(num, type='Factor')
            elements.append(node)
        else:
            node = graph.Node(num, type='Prime')
            elements.append(node)
        
        for factor in self.ut.tree:
            if num % factor == 0 and num / factor != 1:
                prime_factor_node = graph.Node(factor, type="Prime")
                elements.extend([prime_factor_node, graph.Node(num / factor, type="Factor")])
                elements.extend([
                    graph.Edge(prime_factor_node, node, type="Product"),
                    graph.Edge(prime_factor_node, graph.Node(num / factor, type="Factor"), type="Product")
                ])
            elif num % factor == 0 and num / factor == 1 and num != 1:
                prime_factor_node = graph.Node(num / factor, type="Prime")
                previous_prime_node = graph.Node(self.ut.tree[self.ut.tree.index(factor) - 1], type="Prime")
                elements.extend([prime_factor_node, previous_prime_node, graph.Edge(prime_factor_node, previous_prime_node, type='PrimeLine')])
        
        if elements:
            self.out(elements)

    def split(self):
        """
        Connects each number in the list of lone numbers to its factors.
        """
        if not self.lone:
            return 'Nothing to add'
        
        idx = 0
        while idx < len(self.lone):
            if self.ut.check_tree(self.lone[idx]):
                del self.lone[idx]
            try:
                self.connect(self.lone[idx])
            except IndexError:
                message = 'Finished'
                del self.lone
                return message
            idx += 1
        self.lone = []

    def sow(self, element):
        """
        Connects the elements of a series presented by the argument element.
        """
        if isinstance(element, (int, list, tuple, dict)):
            for key in element if isinstance(element, dict) else element:
                self.connect(key)
        elif isinstance(element, str):
            print('No dictionaries for words currently')
        else:
            for key in element:
                self.connect(key)

    def flow(self, timer=None):
        """
        Continuously adds numbers to the graph, starting from 2, at a given time interval.
        """
        timer = timer or 0
        flux = 2
        while True:
            user_input = input('Enter "q" to stop the flow...')
            if user_input == 'q':
                break
            print('Adding {}...'.format(flux))
            for _ in range(flux):
                self.connect(flux)
                time.sleep(timer)
                flux += 1

    def spread(self, timer=None):
        """
        Continuously adds numbers to the graph, starting from the highest prime factor, at a given time interval.
        """
        timer = timer or 0
        max_value = self.ut.tree[-1]
        
        if not timer:
            return

        timing = time.time()
        timeout = timing + timer * max_value
        print(f'This operation will take {timeout - timing} seconds.')
        follow = input('Enter "yes" to continue: ')
        
        if follow != 'yes':
            return
        
        for val in range(max_value, 1, -1):
            self.connect(val)
            time.sleep(timer)

if __name__ == '__main__':
    utx = Utx()
    utx.flow()
