import requests
from bs4 import BeautifulSoup
import pandas as pd
import numpy as np
from time import sleep
from random import randint

# Initialize empty lists to store data for each movie
titles, years, time, imdb_ratings, metascores, votes, us_gross = ([] for _ in range(7))

# Mimic a browser request by setting headers
headers = {"Accept-Language": "en-US, en;q=0.5"}

# Generate page numbers to scrape
pages = np.arange(1, 1001, 50)

# Function to extract text from a BeautifulSoup element or return default value
def extract_text(element, class_name, default='_'):
    return element.find(class_=class_name).get_text() if element.find(class_=class_name) else default

# Loop through each page
for page in pages:
    # Construct URL for the current page
    url = f"https://www.imdb.com/search/title/?groups=top_1000&start={page}&ref_=adv_nxt"
    
    # Get the page content
    response = requests.get(url, headers=headers)
    
    # Pause for a random amount of time to avoid being blocked
    sleep(randint(2, 10))

    # Parse the HTML content
    soup = BeautifulSoup(response.text, 'html.parser')

    # Find all movie containers in the parsed HTML
    movie_div = soup.find_all('div', class_='lister-item mode-advanced')

    # Extract data from each movie container
    for container in movie_div:
        # Movie title
        titles.append(container.h3.a.text)
        
        # Release year of the movie
        years.append(container.h3.find('span', class_='lister-item-year').text)
        
        # Runtime of the movie
        time.append(extract_text(container.p, 'runtime'))
        
        # IMDb rating of the movie
        imdb_ratings.append(float(container.strong.text))
        
        # Metascore of the movie (if available)
        metascores.append(extract_text(container, 'metascore'))
        
        # Votes for the movie
        votes.append(container.find('span', attrs={'name': 'nv'})[0].text)
        
        # US Gross earnings of the movie (if available)
        us_gross.append(extract_text(container, 'nv', '-'))

# Create a DataFrame from the collected data
movies = pd.DataFrame({
    'movie': titles,
    'year': years,
    'timeMin': time,
    'imdb': imdb_ratings,
    'metascore': metascores,
    'votes': votes,
    'us_grossMillions': us_gross
})

# Clean and format the data
movies['year'] = movies['year'].str.extract('(\d+)').astype(int)
movies['timeMin'] = movies['timeMin'].str.extract('(\d+)').astype(int)
movies['votes'] = movies['votes'].str.replace(',', '').astype(int)
movies['metascore'] = pd.to_numeric(movies['metascore'], errors='coerce')
movies['us_grossMillions'] = movies['us_grossMillions'].str.replace('M', '').str.replace('$', '').astype(float)

# Save the DataFrame to a CSV file
movies.to_csv('movie.csv', index=False)

# Print the DataFrame and data types
print(movies.to_string())
print(movies.dtypes)
