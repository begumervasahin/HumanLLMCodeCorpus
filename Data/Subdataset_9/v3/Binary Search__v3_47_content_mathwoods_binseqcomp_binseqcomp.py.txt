from __future__ import print_function
import random
import timeit
import matplotlib.pyplot as plt

def sequential_search(arr, x):
    """
    Perform sequential search to find the index of a given element in an array.

    Arguments:
    arr -- the array to search in
    x -- the element to search for

    Returns:
    index -- the index of the element if found, otherwise -1
    """
    i = 0
    while i < len(arr):
        if arr[i] == x:
            return i
        i += 1
    return -1

def binary_search(arr, x):
    """
    Perform binary search to find the index of a given element in a sorted array.

    Arguments:
    arr -- the sorted array to search in
    x -- the element to search for

    Returns:
    index -- the index of the element if found, otherwise -1
    """
    l, r = 0, len(arr) - 1
    while l <= r:
        m = (l + r) // 2
        if x > arr[m]:
            l = m + 1
        elif x < arr[m]:
            r = m - 1
        else:
            return m
    return -1

def wrapper(func, *args, **kwargs):
    """
    Create a function wrapper to pass arguments to the function.

    Arguments:
    func -- the function to wrap
    *args -- positional arguments for the function
    **kwargs -- keyword arguments for the function

    Returns:
    wrapped -- the wrapped function
    """
    def wrapped():
        return func(*args, **kwargs)
    return wrapped
