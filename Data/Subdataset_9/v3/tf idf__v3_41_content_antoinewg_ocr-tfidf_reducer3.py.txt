import sys
import math

def calculate_tfidf(word_count, word_per_doc, n_doc, N=2):
    """
    Calculate TF-IDF (Term Frequency-Inverse Document Frequency)
    :param word_count: Number of occurrences of the word in the document
    :param word_per_doc: Total number of words in the document
    :param n_doc: Number of documents containing the word
    :param N: Total number of documents
    :return: TF-IDF value
    """
    tf = word_count / float(word_per_doc)  # Calculate Term Frequency (TF)
    idf = math.log(N / n_doc)  # Calculate Inverse Document Frequency (IDF)
    return tf * idf

# Initialize variables
prev_word = None
prev_doc_id = None
prev_word_count = None
prev_word_per_doc = None
n_doc = 1

# Process input from stdin
for line in sys.stdin:
    line = line.strip()
    word, rest = line.split("\t")
    doc_id, word_count, word_per_doc = rest.split(",")
    word_count = int(word_count)
    word_per_doc = int(word_per_doc)
    
    # Check if the word is the same as the previous one
    if prev_word == word:
        n_doc += 1  # Increment document count
    else:
        if prev_word is not None:
            # Calculate TF-IDF for the previous word and print the result
            prev_tfidf = calculate_tfidf(prev_word_count, prev_word_per_doc, n_doc)
            print("(%s,%s)\t%s" % (prev_word, prev_doc_id, prev_tfidf))
        n_doc = 1  # Reset document count for the new word
    
    # Update variables for the next iteration
    prev_word = word
    prev_doc_id = doc_id
    prev_word_count = word_count
    prev_word_per_doc = word_per_doc

# Print TF-IDF for the last word
if prev_word is not None:
    prev_tfidf = calculate_tfidf(prev_word_count, prev_word_per_doc, n_doc)
    print("(%s,%s)\t%s" % (prev_word, prev_doc_id, prev_tfidf))
