def linear_search(lst, key):
    """
    Perform linear search to find a key in a list.

    Parameters:
    - lst: The list to search through.
    - key: The value to search for.

    Returns:
    True if the key is found, False otherwise.
    """
    return key in lst

# Example usage of linear_search
lst_linear = [10, 20, 30, 40, 50]
key_linear = 30
print(f"Linear Search: {linear_search(lst_linear, key_linear)}")

def binary_search(lst, key):
    """
    Perform binary search to find a key in a sorted list.

    Parameters:
    - lst: The sorted list to search through.
    - key: The value to search for.

    Returns:
    True if the key is found, False otherwise.
    """
    if not lst:
        return False

    mid = len(lst) // 2
    if lst[mid] == key:
        return True
    elif key < lst[mid]:
        return binary_search(lst[:mid], key)
    else:
        return binary_search(lst[mid + 1:], key)

# Example usage of binary_search
lst_binary = [10, 20, 30, 50, 60, 70, 80]
key_binary = 50
print(f"Binary Search: {binary_search(lst_binary, key_binary)}")
