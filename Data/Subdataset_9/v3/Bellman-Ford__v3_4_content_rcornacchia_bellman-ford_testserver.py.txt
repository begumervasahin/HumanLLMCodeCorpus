import socket
import sys

# Ensure a command-line argument for UDP port is provided
if len(sys.argv) < 2:
    print("Usage: python udp_server.py <UDP_PORT>")
    sys.exit(1)

# Extract the UDP port from the command-line argument
UDP_PORT = int(sys.argv[1])

# Create a UDP socket
udp_socket = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)

# Bind the UDP socket to the specified IP address and port
server_address = ("160.39.231.6", UDP_PORT)
udp_socket.bind(server_address)

print(f"UDP server is running on port {UDP_PORT}")

# Listen for incoming messages and print them
while True:
    data, client_address = udp_socket.recvfrom(1024)
    message = data.decode('utf-8')
    print(f"Received message from {client_address}: {message}")
