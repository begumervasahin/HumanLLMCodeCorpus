import os
from filecmp import dircmp

def print_diff_files(dcmp):
    """
    Recursively traverse and print differing files in directory comparison.

    Args:
        dcmp (dircmp): Directory comparison object.
    """
    # Print differing files in the current directory
    for name in dcmp.diff_files:
        print(f"Differing file '{name}' found in '{dcmp.left}' and '{dcmp.right}'")
    
    # Recursively traverse subdirectories
    for sub_dcmp in dcmp.subdirs.values():
        # Recursive call to print differing files
        print_diff_files(sub_dcmp)

def main():
    # Define directories to compare
    dir1 = 'dir1'
    dir2 = 'dir2'

    # Check if both directories exist
    if not (os.path.exists(dir1) and os.path.exists(dir2)):
        print("Error: One or both directories do not exist.")
        return

    # Create directory comparison object
    dcmp = dircmp(dir1, dir2)
    
    # Call function to print differing files
    print_diff_files(dcmp)

if __name__ == "__main__":
    # Call main function when the script is executed
    main()
