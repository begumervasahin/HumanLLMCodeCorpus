import sys

def is_hamming_number(x):
    """Check if a number is a Hamming number."""
    if x == 1:
        return True
    # Check divisibility by 2, 3, and 5
    for factor in (2, 3, 5):
        if x % factor == 0:
            return is_hamming_number(x // factor)
    return False

def print_hamming_numbers(limit):
    """Print Hamming numbers up to a given limit."""
    for num in range(1, limit + 1):
        if is_hamming_number(num):
            print(num, end=' ')

def hamming_numbers_main():
    """Main function to print Hamming numbers."""
    print("Hamming Numbers:", end=' ')
    print_hamming_numbers(9830)

if __name__ == "__main__":
    # Increase recursion limit to handle large numbers
    sys.setrecursionlimit(10000)
    hamming_numbers_main()
