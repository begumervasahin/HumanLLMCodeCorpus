from time import time

# Define a decorator to measure the execution time of a function
def measure_execution_time(fn):
    def wrapper(*args, **kwargs):
        start_time = time()  # Record the start time
        result = fn(*args, **kwargs)  # Call the original function
        elapsed_time = time() - start_time  # Calculate the elapsed time
        print(f"Function '{fn.__name__}' took {elapsed_time:.6f} seconds.")  # Print the execution time
        return result
    return wrapper

# Define a function to demonstrate the timing functionality
@measure_execution_time
def perform_some_processing():
    # Simulate some processing
    for _ in range(1000000):
        pass

if __name__ == "__main__":
    # Call the decorated function to test the timing functionality
    perform_some_processing()
