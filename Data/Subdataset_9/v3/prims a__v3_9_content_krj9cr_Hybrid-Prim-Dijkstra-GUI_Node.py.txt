import sys

class Node:
    def __init__(self, x, y, idx):
        self.x = x          # x-coordinate
        self.y = y          # y-coordinate
        self.idx = idx      # index
        self.key = sys.maxsize  # Maximum integer value
        self.parent = None  # Parent node initially set to None

    def __str__(self):
        return f"{self.idx}: ({self.x}, {self.y})"

# Example usage:
node1 = Node(1, 2, 0)
print(node1)  # Output: 0: (1,2)
print("Key:", node1.key)  # Output: Key: 9223372036854775807 (maximum size of an integer)
