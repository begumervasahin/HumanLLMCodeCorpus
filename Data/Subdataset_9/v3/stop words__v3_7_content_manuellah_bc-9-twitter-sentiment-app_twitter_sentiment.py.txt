import tweepy
import progressbar
from colorama import init, Fore
from config import API_KEYS, API_SECRET, ACCESS_TOKEN, ACCESS_TOKEN_SECRET

# Initialize colorama
init()

# List of stop words, you need to define it somewhere in your code
stop_words = []

class TwitterSentiment:
    '''
    This class fetches a particular Twitter user's frequent tweets, analyzes the tweets, and returns the most used words with their frequency.
    '''

    auth = tweepy.OAuthHandler(API_KEYS, API_SECRET)
    auth.secure = True
    auth.set_access_token(ACCESS_TOKEN, ACCESS_TOKEN_SECRET)
    api = tweepy.API(auth)

    def __init__(self, username, number=3200):
        '''
        Initializes the username whose data is to be fetched (tweets).
        '''
        self.username = username
        self.no_tweets = number

    def fetch_data(self):
        '''
        Fetches data from Twitter for a specific user and stores it in a dictionary.
        '''
        try:
            tweet_data = {}
            all_tweets = []

            if self.no_tweets <= 200:
                new_tweets = TwitterSentiment.api.user_timeline(self.username, count=self.no_tweets)
            else:
                new_tweets = TwitterSentiment.api.user_timeline(self.username, count=200)

            all_tweets.extend(new_tweets)
            self.no_tweets -= 200

            while len(new_tweets) > 0 and self.no_tweets > 0:
                if self.no_tweets <= 200:
                    new_tweets = TwitterSentiment.api.user_timeline(self.username, count=self.no_tweets)
                else:
                    new_tweets = TwitterSentiment.api.user_timeline(self.username, count=200, max_id=all_tweets[-1].id - 1)

                all_tweets.extend(new_tweets)
                self.no_tweets -= 200

            for tweet in all_tweets:
                tweet_data[tweet.id] = (tweet.text, tweet.created_at)

            return tweet_data

        except tweepy.error.TweepError as e:
            print(Fore.RED + "Error occurred: {}".format(e))

    def display_tweets(self):
        '''
        Displays fetched tweets.
        '''
        tweet_dict = self.fetch_data()
        header = '{} {} {} {}'.format(Fore.BLUE + "\nTWITTER ID".ljust(30), Fore.BLUE + "DATE POSTED".ljust(30),
                                      Fore.BLUE + "THE TWEET\n\n", "==" * 70)
        print(header)

        try:
            for key, (text, created_at) in tweet_dict.items():
                print('{} {} {}'.format(Fore.YELLOW + str(key).ljust(30), Fore.GREEN + str(created_at).ljust(30),
                                         Fore.RED + text))
        except TypeError as e:
            print(Fore.RED + "No tweets for this particular Twitter handle")

    def analyse_data(self):
        '''
        Analyses the fetched tweet data and returns the most frequently used words, excluding stop words.
        '''
        tweet_dict = self.fetch_data()
        freq_dict = {}

        for key in tweet_dict:
            tweet_words = str(tweet_dict[key][0]).lower().split()

            for word in tweet_words:
                if word in stop_words:
                    continue
                elif word not in freq_dict:
                    freq_dict[word] = 1
                else:
                    freq_dict[word] += 1

        sorted_freq = sorted(freq_dict.items(), key=lambda x: x[1], reverse=True)[:10]

        print('{} {} {} {} {} {}'.format(Fore.BLUE + "\n\n", "COMMON TWEETED WORD".ljust(30),
                                         Fore.BLUE + "THE FREQUENCY\n".ljust(30), '\n\n', "==" * 70, "\n\n"))

        for word, freq in sorted_freq:
            print(' {} {}'.format(Fore.YELLOW + word.ljust(30), Fore.GREEN + str(freq).ljust(30)))


if __name__ == "__main__":
    username = input("Enter the Twitter username: ")
    ts = TwitterSentiment(username)
    ts.display_tweets()
    ts.analyse_data()
