class BinarySearchTree:
    class Node:
        def __init__(self, key, value):
            self.key = key
            self.value = value
            self.left = None
            self.right = None

    def __init__(self):
        self._root = None
        self._count = 0
        self._removed_value = None
        self._traversal_order = "inorder"

    def make_empty(self):
        """Make the binary search tree empty."""
        self._root = None
        self._count = 0
        self._removed_value = None
        self._traversal_order = "inorder"

    def _insert(self, node, key, value):
        """Insert a key-value pair into the binary search tree."""
        if not node:
            new_node = BinarySearchTree.Node(key, value)
            return new_node
        if key < node.key:
            node.left = self._insert(node.left, key, value)
        else:
            node.right = self._insert(node.right, key, value)
        return node

    def insert(self, key, value):
        """Public method to insert a key-value pair into the binary search tree."""
        self._root = self._insert(self._root, key, value)
        self._count += 1

    def _lookup(self, node, key):
        """Lookup the value associated with the given key in the binary search tree."""
        if not node:
            return None
        if key == node.key:
            return node.value
        if key < node.key:
            return self._lookup(node.left, key)
        else:
            return self._lookup(node.right, key)

    def lookup(self, key):
        """Public method to lookup the value associated with the given key."""
        return self._lookup(self._root, key)

    def _remove(self, node, key):
        """Remove a key-value pair from the binary search tree."""
        if not node:
            return None
        if key == node.key:
            if not node.left and not node.right:
                self._removed_value = node.value
                self._count -= 1
                return None
            if not node.right:
                self._removed_value = node.value
                self._count -= 1
                return node.left
            if not node.left:
                self._removed_value = node.value
                self._count -= 1
                return node.right
            minimum_node = self._minimum(node.right)
            saved_value = node.value
            node.key = minimum_node.key
            node.value = minimum_node.value
            node.right = self._remove(node.right, minimum_node.key)
            self._removed_value = saved_value
            return node
        if key < node.key:
            node.left = self._remove(node.left, key)
        else:
            node.right = self._remove(node.right, key)
        return node

    def remove(self, key):
        """Public method to remove a key-value pair from the binary search tree."""
        self._removed_value = None
        self._root = self._remove(self._root, key)
        return self._removed_value

    def _minimum(self, node):
        """Find the minimum node in the binary search tree."""
        current = node
        while current.left:
            current = current.left
        return current

    def preorder(self):
        """Set the traversal order to preorder."""
        self._traversal_order = "preorder"

    def inorder(self):
        """Set the traversal order to inorder."""
        self._traversal_order = "inorder"

    def postorder(self):
        """Set the traversal order to postorder."""
        self._traversal_order = "postorder"

    def __iter__(self):
        """Iterate through the binary search tree based on the traversal order."""
        return self._traverse(self._root)

    def _traverse(self, node):
        """Traverse the binary search tree based on the traversal order."""
        if not node:
            return
        if self._traversal_order == "preorder":
            yield node.value
        if node.left:
            yield from self._traverse(node.left)
        if self._traversal_order == "inorder":
            yield node.value
        if node.right:
            yield from self._traverse(node.right)
        if self._traversal_order == "postorder":
            yield node.value

    def __len__(self):
        """Return the number of elements in the binary search tree."""
        return self._count

# Example usage:
bst = BinarySearchTree()
bst.insert(5, 'five')
bst.insert(3, 'three')
bst.insert(7, 'seven')

print("Inorder traversal:")
bst.inorder()
for value in bst:
    print(value)
