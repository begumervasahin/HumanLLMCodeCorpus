# Import necessary functions and modules
from funcions import factors_primers, invers_modular
import numpy as np
import matplotlib.pyplot as plt
import timeit
from genera_claus import genera_claus

# Function to factorize N given e
def factoritza(N, e):
    # Obtain prime factors of N
    prime_factors = factors_primers(N)

    # If N does not have exactly 2 prime factors, return zeros
    if len(prime_factors) != 2:
        return 0, 0, 0

    # Extract prime factors
    p, q = prime_factors

    # Calculate Euler's totient function of N
    phi_n = (p - 1) * (q - 1)

    # Calculate modular inverse of e modulo phi_n
    d = invers_modular(e, phi_n)

    return p, q, d

# Function to test the factorization process for different key sizes  
def test():
    # Set maximum number of bits for keys
    max_bits = 25

    # Lists to store key sizes and corresponding processing times     
    key_sizes = []
    processing_times = []

    # Iterate over different key sizes
    for i in range(8, max_bits):
        # Generate keys of size i bits
        n, e, d, p, q = genera_claus(i)

        # Start timer
        start_time = timeit.default_timer()

        # Factorize N
        p_desc, q_desc, d_desc = factoritza(n, e)

        # Stop timer
        end_time = timeit.default_timer()

        # Store key size and processing time
        key_sizes.append(i)
        processing_times.append(end_time - start_time)

        # Print factorization results
        print("Factorization completed for N")
        print("Real prime factor p: " + str(p))
        print("Real prime factor q: " + str(q))
        print("Computed prime factor p: " + str(p_desc))
        print("Computed prime factor q: " + str(q_desc))

    # Plot key sizes against processing times
    plt.plot(key_sizes, processing_times, '-gD')
    plt.xlabel("Number of bits")
    plt.ylabel("Seconds")
    plt.show()

# Entry point of the script
if __name__ == "__main__":
    test()
