import random

def generate_dna_sequence(length, filename):
    """Generate a random DNA sequence of specified length and save it to a file."""
    nucleotides = ["A", "C", "T", "G"]
    sequence = ''.join(random.choice(nucleotides) for _ in range(length))
    with open(filename, "w") as file_txt:
        file_txt.write(sequence)

def longest_common_subsequence(X, Y):
    """Find the longest common subsequence (LCS) of two DNA sequences."""
    m, n = len(X), len(Y)
    # Create a 2D table to store the lengths of LCSs
    lcs_lengths = [[0] * (n + 1) for _ in range(m + 1)]

    # Fill in the table to calculate the lengths of LCSs
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if X[i - 1] == Y[j - 1]:
                lcs_lengths[i][j] = lcs_lengths[i - 1][j - 1] + 1
            else:
                lcs_lengths[i][j] = max(lcs_lengths[i - 1][j], lcs_lengths[i][j - 1])

    # Trace back to find the actual LCS
    lcs = []
    i, j = m, n
    while i > 0 and j > 0:
        if X[i - 1] == Y[j - 1]:
            lcs.append(X[i - 1])
            i -= 1
            j -= 1
        elif lcs_lengths[i - 1][j] > lcs_lengths[i][j - 1]:
            i -= 1
        else:
            j -= 1

    # Print the LCS and determine if it's matching or not based on its length
    print("Longest Common Subsequence of X and Y is:", "".join(reversed(lcs)))
    if len(lcs) > 500:
        print("Matching")
    else:
        print("Not matching")

# Generate DNA sequences and save them to files
generate_dna_sequence(1000, "dna-sequence.txt")
generate_dna_sequence(1000, "dna-sequence2.txt")

# Read DNA sequences from files
X = open("dna-sequence.txt", "r").read()
Y = open("dna-sequence2.txt", "r").read()

# Calculate LCS and check if it's matching or not
longest_common_subsequence(X, Y)
