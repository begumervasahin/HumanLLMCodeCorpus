from random import randint
from time import time
from tkinter import *

def selection_sort(values):
    """
    Perform selection sort on the given list.

    Args:
        values (list): The list to be sorted.
    """
    for i in range(len(values)):
        min_pos = i
        for j in range(i + 1, len(values)):
            if values[j] < values[min_pos]:
                min_pos = j
        values[i], values[min_pos] = values[min_pos], values[i]

def collect_measurements(n_min, n_max, runs):
    """
    Collect measurements for sorting algorithms.

    Args:
        n_min (int): Minimum list size.
        n_max (int): Maximum list size.
        runs (int): Number of different measurements to run.

    Returns:
        list: A list of lists containing size and time measurements.
    """
    measurements = []
    n = n_min
    for _ in range(runs):
        values = [randint(1, 1000) for _ in range(n)]
        start_time = time()
        selection_sort(values)
        end_time = time()
        measurements.append([n, round((end_time - start_time), 3)])
        n += (n_max - n_min) // (runs - 1)
        n -= n % 5
    return measurements

def display_results(measurements):
    """
    Display measurement results using Tkinter.

    Args:
        measurements (list): A list of lists containing size and time measurements.
    """
    window = Tk()
    window.title("Results of Sample Runs - Selection Sort")

    Label(text='Sort Size').grid(row=0, column=1, sticky=NSEW)
    Label(text='Seconds to Sort').grid(row=0, column=2, sticky=NSEW)

    for i, (size, time) in enumerate(measurements, start=1):
        Label(text=f'{size}', relief=RIDGE, width=20, height=2).grid(row=i, column=1, sticky=NSEW)
        Label(text=f'{time:.3f}', relief=RIDGE, width=20, height=2).grid(row=i, column=2, sticky=NSEW)

    window.mainloop()

def main():
    print()
    n_min = int(input("Enter the minimum list size: "))
    print()
    n_max = int(input("Enter the maximum list size: "))
    print()
    runs = int(input("Enter the number of different measurements to run: "))

    measurements = collect_measurements(n_min, n_max, runs)
    display_results(measurements)

if __name__ == "__main__":
    main()
