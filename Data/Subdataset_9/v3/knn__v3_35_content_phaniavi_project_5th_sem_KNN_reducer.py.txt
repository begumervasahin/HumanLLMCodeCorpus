import sys

def main():
    distances = []

    # Read lines from standard input
    for line in sys.stdin:
        line = line.strip()
        elements = line.split()
        if len(elements) >= 2:
            distances.append((float(elements[0]), float(elements[1])))

    k = 5
    if k <= 0:
        print("k must be greater than 0")
        return

    # Sort the distances based on the first column
    distances.sort(key=lambda x: x[0])

    # Calculate the average of the first k values in the second column
    total = sum(value for _, value in distances[:k])
    print(total / k)

if __name__ == "__main__":
    main()
