def selection_sort(lst):
    """
    Sorts a list in ascending order using the selection sort algorithm.

    Args:
    - lst (list): The list to be sorted.
    """
    # Iterate over each index in the list
    for i in range(len(lst)):
        # Assume the current index as the minimum
        min_index = i
        # Find the index of the minimum element in the remaining unsorted portion
        for j in range(i + 1, len(lst)):
            if lst[min_index] > lst[j]:
                min_index = j
        # Swap the current element with the minimum element found
        lst[i], lst[min_index] = lst[min_index], lst[i]

def get_user_input():
    """
    Gets user input for the list elements.

    Returns:
    - list: The list of user input elements.
    """
    lst = []
    num_elements = int(input("Enter the number of elements in the list: "))
    for _ in range(num_elements):
        element = int(input("Enter an element: "))
        lst.append(element)
    return lst

# Main code
if __name__ == "__main__":
    # Get user input for the list
    lst = get_user_input()
    print("List before sorting:", lst)
    # Sort the list using selection sort
    selection_sort(lst)
    print("List after sorting:", lst)
