class Edge:
    def __init__(self, vertex1, vertex2, weight, extra):
        """
        Initializes an Edge with two vertices, weight, and additional information.
        """
        self.vertex1 = vertex1
        self.vertex2 = vertex2
        self.weight = weight
        self.extra = extra
        self.selected = False

    def __repr__(self):
        """
        Returns a string representation of the Edge.
        """
        return f"Edge({self.vertex1}, {self.vertex2}, weight={self.weight}, extra={self.extra}, selected={self.selected})"

# Example usage
if __name__ == "__main__":
    edge = Edge(1, 2, 10, 0)
    print(edge)
