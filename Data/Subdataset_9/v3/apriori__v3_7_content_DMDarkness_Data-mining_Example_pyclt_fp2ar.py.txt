import pyclt  # Import the package for the Central Limit Theorem-based algorithm
import fp2ar  # Import the package to form association rules from frequent patterns
import re  # Import the regular expression module for string splitting and extracting items from transactions

def read_transactional_dataset(file_name):
    """
    Read the transactional dataset from the given file and return a list of transactions.
    Each transaction is represented as a list of integers.
    """
    transactions = []
    with open(file_name, 'r') as file:
        for line in file:
            # Split the dataset by space or newline and extract integers
            items = re.findall(r'\d+', line)
            if items:
                transactions.append([int(item) for item in items])
    return transactions

if __name__ == '__main__':
    # Specify the file containing the transactional dataset
    dataset_file = "kosarak.dat"

    # Read the transactional dataset from the file
    dataset = read_transactional_dataset(dataset_file)

    # Mine the dataset with a minimum support of 3%
    min_support = 0.03
    deviation = 0.005
    probability = 0.01
    frequent_patterns, sample_size = pyclt.getFP(dataset, min_support, deviation, probability)

    # Form association rules from frequent patterns
    min_confidence = 0.75
    min_lift = 1
    association_rules = fp2ar.getAR(frequent_patterns, sample_size, min_confidence, min_lift)
