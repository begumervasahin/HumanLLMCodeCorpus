import matplotlib.pyplot as plt
import networkx as nx
from prim import prim

def create_graph():
    """Create a graph with edges and weights."""
    graph = nx.Graph()
    edges = [
        ('A', 'B', 3), ('A', 'D', 5), ('A', 'E', 9),
        ('B', 'C', 5), ('B', 'D', 4), ('B', 'E', 8),
        ('C', 'D', 7), ('C', 'G', 3), ('D', 'F', 8),
        ('D', 'G', 5), ('D', 'H', 6), ('E', 'F', 2),
        ('F', 'H', 10), ('G', 'I', 1), ('H', 'I', 3)
    ]
    graph.add_weighted_edges_from(edges)
    return graph

def main():
    """Main function."""
    # Create graph
    graph = create_graph()

    # Find minimal spanning tree
    start_node = 'A'
    minimal_spanning_tree = prim(graph, start_node)

    # Identify other edges not in the minimal spanning tree
    other_edges = [edge for edge in graph.edges() if edge not in minimal_spanning_tree]

    # Print the minimal spanning tree
    print("Minimal Spanning Tree:", minimal_spanning_tree)

    # Generate layout for the graph
    layout = nx.spring_layout(graph)

    # Draw nodes
    nx.draw_networkx_nodes(graph, layout, node_size=500)

    # Draw edges for the minimal spanning tree
    nx.draw_networkx_edges(graph, layout, edgelist=minimal_spanning_tree, width=6)

    # Draw other edges with dashed lines
    nx.draw_networkx_edges(graph, layout, edgelist=other_edges, width=6, alpha=0.5, edge_color='b', style='dashed')

    # Draw labels for nodes
    nx.draw_networkx_labels(graph, layout, font_size=20, font_family='sans-serif')

    # Hide axis
    plt.axis('off')

    # Show the graph
    plt.show()

if __name__ == "__main__":
    main()
