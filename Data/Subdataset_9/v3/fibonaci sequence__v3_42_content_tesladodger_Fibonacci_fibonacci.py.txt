import os
import datetime

def fibonacci_recursive(n):
    """Calculate the nth Fibonacci number using a recursive method."""
    if n <= 2:
        return 1
    else:
        return fibonacci_recursive(n - 1) + fibonacci_recursive(n - 2)

def get_positive_integer(prompt):
    """Get a positive integer input from the user."""
    while True:
        try:
            value = int(input(prompt))
            if value <= 0:
                print('\nYou must enter a positive integer.\n')
            else:
                return value
        except ValueError:
            print('\nYou must enter an integer.\n')

def generate_fibonacci_n_times(n):
    """Generate the Fibonacci sequence n times."""
    num = 1
    next_num = 1
    print(num)

    for _ in range(n - 1):
        print(next_num)
        num, next_num = next_num, num + next_num

    print('\nS    - Save to a file')
    print('else - Go to the main menu')
    choice = input('==> ')

    return choice.lower() == 's'

def generate_fibonacci_up_to(maxnum):
    """Generate the Fibonacci sequence up to a certain number."""
    num = 1
    next_num = 1
    iterations = 0
    print(num)

    while next_num <= maxnum:
        print(next_num)
        num, next_num = next_num, num + next_num
        iterations += 1

    print('\nS    - Show more information')
    print('else - Go to the main menu')
    choice = input('==> ')

    if choice.lower() == 's':
        if num == maxnum:
            print("The number", maxnum, "is in the Fibonacci sequence")
        print("Number of iterations:       ", iterations)
        print("Your number:                ", maxnum)
        print("Next number in the sequence:", next_num)
        print("Difference to that number:   ", (next_num - maxnum))
        print("\n\n")

def get_time_of_day():
    """Get the time of day for a greeting."""
    hour = datetime.datetime.now().hour
    if hour >= 20:
        return 'night!'
    elif hour >= 12:
        return 'afternoon!'
    elif hour >= 5:
        return 'day!'
    else:
        return 'night!'
