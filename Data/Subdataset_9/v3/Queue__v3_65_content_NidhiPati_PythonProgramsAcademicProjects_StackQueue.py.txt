import sys

class Stack:
    """Implementation of a stack data structure."""

    def __init__(self):
        """Initialize an empty stack."""
        self._items = []

    def push(self, item):
        """Push an item onto the stack."""
        self._items.append(item)

    def peek(self):
        """Return the top item of the stack without removing it."""
        if not self._items:
            raise IndexError("Stack is empty")
        return self._items[-1]

    def pop(self):
        """Remove and return the top item of the stack."""
        if not self._items:
            raise IndexError("Stack is empty")
        return self._items.pop()

    def __len__(self):
        """Return the number of items in the stack."""
        return len(self._items)

    def get_size(self):
        """Return the size of the stack in bytes."""
        return sys.getsizeof(self._items)

class CircularQueue:
    """Implementation of a circular queue data structure."""

    DEFAULT_CAPACITY = 10

    def __init__(self):
        """Initialize an empty circular queue."""
        self._items = [None] * CircularQueue.DEFAULT_CAPACITY
        self._size = 0
        self._front = 0

    def __len__(self):
        """Return the number of items in the queue."""
        return self._size

    def is_empty(self):
        """Check if the queue is empty."""
        return self._size == 0

    def first(self):
        """Return the first item in the queue without removing it."""
        if self.is_empty():
            raise IndexError("Empty Queue")
        return self._items[self._front]

    def dequeue(self):
        """Remove and return the first item in the queue."""
        if self.is_empty():
            raise IndexError("Empty Queue")
        value = self._items[self._front]
        self._items[self._front] = None
        self._front = (self._front + 1) % len(self._items)
        self._size -= 1
        return value

    def enqueue(self, item):
        """Add an item to the end of the queue."""
        if self._size == len(self._items):
            self._resize(len(self._items) * 2)
        next_index = (self._front + self._size) % len(self._items)
        self._items[next_index] = item
        self._size += 1

    def _resize(self, capacity):
        """Resize the internal array of the queue."""
        old_items = self._items
        self._items = [None] * capacity
        current_index = self._front
        for i in range(self._size):
            self._items[i] = old_items[current_index]
            current_index = (current_index + 1) % len(old_items)
        self._front = 0

    def get_size(self):
        """Return the size of the queue in bytes."""
        return sys.getsizeof(self._items)

    stack = Stack()
    stack.push(1)
    stack.push(2)
    stack.push(3)
    print(stack.pop())  # Output: 3

    queue = CircularQueue()
    queue.enqueue(1)
    queue.enqueue(2)
    queue.enqueue(3)
    print(queue.dequeue())  # Output: 1
    print(queue.first())    # Output: 2
