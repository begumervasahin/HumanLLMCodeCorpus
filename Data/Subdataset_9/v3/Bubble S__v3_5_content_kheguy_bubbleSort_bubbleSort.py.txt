import random

class BubbleSort:
    def __init__(self, size_of_array, interval_from, interval_to):
        """
        Initializes a BubbleSort object with a random array of given size and interval.

        Args:
            size_of_array (int): The size of the random array.
            interval_from (int): The lower bound of the random interval.
            interval_to (int): The upper bound of the random interval.
        """
        self.array_size = size_of_array
        self.array = []
        random.seed()
        # Generate a random array
        for _ in range(self.array_size):
            self.array.append(random.randint(interval_from, interval_to))

    def __del__(self):
        """
        Clears the array on object deletion.
        """
        self.array.clear()

    def print_array(self):
        """
        Prints the array.
        """
        print(self.array)

    def sort(self):
        """
        Performs bubble sort on the array.
        """
        is_sorted = False
        while not is_sorted:
            is_sorted = True
            for i in range(self.array_size - 1):
                if self.array[i + 1] < self.array[i]:
                    self.array[i + 1], self.array[i] = self.array[i], self.array[i + 1]
                    is_sorted = False

def main():
    while True:
        # Get input for array size and random interval
        n = int(input('Enter size of array:\n'))
        a = int(input('Enter lower bound for random numbers:\n'))
        b = int(input('Enter upper bound for random numbers:\n'))
        # Create and sort the array
        sort = BubbleSort(n, a, b)
        print("Original array:")
        sort.print_array()
        sort.sort()
        print("Sorted array:")
        sort.print_array()
        del sort

if __name__ == "__main__":
    main()
