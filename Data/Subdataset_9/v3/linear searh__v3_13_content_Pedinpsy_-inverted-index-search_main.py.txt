import os
import time

class HashTable:
    def __init__(self, size, method='division', collision='linear'):  
        self.size = size
        self.table = [[] for _ in range(size)]  # Initialize an empty hash table
        self.method = method  # Set the hashing method
        self.collision = collision  # Set the collision resolution strategy

    def hash_function(self, key):
        if self.method == 'multiplication':
            return self._hash_multiplication(key)
        else:
            return self._hash_division(key)

    def _hash_division(self, key):
        return hash(key) % self.size

    def _hash_multiplication(self, key):
        A = 0.6180339887  # Fractional part of the golden ratio       
        return int(self.size * ((hash(key) * A) % 1))

    def insert(self, key, value):
        index = self.hash_function(key)
        if not self.table[index]:
            self.table[index].append((key, value))
        elif self.collision == 'linear':
            self._resolve_linear_collision(index, key, value)
        elif self.collision == 'quadratic':
            self._resolve_quadratic_collision(index, key, value)

    def _resolve_linear_collision(self, index, key, value):
        next_index = (index + 1) % self.size
        while next_index != index:
            if not self.table[next_index]:
                self.table[next_index].append((key, value))
                return
            next_index = (next_index + 1) % self.size
        print("Table is full. Couldn't insert:", key)

    def _resolve_quadratic_collision(self, index, key, value):
        i = 1
        next_index = (index + i**2) % self.size
        while next_index != index:
            if not self.table[next_index]:
                self.table[next_index].append((key, value))
                return
            i += 1
            next_index = (index + i**2) % self.size
        print("Table is full. Couldn't insert:", key)

    def get_value(self, key):
        index = self.hash_function(key)
        for k, v in self.table[index]:
            if k == key:
                return v
        return None

def list_files(folder):
    return [os.path.join(folder, name) for name in os.listdir(folder) if os.path.isfile(os.path.join(folder, name))]

def build_index(file_list, hashtable):
    for i, file in enumerate(file_list):
        with open(file, 'r') as f:
            words = f.read().lower().replace(",", "").replace(".", "").replace("!", "").replace("?", "").replace("\r", "").replace("\t", "").replace("\n", "").split(" ")
            for word in words:
                count = words.count(word)
                hashtable.insert(word, [count, i + 1])

def main():
    hashtable_mul_linear = HashTable(1000000, method='multiplication', collision='linear')
    hashtable_mul_quadratic = HashTable(1000000, method='multiplication', collision='quadratic')
    hashtable_div_linear = HashTable(1000000, method='division', collision='linear')   
    hashtable_div_quadratic = HashTable(1000000, method='division', collision='quadratic')

    folder = 'base'
    files = list_files(folder)

    for hashtable, method, collision in [(hashtable_mul_linear, 'Multiplication', 'Linear'),
                                         (hashtable_mul_quadratic, 'Multiplication', 'Quadratic'),
                                         (hashtable_div_linear, 'Division', 'Linear'),
                                         (hashtable_div_quadratic, 'Division', 'Quadratic')]:
        start = time.time()
        build_index(files, hashtable)
        end = time.time()
        print(f"Time for {method} Hashing with {collision} Collision: {end - start}")

    while True:
        word = input("Enter a word:\n")
        print("Multiplication Linear:", hashtable_mul_linear.get_value(word))
        print("Multiplication Quadratic:", hashtable_mul_quadratic.get_value(word))
        print("Division Linear:", hashtable_div_linear.get_value(word))
        print("Division Quadratic:", hashtable_div_quadratic.get_value(word))

if __name__ == "__main__":
    main()
