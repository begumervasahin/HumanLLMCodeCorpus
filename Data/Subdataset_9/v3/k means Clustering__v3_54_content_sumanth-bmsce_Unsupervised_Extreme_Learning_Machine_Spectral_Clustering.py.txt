import numpy as np
from sklearn.cluster import SpectralClustering
from itertools import groupby

def spectral_clustering(data, num_clusters):
    # Perform spectral clustering
    spectral_labels = SpectralClustering(n_clusters=num_clusters, eigen_solver='arpack', affinity="nearest_neighbors").fit_predict(data) 

    # Print cluster information for different subsets of data
    print_cluster_info("Samples 1-59", np.sort(spectral_labels[:59]))
    print_cluster_info("Samples 60-130", np.sort(spectral_labels[59:130]))
    print_cluster_info("Samples 131-178", np.sort(spectral_labels[130:178]))

def print_cluster_info(label, cluster_labels):
    # Print cluster labels and sizes
    print(f"Clusters for {label}:")
    for key, group in groupby(cluster_labels):
        print(f"  Cluster {key}: Size {len(list(group))}")
