class ListNode:
    """A class to represent a node in a linked list."""
    
    def __init__(self, value):
        """Initialize a ListNode with a value and a next reference."""
        self.val = value
        self.next = None

class Solution:
    """A class to perform addition of two numbers represented by linked lists."""

    def addTwoNumbers(self, l1, l2):
        """Add two numbers represented by linked lists and return the result."""
        
        # Initialize carry to 0
        carry = 0
        # Create a dummy node to store the result
        result_root = ListNode(0)
        # Initialize a pointer to traverse the result list
        result_ptr = result_root

        # Iterate through both input lists while there are nodes or carry remains
        while l1 or l2 or carry:
            # Add the values of the current nodes and the carry
            if l1:
                carry += l1.val
                l1 = l1.next
            if l2:
                carry += l2.val
                l2 = l2.next

            # Append a new node with the sum of current digits to the result list
            result_ptr.next = ListNode(carry % 10)
            carry //= 10  # Update carry for the next digit addition
            result_ptr = result_ptr.next  # Move to the next node in the result list

        return result_root.next  # Return the result list excluding the dummy node

def create_linked_list(values):
    """Create a linked list from a list of values."""
    head = ListNode(0)  # Dummy node
    current = head
    for value in values:
        current.next = ListNode(value)
        current = current.next
    return head.next

def print_linked_list(node):
    """Print the values of a linked list."""
    while node:
        print(node.val, end=" ")
        node = node.next
    print()

# Example usage:
if __name__ == "__main__":
    # Create linked lists l1 and l2
    l1 = create_linked_list([2, 4, 3])
    l2 = create_linked_list([5, 6, 4])

    # Create a Solution object
    solution = Solution()

    # Add the two numbers represented by the linked lists
    result = solution.addTwoNumbers(l1, l2)

    # Print the result
    print_linked_list(result)
