import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import PolynomialFeatures

# Load dataset
dataset = pd.read_csv('Position_Salaries.csv')
X = dataset['Level'].values.reshape(-1, 1)  # Independent variable (Position Level)
y = dataset['Salary'].values                # Dependent variable (Salary)

# Finding the optimal degree for polynomial regression
best_degree = 1          # Initialize the best degree
min_mse = float('inf')   # Initialize minimum mean squared error

# Loop through different polynomial degrees to find the best fit
for degree in range(1, 11):
    # Transforming the feature matrix to polynomial features
    poly_features = PolynomialFeatures(degree=degree)
    X_poly = poly_features.fit_transform(X)
    
    # Fitting polynomial regression to the dataset
    polynomial_regression = LinearRegression()
    polynomial_regression.fit(X_poly, y)
    
    # Predicting the salary based on the polynomial regression model
    y_pred = polynomial_regression.predict(X_poly)
    
    # Calculating mean squared error
    mse = np.mean((y_pred - y) ** 2)
    
    # Updating the best degree if the current degree has lower mean squared error
    if mse < min_mse:
        min_mse = mse
        best_degree = degree

# Fit the final polynomial regression model with the appropriate degree
poly_features = PolynomialFeatures(degree=best_degree)
X_poly = poly_features.fit_transform(X)
polynomial_regression = LinearRegression()
polynomial_regression.fit(X_poly, y)

# Plotting the results
plt.scatter(X, y, color='red', label='Actual data')
plt.plot(X, polynomial_regression.predict(X_poly), color='blue', label='Polynomial Regression')
plt.title('Truth or Bluff (Polynomial Regression)')
plt.xlabel('Position level')
plt.ylabel('Salary')
plt.legend()
plt.show()
