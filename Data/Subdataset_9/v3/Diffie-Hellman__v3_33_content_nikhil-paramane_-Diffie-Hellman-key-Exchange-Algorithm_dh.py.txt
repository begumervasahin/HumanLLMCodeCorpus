from random import randint

def is_prime(num, count):
    """
    Checks if a number is prime using the Miller-Rabin primality test.

    Args:
        num (int): The number to be checked for primality.
        count (int): The number of iterations for the primality test.

    Returns:
        bool: True if the number is prime, False otherwise.
    """
    if num == 1:
        return False
    if count >= num:
        count = num - 1
    for x in range(count):
        a = randint(1, num - 1)
        if pow(a, num-1, num) != 1:
            return False
    return True

def generate_big_prime(n):
    """
    Generates a large prime number with n bits using the Miller-Rabin primality test.

    Args:
        n (int): The number of bits for the prime number.

    Returns:
        int: A large prime number.
    """
    found_prime = False
    while not found_prime:
        p = randint(2**(n-1), 2**n)
        if is_prime(p, 1000):
            return p

def generate_keys(n_bits):
    """
    Generates two large prime numbers and computes public and secret keys for Alice and Bob.

    Args:
        n_bits (int): The number of bits for the prime numbers.

    Returns:
        tuple: A tuple containing the prime numbers and the keys (prime1, prime2, alice_key, bob_key).
    """
    prime1 = generate_big_prime(n_bits)
    prime2 = generate_big_prime(n_bits)

    alice_private_key = randint(2**10, 2**15)
    bob_private_key = randint(2**10, 2**15)

    a = pow(prime2, alice_private_key, prime1)
    b = pow(prime2, bob_private_key, prime1)

    alice_key = pow(b, alice_private_key, prime1)
    bob_key = pow(a, bob_private_key, prime1)

    return prime1, prime2, alice_key, bob_key
