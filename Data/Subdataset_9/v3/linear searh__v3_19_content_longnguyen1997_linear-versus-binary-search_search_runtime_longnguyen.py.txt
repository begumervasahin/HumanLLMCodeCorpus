import matplotlib.pyplot as plt
import numpy as np
from random import randint
from sys import maxsize
from time import time

def binary_search(array, n, target):
    """
    Finds the index of the target element in a sorted array using binary search.
    Returns -1 if the target is not found.
    """
    left = 0
    right = n - 1
    while left <= right:
        mid = (left + right) // 2
        if array[mid] < target:
            left = mid + 1
        elif array[mid] > target:
            right = mid - 1
        else:
            return mid
    return -1

def linear_search(array, n, target):
    """
    Finds the index of the target element in a sorted array using linear search.
    Returns -1 if the target is not found.
    """
    for i in range(n):
        if array[i] == target:
            return i
        elif array[i] > target:
            return -1
    return -1

def generate_random_array(size):
    """
    Generates an array of random integers of given size.
    """
    return [randint(-maxsize - 1, maxsize) for _ in range(size)]

def calculate_cpu_time(search_function, array, num_trials):
    """
    Measures CPU time taken by a search function for a given number of trials.
    """
    total_time = 0
    for _ in range(num_trials):
        start_time = time()
        search_function(array, len(array), array[randint(0, len(array) - 1)])
        end_time = time()
        total_time += end_time - start_time
    return total_time / num_trials

def run_performance_trials(num_trials=5, benchmark=False, array_size=None):
    """
    Runs performance trials for binary and linear search algorithms.
    """
    threshold = 2 if array_size is None else array_size
    while True:
        binary_search_time = linear_search_time = float('inf')
        array = generate_random_array(threshold)
        array.sort()

        binary_search_time = calculate_cpu_time(binary_search, array, num_trials)
        binary_search_time /= num_trials
        linear_search_time = calculate_cpu_time(linear_search, array, num_trials)
        linear_search_time /= num_trials

        if not benchmark:
            return linear_search_time, binary_search_time
        if benchmark and binary_search_time < linear_search_time:
            return threshold
        threshold += 1

def find_threshold():
    """
    Finds the threshold at which binary search outperforms linear search.
    """
    return sum(run_performance_trials(250, True) for _ in range(100)) / 100

def get_performance_model(linear, data):
    """
    Generates a performance model for the given data.
    """
    if linear:
        min_time = min(data)
        min_index = data.index(min_time)
        max_time = max(data)
        max_index = data.index(max_time)
        slope = (max_time - min_time) / (max_index - min_index)
        return lambda array_size: slope * array_size + min_time
    else:
        slope, intercept = np.polyfit(np.log(np.array([i for i in range(1, len(data) + 1)])), np.array(data), 1)
        return lambda array_size: intercept + (slope * np.log(array_size))

def plot_performance(max_array_size=50, num_trials=250):
    """
    Plots the performance of linear and binary search algorithms.
    """
    linear_times = []
    binary_times = []
    for size in range(1, max_array_size + 1):
        linear_time, binary_time = run_performance_trials(num_trials, array_size=size)
        linear_times.append(linear_time)
        binary_times.append(binary_time)

    linear_model = get_performance_model(True, linear_times)
    binary_model = get_performance_model(False, binary_times)

    plt.plot(linear_times, 'o', label='Linear search')
    plt.plot(binary_times, 'o', label='Binary search')
    plt.plot([i for i in range(1, max_array_size + 1)], [linear_model(i) for i in range(1, max_array_size + 1)])
    plt.plot([i for i in range(1, max_array_size + 1)], [binary_model(i) for i in range(1, max_array_size + 1)])
    plt.legend()
    plt.xlim(left=1)
    plt.title('Linear versus binary search runtime')
    plt.xlabel('Size of array')
    plt.ylabel('CPU time')
    plt.show()

    return linear_model, binary_model

# Perform the plotting
linear_model, binary_model = plot_performance(350)
