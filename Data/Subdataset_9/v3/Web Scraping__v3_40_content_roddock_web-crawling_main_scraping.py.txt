import requests
from bs4 import BeautifulSoup

def fetch_html(url):
    """
    Fetches the HTML content of a webpage given its URL.
    """
    try:
        response = requests.get(url)
        return response.text
    except Exception as e:
        print("Error fetching HTML:", e)
        return None

def get_title(html_content):
    """
    Extracts the title of a webpage from its HTML content.
    """
    try:
        soup = BeautifulSoup(html_content, 'html.parser')
        title = soup.title.string
        return title
    except Exception as e:
        print("Error extracting title:", e)
        return None

def get_body(html_content):
    """
    Extracts the body of a webpage from its HTML content.
    """
    try:
        soup = BeautifulSoup(html_content, 'html.parser')
        body = soup.body
        return body
    except Exception as e:
        print("Error extracting body:", e)
        return None

def get_text_paragraphs(html_content):
    """
    Extracts all text paragraphs from a webpage's HTML content.
    """
    try:
        soup = BeautifulSoup(html_content, 'html.parser')
        text_paragraphs = soup.find_all('p')
        return text_paragraphs
    except Exception as e:
        print("Error extracting text paragraphs:", e)
        return None

def get_head_section(html_content):
    """
    Extracts the head section of a webpage from its HTML content.
    """
    try:
        soup = BeautifulSoup(html_content, 'html.parser')
        head_section = soup.head
        return head_section
    except Exception as e:
        print("Error extracting head section:", e)
        return None

def get_links(html_content):
    """
    Extracts all links from a webpage's HTML content.
    """
    try:
        soup = BeautifulSoup(html_content, 'html.parser')
        links = soup.find_all('a')
        return links
    except Exception as e:
        print("Error extracting links:", e)
        return None

def get_images(html_content):
    """
    Extracts all images from a webpage's HTML content.
    """
    try:
        soup = BeautifulSoup(html_content, 'html.parser')
        images = soup.find_all('img')
        return images
    except Exception as e:
        print("Error extracting images:", e)
        return None

# URL of the web page to scrape
url = "https://www.pea.co.th/"

# Fetch HTML content of the webpage
html_content = fetch_html(url)

if html_content is None:
    print("Failed to fetch HTML content.")
else:
    # Get and display the title of the web page
    title = get_title(html_content)
    print("*** Title: ***")
    print(title if title else "Title not found")

    # Get and display the body of the web page
    body = get_body(html_content)
    print("\n*** Body: ***")
    print(body if body else "Body not found")

    # Get and display all text paragraphs of the web page
    text_paragraphs = get_text_paragraphs(html_content)
    print("\n*** Text Paragraphs: ***")
    if text_paragraphs:
        for paragraph in text_paragraphs:
            print(paragraph.get_text())

    # Get and display the head section of the web page
    head_section = get_head_section(html_content)
    print("\n*** Head Section: ***")
    print(head_section if head_section else "Head section not found")

    # Get and display all links of the web page
    links = get_links(html_content)
    print("\n*** Links: ***")
    if links:
        for link in links:
            if "href" in link.attrs:
                print(link.attrs["href"])

    # Get and display all images of the web page
    images = get_images(html_content)
    print("\n*** Images: ***")
    if images:
        for image in images:
            if "src" in image.attrs:
                print(image.attrs["src"])
