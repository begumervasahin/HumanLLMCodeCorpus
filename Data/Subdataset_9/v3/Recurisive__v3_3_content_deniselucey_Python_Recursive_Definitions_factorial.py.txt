def factorial(n):
    """Calculate the factorial of a non-negative integer."""
    # Base case: factorial of 0 or negative numbers is 1
    if n < 1:
        return 1
    else:
        # Recursively calculate factorial for n-1 and multiply by n
        result = n * factorial(n - 1)
        # Print the intermediate result for each step
        print(f"{n}! = {result}")
        return result

# Test the factorial function with an example input
factorial(4)
