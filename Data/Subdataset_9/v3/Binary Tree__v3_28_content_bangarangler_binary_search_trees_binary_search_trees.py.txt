class BinarySearchTree:
    def __init__(self, value):
        """Initialize a Binary Search Tree with a given value."""
        self.value = value
        self.left = None
        self.right = None

    def insert(self, value):
        """Insert a new value into the Binary Search Tree."""
        if value < self.value:
            if not self.left:
                self.left = BinarySearchTree(value)
            else:
                self.left.insert(value)
        else:
            if not self.right:
                self.right = BinarySearchTree(value)
            else:
                self.right.insert(value)

    def contains(self, target):
        """Check if the Binary Search Tree contains a target value."""
        if target == self.value:
            return True
        elif target < self.value and self.left:
            return self.left.contains(target)
        elif target > self.value and self.right:
            return self.right.contains(target)
        else:
            return False

    def get_max(self):
        """Get the maximum value in the Binary Search Tree."""
        return self.right.get_max() if self.right else self.value

    def for_each(self, cb):
        """Apply a callback function to each node in the Binary Search Tree."""
        cb(self.value)
        if self.left:
            self.left.for_each(cb)
        if self.right:
            self.right.for_each(cb)


# Example usage
def print_node_value(value):
    """Callback function to print the node value."""
    print(value)

if __name__ == "__main__":
    # Example of creating a Binary Search Tree
    bst = BinarySearchTree(5)
    for val in [3, 8, 2, 4, 7, 9]:
        bst.insert(val)

    # Example of using contains method
    print(bst.contains(4))  # Output: True
    print(bst.contains(6))  # Output: False

    # Example of using get_max method
    print(bst.get_max())    # Output: 9

    # Example of using for_each method
    bst.for_each(print_node_value)  # Output: 5 3 2 4 8 7 9
