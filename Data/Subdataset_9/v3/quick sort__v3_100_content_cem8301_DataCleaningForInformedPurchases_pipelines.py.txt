import scrapy

class CraigslistPipeline:
    def process_item(self, item, spider):
        # Process the item here, such as saving it to a database
        return item

class MySpider(scrapy.Spider):
    name = 'my_spider'
    start_urls = ['https://www.example.com']

    custom_settings = {
        'ITEM_PIPELINES': {
            'my_project.pipelines.CraigslistPipeline': 100,
        }
    }

    def parse(self, response):
        # Extract data from the response and yield items
        item = {
            'title': response.css('title::text').get(),
            'url': response.url,
        }
        yield item
