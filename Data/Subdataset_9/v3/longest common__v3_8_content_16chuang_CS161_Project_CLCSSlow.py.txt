import sys
import numpy as np

# Initialize a 2D array to store results of subproblems
memoization_table = np.zeros((2048, 2048), dtype=int)

# Function to find the length of the Longest Common Subsequence (LCS)
def find_LCS_length(A, B):
    m = len(A)
    n = len(B)
    
    # Iterate through each pair of characters in A and B
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if A[i - 1] == B[j - 1]:
                memoization_table[i][j] = memoization_table[i - 1][j - 1] + 1
            else:
                memoization_table[i][j] = max(memoization_table[i - 1][j], memoization_table[i][j - 1])
    
    # Return the length of the LCS
    return memoization_table[m][n]

# Function to find the length of the Longest Common Rotated Subsequence (LCRS)
def find_LCRS_length(A, B):
    longest = find_LCS_length(A, B)
    
    # Rotate A and check for longer common subsequences
    for i in range(0, len(A)):
        rotated = A[i:] + A[:i]
        clrs = find_LCS_length(rotated, B)
        
        # Update longest if a longer common rotated subsequence is found
        if clrs > longest:
            longest = clrs
    
    # Return the length of the Longest Common Rotated Subsequence
    return longest

def main():
    # Ensure correct usage of the script
    if len(sys.argv) != 1:
        sys.exit('Usage: `python LCS.py < input`')
    
    # Read input strings line by line from standard input
    for line in sys.stdin:
        A, B = line.split()
        # Compute and print the length of the Longest Common Rotated Subsequence
        print(find_LCRS_length(A, B))

if __name__ == '__main__':
    main()
