# Importing necessary modules from PySpark
from pyspark import SparkConf, SparkContext
from pyspark.ml.feature import HashingTF, IDF, NGram
from pyspark.sql import SparkSession
import re

# Function to process each file and tokenize its content
def process_file(file_path, content):
    # Preprocessing text: converting to lowercase, removing non-alphanumeric characters, and splitting into words
    words = re.sub('[^a-z0-9]+', ' ', content.lower()).split()
    
    # Extracting file name without path
    file_name = file_path.split("/")[-1]
    
    # Returning file name and list of words
    return (file_name, words)

# Setting up Spark configuration
conf = SparkConf().setAppName("TF-IDF for bigrams").set("spark.executor.memory", "2g")
sc = SparkContext(conf=conf)

# Creating a SparkSession
spark = SparkSession(sc)

# Loading text files from a directory into Spark RDD
lines = sc.wholeTextFiles("/cosc6339_s17/books-longlist/")

# Applying the processing function to each file and converting the result into a DataFrame
tokenized_data = lines.map(lambda context: process_file(*context)).toDF(["bookname", "words"])

# Generating bigrams from the tokenized words
bigram = NGram(n=2, inputCol="words", outputCol="bigrams")
bigram_df = bigram.transform(tokenized_data)

# Computing Term Frequency (TF) for bigrams using HashingTF
hashingTF = HashingTF(inputCol="bigrams", outputCol="bigram-tf")
tf_data = hashingTF.transform(bigram_df)

# Computing Inverse Document Frequency (IDF) for bigrams
idf = IDF(inputCol="bigram-tf", outputCol="bigram-tf-idf")
idf_model = idf.fit(tf_data)
tfidf_data = idf_model.transform(tf_data)

# Saving the TF-IDF data to an output directory
tfidf_data.rdd.saveAsTextFile("/bigd12/output2_2")

# Stopping SparkContext to release resources
sc.stop()
