import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.tsa.arima_model import ARIMA
from sklearn.metrics import mean_squared_error

# Load the data
df = pd.read_csv('Shampoo-Sales.csv')

# Preprocessing
df = df.iloc[:-1]
df['Month'] = '190' + df['Month']
df['Month'] = pd.to_datetime(df['Month'], format='%Y-%m')
df.rename(columns={'Sales of shampoo over a three year period': 'Shampoo Sales'}, inplace=True)
series = df.set_index('Month')['Shampoo Sales']

# Visualize the data
plt.figure(figsize=(20, 10))
plt.title('Shampoo Sales over a 3-year period')
plt.xlabel('Month')
plt.ylabel('Shampoo Sales')
series.plot()
plt.show()

# Decompose the time series
def decompose_series(series):
    sales_cycle, sales_trend = sm.tsa.filters.hpfilter(series)
    sales_df = pd.DataFrame({'Sales': series.values, 'Trend': sales_trend, 'Cycle': sales_cycle}, index=series.index)
    return sales_df

sales_df = decompose_series(series)
sales_df.plot(title='Shampoo Sales, Sales Trend and Sales Cycle over a 3-year period', figsize=(20, 10))
plt.show()

# Exponential Weighted Moving Average (EWMA)
ewma_df = pd.DataFrame({'Actual Series': series, 'EWMA Series': series.ewm(span=3).mean()})
ewma_df.plot(figsize=(20, 10))
plt.show()

# Rolling statistics
def plot_rolling_statistics(series):
    plt.figure(figsize=(20, 10))
    series.rolling(3).mean().plot(label='3 month rolling mean')
    series.rolling(3).std().plot(label='3 month rolling standard deviation')
    series.plot(label='Actual data')
    plt.legend()
    plt.show()

plot_rolling_statistics(series)

# Seasonal Decomposition
def seasonal_decomposition(series):
    ets_result = sm.tsa.seasonal_decompose(series, model='additive')
    ets_result.plot()
    plt.gcf().set_size_inches(20, 10)
    plt.show()

seasonal_decomposition(series)

# Augmented Dickey-Fuller Test
def augmented_dickey_fuller_test(series):
    adfuller_result = sm.tsa.adfuller(series)
    print('Augmented Dickey-Fuller Test:')
    labels = ['ADF Test Statistic', 'p-value']
    for value, label in zip(adfuller_result, labels):
        print(label + ": " + str(value))

augmented_dickey_fuller_test(series)

# Making the time series stationary
def make_stationary(time_series):
    for i in range(0, 10):
        if i == 0:
            print('Actual Time Series')
        else:
            print(str(i) + '-Differenced Time Series')
        print('-' * 60)
        p_stationarity = augmented_dickey_fuller_test(time_series)[1]
        print("\nStationarity:")
        if p_stationarity <= 0.05:
            print('Data is Stationary')
            break
        else:
            print('Data is Non-Stationary\n')
            time_series = (time_series - time_series.shift(1))
            time_series.dropna(inplace=True)
    return i, time_series

d, stationary_series = make_stationary(series)
print('Number of times differenced = ', d)
stationary_series.plot()
plt.show()

# Autocorrelation and Partial Autocorrelation
def plot_autocorrelation(series):

    plt.show()

def plot_partial_autocorrelation(series):
    sm.graphics.tsa.plot_pacf(series)
    plt.show()

plot_autocorrelation(stationary_series)
plot_partial_autocorrelation(stationary_series)

# ARIMA Modeling
def fit_arima_model(series, order=(4, 1, 2)):
    model = ARIMA(series, order=order)
    arima_results = model.fit()
    print(arima_results.summary())
    return arima_results

arima_results = fit_arima_model(series)

# Plotting residuals
def plot_residuals(residuals):
    residuals.plot(figsize=(10, 10))
    plt.show()
    residuals.plot(kind='kde', figsize=(10, 10))
    plt.show()
    print(residuals.describe())

plot_residuals(pd.DataFrame(arima_results.resid))

# Forecasting
predicted_plot = arima_results.plot_predict()
plt.show()

# Rolling Forecast ARIMA Model
def rolling_forecast_arima(series, order=(4, 1, 2)):
    X = series.values
    size = int(len(X) * 0.66)
    train, test = X[0:size], X[size:len(X)]
    history = [x for x in train]
    predictions = list()
    for t in range(len(test)):
        model = ARIMA(history, order=order)
        model_fit = model.fit(disp=0)
        output = model_fit.forecast()
        yhat = output[0]
        predictions.append(yhat)
        obs = test[t]
        history.append(obs)
        print('Predicted=%f, Expected=%f' % (yhat, obs))
    error = mean_squared_error(test, predictions)
    print('Test MSE: %.3f' % error)
    plt.plot(test, color='blue', label='Actuals')
    plt.plot(predictions, color='red', label='Rolling forecast')
    plt.legend()
    plt.show()

rolling_forecast_arima(series)

# Grid search for ARIMA parameters
def grid_search_arima(series):
    X = series.values
    p = 0
    q = 0
    d = 1
    pdq = []
    aic = []
    for p in range(6):
        for q in range(4):
            try:
                model = ARIMA(X, order=(p, d, q)).fit()
                x = model.aic
                x1 = (p, d, q)
                print(x1, x)
                aic.append(x)
                pdq.append(x1)
            except:
                pass
    keys = pdq
    values = aic
    d = dict(zip(keys, values))
    minaic = min(d, key=d.get)
    for i in range(3):
        p = minaic[0]
        d = minaic[1]
        q = minaic[2]
    print("Best Model is :", (p, d, q))
    return p, d, q

p, d, q = grid_search_arima(series)

# Final ARIMA Model
arima_results_final = fit_arima_model(series, order=(p, d, q))

# Rolling Forecast with Final ARIMA Model
rolling_forecast_arima(series, order=(p, d, q))
