import re

def extract_replacements(grammar):
    # Regular expression pattern to extract replacements from the grammar
    syntax_re = r"""(?mx)
        ^(?:
          (?: (?P<comment> \s* $ ) |
              (?P<rule> (?P<pat>.+?) \s+ -> \s+ (?P<repl>.+) (?P<term>\.)? )
          )
        )$
        """

    # Extract replacements from the grammar using regular expressions
    replacements = []
    for matchobj in re.finditer(syntax_re, grammar):
        if matchobj.group('rule'):
            pat = matchobj.group('pat').strip()
            repl = matchobj.group('repl').strip()
            term = bool(matchobj.group('term'))
            replacements.append((pat, repl, term))
    return replacements

def replace(text, replacements):
    # Iterate until no further replacements can be made
    while True:
        for pat, repl, term in replacements:
            if pat in text:
                # Replace the pattern with its corresponding replacement
                text = text.replace(pat, repl, 1)
                if term:
                    # Terminate if the replacement rule is marked as terminating
                    return text
                break
        else:
            # If no replacements were made, return the final text
            return text
