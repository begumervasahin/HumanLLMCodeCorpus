import requests
from bs4 import BeautifulSoup
import os
import traceback

# Function to download an image from a URL and save it to a file
def download_image(url, filename):
    if os.path.exists(filename):
        print('File already exists:', filename)
        return
    
    try:
        response = requests.get(url, stream=True, timeout=60)
        response.raise_for_status()
        
        with open(filename, 'wb') as file:
            for chunk in response.iter_content(chunk_size=1024):
                if chunk:
                    file.write(chunk)
                    file.flush()
        
        print('Downloaded:', filename)
    
    except KeyboardInterrupt:
        if os.path.exists(filename):
            os.remove(filename)
        raise KeyboardInterrupt
    
    except Exception as e:
        print('Error downloading:', e)
        traceback.print_exc()
        if os.path.exists(filename):
            os.remove(filename)

# Ensure the directory to store downloaded images exists
if not os.path.exists('data/anime'):
    os.makedirs('data/anime')

# Define the range of pages to scrape
start_page = 1
end_page = 8000

# Iterate over each page
for page_number in range(start_page, end_page + 1):
    page_url = 'http://konachan.net/post?page=%d&tags=' % page_number
    
    try:
        # Send a GET request to the page URL and parse the HTML
        page_html = requests.get(page_url).text
        soup = BeautifulSoup(page_html, 'html.parser')
        
        # Find all image elements with class 'preview'
        for img in soup.find_all('img', class_="preview"):
            # Extract the image URL
            image_url = 'http:' + img['src']
            
            # Construct the filename to save the image
            filename = os.path.join('data/anime', os.path.basename(image_url))
            
            # Download the image
            download_image(image_url, filename)
        
        # Print progress
        print('Page %d / %d' % (page_number, end_page))
    
    except Exception as e:
        # Handle exceptions
        print('Error accessing page:', e)
