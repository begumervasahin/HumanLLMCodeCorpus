# This script reads data from a CSV file named "distance_matrix_1698.csv".
# It prints the length of the fifth line of the file plus 1698,
# and then prints the third element of that line.

# Author: 's'

# Open the file for reading
with open("distance_matrix_1698.csv", "r+") as file:
    # Initialize variables
    line_counter = 0
    fifth_line_elements = []

    # Iterate through each line in the file
    for line in file:
        # Increment the line counter
        line_counter += 1
        
        # If we've read 5 lines, store the elements of the fifth line
        if line_counter == 5:
            fifth_line_elements = line.split(",")
            break
    
    # Calculate and print the total elements in the fifth line plus 1698
    total_elements = len(fifth_line_elements) + 1698
    print("Total elements in the fifth line plus 1698:", total_elements)
    
    # Print the third element of the fifth line
    third_element = fifth_line_elements[2]
    print("Third element of the fifth line:", third_element)
