def add_value():
    # Get the amount the customer wants to add to their card
    cust_adding = float(input("Enter the amount you want to add to your card:\n"))
    print(f"Customer is adding ${cust_adding:.2f}")

    # Check if the added amount is in 5-cent increments
    if (cust_adding * 1000) % 5 != 0:
        print("Error: Amount must be in 5-cent increments.")
    # Check if the added amount exceeds $80
    elif cust_adding > 80.00:
        print("Sorry, you cannot add more than $80 to your card at one time.")
    else:
        # Decide which function to call based on the added amount
        if cust_adding <= 5.50:
            add_value_small(cust_adding)
        else:
            add_value_big(cust_adding)


def add_value_small(cust_adding):
    print("For amounts less than or equal to $5.50:")

    # Calculate the minimum amount needed to make the total divisible by $2.75
    min_amount = 2.75 - (cust_adding % 2.75)
    print(f"To make the total divisible by $2.75, add ${min_amount:.2f}")


def add_value_big(cust_adding):
    print("For amounts greater than $5.50:")

    # Get the current value on the card
    card_value_1 = float(input("What is the current value on your card?\n"))
    print(f"Current card value: ${card_value_1:.2f}")

    # Calculate the total value after adding the customer's input
    card_value_final = card_value_1 + cust_adding
    print(f"Total card value after adding: ${card_value_final:.2f}")

    # Calculate the next 3 amounts that are divisible by $2.75
    next_divisible = []
    for i in range(1, 4):
        amount = 2.75 * (card_value_final // 2.75 + i)
        next_divisible.append(amount)

    # Display the next 3 amounts that are divisible by $2.75
    print("Next 3 amounts divisible by $2.75:")
    for amount in next_divisible:
        print(f"${amount:.2f}")


if __name__ == "__main__":
    # Check if the card is new or existing
    is_new_card = input("Is this a new card? (Y/N)\n").strip().lower()
    if is_new_card == "y":
        add_value()
    elif is_new_card == "n":
        card_value_1 = float(input("What is the current value on your card?\n"))
        add_value()
    else:
