def kruskal(graph):
    # Ensure that the input graph is a dictionary
    assert isinstance(graph, dict)
    
    # Get a list of nodes in the graph
    nodes = list(graph.keys())
    
    # Initialize an empty set to keep track of visited nodes
    visited = set()
    
    # Initialize an empty list to store the final path
    path = []
    
    # Continue until all nodes are visited
    while len(visited) < len(nodes):
        # Initialize the shortest distance to infinity
        shortest_distance = float('inf')
        
        # Initialize variables to store the nodes of the shortest edge
        source_node = None
        destination_node = None
        
        # Iterate over all pairs of nodes in the graph
        for source in nodes:
            for destination in nodes:
                # Skip if both nodes are already visited or if they are the same node
                if source in visited and destination in visited or source == destination:
                    continue
                
                # Update the shortest distance and edge if a shorter edge is found
                if graph[source][destination] < shortest_distance:
                    shortest_distance = graph[source][destination]
                    source_node = source
                    destination_node = destination
        
        # Add the shortest edge to the path and mark its nodes as visited
        path.append((source_node, destination_node))
        visited.add(source_node)
        visited.add(destination_node)
    
    # Return the final minimum spanning tree path
    return path
