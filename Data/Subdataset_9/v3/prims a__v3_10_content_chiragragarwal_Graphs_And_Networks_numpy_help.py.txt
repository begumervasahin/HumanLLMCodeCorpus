import numpy as np

# Define the matrix as a string
matrix_string = '1 2 3 4; 5 6 7 8; 0 1 2 3'  # Removed unnecessary spaces

# Create a numpy matrix from the string
matrix = np.matrix(matrix_string)

# Print the shape of the matrix
print("Shape of the matrix:", matrix.shape)

# Access and print specific elements in the matrix
element_1_2 = matrix[0, 1]  # Element at row 0, column 1
element_3_4 = matrix[2, 3]  # Element at row 2, column 3
print(f"Element at row 0, column 1: {element_1_2}")
print(f"Element at row 2, column 3: {element_3_4}")

# Modify an element in the matrix
matrix[0, 1] = -8
print("Modified matrix:")
print(matrix)

# Print the second column of the matrix
second_column = matrix[:, 1]
print("Second column of the matrix:")
print(second_column)
