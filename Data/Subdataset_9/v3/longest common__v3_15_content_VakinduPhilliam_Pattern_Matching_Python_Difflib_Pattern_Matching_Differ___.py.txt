from difflib import Differ
from pprint import pprint

# Define the two texts to be compared
text1 = '''  1. Beautiful is better than ugly.
      2. Explicit is better than implicit.
      3. Simple is better than complex.
      4. Complex is better than complicated.
    '''.splitlines(keepends=True)

text2 = '''  1. Beautiful is better than ugly.
      3.   Simple is better than complex.
      4. Complicated is better than complex.
      5. Flat is better than nested.
    '''.splitlines(keepends=True)

# Create a Differ object to find the differences
differ = Differ()

# Generate a list of differences between text1 and text2
differences = list(differ.compare(text1, text2))

# Print the differences in a human-readable format
pprint(differences)

# Write the differences to an output file named 'output.txt'
with open('output.txt', 'w') as output_file:
    output_file.writelines(differences)
