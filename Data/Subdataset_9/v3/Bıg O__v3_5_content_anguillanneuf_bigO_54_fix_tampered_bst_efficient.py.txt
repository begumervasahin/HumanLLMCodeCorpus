class BinaryTreeNode:
    """
    Represents a node in a binary tree.
    """

    def __init__(self, value):
        """Initialize a binary tree node with a given value."""
        self.value = value
        self.left = None
        self.right = None
        self.wrong1 = None  # First wrongly swapped node
        self.wrong2 = None  # Second wrongly swapped node
        self.last_node = None  # Track the last node during traversal

    def in_order_traversal(self, node):
        """Perform an in-order traversal of the binary tree."""
        if node:
            self.in_order_traversal(node.left)
            print(node.value)
            self.in_order_traversal(node.right)

    def find_swapped_nodes(self, node):
        """
        Find the wrongly swapped nodes in the binary tree.

        Args:
            node (BinaryTreeNode): The current node being processed.
        """
        if not node:
            return
        if node.left:
            self.find_swapped_nodes(node.left)
        # Check for wrongly swapped nodes
        if not self.wrong1:
            if self.last_node and node.value < self.last_node.value:
                self.wrong1 = self.last_node
                self.wrong2 = node
        else:
            if self.wrong2:
                if node.value < self.wrong2.value:
                    self.wrong2 = node
        self.last_node = node
        if node.right:
            self.find_swapped_nodes(node.right)

    def swap_wrongly_swapped_nodes(self, root):
        """
        Swap the wrongly swapped nodes in the binary tree.

        Args:
            root (BinaryTreeNode): The root of the binary tree.
        """
        self.find_swapped_nodes(root)
        # Swap the values of the wrongly swapped nodes
        self.wrong1.value, self.wrong2.value = self.wrong2.value, self.wrong1.value


# Create a binary tree and test the methods
bt = BinaryTreeNode(7)
bt.left = BinaryTreeNode(4)
bt.left.left = BinaryTreeNode(1)
bt.left.right = BinaryTreeNode(5)
bt.right = BinaryTreeNode(11)

# Simulate swapping of nodes
bt.value, bt.left.left.value = bt.left.left.value, bt.value

# Perform in-order traversal before and after swapping
print("In-order traversal before swapping:")
bt.in_order_traversal(bt)
print("\nIn-order traversal after swapping:")
bt.swap_wrongly_swapped_nodes(bt)
bt.in_order_traversal(bt)
