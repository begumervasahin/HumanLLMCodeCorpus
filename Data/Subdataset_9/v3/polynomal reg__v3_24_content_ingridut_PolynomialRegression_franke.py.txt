# Importing necessary libraries
import numpy as np
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d import Axes3D
from matplotlib import cm
from matplotlib.ticker import LinearLocator, FormatStrFormatter

# Define the Franke function
def FrankeFunction(x, y):
    term1 = 0.75 * np.exp(-(0.25 * (9 * x - 2) ** 2) - 0.25 * ((9 * y - 2) ** 2))
    term2 = 0.75 * np.exp(-((9 * x + 1) ** 2) / 49.0 - 0.1 * (9 * y + 1))
    term3 = 0.5 * np.exp(-(9 * x - 7) ** 2 / 4.0 - 0.25 * ((9 * y - 3) ** 2))
    term4 = -0.2 * np.exp(-(9 * x - 4) ** 2 - (9 * y - 7) ** 2)
    return term1 + term2 + term3 + term4

# Generating grid points for x and y
x_values = np.arange(0, 1, 0.05)
y_values = np.arange(0, 1, 0.05)
x_grid, y_grid = np.meshgrid(x_values, y_values)  # Create 2D grid of x and y values

# Calculating z values using Franke function
z_values = FrankeFunction(x_grid, y_grid)

# Creating the plot
fig = plt.figure()  # Create a new figure
ax = fig.gca(projection='3d')  # Get current axes in 3D projection
surface_plot = ax.plot_surface(x_grid, y_grid, z_values, cmap=cm.coolwarm, linewidth=0, antialiased=False)  # Plot surface

# Customizing the plot
ax.set_zlim(-0.10, 1.40)  # Set limits for z-axis
ax.zaxis.set_major_locator(LinearLocator(10))  # Set major locator for z-axis
ax.zaxis.set_major_formatter(FormatStrFormatter('%.02f'))  # Set format for z-axis labels
color_bar = fig.colorbar(surface_plot, shrink=0.5, aspect=5)  # Add color bar to the plot

# Showing the plot
plt.show()
