class Node:
    def __init__(self, value):
        self.value = value
        self.left = None
        self.right = None

    def add_left(self, node):
        self.left = node

    def add_right(self, node):
        self.right = node

def countdown_recursive(x):
    """
    Recursively countdown from x to 0.
    """
    print(x)
    if x > 0:
        countdown_recursive(x - 1)

def factorial_recursive(x):
    """
    Compute factorial of x using recursive approach.
    """
    if x == 0:
        return 1
    else:
        return x * factorial_recursive(x - 1)

def factorial_tail_recursive(x, acc=1):
    """
    Compute factorial of x using tail recursion.
    """
    if x == 0:
        return acc
    else:
        return factorial_tail_recursive(x - 1, acc * x)

# Example binary tree
def build_binary_tree():
    root_node = Node(0)
    node1 = Node(1)
    node2 = Node(2)
    node3 = Node(3)
    node4 = Node(4)
    node5 = Node(5)

    root_node.add_left(node1)
    root_node.add_right(node2)
    node1.add_left(node3)
    node2.add_left(node4)
    node2.add_right(node5)
    return root_node

def tree_traverse_pre_order(root):
    """
    Traverse binary tree in pre-order and print node values.
    """
    print(root.value)
    if root.left:
        print("Left")
        tree_traverse_pre_order(root.left)
    if root.right:
        print("Right")
        tree_traverse_pre_order(root.right)

def tree_search(root, val):
    """
    Search for a value in the binary tree.
    """
    if root is None:
        return
    if root.value == val:
        print("Found")
    else:
        tree_search(root.left, val)
        tree_search(root.right, val)

def merge_sort(arr):
    """
    Perform merge sort on a list.
    """
    if len(arr) < 2:
        return arr
    mid = len(arr) // 2
    left_half = merge_sort(arr[:mid])
    right_half = merge_sort(arr[mid:])
    return merge(left_half, right_half)

def merge(left, right):
    """
    Merge two sorted arrays.
    """
    result = []
    i = j = 0
    while i < len(left) and j < len(right):
        if left[i] < right[j]:
            result.append(left[i])
            i += 1
        else:
            result.append(right[j])
            j += 1
    result.extend(left[i:])
    result.extend(right[j:])
    return result
