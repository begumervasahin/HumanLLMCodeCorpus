# Import necessary modules
import pybpsohd as pybp
import re

# Define a function to read the transactional dataset from a file
def read_dataset(file_name):
    """
    Reads a transactional dataset from a file and returns a list of transactions.
    Each transaction is represented as a list of integers.
    """
    transactions = []
    with open(file_name, 'r') as file:
        for line in file:
            # Extract integers from the line using regular expression
            items = re.findall(r'\d+', line)
            if items:
                transactions.append([int(item) for item in items])
    return transactions

if __name__ == '__main__':
    # Specify the file containing the transactional dataset
    dataset_file = "kosarak.dat"

    # Read the transactional dataset from the file
    dataset = read_dataset(dataset_file)

    # Parameters for mining long patterns
    min_support = 0.00001
    population_size = 30
    num_generations = 30
    inertia_weight = 0.5
    acceleration_1 = 1
    acceleration_2 = 1
    pattern_length = 10

    # Mine long patterns from the dataset
    patterns = pybp.getFP(dataset, min_support, population_size, num_generations,
                           inertia_weight, acceleration_1, acceleration_2, pattern_length)

    # Display the mined patterns
    for pattern in patterns:
        print(pattern)
