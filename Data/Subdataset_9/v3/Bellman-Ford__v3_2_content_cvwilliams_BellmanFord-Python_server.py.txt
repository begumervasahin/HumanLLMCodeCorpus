import pickle
import socket
import sys

# Define the maximum integer value
MAXINT = sys.maxsize

def print_table(table):
    """
    Print the table in a formatted manner.
    """
    print("DESTINATION-------INTERFACE------LINKCOST")
    for entry in table:
        dest, interface, cost = entry
        if interface == -1:
            print(f"    {dest}            {interface}           {cost}")
        elif interface == MAXINT:
            print(f"    {dest}            inf            inf")  
        else:
            print(f"    {dest}            {interface}            {cost}")

def receive_table_from_client(socket):
    """
    Receive table from the client.
    """
    table = socket.recv(1024)
    return pickle.loads(table)

def update_table(table1, table0):
    """
    Update the received table based on the local table.
    """
    for i in range(len(table1)):
        for j in range(len(table0)):
            num = table1[i][2] + int(table0[j][2])
            if num < table0[i][2]:
                table0[i][2] = num + 1
    return table0

def main():
    # Initialize the table
    table1 = [[0, 0, 1], [1, -1, 0], [2, 1, 1], [3, MAXINT, MAXINT]]
    print("Initial table: ")
    print_table(table1)

    # Create a socket
    s = socket.socket()
    port = 57171
    host = ''

    # Bind the socket to the port
    try:
        s.bind((host, port))
    except socket.error as msg:
        print(f'Bind failed. Error Code: {msg[0]}. Message: {msg[1]}')
        sys.exit()

    print('Socket bind complete')
    s.listen(1)
    client, addr = s.accept()
    print('Got connection from', addr)

    # Receive table from the client
    table0 = receive_table_from_client(client)

    # Print received and current tables
    print("Current table: ")
    print_table(table1)
    print("Received table: ")
    print_table(table0)

    # Update the received table
    updated_table = update_table(table1, table0)

    # Print updated table
    print("Updated table: ")
    print_table(updated_table)

    # Send the updated table back to the client
    client.send(pickle.dumps(updated_table))
    s.close()

if __name__ == "__main__":
    main()
