class BinaryTree:
    def __init__(self, root_value):
        """
        Initialize a binary tree with a root value.

        Parameters:
            root_value: The value of the root node.
        """
        self.key = root_value
        self.left_child = None
        self.right_child = None

    def insert_left(self, new_value):
        """
        Insert a new node as the left child of this node.

        Parameters:
            new_value: The value of the new left child node.
        """
        if self.left_child is None:
            self.left_child = BinaryTree(new_value)
        else:
            new_tree = BinaryTree(new_value)
            new_tree.left_child = self.left_child
            self.left_child = new_tree

    def insert_right(self, new_value):
        """
        Insert a new node as the right child of this node.

        Parameters:
            new_value: The value of the new right child node.
        """
        if self.right_child is None:
            self.right_child = BinaryTree(new_value)
        else:
            new_tree = BinaryTree(new_value)
            new_tree.right_child = self.right_child
            self.right_child = new_tree

    def get_right_child(self):
        """
        Get the right child of this node.

        Returns:
            The right child of this node.
        """
        return self.right_child

    def get_left_child(self):
        """
        Get the left child of this node.

        Returns:
            The left child of this node.
        """
        return self.left_child

    def set_root_value(self, value):
        """
        Set the value of the root node.

        Parameters:
            value: The new value for the root node.
        """
        self.key = value

    def get_root_value(self):
        """
        Get the value of the root node.

        Returns:
            The value of the root node.
        """
        return self.key

# Test the BinaryTree class
tree = BinaryTree('a')
print("Root value:", tree.get_root_value())
print("Left child:", tree.get_left_child())
tree.insert_left('b')
print("Left child after insertion:", tree.get_left_child())
print("Root value of left child:", tree.get_left_child().get_root_value())
tree.insert_right('c')
print("Right child:", tree.get_right_child())
print("Root value of right child:", tree.get_right_child().get_root_value())
tree.get_right_child().set_root_value('hello')
print("Updated root value of right child:", tree.get_right_child().get_root_value())
