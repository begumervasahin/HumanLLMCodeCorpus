from math import sqrt

# Parameters
DISTANCE_THRESHOLD = 0.1
UNKNOWN_DIRECTION = 0
UP_DIRECTION = 1
DOWN_DIRECTION = 2

def find_peaks(data, max_distance):
    peak_values = [data[0]]  # Initialize list to store peak values
    peak_indices = [0]       # Initialize list to store peak indices
    current_direction = UNKNOWN_DIRECTION  # Initialize direction as unknown

    # Loop through the data points
    for index in range(1, len(data)):
        if current_direction == UNKNOWN_DIRECTION:
            # Determine initial direction of data
            if data[index] > data[index - 1]:
                current_direction = UP_DIRECTION
            elif data[index] < data[index - 1]:
                current_direction = DOWN_DIRECTION
            continue

        if current_direction == UP_DIRECTION and data[index] < data[index - 1]:
            # If direction changes from up to down, check distance threshold
            current_direction = DOWN_DIRECTION
            distance = (data[index - 1] - peak_values[-1]) / max_distance
            if distance >= DISTANCE_THRESHOLD:
                peak_values.append(data[index - 1])
                peak_indices.append(index - 1)
            continue

        if current_direction == DOWN_DIRECTION and data[index] > data[index - 1]:
            # If direction changes from down to up, check distance threshold
            current_direction = UP_DIRECTION
            distance = (peak_values[-1] - data[index - 1]) / max_distance
            if distance >= DISTANCE_THRESHOLD:
                peak_values.append(data[index - 1])
                peak_indices.append(index - 1)
            continue

    return peak_values, peak_indices

def calculate_slopes(peaks):
    slopes = []  # Initialize list to store slopes

    # Calculate slopes between consecutive peaks
    for index in range(1, len(peaks[0])):
        slope = (peaks[0][index] - peaks[0][index - 1]) / (peaks[1][index] - peaks[1][index - 1])
        slopes.append(slope)

    return slopes

def calculate_means(data, peaks):
    means = []  # Initialize list to store mean values

    # Calculate mean values between consecutive peaks
    for i in range(1, len(peaks[0])):
        total = 0.0  # Initialize sum of data points between peaks
        for j in range(peaks[1][i-1], peaks[1][i]):
            total += data[j]  # Sum data points between peaks
        mean = total / (peaks[1][i] - peaks[1][i-1])  # Calculate mean
        means.append(mean)

    return means

def calculate_distances(slopes, peaks, means):
    distances = []  # Initialize list to store distances

    # Calculate distances based on slope and mean between consecutive peaks
    for index in range(1, len(slopes)):
        distance = sqrt((slopes[index] - slopes[index-1])**2 + 
                        (peaks[1][index] - peaks[1][index - 1])**2 + 
                        (means[index] - means[index-1])**2)
        distances.append(distance)

    return distances

# Sample data
data = (10.0, 11.0, 14.0, 13.0, 40.0, 39.0, 38.0, 60.0, 59.0, 62.0, 80.0, 81.0, 79.0, 78.0, 50.0, 52.0, 51.0, 53.0, 10.0, 9.0, 8.0, 50.0)

# Find peaks in the data
peaks = find_peaks(data, 1)

# Calculate slopes between peaks
slopes = calculate_slopes(peaks)

# Calculate mean values between peaks
means = calculate_means(data, peaks)

# Calculate distances between peaks based on slope and mean
distances = calculate_distances(slopes, peaks, means)

# Print the calculated distances
print(distances)
