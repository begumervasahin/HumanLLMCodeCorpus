import time

def add_lists(list1, list2):
    """Add corresponding elements of two lists."""
    return [x + y for x, y in zip(list1, list2)]

def generate_consecutive_range(start, end):
    """Generate a range of consecutive numbers from start to end."""
    return range(start, end)

def main():
    # Generating two lists of consecutive numbers
    nums1 = generate_consecutive_range(0, 9999999)
    nums2 = generate_consecutive_range(100, 10000100)

    # Measuring the start time
    start_time = time.time()

    # Adding the two lists
    result = add_lists(nums1, nums2)

    # Calculating the elapsed time
    elapsed_time = time.time() - start_time

    # Printing the elapsed time and the length of the result list
    print("Time elapsed:", elapsed_time)
    print("Result list length:", len(result))

    # Printing numbers in the result list that are divisible by 777777
    for item in result:
        if item % 777777 == 0:
            print(item)

if __name__ == "__main__":
    main()
