import pandas as pd
import numpy as np

def read_data(filename, separator='\t'):
    """Reads data from a CSV file into a DataFrame."""
    return pd.read_csv(filename, sep=separator)

def calculate_means(data):
    """Calculates the mean values of the variables."""
    return data.mean()

def calculate_sums(data):
    """Calculates the sum values of the variables."""
    return data.sum()

def calculate_sums_of_products(X, X1, Y):
    """Calculates the sum of products between variables."""
    return np.sum(X * Y), np.sum(X1 * Y), np.sum(X * X1)
