import queue
import threading
from collections import defaultdict

class Interface:
    def __init__(self, max_size=0):
        self.in_queue = queue.Queue(max_size)
        self.out_queue = queue.Queue(max_size)

    def get(self, direction):
        try:
            return self.in_queue.get(False) if direction == 'in' else self.out_queue.get(False)
        except queue.Empty:
            return None

    def put(self, packet, direction, block=False):
        queue_to_use = self.out_queue if direction == 'out' else self.in_queue
        queue_to_use.put(packet, block)

class NetworkPacket:
    DST_LENGTH = 5
    PROT_LENGTH = 1

    def __init__(self, destination, protocol, data):
        self.destination = destination
        self.protocol = protocol
        self.data = data

    def __str__(self):
        return self.to_byte_string()

    def to_byte_string(self):
        byte_string = str(self.destination).zfill(self.DST_LENGTH)
        byte_string += '1' if self.protocol == 'data' else '2'
        byte_string += self.data
        return byte_string

    @classmethod
    def from_byte_string(cls, byte_string):
        destination = byte_string[:cls.DST_LENGTH].strip('0')
        protocol = 'data' if byte_string[cls.DST_LENGTH] == '1' else 'control'
        data = byte_string[cls.DST_LENGTH + cls.PROT_LENGTH:]
        return cls(destination, protocol, data)

class Host:
    def __init__(self, address):
        self.address = address
        self.interfaces = [Interface()]
        self.stop = False

    def __str__(self):
        return self.address

    def send_packet(self, destination, data):
        packet = NetworkPacket(destination, 'data', data)
        print(f'{self}: SENDING PACKET "{packet}"')
        self.interfaces[0].put(packet.to_byte_string(), 'out')

    def receive_packet(self):
        packet_string = self.interfaces[0].get('in')
        if packet_string:
            print(f'{self}: RECEIVED PACKET "{packet_string}"')
            if self.address == 'H2':
                self.send_packet('H1', 'The way you look should be a sin, you my sensation')

    def run(self):
        print(f'{self}: Starting')
        while True:
            self.receive_packet()
            if self.stop:
                print(f'{self}: Ending')
                return

class Router:
    def __init__(self, name, neighbor_costs, max_queue_size):
        self.stop = False
        self.name = name
        self.interfaces = [Interface(max_queue_size) for _ in range(len(neighbor_costs))]       
        self.neighbors = neighbor_costs
        self.routing_table = defaultdict(dict)
        self.total_routing_table = defaultdict(dict)
        self.initialize_routing_tables()
        print(f'{self}: Initialized routing table')
        self.print_routing_table()

    def initialize_routing_tables(self):
        for neighbor_name, neighbor_info in self.neighbors.items():
            for interface, cost in neighbor_info.items():
                self.total_routing_table[self.name][neighbor_name] = [cost]

    def __str__(self):
        return self.name

    def process_queues(self):
        for i, interface in enumerate(self.interfaces):
            packet_string = interface.get('in')
            if packet_string:
                packet = NetworkPacket.from_byte_string(packet_string)
                if packet.protocol == 'data':
                    self.forward_packet(packet, i)
                elif packet.protocol == 'control':
                    self.update_routing_table(packet, i)
                else:
                    raise ValueError(f'Unknown packet type in packet {packet}')

    def forward_packet(self, packet, i):
        try:
            route = self.routing_table.get(str(packet.destination))
            interface, cost = next(iter(route.items()))
            self.interfaces[int(interface)].put(packet.to_byte_string(), 'out', True)
            print(f'{self}: forwarding packet "{packet}" from interface {i} to {interface}')
        except queue.Full:
            print(f'{self}: packet "{packet}" lost on interface {i}')

    def send_routing_table(self, i):
        print('sending routing table')
        contents = f'{self.name}->'
        for k, j in self.routing_table.items():
            for link, cost in j.items():
                contents += f'{k}_{link}_{cost} '
        packet = NetworkPacket(0, 'control', contents)
        self.interfaces[i].put(packet.to_byte_string(), 'out', True)

    def update_routing_table(self, packet, i):
        update_flag = 0
        name, route_info = packet.data.split('->')
        for item in route_info.split(' '):
            if item:
                current_node, _, cost = item.split('_')
                cost = int(cost)
                self.total_routing_table[name][current_node] = [cost]
                if current_node == self.name:
                    neighbor_interface = int(list(self.neighbors[name].keys())[0])      
                    self.routing_table[current_node] = {neighbor_interface: 0}
                    self.total_routing_table[self.name][current_node] = [0]
                    continue
                else:
                    if current_node not in self.neighbors and current_node not in self.routing_table:
                        neighbor_cost = list(self.neighbors[name].values())[0]
                        neighbor_interface = int(list(self.neighbors[name].keys())[0])  
                        self.routing_table[current_node] = {neighbor_interface: cost + neighbor_cost}
                        self.total_routing_table[self.name][current_node] = [cost + neighbor_cost]
                        update_flag = 1
                    elif current_node not in self.neighbors and current_node in self.routing_table:
                        neighbor_cost = list(self.neighbors[name].values())[0]
                        neighbor_interface = int(list(self.neighbors[name].keys())[0])  
                        current_cost = list(self.routing_table[current_node].values())[0]
                        if current_cost > neighbor_cost + cost:
                            self.routing_table[current_node] = {neighbor_interface: cost + neighbor_cost}
                            self.total_routing_table[self.name][current_node] = [cost + neighbor_cost]
                            update_flag = 1
        if update_flag == 1:
            for neighbor_name, neighbor_info in self.neighbors.items():
                for interface, cost in neighbor_info.items():
                    if 'R' in neighbor_name:
                        self.send_routing_table(interface)

    def print_routing_table(self):
        print(f'cost table for {self}:')
        sorted_table = OrderedDict(sorted(self.total_routing_table.items()))
        interfaces = sorted(set(link for neighbor_info in sorted_table.values() for link in neighbor_info.keys()))
        for interface in interfaces:
            print(f'{interface}\t', end='')
            for item, neighbor_info in sorted_table.items():
                for link, cost_info in neighbor_info.items():
                    if link == interface:
                        print(f'{cost_info[0]}\t', end='')
        print('\n')

    def run(self):
        print(f'{self}: Starting')
        while True:
            self.process_queues()
            if self.stop:
                print(f'{self}: Ending')
                return
