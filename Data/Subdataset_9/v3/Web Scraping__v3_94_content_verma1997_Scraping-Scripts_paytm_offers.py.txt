# Function to get a random user-agent header
def get_random_user_agent():
    return {'User-Agent': choice(USER_AGENTS)}

# Function to scrape flight offers from Paytm
def scrape_paytm_offers():
    url = "https://paytm.com/flights"
    response = requests.get(url, headers=get_random_user_agent())
    soup = BeautifulSoup(response.text, 'html.parser')

    # Extracting offer details
    offer_elements = soup.find_all("li", {"class": "slider-slide"})
    offers = []
    for element in offer_elements:
        link = element.find("a")
        offer = {
            'link': link['href'],
            'image_link': link.img['src'],
            'promocode': element.find("p", {"class": "PromoCode"}).text.replace("Use promocode : ", "")
        }
        offers.append(offer)

    # Writing offer details to a CSV file
    filename = "paytm_offers.csv"
    with open(filename, "w") as file:
        file.write("Link,Image Link,Promocode\n")
        for offer in offers:
            file.write(f"{offer['link']},{offer['image_link']},{offer['promocode']}\n")

# Main function
if __name__ == "__main__":
    scrape_paytm_offers()
