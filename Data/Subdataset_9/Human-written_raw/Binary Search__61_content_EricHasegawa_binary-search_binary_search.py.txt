def binarySearch():
    while True :
        try:
            target = int(input("Enter an integer: "))
            break
        except ValueError:
            print("You did not enter an integer , try again.")
    numberlist = open('numberlist.txt')
    split_list = numberlist.read().split(", ")
    print(split_list, "HELLO")
    results = list(map(int, split_list))
    sorted_results = sorted(results)
    high = len(sorted_results) - 1
    low = 0
    while (high > low):
        middle = ((high + low)//2)
        if (target == sorted_results[middle]) :
            print ("Target found")
            break
        elif (middle == low)  :
            print ("Target not found")
            break
        elif (target > sorted_results[middle]) :
            low = middle
        elif (target < sorted_results[middle]) :
            high = middle
    while True :
        yn = input("Would you like to test another number? (y/n): ")
        if (yn == 'y'):
            binarySearch()
            break
        elif (yn == 'n') :
            print("Roger Dodger")
            break
        else :
            print("You did not enter y or n, try again.")
binarySearch()
