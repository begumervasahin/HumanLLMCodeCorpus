STFT Method:
The original STFT method in numpy may be too complex for some use cases.
This script provides a simplified version of the Short-Time Fourier Transform (STFT) method.
"""

import math
import numpy as np

def stft(source, winlen, slide):
    """
    Computes the Short-Time Fourier Transform (STFT) of a given signal.

    Args:
        source (list or np.ndarray): Input signal data.
        winlen (int): Length of the window used for the STFT.
        slide (int): Number of samples to slide the window.

    Returns:
        np.ndarray: STFT of the input signal, with shape (winlen, num_windows).
    """
    # Convert source to a numpy array if it isn't already
    data = np.asarray(source)

    # Determine the nearest power of 2 greater than or equal to the length of the signal
    topindex = 2 ** power2_near(len(data))
    
    # Pad the data with zeros if necessary
    if topindex - len(data) > 0:
        data = np.pad(data, (0, topindex - len(data)), mode='constant')

    # Calculate the number of windows
    win_num = int((len(data) - winlen) / slide) + 1
    
    # Create an empty matrix to hold the windowed data
    window_matrix = np.zeros((winlen, win_num))

    # Fill the matrix with windowed segments of the signal
    for i in range(win_num):
        window_matrix[:, i] = data[slide * i:slide * i + winlen]
    
    # Compute the FFT for each window
    complex_map = np.fft.fft(window_matrix, axis=0)
    
    return complex_map

def power2_near(k):
    """
    Finds the smallest power of 2 greater than or equal to k.

    Args:
        k (int): The input number.

    Returns:
        int: The power of 2 such that 2^power >= k.
    """
    assert k > 0, "Input must be greater than 0"
    
    power = 0
    while k > 1:
        k = math.ceil(k / 2)
        power += 1

    return power

# Example usage
if __name__ == "__main__":
    # Example signal
    signal = np.sin(2 * np.pi * np.arange(1000) / 100)
    
    # Parameters for STFT
    window_length = 64
    slide_step = 32
    
    # Compute STFT
    stft_result = stft(signal, window_length, slide_step)
    
    # Print the shape of the result to confirm dimensions
    print("STFT result shape:", stft_result.shape)
