def insertion_sort(arr, verbose=0, desc=False):
    """
    Perform an insertion sort on the provided list.

    Parameters:
    arr (list): The list to be sorted.
    verbose (int): Level of verbosity. 0 for no output, 1 for iteration-level output, 2 for sub-level output.
    desc (bool): If True, the sorted list will be returned in descending order.

    Returns:
    list: The sorted list.
    """
    n = len(arr)
    
    for i in range(1, n):
        key = arr[i]
        j = i - 1

        # Move elements of arr[0..i-1], that are greater than key,
        # to one position ahead of their current position
        while j >= 0 and arr[j] > key:
            arr[j + 1] = arr[j]
            j -= 1
            if verbose == 2:
                print("  sub:", j, " :: ", arr)
        
        arr[j + 1] = key
        
        if verbose >= 1:
            print(f"Iteration {i}: {arr}")
    
    if desc:
        arr = arr[::-1]
    
    return arr

# Example usage:
if __name__ == "__main__":
    sample_array = [12, 11, 13, 5, 6]
    sorted_array = insertion_sort(sample_array, verbose=1, desc=False)
    print("Sorted array:", sorted_array)
