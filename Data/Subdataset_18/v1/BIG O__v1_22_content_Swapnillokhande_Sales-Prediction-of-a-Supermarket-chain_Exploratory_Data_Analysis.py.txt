import pandas as pd
import numpy as np
from sklearn.preprocessing import LabelEncoder

# Load the datasets
train_data = pd.read_csv("D:/Github/Big Mart Sales/Train_Data.csv", na_values='.')
test_data = pd.read_csv("D:/Github/Big Mart Sales/Test_Data.csv")

# Add source column to distinguish between train and test data
train_data['source'] = 'train_data'
test_data['source'] = 'test_data'

# Combine the datasets
comb_data = pd.concat([train_data, test_data], ignore_index=True)

# Print dataset shapes
print("Number of rows and columns in Train dataset:", train_data.shape)
print("Number of rows and columns in Test dataset:", test_data.shape)
print("Number of rows and columns in combined dataset:", comb_data.shape)

# Summary for unique values and missing values
print("\nSummary for unique values\n", comb_data.apply(lambda x: len(x.unique())))
print("\nSummary for missing values\n", comb_data.isnull().sum())
print("\nPercentage of missing values\n", comb_data.isnull().sum() / comb_data.shape[0] * 100)

# Statistical Summary of Item Visibility
print("\nStatistical Summary of Item Visibility\n", comb_data['Item_Visibility'].describe())

# Statistical Summary of the combined data
print("\nStatistical Summary of combined data\n", comb_data.describe())

# Frequency of categories
category_columns = [col for col in comb_data.dtypes.index if comb_data.dtypes[col] == 'object']
category_columns = [col for col in category_columns if col not in ['Item_Identifier', 'Outlet_Identifier', 'source']]
for column in category_columns:
    print(f'\nFrequency of Categories for variable {column}')
    print(comb_data[column].value_counts())

# Impute missing values in Item_Weight
item_avg_weight = comb_data.pivot_table(values='Item_Weight', index='Item_Identifier')
comb_data['Item_Weight'] = comb_data.groupby('Item_Identifier')['Item_Weight'].transform(lambda x: x.fillna(x.mean()))

# Impute missing values in Outlet_Size
outlet_size_mode = comb_data.pivot_table(values='Outlet_Size', columns='Outlet_Type', aggfunc=lambda x: x.mode()[0])
comb_data['Outlet_Size'] = comb_data.groupby('Outlet_Type')['Outlet_Size'].apply(lambda x: x.fillna(x.mode()[0]))

# Replace zeros in Item_Visibility with mean visibility of the same item
comb_data['Item_Visibility'] = comb_data.groupby('Item_Identifier')['Item_Visibility'].transform(lambda x: x.replace(0, x.mean()))

# Add new features
comb_data['Outlet_Years'] = 2013 - comb_data['Outlet_Establishment_Year']
comb_data['Item_Type_Combined'] = comb_data['Item_Identifier'].apply(lambda x: x[0:2])
comb_data['Item_Type_Combined'] = comb_data['Item_Type_Combined'].map({'FD': 'Food', 'NC': 'Non-Consumable', 'DR': 'Drinks'})

# Modify Item_Fat_Content
comb_data['Item_Fat_Content'] = comb_data['Item_Fat_Content'].replace({'LF': 'Low Fat', 'reg': 'Regular', 'low fat': 'Low Fat'})
comb_data.loc[comb_data['Item_Type_Combined'] == "Non-Consumable", 'Item_Fat_Content'] = "Non-Edible"

# Create Item_Visibility_MeanRatio feature
visibility_avg = comb_data.pivot_table(values='Item_Visibility', index='Item_Identifier')
comb_data['Item_Visibility_MeanRatio'] = comb_data.apply(lambda x: x['Item_Visibility'] / visibility_avg.loc[x['Item_Identifier']][0], axis=1)

# Encode categorical variables
le = LabelEncoder()
comb_data['Outlet'] = le.fit_transform(comb_data['Outlet_Identifier'])
var_mod = ['Item_Fat_Content', 'Outlet_Location_Type', 'Outlet_Size', 'Item_Type_Combined', 'Outlet_Type', 'Outlet']
for var in var_mod:
    comb_data[var] = le.fit_transform(comb_data[var])
comb_data = pd.get_dummies(comb_data, columns=var_mod)

# Drop unnecessary columns
comb_data.drop(['Item_Type', 'Outlet_Establishment_Year'], axis=1, inplace=True)

# Split combined data back into train and test sets
train = comb_data[comb_data['source'] == "train_data"].drop(['source'], axis=1)
test = comb_data[comb_data['source'] == "test_data"].drop(['source', 'Item_Outlet_Sales'], axis=1)

# Save the modified datasets
train.to_csv("D:/Github/Big Mart Sales/train_modified.csv", index=False)
test.to_csv("D:/Github/Big Mart Sales/test_modified.csv", index=False)
