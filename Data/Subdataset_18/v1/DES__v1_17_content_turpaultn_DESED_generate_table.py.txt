import re

def concat_files(readme):
    """
    Concatenates the contents of files included in the README file.

    Parameters
    ----------
    readme : str
        The path to the README file.
    """
    result_doc = ""
    matched = False

    # Read the contents of the README file
    with open(readme, "r", encoding="utf-8") as f:
        lines = f.readlines()

    for line in lines:
        if not matched:
            result_doc += line

        match = re.search(r"<!--\s*include\s+(.+?)\s*-->", line)
        if match and not matched:
            matched = True
            include_filename = match.group(1).strip()

            # Read the contents of the included file
            with open(include_filename, "r", encoding="utf-8") as inc_file:
                result_doc += inc_file.read()

        if re.search(r"<!--\s*end\s*-->", line):
            matched = False
            result_doc += "\n" + line

    # Write the concatenated result back to the README file
    with open(readme, "w", encoding="utf-8") as f:
        f.write(result_doc)

if __name__ == '__main__':
    concat_files("README.md")
