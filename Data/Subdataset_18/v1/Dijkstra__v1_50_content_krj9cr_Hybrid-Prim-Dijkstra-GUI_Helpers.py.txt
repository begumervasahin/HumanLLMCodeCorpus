import math
import matplotlib.pyplot as plt
from sympy import cos, sin, pi

class Node:
    def __init__(self, x, y, idx):
        self.x = x
        self.y = y
        self.idx = idx

def points_in_circumference(radius, num_points=100):
    """
    Generate points evenly distributed around a circle.

    Parameters:
    radius (float): The radius of the circle.
    num_points (int): The number of points to generate.

    Returns:
    list: A list of tuples representing the points (x, y) on the circumference.
    """
    return [(0, 0)] + [(cos(2 * pi / num_points * x) * radius, 
                        sin(2 * pi / num_points * x) * radius) 
                       for x in range(num_points + 1)]

def distance(node1, node2):
    """
    Calculate the Euclidean distance between two nodes.

    Parameters:
    node1 (Node): The first node.
    node2 (Node): The second node.

    Returns:
    float: The Euclidean distance between the two nodes.
    """
    return math.sqrt((node1.x - node2.x) ** 2 + (node1.y - node2.y) ** 2)

def generate_graph(radius=1.0, num_points=8):
    """
    Generate a graph where nodes are points on the circumference of a circle.

    Parameters:
    radius (float): The radius of the circle.
    num_points (int): The number of points (nodes) on the circumference.

    Returns:
    tuple: A tuple containing:
        - list of points (x, y)
        - list of Node objects
        - adjacency matrix with distances
    """
    points = points_in_circumference(radius, num_points)
    nodes = [Node(x, y, i) for i, (x, y) in enumerate(points)]
    edges = []

    for i in range(len(nodes)):
        row = []
        for j in range(len(nodes)):
            if i == j:
                row.append(0)
            else:
                row.append(distance(nodes[i], nodes[j]))
        edges.append(row)

    return points, nodes, edges

def plot_graph(points, edges, radius=1.0):
    """
    Plot the graph with nodes and edges.

    Parameters:
    points (list of tuples): List of points (x, y).
    edges (list of tuples): List of edges where each edge is a tuple of Node objects.
    radius (float): The radius of the circle used for setting plot limits.
    """
    x = [p[0] for p in points]
    y = [p[1] for p in points]

    # Plot nodes
    plt.plot(x, y, 'ro')

    # Plot edges
    for edge in edges:
        plt.plot([edge[0].x, edge[1].x], [edge[0].y, edge[1].y], 'b-', lw=1)

    # Set plot limits and aspect ratio
    size = radius * 2
    plt.axis([-size, size, -size, size])
    plt.grid(True)
    plt.gca().set_aspect('equal', 'datalim')

    # Display the plot
    plt.show()

# Example usage
if __name__ == "__main__":
    # Generate graph data
    points, nodes, edges = generate_graph(radius=1.0, num_points=8)

    # Convert edge adjacency matrix to edge list for plotting
    edge_list = [(nodes[i], nodes[j]) for i in range(len(nodes)) for j in range(len(nodes)) if i != j]

    # Plot the graph
    plot_graph(points, edge_list, radius=1.0)
