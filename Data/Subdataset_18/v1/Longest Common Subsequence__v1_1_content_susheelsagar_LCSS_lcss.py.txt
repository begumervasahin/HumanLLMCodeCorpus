def cal_sim(traj1, traj2):
    score = 0
    while traj1 and traj2:
        if traj1[-1][2] == traj2[-1][2] and traj1[-1][0] == traj2[-1][0]:
            score += 1
            traj1.pop()
            traj2.pop()
        elif traj1[-1][2] > traj2[-1][2]:
            traj1.pop()
        elif traj1[-1][2] < traj2[-1][2]:
            traj2.pop()
        else:
            if len(traj1) > 1 and traj1[-2][2] == traj2[-1][2]:
                traj1.pop()
            elif len(traj2) > 1 and traj1[-1][2] == traj2[-2][2]:
                traj2.pop()
            else:
                traj1.pop()
                traj2.pop()
    return score
