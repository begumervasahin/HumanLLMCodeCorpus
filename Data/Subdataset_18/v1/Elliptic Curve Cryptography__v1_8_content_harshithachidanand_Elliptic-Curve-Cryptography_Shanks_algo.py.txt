def print_shanks_algorithm():
    # Print header
    print("Shank's Algorithm!\n")
    
    # Print S table
    print("\n\nS table\n")
    for x in range(0, 46):
        val = (5**(45 * x)) % 2017
        print(f"{x} {val}\n")
    
    # Print T table
    print("\n\nT table\n")
    for y in range(0, 46):
        val = (1736 * (5**y)) % 2017
        print(f"{y}  {val}\n")
    
    # Find matching values between S and T tables
    print("\n\nMatching Values\n")
    s_values = {}
    for x in range(0, 46):
        x_val = (5**(45 * x)) % 2017
        s_values[x_val] = x

    for y in range(0, 46):
        y_val = (1736 * (5**y)) % 2017
        if y_val in s_values:
            print(f"{s_values[y_val]} {y}\n")

# Run the function
print_shanks_algorithm()
