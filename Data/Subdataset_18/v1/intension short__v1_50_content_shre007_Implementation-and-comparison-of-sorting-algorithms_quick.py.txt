import statistics

def find_pivot_position(arr, first, last):
    low = arr[first]
    high = arr[last]
    mid = (first + last) // 2
    
    # Determine the median of the first, middle, and last elements
    pivot_value = statistics.median([low, arr[mid], high])
    
    # Find the index of the pivot value
    if pivot_value == low:
        pivot_index = first
    elif pivot_value == high:
        pivot_index = last
    else:
        pivot_index = mid
    
    # Swap the pivot with the last element
    arr[last], arr[pivot_index] = arr[pivot_index], arr[last]
    pivot = arr[last]
    
    left = first
    right = last - 1
    
    # Partitioning the array around the pivot
    while True:
        while left <= right and arr[left] <= pivot:
            left += 1
        while left <= right and arr[right] >= pivot:
            right -= 1
        if left > right:
            break
        else:
            arr[left], arr[right] = arr[right], arr[left]
    
    arr[first], arr[right] = arr[right], arr[first]
    return right

def quicksort(arr, first, last):
    if first < last:
        pivot = find_pivot_position(arr, first, last)
        quicksort(arr, first, pivot - 1)
        quicksort(arr, pivot + 1, last)
