def partition(arr, low, high):
    i = low - 1
    pivot = arr[high][1]  # Choose the last element's second value as the pivot

    for j in range(low, high):
        if arr[j][1] <= pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]

    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    return i + 1

def quickSort(arr, low, high):
    if low < high:
        pi = partition(arr, low, high)
        quickSort(arr, low, pi - 1)
        quickSort(arr, pi + 1, high)

# Read the data from the file and filter based on the criterion
with open("db2.txt", encoding="utf-8") as text:
    lines = []
    for line in text:
        test = line.split(",")
        if test[3].strip() == "LOS OLIVOS":
            lines.append((float(test[5]), float(test[6])))

# Sort the filtered data using quicksort
quickSort(lines, 0, len(lines) - 1)

# Print the sorted data
for line in lines:
    print(line)

# Write the sorted data to a new file
with open("testSET4.txt", "w") as write:
    for line in lines:
        write.write(f"{line[0]},{line[1]}\n")

# Print the number of filtered and sorted lines
print(len(lines))
