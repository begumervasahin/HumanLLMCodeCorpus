# Importing necessary libraries
import pandas as pd
import matplotlib.pyplot as plt
from apyori import apriori

# Data Preprocessing
def load_data(file_path):
    """
    Load and preprocess the dataset from the given file path.

    Args:
    file_path (str): Path to the CSV file containing the dataset.

    Returns:
    list: List of transactions, where each transaction is a list of items.
    """
    dataset = pd.read_csv(file_path, header=None)
    transactions = []
    for i in range(0, len(dataset)):
        transactions.append([str(dataset.values[i, j]) for j in range(0, 20)])
    return transactions

def train_apriori(transactions, min_support, min_confidence, min_lift, min_length):
    """
    Train the Apriori algorithm on the given transactions.

    Args:
    transactions (list): List of transactions.
    min_support (float): Minimum support.
    min_confidence (float): Minimum confidence.
    min_lift (float): Minimum lift.
    min_length (int): Minimum length of the rules.

    Returns:
    list: List of rules generated by the Apriori algorithm.
    """
    rules = apriori(transactions, min_support=min_support, min_confidence=min_confidence, min_lift=min_lift, min_length=min_length)
    return list(rules)

def visualize_rules(rules):
    """
    Visualize the rules generated by the Apriori algorithm.

    Args:
    rules (list): List of rules generated by the Apriori algorithm.
    """
    for rule in rules:
        items = [x for x in rule.items]
        print(f"Rule: {items}")
        print(f"Support: {rule.support}")
        print(f"Confidence: {rule.ordered_statistics[0].confidence}")
        print(f"Lift: {rule.ordered_statistics[0].lift}")
        print("-" * 20)

def plot_item_frequencies(rules):
    """
    Plot item frequencies from the generated rules.

    Args:
    rules (list): List of rules generated by the Apriori algorithm.
    """
    items = []
    supports = []
    for rule in rules:
        for item in rule.items:
            items.append(item)
            supports.append(rule.support)
    
    plt.figure(figsize=(10, 6))
    plt.bar(items, supports, color='blue')
    plt.xlabel('Items')
    plt.ylabel('Support')
    plt.title('Item Frequencies')
    plt.xticks(rotation=90)
    plt.show()

if __name__ == "__main__":
    # Load data
    file_path = 'F://ML//Machine Learning A-Z Template Folder/Part 5 - Association Rule Learning/Section 28 - Apriori/Apriori-Python/Apriori_Python/Market_Basket_Optimisation.csv'
    transactions = load_data(file_path)

    # Train Apriori on the dataset
    min_support = 0.003
    min_confidence = 0.2
    min_lift = 3
    min_length = 2
    rules = train_apriori(transactions, min_support, min_confidence, min_lift, min_length)

    # Visualize the results
    visualize_rules(rules)

    # Plot item frequencies
    plot_item_frequencies(rules)
