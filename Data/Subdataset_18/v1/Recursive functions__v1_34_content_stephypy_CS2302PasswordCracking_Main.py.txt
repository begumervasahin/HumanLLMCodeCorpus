import hashlib
import time

def hash_with_sha256(s):
    hash_object = hashlib.sha256(s.encode('utf-8'))
    hex_dig = hash_object.hexdigest()
    return hex_dig

def find_password(password):
    with open("password_file.txt") as f:
        for curr_line in f:
            elements = curr_line.split(",")
            hashline = elements[2].strip()
            new_hashline = hash_with_sha256(password + str(elements[1]))
            if new_hashline == hashline:
                print(f"{elements[0]}: {password}")

def get_combination(size, comb):
    if size == 0:
        find_password(comb)
    else:
        for digit in range(10):
            new_comb = comb + str(digit)
            get_combination(size - 1, new_comb)

def hack_file(start, end):
    if not isinstance(start, int) or not isinstance(end, int):
        print('Invalid input. Please try again')
        return
    if start < 3 or end > 7:
        print('Invalid input. Please try again')
        return
    for size in range(start, end + 1):
        get_combination(size, '')

def test_cases():
    start = time.time()
    hack_file(3.0, 7.0)
    end = time.time()
    print('Running time for invalid data type:', end - start, 'seconds\n')

    start = time.time()
    hack_file(2, 7)
    end = time.time()
    print('Running time for invalid start size:', end - start, 'seconds\n')

    start = time.time()
    hack_file(3, 8)
    end = time.time()
    print('Running time for invalid end size:', end - start, 'seconds\n')

def main():
    start = time.time()
    hack_file(3, 7)
    end = time.time()
    print('Running time for normal run: ', end - start, 'seconds\n')
    test_cases()

if __name__ == "__main__":
    main()
