def parse_line(line):
    """
    Parse a CSV line considering possible quoted fields with commas.
    
    Args:
        line (str): A single line from the CSV file.
    
    Returns:
        list: A list of parsed elements from the CSV line.
    """
    elements = []
    skip_until = 0
    csplit = line.rstrip().split(',')

    for i in range(len(csplit)):
        if i <= skip_until:
            continue
        if '"' in csplit[i]:
            new_element = csplit[i]
            j = i + 1
            while '"' not in csplit[j]:
                new_element += ',' + csplit[j]
                j += 1
            new_element += ',' + csplit[j]
            elements.append(new_element.strip('"'))
            skip_until = j
        else:
            elements.append(csplit[i])
    
    return elements

def find_max_values(file_path):
    """
    Find the maximum number of students and the maximum student-staff ratio.
    
    Args:
        file_path (str): The path to the CSV file.
    
    Returns:
        tuple: The maximum number of students and the maximum student-staff ratio.
    """
    max_num_students = 0
    max_stud_staff_ratio = 0.0

    with open(file_path, 'r') as file:
        for line in file:
            parsed_line = parse_line(line)
            if parsed_line[0] == "world_rank":
                continue
            try:
                num_students = int(parsed_line[-5].replace(',', ''))
                student_staff_ratio = float(parsed_line[-4])
                max_num_students = max(max_num_students, num_students)
                max_stud_staff_ratio = max(max_stud_staff_ratio, student_staff_ratio)
            except (ValueError, IndexError):
                continue

    return max_num_students, max_stud_staff_ratio

def process_universities(file_path, max_num_students, max_stud_staff_ratio):
    """
    Process the universities data and write the points, labels, and metadata files.
    
    Args:
        file_path (str): The path to the CSV file.
        max_num_students (int): The maximum number of students.
        max_stud_staff_ratio (float): The maximum student-staff ratio.
    """
    with open(file_path, 'r') as f, \
         open("universities.points", 'w') as pts, \
         open("universities.labels", 'w') as lbls, \
         open("metadata.txt", 'w') as meta:
        
        meta.write(f"max_num_students: {max_num_students}\n")
        meta.write(f"max_stud_staff_ratio: {max_stud_staff_ratio:.6f}\n")

        for line in f:
            parsed_line = parse_line(line)
            if parsed_line[0] == "world_rank":
                continue
            try:
                num_students = float(parsed_line[-5].replace(',', '')) / max_num_students
                student_staff_ratio = 1.0 - (float(parsed_line[-4]) / max_stud_staff_ratio)
                name = parsed_line[1]
                teaching_score = float(parsed_line[2]) / 100
                tot_score = parsed_line[-6]

                lbls.write(name + "\n")

                try:
                    tot_score = float(tot_score) / 100
                except ValueError:
                    tot_score = teaching_score

                pts.write(f"{tot_score} {num_students} {student_staff_ratio}\n")
            except (ValueError, IndexError):
                continue

def main():
    file_path = "timesData.csv"
    max_num_students, max_stud_staff_ratio = find_max_values(file_path)
    process_universities(file_path, max_num_students, max_stud_staff_ratio)

if __name__ == "__main__":
    main()
