import sys

# Set the recursion limit to handle deeper recursive calls
sys.setrecursionlimit(3000)

def ack(M, N):
    """
    Computes the Ackermann function A(M, N).

    Parameters:
    M (int): The first parameter of the Ackermann function.
    N (int): The second parameter of the Ackermann function.

    Returns:
    int: The result of the Ackermann function.
    """
    if M == 0:
        return N + 1
    elif N == 0:
        return ack(M - 1, 1)
    else:
        return ack(M - 1, ack(M, N - 1))

    # Test the Ackermann function with the given values
    result = ack(3, 4)
    print(result)
