class Stack:
    """
    A simple Stack implementation using a list.
    """
    def __init__(self):
        self.list = []
        self.state = set()
    
    def push(self, item):
        """
        Add item to the stack and track its state.
        """
        self.list.append(item)  # Add item to the stack
        self.state.add(tuple(item.list))  # Track the state of the item
    
    def pop(self):
        """
        Remove the item from the top of the stack (LIFO) and return it.
        """
        if not self.isEmpty():
            tos = self.list[-1]  # Get the top item
            del self.list[-1]  # Remove the top item
            self.state.remove(tuple(tos.list))  # Remove the state of the popped item
            return tos
        else:
            return None
    
    def isEmpty(self):
        """
        Check if the stack is empty.
        """
        return len(self.list) == 0
    
    def clear_stack(self):
        """
        Clear the stack and reset its state.
        """
        self.list.clear()
        self.state.clear()

# Example usage of the Stack class
if __name__ == "__main__":
    class PuzzleState:
        def __init__(self, state_list):
            self.list = state_list

        def __repr__(self):
            return str(self.list)

    # Create a Stack instance
    s = Stack()

    # Push some puzzle states onto the stack
    s.push(PuzzleState([1, 2, 3, 4, 5, 6, 7, 8, 0]))
    s.push(PuzzleState([1, 2, 3, 4, 5, 6, 0, 7, 8]))
    s.push(PuzzleState([1, 2, 3, 4, 5, 0, 6, 7, 8]))

    # Pop and print each item
    while not s.isEmpty():
        state = s.pop()
        print("Popped:", state)
    
    # Clear the stack
    s.clear_stack()
    print("Stack cleared. Is empty:", s.isEmpty())
