from itertools import accumulate, count
from exceed import until_exceeded
from time import sleep

def triangle_number(n):
    """Calculates the n-th triangle number using the formula n*(n+1)/2.
    
    Args:
        n (int): The position in the triangle number sequence.
    
    Returns:
        int: The n-th triangle number.
    """
    return n * (n + 1) // 2

def triangle_number_alt(n):
    """Calculates the n-th triangle number using a summation.
    
    Args:
        n (int): The position in the triangle number sequence.
    
    Returns:
        int: The n-th triangle number.
    """
    return sum(range(1, n + 1))

def triangle_numbers():
    """Generator function that yields an infinite sequence of triangle numbers.
    
    Yields:
        int: The next triangle number in the sequence.
    """
    return accumulate(count(1), lambda x, y: x + y)

def is_triangle_number(number):
    """Checks if a given number is a triangle number.
    
    Args:
        number (int): The number to check.
    
    Returns:
        bool: True if the number is a triangle number, False otherwise.
    """
    last = None
    for num in until_exceeded(number, triangle_numbers()):
        last = num
    return number == last
