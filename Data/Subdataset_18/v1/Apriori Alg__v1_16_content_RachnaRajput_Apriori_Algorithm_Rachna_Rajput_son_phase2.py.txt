import json
import sys
import MapReduce

mr = MapReduce.MapReduce()
dict = {}

class Rachna_Rajput_son_phase2:
    @staticmethod
    def mapper(record):
        with open(sys.argv[1]) as f:
            mylist = []
            for line in f:
                chunkdata = json.loads(line)
                count = 0
                length = len(chunkdata)
                for chunkline in chunkdata:
                    if set(chunkline) >= set(record):
                        count += 1
                mylist = [count, length]
                recordtuple = tuple(record)
                dict[recordtuple] = mylist
                mr.emit_intermediate(recordtuple, mylist)

    @staticmethod
    def reducer(key, values):
        count = sum(value[0] for value in values)
        mylist = [key, count]
        mr.emit(mylist)

if __name__ == '__main__':
    inputdata = open(sys.argv[2])
    output = mr.execute(inputdata, Rachna_Rajput_son_phase2.mapper, Rachna_Rajput_son_phase2.reducer)
