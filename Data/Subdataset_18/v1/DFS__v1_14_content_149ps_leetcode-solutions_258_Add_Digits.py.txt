class Solution:
    def addDigits(self, num: int) -> int:
        temp = num
        while temp // 10 > 0:
            temp = (temp // 10) + (temp % 10)
        return temp

# Example usage:
solution = Solution()

# Test cases
print(solution.addDigits(38))  # Output: 2
print(solution.addDigits(123))  # Output: 6
print(solution.addDigits(0))  # Output: 0
