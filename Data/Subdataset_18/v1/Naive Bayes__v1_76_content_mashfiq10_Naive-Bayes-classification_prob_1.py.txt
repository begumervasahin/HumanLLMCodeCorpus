import numpy as np
import matplotlib.pyplot as plt

def load_images(filename):
    with open(filename, 'rb') as f:
        images = f.read()
    images = bytearray(images)
    images = np.array(images[16:], dtype='float64') > 100.
    return images

def load_labels(filename):
    with open(filename, 'rb') as f:
        labels = f.read()
    labels = bytearray(labels)
    labels = np.array(labels[8:])
    return labels

def plot_images(images, num_images=400, cols=20):
    d = 28 * 28
    rows = num_images // cols
    fig = plt.figure(figsize=(10, 10))
    for k in range(num_images):
        a = fig.add_subplot(rows, cols, k + 1)
        plt.imshow(images[k*d:(k+1)*d].reshape(28, 28), cmap=plt.cm.bone)
        a.set_axis_off()
    plt.show()

# Load training images and labels
training_images = load_images('train-images.idx3-ubyte')
training_labels = load_labels('train-labels.idx1-ubyte')

# Plot some training images
plot_images(training_images)

# Reshape training images
d = 28 * 28
n_train = training_images.shape[0] // d
training_images = training_images.reshape(int(n_train), d)

# Calculate priors
k = 10
counts = np.bincount(training_labels)
prior = counts / n_train
log_prior = np.log10(prior)

# Calculate likelihoods
theta = np.zeros((k, d), dtype='float64')
for i in range(k):
    mask = (training_labels == i)
    theta[i] = (np.sum(training_images[mask], axis=0, dtype='float64') + 1.) / (counts[i] + k)
log_theta = np.log10(theta)
log_complement = np.log10(1. - theta)

# Load test images and labels
test_images = load_images('t10k-images.idx3-ubyte')
test_labels = load_labels('t10k-labels.idx1-ubyte')

# Reshape test images
n_test = test_images.shape[0] // d
test_images = test_images.reshape(int(n_test), d)

# Predict labels for test images
test_labels_hat = np.zeros(test_labels.shape, dtype=int)
for i in range(test_images.shape[0]):
    log_likelihood = np.sum(log_theta[:, test_images[i].reshape(-1)], axis=1) + \
                     np.sum(log_complement[:, np.logical_not(test_images[i].reshape(-1))], axis=1)
    log_posterior = log_prior + log_likelihood
    test_labels_hat[i] = np.argmax(log_posterior)

# Calculate accuracy
accuracy = np.mean(test_labels == test_labels_hat)
print("Naive Bayes classifier classification accuracy:", accuracy * 100., '%')
