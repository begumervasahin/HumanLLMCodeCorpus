def calculate_fibonacci(n):
    """Calculate the first n Fibonacci numbers."""
    first_number = 1
    second_number = 0
    fibonacci_list = []
    for _ in range(n):
        answer = first_number + second_number
        fibonacci_list.append(answer)
        first_number = second_number
        second_number = answer
    return fibonacci_list

def print_fibonacci(fibonacci_list):
    """Print the Fibonacci numbers with their calculation steps."""
    for i in range(len(fibonacci_list)):
        if i == 0:
            print(f"{fibonacci_list[0]} + 0 = {fibonacci_list[0]}")
        elif i == 1:
            print(f"0 + {fibonacci_list[1]} = {fibonacci_list[1]}")
        else:
            print(f"{fibonacci_list[i - 2]} + {fibonacci_list[i - 1]} = {fibonacci_list[i]}")

def main():
    number_list = calculate_fibonacci(100)
    print_fibonacci(number_list)

if __name__ == "__main__":
    main()
