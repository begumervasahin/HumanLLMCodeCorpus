import random

# Define the character set
charset = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"

def main():
    """
    Demonstrates the usage of the encryption and decryption functions.
    """
    vector = "Once more into the fray."
    encrypted = encrypt(vector)
    decrypted = decrypt(encrypted[0], encrypted[1])
    
    print("Test Vector: " + vector)
    print("OTP: " + encrypted[0])
    print("Encrypted: " + encrypted[1])
    print("Decrypted: " + decrypted)

def encrypt(plaintext):
    """
    Encrypts the plaintext using a randomized one-time pad.

    Args:
    plaintext (str): The plaintext value to encrypt.

    Returns:
    tuple: A tuple containing the one-time pad and the encrypted message.
    """
    otp = "".join(random.sample(charset, len(charset)))
    result = ""
    for c in plaintext.upper():
        if c not in otp:
            continue
        else:
            result += otp[charset.find(c)]
    return otp, result

def decrypt(otp, secret):
    """
    Decrypts the secret using the one-time pad.

    Args:
    otp (str): The one-time pad used for encryption.
    secret (str): The encrypted message to be decrypted.

    Returns:
    str: The decrypted message.
    """
    result = ""
    for c in secret.upper():
        if c not in otp:
            continue
        else:
            result += charset[otp.find(c)]
    return result

if __name__ == "__main__":
    main()
