import time
import networkx as nx
import heapq
import sys

def dijkstra(digrafo, fonte, destino):
    # Initialize distances and labels
    for node in digrafo:
        digrafo.nodes[node]['distance'] = sys.maxsize
        digrafo.nodes[node]['rotulado'] = False
    digrafo.nodes[fonte]['distance'] = 0

    # Priority queue with (distance, node) tuples
    priority_queue = [(0, fonte)]
    heapq.heapify(priority_queue)
    
    while priority_queue:
        current_distance, current_node = heapq.heappop(priority_queue)
        
        if digrafo.nodes[current_node]['rotulado']:
            continue
        
        digrafo.nodes[current_node]['rotulado'] = True
        
        for neighbor in digrafo.successors(current_node):
            if not digrafo.nodes[neighbor]['rotulado']:
                new_distance = current_distance + digrafo.edges[current_node, neighbor]['weight']
                if new_distance < digrafo.nodes[neighbor]['distance']:
                    digrafo.nodes[neighbor]['distance'] = new_distance
                    digrafo.nodes[neighbor]['precedente'] = current_node
                    heapq.heappush(priority_queue, (new_distance, neighbor))
    
    # Retrieve the shortest path
    path_distance = digrafo.nodes[destino]['distance']
    path = [{'peso': path_distance}]
    node = destino
    
    while node != fonte:
        path.append(node)
        node = digrafo.nodes[node].get('precedente', fonte)
    path.append(fonte)
    
    return path[::-1]
