def text_file_aligning(file_path, align, line_length):
    """
    Splits a text file into lines with the given length and aligns these lines
    as specified.
    
    Parameters:
    file_path (str): Path to the .txt file.
    align (str): Type of alignment. One-character string:
                 'r' - right, 'l' - left, 'c' - center.
    line_length (int): Length of each line after splitting.
    
    Returns:
    None: Prints the aligned lines to the screen.
    """
    # Determine the alignment pattern based on the 'align' parameter
    if align == 'r':
        pattern = '{:>'  # Right alignment
    elif align == 'l':
        pattern = '{:'   # Left alignment
    elif align == 'c':
        pattern = '{:^'  # Center alignment
    else:
        raise ValueError("Invalid alignment option. Use 'r', 'l', or 'c'.")
    
    pattern += str(line_length) + '}'  # Complete the pattern
    
    try:
        with open(file_path, 'r') as file:
            for line in file:
                line = line.rstrip('\n')  # Remove trailing newline characters
                while len(line) > line_length:
                    # Find the last space within the line_length limit
                    split_pos = line[:line_length].rfind(' ')
                    if split_pos == -1:
                        split_pos = line_length
                    # Print the aligned substring
                    print(pattern.format(line[:split_pos]))
                    # Update the line to the remaining part
                    line = line[split_pos:].lstrip()
                # Print the remaining part of the line
                print(pattern.format(line))
    except FileNotFoundError:
        print(f"Error: The file '{file_path}' does not exist.")
    except Exception as e:
        print(f"An error occurred: {e}")

if __name__ == "__main__":
    # Example usage
    import sys
    
    if len(sys.argv) != 4:
        print("Usage: python script.py <file_path> <alignment> <line_length>")
        sys.exit(1)
    
    file_path = sys.argv[1]
    align = sys.argv[2]
    try:
        line_length = int(sys.argv[3])
    except ValueError:
        print("Error: Line length must be an integer.")
        sys.exit(1)
    
    text_file_aligning(file_path, align, line_length)
