# Global variable to count the number of swaps
swaps = 0

def heapify(arr, n, i):
    """
    Ensures the subtree rooted at index i is a max heap.
    
    Parameters:
    arr (list): The list to be heapified.
    n (int): The size of the heap.
    i (int): The root index of the subtree.

    Returns:
    int: The number of comparisons made.
    """
    global swaps
    count = 0
    largest = i  # Initialize largest as root
    l = 2 * i + 1  # left child index
    r = 2 * i + 2  # right child index

    # See if left child of root exists and is greater than root
    if l < n and arr[i] < arr[l]:
        largest = l

    # See if right child of root exists and is greater than root
    if r < n and arr[largest] < arr[r]:
        largest = r

    # Change root if needed
    if largest != i:
        count += 1
        arr[i], arr[largest] = arr[largest], arr[i]  # swap
        swaps += 1
        # Heapify the root
        count += heapify(arr, n, largest)
    
    return count

def heapSort(arr):
    """
    Sorts a list using the heap sort algorithm and counts the number of comparisons and swaps.

    Parameters:
    arr (list): The list to be sorted.

    Returns:
    str: A summary of comparisons and swaps.
    """
    global swaps
    n = len(arr)
    count = 0

    # Build a maxheap.
    for i in range(n // 2 - 1, -1, -1):
        count += heapify(arr, n, i)

    # One by one extract elements
    for i in range(n-1, 0, -1):
        arr[i], arr[0] = arr[0], arr[i]  # swap
        swaps += 1
        count += heapify(arr, i, 0)

    return "HEAP SORT:\nComparisons: " + str(count) + "\nSwaps: " + str(swaps)

# Example usage
if __name__ == "__main__":
    array = [12, 11, 13, 5, 6, 7]
    print("Original array:", array)
    result = heapSort(array)
    print(result)
    print("Sorted array:", array)
