# Initial value computation
val = (5**30) % 2017
print(f"S1 {val}\n")

# Initialize an empty list to store values
my_list = []

# Loop to generate values and classify them
for i in range(1, 10):
    val2 = (val**2) % 2017
    if 1 <= val2 <= 672:
        print(f"S0 {val2}\n")
    elif 673 <= val2 <= 1345:
        print(f"S1 {val2}\n")
    elif 1346 <= val2 <= 2016:
        print(f"S2 {val2}\n")
    else:
        print("oops\n")
    
    # Append the values to the list
    my_list.append(val)
    my_list.append(val2)
    
    # Update val for the next iteration
    val = (1736 * val2) % 2017
    
    if 1 <= val <= 672:
        print(f"S0 {val}\n")
    elif 673 <= val <= 1345:
        print(f"S1 {val}\n")
    elif 1346 <= val <= 2016:
        print(f"S2 {val}\n")
    else:
        print("oops\n")

# Print the list of values
print(my_list)

# Create a dictionary to find the indices of each value in the list
value_indices = {val: [i for i in range(len(my_list)) if my_list[i] == val] for val in my_list}

# Print the dictionary of values and their indices
print(value_indices)
