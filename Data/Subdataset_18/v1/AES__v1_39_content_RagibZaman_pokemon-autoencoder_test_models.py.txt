import unittest
import torch
from torch.autograd import Variable
import models

class TestEncoder(unittest.TestCase):
    def setUp(self):
        self.batch_size = 32
        self.image_size = 64
        self.latent_size = 128
        self.E = models.Encoder(self.latent_size, self.image_size)

    def test_size(self):
        """Test that the Encoder produces the correct size output."""
        tensor_in = Variable(2.0 * (torch.rand((self.batch_size, 3, self.image_size, self.image_size)) - 0.5))
        tensor_out = self.E.forward(tensor_in)
        self.assertEqual(tensor_out.size(), torch.Size([self.batch_size, self.latent_size]))

    def test_layers(self):
        """Test that the Encoder produces the correct size output for each layer."""
        tensor = Variable(2.0 * (torch.rand((self.batch_size, 3, self.image_size, self.image_size)) - 0.5))
        tensor = tensor.view(tensor.size()[0], -1)
        tensor = self.E.hidden_layer(tensor)
        self.assertEqual(tensor.size(), torch.Size([self.batch_size, self.latent_size]))
        tensor = self.E.output_layer(tensor)
        self.assertEqual(tensor.size(), torch.Size([self.batch_size, self.latent_size]))

class TestDecoder(unittest.TestCase):
    def setUp(self):
        self.batch_size = 32
        self.image_size = 64
        self.latent_size = 128
        self.D = models.Decoder(self.latent_size, self.image_size)

    def test_size(self):
        """Test that the Decoder produces the correct size output."""
        tensor_in = Variable(2.0 * (torch.rand((self.batch_size, self.latent_size, 1, 1)) - 0.5))
        tensor_out = self.D.forward(tensor_in)
        self.assertEqual(tensor_out.size(), torch.Size([self.batch_size, 3, self.image_size, self.image_size]))

    def test_layers(self):
        """Test that the Decoder produces the correct size output for each layer."""
        tensor = Variable(2.0 * (torch.rand((self.batch_size, self.latent_size, 1, 1)) - 0.5))
        tensor = self.D.layer1(tensor)
        self.assertEqual(tensor.size(), torch.Size([self.batch_size, 128, 7, 7]))
        tensor = self.D.layer2(tensor)
        self.assertEqual(tensor.size(), torch.Size([self.batch_size, 64, 15, 15]))
        tensor = self.D.layer3(tensor)
        self.assertEqual(tensor.size(), torch.Size([self.batch_size, 64, 31, 31]))
        tensor = self.D.layer4(tensor)
        self.assertEqual(tensor.size(), torch.Size([self.batch_size, 32, 63, 63]))
        tensor = self.D.layer5(tensor)
        self.assertEqual(tensor.size(), torch.Size([self.batch_size, 3, self.image_size, self.image_size]))

if __name__ == '__main__':
    unittest.main()
