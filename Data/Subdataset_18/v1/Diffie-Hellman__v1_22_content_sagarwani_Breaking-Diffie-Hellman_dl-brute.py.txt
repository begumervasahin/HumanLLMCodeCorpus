import sys
import re

def pow_x(g_base, a, p_mod):
    """
    Compute g_base^a % p_mod using an efficient method.
    
    Args:
        g_base (int): The base value.
        a (int): The exponent.
        p_mod (int): The modulus value.
    
    Returns:
        int: Result of (g_base^a) % p_mod.
    """
    x = 1
    bits = "{0:b}".format(a)
    for bit in bits:
        x = (x ** 2) % p_mod
        if bit == '1':
            x = (x * g_base) % p_mod
    return x

def attack(p, g, ga):
    """
    Attack the Diffie-Hellman exchange to find the secret exponent.
    
    Args:
        p (int): The prime modulus.
        g (int): The base.
        ga (int): The result of g^a % p.
    
    Returns:
        int: The secret exponent 'a' if found, otherwise 0.
    """
    for mystry_a in range(p):
        if pow_x(g, mystry_a, p) == ga:
            return mystry_a
    return 0

def main():
    """
    Main function to read input from a file and find the secret exponent.
    """
    if len(sys.argv) != 2:
        print("Usage: python script.py <input_file>")
        sys.exit(1)

    input_file = sys.argv[1]
    
    with open(input_file, 'r') as my_file1:
        content = my_file1.readline()
        x, y, z = content.split(',')

        p = int(re.findall('\d+', x)[0])
        g = int(re.findall('\d+', y)[0])
        ga = int(re.findall('\d+', z)[0])

    a = attack(p, g, ga)
    
    if a:
        print(a)
    else:
        print("Exponent not found.")

if __name__ == '__main__':
    main()
