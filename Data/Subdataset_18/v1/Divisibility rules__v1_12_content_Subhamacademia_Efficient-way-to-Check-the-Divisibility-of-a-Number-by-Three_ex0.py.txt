def is_divisible_by_three(n):
    """
    Check if a number is divisible by 3 using bitwise operations.
    """
    odd_counter = 0
    even_counter = 0
    
    # Base cases
    if n == 0:
        return True
    if n == 1:
        return False
    
    # Use bitwise operations to count the bits
    while n:
        if n & 1:  # Check the least significant bit
            odd_counter += 1
        n = n >> 1
        if n & 1:  # Check the next bit
            even_counter += 1
        n = n >> 1
    
    # Recursively check the difference between odd and even bit counts
    return is_divisible_by_three(abs(odd_counter - even_counter))

# Get user input
n = int(input("Enter an integer: "))

# Check divisibility and print the result
if is_divisible_by_three(abs(n)):
    print(f"{n} is divisible by 3.")
else:
    print(f"{n} is not divisible by 3.")
