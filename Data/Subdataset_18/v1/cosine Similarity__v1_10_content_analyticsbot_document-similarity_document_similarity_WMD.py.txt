import os
import pandas as pd
from word_mover_distance import similarity

def read_file(file_path):
    """Reads the contents of a file."""
    with open(file_path, 'r') as file:
        return file.read()

def calculate_similarity(file_list, file_path, verbose=0):
    """Calculates the similarity between each pair of files using word mover distance."""
    num_files = len(file_list)
    similarity_matrix = pd.DataFrame(index=range(num_files), columns=range(num_files))
    
    for i, file1 in enumerate(file_list):
        print(file1)
        s1 = read_file(os.path.join(file_path, file1))
        for j, file2 in enumerate(file_list):
            if i <= j:
                s2 = read_file(os.path.join(file_path, file2))
                similarity_value = similarity(s1, s2)
                similarity_matrix.loc[i, j] = similarity_value
                similarity_matrix.loc[j, i] = similarity_value  # Symmetric matrix
            if verbose:
                print(f'Currently processing {i}, {j}')
    
    return similarity_matrix

def main():
    print('All modules imported correctly')

    # Parameters
    verbose = 0
    mypath = 'C:\\Users\\Administrator\\Dropbox\\IPO analysis sourcecode\\IPO Project\\Data\\Individual Sections of IPO documents\\newuop'
    output_file = 'document_similarity_business.csv'

    # Get the list of files
    file_list = [f for f in os.listdir(mypath) if os.path.isfile(os.path.join(mypath, f))]

    # Calculate similarities
    similarity_df = calculate_similarity(file_list, mypath, verbose)

    # Assign file names to the DataFrame
    similarity_df.columns = file_list
    similarity_df.index = file_list

    print('All calculations made. Exporting to CSV')

    # Export to CSV
    similarity_df.to_csv(output_file, encoding='utf-8')

    print('Export to CSV done!')

if __name__ == "__main__":
    main()
