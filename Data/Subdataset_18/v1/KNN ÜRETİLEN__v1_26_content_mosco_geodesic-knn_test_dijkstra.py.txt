import numpy as np
import scipy.sparse
import scipy.sparse.csgraph
import dijkstra

def test_dijkstra_against_scipy():
    # Generate a random weighted graph
    W = np.random.random((100, 100))
    W += W.transpose()  # Make the graph symmetric
    W[W < 1.0] = np.inf  # Set some weights to infinity to simulate no connection
    W = scipy.sparse.csr_matrix(W)  # Convert to CSR matrix for efficiency

    for seed in [0, 1, 2]:
        # Compute shortest paths using custom Dijkstra implementation
        result0 = dijkstra.dijkstra(W, seed)

        # Compute shortest paths using SciPy's Dijkstra implementation
        result1 = scipy.sparse.csgraph.dijkstra(W, indices=[seed])[0]

        # Verify that the results are identical
        assert all(result0 == result1), f"Discrepancy found for seed {seed}"

# Example usage: Run the test
test_dijkstra_against_scipy()
