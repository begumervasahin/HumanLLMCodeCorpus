def kruskal_gen(matrix):
    """
    Generate a list of edges from the adjacency matrix and write to a file.
    Only considers upper triangular part of the matrix (excluding diagonal).
    """
    with open('kruskal.txt', 'w') as file:
        for lin in range(len(matrix)):
            for col in range(len(matrix[lin])):
                # Only write the edge if it's non-zero and in the upper triangular part
                if matrix[lin][col] != 0 and col > lin:
                    file.write(f'{lin} {col} {matrix[lin][col]}\n')

if __name__ == '__main__':
    # Read the adjacency matrix from a file
    with open('matrix.txt', 'r') as arquivo:
        lines = arquivo.readlines()

    # Determine the size of the matrix
    size = len(lines)
    mat = [[0 for _ in range(size)] for _ in range(size)]

    # Parse the matrix from the file
    for lin, line in enumerate(lines):
        values = line.strip().split()
        for col in range(size):
            mat[lin][col] = int(values[col])

    # Generate the edges for Kruskal's algorithm
    kruskal_gen(mat)
