import networkx as nx
import matplotlib.pyplot as plt

def draw_graph_with_path(edges, path):
    """
    Draw a directed graph with weighted edges, highlighting a specific path.

    Parameters:
    edges (list of tuples): List of edges in the graph, where each edge is represented as a tuple (source, target, weight).
    path (list of tuples): List of edges to highlight in the graph.
    """
    # Create a directed graph
    G = nx.DiGraph()

    # Add weighted edges to the graph
    G.add_weighted_edges_from(edges)

    # Define node colors based on some predefined value mapping
    value_map = {1: 1.0, 4: 0.5714285714285714, 8: 0.0}
    values = [value_map.get(node, 0.25) for node in G.nodes()]

    # Create edge labels dictionary
    edge_labels = dict(((u, v), d['weight']) for u, v, d in G.edges(data=True))

    # Separate edges into red (highlighted) and black (normal)
    red_edges = path
    edge_colors = ['red' if edge in red_edges else 'black' for edge in G.edges()]

    # Get positions for nodes using spring layout
    pos = nx.spring_layout(G)

    # Draw nodes
    nx.draw_networkx_nodes(G, pos, cmap=plt.get_cmap('jet'), node_color=values, node_size=500)

    # Draw edge labels
    nx.draw_networkx_edge_labels(G, pos, edge_labels=edge_labels)

    # Draw node labels
    nx.draw_networkx_labels(G, pos)

    # Draw edges, with highlighted edges in red and other edges in black
    nx.draw_networkx_edges(G, pos, edgelist=red_edges, edge_color='r', arrows=True)
    nx.draw_networkx_edges(G, pos, edgelist=[e for e in G.edges() if e not in red_edges], arrows=False)

    # Show the plot
    plt.show()

# Example usage
if __name__ == "__main__":
    # Define edges and path
    edges = [(1, 2, 4), (1, 3, 1), (2, 3, 2), (3, 4, 5)]
    path = [(1, 2), (2, 3)]

    # Draw the graph
    draw_graph_with_path(edges, path)
