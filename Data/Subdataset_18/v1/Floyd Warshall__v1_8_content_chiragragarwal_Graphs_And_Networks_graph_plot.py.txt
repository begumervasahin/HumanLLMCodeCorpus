import sys
from pathlib import Path
from read_graph import ReadGraph  # Ensure ReadGraph is implemented in read_graph.py
from plot_graph import PlotGraph  # Ensure PlotGraph is implemented in plot_graph.py

def main(input_file):
    """
    Main function to read a graph from a file, print the graph matrix, and plot the graph.

    Parameters:
    input_file (str): The path to the input file containing the graph data.
    """
    # Read the graph from the input file
    r = ReadGraph(input_file)
    graph_matrix = r.read()

    # Print the graph matrix
    print("Analysing graph matrix..\n")
    print(graph_matrix)

    # Plot the graph
    print("\nPlotting the graph..")
    p = PlotGraph(graph_matrix)
    p.plot()

if __name__ == '__main__':
    # Ensure the input file is provided as a command-line argument
    if len(sys.argv) <= 1:
        raise ValueError("Please enter the file to be read as an argument")

    # Get the input file path from the command-line arguments
    input_file = Path(sys.argv[1])

    # Run the main function with the provided input file
    main(input_file)
