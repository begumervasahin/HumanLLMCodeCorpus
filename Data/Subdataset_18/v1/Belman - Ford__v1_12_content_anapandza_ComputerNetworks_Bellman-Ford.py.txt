import pdb

def initialize(graph, source):
    """Initialize the distance and predecessor dictionaries."""
    d = {}  # Dictionary to hold the shortest distance from source to each node
    p = {}  # Dictionary to hold the predecessor of each node in the path
    for node in graph:
        d[node] = float('Inf')  # Initially, all distances are infinity
        p[node] = None  # Initially, all predecessors are None
    d[source] = 0  # Distance from source to itself is 0
    return d, p

def relax(node, neighbour, graph, d, p):
    """Relax the edge between node and neighbour."""
    if d[neighbour] > d[node] + graph[node][neighbour]:
        d[neighbour] = d[node] + graph[node][neighbour]
        p[neighbour] = node

def bellman_ford(graph, source):
    """Bellman-Ford algorithm to find the shortest path in a graph with possible negative weights."""
    d, p = initialize(graph, source)  # Initialize distance and predecessor
    # Relax all edges len(graph)-1 times
    for i in range(len(graph) - 1):
        for u in graph:
            for v in graph[u]:
                relax(u, v, graph, d, p)
    # Check for negative weight cycles
    for u in graph:
        for v in graph[u]:
            assert d[v] <= d[u] + graph[u][v], "Graph contains a negative weight cycle"
    return d, p

def test():
    """Test the Bellman-Ford algorithm with a sample graph."""
    graph = {
        'a': {'b': -2, 'c': 1, 'd': 4},
        'b': {'e': 3},
        'c': {'b': -3, 'd': 2},
        'd': {'e': -1},
        'e': {'c': 5}
    }
    d, p = bellman_ford(graph, 'a')
    print("The shortest distance from source a to all other nodes is:")
    for u in graph:
        print(f"{u} = {d[u]}")
    print("Paths from source a to all other nodes (predecessors of nodes):")
    for u in graph:
        print(f"{u} = {p[u]}")

if __name__ == '__main__':
    test()
