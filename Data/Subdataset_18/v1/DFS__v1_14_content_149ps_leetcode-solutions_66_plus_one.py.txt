def plus_one(nums):
    """
    Given a list of integers representing digits of a large number, increment the number by one.
    :param nums: List[int]
    :return: List[int]
    """
    # Convert the list of digits to a single integer
    number = int(''.join(str(x) for x in nums))
    
    # Increment the number by one
    number += 1
    
    # Convert the incremented number back to a list of digits
    result = [int(digit) for digit in str(number)]
    
    return result

# Example usage:
nums = [1, 2, 4, 5, 3, 2, 2, 9]
print(plus_one(nums))  # Output should be [1, 2, 4, 5, 3, 2, 3, 0]

# Additional test cases:
nums = [9, 9, 9]
print(plus_one(nums))  # Output should be [1, 0, 0, 0]

nums = [0]
print(plus_one(nums))  # Output should be [1]

nums = [1, 0, 0, 0]
print(plus_one(nums))  # Output should be [1, 0, 0, 1]
