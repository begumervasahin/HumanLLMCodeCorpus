from numberPrime import *
from modular_arithmetic import *

class RSA:
    def __init__(self):
        self.n, self.d, self.e = GenParms(n_bits=100, DEBUG=True)

    def encryption(self, msg):
        """
        Encrypt the message using RSA encryption.
        
        Parameters:
        msg (list): The message to encrypt as a list of integers.
        
        Returns:
        list: The encrypted message as a list of integers.
        """
        encrypted_message = [modExp(i, self.e, self.n) for i in msg]
        print('\nMensagem criptografada:', encrypted_message)
        return encrypted_message

    def decryption(self, msg):
        """
        Decrypt the message using RSA decryption.
        
        Parameters:
        msg (list): The encrypted message to decrypt as a list of integers.
        
        Returns:
        list: The decrypted message as a list of integers.
        """
        decrypted_message = [modExp(i, self.d, self.n) for i in msg]
        print('Mensagem Decriptografada:', decrypted_message)
        return decrypted_message

    def public_key(self):
        """
        Get the public key (n, e).
        
        Returns:
        tuple: The public key.
        """
        return self.n, self.e

    def private_key(self):
        """
        Get the private key (n, d).
        
        Returns:
        tuple: The private key.
        """
        return self.n, self.d
