def next_edges(graph, available_edges, vertex, graph_size):
    for j in range(graph_size):
        l = [vertex]
        if graph[vertex][j] != 0:
            l.append(j)
            available_edges.append(l)
    return

def prim_frontier_edge(graph, available_edges, visited):
    min_cost = 9999999
    selected_edge = None
    for edge in available_edges:
        if graph[edge[0]][edge[1]] != 0:
            if visited[edge[0]] == 1 and visited[edge[1]] == 0 and graph[edge[0]][edge[1]] < min_cost:
                min_cost = graph[edge[0]][edge[1]]
                selected_edge = edge
    if selected_edge:
        visited[selected_edge[1]] = 1
    return selected_edge

def update_frontier_edge(graph, available_edges, new_vertex, graph_size):
    available_edges = [edge for edge in available_edges if edge[1] != new_vertex]
    next_edges(graph, available_edges, new_vertex, graph_size)
    return available_edges

def prims(graph, start):
    vertex = [start]
    graph_size = len(graph)
    spanning_edges = []
    available_edges = []
    next_edges(graph, available_edges, start, graph_size)
    visited = [0] * graph_size
    visited[start] = 1

    for _ in range(graph_size - 1):
        minimum_edge = prim_frontier_edge(graph, available_edges, visited)
        if minimum_edge is not None:
            spanning_edges.append(minimum_edge)
            new_vertex = minimum_edge[1]
            vertex.append(new_vertex)
            available_edges = update_frontier_edge(graph, available_edges, new_vertex, graph_size)
    
    total_cost = sum(graph[edge[0]][edge[1]] for edge in spanning_edges)
    
    mstree = [[0] * graph_size for _ in range(graph_size)]
    for edge in spanning_edges:
        mstree[edge[0]][edge[1]] = graph[edge[0]][edge[1]]
        mstree[edge[1]][edge[0]] = graph[edge[0]][edge[1]]

    print("The total cost is:", total_cost)
    print("The minimum spanning tree is:")
    for row in mstree:
        print(row)
    return mstree

for i in range(graph_size):
