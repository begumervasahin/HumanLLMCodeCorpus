import itertools

def generate_combinations(num):
    """
    Generate all binary combinations of a given length.
    
    Args:
        num (int): The length of the binary strings.
        
    Returns:
        list: A list of binary strings.
    """
    tuples = list(itertools.product([0, 1], repeat=num))
    combinations = [''.join(map(str, tpl)) for tpl in tuples]
    return combinations

def crc(message, divisor):
    """
    Calculate the CRC of a given message using the provided divisor.
    
    Args:
        message (str): The binary message string.
        divisor (str): The binary divisor string.
        
    Returns:
        str: The calculated CRC.
    """
    pad = '0' * (len(divisor) - 1)
    message = list(message + pad)
    divisor = list(divisor)
    
    for i in range(len(message) - len(pad)):
        if message[i] == '1':
            for j in range(len(divisor)):
                message[i + j] = str(int(message[i + j]) ^ int(divisor[j]))
    
    return ''.join(message[-len(pad):])

def calculate_all_collisions(message, divisor, prefix_list=None):
    """
    Calculate all binary combinations that produce the same CRC as the given message.
    
    Args:
        message (str): The binary message string.
        divisor (str): The binary divisor string.
        prefix_list (list, optional): A list of prefix strings to consider.
        
    Returns:
        list: A list of binary strings that produce the same CRC.
    """
    final_crc = crc(message, divisor)
    combinations = generate_combinations(len(divisor))
    results = []

    if prefix_list is None:
        prefix_list = generate_combinations(len(message) - len(divisor))
    
    for prefix in prefix_list:
        for calc_bit in combinations:
            curr_msg = prefix + calc_bit
            if crc(curr_msg, divisor) == final_crc:
                results.append(curr_msg)
    
    return results

# Example usage
if __name__ == "__main__":
    message = "11010110"
    divisor = "10011"
    prefixes = ["111"]

    collisions = calculate_all_collisions(message, divisor, prefixes)
    print("Collisions:")
    for collision in collisions:
        print(collision)
