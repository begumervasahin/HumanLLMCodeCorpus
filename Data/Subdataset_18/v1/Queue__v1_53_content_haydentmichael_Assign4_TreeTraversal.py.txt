class TreeNode(object):
    def __init__(self, x):
        self.val = x
        self.left = None
        self.right = None

def inorderTraversal(a, root):
    if root:
        inorderTraversal(a, root.left)
        if root.val is not None:
            a.append(root.val)
        inorderTraversal(a, root.right)
    return a

def preorderTraversal(a, root):
    if root:
        if root.val is not None:
            a.append(root.val)
        preorderTraversal(a, root.left)
        preorderTraversal(a, root.right)
    return a

def populateTree(arr, root, i, n):
    if i < n:
        temp = TreeNode(arr[i])
        root = temp
        root.left = populateTree(arr, root.left, 2 * i + 1, n)
        root.right = populateTree(arr, root.right, 2 * i + 2, n)
    return root

def printTraversal(traversal, order_type):
    print(f"{order_type} Traversal")
    for x in traversal:
        print(x)
    print(" ")

# Main logic to demonstrate tree creation and traversal
if __name__ == "__main__":
    arrays = [
        [1, None, 2, None, None, 3],
        [1, 2, 3, 4, 5, 6, None],
        [1, 2, 3]
    ]

    for i, arr in enumerate(arrays):
        n = len(arr)
        root = populateTree(arr, None, 0, n)
        
        inorder_result = inorderTraversal([], root)
        preorder_result = preorderTraversal([], root)

        print(f"Array {i + 1}: {arr}")
        printTraversal(inorder_result, "In-order")
        printTraversal(preorder_result, "Pre-order")
