from pyspark import SparkContext
from pprint import pprint

def arrange_edges(edge):
    """
    Arrange edges from a given line in the file. 
    Handles standard edge format.
    
    :param edge: A string representing an edge.
    :return: A tuple of node connections.
    """
    u, v, w = edge.split(' ')
    if float(w) < 0:
        return (u, [v + ":" + w]), (v, []), (-1, [])
    return (u, [v + ":" + w]), (v, [])

def gr_arrange_edges(edge):
    """
    Arrange edges from a given line in the file. 
    Handles .gr format edge.
    
    :param edge: A string representing an edge.
    :return: A tuple of node connections.
    """
    split_list = edge.split(' ')
    if split_list[0] != 'a': 
        return []
    else:
        u, v, w = split_list[1], split_list[2], split_list[3]
        if int(w) < 0:
            return (u, [v + ":" + w]), (v, []), (-1, [])
        return (u, [v + ":" + w]), (v, [])

def get_negative_flag(it):
    """
    Check if there are any negative weights in the partition.
    
    :param it: An iterator over a partition.
    :return: A tuple indicating the presence of negative weights.
    """
    for i in it:
        if i[0] == -1:
            return i
    return (None, None)

def merge_nodes(node1, node2):
    """
    Merge two nodes.
    
    :param node1: First node.
    :param node2: Second node.
    :return: Merged nodes.
    """
    if node1 is None:
        return node2
    else:
        return node1

def get_graph(filePath, sc):
    """
    Build the graph from the input file.
    
    :param filePath: Path to the input file.
    :param sc: SparkContext.
    :return: A tuple of the graph and a flag indicating the presence of negative weights.
    """
    data = sc.textFile(filePath, 8)
    if '.gr' in filePath:
        nodesMapper = data.flatMap(gr_arrange_edges)
    else:
        nodesMapper = data.flatMap(arrange_edges)
    
    flag_exist_negative = True if -1 in nodesMapper.mapPartitions(get_negative_flag).collect() else False
    
    graph = nodesMapper.reduceByKey(lambda x, y: x + y).filter(lambda x: x[0] != -1).sortByKey()
    
    return graph, flag_exist_negative

if __name__ == '__main__':
    sc = SparkContext.getOrCreate()
    graph, has_negative_weights = get_graph('data/input_slides.dat', sc)
    pprint(graph.collect())
    print(f"Contains negative weights: {has_negative_weights}")
