import pandas as pd
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import linear_kernel

# Load dataset
dataset_path = "/home/nikita/Downloads/sample-data.csv"
ds = pd.read_csv(dataset_path)

# Initialize TF-IDF Vectorizer
tfidf_vectorizer = TfidfVectorizer(analyzer='word', ngram_range=(1, 3), min_df=0, stop_words='english')

# Fit and transform the dataset's 'description' column
tfidf_matrix = tfidf_vectorizer.fit_transform(ds['description'])

# Compute cosine similarities between the TF-IDF vectors
cosine_similarities = linear_kernel(tfidf_matrix, tfidf_matrix)

# Store results in a dictionary
results = {}

# Populate the results dictionary with cosine similarity scores
for idx, row in ds.iterrows():
    similar_indices = cosine_similarities[idx].argsort()[:-100:-1]
    similar_items = [(cosine_similarities[idx][i], ds['id'][i]) for i in similar_indices]
    results[row['id']] = similar_items[1:]

print('done!')

def item(id):
    """
    Retrieve the description of the item by its ID.
    
    Args:
        id (int): The ID of the item.
        
    Returns:
        str: The description of the item.
    """
    return ds.loc[ds['id'] == id]['description'].tolist()[0].split(' - ')[0]

def recommend(item_id, num):
    """
    Recommend a specified number of items similar to the given item ID.
    
    Args:
        item_id (int): The ID of the item for which recommendations are needed.
        num (int): The number of recommendations to return.
    """
    print(f"Recommending {num} products similar to {item(item_id)}...")
    print("-------")
    recs = results[item_id][:num]
    for rec in recs:
        print(f"Recommended: {item(rec[1])} (score: {rec[0]})")

# Example usage: Recommend 5 items similar to the item with ID 11
recommend(item_id=11, num=5)
