from collections import Counter
import math

# Set the threshold value
threshold = 0.4

def counter_cosine_similarity(c1, c2):
    """
    Calculate the cosine similarity between two counters.

    Args:
        c1 (Counter): First counter.
        c2 (Counter): Second counter.

    Returns:
        float: Cosine similarity between the two counters.
    """
    terms = set(c1).union(c2)
    dotprod = sum(c1.get(k, 0) * c2.get(k, 0) for k in terms)
    magA = math.sqrt(sum(c1.get(k, 0)**2 for k in terms))
    magB = math.sqrt(sum(c2.get(k, 0)**2 for k in terms))
    return dotprod / (magA * magB)

def length_similarity(c1, c2):
    """
    Calculate the length similarity between two counters.

    Args:
        c1 (Counter): First counter.
        c2 (Counter): Second counter.

    Returns:
        float: Length similarity between the two counters.
    """
    lenc1 = sum(c1.values())
    lenc2 = sum(c2.values())
    return min(lenc1, lenc2) / float(max(lenc1, lenc2))

def similarity_score(l1, l2):
    """
    Calculate the similarity score between two lists of words.

        l1 (list): First list of words.
        l2 (list): Second list of words.
