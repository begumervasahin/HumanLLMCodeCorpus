from queue import Queue
from copy import deepcopy

class Node:
    """
    Node class for the 8-puzzle problem.
    """
    def __init__(self, puzzle, parent=None, move=""):
        self.state = puzzle
        self.parent = parent
        self.depth = 0
        if parent is None:
            self.depth = 0
            self.moves = move
        else:
            self.depth = parent.depth + 1
            self.moves = parent.moves + move

    def goalState(self):
        """
        Check if the node's state is a goal state.
        """
        return self.state.checkPuzzle()

    def succ(self):
        """
        Generate the node's children states.
        """
        succs = Queue()
        for m in self.state.moves:
            p = deepcopy(self.state)
            p.doMove(m)
            if p.zero != self.state.zero:
                succs.put(Node(p, self, m))
        return succs

    def costHeur(self, heuristic):
        """
        Choose between the two available heuristics.
        """
        return self.nWrongTiles() if heuristic == 0 else self.manhattanDistance()

    def nWrongTiles(self):
        """
        First heuristic - number of wrong tiles.
        """
        result = 0
        count = 1
        for i in range(self.state.size):
            for j in range(self.state.size):
                if self.state.puzzle[i][j] != (count % (self.state.size * self.state.size)):
                    result += 1
                count += 1
        return result

    def manhattanDistance(self):
        """
        Second heuristic - distance of wrong tiles to their right position.
        """
        result = 0
        for i in range(self.state.size):
            for j in range(self.state.size):
                index = self.state.puzzle[i][j] - 1
                if index == -1:
                    continue
                correct_row, correct_col = divmod(index, self.state.size)
                result += abs(i - correct_row) + abs(j - correct_col)
        return result

    def __str__(self):
        """
        When printing the node, we obtain the moves from the starting state to this specific one.
        """
        return str(self.moves)

# Example usage of Node class
class Puzzle:
    """
    Example Puzzle class to integrate with the Node class.
    """
    def __init__(self, puzzle):
        self.puzzle = puzzle
        self.size = len(puzzle)
        self.moves = ['U', 'D', 'L', 'R']
        self.zero = self.find_zero()

    def find_zero(self):
        """
        Find the position of the empty tile (0).
        """
        for i, row in enumerate(self.puzzle):
            for j, value in enumerate(row):
                if value == 0:
                    return (i, j)
        return None

    def doMove(self, move):
        """
        Perform a move on the puzzle.
        """
        i, j = self.zero
        if move == 'U' and i > 0:
            self.puzzle[i][j], self.puzzle[i - 1][j] = self.puzzle[i - 1][j], self.puzzle[i][j]
            self.zero = (i - 1, j)
        elif move == 'D' and i < self.size - 1:
            self.puzzle[i][j], self.puzzle[i + 1][j] = self.puzzle[i + 1][j], self.puzzle[i][j]
            self.zero = (i + 1, j)
        elif move == 'L' and j > 0:
            self.puzzle[i][j], self.puzzle[i][j - 1] = self.puzzle[i][j - 1], self.puzzle[i][j]
            self.zero = (i, j - 1)
        elif move == 'R' and j < self.size - 1:
            self.puzzle[i][j], self.puzzle[i][j + 1] = self.puzzle[i][j + 1], self.puzzle[i][j]
            self.zero = (i, j + 1)

    def checkPuzzle(self):
        """
        Check if the puzzle is in the goal state.
        """
        goal = list(range(1, self.size * self.size)) + [0]
        return self.puzzle == [goal[i:i + self.size] for i in range(0, len(goal), self.size)]

if __name__ == "__main__":
    initial_state = Puzzle([[1, 2, 3], [4, 5, 6], [7, 8, 0]])
    node = Node(initial_state)

    print("Initial Node:")
    print(node)
    print("Is Goal State:", node.goalState())
    print("Heuristic (number of wrong tiles):", node.costHeur(0))
    print("Heuristic (Manhattan distance):", node.costHeur(1))

    children = node.succ()
    print("\nChildren States:")
    while not children.empty():
        child_node = children.get()
        print(child_node)
