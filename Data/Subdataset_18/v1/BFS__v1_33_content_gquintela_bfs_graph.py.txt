class Graph:
    def __init__(self):
        self.adjacency_list = {}
        self.size = 0

    def get_vertices(self):
        """
        Get a list of all vertices in the graph.
        """
        return list(self.adjacency_list.keys())

    def get_neighbours(self, v):
        """
        Get a list of all neighbours of a given vertex.
        """
        return self.adjacency_list[v]

    def print_vertices(self):
        """
        Print all vertices in the graph.
        """
        print("Vertices:")
        for vertex in self.adjacency_list:
            print(vertex)
        print('\n')

    def print_neighbours(self, v):
        """
        Print all neighbours of a given vertex.
        """
        print("Neighbours of " + v)
        for neighbour in self.adjacency_list[v]:
            print(neighbour)
        print('\n')

    def add_vertex(self, v):
        """
        Add a vertex to the graph.
        """
        if v not in self.adjacency_list:
            self.adjacency_list[v] = []

    def add_undirected_edge(self, u, v):
        """
        Add an undirected edge to the graph.
        """
        self.add_vertex(u)
        self.add_vertex(v)
        self.adjacency_list[u].append(v)
        self.adjacency_list[v].append(u)

    def add_directed_edge(self, u, v):
        """
        Add a directed edge to the graph.
        """
        self.add_vertex(u)
        self.add_vertex(v)
        self.adjacency_list[u].append(v)

if __name__ == "__main__":
    # Create a graph instance
    graph = Graph()

    # Add vertices and edges
    graph.add_vertex('A')
    graph.add_vertex('B')
    graph.add_vertex('C')
    graph.add_undirected_edge('A', 'B')
    graph.add_undirected_edge('A', 'C')
    graph.add_directed_edge('B', 'C')

    # Print vertices and their neighbours
    graph.print_vertices()
    graph.print_neighbours('A')
    graph.print_neighbours('B')
    graph.print_neighbours('C')
