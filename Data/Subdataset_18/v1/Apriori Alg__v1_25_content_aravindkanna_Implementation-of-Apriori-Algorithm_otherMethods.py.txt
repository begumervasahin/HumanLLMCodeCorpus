import itertools

def generate(currentSizeList):
    """
    Generate the next size list of itemsets from the current size list.
    """
    size = len(currentSizeList)
    nextSizeList = []
    i = 0
    
    while i < size:
        j = i + 1
        while j < size:
            if currentSizeList[i][:-1] == currentSizeList[j][:-1]:
                a = currentSizeList[i][:]
                a.append(currentSizeList[j][-1])
                nextSizeList.append(a)
                j += 1
            else:
                break
        i += 1
    return nextSizeList

def oneLessSubsets(itemset):
    """
    Generate all subsets of an itemset with one less element.
    """
    allSubsets = []
    size = len(itemset)
    for i in range(size):
        s = itemset[:i] + itemset[i+1:]
        allSubsets.append(s)
    return allSubsets

def prune(FreqsTrie, currList):
    """
    Prune the current list of itemsets based on their subsets in FreqsTrie.
    """
    prunedList = []
    for i in currList:
        allSubsets = oneLessSubsets(i)
        flag = True
        for j in allSubsets:
            if not FreqsTrie.hasNode(j):
                flag = False
                break
        if flag:
            prunedList.append(i)
    return prunedList

def itemSetsCount(inFile, currList):
    """
    Count the occurrences of each itemset in the current list within the transactions in the input file.
    """
    counts = [0] * len(currList)
    
    with open(inFile, 'r') as infp:
        for line in infp:
            line = line.strip()
            itemList = line.split(",")
            currTransaction = set(itemList)
            currSize = len(currTransaction)
            
            for i, itemset in enumerate(currList):
                if currSize < len(itemset):
                    continue 
                if currTransaction.issuperset(set(itemset)):
                    counts[i] += 1
    return counts

def printAssociateRules(FreqsTrie, mincon):
    """
    Print the association rules from the frequent itemsets in FreqsTrie based on the minimum confidence.
    """
    count = 0
    for i in FreqsTrie.getItemSets([]):
        for j in range(1, len(i)):
            for k in itertools.combinations(i, j):
                if float(FreqsTrie.getCount(i)) / float(FreqsTrie.getCount(k)) >= mincon:
                    print(",".join(k) + " => " + ",".join(set(i) - set(k)))
                    count += 1
    return count
