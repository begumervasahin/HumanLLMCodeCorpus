from fs.memoryfs import MemoryFS

def create_directories(dfs):
    """
    Create directories in the provided filesystem.
    
    Args:
        dfs (MemoryFS): The memory filesystem instance.
    """
    dfs.makedir('test')
    dfs.makedir('foo')

def list_directory_contents(dfs):
    """
    List the contents of the directories in the provided filesystem.
    
    Args:
        dfs (MemoryFS): The memory filesystem instance.
    """
    for path, info in dfs.walk.info(max_depth=1):
        if info.is_dir:
            print("[dir] {}".format(path))
        else:
            print("[file] {}".format(path))

def main():
    """
    Main function to create directories and list their contents.
    """
    dfs = MemoryFS()
    create_directories(dfs)
    list_directory_contents(dfs)

if __name__ == "__main__":
    main()
