def is_prime(x):
    """
    Check if a number x is a prime number.
    
    Parameters:
    x (int): The number to check.

    Returns:
    int: The number itself if it is prime, otherwise None.
    """
    for i in range(2, x):
        if x % i == 0:
            return None
    return x

def generate_prime_products():
    """
    Generate a list of numbers based on prime numbers between 9 and 200.
    The list includes squares of primes and products of distinct primes.
    
    Returns:
    list: A sorted list of generated numbers.
    """
    primes = list(filter(None, map(is_prime, range(9, 201))))
    products = [1]
    
    for i, prime_i in enumerate(primes):
        square = prime_i * prime_i
        if square < 201:
            products.append(square)
            for j in range(i + 1, len(primes)):
                product = prime_i * primes[j]
                if product < 201:
                    products.append(product)
                else:
                    break
        else:
            break
    
    all_numbers = sorted(set(primes + products))
    return all_numbers

def main():
    """
    Main function to generate and print the list of prime-related numbers.
    """
    result = generate_prime_products()
    print(result)

if __name__ == "__main__":
    main()
