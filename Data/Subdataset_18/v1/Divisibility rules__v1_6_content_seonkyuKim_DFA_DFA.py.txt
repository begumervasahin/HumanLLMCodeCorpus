def DFA(input_string):
    """
    Deterministic Finite Automaton (DFA) to process a binary string.

    - internal states : q0, q1, q2, q3, q4
    - input alphabet : {0, 1}
    - transition function :
        (q0, 0) = q0  /  (q0, 1) = q1
        (q1, 0) = q2  /  (q1, 1) = q3
        (q2, 0) = q4  /  (q2, 1) = q0
        (q3, 0) = q1  /  (q3, 1) = q2
        (q4, 0) = q3  /  (q4, 1) = q4
    - initial state : q0
    - final state : q0
    """
    # Define states
    q0 = 'q0'
    q1 = 'q1'
    q2 = 'q2'
    q3 = 'q3'
    q4 = 'q4'
    
    # Set the initial state
    curr_state = q0

    # Process the input string
    for element in input_string:
        if curr_state == q0:
            curr_state = q0 if element == '0' else q1
        elif curr_state == q1:
            curr_state = q2 if element == '0' else q3
        elif curr_state == q2:
            curr_state = q4 if element == '0' else q0
        elif curr_state == q3:
            curr_state = q1 if element == '0' else q2
        elif curr_state == q4:
            curr_state = q3 if element == '0' else q4
    
    # Check if the final state is the accepting state
    return curr_state == q0

def check_binary(input_string):
    """
    Check if the input string is a valid binary string.

    Parameters:
    input_string (str): The string to check.

    Returns:
    bool: True if the string is binary, False otherwise.
    """
    return set(input_string).issubset({'0', '1'})

def main():
    """
    Main function to execute the DFA check.
    """
    description = "Input a binary value (0 and 1). To exit the program, type 'exit'.\n"
    
    while True:
        user_input = input(description)
        if user_input.lower() == 'exit':
            break
        if not check_binary(user_input):
            print('Error: Input value is not binary.\n')
            continue
        is_accepted = DFA(user_input)
        if is_accepted:
            print('Accept\n')
        else:
            print('Not accept\n')

if __name__ == "__main__":
    main()
