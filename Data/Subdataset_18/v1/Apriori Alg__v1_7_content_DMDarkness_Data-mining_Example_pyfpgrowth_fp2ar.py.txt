import pyfpgrowth as pyfp  # Import the package of FP-Growth
import fp2ar  # Import the package to form association rules from frequent patterns
import re  # Import the regular expression package

def read(dName):
    """
    Reads a transactional dataset from a file and returns a list of transactions.
    Each transaction is a list of item IDs (integers).
    """
    with open(dName, 'r') as f:
        content = f.readlines()  # Read all the lines from the transactional dataset

    Trans = []
    for line in content:
        sline = re.split(' |\n', line)  # Split the dataset by space or newline
        trans = [int(item) for item in sline if item.isdigit()]  # Extract items as integers
        if trans:
            Trans.append(trans)  # Append non-empty transactions to the list
    return Trans

# Read the transactional dataset and store it in dataset
# The dataset kosarak.dat can be downloaded from http://fimi.ua.ac.be/data/
dataset = read("kosarak.dat")

# Mine the dataset with the minimum support of 1%
# After mining, there should be 383 frequent patterns
fi = pyfp.getFP(dataset, 0.01)

# Form the association rules from frequent patterns
# The minimum confidence is set to 75%
# The minimum lift is set to 1
# After running, there should be 431 association rules
ar = fp2ar.getAR(fi, len(dataset), 0.75, 1)

# Print the frequent patterns
print("Frequent Patterns:")
for pattern in fi:
    print(pattern)

# Print the association rules
print("\nAssociation Rules:")
for rule in ar:
    print(rule)
