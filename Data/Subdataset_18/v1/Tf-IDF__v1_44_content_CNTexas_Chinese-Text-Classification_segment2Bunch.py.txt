import os
import jieba
import pickle
from sklearn.datasets import Bunch

def save_file(save_path, content):
    with open(save_path, "wb") as fp:
        fp.write(content)

def read_file(path):
    with open(path, "rb") as fp:
        content = fp.read()
    return content

# Initialize Bunch object
bunch = Bunch(target_name=[], label=[], filenames=[], contents=[])

wordbag_path = "train_word_bag/train_set.dat"
seg_path = "train_corpus_seg/"

# Get category list from segmented path
catelist = os.listdir(seg_path)
bunch.target_name.extend(catelist)

# Process each category
for mydir in catelist:
    class_path = os.path.join(seg_path, mydir)
    file_list = os.listdir(class_path)
    for file_name in file_list:
        full_name = os.path.join(class_path, file_name)
        bunch.label.append(mydir)
        bunch.filenames.append(full_name)
        bunch.contents.append(read_file(full_name).strip())

# Save the Bunch object to a file
with open(wordbag_path, "wb") as file_obj:
    pickle.dump(bunch, file_obj)

print("构建文本对象结束！！！")
