from uncompyle6.main import decompile
import marshal
import sys
import os
import marcode

def decompile_and_save(py_version, output_file, marcode_data):
    """
    Decompile the given marshal data and save it to a Python file.

    Parameters
    ----------
    py_version : float
        The Python version for decompiling.
    output_file : str
        The name of the output file.
    marcode_data : bytes
        The marshal data to decompile.
    """
    try:
        # Decompile the marshal data
        decompiled_code = decompile(py_version, marshal.loads(marcode_data), sys.stdout)
        
        # Save the decompiled code to a file
        with open(f"{output_file}.py", "w") as f:
            f.write(decompiled_code)
        
        print(f"\n\n[result] Saved as \033[95m{output_file}.py")
    except Exception as e:
        print(f"Error: {e}")

def py3():
    """
    Decompile marshal data for Python 3.7.X and save to a file.
    """
    output_file = input("[*] Decompile marshal Python 3.7.X\n[?] File output: ")
    decompile_and_save(3.7, output_file, marcode.py3)

def py2():
    """
    Decompile marshal data for Python 2.7.X and save to a file.
    """
    output_file = input("[*] Decompile marshal Python 2.7.X\n[?] File output: ")
    decompile_and_save(2.7, output_file, marcode.py2)

def main():
    """
    Main function to determine the Python version and call the appropriate decompilation function.
    """
    try:
        os.system('clear')
        print("""
        ;;;;;;;;;;;;;;;;;;;;;
        ; Decompile Marshal ;
        ;   By KANG-NEWBIE  ;
        ;;;;;;;;;;;;;;;;;;;;;
        """)
        if sys.version_info[0] == 3:
            py3()
        elif sys.version_info[0] == 2:
            py2()
        else:
            print("Unsupported Python version.")
    except Exception as e:
        print(f"Error: {e}")

if __name__ == "__main__":
    main()
