def selection_sort(arr):
    """
    Sorts a list of numbers in ascending order using the selection sort algorithm.

    Args:
        arr (list): The list of numbers to be sorted.
    """
    for i in range(len(arr)):
        # Find the index of the smallest element in the remaining unsorted array
        min_index = i
        for j in range(i + 1, len(arr)):
            if arr[j] < arr[min_index]:
                min_index = j
        # Swap the found minimum element with the first element of the unsorted array
        arr[i], arr[min_index] = arr[min_index], arr[i]

def main():
    # Example list of numbers to be sorted
    arr = [49, 25, 26, 2, 99]
    print("Original array:", arr)
    selection_sort(arr)
    print("Sorted array is:")
    for element in arr:
        print(element, end=' ')

if __name__ == "__main__":
    main()
