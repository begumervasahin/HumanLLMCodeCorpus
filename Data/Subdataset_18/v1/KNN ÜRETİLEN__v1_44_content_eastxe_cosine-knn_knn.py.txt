import heapq
import numpy as np
from sklearn.metrics.pairwise import pairwise_distances

class CosineKNN:
    def __init__(self, n_neighbors=5):
        self.n_neighbors = n_neighbors

    def fit(self, X, y):
        """Fit the model using X as training data and y as target values."""
        self.data = np.array(X)
        self.train = np.array(y)

    def predict(self, X):
        """Predict the class labels for the provided data."""
        X = np.array([X])
        X = X.reshape(1, -1)
        simtree = []
        dist_dic = {}

        for data in self.data:
            sim = pairwise_distances(X, np.array(data).reshape(1, -1), metric='cosine')
            dist_dic[float(sim)] = data
            heapq.heappush(simtree, sim)

        near_labels = []
        for _ in range(self.n_neighbors):
            min_sim = heapq.heappop(simtree)
            vector = dist_dic[float(min_sim)]
            for index, item in enumerate(self.data):
                if (item == vector).all():
                    near_labels.append(self.train[index])
                    break

        label_count = {}
        for label in near_labels:
            if label in label_count:
                label_count[label] += 1
            else:
                label_count[label] = 1

        y_pred = max(label_count, key=label_count.get)
        return y_pred

if __name__ == '__main__':
    knn = CosineKNN(n_neighbors=2)
    X = [[0.5, 0.86], [0.86, 0.5], [-0.5, -0.86], [-0.86, -0.5]]
    y = [1, 1, 0, 0]
    knn.fit(X, y)
    print('Answer:', knn.predict([0.7, 0.7]))
    print('Answer:', knn.predict([-0.7, -0.7]))
