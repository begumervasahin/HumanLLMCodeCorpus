def main():
    # Display the introduction
    print("Bienvenido a la máquina de Atwood, consiste en dos masas m1 y m2, conectadas por una cuerda inelástica de masa despreciable con una polea ideal de masa despreciable")
    
    # Input masses from the user
    m1 = float(input("Digite la masa del objeto 1 en kilogramos, considere que este objeto es el más pesado: "))
    m2 = float(input("Digite la masa del objeto 2 en kilogramos: "))
    
    # Gravitational acceleration
    g = 9.81  # m/s^2
    
    # Calculate acceleration
    acceleration = (g * (m1 - m2)) / (m1 + m2)
    print("Recuerde que la gravedad es 9.81 m/s^2")
    print(f"Considerando los datos anteriores, la aceleración es equivalente a: {acceleration:.2f} m/s^2")
    
    # Calculate tension in Newtons
    tension = (g * 2 * (m1 * m2)) / (m1 + m2)
    print(f"La tensión de la cuerda en Newtons es: {tension:.2f} N")
    
    # Calculate tension in dynes
    tension_dyne = (g * 0.01 * 2 * (m1 * m2) * 0.001) / ((m1 + m2) * 0.001)
    print(f"La tensión en dinas es: {tension_dyne:.2f} dyn")

if __name__ == "__main__":
    main()
