import time
import random

def selection_sort(array):
    start_time = time.time()
    for i in range(len(array) - 1):
        position = i
        for j in range(i + 1, len(array)):
            if array[position] > array[j]:
                position = j
        array[i], array[position] = array[position], array[i]
    end_time = time.time()
    print(f"Time taken to sort the array: {end_time - start_time} seconds")
    return array

# Example usage:
if __name__ == "__main__":
    # Generate a random list of 100 numbers for testing
    array = [random.randint(0, 1000) for _ in range(100)]
    print("Original array:", array)
    sorted_array = selection_sort(array)
    print("Sorted array:", sorted_array)
