import tkinter as tk
from tkinter import simpledialog, Label, Entry

class TModelSizeWindow(simpledialog.Dialog):
    """
    Class representing a popup window for entering model size (domain).
    
    Attributes:
        e1 (tk.Entry): Entry widget for the x dimension.
        e2 (tk.Entry): Entry widget for the y dimension.
    """

    def __init__(self, master):
        """
        Initialize the TModelSizeWindow dialog.
        
        Args:
            master (tk.Tk): The master window object.
        """
        super().__init__(master)

    def body(self, master):
        """
        Initialize and arrange the widgets in the dialog.
        
        Args:
            master (tk.Tk): The master window object.
        """
        Label(master, text="x:").grid(row=0, column=0, padx=10, pady=5)
        Label(master, text="y:").grid(row=1, column=0, padx=10, pady=5)

        self.e1 = Entry(master)
        self.e2 = Entry(master)

        self.e1.grid(row=0, column=1, padx=10, pady=5)
        self.e2.grid(row=1, column=1, padx=10, pady=5)

    def apply(self):
        """
        Capture and return the user inputs.
        """
        first = self.e1.get()
        second = self.e2.get()
        self.result = first, second

# Example usage
if __name__ == "__main__":
    def main():
        root = tk.Tk()
        root.withdraw()  # Hide the root window

        dialog = TModelSizeWindow(root)
        root.wait_window(dialog)

        # Output the result
        if dialog.result:
            x, y = dialog.result
            print(f"Model size: x={x}, y={y}")
        else:
            print("No model size entered.")

    main()
