class Node:
    def __init__(self, val):
        self.l = None  # Left child
        self.r = None  # Right child
        self.p = None  # Parent
        self.v = val   # Value

    def addLeft(self, childNode):
        """Add a left child to the current node."""
        self.l = childNode
        childNode.p = self

    def addRight(self, childNode):
        """Add a right child to the current node."""
        self.r = childNode
        childNode.p = self

    def printLeft(self):
        """Print the left child of the current node."""
        print("Left child:", self.l.v if self.l else None)

    def printRight(self):
        """Print the right child of the current node."""
        print("Right child:", self.r.v if self.r else None)

    def printVal(self):
        """Print the value of the current node."""
        print("Value:", self.v)

    def printParent(self):
        """Print the parent of the current node."""
        print("Parent:", self.p.v if self.p else None)

# Example usage
if __name__ == "__main__":
    # Create nodes
    root = Node(10)
    left_child = Node(5)
    right_child = Node(15)
    left_grandchild = Node(3)
    
    # Add children
    root.addLeft(left_child)
    root.addRight(right_child)
    left_child.addLeft(left_grandchild)
    
    # Print values
    root.printVal()
    root.printLeft()
    root.printRight()
    left_child.printVal()
    left_child.printParent()
    left_child.printLeft()
    left_grandchild.printVal()
    left_grandchild.printParent()
