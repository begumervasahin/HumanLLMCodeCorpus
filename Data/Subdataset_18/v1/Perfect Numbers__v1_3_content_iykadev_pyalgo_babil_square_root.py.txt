def babil_square_root(a, eps):
    """
    Calculates the square root of a number using the Babylonian method.

    Parameters:
    a (float): The number to find the square root of.
    eps (float): The tolerance for the approximation.

    Returns:
    None
    """
    x0 = (1 + a) / 2
    x1 = (x0 + a / x0) * 0.5
    y = abs((x1 - x0) / x0)
    
    while y > eps:
        x0 = x1
        x1 = (x0 + a / x0) * 0.5
        y = abs((x1 - x0) / x0)
    
    result = "The square root of the number {} is {:.6f}".format(a, x1)
    print(result)

def babil_square_root_main():
    babil_square_root(5, 0.0001)

if __name__ == "__main__":
    babil_square_root_main()
