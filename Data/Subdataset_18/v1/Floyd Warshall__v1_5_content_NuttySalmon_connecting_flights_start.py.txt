from shortest import Database, ConnectingFlights, console

def main():
    """
    Main function to create a database connection, initialize ConnectingFlights,
    and display the console menu.
    """
    # Connect to the database
    db = Database('localhost', 27017, "connecting_flight")
    
    # Initialize ConnectingFlights with the database
    cf = ConnectingFlights(db)
    
    # Display the console menu
    console.menu(db, cf)
    
    # Clear the console
    console.clear()

if __name__ == '__main__':
    main()
