from itertools import product

import sympy as sp
import numpy as np
from math import floor

def char_poly(M):
    """
    Compute the characteristic polynomial of a matrix.

    Parameters:
    M (scipy.sparse.csr_matrix): The matrix.

    Returns:
    sympy.Poly: The characteristic polynomial.
    """
    temp = M.toarray()
    temp = sp.SparseMatrix(temp)
    return temp.berkowitz_charpoly()

def multiplicity(poly, root, Der):
    """
    Compute the multiplicity of a root in a polynomial.

    Parameters:
    poly (sympy.Poly): The polynomial.
    root (float): The root to check.
    Der (list): List of derivatives.

    Returns:
    list: Multiplicity and list of derivatives.
    """
    D = Der
    i = 1
    flag = 'go'
    while flag == 'go':
        if i <= (len(D) - 1):
            if D[i].eval(root) != 0:
                multiplicity = i
                flag = 'stop'
        else:
            temp = D[-1].diff()
            D.append(temp)
            if temp.eval(root) != 0:
                multiplicity = i
                flag = 'stop'
        i += 1
    return [multiplicity, D]

def divisors(n):
    """
    Compute the divisors of an integer.

    Parameters:
    n (int): The integer.

    Returns:
    list: List of divisors.
    """
    out = []
    for i in range(1, floor(n**0.5) + 1):
        if n % i == 0:
            out.append(i)
    return out

def remove_zeros_at_zero(poly):
    """
    Remove zeros at the origin of a polynomial.

    Parameters:
    poly (sympy.Poly): The polynomial.

    Returns:
    sympy.Poly: The polynomial without zeros at the origin.
    """
    mon = poly.EM()
    mon = mon.as_expr()
    return sp.exquo(poly, mon)

def higman_trace(A):
    """
    Compute the Higman trace of an algebra.

    Parameters:
    A (Algebra): The algebra object.

    Returns:
    scipy.sparse.csr_matrix: The Higman trace matrix.
    """
    dim = A.dim
    if A.casimir_flag == 'no':
        A.GetCasimir()
    mult = A.mult
    V = CreateBasisVectors(dim)
    casimir = A.casimir
    higman_trace = sps.csr_matrix((dim, dim))
    for i, j in product(range(dim), repeat=2):
        if A.casimir[dim * i + j] != 0:
            higman_trace += Left_Action_Matrix(V[i], mult).dot(Right_Action_Matrix(V[j], mult))
    return higman_trace

def compute_M(A):
    """
    Compute the matrix M of an algebra.

    Parameters:
    A (Algebra): The algebra object.

    Returns:
    scipy.sparse.csr_matrix: The computed matrix M.
    """
    dim = A.dim
    if A.casimir_flag == 'no':
        A.GetCasimir()
    mult = A.mult
    casimir = A.casimir
    tensor_mult = Tensor_Mult(A, A)
    C = mult.dot(tensor_mult.dot(np.kron(casimir, casimir)))
    return Left_Action_Matrix(C, A.mult)
