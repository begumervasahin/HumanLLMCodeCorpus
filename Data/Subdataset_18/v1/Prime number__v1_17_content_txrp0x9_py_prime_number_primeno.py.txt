print("Welcome random tester, this program will print all the prime numbers from 2 to n")
n = int(input("Enter n\n"))

# Create a list of boolean values representing the primality of each number
primes = [True for i in range(n+1)]
p = 2

# Implement the Sieve of Eratosthenes algorithm
while p * p <= n:
    if primes[p] == True:
        # Mark all multiples of p as False indicating they are not prime
        for i in range(p * 2, n + 1, p):
            primes[i] = False
    p += 1

# Print all prime numbers from 2 to n
for x in range(2, n + 1):
    if primes[x]:
        print(x)
