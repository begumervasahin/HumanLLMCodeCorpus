import os
import glob
import cv2
import random
import numpy as np
from scipy.spatial import distance
import ypoften as of

def attr_edge(imgpath, tffolder, blur_first=True, blur_size=3, threshold1=100, threshold2=250, adaptive_threshold=True, ratio1=0.4, ratio2=0.8, savetf=True, select_random=True, n_random=1000):
    """Extracts edge attributes from an image."""
    
    # Read the image in grayscale
    img = cv2.imread(imgpath, 0)
    
    # Apply Gaussian blur if specified
    if blur_first:
        img = cv2.GaussianBlur(img, (blur_size, blur_size), 0)
    
    # Compute adaptive thresholds if specified
    if adaptive_threshold:
        threshold1 = min(100, np.quantile(img, q=ratio1))
        threshold2 = max(200, np.quantile(img, q=ratio2))
        print("Edge detection thresholds:", threshold1, threshold2)
    
    # Perform Canny edge detection
    edge = cv2.Canny(img, threshold1=threshold1, threshold2=threshold2)
    
    # Save the edge-detected image if specified
    if savetf:
        imgname = os.path.basename(imgpath)
        imgsavepath = os.path.join(tffolder, "edge_canny", os.path.splitext(imgname)[0] + '.png')
        of.create_path(imgsavepath)
        cv2.imwrite(imgsavepath, edge)
    
    # Extract edge points
    epoints_twodims = np.nonzero(edge)
    epoints = np.transpose(epoints_twodims)
    h, w = edge.shape
    dia = np.sqrt(h**2 + w**2)
    etotal = len(epoints)
    edensity = etotal / (h * w)
    
    if etotal > 0:
        if select_random:
            random.seed(42)
            epoints = random.sample(list(epoints), min(n_random, etotal))
        dists = distance.pdist(epoints, 'euclidean')
        edist = np.mean(dists) / dia
    else:
        edist = -99999
    
    rlist = [edensity, edist]
    print("Edge density & distribution:", rlist)
    return rlist

def main():
    """Main function to process all images in a folder."""
    
    imgfolder = os.path.join('img_all', '')
    tffolder = os.path.join('img_transform', '')
    resultpath = os.path.join('img_result', 'edge.txt')
    
    # Get list of all image paths
    imgpaths = glob.glob(imgfolder + '*')
    
    for j, imgpath in enumerate(imgpaths):
        print("-" * 100)
        imgname = os.path.basename(imgpath)
        print(j, imgname)
        
        # Extract edge attributes and save results
        wlist = [imgname] + attr_edge(imgpath, tffolder)
        of.save_list_to_txt(wlist, resultpath)
    
    print("DONE" * 50)

if __name__ == "__main__":
    main()
