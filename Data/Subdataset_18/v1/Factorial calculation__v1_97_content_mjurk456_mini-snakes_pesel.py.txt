import unittest

def check_pesel(pesel_nr):
    """
    Validates a PESEL number.

    Parameters:
    - pesel_nr (str): The PESEL number to be validated.

    Returns:
    - bool: True if the PESEL number is valid, False otherwise.
    """
    if len(pesel_nr) != 11:
        return False
    if not pesel_nr.isdigit():
        return False

    check_array = [1, 3, 7, 9, 1, 3, 7, 9, 1, 3]
    total_sum = sum(int(pesel_nr[i]) * check_array[i] for i in range(10))

    check_digit = (10 - (total_sum % 10)) % 10
    return check_digit == int(pesel_nr[-1])

class TestPeselValidator(unittest.TestCase):
    def test_correct_pesel(self):
        """Test with valid PESEL numbers."""
        self.assertTrue(check_pesel("92121510595"))
        self.assertTrue(check_pesel("99810235403"))

    def test_wrong_pesel(self):
        """Test with invalid PESEL numbers."""
        self.assertFalse(check_pesel("44051401358"))
        self.assertFalse(check_pesel("51401358"))
        self.assertFalse(check_pesel("12345678901"))  # Invalid PESEL number for additional coverage

if __name__ == "__main__":
    unittest.main()
