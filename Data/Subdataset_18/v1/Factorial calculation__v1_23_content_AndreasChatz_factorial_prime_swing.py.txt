import math
from typing import List

def divide_swing_and_conquer(number: int) -> int:
    """
    Computes the swinging factorial of a number using a divide-and-conquer approach.
    
    :param number: The number to compute the factorial of
    :return: The swinging factorial of the given number
    """
    if number <= 1:
        return 1
    prime_numbers = all_prime_numbers_up_to(number)
    prime_exponents_array = prime_exponents_of_a_swing_number(number, prime_numbers)
    sf = compute_swinging_factorial(prime_numbers, prime_exponents_array)
    return divide_swing_and_conquer(number // 2) ** 2 * sf

def prime_exponents_of_a_swing_number(swing_number: int, prime_numbers: List[int]) -> List[int]:
    """
    Calculates the exponents of prime factors in the swinging number representation.
    
    :param swing_number: The number to calculate the prime exponents for
    :param prime_numbers: List of prime numbers to use for factorization
    :return: List of exponents for each prime number
    """
    k = math.floor(math.log2(swing_number))
    prime_exponents_array = []
    for prime in prime_numbers:
        exponent = 0
        for i in range(1, k + 1):
            exponent += swing_number // (prime ** i) % 2
        prime_exponents_array.append(exponent)
    return prime_exponents_array

def compute_swinging_factorial(prime_numbers: List[int], prime_exponents_array: List[int]) -> int:
    """
    Computes the swinging factorial from the prime numbers and their exponents.
    
    :param prime_numbers: List of prime numbers
    :param prime_exponents_array: List of exponents corresponding to each prime number
    :return: The swinging factorial
    """
    swing_factorial = 1
    for prime, exponent in zip(prime_numbers, prime_exponents_array):
        if exponent != 0:
            swing_factorial *= prime ** exponent
    return swing_factorial

def is_prime(n: int) -> bool:
    """
    Checks if a number is prime.
    
    :param n: The number to check
    :return: True if the number is prime, False otherwise
    """
    if n <= 1:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False
    max_divisor = math.isqrt(n)
    for divisor in range(3, max_divisor + 1, 2):
        if n % divisor == 0:
            return False
    return True

def all_prime_numbers_up_to(n: int) -> List[int]:
    """
    Calculates all prime numbers up to a given number.
    
    :param n: The upper limit for prime number calculation
    :return: List of all prime numbers up to the given number
    """
    return [num for num in range(2, n + 1) if is_prime(num)]
