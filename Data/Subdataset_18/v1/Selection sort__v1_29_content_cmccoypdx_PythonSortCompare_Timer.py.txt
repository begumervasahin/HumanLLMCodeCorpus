import timeit

def wrapper(func, *args, **kwargs):
    """Wrap the function and its arguments."""
    def wrapped():
        return func(*args, **kwargs)
    return wrapped

def TimeFunc(func, *args, **kwargs):
    """Time the execution of a function with the provided arguments."""
    wrapped = wrapper(func, *args, **kwargs)
    return timeit.timeit(wrapped, number=1)

# Example usage:
if __name__ == "__main__":
    # Define a sample function to test
    def sample_function(x, y):
        return x + y

    # Time the sample function with arguments
    time_taken = TimeFunc(sample_function, 3, 5)
    print(f"Time taken to execute sample_function: {time_taken} seconds")
