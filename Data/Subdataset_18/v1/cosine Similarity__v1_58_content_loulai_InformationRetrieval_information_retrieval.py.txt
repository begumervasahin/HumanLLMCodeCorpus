import re
from stop_list import closed_class_stop_words

# Read and process the queries from the file
queries = open("cran.qry", 'r').read()
queries = queries.replace("\s", "")  # Fix replace method usage

def get_query_nums(string):
    """
    Extract query numbers from the input string.
    
    Args:
        string (str): The input string containing query numbers.
    
    Returns:
        list: A list of query numbers.
    """
    query_nums = []
    matches = re.findall(r"\.I \d{3}", string)
    for match in matches:
        query_nums.append(match.replace(".I ", ""))
    return query_nums

def get_query_strings(string):
    """
    Extract query strings from the input string.
    
    Args:
        string (str): The input string containing queries.
    
    Returns:
        list: A list of query strings.
    """
    query_strings = []
    matches = re.findall(r"([a-z ]+\n){1,3}[a-z ]+\.", string)
    for match in matches:
        query_strings.append(match.strip())
    return query_strings

def trim(string):
    """
    Remove stopwords and trim the input string.
    
    Args:
        string (str): The input string to be trimmed.
    
    Returns:
        list: A list of trimmed words.
    """
    words = re.sub(r"\n|\r", " ", string).split()
    trimmed_words = [word for word in words if word not in closed_class_stop_words]
    return trimmed_words

def get_tf(array):
    """
    Calculate term frequency in the given array of words.
    
    Args:
        array (list): A list of words.
    
    Returns:
        dict: A dictionary of term frequencies.
    """
    terms = list(set(array))
    term_frequency = {term: array.count(term) for term in terms}
    return term_frequency

def print_dict(dictionary):
    """
    Print a dictionary's keys and values.
    
    Args:
        dictionary (dict): The dictionary to print.
    """
    for key, value in dictionary.items():
        print(f"{key}: {value}")

    query_strings = get_query_strings(queries)

    trimmed_content = trim(open("cran2.qry", 'r').read())
    print(trimmed_content)
    
    term_frequency = get_tf(trimmed_content)
