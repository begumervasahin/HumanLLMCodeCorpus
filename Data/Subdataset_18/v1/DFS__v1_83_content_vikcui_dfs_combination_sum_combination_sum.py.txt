def combination_sum_aux(candidates, target):
    """
    Find all unique combinations of candidates where the candidate numbers sum up to the target.
    
    Args:
        candidates (list of int): A list of candidate numbers (without duplicates).
        target (int): The target sum.
    
    Returns:
        list of list of int: A list of unique combinations that sum up to the target.
    """
    # Sort the candidates to facilitate the combination search
    candidates.sort()
    
    # Initialize result list and temporary sequence list
    result_list = []
    result_seq = []
    
    # Start the recursive combination search
    combination_sum(candidates, target, 0, result_seq, result_list)
    
    return result_list

def combination_sum(candidates, target, current_index, result_seq, result_list):
    """
    Helper function to find combinations that sum up to the target using recursion.
    
    Args:
        candidates (list of int): A list of candidate numbers.
        target (int): The remaining target sum.
        current_index (int): The current index in the candidates list.
        result_seq (list of int): The current combination sequence being explored.
        result_list (list of list of int): The list to store all valid combinations.
    """
    # Iterate through the candidates starting from the current index
    for j in range(current_index, len(candidates)):
        num = candidates[j]
        
        # If the current number exceeds the target, no need to proceed further
        if num > target:
            break
        
        # Add the number to the current sequence
        result_seq.append(num)
        
        # If the target is still positive, continue the recursion with updated target
        if num < target:
            combination_sum(candidates, target - num, j, result_seq, result_list)
        else:
            # If the target is zero, we found a valid combination
            result_list.append(list(result_seq))
        
        # Backtrack by removing the last added number
        result_seq.pop()

    candidates = [2, 3, 6, 7]
    target = 7
    combinations = combination_sum_aux(candidates, target)
    print("Unique combinations:", combinations)
