class PriorityQueue:
    def __init__(self):
        self.fila = [(0, 0)]
        self.tamanho = 0

    def getTamanho(self):
        return self.tamanho

    def find_min(self):
        if self.tamanho != 0:
            return self.fila[1]

    def remove_min(self):
        minimo = self.find_min()
        self.fila[1] = self.fila[self.tamanho]
        self.tamanho -= 1
        self.fila.pop()
        self.fix_down(1)
        return minimo

    def insert(self, elemento):
        self.fila.append(elemento)
        self.tamanho += 1
        self.fix_up(self.tamanho)

    def increase_key(self, node, valor):
        cont = 0
        for i in self.fila:
            if node == i[1]:
                self.fila[cont] = (valor, node)
                self.fix_up(cont)
                break
            cont += 1

    def __contains__(self, elemento):
        for pair in self.fila:
            if pair == elemento:
                return True
        return False

    def fix_down(self, i):
        while 2 * i <= self.tamanho:
            f = 2 * i
            if f < self.tamanho and self.fila[f][0] > self.fila[f + 1][0]:
                f += 1
            if self.fila[i][0] <= self.fila[f][0]:
                break
            else:
                self.fila[i], self.fila[f] = self.fila[f], self.fila[i]
                i = f

    def build_min_heap(self, lista):
        self.fila = [(0, 0)]
        self.tamanho = len(lista)
        self.fila.extend(lista)
        i = len(lista) // 2
        while i >= 1:
            self.fix_down(i)
            i -= 1

    def fix_up(self, i):
        while i >= 2 and self.fila[i // 2][0] > self.fila[i][0]:
            self.fila[i // 2], self.fila[i] = self.fila[i], self.fila[i // 2]
            i //= 2
