def find_primes(limit):
    n = 2
    rest = []
    primes = [1]
    count = 0
    
    while n <= limit:
        i = 0
        m = primes[i]
        while m < n:
            p = n % m
            rest.append(p)
            count = rest.count(0)
            i += 1
            if (i + 1) > len(primes):
                break
            m = primes[i]
            if count > 1:
                break
        if count == 1:
            print(f"{n} is prime")
            primes.append(n)
        if n == 2:
            n += 1
        else:
            n += 2
        rest = []

if __name__ == "__main__":
    find_primes(3001)
