def three_plus_one(x):
    i = 0
    while x > 1:
        i += 1  # Increment counter
        if x % 2 == 1:  # Check if x is odd
            x = 3 * x + 1
        else:  # If x is even
            x = x // 2  # Use integer division to ensure x remains an integer
        print(x)

def three_plus_one_main(x):
    three_plus_one(x)

if __name__ == "__main__":
    three_plus_one_main(15000)
