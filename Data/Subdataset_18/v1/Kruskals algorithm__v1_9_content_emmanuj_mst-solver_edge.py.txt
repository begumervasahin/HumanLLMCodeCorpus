class Edge:
    def __init__(self, b, e, w):
        """
        Initializes an Edge object.

        Parameters:
        b (int or str): The starting vertex of the edge.
        e (int or str): The ending vertex of the edge.
        w (float or int): The weight of the edge.
        """
        self.u = b  # Start vertex
        self.v = e  # End vertex
        self.weight = w  # Weight of the edge

    def __repr__(self):
        """
        Returns a string representation of the Edge object.
        """
        return "e " + str(self.u) + " " + str(self.v) + " " + str(self.weight)

# Example usage
if __name__ == "__main__":
    # Create an edge from vertex 1 to vertex 2 with a weight of 3.5
    edge = Edge(1, 2, 3.5)
    
    # Print the edge
    print(edge)  # Output: e 1 2 3.5
