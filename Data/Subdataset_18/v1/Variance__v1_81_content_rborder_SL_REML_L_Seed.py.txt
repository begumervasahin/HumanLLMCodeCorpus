import numpy as np
from scipy import linalg as la
from scipy.sparse import linalg as spla

def L_Seed(
        A: spla.LinearOperator,
        B: np.ndarray,
        tol=5e-3,
        p_freq=5,
        maxit=50,
        verbose=True
    ) -> dict:
    """ see alg * in the paper """
    n = A.shape[0]
    t = B.shape[1]
    X = np.zeros((n, t, maxit))
    R = np.zeros((n, t, maxit))
    U = np.zeros((n, t, maxit))
    ρ = np.zeros((t, maxit))
    β = np.zeros((t, maxit))
    ω = np.zeros((t, maxit))
    γ = np.ones((t, maxit))
    δ = np.zeros((t, maxit))

    ρ[:, 0] = np.apply_along_axis(la.norm, 0, B)
    β[:, 0] = ρ[:, 0]
    R[:, :, 0] = U[:, :, 0] = B
    U[:, :, 0] = R[:, :, 0] / β[:, 0]

    j = 0
    cnvg = False

    while j < maxit - 1:
        tmp = A @ U[:, :, j]
        δ[:, j] = np.diag(U[:, :, j].T @ tmp)
        tmp = tmp - U[:, :, j] * δ[:, j] - β[:, j] * U[:, :, j - 1]
        β[:, j + 1] = np.apply_along_axis(la.norm, 0, tmp)
        U[:, :, j + 1] = tmp / β[:, j + 1]
        γ[:, j] = (δ[:, j] - ω[:, j - 1] / γ[:, j - 1]) ** -1
        ω[:, j] = (β[:, j + 1] * γ[:, j]) ** 2
        ρ[:, j + 1] = -β[:, j + 1] * γ[:, j] * ρ[:, j]
        R[:, :, j + 1] = ρ[:, j + 1] * U[:, :, j + 1]
        j += 1

        res_norm = np.amax(np.apply_along_axis(la.norm, 0, R[:, :, j]))
        if verbose and j % p_freq == 0:
            print("Error at step ", j, " is ", res_norm)

        if res_norm <= tol:
            cnvg = True
            break

    if verbose and cnvg:
        print("Converged after ", j, " iterations.")
    elif verbose:
        print("Failed to converge after ", j, " iterations.")

    return {
        "U": U[:, :, :j],
        "β": β[:, :j],
        "δ": δ[:, :j],
        "ρ": ρ[:, :j]
    }
