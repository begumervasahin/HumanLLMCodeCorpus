def new_ascii(ch):
    """
    Converts a character to its ASCII value and adds 100.

    Args:
        ch (str): The character to convert.

    Returns:
        int: The new ASCII value.
    """
    return ord(ch) + 100

def old_ascii(ascii_val):
    """
    Converts a modified ASCII value back to its original ASCII value by subtracting 100.

    Args:
        ascii_val (int): The modified ASCII value.

    Returns:
        int: The original ASCII value.
    """
    return int(ascii_val) - 100

def encode(msg):
    """
    Encodes a message by converting each character to its modified ASCII value.

    Args:
        msg (str): The message to encode.

    Returns:
        str: The encoded message as a string of modified ASCII values.
    """
    return ''.join(str(new_ascii(ch)).zfill(3) for ch in msg)

def decode(encoded_msg):
    """
    Decodes an encoded message back to its original form.

    Args:
        encoded_msg (str): The encoded message as a string of modified ASCII values.

    Returns:
        str: The decoded original message.
    """
    return ''.join(chr(old_ascii(encoded_msg[i:i+3])) for i in range(0, len(encoded_msg), 3))

if __name__ == "__main__":
    # Example usage
    original_message = "Hello, World!"
    print("Original Message:", original_message)

    encoded_message = encode(original_message)
    print("Encoded Message:", encoded_message)

    decoded_message = decode(encoded_message)
    print("Decoded Message:", decoded_message)
