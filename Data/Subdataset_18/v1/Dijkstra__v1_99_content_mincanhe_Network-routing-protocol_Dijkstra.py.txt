import collections

class Graph:
    def __init__(self):
        """
        Initialize an empty graph with nodes, edges, and distances.
        """
        self.nodes = set()
        self.edges = collections.defaultdict(list)
        self.distances = {}

    def add_node(self, value):
        """
        Add a node to the graph.

        Parameters:
        value (any): The value of the node to be added.
        """
        self.nodes.add(value)

    def add_edge(self, from_node, to_node, distance):
        """
        Add an edge to the graph.

        Parameters:
        from_node (any): The starting node of the edge.
        to_node (any): The ending node of the edge.
        distance (float): The distance (weight) of the edge.
        """
        self.edges[from_node].append(to_node)
        self.edges[to_node].append(from_node)
        self.distances[(from_node, to_node)] = distance
        self.distances[(to_node, from_node)] = distance

def dijkstra(graph, initial):
    """
    Compute the shortest paths from the initial node to all other nodes in the graph using Dijkstra's algorithm.

    Parameters:
    graph (Graph): The graph to compute shortest paths for.
    initial (any): The starting node for the algorithm.

    Returns:
    tuple: A tuple containing two dictionaries:
        - distances (dict): Shortest distances from the initial node to each node.
        - path (dict): Predecessors of each node in the shortest path.
    """
    visited = {initial: 0}
    path = {}
    nodes = set(graph.nodes)

    while nodes:
        # Select the unvisited node with the smallest distance
        min_node = None
        for node in nodes:
            if node in visited:
                if min_node is None:
                    min_node = node
                elif visited[node] < visited[min_node]:
                    min_node = node

        if min_node is None:
            break

        nodes.remove(min_node)
        current_weight = visited[min_node]

        # Update the distances to neighboring nodes
        for neighbor in graph.edges[min_node]:
            weight = current_weight + graph.distances[(min_node, neighbor)]
            if neighbor not in visited or weight < visited[neighbor]:
                visited[neighbor] = weight
                path[neighbor] = min_node

    return visited, path
