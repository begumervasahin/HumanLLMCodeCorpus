def create_matrix(rows, cols):
    """
    Creates a matrix initialized with '-' for each cell.

    :param rows: Number of rows in the matrix.
    :param cols: Number of columns in the matrix.
    :return: A 2D list representing the matrix.
    """
    return [["-" for _ in range(cols)] for _ in range(rows)]

def is_cell_free(matrix, row, col):
    """
    Checks if a cell in the matrix is free.

    :param matrix: The matrix to check.
    :param row: Row index.
    :param col: Column index.
    :return: True if the cell is free, False otherwise.
    """
    return matrix[row][col] == "-"

def is_cell_valid(row, col, size):
    """
    Checks if the cell indices are valid within the matrix boundaries.

    :param row: Row index.
    :param col: Column index.
    :param size: Size of the matrix.
    :return: True if the cell indices are valid, False otherwise.
    """
    return 0 <= row < size and 0 <= col < size

def display_matrix(matrix):
    """
    Displays the matrix in a readable format.

    :param matrix: The matrix to display.
    """
    for row in matrix:
        print("\t".join(row))
        print()

def assign_move(matrix, row, col, player):
    """
    Assigns a move to the matrix for the current player.

    :param matrix: The matrix to update.
    :param row: Row index.
    :param col: Column index.
    :param player: The current player (1 or 2).
    """
    marker = "X" if player == 1 else "O"
    matrix[row][col] = marker

def check_rows(matrix, marker):
    """
    Checks for a winning row.

    :param matrix: The matrix to check.
    :param marker: The marker to check (X or O).
    :return: True if there is a winning row, False otherwise.
    """
    for row in matrix:
        if all(cell == marker for cell in row):
            return True
    return False

def check_columns(matrix, marker):
    """
    Checks for a winning column.

    :param matrix: The matrix to check.
    :param marker: The marker to check (X or O).
    :return: True if there is a winning column, False otherwise.
    """
    for col in range(len(matrix)):
        if all(matrix[row][col] == marker for row in range(len(matrix))):
            return True
    return False

def check_diagonals(matrix, marker):
    """
    Checks for a winning diagonal.

    :param matrix: The matrix to check.
    :param marker: The marker to check (X or O).
    :return: True if there is a winning diagonal, False otherwise.
    """
    size = len(matrix)
    if all(matrix[i][i] == marker for i in range(size)):
        return True
    if all(matrix[i][size - 1 - i] == marker for i in range(size)):
        return True
    return False

def check_winner(matrix, player):
    """
    Checks if the current player has won the game.

    :param matrix: The matrix to check.
    :param player: The current player (1 or 2).
    :return: True if the player has won, False otherwise.
    """
    marker = "X" if player == 1 else "O"
    return (check_rows(matrix, marker) or
            check_columns(matrix, marker) or
            check_diagonals(matrix, marker))

def get_move(matrix, player):
    """
    Gets a valid move from the player and updates the matrix.

    :param matrix: The matrix to update.
    :param player: The current player (1 or 2).
    :return: True if the player wins after the move, False otherwise.
    """
    while True:
        try:
            print(f"\nPlayer {player}'s turn")
            row = int(input("Enter the row index (0, 1, 2): "))
            col = int(input("Enter the column index (0, 1, 2): "))

            if is_cell_valid(row, col, len(matrix)) and is_cell_free(matrix, row, col):
                assign_move(matrix, row, col, player)
                return check_winner(matrix, player)
            else:
                print("Invalid or occupied cell, please try again.")
        except ValueError:
            print("Invalid input, please enter integers for row and column.")

def main():
    """
    Main function to run the Tic-Tac-Toe game.
    """
    size = 3
    matrix = create_matrix(size, size)
    moves = 0
    current_player = 1

    while moves < size * size:
        display_matrix(matrix)
        if get_move(matrix, current_player):
            display_matrix(matrix)
            print(f"Congratulations! Player {current_player} wins!")
            return
        
        moves += 1
        current_player = 2 if current_player == 1 else 1

    display_matrix(matrix)
    print("The game is a draw!")

if __name__ == "__main__":
    main()
