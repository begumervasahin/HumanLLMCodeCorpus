def is_prime(digit):
    """Check if a number is prime."""
    if digit < 2:
        return False
    elif digit == 2:
        return True
    for number in range(2, digit):
        if digit % number == 0:
            return False
    return True

def prime(number):
    """Check if a number is prime and print numbers in the range."""
    if number < 2:
        return False
    elif number == 2:
        return False
    else:
        for x in range(2, number):
            print(x)  # This prints numbers from 2 to number-1
            if number % x == 0:
                return False
        return True

def check_if_prime_number(number):
    """Check if a number is prime."""
    if number < 2:
        return False
    for x in range(2, number - 1):
        if number % x == 0:
            return False
    return True
