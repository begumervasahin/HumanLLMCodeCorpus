import numpy as np
from scipy.linalg import svd
from scipy.stats import f_oneway

def swpca(dataset, catvar, k=0.05, trset=None):
    N = dataset.shape[0]
    if trset is None:
        trset = np.ones(N, dtype=bool)
        training = False
    else:
        training = True

    # Subtract subject mean
    subjMean = dataset.mean(axis=1)
    X = (dataset.T - subjMean).T

    # Subtract mean and normalize by variance for training set
    meanTr = X[trset].mean(axis=0)
    X = X - meanTr
    varTr = X[trset].var(axis=0)
    varTr = np.maximum(varTr, 1)
    X = X / varTr

    if training:
        XNTEST = X[~trset]
    XNTRAIN = X[trset]

    # Perform SVD
    U, s, Wt = svd(XNTRAIN, full_matrices=False)
    W = Wt.T

    # Sort by descending singular values
    ind = np.argsort(s)[::-1]
    U = U[:, ind]
    s = s[ind]
    W = W[:, ind]

    STr = XNTRAIN.dot(W)
    if training:
        STe = XNTEST.dot(W)

    # Calculate ANOVA F-statistics and weights
    labs = np.unique(catvar)
    F, p_val = f_oneway(STr[catvar[trset] == labs[0]], STr[catvar[trset] == labs[1]])
    weights = 1 - np.exp(-p_val / k)

    # Reconstruct the data
    A = np.linalg.pinv(W)
    weightMat = np.diag(weights)
    if training:
        XTRhat = STr.dot(weightMat).dot(A)
        XTEhat = STe.dot(weightMat).dot(A)
        Xhat = np.zeros((N, XTRhat.shape[1]))
        Xhat[trset] = XTRhat
        Xhat[~trset] = XTEhat
    else:
        Xhat = STr.dot(weightMat).dot(A)

    Xhat = Xhat * varTr
    Xhat = Xhat + meanTr
    Xhat = (Xhat.T + subjMean).T

    return Xhat, weights, A

# Example usage
if __name__ == "__main__":
    # Example dataset and category variable
    dataset = np.random.rand(100, 10)  # Replace with your dataset
    catvar = np.random.randint(0, 2, 100)  # Replace with your categorical variable

    # Run SWPCA
    Xhat, weights, A = swpca(dataset, catvar)

    # Print the results
    print("Reconstructed Dataset:\n", Xhat)
    print("Weights:\n", weights)
    print("Transformation Matrix:\n", A)
