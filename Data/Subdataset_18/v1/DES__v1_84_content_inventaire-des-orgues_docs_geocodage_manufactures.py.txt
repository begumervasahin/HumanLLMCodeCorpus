import json
import requests
import geojson

# URL for the French government geocoding API
GEOCODING_URL = "https://api-adresse.data.gouv.fr/search/?q="

def geocode_address(address):
    """
    Geocode an address using the French government geocoding API.

    :param address: The address to geocode.
    :return: A tuple (latitude, longitude) of the geocoded address.
    """
    address_query = address.replace(" ", "+")
    response = requests.get(GEOCODING_URL + address_query)
    feature_collection = geojson.loads(response.text)
    point = feature_collection['features'][0]['geometry']['coordinates']
    return point[1], point[0]

def main():
    # Load the data from datas.json
    with open('datas.json') as json_file:
        data_dict = json.load(json_file)

    # Geocode each address and update the data dictionary
    for facteur in data_dict:
        address = facteur['adresse']
        latitude, longitude = geocode_address(address)
        facteur['latitude'] = latitude
        facteur['longitude'] = longitude

    # Save the updated data to facteurs_localisation.json
    with open('facteurs_localisation.json', 'w') as output_file:
        json.dump(data_dict, output_file, ensure_ascii=False, indent=4)

if __name__ == "__main__":
    main()
