import numpy as np

def var1(n):
    avg = n.mean()
    delta = n - avg
    delta_squared_sum = np.dot(delta, delta)
    return delta_squared_sum / n.size

def var2(n):
    avg = n.mean()
    n_sq_sum = np.dot(n, n)
    E_n_sq = n_sq_sum / n.size
    return E_n_sq - avg**2

def rs1(n):
    k = 1
    M = n[0]
    S = 0
    for ni in n:
        M_old = M
        M = M_old + (ni - M_old) / k
        S = S + (ni - M_old) * (ni - M)
        k += 1
    var = S / (k - 1)
    return M, var

def rs2(n):
    M = n[0]
    S = 0
    for k in range(1, n.size):
        M_old = M
        M = M_old + (n[k] - M_old) / k
        S = S + (n[k] - M_old) * (n[k] - M)
        k += 1
    var = S / (k - 1)
    return M, var

def rs3(n):
    i = 0
    M = 0.0
    S = 0.0
    for ni in n:
        i += 1
        delta = ni - M
        M += delta / i
        S += delta * (ni - M)
    if i < 2:
        return n[0], 0
    else:
        return M, S / (i - 1)
