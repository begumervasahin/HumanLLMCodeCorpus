def heapify(H, n, i):
    """
    Ensures the heap property for a subtree rooted at index i.
    
    Parameters:
    H (list): The heap (or array) to be heapified.
    n (int): The size of the heap.
    i (int): The index of the root of the subtree.
    """
    largest = i  # Initialize largest as root
    l = 2 * i + 1  # Left child index
    r = 2 * i + 2  # Right child index

    # If left child exists and is greater than root
    if l < n and H[i] > H[l]:
        largest = l

    # If right child exists and is greater than the largest so far
    if r < n and H[largest] > H[r]:
        largest = r

    # If largest is not root
    if largest != i:
        H[i], H[largest] = H[largest], H[i]  # Swap
        heapify(H, n, largest)  # Recursively heapify the affected subtree

def heap_sort(H):
    """
    Performs heap sort on the provided list.
    
    Parameters:
    H (list): The list to be sorted.
    """
    n = len(H)

    # Build a max heap
    for i in range(n // 2 - 1, -1, -1):
        heapify(H, n, i)

    # Extract elements one by one from the heap
    for i in range(n - 1, 0, -1):
        H[i], H[0] = H[0], H[i]  # Swap current root to end
        heapify(H, i, 0)  # Call heapify on the reduced heap

# Example usage
if __name__ == "__main__":
    H = [12, 11, 13, 5, 6, 7]
    print("Original array:", H)
    heap_sort(H)
    print("Sorted array:", H)
