import heapq
from collections import namedtuple, defaultdict

# Define a simple Edge namedtuple
Edge = namedtuple('Edge', ['vertex', 'weight'])

class DijkstraWithMaxHeap:
    def __init__(self, num_vertices):
        self.num_vertices = num_vertices
        self.status = ['unseen'] * num_vertices
        self.weights = [float('-inf')] * num_vertices
        self.dad = [None] * num_vertices
        self.graph = defaultdict(list)

    def add_edge(self, u, v, weight):
        self.graph[u].append(Edge(vertex=v, weight=weight))
        self.graph[v].append(Edge(vertex=u, weight=weight))  # Assuming undirected graph

    def max_heap_insert(self, heap, vertex, weight):
        heapq.heappush(heap, (-weight, vertex))

    def max_heap_pop(self, heap):
        weight, vertex = heapq.heappop(heap)
        return -weight, vertex

    def run_dijkstra(self, source, destination):
        heap = []
        self.status[source] = 'intree'
        for edge in self.graph[source]:
            self.status[edge.vertex] = 'fringe'
            self.weights[edge.vertex] = edge.weight
            self.max_heap_insert(heap, edge.vertex, edge.weight)
            self.dad[edge.vertex] = source

        while heap:
            max_weight, v = self.max_heap_pop(heap)
            if v == destination:
                break
            self.status[v] = 'intree'
            for edge in self.graph[v]:
                if self.status[edge.vertex] == 'unseen':
                    self.status[edge.vertex] = 'fringe'
                    self.dad[edge.vertex] = v
                    self.weights[edge.vertex] = min(max_weight, edge.weight)
                    self.max_heap_insert(heap, edge.vertex, self.weights[edge.vertex])
                elif self.status[edge.vertex] == 'fringe' and self.weights[edge.vertex] < min(max_weight, edge.weight):
                    self.dad[edge.vertex] = v
                    self.weights[edge.vertex] = min(max_weight, edge.weight)
                    self.max_heap_insert(heap, edge.vertex, self.weights[edge.vertex])

        return self.build_path(destination), self.weights[destination]

    def build_path(self, destination):
        path = []
        while destination is not None:
            path.append(destination)
            destination = self.dad[destination]
        path.reverse()
        return path
