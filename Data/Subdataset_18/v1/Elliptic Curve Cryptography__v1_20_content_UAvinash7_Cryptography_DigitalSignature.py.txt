from Crypto.Hash import SHA256
from Crypto.PublicKey import RSA
from Crypto.Signature import pkcs1_15
from Crypto import Random

def generate_key_pair():
    """
    Generate an RSA key pair.
    
    :return: Tuple containing the private key and public key.
    """
    random_generator = Random.new().read
    key_pair = RSA.generate(1024, random_generator)
    return key_pair, key_pair.publickey()

def create_hash(message):
    """
    Create a SHA-256 hash of a message.
    
    :param message: The message to hash.
    :return: The hash of the message.
    """
    return SHA256.new(message.encode())

def sign_hash(private_key, hash_value):
    """
    Sign a hash using the RSA private key.
    
    :param private_key: The RSA private key.
    :param hash_value: The hash value to sign.
    :return: The digital signature.
    """
    return pkcs1_15.new(private_key).sign(hash_value)

def verify_signature(public_key, hash_value, signature):
    """
    Verify a digital signature using the RSA public key.
    
    :param public_key: The RSA public key.
    :param hash_value: The hash value that was signed.
    :param signature: The digital signature.
    :return: Boolean indicating whether the signature is valid.
    """
    try:
        pkcs1_15.new(public_key).verify(hash_value, signature)
        return True
    except (ValueError, TypeError):
        return False

def main():
    """
    Main function to demonstrate RSA signing and verification.
    """
    # Generate RSA key pair
    private_key, public_key = generate_key_pair()

    # Plain text message
    plain_text = "Hello World"

    # Create hash of the message
    hash_a = create_hash(plain_text)
    print("Hash A:", hash_a.hexdigest(), "\n")

    # Sign the hash
    digital_signature = sign_hash(private_key, hash_a)
    print("Digital Signature:", repr(digital_signature), "\n")

    # Create another hash of the same message
    hash_b = create_hash(plain_text)
    print("Hash B:", hash_b.hexdigest(), "\n")

    # Verify the signature
    if verify_signature(public_key, hash_b, digital_signature):
        print("Match")
    else:
        print("No Match")

if __name__ == "__main__":
    main()
