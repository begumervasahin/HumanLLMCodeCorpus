import tkinter as tk
from tkinter import simpledialog

def busqueda():
    try:
        # Get the number of elements to input
        datos = int(entry_numeros.get())

        # Initialize an empty list to hold the numbers
        vector = []

        # Ask for the number to search
        numerito = simpledialog.askinteger("Buscar Número", "Ingresa el número a buscar:")

        # Get the list of numbers from the user
        for i in range(datos):
            numero = simpledialog.askinteger("Ingresar Número", f"Ingrese el número {i + 1}:")
            vector.append(numero)

        # Display the entered list
        aux_impresion1 = ', '.join(map(str, vector))
        label_resultado_lista.config(text=f"Números ingresados: {aux_impresion1}")

        # Perform a sequential search
        for i, num in enumerate(vector):
            if num == numerito:
                resultado_final = f"Encontrado en la posición: {i}"
                label_resultado_busqueda.config(text=resultado_final)
                return

        # If the number was not found
        label_resultado_busqueda.config(text="Número no encontrado")

    except ValueError:
        label_resultado_busqueda.config(text="Por favor, ingresa un número válido")

# Set up the main application window
app = tk.Tk()
app.title("Búsqueda Secuencial")
app.geometry('300x200')
app.configure(bg='SkyBlue2')

# Create and pack the widgets
label_instruccion = tk.Label(app, text="¿Cuántos números vas a ingresar?", font='Helvetica 12', bg='SkyBlue2')
label_instruccion.pack()

entry_numeros = tk.Entry(app, width=8)
entry_numeros.pack()

button_buscar = tk.Button(app, text="Ok!", command=busqueda)
button_buscar.pack(pady=(10, 0))

label_resultado_lista = tk.Label(app, text="", bg='SkyBlue2')
label_resultado_lista.pack(pady=(10, 0))

label_resultado_busqueda = tk.Label(app, text="", bg='SkyBlue2')
label_resultado_busqueda.pack(pady=(10, 0))

# Run the main loop
app.mainloop()
