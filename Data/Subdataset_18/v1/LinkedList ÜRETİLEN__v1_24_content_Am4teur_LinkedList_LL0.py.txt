class Node:
    def __init__(self, value):
        self.value = value
        self.next = None

if __name__ == "__main__":
    # Create the linked list
    ll = Node(0)           # First node with value 0
    ll.next = Node(1)      # Second node with value 1
    ll.next.next = Node(2) # Third node with value 2

    # Print the values of the linked list nodes
    print(f'value = {ll.value}')             # Output: value = 0
    print(f'value = {ll.next.value}')        # Output: value = 1
    print(f'value = {ll.next.next.value}')   # Output: value = 2
