import numpy as np
import matplotlib.pyplot as plt
from matplotlib.backends.backend_pdf import PdfPages
import sys

if len(sys.argv) != 5:
    print(f"{sys.argv[0]} takes 4 arguments. Not {len(sys.argv)-1}")
    sys.exit()

first = sys.argv[1]
second = sys.argv[2]
third = sys.argv[3]
fourth = sys.argv[4]

# Load data
Xt = np.genfromtxt(first, delimiter=',')
y = np.genfromtxt(second, autostrip=True)

# Get dimensions and unique labels
m, n = Xt.shape
unique_labels = set(y)

# Calculate global mean
mu_global = np.mean(Xt, axis=0)

# Initialize within-class and between-class scatter matrices
W_class = np.zeros((n, n))
B_class = np.zeros((n, n))

# Calculate between-class scatter matrix
for c in unique_labels:
    B1 = np.zeros((n, n))
    BT1 = np.zeros((n, n))
    s1 = np.zeros((1, n))
    C1 = np.zeros((n, n))
    m1 = 0
    
    for i, xt in enumerate(Xt):
        if y[i] == c:
            B1 += np.dot(xt.T, xt)
            s1 += xt
            m1 += 1
            
    mu1 = s1 / m1
    
    for i, xt in enumerate(Xt):
        if y[i] == c:
            ct = xt - mu1
            bt = mu1 - mu_global
            C1 += np.dot(ct.T, ct)
            BT1 += np.dot(bt.T, bt)
    
    B_class += BT1

# Eigen decomposition
evals, evecs = np.linalg.eigh(B_class)
idx = np.argsort(evals)[::-1]
evecs = evecs[:, idx]
evals = evals[idx]

# Select top r eigenvectors
r = 2
V_r = evecs[:, :r]

# Project data onto new feature space
result = np.dot(Xt, V_r)
x1 = result[:, 0]
x2 = result[:, 1]

# Plotting the results
fig, ax = plt.subplots()
ax.scatter(x1, x2)
plt.title("Maximize between class scatter")
plt.xlabel("PC1")
plt.ylabel("PC2")
plotname = 'scatter_2_plot'
pdf = PdfPages(plotname + '.pdf')
pdf.savefig(fig)
pdf.close()

# Save the results
np.savetxt(third, V_r.T, delimiter=',')
np.savetxt(fourth, result, delimiter=',')
