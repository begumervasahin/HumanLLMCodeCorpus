import random

def partition3(a, l, r):
    """
    Perform a three-way partitioning of the array.
    
    :param a: The list to be partitioned.
    :param l: The starting index of the partition.
    :param r: The ending index of the partition.
    :return: A tuple containing the indices for the left and right partitions.
    """
    x = a[l]
    j, o = l, l
    for i in range(l + 1, r + 1):
        if a[i] < x:
            o += 1
            a[i], a[o] = a[o], a[i]
            a[j], a[o] = a[o], a[j]
            j += 1
        elif a[i] == x:
            o += 1
            a[i], a[o] = a[o], a[i]
        else:
            continue
    if j > l:
        a[l], a[j - 1] = a[j - 1], a[l]
    else:
        a[l], a[j] = a[j], a[l]
    return j, o

def randomized_quick_sort3(a, l, r):
    """
    Perform the randomized QuickSort with a three-way partitioning.
    
    :param a: The list to be sorted.
    :param l: The starting index of the sort.
    :param r: The ending index of the sort.
    """
    if l >= r:
        return
    k = random.randint(l, r)
    a[l], a[k] = a[k], a[l]
    m1, m2 = partition3(a, l, r)
    randomized_quick_sort3(a, l, m1 - 1)
    randomized_quick_sort3(a, m2 + 1, r)

if __name__ == '__main__':
    input_data = "5\n2 3 9 2 2"
    input_data = list(map(int, input_data.split()))
    n = input_data[0]
    a = input_data[1:]
    randomized_quick_sort3(a, 0, n - 1)
    for x in a:
        print(x, end=' ')
    print("\nSorted array:", a)
