import random

def partition(a, l, r):
    """
    Partition the subarray a[l:r+1] using Hoare's partition scheme.
    """
    pivot = a[l]
    i = l
    j = r + 1
    
    while True:
        # Move i to the right until a[i] >= pivot
        while True:
            i += 1
            if i >= len(a) or a[i] >= pivot:
                break
        
        # Move j to the left until a[j] <= pivot
        while True:
            j -= 1
            if j < 0 or a[j] <= pivot:
                break
        
        # If two indices crossed, partitioning is complete
        if i >= j:
            break
        
        # Swap elements at i and j
        a[i], a[j] = a[j], a[i]
    
    # Swap pivot element with element at j
    a[l], a[j] = a[j], a[l]
    
    return j

def quicksort_helper(a, l, r):
    """
    Recursively apply quicksort to the subarray a[l:r+1].
    """
    if l < r:
        # Partition the array and get the partition index
        s = partition(a, l, r)
        
        # Recursively apply quicksort to the left and right subarrays
        quicksort_helper(a, l, s - 1)
        quicksort_helper(a, s + 1, r)

def quicksort(a):
    """
    Perform quicksort on the entire array.
    """
    quicksort_helper(a, 0, len(a) - 1)

if __name__ == "__main__":
    arr = [random.randint(1, 100) for _ in range(10)]
    print("Array before sorting:", arr)
    quicksort(arr)
    print("Array after sorting:", arr)
