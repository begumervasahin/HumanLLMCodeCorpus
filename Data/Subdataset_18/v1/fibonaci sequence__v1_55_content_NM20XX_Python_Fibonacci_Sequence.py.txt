def fib(n):
    """Generator function to yield Fibonacci sequence up to n terms."""
    x, y = 0, 1
    for _ in range(n):
        yield x
        x, y = y, x + y

def get_input():
    """Prompt the user to enter a term for the Fibonacci sequence."""
    while True:
        try:
            term = int(input("Enter a term in the Fibonacci sequence: "))
            if term <= 0:
                print("Please enter a positive integer.")
            else:
                return term
        except ValueError:
            print("Error: Please enter an integer value.")

def main():
    input_term = get_input()
    print(f"Fibonacci sequence up to {input_term} terms:")
    for number in fib(input_term):
        print(number, end=", ")
    print()  # For a newline after the sequence

if __name__ == "__main__":
    main()
