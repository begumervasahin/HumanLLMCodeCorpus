def encrypt(plain_text, shift):
    """
    Encrypts the given plain text using a Caesar cipher with the specified shift.

    Parameters
    ----------
    plain_text : str
        The text to be encrypted.
    shift : int
        The number of positions to shift each character in the plain text.

    Returns
    -------
    str
        The encrypted text.
    """
    encrypted_text = ""
    
    # Loop through each character in the plain text
    for char in plain_text:
        # Check if the character is uppercase
        if char.isupper():
            # Encrypt uppercase characters
            encrypted_text += chr((ord(char) + shift - 65) % 26 + 65)
        # Otherwise, encrypt lowercase characters
        else:
            encrypted_text += chr((ord(char) + shift - 97) % 26 + 97)
    
    return encrypted_text

# Example usage
plain_text = "Vineet"
shift = 4
encrypted_text = encrypt(plain_text, shift)
print("ENCRYPTED TEXT: " + encrypted_text)
