class BinarySearchTreeNode:
    def __init__(self, value):
        self.value = value
        self.left = None  # Left child
        self.right = None  # Right child

    def insert(self, value):
        if value < self.value:  # If the value is less than the current node's value
            if not self.left:  # If no left child exists
                self.left = BinarySearchTreeNode(value)  # Create a new left child
            else:
                self.left.insert(value)  # Recursively insert into the left subtree
        elif value > self.value:  # If the value is greater than the current node's value
            if not self.right:  # If no right child exists
                self.right = BinarySearchTreeNode(value)  # Create a new right child
            else:
                self.right.insert(value)  # Recursively insert into the right subtree

    def lookup(self, value):
        if self.value == value:  # If the current node is the one we're looking for
            return self
        elif value < self.value and self.left:  # If the value is less and there's a left child
            return self.left.lookup(value)  # Recursively look in the left subtree
        elif value > self.value and self.right:  # If the value is greater and there's a right child
            return self.right.lookup(value)  # Recursively look in the right subtree
        return None  # If the value is not found


class BinarySearchTree:
    def __init__(self):
        self.root = None  # Start with an empty tree

    def insert(self, value):
        if not self.root:  # If the tree is empty
            self.root = BinarySearchTreeNode(value)  # The first node becomes the root
        else:
            self.root.insert(value)  # Insert the value into the existing tree

    def lookup(self, value):
        if not self.root:  # If the tree is empty
            return None
        else:
            return self.root.lookup(value)  # Look for the value starting from the root


# Example usage
bst = BinarySearchTree()
bst.insert(10)
bst.insert(5)
bst.insert(15)
bst.insert(3)
bst.insert(7)
bst.insert(12)
bst.insert(18)

# Lookup nodes
node = bst.lookup(7)
if node:
    print(f"Node with value {node.value} found in the tree.")
else:
    print("Value not found in the tree.")
