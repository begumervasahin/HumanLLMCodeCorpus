from find_prime_numbers import sieve_of_eratosthenes
from gdc_euclidian_algorithm import modern_euclid_recursive as gcd

def naive_factorization(n):
    """Factorize the number using the naive approach with the Sieve of Eratosthenes."""
    factors = []
    sieve = sieve_of_eratosthenes(n // 2 + 1)
    primes = [i for i in range(2, len(sieve)) if sieve[i]]

    for prime in primes:
        if n % prime == 0:
            factors.append(prime)

    return factors

def g(x, n):
    """Polynomial function used in the Pollard Rho algorithm."""
    return (x**2 + 1) % n

def pollard_rho(n):
    """Factorize the number using the Pollard Rho algorithm."""
    x = 2
    y = 2
    d = 1

    while d == 1:
        x = g(x, n)
        y = g(g(y, n), n)
        d = gcd(abs(x - y), n)

    if d == n:
        return False
    return d

def main():
    """Main function to choose algorithm and factorize numbers."""
    while True:
        alg = input("Do you want to use my naive approach (n) or the Pollard Rho algorithm (p)? ").lower()

        while True:
            n = input("Enter a number (or type 'exit' to choose algorithm again): ").lower()
            if n == 'exit':
                break
            if n.isnumeric():
                n = int(n)
                if alg == 'n':
                    factors = naive_factorization(n)
                    print(f"Factors of {n} using the naive approach: {factors}")
                elif alg == 'p':
                    factor = pollard_rho(n)
                    if factor:
                        print(f"A non-trivial factor of {n} using the Pollard Rho algorithm: {factor}")
                    else:
                        print(f"Pollard Rho algorithm failed to find a non-trivial factor of {n}")
                else:
                    print("Invalid algorithm choice. Please choose 'n' for naive or 'p' for Pollard Rho.")
            else:
                print("Invalid input. Please enter a numeric value.")

if __name__ == "__main__":
    main()
