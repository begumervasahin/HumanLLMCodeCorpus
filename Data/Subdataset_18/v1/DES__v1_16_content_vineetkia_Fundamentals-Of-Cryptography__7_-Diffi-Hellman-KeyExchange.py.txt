def power(a, b, P):
    """
    Compute (a^b) % P

    Parameters
    ----------
    a : int
        Base number.
    b : int
        Exponent.
    P : int
        Modulus.

    Returns
    -------
    int
        Result of (a^b) % P.
    """
    if b == 1:
        return a % P
    else:
        return pow(a, b, P)

# Given values
G = 9
a = 4
b = 3
P = 29

# Display given values
print("P Value:", P, "\n")
print("G Value:", G, "\n")
print("A Private KEY:", a, "\n")

# Calculate public keys
x = power(G, a, P)
print("The private key B:", b, "\n")
y = power(G, b, P)

# Calculate secret keys
ka = power(y, a, P)
kb = power(x, b, P)

# Display secret keys
print("A Secret Key:", ka, "\n")
print("B Secret Key:", kb, "\n")
