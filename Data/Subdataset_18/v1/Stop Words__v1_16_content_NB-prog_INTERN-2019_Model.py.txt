import torch
import torch.nn as nn
import numpy as np

class SkipGram(nn.Module):
    def __init__(self, vocab_size=20001, embedding_dims=50, neg_samples=20, word_freq=None, padding_idx=0):
        super(SkipGram, self).__init__()
        self.vocab_size = vocab_size
        self.embedding_dims = embedding_dims
        self.neg_samples = neg_samples
        
        self.input_embedding = nn.Embedding(self.vocab_size, self.embedding_dims, padding_idx=padding_idx)
        self.output_embedding = nn.Embedding(self.vocab_size, self.embedding_dims, padding_idx=padding_idx)
        
        # Initialize embeddings
        self.input_embedding.weight = nn.Parameter(
            torch.empty(self.vocab_size, self.embedding_dims, dtype=torch.float).uniform_(-0.5 / self.embedding_dims, 0.5 / self.embedding_dims)
        )
        self.output_embedding.weight = nn.Parameter(
            torch.empty(self.vocab_size, self.embedding_dims, dtype=torch.float).uniform_(-0.5 / self.embedding_dims, 0.5 / self.embedding_dims)
        )
        
        self.neg_sampling_dis = self.gen_neg_sampling_dis(word_freq)
        self.device = 'cuda' if torch.cuda.is_available() else 'cpu'
        
    @staticmethod
    def gen_neg_sampling_dis(word_freq):
        neg_sampling_dis = np.power(word_freq, 0.75)
        neg_sampling_dis = neg_sampling_dis / neg_sampling_dis.sum()
        return neg_sampling_dis
    
    def forward(self, target, contexts):
        batch_size = target.shape[0]
        context_size = contexts.shape[1]
        
        # Generate negative samples
        neg_words = torch.from_numpy(
            np.random.choice(self.vocab_size, size=(batch_size, context_size * self.neg_samples), replace=True, p=self.neg_sampling_dis)
        )
        neg_words = neg_words.to(dtype=torch.long, device=self.device)
        
        target_vectors = self.input_embedding(target).unsqueeze(2)  # (batch_size, embedding_dims, 1)
        contexts_vectors = self.output_embedding(contexts)          # (batch_size, context_size, embedding_dims)
        neg_vectors = self.output_embedding(neg_words)              # (batch_size, context_size * neg_samples, embedding_dims)
        
        # Positive loss
        pos_loss = torch.bmm(contexts_vectors, target_vectors).squeeze().sigmoid().log()
        
        # Negative loss
        neg_loss = torch.bmm(-neg_vectors, target_vectors).sigmoid().log().view(-1, context_size, self.neg_samples).sum(2)
        
        # Combine positive and negative loss
        loss = -(pos_loss + neg_loss).mean()
        
        return loss

# Example usage
if __name__ == "__main__":
    # Example word frequencies (for testing purposes)
    word_freq = np.array([1, 2, 3, 4, 5])
    word_freq = word_freq / word_freq.sum()

    # Initialize the model
    model = SkipGram(vocab_size=6, embedding_dims=10, neg_samples=5, word_freq=word_freq)
    
    # Example data
    target = torch.tensor([1, 2, 3])
    contexts = torch.tensor([[2, 3], [3, 4], [4, 5]])
    
    # Forward pass
    loss = model(target, contexts)
    
    print("Loss:", loss.item())
