import json
import pandas as pd
import matplotlib.pyplot as plt

# Path to the tweets data file
tweets_data_path = '/home/nick/projects/twitter_data.txt'

# Load tweets data
tweets_data = []
with open(tweets_data_path, "r") as tweets_file:
    for line in tweets_file:
        try:
            tweet = json.loads(line)
            tweets_data.append(tweet)
        except json.JSONDecodeError:
            continue

print(f"Number of tweets: {len(tweets_data)}")

# Create a DataFrame from the tweets data
tweets = pd.DataFrame(tweets_data)

# Extract relevant information from the tweets
tweets['text'] = tweets_data.apply(lambda tweet: tweet['text'])
tweets['lang'] = tweets_data.apply(lambda tweet: tweet['lang'])
tweets['country'] = tweets_data.apply(lambda tweet: tweet['place']['country'] if tweet['place'] is not None else None)

# Get the count of tweets by language
tweets_by_lang = tweets['lang'].value_counts()

# Plot the top 5 languages
fig, ax = plt.subplots()
ax.tick_params(axis='x', labelsize=15)
ax.tick_params(axis='y', labelsize=10)
ax.set_xlabel('Languages', fontsize=15)
ax.set_ylabel('Number of tweets', fontsize=15)
ax.set_title('Top 5 languages', fontsize=15, fontweight='bold')
tweets_by_lang[:5].plot(ax=ax, kind='bar', color='red')

# Get the count of tweets by country
tweets_by_country = tweets['country'].value_counts()

# Show the plot
plt.show()

print("We are done")
