import math
import pandas as pd
from selenium import webdriver
from selenium.webdriver.firefox.options import Options
from bs4 import BeautifulSoup as soup

# Read city data
city_data = pd.read_csv("geo.csv", encoding="ISO-8859-1")
city_names = city_data['city']
city_lat = city_data['lat']
city_lng = city_data['lng']

# Set up Firefox options
options = Options()
options.set_preference('permissions.default.stylesheet', 2)
options.set_preference('permissions.default.image', 2)
options.set_preference('dom.ipc.plugins.enabled.libflashplayer.so', 'false')
options.headless = True  # Run in headless mode

driver = webdriver.Firefox(options=options)

# Define feature dictionary and categories
feature = {'Juice Bar': 21, 'Macrobiotic': 12, 'Organic': 13, 'Raw food': 14, 'Salad Bar': 20, 'Take Out': 24, 'American': 5,
           'Asian': 28, 'Australian': 47, 'Brazilian': 46, 'British': 30, 'Caribbean': 31, 'Chinese': 7, 'European': 34,
           'French': 35, 'Fusion': 36, 'German': 37, 'Indian': 8, 'International': 9, 'Italian': 10, 'Japanese': 11, 'Latin': 45,
           'Mediterranean': 18, 'Mexican': 25, 'Middle Eastern': 39, 'Spanish': 40, 'Taiwanese': 41, 'Thai': 15, 'Vietnamese': 42,
           'Western': 16}

Cat = ['Juice Bar', 'Macrobiotic', 'Organic', 'Raw food', 'Salad Bar', 'Take Out', 'American', 'Asian', 'Australian', 'Brazilian',
       'British', 'Caribbean', 'Chinese', 'European', 'French', 'Fusion', 'German', 'Indian', 'International', 'Italian', 'Japanese',
       'Latin', 'Mediterranean', 'Mexican', 'Middle Eastern', 'Spanish', 'Taiwanese', 'Thai', 'Vietnamese', 'Western']

# Loop through each category
for category in Cat:
    data = []
    feature_code = feature[category]
    # Loop through each city
    for city_name, lat, lng in zip(city_names, city_lat, city_lng):
        # Construct the initial URL
        url = f"https://www.happycow.net/searchmap?=&filters=vegan-vegetarian&ft={feature_code}&radius=25&metric=mi&limit=81&order=default&lat={lat}&lng={lng}"
        driver.get(url)
        page_html = driver.page_source
        soup_page = soup(page_html, 'html.parser')
        
        # Find the total number of results
        results_span = soup_page.find("span", {"class": "total-results"})
        if results_span:
            results_count = int(results_span.text.strip())
            loop_count = math.ceil(results_count / 81)
            
            # Loop through each page of results
            for count in range(1, loop_count + 1):
                if count > 1:
                    url = f"https://www.happycow.net/searchmap?=&filters=vegan-vegetarian&ft={feature_code}&radius=25&metric=mi&limit=81&order=default&lat={lat}&lng={lng}&page={count}"
                    driver.get(url)
                    page_html = driver.page_source
                    soup_page = soup(page_html, 'html.parser')
                
                # Extract data from each venue item
                for details in soup_page.findAll(attrs={"class": "js-venues venues__item"}):
                    data.append(details["data-id"] + "\n")
    
    # Save data to CSV file
    filename = f"category_{category}.csv"
    with open(filename, "w", encoding="utf-8") as f:
        f.writelines(data)

# Close the browser
driver.quit()
