a = int(input("Enter the value of a: "))
flag = True

def prime(a):
    global flag
    if a <= 1:
        flag = False
    for i in range(2, int(a ** 0.5) + 1):
        if a % i == 0:
            flag = False
            break

prime(a)

if flag:
    print("The given number is a prime")
else:
    print("The given number is not a prime")
