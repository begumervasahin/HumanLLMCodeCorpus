import socket

def powmod(base, exp, mod):
    """
    Compute (base^exp) % mod using an iterative method.
    
    Args:
        base (int): The base value.
        exp (int): The exponent.
        mod (int): The modulus value.
    
    Returns:
        int: Result of (base^exp) % mod.
    """
    result = 1
    while exp != 0:
        result = (result * base) % mod
        exp -= 1
    return result % mod

def main():
    # Set up the socket connection
    s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    host = input("Please enter the host name of the server: ")
    port = 1024
    s.connect((host, port))
    print("Connected to the server\n")

    # Get the private key from the user
    private_key = int(input("Enter client private key: "))

    # Compute the message to send to the server
    base = 17
    mod = 23
    message = str(powmod(base, private_key, mod))
    print()

    # Receive and print the message from the server
    incoming_message = s.recv(1024).decode()
    print("Server:", incoming_message)

    # Send the computed message to the server
    s.send(message.encode())
    print("Message has been sent\n")

    # Compute and print the shared secret key
    incoming_message = int(incoming_message)
    shared_secret = powmod(incoming_message, private_key, mod)
    print("Secret key shared:", shared_secret)

if __name__ == "__main__":
    main()
