import random

# List of the number of vertices for which we want to generate graphs
vertex_counts = [5, 10, 20, 50, 100, 200, 500, 750, 1000, 1250, 1500, 2000, 3000, 5000]

for num_vertices in vertex_counts:
    # Create a matrix with random edge weights between 1 and 10
    matrix = [[random.randint(1, 10) for _ in range(num_vertices)] for _ in range(num_vertices)]
    
    # Open a file to write the graph data
    with open(f"randograph-{num_vertices}.txt", "w") as f:
        # Write the number of vertices as the first line
        f.write(f"{num_vertices}\n")
        
        # Write edges with random weights
        for i in range(num_vertices):
            # Generate 10 random edges for each vertex
            random_edges = random.sample(range(num_vertices), 10)
            for j in random_edges:
                # Write the edge in the format "start_vertex end_vertex weight"
                f.write(f"{i + 1} {j + 1} {matrix[i][j]}\n")
