import sys
import argparse
import os
import datetime
import subprocess

yes_answers = {'yes', 'y', '', 'si', 's'}
no_answers = {'no', 'n'}

# Argument parser setup
parser = argparse.ArgumentParser(description='Mem trace analysis run folder')
parser.add_argument('session_name', action="store", help='Session name for the TMUX session')
parser.add_argument('run_file', action="store", help='Path to the run file')
args = parser.parse_args()

# Generate session name with timestamp
session_name = f"job_{os.path.relpath(args.session_name)}_{datetime.datetime.now().strftime('%Y-%m-%d_%H-%M-%S')}"
session_logname = f"{session_name}.log"

# Check if run file exists
if not os.path.isfile(args.run_file):
    print(f'Error: "{args.run_file}" not a file')
    exit(-1)

print("Session name:", session_name)
print("Runfile:", args.run_file, "\n")
print("Runfile content:")
print("========================================")

# Read and display run file content
with open(args.run_file, 'r') as runfile:
    lines = runfile.readlines()

for idx, line in enumerate(lines):
    linecmd = line.rstrip('\n')
    if linecmd:
        print(f"{idx}: {line}", end='')

print("\n========================================")
print(f"Detected {idx + 1} jobs to run\n")

# Confirm if the run file content is correct
iscorrect = input("Is runfile content correct? [Y/n] ").lower()
if iscorrect not in yes_answers:
    exit(-1)

# Check if running inside TMUX
if os.environ.get("TMUX") is not None:
    insidetmux = input("Running inside TMUX, proceed? [Y/n] ").lower()
    if insidetmux not in yes_answers:
        exit(-1)
    del os.environ['TMUX']

# Check if the TMUX session already exists
session_exists = True
try:
    subprocess.check_output(["tmux", "has-session", "-t", session_name], stderr=subprocess.STDOUT)
except subprocess.CalledProcessError:
    session_exists = False

if session_exists:
    print("Error: session", session_name, "already exists")
    exit(-1)

# Launch a new TMUX session
try:
    tmux_cmds = f"python3 ~/scripts/run_jobs_launcher.py {session_name} {args.run_file} ; read"
    subprocess.check_output(["tmux", "new-session", "-d", "-s", session_name, tmux_cmds])
except subprocess.CalledProcessError as e:
    print("Error: when launching", e.cmd)
    exit(-1)

print("Session launched")

# Ask if the user wants to switch to the new TMUX session
wantstoattach = input("Switch to tmux job session? [Y/n] ").lower()
if wantstoattach in yes_answers:
    subprocess.check_output(["tmux", "switch-client", "-t", session_name])
