class Node:
    def __init__(self, data=None):
        """
        Initialize a new Node.
        
        Args:
            data: The data to be stored in the node.
        """
        self.data = data
        self.next = None

class LinkedList:
    def __init__(self):
        """
        Initialize an empty LinkedList.
        """
        self.head = Node()

    def append(self, data):
        """
        Append a new node with the given data to the end of the list.
        
        Args:
            data: The data to be stored in the new node.
        """
        new_node = Node(data)
        current = self.head
        while current.next is not None:
            current = current.next
        current.next = new_node

    def length(self):
        """
        Calculate the total number of nodes in the list.
        
        Returns:
            int: The total number of nodes.
        """
        current = self.head
        total = 0
        while current.next is not None:
            total += 1
            current = current.next
        return total

    def display(self):
        """
        Print all elements in the list.
        """
        elements = []
        current_node = self.head
        while current_node.next is not None:
            current_node = current_node.next
            elements.append(current_node.data)
        print(elements)

    def get(self, index):
        """
        Retrieve the data at a specific index in the list.
        
        Args:
            index (int): The index of the node whose data is to be retrieved.
        
        Returns:
            The data at the specified index, or None if the index is out of bounds.
        """
        if index >= self.length():
            print('Index out of bounds')
            return None
        current_idx = 0
        current_node = self.head
        while True:
            current_node = current_node.next
            if current_idx == index:
                return current_node.data
            current_idx += 1

    def erase(self, index):
        """
        Remove the node at a specific index from the list.
        
        Args:
            index (int): The index of the node to be removed.
        """
        if index >= self.length():
            print('Index out of bounds')
            return
        current_idx = 0
        current_node = self.head
        while True:
            last_node = current_node
            current_node = current_node.next
            if current_idx == index:
                last_node.next = current_node.next
                return
            current_idx += 1

# Example usage
my_list = LinkedList()
my_list.append(2)
my_list.append(1)
my_list.erase(1)
my_list.display()
