import requests
from bs4 import BeautifulSoup

def get_yelp_data(location, max_pages=200, page_increment=10):
    base_url = "https://www.yelp.com/search?find_desc=Restaurants&find_loc="
    current_page = 0
    file_path = f"yelp-{location}.txt"

    with open(file_path, "w") as textfile:
        while current_page <= max_pages:
            url = base_url + location + "&start=" + str(current_page)
            yelp_r = requests.get(url)
            print(f"Fetching page {current_page // page_increment + 1} status: {yelp_r.status_code}")
            yelp_soup = BeautifulSoup(yelp_r.text, "html.parser")

            businesses = yelp_soup.findAll("div", {"class": "biz-listing-large"})
            for biz in businesses:
                title = biz.findAll("a", {"class": "biz-name"})[0].text.strip()
                print(title)

                try:
                    address = biz.findAll("address")[0].contents
                    first_line = ""
                    second_line = ""
                    for item in address:
                        if "br" in str(item):
                            second_line += item.getText().strip(" \n\tr")
                        else:
                            first_line = item.strip(" \n\tr")
                    print(first_line)
                    print(second_line)
                except:
                    first_line = ""
                    second_line = ""
                
                try:
                    phone = biz.findAll("span", {"class": "biz-phone"})[0].getText().strip(" \n\t\r")
                except:
                    phone = "N/A"
                print(phone)
                
                page_line = f"{title}\n{first_line}\n{second_line}\n{phone}\n\n"
                textfile.write(page_line)
            
            current_page += page_increment

if __name__ == "__main__":
    location = "New York,NY"
    get_yelp_data(location)
