def selection_sort(input_list):
    """
    Performs selection sort on the given list and returns a new sorted list.
    The input list is not modified.

    Args:
        input_list (list): The list to be sorted.

    Returns:
        list: A new sorted list with the contents of the input list.
    """
    # Create a copy of the input list to avoid modifying the original list
    unsorted_list = list(input_list)
    
    # Initialize an empty list to hold the sorted elements
    sorted_list = []
    
    while unsorted_list:
        # Assume the first element is the smallest
        smallest = unsorted_list[0]
        
        # Find the smallest element in the unsorted list
        for item in unsorted_list:
            if item < smallest:
                smallest = item
        
        # Add the smallest element to the sorted list
        sorted_list.append(smallest)
        
        # Remove the smallest element from the unsorted list
        unsorted_list.remove(smallest)
    
    return sorted_list

if __name__ == "__main__":
    # Example usage
    example_list = [64, 25, 12, 22, 11]
    sorted_list = selection_sort(example_list)
    print("Original list:", example_list)
    print("Sorted list:", sorted_list)
