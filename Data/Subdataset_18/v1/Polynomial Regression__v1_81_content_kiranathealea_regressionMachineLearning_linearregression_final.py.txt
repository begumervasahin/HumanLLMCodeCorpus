import numpy as np
import pandas as pd
import os
import glob
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error, r2_score

daftar_kota = ['Bandung', 'Bantul', 'Bekasi', 'Bogor', 'Ciamis', 'Cianjur', 'Cirebon', 'DIY', 'Garut', 'Gunung Kidul', 'Indramayu', 'Karawang', 'Kulon Progo', 'Kuningan', 'Majalengka', 'Purwakarta', 'Sleman', 'Subang', 'Sukabumi', 'Sumedang', 'Tasikmalaya']

for i in range(21):
    folder = '/home/leathea/Downloads/Machine-Learning/PolynomialRegression/tomato_csv_perdaerah/' + daftar_kota[i] + ".csv"
    data_year_x = pd.read_csv(folder, usecols=['Year'])
    x = np.array(data_year_x)
    x_list = x.ravel()
    data_crop_y = pd.read_csv(folder, usecols=['Crop'])
    y = np.array(data_crop_y)

    model = LinearRegression()
    model.fit(x, y)
    y_pred = model.predict(x)

    rmse_ = mean_squared_error(y, y_pred)
    r2_ = r2_score(y, y_pred)
    print(f'The RMSE of the linear regression model for {daftar_kota[i]} is {rmse_}')
    print(f'The R2 score of the linear regression model for {daftar_kota[i]} is {r2_}')

    years = np.array([2015, 2016, 2017, 2018, 2019, 2020])
    year = years[:, np.newaxis]
    y_next = model.predict(year)
    y_next = np.round(y_next, 4)

    x_final = np.concatenate((x_list, years))
    y_final = np.concatenate((y_pred, y_next))
    y_final_print = np.concatenate((y, y_next))

    wilayah = pd.read_csv(folder, usecols=['Entity'])
    wilayah = np.array(wilayah)
    wilayah = wilayah.ravel()
    wil_tambahan = wilayah[:6]
    wilayah = np.concatenate((wilayah, wil_tambahan))

    kode = pd.read_csv(folder, usecols=['Code'])
    kode = np.array(kode)
    kode = kode.ravel()
    kode_tambahan = kode[:6]
    kode = np.concatenate((kode, kode_tambahan))

    recsv = {
        'Entity': wilayah,
        'Code': kode,
        'Year': x_final,
        'crop(tonnes per hectare)': y_final_print.ravel()
    }

    df = pd.DataFrame(recsv, columns=['Entity', 'Code', 'Year', 'crop(tonnes per hectare)'])
    export_csv = df.to_csv(f'/home/leathea/Downloads/Machine-Learning/PolynomialRegression/hasiltomato/linear/{daftar_kota[i]}_tomato.csv', index=None, header=True)

os.chdir("/home/leathea/Downloads/Machine-Learning/PolynomialRegression/hasiltomato/linear")
extension = 'csv'
all_filenames = [i for i in glob.glob('*.{}'.format(extension))]
combined_csv = pd.concat([pd.read_csv(f) for f in all_filenames])
combined_csv.to_csv("/home/leathea/Downloads/Machine-Learning/PolynomialRegression/predicted_tomato.csv", index=False, encoding='utf-8-sig')
print('all done')
