class Graph:
    def __init__(self, nodes):
        self.V = nodes
        self.edges = []
        self.parent = []
        self.rank = []

    def initialize_sets(self):
        self.parent = [i for i in range(self.V)]
        self.rank = [0] * self.V

    def add_edge(self, u, v, w):
        self.edges.append([u, v, w])

    def find(self, i):
        if self.parent[i] == i:
            return i
        self.parent[i] = self.find(self.parent[i])
        return self.parent[i]

    def union(self, x, y):
        root_x = self.find(x)
        root_y = self.find(y)
        if self.rank[root_x] < self.rank[root_y]:
            self.parent[root_x] = root_y
        elif self.rank[root_x] > self.rank[root_y]:
            self.parent[root_y] = root_x
        else:
            self.parent[root_y] = root_x
            self.rank[root_x] += 1

def kruskal(graph):
    graph.initialize_sets()
    result = []  # To store the resultant MST
    i = 0  # Index variable for sorted edges
    e = 0  # Index variable for the result[]

    graph.edges = sorted(graph.edges, key=lambda item: item[2])

    while e < graph.V - 1:
        u, v, w = graph.edges[i]
        i += 1
        x = graph.find(u)
        y = graph.find(v)

        if x != y:
            e += 1
            result.append([u, v, w])
            graph.union(x, y)

    print("Resulting Minimum Spanning Tree (MST):")
    total_weight = 0
    for u, v, w in result:
        print(f"{u} -- {v} == weight: {w}")
        total_weight += w
    print(f"The total weight of the MST is {total_weight}")

    return result

# Example usage
g = Graph(5)
g.add_edge(0, 1, 9)
g.add_edge(0, 2, 7)
g.add_edge(0, 3, 2)
g.add_edge(1, 2, 2)
g.add_edge(2, 3, 2)
g.add_edge(1, 3, 2)
g.add_edge(1, 4, 3)
g.add_edge(3, 4, 3)

kruskal(g)
