def encrypt(plain_text, key):
    """
    Encrypts the given plain text using a Vigenère cipher with the specified key.

    Parameters
    ----------
    plain_text : str
        The text to be encrypted.
    key : str
        The encryption key.

    Returns
    -------
    str
        The encrypted text.
    """
    result = ""

    for i in range(len(plain_text)):
        # Get the current position in the key
        j = i % len(key)

        if plain_text[i].isupper():
            # Encrypt uppercase characters
            new_char = chr((ord(plain_text[i]) + ord(key[j]) - 2 * ord('A')) % 26 + ord('A'))
        elif plain_text[i].islower():
            # Encrypt lowercase characters
            new_char = chr((ord(plain_text[i]) + ord(key[j]) - ord('a') - ord('A')) % 26 + ord('a'))
        else:
            # Leave non-alphabetic characters unchanged
            new_char = plain_text[i]

        result += new_char

    return result

if __name__ == "__main__":
    # Example usage
    plain_text = "VINeeT"
    key = "WOW"
    encrypted_text = encrypt(plain_text, key)
    print("ENCRYPTED TEXT:", encrypted_text)
