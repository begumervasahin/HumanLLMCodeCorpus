import os

def gcf(n1, n2):
    """
    Calculate the greatest common factor (GCF) of two numbers using the Euclidean algorithm.
    
    Parameters:
    n1 (int): The first number.
    n2 (int): The second number.
    
    Returns:
    int: The greatest common factor of n1 and n2.
    """
    if n2 > n1:
        n1, n2 = n2, n1

    while n2 != 0:
        n1, n2 = n2, n1 % n2
    
    return n1

def gcd(p, q):
    """
    Calculate the greatest common divisor (GCD) of two numbers using the Euclidean algorithm.
    
    Parameters:
    p (int): The first number.
    q (int): The second number.
    
    Returns:
    int: The greatest common divisor of p and q.
    """
    while q != 0:
        p, q = q, p % q
    return p

def lcm(n1, n2):
    """
    Calculate the least common multiple (LCM) of two numbers.
    
    Parameters:
    n1 (int): The first number.
    n2 (int): The second number.
    
    Returns:
    int: The least common multiple of n1 and n2.
    """
    return (n1 * n2) // gcf(n1, n2)

def euclidean_extended_gcd(n1, n2):
    """
    Calculate the extended Euclidean algorithm, which finds integers u and v such that:
    n1 * u + n2 * v = gcd(n1, n2)
    
    Parameters:
    n1 (int): The first number.
    n2 (int): The second number.
    
    Returns:
    tuple: A tuple containing the GCD of n1 and n2, and the coefficients u and v.
    """
    u, v, s, t = 1, 0, 0, 1
    if n2 < n1:
        n1, n2 = n2, n1

    while n2 != 0:
        q = n1 // n2
        n1, n2 = n2, n1 - q * n2
        u, s = s, u - q * s
        v, t = t, v - q * t

    return n1, u, v
