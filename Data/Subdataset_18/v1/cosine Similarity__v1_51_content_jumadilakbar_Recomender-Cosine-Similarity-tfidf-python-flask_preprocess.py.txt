import string
import pandas as pd
from Sastrawi.Stemmer.StemmerFactory import StemmerFactory
from Sastrawi.StopWordRemover.StopWordRemoverFactory import StopWordRemoverFactory

class Preprocess:
    def __init__(self):
        # Initialize the stemmer and stop word remover
        self.stemmer = StemmerFactory().create_stemmer()
        self.remover = StopWordRemoverFactory().create_stop_word_remover()

    def preprocess(self, text):
        """
        Preprocess the given text by stemming, removing stopwords, converting to lowercase,
        and removing punctuation.
        
        Args:
            text (str): The text to preprocess.

        Returns:
            list: The preprocessed text as a list of words.
        """
        # Stem the text
        text_stem = self.stemmer.stem(text)
        # Remove stopwords
        text_clean = self.remover.remove(text_stem)
        # Convert to lowercase
        lowercase = text_clean.lower()
        # Remove punctuation and split into words
        preprocessed_text = lowercase.translate(str.maketrans('', '', string.punctuation)).split()
        return preprocessed_text

if __name__ == '__main__':
    # Create an instance of the Preprocess class
    preprocess = Preprocess()
    
    # Load the dataset
    data = pd.read_csv('Coba.csv', sep=',')
    
    # List to hold preprocessed documents
    preprocessed_documents = []
    
    # Get the 'TOPIK' column from the dataset
    titles = data.TOPIK.tolist()
    
    # Preprocess each title
    for title in titles:
        preprocessed = preprocess.preprocess(title)
        preprocessed_documents.append(' '.join(preprocessed))
    
    # Add the preprocessed documents to the dataframe
    data['preprocessed'] = preprocessed_documents
    
    # Save the updated dataframe to a new CSV file
    data.to_csv('Coba_preprocessed.csv', index=False)
