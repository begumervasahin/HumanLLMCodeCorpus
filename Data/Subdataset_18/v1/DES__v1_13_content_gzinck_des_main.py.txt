from cli.menus.main_menu import main_menu
import global_settings
from tempfile import TemporaryDirectory

"""
This main file runs the DES application on the command line.
"""

def run_des_application():
    """
    Run the DES application, initializing settings and handling screen transitions.
    """
    # Initialize global settings
    global_settings.initialize()

    # Initialize screen and automata stacks
    screens = [main_menu]
    automata = []

    # Create a temporary directory to be used within the application
    with TemporaryDirectory() as temp_dir:
        # Continue running while there are screens to display
        while screens:
            # Get the next screen to display
            next_screen = screens[-1]

            # Display the next screen, passing the screens stack, automata stack, and temp directory
            next_screen(screens, automata, temp_dir)

if __name__ == "__main__":
    run_des_application()
