"""
Dijkstra Algorithm.
Find the cheapest path. If the cost is the same for every move, it works like BFS.
"""

from queue import PriorityQueue
from pprint import pprint
import grid_helper as gh

def find_path_dijkstra(grid, start, end):
    """
    Find the cheapest path from start to end using Dijkstra's algorithm.

    Parameters:
        grid (list of lists): The grid with weighted paths.
        start (tuple): The starting position in the grid.
        end (tuple): The ending position in the grid.

    Returns:
        dict: A dictionary representing the path from start to end.
    """
    priority_queue = PriorityQueue()
    priority_queue.put((0, start))
    came_from = {start: None}
    costs = {start: 0}
    
    while not priority_queue.empty():
        current_position = priority_queue.get()[1]
        
        if current_position == end:
            break
        
        neighbors = gh.get_neighbors(grid, current_position[0], current_position[1])
        
        for neighbor in neighbors:
            new_cost = costs[current_position] + gh.get_cost(grid, neighbor)
            if neighbor not in costs or new_cost < costs[neighbor]:
                costs[neighbor] = new_cost
                priority_queue.put((new_cost, neighbor))
                came_from[neighbor] = current_position
    
    return came_from

def display_path(grid, start, end, came_from):
    """
    Find and display the path on the grid from start to end.
    
    Parameters:
        grid (list of lists): The grid with weighted paths.
        start (tuple): The starting position in the grid.
        end (tuple): The ending position in the grid.
        came_from (dict): The directions to reach each cell from the start position.
    """
    path = gh.find_path(start, end, came_from)

    pprint(grid_with_path)
    print(f"Steps: {len(path)}")

def initialize():
    """
    Initialize the grid, find the path using Dijkstra's algorithm, and display the result.
    """
    initial_grid = gh.generate_grid_weighted()
    start, end = (4, 0), (4, 6)
    
    # Find path using Dijkstra's algorithm
    came_from = find_path_dijkstra(initial_grid, start, end)
    
    # Display the path on the grid
    display_path(initial_grid, start, end, came_from)
