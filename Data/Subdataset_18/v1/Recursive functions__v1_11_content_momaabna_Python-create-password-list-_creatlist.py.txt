import sys

hel = """
Usage : creatlist.py Min Max filename [p]
Min :length of minimum password
Max :length of maximum password
p : y for print on screen
Example : creatlist.py 5 8 passwords.txt y
"""

# Check if the number of arguments is correct
if len(sys.argv) <= 4:
    print(hel)
    exit()

# Parse command line arguments
min_length = int(sys.argv[1])
max_length = int(sys.argv[2])
filename = str(sys.argv[3])
print_option = str(sys.argv[4]) if len(sys.argv) > 4 else 'n'

# Read the letters from letters.txt
with open('letters.txt', 'r') as ff:
    letters = ff.read().strip()

# Open the output file for writing
with open(filename, 'w') as f:
    def create(l, pas):
        if len(pas) == l:
            for lett in letters:
                npas = pas + str(lett) + '\n'
                if print_option.lower() == 'y':
                    print(npas, end='')
                f.write(npas)
        elif len(pas) < l:
            for lett in letters:
                npas = pas + str(lett)
                create(l, npas)

    hello = """
Welcome in Passlist Creator From rw-sec.com
Author : Mohammed Mahmood
Start creating list :
"""
    print(hello)

    # Generate passwords
    for i in range(min_length, max_length + 1):
        create(i, '')

print('Password List Created Successfully ..')
