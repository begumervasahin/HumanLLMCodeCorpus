import argparse
import os
import re
import sys
import time
import math
import cProfile

# Compile regular expressions for graph and edge parsing
graphRE = re.compile(r"(\d+)\s(\d+)")
edgeRE = re.compile(r"(\d+)\s(\d+)\s(-?\d+)")

# Initialize global variables
vertices = []
edges = []
pr = cProfile.Profile()

def BellmanFord(G):
    """Bellman-Ford algorithm for finding shortest paths."""
    vertices, edge_weights = G
    pathPairs = []
    edges = []
    l = len(vertices)

    for i in range(l):
        for j in range(l):
            if not math.isinf(float(edge_weights[i][j])):
                e = [(i, j), float(edge_weights[i][j])]
                edges.append(e)

    for k in range(len(vertices)):
        dist = [float("inf")] * l
        dist[k] = 0

        for _ in range(l - 1):
            for u, v in edges:
                if dist[v[1]] > dist[v[0]] + v[1]:
                    dist[v[1]] = dist[v[0]] + v[1]

        for u, v in edges:
            if dist[v[1]] > dist[v[0]] + v[1]:
                print("Negative cycle detected")
                return []

        for x in range(l):
            edge = ((k, x), dist[x])
            pathPairs.append(edge)
    
    print(pathPairs)
    return pathPairs

def FloydWarshall(G):
    """Floyd-Warshall algorithm for finding shortest paths."""
    vertices, edge_weights = G
    pathPairs = []
    l = len(vertices)
    d = [row[:] for row in edge_weights]

    for i in range(l):
        d[i][i] = 0

    for k in range(l):
        for i in range(l):
            for j in range(l):
                if d[i][j] > d[i][k] + d[k][j]:
                    d[i][j] = d[i][k] + d[k][j]

    for i in range(l):
        if d[i][i] < 0:
            print("Negative cycle detected")
            return []

    for i in range(l):
        for j in range(l):
            edge = ((i, j), d[i][j])
            pathPairs.append(edge)

    print(pathPairs)
    return pathPairs

def readFile(filename):
    """Reads the input file and parses the graph."""
    global vertices
    global edges
    with open(filename, 'r') as inFile:
        line1 = inFile.readline()
        graphMatch = graphRE.match(line1)
        if not graphMatch:
            print(f"{line1} not properly formatted")
            sys.exit(1)
        
        vertices = list(range(int(graphMatch.group(1))))
        edges = [[float("inf")] * len(vertices) for _ in range(len(vertices))]

        for line in inFile.readlines():
            edgeMatch = edgeRE.match(line.strip())
            if edgeMatch:
                source, sink, weight = map(int, edgeMatch.groups())
                if source > len(vertices) or sink > len(vertices):
                    print(f"Attempting to insert an edge between {source} and {sink} in a graph with {len(vertices)} vertices")
                    sys.exit(1)
                edges[source - 1][sink - 1] = weight
    
    return vertices, edges

def matrixEquality(a, b):
    """Checks equality of two matrices."""
    if len(a) != len(b) or not a or not b:
        return False
    for i, row in enumerate(a):
        for j, value in enumerate(row):
            if value != b[i][j]:
                return False
    return True

def main(filename, algorithm):
    """Main function to execute the algorithms based on user input."""
    G = readFile(filename)
    start = time.time()

    if algorithm == 'a':
        print('Running both Bellman-Ford and Floyd-Warshall algorithms')
        pathPairsBellman = BellmanFord(G)
        pathPairsFloyd = FloydWarshall(G)
        print("--- %s seconds ---" % (time.time() - start))
    elif algorithm == 'b':
        print('Running Bellman-Ford algorithm')
        pathPairsBellman = BellmanFord(G)
        print("--- %s seconds ---" % (time.time() - start))
    elif algorithm == 'f':
        print('Running Floyd-Warshall algorithm')
        pathPairsFloyd = FloydWarshall(G)
        print("--- %s seconds ---" % (time.time() - start))
    else:
        print("Unknown algorithm selected. Use 'a', 'b', or 'f'.")

if __name__ == '__main__':
    parser = argparse.ArgumentParser(description='Calculate the shortest path between all pairs of vertices in a graph')
    parser.add_argument('--algorithm', default='a', help='Algorithm: Select the algorithm to run, default is all. (a)ll, (b)ellman-ford only, or (f)loyd-warshall only')
    parser.add_argument('-v', '--verbose', action='store_true')
    parser.add_argument('--profile', action='store_true')
    parser.add_argument('filename', metavar='<filename>', help='Input file containing graph')
    
    args = parser.parse_args()
    if args.profile:
        pr.enable()

    main(args.filename, args.algorithm)
    
    if args.profile:
        pr.print_stats(sort='time')
