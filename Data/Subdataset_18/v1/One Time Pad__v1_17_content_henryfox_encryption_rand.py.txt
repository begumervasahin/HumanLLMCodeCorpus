import random

def rand(dig):
    """
    Generates a random number string of a specified length.
    
    Parameters:
    dig (int): The length of the random number string.
    
    Returns:
    str: The generated random number string.
    """
    nums = []
    for x in range(dig):
        nums.append(str(random.randint(0, 9)))
    num = "".join(nums)
    return num

def main():
    # Example usage: generate a random number string of length 10
    length = 10
    random_number = rand(length)
    print(f"Random number of length {length}: {random_number}")

if __name__ == "__main__":
    main()
