import torch
import matplotlib.pyplot as plt

class Stats:
    """
    This class is used for computing statistics
    such as the number of non-zeros, objective gap,
    and effective passes. Only the compute method
    is the public method.
    """
    def __init__(self):
        self.num_non_zeros = []
        self.objective_gap = []
        self.effective_passes = []
        self.current_iteration = 0

    def _compute_num_non_zeros(self, w):
        non_zero_count = int((torch.abs(w) > 1e-5).int().sum().data.cpu().numpy())
        self.num_non_zeros.append(non_zero_count)

    def _compute_objective_gap(self, loss):
        self.objective_gap.append(loss)

    def _compute_effective_passes(self):
        self.effective_passes.append(self.current_iteration)

    def compute(self, w, loss):
        """
        Computes the statistics for the given weights and loss.
        
        Parameters:
        w: tensor of weights [1 x d]
        loss: float
        """
        self._compute_num_non_zeros(w)
        self._compute_objective_gap(loss)
        self._compute_effective_passes()
        self.current_iteration += 1

    def plot(self, title=""):
        """
        Plots the number of non-zeros and objective gap against effective passes.
        
        Parameters:
        title: str, optional title for the plots
        """
        # Plotting number of non-zeros
        plt.plot(self.num_non_zeros)
        plt.xlabel("Effective Passes")
        plt.ylabel("Number of Non-Zeros")
        plt.title(f"{title} Number of Non-Zeros")
        plt.xticks(range(len(self.num_non_zeros)))
        plt.savefig(f"{title}_NonZeros.png")
        plt.clf()

        # Plotting objective gap
        plt.plot(self.objective_gap)
        plt.xlabel("Effective Passes")
        plt.ylabel("Objective Loss")
        plt.title(f"{title} Objective Loss")
        plt.xticks(range(len(self.num_non_zeros)))
        plt.savefig(f"{title}_Loss.png")
        plt.clf()

# Example usage
if __name__ == "__main__":
    stats = Stats()
    
    # Example data (normally you'd have real weight tensors and loss values)
    for i in range(10):
        weights = torch.randn(1, 100)
        loss = torch.abs(torch.randn(1)).item()
        stats.compute(weights, loss)
    
    stats.plot("Example")
