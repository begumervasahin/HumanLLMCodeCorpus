from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives.asymmetric import dh
import Seed_Generator

def generate_access_key():
    """
    Generate an access key using the Diffie-Hellman key exchange protocol.
    
    Returns:
        bytes or bool: The shared key if successful, otherwise False.
    """
    # Generate Diffie-Hellman parameters
    parameters = dh.generate_parameters(generator=5, key_size=1024, backend=default_backend())
    
    # Generate private keys and public keys for both parties
    a_private_key = parameters.generate_private_key()
    a_peer_public_key = a_private_key.public_key()
    
    b_private_key = parameters.generate_private_key()
    b_peer_public_key = b_private_key.public_key()
    
    # Exchange keys to compute shared keys
    a_shared_key = a_private_key.exchange(b_peer_public_key)
    b_shared_key = b_private_key.exchange(a_peer_public_key)
    
    # Check if shared keys match
    if a_shared_key == b_shared_key:
        # Generate seed using the Seed_Generator module
        Seed_Generator.generate_seed(a_private_key, b_private_key, a_peer_public_key, b_peer_public_key)
        return a_shared_key
    else:
        return False

# Execute the function to generate the access key
shared_key = generate_access_key()
if shared_key:
    print("Access key generated successfully.")
else:
    print("Failed to generate access key.")
