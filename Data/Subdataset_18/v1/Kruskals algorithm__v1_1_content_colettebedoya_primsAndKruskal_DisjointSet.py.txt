class DisjointSet:
    def __init__(self, size):
        """
        Initializes a disjoint set of the specified size.
        Each element is initially in its own set.
        """
        if size < 0:
            raise ValueError("size must be >= 0")
        self.size = size
        self.set = [-1] * (size + 1)  # Using -1 to indicate that the element is a root

    def union(self, root1, root2):
        """
        Performs a union of the sets containing root1 and root2.
        root2's set will be merged into root1's set.
        """
        if root1 < 0 or root1 > self.size or root2 < 0 or root2 > self.size:
            raise ValueError("Illegal value")
        root1_set = self.find(root1)
        root2_set = self.find(root2)
        if root1_set != root2_set:  # Merge only if they are in different sets
            self.set[root2_set] = root1_set

    def find(self, root):
        """
        Performs a find operation to locate the root of the set containing the specified element.
        Implements path compression to flatten the structure, speeding up future operations.
        """
        if root < 0 or root > self.size:
            raise ValueError("Illegal value")
        if self.set[root] < 0:  # This means the element is a root
            return root
        else:
            # Path compression: make the found root the parent of the current element
            self.set[root] = self.find(self.set[root])
            return self.set[root]

if __name__ == "__main__":
    # Example usage:
    dj = DisjointSet(10)

    # Union some sets
    dj.union(1, 2)
    dj.union(3, 4)
    dj.union(2, 3)

    # Find the roots of the sets
    print("Root of 1:", dj.find(1))  # Should print the root of the set containing 1
    print("Root of 2:", dj.find(2))  # Should print the same root as for 1
    print("Root of 3:", dj.find(3))  # Should print the same root as for 1 and 2
    print("Root of 4:", dj.find(4))  # Should print the same root as for 1, 2, and 3

    # Trying to find a root outside the range should raise an error
    try:
        print(dj.find(11))
    except ValueError as e:
        print(e)
