import random
import sys

def encrypt(byte):
    """
    Encrypts a single byte using a random key and XOR operation.

    Args:
    byte (bytes): The byte to encrypt.

    Returns:
    list: A list containing the random key and the encrypted byte.
    """
    byte = ord(byte)
    keys = []
    keys.append(random.randint(0, 255))
    keys.append(byte ^ keys[0])
    return keys

def decrypt(byte0, byte1):
    """
    Decrypts a pair of bytes using XOR operation.

    Args:
    byte0 (bytes): The first byte (random key).
    byte1 (bytes): The second byte (encrypted byte).

    Returns:
    int: The decrypted byte.
    """
    keys = []
    keys.append(ord(byte0))
    keys.append(ord(byte1))
    aux = keys[0] ^ keys[1]
    return aux

def encryptFile(input_file, output1, output2):
    """
    Encrypts the contents of the input file and writes the result to two output files.

    Args:
    input_file (str): The name of the input file to encrypt.
    output1 (str): The name of the first output file to store the random keys.
    output2 (str): The name of the second output file to store the encrypted bytes.
    """
    with open(input_file, "rb") as f, open(output1, 'wb') as key1File, open(output2, 'wb') as key2File:
        byte = f.read(1)
        while byte:
            aux = encrypt(byte)
            key1File.write(aux[0].to_bytes(1, byteorder="big", signed=False))
            key2File.write(aux[1].to_bytes(1, byteorder="big", signed=False))
            byte = f.read(1)

def decryptFile(input1, input2, output_file):
    """
    Decrypts the contents of two input files and writes the result to an output file.

    Args:
    input1 (str): The name of the first input file containing the random keys.
    input2 (str): The name of the second input file containing the encrypted bytes.
    output_file (str): The name of the output file to store the decrypted bytes.
    """
    with open(input1, "rb") as f1, open(input2, "rb") as f2, open(output_file, 'wb') as output:
        byte1 = f1.read(1)
        byte2 = f2.read(1)
        while byte1 and byte2:
            output.write(decrypt(byte1, byte2).to_bytes(1, byteorder="big", signed=False))
            byte1 = f1.read(1)
            byte2 = f2.read(1)

if __name__ == "__main__":
    print("Starting script...")
    if len(sys.argv) == 2:
        encryptFile(sys.argv[1], sys.argv[1] + ".0", sys.argv[1] + ".1")
    elif len(sys.argv) == 3:
        decryptFile(sys.argv[1], sys.argv[2], sys.argv[1][:-2])
    else:
        print("Argument parsing error.")
    print("Script finished.")
