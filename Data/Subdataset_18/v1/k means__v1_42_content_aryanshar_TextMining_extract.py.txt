import re
from textblob import TextBlob

def clean_tweet(tweet):
    '''
    Utility function to clean tweet text by removing links, special characters
    using simple regex statements.
    '''
    return re.sub(r'http\S+|www\S+|https\S+', '', tweet, flags=re.MULTILINE)

def get_tweet_sentiment(tweet):
    '''
    Utility function to classify sentiment of passed tweet
    using textblob's sentiment method
    '''
    analysis = TextBlob(clean_tweet(tweet))
    if analysis.sentiment.polarity > 0:
        return 'positive'
    elif analysis.sentiment.polarity == 0:
        return 'neutral'
    else:
        return 'negative'

# Lists to store tweets with and without URLs
tweets_url_yes = []
tweets_url_no = []

# Reading tweets from the file
with open("tweets_123.txt", 'r') as data:
    tweets = [line.strip() for line in data]

# Classifying tweets and storing them in respective lists
for line in tweets:
    tweet_with_url = {}
    tweet_without_url = {}
    urls = re.search('http[s]?://(?:[a-zA-Z]|[0-9]|[$-_@.&+]|[!*\(\),]|(?:%[0-9a-fA-F][0-9a-fA-F]))+', line)
    if urls is not None:
        tweet_with_url['text'] = line
        tweet_with_url['sentiment'] = get_tweet_sentiment(line)
        tweets_url_yes.append(tweet_with_url)
    else:
        tweet_without_url['text'] = line
        tweet_without_url['sentiment'] = get_tweet_sentiment(line)
        tweets_url_no.append(tweet_without_url)

# Calculating and printing sentiment percentages
if len(tweets_url_yes) > 0:
    ptweets = [tweet for tweet in tweets_url_yes if tweet['sentiment'] == 'positive']
    ntweets = [tweet for tweet in tweets_url_yes if tweet['sentiment'] == 'negative']
    neuttweets = [tweet for tweet in tweets_url_yes if tweet['sentiment'] == 'neutral']

    print("Positive tweets percentage: {} %".format(100*len(ptweets)/len(tweets_url_yes)))
    print("Negative tweets percentage: {} %".format(100*len(ntweets)/len(tweets_url_yes)))
    print("Neutral tweets percentage: {} %".format(100*len(neuttweets)/len(tweets_url_yes)))

    print("\n\nPositive tweets:")
    for tweet in ptweets:
        print("%s" % tweet['text'])

    print("\n\nNegative tweets:")
    for tweet in ntweets:
        print("%s" % tweet['text'])

    print("\n\nNeutral tweets:")
    for tweet in neuttweets:
        print("%s" % tweet['text'])
else:
    print("No tweets with URLs found.")

if len(tweets_url_no) > 0:
    ptweets_no = [tweet for tweet in tweets_url_no if tweet['sentiment'] == 'positive']
    ntweets_no = [tweet for tweet in tweets_url_no if tweet['sentiment'] == 'negative']
    neuttweets_no = [tweet for tweet in tweets_url_no if tweet['sentiment'] == 'neutral']

    print("\n\nPositive tweets without URLs:")
    for tweet in ptweets_no:
        print("%s" % tweet['text'])

    print("\n\nNegative tweets without URLs:")
    for tweet in ntweets_no:
        print("%s" % tweet['text'])

    print("\n\nNeutral tweets without URLs:")
    for tweet in neuttweets_no:
        print("%s" % tweet['text'])
else:
    print("No tweets without URLs found.")
