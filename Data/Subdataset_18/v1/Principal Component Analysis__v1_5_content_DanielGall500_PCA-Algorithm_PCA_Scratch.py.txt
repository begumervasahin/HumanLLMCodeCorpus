import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# Step 1: Generate random data
mean = [5, 5]
cov = [[1, 0], [0, 100]]
data = pd.DataFrame(np.random.multivariate_normal(mean, cov, 100))

# Step 2: Scatter plot of the original data
plt.scatter(data[0], data[1], color='b', label='Original Data')

# Step 3: Function to subtract the mean from each column
def subtract_mean(matrix):
    for col in matrix.columns:
        mean = np.mean(matrix[col])
        sub_mean = lambda x: x - mean
        matrix[col] = matrix[col].apply(sub_mean)
    return matrix

# Step 4: Function to calculate the covariance matrix
def covariance(matrix):
    cov_list = np.array([])
    cols = matrix.columns
    c_header1, c_header2 = 'x', 'y'
    combinations = [[col, f] for f in cols for col in cols]
    for c1, c2 in combinations:
        comb_data = matrix[[c1, c2]]
        comb_data.columns = [c_header1, c_header2]
        multiplied_columns = [np.multiply(i, j) for i, j in comb_data.itertuples(index=False)]
        n = comb_data.shape[0]
        covariance = np.sum(multiplied_columns) / (n - 1)
        cov_list = np.append(cov_list, covariance)
    cov_matrix = np.matrix(cov_list).reshape([len(cols), len(cols)])
    return cov_matrix

# Step 5: Function to get the principal components
def get_principle_comp(eig_vals, eig_vecs, dimensions):
    eigval_max = np.argsort(-eig_vals)[:dimensions]
    eigvec_max = eig_vecs[:, eigval_max]
    return eigvec_max

# Step 6: Adjust the data by subtracting the mean
mean_adjusted = subtract_mean(data.copy())

# Step 7: Calculate the covariance matrix
covariance_data = covariance(mean_adjusted)

# Step 8: Calculate the eigenvalues and eigenvectors
eig_vals, eig_vecs = np.linalg.eig(covariance_data)

# Step 9: Get the principal components
principle_comp = get_principle_comp(eig_vals, eig_vecs, 2)

# Step 10: Transform the data using the principal components
feature_vector = np.dot(principle_comp.T, mean_adjusted.T)

# Step 11: Scatter plot of the transformed data
plt.scatter(feature_vector[0], feature_vector[1], color='r', marker='x', label='PCA Transformed Data')

# Step 12: Show the plots
plt.xlabel('X-axis')
plt.ylabel('Y-axis')
plt.title('PCA Transformation')
plt.legend()
plt.show()

# Print the transformed feature vector
print(feature_vector.T)
