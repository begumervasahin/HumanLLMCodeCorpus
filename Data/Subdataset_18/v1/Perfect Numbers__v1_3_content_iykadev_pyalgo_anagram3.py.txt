def anagram(s1, s2):
    # Initialize two lists to count the frequency of each letter in both strings
    c1 = [0] * 26
    c2 = [0] * 26
    
    # Count the frequency of each letter in the first string
    for char in s1:
        pos = ord(char) - ord('a')
        c1[pos] += 1
    
    # Count the frequency of each letter in the second string
    for char in s2:
        pos = ord(char) - ord('a')
        c2[pos] += 1
    
    # Check if the frequency counts are the same for both strings
    for i in range(26):
        if c1[i] != c2[i]:
            return False
    
    return True

# Test the function
print(anagram('pide', 'epid'))  # This should return True
print(anagram('hello', 'bello'))  # This should return False
