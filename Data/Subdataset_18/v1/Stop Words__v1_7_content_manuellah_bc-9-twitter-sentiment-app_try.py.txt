def get_all_tweets(screen_name):
    """
    Get all tweets from a specific Twitter user.

    Args:
        screen_name (str): Twitter handle of the user.

    Returns:
        dict: A dictionary where keys are tweet IDs and values are tuples of tweet text and creation date.
    """
    my_dict = {}
    alltweets = []

    new_tweets = api.user_timeline(screen_name=screen_name, count=200)
    alltweets.extend(new_tweets)

    if not alltweets:
        my_dict[0] = ("No tweet so far", "00/00/00")
        return my_dict

    oldest = alltweets[-1].id - 1

    while new_tweets:
        new_tweets = api.user_timeline(screen_name=screen_name, count=200, max_id=oldest)
        alltweets.extend(new_tweets)
        oldest = alltweets[-1].id - 1

    for tweet in alltweets:
        my_dict[tweet.id] = (tweet.text, tweet.created_at)

    return my_dict


def display_tweets(screen_name):
    """
    Display all tweets from a specific Twitter user.

    Args:
        screen_name (str): Twitter handle of the user.
    """
    tweet_dict = get_all_tweets(screen_name)
    header = '{:<30} {:<30} {}'.format("Tweeter ID", "Date Posted", "The Tweet")
    print(header)
    for key in tweet_dict:
        print('{:<30} {:<30} {}'.format(str(key), str(tweet_dict[key][1]), tweet_dict[key][0]))


def frequency_analyzer(screen_name):
    """
    Analyze word frequency in tweets from a specific Twitter user.

    Args:
        screen_name (str): Twitter handle of the user.
    """
    tweet_dict = get_all_tweets(screen_name)
    freq_dict = {}
    stop_words = set()  # Add any stop words you want to exclude here

    for key in tweet_dict:
        tweet_list = str(tweet_dict[key][0]).split()
        for word in tweet_list:
            if word in stop_words:
                continue
            if word not in freq_dict:
                freq_dict[word] = 1
            else:
                freq_dict[word] += 1

    for word, count in freq_dict.items():
        print(f"{word}: {count}")
