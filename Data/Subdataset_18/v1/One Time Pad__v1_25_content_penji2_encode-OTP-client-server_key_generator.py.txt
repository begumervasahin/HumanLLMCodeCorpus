import random
import string

def main():
    # Open the files for writing
    with open("key.dat", mode="w") as f, open("key1.dat", mode="w") as f1:
        for _ in range(1000000):
            # Choose a random ASCII letter
            b = random.choice(string.ascii_letters)
            # Write the letter to key.dat
            f.write(b)
            # Convert the letter to its ASCII value, then to an 8-bit binary string
            dec = ord(b)
            b_bin = '{0:08b}'.format(dec)
            # Write the binary string to key1.dat
            f1.write(b_bin)

if __name__ == "__main__":
    main()
