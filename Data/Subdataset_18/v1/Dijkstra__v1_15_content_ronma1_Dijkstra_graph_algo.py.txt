import networkx as nx
from Graph import Graph

class GraphAlgo:
    """Graph algorithms using NetworkX."""

    def __init__(self, filename):
        """Initialize the graph from a file."""
        g = Graph(filename)
        self.graph = g.graph

    def calc_dist(self, v1, v2):
        """Calculate the shortest distance between two nodes using Dijkstra's algorithm."""
        return nx.dijkstra_path_length(self.graph, v1, v2)

    def calc_path(self, v1, v2):
        """Calculate the shortest path between two nodes using Dijkstra's algorithm."""
        return nx.dijkstra_path(self.graph, v1, v2)

    def calc_path_with_blacklist(self, v1, v2, blacklist):
        """Calculate the shortest path between two nodes with certain nodes blacklisted."""
        temp_graph = self.graph.copy()
        for node in blacklist:
            temp_graph.remove_node(node)
        return nx.dijkstra_path(temp_graph, v1, v2)

    def diameter(self):
        """Calculate the diameter of the graph."""
        return nx.diameter(self.graph)

    def radius(self):
        """Calculate the radius of the graph."""
        return nx.radius(self.graph)

    def is_triangle_inequality(self):
        """Check if the graph satisfies the triangle inequality."""
        nodes = list(self.graph.nodes)
        for i in range(len(nodes)):
            for j in range(len(nodes)):
                if nx.dijkstra_path_length(self.graph, nodes[i], nodes[j]) > 1:
                    return False
        return True
