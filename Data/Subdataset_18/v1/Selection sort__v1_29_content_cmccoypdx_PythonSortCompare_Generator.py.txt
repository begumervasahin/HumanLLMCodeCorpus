import random

def GenArray(size):
    """Generate an array of given size with random integers ranging from -size to size."""
    a = []
    for _ in range(size):
        a.append(random.randint(-size, size))
    return a

def GenPayload():
    """Generate a list of arrays with predefined sizes."""
    p = []
    sizes = [10, 100, 1000, 10000, 100000, 1000000, 10000000]
    for size in sizes:
        p.append(GenArray(size))
    return p

# Generate the payload
payload = GenPayload()

# Optionally, print the payload or its summary
# Printing the entire payload for large sizes can be overwhelming,
# so here is a summary of the sizes of generated arrays.
for i, arr in enumerate(payload):
    print(f"Array {i + 1} size: {len(arr)}")
