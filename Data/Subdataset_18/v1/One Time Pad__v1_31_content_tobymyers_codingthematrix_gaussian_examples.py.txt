from mat import Mat
from GF2 import one

def main():
    # Define the dimensions and elements of matrix A
    Ad = (set([1, 2, 3, 4]), set(['A', 'B', 'C', 'D']))
    Af = {
        (2, 'A'): one, (2, 'C'): one, (2, 'D'): one,
        (1, 'C'): one, (1, 'D'): one,
        (3, 'A'): one, (3, 'D'): one,
        (4, 'A'): one, (4, 'B'): one, (4, 'C'): one, (4, 'D'): one
    }
    A = Mat(Ad, Af)

    # Define the dimensions and elements of matrix B
    Bd = (set([1, 2, 3, 4]), set(['A', 'B', 'C', 'D']))
    Bf = {
        (1, 'A'): one, (1, 'B'): one,
        (2, 'A'): one, (2, 'C'): one,
        (3, 'B'): one, (3, 'C'): one, (3, 'D'): one,
        (4, 'A'): one
    }
    B = Mat(Bd, Bf)

    # Print matrices A and B (optional, for verification)
    print("Matrix A:")
    print(A)
    print("\nMatrix B:")
    print(B)

if __name__ == "__main__":
    main()
