def xor(a, b):
    """Perform XOR operation between two hexadecimal strings."""
    d = int(b, 16)
    e = a ^ d
    return e

def key_schedule(password):
    """Generate a key schedule from a password."""
    first_pass = hashlib.sha256(password.encode()).hexdigest()
    second_pass = hashlib.sha256(first_pass.encode()).hexdigest()
    
    first_pass_bytes = [first_pass[i:i+2] for i in range(0, len(first_pass), 2)]
    second_pass_bytes = [second_pass[i:i+2] for i in range(0, len(second_pass), 2)]
    
    key = first_pass_bytes + second_pass_bytes
    return key

def encrypt(image_path, key):
    """Encrypt an image using a generated key."""
    image = Image.open(image_path)
    width, height = image.size
    print(f'Width: {width}, Height: {height}')
    
    private_key = []
    encrypted_image = Image.new('RGB', (width, height), color='black')
    
    for y in range(height):
        for x in range(width):
            cc = image.getpixel((x, y))
            private_key.append(cc)
            encrypted_image.putpixel((x, y), (0, 0, 0))

    encrypted_image.save('encrypted.jpg')
    
    with open('key.txt', 'w') as private_file:
        e = 0
        for pixel in private_key:
            for value in pixel:
                private_file.write(f'{xor(value, key[e]):x}\n')
                e = (e + 1) % len(key)

def decrypt(image_path, key_file):
    """Decrypt an image using a key file."""
    key = key_schedule("password")  # Hardcoded password; adjust if needed

    private_key = []
    with open(key_file) as pk:
        for line in pk:
            private_key.append(int(line.strip(), 16))
    
    image = Image.open(image_path)
    width, height = image.size
    print(f'Width: {width}, Height: {height}')
    
    decrypted_image = Image.new('RGB', (width, height))

    k = 0
    for y in range(height):
        for x in range(width):
            decrypted_image.putpixel(
                (x, y),
                (private_key[k], private_key[k+1], private_key[k+2])
            )
            k += 3
    
    decrypted_image.save('decrypted.jpg')

def browse_file():
    """Open file dialog to select a file."""
    filename = filedialog.askopenfilename()
    return filename

def main():
    """Main function for command-line or GUI interaction."""
    root = Tk()
    root.withdraw()  # Hide the root window

    action = input('e for encryption, d for decryption: ').strip().lower()

    if action == 'e':
        image_path = browse_file()
        password = input('Give me a password: ')
        key = key_schedule(password + '.jpg')
        encrypt(image_path, key)
        print('Encryption completed.')
    elif action == 'd':
        image_path = browse_file()
        key_file = input('Give the private key file: ')
        decrypt(image_path, key_file)
        print('Decryption completed.')
    else:
        print('Input Error')

if __name__ == "__main__":
    main()
