import random
import math

def RSA(p, q):
    n = p * q
    phi_n = (p - 1) * (q - 1)
    relative_primes = []
    
    # Find all relative primes for e
    for e in range(2, phi_n):
        print("Working out the possible relative Primes...", int((e / phi_n) * 100), "%")
        
        # Check if e is coprime with n, p, q, and phi_n
        if math.gcd(e, n) == 1 and math.gcd(e, p) == 1 and math.gcd(e, q) == 1 and math.gcd(e, phi_n) == 1:
            relative_primes.append(e)
    
    relative_primes.sort()
    d = False
    
    # Find a suitable d for the chosen e
    while not d:
        e = random.choice(relative_primes)
        d = d_check(e, phi_n)
    
    return relative_primes, phi_n, n, d, e

def d_check(e, phi_n):
    # Find d such that (d * e) % phi_n == 1
    for d in range(1, phi_n):
        if (d * e) % phi_n == 1:
            return d
