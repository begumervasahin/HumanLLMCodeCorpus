class Graph:
    """
    A simple Graph class for demonstration purposes.
    
    Attributes:
        num_vertices (int): The number of vertices in the graph.
        adj_list (dict): Adjacency list representing the graph.
        path (list): List to store the DFS traversal path.
    """

    def __init__(self, num_vertices):
        """
        Initializes the graph with a given number of vertices.
        
        Args:
            num_vertices (int): The number of vertices in the graph.
        """
        self.num_vertices = num_vertices
        self.adj_list = {i: [] for i in range(num_vertices)}
        self.path = []
        self._initialize_graph()

    def _initialize_graph(self):
        """
        Initializes the graph with some edges for demonstration purposes.
        """
        # Example edges (for demonstration, this should be customized as needed)
        for i in range(self.num_vertices):
            if i < self.num_vertices - 1:
                self.add_edge(i, i + 1)
            if i % 2 == 0 and i + 2 < self.num_vertices:
                self.add_edge(i, i + 2)

    def add_edge(self, u, v):
        """
        Adds an edge to the graph between vertices u and v.

        Args:
            u (int): The start vertex.
            v (int): The end vertex.
        """
        self.adj_list[u].append(v)
        self.adj_list[v].append(u)  # Assuming an undirected graph

    def dfs(self):
        """
        Performs Depth-First Search (DFS) traversal of the graph.
        """
        visited = set()
        self.path = []

        def _dfs(v):
            visited.add(v)
            self.path.append(v)
            for neighbor in self.adj_list[v]:
                if neighbor not in visited:
                    _dfs(neighbor)

        # Start DFS from vertex 0 (or any other vertex)
        _dfs(0)

# Main execution
if __name__ == "__main__":
    # Create a graph with 10 vertices
    G = Graph(10)
    
    # Print adjacency list
    print("Adjacency List:", G.adj_list)
    
    # Perform DFS
    G.dfs()
    
    # Print DFS traversal path
    print("DFS Path:", G.path)
