Encapsulates node data
'''

class Node:
    def __init__(self, val):
        """
        Initializes a Node object.

        Parameters:
        val (int or str): The value associated with the node.
        """
        self.value = val  # The value of the node
        self.parent = self  # The parent of the node (initially itself, for union-find structures)
        self.rank = 0  # The rank of the node (used in union-find to keep the tree flat)

    def __repr__(self):
        """
        Returns a string representation of the Node object.
        """
        return "n " + str(self.value) + " r " + str(self.rank) + " p " + str(self.parent.value)

# Example usage
if __name__ == "__main__":
    # Create a node with value 5
    node = Node(5)
    
    # Print the node
    print(node)  # Output: n 5 r 0 p 5
