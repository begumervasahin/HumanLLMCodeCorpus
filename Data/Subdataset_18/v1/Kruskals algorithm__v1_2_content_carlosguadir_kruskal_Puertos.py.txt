# Function to create a list of all ports
def get_all_ports():
    ports = ['-Vacio-']
    for port_list in countries_and_ports.values():
        ports.extend(port_list)
    return ports

# Function to get the country corresponding to a port
def get_country_by_port(port):
    for country, ports in countries_and_ports.items():
        if port in ports:
            return country
    return None

# Example usage
all_ports = get_all_ports()

country = get_country_by_port('Miami')
