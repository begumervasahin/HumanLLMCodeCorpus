import base64

def main():
    # Read the image file in binary mode and encode it to a base64 string
    with open("rode_in_club.png", "rb") as imageFile:
        encoded_string = base64.b64encode(imageFile.read())
        print(encoded_string)
        
        # Convert the base64 encoded string to an integer
        encoded_integer = int.from_bytes(encoded_string, "big")
        print(encoded_integer)
        
        # Decode the base64 string back to binary
        decoded_bytes = base64.b64decode(encoded_string)
        
        # Write the decoded bytes to a new image file
        with open("new_rode_in_club.png", "wb") as f:
            f.write(decoded_bytes)

if __name__ == "__main__":
    main()
