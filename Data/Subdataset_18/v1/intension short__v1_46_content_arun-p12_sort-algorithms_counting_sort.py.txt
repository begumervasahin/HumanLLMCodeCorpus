def counting_sort(arr, verbose=0, desc=False):
    # Find the largest and smallest elements in the array
    largest = max(arr)
    smallest = min(arr)
    
    # Normalize the data by shifting the smallest number to 0
    if smallest < 0:
        arr = [x - smallest for x in arr]
    
    if verbose:
        print("Normalized array:", arr)
    
    # Create a count array with the range from 0 to largest - smallest
    range_of_elements = largest - smallest + 1
    count = [0] * range_of_elements
    
    # Store the frequency of each element in the count array
    for number in arr:
        count[number] += 1
    
    # Update the count array to hold the cumulative count of each element
    for i in range(1, len(count)):
        count[i] += count[i - 1]
        if verbose == 2:
            print(f"Count after processing index {i}: {count}")
    
    # Output array that will hold the sorted order
    output = [0] * len(arr)
    
    # Build the output array by scanning the original array from right to left
    for i in range(len(arr) - 1, -1, -1):
        current_number = arr[i]
        count[current_number] -= 1
        output[count[current_number]] = current_number
    
    # If there was a normalization shift, shift the numbers back
    if smallest < 0:
        output = [x + smallest for x in output]
    
    # If descending order is requested, reverse the sorted array
    if desc:
        output = output[::-1]
    
    return output

# Example usage:
if __name__ == "__main__":
    A = [4, 2, 2, 8, 3, 3, 1]
    sorted_A = counting_sort(A, verbose=1, desc=False)
    print("Sorted array:", sorted_A)
