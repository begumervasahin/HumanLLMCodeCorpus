import argparse
from Crypto.Cipher import AES
from util_function import hexxor, print_ascii

# Modes for CBC and CTR
MODE_CBC = 1
MODE_CTR = 0

def split_cipher_text(cipher_text):
    """
    Split cipher text into blocks of 32 characters.
    """
    if len(cipher_text) % 32 != 0:
        raise Exception("Cipher text in CBC mode should be padded to a multiple of 32!")
    
    block_num = len(cipher_text) // 32
    return [cipher_text[32 * i: 32 * (i + 1)] for i in range(block_num)]

def eliminate_PKCS5_padding(text):
    """
    Eliminate PKCS5 padding from the text.
    """
    last_byte = int(text[-2]) * 16 + int(text[-1])
    return text[:-last_byte * 2]

def CBC_decipher(key, cipher_text, IV):
    """
    Decipher the given cipher text in CBC mode.
    """
    cipher_text_splitted = split_cipher_text(cipher_text)
    middle_level = []
    plain_text = ''
    aes_cipher = AES.new(bytes.fromhex(key), AES.MODE_ECB)
    
    for current_text in reversed(cipher_text_splitted):
        aes_decipher_result = aes_cipher.decrypt(bytes.fromhex(current_text)).hex()
        middle_level.append(aes_decipher_result)
    
    for i, opera2 in enumerate(reversed(middle_level)):
        opera1 = IV if i == 0 else cipher_text_splitted[i - 1]
        plain_text += hexxor(opera1, opera2)
    
    print_ascii(eliminate_PKCS5_padding(plain_text))

def CTR_decipher(key, cipher_text, IV):
    """
    Decipher the given cipher text in CTR mode.
    """
    aes_cipher = AES.new(bytes.fromhex(key), AES.MODE_ECB)
    block_length = len(cipher_text) // 32 + 1
    opera1 = ""
    
    for i in range(block_length):
        message_encrypting = hex(int(IV, 16) + i)[2:]
        aes_decipher_result = aes_cipher.encrypt(bytes.fromhex(message_encrypting)).hex()
        opera1 += aes_decipher_result
    
    print_ascii(hexxor(opera1, cipher_text))

def main():
    parser = argparse.ArgumentParser(description='Process arguments.')
    parser.add_argument('--mode', type=str, help='Mode for decryption: CBC or CTR')
    args = parser.parse_args()

    mode = MODE_CBC if args.mode is None or args.mode.upper() == 'CBC' else MODE_CTR

    key = input("Enter the key: ")
    cipher_text = input("Enter the cipher text: ")
    IV = cipher_text[:32]
    cipher_text = cipher_text[32:]

    if mode == MODE_CBC:
        CBC_decipher(key, cipher_text, IV)
    else:
        CTR_decipher(key, cipher_text, IV)

if __name__ == "__main__":
    main()
