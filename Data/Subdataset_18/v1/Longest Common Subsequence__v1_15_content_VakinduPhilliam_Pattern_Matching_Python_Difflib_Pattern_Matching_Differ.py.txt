import sys
import difflib

def context_diff(s1, s2, fromfile, tofile):
    """
    Generates a context diff of two sequences of lines.

    Parameters:
    s1 : list of str
        The first sequence of lines.
    s2 : list of str
        The second sequence of lines.
    fromfile : str
        The name of the first file (for diff output).
    tofile : str
        The name of the second file (for diff output).

    Returns:
    generator
        A generator that yields lines of the context diff.
    """
    return difflib.context_diff(s1, s2, fromfile=fromfile, tofile=tofile)

# Define the lists
s1 = ['bacon\n', 'eggs\n', 'ham\n', 'guido\n']
s2 = ['python\n', 'eggy\n', 'hamster\n', 'guido\n']

# Generate the context diff
diff = context_diff(s1, s2, fromfile='before.py', tofile='after.py')

# Write the diff to the standard output
sys.stdout.writelines(diff)
