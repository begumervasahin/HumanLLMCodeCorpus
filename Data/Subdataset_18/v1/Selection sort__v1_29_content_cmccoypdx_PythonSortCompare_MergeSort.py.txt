def merge(a, l, r):
    """
    Merge two sorted subarrays l and r into array a.
    """
    i = j = n = 0
    
    # Merge the two subarrays into a
    while i < len(l) and j < len(r):
        if l[i] < r[j]:
            a[n] = l[i]
            i += 1
        else:
            a[n] = r[j]
            j += 1
        n += 1
    
    # Copy any remaining elements from l
    while i < len(l):
        a[n] = l[i]
        i += 1
        n += 1
    
    # Copy any remaining elements from r
    while j < len(r):
        a[n] = r[j]
        j += 1
        n += 1
    
    return a

def merge_sort(a):
    """
    Perform merge sort on the given array a.
    """
    if len(a) > 1:
        mid = len(a) // 2
        left_half = a[:mid]
        right_half = a[mid:]
        
        # Recursively sort the two halves
        merge_sort(left_half)
        merge_sort(right_half)
        
        # Merge the sorted halves
        merge(a, left_half, right_half)

    arr = [54, 45, 67, 12, 34, 98, 66]
