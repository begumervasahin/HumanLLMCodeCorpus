a = [0, 9, 1, 4, 6, 7, 2, 1, 8, 7, 7, 4, 3, 5, 6, 1, 7, 8, 0, 3]
l = len(a)
li = l
x = 0

# Calculate the largest power of 2 that divides the length of the list
while li % 2 == 0:
    li //= 2
    x += 1

p = 2 ** x
tamis = l // p
inters = l // tamis

print('Numero de intervalos =', inters)
print('Tamano intervalo =', tamis)
print('Longitud de lista =', l)
print(a)

s = 0
g = 1

for h in range(s, l, tamis):
    for k in range(s, g * li - 1):
        b = []
        print('b =', b, '\ng =', g)
        n = 0
        m = 0
        for i in range(s, g * li - 1):
            for j in range(s, g * li):
                if i >= n:
                    if i < j:
                        if a[i] <= a[j]:
                            b.append(a[i])
                            n = j
                            if j == li - 1:
                                b.append(a[j])
                            else:
                                break
                        else:
                            b.append(a[j])
                            if j == li - 1:
                                m += 1
                                b.append(a[i])
                                break
                else:
                    break
            if m > 0:
                break
        a = list(b)
        print('a =', a, '\nb =', b, '\nk =', k, '\ni =', i, '\nj =', j, '\nh =', h, '\ng*li-1 =', g * li - 1)
    g += 1
    s += li
