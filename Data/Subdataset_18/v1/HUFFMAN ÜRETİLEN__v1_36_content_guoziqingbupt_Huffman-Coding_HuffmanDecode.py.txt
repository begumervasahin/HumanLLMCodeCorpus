class Node:
    """A simple Node class for a Huffman tree."""
    def __init__(self, val=None, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

def isLeaf(node):
    """Determine whether a tree node is a leaf."""
    return node.left is None and node.right is None

def huffmanDecode(text, root):
    """
    Decode a Huffman-encoded string using the provided Huffman tree.
    
    :param text: The Huffman-encoded string (binary string).
    :param root: The root node of the Huffman tree.
    :return: The decoded string.
    """
    result = ""
    cur = root
    
    for bit in text:
        if not isLeaf(cur):
            if bit == "0":
                cur = cur.left
            elif bit == "1":
                cur = cur.right
        
        if isLeaf(cur):
            result += cur.val
            cur = root  # Return to the root for the next character

    return result

# Example usage
if __name__ == "__main__":
    # Build a simple Huffman tree for demonstration
    # Example tree:
    #       *
    #      / \
    #     *   c
    #    / \
    #   a   b
    
    root = Node()
    root.left = Node()
    root.right = Node('c')
    root.left.left = Node('a')
    root.left.right = Node('b')
    
    # Example encoded text using the tree
    encoded_text = "000110"  # should decode to 'abc'
    
    # Decode the text
    decoded_text = huffmanDecode(encoded_text, root)
    print("Decoded text:", decoded_text)  # Output should be 'abc'
