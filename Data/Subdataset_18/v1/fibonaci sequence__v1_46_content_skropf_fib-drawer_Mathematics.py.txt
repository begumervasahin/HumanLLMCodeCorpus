def get_doubling_sequence(start, base):
    """
    Generate a sequence of numbers, doubling the start value each time, and converting to the specified base.

    Args:
        start (int): The starting value.
        base (int): The base to convert to.

    Yields:
        str: The converted number in the specified base.
    """
    while True:
        yield convert_decimal_to_base(start, base)
        start *= 2

def convert_decimal_to_base(number, base, converted=''):
    """
    Convert a decimal number to a specified base.

    Args:
        number (int): The decimal number to convert.
        base (int): The base to convert to.
        converted (str): The converted number as a string.

    Returns:
        str: The number converted to the specified base.
    """
    if number:
        if 2 <= base <= 36:
            return convert_decimal_to_base(number // base, base, CHARACTER_MAP[number % base] + converted)
        else:
            return convert_decimal_to_base(number // base, base, str(number % base) + '.' + converted if converted else str(number % base))
    return converted

def convert_base_to_decimal(number, base, converted=0, power=0):
    """
    Convert a number from a specified base to decimal.

    Args:
        number (str): The number in the specified base.
        base (int): The base of the number.
        converted (int): The accumulated converted decimal value.
        power (int): The current power of the base.

    Returns:
        int: The number converted to decimal.
    """
    number = str(number)
    if number:
        if 2 <= base <= 36:
            return convert_base_to_decimal(number[:-1], base, converted + list(CHARACTER_MAP.values()).index(number[-1]) * (base ** power), power + 1)
        else:
            return convert_base_to_decimal('.'.join(number.split('.')[:-1]), base, converted + int(number.split('.')[-1]) * (base ** power), power + 1)
    return converted

def get_digital_root(number, base):
    """
    Get the digital root of a number in a specified base.

    Args:
        number (str): The number in the specified base.
        base (int): The base of the number.

    Returns:
        str: The digital root of the number.
    """
    if convert_base_to_decimal(number, base) >= base:
        if 2 <= base <= 36:
            return get_digital_root(convert_decimal_to_base(sum([list(CHARACTER_MAP.values()).index(elem) for elem in list(str(number))]), base), base)
        else:
            return get_digital_root(convert_decimal_to_base(sum([int(elem) for elem in str(number).split('.')]), base), base)
    return number
