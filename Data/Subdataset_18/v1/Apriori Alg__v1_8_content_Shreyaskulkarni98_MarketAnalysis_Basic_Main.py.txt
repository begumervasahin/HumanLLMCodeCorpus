import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from apyori import apriori

# Load the dataset
store_data = pd.read_csv('D:/Shreyas Kulkarni/Documents/PycharmProjects/MarketAnalysis_Apriori_Basic/store_data.csv', header=None)

# Display the first few rows of the dataset
print(store_data.head())

# Convert pandas dataframe into a list of lists
records = []
for i in range(0, 7501):
    records.append([str(store_data.values[i, j]) for j in range(0, 20)])

# Apply the apriori algorithm to find association rules
association_rules = apriori(records, min_support=0.0045, min_confidence=0.2, min_lift=3, min_length=2)
association_results = list(association_rules)

# Print the number of association rules found
print(f"Number of association rules: {len(association_results)}")

# Print the first association rule
if association_results:
    print(f"First association rule: {association_results[0]}")

# Print all the association rules
for item in association_results:
    # First index of the inner list contains the base item and the add item
    pair = item[0]
    items = [x for x in pair]
    print(f"Rule: {items[0]} -> {items[1]}")

    # Second index of the inner list contains the support
    print(f"Support: {item[1]}")

    # Third index of the list located at 0th of the third index of the inner list
    print(f"Confidence: {item[2][0][2]}")
    print(f"Lift: {item[2][0][3]}")
    print("=====================================")
