import random
import time

def binarySearch(collection, target):
    """Performs binary search on a sorted collection."""
    low = 0
    high = len(collection) - 1
    while high >= low:
        mid = (high + low) // 2
        if target == collection[mid]:
            return mid
        if target < collection[mid]:
            high = mid - 1
        else:
            low = mid + 1
    return -1

def trinarySearch(collection, target):
    """Performs trinary search on a sorted collection."""
    low = 0
    high = len(collection) - 1
    while high >= low:
        one_third = low + (high - low) // 3
        two_thirds = low + 2 * (high - low) // 3
        if target == collection[one_third]:
            return one_third
        elif target < collection[one_third]:
            high = one_third - 1
        elif target == collection[two_thirds]:
            return two_thirds
        elif target < collection[two_thirds]:
            low = one_third + 1
            high = two_thirds - 1
        else:
            low = two_thirds + 1
    return -1

def selectionSort(aList):
    """Sorts a list using selection sort algorithm."""
    for fill in range(len(aList) - 1, 0, -1):
        maxPosition = 0
        for index in range(1, fill + 1):
            if aList[index] > aList[maxPosition]:
                maxPosition = index
        aList[fill], aList[maxPosition] = aList[maxPosition], aList[fill]

def createList1(n):
    """Creates a list of even integers."""
    aList = random.sample(range(1, 16001), n)
    for index in range(len(aList)):
        if aList[index] % 2 != 0:
            aList[index] += 1
    return aList

def createOddList(n):
    """Creates a list of odd integers."""
    aList = random.sample(range(1, 1000000), 10 * n)
    for index in range(len(aList)):
        if aList[index] % 2 == 0:
            aList[index] += 1
    return aList

def createList2(list1):
    """Creates a list by repeating each element of list1 ten times."""
    list2 = []
    for integer in list1:
        list2.extend([integer] * 10)
    return list2

def checkTime(list1, list2):
    """Measures and prints the time taken by binary and trinary search."""
    t0 = time.time()
    for index in list2:
        result1 = binarySearch(list1, index)
    print("Binary Search time: {:.6f} seconds".format(time.time() - t0))

    t0 = time.time()
    for index in list2:
        result2 = trinarySearch(list1, index)
    print("Trinary Search time: {:.6f} seconds".format(time.time() - t0))
    print()

def testing(n):
    """Runs the experiment with lists of size n."""
    print("Experiment with even integer list:")
    list1 = createList1(n)
    selectionSort(list1)
    list2 = createList2(list1)
    checkTime(list1, list2)

    print("Experiment with odd integer list:")
    list2 = createOddList(n)
    checkTime(list1, list2)

def main():
    """Main function to execute the experiments with various sizes of n."""
    nValues = [1000, 2000, 4000, 8000, 16000]
    for index in nValues:
        print(f"For n = {index}\n")
        testing(index)

if __name__ == "__main__":
    main()
