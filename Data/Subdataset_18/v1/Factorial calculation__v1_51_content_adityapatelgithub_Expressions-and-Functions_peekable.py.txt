class Peekable:
    """
    An iterator with the ability to examine a value without advancing.
    """
    def __init__(self, iterator):
        """
        Initializes the Peekable iterator.

        :param iterator: An existing iterator to enhance with peek functionality.
        """
        self._iterator = iterator
        self._peeked = None

    def __iter__(self):
        return self

    def __next__(self):
        """
        Returns the next element of the data. If the element has been peeked at, 
        no advancement occurs.

        :return: The next element from the iterator.
        """
        if self._peeked is None:
            self._peeked = next(self._iterator)
        ans = self._peeked
        self._peeked = None
        return ans

    def peek(self):
        """
        Peeks at the next element of the data without advancing the iterator.
        Only advances if that next item has not yet been peeked at.

        :return: The next element from the iterator.
        """
        if self._peeked is None:
            self._peeked = next(self._iterator)
        return self._peeked

def peek(iterator):
    """
    Returns the next element of the iterator without advancing it.

    :param iterator: A Peekable iterator.
    :return: The next element from the iterator.
    """
    return iterator.peek()

# Example usage
if __name__ == "__main__":
    i = Peekable(iter([1, 2, 3, 4, 5]))

    print(peek(i))  # Peek at the next item (1)
    print(peek(i))  # Peek at the next item again (1)
    print(next(i))  # Get the next item (1)
    print(next(i))  # Get the next item (2)
    print(next(i))  # Get the next item (3)
    print(peek(i))  # Peek at the next item (4)
    print(next(i))  # Get the next item (4)
    print(list(Peekable(iter([1, 2, 3, 4, 5]))))  # Convert to list and print all items
