# Define infinity as a large enough value. This represents the absence of an edge between nodes
INF = float("inf")

# Define the graph as an adjacency matrix
graph = [
    [0, 5, INF, 10],
    [INF, 0, 3, INF],
    [INF, INF, 0, 1],
    [INF, INF, INF, 0]
]

def floyd_warshall(graph):
    """
    Function to perform the Floyd-Warshall algorithm and print the shortest path matrix.
    
    Parameters:
    graph (list): 2D list representing the graph as an adjacency matrix.
    """
    # Initialize the distance matrix with the given graph
    distance = [row[:] for row in graph]
    len_matrix = len(graph)
    
    # Run the Floyd-Warshall algorithm
    for k in range(len_matrix):
        for i in range(len_matrix):
            for j in range(len_matrix):
                distance[i][j] = min(distance[i][j], distance[i][k] + distance[k][j])
    
    # Print the resulting distance matrix
    for row in distance:
        print("\t".join(map(lambda x: f"{x if x != INF else 'INF'}", row)))

if __name__ == '__main__':
    floyd_warshall(graph)
