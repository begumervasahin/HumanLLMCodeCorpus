class Vertex:
    def __init__(self, vertex):
        """
        Initializes a Vertex object.

        Parameters:
        vertex (str or int): The name or identifier of the vertex.
        """
        self.name = vertex  # The name of the vertex
        self.neighbors = []  # A list to store the names of adjacent vertices

    def add_neighbor(self, neighbor):
        """
        Adds a neighbor to the vertex's adjacency list.

        Parameters:
        neighbor (Vertex): The neighboring vertex to be added.

        Returns:
        bool: False if the neighbor is not a Vertex instance, otherwise None.
        """
        if isinstance(neighbor, Vertex):
            if neighbor.name not in self.neighbors:
                # Add the neighbor's name to the current vertex's neighbors
                self.neighbors.append(neighbor.name)
                # Add this vertex's name to the neighbor's neighbors
                neighbor.neighbors.append(self.name)
                # Keep neighbors lists sorted
                self.neighbors = sorted(self.neighbors)
                neighbor.neighbors = sorted(neighbor.neighbors)
        else:
            return False

    def __repr__(self):
        """
        Returns a string representation of the vertex's neighbors.
        """
        return str(self.neighbors)

# Example usage
if __name__ == "__main__":
    # Create vertex instances
    v1 = Vertex("A")
    v2 = Vertex("B")
    v3 = Vertex("C")

    # Add neighbors
    v1.add_neighbor(v2)
    v1.add_neighbor(v3)

    # Display the neighbors of each vertex
    print(f"Neighbors of {v1.name}: {v1}")
    print(f"Neighbors of {v2.name}: {v2}")
    print(f"Neighbors of {v3.name}: {v3}")
