import math

class Edge:
    def __init__(self, weight):
        self.weight = weight

    def __repr__(self):
        return f"Edge({self.weight})"

class Heapq:
    def __init__(self, d):
        """
        Initialize the heap with a branching factor of 'd'.
        """
        self.data = []
        self.d = d

    def make_heap(self, lst):
        """
        Build the heap from an unordered list.
        """
        self.data = lst
        for k in range(self.size() - 1, -1, -1):
            self.siftdown(k)

    def find_min(self):
        """
        Return the minimum element (the root of the heap).
        """
        return self.data[0] if self.size() > 0 else None

    def delete(self, k):
        """
        Delete the element at index 'k'.
        """
        if k == self.size() - 1:
            self.data.pop()
        else:
            last = self.data.pop()
            item = self.data[k]
            self.data[k] = last
            if last.weight < item.weight:
                self.siftup(k)
            else:
                self.siftdown(k)

    def delete_min(self):
        """
        Remove the minimum element (the root of the heap).
        """
        if self.size() > 0:
            self.delete(0)

    def size(self):
        """
        Return the size of the heap.
        """
        return len(self.data)

    def siftup(self, k):
        """
        Move the element at index 'k' up the heap to restore heap order.
        """
        while k > 0:
            parent = (k - 1) // self.d
            if self.data[k].weight < self.data[parent].weight:
                self.data[k], self.data[parent] = self.data[parent], self.data[k]
                k = parent
            else:
                break

    def siftdown(self, k):
        """
        Move the element at index 'k' down the heap to restore heap order.
        """
        while True:
            j = self.minchild(k)
            if j is None or self.data[j].weight >= self.data[k].weight:
                break
            self.data[k], self.data[j] = self.data[j], self.data[k]
            k = j

    def minchild(self, k):
        """
        Return the index of the smallest child for the element at index 'k'.
        """
        start = self.d * k + 1
        end = min(self.d * k + self.d, self.size() - 1)
        if start > end:
            return None
        min_idx = start
        min_weight = self.data[start].weight
        for i in range(start + 1, end + 1):
            if self.data[i].weight < min_weight:
                min_idx = i
                min_weight = self.data[i].weight
        return min_idx

    def __repr__(self):
        """
        String representation of the heap.
        """
        return str(self.data)

# Example usage:
if __name__ == "__main__":
    edges = [Edge(weight) for weight in [5, 3, 8, 1, 2, 9, 7]]
    heap = Heapq(d=2)
    heap.make_heap(edges)
    print("Heap after creation:", heap)
    print("Minimum element:", heap.find_min())
    heap.delete_min()
    print("Heap after deleting minimum element:", heap)
