import utils
import filenames

class Log:
    def __init__(self, k, des_name, svm_kernel):
        self.text = ""
        self.k = k
        self.des_name = des_name
        self.svm_kernel = svm_kernel

    def save(self):
        with open(filenames.log(self.k, self.des_name, self.svm_kernel), "w") as file:
            file.write(self.text)

    def train_des_time(self, time):
        elapsed_time = utils.humanize_time(time)
        self.text += f"Time for getting all the local descriptors of the training images was {elapsed_time}.\n"

    def codebook_time(self, time):
        elapsed_time = utils.humanize_time(time)
        self.text += f"Time for generating the codebook with k-means was {elapsed_time}.\n"

    def train_vlad_time(self, time):
        self.vlad_time(time, "training")

    def svm_time(self, time):
        elapsed_time = utils.humanize_time(time)
        self.text += f"Time for calculating the SVM was {elapsed_time}.\n"

    def test_vlad_time(self, time):
        self.vlad_time(time, "testing")

    def predict_time(self, time):
        elapsed_time = utils.humanize_time(time)
        self.text += f"Elapsed time predicting the testing set is {elapsed_time}\n"

    def accuracy(self, accuracy):
        self.text += f"Accuracy = {accuracy}.\n"

    def classes(self, classes):
        self.text += f"Classes = {classes}\n"

    def classes_counts(self, classes_counts):
        self.text += f"Classes Local Descriptors Counts = {classes_counts}\n"

    def confusion_matrix(self, conf_mat):
        self.text += f"Confusion Matrix =\n{conf_mat}"

    def vlad_time(self, time, set):
        elapsed_time = utils.humanize_time(time)
        self.text += f"Time for getting VLAD global descriptors of the {set} images was {elapsed_time}.\n"

# Example usage
if __name__ == "__main__":
    log = Log(k=10, des_name="SIFT", svm_kernel="linear")
    log.train_des_time(120)
    log.codebook_time(300)
    log.train_vlad_time(180)
    log.svm_time(60)
    log.test_vlad_time(90)
    log.predict_time(30)
    log.accuracy(0.85)
    log.classes(["cat", "dog"])
    log.classes_counts({"cat": 100, "dog": 120})
    log.confusion_matrix([[90, 10], [20, 100]])
    log.save()
