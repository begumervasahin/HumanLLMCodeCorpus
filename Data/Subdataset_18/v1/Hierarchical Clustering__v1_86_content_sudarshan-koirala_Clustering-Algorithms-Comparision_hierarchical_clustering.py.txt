import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from scipy.cluster.hierarchy import dendrogram, linkage
from sklearn.cluster import AgglomerativeClustering

def load_dataset(file_path):
    """
    Load the dataset from a CSV file.

    Args:
        file_path (str): Path to the CSV file.

    Returns:
        np.ndarray: Feature matrix X for clustering.
    """
    dataset = pd.read_csv(file_path)
    return dataset.iloc[:, [3, 4]].values

def plot_dendrogram(X):
    """
    Plot the dendrogram for hierarchical clustering.

    Args:
        X (np.ndarray): Feature matrix used for clustering.
    """
    plt.figure(figsize=(10, 7))
    dendrogram(linkage(X, method='ward'))
    plt.title('Dendrogram')
    plt.xlabel('Customers')
    plt.ylabel('Euclidean distances')
    plt.show()

def perform_agglomerative_clustering(X, n_clusters=5):
    """
    Perform Agglomerative Clustering and return the cluster labels.

    Args:
        X (np.ndarray): Feature matrix used for clustering.
        n_clusters (int): Number of clusters to form.

    Returns:
        np.ndarray: Cluster labels for each point in the dataset.
    """
    hc = AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage='ward')
    return hc.fit_predict(X)

def plot_clusters(X, y_hc):
    """
    Plot the clusters identified by Agglomerative Clustering.

    Args:
        X (np.ndarray): Feature matrix used for clustering.
        y_hc (np.ndarray): Cluster labels for each point in the dataset.
    """
    plt.figure(figsize=(10, 7))
    plt.scatter(X[y_hc == 0, 0], X[y_hc == 0, 1], s=100, c='red', label='Careful')
    plt.scatter(X[y_hc == 1, 0], X[y_hc == 1, 1], s=100, c='blue', label='Standard')
    plt.scatter(X[y_hc == 2, 0], X[y_hc == 2, 1], s=100, c='green', label='Target')
    plt.scatter(X[y_hc == 3, 0], X[y_hc == 3, 1], s=100, c='cyan', label='Careless')
    plt.scatter(X[y_hc == 4, 0], X[y_hc == 4, 1], s=100, c='magenta', label='Sensible')
    plt.title('Clusters of clients (Hierarchical clustering)')
    plt.xlabel('Annual Income (k$)')
    plt.ylabel('Spending Score (1-100)')
    plt.legend()
    plt.show()

def main():
    """
    Main function to execute the clustering analysis and visualization.
    """
    # Load the dataset
    X = load_dataset('Mall_Customers.csv')
    
    # Plot the dendrogram to determine the optimal number of clusters
    plot_dendrogram(X)
    
    # Perform Agglomerative Clustering
    y_hc = perform_agglomerative_clustering(X)
    
    # Plot the clusters
    plot_clusters(X, y_hc)

if __name__ == "__main__":
    main()
