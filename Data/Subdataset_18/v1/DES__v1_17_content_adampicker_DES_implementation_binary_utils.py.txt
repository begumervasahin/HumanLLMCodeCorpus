def read_binary_file(filename_path):
    """
    Reads a binary file and returns its content as a binary string.

    Parameters
    ----------
    filename_path : str
        The path to the binary file.

    Returns
    -------
    str
        The binary content of the file as a string.
    """
    string_binary = ''
    with open(filename_path, mode='rb') as file:
        file_bytes = file.read()
    for byte in file_bytes:
        binary_str = bin(byte)[2:].zfill(8)
        string_binary += binary_str
    return string_binary

def write_binary_file(binary):
    """
    Converts a binary string to a bytearray.

    Parameters
    ----------
    binary : str
        The binary string to be converted.

    Returns
    -------
    bytearray
        The bytearray representation of the binary string.
    """
    byte_array = bytearray()
    for i in range(0, len(binary), 8):
        byte = binary[i:i+8]
        byte_array.append(int(byte, 2))
    return byte_array

def hex_key_to_binary(hex_key):
    """
    Converts a hexadecimal key to a binary string.

    Parameters
    ----------
    hex_key : str
        The hexadecimal key.

    Returns
    -------
    str
        The binary string representation of the hexadecimal key.
    """
    binary_key = ''.join("{:04b}".format(int(char, 16)) for char in hex_key)
    return binary_key

# Example usage
if __name__ == "__main__":
    # Example file read and write
    filename_path = 'example.bin'
    
    # Read binary file
    binary_content = read_binary_file(filename_path)
    print("Binary content of file:", binary_content)
    
    # Convert binary string to bytearray
    byte_array = write_binary_file(binary_content)
    print("Bytearray:", byte_array)
    
    # Example hex key to binary
    hex_key = '1a2b3c'
    binary_key = hex_key_to_binary(hex_key)
    print("Binary key:", binary_key)
