from __future__ import print_function
import CPUtimer
from numpy.polynomial import polynomial as P
from data3 import instance_iterator, print_solution

def solve(instance_path):
    """Solve the polynomial multiplication problem for each instance in the given path."""
    timer = CPUtimer.CPUTimer()
    
    for instance in instance_iterator(instance_path):
        instance_name, degree, p1, p2 = instance
        
        timer.reset()
        timer.start()
        
        for _ in range(2):  # Looping twice to measure execution time
            result = P.polymul(p1, p2)
            timer.lap()
        
        timer.stop()
        print_solution(result, instance_name, '3b', timer)

def mult(degree, x, y):
    """Perform polynomial multiplication using the Karatsuba algorithm."""
    product = []
    for i in range(degree):
        product.append(karatsuba(int(x[i]), int(y[i])))
    return product

def karatsuba(x, y):
    """Karatsuba algorithm for fast multiplication."""
    if len(str(x)) == 1 or len(str(y)) == 1:
        return x * y
    else:
        n = max(len(str(x)), len(str(y)))
        nby2 = n // 2
        a = x // 10**nby2
        b = x % 10**nby2
        c = y // 10**nby2
        d = y % 10**nby2
        
        ac = karatsuba(a, c)
        bd = karatsuba(b, d)
        ad_plus_bc = karatsuba(a + b, c + d) - ac - bd
        
        return ac * 10**(2 * nby2) + ad_plus_bc * 10**nby2 + bd
