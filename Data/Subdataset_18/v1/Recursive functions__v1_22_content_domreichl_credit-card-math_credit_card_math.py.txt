def calc_balance(balance, min_payment_rate, monthly_interest_rate, months, payments):
    """
    Calculates the credit card balance after X months for
    a person who only pays the minimum monthly payment.
    It also builds a list of monthly payments.
    """
    if months <= 0:
        return round(balance, 2), payments
    else:
        min_payment = min_payment_rate * balance
        payments.append(min_payment)
        unpaid_balance = balance - min_payment
        balance = unpaid_balance + monthly_interest_rate * unpaid_balance
        return calc_balance(balance, min_payment_rate, monthly_interest_rate, months - 1, payments)

def pay_off_debt(balance, monthly_interest_rate, months, lower, upper):
    """
    Calculates the minimum fixed monthly payment needed in order to pay off
    a credit card balance within X months using bisection search.
    """
    guess = (lower + upper) / 2
    if months <= 0:
        if round(balance) == 0:
            return round(guess, 2)
        elif balance < 0:
            return pay_off_debt(balance, monthly_interest_rate, months, lower, guess)
        else:
            return pay_off_debt(balance, monthly_interest_rate, months, guess, upper)
    else:
        unpaid_balance = balance - guess
        balance = unpaid_balance + monthly_interest_rate * unpaid_balance
        return pay_off_debt(balance, monthly_interest_rate, months - 1, lower, upper)

def main():
    balance = 100000
    months = 12
    annual_interest_rate = 0.22
    min_payment_rate = 0.05
    monthly_interest_rate = annual_interest_rate / 12.0

    # Calculate balance after making only minimum payments
    payments = []
    new_balance, payments = calc_balance(balance, min_payment_rate, monthly_interest_rate, months, payments)
    average_payment = round(sum(payments) / len(payments), 2)
    loss1 = round(abs(balance - new_balance - sum(payments)), 2)

    # Calculate the fixed monthly payment needed to pay off debt
    min_guess = balance / 12
    max_guess = (balance * (1 + monthly_interest_rate) ** months) / months
    payment = pay_off_debt(balance, monthly_interest_rate, months, min_guess, max_guess)
    loss2 = round(abs(balance - 0 - payment * months), 2)

    # Output the results
    print('Current Balance:', balance, '€')
    print('Duration:', months, 'months')
    print('With a monthly payment rate of {:.2f}:'.format(min_payment_rate))
    print('   - You will pay', average_payment, '€ per month on average.')
    print('   - Your loss will be', loss1, '€ for that time period.')
    print('   - Your remaining balance will be', new_balance, '€.')
    print('If you want to pay off your balance within', months, 'months:')
    print('   - You will have to pay', payment, '€ per month.')
    print('   - Your loss will be', loss2, '€ in total.')
    print('   - Your remaining balance will be 0 €.')

if __name__ == "__main__":
    main()
