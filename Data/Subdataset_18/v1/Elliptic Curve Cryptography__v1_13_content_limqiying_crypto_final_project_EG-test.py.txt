from ecc_simple import *
import random

def main():
    """
    ElGamal encryption and decryption demonstration.
    """
    set_p256_param()

    # Parameters for P-256 curve
    p = 115792089210356248762697446949407573530086143415290314195533631308867097853951
    x = 48439561293906451759052585252797914202762949526041747995844080717082404635286
    y = 36134250956749795798585127919587881956611106672985015071877198253568414405109
    g = (x, y)

    # Generate private keys for Alice and Bob
    a = random.randrange(1, p)
    b = random.randrange(1, p)

    # Generate public keys for Bob and Alice
    public_Bob = times(b, g)
    public_Alice = times(a, g)

    # Alice's message to be encrypted
    m = 32454324564335645345643

    # Alice calculates the shared key and ciphertext
    shared_key_A = times(a, public_Bob)
    c_1 = public_Alice
    c_2 = add(encode(m, 100), shared_key_A)
    ciphertext = (c_1, c_2)

    # Bob calculates the shared key and decrypts the message
    shared_key_B = times(b, ciphertext[0])
    plaintext = subtract(ciphertext[1], shared_key_B)
    decrypted_message = decode(plaintext, 100)

    # Print the decrypted message
    print(decrypted_message)

if __name__ == "__main__":
    main()
