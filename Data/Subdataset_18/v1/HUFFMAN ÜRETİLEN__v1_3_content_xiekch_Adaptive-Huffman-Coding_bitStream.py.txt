class BitStream:
    def __init__(self, file_name, mode):
        """Initialize the BitStream object with a file and mode."""
        self.file = open(file_name, mode)
        self.word = 0
        self.pos = 7 if mode[0] == 'w' else -1
        self.mode = mode

    def read(self, size):
        """Read a specified number of bits from the file."""
        ret = ''
        for _ in range(size):
            if self.pos == -1:
                self.word = self.file.read(1)
                if not self.word:
                    return ret
                self.word = ord(self.word)
                self.pos = 7
            ret += '1' if self.word & (1 << self.pos) else '0'
            self.pos -= 1
        return ret

    def write(self, string):
        """Write a string of bits to the file."""
        for char in string:
            if char not in ('0', '1'):
                continue
            self.word = (self.word << 1) | (1 if char == '1' else 0)
            self.pos -= 1
            if self.pos == -1:
                self.flush()

    def flush(self):
        """Flush the remaining bits to the file."""
        if self.pos != 7:
            self.word <<= self.pos + 1
            self.file.write(self.word.to_bytes(1, byteorder='little'))
        self.word = 0
        self.pos = 7

    def close(self):
        """Flush the remaining bits and close the file."""
        if self.mode[0] == 'w':
            self.flush()
        self.file.close()
