import fitz  # PyMuPDF
from storage import enumrateFilenames

def read_all_pdfs():
    """
    Read all the PDF files from storage and return the text from all in a list along with the file names.

    Returns:
        list: A list of tuples, each containing a PDF name and the extracted text data from that PDF.
    """
    pages = []
    for pdf in enumrateFilenames():
        with fitz.open(pdf) as infile:
            for page in infile:
                pages.append((pdf, page.get_text()))
    return pages

def read_pdf(pdf_name):
    """
    Read a PDF file from storage and return the text from it in a list along with the file name.

    Args:
        pdf_name (str): Path to the PDF file.

    Returns:
        list: A list of tuples, each containing the PDF name and the extracted text data from the PDF.
    """
    pages = []
    with fitz.open(pdf_name) as infile:
        for page in infile:
            pages.append((pdf_name, page.get_text()))
    return pages
