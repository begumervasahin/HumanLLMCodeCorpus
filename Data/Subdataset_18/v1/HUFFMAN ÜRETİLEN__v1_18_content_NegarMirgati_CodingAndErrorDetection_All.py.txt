import random

# Placeholder class for Huffman coding
class HuffmanCoding:
    def encode(self, data):
        # A simple mock encoding for demonstration
        return ''.join(format(ord(char), '08b') for char in data)

    def decode(self, encoded_data):
        # Decodes binary string to original string
        chars = [encoded_data[i:i + 8] for i in range(0, len(encoded_data), 8)]
        return ''.join(chr(int(char, 2)) for char in chars)

# Placeholder class for Convolutional coding
class Convolutional:
    def encode(self, data):
        # A simple mock convolutional encoding (identity function)
        return data

    def decode(self, encoded_data):
        # A simple mock convolutional decoding (identity function)
        return encoded_data

def noise(input_data):
    output = ''
    for i in range(len(input_data)):
        r = random.uniform(0, 1)
        if r < 0.1:  # Introduce noise with a 10% probability
            output += str(1 - int(input_data[i]))  # Flip the bit
        else:
            output += input_data[i]
    return output

def main():
    h = HuffmanCoding()
    c = Convolutional()
    
    # Input string
    input_data = "negarmirgati"
    print('Input:', input_data)
    
    # Huffman encoding
    h_encoded = h.encode(input_data)
    print('Huffman encoded:', h_encoded)
    
    # Convolutional encoding
    c_encoded = c.encode(h_encoded)
    print('Convolutional encoded:', c_encoded)
    
    # Adding noise
    input_with_noise = noise(c_encoded)
    print('Noise added:', input_with_noise)
    
    # Convolutional decoding
    c_decoded = c.decode(input_with_noise)
    print('Convolutional decoded:', c_decoded)
    
    # Huffman decoding
    h_decoded = h.decode(c_decoded)
    print('Huffman decoded:', h_decoded)

if __name__ == "__main__":
    main()
