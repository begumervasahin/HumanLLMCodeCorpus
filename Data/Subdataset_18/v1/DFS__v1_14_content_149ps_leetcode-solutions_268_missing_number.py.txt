def missing_number(nums):
    l = 0
    u = max(nums)
    length = u - l + 1
    print(l, u, length)
    
    i = 0
    while length != 0:
        if l + i in nums:
            i += 1
            length -= 1
        else:
            return l + i
    
    return l + i

# Example usage
nums = [0]
print(missing_number(nums))  # Output should be 1

# Additional test cases
nums = [3, 0, 1]
print(missing_number(nums))  # Output should be 2

nums = [0, 1]
print(missing_number(nums))  # Output should be 2

nums = [9,6,4,2,3,5,7,0,1]
print(missing_number(nums))  # Output should be 8
