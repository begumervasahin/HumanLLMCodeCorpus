def factorial_loop(n):
    """
    Calculates the factorial of a positive integer n using a loop.

    :param n: The number to calculate the factorial for.
    :return: The factorial of the number.
    """
    factorial = 1
    for i in range(n, 0, -1):
        factorial *= i
    return factorial

def factorial_recursive(n):
    """
    Calculates the factorial of a positive integer n using recursion.

    :param n: The number to calculate the factorial for.
    :return: The factorial of the number.
    """
    if n == 0:
        return 1
    elif n == 1:
        return 1
    else:
        return n * factorial_recursive(n - 1)

def get_positive_integer(prompt):
    """
    Prompts the user to input a positive integer and validates the input.

    :param prompt: The prompt message to display to the user.
    :return: A valid positive integer entered by the user.
    """
    while True:
        user_input = input(prompt)
        if user_input.isdigit():
            return int(user_input)
        else:
            print("Invalid input. Please enter a whole number.")

def main():
    """
    Main function to demonstrate the factorial calculations using loop and recursion.
    """
    # Loop method
    number = get_positive_integer("Loop Method: \nWhat number would you like to find the factorial for? \nEnter a whole number: ")
    loop_result = factorial_loop(number)
    print(f"Factorial (loop method) of {number} is: {loop_result}")

    # Recursion method
    number = get_positive_integer("Recursion Method: \nWhat number would you like to find the factorial for? \nEnter a whole number: ")
    recursion_result = factorial_recursive(number)
    print(f"Factorial (recursion method) of {number} is: {recursion_result}")

if __name__ == "__main__":
    main()
