import numpy as np

def phi(w):
    """Sigmoid function."""
    return 1 / (1 + np.exp(-w))

def phiprim(w):
    """Derivative of the sigmoid function."""
    exp_neg_w = np.exp(-w)
    
    if exp_neg_w < 10**16:
        if exp_neg_w > 10**-16:
            return exp_neg_w / ((1 + exp_neg_w) ** 2)
        else:
            return exp_neg_w
    else:
        return np.exp(w)

def phi2prim(w):
    """Second derivative of the sigmoid function."""
    exp_neg_w = np.exp(-w)
    
    if exp_neg_w < 10**16:
        if exp_neg_w > 10**-16:
            exp_w = np.exp(w)
            return -exp_w * (exp_w - 1) / ((1 + exp_w) ** 3)
        else:
            return -exp_neg_w
    else:
        return np.exp(w)

def loss(x, S, A, y, arg=1, penalization=False):
    """
    Compute the loss function and optionally its gradient and Hessian.
    
    Parameters:
    x (array): Point for which the loss must be computed.
    S (list): Subset of indices.
    A (array): Matrix of the datapoints.
    y (array): Vector of labels.
    arg (int): Number of output parameters. 
               1 for loss, 2 for loss and gradient, 3 for loss, gradient, and Hessian.
    penalization (bool): Use penalized empirical loss function or not.
    
    Returns:
    Depending on arg:
    - Value of the loss function.
    - Value of the loss function and its gradient.
    - Value of the loss function, its gradient, and its Hessian.
    """
    d = A.shape[1]
    card_S = len(S)
    f = 0
    
    if arg >= 2:
        g = np.zeros(d)
    if arg == 3:
        H = np.zeros((d, d))
    
    for i in range(card_S):
        y_S_i = y[S[i]]
        A_S_i = A[S[i]]
        scal_prod = np.dot(x, A_S_i)
        
        f += (y_S_i - phi(scal_prod)) ** 2
        
        if arg >= 2:
            g += 2 * A_S_i * phiprim(scal_prod) * (phi(scal_prod) - y_S_i)
        if arg == 3:
            H += 2 * ((phi(scal_prod) - y_S_i) * phi2prim(scal_prod) + 
                      phiprim(scal_prod) ** 2) * np.outer(A_S_i, A_S_i)
    
    if penalization:
        f += 0.5 * np.linalg.norm(x) ** 2
    
    f /= card_S
    
    if arg == 3:
        H = 0.5 * (H + H.T)
        if penalization:
            g += x
            H += np.eye(d)
        g /= card_S
        H /= card_S
        return f, g, H
    elif arg == 2:
        if penalization:
            g += x
        g /= card_S
        return f, g
    else:
        return f
