class Node:
    def __init__(self, value):
        """Initialize the node with a value, leftChild, and rightChild."""
        self.value = value
        self.leftChild = None
        self.rightChild = None

    def insert(self, data):
        """
        Insert a new node with 'data' into the BST.
        Returns True if the node is inserted, False if the value already exists.
        """
        if self.value == data:
            return False
        elif self.value > data:
            if self.leftChild:
                return self.leftChild.insert(data)
            else:
                self.leftChild = Node(data)
                return True
        else:
            if self.rightChild:
                return self.rightChild.insert(data)
            else:
                self.rightChild = Node(data)
                return True

    def find(self, data):
        """
        Search for a node with 'data' in the BST.
        Returns True if found, False otherwise.
        """
        if self.value == data:
            return True
        elif self.value > data:
            if self.leftChild:
                return self.leftChild.find(data)
            else:
                return False
        else:
            if self.rightChild:
                return self.rightChild.find(data)
            else:
                return False

    def preorder(self):
        """Traverse the tree in preorder and print each node's value."""
        print(str(self.value))
        if self.leftChild:
            self.leftChild.preorder()
        if self.rightChild:
            self.rightChild.preorder()

    def inorder(self):
        """Traverse the tree in inorder and print each node's value."""
        if self.leftChild:
            self.leftChild.inorder()
        print(str(self.value))
        if self.rightChild:
            self.rightChild.inorder()

    def postorder(self):
        """Traverse the tree in postorder and print each node's value."""
        if self.leftChild:
            self.leftChild.postorder()
        if self.rightChild:
            self.rightChild.postorder()
        print(str(self.value))

class Tree:
    def __init__(self):
        """Initialize the tree with an empty root."""
        self.root = None

    def insert(self, data):
        """Insert 'data' into the BST. If the tree is empty, create the root."""
        if self.root:
            return self.root.insert(data)
        else:
            self.root = Node(data)
            return True

    def find(self, data):
        """Search for 'data' in the BST."""
        if self.root:
            return self.root.find(data)
        else:
            return False

    def preorder(self):
        """Print the preorder traversal of the BST."""
        if self.root:
            print("Preorder")
            self.root.preorder()

    def inorder(self):
        """Print the inorder traversal of the BST."""
        if self.root:
            print("Inorder")
            self.root.inorder()

    def postorder(self):
        """Print the postorder traversal of the BST."""
        if self.root:
            print("Postorder")
            self.root.postorder()

# Example usage:
bst = Tree()
bst.insert(1)
bst.insert(2)
bst.insert(3)
bst.preorder()
bst.inorder()
bst.postorder()
