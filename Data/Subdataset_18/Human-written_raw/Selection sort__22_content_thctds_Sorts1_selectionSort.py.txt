a = list()
num = raw_input("Enter how many elements you want:")
print 'Enter numbers in array: '
for i in range(int(num)):
    n = raw_input("num :")
    a.append(int(n))
print 'ARRAY: ',a
i = 0
while i<len(a):
    smallest = min(a[i:])
    index_of_smallest = a.index(smallest)
    a[i],a[index_of_smallest] = a[index_of_smallest],a[i]
    i=i+1
print 'SORTED ARRAY:',(a)
