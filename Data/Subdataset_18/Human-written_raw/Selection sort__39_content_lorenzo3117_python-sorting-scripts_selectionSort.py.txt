def selectionSort(arr):
    comparisons = 0
    swaps = 0
    for i in range(0, len(arr) - 1):
        min_idx = i
        for j in range(i+1, len(arr)):
            comparisons += 1
            if arr[j] < arr[min_idx]:
                min_idx = j
        arr[i], arr[min_idx] = arr[min_idx], arr[i]
        swaps += 1
    return "SELECTION SORT:\nComparisons: " + str(comparisons) + "\nSwaps: " + str(swaps)
