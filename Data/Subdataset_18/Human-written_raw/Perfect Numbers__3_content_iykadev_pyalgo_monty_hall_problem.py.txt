import random
stay = 0
switch = 0
for i in range(1000):
    lst = [1,0,0]
    random.shuffle(lst)
    ran = random.randrange(3)
    user = lst[ran]
    del(lst[ran])
    huh = 0
    for i in lst:
        if i ==0:
            del(lst[huh])
            break
        huh+=1
    if user ==1:
        stay+=1
    if lst[0] == 1:
        switch+=1
print("Stay =",stay)
print("Switch = ",switch)
