def selectionSort(ar):
    i = 0
    while i < len(ar):
        smallest = min(ar[i:])
        index_of_smallest = ar.index(smallest)
        ar[i], ar[index_of_smallest] = ar[index_of_smallest], ar[i]
        i = i + 1
    for elements in ar:
        print(elements)
ar = [49, 25, 26, 2, 99]
print("Sorted array is:")
selectionSort(ar)
