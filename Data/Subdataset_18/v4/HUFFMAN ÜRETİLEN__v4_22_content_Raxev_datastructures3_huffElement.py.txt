class HuffElement(Comparable):
    """
    Represents an element used in conjunction with the HuffMap class.
    Stores a character, its frequency in a string, and its corresponding Huffman code.
    """

    def __init__(self, char: str):
        """
        Initializes a HuffElement with the given character.
        The frequency is set to zero and the code is set to an empty string.
        """
        self._char = char
        self._frequency = 0
        self._code = ""

    def increment_frequency(self):
        """Increments the frequency count of the character."""
        self._frequency += 1

    @property
    def frequency(self) -> int:
        """Returns the frequency count of the character."""
        return self._frequency

    @frequency.setter
    def frequency(self, count: int):
        """Sets the frequency count of the character."""
        self._frequency = count

    @property
    def code(self) -> str:
        """Returns the Huffman code for the character."""
        return self._code

    @code.setter
    def code(self, huffman_code: str):
        """Sets the Huffman code for the character."""
        self._code = huffman_code

    @property
    def char(self) -> str:
        """Returns the character associated with this element."""
        return self._char

    @char.setter
    def char(self, character: str):
        """Sets the character associated with this element."""
        self._char = character

    def compare(self, other: 'HuffElement') -> int:
        """
        Compares this HuffElement with another based on frequency.
        Returns 1 if this element's frequency is greater,
        -1 if less, and 0 if equal.
        """
        if self._frequency > other.frequency:
            return 1
        elif self._frequency < other.frequency:
            return -1
        return 0

    def __str__(self) -> str:
        """Returns a string representation of this HuffElement."""
        return f"Char: {self._char} Code: {self._code} Frequency: {self._frequency}"
