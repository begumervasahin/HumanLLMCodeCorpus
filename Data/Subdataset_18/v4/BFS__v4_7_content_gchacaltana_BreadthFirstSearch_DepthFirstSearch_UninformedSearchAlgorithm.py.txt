class UninformedSearchAlgorithm:
    """
    A base class for uninformed search algorithms such as BFS and DFS.
    """

    def __init__(self, list_nodes, target):
        """
        Initialize the search algorithm with a list of nodes and the target node.

        Args:
            list_nodes (list): List of nodes in the graph.
            target (str): The target node to search for.
        """
        self.list_nodes = list_nodes
        self.target = target
        self.create_queue()

    def create_queue(self):
        """
        Initialize the queue with the starting node.
        """
        self.queue = [self.list_nodes[0]]

    def add_to_queue(self, node):
        """
        Add a node to the queue. To be implemented by subclasses.
        """
        pass

    def read_from_queue(self):
        """
        Read and remove the first node from the queue.

        Returns:
            Node: The node at the front of the queue.
        """
        return self.queue.pop(0)

    def get_queue_length(self):
        """
        Get the current length of the queue.

        Returns:
            int: The length of the queue.
        """
        return len(self.queue)

    def get_node_by_name(self, name):
        """
        Get a node from the list by its name.

        Args:
            name (str): The name of the node.

        Returns:
            Node: The node with the given name.
        """
        for node in self.list_nodes:
            if node.name == name:
                return node
        return None

    def validate_queue_length(self):
        """
        Validate if the queue is empty and raise an exception if it is.

        Raises:
            Exception: If the queue is empty.
        """
        if self.get_queue_length() == 0:
            raise Exception("The queue is empty")

    def check_if_target(self, node_name):
        """
        Check if the given node name matches the target.

        Args:
            node_name (str): The name of the node.

        Raises:
            Exception: If the node name matches the target.
        """
        if node_name == self.target:
            raise Exception(f"Target city found: {node_name}")

    def search(self):
        """
        Perform the search. To be implemented by subclasses.
        """
        pass

    def insert_children_into_queue(self, node):
        """
        Insert the children of a node into the queue.

        Args:
            node (Node): The node whose children are to be inserted.
        """
        children = node.get_children()
        for child in children:
            child_node = self.get_node_by_name(child.name)
            if isinstance(child_node, Node):
                self.add_to_queue(child_node)
