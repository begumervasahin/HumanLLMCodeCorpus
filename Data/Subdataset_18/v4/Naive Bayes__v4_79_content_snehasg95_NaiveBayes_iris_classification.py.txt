"""
This script uses the Iris dataset from https://archive.ics.uci.edu/ml/datasets/iris
to classify different types of Iris flowers using the Gaussian Naive Bayes algorithm.
"""

import pandas as pd
import numpy as np
from sklearn import datasets
from sklearn.naive_bayes import GaussianNB
from sklearn.metrics import accuracy_score
from sklearn.model_selection import train_test_split

# Load the Iris dataset
iris = datasets.load_iris()
class_names = iris.target_names

# Create a DataFrame from the Iris data
iris_df = pd.DataFrame(iris.data, columns=iris.feature_names)
iris_df['target'] = iris.target

# Split the data into training and test sets
X_train, X_test, y_train, y_test = train_test_split(iris.data, iris.target, test_size=0.25, random_state=0)

# Uncomment the lines below to print the shapes of the datasets
# print("Dataset shapes:")
# print("Full dataset:", iris_df.shape)
# print("Training data:", X_train.shape)
# print("Test data:", X_test.shape)
# print("Training labels:", y_train.shape)
# print("Test labels:", y_test.shape)

# Initialize and train the Gaussian Naive Bayes classifier
clf = GaussianNB()
clf.fit(X_train, y_train)

# Predict the test set results
y_predict = clf.predict(X_test)

# Calculate and print the accuracy score
score = accuracy_score(y_predict, y_test)
print(f"Accuracy score: {score:.2f}")
