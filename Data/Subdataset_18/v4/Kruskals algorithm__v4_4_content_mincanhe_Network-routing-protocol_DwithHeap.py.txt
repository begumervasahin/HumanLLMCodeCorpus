from collections import defaultdict
from heapq import heappop, heappush

def dijkstra(edges, start, target):
    """
    Implements Dijkstra's algorithm to find the shortest path between two nodes.

    Parameters:
        edges: A list of tuples representing the graph's edges, where each tuple is (from_node, to_node, cost).
        start: The starting node for the path.
        target: The target node for the path.

    Returns:
        A tuple containing the total cost of the shortest path and the path itself as a linked tuple structure.
        If no path exists, returns infinity.
    """
    # Create a graph as an adjacency list
    graph = defaultdict(list)
    for from_node, to_node, cost in edges:
        graph[from_node].append((cost, to_node))

    # Priority queue to store (cost, current_node, path)
    priority_queue = [(0, start, ())]

    # Set to keep track of visited nodes
    visited = set()

    while priority_queue:
        # Pop the node with the smallest cost
        current_cost, current_node, path = heappop(priority_queue)

        if current_node not in visited:
            visited.add(current_node)
            path = (current_node, path)  # Update the path

            # If the target node is reached, return the cost and path
            if current_node == target:
                return (current_cost, path)

            # Explore neighbors
            for neighbor_cost, neighbor in graph.get(current_node, ()):
                if neighbor not in visited:
                    heappush(priority_queue, (current_cost + neighbor_cost, neighbor, path))

    # If the target node is not reachable, return infinity
    return float("inf")

# edges = [
#     ('A', 'B', 1),
#     ('B', 'C', 2),
#     ('A', 'C', 4)
# ]
