import json
from time import perf_counter
from typing import List
from sys import stderr
import spotipy
import spotipy.util as util

# Assuming `chunker` is a custom function for chunking lists
from chunker import chunker
import config

def load_tracks_from_json(json_file_path: str) -> List[dict]:
    """Load tracks from a JSON file."""
    with open(json_file_path) as json_file:
        return json.load(json_file)

def retrieve_spotify_token(username: str, scope: str, client_id: str, client_secret: str, redirect_uri: str) -> str:
    """Retrieve Spotify token using Spotipy's util module."""
    try:
        return util.prompt_for_user_token(username, scope, client_id, client_secret, redirect_uri)
    except Exception as e:
        print(f"Token retrieval failed for {username}: {e}", file=stderr)
        return None

def retrieve_audio_features(sp: spotipy.Spotify, track_ids: List[str]) -> List[dict]:
    """Retrieve audio features for a list of track IDs using Spotify API."""
    audio_features = []
    for chunk in chunker(track_ids, 50):  # Spotify API allows max 100 IDs per request, chunked to 50 for safety
        audio_features.extend(sp.audio_features(chunk))
    return [af for af in audio_features if af]  # Filter out any None results

def main():
    # Load tracks from a JSON file
    tracks = load_tracks_from_json("json/tracks.json")

    # Retrieve Spotify token
    token = retrieve_spotify_token(
        config.username, config.scope, config.client_id, config.client_secret, config.redirect_uri
    )

    if not token:
        return

    # Initialize Spotify client with the token
    sp = spotipy.Spotify(auth=token)

    print("Start retrieving audio features...")
    start_time = perf_counter()

    # Retrieve audio features for the tracks
    audio_features = retrieve_audio_features(sp, [track["id"] for track in tracks])
    elapsed_time = perf_counter() - start_time

    print(f"Retrieved {len(audio_features)} audio features in {elapsed_time:.2f} seconds.")

    # Save the retrieved audio features to a JSON file
    with open("json/audioFeatures.json", "w") as outfile:
        json.dump(audio_features, outfile)

if __name__ == "__main__":
    main()
