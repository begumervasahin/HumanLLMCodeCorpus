class Graph:
    def __init__(self, n):
        """
        Initializes a graph with n vertices (numbered from 0 to n-1) and no edges.
        
        :param n: Number of vertices in the graph
        """
        self.nrVertices = n
        self.nrEdges = 0
        self._dictOut = {i: [] for i in range(n)}

    def isVertex(self, x):
        """
        Checks if x is a vertex in the graph.
        
        :param x: Vertex to check
        :return: True if x is a vertex, False otherwise
        """
        return x in self._dictOut

    def parseAll(self):
        """
        Returns a list of all vertices in the graph.
        
        :return: List of vertices
        """
        return list(self._dictOut.keys())

    def parseNeighbours(self, x):
        """
        Returns a list of neighbors of vertex x.
        
        :param x: Vertex whose neighbors are to be returned
        :return: List of neighbors if x is a vertex, otherwise False
        """
        if self.isVertex(x):
            return self._dictOut[x]
        return False

    def isEdge(self, x, y):
        """
        Checks if there is an edge from vertex x to vertex y.
        
        :param x: Starting vertex
        :param y: Ending vertex
        :return: True if there is an edge from x to y, False otherwise
        """
        if self.isVertex(x):
            return y in self._dictOut[x]
        return False

    def addEdge(self, x, y):
        """
        Adds an edge between vertex x and vertex y.
        
        :param x: Starting vertex
        :param y: Ending vertex
        :return: True if the edge was added, False otherwise
        """
        if not self.isEdge(x, y):
            self._dictOut[x].append(y)
            self._dictOut[y].append(x)
            self.nrEdges += 1
            return True
        return False

    def getNrVertices(self):
        """
        Returns the number of vertices in the graph.
        
        :return: Number of vertices
        """
        return self.nrVertices

    def getNrEdges(self):
        """
        Returns the number of edges in the graph.
        
        :return: Number of edges
        """
        return self.nrEdges

    def getDegree(self, x):
        """
        Returns the degree of vertex x.
        
        :param x: Vertex whose degree is to be returned
        :return: Degree of the vertex if it exists, otherwise False
        """
        if self.isVertex(x):
            return len(self._dictOut[x])
        return False
