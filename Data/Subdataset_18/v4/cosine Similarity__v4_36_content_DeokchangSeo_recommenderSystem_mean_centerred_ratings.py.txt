"""
This module calculates mean-centered ratings from a dataset using Surprise library and numpy.
"""
from surprise import Dataset, Reader
from numpy import zeros
from six import iteritems

def mean_centered_ratings(dataset_path):
    # Load dataset
    file_path = dataset_path
    reader = Reader(line_format='user item rating', rating_scale=(1, 7), sep='\t')
    data = Dataset.load_from_file(file_path, reader)
    trainset = data.build_full_trainset()
    
    # Extract necessary information from the dataset
    number_users = trainset.n_users
    number_items = trainset.n_items
    user_ratings = trainset.ur
    
    # Initialize matrices for user-item ratings and average calculations
    u_matrix = zeros((number_users, number_items), dtype=float)
    avg_rating = zeros((number_users, 3), dtype=float)
    avg_matrix = zeros((number_users, number_items), dtype=float)
    
    # Calculate total ratings per user and sum for average calculation
    for user, ratings in iteritems(user_ratings):
        for item, rating in ratings:
            if rating != 0:
                u_matrix[user, item] += rating
                avg_rating[user, 1] += rating
                avg_rating[user, 2] += 1
    
    # Print original user ratings matrix
    print('Original user ratings:\n', u_matrix, '\n')
    
    # Compute average rating per item per user
    for user, ratings in iteritems(user_ratings):
        for item, rating in ratings:
            if rating != 0:
                avg_matrix[user, item] += avg_rating[user, 1] / avg_rating[user, 2]
    
    # Compute mean-centered ratings matrix
    mean_centered_matrix = u_matrix - avg_matrix
    return mean_centered_matrix

# Example usage
file_path = 'data/table2.1_ratings.txt'
mean_centered = mean_centered_ratings(file_path)
print('The mean centered ratings:\n', mean_centered)
