import argparse
from util import labels_to_key, parse_federalist_papers, labels_to_y, find_zero_rule_class, apply_zero_rule
from sklearn.metrics import accuracy_score
from sklearn.model_selection import train_test_split

def main(data_file, random_seed):
    """
    Main function to process the Federalist Papers dataset, split it into training and testing sets,
    apply the zero rule classifier, and evaluate its accuracy.

    Parameters:
    - data_file (str): Path to the dataset file.
    - random_seed (int): Random seed for reproducibility of the dataset split.
    """
    print(f"Data file: {data_file}")

    # Parse the dataset
    authors, essays, essay_ids = parse_federalist_papers(data_file)
    num_essays = len(essays)
    print(f"1: Working with {num_essays} essays")

    # Convert author labels to keys
    author_key = labels_to_key(authors)
    print(f"2: Author key: {author_key}")

    # Convert labels to numeric format
    y = labels_to_y(authors, author_key)
    assert y.size == len(authors), f"Size of label array (y.size) must equal number of labels {len(authors)}"

    # Split the data into training and testing sets
    train_X, test_X, train_y, test_y = train_test_split(essays, y, test_size=0.3, random_state=random_seed)
    print(f"{len(train_X)} in train; {len(test_X)} in test")

    # Find the most frequent class in the training labels
    most_frequent_class = find_zero_rule_class(train_y)
    reverse_author_key = {v: k for k, v in author_key.items()}
    print(f"3: The most frequent class is {reverse_author_key[most_frequent_class]}")

    # Apply the zero rule classifier to the test set
    test_predictions = apply_zero_rule(test_X, most_frequent_class)
    print(f"4: Zero rule predictions on held-out data: {test_predictions}")

    # Calculate and print the accuracy of the zero rule classifier
    test_accuracy = accuracy_score(test_y, test_predictions)
    print(f"5: Accuracy of zero rule: {test_accuracy:.3f}")

if __name__ == '__main__':
    # Set up argument parser
    parser = argparse.ArgumentParser(description='Test supervised learning utilities')
    parser.add_argument('--path', type=str, default="federalist_dev.json",
                        help='Path to author dataset')
    parser.add_argument('--seed', type=int, default=7,
                        help='Random seed for dataset split')

    # Parse arguments and call the main function
    args = parser.parse_args()
    main(args.path, args.seed)
