from selenium import webdriver
import re

# Function to find and process results
def find_results(pattern, start, end):
    global detained
    base_htno = pattern + "000"
    
    for i in range(start, end + 1):
        try:
            # Format the hall ticket number correctly
            htno = base_htno[:-len(str(i))] + str(i)
            
            # Enter hall ticket number and submit
            browser.find_element_by_name('htno').send_keys(htno)
            browser.find_element_by_name('Submit').click()
            
            # Extract details from the results page
            details_table = browser.find_element_by_id('AutoNumber3').text     
            pin_area = re.search(r"\d{12}", details_table).span()
            pin_no = details_table[pin_area[0]:pin_area[1]]
            name_span = re.search(r"Name [A-Z .]*\s", details_table).span()      
            name = details_table[name_span[0]:name_span[1]]
            
            file.writelines("\n\n-------------------------------------------------------------\n")
            file.writelines(f"PIN NO : {pin_no}\t{name}\n")
            file.writelines("-------------------------------------------------------------\n")
            
            # Extract and write marks details
            marks_table = browser.find_element_by_id('AutoNumber4').text       
            marks_list = marks_table.splitlines()
            for subject in marks_list[2:]:
                subject_code = re.search(r"\d{3}", subject).group()
                subject_name = re.findall(r"[A-Z -.&]*", subject)[4]
                subject_grade_secured = subject[-1]
                gpa_scored = re.findall(r"\d+", subject)[-1]
                file.writelines(f"{subject_code}\t{subject_name}\t\t{gpa_scored}\n")
            
            # Extract and write result details
            result_table = browser.find_element_by_id('AutoNumber5').text      
            result_lines = result_table.splitlines()
            if "DETAINED" in result_lines[-1]:
                detained += 1
            file.writelines(f"RESULT : {result_lines[-1]}\n")
            file.writelines("-------------------------------------------------------------\n\n\n\n")
        
        except Exception as e:
            print(e)
            file.writelines("---------SKIPPING ONE PIN HERE-------------\n\n\n")

# Main script execution
if __name__ == "__main__":
    file_name = input("ENTER A FILE NAME TO BE SAVED: ")
    pattern = input("ENTER PATTERN: ")
    
    with open(f"{file_name}.txt", "w") as file:
        PATH = "/usr/lib/chromium-browser/chromedriver"
        browser = webdriver.Chrome(executable_path=PATH)
        url = "https://www.osmania.ac.in/res07/20190855.jsp"
        browser.get(url)
        
        detained = 0
        find_results(pattern, 1, 59)
        find_results(pattern, 301, 311)
        
        file.seek(0, 0)
        file.writelines(f"NO. OF STUDENTS DETAINED: {detained}\n")
        
        browser.quit()
