# Definition for a Node.
class Node:
    def __init__(self, val, children=[]):
        """
        Initialize a Node.

        :param val: The value of the node.
        :param children: A list of child nodes.
        """
        self.val = val
        self.children = children

class Solution:
    def levelOrder(self, root):
        """
        Perform level order traversal on an N-ary tree.

        :param root: The root node of the N-ary tree.
        :return: A list of lists, where each list contains the values of nodes at that level.
        
        Example:
              1
           /  |  \
          3   2   4
         / \
        5   6 

        Output: [[1], [3, 2, 4], [5, 6]]
        """
        if not root:
            return []

        result = []
        current_level = [root]

        while current_level:
            level_values = []
            next_level = []
            for node in current_level:
                level_values.append(node.val)
                next_level.extend(node.children)
            result.append(level_values)
            current_level = next_level

        return result

# Example usage
if __name__ == "__main__":
    # Create an example N-ary tree
    root = Node(1, [
        Node(3, [
            Node(5),
            Node(6)
        ]),
        Node(2),
        Node(4)
    ])

    solution = Solution()
    print("Level Order Traversal:", solution.levelOrder(root))
