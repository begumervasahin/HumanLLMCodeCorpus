def bubble_sort(arr):
    while True:
        swapped = False
        for i in range(len(arr) - 1):
            if arr[i] > arr[i + 1]:
                arr[i], arr[i + 1] = arr[i + 1], arr[i]
                swapped = True
        if not swapped:
            break
    return arr

# Testing the bubble_sort function with different inputs
print(bubble_sort([1, 2, 3, 4, 5, 6]))  # Already sorted
print(bubble_sort([2, 1, 4, 3, 6, 5]))  # Partially unsorted
print(bubble_sort([6, 5, 4, 3, 2, 1]))  # Completely unsorted
