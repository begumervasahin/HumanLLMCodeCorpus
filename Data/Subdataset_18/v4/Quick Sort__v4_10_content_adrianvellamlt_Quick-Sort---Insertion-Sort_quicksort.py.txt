import insertionsort as ins

class QuickSort:
    def __init__(self):
        pass

    def get_median_of_three_index(self, array, start, end):
        def get_median_index(start, length):
            return start + (length // 2) if length % 2 != 0 else start + (length // 2) - 1

        def median(i, j, k):
            return j if i < j < k or k < j < i else k if j < k < i or i < k < j else i

        median_index = get_median_index(start, end - start)
        start_value, end_value, median_value = array[start], array[end - 1], array[median_index]
        return median(start_value, end_value, median_value)

    def partition(self, array, start, end, use_median_of_three):
        median_index = self.get_median_of_three_index(array, start, end) if use_median_of_three else start
        pivot = array[median_index]
        array[median_index], array[start] = array[start], array[median_index]

        i = start + 1
        for j in range(start + 1, end):
            if array[j] < pivot:
                array[j], array[i] = array[i], array[j]
                i += 1

        array[start], array[i - 1] = array[i - 1], array[start]
        return i - 1

    def sort(self, arr, use_median_of_three):
        stack = [0, len(arr)]
        while stack:
            end = stack.pop()
            start = stack.pop()
            if end - start < 2:
                continue

            p = self.partition(arr, start, end, use_median_of_three)
            stack.extend([p + 1, end, start, p])

    def quick_sort(self, arr):
        if len(arr) <= 1:
            return arr
        self.sort(arr, use_median_of_three=False)
        return arr

    def sort_median_of_three(self, arr):
        if len(arr) <= 1:
            return arr
        self.sort(arr, use_median_of_three=True)
        return arr

    def optimised_sort(self, arr):
        if len(arr) > 16:
            return self.sort_median_of_three(arr)
        return ins.InsertionSort().sort(arr)

# Example usage
if __name__ == "__main__":
    sorter = QuickSort()
    arr = [3, 6, 8, 10, 1, 2, 1]
    print("Original array:", arr)
    print("Sorted array (QuickSort):", sorter.quick_sort(arr[:]))
    print("Sorted array (Median of Three):", sorter.sort_median_of_three(arr[:]))
    print("Optimized Sorted array:", sorter.optimised_sort(arr[:]))
