import matplotlib.pyplot as plt
import numpy as np

def get(signal_chunks, showtemplate):
    n_signal_chunks = len(signal_chunks)
    signal_average = np.mean(signal_chunks, axis=0)
    
    # Calculate the standard deviation for the first quarter of the signal chunks
    std_quarter = np.mean(np.std(signal_chunks[:int(n_signal_chunks / 4)], axis=0), axis=0)
    
    # Create a test signal and manipulate the first half
    test = signal_average.copy()
    test[:len(test) // 2] = abs(test[:len(test) // 2])
    
    # Calculate upper and lower limits
    upper_limit = abs(test) + (5 * std_quarter)
    lower_limit = signal_average - (7 * std_quarter)
    
    # Expand the upper limit signal
    expanded_upper_limit = np.repeat(upper_limit, 4)
    start_idx = len(expanded_upper_limit) // (4 // 1.5) + int(4 * 1.5)
    upper_limit = expanded_upper_limit[start_idx:start_idx + len(upper_limit)]
    
    if showtemplate:
        plt.plot(signal_average, label='Signal Average')
        plt.plot(upper_limit, 'r--', label='Upper Limit')
        plt.plot(lower_limit, 'r--', label='Lower Limit')
        plt.legend()
        plt.show()
    
    # Filter the signal chunks based on the upper and lower limits
    valid_signals = []
    for i in range(n_signal_chunks):
        curr_trial = signal_chunks[i]
        is_within_limits = np.all((curr_trial > lower_limit) & (curr_trial < upper_limit))
        valid_signals.append(is_within_limits)
    
    valid_signals = np.array(valid_signals)
    valid_indices = np.where(valid_signals)[0]
    
    print(f'{n_signal_chunks} signals have been reduced to {len(valid_indices)}.')
    return valid_indices
