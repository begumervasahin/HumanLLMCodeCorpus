from heap import Min_Heap
from graph import Graph
from union_find import Set

def MST_Prim(graph):
    """
    Applies Prim's algorithm to compute a minimum spanning tree.
    """
    # Create an empty graph for the minimum spanning tree
    minSpanTree = Graph(numVertices=graph.numVertices(), numEdges=0, weightRange=0)
    
    # Initialize a priority queue (min-heap) and a parent dictionary
    queue = Min_Heap()
    parent = {}

    # Set the initial key values for all vertices to infinity and insert them into the queue
    for vertex in graph.vertices():
        parent[vertex] = None
        queue.insert(float('inf'), vertex)

    while queue:
        # Extract the vertex with the minimum key value
        key, vertex = queue.extract_min()
        predecessor = parent[vertex]

        # Add the edge to the minimum spanning tree if it exists
        if predecessor is not None:
            weight = graph.weight[(predecessor, vertex)]
            minSpanTree.addUndirectedEdge(predecessor, vertex, weight)

        # Update the key values of the adjacent vertices
        for neighbor in graph.adjacent[vertex]:
            if neighbor in queue:
                new_key = graph.weight[(vertex, neighbor)]
                if new_key < queue.key(neighbor):
                    queue.decrease_key(neighbor, new_key)
                    parent[neighbor] = vertex

    return minSpanTree

def MST_Kruskal(graph):
    """
    Applies Kruskal's algorithm to compute a minimum spanning tree.
    """
    # Create a disjoint set for each vertex
    component = {vertex: Set(vertex) for vertex in graph.vertices()}
    
    # Create an empty graph for the minimum spanning tree
    minSpanTree = Graph(numVertices=graph.numVertices(), numEdges=0, weightRange=0)

    # Sort the edges by weight and process them
    for (u, v) in sorted(graph.edges(), key=lambda e: graph.weight[e]):
        if component[u].findSet() != component[v].findSet():
            # Union the sets and add the edge to the minimum spanning tree
            component[u].union(component[v])
            minSpanTree.addUndirectedEdge(u, v, graph.weight[(u, v)])

    return minSpanTree

def test(numVertices, numEdges, weightRange):
    """
    Checks whether Prim's and Kruskal's algorithms produce spanning trees with the same weight.
    """
    # Generate a random graph
    graph = Graph(numVertices=numVertices, numEdges=numEdges, weightRange=weightRange, directed=False)

    # Compute the MST using Prim's algorithm and calculate its weight
    mst = MST_Prim(graph)
    primsWeight = sum(graph.weight[e] for e in mst.edges())

    # Compute the MST using Kruskal's algorithm and calculate its weight
    mst = MST_Kruskal(graph)
    kruskalsWeight = sum(graph.weight[e] for e in mst.edges())

    return primsWeight == kruskalsWeight

def runTests(numVertices, numEdges, weightRange, numTests):
    """
    Runs a sequence of tests to check if Prim's and Kruskal's algorithms produce spanning trees with the same weight.
    """
    return all(test(numVertices, numEdges, weightRange) for _ in range(numTests))
