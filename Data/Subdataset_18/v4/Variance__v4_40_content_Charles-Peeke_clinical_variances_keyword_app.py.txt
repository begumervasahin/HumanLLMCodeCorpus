import os
import time
import psycopg2
import pandas as pd
import matplotlib.pyplot as plt
from dotenv import load_dotenv

# Global variables
conn = None
df_keywords = None

def create_env_vars():
    """Load environment variables for database connection."""
    global hostname, port, database, username, password
    hostname = os.getenv('hostname')
    port = os.getenv('port')
    database = os.getenv('database')
    username = os.getenv('username')
    password = os.getenv('password')

def get_keywords():
    """Fetch keywords from the database and store them in a DataFrame."""
    print('Computing keywords...')
    start_time = time.time()
    global df_keywords
    query = '''
    SELECT studies.nct_id, downcase_name 
    FROM studies 
    RIGHT JOIN keywords 
    ON studies.nct_id = keywords.nct_id
    '''
    df_keywords = pd.read_sql(query, con=conn)
    print(f'Selected keywords: {time.time() - start_time:.2f} sec')

def collect_keyword_variances():
    """Collect and plot keyword variances."""
    print('Collecting Keyword variances...')
    start_time = time.time()
    key_dict = {}

    for nct_id, keyword in zip(df_keywords['nct_id'], df_keywords['downcase_name']):
        key_dict[nct_id] = key_dict.get(nct_id, 0) + 1

    print(f'Keyword variances Calculated: {time.time() - start_time:.2f} sec')

    # Plotting the results
    ids = key_dict.keys()
    counts = key_dict.values()
    plt.bar(ids, counts)
    plt.xlabel('NCT ID')
    plt.ylabel('Keyword Count')
    plt.title('Keyword Variances')
    plt.show()

if __name__ == '__main__':
    load_dotenv()
    create_env_vars()
    conn = psycopg2.connect(host=hostname, database=database, user=username, password=password)
    
    get_keywords()
    collect_keyword_variances()
