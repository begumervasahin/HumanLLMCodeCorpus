import numpy as np
import matplotlib.pyplot as plt
from matplotlib.backends.backend_pdf import PdfPages
import sys

# Check if the correct number of arguments is provided
if len(sys.argv) != 5:
    print(f"{sys.argv[0]} takes 4 arguments. Not {len(sys.argv) - 1}")
    sys.exit()

# Parse command line arguments
first = sys.argv[1]
second = sys.argv[2]
third = sys.argv[3]
fourth = sys.argv[4]

# Load data
Xt = np.genfromtxt(first, delimiter=',')
y = np.genfromtxt(second, autostrip=True)

# Get dimensions of the input data
m, n = Xt.shape
unique_labels = set(y)
mu_global = np.mean(Xt, axis=0)

# Initialize within-class scatter and between-class scatter matrices
W_class = np.zeros((n, n))
B_class = np.zeros((n, n))

# Compute within-class scatter and between-class scatter matrices
for c in unique_labels:
    B1 = np.zeros((n, n))
    BT1 = np.zeros((n, n))
    s1 = np.zeros((1, n))
    C1 = np.zeros((n, n))
    m1 = 0

    # Accumulate class-specific data
    for i, xt in enumerate(Xt):
        if y[i] == c:
            B1 += np.dot(xt.T, xt)
            s1 += xt
            m1 += 1

    mu1 = s1 / m1

    # Compute scatter matrices
    for i, xt in enumerate(Xt):
        if y[i] == c:
            ct = xt - mu1
            bt = mu1 - mu_global
            C1 += np.dot(ct.T, ct)
            BT1 += np.dot(bt.T, bt)

    B_class += BT1

# Perform eigen decomposition on between-class scatter matrix
evals, evecs = np.linalg.eigh(B_class)
idx = np.argsort(evals)[::-1]
evecs = evecs[:, idx]
evals = evals[idx]

# Select the top r eigenvectors
r = 2
V_r = evecs[:, :r]

# Project the data onto the new feature space
result = np.dot(Xt, V_r)
x1 = result[:, 0]
x2 = result[:, 1]

# Plot the results
fig, ax = plt.subplots()
ax.scatter(x1, x2)
plt.title("Maximize between class scatter")
plt.xlabel("PC1")
plt.ylabel("PC2")

# Save the plot as a PDF
plotname = 'scatter_2_plot'
pdf = PdfPages(f'{plotname}.pdf')
pdf.savefig(fig)
pdf.close()

# Save the transformation matrix and the result
np.savetxt(third, V_r.T, delimiter=',')
np.savetxt(fourth, result, delimiter=',')
