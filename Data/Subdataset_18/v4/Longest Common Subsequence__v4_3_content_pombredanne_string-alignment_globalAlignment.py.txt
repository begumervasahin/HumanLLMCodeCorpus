def globalAlignmentScore(seq1, seq2):
    MATCH = 5
    MISMATCH = -4
    GAP = -6

    numRows = len(seq2) + 1
    numCols = len(seq1) + 1

    # Initialize costs and directions tables
    costs = createTable(numRows, numCols, 0)
    directions = createTable(numRows, numCols, "A")

    # Fill the first row and first column with initial gap penalties
    for col in range(numCols):
        costs[0][col] = GAP * col
        directions[0][col] = "L"
    
    for row in range(1, numRows):
        costs[row][0] = GAP * row
        directions[row][0] = "T"
    
    directions[0][0] = 'F'

    # Fill the rest of the table
    for row in range(1, numRows):
        for col in range(1, numCols):
            top = costs[row-1][col] + GAP
            left = costs[row][col-1] + GAP
            diag = costs[row-1][col-1] + (MATCH if seq1[col-1] == seq2[row-1] else MISMATCH)

            max_val = max(top, left, diag)
            costs[row][col] = max_val

            if max_val == left:
                directions[row][col] = "L"
            elif max_val == diag:
                directions[row][col] = "D"
            else:
                directions[row][col] = "T"

    # Align sequences and save to file
    align(directions, seq1, seq2, "alignment.txt")
    return costs[numRows-1][numCols-1]

def createTable(numRows, numCols, initialValue):
    return [[initialValue for _ in range(numCols)] for _ in range(numRows)]

def printTable(table, filename):
    with open(filename, 'w') as file:
        for row in table:
            file.write("\t".join(map(str, row)) + "\n")

def align(directions, seq1, seq2, filename):
    with open(filename, 'w') as file:
        row, col = len(directions) - 1, len(directions[0]) - 1
        topSeq, botSeq = "", ""

        while directions[row][col] != "F":
            if directions[row][col] == "T":
                topSeq = "-" + topSeq
                botSeq = seq2[row-1] + botSeq
                row -= 1
            elif directions[row][col] == "L":
                topSeq = seq1[col-1] + topSeq
                botSeq = "-" + botSeq
                col -= 1
            elif directions[row][col] == "D":
                topSeq = seq1[col-1] + topSeq
                botSeq = seq2[row-1] + botSeq
                row -= 1
                col -= 1

        for i in range(0, len(topSeq), 50):
            file.write(topSeq[i:i+50] + "\n")
            file.write(botSeq[i:i+50] + "\n")
            file.write("\n")

# Example usage
s1 = "AGCGTCTA"
s2 = "TGCATCTCG"
optimalScore = globalAlignmentScore(s1, s2)
print(s1)
print(s2)
print("Global alignment score: " + str(optimalScore))
