from time import time

def lcs(X, Y):
    m = len(X)
    n = len(Y)
    L = [[0] * (n + 1) for _ in range(m + 1)]

    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if X[i - 1] == Y[j - 1]:
                L[i][j] = L[i - 1][j - 1] + 1
            else:
                L[i][j] = max(L[i - 1][j], L[i][j - 1])

    print("\nL table")
    for row in L:
        print("\t".join(map(str, row)))

    index = L[m][n]
    lcs_str = [""] * (index + 1)
    lcs_str[index] = ""

    i, j = m, n
    while i > 0 and j > 0:
        if X[i - 1] == Y[j - 1]:
            lcs_str[index - 1] = X[i - 1]
            i -= 1
            j -= 1
            index -= 1
        elif L[i - 1][j] >= L[i][j - 1]:
            i -= 1
        else:
            j -= 1

    print(f"\nLength of LCS is {L[m][n]}")
    print(f"LCS of {X} and {Y} is {''.join(lcs_str)}")

el_time = []
mn = []

X = ["AGGTAB", "ABRAC", "BACDB", "AYUSH", "KAMLESH", "SUNIL", "BIBASH", "ARAJU", "MANASI", "DEEPESH"]
Y = ["GXTXAYB", "YABBAD", "BDCB", "SHAHA", "MAHES", "UNATTI", "SHOWIN", "ARUNADHA", "ANSI", "DISH"] 

for i in range(10):
    start_time = time()
    lcs(X[i], Y[i])
    end_time = time()
    mn.append(len(X[i]) * len(Y[i]))
    el_time.append(end_time - start_time)

print("\nmn\tTime")
for mn_val, time_val in zip(mn, el_time):
    print(f"{mn_val}\t{time_val:.5f}")
