def quick_sort(A, verbose=0, desc=0):
    def partition(low, high):
        pivot = A[low]
        left = low + 1
        right = high

        while True:
            while left <= right and A[left] <= pivot:
                left += 1
            while left <= right and A[right] > pivot:
                right -= 1
            if left <= right:
                A[left], A[right] = A[right], A[left]
                if verbose == 2:
                    print("  sub:", pivot, left, right, " :: ", A)
            else:
                break

        A[low], A[right] = A[right], A[low]
        return right

    def quicksort_recursive(low, high):
        if low < high:
            pivot_index = partition(low, high)
            if verbose:
                print("iter :", pivot_index, low, high, " :: ", A)
            quicksort_recursive(low, pivot_index - 1)
            quicksort_recursive(pivot_index + 1, high)

    quicksort_recursive(0, len(A) - 1)
    
    if desc:
        A.reverse()
    
    return A
