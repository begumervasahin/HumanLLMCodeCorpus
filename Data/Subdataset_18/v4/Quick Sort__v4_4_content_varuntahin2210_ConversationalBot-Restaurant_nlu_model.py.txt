from rasa_nlu.training_data import load_data
from rasa_nlu.config import RasaNLUModelConfig
from rasa_nlu.model import Trainer, Metadata, Interpreter
from rasa_nlu import config
from rasa_nlu.components import ComponentBuilder

# Initialize the ComponentBuilder with caching enabled
builder = ComponentBuilder(use_cache=True)

def train_nlu(data_path, config_path, model_dir):
    """
    Train the Rasa NLU model.

    Parameters:
    data_path (str): Path to the training data.
    config_path (str): Path to the config file.
    model_dir (str): Directory to save the trained model.
    """
    # Load training data
    training_data = load_data(data_path)
    
    # Create a Trainer with the specified config
    trainer = Trainer(config.load(config_path), builder)
    
    # Train the model
    trainer.train(training_data)
    
    # Persist the model to the specified directory
    model_directory = trainer.persist(model_dir, fixed_model_name='restaurantnlu')

def run_nlu(model_dir='./models/nlu/default/restaurantnlu'):
    """
    Run the Rasa NLU interpreter.

    Parameters:
    model_dir (str): Directory where the trained model is stored.
    """
    # Load the interpreter from the model directory
    interpreter = Interpreter.load(model_dir, builder)
    
    # Parse a sample text
    result = interpreter.parse("can you please suggest food")
    print(result)

if __name__ == '__main__':
    # Train the NLU model
    train_nlu('./data/data.json', 'config_spacy.json', './models/nlu')
    
    # Run the NLU interpreter
    run_nlu()
