def combination_sum(candidates, target):
    """
    Find all unique combinations of candidate numbers that sum up to the target.
    
    Args:
        candidates (list of int): List of candidate numbers (without duplicates).
        target (int): The target sum.
    
    Returns:
        list of list of int: A list of unique combinations that sum up to the target.
    """
    # Sort candidates to streamline the combination process
    candidates.sort()
    
    # Initialize result list and temporary combination list
    result_list = []
    current_combination = []
    
    # Start the recursive search for combinations
    _find_combinations(candidates, target, 0, current_combination, result_list)
    
    return result_list

def _find_combinations(candidates, target, start_index, current_combination, result_list):
    """
    Recursive helper function to find combinations that sum up to the target.
    
    Args:
        candidates (list of int): List of candidate numbers.
        target (int): Remaining sum needed to reach the target.
        start_index (int): The index to start from in the candidates list.
        current_combination (list of int): The current sequence of numbers being considered.
        result_list (list of list of int): List where valid combinations are stored.
    """
    for i in range(start_index, len(candidates)):
        num = candidates[i]
        
        # If the number exceeds the target, skip further checks
        if num > target:
            break
        
        # Add the number to the current combination
        current_combination.append(num)
        
        # Recur with updated target; allow repeated use of the same number
        if num < target:
            _find_combinations(candidates, target - num, i, current_combination, result_list)
        else:
            # If the number matches the target, store the combination
            result_list.append(list(current_combination))
        
        # Backtrack by removing the last added number
        current_combination.pop()

# Example usage
if __name__ == "__main__":
    candidates = [2, 3, 6, 7]
    target = 7
    combinations = combination_sum(candidates, target)
    print("Unique combinations:", combinations)
