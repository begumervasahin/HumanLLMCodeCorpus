Implementation of an edge priority queue using a d-ary heap.
"""

from edge import Edge  # Assuming Edge is a class with a 'weight' attribute

class Heapq:
    def __init__(self, d):
        """Initialize the d-ary heap."""
        self.data = []
        self.d = d

    def make_heap(self, lst):
        """Convert a list into a heap."""
        self.data = lst
        for k in range(self.size() - 1, -1, -1):
            self.siftdown(k)

    def find_min(self):
        """Return the minimum element in the heap."""
        return self.data[0] if self.data else None

    def delete(self, k):
        """Delete the element at index k."""
        if k == self.size() - 1:
            self.data.pop()
        else:
            last_item = self.data.pop()
            self.data[k] = last_item
            if last_item.weight < self.data[k].weight:
                self.siftup(k)
            else:
                self.siftdown(k)

    def delete_min(self):
        """Delete the minimum element (root) from the heap."""
        if self.size() > 0:
            self.delete(0)

    def size(self):
        """Return the size of the heap."""
        return len(self.data)

    def siftup(self, k):
        """Perform sift-up operation to maintain heap property."""
        while k > 0:
            parent_index = (k - 1) // self.d
            if self.data[parent_index].weight <= self.data[k].weight:
                break
            self.data[k], self.data[parent_index] = self.data[parent_index], self.data[k]
            k = parent_index

    def siftdown(self, k):
        """Perform sift-down operation to maintain heap property."""
        while True:
            min_child_index = self.minchild(k)
            if min_child_index is None or self.data[k].weight <= self.data[min_child_index].weight:
                break
            self.data[k], self.data[min_child_index] = self.data[min_child_index], self.data[k]
            k = min_child_index

    def minchild(self, k):
        """Return the index of the smallest child of k, or None if no children exist."""
        start_child = self.d * k + 1
        end_child = min(self.d * k + self.d, self.size() - 1)
        if start_child >= self.size():
            return None

        min_child_index = start_child
        for i in range(start_child + 1, end_child + 1):
            if self.data[i].weight < self.data[min_child_index].weight:
                min_child_index = i

        return min_child_index

    def __repr__(self):
        """String representation of the heap."""
        return str(self.data)
