class DisjointSets:
    def __init__(self, elements):
        """
        Initializes the disjoint set structure.

        Parameters:
            elements: A list or set of elements to be managed in disjoint sets.
        """
        self.elements = list(elements)
        self.parent = list(range(len(elements)))  # Initially, each element is its own parent
        self.rank = [1] * len(elements)  # Initialize ranks with 1 for each element

    def find(self, x):
        """
        Finds the representative (root) of the set containing element x with path compression.

        Parameters:
            x: The element whose set representative is to be found.

        Returns:
            The index of the representative element in the disjoint set.
        """
        idx = root = self.elements.index(x)

        # Find the root of the set containing x
        while self.parent[root] != root:
            root = self.parent[root]

        # Path compression: make all nodes on the path point directly to the root
        while idx != root:
            next_idx = self.parent[idx]
            self.parent[idx] = root
            idx = next_idx

        return root

    def merge(self, set_a, set_b):
        """
        Merges the sets containing set_a and set_b. Assumes that set_a and set_b are different.

        Parameters:
            set_a: The representative (root) of the first set.
            set_b: The representative (root) of the second set.
        """
        # Merge by rank
        if self.rank[set_a] == self.rank[set_b]:
            self.rank[set_a] += 1
            self.parent[set_b] = set_a
        elif self.rank[set_a] > self.rank[set_b]:
            self.parent[set_b] = set_a
        else:
            self.parent[set_a] = set_b
