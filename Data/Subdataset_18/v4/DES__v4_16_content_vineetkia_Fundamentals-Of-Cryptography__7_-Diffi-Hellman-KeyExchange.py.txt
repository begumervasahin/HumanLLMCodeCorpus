def power(base, exponent, modulus):
    """
    Compute (base^exponent) % modulus.

    Parameters
    ----------
    base : int
        The base number.
    exponent : int
        The exponent.
    modulus : int
        The modulus.

    Returns
    -------
    int
        The result of (base^exponent) % modulus.
    """
    return pow(base, exponent, modulus)

def display_values(label, value):
    """
    Display a label and its corresponding value.

    Parameters
    ----------
    label : str
        The label to display.
    value : any
        The value to display.
    """
    print(f"{label}: {value}\n")

def main():
    # Given values
    G = 9   # Base number
    a = 4   # Private key A
    b = 3   # Private key B
    P = 29  # Modulus

    # Display given values
    display_values("P Value", P)
    display_values("G Value", G)
    display_values("A Private KEY", a)

    # Calculate public keys
    x = power(G, a, P)
    display_values("The private key B", b)
    y = power(G, b, P)

    # Calculate secret keys
    ka = power(y, a, P)
    kb = power(x, b, P)

    # Display secret keys
    display_values("A Secret Key", ka)
    display_values("B Secret Key", kb)

if __name__ == "__main__":
    main()
