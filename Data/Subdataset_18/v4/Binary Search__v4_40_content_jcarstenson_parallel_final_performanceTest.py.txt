import sys
import random
from time import time
from BST import BST
from linkedList import LinkedList as LL
from willowCuttingHashTable import WillowCuttingHashTable
from chainedHashTable import ChainedHashTable

def main():
    # Increase the recursion limit to handle deep recursions in BSTs
    sys.setrecursionlimit(10000)

    # Validate the command-line arguments
    if len(sys.argv) != 4:
        print("Invalid Syntax\nUsage: performanceTest.py filename chainedLoad willowLoad")
        sys.exit(1)

    # Parse command-line arguments
    filename = sys.argv[1]
    chainedLoad = int(sys.argv[2])
    willowLoad = int(sys.argv[3])

    # Read the file and split its content into words
    with open(filename, "r") as f:
        data = f.read().split()

    size = len(data)
    print(f"Using {size} words")

    # Initialize the hash tables with the given load factors
    chained = ChainedHashTable(12, chainedLoad, BST)
    willow = WillowCuttingHashTable(willowLoad, 0, size)

    # Prepare shuffled keys for random access timing
    shuffled_keys = list(range(size))
    random.shuffle(shuffled_keys)

    # Measure the time taken to insert (put) words into both hash tables
    print("Timing put")
    
    start = time()
    for i in range(size):
        chained[i] = data[i]
    chained_put_time = time() - start

    start = time()
    for i in range(size):
        willow[i] = data[i]
    willow_put_time = time() - start

    print(f"Willow time: {willow_put_time:.6f} Chained time: {chained_put_time:.6f}")

    # Measure the time taken to randomly access (get) words from both hash tables
    print("Timing get (random)")

    start = time()
    for i in shuffled_keys:
        x = chained[i]
    chained_get_random_time = time() - start

    start = time()
    for i in shuffled_keys:
        x = willow[i]
    willow_get_random_time = time() - start

    print(f"Willow time: {willow_get_random_time:.6f} Chained time: {chained_get_random_time:.6f}")

    # Measure the time taken to access (get) words in order from both hash tables
    print("Timing get (in order)")

    start = time()
    for i in range(size):
        x = chained[i]
    chained_get_order_time = time() - start

    start = time()
    for i in range(size):
        x = willow[i]
    willow_get_order_time = time() - start

    print(f"Willow time: {willow_get_order_time:.6f} Chained time: {chained_get_order_time:.6f}")

if __name__ == "__main__":
    main()
