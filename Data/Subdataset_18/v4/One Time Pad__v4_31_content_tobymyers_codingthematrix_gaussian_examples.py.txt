from mat import Mat
from GF2 import one

def create_matrix(domain, function_set):
    """
    Create a matrix using the given domain and function set.
    
    Parameters:
    domain (tuple): A tuple containing two sets representing the row and column domains.
    function_set (set): A set of tuples representing the non-zero entries of the matrix.
    
    Returns:
    Mat: The created matrix.
    """
    function_dict = {k: one for k in function_set}
    return Mat(domain, function_dict)

# Define the domains for matrices A and B
Ad = (set([1, 2, 3, 4]), set(['A', 'B', 'C', 'D']))
Bd = (set([1, 2, 3, 4]), set(['A', 'B', 'C', 'D']))

# Define the non-zero entries for matrices A and B
Af_set = {(2, 'A'), (2, 'C'), (2, 'D'), (1, 'C'), (1, 'D'), 
          (3, 'A'), (3, 'D'), (4, 'A'), (4, 'B'), (4, 'C'), (4, 'D')}
Bf_set = {(1, 'A'), (1, 'B'), (2, 'A'), (2, 'C'), (3, 'B'), 
          (3, 'C'), (3, 'D'), (4, 'A')}

# Create matrices A and B
A = create_matrix(Ad, Af_set)
B = create_matrix(Bd, Bf_set)

# Print the matrices for verification
print("Matrix A:")
print(A)
print("\nMatrix B:")
print(B)
