import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from scipy.cluster.hierarchy import dendrogram, linkage
from sklearn.cluster import AgglomerativeClustering

def load_dataset(file_path):
    """
    Load the dataset from a CSV file.

    Args:
        file_path (str): Path to the CSV file.

    Returns:
        np.ndarray: Feature matrix X containing the selected features for clustering.
    """
    dataset = pd.read_csv(file_path)
    X = dataset.iloc[:, [3, 4]].values  # Extract 'Annual Income (k$)' and 'Spending Score (1-100)'
    return X

def plot_dendrogram(X):
    """
    Plot the dendrogram for hierarchical clustering.

    Args:
        X (np.ndarray): Feature matrix used for clustering.
    """
    plt.figure(figsize=(10, 7))
    dendrogram(linkage(X, method='ward'))
    plt.title('Dendrogram')
    plt.xlabel('Customers')
    plt.ylabel('Euclidean Distances')
    plt.show()

def perform_agglomerative_clustering(X, n_clusters=5):
    """
    Perform Agglomerative Clustering and return the cluster labels.

    Args:
        X (np.ndarray): Feature matrix used for clustering.
        n_clusters (int): Number of clusters to form.

    Returns:
        np.ndarray: Cluster labels for each data point in the dataset.
    """
    clustering_model = AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage='ward')
    return clustering_model.fit_predict(X)

def plot_clusters(X, cluster_labels):
    """
    Plot the clusters identified by Agglomerative Clustering.

    Args:
        X (np.ndarray): Feature matrix used for clustering.
        cluster_labels (np.ndarray): Cluster labels for each data point in the dataset.
    """
    plt.figure(figsize=(10, 7))
    
    # Define cluster colors and labels
    cluster_colors = ['red', 'blue', 'green', 'cyan', 'magenta']
    cluster_names = ['Careful', 'Standard', 'Target', 'Careless', 'Sensible']
    
    for i, color in enumerate(cluster_colors):
        plt.scatter(X[cluster_labels == i, 0], X[cluster_labels == i, 1], 
                    s=100, c=color, label=cluster_names[i])
    
    plt.title('Clusters of Clients (Hierarchical Clustering)')
    plt.xlabel('Annual Income (k$)')
    plt.ylabel('Spending Score (1-100)')
    plt.legend()
    plt.show()

def main():
    """
    Execute the full workflow: load data, perform clustering, and plot results.
    """
    # Load the dataset
    X = load_dataset('Mall_Customers.csv')
    
    # Plot the dendrogram to determine the optimal number of clusters
    plot_dendrogram(X)
    
    # Perform Agglomerative Clustering on the dataset
    cluster_labels = perform_agglomerative_clustering(X)
    
    # Plot the resulting clusters
    plot_clusters(X, cluster_labels)

if __name__ == "__main__":
    main()
