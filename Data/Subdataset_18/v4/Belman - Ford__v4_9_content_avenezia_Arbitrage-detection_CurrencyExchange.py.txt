class CurrencyExchange:
    def __init__(self, from_currency, to_currency, exchange_rate):
        """
        Initialize a CurrencyExchange object.

        Args:
            from_currency (str): The source currency code.
            to_currency (str): The target currency code.
            exchange_rate (float): The exchange rate from source to target currency.
        """
        self.__from_currency = from_currency
        self.__to_currency = to_currency
        self.__exchange_rate = float(exchange_rate)

    def __str__(self):
        """
        Return the string representation of the CurrencyExchange object.

        Returns:
            str: A string representation of the object in the format 'from_currency -> to_currency: exchange_rate'.
        """
        return f"{self.__from_currency} -> {self.__to_currency}: {self.__exchange_rate}"

    def __repr__(self):
        """
        Return the official string representation of the CurrencyExchange object.

        Returns:
            str: The string representation of the object.
        """
        return self.__str__()

    @property
    def from_currency(self):
        """
        Return the source currency code.

        Returns:
            str: The source currency code.
        """
        return self.__from_currency

    @property
    def to_currency(self):
        """
        Return the target currency code.

        Returns:
            str: The target currency code.
        """
        return self.__to_currency

    @property
    def exchange_rate(self):
        """
        Return the exchange rate from source to target currency.

        Returns:
            float: The exchange rate.
        """
        return self.__exchange_rate

def main():
    """
    Main function to demonstrate the usage of the CurrencyExchange class.
    """
    # Create an instance of CurrencyExchange
    exchange = CurrencyExchange("USD", "EUR", 0.85)
    
    # Print the string representation of the CurrencyExchange object
    print(exchange)  # Output: USD -> EUR: 0.85
    
    # Print the individual properties
    print(f"From Currency: {exchange.from_currency}")  # Output: From Currency: USD
    print(f"To Currency: {exchange.to_currency}")  # Output: To Currency: EUR
    print(f"Exchange Rate: {exchange.exchange_rate}")  # Output: Exchange Rate: 0.85

if __name__ == "__main__":
    main()
