import binary

def countdown(x):
    """
    Print numbers from x down to 0 recursively.
    """
    print(x)
    if x > 0:
        countdown(x - 1)

def factorial(x):
    """
    Calculate the factorial of x recursively.
    """
    if x == 0:
        return 1
    return x * factorial(x - 1)

def tail_factorial(x, acc=1):
    """
    Calculate the factorial of x using tail recursion.
    """
    if x == 0:
        return acc
    return tail_factorial(x - 1, acc * x)

def tree_traverse(root):
    """
    Traverse the binary tree in a depth-first manner and print the nodes.
    """
    print(root.v)
    if root.l or root.r:
        if root.l:
            print("Left")
            tree_traverse(root.l)
        if root.r:
            print("Right")
            tree_traverse(root.r)
    else:
        print("Up")

def tree_search(root, val):
    """
    Search for a value in the binary tree.
    """
    if root.v == val:
        print("Found")
        return
    if root.l:
        tree_search(root.l, val)
    if root.r:
        tree_search(root.r, val)

def merge_sort(x):
    """
    Perform merge sort on the list x.
    """
    if len(x) < 2:
        return x
    
    mid = len(x) // 2
    left = merge_sort(x[:mid])
    right = merge_sort(x[mid:])
    
    return merge(left, right)

def merge(left, right):
    """
    Merge two sorted lists into one sorted list.
    """
    result = []
    i = j = 0
    
    while i < len(left) and j < len(right):
        if left[i] < right[j]:
            result.append(left[i])
            i += 1
        else:
            result.append(right[j])
            j += 1
    
    result.extend(left[i:])
    result.extend(right[j:])
    
    return result
