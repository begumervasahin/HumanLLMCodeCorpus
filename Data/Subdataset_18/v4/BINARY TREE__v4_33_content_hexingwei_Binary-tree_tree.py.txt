import sys
from collections import defaultdict
import gflags as flags

FLAGS = flags.FLAGS

class Tree:
    def __init__(self, label, span, wrd=None, subs=None):
        assert (wrd is None) ^ (subs is None), f"Bad tree {label} {wrd} {subs}"
        self.label = label
        self.span = span
        self.word = wrd
        self.subs = subs
        self._str = None
        self._hash = None

    def is_terminal(self):
        return self.word is not None

    def dostr(self):
        if self.is_terminal():
            return f"({self.label} {self.word})"
        else:
            subs_str = " ".join(map(str, self.subs))
            return f"({self.label} {subs_str})"

    def __str__(self):
        if self._str is None:
            self._str = self.dostr()
        return self._str

    __repr__ = __str__

    def __hash__(self):
        if self._hash is None:
            self._hash = hash(str(self))
        return self._hash

    def __eq__(self, other):
        return str(self) == str(other)

    def span_width(self):
        return self.span[1] - self.span[0]

    __len__ = span_width

    def arity(self):
        return len(self.subs)

    def labelspan(self):
        return f"{self.label} [{self.span[0]}-{self.span[1]}]"

    def spanlabel(self):
        return f"[{self.span[0]}-{self.span[1]}]: {self.label}"

    @staticmethod
    def _parse(line, pos=0, wrdidx=0, trunc=True):
        """
        Parse a line of text into a Tree object.
        Returns a triple:
            ((pos, wordindex), is_empty, tree)
        """
        assert line[pos] == '(', f"Tree must start with '('! line={line}, pos={pos}, line[pos]={line[pos]}"
        empty = False
        space = line.find(" ", pos)
        label = line[pos + 1 : space]
        
        if trunc:
            if label[0] != "-":
                for sep in ["-", "=", "|"]:
                    dashpos = label.find(sep)
                    if dashpos >= 0:
                        label = label[:dashpos]
            elif label == "-NONE-":
                empty = True

        newpos = space + 1
        newidx = wrdidx

        if line[newpos] == '(':
            subtrees = []
            while line[newpos] != ')':
                if line[newpos] == " ":
                    newpos += 1
                (newpos, newidx), emp, sub = Tree._parse(line, newpos, newidx, trunc)
                if not emp:
                    subtrees.append(sub)
            return (newpos + 1, newidx), subtrees == [], Tree(label, (wrdidx, newidx), subs=subtrees)
        else:
            finalpos = line.find(")", newpos)
            word = line[newpos : finalpos]
            return (finalpos + 1, wrdidx + 1 if not empty else wrdidx), empty, Tree(label, (wrdidx, wrdidx + 1), wrd=word)

    @staticmethod
    def parse(line, trunc=False):
        """
        Parse a line of text into a Tree object.
        """
        _, is_empty, tree = Tree._parse(line, 0, 0, trunc)
        assert not is_empty, "The whole tree is empty! " + line
        if tree.label != "TOP":
            tree = Tree(label="TOP", span=tree.span, subs=[tree])
        return tree

    def all_label_spans(self):
        """
        Get a list of all labeled spans for PARSEVAL.
        """
        if self.is_terminal():
            return []
        spans = [(self.label, self.span)]
        for sub in self.subs:
            spans.extend(sub.all_label_spans())
        return spans

    def label_span_counts(self):
        """
        Return a dict mapping (label, span) -> count.
        """
        counts = defaultdict(int)
        for span in self.all_label_spans():
            counts[span] += 1
        return counts

    def pp(self, level=0):
        """
        Pretty print the tree.
        """
        if self.is_terminal():
            print(f"{'| ' * level}{self.labelspan()} {self.word}")
        else:
            print(f"{'| ' * level}{self.labelspan()}")
            for sub in self.subs:
                sub.pp(level + 1)

    def height(self):
        """
        Calculate the height of the tree.
        """
        if self.is_terminal():
            return 1
        return max(sub.height() for sub in self.subs) + 1

    def binarize(self):
        """
        Binarize the tree.
        """
        if self.subs:
            if len(self.subs) > 2:
                new_label = self.label
                if new_label[-1] != "'":
                    new_label += "'"
                t2 = Tree(label=new_label, span=self.span, subs=self.subs[1:])
                self.subs = [self.subs[0], t2]
            for child in self.subs:
                child.binarize()

    def deBinarize(self):
        """
        De-binarize the tree.
        Assumes that this is a binary tree.
        """
        if self.subs:
            while self.subs and self.subs[-1].label[-1] == "'":
                rhs_node = self.subs.pop()
                self.subs.extend(rhs_node.subs)
            for child in self.subs:
                child.deBinarize()

    def getProductions(self):
        """
        Get the productions of the tree.
        """
        prods = []
        if self.subs:
            if len(self.subs) == 2:
                child1, child2 = self.subs
                prod = (self.label, f"{child1.label} {child2.label}")
                prods.append(prod)
            elif len(self.subs) == 1:
                prod = (self.label, self.subs[0].label)
                prods.append(prod)
            for child in self.subs:
                prods.extend(child.getProductions())
        elif self.word:
            prod = (self.label, self.word)
            prods.append(prod)
        return prods

if __name__ == "__main__":
    flags.DEFINE_integer("max_len", 400, "Maximum sentence length")
    flags.DEFINE_boolean("pp", True, "Pretty print")
    flags.DEFINE_boolean("height", False, "Output the height of each tree")
    flags.DEFINE_boolean("clean", False, "Clean up functional tags and empty nodes")

    argv = FLAGS(sys.argv)
    for i, line in enumerate(sys.stdin):
        tree = Tree.parse(line.strip(), trunc=FLAGS.clean)
        tree.binarize()
        
        if len(tree) <= FLAGS.max_len:
            if FLAGS.pp:
                tree.pp()
                print(tree)
            elif FLAGS.height:
                print(f"{len(tree)}\t{tree.height()}")
            else:
                print(tree)
        
        tree.deBinarize()
        
        if len(tree) <= FLAGS.max_len:
            if FLAGS.pp:
                tree.pp()
                print(tree)
            elif FLAGS.height:
                print(f"{len(tree)}\t{tree.height()}")
            else:
                print(tree)
