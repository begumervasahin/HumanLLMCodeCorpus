class State:
    def __init__(self, state, parent=None, move=None, depth=0, cost=0, key=None):
        """
        Initialize a new state for the 8-puzzle solver.

        Args:
            state (list): The current state of the puzzle.
            parent (State, optional): The parent state. Defaults to None.
            move (str, optional): The move taken to reach this state. Defaults to None.
            depth (int, optional): The depth of this state in the search tree. Defaults to 0.
            cost (int, optional): The cost to reach this state. Defaults to 0.
            key (str, optional): A unique key for this state. Defaults to None.
        """
        self.state = state
        self.parent = parent
        self.move = move
        self.depth = depth
        self.cost = cost
        self.key = key

        if self.state:
            self.map = ''.join(str(e) for e in self.state)

    def __eq__(self, other):
        """
        Compare two states for equality based on their string representation.

        Args:
            other (State): The other state to compare against.

        Returns:
            bool: True if the states are equal, False otherwise.
        """
        return self.map == other.map

    def __lt__(self, other):
        """
        Compare two states for ordering based on their string representation.

        Args:
            other (State): The other state to compare against.

        Returns:
            bool: True if this state is less than the other state, False otherwise.
        """
        return self.map < other.map
