class ListNode:
    """
    Class to represent a node in a singly linked list.
    """
    def __init__(self, x):
        self.val = x
        self.next = None

class Solution:
    """
    Class to provide a solution for adding two numbers represented by linked lists.
    """
    def addTwoNumbers(self, l1, l2):
        """
        Adds two numbers represented by linked lists and returns the result as a linked list.
        The digits are stored in reverse order and each node contains a single digit.
        
        Args:
        l1 (ListNode): The head node of the first linked list.
        l2 (ListNode): The head node of the second linked list.
        
        Returns:
        ListNode: The head node of the resulting linked list.
        """
        carry = 0
        root = ListNode(0)  # Placeholder for the result linked list
        dummy = root
        
        while l1 or l2 or carry:
            if l1:
                carry += l1.val
                l1 = l1.next
            if l2:
                carry += l2.val
                l2 = l2.next
            
            # Create a new node with the digit value of (carry % 10)
            dummy.next = ListNode(carry % 10)
            carry //= 10  # Update carry for the next iteration
            dummy = dummy.next  # Move to the next node
        
        return root.next

# Example usage:
if __name__ == "__main__":
    # Create the first linked list: 2 -> 4 -> 3
    l1 = ListNode(2)
    l1.next = ListNode(4)
    l1.next.next = ListNode(3)
    
    # Create the second linked list: 5 -> 6 -> 4
    l2 = ListNode(5)
    l2.next = ListNode(6)
    l2.next.next = ListNode(4)
    
    # Add the two numbers
    solution = Solution()
    result = solution.addTwoNumbers(l1, l2)
    
    # Print the result linked list
    while result:
        print(result.val, end=" -> " if result.next else "\n")
        result = result.next
