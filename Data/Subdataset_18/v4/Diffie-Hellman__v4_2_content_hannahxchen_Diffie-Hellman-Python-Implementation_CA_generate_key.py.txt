from Crypto.PublicKey import RSA
from Crypto.Random import get_random_bytes

def generate_rsa_key(bits=1024):
    """
    Generate an RSA key pair.

    Args:
        bits (int): The number of bits for the RSA key. Default is 1024.

    Returns:
        tuple: (private_key, public_key) - RSA keys.
    """
    key = RSA.generate(bits, get_random_bytes)
    return key, key.publickey()

def save_key_to_file(filename, key):
    """
    Save an RSA key to a file.

    Args:
        filename (str): Path to the file where the key will be saved.
        key (RSA key): The RSA key to save.
    """
    with open(filename, "wb") as file:
        file.write(key.export_key())

def main():
    # Generate RSA key pairs for two users
    private_key1, public_key1 = generate_rsa_key()
    private_key2, public_key2 = generate_rsa_key()
    
    # File paths for storing keys
    file_paths = {
        "privatekey1": "privatekey1.pem",
        "publickey1": "publickey1.pem",
        "privatekey2": "privatekey2.pem",
        "publickey2": "publickey2.pem"
    }
    
    # Save keys to files
    save_key_to_file(file_paths["privatekey1"], private_key1)
    save_key_to_file(file_paths["publickey1"], public_key1)
    save_key_to_file(file_paths["privatekey2"], private_key2)
    save_key_to_file(file_paths["publickey2"], public_key2)
    
    print("Finished generating key pairs.")

if __name__ == "__main__":
    main()
