import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
import datetime
import numpy as np

def cluster_analysis(data_array, k):
    """
    Perform K-Means clustering on the input data points and visualize the clusters.

    Parameters:
        data_array (list of lists): Input data points.
        k (int): Number of clusters to create.

    Returns:
        list: Cluster labels for each data point.
    """
    # Convert the input data list to a numpy array for easier manipulation
    data_array = np.array(data_array)

    # Initialize the KMeans model with the specified number of clusters
    model = KMeans(n_clusters=k)

    # Fit the model to the data and predict the cluster labels
    labels = model.fit_predict(data_array)

    # Extract x and y coordinates from the data array
    xs = data_array[:, 0]
    ys = data_array[:, 1]

    # Extract the coordinates of the cluster centers
    centroids = model.cluster_centers_
    centroids_x = centroids[:, 0]
    centroids_y = centroids[:, 1]

    # Create a scatter plot of the data points colored by cluster
    plt.scatter(xs, ys, c=labels, alpha=0.5, label='Data Points')

    # Plot the cluster centers
    plt.scatter(centroids_x, centroids_y, marker='D', s=100, c='red', label='Cluster Centers')

    # Set the plot title to include the current date and time
    plt.title(f"Cluster Analysis ({datetime.datetime.now():%Y-%m-%d %H:%M:%S})")
    plt.xlabel('X')
    plt.ylabel('Y')
    plt.legend()

    # Inform the user to inspect the plot
    print("\nInspect the cluster plot shown.\nWhen ready, close the plot to proceed.")      

    # Show the plot
    plt.show()

    # Return the cluster labels
    return labels

# Example usage
if __name__ == "__main__":
    # Example data
    data_array = [[1, 2], [2, 3], [3, 4], [10, 9], [8, 7], [7, 6]]
    k = 2  # Number of clusters

    # Call the function
    cluster_analysis(data_array, k)
