# Assuming the classes Fp, EllipticCurve, and RationalPointInEC are imported from Mods.Math.Fp,
# Mods.EC.Elliptic_Curve_Fp, and Mods.EC.Rational_Point_In_ECFp respectively

from Mods.Math.Fp import Fp
from Mods.EC.Elliptic_Curve_Fp import EllipticCurve
from Mods.EC.Rational_Point_In_ECFp import RationalPointInEC

def main():
    # Initialize finite field instance with modulus 5
    fp_instance = Fp.get_instance(5)
    print(fp_instance)

    # Initialize elliptic curve instance with coefficients a=0, b=1, and modulus 1
    elliptic_curve = EllipticCurve.get_instance(0, 1, 1)
    print(elliptic_curve)

    # Initialize rational points on the elliptic curve
    o = RationalPointInEC()
    p1 = RationalPointInEC(0, 1)
    p2 = RationalPointInEC(0, 4)
    p3 = RationalPointInEC(2, 1)
    p4 = RationalPointInEC(2, 4)
    p5 = RationalPointInEC(3, 1)
    p6 = RationalPointInEC(3, 4)

    # Print initialized points
    points = [o, p1, p2, p3, p4, p5, p6]
    for point in points:
        print(point)

    # Perform and print calculations with points on the elliptic curve
    print('Calculations:')
    print(f'2 * {p3} = {2 * p3}')
    print(f'{o} + {p1} = {o + p1}')
    print(f'{p3} + {p1} = {p3 + p1}')
    print(f'{p5} + {p3} = {p5 + p3}')

if __name__ == '__main__':
    main()
