def fact_rec(n):
    """Recursive factorial function."""
    if n == 0:
        return 1
    else:
        return n * fact_rec(n-1)

def fact_cps(n, cont):
    """Factorial function using continuation-passing style (CPS)."""
    if n == 0:
        return cont(1)
    else:
        return fact_cps(n-1, lambda value: cont(n * value))

def end_cont(n):
    """End continuation function."""
    return n

def trampoline(f, *args):
    """Trampoline function to handle CPS with thunks."""
    v = f(*args)
    while callable(v):
        v = v()
    return v

def fact_cps_thunked(n, cont):
    """Factorial function using CPS with thunks."""
    if n == 0:
        return cont(1)
    else:
        return lambda: fact_cps_thunked(n - 1, lambda value: lambda: cont(n * value))

def fib_rec(n):
    """Recursive Fibonacci function."""
    if n <= 2:
        return 1
    else:
        return fib_rec(n-1) + fib_rec(n-2)

def fib_cps(n, cont):
    """Fibonacci function using CPS."""
    if n <= 2:
        return cont(1)
    else:
        return fib_cps(n-1, lambda value1: fib_cps(n-2, lambda value2: cont(value1 + value2)))

def fib_cps_thunked(n, cont):
    """Fibonacci function using CPS with thunks."""
    if n <= 2:
        return cont(1)
    else:
        return lambda: fib_cps_thunked(n-1, lambda value1: lambda: fib_cps_thunked(n-2, lambda value2: cont(value1 + value2)))

def main():
    """Main function to test factorial and Fibonacci implementations."""
    try:
        print(fact_rec(1000))
    except RecursionError as e:
        print(f'RecursionError: {e}')

    try:
        print(fact_cps(1000, end_cont))
    except RecursionError as e:
        print(f'RecursionError: {e}')

    try:
        print(trampoline(fact_cps_thunked(1000, end_cont)))
    except RecursionError as e:
        print(f'RecursionError: {e}')

    try:
        print(fib_cps(43, end_cont))
    except RecursionError as e:
        print(f'RecursionError: {e}')

    try:
        print(trampoline(fib_cps_thunked(43, end_cont)))
    except RecursionError as e:
        print(f'RecursionError: {e}')

if __name__ == '__main__':
    main()
