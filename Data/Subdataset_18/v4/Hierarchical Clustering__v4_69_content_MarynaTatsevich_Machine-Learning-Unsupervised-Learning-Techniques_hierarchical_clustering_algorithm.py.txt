import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import scipy.cluster.hierarchy as sch
from sklearn.cluster import AgglomerativeClustering

def load_data(file_path):
    """Load the dataset from a CSV file."""
    return pd.read_csv(file_path)

def plot_dendrogram(X, method='ward'):
    """
    Plot a dendrogram to visualize the hierarchical clustering process.

    Args:
        X (np.ndarray): The data used for clustering.
        method (str): The linkage method to be used. Default is 'ward'.
    """
    plt.figure(figsize=(10, 7))
    sch.dendrogram(sch.linkage(X, method=method))
    plt.title('Dendrogram')
    plt.xlabel('Customers')
    plt.ylabel('Euclidean Distances')
    plt.show()

def perform_agglomerative_clustering(X, n_clusters=5, affinity='euclidean', linkage='ward'):
    """
    Apply Agglomerative Clustering to the dataset.

    Args:
        X (np.ndarray): The data to be clustered.
        n_clusters (int): The number of clusters to form. Default is 5.
        affinity (str): Metric used to compute the linkage. Default is 'euclidean'.
        linkage (str): Which linkage criterion to use. Default is 'ward'.

    Returns:
        np.ndarray: Cluster labels for each point in the dataset.
    """
    model = AgglomerativeClustering(n_clusters=n_clusters, affinity=affinity, linkage=linkage)
    return model.fit_predict(X)

def plot_clusters(X, labels, cluster_colors=None, cluster_names=None):
    """
    Plot the clusters identified by Agglomerative Clustering.

    Args:
        X (np.ndarray): The data points.
        labels (np.ndarray): Cluster labels for each data point.
        cluster_colors (list): Colors for each cluster. Default is None.
        cluster_names (list): Names for each cluster. Default is None.
    """
    if cluster_colors is None:
        cluster_colors = ['red', 'blue', 'green', 'black', 'magenta']
    if cluster_names is None:
        cluster_names = ['Careful', 'Standard', 'Target', 'Careless', 'Sensible']

    plt.figure(figsize=(10, 7))
    for i, color in enumerate(cluster_colors):
        plt.scatter(X[labels == i, 0], X[labels == i, 1], s=100, c=color, label=cluster_names[i])

    plt.title('Clusters of Clients')
    plt.xlabel('Annual Income (K$)')
    plt.ylabel('Spending Score (1-100)')
    plt.legend()
    plt.show()

def main():
    """Main function to execute the clustering analysis."""
    # Load the dataset
    dataset = load_data('Mall_Customers.csv')
    
    # Extract the relevant features for clustering
    X = dataset.iloc[:, [3, 4]].values
    
    # Plot the dendrogram to determine the optimal number of clusters
    plot_dendrogram(X)
    
    # Perform Agglomerative Clustering
    cluster_labels = perform_agglomerative_clustering(X)
    
    # Plot the resulting clusters
    plot_clusters(X, cluster_labels)

if __name__ == "__main__":
    main()
