import hashlib
import time

def hash_with_sha256(input_str):
    """
    Generate a SHA-256 hash for the given input string.
    """
    hash_object = hashlib.sha256(input_str.encode('utf-8'))
    return hash_object.hexdigest()

def find_password(password):
    """
    Find and print the password if its hash matches the stored hash in the password file.
    """
    with open("password_file.txt") as f:
        for line in f:
            user, salt, stored_hash = line.strip().split(",")
            computed_hash = hash_with_sha256(password + salt)
            if computed_hash == stored_hash:
                print(f"{user}: {password}")

def get_combination(size, comb):
    """
    Recursively generate combinations of digits of a given size and find matching passwords.
    """
    if size == 0:
        find_password(comb)
    else:
        for digit in range(10):
            new_comb = comb + str(digit)
            get_combination(size - 1, new_comb)

def hack_file(start, end):
    """
    Generate all combinations of digits of length between start and end (inclusive)
    and attempt to find matching passwords.
    """
    if not isinstance(start, int) or not isinstance(end, int):
        print('Invalid input. Please try again.')
        return

    if start < 3 or end > 7:
        print('Invalid input. Please try again.')
        return

    for size in range(start, end + 1):
        get_combination(size, '')

def test_cases():
    """
    Run test cases to measure the execution time for different scenarios.
    """
    start = time.time()
    hack_file(3.0, 7.0)
    end = time.time()
    print('Running time for invalid data type:', end - start, 'seconds\n')

    start = time.time()
    hack_file(2, 7)
    end = time.time()
    print('Running time for invalid range:', end - start, 'seconds\n')

    start = time.time()
    hack_file(3, 8)
    end = time.time()
    print('Running time for invalid range:', end - start, 'seconds\n')

def main():
    """
    Main function to execute the normal run and test cases.
    """
    start = time.time()
    hack_file(3, 7)
    end = time.time()
    print('Running time for normal run:', end - start, 'seconds\n')

    test_cases()

if __name__ == "__main__":
    main()
