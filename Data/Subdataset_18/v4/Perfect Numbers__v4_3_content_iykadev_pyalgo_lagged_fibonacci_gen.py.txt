# Initial values
j = 3
k = 7
s = [8, 6, 7, 5, 3, 0, 9]

# Loop for 10 iterations
for n in range(10):
    for i in range(len(s)):
        if i == 0:
            # Calculate the output value based on the elements at positions j-1 and k-1
            out = (s[j - 1] + s[k - 1]) % 10
        elif 0 < i < len(s) - 1:
            # Shift elements to the left
            s[i] = s[i + 1]
        else:
            # Replace the last element with the calculated output value
            s[i] = out
            print(s[i])
