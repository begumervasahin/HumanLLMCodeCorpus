import socket as mysoc
import sys

def rs(com_host, edu_host, file_name):
    try:
        rssd = mysoc.socket(mysoc.AF_INET, mysoc.SOCK_STREAM)
    except mysoc.error as err:
        print('[RS] RS server socket open error:', err)
        return

    try:
        rstotscom = mysoc.socket(mysoc.AF_INET, mysoc.SOCK_STREAM)
    except mysoc.error as err:
        print('Socket open error:', err)
        return

    try:
        rstotsedu = mysoc.socket(mysoc.AF_INET, mysoc.SOCK_STREAM)
    except mysoc.error as err:
        print('Socket open error:', err)
        return

    try:
        with open(file_name, "r") as fr:
            RS_table = {}
            RS_table[edu_host] = {'ip': mysoc.gethostbyname(edu_host), 'flag': 'NS'}
            RS_table[com_host] = {'ip': mysoc.gethostbyname(com_host), 'flag': 'NS'}

            for line in fr:
                tokenize = line.split()
                if tokenize[1].strip() != '-':
                    RS_table[tokenize[0].strip()] = {
                        'ip': tokenize[1].strip(),
                        'flag': tokenize[2].strip()
                    }

    except IOError as err:
        print('File Open Error:', err)
        print("Please ensure the desired file exists in the source folder")
        return

    server_binding = ('', 50008)
    rssd.bind(server_binding)
    rssd.listen(1)

    host = mysoc.gethostname()
    localhost_ip = mysoc.gethostbyname(host)
    print(f"[S]: Server host name is: {host}")
    print(f"[S]: Attempting to connect to client.\n[S]: Server IP address is {localhost_ip}")

    crsd, addr = rssd.accept()
    print(f"[S]: Got a connection request from a client at {addr}")

    first_edu = True
    first_com = True

    while True:
        data = crsd.recv(100)
        hnstring = data.decode('utf-8')
        if not hnstring:
            break

        entry = ""
        if hnstring in RS_table:
            entry = f"{hnstring} {RS_table[hnstring]['ip']} {RS_table[hnstring]['flag']}"
        else:
            if ".edu" in hnstring:
                if not edu_host:
                    entry = f"{hnstring} - Error: HOST NOT FOUND"
                else:
                    if first_edu:
                        first_edu = False
                        try:
                            sa_sameas_myaddr = RS_table[edu_host]['ip']
                            port = 5677
                            server_binding = (sa_sameas_myaddr, port)
                            rstotsedu.connect(server_binding)
                        except mysoc.error as err:
                            print('TSEDU connect error:', err)
                            return
                    rstotsedu.send(hnstring.strip().encode('utf-8'))
                    dataFromTS = rstotsedu.recv(100).decode('utf-8')
                    if not dataFromTS:
                        break
                    entry = dataFromTS
            elif ".com" in hnstring:
                if not com_host:
                    entry = f"{hnstring} - Error: HOST NOT FOUND"
                else:
                    if first_com:
                        first_com = False
                        try:
                            sa_sameas_myaddr = RS_table[com_host]['ip']
                            port = 5678
                            server_binding = (sa_sameas_myaddr, port)
                            rstotscom.connect(server_binding)
                        except mysoc.error as err:
                            print('TSCOM connect error:', err)
                            return
                    rstotscom.send(hnstring.strip().encode('utf-8'))
                    dataFromTS = rstotscom.recv(100).decode('utf-8')
                    if not dataFromTS:
                        break
                    entry = dataFromTS
            else:
                entry = f"{hnstring} - Error: HOST NOT FOUND"

        crsd.send(entry.encode('utf-8'))

    rssd.close()
    rstotsedu.close()
    rstotscom.close()

if __name__ == '__main__':
    com_host = sys.argv[1]
    edu_host = sys.argv[2]
    file_name = sys.argv[3]
    rs(com_host, edu_host, file_name)
