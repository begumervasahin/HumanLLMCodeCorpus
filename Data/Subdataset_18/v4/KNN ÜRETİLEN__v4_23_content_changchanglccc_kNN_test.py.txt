import numpy as np
import operator

# Basic addition
a = 1
b = 2
c = a + b
print(c)  # Output: 3

# Create a numpy array and get its shape
b = np.array([6, 7, 8])
c = b.shape[0]
print(c)  # Output: 3

# Sum of all elements in a 2D array
a = np.sum([[0, 1, 2], [2, 1, 3]])
print(a)  # Output: 9

# Sum of elements along axis 0 (column-wise)
a = np.sum([[0, 1, 2], [2, 1, 3]], axis=0)
print(a)  # Output: [2 2 5]

# Sum of elements along axis 1 (row-wise)
a = np.sum([[8, 1, 2], [2, 1, 3], [0, 0, 0]], axis=1)
print(a)  # Output: [11  6  0]

# Argsort the result to get the indices that would sort the array
d = a.argsort()
print(d)  # Output: [2 1 0]

# Sort a dictionary by value in descending order
classCount = {"A": 6, "B": 1, "C": 0, "D": 2}
sortedClassCount = sorted(classCount.items(), key=operator.itemgetter(1), reverse=True)
print(sortedClassCount)  # Output: [('A', 6), ('D', 2), ('B', 1), ('C', 0)]

# Function to convert an image to a vector (example)
def imgToVector(matrix):
    rows, cols = matrix.shape
    myVector = np.zeros((1, rows * cols))
    for i in range(rows):
        for j in range(cols):
            myVector[0, cols * i + j] = int(matrix[i, j])
    return myVector
