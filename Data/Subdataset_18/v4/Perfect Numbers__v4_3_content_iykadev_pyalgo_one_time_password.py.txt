import hmac
import base64
import struct
import hashlib
import time

def get_hotp_token(secret, intervals_no):
    """Generates an HMAC-based One-Time Password (HOTP) token."""
    # Decode the base32 secret to bytes
    key = base64.b32decode(secret, True)
    
    # Pack the intervals number into a byte array
    msg = struct.pack(">Q", intervals_no)
    
    # Create the HMAC-SHA1 hash of the message using the secret key
    hmac_hash = hmac.new(key, msg, hashlib.sha1).digest()
    
    # Dynamically truncate the hash to extract a 4-byte string
    offset = hmac_hash[19] & 0x0F
    code = (struct.unpack(">I", hmac_hash[offset:offset + 4])[0] & 0x7FFFFFFF) % 1000000
    
    return code

def get_totp_token(secret):
    """Generates a Time-based One-Time Password (TOTP) token."""
    # Calculate the current interval number based on the current time
    intervals_no = int(time.time()) // 30
    return get_hotp_token(secret, intervals_no)

# Example usage
if __name__ == "__main__":
    secret = 'MZXW633PN5XW6MZX'
    
    print("HOTP tokens:")
    for i in range(1, 30):
        print(f"Interval {i}: {get_hotp_token(secret, intervals_no=i)}")
    
    print("\nCurrent TOTP token:")
    print(get_totp_token(secret))
