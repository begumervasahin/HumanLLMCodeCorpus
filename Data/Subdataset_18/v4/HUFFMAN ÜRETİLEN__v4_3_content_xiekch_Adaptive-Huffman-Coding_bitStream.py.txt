class BitStream:
    def __init__(self, file_name, mode):
        self.file = open(file_name, mode)
        self.word = 0
        self.pos = 7 if mode.startswith('w') else -1
        self.mode = mode

    def read(self, size):
        bits = []
        for _ in range(size):
            if self.pos == -1:
                byte = self.file.read(1)
                if not byte:
                    return ''.join(bits)
                self.word = ord(byte)
                self.pos = 7
            bits.append('1' if self.word & (1 << self.pos) else '0')
            self.pos -= 1
        return ''.join(bits)

    def write(self, bit_string):
        for bit in bit_string:
            self.word = (self.word << 1) | int(bit)
            self.pos -= 1
            if self.pos == -1:
                self.flush()

    def close(self):
        if self.mode.startswith('w'):
            self.flush()
        self.file.close()

    def flush(self):
        if self.pos != 7:
            self.word <<= self.pos + 1
            self.file.write(self.word.to_bytes(1, byteorder='little'))
        self.word = 0
        self.pos = 7
