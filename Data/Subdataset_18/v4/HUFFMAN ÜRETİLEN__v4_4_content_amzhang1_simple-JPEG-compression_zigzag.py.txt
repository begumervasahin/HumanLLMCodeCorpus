import numpy as np

def zigzag(input_matrix):
    rows, cols = input_matrix.shape
    output = np.zeros(rows * cols, dtype=input_matrix.dtype)
    
    i = 0
    for diag_sum in range(rows + cols - 1):
        if diag_sum % 2 == 0:
            r = min(diag_sum, rows - 1)
            c = diag_sum - r
            while r >= 0 and c < cols:
                output[i] = input_matrix[r, c]
                i += 1
                r -= 1
                c += 1
        else:
            c = min(diag_sum, cols - 1)
            r = diag_sum - c
            while c >= 0 and r < rows:
                output[i] = input_matrix[r, c]
                i += 1
                r += 1
                c -= 1

    return output

def inverse_zigzag(input_array, rows, cols):
    output_matrix = np.zeros((rows, cols), dtype=input_array.dtype)
    
    i = 0
    for diag_sum in range(rows + cols - 1):
        if diag_sum % 2 == 0:
            r = min(diag_sum, rows - 1)
            c = diag_sum - r
            while r >= 0 and c < cols:
                output_matrix[r, c] = input_array[i]
                i += 1
                r -= 1
                c += 1
        else:
            c = min(diag_sum, cols - 1)
            r = diag_sum - c
            while c >= 0 and r < rows:
                output_matrix[r, c] = input_array[i]
                i += 1
                r += 1
                c -= 1

    return output_matrix
