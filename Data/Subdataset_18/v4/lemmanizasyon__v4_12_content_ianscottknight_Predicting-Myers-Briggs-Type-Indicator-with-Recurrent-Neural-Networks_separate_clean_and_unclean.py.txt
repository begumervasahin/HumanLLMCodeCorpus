import os
import collections
import pandas as pd
import csv

# Define paths for data files
DATA_DIR = "data"
MBTI_RAW_CSV_PATH = os.path.join(DATA_DIR, "mbti_1.csv")
MBTI_CLEAN_CSV_PATH = os.path.join(DATA_DIR, "mbti_clean.csv")
MBTI_UNCLEAN_CSV_PATH = os.path.join(DATA_DIR, "mbti_unclean.csv")

# Define MBTI type frequencies
MBTI_TO_FREQUENCY_DICT = {
    "ISTJ": 0.11,
    "ISFJ": 0.09,
    "INFJ": 0.04,
    "INTJ": 0.05,
    "ISTP": 0.05,
    "ISFP": 0.05,
    "INFP": 0.06,
    "INTP": 0.06,
    "ESTP": 0.04,
    "ESFP": 0.04,
    "ENFP": 0.08,
    "ENTP": 0.06,
    "ESTJ": 0.08,
    "ESFJ": 0.09,
    "ENFJ": 0.05,
    "ENTJ": 0.05,
}

# Read the raw MBTI data
df = pd.read_csv(MBTI_RAW_CSV_PATH)

# Count occurrences of each MBTI type
counts = collections.defaultdict(int)
for mbti in df["type"]:
    counts[mbti] += 1

# Identify the limiting MBTI type based on minimum size ratio
limiting_type = None
min_size = float("infinity")
for mbti, count in counts.items():
    size = count / MBTI_TO_FREQUENCY_DICT[mbti]
    if size < min_size:
        min_size = size
        limiting_type = mbti

# Group rows by MBTI type
type_to_rows = collections.defaultdict(list)
for _, row in df.iterrows():
    type_to_rows[row["type"]].append(row)

# Write cleaned and uncleaned data to separate CSV files
unclean_data = []
with open(MBTI_CLEAN_CSV_PATH, "w", newline='') as clean_file:
    writer = csv.writer(clean_file)
    writer.writerow(["type", "posts"])
    for mbti, frequency in MBTI_TO_FREQUENCY_DICT.items():
        rows = type_to_rows[mbti]
        sample_size = int(round(min_size * frequency))
        writer.writerows(rows[:sample_size])
        unclean_data.extend(rows[sample_size:])

with open(MBTI_UNCLEAN_CSV_PATH, "w", newline='') as unclean_file:
    writer = csv.writer(unclean_file)
    writer.writerow(["type", "posts"])
    writer.writerows(unclean_data)
