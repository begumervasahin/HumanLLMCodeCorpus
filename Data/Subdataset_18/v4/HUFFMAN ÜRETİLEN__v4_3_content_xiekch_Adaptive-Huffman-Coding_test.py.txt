def write_file(file_name, data):
    """Write the given data to a file in binary mode."""
    with open(file_name, 'wb') as file:
        for ch in data:
            file.write(ch.encode('utf-8'))  # Encode the character to bytes using UTF-8

def read_file(file_name):
    """Read and print the contents of a file in binary mode."""
    with open(file_name, 'rb') as file:
        while (byte := file.read(1)):  # Read one byte at a time
            print(byte.decode('utf-8', errors='replace'), end='')  # Decode and print the byte

if __name__ == '__main__':
    file_name = 'test.txt'
    data = '中文'  # Unicode string containing non-ASCII characters
    write_file(file_name, data)
    read_file(file_name)
