from LinkedList import Node, LinkedList

def locate_middle_node(ll):
    """
    Locate the middle node of the linked list.
    """
    cursor, ds_cursor = ll.head, ll.head
    prev_cursor = None

    while ds_cursor and ds_cursor.next_node:
        prev_cursor = cursor
        cursor = cursor.next_node
        ds_cursor = ds_cursor.next_node.next_node

    # Disconnect the first half from the second half
    if prev_cursor:
        prev_cursor.next_node = None

    return cursor

def reverse_second_half(middle_node):
    """
    Reverse the second half of the linked list starting from the middle node.
    """
    cursor = middle_node
    next_cursor = cursor.next_node
    cursor.next_node = None
    prev_cursor = cursor

    while next_cursor:
        cursor = next_cursor
        next_cursor = cursor.next_node
        cursor.next_node = prev_cursor
        prev_cursor = cursor

    return cursor

def reorder_original_list(ll, head_of_second_half):
    """
    Reorder the original linked list by alternating nodes from the first and second halves.
    """
    c1 = ll.head
    c2 = head_of_second_half

    while c1 and c2:
        tmp_node1 = c1.next_node
        tmp_node2 = c2.next_node

        c1.next_node = c2
        if not tmp_node1:  # If the end of the first half is reached
            break

        c2.next_node = tmp_node1
        c1 = tmp_node1
        c2 = tmp_node2

def reorder_linked_list(ll):
    """
    Reorder the linked list by:
    1. Finding the middle node
    2. Reversing the second half
    3. Reordering the original list
    """
    if not ll.head or not ll.head.next_node:
        return

    middle_node = locate_middle_node(ll)
    head_of_second_half = reverse_second_half(middle_node)
    reorder_original_list(ll, head_of_second_half)

# Example usage:
ll = LinkedList()
ll.add_to_end(1)
ll.add_to_end(2)
ll.add_to_end(3)
ll.add_to_end(4)
ll.add_to_end(5)

print("Original list:")
ll.display()

reorder_linked_list(ll)

print("Reordered list:")
ll.display()
