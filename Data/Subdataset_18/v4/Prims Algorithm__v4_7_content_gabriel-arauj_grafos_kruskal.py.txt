import networkx as nx

class KruskalAlgorithm:
    def __init__(self):
        self.parent = {}
        self.rank = {}

    def create_set(self, vertex):
        """
        Initializes a set for the given vertex and sets its rank to 0.
        """
        self.parent[vertex] = vertex
        self.rank[vertex] = 0

    def find(self, vertex):
        """
        Finds the representative of the set containing the given vertex.
        Implements path compression to flatten the structure for efficiency.
        """
        if self.parent[vertex] != vertex:
            self.parent[vertex] = self.find(self.parent[vertex])
        return self.parent[vertex]

    def union(self, vertex1, vertex2):
        """
        Unites the sets containing vertex1 and vertex2 using rank to keep the tree flat.
        """
        root1 = self.find(vertex1)
        root2 = self.find(vertex2)

        if root1 != root2:
            if self.rank[root1] > self.rank[root2]:
                self.parent[root2] = root1
            else:
                self.parent[root1] = root2
                if self.rank[root1] == self.rank[root2]:
                    self.rank[root2] += 1

    def calculate_weight(self, graph):
        """
        Calculates the total weight of all edges in the graph.
        """
        return sum(weight for _, _, weight in graph.edges.data('weight'))

    def kruskal(self, graph):
        """
        Implements Kruskal's algorithm to find the minimum spanning tree (MST) of the given graph.
        Returns the MST as a new graph.
        """
        # Initialize sets for each vertex
        for vertex in graph.nodes():
            self.create_set(vertex)

        # Create an empty graph to store the MST
        mst = nx.Graph()

        # Sort edges by weight
        edges = sorted(graph.edges.data('weight'), key=lambda x: x[2])

        # Process each edge in order of increasing weight
        for vertex1, vertex2, weight in edges:
            if self.find(vertex1) != self.find(vertex2):
                self.union(vertex1, vertex2)
                mst.add_edge(vertex1, vertex2, weight=weight)

        return mst
