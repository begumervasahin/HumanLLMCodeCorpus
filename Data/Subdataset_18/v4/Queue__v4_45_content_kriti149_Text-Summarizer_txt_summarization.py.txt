import bs4 as bs
import urllib.request
import re
import nltk
import heapq

# Download necessary NLTK resources
nltk.download('stopwords')
nltk.download('punkt')

def fetch_wikipedia_article(url):
    """
    Fetches the content of a Wikipedia article.

    Parameters:
    url (str): The URL of the Wikipedia article.

    Returns:
    str: The raw text content of the article.
    """
    source = urllib.request.urlopen(url).read()
    soup = bs.BeautifulSoup(source, 'lxml')
    text = ""
    for paragraph in soup.find_all('p'):
        text += paragraph.text
    return text

def clean_text(text):
    """
    Cleans the text by removing references, non-alphabetic characters, and extra spaces.

    Parameters:
    text (str): The raw text to clean.

    Returns:
    str: The cleaned text.
    """
    text = re.sub(r'\[[0-9]*\]', ' ', text)
    text = re.sub(r'\s+', ' ', text)
    clean_text = text.lower()
    clean_text = re.sub(r'\W', ' ', clean_text)
    clean_text = re.sub(r'\d', ' ', clean_text)
    clean_text = re.sub(r'\s+', ' ', clean_text)
    return clean_text

def score_sentences(sentences, word2count):
    """
    Scores sentences based on word frequency.

    Parameters:
    sentences (list): List of sentences to score.
    word2count (dict): Dictionary of word frequencies.

    Returns:
    dict: Dictionary of sentence scores.
    """
    sent2score = {}
    for sentence in sentences:
        for word in nltk.word_tokenize(sentence.lower()):
            if word in word2count:
                if len(sentence.split(' ')) < 25:
                    if sentence not in sent2score:
                        sent2score[sentence] = word2count[word]
                    else:
                        sent2score[sentence] += word2count[word]
    return sent2score

def main():
    # Fetch and clean the Wikipedia article
    article_url = 'https://en.wikipedia.org/wiki/Global_warming'
    raw_text = fetch_wikipedia_article(article_url)
    clean_text_content = clean_text(raw_text)

    # Tokenize sentences and remove stopwords
    sentences = nltk.sent_tokenize(raw_text)
    stop_words = set(nltk.corpus.stopwords.words('english'))

    # Calculate word frequencies
    word2count = {}
    for word in nltk.word_tokenize(clean_text_content):
        if word not in stop_words:
            if word not in word2count:
                word2count[word] = 1
            else:
                word2count[word] += 1

    # Normalize word frequencies
    max_freq = max(word2count.values())
    for word in word2count:
        word2count[word] = word2count[word] / max_freq

    # Score sentences
    sent2score = score_sentences(sentences, word2count)

    # Select the best sentences
    best_sentences = heapq.nlargest(5, sent2score, key=sent2score.get)
    
    # Print the summary
    print('------------------------------------------------------------')
    for sentence in best_sentences:
        print(sentence)

if __name__ == '__main__':
    main()
