def heapify(arr, n, i):
    largest = i        # Initialize largest as the root
    left = 2 * i + 1   # Left child index
    right = 2 * i + 2  # Right child index

    # If the left child exists and is smaller than the root, update largest
    if left < n and arr[i] > arr[left]:
        largest = left

    # If the right child exists and is smaller than the current largest, update largest
    if right < n and arr[largest] > arr[right]:
        largest = right

    # If largest is not the root, swap and continue heapifying
    if largest != i:
        arr[i], arr[largest] = arr[largest], arr[i]
        heapify(arr, n, largest)

def heap_sort(arr):
    n = len(arr)

    # Build a max heap
    for i in range(n // 2 - 1, -1, -1):
        heapify(arr, n, i)

    # Extract elements one by one from the heap
    for i in range(n - 1, 0, -1):
        arr[i], arr[0] = arr[0], arr[i]  # Move current root to the end
        heapify(arr, i, 0)               # Call heapify on the reduced heap
