This program checks if selected Turkish 'stop words' (filler words) are among the top 10,000 most frequently used words.
The word 'zxtest' is included specifically for control purposes.
"""

# Create a dictionary to hold the frequency of words
dictionary = {}

# Open and read the frequency file
with open("dosyalar/derlemtr2016-10000.txt", 'r', encoding='utf-8') as fdict:
    for line in fdict:
        # Skip lines that do not start with '0' or '1'
        if line[0] not in ['0', '1']:
            continue
        freq, word = line.strip().split()
        dictionary[word] = int(freq)

def is_stop_word(word):
    """
    Checks if the given word is in the dictionary.

    Parameters:
        word (str): The word to check.

    Returns:
        bool: True if the word is in the dictionary, False otherwise.
    """
    return word in dictionary

def get_word_freq(word):
    """
    Returns the frequency of the given word.

    Parameters:
        word (str): The word to check.

    Returns:
        int: The frequency of the word if it is in the dictionary, 0 otherwise.
    """
    return dictionary.get(word, 0)

if __name__ == '__main__':
    # Open and read the stop words file
    with open("dosyalar/turkce-stop-words", 'r', encoding='utf-8') as ftest:
        for word in ftest:
            word = word.strip()
            if not is_stop_word(word):
                print(f'{word} is not in the dictionary')
