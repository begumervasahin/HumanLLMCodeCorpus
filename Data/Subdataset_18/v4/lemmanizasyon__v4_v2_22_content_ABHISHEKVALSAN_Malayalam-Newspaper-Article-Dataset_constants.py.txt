import requests
from bs4 import BeautifulSoup

# Define the base URL of the news website
URL_PREFIX = 'https://www.janmabhumidaily.com/news'

def fetch_news_content(url):
    """
    Fetches the content of the provided URL.

    Parameters:
    url (str): The URL of the news website.

    Returns:
    content (str or None): The content of the response if successful, None otherwise.
    """
    try:
        response = requests.get(url)
        if response.status_code == 200:
            return response.content
        else:
            print(f"Failed to fetch content. Status code: {response.status_code}")
            return None
    except Exception as e:
        print(f"An error occurred: {e}")
        return None

def parse_news_content(content):
    """
    Parses and processes the news content to extract and print headlines.

    Parameters:
    content (str): The HTML content of the news website.
    """
    if content:
        soup = BeautifulSoup(content, 'html.parser')
        headlines = soup.find_all('h2', class_='entry-title')

        for headline in headlines:
            print(headline.text.strip())
    else:
        print("No content to parse.")

def main():
    """
    Main function to orchestrate fetching and parsing news content.
    """
    full_url = URL_PREFIX
    content = fetch_news_content(full_url)
    parse_news_content(content)

if __name__ == "__main__":
    main()
