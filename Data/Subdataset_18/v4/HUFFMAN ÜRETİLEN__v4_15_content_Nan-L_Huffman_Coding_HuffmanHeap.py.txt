class HuffmanHeap:
    """
    A heap structure to manage Huffman trees during the encoding process.
    
    Attributes:
        old (list): The old list of Huffman trees, sorted by frequency.
        new (list): The new list of Huffman trees, sorted by frequency.
    """
    
    def __init__(self, old, new=None):
        """
        Initialize the HuffmanHeap with old and new Huffman trees.
        
        Args:
            old (list): A list of sorted Huffman trees.
            new (list, optional): A list of sorted Huffman trees. Defaults to an empty list.
        """
        self.old = old
        self.new = new if new is not None else []

    def enqueue(self, item):
        """
        Add a Huffman tree to the new list.
        
        Args:
            item: A Huffman tree to be added to the heap.
        """
        self.new.append(item)

    def dequeue(self):
        """
        Remove and return the smallest item from the heap.
        
        Returns:
            The Huffman tree with the smallest frequency. Returns None if both heaps are empty.
        """
        if not self.old and self.new:
            return self.new.pop(0)
        elif not self.new and self.old:
            return self.old.pop(0)
        elif self.old and self.new:
            if self.old[0].get_freq() >= self.new[0].get_freq():
                return self.new.pop(0)
            else:
                return self.old.pop(0)
        else:
            print("Both old and new heaps are empty!")
            return None
