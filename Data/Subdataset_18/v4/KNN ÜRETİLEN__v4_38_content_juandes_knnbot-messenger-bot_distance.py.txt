import math

def distance(data1, data2):
    """
    Calculate the Euclidean distance between two data points.
    
    Parameters:
    data1 (list): The first data point.
    data2 (list): The second data point.
    
    Returns:
    float: The Euclidean distance between the two data points.
    """
    
    # Zip the two data points to get pairs of corresponding elements
    paired_points = zip(data1, data2)
    
    # Calculate the squared differences for each pair of elements
    squared_differences = [(a - b) ** 2 for a, b in paired_points]
    
    # Sum the squared differences and take the square root to get the distance
    return math.sqrt(sum(squared_differences))

# Example usage
point1 = [2, 3, 4]
point2 = [5, 6, 7]
print("Distance:", distance(point1, point2))
