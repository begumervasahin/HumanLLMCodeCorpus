import sys
import csv
import time

# List of keywords to search for in tweets
keywords = [
    'abuse', 'eatingdisorders', 'nostigma', 'presspause', 'addiction', 'endthestigma', 'nostigmas', 'mentalhealthmatters',
    'alzheimers', 'IAmStigmaFree', '1SmallAct', 'ocd', 'anxiety', 'mentalhealth', 'psychology', 'suicideprevention', 'bipolar',
    'pts', 'mhchat', 'therapy', 'bpd', 'schizophrenia', 'trauma', 'Operationalstress', 'ptsd', 'mhsm', 'worldmentalhealthday',
    'AA', 'stress', 'spsm', 'wellbeing', 'alcoholism', 'mentalhealthawareness', 'stopsuicide', 'adhd', 'depressed', 'mentalillness',
    'suicide', 'MH', 'shellshock', 'bts'
]

# Open the output CSV file to store filtered tweets
with open('filtered_tweets_per_abused_user.csv', 'w', newline='') as csvStorage:
    original_time = time.time()
    csv_writer = csv.writer(csvStorage)

    # Read input data from standard input
    data = csv.reader(iter(sys.stdin.readline, ''))

    # Process each row in the input data
    for row in data:
        added = False
        # Split the tweet text into words and check for keywords
        for word in row[1].split():
            if word in keywords:
                # If a keyword is found and the tweet has not been added yet, write it to the CSV
                if not added:
                    csv_writer.writerow([row[0], row[1]])
                    added = True
                    break
