class Node:
    def __init__(self, data=None):
        self.data = data
        self.next = None

class LinkedList:
    def __init__(self):
        self.head = Node()

    def append(self, data):
        """Append a new node with the given data to the end of the list."""
        new_node = Node(data)
        cur = self.head
        while cur.next:
            cur = cur.next
        cur.next = new_node

    def length(self):
        """Return the length of the linked list."""
        cur = self.head
        total = 0
        while cur.next:
            total += 1
            cur = cur.next
        return total

    def display(self):
        """Print all elements in the linked list."""
        elems = []
        cur_node = self.head
        while cur_node.next:
            cur_node = cur_node.next
            elems.append(cur_node.data)
        print(elems)

    def get(self, index):
        """Get the data of the node at the given index."""
        if index >= self.length():
            print('Index out of range')
            return None
        cur_idx = 0
        cur_node = self.head
        while cur_node.next:
            cur_node = cur_node.next
            if cur_idx == index:
                return cur_node.data
            cur_idx += 1

    def erase(self, index):
        """Erase the node at the given index."""
        if index >= self.length():
            print('Index out of range')
            return
        cur_idx = 0
        cur_node = self.head
        while cur_node.next:
            last_node = cur_node
            cur_node = cur_node.next
            if cur_idx == index:
                last_node.next = cur_node.next
                return
            cur_idx += 1

# Example usage
my_list = LinkedList()
my_list.append(2)
my_list.append(1)
my_list.erase(1)
my_list.display()
