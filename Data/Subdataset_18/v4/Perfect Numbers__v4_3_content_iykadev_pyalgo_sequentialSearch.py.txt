def sequential_search(alist, item):
    """
    Perform a sequential search to find an item in a list.
    
    Parameters:
    alist (list): The list to search through.
    item: The item to search for in the list.
    
    Returns:
    bool: True if the item is found, False otherwise.
    """
    pos = 0
    found = False
    
    while pos < len(alist) and not found:
        if alist[pos] == item:
            found = True
        else:
            pos += 1
    
    return found

# Example usage
testlist = [1, 2, 32, 8, 17, 19, 42, 13, 0]
print(sequential_search(testlist, 3))  # Output: False
print(sequential_search(testlist, 13)) # Output: True
