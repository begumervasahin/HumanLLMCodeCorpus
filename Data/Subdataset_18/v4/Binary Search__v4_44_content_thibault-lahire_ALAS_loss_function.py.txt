import numpy as np

def phi(w):
    """
    Sigmoid function.
    """
    return 1 / (1 + np.exp(-w))

def phiprim(w):
    """
    Derivative of the sigmoid function.
    """
    exp_neg_w = np.exp(-w)
    
    if exp_neg_w < 10**16:
        if exp_neg_w > 10**-16:
            # General case
            result = exp_neg_w / ((1 + exp_neg_w) ** 2)
        else:
            # Handles small exponentials to avoid underflow
            result = exp_neg_w
    else:
        # Handles large exponentials to avoid overflow
        result = np.exp(w)
    
    return result

def phi2prim(w):
    """
    Second derivative of the sigmoid function.
    """
    exp_neg_w = np.exp(-w)
    
    if exp_neg_w < 10**16:
        if exp_neg_w > 10**-16:
            # General case
            result = -np.exp(w) * (np.exp(w) - 1) / ((1 + np.exp(w)) ** 3)
        else:
            # Handles small exponentials to avoid underflow
            result = -exp_neg_w
    else:
        # Handles large exponentials to avoid overflow
        result = np.exp(w)
    
    return result

def loss(x, S, A, y, arg=1, penalization=False):
    """
    Compute the empirical loss function, its gradient, and Hessian.

    Parameters:
    x: ndarray, point at which the loss is computed
    S: list of int, subset of indices
    A: ndarray, matrix of datapoints
    y: ndarray, vector of labels
    arg: int, number of outputs (1 for loss, 2 for loss and gradient, 3 for loss, gradient, and Hessian)
    penalization: bool, if True, use a penalized loss function

    Returns:
    Depending on the value of `arg`, returns:
    - The loss value (arg == 1)
    - The loss value and its gradient (arg == 2)
    - The loss value, its gradient, and its Hessian (arg == 3)
    """
    d = A.shape[1]  # Number of features
    card_S = len(S)  # Size of subset S

    # Initialize the loss, gradient, and Hessian
    f = 0
    g = np.zeros(d) if arg >= 2 else None
    H = np.zeros((d, d)) if arg == 3 else None

    for i in range(card_S):
        y_S_i = y[S[i]]
        A_S_i = A[S[i]]
        scal_prod = np.dot(x, A_S_i)
        
        # Compute the loss contribution
        f += (y_S_i - phi(scal_prod)) ** 2
        
        if arg >= 2:
            # Compute the gradient contribution
            g += 2 * A_S_i * phiprim(scal_prod) * (phi(scal_prod) - y_S_i)
        
        if arg == 3:
            # Compute the Hessian contribution
            H += 2 * ((phi(scal_prod) - y_S_i) * phi2prim(scal_prod) + \
                      phiprim(scal_prod) ** 2) * np.outer(A_S_i, A_S_i)
    
    # Apply penalization if requested
    if penalization:
        f += 0.5 * np.linalg.norm(x) ** 2
    
    # Normalize the loss, gradient, and Hessian by the size of subset S
    f /= card_S
    if arg >= 2:
        if penalization:
            g += x
        g /= card_S
    if arg == 3:
        H = (H + H.T) / (2 * card_S)  # Ensure the Hessian is symmetric
        if penalization:
            H += np.eye(d) / card_S
    
    # Return the results based on the requested argument
    if arg == 3:
        return f, g, H
    elif arg == 2:
        return f, g
    else:
        return f
