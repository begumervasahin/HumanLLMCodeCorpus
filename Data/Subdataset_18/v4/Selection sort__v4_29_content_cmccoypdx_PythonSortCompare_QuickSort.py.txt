import random

def hoare_partition(arr, low, high):
    """Partition the array using Hoare's partition scheme."""
    pivot = arr[low]
    i = low
    j = high + 1
    
    while True:
        while True:
            i += 1
            if i >= len(arr) or arr[i] >= pivot:
                break
        
        while True:
            j -= 1
            if j <= 0 or arr[j] <= pivot:
                break
        
        if i >= j:
            break
        
        arr[i], arr[j] = arr[j], arr[i]
    
    arr[low], arr[j] = arr[j], arr[low]
    return j

def quick_sort(arr, low, high):
    """Recursive quicksort function."""
    if low < high:
        pivot_index = hoare_partition(arr, low, high)
        quick_sort(arr, low, pivot_index - 1)
        quick_sort(arr, pivot_index + 1, high)

def quicksort(arr):
    """Sort the array using quicksort algorithm."""
    quick_sort(arr, 0, len(arr) - 1)

if __name__ == "__main__":
    sample_list = [random.randint(0, 100) for _ in range(10)]
    print(f"Unsorted list: {sample_list}")
    quicksort(sample_list)
    print(f"Sorted list: {sample_list}")
