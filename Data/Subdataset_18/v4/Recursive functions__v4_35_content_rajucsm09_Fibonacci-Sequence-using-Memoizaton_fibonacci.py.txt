# Declare a Python dictionary to work as a cache for storing and searching Fibonacci calculations
fibonacci_cache = {}

def fibonacci(num):
    """
    Calculate the Fibonacci number using memoization to cache results.

    Args:
        num (int): The position in the Fibonacci sequence.

    Returns:
        int: The Fibonacci number at the specified position.
        bool: False if the input is not an integer.
    """
    if not isinstance(num, int):
        return False

    if num in fibonacci_cache:
        return fibonacci_cache[num]

    if num == 0:
        value = 0
    elif num == 1:
        value = 1
    else:
        value = fibonacci(num - 1) + fibonacci(num - 2)

    fibonacci_cache[num] = value
    return value

# Print the first 1000 Fibonacci numbers and observe system behavior
for num in range(1000):
    print(f"{num}: {fibonacci(num)}")
