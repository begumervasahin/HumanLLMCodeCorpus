def partition(arr, left, right):
    # Base case: if the left index is greater than or equal to the right, no partition is needed
    if left >= right:
        return right
    
    # Initialize the pivot and split index
    pivot = arr[right]
    split = left

    # Debugging output
    print(f"Pivot: {pivot}")

    # Partition the array based on the pivot value
    for i in range(left, right + 1):
        print(f"Index: {i}")

        # If the current element is less than or equal to the pivot
        if arr[i] <= pivot:
            # Swap elements if necessary
            if i != split:
                arr[i], arr[split] = arr[split], arr[i]
            split += 1
    
    # Return the split index minus one, as the pivot is now correctly placed
    return split - 1

def quicksort(arr, left, right):
    # Base case: If the subarray has more than one element
    if left < right:
        # Partition the array and get the split index
        split = partition(arr, left, right)
        
        # Debugging output
        print(f"Split index: {split}, Left: {left}, Right: {right}")
        print(f"Array after partition: {arr}")
        
        # Recursively apply quicksort to the left and right subarrays
        if split >= left + 1:
            quicksort(arr, left, split - 1)
        if split <= right - 1:
            quicksort(arr, split + 1, right)

# Example usage
arr = [3, 6, 8, 10, 1, 2, 1]
quicksort(arr, 0, len(arr) - 1)
print(f"Sorted array: {arr}")
