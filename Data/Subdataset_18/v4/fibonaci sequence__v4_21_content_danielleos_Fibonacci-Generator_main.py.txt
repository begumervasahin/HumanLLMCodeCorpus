def generate_fibonacci_sequence(count):
    """
    Generate a list of Fibonacci numbers.

    Args:
        count (int): The number of Fibonacci numbers to generate.

    Returns:
        list: A list containing the Fibonacci sequence.
    """
    if count < 1:
        return []

    fib_list = [0, 1]
    while len(fib_list) < count:
        fib_list.append(fib_list[-1] + fib_list[-2])
    
    return fib_list[:count]

def main():
    while True:
        try:
            count = int(input("How many Fibonacci numbers do you want to see? "))
            if count <= 0:
                raise ValueError("Number must be positive.")
        except ValueError:
            print("Sorry, I did not understand that input, please input a positive whole number.")
            continue

        print(f"Producing {count} Fibonacci numbers...")
        fib_sequence = generate_fibonacci_sequence(count)
        print(fib_sequence)
        break

if __name__ == "__main__":
    main()
