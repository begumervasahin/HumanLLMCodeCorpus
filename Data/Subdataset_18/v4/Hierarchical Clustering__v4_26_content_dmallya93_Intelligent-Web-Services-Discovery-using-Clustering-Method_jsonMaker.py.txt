import os
import json

# Dictionary to store cluster names
cluster_names = {}

def read_cluster_names(filename="ClusterNames.csv"):
    """
    Reads cluster names from a CSV file and stores them in a global dictionary.

    Parameters:
    filename (str): The path to the CSV file containing cluster names.
    """
    global cluster_names
    
    with open(filename, "r") as file:
        for line in file:
            fields = line.strip().split(",")
            if len(fields) == 2:
                cluster_id, cluster_name = fields
                cluster_names[cluster_id] = cluster_name

def generate_json(input_file="Hierarchies.csv", output_file="newjson.json"):
    """
    Generates a JSON structure from hierarchy data and writes it to a file.

    Parameters:
    input_file (str): The path to the CSV file containing hierarchy data.
    output_file (str): The path to the output JSON file.
    """
    json_data = {
        "name": "Entire Dataset",
        "children": []
    }

    with open(input_file, "r") as file:
        for line in file:
            fields = line.strip().split(",")
            if len(fields) == 2:
                parent_name, clusters_str = fields
                cluster_ids = clusters_str.split()

                children = [
                    {"name": cluster_names.get(cluster, cluster), "size": 3000}
                    for cluster in cluster_ids
                ]
                
                json_data["children"].append({
                    "name": parent_name,
                    "children": children
                })

    # Write the JSON data to a file
    with open(output_file, "w") as output:
        json.dump(json_data, output, indent=4)

def main():
    # Read cluster names from the CSV file
    read_cluster_names()
    print("Cluster Names:", cluster_names)
    
    # Generate JSON and write it to a file
    generate_json()

if __name__ == "__main__":
    main()
