class Clustering:
    """ 
    This class applies the K-Means algorithm on the iris dataset to group similar species together. 
    """
    
    def __init__(self, clustering_name):
        self.clustering_name = clustering_name
    
    def cluster(self, X_train, X_test, number_clusters):
        """
        Trains the model and clusters the X_test dataset based on the provided number of clusters.
        
        Parameters:
        X_train (array-like): Training data.
        X_test (array-like): Test data to be clustered.
        number_clusters (int): Number of clusters to form.
        
        Returns:
        array-like: Predicted cluster labels for X_test.
        """
        if self.clustering_name == "K-Means":
            from sklearn.cluster import KMeans
            kmeans = KMeans(n_clusters=number_clusters, random_state=100)
            kmeans.fit(X_train)
            return kmeans.predict(X_test)
    
    def tune_parameters(self, X_train):
        """
        Tunes the number of clusters parameter by running multiple iterations 
        and calculating the inertia for different values of the number of clusters.
        
        Parameters:
        X_train (array-like): Training data.
        
        Returns:
        tuple: The number of clusters with the lowest inertia and the corresponding inertia.
        """
        from sklearn.cluster import KMeans
        
        n_clusters = [1, 2, 3, 4]
        min_inertia = float('inf')
        best_n_clusters = n_clusters[0]
        
        print("\nTuning the number of clusters parameter...\n")
        
        for n in n_clusters:
            kmeans = KMeans(n_clusters=n, random_state=100)
            kmeans.fit(X_train)
            inertia = kmeans.inertia_
            
            print(f"The number of clusters: {n}, corresponding inertia: {inertia}")
            
            if inertia < min_inertia:
                min_inertia = inertia
                best_n_clusters = n
        
        print(f"\nThe chosen number of clusters is: {best_n_clusters} with corresponding inertia of: {min_inertia}")
        
        return best_n_clusters, min_inertia
