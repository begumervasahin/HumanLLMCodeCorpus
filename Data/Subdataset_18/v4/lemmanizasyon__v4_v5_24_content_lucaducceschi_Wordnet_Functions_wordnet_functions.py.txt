import nltk
from nltk.corpus import wordnet as wn

def get_hyponyms(noun):
    """
    Get hyponyms (more specific terms) for a given noun.
    
    Args:
    noun (str): The noun for which to find hyponyms.
    
    Returns:
    list: A list of lists containing hyponyms for each synset of the noun.
    """
    # Retrieve synsets containing the noun
    synsets_with_noun = wn.synsets(noun)

    # Filter synsets to include only those that contain the noun as a lemma
    filtered_synsets = [synset for synset in synsets_with_noun if noun in synset.lemma_names()]

    # Define a closure function to get hyponyms of a synset
    def hyponyms_closure(synset):
        return synset.hyponyms()

    # Get hyponyms for each filtered synset
    hyponyms_list = [list(synset.closure(hyponyms_closure, depth=1)) for synset in filtered_synsets]

    return hyponyms_list

def get_hypernyms(noun):
    """
    Get hypernyms (more general terms) for a given noun.
    
    Args:
    noun (str): The noun for which to find hypernyms.
    
    Returns:
    tuple: A tuple containing a list of synset strings and a list of lists containing hypernyms for each synset.
    """
    # Retrieve synsets containing the noun
    synsets_with_noun = wn.synsets(noun)

    # Extract strings representing synsets containing the noun
    synset_strings = [str(synset) for synset in synsets_with_noun if noun in synset.lemma_names()]

    # Define a closure function to get hypernyms of a synset
    def hypernyms_closure(synset):
        return synset.hypernyms()

    # Get hypernyms for each synset string
    hypernyms_list = [list(wn.synset(synset_str).closure(hypernyms_closure, depth=8)) for synset_str in synset_strings]

    return synset_strings, hypernyms_list

def universal_pos_converter(tag):
    """
    Convert universal POS tags to WordNet POS tags.
    
    Args:
    tag (str): Universal POS tag.
    
    Returns:
    str: Corresponding WordNet POS tag.
    """
    pos_tags_mapping = {
        "NOUN": "n",
        "VERB": "v",
        "ADJ": "a",
        "ADV": "r"
    }
    return pos_tags_mapping.get(tag)

def lemmatize_word(word_tuple):
    """
    Lemmatize a word based on its POS tag.
    
    Args:
    word_tuple (tuple): A tuple containing the word and its POS tag.
    
    Returns:
    tuple: A tuple containing the lemmatized word and its original POS tag.
    """
    word, pos_tag = word_tuple
    lemma = wn.morphy(word, universal_pos_converter(pos_tag))
    return (lemma, pos_tag) if lemma else word_tuple

def tag_text(corpus):
    """
    Lemmatize each word in the corpus.
    
    Args:
    corpus (list): A list of lists containing tuples of words and their POS tags.
    
    Returns:
    list: A list of lists containing tuples of lemmatized words and their POS tags.
    """
    return [[lemmatize_word(word) for word in line] for line in corpus]

if __name__ == "__main__":
    # Example usage
    noun = "dog"
    print(f"Hyponyms of {noun}: {get_hyponyms(noun)}")
    print(f"Hypernyms of {noun}: {get_hypernyms(noun)}")
    
    text = [[('running', 'VERB'), ('quickly', 'ADV')], [('dogs', 'NOUN'), ('barking', 'VERB')]]
    print(f"Lemmatized text: {tag_text(text)}")
