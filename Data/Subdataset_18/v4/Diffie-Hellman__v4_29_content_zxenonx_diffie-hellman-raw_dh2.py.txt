import random
import hashlib

# Define shared parameters for the key exchange
BASE = 9  # Shared base
PRIME = 1001  # Prime number

# Generate random secrets for Alice and Bob
alice_secret = random.randint(5, 10)
bob_secret = random.randint(10, 20)

# Compute public values
alice_public_value = pow(BASE, alice_secret, PRIME)
bob_public_value = pow(BASE, bob_secret, PRIME)

# Display shared parameters and computed values
print(f"Shared Parameters:")
print(f"  Base (g): {BASE}")
print(f"  Prime (p): {PRIME}")

print("\nAlice's Calculations:")
print(f"  Alice's secret (a): {alice_secret}")
print(f"  Alice's public value (A): {alice_public_value} (g^a mod p)")

print("\nBob's Calculations:")
print(f"  Bob's secret (b): {bob_secret}")
print(f"  Bob's public value (B): {bob_public_value} (g^b mod p)")

# Compute shared keys and display them
print("\nShared Key Computation:")

# Alice computes the shared key using Bob's public value
alice_shared_key = pow(bob_public_value, alice_secret, PRIME)
print(f"  Alice's shared key: {alice_shared_key} (B^a mod p)")
print(f"  SHA-256 hash of Alice's shared key: {hashlib.sha256(str(alice_shared_key).encode()).hexdigest()}")

# Bob computes the shared key using Alice's public value
bob_shared_key = pow(alice_public_value, bob_secret, PRIME)
print(f"  Bob's shared key: {bob_shared_key} (A^b mod p)")
print(f"  SHA-256 hash of Bob's shared key: {hashlib.sha256(str(bob_shared_key).encode()).hexdigest()}")
