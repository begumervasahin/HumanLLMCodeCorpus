from collections import Counter
from minheap import MinHeap

class TreeLeafEndMessage:
    pass

class TreeLeaf:
    def __init__(self, value):
        self.value = value

class TreeBranch:
    def __init__(self, left, right):
        self.left = left
        self.right = right

def make_tree(freq_table):
    trees = MinHeap()
    
    # Adding a special end message to the heap
    trees.add(1, TreeLeafEndMessage())
    
    # Adding each symbol with its frequency as a tree leaf
    for symbol, freq in freq_table.items():
        trees.add(freq, TreeLeaf(symbol))
    
    # Combining trees until one remains
    while len(trees) > 1:
        rfreq, right = trees.pop_min()
        lfreq, left = trees.pop_min()
        trees.add(lfreq + rfreq, TreeBranch(left, right))
    
    # The last remaining tree is the complete Huffman tree
    total_freq, huffman_tree = trees.pop_min()
    return huffman_tree

def decode(tree, bitreader):
    while True:
        if isinstance(tree, TreeLeafEndMessage):
            return None
        elif isinstance(tree, TreeLeaf):
            return tree.value
        elif isinstance(tree, TreeBranch):
            tree = tree.left if bitreader.readbit() == 0 else tree.right
        else:
            raise TypeError(f'{type(tree)} is not a valid tree type')

def make_encoding_table(huffman_tree):
    encoding_table = {}

    def recurse(tree, path):
        if isinstance(tree, TreeLeafEndMessage):
            encoding_table[None] = path
        elif isinstance(tree, TreeLeaf):
            encoding_table[tree.value] = path
        elif isinstance(tree, TreeBranch):
            recurse(tree.left, path + (False,))
            recurse(tree.right, path + (True,))
        else:
            raise TypeError(f'{type(tree)} is not a valid tree type')

    recurse(huffman_tree, ())
    return encoding_table

def make_freq_table(stream):
    freqs = Counter()
    buffer = bytearray(512)
    
    while True:
        count = stream.readinto(buffer)
        freqs.update(buffer[:count])
        if count < len(buffer):
            break
    
    return freqs
