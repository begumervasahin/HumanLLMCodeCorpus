import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
import datetime
import numpy as np

def cluster_analysis(data_array, k):
    """
    Performs K-Means clustering on the input data and plots the resulting clusters along with cluster centers.

    Parameters:
        data_array (list of lists or np.ndarray): The input data points.
        k (int): The number of clusters to create.

    Returns:
        list: Cluster labels for each data point.
    """
    # Ensure data_array is a NumPy array for consistency
    data_array = np.array(data_array)

    # Initialize KMeans model with specified number of clusters
    model = KMeans(n_clusters=k, random_state=42)

    # Fit the model to the data and predict cluster labels
    labels = model.fit_predict(data_array)

    # Extract x and y coordinates from data_array
    xs = data_array[:, 0]
    ys = data_array[:, 1]

    # Extract cluster centers
    centroids = model.cluster_centers_
    centroids_x = centroids[:, 0]
    centroids_y = centroids[:, 1]

    # Plot data points with cluster colors
    plt.scatter(xs, ys, c=labels, alpha=0.5, cmap='viridis')

    # Plot cluster centers
    plt.scatter(centroids_x, centroids_y, marker='D', s=100, c='red', label='Centroids')

    # Set plot title with current date and time
    plt.title("Cluster Analysis - " + datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S"))
    plt.xlabel("Feature 1")
    plt.ylabel("Feature 2")
    plt.legend()

    # Display the plot
    plt.show()

    # Return the cluster labels
    return labels

# Example usage:
if __name__ == "__main__":
    # Example data
    data_array = np.array([[1, 2], [2, 3], [3, 4], [10, 9], [8, 7], [7, 6]])
    k = 2  # Number of clusters

    # Call the function
    cluster_labels = cluster_analysis(data_array, k)
    print("Cluster Labels:", cluster_labels)
