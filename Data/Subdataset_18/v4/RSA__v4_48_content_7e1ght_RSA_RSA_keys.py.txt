import random
import math

class Key:
    def __init__(self, bit_size):
        self.bit_size = bit_size

    def is_prime(self, n):
        """Check if a number is a prime."""
        if n in (2, 3):
            return True
        if n % 2 == 0 or n < 2:
            return False
        for i in range(3, int(math.sqrt(n)) + 1, 2):
            if n % i == 0:
                return False
        return True

    def generate_random_number(self):
        """Generate a random number with the specified bit size."""
        return random.randint(2**(self.bit_size - 1) + 1, 2**self.bit_size)

    def generate_prime(self):
        """Generate a prime number."""
        num = self.generate_random_number()
        while not self.is_prime(num):
            num = self.generate_random_number()
        return num

    def is_relatively_prime(self, a, b):
        """Check if two numbers are relatively prime."""
        return math.gcd(a, b) == 1

    def find_relative_prime(self, a):
        """Find a number that is relatively prime to 'a'."""
        for i in range(3, int(math.sqrt(a)) + 1, 2):
            if self.is_relatively_prime(i, a):
                return i
        raise Exception("Failed to find a relatively prime number.")

    def extended_gcd(self, a, b):
        """Extended Euclidean Algorithm to find the greatest common divisor."""
        if b == 0:
            return a, 1, 0
        else:
            d, x1, y1 = self.extended_gcd(b, a % b)
            x = y1
            y = x1 - (a // b) * y1
            return d, x, y
