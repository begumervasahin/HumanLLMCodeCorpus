from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from time import sleep

def setup_driver():
    """
    Sets up the Selenium WebDriver with Firefox.
    
    Returns:
        webdriver.Firefox: Configured WebDriver instance.
    """
    driver_path = 'D:/geckodriver/geckodriver.exe'
    return webdriver.Firefox(executable_path=driver_path)

def select_date(driver, date):
    """
    Selects a date in the date input field.

    Args:
        driver (webdriver.Firefox): The WebDriver instance.
        date (str): The date to input in 'YYYY-MM-DD' format.
    """
    date_input = driver.find_element(By.ID, 'fecha_salida')
    date_input.click()
    date_input.clear()  # Clear any existing value
    date_input.send_keys(date)

def print_dropdown_options(driver, dropdown_id):
    """
    Prints the values of options in a dropdown menu.

    Args:
        driver (webdriver.Firefox): The WebDriver instance.
        dropdown_id (str): The ID of the dropdown element.
    """
    dropdown = WebDriverWait(driver, 10).until(
        EC.presence_of_element_located((By.ID, dropdown_id))
    )
    options = dropdown.find_elements(By.TAG_NAME, 'option')
    for option in options:
        print(option.get_attribute('value'))
        sleep(1)  # Pause between each option for readability

def main():
    """
    Main function to execute the script logic.
    """
    driver = setup_driver()
    try:
        driver.get("http://checkyourflight.info")

        # Set the departure date
        select_date(driver, '2019-08-14')
        sleep(1)  # Allow time for the page to update
        
        # Print flight options
        print("Flight options:")
        print_dropdown_options(driver, 'flight')
        
        # Print hotel options for each flight option
        flight_dropdown = WebDriverWait(driver, 10).until(
            EC.presence_of_element_located((By.ID, 'flight'))
        )
        flights = flight_dropdown.find_elements(By.TAG_NAME, 'option')
        for flight in flights:
            flight.click()
            print(f"Selected flight: {flight.get_attribute('value')}")
            
            # Print hotel options related to the selected flight
            print("Hotel options:")
            print_dropdown_options(driver, 'hoteles')

            sleep(1)  # Pause between each flight for readability

        # Pause to observe results before closing
        sleep(100)
    finally:
        driver.quit()  # Ensure the browser is closed properly

if __name__ == "__main__":
    main()
