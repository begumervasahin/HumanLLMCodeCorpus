import random

def insertion_sort(arr):
    """
    Perform insertion sort on a list.
    Each element is compared to those preceding it, and greater elements are shifted right.
    """
    if not isinstance(arr, list):
        raise TypeError('The input must be a list containing numbers.')
    
    n = len(arr)
    if n < 2:
        return arr
    
    for i in range(1, n):
        key = arr[i]
        j = i - 1
        
        while j >= 0 and arr[j] > key:
            arr[j + 1] = arr[j]
            j -= 1
        arr[j + 1] = key
    
    return arr

def choose_pivot(arr, low, high, pivot_type):
    """
    Choose the pivot element based on the provided pivot type.
    The pivot is moved to the last position in the current sublist.
    
    pivot_type options: 'first', 'last', 'middle', 'random'.
    """
    pivot_index = {
        'first': low,
        'last': high,
        'middle': (low + high) // 2,
        'random': random.randint(low, high)
    }.get(pivot_type, low)  # Default to 'first' if pivot_type is invalid

    arr[pivot_index], arr[high] = arr[high], arr[pivot_index]
    return arr[high], arr

def partition(arr, low, high, pivot_type):
    """
    Partition the list around the pivot, placing elements smaller than the pivot to its left and greater to its right.
    """
    pivot, arr = choose_pivot(arr, low, high, pivot_type)
    i = low - 1

    for j in range(low, high):
        if arr[j] <= pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]

    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    return i + 1

def quicksort(arr, low=0, high=None, pivot_type='last', use_insertion_sort=False, threshold=10):
    """
    Perform quicksort on a list recursively. Insertion sort is used for small sublists when specified.
    
    Parameters:
        arr: The list to sort.
        low: Starting index of the list/sublist.
        high: Ending index of the list/sublist.
        pivot_type: Strategy to select the pivot ('first', 'last', 'middle', 'random').
        use_insertion_sort: If True, use insertion sort for sublists smaller than `threshold`.
        threshold: Maximum size of the sublist to use insertion sort.
    """
    if not isinstance(arr, list):
        raise TypeError('The input must be a list containing numbers.')
    
    if high is None:
        high = len(arr) - 1
    
    if low < high:
        if use_insertion_sort and (high - low) < threshold:
            arr[low:high + 1] = insertion_sort(arr[low:high + 1])
        else:
            pivot_index = partition(arr, low, high, pivot_type)
            quicksort(arr, low, pivot_index - 1, pivot_type, use_insertion_sort, threshold)
            quicksort(arr, pivot_index + 1, high, pivot_type, use_insertion_sort, threshold)

    return arr
