import numpy as np

def main():
    """
    Demonstrates operations on a NumPy matrix created from a formatted string.
    """
    # Define the matrix as a string with semicolons separating rows
    matrix_str = '1 2 3 4 ; 5 6 7 8 ; 0 1 2 3'
    
    # Replace semicolons with newlines to convert the string into a matrix format
    formatted_str = matrix_str.replace(';', '\n')
    
    # Create a NumPy matrix from the formatted string
    matrix = np.matrix(formatted_str)
    
    # Print the shape of the matrix
    print("Matrix shape:", matrix.shape)
    
    # Access and print specific elements from the matrix
    print("Element at position (0, 1):", matrix[0, 1])
    print("Element at position (2, 3):", matrix[2, 3])
    
    # Modify an element in the matrix
    matrix[0, 1] = -8
    
    # Print the modified row
    print("Modified row 1:", matrix[1])
    
    # Print the entire column
    print("Column 1:", matrix[:, 1])

if __name__ == '__main__':
    main()
