def hex_to_nums(msg):
    """
    Converts a message of hexadecimal pairs to a list of decimal numbers.

    Args:
        msg (str): String containing a stream of hexadecimal bytes.

    Returns:
        list: List of integers representing the decimal equivalents of the hex pairs.
    """
    len_msg = len(msg) // 2
    bytes_array = [0] * len_msg

    for i in range(len_msg):
        digit1 = hex_num_helper(msg[2 * i])
        digit2 = hex_num_helper(msg[2 * i + 1])
        bytes_array[i] = 16 * digit1 + digit2

    return bytes_array

def nums_to_hex(nums):
    """
    Converts a list of integers to their 2-digit hexadecimal string representations.

    Args:
        nums (list): List of integers.

    Returns:
        list: List of strings representing the hexadecimal values of the integers.
    """
    return [hex(n)[2:].upper().zfill(2) for n in nums]

def hex_num_helper(hex_val):
    """
    Determines the decimal equivalent of a hex digit.

    Args:
        hex_val (str): A single hexadecimal digit.

    Returns:
        int: Decimal equivalent of the hex digit.
    """
    if hex_val.isdigit():
        return int(hex_val)
    return ord(hex_val.upper()) - 55

def bytes_to_chars(bytes_array):
    """
    Converts a list of bytes to their corresponding character representations.

    Args:
        bytes_array (list): List of byte values.

    Returns:
        list: List of characters corresponding to the byte values.
    """
    return [chr(b) for b in bytes_array]

def xor_chrs(chrs):
    """
    Returns the XOR of all characters in the input list.

    Args:
        chrs (list): List of characters.

    Returns:
        str: A single character resulting from the XOR operation.
    """
    xor_result = 0
    for c in chrs:
        xor_result ^= ord(c)
    return chr(xor_result)
