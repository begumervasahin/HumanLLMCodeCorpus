import queue as Q

# Define a custom comparison method for strings based on their length
def string_length_comparator(self, other):
    return len(self) < len(other)

# Attach the comparison method to the str class
str.__lt__ = string_length_comparator

# Initialize a priority queue
heap = Q.PriorityQueue()

# Add elements to the priority queue
heap.put('jeej')
heap.put('kek')
heap.put('topkek')
heap.put('non')

# Print the shortest string in the queue
shortest_string = sorted(heap.queue)[0]
print(shortest_string)
