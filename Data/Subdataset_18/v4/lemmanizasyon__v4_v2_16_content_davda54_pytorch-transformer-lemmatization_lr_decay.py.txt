class LearningRateDecay:
    """
    A class to handle learning rate decay with warmup for optimizers.
    """
    def __init__(self, optimizers, hidden_dim, base_lr, warmup_steps, initial_step=0):
        """
        Initializes the LearningRateDecay instance.

        Args:
            optimizers (list): List of optimizers to update.
            hidden_dim (int): Hidden dimension size.
            base_lr (float): Base learning rate.
            warmup_steps (int): Number of warmup steps.
            initial_step (int, optional): Initial step count. Defaults to 0.
        """
        self.optimizers = optimizers
        self.hidden_dim = hidden_dim
        self.base_lr = base_lr
        self.warmup_steps = warmup_steps
        self.step = initial_step

    def __call__(self):
        """
        Updates the learning rate based on the current step and applies it to all optimizers.

        Returns:
            float: The updated learning rate.
        """
        # Increment the step counter
        self.step += 1

        # Calculate the learning rate based on the current step
        if self.step < self.warmup_steps:
            lr = (self.hidden_dim ** -0.5) * self.step * (self.warmup_steps ** -1.5) * self.base_lr
        else:
            lr = 0.5 * (self.hidden_dim ** -0.5) * (self.step ** -0.5) * self.base_lr

        # Update the learning rate for each optimizer
        for optimizer in self.optimizers:
            for param_group in optimizer.param_groups:
                param_group["lr"] = lr

        return lr


# Example usage:
# Assuming optimizer1 and optimizer2 are defined and instantiated elsewhere
optimizers = [optimizer1, optimizer2]
hidden_dim = 512
base_lr = 0.001
warmup_steps = 1000

# Initialize the LearningRateDecay object
lr_decay = LearningRateDecay(optimizers, hidden_dim, base_lr, warmup_steps)

# Call the LearningRateDecay object to update and retrieve the current learning rate
current_learning_rate = lr_decay()
print("Current learning rate:", current_learning_rate)
