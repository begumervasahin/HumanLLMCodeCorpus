import networkx as nx
import math
import matplotlib.pyplot as plt
from fibonacci_heap import FibonacciHeap

# Function to build a heap from a set of nodes and their keys
def build_heap(V, k, heap):
    nodes_heap = dict()
    for v in V:
        nodes_heap[v] = heap.insert(k[v], v)
    return nodes_heap

# Create a graph
G = nx.Graph()
node_set = ('a', 'b', 'c', 'd')

# Set node positions
positions = {'a': (0, 2), 'b': (1, 0), 'c': (2, 2), 'd': (3, 0)}
for node in node_set:
    G.add_node(node, pos=positions[node], key=math.inf, pred=None, flag=False)

# Add edges with weights
edges = [
    ('a', 'b', 2),
    ('a', 'c', 10),
    ('b', 'c', 9),
    ('b', 'd', 1),
    ('d', 'a', 3)
]
G.add_weighted_edges_from(edges)

# Get edge weights and node positions
weights = nx.get_edge_attributes(G, 'weight')
positions = nx.get_node_attributes(G, 'pos')

# Draw the initial graph
plt.figure(figsize=(12, 6))
plt.subplot(121)
nx.draw(G, positions, with_labels=True, node_color='lightblue', node_size=500)
nx.draw_networkx_edge_labels(G, positions, edge_labels=weights)

# Initialize the starting node
start_node = 'a'
G.nodes[start_node]['key'] = 0

# Create a Fibonacci heap and build the initial heap of nodes
Q = FibonacciHeap()
keys = nx.get_node_attributes(G, 'key')
nodes_heap = build_heap(node_set, keys, Q)

# Prim's algorithm to find the Minimum Spanning Tree (MST)
while Q.total_nodes != 0:
    u = Q.extract_min().value
    G.nodes[u]['flag'] = True
    for v in G.neighbors(u):
        if not G.nodes[v]['flag'] and G[u][v]['weight'] < G.nodes[v]['key']:
            G.nodes[v]['key'] = G[u][v]['weight']
            G.nodes[v]['pred'] = u
            Q.decrease_key(nodes_heap[v], G[u][v]['weight'])

# Construct the MST graph
MST = nx.Graph()
for v in node_set:
    if G.nodes[v]['pred'] is not None:
        MST.add_edge(G.nodes[v]['pred'], v, weight=G[v][G.nodes[v]['pred']]['weight'])

# Get MST edge weights
mst_weights = nx.get_edge_attributes(MST, 'weight')

# Draw the MST graph
plt.subplot(122)
nx.draw(MST, positions, with_labels=True, node_color='lightgreen', node_size=500)
nx.draw_networkx_edge_labels(MST, positions, edge_labels=mst_weights)
plt.title('Minimum Spanning Tree')
plt.show()
