import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.preprocessing import PolynomialFeatures
from sklearn.feature_selection import f_regression, mutual_info_regression
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn import metrics

# Load the dataset
customers = pd.read_csv("Ecommerce Customers")

# Display the first few rows and descriptive statistics
print(customers.head())
print(customers.describe())

# Set the visual style
sns.set_palette("GnBu_d")
sns.set_style('whitegrid')

# Create visualizations
sns.jointplot(x='Time on Website', y='Yearly Amount Spent', data=customers)
sns.jointplot(x='Time on App', y='Yearly Amount Spent', data=customers)
sns.jointplot(x='Time on App', y='Length of Membership', kind='hex', data=customers)
sns.pairplot(customers)
sns.lmplot(x='Length of Membership', y='Yearly Amount Spent', data=customers)

# Define the target variable and features
y = customers['Yearly Amount Spent']
X = customers[['Avg. Session Length', 'Time on App', 'Time on Website', 'Length of Membership']]

# Apply polynomial features transformation
poly = PolynomialFeatures(degree=2, interaction_only=False, include_bias=True)
X_poly = poly.fit_transform(X)

# Feature selection using F-regression and mutual information regression
f_values, p_values = f_regression(X, y, center=True)
print("F-regression values:", f_values)
print("F-regression p-values:", p_values)

mi_scores = mutual_info_regression(X, y, discrete_features='auto', n_neighbors=3, copy=True, random_state=None)
print("Mutual information scores:", mi_scores)

# Split the data into training and testing sets
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=101)
X_poly_train, X_poly_test = train_test_split(X_poly, test_size=0.3, random_state=101)

# Drop 'Time on App' for another model
X_reduced = X.drop('Time on App', axis=1)
X_reduced_train, X_reduced_test = train_test_split(X_reduced, test_size=0.3, random_state=101)

# Initialize and train the linear regression models
lm = LinearRegression()
lm.fit(X_train, y_train)

lm_reduced = LinearRegression()
lm_reduced.fit(X_reduced_train, y_train)

lm_poly = LinearRegression()
lm_poly.fit(X_poly_train, y_train)

# Print the coefficients of the models
print('Model with all features Coefficients:', lm.coef_)
print('Model without "Time on App" Coefficients:', lm_reduced.coef_)
print('Polynomial Features Model Coefficients:', lm_poly.coef_)

# Make predictions using the models
predictions = lm.predict(X_test)
predictions_reduced = lm_reduced.predict(X_reduced_test)
predictions_poly = lm_poly.predict(X_poly_test)

# Plot the predictions
plt.figure(figsize=(14, 6))

plt.subplot(1, 3, 1)
plt.scatter(y_test, predictions)
plt.xlabel('Y Test')
plt.ylabel('Predicted Y')
plt.title('All Features')

plt.subplot(1, 3, 2)
plt.scatter(y_test, predictions_reduced)
plt.xlabel('Y Test')
plt.ylabel('Predicted Y')
plt.title('Without "Time on App"')

plt.subplot(1, 3, 3)
plt.scatter(y_test, predictions_poly)
plt.xlabel('Y Test')
plt.ylabel('Predicted Y')
plt.title('Polynomial Features')

plt.tight_layout()
plt.show()

# Evaluate the models
def evaluate_model(model_name, y_test, predictions):
    print(f'{model_name} Model Performance:')
    print('MAE:', metrics.mean_absolute_error(y_test, predictions))
    print('MSE:', metrics.mean_squared_error(y_test, predictions))
    print('RMSE:', np.sqrt(metrics.mean_squared_error(y_test, predictions)))
    print('\n')

evaluate_model('All Features', y_test, predictions)
evaluate_model('Without "Time on App"', y_test, predictions_reduced)
evaluate_model('Polynomial Features', y_test, predictions_poly)
