import sys
from math import sqrt, log

def is_prime(p):
    """Check if a number p is prime."""
    if p == 2:
        return True
    if p <= 1 or p % 2 == 0:
        return False
    for i in range(3, int(sqrt(p)) + 1, 2):
        if p % i == 0:
            return False
    return True

def is_mersenne_prime(p):
    """Check if 2^p - 1 is a Mersenne prime."""
    if p == 2:
        return True
    m_p = (1 << p) - 1
    s = 4
    for _ in range(3, p + 1):
        s = (s ** 2 - 2) % m_p
    return s == 0

def find_mersenne_primes(upb_prime, upb_count):
    """Find and print Mersenne primes up to a given upper bound."""
    print(f"Finding Mersenne primes in M[2..{upb_prime}]:")
    count = 0
    for p in range(2, upb_prime + 1):
        if is_prime(p) and is_mersenne_prime(p):
            print(f"M{p}", end=' ')
            sys.stdout.flush()
            count += 1
        if count >= upb_count:
            break
    print()

if __name__ == '__main__':
    precision = 20000
    long_bits_width = precision * log(10, 2)
    upb_prime = int(long_bits_width - 1) // 2
    upb_count = 45

    find_mersenne_primes(upb_prime, upb_count)
