def to_string(lst):
    """Convert a list to a string."""
    return ''.join(lst)

def print_to_n(n):
    """Print all numbers from 1 to n in increasing order."""
    if n >= 1:
        print_to_n(n - 1)
        print(n)

def print_reversed(n):
    """Print all numbers from n to 1 in decreasing order."""
    if n >= 1:
        print(n)
        print_reversed(n - 1)

def has_divisor_smaller_than(n, i):
    """Check if a number has a divisor smaller than the given number."""
    if i == 1 or i == 0:
        return False
    elif n % i == 0:
        return True
    return has_divisor_smaller_than(n, i - 1)

def is_prime(n):
    """Check if a number is prime."""
    if n < 2:
        return False
    i = int(n ** 0.5)
    return not has_divisor_smaller_than(n, i)

def factorial(n):
    """Calculate the factorial of a number."""
    if n == 1:
        return 1
    return n * factorial(n - 1)

def exp_n_x(n, x):
    """Calculate the value of e^x as n goes to infinity."""
    if n == 0:
        return 1
    return (x ** n) / factorial(n) + exp_n_x(n - 1, x)

def play_hanoi(hanoi, n, src, dest, temp):
    """Solve the Towers of Hanoi game in minimum steps using a graphic class."""
    if n < 1:
        return
    elif n == 1:
        hanoi.move(src, dest)
        return
    play_hanoi(hanoi, n - 1, src, temp, dest)
    hanoi.move(src, dest)
    play_hanoi(hanoi, n - 1, temp, dest, src)

def print_sequences(char_list, n):
    """Print all possible words of length n with characters from the given list."""
    if n == 0:
        return
    unique_chars = list(set(char_list))
    _print_sequences_rec(unique_chars, "", len(unique_chars), n)

def _print_sequences_rec(char_list, prefix, n, k):
    """Helper function to print all possible words of length k."""
    if k == 0:
        print(prefix)
        return
    for i in range(n):
        new_prefix = prefix + char_list[i]
        _print_sequences_rec(char_list, new_prefix, n, k - 1)

def print_no_repetition_sequences(char_list, n):
    """Print all possible strings of length n with characters from the list, without repetitions."""
    if n == 0:
        return
    _print_no_repetition_sequences_rec(char_list, "", n)

def _print_no_repetition_sequences_rec(char_list, prefix, n):
    """Helper function to print all possible strings of length n without repetitions."""
    if n == 0:
        print(prefix)
        return
    for i, char in enumerate(char_list):
        new_prefix = prefix + char
        _print_no_repetition_sequences_rec(char_list[:i] + char_list[i+1:], new_prefix, n - 1)

def parentheses(n):
    """Return a list of properly opened and closed combinations of n pairs of parentheses."""
    result_list = []
    if n > 0:
        _parentheses_rec([""] * 2 * n, 0, n, 0, 0, result_list)
    return result_list

def _parentheses_rec(curr, pos, n, open_count, close_count, result_list):
    """Helper function to build a list of properly opened and closed combinations of n pairs of parentheses."""
    if close_count == n:
        result_list.append("".join(curr))
        return
    if open_count > close_count:
        curr[pos] = ')'
        _parentheses_rec(curr, pos + 1, n, open_count, close_count + 1, result_list)
    if open_count < n:
        curr[pos] = '('
        _parentheses_rec(curr, pos + 1, n, open_count + 1, close_count, result_list)

def up_and_right(n, k):
    """Print all possible paths from (0,0) to (n,k) consisting only of 'u' (up) and 'r' (right)."""
    if n < 0 or k < 0:
        return None
    _up_and_right_rec(n, k, "")

def _up_and_right_rec(n, k, prefix):
    """Helper function to print all possible paths from (0,0) to (n,k) consisting only of 'u' and 'r'."""
    if n == 0 and k == 0:
        print(prefix)
        return
    if n > 0:
        _up_and_right_rec(n - 1, k, prefix + 'r')
    if k > 0:
        _up_and_right_rec(n, k - 1, prefix + 'u')

def flood_fill(image, start):
    """Change every '.' connected to the start point (x,y) to '*'."""
    x, y = start
    _flood_fill_rec(image, x, y, '.', '*')

def _flood_fill_rec(image, x, y, empty_char, fill_char):
    """Helper function to recursively fill connected '.' characters with '*'."""
    image_width = len(image)
    image_height = len(image[0])
    if x < 0 or x >= image_width or y < 0 or y >= image_height or image[x][y] != empty_char:
        return
    image[x][y] = fill_char
    _flood_fill_rec(image, x - 1, y, empty_char, fill_char)
    _flood_fill_rec(image, x + 1, y, empty_char, fill_char)
    _flood_fill_rec(image, x, y - 1, empty_char, fill_char)
    _flood_fill_rec(image, x, y + 1, empty_char, fill_char)
