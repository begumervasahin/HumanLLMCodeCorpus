import numpy as np
from sklearn.model_selection import train_test_split

# Load the dataset
data = np.loadtxt(open('D:\\diabetes.csv'), delimiter=',', skiprows=1, encoding='utf8', dtype=float)

def calculate_mean_std(data):
    """
    Calculate the mean and standard deviation for each feature in the dataset.
    """
    num_features = data.shape[1] - 1  # Exclude the label column
    means = []
    stds = []
    
    for i in range(num_features):
        feature_values = data[:, i]
        means.append(np.mean(feature_values))
        stds.append(np.std(feature_values))
        
    return means, stds, num_features

def gauss_probability(x, mean, std):
    """
    Calculate the Gaussian probability density function.
    """
    exponent = np.exp(-((x - mean) ** 2) / (2 * (std ** 2)))
    return (1 / (np.sqrt(2 * np.pi) * std)) * exponent

def class_probability(test_data, means, stds):
    """
    Calculate the probability of the test data belonging to a class.
    """
    probability = 1
    for i in range(len(means)):
        probability *= gauss_probability(test_data[i], means[i], stds[i])
    return probability

def predict(train_data, test_data):
    """
    Predict the probability of the test data belonging to each class.
    """
    means, stds, _ = calculate_mean_std(train_data)
    return class_probability(test_data, means, stds)

# Split the data into training and testing sets
train_data, test_data = train_test_split(data, test_size=0.10)

# Separate the training data into two classes
class1 = [row for row in train_data if row[-1] == 1.0]
class2 = [row for row in train_data if row[-1] == 0.0]

# Perform predictions and calculate accuracy
correct_count = 0

for test_instance in test_data:
    prob_class1 = predict(class1, test_instance)
    prob_class2 = predict(class2, test_instance)
    
    predicted_label = 1.0 if prob_class1 > prob_class2 else 0.0
    
    if predicted_label == test_instance[-1]:
        correct_count += 1

# Calculate accuracy
accuracy = (correct_count / float(len(test_data))) * 100.0
print(f"Accuracy: {accuracy:.2f}%")
