def find_root(vertex):
    """Find the root of the vertex with path compression."""
    while root_ids[vertex] != vertex:
        root_ids[vertex] = root_ids[root_ids[vertex]]  # Path compression
        vertex = root_ids[vertex]
    return vertex

def union(u, v):
    """Union the sets containing vertices u and v."""
    root_u = find_root(u)
    root_v = find_root(v)
    if root_u != root_v:
        root_ids[root_v] = root_ids[root_u]

def kruskal(graph_edges):
    """Kruskal's algorithm to find the Minimum Spanning Tree (MST).
    
    Returns:
        - The total weight of the MST.
    """
    global root_ids
    total_weight = 0

    # Process each edge in increasing order of weight
    for edge in graph_edges:
        u, v, weight = edge
        if find_root(u) != find_root(v):
            total_weight += weight
            union(u, v)
    
    return total_weight

# Input the number of vertices and edges
num_vertices, num_edges = map(int, input().split())
graph_edges = []

# Input each edge with its weight
for _ in range(num_edges):
    u, v, weight = map(int, input().split())
    graph_edges.append([u, v, weight])

# Sort edges by their weight
graph_edges = sorted(graph_edges, key=lambda x: x[2])

# Initialize root ids for each vertex
root_ids = list(range(num_vertices + 1))

# Run Kruskal's algorithm and print the total weight of the MST
print(kruskal(graph_edges))
