class Node:
    def __init__(self, data=None, post=None, pred=None):
        """
        Initialize a new Node.

        Args:
            data: The data to be stored in the node.
            post: The reference to the next node.
            pred: The reference to the previous node.
        """
        self.data = data
        self.post = post
        self.pred = pred

    def remove(self, data):
        """
        Remove the node with the specified data from the list.

        Args:
            data: The data of the node to be removed.

        Returns:
            int: 0 if the node was removed successfully, -1 otherwise.
        """
        if self.pred is None:
            if self.data == data:
                if self.post is None:
                    self.data = None
                else:
                    hold = self.post
                    self.data = self.post.data
                    self.post = self.post.post
                    del hold
                return 0
            elif self.post is None:
                return -1
            else:
                return self.post.remove(data)
        else:
            if self.post is None:
                if self.data == data:
                    hold = self
                    temp = self.pred
                    temp.post = None
                    del hold
                    return 0
                else:
                    return -1
            else:
                if self.data == data:
                    hold = self
                    temp = self.pred
                    temp.post = self.post
                    del hold
                    return 0
                else:
                    return self.post.remove(data)

    def insert(self, data):
        """
        Insert a new node with the given data in sorted order.

        Args:
            data: The data to be inserted.
        """
        if data < self.data:
            new_node = Node(data, self, self.pred)
            if self.pred:
                self.pred.post = new_node
            self.pred = new_node
        elif self.post is None:
            new_node = Node(data, None, self)
            self.post = new_node
        else:
            self.post.insert(data)

    def contain(self, s):
        """
        Helper function to collect the string representation of the list.

        Args:
            s: The current string representation of the list.
        """
        if self.data is not None:
            s += ", " + str(self.data)
        if self.post is None:
            print(s)
        else:
            self.post.contain(s)

    def contains(self):
        """
        Print all elements in the list.
        """
        if self.post is None:
            if self.data is None:
                print("Empty.")
            else:
                print(self.data)
        else:
            self.post.contain(str(self.data))


def test():
    base = Node(0)
    print("contains:")
