class Node:
    def __init__(self, value):
        self.value = value       # The value stored in the node
        self.parent = self       # The parent node (initially points to itself)
        self.rank = 0            # The rank of the node (used in union-find algorithms)

    def __repr__(self):
        return (
            f"Node(value: {self.value}, "
            f"rank: {self.rank}, "
            f"parent: {self.parent.value})"
        )
