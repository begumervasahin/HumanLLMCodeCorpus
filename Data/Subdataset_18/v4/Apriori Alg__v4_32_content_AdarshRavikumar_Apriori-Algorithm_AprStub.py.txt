import re
from Apriori import Apriori

def load_dataset(file_path):
    """
    Load and preprocess the dataset from a text file.

    Parameters:
    file_path (str): Path to the text file containing the dataset.

    Returns:
    list: List of transactions where each transaction is a list of items.
    """
    alpha = [chr(i) for i in range(ord('a'), ord('z') + 1)]
    transactions = []

    with open(file_path, 'r') as file:
        for line in file:
            cleaned_line = re.sub(r"[?\s]", 'a', line.strip())
            items = [item for item in cleaned_line.split(',') if item in alpha]
            transactions.append(items)

    return transactions

    min_confidence = float(input("Minimum confidence: "))
