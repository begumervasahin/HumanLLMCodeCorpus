#!/usr/bin/env python
import os
import sys

def main():
    """
    The main function to set the Django settings module and execute the command line.
    """
    os.environ.setdefault("DJANGO_SETTINGS_MODULE", "fypProject.settings")
    
    try:
        from django.core.management import execute_from_command_line
    except ImportError as exc:
        handle_import_error(exc)
    
    execute_from_command_line(sys.argv)

def handle_import_error(exc):
    """
    Handle ImportError, providing a clear message if Django is not installed.
    
    Parameters:
    exc (ImportError): The ImportError exception instance.
    
    Raises:
    ImportError: If Django is not installed or another import error occurs.
    """
    try:
        import django
    except ImportError:
        raise ImportError(
            "Couldn't import Django. Are you sure it's installed and "
            "available on your PYTHONPATH environment variable? Did you "
            "forget to activate a virtual environment?"
        ) from exc
    raise exc

if __name__ == "__main__":
    main()
