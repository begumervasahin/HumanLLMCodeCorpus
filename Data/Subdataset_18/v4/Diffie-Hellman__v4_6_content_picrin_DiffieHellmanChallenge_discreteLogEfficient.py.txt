import sys
import random

def read_prime_from_file(filename):
    """Read the prime number from the specified file."""
    with open(filename, "r") as f:
        prime = int(f.readline().strip())
    return prime

def extended_euclidean_algorithm(a, b):
    """
    Perform the Extended Euclidean Algorithm to find the GCD and coefficients.

    Parameters:
    a (int): First integer.
    b (int): Second integer.

    Returns:
    tuple: A tuple containing the GCD and the coefficients (gcd, x, y).
    """
    reconstruct = {a: [1, 0], b: [0, 1]}
    current_left, current_right = a, b
    remainder = -1

    while remainder != 0:
        remainder = current_left % current_right
        multiplicity = current_left // current_right
        reconstruct_left = reconstruct[current_left]
        reconstruct_right = reconstruct[current_right]
        i = multiplicity * reconstruct_right[0]
        j = multiplicity * reconstruct_right[1]
        reconstruct[remainder] = [reconstruct_left[0] - i, reconstruct_left[1] - j]
        current_left, current_right = current_right, remainder

    gcd = current_left
    x, y = reconstruct[gcd]
    return gcd, x, y

def multiplicative_inverse(a, p):
    """
    Calculate the multiplicative inverse of a modulo p using the Extended Euclidean Algorithm.

    Parameters:
    a (int): The integer whose multiplicative inverse is to be found.
    p (int): The prime modulus.

    Returns:
    int: The multiplicative inverse of a modulo p.
    """
    gcd, x, _ = extended_euclidean_algorithm(p, a)
    if gcd != 1:
        raise ValueError("No multiplicative inverse exists for the given inputs.")
    return (x + p) % p

def main():
    if len(sys.argv) != 3:
        print("Usage: python script.py <filename_with_prime> <a>")
        sys.exit(1)

    filename_with_prime = sys.argv[1]
    a = int(sys.argv[2])

    # Read the prime number from the file
    p = read_prime_from_file(filename_with_prime)

    # Randomly select an integer g between 1 and p-1
    g = random.randint(1, p)

    # Calculate the multiplicative inverse of g modulo p
    inverse_g = multiplicative_inverse(g, p)

    # Calculate the result of a * multiplicative inverse of g modulo p
    result = (a * inverse_g) % p

    print(f"The result of {a} * multiplicative inverse of {g} modulo {p} is {result}")

if __name__ == '__main__':
    main()
