import numpy as np
import pandas as pd
from sklearn.feature_extraction.text import CountVectorizer, TfidfTransformer
from sklearn.metrics.pairwise import cosine_similarity

# Load article data from a CSV file
df_articles = pd.read_csv('article_data.csv')

# Extract the target text (the first article)
target_text = df_articles.content.iloc[0]

# Extract all article texts
texts = df_articles.content.tolist()

# Initialize the CountVectorizer with English stop words
vectorizer = CountVectorizer(stop_words='english')

# Fit the vectorizer to the target text
vectorizer.fit_transform([target_text])

# Transform the texts into a frequency term matrix
freq_term_matrix = vectorizer.transform(texts)

# Initialize the TfidfTransformer
tfidf_transformer = TfidfTransformer(norm="l2")

# Fit the transformer to the frequency term matrix
tfidf_transformer.fit(freq_term_matrix)

# Transform the frequency term matrix to a TF-IDF matrix
tf_idf_matrix = tfidf_transformer.transform(freq_term_matrix)

# Compute the cosine similarity between the target text and all texts
cosine_similarities = cosine_similarity(tf_idf_matrix[0:1], tf_idf_matrix)

# Create a DataFrame for the cosine similarity scores
cos_sim_df = pd.DataFrame(cosine_similarities[0], columns=['Cos_Sim'])

# Concatenate the original DataFrame with the cosine similarity scores
df_final = pd.concat([df_articles, cos_sim_df], axis=1)

# Sort the DataFrame by cosine similarity scores in descending order
df_final_sorted = df_final.sort_values(by='Cos_Sim', ascending=False)

# Print the top 6 results
print(df_final_sorted.head(6))
