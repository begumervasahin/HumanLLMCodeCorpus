class Node:
    def __init__(self, val):
        self.value = val
        self.next = None
        self.prev = None

class DoublyList:
    def __init__(self, val):
        self.head = Node(val)
        self.tail = self.head

    def add_node_last(self, val):
        """Add a node with value `val` at the end of the list."""
        new_node = Node(val)
        self.tail.next = new_node
        new_node.prev = self.tail
        self.tail = new_node

    def insert_node(self, val, new_val):
        """Insert a node with value `new_val` after the node with value `val`."""
        current = self.head
        while current and current.value != val:
            current = current.next

        if not current:
            print(f"Value {val} not found in the list.")
            return

        new_node = Node(new_val)
        new_node.next = current.next
        new_node.prev = current

        if current.next:
            current.next.prev = new_node

        current.next = new_node

        if current == self.tail:
            self.tail = new_node

    def remove_node(self, val):
        """Remove the node with value `val` from the list."""
        current = self.head
        while current and current.value != val:
            current = current.next

        if not current:
            print(f"Value {val} not found in the list.")
            return

        if current.prev:
            current.prev.next = current.next

        if current.next:
            current.next.prev = current.prev

        if current == self.head:
            self.head = current.next

        if current == self.tail:
            self.tail = current.prev

    def show_reverse(self):
        """Print the list in reverse order."""
        current = self.tail
        while current:
            print(current.value)
            current = current.prev

    def show(self):
        """Print the list in order."""
        current = self.head
        while current:
            print(current.value)
            current = current.next

# Example usage
if __name__ == "__main__":
    d_list = DoublyList(10)
    d_list.add_node_last(20)
    d_list.add_node_last(30)
    d_list.add_node_last(40)
    d_list.remove_node(40)
    print("List in order:")
    d_list.show()
    print("List in reverse order:")
    d_list.show_reverse()
