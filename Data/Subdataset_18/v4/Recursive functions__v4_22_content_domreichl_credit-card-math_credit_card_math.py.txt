def calc_balance(balance, min_payment_rate, monthly_interest_rate, months, payments):
    """
    Calculates the credit card balance after a given number of months for
    a person who only pays the minimum monthly payment.
    It also builds a list of monthly payments.

    Args:
        balance (float): Initial balance.
        min_payment_rate (float): Minimum payment rate.
        monthly_interest_rate (float): Monthly interest rate.
        months (int): Number of months.
        payments (list): List to store monthly payments.

    Returns:
        tuple: Remaining balance and list of payments.
    """
    months -= 1
    if months < 0:
        return round(balance, 2), payments
    else:
        min_payment = min_payment_rate * balance
        payments.append(min_payment)
        unpaid_balance = balance - min_payment
        balance = unpaid_balance + monthly_interest_rate * unpaid_balance
        return calc_balance(balance, min_payment_rate, monthly_interest_rate, months, payments)

def pay_off_debt(balance, monthly_interest_rate, months, lower, upper):
    """
    Calculates the minimum fixed monthly payment needed in order to pay off
    a credit card balance within a given number of months using bisection search.

    Args:
        balance (float): Initial balance.
        monthly_interest_rate (float): Monthly interest rate.
        months (int): Number of months.
        lower (float): Lower bound of the monthly payment.
        upper (float): Upper bound of the monthly payment.

    Returns:
        float: Minimum fixed monthly payment.
    """
    guess = (lower + upper) / 2
    months -= 1
    if months < 0:
        if round(balance) == 0:
            return round(guess, 2)
        elif balance < 0:
            return pay_off_debt(balance, monthly_interest_rate, months + 1, lower, guess)
        else:
            return pay_off_debt(balance, monthly_interest_rate, months + 1, guess, upper)
    else:
        unpaid_balance = balance - guess
        balance = unpaid_balance + monthly_interest_rate * unpaid_balance
        return pay_off_debt(balance, monthly_interest_rate, months, lower, upper)

    balance = 100000
    months = 12
    annual_interest_rate = 0.22
    min_payment_rate = 0.05
    monthly_interest_rate = annual_interest_rate / 12
    payments = []

    new_balance, payments = calc_balance(balance, min_payment_rate, monthly_interest_rate, months, payments)
    average_payment = round(sum(payments) / len(payments), 2)
    loss1 = round(abs(balance - new_balance - sum(payments)), 2)

    min_guess = balance / 12
    max_guess = (balance * (1 + monthly_interest_rate) ** months) / months
    payment = pay_off_debt(balance, monthly_interest_rate, months, min_guess, max_guess)
    loss2 = round(abs(balance - 0 - payment * months), 2)

    print(f"Current Balance: {balance} €")
    print(f"Duration: {months} months")
    print(f"With a monthly payment rate of {min_payment_rate:.2f}:")
    print(f"   - You will pay {average_payment} € per month on average.")
    print(f"   - Your loss will be {loss1} € for that time period.")
    print(f"   - Your remaining balance will be {new_balance} €.")
    print(f"If you want to pay off your balance within {months} months:")
    print(f"   - You will have to pay {payment} € per month.")
    print(f"   - Your loss will be {loss2} € in total.")
    print(f"   - Your remaining balance will be 0 €.")
