import random

class BSTVectorBase:
    """Base class for the BST-based vector structure."""
    
    def set(self, index, value):
        """Set the value at the specified index."""
        pass
    
    def get(self, index):
        """Get the value at the specified index."""
        pass
    
    def sample(self):
        """Sample an index from the vector based on its values."""
        pass
    
    @property
    def norm2(self):
        """Return the L2 norm squared of the vector."""
        return self._norm2
    
    def __repr__(self):
        return f"<BSTVector of dimension {self._dim}: {self}>"
    
class BSTVectorLeaf(BSTVectorBase):
    """Leaf node class for the BST-based vector."""
    
    def __init__(self):
        self.value = 0.0
        self._norm2 = 0.0
        self._dim = 1
    
    def set(self, index, value):
        """Set the value at the leaf node."""
        if index != 0:
            raise IndexError("Index out of bounds for leaf node.")
        self.value = float(value)
        self._update_norm2()
    
    def get(self, index):
        """Get the value at the leaf node."""
        if index != 0:
            raise IndexError("Index out of bounds for leaf node.")
        return self.value
    
    def sample(self):
        """Sample the index (always 0 for leaf node)."""
        return 0
    
    def _update_norm2(self):
        """Update the L2 norm squared for the leaf node."""
        self._norm2 = self.value ** 2
    
    def __str__(self):
        return str(self.value)

class BSTVectorNode(BSTVectorBase):
    """Internal node class for the BST-based vector."""
    
    def __init__(self, dim):
        self._dim = dim
        self._norm2 = 0.0
        self.left = None
        self.right = None
    
    def set(self, index, value):
        """Set the value at the specified index within the node."""
        if index < self.cutoff:
            child_side = 'left'
            child_index = index
        else:
            child_side = 'right'
            child_index = index - self.cutoff
        
        if getattr(self, child_side) is None:
            setattr(self, child_side, BSTVector(self.cutoff if child_side == 'left' else self._dim - self.cutoff))
        
        child = getattr(self, child_side)
        child.set(child_index, value)
        
        # Remove child if its norm2 becomes zero
        if child.norm2 == 0.0:
            setattr(self, child_side, None)
        
        self._update_norm2()
    
    def get(self, index):
        """Get the value at the specified index within the node."""
        if index >= self._dim:
            raise IndexError("Index out of bounds for BSTVector.")
        
        if index < self.cutoff:
            child_side = 'left'
            child_index = index
        else:
            child_side = 'right'
            child_index = index - self.cutoff
        
        child = getattr(self, child_side)
        return 0.0 if child is None else child.get(child_index)
    
    def sample(self, seed=None):
        """Sample an index from the vector based on its values."""
        if self.norm2 == 0.0:
            raise ValueError("No nonzero entries to sample.")
        
        left_norm2 = self.left.norm2 if self.left is not None else 0.0
        
        if seed is not None:
            random.seed(seed)
        
        if random.uniform(0, self.norm2) < left_norm2:
            return self.left.sample()
        return self.cutoff + self.right.sample()
    
    def _update_norm2(self):
        """Update the L2 norm squared for the internal node."""
        self._norm2 = sum(child.norm2 for child in [self.left, self.right] if child is not None)
    
    @property
    def cutoff(self):
        """Return the index cutoff between left and right children."""
        return self._dim // 2
    
    def __str__(self):
        left_str = str(self.left) if self.left is not None else "-" * self.cutoff
        right_str = str(self.right) if self.right is not None else "-" * (self._dim - self.cutoff)
        return f"({left_str} {right_str})"

def BSTVector(dim):
    """Factory function to create a BSTVector of the specified dimension."""
    if dim == 1:
        return BSTVectorLeaf()
    return BSTVectorNode(dim)
