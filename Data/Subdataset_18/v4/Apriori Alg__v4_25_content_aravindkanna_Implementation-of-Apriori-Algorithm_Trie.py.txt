from queue import Queue

class TrieNode:
    """
    Class representing a node in the Trie.
    """
    def __init__(self, count=1, name="root"):
        self.child = {}
        self.count = count
        self.name = name

    def insert_node(self, itemset, count):
        """
        Insert a node into the Trie.

        :param itemset: List of items to be inserted
        :param count: Count associated with the itemset
        """
        current_node = self.child
        for item in itemset:
            if item in current_node:
                current_node = current_node[item].child
            else:
                new_node = TrieNode(count, item)
                current_node[item] = new_node
                current_node = new_node.child

    def has_node(self, itemset):
        """
        Check if a node exists in the Trie.

        :param itemset: List of items to be checked
        :return: True if the itemset exists, False otherwise
        """
        current_node = self.child
        for item in itemset:
            if item in current_node:
                current_node = current_node[item].child
            else:
                return False
        return True

    def insert_all(self, freq_itemsets, counts):
        """
        Insert multiple itemsets into the Trie.

        :param freq_itemsets: List of frequent itemsets
        :param counts: List of counts corresponding to the itemsets
        """
        for itemset, count in zip(freq_itemsets, counts):
            self.insert_node(itemset, count)

    def print_all(self, prefix=""):
        """
        Print all nodes in the Trie.

        :param prefix: Prefix string to be appended before each item
        """
        for item, node in self.child.items():
            current_str = prefix + item
            print(current_str)
            node.print_all(current_str + ",")

    def get_itemsets(self, prefix=None):
        """
        Get all itemsets in the Trie.

        :param prefix: Prefix list to be appended before each itemset
        :return: Generator yielding itemsets
        """
        if prefix is None:
            prefix = []
        for item, node in self.child.items():
            current_list = prefix + [item]
            yield current_list
            yield from node.get_itemsets(current_list)

    def get_count(self, itemset):
        """
        Get the count of a specific itemset in the Trie.

        :param itemset: List of items whose count is to be retrieved
        :return: Count of the itemset
        """
        current_node = self.child
        for item in itemset:
            current_node = current_node[item]
        return current_node.count
