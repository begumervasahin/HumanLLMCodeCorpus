import numpy as np
import matplotlib.pyplot as plt
import os
import importlib

# Configuration options
do_paleo = True
cal_individual_misfits = True
show_plots = False
save_to_pdf = True
print_info = True

# Import and reload modules to ensure the latest versions are used
import config as cf
import tools as tl
import get_score as gs
import get_paleo as ps
import compare_score as cs

modules = [cf, tl, gs, ps, cs]
for module in modules:
    importlib.reload(module)

# Define ensemble range
ens_min = cf.ens_min
ens_max = cf.ens_max
ensnums = np.arange(ens_min, ens_max + 1)

# Plot and print options
shownprint = [show_plots, save_to_pdf, print_info]

# Create output directory for plots if saving to PDF
if save_to_pdf:
    plots_path = os.path.join(cf.output_data_path, "plots")
    os.makedirs(plots_path, exist_ok=True)

# Load observations
obsfile = cf.obsfile
velobsfile = cf.velobsfile
observations = tl.load_observations(obsfile, velobsfile)

# Calculate individual misfits if required
if cal_individual_misfits:
    for ensnum in ensnums:
        print(f"Ensemble number: {ensnum}")
        pds = gs.pd_score(ensnum, observations, shownprint)
        
        if do_paleo:
            pls = ps.paleo_score(ensnum, observations, shownprint)
        
        if show_plots:
            plt.show()

# Compare scores across ensembles
ens = cs.compare_score(ensnums, do_paleo, shownprint)
score = ens.score

# Output the final score
if print_info:
    print(f"Final score: {score}")
