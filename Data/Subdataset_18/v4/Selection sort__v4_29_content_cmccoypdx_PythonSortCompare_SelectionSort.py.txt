"""
Module for holding the selection sort algorithm
"""

def selection_sort(input_list):
    """
    Returns a sorted list with contents of the input list.
    Does not modify the input list.

    Args:
        input_list (list): The list to be sorted.

    Returns:
        list: A new list containing the sorted elements.
    """
    # Create a copy of the input list to avoid modifying the original list
    unsorted_list = list(input_list)
    sorted_list = []

    while unsorted_list:
        # Assume the first element is the smallest
        smallest = unsorted_list[0]

        # Find the smallest element in the unsorted list
        for element in unsorted_list:
            if element < smallest:
                smallest = element

        # Append the smallest element to the sorted list and remove it from the unsorted list
        sorted_list.append(smallest)
        unsorted_list.remove(smallest)

    return sorted_list

# Example usage:
if __name__ == "__main__":
    sample_list = [34, 23, 12, 45, 9, 1, 24]
    print("Original list:", sample_list)
    sorted_sample_list = selection_sort(sample_list)
    print("Sorted list:", sorted_sample_list)
