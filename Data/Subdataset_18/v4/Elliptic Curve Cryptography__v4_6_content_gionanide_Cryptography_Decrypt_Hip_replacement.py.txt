import timeit

def decrypt_caesar_cipher(text, shift):
    """
    Decrypts a Caesar cipher encrypted text by shifting characters forward.

    Args:
        text (str): The encrypted text.
        shift (int): The number of positions to shift characters forward.

    Returns:
        str: The decrypted text.
    """
    decrypted_chars = []

    for char in text:
        if 'A' <= char <= 'Z':
            # Shift the character forward in the alphabet
            shifted_char_code = (ord(char) - ord('A') + shift) % 26 + ord('A')
            decrypted_chars.append(chr(shifted_char_code))
        else:
            # Non-alphabetic characters remain unchanged
            decrypted_chars.append(char)
    
    return ''.join(decrypted_chars)

def main():
    encrypted_text = 'YMJJDJXHFSRNXQJFIYMJXRNQJHFSQNJGZYYMJXMTJXYJQQYMJYWZYM'
    print('Encrypted message:', encrypted_text)

    # Try all possible shifts from 1 to 25
    for shift in range(1, 26):
        decrypted_message = decrypt_caesar_cipher(encrypted_text, shift)
        print(f'Shift {shift}: {decrypted_message}')

if __name__ == "__main__":
    main()
