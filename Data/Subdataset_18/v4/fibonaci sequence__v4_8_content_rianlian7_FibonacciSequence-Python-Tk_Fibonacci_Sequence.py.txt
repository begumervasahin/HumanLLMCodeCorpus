import tkinter as tk

def calculate_fibonacci_term():
    """
    Calculate the nth Fibonacci term based on user input and display the result.
    """
    try:
        # Get the user input from the entry widget
        term_number = int(num_entry.get())
        
        # Initialize the Fibonacci sequence with the first two numbers
        fibonacci_sequence = [0, 1]
        
        # Calculate the Fibonacci sequence up to the desired term
        for _ in range(2, term_number + 1):
            next_term = fibonacci_sequence[-1] + fibonacci_sequence[-2]
            fibonacci_sequence.append(next_term)
        
        # Determine the result based on the input term number
        if term_number == 0:
            result = 0
        elif term_number == 1:
            result = 1
        else:
            result = fibonacci_sequence[-1]
        
        # Display the result in the result label
        result_label.config(text=f"The {term_number}th term is: {result}")
    
    except ValueError:
        result_label.config(text="Please enter a valid integer.")

def setup_gui():
    """
    Set up the main Tkinter window and its widgets.
    """
    # Create the main window
    main_window = tk.Tk()
    main_window.title("Fibonacci Sequence Calculator")
    
    # Create an entry widget for user input
    tk.Label(main_window, text="Enter term number:").grid(row=0, column=0, padx=5, pady=5)
    global num_entry
    num_entry = tk.Entry(main_window)
    num_entry.grid(row=0, column=1, padx=5, pady=5)
    num_entry.focus()
    
    # Create a button to trigger the Fibonacci calculation
    calculate_button = tk.Button(main_window, text="Find Fibonacci Term", command=calculate_fibonacci_term)
    calculate_button.grid(row=1, column=0, columnspan=2, padx=5, pady=5)
    
    # Create a label to display the result
    global result_label
    result_label = tk.Label(main_window, text="Result")
    result_label.grid(row=2, column=0, columnspan=2, sticky="w", padx=5, pady=5)
    
    # Start the Tkinter event loop
    main_window.mainloop()

if __name__ == "__main__":
    setup_gui()
