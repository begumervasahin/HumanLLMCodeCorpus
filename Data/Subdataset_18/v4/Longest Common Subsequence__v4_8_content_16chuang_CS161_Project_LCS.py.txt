import sys
import numpy as np

# Initialize a 2D array to store the length of LCS
arr = np.zeros((2048, 2048), dtype=int)

def LCS(A, B):
    """
    Compute the length of the longest common subsequence (LCS) of two strings A and B.
    
    Parameters:
    A (str): First string.
    B (str): Second string.
    
    Returns:
    int: Length of the LCS.
    """
    m = len(A)
    n = len(B)
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if A[i - 1] == B[j - 1]:
                arr[i][j] = arr[i - 1][j - 1] + 1
            else:
                arr[i][j] = max(arr[i - 1][j], arr[i][j - 1])
    return arr[m][n]

def main():
    """
    Main function to read input and compute LCS for each pair of strings.
    """
    if len(sys.argv) != 1:
        sys.exit('Usage: python LCS.py < input')
    
    for line in sys.stdin:
        A, B = line.strip().split()
        print(LCS(A, B))

if __name__ == '__main__':
    main()
