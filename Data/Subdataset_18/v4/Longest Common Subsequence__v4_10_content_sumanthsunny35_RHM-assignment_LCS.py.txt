import re

def verify(sequence):
    """
    Verify if the given DNA sequence only contains the characters A, T, C, G, and spaces.
    """
    return re.match("^[ATCG ]*$", sequence)

def lcs(X, Y, m, n):
    """
    Find the Longest Common Subsequence (LCS) of two DNA sequences X and Y.
    """
    # Create a 2D array to store lengths of longest common subsequence.
    L = [[0 for _ in range(n + 1)] for _ in range(m + 1)]
    
    # Build the L[m+1][n+1] table in bottom-up fashion.
    for i in range(m + 1):
        for j in range(n + 1):
            if i == 0 or j == 0:
                L[i][j] = 0
            elif X[i - 1] == Y[j - 1]:
                L[i][j] = L[i - 1][j - 1] + 1
            else:
                L[i][j] = max(L[i - 1][j], L[i][j - 1])
    
    # The length of LCS is L[m][n]
    index = L[m][n]
    
    # Create a list to store the LCS sequence
    lcs_sequence = [""] * (index + 1)
    lcs_sequence[index] = "\0"  # Set the end of the string
    
    # Start from the right-most-bottom-most corner and find the LCS
    i, j = m, n
    while i > 0 and j > 0:
        if X[i - 1] == Y[j - 1]:
            lcs_sequence[index - 1] = X[i - 1]
            i -= 1
            j -= 1
            index -= 1
        elif L[i - 1][j] > L[i][j - 1]:
            i -= 1
        else:
            j -= 1
    
    # Join the list to form the LCS string
    lcs_str = "".join(lcs_sequence).strip("\0")
    print(f"\n\nLCS of given two DNA sequences is {lcs_str} \nand length of it is: {len(lcs_str)}")

def main():
    s1 = input("Enter the First Sequence\n").upper()
    s2 = input("Enter the Querying Sequence\n").upper()
    
    if verify(s1) and verify(s2):
        if len(s1) >= len(s2):
            m, n = len(s1), len(s2)
            lcs(s1, s2, m, n)
        else:
            print("Querying Sequence should be smaller than First Sequence")
    else:
        print("The Sequences should contain 'A', 'T', 'C', 'G', or ' ' only")

if __name__ == "__main__":
    main()
