import os
import collections
import pandas as pd
import csv

# Define file paths
DATA_DIR = "data"
MBTI_RAW_CSV_PATH = os.path.join(DATA_DIR, "mbti_1.csv")
MBTI_CLEAN_CSV_PATH = os.path.join(DATA_DIR, "mbti_clean.csv")
MBTI_UNCLEAN_CSV_PATH = os.path.join(DATA_DIR, "mbti_unclean.csv")

# Frequency dictionary for MBTI types
MBTI_TO_FREQUENCY_DICT = {
    "ISTJ": 0.11,
    "ISFJ": 0.09,
    "INFJ": 0.04,
    "INTJ": 0.05,
    "ISTP": 0.05,
    "ISFP": 0.05,
    "INFP": 0.06,
    "INTP": 0.06,
    "ESTP": 0.04,
    "ESFP": 0.04,
    "ENFP": 0.08,
    "ENTP": 0.06,
    "ESTJ": 0.08,
    "ESFJ": 0.09,
    "ENFJ": 0.05,
    "ENTJ": 0.05,
}

# Read the raw data
df = pd.read_csv(MBTI_RAW_CSV_PATH)

# Count occurrences of each MBTI type
counts = collections.defaultdict(int)
for mbti in df["type"]:
    counts[mbti] += 1

# Find the MBTI type with the smallest relative frequency
limiting_type = None
min_size = float("inf")
for mbti, count in counts.items():
    size = count / MBTI_TO_FREQUENCY_DICT[mbti]
    if size < min_size:
        min_size = size
        limiting_type = mbti

# Group data by MBTI type
grouped_data = collections.defaultdict(list)
for index, row in df.iterrows():
    grouped_data[row["type"]].append(row)

# Prepare data for unclean CSV
unclean_list = []
for mbti in MBTI_TO_FREQUENCY_DICT.keys():
    data_list = grouped_data[mbti]
    start_index = int(round(min_size * MBTI_TO_FREQUENCY_DICT[mbti]))
    unclean_list.append(data_list[start_index:])

# Write unclean data to CSV
with open(MBTI_UNCLEAN_CSV_PATH, "w", newline='') as f:
    writer = csv.writer(f)
    writer.writerow(["type", "posts"])
    for mbti_list in unclean_list:
        for row in mbti_list:
            writer.writerow([row["type"], row["posts"]])

# Write clean data to CSV
with open(MBTI_CLEAN_CSV_PATH, "w", newline='') as f:
    writer = csv.writer(f)
    writer.writerow(["type", "posts"])
    for mbti, posts in grouped_data.items():
        if mbti != limiting_type:
            for post in posts:
                writer.writerow([post["type"], post["posts"]])
