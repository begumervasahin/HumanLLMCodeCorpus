import pandas as pd

# Load the prediction data from the CSV file
data_load = pd.read_csv("y_prediction_bts.csv").values.flatten()
pred_data = list(data_load)

# Initialize lists to store the start and end indices of valid sequences
cut_start_data = []
cut_end_data = []

# Initialize the index and determine the length of the data
idx = 0
data_length = len(pred_data)

# Loop through the prediction data
while idx < data_length - 1:
    count = 0
    most_dyn_count = 0
    
    # Check if the current value is 0.5 or 1
    if pred_data[idx] in [0.5, 1]:
        start_idx = idx
        
        # Continue looping while the value is 0.5 or 1
        while idx < data_length and pred_data[idx] in [0.5, 1]:
            count += 1
            if pred_data[idx] == 1:
                most_dyn_count += 1
            idx += 1
            print("TEST")  # Debug print statement
            
        end_idx = idx
        
        # Check if the sequence is valid
        if count > 3 and most_dyn_count > 0:
            print("SUCCESS")  # Debug print statement
            cut_start_data.append(start_idx)
            cut_end_data.append(end_idx)
    
    idx += 1

# Create DataFrames for the start and end times
cut_start_df = pd.DataFrame(cut_start_data, columns=["Start Time"])
cut_end_df = pd.DataFrame(cut_end_data, columns=["End Time"])

# Concatenate the start and end times into a single DataFrame
time_data = pd.concat([cut_start_df, cut_end_df], axis=1)

# Save the resulting DataFrame to a CSV file
time_data.to_csv("timedata_bts.csv", index=False)
