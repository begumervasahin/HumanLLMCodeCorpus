import os

class DatasetClass:
    def __init__(self, required_no):
        dataset_name = "ORL"
        dir_path = os.path.join("images", dataset_name)

        self.image_path_for_training = []
        self.labels_for_training = []
        self.no_of_images_for_training = []
        self.image_path_for_testing = []
        self.labels_for_testing = []
        self.no_of_images_for_testing = []
        self.images_target = []

        person_index = 0

        for person_name in os.listdir(dir_path):
            person_dir_path = os.path.join(dir_path, person_name)

            if os.path.isdir(person_dir_path) and len(os.listdir(person_dir_path)) >= required_no:
                images = os.listdir(person_dir_path)

                for i, img_name in enumerate(images):
                    img_path = os.path.join(person_dir_path, img_name)

                    if i < required_no:
                        self.image_path_for_training.append(img_path)
                        self.labels_for_training.append(person_index)

                        if len(self.no_of_images_for_training) > person_index:
                            self.no_of_images_for_training[person_index] += 1
                        else:
                            self.no_of_images_for_training.append(1)

                        if i == 0:
                            self.images_target.append(person_name)
                    else:
                        self.image_path_for_testing.append(img_path)
                        self.labels_for_testing.append(person_index)

                        if len(self.no_of_images_for_testing) > person_index:
                            self.no_of_images_for_testing[person_index] += 1
                        else:
                            self.no_of_images_for_testing.append(1)

                person_index += 1
