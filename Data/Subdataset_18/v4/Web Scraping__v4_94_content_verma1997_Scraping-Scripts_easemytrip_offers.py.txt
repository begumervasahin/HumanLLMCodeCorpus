URL = "https://www.easemytrip.com/deals.html"
FILENAME = "easemytrip_offers.csv"

def get_random_header():
    """Return a random User-Agent header."""
    return {'User-Agent': choice(HEADERS)}

def fetch_html(url):
    """Fetch HTML content from the given URL using a random header."""
    response = requests.get(url, headers=get_random_header())
    response.raise_for_status()
    return response.text

def parse_offers(html):
    """Parse offers from the HTML content and return them as a list of dictionaries."""
    soup = BeautifulSoup(html, 'html.parser')
    containers = soup.findAll("div", {"class": "deal"})
    offers = []

    for container in containers:
        image = container.find("img")['src']
        name = container.find("div", {"class": "dealName"}).text
        description = container.find("p").text
        link = container.find("a")['href']

        offers.append({
            "name": name,
            "description": description,
            "image": image,
            "link": link
        })
    return offers

def save_offers_to_csv(offers, filename):
    """Save the offers to a CSV file."""
    with open(filename, "w") as file:
        file.write("Offer Name,Link,Description,Image Link\n")
        for offer in offers:
            file.write(f'{offer["name"]},{offer["link"]},{offer["description"]},{offer["image"]}\n')

def main():
    """Main function to fetch, parse, and save offers."""
    html = fetch_html(URL)
    offers = parse_offers(html)
    save_offers_to_csv(offers, FILENAME)

if __name__ == "__main__":
    main()
