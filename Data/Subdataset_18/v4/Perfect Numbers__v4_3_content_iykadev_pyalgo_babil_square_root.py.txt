def babylonian_square_root(a, eps):
    """
    Compute the square root of a number using the Babylonian method (also known as Heron's method).
    
    Parameters:
    a (float): The number to find the square root of.
    eps (float): The desired precision for the square root calculation.
    
    Returns:
    None: The function prints the square root of the number.
    """
    # Initial guess
    x0 = (1 + a) / 2
    x1 = (x0 + a / x0) * 0.5
    
    # Calculate the relative error
    y = abs((x1 - x0) / x0)
    
    # Iterate until the relative error is within the desired precision
    while y > eps:
        x0 = x1
        x1 = (x0 + a / x0) * 0.5
        y = abs((x1 - x0) / x0)
    
    result = f"The square root of the number {a} is approximately {x1}"
    print(result)

def main():
    """
    Main function to test the Babylonian square root calculation.
    
    Returns:
    None
    """
    babylonian_square_root(5, 0.0001)

# Run the main function
main()
