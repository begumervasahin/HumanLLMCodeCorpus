def tag_visible(element):
    """
    Check if a BeautifulSoup element is visible.
    
    Args:
        element: The BeautifulSoup element to check.
        
    Returns:
        bool: True if the element is visible, False otherwise.
    """
    if element.parent.name in ['style', 'script', 'head', 'title', 'meta', '[document]']:
        return False
    if isinstance(element, Comment):
        return False
    return True

def text_from_html(body):
    """
    Extract visible text from an HTML document.
    
    Args:
        body (str): The HTML document.
        
    Returns:
        str: The visible text in the HTML document.
    """
    soup = BeautifulSoup(body, 'html.parser')
    texts = soup.findAll(text=True)
    visible_texts = filter(tag_visible, texts)
    return u" ".join(t.strip() for t in visible_texts)

def fetch_and_store_url_content(url):
    """
    Fetch and store the content of a URL.
    
    Args:
        url (str): The URL to fetch.
        
    Returns:
        None
    """
    try:
        response = urllib.request.urlopen(url)
        html = response.read()
        content = text_from_html(html)
        
        record = {
            "link": url,
            "content": content
        }
        record_id = mydb.mytable.insert_one(record).inserted_id
        print(f"Record inserted with ID: {record_id}")
        print(f"Collections in the database: {mydb.list_collection_names()}")
        print(content)
    except HTTPError as e:
        print(f"Failed to fetch URL: {url} with error: {e}")

def main():
    """
    Main function to fetch and store content from all valid links on the NYTimes homepage.
    
    Returns:
        None
    """
    url = 'http://www.nytimes.com'
    html = urllib.request.urlopen(url).read()
    soup = BeautifulSoup(html, "html.parser")
    links = soup.select('a[href^="http"]')
    
    for link in links:
        href = link.get('href')
        if href:
            print(href)
            fetch_and_store_url_content(href)

if __name__ == '__main__':
    main()
