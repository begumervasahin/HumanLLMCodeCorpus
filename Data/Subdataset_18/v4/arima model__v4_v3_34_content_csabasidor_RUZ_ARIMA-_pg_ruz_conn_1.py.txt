def connect_to_database(params):
    """
    Establishes a connection to the PostgreSQL database.

    Parameters:
    params (dict): A dictionary containing database connection parameters.

    Returns:
    connection: A connection object if successful, None otherwise.
    """
    try:
        connection = pg.connect(**params)
        print("Connected to the database!")
        return connection
    except Exception as error:
        print(f"Error connecting to the database: {error}")
        return None

def execute_query(connection, query):
    """
    Executes a SQL query and retrieves the results into a Pandas DataFrame.

    Parameters:
    connection: A connection object to the database.
    query (str): The SQL query to execute.

    Returns:
    pd.DataFrame: A DataFrame containing the query results, None if an error occurs.
    """
    try:
        data_frame = pd.read_sql(query, connection)
        return data_frame
    except Exception as error:
        print(f"Error executing query: {error}")
        return None

def close_connection(connection):
    """
    Closes the database connection.

    Parameters:
    connection: A connection object to the database.
    """
    try:
        if connection is not None:
            connection.close()
            print("Connection closed.")
    except Exception as error:
        print(f"Error closing the connection: {error}")
