from cp_otp import strxor
from cp_dictionary import list_of_words
from itertools import combinations

# Given ciphertexts
c1 = b'\x0b\x0e\x1e\x0b\x17'
c2 = b'\x15\x0a\x1b\x1d\x01'

# Generate a list of words with the same length as the ciphertexts
word_length = len(c1)
words = list_of_words(of_length=word_length)

# Create all possible pairs of words from the list
pairs = combinations(words, r=2)

# Example usage (print pairs)
for pair in pairs:
    print(pair)
