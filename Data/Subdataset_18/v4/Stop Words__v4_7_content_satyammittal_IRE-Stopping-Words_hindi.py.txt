import os
import json
import operator

def list_files(dir):
    """
    List all files in the given directory and its subdirectories.
    """
    file_list = []
    for subdir, _, files in os.walk(dir):
        for file in files:
            file_list.append(os.path.join(subdir, file))
    return file_list

# List all files in the 'extracted' directory
file_paths = list_files('extracted')
word_count = {}
document_count = {}
total_documents = 0

# Iterate through each file and count word occurrences
for file in file_paths:
    with open(file) as f:
        for line in f:
            word_in_document = {}
            json_line = json.loads(line)
            text = json_line.get('text')
            if text:
                total_documents += 1
                for word in text.split():
                    # Update word count
                    word_count[word] = word_count.get(word, 0) + 1
                    # Track word occurrence in the current document
                    word_in_document[word] = word_in_document.get(word, 0) + 1
            
            # Update document count for each word
            for word in word_in_document:
                document_count[word] = document_count.get(word, 0) + 1

# Calculate the result based on the provided formula
result = {}
for word in document_count:
    result[word] = word_count[word] * document_count[word] ** 2

# Sort the results in descending order
sorted_result = sorted(result.items(), key=operator.itemgetter(1), reverse=True)

# Print the top 50 words and their scores
for i, (word, score) in enumerate(sorted_result[:50]):
    print(f"{word}: {score}")

# Print the top 50 words only
for word, _ in sorted_result[:50]:
    print(word)
