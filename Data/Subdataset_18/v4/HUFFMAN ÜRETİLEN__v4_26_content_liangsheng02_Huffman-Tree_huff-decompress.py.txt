import os
import pickle
import argparse
import time

class Node:
    def __init__(self, char=None):
        self.char = char
        self.left = None
        self.right = None

class HuffmanDecoder:
    def __init__(self, file):
        self.file = os.path.splitext(file)[0]
        self.text_code, self.text_string = self._read_file()
        self.root = self._rebuild_tree()
        self.final_text = self._decompress()
        print('Decompression complete:', len(self.final_text), 'characters')

    def _read_file(self):
        """
        Load the symbol model (pkl file) and the binary data (bin file).
        Convert the binary data into a long string.
        
        Returns:
            dict: A dictionary representing the Huffman codes.
            str: A string representing the compressed binary data.
        """
        with open(self.file + '-symbol-model.pkl', 'rb') as pkl_file:
            text_code = pickle.load(pkl_file)
        
        padding_length = text_code.pop('padding_length')
        with open(self.file + '.bin', 'rb') as bin_file:
            text_bin = bin_file.read()
        
        text_string = ''.join(bin(byte)[2:].zfill(8) for byte in text_bin)[:-padding_length]
        return text_code, text_string

    def _rebuild_tree(self):
        """
        Rebuild the Huffman tree from the dictionary of Huffman codes.
        
        Returns:
            Node: The root node of the reconstructed Huffman tree.
        """
        root = Node()
        for char, code in self.text_code.items():
            node = root
            for bit in code:
                if bit == '0':
                    if not node.left:
                        node.left = Node()
                    node = node.left
                else:
                    if not node.right:
                        node.right = Node()
                    node = node.right
            node.char = char
        return root

    def _decompress(self):
        """
        Decompress the encoded text by traversing the Huffman tree.
        Saves the decompressed text to a file.
        
        Returns:
            str: The decompressed text.
        """
        final_text = []
        node = self.root
        for bit in self.text_string:
            node = node.left if bit == "0" else node.right
            if node.char:
                final_text.append(node.char)
                node = self.root
        
        final_text_str = ''.join(final_text)
        with open(self.file + "-decompressed.txt", 'w', encoding='utf-8', newline='\n') as output_file:
            output_file.write(final_text_str)
        
        return final_text_str

def main():
    parser = argparse.ArgumentParser(description="Huffman Decompression Utility")
    parser.add_argument('infile', type=str, help="Input file for decompression")
    args = parser.parse_args()

    start_time = time.time()
    HuffmanDecoder(args.infile)
    end_time = time.time()

    print("Time taken to decode the compressed file: {:.2f} seconds".format(end_time - start_time))

if __name__ == '__main__':
    main()
