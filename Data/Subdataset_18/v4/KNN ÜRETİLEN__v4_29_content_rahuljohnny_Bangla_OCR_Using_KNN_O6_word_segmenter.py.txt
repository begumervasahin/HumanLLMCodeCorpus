import os
import array
import numpy as np
import cv2

arr2 = []

def word_segmenter(count2):
    """
    Segment words from the image and save the segmented images based on count2.
    """
    img_path = "words.png"
    img_testing_numbers2 = cv2.imread(img_path)
    
    if img_testing_numbers2 is None:
        print("Error: image not read from file\n\n")
        os.system("pause")
        return

    height, width, channels = img_testing_numbers2.shape
    print(f"Height: {height}")
    print(f"Width: {width}")

    word_ratio = float(width) / float(height)
    print(f"Ratio: {word_ratio}")

    if word_ratio >= 0.9:
        max_val = 9999
        min_val = -999
        c = array.array('i', (0,) * 100)

        inew = -1
        inew0 = -1

        for i in range(0, height // 2):
            for j in range(width):
                if np.any(img_testing_numbers2[i, j] == 0):
                    c[i] += 1
            
            if c[i] < max_val:
                max_val = c[i]
                inew = i
            elif c[i] - max_val <= width // 3:
                inew0 = i

        irange = max(inew, inew0)
        for i in range(inew - 2, irange + 2):
            for j in range(width):
                img_testing_numbers2[i, j] = 1

    img_gray1 = cv2.cvtColor(img_testing_numbers2, cv2.COLOR_BGR2GRAY)
    img_thresh1 = cv2.adaptiveThreshold(
        img_gray1,
        255,
        cv2.ADAPTIVE_THRESH_MEAN_C,
        cv2.THRESH_BINARY_INV,
        111,
        2
    )

    arr2.append(img_thresh1)
    save_segmented_image(count2, arr2[count2 - 1])

    cv2.imwrite("wordseg.png", arr2[count2])
    cv2.imshow("Segmented Image", img_thresh1)
    cv2.waitKey(0)
    return img_thresh1

def save_segmented_image(count2, image):
    """
    Save the segmented image based on the count2 value.
    """
    save_path = f"CreS2//c{count2 - 1}.png" if count2 > 0 else "CreS2//c-1.png"
    cv2.imwrite(save_path, image)

if __name__ == "__main__":
    count = int(input("Enter count: "))
    word_segmenter(count)
