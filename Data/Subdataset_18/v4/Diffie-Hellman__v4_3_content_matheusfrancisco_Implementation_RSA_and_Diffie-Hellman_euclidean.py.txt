"""
This script includes implementations of various mathematical algorithms
such as GCF, GCD, LCM, and the extended Euclidean algorithm.
"""

def gcf(n1, n2):
    """
    Calculate the greatest common factor (GCF) of two numbers using the Euclidean algorithm.
    
    Parameters:
    n1 (int): The first number.
    n2 (int): The second number.
    
    Returns:
    int: The greatest common factor of n1 and n2.
    """
    while n2:
        n1, n2 = n2, n1 % n2
    return n1

def gcd(p, q):
    """
    Calculate the greatest common divisor (GCD) of two numbers using the Euclidean algorithm.
    
    Parameters:
    p (int): The first number.
    q (int): The second number.
    
    Returns:
    int: The greatest common divisor of p and q.
    """
    while q:
        p, q = q, p % q
    return p

def lcm(n1, n2):
    """
    Calculate the least common multiple (LCM) of two numbers.
    
    Parameters:
    n1 (int): The first number.
    n2 (int): The second number.
    
    Returns:
    int: The least common multiple of n1 and n2.
    """
    return (n1 * n2) // gcf(n1, n2)

def extended_gcd(n1, n2):
    """
    Calculate the extended Euclidean algorithm, which finds integers u and v such that:
    n1 * u + n2 * v = gcd(n1, n2)
    
    Parameters:
    n1 (int): The first number.
    n2 (int): The second number.
    
    Returns:
    tuple: A tuple containing the GCD of n1 and n2, and the coefficients u and v.
    """
    u, v, s, t = 1, 0, 0, 1
    while n2:
        q = n1 // n2
        n1, n2 = n2, n1 - q * n2
        u, s = s, u - q * s
        v, t = t, v - q * t
    return n1, u, v

def main():
    """
    Main function to test the implemented algorithms.
    """
    print('Testing the algorithms:')
    
    gcd_result = gcd(18, 102)
    extended_gcd_result = extended_gcd(5023, 1487)

    print('GCD Result:', gcd_result)
    print('Extended GCD Result:', extended_gcd_result)

if __name__ == '__main__':
    main()
