"""
Overflow Algorithm Implementation
This module implements the overflow algorithm as described in Algorithm 2 of the paper:
https://arxiv.org/pdf/2001.09611v1, referred to as the overflow paper.
The implementation consists of two main functions:
1. solve_linearized_equation(): Solves the linearized overflow equation (6) from the paper.
2. solve_overflow_traffic_equation(): Implements the actual overflow algorithm (Algorithm 2) from the paper.
"""

import copy
import numpy as np
import scipy.linalg as linalg

def solve_linearized_equation(n, alpha, mu, P, Q, A_set, B_set):
    """
    Solves the linearized overflow equation (6) from the overflow paper.
    
    Args:
        n (int): Number of nodes in the network.
        alpha (array-like): Exogenous arrival rates (size nx1).
        mu (array-like): Service capacities (size nx1).
        P (array-like): Substochastic routing matrix (size nxn).
        Q (array-like): Substochastic overflow matrix (size nxn).
        A_set (set): Set of potentially stable nodes.
        B_set (set): Set of overflowing nodes.

    Returns:
        numpy.ndarray: Solution array (size nx1) to the linearized overflow equation (6).
    """
    alpha, mu, P, Q = map(np.asarray, (alpha, mu, P, Q))
    
    P_A = np.zeros((n, n))
    Q_B = np.zeros((n, n))
    
    for i in range(n):
        if i in A_set:
            P_A[i, :] = P[i, :]
        if i in B_set:
            Q_B[i, :] = Q[i, :]
    
    P_A_bar = P - P_A
    I = np.eye(n)
    X = np.transpose(I - P_A - Q_B)
    z = np.transpose(alpha + mu @ (P_A_bar - Q_B))
    
    y = linalg.solve(X, z)
    return y.T

def solve_overflow_traffic_equation(alpha, mu, P, Q, count=False):
    """
    Solves the overflow equation (3) from the overflow paper using the overflow algorithm (Algorithm 2).
    
    Args:
        alpha (array-like): Exogenous arrival rates (size nx1).
        mu (array-like): Service capacities (size nx1).
        P (array-like): Substochastic routing matrix (size nxn).
        Q (array-like): Substochastic overflow matrix (size nxn).
        count (bool): If True, the function also returns iteration counts.

    Returns:
        numpy.ndarray: Solution array (size nx1) to the overflow equation (3).
        int (optional): Number of iterations to compute the solution.
        int (optional): Maximum number of iterations allowed.
    """
    n = len(alpha)
    alpha, mu, P, Q = map(np.asarray, (alpha, mu, P, Q))
    
    solve_count = 0
    N_set = set(range(n))
    B_set = set()
    previous_B_set = copy.deepcopy(N_set)
    
    while B_set != previous_B_set:
        previous_B_set = copy.deepcopy(B_set)
        A_set = set()
        previous_A_set = copy.deepcopy(N_set)
        
        while A_set != previous_A_set:
            previous_A_set = copy.deepcopy(A_set)
            labda = solve_linearized_equation(n, alpha, mu, P, Q, A_set, B_set)
            A_set = {i for i in range(n) if labda[i] < mu[i]}
            solve_count += 1
        
        B_set = N_set - A_set
    
    zeros = np.zeros(n)
    labda_check = alpha + np.minimum(labda, mu) @ P + np.maximum(labda - mu, zeros) @ Q
    
    if not np.allclose(labda, labda_check):
        print("\nWarning: solution may be incorrect\n")
    
    if count:
        max_count = int(1 + (0.5 * n * (n + 1)))
        return labda, solve_count, max_count
    else:
        return labda
