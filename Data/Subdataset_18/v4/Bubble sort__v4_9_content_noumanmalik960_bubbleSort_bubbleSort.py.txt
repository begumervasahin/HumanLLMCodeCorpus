def bubble_sort(arr):
    swapped = True
    while swapped:
        swapped = False
        for i in range(len(arr) - 1):
            if arr[i] > arr[i + 1]:
                arr[i], arr[i + 1] = arr[i + 1], arr[i]
                swapped = True

def main():
    arr = [9, 8, 6, 6, 5, 4, 3, 2, 1, 0]
    print("Original array:", arr)
    bubble_sort(arr)
    print("Sorted array:", arr)

if __name__ == "__main__":
    main()
