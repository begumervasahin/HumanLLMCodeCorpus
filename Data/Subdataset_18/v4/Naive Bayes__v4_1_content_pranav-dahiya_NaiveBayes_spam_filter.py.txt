
import glob
import pickle
import numpy as np
from multiprocessing import Process, Lock
from nltk.stem import WordNetLemmatizer

def compute_probability(vocabulary, ignore=0):
    """
    Computes the conditional and prior probabilities for spam and ham emails.

    Args:
        vocabulary (dict): The vocabulary of words.
        ignore (int): The part to ignore for cross-validation.

    Returns:
        tuple: Conditional probabilities and prior probabilities.
    """
    conditional_probability = [{key: 1 for key in vocabulary.keys()}, {key: 1 for key in vocabulary.keys()}]
    priori_probability = [0, 0]
    lemmatizer = WordNetLemmatizer()

    for i in range(1, 11):
        if i != ignore:
            filenames = glob.glob(f"lingspam/part{i}/*.txt")
            for filename in filenames:
                flag = {key: True for key in vocabulary.keys()}
                label = int("spmsg" in filename)
                priori_probability[label] += 1

                with open(filename, "rb") as f:
                    text = f.readlines()
                    for line in text:
                        for word in line.decode().split():
                            word = lemmatizer.lemmatize(word)
                            if word in vocabulary and flag[word]:
                                conditional_probability[label][word] += 1
                                flag[word] = False

    for label in [0, 1]:
        total = priori_probability[label]
        conditional_probability[label] = {key: value / total for key, value in conditional_probability[label].items()}

    total_samples = sum(priori_probability)
    priori_probability = [count / total_samples for count in priori_probability]

    return conditional_probability, priori_probability

def classify(conditional_probability, priori_probability, filename):
    """
    Classifies a given email as spam or ham.

    Args:
        conditional_probability (list): Conditional probabilities for spam and ham.
        priori_probability (list): Prior probabilities for spam and ham.
        filename (str): The path to the email file.

    Returns:
        int: 1 if spam, 0 if ham.
    """
    spam_prob, ham_prob = 1, 1
    words = list(conditional_probability[0].keys())
    lemmatizer = WordNetLemmatizer()

    with open(filename, "rb") as f:
        text = f.readlines()
        for line in text:
            for word in line.decode().split():
                word = lemmatizer.lemmatize(word)
                if word in words:
                    ham_prob *= conditional_probability[0][word]
                    spam_prob *= conditional_probability[1][word]
                    words.remove(word)

    for word in words:
        ham_prob *= (1 - conditional_probability[0][word])
        spam_prob *= (1 - conditional_probability[1][word])

    ham_prob *= priori_probability[0]
    spam_prob *= priori_probability[1]

    return int(spam_prob > ham_prob)

def test(conditional_probability, priori_probability, folder):
    """
    Tests the classifier on a specific folder.

    Args:
        conditional_probability (list): Conditional probabilities for spam and ham.
        priori_probability (list): Prior probabilities for spam and ham.
        folder (int): The folder number to test.

    Returns:
        tuple: True positives, false positives, true negatives, false negatives.
    """
    filenames = glob.glob(f"lingspam/part{folder}/*.txt")
    TP, FP, TN, FN = 0, 0, 0, 0

    for filename in filenames:
        label = int("spmsg" in filename)
        predicted_label = classify(conditional_probability, priori_probability, filename)

        if label and predicted_label:
            TP += 1
        elif label and not predicted_label:
            FN += 1
        elif not label and not predicted_label:
            TN += 1
        else:
            FP += 1

    return TP, FP, TN, FN

def validate(vocabulary, part, lock):
    """
    Validates the model using cross-validation.

    Args:
        vocabulary (dict): The vocabulary of words.
        part (int): The part to validate.
        lock (Lock): A multiprocessing lock for synchronized printing.
    """
    conditional_probability, priori_probability = compute_probability(vocabulary, part)
    TP, FP, TN, FN = test(conditional_probability, priori_probability, part)

    with lock:
        print(f"{part}, {TP}, {FP}, {TN}, {FN},")

def main():
    """
    Main function to load vocabularies and start the validation process.
    """
    for v in range(1, 5):
        filename = f"vocabulary{v}.pickle"
        print(filename)

        with open(filename, "rb") as f:
            vocabulary = pickle.load(f)

        lock = Lock()
        processes = [Process(target=validate, args=(vocabulary, i, lock)) for i in range(1, 11)]

        for process in processes:
            process.start()
        for process in processes:
            process.join()

if __name__ == '__main__':
    main()
