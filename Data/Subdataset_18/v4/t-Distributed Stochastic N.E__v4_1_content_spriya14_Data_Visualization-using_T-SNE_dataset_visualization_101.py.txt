# Importing required libraries
import pandas as pd
import numpy as np
from sklearn.preprocessing import LabelEncoder, StandardScaler
from sklearn.model_selection import train_test_split
from sklearn.manifold import TSNE
import matplotlib.pyplot as plt

# Load the dataset
data_url = "https://d396qusza40orc.cloudfront.net/predmachlearn/pml-training.csv"
data = pd.read_csv(data_url)

# Display the first few rows of the dataset
print("Initial Data Preview:")
print(data.head())

# Data Cleaning

# Count the number of missing elements in each column
missing_counts = data.isnull().sum()

# Select columns without any missing values
columns_without_null = missing_counts[missing_counts == 0].index
data_cleaned = data[columns_without_null]

# Remove the first seven columns (assumed to be irrelevant)
data_cleaned = data_cleaned.iloc[:, 7:]

# Print the list of remaining columns
print("\nRemaining Columns After Cleaning:")
print(data_cleaned.columns)

# Feature Extraction and Label Encoding

# Extract feature columns
X = data_cleaned.iloc[:, :-1].values

# Standardize the features
scaler = StandardScaler()
X_standardized = scaler.fit_transform(X)

# Extract the target labels
y = data_cleaned.iloc[:, -1].values

# Encode the target labels to numeric values
label_encoder = LabelEncoder()
y_encoded = label_encoder.fit_transform(y)

# Display unique class labels
class_labels = np.unique(y)
print("\nClass Labels:")
print(class_labels)

# Split the data into training and testing sets
test_size = 0.1  # 10% of the data will be used for testing
X_train, X_test, y_train, y_test = train_test_split(X_standardized, y_encoded, test_size=test_size, random_state=0)

# t-SNE Visualization

# Initialize t-SNE with 2 components
tsne = TSNE(n_components=2, random_state=0)

# Fit and transform the test data
X_test_2d = tsne.fit_transform(X_test)

# Scatter Plot for t-SNE Visualization

# Define markers and colors for different classes
markers = ['s', 'd', 'o', '^', 'v']
colors = ['red', 'blue', 'lightgreen', 'purple', 'cyan']
color_map = dict(zip(np.unique(y_test), colors))

# Create the plot
plt.figure(figsize=(10, 8))
for idx, class_label in enumerate(np.unique(y_test)):
    plt.scatter(X_test_2d[y_test == class_label, 0], X_test_2d[y_test == class_label, 1], 
                c=color_map[class_label], marker=markers[idx], label=label_encoder.inverse_transform([class_label])[0])

# Label the axes and set the title
plt.xlabel('t-SNE Component 1')
plt.ylabel('t-SNE Component 2')
plt.legend(loc='upper left')
plt.title('t-SNE Visualization of Test Data')

# Show the plot
plt.show()
