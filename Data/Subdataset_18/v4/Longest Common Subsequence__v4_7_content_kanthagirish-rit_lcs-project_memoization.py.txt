import sys
import numpy as np

num_recursive_calls = 0

def memoization(x, y):
    """
    Computes the length of the longest common subsequence (LCS) between two strings using a memoized algorithm.
    
    :param x: First string sequence
    :param y: Second string sequence
    :return: Length of the longest common subsequence between x and y
    """
    m = len(x)
    n = len(y)
    c = np.ones((m, n), dtype='int64') * -1
    return memoized_LCS(x, y, c)

def memoized_LCS(x, y, c):
    """
    Helper function to compute LCS with memoization.
    
    :param x: First string sequence
    :param y: Second string sequence
    :param c: 2D numpy array of size (len(x), len(y)) initialized to -1
    :return: Length of the longest common subsequence of x and y
    """
    global num_recursive_calls
    num_recursive_calls += 1

    if len(x) == 0 or len(y) == 0:
        return 0

    i = len(x) - 1
    j = len(y) - 1

    if c[i, j] > -1:
        return c[i, j]

    if x[-1] == y[-1]:
        q = 1 + memoized_LCS(x[:-1], y[:-1], c)
    else:
        q = max(memoized_LCS(x, y[:-1], c), memoized_LCS(x[:-1], y, c))

    c[i, j] = q
    return q

def test(x, y):
    """
    Test function to compute the LCS length of two strings.
    
    :param x: First string sequence
    :param y: Second string sequence
    :return: Length of the longest common subsequence of x and y
    """
    return memoization(x, y)

if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Usage: python3 " + __file__ + " <input_file>")
    else:
        input_file = sys.argv[1]
        with open(input_file, 'r') as f:
            x = f.readline().strip()
            y = f.readline().strip()
            print("LCS length: " + str(test(x, y)))
            print("Number of recursive calls: " + str(num_recursive_calls))
