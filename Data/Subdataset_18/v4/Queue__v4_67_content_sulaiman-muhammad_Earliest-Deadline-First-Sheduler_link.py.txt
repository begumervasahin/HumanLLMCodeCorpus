class Node:
    def __init__(self, val=None, nxt=None):
        """
        Initialize a new node.
        
        :param val: The value of the node.
        :param nxt: The next node in the linked list.
        """
        self.val = val
        self.nxt = nxt

class LinkedList:
    def __init__(self):
        """Initialize an empty linked list."""
        self.head = None
        self.tail = None

    def insert(self, x, pos):
        """
        Insert a new node with value x after the node pos.
        
        :param x: The value to be inserted.
        :param pos: The position after which the new node is to be inserted.
        """
        temp = Node(x)
        temp.nxt = pos.nxt
        pos.nxt = temp

    def append(self, x):
        """
        Append a new node with value x to the end of the linked list.
        
        :param x: The value to be appended.
        """
        temp = Node(x)
        if self.head is None:
            self.head = temp
            self.tail = temp
        else:
            self.tail.nxt = temp
            self.tail = temp

    def search(self, x):
        """
        Search for a node with value x in the linked list.
        
        :param x: The value to be searched for.
        :return: The node with value x if found, else False.
        """
        cur = self.head
        while cur is not None and cur.val != x:
            cur = cur.nxt
        return cur if cur is not None else False

    def print_list(self):
        """Print the values of all nodes in the linked list."""
        cur = self.head
        while cur:
            print(cur.val)
            cur = cur.nxt

    def findpos(self, x):
        """
        Insert a new node with value x in the linked list, maintaining sorted order based on CPU attribute.
        
        :param x: The value to be inserted.
        """
        temp = Node(x)
        if self.head is None:
            self.head = temp
            self.tail = temp
        elif self.head.val.cpu > x.cpu:
            temp.nxt = self.head
            self.head = temp
        else:
            cur = self.head
            while cur.nxt is not None and cur.nxt.val.cpu < x.cpu:
                cur = cur.nxt
            temp.nxt = cur.nxt
            cur.nxt = temp
            if temp.nxt is None:
                self.tail = temp

    def reverse(self):
        """Reverse the linked list."""
        prev = None
        cur = self.head
        while cur:
            nxt = cur.nxt
            cur.nxt = prev
            prev = cur
            cur = nxt
        self.head = prev
        self.print_list()

    def insert_head(self, x):
        """
        Insert a new node with value x at the head of the linked list.
        
        :param x: The value to be inserted.
        """
        temp = Node(x)
        if self.head is None:
            self.head = temp
            self.tail = temp
        else:
            temp.nxt = self.head
            self.head = temp

    def remove(self):
        """
        Remove and return the node at the head of the linked list.
        
        :return: The removed node.
        """
        if self.head is None:
            print("empty")
            return None
        temp = self.head
        self.head = self.head.nxt
        if self.head is None:
            self.tail = None
        temp.nxt = None
        return temp

def main():
    l = LinkedList()
    l.insert_head(1)
    l.findpos(Node(3))  # Node should be created with a value that has a 'cpu' attribute
    l.print_list()
    print("tail", l.tail.val)
    l.remove()
    l.remove()
    l.remove()
    l.remove()
