import numpy as np

def as_row_matrix(X):
    """
    Convert a list of vectors into a row matrix.
    """
    if len(X) == 0:
        return np.array([])
    mat = np.empty((0, X[0].size), dtype=X[0].dtype)
    for row in X:
        mat = np.vstack((mat, np.asarray(row).reshape(1, -1)))
    return mat

def as_column_matrix(X):
    """
    Convert a list of vectors into a column matrix.
    """
    if len(X) == 0:
        return np.array([])
    mat = np.empty((X[0].size, 0), dtype=X[0].dtype)
    for col in X:
        mat = np.hstack((mat, np.asarray(col).reshape(-1, 1)))
    return mat

def pca(X, num_components=0):
    """
    Perform Principal Component Analysis (PCA) on the dataset X.
    """
    n, d = X.shape
    if num_components <= 0 or num_components > n:
        num_components = n

    # Center the data
    mu = X.mean(axis=0)
    X = X - mu

    # Calculate covariance matrix and its eigenvalues and eigenvectors
    if n > d:
        C = np.dot(X.T, X)
        eigenvalues, eigenvectors = np.linalg.eigh(C)
    else:
        C = np.dot(X, X.T)
        eigenvalues, eigenvectors = np.linalg.eigh(C)
        eigenvectors = np.dot(X.T, eigenvectors)

    # Normalize eigenvectors
    eigenvectors = np.apply_along_axis(lambda v: v / np.linalg.norm(v), 0, eigenvectors)

    # Sort eigenvalues and eigenvectors
    idx = np.argsort(-eigenvalues)
    eigenvalues = eigenvalues[idx]
    eigenvectors = eigenvectors[:, idx]

    # Select the top num_components eigenvalues and eigenvectors
    eigenvalues = eigenvalues[:num_components].copy()
    eigenvectors = eigenvectors[:, :num_components].copy()

    return eigenvalues, eigenvectors, mu

def project(W, X, mu=None):
    """
    Project data X onto the PCA subspace defined by W.
    """
    if mu is None:
        return np.dot(X, W)
    return np.dot(X - mu, W)

def reconstruct(W, Y, mu=None):
    """
    Reconstruct the data from its PCA projection.
    """
    if mu is None:
        return np.dot(Y, W.T)
    return np.dot(Y, W.T) + mu
