import inspect

class DPifier:
    def __init__(self, f, lambda_mode=False):
        self.dp = {}
        
        # The Y combinator to enable memoization
        def Y(f):
            def g(*args):
                if args in self.dp:
                    return self.dp[args]
                else:
                    self.dp[args] = f(Y(f))(*args)
                    return self.dp[args]
            return g
        
        self.Y = Y
        
        # Determine the function handling based on lambda_mode
        if lambda_mode:
            self.f = f
        else:
            self.f = self.__parseFunction(f)
    
    def __call__(self, *args):
        return self.Y(self.f)(*args)
    
    def __parseFunction(self, f):
        functionName = f.__name__
        s = inspect.getsource(f)
        
        # Replace the function name with a generic one to avoid conflicts
        s = s.replace(functionName, 'h', 1)
        s = s.replace(functionName + '(', 'abc(', 1)
        
        # Define the inner function to execute the modified code
        def g(abc):
            exec(s, {'abc': abc}, globals())
            return h
        
        return g
