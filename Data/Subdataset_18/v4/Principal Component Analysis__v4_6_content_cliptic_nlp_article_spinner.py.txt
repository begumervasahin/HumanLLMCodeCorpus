from __future__ import print_function, division
from future.utils import iteritems
from builtins import range
import nltk
import random
import numpy as np
from bs4 import BeautifulSoup

# Load positive reviews
with open('electronics/positive.review') as f:
    positive_reviews = BeautifulSoup(f.read(), 'html.parser')
positive_reviews = positive_reviews.findAll('review_text')

# Build the trigrams model
trigrams = {}
for review in positive_reviews:
    review_text = review.text.lower()
    tokens = nltk.tokenize.word_tokenize(review_text)
    for i in range(len(tokens) - 2):
        key = (tokens[i], tokens[i+2])
        if key not in trigrams:
            trigrams[key] = []
        trigrams[key].append(tokens[i+1])

# Calculate trigram probabilities
trigram_probabilities = {}
for key, words in iteritems(trigrams):
    if len(set(words)) > 1:
        word_counts = {}
        total_count = 0
        for word in words:
            if word not in word_counts:
                word_counts[word] = 0
            word_counts[word] += 1
            total_count += 1
        word_probabilities = {word: float(count) / total_count for word, count in iteritems(word_counts)}
        trigram_probabilities[key] = word_probabilities

# Function to sample a word based on its probability distribution
def random_sample(probabilities):
    r = random.random()
    cumulative = 0
    for word, probability in iteritems(probabilities):
        cumulative += probability
        if r < cumulative:
            return word

# Function to test the text spinner
def test_spinner():
    review = random.choice(positive_reviews)
    original_text = review.text.lower()
    print("Original:", original_text)
    
    tokens = nltk.tokenize.word_tokenize(original_text)
    for i in range(len(tokens) - 2):
        if random.random() < 0.2:
            key = (tokens[i], tokens[i+2])
            if key in trigram_probabilities:
                new_word = random_sample(trigram_probabilities[key])
                tokens[i+1] = new_word
    
    spun_text = " ".join(tokens).replace(" .", ".").replace(" '", "'").replace(" ,", ",").replace("$ ", "$").replace(" !", "!")
    print("Spun:", spun_text)

if __name__ == '__main__':
    test_spinner()
