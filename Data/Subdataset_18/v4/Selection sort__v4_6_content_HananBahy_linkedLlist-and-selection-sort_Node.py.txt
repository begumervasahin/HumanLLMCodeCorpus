class Node:
    def __init__(self, init_data):
        """
        Initialize a new node with the given data.
        
        Args:
            init_data: The initial data for the node.
        """
        self.data = init_data
        self.next = None

    def get_data(self):
        """
        Get the data stored in the node.
        
        Returns:
            The data stored in the node.
        """
        return self.data

    def get_next(self):
        """
        Get the next node.
        
        Returns:
            The next node.
        """
        return self.next

    def set_data(self, new_data):
        """
        Set the data stored in the node.
        
        Args:
            new_data: The new data to store in the node.
        """
        self.data = new_data

    def set_next(self, new_next):
        """
        Set the next node.
        
        Args:
            new_next: The new next node.
        """
        self.next = new_next
