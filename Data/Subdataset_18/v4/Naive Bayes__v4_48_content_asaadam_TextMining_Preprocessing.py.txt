import string
import re
from Sastrawi.Stemmer.StemmerFactory import StemmerFactory
import collections

class Preprocessing:
    stopword = open("stopword-list.txt", "r").read().split('\n')

    @staticmethod
    def cleaning(text):
        """
        Clean the input text by removing links, punctuation, and numbers.
        
        :param text: Input text
        :return: Cleaned text
        """
        # Remove links
        text = re.sub(r'[\S]+\.(net|com|org|info|edu|gov|uk|de|ca|jp|fr|au|us|ru|ch|it|nl|se|no|es|mil)[\S]*', ' ', text)
        
        # Replace punctuation with spaces
        replace_punctuation = str.maketrans(string.punctuation, ' ' * len(string.punctuation))
        text = text.translate(replace_punctuation)
        
        # Remove numbers
        text = text.translate(str.maketrans('', '', '1234567890'))
        
        # Remove extra whitespace
        text = re.sub(r'\s+', ' ', text).strip()
        
        return text

    @staticmethod
    def case_folding(text):
        """
        Convert the text to lowercase.
        
        :param text: Input text
        :return: Lowercased text
        """
        return text.casefold()

    @staticmethod
    def tokenisasi(text):
        """
        Tokenize the text into words.
        
        :param text: Input text
        :return: List of words
        """
        return text.split()

    @staticmethod
    def filtering(words):
        """
        Remove stopwords from the list of words.
        
        :param words: List of words
        :return: List of words without stopwords
        """
        return [word for word in words if word not in Preprocessing.stopword]

    @staticmethod
    def type(words):
        """
        Remove duplicate words while preserving order.
        
        :param words: List of words
        :return: List of unique words
        """
        return list(collections.OrderedDict.fromkeys(words))

    @staticmethod
    def stemming(words):
        """
        Perform stemming on the list of words.
        
        :param words: List of words
        :return: List of stemmed words
        """
        factory = StemmerFactory()
        stemmer = factory.create_stemmer()
        return [stemmer.stem(word) for word in words]

    @staticmethod
    def all_in_one(text):
        """
        Perform all preprocessing steps: cleaning, case folding, tokenizing, type, filtering, and stemming.
        
        :param text: Input text
        :return: List of preprocessed words
        """
        cleaned_text = Preprocessing.cleaning(text)
        folded_text = Preprocessing.case_folding(cleaned_text)
        tokens = Preprocessing.tokenisasi(folded_text)
        unique_tokens = Preprocessing.type(tokens)
        stemmed_words = Preprocessing.stemming(unique_tokens)
        filtered_words = Preprocessing.filtering(stemmed_words)
        return filtered_words

    @staticmethod
    def all_in_one_without_type(text):
        """
        Perform all preprocessing steps except removing duplicate words.
        
        :param text: Input text
        :return: List of preprocessed words
        """
        print('Preprocessing...')
        print(text)
        cleaned_text = Preprocessing.cleaning(text)
        folded_text = Preprocessing.case_folding(cleaned_text)
        tokens = Preprocessing.tokenisasi(folded_text)
        stemmed_words = Preprocessing.stemming(tokens)
        filtered_words = Preprocessing.filtering(stemmed_words)
        return filtered_words
