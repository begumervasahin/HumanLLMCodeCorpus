def insertion_sort(array):
    """Sorts a list of integers in place using the insertion sort algorithm."""
    for i in range(1, len(array)):
        current_value = array[i]
        j = i - 1
        # Move elements of array[0..i-1], that are greater than current_value,
        # to one position ahead of their current position
        while j >= 0 and array[j] > current_value:
            array[j + 1] = array[j]
            j -= 1
        array[j + 1] = current_value

def read_data(file_path):
    """Reads a file and returns a list of lists, where each inner list contains integers."""
    number_list = []
    with open(file_path, "r") as file:
        for line in file:
            numbers = list(map(int, line.split()[1:]))  # Skip the first value and convert to integers
            number_list.append(numbers)
    return number_list

def write_data(file_path, sorted_lists):
    """Writes the sorted lists of integers to a specified file."""
    with open(file_path, "w") as file:
        for lst in sorted_lists:
            file.write(" ".join(map(str, lst)) + "\n")

if __name__ == "__main__":
    input_file = "data.txt"
    output_file = "insertionsort.txt"

    # Read data from the input file
    unsorted_lists = read_data(input_file)

    # Sort each list using insertion sort
    for lst in unsorted_lists:
        insertion_sort(lst)

    # Write the sorted lists to the output file
    write_data(output_file, unsorted_lists)
