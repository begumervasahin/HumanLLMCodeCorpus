from difflib import ndiff, restore

# Define the input strings and split them into lines
string1 = 'one\ntwo\nthree\n'
string2 = 'ore\ntree\nemu\n'

# Split the strings into lines and keep the newline characters
lines1 = string1.splitlines(keepends=True)
lines2 = string2.splitlines(keepends=True)

# Compute the differences between the two sets of lines
diff = ndiff(lines1, lines2)

# Convert the diff object to a list
diff_list = list(diff)

# Restore the first and second sets of lines from the diff
restored1 = ''.join(restore(diff_list, 1))
restored2 = ''.join(restore(diff_list, 2))

# Print the restored strings
print(restored1, end="")
print(restored2, end="")
