import random

def generate_array(size):
    """
    Generate an array of given size with random integers 
    in the range from -size to size.
    """
    return [random.randint(-size, size) for _ in range(size)]

def generate_payload():
    """
    Generate a payload consisting of arrays of different sizes.
    The sizes are 10, 100, 1000, 10000, 100000, 1000000, 10000000.
    """
    sizes = [10, 100, 1000, 10000, 100000, 1000000, 10000000]
    payload = [generate_array(size) for size in sizes]
    return payload

# Example usage
if __name__ == "__main__":
    payload = generate_payload()
    for i, arr in enumerate(payload):
        print(f"Array of size {len(arr)}: {arr[:10]}...")  # Print only the first 10 elements for brevity
