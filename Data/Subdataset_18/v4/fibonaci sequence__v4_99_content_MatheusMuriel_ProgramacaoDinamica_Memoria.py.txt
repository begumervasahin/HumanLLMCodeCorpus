class Cerebro:
    def __init__(self, limite, taxa_de_limpeza):
        """
        Initialize the Cerebro class.

        Args:
            limite (int): The memory limit for the Cerebro instance.
            taxa_de_limpeza (float): The cleaning rate for the Cerebro instance.
        """
        self.memoria = {}
        self.usos = {}
        self.limite = limite
        self.taxa_de_limpeza = taxa_de_limpeza

    def nova_memoria(self, nome, valor):
        """
        Add a new memory entry.

        Args:
            nome (str): The key for the memory entry.
            valor (int): The value for the memory entry.
        """
        nome = str(nome)
        valor = int(valor)

        if len(self.memoria) >= self.limite:
            self.limpar_memoria()

        self.memoria[nome] = valor
        self.usos[nome] = 1

    def get_lembranca(self, nome):
        """
        Retrieve a memory entry if it exists.

        Args:
            nome (str): The key for the memory entry.

        Returns:
            int: The value of the memory entry if it exists, otherwise -1.
        """
        if nome in self.memoria:
            self.usos[nome] += 1
            return self.memoria[nome]
        return -1

    def limpar_memoria(self):
        """
        Clean the memory to reduce its size when the limit is reached.
        """
        print("Executing memory cleanup.")
        target_size = int(self.limite * self.taxa_de_limpeza)
        while len(self.memoria) > target_size:
            menor_registro = min(self.usos.items(), key=lambda x: x[1])
            self.memoria.pop(menor_registro[0])
            self.usos.pop(menor_registro[0])

if __name__ == "__main__":
    # Example usage of the Cerebro class
    cerebro = Cerebro(limite=5, taxa_de_limpeza=0.7)
    cerebro.nova_memoria("1", 1)
    cerebro.nova_memoria("2", 1)
    cerebro.nova_memoria("3", 2)
    cerebro.nova_memoria("4", 3)
    cerebro.nova_memoria("5", 5)
    
    print(cerebro.get_lembranca("1"))  # Output: 1
    
    cerebro.nova_memoria("6", 8)  # This should trigger memory cleanup
    
    print(cerebro.get_lembranca("1"))  # Output: -1 if "1" was cleaned
    print(cerebro.get_lembranca("6"))  # Output: 8
