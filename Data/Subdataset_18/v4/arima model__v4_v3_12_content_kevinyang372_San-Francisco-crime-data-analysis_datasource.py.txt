import pandas as pd

def load_data(file_name):
    """
    Load data from a CSV file.

    Parameters:
    file_name (str): The name or path of the CSV file to load.

    Returns:
    pd.DataFrame: The data loaded from the CSV file.
    """
    # Read the CSV file using pandas
    data = pd.read_csv(file_name)
    return data

# Example usage:
data_file_name = "your_data_file.csv"  # Specify your CSV file path here

# Load the data from the CSV file
loaded_data = load_data(data_file_name)

# Display the first few rows of the loaded data
print("Preview of the loaded data:")
print(loaded_data.head())
