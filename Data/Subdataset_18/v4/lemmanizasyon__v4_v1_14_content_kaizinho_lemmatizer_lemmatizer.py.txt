import re
import json

def lemmatize(sentence, word_list):
    """
    Replace words in a sentence with their lemmatized forms based on the provided word list.

    :param sentence: The input sentence to be lemmatized.
    :param word_list: A dictionary mapping words to their lemmatized forms.
    :return: The lemmatized sentence.
    """
    lemmatized_sentence = []
    for word in sentence.split():
        lemmatized_sentence.append(word_list.get(word, word))
    return ' '.join(lemmatized_sentence)

def extract_words(raw_data_file, output_file="data/swedish_words.json"):
    """
    Extracts words and their lemmatizations from a raw data file and saves them to a JSON file.

    :param raw_data_file: Path to the input file containing raw data.
    :param output_file: Path to the output JSON file where the extracted data will be saved.
    """
    # Regular expression patterns for matching different lines in the raw data file
    section_pattern = r"^(===(Verb|Substantiv|Adjektiv|Adverb|Konjuktion|Pronomen|Räkneord|Interjektion|Konjugation)===)$"
    word_pattern = r"^'''([a-zA-ZåäöÅÄÖ]+)'''$"
    word_with_lemma_pattern = r"^'''([a-zA-ZåäöÅÄÖ]+)''' \(([^)]+)\)$"

    data = {}
    current_type = None

    # Open and read the raw data file
    with open(raw_data_file, "r", encoding='utf-8') as input_file:
        for line in input_file:
            # Check if the line matches the section pattern
            match = re.match(section_pattern, line)
            if match:
                current_type = match.group(2)
                data.setdefault(current_type, {})
            # Check if the line matches the word pattern (without lemma)
            elif re.match(word_pattern, line) and current_type:
                word = re.match(word_pattern, line).group(1)
                data[current_type][word] = ""
            # Check if the line matches the word with lemma pattern
            elif re.match(word_with_lemma_pattern, line) and current_type:
                match = re.match(word_with_lemma_pattern, line)
                word = match.group(1)
                lemma = match.group(2)
                data[current_type][word] = lemma

    # Save the extracted data to the output JSON file
    with open(output_file, "w", encoding='utf-8') as output_file:
        json.dump(data, output_file, ensure_ascii=False, indent=4)

# Example usage
extract_words("raw_data_file.txt", "swedish_words.json")
