def shell_sort(alist):
    """Sorts a list using the Shell sort algorithm."""
    sublist_count = len(alist) // 2
    while sublist_count > 0:
        for start_position in range(sublist_count):
            gap_insertion_sort(alist, start_position, sublist_count)
        print(f"After increments of size {sublist_count}, the list is {alist}")
        sublist_count //= 2

def gap_insertion_sort(alist, start, gap):
    """Helper function to perform a gapped insertion sort."""
    for i in range(start + gap, len(alist), gap):
        current_value = alist[i]
        position = i
        while position >= gap and alist[position - gap] > current_value:
            alist[position] = alist[position - gap]
            position -= gap
        alist[position] = current_value
