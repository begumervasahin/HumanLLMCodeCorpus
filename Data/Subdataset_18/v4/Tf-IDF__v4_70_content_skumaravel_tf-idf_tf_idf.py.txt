import math
import string

def tf_idf(corpus, document, term):
    """
    Calculate the TF-IDF score for a given term in a document within a corpus.

    :param corpus: List of documents (each document is a string)
    :param document: The document in which to calculate the TF-IDF for the term
    :param term: The term for which to calculate the TF-IDF
    :return: TF-IDF score for the term in the document
    """
    # Convert the document to lowercase and remove punctuation
    document = document.lower()
    exclude = set(string.punctuation)
    document = ''.join(ch for ch in document if ch not in exclude)
    
    # Split the document into terms
    doc_terms = document.split()
    total_terms_in_doc = len(doc_terms)
    
    # Calculate term frequency (TF)
    term_frequency = doc_terms.count(term) / total_terms_in_doc
    
    # Calculate the number of documents in the corpus
    total_documents = len(corpus)
    
    # Calculate the number of documents containing the term
    docs_with_term = sum(1 for doc in corpus if term in ''.join(ch for ch in doc.lower() if ch not in exclude).split())
    
    # Calculate inverse document frequency (IDF)
    if docs_with_term != 0:
        idf = math.log(total_documents / docs_with_term)
    else:
        idf = 0
    
    # Return the TF-IDF score
    return term_frequency * idf

# Example usage
corpus = [
    "This is a sample document.",
    "This document is another sample document.",
    "And this is a different document."
]
document = "This is a sample document."
term = "sample"

tf_idf_score = tf_idf(corpus, document, term)
print(f"TF-IDF score for term '{term}' in the document: {tf_idf_score}")
