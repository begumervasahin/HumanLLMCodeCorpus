import timeit

def wrapper(func, *args, **kwargs):
    """
    Wrap the function and its arguments in another function
    to delay its execution.
    """
    def wrapped():
        return func(*args, **kwargs)
    return wrapped

def time_function(func, *args, **kwargs):
    """
    Measure the execution time of a function with the provided arguments.
    """
    wrapped = wrapper(func, *args, **kwargs)
    return timeit.timeit(wrapped, number=1)

# Example usage
if __name__ == "__main__":
    # Define a sample function for demonstration
    def sample_function(x, y):
        return x + y

    # Time the sample function with arguments 10 and 20
    elapsed_time = time_function(sample_function, 10, 20)
    print(f"Elapsed time: {elapsed_time} seconds")
