a_unsorted = [6, 2, 7, 8, 3, 1, 10, 5, 4, 9]
a_sorted = []

while a_unsorted:
    a_min = a_unsorted[0]
    min_index = 0

    for i in range(len(a_unsorted)):
        if a_unsorted[i] < a_min:
            a_min = a_unsorted[i]
            min_index = i

    a_unsorted.pop(min_index)
    a_sorted.append(a_min)

print(a_sorted)
