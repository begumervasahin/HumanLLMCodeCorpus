import numpy as np
import scipy.sparse.csgraph as csgraph
import dijkstra

def test_dijkstra_against_scipy():
    """
    Test the custom Dijkstra implementation against SciPy's implementation.
    """
    # Generate a random weighted graph
    W = np.random.random((100, 100))
    W = (W + W.T) / 2  # Make the graph symmetric
    W[W < 1.0] = np.inf  # Set weights less than 1 to infinity to represent no path
    W = scipy.sparse.csr_matrix(W)  # Convert to CSR format for efficient processing

    # Test for different seed values
    for seed in [0, 1, 2]:
        result_custom = dijkstra.dijkstra(W, seed)
        result_scipy = csgraph.dijkstra(W, indices=seed)[0]
        
        # Ensure the results from both implementations are the same
        assert np.allclose(result_custom, result_scipy), f"Mismatch found for seed {seed}"

if __name__ == "__main__":
    test_dijkstra_against_scipy()
