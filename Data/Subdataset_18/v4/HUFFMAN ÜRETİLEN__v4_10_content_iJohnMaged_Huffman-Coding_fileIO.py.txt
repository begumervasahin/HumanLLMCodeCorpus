import errno
import os

def splitext_(path):
    """Splits the file path into file name and extension."""
    file_name, ext = os.path.splitext(path)
    return file_name, ext[1:]

def get_basename(path):
    """Returns the base file name without its extension."""
    return splitext_(os.path.basename(path))[0]

def get_filename(path):
    """Returns the file name with its extension."""
    return os.path.basename(path)

def get_file_extension(path):
    """Returns the file extension without the leading period."""
    return os.path.splitext(path)[1][1:]

def add_to_filename(path, suffix):
    """Appends a suffix to the file name before the extension."""
    file_name, ext = os.path.splitext(path)
    return f"{file_name}{suffix}{ext}"

def change_extension(path, new_extension):
    """Changes the file extension to the specified new extension."""
    file_name, _ = os.path.splitext(path)
    return f"{file_name}.{new_extension}"

def get_directory_name(path):
    """Returns the directory name of the provided file path."""
    return os.path.dirname(os.path.abspath(path))

def get_file_size(path):
    """Returns the size of the file in bytes."""
    return os.path.getsize(path)

def get_directory_size(path):
    """Calculates the total size of all files in the directory."""
    total_size = 0
    for root, _, files in os.walk(path):
        for file in files:
            total_size += get_file_size(os.path.join(root, file))
    return total_size

def generate_output_file_path(directory_path):
    """Generates an output file path with a .huffman extension."""
    directory_path = os.path.join(directory_path, '')
    return os.path.join(os.getcwd(), '') + os.path.basename(os.path.dirname(directory_path)) + ".huffman"

def get_huffman_file_path(path, is_directory=False):
    """Generates the appropriate Huffman file path based on whether it's a file or directory."""
    if is_directory:
        return generate_output_file_path(path)
    return os.path.join(os.getcwd(), '') + change_extension(path, "huffman")

def prepare_files_for_processing(path):
    """Identifies and prepares files for Huffman processing."""
    if os.path.isdir(path):
        files, original_size = list_files_in_directory(path)
        output_path = get_huffman_file_path(path, is_directory=True)
    elif os.path.isfile(path):
        files = {os.path.basename(path): path}
        original_size = get_file_size(path)
        output_path = get_huffman_file_path(path)
    else:
        raise FileNotFoundError(f"The specified path does not exist: {path}")
    return files, original_size, output_path

def list_files_in_directory(path):
    """Lists all files in the directory, returning their relative paths and total size."""
    files = {}
    total_size = 0
    path = path.rstrip(os.sep)
    parent_dir = os.path.dirname(path)
    
    for root, _, file_names in os.walk(path):
        for file_name in file_names:
            file_path = os.path.join(root, file_name)
            total_size += get_file_size(file_path)
            relative_path = os.path.relpath(file_path, parent_dir)
            files[relative_path] = file_path
    
    return files, total_size

def create_directory_if_not_exists(path):
    """Creates the directory if it does not already exist."""
    directory = os.path.dirname(path)
    if not directory:
        return
    
    try:
        os.makedirs(directory)
    except OSError as e:
        if e.errno != errno.EEXIST:
            raise

def invert_dictionary(d):
    """Inverts the keys and values of a dictionary."""
    return {v: k for k, v in d.items()}
