import pandas as pd
import numpy as np
import re

# Load data
df1 = pd.read_excel('results_25july.xlsx')

# Split and expand the 'all' column
df1['sum_all'] = df1['all'].str.split(':%')
dfn = df1[['sum_all']].stack().apply(pd.Series)

# Load and rename columns of the second DataFrame
df2 = pd.read_excel('results_25july_2.xlsx')  # Assuming another file for df2
df2.columns = [
    'ctri_id', 'study_design', 'public_title', 'scientific_title', 'country', 
    'sample_size', 'phase', 'trial_type', 'sponsor_1', 'sponsor_2', 
    'condition', 'intervention', 'primary_outcome'
]

# Clean the 'ctri_id' column
df2['ctri'] = df2['ctri_id'].str.split('\\\\').map(lambda x: x[0]).str.replace("^\W+", '')

# Clean the 'country' column
df2['country'] = df2['country'].str.replace('(?<=[a-z])(?=[A-Z])', ' ').str.replace('\W+', ' ')

# Clean the 'sample_size' column
df2['sample_size'] = (df2['sample_size'].str.split('=', n=1).map(lambda x: x[-1])
                      .str.split('Sample Size from India').map(lambda x: x[0])
                      .str.replace('\W+', ''))

# Clean the 'sponsor_1' column
df2['sponsor_1'] = df2['sponsor_1'].replace('Address', np.nan, regex=True).astype(str).str.split(',').map(lambda x: x[0])

# Clean the 'sponsor_2' column
df2['sponsor_20'] = df2['sponsor_2'].str.split(',\s+{0:').map(lambda x: x[0]).str.split(',').map(lambda x: x[0])
df2['sponsor_21'] = df2['sponsor_2'].str.split(',\s+{0:').map(lambda x: x[-1]).str.split(',').map(lambda x: x[0])
df2['sponsor_2'] = df2['sponsor_20'] + df2['sponsor_21']

# Define regex patterns for cleaning
io = re.compile('(?i)(^\W*name|none|nil\W*$)')
io1 = re.compile('(?i)(^\W*name|none|NIL\W*\w+$)')

# Clean 'sponsor_2', 'condition', and 'intervention' columns
df2['sponsor_2'] = (df2['sponsor_20'].str.replace('0', '').str.replace('\W+', ' ')
                    .replace(io, np.nan, regex=True))

df2['condition'] = df2['condition'].replace(io, np.nan, regex=True)

df2['intervention'] = (dfn[11].str.split('Intervention').map(lambda x: x[-1])
                       .str.replace('(0|1|2)(:)', '').str.replace('(0|1|2)(\.)', '')
                       .str.replace('[^A-Za-z0-9-/,:]', ' ')
                       .replace('^.*xa0Year.*$', np.nan, regex=True)
                       .replace('^.*Type\s+Name\s+Details.*$', np.nan, regex=True)
                       .str.strip())

# Clean the 'primary_outcome' column
df2['primary_outcome'] = (df2['primary_outcome'].str.replace('([0123])(\.|\:)', '')
                          .astype(str).str.replace('[^A-Za-z0-9-/:]', ' ')
                          .str.strip().str.replace('\s+', ' '))

# Further clean the 'intervention' column
df2['intervention'] = (df2['intervention'].astype(str).str.split(',', n=1)
                       .map(lambda x: x[-1]).str.split(',').map(lambda x: x[0])
                       .str.strip())

# Select and save final columns
df3 = df2[['ctri', 'public_title', 'scientific_title', 'sponsor_1',
           'sponsor_2', 'condition', 'intervention', 'primary_outcome']]
df3.to_excel('last_4663_ctri_25july.xlsx', index=False)
df2.to_excel('last_4663_ctri_allfields_25july.xlsx', index=False)

# Load, clean, and combine another DataFrame
df4 = pd.read_excel('clean_ctri_19123_24july.xlsx')
df4.drop(['ctri_y', 'intervention_y'], axis=1, inplace=True)
df4.columns = ['ctri', 'sponsor_1', 'sponsor_2', 'primary_outcome', 
               'condition', 'public_title', 'scientific_title', 'intervention']

# Reset index and clean up combined DataFrame
df3.reset_index(inplace=True)
df3.drop(['level_1', 'level_0'], axis=1, inplace=True)
ctri_19519 = pd.concat([df3, df4]).drop_duplicates(subset='ctri')
ctri_19519.to_excel('ctri_clean_total_19519.xlsx', index=False)
