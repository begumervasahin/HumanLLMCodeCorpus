import json
import requests
import geojson

"""
Geocoding with the api-adresse.data.gouv API.
For each address, a list of points is returned. These points are ranked in descending order based on a confidence score.
The latitudes and longitudes used are those of the point with the highest score.
'datas.json' is the result of the 'requetes_facteurs.py' program.
"""

GEOCODING_URL = "https://api-adresse.data.gouv.fr/search/?q="

def geocode_address(address):
    """
    Geocode an address using the French government geocoding API.

    :param address: The address to geocode.
    :return: A tuple (latitude, longitude) of the geocoded address.
    """
    address_query = address.replace(" ", "+")  # URL encode the address
    response = requests.get(GEOCODING_URL + address_query)  # Send request to the API
    feature_collection = geojson.loads(response.text)  # Parse the response as GeoJSON
    point = feature_collection['features'][0]['geometry']['coordinates']  # Extract coordinates
    return point[1], point[0]  # Return latitude and longitude

def update_addresses_with_geolocation(data):
    """
    Geocode each address in the data dictionary and update it with latitude and longitude.

    :param data: List of dictionaries containing address data.
    :return: Updated list of dictionaries with geolocation.
    """
    for entry in data:
        address = entry['adresse']
        latitude, longitude = geocode_address(address)
        entry['latitude'] = latitude
        entry['longitude'] = longitude
    return data

def main():
    """
    Main function to load addresses, geocode them, and save the updated data.
    """
    # Load the data from datas.json
    with open('datas.json', 'r', encoding='utf-8') as json_file:
        data = json.load(json_file)

    # Geocode and update addresses
    updated_data = update_addresses_with_geolocation(data)

    # Save the updated data to facteurs_localisation.json
    with open('facteurs_localisation.json', 'w', encoding='utf-8') as output_file:
        json.dump(updated_data, output_file, ensure_ascii=False, indent=4)

if __name__ == "__main__":
    main()
