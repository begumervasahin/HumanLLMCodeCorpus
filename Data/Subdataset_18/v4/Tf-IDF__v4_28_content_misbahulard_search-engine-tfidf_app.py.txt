from flask import Flask, render_template, request
from forms import SearchForm
import math
from tfidf import search as Searching

# Initialize the Flask application
app = Flask(__name__)
app.secret_key = 'development'

@app.route("/")
def index():
    """Render the index page with the search form."""
    form = SearchForm()
    return render_template("index.html", form=form)

@app.route("/search")
def search():
    """Handle the search functionality and render the search results page."""
    form = SearchForm()
    query = request.args.get('keyword')
    page = int(request.args.get('page', 1))  # Default to page 1 if not specified
    
    # Perform the search using the provided query
    data = Searching.search_for(query)
    data_length = len(data)
    
    # Pagination settings
    pagination_size = 5
    start = (page - 1) * pagination_size
    end = min(page * pagination_size, data_length)
    total_pages = math.ceil(data_length / pagination_size)
    
    return render_template(
        "search.html",
        form=form,
        data=data[start:end],
        keyword=query,
        page=page,
        start=start,
        end=end,
        total=total_pages
    )

if __name__ == '__main__':
    app.run(debug=True)
