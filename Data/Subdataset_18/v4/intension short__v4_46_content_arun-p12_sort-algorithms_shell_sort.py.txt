def shell_sort(arr, verbose=0, desc=False):
    def insertion_sort_with_gap(arr, gap, start_index):
        for i in range(start_index + gap, len(arr), gap):
            current_value = arr[i]
            j = i
            while j >= gap and arr[j - gap] > current_value:
                arr[j] = arr[j - gap]
                j -= gap
            arr[j] = current_value
            if verbose == 2:
                print(f"  Subarray with gap {gap}: {arr}")

    gap = len(arr) // 2
    while gap > 0:
        if verbose:
            print(f"Gap: {gap}")
        for start_index in range(gap):
            insertion_sort_with_gap(arr, gap, start_index)
        gap //= 2

    if desc:
        arr.reverse()

    return arr
