import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
import datetime

def cluster_analysis(data_array, k):
    """
    Performs K-Means clustering on the given data and plots the results.

    Parameters:
    data_array (array-like): The input data array containing samples, shape (n_samples, n_features).
    k (int): The number of clusters to form.

    Returns:
    labels (array): Cluster labels for each sample, shape (n_samples,).
    """
    # Initialize the KMeans model with the specified number of clusters
    kmeans = KMeans(n_clusters=k)
    
    # Fit the model to the data and predict cluster labels
    labels = kmeans.fit_predict(data_array)
    
    # Extract x and y coordinates from the data array for plotting
    xs = data_array[:, 0]
    ys = data_array[:, 1]
    
    # Get the coordinates of the cluster centers
    centroids = kmeans.cluster_centers_
    centroids_x = centroids[:, 0]
    centroids_y = centroids[:, 1]
    
    # Create a scatter plot of the data points colored by their cluster labels
    plt.scatter(xs, ys, c=labels, alpha=0.5, label='Data Points')
    # Plot the cluster centers as diamonds
    plt.scatter(centroids_x, centroids_y, marker='D', s=100, color='red', label='Centroids')
    
    # Add a title with the current date and time
    plt.title(f"Cluster Analysis - {datetime.datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    plt.legend()
    
    # Show the plot
    plt.show()
    
    print("\nInspect the cluster plot shown. When ready, close the plot and the images will be sorted "
          "and copied into subdirectories.")
    
    return labels
