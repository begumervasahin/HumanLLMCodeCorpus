import operator
from math import sqrt
from functools import reduce

def gcd(x, y):
    """Calculate the greatest common divisor of x and y."""
    return x if y == 0 else gcd(y, x % y)

def dumb_factor(x, primeset):
    """Factorize x over a given set of primes.

    Args:
        x (int): The number to factorize.
        primeset (set): A set of prime numbers.

    Returns:
        list: A list of tuples (p_i, a_i) where p_i is a prime factor and a_i is its exponent.
              Returns an empty list if x cannot be fully factorized over the primeset.
    """
    factors = []
    for p in primeset:
        exponent = 0
        while x % p == 0:
            exponent += 1
            x //= p
        if exponent > 0:
            factors.append((p, exponent))
    return factors if x == 1 else []

def primes(limit):
    """Generate a set of prime numbers up to a given limit using the Sieve of Eratosthenes.

    Args:
        limit (int): The upper limit for generating prime numbers.

    Returns:
        set: A set of prime numbers up to the given limit.
    """
    primeset = set()
    sieve = [True] * limit
    sieve[0] = sieve[1] = False
    for i, is_prime in enumerate(sieve):
        if is_prime:
            primeset.add(i)
            for n in range(i * i, limit, i):
                sieve[n] = False
    return primeset

def intsqrt(x):
    """Calculate the integer square root of x.

    Args:
        x (int): The number to calculate the square root of.

    Returns:
        int: The integer square root of x.
    """
    L, H = 1, x
    if H < L:
        L, H = H, L
    while H - L > 1:
        m = (L + H) // 2
        if x // m > m:
            L = m
        else:
            H = m
    return L if L * L == x else H

def prod(factors):
    """Calculate the product of a list of numbers.

    Args:
        factors (list): A list of numbers to multiply.

    Returns:
        int: The product of the numbers in the list.
    """
    return reduce(operator.mul, factors, 1)
