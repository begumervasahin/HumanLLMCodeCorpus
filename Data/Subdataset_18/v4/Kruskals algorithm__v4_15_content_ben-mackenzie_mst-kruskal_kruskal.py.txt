from DisJSets import DisJSet
from heapqriorityqueue import HeapPriorityQueue

def MST_Kruskal(graph):
    """
    Computes a Minimum Spanning Tree (MST) of a graph using Kruskal's algorithm.
    
    Returns:
        - A list of edges that comprise the MST.
        - The total weight of the MST.
        - A set of vertices with their associated edges.
        
    Assumptions:
        - The elements of the graph's edges represent weights.
    """
    mst = []  # List to store the edges of the MST
    priority_queue = HeapPriorityQueue()  # Priority queue to select the edge with the smallest weight
    disjoint_set = DisJSet()  # Disjoint set to manage connected components
    vertex_to_group = {}  # Dictionary to map each vertex to its group in the disjoint set

    # Initialize each vertex as its own group
    for vertex in graph.vertices():
        vertex_to_group[vertex] = disjoint_set.make_group(vertex)

    # Insert all edges of the graph into the priority queue
    for edge in graph.edges():
        priority_queue.insert(edge)

    num_vertices = graph.vertex_count()

    # Process edges in order of increasing weight
    while len(mst) < num_vertices - 1 and not priority_queue.is_empty():
        edge = priority_queue.deleteMin()  # Get the edge with the smallest weight
        u, v = edge.endpoints()  # Get the vertices connected by the edge
        group_u = disjoint_set.find(vertex_to_group[u])
        group_v = disjoint_set.find(vertex_to_group[v])

        # If u and v are in different groups, add the edge to the MST and unite the groups
        if group_u != group_v:
            mst.append(edge)
            disjoint_set.unionSets(group_u, group_v)

    return mst
