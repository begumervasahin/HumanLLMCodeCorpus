import re
from urllib import robotparser
from urllib.parse import urljoin
from downloader import Downloader

def get_robots_parser(robots_url):
    """
    Return the robots parser object using the provided robots_url.
    """
    rp = robotparser.RobotFileParser()
    rp.set_url(robots_url)
    rp.read()
    return rp

def get_links(html):
    """
    Return a list of links extracted from the HTML content using a regular expression.
    """
    webpage_regex = re.compile(r'<a[^>]+href=["\'](.*?)["\']', re.IGNORECASE)
    return webpage_regex.findall(html)

def link_crawler(start_url, link_regex, robots_url=None, user_agent='wswp',
                 proxies=None, delay=3, max_depth=4, num_retries=2, cache={}, scraper_callback=None):
    """
    Crawl from the given start URL, following links that match the provided link_regex.
    
    Args:
        start_url (str): The initial URL to start crawling from.
        link_regex (str): Regular expression to match links for crawling.
        robots_url (str, optional): URL of the site's robots.txt. Defaults to start_url + '/robots.txt'.
        user_agent (str, optional): User agent for the crawler. Defaults to 'wswp'.
        proxies (list of dicts, optional): List of possible proxies for HTTP/HTTPS requests.
        delay (int, optional): Delay in seconds between requests to the same domain. Defaults to 3.
        max_depth (int, optional): Maximum crawl depth to avoid traps. Defaults to 4.
        num_retries (int, optional): Number of retry attempts for failed requests. Defaults to 2.
        cache (dict, optional): Cache dictionary with URLs as keys and response dicts as values. Defaults to {}.
        scraper_callback (function, optional): Callback function to be called with URL and HTML content. Defaults to None.
    """
    crawl_queue = [start_url]
    seen = {}
    
    # Determine robots.txt URL if not provided
    if not robots_url:
        robots_url = f'{start_url}/robots.txt'
    
    # Initialize the robots parser
    rp = get_robots_parser(robots_url)
    
    # Initialize the downloader
    downloader = Downloader(delay=delay, user_agent=user_agent, proxies=proxies, cache=cache)
    
    while crawl_queue:
        url = crawl_queue.pop()
        
        # Check robots.txt rules
        if rp.can_fetch(user_agent, url):
            depth = seen.get(url, 0)
            
            if depth == max_depth:
                print(f'Skipping {url} due to depth')
                continue
            
            # Download the page content
            html = downloader(url, num_retries=num_retries)
            
            if not html:
                continue
            
            # If a scraper callback is provided, use it to extract links
            if scraper_callback:
                links = scraper_callback(url, html) or []
            else:
                links = []
            
            # Extract links from the HTML content
            for link in get_links(html) + links:
                if re.match(link_regex, link):
                    abs_link = urljoin(start_url, link)
                    
                    if abs_link not in seen:
                        seen[abs_link] = depth + 1
                        crawl_queue.append(abs_link)
        else:
            print(f'Blocked by robots.txt: {url}')
