class BinarySearchTree:
    def __init__(self):
        self.root = None  # Initialize the root of the tree to None

    def insert(self, value):
        if not self.root:
            # If the tree is empty, set the root to a new node with the given value
            self.root = BinarySearchTreeNode(value)
        else:
            # Otherwise, insert the value in the appropriate position in the tree
            self.root.insert(value)
        return self.root  # Return the root of the tree after insertion

    def lookup(self, value):
        if not self.root:
            return None  # If the tree is empty, return None
        elif self.root.value == value:
            return self.root  # If the root's value matches, return the root
        else:
            # Otherwise, search for the value starting from the root
            return self.root.lookup(value)


class BinarySearchTreeNode:
    def __init__(self, value):
        self.value = value  # Node's value
        self.left = None    # Left child node
        self.right = None   # Right child node

    def insert(self, value):
        if value < self.value:
            if not self.left:
                # If the left child doesn't exist, create a new node there
                self.left = BinarySearchTreeNode(value)
                return self.left
            else:
                # Otherwise, recursively insert the value into the left subtree
                self.left.insert(value)
        elif value > self.value:
            if not self.right:
                # If the right child doesn't exist, create a new node there
                self.right = BinarySearchTreeNode(value)
                return self.right
            else:
                # Otherwise, recursively insert the value into the right subtree
                self.right.insert(value)
        return None  # Return None if the value is already in the tree

    def lookup(self, value):
        if self.value == value:
            return self  # If the current node's value matches, return the node
        elif value < self.value and self.left:
            # If the value is less, search the left subtree
            return self.left.lookup(value)
        elif value > self.value and self.right:
            # If the value is greater, search the right subtree
            return self.right.lookup(value)
        return None  # Return None if the value is not found
