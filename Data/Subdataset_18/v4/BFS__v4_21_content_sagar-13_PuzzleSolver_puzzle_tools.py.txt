"""
This module contains functions and classes for working with puzzles,
including solving puzzles using BFS and DFS algorithms.
"""

from puzzle import Puzzle
from collections import deque
import sys

# Set a higher recursion limit
sys.setrecursionlimit(10**6)

def depth_first_solve(puzzle):
    """
    Solve the given puzzle using Depth-First Search (DFS).

    Args:
        puzzle (Puzzle): The puzzle to be solved.

    Returns:
        PuzzleNode: A node containing the solution, or None if no solution is found.
    """
    visited = set()
    stack = deque([PuzzleNode(puzzle)])

    while stack:
        node = stack.pop()

        for config in node.puzzle.extensions():
            if str(config) in visited:
                continue
            if config.is_solved():
                return PuzzleNode(config)
            if config.fail_fast() or not config.extensions():
                continue

            child_node = PuzzleNode(config)
            visited.add(str(child_node.puzzle))
            stack.append(child_node)

    return None

def breadth_first_solve(puzzle):
    """
    Solve the given puzzle using Breadth-First Search (BFS).

    Args:
        puzzle (Puzzle): The puzzle to be solved.

    Returns:
        PuzzleNode: A node containing the solution, or None if no solution is found.
    """
    visited = set()
    queue = deque([PuzzleNode(puzzle)])

    while queue:
        node = queue.popleft()

        for config in node.puzzle.extensions():
            if str(config) in visited:
                continue
            if config.is_solved():
                return PuzzleNode(config)
            if config.fail_fast() or not config.extensions():
                continue

            child_node = PuzzleNode(config)
            visited.add(str(child_node.puzzle))
            queue.append(child_node)

    return None

class PuzzleNode:
    """
    A node in a tree of puzzle configurations.
    Each node refers to other configurations that it can be extended to.
    """

    def __init__(self, puzzle=None, children=None, parent=None):
        """
        Initialize a new PuzzleNode.

        Args:
            puzzle (Puzzle): The puzzle configuration for this node.
            children (list[PuzzleNode]): List of child nodes.
            parent (PuzzleNode): The parent node.

        Returns:
            None
        """
        self.puzzle = puzzle
        self.parent = parent
        self.children = children[:] if children else []

    def __eq__(self, other):
        """
        Check if this PuzzleNode is equivalent to another.

        Args:
            other (PuzzleNode): The node to compare with.

        Returns:
            bool: True if the nodes are equivalent, False otherwise.
        """
        return (
            isinstance(other, PuzzleNode) and
            self.puzzle == other.puzzle and
            set(self.children) == set(other.children)
        )

    def __str__(self):
        """
        Return a human-readable string representation of this PuzzleNode.

        Returns:
            str: A string representation of this node.
        """
        children_str = "\n".join(str(child) for child in self.children)
        return f"{self.puzzle}\n\n{children_str}"
