import os
import re
import nltk
import codecs
import string
import subprocess
import unicodedata

# Set up NLTK data path
nltk_data_path = os.path.expanduser('~/nltk_data')
if not os.path.exists(nltk_data_path):
    os.mkdir(nltk_data_path)
print("Does path exist:", os.path.exists(nltk_data_path))

# Define paths
DOCS_PDF = os.path.join(os.getcwd(), "pdf")
CORPUS = os.path.join(os.getcwd(), "corpus")

# Function to get PDF documents
def get_pdf_docs(path=DOCS_PDF):
    """
    Returns a filtered list of paths to PDF files.
    """
    print(path)
    for name in os.listdir(path):
        if name.endswith('.pdf'):
            yield os.path.join(path, name)

# Count the number of PDF documents
print(len(list(get_pdf_docs())))

# Function to extract text corpus from PDF documents
def extract_pdf_corpus(DOCS_PDF=DOCS_PDF, corpusval=CORPUS):
    """
    Extracts a text corpus from the PDF documents.
    """
    if not os.path.exists(corpusval):
        os.mkdir(corpusval)

    for path in get_pdf_docs(DOCS_PDF):
        tp = os.path.join(os.getcwd(), 'pdf2txt.py')
        document = subprocess.check_output(['python', tp, path])
        
        filen = os.path.splitext(os.path.basename(path))[0] + ".txt"
        outp = os.path.join(corpusval, filen)

        document = document.decode("utf-8")
        document = document.replace('\n', ' ').replace('\r', ' ')
        document = document.replace(',', ' ').replace('-', '').replace(':', ' ')

        with codecs.open(outp, 'w', encoding='utf-8') as f:
            f.write(document)

# Extract PDF corpus
extract_pdf_corpus()

# Load the corpus with NLTK
kddcorpus = nltk.corpus.PlaintextCorpusReader(CORPUS, '.*\.txt')
wordsvals = nltk.FreqDist(kddcorpus.words())

# Get word count and vocabulary size
count = sum(wordsvals.values())
vocab = len(wordsvals)

# Print corpus statistics
print('----------------------------------------------------------------------')
print("Corpus contains a vocabulary of {} and a word count of {}.".format(vocab, count))
print('----------------------------------------------------------------------')
print(wordsvals.hapaxes())
print('----------------------------------------------------------------------')
print(wordsvals.most_common())
