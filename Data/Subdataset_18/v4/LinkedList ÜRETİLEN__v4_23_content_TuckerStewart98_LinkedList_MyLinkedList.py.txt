class Node:
    """
    Node class to represent each element in the linked list.
    """
    def __init__(self, data, next_node=None):
        self._data = data
        self._next = next_node

    def get_data(self):
        return self._data

    def get_next(self):
        return self._next

    def set_next(self, next_node):
        self._next = next_node

    def __str__(self):
        return str(self._data)


class MyLinkedList:
    """
    Linked list class to represent a singly linked list.
    """
    def __init__(self):
        self._head = None
        self._last = None
        self._size = 0

    def size(self):
        """
        Returns the number of elements in the linked list.
        """
        return self._size

    def add(self, data):
        """
        Adds a new node with the given data to the end of the linked list.
        """
        new_node = Node(data)
        if self._head is None:
            self._head = new_node
            self._last = new_node
        else:
            self._last.set_next(new_node)
            self._last = new_node
        self._size += 1

    def get(self, index):
        """
        Returns the data of the node at the given index.
        Raises an exception if the index is out of bounds.
        """
        if index >= self._size or index < 0:
            raise IndexError('Index out of bounds')
        current_node = self._head
        for _ in range(index):
            current_node = current_node.get_next()
        return current_node.get_data()

if __name__ == "__main__":
    list1 = MyLinkedList()
    list1.add(1)
    list1.add(5)
    list1.add(-7)
    for i in range(list1.size()):
        print(list1.get(i))
