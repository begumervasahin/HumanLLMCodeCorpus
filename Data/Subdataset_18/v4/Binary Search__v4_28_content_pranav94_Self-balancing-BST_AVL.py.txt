from BST import BST
from TreeNode import TreeNode

class AVLTreeNode(TreeNode):
    """Node class for the AVL tree, extending the base TreeNode class with a height attribute."""
    
    def __init__(self, val):
        super().__init__(val)
        self.height = 0  # Initialize the height of the node to 0

class AVL(BST):
    """AVL Tree class extending the basic Binary Search Tree (BST) class."""

    def rotate_left(self, node):
        """Perform a left rotation around the given node."""
        new_root = node.right
        node.right = new_root.left
        new_root.left = node

        # Update the heights of the affected nodes
        self.calculate_height(node)
        self.calculate_height(new_root)

        return new_root

    def rotate_right(self, node):
        """Perform a right rotation around the given node."""
        new_root = node.left
        node.left = new_root.right
        new_root.right = node

        # Update the heights of the affected nodes
        self.calculate_height(node)
        self.calculate_height(new_root)

        return new_root

    def rebalance(self, node):
        """Rebalance the tree if it becomes unbalanced."""
        balance_factor = self.balance(node)

        if balance_factor > 1:  # Left-heavy
            if self.balance(node.left) >= 0:  # Left-Left case
                return self.rotate_right(node)
            else:  # Left-Right case
                node.left = self.rotate_left(node.left)
                return self.rotate_right(node)

        if balance_factor < -1:  # Right-heavy
            if self.balance(node.right) <= 0:  # Right-Right case
                return self.rotate_left(node)
            else:  # Right-Left case
                node.right = self.rotate_right(node.right)
                return self.rotate_left(node)

        return node  # No rotation needed

    def insert_node(self, node, val):
        """Recursive helper method to insert a value into the AVL tree."""
        if not node:
            return AVLTreeNode(val)  # Create a new node if the current position is empty

        if val < node.val:
            node.left = self.insert_node(node.left, val)
        else:
            node.right = self.insert_node(node.right, val)

        self.calculate_height(node)  # Update the height of the current node
        return self.rebalance(node)  # Rebalance the tree if necessary

    def insert(self, val):
        """Insert a value into the AVL tree."""
        self.root = self.insert_node(self.root, val)

    def delete_node(self, node, val):
        """Recursive helper method to delete a value from the AVL tree."""
        if not node:
            return None

        if val < node.val:
            node.left = self.delete_node(node.left, val)
        elif val > node.val:
            node.right = self.delete_node(node.right, val)
        else:
            # Node to be deleted found
            if not node.left and not node.right:  # No children
                return None
            elif not node.left:  # One child (right)
                return node.right
            elif not node.right:  # One child (left)
                return node.left
            else:  # Two children
                predecessor = self.predecessor(node)
                node.val = predecessor.val
                node.left = self.delete_node(node.left, predecessor.val)

        self.calculate_height(node)  # Update the height of the current node
        return self.rebalance(node)  # Rebalance the tree if necessary

    def delete(self, val):
        """Delete a value from the AVL tree."""
        self.root = self.delete_node(self.root, val)

    def calculate_height(self, node):
        """Calculate and update the height of a node."""
        node.height = 1 + max(self.height(node.left), self.height(node.right))

    @staticmethod
    def height(node):
        """Return the height of a node."""
        return node.height if node else 0

    def balance(self, node):
        """Calculate the balance factor of a node."""
        return self.height(node.left) - self.height(node.right) if node else 0
