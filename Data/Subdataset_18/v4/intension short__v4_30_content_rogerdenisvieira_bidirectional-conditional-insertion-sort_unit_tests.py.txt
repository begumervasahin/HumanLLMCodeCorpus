import unittest
from sorters import QS, BCIS, IS
import random

class SortersTestCase(unittest.TestCase):
    def setUp(self):
        """Set up the initial conditions for each test."""
        self.expected = list(range(10))  # Expected sorted list
        self.non_sorted = self.expected.copy()  # Copy the expected list for shuffling
        random.shuffle(self.non_sorted)  # Shuffle to create a non-sorted list
        print(f"Setting things up... Expected: {self.expected} Non-Sorted: {self.non_sorted}")
    
    def test_bcis_sorting(self):
        """Test the BCIS sorting algorithm."""
        sorter = BCIS()
        sorter.sort(self.non_sorted, 0, len(self.non_sorted) - 1)
        self.assertEqual(self.expected, self.non_sorted)
    
    def test_is_sorting(self):
        """Test the IS sorting algorithm."""
        sorter = IS()
        sorter.sort(self.non_sorted)
        self.assertEqual(self.expected, self.non_sorted)
    
    def test_qs_sorting(self):
        """Test the QS sorting algorithm."""
        sorter = QS()
        sorter.sort(self.non_sorted, 0, len(self.non_sorted) - 1)
        self.assertEqual(self.expected, self.non_sorted)
    
    def test_fake(self):
        """An intentionally failing test to demonstrate test failures."""
        print("This test is expected to fail.")
        self.assertEqual("foo", 1)

if __name__ == '__main__':
    unittest.main()
