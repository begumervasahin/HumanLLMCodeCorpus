os.environ["CUDA_DEVICE_ORDER"] = "PCI_BUS_ID"

# Initialize NLTK tools
stop_words = stopwords.words('english')
porter_stemmer = PorterStemmer()
lancaster_stemmer = LancasterStemmer()
wordnet_lemmatizer = WordNetLemmatizer()

# Number of neighbors for k-NN
K = 250

def get_word_embeddings(data):
    """
    Load the Word2Vec model and generate sentence embeddings.
    """
    model = gensim.models.KeyedVectors.load_word2vec_format('GoogleNews-vectors-negative300.bin', binary=True)
    out_sent_vectors = []

    for sent in data:
        sent_vec = [model[word] for word in sent.split() if word in model]
        sent_vec = np.mean(sent_vec, axis=0) if sent_vec else np.zeros(300)
        out_sent_vectors.append(sent_vec)

    return np.array(out_sent_vectors)

def clean_data(input_str, remove_stop=True):
    """
    Clean the input string by removing HTML tags and non-alphanumeric characters.
    """
    input_str = re.sub('<[^<]+?>', ' ', input_str)
    input_str = re.sub('[^A-Za-z0-9]+', ' ', input_str.lower())
    return input_str

def extract_features(train_x, train_y, test_x):
    """
    Extract TF-IDF and Word2Vec features from the data.
    """
    vectorizer = TfidfVectorizer(min_df=0.001, max_df=0.98, ngram_range=(1, 4))
    selector = SelectKBest(chi2, k=7000)

    train_x_tfidf = vectorizer.fit_transform(train_x).toarray()
    train_x_tfidf = selector.fit_transform(train_x_tfidf, train_y)
    test_x_tfidf = vectorizer.transform(test_x).toarray()
    test_x_tfidf = selector.transform(test_x_tfidf)

    train_x_no_stop = [' '.join([word for word in sent.split() if word not in stop_words]) for sent in train_x]
    test_x_no_stop = [' '.join([word for word in sent.split() if word not in stop_words]) for sent in test_x]

    train_x_embed = get_word_embeddings(train_x_no_stop)
    test_x_embed = get_word_embeddings(test_x_no_stop)

    train_x_final = np.concatenate((train_x_tfidf, train_x_embed), axis=1)
    test_x_final = np.concatenate((test_x_tfidf, test_x_embed), axis=1)

    return train_x_final, test_x_final

def process_data():
    """
    Process the training and testing data.
    """
    with open('train.dat') as f_train:
        train_data = [line.strip().split('\t') for line in f_train]

    with open('test_data.dat') as f_test:
        test_data = [line.strip() for line in f_test]

    train_x = [clean_data(sample[1], True) for sample in train_data]
    train_y = [int(sample[0]) for sample in train_data]
    test_x = [clean_data(sample, True) for sample in test_data]

    train_x_fit, test_x_fit = extract_features(train_x, train_y, test_x)

    print(train_x_fit.shape, test_x_fit.shape, train_x_fit.dtype, test_x_fit.dtype)

    return np.array(train_y), train_x_fit, test_x_fit

def test_model_locally(train_y, train_x, test_x):
    """
    Test the k-NN model locally.
    """
    X_train, X_test, Y_train, Y_test = train_test_split(train_x, train_y, test_size=0.20, random_state=42)

    x_keys = tf.placeholder(tf.float32, [None, train_x.shape[1]])
    x_queries = tf.placeholder(tf.float32, [None, train_x.shape[1]])

    normalized_keys = tf.nn.l2_normalize(x_keys, axis=1)
    normalized_query = tf.nn.l2_normalize(x_queries, axis=1)

    query_result = tf.matmul(normalized_query, tf.transpose(normalized_keys))
    pred = tf.argmax(query_result, axis=1)

    init = tf.global_variables_initializer()

    with tf.Session() as sess:
        sess.run(init)
        print("Training the model")

        preds = sess.run(query_result, feed_dict={x_keys: X_train, x_queries: X_test})
        preds = tf.transpose(preds)

        values, indices = sess.run(tf.nn.top_k(preds, K))
        y_preds = []

        for top in indices:
            sample_label = [Y_train[neighbor] for neighbor in top]
            y_preds.append(mode(sample_label)[0][0])

        accuracy = np.mean(y_preds == Y_test)
        print(f"Accuracy: {accuracy * 100:.2f}%")

def generate_predictions(train_y, train_x, test_x, prediction_output_file_name):
    """
    Generate predictions for the test set.
    """
    batches_x_test = [test_x[i:i+5000] for i in range(0, len(test_x), 5000)]

    with open(prediction_output_file_name, "w") as f_out:
        for batch in batches_x_test:
            x_keys = tf.placeholder(tf.float32, [None, train_x.shape[1]])
            x_queries = tf.placeholder(tf.float32, [None, train_x.shape[1]])

            normalized_keys = tf.nn.l2_normalize(x_keys, axis=1)
            normalized_query = tf.nn.l2_normalize(x_queries, axis=1)

            query_result = tf.matmul(normalized_query, tf.transpose(normalized_keys))
            pred = tf.argmax(query_result, axis=1)

            init = tf.global_variables_initializer()

            with tf.Session() as sess:
                sess.run(init)
                print("Evaluating tensorflow KNN")

                preds = sess.run(query_result, feed_dict={x_keys: train_x, x_queries: batch})
                preds = tf.transpose(preds)

                values, indices = sess.run(tf.nn.top_k(preds, K))
                y_preds = []

                for top in indices:
                    sample_label = [train_y[neighbor] for neighbor in top]
                    y_preds.append(mode(sample_label)[0][0])

                for pred in y_preds:
                    f_out.write('+1\n' if pred > 0 else '-1\n')

            ops.reset_default_graph()

def run(evaluate_model_locally, prediction_output_file_name):
    """
    Run the main process.
    """
    train_y, train_x, test_x = process_data()

    if evaluate_model_locally:
        test_model_locally(train_y, train_x, test_x)
    else:
        generate_predictions(train_y, train_x, test_x, prediction_output_file_name)

if __name__ == "__main__":
    evaluate_model_locally = False
    prediction_output_file_name = "test.dat"
    run(evaluate_model_locally, prediction_output_file_name)
