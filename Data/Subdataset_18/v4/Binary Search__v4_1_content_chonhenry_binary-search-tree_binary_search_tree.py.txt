class Node:
    """Represents a single node in a binary search tree."""
    def __init__(self, value):
        self.value = value
        self.left = None
        self.right = None

class BinarySearchTree:
    """Implements a binary search tree with various operations."""
    def __init__(self):
        self.root = None

    def insert(self, value):
        """Inserts a new value into the binary search tree."""
        new_node = Node(value)
        if self.root is None:
            self.root = new_node
        else:
            current = self.root
            while True:
                if value < current.value:
                    if current.left is None:
                        current.left = new_node
                        break
                    current = current.left
                elif value > current.value:
                    if current.right is None:
                        current.right = new_node
                        break
                    current = current.right
                else:
                    break  # Value already exists in the tree

    def lookup(self, value):
        """Searches for a node with the given value."""
        current = self.root
        while current:
            if value < current.value:
                current = current.left
            elif value > current.value:
                current = current.right
            else:
                # Value found
                print(f"Value {value} found.")
                left_val = current.left.value if current.left else 'None'
                right_val = current.right.value if current.right else 'None'
                print(f"Left child: {left_val}")
                print(f"Right child: {right_val}")
                return current
        print("Value not found.")
        return None

    def remove(self, value):
        """Removes a node with the given value from the tree."""
        parent = None
        current = self.root

        while current:
            if value < current.value:
                parent = current
                current = current.left
            elif value > current.value:
                parent = current
                current = current.right
            else:
                # Node to be deleted found
                if current.left is None and current.right is None:
                    # Case 1: Node has no children (leaf node)
                    if parent is None:
                        self.root = None
                    elif parent.left == current:
                        parent.left = None
                    else:
                        parent.right = None
                    del current
                elif current.left is None:
                    # Case 2: Node has only a right child
                    if parent is None:
                        self.root = current.right
                    elif parent.left == current:
                        parent.left = current.right
                    else:
                        parent.right = current.right
                    del current
                elif current.right is None:
                    # Case 3: Node has only a left child
                    if parent is None:
                        self.root = current.left
                    elif parent.left == current:
                        parent.left = current.left
                    else:
                        parent.right = current.left
                    del current
                else:
                    # Case 4: Node has two children
                    smallest_parent = current
                    smallest = current.right
                    while smallest.left:
                        smallest_parent = smallest
                        smallest = smallest.left
                    current.value = smallest.value
                    if smallest_parent.left == smallest:
                        smallest_parent.left = smallest.right
                    else:
                        smallest_parent.right = smallest.right
                    del smallest
                return
        print("Value not found in the tree.")

    def breadth_first_search(self):
        """Performs breadth-first search (BFS) traversal of the tree."""
        if self.root is None:
            return []
        
        current = self.root
        queue = [current]
        result = []

        while queue:
            current = queue.pop(0)
            result.append(current.value)
            if current.left:
                queue.append(current.left)
            if current.right:
                queue.append(current.right)

        return result

    def breadth_first_search_recursive(self, queue, arr):
        """Performs BFS traversal using recursion."""
        if len(queue) == 0:
            return arr

        current = queue.pop(0)
        arr.append(current.value)

        if current.left:
            queue.append(current.left)
        if current.right:
            queue.append(current.right)

        return self.breadth_first_search_recursive(queue, arr)

    def height(self, node):
        """Calculates the height of the tree."""
        if node is None:
            return 0
        return 1 + max(self.height(node.left), self.height(node.right))

def in_order_traversal(root):
    """Performs in-order traversal of the tree."""
    if root:
        in_order_traversal(root.left)
        print(root.value)
        in_order_traversal(root.right)

def pre_order_traversal(root):
    """Performs pre-order traversal of the tree."""
    if root:
        print(root.value)
        pre_order_traversal(root.left)
        pre_order_traversal(root.right)

def post_order_traversal(root):
    """Performs post-order traversal of the tree."""
    if root:
        post_order_traversal(root.left)
        post_order_traversal(root.right)
        print(root.value)

    tree = BinarySearchTree()
    values = [50, 28, 78, 12, 40, 66, 90, 9, 18, 37, 43, 57, 69, 81, 93, 3, 10, 15, 21, 30, 38, 42, 46, 54, 60, 67, 72, 80, 84, 92, 96, 13, 29, 55, 56]

    # Perform various operations on the tree
    tree.remove(15)
    print("Breadth-first search:", tree.breadth_first_search())
    print("Height of the tree:", tree.height(tree.root))
    print("Breadth-first search (recursive):", tree.breadth_first_search_recursive([tree.root], []))

    # Traversals
    print("\nIn-order traversal:")
    in_order_traversal(tree.root)

    print("\nPre-order traversal:")
