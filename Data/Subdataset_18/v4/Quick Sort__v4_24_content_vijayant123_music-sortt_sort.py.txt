import subprocess
import os
import sys
from os import listdir
from os.path import isfile, join
from pygame import mixer

# Define paths
source_path = "/home/error-404/Desktop/songs"
save_path = "/home/error-404/Desktop/dance"
alt_path = "/home/error-404/Desktop/others"

# Get a list of all files in the source path
song_files = [f for f in listdir(source_path) if isfile(join(source_path, f))]

# Initialize the mixer
mixer.init()

# Flag to track play/pause state
is_paused = False

# Loop through each file in the source path
for song in song_files:
    print(song)
    # Load and play the song
    mixer.music.load(join(source_path, song))
    mixer.music.play(0, 40)
    
    while True:
        user_input = input()
        
        if user_input == 'q':
            sys.exit(0)  # Quit the program
        
        elif user_input.isdigit():
            # Rewind and play from the specified time
            mixer.music.rewind()
            mixer.music.play(0, int(user_input) * 10)
        
        elif user_input == '':
            # Stop the music and move the file to alt_path
            mixer.music.stop()
            os.rename(join(source_path, song), join(alt_path, song))
            break
        
        elif user_input == '/':
            # Stop the music and move the file to save_path
            mixer.music.stop()
            os.rename(join(source_path, song), join(save_path, song))
            break
        
        elif user_input == 'p':
            if not is_paused:
                mixer.music.pause()  # Pause the music
                is_paused = True
            else:
                mixer.music.unpause()  # Unpause the music
                is_paused = False
