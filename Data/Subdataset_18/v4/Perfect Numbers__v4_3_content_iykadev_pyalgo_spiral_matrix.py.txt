def generate_spiral(n):
    """Generate an n x n spiral matrix."""
    def spiral_value(x, y, n):
        """Recursively calculate the value at position (x, y) in an n x n spiral."""
        if x == -1 and y == 0:
            return -1
        if y == (x + 1) and x < (n // 2):
            return spiral_value(x - 1, y - 1, n - 1) + 4 * (n - y)
        if x < (n - y) and y <= x:
            return spiral_value(y - 1, y, n) + (x - y) + 1
        if x >= (n - y) and y <= x:
            return spiral_value(x, y - 1, n) + 1
        if x >= (n - y) and y > x:
            return spiral_value(x + 1, y, n) + 1
        if x < (n - y) and y > x:
            return spiral_value(x, y - 1, n) - 1

    # Initialize the n x n matrix with zeros
    array = [[0] * n for _ in range(n)]
    
    # Fill the matrix with spiral values
    for x in range(n):
        for y in range(n):
            array[x][y] = spiral_value(y, x, n)
    
    return array

def print_spiral(matrix):
    """Print the spiral matrix in a formatted way."""
    for row in matrix:
        print(" ".join(f"{x:2}" for x in row))

if __name__ == "__main__":
    n = 6
    spiral_matrix = generate_spiral(n)
    print_spiral(spiral_matrix)
