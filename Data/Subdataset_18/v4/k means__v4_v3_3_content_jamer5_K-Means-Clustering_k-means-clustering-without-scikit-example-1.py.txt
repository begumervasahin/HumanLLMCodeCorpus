import numpy as np
import os

def compute_euclidean_distance(point, centroid):
    """Compute the Euclidean distance between a point and a centroid."""
    return np.linalg.norm(point - centroid)

def assign_label_cluster(distances, data_point, centroids):
    """Assign a data point to the nearest centroid."""
    index_of_minimum = min(distances, key=distances.get)
    return [index_of_minimum, data_point, centroids[index_of_minimum]]

def compute_new_centroids(cluster_points):
    """Compute the new centroid as the mean of the cluster points."""
    return np.mean(cluster_points, axis=0)

def iterate_k_means(data_points, centroids, total_iterations):
    """Perform the k-means clustering algorithm."""
    cluster_labels = []
    k = len(centroids)

    for iteration in range(total_iterations):
        clusters = {i: [] for i in range(k)}
        
        # Assign each data point to the nearest centroid
        for data_point in data_points:
            distances = {i: compute_euclidean_distance(data_point, centroids[i]) for i in range(k)}
            label = assign_label_cluster(distances, data_point, centroids)
            clusters[label[0]].append(data_point)
            
            if iteration == total_iterations - 1:
                cluster_labels.append(label)
        
        # Update centroids
        for i in range(k):
            if clusters[i]:
                centroids[i] = compute_new_centroids(clusters[i])

    return cluster_labels, centroids

def print_label_data(cluster_labels, centroids):
    """Print the results of the k-means clustering."""
    print("Result of k-Means Clustering:\n")
    for label in cluster_labels:
        print("Data point:", label[1])
        print("Cluster number:", label[0], "\n")
    print("Last centroids position:\n", centroids)

def create_centroids():
    """Create initial centroids."""
    return np.array([[5.0, 0.0], [45.0, 70.0], [50.0, 90.0]])

if __name__ == "__main__":
    filename = os.path.join(os.path.dirname(__file__), "data-example-1.csv")
    data_points = np.genfromtxt(filename, delimiter=",")
    centroids = create_centroids()
    total_iterations = 100
    
    cluster_labels, new_centroids = iterate_k_means(data_points, centroids, total_iterations)
    print_label_data(cluster_labels, new_centroids)
