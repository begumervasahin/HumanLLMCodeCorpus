# Second Task: Find the index of the element with the minimum in one list and maximum in another list

a = []
b = []
n = int(input())  # Number of elements

# Input elements into lists a and b
for i in range(n):
    x, y = map(int, input().split())
    a.append(x)
    b.append(y)

# Find the index of the element that is the minimum in a and the maximum in b
answer = -1
min_value, max_value = min(a), max(b)
for i in range(n):
    if a[i] == min_value and b[i] == max_value:
        answer = i + 1
        break
print(answer)

# Third Task: Mark elements based on comparison and print the final lists

n = int(input())
a = []
b = []

# Input elements into lists a and b
for i in range(n):
    x, y = map(int, input().split())
    a.append(x)
    b.append(y)

final_1 = [False] * n
final_2 = [False] * n
i = j = 0
chosen = 0

# Compare elements from a and b to mark them in final_1 and final_2
while chosen < n:
    if a[i] < b[j]:
        final_1[i] = True
        i += 1
    else:
        final_2[j] = True
        j += 1
    chosen += 1

# Print the results
for i in range(n):
    if final_1[i] or i < n // 2:
        print(1, end='')
    else:
        print(0, end='')
print()

for i in range(n):
    if final_2[i] or i < n // 2:
        print(1, end='')
    else:
        print(0, end='')

# Fourth Task: Count the number of ways to pick two different genres

n, m = map(int, input().split())  # Number of books and genres
a = list(map(int, input().split())) # Input list of genres for the books

count = [0] * (m + 1)

# Count occurrences of each genre
for genre in a:
    count[genre] += 1

ways = 0

# Calculate the number of ways to pick two different genres
for i in range(1, m):
    for j in range(i + 1, m + 1):
        ways += count[i] * count[j]

print(ways)
