class Queue:
    """A simple implementation of a queue with state tracking."""
    
    def __init__(self):
        self.list = []
        self.state = set()
        
    def enqueue(self, item):
        """Add item to the queue."""
        self.list.append(item)
        self.state.add(tuple(item.list))
        
    def dequeue(self):
        """Remove and return the first item from the queue (FIFO)."""
        if not self.is_empty():
            item = self.list.pop(0)
            self.state.remove(tuple(item.list))
            return item
        else:
            return None
        
    def is_empty(self):
        """Check if the queue is empty."""
        return len(self.list) == 0

# Example usage of the Queue class
if __name__ == "__main__":
    class PuzzleState:
        """Example PuzzleState class to be used with Queue."""
        def __init__(self, state_list):
            self.list = state_list
            
        def __repr__(self):
            return f"PuzzleState({self.list})"
    
    q = Queue()
    q.enqueue(PuzzleState([1, 2, 3, 4, 5, 6, 7, 8, 0]))
    q.enqueue(PuzzleState([1, 2, 3, 4, 5, 6, 7, 0, 8]))
    
    print(f"Queue is empty: {q.is_empty()}")
    
    dequeued_item = q.dequeue()
    print(f"Dequeued: {dequeued_item}")
    
    print(f"Queue is empty: {q.is_empty()}")
    
    dequeued_item = q.dequeue()
    print(f"Dequeued: {dequeued_item}")
    
    print(f"Queue is empty: {q.is_empty()}")
