def partition(arr, start, end):
    """
    Partition the array for the QuickSort algorithm.
    
    Args:
        arr (list): The list to be partitioned.
        start (int): The starting index of the segment to be partitioned.
        end (int): The ending index of the segment to be partitioned.
    
    Returns:
        int: The index of the pivot element after partitioning.
    """
    left = start
    right = end
    pivot = arr[start]
    
    while left < right:
        while left < right and arr[left] <= pivot:
            left += 1
        while arr[right] > pivot:
            right -= 1
        if left < right:
            arr[left], arr[right] = arr[right], arr[left]
    
    arr[start], arr[right] = arr[right], arr[start]
    return right

def quick_sort(arr, start, end):
    """
    Perform QuickSort on the given array.
    
    Args:
        arr (list): The list to be sorted.
        start (int): The starting index of the segment to be sorted.
        end (int): The ending index of the segment to be sorted.
    """
    if start < end:
        pivot_index = partition(arr, start, end)
        quick_sort(arr, start, pivot_index - 1)
        quick_sort(arr, pivot_index + 1, end)

def main():
    """
    Main function to run the QuickSort algorithm on user input.
    """
    arr = []
    print("Enter the size of the array:")
    size = int(input())
    print(f"Enter {size} elements:")

    for _ in range(size):
        num = int(input())
        arr.append(num)

    print("Sorted array:")
    quick_sort(arr, 0, len(arr) - 1)
    for num in arr:
        print(num)

if __name__ == "__main__":
    main()
