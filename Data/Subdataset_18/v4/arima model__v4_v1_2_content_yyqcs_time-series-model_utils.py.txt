import torch
import torch.nn as nn
from torch.nn import init

class YourModel(nn.Module):
    def __init__(self):
        super(YourModel, self).__init__()
        # Define a linear layer with 10 input features and 5 output features
        self.linear_layer = nn.Linear(in_features=10, out_features=5)

    def forward(self, x):
        # Implement the forward pass
        # Currently, it is just a placeholder and does nothing
        return self.linear_layer(x)

def initialize_weights(m):
    # Initialize the weights of linear layers using Xavier uniform distribution
    if isinstance(m, nn.Linear):
        init.xavier_uniform_(m.weight)

# Create an instance of YourModel
your_model = YourModel()

# Apply the weight initialization function to the model
your_model.apply(initialize_weights)

# Verify the initialized weights
for name, param in your_model.named_parameters():
    if 'weight' in name:
        print(f'Initialized {name} with shape {param.shape}')
