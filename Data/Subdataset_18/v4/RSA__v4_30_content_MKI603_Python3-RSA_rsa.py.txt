import math
from Crypto.Util import number
import os

def gcd(a, b):
    """Calculate the greatest common divisor of a and b."""
    while b:
        a, b = b, a % b
    return a

def extended_euclidean(a, b):
    """Perform the extended Euclidean algorithm to find the gcd and the coefficients of Bézout's identity."""
    if b == 0:
        return 1, 0, a
    else:
        x, y, gcd = extended_euclidean(b, a % b)
        x, y = y, x - (a // b) * y
        return x, y, gcd

def generate_keys(p, q):
    """Generate RSA keys given two primes p and q."""
    N = p * q
    phi_n = (p - 1) * (q - 1)

    e = 2
    while gcd(e, phi_n) != 1:
        e += 1

    d, _, _ = extended_euclidean(e, phi_n)
    d = d % phi_n
    if d < 0:
        d += phi_n

    return N, phi_n, e, d

def fast_exp_mod(base, exp, mod):
    """Perform fast modular exponentiation."""
    result = 1
    while exp > 0:
        if exp % 2 == 1:
            result = (result * base) % mod
        base = (base * base) % mod
        exp //= 2
    return result

def encrypt(message, e, N):
    """Encrypt a message using RSA encryption."""
    print("[*] Encrypting...")
    return fast_exp_mod(message, e, N)

def decrypt(ciphertext, d, N):
    """Decrypt a message using RSA encryption."""
    print("[*] Decrypting...")
    return fast_exp_mod(ciphertext, d, N)

def text_to_num(text):
    """Convert text to a numerical representation."""
    binary_string = ''.join(format(ord(char), '08b') for char in text)
    return int(binary_string, 2)

def num_to_text(num):
    """Convert a numerical representation back to text."""
    binary_string = bin(num)[2:]
    binary_string = binary_string.zfill((8 - len(binary_string) % 8) % 8 + len(binary_string))
    text = ''.join(chr(int(binary_string[i:i+8], 2)) for i in range(0, len(binary_string), 8))
    return text

def RSA(text):
    """Perform RSA encryption and decryption on the provided text."""
    m = text_to_num(text)
    print('[*] Numeric representation:', m)

    p = number.getPrime(256, os.urandom)
    print("[*] Generated prime p:", p)

    q = number.getPrime(256, os.urandom)
    print("[*] Generated prime q:", q)

    print("[*] Starting key generation...")
    N, phi_n, e, d = generate_keys(p, q)

    ciphertext = encrypt(m, e, N)
    decrypted_message = decrypt(ciphertext, d, N)
    result = num_to_text(decrypted_message)
    print("[*] Decrypted message:", result)

    RSA(message)
