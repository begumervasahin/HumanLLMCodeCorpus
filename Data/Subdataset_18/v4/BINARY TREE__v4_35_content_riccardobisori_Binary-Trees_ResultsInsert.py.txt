import pickle
import TestInsert
import matplotlib.pyplot as plt

# Define test values
elements = (10, 50, 1000, 10000, 50000)
range_elements = (30, 1500, 1000000000)
test_values = (elements, range_elements)

# Save test values to a file
with open("testInsert.p", "wb") as file:
    pickle.dump(test_values, file)

# Run the test insertion function
TestInsert.testInsert()

# Load the results from the test
with open("resultInsertABR.p", "rb") as file:
    result_insert_abr = pickle.load(file)

with open("resultInsertRB.p", "rb") as file:
    result_insert_rb = pickle.load(file)

# Plot the results for Binary Search Tree (ABR)
x = (30, 100, 500, 1000)
plt.plot(x, result_insert_abr[1])
plt.xlabel('Range of Elements')
plt.ylabel('Insertion Time of (max(range) + 1)')
plt.title("Insertion Times in a Binary Search Tree\n with Elements Previously Inserted in Ascending Order")
plt.show()

# Plot the results for Red-Black Tree (RB)
x = (30, 100, 500, 1000, 1500)
plt.plot(x, result_insert_rb[1])
plt.xlabel('Range of Elements')
plt.ylabel('Insertion Time of (max(range) + 1)')
plt.title("Insertion Times in a Red-Black Tree\n with Elements Previously Inserted in Ascending Order")
plt.show()
