import time
import numpy as np
import hashlib
import quicksort as qsort
import sys
import os
from random import randint

# Initialize QuickSort instance
qs = qsort.QuickSort()

# Global variables to store time and array information
timeArray = []
arr = []
chosenAlgorithm = ''
arrSize = 0
isUnique = 1
shuffle = 0
repeat = 0

# File paths and names
definitions = {
    'CSVFileName': 'randomNumDump.csv',
    'Output': 'output.csv',
    'TimeLog': 'timelog.csv'
}

def timed_sort_function(function):
    """
    Times the execution of the chosen sorting function and logs the results.

    Parameters:
        function (str): The name of the sorting function to use.
    """
    global timeArray
    global arr
    global repeat

    for i in range(int(repeat)):
        arr = setup_array()
        start_time = time.time()
        sort_func = choose_algorithm(function)
        arr = sort_func(arr)
        elapsed_time = (time.time() - start_time) * 1000  # Convert to milliseconds

        # Print SHA-512 hash of the sorted array and elapsed time
        print(hashlib.sha512(', '.join(map(str, arr)).encode('utf-8')).hexdigest())
        print(f'{i}. function [{sort_func.__name__}] finished in {elapsed_time:.2f} ms')

        timeArray.append(elapsed_time)

    # Calculate and log mean elapsed time
    mean_time = sum(timeArray) / len(timeArray)
    timeArray.append(f'Mean :: {mean_time:.2f}')
    np.savetxt(definitions['TimeLog'], timeArray, fmt='%s', delimiter=',')
    np.savetxt(definitions['Output'], arr, fmt='%d', delimiter=',')

def new_array_to_csv(size, is_unique, shuffle):
    """
    Generates a new random array and saves it to a CSV file.

    Parameters:
        size (int): The size of the array.
        is_unique (bool): Whether the array elements should be unique.
        shuffle (int): The level of shuffling to apply to the array.
    """
    arr = list(np.random.choice(size, size, replace=not is_unique))
    print('Array Size ::', len(arr))

    if shuffle == '0':
        arr.sort()
    elif shuffle == '1':
        arr.sort()
        arr = slightly_shuffle(arr)

    np.savetxt(definitions['CSVFileName'], arr, fmt='%d', delimiter=',')

def csv_to_array(path):
    """
    Loads an array from a CSV file.

    Parameters:
        path (str): The path to the CSV file.

    Returns:
        list: The loaded array.
    """
    return list(np.genfromtxt(path, delimiter=',', dtype=int))

def choose_algorithm(func):
    """
    Selects the appropriate sorting function based on the given name.

    Parameters:
        func (str): The name of the sorting function to use.

    Returns:
        function: The selected sorting function.
    """
    return {
        'QuickSort': qs.QuickSort,
        'SortMedianOfThree': qs.SortMedianOfThree,
        'OptimisedSort': qs.OptimisedSort
    }.get(func, qs.QuickSort)

def initial_setup():
    """
    Initializes global variables based on command line arguments or user input.
    """
    global chosenAlgorithm
    global arrSize
    global isUnique
    global shuffle
    global repeat

    if len(sys.argv) > 1:
        chosenAlgorithm = sys.argv[1]
        arrSize = sys.argv[2]
        isUnique = sys.argv[3]
        shuffle = sys.argv[4]
        repeat = sys.argv[5]
        definitions['CSVFileName'] = sys.argv[6]
        definitions['Output'] = sys.argv[7]
    else:
        chosenAlgorithm = input('Write the algorithm to be used: (QuickSort, SortMedianOfThree, OptimisedSort):\n')
        arrSize = input('Write the size of the array:\n')
        isUnique = input('All unique elements? (0/1)\n')
        shuffle = input('How shuffled should the list be? (0/1/2)\n')
        repeat = input('Repeat sort how many times?\n')
        definitions['CSVFileName'] = input('Write input file name:\n')
        definitions['Output'] = input('Write output file name:\n')

    # Ensure filenames have .csv extension
    if not definitions['CSVFileName'].endswith('.csv'):
        definitions['CSVFileName'] += '.csv'
    if not definitions['Output'].endswith('.csv'):
        definitions['Output'] += '.csv'

    # Set absolute paths
    base_path = os.path.abspath(os.path.dirname(sys.argv[0]))
    definitions['Output'] = os.path.join(base_path, definitions['Output'])
    if arrSize != '0':
        definitions['CSVFileName'] = os.path.join(base_path, definitions['CSVFileName'])

    return chosenAlgorithm

def setup_array():
    """
    Sets up the array based on the initial configuration.

    Returns:
        list: The configured array.
    """
    global arrSize
    global isUnique
    global shuffle

    if arrSize != '0':
        new_array_to_csv(int(arrSize), isUnique == '1', shuffle)
    return csv_to_array(definitions['CSVFileName'])

def slightly_shuffle(arr):
    """
    Slightly shuffles the array.

    Parameters:
        arr (list): The array to shuffle.

    Returns:
        list: The shuffled array.
    """
    for _ in range(len(arr) // 7):
        rand_no1 = randint(0, len(arr) - 1)
        rand_no2 = randint(0, len(arr) - 1)
        arr[rand_no1], arr[rand_no2] = arr[rand_no2], arr[rand_no1]
    return arr

if __name__ == "__main__":
    if len(sys.argv) > 1 and sys.argv[1] == "runtests":
        print('Running tests')
    else:
        initial_setup()
        timed_sort_function(chosenAlgorithm)
