import collections
import heapq


class PriorityQueue:
    """A priority queue implemented with a heap."""
    def __init__(self):
        self.elements = []

    def is_empty(self) -> bool:
        """Check if the priority queue is empty."""
        return not self.elements

    def put(self, item, priority):
        """Add an item with a given priority to the queue."""
        heapq.heappush(self.elements, (priority, item))

    def get(self):
        """Remove and return the item with the highest priority."""
        return heapq.heappop(self.elements)[1]

    def __len__(self) -> int:
        """Return the number of items in the queue."""
        return len(self.elements)


class Queue:
    """A first-in, first-out queue implemented with collections.deque."""
    def __init__(self):
        self.elements = collections.deque()

    def is_empty(self) -> bool:
        """Check if the queue is empty."""
        return not self.elements

    def put(self, item):
        """Add an item to the queue."""
        self.elements.append(item)

    def get(self):
        """Remove and return the item from the front of the queue."""
        return self.elements.popleft()

    def __len__(self) -> int:
        """Return the number of items in the queue."""
        return len(self.elements)


def bfs(start, end):
    """Perform BFS from the start tile to the end tile."""
    frontier = Queue()
    frontier.put(start)
    came_from = {start: None}

    while not frontier.is_empty():
        current = frontier.get()
        current.visit()

        if current == end:
            print("WE MADE IT!!!")
            break

        for next_tile in current.neighbors:
            if next_tile not in came_from:
                frontier.put(next_tile)
                came_from[next_tile] = current

    return came_from


class Tile:
    """Class representing a tile in a grid."""
    def __init__(self, name):
        self.name = name
        self.neighbors = []

    def add_neighbor(self, neighbor):
        """Add a neighboring tile."""
        self.neighbors.append(neighbor)

    def visit(self):
        """Visit the tile and print its name."""
        print(f"Visiting {self.name}")


def create_demo_tiles():
    """Create a simple grid of tiles for demonstration."""
    tile_A = Tile('A')
    tile_B = Tile('B')
    tile_C = Tile('C')
    tile_D = Tile('D')

    tile_A.add_neighbor(tile_B)
    tile_B.add_neighbor(tile_C)
    tile_C.add_neighbor(tile_D)

    return tile_A, tile_D


def print_path(came_from, end):
    """Print the path from the start to the end tile."""
    print("\nPath taken:")
    current = end
    while current is not None:
        print(current.name)
        current = came_from[current]


if __name__ == "__main__":
    start_tile, end_tile = create_demo_tiles()
    came_from = bfs(start_tile, end_tile)
    print_path(came_from, end_tile)
