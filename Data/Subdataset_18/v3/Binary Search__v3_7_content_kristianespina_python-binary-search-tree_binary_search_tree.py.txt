from random import randint

class Node:
    """Represents a node in a binary search tree."""
    def __init__(self, value):
        self.value = value
        self.left = None
        self.right = None

    def insert(self, value):
        """Inserts a new value into the subtree rooted at this node."""
        if value < self.value:
            if self.left is None:
                self.left = Node(value)
            else:
                self.left.insert(value)
        elif value > self.value:
            if self.right is None:
                self.right = Node(value)
            else:
                self.right.insert(value)

    def to_array(self, array=None):
        """
        Converts the subtree rooted at this node into an array using a
        depth-first search (preorder traversal).
        """
        if array is None:
            array = []
        array.append(self.value)
        if self.left:
            self.left.to_array(array)
        if self.right:
            self.right.to_array(array)
        return array

class BinaryTree:
    """Represents a binary search tree."""
    def __init__(self):
        self.root = None

    def insert(self, value):
        """Inserts a new value into the binary search tree."""
        if self.root is None:
            self.root = Node(value)
        else:
            self.root.insert(value)

    def find(self, value):
        """Searches for a value in the binary search tree."""
        return self._find_node(self.root, value)

    def _find_node(self, node, value):
        """Helper function to recursively search for a value in the tree."""
        if node is None:
            return False
        if value == node.value:
            return True
        elif value < node.value:
            return self._find_node(node.left, value)
        else:
            return self._find_node(node.right, value)

    def to_array(self):
        """Converts the entire binary search tree to an array."""
        return self.root.to_array() if self.root else []

    def rearrange(self):
        """
        Re-arranges the binary tree such that the median value becomes the root.
        Returns the binary tree in depth-first search (DFS) array form after rearrangement.
        """
        nodes = self.to_array()
        nodes.sort()
        
        # Rebuild the tree with the median as the root
        self.root = Node(nodes.pop(len(nodes) // 2))
        for value in nodes:
            self.insert(value)
        
        return self.to_array()

# Example usage
if __name__ == "__main__":
    tree = BinaryTree()
    
    # Insert 50 random integers between 0 and 100
    for _ in range(50):
        tree.insert(randint(0, 100))
    
    print("Tree in DFS Array Form:")
    print(tree.to_array())
    
    print("\nRe-arranged Tree with Median as Root:")
    print(tree.rearrange())
    
    # Search for a specific value in the tree
    value_to_find = 55
    print(f"\nSearching for {value_to_find} in the tree:")
    print("Found!" if tree.find(value_to_find) else "Not Found!")
