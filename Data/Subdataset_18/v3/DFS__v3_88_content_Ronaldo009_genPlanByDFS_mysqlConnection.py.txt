import pymysql.cursors

def create_connection(host, port, user, password, db):
    """
    Creates and returns a connection to a MySQL database.
    
    Args:
        host (str): The host address of the MySQL server.
        port (int): The port number of the MySQL server.
        user (str): The username to connect to the MySQL server.
        password (str): The password for the user.
        db (str): The database name to connect to.

    Returns:
        pymysql.connections.Connection: Connection object for the MySQL database.
    """
    return pymysql.connect(
        host=host,
        port=port,
        user=user,
        password=password,
        db=db,
        charset='utf8',
        cursorclass=pymysql.cursors.DictCursor
    )

def connect_and_print_status(connection_func, name):
    """
    Attempts to connect to the database using the provided function
    and prints the status of the connection.
    
    Args:
        connection_func (function): Function to create the database connection.
        name (str): Name of the database connection for status printing.
    
    Returns:
        pymysql.connections.Connection: Connection object if successful, otherwise None.
    """
    try:
        conn = connection_func()
        print(f"Successfully connected to the {name} database.")
        return conn
    except Exception as e:
        print(f"Failed to connect to the {name} database: {e}")
        return None

def close_connection(conn, name):
    """
    Closes the given database connection and prints a status message.
    
    Args:
        conn (pymysql.connections.Connection): The connection to close.
        name (str): Name of the database connection for status printing.
    """
    if conn:
        conn.close()
        print(f"Closed the connection to the {name} database.")
