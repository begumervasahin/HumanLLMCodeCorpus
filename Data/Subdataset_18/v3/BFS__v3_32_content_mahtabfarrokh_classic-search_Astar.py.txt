import heapq

class AStar:
    def __init__(self, initial_state, actions, result, goal_test, get_cost, heuristic):
        self.frontier = []  # Priority queue (min-heap) for frontier nodes
        self.explored = set()  # Set of explored nodes
        self.visited = set()   # Set of visited nodes
        self.initial_state = initial_state
        self.actions = actions
        self.result = result
        self.goal_test = goal_test
        self.get_cost = get_cost
        self.heuristic = heuristic
        self.max_memory = 0

    def search(self):
        while self.frontier:
            # Update maximum memory usage
            self.max_memory = max(self.max_memory, len(self.frontier) + len(self.explored))
            
            # Pop the node with the lowest cost from the priority queue
            current_f, current_g, current_node, current_path, current_actions = heapq.heappop(self.frontier)

            if self.goal_test(current_node):
                return current_actions, current_path, current_g

            self.explored.add(current_node)

            for action in self.actions(current_node):
                successor = self.result(current_node, action)
                if successor not in self.explored:
                    new_g = current_g + self.get_cost(current_node, successor)
                    new_h = self.heuristic(successor)
                    new_f = new_g + new_h
                    new_path = current_path + [successor]
                    new_actions = current_actions + [action]
                    
                    heapq.heappush(self.frontier, (new_f, new_g, successor, new_path, new_actions))
                    self.visited.add(successor)
        
        return None

    def search_astar(self):
        start_state = self.initial_state()
        start_h = self.heuristic(start_state)
        heapq.heappush(self.frontier, (start_h, 0, start_state, [start_state], []))
        
        result = self.search()
        
        if result:
            actions, path, cost = result
            print("Path found:")
            print("Actions:", actions)
            print("Number of visited nodes:", len(self.visited))
            print("Number of nodes in closed list:", len(self.explored))
            print("Max memory used:", self.max_memory)
            print("Total path cost:", cost)
        else:
            print("No path found.")

# Example usage for an 8-puzzle problem
def initial_state():
    return (1, 2, 3, 4, 5, 6, 7, 8, 0)  # Example initial state of 8-puzzle

def actions(state):
    zero_index = state.index(0)
    possible_actions = []
    if zero_index % 3 > 0:  # Can move left
        possible_actions.append('Left')
    if zero_index % 3 < 2:  # Can move right
        possible_actions.append('Right')
    if zero_index > 2:  # Can move up
        possible_actions.append('Up')
    if zero_index < 6:  # Can move down
        possible_actions.append('Down')
    return possible_actions

def result(state, action):
    zero_index = state.index(0)
    new_state = list(state)
    if action == 'Left':
        new_state[zero_index], new_state[zero_index - 1] = new_state[zero_index - 1], new_state[zero_index]
    elif action == 'Right':
        new_state[zero_index], new_state[zero_index + 1] = new_state[zero_index + 1], new_state[zero_index]
    elif action == 'Up':
        new_state[zero_index], new_state[zero_index - 3] = new_state[zero_index - 3], new_state[zero_index]
    elif action == 'Down':
        new_state[zero_index], new_state[zero_index + 3] = new_state[zero_index + 3], new_state[zero_index]
    return tuple(new_state)

def goal_test(state):
    return state == (0, 1, 2, 3, 4, 5, 6, 7, 8)

def get_cost(state1, state2):
    return 1  # Uniform cost for each move

def heuristic(state):
    goal = (0, 1, 2, 3, 4, 5, 6, 7, 8)
    return sum(abs(s % 3 - g % 3) + abs(s // 3 - g // 3) for s, g in zip(state, goal))

if __name__ == "__main__":
    astar_solver = AStar(initial_state, actions, result, goal_test, get_cost, heuristic)
    astar_solver.search_astar()
