import unittest

def fizzbuzz(list1, list2):
    """
    Calculate the FizzBuzz result based on the combined length of two lists.

    Parameters:
    list1 (list): The first list of elements.
    list2 (list): The second list of elements.

    Returns:
    str: 'Fizz' if the combined length is divisible by 3.
         'Buzz' if the combined length is divisible by 5.
         'FizzBuzz' if the combined length is divisible by both 3 and 5.
         'Invalid input' if either input is not a list.
         Otherwise, return the combined length as a string.
    """
    if not isinstance(list1, list) or not isinstance(list2, list):
        return "Invalid input"
    
    total_length = len(list1) + len(list2)

    if total_length % 3 == 0 and total_length % 5 == 0:
        return 'FizzBuzz'
    elif total_length % 3 == 0:
        return 'Fizz'
    elif total_length % 5 == 0:
        return 'Buzz'
    else:
        return str(total_length)

class TestFizzBuzz(unittest.TestCase):
    def test_non_list_inputs(self):
        self.assertEqual(fizzbuzz(5, 8), "Invalid input")
    
    def test_one_non_list_input(self):
        self.assertEqual(fizzbuzz([2, 3, 4], 7), "Invalid input")
    
    def test_fizz(self):
        self.assertEqual(fizzbuzz([1, 3, 4], ['a', 'b', 'c']), "Fizz")
    
    def test_buzz(self):
        self.assertEqual(fizzbuzz([4, 5, 4], [3, 4]), "Buzz")
    
    def test_fizzbuzz(self):
        self.assertEqual(fizzbuzz([1, 2, 3, 3, 4, 5, 5, 5, 5, 6], [7, 7, 4, 5, 9]), "FizzBuzz")
    
    def test_combined_length(self):
        self.assertEqual(fizzbuzz([4, 5, 6], [3]), '4')
    
    def test_empty_second_list(self):
        self.assertEqual(fizzbuzz([1, 3, 5], []), "Fizz")

if __name__ == '__main__':
    unittest.main()
