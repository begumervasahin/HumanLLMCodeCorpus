import sys
import argparse
import importlib

# Mapping of algorithm names to module names
ALGORITHM_MODULES = {
    'bubble': 'bubbleSort',
    'selection': 'selectionSort',
    'merge': 'mergeSort',
    'quick': 'quickSort'
}

def display_list(lst):
    """
    Displays the list in a formatted manner.
    
    Args:
        lst (list): The list of integers to be displayed.
    """
    formatted_list = '| ' + ' | '.join(map(str, lst)) + ' |'
    print(formatted_list if lst else '| |')
    input("Press Enter to continue...")

def parse_arguments():
    """
    Parses command-line arguments.
    
    Returns:
        argparse.Namespace: Parsed command-line arguments.
    """
    parser = argparse.ArgumentParser(description="Sort a list of integers using the specified algorithm.")
    parser.add_argument("input_list", nargs='+', type=int, help="List of integers separated by space")
    parser.add_argument("alg", type=str, choices=ALGORITHM_MODULES.keys(), help="Sorting algorithm: bubble, selection, merge, or quick")
    parser.add_argument("-t", "--trace", action="store_true", help="Print the sorting algorithm step-by-step")
    return parser.parse_args()

def main(argv):
    """
    Main function to parse arguments and execute the chosen sorting algorithm.
    
    Args:
        argv (list): List of command-line arguments.
    """
    args = parse_arguments()

    print('Input:', args.input_list)

    # Dynamically import the module and get the sorter class
    module_name = ALGORITHM_MODULES[args.alg]
    module = importlib.import_module(module_name)
    sorter_class = getattr(module, module_name)

    # Create an instance of the sorter class
    sorter = sorter_class(args.input_list, args.trace)
    result = sorter.sort()

    print('Result:', result)

if __name__ == '__main__':
    main(sys.argv[1:])
