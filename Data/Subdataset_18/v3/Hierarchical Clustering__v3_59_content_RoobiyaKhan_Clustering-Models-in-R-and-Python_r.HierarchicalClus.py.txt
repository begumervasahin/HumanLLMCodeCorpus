import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
import scipy.cluster.hierarchy as sch
from sklearn.cluster import AgglomerativeClustering

def load_data(filepath):
    """
    Load the dataset from a CSV file and extract relevant features.

    Parameters:
    - filepath (str): Path to the CSV file.

    Returns:
    - ndarray: The feature matrix containing Annual Income and Spending Score.
    """
    dataset = pd.read_csv(filepath)
    X = dataset.iloc[:, [3, 4]].values  # Extract Annual Income and Spending Score
    return X

def plot_dendrogram(X):
    """
    Plot a dendrogram to help determine the optimal number of clusters.

    Parameters:
    - X (ndarray): The feature matrix.
    """
    plt.figure(figsize=(10, 7))
    sch.dendrogram(sch.linkage(X, method='ward'))
    plt.title('Dendrogram')
    plt.xlabel('Customers')
    plt.ylabel('Euclidean Distances')
    plt.show()

def perform_agglomerative_clustering(X, n_clusters=5):
    """
    Apply Agglomerative Clustering to the dataset.

    Parameters:
    - X (ndarray): The feature matrix.
    - n_clusters (int): The number of clusters to form.

    Returns:
    - ndarray: Cluster labels for each data point.
    """
    clustering_model = AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage='ward')
    y_hc = clustering_model.fit_predict(X)
    return y_hc

def plot_clusters(X, cluster_labels):
    """
    Visualize the clusters formed by Agglomerative Clustering.

    Parameters:
    - X (ndarray): The feature matrix.
    - cluster_labels (ndarray): The cluster labels for each data point.
    """
    plt.figure(figsize=(10, 7))
    colors = ['red', 'blue', 'green', 'cyan', 'magenta']
    labels = ['Careful', 'Standard', 'Target', 'Careless', 'Sensible']

    for i, color in enumerate(colors):
        plt.scatter(X[cluster_labels == i, 0], X[cluster_labels == i, 1], 
                    s=100, c=color, label=labels[i])

    plt.title('Clusters of Customers')
    plt.xlabel('Annual Income (k$)')
    plt.ylabel('Spending Score (1-100)')
    plt.legend()
    plt.show()

def main():
    # File path to the dataset
    filepath = 'Mall_Customers.csv'
    
    # Load the dataset
    X = load_data(filepath)

    # Plot the dendrogram to find the optimal number of clusters
    plot_dendrogram(X)

    # Perform Agglomerative Clustering
    cluster_labels = perform_agglomerative_clustering(X)

    # Visualize the clusters
    plot_clusters(X, cluster_labels)

if __name__ == "__main__":
    main()
