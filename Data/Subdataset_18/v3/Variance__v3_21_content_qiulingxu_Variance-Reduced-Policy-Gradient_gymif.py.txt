import gym
import copy
from VRRL import VRRL, montocarloevaluation, policy_neural_net, sgd
from log import log

class Agent:
    def __init__(self, env_name):
        self.env = gym.make(env_name)
        self.curr_state = None
        self.step = 0
        self.done = False

    def start_exp(self):
        self.curr_state = self.env.reset()
        self.step = 0
        self.done = False
        return self.get_current_state()

    def get_current_state(self):
        return copy.deepcopy(self.curr_state)

    def perform_action(self, action):
        observation, reward, done, info = self.env.step(action)
        self.curr_state = observation
        self.step += 1
        self.done = done or self.step >= self.env.spec.max_episode_steps
        return self.get_current_state(), float(reward)

    def is_experiment_done(self):
        return self.done

def train_vrrl_for_env(env_name):
    env = gym.make(env_name)
    input_layer_size = env.observation_space.shape[0]
    output_layer_size = env.action_space.n
    batch_size = 64

    network_structure = [input_layer_size, 8, 8, output_layer_size]
    log_path = log(env_name)
    policy_net = policy_neural_net(network_structure, batch_size)

    for baseline in [True]:
        for vareduce in [True, False]:
            monte_carlo_eval = montocarloevaluation(log_path, batch_size=batch_size, baseline=baseline, vareduce=vareduce)
            optimizer = sgd(0.2)
            agent = Agent(env_name)
            vrrl = VRRL(policy_net, monte_carlo_eval, batch_size, agent, optimizer)
            vrrl.train(2000)

if __name__ == "__main__":
    experiments = ['CartPole-v0', 'Acrobot-v1', 'MountainCar-v0', 'Pendulum-v0']
    for exp_name in experiments:
        train_vrrl_for_env(exp_name)
