def recursive_fibo(num):
    """
    Calculate the Fibonacci value at a given position.

    Args:
        num (int): The position in the Fibonacci sequence.

    Returns:
        int: The Fibonacci value at the given position.
    """
    if num <= 1:
        return num
    return recursive_fibo(num - 2) + recursive_fibo(num - 1)

def get_positive_integer(prompt):
    """
    Prompt the user to enter a positive integer.

    Args:
        prompt (str): The prompt message to display.

    Returns:
        int: The positive integer entered by the user.
    """
    while True:
        try:
            value = int(input(prompt))
            if value > 0:
                return value
            else:
                print("Please enter a positive integer.")
        except ValueError:
            print("Please only input an integer.")

def print_fibonacci_sequence(term):
    """
    Print the Fibonacci sequence up to the given number of terms.

    Args:
        term (int): The number of terms to print.
    """
    print("\nFibonacci sequence:\n")
    for num in range(term + 1):
        print(recursive_fibo(num))

def main():
    """
    Main function to run the Fibonacci sequence program.
    """
    term = get_positive_integer('Up to how many terms to return Fibonacci?: ')
    print_fibonacci_sequence(term)

if __name__ == "__main__":
    main()
