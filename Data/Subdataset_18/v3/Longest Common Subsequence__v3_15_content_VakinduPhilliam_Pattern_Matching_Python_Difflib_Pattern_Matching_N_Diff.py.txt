import difflib

def compare_strings(str1, str2):
    """
    Compares two strings line by line and prints the differences.

    Parameters:
    - str1: First string to be compared.
    - str2: Second string to be compared.
    """
    # Split the strings into lines, keeping the end-of-line characters
    lines1 = str1.splitlines(keepends=True)
    lines2 = str2.splitlines(keepends=True)

    # Create a Differ object to compute the differences
    differ = difflib.Differ()
    diff = differ.ndiff(lines1, lines2)

    # Print the computed differences
    print(''.join(diff), end="")

# Define the two strings to be compared
string1 = 'one\ntwo\nthree\n'
string2 = 'ore\ntree\nemu\n'

# Compare the strings and print the differences
compare_strings(string1, string2)
