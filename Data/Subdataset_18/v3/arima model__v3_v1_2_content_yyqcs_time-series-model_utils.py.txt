import torch
import torch.nn as nn
from torch.nn import init

class YourModel(nn.Module):
    def __init__(self):
        super(YourModel, self).__init__()
        self.linear_layer = nn.Linear(in_features=10, out_features=5)  # Define the linear layer with specified input and output features

    def forward(self, x):
        # Define the forward pass through the linear layer
        return self.linear_layer(x)

def initialize_weights(m):
    # Initialize weights of linear layers using Xavier uniform initialization
    if isinstance(m, nn.Linear):
        init.xavier_uniform_(m.weight.data)

def main():
    # Instantiate the model
    model = YourModel()

    # Apply the weight initialization function to the model
    model.apply(initialize_weights)

    # Print the initialized weights for verification
    for name, param in model.named_parameters():
        if 'weight' in name:
            print(f'Initialized {name} with shape {param.shape}')

    # Create a dummy input tensor with a batch size of 2 and 10 features
    dummy_input = torch.randn(2, 10)
    
    # Perform a forward pass with the dummy input
    output = model(dummy_input)
    
    # Print the output of the model
    print('Output:', output)

if __name__ == "__main__":
    main()
