from __future__ import print_function, division
from future.utils import iteritems
from builtins import range
import nltk
import random
from bs4 import BeautifulSoup

# Function to read and parse positive reviews
def read_positive_reviews(filepath):
    """Reads and parses the positive reviews from the specified file."""
    with open(filepath, 'r', encoding='utf-8') as file:
        return BeautifulSoup(file.read(), 'html.parser').findAll('review_text')

# Function to construct trigrams from tokenized reviews
def construct_trigrams(reviews):
    """Constructs trigrams from the given reviews."""
    trigrams = {}
    for review in reviews:
        tokens = nltk.tokenize.word_tokenize(review.text.lower())
        for i in range(len(tokens) - 2):
            k = (tokens[i], tokens[i+2])
            if k not in trigrams:
                trigrams[k] = []
            trigrams[k].append(tokens[i+1])
    return trigrams

# Function to calculate trigram probabilities
def calculate_trigram_probabilities(trigrams):
    """Calculates the probabilities for each trigram."""
    trigram_probabilities = {}
    for k, words in iteritems(trigrams):
        if len(set(words)) > 1:
            word_count = {}
            total_count = 0
            for w in words:
                if w not in word_count:
                    word_count[w] = 0
                word_count[w] += 1
                total_count += 1
            word_probabilities = {w: float(c) / total_count for w, c in iteritems(word_count)}
            trigram_probabilities[k] = word_probabilities
    return trigram_probabilities

# Function to randomly sample from a dictionary of probabilities
def random_sample(d):
    """Randomly samples a word based on the given probabilities."""
    r = random.random()
    cumulative = 0
    for w, p in iteritems(d):
        cumulative += p
        if r < cumulative:
            return w

# Function to test the spinner
def test_spinner(reviews, trigram_probabilities):
    """Tests the spinner by replacing words in a random review based on trigram probabilities."""
    review = random.choice(reviews)
    original_text = review.text.lower()
    print("Original:", original_text)
    tokens = nltk.tokenize.word_tokenize(original_text)
    for i in range(len(tokens) - 2):
        if random.random() < 0.2:
            k = (tokens[i], tokens[i+2])
            if k in trigram_probabilities:
                tokens[i+1] = random_sample(trigram_probabilities[k])
    spun_text = " ".join(tokens).replace(" .", ".").replace(" '", "'").replace(" ,", ",").replace("$ ", "$").replace(" !", "!")
    print("Spun:")
    print(spun_text)

if __name__ == '__main__':
    # Filepath to the positive reviews
    filepath = 'electronics/positive.review'
    
    # Reading and parsing the positive reviews
    positive_reviews = read_positive_reviews(filepath)
    
    # Constructing trigrams from the reviews
    trigrams = construct_trigrams(positive_reviews)
    
    # Calculating trigram probabilities
    trigram_probabilities = calculate_trigram_probabilities(trigrams)
    
    # Testing the spinner
    test_spinner(positive_reviews, trigram_probabilities)
