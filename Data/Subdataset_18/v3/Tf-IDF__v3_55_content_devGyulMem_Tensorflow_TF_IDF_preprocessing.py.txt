import pickle
import os
import utils.parameter as parameter
import utils.process_data as process_data

def preprocessing():
    """
    Main preprocessing function that sets parameters and datasets,
    then prints the results.
    """
    print("  >> It will take about one minute. Please wait.......")
    
    parameters = parameter_setting()
    res1, res2 = dataset_setting(parameters)
    
    print(f"  >> Result1: {res1}")
    print(f"  >> Result2: {res2}")

def parameter_setting():
    """
    Loads parameters and configuration, saves the parameters to a pickle file,
    and returns the parameters.
    """
    parameters, _ = parameter.load_parameter()
    save_to_pickle('parameters.bin', parameters)
    return parameters

def dataset_setting(parameters):
    """
    Processes the dataset using the given parameters, saves the data information
    to a pickle file, and returns results based on the processed data.
    """
    data_info = process_data.Data(parameters)
    data_info.data_preprocessing()
    save_to_pickle("data_info.bin", data_info)
    
    res1 = len(data_info.train_data)
    res2 = data_info.check_flag
    
    return res1, res2

def save_to_pickle(filename, data):
    """
    Saves the given data object to a pickle file in the 'Pickle' directory.
    """
    os.makedirs('Pickle', exist_ok=True)  # Ensure the directory exists
    
    file_path = os.path.join('Pickle', filename)
    with open(file_path, "wb") as file:
        pickle.dump(data, file)

if __name__ == '__main__':
    preprocessing()
