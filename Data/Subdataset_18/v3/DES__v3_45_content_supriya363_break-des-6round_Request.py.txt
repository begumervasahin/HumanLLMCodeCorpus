import requests
import json
import warnings

# Suppress warnings about unverified HTTPS requests
warnings.filterwarnings('ignore')

# Define constants
URL = "https://172.27.26.163:9999/des"
HEADERS = {
    'Content-type': 'application/json',
    'Origin': 'https://172.27.26.163:9999',
    'Referer': 'https://172.27.26.163:9999/game/caves.swf'
}
DATA_TEMPLATE = {
    "password": "555acfd4d3c1b7a398d71e4c1538ad4d",
    "teamname": "Amalgam",
    "plaintext": ""
}

def encrypt_and_write(plaintext, outfile, count):
    """Encrypt the plaintext and write the ciphertext to the output file."""
    DATA_TEMPLATE["plaintext"] = plaintext
    response = requests.post(URL, json=DATA_TEMPLATE, headers=HEADERS, verify=False)

    if response.status_code == 200:
        response_json = response.json()
        ciphertext = response_json.get("ciphertext")
        if ciphertext:
            print(f"Processed line {count}")
            outfile.write(f"{ciphertext}\n")
        else:
            print(f"No ciphertext found in the response for line {count}")
    else:
        print(f"Request failed for line {count} with status code {response.status_code}")

def process_lines(input_file, output_file):
    """
    Read lines from the input file, make POST requests to the URL, 
    and write the received ciphertexts to the output file.
    """
    with open(input_file, 'r') as infile, open(output_file, 'w') as outfile:
        for count, line in enumerate(infile, start=1):
            plaintext = line.strip().split(',')[0]
            encrypt_and_write(plaintext, outfile, count)

def main():
    """Main function to process the input file and write the ciphertexts to the output file."""
    input_file = 'input.txt'
    output_file = 'response.txt'
    process_lines(input_file, output_file)

if __name__ == "__main__":
    main()
