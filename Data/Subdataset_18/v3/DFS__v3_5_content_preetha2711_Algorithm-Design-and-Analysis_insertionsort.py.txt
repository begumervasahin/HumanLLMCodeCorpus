def insertion_sort(array):
    """
    Sort an array using the insertion sort algorithm.

    Args:
        array (list): The list of elements to be sorted.
    """
    for i in range(1, len(array)):
        value = array[i]
        hole = i
        # Shift elements of the sorted segment forward to find the correct position for the current element
        while hole > 0 and array[hole - 1] > value:
            array[hole] = array[hole - 1]
            hole -= 1
        # Place the current element in its correct position
        array[hole] = value

def main():
    """
    Main function to demonstrate the insertion sort algorithm.
    """
    # Define the array to be sorted
    arr = [12, 11, 4, 3, 1, 1, 6, 7]

    # Sort the array using insertion sort
    insertion_sort(arr)

    # Print the sorted array
    print("Sorted array:", arr)

if __name__ == "__main__":
    main()
