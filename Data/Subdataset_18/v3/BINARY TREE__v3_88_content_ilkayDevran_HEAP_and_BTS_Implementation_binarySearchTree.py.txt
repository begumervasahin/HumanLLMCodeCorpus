from random import randint

class Node:
    def __init__(self, key):
        self.key = key
        self.left = None
        self.right = None
        self.parent = None

class Tree:
    def __init__(self):
        self.root = None

    def inorder_tree_walk(self, node):
        """Perform an in-order traversal of the tree."""
        if node:
            self.inorder_tree_walk(node.left)
            print(node.key, end=' ')
            self.inorder_tree_walk(node.right)

    def tree_search(self, key, node=None):
        """Search for a node with a specific key."""
        if node is None:
            node = self.root
        if node is None or node.key == key:
            return node
        if key < node.key:
            return self.tree_search(key, node.left)
        return self.tree_search(key, node.right)

    def tree_minimum(self, node=None):
        """Find the node with the minimum key."""
        if node is None:
            node = self.root
        while node.left:
            node = node.left
        return node

    def tree_maximum(self, node=None):
        """Find the node with the maximum key."""
        if node is None:
            node = self.root
        while node.right:
            node = node.right
        return node

    def tree_successor(self, key):
        """Find the successor of the node with the given key."""
        node = self.tree_search(key)
        if node.right:
            return self.tree_minimum(node.right)
        parent_node = node.parent
        while parent_node and node == parent_node.right:
            node = parent_node
            parent_node = parent_node.parent
        return parent_node

    def tree_predecessor(self, key):
        """Find the predecessor of the node with the given key."""
        node = self.tree_search(key)
        if node.left:
            return self.tree_maximum(node.left)
        parent_node = node.parent
        while parent_node and node == parent_node.left:
            node = parent_node
            parent_node = parent_node.parent
        return parent_node

    def tree_insert(self, key, node=None):
        """Insert a new node with the given key."""
        if self.root is None:
            self.root = Node(key)
            return
        if node is None:
            node = self.root
        if key <= node.key:
            if node.left is None:
                node.left = Node(key)
                node.left.parent = node
            else:
                self.tree_insert(key, node.left)
        else:
            if node.right is None:
                node.right = Node(key)
                node.right.parent = node
            else:
                self.tree_insert(key, node.right)

    def tree_delete(self, key):
        """Delete the node with the given key."""
        node = self.tree_search(key)
        if node is None:
            return None
        parent_node = node.parent
        if node.left is None and node.right is None:
            if node == self.root:
                self.root = None
            elif node == parent_node.left:
                parent_node.left = None
            else:
                parent_node.right = None
        elif node.left is None:
            if node == self.root:
                self.root = node.right
            elif node == parent_node.left:
                parent_node.left = node.right
            else:
                parent_node.right = node.right
            node.right.parent = parent_node
        elif node.right is None:
            if node == self.root:
                self.root = node.left
            elif node == parent_node.left:
                parent_node.left = node.left
            else:
                parent_node.right = node.left
            node.left.parent = parent_node
        else:
            successor = self.tree_minimum(node.right)
            if successor.parent != node:
                self.tree_delete(successor.key)
                successor.right = node.right
                successor.right.parent = successor
            if node == self.root:
                self.root = successor
            elif node == parent_node.left:
                parent_node.left = successor
            else:
                parent_node.right = successor
            successor.left = node.left
            successor.left.parent = successor
            successor.parent = parent_node

    def tree_max_path_length(self, node):
        """Compute the maximum path length from the given node."""
        if node is None:
            return 0
        return 1 + max(self.tree_max_path_length(node.left), self.tree_max_path_length(node.right))

    def tree_min_path_length(self, node):
        """Compute the minimum path length from the given node."""
        if node is None:
            return 0
        return 1 + min(self.tree_min_path_length(node.left), self.tree_min_path_length(node.right))

    def tree_ratio_length(self):
        """Compute the ratio of minimum to maximum path length of the tree."""
        return self.tree_min_path_length(self.root) / self.tree_max_path_length(self.root)

def create_bst(lst):
    """Create a binary search tree and display various tree properties."""
    tree = Tree()
    for key in lst:
        tree.tree_insert(key)
    
    print("\n---TREE INFO---\n")
    print("INORDER WALK: ", end='')
    tree.inorder_tree_walk(tree.root)
    print("\nSEARCH for", lst[0], "-->", "Key EXISTS!" if tree.tree_search(lst[0]) else "Key does NOT exist!")
    print("SEARCH for", lst[5], "-->", "Key EXISTS!" if tree.tree_search(lst[5]) else "Key does NOT exist!")
    print("SEARCH for", lst[0] * 2, "-->", "Key EXISTS!" if tree.tree_search(lst[0] * 2) else "Key does NOT exist!")
    print("Minimum key in the Tree:", tree.tree_minimum().key)
    print("Maximum key in the Tree:", tree.tree_maximum().key)
    print("Successor of", lst[4], "is", tree.tree_successor(lst[4]).key)
    print("Predecessor of", lst[4], "is", tree.tree_predecessor(lst[4]).key)
    print("Delete root", lst[0], ":", end=' ')
    tree.tree_delete(lst[0])
    tree.inorder_tree_walk(tree.root)
    print("\nHeight of the Tree:", tree.tree_max_path_length(tree.root))
    print("Depth of the Tree:", tree.tree_min_path_length(tree.root))
    print("Ratio Depth/Height:", tree.tree_ratio_length())
    print("\n")

def generate_random_array(length, rng):
    """Generate a random array of the specified length within the given range."""
    return [randint(0, rng) for _ in range(length)]

def main():
    """Main function to create and display binary search trees."""
    create_bst(generate_random_array(30, 100))
    create_bst(generate_random_array(50, 1000))
    create_bst(generate_random_array(100, 500))

if __name__ == '__main__':
    main()
