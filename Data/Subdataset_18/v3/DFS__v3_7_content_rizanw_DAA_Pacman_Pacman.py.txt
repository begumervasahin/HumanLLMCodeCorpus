from Util.Algorithms import Algorithm
from Map.map import Map

def print_solution(dfs_result, bfs_result):
    """
    Display the results of the DFS and BFS algorithms.

    Args:
        dfs_result (list): The path found by the DFS algorithm.
        bfs_result (list): The path found by the BFS algorithm.
    """
    print("Solution:")
    print(f"  DFS: {dfs_result}")
    print(f"  BFS: {bfs_result}")

def execute_algorithm(map_choice, start, goal):
    """
    Execute DFS and BFS algorithms based on the chosen map and positions.

    Args:
        map_choice (str): The chosen map identifier ('1' or '2').
        start (tuple): The starting position (x, y).
        goal (tuple): The goal position (x, y).
    
    Returns:
        tuple: Results from DFS and BFS algorithms.
    """
    if map_choice == '1':
        graph = Map.graph1
    elif map_choice == '2':
        graph = Map.graph
    else:
        raise ValueError("Invalid map choice")

    dfs_result = Algorithm.dfs(graph, start, goal)
    bfs_result = Algorithm.bfs(graph, start, goal)
    return dfs_result, bfs_result

def main():
    """
    Main function to allow the user to select a map, execute algorithms, and display results.
    """
    print("== Pacman Solver ==")
    print("Select a map:")
    print(" 1. Simple 3x4 Map")
    print(" 2. Hard 6x6 Map")

    choice = input("Enter your choice (1 or 2): ").strip()
    
    if choice == '1':
        print("== Simple 3x4 Map selected ==")
        start_position = (1, 1)
        goal_position = (2, 3)
    elif choice == '2':
        print("== Hard 6x6 Map selected ==")
        start_position = (1, 1)
        goal_position = (6, 6)
    else:
        print("Invalid choice. Please select either '1' or '2'.")
        return

    print(f" - Start position: {start_position}")
    print(f" - Dot objective: {goal_position}")

    # Execute algorithms and get results
    dfs_result, bfs_result = execute_algorithm(choice, start_position, goal_position)
    
    # Print the results
    print_solution(dfs_result, bfs_result)

if __name__ == '__main__':
    main()
