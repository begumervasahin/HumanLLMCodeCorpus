import random
from collections import defaultdict

def create_grid_and_edges(size=9):
    """Creates a grid of nodes and generates edges between neighbors."""
    edges = []
    tree = defaultdict(list)
    node_id = 0

    # Populate the grid and create edges between adjacent nodes
    for x in range(size):
        for y in range(size):
            tree[node_id].append((x, y))
            node_id += 1

            # Generate edges to adjacent nodes, excluding diagonals
            for dx in range(-1, 2):
                for dy in range(-1, 2):
                    if abs(dx) == abs(dy):  # Skip diagonal connections
                        continue
                    
                    neighbor = (x + dx, y + dy)
                    if 0 <= neighbor[0] < size and 0 <= neighbor[1] < size:
                        edge = sorted([(x, y), neighbor])  # Ensure edges are in a consistent order
                        if edge not in edges:
                            edges.append(edge)
    
    return tree, edges

def find_trees_containing_edge(tree, edge):
    """Finds the tree IDs containing the two nodes of the edge."""
    tree1, tree2 = None, None
    for tree_id, nodes in tree.items():
        if edge[0] in nodes:
            tree1 = tree_id
        if edge[1] in nodes:
            tree2 = tree_id
        if tree1 is not None and tree2 is not None:
            break
    return tree1, tree2

def merge_trees(tree, tree1_id, tree2_id):
    """Merges two trees into one and removes the second tree."""
    tree[tree1_id].extend(tree[tree2_id])
    tree.pop(tree2_id)

def main():
    # Initialize the grid and edges
    tree, edges = create_grid_and_edges(size=9)

    # Randomly connect nodes until only one tree remains
    while len(tree) > 1:
        random_index = random.randint(0, len(edges) - 1)
        selected_edge = edges[random_index]
        
        # Identify the trees containing the selected edge's endpoints
        tree1_id, tree2_id = find_trees_containing_edge(tree, selected_edge)

        # Skip if the selected edge connects nodes within the same tree
        if tree1_id == tree2_id:
            continue

        # Merge the two trees
        print(f"Connecting edge: {selected_edge}")
        print(f"Before merging, tree {tree1_id}: {tree[tree1_id]}")
        print(f"Before merging, tree {tree2_id}: {tree[tree2_id]}")
        
        merge_trees(tree, tree1_id, tree2_id)

        # Remove the selected edge from the list
        edges.pop(random_index)
        
        # Display the current state of the trees and remaining edges
        print(f"Remaining edges: {len(edges)}, Remaining trees: {len(tree)}")
        print()

    # Display the final tree structure
    for tree_id, nodes in tree.items():
        print(f"Tree {tree_id}: {nodes}")

if __name__ == "__main__":
    main()
