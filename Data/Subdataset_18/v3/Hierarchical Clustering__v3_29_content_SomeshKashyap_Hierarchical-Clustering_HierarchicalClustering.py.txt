import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import scipy.cluster.hierarchy as sch
from sklearn.cluster import AgglomerativeClustering

def load_dataset(filename):
    """
    Loads the dataset from a CSV file and returns it as a pandas DataFrame.

    Parameters:
    filename (str): The path to the CSV file.

    Returns:
    pd.DataFrame: The loaded dataset.
    """
    return pd.read_csv(filename)

def plot_dendrogram(data):
    """
    Plots a dendrogram to help determine the optimal number of clusters.

    Parameters:
    data (ndarray): The data to be clustered.
    """
    plt.figure(figsize=(10, 7))
    dendrogram = sch.dendrogram(sch.linkage(data, method='ward'))
    plt.title('Dendrogram')
    plt.xlabel('Customers')
    plt.ylabel('Euclidean Distance')
    plt.show()

def perform_agglomerative_clustering(data, n_clusters=5):
    """
    Performs agglomerative clustering on the dataset.

    Parameters:
    data (ndarray): The data to be clustered.
    n_clusters (int): The number of clusters to form.

    Returns:
    ndarray: Cluster labels for each data point.
    """
    clustering_model = AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage='ward')
    return clustering_model.fit_predict(data)

def plot_clusters(data, cluster_labels):
    """
    Plots the data points color-coded by their cluster labels.

    Parameters:
    data (ndarray): The data points.
    cluster_labels (ndarray): The cluster labels for each data point.
    """
    plt.figure(figsize=(10, 7))
    colors = ['red', 'blue', 'green', 'cyan', 'magenta']
    labels = ['Careful', 'Standard', 'Target', 'Careless', 'Sensible']
    
    for i in range(len(labels)):
        plt.scatter(data[cluster_labels == i, 0], data[cluster_labels == i, 1], 
                    s=100, c=colors[i], label=labels[i])
    
    plt.title('Clusters of Clients')
    plt.xlabel('Annual Income (K$)')
    plt.ylabel('Spending Score (1-100)')
    plt.legend()
    plt.show()

def main():
    """
    Main function to load the dataset, perform clustering, and plot the results.
    """
    # Load the dataset
    dataset = load_dataset('Mall_Customers.csv')
    data = dataset.iloc[:, [3, 4]].values  # Select Annual Income and Spending Score columns

    # Plot the dendrogram to determine the optimal number of clusters
    plot_dendrogram(data)

    # Perform agglomerative clustering
    cluster_labels = perform_agglomerative_clustering(data, n_clusters=5)

    # Plot the clusters
    plot_clusters(data, cluster_labels)

if __name__ == "__main__":
    main()
