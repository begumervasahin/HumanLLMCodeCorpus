import sys
from RSA import primality3, generatePrime, stringToInt, RSA

# Increase the recursion limit to handle large inputs
sys.setrecursionlimit(10000000)

def main():
    while True:
        try:
            option = int(input(
                "Input 1-4 from the following options:\n"
                "1. Check for primality\n"
                "2. Generate a prime\n"
                "3. Encrypt a message\n"
                "4. Quit\n"
                "Your choice: "
            ))
            if 1 <= option <= 4:
                print(f"You chose option {option}")
                if option == 4:
                    print("Goodbye!")
                    break
                problemSelector(option)
            else:
                print("Invalid option. Please enter a number between 1 and 4.")
        except ValueError:
            print("Please enter a valid integer.")

def problemSelector(option):
    if option == 1:
        checkPrimality()
    elif option == 2:
        generatePrimeNumber()
    elif option == 3:
        encryptMessage()

def checkPrimality():
    N = getPositiveInteger("Choose an input N to check for primality (must be positive): ")
    K = getPositiveInteger("Choose K amount of times you want it tested (must be positive): ")
    isPrime = primality3(N, K)
    print(f"The number {N} is {'prime' if isPrime else 'not prime'}.")

def generatePrimeNumber():
    N = getPositiveInteger("Choose an input size N to make a binary prime number of N bits (must be positive): ")
    K = getPositiveInteger("Choose K amount of times you want it tested (must be positive): ")
    primeNum = generatePrime(N, K)
    print(f"Generated prime number of {N} bits: {primeNum}")

def encryptMessage():
    N = getPositiveInteger("Choose an input N to make a prime number of N bits (must be positive, larger input required for encryption of longer messages): ")
    K = getPositiveInteger("Choose K amount of times you want it tested for primality (must be positive): ")
    M = input("Choose a string or integer M to be encrypted: ")
    
    try:
        M = int(M)
        stringFlag = 0
    except ValueError:
        M = stringToInt(M)
        stringFlag = 1
    
    RSA(N, K, M, stringFlag)

def getPositiveInteger(prompt):
    while True:
        try:
            value = int(input(prompt))
            if value > 0:
                return value
            else:
                print("The number must be positive.")
        except ValueError:
            print("Please enter a valid integer.")

if __name__ == "__main__":
    main()
