from typing import Any, Tuple, List
import numpy as np
import logging

# Set up logging
logger = logging.getLogger(__name__)
logging.basicConfig(level=logging.INFO)

# Define Weightable class and its subclasses
class Weightable:
    def get_weight(self) -> float:
        pass

class SimpleWeight(Weightable):
    def __init__(self, weight: float):
        self.weight = weight
    
    def get_weight(self) -> float:
        return self.weight

# Define LCS class for Longest Common Subsequence calculation
class LCS:
    def __init__(self, X: List[Any], Y: List[Any], compare):
        self.X = X
        self.Y = Y
        self.compare = compare
        self.m = len(X)
        self.n = len(Y)
        self.matrix = np.zeros((self.m + 1, self.n + 1), dtype=float)
        self.lcs_length = 0
        self.threshold = 0.5
        self._calculate_lcs()
    
    def _calculate_lcs(self):
        for i in range(1, self.m + 1):
            for j in range(1, self.n + 1):
                if self.compare(self.X[i-1], self.Y[j-1]).get_weight() > 0:
                    self.matrix[i][j] = self.matrix[i-1][j-1] + 1
                else:
                    self.matrix[i][j] = max(self.matrix[i-1][j], self.matrix[i][j-1])
        self.lcs_length = self.matrix[self.m][self.n]
    
    def backtrack_indexes(self, i: int, j: int) -> List[Tuple[int, int, Weightable]]:
        indexes = []
        while i > 0 and j > 0:
            if self.compare(self.X[i-1], self.Y[j-1]).get_weight() > 0:
                indexes.append((i-1, j-1, self.compare(self.X[i-1], self.Y[j-1])))
                i -= 1
                j -= 1
            elif self.matrix[i-1][j] >= self.matrix[i][j-1]:
                i -= 1
            else:
                j -= 1
        indexes.reverse()
        return indexes
    
    def backtrack_full(self) -> Tuple[Tuple[int, int], Tuple[int, int], float]:
        i, j = self.m, self.n
        indexes = self.backtrack_indexes(i, j)
        if indexes:
            (i1, j1, w1) = indexes[0]
            (i2, j2, w2) = indexes[-1]
            s1 = (i1, i2 + 1)
            s2 = (j1, j2 + 1)
            total_weight = sum(wi.get_weight() for _, _, wi in indexes)
            weight = total_weight / self.lcs_length
            if weight < self.threshold:
                return None, None, 0
            return s1, s2, weight
        return None, None, 0

# Define EmbeddingContext classes
class EmbeddingContext:
    def get_embedding_tensor(self, text: str) -> np.ndarray:
        pass
    
    def get_embedding_tensors(self, texts: List[str]) -> List[np.ndarray]:
        return [self.get_embedding_tensor(text) for text in texts]
    
    def get_compare_func(self):
        pass

class CharEmbeddingContext(EmbeddingContext):
    def get_embedding_tensor(self, text: str) -> np.ndarray:
        return np.asarray(list(text))
    
    def get_compare_func(self):
        def simple_compare(x, y):
            return SimpleWeight(1.0) if x == y else SimpleWeight(0.0)
        return simple_compare

class SimpleTokenEmbeddingContext(CharEmbeddingContext):
    def get_embedding_tensor(self, text: str) -> np.ndarray:
        return np.asarray(text.split())

# Define Pattern class
class Pattern:
    def __init__(self, text: str, start: int, stop: int, embedding_context: EmbeddingContext = CharEmbeddingContext(), var_positions=[]):
        self.start = start
        self.stop = stop
        self.var_positions = var_positions
        self.embedding_context = embedding_context
        self.embedding = self.embedding_context.get_embedding_tensor(text)
        logger.info(f"Pattern shape: {self.get_pattern_embedding().shape}")
    
    def get_pattern_embedding(self) -> np.ndarray:
        return np.asarray(self.embedding[self.start:self.stop])
    
    def get_matcher(self, text_embedding_list: List[Any]) -> 'Matcher':
        return Matcher(self, text_embedding_list)
    
    def get_matcher_str(self, text: str) -> 'Matcher':
        text_emb_list = self.embedding_context.get_embedding_tensor(text)
        return self.get_matcher(text_emb_list)
    
    def get_pattern_len(self) -> int:
        return self.stop - self.start

# Define Matcher class
class Matcher:
    def __init__(self, pattern: Pattern, text_embedding_list: List[Any]) -> None:
        self.pattern = pattern
        self.text_emb = text_embedding_list
        self.lcs = LCS(text_embedding_list, pattern.get_pattern_embedding(), compare=pattern.embedding_context.get_compare_func())
        self.i, self.j = self.lcs.m - 1, self.lcs.n - 1
    
    def find(self) -> Tuple[float, Tuple[int, int]]:
        indexes = self.lcs.backtrack_indexes(self.i, self.j)
        logger.info(indexes)
        logger.info(f'LCS length: {self.lcs.lcs_length}')
        if indexes:
            (i1, j1, w1) = indexes[0]
            (i2, j2, w2) = indexes[-1]
            s1 = (i1, i2 + 1)
            s2 = (j1, j2 + 1)
            total_weight = sum(wi.get_weight() for _, _, wi in indexes)
            weight = total_weight / self.pattern.get_pattern_len()
            logger.info(f'Weight = {weight}')
            if weight < self.lcs.threshold:
                return None, None
            self.i, self.j = s1[0], self.lcs.n - 1
            self.lcs.lcs_length = self.lcs.matrix[self.i, self.j]
            return weight, s1
        return None, None

# Define helper functions
def find_fuzzy_pattern(pattern: Pattern, text: str) -> Tuple[float, Tuple[int, int]]:
    text_emb_list = pattern.embedding_context.get_embedding_tensor(text)
    return find_fuzzy_pattern_emb(pattern, text_emb_list)

def find_fuzzy_pattern_emb(pattern: Pattern, text_emb_list: List[Any]) -> Tuple[float, Tuple[int, int]]:
    lcs = LCS(text_emb_list, pattern.get_pattern_embedding(), compare=pattern.embedding_context.get_compare_func())
    return lcs.backtrack_full()

def get_string_from_tuple(result: Tuple[float, Tuple[int, int]], s: str, delimiter: str = ' ') -> str:
    _, span = result
    return get_string_from_span(span, s, delimiter=delimiter)

def get_string_from_span(span: Tuple[int, int], s: str, delimiter: str = ' ') -> str:
    return delimiter.join(s[span[0]:span[1]])
