def record_user_click(index, keyword, url):
    """Increment the click count for a URL associated with a keyword in the index."""
    urls = lookup(index, keyword)
    if urls:
        for entry in urls:
            if entry[0] == url:
                entry[1] += 1

def lookup(index, keyword):
    """Lookup URLs associated with a keyword in the index."""
    for entry in index:
        if entry[0] == keyword:
            return entry[1]
    return None

def add_to_index(index, keyword, url):
    """Add a URL to the index under a specified keyword."""
    for entry in index:
        if entry[0] == keyword:
            for urls in entry[1]:
                if urls[0] == url:
                    return
            entry[1].append([url, 0])
            return
    index.append([keyword, [[url, 0]]])

def union(a, b):
    """Add all elements of list b to list a if they are not already in a."""
    for e in b:
        if e not in a:
            a.append(e)

def get_next_target(page):
    """Return the next URL and its end position from a page's HTML content."""
    start_link = page.find('<a href=')
    if start_link == -1:
        return None, 0
    start_quote = page.find('"', start_link)
    end_quote = page.find('"', start_quote + 1)
    url = page[start_quote + 1:end_quote]
    return url, end_quote

def get_all_links(page):
    """Return a list of all URLs found in a page's HTML content."""
    links = []
    while True:
        url, endpos = get_next_target(page)
        if url:
            links.append(url)
            page = page[endpos:]
        else:
            break
    return links

def crawl_web(seed):
    """Crawl the web starting from the seed URL and build an index."""
    tocrawl = [seed]
    crawled = []
    index = []
    while tocrawl:
        page = tocrawl.pop()
        if page not in crawled:
            content = get_page(page)
            add_page_to_index(index, page, content)
            union(tocrawl, get_all_links(content))
            crawled.append(page)
    return index

def add_page_to_index(index, url, content):
    """Add all words in a page's content to the index under the URL."""
    words = content.split()
    for word in words:
        add_to_index(index, word, url)
