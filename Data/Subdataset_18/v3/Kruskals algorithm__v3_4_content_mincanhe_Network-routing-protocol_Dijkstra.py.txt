import collections

class Graph:
    def __init__(self):
        self.nodes = set()
        self.edges = collections.defaultdict(list)
        self.distances = {}

    def add_node(self, value):
        """Add a node to the graph."""
        self.nodes.add(value)

    def add_edge(self, from_node, to_node, distance):
        """Add an edge between two nodes with a given distance."""
        self.edges[from_node].append(to_node)
        self.edges[to_node].append(from_node)
        self.distances[(from_node, to_node)] = distance
        self.distances[(to_node, from_node)] = distance

def dijkstra(graph, start_node):
    """
    Implementation of Dijkstra's algorithm to find the shortest paths from the start_node to all other nodes in the graph.
    
    Parameters:
        graph: Instance of the Graph class.
        start_node: The node from which to calculate the shortest paths.
    
    Returns:
        visited: A dictionary of nodes with their minimum distances from the start_node.
        path: A dictionary representing the shortest path tree.
    """
    visited = {start_node: 0}  # Tracks minimum distance to each node from start_node
    path = {}  # Tracks the optimal path tree
    nodes = set(graph.nodes)

    while nodes:
        current_node = None

        # Find the node with the smallest known distance
        for node in nodes:
            if node in visited:
                if current_node is None or visited[node] < visited[current_node]:
                    current_node = node

        if current_node is None:
            break

        nodes.remove(current_node)
        current_weight = visited[current_node]

        # Explore neighbors of the current node
        for neighbor in graph.edges[current_node]:
            weight = current_weight + graph.distances[(current_node, neighbor)]
            if neighbor not in visited or weight < visited[neighbor]:
                visited[neighbor] = weight
                path[neighbor] = current_node

    return visited, path

def main():
    # Create a graph instance
    graph = Graph()
    
    # Add nodes to the graph
    nodes = ['A', 'B', 'C', 'D', 'E']
    for node in nodes:
        graph.add_node(node)

    # Add edges between nodes
    edges = [
        ('A', 'B', 1),
        ('A', 'C', 3),
        ('B', 'C', 1),
        ('B', 'D', 6),
        ('C', 'D', 4),
        ('C', 'E', 2),
        ('D', 'E', 1)
    ]
    
    for edge in edges:
        graph.add_edge(*edge)

    # Run Dijkstra's algorithm from node 'A'
    visited, path = dijkstra(graph, 'A')

    # Output the results
    print("Visited Nodes and Distances:", visited)
    print("Shortest Paths:", path)

if __name__ == '__main__':
    main()
