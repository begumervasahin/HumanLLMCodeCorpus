import click
import jieba
import jieba.analyse
from pymongo import MongoClient

def get_db():
    """
    Establishes a connection to the MongoDB instance and returns the database.
    Replace 'your_database_name' with the actual name of your database.
    """
    client = MongoClient('mongodb://localhost:27017/')
    return client['your_database_name']

# List of available collections in the database
collections = ['article-juejin']

# Get a reference to the database
db = get_db()

def process_text_rank_for_collection(collection_name):
    """
    Processes a specified collection, computes TextRank for each document,
    and updates the document with the computed ranks.
    
    :param collection_name: The name of the collection to process
    """
    collection = db[collection_name]
    jieba.analyse.set_stop_words('stopword.txt')

    # Iterate over all documents in the collection
    for document in collection.find():
        body = document.get('body')
        doc_id = document.get('document_id')
        print(f"Processing document_id: {doc_id}")

        # Compute TextRank for the document body
        ranks = [
            {keyword: weight}
            for keyword, weight in jieba.analyse.textrank(
                body, topK=30, withWeight=True, allowPOS=('ns', 'n')
            )
        ]

        # Prepare update data
        update_data = {
            '$set': {
                'jieba_text_rank': ranks
            }
        }

        # Update the document with the computed TextRank
        collection.update_one({'document_id': doc_id}, update_data, upsert=True)

@click.command()
@click.option('--collection', '-c', type=click.Choice(collections), multiple=True, help="Specify collections to process")
def textrank(collection):
    """
    CLI command to process specified collections. If no collections are specified,
    it processes all collections listed in 'collections'.
    
    :param collection: Tuple of collection names to process
    """
    selected_collections = collection if collection else collections
    print(f"Selected collections: {selected_collections}")

    # Process each selected collection
    for collection_name in selected_collections:
        if collection_name in collections:
            process_text_rank_for_collection(collection_name)

if __name__ == '__main__':
    textrank()
