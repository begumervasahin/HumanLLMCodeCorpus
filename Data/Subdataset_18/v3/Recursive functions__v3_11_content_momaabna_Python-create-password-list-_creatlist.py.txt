import sys

HELP_TEXT = """
Usage: createlist.py Min Max filename [p]
Min: length of minimum password
Max: length of maximum password
p: y to print on screen (optional)
Example: createlist.py 5 8 passwords.txt y
"""

def display_help_and_exit():
    """Display help text and exit the program."""
    print(HELP_TEXT)
    sys.exit()

def parse_arguments():
    """
    Parse command line arguments and return them.

    Returns:
        tuple: (min_length, max_length, filename, print_option)
    """
    if len(sys.argv) < 4:
        display_help_and_exit()

    try:
        min_length = int(sys.argv[1])
        max_length = int(sys.argv[2])
    except ValueError:
        display_help_and_exit()

    filename = sys.argv[3]
    print_option = sys.argv[4].lower() if len(sys.argv) > 4 else 'n'
    return min_length, max_length, filename, print_option

def read_letters():
    """
    Read the letters from 'letters.txt' and return them as a string.

    Returns:
        str: Contents of 'letters.txt'
    """
    try:
        with open('letters.txt', 'r') as file:
            return file.read().strip()
    except FileNotFoundError:
        print("Error: 'letters.txt' not found.")
        sys.exit(1)

def create_passwords(length, current_password, letters, print_option, file_handle):
    """
    Recursively create passwords of a given length.

    Args:
        length (int): Desired password length.
        current_password (str): Current password being built.
        letters (str): Possible letters to use in the passwords.
        print_option (str): Whether to print passwords to screen.
        file_handle (file): File handle to write passwords to.
    """
    if len(current_password) == length:
        for letter in letters:
            password = current_password + letter
            if print_option == 'y':
                print(password)
            file_handle.write(password + '\n')
    else:
        for letter in letters:
            create_passwords(length, current_password + letter, letters, print_option, file_handle)

def main():
    """Main function to generate passwords and write them to a file."""
    min_length, max_length, filename, print_option = parse_arguments()
    letters = read_letters()

    print("""
Welcome to Passlist Creator From rw-sec.com
Author: Mohammed Mahmood
Start creating list:
""")

    with open(filename, 'w') as file:
        for length in range(min_length, max_length + 1):
            create_passwords(length, '', letters, print_option, file)

    print('Password List Created Successfully.')

if __name__ == '__main__':
    main()
