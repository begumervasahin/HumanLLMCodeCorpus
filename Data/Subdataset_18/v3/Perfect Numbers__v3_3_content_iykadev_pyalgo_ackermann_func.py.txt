import sys

# Set the recursion limit to handle deeper recursive calls
sys.setrecursionlimit(3000)

def ackermann(m: int, n: int) -> int:
    """
    Computes the Ackermann function A(m, n).

    The Ackermann function is a well-known recursive function that grows very quickly.

    Parameters:
    m (int): The first parameter of the Ackermann function.
    n (int): The second parameter of the Ackermann function.

    Returns:
    int: The result of the Ackermann function.
    """
    if m == 0:
        return n + 1
    elif n == 0:
        return ackermann(m - 1, 1)
    else:
        return ackermann(m - 1, ackermann(m, n - 1))

def main() -> None:
    """
    Main function to test the Ackermann function.
    """
    # Test the Ackermann function with the given values
    m = 3
    n = 4
    result = ackermann(m, n)
    print(f"Ackermann({m}, {n}) = {result}")

if __name__ == "__main__":
    main()
