import base58

# TypeConversion module
class TypeConversion:
    @staticmethod
    def string_to_big_int(s):
        """
        Convert a string to a big integer.

        Args:
            s (str): The string to convert.

        Returns:
            int: The resulting big integer.
        """
        return int.from_bytes(s.encode(), 'big')

    @staticmethod
    def big_int_to_string(num):
        """
        Convert a big integer to a string.

        Args:
            num (int): The big integer to convert.

        Returns:
            str: The resulting string.
        """
        byte_length = (num.bit_length() + 7) // 8
        return num.to_bytes(byte_length, 'big').decode()

    @staticmethod
    def int_to_b58(number):
        """
        Convert an integer to a base-58 string.

        Args:
            number (int): The integer to convert.

        Returns:
            str: The base-58 encoded string.
        """
        return base58.b58encode_int(number).decode()

    @staticmethod
    def b58_to_int(b58_string):
        """
        Convert a base-58 string to an integer.

        Args:
            b58_string (str): The base-58 string to convert.

        Returns:
            int: The resulting integer.
        """
        return base58.b58decode_int(b58_string)

# Encryption and Decryption functions
def xor_encrypt(message, symmetric_key_b58):
    """
    Encrypt a message using XOR with a symmetric key.

    Args:
        message (str): The plaintext message.
        symmetric_key_b58 (str): The base-58 encoded symmetric key.

    Returns:
        str: The base-58 encoded encrypted message.
    """
    message_int = TypeConversion.string_to_big_int(message)
    symmetric_key_int = TypeConversion.b58_to_int(symmetric_key_b58)
    encrypted_message_int = message_int ^ symmetric_key_int
    return TypeConversion.int_to_b58(encrypted_message_int)

def xor_decrypt(encrypted_message_b58, symmetric_key_b58):
    """
    Decrypt a message using XOR with a symmetric key.

    Args:
        encrypted_message_b58 (str): The base-58 encoded encrypted message.
        symmetric_key_b58 (str): The base-58 encoded symmetric key.

    Returns:
        str: The decrypted plaintext message.
    """
    encrypted_message_int = TypeConversion.b58_to_int(encrypted_message_b58)
    symmetric_key_int = TypeConversion.b58_to_int(symmetric_key_b58)
    decrypted_message_int = encrypted_message_int ^ symmetric_key_int
    return TypeConversion.big_int_to_string(decrypted_message_int)

# Example usage
if __name__ == "__main__":
    # Define a message and a symmetric key
    message = "Hello, World!"
    symmetric_key = "5HueCGU8rMjxEXxQk6t6v4F5FzbgQqTjZKLRkxakj89vC"  # Base-58 encoded key

    # Encrypt the message
    encrypted_message_b58 = xor_encrypt(message, symmetric_key)
    print(f"Encrypted message (base-58): {encrypted_message_b58}")

    # Decrypt the message
    decrypted_message = xor_decrypt(encrypted_message_b58, symmetric_key)
    print(f"Decrypted message: {decrypted_message}")
