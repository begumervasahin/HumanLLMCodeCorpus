import random

def partition3(arr, left, right):
    """
    Perform a three-way partitioning of the array.

    This function rearranges the elements in the array such that elements less than the pivot
    are on its left, elements equal to the pivot are in the middle, and elements greater than the pivot
    are on its right.

    :param arr: The list to be partitioned.
    :param left: The starting index of the partition.
    :param right: The ending index of the partition.
    :return: A tuple containing the indices for the left and right partitions.
    """
    pivot = arr[left]
    less = left
    equal = left

    for i in range(left + 1, right + 1):
        if arr[i] < pivot:
            less += 1
            arr[i], arr[less] = arr[less], arr[i]
            arr[less], arr[equal] = arr[equal], arr[less]
            equal += 1
        elif arr[i] == pivot:
            equal += 1
            arr[i], arr[equal] = arr[equal], arr[i]

    arr[left], arr[less] = arr[less], arr[left]

    return less, equal

def randomized_quick_sort3(arr, left, right):
    """
    Perform the randomized QuickSort with a three-way partitioning.

    This function sorts the array using the QuickSort algorithm with a randomized pivot and three-way partitioning
    to handle duplicates more efficiently.

    :param arr: The list to be sorted.
    :param left: The starting index of the sort.
    :param right: The ending index of the sort.
    """
    if left >= right:
        return

    pivot_index = random.randint(left, right)
    arr[left], arr[pivot_index] = arr[pivot_index], arr[left]

    m1, m2 = partition3(arr, left, right)

    randomized_quick_sort3(arr, left, m1 - 1)
    randomized_quick_sort3(arr, m2 + 1, right)

def main():
    """
    Main function to read input data, perform QuickSort, and print the sorted array.
    """
    input_data = "5\n2 3 9 2 2"
    input_lines = input_data.split('\n')
    n = int(input_lines[0])
    arr = list(map(int, input_lines[1].split()))

    randomized_quick_sort3(arr, 0, n - 1)

    print("Sorted array:", arr)

if __name__ == '__main__':
    main()
