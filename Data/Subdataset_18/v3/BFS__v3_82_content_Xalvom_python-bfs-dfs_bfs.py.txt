# Define the graph
graph = {
    'A': {'B'},
    'B': {'A', 'C'},
    'C': {'B', 'D', 'H', 'I'},
    'D': {'C', 'E', 'F', 'H', 'M'},
    'E': {'D'},
    'F': {'D', 'G', 'M'},
    'G': {'F', 'H'},
    'H': {'C', 'D', 'G', 'L'},
    'I': {'C', 'J', 'K'},
    'J': {'I'},
    'K': {'I', 'L'},
    'L': {'H', 'K'},
    'M': {'D', 'F'}
}

def bfs(graph, start, goal):
    """
    Perform Breadth-First Search (BFS) to find the shortest path from start to goal.

    :param graph: The graph represented as an adjacency list.
    :param start: The starting node for the BFS.
    :param goal: The goal node to reach.
    :return: A list representing the path from start to goal if found, otherwise "Path not found".
    """
    queue = [[start]]  # Initialize the queue with the start node
    visited = set()  # Initialize a set to keep track of visited nodes

    while queue:
        path = queue.pop(0)  # Get the first path from the queue
        node = path[-1]  # Get the last node from the path

        if node == goal:
            return path  # Return the path if the goal is reached

        elif node not in visited:
            for neighbor in graph.get(node, []):
                new_path = list(path)  # Create a new path
                new_path.append(neighbor)  # Append the neighbor to the path
                queue.append(new_path)  # Add the new path to the queue

            visited.add(node)  # Mark the node as visited

    return "Path not found"

    print()

    result = bfs(graph, start_node, goal_node)
