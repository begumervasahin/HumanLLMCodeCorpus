def selection_sort(arr):
    """Sorts an array using the selection sort algorithm."""
    n = len(arr)
    for i in range(n):
        # Assume the current position holds the smallest element
        min_index = i
        # Find the smallest element in the remaining unsorted portion of the array
        for j in range(i + 1, n):
            if arr[min_index] > arr[j]:
                min_index = j
        # Swap the found minimum element with the element at the current position
        if i != min_index:
            arr[i], arr[min_index] = arr[min_index], arr[i]
    return arr

def print_array(arr):
    """Prints the elements of the array."""
    print(arr)

# Example usage
if __name__ == "__main__":
    arr = [6, 3, 1, 9, 2, 5, 8, 7, 4]
    print("Original array:")
    print_array(arr)
    
    sorted_arr = selection_sort(arr)
    
    print("Sorted array:")
    print_array(sorted_arr)
