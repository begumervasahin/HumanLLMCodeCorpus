import gensim

def load_word2vec_model(model_path):
    """
    Load the pre-trained Word2Vec model from the specified path.

    Args:
    - model_path (str): Path to the Word2Vec model file.

    Returns:
    - gensim.models.KeyedVectors: Loaded Word2Vec model.
    """
    return gensim.models.KeyedVectors.load_word2vec_format(model_path, binary=True)

def get_similar_words(model, word, topn=10):
    """
    Get the top N most similar words for the given word using the specified Word2Vec model.

    Args:
    - model (gensim.models.KeyedVectors): Pre-trained Word2Vec model.
    - word (str): The word to find similar words for.
    - topn (int): Number of similar words to retrieve (default is 10).

    Returns:
    - list of str: List of similar words.
    """
    similar_words = model.most_similar(positive=[word], topn=topn)
    return [word for word, _ in similar_words]

def calculate_cosine_similarities(model, word, similar_words):
    """
    Calculate cosine similarities between the given word and each word in the similar words list.

    Args:
    - model (gensim.models.KeyedVectors): Pre-trained Word2Vec model.
    - word (str): The reference word.
    - similar_words (list of str): List of words to calculate similarities with.

    Returns:
    - str: Formatted string with cosine similarities.
    """
    similarities = [
        f"{word} - {similar_word}: {model.similarity(word, similar_word):.4f}"
        for similar_word in similar_words
    ]
    return "\n".join(similarities)

def main():
    # Load the Word2Vec model
    model_path = 'GoogleNews-vectors-negative300.bin.gz'
    model = load_word2vec_model(model_path)

    # Define words of interest
    word1 = "rebellion"
    word2 = "slave"

    # Get similar words for each word
    similar_words1 = get_similar_words(model, word1)
    similar_words2 = get_similar_words(model, word2)

    # Calculate cosine similarities
    cosine_similarities1 = calculate_cosine_similarities(model, word1, similar_words1)
    cosine_similarities2 = calculate_cosine_similarities(model, word2, similar_words2)

    # Print results
    print(f"Top ten similar words for '{word1}':\n")
    print(", ".join(similar_words1))
    print(f"\nTop ten similar words for '{word2}':\n")
    print(", ".join(similar_words2))
    print(f"\nCosine similarities for '{word1}':\n")
    print(cosine_similarities1)
    print(f"\nCosine similarities for '{word2}':\n")
    print(cosine_similarities2)

if __name__ == "__main__":
    main()
