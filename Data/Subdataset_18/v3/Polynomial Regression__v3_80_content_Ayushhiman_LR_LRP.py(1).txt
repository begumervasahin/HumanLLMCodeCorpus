import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.preprocessing import PolynomialFeatures
from sklearn.feature_selection import f_regression, mutual_info_regression
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn import metrics

# Function to load and display dataset
def load_and_display_data(filepath):
    df = pd.read_csv(filepath)
    print(df.head())
    print(df.describe())
    return df

# Function to perform exploratory data analysis
def exploratory_data_analysis(df):
    sns.set_palette("GnBu_d")
    sns.set_style('whitegrid')

    sns.jointplot(x='Time on Website', y='Yearly Amount Spent', data=df)
    sns.jointplot(x='Time on App', y='Yearly Amount Spent', data=df)
    sns.jointplot(x='Time on App', y='Length of Membership', kind='hex', data=df)
    sns.pairplot(df)
    sns.lmplot(x='Length of Membership', y='Yearly Amount Spent', data=df)
    plt.show()

# Function to prepare features and target variable
def prepare_features(df):
    y = df['Yearly Amount Spent']
    X = df[['Avg. Session Length', 'Time on App', 'Time on Website', 'Length of Membership']]
    return X, y

# Function to generate polynomial features
def generate_polynomial_features(X, degree=2):
    poly = PolynomialFeatures(degree=degree, interaction_only=False, include_bias=True)
    return poly.fit_transform(X)

# Function to perform feature selection
def feature_selection(X, y):
    print(f_regression(X, y, center=True))
    print(mutual_info_regression(X, y, discrete_features='auto', n_neighbors=3, copy=True, random_state=None))

# Function to split data into training and testing sets
def split_data(X, y, test_size=0.3, random_state=101):
    return train_test_split(X, y, test_size=test_size, random_state=random_state)

# Function to initialize and train linear regression models
def train_models(X_train, y_train, X1_train, X2_train):
    lm = LinearRegression()
    lm1 = LinearRegression()
    lm2 = LinearRegression()

    lm.fit(X_train, y_train)
    lm1.fit(X1_train, y_train)
    lm2.fit(X2_train, y_train)

    return lm, lm1, lm2

# Function to display model coefficients
def display_coefficients(models):
    for i, model in enumerate(models, 1):
        print(f'Model {i} Coefficients:', model.coef_)

# Function to generate predictions
def generate_predictions(models, X_tests):
    return [model.predict(X_test) for model, X_test in zip(models, X_tests)]

# Function to plot predictions vs actual values
def plot_predictions(y_test, predictions, title):
    plt.scatter(y_test, predictions)
    plt.xlabel('Y Test')
    plt.ylabel('Predicted Y')
    plt.title(title)
    plt.show()

# Function to evaluate model performance
def evaluate_model(y_test, predictions, model_name):
    print(f'{model_name}')
    print('MAE:', metrics.mean_absolute_error(y_test, predictions))
    print('MSE:', metrics.mean_squared_error(y_test, predictions))
    print('RMSE:', np.sqrt(metrics.mean_squared_error(y_test, predictions)))
    print()

# Main execution
if __name__ == "__main__":
    filepath = "Ecommerce Customers"
    customers = load_and_display_data(filepath)
    exploratory_data_analysis(customers)

    X, y = prepare_features(customers)
    X_poly = generate_polynomial_features(X)
    feature_selection(X, y)

    X_train, X_test, y_train, y_test = split_data(X, y)
    X1 = X.drop('Time on App', axis=1)
    X1_train, X1_test, y_train, y_test = split_data(X1, y)
    X2_train, X2_test, y_train, y_test = split_data(X_poly, y)

    lm, lm1, lm2 = train_models(X_train, y_train, X1_train, X2_train)
    display_coefficients([lm, lm1, lm2])

    predictions, predictions1, predictions2 = generate_predictions([lm, lm1, lm2], [X_test, X1_test, X2_test])
    
    plot_predictions(y_test, predictions, 'Model 1 Predictions')
    plot_predictions(y_test, predictions1, 'Model 2 Predictions')
    plot_predictions(y_test, predictions2, 'Model 3 Predictions')

    evaluate_model(y_test, predictions, 'Model 1')
    evaluate_model(y_test, predictions1, 'Model 2')
    evaluate_model(y_test, predictions2, 'Model 3')
