from collections import deque

# Set a higher recursion limit if needed
sys.setrecursionlimit(10**6)

class Puzzle:
    """
    A generic puzzle class that must be implemented with specific puzzle logic.
    """
    def extensions(self):
        """
        Return a list of possible extensions of the current puzzle state.
        """
        pass
    
    def is_solved(self):
        """
        Return whether the puzzle is in a solved state.
        """
        pass
    
    def fail_fast(self):
        """
        Return whether the current state is a dead-end.
        """
        pass

class PuzzleNode:
    """
    A node in the puzzle-solving process.
    """
    def __init__(self, puzzle=None, children=None, parent=None):
        self.puzzle = puzzle
        self.parent = parent
        self.children = children[:] if children else []
        self.visited = False
    
    def __eq__(self, other):
        return (type(self) == type(other) and
                self.puzzle == other.puzzle and
                all(x in self.children for x in other.children) and
                all(x in other.children for x in self.children))
    
    def __str__(self):
        return "{}\n\n{}".format(self.puzzle, "\n".join(str(x) for x in self.children))

def depth_first_solve(puzzle):
    """
    Solve the puzzle using depth-first search (DFS).
    """
    visited = set()
    stack = deque([PuzzleNode(puzzle)])
    
    while stack:
        node = stack.pop()
        
        for config in node.puzzle.extensions():
            if str(config) in visited:
                continue
            if config.is_solved():
                return PuzzleNode(config)
            if config.fail_fast() or not config.extensions():
                continue
            
            new_node = PuzzleNode(config)
            visited.add(str(new_node.puzzle))
            stack.append(new_node)
    
    return None

def breadth_first_solve(puzzle):
    """
    Solve the puzzle using breadth-first search (BFS).
    """
    visited = set()
    queue = deque([PuzzleNode(puzzle)])
    
    while queue:
        node = queue.popleft()
        
        for child in node.puzzle.extensions():
            if str(child) in visited:
                continue
            if child.is_solved():
                return PuzzleNode(child)
            if child.fail_fast() or not child.extensions():
                continue
            
            new_node = PuzzleNode(child)
            visited.add(str(new_node.puzzle))
            queue.append(new_node)
    
    return None
