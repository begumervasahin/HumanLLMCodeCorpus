import random

def insertion_sort(arr):
    """Sort a list using the insertion sort algorithm."""
    for i in range(1, len(arr)):
        key = arr[i]
        j = i - 1
        # Shift elements of arr[0..i-1] that are greater than key to one position ahead
        while j >= 0 and arr[j] > key:
            arr[j + 1] = arr[j]
            j -= 1
        arr[j + 1] = key
    return arr

def choose_pivot(arr, low, high, pivot_type):
    """Select a pivot element based on the specified type and move it to the end."""
    pivot_selection = {
        'first': low,
        'last': high,
        'middle': (low + high) // 2,
        'random': random.randint(low, high)
    }
    pivot_index = pivot_selection.get(pivot_type, high)
    arr[pivot_index], arr[high] = arr[high], arr[pivot_index]
    return arr[high]

def partition(arr, low, high, pivot_type):
    """Partition the array into elements less than or equal to the pivot and those greater."""
    pivot = choose_pivot(arr, low, high, pivot_type)
    i = low - 1  # Index of smaller element
    for j in range(low, high):
        if arr[j] <= pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    return i + 1

def quicksort(arr, low=0, high=None, pivot_type='last', use_insertion_sort=False, threshold=10):
    """
    Sort the array using the quicksort algorithm.
    Optionally, use insertion sort for small segments to optimize performance.
    """
    if high is None:
        high = len(arr) - 1

    if low < high:
        if use_insertion_sort and (high - low + 1) <= threshold:
            # Apply insertion sort for small segments
            insertion_sort(arr[low:high + 1])
        else:
            # Partition the array and recursively apply quicksort
            pivot_index = partition(arr, low, high, pivot_type)
            quicksort(arr, low, pivot_index - 1, pivot_type, use_insertion_sort, threshold)
            quicksort(arr, pivot_index + 1, high, pivot_type, use_insertion_sort, threshold)

    return arr

# Example usage:
if __name__ == "__main__":
    data = [10, 7, 8, 9, 1, 5]
    print("Unsorted:", data)
    sorted_data = quicksort(data, pivot_type='middle', use_insertion_sort=True, threshold=10)
    print("Sorted:", sorted_data)
