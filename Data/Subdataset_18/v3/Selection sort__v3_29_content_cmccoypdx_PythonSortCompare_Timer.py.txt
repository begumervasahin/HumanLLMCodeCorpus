import timeit

def wrapper(func, *args, **kwargs):
    """
    Wrap the function and its arguments for timing.
    
    Parameters:
    func (callable): The function to wrap.
    *args: Variable length argument list for the function.
    **kwargs: Arbitrary keyword arguments for the function.
    
    Returns:
    callable: A wrapped function that can be timed.
    """
    def wrapped():
        return func(*args, **kwargs)
    return wrapped

def time_function(func, *args, **kwargs):
    """
    Time the execution of a function with the provided arguments.
    
    Parameters:
    func (callable): The function to time.
    *args: Variable length argument list for the function.
    **kwargs: Arbitrary keyword arguments for the function.
    
    Returns:
    float: The time taken to execute the function.
    """
    wrapped = wrapper(func, *args, **kwargs)
    return timeit.timeit(wrapped, number=1)

def sample_function(x, y):
    """
    A sample function to demonstrate the timing functionality.
    
    Parameters:
    x (int): The first number.
    y (int): The second number.
    
    Returns:
    int: The sum of x and y.
    """
    return x + y

if __name__ == "__main__":
    # Time the sample function with arguments
    time_taken = time_function(sample_function, 3, 5)
    print(f"Time taken to execute sample_function: {time_taken:.6f} seconds")
