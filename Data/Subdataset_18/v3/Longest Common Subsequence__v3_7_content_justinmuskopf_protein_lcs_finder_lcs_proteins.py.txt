from sys import argv
from os.path import isfile

LENGTH = 0
NAME = 0
SEQUENCE = 1

def die(message=None, code=1):
    """Print an error message and exit."""
    if message:
        print(f"Error: {message}")
    else:
        print("Error: An undefined error occurred!")
    exit(code)

def read_sequences(filename):
    """Read sequences from a file and return them as a list of tuples (name, sequence)."""
    try:
        with open(filename, "r") as file:
            file_content = file.read()
    except IOError:
        die(f"Error! Could not open {filename}")

    sequences = file_content.strip().split("\n\n")
    if not sequences:
        die("Could not find any sequences!")
    
    if sequences[0][0] != '>':
        sequences = sequences[1:]
    
    return [parse_sequence(seq) for seq in sequences if seq]

def parse_sequence(sequence):
    """Parse a sequence entry into name and sequence data."""
    first_newline_index = sequence.find('\n')
    name = sequence[1:first_newline_index]
    sequence_data = sequence[first_newline_index + 1:].replace("\n", "")
    return name, sequence_data

def compute_lcs(seq1, seq2):
    """Compute the Longest Common Subsequence (LCS) of two sequences."""
    len1, len2 = len(seq1), len(seq2)
    lcs_matrix = [[""] * (len2 + 1) for _ in range(len1 + 1)]
    
    for i in range(1, len1 + 1):
        for j in range(1, len2 + 1):
            if seq1[i - 1] == seq2[j - 1]:
                lcs_matrix[i][j] = lcs_matrix[i - 1][j - 1] + seq1[i - 1]
            else:
                lcs_matrix[i][j] = max(lcs_matrix[i - 1][j], lcs_matrix[i][j - 1], key=len)
    
    lcs_string = lcs_matrix[len1][len2]
    return len(lcs_string), lcs_string

def find_max_lcs(sequence_dict):
    """Find the sequence with the maximum LCS length."""
    max_lcs_length = 0
    max_lcs_entry = None
    
    for key, seq_object in sequence_dict.items():
        lcs_length = seq_object[LENGTH]
        if lcs_length > max_lcs_length:
            max_lcs_entry = (key, seq_object)
            max_lcs_length = lcs_length
    
    return max_lcs_entry

def add_sequence_to_group(group_name, sequence, groups):
    """Add a sequence to the appropriate group."""
    if group_name not in groups:
        groups[group_name] = []
    groups[group_name].append(sequence)

def group_sequences_by_lcs(sequences):
    """Group sequences based on their LCS lengths."""
    groups = {}
    
    for seq in sequences:
        max_seq_name, max_seq_object = find_max_lcs(sequences[seq])
        sequence_to_add = (seq, max_seq_object[LENGTH], max_seq_object[SEQUENCE])
        add_sequence_to_group(max_seq_name, sequence_to_add, groups)
    
    return groups

def get_filename_from_args(args):
    """Retrieve the filename from command line arguments or prompt the user."""
    if len(args) < 2:
        return input("Please enter the filename containing protein sequences: ")
    else:
        filename = args[1]
    
    if not isfile(filename):
        die(f"Could not find file '{filename}'")
    
    return filename

def compute_lcs_for_all_pairs(sequences):
    """Compute LCS for all pairs of sequences."""
    lcs_dict = {name: {} for name, _ in sequences}
    num_sequences = len(sequences)
    
    for idx, (first_name, first_seq) in enumerate(sequences):
        print(f"[{idx + 1}/{num_sequences}] {first_name}...")
        for second_name, second_seq in sequences[idx:]:
            if first_name != second_name:
                lcs_length, lcs_seq = compute_lcs(first_seq, second_seq)
                lcs_dict[first_name][second_name] = (lcs_length, lcs_seq)
                lcs_dict[second_name][first_name] = (lcs_length, lcs_seq)
                print(f"   {second_name}... LCS Length: {lcs_length}")
        print()
    
    return lcs_dict

def print_grouped_sequence(grouped_item, padding=2, offset=20):
    """Print a grouped sequence item."""
    padded_space = " " * padding
    name_str = f"Name: {grouped_item[NAME]}"
    offset_space = " " * max(offset - len(name_str), 0)
    print(f"{padded_space} {name_str}{offset_space} Length of LCS: {grouped_item[1]}")

def main(args):
    """Main function to execute the script."""
    filename = get_filename_from_args(args)
    sequences = read_sequences(filename)
    lcs_dict = compute_lcs_for_all_pairs(sequences)
    grouped_sequences = group_sequences_by_lcs(lcs_dict)
    
    for group, items in grouped_sequences.items():
        print(f"Group {group}:")
        for item in items:
            print_grouped_sequence(item)

if __name__ == "__main__":
    main(argv)
