def partition(arr, low, high):
    """
    Partition the array for quicksort. The pivot is chosen as the second value of the last element in the array.
    Elements less than or equal to the pivot are moved to the left of the pivot, and greater elements to the right.
    
    Parameters:
        arr: List of tuples to be sorted
        low: Starting index
        high: Ending index (pivot)
    
    Returns:
        The partition index where the pivot is placed.
    """
    i = low - 1
    pivot = arr[high][1]  # Choose the second value of the last element as the pivot

    for j in range(low, high):
        if arr[j][1] <= pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]  # Swap if element is less than or equal to pivot

    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # Place pivot in the correct position
    return i + 1

def quick_sort(arr, low, high):
    """
    Perform quicksort on the array.
    
    Parameters:
        arr: List of tuples to be sorted
        low: Starting index for the sort
        high: Ending index for the sort
    """
    if low < high:
        pi = partition(arr, low, high)
        quick_sort(arr, low, pi - 1)  # Sort the left partition
        quick_sort(arr, pi + 1, high)  # Sort the right partition

def filter_data(filename, location):
    """
    Filter data from the file based on the location and extract specific columns.
    
    Parameters:
        filename: The file to read from
        location: The location to filter by
    
    Returns:
        A list of tuples containing the filtered data
    """
    filtered_data = []
    with open(filename, encoding="utf-8") as file:
        for line in file:
            columns = line.split(",")
            if columns[3].strip() == location:
                filtered_data.append((float(columns[5]), float(columns[6])))
    return filtered_data

def write_data_to_file(data, output_filename):
    """
    Write the sorted data to a new file.
    
    Parameters:
        data: The data to write
        output_filename: The file to write to
    """
    with open(output_filename, "w") as output_file:
        for item in data:
            output_file.write(f"{item[0]},{item[1]}\n")

def main():
    input_filename = "db2.txt"
    output_filename = "testSET4.txt"
    location = "LOS OLIVOS"

    # Filter the data based on location
    filtered_data = filter_data(input_filename, location)

    # Sort the filtered data using quicksort
    quick_sort(filtered_data, 0, len(filtered_data) - 1)

    # Print the sorted data
    for item in filtered_data:
        print(item)

    # Write the sorted data to a new file
    write_data_to_file(filtered_data, output_filename)

    # Print the number of filtered and sorted lines
    print(f"Number of filtered and sorted lines: {len(filtered_data)}")

if __name__ == "__main__":
    main()
