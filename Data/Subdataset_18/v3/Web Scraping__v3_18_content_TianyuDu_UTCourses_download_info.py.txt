import numpy as np
import pandas as pd
import selenium
from selenium import webdriver
from selenium.webdriver.common.by import By
import time

class TimetableBot:
    def __init__(self, driver_path: str, url: str):
        """Initialize the webdriver and open the specified URL."""
        self.driver = webdriver.Chrome(driver_path)
        self.driver.get(url)
        
    def close(self):
        """Close the webdriver."""
        self.driver.close()
        
    def batch_retrieve(self, department: str) -> pd.DataFrame:
        """Retrieve course information for a specified department."""
        department = department.upper()
        
        # Locate and interact with search elements
        search_box = self.driver.find_element(By.ID, "courseCode")
        search_button = self.driver.find_element(By.ID, "searchButton")
        
        search_box.clear()
        search_box.send_keys(department)
        search_button.click()
        
        # Wait for search results to load
        course_elements = []
        while not course_elements:
            time.sleep(1.0)
            course_elements = self.driver.find_elements(By.CLASS_NAME, "perCourse")
        
        # Extract course information
        course_info_list = []
        print(f"Total courses found: {len(course_elements)}")
        for course_element in course_elements:
            code, title = course_element.find_element(By.CLASS_NAME, "courseTitle").text.split("   ")
            print(f"{code}\t{title}")
            meeting_elements = course_element.find_elements(By.CLASS_NAME, "perMeeting")
            for meeting_element in meeting_elements:
                meeting_code = meeting_element.find_element(By.CLASS_NAME, "colCode").text
                print(f"\t{meeting_code}")
                meeting_info = self.get_meeting_info(meeting_element, meeting_code)
                course_info_list.append([code, title, meeting_code, meeting_info])
        
        # Create a DataFrame from the extracted information
        course_info_df = pd.DataFrame(
            course_info_list,
            columns=["Code", "Title", "Session", "Details"]
        )
        return course_info_df

    def get_meeting_info(self, meeting_element, meeting_code):
        """Get meeting information based on meeting code."""
        try:
            if meeting_code.startswith("LEC"):
                return meeting_element.find_element(By.CLASS_NAME, "secLec").text
            else:
                return meeting_element.find_element(By.CLASS_NAME, "secTut").text
        except selenium.common.exceptions.NoSuchElementException:
            return meeting_element.find_element(By.CLASS_NAME, "secPra").text
    
    def batch_download(self, save_dir: str) -> None:
        """Download course information for all departments and save to a CSV file."""
        department_options = self.driver.find_elements(By.CLASS_NAME, "option")
        department_codes = [
            option.text.split(" ")[0][1:-1]
            for option in department_options
            if option.text.startswith("(") and option.text.endswith(")")
        ]
        
        # Retrieve information for each department and concatenate into a single DataFrame
        all_courses = []
        for code in department_codes:
            df = self.batch_retrieve(code)
            all_courses.append(df)
        
        final_df = pd.concat(all_courses, axis=0)
        print(final_df.head())
        print(final_df.shape)
        final_df.to_csv(save_dir, index=False)

if __name__ == "__main__":
    # Initialize the bot with the appropriate driver path and URL
    bot = TimetableBot(driver_path="./chromedriver", url="https://timetable.iit.artsci.utoronto.ca/")
    
    # Perform batch download and save results to CSV
    bot.batch_download(save_dir="./results.csv")
    
    # Close the bot
    bot.close()
