import matplotlib.pyplot as plt

def plot_path_with_annotations(db, reco, cost, time, x_median, y_median):
    """
    Plots the path based on the provided data, annotates it with distance and time, 
    and marks the starting point and median lines.
    
    Parameters:
        db: Dictionary of points with indices as keys and (longitude, latitude) tuples as values.
        reco: List of tuples representing the path, with each tuple containing the start and end point indices.
        cost: The total distance of the path.
        time: The total time associated with the path.
        x_median: The median value for the x-axis (longitude).
        y_median: The median value for the y-axis (latitude).
    """
    # Initialize lists to store the x and y coordinates of the path
    x_coords = []
    y_coords = []

    # Collect coordinates based on the path indices (reco)
    for start, end in reco:
        x_coords.append(db[start][0])
        y_coords.append(db[start][1])

    # Closing the path by adding the starting point at the end
    x_coords.append(x_coords[0])
    y_coords.append(y_coords[0])

    # Plot the path
    plt.plot(x_coords, y_coords, label='Path')

    # Annotate the plot with distance and time information
    plt.text(x_median - 0.2, y_median - 0.2, f"Distancia: {cost}")
    plt.text(x_median - 0.2, y_median - 0.4, f"Tiempo: {time}")

    # Plot the points and highlight the starting point
    plt.plot(x_coords, y_coords, 'o', label='Points')
    plt.plot(x_coords[0], y_coords[0], 'rx', label='Start Point')

    # Plot median lines for x and y
    plt.axhline(y_median, color='black', linestyle='--', label='Y Median')
    plt.axvline(x_median, color='black', linestyle='--', label='X Median')

    # Add labels and a legend
    plt.xlabel('Longitude')
    plt.ylabel('Latitude')
    plt.legend()

    # Display the plot
    plt.show()
