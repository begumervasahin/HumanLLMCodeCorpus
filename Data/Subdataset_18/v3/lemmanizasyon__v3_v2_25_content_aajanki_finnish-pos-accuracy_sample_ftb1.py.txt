import hashlib
import re

def main():
    # Define input and output file paths
    input_file_path = 'data/test/ftb1/ftb1u.tsv'
    output_file_path = 'data/test/ftb1/ftb1u_sample.tsv'

    # Read sentences from the input file
    sentences = read_sentences_from_file(input_file_path)

    # Select sentences with SHA-256 digest starting with '0'
    selected_sentences = select_sentences_with_digest_starting_with_zero(sentences)

    # Print summary of sampled sentences
    print(f'Sampled {len(selected_sentences)} out of {len(sentences)} sentences')

    # Write selected sentences to the output file
    write_sentences_to_file(output_file_path, selected_sentences)

    # Print confirmation message
    print(f'Wrote samples to {output_file_path}')

def read_sentences_from_file(file_path):
    """
    Read lines from a file and split them into sentences.

    Args:
        file_path (str): Path to the input file.

    Returns:
        list of str: List of sentences.
    """
    with open(file_path, 'r') as file:
        lines = file.readlines()
    return list(split_sentences(lines))

def split_sentences(lines):
    """
    Split lines into sentences using a regular expression to match the start of a sentence.

    Args:
        lines (list of str): Lines from the input file.

    Yields:
        str: A complete sentence.
    """
    sentence = []
    start_of_sentence_re = re.compile(r'^')

    for line in lines:
        # Check if the line matches the start of a sentence
        if start_of_sentence_re.match(line):
            # If a sentence is being accumulated, yield it
            if sentence:
                yield ''.join(sentence)
                sentence = []  # Reset sentence buffer
        # Add non-empty lines to the current sentence
        if line.strip():
            sentence.append(line)

    # Yield the last accumulated sentence if any
    if sentence:
        yield ''.join(sentence)

def is_selected_sentence(sentence):
    """
    Determine if the sentence should be selected based on its SHA-256 digest.

    Args:
        sentence (str): The sentence to check.

    Returns:
        bool: True if the SHA-256 digest of the sentence starts with '0', False otherwise.
    """
    digest = hashlib.sha256(sentence.encode('utf-8')).hexdigest()
    return digest.startswith('0')

def select_sentences_with_digest_starting_with_zero(sentences):
    """
    Select sentences whose SHA-256 digest starts with '0'.

    Args:
        sentences (list of str): List of sentences to check.

    Returns:
        list of str: List of selected sentences.
    """
    return [sentence for sentence in sentences if is_selected_sentence(sentence)]

def write_sentences_to_file(file_path, sentences):
    """
    Write sentences to a file.

    Args:
        file_path (str): Path to the output file.
        sentences (list of str): List of sentences to write.
    """
    with open(file_path, 'w') as file:
        file.write('\n'.join(sentences))

if __name__ == '__main__':
    main()
