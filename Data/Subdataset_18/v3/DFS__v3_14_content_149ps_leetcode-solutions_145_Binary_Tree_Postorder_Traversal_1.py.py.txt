class TreeNode(object):
    def __init__(self, x):
        self.val = x
        self.left = None  # Initialize the left child as None
        self.right = None  # Initialize the right child as None

class Solution(object):
    def postorderTraversal(self, root):
        """
        Perform a post-order traversal of a binary tree.
        :type root: TreeNode
        :rtype: List[int]
        """
        result = []  # List to store the post-order traversal
        stack = []  # Stack to manage tree nodes
        visited = set()  # Set to keep track of visited nodes
        
        current = root  # Start with the root node
        
        while current or stack:
            # Traverse the left subtree
            while current:
                stack.append(current)  # Push the current node onto the stack
                current = current.left  # Move to the left child
            
            current = stack.pop()  # Pop a node from the stack

            # If the right child exists and has not been visited yet
            if current.right and current.right not in visited:
                stack.append(current)  # Push the current node back onto the stack
                current = current.right  # Move to the right child
            else:
                visited.add(current)  # Mark the node as visited
                result.append(current.val)  # Add the node's value to the result list
                current = None  # Reset the current node to continue with the stack

        return result

# Example usage:
# Constructing the binary tree
#        1
#         \
#          2
#         /
#        3

root = TreeNode(1)
root.right = TreeNode(2)
root.right.left = TreeNode(3)

solution = Solution()
result = solution.postorderTraversal(root)
print(result)  # Output: [3, 2, 1]
