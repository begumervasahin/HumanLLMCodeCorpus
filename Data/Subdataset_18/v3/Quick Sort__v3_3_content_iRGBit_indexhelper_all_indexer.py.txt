import sys
import string
from collections import Counter

# Default file paths
DEFAULT_FILE = 'files/sample.txt'
DEFAULT_STOP_WORDS = 'stop_words.txt'

def main():
    """
    Main function to read input files and call the index_them function.
    If no command-line arguments are provided, default files are used.
    """
    if len(sys.argv) > 3:
        print("\nUsage: python indexer.py <yourFile> <stopWords>")
        print(f"If no arguments are given, {DEFAULT_FILE} and {DEFAULT_STOP_WORDS} will be used as default files\n")
        sys.exit()

    input_file = sys.argv[1] if len(sys.argv) > 1 else DEFAULT_FILE
    stop_words_file = sys.argv[2] if len(sys.argv) > 2 else DEFAULT_STOP_WORDS
    
    print(f'Using {input_file} as the input file and {stop_words_file} as the stop words reference.\n')
    
    index_words(input_file, stop_words_file)

def index_words(input_file, stop_words_file):
    """
    Index the words from the input file excluding the stop words.
    
    :param input_file: Path to the input text file.
    :param stop_words_file: Path to the stop words file.
    """
    book_words = read_and_process_file(input_file)
    stop_words = read_stop_words(stop_words_file)
    filtered_words = filter_stop_words(book_words, stop_words)
    word_counts = count_words(filtered_words)
    print_word_counts(word_counts)

def read_and_process_file(file_path):
    """
    Read and process the input file.
    
    :param file_path: Path to the input text file.
    :return: List of processed words.
    """
    with open(file_path, 'r', encoding='unicode_escape') as file:
        words = file.read().lower().split()
        processed_words = [word.strip(string.punctuation) for word in words]
    return processed_words

def read_stop_words(file_path):
    """
    Read the stop words from the stop words file.
    
    :param file_path: Path to the stop words file.
    :return: List of stop words.
    """
    with open(file_path, 'r', encoding='utf-8-sig') as file:
        stop_words = file.read().splitlines()
    return stop_words

def filter_stop_words(words, stop_words):
    """
    Filter out the stop words from the list of words.
    
    :param words: List of words to be filtered.
    :param stop_words: List of stop words.
    :return: List of words excluding the stop words.
    """
    return [word for word in words if word not in stop_words]

def count_words(words):
    """
    Count the occurrences of each word in the list.
    
    :param words: List of words.
    :return: Counter object with word counts.
    """
    return Counter(words)

def print_word_counts(word_counts):
    """
    Print the word counts in sorted order.
    
    :param word_counts: Counter object with word counts.
    """
    sorted_word_counts = sorted(word_counts.items())
    for word, count in sorted_word_counts:
        print(f'{word}: {count}')

if __name__ == '__main__':
    main()
