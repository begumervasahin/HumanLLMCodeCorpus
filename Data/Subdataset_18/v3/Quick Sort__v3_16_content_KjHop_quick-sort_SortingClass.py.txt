class Sorting:
    def __init__(self):
        """
        Initialize an empty list to store numbers.
        """
        self.numbers = []

    def add_number(self, number):
        """
        Add a number to the list.

        Args:
            number (int): The number to add to the list.
        """
        self.numbers.append(number)

    def get_numbers(self):
        """
        Return the list of numbers.

        Returns:
            list: The list of numbers.
        """
        return self.numbers

    def print_numbers(self):
        """
        Print each element of the list.
        """
        for number in self.numbers:
            print(number)

    def quick_sort(self, arr):
        """
        Perform quick sort on the list.

        Args:
            arr (list): The list to sort.

        Returns:
            list: The sorted list.
        """
        if len(arr) <= 1:
            return arr
        else:
            pivot = arr[-1]
            less_than_pivot = [num for num in arr if num < pivot]
            equal_to_pivot = [num for num in arr if num == pivot]
            greater_than_pivot = [num for num in arr if num > pivot]
            return self.quick_sort(less_than_pivot) + equal_to_pivot + self.quick_sort(greater_than_pivot)

if __name__ == "__main__":
    sorting = Sorting()
    numbers_to_add = [34, 7, 23, 32, 5, 62, 32, 7]

    # Add numbers to the list
    for number in numbers_to_add:
        sorting.add_number(number)

    # Print the list
    print("Original list:")
    sorting.print_numbers()

    # Perform quick sort and print the sorted list
    sorted_list = sorting.quick_sort(sorting.get_numbers())
    print("\nSorted list:")
    for number in sorted_list:
        print(number)
