import queue

INF = int(1e9)

class Node:
    def __init__(self, vertex, weight):
        self.vertex = vertex
        self.weight = weight

    def __lt__(self, other):
        return self.weight <= other.weight

def dijkstra(graph, start, distances):
    """
    Implementation of Dijkstra's algorithm to find the shortest paths from a starting node.

    Args:
        graph (list): The adjacency list of the graph.
        start (int): The starting node.
        distances (list): The list to store the shortest distances from the start node.
    """
    pq = queue.PriorityQueue()
    pq.put(Node(start, 0))
    distances[start] = 0
    
    while not pq.empty():
        top = pq.get()
        u = top.vertex
        current_distance = top.weight
        
        for neighbor in graph[u]:
            if current_distance + neighbor.weight < distances[neighbor.vertex]:
                distances[neighbor.vertex] = current_distance + neighbor.weight
                pq.put(Node(neighbor.vertex, distances[neighbor.vertex]))

def bfs(graph, start, distances):
    """
    Implementation of Breadth-First Search to find the shortest path in an unweighted graph.

    Args:
        graph (list): The adjacency list of the graph.
        start (int): The starting node.
        distances (list): The list to store the shortest distances from the start node.
    """
    q = queue.Queue()
    q.put(start)
    distances[start] = 0
    
    while not q.empty():
        u = q.get()
        for neighbor in graph[u]:
            if distances[neighbor] == INF:
                distances[neighbor] = distances[u] + 1
                q.put(neighbor)

def problem_1():
    """
    Problem 1: Solve using Dijkstra's algorithm to determine the number of nodes reachable within a given time.
    """
    n = int(input("Enter number of nodes: "))
    e = int(input("Enter the end node: "))
    t = int(input("Enter the time limit: "))
    m = int(input("Enter number of edges: "))
    
    graph = [[] for _ in range(n + 1)]
    time = [INF for _ in range(n + 1)]
    
    for _ in range(m):
        a, b, w = map(int, input("Enter edge (a, b, weight): ").split())
        graph[b].append(Node(a, w))
    
    dijkstra(graph, e, time)
    count = sum(1 for i in range(1, n + 1) if time[i] <= t)
    print("Number of reachable nodes within time limit:", count)

def problem_2():
    """
    Problem 2: Handle multiple test cases, using Dijkstra's algorithm to find the shortest path costs between cities.
    """
    tc = int(input("Enter number of test cases: "))
    for _ in range(tc):
        n = int(input("Enter number of cities: "))
        graph = [[] for _ in range(n + 1)]
        cities = []
        
        for i in range(n):
            city = input("Enter city name: ").strip()
            cities.append(city)
            p = int(input(f"Enter number of neighbors for {city}: "))
            for _ in range(p):
                nr, c = map(int, input("Enter neighbor (nr, cost): ").split())
                graph[i + 1].append(Node(nr, c))
        
        r = int(input("Enter number of routes: "))
        for _ in range(r):
            cost = [INF for _ in range(n + 1)]
            source, destination = input("Enter source and destination: ").split()
            start = cities.index(source) + 1
            end = cities.index(destination) + 1
            dijkstra(graph, start, cost)
            print(f"Cost from {source} to {destination}:", cost[end])
        input()  # Consume the extra input

def problem_3():
    """
    Problem 3: Find the minimum time required to travel between two nodes, considering a set of key nodes.
    """
    n, m, k, x = map(int, input("Enter n, m, k, x: ").split())
    ks = list(map(int, input("Enter key nodes: ").split()))
    
    graph = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v, d = map(int, input("Enter edge (u, v, distance): ").split())
        graph[u].append(Node(v, d))
        graph[v].append(Node(u, d))
    
    a, b = map(int, input("Enter nodes a and b: ").split())
    
    distA = [INF for _ in range(n + 1)]
    distB = [INF for _ in range(n + 1)]
    
    dijkstra(graph, a, distA)
    dijkstra(graph, b, distB)
    
    min_time = min((distA[k] + distB[k] for k in ks if distB[k] <= x), default=INF)
    print("Minimum time required:", min_time if min_time < INF else -1)

def problem_4():
    """
    Problem 4: Find the maximum shortest path between two nodes considering all nodes.
    """
    t = int(input("Enter number of test cases: "))
    for c in range(t):
        n = int(input("Enter number of nodes: "))
        graph = [[] for _ in range(n)]
        r = int(input("Enter number of edges: "))
        
        for _ in range(r):
            u, v = map(int, input("Enter edge (u, v): ").split())
            graph[u].append(v)
            graph[v].append(u)
        
        s, d = map(int, input("Enter nodes s and d: ").split())
        
        distS = [INF for _ in range(n)]
        distD = [INF for _ in range(n)]
        
        bfs(graph, s, distS)
        bfs(graph, d, distD)
        
        max_time = max(distS[i] + distD[i] for i in range(n) if distS[i] != INF and distD[i] != INF)
        print(f'Case {c + 1}: {max_time}')
