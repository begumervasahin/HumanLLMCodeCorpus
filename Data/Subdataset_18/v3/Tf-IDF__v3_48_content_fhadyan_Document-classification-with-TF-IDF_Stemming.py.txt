import os
from nltk.stem import PorterStemmer
from UniqueWord import UniqueWord

class Stemming:
    def __init__(self, source_path="output/training/stopword/", out_path="output/training/stem/"):
        self.source_path = source_path
        self.out_path = out_path
        self.ensure_source_path_not_empty()

    def ensure_source_path_not_empty(self):
        """Generate unique words if the source path directory is empty."""
        if not os.listdir(self.source_path):
            unique_word_generator = UniqueWord()
            unique_word_generator.output()

    def stem_file(self, input_filepath, output_filepath):
        """Stem words in a single file and write the results to the output file."""
        stemmer = PorterStemmer()

        with open(input_filepath, 'r', encoding='utf-8') as infile:
            lines = infile.readlines()
        
        with open(output_filepath, 'w', encoding='utf-8') as outfile:
            for line in lines:
                line = line.strip()
                if len(line) > 5:  # Ensure the line is long enough to avoid index errors
                    stemmed_word = stemmer.stem(line[:-5])
                    outfile.write(f"{stemmed_word} - {line[-2]}\n")

    def stem(self):
        """Process each file in the source path directory and stem their contents."""
        for filename in os.listdir(self.source_path):
            if filename.endswith('.txt'):
                input_filepath = os.path.join(self.source_path, filename)
                output_filepath = os.path.join(self.out_path, filename)
                self.stem_file(input_filepath, output_filepath)

if __name__ == "__main__":
    stemming = Stemming()
    stemming.stem()
