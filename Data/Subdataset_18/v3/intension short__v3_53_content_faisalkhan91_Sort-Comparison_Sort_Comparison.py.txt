import random
import timeit

def print_array(arr):
    """Prints the array elements separated by spaces."""
    print(' '.join(map(str, arr)))

def insertion_sort(arr):
    """Sorts the array using the insertion sort algorithm."""
    for i in range(1, len(arr)):
        current_value = arr[i]
        position = i

        # Shift elements of arr[0..i-1], that are greater than current_value, to one position ahead
        while position > 0 and arr[position - 1] > current_value:
            arr[position] = arr[position - 1]
            position -= 1

        # Place the current_value in its correct position
        arr[position] = current_value

def merge_sort(arr):
    """Sorts the array using the merge sort algorithm."""
    if len(arr) > 1:
        mid = len(arr) // 2
        left_half = arr[:mid]
        right_half = arr[mid:]

        # Recursively sort the halves
        merge_sort(left_half)
        merge_sort(right_half)

        # Indices for iterating through the left_half, right_half, and the original array
        i = j = k = 0

        # Merge the sorted halves back into the original array
        while i < len(left_half) and j < len(right_half):
            if left_half[i] < right_half[j]:
                arr[k] = left_half[i]
                i += 1
            else:
                arr[k] = right_half[j]
                j += 1
            k += 1

        # Copy any remaining elements of left_half, if any
        while i < len(left_half):
            arr[k] = left_half[i]
            i += 1
            k += 1

        # Copy any remaining elements of right_half, if any
        while j < len(right_half):
            arr[k] = right_half[j]
            j += 1
            k += 1

def unh_sort(arr):
    """Sorts the array using a custom sorting algorithm (possibly a bubble sort variant)."""
    for i in range(len(arr) - 1, 0, -1):
        for j in range(1, i):
            if arr[j] > arr[j + 1]:
                # Swap arr[j] and arr[j + 1] if they are in the wrong order
                arr[j], arr[j + 1] = arr[j + 1], arr[j]

def measure_sorting_time(sort_function, arr):
    """Measures the time taken by a sorting function to sort an array."""
    start_time = timeit.default_timer()
    sort_function(arr)
    end_time = timeit.default_timer()
    return end_time - start_time

def main():
    # Generate a random array of 10,000 unique elements
    array_size = 10000
    random_array = random.sample(range(array_size), array_size)

    # Create separate copies for each sorting algorithm
    insertion_randoms = random_array.copy()
    merge_randoms = random_array.copy()
    unh_randoms = random_array.copy()

    # Display the initial array (optional, can be commented out for large arrays)
    print("Initial Array:")
    print_array(random_array)

    # Measure and print the time taken by each sorting algorithm
    insertion_sort_time = measure_sorting_time(insertion_sort, insertion_randoms)
    print("Insertion Sort Time: {:.5f} seconds".format(insertion_sort_time))

    merge_sort_time = measure_sorting_time(merge_sort, merge_randoms)
    print("Merge Sort Time: {:.5f} seconds".format(merge_sort_time))

    unh_sort_time = measure_sorting_time(unh_sort, unh_randoms)
    print("UNH Sort Time: {:.5f} seconds".format(unh_sort_time))

if __name__ == "__main__":
    main()
