import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
import scipy.cluster.hierarchy as sch
from sklearn.cluster import AgglomerativeClustering

def load_data(file_path):
    """Load the dataset from a CSV file."""
    return pd.read_csv(file_path)

def extract_features(data, columns):
    """
    Extract specified columns for clustering.

    Args:
        data (DataFrame): The dataset containing all features.
        columns (list): List of column indices to extract.

    Returns:
        np.ndarray: The extracted feature matrix.
    """
    return data.iloc[:, columns].values

def plot_dendrogram(X):
    """
    Plot a dendrogram to visualize hierarchical clustering.

    Args:
        X (np.ndarray): The data matrix used for clustering.
    """
    plt.figure(figsize=(10, 7))
    dendrogram = sch.dendrogram(sch.linkage(X, method='ward'))
    plt.title('Dendrogram')
    plt.xlabel('Customers')
    plt.ylabel('Euclidean Distances')
    plt.show()

def apply_agglomerative_clustering(X, n_clusters=5):
    """
    Apply Agglomerative Clustering to the data.

    Args:
        X (np.ndarray): The data matrix to cluster.
        n_clusters (int): The number of clusters to form.

    Returns:
        np.ndarray: Cluster labels for each point in the dataset.
    """
    hc = AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage='ward')
    return hc.fit_predict(X)

def plot_clusters(X, cluster_labels, cluster_colors=None):
    """
    Visualize the clusters of customers.

    Args:
        X (np.ndarray): The data matrix containing the features.
        cluster_labels (np.ndarray): Cluster labels for each point.
        cluster_colors (list): List of colors to use for the clusters.
    """
    plt.figure(figsize=(10, 7))
    
    if cluster_colors is None:
        cluster_colors = ['red', 'blue', 'green', 'cyan', 'magenta']

    for i, color in enumerate(cluster_colors):
        plt.scatter(X[cluster_labels == i, 0], X[cluster_labels == i, 1], 
                    s=100, c=color, label=f'Cluster {i+1}')
    
    plt.title('Clusters of Customers')
    plt.xlabel('Annual Income (k$)')
    plt.ylabel('Spending Score (1-100)')
    plt.legend()
    plt.show()

def main(file_path='Mall_Customers.csv', feature_columns=[3, 4], n_clusters=5):
    """
    Main function to perform hierarchical clustering on customer data.

    Args:
        file_path (str): Path to the CSV file containing the dataset.
        feature_columns (list): List of columns to use for clustering.
        n_clusters (int): Number of clusters to form.
    """
    # Load the dataset
    data = load_data(file_path)
    
    # Select the features for clustering
    X = extract_features(data, feature_columns)
    
    # Plot the Dendrogram to find the optimal number of clusters
    plot_dendrogram(X)
    
    # Apply Agglomerative Clustering
    cluster_labels = apply_agglomerative_clustering(X, n_clusters)
    
    # Visualize the clusters
    plot_clusters(X, cluster_labels)

if __name__ == "__main__":
    main()
