import torch
import torch.nn as nn
from torch.nn import init

class YourModel(nn.Module):
    """
    A simple neural network model with a single linear layer.
    """
    def __init__(self):
        super(YourModel, self).__init__()
        # Define a linear layer with 10 input features and 5 output features
        self.linear_layer = nn.Linear(in_features=10, out_features=5)

    def forward(self, x):
        """
        Forward pass of the model.
        Args:
            x (Tensor): Input tensor.
        Returns:
            Tensor: Output tensor after applying the linear layer.
        """
        return self.linear_layer(x)

def initialize_weights(m):
    """
    Initialize weights of the model using Xavier uniform initialization.
    Args:
        m (nn.Module): A layer in the neural network model.
    """
    if isinstance(m, nn.Linear):
        init.xavier_uniform_(m.weight.data)

def main():
    """
    Main function to create the model, apply weight initialization,
    and demonstrate an example usage of the model.
    """
    # Create an instance of the model
    model = YourModel()

    # Apply the weight initialization function to the model
    model.apply(initialize_weights)

    # Print the initialized weights of the linear layers
    for name, param in model.named_parameters():
        if 'weight' in name:
            print(f'Initialized weights for {name} with shape {param.shape}')

    # Example usage: create a random input tensor and pass it through the model
    example_input = torch.randn(1, 10)  # Batch size of 1, 10 input features
    output = model(example_input)
    print(f'Output of the model: {output}')

if __name__ == "__main__":
    main()
