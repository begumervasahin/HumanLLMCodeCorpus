import numpy as np
import timeit
import matplotlib.pyplot as plt

# Use the 'ggplot' style for better visualization
plt.style.use('ggplot')

def linear_search(arr):
    """
    Perform a linear search on the array to find the point where the order breaks.
    """
    comparisons = 0
    previous_element = -np.inf

    for current_element in arr:
        if current_element < previous_element:
            return comparisons
        comparisons += 1
        previous_element = current_element

def binary_search(arr, start_index=0):
    """
    Perform a binary search on the array to find the pivot element index.
    """
    middle_index = len(arr) // 2
    pivot = arr[middle_index]

    if pivot == arr[0]:
        return start_index + 1
    elif pivot < arr[0]:
        return binary_search(arr[:middle_index], start_index)
    else:
        return binary_search(arr[middle_index:], start_index + middle_index)

def time_function(func, *args, **kwargs):
    """
    Measure execution time of the given function.
    """
    def wrapped():
        return func(*args, **kwargs)
    return wrapped

def measure_search_times(array_sizes, num_iterations=10, num_trials=100):
    """
    Measure and return the average execution times for linear and binary search
    over multiple array sizes.
    """
    linear_times = []
    binary_times = []

    for size in array_sizes:
        total_time_binary = 0
        total_time_linear = 0

        for _ in range(num_trials):
            arr = np.roll(np.arange(size), np.random.randint(size))
            timed_linear_search = time_function(linear_search, arr)
            timed_binary_search = time_function(binary_search, arr)

            total_time_binary += timeit.timeit(timed_binary_search, number=num_iterations)     
            total_time_linear += timeit.timeit(timed_linear_search, number=num_iterations)     

        avg_time_binary = total_time_binary / num_trials
        avg_time_linear = total_time_linear / num_trials

        binary_times.append(avg_time_binary)
        linear_times.append(avg_time_linear)

    return linear_times, binary_times

def normalize_times(times, max_value):
    """
    Normalize the list of times by the provided maximum value.
    """
    return np.array(times) / max_value

def plot_search_times(array_sizes, linear_times, binary_times):
    """
    Plot the search times for linear and binary search along with reference trends.
    """
    max_linear_time = max(linear_times)
    normalized_linear_times = normalize_times(linear_times, max_linear_time)
    normalized_binary_times = normalize_times(binary_times, max_linear_time)
    normalized_linear_trend = normalize_times(array_sizes, max(array_sizes))
    normalized_logarithmic_trend = normalize_times(np.log2(array_sizes), max(array_sizes))

    plt.plot(array_sizes, normalized_linear_times, '-o', label='Linear Search')
    plt.plot(array_sizes, normalized_binary_times, '-o', label='Binary Search')
    plt.plot(array_sizes, normalized_linear_trend, '--k', label='Linear')
    plt.plot(array_sizes, normalized_logarithmic_trend, '--k', label='Logarithmic')
    
    plt.legend()
    plt.xlabel('Array Size (n)')
    plt.ylabel('Normalized Time')
    plt.title('Linear Search vs Binary Search Time Complexity')
    plt.show()

if __name__ == '__main__':
    array_sizes = [10, 50, 100, 200, 500, 1000, 5000, 10000]
    linear_times, binary_times = measure_search_times(array_sizes)
    plot_search_times(array_sizes, linear_times, binary_times)
