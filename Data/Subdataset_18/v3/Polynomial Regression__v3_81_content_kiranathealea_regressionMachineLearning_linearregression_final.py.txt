import numpy as np
import pandas as pd
import os
import glob
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error, r2_score

def load_data(folder):
    """Load year and crop data from a CSV file."""
    data_year_x = pd.read_csv(folder, usecols=['Year'])
    data_crop_y = pd.read_csv(folder, usecols=['Crop'])
    return data_year_x.values.reshape(-1, 1), data_crop_y.values.reshape(-1, 1)

def perform_regression(x, y):
    """Perform linear regression and return model, predictions, RMSE, and R2 score."""
    model = LinearRegression()
    model.fit(x, y)
    y_pred = model.predict(x)
    rmse = mean_squared_error(y, y_pred)
    r2 = r2_score(y, y_pred)
    return model, y_pred, rmse, r2

def predict_future(model, years):
    """Predict future values using the regression model."""
    future_predictions = np.round(model.predict(years.reshape(-1, 1)), 4)
    return future_predictions

def combine_data(x, y_pred, future_years, future_predictions):
    """Combine original and future data."""
    x_combined = np.concatenate((x, future_years))
    y_combined = np.concatenate((y_pred, future_predictions))
    return x_combined, y_combined

def extend_data(original_data):
    """Extend the data by repeating the first 6 values for future predictions."""
    return np.concatenate((original_data, original_data[:6]))

def create_dataframe(city, x_combined, y_combined_print, wilayah, kode):
    """Create a dataframe with the combined data."""
    wilayah_extended = extend_data(wilayah)
    kode_extended = extend_data(kode)
    result_df = pd.DataFrame({
        'Entity': wilayah_extended,
        'Code': kode_extended,
        'Year': x_combined.ravel(),
        'crop(tonnes per hectare)': y_combined_print.ravel()
    })
    return result_df

def save_to_csv(df, path):
    """Save dataframe to a CSV file."""
    df.to_csv(path, index=False)

def main():
    daftar_kota = [
        'Bandung', 'Bantul', 'Bekasi', 'Bogor', 'Ciamis', 'Cianjur', 'Cirebon', 'DIY', 'Garut', 
        'Gunung Kidul', 'Indramayu', 'Karawang', 'Kulon Progo', 'Kuningan', 'Majalengka', 
        'Purwakarta', 'Sleman', 'Subang', 'Sukabumi', 'Sumedang', 'Tasikmalaya'
    ]

    for city in daftar_kota:
        folder = f'/home/leathea/Downloads/Machine-Learning/PolynomialRegression/tomato_csv_perdaerah/{city}.csv'
        x, y = load_data(folder)
        
        model, y_pred, rmse, r2 = perform_regression(x, y)
        
        print(f'The RMSE of the linear regression model for {city} is {rmse}')
        print(f'The R2 score of the linear regression model for {city} is {r2}')
        
        future_years = np.array([2015, 2016, 2017, 2018, 2019, 2020])
        future_predictions = predict_future(model, future_years)
        
        x_combined, y_combined = combine_data(x, y_pred, future_years, future_predictions)
        y_combined_print = np.concatenate((y, future_predictions))
        
        wilayah = pd.read_csv(folder, usecols=['Entity']).values.ravel()
        kode = pd.read_csv(folder, usecols=['Code']).values.ravel()
        
        result_df = create_dataframe(city, x_combined, y_combined_print, wilayah, kode)
        save_to_csv(result_df, f'/home/leathea/Downloads/Machine-Learning/PolynomialRegression/hasiltomato/linear/{city}_tomato.csv')
    
    output_dir = "/home/leathea/Downloads/Machine-Learning/PolynomialRegression/hasiltomato/linear"
    os.chdir(output_dir)
    all_filenames = glob.glob('*.csv')
    
    combined_df = pd.concat([pd.read_csv(f) for f in all_filenames])
    save_to_csv(combined_df, "/home/leathea/Downloads/Machine-Learning/PolynomialRegression/predicted_tomato.csv")
    
    print('All done')

if __name__ == "__main__":
    main()
