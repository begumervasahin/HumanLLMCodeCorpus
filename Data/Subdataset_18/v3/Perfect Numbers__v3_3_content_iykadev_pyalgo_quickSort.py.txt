def quick_sort(alist):
    """
    Sorts alist in place using the quicksort algorithm.
    """
    quick_sort_helper(alist, 0, len(alist) - 1)

def quick_sort_helper(alist, first, last):
    """
    Recursive helper function for the quicksort algorithm.
    """
    if first < last:
        split_point = partition(alist, first, last)
        quick_sort_helper(alist, first, split_point - 1)
        quick_sort_helper(alist, split_point + 1, last)

def partition(alist, first, last):
    """
    Partitions the list and returns the index of the split point.
    """
    pivot_value = alist[first]
    left_mark = first + 1
    right_mark = last

    while True:
        # Move left_mark to the right until an element larger than the pivot is found
        while left_mark <= right_mark and alist[left_mark] <= pivot_value:
            left_mark += 1

        # Move right_mark to the left until an element smaller than the pivot is found
        while left_mark <= right_mark and alist[right_mark] >= pivot_value:
            right_mark -= 1

        # If right_mark crosses left_mark, break the loop
        if right_mark < left_mark:
            break
        else:
            # Swap the elements at left_mark and right_mark
            alist[left_mark], alist[right_mark] = alist[right_mark], alist[left_mark]

    # Swap the pivot element with the element at right_mark
    alist[first], alist[right_mark] = alist[right_mark], alist[first]

    return right_mark

# Example usage
alist = [54, 26, 93, 17, 77, 31, 44, 55, 20]
quick_sort(alist)
print(alist)
