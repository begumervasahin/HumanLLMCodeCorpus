def selection_sort(alist):
    """
    Sorts a list in ascending order using the selection sort algorithm.
    
    Args:
    alist (list): The list to be sorted.
    """
    n = len(alist)
    
    # Traverse through the list
    for fillslot in range(n - 1, 0, -1):
        position_of_max = 0
        
        # Find the maximum element in the unsorted part of the list
        for location in range(1, fillslot + 1):
            if alist[location] > alist[position_of_max]:
                position_of_max = location
        
        # Swap the found maximum element with the last element in the unsorted part
        alist[fillslot], alist[position_of_max] = alist[position_of_max], alist[fillslot]

# Test the selection sort function
test_list = [54, 26, 93, 17, 77, 31, 44, 55, 20]
selection_sort(test_list)
print(test_list)  # Output should be a sorted list
