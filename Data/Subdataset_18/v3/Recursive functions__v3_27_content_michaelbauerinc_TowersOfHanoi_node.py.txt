class Node:
    def __init__(self, value, next_node=None):
        """
        Initialize a Node with a value, optional next node, and other attributes.
        
        Args:
            value: The value to store in the node.
            next_node (Node, optional): The next node in the sequence. Defaults to None.
        """
        self.value = value
        self.next_node = next_node
        self.previous_node = None
        self.image = None

    def set_next_node(self, next_node):
        """Set the next node."""
        self.next_node = next_node

    def get_next_node(self):
        """Get the next node."""
        return self.next_node

    def set_previous_node(self, previous_node):
        """Set the previous node."""
        self.previous_node = previous_node

    def get_previous_node(self):
        """Get the previous node."""
        return self.previous_node

    def get_value(self):
        """Get the value of the node."""
        return self.value

    def assign_image(self, image):
        """Assign an image to the node."""
        self.image = image

    def get_image(self):
        """Get the image assigned to the node."""
        return self.image

def main():
    """
    Main function to create nodes, link them, assign images, and print their details.
    """
    # Create nodes
    node1 = Node(1)
    node2 = Node(2)
    node3 = Node(3)

    # Link nodes
    node1.set_next_node(node2)
    node2.set_previous_node(node1)
    node2.set_next_node(node3)
    node3.set_previous_node(node2)

    # Assign images
    node1.assign_image('image1.png')
    node2.assign_image('image2.png')
    node3.assign_image('image3.png')

    # Print node values and images
    current_node = node1
    while current_node:
        print(f'Node Value: {current_node.get_value()}')
        print(f'Node Image: {current_node.get_image()}')
        current_node = current_node.get_next_node()

if __name__ == '__main__':
    main()
