class PrefixMatcher:
    def __init__(self, options):
        """
        Initialize the PrefixMatcher with a list of options.

        Args:
            options (list): List of options to match against.
        """
        self._opts = options

    def matchunique(self, value):
        """
        Match the given value uniquely against the options.

        Args:
            value (str): The value to match.

        Returns:
            str: The unique matched option.

        Raises:
            Exception: If no matches or multiple matches are found.
        """
        result = self.match(value)
        if len(result) == 0:
            raise Exception(f"'{value}' did not match any options.")
        elif len(result) > 1:
            raise Exception(f"'{value}' is ambiguous. Please clarify further. Available: {', '.join(sorted(result))}")
        return result[0]

    def match(self, value):
        """
        Match the given value against the options.

        Args:
            value (str): The value to match.

        Returns:
            list: A list of matched options.
        """
        return [option for option in self._opts if option.startswith(value)]

if __name__ == "__main__":
    # Initialize the PrefixMatcher with a list of options
    options = ["import", "install", "foo"]
    pm = PrefixMatcher(options)

    # Test the match method
    matches = pm.match("i")
    print("Matches for 'i':", matches)  # Output: ['import', 'install']

    # Test the matchunique method with ambiguous input
    try:
        unique_match = pm.matchunique("i")
        print("Unique match for 'i':", unique_match)
    except Exception as e:
        print("Error:", e)  # Output: "'i' is ambiguous. Please clarify further. Available: import, install"

    # Test the matchunique method with a unique match
    try:
        unique_match = pm.matchunique("im")
        print("Unique match for 'im':", unique_match)  # Output: 'import'
    except Exception as e:
        print("Error:", e)
