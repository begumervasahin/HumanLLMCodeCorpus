import random
from time import time

# Set the maximum range for generating random numbers
MAX_RANGE = 10001

# Generate an unsorted list of random numbers
unsorted_list = random.sample(range(1, MAX_RANGE), MAX_RANGE - 1)

def bubble_sort(input_list):
    """
    Sorts a list using the bubble sort algorithm.

    Parameters:
    input_list (list): The list to be sorted.

    Returns:
    list: The sorted list.
    """
    n = len(input_list)
    for i in range(n):
        swapped = False
        for j in range(0, n - i - 1):
            if input_list[j] > input_list[j + 1]:
                input_list[j], input_list[j + 1] = input_list[j + 1], input_list[j]
                swapped = True
        if not swapped:
            break
    return input_list

def measure_time(sort_function, data):
    """
    Measures the time taken by a sorting function to sort a list.

    Parameters:
    sort_function (function): The sorting function.
    data (list): The list to be sorted.

    Returns:
    float: The time taken to sort the list.
    """
    start = time()
    sort_function(data)
    return time() - start

if __name__ == "__main__":
    # Copy the unsorted list for bubble sort
    bubble_sort_list = unsorted_list.copy()

    # Measure time taken by bubble sort
    bubble_sort_time = measure_time(bubble_sort, bubble_sort_list)
    print(f"Bubble sort algorithm ends in {bubble_sort_time:.3f} seconds to sort a list of {MAX_RANGE - 1} items.")

    # Copy the unsorted list for built-in sort
    timsort_list = unsorted_list.copy()

    # Measure time taken by Python's built-in sort() method (Timsort)
    timsort_time = measure_time(timsort_list.sort, timsort_list)
    print(f"System Python sort() Timsort algorithm ends in {timsort_time:.3f} seconds to sort a list of {MAX_RANGE - 1} items.")
