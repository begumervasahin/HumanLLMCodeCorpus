def selection_sort(arr):
    """Sorts an array using the selection sort algorithm."""
    for i in range(len(arr)):
        # Assume the current position holds the smallest element
        min_idx = i
        # Find the smallest element in the remaining unsorted portion of the array
        for j in range(i + 1, len(arr)):
            if arr[min_idx] > arr[j]:
                min_idx = j
        # Swap the found minimum element with the element at the current position
        arr[i], arr[min_idx] = arr[min_idx], arr[i]
    return arr

def print_sorted_array(arr):
    """Prints the sorted array."""
    sorted_arr = selection_sort(arr)
    print("Sorted array:", sorted_arr)

# Example usage
if __name__ == "__main__":
    my_arr = [8, 5, 2, 6, 9, 3, 1, 4, 0, 7]
    print("Original array:", my_arr)
    print_sorted_array(my_arr)
