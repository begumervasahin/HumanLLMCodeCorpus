from collections import defaultdict, deque

class EdmondsKarp:
    def __init__(self, graph, num_vertices):
        """
        Initialize the Edmonds-Karp algorithm with a graph and the number of vertices.

        :param graph: The adjacency matrix of the graph.
        :param num_vertices: The number of vertices in the graph.
        """
        self.graph = graph
        self.num_vertices = num_vertices

    def bfs(self, source, sink, parent):
        """
        Perform Breadth-First Search (BFS) to find an augmenting path.

        :param source: The source vertex.
        :param sink: The sink vertex.
        :param parent: List to store the path.
        :return: Boolean indicating if an augmenting path was found.
        """
        visited = [False] * self.num_vertices
        queue = deque([source])
        visited[source] = True

        while queue:
            u = queue.popleft()

            for v in range(self.num_vertices):
                if not visited[v] and self.graph[u][v] > 0:
                    queue.append(v)
                    visited[v] = True
                    parent[v] = u
                    if v == sink:
                        return True

        return False

    def edmonds_karp(self, source, sink):
        """
        Run the Edmonds-Karp algorithm to find the maximum flow from source to sink.

        :param source: The source vertex.
        :param sink: The sink vertex.
        :return: The maximum flow value.
        """
        parent = [-1] * self.num_vertices
        max_flow = 0

        while self.bfs(source, sink, parent):
            path_flow = float('Inf')
            s = sink

            while s != source:
                path_flow = min(path_flow, self.graph[parent[s]][s])
                s = parent[s]

            max_flow += path_flow
            v = sink

            while v != source:
                u = parent[v]
                self.graph[u][v] -= path_flow
                self.graph[v][u] += path_flow
                v = parent[v]

        return max_flow

    def min_cut(self, source):
        """
        Find the vertices on the source side of the minimum cut.

        :param source: The source vertex.
        :return: List of vertices on the source side of the minimum cut.
        """
        visited = [False] * self.num_vertices
        queue = deque([source])
        visited[source] = True
        min_cut_vertices = []

        while queue:
            u = queue.popleft()
            min_cut_vertices.append(u)

            for v in range(self.num_vertices):
                if not visited[v] and self.graph[u][v] > 0:
                    queue.append(v)
                    visited[v] = True

        return min_cut_vertices

def read_graph(filename):
    """
    Read the graph from a file and create an adjacency matrix.

    :param filename: The name of the file containing the graph.
    :return: The graph as an adjacency matrix and the number of vertices.
    """
    with open(filename, 'r') as file:
        first_line = file.readline().strip().split()
        num_vertices = int(first_line[0])
        num_edges = int(first_line[1])

        graph = [[0] * num_vertices for _ in range(num_vertices)]

        for _ in range(num_edges):
            line = file.readline().strip().split()
            u, v, weight = map(int, line)
            graph[u][v] = weight
            graph[v][u] = weight  # Assuming undirected graph

    return graph, num_vertices

def main():
    input_file = 'mincut_input/XXXX.in'  # Replace with your input file path
    output_file = 'output_file.txt'  # Replace with your output file path

    graph, num_vertices = read_graph(input_file)
    source = 0  # Assuming the source vertex is 0, adjust if necessary
    sink = num_vertices - 1  # Assuming the sink vertex is the last vertex, adjust if necessary

    ek = EdmondsKarp(graph, num_vertices)
    max_flow = ek.edmonds_karp(source, sink)
    min_cut_vertices = ek.min_cut(source)

    with open(output_file, 'w') as file:
        file.write(f"{len(min_cut_vertices)}\n")
        file.write(" ".join(map(str, min_cut_vertices)) + "\n")
        file.write(f"{max_flow}\n")

    print("Successfully wrote output to", output_file)

if __name__ == "__main__":
    main()
