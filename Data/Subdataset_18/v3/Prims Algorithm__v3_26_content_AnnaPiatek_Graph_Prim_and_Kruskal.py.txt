import graphviz as gv
from graphviz import Graph

# Define styles for the graph, nodes, and edges
styles = {
    'graph': {
        'label': 'Graph',
        'fontsize': '16',
        'fontcolor': 'black',
        'bgcolor': '#ffffff',
    },
    'nodes': {
        'shape': 'circle',
        'fontcolor': 'black',
        'color': 'black',
        'style': 'filled',
        'fillcolor': '#ffffff',
    },
    'edges': {
        'color': 'black',
        'fontcolor': 'black',
        'fontsize': '12',
    }
}

def apply_styles(graph, styles):
    """Apply styles to the graph."""
    graph.graph_attr.update(styles.get('graph', {}))
    graph.node_attr.update(styles.get('nodes', {}))
    graph.edge_attr.update(styles.get('edges', {}))
    return graph

class Graph_0:
    """A class representing a graph."""
    
    def __init__(self):
        self.edges = []
    
    def add_edge(self, node1, node2, weight):
        """Add an edge to the graph."""
        self.edges.append((node1, node2, weight))
    
    def branches_building(self):
        """
        This function should implement Prim's or Kruskal's algorithm to find the Minimum Spanning Tree (MST).
        For simplicity, it just returns the edges sorted by weight.
        """
        return sorted(self.edges, key=lambda x: x[2])

class Drawing:
    """A class to draw the graph and highlight paths."""
    
    def __init__(self, edges):
        """Initialize with the edges of the graph."""
        self.edges = edges
    
    def draw_initial_graph(self):
        """Draw the initial graph with all edges."""
        graph = Graph(format='png')
        for edge in self.edges:
            graph.edge(edge[0], edge[1], label=str(edge[2]), color='black')
        graph = apply_styles(graph, styles)
        graph.render(filename='graph_initial', view=True)

    def draw_highlighted_paths(self, path):
        """Draw the graph highlighting the edges in the given path."""
        path_edges = []
        
        for step, edge in enumerate(path, start=1):
            graph = Graph(format='png')
            path_edges.append([str(edge[0]), str(edge[1]), str(edge[2])])
            
            for edge in self.edges:
                node_0, node_1, weight = map(str, edge)
                color = 'red' if [node_0, node_1, weight] in path_edges or [node_1, node_0, weight] in path_edges else 'black'
                graph.edge(node_0, node_1, label=weight, color=color)
            
            graph = apply_styles(graph, styles)
            graph.render(filename=f'graph_{step}', view=True)

if __name__ == "__main__":
    # Create a graph and add edges
    g = Graph_0()
    edges = [
        ('A', 'B', 5), ('B', 'E', 4), ('E', 'D', 1),
        ('D', 'B', 3), ('B', 'C', 2), ('C', 'D', 3),
        ('C', 'A', 6)
    ]
    for edge in edges:
        g.add_edge(*edge)
    
    # Build the Minimum Spanning Tree (MST)
    mst = g.branches_building()
    print(mst)
    
    # Draw the graph with the MST highlighted
    d = Drawing(edges)
    d.draw_initial_graph()
    d.draw_highlighted_paths(mst)
