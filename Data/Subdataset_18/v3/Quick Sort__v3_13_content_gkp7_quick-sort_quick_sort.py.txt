def quicksort(alist):
    """
    Sorts a list in place using the Quick Sort algorithm.

    Args:
    alist (list): The list to be sorted.
    """
    quicksort_helper(alist, 0, len(alist) - 1)

def quicksort_helper(alist, first, last):
    """
    Helper function to recursively sort sublists.

    Args:
    alist (list): The list to be sorted.
    first (int): The starting index of the sublist.
    last (int): The ending index of the sublist.
    """
    if first < last:
        # Partition the list and get the split point
        split_point = partition(alist, first, last)
        
        # Recursively sort the two halves
        quicksort_helper(alist, first, split_point - 1)
        quicksort_helper(alist, split_point + 1, last)

def partition(alist, first, last):
    """
    Partition the sublist and return the split point.

    Args:
    alist (list): The list to be partitioned.
    first (int): The starting index of the sublist.
    last (int): The ending index of the sublist.

    Returns:
    int: The split point index.
    """
    pivot_value = alist[first]  # Choose the first element as the pivot
    left_mark = first + 1
    right_mark = last
    done = False

    while not done:
        # Move left_mark to the right as long as it points to a value less than the pivot
        while left_mark <= right_mark and alist[left_mark] <= pivot_value:
            left_mark += 1

        # Move right_mark to the left as long as it points to a value greater than the pivot
        while alist[right_mark] >= pivot_value and right_mark >= left_mark:
            right_mark -= 1

        # If left_mark and right_mark cross, we are done
        if right_mark < left_mark:
            done = True
        else:
            # Swap the elements at left_mark and right_mark
            alist[left_mark], alist[right_mark] = alist[right_mark], alist[left_mark]

    # Swap the pivot value with the element at right_mark
    alist[first], alist[right_mark] = alist[right_mark], alist[first]

    # Return the split point
    return right_mark
