import random

class Vertex:
    def __init__(self, label):
        """
        Initialize a vertex with a label and a list of connections.
        Each connection is a tuple of (connected_vertex_label, weight).
        
        Args:
            label (int): The label of the vertex.
        """
        self.label = label
        self.connections = []

    def add_connection(self, vertex_label, weight):
        """
        Add a connection to another vertex with a specified weight.
        
        Args:
            vertex_label (int): The label of the connected vertex.
            weight (int): The weight of the edge connecting the vertices.
        """
        self.connections.append((vertex_label, weight))

    def remove_connection(self, vertex_label):
        """
        Remove a connection to another vertex by its label.
        
        Args:
            vertex_label (int): The label of the vertex to disconnect from.
        """
        self.connections = [conn for conn in self.connections if conn[0] != vertex_label]

    def has_connection(self, vertex_label):
        """
        Check if a connection to the specified vertex exists.
        
        Args:
            vertex_label (int): The label of the vertex to check the connection with.
        
        Returns:
            bool: True if a connection exists, False otherwise.
        """
        return any(conn[0] == vertex_label for conn in self.connections)

def defined_degree_graph(num_vertices, degree):
    """
    Generate a random graph with a specified number of vertices and degree per vertex.

    Each vertex is connected to 'degree / 2' neighbors on both sides in a cyclic manner.
    The graph is randomized by shuffling vertex indices before connecting them.
    
    Args:
        num_vertices (int): The number of vertices in the graph.
        degree (int): The degree of each vertex.
    
    Returns:
        List[Vertex]: A list of vertices representing the graph.
    """
    vertices = [Vertex(i) for i in range(num_vertices)]
    edges = []

    # Shuffle vertices to randomize connections
    vertex_indices = list(range(num_vertices))
    random.shuffle(vertex_indices)

    for i in range(num_vertices):
        _connect_neighbors(vertices, vertex_indices, i, num_vertices, degree, edges)
    
    _write_edges_to_file(edges, './edges.txt')

    return vertices

def _connect_neighbors(vertices, vertex_indices, i, num_vertices, degree, edges):
    """
    Connect each vertex to its neighbors to achieve the desired degree.
    
    Args:
        vertices (List[Vertex]): The list of vertices.
        vertex_indices (List[int]): The shuffled list of vertex indices.
        i (int): The index of the current vertex.
        num_vertices (int): The total number of vertices.
        degree (int): The degree of each vertex.
        edges (List[Tuple[int, int, int]]): The list to store edges.
    """
    left, right = i, i
    
    for _ in range(degree // 2):
        left = (left + 1) % num_vertices
        right = (right - 1) % num_vertices
        
        _create_edge(vertices, vertex_indices, i, left, edges)
        _create_edge(vertices, vertex_indices, i, right, edges)

def _create_edge(vertices, vertex_indices, i, neighbor_index, edges):
    """
    Create an edge between the current vertex and a neighbor.
    
    Args:
        vertices (List[Vertex]): The list of vertices.
        vertex_indices (List[int]): The shuffled list of vertex indices.
        i (int): The index of the current vertex.
        neighbor_index (int): The index of the neighbor vertex.
        edges (List[Tuple[int, int, int]]): The list to store edges.
    """
    if neighbor_index > i:
        weight = random.randint(0, 100)
        current_vertex = vertices[vertex_indices[i]]
        neighbor_vertex = vertices[vertex_indices[neighbor_index]]
        
        current_vertex.add_connection(vertex_indices[neighbor_index], weight)
        neighbor_vertex.add_connection(vertex_indices[i], weight)
        
        edges.append((
            min(vertex_indices[i], vertex_indices[neighbor_index]),
            max(vertex_indices[i], vertex_indices[neighbor_index]),
            weight
        ))

def _write_edges_to_file(edges, filename):
    """
    Write the edges to a file.
    
    Args:
        edges (List[Tuple[int, int, int]]): The list of edges.
        filename (str): The path to the file where edges will be written.
    """
    with open(filename, 'a') as file:
        for edge in edges:
            file.write(f"{edge[0]} {edge[1]} {edge[2]}\n")

# Example usage:
defined_degree_graph(5000, 6)
defined_degree_graph(5000, 1000)
