def is_prime(x):
    """
    Check if a number is prime.
    
    Args:
        x (int): The number to check.
        
    Returns:
        bool: True if x is a prime number, False otherwise.
    """
    if x < 2:
        return False
    for i in range(2, int(x ** 0.5) + 1):
        if x % i == 0:
            return False
    return True

def all_primes(n):
    """
    Get a list of all prime numbers up to n (inclusive).
    
    Args:
        n (int): The upper limit to check for prime numbers.
        
    Returns:
        list: A list of prime numbers up to n.
    """
    primes = []
    for i in range(2, n + 1):
        if is_prime(i):
            primes.append(i)
    return primes

def main():
    """
    Main function to print all prime numbers up to a specified limit.
    """
    primes_up_to_100 = all_primes(100)
    print(primes_up_to_100)

if __name__ == '__main__':
    main()
