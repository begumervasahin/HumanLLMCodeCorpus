def fizz_buzz(number):
    """
    Returns 'Fizz', 'Buzz', 'FizzBuzz', or the number itself based on its divisibility by 3 and 5.

    Args:
        number (int): The number to be checked.

    Returns:
        str or int: 'FizzBuzz' if divisible by both 3 and 5, 'Fizz' if divisible by 3,
                    'Buzz' if divisible by 5, otherwise the number itself.
    """
    if number % 3 == 0 and number % 5 == 0:
        return 'FizzBuzz'
    if number % 3 == 0:
        return 'Fizz'
    if number % 5 == 0:
        return 'Buzz'
    return number

def print_fizz_buzz(start, end):
    """
    Print the results of the fizz_buzz function for numbers in the specified range.

    Args:
        start (int): The starting number of the range.
        end (int): The ending number of the range.
    """
    for i in range(start, end + 1):
        print(fizz_buzz(i))

def main():
    """
    Main function to print the results of fizz_buzz for numbers from 1 to 100.
    """
    print_fizz_buzz(1, 100)

if __name__ == "__main__":
    main()
