import json
import pandas as pd
import matplotlib.pyplot as plt

# Path to the tweets data file
tweets_data_path = '/home/nick/projects/twitter_data.txt'

# Load tweets data from file
tweets_data = []
with open(tweets_data_path, "r") as tweets_file:
    for line in tweets_file:
        try:
            tweet = json.loads(line)
            tweets_data.append(tweet)
        except json.JSONDecodeError:
            continue

# Print the number of tweets loaded
print(f"Number of tweets loaded: {len(tweets_data)}")

# Create a DataFrame from the tweets data
tweets = pd.DataFrame(tweets_data)

# Extract relevant information from the tweets
tweets['text'] = tweets['text']
tweets['lang'] = tweets['lang']
tweets['country'] = tweets['place'].apply(lambda place: place['country'] if place is not None else None)

# Count tweets by language
tweets_by_lang = tweets['lang'].value_counts()

# Plot the top 5 languages
fig, ax = plt.subplots()
ax.tick_params(axis='x', labelsize=12)  # X-axis label size
ax.tick_params(axis='y', labelsize=12)  # Y-axis label size
ax.set_xlabel('Languages', fontsize=15)  # X-axis label
ax.set_ylabel('Number of Tweets', fontsize=15)  # Y-axis label
ax.set_title('Top 5 Languages in Tweets', fontsize=18, fontweight='bold')  # Plot title
tweets_by_lang[:5].plot(ax=ax, kind='bar', color='skyblue')  # Plotting top 5 languages

# Display the plot
plt.tight_layout()  # Ensures labels fit within the plot area
plt.show()

# Indicate completion
print("Visualization complete.")
