class Stack:
    """
    A simple Stack implementation using a list.
    """
    def __init__(self):
        self.items = []
        self.state = set()
    
    def push(self, item):
        """
        Add an item to the stack and track its state.

        Args:
            item: The item to be added to the stack.
        """
        self.items.append(item)  # Add item to the stack
        self.state.add(tuple(item.state_list))  # Track the state of the item
    
    def pop(self):
        """
        Remove the item from the top of the stack (LIFO) and return it.

        Returns:
            The item from the top of the stack, or None if the stack is empty.
        """
        if not self.is_empty():
            item = self.items.pop()  # Remove and return the top item
            self.state.remove(tuple(item.state_list))  # Remove the state of the popped item
            return item
        return None
    
    def is_empty(self):
        """
        Check if the stack is empty.

        Returns:
            bool: True if the stack is empty, False otherwise.
        """
        return len(self.items) == 0
    
    def clear(self):
        """
        Clear the stack and reset its state.
        """
        self.items.clear()
        self.state.clear()

class PuzzleState:
    """
    A simple class to represent a puzzle state.
    """
    def __init__(self, state_list):
        self.state_list = state_list

    def __repr__(self):
        return str(self.state_list)

# Example usage of the Stack class
if __name__ == "__main__":
    # Create a Stack instance
    stack = Stack()

    # Push some puzzle states onto the stack
    stack.push(PuzzleState([1, 2, 3, 4, 5, 6, 7, 8, 0]))
    stack.push(PuzzleState([1, 2, 3, 4, 5, 6, 0, 7, 8]))
    stack.push(PuzzleState([1, 2, 3, 4, 5, 0, 6, 7, 8]))

    # Pop and print each item
    while not stack.is_empty():
        state = stack.pop()
        print("Popped:", state)
    
    # Clear the stack
    stack.clear()
    print("Stack cleared. Is empty:", stack.is_empty())
