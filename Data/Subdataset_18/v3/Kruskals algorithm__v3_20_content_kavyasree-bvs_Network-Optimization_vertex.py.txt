class Vertex:
    def __init__(self, name):
        """
        Initializes a Vertex object.

        Parameters:
        - name (str or int): The name or identifier of the vertex.
        """
        self.name = name  # The name or identifier of the vertex
        self.neighbors = []  # A list to store the names of adjacent vertices

    def add_neighbor(self, neighbor):
        """
        Adds a neighbor to the vertex's adjacency list.

        Parameters:
        - neighbor (Vertex): The neighboring vertex to be added.

        Returns:
        - bool: False if the neighbor is not an instance of the Vertex class, otherwise None.
        """
        if not isinstance(neighbor, Vertex):
            return False

        if neighbor.name not in self.neighbors:
            # Add the neighbor's name to the current vertex's neighbors
            self.neighbors.append(neighbor.name)
            # Add this vertex's name to the neighbor's neighbors
            neighbor.neighbors.append(self.name)
            # Keep both neighbors lists sorted for consistency
            self.neighbors.sort()
            neighbor.neighbors.sort()

    def __repr__(self):
        """
        Returns a string representation of the vertex's neighbors.
        """
        return f"{self.neighbors}"

# Example usage
if __name__ == "__main__":
    # Create vertex instances
    vertex_a = Vertex("A")
    vertex_b = Vertex("B")
    vertex_c = Vertex("C")

    # Add neighbors to vertex_a
    vertex_a.add_neighbor(vertex_b)
    vertex_a.add_neighbor(vertex_c)

    # Display the neighbors of each vertex
    print(f"Neighbors of {vertex_a.name}: {vertex_a}")
    print(f"Neighbors of {vertex_b.name}: {vertex_b}")
    print(f"Neighbors of {vertex_c.name}: {vertex_c}")
