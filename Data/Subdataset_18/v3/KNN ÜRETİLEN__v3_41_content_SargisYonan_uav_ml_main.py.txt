import map_driver
import knn
import matplotlib.pyplot as plt
import matplotlib.animation as animation

def initialize_simulation(uav_count, interest_points):
    """
    Initialize UAVs and place interest points on the map.
    
    Parameters:
    - uav_count (int): Number of UAVs to initialize.
    - interest_points (list of tuples): List of interest points where each tuple is (x, y, intensity).
    """
    map_driver.initialize_uavs(uav_count)
    for point in interest_points:
        map_driver.place_interest_point(*point)

def run_uav_simulation(predicted_map, output_file):
    """
    Run the UAV flight simulation and save the output as a video file.
    
    Parameters:
    - predicted_map (numpy.ndarray): The predicted map for the UAVs to follow.
    - output_file (str): The name of the output video file.
    """
    map_driver.uav_flight(predicted_map, output_file)

def plot_predicted_map(predicted_map):
    """
    Plot the predicted map using matplotlib.
    
    Parameters:
    - predicted_map (numpy.ndarray): The predicted map to plot.
    """
    map_driver.plot_map(predicted_map)

def main():
    """
    Main function to initialize UAVs, place interest points, run a UAV flight simulation,
    and plot the predicted map.
    """
    # Define number of UAVs and interest points
    uav_count = 3
    interest_points = [(25, 25, 20), (19, 25, 20)]

    # Initialize simulation
    initialize_simulation(uav_count, interest_points)

    # Run UAV flight simulation and save as 'predicted.mp4'
    run_uav_simulation(knn.predicted_map, 'predicted.mp4')

    # Plot the predicted map
    plot_predicted_map(knn.predicted_map)

if __name__ == "__main__":
    main()
