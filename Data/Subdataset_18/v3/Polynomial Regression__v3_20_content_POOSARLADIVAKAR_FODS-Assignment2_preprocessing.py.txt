"""
This file contains the classes required for processing the data
before applying machine learning algorithms.
"""

import numpy as np

class NormalScaler:
    """
    The NormalScaler class transforms the given data
    to follow a normal distribution.
    """
    
    def fit(self, arr):
        """
        Computes the mean and standard deviation of the data.

        Parameters:
        arr (array-like): The data to compute the mean and standard deviation for.
        """
        self.mean = np.mean(arr)
        self.std = np.std(arr)
    
    def transform(self, arr):
        """
        Applies the normal transformation to the data.

        Parameters:
        arr (array-like): The data to be transformed.

        Returns:
        array-like: The transformed data.
        """
        return (arr - self.mean) / self.std


class MinMaxScaler:
    """
    The MinMaxScaler class transforms the given data
    so that all data points fall between 0 and 1.
    """
    
    def fit(self, arr):
        """
        Computes the minimum and maximum values of the data.

        Parameters:
        arr (array-like): The data to compute the minimum and maximum for.
        """
        self.min = np.min(arr)
        self.max = np.max(arr)
    
    def transform(self, arr):
        """
        Applies the min-max transformation to the data.

        Parameters:
        arr (array-like): The data to be transformed.

        Returns:
        array-like: The transformed data.
        """
        return (arr - self.min) / (self.max - self.min)
    
    def inverse_transform(self, arr):
        """
        Applies the inverse transformation to return the original data.

        Parameters:
        arr (array-like): The transformed data to be inverted.

        Returns:
        array-like: The original data.
        """
        return arr * (self.max - self.min) + self.min
