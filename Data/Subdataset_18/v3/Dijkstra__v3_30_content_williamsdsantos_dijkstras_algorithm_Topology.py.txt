class Topology:
    def __init__(self, file_path):
        """
        Initializes the Topology object by reading the topology file and extracting nodes and links.

        Parameters:
        file_path (str): The path to the topology file.
        """
        self.nodes = []
        self.links = []
        self._read_topology_file(file_path)

    def _read_topology_file(self, file_path):
        """
        Reads the topology file and populates the nodes and links lists.

        Parameters:
        file_path (str): The path to the topology file.
        """
        with open(file_path, 'r') as file:
            # Skip the first three header lines
            self._skip_lines(file, 3)
            # Read nodes
            self._read_nodes(file)
            # Skip the next three header lines
            self._skip_lines(file, 3)
            # Read links
            self._read_links(file)

    def _skip_lines(self, file, num_lines):
        """
        Skips a specified number of lines in the given file.

        Parameters:
        file (file object): The open file.
        num_lines (int): The number of lines to skip.
        """
        for _ in range(num_lines):
            next(file)

    def _read_nodes(self, file):
        """
        Reads the nodes from the topology file.

        Parameters:
        file (file object): The open topology file.
        """
        for line in file:
            data = line.split()
            if len(data) < 3:
                break
            node_id = data[2]
            self.nodes.append(node_id)

    def _read_links(self, file):
        """
        Reads the links from the topology file.

        Parameters:
        file (file object): The open topology file.
        """
        for line in file:
            data = line.split()
            if len(data) < 5:
                break
            origin = data[2]
            destination = data[3]
            length = data[4]
            self.links.append([origin, destination, length])

def main():
    file_path = 'data/topologies/PacificBell'
    topology = Topology(file_path)
    print("Nodes:", topology.nodes)
    print("Links:", topology.links)

if __name__ == "__main__":
    main()
