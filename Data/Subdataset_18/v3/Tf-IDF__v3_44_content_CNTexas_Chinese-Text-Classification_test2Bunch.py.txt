import os
import pickle
from sklearn.utils import Bunch
import jieba

def save_file(file_path, content):
    """
    Saves the given content to the specified file path.
    
    Parameters:
    file_path (str): The path where the content will be saved.
    content (bytes): The content to save.
    """
    with open(file_path, "wb") as file:
        file.write(content)

def read_file(file_path):
    """
    Reads and returns the content from the specified file path.
    
    Parameters:
    file_path (str): The path of the file to read.
    
    Returns:
    bytes: The content read from the file.
    """
    with open(file_path, "rb") as file:
        return file.read()

def create_bunch_from_files(segmented_path, output_path):
    """
    Creates a Bunch object from files in the segmented path and saves it to the output path.
    
    Parameters:
    segmented_path (str): The path containing the segmented text files categorized in directories.
    output_path (str): The path where the Bunch object will be saved.
    """
    # Initialize the Bunch object
    bunch = Bunch(target_name=[], label=[], filenames=[], contents=[])
    
    # Get list of categories (directories) in the segmented path
    categories = os.listdir(segmented_path)
    bunch.target_name.extend(categories)

    # Process each category directory
    for category in categories:
        category_path = os.path.join(segmented_path, category)
        files = os.listdir(category_path)

        # Process each file in the category directory
        for file_name in files:
            file_path = os.path.join(category_path, file_name)
            content = read_file(file_path).strip()
            
            bunch.label.append(category)
            bunch.filenames.append(file_path)
            bunch.contents.append(content)

    # Save the Bunch object to the output path
    with open(output_path, "wb") as output_file:
        pickle.dump(bunch, output_file)

    print("Text object construction completed!")

if __name__ == "__main__":
    SEGMENTED_PATH = "test_corpus_seg/"
    OUTPUT_PATH = "test_word_bag/test_set.dat"

    create_bunch_from_files(SEGMENTED_PATH, OUTPUT_PATH)
