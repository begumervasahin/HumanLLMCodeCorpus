import requests
from bs4 import BeautifulSoup
import re

def get_html(url):
    """
    Fetch HTML content from the given URL.

    Args:
        url (str): The URL to fetch HTML content from.

    Returns:
        str: The HTML content of the page.
    """
    headers = {
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; WOW64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/47.0.2526.80 Safari/537.36',
        'Accept': '*/*',
        'Connection': 'keep-alive',
        'Host': 'www.plantcell.org'
    }
    response = requests.get(url, headers=headers)
    response.encoding = 'utf-8'
    return response.text

def analyse(html):
    """
    Analyse the HTML content to extract paragraphs, authors, and addresses.

    Args:
        html (str): The HTML content to analyse.

    Returns:
        tuple: A tuple containing content, authors, and addresses.
    """
    soup = BeautifulSoup(html, 'lxml')
    
    # Extract content paragraphs
    contents = []
    for paragraph in soup.find_all('p'):
        try:
            if "p-" in paragraph.get('id', ''):
                content = paragraph.get_text(strip=True)
                content = re.sub(r' +', ' ', content)
                if len(content) > 250:
                    contents.append(content)
        except Exception as e:
            print(f"Error extracting content: {e}")

    # Extract authors and addresses
    address_list = []
    author_list = []
    count = 1

    for item in soup.find_all('li'):
        try:
            if 'last' in item.get('class', []) and 'name' in str(item):
                author = item.find('a').string
                author_list.append(author)
            elif 'aff' in item.get('class', []):
                address = item.find('address').get_text(separator=' ', strip=True)
                if re.match(r'[a-z]', address[0]):
                    address = address[1:]
                address_list.append(f"{count}\t{address}\n")
                count += 1
        except Exception as e:
            print(f"Error extracting author/address: {e}")

    # Check for content error
    if len(contents) > 2:
        print("Content error")
        return '', '', ''

    return ''.join(contents), ''.join(author_list), ''.join(address_list)

def save_data(vol, page, content, author, address):
    """
    Save the extracted content, authors, and addresses to respective CSV files.

    Args:
        vol (int): The volume number.
        page (int): The page number.
        content (str): The content extracted from the page.
        author (str): The author information extracted from the page.
        address (str): The address information extracted from the page.
    """
    with open(f"{vol}-content.csv", 'a', encoding='utf-8') as content_file, \
         open(f"{vol}-author.csv", 'a', encoding='utf-8') as author_file, \
         open(f"{vol}-address.csv", 'a', encoding='utf-8') as address_file:
        
        if content and content[-1] != "\n":
            content += "\n"
        if author and author[-1] != "\n":
            author += "\n"
        if address and address[-1] != "\n":
            address += "\n"
        
        content_file.write(f">{page}\n{content}")
        author_file.write(f">{page}\n{author}")
        address_file.write(f">{page}\n{address}")

def main(vol, page):
    """
    Main function to fetch, analyse, and save the data for a given volume and page.

    Args:
        vol (int): The volume number.
        page (int): The page number.
    """
    url = f"http://www.plantcell.org/content/28/{vol}/{page}.abstract"
    html = get_html(url)
    content, author, address = analyse(html)
    save_data(vol, page, content, author, address)
