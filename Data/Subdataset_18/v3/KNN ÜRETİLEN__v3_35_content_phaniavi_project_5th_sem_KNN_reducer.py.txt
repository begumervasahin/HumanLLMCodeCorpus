import sys
from operator import itemgetter

def read_distances():
    """
    Read distances from standard input.

    Returns:
    list of list: A list of distances read from standard input.
    """
    distances = []
    for line in sys.stdin:
        elements = line.strip().split()
        distances.append(elements)
    return distances

def calculate_average_of_smallest(distances, k):
    """
    Calculate the average of the k smallest distances.

    Parameters:
    distances (list of list): The list of distances.
    k (int): The number of smallest distances to consider.

    Returns:
    float: The average of the k smallest distances.
    """
    # Sort distances based on the first element
    sorted_distances = sorted(distances, key=itemgetter(0))

    # Sum the k smallest values
    estimate_val = 0.0
    valid_entries = 0
    for i in range(k):
        try:
            estimate_val += float(sorted_distances[i][1])
            valid_entries += 1
        except (ValueError, IndexError):
            # Skip entries that cannot be converted to float or are out of range
            continue

    # Calculate and return the average
    if valid_entries > 0:
        return estimate_val / valid_entries
    else:
        return 0.0

def main():
    """
    Main function to read distances, sort them, and calculate the average of the k smallest distances.
    """
    k = 5  # Number of smallest distances to consider
    distances = read_distances()
    average = calculate_average_of_smallest(distances, k)
    print(average)

if __name__ == "__main__":
    main()
