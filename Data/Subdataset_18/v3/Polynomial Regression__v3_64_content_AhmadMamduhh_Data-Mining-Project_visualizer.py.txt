import matplotlib.pyplot as plt
from sklearn.decomposition import PCA
from sklearn.cluster import KMeans
from sklearn.datasets import load_iris

class Visualizer:
    """This class contains methods to help visualize data so the inputs and results can be understood clearly."""
    
    def plot_classifier_regressor(self, y_test, y_result, method_identifier):
        """Plot actual data against predicted data.

        Parameters:
        y_test (array-like): Actual target values.
        y_result (array-like): Predicted target values.
        method_identifier (int): Identifier to specify the type of plot (1 for classification, 2 for regression).
        """
        if method_identifier == 2:
            self._plot_regression_results(y_test, y_result)
        elif method_identifier == 1:
            self._plot_classification_results(y_test, y_result)
    
    def _plot_regression_results(self, y_test, y_result):
        """Helper method to plot regression results."""
        plt.scatter(y_test, y_result, c='blue')
        plt.title('Actual price vs. Predicted price')
        plt.xlabel('Actual Price')
        plt.ylabel('Predicted Price')
        plt.show()

    def _plot_classification_results(self, y_test, y_result):
        """Helper method to plot classification results."""
        plt.hist(y_test, bins=2, alpha=0.7, label='Actual')
        plt.hist(y_result, bins=2, alpha=0.7, label='Predicted')
        plt.title('Malignant vs. Benign (Classification Results)')
        plt.xlabel('0 = Benign   1 = Malignant')
        plt.ylabel('Number of patients')
        plt.legend()
        plt.grid()
        plt.show()
    
    def plot_clustering(self, iris_data, clusters):
        """Plot the Iris dataset before and after clustering using PCA for dimensionality reduction.

        Parameters:
        iris_data (Bunch): The Iris dataset.
        clusters (array-like): Cluster labels for the data points.
        """
        pca_2d = self._reduce_dimensions(iris_data.data)
        
        self._plot_iris_distribution(pca_2d)
        self._plot_iris_clustering(pca_2d, clusters)
    
    def _reduce_dimensions(self, data):
        """Helper method to reduce data dimensions using PCA."""
        pca = PCA(n_components=2).fit(data)
        return pca.transform(data)

    def _plot_iris_distribution(self, pca_2d):
        """Helper method to plot the Iris dataset distribution."""
        plt.figure('Reference Plot')
        plt.title('Iris Test Dataset Distribution')
        plt.scatter(pca_2d[:, 0], pca_2d[:, 1])
        plt.show()

    def _plot_iris_clustering(self, pca_2d, clusters):
        """Helper method to plot the Iris dataset after clustering."""
        plt.figure('K-means')
        plt.title('Iris Test Dataset After Clustering')
        colors = ['r', 'g', 'b', 'y']
        markers = ['+', 'o', '*', 'X']
        list_clusters = []
        cluster_names = []

        for i in range(pca_2d.shape[0]):
            cluster_idx = clusters[i]
            scatter = plt.scatter(pca_2d[i, 0], pca_2d[i, 1], c=colors[cluster_idx], marker=markers[cluster_idx])
            cluster_label = f'Cluster {cluster_idx + 1}'
            if cluster_label not in cluster_names:
                cluster_names.append(cluster_label)
                list_clusters.append(scatter)
        
        plt.legend(list_clusters, cluster_names)
        plt.show()

# Example usage
if __name__ == "__main__":
    visualizer = Visualizer()
    
    # Example data for classification/regression plotting
    y_test = [0, 0, 1, 1, 0, 1, 0, 1]
    y_result = [0, 1, 1, 1, 0, 0, 0, 1]
    visualizer.plot_classifier_regressor(y_test, y_result, method_identifier=1)

    # Example data for clustering
    iris = load_iris()
    kmeans = KMeans(n_clusters=3, random_state=42)
    kmeans.fit(iris.data)
    clusters = kmeans.predict(iris.data)
    visualizer.plot_clustering(iris, clusters)
