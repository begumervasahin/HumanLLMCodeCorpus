class Student:
    def __init__(self, name, rating, growth):
        self.name = name
        self.rating = rating
        self.growth = growth

    def __repr__(self):
        return f"{self.name} (Rating: {self.rating}, Growth: {self.growth})"

# Global variables for counting comparisons and swaps
comparison_count = 0
swap_count = 0

def bubble_sort(arr):
    """Perform bubble sort on the list based on student rating."""
    global comparison_count, swap_count
    n = len(arr)
    for i in range(n):
        swapped = False
        for j in range(0, n - i - 1):
            comparison_count += 1
            if arr[j].rating < arr[j + 1].rating:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]
                swap_count += 1
                swapped = True
        if not swapped:
            break
    print(f"BubbleSort\nComparison count: {comparison_count}\nSwap count: {swap_count}")
    comparison_count, swap_count = 0, 0

def quick_sort(arr, low, high):
    """Perform quick sort on the list based on student growth."""
    if low < high:
        pi = partition(arr, low, high)
        quick_sort(arr, low, pi - 1)
        quick_sort(arr, pi + 1, high)

def partition(arr, low, high):
    """Partition the array and return the pivot index."""
    global comparison_count, swap_count
    pivot = arr[high].growth
    i = low - 1
    for j in range(low, high):
        comparison_count += 1
        if arr[j].growth < pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
            swap_count += 1
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    swap_count += 1
    return i + 1

def reset_comparison_and_swap_counts():
    """Reset the global comparison and swap counts."""
    global comparison_count, swap_count
    comparison_count, swap_count = 0, 0

def print_students(title, students):
    """Print the list of students with a title."""
    print(title)
    for student in students:
        print(student)
    print()

if __name__ == '__main__':
    students = [
        Student("Alice", 85, 160),
        Student("Bob", 70, 170),
        Student("Charlie", 90, 155),
        Student("David", 75, 180)
    ]

    print_students("Original list:", students)

    # Perform bubble sort based on rating
    bubble_sort(students)
    print_students("Sorted list by rating (Bubble Sort):", students)

    # Reset the list for quick sort
    students = [
        Student("Alice", 85, 160),
        Student("Bob", 70, 170),
        Student("Charlie", 90, 155),
        Student("David", 75, 180)
    ]

    # Perform quick sort based on growth
    quick_sort(students, 0, len(students) - 1)
    print(f"QuickSort\nComparison count: {comparison_count}\nSwap count: {swap_count}")
    reset_comparison_and_swap_counts()

    print_students("Sorted list by growth (Quick Sort):", students)
