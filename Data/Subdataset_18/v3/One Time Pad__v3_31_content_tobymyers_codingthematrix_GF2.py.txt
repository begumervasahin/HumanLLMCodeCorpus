from numbers import Number

class One:
    """
    A class representing a special numeric object that behaves like 'one'
    with customized arithmetic and comparison operations.
    """
    
    def __add__(self, other):
        """
        Custom addition: returns 'one' if other is 0, else returns 0.
        """
        return self if other == 0 else 0

    __sub__ = __add__

    def __mul__(self, other):
        """
        Custom multiplication: returns 0 if other is 0, else returns 'one'.
        """
        if isinstance(other, Number):
            return 0 if other == 0 else self
        return other

    def __truediv__(self, other):
        """
        Custom division: returns 'one' if other is not 0, raises ZeroDivisionError otherwise.
        """
        if other == 0:
            raise ZeroDivisionError("Division by zero")
        return self

    __div__ = __truediv__

    def __rtruediv__(self, other):
        """
        Custom reverse division: returns the other operand.
        """
        return other

    __rdiv__ = __rtruediv__
    __radd__ = __add__
    __rsub__ = __add__
    __rmul__ = __mul__

    def __lt__(self, other):
        """
        Custom less-than comparison: always returns False.
        """
        return False

    def __eq__(self, other):
        """
        Custom equality comparison: returns True if other is 'one', False if other is 0,
        raises TypeError for unsupported comparisons.
        """
        if isinstance(other, One) or other == 0:
            return other != 0
        else:
            raise TypeError(f"Unsupported comparison between 'One' and '{type(other).__name__}'")

    def __hash__(self):
        """
        Custom hash function: returns a constant hash value.
        """
        return 1

    def __str__(self):
        """
        String representation of the object.
        """
        return 'one'

    __repr__ = __str__

    def __neg__(self):
        """
        Unary negation: returns 'one'.
        """
        return self

    def __bool__(self):
        """
        Boolean representation: always True.
        """
        return True

    def __format__(self, format_spec):
        """
        Custom format: formats the string representation according to format_spec.
        """
        return format(str(self), format_spec)

# Create an instance of One
one = One()

# Examples to demonstrate the functionality
print(one + 0)          # Output: one
print(one + 1)          # Output: 0
print(one - 0)          # Output: one
print(one * 0)          # Output: 0
print(one * 1)          # Output: one
print(one / 1)          # Output: one
print(one == one)       # Output: True
print(one == 0)         # Output: False
print(one < 1)          # Output: False
print(one > 1)          # Output: False
print(-one)             # Output: one
print(bool(one))        # Output: True
print(format(one))      # Output: one
print(hash(one))        # Output: 1
