# Import necessary utility functions from des_utils and data_utils
from des_utils import generateKeys, get_bits, add_pads_if_necessary, encryptBlock, bin_to_hex
from data_utils import joinNumbers

def encrypt(msg: str, key: str) -> str:
    """
    Encrypt a message using DES encryption with the provided key.
    
    :param msg: The message to encrypt.
    :param key: The encryption key.
    :return: The encrypted message in hexadecimal format.
    """
    # Generate the DES keys from the provided key
    keys = generateKeys(key)
    
    # Convert the message to bits and add necessary padding
    text_bits = add_pads_if_necessary(get_bits(msg))
    
    # Process each 64-bit block of the message
    final_cipher = []
    for i in range(0, len(text_bits), 64):
        block = text_bits[i:i+64]
        final_cipher.extend(encryptBlock(block, keys))
    
    # Convert the final cipher bits to hexadecimal format
    hex_cipher = ''.join(bin_to_hex(joinNumbers(final_cipher[i:i+4])) for i in range(0, len(final_cipher), 4))
    
    return hex_cipher

def decrypt(cipher: str, key: str) -> str:
    """
    Decrypt a message using DES encryption with the provided key.
    
    :param cipher: The encrypted message in hexadecimal format.
    :param key: The decryption key.
    :return: The decrypted message.
    """
    # Generate the DES keys from the provided key and reverse them for decryption
    keys = generateKeys(key)
    keys.reverse()
    
    # Convert the cipher from hexadecimal to bits
    text_bits = get_bits(cipher)
    
    # Process each 64-bit block of the cipher
    decrypted_bits = []
    for i in range(0, len(text_bits), 64):
        block = text_bits[i:i+64]
        decrypted_bits.extend(encryptBlock(block, keys))
    
    # Convert the decrypted bits to hexadecimal format and remove padding
    hex_msg = ''.join(bin_to_hex(joinNumbers(decrypted_bits[i:i+4])) for i in range(0, len(decrypted_bits), 4))
    
    return hex_msg.rstrip('0')

# Example usage of the encryption and decryption functions
if __name__ == "__main__":
    message = "This is a test message."
    encryption_key = "mysecretkey"
    
    # Encrypt the message
    encrypted_message = encrypt(message, encryption_key)
    print(f"Encrypted Message: {encrypted_message}")
    
    # Decrypt the message
    decrypted_message = decrypt(encrypted_message, encryption_key)
    print(f"Decrypted Message: {decrypted_message}")
