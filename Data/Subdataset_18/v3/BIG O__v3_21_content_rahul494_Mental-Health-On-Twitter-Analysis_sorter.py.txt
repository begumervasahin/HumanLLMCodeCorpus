import sys
import operator

def read_input():
    """
    Read key-value pairs from standard input and store them in a dictionary.

    Returns:
        dict: A dictionary with keys as strings and values as integers.
    """
    storage = {}
    for line in sys.stdin:
        value, key = line.split()
        storage[key] = int(value)
    return storage

def sort_storage(storage):
    """
    Sort the dictionary by values in descending order.

    Args:
        storage (dict): The dictionary to be sorted.

    Returns:
        list: A list of tuples containing the sorted key-value pairs.
    """
    return sorted(storage.items(), key=operator.itemgetter(1), reverse=True)

def print_sorted_storage(sorted_storage):
    """
    Print the sorted key-value pairs.

    Args:
        sorted_storage (list): A list of tuples containing the sorted key-value pairs.
    """
    for key, value in sorted_storage:
        print(f"{key},{value}")

def main():
    """
    Main function to read input, sort the storage, and print the sorted key-value pairs.
    """
    storage = read_input()
    sorted_storage = sort_storage(storage)
    print_sorted_storage(sorted_storage)

if __name__ == "__main__":
    main()
