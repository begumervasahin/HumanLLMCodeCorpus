from itertools import islice
from decorators import memoised

def _fibonacci():
    """
    Generator function to yield an infinite Fibonacci sequence.
    """
    a, b = 0, 1
    while True:
        yield a
        a, b = b, a + b

def take(n, iterable):
    """
    Take n items from the given iterable.

    :param n: Number of items to take from the iterable.
    :param iterable: The iterable to take items from.
    :return: A list of the first n items from the iterable.
    """
    return list(islice(iterable, n))

def fibonacci_numbers(n=None):
    """
    Generate a specified number of Fibonacci numbers.
    
    :param n: Number of Fibonacci numbers to generate. If None, generate indefinitely.
    :return: A list of the first n Fibonacci numbers, or an infinite generator if n is None.
    """
    if n is not None:
        return take(n, _fibonacci())
    else:
        return _fibonacci()

@memoised
def fibonacci_number(num):
    """
    Return the nth Fibonacci number using a recursive method.
    
    :param num: The position of the Fibonacci number to return.
    :return: The nth Fibonacci number.
    """
    if num < 0:
        raise ValueError('Fibonacci numbers are not defined for negative numbers.')
    if num < 2:
        return num
    return fibonacci_number(num - 2) + fibonacci_number(num - 1)

def display_fibonacci_sequence():
    """Display the first few Fibonacci numbers."""
    print('Fibonacci Sequence:\n')
    print('First Few:')
    for num in fibonacci_numbers(13):
        print(num)
    print()

def display_infinite_fibonacci_sequence():
    """Display an infinite Fibonacci sequence."""
    from time import sleep
    
    print('Infinite (Ctrl-C to exit):')
    try:
        for num in fibonacci_numbers():
            print(num)
            sleep(0.42)
    except KeyboardInterrupt:
        print("\nStopped.")

if __name__ == "__main__":
    display_fibonacci_sequence()
    display_infinite_fibonacci_sequence()
