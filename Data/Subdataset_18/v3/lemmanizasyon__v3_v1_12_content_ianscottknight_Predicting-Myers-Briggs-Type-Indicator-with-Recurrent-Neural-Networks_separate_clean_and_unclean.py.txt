import os
import collections
import pandas as pd
import csv

# Define paths and frequency dictionary
DATA_DIR = "data"
MBTI_RAW_CSV_PATH = os.path.join(DATA_DIR, "mbti_1.csv")
MBTI_CLEAN_CSV_PATH = os.path.join(DATA_DIR, "mbti_clean.csv")
MBTI_UNCLEAN_CSV_PATH = os.path.join(DATA_DIR, "mbti_unclean.csv")
MBTI_TO_FREQUENCY_DICT = {
    "ISTJ": 0.11,
    "ISFJ": 0.09,
    "INFJ": 0.04,
    "INTJ": 0.05,
    "ISTP": 0.05,
    "ISFP": 0.05,
    "INFP": 0.06,
    "INTP": 0.06,
    "ESTP": 0.04,
    "ESFP": 0.04,
    "ENFP": 0.08,
    "ENTP": 0.06,
    "ESTJ": 0.08,
    "ESFJ": 0.09,
    "ENFJ": 0.05,
    "ENTJ": 0.05,
}

def read_data(file_path):
    return pd.read_csv(file_path)

def count_mbti_types(dataframe):
    counts = collections.defaultdict(int)
    for mbti_type in dataframe["type"]:
        counts[mbti_type] += 1
    return counts

def find_limiting_type(counts, frequency_dict):
    limiting_type = None
    min_size = float("infinity")
    for mbti_type, count in counts.items():
        size = count / frequency_dict[mbti_type]
        if size < min_size:
            min_size = size
            limiting_type = mbti_type
    return limiting_type, min_size

def group_data_by_type(dataframe):
    grouped_data = collections.defaultdict(list)
    for _, row in dataframe.iterrows():
        grouped_data[row["type"]].append(row)
    return grouped_data

def prepare_unclean_data(grouped_data, min_size, frequency_dict):
    unclean_data = []
    for mbti_type, freq in frequency_dict.items():
        type_data = grouped_data[mbti_type]
        start_idx = int(round(min_size * freq))
        unclean_data.extend(type_data[start_idx:])
    return unclean_data

def write_to_csv(file_path, data, header):
    with open(file_path, "w", newline='') as f:
        writer = csv.writer(f)
        writer.writerow(header)
        for entry in data:
            writer.writerow([entry["type"], entry["posts"]])

def main():
    # Read the raw data
    df = read_data(MBTI_RAW_CSV_PATH)
    
    # Count occurrences of each MBTI type
    counts = count_mbti_types(df)
    
    # Find the MBTI type with the smallest relative frequency
    limiting_type, min_size = find_limiting_type(counts, MBTI_TO_FREQUENCY_DICT)
    
    # Group data by MBTI type
    grouped_data = group_data_by_type(df)
    
    # Prepare data for unclean CSV
    unclean_data = prepare_unclean_data(grouped_data, min_size, MBTI_TO_FREQUENCY_DICT)
    
    # Write unclean data to CSV
    write_to_csv(MBTI_UNCLEAN_CSV_PATH, unclean_data, ["type", "posts"])
    
    # Prepare data for clean CSV
    clean_data = []
    for mbti_type, type_data in grouped_data.items():
        if mbti_type != limiting_type:
            clean_data.extend(type_data)
    
    # Write clean data to CSV
    write_to_csv(MBTI_CLEAN_CSV_PATH, clean_data, ["type", "posts"])

if __name__ == "__main__":
    main()
