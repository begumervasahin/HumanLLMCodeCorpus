import pygame
import random

# Define some colors
BLACK = (0, 0, 0)
WHITE = (255, 255, 255)
RED = (255, 0, 0)

# Set the width and height of each grid location
WIDTH = 20
HEIGHT = 20

# Set the margin between each cell
MARGIN = 0

# Set the dimensions of the grid
GRID_WIDTH = 34
GRID_HEIGHT = 24

# Set the dimensions of the window
WINDOW_SIZE = [680, 480]

# Create a 2D array to store the grid
grid = [[0 for _ in range(GRID_WIDTH)] for _ in range(GRID_HEIGHT)]

# Initialize Pygame
pygame.init()
screen = pygame.display.set_mode(WINDOW_SIZE)
pygame.display.set_caption("RANDOM MAZE")

# Manage how fast the screen updates
clock = pygame.time.Clock()

# Initialize the frontier list
frontier = []

# Start at a random position in the grid
start_i = random.randint(2, GRID_HEIGHT - 3)
start_j = random.randint(2, GRID_WIDTH - 3)
grid[start_i][start_j] = 1

# Function to add frontier cells
def add_frontier(x, y):
    if 2 <= x < GRID_HEIGHT - 2 and 2 <= y < GRID_WIDTH - 2 and grid[x][y] == 0:
        grid[x][y] = 2
        frontier.append((x, y))

# Add initial frontier cells
add_frontier(start_i + 2, start_j)
add_frontier(start_i - 2, start_j)
add_frontier(start_i, start_j + 2)
add_frontier(start_i, start_j - 2)

# Function to draw the grid
def draw_grid():
    for row in range(GRID_HEIGHT):
        for column in range(GRID_WIDTH):
            color = BLACK
            if grid[row][column] == 1:
                color = WHITE
            elif grid[row][column] == 2:
                color = RED
            pygame.draw.rect(screen, color, 
                             [(MARGIN + WIDTH) * column + MARGIN,
                              (MARGIN + HEIGHT) * row + MARGIN, WIDTH, HEIGHT])

# Draw the initial grid
draw_grid()
pygame.display.flip()

# Main loop
done = False
while not done:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            done = True
    
    while frontier:
        # Select a random frontier cell
        f = random.randint(0, len(frontier) - 1)
        x, y = frontier[f]

        # Find neighbors
        neighbors = []
        if grid[x - 2][y] == 1:
            neighbors.append((x - 2, y))
        if grid[x + 2][y] == 1:
            neighbors.append((x + 2, y))
        if grid[x][y - 2] == 1:
            neighbors.append((x, y - 2))
        if grid[x][y + 2] == 1:
            neighbors.append((x, y + 2))

        if neighbors:
            p, q = random.choice(neighbors)
            grid[x][y] = 1
            if p == x:
                if q > y:
                    grid[x][y + 1] = 1
                else:
                    grid[x][y - 1] = 1
            else:
                if p > x:
                    grid[x + 1][y] = 1
                else:
                    grid[x - 1][y] = 1

        # Add new frontier cells
        add_frontier(x + 2, y)
        add_frontier(x - 2, y)
        add_frontier(x, y + 2)
        add_frontier(x, y - 2)
        
        # Draw the grid
        draw_grid()
        pygame.display.flip()
        del frontier[f]

    clock.tick(10)

# Quit Pygame
pygame.quit()
