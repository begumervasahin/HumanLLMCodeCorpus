import os
import pandas as pd
import re

# Define the encoding used in the text files
ENCODING = 'utf-8'  # Adjust this based on the actual encoding used

def load_data(path_to_dir):
    """
    Loads data from text files in a directory.

    :param path_to_dir: Path to the directory containing text files.
    :return: A list of lists containing the first three floating point numbers found in each line.
    """
    text_files = _get_text_files(path_to_dir)
    all_data = []

    for file in text_files:
        file_path = os.path.join(path_to_dir, file)
        file_content = _read_file_content(file_path)
        all_data.extend(file_content)

    filtered_data = [_extract_numbers(line) for line in all_data]
    return filtered_data

def _get_text_files(directory):
    """
    Retrieves all .txt files in a given directory.

    :param directory: Path to the directory.
    :return: List of .txt file names.
    """
    return [
        file for file in os.listdir(directory)
        if os.path.isfile(os.path.join(directory, file)) and file.endswith(".txt")
    ]

def _read_file_content(file_path):
    """
    Reads the content of a text file.

    :param file_path: Path to the text file.
    :return: List of lines in the file.
    """
    with open(file_path, "r", encoding=ENCODING) as file:
        return file.read().splitlines()

def _extract_numbers(text):
    """
    Extracts the first three floating point numbers from a string.

    :param text: Input text string.
    :return: List of the first three floating point numbers found in the string.
    """
    pattern = r"[0-9]+"
    matches = re.findall(pattern, text)
    numbers = [float(match) for match in matches[:3]]
    return numbers

def save_to_csv(data, file_path):
    """
    Saves data to a CSV file.

    :param data: Data to be saved.
    :param file_path: Path to save the CSV file.
    """
    df = pd.DataFrame(data, columns=['x', 'y', 'z'])
    df.to_csv(file_path, index=False, encoding=ENCODING)
