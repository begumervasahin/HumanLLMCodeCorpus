def fibonacci(n):
    """
    Calculate the nth Fibonacci number using Binet's formula.

    Args:
        n (int): The position of the Fibonacci number to generate.

    Returns:
        int: The nth Fibonacci number.
    """
    phi = (1 + sqrt(5)) / 2
    psi = (1 - sqrt(5)) / 2
    return int((phi**n - psi**n) / sqrt(5))

def generate_fibonacci_sequence(number, binary=False):
    """
    Generate a Fibonacci sequence of a given length.

    Args:
        number (int): The number of Fibonacci numbers to generate.
        binary (bool): Whether to output the numbers in binary format.

    Returns:
        list: A list containing the Fibonacci sequence.
    """
    sequence = []
    for x in range(1, number + 1):
        fib_num = fibonacci(x)
        if binary:
            sequence.append(bin(fib_num)[2:])
        else:
            sequence.append(str(fib_num))
    return sequence

def main():
    parser = argparse.ArgumentParser(description='Generate Fibonacci sequence')
    parser.add_argument(
        "-b", "--binary",
        action="store_true",
        help="Print output in binary"
    )
    parser.add_argument(
        "number",
        type=int,
        help="Generate n length sequence"
    )
    args = parser.parse_args()

    sequence = generate_fibonacci_sequence(args.number, args.binary)
    print(" ".join(sequence))

if __name__ == '__main__':
    main()
