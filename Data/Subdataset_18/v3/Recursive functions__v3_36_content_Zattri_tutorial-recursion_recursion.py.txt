# Countdown function using recursion
def countdown(x):
    """Prints numbers from x down to 0."""
    print(x)
    if x > 0:
        countdown(x - 1)

# Factorial function using recursion
def factorial(x):
    """Calculates factorial of x using recursion."""
    if x == 0:
        return 1
    else:
        return x * factorial(x - 1)

# Tail-recursive factorial function
def tail_factorial(x, acc=1):
    """Calculates factorial of x using tail recursion."""
    if x == 0:
        return acc
    else:
        return tail_factorial(x - 1, acc * x)

# Binary tree node class
class Node:
    def __init__(self, value):
        self.value = value
        self.left = None
        self.right = None

    def add_left(self, node):
        """Adds a node to the left child."""
        self.left = node

    def add_right(self, node):
        """Adds a node to the right child."""
        self.right = node

# Creating a binary tree
def create_sample_tree():
    root_node = Node(0)
    node1 = Node(1)
    node2 = Node(2)
    node3 = Node(3)
    node4 = Node(4)
    node5 = Node(5)

    root_node.add_left(node1)
    root_node.add_right(node2)
    node1.add_left(node3)
    node2.add_left(node4)
    node2.add_right(node5)

    return root_node

# Tree traversal function
def tree_traverse(root):
    """Traverses the binary tree in pre-order."""
    if root is not None:
        print(root.value)
        if root.left:
            tree_traverse(root.left)
        if root.right:
            tree_traverse(root.right)

# Tree search function
def tree_search(root, val):
    """Searches for a value in the binary tree."""
    if root is not None:
        if root.value == val:
            print(f"Found {val}")
        else:
            tree_search(root.left, val)
            tree_search(root.right, val)

# Merge sort function
def merge_sort(arr):
    """Sorts an array using the merge sort algorithm."""
    if len(arr) < 2:
        return arr

    mid = len(arr) // 2
    left_half = merge_sort(arr[:mid])
    right_half = merge_sort(arr[mid:])

    return merge(left_half, right_half)

def merge(left, right):
    """Merges two sorted arrays."""
    result = []
    i = j = 0

    while i < len(left) and j < len(right):
        if left[i] > right[j]:
            result.append(right[j])
            j += 1
        else:
            result.append(left[i])
            i += 1

    result.extend(left[i:])
    result.extend(right[j:])

    return result

    # Tree search
