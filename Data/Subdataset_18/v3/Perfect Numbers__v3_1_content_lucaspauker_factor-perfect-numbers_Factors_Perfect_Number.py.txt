def classify_number(n):
    """
    Classifies a number as Perfect, Deficient, or Abundant based on the sum of its proper divisors.
    
    Args:
    n (int): The number to classify.
    
    Returns:
    str: The classification of the number ('Perfect', 'Deficient', or 'Abundant').
    """
    total_sum = 0
    factors = []
    
    # Find all proper divisors and sum them
    for x in range(1, n // 2 + 1):
        if n % x == 0:
            total_sum += x
            factors.append(x)
    
    print('Factors:', factors)
    print('Sum of factors:', total_sum)
    print('Number of factors:', len(factors))
    
    # Determine the classification
    if total_sum == n:
        return 'Perfect'
    elif total_sum < n:
        return 'Deficient'
    else:
        return 'Abundant'

def find_perfect_square_factors(n):
    """
    Finds and prints all perfect square factors of a number.
    
    Args:
    n (int): The number to find perfect square factors for.
    """
    square_factors = []
    
    # Find all perfect square factors
    for x in range(1, int(n**0.5) + 1):
        if n % (x ** 2) == 0:
            square_factors.append(x)
    
    print('Perfect square factors:', square_factors)
    print('Number of perfect square factors:', len(square_factors))

# Example usage
number = 28
print(classify_number(number))
find_perfect_square_factors(number)
