import tkinter as tk
from socket import AF_INET, socket, SOCK_STREAM
from threading import Thread
from Crypto.Cipher import AES
import base64

# AES Encryption/Decryption functions
BLOCK_SIZE = 16

def pad(data):
    """Pads the data to be a multiple of BLOCK_SIZE."""
    padding_length = BLOCK_SIZE - (len(data) % BLOCK_SIZE)
    padding = chr(padding_length) * padding_length
    return data + padding.encode()

def unpad(data):
    """Removes padding from the data."""
    padding_length = ord(data[-1:])
    return data[:-padding_length]

def encrypt(key, data):
    """Encrypts the data using AES encryption."""
    data = pad(data)
    cipher = AES.new(key, AES.MODE_ECB)
    encrypted_data = cipher.encrypt(data)
    return base64.b64encode(encrypted_data)

def decrypt(key, enc_data):
    """Decrypts the data using AES encryption."""
    enc_data = base64.b64decode(enc_data)
    cipher = AES.new(key, AES.MODE_ECB)
    decrypted_data = cipher.decrypt(enc_data)
    return unpad(decrypted_data)

# Chat client functions
def receive():
    """Handles receiving of messages."""
    while True:
        try:
            msg = client_socket.recv(BUFSIZ)
            decrypted_msg = decrypt(key, msg).decode("utf-8")
            msg_list.insert(tk.END, decrypted_msg)
        except OSError:  # Possibly client has left the chat.
            break

def send(event=None):
    """Handles sending of messages."""
    msg = my_msg.get()
    my_msg.set("")  # Clears input field.
    encrypted_msg = encrypt(key, msg.encode('utf-8'))
    client_socket.send(encrypted_msg)
    if msg == "{quit}":
        client_socket.close()
        top.destroy()

def on_closing(event=None):
    """This function is to be called when the window is closed."""
    my_msg.set("{quit}")
    send()

# GUI setup
def setup_gui():
    """Sets up the GUI components."""
    top.title("Chat Room")

    messages_frame = tk.Frame(top)
    scrollbar = tk.Scrollbar(messages_frame)  # To navigate through past messages.

    global msg_list
    msg_list = tk.Listbox(messages_frame, height=15, width=60, yscrollcommand=scrollbar.set)
    scrollbar.pack(side=tk.RIGHT, fill=tk.Y)
    msg_list.pack(side=tk.LEFT, fill=tk.BOTH, expand=True)
    messages_frame.pack()

    entry_field = tk.Entry(top, textvariable=my_msg)
    entry_field.bind("<Return>", send)
    entry_field.pack()

    send_button = tk.Button(top, text="Send", command=send)
    send_button.pack()

    top.protocol("WM_DELETE_WINDOW", on_closing)

# Network setup
def setup_network():
    """Sets up the network connection."""
    global client_socket, BUFSIZ, ADDR, key

    HOST = input('Enter host: ')
    PORT = input('Enter port: ')
    if not PORT:
        PORT = 33000
    else:
        PORT = int(PORT)

    BUFSIZ = 1024
    ADDR = (HOST, PORT)
    key = b'master key'  # 16-byte AES key

    client_socket = socket(AF_INET, SOCK_STREAM)
    client_socket.connect(ADDR)

    receive_thread = Thread(target=receive)
    receive_thread.start()

# Initialize main application window
top = tk.Tk()
my_msg = tk.StringVar()  # For the messages to be sent.
my_msg.set("<Type your message here>")

# Setup GUI and network
setup_gui()
setup_network()

# Start the GUI main loop
tk.mainloop()
