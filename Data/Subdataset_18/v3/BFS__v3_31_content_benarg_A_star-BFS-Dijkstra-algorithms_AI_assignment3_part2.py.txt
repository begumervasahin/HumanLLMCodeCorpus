import os
from PIL import Image, ImageDraw

# Constants for terrain costs and colors
TERRAIN_COSTS = {'w': 100, 'm': 50, 'f': 10, 'g': 5, 'r': 1, 'A': 1, 'B': 1}
TERRAIN_COLORS = {
    'w': (73, 216, 245), 'm': (99, 99, 99), 'f': (3, 82, 0),
    'g': (50, 200, 50), 'r': (114, 80, 41), 'A': (90, 180, 90), 'B': (255, 90, 90)
}

# Classes for State and SearchNode
class State:
    def __init__(self, x, y, terrain):
        self.x = x
        self.y = y
        self.terrain = terrain

class SearchNode:
    def __init__(self, state, cost):
        self.state = state
        self.cost = cost
        self.g = float('inf')
        self.h = 0
        self.f = float('inf')
        self.parent = None
        self.kids = []

# Functions for reading the board and setting up the search
def read_board(filename):
    with open(filename, 'r') as file:
        lines = file.read().strip().split('\n')

    board = []
    start, end = None, None
    for y, line in enumerate(lines):
        row = []
        for x, char in enumerate(line):
            node = SearchNode(State(x, len(lines) - y - 1, char), TERRAIN_COSTS[char])
            row.append(node)
            if char == 'A':
                start = (x, len(lines) - y - 1)
            elif char == 'B':
                end = (x, len(lines) - y - 1)
        board.append(row)
    board.reverse()
    return start, end, board

def heuristic(s1, s2):
    return abs(s2.x - s1.x) + abs(s2.y - s1.y)

def solution(s1, s2):
    return s1.x == s2.x and s1.y == s2.y

def generate_all_successors(s, board):
    successors = []
    if s.x > 0:
        successors.append(board[s.y][s.x - 1])
    if s.x < len(board[s.y]) - 1:
        successors.append(board[s.y][s.x + 1])
    if s.y > 0:
        successors.append(board[s.y - 1][s.x])
    if s.y < len(board) - 1:
        successors.append(board[s.y + 1][s.x])
    return successors

def attach_and_eval(child, parent, goal):
    child.parent = parent
    child.g = parent.g + child.cost
    child.h = heuristic(child.state, goal)
    child.f = child.g + child.h

def propagate_path_improvement(parent):
    for child in parent.kids:
        if parent.g + child.cost < child.g:
            attach_and_eval(child, parent, parent.state)
            propagate_path_improvement(child)

def draw_image(board, path, filename):
    img = Image.new('RGB', (len(board[0]) * 20, len(board) * 20), "white")
    draw = ImageDraw.Draw(img)

    for y, row in enumerate(board):
        for x, node in enumerate(row):
            draw.rectangle([(x * 20, y * 20), (x * 20 + 20, y * 20 + 20)], fill=TERRAIN_COLORS[node.state.terrain], outline=(0, 0, 0))
            if node in path:
                draw.rectangle([(x * 20 + 6, y * 20 + 6), (x * 20 + 14, y * 20 + 14)], fill=(107, 97, 255), outline=(0, 0, 0))
    
    img.save(filename, "PNG")

def visualize_path(start_node, end_node, board, filename):
    path = []
    current = end_node
    while current != start_node:
        path.append(current)
        current = current.parent
    path.append(start_node)
    path.reverse()
    draw_image(board, path, filename)

def best_first_search(start, end, board, filename):
    open_list = []
    closed_list = []
    start_node = board[start[1]][start[0]]
    end_node = board[end[1]][end[0]]
    start_node.g = 0
    start_node.h = heuristic(start_node.state, end_node.state)
    start_node.f = start_node.h
    open_list.append(start_node)

    while open_list:
        open_list.sort(key=lambda node: node.f)
        current_node = open_list.pop(0)
        closed_list.append(current_node)

        if solution(current_node.state, end_node.state):
            visualize_path(start_node, end_node, board, filename)
            return 'SUCCEEDED'

        successors = generate_all_successors(current_node.state, board)
        for successor in successors:
            current_node.kids.append(successor)
            if successor not in open_list and successor not in closed_list:
                attach_and_eval(successor, current_node, end_node.state)
                open_list.append(successor)
            elif current_node.g + successor.cost < successor.g:
                attach_and_eval(successor, current_node, end_node.state)
                if successor in closed_list:
                    propagate_path_improvement(successor)

    return 'FAILED'

if __name__ == "__main__":
    os.makedirs("assignment3_images", exist_ok=True)
    boards = ['board-2-1.txt', 'board-2-2.txt', 'board-2-3.txt', 'board-2-4.txt']
    for board_file in boards:
        start, end, board = read_board(f'boards/{board_file}')
        result = best_first_search(start, end, board, f'assignment3_images/part2_{board_file}')
        print(f'Result for {board_file}: {result}')
