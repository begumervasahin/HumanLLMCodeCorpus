from sklearn.metrics import confusion_matrix, accuracy_score, average_precision_score, roc_auc_score
from sklearn.naive_bayes import MultinomialNB

def train_and_evaluate_multinomial_nb(x_train, y_train, x_test, y_test, input_vector):
    """
    Trains and evaluates a Multinomial Naive Bayes model.
    Prints the confusion matrix, accuracy score, ROC AUC score, and average precision score.
    Predicts the class of a given input vector.
    
    Parameters:
    x_train (array-like): Training features
    y_train (array-like): Training labels
    x_test (array-like): Testing features
    y_test (array-like): Testing labels
    input_vector (array-like): Vector to predict class for
    
    Returns:
    str: "Positive" if the prediction for input_vector is 1, otherwise "Negative"
    """
    print("Multinomial Naive Bayes Model Evaluation")

    # Initialize and train the model
    model = MultinomialNB()
    model.fit(x_train, y_train)
    
    # Predict the test set
    y_pred = model.predict(x_test)
    
    # Evaluate the model
    print("Confusion Matrix:")
    print(confusion_matrix(y_test, y_pred))
    print(f"Accuracy Score: {accuracy_score(y_test, y_pred):.2f}")
    print(f"ROC AUC Score: {roc_auc_score(y_test, y_pred):.2f}")
    print(f"Average Precision Score: {average_precision_score(y_test, y_pred):.2f}")
    
    # Predict the class of the input vector
    prediction = model.predict(input_vector)
    return "Positive" if prediction == [1] else "Negative"

# Example usage
if __name__ == "__main__":
    # Sample data (replace with actual data)
    x_train = [[0, 1, 2], [1, 0, 3], [2, 1, 0]]
    y_train = [0, 1, 0]
    x_test = [[0, 1, 2], [2, 0, 1]]
    y_test = [0, 1]
    input_vector = [[1, 0, 2]]

    result = train_and_evaluate_multinomial_nb(x_train, y_train, x_test, y_test, input_vector)
    print("Prediction for input vector:", result)
