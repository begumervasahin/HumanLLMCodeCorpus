import json
import time
import requests

# Headers for the API requests
HEADERS = {
    'Content-Type': 'application/json',
    'Ocp-Apim-Subscription-Key': 'b3f5d9f8d81046598dedc07a7541e2c9',
}

def prepare_request_body(documents):
    """
    Prepares the request body for the API call.
    
    Parameters:
    documents (list of str): List of documents to be included in the request.
    
    Returns:
    dict: Request body for the API call.
    """
    return {
        'documents': [
            {'id': str(i), 'text': doc} 
            for i, doc in enumerate(documents) if doc
        ]
    }

def documents_to_sentiments(documents):
    """
    Sends a list of documents to the Azure Text Analytics API for sentiment analysis.
    
    Parameters:
    documents (list of str): List of documents to analyze.
    
    Returns:
    dict: Contains the request body and the API response.
    """
    request_body = prepare_request_body(documents)

    response = requests.post(
        'https://westus.api.cognitive.microsoft.com/text/analytics/v2.0/sentiment',
        headers=HEADERS,
        data=json.dumps(request_body)
    )
    
    return {
        'body': request_body,
        'result': response.json()
    }

def documents_to_topics(documents):
    """
    Sends a list of documents to the Azure Text Analytics API for topic detection.
    
    Parameters:
    documents (list of str): List of documents to analyze.
    
    Returns:
    dict: Contains the request body and the API response.
    """
    request_body = {
        'stopWords': [],
        'topicsToExclude': [],
        'documents': [
            {'id': str(i), 'text': doc}
            for i, doc in enumerate(documents)
        ]
    }

    response = requests.post(
        'https://westus.api.cognitive.microsoft.com/text/analytics/v2.0/topics',
        headers=HEADERS,
        data=json.dumps(request_body)
    )

    operation_location = response.headers['operation-location']
    operation_path = operation_location[operation_location.index('.com') + 4:]
    result = {'body': request_body}
    
    while True:
        time.sleep(5)  # Wait for 5 seconds before the next status check
        status_response = requests.get(
            'https://westus.api.cognitive.microsoft.com' + operation_path,
            headers=HEADERS
        )
        status_data = status_response.json()
        
        if status_data.get('status') == 'Succeeded':
            result['result'] = status_data
            break

    return result

# Example usage
if __name__ == "__main__":
    documents = ["This is a sample document.", "Another document here.", ""]  # Add your documents here
    print(documents_to_sentiments(documents))
    print(documents_to_topics(documents))
