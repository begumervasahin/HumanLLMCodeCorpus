from searcher import Searcher

class BFS(Searcher):
    """
    Breadth-First Search (BFS) class that extends the Searcher base class.
    """
    name = '2.A_bfs.txt'

    def merge_lists(self, list1, list2):
        """
        Merge two lists into a single list.

        Args:
            list1 (list): The first list to be merged.
            list2 (list): The second list to be merged.

        Returns:
            list: A new list containing all elements from both input lists.
        """
        return list1 + list2

def main():
    """
    Main function to demonstrate the functionality of the BFS class.
    """
    # Create an instance of the BFS class
    bfs_instance = BFS()
    
    # Define two lists to be merged
    list1 = [1, 2, 3]
    list2 = [4, 5, 6]
    
    # Merge the lists using the BFS instance
    merged_list = bfs_instance.merge_lists(list1, list2)
    
    # Print the results
    print("Merged List:", merged_list)
    print("BFS Name:", bfs_instance.name)

if __name__ == "__main__":
    main()
