import itertools
from solver import Solver
from board import Board

class IDDFS(Solver):
    """
    Iterative Deepening Depth-First Search (IDDFS) solver for the 8-puzzle problem.
    """
    def __init__(self, initial_state):
        super(IDDFS, self).__init__(initial_state)
        self.frontier = []
        self.explored_nodes = set()
        self.max_depth = 0

    def dls(self, limit):
        """
        Depth-Limited Search (DLS) function to explore nodes up to a given depth limit.

        Args:
            limit (int): The depth limit for the search.

        Returns:
            list: The solution path if found, otherwise None.
        """
        self.frontier.append(self.initial_state)
        while self.frontier:
            board = self.frontier.pop()
            self.explored_nodes.add(tuple(board.state))
            if board.goal_test():
                self.set_solution(board)
                return self.solution
            if board.depth < limit:
                for neighbor in reversed(board.neighbors()):
                    if tuple(neighbor.state) not in self.explored_nodes:
                        self.frontier.append(neighbor)
                        self.explored_nodes.add(tuple(neighbor.state))
                        self.max_depth = max(self.max_depth, neighbor.depth)
        return None

    def solve(self):
        """
        Solve the puzzle using the IDDFS algorithm.

        Returns:
            list: The solution path if found.
        """
        for depth in itertools.count():
            self.frontier = []
            self.explored_nodes = set()
            self.max_depth = 0
            solution = self.dls(depth)
            if solution is not None:
                return solution
        return self.solution

if __name__ == "__main__":
    # Example usage of IDDFS class
    initial_state = Board([1, 2, 5, 3, 4, 0, 6, 7, 8])
    solver = IDDFS(initial_state)
    solution = solver.solve()

    if solution:
        print("Solution found:")
        for step in solution:
            print(step)
    else:
        print("No solution found.")
