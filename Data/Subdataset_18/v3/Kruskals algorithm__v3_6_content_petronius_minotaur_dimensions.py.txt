# Color tuples with their corresponding RGB values
COLORS = [

def get_message(dimension, direction):
    """
    Retrieve the message based on the dimension and direction.
    
    Args:
        dimension (int): The dimension index (0-based).
        direction (int): The direction (1 for forward, -1 for backward).
    
    Returns:
        str: The corresponding message if found, otherwise None.
    """
    if 0 <= dimension < len(MESSAGES):
        return MESSAGES[dimension].get(direction)
    return None

def get_color_rgb(index):
    """
    Retrieve the RGB value of a color by its index.
    
    Args:
        index (int): The index of the color in the COLORS list.
    
    Returns:
        tuple: The RGB value corresponding to the color.
    """
    if 0 <= index < len(COLORS):
        return COLORS[index][1]
    return None
