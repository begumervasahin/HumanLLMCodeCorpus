import copy

# Constants (define MIN_CONTOUR_AREA according to your needs)
MIN_CONTOUR_AREA = 100  # Example value

class ContourWithData:
    """
    A class to represent contour data with related properties and methods.
    """
    def __init__(self):
        self.npaContour = None
        self.boundingRect = None
        self.intRectX = 0
        self.intRectY = 0
        self.intRectWidth = 0
        self.intRectHeight = 0
        self.fltArea = 0.0

    def calculateRectTopLeftPointAndWidthAndHeight(self):
        """
        Calculate the top-left point, width, and height of the bounding rectangle.
        """
        [intX, intY, intWidth, intHeight] = self.boundingRect
        self.intRectX = intX
        self.intRectY = intY
        self.intRectWidth = intWidth
        self.intRectHeight = intHeight

    def checkIfContourIsValid(self):
        """
        Check if the contour is valid based on its area.
        
        Returns:
        bool: True if the contour area is above the minimum threshold, False otherwise.
        """
        return self.fltArea >= MIN_CONTOUR_AREA

    def adjustIntRectY(self, adjustment):
        """
        Adjust the y-coordinate of the top-left point of the bounding rectangle.
        
        Parameters:
        adjustment (int): The value to adjust the y-coordinate by.
        """
        self.intRectY -= adjustment

def ndlist(init, *dims):
    """
    Create a multi-dimensional list initialized with a given value.
    
    Parameters:
    init: The initial value to fill the list.
    *dims: The dimensions of the list.
    
    Returns:
    list: A multi-dimensional list.
    """
    if not dims:
        return init
    return [ndlist(init, *dims[1:]) for _ in range(dims[0])]

if __name__ == "__main__":
    # Create a ContourWithData object
    contour = ContourWithData()
    contour.boundingRect = [50, 100, 200, 300]  # Example values
    contour.fltArea = 150.0  # Example area
    
    # Calculate rectangle properties
    contour.calculateRectTopLeftPointAndWidthAndHeight()
    print(f"Rectangle X: {contour.intRectX}, Y: {contour.intRectY}, Width: {contour.intRectWidth}, Height: {contour.intRectHeight}")
    
    # Check if contour is valid
    is_valid = contour.checkIfContourIsValid()
    print(f"Is Contour Valid? {'Yes' if is_valid else 'No'}")
    
    # Adjust the intRectY
    contour.adjustIntRectY(555)
    print(f"Adjusted Rectangle Y: {contour.intRectY}")
