def get_target_input():
    """Prompt the user to enter an integer and validate the input."""
    while True:
        try:
            return int(input("Enter an integer to search: "))
        except ValueError:
            print("Invalid input. Please enter an integer.")


def read_and_process_number_list(filename='numberlist.txt'):
    """Read a list of numbers from a file and return it as a sorted list of integers."""
    with open(filename, 'r') as file:
        number_list = file.read().split(", ")
    return sorted(map(int, number_list))


def perform_binary_search(numbers, target):
    """Perform a binary search for the target in the sorted list of numbers."""
    low, high = 0, len(numbers) - 1

    while low <= high:
        mid = (low + high) // 2
        if numbers[mid] == target:
            return True
        elif numbers[mid] < target:
            low = mid + 1
        else:
            high = mid - 1
    return False


def ask_retry():
    """Ask the user if they want to search again and validate the input."""
    while True:
        retry = input("Would you like to search for another number? (y/n): ").strip().lower()
        if retry in ('y', 'n'):
            return retry == 'y'
        print("Invalid input. Please enter 'y' for yes or 'n' for no.")


def binary_search():
    """Main function to perform the binary search."""
    while True:
        target = get_target_input()
        numbers = read_and_process_number_list()

        if perform_binary_search(numbers, target):
            print(f"Target {target} found in the list!")
        else:
            print(f"Target {target} not found in the list.")

        if not ask_retry():
            print("Exiting program. Goodbye!")
            break


# Start the binary search
binary_search()
