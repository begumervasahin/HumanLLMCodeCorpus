import numpy as np
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error, r2_score
from sklearn.preprocessing import PolynomialFeatures

def load_data():
    """Load the observed data."""
    years = np.array([1961, 1962, 1963, 1964, 1965, 1966, 1967, 1968, 1969, 1970, 1971, 1972, 1973, 1974, 1975, 1976, 1977, 1978, 1979, 1980, 1981, 1982, 1983, 1984, 1985, 1986, 1987, 1988, 1989, 1990, 1991, 1992, 1993, 1994, 1995, 1996, 1997, 1998, 1999, 2000, 2001, 2002, 2003, 2004, 2005, 2006, 2007, 2008, 2009, 2010, 2011, 2012, 2013, 2014])
    values = np.array([1.4, 1.4, 1.426, 1.4257, 1.44, 1.44, 1.4144, 1.7064, 1.7177, 1.4757, 1.34, 1.5652, 1.617, 1.617, 1.6116, 1.6598, 1.5833, 1.6183, 1.6102, 1.6711, 1.669, 1.6658, 1.6641, 1.6612, 1.6652, 1.6875, 1.702, 1.7034, 1.6963, 1.7582, 1.68, 1.5, 1.6786, 1.6667, 1.6563, 1.6364, 1.5, 1.65, 1.5, 1.1979, 2, 2.98, 0.84, 1.6, 1.2069, 2.6204, 2.6277, 2.6277, 2.1429, 1.6448, 1.64, 2.1986, 2.1972, 2.4882])
    return years, values

def preprocess_data(years, values):
    """Reshape data for the model and transform to polynomial features."""
    x = years[:, np.newaxis]
    y = values[:, np.newaxis]
    polynomial_features = PolynomialFeatures(degree=2)
    x_poly = polynomial_features.fit_transform(x)
    return x, y, x_poly, polynomial_features

def train_model(x_poly, y):
    """Train the linear regression model on polynomial features."""
    model = LinearRegression()
    model.fit(x_poly, y)
    y_poly_pred = model.predict(x_poly)
    return model, y_poly_pred

def evaluate_model(y, y_poly_pred):
    """Evaluate model performance."""
    rmse = mean_squared_error(y, y_poly_pred)
    r2 = r2_score(y, y_poly_pred)
    print(f'The RMSE of the polynomial regression model is {rmse}')
    print(f'The R2 score of the polynomial regression model is {r2}')

def predict_future_values(model, polynomial_features):
    """Predict future values."""
    future_years = np.array([2015, 2016, 2017, 2018, 2019])
    future_years_poly = polynomial_features.fit_transform(future_years[:, np.newaxis])
    y_future_pred = model.predict(future_years_poly)
    print(f'Predicted values for future years: {y_future_pred.flatten()}')
    return future_years, y_future_pred

def plot_results(x, y, future_years, y_future_pred, y_poly_pred):
    """Plot the results."""
    x_combined = np.concatenate((x.flatten(), future_years))
    y_combined = np.concatenate((y_poly_pred.flatten(), y_future_pred.flatten()))

    plt.scatter(x, y, s=10, color='c', label='Observed Data')
    plt.scatter(future_years, y_future_pred, s=10, color='r', label='Predicted Data')
    plt.plot(x_combined, y_combined, color='m', label='Polynomial Regression (degree 2)')
    plt.xlabel('Year')
    plt.ylabel('Value')
    plt.title('Polynomial Regression')
    plt.legend()
    plt.show()

def main():
    years, values = load_data()
    x, y, x_poly, polynomial_features = preprocess_data(years, values)
    model, y_poly_pred = train_model(x_poly, y)
    evaluate_model(y, y_poly_pred)
    future_years, y_future_pred = predict_future_values(model, polynomial_features)
    plot_results(x, y, future_years, y_future_pred, y_poly_pred)

if __name__ == "__main__":
    main()
