import numpy as np
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
import pandas as pd

# Load the dataset
dataset = pd.read_csv("Churn_Modelling.csv")

# Select relevant features: Balance and Estimated Salary
features = dataset.iloc[:, [8, 12]].values

# Function to perform the Elbow Method
def find_optimal_clusters(data, max_clusters):
    wcss = []  # Within-cluster sum of squares
    for num_clusters in range(1, max_clusters + 1):
        kmeans = KMeans(n_clusters=num_clusters, init='k-means++', random_state=0)
        kmeans.fit(data)
        wcss.append(kmeans.inertia_)
    return wcss

# Determine the optimal number of clusters
max_clusters = 15
wcss = find_optimal_clusters(features, max_clusters)

# Plot the Elbow Method to visualize the optimal number of clusters
def plot_elbow_method(wcss, max_clusters):
    plt.figure(figsize=(10, 6))
    plt.plot(range(1, max_clusters + 1), wcss, marker='o')
    plt.title('Elbow Method')
    plt.xlabel('Number of Clusters')
    plt.ylabel('WCSS')
    plt.grid(True)
    plt.show()

plot_elbow_method(wcss, max_clusters)

# Apply k-means clustering with the optimal number of clusters
optimal_num_clusters = 4
kmeans = KMeans(n_clusters=optimal_num_clusters, init='k-means++', random_state=0)
cluster_labels = kmeans.fit_predict(features)

# Function to visualize the clusters
def plot_clusters(features, cluster_labels, centroids):
    plt.figure(figsize=(10, 6))
    colors = ['red', 'blue', 'green', 'pink']
    for i in range(len(np.unique(cluster_labels))):
        plt.scatter(features[cluster_labels == i, 0], features[cluster_labels == i, 1], 
                    s=25, c=colors[i], label=f'Cluster {i + 1}')
    plt.scatter(centroids[:, 0], centroids[:, 1], 
                s=100, c='yellow', label='Centroids', edgecolor='black')
    plt.title('K-Means Clustering')
    plt.xlabel('Balance')
    plt.ylabel('Estimated Salary in $')
    plt.legend()
    plt.grid(True)
    plt.show()

# Plot the clusters
plot_clusters(features, cluster_labels, kmeans.cluster_centers_)
