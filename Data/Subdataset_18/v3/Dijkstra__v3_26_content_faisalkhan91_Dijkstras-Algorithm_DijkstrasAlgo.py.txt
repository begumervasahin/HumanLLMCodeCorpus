# Define the nodes and distances for the graph
nodes = ('A', 'B', 'C', 'D', 'E', 'F', 'G')
distances = {
    'A': {'B': 5, 'D': 3, 'E': 12, 'F': 5},
    'B': {'A': 5, 'D': 1, 'G': 2},
    'C': {'G': 2, 'E': 1, 'F': 16},
    'D': {'A': 3, 'B': 1, 'E': 1, 'G': 1},
    'E': {'A': 12, 'C': 1, 'D': 1, 'F': 2},
    'F': {'A': 5, 'C': 16, 'E': 2},
    'G': {'B': 2, 'C': 2, 'D': 1}
}

def dijkstra(nodes, distances, start):
    """
    Implements Dijkstra's algorithm to find the shortest path from the start node to all other nodes in the graph.

    Parameters:
    nodes (tuple): A tuple containing all the nodes in the graph.
    distances (dict): A dictionary representing the graph with adjacency lists and edge weights.
    start (str): The starting node.

    Returns:
    dict: A dictionary containing the shortest distances from the start node to each other node.
    """
    # Initialize dictionaries for unvisited and visited nodes
    unvisited = {node: float('inf') for node in nodes}
    visited = {}
    current_node = start
    current_distance = 0
    unvisited[current_node] = current_distance

    while unvisited:
        # Update the distances to neighboring nodes
        for neighbor, distance in distances[current_node].items():
            new_distance = current_distance + distance
            if new_distance < unvisited[neighbor]:
                unvisited[neighbor] = new_distance

        # Move the current node to the visited set
        visited[current_node] = current_distance
        del unvisited[current_node]

        # Exit if all nodes have been visited
        if not unvisited:
            break

        # Select the unvisited node with the smallest known distance
        current_node, current_distance = min(unvisited.items(), key=lambda x: x[1])

    return visited

def main():
    start_node = 'B'
    shortest_path_tree = dijkstra(nodes, distances, start_node)
    print(f"\nThe shortest path tree using Dijkstra's algorithm starting from node '{start_node}':")
    print(shortest_path_tree)

if __name__ == "__main__":
    main()
