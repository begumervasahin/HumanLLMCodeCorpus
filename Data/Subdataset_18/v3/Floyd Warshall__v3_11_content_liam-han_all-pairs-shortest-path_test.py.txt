import argparse
import re
import sys
import time
import math

# Compile regular expressions for graph and edge parsing
GRAPH_RE = re.compile(r"(\d+)\s(\d+)")
EDGE_RE = re.compile(r"(\d+)\s(\d+)\s(-?\d+)")

class EdgeNode:
    """Class to represent an edge in the graph."""
    def __init__(self, u, v, w):
        self.u = u
        self.v = v
        self.w = w

    def __repr__(self):
        return f"(({self.u}, {self.v}), {self.w})"

def read_file(filename):
    """Reads the input file and parses the graph."""
    with open(filename, 'r') as infile:
        first_line = infile.readline()
        graph_match = GRAPH_RE.match(first_line)
        if not graph_match:
            print(f"{first_line.strip()} not properly formatted")
            sys.exit(1)
        
        num_vertices = int(graph_match.group(1))
        vertices = list(range(num_vertices))
        edges = [[float("inf")] * num_vertices for _ in range(num_vertices)]

        for line in infile:
            edge_match = EDGE_RE.match(line.strip())
            if edge_match:
                source, sink, weight = map(int, edge_match.groups())
                if source >= num_vertices or sink >= num_vertices:
                    print(f"Attempting to insert an edge between {source} and {sink} in a graph with {num_vertices} vertices")
                    sys.exit(1)
                edges[source - 1][sink - 1] = weight

    return vertices, edges

def bellman_ford(graph):
    """Bellman-Ford algorithm for finding shortest paths."""
    vertices, edges = graph
    path_pairs = []
    edge_list = []

    for i in range(len(vertices)):
        for j in range(len(vertices)):
            if not math.isinf(edges[i][j]):
                edge_list.append(EdgeNode(i, j, edges[i][j]))

    for src in range(len(vertices)):
        dist = [float("inf")] * len(vertices)
        dist[src] = 0

        for _ in range(len(vertices) - 1):
            for edge in edge_list:
                if dist[edge.v] > dist[edge.u] + edge.w:
                    dist[edge.v] = dist[edge.u] + edge.w

        for edge in edge_list:
            if dist[edge.v] > dist[edge.u] + edge.w:
                print("There is a negative cycle")
                return []

        for dest in range(len(vertices)):
            path_pairs.append(EdgeNode(src, dest, dist[dest]))

    return path_pairs

def floyd_warshall(graph):
    """Floyd-Warshall algorithm for finding shortest paths."""
    vertices, edges = graph
    path_pairs = []
    dist = [[edges[i][j] for j in range(len(vertices))] for i in range(len(vertices))]

    for i in range(len(vertices)):
        dist[i][i] = 0

    for k in range(len(vertices)):
        for i in range(len(vertices)):
            for j in range(len(vertices)):
                if dist[i][j] > dist[i][k] + dist[k][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]

    for i in range(len(vertices)):
        for j in range(len(vertices)):
            path_pairs.append(EdgeNode(i, j, dist[i][j]))

    return path_pairs

def main(filename, algorithm):
    """Main function to execute the algorithms based on user input."""
    graph = read_file(filename)

    if algorithm == 'b':
        result = bellman_ford(graph)
        print(result)
    elif algorithm == 'f':
        result = floyd_warshall(graph)
        print(result)
    elif algorithm == 'both':
        start = time.perf_counter()
        result_bf = bellman_ford(graph)
        bf_time = time.perf_counter() - start
        print(result_bf)
        print(f"Bellman-Ford timing: {bf_time:.6f} seconds")

        start = time.perf_counter()
        result_fw = floyd_warshall(graph)
        fw_time = time.perf_counter() - start
        print(result_fw)
        print(f"Floyd-Warshall timing: {fw_time:.6f} seconds")
    else:
        print("Unknown algorithm selected. Use 'b' for Bellman-Ford, 'f' for Floyd-Warshall, or 'both' for both.")

if __name__ == '__main__':
    parser = argparse.ArgumentParser(description='Calculate the shortest path between all pairs of vertices in a graph')
    parser.add_argument('--algorithm', required=True, choices=['b', 'f', 'both'], help='Algorithm to run: (b)ellman-ford, (f)loyd-warshall, or (both)')
    parser.add_argument('filename', metavar='<filename>', help='Input file containing graph')
    
    args = parser.parse_args()
    main(args.filename, args.algorithm)
