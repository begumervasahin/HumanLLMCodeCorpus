import collections

class Dijkstra:
    """Class to represent a graph and perform Dijkstra's algorithm."""

    def __init__(self, *nodes):
        """Initialize the graph with given nodes."""
        self.nodes = list(nodes)
        self.distances = {node: float('inf') for node in self.nodes}
        self.edges = collections.defaultdict(dict)

    def add_edge(self, src, dest, cost):
        """Add an edge with a cost between two nodes."""
        self.edges[src][dest] = cost

    def calculate(self, source):
        """Calculate shortest paths from the source node to all other nodes."""
        self.distances[source] = 0
        visited = set()
        nodes_to_visit = [source]

        while nodes_to_visit:
            current_node = nodes_to_visit.pop(0)
            if current_node in visited:
                continue
            visited.add(current_node)

            for neighbor, cost in self.edges[current_node].items():
                new_distance = self.distances[current_node] + cost
                if new_distance < self.distances[neighbor]:
                    self.distances[neighbor] = new_distance
                    nodes_to_visit.append(neighbor)

        return self.distances

def main():
    """Main function to demonstrate Dijkstra's algorithm."""
    # Create a graph with nodes 'a', 'b', 'c', 'd', 'e'
    graph = Dijkstra('a', 'b', 'c', 'd', 'e')
    
    # Add edges between the nodes
    graph.add_edge('a', 'b', 1)
    graph.add_edge('a', 'c', 4)
    graph.add_edge('b', 'c', 2)
    graph.add_edge('b', 'd', 5)
    graph.add_edge('c', 'd', 1)
    graph.add_edge('d', 'e', 3)
    
    # Calculate shortest paths from node 'a'
    shortest_paths = graph.calculate('a')
    
    print("Shortest paths from node 'a':")
    for node, distance in shortest_paths.items():
        print(f"Distance to {node}: {distance}")

if __name__ == '__main__':
    main()
