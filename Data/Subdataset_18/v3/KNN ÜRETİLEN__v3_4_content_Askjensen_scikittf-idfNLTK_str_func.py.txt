import base64
import csv
from collections import defaultdict
from stop_words import get_stop_words
from sklearn.feature_extraction.text import TfidfVectorizer
import gensim


def cosine_sim(text1, text2):
    """
    Calculate the cosine similarity between two texts.

    Args:
    text1 (str): First text.
    text2 (str): Second text.

    Returns:
    float: Cosine similarity between text1 and text2.
    """
    vectorizer = TfidfVectorizer(min_df=1, stop_words='danish')
    tfidf = vectorizer.fit_transform([text1, text2])
    return (tfidf * tfidf.T).A[0, 1]


def write_csv_file_from_text_dict(outpath, datadict):
    """
    Write a CSV file from a dictionary of text data.

    Args:
    outpath (str): Output file path.
    datadict (dict): Dictionary containing text data.
    """
    with open(outpath, "w", newline='', encoding='utf-8') as outfile:
        csv_file = csv.writer(outfile, delimiter=';', dialect='excel')
        for i, text in datadict.items():
            row = [str(i), text.encode('utf-8'), str(datadict.dfs[i])]
            csv_file.writerow(row)


def filter_words_in_documents(documents):
    """
    Filter words in a list of documents by removing stopwords and infrequent words.

    Args:
    documents (list of str): List of documents.

    Returns:
    list of list of str: List of filtered documents.
    """
    stopwords = get_stop_words('danish')
    texts = [
        [word for word in document.lower().split() if word not in stopwords]
        for document in documents
    ]
    
    frequency = defaultdict(int)
    for text in texts:
        for token in text:
            frequency[token] += 1

    filtered_texts = [
        [token for token in text if frequency[token] > 1]
        for text in texts
    ]

    return [text for text in filtered_texts if text]


def create_dictionary(texts):
    """
    Create a Gensim dictionary from texts.

    Args:
    texts (list of list of str): List of tokenized texts.

    Returns:
    gensim.corpora.Dictionary: Gensim dictionary.
    """
    return gensim.corpora.Dictionary(texts)


def create_corpus(dictionary, texts):
    """
    Create a corpus from a dictionary and texts.

    Args:
    dictionary (gensim.corpora.Dictionary): Gensim dictionary.
    texts (list of list of str): List of tokenized texts.

    Returns:
    list of list of (int, int): Corpus.
    """
    return [dictionary.doc2bow(text) for text in texts]


def decrypt_text(text, cipher):
    """
    Decrypt a base64 encoded text using a cipher.

    Args:
    text (str): Encrypted text.
    cipher (Cipher): Decryption cipher.

    Returns:
    str: Decrypted text.
    """
    try:
        return cipher.decrypt(base64.b64decode(text))
    except Exception as e:
        print(f"Decryption failed: {e}")
        return ''
