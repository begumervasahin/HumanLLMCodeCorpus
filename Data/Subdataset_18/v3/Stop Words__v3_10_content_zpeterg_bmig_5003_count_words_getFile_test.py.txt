import unittest
from time import time

def getFile(filename, callback):
    """
    Reads a file and applies a callback to each word.

    Parameters:
    filename (str): The name of the file to read.
    callback (function): The callback function to apply to each word.

    Returns:
    list: A list of processed words.

    Raises:
    FileNotFoundError: If the file is not found.
    """
    try:
        with open(filename, 'r') as file:
            words = file.read().split()
        return [callback(word) for word in words if callback(word) is not None]
    except FileNotFoundError:
        raise FileNotFoundError(f"File {filename} not found")

# Sample list of words for comparison in tests
words = [
    'Fish', 'hat', 'foo', 'cow', 'Cow', 'siamese.', 'Wonderland', 'foo',
    'toothpaste', 'bar', 'umbrella', 'foo\n', 'milky', '\'"flight-manual"\'',
    'toothpick', 'bar', 'enough', 'event-horizon', 'bar!'
]

def on_word(word):
    """
    Callback function to append 'hi' to each word, except for 'umbrella'.

    Parameters:
    word (str): The word to process.

    Returns:
    str: The processed word, or None if the word is 'umbrella'.
    """
    if word == 'umbrella':
        return None
    return word + 'hi'

class GetFileTest(unittest.TestCase):
    """Unit test class for the getFile function."""

    def test_getFile_basic(self):
        """Test getFile with basic functionality."""
        self.assertEqual(words, getFile('small_test.txt', lambda a: a))
    
    def test_getFile_no_file(self):
        """Test getFile with a non-existent file."""
        self.assertRaises(FileNotFoundError, getFile, 'aaa.txt', lambda a: a)
    
    def test_getFile_with_callback(self):
        """Test getFile with the on_word callback."""
        expected_result = [
            'Fishhi', 'hathi', 'foohi', 'cowhi', 'Cowhi', 'siamese.hi',
            'Wonderlandhi', 'foohi', 'toothpastehi', 'barhi'
        ]
        self.assertEqual(expected_result, getFile('small_test.txt', on_word))
    
    def test_getFile_moby(self):
        """Test getFile with a large file and measure time taken."""
        start_time = time()
        self.assertLess(100000, len(getFile('moby_test.txt', on_word)))
        end_time = time()
        print('Moby import time (seconds):', end_time - start_time)

if __name__ == '__main__':
    unittest.main()
