def selection_sort(arr):
    """
    Perform selection sort on the given array, counting the number of comparisons and swaps.
    """
    comparisons = 0
    swaps = 0
    
    n = len(arr)
    for i in range(n - 1):
        min_idx = i
        
        # Find the minimum element in the unsorted portion of the array
        for j in range(i + 1, n):
            comparisons += 1
            if arr[j] < arr[min_idx]:
                min_idx = j
        
        # Swap the found minimum element with the first element
        if min_idx != i:
            arr[i], arr[min_idx] = arr[min_idx], arr[i]
            swaps += 1
    
    return comparisons, swaps

def main():
    """
    Main function to demonstrate selection sort.
    """
    # Example array to be sorted
    arr = [64, 25, 12, 22, 11]
    
    # Print the array before sorting
    print("Array before sorting:")
    print(arr)
    
    # Sort the array using selection sort
    comparisons, swaps = selection_sort(arr)
    
    # Print the sorted array
    print("Sorted array is:")
    print(arr)
    
    # Print the number of comparisons and swaps
    print(f"\nSELECTION SORT:\nComparisons: {comparisons}\nSwaps: {swaps}")

if __name__ == "__main__":
    main()
