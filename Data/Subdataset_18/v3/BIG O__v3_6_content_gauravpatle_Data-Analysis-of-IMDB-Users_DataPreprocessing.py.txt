import pandas as pd

def load_data(file_path):
    """
    Load the data from the specified file.
    
    Args:
        file_path (str): The path to the data file.
    
    Returns:
        pd.DataFrame: The loaded data as a DataFrame.
    """
    return pd.read_csv(file_path, sep='\t', encoding='unicode_escape', header=None)

def extract_genre_columns(data):
    """
    Extract the genre columns from the data and assign appropriate column names.
    
    Args:
        data (pd.DataFrame): The original data.
    
    Returns:
        pd.DataFrame: A DataFrame containing only the genre columns.
    """
    genre_columns = [
        "unknown", "Action", "Adventure", "Animation", "Children", "Comedy",
        "Crime", "Documentary", "Drama", "Fantasy", "Film_Noir", "Horror",
        "Musical", "Mystery", "Romance", "Sci_Fi", "Thriller", "War", "Western"
    ]
    genre_data = data.iloc[:, 4:23]
    genre_data.columns = genre_columns
    return genre_data

def concatenate_genres(genre_data):
    """
    Concatenate genre names for each row where the genre indicator is 1.
    
    Args:
        genre_data (pd.DataFrame): The DataFrame containing genre columns.
    
    Returns:
        pd.Series: A Series with concatenated genre names.
    """
    genre_data['GENRE'] = ""
    for index, row in genre_data.iterrows():
        genres = [col for col in genre_data.columns if row[col] == 1]
        genre_data.at[index, 'GENRE'] = ','.join(genres)
    return genre_data['GENRE']

def combine_data(original_data, genre_column):
    """
    Combine the first three columns of the original data with the new GENRE column.
    
    Args:
        original_data (pd.DataFrame): The original data.
        genre_column (pd.Series): The concatenated genre column.
    
    Returns:
        pd.DataFrame: The final combined data.
    """
    data_part1 = original_data.iloc[:, 0:3]
    return pd.concat([data_part1, genre_column], axis=1)

def save_data(data, output_file):
    """
    Save the processed data to a new file.
    
    Args:
        data (pd.DataFrame): The final combined data.
        output_file (str): The path to the output file.
    """
    data.to_csv(output_file, sep='\t', index=False, header=False)

def main():
    input_file = 'u_item_cleaned.txt'
    output_file = 'movie_details.csv'

    # Load the data
    data = load_data(input_file)

    # Extract and process genre columns
    genre_data = extract_genre_columns(data)
    genre_column = concatenate_genres(genre_data)

    # Combine the data and save to file
    final_data = combine_data(data, genre_column)
    save_data(final_data, output_file)

if __name__ == "__main__":
    main()
