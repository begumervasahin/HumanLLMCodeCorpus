import matplotlib.pyplot as plt
import networkx as nx
import sys

def display_graph(mst_list, vert_list):
    """Display the Minimum Spanning Tree using NetworkX and Matplotlib."""
    graph = nx.Graph()

    for edge in mst_list:
        graph.add_edge(vert_list[edge[0]], vert_list[edge[1]], weight=int(edge[2]))

    pos = nx.spring_layout(graph, k=20, iterations=150, weight='weight', scale=1.0)
    edge_labels = {(u, v): str(d['weight']) for u, v, d in graph.edges(data=True)}

    nx.draw_networkx_edge_labels(graph, pos, edge_labels=edge_labels)
    node_size = max(len(vert_list) * 180, 100)
    nx.draw_networkx_nodes(graph, pos, node_size=node_size, node_shape='h', alpha=0.5)
    nx.draw_networkx_edges(graph, pos, edgelist=graph.edges, width=2, edge_color='b', alpha=0.5)
    nx.draw_networkx_labels(graph, pos, font_size=9, font_family='sans-serif')

    plt.axis('off')
    plt.show()

def sort_edges(weighted_graph):
    """Sort edges based on their weight."""
    return sorted(weighted_graph, key=lambda x: x[2])

def find(parent, vertex):
    """Find the root of the set in which the vertex is located."""
    if parent[vertex] != vertex:
        parent[vertex] = find(parent, parent[vertex])
    return parent[vertex]

def union(parent, rank, root1, root2):
    """Perform the union of two sets."""
    if rank[root1] > rank[root2]:
        parent[root2] = root1
    elif rank[root1] < rank[root2]:
        parent[root1] = root2
    else:
        parent[root2] = root1
        rank[root1] += 1

def kruskal(weighted_graph, vertex_count):
    """Kruskal's algorithm to find the Minimum Spanning Tree."""
    mst_list = []
    sorted_graph = sort_edges(weighted_graph)
    parent = list(range(vertex_count))
    rank = [0] * vertex_count

    for edge in sorted_graph:
        u, v, weight = edge
        root_u = find(parent, u)
        root_v = find(parent, v)
        if root_u != root_v:
            mst_list.append(edge)
            union(parent, rank, root_u, root_v)
            if len(mst_list) == vertex_count - 1:
                break

    return mst_list

def read_graph_from_file(file_name):
    """Read graph edges from a file."""
    with open(file_name) as file:
        edges = [line.strip().split() for line in file]

    vertices = {edge[0] for edge in edges} | {edge[1] for edge in edges}
    vertex_list = list(vertices)
    weighted_graph = [[vertex_list.index(edge[0]), vertex_list.index(edge[1]), int(edge[2])] for edge in edges]

    return vertex_list, weighted_graph

def main(file_name):
    """Main function to read input file and find/display the MST."""
    vertex_list, weighted_graph = read_graph_from_file(file_name)
    mst_list = kruskal(weighted_graph, len(vertex_list))
    total_weight = sum(edge[2] for edge in mst_list)

    print("The minimum spanning tree is as follows:")
    for edge in mst_list:
        print(f"{vertex_list[edge[0]]} to {vertex_list[edge[1]]} = {edge[2]} miles")
    print(f"Total weight: {total_weight} miles.")

    display_graph(mst_list, vertex_list)

if __name__ == "__main__":
    if len(sys.argv) != 2:
        print("Usage: python3 prim.py city-pairs.txt")
        sys.exit(1)
    main(sys.argv[1])
