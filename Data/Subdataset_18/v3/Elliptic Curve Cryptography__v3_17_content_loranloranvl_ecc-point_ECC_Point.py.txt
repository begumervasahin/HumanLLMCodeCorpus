BASE = 23
INFINITE_POINT = (BASE, BASE)

# Generate the list of points on the curve
F_BASE = [
    (x, y) for x in range(BASE) for y in range(BASE)
    if (x ** 3 + x * 2 + 7) % BASE == y * y % BASE
]

def get_reciprocal(x):
    """Find the multiplicative inverse of x modulo BASE."""
    for i in range(BASE):
        if (i * x) % BASE == 1:
            return i
    raise ValueError('Reciprocal not found')

class ECC_Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y
        if self.val() not in F_BASE and self.val() != INFINITE_POINT:
            raise ValueError('Invalid initialization')

    def val(self):
        """Return the point as a tuple."""
        return (self.x, self.y)

    def show_val(self):
        """Print the coordinates of the point."""
        print(self.val())

    def __add__(self, other):
        """Add two points on the elliptic curve."""
        if self.val() == INFINITE_POINT:
            return other
        if other.val() == INFINITE_POINT:
            return self

        x1, y1 = self.val()
        x2, y2 = other.val()

        if x1 == x2 and (y1 + y2) % BASE == 0:
            return ECC_Point(*INFINITE_POINT)

        if self.val() == other.val():
            lamda = (3 * x1 ** 2 + 2) * get_reciprocal(2 * y1) % BASE
        else:
            lamda = (y2 - y1) * get_reciprocal(x2 - x1) % BASE

        x3 = (lamda ** 2 - x1 - x2) % BASE
        y3 = (lamda * (x1 - x3) - y1) % BASE

        return ECC_Point(x3, y3)

    def __mul__(self, n):
        """Multiply a point by an integer."""
        result = ECC_Point(self.x, self.y)
        for _ in range(n - 1):
            result += self
        return result

# Example usage
def main():
    point1 = ECC_Point(3, 10)
    point2 = ECC_Point(9, 7)

    # Adding two points
    sum_point = point1 + point2
    sum_point.show_val()

    # Multiplying a point by a scalar
    product_point = point1 * 3
    product_point.show_val()

if __name__ == "__main__":
    main()
