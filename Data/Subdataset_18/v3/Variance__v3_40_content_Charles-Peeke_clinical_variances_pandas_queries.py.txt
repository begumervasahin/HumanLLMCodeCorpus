from dotenv import load_dotenv
import os
import psycopg2
import pandas as pd
import time

# Initialize global variables for the database connection
conn = None

def load_environment_variables():
    """
    Load database connection parameters from environment variables.
    """
    global hostname, port, database, username, password
    hostname = os.getenv('hostname')
    port = os.getenv('port')
    database = os.getenv('database')
    username = os.getenv('username')
    password = os.getenv('password')

def connect_to_database():
    """
    Establish a connection to the database using the loaded environment variables.
    """
    global conn
    conn = psycopg2.connect(
        host=hostname, 
        database=database, 
        user=username, 
        password=password
    )

def fetch_table_data(table_name):
    """
    Fetch data from the specified table in the database.
    """
    query = f'SELECT * FROM {table_name}'
    return pd.read_sql(query, con=conn)

def merge_dataframes(df1, df2, key, how='right'):
    """
    Merge two dataframes on the specified key using the specified method.
    """
    return pd.merge(df1, df2, how=how, on=key)

def measure_execution_time(func):
    """
    Decorator to measure the execution time of a function.
    """
    def wrapper(*args, **kwargs):
        start_time = time.time()
        result = func(*args, **kwargs)
        elapsed_time = time.time() - start_time
        print(f"Time taken: {elapsed_time} seconds")
        return result
    return wrapper

@measure_execution_time
def main():
    # Load environment variables
    load_dotenv()
    load_environment_variables()
    
    # Establish the database connection
    connect_to_database()
    
    # Fetch data from the studies and keywords tables
    studies_df = fetch_table_data('studies')
    keywords_df = fetch_table_data('keywords')
    
    # Merge the studies and keywords dataframes on the 'nct_id' column
    merged_result = merge_dataframes(studies_df, keywords_df, 'nct_id')
    
    # Print the merged result
    print(merged_result)

# Entry point of the script
if __name__ == '__main__':
    main()
