import sys

def merge(left, right):
    """
    Merge two sorted lists into one sorted list.

    Args:
        left (list): The first sorted list.
        right (list): The second sorted list.

    Returns:
        list: A merged and sorted list.
    """
    merged_list = []
    while left and right:
        if left[0] <= right[0]:
            merged_list.append(left.pop(0))
        else:
            merged_list.append(right.pop(0))
    
    # Append any remaining elements from both lists
    merged_list.extend(left)
    merged_list.extend(right)
    return merged_list

def merge_sort(lst):
    """
    Sort a list using the merge sort algorithm.

    Args:
        lst (list): The list to be sorted.

    Returns:
        list: The sorted list.
    """
    if len(lst) <= 1:
        return lst
    
    mid = len(lst) // 2
    left_half = merge_sort(lst[:mid])
    right_half = merge_sort(lst[mid:])
    
    return merge(left_half, right_half)

def main():
    """
    Main function to read command-line arguments and sort the provided list.
    """
    if len(sys.argv) > 1:
        try:
            # Read the list of numbers from command-line arguments
            input_list = [int(x) for x in sys.argv[1:]]
            print("Original list:", input_list)
            
            # Sort the list using merge sort
            sorted_list = merge_sort(input_list)
            print("Sorted list:", sorted_list)
        except ValueError:
            print("Please provide a valid list of integers.")
    else:
        print("Usage: python script.py <num1> <num2> ... <numN>")

if __name__ == "__main__":
    main()
