import sys
import pickle
import time
import numpy as np
from scipy.io import loadmat
from S3_BSTP import wild_bstp

def load_data(input_dir, interm_dir):
    """Load the necessary data from specified directories."""
    y_design = loadmat(input_dir + "img_data_" + LorR + ".mat")['img_data']
    if y_design.ndim == 2:
        y_design = y_design[np.newaxis, :, :]

    proj_mat = pickle.load(open(interm_dir + 'proj_mat.dat', 'rb'))
    efit_eta = pickle.load(open(interm_dir + 'efit_eta.dat', 'rb'))
    coord_data = pickle.load(open(interm_dir + 'coord_data.dat', 'rb'))
    h_opt = pickle.load(open(interm_dir + 'h_opt.dat', 'rb'))

    img_size = np.loadtxt(input_dir + 'img_size.txt').astype(int)
    img_idx = (np.loadtxt(input_dir + 'img_idx.txt').astype(int) - 1).astype(int)
    snp = np.loadtxt(input_dir + 'snp_data.txt')

    return y_design, proj_mat, efit_eta, coord_data, h_opt, img_size, img_idx, snp

def project_y_design(y_design, proj_mat):
    """Project the image data using the projection matrix."""
    proj_y_design = np.zeros_like(y_design)
    for mii in range(y_design.shape[0]):
        proj_y_design[mii, :, :] = np.dot(proj_mat, np.squeeze(y_design[mii, :, :]))
    return proj_y_design

def save_results(bstp_dir, i, max_gstat_bstp, max_lstat_bstp, max_area_bstp):
    """Save the bootstrap results to the specified directory."""
    np.savetxt(bstp_dir + f'max_gstat_bstp_{i}', max_gstat_bstp)
    np.savetxt(bstp_dir + f'max_lstat_bstp_{i}', max_lstat_bstp)
    np.savetxt(bstp_dir + f'max_area_bstp_{i}', max_area_bstp)

def main(LorR, i):
    # Define directories
    input_dir = 'data/'
    interm_dir = f'res/{LorR}vars/'
    bstp_dir = f'res/{LorR}bstp/'

    # Load necessary data
    y_design, proj_mat, efit_eta, coord_data, h_opt, img_size, img_idx, snp = load_data(input_dir, interm_dir)

    # Project the image data
    proj_y_design = project_y_design(y_design, proj_mat)
    print(f'The matrix dimension of image data is {y_design.shape}')

    # Set parameters for the bootstrap procedure
    alpha = 0.005
    alpha_log10 = -np.log10(alpha)
    b_num = 25
    g_num = 2000

    # Perform wild bootstrap
    start_time = time.time()
    max_gstat_bstp, max_lstat_bstp, max_area_bstp = wild_bstp(
        snp, proj_y_design, efit_eta, proj_mat, coord_data, h_opt, img_size, img_idx, alpha_log10, g_num, b_num
    )
    end_time = time.time()
    print(f'Elapsed time in wild_bstp is {end_time - start_time} seconds')

    # Save the bootstrap results
    save_results(bstp_dir, i, max_gstat_bstp, max_lstat_bstp, max_area_bstp)

if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("Usage: python script.py <LorR> <i>")
    else:
        LorR = sys.argv[1]
        i = sys.argv[2]
        main(LorR, i)
