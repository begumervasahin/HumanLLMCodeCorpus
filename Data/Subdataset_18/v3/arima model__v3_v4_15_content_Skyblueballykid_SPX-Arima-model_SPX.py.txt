import numpy as np
import pandas as pd
from statsmodels.tsa.arima.model import ARIMA

def load_data(file_path):
    """
    Load data from a CSV file into a pandas DataFrame.
    
    Args:
    - file_path (str): The path to the CSV file.
    
    Returns:
    - DataFrame: Loaded data as a pandas DataFrame.
    """
    return pd.read_csv(file_path)

def display_head(dataframe, n=5):
    """
    Display the first n rows of a DataFrame.
    
    Args:
    - dataframe (DataFrame): The DataFrame to display.
    - n (int): The number of rows to display. Default is 5.
    """
    print(f"First {n} rows of the DataFrame:")
    print(dataframe.head(n))

def create_arima_model(data, order):
    """
    Create and fit an ARIMA model to the given data.
    
    Args:
    - data (DataFrame or Series): The data to fit the ARIMA model on.
    - order (tuple): The (p,d,q) order of the ARIMA model.
    
    Returns:
    - ARIMAResults: Fitted ARIMA model.
    """
    model = ARIMA(data, order=order)
    return model.fit()

def forecast_next_value(fitted_model):
    """
    Forecast the next value using the fitted ARIMA model.
    
    Args:
    - fitted_model (ARIMAResults): The fitted ARIMA model.
    
    Returns:
    - float: The forecasted next value.
    """
    return fitted_model.forecast()[0]

def display_model_summary(fitted_model, rows=5):
    """
    Display the specified number of rows from the ARIMA model summary.
    
    Args:
    - fitted_model (ARIMAResults): The fitted ARIMA model.
    - rows (int): The number of rows to display from the summary.
    """
    print(f"First {rows} rows of the ARIMA model summary:")
    print(fitted_model.summary().tables[1].as_html())

def main():
    # Define the path to the CSV file
    csv_file_path = "SPX Weekly 2000-2017.csv"
    
    # Load data from the CSV file
    spx_data = load_data(csv_file_path)
    
    # Display the first 5 rows of the DataFrame
    display_head(spx_data)
    
    # Define the ARIMA model order
    arima_order = (0, 1, 0)
    
    # Create and fit the ARIMA model
    fitted_arima_model = create_arima_model(spx_data, arima_order)
    
    # Forecast the next data point using the fitted model
    forecasted_value = forecast_next_value(fitted_arima_model)
    print("Forecasted value:", forecasted_value)
    
    # Display the first 5 rows of the model summary
    display_model_summary(fitted_arima_model)

if __name__ == "__main__":
    main()
