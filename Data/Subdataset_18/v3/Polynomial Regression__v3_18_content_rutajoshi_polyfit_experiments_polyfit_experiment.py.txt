import numpy as np
import matplotlib.pyplot as plt

# Dummy plotting functions for completeness
# Replace these with actual functions from plot_functions.py if available
def set_axis_ranges(x, y):
    plt.xlim(min(x) - 1, max(x) + 1)
    plt.ylim(min(y) - 1, max(y) + 1)

def plot_function(x, func, style, color, label):
    y = func(x)
    plt.plot(x, y, style, color=color, label=label)

def plot_data(x, y, style, color, label):
    plt.plot(x, y, style, color=color, label=label)

def show_all(legend_loc='upper right'):
    plt.legend(loc=legend_loc)
    plt.show()

# Function to create a polynomial function based on coefficients
def make_function(w):
    def func(x):
        return evaluate_function(x, w)
    return func

# Evaluate polynomial function at a given x value
def evaluate_function(x, w):
    return sum(w[i] * (x ** i) for i in range(len(w)))

# Evaluate polynomial function for an array of x values
def evaluate_function_vec(x_values, w):
    return np.array([evaluate_function(x, w) for x in x_values])

# Sample data points based on a polynomial function
def sample_function(d, w, n, lower=-100, upper=100):
    assert len(w) == d + 1, "Must pass in d+1 coefficients"
    x_values = np.linspace(lower, upper, n)
    y_values = evaluate_function_vec(x_values, w)
    return x_values, y_values

# Perform polynomial regression to fit a model to the data
def polynomial_regression(x_train, y_train, d, noise_mean=0, noise_std=1):
    X_mtx = np.array([[x ** i for i in range(d + 1)] for x in x_train])
    y_train_noisy = y_train + np.random.normal(noise_mean, noise_std, y_train.shape)
    w = np.linalg.inv(X_mtx.T @ X_mtx) @ (X_mtx.T @ y_train_noisy)
    return w

# Run an experiment to sample data, fit a polynomial, and plot the results
def run_experiment(sample_coefficients, model_degree, n, noise_mean=0, noise_std=0.5):
    x_train, y_train = sample_function(len(sample_coefficients) - 1, sample_coefficients, n, lower=-5, upper=5)
    polyfit_coefficients = polynomial_regression(x_train, y_train, model_degree, noise_mean, noise_std)
    
    true_function = make_function(sample_coefficients)
    learned_function = make_function(polyfit_coefficients)
    
    y_model = evaluate_function_vec(x_train, polyfit_coefficients)
    variance = np.mean((y_model - y_train) ** 2)
    
    print("\nVariance of the polyfit = " + str(variance) + "\n")
    
    set_axis_ranges(x_train, y_train)
    plot_function(x_train, true_function, '-', color='r', label='ground truth function')
    plot_function(x_train, learned_function, '--', color='b', label='model fit function')
    plot_data(x_train, y_train, '.', color='g', label='sample points')
    show_all(legend_loc='upper right')

# Main function to run the experiment with given polynomial coefficients
def main():
    w = [3, 1, 1, -2]
    run_experiment(w, 3, 6, noise_mean=0, noise_std=0.5)

if __name__ == "__main__":
    main()
