import networkx as nx
import heapq

def parallel_dijkstra(graph, sources, mode):
    """
    Perform Dijkstra's algorithm from multiple source nodes on a graph.

    Parameters:
    graph (nx.Graph): The graph on which to perform the algorithm.
    sources (list): List of source nodes for Dijkstra's algorithm.
    mode (int): Mode of operation for handling edge directions:
                1 - Undirected graph.
                2 - Directed graph.

    Returns:
    tuple: A tuple containing:
        - The updated graph with shortest path distances and labels.
        - A list of labels assigned to the nodes.
    """
    # Initialize node attributes
    nx.set_node_attributes(graph, float('inf'), 'Dv')  # Distance values
    nx.set_node_attributes(graph, 'N', 'V')  # Vertex labels
    nx.set_node_attributes(graph, 0, 'visit')  # Visit status

    # Priority queue and node labels
    priority_queue = []
    tags = []

    # Initialize the source nodes in the priority queue
    for i, source in enumerate(sources):
        if source in graph:
            graph.nodes[source]['Dv'] = 0
            graph.nodes[source]['V'] = f'V{i}'
            tags.append(f'V{i}')
            heapq.heappush(priority_queue, (0, source))

    # Execute Dijkstra's algorithm
    while priority_queue:
        current_distance, current_node = heapq.heappop(priority_queue)
        graph.nodes[current_node]['visit'] = 1

        # Get neighbors based on the mode
        neighbors = get_neighbors(graph, current_node, mode)

        for neighbor in neighbors:
            edge_data = graph.get_edge_data(current_node, neighbor)
            weight = edge_data.get('weight', float('inf'))

            # Compute the new distance
            new_distance = current_distance + weight
            if new_distance < graph.nodes[neighbor]['Dv']:
                graph.nodes[neighbor]['Dv'] = new_distance
                graph.nodes[neighbor]['V'] = graph.nodes[current_node]['V']
                heapq.heappush(priority_queue, (new_distance, neighbor))

    return graph, tags

def get_neighbors(graph, node, mode):
    """
    Get neighbors of a node based on the graph mode.

    Parameters:
    graph (nx.Graph): The graph from which to retrieve neighbors.
    node: The node for which to get neighbors.
    mode (int): Mode of operation:
                1 - Undirected graph.
                2 - Directed graph.

    Returns:
    list: List of neighbors for the given node.
    """
    if mode == 1:  # Undirected graph
        return list(graph.neighbors(node))
    elif mode == 2:  # Directed graph
        return list(graph.predecessors(node))
    else:
        raise ValueError("Invalid mode. Use 1 for undirected or 2 for directed.")

# Example usage
if __name__ == "__main__":
    # Create a sample directed graph
    G = nx.DiGraph()
    G.add_weighted_edges_from([(1, 2, 4), (1, 3, 1), (2, 3, 2), (3, 4, 5)])

    # Define sources and mode
    sources = [1]
    mode = 2  # Directed graph mode

    # Run the algorithm
    updated_graph, tags = parallel_dijkstra(G, sources, mode)
    
    # Output the results
    print("Node labels and distances:")
    for node, data in updated_graph.nodes(data=True):
        print(f"Node {node}: {data}")
    print("Tags:", tags)
