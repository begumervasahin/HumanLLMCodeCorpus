class Edge:
    def __init__(self, start, target, weight):
        """
        Initialize an Edge instance.

        :param start: The starting node of the edge.
        :param target: The target node of the edge.
        :param weight: The weight of the edge.
        """
        self.start = start
        self.target = target
        self.weight = weight
    
    def get_weight(self):
        """
        Get the weight of the edge.

        :return: The weight of the edge.
        """
        return self.weight
    
    def set_weight(self, weight):
        """
        Set a new weight for the edge.

        :param weight: The new weight of the edge.
        """
        self.weight = weight
    
    def get_start(self):
        """
        Get the starting node of the edge.

        :return: The starting node.
        """
        return self.start
    
    def set_start(self, start):
        """
        Set a new starting node for the edge.

        :param start: The new starting node.
        """
        self.start = start
    
    def get_target(self):
        """
        Get the target node of the edge.

        :return: The target node.
        """
        return self.target
    
    def set_target(self, target):
        """
        Set a new target node for the edge.

        :param target: The new target node.
        """
        self.target = target

# Example usage:
if __name__ == "__main__":
    # Create an edge from node 'A' to node 'B' with a weight of 5
    edge = Edge("A", "B", 5)
    
    # Print the initial values
    print("Start:", edge.get_start())  # Output: Start: A
    print("Target:", edge.get_target())  # Output: Target: B
    print("Weight:", edge.get_weight())  # Output: Weight: 5

    # Update the edge's properties
    edge.set_start("C")
    edge.set_target("D")
    edge.set_weight(10)
    
    # Print the updated values
    print("Updated Start:", edge.get_start())  # Output: Updated Start: C
    print("Updated Target:", edge.get_target())  # Output: Updated Target: D
    print("Updated Weight:", edge.get_weight())  # Output: Updated Weight: 10
