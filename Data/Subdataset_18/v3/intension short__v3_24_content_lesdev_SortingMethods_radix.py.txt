import tkinter as tk
from tkinter import simpledialog

def counting_sort(numbers, exp):
    """
    Perform counting sort on the list of numbers based on the digit represented by exp.
    """
    n = len(numbers)
    output = [0] * n
    count = [0] * 10

    # Count occurrences of each digit in the given place value
    for number in numbers:
        index = (number // exp) % 10
        count[index] += 1

    # Update count so that it contains the actual position of this digit in output
    for i in range(1, 10):
        count[i] += count[i - 1]

    # Build the output array
    for i in reversed(range(n)):
        index = (numbers[i] // exp) % 10
        output[count[index] - 1] = numbers[i]
        count[index] -= 1

    # Copy the sorted numbers back to the original list
    for i in range(n):
        numbers[i] = output[i]

def radix_sort(numbers):
    """
    Perform radix sort on the provided list of numbers.
    """
    max_number = max(numbers)
    exp = 1
    while max_number // exp > 0:
        counting_sort(numbers, exp)
        exp *= 10

def get_numbers_from_user():
    """
    Prompt the user to input the numbers they want to sort and return the list of numbers.
    """
    num_elements = int(entry_num.get())
    numbers = []
    
    for _ in range(num_elements):
        number = simpledialog.askinteger("Input", "Enter a number:")
        if number is not None:
            numbers.append(number)
    
    return numbers

def display_sorted_numbers():
    """
    Get numbers from the user, perform radix sort, and display the sorted numbers.
    """
    numbers = get_numbers_from_user()
    
    # Display the original list of numbers
    label_original.config(text=f"Original List: {numbers}")
    
    # Sort the numbers using radix sort
    radix_sort(numbers)
    
    # Display the sorted list of numbers
    label_sorted.config(text=f"Sorted List: {numbers}")

# Set up the main application window
app = tk.Tk()
app.title("Radix Sort Application")
app.geometry('300x200')
app.configure(bg='SkyBlue2')

# Create and place the UI elements
label_prompt = tk.Label(app, text="How many numbers will you enter?", font='Helvetica 12', bg='SkyBlue2')
label_prompt.pack(pady=10)

entry_num = tk.Entry(app, width=8)
entry_num.pack()

button_sort = tk.Button(app, text="Sort", command=display_sorted_numbers)
button_sort.pack(pady=10)

label_original = tk.Label(app, text="", bg='SkyBlue2')
label_original.pack(pady=5)

label_sorted = tk.Label(app, text="", bg='SkyBlue2')
label_sorted.pack(pady=5)

# Start the Tkinter event loop
app.mainloop()
