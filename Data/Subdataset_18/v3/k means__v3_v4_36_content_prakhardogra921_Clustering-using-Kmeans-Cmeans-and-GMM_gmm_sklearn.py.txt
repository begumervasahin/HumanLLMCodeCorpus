import numpy as np
from sklearn import mixture
import matplotlib.pyplot as plt
import random

def generate_synthetic_data(size, params):
    """
    Generates synthetic data based on normal distributions.

    Parameters:
    size (int): Size of each data set.
    params (list of tuples): Each tuple contains the mean (loc) and standard deviation (scale)
                             of the normal distribution for one data set.

    Returns:
    list: List of synthetic data sets.
    """
    return [np.random.normal(loc=loc, scale=scale, size=size) for loc, scale in params]

def combine_data_sets(data_sets, proportions, size):
    """
    Combines multiple data sets based on specified proportions.

    Parameters:
    data_sets (list of array-like): Data sets to combine.
    proportions (list of float): Proportions for each data set.
    size (int): Total size of the combined data set.

    Returns:
    np.array: Combined data set.
    """
    combined_data = np.concatenate([
        random.sample(list(data_set), int(prop * size)) 
        for data_set, prop in zip(data_sets, proportions)
    ])
    return combined_data

def plot_gmm(data):
    """
    Fits a Gaussian Mixture Model (GMM) to the data and generates a scatter plot.

    Parameters:
    data (np.array): Data to fit the GMM.
    """
    gmm = mixture.GaussianMixture(n_components=3)
    gmm.fit(data.reshape(-1, 1))
    y = gmm.predict(data.reshape(-1, 1))
    plt.scatter(data, y, c=y, cmap='viridis', marker='o', edgecolor='k')
    plt.title('Gaussian Mixture Model')
    plt.xlabel('Data')
    plt.ylabel('Cluster')
    plt.show()

def main():
    # Parameters
    size = 500
    proportions = [0.25, 0.5, 0.25]

    # Data generation parameters
    params1 = [(1, 0.1), (1.5, 0.1), (2, 0.2)]
    params2 = [(1, 0.3), (1.5, 0.4), (2, 0.3)]

    # Generate and combine first set of synthetic data
    data_sets1 = generate_synthetic_data(size, params1)
    dset1 = combine_data_sets(data_sets1, proportions, size)

    # Generate and combine second set of synthetic data
    data_sets2 = generate_synthetic_data(size, params2)
    dset2 = combine_data_sets(data_sets2, proportions, size)

    # Generate GMM plot for the first data set
    plot_gmm(dset1)

    # Uncomment the following line to generate GMM plot for the second data set
    # plot_gmm(dset2)

if __name__ == "__main__":
    main()
