# Define the file paths for input and output
INPUT_FILE_PATH = "tvshow_list_unsorted.txt"
OUTPUT_FILE_PATH = "tvshow_list_sorted.txt"

def read_tv_shows(file_path):
    """
    Read TV shows from the given file and return a list of tuples (show_name, show_detail).

    Args:
    file_path (str): The path to the input file.

    Returns:
    list: A list of tuples containing show_name and show_detail.
    """
    tv_shows = []
    with open(file_path, "r") as file:
        for line in file:
            show_name, show_detail = line.strip().split('_')
            tv_shows.append((show_name, show_detail))
    return tv_shows

def write_tv_shows(file_path, tv_shows):
    """
    Write the sorted TV shows to the given file.

    Args:
    file_path (str): The path to the output file.
    tv_shows (list): A list of tuples containing show_name and show_detail.
    """
    with open(file_path, 'w') as file:
        for show in tv_shows:
            file.write(f"{show[0]}_{show[1]}\n")

def main():
    """
    Main function to read, sort, and write the TV shows.
    """
    # Read the unsorted list of TV shows from the input file
    tv_shows = read_tv_shows(INPUT_FILE_PATH)

    # Sort the list first by show_detail and then by show_name
    tv_shows.sort(key=lambda show: (show[1], show[0]))

    # Write the sorted list to the output file
    write_tv_shows(OUTPUT_FILE_PATH, tv_shows)

if __name__ == "__main__":
    main()
