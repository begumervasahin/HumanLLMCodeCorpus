import os
import string
import numpy as np
import pandas as pd
from time import time
from nltk.corpus import stopwords
from nltk.stem.wordnet import WordNetLemmatizer

def load_words_list(file_path):
    """Load the list of words from the given CSV file."""
    df = pd.read_csv(file_path)
    return df['word'].tolist()

def write_headers(file_path, headers):
    """Write headers to the given CSV file."""
    with open(file_path, "w") as file:
        file.write(','.join(headers) + ',output\n')

def process_email_file(file_path, words, lmtzr, stop_words):
    """Process a single email file and return the word frequency array."""
    word_freq = np.zeros(len(words))
    
    with open(file_path, "r", encoding='utf-8', errors='ignore') as file:
        for word in file.read().split():
            word = lmtzr.lemmatize(word.lower())
            if word in stop_words or word in string.punctuation or len(word) <= 2 or word.isdigit():
                continue
            if word in words:
                word_freq[words.index(word)] += 1
    
    return word_freq

def determine_output_label(file_name):
    """Determine the output label based on the file name length."""
    return "1" if len(file_name) == 71 else "-1"

def append_to_csv(file_path, word_freq, output_label):
    """Append the word frequencies and output label to the CSV file."""
    with open(file_path, "a") as file:
        file.write(','.join(map(str, map(int, word_freq))) + ',')
        file.write(output_label + '\n')

def main():
    # Start the timer
    start_time = time()

    # Load the list of words
    words = load_words_list('wordslist.csv')
    lmtzr = WordNetLemmatizer()
    stop_words = set(stopwords.words('english'))

    # Define the directory containing the email files
    directory_path = "emails/"
    directory = os.fsencode(directory_path)

    # Create and write headers to the frequency.csv file
    write_headers("frequency.csv", words)

    # Initialize email processing count
    processed_count = 0

    # Process each email file in the directory
    for file in os.listdir(directory):
        # Decode the file name and determine its path
        file_name = os.fsdecode(file)
        file_path = os.path.join(directory_path, file_name)
        processed_count += 1

        # Process the email file and get word frequencies
        word_freq = process_email_file(file_path, words, lmtzr, stop_words)

        # Determine the output label and append data to the CSV file
        output_label = determine_output_label(file_name)
        append_to_csv("frequency.csv", word_freq, output_label)

        # Print progress every 100 files processed
        if processed_count % 100 == 0:
            print(f"Processed {processed_count} files")

    # Print the total time taken for processing
    elapsed_time = round(time() - start_time, 2)
    print(f"Time (in seconds) to process the entire dataset: {elapsed_time}")

if __name__ == "__main__":
    main()
