import pickle
import numpy as np
from sklearn.pipeline import Pipeline
from sklearn import svm

# Set a seed for reproducibility
np.random.seed(1000)

def load_classifier(filename):
    """
    Load a pre-trained classifier from a file.
    
    Parameters:
    filename (str): The path to the saved classifier file.
    
    Returns:
    object: The loaded classifier.
    """
    with open(filename, 'rb') as file:
        classifier = pickle.load(file)
    return classifier

def classify_text(classifier, text):
    """
    Classify the input text using the provided classifier.
    
    Parameters:
    classifier (object): The pre-trained classifier.
    text (str): The text to classify.
    
    Returns:
    array: The predicted class for the input text.
    """
    test_data = [text]
    print("Test data:", test_data, '\n')
    
    print("Classifying test set...")
    prediction = classifier.predict(test_data)
    
    print("Prediction:", prediction, '\n')
    return prediction

if __name__ == "__main__":
    # Define the model filename
    model_filename = 'Amz_5000_.sav'
    
    # Load the classifier
    print("Loading classifier...")
    classifier = load_classifier(model_filename)
    print("Classifier loaded!")
    
    # Prompt the user to enter text for classification
    input_text = input("Enter the text to classify: ")
    
    # Classify the input text
    result = classify_text(classifier, input_text)
    
    # Display the classification result
    print("Result:", result)
