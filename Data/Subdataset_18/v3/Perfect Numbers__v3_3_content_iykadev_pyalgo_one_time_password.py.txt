import hmac
import base64
import struct
import hashlib
import time

def get_hotp_token(secret, intervals_no):
    """
    Generate an HMAC-based One-Time Password (HOTP).
    
    Args:
    secret (str): The shared secret in base32 encoding.
    intervals_no (int): The counter value (time step or event counter).
    
    Returns:
    int: The generated HOTP token.
    """
    # Decode the base32 encoded secret
    key = base64.b32decode(secret, True)
    
    # Pack the interval number into a binary structure
    msg = struct.pack(">Q", intervals_no)
    
    # Create the HMAC hash using SHA-1
    hmac_hash = hmac.new(key, msg, hashlib.sha1).digest()
    
    # Extract a dynamic binary code from the HMAC hash
    offset = hmac_hash[19] & 15
    binary_code = struct.unpack(">I", hmac_hash[offset:offset+4])[0] & 0x7fffffff
    
    # Generate the HOTP value
    hotp_token = binary_code % 1000000
    return hotp_token

def get_totp_token(secret):
    """
    Generate a Time-based One-Time Password (TOTP).
    
    Args:
    secret (str): The shared secret in base32 encoding.
    
    Returns:
    int: The generated TOTP token.
    """
    # Calculate the current time step
    time_step = int(time.time()) // 30
    return get_hotp_token(secret, intervals_no=time_step)

def main():
    # Define the shared secret
    secret = 'MZXW633PN5XW6MZX'

    # Generate and print HOTP tokens for intervals 1 through 29
    print("HOTP Tokens:")
    for interval in range(1, 30):
        hotp_token = get_hotp_token(secret, intervals_no=interval)
        print(f"Interval {interval}: {hotp_token}")

    # Generate and print the current TOTP token
    current_totp_token = get_totp_token(secret)
    print(f"Current TOTP token: {current_totp_token}")

if __name__ == "__main__":
    main()
