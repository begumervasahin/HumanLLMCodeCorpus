import random

# Global counter to track the number of comparisons
comparison_counter = 0

def merge_sort(lst):
    """
    Recursively performs merge sort on the provided list and counts the number of comparisons.

    Parameters:
    lst (list): The list of elements to sort.

    Returns:
    list: A sorted version of the input list.
    """
    global comparison_counter

    if len(lst) <= 1:
        comparison_counter += 1  # Counting even for trivial cases
        return lst

    # Divide the list into two halves
    mid = len(lst) // 2
    left = merge_sort(lst[:mid])
    right = merge_sort(lst[mid:])

    # Merge the sorted halves
    return merge(left, right)

def merge(left, right):
    """
    Merges two sorted lists into one sorted list, counting comparisons.

    Parameters:
    left (list): The left half of the list.
    right (list): The right half of the list.

    Returns:
    list: A merged and sorted list from the two input halves.
    """
    global comparison_counter
    merged = []
    i, j = 0, 0

    # Merge elements from both lists in sorted order
    while i < len(left) and j < len(right):
        if left[i] < right[j]:
            merged.append(left[i])
            i += 1
        else:
            merged.append(right[j])
            j += 1
        comparison_counter += 1  # Increment comparison counter for each comparison

    # Append remaining elements from left and right lists
    merged.extend(left[i:])
    merged.extend(right[j:])
    comparison_counter += 1  # Count the final merge step

    return merged

def test_merge_sort(lines, num_words):
    """
    Tests merge sort on a subset of words in different configurations (sorted, reversed, random).

    Parameters:
    lines (list): The list of words to test.
    num_words (int): The number of words to use for the test.
    """
    global comparison_counter

    print(f'\nTesting on {num_words} words')

    # Test on sorted words
    print(f'{num_words} words sorted')
    words_sorted = lines[:num_words]
    merge_sort(words_sorted)
    print('Number of comparisons:', comparison_counter)

    # Test on reversed words
    comparison_counter = 0
    print(f'{num_words} words reversed')
    words_reversed = list(reversed(words_sorted))
    merge_sort(words_reversed)
    print('Number of comparisons:', comparison_counter)

    # Test on randomly shuffled words
    comparison_counter = 0
    print(f'{num_words} words random')
    random_start = random.randint(0, len(lines) - num_words)
    words_random = lines[random_start:random_start + num_words]
    random.shuffle(words_random)
    merge_sort(words_random)
    print('Number of comparisons:', comparison_counter)

def main():
    """
    Reads words from a file and tests merge sort on various word counts.
    """
    with open('words.txt') as file:
        lines = [line.strip() for line in file]

    print(f'Total words: {len(lines)}')

    # Define test cases with different word counts
    test_cases = [10, 30, 100, 300, 1000, 3000, 10000, len(lines)]

    # Run tests for each case
    for num_words in test_cases:
        test_merge_sort(lines, num_words)

if __name__ == "__main__":
    main()
