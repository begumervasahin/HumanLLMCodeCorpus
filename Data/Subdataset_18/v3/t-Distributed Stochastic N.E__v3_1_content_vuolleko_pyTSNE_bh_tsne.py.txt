import numpy as np
from time import time
from vptree import VPnode
from quadtree import Quadnode
from tsne import TSNE

class BH_TSNE(TSNE):
    """
    Barnes-Hut accelerated t-Distributed Stochastic Neighbor Embedding (t-SNE).
    
    This class extends the base TSNE class, incorporating the Barnes-Hut approximation 
    to optimize performance for large datasets.
    """

    def __init__(self, bh_threshold=0.5, *args, **kwargs):
        """
        Initialize the Barnes-Hut t-SNE algorithm with a specified threshold.

        Args:
            bh_threshold (float): Threshold for the Barnes-Hut approximation.
            *args, **kwargs: Additional arguments passed to the TSNE base class.
        """
        self.bh_threshold = bh_threshold
        super().__init__(*args, **kwargs)
        self.cost_min_since_max = 0  # Not implemented for Barnes-Hut

    def _set_affin_hd(self):
        """
        Compute pairwise affinities for nearest neighbors in high-dimensional data
        using a Vantage-point tree (VP-tree).
        """
        # Initialize an empty affinity matrix
        affinities = np.zeros((self.n_samples, self.n_samples))

        # Construct a VP-tree from the dataset
        start_time = time()
        print("Constructing a Vantage-point tree...")
        vp_tree_root = VPnode(np.arange(self.n_samples), self.data)
        print(f"VP-tree constructed in {time() - start_time:.2f} seconds.")

        # Define the number of neighbors to consider based on perplexity
        n_neighbors = int(3 * self.perplexity)
        distances = np.full(n_neighbors, np.inf)  # Initialize distances to infinity

        # Calculate conditional probabilities for each data point
        for i in range(self.n_samples):
            neighbors = np.zeros(n_neighbors, dtype=int)  # Array to store neighbor indices
            vp_tree_root.find_neighbors(i, neighbors, distances, self.data)
            affinities[i, neighbors] = self._affin_bin_search_sigma(distances)

        # Make the affinity matrix symmetric
        affinities = (affinities + affinities.T) / (2.0 * self.n_samples)

        # Store the computed affinities
        self.affin_hd = affinities

    def _set_gradient(self):
        """
        Estimate the gradient using a Quadtree for efficient computation of repulsive forces.
        """
        # Construct a Quadtree from the low-dimensional coordinates
        center_y, center_x = np.median(self.coord, axis=0)
        quad_width = 2.0 * (np.max(self.coord) - np.min(self.coord))
        quadtree_root = Quadnode(self.coord, center_x, center_y, quad_width)

        # Calculate pairwise differences between coordinates
        pairwise_diffs = self.coord[:, np.newaxis, :] - self.coord[np.newaxis, :, :]

        # Compute attractive forces based on affinities
        F_attr = np.sum(
            (self.affin_hd[:, :, np.newaxis] / (1.0 + np.sum(pairwise_diffs ** 2, axis=2))[:, :, np.newaxis]) 
            * pairwise_diffs, axis=1
        )

        # Compute repulsive forces using the Quadtree
        F_rep = np.empty_like(F_attr)
        total_student_sum = 0.0
        for i in range(self.n_samples):
            summary = quadtree_root.summarize(self.coord[i, :], self.bh_threshold)
            F_rep[i, :] = summary[0]
            total_student_sum += summary[1]

        # Normalize the repulsive forces
        F_rep /= total_student_sum

        # Compute the final gradient
        self.gradient = 4.0 * (F_attr + F_rep)
