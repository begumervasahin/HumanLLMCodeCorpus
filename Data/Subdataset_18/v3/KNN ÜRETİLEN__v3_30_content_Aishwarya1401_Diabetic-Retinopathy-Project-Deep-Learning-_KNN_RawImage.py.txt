import os
import numpy as np
import pandas as pd
from PIL import Image
from sklearn.decomposition import PCA
from sklearn.neighbors import KNeighborsClassifier
from sklearn.model_selection import train_test_split

# Define the path to the images and load the labels
IMAGE_PATH = "gray2"
LABELS_FILE = 'trainLabels.csv'

# Load labels from the CSV file
df_label = pd.read_csv(LABELS_FILE)

def get_label(image_name):
    """
    Get the label for a given image based on its filename.
    
    Parameters:
    image_name (str): The filename of the image.
    
    Returns:
    int: The label of the image.
    """
    image_id = image_name.split(os.path.sep)[-1].split(".")[0]
    label = df_label.loc[df_label['id'] == image_id, 'label'].values[0]
    return int(label)

def load_images_and_labels(image_path):
    """
    Load images and their corresponding labels from the specified path.
    
    Parameters:
    image_path (str): The directory containing the images.
    
    Returns:
    tuple: A tuple containing the image vectors and their labels.
    """
    files = os.listdir(image_path)
    image_vectors = []
    labels = []

    for file in files:
        image = Image.open(os.path.join(image_path, file))
        image_vector = np.array(image).flatten()
        image_vectors.append(image_vector)
        
        label = get_label(file)
        labels.append(label)

    return np.array(image_vectors, dtype='float32'), np.array(labels)

def evaluate_knn_classifier(train_vectors, train_labels, test_vectors, test_labels, k_values):
    """
    Evaluate the k-NN classifier with different values of k.
    
    Parameters:
    train_vectors (np.ndarray): Training image vectors.
    train_labels (np.ndarray): Training labels.
    test_vectors (np.ndarray): Testing image vectors.
    test_labels (np.ndarray): Testing labels.
    k_values (list): List of k values to evaluate.
    """
    for k in k_values:
        print(f"Evaluating raw pixel accuracy with k = {k}")
        model = KNeighborsClassifier(n_neighbors=k)
        model.fit(train_vectors, train_labels)
        accuracy = model.score(test_vectors, test_labels)
        print(f"Raw pixel accuracy: {accuracy * 100:.2f}%")

def main():
    # Load image vectors and labels
    img_vectors, labels = load_images_and_labels(IMAGE_PATH)

    # Split the data into training and testing sets
    train_vectors, test_vectors, train_labels, test_labels = train_test_split(img_vectors, labels, test_size=0.25, random_state=42)

    # Evaluate the classifier with different values of k
    k_values = [1, 3, 5, 10, 20, 50, 100]
    evaluate_knn_classifier(train_vectors, train_labels, test_vectors, test_labels, k_values)

if __name__ == "__main__":
    main()
