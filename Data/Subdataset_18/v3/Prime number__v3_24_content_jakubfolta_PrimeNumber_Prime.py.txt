def is_prime(digit):
    """
    Check if a number is prime.
    """
    if digit < 2:
        return False
    for number in range(2, int(digit ** 0.5) + 1):
        if digit % number == 0:
            return False
    return True

def print_and_check_prime(number):
    """
    Print numbers from 2 to number-1 and check if the number is prime.
    """
    if number < 2:
        return False

    for x in range(2, number):
        print(x)
        if number % x == 0:
            return False
    return True

def check_if_prime(number):
    """
    Check if a number is prime.
    """
    if number < 2:
        return False
    for x in range(2, int(number ** 0.5) + 1):
        if number % x == 0:
            return False
    return True

# Testing the functions
print(is_prime(21))                # Output should be False
