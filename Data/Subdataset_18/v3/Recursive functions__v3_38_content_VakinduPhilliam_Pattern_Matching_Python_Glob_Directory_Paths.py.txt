import glob

def find_files():
    """
    Finds and prints various types of files in the current directory using glob patterns.
    """
    # Find files in the current directory with names starting with a single digit followed by any extension
    digit_files = glob.glob('./[0-9].*')
    print('Files starting with a digit:', digit_files)

    # Find all .gif files in the current directory
    gif_files = glob.glob('*.gif')
    print('All .gif files:', gif_files)

    # Find all .gif files in the current directory with a single character name
    single_char_gif_files = glob.glob('?.gif')
    print('.gif files with a single character name:', single_char_gif_files)

    # Recursively find all .txt files in the current directory and its subdirectories
    txt_files_recursive = glob.glob('**/*.txt', recursive=True)
    print('All .txt files recursively:', txt_files_recursive)

    # Recursively find all directories in the current directory and its subdirectories
    directories_recursive = glob.glob('./**/', recursive=True)
    print('All directories recursively:', directories_recursive)

def main():
    """
    Main function to execute the file search and print results.
    """
    find_files()

if __name__ == '__main__':
    main()
