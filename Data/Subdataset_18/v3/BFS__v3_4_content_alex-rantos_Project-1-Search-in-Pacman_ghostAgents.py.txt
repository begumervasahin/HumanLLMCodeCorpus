from game import Agent, Actions, Directions
import random
from util import manhattanDistance
import util

class GhostAgent(Agent):
    """
    A base class for ghost agents in the game.
    """
    def __init__(self, index):
        self.index = index

    def getAction(self, state):
        """
        Determines the action for the ghost to take based on a distribution of possible actions.
        """
        dist = self.getDistribution(state)
        if len(dist) == 0:
            return Directions.STOP
        else:
            return util.chooseFromDistribution(dist)

    def getDistribution(self, state):
        """
        Returns a Counter encoding a distribution over actions from the provided state.
        To be implemented by subclasses.
        """
        util.raiseNotDefined()

class RandomGhost(GhostAgent):
    """
    A ghost that chooses a legal action uniformly at random.
    """
    def getDistribution(self, state):
        dist = util.Counter()
        legalActions = state.getLegalActions(self.index)
        for action in legalActions:
            dist[action] = 1.0
        dist.normalize()
        return dist

class DirectionalGhost(GhostAgent):
    """
    A ghost that prefers to rush Pacman, or flee when scared.
    """
    def __init__(self, index, prob_attack=0.8, prob_scaredFlee=0.8):
        super().__init__(index)
        self.prob_attack = prob_attack
        self.prob_scaredFlee = prob_scaredFlee

    def getDistribution(self, state):
        """
        Returns a distribution over actions based on the state of the ghost and Pacman.
        """
        ghostState = state.getGhostState(self.index)
        legalActions = state.getLegalActions(self.index)
        ghostPos = state.getGhostPosition(self.index)
        pacmanPos = state.getPacmanPosition()
        isScared = ghostState.scaredTimer > 0

        # Adjust speed based on scared state
        speed = 0.5 if isScared else 1

        # Calculate new positions after applying legal actions
        actionVectors = [Actions.directionToVector(action, speed) for action in legalActions]
        newPositions = [(ghostPos[0] + vector[0], ghostPos[1] + vector[1]) for vector in actionVectors]

        # Calculate distances to Pacman for each new position
        distancesToPacman = [manhattanDistance(pos, pacmanPos) for pos in newPositions]

        # Determine the best actions based on the state
        if isScared:
            bestScore = max(distancesToPacman)
            bestProb = self.prob_scaredFlee
        else:
            bestScore = min(distancesToPacman)
            bestProb = self.prob_attack

        bestActions = [action for action, distance in zip(legalActions, distancesToPacman) if distance == bestScore]

        # Construct the action distribution
        dist = util.Counter()
        for action in bestActions:
            dist[action] = bestProb / len(bestActions)
        for action in legalActions:
            dist[action] += (1 - bestProb) / len(legalActions)
        dist.normalize()
        
        return dist
