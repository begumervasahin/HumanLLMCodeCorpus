import numpy as np

def load_data(filename: str) -> np.array:
    """
    Loads the dataset from the specified file and returns it as a numpy array.
    
    Parameters:
    - filename: The path to the data file.
    
    Returns:
    - A numpy array containing the data.
    """
    data = []
    with open(filename, "r") as file:
        for line in file.readlines():
            data.append([float(value) for value in line.split('\t')] + [0])
    return np.asarray(data)

def compute_error(X: np.array, centroids: np.array) -> float:
    """
    Computes the quality of clustering.
    
    Parameters:
    - X: The dataset with assigned cluster IDs.
    - centroids: The current centroids.
    
    Returns:
    - The computed error of clustering.
    """
    total_error = 0.0
    for point in X:
        cluster_id = int(point[-1])
        total_error += np.linalg.norm(point[:-1] - centroids[cluster_id])
    return total_error / X.shape[0]

def calculate_new_centroids(X: np.array, centroids: np.array) -> np.array:
    """
    Calculates the new centroids based on the current clusters.
    
    Parameters:
    - X: The dataset with assigned cluster IDs.
    - centroids: The current centroids.
    
    Returns:
    - The new centroids as a numpy array.
    """
    new_centroids = np.zeros(centroids.shape)
    cluster_counts = np.zeros(centroids.shape[0], dtype=int)

    for point in X:
        cluster_id = int(point[-1])
        new_centroids[cluster_id] += point[:-1]
        cluster_counts[cluster_id] += 1

    for i in range(centroids.shape[0]):
        if cluster_counts[i] > 0:
            new_centroids[i] /= cluster_counts[i]

    return new_centroids

def assign_clusters(X: np.array, centroids: np.array) -> np.array:
    """
    Assigns each data point to the nearest cluster based on Euclidean distance.
    
    Parameters:
    - X: The dataset.
    - centroids: The current centroids.
    
    Returns:
    - The dataset with updated cluster assignments.
    """
    for point in X:
        distances = np.linalg.norm(point[:-1] - centroids, axis=1)
        point[-1] = np.argmin(distances)
    return X

def kmeans_clustering(filename: str, num_clusters: int, max_iterations: int = 100) -> (np.array, float):
    """
    Perform K-means clustering on the dataset provided in the filename.
    
    Parameters:
    - filename: The path to the data file.
    - num_clusters: The number of clusters to form.
    - max_iterations: The maximum number of iterations to run the algorithm.
    
    Returns:
    - The final centroids and the error of the clustering.
    """
    # Load data
    X = load_data(filename)
    
    # Initialize centroids randomly from the dataset
    initial_centroids_idx = np.random.choice(X.shape[0], num_clusters, replace=False)
    centroids = X[initial_centroids_idx, :-1]

    for iteration in range(max_iterations):
        # Assign clusters
        X = assign_clusters(X, centroids)
        
        # Calculate new centroids
        new_centroids = calculate_new_centroids(X, centroids)
        
        # Check for convergence
        if np.allclose(centroids, new_centroids):
            print(f"Converged after {iteration} iterations.")
            break
        
        centroids = new_centroids

    # Compute final error
    error = compute_error(X, centroids)
    return centroids, error
