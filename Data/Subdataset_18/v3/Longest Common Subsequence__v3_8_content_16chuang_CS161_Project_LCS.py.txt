import sys
import numpy as np

def initialize_array(size):
    """
    Initialize a 2D array with the given size filled with zeros.
    """
    return np.zeros((size, size), dtype=int)

def LCS(A, B, arr):
    """
    Compute the length of the Longest Common Subsequence (LCS) of strings A and B.
    
    Parameters:
    A (str): First string.
    B (str): Second string.
    arr (np.ndarray): 2D array to store intermediate results.
    
    Returns:
    int: Length of the Longest Common Subsequence.
    """
    m, n = len(A), len(B)
    
    # Fill the array using dynamic programming
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if A[i - 1] == B[j - 1]:
                arr[i][j] = arr[i - 1][j - 1] + 1
            else:
                arr[i][j] = max(arr[i - 1][j], arr[i][j - 1])
    
    return arr[m][n]

def main():
    """
    Main function to read input from stdin and print the LCS length for each pair of strings.
    """
    if len(sys.argv) != 1:
        sys.exit('Usage: python LCS.py < input')
    
    # Initialize the array once outside the loop
    arr = initialize_array(2048)
    
    for line in sys.stdin:
        A, B = line.strip().split()
        print(LCS(A, B, arr))

if __name__ == '__main__':
    main()
