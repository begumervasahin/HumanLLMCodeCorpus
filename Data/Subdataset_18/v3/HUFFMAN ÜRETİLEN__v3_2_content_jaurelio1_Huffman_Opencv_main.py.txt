import cv2
from Heap import Heap
from Huffman import Huffman
from Node import Node
import zipfile

def resize_image(image_path, scale_percent=60):
    """Resize the image by a specified scale percentage."""
    image = cv2.imread(image_path)
    width = int(image.shape[1] * scale_percent / 100)
    height = int(image.shape[0] * scale_percent / 100)
    resized_image = cv2.resize(image, (width, height), interpolation=cv2.INTER_AREA)
    return resized_image

def convert_to_grayscale(image):
    """Convert the input image to grayscale."""
    return cv2.cvtColor(image, cv2.COLOR_RGB2GRAY)

def generate_pixel_frequencies(grayscale_image):
    """Generate the frequency of pixels in the grayscale image using the Node class."""
    height, width = grayscale_image.shape
    node = Node()
    node.setFrequencePixels(grayscale_image, height, width)
    return node.returnArrayNode()

def build_huffman_tree(node_array):
    """Build the Huffman tree using the Heap class."""
    heap = Heap(node_array)
    smallest_node = heap.returnHeapMinimum()
    huffman = Huffman(smallest_node)
    return huffman.returnHuff()

def compress_image(grayscale_image, huffman_tree_root):
    """Compress the grayscale image using Huffman coding."""
    height, width = grayscale_image.shape
    huffman = Huffman(huffman_tree_root)
    huffman_dict = huffman.goThroughTree(huffman_tree_root)
    compressed_data, data_length = huffman.compressOp(grayscale_image, width, height, huffman_dict)
    return compressed_data, data_length, huffman_dict

def write_compressed_data(compressed_data, data_length, output_file):
    """Write the compressed data to a file."""
    with open(output_file, 'w') as file:
        file.write(' '.join(map(str, compressed_data[:data_length])))

def decompress_image(file_path, huffman_tree_root):
    """Decompress the image data using Huffman decoding."""
    huffman = Huffman(huffman_tree_root)
    with open(file_path, 'r') as file:
        return huffman.decompressOp(file, huffman_tree_root)

def display_images(original_image, grayscale_image):
    """Display the original and grayscale images."""
    cv2.imshow("Original Image", original_image)
    cv2.imshow("Resized Grayscale Image", grayscale_image)
    cv2.waitKey(0)
    cv2.destroyAllWindows()

def main():
    image_path = "img1.tif"
    compressed_file_path = 'compressed_image.txt'

    # Step 1: Read and resize the image
    resized_image = resize_image(image_path)

    # Step 2: Convert the resized image to grayscale
    grayscale_image = convert_to_grayscale(resized_image)

    # Step 3: Generate frequency of pixels
    node_array = generate_pixel_frequencies(grayscale_image)

    # Step 4: Build the Huffman tree
    huffman_tree_root = build_huffman_tree(node_array)

    # Step 5: Compress the image data using Huffman coding
    compressed_data, data_length, _ = compress_image(grayscale_image, huffman_tree_root)

    # Step 6: Write the compressed data to a file
    write_compressed_data(compressed_data, data_length, compressed_file_path)

    # Step 7: Decompress the image data (optional display, as per requirement)
    decompressed_data = decompress_image(compressed_file_path, huffman_tree_root)

    # Step 8: Display the original and grayscale images
    display_images(cv2.imread(image_path), grayscale_image)

if __name__ == '__main__':
    main()
