def write_file(file_name: str, data: str) -> None:
    """
    Writes the given string data to a file in binary mode, encoding each character as a byte.

    Args:
        file_name (str): The name of the file to write to.
        data (str): The string data to write to the file.
    """
    with open(file_name, 'wb') as file:
        file.write(data.encode('utf-8'))

def read_file(file_name: str) -> None:
    """
    Reads and prints each byte from the file.

    Args:
        file_name (str): The name of the file to read from.
    """
    with open(file_name, 'rb') as file:
        for byte in iter(lambda: file.read(1), b''):
            print(byte)

if __name__ == '__main__':
    file_name = 'test.txt'
    data = '中文'
    write_file(file_name, data)
    read_file(file_name)
