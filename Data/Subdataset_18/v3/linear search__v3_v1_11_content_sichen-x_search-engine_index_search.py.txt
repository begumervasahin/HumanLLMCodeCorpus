from collections import defaultdict
from words import get_text, words  # Assuming these functions are implemented correctly

def create_index(files):
    """
    Build an index mapping words to sets of document IDs based on the provided list of filenames.
    
    Args:
        files (list of str): List of fully-qualified filenames.
        
    Returns:
        dict: A dictionary where keys are words and values are sets of document IDs.
    """
    if not files:
        return None

    index = defaultdict(set)
    for doc_id, file_path in enumerate(files):
        file_content = get_text(file_path)
        key_words = words(file_content)
        for word in key_words:
            index[word].add(doc_id)
    return index

def index_search(files, index, terms):
    """
    Search for documents containing all words in the provided terms list using the given index.
    
    Args:
        files (list of str): List of fully-qualified filenames.
        index (dict): Index mapping words to sets of document IDs.
        terms (list of str): List of search terms.
        
    Returns:
        list: List of filenames containing all search terms.
    """
    if not (files and index and terms):
        return None

    doc_id_sets = [index.get(term) for term in terms if term in index]

    if not doc_id_sets or len(doc_id_sets) != len(terms):
        return None

    matching_doc_ids = set.intersection(*doc_id_sets)
    return [files[doc_id] for doc_id in matching_doc_ids]

# Example usage
if __name__ == "__main__":
    files = ["file1.txt", "file2.txt", "file3.txt"]

    # Create the index
    index = create_index(files)

    # Search for specific terms
    search_terms = ["term1", "term2"]
    result = index_search(files, index, search_terms)

    print("Matching files:", result)
