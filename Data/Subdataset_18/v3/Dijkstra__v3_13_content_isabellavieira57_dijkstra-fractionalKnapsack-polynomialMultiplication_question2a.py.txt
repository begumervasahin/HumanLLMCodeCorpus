from __future__ import print_function, division
import CPUtimer
from data2 import instance_iterator, print_solution

def solve(instance_path):
    """Solve the knapsack problem for each instance in the given path."""
    timer = CPUtimer.CPUTimer()
    
    for instance in instance_iterator(instance_path):
        instance_name, capacity, profits, weights = instance
        
        timer.reset()
        timer.start()
        
        for _ in range(2):  # Looping twice to measure execution time
            items = knapsack(capacity, profits, weights)
            timer.lap()
        
        timer.stop()
        print_solution(profits, weights, items, instance_name, '2a', timer)

def knapsack(capacity, profits, weights):
    """Implement the fractional knapsack problem using a greedy approach."""
    # Create a list of items with their indices, weights, and profit-to-weight ratios
    items = [(i + 1, w, p / w) for i, (p, w) in enumerate(zip(profits, weights))]
    # Sort items based on their profit-to-weight ratios in descending order
    items = sorted(items, key=lambda item: item[2], reverse=True)
    
    x = [0] * len(profits)
    total_weight = 0
    
    # Iterate through the sorted items and fill the knapsack
    for i, w, _ in items:
        if total_weight + w <= capacity:
            x[i - 1] = 1
            total_weight += w
        else:
            x[i - 1] = (capacity - total_weight) / w
            total_weight = capacity
            break

    # Return a list of items that are included in the knapsack with their respective fractions
    return [(i + 1, frac) for i, frac in enumerate(x) if frac > 0]
