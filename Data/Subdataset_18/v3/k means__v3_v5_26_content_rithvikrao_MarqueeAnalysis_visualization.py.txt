import numpy as np
import matplotlib.pyplot as plt

def load_data(file_path):
    """
    Load data from a .npy file.
    
    Parameters:
    file_path (str): Path to the .npy file containing the data.
    
    Returns:
    numpy.ndarray: Loaded data array.
    """
    return np.load(file_path)

def extract_scores(data, ticker):
    """
    Extract scores for a given ticker from the data array.
    
    Parameters:
    data (numpy.ndarray): Array containing the data.
    ticker (str): Ticker symbol of the company to extract data for.
    
    Returns:
    tuple: Company name and lists of financial returns, growth, multiple, and integrated scores.
    """
    financial_returns_scores = []
    growth_scores = []
    multiple_scores = []
    integrated_scores = []
    company_name = ""

    for row in data:
        if row[1] == ticker:
            company_name = row[0]
            financial_returns_scores.append(row[5])
            growth_scores.append(row[6])
            multiple_scores.append(row[7])
            integrated_scores.append(row[8])

    return company_name, financial_returns_scores, growth_scores, multiple_scores, integrated_scores

def plot_scores(company_name, start_time, end_time, ticker, financial_returns_scores, growth_scores, multiple_scores, integrated_scores):
    """
    Plot scores and save the plot as an image file.
    
    Parameters:
    company_name (str): Name of the company.
    start_time (str): Start time for the plot.
    end_time (str): End time for the plot.
    ticker (str): Ticker symbol of the company.
    financial_returns_scores (list): List of financial return scores.
    growth_scores (list): List of growth scores.
    multiple_scores (list): List of multiple scores.
    integrated_scores (list): List of integrated scores.
    """
    plt.figure(figsize=(10, 6))
    plt.plot(financial_returns_scores, label='Financial Return Score')
    plt.plot(growth_scores, label='Growth Score')
    plt.plot(multiple_scores, label='Multiple Score')
    plt.plot(integrated_scores, label='Integrated Score')

    plt.title(f"Metrics for {company_name} from {start_time} to {end_time}")
    plt.legend()
    plt.ylabel("Score")
    plt.xlabel("Time")

    output_filename = f"{ticker}_data.png"
    plt.savefig(output_filename)
    plt.show()

    print(f"Plot saved as {output_filename}")

def get_graph(ticker, start_time, end_time):
    """
    Generate a graph for a given ticker and time range.
    
    Parameters:
    ticker (str): Ticker symbol of the company.
    start_time (str): Start time for the plot.
    end_time (str): End time for the plot.
    """
    data = load_data('ticker_data_file.npy')
    company_name, financial_returns_scores, growth_scores, multiple_scores, integrated_scores = extract_scores(data, ticker)
    plot_scores(company_name, start_time, end_time, ticker, financial_returns_scores, growth_scores, multiple_scores, integrated_scores)

# Example usage
if __name__ == "__main__":
    get_graph('LULU', '2012-01-01', '2018-01-01')
