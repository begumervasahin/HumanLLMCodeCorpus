import utils
import filenames

class Log:
    def __init__(self, k, des_name, svm_kernel):
        self.text = ""
        self.k = k
        self.des_name = des_name
        self.svm_kernel = svm_kernel

    def save(self):
        """Saves the log text to a file."""
        log_filename = filenames.log(self.k, self.des_name, self.svm_kernel)
        with open(log_filename, "w") as file:
            file.write(self.text)

    def _log_time(self, description, time):
        """Helper method to format and log the time taken for a given process."""
        elapsed_time = utils.humanize_time(time)
        self.text += f"{description} {elapsed_time}.\n"

    def train_des_time(self, time):
        """Logs the time taken to get local descriptors of the training images."""
        self._log_time("Time for getting all the local descriptors of the training images was", time)

    def codebook_time(self, time):
        """Logs the time taken to generate the codebook with k-means."""
        self._log_time("Time for generating the codebook with k-means was", time)

    def train_vlad_time(self, time):
        """Logs the time taken to get VLAD global descriptors for training images."""
        self._log_time("Time for getting VLAD global descriptors of the training images was", time)

    def svm_time(self, time):
        """Logs the time taken to calculate the SVM."""
        self._log_time("Time for calculating the SVM was", time)

    def test_vlad_time(self, time):
        """Logs the time taken to get VLAD global descriptors for testing images."""
        self._log_time("Time for getting VLAD global descriptors of the testing images was", time)

    def predict_time(self, time):
        """Logs the elapsed time for predicting the testing set."""
        self._log_time("Elapsed time predicting the testing set is", time)

    def accuracy(self, accuracy):
        """Logs the accuracy of the model."""
        self.text += f"Accuracy = {accuracy}.\n"

    def classes(self, classes):
        """Logs the classes involved in the classification."""
        self.text += f"Classes = {classes}\n"

    def classes_counts(self, classes_counts):
        """Logs the counts of local descriptors for each class."""
        self.text += f"Classes Local Descriptors Counts = {classes_counts}\n"

    def confusion_matrix(self, conf_mat):
        """Logs the confusion matrix."""
        self.text += f"Confusion Matrix =\n{conf_mat}\n"

# Example usage
if __name__ == "__main__":
    log = Log(k=10, des_name="SIFT", svm_kernel="linear")
    
    # Log various times and metrics
    log.train_des_time(120)
    log.codebook_time(300)
    log.train_vlad_time(180)
    log.svm_time(60)
    log.test_vlad_time(90)
    log.predict_time(30)
    
    # Log accuracy and class information
    log.accuracy(0.85)
    log.classes(["cat", "dog"])
    log.classes_counts({"cat": 100, "dog": 120})
    log.confusion_matrix([[90, 10], [20, 100]])
    
    # Save the log to a file
    log.save()
