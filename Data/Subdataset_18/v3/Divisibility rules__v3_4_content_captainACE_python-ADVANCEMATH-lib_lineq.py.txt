def solve_linear_equation_cramers_rule():
    """
    Solve a system of linear equations using Cramer's Rule.
    The equations are of the form:
    a*x + b*y = c
    k*x + l*y = m
    """
    print("SOLVE A LINEAR EQUATION BY CRAMER'S RULE OF DISCRIMINANTS")

    # Input for the first equation
    print("Equation no. 1")
    a = get_coefficient("a")
    b = get_coefficient("b")
    c = get_coefficient("c")
    print_equation(a, b, c)

    print("----------------------------")

    # Input for the second equation
    print("Equation no. 2")
    k = get_coefficient("k")
    l = get_coefficient("l")
    m = get_coefficient("m")
    print_equation(k, l, m)

    print("--------------------------------------------")
    print("The equations will be solved by Cramer's rule.")

    # Calculate determinants
    d = calculate_determinant(a, b, k, l)
    dx = calculate_determinant(c, b, m, l)
    dy = calculate_determinant(a, c, k, m)

    # Check if there is a unique solution
    if d == 0:
        if dx == 0 and dy == 0:
            print("The system has infinitely many solutions.")
        else:
            print("The system has no solution.")
    else:
        x = dx / d
        y = dy / d
        print(f'd = {d}')
        print(f'dx = {dx}')
        print(f'dy = {dy}')
        print(f'x = {x}')
        print(f'y = {y}')

def get_coefficient(name):
    """
    Prompt the user to enter a coefficient for the linear equation.

    Args:
        name (str): The name of the coefficient (e.g., 'a', 'b', 'c').

    Returns:
        int: The coefficient entered by the user.
    """
    return int(input(f"Enter the coefficient '{name}': "))

def calculate_determinant(a, b, c, d):
    """
    Calculate the determinant of a 2x2 matrix.

    Args:
        a, b, c, d (int): The elements of the matrix.

    Returns:
        int: The determinant of the matrix.
    """
    return a * d - b * c

def print_equation(a, b, c):
    """
    Print the linear equation in the form ax + by = c.
    
    Args:
        a (int): Coefficient of x.
        b (int): Coefficient of y.
        c (int): Constant term.
    """
    equation = []
    if a != 0:
        equation.append(f"{a}x")
    if b != 0:
        if b > 0 and a != 0:
            equation.append(f"+ {b}y")
        else:
            equation.append(f"{b}y")
    equation.append(f"= {c}")
    print(" ".join(equation))

if __name__ == "__main__":
    solve_linear_equation_cramers_rule()
