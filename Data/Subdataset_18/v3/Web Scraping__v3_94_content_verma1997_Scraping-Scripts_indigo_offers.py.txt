def get_random_header():
    """Return a random user agent header from the predefined list."""
    return {'User-Agent': choice(HEADERS)}

def scrape_indigo_offers(url):
    """Scrape Indigo offers from the provided URL and return a list of offer details."""
    response = requests.get(url, headers=get_random_header())
    soup = BeautifulSoup(response.text, 'html.parser')

    offers = soup.findAll("div", {"class": "commonBottomAdon"})
    offer_data = []

    for offer in offers:
        link = offer.find("a")['href']
        image = "https://www.goindigo.in" + offer.find("img")['src']
        name = offer.find("h6").text.replace(",", "")
        description = offer.find("p").text.replace(",", "")
        
        offer_data.append((name, link, description, image))
    
    return offer_data

def save_offers_to_csv(offer_data, filename="indigo_offers.csv"):
    """Save the scraped offer data to a CSV file."""
    with open(filename, "w", encoding="utf-8") as file:
        headers = "Offer Name,Link,Description,Image Link\n"
        file.write(headers)
        
        for offer in offer_data:
            file.write(",".join(offer) + "\n")

def main():
    """Main function to scrape Indigo offers and save them to a CSV file."""
    url = "https://www.goindigo.in/campaigns/indigo-offers.html"
    offers = scrape_indigo_offers(url)
    save_offers_to_csv(offers)

if __name__ == "__main__":
    main()
