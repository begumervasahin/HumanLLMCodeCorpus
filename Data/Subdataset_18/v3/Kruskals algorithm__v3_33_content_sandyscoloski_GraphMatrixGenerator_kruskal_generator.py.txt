def generate_kruskal_edges(matrix, output_file='kruskal.txt'):
    """
    Generate a list of edges from the adjacency matrix and write them to a file.
    Only considers the upper triangular part of the matrix (excluding the diagonal).
    
    Parameters:
        matrix: 2D list representing the adjacency matrix of the graph
        output_file: Name of the file to write the edges to (default is 'kruskal.txt')
    """
    with open(output_file, 'w') as file:
        size = len(matrix)
        for i in range(size):
            for j in range(i + 1, size):
                if matrix[i][j] != 0:  # Consider only non-zero edges
                    file.write(f'{i} {j} {matrix[i][j]}\n')

def read_adjacency_matrix(file_name):
    """
    Read an adjacency matrix from a file and return it as a 2D list.
    
    Parameters:
        file_name: Name of the file containing the adjacency matrix
    
    Returns:
        matrix: A 2D list representing the adjacency matrix of the graph
    """
    with open(file_name, 'r') as file:
        lines = file.readlines()

    matrix = [list(map(int, line.strip().split())) for line in lines]

    return matrix

def main():
    # File containing the adjacency matrix
    matrix_file = 'matrix.txt'

    # Read the adjacency matrix from the file
    adjacency_matrix = read_adjacency_matrix(matrix_file)

    # Generate the edges for Kruskal's algorithm and write them to a file
    generate_kruskal_edges(adjacency_matrix)

if __name__ == '__main__':
    main()
