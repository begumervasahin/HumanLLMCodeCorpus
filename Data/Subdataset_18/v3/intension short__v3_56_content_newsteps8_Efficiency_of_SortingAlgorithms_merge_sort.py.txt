import time

def merge_sort(arr):
    """Recursively divides and merges the array, counting the number of comparisons."""
    def merge(left, right):
        """Merge two sorted halves into a single sorted array."""
        merged = []
        i = j = 0
        while i < len(left) and j < len(right):
            merge_sort.comparison_count += 1
            if left[i] < right[j]:
                merged.append(left[i])
                i += 1
            else:
                merged.append(right[j])
                j += 1
        
        # Add remaining elements from both halves, if any
        merged.extend(left[i:])
        merged.extend(right[j:])
        return merged

    if len(arr) <= 1:
        return arr

    mid = len(arr) // 2
    left_half = merge_sort(arr[:mid])
    right_half = merge_sort(arr[mid:])
    return merge(left_half, right_half)

def print_list(arr):
    """Prints the elements of the array in a single line."""
    print(" ".join(map(str, arr)))

def main():
    # Start the timer
    start_time = time.time()

    # Read the input file "case"
    with open("case", "r") as file:
        arr = file.read().split()

    # Convert the read strings to integers
    array_to_sort = [int(num) for num in arr]

    # Initialize comparison count
    merge_sort.comparison_count = 0

    # Sort the array using merge sort
    sorted_array = merge_sort(array_to_sort)

    # Print the sorted array
    print("Sorted array is:")
    print_list(sorted_array)

    # Print the total number of comparisons made
    print(f"Number of comparisons: {merge_sort.comparison_count}")

    # Stop the timer and print the elapsed time
    end_time = time.time()
    print(f"Time taken: {end_time - start_time:.6f} seconds")

if __name__ == "__main__":
    main()
