def tuple_sum(A, B):
    """
    Sums corresponding elements of tuples in two lists.
    
    Args:
      A (list of tuples): The first list of tuples.
      B (list of tuples): The second list of tuples.
      
    Returns:
      list of tuples: A list where each tuple contains the sum of corresponding elements from A and B.
    
    Examples:
    >>> tuple_sum([(1,2), (10,20)],[(3,4), (30,40)])
    [(4, 6), (40, 60)]
    >>> tuple_sum([(0,1),(-1,0),(2,2)], [(3,4),(5,6),(7,8)])
    [(3, 5), (4, 6), (9, 10)]
    """
    return [(a[0] + b[0], a[1] + b[1]) for a, b in zip(A, B)]

def inv_dict(d):
    """
    Inverts a dictionary.
    
    Args:
      d (dict): A dictionary representing an invertible function.
    
    Returns:
      dict: A dictionary representing the inverse of the input dictionary.
    
    Examples:
    >>> inv_dict({'goodbye': 'au revoir', 'thank you': 'merci'})
    {'au revoir': 'goodbye', 'merci': 'thank you'}
    """
    return {v: k for k, v in d.items()}

def row(p, n):
    """
    Generates a list of n elements starting from p.
    
    Args:
      p (int): The starting number.
      n (int): The number of elements to generate.
    
    Returns:
      list of int: A list where each element is p + i for i in range(n).
    
    Examples:
    >>> row(10, 4)
    [10, 11, 12, 13]
    """
    return [p + i for i in range(n)]

# List comprehensions to generate rows
comprehension_with_row = [row(i, 20) for i in range(15)]
comprehension_without_row = [[i + j for j in range(20)] for i in range(15)]

# Functions for probability calculations
def f(n):
    return n % 2

def g(n):
    return n % 3

# Calculate probabilities
n_values = list(range(1, 101))

Pr_f_is_even = len([n for n in n_values if f(n) == 0]) / len(n_values)
Pr_f_is_odd = len([n for n in n_values if f(n) != 0]) / len(n_values)
Pr_g_is_1 = len([n for n in n_values if g(n) == 1]) / len(n_values)
Pr_g_is_0or2 = len([n for n in n_values if g(n) == 0 or g(n) == 2]) / len(n_values)
