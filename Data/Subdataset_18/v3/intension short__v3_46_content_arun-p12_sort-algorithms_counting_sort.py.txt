def counting_sort(arr, verbose=0, desc=False):
    """
    Perform counting sort on the given list of integers.

    Parameters:
    arr (list): The list of integers to sort.
    verbose (int): Verbosity level for debugging (0: no output, 1: basic, 2: detailed).
    desc (bool): If True, sorts the array in descending order.

    Returns:
    list: A sorted list of integers.
    """
    if not arr:
        return []

    # Identify the smallest and largest elements
    min_value = min(arr)
    max_value = max(arr)
    
    # Normalize the array by shifting the minimum value to zero
    normalized_arr = [x - min_value for x in arr] if min_value < 0 else arr

    if verbose:
        print("Normalized array:", normalized_arr)
    
    # Initialize count array based on the range of the input array
    range_size = max_value - min_value + 1
    count = [0] * range_size

    # Populate the count array with frequency of each element
    for number in normalized_arr:
        count[number] += 1
    
    # Convert count array to cumulative count array
    for i in range(1, range_size):
        count[i] += count[i - 1]
        if verbose == 2:
            print(f"Cumulative count at index {i}: {count}")
    
    # Build the sorted output array
    output = [0] * len(arr)
    for number in reversed(normalized_arr):
        count[number] -= 1
        output[count[number]] = number

    # Reverse the normalization by shifting values back
    sorted_arr = [x + min_value for x in output] if min_value < 0 else output
    
    # Reverse the array if descending order is requested
    if desc:
        sorted_arr.reverse()
    
    return sorted_arr

# Example usage
if __name__ == "__main__":
    sample_array = [4, 2, 2, 8, 3, 3, 1]
    sorted_array = counting_sort(sample_array, verbose=1)
    print("Sorted array:", sorted_array)
