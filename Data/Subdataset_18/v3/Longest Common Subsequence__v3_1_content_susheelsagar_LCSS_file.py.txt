import numpy as np

def write_to_file(filename, data, message):
    """
    Write data and message to a file.
    
    Parameters:
    filename (str): The name of the file.
    data (list): The data to write.
    message (str): The message to write.
    """
    with open(filename, "r+") as f:
        f.write(str(data))      # Write the list as a string
        f.write(message)        # Write the message
        f.write("\n")           # Write a newline character
        f.write(str(data))      # Write the list as a string again

def create_and_modify_array():
    """
    Create a 2x2 numpy array, modify its elements, and print specific values.
    
    Returns:
    tuple: The modified values at specific positions in the array.
    """
    similarity = np.zeros((2, 2))  # Create a 2x2 array filled with zeros
    
    similarity[1, 0] = 2
    similarity[0, 1] = similarity[1, 0]  # Set the symmetric element
    similarity[1, 0] -= 1                # Decrement the original element
    
    return similarity[1, 0], similarity[0, 1]

if __name__ == "__main__":
    # Define the data and message
    data = [0, ",", 1]
    message = "hello"

    # Write data and message to the file
    write_to_file("distance2.csv", data, message)

    # Create and modify the array, then print the results
    modified_values = create_and_modify_array()
