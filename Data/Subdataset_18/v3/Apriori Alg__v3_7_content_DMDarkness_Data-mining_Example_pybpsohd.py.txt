import pybpsohd as pybp  # Import the package of BPSO-HD
import re  # Import the regular expression module to split the string and extract items from transactions

def read_dataset(file_path):
    """
    Read the transactional dataset and store it in a list of transactions.

    Args:
        file_path (str): Path to the dataset file.

    Returns:
        list: List of transactions, where each transaction is a list of items.
    """
    transactions = []
    with open(file_path, 'r') as file:
        content = file.readlines()  # Read all the lines from the dataset
        for line in content:
            items = re.split(r' |\n', line)  # Split the dataset by space and newline
            transaction = [int(item) for item in items if item.isdigit()]  # Extract items that are digits
            if transaction:
                transactions.append(transaction)
    return transactions

if __name__ == "__main__":
    # Path to the transactional dataset
    dataset_path = "kosarak.dat"
    
    # Read the transactional dataset and store it in the dataset variable
    dataset = read_dataset(dataset_path)

    # Mine frequent itemsets from the dataset using BPSO-HD
    # Minimum support: 0.00001
    # Population size: 30
    # Number of generations: 30
    # Inertia weight: 0.5
    # Acceleration coefficients: 1, 1
    # Minimum length of itemsets: 10
    frequent_itemsets = pybp.getFP(dataset, 0.00001, 30, 30, 0.5, 1, 1, 10)

    # Print the frequent itemsets
    for itemset in frequent_itemsets:
        print(itemset)
