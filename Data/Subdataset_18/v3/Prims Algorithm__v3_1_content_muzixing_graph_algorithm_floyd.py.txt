def floyd(graph):
    length = len(graph)
    path = {i: {} for i in range(length)}

    for i in range(length):
        for j in range(length):
            if i != j:
                path[i][j] = [i, j]
                for k in range(length):
                    if k != j and graph[i][j] > graph[i][k] + graph[k][j]:
                        graph[i][j] = graph[i][k] + graph[k][j]
                        path[i][j].insert(-1, k)
    
    return graph, path

def floyd_dict(graph):
    path = {src: {} for src in graph}

    for src in graph:
        for dst in graph[src]:
            if src != dst:
                path[src][dst] = [src, dst]
                for mid in graph:
                    if mid != dst and graph[src][dst] > graph[src][mid] + graph[mid][dst]:
                        graph[src][dst] = graph[src][mid] + graph[mid][dst]
                        path[src][dst].insert(-1, mid)
    
    return graph, path
