import numpy as np
import copy

def similarity_lcss(traj1, traj2):
    """Calculate the LCSS-based similarity between two trajectories."""
    score = 0
    min_traj_len = min(len(traj1), len(traj2))
    days = ["Fri", "Sat", "Sun", "Mon", "Tue", "Wed", "Thu"]

    traj1_days = {day: [] for day in days}
    traj2_days = {day: [] for day in days}

    # Distribute trajectory points into day-specific lists
    for row in traj1:
        traj1_days[row[1]].append(row)
    for row in traj2:
        traj2_days[row[1]].append(row)

    # Calculate similarity scores for each day
    for day in days:
        if len(traj1_days[day]) < len(traj2_days[day]):
            score += cal_sim(traj1_days[day], traj2_days[day])
        else:
            score += cal_sim(traj2_days[day], traj1_days[day])

    # Calculate normalized distance
    distance = 1 - (float(score) / min_traj_len)
    return distance

def cal_sim(traj1, traj2):
    """Calculate similarity score between two trajectories based on matching points."""
    score = 0
    for point in traj1:
        if any(traj2_point[2] == point[2] and traj2_point[0] == point[0] for traj2_point in traj2):
            score += 1
    return score

def read_trajectories(file_path):
    """Read trajectories from a CSV file and organize them by user."""
    traj1 = []
    lst_users = []
    traj2 = []

    with open(file_path, 'r') as f:
        next(f)  # Skip header
        for line in f:
            m = line.strip().split(",")
            user_id = int(m[0])
            tuple = [m[1], m[2], m[3]]

            if lst_users and user_id == int(lst_users[-1]):
                traj2.append(tuple)
            else:
                if traj2:
                    traj1.append(traj2)
                traj2 = [tuple]
                lst_users.append(m[0])

        if traj2:
            traj1.append(traj2)

    return traj1, lst_users

def write_distance_matrix(file_path, lst_users, distance_matrix, start_index=1698):
    """Write the distance matrix to a CSV file."""
    with open(file_path, 'w') as f:
        for j in range(start_index, len(lst_users) - 1):
            f.write(f"{lst_users[j]},{','.join(map(str, distance_matrix[j][j:]))}\n")

def main():
    # Initialize parameters
    num_users = 8357
    start_index = 1698
    distance_matrix = np.zeros((num_users, num_users))

    # File paths
    file_path = 'v01_anonymized_mobility.csv'
    output_file_with_id = 'distance_matrix_1698.csv'
    output_file_without_id = 'distance_matrix_1698_withoutid.csv'

    # Read trajectories
    traj1, lst_users = read_trajectories(file_path)

    # Print the 1697th user ID for verification
    print(lst_users[1697])

    # Calculate the distance matrix
    for j in range(start_index, num_users - 1):
        print(f"Processing user {j}...")
        user1 = copy.deepcopy(traj1[j])
        for i in range(j + 1, num_users):
            user2 = copy.deepcopy(traj1[i])
            distance_matrix[j][i] = similarity_lcss(user1, user2)
            distance_matrix[i][j] = distance_matrix[j][i]

    # Write the distance matrix to CSV files
    write_distance_matrix(output_file_with_id, lst_users, distance_matrix, start_index)
    write_distance_matrix(output_file_without_id, lst_users, distance_matrix, start_index)

if __name__ == "__main__":
    main()
