class Node:
    def __init__(self, data):
        self.data = data
        self.left = None
        self.right = None

def print_level_order(root):
    """Prints the level-order traversal of the tree."""
    if root is None:
        return

    queue = [root]
    while queue:
        node = queue.pop(0)
        print(node.data, end=' ')
        
        if node.left:
            queue.append(node.left)
        if node.right:
            queue.append(node.right)

def inorder_traversal(root):
    """Prints the in-order traversal of the tree."""
    if root is None:
        return
    inorder_traversal(root.left)
    print(root.data, end=' ')
    inorder_traversal(root.right)

def delete_deepest(root, del_node):
    """Deletes the deepest node (del_node) in the tree."""
    if root is None:
        return

    queue = [root]
    while queue:
        node = queue.pop(0)

        if node == del_node:
            node = None
            return

        if node.right:
            if node.right == del_node:
                node.right = None
                return
            else:
                queue.append(node.right)
        
        if node.left:
            if node.left == del_node:
                node.left = None
                return
            else:
                queue.append(node.left)

def delete_node(root, key):
    """Deletes the node with the given key from the tree."""
    if root is None:
        return

    queue = [root]
    key_node = None
    last_node = None

    while queue:
        last_node = queue.pop(0)

        if last_node.data == key:
            key_node = last_node

        if last_node.left:
            queue.append(last_node.left)
        if last_node.right:
            queue.append(last_node.right)

    if key_node:
        key_node.data = last_node.data
        delete_deepest(root, last_node)

if __name__ == '__main__':
    # Create the tree
    root = Node(10)
    root.left = Node(11)
    root.left.left = Node(7)
    root.left.right = Node(12)
    root.right = Node(9)
    root.right.left = Node(15)
    root.right.right = Node(8)

    print("The tree before the deletion:")
    inorder_traversal(root)
    print()
    print_level_order(root)
    print()

    # Key to be deleted
    key = 11
    delete_node(root, key)

    print("The tree after the deletion:")
    inorder_traversal(root)
    print()
    print_level_order(root)
    print()
