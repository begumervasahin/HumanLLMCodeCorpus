import argparse

def run(k, mode='knn', distance='euclidean', keep_punc=False, keep_stopwords=False):
    """Run the specified classifier with the given parameters."""
    from knn import KNNClassifier
    from ncc import NearestCentroidClassifier

    if mode == 'knn':
        clf = KNNClassifier(k, distance=distance, keep_stopwords=keep_stopwords, keep_punc=keep_punc)
    elif mode == 'ncc':
        clf = NearestCentroidClassifier(k, distance=distance, keep_stopwords=keep_stopwords, keep_punc=keep_punc)
    
    clf.score()

def main():
    """Parse command-line arguments and run the classifier."""
    parser = argparse.ArgumentParser(description="Run KNN or Nearest Centroid Classifier.")
    
    punc_group = parser.add_mutually_exclusive_group()
    punc_group.add_argument('--punc', dest='keep_punc', action='store_true', help="Keep punctuation in the text.")
    punc_group.add_argument('--no-punc', dest='keep_punc', action='store_false', help="Remove punctuation from the text.")
    
    stop_group = parser.add_mutually_exclusive_group()
    stop_group.add_argument('--stop', dest='keep_stopwords', action='store_true', help="Keep stopwords in the text.")
    stop_group.add_argument('--no-stop', dest='keep_stopwords', action='store_false', help="Remove stopwords from the text.")
    
    parser.add_argument('--distance', dest='distance', default='euclidean', help="Distance metric to use.")
    parser.add_argument('-k', dest='k', type=int, required=True, help="Number of neighbors or centroids.")
    parser.add_argument('--mode', dest='mode', default='knn', choices=['knn', 'ncc'], help="Classifier mode to use.")
    
    parser.set_defaults(keep_punc=False, keep_stopwords=False)
    
    args = parser.parse_args()
    
    run(k=args.k, mode=args.mode, distance=args.distance, keep_punc=args.keep_punc, keep_stopwords=args.keep_stopwords)

if __name__ == '__main__':
    main()
