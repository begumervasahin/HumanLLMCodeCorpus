import math
import matplotlib.pyplot as plt
from pandas import read_csv, to_datetime

def parser(date_str):
    """Parse date string to datetime object."""
    return to_datetime(date_str, format='%Y-%m-%d %H:%M:%S')

def mean_absolute_percentage_error(y_true, y_pred):
    """
    Calculate Mean Absolute Percentage Error (MAPE).

    Parameters:
    y_true (list of float): List of true values.
    y_pred (list of float): List of predicted values.

    Returns:
    float: MAPE as a percentage.
    """
    sum_error = 0
    for true, pred in zip(y_true, y_pred):
        sum_error += abs((pred - true) / true)
    return (sum_error / len(y_true)) * 100

def plot_time_series(data, title, x_label, y_label):
    """
    Plot time series data.

    Parameters:
    data (array-like): Time series data to plot.
    title (str): Title of the plot.
    x_label (str): Label for the x-axis.
    y_label (str): Label for the y-axis.
    """
    plt.plot(data, '-', label="Real Flow")
    plt.legend(loc='upper right')
    plt.xlabel(x_label)
    plt.ylabel(y_label)
    plt.title(title)
    plt.show()

def main():
    # Path to the CSV file
    csv_file_path = '/Users/daihanru/Desktop/arima-lstm/DataSet/FEB15.csv'

    # Read CSV file, parse dates, and set the first column as the index
    series = read_csv(
        csv_file_path, 
        usecols=[2, 3], 
        header=0, 
        parse_dates=[0], 
        index_col=0, 
        squeeze=True, 
        date_parser=parser
    )

    # Extract the first 1000 values for plotting
    data_to_plot = series.values[:1000]

    # Plot the time series data
    plot_time_series(
        data=data_to_plot,
        title="Vehicle Volume Over Time",
        x_label="Period (15-minute intervals)",
        y_label="Volume (vehicle/period)"
    )

    # Example data for MAPE calculation
    y_true = [1, 2, 3, 4, 5]
    y_pred = [0.9, 2.1, 3.2, 4.3, 5.1]

    # Calculate and print MAPE
    mape = mean_absolute_percentage_error(y_true, y_pred)
    print(f"Mean Absolute Percentage Error (MAPE): {mape:.2f}%")

if __name__ == "__main__":
    main()
