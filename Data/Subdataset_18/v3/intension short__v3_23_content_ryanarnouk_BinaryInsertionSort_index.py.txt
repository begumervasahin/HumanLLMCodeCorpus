def binary_search(arr, val, start, end):
    """
    Perform a binary search to find the index at which `val` should be inserted
    in the sorted portion of `arr` to maintain sorted order.
    """
    while start <= end:
        mid = (start + end) // 2
        if arr[mid] < val:
            start = mid + 1
        elif arr[mid] > val:
            end = mid - 1
        else:
            return mid
    return start

def insertion_sort(arr):
    """
    Sorts the array `arr` using an insertion sort algorithm enhanced with binary search.
    This reduces the number of comparisons needed to find the correct position.
    """
    for i in range(1, len(arr)):
        val = arr[i]
        # Find the position where the current element should be inserted
        correct_position = binary_search(arr, val, 0, i - 1)
        # Move the elements to create space for insertion
        arr = arr[:correct_position] + [val] + arr[correct_position:i] + arr[i+1:]

    return arr

if __name__ == "__main__":
    # Example usage
    unsorted_arr = [37, 23, 0, 17, 12, 72, 31, 46, 100, 88, 54]
    sorted_arr = insertion_sort(unsorted_arr)
    print("Sorted array:", sorted_arr)
