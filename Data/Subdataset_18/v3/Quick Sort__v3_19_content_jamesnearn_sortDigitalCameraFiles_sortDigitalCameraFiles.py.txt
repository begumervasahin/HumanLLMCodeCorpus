import getopt
import os
import time
import sys

def sort_files(directory, separator):
    """
    Sort media files in the given directory into subdirectories based on their modification date.

    Args:
        directory (str): The directory to sort files in.
        separator (str): The separator to use in the new directory names.
    """
    for dirpath, _, filenames in os.walk(directory):
        for filename in filenames:
            if filename.lower().endswith((".jpg", ".mp4", ".mts", ".mov")):
                move_file_to_date_folder(dirpath, filename, separator)

def move_file_to_date_folder(directory_path, filename, separator):
    """
    Move a file to a new directory based on its modification date.

    Args:
        directory_path (str): The path of the directory containing the file.
        filename (str): The name of the file to move.
        separator (str): The separator to use in the new directory names.
    """
    full_filepath = os.path.join(directory_path, filename)
    print(f"Processing file: {full_filepath}")

    destination_path = get_destination_path(full_filepath, separator)
    destination_directory = os.path.join(directory_path, destination_path)
    print(f"Destination directory: {destination_directory}")

    if not os.path.exists(destination_directory):
        os.makedirs(destination_directory)
    
    new_file_path = os.path.join(destination_directory, filename)
    print(f"Moving file to: {new_file_path}")
    os.rename(full_filepath, new_file_path)

def get_destination_path(filepath, separator):
    """
    Get the destination path for a file based on its modification date.

    Args:
        filepath (str): The full path of the file.

    Returns:
        str: The destination path for the file.
    """
    modification_time = os.path.getmtime(filepath)
    destination_path = time.strftime(f'%Y{separator}%m{separator}%d', time.localtime(modification_time))
    return destination_path

def main(argv):
    """
    Main function to parse command line arguments and call the sorting function.

    Args:
        argv (list): The list of command line arguments.
    """
    separator = os.sep  # Default separator based on OS
    directory = ""

    try:
        opts, _ = getopt.getopt(argv, "h", ["separator=", "path="])
    except getopt.GetoptError:
        print_usage()
        sys.exit(2)

    for opt, arg in opts:
        if opt in ("-h", "--help"):
            print_usage()
            sys.exit()
        elif opt == "--separator":
            separator = arg
        elif opt == "--path":
            directory = arg

    if not directory:
        print("Error: Path is required")
        print_usage()
        sys.exit(2)

    print(f'Separator: {separator}')
    print(f'Path: {directory}')
    sort_files(directory, separator)

def print_usage():
    """
    Print the usage information for the script.
    """
    print('Usage: SortFiles.py --separator <value> --path <path>')

if __name__ == "__main__":
    if len(sys.argv) > 1 and sys.argv[1] == "runtests":
        print('Running tests')
    else:
        main(sys.argv[1:])
