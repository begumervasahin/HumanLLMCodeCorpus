import pandas as pd
from naive_bayes import loocv, et

# Initialize DataFrame to store results
results_df = pd.DataFrame(columns=['Dataset', 'Accuracy(%)'])

# Define datasets and associated parameters
datasets = [
    ("tic-tac-toe.data", 'Tic-tac-toe Endgame', loocv, False, None),
    ("SPECT.train", 'SPECT Heart Dataset', et, True, "SPECT.test"),
    ("soybean-small.data", 'Soybean (small)', loocv, False, None),
    ("shuttle-landing-control.data", 'Shuttle Landing Control', loocv, False, None),
    ("monks-1.train", 'Monks-1', et, True, "monks-1.test"),
    ("monks-2.train", 'Monks-2', et, True, "monks-2.test"),
    ("monks-3.train", 'Monks-3', et, True, "monks-3.test")
]

def load_data(file, has_test, test_file=None):
    """Load dataset and split into features and labels."""
    data = pd.read_csv(file, header=None)
    if has_test:
        test_data = pd.read_csv(test_file, header=None)
        return data, test_data
    return data, None

def split_features_labels(data, dataset_name):
    """Split data into features and labels based on the dataset name."""
    if dataset_name == 'Shuttle Landing Control':
        x = data.iloc[:, 1:].values.tolist()
        y = data.iloc[:, 0].values.tolist()
    else:
        x = data.iloc[:, :-1].values.tolist()
        y = data.iloc[:, -1].values.tolist()
    return x, y

def process_dataset(index, train_file, dataset_name, method, has_test, test_file):
    """Process each dataset and compute accuracy."""
    train_data, test_data = load_data(train_file, has_test, test_file)
    
    if has_test:
        x_train, y_train = split_features_labels(train_data, dataset_name)
        x_test, y_test = split_features_labels(test_data, dataset_name)
        unique_labels = set(y_train)
        accuracy = method(x_train, x_test, y_train, y_test, unique_labels)
    else:
        x, y = split_features_labels(train_data, dataset_name)
        unique_labels = set(y)
        accuracy = method(x, y, unique_labels, train_data.shape[0])
    
    results_df.loc[index, 'Dataset'] = dataset_name
    results_df.loc[index, 'Accuracy(%)'] = accuracy
    print(f"{dataset_name}: {accuracy:.2f}%")

# Iterate through each dataset and compute accuracy
for index, dataset in enumerate(datasets):
    train_file, dataset_name, method, has_test, test_file = dataset
    process_dataset(index, train_file, dataset_name, method, has_test, test_file)

# Save results to CSV
results_df.to_csv("Data.csv", index=False)
