from Team import Team

def display_team_details(teams):
    """
    Display details of each team from the list.

    Args:
        teams (list of Team): List of Team objects to display.
    """
    print("List of Teams:")
    for team in teams:
        abbreviations = ', '.join(team.abbreviations)
        print(f"Abbreviations: {abbreviations}, Name: {team.name}")
