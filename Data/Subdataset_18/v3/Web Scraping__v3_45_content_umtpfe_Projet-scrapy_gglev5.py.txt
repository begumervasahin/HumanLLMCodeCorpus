from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.firefox.options import Options
import time

def initialize_driver():
    """
    Initialize the Firefox WebDriver in headless mode.
    
    Returns:
        WebDriver: An instance of the Firefox WebDriver.
    """
    options = Options()
    options.headless = True  # Run in headless mode
    driver = webdriver.Firefox(options=options)
    return driver

def extract_search_results(driver):
    """
    Extract search results from the current Google search results page.
    
    Args:
        driver (WebDriver): The WebDriver instance.
    
    Returns:
        list of tuple: A list of tuples containing the title and link of each search result.
    """
    results = []
    elements = driver.find_elements(By.XPATH, '//*[@id="rso"]//div[@class="g"]')
    
    for element in elements:
        try:
            title = element.find_element(By.TAG_NAME, "h3").text
            link = element.find_element(By.TAG_NAME, "a").get_attribute("href")
            results.append((title, link))
        except Exception as e:
            print(f"An error occurred while processing an element: {e}")
    
    return results

def click_next_button(driver):
    """
    Click the 'Next' button on the Google search results page to navigate to the next page.
    
    Args:
        driver (WebDriver): The WebDriver instance.
    
    Returns:
        bool: True if the 'Next' button was clicked, False otherwise.
    """
    next_buttons = driver.find_elements(By.XPATH, '//*[@id="pnnext"]')
    if next_buttons:
        next_buttons[0].click()
        time.sleep(2)  # Wait for 2 seconds to load the next page
        return True
    return False

def scrape_google_search(query):
    """
    Scrape Google search results for a given query.
    
    Args:
        query (str): The search query.
    
    Returns:
        list of tuple: A list of tuples containing the title and link of each search result.
    """
    driver = initialize_driver()
    url = f"http://www.google.com/search?q={query}"
    driver.get(url)

    all_results = []
    try:
        while True:
            results = extract_search_results(driver)
            all_results.extend(results)
            
            if not click_next_button(driver):
                break  # No more pages to navigate
    except Exception as e:
        print(f"An error occurred: {e}")
    finally:
        driver.quit()
    
    return all_results

if __name__ == "__main__":
    query = "chanel"
    results = scrape_google_search(query)
    for title, link in results:
        print(f"Title: {title}\nLink: {link}\n")
