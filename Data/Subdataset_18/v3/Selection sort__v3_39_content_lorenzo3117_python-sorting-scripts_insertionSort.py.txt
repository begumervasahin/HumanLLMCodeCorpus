def insertion_sort(array):
    """
    Perform insertion sort on the given array, counting the number of comparisons and swaps.
    """
    swaps_made = 0
    checks_made = 0

    # Traverse through 1 to len(array)
    for i in range(1, len(array)):
        value = array[i]
        value_index = i

        # Count the initial comparison to enter the while loop
        checks_made += 1
        
        # Shift elements of array[0..i-1] that are greater than value to one position ahead of their current position
        while value_index > 0 and value < array[value_index - 1]:
            array[value_index] = array[value_index - 1]
            value_index -= 1
            checks_made += 1
            swaps_made += 1
        
        # Place value at its correct position
        array[value_index] = value
    
    return checks_made, swaps_made

def main():
    """
    Main function to demonstrate insertion sort.
    """
    # Example array to be sorted
    arr = [12, 11, 13, 5, 6]
    
    # Print the array before sorting
    print("Array before sorting:")
    print(arr)
    
    # Sort the array using insertion sort
    comparisons, swaps = insertion_sort(arr)
    
    # Print the sorted array
    print("Sorted array is:")
    print(arr)
    
    # Print the number of comparisons and swaps
    print(f"\nINSERTION SORT:\nComparisons: {comparisons}\nSwaps: {swaps}")

if __name__ == "__main__":
    main()
