import numpy as np

def create_matrix_from_string(matrix_str):
    """
    Converts a formatted string into a NumPy matrix.

    Parameters:
    matrix_str (str): The input string representing the matrix with semicolons as row delimiters.

    Returns:
    np.matrix: The created NumPy matrix.
    """
    # Replace semicolons with newlines to format the string for NumPy matrix creation
    formatted_str = matrix_str.replace(';', '\n')
    return np.matrix(formatted_str)

def main():
    """
    Demonstrates various operations on a NumPy matrix created from a formatted string.
    """
    # Define the matrix as a string with semicolons as row delimiters
    matrix_str = '1 2 3 4 ; 5 6 7 8 ; 0 1 2 3'
    
    # Create a NumPy matrix from the formatted string
    matrix = create_matrix_from_string(matrix_str)
    
    # Print the shape of the matrix
    print(f"Matrix shape: {matrix.shape}")
    
    # Access and print specific elements from the matrix
    print(f"Element at position (0, 1): {matrix[0, 1]}")
    print(f"Element at position (2, 3): {matrix[2, 3]}")
    
    # Modify an element in the matrix
    matrix[0, 1] = -8
    
    # Print the modified row
    print(f"Modified row 1: {matrix[1]}")
    
    # Print the entire column
    print(f"Column 1: {matrix[:, 1]}")

if __name__ == '__main__':
    main()
