import unittest

def check_pesel(pesel_nr):
    """
    Validates a PESEL number using the standard PESEL validation algorithm.

    Parameters:
    - pesel_nr (str): The PESEL number to be validated.

    Returns:
    - bool: True if the PESEL number is valid, False otherwise.
    """
    # Ensure PESEL number is exactly 11 digits
    if len(pesel_nr) != 11 or not pesel_nr.isdigit():
        return False

    # Define the weights for each digit position
    weights = [1, 3, 7, 9, 1, 3, 7, 9, 1, 3]
    
    # Calculate the weighted sum of the first 10 digits
    weighted_sum = sum(int(pesel_nr[i]) * weights[i] for i in range(10))

    # Determine the expected check digit
    expected_check_digit = (10 - (weighted_sum % 10)) % 10

    # Validate the check digit against the last digit of the PESEL number
    return expected_check_digit == int(pesel_nr[-1])

class TestPeselValidator(unittest.TestCase):
    def test_valid_pesel(self):
        """Test with valid PESEL numbers."""
        self.assertTrue(check_pesel("92121510595"))
        self.assertTrue(check_pesel("99810235403"))

    def test_invalid_pesel(self):
        """Test with invalid PESEL numbers."""
        self.assertFalse(check_pesel("44051401358"))  # Incorrect check digit
        self.assertFalse(check_pesel("51401358"))     # Too short
        self.assertFalse(check_pesel("12345678901"))  # Incorrect check digit

if __name__ == "__main__":
    unittest.main()
