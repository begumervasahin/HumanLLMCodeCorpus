class TreeNode:
    """
    A class representing a node in a binary tree.
    
    Attributes:
        value (int): The value stored in the node.
        left (TreeNode): Reference to the left child node.
        right (TreeNode): Reference to the right child node.
    """
    def __init__(self, value):
        self.value = value
        self.left = None
        self.right = None

def inorder_traversal(node, result_list=None):
    """
    Perform an in-order traversal of the binary tree.

    Args:
        node (TreeNode): The root node of the binary tree.
        result_list (list, optional): List to store traversal results. Defaults to None.

    Returns:
        list: The in-order traversal of the binary tree.
    """
    if result_list is None:
        result_list = []

    if node:
        inorder_traversal(node.left, result_list)
        if node.value is not None:
            result_list.append(node.value)
        inorder_traversal(node.right, result_list)
        
    return result_list

def preorder_traversal(node, result_list=None):
    """
    Perform a pre-order traversal of the binary tree.

    Args:
        node (TreeNode): The root node of the binary tree.
        result_list (list, optional): List to store traversal results. Defaults to None.

    Returns:
        list: The pre-order traversal of the binary tree.
    """
    if result_list is None:
        result_list = []

    if node:
        if node.value is not None:
            result_list.append(node.value)
        preorder_traversal(node.left, result_list)
        preorder_traversal(node.right, result_list)
        
    return result_list

def build_tree_from_list(values, index=0):
    """
    Build a binary tree from a list of values.

    Args:
        values (list): List of values to create the binary tree.
        index (int, optional): Current index in the list of values. Defaults to 0.

    Returns:
        TreeNode: The root of the binary tree.
    """
    if index >= len(values) or values[index] is None:
        return None

    root = TreeNode(values[index])
    root.left = build_tree_from_list(values, 2 * index + 1)
    root.right = build_tree_from_list(values, 2 * index + 2)
    
    return root

def print_traversal(traversal, traversal_type):
    """
    Print the traversal of the binary tree.

    Args:
        traversal (list): List of values representing the traversal.
        traversal_type (str): The type of traversal ("In-order" or "Pre-order").
    """
    print(f"{traversal_type} Traversal:")
    for value in traversal:
        print(value, end=' ')
    print('\n')

if __name__ == "__main__":
    tree_values_list = [
        [1, None, 2, None, None, 3],
        [1, 2, 3, 4, 5, 6, None],
        [1, 2, 3]
    ]

    for index, tree_values in enumerate(tree_values_list):
        root = build_tree_from_list(tree_values)
        
        inorder_result = inorder_traversal(root)
        preorder_result = preorder_traversal(root)

        print(f"Tree {index + 1}: {tree_values}")
        print_traversal(inorder_result, "In-order")
        print_traversal(preorder_result, "Pre-order")
