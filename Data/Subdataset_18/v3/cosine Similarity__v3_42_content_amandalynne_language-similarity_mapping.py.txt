class Mapping:
    def __init__(self, items):
        """
        Initialize a Mapping object with a list of items.

        Args:
        - items (list): List of items to initialize the mapping.

        Attributes:
        - item_list (list): Original list of items.
        - items (dict): Mapping of index to item.
        - index (dict): Mapping of item to index.
        """
        self.item_list = items
        self.items = {index: item for index, item in enumerate(items)}
        self.index = {item: index for index, item in enumerate(items)}

    def __contains__(self, item):
        """
        Check if an item exists in the mapping.

        Args:
        - item: Item to check for existence.

        Returns:
        - bool: True if item exists in the mapping, False otherwise.
        """
        return item in self.index

    def __getitem__(self, item):
        """
        Get the index of an item.

        Args:
        - item: Item whose index needs to be retrieved.

        Returns:
        - int: Index of the item.
        """
        return self.index[item]

    def __iter__(self):
        """
        Iterate over the items in the mapping.

        Returns:
        - iter: Iterator over the items.
        """
        return iter(self.index)

    def __len__(self):
        """
        Get the number of items in the mapping.

        Returns:
        - int: Number of items in the mapping.
        """
        return len(self.index)

    def __repr__(self):
        """
        Return a string representation of the Mapping object.

        Returns:
        - str: String representation of the Mapping object.
        """
        return f"<{self.__class__.__name__} with {len(self)} entries>"

    def add(self, item):
        """
        Add an item to the mapping if it doesn't already exist.

        Args:
        - item: Item to add to the mapping.

        Returns:
        - item: The added item.
        """
        if item not in self:
            index = len(self)
            self.items[index] = item
            self.index[item] = index
        return item

    def get(self, item, default=None):
        """
        Get the index of an item or return a default value if not found.

        Args:
        - item: Item whose index needs to be retrieved.
        - default: Default value to return if item is not found (default: None).

        Returns:
        - int or default: Index of the item or default value.
        """
        return self.index.get(item, default)

    def item(self, index):
        """
        Get the item at a specific index.

        Args:
        - index: Index of the item to retrieve.

        Returns:
        - item: Item at the specified index.
        """
        return self.items[index]

    def keys(self):
        """
        Return the original list of items in their original order.

        Returns:
        - list: Original list of items.
        """
        return self.item_list

# Example usage:
items = ['apple', 'banana', 'cherry']
mapping = Mapping(items)

# Demonstrate various methods of the Mapping class
print(f"Mapping: {mapping}")
print(f"Contains 'banana': {'banana' in mapping}")
print(f"Index of 'cherry': {mapping['cherry']}")
print(f"Number of entries: {len(mapping)}")
print(f"Item at index 1: {mapping.item(1)}")

# Add a new item and show the updated mapping
mapping.add('date')
print(f"Mapping after adding 'date': {mapping}")

# Show the keys of the mapping in their original order
print(f"Keys in original order: {mapping.keys()}")
