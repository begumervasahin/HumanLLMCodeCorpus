# Initialize a global counter for moves
move_counter = 1

def tower_of_hanoi(height, start, end, auxiliary):
    """
    Solve the Tower of Hanoi puzzle and print each move.

    Args:
        height (int): The number of disks.
        start (int): The starting rod.
        end (int): The destination rod.
        auxiliary (int): The auxiliary rod.
    """
    global move_counter
    if height >= 1:
        # Move a tower of height-1 from the start rod to the auxiliary rod
        tower_of_hanoi(height - 1, start, auxiliary, end)
        
        # Move the remaining disk from the start rod to the end rod
        print(f"{move_counter}: Move from {start} to {end}")
        move_counter += 1
        
        # Move the tower of height-1 from the auxiliary rod to the end rod
        tower_of_hanoi(height - 1, auxiliary, end, start)

def main():
    """Main function to set the number of disks and solve the Tower of Hanoi puzzle."""
    num_disks = 3  # Set the number of disks
    tower_of_hanoi(num_disks, 'A', 'C', 'B')  # Solve the puzzle

if __name__ == "__main__":
    main()
