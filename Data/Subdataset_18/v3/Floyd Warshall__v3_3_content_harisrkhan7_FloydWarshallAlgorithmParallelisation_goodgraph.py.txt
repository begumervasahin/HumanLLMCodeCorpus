import random

def generate_random_graph(num_vertices):
    """
    Generate a random adjacency matrix for a graph with the given number of vertices.
    
    Parameters:
    num_vertices (int): The number of vertices in the graph.
    
    Returns:
    list of list of int: The adjacency matrix with random edge weights.
    """
    return [[random.randint(1, 10) for _ in range(num_vertices)] for _ in range(num_vertices)]

def write_graph_to_file(filename, num_vertices, matrix):
    """
    Write the graph data to a file in the specified format.
    
    Parameters:
    filename (str): The name of the file to write the graph data to.
    num_vertices (int): The number of vertices in the graph.
    matrix (list of list of int): The adjacency matrix with edge weights.
    """
    with open(filename, "w") as file:
        file.write(f"{num_vertices}\n")
        for i in range(num_vertices):
            random_edges = random.sample(range(num_vertices), 10)
            for j in random_edges:
                file.write(f"{i + 1} {j + 1} {matrix[i][j]}\n")

def main():
    """
    Main function to generate random graphs and write them to files.
    """
    vertex_counts = [5, 10, 20, 50, 100, 200, 500, 750, 1000, 1250, 1500, 2000, 3000, 5000]
    
    for num_vertices in vertex_counts:
        # Generate the adjacency matrix for the given number of vertices
        matrix = generate_random_graph(num_vertices)
        # Define the filename based on the number of vertices
        filename = f"randograph-{num_vertices}.txt"
        # Write the graph data to the file
        write_graph_to_file(filename, num_vertices, matrix)

if __name__ == '__main__':
    main()
