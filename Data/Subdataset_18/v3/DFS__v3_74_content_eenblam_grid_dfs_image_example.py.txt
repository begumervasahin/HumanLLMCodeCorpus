from PIL import Image

def create_image():
    """
    Create and display a 100x100 image with a repeating color pattern.
    """
    # Define the color pattern
    color_pattern = [
        (0, 0, 0),       # Black
        (0, 0, 0),       # Black
        (128, 128, 128), # Gray
        (255, 255, 255), # White
        (255, 255, 255)  # White
    ]
    
    # Calculate the number of pixels needed for the image
    num_pixels = 100 * 100
    
    # Repeat the color pattern to cover the entire image
    data = (color_pattern * (num_pixels // len(color_pattern)))[:num_pixels]
    
    # Create a new RGB image with the specified size
    image = Image.new("RGB", (100, 100), "white")
    
    # Populate the image with the color data
    image.putdata(data)
    
    # Display the image
    image.show()

if __name__ == "__main__":
    create_image()
