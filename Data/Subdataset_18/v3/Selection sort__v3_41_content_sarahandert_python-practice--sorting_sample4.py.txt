import random

def generate_random_numbers(filename, count):
    """
    Generates a file with 'count' random numbers between 0 and 99.

    Args:
        filename (str): The name of the file where the numbers will be written.
        count (int): The number of random numbers to generate.
    """
    random.seed(0)  # Ensure reproducibility
    with open(filename, 'w') as file:
        for _ in range(count):
            number = random.randint(0, 99)
            file.write(f"{number}\n")

def get_user_input(prompt):
    """
    Prompts the user for input and returns the result.

    Args:
        prompt (str): The prompt message to display to the user.

    Returns:
        str: The user input.
    """
    return input(prompt)

def main():
    """
    Main function to take user inputs and generate the file with random numbers.
    """
    filename = get_user_input('Enter the filename: ')
    count = int(get_user_input('Enter the number of random numbers to generate: '))
    generate_random_numbers(filename, count)
    print(f"Generated {count} random numbers and saved them to {filename}")

if __name__ == "__main__":
    main()
