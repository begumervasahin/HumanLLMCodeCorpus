from ecc import EllipticCurve, Point
from typing import Tuple

def generate_public_key(p: int, a: int, b: int, G: Tuple[int, int], private_key: int) -> Point:
    """
    Generate a public key using the Diffie-Hellman Algorithm on an elliptic curve.

    Args:
        p (int): Prime field size.
        a (int): Coefficient of x^1 in the elliptic curve equation.
        b (int): Coefficient of x^0 in the elliptic curve equation.
        G (Tuple[int, int]): Coordinates of the generator point on the curve.
        private_key (int): The private key.

    Returns:
        Point: The generated public key.
    """
    elliptic_curve = EllipticCurve(a, b, p)
    generator_point = Point(elliptic_curve, G[0], G[1], "Generator")
    public_key = generator_point * private_key
    return public_key

def main():
    """
    Main function to demonstrate elliptic curve Diffie-Hellman key exchange.
    """
    # Define private keys for Alice and Bob
    alice_private_key = 4
    bob_private_key = 7

    # Define elliptic curve parameters
    p = 37  # Prime field size
    a = 7   # Coefficient of x^1
    b = 3   # Coefficient of x^0
    G = (2, 5)  # Generator point coordinates

    # Create the elliptic curve and generator point
    ecc = EllipticCurve(a, b, p)
    generator_point = Point(ecc, G[0], G[1], "Generator")

    # Generate public keys for Alice and Bob
    alice_public_key = generate_public_key(p, a, b, G, alice_private_key)
    alice_public_key.name = "Alice Public Key"
    
    bob_public_key = generate_public_key(p, a, b, G, bob_private_key)
    bob_public_key.name = "Bob Public Key"

    # Generate shared secrets
    shared_secret_from_bob = alice_public_key * bob_private_key
    shared_secret_from_alice = bob_public_key * alice_private_key

    # Ensure both shared secrets are equal
    assert shared_secret_from_alice == shared_secret_from_bob
    shared_secret_from_alice.name = "Shared Secret"

    # Plot points on the elliptic curve
    ecc.plot_points([generator_point, alice_public_key, bob_public_key, shared_secret_from_alice])

if __name__ == '__main__':
    main()
