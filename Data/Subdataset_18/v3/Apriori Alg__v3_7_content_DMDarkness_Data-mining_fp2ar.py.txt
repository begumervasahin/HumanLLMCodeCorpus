Transforming frequent itemsets into association rules based
on specified minimum confidence and minimum lift.

def get_subsets(pat, L, base=0, temp=None):
    """
    Get all the subsets with length L and len(pat)-L of the object pattern pat.

    Args:
        pat (list): The pattern to find subsets of.
        L (int): The length of the subsets to generate.
        base (int): The base index to start generating subsets.
        temp (list): A temporary list to store intermediate results.

    Returns:
        tuple: Two lists containing subsets and their complementary subsets.
    """
    if temp is None:
        temp = []
    
    if L < 1:
        sub1, sub2 = [sorted(temp)], [sorted(pat)]
    else:
        sub1, sub2 = [], []
        for i in range(base, len(pat)):
            t1, t2 = get_subsets(pat[:i] + pat[i+1:], L-1, i, temp + [pat[i]])
            sub1.extend(t1)
            sub2.extend(t2)
    return sub1, sub2

def generate_association_rules(fi, N, min_conf, min_lift):
    """
    Generate association rules from frequent itemsets.

    Args:
        fi (list): The list of frequent itemsets.
        N (int): The total number of transactions.
        min_conf (float): The minimum confidence threshold.
        min_lift (float): The minimum lift threshold.

    Returns:
        list: A list of generated association rules.
    """
    rules = []
    fi_dict = {tuple(sorted(itemset[:-1])): itemset[-1] for itemset in fi}

    for itemset in fi:
        pattern = itemset[:-1]
        support = itemset[-1]
        subset_len = int((len(pattern) - 1) / 2)

        for L in range(1, subset_len + 1):
            sub1, sub2 = get_subsets(pattern, L)

            for s1, s2 in zip(sub1, sub2):
                if s1 in sub2:
                    s1, s2 = [-1, -9], [-2, -5]
                
                pat1, pat2 = tuple(sorted(s1)), tuple(sorted(s2))
                if pat1 in fi_dict and pat2 in fi_dict:
                    sup1, sup2, sup3 = support, fi_dict[pat1], fi_dict[pat2]

                    # Check lift and confidence
                    lift = (sup1 * N) / (sup2 * sup3)
                    conf1, conf2 = sup1 / sup2, sup1 / sup3

                    if lift > min_lift:
                        if conf1 > min_conf:
                            rules.append([list(pat1), list(pat2), sup1, conf1, lift])
                        if conf2 > min_conf:
                            rules.append([list(pat2), list(pat1), sup1, conf2, lift])

    return rules

if __name__ == "__main__":
    # Example frequent itemsets with their support counts
    frequent_itemsets = [
        [['A'], 0.5],
        [['B'], 0.6],
        [['A', 'B'], 0.3],
        [['A', 'C'], 0.2],
        [['B', 'C'], 0.25],
        [['A', 'B', 'C'], 0.1]
    ]

    # Total number of transactions
    total_transactions = 1000

    # Minimum confidence and lift thresholds
    min_confidence = 0.7
    min_lift = 1.2

    # Generate association rules
    rules = generate_association_rules(frequent_itemsets, total_transactions, min_confidence, min_lift)

    # Print the generated rules
    for rule in rules:
        print(f"Rule: {rule[0]} -> {rule[1]}, Support: {rule[2]:.4f}, Confidence: {rule[3]:.4f}, Lift: {rule[4]:.4f}")
