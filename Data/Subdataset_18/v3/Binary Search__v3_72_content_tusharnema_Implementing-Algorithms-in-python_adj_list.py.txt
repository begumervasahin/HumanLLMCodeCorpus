from collections import defaultdict, deque

class Graph:
    """Graph data structure, undirected by default."""

    def __init__(self, connections, directed=False):
        self.graph = defaultdict(set)
        self.directed = directed
        self._add_connections(connections)

    def _add_connections(self, connections):
        """Add a list of connections (tuples) to the graph."""
        for node1, node2 in connections:
            self.add_edge(node1, node2)

    def add_edge(self, node1, node2):
        """Add an edge between node1 and node2."""
        self.graph[node1].add(node2)
        if not self.directed:
            self.graph[node2].add(node1)

    def remove_node(self, node):
        """Remove all edges associated with a node."""
        self.graph.pop(node, None)
        for neighbors in self.graph.values():
            neighbors.discard(node)

    def is_connected(self, node1, node2):
        """Check if node1 is directly connected to node2."""
        return node2 in self.graph[node1]

    def find_path(self, start, end, path=None):
        """Find any path between start and end (may not be the shortest)."""
        if path is None:
            path = []

        path.append(start)

        if start == end:
            return path

        if start not in self.graph:
            return None

        for neighbor in self.graph[start]:
            if neighbor not in path:
                new_path = self.find_path(neighbor, end, path)
                if new_path:
                    return new_path

        path.pop()
        return None

    def bfs(self, start):
        """Perform Breadth-First Search (BFS) starting from a given node."""
        visited = set()
        queue = deque([start])

        while queue:
            node = queue.popleft()
            if node not in visited:
                print(node, end=" --> ")
                visited.add(node)
                queue.extend(self.graph[node] - visited)
        print("End")

    def dfs(self, start):
        """Perform Depth-First Search (DFS) starting from a given node."""
        visited = set()
        self._dfs_recursive(start, visited)
        print("End")

    def _dfs_recursive(self, node, visited):
        """Helper function for DFS to track visited nodes."""
        if node not in visited:
            print(node, end=" --> ")
            visited.add(node)
            for neighbor in self.graph[node]:
                self._dfs_recursive(neighbor, visited)

    def __str__(self):
        return f"{self.__class__.__name__}({dict(self.graph)})"

    def print_graph(self):
        """Print the graph structure."""
        for node, connections in self.graph.items():
            connections_str = ', '.join(map(str, connections))
            print(f"{node} ---> {connections_str}")

# Example usage:
if __name__ == "__main__":
    connections = [
        (0, 1), (1, 2), (2, 3), (2, 4),
        (3, 4), (5, 6), (7, 3)
    ]
    
    graph = Graph(connections)
    
    # Add additional connections
    graph.add_edge(6, 8)
    graph.add_edge(8, 4)
    graph.add_edge(9, 10)

    # Print the graph structure
    graph.print_graph()
    print()

    # Perform BFS and DFS starting from node 0
    print("BFS traversal starting from node 0:")
    graph.bfs(0)
    print()

    print("DFS traversal starting from node 0:")
    graph.dfs(0)
