Encapsulates node data for use in data structures such as union-find.
"""

class Node:
    def __init__(self, value):
        """
        Initializes a Node object.

        Parameters:
        - value (int or str): The value associated with the node.
        """
        self.value = value  # The value of the node
        self.parent = self  # Initially, the node is its own parent (self-referential)
        self.rank = 0  # The rank of the node, used to keep the union-find tree balanced

    def __repr__(self):
        """
        Returns a string representation of the Node object.
        Format: "n value r rank p parent_value"
        """
        return f"n {self.value} r {self.rank} p {self.parent.value}"

# Example usage
if __name__ == "__main__":
    # Create a node with the value 5
    node = Node(5)
    
    # Print the node's details
    print(node)  # Expected output: n 5 r 0 p 5
