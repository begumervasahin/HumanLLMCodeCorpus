import sys
from time import time
import pickle
import os

def process_pubtator_file(input_file_path):
    """
    Process the PubTator file to create a dictionary mapping PMIDs to diseases.

    Parameters:
    input_file_path (str): Path to the input PubTator file.

    Returns:
    dict: A dictionary where keys are PMIDs and values are lists of diseases.
    """
    pmid_disease_dict = {}  # Initialize the dictionary to store PMID-disease mappings

    with open(input_file_path, "r") as file:
        lines = file.readlines()

    start_time = time()  # Start the timer to measure processing time

    for line in lines:
        tokens = line.strip().split("\t")  # Remove leading/trailing whitespaces and split the line into tokens

        if len(tokens) == 4:  # Ensure the line has the required number of fields
            pmid, _, diseases_mentioned, _ = tokens
            diseases = diseases_mentioned.split("|")  # Split the diseases_mentioned field into individual diseases

            if pmid in pmid_disease_dict:
                pmid_disease_dict[pmid].extend(diseases)
            else:
                pmid_disease_dict[pmid] = diseases

    print(f"Processing completed in {time() - start_time:.4f} seconds\n")  # Print the time taken to process the file

    return pmid_disease_dict

def save_dictionary(dictionary, file_path):
    """
    Save the dictionary to a file using pickle.

    Parameters:
    dictionary (dict): The dictionary to save.
    file_path (str): The path to the output file.
    """
    os.makedirs(os.path.dirname(file_path), exist_ok=True)  # Ensure the directory exists

    with open(file_path, 'wb') as file:
        pickle.dump(dictionary, file, protocol=pickle.HIGHEST_PROTOCOL)  # Serialize and save the dictionary using pickle

def run():
    """
    Main function to process the PubTator file and save the PMID-disease dictionary.
    """
    if len(sys.argv) != 2:  # Ensure the input file path is provided as a command-line argument
        print("Usage: python script_name.py <input_file_path>")
        sys.exit(1)

    input_file_path = sys.argv[1]  # Get the input file path from the command-line arguments

    pmid_disease_dict = process_pubtator_file(input_file_path)  # Process the PubTator file

    output_file_path = './pubtator_dumps/pmid_disease_dict.pickle'  # Define the path for storing the dictionary

    save_dictionary(pmid_disease_dict, output_file_path)  # Save the dictionary to a file

if __name__ == '__main__':
    run()  # Execute the main function
