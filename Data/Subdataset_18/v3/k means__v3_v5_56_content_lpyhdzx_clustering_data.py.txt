import pandas as pd

class DataLoader:
    """
    Responsible for loading data from a CSV file based on specified target values.

    Attributes:
    target (list): List of target values to filter the data.
    path (str): Path to the CSV file.
    """

    def __init__(self, target=None, path=None):
        self.target = target
        self.path = path

    def load_data(self):
        """
        Load and filter data from the CSV file.

        Returns:
        pd.DataFrame: Filtered data based on the provided target values.
        """
        df = pd.read_csv(self.path)
        filtered_data = df[df['V1'].isin(self.target)].iloc[:, 1:]
        return filtered_data

class ThresholdDataLoader(DataLoader):
    """
    Extends DataLoader by applying thresholding to the data.

    Attributes:
    target (list): List of target values to filter the data.
    path (str): Path to the CSV file.
    """

    def load_data(self):
        """
        Load and filter data, then apply thresholding.

        Returns:
        pd.DataFrame: Filtered and thresholded data based on the provided target values.
        """
        filtered_data = super().load_data()
        filtered_data[filtered_data < 0.8] = -1
        return filtered_data

def main():
    """
    Main function to load and print training and test data.
    """
    # Load and print training data
    train_loader = DataLoader(target=[5, 6], path='zip_train.csv')
    train_data = train_loader.load_data()
    print("Training Data:")
    print(train_data)

    # Load and print test data
    test_loader = DataLoader(target=[5, 6], path='zip_test.csv')
    test_data = test_loader.load_data()
    print("Test Data:")
    print(test_data)

    # Load and print thresholded training data
    threshold_train_loader = ThresholdDataLoader(target=[5, 6], path='zip_train.csv')
    threshold_train_data = threshold_train_loader.load_data()
    print("Thresholded Training Data:")
    print(threshold_train_data)

    # Load and print thresholded test data
    threshold_test_loader = ThresholdDataLoader(target=[5, 6], path='zip_test.csv')
    threshold_test_data = threshold_test_loader.load_data()
    print("Thresholded Test Data:")
    print(threshold_test_data)

if __name__ == '__main__':
    main()
