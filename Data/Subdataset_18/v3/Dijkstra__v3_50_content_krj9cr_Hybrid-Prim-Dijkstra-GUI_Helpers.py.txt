import math
import matplotlib.pyplot as plt
from sympy import cos, sin, pi

class Node:
    def __init__(self, x, y, idx):
        self.x = x
        self.y = y
        self.idx = idx

def generate_points_on_circle(radius, num_points=100):
    """
    Generate evenly distributed points around the circumference of a circle.

    Parameters:
    radius (float): The radius of the circle.
    num_points (int): The number of points to generate.

    Returns:
    list of tuples: Coordinates of points on the circumference.
    """
    return [
        (cos(2 * pi / num_points * i) * radius, 
         sin(2 * pi / num_points * i) * radius)
        for i in range(num_points)
    ]

def calculate_distance(node1, node2):
    """
    Compute the Euclidean distance between two nodes.

    Parameters:
    node1 (Node): The first node.
    node2 (Node): The second node.

    Returns:
    float: The Euclidean distance between node1 and node2.
    """
    return math.sqrt((node1.x - node2.x) ** 2 + (node1.y - node2.y) ** 2)

def create_graph(radius=1.0, num_points=8):
    """
    Generate a graph with nodes positioned on the circumference of a circle.

    Parameters:
    radius (float): The radius of the circle.
    num_points (int): The number of points (nodes) on the circumference.

    Returns:
    tuple: A tuple containing:
        - A list of node coordinates.
        - A list of Node objects.
        - An adjacency matrix representing distances between nodes.
    """
    points = generate_points_on_circle(radius, num_points)
    nodes = [Node(x, y, i) for i, (x, y) in enumerate(points)]
    
    # Create the adjacency matrix for distances
    distances = [
        [0 if i == j else calculate_distance(nodes[i], nodes[j])
         for j in range(num_points)]
        for i in range(num_points)
    ]
    
    return points, nodes, distances

def plot_graph(points, edges, radius=1.0):
    """
    Plot nodes and edges of the graph.

    Parameters:
    points (list of tuples): Node coordinates.
    edges (list of tuples): List of edges represented as tuples of Node objects.
    radius (float): The radius of the circle to set plot limits.
    """
    x_coords, y_coords = zip(*points)

    # Plot nodes
    plt.plot(x_coords, y_coords, 'ro', label='Nodes')

    # Plot edges
    for edge in edges:
        plt.plot(
            [edge[0].x, edge[1].x], 
            [edge[0].y, edge[1].y], 
            'b-', lw=1
        )

    # Set plot limits and aspect ratio
    plt.axis([-radius * 1.5, radius * 1.5, -radius * 1.5, radius * 1.5])
    plt.grid(True)
    plt.gca().set_aspect('equal', 'datalim')

    # Add a legend
    plt.legend()

    # Display the plot
    plt.show()

# Example usage
if __name__ == "__main__":
    # Generate graph data
    points, nodes, _ = create_graph(radius=1.0, num_points=8)

    # Convert the distance matrix to an edge list
    edge_list = [
        (nodes[i], nodes[j])
        for i in range(len(nodes))
        for j in range(i + 1, len(nodes))
    ]

    # Plot the graph
    plot_graph(points, edge_list, radius=1.0)
