import random
import math

def generate_keys(p, q):
    """Generate RSA public and private keys based on two prime numbers."""
    n = p * q
    phi_n = (p - 1) * (q - 1)
    
    e = choose_e(phi_n, n)
    d = find_d(e, phi_n)
    
    return n, e, d

def choose_e(phi_n, n):
    """Choose an encryption exponent e that is coprime with phi_n and n."""
    relative_primes = [e for e in range(2, phi_n) if math.gcd(e, phi_n) == 1 and math.gcd(e, n) == 1]
    return random.choice(relative_primes)

def find_d(e, phi_n):
    """Find the decryption exponent d that satisfies (d * e) % phi_n == 1."""
    for d in range(1, phi_n):
        if (d * e) % phi_n == 1:
            return d
    return None
