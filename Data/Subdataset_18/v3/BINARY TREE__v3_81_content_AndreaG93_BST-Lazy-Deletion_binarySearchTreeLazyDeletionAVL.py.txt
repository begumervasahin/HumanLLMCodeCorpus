class AVLNode:
    """
    Class to represent a node in an AVL tree with lazy deletion.
    """
    def __init__(self, key, value, parent=None):
        self.key = key
        self.value = value
        self.left = None
        self.right = None
        self.parent = parent
        self.is_valid = True
        self.height = 0
        self.is_left_son = None
    
    def has_right_son(self):
        return self.right is not None
    
    def has_left_son(self):
        return self.left is not None
    
    def has_only_one_son(self):
        return (self.left is None) != (self.right is None)

class AVLTree:
    """
    Class to represent an AVL tree.
    """
    def __init__(self, root_node=None):
        self.root_node = root_node
    
    def update_node_heights(self, node):
        """
        Update the heights of the nodes along the tree.
        """
        while node:
            left_height = node.left.height if node.left else -1
            right_height = node.right.height if node.right else -1
            node.height = max(left_height, right_height) + 1
            node = node.parent
    
    def delete_node(self, node):
        """
        Delete a node from the tree.
        """
        if node.is_left_son:
            node.parent.left = None
        else:
            node.parent.right = None
        self.update_node_heights(node.parent)

class AVLTreeWithLazyDeletion(AVLTree):
    """
    Class to represent an AVL tree with lazy deletion.
    """
    def __init__(self, root_node=None):
        """
        Constructs a newly allocated 'AVLTreeWithLazyDeletion' object.
        """
        super().__init__(root_node)

    def insert(self, key, value):
        """
        Insert a new (key, value) pair into the tree.
        """
        if not self.root_node:
            self.root_node = AVLNode(key, value)
            return
        
        current_node = self.root_node
        parent_node = None
        insert_to_left = True
        
        while current_node:
            parent_node = current_node
            if current_node.is_valid:
                if key <= current_node.key:
                    current_node = current_node.left
                    insert_to_left = True
                else:
                    current_node = current_node.right
                    insert_to_left = False
            else:
                if current_node.has_right_son() and key > current_node.right.key:
                    current_node = current_node.right
                    insert_to_left = False
                elif current_node.has_left_son() and key < current_node.left.key:
                    current_node = current_node.left
                    insert_to_left = True
                else:
                    current_node.is_valid = True
                    current_node.key = key
                    current_node.value = value
                    return
        
        new_node = AVLNode(key, value, parent=parent_node)
        if insert_to_left:
            parent_node.left = new_node
            new_node.is_left_son = True
        else:
            parent_node.right = new_node
            new_node.is_left_son = False
        
        if parent_node.has_only_one_son():
            self.update_node_heights(new_node)

    def delete(self, key):
        """
        Delete the first occurrence of a node with the specified key from the tree.
        """
        current_node = self.root_node
        while current_node:
            if key < current_node.key:
                current_node = current_node.left
            elif key > current_node.key:
                current_node = current_node.right
            else:
                if current_node.is_valid:
                    current_node.is_valid = False
                    return True
                current_node = current_node.left
        return False

    def search(self, key, allow_restructuring=True):
        """
        Search for the first occurrence of a node with the specified key in the tree.
        """
        invalid_node_list = []
        current_node = self.root_node
        
        while current_node:
            if allow_restructuring and not current_node.is_valid:
                invalid_node_list.append(current_node)
            if key < current_node.key:
                current_node = current_node.left
            elif key > current_node.key:
                current_node = current_node.right
            else:
                if current_node.is_valid:
                    break
                current_node = current_node.left
        
        for item in invalid_node_list:
            self.delete_node(item)
        
        return current_node
