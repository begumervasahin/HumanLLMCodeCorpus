import socket
from Crypto.Cipher import AES
from binascii import b2a_hex, a2b_hex

def quick_pow_mod(base, exponent, modulus):
    """Perform quick exponentiation with modulus."""
    result = 1
    while exponent:
        if exponent & 1:
            result = (result * base) % modulus
        exponent >>= 1
        base = (base * base) % modulus
    return result

def bin_to_str(binary):
    """Convert a binary string to a character string."""
    return ''.join(chr(int(binary[i:i+8], 2)) for i in range(0, len(binary), 8))

def brute_force_aes_key(sock, addr, C, e, N):
    """Brute force the AES key bit by bit."""
    current_known = ""
    
    for b in range(127, -1, -1):
        print("*" * 60)
        print(f"b = {b}")
        
        # Calculate C_b
        C_b = (C * quick_pow_mod(2, b * e, N)) % N
        
        # Form the candidate AES key
        aes_key_candidate = "0" + current_known + "0" * b
        print(f"Trying key: {aes_key_candidate}")
        
        # Convert the binary string to character string
        aes_key_candidate_str = bin_to_str(aes_key_candidate)
        
        # Encrypt the test string using the candidate AES key
        crypto = AES.new(aes_key_candidate_str.encode('utf-8'), AES.MODE_ECB)
        encrypted_wup = crypto.encrypt(b"WUP0WUP1WUP2WUP3")
        
        # Send the candidate C_b and encrypted WUP to the server
        sock.sendto(str(C_b).encode("utf-8"), addr)
        sock.sendto(b2a_hex(encrypted_wup), addr)
        
        # Receive the server's response
        data_msg, _ = sock.recvfrom(1024)
        response = data_msg.decode("utf-8")
        
        # Update the known AES key bits
        if response == "valid":
            current_known = "0" + current_known
        else:
            current_known = "1" + current_known
        
        print(f"Current known AES key: {current_known}")
    
    return bin_to_str(current_known)

def main():
    # Server details
    host = "192.168.146.128"
    port = 25535
    addr = (host, port)
    
    # Initialize socket
    sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    
    # Initial message exchange
    while True:
        msg = input("Shall we start?\n")
        sock.sendto(msg.encode("utf-8"), addr)
        if msg == "Let's get it started":
            break
    
    # RSA public exponent
    e = 65537
    
    # Receive data from server
    data_N, _ = sock.recvfrom(1024)
    data_encrypted_aes_key, _ = sock.recvfrom(1024)
    data_history_encrypted_wup, _ = sock.recvfrom(1024)
    
    # Convert received data
    N = int(data_N.decode("utf-8"))
    C = int(data_encrypted_aes_key.decode("utf-8"))
    
    print(f"I got N: {N}")
    print(f"I got C: {C}")
    
    # Brute force the AES key
    aes_key = brute_force_aes_key(sock, addr, C, e, N)
    print(f"I guess the AES key is: {aes_key}")
    
    # Decrypt the historical encrypted WUP using the guessed AES key
    crypto_aes_key = AES.new(aes_key.encode('utf-8'), AES.MODE_ECB)
    history_decrypted_wup = crypto_aes_key.decrypt(a2b_hex(data_history_encrypted_wup.decode("utf-8"))).decode("utf-8")
    
    print(f"Decrypted history WUP: {history_decrypted_wup}")

if __name__ == "__main__":
    main()
