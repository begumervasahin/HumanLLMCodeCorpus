def insertion_sort(arr):
    """
    Sorts a list of integers using the insertion sort algorithm.

    Args:
        arr (list of int): The list of integers to be sorted.
    """
    for i in range(1, len(arr)):
        current_value = arr[i]
        position = i - 1

        # Shift elements of the sorted portion of the list to the right
        while position >= 0 and arr[position] > current_value:
            arr[position + 1] = arr[position]
            position -= 1

        # Insert the current value into its correct position
        arr[position + 1] = current_value

def read_data_from_file(filename):
    """
    Reads unsorted lists of integers from a file and returns them as a list of lists.

    Args:
        filename (str): The name of the file to read from.

    Returns:
        list of list of int: A list where each element is a list of integers from the file.
    """
    lists_of_numbers = []
    
    with open(filename, "r") as file:
        for line in file:
            # Convert the line into a list of integers, skipping the first value (assumed placeholder)
            numbers = [int(num) for num in line.split()[1:]]
            lists_of_numbers.append(numbers)
    
    return lists_of_numbers

def write_data_to_file(filename, data):
    """
    Writes a list of lists of integers to a file, with each inner list on a new line.

    Args:
        filename (str): The name of the file to write to.
        data (list of list of int): The sorted lists of integers to write.
    """
    with open(filename, "w") as file:
        for number_list in data:
            file.write(" ".join(map(str, number_list)) + "\n")

def main():
    """
    Main function that orchestrates reading, sorting, and writing data.
    """
    # Step 1: Read unsorted lists from 'data.txt'
    unsorted_data = read_data_from_file("data.txt")
    
    # Step 2: Sort each list using the insertion sort algorithm
    for numbers in unsorted_data:
        insertion_sort(numbers)
    
    # Step 3: Write the sorted lists to 'insertionsort.txt'
    write_data_to_file("insertionsort.txt", unsorted_data)

if __name__ == "__main__":
    main()
