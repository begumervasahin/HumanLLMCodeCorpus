def longest_subsequence_Onlogn(arr):
    """
    Find the longest increasing subsequence in O(n log n) time.
    
    Parameters:
    arr (list): A list of integers.
    
    Returns:
    list: The longest increasing subsequence.
    """
    if len(arr) < 2:
        return arr
    
    # M will store the index of the smallest possible end element for subsequences of different lengths
    M = [None] * len(arr)
    # P will store the predecessor of each element in the longest increasing subsequence ending at that element
    P = [None] * len(arr)
    L = 1
    M[0] = 0

    for i in range(1, len(arr)):
        if arr[i] > arr[M[L-1]]:
            j = L
        else:
            lo, hi = 0, L - 1
            while lo < hi:
                mid = (lo + hi) // 2
                if arr[i] > arr[M[mid]]:
                    lo = mid + 1
                else:
                    hi = mid
            j = lo
        
        P[i] = M[j-1] if j > 0 else None
        M[j] = i
        if j + 1 > L:
            L = j + 1
    
    # Reconstruct the longest increasing subsequence
    k = M[L-1]
    results = []
    for _ in range(L):
        results.append(arr[k])
        k = P[k]
    
    return results[::-1]

def longest_subsequence_On2(arr):
    """
    Find the longest increasing subsequence in O(n^2) time.
    
    Parameters:
    arr (list): A list of integers.
    
    Returns:
    list: The longest increasing subsequence.
    """
    if not arr:
        return []

    # mem[i] will store the longest increasing subsequence that ends with arr[i]
    mem = [[] for _ in range(len(arr))]
    best = []

    for i in range(len(arr)):
        for j in range(i):
            if arr[i] > arr[j] and len(mem[j]) + 1 > len(mem[i]):
                mem[i] = mem[j] + [arr[i]]
        if not mem[i]:
            mem[i].append(arr[i])
        if len(mem[i]) > len(best):
            best = mem[i]
    
    return best

    arr1 = [30, 10, 20, 50, 40, 60]
    print("Longest increasing subsequence using O(n log n) approach:")
    print(longest_subsequence_Onlogn(arr1))

    print("\nLongest increasing subsequence using O(n^2) approach:")
    print(longest_subsequence_On2(arr2))
