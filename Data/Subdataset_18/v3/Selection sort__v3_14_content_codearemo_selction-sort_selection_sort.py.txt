def selection_sort(items):
    """
    Sorts the input list using the selection sort algorithm.

    Args:
        items (list): The list of integers to be sorted.

    Returns:
        list: The sorted list.
    """
    n = len(items)
    for step in range(n):
        min_idx = step
        for location in range(step + 1, n):
            if items[location] < items[min_idx]:
                min_idx = location
        # Swap the found minimum element with the first element
        items[step], items[min_idx] = items[min_idx], items[step]

    return items

def get_user_input():
    """
    Prompts the user to enter a list of numbers to be sorted.

    Returns:
        list: The list of integers entered by the user.
    """
    try:
        user_input = input("Enter the numbers to be sorted, separated by spaces: ")
        number_list = [int(x) for x in user_input.split()]
        return number_list
    except ValueError:
        print("Invalid input. Please enter a list of integers separated by spaces.")
        return []

def main():
    """
    Main function to handle user input and sort the list using selection sort.
    """
    number_list = get_user_input()
    if number_list:
        sorted_list = selection_sort(number_list)
        print("Sorted list:", sorted_list)
        print("Total number of items:", len(sorted_list))

if __name__ == "__main__":
    main()
