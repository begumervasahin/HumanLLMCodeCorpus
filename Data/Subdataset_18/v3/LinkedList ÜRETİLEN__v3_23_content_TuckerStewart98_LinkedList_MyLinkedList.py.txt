class Node:
    def __init__(self, data, next_node=None):
        """Initialize the node with data and the reference to the next node."""
        self.__data = data
        self.__next = next_node

    def get_data(self):
        """Get the data stored in the node."""
        return self.__data

    def get_next(self):
        """Get the reference to the next node."""
        return self.__next

    def set_next(self, next_node):
        """Set the reference to the next node."""
        self.__next = next_node

    def __str__(self):
        """String representation of the node."""
        return str(self.__data)


class MyLinkedList:
    def __init__(self):
        """Initialize the linked list with head, tail, and size."""
        self.__head = None
        self.__tail = None
        self.__size = 0

    def size(self):
        """Get the size of the linked list."""
        return self.__size

    def add(self, data):
        """Add a node with the given data to the end of the linked list."""
        new_node = Node(data)
        if self.__head is None:
            self.__head = new_node
            self.__tail = new_node
        else:
            self.__tail.set_next(new_node)
            self.__tail = new_node
        self.__size += 1

    def get(self, index):
        """Get the data of the node at the specified index."""
        if index < 0 or index >= self.__size:
            raise IndexError('Index out of bounds')
        current_node = self.__head
        for _ in range(index):
            current_node = current_node.get_next()
        return current_node.get_data()

    def __str__(self):
        """String representation of the linked list."""
        current_node = self.__head
        elements = []
        while current_node:
            elements.append(str(current_node.get_data()))
            current_node = current_node.get_next()
        return " -> ".join(elements)


# Example usage
if __name__ == "__main__":
    linked_list = MyLinkedList()
    linked_list.add(1)
    linked_list.add(5)
    linked_list.add(-7)

    for i in range(linked_list.size()):
        print(linked_list.get(i))

    print("Linked List:", linked_list)
