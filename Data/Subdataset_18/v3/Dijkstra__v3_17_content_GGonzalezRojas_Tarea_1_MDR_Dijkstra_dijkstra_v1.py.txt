from collections import defaultdict
from heapq import heappop, heappush

def dijkstra(graph, start, end):
    """
    Compute the shortest path from start to end using Dijkstra's algorithm.

    Args:
    - graph (list of tuples): Each tuple represents an edge with (start, end, weight).
    - start (str): The starting node.
    - end (str): The destination node.

    Returns:
    - tuple: A tuple containing the total cost and the path as a linked list.
    """
    adjacency_list = defaultdict(list)
    
    # Build the adjacency list from the list of edges
    for src, dest, weight in graph:
        adjacency_list[src].append((weight, dest))
    
    # Priority queue to store (cost, vertex, path)
    priority_queue = [(0, start, ())]  # Start with the source node
    visited = set()
    
    while priority_queue:
        # Get the node with the lowest cost
        cost, current_node, path = heappop(priority_queue)
        
        if current_node not in visited:
            visited.add(current_node)
            path = (current_node, path)  # Append the current node to the path
            
            # If we have reached the destination node
            if current_node == end:
                return cost, path
            
            # Add neighboring nodes to the priority queue
            for weight, neighbor in adjacency_list[current_node]:
                if neighbor not in visited:
                    heappush(priority_queue, (cost + weight, neighbor, path))
    
    return float("inf"), ()  # Return infinity and empty path if there's no path

def print_result(result):
    """
    Helper function to print the result.

    Args:
    - result (tuple): A tuple containing the total cost and the path as a linked list.
    """
    cost, path = result
    node_list = []
    
    # Reconstruct the path
    while path:
        node, path = path
        node_list.append(node)
    
    print(f"Cost: {cost}")
    print("Path:", " -> ".join(reversed(node_list)))
