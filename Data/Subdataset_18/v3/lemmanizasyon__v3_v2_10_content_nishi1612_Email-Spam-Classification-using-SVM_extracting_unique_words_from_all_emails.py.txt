import os
import time
import string
import operator
from nltk.corpus import stopwords
from nltk.stem.wordnet import WordNetLemmatizer

def text_cleanup(text):
    """
    Cleans up the input text by removing punctuation, stopwords, and converting to lowercase.

    Args:
        text (str): The text to clean up.

    Returns:
        list: A list of cleaned words.
    """
    # Remove punctuation
    text_without_punctuation = ''.join(char for char in text if char not in string.punctuation)

    # Remove stopwords
    text_without_stopwords = ' '.join(
        word for word in text_without_punctuation.split() if word.lower() not in stopwords.words('english')
    )

    # Convert to lowercase and split into words
    cleaned_text = [word.lower() for word in text_without_stopwords.split()]
    return cleaned_text

def process_emails(directory_path, output_file_path, lemmatizer):
    """
    Processes email files in a directory, counts word occurrences, and writes the results to a CSV file.

    Args:
        directory_path (str): The path to the directory containing email files.
        output_file_path (str): The path to the output CSV file.
        lemmatizer (WordNetLemmatizer): The lemmatizer to use for word lemmatization.
    """
    start_time = time.time()
    word_count = {}
    processed_files_count = 0

    for file_name in os.listdir(directory_path):
        file_path = os.path.join(directory_path, file_name)

        with open(file_path, "r", encoding='utf-8', errors='ignore') as file_reading:
            words = text_cleanup(file_reading.read())
            for word in words:
                if not (word.isdigit() or len(word) <= 2):
                    # Lemmatize words
                    lemmatized_word = lemmatizer.lemmatize(word)
                    word_count[lemmatized_word] = word_count.get(lemmatized_word, 0) + 1

        processed_files_count += 1
        if processed_files_count % 100 == 0:
            print(f"Processed {processed_files_count} files")

    # Sort the word count dictionary by word frequency in descending order
    sorted_word_count = sorted(word_count.items(), key=operator.itemgetter(1), reverse=True)

    # Write the word counts to a CSV file
    write_word_count_to_csv(output_file_path, sorted_word_count)

    processing_time = round(time.time() - start_time, 2)
    print(f'Time taken to preprocess emails: {processing_time} seconds')

def write_word_count_to_csv(output_file_path, word_count):
    """
    Writes the word counts to a CSV file.

    Args:
        output_file_path (str): The path to the output CSV file.
        word_count (dict): The dictionary containing word counts.
    """
    with open(output_file_path, "w+", encoding='utf-8') as output_file:
        output_file.write('word,count\n')
        for word, count in word_count:
            if count < 100:
                break
            output_file.write(f"{word},{count}\n")

    # Path to the directory containing email files
    directory_path = "emails/"
    
    # Path to the output CSV file
    output_file_path = "wordslist.csv"
    
    # Initialize the lemmatizer
