import os
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import statsmodels.api as sm
import seaborn as sb
from statsmodels.tsa.stattools import acf, pacf
from statsmodels.tsa.seasonal import seasonal_decompose

# Set the style for seaborn
sb.set_style('darkgrid')

# Define the path to the data file
data_path = os.path.join(os.getcwd(), 'q_table.csv')

# Load the data into a pandas DataFrame
stock_data = pd.read_csv(data_path)

# Convert the 'Date' column to datetime format and sort by date
stock_data['Date'] = pd.to_datetime(stock_data['Date'])
stock_data.sort_values(by='Date', inplace=True)
stock_data.set_index('Date', inplace=True)

# Calculate additional columns for analysis
stock_data['First Difference'] = stock_data['Close'].diff()
stock_data['Natural Log'] = np.log(stock_data['Close'])
stock_data['Original Variance'] = stock_data['Close'].rolling(window=30, center=True).var()
stock_data['Log Variance'] = stock_data['Natural Log'].rolling(window=30, center=True).var()
stock_data['Logged First Difference'] = stock_data['Natural Log'].diff()
stock_data['Lag 20'] = stock_data['Logged First Difference'].shift(20)

# Calculate autocorrelation and partial autocorrelation
logged_first_diff = stock_data['Logged First Difference'].dropna()
lag_corr = acf(logged_first_diff)
lag_partial_corr = pacf(logged_first_diff)

# Decompose the time series
decomposition = seasonal_decompose(stock_data['Natural Log'], model='additive', period=30)

# Fit the ARIMA model
arima_order = (0, 0, 1)
arima_model = sm.tsa.ARIMA(logged_first_diff, order=arima_order)
arima_results = arima_model.fit(disp=-1)

# Add the forecasted values to the DataFrame
stock_data['Forecast'] = arima_results.fittedvalues

# Plot the original and forecasted data
plt.figure(figsize=(16, 12))
stock_data[['Logged First Difference', 'Forecast']].plot()
plt.title('Logged First Difference and ARIMA Forecast')
plt.show()
