import glob

def find_files(pattern, description):
    """Find and print files or directories based on the specified pattern and description."""
    matches = glob.glob(pattern, recursive=True)
    print(f"{description}: {matches}")

def find_files_and_directories():
    """Find and print files and directories based on specific patterns."""
    # Find files in the current directory that start with a digit followed by any character(s)
    find_files('./[0-9].*', "Files starting with a digit followed by any character(s)")
    
    # Find all .gif files in the current directory
    find_files('*.gif', "All .gif files in the current directory")
    
    # Find all single-character .gif files in the current directory
    find_files('?.gif', "Single-character .gif files in the current directory")
    
    # Recursively find all .txt files in the current directory and subdirectories
    find_files('**/*.txt', "All .txt files in the current directory and subdirectories")
    
    # Recursively find all directories in the current directory and subdirectories
    find_files('./**/', "All directories in the current directory and subdirectories")

if __name__ == "__main__":
    find_files_and_directories()
