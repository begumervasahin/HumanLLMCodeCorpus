import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler

def perform_pca(df, n_components=150):
    """
    Perform Principal Component Analysis (PCA) on the given DataFrame.
    
    Parameters:
    - df: DataFrame to perform PCA on
    - n_components: Number of principal components to retain
    
    Returns:
    - df_out: DataFrame containing the reduced data
    - top_eigenvectors: The top eigenvectors used for the reduction
    - column_names: Names of the principal components
    """
    # Transpose the DataFrame and extract labels
    df_transposed = df.transpose()
    labels = df_transposed.iloc[:, -1].values

    # Standardize the data
    data_matrix = StandardScaler().fit_transform(df_transposed.iloc[:, :-1])

    # Calculate the covariance matrix
    covariance_matrix = np.cov(data_matrix.T)

    # Calculate eigenvalues and eigenvectors
    eigenvalues, eigenvectors = np.linalg.eig(covariance_matrix)

    # Sort eigenvalues and eigenvectors in descending order
    eigens = sorted([(np.abs(eigenvalues[i]), eigenvectors[:, i]) for i in range(len(eigenvalues))], 
                    key=lambda x: x[0], reverse=True)

    # Calculate the cumulative sum of eigenvalues
    eigen_total = sum(eigenvalues)
    cumulative_sum = np.cumsum([eigen[0] for eigen in eigens]) / eigen_total

    # Plot the cumulative sum of eigenvalues
    plt.plot(cumulative_sum, marker='o')
    plt.xlabel("Number of Components")
    plt.ylabel("Cumulative Sum of Eigenvalues / Total Eigenvalue")
    plt.title("PCA Cumulative Sum of Eigenvalues")
    plt.show()

    # Select the top n_components eigenvectors
    top_eigenvectors = np.array([eigens[i][1] for i in range(n_components)]).T

    # Reduce the data matrix
    reduced_data = data_matrix.dot(top_eigenvectors)
    reduced_data = np.real(reduced_data)

    # Create the output DataFrame
    column_names = [f"PC{i+1}" for i in range(n_components)]
    df_out = pd.DataFrame(reduced_data, columns=column_names)
    df_out['label'] = labels

    return df_out, top_eigenvectors, column_names

def apply_pca(df, model, column_names):
    """
    Apply the PCA model to a test DataFrame.
    
    Parameters:
    - df: Test DataFrame to transform
    - model: Eigenvectors used for the PCA transformation
    - column_names: Names of the principal components
    
    Returns:
    - df_out: DataFrame containing the reduced test data
    """
    # Transpose the DataFrame and extract labels
    df_transposed = df.transpose()
    labels = df_transposed.iloc[:, -1].values

    # Standardize the data
    data_matrix = StandardScaler().fit_transform(df_transposed.iloc[:, :-1])

    # Reduce the data matrix using the PCA model
    reduced_data = data_matrix.dot(model)
    reduced_data = np.real(reduced_data)

    # Create the output DataFrame
    df_out = pd.DataFrame(reduced_data, columns=column_names)
    df_out['label'] = labels

    return df_out

if __name__ == '__main__':
    # Load training data
    train_df = pd.read_csv('train.csv')

    # Perform PCA on the training data
    pca_train_df, pca_model, pca_column_names = perform_pca(train_df)

    # Save the reduced training data
    pca_train_df.to_csv("pca_out.csv", index=False)

    # Load test data
    test_df = pd.read_csv('test.csv')

    # Apply PCA model to the test data
    pca_test_df = apply_pca(test_df, pca_model, pca_column_names)

    # Save the reduced test data
    pca_test_df.to_csv("pca_test_out.csv", index=False)
