def quick_sort(arr, left, right):
    """
    Sorts the list 'arr' using the Quick Sort algorithm.

    Args:
    arr (list): The list to be sorted.
    left (int): The starting index of the segment to be sorted.
    right (int): The ending index of the segment to be sorted.
    """
    if left < right:
        pivot_index = partition(arr, left, right)
        quick_sort(arr, left, pivot_index)  # Recursively sort the left partition
        quick_sort(arr, pivot_index + 1, right)  # Recursively sort the right partition

def partition(arr, left, right):
    """
    Partition the array around the pivot and return the partition index.

    Args:
    arr (list): The list to be partitioned.
    left (int): The starting index of the segment to be partitioned.
    right (int): The ending index of the segment to be partitioned.

    Returns:
    int: The index of the partition.
    """
    pivot = arr[left]  # Choose the pivot element
    s = left  # Initialize the partitioning index

    for i in range(left + 1, right):
        if arr[i] < pivot:
            s += 1
            arr[s], arr[i] = arr[i], arr[s]  # Swap elements

    arr[left], arr[s] = arr[s], arr[left]  # Place the pivot in the correct position
    return s

def main():
    """
    Main function to demonstrate the Quick Sort algorithm.
    """
    # Example usage
    array = [21, 4, 1, 3, 9, 20, 25, 6, 21, 14]
    print("Original array:", array)

    quick_sort(array, 0, len(array))  # Perform quick sort on the array

    print("Sorted array:", array)

if __name__ == "__main__":
    main()
