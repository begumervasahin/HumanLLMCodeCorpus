import numpy as np
import statsmodels.api as sm
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d import Axes3D
from sklearn import metrics
import scipy.optimize

def get_power_combinations(dim, order):
    """Generate power combinations for polynomial terms."""
    p = (order + 1) ** dim
    combinations = [int(np.base_repr(i, order + 1)) for i in range(p)]
    formatted_combinations = ["{:02d}".format(int(v)) for v in combinations]
    
    power_combinations = []
    for combo in formatted_combinations:
        power_combinations.append([int(digit) for digit in list(combo)])
    
    return power_combinations

def parse_line_to_float_list(line):
    """Convert a line of text into a list of floats."""
    return list(map(float, line.strip().split()))

def read_input_file(file_name):
    """Extract parameters from the input file."""
    with open(file_name, 'r') as file:
        A = parse_line_to_float_list(file.readline())
        B = parse_line_to_float_list(file.readline())
        Y = [parse_line_to_float_list(line) for line in file]
    
    A, B = np.array(A), np.array(B)
    a1, a2 = A, B
    A, B = np.meshgrid(A, B)
    A, B = A.flatten(), B.flatten()
    Y = np.array(Y).T.flatten()
    
    return A, B, a1, a2, Y

def fit_polynomial_regression(A, B, Y, order):
    """Perform polynomial regression and return fitted parameters."""
    dim = 2
    num_samples = len(Y)
    num_terms = (order + 1) ** dim
    X = np.ones((num_samples, 1))
    power_combinations = get_power_combinations(dim, order)
    
    for j in range(1, num_terms):
        power_A, power_B = power_combinations[j]
        X = np.column_stack((X, (A**power_A) * (B**power_B)))
    
    model = sm.OLS(Y, X)
    results = model.fit()
    
    return results, power_combinations

def evaluate_polynomial(x, params, power_combinations):
    """Compute polynomial function value."""
    terms = [(x[0]**power_combinations[i][0]) * (x[1]**power_combinations[i][1]) for i in range(len(params))]
    return np.dot(terms, params)

def plot_3d_surface(A, B, Y, polynomial_func):
    """Visualize data points and fitted polynomial surface."""
    fig = plt.figure()
    ax = fig.add_subplot(111, projection='3d')
    ax.scatter(A, B, Y)
    
    x1 = np.linspace(A.min(), A.max(), 50)
    x2 = np.linspace(B.min(), B.max(), 50)
    x1, x2 = np.meshgrid(x1, x2)
    y = np.array([[polynomial_func([x1i, x2i]) for x1i in x1] for x2i in x2])
    
    ax.plot_surface(x1, x2, y, cmap='viridis', alpha=0.5)
    plt.show()

def main():
    print('Reading input file')
    file_name = input('Please input file name:\n')
    
    print('Initializing inputs')
    A, B, a1, a2, Y = read_input_file(file_name)
    
    print('Please input the order for polynomial regression:')
    order = int(input())
    
    results, power_combinations = fit_polynomial_regression(A, B, Y, order)
    params = results.params
    
    print("\n========")
    print("Parameters:")
    for i, param in enumerate(params):
        power_A, power_B = power_combinations[i]
        print(f"Beta{i} (x{power_A}y{power_B}) = {param}")
    print(results.summary())
    
    polynomial_func = lambda x: evaluate_polynomial(x, params, power_combinations)
    
    plot_3d_surface(A, B, Y, polynomial_func)
    
    Y_predicted = np.array([polynomial_func([a, b]) for a, b in zip(A, B)])
    print("R squared score: ", metrics.r2_score(Y, Y_predicted))
    
    initial_guess = np.array([A[0], B[0]])
    bounds = [(A.min(), A.max()), (B.min(), B.max())]
    result = scipy.optimize.minimize(polynomial_func, initial_guess, method='L-BFGS-B', bounds=bounds)
    x_opt, y_opt = result.x
    print("Optima point:")
    print("1st variable = ", x_opt, "; 2nd variable =", y_opt)
    
    fig = plt.figure()
    ax = fig.add_subplot(111, projection='3d')
    ax.scatter(A, B, Y)
    ax.scatter(x_opt, y_opt, polynomial_func([x_opt, y_opt]), color='r', s=100)
    plt.show()

if __name__ == '__main__':
    main()
