def insertion_sort(array):
    """
    Sorts an array in ascending order using the insertion sort algorithm.
    
    :param array: List of elements to be sorted.
    :return: Sorted list.
    """
    for i in range(1, len(array)):
        current_value = array[i]
        position = i
        
        # Shift elements of the sorted portion to the right to create a position for current_value
        while position > 0 and array[position - 1] > current_value:
            array[position] = array[position - 1]
            position -= 1
        
        # Place the current value in its correct position
        array[position] = current_value
    
    return array

def merge_sort(array):
    """
    Sorts an array in ascending order using the merge sort algorithm.
    
    :param array: List of elements to be sorted.
    :return: Sorted list.
    """
    # Base case: A list of zero or one elements is already sorted
    if len(array) <= 1:
        return array
    
    # Recursive case: Split the list into halves and sort each half
    mid = len(array) // 2
    left_half = merge_sort(array[:mid])
    right_half = merge_sort(array[mid:])
    
    # Merge the sorted halves
    return merge(left_half, right_half)

def merge(left, right):
    """
    Merges two sorted lists into one sorted list.
    
    :param left: Sorted left half.
    :param right: Sorted right half.
    :return: Merged and sorted list.
    """
    merged_array = []
    i = j = 0
    
    # Merge elements from both lists in sorted order
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            merged_array.append(left[i])
            i += 1
        else:
            merged_array.append(right[j])
            j += 1
    
    # Append any remaining elements from both lists
    merged_array.extend(left[i:])
    merged_array.extend(right[j:])
    
    return merged_array
