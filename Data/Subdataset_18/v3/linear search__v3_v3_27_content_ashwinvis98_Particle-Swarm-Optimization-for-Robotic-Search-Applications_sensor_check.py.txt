import cv2
import cv2.aruco as aruco
from nxt.motor import *
from nxt.sensor import *
import tty
import sys
import termios
import nxt.locator

def get_single_char():
    """
    Function to get a single character from the keyboard input.
    Temporarily sets the terminal to raw mode to read a single character.
    """
    fd = sys.stdin.fileno()  # Get the file descriptor for standard input
    old_settings = termios.tcgetattr(fd)  # Save the current terminal settings
    try:
        tty.setraw(fd)  # Set the terminal to raw mode
        char = sys.stdin.read(1)  # Read a single character
    finally:
        termios.tcsetattr(fd, termios.TCSADRAIN, old_settings)  # Restore terminal settings
    return char

def initialize_robot():
    """
    Function to initialize the LEGO Mindstorms robot by connecting to the brick
    and setting up motors and sensors.
    """
    # Connect to the LEGO Mindstorms brick using its Bluetooth address
    brick = nxt.locator.find_one_brick(host='00:16:53:0F:0F:D8')

    # Initialize motors
    left_motor = Motor(brick, PORT_C)  # Left motor connected to port C
    right_motor = Motor(brick, PORT_A)  # Right motor connected to port A

    # Initialize light sensor
    light_sensor = Light(brick, PORT_1, False)  # Light sensor connected to port 1

    # Initialize synchronized motors for movement
    both_motors = SynchronizedMotors(left_motor, right_motor, 0)  # Synchronized motors for forward/backward
    right_turn_motors = SynchronizedMotors(left_motor, right_motor, 50)  # Synchronized motors for right turn
    left_turn_motors = SynchronizedMotors(right_motor, left_motor, 50)  # Synchronized motors for left turn

    print("Robot is ready to move.")
    print("Connected to:", brick)

    return brick, left_motor, right_motor, light_sensor, both_motors, right_turn_motors, left_turn_motors

def control_robot():
    """
    Main function to control the LEGO Mindstorms robot based on user input.
    The robot moves forward, backward, turns left, or turns right depending on the input.
    """
    brick, left_motor, right_motor, light_sensor, both_motors, right_turn_motors, left_turn_motors = initialize_robot()

    user_input = ' '  # Initialize user input variable
    while user_input != 'q':  # Loop until 'q' is pressed
        user_input = get_single_char()  # Get user input

        # Read light sensor value
        light_value = light_sensor.get_input_values().scaled_value
        print("Light sensor value:", light_value)

        # Perform actions based on user input
        if user_input == 'w':
            both_motors.turn(75, 360, True)  # Move forward
        elif user_input == 's':
            both_motors.turn(-75, 90, True)  # Move backward
        elif user_input == 'a':
            left_turn_motors.turn(75, 90, True)  # Turn left
        elif user_input == 'd':
            right_turn_motors.turn(75, 90, True)  # Turn right

    print("Movement aborted.")

# Entry point of the script
if __name__ == "__main__":
    control_robot()
