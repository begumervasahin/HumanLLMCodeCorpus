import numpy as np
from os import listdir

def knn_classifier(unlabeled_sample, dataset, labels, k):
    """Classify the unlabeled sample using the k-Nearest Neighbors algorithm."""
    num_samples = dataset.shape[0]
    diff = np.tile(unlabeled_sample, (num_samples, 1)) - dataset
    distances = np.sqrt(np.sum(diff ** 2, axis=1))
    sorted_dist_indices = np.argsort(distances)
    
    class_count = {}
    for i in range(k):
        vote_label = labels[sorted_dist_indices[i]]
        class_count[vote_label] = class_count.get(vote_label, 0) + 1
    
    sorted_class_count = sorted(class_count.items(), key=lambda item: item[1], reverse=True)
    return sorted_class_count[0][0]

def image_to_vector(file_name):
    """Convert a 32x32 image to a 1x1024 vector."""
    vector = np.zeros((1, 1024))
    with open(file_name) as img_file:
        for i in range(32):
            line = img_file.readline()
            for j in range(32):
                vector[0, 32 * i + j] = int(line[j])
    return vector

def load_dataset(directory):
    """Load the dataset from the given directory and return the data matrix and labels."""
    labels = []
    file_list = listdir(directory)
    num_samples = len(file_list)
    data_matrix = np.zeros((num_samples, 1024))

    for i, file_name in enumerate(file_list):
        class_label = int(file_name.split('_')[0])
        labels.append(class_label)
        data_matrix[i, :] = image_to_vector(f'{directory}/{file_name}')
    
    return data_matrix, labels

def handwriting_recognition_test():
    """Test the k-Nearest Neighbors classifier on the handwriting dataset."""
    training_matrix, training_labels = load_dataset('digits/trainingDigits')
    test_matrix, test_labels = load_dataset('digits/testDigits')

    num_test_samples = len(test_labels)
    error_count = 0.0

    for i in range(num_test_samples):
        predicted_label = knn_classifier(test_matrix[i], training_matrix, training_labels, k=3)
        actual_label = test_labels[i]

        print(f'The KNN classified result is: {predicted_label}, the actual class is: {actual_label}')
        if predicted_label != actual_label:
            error_count += 1

    print(f'\nThe total number of incorrectly classified samples is: {error_count}')
    print(f'The error rate is: {error_count / num_test_samples:.6f}')

if __name__ == '__main__':
    handwriting_recognition_test()
