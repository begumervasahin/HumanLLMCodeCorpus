import itertools

def floyd_warshall(graph, vertex_count):
    """
    Perform the Floyd-Warshall algorithm to find the shortest paths between all pairs of nodes.
    
    Parameters:
    graph (list of list of int): The adjacency matrix of the graph.
    vertex_count (int): The number of vertices in the graph.
    """
    for k in range(vertex_count):
        for i, j in itertools.product(range(vertex_count), range(vertex_count)):
            if graph[i][j] > graph[i][k] + graph[k][j]:
                graph[i][j] = graph[i][k] + graph[k][j]

def initialize_graph(vertex_count):
    """
    Initialize the graph with default values and specific edges.
    
    Parameters:
    vertex_count (int): The number of vertices in the graph.
    
    Returns:
    list of list of int: The initialized adjacency matrix.
    """
    INF = 1000  # A large number representing infinity
    graph = [[INF for _ in range(vertex_count)] for _ in range(vertex_count)]
    
    # Set the diagonal to zero
    for i in range(vertex_count):
        graph[i][i] = 0

    # Define specific edges
    edges = [
        (0, 2, -2),
        (1, 0, 4),
        (1, 2, 3),
        (2, 3, 2),
        (3, 1, -1)
    ]
    
    for u, v, w in edges:
        graph[u][v] = w
    
    return graph

def print_graph(graph):
    """
    Print the adjacency matrix of the graph.
    
    Parameters:
    graph (list of list of int): The adjacency matrix to print.
    """
    for row in graph:
        print(row)

def main():
    """
    Main function to initialize the graph, perform the Floyd-Warshall algorithm,
    and print the initial and resulting adjacency matrices.
    """
    vertex_count = 4
    graph = initialize_graph(vertex_count)
    
    print("Initial graph:")
    print_graph(graph)
    
    print("\nAfter applying Floyd-Warshall algorithm:")
    floyd_warshall(graph, vertex_count)
    print_graph(graph)

if __name__ == '__main__':
    main()
