import torch
import matplotlib.pyplot as plt

class Stats:
    """
    A class for computing and plotting statistics such as the number of non-zeros,
    objective gap, and effective passes during training.
    """

    def __init__(self):
        self.num_non_zeros = []
        self.objective_gap = []
        self.effective_passes = []
        self.current_iteration = 0

    def _compute_num_non_zeros(self, weights):
        """
        Computes the number of non-zero elements in the weights tensor.

        Parameters:
        weights: tensor of weights [1 x d]
        """
        non_zero_count = (torch.abs(weights) > 1e-5).int().sum().item()
        self.num_non_zeros.append(non_zero_count)

    def _compute_objective_gap(self, loss):
        """
        Records the objective loss.

        Parameters:
        loss: float
        """
        self.objective_gap.append(loss)

    def _compute_effective_passes(self):
        """Records the current iteration as an effective pass."""
        self.effective_passes.append(self.current_iteration)

    def compute(self, weights, loss):
        """
        Computes the statistics for the given weights and loss.

        Parameters:
        weights: tensor of weights [1 x d]
        loss: float
        """
        self._compute_num_non_zeros(weights)
        self._compute_objective_gap(loss)
        self._compute_effective_passes()
        self.current_iteration += 1

    def plot(self, title=""):
        """
        Plots the number of non-zeros and objective gap against effective passes.

        Parameters:
        title: str, optional title for the plots
        """
        self._plot_statistic(self.num_non_zeros, "Number of Non-Zeros", title, "NonZeros")
        self._plot_statistic(self.objective_gap, "Objective Loss", title, "Loss", color='r')

    def _plot_statistic(self, data, ylabel, title, filename_suffix, color='b'):
        """
        Plots a given statistic.

        Parameters:
        data: list of data points to plot
        ylabel: str, label for the y-axis
        title: str, title for the plot
        filename_suffix: str, suffix for the saved plot filename
        color: str, optional color for the plot line
        """
        plt.figure()
        plt.plot(data, marker='o', color=color)
        plt.xlabel("Effective Passes")
        plt.ylabel(ylabel)
        plt.title(f"{title} {ylabel}")
        plt.xticks(range(len(data)))
        plt.grid(True)
        plt.savefig(f"{title}_{filename_suffix}.png")
        plt.close()

# Example usage
if __name__ == "__main__":
    stats = Stats()

    # Example data (normally you'd have real weight tensors and loss values)
    for _ in range(10):
        weights = torch.randn(1, 100)
        loss = torch.abs(torch.randn(1)).item()
        stats.compute(weights, loss)

    stats.plot("Example")
