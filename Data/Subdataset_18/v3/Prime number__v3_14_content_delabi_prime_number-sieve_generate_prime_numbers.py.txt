def generate_prime_numbers(limit):
    """
    Generate prime numbers up to a specified limit using the Sieve of Eratosthenes algorithm.
    """
    if limit is None:
        return "Please enter a positive integer as the limit!"

    if not isinstance(limit, int):
        return "That is not an integer. Please enter a number without a decimal."

    if limit < 2:
        return []

    if limit == 2:
        return [2]

    sieve = list(range(3, limit + 1, 2))  # Create a list of odd numbers from 3 to limit
    sqrt_limit = int(limit ** 0.5)
    half_limit = (limit + 1) // 2 - 1
    i = 0
    prime = 3

    while prime <= sqrt_limit:
        if sieve[i]:  # If sieve[i] is not zero, it is a prime
            for j in range((prime * prime - 3) // 2, half_limit, prime):
                sieve[j] = 0  # Mark multiples of prime as zero
        i += 1
        prime = 2 * i + 3

    return [2] + [num for num in sieve if num]  # Add 2 and filter out non-primes

# Example usage
if __name__ == "__main__":
    limit = 50  # Replace with the desired limit
    primes = generate_prime_numbers(limit)
    print(f"Prime numbers up to {limit}: {primes}")
