import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from scipy.cluster.hierarchy import dendrogram, linkage
from sklearn.cluster import AgglomerativeClustering

def load_dataset(file_path):
    """
    Load and preprocess the dataset from a CSV file.

    Args:
        file_path (str): Path to the CSV file containing the dataset.

    Returns:
        np.ndarray: Feature matrix containing 'Annual Income (k$)' and 'Spending Score (1-100)'.
    """
    dataset = pd.read_csv(file_path)
    X = dataset.iloc[:, [3, 4]].values  # Extract relevant features for clustering
    return X

def plot_dendrogram(X):
    """
    Plot a dendrogram to visualize the hierarchical clustering process.

    Args:
        X (np.ndarray): Feature matrix used for clustering.
    """
    plt.figure(figsize=(10, 7))
    dendrogram(linkage(X, method='ward', metric='euclidean'))
    plt.title('Dendrogram')
    plt.xlabel('Customers')
    plt.ylabel('Euclidean Distance (Ward Method)')
    plt.show()

def perform_clustering(X, n_clusters=5):
    """
    Perform Agglomerative Hierarchical Clustering on the dataset.

    Args:
        X (np.ndarray): Feature matrix used for clustering.
        n_clusters (int): Number of clusters to form.

    Returns:
        np.ndarray: Cluster labels for each data point in the dataset.
    """
    model = AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage='ward')
    labels = model.fit_predict(X)
    return labels

def plot_clusters(X, labels):
    """
    Visualize the clusters identified by the Agglomerative Clustering.

    Args:
        X (np.ndarray): Feature matrix used for clustering.
        labels (np.ndarray): Cluster labels assigned to each data point.
    """
    plt.figure(figsize=(10, 7))

    cluster_colors = ['red', 'blue', 'green', 'magenta', 'cyan']
    cluster_names = ['Low Spenders', 'Standard', 'Target', 'Low Earners', 'Out of Target']

    for i, color in enumerate(cluster_colors):
        plt.scatter(X[labels == i, 0], X[labels == i, 1], s=100, color=color, label=cluster_names[i])

    plt.title('Clusters of Clients')
    plt.xlabel('Annual Income (k$)')
    plt.ylabel('Spending Score (1-100)')
    plt.legend()
    plt.show()

def main():
    """
    Execute the clustering analysis workflow: load data, perform clustering, and plot the results.
    """
    # Load the dataset
    X = load_dataset('Mall_Customers.csv')

    # Plot the dendrogram to help determine the optimal number of clusters
    plot_dendrogram(X)

    # Perform Agglomerative Clustering
    cluster_labels = perform_clustering(X)

    # Plot the resulting clusters
    plot_clusters(X, cluster_labels)

if __name__ == "__main__":
    main()
