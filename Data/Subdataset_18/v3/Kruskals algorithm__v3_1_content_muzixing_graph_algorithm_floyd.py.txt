def floyd(graph):
    """
    Floyd-Warshall algorithm for finding the shortest paths in a graph represented as an adjacency matrix.

    :param graph: Adjacency matrix (list of lists) where graph[i][j] is the weight of the edge from node i to node j.
    :return: A tuple containing the modified graph with shortest paths and the path reconstruction dictionary.
    """
    length = len(graph)
    path = {i: {j: [i, j] for j in range(length) if i != j} for i in range(length)}
    
    for k in range(length):
        for i in range(length):
            for j in range(length):
                if graph[i][j] > graph[i][k] + graph[k][j]:
                    graph[i][j] = graph[i][k] + graph[k][j]
                    path[i][j] = path[i][k] + path[k][j][1:]

    return graph, path

def floyd_dict(graph):
    """
    Floyd-Warshall algorithm for finding the shortest paths in a graph represented as a dictionary.

    :param graph: Dictionary where graph[src][dst] is the weight of the edge from node src to node dst.
    :return: A tuple containing the modified graph with shortest paths and the path reconstruction dictionary.
    """
    path = {src: {dst: [src, dst] for dst in graph[src] if src != dst} for src in graph}
    
    for mid in graph:
        for src in graph:
            for dst in graph[src]:
                if graph[src][dst] > graph[src][mid] + graph[mid][dst]:
                    graph[src][dst] = graph[src][mid] + graph[mid][dst]
                    path[src][dst] = path[src][mid] + path[mid][dst][1:]

    return graph, path
