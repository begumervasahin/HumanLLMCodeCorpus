from typing import Tuple

def modulo_multiply(a: int, b: int, mod: int) -> int:
    """
    Evaluates (a * b) % mod.

    Args:
        a (int): First operand.
        b (int): Second operand.
        mod (int): Modulus.

    Returns:
        int: Result of (a * b) % mod.

    Raises:
        ValueError: If mod is zero.
    """
    if mod == 0:
        raise ValueError("Modulus cannot be zero")
    return ((a % mod) * (b % mod)) % mod

def modulo_pow(a: int, b: int, mod: int) -> int:
    """
    Evaluates a^b % mod.

    Args:
        a (int): Base.
        b (int): Exponent.
        mod (int): Modulus.

    Returns:
        int: Result of (a^b) % mod.
    """
    result = 1
    base = a % mod
    while b > 0:
        if b % 2 == 1:  # If b is odd, multiply base with the result
            result = modulo_multiply(result, base, mod)
        base = modulo_multiply(base, base, mod)  # Square the base
        b //= 2
    return result

def egcd(a: int, b: int) -> Tuple[int, int, int]:
    """
    Extended Euclidean algorithm to compute the gcd.

    Args:
        a (int): First number.
        b (int): Second number.

    Returns:
        Tuple[int, int, int]: A tuple (g, x, y) where a*x + b*y = gcd(a, b).
    """
    if a == 0:
        return b, 0, 1
    g, x, y = egcd(b % a, a)
    return g, y - (b // a) * x, x

def modulo_div(a: int, b: int, mod: int) -> int:
    """
    Evaluates (a / b) % mod.

    Args:
        a (int): Numerator.
        b (int): Denominator.
        mod (int): Modulus.

    Returns:
        int: Result of (a / b) % mod.
    """
    return modulo_multiply(a, mulinv(b, mod), mod)

def mulinv(b: int, n: int) -> int:
    """
    Multiplicative inverse of b modulo n.

    Args:
        b (int): The number to find the inverse of.
        n (int): Modulus.

    Returns:
        int: The multiplicative inverse of b modulo n.

    Raises:
        ValueError: If the modular inverse does not exist.
    """
    g, x, _ = egcd(b, n)
    if g == 1:
        return x % n
    raise ValueError("Modular inverse does not exist")

# Example usage
if __name__ == '__main__':
    a = 3
    b = 4
    mod = 5

    print(f"Modulo Multiply: {modulo_multiply(a, b, mod)}")  # Output: 2
    print(f"Modulo Power: {modulo_pow(a, b, mod)}")  # Output: 1
    print(f"Extended GCD: {egcd(a, mod)}")  # Output: (1, -1, 1)
    print(f"Modulo Division: {modulo_div(a, b, mod)}")  # Output: 2
    print(f"Multiplicative Inverse: {mulinv(b, mod)}")  # Output: 4
