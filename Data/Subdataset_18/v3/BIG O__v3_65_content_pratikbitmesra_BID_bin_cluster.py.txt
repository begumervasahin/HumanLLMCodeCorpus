import os
import sys
import csv
from itertools import groupby

# Constants
SIZE_OF_DISK_GB = 250  # Size of disk in GB
TOTAL_NO_BINS = SIZE_OF_DISK_GB * 1024 / 128  # Total number of bins

def read_extracted_csv(file_path):
    """Read the extracted CSV file and process the data."""
    lba_list = []
    xfrlen_list = []
    timestamp_list = []
    operation_list = []
    bin_list = []

    # Read the extracted CSV file
    with open(file_path, 'r') as extracted_csv:
        reader = csv.reader(extracted_csv)
        next(reader)  # Skip the header
        for row in reader:
            bin_list.append(int(row[0]))
            lba_list.append(int(row[1]))
            xfrlen_list.append(int(row[2]))
            operation_list.append(row[3])
            timestamp_list.append(float(row[4]))

    # Cluster bins
    clustered_bins = [list(group) for _, group in groupby(bin_list)]
    
    bin_number = []
    bin_count = []
    end_time = []
    xfrlen_bin_sum = []
    operation_bin = []
    
    counter = 0
    flag_counter = 0

    for bin_cluster in clustered_bins:
        bin_number.append(bin_cluster[-1])
        bin_count.append(len(bin_cluster))
        counter += len(bin_cluster)
        end_time.append(timestamp_list[counter - 1])
        operation_bin.append(operation_list[counter - 1])
        xfrlen_bin_sum.append(sum(xfrlen_list[flag_counter:counter]) / (2 * 1024.0))  # Convert to MB
        flag_counter += len(bin_cluster)

    single_bin_info = list(zip(bin_number, bin_count, xfrlen_bin_sum, operation_bin, end_time))

    # Write the processed information to a new CSV file
    output_file = file_path + '_bin_info.csv'
    with open(output_file, 'w', newline='') as bin_info_file:
        writer = csv.writer(bin_info_file)
        writer.writerow(["bin_number", "bin_count", "xfrlen_bin_sum(MB)", "operation_bin", "end_time"])
        writer.writerows(single_bin_info)

def main():
    if len(sys.argv) < 2:
        print("Usage: python script.py <input_csv_file>")
        sys.exit(1)

    input_file = sys.argv[1]
    read_extracted_csv(input_file)

if __name__ == "__main__":
    main()
