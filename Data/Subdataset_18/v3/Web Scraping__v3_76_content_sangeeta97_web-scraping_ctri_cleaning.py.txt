import pandas as pd
import numpy as np
import re

# Function to clean 'ctri_id' column
def clean_ctri_id(series):
    return series.str.split('\\\\').str[0].str.replace(r"^\W+", '', regex=True)

# Function to clean 'country' column
def clean_country(series):
    return series.str.replace(r'(?<=[a-z])(?=[A-Z])', ' ', regex=True).str.replace(r'\W+', ' ', regex=True)

# Function to clean 'sample_size' column
def clean_sample_size(series):
    return series.str.split('=', n=1).str[-1].str.split('Sample Size from India').str[0].str.replace(r'\W+', '', regex=True)

# Function to clean 'sponsor_1' column
def clean_sponsor_1(series):
    return series.replace('Address', np.nan, regex=True).astype(str).str.split(',').str[0]

# Function to split and clean 'sponsor_2' column
def split_and_clean_sponsor_2(series):
    sponsor_20 = series.str.split(r',\s+\{0:', regex=True).str[0].str.split(',').str[0]
    sponsor_21 = series.str.split(r',\s+\{0:', regex=True).str[-1].str.split(',').str[0]
    return sponsor_20 + sponsor_21

# Function to clean text columns using a regular expression
def clean_text_column(series, regex):
    return series.replace(regex, np.nan, regex=True).str.replace('0', '').str.replace(r'\W+', ' ', regex=True)

# Function to clean 'intervention' column
def clean_intervention(series, dfn):
    intervention = dfn[11].str.split('Intervention').str[-1]
    intervention = intervention.str.replace(r'(0|1|2)(\:)', '', regex=True)
    intervention = intervention.str.replace(r'(0|1|2)(\.)', '', regex=True)
    intervention = intervention.str.replace(r'[^A-Za-z0-9-/:\,]', ' ', regex=True)
    intervention = intervention.replace(r'^.*xa0Year.*$', np.nan, regex=True)
    intervention = intervention.replace(r'^.*Type\s+Name\s+Details.*$', np.nan, regex=True)
    return intervention.str.strip()

# Load the first Excel file
df1 = pd.read_excel('results_25july.xlsx')

# Split the 'all' column into lists and stack them into a DataFrame
df1['sum_all'] = df1['all'].str.split(':%')
dfn = df1[['sum_all']].stack().apply(pd.Series)

# Load the second Excel file and rename columns
df2 = pd.read_excel('second_file.xlsx')  # Replace 'second_file.xlsx' with the actual file name
df2.columns = [
    'ctri_id', 'study_design', 'public_title', 'scientific_title', 'country', 
    'sample_size', 'phase', 'trial_type', 'sponsor_1', 'sponsor_2', 'condition', 
    'intervention', 'primary_outcome'
]

# Apply cleaning functions to the relevant columns
df2['ctri'] = clean_ctri_id(df2.ctri_id)
df2['country'] = clean_country(df2.country)
df2['sample_size'] = clean_sample_size(df2.sample_size)
df2['sponsor_1'] = clean_sponsor_1(df2.sponsor_1)
df2['sponsor_2'] = split_and_clean_sponsor_2(df2.sponsor_2)

# Compile regular expressions for cleaning text
io = re.compile(r'(?i)(^\W*name|none|nil\W*$)')

# Apply cleaning to 'sponsor_2', 'condition', and 'intervention' columns
df2['sponsor_2'] = clean_text_column(df2['sponsor_20'], io)
df2['condition'] = clean_text_column(df2.condition, io)
df2['intervention'] = clean_intervention(df2.intervention, dfn)

# Clean 'primary_outcome' column
df2['primary_outcome'] = df2.primary_outcome.str.replace(r'([0123])(\.|\:)', '', regex=True).astype(str)
df2['primary_outcome'] = df2['primary_outcome'].str.replace(r'[^A-Za-z0-9-/:]', ' ', regex=True).str.strip()
df2['primary_outcome'] = df2['primary_outcome'].str.replace(r'\s+', ' ', regex=True)

# Further clean 'intervention' column
df2['intervention'] = df2.intervention.astype(str).str.split(',', n=1).str[-1].str.split(',').str[0].str.strip()

# Select specific columns and save to an Excel file
df3 = df2[['ctri', 'public_title', 'scientific_title', 'sponsor_1', 'sponsor_2', 'condition', 'intervention', 'primary_outcome']]
df3.to_excel('last_4663_ctri_25july.xlsx', index=False)
df2.to_excel('last_4663_ctri_allfields_25july.xlsx', index=False)

# Load another Excel file and clean it
df4 = pd.read_excel('clean_ctri_19123_24july.xlsx')
df4.drop(['ctri_y', 'intervention_y'], axis=1, inplace=True)
df4.columns = ['ctri', 'sponsor_1', 'sponsor_2', 'primary_outcome', 'condition', 'public_title', 'scientific_title', 'intervention']

# Reset index and drop unnecessary columns in df3
df3 = df3.reset_index().drop(['level_1', 'level_0'], axis=1)

# Concatenate df3 and df4, drop duplicates, and save to an Excel file
ctri_19519 = pd.concat([df3, df4]).drop_duplicates(subset='ctri')
ctri_19519.to_excel('ctri_clean_total_19519.xlsx', index=False)
