def selection_sort(arr):
    """
    Perform a selection sort on the given array.
    
    Parameters:
    arr (list): The list of elements to be sorted.
    
    Returns:
    list: The sorted list.
    """
    n = len(arr)
    for i in range(n):
        min_index = i  # Assume the current element is the minimum
        for j in range(i + 1, n):
            if arr[j] < arr[min_index]:
                min_index = j
        arr[i], arr[min_index] = arr[min_index], arr[i]  # Swap the found minimum element with the current element
        print(f"Array after swap {i+1}: {arr}")  # Print the array after each swap
    return arr

# Initialize the array
arr = [100, 48, 70, 29, 2, 17]

# Call the selection sort function and get the sorted array
sorted_arr = selection_sort(arr)

# Print the final sorted array
print("Sorted array:")
print(sorted_arr)
