def fibonacci_sequence(limit):
    """
    Generates the Fibonacci sequence up to a given limit.

    Parameters:
    limit (int): The upper bound for the sequence values.

    Returns:
    list: A list containing the Fibonacci sequence up to the limit.
    """
    sequence = []
    a, b = 0, 1
    while b < limit:
        sequence.append(b)
        a, b = b, a + b
    return sequence

def main():
    """
    Main function to demonstrate the Fibonacci sequence generation.
    """
    limit = 10
    fib_sequence = fibonacci_sequence(limit)
    print(f"Fibonacci sequence up to {limit}: {fib_sequence}")

if __name__ == '__main__':
    main()
