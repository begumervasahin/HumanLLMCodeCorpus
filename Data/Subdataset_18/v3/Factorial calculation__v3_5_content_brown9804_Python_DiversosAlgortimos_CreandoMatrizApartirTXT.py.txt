import numpy as np
import sys

def read_file(file_name, expected_rows, expected_cols):
    """
    Reads a CSV file and converts it into a matrix.

    :param file_name: The name of the file to read, including its extension.
    :param expected_rows: Number of elements expected per row.
    :param expected_cols: Number of rows expected in the matrix.
    :return: A 2D list representing the matrix.
    """
    matrix = []
    try:
        with open(file_name, "r") as file:
            for line in file:
                line_values = line.strip().split(",")
                
                if len(line_values) != expected_rows:
                    print(f"Warning: Expected {expected_rows} elements per row, found {len(line_values)}.")
                
                matrix.append(line_values)
                
                if len(matrix) >= expected_cols:
                    break
    except FileNotFoundError:
        print("Error: The specified file was not found.")
        sys.exit()
    except ValueError:
        print("Error: There was a problem with the data in the file.")
        sys.exit()
    
    return matrix

def main():
    """
    Main function to handle user input, process the file, and display the matrix.
    """
    file_name = input("Enter the name of the file to read (including its extension, e.g., .txt):\n")
    
    try:
        expected_rows = int(input("Enter the number of elements per row:\n"))
        expected_cols = int(input("Enter the number of rows:\n"))
    except ValueError:
        print("Error: Please enter valid integers for the number of rows and columns.")
        sys.exit()
    
    matrix = read_file(file_name, expected_rows, expected_cols)
    
    print("Requested matrix (NxM):")
    matrix_np = np.array(matrix)
    print(matrix_np)

if __name__ == "__main__":
    main()
