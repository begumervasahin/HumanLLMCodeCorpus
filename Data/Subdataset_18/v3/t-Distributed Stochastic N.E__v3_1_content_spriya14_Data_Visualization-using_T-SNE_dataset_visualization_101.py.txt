import pandas as pd
import numpy as np
from sklearn.preprocessing import LabelEncoder, StandardScaler
from sklearn.model_selection import train_test_split
from sklearn.manifold import TSNE
import matplotlib.pyplot as plt

def load_and_clean_data(url):
    """
    Load and clean the dataset by removing columns with null values 
    and discarding the first seven columns.

    Args:
        url (str): URL to the CSV file.

    Returns:
        pd.DataFrame: Cleaned DataFrame with selected columns.
    """
    # Load dataset from the URL
    data = pd.read_csv(url)

    # Filter out columns with any null values
    data_cleaned = data.dropna(axis=1)

    # Remove the first seven columns
    data_cleaned = data_cleaned.iloc[:, 7:]

    return data_cleaned

def preprocess_data(data):
    """
    Preprocess the data by standardizing the features and encoding the labels,
    then split into training and testing sets.

    Args:
        data (pd.DataFrame): The cleaned DataFrame.

    Returns:
        tuple: Arrays of training and testing features and labels.
    """
    # Separate features and labels
    features = data.iloc[:, :-1].values
    labels = data.iloc[:, -1].values

    # Standardize the features
    scaler = StandardScaler()
    features_scaled = scaler.fit_transform(features)

    # Encode the labels
    label_encoder = LabelEncoder()
    labels_encoded = label_encoder.fit_transform(labels)

    # Split the data into training and test sets
    x_train, x_test, y_train, y_test = train_test_split(
        features_scaled, labels_encoded, test_size=0.1, random_state=0
    )

    return x_train, x_test, y_train, y_test

def visualize_tsne(x_test, y_test):
    """
    Visualize the test data using t-SNE to reduce the dimensionality to two components,
    and plot the data points.

    Args:
        x_test (np.ndarray): Test features.
        y_test (np.ndarray): Test labels.
    """
    # Reduce dimensions using t-SNE
    tsne = TSNE(n_components=2, random_state=0)
    x_test_2d = tsne.fit_transform(x_test)

    # Define markers and colors for the different classes
    markers = ['s', 'd', 'o', '^', 'v']
    colors = ['red', 'blue', 'lightgreen', 'purple', 'cyan']
    unique_classes = np.unique(y_test)

    # Plot the 2D t-SNE results
    plt.figure(figsize=(8, 6))
    for idx, class_label in enumerate(unique_classes):
        plt.scatter(
            x=x_test_2d[y_test == class_label, 0], 
            y=x_test_2d[y_test == class_label, 1],
            c=colors[idx], marker=markers[idx], label=class_label
        )

    plt.xlabel('t-SNE Component 1')
    plt.ylabel('t-SNE Component 2')
    plt.legend(loc='upper left')
    plt.title('t-SNE Visualization of Test Data')
    plt.show()

def main():
    """
    Main function to execute the data loading, preprocessing, and visualization steps.
    """
    # URL to the dataset
    data_url = "https://d396qusza40orc.cloudfront.net/predmachlearn/pml-training.csv"

    # Load and clean the data
    cleaned_data = load_and_clean_data(data_url)

    # Preprocess the data
    x_train, x_test, y_train, y_test = preprocess_data(cleaned_data)

    # Visualize the test data using t-SNE
    visualize_tsne(x_test, y_test)

if __name__ == "__main__":
    main()
