# vertice.py
class Vertex:
    def __init__(self, id):
        """
        Initialize a vertex with a given ID.
        :param id: The identifier for the vertex.
        """
        self.id = id
        self.adjacent = {}  # Dictionary of adjacent vertices and edge weights
        self.distance = float('inf')  # Distance from source vertex
        self.visited = False  # Visited status

    def add_adjacent(self, neighbor, weight=0):
        """
        Add a neighbor to the vertex.
        :param neighbor: The neighbor vertex.
        :param weight: The weight of the edge to the neighbor.
        """
        self.adjacent[neighbor] = weight

    def get_adjacent(self):
        """
        Get the neighbors of the vertex.
        :return: A list of neighbor vertices.
        """
        return self.adjacent.keys()

    def get_id(self):
        """
        Get the ID of the vertex.
        :return: The ID of the vertex.
        """
        return self.id

    def get_weight(self, neighbor):
        """
        Get the weight of the edge to a neighbor.
        :param neighbor: The neighbor vertex.
        :return: The weight of the edge.
        """
        return self.adjacent[neighbor]

    def set_distance(self, dist):
        """
        Set the distance of the vertex from the source.
        :param dist: The distance to be set.
        """
        self.distance = dist

    def get_distance(self):
        """
        Get the distance of the vertex from the source.
        :return: The distance of the vertex.
        """
        return self.distance

    def set_visited(self):
        """
        Mark the vertex as visited.
        """
        self.visited = True

    def is_visited(self):
        """
        Check if the vertex has been visited.
        :return: True if the vertex has been visited, False otherwise.
        """
        return self.visited


# grafo.py
from vertice import Vertex

class Graph:
    def __init__(self, directed=False):
        """
        Initialize the graph with an optional directed parameter.
        :param directed: Boolean indicating if the graph is directed.
        """
        self.vertices = {}
        self.directed = directed

    def add_vertex(self, id):
        """
        Add a vertex to the graph.
        :param id: The ID of the vertex to be added.
        :return: The created vertex.
        """
        vertex = Vertex(id)
        self.vertices[id] = vertex
        return vertex

    def add_edge(self, from_vert, to_vert, weight=0):
        """
        Add an edge to the graph.
        :param from_vert: The starting vertex of the edge.
        :param to_vert: The ending vertex of the edge.
        :param weight: The weight of the edge.
        """
        if from_vert not in self.vertices:
            self.add_vertex(from_vert)
        if to_vert not in self.vertices:
            self.add_vertex(to_vert)
        self.vertices[from_vert].add_adjacent(self.vertices[to_vert], weight)
        if not self.directed:
            self.vertices[to_vert].add_adjacent(self.vertices[from_vert], weight)

    def get_vertices(self):
        """
        Get the vertices of the graph.
        :return: A list of vertices.
        """
        return list(self.vertices.values())

    def get_vertex(self, id):
        """
        Get a specific vertex by ID.
        :param id: The ID of the vertex.
        :return: The vertex with the specified ID.
        """
        return self.vertices.get(id)

    def get_edges(self):
        """
        Get the edges of the graph.
        :return: A set of edges.
        """
        edges = set()
        for vertex in self.vertices.values():
            for neighbor in vertex.get_adjacent():
                edges.add((vertex, neighbor))
        return edges

    def __iter__(self):
        """
        Iterate over the vertices of the graph.
        :return: An iterator over the vertices.
        """
        return iter(self.vertices.values())


# Example usage
if __name__ == "__main__":
    graph = Graph(directed=False)

    # Adding vertices and edges
    edges = [
        (0, 1, 4),
        (0, 2, 1),
        (2, 1, 2),
        (1, 3, 1),
        (2, 3, 5),
        (3, 4, 3)
    ]
    
    for from_vert, to_vert, weight in edges:
        graph.add_edge(from_vert, to_vert, weight)

    # Print vertices
    print("Vertices of graph:")
    for vertex in graph:
        print(f"Vertex {vertex.get_id()}")

    # Print edges
    print("\nEdges of graph:")
    for edge in graph.get_edges():
        print(f"Edge from {edge[0].get_id()} to {edge[1].get_id()} with weight {edge[0].get_weight(edge[1])}")

    # Get specific vertex
    vertex_id = 2
    vertex = graph.get_vertex(vertex_id)
    if vertex:
        print(f"\nVertex {vertex.get_id()} found in graph")
    else:
        print(f"\nVertex {vertex_id} not found in graph")
