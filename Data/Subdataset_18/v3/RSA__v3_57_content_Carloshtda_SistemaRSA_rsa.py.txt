import secrets
import sympy

def is_prime(num):
    """Check if a number is prime using sympy library."""
    return sympy.isprime(num)

def mulinv(e, totient):
    """Calculate the modular inverse of e modulo totient using Extended Euclidean Algorithm."""
    g, x, y = egcd(e, totient)
    if g != 1:
        raise Exception('Modular inverse does not exist')
    else:
        return x % totient

def egcd(a, b):
    """Extended Euclidean Algorithm to find the greatest common divisor and coefficients."""
    if a == 0:
        return (b, 0, 1)
    else:
        g, y, x = egcd(b % a, a)
        return (g, x - (b // a) * y, y)

class Rsa:
    def __init__(self):
        self.__p = 0
        self.__q = 0
        self.__n = 0
        self.__e = 3
        self.__d = 0
        self.__minPrime = 0
        self.__maxPrime = 1000

    def file_setup(self, file_control):
        """Setup RSA keys from a provided file control object."""
        self.__p = int(file_control.keys[0])
        self.__q = int(file_control.keys[1])
        self.__d = int(file_control.keys[2])
        self.__e = int(file_control.keys[3])
        self.__n = int(file_control.keys[4])
        return self.__e, self.__n

    def setup(self):
        """Generate RSA keys using prime numbers."""
        primes = [i for i in range(self.__minPrime, self.__maxPrime) if is_prime(i)]
        
        self.__p = self.__choose_prime(primes)
        self.__q = self.__choose_prime(primes, exclude=self.__p)
        
        self.__n = self.__p * self.__q
        totient = (self.__p - 1) * (self.__q - 1)
        self.__d = mulinv(self.__e, totient)
        
        return self.__e, self.__n

    def __choose_prime(self, primes, exclude=None):
        """Select a suitable prime number from the list of primes."""
        prime = secrets.choice(primes)
        while (prime - 5) % 6 != 0 or prime == exclude:
            prime = secrets.choice(primes)
        return prime

    def encrypt(self, msg):
        """Encrypt a message using the public key."""
        encoded_msg = [pow(ord(char), self.__e, self.__n) for char in msg]
        return ' '.join(map(str, encoded_msg))

    def decrypt(self, encrypted_msg):
        """Decrypt a message using the private key."""
        encrypted_list = map(int, encrypted_msg.split())
        decrypted_list = [chr(pow(num, self.__d, self.__n)) for num in encrypted_list]
        return ''.join(decrypted_list)

    def get_public_n(self):
        """Return the value of n."""
        return self.__n

    def keys_to_string(self):
        """Return the keys as a string."""
        keys = [self.__p, self.__q, self.__d, self.__e, self.__n]
        return ' '.join(map(str, keys))

if __name__ == "__main__":
    rsa = Rsa()
    public_keys = rsa.setup()
    print(f"Public keys: {public_keys}")

    message = "Hello, World!"
    encrypted_message = rsa.encrypt(message)
    print(f"Encrypted message: {encrypted_message}")

    decrypted_message = rsa.decrypt(encrypted_message)
    print(f"Decrypted message: {decrypted_message}")
