import sys
import string
from collections import Counter

# Available languages and default settings
LANGUAGES = ['EN', 'DE']
DEFAULT_LANGUAGE = 'EN'
DEFAULT_INPUT_FILE = 'files/sample.txt'
DEFAULT_OUTPUT_FILE = 'out.txt'

def get_stopwords_file(language):
    """Return the stopwords file path for the given language."""
    return f'stopwords/stop_words_{language}.txt'

def get_language_selection():
    """Prompt the user to select a language and return the corresponding stopwords file."""
    selected_language = input(f"Select language from the following: {', '.join(LANGUAGES)} (default is {DEFAULT_LANGUAGE}): ").upper()
    if selected_language in LANGUAGES:
        print(f"Parsing your text with the {selected_language} stopwords.")
        return get_stopwords_file(selected_language)
    else:
        print("Invalid language. Assuming English...")
        return get_stopwords_file(DEFAULT_LANGUAGE)

def get_output_file():
    """Prompt the user to select an output file name and return it."""
    output_file = input(f"Select name of output text file (default is {DEFAULT_OUTPUT_FILE}): ")
    if not output_file:
        return DEFAULT_OUTPUT_FILE
    if not output_file.endswith('.txt'):
        output_file += '.txt'
    print(f"Printing your results to {output_file}.")
    return output_file

def get_input_file():
    """Determine the input file based on command-line arguments or use the default."""
    if len(sys.argv) > 2:
        print("\nUsage: python indexer.py <yourFile>")
        print(f"If no arguments are given, {DEFAULT_INPUT_FILE} and the selected stopwords file will be used as default files.\n")
        sys.exit()
    elif len(sys.argv) == 2:
        return sys.argv[1]
    else:
        return DEFAULT_INPUT_FILE

def index_words(input_file, stopwords_file, output_file):
    """Read input file, remove stopwords, and write word frequencies to output file."""
    with open(input_file, 'r', encoding='utf-8') as file:
        book_words = file.read().lower().split()
        book_words = [word.strip(string.punctuation) for word in book_words]

    with open(stopwords_file, 'r', encoding='utf-8') as file:
        stopwords = set(file.read().splitlines())

    final_words = [word for word in book_words if word not in stopwords]

    word_counts = Counter(final_words)
    total_words = sum(word_counts.values())
    average_frequency = total_words / len(word_counts)
    frequent_words = {word: count for word, count in word_counts.items() if count >= average_frequency}
    sorted_frequent_words = sorted(frequent_words.items())

    with open(output_file, 'w') as file:
        for word, count in sorted_frequent_words:
            file.write(f'{word}: {count}\n')

def main():
    stopwords_file = get_language_selection()
    output_file = get_output_file()
    input_file = get_input_file()
    
    print(f'Using {input_file} as input file and {stopwords_file} as stop words reference. Results will be printed to {output_file}.\n')
    
    index_words(input_file, stopwords_file, output_file)

if __name__ == '__main__':
    main()
