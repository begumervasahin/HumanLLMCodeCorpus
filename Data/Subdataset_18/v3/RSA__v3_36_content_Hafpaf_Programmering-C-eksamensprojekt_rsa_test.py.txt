import os
import random
from numpy import gcd
from library.Sieve_of_Eratosthenes import gen_primes

def prime_limit(interval):
    """Generate prime numbers below the given interval using the Sieve of Eratosthenes."""
    prime_array = []
    for prime in gen_primes():
        if prime < interval:
            prime_array.append(prime)
        else:
            break
    return prime_array

def prime_numbers(prime_array):
    """Select two distinct prime numbers from the generated prime array."""
    while True:
        p = random.choice(prime_array)
        q = random.choice(prime_array)
        print(f'Random prime_p below {prime_interval} is: {p}')
        print(f'Random prime_q below {prime_interval} is: {q}')
        if p != q:
            print("Primes are not equal: Pass")
            return p, q
        print("Primes must not be equal, generating new primes")

def calculate_n(p, q):
    """Calculate the modulus n."""
    n = p * q
    print("n: ", n)
    return n

def phi(p, q):
    """Calculate Euler's Totient function."""
    phi_result = (p - 1) * (q - 1)
    print("phi_n: ", phi_result)
    return phi_result

def number_e(phi_n):
    """Generate a random number e that is coprime with phi_n."""
    while True:
        e = random.randint(2, phi_n - 1)
        if gcd(phi_n, e) == 1:
            print("e: ", e)
            return e

def mod_inverse(e, phi_n):
    """Calculate the modular inverse of e mod phi_n."""
    d = pow(e, -1, phi_n)
    print("d: ", d)
    return d

def main():
    # Prompt the user to define the upper limit for prime number generation
    prime_interval = int(input("Generate primes below: "))

    # Generate prime numbers up to the specified interval
    prime_array = prime_limit(prime_interval)

    # Select two distinct prime numbers
    prime_p, prime_q = prime_numbers(prime_array)

    # Calculate n and phi(n)
    number_n = calculate_n(prime_p, prime_q)
    phi_n = phi(prime_p, prime_q)

    # Generate e and verify gcd(phi_n, e) is 1
    number_e_value = number_e(phi_n)

    # Calculate d, the modular inverse of e
    number_d = mod_inverse(number_e_value, phi_n)

    # Output the public and private keys
    print("Public key (e, n):", (number_e_value, number_n))
    print("Private key (d, n):", (number_d, number_n))

if __name__ == "__main__":
    main()
