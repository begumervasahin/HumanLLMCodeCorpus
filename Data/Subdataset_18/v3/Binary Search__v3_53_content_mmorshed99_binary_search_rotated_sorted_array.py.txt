class Solution:
    def search(self, A, B):
        def find_break_point(start, end, A):
            """Find the break point in the rotated sorted array."""
            if len(A) < 3:
                return start  # Array too small to have a valid break point

            while start <= end:
                mid = start + (end - start) // 2
                
                # Check if the midpoint is a break point
                if mid > 0 and A[mid] < A[mid - 1]:
                    return mid
                elif A[mid] >= A[start]:
                    if A[mid] > A[end]:
                        start = mid + 1
                    else:
                        break  # No break point, array is sorted
                else:
                    end = mid - 1
            
            return start  # Default return if no explicit break point found

        def binary_search(start, end, target, A):
            """Perform binary search for the target in the subarray."""
            while start <= end:
                mid = start + (end - start) // 2
                if A[mid] == target:
                    return mid
                elif A[mid] < target:
                    start = mid + 1
                else:
                    end = mid - 1
            return -1

        target = B
        n = len(A)

        # Find the break point in the array
        break_point = find_break_point(0, n - 1, A)
        
        # Perform binary search in the appropriate subarray
        if break_point == 0:  # No rotation
            return binary_search(0, n - 1, target, A)
        elif target >= A[0]:
            return binary_search(0, break_point - 1, target, A)
        else:
            return binary_search(break_point, n - 1, target, A)

# Example usage:
if __name__ == "__main__":
    solution = Solution()
    A = [4, 5, 6, 7, 0, 1, 2]  # Rotated sorted array
    B = 1  # Target to search for
    result = solution.search(A, B)
    print(f"Index of target {B}: {result}")
