import numpy as np
import pandas as pd
from sklearn.preprocessing import LabelEncoder, OneHotEncoder
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
import statsmodels.api as sm

# Load the dataset
dataset = pd.read_csv('50_Startups.csv')
X = dataset.iloc[:, :-1].values  # Independent variables
y = dataset.iloc[:, 4].values    # Dependent variable

# Encode categorical data (the 'State' column)
labelencoder = LabelEncoder()
X[:, 3] = labelencoder.fit_transform(X[:, 3])

onehotencoder = OneHotEncoder()
X = onehotencoder.fit_transform(X).toarray()

# Avoid the dummy variable trap by removing one dummy variable
X = X[:, 1:]

# Split the dataset into training and test sets
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=0)

# Train the multiple linear regression model on the training set
regressor = LinearRegression()
regressor.fit(X_train, y_train)

# Predict the test set results
y_pred = regressor.predict(X_test)

# Add a column of ones to X for the intercept term (for statsmodels)
X = np.append(arr=np.ones((X.shape[0], 1)).astype(int), values=X, axis=1)

# Perform backward elimination to find the optimal model
def backward_elimination(x, y, sl):
    numVars = x.shape[1]
    while numVars > 0:
        regressor_OLS = sm.OLS(y, x).fit()
        maxPVal = max(regressor_OLS.pvalues).astype(float)
        if maxPVal > sl:
            for j in range(numVars):
                if (regressor_OLS.pvalues[j].astype(float) == maxPVal):
                    x = np.delete(x, j, 1)
                    numVars -= 1
        else:
            break
    regressor_OLS.summary()
    return x

# Set the significance level to 0.05
SL = 0.05
X_opt = X[:, [0, 1, 2, 3, 4, 5]]

# Run backward elimination
X_Modeled = backward_elimination(X_opt, y, SL)

# Print the summary of the final model
regressor_OLS = sm.OLS(endog=y, exog=X_Modeled).fit()
print(regressor_OLS.summary())
