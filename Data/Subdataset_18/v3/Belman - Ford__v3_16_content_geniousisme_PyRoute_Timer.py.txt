import time
from threading import Thread, Timer

class CountDownTimer(Thread):
    def __init__(self, interval, target):
        """
        Initialize the countdown timer.
        :param interval: The time interval between target function calls.
        :param target: The target function to be called at each interval.
        """
        super().__init__()
        self.target = target
        self.interval = interval
        self.stopped = False
        self.daemon = True

    def run(self):
        """
        Run the countdown timer, calling the target function at each interval.
        """
        while not self.stopped:
            time.sleep(self.interval)
            self.target()

class ResetTimer:
    def __init__(self, interval, func_ptr, args=None):
        """
        Initialize the reset timer.
        :param interval: The time interval for the timer.
        :param func_ptr: The function to be called when the timer expires.
        :param args: The arguments to be passed to the function.
        """
        self.func_ptr = func_ptr
        self.interval = interval
        self.args = args if args else []
        self.countdown_timer = self._create_new_timer()

    def _create_new_timer(self):
        """
        Create a new timer instance.
        :return: A new Timer object.
        """
        timer = Timer(self.interval, self.func_ptr, self.args)
        timer.daemon = True
        return timer

    def reset(self):
        """
        Reset the timer.
        """
        self.stop()
        self.countdown_timer = self._create_new_timer()
        self.start()

    def start(self):
        """
        Start the timer.
        """
        self.countdown_timer.start()

    def stop(self):
        """
        Stop the timer.
        """
        self.countdown_timer.cancel()

# Example usage of CountDownTimer and ResetTimer
def example_function():
    print("Function called!")

if __name__ == "__main__":
    # Example usage of CountDownTimer
    countdown_timer = CountDownTimer(1, example_function)
    countdown_timer.start()

    # Let it run for 5 seconds and then stop
    time.sleep(5)
    countdown_timer.stopped = True

    # Example usage of ResetTimer
    reset_timer = ResetTimer(2, example_function)
    reset_timer.start()

    # Reset the timer after 3 seconds
    time.sleep(3)
    reset_timer.reset()

    # Let it run for another 3 seconds and then stop
    time.sleep(3)
    reset_timer.stop()
