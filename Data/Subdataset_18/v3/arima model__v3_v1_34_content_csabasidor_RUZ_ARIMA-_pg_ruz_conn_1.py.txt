def connect_to_db(params):
    """
    Establish a connection to the PostgreSQL database.

    Parameters:
    params (dict): Database connection parameters.

    Returns:
    connection: A psycopg2 database connection object.
    """
    try:
        connection = pg.connect(**params)
        print("Connected to the database!")
        return connection
    except Exception as error:
        print(f"Error connecting to the database: {error}")
        return None

def fetch_data_from_db(query, connection):
    """
    Execute a SQL query and fetch the results into a Pandas DataFrame.

    Parameters:
    query (str): The SQL query to execute.
    connection (object): A psycopg2 database connection object.

    Returns:
    DataFrame: A Pandas DataFrame containing the query results.
    """
    try:
        df = pd.read_sql(query, connection)
        return df
    except Exception as error:
        print(f"Error executing query: {error}")
        return None

        df = fetch_data_from_db(query, connection)
