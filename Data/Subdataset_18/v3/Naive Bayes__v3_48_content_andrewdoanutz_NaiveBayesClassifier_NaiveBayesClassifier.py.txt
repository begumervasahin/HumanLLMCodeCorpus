import sys
import gaussianBayes
import multi

def print_usage_and_exit():
    """Prints the usage information and exits the script."""
    print("Usage: script.py <train_file> <test_file> <vectorizer_train> <vectorizer_test>")
    sys.exit(1)

def main():
    # Ensure correct number of arguments are provided
    if len(sys.argv) != 5:
        print_usage_and_exit()
    
    train_file = sys.argv[1]
    test_file = sys.argv[2]
    vectorizer_train = sys.argv[3]
    vectorizer_test = sys.argv[4]
    
    # Multinomial with Bag of Words
    print("\nMultinomial with Bag of Words")
    print("-" * 20)
    bag_mn = multi.MN(train_file, test_file)
    bag_mn.predict(vectorizer_train, vectorizer_test)
    
    # Gaussian with Bag of Words
    print("\nGaussian with Bag of Words")
    print("-" * 20)
    bag_gn = gaussianBayes.GN("bag", train_file, test_file)
    bag_gn.predict(vectorizer_train, vectorizer_test, "bag")
    
    # Gaussian with TF-IDF
    print("\nGaussian with TF-IDF")
    print("-" * 20)
    tf_gn = gaussianBayes.GN("tf", train_file, test_file)
    tf_gn.predict(vectorizer_train, vectorizer_test, "tf")

if __name__ == "__main__":
    main()
