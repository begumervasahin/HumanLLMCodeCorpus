from sklearn.neighbors import NearestNeighbors
import numpy as np

def create_sample_dataset():
    """
    Create and return a sample dataset of 2D points.
    
    Returns:
        np.ndarray: A sample dataset.
    """
    return np.array([[-1, -1], [-2, -1], [-3, -2], [1, 1], [2, 1], [3, 2]])

def train_nearest_neighbors_model(X, n_neighbors=2, algorithm='ball_tree'):
    """
    Train the NearestNeighbors model using the given dataset.
    
    Args:
        X (np.ndarray): The dataset to train the model.
        n_neighbors (int): Number of neighbors to use.
        algorithm (str): Algorithm to use for computing the nearest neighbors.
    
    Returns:
        NearestNeighbors: The trained NearestNeighbors model.
    """
    model = NearestNeighbors(n_neighbors=n_neighbors, algorithm=algorithm)
    model.fit(X)
    return model

def get_nearest_neighbors(model, X):
    """
    Get the nearest neighbors for the given dataset using the trained model.
    
    Args:
        model (NearestNeighbors): The trained NearestNeighbors model.
        X (np.ndarray): The dataset to find the nearest neighbors for.
    
    Returns:
        tuple: Distances and indices of the nearest neighbors.
    """
    distances, indices = model.kneighbors(X)
    return distances, indices

def main():
    """Main function to demonstrate the usage of NearestNeighbors."""
    # Create a sample dataset
    X = create_sample_dataset()
    
    # Train the NearestNeighbors model
    model = train_nearest_neighbors_model(X)
    
    # Print the trained model
    print("Trained NearestNeighbors model:\n", model)
    
    # Get the nearest neighbors
    distances, indices = get_nearest_neighbors(model, X)
    
    # Print the distances and indices of the nearest neighbors
    print("\nDistances to nearest neighbors:\n", distances)
    print("\nIndices of nearest neighbors:\n", indices)

if __name__ == "__main__":
    main()
