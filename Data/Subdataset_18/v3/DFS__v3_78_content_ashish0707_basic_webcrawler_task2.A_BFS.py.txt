from webcrawler import Crawler  # Import the Crawler class from the webcrawler module
from bfs import BFS  # Import the BFS class from the bfs module

def configure_crawler(crawler):
    """
    Configure the web crawler with specific rules and parameters.
    
    Args:
        crawler (Crawler): The Crawler instance to configure.
    """
    # Add paths that must be included in the crawl
    crawler.add_to_mandatory_list('/wiki/')
    
    # Add paths that must be excluded from the crawl
    crawler.add_to_disallowed_list(':')
    
    # Set the text that the crawler should focus on
    crawler.set_focused_text('solar')

def main():
    """
    Main function to set up and start the web crawler with BFS search.
    """
    # Define the target URL for the web crawler
    target_url = "https://en.wikipedia.org/wiki/Sustainable_energy"
    
    # Create an instance of BFS searcher
    bfs_searcher = BFS()
    
    # Create and configure the Crawler instance
    crawler = Crawler(target_url, bfs_searcher)
    configure_crawler(crawler)
    
    # Start the web crawling process
    crawler.start()

if __name__ == "__main__":
    main()
