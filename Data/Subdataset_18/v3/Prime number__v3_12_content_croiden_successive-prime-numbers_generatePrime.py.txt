import sys
import math

def is_prime(num, primes):
    """
    Check if a number is prime.
    """
    sqt = int(math.sqrt(num)) + 1
    for prime in primes:
        if prime > sqt:
            break
        if num % prime == 0:
            return False
    return True

def generate_prime_substring(index):
    """
    Generate a string of prime numbers concatenated together and extract a substring from a specified index.
    """
    SUBSTRING_LENGTH = 5
    concatenated_primes = "23"  # Start with the first two primes: 2 and 3
    i = 5  # Start checking for primes from 5 onwards
    primes = [3]  # List of known primes to check divisibility

    while True:
        if is_prime(i, primes):
            primes.append(i)
            concatenated_primes += str(i)
            # Check if we have enough primes concatenated to get the required substring
            if len(concatenated_primes) > index + SUBSTRING_LENGTH:
                return concatenated_primes[index:index + SUBSTRING_LENGTH]
        i += 2  # Check only odd numbers for primality

def main():
    if len(sys.argv) != 2:
        print("Usage: python script.py <index>")
        return
    
    try:
        index = int(sys.argv[1])
    except ValueError:
        print("Please provide a valid integer index.")
        return
    
    result = generate_prime_substring(index)
    print(f"minion[{index}] = {result}")

if __name__ == "__main__":
    main()
