import sys

def verify(actual, expected):
    """
    Compare the actual and expected outputs and report any mismatches.
    
    Parameters:
    actual (list of int): The actual output.
    expected (list of int): The expected output.
    """
    failed = 0
    
    for i, (a, e) in enumerate(zip(actual, expected)):
        if a != e:
            print(f"Test case {i + 1} failed: expected {e}, got {a}")
            failed += 1
    
    print("---------------------------------")
    if failed == 0:
        print("Passed all test cases!")
    else:
        print(f"Failed {failed}/{len(actual)} test cases.")
    print("---------------------------------")

def read_input():
    """
    Read the actual output from stdin and convert it to a list of integers.
    
    Returns:
    list of int: The actual output.
    """
    return [int(line.strip()) for line in sys.stdin]

def read_expected(filename):
    """
    Read the expected output from a file and convert it to a list of integers.
    
    Parameters:
    filename (str): The name of the file containing the expected output.
    
    Returns:
    list of int: The expected output.
    """
    with open(filename, 'r') as f:
        return [int(line.strip()) for line in f]

def main():
    """
    The main function that orchestrates the reading of inputs, comparison, and result reporting.
    """
    if len(sys.argv) != 2:
        sys.exit("Usage: python judge.py EXPECTED < ACTUAL")
    
    expected_file = sys.argv[1]
    actual = read_input()
    expected = read_expected(expected_file)
    
    if len(actual) != len(expected):
        sys.exit("Error: The expected and actual outputs have different lengths")
    
    verify(actual, expected)

if __name__ == '__main__':
    main()
