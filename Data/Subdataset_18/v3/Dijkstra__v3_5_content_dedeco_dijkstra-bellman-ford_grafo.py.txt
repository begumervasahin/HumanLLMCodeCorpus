class Vertex:
    def __init__(self, id):
        """
        Initialize a Vertex with a unique identifier.

        Parameters:
        id (int or str): The unique identifier for the vertex.
        """
        self.id = id
        self._adjacent_vertices = {}

    def add_adjacent_vertex(self, vertex, weight):
        """
        Add an adjacent vertex with a specified weight.

        Parameters:
        vertex (Vertex): The adjacent vertex to be added.
        weight (float): The weight of the edge connecting this vertex to the adjacent vertex.
        """
        if not isinstance(vertex, Vertex):
            raise ValueError("The vertex must be an instance of the Vertex class.")
        if not isinstance(weight, (int, float)):
            raise ValueError("The weight must be a numeric value.")

        self._adjacent_vertices[vertex] = weight

    def __repr__(self):
        """
        Return a string representation of the Vertex.

        Returns:
        str: A string that represents the Vertex.
        """
        return f"Vertex(id={self.id})"

    def get_adjacent_vertices(self):
        """
        Get a dictionary of adjacent vertices and their respective weights.

        Returns:
        dict: A dictionary where keys are adjacent Vertex objects and values are edge weights.
        """
        return self._adjacent_vertices
