def find_perfect_numbers(limit: int) -> None:
    """
    Finds and prints all perfect numbers less than the specified limit.

    A perfect number is a positive integer that is equal to the sum of its proper divisors.

    Parameters:
    limit (int): The upper limit for finding perfect numbers.
    """
    for number in range(2, limit):
        sum_of_divisors = sum(divisor for divisor in range(1, number) if number % divisor == 0)
        if sum_of_divisors == number:
            print(f"Perfect number found: {number}")

def main() -> None:
    """
    Main function to prompt the user for a range and find perfect numbers within that range.
    """
    try:
        # Get the range number from the user
        limit = int(input("Input the range number: "))
        if limit < 2:
            print("Please enter a number greater than or equal to 2.")
            return
        find_perfect_numbers(limit)
    except ValueError:
        print("Please enter a valid integer.")

if __name__ == "__main__":
    main()
