import math
import matplotlib.pyplot as plt
import pandas as pd
from datetime import datetime

def parser(x):
    """Convert a string to a datetime object."""
    return datetime.strptime(x, '%Y-%m-%d %H:%M:%S')

def mean_absolute_percentage_error(y_true, y_pred):
    """
    Calculate the Mean Absolute Percentage Error (MAPE).

    Parameters:
    y_true: list or array of true values
    y_pred: list or array of predicted values

    Returns:
    float: MAPE value
    """
    total_error = sum(abs(pred - true) / true for true, pred in zip(y_true, y_pred))
    return total_error / len(y_true)

def load_data(file_path, usecols, date_parser):
    """
    Load the dataset from a CSV file.

    Parameters:
    file_path: str, path to the CSV file
    usecols: list, columns to use from the CSV file
    date_parser: function, function to parse dates

    Returns:
    DataFrame: pandas DataFrame with the loaded data
    """
    return pd.read_csv(file_path, usecols=usecols, header=0, parse_dates=[0], index_col=0, date_parser=date_parser)

def plot_data(data, title, xlabel, ylabel, legend_label, figsize=(12, 6)):
    """
    Plot the data.

    Parameters:
    data: array-like, data to plot
    title: str, title of the plot
    xlabel: str, label for the x-axis
    ylabel: str, label for the y-axis
    legend_label: str, label for the legend
    figsize: tuple, size of the figure
    """
    plt.figure(figsize=figsize)
    plt.plot(data, '-', label=legend_label)
    plt.legend(loc='upper right')
    plt.xlabel(xlabel)
    plt.ylabel(ylabel)
    plt.title(title)
    plt.show()

# Constants
FILE_PATH = '/Users/daihanru/Desktop/arima-lstm/DataSet/FEB15.csv'
USECOLS = [2, 3]

# Load the dataset
series = load_data(FILE_PATH, USECOLS, DATE_PARSER)
