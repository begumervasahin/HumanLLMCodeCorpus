class Graph:
    def __init__(self):
        """
        Initialize a new Graph instance with an empty adjacency list.
        """
        self.adjacency_list = {}

    def get_vertices(self):
        """
        Get a list of all vertices in the graph.

        Returns:
            list: A list of all vertices in the graph.
        """
        return list(self.adjacency_list.keys())

    def get_neighbours(self, vertex):
        """
        Get a list of all neighbours of a given vertex.

        Args:
            vertex (str): The vertex to get neighbours for.

        Returns:
            list: A list of all neighbours of the given vertex.
        """
        return self.adjacency_list.get(vertex, [])

    def print_vertices(self):
        """
        Print all vertices in the graph.
        """
        print("Vertices:")
        for vertex in self.get_vertices():
            print(vertex)
        print()

    def print_neighbours(self, vertex):
        """
        Print all neighbours of a given vertex.

        Args:
            vertex (str): The vertex to print neighbours for.
        """
        print(f"Neighbours of {vertex}:")
        for neighbour in self.get_neighbours(vertex):
            print(neighbour)
        print()

    def add_vertex(self, vertex):
        """
        Add a vertex to the graph if it does not already exist.

        Args:
            vertex (str): The vertex to add.
        """
        if vertex not in self.adjacency_list:
            self.adjacency_list[vertex] = []

    def add_undirected_edge(self, u, v):
        """
        Add an undirected edge to the graph.

        Args:
            u (str): One vertex of the edge.
            v (str): The other vertex of the edge.
        """
        self.add_vertex(u)
        self.add_vertex(v)
        if v not in self.adjacency_list[u]:
            self.adjacency_list[u].append(v)
        if u not in self.adjacency_list[v]:
            self.adjacency_list[v].append(u)

    def add_directed_edge(self, u, v):
        """
        Add a directed edge to the graph.

        Args:
            u (str): The start vertex of the edge.
            v (str): The end vertex of the edge.
        """
        self.add_vertex(u)
        self.add_vertex(v)
        if v not in self.adjacency_list[u]:
            self.adjacency_list[u].append(v)

if __name__ == "__main__":
    # Create a graph instance
    graph = Graph()

    # Add vertices and edges
    graph.add_vertex('A')
    graph.add_vertex('B')
    graph.add_vertex('C')
    graph.add_undirected_edge('A', 'B')
    graph.add_undirected_edge('A', 'C')
    graph.add_directed_edge('B', 'C')

    # Print vertices and their neighbours
    graph.print_vertices()
    graph.print_neighbours('A')
    graph.print_neighbours('B')
    graph.print_neighbours('C')
