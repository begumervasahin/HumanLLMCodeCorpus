import os
import time
import string
import operator
import nltk
from nltk.corpus import stopwords
from nltk.stem.wordnet import WordNetLemmatizer

# Ensure necessary NLTK data is downloaded
nltk.download('stopwords')
nltk.download('wordnet')

def text_cleanup(text):
    """
    Cleans up the input text by removing punctuation, stopwords, 
    and converting to lowercase.
    """
    text_without_punctuation = ''.join([char for char in text if char not in string.punctuation])
    text_without_stopwords = ' '.join(
        [word for word in text_without_punctuation.split() if word.lower() not in stopwords.words('english')]
    )
    cleaned_text = [word.lower() for word in text_without_stopwords.split()]
    return cleaned_text

def process_emails(directory_path):
    """
    Processes all emails in the specified directory, cleans up the text, 
    lemmatizes words, and counts word occurrences.
    """
    lemmatizer = WordNetLemmatizer()
    word_count = {}
    processed_files = 0

    for file_name in os.listdir(directory_path):
        file_path = os.path.join(directory_path, file_name)
        with open(file_path, "r", encoding='utf-8', errors='ignore') as file:
            words = text_cleanup(file.read())
            for word in words:
                if not word.isdigit() and len(word) > 2:
                    lemmatized_word = lemmatizer.lemmatize(word)
                    word_count[lemmatized_word] = word_count.get(lemmatized_word, 0) + 1

        processed_files += 1
        if processed_files % 100 == 0:
            print(f"Processed {processed_files} files")

    return word_count

def save_word_counts(word_counts, output_file, min_count=100):
    """
    Saves the word counts to a CSV file, filtering out words that occur
    less than the specified minimum count.
    """
    sorted_word_counts = sorted(word_counts.items(), key=operator.itemgetter(1), reverse=True)

    with open(output_file, "w+") as file:
        file.write('word,count\n')
        for word, count in sorted_word_counts:
            if count < min_count:
                break
            file.write(f"{word},{count}\n")

def main():
    start_time = time.time()
    directory_path = "emails/"
    output_file = "wordslist.csv"

    word_counts = process_emails(directory_path)
    save_word_counts(word_counts, output_file)

    elapsed_time = round(time.time() - start_time, 2)
    print(f"Time (in seconds) to pre-process the emails: {elapsed_time}")

if __name__ == "__main__":
    main()
