from openerp import models, fields

class ResRegion(models.Model):
    """
    Model representing a region.
    """
    _name = 'res.region'
    _description = 'Region'

    name = fields.Char(string='Region', required=True)


class ResPartner(models.Model):
    """
    Extends the res.partner model to include a reference to res.region.
    """
    _inherit = 'res.partner'

    region_id = fields.Many2one('res.region', string='Region Client')
