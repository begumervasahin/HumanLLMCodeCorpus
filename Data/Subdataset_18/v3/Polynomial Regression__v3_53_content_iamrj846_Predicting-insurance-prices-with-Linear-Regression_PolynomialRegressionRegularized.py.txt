import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import LabelEncoder, OneHotEncoder, StandardScaler, PolynomialFeatures
from sklearn.linear_model import Lasso
from sklearn.model_selection import train_test_split
from sklearn.metrics import r2_score

# Load the dataset
dataset = pd.read_csv('insurance.csv')

# Separate features and target variable
X = dataset.iloc[:, :5].values
y = dataset.iloc[:, 6].values

# Encode categorical features ('sex' and 'smoker')
labelencoder = LabelEncoder()
X[:, 1] = labelencoder.fit_transform(X[:, 1])  # Encode 'sex' column
X[:, 4] = labelencoder.fit_transform(X[:, 4])  # Encode 'smoker' column

# Apply OneHotEncoder to categorical columns to avoid any ordinal nature
onehotencoder = OneHotEncoder(categorical_features=[1, 4])
X = onehotencoder.fit_transform(X).toarray()

# Avoid the dummy variable trap by removing the first dummy column for each categorical feature
X = X[:, [1, 3, 4, 5, 6]]

# Feature scaling for features (X) and target variable (y)
scaler_X = StandardScaler()
X = scaler_X.fit_transform(X)

scaler_y = StandardScaler()
y = scaler_y.fit_transform(y.reshape(-1, 1))

# Polynomial feature transformation
poly_features = PolynomialFeatures(degree=4)
X_poly = poly_features.fit_transform(X)

# Split the dataset into training and testing sets
X_train, X_test, y_train, y_test = train_test_split(X_poly, y, test_size=0.2, random_state=0)

# Train the Lasso regression model
lasso_regressor = Lasso(alpha=0.01, fit_intercept=False)
lasso_regressor.fit(X_train, y_train)

# Predict on training set
y_train_pred = lasso_regressor.predict(X_train)
y_train_pred = scaler_y.inverse_transform(y_train_pred)
y_train = scaler_y.inverse_transform(y_train)

# Predict on testing set
y_test_pred = lasso_regressor.predict(X_test)
y_test_pred = scaler_y.inverse_transform(y_test_pred)
y_test = scaler_y.inverse_transform(y_test)

# Calculate R² score for training and testing sets
r2_score_train = r2_score(y_train, y_train_pred) * 100
r2_score_test = r2_score(y_test, y_test_pred) * 100

# Print the R² scores
print(f'R2 Score for Training Set: {r2_score_train:.2f}%')
print(f'R2 Score for Testing Set: {r2_score_test:.2f}%')

# Optional: Plotting the results
plt.scatter(y_test, y_test_pred, color='blue', label='Predicted vs Actual')
plt.xlabel('Actual values')
plt.ylabel('Predicted values')
plt.title('Lasso Regression Predicted vs Actual')
plt.legend()
plt.show()
