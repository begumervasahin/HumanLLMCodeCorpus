def missing_number(nums):
    """
    Given a list of numbers containing n distinct numbers in the range [0, n], find the missing number.
    :param nums: List[int]
    :return: int
    """
    lower_bound = 0
    upper_bound = max(nums)
    expected_length = upper_bound - lower_bound + 1

    print(lower_bound, upper_bound, expected_length)  # Debugging statement to print range bounds and expected length
    
    index = 0
    while expected_length != 0:
        current_number = lower_bound + index
        if current_number in nums:
            index += 1
            expected_length -= 1
        else:
            return current_number
    
    return lower_bound + index

# Example usage:
nums = [0]
print(missing_number(nums))  # Output should be 1

# Additional test cases
nums = [3, 0, 1]
print(missing_number(nums))  # Output should be 2

nums = [0, 1]
print(missing_number(nums))  # Output should be 2

nums = [9, 6, 4, 2, 3, 5, 7, 0, 1]
print(missing_number(nums))  # Output should be 8
