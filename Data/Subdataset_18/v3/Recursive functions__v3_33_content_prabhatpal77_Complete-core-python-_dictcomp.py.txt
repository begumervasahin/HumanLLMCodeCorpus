# Create a dictionary with each key-value pair (p, p*p) for p in the range 0 to 9
squares = {p: p * p for p in range(10)}
print("Squares (0 to 9):", squares)

# Create a dictionary with each key-value pair (q, q**2) for q in the range 10 to 19, only if q is even
even_squares = {q: q ** 2 for q in range(10, 20) if q % 2 == 0}
print("Even squares (10 to 19):", even_squares)

# Create a dictionary with each key-value pair (r, r*2) for r in the range 20 to 29, only if r is odd
odd_doubles = {r: r * 2 for r in range(20, 30) if r % 2 != 0}
print("Odd doubles (20 to 29):", odd_doubles)
