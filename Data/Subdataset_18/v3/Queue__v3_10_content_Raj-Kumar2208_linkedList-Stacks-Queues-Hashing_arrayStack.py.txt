class StackArray:
    def __init__(self, limit=10):
        """
        Initialize the stack with a specified limit.
        
        Parameters:
        limit (int): The maximum number of elements that can be held in the stack.
        """
        self.limit = limit
        self.stack = []

    def push(self, data):
        """
        Push a new element onto the stack.
        
        Parameters:
        data: The element to be added to the stack.
        
        Raises:
        ValueError: If the stack is already at its limit.
        """
        if self.size() >= self.limit:
            raise ValueError("Stack Overflow")
        self.stack.append(data)
        print(f"Stack After Push: {self.stack}")

    def is_empty(self):
        """
        Check if the stack is empty.
        
        Returns:
        bool: True if the stack is empty, False otherwise.
        """
        return len(self.stack) == 0

    def pop(self):
        """
        Remove and return the top element from the stack.
        
        Returns:
        The top element of the stack.
        
        Raises:
        ValueError: If the stack is empty.
        """
        if self.is_empty():
            raise ValueError("Stack Underflow")
        return self.stack.pop()

    def peek(self):
        """
        Return the top element of the stack without removing it.
        
        Returns:
        The top element of the stack.
        
        Raises:
        ValueError: If the stack is empty.
        """
        if self.is_empty():
            raise ValueError("Stack Underflow")
        return self.stack[-1]

    def size(self):
        """
        Return the number of elements in the stack.
        
        Returns:
        int: The number of elements in the stack.
        """
        return len(self.stack)

    def print_stack(self):
        """
        Print the current elements in the stack.
        """
        print(self.stack)

if __name__ == "__main__":
    stack = StackArray(5)
    print(f"Is stack empty? {stack.is_empty()}")
    stack.push(5)
    stack.push(6)
    stack.push(7)
    print(f"Top element is: {stack.peek()}")
    stack.pop()
    stack.print_stack()
