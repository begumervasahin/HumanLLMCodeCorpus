import csv
import math
import operator

# Function to load training data from a CSV file
def load_train_data(file_name):
    training_data = []
    try:
        with open(file_name) as train_file:
            reader = csv.reader(train_file)
            for row in reader:
                temp_list = [float(value) if index != len(row) - 1 else value for index, value in enumerate(row)]
                training_data.append(temp_list)
        print("Training data loaded successfully.")
    except IOError:
        print("File not available. Please check the filename.")
        exit()
    return training_data

# Function to calculate Euclidean distance between test instance and all training instances
def calculate_distance(train_set, test_instance):
    distances = []
    for train_instance in train_set:
        if len(train_instance) - 1 != len(test_instance):
            print("Dimensions of test and training data do not match.")
            break
        distance = math.sqrt(sum((test_instance[i] - train_instance[i]) ** 2 for i in range(len(test_instance))))
        distances.append(round(distance, 3))
    return distances

# Function to find the k-nearest neighbors and predict the class
def find_neighbors(training_data, distances, num_neighbors):
    top_indices = sorted(range(len(distances)), key=lambda i: distances[i])[:num_neighbors]
    class_count = {}
    for index in top_indices:
        label = training_data[index][-1]
        class_count[label] = class_count.get(label, 0) + 1
    predicted_class = max(class_count.items(), key=operator.itemgetter(1))[0]
    return predicted_class

if __name__ == "__main__":
    print("Place the input dataset CSV file in the same directory as this Python module.")
    file_name = input("Enter the name of the training dataset CSV file: ")
    training_data = load_train_data(file_name)
    
    test_input = input("Enter the test values separated by commas: ")
    test_instance = [float(value) for value in test_input.split(",")]
    
    k = int(input("Enter the number of neighbors to consider: "))
    
    distances = calculate_distance(training_data, test_instance)
    if distances:
        predicted_class = find_neighbors(training_data, distances, k)
        print(f"The predicted class for the test instance is: {predicted_class}")
