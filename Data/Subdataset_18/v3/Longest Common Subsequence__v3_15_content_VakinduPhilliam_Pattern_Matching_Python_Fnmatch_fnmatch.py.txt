import fnmatch
import os

def list_txt_files(directory='.'):
    """
    Lists all .txt files in the specified directory.

    Parameters:
    directory (str): The directory to search for .txt files. Defaults to the current directory.

    Returns:
    list: A list of .txt files in the specified directory.
    """
    # Get the list of all files in the directory
    files_in_directory = os.listdir(directory)
    
    # Filter the list to include only .txt files
    txt_files = [file for file in files_in_directory if fnmatch.fnmatch(file, '*.txt')]
    
    return txt_files

def print_txt_files(directory='.'):
    """
    Prints all .txt files in the specified directory.

    Parameters:
    directory (str): The directory to search for .txt files. Defaults to the current directory.
    """
    txt_files = list_txt_files(directory)
    for file in txt_files:
        print(file)

if __name__ == "__main__":
    print_txt_files()
