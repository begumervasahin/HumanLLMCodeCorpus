import os
import csv

def parse_and_create_map(coordinates_directory):
    """
    Parse coordinates from CSV files in the specified directory and create a map.

    Parameters:
    coordinates_directory (str): The path to the directory containing CSV files with coordinates.

    Returns:
    list of lists: The generated map, represented as a list of lists.
    """
    points = []

    # Read and process all CSV files in the directory
    for filename in os.listdir(coordinates_directory):
        filepath = os.path.join(coordinates_directory, filename)
        
        with open(filepath, 'r') as file:
            reader = csv.reader(file, delimiter=',')
            for row in reader:
                cleaned_row = [entry.strip() for entry in row if entry.strip()]
                points.extend(cleaned_row)

    # Extract and parse coordinates from the cleaned points
    x_coords, y_coords = zip(*[
        (int(x_str), int(y_str))
        for point in points
        if 'y' in point
        for x_str, y_str in [point[1:].split('y')]
    ]) if points else ([], [])

    # Determine the dimensions of the map
    max_x = max(x_coords, default=0)
    max_y = max(y_coords, default=0)
    
    # Create and initialize the map
    map_ = [['.' for _ in range(max_x + 1)] for _ in range(max_y + 1)]
    
    # Fill the map with 'x' at the specified coordinates
    for x, y in zip(x_coords, y_coords):
        map_[y][x] = 'x'
    
    # Print the map to the console
    for row in map_:
        print(''.join(row))
    
    # Save the map to a text file
    with open('map.txt', 'w') as file:
        for row in map_:
            file.write(''.join(row) + '\n')

    return map_

# Example usage
if __name__ == "__main__":
    # Replace 'CoordinateSystem' with the path to your coordinates directory
    map_result = parse_and_create_map("CoordinateSystem")
