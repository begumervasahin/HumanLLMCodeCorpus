import random

def simulate_monty_hall(num_simulations=1000):
    """
    Simulates the Monty Hall problem and counts the number of wins for stay and switch strategies.
    
    Args:
    num_simulations (int): The number of times to simulate the game.
    
    Returns:
    tuple: A tuple containing the number of wins for stay and switch strategies.
    """
    stay_wins = 0
    switch_wins = 0

    for _ in range(num_simulations):
        # Set up the game with one prize (1) and two goats (0)
        doors = [1, 0, 0]
        random.shuffle(doors)

        # Contestant makes an initial choice
        initial_choice = random.randrange(3)
        chosen_door = doors[initial_choice]

        # Host reveals a goat by removing one 0 from the remaining doors
        remaining_doors = [door for i, door in enumerate(doors) if i != initial_choice]
        host_reveal_index = remaining_doors.index(0)
        del remaining_doors[host_reveal_index]

        # Check the outcomes for staying or switching
        if chosen_door == 1:
            stay_wins += 1
        if remaining_doors[0] == 1:
            switch_wins += 1

    return stay_wins, switch_wins

# Run the simulation and print the results
stay_wins, switch_wins = simulate_monty_hall(1000)
print(f"Stay wins = {stay_wins}")
print(f"Switch wins = {switch_wins}")
