def normalize_date(date_str):
    """
    Normalize a date string to the format 'YYYY-MM-DD'.
    Accepted formats are: 'YYYY-MM-DD', 'YYYY/MM/DD', and 'DD.MM.YYYY'.
    
    :param date_str: The date string to normalize.
    :return: Normalized date string in the format 'YYYY-MM-DD'.
    :raises DateError: If the date format is not allowed.
    """
    if re.match(r'^\d{1,2}\.\d{1,2}\.\d{4}$', date_str):
        date_obj = datetime.strptime(date_str, '%d.%m.%Y')
    elif re.match(r'^\d{4}\-\d{1,2}\-\d{1,2}$', date_str):
        date_obj = datetime.strptime(date_str, '%Y-%m-%d')
    elif re.match(r'^\d{4}\/\d{1,2}\/\d{1,2}$', date_str):
        date_obj = datetime.strptime(date_str, '%Y/%m/%d')
    else:
        raise DateError(f"Date format is not allowed: {date_str}")
    return date_obj.strftime("%Y-%m-%d")
