import socket
import DiffieHellman
import nacl.secret
import nacl.utils
from binascii import hexlify

def initialize_client(server_address=('localhost', 5555)):
    """
    Initialize the Diffie-Hellman client and set up the socket connection.

    Args:
        server_address (tuple): A tuple containing the server IP address and port number.

    Returns:
        tuple: A tuple containing the client object and the socket object.
    """
    client = DiffieHellman.D_H()
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.connect(server_address)
    print(f"Connected to {server_address[0]} on port {server_address[1]}")
    return client, sock

def exchange_keys(client, sock):
    """
    Exchange public keys with the server and generate the shared secret key.

    Args:
        client: The Diffie-Hellman client object.
        sock: The socket object.

    Returns:
        bytes: The shared secret key.
    """
    # Send client's public key to the server
    sock.sendall(str(client.pubKey).encode())

    # Receive server's public key and generate shared secret key
    server_pub_key = int(sock.recv(9000).decode())
    client.genKey(server_pub_key)
    secret_key = client.getKey()
    print("Secret key:", hexlify(secret_key).decode())
    return secret_key

def communicate(sock, box):
    """
    Handle the communication with the server, encrypting and decrypting messages.

    Args:
        sock: The socket object.
        box: The NaCl SecretBox object for encryption and decryption.
    """
    while True:
        message = input('Enter a message or "exit" to close the program: ')
        
        if message.lower() == 'exit':
            print('Closing socket.')
            sock.close()
            break
        
        # Encrypt the message and send it to the server
        encrypted_message = box.encrypt(message.encode())
        sock.sendall(encrypted_message)

        # Receive and decrypt the response from the server
        data = sock.recv(9000)
        decrypted_message = box.decrypt(data).decode()
        print(f'Received data: "{decrypted_message}"')

def main():
    """
    Main function to initialize the client, exchange keys, and communicate with the server.
    """
    client, sock = initialize_client()
    secret_key = exchange_keys(client, sock)
    box = nacl.secret.SecretBox(secret_key)
    communicate(sock, box)

if __name__ == "__main__":
    main()
