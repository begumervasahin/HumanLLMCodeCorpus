import knn
import time
import gc
import numpy as np

def evaluate_knn(k, knn_func, knn_func_name, label_func, label_func_name, pca_param, train_samples, train_labels, test_samples, ground_labels, result_file):
    """
    Evaluate the KNN algorithm with the given parameters and log the results.

    Args:
        k (int): Number of neighbors.
        knn_func (function): KNN distance function.
        knn_func_name (str): Name of the KNN distance function.
        label_func (function): Label function.
        label_func_name (str): Name of the label function.
        pca_param (int): PCA parameter.
        train_samples (array): Training samples.
        train_labels (array): Training labels.
        test_samples (array): Test samples.
        ground_labels (array): Ground truth labels.
        result_file (file): File to log the results.

    Returns:
        list: Single result with evaluation metrics.
    """
    single_result = [k, knn_func_name, label_func_name, pca_param]

    print(f'\n******** Conditions ******', file=result_file)
    print(f'k = {k}', file=result_file)
    print(f'knn_function = {knn_func_name}', file=result_file)
    print(f'label_function = {label_func_name}', file=result_file)
    print(f'pca_parameters = {pca_param}', file=result_file)
    print('**************************', file=result_file)

    start_time = time.time()
    result_labels, dimension = knn.get_test_samples_labels(
        k, train_samples, train_labels, test_samples, knn_func, label_func, pca_param
    )
    elapsed_time = time.time() - start_time

    accuracy, precision, recall, f1, macro_precision, macro_recall, macro_f1 = knn.result_evaluate(
        ground_labels, result_labels
    )

    print(f'accuracy = {accuracy}', file=result_file)
    single_result.append(accuracy)
    print(f'precision = {precision}', file=result_file)
    print(f'recall = {recall}', file=result_file)
    print(f'F1 = {f1}', file=result_file)
    print(f'macro_precision = {macro_precision}', file=result_file)
    single_result.append(macro_precision)
    print(f'macro_recall = {macro_recall}', file=result_file)
    single_result.append(macro_recall)
    print(f'macro_F1 = {macro_f1}', file=result_file)
    single_result.append(macro_f1)
    print(f'execution time = {elapsed_time}', file=result_file)
    single_result.append(elapsed_time)
    print(f'dimension = {dimension}', file=result_file)
    single_result.append(dimension)
    print(file=result_file)

    return single_result

def main():
    experiment_start_time = time.time()
    train_dir = './digits/trainingDigits'
    test_dir = './digits/testDigits'
    result_file = open('result.txt', 'w')

    train_samples, train_labels = knn.load_sample_set(train_dir)
    test_samples, ground_labels = knn.load_sample_set(test_dir)

    ks = [3, 5, 7, 9]
    knn_functions = [
        (knn.get_knn_e_dist, 'get_knn_e_dist'),
        (knn.get_knn_e_dist_with_kdtree, 'get_knn_e_dist_with_kdtree'),
        (knn.get_knn_m_dist, 'get_knn_m_dist')
    ]
    label_functions = [
        (knn.get_label_by_knn, 'get_label_by_knn'),
        (knn.get_label_by_wknn, 'get_label_by_wknn')
    ]
    pca_parameters = [0, 4, 8, 16, 32, 64, 128, 256]

    results = [
        ['k', 'knn_function', 'label_function', 'pca_parameters', 'accuracy', 'macro_precision', 'macro_recall',
         'macro_F1', 'execution_time', 'dimension']
    ]

    for k in ks:
        for knn_func, knn_func_name in knn_functions:
            for label_func, label_func_name in label_functions:
                for pca_param in pca_parameters:
                    single_result = evaluate_knn(
                        k, knn_func, knn_func_name, label_func, label_func_name, pca_param,
                        np.copy(train_samples), np.copy(train_labels), np.copy(test_samples), np.copy(ground_labels),
                        result_file
                    )
                    results.append(single_result)
                    gc.collect()

    experiment_end_time = time.time()
    print('***********************', file=result_file)
    print(f'Total Time = {experiment_end_time - experiment_start_time}', file=result_file)
    print('***********************', file=result_file)
    result_file.close()

    with open('result_table.csv', 'w') as csv_file:
        for row in results:
            csv_file.write(', '.join(map(str, row)) + '\n')

if __name__ == '__main__':
    main()
