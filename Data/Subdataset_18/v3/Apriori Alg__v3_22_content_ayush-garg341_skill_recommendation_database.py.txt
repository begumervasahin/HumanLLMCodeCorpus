import pandas as pd
from apyori import apriori

def get_skills_from_csv(file_path):
    """
    Read skills from a CSV file.

    Args:
        file_path (str): Path to the CSV file containing the skills data.

    Returns:
        list: List of lists containing skills.
    """
    data = pd.read_csv(file_path)
    skills = data['skills'].apply(lambda x: x.split(',')).tolist()
    return skills

def preprocess_skills(skills):
    """
    Preprocess the list of skills by removing unwanted characters.

    Args:
        skills (list): List of lists containing skills.

    Returns:
        list: Cleaned list of lists containing skills.
    """
    cleaned_skills = [[skill.strip().lower() for skill in skill_set] for skill_set in skills]
    return cleaned_skills

def apply_apriori(skills, min_support=0.004, min_confidence=0.5):
    """
    Apply Apriori algorithm to find frequent skill sets.

    Args:
        skills (list): List of lists containing skills.
        min_support (float): Minimum support value.
        min_confidence (float): Minimum confidence value.

    Returns:
        list: List of rules generated by the Apriori algorithm.
    """
    rules = apriori(skills, min_support=min_support, min_confidence=min_confidence)
    return list(rules)

def save_rules_to_csv(rules, file_path):
    """
    Save the Apriori rules to a CSV file.

    Args:
        rules (list): List of rules generated by the Apriori algorithm.
        file_path (str): Path to the output CSV file.
    """
    with open(file_path, 'w') as file:
        for rule in rules:
            file.write(f"{rule}\n")
import pandas as pd

def recommend_skills(skill, file_path):
    """
    Recommend related skills based on a particular skill.

    Args:
        skill (str): The skill to find recommendations for.
        file_path (str): Path to the CSV file containing the Apriori rules.

    Returns:
        list: List of recommended skills.
    """
    rules = pd.read_csv(file_path, header=None)
    recommendations = []
    for index, row in rules.iterrows():
        if skill in row[0]:
            recommendations.append(row[1])
    return recommendations
