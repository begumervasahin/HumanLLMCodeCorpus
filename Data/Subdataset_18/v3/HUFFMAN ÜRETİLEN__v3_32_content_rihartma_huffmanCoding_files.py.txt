class Files:
    def load_file(self, filename):
        """
        Loads the content of a file and returns it as a string.

        Parameters
        ----------
        filename : str
            The name of the file to load.

        Returns
        -------
        str
            The content of the file as a string.
        """
        with open(filename, 'r') as file:
            return file.read()

    def save_table(self, table_text, filename):
        """
        Saves a string (Huffman code table) to a file.

        Parameters
        ----------
        table_text : str
            The text to save (Huffman code table).
        filename : str
            The name of the file to save the table to.
        """
        with open(filename, 'w') as file:
            file.write(table_text)

    def save_encoded_file(self, filename, text, codes):
        """
        Saves the encoded text to a file using Huffman codes.

        Parameters
        ----------
        filename : str
            The name of the file to save.
        text : str
            The original text to encode.
        codes : dict
            A dictionary mapping characters to their Huffman codes.
        """
        binary_string = ''.join(codes[char] for char in text)
        
        # Convert binary string to ASCII characters
        encoded_text = ''.join(
            chr(int(binary_string[i:i + 8], 2)) 
            for i in range(0, len(binary_string) - len(binary_string) % 8, 8)
        )
        
        # Handle any remaining bits
        remaining_bits = len(binary_string) % 8
        if remaining_bits != 0:
            encoded_text += f"~~~{binary_string[-remaining_bits:]}"
        
        with open(filename, 'w') as file:
            file.write(encoded_text)

    def load_table(self, filename):
        """
        Loads a Huffman code table from a file and returns it as a dictionary.

        Parameters
        ----------
        filename : str
            The name of the file to load the table from.

        Returns
        -------
        dict
            A dictionary mapping Huffman codes to their respective characters.
        """
        with open(filename, "r") as file:
            content = file.read()
        
        codes = {}
        pairs = content.split(",")

        for pair in pairs:
            if pair:  # Ensure the pair is not empty
                char, code = pair.split("-")
                codes[code] = char
        
        return codes

# Example usage:
if __name__ == "__main__":
    files = Files()
    
    # Load text from a file
    original_text = files.load_file("example.txt")
    
    # Example Huffman code table
    huffman_codes = {
        'a': '101',
        'b': '111',
        'c': '000',
        # Additional mappings...
    }
    
    # Save the Huffman code table
    files.save_table("a-101,b-111,c-000", "huffman_table.txt")
    
    # Save the encoded file
    files.save_encoded_file("encoded_file.txt", original_text, huffman_codes)
    
    # Load the Huffman code table
    loaded_codes = files.load_table("huffman_table.txt")
    print(loaded_codes)
