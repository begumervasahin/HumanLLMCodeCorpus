def new_ascii(char):
    """
    Convert a character to its ASCII value and add 100.

    Args:
        char (str): The character to convert.

    Returns:
        int: The new ASCII value.
    """
    return ord(char) + 100

def old_ascii(modified_ascii):
    """
    Convert a modified ASCII value back to its original ASCII value by subtracting 100.

    Args:
        modified_ascii (int): The modified ASCII value.

    Returns:
        int: The original ASCII value.
    """
    return int(modified_ascii) - 100

def encode(message):
    """
    Encode a message by converting each character to its modified ASCII value.

    Args:
        message (str): The message to encode.

    Returns:
        str: The encoded message as a string of modified ASCII values.
    """
    return ''.join(str(new_ascii(char)).zfill(3) for char in message)

def decode(encoded_message):
    """
    Decode an encoded message back to its original form.

    Args:
        encoded_message (str): The encoded message as a string of modified ASCII values.

    Returns:
        str: The decoded original message.
    """
    return ''.join(chr(old_ascii(encoded_message[i:i+3])) for i in range(0, len(encoded_message), 3))

if __name__ == "__main__":
    # Example usage
    original_message = "Hello, World!"
    print("Original Message:", original_message)

    encoded_message = encode(original_message)
    print("Encoded Message:", encoded_message)

    decoded_message = decode(encoded_message)
    print("Decoded Message:", decoded_message)
