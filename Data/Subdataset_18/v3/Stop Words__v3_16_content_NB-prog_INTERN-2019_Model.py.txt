import torch
import torch.nn as nn
import numpy as np

class SkipGram(nn.Module):
    def __init__(self, vocab_size=20001, embedding_dims=50, neg_samples=20, word_freq=None, padding_idx=0):
        super(SkipGram, self).__init__()
        
        self.vocab_size = vocab_size
        self.embedding_dims = embedding_dims
        self.neg_samples = neg_samples
        
        self.input_embedding = nn.Embedding(vocab_size, embedding_dims, padding_idx=padding_idx)
        self.output_embedding = nn.Embedding(vocab_size, embedding_dims, padding_idx=padding_idx)
        
        self._initialize_weights()
        
        self.neg_sampling_dis = self._generate_neg_sampling_distribution(word_freq)
        self.device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
        
    def _initialize_weights(self):
        init_range = 0.5 / self.embedding_dims
        self.input_embedding.weight.data.uniform_(-init_range, init_range)
        self.output_embedding.weight.data.uniform_(-init_range, init_range)
        
    @staticmethod
    def _generate_neg_sampling_distribution(word_freq):
        neg_sampling_distribution = np.power(word_freq, 0.75)
        neg_sampling_distribution /= neg_sampling_distribution.sum()
        return neg_sampling_distribution
    
    def forward(self, target, contexts):
        batch_size, context_size = target.shape[0], contexts.shape[1]
        
        negative_samples = self._generate_negative_samples(batch_size, context_size)
        
        target_vectors = self.input_embedding(target).unsqueeze(2)  # (batch_size, embedding_dims, 1)
        context_vectors = self.output_embedding(contexts)           # (batch_size, context_size, embedding_dims)
        negative_vectors = self.output_embedding(negative_samples)  # (batch_size, context_size * neg_samples, embedding_dims)
        
        positive_loss = self._compute_positive_loss(context_vectors, target_vectors)
        negative_loss = self._compute_negative_loss(negative_vectors, target_vectors, context_size)
        
        total_loss = -(positive_loss + negative_loss).mean()
        
        return total_loss
    
    def _generate_negative_samples(self, batch_size, context_size):
        neg_samples = np.random.choice(
            self.vocab_size, 
            size=(batch_size, context_size * self.neg_samples), 
            replace=True, 
            p=self.neg_sampling_dis
        )
        return torch.from_numpy(neg_samples).to(dtype=torch.long, device=self.device)
    
    def _compute_positive_loss(self, context_vectors, target_vectors):
        return torch.bmm(context_vectors, target_vectors).squeeze().sigmoid().log()
    
    def _compute_negative_loss(self, negative_vectors, target_vectors, context_size):
        return torch.bmm(-negative_vectors, target_vectors).sigmoid().log().view(-1, context_size, self.neg_samples).sum(2)

if __name__ == "__main__":
    # Example word frequencies (for testing purposes)
    word_freq = np.array([1, 2, 3, 4, 5], dtype=np.float32)
    word_freq /= word_freq.sum()

    # Initialize the model
    model = SkipGram(vocab_size=6, embedding_dims=10, neg_samples=5, word_freq=word_freq)
    
    # Example data
    target = torch.tensor([1, 2, 3], dtype=torch.long)
    contexts = torch.tensor([[2, 3], [3, 4], [4, 5]], dtype=torch.long)
    
    # Forward pass
    loss = model(target, contexts)
    
    print("Loss:", loss.item())
