import numpy as np
from sklearn.linear_model import LinearRegression      
from sklearn.preprocessing import PolynomialFeatures   
from sklearn.metrics import mean_squared_error, r2_score

class LinearModel:
    def __init__(self, model):
        self.model = model
        self.rmse_ = None
        self.r2_ = None

    def compute_metrics(self, x, y):
        y_pred = self.model.predict(x)
        self.rmse_ = np.sqrt(mean_squared_error(y, y_pred))
        self.r2_ = r2_score(y, y_pred)

def generate_data_set():
    years = np.array([

    ])

    values = np.array([
        1.4, 1.4, 1.426, 1.4257, 1.44, 1.44, 1.4144, 1.7064, 1.7177, 1.4757,
        1.34, 1.5652, 1.617, 1.617, 1.6116, 1.6598, 1.5833, 1.6183, 1.6102,
        1.6711, 1.669, 1.6658, 1.6641, 1.6612, 1.6652, 1.6875, 1.702, 1.7034,
        1.6963, 1.7582, 1.68, 1.5, 1.6786, 1.6667, 1.6563, 1.6364, 1.5, 1.65,
        1.5, 1.1979, 2, 2.98, 0.84, 1.6, 1.2069, 2.6204, 2.6277, 2.6277,
        2.1429, 1.6448, 1.64, 2.1986, 2.1972, 2.4882   
    ])

    x = years[:, np.newaxis]
    y = values[:, np.newaxis]

    return x, y

def train_and_evaluate_model(x_train, y_train, degree=1):
    if degree == 1:
        model = LinearRegression()
        model.fit(x_train, y_train)
        x_transformed = x_train
    else:
        poly_features = PolynomialFeatures(degree=degree, include_bias=False)
        x_transformed = poly_features.fit_transform(x_train)
        model = LinearRegression()
        model.fit(x_transformed, y_train)
    
    linear_model = LinearModel(model)
    linear_model.compute_metrics(x_transformed, y_train)
    
    return linear_model

if __name__ == "__main__":
    x_train, y_train = generate_data_set()

    # Linear Regression
    linear_regression = train_and_evaluate_model(x_train, y_train)
    print(f'The RMSE of the linear regression model is {linear_regression.rmse_}')
    print(f'The R2 score of the linear regression model is {linear_regression.r2_}')

    # Polynomial Regression
    polynomial_regression = train_and_evaluate_model(x_train, y_train, degree=2)
    print(f'The RMSE of the polynomial regression model is {polynomial_regression.rmse_}')
    print(f'The R2 score of the polynomial regression model is {polynomial_regression.r2_}')
