import glob

def find_files(pattern):
    """Search for files matching the given pattern in the current directory."""
    return glob.glob(pattern)

# Search for all .gif files in the current directory
gif_files = find_files('*.gif')
print("GIF files found:", gif_files)

# Search for all files starting with .c in the current directory
c_files = find_files('.c*')
print("Files starting with .c found:", c_files)
