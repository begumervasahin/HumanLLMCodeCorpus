import socket

def create_socket():
    """
    Create and return a TCP/IP socket.
    
    Returns:
        socket.socket: The created TCP/IP socket.
    """
    return socket.socket(socket.AF_INET, socket.SOCK_STREAM)

def connect_socket(sock, ip, port):
    """
    Connect the socket to the given IP address and port.
    
    Args:
        sock (socket.socket): The socket to connect.
        ip (str): The IP address to connect to.
        port (int): The port number to connect to.
    """
    sock.connect((ip, port))
    print(f'Connected to {ip} on port {port}')

def send_message(sock, message):
    """
    Send a message through the socket.
    
    Args:
        sock (socket.socket): The socket to send the message through.
        message (str): The message to send.
    """
    sock.sendall(message.encode())

def receive_message(sock, buffer_size):
    """
    Receive a message from the socket.
    
    Args:
        sock (socket.socket): The socket to receive the message from.
        buffer_size (int): The size of the buffer to use for receiving the message.
    
    Returns:
        str: The received message.
    """
    data = sock.recv(buffer_size)
    return data.decode()

def main():
    """Main function to demonstrate socket communication."""
    # Connection parameters
    TCP_IP = '52.43.121.77'
    TCP_PORT = 2329
    BUFFER_SIZE = 1024
    MESSAGE = "Hello, World!"
    
    # Create and connect the socket
    with create_socket() as sock:
        connect_socket(sock, TCP_IP, TCP_PORT)
        
        # Send the message
        send_message(sock, MESSAGE)
        
        # Receive the response
        data = receive_message(sock, BUFFER_SIZE)
        
        # Print the received data
        print("Received data:", data)

if __name__ == "__main__":
    main()
