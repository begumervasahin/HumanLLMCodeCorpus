import numpy as np

class ReadGraph:
    """
    A class to read a graph matrix from a text file and store it as a NumPy matrix.

    Attributes:
    input_file (str): The path to the text file containing the graph matrix.
    """

    def __init__(self, input_file):
        """
        Initializes the ReadGraph instance with the path to the input file.

        Parameters:
        input_file (str): Path to the text file containing the graph matrix.
        """
        self.input_file = input_file

    def read(self):
        """
        Reads the contents of the file and converts it into a NumPy matrix.

        Returns:
        np.matrix: The graph matrix read from the file.
        """
        # Read the file contents as a string
        with open(self.input_file, 'r') as file:
            graph_string = file.read()

        # Convert the string to a NumPy matrix
        graph_mat = np.matrix(graph_string)

        return graph_mat

# Example usage
if __name__ == '__main__':
    # Define the path to the text file containing the graph matrix
    file_path = 'graph_matrix.txt'
    
    # Create an instance of ReadGraph
    graph_reader = ReadGraph(file_path)
    
    # Read and display the matrix
    graph_matrix = graph_reader.read()
    print("Graph matrix:\n", graph_matrix)
