class HeapItem:
    """
    Heap item data structure.
    """
    def __init__(self, key, value):
        """
        Generates a heap item with key and value.
        """
        self.key = key
        self.value = value
    
    def __repr__(self):
        """
        Represents a heap item.
        """
        return f"({self.key}, {self.value})"


class MinHeap:
    """
    Min-Heap data structure.
    """
    def __init__(self):
        """
        Generates an empty min-heap.
        """
        self.heap = []
        self.index = {}
    
    def key(self, value):
        """
        Returns the key of the heap element with the given value.
        """
        return self.heap[self.index[value]].key
    
    def left(self, i):
        """
        Returns the index of the left child of the i-th item.
        """
        return 2 * i + 1
    
    def right(self, i):
        """
        Returns the index of the right child of the i-th item.
        """
        return 2 * (i + 1)
    
    def parent(self, i):
        """
        Returns the index of the parent of the i-th item.
        """
        return (i - 1) // 2
    
    def min_heapify(self, i):
        """
        Restores the min-heap property at the i-th item, assuming
        its left and right sub-heaps are valid min-heaps.
        """
        l = self.left(i)
        r = self.right(i)
        smallest = i

        if l < len(self.heap) and self.heap[l].key < self.heap[smallest].key:
            smallest = l
        if r < len(self.heap) and self.heap[r].key < self.heap[smallest].key:
            smallest = r
        if smallest != i:
            self.swap(i, smallest)
            self.min_heapify(smallest)
    
    def extract_min(self):
        """
        Extracts the (key, value) pair with the smallest key from the heap.
        """
        if not self:
            return None
        self.swap(0, len(self.heap) - 1)
        min_item = self.heap.pop()
        del self.index[min_item.value]
        self.min_heapify(0)
        return min_item.key, min_item.value
    
    def insert(self, key, value):
        """
        Inserts a (key, value) pair into the heap.
        """
        self.heap.append(HeapItem(key, value))
        self.index[value] = len(self.heap) - 1
        self.decrease_key(value, key)
    
    def decrease_key(self, value, new_key):
        """
        Decreases the key of the heap element with the given value to new_key.
        """
        i = self.index[value]
        self.heap[i].key = new_key
        while i > 0 and self.heap[i].key < self.heap[self.parent(i)].key:
            self.swap(i, self.parent(i))
            i = self.parent(i)
    
    def swap(self, i, j):
        """
        Swaps the i-th and j-th items in the heap.
        """
        self.index[self.heap[i].value], self.index[self.heap[j].value] = j, i
        self.heap[i], self.heap[j] = self.heap[j], self.heap[i]
    
    def __len__(self):
        """
        Returns the number of items in the heap.
        """
        return len(self.heap)
    
    def __bool__(self):
        """
        Returns True if the heap is not empty, False otherwise.
        """
        return bool(self.heap)
    
    def __contains__(self, value):
        """
        Checks whether the value is present in the heap.
        """
        return value in self.index


# Example usage
if __name__ == "__main__":
    heap = MinHeap()
    heap.insert(3, "apple")
    heap.insert(1, "banana")
    heap.insert(2, "cherry")
    
    print("Heap after inserts:", heap.heap)
    print("Extracting min:", heap.extract_min())
    print("Heap after extracting min:", heap.heap)
    heap.decrease_key("cherry", 0)
    print("Heap after decreasing key:", heap.heap)
