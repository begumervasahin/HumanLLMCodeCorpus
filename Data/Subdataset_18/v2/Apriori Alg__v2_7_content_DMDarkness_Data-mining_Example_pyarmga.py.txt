import pyarmga as ga  # Import the package of BPSO-HD
import re  # Import the regular expression package

def read_transactional_dataset(file_name):
    """
    Reads a transactional dataset from a file and returns a list of transactions.
    Each transaction is a list of item IDs (integers).

    Args:
    file_name (str): The path to the transactional dataset file.

    Returns:
    list: A list of transactions, where each transaction is a list of integers.
    """
    with open(file_name, 'r') as file:
        content = file.readlines()

    transactions = []
    for line in content:
        items = re.split(r' |\n', line)
        transaction = [int(item) for item in items if item.isdigit()]
        if transaction:
            transactions.append(transaction)
    
    return transactions

def mine_association_rules(dataset, min_confidence, min_lift, population_size, generations, mutation_prob, selection_prob, crossover_prob, rule_length):
    """
    Mines association rules from the dataset using the pyarmga library.

    Args:
    dataset (list): A list of transactions.
    min_confidence (float): Minimum confidence for the rules.
    min_lift (float): Minimum lift for the rules.
    population_size (int): Size of the population.
    generations (int): Number of generations.
    mutation_prob (float): Mutation probability.
    selection_prob (float): Selection probability.
    crossover_prob (float): Crossover probability.
    rule_length (int): Length of the rules.

    Returns:
    list: A list of association rules.
    """
    return ga.getAR(dataset, min_confidence, min_lift, population_size, generations, mutation_prob, selection_prob, crossover_prob, rule_length)

def main():
    # Path to the transactional dataset
    dataset_path = "kosarak.dat"
    
    # Read the transactional dataset
    dataset = read_transactional_dataset(dataset_path)

    # Parameters for mining association rules
    min_confidence = 0.7
    min_lift = 1
    population_size = 30
    generations = 30
    mutation_prob = 0.25
    selection_prob = 1
    crossover_prob = 1
    rule_length = 10

    # Mine association rules from the dataset
    association_rules = mine_association_rules(
        dataset,
        min_confidence,
        min_lift,
        population_size,
        generations,
        mutation_prob,
        selection_prob,
        crossover_prob,
        rule_length
    )

    # Print the association rules
    for rule in association_rules:
        print(rule)

if __name__ == "__main__":
    main()
