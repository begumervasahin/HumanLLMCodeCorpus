import nltk
import numpy as np
import matplotlib.pyplot as plt
from nltk.stem import WordNetLemmatizer
from sklearn.decomposition import TruncatedSVD

# Initialize the lemmatizer
wordnet_lemmatizer = WordNetLemmatizer()

# Read book titles from file
with open('all_book_titles.txt', 'r') as f:
    titles = [line.rstrip() for line in f]

# Read stopwords from file
with open('stopwords.txt', 'r') as f:
    stopwords = set(w.rstrip() for w in f)

# Add custom stopwords
custom_stopwords = {
    'introduction', 'edition', 'series', 'application',
    'approach', 'card', 'access', 'package', 'plus', 'etext',
    'brief', 'vol', 'fundamental', 'guide', 'essential', 'printed',
    'third', 'second', 'fourth',
}
stopwords = stopwords.union(custom_stopwords)

# Tokenizer function
def my_tokenizer(s):
    s = s.lower()
    tokens = nltk.tokenize.word_tokenize(s)
    tokens = [t for t in tokens if len(t) > 2]
    tokens = [wordnet_lemmatizer.lemmatize(t) for t in tokens]
    tokens = [t for t in tokens if t not in stopwords]
    tokens = [t for t in tokens if not any(c.isdigit() for c in t)]
    return tokens

# Initialize variables
word_index_map = {}
current_index = 0
all_tokens = []
all_titles = []
index_word_map = []

# Process each title
for title in titles:
    try:
        # Encode to ASCII and decode to UTF-8 to handle special characters
        title = title.encode('ascii', 'ignore').decode('utf-8')
        all_titles.append(title)
        
        # Tokenize the title
        tokens = my_tokenizer(title)
        all_tokens.append(tokens)
        
        # Create a mapping of tokens to indices
        for token in tokens:
            if token not in word_index_map:
                word_index_map[token] = current_index
                current_index += 1
                index_word_map.append(token)
    except Exception as e:
        print(f"Error processing title: {title} - {e}")

# Function to convert tokens to vector
def tokens_to_vector(tokens):
    x = np.zeros(len(word_index_map))
    for t in tokens:
        if t in word_index_map:  # Check if token exists in the word_index_map
            i = word_index_map[t]
            x[i] = 1
    return x

# Create the matrix X
N = len(all_tokens)
D = len(word_index_map)
X = np.zeros((D, N))

# Fill the matrix X with token vectors
for i, tokens in enumerate(all_tokens):
    X[:, i] = tokens_to_vector(tokens)

# Perform Truncated SVD
svd = TruncatedSVD(n_components=2)
Z = svd.fit_transform(X)

# Plot the results
plt.figure(figsize=(10, 8))
plt.scatter(Z[:, 0], Z[:, 1])

# Annotate the plot with token labels
for i in range(D):
    plt.annotate(s=index_word_map[i], xy=(Z[i, 0], Z[i, 1]))

plt.title('Truncated SVD of Book Titles')
plt.xlabel('Component 1')
plt.ylabel('Component 2')
plt.grid(True)
plt.show()
