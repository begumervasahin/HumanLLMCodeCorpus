import random
from util_function import strxor  # Assuming strxor is defined in util_function.py

def read_input():
    """
    Reads a number of messages from the user and returns them as a list.
    
    Returns:
        list: A list of input messages.
    """
    message_num = int(input("Enter the number of messages: "))
    message_set = [input("Enter message: ") for _ in range(message_num)]
    return message_set

def encrypt(key, message):
    """
    Encrypts the message using the provided key with the XOR operation.
    
    Args:
        key (str): The encryption key.
        message (str): The message to encrypt.
    
    Returns:
        str: The encrypted message.
    """
    return strxor(key, message)

def print_single_encrypted(cipher_text):
    """
    Prints the encrypted message in hexadecimal format.
    
    Args:
        cipher_text (str): The encrypted message.
    """
    cipher_text_hex = "".join(f"{ord(char):02x}" for char in cipher_text)
    print(cipher_text_hex)

def generate_random_key(length):
    """
    Generates a random key of the given length.
    
    Args:
        length (int): The length of the key.
    
    Returns:
        str: The generated random key.
    """
    random_ascii = [chr(random.randint(0, 255)) for _ in range(length)]
    return ''.join(random_ascii)

def main():
    """
    Main function to handle the encryption process.
    """
    # Read input messages from the user
    message_set = read_input()
    message_num = len(message_set)
    
    # Determine the length of the key based on the longest message
    max_size = len(max(message_set, key=len))
    
    # Generate a random key
    key = generate_random_key(max_size)
    
    print("Number of messages:", message_num)
    
    # Encrypt each message and print the result
    for message in message_set:
        cipher_text = encrypt(key, message)
        print_single_encrypted(cipher_text)

if __name__ == "__main__":
    main()
