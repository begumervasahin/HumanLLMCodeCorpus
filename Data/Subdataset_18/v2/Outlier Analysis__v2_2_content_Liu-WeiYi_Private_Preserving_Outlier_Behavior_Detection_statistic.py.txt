import json
import sys

def load_users(filename):
    """Load user IDs from a file."""
    with open(filename, 'r') as f:
        return [line.strip() for line in f.readlines()]

def process_devices_info(user_info_file):
    """Process and display the number of unique devices used by all users."""
    print('Processing devices group...')
    with open(user_info_file) as f:
        all_user_info = json.load(f)

    devices = set()
    total_users = len(all_user_info)

    for count, user in enumerate(all_user_info.keys(), start=1):
        percentage = 100 * count / total_users
        sys.stdout.write('\r>> Processing Users............ %.2f %%' % percentage)
        sys.stdout.flush()

        reach_time_info = all_user_info[user]["Reach_Time"]
        for time in reach_time_info.keys():
            current_devices = reach_time_info[time].keys()
            devices.update(current_devices)

    num_devices = len(devices)
    print(f'\nTotal number of devices: {num_devices}')

def process_features_info(user_info_file):
    """Process and display total and average number of keywords and IP addresses per user."""
    print('Processing features...')
    with open(user_info_file) as f:
        all_user_info = json.load(f)

    total_keywords = 0
    total_ips = 0
    total_users = len(all_user_info)

    for count, user in enumerate(all_user_info.keys(), start=1):
        percentage = 100 * count / total_users
        sys.stdout.write('\r>> Processing Users............ %.2f %%' % percentage)
        sys.stdout.flush()

        reach_time_info = all_user_info[user]["Reach_Time"]
        for time in reach_time_info.keys():
            for user_detail in reach_time_info[time].keys():
                total_keywords += len(reach_time_info[time][user_detail]["Keyword"])
                total_ips += len(reach_time_info[time][user_detail]["Device_IP"])

    avg_keywords = total_keywords / total_users
    avg_ips = total_ips / total_users

    print(f'\nTotal keywords: {total_keywords}')
    print(f'Average keywords per user: {avg_keywords:.2f}')
    print(f'Total IP addresses: {total_ips}')
    print(f'Average IP addresses per user: {avg_ips:.2f}')

def main():
    """Main function to process user information based on the provided argument."""
    if len(sys.argv) < 2:
        print("Please provide an argument (0, 1, or 2).")
