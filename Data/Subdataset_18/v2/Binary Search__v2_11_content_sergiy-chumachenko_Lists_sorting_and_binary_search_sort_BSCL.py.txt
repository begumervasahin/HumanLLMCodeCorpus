class MySorting:
    def __init__(self, nums):
        """Initialize the MySorting class with a list of numbers."""
        self.nums = nums

    def __str__(self):
        """Return a string representation of the sorting object."""
        return f"MySorting instance with the list: {self.nums}"

    def asc(self):
        """Sort the list in ascending order using bubble sort."""
        n = len(self.nums)
        for i in range(n):
            swapped = False
            for j in range(n - i - 1):
                if self.nums[j] > self.nums[j + 1]:
                    self.nums[j], self.nums[j + 1] = self.nums[j + 1], self.nums[j]
                    swapped = True
            if not swapped:
                break
        return self.nums

    def desc(self):
        """Sort the list in descending order using bubble sort."""
        n = len(self.nums)
        for i in range(n):
            swapped = False
            for j in range(n - i - 1):
                if self.nums[j] < self.nums[j + 1]:
                    self.nums[j], self.nums[j + 1] = self.nums[j + 1], self.nums[j]
                    swapped = True
            if not swapped:
                break
        return self.nums

def binary_search(lst, n):
    """
    Perform binary search to find the index where the element n should be inserted.

    Args:
        lst (list): The sorted list where the search is performed.
        n (int): The number to search for or insert.

    Returns:
        int: The index where n should be inserted to maintain sorted order.
    """
    left, right = 0, len(lst) - 1
    while left <= right:
        middle = (left + right) // 2
        if lst[middle] < n:
            left = middle + 1
        elif lst[middle] > n:
            right = middle - 1
        else:
            return middle
    return left

def insert_in_sorted_list(n, lst):
    """
    Insert a number n into the sorted list lst, maintaining the sorted order.

    Args:
        n (int): The number to insert.
        lst (list): The sorted list to insert the number into.

    Returns:
        list: The updated list with n inserted.
    """
    if not lst:
        lst.append(n)
    else:
        index = binary_search(lst, n)
        lst.insert(index, n)
    return lst

# Example usage:
if __name__ == "__main__":
    nums = [3, 1, 4, 1, 5, 9, 2, 6, 5, 3, 5]
    sorter = MySorting(nums)

    print("Original list:", nums)
    print("Sorted list (ascending):", sorter.asc())
    print("Sorted list (descending):", sorter.desc())

    # Inserting into a sorted list
    sorted_list = [1, 3, 5, 7, 9]
    new_number = 6
    print(f"Inserting {new_number} into {sorted_list}:")
    updated_list = insert_in_sorted_list(new_number, sorted_list)
    print("Updated list:", updated_list)
