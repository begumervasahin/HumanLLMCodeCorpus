import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
import scipy.cluster.hierarchy as sch
from sklearn.cluster import AgglomerativeClustering

# Load the dataset
def load_dataset(file_path):
    """
    Load the dataset and extract relevant features.

    Args:
        file_path (str): Path to the CSV file.

    Returns:
        np.ndarray: Extracted features (Annual Income and Spending Score).
    """
    dataset = pd.read_csv(file_path)
    X = dataset.iloc[:, [3, 4]].values  # Extract Annual Income and Spending Score
    return X

# Plot the dendrogram
def plot_dendrogram(X):
    """
    Plot the dendrogram to find the optimal number of clusters.

    Args:
        X (np.ndarray): The data to be used for plotting the dendrogram.
    """
    plt.figure(figsize=(10, 7))
    dendrogram = sch.dendrogram(sch.linkage(X, method='ward'))
    plt.title('Dendrogram')
    plt.xlabel('Customers')
    plt.ylabel('Euclidean Distance')
    plt.show()

# Perform Agglomerative Clustering
def perform_clustering(X, n_clusters=5):
    """
    Apply Agglomerative Clustering to the dataset.

    Args:
        X (np.ndarray): The data to be clustered.
        n_clusters (int): The number of clusters.

    Returns:
        np.ndarray: Cluster labels for each point in the dataset.
    """
    hc = AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage='ward')
    y_hc = hc.fit_predict(X)
    return y_hc

# Visualize the clusters
def visualize_clusters(X, y_hc):
    """
    Visualize the clusters formed by Agglomerative Clustering.

    Args:
        X (np.ndarray): The data used for clustering.
        y_hc (np.ndarray): Cluster labels for the data points.
    """
    plt.figure(figsize=(10, 7))
    plt.scatter(X[y_hc == 0, 0], X[y_hc == 0, 1], s=100, c='red', label='Careful')
    plt.scatter(X[y_hc == 1, 0], X[y_hc == 1, 1], s=100, c='blue', label='Standard')
    plt.scatter(X[y_hc == 2, 0], X[y_hc == 2, 1], s=100, c='green', label='Targets')
    plt.scatter(X[y_hc == 3, 0], X[y_hc == 3, 1], s=100, c='cyan', label='Careless')
    plt.scatter(X[y_hc == 4, 0], X[y_hc == 4, 1], s=100, c='magenta', label='Sensible')
    plt.title('Clusters of Clients')
    plt.xlabel('Annual Income (k$)')
    plt.ylabel('Spending Score (1-100)')
    plt.legend()
    plt.show()

def main():
    # File path to the dataset
    file_path = 'Mall_Customers.csv'

    # Load dataset and extract features
    X = load_dataset(file_path)

    # Plot the dendrogram
    plot_dendrogram(X)

    # Perform Agglomerative Clustering
    y_hc = perform_clustering(X, n_clusters=5)

    # Visualize the clusters
    visualize_clusters(X, y_hc)

if __name__ == "__main__":
    main()
