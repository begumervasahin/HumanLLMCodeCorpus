def missing_number(nums):
    """
    Given a list of numbers containing n distinct numbers in the range [0, n], find the missing number.
    :param nums: List[int]
    :return: int
    """
    l = 0  # Lower bound of the range
    u = max(nums)  # Upper bound of the range
    expected_length = u - l + 1  # Expected number of elements in the range
    print(l, u, expected_length)  # Debugging statement to print range bounds and expected length
    
    i = 0
    while expected_length != 0:
        current_num = l + i
        if current_num in nums:
            i += 1
            expected_length -= 1
        else:
            return current_num
    
    return l + i

# Example usage:
nums = [0]
print(missing_number(nums))  # Output should be 1

# Additional test cases
nums = [3, 0, 1]
print(missing_number(nums))  # Output should be 2

nums = [0, 1]
print(missing_number(nums))  # Output should be 2

nums = [9, 6, 4, 2, 3, 5, 7, 0, 1]
print(missing_number(nums))  # Output should be 8
