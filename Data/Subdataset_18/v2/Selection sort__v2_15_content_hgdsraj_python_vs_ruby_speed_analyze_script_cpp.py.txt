# Open the file 'cpp_res.txt' and read all lines
with open('cpp_res.txt') as file:
    lines = file.readlines()

# Initialize an empty list to store the extracted values
extracted_values = []

# Loop through each line in the file
for line in lines:
    # Check if the line contains the word "real"
    if "real" in line:
        # Extract the value from the line and append it to the list
        extracted_values.append(line[9:12])

# Initialize a variable to store the sum of the values
total_sum = 0

# Loop through each value in the list and add it to the sum
for value in extracted_values:
    total_sum += float(value) / 1000

# Calculate the average of the values
average = total_sum / len(extracted_values)  # Using len(extracted_values) to handle dynamic length

# Print the average
print("The average was:")
print(f"{average} seconds")
