class Node:
    """Node class for a singly linked list."""
    
    def __init__(self, data=None):
        """
        Initialize a node with data and a pointer to the next node.
        
        Args:
            data (any): The data to be stored in the node.
        """
        self.data = data
        self.next = None

    def __repr__(self):
        """String representation of the node."""
        return f"Node({self.data})"

class LinkedList:
    """Class to represent a singly linked list."""
    
    def __init__(self):
        """Initialize an empty linked list with head and tail pointers."""
        self.head = None
        self.tail = None

    def is_empty(self):
        """
        Check if the linked list is empty.
        
        Returns:
            bool: True if the list is empty, False otherwise.
        """
        return self.head is None

    def append(self, item):
        """
        Append a new item to the end of the linked list.
        
        Args:
            item (any): The data to append to the list.
        """
        new_node = Node(item)
        if self.is_empty():
            self.head = new_node
            self.tail = new_node
        else:
            self.tail.next = new_node
            self.tail = new_node

    def size(self):
        """
        Get the number of elements in the linked list.
        
        Returns:
            int: The number of elements in the list.
        """
        count = 0
        current = self.head
        while current:
            count += 1
            current = current.next
        return count

    def search(self, item):
        """
        Search for an item in the linked list.
        
        Args:
            item (any): The data to search for in the list.
        
        Returns:
            bool: True if the item is found, False otherwise.
        """
        current = self.head
        while current:
            if current.data == item:
                return True
            current = current.next
        return False

    def remove(self, item):
        """
        Remove the first occurrence of an item from the linked list.
        
        Args:
            item (any): The data to remove from the list.
        """
        current = self.head
        previous = None
        found = False

        while current and not found:
            if current.data == item:
                found = True
            else:
                previous = current
                current = current.next

        if found:
            if previous is None:  # Removing the head node
                self.head = current.next
            else:
                previous.next = current.next

            if current == self.tail:  # Removing the tail node
                self.tail = previous

    def display(self):
        """Display the elements of the linked list."""
        current = self.head
        while current:
            print(current.data, end=" -> " if current.next else "\n")
            current = current.next

# Example usage
if __name__ == "__main__":
    linked_list = LinkedList()
    linked_list.append(1)
    linked_list.append(2)
    linked_list.append(3)
    
    linked_list.display()  # Output: 1 -> 2 -> 3
    
    print("Size:", linked_list.size())  # Output: Size: 3
    
    print("Search 2:", linked_list.search(2))  # Output: Search 2: True
    print("Search 5:", linked_list.search(5))  # Output: Search 5: False
    
    linked_list.remove(2)
    linked_list.display()  # Output: 1 -> 3
