import itertools

def generate_combinations(length):
    """
    Generate all binary combinations of a given length.

    Args:
        length (int): The length of the binary strings to generate.

    Returns:
        list: A list of binary strings.
    """
    # Generate all possible binary combinations of the specified length
    combinations = [''.join(map(str, bits)) for bits in itertools.product([0, 1], repeat=length)]
    return combinations

def calculate_crc(message, divisor):
    """
    Calculate the CRC of a given binary message using a binary divisor.

    Args:
        message (str): The binary message.
        divisor (str): The binary divisor.

    Returns:
        str: The CRC result as a binary string.
    """
    # Add padding to the message
    padding = '0' * (len(divisor) - 1)
    padded_message = list(message + padding)
    divisor = list(divisor)

    # Perform the CRC calculation
    for i in range(len(padded_message) - len(padding)):
        if padded_message[i] == '1':
            for j in range(len(divisor)):
                padded_message[i + j] = str(int(padded_message[i + j]) ^ int(divisor[j]))

    # Return the CRC result
    return ''.join(padded_message[-(len(divisor) - 1):])

def find_crc_collisions(message, divisor, prefix_list=None):
    """
    Find all binary combinations that produce the same CRC as the given message.

    Args:
        message (str): The binary message.
        divisor (str): The binary divisor.
        prefix_list (list, optional): A list of prefix strings to include in the search.

    Returns:
        list: A list of binary strings that produce the same CRC as the given message.
    """
    # Compute the CRC of the original message
    target_crc = calculate_crc(message, divisor)
    # Generate all possible combinations of the same length as the divisor
    all_combinations = generate_combinations(len(divisor))
    collisions = []

    # Generate all possible prefixes if not provided
    if prefix_list is None:
        prefix_list = generate_combinations(len(message) - len(divisor))

    # Check each prefix and combination for matching CRC
    for prefix in prefix_list:
        for combination in all_combinations:
            test_message = prefix + combination
            if calculate_crc(test_message, divisor) == target_crc:
                collisions.append(test_message)

    return collisions

# Example usage
if __name__ == "__main__":
    # Define the message and divisor
    message = "11010110"
    divisor = "10011"
    prefixes = ["111"]

    # Find and print all collisions
    collisions = find_crc_collisions(message, divisor, prefixes)
    print("Collisions:")
    for collision in collisions:
        print(collision)
