def bubble_sort(arr):
    """
    Sorts an array using the bubble sort algorithm.
    
    Parameters:
    arr (list): The list to be sorted.
    
    Returns:
    list: The sorted list.
    """
    while True:
        swapped = False
        for i in range(len(arr) - 1):
            if arr[i] > arr[i + 1]:
                arr[i], arr[i + 1] = arr[i + 1], arr[i]
                swapped = True
        if not swapped:
            break
    return arr

# Test the bubble_sort function
print(bubble_sort([1, 2, 3, 4, 5, 6]))  # Already sorted list
print(bubble_sort([2, 1, 4, 3, 6, 5]))  # Unsorted list
print(bubble_sort([6, 5, 4, 3, 2, 1]))  # Reverse sorted list
