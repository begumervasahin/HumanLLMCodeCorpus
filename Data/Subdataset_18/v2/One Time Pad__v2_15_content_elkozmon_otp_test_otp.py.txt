def encrypt(plaintext, key, alphabet):
    """
    Encrypt the plaintext using the One-Time Pad (OTP) encryption method.

    Args:
        plaintext (str): The message to be encrypted.
        key (str): The key for the encryption.
        alphabet (str): The alphabet to use for encryption.

    Returns:
        str: The encrypted message (ciphertext).
    """
    ciphertext = []
    alphabet_size = len(alphabet)

    for p, k in zip(plaintext, key):
        p_index = alphabet.index(p)
        k_index = alphabet.index(k)
        c_index = (p_index + k_index) % alphabet_size
        ciphertext.append(alphabet[c_index])
    
    return ''.join(ciphertext)

def decrypt(ciphertext, key, alphabet):
    """
    Decrypt the ciphertext using the One-Time Pad (OTP) decryption method.

    Args:
        ciphertext (str): The encrypted message.
        key (str): The key for the decryption.
        alphabet (str): The alphabet to use for decryption.

    Returns:
        str: The decrypted message (plaintext).
    """
    plaintext = []
    alphabet_size = len(alphabet)

    for c, k in zip(ciphertext, key):
        c_index = alphabet.index(c)
        k_index = alphabet.index(k)
        p_index = (c_index - k_index) % alphabet_size
        plaintext.append(alphabet[p_index])
    
    return ''.join(plaintext)

import unittest

class OtpTestCase(unittest.TestCase):
    """Tests for the One-Time Pad (OTP) encryption and decryption functions."""

    def test_encrypt_hello(self):
        """Test encrypting 'HELLO' with key 'XMCKL' results in ciphertext 'EQNVZ'."""
        self.assertEqual(encrypt("HELLO", "XMCKL", "ABCDEFGHIJKLMNOPQRSTUVWXYZ"), "EQNVZ")

    def test_decrypt_hello(self):
        """Test decrypting 'EQNVZ' with key 'XMCKL' results in plaintext 'HELLO'."""
        self.assertEqual(decrypt("EQNVZ", "XMCKL", "ABCDEFGHIJKLMNOPQRSTUVWXYZ"), "HELLO")

if __name__ == '__main__':
    unittest.main()
