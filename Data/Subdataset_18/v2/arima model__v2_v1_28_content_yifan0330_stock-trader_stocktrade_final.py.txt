import stocktrader

# Load the portfolio from a CSV file
portfolio_file = 'portfolio.csv'
stocktrader.loadPortfolio(portfolio_file)

# Load historical stock data for all stocks
stocktrader.loadAllStocks()

# Define the date for portfolio valuation
valuation_date = '2024-03-28'

# Evaluate the total value of the portfolio on the specified date
total_value = stocktrader.valuatePortfolio(date=valuation_date, verbose=True)
print(f"Total portfolio value on {valuation_date}: {total_value:.2f}")

# Execute trading strategies

# Strategy 1: Buy as much as possible and then sell everything
print("Executing Strategy 1: Buy as much as possible and sell all")
stocktrader.tradeStrategy1(verbose=True)

# Strategy 2: Buy stocks predicted to increase in price and sell those predicted to decrease
print("Executing Strategy 2: Buy predicted gainers and sell predicted losers")
stocktrader.tradeStrategy2(verbose=True)

# Save the updated portfolio to a new CSV file
updated_portfolio_file = 'updated_portfolio.csv'
stocktrader.savePortfolio(updated_portfolio_file)

print(f"Updated portfolio saved to {updated_portfolio_file}")
