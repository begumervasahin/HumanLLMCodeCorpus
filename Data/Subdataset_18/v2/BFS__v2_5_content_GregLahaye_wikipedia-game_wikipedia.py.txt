import requests

# Base URL for the Wikipedia API
API_URL = "https://en.wikipedia.org/w/api.php"
s = requests.session()

def random(num=1):
    """
    Get random Wikipedia page titles.

    Args:
        num (int): Number of random titles to retrieve. Default is 1.

    Returns:
        list: A list of random Wikipedia page titles.
    """
    params = {
        "format": "json",
        "action": "query",
        "generator": "random",
        "grnnamespace": "0",
        "grnlimit": num
    }
    
    response = s.get(API_URL, params=params).json()
    titles = [item["title"] for item in response["query"]["pages"].values()]
    return titles

def check(title):
    """
    Check if a Wikipedia page title is valid and not a disambiguation page.

    Args:
        title (str): The Wikipedia page title to check.

    Returns:
        str or bool: The valid title if the page is valid, otherwise False.
    """
    params = {
        "action": "query",
        "titles": title,
        "prop": "categories",
        "clcategories": "Category:All disambiguation pages",
        "format": "json",
        "redirects": "true"
    }
    
    response = s.get(API_URL, params=params).json()
    pageid = list(response["query"]["pages"].keys())[0]
    valid = False
    
    if "redirects" in response["query"]:
        valid = response["query"]["redirects"][0]["to"]
    elif "categories" in response["query"]["pages"][pageid]:
        print(f"'{title}' is a disambiguation page")
    elif pageid == "-1": 
        print(f"'{title}' is not a valid article")
    else:
        valid = title

    return valid

def get_links(title):
    """
    Retrieve all linked page titles from a given Wikipedia page.

    Args:
        title (str): The Wikipedia page title.

    Returns:
        list: A list of linked Wikipedia page titles.
    """
    params = {
        "action": "query",
        "titles": title,
        "prop": "links",
        "pllimit": "max",
        "format": "json",
        "plnamespace": 0
    }

    links = []
    done = False

    while not done:
        try:
            response = s.get(API_URL, params=params).json()
            pageid = list(response["query"]["pages"].keys())[0]

            if "links" in response["query"]["pages"][pageid]:
                links += [link["title"] for link in response["query"]["pages"][pageid]["links"]]

                # Handle pagination
                while "continue" in response:
                    params["plcontinue"] = response["continue"]["plcontinue"]    
                    response = s.get(API_URL, params=params).json()
                    pageid = list(response["query"]["pages"].keys())[0]
                    links += [link["title"] for link in response["query"]["pages"][pageid]["links"]]

            done = True
        except requests.exceptions.ConnectionError:
            input("Connection Error, <Enter> to try again")
    
    return links    

if __name__ == "__main__":
    # Example usage of the functions
    print("Random Wikipedia page:", random())
    page = input("Enter a Wikipedia page title to check: ")
    print("Checking page:", check(page))
    print("Links on the page:", get_links(page))
