class Graph:
    """
    Represents an undirected graph using an adjacency list.

    Attributes:
        num_vertices (int): The number of vertices in the graph.
        adj_list (dict): Adjacency list where key is a vertex and value is a list of adjacent vertices.
        path (list): List of vertices visited during the Depth-First Search (DFS) traversal.
    """

    def __init__(self, num_vertices):
        """
        Initializes the graph with a given number of vertices and sets up an initial graph structure.

        Args:
            num_vertices (int): Number of vertices in the graph.
        """
        self.num_vertices = num_vertices
        self.adj_list = {i: [] for i in range(num_vertices)}
        self.path = []
        self._initialize_graph()

    def _initialize_graph(self):
        """
        Populates the graph with example edges for demonstration purposes.
        """
        for i in range(self.num_vertices):
            if i < self.num_vertices - 1:
                self.add_edge(i, i + 1)
            if i % 2 == 0 and i + 2 < self.num_vertices:
                self.add_edge(i, i + 2)

    def add_edge(self, u, v):
        """
        Adds an undirected edge between vertices u and v.

        Args:
            u (int): The start vertex.
            v (int): The end vertex.
        """
        self.adj_list[u].append(v)
        self.adj_list[v].append(u)

    def dfs(self):
        """
        Performs Depth-First Search (DFS) starting from vertex 0 and records the traversal path.
        """
        visited = set()
        self.path = []

        def _dfs(v):
            visited.add(v)
            self.path.append(v)
            for neighbor in self.adj_list[v]:
                if neighbor not in visited:
                    _dfs(neighbor)

        # Start DFS from vertex 0
        if self.num_vertices > 0:
            _dfs(0)

# Main execution
if __name__ == "__main__":
    # Create a graph with 10 vertices
    graph = Graph(10)
    
    # Print the adjacency list of the graph
    print("Adjacency List:", graph.adj_list)
    
    # Perform Depth-First Search
    graph.dfs()
    
    # Print the DFS traversal path
    print("DFS Path:", graph.path)
