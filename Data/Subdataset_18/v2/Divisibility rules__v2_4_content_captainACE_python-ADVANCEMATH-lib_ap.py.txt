def calculate_last_term_of_ap():
    """
    This function calculates the last number in an arithmetic progression (AP).
    It prompts the user to enter the first term, common difference, and number of terms.
    Then, it computes and prints the last term of the AP.
    """
    print("Welcome to the ARITHMETIC PROGRESSION module.")
    print("This module helps you find the last number in a sequence of numbers.")
    print("To determine the last number, you need to enter the first term, the common difference, and the number of terms in the AP.")

    while True:
        try:
            a = int(input("Please enter the first term of the sequence: "))
            d = int(input("Please enter the common difference of the sequence: "))
            n = int(input("Please enter the number of terms: "))

            last_term = a + (n - 1) * d
            print(f"The last term of the arithmetic progression is: {last_term}")
            break
        except ValueError:
            print("Invalid input. Please enter valid integer values for the first term, common difference, and number of terms.")
        except Exception as e:
            print(f"An error occurred: {e}. Please try again.")

if __name__ == "__main__":
    calculate_last_term_of_ap()
