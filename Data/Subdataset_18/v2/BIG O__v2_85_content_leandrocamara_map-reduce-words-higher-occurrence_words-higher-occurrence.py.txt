import csv
import glob
import sys
import mincemeat
import os

# Path to the directory containing the data files
DATA_PATH = '/home/leandro/Documents/puc/06-solutions/activities/map-reduce-words-higher-occurrence/data'

# Get all text files in the specified directory
text_files = glob.glob(os.path.join(DATA_PATH, '*'))

def file_contents(file_name):
    """Reads the contents of the given file."""
    with open(file_name, 'r') as f:
        return f.read()

def mapfn(k, v):
    """
    Map function to process each line and emit word-author pairs.
    
    Args:
        k: Key (file name)
        v: Value (file contents)
    
    Yields:
        word, author: Pairs of words and their respective authors.
    """
    from stopwords import allStopWords
    for line in v.splitlines():
        fields = line.split(':::')
        authors = fields[1]
        title = fields[2]
        
        for word in title.split():
            word = word.strip('.,:?"()')
            if word.lower() not in allStopWords:
                for author in authors.split('::'):
                    if author in ['Grzegorz Rozenberg', 'Philip S. Yu']:
                        print(f'author: {author}; title: {title}')
                        yield word, author

def reducefn(k, v):
    """
    Reduce function to aggregate the results and filter by occurrence count.
    
    Args:
        k: Key (word)
        v: Values (list of authors)
    
    Returns:
        List of authors with occurrence counts if they exceed the threshold, else None.
    """
    author_counts = {'Grzegorz Rozenberg': 0, 'Philip S. Yu': 0}
    
    for author in v:
        if author in author_counts:
            author_counts[author] += 1
    
    result = []
    for author, count in author_counts.items():
        if count > 5:
            result.append(f'{author}: {count}')
    
    return result if result else None

# Create a data source dictionary mapping file names to their contents
source = {file_name: file_contents(file_name) for file_name in text_files}

# Set up the MapReduce server
server = mincemeat.Server()
server.datasource = source
server.mapfn = mapfn
server.reducefn = reducefn

# Run the server and get the results
results = server.run_server(password="mapreduce")

# Write the results to a CSV file
output_file = os.path.join(DATA_PATH, 'result.csv')
with open(output_file, 'w', newline='') as csvfile:
    writer = csv.writer(csvfile)
    for word, authors in results.items():
        if authors:
            writer.writerow([word, ', '.join(authors)])
