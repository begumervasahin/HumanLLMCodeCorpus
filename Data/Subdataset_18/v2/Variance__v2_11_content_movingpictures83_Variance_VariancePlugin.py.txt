import numpy as np

class VariancePlugin:
    def __init__(self):
        self.bacteria = []
        self.ADJ = []

    def input(self, filename):
        """Reads the input file name and stores it."""
        self.myfile = filename

    def run(self):
        """Processes the input file to extract bacteria names and their corresponding values."""
        with open(self.myfile, 'r') as file:
            # Read the first line to get bacteria names
            firstline = file.readline().strip()
            self.bacteria = [item for item in firstline.split(',') if item != '\"\"']
            self.n = len(self.bacteria)
            self.ADJ = [[] for _ in range(self.n)]

            # Read subsequent lines to get the data values
            for line in file:
                contents = line.strip().split(',')
                for j in range(1, len(contents)):
                    value = float(contents[j])
                    self.ADJ[j-1].append(value)

    def output(self, filename):
        """Calculates the variance for each bacteria and writes the results to the output file."""
        variances = []

        # Calculate variance for each set of values
        for i in range(self.n):
            if len(self.ADJ[i]) > 0:
                values = np.array(self.ADJ[i])
                variance = np.var(values)
                variances.append((variance, self.bacteria[i]))

        # Sort the variances in descending order
        variances.sort(reverse=True)

        # Write the results to the output file
        with open(filename, 'w') as file:
            file.write("Element\tVariance\n\n")
            for variance, name in variances:
                file.write(f"{name}\t{variance:.6f}\n")
