def is_prime(n):
    """
    Check if a number is prime.

    Args:
        n (int): The number to check.

    Returns:
        bool: True if the number is prime, False otherwise.
    """
    if n <= 1:
        return False
    for x in range(2, n):
        if n % x == 0:
            return False
    return True

def process_number(num):
    """
    Perform various checks and print results based on the input number.

    Args:
        num (int): The input number.
    """
    results = []
    if num % 3 == 0:
        results.append("Buzz")
    if num % 5 == 0:
        results.append("Fizz")
    if num % 15 == 0:
        results.append("FizzBuzz")
    if is_prime(num):
        results.append("BuzzFizz")
    
    if results:
        for item in results:
            print(item)
    else:
        fibonacci_sequence = generate_fibonacci(num)
        print(fibonacci_sequence)

def generate_fibonacci(n):
    """
    Generate a Fibonacci sequence of n numbers.

    Args:
        n (int): The number of Fibonacci numbers to generate.

    Returns:
        list: A list containing the Fibonacci sequence.
    """
    sequence = []
    a, b = 0, 1
    for i in range(n):
        sequence.append(a)
        a, b = b, a + b
    return sequence

    n = int(input("How many Fibonacci numbers do you want to see? "))
    process_number(n)
