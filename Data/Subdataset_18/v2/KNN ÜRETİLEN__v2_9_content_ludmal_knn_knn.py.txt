import csv
import random
import math
import operator

def load_dataset(filename, split_ratio, signal=4):
    """
    Load the dataset from a CSV file and split it into training and testing sets.
    
    Args:
        filename (str): The name of the CSV file.
        split_ratio (float): The ratio to split the dataset into training and testing sets.
        signal (int): The number of features in the dataset.
    
    Returns:
        tuple: A tuple containing the training and testing sets.
    """
    train_set, test_set = [], []
    
    with open(filename, 'r') as csvfile:
        lines = csv.reader(csvfile)
        dataset = list(lines)
        
        for row in dataset[:-1]:
            for i in range(signal):
                row[i] = float(row[i])
            if random.random() < split_ratio:
                train_set.append(row)
            else:
                test_set.append(row)
    
    return train_set, test_set

def calculate_distance(instance1, instance2, length):
    """
    Calculate the Euclidean distance between two instances.
    
    Args:
        instance1 (list): The first instance.
        instance2 (list): The second instance.
        length (int): The number of features to consider.
    
    Returns:
        float: The Euclidean distance between the two instances.
    """
    return math.sqrt(sum(pow(instance1[i] - instance2[i], 2) for i in range(length)))

def get_neighbors(train_set, test_instance, k):
    """
    Get the k nearest neighbors to the test instance.
    
    Args:
        train_set (list): The training set.
        test_instance (list): The test instance.
        k (int): The number of neighbors to retrieve.
    
    Returns:
        list: The k nearest neighbors.
    """
    distances = [(train_instance, calculate_distance(test_instance, train_instance, len(test_instance) - 1)) for train_instance in train_set]
    distances.sort(key=operator.itemgetter(1))
    return [distances[i][0] for i in range(k)]

def get_response(neighbors):
    """
    Get the majority class from the neighbors.
    
    Args:
        neighbors (list): The neighbors.
    
    Returns:
        The majority class.
    """
    class_votes = {}
    for neighbor in neighbors:
        response = neighbor[-1]
        class_votes[response] = class_votes.get(response, 0) + 1
    
    sorted_votes = sorted(class_votes.items(), key=operator.itemgetter(1), reverse=True)
    return sorted_votes[0][0]

def calculate_accuracy(test_set, predictions):
    """
    Calculate the accuracy of the predictions.
    
    Args:
        test_set (list): The test set.
        predictions (list): The predictions.
    
    Returns:
        float: The accuracy percentage.
    """
    correct = sum(1 for i in range(len(test_set)) if test_set[i][-1] == predictions[i])
    return (correct / float(len(test_set))) * 100.0

def main():
    """
    Main function to run the k-NN algorithm.
    """
    filename = 'traindata.csv'
    split_ratio = 0.67
    k = 3
    
    train_set, test_set = load_dataset(filename, split_ratio)
    print(f'Train set: {len(train_set)}')
    print(f'Test set: {len(test_set)}')
    
    predictions = []
    for test_instance in test_set:
        neighbors = get_neighbors(train_set, test_instance, k)
        result = get_response(neighbors)
        predictions.append(result)
        print(f'> predicted={result}, actual={test_instance[-1]}')
    
    accuracy = calculate_accuracy(test_set, predictions)
    print(f'Accuracy: {accuracy:.2f}%')

if __name__ == '__main__':
    main()
