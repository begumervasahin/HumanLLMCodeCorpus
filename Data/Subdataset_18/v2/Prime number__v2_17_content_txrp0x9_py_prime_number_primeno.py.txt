def sieve_of_eratosthenes(n):
    """
    Generate a list of prime numbers up to a given number n using the Sieve of Eratosthenes algorithm.
    
    :param n: The upper limit to generate prime numbers.
    :return: A list of prime numbers up to n.
    """
    # Initialize a list of boolean values representing the primality of each number
    primes = [True] * (n + 1)
    p = 2

    while p * p <= n:
        if primes[p]:
            # Mark all multiples of p as False indicating they are not prime
            for i in range(p * p, n + 1, p):
                primes[i] = False
        p += 1

    # Collect all prime numbers from 2 to n
    prime_numbers = [p for p in range(2, n + 1) if primes[p]]
    return prime_numbers

# Example usage
print("Welcome random tester, this program will print all the prime numbers from 2 to n")
n = int(input("Enter n: "))
prime_numbers = sieve_of_eratosthenes(n)
print("Prime numbers up to", n, "are:", prime_numbers)
