import random

def generate_configurations(algoritmo, num_samples, sample_size, file_prefix):
    """
    Generates random configurations, writes them to a file, and returns the configurations.
    
    Args:
        algoritmo (str): The name of the algorithm (used in the file name).
        num_samples (int): The number of samples to generate.
        sample_size (int): The size of each sample.
        file_prefix (str): The prefix for the output file name.
    
    Returns:
        list: A list of generated configurations.
    """
    configurations = []
    for n in range(1, num_samples + 1):
        configuration = random.sample(range(1000000000, 9999999999), sample_size)
        configurations.append(configuration)
        print(n)
    
    file_name = f'{file_prefix}Inicial{algoritmo}.txt'
    with open(file_name, 'a') as file:
        for config in configurations:
            file.write(f'{config}\n')
    
    return configurations

def confA(algoritmo):
    return generate_configurations(algoritmo, 100000, 10, 'confA')

def confB(algoritmo):
    return generate_configurations(algoritmo, 10000, 100, 'confB')

def confC(algoritmo):
    return generate_configurations(algoritmo, 1000, 1000, 'confC')

def confD(algoritmo):
    return generate_configurations(algoritmo, 100, 10000, 'confD')

def confE(algoritmo):
    return generate_configurations(algoritmo, 10, 100000, 'confE')

def confF(algoritmo):
    return generate_configurations(algoritmo, 1, 1000000, 'confF')
