import hashlib
import re

def main():
    # Define input and output file paths
    inputfile = 'data/test/ftb1/ftb1u.tsv'
    outputfile = 'data/test/ftb1/ftb1u_sample.tsv'
    
    # Read sentences from input file
    with open(inputfile) as f:
        sentences = list(split_sentences(f.readlines()))
    
    # Select sentences with SHA-256 digest starting with '0'
    selected = []
    for sentence in sentences:
        digest = hashlib.sha256(sentence.encode('utf-8')).hexdigest()
        if digest.startswith('0'):
            selected.append(sentence)
    
    # Print summary of sampled sentences
    print(f'Sampled {len(selected)} out of {len(sentences)} sentences')
    
    # Write selected sentences to output file
    with open(outputfile, 'w') as f:
        f.write('\n'.join(selected))
    
    # Print confirmation message
    print(f'Wrote samples to {outputfile}')

def split_sentences(lines):
    # Regular expression to match start of a sentence
    start_re = re.compile(r'^')
    sentence = []
    
    # Iterate through lines to split sentences
    for line in lines:
        # If line matches the start of a sentence
        if start_re.match(line):
            # If there's a sentence being accumulated, yield it
            if sentence:
                yield ''.join(sentence)
                sentence = []  # Reset sentence buffer
        # If line is not empty, add it to the current sentence
        if line.strip():
            sentence.append(line)
    
    # Yield the last accumulated sentence if any
    if sentence:
        yield ''.join(sentence)

if __name__ == '__main__':
    main()
