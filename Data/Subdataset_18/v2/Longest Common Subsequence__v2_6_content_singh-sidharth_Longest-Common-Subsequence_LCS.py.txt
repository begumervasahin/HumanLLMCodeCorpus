def lcs(x, y):
    # Get the lengths of the two input strings
    m = len(x)
    n = len(y)
    
    # Create a 2D list to store lengths of longest common subsequences
    L = [[0] * (n + 1) for _ in range(m + 1)]
    
    # Build the LCS table in bottom-up fashion
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if x[i - 1] == y[j - 1]:  # Check if characters match
                L[i][j] = L[i - 1][j - 1] + 1
            else:  # If no match, take the maximum of the two possibilities
                L[i][j] = max(L[i - 1][j], L[i][j - 1])
    
    # The length of the LCS is in the bottom-right cell
    return L[m][n]

if __name__ == "__main__":
    # Prompt the user to enter the two strings
    x = input("First string: ")
    y = input("Second string: ")
    
    # Compute and print the length of the LCS
    print("The length of LCS is:", lcs(x, y))
