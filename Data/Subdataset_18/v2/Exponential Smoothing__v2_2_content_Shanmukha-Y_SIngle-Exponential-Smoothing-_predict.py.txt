import matplotlib.pyplot as plt

def read_data(file_path):
    """
    Reads data from a file and returns it as a list of floats.
    
    Parameters:
    file_path (str): Path to the data file.
    
    Returns:
    list: List of float values read from the file.
    """
    with open(file_path, 'r') as file:
        data = list(map(float, file.read().strip().split("\n")))
    return data

def simple_exponential_smoothing(data, alpha):
    """
    Applies simple exponential smoothing to a list of data.
    
    Parameters:
    data (list): List of float values to smooth.
    alpha (float): Smoothing factor.
    
    Returns:
    list: List of smoothed values.
    """
    smoothed_data = [data[0], data[1]]  # Initialize the first two values
    for i in range(2, len(data)):
        smoothed_value = alpha * data[i - 1] + (1 - alpha) * smoothed_data[i - 2]
        smoothed_data.append(round(smoothed_value, 3))
    return smoothed_data

def calculate_mean_absolute_error(data, smoothed_data):
    """
    Calculates the mean absolute error between original and smoothed data.
    
    Parameters:
    data (list): List of original data values.
    smoothed_data (list): List of smoothed data values.
    
    Returns:
    float: Mean absolute error.
    """
    total_error = sum(abs(data[i] - smoothed_data[i]) for i in range(len(data)))
    mean_error = total_error / len(data)
    return mean_error

def plot_data(original_data, smoothed_data):
    """
    Plots the original and smoothed data.
    
    Parameters:
    original_data (list): List of original data values.
    smoothed_data (list): List of smoothed data values.
    """
    plt.plot(original_data, color="blue", label="Original Data")
    plt.plot(smoothed_data, color="red", label="Smoothed Data")
    plt.legend()
    plt.xlabel('Time')
    plt.ylabel('Value')
    plt.title('Original vs Smoothed Data')
    plt.show()

def main():
    """
    Main function to read data, apply exponential smoothing, calculate error, and plot results.
    """
    file_path = "data.txt"  # Replace with your file path
    alpha = 0.98
    
    # Read data from file
    data = read_data(file_path)
    
    # Apply simple exponential smoothing
    smoothed_data = simple_exponential_smoothing(data, alpha)
    
    # Calculate mean absolute error
    mean_error = calculate_mean_absolute_error(data, smoothed_data)
    
    # Print the mean absolute error
    print(f"Mean Absolute Error: {mean_error}")
    
    # Plot the original and smoothed data
    plot_data(data, smoothed_data)

if __name__ == "__main__":
    main()
