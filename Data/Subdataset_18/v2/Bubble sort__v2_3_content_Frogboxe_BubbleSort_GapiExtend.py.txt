def type_of(text):
    """
    Returns the corresponding type based on the input string.
    """
    if text == "int":
        return int
    elif text == "str":
        return str
    elif text == "flt":
        return float
    elif text == "Non":
        return pass_function

def root_check(root):
    """
    Checks if the given path exists.
    """
    return path.exists(root)

def load_dict(root, types):
    """
    Loads a dictionary from a file.
    """
    if not root_check(root):
        raise FileNotFoundError("File not found")
    
    result = {}
    with open(root, "r") as file:
        for line in file.readlines():
            line = line.strip()
            if line == "ENDKEK":
                return result
            key, val = line.split(":")
            key, val = types[0](key), types[1](val)
            result[key] = val
    return result

def save_dict(item, root):
    """
    Saves a dictionary to a file.
    """
    with open(root, "w") as file:
        for key, val in range_iter(item):
            file.write(f"{key}:{val}\n")
        file.write("ENDKEK")

def load_set(root):
    """
    Loads a set from a file.
    """
    if not root_check(root):
        raise FileNotFoundError("File not found")
    
    result = []
    with open(root, "r") as file:
        current = {}
        for line in file.readlines():
            line = line.strip()
            if line == "ENDOBJ":
                result.append(current)
                current = {}
            elif line == "ENDKEK":
                return result
            else:
                cls, attr, val = (type_of(line.split(".")[0]),
                                  line.split(":")[0].split(".")[1],
                                  line.split(":")[1])
                current[attr] = cls(val)
    return result

def save_set(root, items):
    """
    Saves a set to a file.
    """
    with open(root, "w") as file:
        for item in items:
            for key, val in range_iter(item):
                file.write(f"{type(val).__name__[:3]}.{key}:{val}\n")
            file.write("ENDOBJ\n")
        file.write("ENDKEK\n")

def load_list(root, _type):
    """
    Loads a list from a file.
    """
    if not root_check(root):
        raise FileNotFoundError("File not found")
    
    result = []
    with open(root, "r") as file:
        for line in file.readlines():
            line = line.strip()
            if line == "ENDKEK":
                return result
            result.append(_type(line))
    return result

def load_string(root):
    """
    Loads a string from a file.
    """
    if not root_check(root):
        raise FileNotFoundError("File not found")
    
    with open(root, "r") as file:
        return "\n".join(file.readlines())

def save_string(item, root):

def exclude(_set, map_id, index=0):
    """
    Excludes items from the set based on the map_id.
    """
    temp = set()
    for item in _set:
        if item[index] == map_id:
            temp.add(item)
    for item in temp:
        _set.remove(item)
    return _set

def range_iter(x, y=None, z=False):
    """
    Generates a range of values or items.
    """
    if y is None:
        if isinstance(x, int):
            for i in range(x):
                yield i
        elif isinstance(x, dict):
            for key, val in x.items():
                yield key, val
        else:
            for item, n in zip(x, range(len(x))):
                yield item, n
    elif not z:
        for ix in range(x):
            for iy in range(y):
                yield ix, iy
    elif z:
        for ix in range(x):
            for iy in range(y):
                yield ix, iy, (ix * x) + iy

class KeyMapper:
    """
    Class for defining and populating key mappings.
    """
    @staticmethod
    def define_key_map():
        for char in range(26):
            setattr(KeyMapper, chr(char + 97).upper(), char + 97)
        for char in range(12):
            setattr(KeyMapper, f"F{char + 1}", char + 282)

    @staticmethod
    def populate(pop):
        KeyMapper.MAP = pop
        for key, val in range_iter(pop):
            setattr(KeyMapper, val, key)

    @staticmethod
    def is_number(k_key):
        k_key -= 48
        return 0 <= k_key <= 9

    @staticmethod
    def number(k_key):
        return k_key - 48

def color_to_rgba(col):
    """
    Converts an integer to an RGBA color.
    """
    col = int(col)
    r = col // (256 ** 3)
    col -= r * (256 ** 3)
    g = col // (256 ** 2)
    col -= g * (256 ** 2)
    b = col // 256
    col -= b * 256
    a = col
    return (r, g, b, a)

def rgba_to_int(col):
    """
    Converts an RGBA color to an integer.
    """
    return sum(col[i] * (256 ** (3 - i)) for i in range(4))

class ColorMapper:
    """
    Class for populating color mappings.
    """
    @staticmethod
    def populate(pop):
        for key, val in range_iter(pop):
            setattr(ColorMapper, key, val)
