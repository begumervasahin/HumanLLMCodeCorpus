import string
import re
from Sastrawi.Stemmer.StemmerFactory import StemmerFactory
import collections

class Preprocessing:
    # Load stopwords from file
    with open("stopword-list.txt", "r") as file:
        stopwords = file.read().split('\n')

    @staticmethod
    def clean_text(text):
        """
        Cleans the input text by removing links, punctuation, numbers,
        and extra whitespace.
        """
        # Remove URLs
        text = re.sub(r'\b\w+://\S+', ' ', text)
        # Remove punctuation
        text = text.translate(str.maketrans(string.punctuation, ' ' * len(string.punctuation)))
        # Remove numbers
        text = text.translate(str.maketrans('', '', '0123456789'))
        # Remove extra spaces
        text = re.sub(r'\s+', ' ', text).strip()
        return text

    @staticmethod
    def case_fold(text):
        """
        Converts text to lowercase.
        """
        return text.lower()

    @staticmethod
    def tokenize(text):
        """
        Tokenizes text into words.
        """
        return text.split()

    @staticmethod
    def remove_stopwords(words):
        """
        Removes stopwords from the list of words.
        """
        return [word for word in words if word not in Preprocessing.stopwords]

    @staticmethod
    def get_unique_words(words):
        """
        Returns unique words maintaining their order of appearance.
        """
        return list(collections.OrderedDict.fromkeys(words))

    @staticmethod
    def stem_words(words):
        """
        Performs stemming on the list of words.
        """
        factory = StemmerFactory()
        stemmer = factory.create_stemmer()
        return [stemmer.stem(word) for word in words]

    @staticmethod
    def preprocess(text):
        """
        Applies all preprocessing steps: cleaning, case folding, tokenization,
        removing stopwords, getting unique words, and stemming.
        """
        text = Preprocessing.clean_text(text)
        text = Preprocessing.case_fold(text)
        words = Preprocessing.tokenize(text)
        words = Preprocessing.remove_stopwords(words)
        words = Preprocessing.get_unique_words(words)
        words = Preprocessing.stem_words(words)
        return words

    @staticmethod
    def preprocess_without_unique(text):
        """
        Applies all preprocessing steps except for getting unique words.
        """
        text = Preprocessing.clean_text(text)
        text = Preprocessing.case_fold(text)
        words = Preprocessing.tokenize(text)
        words = Preprocessing.remove_stopwords(words)
        words = Preprocessing.stem_words(words)
        return words

# Example usage
if __name__ == "__main__":
    sample_text = "This is a sample text to demonstrate preprocessing. Visit www.example.com for more info!"
    preprocessed_text = Preprocessing.preprocess(sample_text)
    print("Preprocessed text:", preprocessed_text)
