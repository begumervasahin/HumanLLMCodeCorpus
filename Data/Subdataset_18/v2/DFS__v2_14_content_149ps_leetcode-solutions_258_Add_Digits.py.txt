class Solution:
    def addDigits(self, num: int) -> int:
        """
        Given a non-negative integer num, repeatedly add all its digits until the result has only one digit.
        :type num: int
        :rtype: int
        """
        temp = num
        while temp >= 10:  # Continue the process while temp has more than one digit
            next_temp = 0
            while temp > 0:
                next_temp += temp % 10  # Add the last digit of temp to next_temp
                temp //= 10  # Remove the last digit from temp
            temp = next_temp  # Update temp to the sum of its digits
        return temp

# Example usage:
solution = Solution()

# Test cases
print(solution.addDigits(38))   # Output: 2
print(solution.addDigits(123))  # Output: 6
print(solution.addDigits(0))    # Output: 0
