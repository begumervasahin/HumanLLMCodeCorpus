import unittest
from time import time

# Function to read a file and apply a callback to each word
def getFile(filename, callback):
    try:
        with open(filename, 'r') as f:
            words = f.read().split()
        # Apply the callback to each word and filter out None results
        return [callback(word) for word in words if callback(word) is not None]
    except FileNotFoundError:
        raise FileNotFoundError(f"File {filename} not found")

# Sample list of words for comparison in tests
words = [
    'Fish',
    'hat',
    'foo',
    'cow',
    'Cow',
    'siamese.',
    'Wonderland',
    'foo',
    'toothpaste',
    'bar',
    'umbrella',
    'foo\n',
    'milky',
    '\'"flight-manual"\'',
    'toothpick',
    'bar',
    'enough',
    'event-horizon',
    'bar!'
]

# Callback function to append 'hi' to each word, except for 'umbrella'
def on_word(word):
    if word == 'umbrella':
        return None
    return word + 'hi'

# Unit test class for the getFile function
class GetFileTest(unittest.TestCase):
    # Test getFile with basic functionality
    def test_getFile_basic(self):
        self.assertEqual(words, getFile('small_test.txt', lambda a: a))
    
    # Test getFile with a non-existent file
    def test_getFile_no_file(self):
        self.assertRaises(FileNotFoundError, getFile, 'aaa.txt', lambda a: a)
    
    # Test getFile with the on_word callback
    def test_getFile_with_callback(self):
        res = ['Fishhi', 'hathi', 'foohi', 'cowhi', 'Cowhi', 'siamese.hi', 'Wonderlandhi', 'foohi', 'toothpastehi', 'barhi']
        self.assertEqual(res, getFile('small_test.txt', on_word))
    
    # Test getFile with a large file and measure time taken
    def test_getFile_moby(self):
        t0 = time()
        self.assertLess(100000, len(getFile('moby_test.txt', on_word)))
        t1 = time()
        print('Moby import time sec:', t1 - t0)

# Run the unit tests
if __name__ == '__main__':
    unittest.main()
