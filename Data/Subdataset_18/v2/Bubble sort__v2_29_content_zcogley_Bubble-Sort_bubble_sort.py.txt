def bubble_sort(lst):
    # Create a copy of the list to avoid modifying the original list
    new_list = lst[:]
    is_sorted = False  # Flag to keep track of whether the list is sorted

    while not is_sorted:
        idx = 0
        nswaps = 0  # Counter to keep track of the number of swaps in each pass
        lst_range = len(new_list) if len(new_list) % 2 == 0 else len(new_list) - 1

        # Loop through the list up to the second-to-last element
        while idx < lst_range - 1:
            if new_list[idx] > new_list[idx + 1]:
                # Swap the elements if they are in the wrong order
                new_list[idx], new_list[idx + 1] = new_list[idx + 1], new_list[idx]
                nswaps += 1
            idx += 1

        # If no swaps were made in this pass, the list is sorted
        if nswaps == 0:
            is_sorted = True

    return new_list

# Example usage
if __name__ == "__main__":
    lst = [64, 25, 12, 22, 11]
    sorted_lst = bubble_sort(lst)
    print("Sorted list:", sorted_lst)
