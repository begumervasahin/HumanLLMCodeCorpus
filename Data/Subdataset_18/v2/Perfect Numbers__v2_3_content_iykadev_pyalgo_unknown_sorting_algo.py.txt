# Initial unsorted list
unsorted_list = [6, 2, 7, 8, 3, 1, 10, 5, 4, 9]
sorted_list = []

while unsorted_list:
    # Assume the first element is the minimum
    min_value = unsorted_list[0]
    min_index = 0
    
    # Find the minimum element and its index in the unsorted list
    for i in range(len(unsorted_list)):
        if unsorted_list[i] < min_value:
            min_value = unsorted_list[i]
            min_index = i
    
    # Remove the minimum element from the unsorted list
    del unsorted_list[min_index]
    
    # Append the minimum element to the sorted list
    sorted_list.append(min_value)

# Print the sorted list
print(sorted_list)
