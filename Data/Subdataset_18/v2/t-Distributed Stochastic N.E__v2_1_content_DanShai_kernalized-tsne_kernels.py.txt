import numpy as np

class Kernels:
    def __init__(self, data, kernel_options=None):
        """
        Initializes the Kernels class with data and kernel options.
        
        :param data: Input data (numpy array).
        :param kernel_options: Dictionary of kernel options including type, gamma, degree, and number of components.
        """
        self.X = data.copy()
        self.k_opts = kernel_options if kernel_options else {"kernel": "pca", "gamma": 0.5, "degree": 1, "pcomp": 4}

    def process_data(self):
        """
        Processes the input data using the specified kernel.
        
        :return: Processed data after applying the chosen kernel method.
        """
        gamma = self.k_opts["gamma"]
        degree = self.k_opts["degree"]
        n_components = self.k_opts["pcomp"]
        kernel_type = self.k_opts["kernel"]

        if kernel_type == "poly":
            return self.poly(self.X, gamma=gamma, degree=degree, n_components=n_components)
        elif kernel_type == "anova":
            return self.anova(self.X, gamma=gamma, degree=degree, n_components=n_components)
        elif kernel_type == "rbf":
            return self.rbf(self.X, gamma=gamma, n_components=n_components)
        elif kernel_type == "cosine":
            return self.cosine(self.X, n_components=n_components)
        elif kernel_type == "iquad":
            return self.iquad(self.X, gamma=gamma, degree=degree, n_components=n_components)
        elif kernel_type == "cauchy":
            return self.cauchy(self.X, gamma=gamma, n_components=n_components)
        elif kernel_type == "fourier":
            return self.fourier(self.X, gamma=gamma, n_components=n_components)
        else:
            return self.pca(self.X, n_components=n_components)

    def eigens(self, matrix, n_components=4):
        """
        Computes the eigenvalues and eigenvectors of the given matrix, returning the top components.
        
        :param matrix: The matrix to decompose.
        :param n_components: The number of components to retain.
        :return: Matrix of eigenvectors corresponding to the top eigenvalues.
        """
        eigenvalues, eigenvectors = np.linalg.eig(matrix)
        sorted_indices = np.argsort(eigenvalues)[::-1]  # Sort in descending order
        top_eigenvectors = eigenvectors[:, sorted_indices][:, :n_components]
        return top_eigenvectors.real

    def centerK(self, K):
        """
        Centers the kernel matrix.
        
        :param K: Kernel matrix to be centered.
        :return: Centered kernel matrix.
        """
        N = K.shape[0]
        one_n = np.ones((N, N)) / N
        return K - one_n @ K - K @ one_n + one_n @ K @ one_n

    def pca(self, X, n_components=2):
        """
        Performs Principal Component Analysis (PCA) on the data.
        
        :param X: Input data matrix.
        :param n_components: Number of principal components to retain.
        :return: Data transformed into the principal component space.
        """
        X -= np.mean(X, axis=0)
        covariance_matrix = np.cov(X.T)
        return np.dot(X, self.eigens(covariance_matrix, n_components=n_components))

    def poly(self, X, gamma=1, degree=2, n_components=2):
        """
        Applies a polynomial kernel to the data.
        
        :param X: Input data matrix.
        :param gamma: Kernel coefficient.
        :param degree: Degree of the polynomial.
        :param n_components: Number of components to retain.
        :return: Data transformed by the polynomial kernel.
        """
        X -= np.mean(X, axis=0)
        K = (gamma * X @ X.T + 1) ** degree
        return self.eigens(self.centerK(K), n_components=n_components)

    def rbf(self, X, gamma=0.1, n_components=2):
        """
        Applies a Radial Basis Function (RBF) kernel to the data.
        
        :param X: Input data matrix.
        :param gamma: Kernel coefficient.
        :param n_components: Number of components to retain.
        :return: Data transformed by the RBF kernel.
        """
        X -= np.mean(X, axis=0)
        squared_distances = np.sum((X[:, None, :] - X[None, :, :]) ** 2, axis=-1)
        K = np.exp(-gamma * squared_distances)
        return self.eigens(self.centerK(K), n_components=n_components)

    def cosine(self, X, n_components=2):
        """
        Applies a cosine similarity kernel to the data.
        
        :param X: Input data matrix.
        :param n_components: Number of components to retain.
        :return: Data transformed by the cosine kernel.
        """
        X -= np.mean(X, axis=0)
        norm_X = np.linalg.norm(X, axis=1, keepdims=True)
        K = X @ X.T / (norm_X @ norm_X.T)
        return self.eigens(self.centerK(K), n_components=n_components)

    def iquad(self, X, gamma=1, degree=1, n_components=2):
        """
        Applies an inverse quadratic kernel to the data.
        
        :param X: Input data matrix.
        :param gamma: Kernel coefficient.
        :param degree: Degree of the inverse quadratic function.
        :param n_components: Number of components to retain.
        :return: Data transformed by the inverse quadratic kernel.
        """
        X -= np.mean(X, axis=0)
        squared_distances = np.sum((X[:, None, :] - X[None, :, :]) ** 2, axis=-1)
        K = 1 / (squared_distances + gamma ** 2) ** degree
        return self.eigens(self.centerK(K), n_components=n_components)

    def cauchy(self, X, gamma=0.2, n_components=2):
        """
        Applies a Cauchy kernel to the data.
        
        :param X: Input data matrix.
        :param gamma: Kernel coefficient.
        :param n_components: Number of components to retain.
        :return: Data transformed by the Cauchy kernel.
        """
        X -= np.mean(X, axis=0)
        squared_distances = np.sum((X[:, None, :] - X[None, :, :]) ** 2, axis=-1)
        K = 1 / (1 + gamma * squared_distances)
        return self.eigens(self.centerK(K), n_components=n_components)

    def anova(self, X, gamma=0.01, degree=1, n_components=2):
        """
        Applies an ANOVA kernel to the data.
        
        :param X: Input data matrix.
        :param gamma: Kernel coefficient.
        :param degree: Degree of the ANOVA function.
        :param n_components: Number of components to retain.
        :return: Data transformed by the ANOVA kernel.
        """
        X -= np.mean(X, axis=0)
        K = np.zeros((X.shape[0], X.shape[0]))
        for d in range(X.shape[1]):
            X_d = X[:, d].reshape(-1, 1)
            K += np.exp(-gamma * (X_d - X_d.T) ** 2) ** degree
        return self.eigens(self.centerK(K), n_components=n_components)

    def fourier(self, X, gamma=0.1, n_components=2):
        """
        Applies a Fourier kernel to the data.
        
        :param X: Input data matrix.
        :param gamma: Kernel coefficient.
        :param n_components: Number of components to retain.
        :return: Data transformed by the Fourier kernel.
        """
        X -= np.mean(X, axis=0)
        K = np.ones((X.shape[0], X.shape[0]))
        gamma = min(0.1, gamma)
        for d in range(X.shape[1]):
            X_d = X[:, d].reshape(-1, 1)
            K *= (1 - gamma ** 2) / (2 * (1 - 2 * gamma * np.cos(X_d - X_d.T)) + gamma ** 2)
        return self.eigens(self.centerK(K), n_components=n_components)

if __name__ == "__main__":
    # Generate random data for testing
    data = np.random.rand(100, 10)

    # Initialize the kernel processor with specific options
    kernel_processor = Kernels(data, kernel_options={"kernel": "rbf", "gamma": 0.5, "pcomp": 3})

    # Process the data using the chosen kernel
    processed_data = kernel_processor.process_data()

    print("Processed data shape:", processed_data.shape)
