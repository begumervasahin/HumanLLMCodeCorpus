import re
from text2num import text2num, Small, Magnitude

# Dictionary to map specific words to their corresponding numeric values
Translations = {
    'a': 'one',
    'dozen': 'twelve',
    'and': '',
}

# Create a set of valid tokens which includes small numbers, magnitudes, and custom translations
valid_tokens = set(["hundred"])
for dictionary in [Small, Magnitude, Translations]:
    valid_tokens.update(dictionary.keys())

# Function to preprocess text: replace non-alphanumeric characters and translate specific words
def prep(text):
    # Replace non-word characters and underscores with spaces
    text = re.sub(r"[\W_]", ' ', text)
    tokens = []
    for token in text.split():
        translated = Translations.get(token)
        # Only append the translation if it's not empty
        if translated is not None and translated != '':
            tokens.append(translated)
        else:
            tokens.append(token)
    return tokens

# Function to extract numeric values from text
def extract(text):
    tokens = prep(text)
    first_valid_index = len(tokens)
    # Find the position of the first valid token from the end of the list
    for i in reversed(range(len(tokens))):
        if tokens[i] in valid_tokens:
            first_valid_index = i
        else:
            break
    # Convert the relevant part of the token list to a number
    return text2num(tokens[first_valid_index:])

# Function to convert an entire text to an integer
def atoi(text):
    tokens = prep(text)
    return text2num(tokens)

# Example usage
if __name__ == "__main__":
    example_text = "I have a dozen apples and twenty-five oranges"
    print(f"Extracted number: {extract(example_text)}")
    print(f"Converted number: {atoi(example_text)}")
