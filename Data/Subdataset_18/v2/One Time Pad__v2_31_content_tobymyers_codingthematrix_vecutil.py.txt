class Vec:
    def __init__(self, domain, function):
        """
        Initialize a vector.
        
        Args:
            domain (set): The set of indices for the vector.
            function (dict): A dictionary mapping indices to values.
        """
        self.D = domain
        self.f = function

    def __repr__(self):
        """
        Return a string representation of the vector.
        """
        return f"Vec({self.D},{self.f})"

    def __getitem__(self, key):
        """
        Get an item from the vector.
        
        Args:
            key: The index of the item.
        
        Returns:
            The value at the specified index.
        """
        return self.f.get(key, 0)

    def __setitem__(self, key, value):
        """
        Set an item in the vector.
        
        Args:
            key: The index of the item.
            value: The value to set at the specified index.
        """
        self.f[key] = value

def list2vec(L):
    """
    Given a list L of field elements, return a Vec with domain {0...len(L)-1}
    whose entry i is L[i].
    
    Args:
        L (list): The list of field elements.
    
    Returns:
        Vec: A vector with domain {0...len(L)-1} and corresponding values.
    
    Example:
    >>> list2vec([10, 20, 30])
    Vec({0, 1, 2},{0: 10, 1: 20, 2: 30})
    """
    return Vec(set(range(len(L))), {k: L[k] for k in range(len(L))})

def zero_vec(D):
    """
    Returns a zero vector with the given domain.
    
    Args:
        D (set): The domain of the vector.
    
    Returns:
        Vec: A vector with domain D and all values set to 0.
    """
    return Vec(D, {d: 0 for d in D})

def main():
    """
    Main function to test the Vec class and associated functions.
    """
    # Test list2vec function
    vec1 = list2vec([10, 20, 30])
    print(vec1)  # Output: Vec({0, 1, 2},{0: 10, 1: 20, 2: 30})

    # Test zero_vec function
    vec2 = zero_vec({0, 1, 2, 3})
    print(vec2)  # Output: Vec({0, 1, 2, 3},{0: 0, 1: 0, 2: 0, 3: 0})

if __name__ == "__main__":
    main()
