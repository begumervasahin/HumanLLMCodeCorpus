# db.py

import pymysql.cursors

def create_connection():
    """
    Creates and returns a database connection object.

    Returns:
        pymysql.connections.Connection: A connection object to the database.
    """
    return pymysql.connect(
        host='localhost',
        user='root',
        password='password',
        db='test_db',
        charset='utf8',
        cursorclass=pymysql.cursors.DictCursor
    )

def get_cursor():
    """
    Obtains a database cursor object for executing queries.

    Returns:
        pymysql.cursors.Cursor: A cursor object for database operations.
    """
    connection = create_connection()
    return connection.cursor()
