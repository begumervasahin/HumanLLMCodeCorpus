def babylonian_square_root(a: float, eps: float) -> float:
    """
    Calculates the square root of a number using the Babylonian method.

    The Babylonian method (also known as Heron's method) is an iterative algorithm to find
    the square root of a number by successive approximations.

    Parameters:
    a (float): The number to find the square root of.
    eps (float): The tolerance for the approximation.

    Returns:
    float: The approximated square root of the number.
    """
    # Initial guess
    x0 = (1 + a) / 2
    x1 = (x0 + a / x0) * 0.5
    y = abs((x1 - x0) / x0)
    
    # Iterate until the approximation error is within the tolerance
    while y > eps:
        x0 = x1
        x1 = (x0 + a / x0) * 0.5
        y = abs((x1 - x0) / x0)
    
    return x1

def main() -> None:
    """
    Main function to test the Babylonian square root calculation.
    """
    number = 5
    tolerance = 0.0001
    result = babylonian_square_root(number, tolerance)
    print(f"The square root of {number} is approximately {result:.6f}")

if __name__ == "__main__":
    main()
