from collections import deque

class TreeNode:
    def __init__(self, val=-1, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

class BinaryTree:
    """
    A binary tree representation that supports deserialization from strings of serialized binary trees,
    as well as pre-order, in-order, post-order, and BFS traversal.
    """
    def __init__(self, data=None):
        if data is None:
            self.root = None
        else:
            self.root = self.deserialize(data)

    def __iter__(self):
        """
        Default iterator set to pre-order iterator
        """
        return self.preOrderIter()

    @staticmethod
    def deserialize(data):
        """
        Decodes a serialized string to construct a binary tree.
        :param data: str
        :return: TreeNode
        """
        if not isinstance(data, str):
            raise TypeError('Input must be a string')
        if len(data) < 3 or not (data.startswith('[') and data.endswith(']')):
            raise ValueError('Input must be surrounded by brackets and cannot be empty')
        
        data = data[1:-1].split(', ')
        if data[0] == 'null':
            return None
        
        root = TreeNode(int(data[0]))
        queue = deque([root])
        i = 1
        
        while i < len(data):
            node = queue.popleft()
            if data[i] != 'null':
                node.left = TreeNode(int(data[i]))
                queue.append(node.left)
            i += 1
            if i < len(data) and data[i] != 'null':
                node.right = TreeNode(int(data[i]))
                queue.append(node.right)
            i += 1
        
        return root

    def inOrderIter(self):
        """
        In-order iterator
        """
        if self.root:
            stack, visited = [], set()
            cur = self.root
            while stack or cur:
                while cur:
                    stack.append(cur)
                    cur = cur.left
                cur = stack.pop()
                yield cur.val
                cur = cur.right

    def preOrderIter(self):
        """
        Pre-order iterator
        """
        if self.root:
            stack = [self.root]
            while stack:
                cur = stack.pop()
                yield cur.val
                if cur.right:
                    stack.append(cur.right)
                if cur.left:
                    stack.append(cur.left)

    def postOrderIter(self):
        """
        Post-order iterator
        """
        if self.root:
            stack, visited = [self.root], set()
            while stack:
                cur = stack[-1]
                if (not cur.left or cur.left in visited) and (not cur.right or cur.right in visited):
                    stack.pop()
                    visited.add(cur)
                    yield cur.val
                else:
                    if cur.right and cur.right not in visited:
                        stack.append(cur.right)
                    if cur.left and cur.left not in visited:
                        stack.append(cur.left)

    def bfsIter(self):
        """
        BFS iterator
        """
        if self.root:
            queue = deque([self.root])
            while queue:
                cur = queue.popleft()
                yield cur.val
                if cur.left:
                    queue.append(cur.left)
                if cur.right:
                    queue.append(cur.right)

# Example usage:
if __name__ == '__main__':
    # Example serialized tree: "[1, 2, 3, null, null, 4, 5]"
    data = "[1, 2, 3, null, null, 4, 5]"
    bt = BinaryTree(data)
    
    print("Pre-order traversal:")
    for value in bt.preOrderIter():
        print(value, end=' ')
    print("\nIn-order traversal:")
    for value in bt.inOrderIter():
        print(value, end=' ')
    print("\nPost-order traversal:")
    for value in bt.postOrderIter():
        print(value, end=' ')
    print("\nBFS traversal:")
    for value in bt.bfsIter():
        print(value, end=' ')
