import numpy as np
import pandas as pd
import selenium
from selenium import webdriver
from selenium.webdriver.common.by import By
import time

class Bot:
    def __init__(self):
        # Initialize the webdriver and open the timetable website
        self.driver = webdriver.Chrome("./chromedriver")
        self.driver.get("https://timetable.iit.artsci.utoronto.ca/")
        
    def close(self):
        # Close the webdriver
        self.driver.close()
        
    def batch_retrieve(self, department: str) -> pd.DataFrame:
        # Convert department code to uppercase
        department = department.upper()
        
        # Locate the search box and search button
        search_box = self.driver.find_element(By.ID, "courseCode")
        search_button = self.driver.find_element(By.ID, "searchButton")
        
        # Clear the search box, enter the department code, and click the search button
        search_box.clear()
        search_box.click()
        search_box.send_keys(department)
        search_button.click()
        
        # Wait for the search results to load
        course_lst = []
        while not course_lst:
            time.sleep(1.0)
            course_lst = self.driver.find_elements(By.CLASS_NAME, "perCourse")
        
        # Retrieve course information
        course_info_lst = []
        print(f"Total courses found: {len(course_lst)}")
        for course in course_lst:
            code, title = course.find_element(By.CLASS_NAME, "courseTitle").text.split("   ")
            print(f"{code}\t{title}")
            meeting_lst = course.find_elements(By.CLASS_NAME, "perMeeting")
            for meeting in meeting_lst:
                meeting_code = meeting.find_element(By.CLASS_NAME, "colCode").text
                print(f"\t{meeting_code}")
                try:
                    meeting_info = meeting.find_element(By.CLASS_NAME, "secLec" if meeting_code.startswith("LEC") else "secTut").text
                except selenium.common.exceptions.NoSuchElementException:
                    meeting_info = meeting.find_element(By.CLASS_NAME, "secPra").text
                info = [code, title, meeting_code, meeting_info]
                course_info_lst.append(info)
        
        # Create a DataFrame from the retrieved information
        course_info_df = pd.DataFrame(
            np.array(course_info_lst),
            columns=["Code", "Title", "Session", "Details"]
        )
        return course_info_df
    
    def batch_download(self, code_lst: list, save_dir: str) -> None:
        # Get the list of department options
        department_lst = [
            x.text
            for x in self.driver.find_elements(By.CLASS_NAME, "option")
        ]
        print(department_lst)
        
        # Extract department codes from the options
        code_lst = [
            x[1:-1]
            for dep in department_lst
            for x in dep.split(" ")
            if x.startswith("(") and x.endswith(")")
        ]
        
        # Retrieve information for each department and concatenate into a single DataFrame
        all_courses = []
        for code in code_lst:
            df = self.batch_retrieve(code)
            all_courses.append(df)
        
        # Concatenate all course information and save to a CSV file
        final_df = pd.concat(all_courses, axis=0)
        print(final_df.head())
        print(final_df.shape)
        final_df.to_csv(save_dir, index=False)

if __name__ == "__main__":
    # Create an instance of the bot
    bot = Bot()
    
    # Define the list of department codes to search
    code_lst = ["MAT"]
    
    # Perform batch download and save results to CSV
    bot.batch_download(code_lst, save_dir="./results.csv")
    
    # Close the bot
    bot.close()
