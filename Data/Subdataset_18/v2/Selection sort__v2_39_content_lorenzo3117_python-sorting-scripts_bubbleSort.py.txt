def bubble_sort(arr):
    """
    Perform bubble sort on the given array, counting the number of comparisons and swaps.
    """
    n = len(arr)
    comparisons = 0
    swaps = 0
    
    for i in range(n):
        swapped = False
        for j in range(0, n - i - 1):
            comparisons += 1
            if arr[j] > arr[j + 1]:
                swaps += 1
                arr[j], arr[j + 1] = arr[j + 1], arr[j]
                swapped = True
        # If no elements were swapped in the inner loop, the array is already sorted
        if not swapped:
            break
    
    return comparisons, swaps

def main():
    # Example array to be sorted
    arr = [64, 34, 25, 12, 22, 11, 90]
    
    # Print the array before sorting
    print("Array before sorting:")
    print(arr)
    
    # Sort the array using bubble sort
    comparisons, swaps = bubble_sort(arr)
    
    # Print the sorted array
    print("Sorted array is:")
    print(arr)
    
    # Print the number of comparisons and swaps
    print(f"\nBubble Sort:\nComparisons: {comparisons}\nSwaps: {swaps}")

if __name__ == "__main__":
    main()
