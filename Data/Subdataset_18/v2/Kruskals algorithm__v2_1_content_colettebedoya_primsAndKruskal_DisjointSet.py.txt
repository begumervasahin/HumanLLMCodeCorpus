class DisjointSet:
    def __init__(self, size):
        """
        Initializes a disjoint set with the specified size.
        Each element starts in its own set.
        
        Parameters:
            size: The number of elements in the disjoint set.
        """
        if size < 0:
            raise ValueError("Size must be >= 0")
        self.size = size
        self.parent = [-1] * (size + 1)  # -1 indicates that the element is a root

    def find(self, element):
        """
        Find the root of the set containing the specified element.
        Implements path compression to speed up future operations.
        
        Parameters:
            element: The element to find the root of.
        
        Returns:
            The root of the set containing 'element'.
        
        Raises:
            ValueError: If 'element' is out of bounds.
        """
        if element < 0 or element > self.size:
            raise ValueError("Element out of bounds")
        
        if self.parent[element] < 0:  # This means the element is a root
            return element
        else:
            # Path compression: directly link the element to its root
            self.parent[element] = self.find(self.parent[element])
            return self.parent[element]

    def union(self, element1, element2):
        """
        Perform a union of the sets containing element1 and element2.
        The set containing element2 will be merged into the set containing element1.
        
        Parameters:
            element1: The first element.
            element2: The second element.
        
        Raises:
            ValueError: If either element is out of bounds.
        """
        if element1 < 0 or element1 > self.size or element2 < 0 or element2 > self.size:
            raise ValueError("Element out of bounds")

        root1 = self.find(element1)
        root2 = self.find(element2)

        if root1 != root2:  # Only merge if they are in different sets
            self.parent[root2] = root1

if __name__ == "__main__":
    # Example usage:
    ds = DisjointSet(10)

    # Perform unions of some sets
    ds.union(1, 2)
    ds.union(3, 4)
    ds.union(2, 3)

    # Find the roots of the sets
    print("Root of 1:", ds.find(1))  # Should print the root of the set containing 1
    print("Root of 2:", ds.find(2))  # Should print the same root as for 1
    print("Root of 3:", ds.find(3))  # Should print the same root as for 1 and 2
    print("Root of 4:", ds.find(4))  # Should print the same root as for 1, 2, and 3

    # Attempting to find a root outside the valid range should raise an error
    try:
        print(ds.find(11))
    except ValueError as e:
        print("Error:", e)
