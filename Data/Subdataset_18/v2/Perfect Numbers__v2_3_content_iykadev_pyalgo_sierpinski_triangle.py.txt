import functools

def sierpinski(order: int) -> list:
    """
    Generates the Sierpinski triangle of a given order.

    Parameters:
    order (int): The order of the Sierpinski triangle.

    Returns:
    list: A list of strings representing the Sierpinski triangle.
    """
    def aggregate(triangle: list, iteration: int) -> list:
        """
        Helper function to aggregate the Sierpinski triangle.

        Parameters:
        triangle (list): The current state of the triangle.
        iteration (int): The current iteration number.

        Returns:
        list: The updated state of the triangle.
        """
        space = " " * (2 ** iteration)
        return [space + row + space for row in triangle] + [row + " " + row for row in triangle]
    
    return functools.reduce(aggregate, range(order), ["*"])

def main() -> None:
    """
    Main function to generate and print the Sierpinski triangle.
    """
    # Generate the Sierpinski triangle of order 4 and print it
    order = 4
    triangle = sierpinski(order)
    print("\n".join(triangle))

if __name__ == "__main__":
    main()
