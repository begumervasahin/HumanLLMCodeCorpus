import math
from typing import List

def divide_swing_and_conquer(number: int) -> int:
    """
    Computes the swinging factorial of a number using a divide-and-conquer approach.
    
    :param number: The number for which to compute the swinging factorial
    :return: The swinging factorial of the specified number
    """
    if number <= 1:
        return 1

    # Get all prime numbers up to the given number
    primes = all_prime_numbers_up_to(number)

    # Calculate the prime exponents for the given number
    exponents = prime_exponents_of_swing_number(number, primes)

    # Compute the swinging factorial using these prime numbers and exponents
    swinging_factorial = compute_swinging_factorial(primes, exponents)

    # Recursively compute the swinging factorial for number // 2
    return divide_swing_and_conquer(number // 2) ** 2 * swinging_factorial

def prime_exponents_of_swing_number(swing_number: int, primes: List[int]) -> List[int]:
    """
    Determines the exponents of prime factors in the swinging number representation.
    
    :param swing_number: The number to analyze
    :param primes: List of prime numbers for factorization
    :return: List of exponents for each prime number
    """
    max_power = math.floor(math.log2(swing_number))
    exponents = []

    for prime in primes:
        exponent = 0
        for i in range(1, max_power + 1):
            exponent += swing_number // (prime ** i) % 2
        exponents.append(exponent)

    return exponents

def compute_swinging_factorial(primes: List[int], exponents: List[int]) -> int:
    """
    Computes the swinging factorial from given primes and their exponents.
    
    :param primes: List of prime numbers
    :param exponents: List of exponents for each prime number
    :return: The resulting swinging factorial
    """
    result = 1
    for prime, exponent in zip(primes, exponents):
        if exponent > 0:
            result *= prime ** exponent
    return result

def is_prime(n: int) -> bool:
    """
    Checks if a number is a prime number.
    
    :param n: The number to check
    :return: True if the number is prime, otherwise False
    """
    if n <= 1:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False

    max_divisor = math.isqrt(n)
    for divisor in range(3, max_divisor + 1, 2):
        if n % divisor == 0:
            return False
    return True

def all_prime_numbers_up_to(n: int) -> List[int]:
    """
    Generates a list of all prime numbers up to a given number.
    
    :param n: The upper limit for prime number generation
    :return: List of all prime numbers up to the specified number
    """
    return [num for num in range(2, n + 1) if is_prime(num)]
