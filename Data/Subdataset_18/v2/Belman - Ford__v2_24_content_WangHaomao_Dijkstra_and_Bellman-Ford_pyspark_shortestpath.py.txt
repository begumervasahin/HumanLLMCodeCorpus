import sys
from pyspark import SparkContext
from data_operation import get_graph
from pyspark_dijkstra import Dijkstra
from pyspark_bellman_ford import Bellman_Ford
from pprint import pprint

def main():
    """
    Main function to process the graph file and run the appropriate algorithm
    based on the presence of negative weights in the graph.
    """
    input_info = sys.argv
    input_len = len(input_info)
    
    if input_len < 2:
        raise ValueError('At least one input file (graph) is required')
    
    graph_path = input_info[1]
    sc = SparkContext.getOrCreate()
    
    print(f"Processing graph file: {graph_path}")
    
    graph, exist_negative = get_graph(graph_path, sc)
    
    # Choose the appropriate algorithm based on the presence of negative weights
    if exist_negative:
        res = Bellman_Ford('1', graph, sc)
    else:
        res = Dijkstra('1', graph, sc)
    
    # Save results to file or print to console
    if input_len == 3:
        output_path = input_info[2]
        res.saveAsTextFile(output_path)
        print(f"Results saved to: {output_path}")
    else:
        pprint(res.collect())

if __name__ == '__main__':
    main()
