import time
import numpy as np
import networkx as nx
from forex_python.converter import CurrencyRates
import matplotlib.pyplot as plt

# List of currency codes to include in the graph
currency_codes = ['EUR', 'USD', 'GBP', 'JPY', 'CNY', 'INR', 'AUD']

def fetch_currency_rates(currencies, delay=0):
    """
    Fetch exchange rates between the specified currencies and return a dictionary
    where keys are currencies and values are dictionaries of rates to other currencies.

    Parameters:
    currencies (list): List of currency codes to include in the graph.
    delay (float): Time to sleep between requests to avoid rate limiting.

    Returns:
    dict: Dictionary of exchange rates with currencies as keys.
    """
    print('Fetching real-time exchange rates...')
    c = CurrencyRates()
    rates = {}

    for currency in currencies:
        print(f'Processing rates for {currency}...')
        rates[currency] = {}
        for other_currency in currencies:
            if currency == other_currency:
                continue
            
            try:
                rate = c.get_rate(currency, other_currency)
                rates[currency][other_currency] = np.log(rate)
            except Exception as e:
                print(f"Error fetching rate for {currency} to {other_currency}: {e}")
                rates[currency][other_currency] = None
        
        time.sleep(delay)

    return rates

def create_currency_graph(rates):
    """
    Create a network graph from the currency rates dictionary.

    Parameters:
    rates (dict): Dictionary of exchange rates.

    Returns:
    networkx.Graph: Graph representing the currency exchange rates.
    """
    print('Creating graph from exchange rates...')
    G = nx.from_dict_of_dicts(rates)
    return G

def plot_graph(G):
    """
    Plot and save the graph to a file.

    Parameters:
    G (networkx.Graph): Graph to be plotted.
    """
    print('Plotting and saving graph...')
    plt.figure(figsize=(10, 8))
    pos = nx.spring_layout(G)
    nx.draw_networkx(G, pos, with_labels=True, node_size=3000, node_color='skyblue', font_size=12, font_weight='bold')
    plt.title('Currency Exchange Rates Network')
    plt.savefig('currency_graph.png')
    plt.show()

if __name__ == '__main__':
    # Fetch exchange rates and create the graph
    currency_rates = fetch_currency_rates(currency_codes, delay=0.5)
    currency_graph = create_currency_graph(currency_rates)
    
    # Plot the graph
    plot_graph(currency_graph)
