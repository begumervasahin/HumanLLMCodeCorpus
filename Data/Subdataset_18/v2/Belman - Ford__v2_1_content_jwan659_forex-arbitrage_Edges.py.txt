class Edge:
    def __init__(self, start, target, weight):
        """
        Initializes an Edge instance.

        :param start: The start node of the edge.
        :param target: The target node of the edge.
        :param weight: The weight of the edge.
        """
        self.start = start
        self.target = target
        self.weight = weight
    
    def get_weight(self):
        """
        Returns the weight of the edge.
        :return: The weight of the edge.
        """
        return self.weight
    
    def set_weight(self, weight):
        """
        Sets the weight of the edge.
        :param weight: The new weight of the edge.
        """
        self.weight = weight
    
    def get_start(self):
        """
        Returns the start node of the edge.
        :return: The start node.
        """
        return self.start
    
    def set_start(self, start):
        """
        Sets the start node of the edge.
        :param start: The new start node.
        """
        self.start = start
    
    def get_target(self):
        """
        Returns the target node of the edge.
        :return: The target node.
        """
        return self.target
    
    def set_target(self, target):
        """
        Sets the target node of the edge.
        :param target: The new target node.
        """
        self.target = target

# Example usage:
if __name__ == "__main__":
    # Create an edge from node 'A' to node 'B' with a weight of 5
    edge = Edge("A", "B", 5)
    
    # Print the initial values
    print("Start:", edge.get_start())  # Output: A
    print("Target:", edge.get_target())  # Output: B
    print("Weight:", edge.get_weight())  # Output: 5

    # Update the edge's properties
    edge.set_start("C")
    edge.set_target("D")
    edge.set_weight(10)
    
    # Print the updated values
    print("Updated Start:", edge.get_start())  # Output: C
    print("Updated Target:", edge.get_target())  # Output: D
    print("Updated Weight:", edge.get_weight())  # Output: 10
