import itertools

def generate_binary_combinations(num_bits):
    """
    Generate all binary combinations of a given length.
    
    Parameters:
    num_bits (int): The length of the binary strings to generate.
    
    Returns:
    list of str: A list of binary strings.
    """
    # Generate all possible combinations of 0s and 1s
    tuples = itertools.product('01', repeat=num_bits)
    return [''.join(t) for t in tuples]

def compute_crc(message, divisor):
    """
    Compute the CRC checksum for a given message and divisor.
    
    Parameters:
    message (str): The binary message to be checked.
    divisor (str): The binary divisor used for CRC calculation.
    
    Returns:
    str: The computed CRC checksum.
    """
    # Append zeros to the message for CRC calculation
    pad = '0' * (len(divisor) - 1)
    message = message + pad
    message = list(message)
    divisor = list(divisor)
    
    # Perform the CRC division
    for i in range(len(message) - len(pad)):
        if message[i] == '1':  # Division is performed only if the current bit is 1
            for j in range(len(divisor)):
                message[i + j] = str(int(message[i + j]) ^ int(divisor[j]))

    return ''.join(message[-len(pad):])

def find_crc_collisions(message, divisor, prefix_list=None):
    """
    Find all binary strings that produce the same CRC as the given message.
    
    Parameters:
    message (str): The original binary message.
    divisor (str): The binary divisor used for CRC calculation.
    prefix_list (list of str, optional): Prefixes to prepend to the binary strings. 
                                          If None, all possible prefixes are used.
    
    Returns:
    list of str: Binary strings that produce the same CRC as the given message.
    """
    # Compute the CRC for the original message
    target_crc = compute_crc(message, divisor)
    
    # Generate binary combinations for collision candidates
    candidate_combinations = generate_binary_combinations(len(divisor))
    
    # Initialize result list
    collisions = []

    # If no prefix list is provided, generate all possible prefixes
    if prefix_list is None:
        prefix_list = generate_binary_combinations(len(message) - len(divisor))
    
    # Check for collisions
    for prefix in prefix_list:
        for candidate in candidate_combinations:
            combined_message = prefix + candidate
            if compute_crc(combined_message, divisor) == target_crc:
                collisions.append(combined_message)
    
    return collisions

# Example usage
if __name__ == "__main__":
    original_message = "11010110"
    crc_divisor = "10011"
    prefixes = ["111"]
    
    # Find collisions
    collision_results = find_crc_collisions(original_message, crc_divisor, prefixes)
    
    # Display results
    print("Collisions found:")
    for collision in collision_results:
        print(collision)
