import csv
import numpy as np

def split_data(ifname, delimiter=None, has_header=False, columns=None, seed=42, fraction=0.15):
    """
    Imports a tab/comma/semi-colon/... separated data file as an array of
    floating point numbers.
    
    Parameters
    ----------
    ifname : str
        Filename/path of the data file.
    delimiter : str, optional
        Delimiter of data values (default is '\t').
    has_header : bool, optional
        Indicates if the data-file has a header line (default is False).
    columns : list of int, optional
        A list of integers specifying which columns of the file to import (counting from 0).
    seed : int, optional
        The seed for the pseudo-random number generator (default is 42).
    fraction : float, optional
        The fraction of data to be used for testing (default is 0.15).
    
    Returns
    -------
    training_data_as_array : numpy.ndarray
        The training data as a numpy array.
    test_data_as_array : numpy.ndarray
        The test data as a numpy array.
    field_names : list of str or None
        The header fields if `has_header` is True, else None.
    """
    
    # Set the random seed for reproducibility
    np.random.seed(seed)
    
    # Read the data from the file
    with open(ifname, 'r') as ifile:
        datareader = csv.reader(ifile, delimiter=delimiter if delimiter else '\t')
        
        # Read the header if present
        if has_header:
            field_names = next(datareader)
        else:
            field_names = None
        
        # Collect the data into a list
        data = list(datareader)
    
    # Select only the specified columns if provided
    if columns is not None:
        data = [[row[c] for c in columns] for row in data]
    
    # Convert data to numpy array of floats
    data = np.array(data).astype(float)
    
    # Get the number of rows in the data
    num_rows = data.shape[0]
    
    # Generate a list of row indices and shuffle them
    indices = np.arange(num_rows)
    np.random.shuffle(indices)
    
    # Determine the size of the test set
    test_size = int(fraction * num_rows)
    
    # Split the indices into test and training indices
    test_indices = indices[:test_size]
    train_indices = indices[test_size:]
    
    # Split the data into test and training sets using the indices
    test_data = data[test_indices]
    training_data = data[train_indices]
    
    # Print some information about the split
    print(f"There are {len(training_data)} training entries")
    print(f"There are {len(test_data)} test entries")
    print(f"Each row has {len(training_data[0])} elements")
    
    return training_data, test_data, field_names
