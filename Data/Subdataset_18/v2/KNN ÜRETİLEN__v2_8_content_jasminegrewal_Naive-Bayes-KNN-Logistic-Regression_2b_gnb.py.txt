import csv
import random
import math

def divide(data):
    """
    Divide the data by class labels.

    Parameters:
    data (list): The dataset.

    Returns:
    dict: A dictionary where keys are class labels and values are lists of instances.
    """
    label = {}
    for element in data:
        if element[-1] not in label:
            label[element[-1]] = []
        label[element[-1]].append(element)
    return label

def mean(numbers):
    """
    Calculate the mean of a list of numbers.

    Parameters:
    numbers (list): The list of numbers.

    Returns:
    float: The mean of the list of numbers.
    """
    return sum(numbers) / float(len(numbers))

def stdev(numbers):
    """
    Calculate the standard deviation of a list of numbers.

    Parameters:
    numbers (list): The list of numbers.

    Returns:
    float: The standard deviation of the list of numbers.
    """
    avg = mean(numbers)
    variance = sum([pow(x - avg, 2) for x in numbers]) / float(len(numbers) - 1)
    return math.sqrt(variance)

def calcparams(data):
    """
    Calculate the mean and standard deviation for each feature.

    Parameters:
    data (list): The dataset.

    Returns:
    list: A list of tuples containing the mean and standard deviation for each feature.
    """
    params = [(mean(feature), stdev(feature)) for feature in zip(*data)]
    del params[-1]
    return params

def divbylabel(data):
    """
    Calculate the parameters for each class label.

    Parameters:
    data (list): The dataset.

    Returns:
    dict: A dictionary where keys are class labels and values are lists of parameters.
    """
    classes = divide(data)
    params = {}
    for class_label, instances in classes.items():
        params[class_label] = calcparams(instances)
    return params

def calcProb(x, mean, stdev):
    """
    Calculate the Gaussian probability density function for a given value.

    Parameters:
    x (float): The value.
    mean (float): The mean of the feature.
    stdev (float): The standard deviation of the feature.

    Returns:
    float: The probability density.
    """
    exponent = math.exp(-(math.pow(x - mean, 2) / (2 * math.pow(stdev, 2))))
    return (1 / (math.sqrt(2 * math.pi) * stdev)) * exponent

def calcClssProb(parameters, sample):
    """
    Calculate the class probabilities for a given sample.

    Parameters:
    parameters (dict): The parameters for each class.
    sample (list): The sample to classify.

    Returns:
    dict: The probabilities for each class.
    """
    probs = {}
    for class_label, params in parameters.items():
        probs[class_label] = 1
        for i in range(len(params)):
            mean, stdev = params[i]
            x = sample[i]
            prob_density = calcProb(x, mean, stdev)
            probs[class_label] *= prob_density
    return probs

def predict(parameters, sample):
    """
    Predict the class label for a given sample.

    Parameters:
    parameters (dict): The parameters for each class.
    sample (list): The sample to classify.

    Returns:
    int: The predicted class label.
    """
    probabilities = calcClssProb(parameters, sample)
    best_label, best_prob = None, -1
    for class_label, probability in probabilities.items():
        if best_label is None or probability > best_prob:
            best_prob = probability
            best_label = class_label
    return best_label

def getPredictions(parameters, test_set):
    """
    Get predictions for the test set.

    Parameters:
    parameters (dict): The parameters for each class.
    test_set (list): The test set.

    Returns:
    list: The predicted class labels.
    """
    return [predict(parameters, sample) for sample in test_set]

def main():
    """
    Main function to load the data, train the model, and make predictions.
    """
    data = []
    with open('data.csv', 'r') as csvfile:
        reader = csv.reader(csvfile)
        raw_data = list(reader)
        for row in raw_data:
            for i in range(4):
                if i == 3:
                    row[i] = 1 if row[i] == 'M' else 2
                row[i] = float(row[i])
            data.append(row)

    # Calculate parameters for each class
    parameters = divbylabel(data)
    print('Summary by class value:', parameters)

    # Example test set for prediction
    test_set = [[5.1, 3.5, 1.4, 0.2]]  # Replace with actual test data input

    # Get predictions
    predictions = getPredictions(parameters, test_set)
    for prediction in predictions:
        print('M' if prediction == 1 else 'W')

if __name__ == "__main__":
    main()
