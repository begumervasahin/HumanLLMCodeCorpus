import os
import errno

def split_extension(path):
    """
    Splits the file path into the file name and extension.

    Args:
        path (str): The file path.

    Returns:
        tuple: A tuple containing the file name and extension.
    """
    file_name, ext = os.path.splitext(path)
    return file_name, ext[1:]

def get_basename(path):
    """
    Returns the base name of the file without the extension.

    Args:
        path (str): The file path.

    Returns:
        str: The base name of the file.
    """
    file_name, _ = split_extension(os.path.basename(path))
    return file_name

def get_filename(path):
    """
    Returns the file name including the extension.

    Args:
        path (str): The file path.

    Returns:
        str: The file name.
    """
    return os.path.basename(path)

def get_file_extension(path):
    """
    Returns the file extension.

    Args:
        path (str): The file path.

    Returns:
        str: The file extension.
    """
    return split_extension(path)[1]

def append_to_filename(path, suffix):
    """
    Appends a suffix to the file name before the extension.

    Args:
        path (str): The file path.
        suffix (str): The suffix to add.

    Returns:
        str: The modified file path.
    """
    file_name, ext = os.path.splitext(path)
    return f"{file_name}{suffix}{ext}"

def change_extension(path, new_extension):
    """
    Changes the file extension.

    Args:
        path (str): The file path.
        new_extension (str): The new file extension.

    Returns:
        str: The file path with the new extension.
    """
    file_name, _ = os.path.splitext(path)
    return f"{file_name}.{new_extension}"

def get_directory(path):
    """
    Returns the absolute path of the directory containing the file.

    Args:
        path (str): The file path.

    Returns:
        str: The directory path.
    """
    return os.path.dirname(os.path.abspath(path))

def get_file_size(path):
    """
    Returns the size of the file in bytes.

    Args:
        path (str): The file path.

    Returns:
        int: The file size in bytes.
    """
    return os.path.getsize(path)

def get_directory_size(path):
    """
    Returns the total size of all files in a directory and its subdirectories.

    Args:
        path (str): The directory path.

    Returns:
        int: The total size in bytes.
    """
    total_size = 0
    for dirpath, _, filenames in os.walk(path):
        for filename in filenames:
            file_path = os.path.join(dirpath, filename)
            total_size += get_file_size(file_path)
    return total_size

def generate_output_file_path(directory):
    """
    Generates the output file path with the '.huffman' extension in the current working directory.

    Args:
        directory (str): The input directory path.

    Returns:
        str: The output file path.
    """
    dir_name = os.path.basename(os.path.normpath(directory))
    return os.path.join(os.getcwd(), f"{dir_name}.huffman")

def generate_huffman_file_path(path, is_directory=False):
    """
    Generates the output file path with the '.huffman' extension.

    Args:
        path (str): The input file or directory path.
        is_directory (bool): Whether the input is a directory.

    Returns:
        str: The output file path.
    """
    if is_directory:
        return generate_output_file_path(path)
    return change_extension(path, "huffman")

def get_files_to_process(path):
    """
    Determines the files to process and their total size based on the input path.

    Args:
        path (str): The input file or directory path.

    Returns:
        tuple: A tuple containing the files dictionary, total size, and output file path.
    """
    if os.path.isdir(path):
        files, total_size = list_files(path)
        output_path = generate_huffman_file_path(path, is_directory=True)
    elif os.path.isfile(path):
        files = {get_filename(path): path}
        total_size = get_file_size(path)
        output_path = generate_huffman_file_path(path)
    else:
        raise FileNotFoundError(f"The specified path does not exist: {path}")
    
    return files, total_size, output_path

def list_files(path):
    """
    Lists all files in a directory and its subdirectories, along with their total size.

    Args:
        path (str): The directory path.

    Returns:
        tuple: A tuple containing the files dictionary and total size in bytes.
    """
    files = {}
    total_size = 0
    parent_dir = os.path.dirname(os.path.abspath(path))
    
    for dirpath, _, filenames in os.walk(path):
        for filename in filenames:
            file_path = os.path.join(dirpath, filename)
            relative_path = os.path.relpath(file_path, parent_dir)
            files[relative_path] = file_path
            total_size += get_file_size(file_path)
    
    return files, total_size

def create_directory_if_not_exists(path):
    """
    Creates the directory for the specified path if it does not exist.

    Args:
        path (str): The file or directory path.
    """
    directory = os.path.dirname(path)
    if directory and not os.path.exists(directory):
        try:
            os.makedirs(directory)
        except OSError as e:
            if e.errno != errno.EEXIST:
                raise

def invert_dictionary(d):
    """
    Inverts the keys and values of a dictionary.

    Args:
        d (dict): The dictionary to invert.

    Returns:
        dict: The inverted dictionary.
    """
    return {v: k for k, v in d.items()}
