from dotenv import load_dotenv
import os
import psycopg2
import pandas as pd
import time

# Initialize global variables for the database connection
conn = None

def load_environment_variables():
    """
    Load database connection parameters from environment variables.
    """
    global hostname, port, database, username, password
    hostname = os.getenv('hostname')
    port = os.getenv('port')
    database = os.getenv('database')
    username = os.getenv('username')
    password = os.getenv('password')

def connect_to_database():
    """
    Establish a connection to the database using the loaded environment variables.
    """
    global conn
    conn = psycopg2.connect(
        host=hostname, 
        database=database, 
        user=username, 
        password=password
    )

def fetch_table_data(query):
    """
    Fetch data from the database using the provided SQL query.
    """
    return pd.read_sql(query, con=conn)

def main():
    # Load environment variables
    load_dotenv()
    load_environment_variables()
    
    # Establish the database connection
    connect_to_database()
    
    # Measure the time taken to retrieve and merge data
    start_time = time.time()
    
    # Fetch data from the studies and keywords tables
    studies_df = fetch_table_data('SELECT * FROM studies')
    keywords_df = fetch_table_data('SELECT * FROM keywords')
    
    # Merge the studies and keywords dataframes on the 'nct_id' column
    merged_result = pd.merge(studies_df, keywords_df, how='right', on='nct_id')
    
    # Calculate the time taken to perform the operations
    select_time = time.time()
    elapsed_time = select_time - start_time
    
    # Print the elapsed time and the merged result
    print(f"Time taken: {elapsed_time} seconds")
    print(merged_result)

# Entry point of the script
if __name__ == '__main__':
    main()
