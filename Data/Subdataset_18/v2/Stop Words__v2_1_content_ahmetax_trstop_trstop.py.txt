# Dictionary to store word frequencies
dictionary = {}

# Load the dictionary from the file
with open("dosyalar/derlemtr2016-10000.txt", 'r', encoding='utf-8') as fdict:
    for line in fdict:
        if line[0] not in ['0', '1']:
            continue
        freq, word = line.strip().split()
        dictionary[word] = int(freq)

def is_stop_word(word):
    """
    Checks if the word is in the dictionary, returns True if so.
    
    Args:
    word (str): The word to check.
    
    Returns:
    bool: True if the word is in the dictionary, False otherwise.
    """
    return word in dictionary

def get_word_freq(word):
    """
    Returns the usage frequency of the word. Returns 0 if the word is not in the dictionary.
    
    Args:
    word (str): The word to check.
    
    Returns:
    int: The frequency of the word in the dictionary, or 0 if the word is not found.
    """
    return dictionary.get(word, 0)

if __name__ == '__main__':
    with open("dosyalar/turkce-stop-words", 'r', encoding='utf-8') as ftest:
        for word in ftest:
            word = word.strip()
            if not is_stop_word(word):
                print(f'{word} sözlükte yok')
