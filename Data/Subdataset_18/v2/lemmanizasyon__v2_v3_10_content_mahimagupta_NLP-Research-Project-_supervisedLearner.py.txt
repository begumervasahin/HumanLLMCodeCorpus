# Define input and output file paths
hindi_dic_file = './datasets/hi_hdtb-ud-dev.conllu'
hindi_out_file = './outputs/train_data.txt'

def is_lemma_in(lemmas, lemma):
    """
    Check if a lemma is already present in the dataset.

    Parameters:
    lemmas (list): A list of existing lemmas.
    lemma (dict): A lemma to check for existence.

    Returns:
    bool: True if the lemma is present, False otherwise.
    """
    for existing_lemma in lemmas:
        if all(existing_lemma[key] == lemma[key] for key in ("root", "tag1", "tag2")):
            return True
    return False

def serialize_lemma(lemma):
    """
    Serialize a lemma into a string format.

    Parameters:
    lemma (dict): A lemma to serialize.

    Returns:
    str: The serialized string representation of the lemma.
    """
    return '[{}, {}, {}]'.format(lemma["root"], lemma["tag1"], lemma["tag2"])

def read_training_data(input_training_file, dest_file=None):
    """
    Read training data from a file and return a dataset.

    Parameters:
    input_training_file (str): Path to the input training file.
    dest_file (str, optional): Path to the destination file to save processed data.

    Returns:
    dict: A dictionary containing the dataset.
    """
    dataset = {}
    
    with open(input_training_file, 'r', encoding='utf-8') as hindi_file:
        for line in hindi_file:
            # Skip lines that are too short or are comments
            if len(line) < 5 or line.startswith('#'):
                continue
            
            # Extract the relevant fields from the line
            word, root, tag1, tag2 = line.split("\t")[1:1+4]
            
            # Initialize the word entry if it does not exist
            if word not in dataset:
                dataset[word] = []
            
            # Create a new lemma dictionary
            lemma = {"root": root, "tag1": tag1, "tag2": tag2}
            
            # Add the lemma if it's not already in the list
            if not is_lemma_in(dataset[word], lemma):
                dataset[word].append(lemma)

    # Save the processed dataset to the destination file if provided
    if dest_file is not None:
        with open(dest_file, 'w', encoding='utf-8') as output_file:
            for word, lemmas in dataset.items():
                serialized_lemmas = ', '.join(serialize_lemma(lemma) for lemma in lemmas)
                output_file.write(f'{word}:{serialized_lemmas}\n')

    return dataset

def supervised_learn():
    """
    Perform supervised learning by reading and processing training data,
    then printing details about each word and its corresponding lemmas.
    """
    training_data = read_training_data(hindi_dic_file, hindi_out_file)
    
    for word, lemmas in training_data.items():
        print(f'Processing word: {word} with {len(lemmas)} lemmas')

# Execute the supervised learning function
if __name__ == "__main__":
    supervised_learn()
