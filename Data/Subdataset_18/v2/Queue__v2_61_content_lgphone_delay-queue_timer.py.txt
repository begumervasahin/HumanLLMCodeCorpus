import time
import json
import uuid
from delay_queue.utils import redis_client, release_redis_lock
from delay_queue.config import (
    JOB_POOL_KEY, DELAY_POOL_KEY, READY_POOL_KEY,
    TIMER_LOCK_KEY, TIMER_LOCK_EXP_TIME,
    JOB_LOCK_KEY_PREFIX, JOB_LOCK_EXP_TIME
)

def process_tasks():
    """
    Processes tasks from the DELAY_POOL_KEY and moves them to READY_POOL_KEY
    if they are ready to be executed. Updates the task status and handles Redis locks.
    """
    lock_value = uuid.uuid4().hex[:20]
    
    # Acquire lock for the timer function
    if redis_client.set(TIMER_LOCK_KEY, lock_value, ex=TIMER_LOCK_EXP_TIME, nx=True):
        current_time = int(time.time() * 1000)  # Current time in milliseconds
        
        # Get tasks from the delay pool that are due for processing
        task_ids = redis_client.zrangebyscore(DELAY_POOL_KEY, '-inf', current_time)
        task_ids = [task_id.decode('utf-8') for task_id in task_ids]
        
        for task_id in task_ids:
            print(f'Processing delayed task: {task_id}')
            
            # Get task details from the job pool
            task_data_raw = redis_client.hget(JOB_POOL_KEY, task_id)
            if task_data_raw:
                task_data = json.loads(task_data_raw.decode('utf-8'))
                
                # Check if the task is still in 'delay' status
                if task_data['status'] == 'delay':
                    job_lock_key = JOB_LOCK_KEY_PREFIX + task_data['id']
                    job_lock_value = uuid.uuid4().hex[:20]
                    
                    # Acquire lock for the job
                    if redis_client.set(job_lock_key, job_lock_value, ex=JOB_LOCK_EXP_TIME, nx=True):
                        # Move task from delay pool to ready pool
                        redis_client.zrem(DELAY_POOL_KEY, task_data['id'])
                        redis_client.lpush(READY_POOL_KEY, task_data['id'])
                        
                        # Update task status to 'ready'
                        task_data['status'] = 'ready'
                        redis_client.hset(JOB_POOL_KEY, task_data['id'], json.dumps(task_data))
                        
                        # Release job lock
                        if not release_redis_lock(job_lock_key, job_lock_value):
                            print('Failed to release job lock')
        
        # Release timer lock
        if not release_redis_lock(TIMER_LOCK_KEY, lock_value):
            print('Failed to release timer lock')

def timer_func():
    """
    Continuously checks and processes tasks every 100 milliseconds.
    """
    while True:
        process_tasks()
        time.sleep(0.1)

if __name__ == "__main__":
    print('Starting timer function')
    try:
        timer_func()
    except KeyboardInterrupt:
        print('Stopping timer function')
        exit(0)
