def fizz_buzz_counter():
    """
    Counts and categorizes numbers in a user-defined range according to FizzBuzz rules.
    """
    # Initialize counters
    fizzes = 0
    buzzes = 0
    fizzbuzzes = 0
    normal = 0
    
    while True:
        try:
            # Prompt user for input and parse start and stop values
            input_values = input("Enter start and stop values (separated by a space): ").split()
            min_limit, max_limit = int(input_values[0]), int(input_values[1])
            break
        except (ValueError, IndexError):
            print("Invalid input. Please enter two integers separated by a space.")
    
    # Determine the step direction based on range
    step = 1 if min_limit <= max_limit else -1
    
    # Process each number in the specified range
    for number in range(min_limit, max_limit + step, step):
        if number % 3 == 0 and number % 5 != 0:
            print("fizz", end=" ")
            fizzes += 1
        elif number % 5 == 0 and number % 3 != 0:
            print("buzz", end=" ")
            buzzes += 1
        elif number % 3 == 0 and number % 5 == 0:
            print("fizzbuzz", end=" ")
            fizzbuzzes += 1
        else:
            print(number, end=" ")
            normal += 1
    
    # Output the results
    print("\nFizzes in range:", fizzes)
    print("Buzzes in range:", buzzes)
    print("Fizzbuzzes in range:", fizzbuzzes)
    print("Normal numbers in range:", normal)

if __name__ == "__main__":
    fizz_buzz_counter()
