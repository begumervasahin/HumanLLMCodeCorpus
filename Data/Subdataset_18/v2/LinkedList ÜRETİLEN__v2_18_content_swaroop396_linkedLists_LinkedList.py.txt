class Node:
    def __init__(self, data=None, link=None):
        """
        Initialize a new node with data and an optional link to the next node.

        Args:
            data: The data to store in the node.
            link (Node): The next node in the linked list.
        """
        self.data = data
        self.link = link

    def update_data(self, data):
        """
        Update the data stored in the node.

        Args:
            data: The new data to store in the node.
        """
        self.data = data

    def set_link(self, node):
        """
        Set the link to the next node.

        Args:
            node (Node): The next node in the linked list.
        """
        self.link = node

    def get_data(self):
        """
        Get the data stored in the node.

        Returns:
            The data stored in the node.
        """
        return self.data

    def get_next_node(self):
        """
        Get the next node in the linked list.

        Returns:
            Node: The next node in the linked list.
        """
        return self.link

class LinkedList:
    def __init__(self):
        """
        Initialize an empty linked list.
        """
        self.head = None

    def add_to_start(self, data):
        """
        Add a new node with the given data to the start of the linked list.

        Args:
            data: The data to store in the new node.
        """
        new_node = Node(data)
        new_node.set_link(self.head)
        self.head = new_node

    def add_to_end(self, data):
        """
        Add a new node with the given data to the end of the linked list.

        Args:
            data: The data to store in the new node.
        """
        new_node = Node(data)
        if not self.head:
            self.head = new_node
            return
        current = self.head
        while current.get_next_node():
            current = current.get_next_node()
        current.set_link(new_node)

    def display(self):
        """
        Display the data in the linked list.
        """
        current = self.head
        if not current:
            print("Empty List!!!")
            return
        while current:
            print(current.get_data(), end=" ")
            current = current.get_next_node()
            if current:
                print("-->", end=" ")
        print()

    def length(self):
        """
        Get the length of the linked list.

        Returns:
            int: The number of nodes in the linked list.
        """
        current = self.head
        size = 0
        while current:
            size += 1
            current = current.get_next_node()
        return size

    def index(self, data):
        """
        Get the index of the node containing the given data.

        Args:
            data: The data to find in the linked list.

        Returns:
            int: The index of the node containing the data, or -1 if not found.
        """
        current = self.head
        position = 0
        while current:
            if current.get_data() == data:
                return position
            current = current.get_next_node()
            position += 1
        return -1

    def remove(self, item):
        """
        Remove the first occurrence of the node containing the given data.

        Args:
            item: The data to remove from the linked list.

        Returns:
            bool: True if the item was found and removed, False otherwise.
        """
        current = self.head
        previous = None
        found = False
        while current and not found:
            if current.get_data() == item:
                found = True
            else:
                previous = current
                current = current.get_next_node()
        if found:
            if previous is None:
                self.head = current.get_next_node()
            else:
                previous.set_link(current.get_next_node())
        return found

    def get_max(self):
        """
        Get the maximum value in the linked list.

        Returns:
            The maximum value in the linked list, or None if the list is empty.
        """
        if not self.head:
            return None
        current = self.head
        largest = current.get_data()
        while current:
            if current.get_data() > largest:
                largest = current.get_data()
            current = current.get_next_node()
        return largest

    def get_min(self):
        """
        Get the minimum value in the linked list.

        Returns:
            The minimum value in the linked list, or None if the list is empty.
        """
        if not self.head:
            return None
        current = self.head
        smallest = current.get_data()
        while current:
            if current.get_data() < smallest:
                smallest = current.get_data()
            current = current.get_next_node()
        return smallest

    def push(self, data):
        """
        Add a new node with the given data to the end of the linked list.

        Args:
            data: The data to store in the new node.

        Returns:
            bool: True if the node was added successfully.
        """
        self.add_to_end(data)
        return True

    def pop(self):
        """
        Remove and return the last node in the linked list.

        Returns:
            The data of the removed node, or None if the list is empty.
        """
        if not self.head:
            return None
        current = self.head
        previous = None
        while current.get_next_node():
            previous = current
            current = current.get_next_node()
        if previous:
            previous.set_link(None)
        else:
            self.head = None
        return current.get_data()

    def at_index(self, position):
        """
        Get the data at a specific index in the linked list.

        Args:
            position (int): The index to retrieve the data from.

        Returns:
            The data at the specified index, or None if the index is out of range.
        """
        if position < 0 or not self.head:
            return None
        current = self.head
        for _ in range(position):
            if not current.get_next_node():
                return None
            current = current.get_next_node()
        return current.get_data()

    def copy(self):
        """
        Create a copy of the linked list.

        Returns:
            LinkedList: A new linked list that is a copy of the current list.
        """
        new_list = LinkedList()
        current = self.head
        while current:
            new_list.add_to_end(current.get_data())
            current = current.get_next_node()
        return new_list

    def clear(self):
        """
        Clear the linked list.

        Returns:
            bool: True if the list was cleared successfully.
        """
        self.head = None
        return True

    def remove_at_position(self, position):
        """
        Remove the node at a specific index in the linked list.

        Args:
            position (int): The index to remove the node from.

        Returns:
            The data of the removed node, or None if the index is out of range.
        """
        data = self.at_index(position)
        if data is not None:
            self.remove(data)
        return data

    def to_string(self, separator=""):
        """
        Convert the linked list to a string representation.

        Args:
            separator (str): The separator to use between node values.

        Returns:
            str: A string representation of the linked list.
        """
        current = self.head
        result = ""
        while current:
            result += str(current.get_data())
            current = current.get_next_node()
            if current:
                result += separator
        return result

    def count(self, element):
        """
        Count the occurrences of a specific element in the linked list.

        Args:
            element: The element to count.

        Returns:
            int: The number of occurrences of the element in the list.
        """
        current = self.head
        count = 0
        while current:
            if current.get_data() == element:
                count += 1
            current = current.get_next_node()
        return count

    def to_list(self):
        """
        Convert the linked list to a Python list.

        Returns:
            list: A list containing the data of the linked list nodes.
        """
        current = self.head
        result = []
        while current:
            result.append(current.get_data())
            current = current.get_next_node()
        return result

    def to_set(self):
        """
        Convert the linked list to a Python set.

        Returns:
            set: A set containing the data of the linked list nodes.
        """
        current = self.head
        result = set()
        while current:
            result.add(current.get_data())
            current = current.get_next_node()
        return result

    def reverse(self):
        """
        Reverse the linked list in place.
        """
        previous = None
        current = self.head
        while current:
            next_node = current.get_next_node()
            current.set_link(previous)
            previous = current
            current = next_node
        self.head = previous

    def sort(self):
        """
        Sort the linked list in place.
        """
        if not self.head or not self.head.get_next_node():
            return
        sorted_list = None
        current = self.head
        while current:
            next_node = current.get_next_node()
            sorted_list = self.sorted_insert(sorted_list, current)
            current = next_node
        self.head = sorted_list

    def sorted_insert(self, head, node):
        """
        Helper function to insert a node into a sorted linked list.

        Args:
            head (Node): The head of the sorted linked list.
            node (Node): The node to insert.

        Returns:
            Node: The head of the updated sorted linked list.
        """
        if not head or head.get_data() >= node.get_data():
            node.set_link(head)
            return node
        current = head
        while current.get_next_node() and current.get_next_node().get_data() < node.get_data():
            current = current.get_next_node()
        node.set_link(current.get_next_node())
        current.set_link(node)
        return head

    def sorted(self):
        """
        Create a sorted copy of the linked list.

        Returns:
            LinkedList: A new sorted linked list.
        """
        new_list = self.copy()
        new_list.sort()
        return new_list
