from sklearn.neighbors import NearestNeighbors
import numpy as np

def create_dataset():
    """
    Create and return a sample dataset.
    
    Returns:
        np.ndarray: A sample dataset of 2D points.
    """
    return np.array([[-1, -1], [-2, -1], [-3, -2], [1, 1], [2, 1], [3, 2]])

def fit_nearest_neighbors(X):
    """
    Fit the NearestNeighbors model to the dataset.
    
    Args:
        X (np.ndarray): The dataset to fit the model.
    
    Returns:
        NearestNeighbors: The fitted NearestNeighbors model.
    """
    nbrs = NearestNeighbors(n_neighbors=2, algorithm='ball_tree')
    nbrs.fit(X)
    return nbrs

def find_nearest_neighbors(nbrs, X):
    """
    Find the nearest neighbors for the dataset.
    
    Args:
        nbrs (NearestNeighbors): The fitted NearestNeighbors model.
        X (np.ndarray): The dataset to find nearest neighbors for.
    
    Returns:
        tuple: Distances and indices of the nearest neighbors.
    """
    distances, indices = nbrs.kneighbors(X)
    return distances, indices

def main():
    """Main function to demonstrate the usage of NearestNeighbors."""
    # Create a sample dataset
    X = create_dataset()
    
    # Fit the NearestNeighbors model
    nbrs = fit_nearest_neighbors(X)
    
    # Print the fitted model
    print("Fitted NearestNeighbors model:\n", nbrs)
    
    # Find the nearest neighbors
    distances, indices = find_nearest_neighbors(nbrs, X)
    
    # Print the distances and indices of the nearest neighbors
    print("\nDistances to nearest neighbors:\n", distances)
    print("\nIndices of nearest neighbors:\n", indices)

if __name__ == "__main__":
    main()
