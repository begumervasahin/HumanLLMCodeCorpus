import base64

def encode_image_to_base64(file_path: str) -> bytes:
    """
    Reads an image file in binary mode and encodes it to a base64 string.
    
    Parameters:
    file_path (str): The path to the image file.
    
    Returns:
    bytes: The base64 encoded string.
    """
    with open(file_path, "rb") as image_file:
        return base64.b64encode(image_file.read())

def base64_to_integer(encoded_string: bytes) -> int:
    """
    Converts a base64 encoded string to an integer.
    
    Parameters:
    encoded_string (bytes): The base64 encoded string.
    
    Returns:
    int: The corresponding integer value.
    """
    return int.from_bytes(encoded_string, "big")

def decode_base64_to_image(encoded_string: bytes, output_path: str) -> None:
    """
    Decodes a base64 string back to binary and writes it to a new image file.
    
    Parameters:
    encoded_string (bytes): The base64 encoded string.
    output_path (str): The path to save the decoded image file.
    """
    decoded_bytes = base64.b64decode(encoded_string)
    with open(output_path, "wb") as output_file:
        output_file.write(decoded_bytes)

def main() -> None:
    """
    Main function to demonstrate encoding an image to base64, converting to an integer,
    decoding back to binary, and saving to a new image file.
    """
    input_image_path = "rode_in_club.png"
    output_image_path = "new_rode_in_club.png"

    # Encode the image to a base64 string
    encoded_string = encode_image_to_base64(input_image_path)
    print(encoded_string)
    
    # Convert the base64 encoded string to an integer
    encoded_integer = base64_to_integer(encoded_string)
    print(encoded_integer)
    
    # Decode the base64 string back to binary and save to a new image file
    decode_base64_to_image(encoded_string, output_image_path)

if __name__ == "__main__":
    main()
