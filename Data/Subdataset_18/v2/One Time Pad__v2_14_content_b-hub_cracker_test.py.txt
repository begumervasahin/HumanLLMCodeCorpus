import sys

def convert_to_column_id(n: int) -> str:
    """
    Converts a number to an Excel-style column ID.
    
    Parameters:
    n (int): The number to convert.
    
    Returns:
    str: The corresponding Excel-style column ID.
    """
    column_id = ""
    while n > 0:
        n -= 1
        column_id = chr(ord('A') + n % 26) + column_id
        n //= 26
    return column_id

def process_input_file(input_file: str) -> None:
    """
    Processes the input file, converting each number to an Excel-style column ID and printing the result.
    
    Parameters:
    input_file (str): The path to the input file containing the numbers.
    """
    try:
        with open(input_file, 'r') as file:
            for line in file:
                try:
                    n = int(line.strip())
                    column_id = convert_to_column_id(n)
                    print(f"{n}: {column_id}")
                except ValueError:
                    print(f"Error: Invalid number '{line.strip()}'. Ensure the input file contains integers only.")
    except FileNotFoundError:
        print(f"Error: File '{input_file}' not found.")

def main() -> None:
    """
    Main function to handle command-line arguments and process the input file.
    """
    if len(sys.argv) != 2:
        print("Usage: python script.py <input_file>")
        return

    input_file = sys.argv[1]
    process_input_file(input_file)

if __name__ == "__main__":
    main()
