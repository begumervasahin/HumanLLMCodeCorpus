import timeit
import random

# Function to copy a section of the source array to the destination array
def array_copy(src, src_pos, dest, dest_pos, length):
    for i in range(length):
        dest[i + dest_pos] = src[i + src_pos]

# Recursive function to sort an array by splitting it into smaller sub-arrays
def split_sort(array):
    # Base case: an array with 1 or fewer elements is already sorted
    if len(array) <= 1:
        return array
    
    # Split the array into two halves
    mid = len(array) // 2
    left = array[:mid]
    right = array[mid:]
    
    # Recursively sort both halves
    split_sort(left)
    split_sort(right)
    
    # Merge the sorted halves back into the original array
    merge_sort(left, right, array)
    
    return array

# Function to merge two sorted arrays into one sorted array
def merge_sort(left, right, array):
    left_index, right_index, merged_index = 0, 0, 0
    
    # Compare elements from the left and right arrays and merge them into the main array
    while left_index < len(left) and right_index < len(right):
        if left[left_index] <= right[right_index]:
            array[merged_index] = left[left_index]
            left_index += 1
        else:
            array[merged_index] = right[right_index]
            right_index += 1
        merged_index += 1
    
    # Copy any remaining elements from the left and right arrays
    array_copy(left, left_index, array, merged_index, len(left) - left_index)
    array_copy(right, right_index, array, merged_index, len(right) - right_index)

# Generate a list of random numbers, starting with a predefined list
numbers = [7, 4, 3, 8, 1, 12, 9, 14]
for _ in range(5000):
    rand = random.randint(1, 99999)
    numbers.append(rand)

# Measure the time taken to sort the array using the merge sort algorithm
start_time = timeit.default_timer()
split_sort(numbers)
elapsed_time = timeit.default_timer() - start_time

# Print the results: time taken and the sorted array
print("Merge Sort Time: {:.6f} seconds".format(elapsed_time))
print(numbers)
