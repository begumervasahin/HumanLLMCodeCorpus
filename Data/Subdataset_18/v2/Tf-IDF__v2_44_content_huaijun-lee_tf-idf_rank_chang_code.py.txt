import os
from os import path

__author__ = 'huaijun'

def load_txt(file):
    """
    Load the content of a text file.

    Args:
        file (str): Path to the text file.

    Returns:
        str: Content of the file.

    Raises:
        Exception: If any error occurs while reading the file.
    """
    try:
        with open(file, 'r', encoding='utf-8') as file_handle:
            content = file_handle.read()
        return content
    except Exception as e:
        raise e

def main():
    # Get the current working directory
    rootdir = os.getcwd()
    
    # List all directories and files in the current working directory
    dirs = os.listdir(rootdir)
    
    # Filter directories that start with 'C0'
    filedirs = [d for d in dirs if d.startswith('C0')]
    
    # Define the path for the 'sample' directory
    sample_dir = path.join(rootdir, 'sample')
    
    # Create 'sample' directory if it doesn't exist
    if not path.isdir(sample_dir):
        os.mkdir(sample_dir)
    
    # Open the log file for recording errors
    with open(path.join(rootdir, 'wrong_code.txt'), 'a+', encoding='utf-8') as log_file:
        # Iterate over each filtered directory
        for directory in filedirs:
            # Define the path for the done directory inside 'sample'
            done_dir = path.join(sample_dir, directory)
            
            # Create done directory if it doesn't exist
            if not path.isdir(done_dir):
                os.mkdir(done_dir)
            
            # Define the path of the current directory
            dir_path = path.join(rootdir, directory)
            
            # List all files in the current directory
            files = os.listdir(dir_path)
            
            # Process each file
            for file_name in files:
                file_path = path.join(dir_path, file_name)
                try:
                    # Load the content of the file
                    content = load_txt(file_path)
                    
                    # Write the content to a new file in the done directory
                    with open(path.join(done_dir, file_name), 'w', encoding='utf-8') as new_file:
                        new_file.write(content)
                except (IOError, UnicodeDecodeError, FileNotFoundError) as e:
                    # Log any errors that occur
                    log_file.write(f'Error processing {directory}/{file_name}: {str(e)}\n')

if __name__ == "__main__":
    main()
