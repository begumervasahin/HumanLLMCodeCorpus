import heapq
import time
import argparse

def parse_file(filename):
    with open(filename, 'r') as file:
        raw_data = [list(map(float, line.split())) for line in file.readlines()]
    return raw_data

def build_adjacency_list(raw_data):
    min_key = int(min([v[0] for v in raw_data[1:]] + [v[1] for v in raw_data[1:]]))
    max_key = int(max([v[0] for v in raw_data[1:]] + [v[1] for v in raw_data[1:]]))
    adjacency_list = {k: [(v[2], v[0], v[1]) for v in raw_data[1:] if v[0] == k or v[1] == k] for k in range(min_key, max_key + 1)}
    return adjacency_list

def get_valid_source_node(raw_data):
    valid_nodes = set([v[0] for v in raw_data[1:]] + [v[1] for v in raw_data[1:]])
    while True:
        try:
            source = int(input("Source node: "))
            if source in valid_nodes:
                return source
            else:
                print("No such node, please try again.")
        except ValueError:
            print("Invalid input, please enter a valid node.")

def prims_algorithm(adjacency_list, source, total_vertices):
    costs = []
    vertices = set()
    vertices.add(float(source))
    priority_queue = []
    mst_structure = []

    while len(vertices) < total_vertices:
        for item in adjacency_list[source]:
            heapq.heappush(priority_queue, item)
        
        while True:
            weight, u, v = heapq.heappop(priority_queue)
            if (u in vertices) != (v in vertices):
                break
        
        adjacency_list[u].remove((weight, u, v))
        adjacency_list[v].remove((weight, u, v))

        new_vertex = v if u in vertices else u
        vertices.add(new_vertex)
        source = new_vertex

        mst_structure.append((u, v, weight))
        costs.append(weight)

    return sum(costs), mst_structure

def main():
    start_time = time.process_time()
    
    parser = argparse.ArgumentParser(description="Implements Prim's minimum spanning tree algorithm.")
    parser.add_argument("-s", help="Display the structure of the MST", action="store_true")        
    parser.add_argument("-t", help="Display execution time", action="store_true")
    parser.add_argument("filename", help="Path to the input file")
    args = parser.parse_args()
    
    raw_data = parse_file(args.filename)
    total_vertices = int(raw_data[0][0])
    adjacency_list = build_adjacency_list(raw_data)
    source_node = get_valid_source_node(raw_data)
    
    mst_cost, mst_structure = prims_algorithm(adjacency_list, source_node, total_vertices)
    
    print(f"The overall cost of the MST: {mst_cost}")
    if args.s:
        print(f"The structure of the MST: {mst_structure}")
    if args.t:
        print(f"--- {time.process_time() - start_time} seconds ---")

if __name__ == "__main__":
    main()
