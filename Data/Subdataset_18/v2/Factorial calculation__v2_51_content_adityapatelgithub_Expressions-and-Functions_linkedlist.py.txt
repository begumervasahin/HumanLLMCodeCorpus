class LinkedList:
    class Node:
        __slots__ = "_value", "_next"

        def __init__(self, value, next_node=None):
            self._value = value
            self._next = next_node

    def __init__(self):
        self._head = None  # Head of the linked list
        self._tail = None  # Tail of the linked list
        self._length = 0   # Length of the linked list

    def __iter__(self):
        current = self._head
        while current:
            yield current._value
            current = current._next

    def __len__(self):
        return self._length

    def __str__(self):
        return "".join([str(value) for value in self])

    def push(self, value):
        """
        Pushes a value onto the head of the linked list.

        :param value: Value to be added to the list.
        """
        new_node = self.Node(value)
        if not self._head:
            self._head = self._tail = new_node
        else:
            new_node._next = self._head
            self._head = new_node
        self._length += 1

    def pop(self):
        """
        Removes and returns the value at the head of the linked list.

        :return: Value at the head of the list.
        :raises IndexError: If the list is empty.
        """
        if not self._head:
            raise IndexError("pop from empty list")
        value = self._head._value
        self._head = self._head._next
        self._length -= 1
        if not self._head:
            self._tail = None
        return value

    def top(self):
        """
        Returns the value at the tail of the linked list without removing it.

        :return: Value at the tail of the list.
        :raises IndexError: If the list is empty.
        """
        if not self._tail:
            raise IndexError("top from empty list")
        return self._tail._value

    def is_empty(self):
        """
        Checks if the linked list is empty.

        :return: True if the list is empty, False otherwise.
        """
        return self._length == 0


# Example usage
if __name__ == "__main__":
    ll = LinkedList()
    ll.push('a')
    ll.push('b')
    ll.push('c')
    print("List:", str(ll))  # Output: List: cba
    print("Length:", len(ll))  # Output: Length: 3
    print("Top:", ll.top())  # Output: Top: a
    print("Pop:", ll.pop())  # Output: Pop: c
    print("List after pop:", str(ll))  # Output: List after pop: ba
    print("Is empty:", ll.is_empty())  # Output: Is empty: False
