def word_freq(sents_list):
    """
    Calculates word frequency in a list of sentences.
    
    Args:
        sents_list (list): List of sentences to process.
        
    Returns:
        dict: Dictionary with words as keys and their frequencies as values.
    """
    word_freq = {}
    f_word = set()
    
    for sentence in sents_list:
        sentence = str(sentence)
        if "(Read more about how we rate cars.)" in sentence:
            sentence = sentence.replace("(Read more about how we rate cars.)", "")
        
        sentence = sentence.lower()
        punctuation = '[,，。.!?:"@]'
        sentence = re.sub(punctuation, '', sentence)
        
        for word in sentence.split():
            if word in f_word:
                word_freq[word] += 1
            else:
                word_freq[word] = 1
                f_word.add(word)
                
    return word_freq

def en_freq():
    """Fetches English data from the database and writes word frequencies to a file."""
    conn = connect_to_db()
    cursor = conn.cursor()
    
    sql = "SELECT data FROM RAW_DATA"
    cursor.execute(sql)
    result = [row[0] for row in cursor.fetchall()]
    
    cursor.close()
    conn.close()
    
    word_freq_dict = word_freq(result)
    word_freq_sort = sorted(word_freq_dict.items(), key=lambda x: x[1], reverse=True)
    
    with open('./word_freq_en_v2.txt', 'w') as f:
        for index, (key, value) in enumerate(word_freq_sort, start=1):
            f.write(f"{index} : {key} : {value}\n")

def process_cn_translations():
    """Processes Chinese translations to get word frequencies and writes them to a file."""
    with open('./cn_translations.txt', 'r', encoding='utf8') as f:
        k_list = get_words(f.read())
    
    with open('./word_freq_cn_v1.txt', 'w') as f:
        for k in k_list:
