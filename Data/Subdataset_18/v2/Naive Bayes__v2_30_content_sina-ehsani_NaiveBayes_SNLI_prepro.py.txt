import json
import re
import numpy as np
from nltk.stem import PorterStemmer
from nltk.tokenize import word_tokenize
from nltk.stem import WordNetLemmatizer
from data_utils import savedata

# Initialize lemmatizer and stemmer
lemmatizer = WordNetLemmatizer()
ps = PorterStemmer()

def extract_tokens_from_binary_parse(parse):
    """
    Extracts tokens from binary parse strings.
    
    :param parse: Binary parse string
    :return: List of tokens
    """
    return parse.replace('(', ' ').replace(')', ' ').replace('-LRB-', '(').replace('-RRB-', ')').split()

def yield_examples(file_name, skip_no_majority=True, limit=None):
    """
    Yields examples from a JSONL file.
    
    :param file_name: Path to the JSONL file
    :param skip_no_majority: Whether to skip examples with no majority label
    :param limit: Maximum number of examples to yield
    :yield: Tuple containing label, sentence1, and sentence2
    """
    with open(file_name, 'r') as file:
        for i, line in enumerate(file):
            if limit and i >= limit:
                break
            data = json.loads(line)
            label = data['gold_label']
            s1 = ' '.join(extract_tokens_from_binary_parse(data['sentence1_binary_parse']))
            s2 = ' '.join(extract_tokens_from_binary_parse(data['sentence2_binary_parse']))
            if skip_no_majority and label == '-':
                continue
            yield (label, s1, s2)

def clean_data(sentence_list):
    """
    Cleans and preprocesses a list of sentences.
    
    :param sentence_list: List of sentences
    :return: List of cleaned sentences
    """
    clean_sentences = []
    for sentence in sentence_list:
        sentence = re.sub(r"[^a-zA-Z]", " ", sentence)
        words = sentence.lower().split()
        words = [ps.stem(word) for word in words]
        clean_sentences.append(" ".join(words))
    return clean_sentences

def preprocess(file_name, limit=None):
    """
    Preprocesses the dataset.
    
    :param file_name: Path to the dataset file
    :param limit: Maximum number of examples to process
    :return: Tuple containing preprocessed sentence1 list, sentence2 list, and labels array
    """
    raw_data = list(yield_examples(file_name, limit=limit))
    left_sentences = [s1 for _, s1, s2 in raw_data]
    right_sentences = [s2 for _, s1, s2 in raw_data]
    left_cleaned = clean_data(left_sentences)
    right_cleaned = clean_data(right_sentences)
    labels = {'contradiction': 0, 'neutral': 1, 'entailment': 2}
    Y = np.array([labels[label] for label, _, _ in raw_data])
    return left_cleaned, right_cleaned, Y

def main():
    train_data = preprocess('snli_1.0/snli_1.0_train.jsonl')
    test_data = preprocess('snli_1.0/snli_1.0_test.jsonl')
    dev_data = preprocess('snli_1.0/snli_1.0_dev.jsonl')
    savedata(test_data, "test_stem")
    savedata(train_data, "train_stem")
    savedata(dev_data, "dev_stem")

if __name__ == '__main__':
    main()
