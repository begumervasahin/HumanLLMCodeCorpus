import numpy as np
import tensorflow as tf
import matplotlib.pyplot as plt
import matplotlib

# Data preparation
n_observations = 10
xs = np.array([1.1, 1.2, 2.7, 3.2, 4.1, 4.8, 6.1, 7.5, 9.5, 9.8])
ys = np.array([0.6, 2.5, 2.6, 3.8, 8.0, 5.9, 7.5, 5.5, 7.0, 9.8])
ws = np.ones(n_observations)

# TensorFlow placeholders and variables
X = tf.placeholder(tf.float32, name='X')
Y = tf.placeholder(tf.float32, name='Y')
TW = tf.placeholder(tf.float32, name='TW')

# Polynomial regression model initialization
Y_pred = tf.Variable([-0.00029278], name='bias')
weights = [0.46918184, 1.00643885, -0.23585591, 0.01418969]

for pow_i, weight in enumerate(weights, 1):
    W = tf.Variable(weight, name=f'weight_{pow_i}')
    Y_pred = tf.add(tf.multiply(tf.pow(X, pow_i), W), Y_pred)

# Cost function and optimizer
cost = tf.reduce_sum(tf.multiply(tf.pow(Y_pred - Y, 2), TW)) / (n_observations - 1)
optimizer = tf.train.GradientDescentOptimizer(learning_rate=0.1e-8).minimize(cost)

# TensorFlow session
with tf.Session() as sess:
    sess.run(tf.global_variables_initializer())

    # Training the model
    n_epochs = 1
    for epoch_i in range(n_epochs):
        for x, y, w in zip(xs, ys, ws):
            sess.run(optimizer, feed_dict={X: x, Y: y, TW: w})
        
        training_cost = sess.run(cost, feed_dict={X: xs, Y: ys, TW: ws})
        print(f"Epoch {epoch_i}: {training_cost}")

    # Plotting the results
    fig, ax = plt.subplots(1, 1)
    xss = np.linspace(0, 10, 1000)
    y_pred_vals = [sess.run(Y_pred, feed_dict={X: x}) for x in xss]
    ax.plot(xss, y_pred_vals, color='red', label='CWP')

    # Custom plot elements
    cx, cy = xs[3], ys[3]
    ax.plot([0, cx], [0, cy], color='blue', linestyle='dashed', label='General Path')
    ax.plot(np.concatenate([[cx], xs[3:]]), np.concatenate([[cy], ys[3:]]), color='blue', linestyle='dashed')

    # Markers and scatter points
    ax.scatter(0.1, 0.15, color='black', marker='^', label='H-AP', 
               s=matplotlib.rcParams['lines.markersize'] ** 2 + 50, zorder=10)
    ax.scatter(xs[:3], ys[:3], color='black', label='HGN', 
               s=matplotlib.rcParams['lines.markersize'] ** 2 + 10, zorder=10)
    ax.scatter(xs[3:], ys[3:], color='black', marker='*', 
               s=matplotlib.rcParams['lines.markersize'] ** 2 + 50, label='LGN', zorder=10)

    # Grid, labels, and legend
    major_ticks = np.arange(0, 11, 1)
    ax.set_xticks(major_ticks)
    ax.set_yticks(major_ticks)
    ax.grid(which='both')
    ax.grid(which='minor', alpha=0.2)
    ax.grid(which='major', alpha=0.5)
    ax.set_ylim([0, 10])
    ax.set_xlim([0, 10])
    plt.xlabel('x(m)')
    plt.ylabel('y(m)')
    plt.legend(loc='lower right', fontsize='small', framealpha=1)
    plt.grid(ls='dashed')

    # Save and show the plot
    plt.savefig('test.eps', format='eps', dpi=1000)
    plt.show()

    # Print the final weights and bias
    weights_bias = [sess.run(var) for var in tf.trainable_variables()]
    print("Weights and bias:", weights_bias)
