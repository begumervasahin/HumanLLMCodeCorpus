# Initialize the array
arr = [1, 3, 5, 7, 9, 2, 4, 6, 8, 0]

# Implement a simple sorting algorithm (Bubble Sort)
for i in range(len(arr)):
    for j in range(len(arr) - 1):
        if arr[j] > arr[j + 1]:
            # Swap the elements
            arr[j], arr[j + 1] = arr[j + 1], arr[j]

# Print the sorted array
print("Sorted array:")
print(arr)
