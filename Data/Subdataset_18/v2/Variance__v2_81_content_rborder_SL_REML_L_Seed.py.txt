import numpy as np
from scipy import linalg as la
from scipy.sparse import linalg as spla

def L_Seed(
        A: spla.LinearOperator,
        B: np.ndarray,
        tol=5e-3,
        p_freq=5,
        maxit=50,
        verbose=True
    ) -> dict:
    """
    This function implements an iterative algorithm to solve a linear system.
    
    Parameters:
    A (spla.LinearOperator): The linear operator.
    B (np.ndarray): The right-hand side matrix.
    tol (float): The convergence tolerance. Default is 5e-3.
    p_freq (int): The frequency of printing progress. Default is every 5 iterations.
    maxit (int): The maximum number of iterations. Default is 50.
    verbose (bool): If True, prints progress information. Default is True.

    Returns:
    dict: A dictionary containing the matrices U, β, δ, and ρ from the iterations.
    """
    n, t = A.shape[0], B.shape[1]
    
    # Initialize matrices and vectors
    X = np.zeros((n, t, maxit))
    R = np.zeros((n, t, maxit))
    U = np.zeros((n, t, maxit))
    ρ = np.zeros((t, maxit))
    β = np.zeros((t, maxit))
    ω = np.zeros((t, maxit))
    γ = np.ones((t, maxit))
    δ = np.zeros((t, maxit))
    
    # Initial calculations
    ρ[:, 0] = np.apply_along_axis(la.norm, 0, B)
    β[:, 0] = ρ[:, 0]
    R[:, :, 0] = B
    U[:, :, 0] = B / β[:, 0]
    
    # Iterative process
    j = 0
    converged = False
    
    while j < maxit - 1:
        tmp = A @ U[:, :, j]
        δ[:, j] = np.diag(U[:, :, j].T @ tmp)
        tmp = tmp - U[:, :, j] * δ[:, j] - β[:, j] * U[:, :, j - 1]
        β[:, j + 1] = np.apply_along_axis(la.norm, 0, tmp)
        U[:, :, j + 1] = tmp / β[:, j + 1]
        γ[:, j] = 1 / (δ[:, j] - ω[:, j - 1] / γ[:, j - 1])
        ω[:, j] = (β[:, j + 1] * γ[:, j]) ** 2
        ρ[:, j + 1] = -β[:, j + 1] * γ[:, j] * ρ[:, j]
        R[:, :, j + 1] = ρ[:, j + 1] * U[:, :, j + 1]
        j += 1
        
        # Check convergence
        res_norm = np.amax(np.apply_along_axis(la.norm, 0, R[:, :, j]))
        if verbose and j % p_freq == 0:
            print(f"Error at step {j} is {res_norm:.4f}")
        
        if res_norm <= tol:
            converged = True
            break
    
    # Print convergence status
    if verbose:
        if converged:
            print(f"Converged after {j} iterations.")
        else:
            print(f"Failed to converge after {j} iterations.")
    
    return {
        "U": U[:, :, :j],
        "β": β[:, :j],
        "δ": δ[:, :j],
        "ρ": ρ[:, :j]
    }
