import imaplib
import email
from email.header import decode_header

# Define email credentials and server
EMAIL = 'youremailinhere'
PASSWORD = 'yourpasswordinhere'
IMAP_SERVER = 'imap.gmail.com'

def login_to_email():
    """
    Logs in to the email server using provided credentials.
    
    Returns:
        IMAP4_SSL object if login is successful, None otherwise.
    """
    mail = imaplib.IMAP4_SSL(IMAP_SERVER)
    try:
        mail.login(EMAIL, PASSWORD)
        print("Logged in successfully")
    except imaplib.IMAP4.error:
        print("Login failed")
        return None
    return mail

def search_emails(mail, sender):
    """
    Searches for emails from a specific sender.
    
    Args:
        mail: The IMAP4_SSL object.
        sender (str): The sender's email address.
    
    Returns:
        list: A list of email numbers from the specified sender.
    """
    mail.select("INBOX")
    typ, data = mail.search(None, f'(FROM "{sender}")')
    return data[0].split()

def process_email_body(data):
    """
    Processes the email body to extract the amount of money.
    
    Args:
        data (str): The email body data.
    
    Returns:
        float: The extracted amount of money.
    """
    data = str(data)
    if "=28=C2=A3" in data:
        amount = data.split('=28=C2=A3')[-1].split('=')[0]
    elif "=28=E2=82=AC" in data:
        amount = data.split('=28=E2=82=AC')[-1].split('=')[0]
        amount = float(amount) * 0.88  # Convert EUR to GBP
    else:
        amount = data.split('(US$')[-1].split(')')[0]
        amount = float(amount) * 0.77  # Convert USD to GBP
    return round(float(amount), 2)

def fetch_email_subject(mail, num):
    """
    Fetches the email subject.
    
    Args:
        mail: The IMAP4_SSL object.
        num: The email number.
    
    Returns:
        str: The email subject.
    """
    typ, data = mail.fetch(num, '(BODY.PEEK[HEADER.FIELDS (SUBJECT)])')
    return [x[-1] for x in data]

def fetch_email_date(mail, num):
    """
    Fetches the email date.
    
    Args:
        mail: The IMAP4_SSL object.
        num: The email number.
    
    Returns:
        str: The formatted email date.
    """
    typ, data = mail.fetch(num, '(BODY.PEEK[HEADER.FIELDS (DATE)])')
    msg = email.message_from_bytes(data[0][1])
    date_tuple = email.utils.parsedate_tz(msg["Date"])
    return email.utils.formatdate(email.utils.mktime_tz(date_tuple), localtime=True)

def label_and_delete_email(mail, num, label):
    """
    Labels and deletes the specified email.
    
    Args:
        mail: The IMAP4_SSL object.
        num: The email number.
        label (str): The label to assign to the email.
    """
    mail.store(num, '+X-GM-LABELS', label)
    mail.store(num, '+FLAGS', '\\Deleted')

def main():
    mail = login_to_email()
    if mail is None:
        return

    sender = "congratulations@redbubble.com"
    email_numbers = search_emails(mail, sender)
    amounts = []
    dates = []

    for num in email_numbers:
        subject_data = fetch_email_subject(mail, num)
        amount = process_email_body(subject_data)
        date = fetch_email_date(mail, num)

        amounts.append(amount)
        dates.append(date)

        label_and_delete_email(mail, num, "yourdestinationfolder")

    # Summarize amounts by date
    unique_dates = list(set(dates))
    date_amount_dict = {date: 0 for date in unique_dates}

    for i, date in enumerate(dates):
        date_amount_dict[date] += amounts[i]

    # Print results
    for date, amount in date_amount_dict.items():
        print(f"{date}: £{amount:.2f}")

    mail.expunge()
    mail.close()
    mail.logout()

if __name__ == '__main__':
    main()
