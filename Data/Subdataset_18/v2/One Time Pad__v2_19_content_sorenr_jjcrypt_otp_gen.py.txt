import os
import sys
import string
import random
import math

# Define the set of uppercase letters
LETTERS = string.ascii_uppercase

def generate_otp(num_chars):
    """
    Generate a One-Time Pad (OTP) with the specified number of characters.

    Args:
        num_chars (int): The number of characters to generate.
    """
    just = int(math.log(num_chars) / math.log(10)) + 1
    pad = []

    # Generate the OTP characters and their values
    for i in range(num_chars):
        n = ord(os.urandom(1)) % len(LETTERS)
        c = LETTERS[n]
        v = ' '.join([str(i).rjust(just), str(ord(c) - ord('A')).rjust(2)])
        pad.append(v)

    # Calculate the number of rows and columns for formatted output
    cols = 4
    rows = int(math.ceil(len(pad) / float(cols)))

    # Print the OTP in a formatted grid
    for r in range(rows):
        for c in range(cols):
            p = c * rows + r
            if p < len(pad):
                print(pad[p], end=' ')
            if c < cols - 1:
                print(' ', end=' ')
        print()

def main():
    if len(sys.argv) <= 1:
        print(f"Usage: python {sys.argv[0]} [chars required] > otp.txt")
        sys.exit(-1)
    
    num_chars = int(sys.argv[1])
    generate_otp(num_chars)

if __name__ == "__main__":
    main()
