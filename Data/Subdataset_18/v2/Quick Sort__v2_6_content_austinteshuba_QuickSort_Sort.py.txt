def partition(my_list, start, end):
    """
    Partition the list into two halves around the pivot element.

    :param my_list: The list to be partitioned.
    :param start: The starting index of the partition.
    :param end: The ending index of the partition.
    :return: The index of the pivot element after partitioning.
    """
    pivot = my_list[start]
    left = start + 1
    right = end
    done = False
    
    while not done:
        while left <= right and my_list[left] <= pivot:
            left += 1
        while right >= left and my_list[right] >= pivot:
            right -= 1
        if right < left:
            done = True
        else:
            my_list[left], my_list[right] = my_list[right], my_list[left]
    
    my_list[start], my_list[right] = my_list[right], my_list[start]
    return right

operations_count = 0  # Global variable to count operations

def quicksort(my_list, start, end):
    """
    Perform the QuickSort algorithm on the list.

    :param my_list: The list to be sorted.
    :param start: The starting index of the sort.
    :param end: The ending index of the sort.
    :return: The sorted list.
    """
    global operations_count
    if start < end:
        operations_count += end - start  # Count the operations
        split = partition(my_list, start, end)
        operations_count += abs(start - (split - 1))
        quicksort(my_list, start, split - 1)
        operations_count += abs((split + 1) - end)
        quicksort(my_list, split + 1, end)
    return my_list

def main():
    """
    Main function to read the list from a file, sort it using QuickSort, and print the result.
    """
    try:
        with open("list.txt", "r") as file:
            my_list = [int(line.strip()) for line in file.readlines()]
    except FileNotFoundError:
        print("Error: The file 'list.txt' was not found.")
        return
    except ValueError:
        print("Error: The file 'list.txt' contains non-integer values.")
        return

    sorted_list = quicksort(my_list, 0, len(my_list) - 1)
    print("Sorted list:", sorted_list)
    print("Total operations:", operations_count)

if __name__ == '__main__':
    main()
