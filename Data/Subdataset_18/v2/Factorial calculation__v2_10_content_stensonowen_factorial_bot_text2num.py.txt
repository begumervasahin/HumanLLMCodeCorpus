# Dictionary for small numbers
SMALL_NUMBERS = {
    'zero': 0,
    'one': 1,
    'two': 2,
    'three': 3,
    'four': 4,
    'five': 5,
    'six': 6,
    'seven': 7,
    'eight': 8,
    'nine': 9,
    'ten': 10,
    'eleven': 11,
    'twelve': 12,
    'thirteen': 13,
    'fourteen': 14,
    'fifteen': 15,
    'sixteen': 16,
    'seventeen': 17,
    'eighteen': 18,
    'nineteen': 19,
    'twenty': 20,
    'thirty': 30,
    'forty': 40,
    'fifty': 50,
    'sixty': 60,
    'seventy': 70,
    'eighty': 80,
    'ninety': 90
}

# Dictionary for magnitudes
MAGNITUDES = {
    'thousand': 1000,
    'million': 1000000,
    'billion': 1000000000,
    'trillion': 1000000000000,
    'quadrillion': 1000000000000000,
    'quintillion': 1000000000000000000,
    'sextillion': 1000000000000000000000,
    'septillion': 1000000000000000000000000,
    'octillion': 1000000000000000000000000000,
    'nonillion': 1000000000000000000000000000000,
    'decillion': 1000000000000000000000000000000000,
}

class NumberException(Exception):
    """
    Custom exception for unknown number tokens.
    """
    def __init__(self, msg):
        super().__init__(msg)

def text_to_number(tokens):
    """
    Convert a list of tokens representing a number in words to its numerical value.
    """
    n = 0
    g = 0
    for token in tokens:
        small_value = SMALL_NUMBERS.get(token, None)
        if small_value is not None:
            g += small_value
        elif token == "hundred" and g != 0:
            g *= 100
        else:
            magnitude_value = MAGNITUDES.get(token, None)
            if magnitude_value is not None:
                n += g * magnitude_value
                g = 0
            else:
                raise NumberException(f"Unknown number: {token}")
    return n + g

if __name__ == "__main__":
    # Test cases to verify the functionality
    test_cases = {

        "one hundred twenty three billion four hundred fifty six million seven hundred eighty nine thousand twelve": 123456789012,
        "four decillion": 4000000000000000000000000000000000
    }

    for text, expected in test_cases.items():
        assert text_to_number(text.split()) == expected, f"Test failed for: {text}"
