import random

# Generate a list of 10 random integers between 0 and 99
arr = [random.randrange(100) for _ in range(10)]

def is_sorted(array) -> bool:
    """
    Checks if the array is sorted in ascending order.
    
    Parameters:
    array (list): The list to check.
    
    Returns:
    bool: True if the array is sorted, False otherwise.
    """
    for i in range(len(array) - 1):
        if array[i] > array[i + 1]:
            return False
    return True

def insertion_sort(array):
    """
    Sorts the array using the insertion sort algorithm.
    
    Parameters:
    array (list): The list to be sorted.
    
    Returns:
    list: The sorted list.
    """
    for i in range(1, len(array)):
        key = array[i]
        j = i - 1
        while j >= 0 and array[j] > key:
            array[j + 1] = array[j]
            j -= 1
        array[j + 1] = key
    return array

# Print the original array
print("Original array:", arr)

# Perform insertion sort and print the sorted array
sorted_arr = insertion_sort(arr.copy())
print("Sorted array:", sorted_arr)

# Check if the sorted array is actually sorted and print the result
is_sorted_correctly = is_sorted(sorted_arr)
print("Is the array sorted correctly?", is_sorted_correctly)
