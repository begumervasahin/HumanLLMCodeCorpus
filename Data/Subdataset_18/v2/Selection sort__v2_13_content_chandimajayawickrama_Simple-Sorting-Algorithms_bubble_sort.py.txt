import time

def bubble_sort(array):
    """
    Sorts the input array using the bubble sort algorithm.

    Args:
        array (list): The list of integers to be sorted.

    Returns:
        list: The sorted list.
    """
    start_time = time.time()
    n = len(array)
    
    # If the array is empty, return it as is
    if n == 0:
        return array

    for i in range(n):
        swapped = False
        for j in range(0, n - i - 1):
            if array[j] > array[j + 1]:
                # Swap the elements
                array[j], array[j + 1] = array[j + 1], array[j]
                swapped = True
        # If no elements were swapped, the array is already sorted
        if not swapped:
            break

    print(f"Sorting took {time.time() - start_time:.6f} seconds")
    return array

def main():
    """
    Main function to handle user input and sort the list using bubble sort.
    """
    try:
        # Get user input and convert it to a list of integers
        user_input = input("Enter the numbers to be sorted, separated by spaces: ")
        number_list = [int(x) for x in user_input.split()]
        sorted_list = bubble_sort(number_list)
        print("Sorted list:", sorted_list)
    except ValueError:
        print("Invalid input. Please enter a list of integers separated by spaces.")

if __name__ == "__main__":
    main()
