import random
import timeit
import matplotlib.pyplot as plt

def seq_search(arr, x):
    """Perform a sequential search for x in arr."""
    for i, val in enumerate(arr):
        if val == x:
            return i
    return -1

def bin_search(arr, x):
    """Perform a binary search for x in arr."""
    l, r = 0, len(arr) - 1
    while l <= r:
        m = (l + r) // 2
        if arr[m] < x:
            l = m + 1
        elif arr[m] > x:
            r = m - 1
        else:
            return m
    return -1

def wrapper(func, *args, **kwargs):
    """Wrap a function to pass to timeit."""
    def wrapped():
        return func(*args, **kwargs)
    return wrapped

def perform_search_experiments():
    print("\nSEQUENTIAL SEARCH VS BINARY SEARCH")
    results = []
    timeit_loops = 10

    # Loop over increasing sizes of arrays (10^i)
    for i in range(2, 7):
        array_size = 10 ** i
        arr = tuple(range(array_size))
        print(f"\nARRAY SIZE = {array_size} (10^{i})")

        # Perform 3 trials for each array size
        for trial in range(1, 4):
            random_value = random.randint(0, array_size - 1)
            print(f"TRIAL {trial}")
            print(f"Random generated value: {random_value}")
            
            # Time sequential search
            print("Running SEQ SEARCH...", end="")
            wrapped_seq = wrapper(seq_search, arr, random_value)
            seq_time = timeit.timeit(wrapped_seq, number=timeit_loops)
            print("DONE!")
            print(f"Total time for {timeit_loops} searches: {seq_time:.6f} seconds.")
            
            # Time binary search
            print("Running BIN SEARCH...", end="")
            wrapped_bin = wrapper(bin_search, arr, random_value)
            bin_time = timeit.timeit(wrapped_bin, number=timeit_loops)
            print("DONE!")
            print(f"Total time for {timeit_loops} searches: {bin_time:.6f} seconds.")
            
            # Store results
            results.append({
                'power': i, 
                'trial': trial, 
                'array_size': array_size,
                'random_value': random_value, 
                'seq_time': seq_time, 
                'bin_time': bin_time,
            })

    return results

def plot_results(results):
    """Plot the search times for sequential and binary searches."""
    seq_times = []
    bin_times = []
    powers = []

    for entry in results:
        powers.append(entry['power'])
        seq_times.append(entry['seq_time'])
        bin_times.append(entry['bin_time'])

    # Plot the results
    plt.plot(powers, seq_times, 'rD', label='Seq Search')
    plt.plot(powers, bin_times, 'bo', label='Bin Search')

    plt.xlabel('Size of array (powers of 10)')
    plt.ylabel('Search time (sec)')
    plt.title('Sequential vs. Binary Search Performance')
    plt.grid(True)
    plt.legend(loc='upper left', numpoints=1)

    # Define the axis limits based on the data
    plt.axis([min(powers) - 1, max(powers) + 1, min(min(seq_times), min(bin_times)) - 0.01, max(seq_times) + 0.1])

    # Show the plot
    plt.show()

def main():
    results = perform_search_experiments()
    print("\nTo end the program, close the graph window.\n")
    plot_results(results)

if __name__ == "__main__":
    main()
