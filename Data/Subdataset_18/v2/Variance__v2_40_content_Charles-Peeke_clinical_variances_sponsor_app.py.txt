import os
import psycopg2
import pandas as pd
import matplotlib.pyplot as plt
from dotenv import load_dotenv

# Load environment variables from a .env file
load_dotenv()

# Fetch database connection parameters from environment variables
hostname = os.getenv('hostname')
port = os.getenv('port')
database = os.getenv('database')
username = os.getenv('username')
password = os.getenv('password')

def compute_results():
    """
    Connect to the database, execute a query to join studies and sponsors,
    and return the results as a DataFrame.
    """
    print('Computing results...')
    try:
        # Establish the connection
        conn = psycopg2.connect(
            host=hostname,
            port=port,
            database=database,
            user=username,
            password=password
        )
        # Execute the query and fetch the data
        query = 'SELECT name FROM studies RIGHT JOIN sponsors ON studies.nct_id = sponsors.nct_id'
        df = pd.read_sql(query, con=conn)
    except Exception as e:
        print(f"Error occurred: {e}")
    finally:
        # Always close the connection
        conn.close()
    
    print('Done')
    return df

def get_unique_sponsor_names(df):
    """
    Extract and return a list of unique sponsor names from the DataFrame.
    """
    return df['name'].unique().tolist()

def write_list_to_file(lst, filename='unique_sponsor_names.txt'):
    """
    Write a list of items to a file, each item on a new line.
    """
    with open(filename, 'w') as file:
        for item in lst:
            file.write(f"{item}\n")
    print("New list created")

def plot_merck_names(df):
    """
    Create and display a pie chart of sponsor names containing 'Merck' or 'MSD'.
    """
    merck_names = {}
    for name in df['name']:
        if "Merck" in name or "MSD" in name:
            merck_names[name] = merck_names.get(name, 0) + 1

    # Create a pie chart
    plt.pie(merck_names.values(), labels=merck_names.keys(), autopct='%1.1f%%')
    plt.title('Distribution of Merck and MSD Sponsor Names')
    plt.show()

if __name__ == '__main__':
    # Compute the results from the database
    df = compute_results()
    
    # Get unique sponsor names and write them to a file
    unique_names = get_unique_sponsor_names(df)
    write_list_to_file(unique_names)
    
    # Plot the names related to 'Merck' or 'MSD'
    plot_merck_names(df)
