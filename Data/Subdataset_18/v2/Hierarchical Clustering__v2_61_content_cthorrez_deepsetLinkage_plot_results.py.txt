import sys
import json
import numpy as np
import matplotlib.pyplot as plt
from utils import unequal_mean

def load_config(path):
    """Load configuration parameters from a JSON file."""
    config_path = f'{path}/config.json'
    with open(config_path, 'r') as file:
        config = json.load(file)
    return config

def load_loss_arrays(path, n_trials, loss_type):
    """Load loss arrays (training/validation) for each trial."""
    return [np.load(f'{path}/{loss_type}_losses_{trial}.npy') for trial in range(n_trials)]

def calculate_mean_test_f1(path, n_trials):
    """Calculate the mean test F1 score across all trials."""
    test_f1_scores = [np.load(f'{path}/test_f1_{trial}.npy') for trial in range(n_trials)]
    return np.mean(test_f1_scores)

def plot_training_curve(train_means, val_means, path):
    """Plot and save the training and validation loss curves."""
    epochs = np.arange(len(train_means))
    plt.plot(epochs, train_means, color='blue', label='Training Loss')
    plt.plot(epochs, val_means, color='red', label='Validation Loss')
    plt.xlabel('Epoch')
    plt.ylabel('Loss')
    plt.legend()
    plt.savefig(f'{path}/training_curve.png')
    plt.close()

def main(path):
    # Load the experiment configuration
    config = load_config(path)

    # Load training and validation loss arrays
    train_arrays = load_loss_arrays(path, config['n_trials'], 'train')
    val_arrays = load_loss_arrays(path, config['n_trials'], 'val')

    # Compute and print the mean test F1 score
    mean_test_f1 = calculate_mean_test_f1(path, config['n_trials'])
    print(f'{path} test F1: {mean_test_f1}')

    # Calculate the mean losses using the unequal_mean function
    train_means = unequal_mean(train_arrays)
    val_means = unequal_mean(val_arrays)

    # Plot and save the training curve
    plot_training_curve(train_means, val_means, path)

if __name__ == '__main__':
    if len(sys.argv) != 2:
        print("Usage: python script_name.py <path_to_directory>")
        sys.exit(1)

    # Run the main function with the provided directory path
    main(sys.argv[1])
