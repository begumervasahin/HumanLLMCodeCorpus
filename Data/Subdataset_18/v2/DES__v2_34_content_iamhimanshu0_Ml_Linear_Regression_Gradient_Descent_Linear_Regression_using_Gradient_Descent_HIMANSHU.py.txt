import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# Set the figure size for plots
plt.rcParams['figure.figsize'] = (5.0, 5.0)

# Load the data from a CSV file
data = pd.read_csv('data.csv')
X = data.iloc[:, 0]
Y = data.iloc[:, 1]

# Scatter plot of the data points
plt.scatter(X, Y)
plt.xlabel('X')
plt.ylabel('Y')
plt.title('Data Points')

# Calculate the mean of X and Y
xi = np.mean(X)
yi = np.mean(Y)

# Calculate the differences from the mean
x_xi = X - xi
y_yi = Y - yi

# Calculate the slope (m) and intercept (c) for the least squares regression line
m = np.sum(x_xi * y_yi) / np.sum(x_xi ** 2)
c = yi - m * xi

# Calculate the predicted Y values for the regression line
Y_pred_initial = m * X + c

# Plot the initial regression line
plt.plot(X, Y_pred_initial, color='red', label='Initial Regression Line')
plt.scatter(X, Y, label='Data Points')
plt.xlabel('X')
plt.ylabel('Y')
plt.title('Data Points with Initial Regression Line')
plt.legend()
plt.show()

# Initialize parameters for gradient descent
m = 0
c = 0
learning_rate = 0.0001
epochs = 1000
n = float(len(X))

# Perform gradient descent to find the optimal values of m and c
for _ in range(epochs):
    Y_pred = m * X + c
    D_m = (-2 / n) * sum(X * (Y - Y_pred))
    D_c = (-2 / n) * sum(Y - Y_pred)
    m = m - learning_rate * D_m
    c = c - learning_rate * D_c

# Print the final values of m and c
print(f"Final values: m = {m}, c = {c}")

# Calculate the final predicted Y values
Y_pred = m * X + c

# Scatter plot of the data points with the final regression line
plt.scatter(X, Y, label='Data Points')
plt.plot(X, Y_pred, color='red', label='Final Regression Line')
plt.xlabel('X')
plt.ylabel('Y')
plt.title('Data Points with Final Regression Line')
plt.legend()
plt.show()
