import pandas as pd
import numpy as np
import time


def to_float(x):
    """Convert a value to float."""
    return float(x)


def mutate_dict(f, d):
    """Apply a function to all values in a dictionary."""
    for k, v in d.items():
        d[k] = f(v)


def new_cluster(key, current_tuple):
    """
    Create a new cluster with the given key and tuple.
    
    Args:
        key: The key identifying the new cluster.
        current_tuple: The initial tuple of the new cluster.
    
    Returns:
        A new cluster structure.
    """
    summary = []
    cluster_structure = []
    for k in current_tuple:
        value_summary = {current_tuple[k]: 1}
        summary.append([value_summary])
    cluster_structure.append([key])
    cluster_structure.append(summary)
    return cluster_structure


def add_same(cluster, current_tuple):
    """
    Update the cluster with the same tuple values.
    
    Args:
        cluster: The cluster to be updated.
        current_tuple: The current tuple to update the cluster with.
    """
    for k in current_tuple:
        if current_tuple[k] in cluster[k][0]:
            cluster[k][0][current_tuple[k]] += 1
        else:
            cluster[k][0][current_tuple[k]] = 1


def calculate_similarity(cluster, current_tuple):
    """
    Calculate similarity between the cluster and current tuple.
    
    Args:
        cluster: The cluster to compare against.
        current_tuple: The current tuple to be compared.
    
    Returns:
        The similarity score.
    """
    similarity = 0
    for k in current_tuple:
        support = cluster[k][0].get(current_tuple[k], 0)
        total_support = sum(cluster[k][0].values())
        similarity += support / float(total_support)
    return similarity
