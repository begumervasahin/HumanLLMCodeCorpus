class Node:
    def __init__(self, data):
        """
        Initialize a Node with the given data and set the next node reference to None.

        Args:
            data: The initial data to store in the node.
        """
        self.data = data
        self.next = None

    def get_data(self):
        """
        Get the data stored in the node.

        Returns:
            The data stored in the node.
        """
        return self.data

    def get_next(self):
        """
        Get the reference to the next node.

        Returns:
            The next node reference.
        """
        return self.next

    def set_data(self, data):
        """
        Set the data stored in the node.

        Args:
            data: The new data to store in the node.
        """
        self.data = data

    def set_next(self, next_node):
        """
        Set the reference to the next node.

        Args:
            next_node: The new reference to the next node.
        """
        self.next = next_node
