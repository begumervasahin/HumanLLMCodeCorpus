def main() -> None:
    """
    Main function to print the roster of addresses and names.
    """
    roster = (
        ('ipns address one', 'Name one'),
        ('ipns address two', 'Name two'),
        ('etc', 'etc')
    )

    # Print the roster to verify its contents
    for address, name in roster:
        print(f"Address: {address}, Name: {name}")

if __name__ == "__main__":
    main()
