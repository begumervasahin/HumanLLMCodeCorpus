class MaxHeap:
    def __init__(self):
        """
        Initialize an empty max heap.
        """
        self.heap = []
        self.size = 0

    def _left_child(self, index):
        """
        Return the index of the left child of the node at the given index.

        Args:
            index (int): The index of the node.
        
        Returns:
            int or None: The index of the left child, or None if it doesn't exist.
        """
        left_index = 2 * index + 1
        return left_index if left_index < self.size else None

    def _right_child(self, index):
        """
        Return the index of the right child of the node at the given index.

        Args:
            index (int): The index of the node.
        
        Returns:
            int or None: The index of the right child, or None if it doesn't exist.
        """
        right_index = 2 * index + 2
        return right_index if right_index < self.size else None

    def _max_heapify(self, index):
        """
        Ensure the max-heap property is maintained starting from the given node.

        Args:
            index (int): The index of the node to start heapifying from.
        """
        largest = index
        left = self._left_child(index)
        right = self._right_child(index)

        if left is not None and self.heap[left] > self.heap[largest]:
            largest = left

        if right is not None and self.heap[right] > self.heap[largest]:
            largest = right

        if largest != index:
            self.heap[index], self.heap[largest] = self.heap[largest], self.heap[index]
            self._max_heapify(largest)

    def build_heap(self, array):
        """
        Build a max-heap from the given array.

        Args:
            array (list): The array to build the heap from.
        """
        self.size = len(array)
        self.heap = array[:]
        for i in range(self.size // 2 - 1, -1, -1):
            self._max_heapify(i)

    def extract_max(self):
        """
        Remove and return the maximum element from the heap.

        Returns:
            The maximum element in the heap, or None if the heap is empty.
        """
        if self.size == 0:
            return None

        max_elem = self.heap[0]
        self.heap[0] = self.heap[self.size - 1]
        self.size -= 1
        self._max_heapify(0)
        return max_elem

    def heap_sort(self):
        """
        Perform heap sort and store the sorted elements back in the heap.
        """
        original_size = self.size
        for i in range(self.size - 1, 0, -1):
            self.heap[0], self.heap[i] = self.heap[i], self.heap[0]
            self.size -= 1
            self._max_heapify(0)
        self.size = original_size

    def insert(self, data):
        """
        Insert a new element into the heap.

        Args:
            data: The data to insert into the heap.
        """
        self.heap.append(data)
        self.size += 1
        current = self.size - 1
        while current > 0 and self.heap[current] > self.heap[(current - 1) // 2]:
            self.heap[current], self.heap[(current - 1) // 2] = self.heap[(current - 1) // 2], self.heap[current]
            current = (current - 1) // 2

    def display(self):
        """
        Display the heap.
        """
        print(self.heap)

def main():
    """
    Main function to demonstrate the heap operations.
    """
    print("=== MaxHeap ===")
    sentence = input("Enter a sentence: ")
    words = sentence.split()
    heap = MaxHeap()
    heap.build_heap(words)
    heap.heap_sort()
    heap.display()

if __name__ == '__main__':
    main()
