from collections import defaultdict
from heapq import heappop, heappush

def dijkstra(edges, start, end):
    """Compute the shortest path from start to end using Dijkstra's algorithm."""
    graph = defaultdict(list)
    
    # Build the graph from the list of edges
    for left, right, cost in edges:
        graph[left].append((cost, right))
    
    # Priority queue to store (cost, vertex, path)
    queue = [(0, start, ())]  # Starting with the source node
    seen = set()
    
    while queue:
        # Get the node with the lowest cost
        (cost, vertex, path) = heappop(queue)
        
        if vertex not in seen:
            seen.add(vertex)
            path = (vertex, path)  # Append the current node to the path
            
            # If we have reached the destination node
            if vertex == end:
                return (cost, path)
            
            # Add neighboring nodes to the priority queue
            for cost_to_neighbor, neighbor in graph[vertex]:
                if neighbor not in seen:
                    heappush(queue, (cost + cost_to_neighbor, neighbor, path))
    
    return float("inf")  # Return infinity if there's no path

def print_result(result):
    """Helper function to print the result."""
    cost, path = result
    node_list = []
    
    # Reconstruct the path
    while path:
        node, path = path
        node_list.append(node)
    
    print(f"Cost: {cost}")
    print("Path:", " -> ".join(reversed(node_list)))
