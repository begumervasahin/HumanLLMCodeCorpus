import graphviz
from math import inf

class TemporalEdge:
    def __init__(self, source, destination, time):
        self.source = source
        self.destination = destination
        self.time = time
        self.is_traversed = False

class TemporalGraph:
    def __init__(self):
        self.edges = []
        self.nodes = set()

    def add_edge(self, source, destination, time):
        """
        Add a temporal edge to the graph.

        Args:
            source (str): The source node.
            destination (str): The destination node.
            time (int): The time associated with the edge.
        """
        self.edges.append(TemporalEdge(source, destination, time))
        self.nodes.add(source)
        self.nodes.add(destination)

    def get_nodes(self):
        """
        Get a list of all nodes in the graph.

        Returns:
            list: List of nodes.
        """
        return list(self.nodes)

    def get_neighbors(self, node):
        """
        Get the neighbors of a given node.

        Args:
            node (str): The node to get neighbors for.

        Returns:
            list: List of neighboring nodes.
        """
        neighbors = set()
        for edge in self.edges:
            if edge.source == node:
                neighbors.add(edge.destination)
        return list(neighbors)

    def get_edge_neighbor(self, node, neighbor):
        """
        Get the edges between a node and its neighbor.

        Args:
            node (str): The source node.
            neighbor (str): The neighboring node.

        Returns:
            list: List of TemporalEdge objects.
        """
        return [edge for edge in self.edges if edge.source == node and edge.destination == neighbor]

class TreeNode:
    def __init__(self, name, time):
        self.name = name
        self.time = time
        self.children = []

    def add_node(self, node):
        """
        Add a child node.

        Args:
            node (TreeNode): The child node to be added.
        """
        self.children.append(node)

def draw_tree(root, filename):
    """
    Draw the tree using graphviz.

    Args:
        root (TreeNode): The root of the tree.
        filename (str): The output filename.
    """
    dot = graphviz.Digraph(comment='DFS_v1 Tree')

    def add_edges(node):
        for child in node.children:
            dot.edge(f"{node.name} ({node.time})", f"{child.name} ({child.time})")
            add_edges(child)

    add_edges(root)
    dot.render(filename, view=True)

def find_edge_with_min_time(edges):
    """
    Find the edge with the minimum time.

    Args:
        edges (list): List of TemporalEdge objects.

    Returns:
        TemporalEdge: The edge with the minimum time.
    """
    return min(edges, key=lambda edge: edge.time)

def dfs_v1(current_node):
    """
    Perform Depth-First Search (DFS) on the temporal graph.

    Args:
        current_node (str): The current node in the DFS traversal.
    """

    for neighbor in graph.get_neighbors(current_node):

            edge_min.is_traversed = True

                next_tree_node = TreeNode(edge_min.destination, edge_min.time)
                predecessor[next_tree_node.name] = current_tree_node.name
                current_tree_node.add_node(next_tree_node)
                current_tree_node = next_tree_node
                sigma[edge_min.destination] = edge_min.time
                dfs_v1(edge_min.destination)
