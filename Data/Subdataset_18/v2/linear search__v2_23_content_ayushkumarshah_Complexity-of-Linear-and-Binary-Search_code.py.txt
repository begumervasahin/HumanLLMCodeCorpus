import random
from time import time

def linearSearch(theValues, target):
    """Performs a linear search on the list for the target value."""
    for value in theValues:
        if value == target:
            return True
    return False

def sortedLinearSearch(theValues, target):
    """Performs a linear search on the sorted list for the target value."""
    for value in theValues:
        if value == target:
            return True
        elif value > target:
            return False
    return False

def findSmallest(theValues):
    """Finds the smallest element in the list."""
    smallest = theValues[0]
    for value in theValues[1:]:
        if value < smallest:
            smallest = value
    return smallest

def binarySearch(theValues, target):
    """Performs a binary search on the sorted list for the target value."""
    low, high = 0, len(theValues) - 1
    while low <= high:
        mid = (high + low) // 2
        if theValues[mid] == target:
            return True
        elif target < theValues[mid]:
            high = mid - 1
        else:
            low = mid + 1
    return False

# Generate a list of 100,000 random numbers
myrandomnumber = random.sample(range(1000000), 100000)
mysortedrandomnumber = sorted(myrandomnumber)
randomnumberfromthelist = random.choice(mysortedrandomnumber)

elapsed_unsorted = []
elapsed_sorted = []
elapsed_smallest = []
elapsed_binary = []

# Measure time for each algorithm
for i in range(10000, 100001, 10000):
    # Measure time for unsorted linear search
    start_time = time()
    linearSearch(myrandomnumber[:i], randomnumberfromthelist)
    end_time = time()
    elapsed_unsorted.append(end_time - start_time)
    
    # Measure time for sorted linear search
    start_time = time()
    sortedLinearSearch(mysortedrandomnumber[:i], randomnumberfromthelist)
    end_time = time()
    elapsed_sorted.append(end_time - start_time)
    
    # Measure time for finding the smallest element
    start_time = time()
    findSmallest(myrandomnumber[:i])
    end_time = time()
    elapsed_smallest.append(end_time - start_time)
    
    # Measure time for binary search
    start_time = time()
    binarySearch(mysortedrandomnumber[:i], randomnumberfromthelist)
    end_time = time()
    elapsed_binary.append(end_time - start_time)

# Print results
print("\nUnsorted Linear Search times")
for time_elapsed in elapsed_unsorted:
    print(time_elapsed)

print("\nSorted Linear Search times")
for time_elapsed in elapsed_sorted:
    print(time_elapsed)

print("\nFinding Smallest Element times")
for time_elapsed in elapsed_smallest:
    print(time_elapsed)

print("\nBinary Search times")
for time_elapsed in elapsed_binary:
    print(time_elapsed)
