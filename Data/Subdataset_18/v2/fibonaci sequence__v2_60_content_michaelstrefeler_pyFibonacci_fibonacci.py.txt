from turtle import *

def get_fibonacci_sequence(n):
    """
    Generate a Fibonacci sequence up to n terms.
    
    :param n: Number of terms in the Fibonacci sequence.
    :return: A list containing the Fibonacci sequence.
    """
    sequence = [0, 1]
    while len(sequence) < n:
        sequence.append(sequence[-1] + sequence[-2])
    return sequence

def draw_fibonacci_spiral(sequence):
    """
    Draw the Fibonacci spiral using the turtle graphics.
    
    :param sequence: A list containing the Fibonacci sequence.
    """
    penup()
    goto(135, -20)
    pendown()
    first_one = True
    
    for fib in sequence:
        if fib == 0:
            continue
        if fib == 1:
            draw_square(fib, first_one)
            first_one = False
        elif fib == 21:
            draw_final_square(fib)
            break
        else:
            draw_special_squares(fib)
    
    draw_spiral(sequence)
    done()

def draw_square(fib, first_one):
    """
    Draw a square for the given Fibonacci number.
    
    :param fib: The Fibonacci number to draw.
    :param first_one: A boolean indicating if it is the first '1' in the sequence.
    """
    if first_one:
        forward(25 * fib / 2)
    else:
        forward(25 * fib)
    write(fib)
    forward(25 * fib / 2)
    left(90)
    for _ in range(3):
        forward(25 * fib)
        left(90)

def draw_final_square(fib):
    """
    Draw the final square in the sequence.
    
    :param fib: The final Fibonacci number to draw.
    """
    forward(25 * fib)
    left(90)
    forward(25 * fib)
    left(90)
    forward(25 * fib / 2)
    write(fib)
    forward(25 * fib / 2)
    left(90)

def draw_special_squares(fib):
    """
    Draw squares for special Fibonacci numbers.
    
    :param fib: The special Fibonacci number to draw.
    """
    if fib in [2, 13]:
        draw_special_pattern(fib, "left")
    elif fib == 3:
        draw_special_pattern(fib, "bottom")
    elif fib == 5:
        draw_special_pattern(fib, "top")
    else:
        draw_standard_square(fib)

def draw_special_pattern(fib, direction):
    """
    Draw a special pattern for the given Fibonacci number.
    
    :param fib: The Fibonacci number to draw.
    :param direction: The direction for special pattern.
    """
    forward(25 * fib)
    left(90)
    forward(25 * fib)
    left(90)
    if direction == "left":
        forward(25 * fib)
        left(90)
        forward(25 * fib / 2)
    elif direction == "bottom":
        forward(25 * fib / 2)
        left(90)
        forward(25 * fib)
    elif direction == "top":
        forward(25 * fib / 2)
    write(fib)
    forward(25 * fib / 2)
    left(90)

def draw_standard_square(fib):
    """
    Draw a standard square for the given Fibonacci number.
    
    :param fib: The Fibonacci number to draw.
    """
    forward(25 * fib / 2)
    write(fib)
    forward(25 * fib / 2)
    left(90)
    for _ in range(3):
        forward(25 * fib)
        left(90)

def draw_spiral(sequence):
    """
    Draw the Fibonacci spiral using the turtle graphics.
    
    :param sequence: A list containing the Fibonacci sequence.
    """
    penup()
    goto(135, 5)
    pendown()
    color('red')
    shape('turtle')
    pensize(2)
    setheading(180)
    
    for fib in sequence:
        if fib < 9:
            circle(25 * fib, 90)
        else:
            break

def main():
    """
    Main function to execute the Fibonacci spiral drawing.
    """
    while True:
        amount = input('Choose up to which number of the Fibonacci sequence you want to go up to (min 5): ')
        if amount.isdecimal() and int(amount) >= 5:
            amount = int(amount)
            break
        print('Please enter a number bigger than 4.')

    fibonacci_sequence = get_fibonacci_sequence(amount)
    
    for index, number in enumerate(fibonacci_sequence):
        print(f'F°{index}: {number}')
    
    draw_fibonacci_spiral(fibonacci_sequence)

if __name__ == "__main__":
    main()
