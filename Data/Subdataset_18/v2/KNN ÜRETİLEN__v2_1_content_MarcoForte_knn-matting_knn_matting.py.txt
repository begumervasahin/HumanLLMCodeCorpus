import numpy as np
import sklearn.neighbors
import scipy.sparse
import warnings
import matplotlib.pyplot as plt
from imageio import imread, imsave

def knn_matte(img, trimap, mylambda=100):
    """
    Perform K-Nearest Neighbors matting on the given image using the trimap.

    Parameters:
    img (ndarray): The input image.
    trimap (ndarray): The trimap indicating foreground, background, and unknown regions.
    mylambda (float): Regularization parameter.

    Returns:
    alpha (ndarray): The alpha matte of the image.
    """
    # Normalize the image and trimap
    img, trimap = img / 255.0, trimap / 255.0
    [m, n, c] = img.shape

    # Create masks for foreground and background
    foreground = (trimap > 0.99).astype(int)
    background = (trimap < 0.01).astype(int)
    all_constraints = foreground + background

    print('Finding nearest neighbors')
    # Create feature vector including image pixels and their spatial coordinates
    a, b = np.unravel_index(np.arange(m * n), (m, n))
    feature_vec = np.append(img.reshape(m * n, c).T, [a, b] / np.sqrt(m * m + n * n), axis=0).T

    # Fit the KNN model and find neighbors
    nbrs = sklearn.neighbors.NearestNeighbors(n_neighbors=10, n_jobs=4).fit(feature_vec)
    knns = nbrs.kneighbors(feature_vec)[1]

    print('Computing sparse matrix A')
    row_inds = np.repeat(np.arange(m * n), 10)
    col_inds = knns.reshape(m * n * 10)
    vals = 1 - np.linalg.norm(feature_vec[row_inds] - feature_vec[col_inds], axis=1) / (c + 2)
    A = scipy.sparse.coo_matrix((vals, (row_inds, col_inds)), shape=(m * n, m * n))

    # Construct the Laplacian matrix
    D_script = scipy.sparse.diags(np.ravel(A.sum(axis=1)))
    L = D_script - A

    # Construct the diagonal constraint matrix
    D = scipy.sparse.diags(np.ravel(all_constraints[:, :, 0]))
    v = np.ravel(foreground[:, :, 0])
    c = 2 * mylambda * v
    H = 2 * (L + mylambda * D)

    print('Solving linear system for alpha')
    warnings.filterwarnings('error')
    try:
        # Solve the linear system to get the alpha matte
        alpha = np.clip(scipy.sparse.linalg.spsolve(H, c), 0, 1).reshape(m, n)
    except Warning:
        # Use least-squares solution if direct solving fails
        x = scipy.sparse.linalg.lsqr(H, c)
        alpha = np.clip(x[0], 0, 1).reshape(m, n)
    
    return alpha

def main():
    # Read the input image and trimap
    img = imread('donkey.png')[:, :, :3]
    trimap = imread('donkeyTrimap.png')[:, :, :3]

    # Generate the alpha matte
    alpha = knn_matte(img, trimap)

    # Save and display the alpha matte
    imsave('donkeyAlpha.png', alpha)
    plt.title('Alpha Matte')
    plt.imshow(alpha, cmap='gray')
    plt.show()

if __name__ == '__main__':
    main()
