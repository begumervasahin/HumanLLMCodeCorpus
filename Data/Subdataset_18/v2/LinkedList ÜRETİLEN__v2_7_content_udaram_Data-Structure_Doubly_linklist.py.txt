class Node:
    """A node in a doubly linked list."""

    def __init__(self, data):
        self.data = data
        self.next = None
        self.prev = None

class DoublyLinkedList:
    """A doubly linked list implementation."""

    def __init__(self):
        self.head = None
        self.tail = None

    def insert(self, data):
        """Insert a new node at the end of the list."""
        new_node = Node(data)
        if self.head is None:
            self.head = self.tail = new_node
        else:
            self.tail.next = new_node
            new_node.prev = self.tail
            self.tail = new_node

    def insert_at_beginning(self, data):
        """Insert a new node at the beginning of the list."""
        new_node = Node(data)
        if self.head is None:
            self.head = self.tail = new_node
        else:
            new_node.next = self.head
            self.head.prev = new_node
            self.head = new_node

    def insert_at_position(self, pos, data):
        """Insert a new node at the specified position in the list."""
        if pos == 0:
            self.insert_at_beginning(data)
            return

        new_node = Node(data)
        current = self.head
        index = 0
        while current is not None and index < pos - 1:
            current = current.next
            index += 1

        if current is None:
            print("Invalid position")
            return

        new_node.next = current.next
        new_node.prev = current
        if current.next is not None:
            current.next.prev = new_node
        current.next = new_node
        if new_node.next is None:
            self.tail = new_node

    def display(self):
        """Display the list from head to tail."""
        current = self.head
        print("Doubly Linked List:", end=" ")
        if current is None:
            print("Empty!")
        else:
            while current is not None:
                print(f"<=> {current.data}", end=" ")
                current = current.next
            print()

    def display_reverse(self):
        """Display the list from tail to head."""
        current = self.tail
        print("List in reverse order is:", end=" ")
        if current is None:
            print("Empty!")
        else:
            while current is not None:
                print(f"<=> {current.data}", end=" ")
                current = current.prev
            print()

    def delete_from_beginning(self):
        """Delete the node from the beginning of the list."""
        if self.head is None:
            print("List is empty")
        else:
            if self.head == self.tail:
                self.head = self.tail = None
            else:
                self.head = self.head.next
                self.head.prev = None

    def delete_from_end(self):
        """Delete the node from the end of the list."""
        if self.head is None:
            print("List is empty")
        else:
            if self.head == self.tail:
                self.head = self.tail = None
            else:
                self.tail = self.tail.prev
                self.tail.next = None

    def delete_from_position(self, pos):
        """Delete the node from the specified position in the list."""
        if self.head is None:
            print("List is empty")
        elif pos == 0:
            self.delete_from_beginning()
        else:
            current = self.head
            index = 0
            while current is not None and index < pos - 1:
                current = current.next
                index += 1

            if current is None or current.next is None:
                print("Invalid position")
            else:
                to_delete = current.next
                current.next = to_delete.next
                if to_delete.next is not None:
                    to_delete.next.prev = current
                if to_delete == self.tail:
                    self.tail = current

def main():
    """Main function to interact with the doubly linked list."""
    dll = DoublyLinkedList()
    while True:
        print('''\n1. Insert at end
2. Insert at beginning
3. Insert at position
4. Display list
5. Display in reverse order
6. Delete from beginning
7. Delete from end
8. Delete from position
9. EXIT''')
        option = int(input("Enter option: "))
        if option == 1:
            data = int(input("Enter data: "))
            dll.insert(data)
        elif option == 2:
            data = int(input("Enter data: "))
            dll.insert_at_beginning(data)
        elif option == 3:
            pos = int(input("Enter position (0 ......n): "))
            data = int(input("Enter data: "))
            dll.insert_at_position(pos, data)
        elif option == 4:
            dll.display()
        elif option == 5:
            dll.display_reverse()
        elif option == 6:
            dll.delete_from_beginning()
        elif option == 7:
            dll.delete_from_end()
        elif option == 8:
            pos = int(input("Enter position (0 ......n): "))
            dll.delete_from_position(pos)
        elif option == 9:
            break
        else:
            print("Invalid option! Please try again.")

if __name__ == "__main__":
    main()
