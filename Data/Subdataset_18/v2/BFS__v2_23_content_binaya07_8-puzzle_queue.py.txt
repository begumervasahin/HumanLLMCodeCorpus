class Queue:
    """
    A simple Queue implementation using a list.
    """
    def __init__(self):
        self.list = []
        self.state = set()
        
    def enqueue(self, item):
        """
        Add an item to the queue and track its state.

        Args:
            item: The item to be added to the queue.
        """
        self.list.append(item)  # Add item to the queue
        self.state.add(tuple(item.list))  # Track the state of the item
        
    def dequeue(self):
        """
        Remove the first item from the queue (FIFO) and return it.

        Returns:
            The first item in the queue.
        """
        if not self.is_empty():
            item = self.list.pop(0)  # Remove and return the first item
            self.state.remove(tuple(item.list))  # Remove the state of the dequeued item
            return item
        
    def is_empty(self):
        """
        Check if the queue is empty.

        Returns:
            bool: True if the queue is empty, False otherwise.
        """
        return len(self.list) == 0

    while not q.is_empty():
