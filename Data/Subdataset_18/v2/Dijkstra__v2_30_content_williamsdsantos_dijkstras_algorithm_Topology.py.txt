class Topology:
    def __init__(self, file_path):
        """
        Initializes the Topology object by reading the topology file and extracting nodes and links.

        Parameters:
        file_path (str): The path to the topology file.
        """
        self.nodes = []
        self.links = []

        self._read_topology_file(file_path)

    def _read_topology_file(self, file_path):
        """
        Reads the topology file and populates the nodes and links lists.

        Parameters:
        file_path (str): The path to the topology file.
        """
        with open(file_path, 'r') as topology_data:
            # Skip the first three header lines
            for _ in range(3):
                next(topology_data)
            
            # Read nodes
            self._read_nodes(topology_data)

            # Skip the next three header lines
            for _ in range(3):
                next(topology_data)

            # Read links
            self._read_links(topology_data)

    def _read_nodes(self, topology_data):
        """
        Reads the nodes from the topology file.

        Parameters:
        topology_data (file object): The open topology file.
        """
        for line in topology_data:
            data = line.split()
            if len(data) < 3:
                break
            node_id = data[2]
            self.nodes.append(node_id)

    def _read_links(self, topology_data):
        """
        Reads the links from the topology file.

        Parameters:
        topology_data (file object): The open topology file.
        """
        for line in topology_data:
            data = line.split()
            if len(data) < 5:
                break
            origin = data[2]
            destination = data[3]
            length = data[4]
            elem_link = [origin, destination, length]
            self.links.append(elem_link)

def main():
    file_path = 'data/topologies/PacificBell'
    topology = Topology(file_path)
    print("Nodes:", topology.nodes)
    print("Links:", topology.links)

if __name__ == "__main__":
    main()
