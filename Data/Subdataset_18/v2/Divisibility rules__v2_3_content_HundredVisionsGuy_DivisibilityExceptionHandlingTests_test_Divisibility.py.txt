def divisibility(a, b):
    """
    Check if `a` is divisible by `b` and handle potential errors.

    Args:
        a (int): The dividend.
        b (int): The divisor.

    Returns:
        str: 'divides evenly' if `a` is divisible by `b`,
             "doesn't divide evenly" if `a` is not divisible by `b`,
             'error' if there is a ZeroDivisionError or ValueError.
    """
    try:
        if b == 0:
            raise ZeroDivisionError("Cannot divide by zero")
        if not isinstance(a, int) or not isinstance(b, int):
            raise ValueError("Both a and b must be integers")
        
        if a % b == 0:
            return 'divides evenly'
        else:
            return "doesn't divide evenly"
    except ZeroDivisionError:
        return 'error'
    except ValueError:
        return 'error'
