def partition(array, left, right):
    """
    Partition the array by selecting the last element as the pivot.
    Elements smaller than or equal to the pivot are moved to the left,
    while larger elements are moved to the right.
    """
    pivot = array[right]
    print(f"Pivot: {pivot}")

    split_index = left
    for i in range(left, right + 1):
        print(f"Inspecting element at index {i}: {array[i]}")
        if array[i] <= pivot:
            if i != split_index:
                # Swap elements at index i and split_index
                array[i], array[split_index] = array[split_index], array[i]
            split_index += 1
    
    return split_index - 1

def quicksort(array, left, right):
    """
    Recursively apply quicksort to the array by partitioning it
    around a pivot and sorting the subarrays.
    """
    if left < right:
        split_index = partition(array, left, right)
        print(f"Split at index {split_index}: {array}")

        # Recursively sort elements before and after the partition
        quicksort(array, left, split_index - 1)
        quicksort(array, split_index + 1, right)

if __name__ == "__main__":
    # Example usage
    array = [3, 6, 8, 10, 1, 2, 1]
    quicksort(array, 0, len(array) - 1)
    print("Sorted array:", array)
