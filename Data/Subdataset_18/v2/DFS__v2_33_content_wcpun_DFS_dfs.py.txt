from fs.memoryfs import MemoryFS

def create_directories(fs):
    """
    Create directories in the provided filesystem.

    Args:
        fs (MemoryFS): The memory filesystem instance.
    """
    fs.makedir('test')
    fs.makedir('foo')

def list_directory_contents(fs):
    """
    List the contents of the directories in the provided filesystem.

    Args:
        fs (MemoryFS): The memory filesystem instance.
    """
    for path, info in fs.walk.info(max_depth=1):
        if info.is_dir:
            print(f"[dir] {path}")
        else:
            print(f"[file] {path}")

def main():
    """
    Main function to create directories and list their contents.
    """
    fs = MemoryFS()
    create_directories(fs)
    list_directory_contents(fs)

if __name__ == "__main__":
    main()
