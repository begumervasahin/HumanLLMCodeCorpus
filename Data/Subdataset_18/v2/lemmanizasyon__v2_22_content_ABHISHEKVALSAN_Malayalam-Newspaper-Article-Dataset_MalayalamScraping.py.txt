import requests
from bs4 import BeautifulSoup
import os

# Constants
URL_PREFIX = "http://example.com/page="  # Replace with the actual URL prefix

def month_to_digit(month):
    """Convert month name to its corresponding two-digit number."""
    month_dict = {
        'January': '01', 'February': '02', 'March': '03',
        'April': '04', 'May': '05', 'June': '06', 'July': '07',
        'August': '08', 'September': '09', 'October': '10',
        'November': '11', 'December': '12'
    }
    return month_dict.get(month, '00')

def conv_to_fname(date_str):
    """Convert date string to a filename-friendly format."""
    flist = date_str.split()
    month = month_to_digit(flist[2])
    day = flist[1].zfill(2)  # Ensure day is two digits
    filename = "1" + flist[3][2:] + month + day
    return filename, flist[3]

def get_data(page_id):
    """Fetch and process data from the given page ID."""
    url = URL_PREFIX + str(page_id)
    response = requests.get(url)
    
    if response.status_code == 200:
        soup = BeautifulSoup(response.text, 'html.parser')
        links = soup.findAll("p", {"class": "text-center"})
        
        if len(links) != 1:
            title = soup.findAll("h1")[0].text
            date = soup.findAll("div", {"class": "myd"})[1].text
            content = soup.findAll("p", {"class": ""})[0].text
            
            filename, year = conv_to_fname(date)
            path = "DataSet/"
            os.makedirs(path, exist_ok=True)  # Ensure the directory exists
            filename_with_path = os.path.join(path, f"{filename}{page_id}.utf8")
            
            with open(filename_with_path, 'w', encoding='utf-8') as f:
                f.write(f"<DOC>\n<DOCNO>{filename}{page_id}.utf8</DOCNO>\n<TEXT>\n{title}\n{date}\n{content}\n</TEXT>\n</DOC>")
            
            print(f"Writing file {filename}{page_id}.utf8 to {path}")
        else:
            print(f"File missing at {url}")
    else:
        print(f"Error: Unable to fetch data from {url} (Status code: {response.status_code})")
