import string
from typing import List, Dict

# Constants for the rows, columns, and boxes in the Sudoku grid
rows = 'ABCDEFGHI'
cols = '123456789'
boxes = [r + c for r in rows for c in cols]

def grid_values(grid: str) -> Dict[str, str]:
    """
    Convert grid into a dictionary of {square: char} with '123456789' for empty squares.
    
    Args:
        grid (str): A string representation of the Sudoku grid.
    
    Returns:
        Dict[str, str]: A dictionary representation of the Sudoku grid.
    """
    digits = '123456789'
    chars = [c if c in digits else digits for c in grid]
    assert len(chars) == 81, "Input grid must be a string of length 81"
    return dict(zip(boxes, chars))

def display(values: Dict[str, str]):
    """
    Display the values as a 2-D grid.
    
    Args:
        values (Dict[str, str]): The Sudoku grid in dictionary form.
    """
    width = 1 + max(len(values[s]) for s in boxes)
    line = '+'.join(['-' * (width * 3)] * 3)
    for r in rows:
        print(''.join(values[r + c].center(width) + ('|' if c in '36' else '') for c in cols))
        if r in 'CF': 
            print(line)
    print()

def search(values: Dict[str, str]) -> Dict[str, str]:
    """
    Using depth-first search and propagation, try all possible values.
    
    Args:
        values (Dict[str, str]): The current state of the Sudoku grid.
    
    Returns:
        Dict[str, str]: The solved Sudoku grid or False if no solution exists.
    """
    values = reduce_puzzle(values)
    if values is False:
        return False  # Failed earlier
    if all(len(values[s]) == 1 for s in boxes):
        return values  # Solved!
    
    # Choose one of the unfilled squares with the fewest possibilities
    n, s = min((len(values[s]), s) for s in boxes if len(values[s]) > 1)
    
    # Use recurrence to solve each one of the resulting sudokus
    for value in values[s]:
        new_sudoku = values.copy()
        new_sudoku[s] = value
        attempt = search(new_sudoku)
        if attempt:
            return attempt

def reduce_puzzle(values: Dict[str, str]) -> Dict[str, str]:
    """
    Apply eliminate, only choice, and naked twins strategies to reduce the puzzle.
    
    Args:
        values (Dict[str, str]): The current state of the Sudoku grid.
    
    Returns:
        Dict[str, str]: The reduced Sudoku grid or False if no solution exists.
    """
    stalled = False
    while not stalled:
        solved_values_before = len([box for box in values if len(values[box]) == 1])
        values = eliminate(values)
        values = only_choice(values)
        values = naked_twins(values)
        solved_values_after = len([box for box in values if len(values[box]) == 1])
        stalled = solved_values_before == solved_values_after
        
        # Check for boxes with zero available values
        if any(len(values[box]) == 0 for box in boxes):
            return False
    return values

def eliminate(values: Dict[str, str]) -> Dict[str, str]:
    """
    Eliminate values from peers of each box with a single value.
    
    Args:
        values (Dict[str, str]): The current state of the Sudoku grid.
    
    Returns:
        Dict[str, str]: The updated Sudoku grid after applying the eliminate strategy.
    """
    for box in boxes:
        if len(values[box]) == 1:
            digit = values[box]
            for peer in peers[box]:
                values[peer] = values[peer].replace(digit, '')
    return values

def only_choice(values: Dict[str, str]) -> Dict[str, str]:
    """
    Finalize all values that are the only choice for a unit.
    
    Args:
        values (Dict[str, str]): The current state of the Sudoku grid.
    
    Returns:
        Dict[str, str]: The updated Sudoku grid after applying the only choice strategy.
    """
    for unit in unitlist:
        for digit in '123456789':
            dplaces = [box for box in unit if digit in values[box]]
            if len(dplaces) == 1:
                values[dplaces[1]] = digit
    return values

def naked_twins(values: Dict[str, str]) -> Dict[str, str]:
    """
    Eliminate values using the naked twins strategy.
    
    Args:
        values (Dict[str, str]): The current state of the Sudoku grid.
    
    Returns:
        Dict[str, str]: The updated Sudoku grid after applying the naked twins strategy.
    """
    # Find all instances of naked twins
    for unit in unitlist:
        # Find all boxes with a pair of identical digits
        pairs = [box for box in unit if len(values[box]) == 2]
        naked_twins = [(box1, box2) for i, box1 in enumerate(pairs) for box2 in pairs[i + 1:] if values[box1] == values[box2]]
        for box1, box2 in naked_twins:
            digit1, digit2 = values[box1]
            for box in unit:
                if box != box1 and box != box2:
                    values[box] = values[box].replace(digit1, '').replace(digit2, '')
    return values

# Define peers, units, and unitlist to be used in the strategies
def cross(A, B):
    return [a + b for a in A for b in B]

# Define units
row_units = [cross(r, cols) for r in rows]
column_units = [cross(rows, c) for c in cols]
square_units = [cross(rs, cs) for rs in ('ABC', 'DEF', 'GHI') for cs in ('123', '456', '789')]
unitlist = row_units + column_units + square_units

# Units and peers for each box
units = dict((s, [u for u in unitlist if s in u]) for s in boxes)
peers = dict((s, set(sum(units[s], [])) - set([s])) for s in boxes)
