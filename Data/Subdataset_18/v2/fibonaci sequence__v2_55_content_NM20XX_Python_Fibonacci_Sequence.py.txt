def fib(n):
    """
    Generator function to yield Fibonacci sequence up to n terms.
    :param n: Number of terms in the Fibonacci sequence.
    :yield: Next number in the Fibonacci sequence.
    """
    x, y = 0, 1
    for _ in range(n):
        yield x
        x, y = y, x + y

def get_input():
    """
    Prompt the user to enter a term for the Fibonacci sequence.
    :return: A positive integer entered by the user.
    """
    while True:
        try:
            term = int(input("Enter a term in the Fibonacci sequence: "))
            if term <= 0:
                print("Please enter a positive integer.")
            else:
                return term
        except ValueError:
            print("Error: Please enter an integer value.")

def main():
    """
    Main function to get user input and print the Fibonacci sequence.
    """
    input_term = get_input()
    print(f"Fibonacci sequence up to {input_term} terms:")
    for number in fib(input_term):
        print(number, end=", ")
    print()  # Print a newline after the sequence

if __name__ == "__main__":
    main()
