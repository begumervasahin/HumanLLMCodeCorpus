import sqlite3
import datetime
import pytz

# Initialize the SQLite database connection
db = sqlite3.connect("accounts.sqlite", detect_types=sqlite3.PARSE_DECLTYPES)

# Create necessary tables and views if they do not exist
db.execute("""
    CREATE TABLE IF NOT EXISTS accounts (
        name TEXT PRIMARY KEY NOT NULL, 
        balance INTEGER NOT NULL
    )
""")
db.execute("""
    CREATE TABLE IF NOT EXISTS history (
        time TIMESTAMP NOT NULL, 
        account TEXT NOT NULL, 
        amount INTEGER NOT NULL, 
        PRIMARY KEY(time, account)
    )
""")
db.execute("""
    CREATE VIEW IF NOT EXISTS localhistory AS 
    SELECT 
        strftime('%Y-%m-%d %H:%M:%f', history.time, 'localtime') AS localtime,
        history.account, 
        history.amount 
    FROM history 
    ORDER BY history.time
""")

class Account:
    """
    A class to represent a bank account.
    
    Attributes
    ----------
    name : str
        The name of the account holder.
    _balance : int
        The current balance of the account in cents.
    """

    def __init__(self, name: str, opening_balance: int = 0):
        cursor = db.execute("SELECT name, balance FROM accounts WHERE name = ?", (name,))
        row = cursor.fetchone()
        if row:
            self.name, self._balance = row
            print(f"Retrieved balance for {self.name}")
        else:
            self.name = name
            self._balance = opening_balance
            cursor.execute("INSERT INTO accounts (name, balance) VALUES (?, ?)", (name, opening_balance))
            cursor.connection.commit()
            print(f"Account created for {self.name}")
        self.show_balance()

    def show_balance(self):
        """Displays the current balance of the account."""
        print(f"Balance on account {self.name} is {self._balance / 100:.2f}")

    @staticmethod
    def current_time():
        """Returns the current time in UTC, localized to UTC."""
        return pytz.utc.localize(datetime.datetime.utcnow())

    def deposit(self, amount: int) -> float:
        """Deposits a specified amount into the account."""
        if amount > 0:
            self._update_balance(amount)
            print(f"{amount / 100:.2f} deposited")
        return self._balance / 100

    def withdraw(self, amount: int) -> float:
        """Withdraws a specified amount from the account if sufficient balance exists."""
        if 0 < amount <= self._balance:
            self._update_balance(-amount)
            print(f"{amount / 100:.2f} withdrawn")
        else:
            print("Amount must be more than zero and less than or equal to your account balance.")
        return self._balance / 100

    def _update_balance(self, amount: int):
        """Updates the balance and records the transaction in the history."""
        new_balance = self._balance + amount
        transaction_time = self.current_time()
        db.execute("UPDATE accounts SET balance = ? WHERE name = ?", (new_balance, self.name))
        db.execute("INSERT INTO history (time, account, amount) VALUES (?, ?, ?)", (transaction_time, self.name, amount))
        db.commit()
        self._balance = new_balance
