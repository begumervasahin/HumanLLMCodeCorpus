import time

def binary_search(nums, item):
    """Perform a binary search to find the item in the sorted list nums."""
    first, last = 0, len(nums) - 1
    found = False
    
    while first <= last and not found:
        middle = (first + last) // 2
        if nums[middle] == item:
            found = True
        elif item < nums[middle]:
            last = middle - 1
        else:
            first = middle + 1
                
    return found

def calculate_runtime(search_func, nums, item):
    """Calculate the runtime of the search function in microseconds."""
    start = time.perf_counter()
    result = search_func(nums, item)
    end = time.perf_counter()
    elapsed_time = (end - start) * 1_000_000  # Convert to microseconds
    return elapsed_time

if __name__ == "__main__":
    # Sample sorted list for binary search
    nums_list = [-125, -50, -1, 0, 1, 5, 7, 10, 12, 20, 26, 30, 50, 127, 500, 833, 900, 1000, 12345, 123456]
    
    # Get the target number from the user
    num_to_find = int(input("Enter the number you want to find: "))
    
    # Perform the binary search
    is_found = binary_search(nums_list, num_to_find)
    
    # Calculate the time taken to perform the search
    time_to_find = calculate_runtime(binary_search, nums_list, num_to_find)
    
    # Display the search results to the user
    if is_found:
        print("The number is in the list.")
    else:
        print("The number is not in the list.")
    
    print(f"Time spent searching: {time_to_find:.2f} microseconds")
