import random

def quicksort(arr):
    if len(arr) <= 1:
        return arr
    
    # Select a random pivot
    pivot_index = random.randint(0, len(arr) - 1)
    pivot = arr[pivot_index]
    print(f"Pivoting {pivot} in {arr}")

    # Partition the array into left (less than pivot), right (greater than pivot), and pivot duplicates
    left = []
    right = []
    pivot_clones = []

    for i, value in enumerate(arr):
        if i == pivot_index:
            pivot_clones.append(pivot)
        elif value < pivot:
            left.append(value)
        elif value > pivot:
            right.append(value)
        else:
            pivot_clones.append(value)

    print(f"Left becomes {left}; right becomes {right}")

    # Recursively sort left and right partitions and combine them with pivot duplicates
    return quicksort(left) + pivot_clones + quicksort(right)

# Generate a random list of integers to sort
unsorted_list = [random.randint(1, 100) for _ in range(10)]

# Output the unsorted list
print("Unsorted list:", unsorted_list)

# Perform quicksort and output the sorted list
sorted_list = quicksort(unsorted_list)
print("Sorted list:", sorted_list)
