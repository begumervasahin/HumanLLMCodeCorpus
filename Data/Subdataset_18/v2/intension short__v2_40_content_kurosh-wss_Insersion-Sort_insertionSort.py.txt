import random
import time
import numpy as np
import matplotlib.pyplot as plt

# Class for performing insertion sort on a Python list
class InsertionSorter:
    @staticmethod
    def sort(lst):
        """Sorts a list using the insertion sort algorithm."""
        for i in range(1, len(lst)):
            key = lst[i]
            j = i - 1
            while j >= 0 and lst[j] > key:
                lst[j + 1] = lst[j]
                j -= 1
            lst[j + 1] = key

# Class for generating lists of random numbers
class NumberGenerator:
    @staticmethod
    def generate(number_of_lists, min_range, max_range, lower_bound, upper_bound):
        """Generates a specified number of lists with random integers within a range."""
        return [
            [random.randint(lower_bound, upper_bound) for _ in range(random.randint(min_range, max_range))]
            for _ in range(number_of_lists)
        ]

# Node class for the doubly linked list
class Node:
    def __init__(self, data):
        self.data = data
        self.next = None
        self.prev = None

# Doubly Linked List class with sorting functionality
class DoublyLinkedList:
    def __init__(self):
        self.head = None
        self.size = 0

    def append(self, data):
        """Appends a new node with the given data to the end of the list."""
        new_node = Node(data)
        if self.head is None:
            self.head = new_node
        else:
            current = self.head
            while current.next:
                current = current.next
            current.next = new_node
            new_node.prev = current
        self.size += 1

    def insert_sorted(self, data):
        """Inserts data into the list while maintaining sorted order."""
        new_node = Node(data)
        if self.head is None:
            self.head = new_node
            self.size += 1
            return
        
        current = self.head
        while current and current.data < data:
            current = current.next

        if current is None:  # Insert at the end
            self.append(data)
        elif current.prev is None:  # Insert at the beginning
            new_node.next = self.head
            self.head.prev = new_node
            self.head = new_node
        else:  # Insert in the middle
            new_node.next = current
            new_node.prev = current.prev
            current.prev.next = new_node
            current.prev = new_node
        self.size += 1

    def insertion_sort(self):
        """Sorts the linked list using insertion sort."""
        sorted_list = DoublyLinkedList()
        current = self.head
        while current:
            sorted_list.insert_sorted(current.data)
            current = current.next
        self.head = sorted_list.head

    def to_list(self):
        """Converts the linked list to a Python list."""
        result = []
        current = self.head
        while current:
            result.append(current.data)
            current = current.next
        return result

# Function to measure and compare sorting times
def measure_time_sorting(num_lists, min_range, max_range, generator_func, sort_func):
    """Measures sorting time for a list and a linked list."""
    numbers = generator_func(num_lists, min_range, max_range, 10, 1000)
    numbers.sort(key=len)
    simple_times = []
    linked_list_times = []
    lengths = [len(lst) for lst in numbers]

    # Measure time for simple list insertion sort
    for lst in numbers:
        start = time.process_time()
        sort_func(lst)
        simple_times.append(time.process_time() - start)

    # Measure time for doubly linked list insertion sort
    for lst in numbers:
        linked_list = DoublyLinkedList()
        for item in lst:
            linked_list.append(item)
        start = time.process_time()
        linked_list.insertion_sort()
        linked_list_times.append(time.process_time() - start)

    return simple_times, linked_list_times, lengths

# Function to plot the results of the sorting time comparison
def plot_results(results, title):
    """Plots the comparison of sorting times."""
    simple_times, linked_list_times, lengths = results
    ind = np.arange(len(lengths))
    width = 0.35

    fig, ax = plt.subplots()
    rects1 = ax.bar(ind - width / 2, simple_times, width, label='Simple Insertion Sort')
    rects2 = ax.bar(ind + width / 2, linked_list_times, width, label='Linked List Insertion Sort')

    ax.set_ylabel('Time (seconds)')
    ax.set_title(title)
    ax.set_xticks(ind)
    ax.set_xticklabels(lengths)
    ax.legend()

    def autolabel(rects):
        """Attach a text label above each bar displaying its height."""
        for rect in rects:
            height = rect.get_height()
            ax.annotate(f'{height:.4f}',
                        xy=(rect.get_x() + rect.get_width() / 2, height),
                        xytext=(0, 3),  # 3 points vertical offset
                        textcoords="offset points",
                        ha='center', va='bottom')

    autolabel(rects1)
    autolabel(rects2)

    plt.show()

# Main function to execute the sorting and plotting
def main():
    # Small range comparison
    plot_results(
        measure_time_sorting(100, 1000, 2000, NumberGenerator.generate, InsertionSorter.sort),
        'Time Table for Linked List and Simple Insertion Sort (Small Ranges)'
    )
    # Large range comparison
    plot_results(
        measure_time_sorting(100, 1000, 20000, NumberGenerator.generate, lambda x: sorted(x)),
        'Time Table for Linked List and Simple Insertion Sort (Large Ranges)'
    )

if __name__ == "__main__":
    main()
