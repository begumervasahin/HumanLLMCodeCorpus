import utils
import filenames

class Log:
    def __init__(self, k, des_name, svm_kernel):
        self.text = ""
        self.k = k
        self.des_name = des_name
        self.svm_kernel = svm_kernel

    def save(self):
        # Generate the log filename using the provided parameters and save the log text to this file
        log_filename = filenames.log(self.k, self.des_name, self.svm_kernel)
        with open(log_filename, "w") as file:
            file.write(self.text)

    def log_time(self, activity, time, set_name=None):
        # Format the time using a human-readable format
        time_str = utils.humanize_time(time)
        if set_name:
            message = f"Time for {activity} {set_name} was {time_str}.\n"
        else:
            message = f"Time for {activity} was {time_str}.\n"
        self.text += message

    def train_des_time(self, time):
        # Log the time taken for extracting local descriptors from the training images
        self.log_time("getting all the local descriptors of the training images", time)

    def codebook_time(self, time):
        # Log the time taken for generating the codebook using k-means
        self.log_time("generating the codebook with k-means", time)

    def train_vlad_time(self, time):
        # Log the time taken for extracting VLAD global descriptors from the training images
        self.log_time("getting VLAD global descriptors of the", time, "training")

    def svm_time(self, time):
        # Log the time taken for training the SVM
        self.log_time("calculating the SVM", time)

    def test_vlad_time(self, time):
        # Log the time taken for extracting VLAD global descriptors from the testing images
        self.log_time("getting VLAD global descriptors of the", time, "testing")

    def predict_time(self, time):
        # Log the time taken for predicting the testing set
        self.log_time("predicting the testing set", time)

    def accuracy(self, accuracy):
        # Log the accuracy of the model
        self.text += f"Accuracy = {accuracy}.\n"

    def classes(self, classes):
        # Log the class labels
        self.text += f"Classes = {classes}\n"

    def classes_counts(self, classes_counts):
        # Log the counts of local descriptors for each class
        self.text += f"Classes Local Descriptors Counts = {classes_counts}\n"

    def confusion_matrix(self, conf_mat):
        # Log the confusion matrix
        self.text += f"Confusion Matrix =\n{conf_mat}\n"
