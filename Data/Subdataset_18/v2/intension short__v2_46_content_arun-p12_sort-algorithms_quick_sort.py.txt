def quick_sort(arr, verbose=0, descending=False):
    """
    Sorts the list arr using the quicksort algorithm.

    Parameters:
    arr (list): The list to be sorted.
    verbose (int): Verbosity level (0: no output, 1: iteration output, 2: detailed output).
    descending (bool): If True, sorts the list in descending order.

    Returns:
    list: The sorted list.
    """
    
    def partition(low, high):
        pivot = arr[low]
        left = low + 1
        right = high
        
        while True:
            # Move the left pointer to the right while elements are less than the pivot
            while left <= right and arr[left] <= pivot:
                left += 1
            # Move the right pointer to the left while elements are greater than the pivot
            while left <= right and arr[right] > pivot:
                right -= 1
            if left <= right:
                arr[left], arr[right] = arr[right], arr[left]
                if verbose == 2:
                    print(f"Sub-array after swap: pivot={pivot}, left={left}, right={right} :: {arr}")
            else:
                break
        
        # Place the pivot element in the correct position
        arr[low], arr[right] = arr[right], arr[low]
        return right
    
    def quicksort_recursive(low, high):
        if low < high:
            partition_idx = partition(low, high)
            if verbose:
                print(f"Partition index={partition_idx}, low={low}, high={high} :: {arr}")
            quicksort_recursive(low, partition_idx - 1)
            quicksort_recursive(partition_idx + 1, high)
    
    quicksort_recursive(0, len(arr) - 1)
    
    if descending:
        arr.reverse()
    
    return arr

# Example usage
if __name__ == "__main__":
    data = [24, 3, 45, 12, 56, 18, 39, 17, 8]
    sorted_data = quick_sort(data, verbose=1, descending=False)
    print("Sorted array:", sorted_data)
