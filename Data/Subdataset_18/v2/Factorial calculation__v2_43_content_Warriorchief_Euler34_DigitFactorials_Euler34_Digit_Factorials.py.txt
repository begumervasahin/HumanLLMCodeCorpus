Digit Factorials
Problem 34

145 is a curious number, as 1! + 4! + 5! = 1 + 24 + 120 = 145.
Find the sum of all numbers which are equal to the sum of the factorial of their digits.
Note: as 1! = 1 and 2! = 2 are not sums they are not included.
"""

def factorial(x):
    """
    Computes the factorial of a given number x.
    
    :param x: Integer
    :return: Factorial of x
    """
    result = 1
    for i in range(2, x + 1):
        result *= i
    return result

def digits_of_number(num):
    """
    Separates a number into its individual digits.
    
    :param num: Integer
    :return: List of digits
    """
    return [int(digit) for digit in str(num)]

def is_curious_number(num):
    """
    Checks if a number is equal to the sum of the factorial of its digits.
    
    :param num: Integer
    :return: True if the number is curious, False otherwise
    """
    return num == sum(factorial(digit) for digit in digits_of_number(num))

def find_curious_numbers(limit):
    """
    Finds all curious numbers less than a given limit.
    
    :param limit: Maximum number to check
    :return: List of curious numbers
    """
    curious_numbers = []
    for i in range(3, limit):
        if i % 100000 == 0:
            print(f"Checked up to {i} numbers...")
        if is_curious_number(i):
            curious_numbers.append(i)
    
    print(f"Curious numbers are: {curious_numbers}")
    print(f"There are {len(curious_numbers)} curious numbers less than {limit}.")
    return curious_numbers

if __name__ == "__main__":
    find_curious_numbers(100000000)
