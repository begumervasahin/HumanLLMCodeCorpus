import sys
from operator import itemgetter

def main():
    """
    Main function to read distances from standard input, sort them,
    and calculate the average of the k smallest distances.
    """
    distances = []
    k = 5  # Number of smallest distances to consider

    # Read input from standard input (e.g., piped or redirected from a file)
    for line in sys.stdin:
        line = line.strip()
        elements = line.split()
        distances.append(elements)

    # Sort distances based on the first element
    distances = sorted(distances, key=itemgetter(0))

    # Sum the k smallest values
    estimate_val = 0.0
    valid_entries = 0
    for i in range(k):
        try:
            estimate_val += float(distances[i][1])
            valid_entries += 1
        except (ValueError, IndexError):
            # Skip entries that cannot be converted to float or are out of range
            continue

    # Calculate and print the average
    if valid_entries > 0:
        average = estimate_val / valid_entries
    else:
        average = 0.0
    print(average)

if __name__ == "__main__":
    main()
