import pyDes
import base64

# Data to be encrypted
data = "abc123"

# Initialize the DES key and cipher parameters
key = 'g9G16nTs'
iv = 'g9G16nTs'  # Initialization vector must be the same length as the key

# Create a DES cipher object with CBC mode and PKCS5 padding
cipher = pyDes.des(key, pyDes.CBC, iv, pad=None, padmode=pyDes.PAD_PKCS5)

# Encrypt the data
encrypted_data = cipher.encrypt(data)

# Encode the encrypted data with base64
encoded_encrypted_data = base64.b64encode(encrypted_data)

# Decode the base64 encoded encrypted data
decoded_encrypted_data = base64.b64decode(encoded_encrypted_data)

# Decrypt the decoded encrypted data
decrypted_data = cipher.decrypt(decoded_encrypted_data)

# Print the results
print("Original Data:", data)
print("Encrypted (base64 encoded):", encoded_encrypted_data.decode())
print("Decrypted Data:", decrypted_data.decode())
