class BinaryTreeNode:
    """
    A class representing a node in a Binary Search Tree (BST).
    
    Binary Search Trees (BST) allow quick lookups with O(logN) time complexity.
    """

    def __init__(self, value):
        """
        Initialize a new BinaryTreeNode.
        
        Args:
            value (int): The value of the node.
        """
        self.value = value
        self.left = None
        self.right = None

    def insert_left(self, value):
        """
        Insert a new node with the given value to the left of the current node.
        
        Args:
            value (int): The value to insert.
        
        Returns:
            BinaryTreeNode: The new left child node.
        """
        self.left = BinaryTreeNode(value)
        return self.left

    def insert_right(self, value):
        """
        Insert a new node with the given value to the right of the current node.
        
        Args:
            value (int): The value to insert.
        
        Returns:
            BinaryTreeNode: The new right child node.
        """
        self.right = BinaryTreeNode(value)
        return self.right

    def check_binary_search_tree_iterative(self):
        """
        Check if the binary tree is a valid binary search tree (BST) using an iterative approach.
        
        Returns:
            bool: True if the tree is a valid BST, False otherwise.
        """
        stack = [(self, -float('inf'), float('inf'))]
        
        while stack:
            node, lbound, ubound = stack.pop()
            if node.value <= lbound or node.value >= ubound:
                return False
            if node.left:
                stack.append((node.left, lbound, node.value))
            if node.right:
                stack.append((node.right, node.value, ubound))
                
        return True

    def check_bst_recursive(self, node, lower, upper):
        """
        Check if the binary tree is a valid binary search tree (BST) using a recursive approach.
        
        Args:
            node (BinaryTreeNode): The current node being checked.
            lower (float): The lower bound for the node value.
            upper (float): The upper bound for the node value.
        
        Returns:
            bool: True if the tree is a valid BST, False otherwise.
        """
        if not node:
            return True
        if node.value <= lower or node.value >= upper:
            return False
        return self.check_bst_recursive(node.left, lower, node.value) and \
               self.check_bst_recursive(node.right, node.value, upper)

# Example usage
if __name__ == "__main__":
    # Create a binary tree and insert nodes
    bt = BinaryTreeNode(5)
    bt.insert_left(3)
    bt.insert_right(7)
    bt.left.insert_left(1)
    bt.left.insert_right(4)  # Changed to 4 from 9 to make the tree a valid BST

    # Check if the binary tree is a valid BST using both iterative and recursive methods
    print("Iterative check:", bt.check_binary_search_tree_iterative())
    print("Recursive check:", bt.check_bst_recursive(bt, -float('inf'), float('inf')))
