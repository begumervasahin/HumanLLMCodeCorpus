import csv
import os
import re
import numpy as np
import pandas as pd
from fastdtw import fastdtw
from auxiliaryfunctions import haversine_np, print_results

# Load the DataFrame from the pickle file
df = pd.read_pickle('./final_cleaned.df')

# Read the test set trajectories from the CSV file
trajectories_read = []
with open('./test_set_a1.csv', 'r') as f:
    reader = csv.reader(f)
    for i, row in enumerate(reader):
        if i == 0:  # Skip the header row
            continue
        trajectory = []
        for j in range(0, len(row), 3):
            lat = float(re.sub(r'[\[]', '', row[j]))
            lon = float(row[j + 1])
            timestamp = float(re.sub(r'[\]]', '', row[j + 2]))
            trajectory.append([lat, lon, timestamp])
        trajectories_read.append(trajectory)

print("Dataset read successfully.")

# Create a directory to store the DTW results
os.makedirs("DTWresults", exist_ok=True)

# Perform DTW and store the results
for j, traj in enumerate(trajectories_read):
    distances = []
    for elem in df['timestamp_longitude_latitude']:
        distance, path = fastdtw(elem, traj, dist=haversine_np)
        distances.append(distance)
    
    distancesnp = np.array(distances)
    sorted_ind = np.argsort(distancesnp)[:5]
    
    for k, elem in enumerate(sorted_ind):
        result_path = f"./DTWresults/trajectory{j}neighbor{k}"
        print_results(df, distancesnp, elem, result_path)

print("DTW analysis completed.")
