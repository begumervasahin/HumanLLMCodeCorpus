# Define global variables
a = 1000
b = 2000

def f1():
    """
    Function f1 demonstrating the use of local and global variables.
    """
    # Define local variables
    p = 3000
    q = 4000
    
    # Print statements inside f1
    print("Inside f1()")
    print(f"a: {a}")  # Accessing global variable a
    print(f"b: {b}")  # Accessing global variable b
    print(f"p: {p}")  # Accessing local variable p
    print(f"q: {q}")  # Accessing local variable q

def f2():
    """
    Function f2 demonstrating the use of local and global variables,
    and handling NameError exceptions for undefined local variables.
    """
    # Define local variables
    x = 5000
    y = 6000
    
    # Print statements inside f2
    print("Inside f2()")
    print(f"a: {a}")  # Accessing global variable a
    print(f"b: {b}")  # Accessing global variable b
    print(f"x: {x}")  # Accessing local variable x
    print(f"y: {y}")  # Accessing local variable y
    
    try:
        # Attempt to access local variable p from f1 (will raise NameError)
        print(f"p: {p}")
    except NameError as e:
        print(e)  # Print the error message
    
    try:
        # Attempt to access local variable q from f1 (will raise NameError)
        print(f"q: {q}")
    except NameError as e:
        print(e)  # Print the error message

# Call the functions to demonstrate variable scope
f1()
f2()
