from utils import initialize_single_source, extract_min, relax

def dijkstra(graph, source):
    """
    Dijkstra's algorithm to find the shortest paths from the source vertex to all other vertices.
    :param graph: The graph on which to run the algorithm.
    :param source: The starting vertex.
    """
    # Step 1: Initialize the graph
    initialize_single_source(graph, source)
    
    # Step 2: Create the set of visited nodes and the priority queue
    visited_nodes = []
    priority_queue = list(graph.get_vertices())
    
    # Step 3: Iterate while the priority queue is not empty
    while priority_queue:
        # Extract the node with the minimum distance
        u = extract_min(priority_queue)
        
        # Mark the node as visited
        u.set_visited()
        visited_nodes.append(u)
        
        # Relax all adjacent vertices
        for v in u.get_adjacent_vertices():
            if not v.is_visited():
                relax(u, v)

    source_vertex = graph.get_vertex(0)
    
    # Run the Dijkstra's algorithm
    dijkstra(graph, source_vertex)
