import itertools
from collections import defaultdict

def get_itemset_transaction_list(data):
    """
    Generate the initial set of items and the transaction list.
    
    Args:
        data (list of list of str): The dataset where each transaction is a list of items.
    
    Returns:
        itemSet (set): The set of individual items.
        transactionList (list of frozenset): The list of transactions as frozensets.
    """
    transaction_list = []
    item_set = set()
    
    for record in data:
        transaction = frozenset(record)
        transaction_list.append(transaction)
        for item in transaction:
            item_set.add(frozenset([item]))
    
    return item_set, transaction_list

def return_items_minsup(item_set, transaction_list, min_sup, freq_set):
    """
    Calculate the itemsets that meet the minimum support threshold.
    
    Args:
        item_set (set): The set of candidate itemsets.
        transaction_list (list of frozenset): The list of transactions.
        min_sup (int): The minimum support threshold.
        freq_set (defaultdict): The frequency count of itemsets.
    
    Returns:
        set: The set of itemsets that meet the minimum support threshold.
    """
    qualified_item_set = set()
    local_set = defaultdict(int)
    
    for item in item_set:
        for transaction in transaction_list:
            if item.issubset(transaction):
                freq_set[item] += 1
                local_set[item] += 1
    
    for item, count in local_set.items():
        if count >= min_sup:
            qualified_item_set.add(item)
    
    return qualified_item_set

def join_set(item_set, length):
    """
    Generate candidate itemsets of a given length from the current itemsets.
    
    Args:
        item_set (set): The set of current itemsets.
        length (int): The length of the candidate itemsets to generate.
    
    Returns:
        set: The set of candidate itemsets of the given length.
    """
    return set([i.union(j) for i in item_set for j in item_set if len(i.union(j)) == length])

def apriori(data, min_sup):
    """
    Perform the Apriori algorithm to find frequent itemsets.
    
    Args:
        data (list of list of str): The dataset where each transaction is a list of items.
        min_sup (int): The minimum support threshold.
    
    Returns:
        dict: A dictionary where the keys are the lengths of the itemsets and the values are the frequent itemsets of that length.
    """
    item_set, transaction_list = get_itemset_transaction_list(data)
    freq_set = defaultdict(int)
    large_set = dict()
    
    one_c_set = return_items_minsup(item_set, transaction_list, min_sup, freq_set)
    current_l_set = one_c_set
    k = 2
    
    while current_l_set:
        large_set[k-1] = current_l_set
        current_l_set = join_set(current_l_set, k)
        current_c_set = return_items_minsup(current_l_set, transaction_list, min_sup, freq_set)
        current_l_set = current_c_set
        k += 1
    
    return large_set

def print_results(items, freq_set):
    """
    Print the frequent itemsets and their frequencies.
    
    Args:
        items (dict): The dictionary of frequent itemsets.
        freq_set (defaultdict): The frequency count of itemsets.
    """
    for k, item_set in items.items():
        print(f"Frequent {k}-itemsets:")
        for item in item_set:
            print(f"{item}: {freq_set[item]}")

    print_results(large_set, freq_set)
