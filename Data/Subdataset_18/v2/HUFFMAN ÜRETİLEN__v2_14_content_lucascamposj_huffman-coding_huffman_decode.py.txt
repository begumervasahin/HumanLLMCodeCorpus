import sys
import os
import math
from bitstring import BitArray

def read_file_metadata(file_data):
    """
    Extracts and returns the metadata from the file data, including the
    maximum code length, the number of bits required to represent the code size,
    and the padding size.
    
    Parameters
    ----------
    file_data : BitArray
        The BitArray containing the file data.
    
    Returns
    -------
    tuple
        A tuple containing the maximum code length, the number of bits for the code size, 
        and the padding size.
    """
    max_code_length = int(file_data[:8].uint)
    max_code_bits = int(math.ceil(math.log(max_code_length + 1, 2)))
    padding_size = int(file_data[8:11].uint)
    
    return max_code_length, max_code_bits, padding_size

def parse_symbol_codes(symbols_data, max_code_bits):
    """
    Parses the symbol codes from the provided BitArray.
    
    Parameters
    ----------
    symbols_data : BitArray
        The BitArray containing the symbol codes.
    max_code_bits : int
        The number of bits required to represent the code size.
    
    Returns
    -------
    dict
        A dictionary where keys are code sizes and values are dictionaries mapping
        binary strings to corresponding symbols.
    """
    symbols = {}
    pos = 0
    
    for s in range(256):
        if symbols_data[pos]:
            pos += 1
            code_size = int(symbols_data[pos:pos + max_code_bits].uint)
            code = symbols_data[pos + max_code_bits:pos + max_code_bits + code_size]
            pos += max_code_bits + code_size
            
            if code_size not in symbols:
                symbols[code_size] = {}
            symbols[code_size][code.bin] = BitArray(uint=s, length=8)
        else:
            pos += 1
    
    return symbols, pos

def decompress_data(data, symbols, max_code_length, output_file_name):
    """
    Decompresses the data using the Huffman codes and writes the output to a file.
    
    Parameters
    ----------
    data : BitArray
        The BitArray containing the compressed data.
    symbols : dict
        The dictionary containing the Huffman codes for decompression.
    max_code_length : int
        The maximum length of the Huffman codes.
    output_file_name : str
        The name of the file to write the decompressed data to.
    """
    final_data = BitArray()
    i = 0
    with open(output_file_name, "wb") as output_file:
        while i < data.length:
            buffer = BitArray(data[i:i + max_code_length])
            for size, codes in symbols.items():
                code = buffer[:size]
                if code.bin in codes:
                    codes[code.bin].tofile(output_file)
                    i += size
                    break

def main():
    if len(sys.argv) < 3:
        print("Usage: python decompress.py <input_file> <output_file>")
        return
    
    input_file_name = sys.argv[1]
    output_file_name = sys.argv[2]

    print(f"Input File: {input_file_name}")
    input_file_size = os.path.getsize(input_file_name)
    print(f"Input File Size: {input_file_size} Bytes")
    print("\nReading File...")

    # Read the input file as a BitArray
    with open(input_file_name, "rb") as input_file:
        file_data = BitArray(bytes=input_file.read())

    # Extract metadata
    max_code_length, max_code_bits, padding_size = read_file_metadata(file_data)
    
    print("Reading Symbol Codes...")
    
    # Parse the symbol codes
    symbols_data = file_data[11:]
    symbols, pos = parse_symbol_codes(symbols_data, max_code_bits)
    
    # Adjust the position by the padding size
    pos += padding_size
    
    print("Reading and Writing Compressed Data...")
    
    # Decompress the data
    data = symbols_data[pos:]
    decompress_data(data, symbols, max_code_length, output_file_name)
    
    # Print output file size
    output_file_size = os.path.getsize(output_file_name)
    print(f"\nOutput File: {output_file_name}")
    print(f"Output File Size: {output_file_size} Bytes")

if __name__ == "__main__":
    main()
