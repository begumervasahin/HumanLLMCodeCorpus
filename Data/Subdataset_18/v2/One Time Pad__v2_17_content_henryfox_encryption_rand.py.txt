import random

def generate_random_number_string(length: int) -> str:
    """
    Generates a random number string of a specified length.
    
    Parameters:
    length (int): The length of the random number string.
    
    Returns:
    str: The generated random number string.
    """
    return ''.join(str(random.randint(0, 9)) for _ in range(length))

def main() -> None:
    """
    Main function to demonstrate the generation of a random number string.
    """
    # Example usage: generate a random number string of length 10
    length = 10
    random_number = generate_random_number_string(length)
    print(f"Random number of length {length}: {random_number}")

if __name__ == "__main__":
    main()
