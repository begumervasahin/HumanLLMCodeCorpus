class Peekable:
    """
    An iterator that allows peeking at the next value without advancing the iterator.
    """
    def __init__(self, iterator):
        """
        Initializes the Peekable iterator.

        :param iterator: An existing iterator to enhance with peek functionality.
        """
        self._iterator = iterator
        self._peeked = None

    def __iter__(self):
        """
        Returns the iterator object itself.
        """
        return self

    def __next__(self):
        """
        Returns the next element from the iterator. If the element has been peeked at,
        it will return the peeked value and reset the peek.

        :return: The next element from the iterator.
        :raises StopIteration: If the iterator is exhausted.
        """
        if self._peeked is None:
            self._peeked = next(self._iterator)
        result = self._peeked
        self._peeked = None
        return result

    def peek(self):
        """
        Peeks at the next element of the iterator without advancing it. If the element
        has not been peeked at yet, it will retrieve it from the iterator.

        :return: The next element from the iterator.
        """
        if self._peeked is None:
            self._peeked = next(self._iterator)
        return self._peeked

def peek(iterator):
    """
    A helper function to peek at the next element of a Peekable iterator.

    :param iterator: A Peekable iterator instance.
    :return: The next element from the iterator.
    """
    return iterator.peek()

# Example usage
if __name__ == "__main__":
    # Create a Peekable iterator with a list of numbers
    i = Peekable(iter([1, 2, 3, 4, 5]))

    # Demonstrate peek and next functionality
    print(peek(i))  # Output: 1
    print(peek(i))  # Output: 1
    print(next(i))  # Output: 1
    print(next(i))  # Output: 2
    print(next(i))  # Output: 3
    print(peek(i))  # Output: 4
    print(next(i))  # Output: 4
    print(list(Peekable(iter([1, 2, 3, 4, 5]))))  # Output: [1, 2, 3, 4, 5]
