def main() -> None:
    """
    Main function to demonstrate the transformation of a list based on specific rules.
    """
    j = 3
    k = 7
    s = [8, 6, 7, 5, 3, 0, 9]

    for _ in range(10):
        for i in range(len(s)):
            if i == 0:
                # Calculate the new value to be inserted at the end of the list
                out = (s[j - 1] + s[k - 1]) % 10
            elif 0 < i < len(s) - 1:
                # Shift the elements to the left
                s[i] = s[i + 1]
            else:
                # Insert the calculated value at the end of the list
                s[i] = out
                print(s[i])

if __name__ == "__main__":
    main()
