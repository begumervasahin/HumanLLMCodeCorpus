import pandas as pd

# URL template to fetch historical data from Poloniex
url_template = 'https://poloniex.com/public?command=returnChartData&currencyPair={}&start=1439020800&end=1511202699&period=14400'

# List of cryptocurrencies to track against BTC
cryptos = [
    'ETH', 'VTC', 'STR', 'XRP', 'DASH', 'LTC', 'ETC', 'XMR', 'ZEC',
    'STRAT', 'XEM', 'VIA', 'SYS', 'LSK', 'FCT', 'CVC', 'BTS', 'DOGE',
    'DGB', 'DCR', 'SC', 'EXP', 'GNO', 'XCP', 'MAID', 'NXC', 'GNT', 
    'STEEM', 'ZRX', 'GAME', 'FLO', 'NAV', 'CLAM', 'LBC', 'HUC', 'OMNI',
    'POT', 'BURST', 'BCY', 'NXT', 'FLDC', 'ARDR', 'BLK', 'EMC2', 
    'NEOS', 'AMP', 'BCN', 'RADS', 'VRC', 'XVC', 'REP', 'BTCD', 'XBC', 
    'RIC', 'PASC', 'NMC', 'PPC', 'PINK', 'XPM', 'SBD', 'GRC', 'BELA', 
    'BTM'
]

def coin_lookup(coin):
    """
    Fetch historical data for a specific cryptocurrency from Poloniex.

    Args:
        coin (str): The symbol of the cryptocurrency (e.g., 'ETH').

    Returns:
        pd.DataFrame: DataFrame containing the historical data for the specified coin.
    """
    # Construct the currency pair string
    currency_pair = f'BTC_{coin}'
    
    # Format the URL with the currency pair
    url = url_template.format(currency_pair)
    
    # Fetch the data from the URL and load it into a DataFrame
    data = pd.read_json(url)
    
    return data

if __name__ == "__main__":
    # Example usage: Fetch data for Ethereum (ETH)
    eth_data = coin_lookup('ETH')
    
    # Display the first few rows of the DataFrame
    print(eth_data.head())

    # You can replace 'ETH' with any other cryptocurrency symbol from the `cryptos` list to fetch its data.
