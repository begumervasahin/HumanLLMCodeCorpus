from dotenv import load_dotenv
import os
import psycopg2
import pandas as pd
import time

# Load environment variables from a .env file
def load_env_vars():
    load_dotenv()
    global hostname, port, database, username, password
    hostname = os.getenv('hostname')
    port = os.getenv('port')
    database = os.getenv('database')
    username = os.getenv('username')
    password = os.getenv('password')

# Establish a connection to the PostgreSQL database
def create_connection():
    return psycopg2.connect(host=hostname, port=port, database=database, user=username, password=password)

# Fetch all records from the studies table
def fetch_studies(conn):
    return pd.read_sql('SELECT * FROM ctgov.studies', con=conn)

# Fetch all records from the keywords table
def fetch_keywords(conn):
    return pd.read_sql('SELECT * FROM ctgov.keywords', con=conn)

# Fetch all records from the sponsors table
def fetch_sponsors(conn):
    return pd.read_sql('SELECT * FROM ctgov.sponsors', con=conn)

# Perform a right join between two tables on the nct_id column
def join_tables(conn, first_table_name, second_table_name):
    query = f"""
        SELECT * FROM {first_table_name}
        RIGHT JOIN {second_table_name}
        ON {first_table_name}.nct_id = {second_table_name}.nct_id
    """
    return pd.read_sql(query, con=conn)

if __name__ == '__main__':
    # Load environment variables
    load_env_vars()

    # Create a database connection
    conn = create_connection()

    # Measure and print the time taken for the join operation using the join_tables function
    start_time = time.time()
    joined_results = join_tables(conn, 'ctgov.studies', 'ctgov.keywords')
    end_time = time.time()
    print("Join operation time using function:", end_time - start_time)

    # Measure and print the time taken for the join operation using a raw SQL query
    start_time = time.time()
    pd.read_sql('SELECT * FROM ctgov.studies RIGHT JOIN ctgov.keywords ON ctgov.studies.nct_id = ctgov.keywords.nct_id', con=conn)
    end_time = time.time()
    print("Join operation time using raw SQL:", end_time - start_time)

    # Close the database connection
    conn.close()
