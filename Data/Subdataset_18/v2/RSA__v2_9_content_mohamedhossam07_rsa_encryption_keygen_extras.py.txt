from random import randint

def power(x, y, p):
    """Compute (x^y) % p efficiently."""
    return (x ** y) % p

def decimal(decimalin):
    """Convert a decimal number to its binary representation."""
    decimalin = int(decimalin)
    binary_rep = ""
    while decimalin > 0:
        binary_rep = str(decimalin % 2) + binary_rep
        decimalin //= 2
    return binary_rep or "0"

def size_be_R(key, size):
    """Adjust the size of a binary string to a specified size by padding or truncating."""
    if len(key) < size:
        return key.rjust(size, '0')
    elif len(key) > size:
        return key[:size]
    return key

def binary2dec(binaryin):
    """Convert a binary string to its decimal representation."""
    binaryin = binaryin[::-1]
    decimal_value = sum(int(bit) * (2 ** idx) for idx, bit in enumerate(binaryin))
    return decimal_value

def rand_bin(length):
    """Generate a random binary number of a given length and convert it to decimal."""
    binary_str = ''.join(size_be_R(bin(randint(0, 10))[2:], 8) for _ in range(length // 8))
    binary_str = size_be_R(binary_str, length)
    return binary2dec(binary_str)

def multiplicative_inverse(a, b):
    """Find the multiplicative inverse of a modulo b using the Extended Euclidean Algorithm."""
    x, y, lx, ly = 0, 1, 1, 0
    oa, ob = a, b
    while b != 0:
        q = a // b
        a, b = b, a % b
        x, lx = lx - q * x, x
        y, ly = ly - q * y, y
    if lx < 0:
        lx += ob
    if ly < 0:
        ly += oa
    return lx
