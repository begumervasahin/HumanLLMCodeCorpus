def merge_sort(arr):
    """Recursively sorts an array using the merge sort algorithm."""
    if len(arr) <= 1:
        return

    # Split the array into two halves
    mid = len(arr) // 2
    left_half = arr[:mid]
    right_half = arr[mid:]

    # Recursively sort both halves
    merge_sort(left_half)
    merge_sort(right_half)

    # Merge the sorted halves back into the original array
    merge(arr, left_half, right_half)

def merge(arr, left, right):
    """Merges two sorted arrays into a single sorted array."""
    i = j = k = 0

    # Compare each element and merge them back into `arr`
    while i < len(left) and j < len(right):
        if left[i] < right[j]:
            arr[k] = left[i]
            i += 1
        else:
            arr[k] = right[j]
            j += 1
        k += 1

    # Copy any remaining elements from the left half
    while i < len(left):
        arr[k] = left[i]
        i += 1
        k += 1

    # Copy any remaining elements from the right half
    while j < len(right):
        arr[k] = right[j]
        j += 1
        k += 1

def main():
    """Main function to execute the merge sort algorithm."""
    num_elements = int(input("How many elements do you want in the list? "))
    arr = [int(input(f"Enter element {i + 1}: ")) for i in range(num_elements)]

    merge_sort(arr)

    print("Sorted list is:", arr)

if __name__ == "__main__":
    main()
