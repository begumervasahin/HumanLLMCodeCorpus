import doctest

def fizzbuzz(n):
    """
    Return 'fizz' if n is divisible by three.
    Return 'buzz' if n is divisible by five.
    Return 'fizzbuzz' if n is divisible by both three and five.
    In other cases return n as a string.
    
    >>> fizzbuzz(1)
    '1'
    >>> fizzbuzz(2)
    '2'
    >>> fizzbuzz(3)
    'fizz'
    >>> fizzbuzz(4)
    '4'
    >>> fizzbuzz(5)
    'buzz'
    >>> fizzbuzz(6)
    'fizz'
    >>> fizzbuzz(14)
    '14'
    >>> fizzbuzz(15)
    'fizzbuzz'
    >>> fizzbuzz(16)
    '16'
    """
    if n % 3 == 0 and n % 5 == 0:
        return 'fizzbuzz'
    elif n % 3 == 0:
        return 'fizz'
    elif n % 5 == 0:
        return 'buzz'
    else:
        return str(n)

if __name__ == '__main__':
    # Run doctests to validate the function
    doctest.testmod()
    
    # Print the results for numbers 1 through 100
    for n in range(1, 101):
        print(fizzbuzz(n))
