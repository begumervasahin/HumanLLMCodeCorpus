import time
from selenium import webdriver
from selenium.webdriver.common.by import By

# Constants
URL = 'https://qainterview.pythonanywhere.com/'
TEXTBOX_ID = 'number'
CALCULATE_BUTTON_XPATH = "//*[@id='getFactorial']"
RESULT_XPATH = "//p[@id='resultDiv']"
EXPECTED_RESULT = "The factorial of 12 is: 479001600"

def main():
    # Step 1: Launch Firefox Driver
    driver = webdriver.Firefox()

    try:
        # Step 2: Navigate to the page
        driver.get(URL)

        # Step 3: Enter integer value in the textbox
        driver.find_element(By.ID, TEXTBOX_ID).send_keys('12')

        # Step 4: Click on the calculate button
        driver.find_element(By.XPATH, CALCULATE_BUTTON_XPATH).click()

        # Step 5: Check the validation message
        time.sleep(3)  # Wait for the result to be displayed
        validation_result = driver.find_element(By.XPATH, RESULT_XPATH).text

        # Step 6: Verify the validation message
        if validation_result == EXPECTED_RESULT:
            print("Success: Message displayed successfully")
        else:
            print("Failed: Validation message is incorrect")

    finally:
        # Step 7: Close the browser
        driver.quit()

if __name__ == "__main__":
    main()
