import tkinter as tk
import tkinter.messagebox
import os
from sympy.ntheory import isprime

def pqeValidacao(p, q, e):
    """Validates if p and q are prime and e is in the correct range."""
    if not (p.isdigit() and q.isdigit() and e.isdigit()):
        return "Todos os valores devem ser números inteiros."
    
    p, q, e = int(p), int(q), int(e)
    
    if not (isprime(p) and isprime(q)):
        return "p e q devem ser números primos."
    
    if not (1 < e < (p - 1) * (q - 1)):
        return "e deve estar no intervalo (1, (p-1)*(q-1))."
    
    return ""

class Option01(tk.Frame):
    """GUI frame for generating a public key."""
    
    def __init__(self, master):
        super().__init__(master)
        
        self.setup_ui()

    def setup_ui(self):
        """Sets up the UI components."""
        tk.Label(self, text="Gerar chave publica", font=('Verdana', 22)).pack(side="top", fill="x")
        tk.Label(self, text="Insira os valores abaixo", font=('Verdana', 10)).pack(side="top", fill="x")
        
        # Adding some vertical space
        for _ in range(5):
            tk.Label(self, text="", font=('Verdana', 22)).pack(side="top", fill="x")
        
        # Labels for p, q, and e
        self.create_label("p:", 80, 65)
        self.create_label("q:", 80, 90)
        self.create_label("e:", 80, 115)
        
        # Entry fields for p, q, and e
        self.p_entry = self.create_entry(100, 65)
        self.q_entry = self.create_entry(100, 90)
        self.e_entry = self.create_entry(100, 115)
        
        # Button to generate the key
        ok_button = tk.Button(self, text="Gerar chave", command=self.generate_key)
        ok_button.pack()
        ok_button.place(x=95, y=145)

    def create_label(self, text, x, y):
        """Creates and places a label."""
        label = tk.Label(self, text=text)
        label.pack()
        label.place(x=x, y=y)

    def create_entry(self, x, y):
        """Creates and places an entry field."""
        entry = tk.Entry(self)
        entry.pack()
        entry.place(x=x, y=y)
        return entry
    
    def generate_key(self):
        """Generates the public key based on p, q, and e."""
        p = self.p_entry.get()
        q = self.q_entry.get()
        e = self.e_entry.get()
        
        error_message = pqeValidacao(p, q, e)
        title = "ERRO!" if error_message else "Concluido"
        
        if not error_message:
            if os.path.exists("chavePublica.txt"):
                os.remove("chavePublica.txt")
                
            with open("chavePublica.txt", "w") as file:
                file.write(str((int(p) * int(q), int(e))))
            
            error_message = "Chave Gerada com Sucesso!"
        
        tk.messagebox.showinfo(title, error_message)

if __name__ == "__main__":
    root = tk.Tk()
    root.geometry("300x200")
    Option01(root).pack(side="top", fill="both", expand=True)
    root.mainloop()
