from selenium import webdriver
from selenium.webdriver.common.by import By
import re

def get_input(prompt):
    return input(prompt)

def initialize_browser(path):
    return webdriver.Chrome(executable_path=path)

def fetch_details(browser, htno):
    htno_input = browser.find_element(By.NAME, 'htno')
    htno_input.clear()
    htno_input.send_keys(htno)
    browser.find_element(By.NAME, 'Submit').click()
    return browser.find_element(By.ID, 'AutoNumber3').text

def extract_student_info(details_table):
    pin_no = re.search(r"\d{12}", details_table).group()
    name = re.search(r"Name [A-Z .]*\s", details_table).group()
    return pin_no, name

def extract_marks(marks_table):
    marks_list = marks_table.splitlines()[2:]
    marks = []
    for subject in marks_list:
        subject_code = re.search(r"\d{3}", subject).group()
        subject_name = re.findall(r"[A-Z -.&]*", subject)[4]
        gpa_scored = re.findall(r"\d+", subject)[-1]
        marks.append((subject_code, subject_name, gpa_scored))
    return marks

def extract_result(result_table):
    return result_table.splitlines()[-1]

def write_student_info(file, pin_no, name, marks, result):
    file.write("\n\n-------------------------------------------------------------\n")
    file.write(f"PIN NO: {pin_no}\t{name}\n")
    file.write("-------------------------------------------------------------\n")
    for subject_code, subject_name, gpa_scored in marks:
        file.write(f"{subject_code}\t{subject_name}\t\t{gpa_scored}\n")
    file.write(f"RESULT: {result}\n")
    file.write("-------------------------------------------------------------\n\n\n\n")

def find_results(browser, file, pattern, start, end):
    detained = 0
    for i in range(start, end + 1):
        try:
            htno = pattern + str(i).zfill(3)
            details_table = fetch_details(browser, htno)
            pin_no, name = extract_student_info(details_table)
            marks_table = browser.find_element(By.ID, 'AutoNumber4').text
            marks = extract_marks(marks_table)
            result_table = browser.find_element(By.ID, 'AutoNumber5').text
            result = extract_result(result_table)
            if "DETAINED" in result:
                detained += 1
            write_student_info(file, pin_no, name, marks, result)
        except Exception as e:
            print(e)
            file.write("---------SKIPPING ONE PIN HERE-------------\n\n\n")
    return detained

def main():
    file_name = get_input("Enter a file name to be saved: ")
    pattern = get_input("Enter pattern: ")
    path = "/usr/lib/chromium-browser/chromedriver"
    url = "https://www.osmania.ac.in/res07/20190855.jsp"

    with open(f"{file_name}.txt", "w") as file:
        browser = initialize_browser(path)
        browser.get(url)
        
        detained_count = find_results(browser, file, pattern, 1, 59)
        detained_count += find_results(browser, file, pattern, 301, 311)
        
        file.seek(0, 0)
        file.write(f"NO. OF STUDENTS DETAINED: {detained_count}\n")
        
        browser.quit()

if __name__ == "__main__":
    main()
