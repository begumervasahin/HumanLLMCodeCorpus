# Open the file "distance_matrix_1698.csv" in read and write mode
with open("distance_matrix_1698.csv", "r+") as file:
    lines = []  # List to store lines from the file
    line_count = 0  # Counter for the number of lines read

    # Iterate over the lines in the file
    for line in file:
        lines.append(line.strip())  # Add each line to the list after stripping newline characters
        line_count += 1  # Increment the line counter

        # Break the loop if 5 lines have been read
        if line_count == 5:
            break

    # Split the fifth line by commas to get a list of values
    if line_count == 5:
        values = lines[-1].split(",")

# Print the length of the list 'values' plus 1698
print(len(values) + 1698)

# Print the third element in the list 'values'
print(values[2])
