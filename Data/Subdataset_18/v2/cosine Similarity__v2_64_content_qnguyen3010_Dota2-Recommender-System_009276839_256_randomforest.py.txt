import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import confusion_matrix, accuracy_score, f1_score

def load_data(train_path, test_path):
    """
    Load training and testing datasets from CSV files.
    
    Parameters:
        train_path (str): Path to the training dataset.
        test_path (str): Path to the testing dataset.
    
    Returns:
        X_train (ndarray): Training features.
        y_train (ndarray): Training labels.
        X_test (ndarray): Testing features.
        y_test (ndarray): Testing labels.
    """
    train_data = pd.read_csv(train_path)
    test_data = pd.read_csv(test_path)
    
    y_train = train_data.iloc[:, 0].values
    X_train = train_data.iloc[:, 4:].values
    y_test = test_data.iloc[:, 0].values
    X_test = test_data.iloc[:, 4:].values
    
    return X_train, y_train, X_test, y_test

def train_classifier(X_train, y_train):
    """
    Train a Random Forest classifier on the training data.
    
    Parameters:
        X_train (ndarray): Training features.
        y_train (ndarray): Training labels.
    
    Returns:
        classifier (RandomForestClassifier): Trained classifier.
    """
    classifier = RandomForestClassifier(max_features='log2', n_estimators=100, criterion='entropy', random_state=0)
    classifier.fit(X_train, y_train)
    return classifier

def evaluate_classifier(classifier, X_test, y_test):
    """
    Evaluate the classifier on the testing data and print metrics.
    
    Parameters:
        classifier (RandomForestClassifier): Trained classifier.
        X_test (ndarray): Testing features.
        y_test (ndarray): Testing labels.
    """
    y_pred = classifier.predict(X_test)
    
    # Compute the confusion matrix
    cm = confusion_matrix(y_test, y_pred)
    print("Confusion Matrix:")
    print(cm)
    
    # Compute and print the accuracy
    accuracy = accuracy_score(y_test, y_pred)
    print("\nAccuracy:", accuracy)
    
    # Compute and print the F1 score
    f1 = f1_score(y_test, y_pred, average='macro')
    print("F1 Score:", f1)

def main():
    # Paths to the datasets
    train_path = 'dota2Train.csv'
    test_path = 'dota2Test.csv'
    
    # Load data
    X_train, y_train, X_test, y_test = load_data(train_path, test_path)
    
    # Train classifier
    classifier = train_classifier(X_train, y_train)
    
    # Evaluate classifier
    evaluate_classifier(classifier, X_test, y_test)

if __name__ == "__main__":
    main()
