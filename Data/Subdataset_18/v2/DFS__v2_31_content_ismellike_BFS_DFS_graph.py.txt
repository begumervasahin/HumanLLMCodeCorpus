class Node:
    def __init__(self, index):
        self.index = index
        self.children = []
        self.discovered = False

    def add_child(self, child_index):
        """
        Adds a child node index to the node's children list.
        
        Args:
            child_index (int): The index of the child node.
        """
        self.children.append(child_index)

class Graph:
    def __init__(self, head_index):
        """
        Initializes the Graph with a head node and an empty list of nodes.
        
        Args:
            head_index (int): The index of the head node.
        """
        self.head_index = head_index
        self.nodes = []

    def add_node(self, node):
        """
        Adds a node to the graph's list of nodes.
        
        Args:
            node (Node): The node to add to the graph.
        """
        self.nodes.append(node)

def read_graph_from_file(file_path):
    """
    Reads a file and constructs a graph from the data.
    
    Args:
        file_path (str): The path to the file containing the graph data.
    
    Returns:
        Graph: The constructed graph with nodes and edges.
    """
    with open(file_path, 'r') as file:
        head_index = int(file.readline().strip())
        graph = Graph(head_index)
        
        current_index = -1
        current_node = None

        for line in file:
            parent_index, child_index = map(int, line.strip().split(','))
            if current_index == parent_index:
                current_node.add_child(child_index)
            else:
                if current_node is not None:
                    graph.add_node(current_node)
                current_index = parent_index
                current_node = Node(parent_index)
                current_node.add_child(child_index)
        
        if current_node is not None:
            graph.add_node(current_node)

    return graph
