import random
from collections import defaultdict

# Initialize the data structures
edges = []
tree = defaultdict(list)
node_id = 0

# Create a 9x9 grid of nodes and corresponding edges
for x in range(9):
    for y in range(9):
        tree[node_id].append((x, y))
        node_id += 1
        
        # Generate edges to connect each node to its neighbors (excluding diagonals)
        for dx in range(-1, 2):
            for dy in range(-1, 2):
                if abs(dx) == abs(dy): 
                    continue
                
                neighbor = (x + dx, y + dy)
                if 0 <= neighbor[0] <= 8 and 0 <= neighbor[1] <= 8:
                    if [neighbor, (x, y)] in edges:
                        continue
                    edges.append([(x, y), neighbor])

# Randomly connect nodes until only one tree remains
selected_edges = []
while len(tree) > 1:
    random_index = random.randint(0, len(edges) - 1)
    
    # Identify the trees to which the endpoints of the selected edge belong
    tree1, tree2 = None, None
    for tree_id, nodes in tree.items():
        if edges[random_index][0] in nodes:
            tree1 = (tree_id, nodes)
        if edges[random_index][1] in nodes:
            tree2 = (tree_id, nodes)
        if tree1 and tree2:
            break

    # If both endpoints are already in the same tree, skip this edge
    if tree1[0] == tree2[0]:
        continue

    # Merge tree2 into tree1
    print(f"Connecting edge: {edges[random_index]}")
    print(f"Before merging, tree1: {tree1[1]}")
    print(f"Before merging, tree2: {tree2[1]}")
    
    tree1[1].extend(tree2[1])
    tree.pop(tree2[0])
    
    # Remove the selected edge from the list and track it
    edges.pop(random_index)
    selected_edges.append(random_index)
    
    # Display the current state of the trees and remaining edges
    for nodes in tree.values():
        print(nodes)
    
    print(f"Remaining edges: {len(edges)}, Remaining trees: {len(tree)}, Selected edge index: {random_index}")
    print()

# Display the final tree structure
for tree_id, nodes in tree.items():
    print(f"Tree {tree_id}: {nodes}")
