from __future__ import print_function

# Publicly shared parameters
shared_prime = 23
shared_base = 5

# Alice's and Bob's private secrets
alice_secret = 6
bob_secret = 15

# Display publicly shared parameters
print("Publicly Shared Parameters:")
print(f"    Prime: {shared_prime}")
print(f"    Base:  {shared_base}")

# Compute Alice's public value
alice_public = pow(shared_base, alice_secret, shared_prime)
print(f"\n  Alice's Public Value (sent to Bob): {alice_public}")

# Compute Bob's public value
bob_public = pow(shared_base, bob_secret, shared_prime)
print(f"  Bob's Public Value (sent to Alice): {bob_public}")

print("\n------------\n")

# Compute the shared secret privately
print("Privately Calculated Shared Secrets:")

# Alice calculates the shared secret using Bob's public value
alice_shared_secret = pow(bob_public, alice_secret, shared_prime)
print(f"    Alice's Shared Secret: {alice_shared_secret}")

# Bob calculates the shared secret using Alice's public value
bob_shared_secret = pow(alice_public, bob_secret, shared_prime)
print(f"    Bob's Shared Secret: {bob_shared_secret}")
