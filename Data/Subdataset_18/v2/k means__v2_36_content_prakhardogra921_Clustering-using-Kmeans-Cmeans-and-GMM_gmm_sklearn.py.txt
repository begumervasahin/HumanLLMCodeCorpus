import numpy as np
from sklearn import mixture
import matplotlib.pyplot as plt
from scipy import linalg
import itertools
import matplotlib as mpl
import random

def generate_dataset(size, loc1, scale1, loc2, scale2, loc3, scale3, p1, p2, p3):
    set1 = np.random.normal(loc=loc1, scale=scale1, size=size)
    set2 = np.random.normal(loc=loc2, scale=scale2, size=size)
    set3 = np.random.normal(loc=loc3, scale=scale3, size=size)
    
    dataset = np.array(
        random.sample(list(set1), int(p1 * size)) + 
        random.sample(list(set2), int(p2 * size)) + 
        random.sample(list(set3), int(p3 * size))
    )
    return dataset

def plot_gmm_1d(dataset, n_components=3):
    gmm = mixture.GaussianMixture(n_components=n_components)
    gmm.fit(dataset.reshape(-1, 1))
    y = gmm.predict(dataset.reshape(-1, 1))
    plt.scatter(dataset, y)
    plt.title('GMM')
    plt.show()

def plot_gmm_with_ellipses(data, n_components=2):
    gmm = mixture.GaussianMixture(n_components=n_components)
    gmm.fit(data.reshape(-1, 1))
    y = gmm.predict(data.reshape(-1, 1))

    fig, ax = plt.subplots()
    color_iter = itertools.cycle(['navy', 'turquoise', 'cornflowerblue', 'darkorange'])
    
    for i, (mean, cov, color) in enumerate(zip(gmm.means_, gmm.covariances_, color_iter)):      
        if not np.any(y == i):
            continue
        plt.scatter(data[y == i], np.zeros_like(data[y == i]), 0.8, color=color)
        v, w = linalg.eigh(cov)
        angle = np.arctan2(w[0][1], w[0][0])
        angle = 180. * angle / np.pi
        v = 2. * np.sqrt(2.) * np.sqrt(v)
        ell = mpl.patches.Ellipse(mean, v[0], v[1], 180. + angle, color=color)
        ell.set_clip_box(ax.bbox)
        ell.set_alpha(0.5)
        ax.add_artist(ell)
    
    plt.xticks(())
    plt.yticks(())
    plt.title('GMM with Ellipses')
    plt.show()

# Parameters
size = 500
p1, p2, p3 = 0.25, 0.5, 0.25

# Generate datasets
dset1 = generate_dataset(size, loc1=1, scale1=0.1, loc2=1.5, scale2=0.1, loc3=2, scale3=0.2, p1=p1, p2=p2, p3=p3)
dset2 = generate_dataset(size, loc1=1, scale1=0.3, loc2=1.5, scale2=0.4, loc3=2, scale3=0.3, p1=p1, p2=p2, p3=p3)

# Plot results
plot_gmm_1d(dset1)
plot_gmm_with_ellipses(dset1)
