import numpy as np
import matplotlib.pyplot as plt

# Load data from the file
data = np.loadtxt("crash.txt")

# Split data into training and test sets
training = data[1::2]
test = data[0:-1:2]

# Separate features and targets for training and test sets
training_x = training[:, 0].reshape(-1, 1)
training_t = training[:, 1].reshape(-1, 1)
test_x = test[:, 0].reshape(-1, 1)
test_t = test[:, 1].reshape(-1, 1)

# Initialize variables for storing RMS errors and best models
Erms_training = np.zeros(20)
Erms_test = np.zeros(20)
best_training_error = float('inf')
best_test_error = float('inf')
best_w_training = None
best_L_training = 0
best_w_test = None
best_L_test = 0

# Function to compute the design matrix phi
def design_matrix(x, L):
    return np.hstack([x**i for i in range(L)])

# Loop over different polynomial degrees
for L in range(1, 21):
    phi_train = design_matrix(training_x, L)
    w = np.linalg.solve(phi_train.T.dot(phi_train), phi_train.T.dot(training_t))
    
    E_training = 0.5 * np.square(np.linalg.norm(training_t - phi_train.dot(w)))
    Erms_training[L-1] = np.sqrt(2. * E_training / len(training))
    
    if Erms_training[L-1] < best_training_error:
        best_training_error = Erms_training[L-1]
        best_L_training = L
        best_w_training = w
    
    phi_test = design_matrix(test_x, L)
    E_test = 0.5 * np.square(np.linalg.norm(test_t - phi_test.dot(w)))
    Erms_test[L-1] = np.sqrt(2. * E_test / len(test))
    
    if Erms_test[L-1] < best_test_error:
        best_test_error = Erms_test[L-1]
        best_L_test = L
        best_w_test = w

# Print the RMS errors
print('Maximum likelihood RMS error between the actual data and the models prediction (for Training sets): \n')
print(Erms_training)
print('Maximum likelihood RMS error between the actual data and the models prediction (for Test sets): \n')
print(Erms_test)

# Plot the RMS errors
plt.figure(figsize=(16, 12))
plt.plot(Erms_training, '-o', markerfacecolor='none', color='b', label='Training')
plt.plot(Erms_test, '-o', markerfacecolor='none', color='r', label='Test')
plt.suptitle('Maximum likelihood RMS error between the actual data and the models prediction', fontsize=24)
plt.legend(fontsize=22)
plt.xlabel("M ", fontsize=22)
plt.ylabel("Erms", fontsize=22)
plt.show()

# Plot the best fit for the training set
x = np.linspace(np.min(training_x), np.max(training_x), 100).reshape(-1, 1)
phi = design_matrix(x, best_L_training)
y = phi.dot(best_w_training)
print('Lowest RMS L for training data:', best_L_training)
plt.figure(figsize=(16, 12))
plt.plot(training_x, training_t, 'o', color='b', label='Training data')
plt.plot(x, y, color='r', label='Lowest RMS model output')
plt.suptitle('Best fit on the training set', fontsize=24)
plt.legend(fontsize=22)
plt.xlabel("time", fontsize=22)
plt.ylabel("acceleration", fontsize=22)
plt.show()

# Plot the best fit for the test set
x = np.linspace(np.min(test_x), np.max(test_x), 100).reshape(-1, 1)
phi = design_matrix(x, best_L_test)
y = phi.dot(best_w_test)
print('Lowest RMS L for test data:', best_L_test)
plt.figure(figsize=(16, 12))
plt.plot(test_x, test_t, 'o', color='b', label='Test data')
plt.plot(x, y, color='r', label='Lowest RMS model output')
plt.suptitle('Best fit on the test set', fontsize=24)
plt.legend(fontsize=22)
plt.xlabel("time", fontsize=22)
plt.ylabel("acceleration", fontsize=22)
plt.show()
