import gmplot
from math import sin, cos, sqrt, atan2, radians

def haversine_np(pointa, pointb):
    R = 6373.0
    lat1, lon1 = radians(pointa[2]), radians(pointa[1])
    lat2, lon2 = radians(pointb[2]), radians(pointb[1])
    dlon, dlat = lon2 - lon1, lat2 - lat1
    a = sin(dlat / 2) ** 2 + cos(lat1) * cos(lat2) * sin(dlon / 2) ** 2
    c = 2 * atan2(sqrt(a), sqrt(1 - a))
    return R * c

def compute_distances(trajectory):
    total_distance, max_distance = 0.0, 0.0
    for i in range(len(trajectory) - 1):
        distance = haversine_np(trajectory[i], trajectory[i + 1])
        total_distance += distance
        if distance > max_distance:
            max_distance = distance
    return total_distance, max_distance

def plot_traj(trajectory, plotname):
    longitudes = [point[1] for point in trajectory]
    latitudes = [point[2] for point in trajectory]
    gmap = gmplot.GoogleMapPlotter(latitudes[0], longitudes[0], 12)
    gmap.plot(latitudes, longitudes, 'cornflowerblue', edge_width=10)
    gmap.draw(f"{plotname}.html")

def plot_traj_red(common_points, trajectory, plotname):
    longitudes = [point[1] for point in trajectory]
    latitudes = [point[2] for point in trajectory]
    gmap = gmplot.GoogleMapPlotter(latitudes[0], longitudes[0], 12)
    gmap.plot(latitudes, longitudes, 'cornflowerblue', edge_width=10)
    
    longitudes_red = [point[1] for point in common_points]
    latitudes_red = [point[2] for point in common_points]
    gmap.plot(latitudes_red, longitudes_red, 'red', edge_width=10)
    
    gmap.draw(f"{plotname}.html")

def print_results(df_traj, distances, index, name):
    target_traj = df_traj.iloc[index]
    traj_coordinates = target_traj['timestamp_longitude_latitude']
    journeypattid = target_traj['JourneyPatternId']
    print(f"Traj: {journeypattid} Distance is {distances[index]:.4f} km's.")
    plot_traj(traj_coordinates, name)

def lcss_trigger(traj, df, trigger, index=None):
    if trigger == 1:
        matching_points = []
        for elem in df['timestamp_longitude_latitude']:
            n0, n1 = len(traj), len(elem)
            C = [[0] * (n1 + 1) for _ in range(n0 + 1)]
            for i in range(1, n0 + 1):
                for j in range(1, n1 + 1):
                    if haversine_np(traj[i - 1], elem[j - 1]) <= 0.2:
                        C[i][j] = C[i - 1][j - 1] + 1
                    else:
                        C[i][j] = max(C[i][j - 1], C[i - 1][j])
            matching_points.append(C[n0][n1])
        return matching_points

    elif trigger == 2 and index is not None:
        common_points = []
        elem = df['timestamp_longitude_latitude'].iloc[index]
        n0, n1 = len(traj), len(elem)
        C = [[0] * (n1 + 1) for _ in range(n0 + 1)]
        for i in range(1, n0 + 1):
            for j in range(1, n1 + 1):
                if haversine_np(traj[i - 1], elem[j - 1]) <= 0.2:
                    C[i][j] = C[i - 1][j - 1] + 1
                    if elem[j - 1] not in common_points:
                        common_points.append(elem[j - 1])
                else:
                    C[i][j] = max(C[i][j - 1], C[i - 1][j])
        return common_points
