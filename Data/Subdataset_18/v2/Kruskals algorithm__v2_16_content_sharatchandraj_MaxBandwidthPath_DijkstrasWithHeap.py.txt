import heapq
from collections import namedtuple, defaultdict

# Define an Edge class using namedtuple for simplicity
Edge = namedtuple('Edge', ['vertex', 'weight'])

class DijkstraWithMaxHeap:
    def __init__(self, num_vertices):
        self.num_vertices = num_vertices
        self.status = ['unseen'] * num_vertices
        self.weights = [float('-inf')] * num_vertices
        self.parents = [None] * num_vertices
        self.graph = defaultdict(list)

    def add_edge(self, u, v, weight):
        """Adds an undirected edge between vertices u and v with the given weight."""
        self.graph[u].append(Edge(vertex=v, weight=weight))
        self.graph[v].append(Edge(vertex=u, weight=weight))  # Undirected graph

    def _insert_into_max_heap(self, heap, vertex, weight):
        """Inserts a vertex into the max heap with the given weight."""
        heapq.heappush(heap, (-weight, vertex))

    def _pop_from_max_heap(self, heap):
        """Pops the vertex with the maximum weight from the heap."""
        weight, vertex = heapq.heappop(heap)
        return -weight, vertex

    def run_dijkstra(self, source, destination):
        """Runs Dijkstra's algorithm from source to destination to find the maximum bandwidth path."""
        heap = []
        self.status[source] = 'intree'

        # Initialize the heap with edges from the source vertex
        for edge in self.graph[source]:
            self.status[edge.vertex] = 'fringe'
            self.weights[edge.vertex] = edge.weight
            self._insert_into_max_heap(heap, edge.vertex, edge.weight)
            self.parents[edge.vertex] = source

        while heap:
            max_weight, current_vertex = self._pop_from_max_heap(heap)

            if current_vertex == destination:
                break

            self.status[current_vertex] = 'intree'

            for edge in self.graph[current_vertex]:
                if self.status[edge.vertex] == 'unseen':
                    self.status[edge.vertex] = 'fringe'
                    self.parents[edge.vertex] = current_vertex
                    self.weights[edge.vertex] = min(max_weight, edge.weight)
                    self._insert_into_max_heap(heap, edge.vertex, self.weights[edge.vertex])

                elif self.status[edge.vertex] == 'fringe' and self.weights[edge.vertex] < min(max_weight, edge.weight):
                    self.parents[edge.vertex] = current_vertex
                    self.weights[edge.vertex] = min(max_weight, edge.weight)
                    self._insert_into_max_heap(heap, edge.vertex, self.weights[edge.vertex])

        return self._build_path(destination), self.weights[destination]

    def _build_path(self, destination):
        """Reconstructs the path from the source to the destination."""
        path = []
        while destination is not None:
            path.append(destination)
            destination = self.parents[destination]
        return path[::-1]  # Reverse the path to start from the source
