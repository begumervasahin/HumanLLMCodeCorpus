import re
import urllib.parse
import subprocess
import requests
from crossref.restful import Works

import metainfoscrapper

# Initialize CrossRef Works
works = Works()

# List to store citation URLs
citation_urls = []

def get_doi_by_title(title):
    """
    Fetch the DOI for a given title using CrossRef search.
    
    Args:
        title (str): The title of the paper to search for.
    
    Returns:
        str: The DOI of the paper, if found. None otherwise.
    """
    try:
        # Prepare the search URL
        args = {"q": title}
        url = f"https://search.crossref.org/?{urllib.parse.urlencode(args)}"
        
        # Send the GET request
        response = requests.get(url)
        response.raise_for_status()
        
        # Parse the response content
        soup = BeautifulSoup(response.content, 'html.parser')
        
        # Find the DOI link
        doi_link = soup.find(href=re.compile("https://doi.org/"))
        if doi_link:
            match = re.search("(?P<url>https?://[^\s]+)", str(doi_link))
            if match:
                return match.group("url")[16:-2]
    
    except Exception as e:
        print(f"Error fetching DOI: {e}")
    
    return None

def get_citations(title):
    """
    Get citations for a given title using a Scholar script.
    
    Args:
        title (str): The title of the paper to search for citations.
    """
    try:
        # Prepare the command for the Scholar script
        command = f'python scholar.py -c 1 -A "{title}"'
        
        # Execute the command and capture the output
        result = subprocess.check_output(command, shell=True)
        if not result:
            print("Result fetching Error! IP blocking!")
            return
        
        # Decode the result and extract citation URLs
        result_str = result.decode('windows-1252')
        citation_urls.extend(re.findall(r'http://scholar.google.com/scholar\?cites=[^\s]+', result_str))
        
        # Fetch details of the first citation URL
        if citation_urls:
            fetch_citation_details(citation_urls[0])
    
    except Exception as e:
        print(f"Error fetching citations: {e}")

def fetch_citation_details(main_url):
    """
    Fetch the details of citations from the provided URL.
    
    Args:
        main_url (str): The main URL to fetch citation details from.
    """
    i = 0
    while True:
        try:
            # Prepare the URL with pagination
            url = f"{main_url}&start={i}"
            
            # Send the GET request
            response = requests.get(url)
            response.raise_for_status()
            
            # Parse the response content
            soup = BeautifulSoup(response.content, 'html.parser')
            
            # Find the citation list
            citation_list = soup.find_all('h3', class_='gs_rt')
            if not citation_list:
                break
            
            # Process each citation
            for paper in citation_list:
                name = paper.get_text().split(' ', 1)[1] if paper.get_text().startswith('[') else paper.get_text()
                doi = get_doi_by_title(name)

def main_search(doi):
    """
    Main function to start the search process using a given DOI.
    
    Args:
        doi (str): The DOI to start the search with.
    """
    try:
        # Fetch the title of the paper using the DOI
        title = works.doi(doi)['title']
        
        # Get citations for the fetched title
        get_citations(title)
    
    except Exception as e:
        print(f"Error in main search: {e}")
