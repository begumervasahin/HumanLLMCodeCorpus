from functools import wraps
from typing import Callable, Dict, Generic, Tuple, Type, TypeVar, Union, cast

__all__ = (
    'Recursion',
    'tail_recursive',
)

# Type variables for generic typing
VT = TypeVar('VT')
KT = TypeVar('KT')
RT = TypeVar('RT')

# TailRecFunction type alias
TailRecFunction = Callable[..., RT]

class NotSet(type):
    """A placeholder indicating an unset value distinct from None."""
    pass

class Recursion(BaseException, Generic[VT, KT]):
    """Exception used to indicate a tail-recursive call."""
    def __init__(self, *args: VT, **kwargs: KT) -> None:
        super().__init__()
        self.args = args
        self.kwargs = kwargs

def virtual_recurse(func: TailRecFunction, *args: VT, **kwargs: KT) -> RT:
    """
    Recursively calls a function with the arguments passed via Recursion exception
    until a non-recursive result is obtained.

    Args:
        func: The function to call recursively.
        args: Positional arguments for the function.
        kwargs: Keyword arguments for the function.

    Returns:
        The final result after resolving all recursive calls.
    """
    result: Union[Type[NotSet], RT] = NotSet
    while result is NotSet:
        try:
            result = func(*args, **kwargs)
        except Recursion as next_call:
            args = cast(Tuple[VT, ...], next_call.args)
            kwargs = cast(Dict[str, KT], next_call.kwargs)
    return cast(RT, result)

def tail_recursive(func: TailRecFunction) -> TailRecFunction:
    """
    Decorator to mark a function as tail-recursive, allowing the use of `raise Recursion`.

    Args:
        func: The function to decorate.

    Returns:
        A wrapped function that supports tail recursion.
    """
    @wraps(func)
    def _trampolined(*args, **kwargs):
        return virtual_recurse(func, *args, **kwargs)
    return _trampolined

# Example usage:
@tail_recursive
def factorial(n: int, acc: int = 1) -> int:
    """
    Compute the factorial of a number using tail recursion.

    Args:
        n: The number to compute the factorial of.
        acc: The accumulator for the factorial result.

    Returns:
        The factorial of n.
    """
    if n == 0:
        return acc
    else:
        raise Recursion(n - 1, acc * n)

@tail_recursive
def fibonacci(n: int, a: int = 0, b: int = 1) -> int:
    """
    Compute the nth Fibonacci number using tail recursion.

    Args:
        n: The position in the Fibonacci sequence.
        a: The first number in the Fibonacci sequence.
        b: The second number in the Fibonacci sequence.

    Returns:
        The nth Fibonacci number.
    """
    if n == 0:
        return a
    elif n == 1:
        return b
    else:
        raise Recursion(n - 1, b, a + b)

if __name__ == "__main__":
    print(factorial(10))  # Should print 3628800
    print(fibonacci(10))  # Should print 55
