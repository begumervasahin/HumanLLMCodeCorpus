from time import time

def lcs(X, Y):
    m, n = len(X), len(Y)
    L = [[0] * (n + 1) for _ in range(m + 1)]

    for i in range(m + 1):
        for j in range(n + 1):
            if i == 0 or j == 0:
                L[i][j] = 0
            elif X[i - 1] == Y[j - 1]:
                L[i][j] = L[i - 1][j - 1] + 1
            else:
                L[i][j] = max(L[i - 1][j], L[i][j - 1])

    # Display L table
    print("\nL table")
    for row in L:
        print("\t".join(map(str, row)))

    index = L[m][n]
    lcs_result = [""] * (index + 1)
    lcs_result[index] = ""

    i, j = m, n
    while i > 0 and j > 0:
        if X[i - 1] == Y[j - 1]:
            lcs_result[index - 1] = X[i - 1]
            i -= 1
            j -= 1
            index -= 1
        elif L[i - 1][j] >= L[i][j - 1]:
            i -= 1
        else:
            j -= 1

    lcs_str = "".join(lcs_result)
    print(f"\nLength of LCS is {len(lcs_str)}")
    print(f"LCS of {X} and {Y} is {lcs_str}")

# Initialize lists and time measurement variables
el_time = []
mn = []

X = ["AGGTAB", "ABRAC", "BACDB", "AYUSH", "KAMLESH", "SUNIL", "BIBASH", "ARAJU", "MANASI", "DEEPESH"]
Y = ["GXTXAYB", "YABBAD", "BDCB", "SHAHA", "MAHES", "UNATTI", "SHOWIN", "ARUNADHA", "ANSI", "DISH"]

# Compute LCS for each pair and measure execution time
for x_str, y_str in zip(X, Y):
    m, n = len(x_str), len(y_str)
    start_time = time()
    lcs(x_str, y_str)
    end_time = time()
    mn.append(m * n)
    el_time.append(end_time - start_time)

# Print the mn and elapsed time results
print("\nmn\tTime")
for m_n, elapsed in zip(mn, el_time):
    print(f"{m_n}\t{elapsed}")
