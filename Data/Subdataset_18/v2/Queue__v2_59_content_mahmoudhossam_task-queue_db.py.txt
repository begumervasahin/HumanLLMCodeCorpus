import os
from sqlalchemy import create_engine, Column, String, Integer
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker
from sqlalchemy_utils import database_exists, create_database

# Define the base class for declarative models
Base = declarative_base()

class User(Base):
    """
    Represents the User table in the database.
    """
    __tablename__ = "users"
    id = Column(Integer, primary_key=True)
    name = Column(String(50))
    email = Column(String(50), unique=True)

class Database:
    """
    Handles the database connection and operations.
    """
    def __init__(self):
        """
        Initializes the database connection and sets up the session.
        """
        database_url = os.getenv("DATABASE_URL")
        self.engine = create_engine(database_url)
        Session = sessionmaker(bind=self.engine)
        self.session = Session()

    def add_user(self, user):
        """
        Performs an upsert operation: updates if the email exists, inserts otherwise.
        
        Args:
            user (User): The user object to be added or updated.
        """
        existing_user = self.session.query(User).filter(User.email == user.email).first()
        if existing_user:
            existing_user.name = user.name
            self.session.add(existing_user)
        else:
            self.session.add(user)
        self.session.commit()

def setup_database():
    """
    Creates the database if it does not exist and sets up the users table.
    """
    db = Database()
    if not database_exists(db.engine.url):
        create_database(db.engine.url)
    Base.metadata.create_all(db.engine)

if __name__ == "__main__":
    setup_database()
