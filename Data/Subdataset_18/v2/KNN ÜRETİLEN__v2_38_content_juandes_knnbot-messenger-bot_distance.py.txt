import math

def euclidean_distance(point1, point2):
    """
    Calculate the Euclidean distance between two data points.
    
    Parameters:
    - point1 (list): The first data point.
    - point2 (list): The second data point.
    
    Returns:
    - float: The Euclidean distance between the two data points.
    """
    squared_difference = [(a - b) ** 2 for a, b in zip(point1, point2)]
    return math.sqrt(sum(squared_difference))

# Example usage:
if __name__ == "__main__":
    point1 = [1, 2, 3]
    point2 = [4, 5, 6]
    distance = euclidean_distance(point1, point2)
    print(f"The Euclidean distance between {point1} and {point2} is {distance:.2f}")
