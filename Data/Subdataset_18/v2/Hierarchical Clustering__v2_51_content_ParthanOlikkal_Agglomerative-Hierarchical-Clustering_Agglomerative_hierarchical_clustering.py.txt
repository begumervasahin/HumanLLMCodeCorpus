import numpy as np
import pandas as pd
from scipy.spatial import distance_matrix
from scipy.cluster import hierarchy
from matplotlib import pyplot as plt
from sklearn.cluster import AgglomerativeClustering
from sklearn.datasets import make_blobs

def generate_data(n_samples, centers, cluster_std):
    """Generate synthetic data using make_blobs."""
    X, y = make_blobs(n_samples=n_samples, centers=centers, cluster_std=cluster_std)
    return X, y

def plot_data(X, title, xlabel, ylabel):
    """Plot the data points."""
    plt.scatter(X[:, 0], X[:, 1], marker='o')
    plt.title(title)
    plt.xlabel(xlabel)
    plt.ylabel(ylabel)
    plt.show()

def apply_agglomerative_clustering(X, n_clusters, linkage_method):
    """Apply Agglomerative Clustering to the data."""
    agglom = AgglomerativeClustering(n_clusters=n_clusters, linkage=linkage_method)
    agglom.fit(X)
    return agglom

def plot_clustered_data(X, y, labels, title):
    """Plot the clustered data with labels."""
    plt.figure(figsize=(6, 4))
    x_min, x_max = np.min(X, axis=0), np.max(X, axis=0)
    X_normalized = (X - x_min) / (x_max - x_min)

    for i in range(X_normalized.shape[0]):
        plt.text(X_normalized[i, 0], X_normalized[i, 1], str(y[i]),
                 color=plt.cm.nipy_spectral(labels[i] / 10.),
                 fontdict={'weight': 'bold', 'size': 9})

    plt.xticks([])
    plt.yticks([])
    plt.scatter(X_normalized[:, 0], X_normalized[:, 1], marker='.')
    plt.title(title)
    plt.show()

def compute_distance_matrix(X):
    """Compute the distance matrix for the data."""
    return distance_matrix(X, X)

def plot_dendrogram(Z, title, xlabel, ylabel):
    """Plot a dendrogram for hierarchical clustering."""
    plt.figure(figsize=(10, 7))
    hierarchy.dendrogram(Z)
    plt.title(title)
    plt.xlabel(xlabel)
    plt.ylabel(ylabel)
    plt.show()

def main():
    """Main function to execute the clustering and visualization."""
    # Generate synthetic data
    X1, y1 = generate_data(n_samples=50, centers=[[4, 4], [-2, -1], [1, 1], [10, 4]], cluster_std=0.9)

    # Plot the generated data points
    plot_data(X1, title='Generated Data Points', xlabel='Feature 1', ylabel='Feature 2')

    # Apply Agglomerative Clustering
    agglom = apply_agglomerative_clustering(X1, n_clusters=4, linkage_method='average')

    # Plot the clustered data
    plot_clustered_data(X1, y1, agglom.labels_, title='Agglomerative Clustering')

    # Compute the distance matrix
    dist_matrix = compute_distance_matrix(X1)
    print("Distance Matrix:\n", dist_matrix)

    # Perform hierarchical clustering and plot the dendrogram
    Z = hierarchy.linkage(dist_matrix, 'complete')
    plot_dendrogram(Z, title='Dendrogram', xlabel='Sample Index', ylabel='Distance')

if __name__ == "__main__":
    main()
