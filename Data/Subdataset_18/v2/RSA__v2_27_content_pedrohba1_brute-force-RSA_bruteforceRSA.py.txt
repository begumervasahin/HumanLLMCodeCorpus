import math
import time

def nearest_odd_under(number):
    """
    Calculate the nearest odd number below the given number.
    
    Parameters:
    number (int): The number to find the nearest odd number below.
    
    Returns:
    int: The nearest odd number below the given number.
    """
    return number - 1 if number % 2 == 0 else number - 2

def egcd(a, b):
    """
    Extended Euclidean Algorithm to find the greatest common divisor.
    
    Parameters:
    a, b (int): Two integers to find the GCD of.
    
    Returns:
    tuple: A tuple containing the GCD and the coefficients.
    """
    if a == 0:
        return (b, 0, 1)
    else:
        g, y, x = egcd(b % a, a)
        return (g, x - (b // a) * y, y)

def modinv(a, m):
    """
    Find the modular inverse of a modulo m.
    
    Parameters:
    a, m (int): The integer and the modulo.
    
    Returns:
    int: The modular inverse of a modulo m.
    
    Raises:
    Exception: If the modular inverse does not exist.
    """
    g, x, y = egcd(a, m)
    if g != 1:
        raise Exception('Modular inverse does not exist')
    return x % m

def brute_rsa(n, e):
    """
    Brute-force RSA decryption to find the private key.
    
    Parameters:
    n, e (int): The modulus and the public exponent.
    
    Returns:
    tuple: The private key and the time taken to find it.
    
    Raises:
    Exception: If factorization or modular inverse calculation fails.
    """
    start_time = time.time()
    c = nearest_odd_under(math.isqrt(n))

    # Discovering p by brute force
    for i in range(c, 1, -2):
        if n % i == 0:
            p = i
            break

    # Discovering q
    q = n // p

    # Checking if the calculation is correct
    if n != p * q:
        raise Exception('Factorization failed')

    # Calculating d
    phin = (p - 1) * (q - 1)
    d = modinv(e, phin)
    if d * e % phin != 1:
        raise Exception('Modular inverse calculation failed')

    total_time = time.time() - start_time
    return d, total_time

# List of RSA keys (n, e)
keys = [
    (1325147, 79), (13339787, 351047), (87411743, 11), (153988391, 365),
    (642281891, 80105), (3662937263, 80273), (2461987247, 78703),
    (10988963221, 118297), (9979645019, 9764819), (36207914857, 89)
]

total_time = 0
for n, e in keys:
    d, elapsed_time = brute_rsa(n, e)
    print(f'Key pair (n, e): ({n}, {e})')
    print(f'Private key d: {d}')
    print(f'Time taken: {elapsed_time:.6f} seconds')
    total_time += elapsed_time

print(f'Total time: {total_time:.6f} seconds')
