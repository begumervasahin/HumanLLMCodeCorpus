import numpy as np
import sys

def dp_lcs(x, y, reconstruct=False):
    """
    Computes the Longest Common Subsequence (LCS) between two strings using dynamic programming.

    Parameters:
        x (str): The first string.
        y (str): The second string.
        reconstruct (bool): Whether to reconstruct the LCS string. Default is False.

    Returns:
        int or str: The length of the LCS if reconstruct is False, otherwise the LCS string.
    """
    m, n = len(x), len(y)
    if m == 0 or n == 0:
        return 0

    # Create a matrix to store lengths of LCS
    l = np.zeros((m + 1, n + 1), dtype=int)

    # Optional matrix to store directions for reconstructing the LCS
    if reconstruct:
        b = np.empty((m + 1, n + 1), dtype=str)

    # Build the l matrix in bottom-up fashion
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if x[i - 1] == y[j - 1]:
                l[i, j] = l[i - 1, j - 1] + 1
                if reconstruct:
                    b[i, j] = "d"  # diagonal
            elif l[i - 1, j] >= l[i, j - 1]:
                l[i, j] = l[i - 1, j]
                if reconstruct:
                    b[i, j] = "u"  # up
            else:
                l[i, j] = l[i, j - 1]
                if reconstruct:
                    b[i, j] = "l"  # left

    if reconstruct:
        # Reconstruct the LCS from the b matrix
        lcs = []
        i, j = m, n
        while i > 0 and j > 0:
            if b[i, j] == "d":
                lcs.append(x[i - 1])
                i -= 1
                j -= 1
            elif b[i, j] == "u":
                i -= 1
            else:
                j -= 1
        return ''.join(reversed(lcs))
    else:
        return l[m, n]

def test(x, y, reconstruct):
    """
    Tests the dp_lcs function.

    Parameters:
        x (str): The first string.
        y (str): The second string.
        reconstruct (bool): Whether to reconstruct the LCS string.

    Returns:
        int or str: The length of the LCS or the LCS string if reconstruct is True.
    """
    return dp_lcs(x, y, reconstruct)

if __name__ == "__main__":
    if len(sys.argv) < 2:
        print(f"Usage: python {__file__} <input_file> [0/1]")
        print("<input_file> - file containing two sequences")
        print("0 - without reconstruction, 1 - with reconstruction")
    else:
        input_file = sys.argv[1]
        with open(input_file, 'r') as f:
            x = f.readline().strip()
            y = f.readline().strip()
            reconstruct = len(sys.argv) == 3 and int(sys.argv[2]) == 1
            if reconstruct:
                lcs = test(x, y, reconstruct)
                print(f"LCS length: {len(lcs)}")
                print(f"LCS: {lcs}")
            else:
                print(f"LCS length: {test(x, y, reconstruct)}")
