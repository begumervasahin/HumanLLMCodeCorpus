import unittest
from binarytreevis import BinarySearchTree

class GivenAsg3Tests(unittest.TestCase):
    """
    Unit tests provided for Assignment 3 to verify the functionality of the BinarySearchTree class.
    """

    def test_binsearchtree_constructor1(self):
        """
        Test constructor with an initial list of contents and verify the tree structure using various traversal methods.
        """
        bt1 = BinarySearchTree(contents=[0, 1, 2, 5, 90, -1])
        bt2 = BinarySearchTree(contents=[0, 1, 2, 5, 90, -1])
        self.assertListEqual(bt1.inorder(), bt2.inorder())
        self.assertListEqual(bt1.preorder(), bt2.preorder())
        self.assertListEqual(bt1.postorder(), bt2.postorder())
        self.assertListEqual(bt1.levelorder(), bt2.levelorder())

    def test_binsearchtree_inorder1(self):
        """
        Test inorder traversal on a BinarySearchTree with specific contents.
        """
        bt = BinarySearchTree(contents=[0, -1, 1])
        self.assertListEqual(bt.inorder(), [-1, 0, 1])

    def test_binsearchtree_preorder1(self):
        """
        Test preorder traversal on a BinarySearchTree with specific contents.
        """
        bt = BinarySearchTree(contents=[2, 0, 1])
        self.assertListEqual(bt.preorder(), [2, 0, 1])

    def test_binsearchtree_postorder1(self):
        """
        Test postorder traversal on a BinarySearchTree with specific contents.
        """
        bt = BinarySearchTree(contents=[2, 0, 1])
        self.assertListEqual(bt.postorder(), [1, 0, 2])

    def test_binsearchtree_levelorder1(self):
        """
        Test level order traversal on a BinarySearchTree with specific contents.
        """
        bt = BinarySearchTree(contents=[2, 0, 1, 6, 10])
        self.assertListEqual(bt.levelorder(), [2, 0, 6, 1, 10])

    def test_binsearchtree_insert1(self):
        """
        Test insertion of elements into a BinarySearchTree and verify with inorder traversal.
        """
        bt = BinarySearchTree()
        bt.insert(0)
        bt.insert(-1)
        bt.insert(1)
        self.assertListEqual(bt.inorder(), [-1, 0, 1])

    def test_binsearchtree_delete1(self):
        """
        Test deletion of an element from a BinarySearchTree and verify with inorder traversal.
        """
        bt = BinarySearchTree(contents=[0, 1, 2])
        bt.delete(0)
        self.assertListEqual(bt.inorder(), [1, 2])

    def test_binsearchtree_contains1(self):
        """
        Test the `in` operator for checking element existence in a BinarySearchTree.
        """
        bt = BinarySearchTree(contents=[0, 1, 2])
        self.assertTrue(0 in bt)
        bt.delete(0)
        self.assertFalse(0 in bt)
