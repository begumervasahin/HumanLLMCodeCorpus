import string

def decrypt_caesar_cipher(text, shift):
    """
    Decrypts a Caesar cipher encrypted text by shifting the characters back.

    Args:
        text (str): The text encrypted with a Caesar cipher.
        shift (int): The number of positions to shift characters back.

    Returns:
        str: The decrypted text.
    """
    decrypted_text = []
    
    for char in text:
        if char in string.ascii_uppercase:
            # Compute the decrypted character
            shifted = (ord(char) - shift - 65) % 26 + 65
            decrypted_text.append(chr(shifted))
        else:
            # Keep non-alphabetic characters unchanged
            decrypted_text.append(char)
    
    return ''.join(decrypted_text)

def main():
    encrypted_message = 'YMJJDJXHFSRNXQJFIYMJXRNQJHFSQNJGZYYMJXMTJXYJQQYMJYWZYM'
    print('Encrypted message:', encrypted_message)

    # Try all possible shifts from 1 to 25
    for shift in range(1, 26):
        decrypted_message = decrypt_caesar_cipher(encrypted_message, shift)
        print(f'Shift {shift}: {decrypted_message}')

if __name__ == "__main__":
    main()
