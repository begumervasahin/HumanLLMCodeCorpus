def search(matrix, i, j, rows, cols):
    """
    Recursive approach to mark the visited parts of the island.
    
    Args:
        matrix (list of list of int): The binary matrix.
        i (int): The current row index.
        j (int): The current column index.
        rows (int): The total number of rows in the matrix.
        cols (int): The total number of columns in the matrix.
    """
    if 0 <= i < rows and 0 <= j < cols and matrix[i][j] == 1:
        matrix[i][j] = 0  # Mark the cell as visited
        search(matrix, i + 1, j, rows, cols)
        search(matrix, i - 1, j, rows, cols)
        search(matrix, i, j + 1, rows, cols)
        search(matrix, i, j - 1, rows, cols)

def get_number_of_islands_recursive(matrix):
    """
    Counts the number of islands using the recursive approach.
    
    Args:
        matrix (list of list of int): The binary matrix.
    
    Returns:
        int: The number of islands in the matrix.
    """
    rows, cols = len(matrix), len(matrix[0])
    island_count = 0
    for i in range(rows):
        for j in range(cols):
            if matrix[i][j] == 1:
                search(matrix, i, j, rows, cols)
                island_count += 1
    return island_count

def dfs_helper(stack, matrix, i, j, rows, cols):
    """
    Helper function to add valid neighbors to the stack.
    
    Args:
        stack (list of tuple): The stack used in DFS.
        matrix (list of list of int): The binary matrix.
        i (int): The current row index.
        j (int): The current column index.
        rows (int): The total number of rows in the matrix.
        cols (int): The total number of columns in the matrix.
    """
    if 0 <= i < rows and 0 <= j < cols and matrix[i][j] == 1:
        stack.append((i, j))

def depth_first_search(matrix, i, j, rows, cols):
    """
    Iterative approach to mark the visited parts of the island using DFS.
    
    Args:
        matrix (list of list of int): The binary matrix.
        i (int): The starting row index.
        j (int): The starting column index.
        rows (int): The total number of rows in the matrix.
        cols (int): The total number of columns in the matrix.
    """
    stack = [(i, j)]
    while stack:
        i, j = stack.pop()
        matrix[i][j] = 0  # Mark the cell as visited
        dfs_helper(stack, matrix, i + 1, j, rows, cols)
        dfs_helper(stack, matrix, i - 1, j, rows, cols)
        dfs_helper(stack, matrix, i, j + 1, rows, cols)
        dfs_helper(stack, matrix, i, j - 1, rows, cols)

def get_number_of_islands_iterative(matrix):
    """
    Counts the number of islands using the iterative approach.
    
    Args:
        matrix (list of list of int): The binary matrix.
    
    Returns:
        int: The number of islands in the matrix.
    """
    rows, cols = len(matrix), len(matrix[0])
    island_count = 0
    for i in range(rows):
        for j in range(cols):
            if matrix[i][j] == 1:
                island_count += 1
                depth_first_search(matrix, i, j, rows, cols)
    return island_count

def print_matrix(matrix):
    """Print the matrix in a readable format."""
    for row in matrix:
        print(" ".join(str(cell) for cell in row))
    print()
