import os
import collections
import pandas as pd
import csv
import re

DATA_DIR = "data"
MBTI_UNCLEAN_CSV_PATH = os.path.join(DATA_DIR, "mbti_unclean.csv")
DIMENSIONS = ("IE", "NS", "TF", "PJ")

# Read the CSV file into a DataFrame
df = pd.read_csv(MBTI_UNCLEAN_CSV_PATH)

# Initialize a dictionary to count the occurrences of each letter
counts = collections.defaultdict(int)

# Count the occurrences of each letter in DIMENSIONS
for dimension in DIMENSIONS:
    letter_1, letter_2 = dimension
    for index, row in df.iterrows():
        mbti_type = row["type"]
        posts = row["posts"].split("|||")
        for post in posts:
            if ("http" in post) or (post.strip() == "") or (not re.search("[a-zA-Z]", post)):
                continue
            if letter_1 in mbti_type:
                counts[letter_1] += 1
            if letter_2 in mbti_type:
                counts[letter_2] += 1

# Create balanced datasets for each letter in DIMENSIONS
for dimension in DIMENSIONS:
    letter_1, letter_2 = dimension
    limit = min(counts[letter_1], counts[letter_2])

    for letter in [letter_1, letter_2]:
        collected_posts = []
        post_count = 0
        for index, row in df.iterrows():
            if letter in row["type"]:
                posts = row["posts"].split("|||")
                for post in posts:
                    if post_count >= limit:
                        break
                    if ("http" in post) or (post.strip() == "") or (not re.search("[a-zA-Z]", post)):
                        continue
                    collected_posts.append(post)
                    post_count += 1

        # Write the collected posts to a CSV file
        train_csv_path = os.path.join(DATA_DIR, f"train_{letter}.csv")
        with open(train_csv_path, "w", newline='', encoding='utf-8') as csv_file:
            writer = csv.writer(csv_file)
            for post in collected_posts:
                writer.writerow([post])
