import random

# Initialize a global counter to keep track of comparisons made during sorting
comparison_counter = 0

def merge_sort(lst):
    """
    Recursively performs merge sort on the provided list, while counting the number of comparisons.
    
    Parameters:
    lst (list): The list of elements to sort.

    Returns:
    list: A sorted version of the input list.
    """
    global comparison_counter

    # Base case: if the list is of length 0 or 1, it is already sorted
    if len(lst) <= 1:
        comparison_counter += 1
        return lst

    # Split the list into two halves
    mid = len(lst) // 2
    left = merge_sort(lst[:mid])
    right = merge_sort(lst[mid:])

    # Merge the sorted halves and return
    return merge(left, right)

def merge(left, right):
    """
    Merges two sorted lists into one sorted list while counting comparisons.
    
    Parameters:
    left (list): The left half of the list.
    right (list): The right half of the list.

    Returns:
    list: A merged and sorted list from the two input halves.
    """
    global comparison_counter
    merged = []
    i = 0
    j = 0

    # Compare elements from both lists and merge them in sorted order
    while i < len(left) and j < len(right):
        if left[i] < right[j]:
            comparison_counter += 1
            merged.append(left[i])
            i += 1
        else:
            merged.append(right[j])
            j += 1

    # Add remaining elements from either list
    comparison_counter += 1
    merged += left[i:]
    merged += right[j:]

    return merged

def test_merge_sort_on_words(lines, num_words):
    """
    Tests merge sort on a subset of words with various configurations:
    sorted, reversed, and randomly shuffled.
    
    Parameters:
    lines (list): The list of words to test.
    num_words (int): The number of words to use for the test.
    """
    global comparison_counter
    print(f'Testing on {num_words} words')
    
    # Test on sorted words
    print(f'{num_words} words sorted')
    words_sorted = lines[:num_words]
    merge_sort(words_sorted)
    print('Number of comparisons:', comparison_counter)
    
    # Test on reversed words
    print(f'{num_words} words reversed')
    comparison_counter = 0
    words_reversed = list(reversed(words_sorted))
    merge_sort(words_reversed)
    print('Number of comparisons:', comparison_counter)
    
    # Test on randomly shuffled words
    print(f'{num_words} words random')
    comparison_counter = 0
    random_start_index = random.randint(0, len(lines) - num_words)
    words_random = lines[random_start_index:random_start_index + num_words]
    random.shuffle(words_random)
    merge_sort(words_random)
    print('Number of comparisons:', comparison_counter)
    print('-------------')

def main():
    """
    Main function that reads words from a file and tests merge sort on varying sizes of word subsets.
    """
    # Read the list of words from the file
    with open('words.txt') as file:
        lines = [line.strip() for line in file]

    print(f'Total words: {len(lines)}')
    
    # Define different test cases based on word count
    test_cases = [10, 30, 100, 300, 1000, 3000, 10000, len(lines)]

    # Perform merge sort testing for each case
    for num_words in test_cases:
        test_merge_sort_on_words(lines, num_words)

if __name__ == "__main__":
    main()
