from graph import Graph

def dfs_aux(graph, vertex, vertices_marked, mark, dfs_order):
    """
    Auxiliary function for Depth First Search (DFS).
    
    Args:
        graph (Graph): The graph object.
        vertex: The current vertex.
        vertices_marked (dict): A dictionary to keep track of marked vertices.
        mark: The mark used to indicate the vertex has been visited.
        dfs_order (list): The list to store the DFS order of vertices.
    """
    if vertices_marked[vertex] is not None:
        return
    
    vertices_marked[vertex] = mark
    dfs_order.append(vertex)
    
    for neighbor in graph.get_neighbours(vertex):
        dfs_aux(graph, neighbor, vertices_marked, mark, dfs_order)

def dfs(graph):
    """
    Perform Depth First Search (DFS) on the graph.
    
    Args:
        graph (Graph): The graph object.
    
    Returns:
        list: The DFS order of vertices and the dictionary of marked vertices.
    """
    dfs_order = []
    vertices = graph.get_vertices()
    vertices_marked = {vertex: None for vertex in vertices}
    
    for vertex in vertices:
        if vertices_marked[vertex] is None:
            dfs_aux(graph, vertex, vertices_marked, vertex, dfs_order)
    
    return [dfs_order, vertices_marked]
