import time
import random

def merge_sort(arr):
    """Recursively sorts an array using the merge sort algorithm."""
    if len(arr) > 1:
        # Find the midpoint and split the array into two halves
        mid = len(arr) // 2
        left_half = arr[:mid]
        right_half = arr[mid:]

        # Recursively sort both halves
        merge_sort(left_half)
        merge_sort(right_half)

        # Merge the sorted halves
        i = j = k = 0

        while i < len(left_half) and j < len(right_half):
            if left_half[i] < right_half[j]:
                arr[k] = left_half[i]
                i += 1
            else:
                arr[k] = right_half[j]
                j += 1
            k += 1

        # Copy any remaining elements from the left half
        while i < len(left_half):
            arr[k] = left_half[i]
            i += 1
            k += 1

        # Copy any remaining elements from the right half
        while j < len(right_half):
            arr[k] = right_half[j]
            j += 1
            k += 1

def insertion_sort(arr):
    """Sorts an array in place using the insertion sort algorithm."""
    for i in range(1, len(arr)):
        current_value = arr[i]
        pos = i

        # Shift elements of the sorted segment to the right to make space
        while pos > 0 and arr[pos - 1] > current_value:
            arr[pos] = arr[pos - 1]
            pos -= 1

        # Place the current value into its correct position
        arr[pos] = current_value

def run_sorting_tests(array_sizes):
    """Runs timing tests on merge sort and insertion sort for various array sizes."""
    for size in array_sizes:
        # Generate a random array of the specified size
        arr = [random.randint(1, 1000) for _ in range(size)]

        # Time the merge sort algorithm
        start_time = time.time()
        merge_sort(arr.copy())
        merge_time = time.time() - start_time

        # Time the insertion sort algorithm
        start_time = time.time()
        insertion_sort(arr.copy())
        insertion_time = time.time() - start_time

        # Output the results
        print(f"Array Size: {size}")
        print(f"Merge Sort Time: {merge_time:.6f} seconds")
        print(f"Insertion Sort Time: {insertion_time:.6f} seconds")
        print("-" * 40)

# Define the array sizes to test
array_sizes = [2000, 8000, 32000, 128000, 512000, 1024000, 4096000]

# Run the tests
run_sorting_tests(array_sizes)
