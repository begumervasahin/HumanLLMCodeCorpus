from searcher import Searcher

class BFS(Searcher):
    """
    Breadth-First Search (BFS) class extending Searcher.
    """
    name = '2.A_bfs.txt'
    
    def merge_list(self, list1, list2):
        """
        Merge two lists into one.

        Args:
            list1 (list): The first list to merge.
            list2 (list): The second list to merge.

        Returns:
            list: A new list containing elements from both input lists.
        """
        return list1 + list2

# Example usage
if __name__ == "__main__":
    # Create an instance of BFS
    bfs = BFS()
    
    # Define two lists to merge
    list1 = [1, 2, 3]
    list2 = [4, 5, 6]
    
    # Merge the lists using the BFS instance
    merged_list = bfs.merge_list(list1, list2)
    
    # Print the results
    print("Merged List:", merged_list)
    print("BFS Name:", bfs.name)
