def encrypt(plain_text, shift):
    """
    Encrypts the given plain text using a Caesar cipher with the specified shift.

    Parameters
    ----------
    plain_text : str
        The text to be encrypted.
    shift : int
        The number of positions to shift each character in the plain text.

    Returns
    -------
    str
        The encrypted text.
    """
    encrypted_text = []

    for char in plain_text:
        if char.isupper():
            # Encrypt uppercase characters
            new_char = chr((ord(char) + shift - ord('A')) % 26 + ord('A'))
        elif char.islower():
            # Encrypt lowercase characters
            new_char = chr((ord(char) + shift - ord('a')) % 26 + ord('a'))
        else:
            # Leave non-alphabetic characters unchanged
            new_char = char

        encrypted_text.append(new_char)

    return ''.join(encrypted_text)

if __name__ == "__main__":
    # Example usage
    plain_text = "Vineet"
    shift = 4
    encrypted_text = encrypt(plain_text, shift)
    print("ENCRYPTED TEXT:", encrypted_text)
