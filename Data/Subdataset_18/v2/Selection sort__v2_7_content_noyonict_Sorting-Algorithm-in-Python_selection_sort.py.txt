def selection_sort(arr):
    """
    Perform selection sort on the given array to sort it in ascending order.
    
    Parameters:
    - arr: The list of elements to be sorted.
    """
    n = len(arr)
    for i in range(n - 1):
        min_index = i
        # Find the minimum element in the unsorted portion of the array
        for j in range(i + 1, n):
            if arr[j] < arr[min_index]:
                min_index = j
        # Swap the found minimum element with the first element of the unsorted portion
        if min_index != i:
            arr[i], arr[min_index] = arr[min_index], arr[i]

def main():
    """
    Main function to demonstrate the selection_sort function.
    """
    array = [12, 3, 43, 43, 9, 3, 2, 2, 43, 23, 53, 3, 3, 23, 23, 12, 53, 91, 1]
    
    print("Array before sorting:")
    print(array)
    
    selection_sort(array)
    
    print("Array after sorting:")
    print(array)

if __name__ == "__main__":
    main()
