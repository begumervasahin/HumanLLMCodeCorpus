def print_shanks_algorithm():
    """
    Prints Shank's Algorithm tables and finds matching values between S and T tables.
    """
    # Print the header for the algorithm
    print("Shank's Algorithm!\n")
    
    # Print the S table
    print("S table:\n")
    s_values = {}  # Dictionary to store S table values
    for x in range(46):
        # Compute the value for the S table
        val = (5**(45 * x)) % 2017
        s_values[val] = x  # Store value with its index
        print(f"{x} {val}")
    
    # Print the T table
    print("\nT table:\n")
    for y in range(46):
        # Compute the value for the T table
        val = (1736 * (5**y)) % 2017
        print(f"{y}  {val}")
    
    # Find and print matching values between S and T tables
    print("\nMatching Values:\n")
    for y in range(46):
        # Compute the value for the T table
        y_val = (1736 * (5**y)) % 2017
        if y_val in s_values:
            # Print matching x and y values
            print(f"{s_values[y_val]} {y}")

# Execute the function
print_shanks_algorithm()
