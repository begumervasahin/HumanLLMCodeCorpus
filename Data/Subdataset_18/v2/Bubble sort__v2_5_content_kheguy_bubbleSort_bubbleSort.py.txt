import random

class BubbleSort:
    def __init__(self, size_of_array, interval_from, interval_to):
        """
        Initialize the BubbleSort object with a randomly generated array.
        
        Args:
            size_of_array (int): The size of the array to be sorted.
            interval_from (int): The lower bound of the random number interval.
            interval_to (int): The upper bound of the random number interval.
        """
        self.array_size = size_of_array
        self.array = [random.randint(interval_from, interval_to) for _ in range(size_of_array)]
    
    def __del__(self):
        """
        Clear the array when the object is deleted.
        """
        self.array.clear()
    
    def print_array(self):
        """
        Print the array.
        """
        print(self.array)
    
    def sort(self):
        """
        Sort the array using the Bubble Sort algorithm.
        """
        is_sorted = False
        while not is_sorted:
            is_sorted = True
            for i in range(self.array_size - 1):
                if self.array[i + 1] < self.array[i]:
                    self.array[i], self.array[i + 1] = self.array[i + 1], self.array[i]
                    is_sorted = False

def main():
    """
    Main function to demonstrate the Bubble Sort algorithm.
    """
    while True:
        try:
            n = int(input('Enter size of array:\n'))
            a = int(input('Enter primary position for random:\n'))
            b = int(input('Enter final position for random:\n'))
            
            if a > b:
                print("Invalid input: the primary position should be less than or equal to the final position.")
                continue
            
            sorter = BubbleSort(n, a, b)
            print("Original array:")
            sorter.print_array()
            
            sorter.sort()
            print("Sorted array:")
            sorter.print_array()
            
            del sorter
        except ValueError:
            print("Invalid input. Please enter valid integers.")

if __name__ == "__main__":
    main()
