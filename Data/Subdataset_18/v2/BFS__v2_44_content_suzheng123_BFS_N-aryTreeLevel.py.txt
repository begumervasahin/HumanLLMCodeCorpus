class Node:
    """
    A class to represent a node in an N-ary tree.
    
    Attributes:
        val (int): The value of the node.
        children (list): A list of child nodes.
    """
    def __init__(self, val, children=None):
        self.val = val
        self.children = children if children is not None else []

class Solution:
    def levelOrder(self, root):
        """
        Perform a level order traversal of an N-ary tree.

        Args:
            root (Node): The root node of the N-ary tree.
        
        Returns:
            List[List[int]]: A list of lists, where each inner list contains the values of the nodes at that level.
        """
        if not root:
            return []

        queue = [root]  # Initialize the queue with the root node
        result = []     # This will store the final level order traversal

        while queue:
            level = []         # List to store the current level's values
            next_queue = []    # Temporary list to store the next level's nodes
            
            for node in queue:
                level.append(node.val)  # Add the current node's value to the level list
                next_queue.extend(node.children)  # Add all children to the next level queue
            
            result.append(level)  # Add the current level's values to the result
            queue = next_queue    # Move to the next level
        
        return result

# Example usage:
if __name__ == "__main__":
    # Creating an example N-ary tree:
    #            1
    #         /  |  \
    #        3   2   4
    #       / \
    #      5   6 
    node5 = Node(5)
    node6 = Node(6)
    node3 = Node(3, [node5, node6])
    node2 = Node(2)
    node4 = Node(4)
    root = Node(1, [node3, node2, node4])

    solution = Solution()
    result = solution.levelOrder(root)
    print(result)  # Output: [[1], [3, 2, 4], [5, 6]]
