import sys
from time import time

def main(begin, end, n_jobs=8):
    """
    Main function to perform a series of operations based on the provided range (begin to end)
    and the number of parallel jobs (n_jobs).
    
    Parameters:
    begin (int): The starting step of the operations.
    end (int): The ending step of the operations.
    n_jobs (int): The number of parallel jobs for the PCA calculation (default is 8).
    """
    t = time()  # Start timing

    # Step 0: Import and delete vox module if within range
    if begin <= 0 <= end:
        print('0: vox')
        import vox
        del vox

    # Step 1: Import and delete mn module if within range
    if begin <= 1 <= end:
        print('1: mn')
        import mn
        del mn

    # Step 2: Perform PCA calculation and save results if within range
    if begin <= 2 <= end:
        print('2: pcamn')
        from numpy import log2, array, load, save
        from sklearn.decomposition import PCA
        from joblib import Parallel, delayed
        from open3d import read_point_cloud

        def log_var(pts):
            """
            Calculate the log variance of the PCA components.
            
            Parameters:
            pts (ndarray): The input points for PCA.
            
            Returns:
            ndarray: The log variance of the PCA components.
            """
            pca = PCA()
            pca.fit(pts)
            pca.explained_variance_[:-1] -= pca.explained_variance_[1:]
            return log2(pca.explained_variance_)

        def export_pca(ply_vox, npy_knn, temp_out):
            """
            Perform PCA on the given point cloud data and save the results.
            
            Parameters:
            ply_vox (str): The path to the input PLY file.
            npy_knn (str): The path to the input NPY file.
            temp_out (str): The path to save the output NPY file.
            """
            vox = array(read_point_cloud(ply_vox).points)
            knn = load(npy_knn)
            save(temp_out, Parallel(n_jobs, verbose=1)(
                delayed(log_var)(vox[n]) for n in knn))

        # Export PCA results for specified files
        export_pca('vox1.ply', '128_1.npy', 'pca_1')
        export_pca('vox2.ply', '128_2.npy', 'pca_2')
        export_pca('vox4.ply', '128_4.npy', 'pca_4')

    # Step 3: Import and delete concat module if within range
    if begin <= 3 <= end:
        print('3: concat')
        import concat
        del concat

    # Step 4: Import and delete propagate module if within range
    if begin <= 4 <= end:
        print('4: propagate')
        import propagate
        del propagate

    # Step 5: Import and delete test module if within range
    if begin <= 5 <= end:
        print('5: test')
        import test
        del test

    # Step 6: Import and delete vis module if within range
    if begin <= 6 <= end:
        print('6: vis')
        import vis
        del vis

    # Step 7: Import and delete treetop module if within range
    if begin <= 7 <= end:
        print('7: treetop')
        import treetop
        del treetop

    # Write the total time taken to a file
    with open('time.txt', 'w') as w:
        w.write(f'{time() - t} sec\n')

if __name__ == "__main__":
    if len(sys.argv) < 3:
        print("Usage: python script.py <begin> <end> [n_jobs]")
        sys.exit(1)

    begin = int(sys.argv[1])
    end = int(sys.argv[2])
    n_jobs = int(sys.argv[3]) if len(sys.argv) > 3 else 8

    main(begin, end, n_jobs)
