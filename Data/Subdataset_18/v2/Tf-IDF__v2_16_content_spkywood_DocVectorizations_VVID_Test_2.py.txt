import codecs
import re

# Define the regex pattern to search for VVIDs
pattern = r'vvid=[A-Z0-9-]{36}'

# Open the log file and read its contents
with codecs.open('app.log', 'r', 'utf-8', 'ignore') as log_file:
    log_lines = log_file.readlines()

# Reverse the order of the log lines
reversed_log_lines = log_lines[::-1]

# Write the reversed log lines to a new file
with codecs.open('app_convert.log', 'w+', 'utf-8', 'ignore') as result_log:
    result_log.writelines(reversed_log_lines)

# Get the total number of lines in the reversed log
total_lines = len(reversed_log_lines)
print(f"All logfile lines: {total_lines}\n")

# Initialize lists to store VVIDs and their corresponding line numbers
vvid_list = []
vvid_line_numbers = []

# Search for the VVID pattern in each line and store matches
for i, line in enumerate(reversed_log_lines):
    match = re.search(pattern, line)
    if match:
        vvid_list.append(match.group())
        vvid_line_numbers.append(i)

# Get the total number of VVIDs found
total_vvids = len(vvid_line_numbers)
print(f"The number of all VVIDs: {total_vvids}\n")

# Print the line numbers and corresponding VVIDs
for line_number, vvid in zip(vvid_line_numbers, vvid_list):
    print(line_number)
    print(vvid)

# Find and print the line numbers where the VVID changes
cut_line_numbers = []
for i in range(1, total_vvids):
    if vvid_list[i] != vvid_list[i - 1]:
        cut_line_numbers.append(vvid_line_numbers[i - 1])

# Print the cut line numbers
print("Cut line numbers where VVID changes:")
for cut_line in cut_line_numbers:
    print(cut_line)
