import pandas as pd
import matplotlib.pyplot as plt
from sklearn.cluster import AgglomerativeClustering
import scipy.cluster.hierarchy as sch

def load_dataset(file_path):
    """Load the dataset from a CSV file."""
    return pd.read_csv(file_path)

def perform_clustering(X, n_clusters=4):
    """
    Apply various Agglomerative Clustering methods and return their predictions.

    Args:
        X (np.ndarray): The data to be clustered.
        n_clusters (int): The number of clusters to form. Default is 4.

    Returns:
        dict: A dictionary containing the clustering labels for each method.
    """
    clustering_methods = {
        'Ward': AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage='ward'),
        'Complete': AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage='complete'),
        'Average': AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage='average'),
        'Single': AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage='single')
    }

    return {method: model.fit_predict(X) for method, model in clustering_methods.items()}

def plot_clusters(X, predictions, n_clusters=4):
    """
    Plot clusters for each clustering method.

    Args:
        X (np.ndarray): The data points.
        predictions (dict): A dictionary containing the clustering labels for each method.
        n_clusters (int): The number of clusters. Default is 4.
    """
    colors = ['red', 'blue', 'green', 'yellow', 'purple', 'orange']
    
    for method, labels in predictions.items():
        plt.figure(figsize=(10, 7))
        for i in range(n_clusters):
            plt.scatter(X[labels == i, 0], X[labels == i, 1], s=100, c=colors[i], label=f'Cluster {i+1}')
        plt.title(f'{method} Method')
        plt.xlabel('Feature 1')
        plt.ylabel('Feature 2')
        plt.legend()
        plt.show()

def plot_dendrograms(X):
    """
    Plot dendrograms for different linkage methods.

    Args:
        X (np.ndarray): The data to be used for generating dendrograms.
    """
    methods = ['ward', 'complete', 'average', 'single']
    
    for method in methods:
        plt.figure(figsize=(10, 7))
        sch.dendrogram(sch.linkage(X, method=method))
        plt.title(f'{method.capitalize()} Dendrogram')
        plt.xlabel("Data Points")
        plt.ylabel("Euclidean Distance")
        plt.show()

def main():
    """Main function to execute the clustering analysis and plotting."""
    # Load the dataset
    dataset = load_dataset('customers_dataset.csv')
    
    # Extract relevant features for clustering
    X = dataset.iloc[:, 2:8].values
    
    # Perform clustering using various methods
    predictions = perform_clustering(X)
    
    # Plot the clusters for each method
    plot_clusters(X, predictions)
    
    # Plot the dendrograms for each linkage method
    plot_dendrograms(X)

if __name__ == "__main__":
    main()
