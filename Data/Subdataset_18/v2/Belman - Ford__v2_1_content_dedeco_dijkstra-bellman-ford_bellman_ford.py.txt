from utils import initialize_single_source, relax

class Graph:
    def __init__(self):
        """
        Initialize the graph with an empty set of vertices and edges.
        """
        self.vertices = {}
        self.edges = []

    def add_vertex(self, vertex):
        """
        Add a vertex to the graph.
        :param vertex: The vertex to be added.
        """
        if vertex not in self.vertices:
            self.vertices[vertex] = Vertex(vertex)

    def add_edge(self, u, v, weight):
        """
        Add an edge to the graph.
        :param u: The starting vertex of the edge.
        :param v: The ending vertex of the edge.
        :param weight: The weight of the edge.
        """
        if u not in self.vertices:
            self.add_vertex(u)
        if v not in self.vertices:
            self.add_vertex(v)
        self.edges.append((u, v, weight))
        self.vertices[u].add_neighbor(self.vertices[v], weight)

    def get_vertices(self):
        """
        Get the vertices of the graph.
        :return: A list of vertices.
        """
        return self.vertices.values()

    def get_edges(self):
        """
        Get the edges of the graph.
        :return: A list of edges.
        """
        return self.edges

class Vertex:
    def __init__(self, key):
        """
        Initialize the vertex with a key.
        :param key: The identifier for the vertex.
        """
        self.id = key
        self.connected_to = {}
        self.distance = float('Inf')
        self.visited = False

    def add_neighbor(self, neighbor, weight=0):
        """
        Add a neighbor to the vertex.
        :param neighbor: The neighbor vertex.
        :param weight: The weight of the edge to the neighbor.
        """
        self.connected_to[neighbor] = weight

    def get_connections(self):
        """
        Get the neighbors of the vertex.
        :return: A list of neighbor vertices.
        """
        return self.connected_to.keys()

    def get_id(self):
        """
        Get the id of the vertex.
        :return: The id of the vertex.
        """
        return self.id

    def get_weight(self, neighbor):
        """
        Get the weight of the edge to a neighbor.
        :param neighbor: The neighbor vertex.
        :return: The weight of the edge.
        """
        return self.connected_to[neighbor]

    def set_distance(self, dist):
        """
        Set the distance of the vertex from the source.
        :param dist: The distance to be set.
        """
        self.distance = dist

    def get_distance(self):
        """
        Get the distance of the vertex from the source.
        :return: The distance of the vertex.
        """
        return self.distance

    def set_visited(self):
        """
        Mark the vertex as visited.
        """
        self.visited = True

    def get_visited(self):
        """
        Check if the vertex has been visited.
        :return: True if the vertex has been visited, False otherwise.
        """
        return self.visited

def bellman_ford(graph, start):
    """
    Bellman-Ford algorithm to find the shortest paths from the source vertex to all other vertices.
    :param graph: The graph on which to run the algorithm.
    :param start: The starting vertex.
    :return: True if no negative weight cycles are detected, False otherwise.
    """
    initialize_single_source(graph, start)
    for _ in range(len(graph.get_vertices()) - 1):
        for u, v, weight in graph.get_edges():
            if v.get_visited():
                continue
            relax(u, v, weight)

    for u, v, weight in graph.get_edges():
        if v.get_distance() > u.get_distance() + u.get_weight(v):
            print("Graph contains a negative weight cycle")
            return False
    return True

# Example usage
if __name__ == "__main__":
    graph = Graph()
    edges = [
        (0, 1, -1),
        (0, 2, 4),
        (1, 2, 3),
        (1, 3, 2),
        (1, 4, 2),
        (3, 2, 5),
        (3, 1, 1),
        (4, 3, -3)
    ]

    for u, v, w in edges:
        graph.add_edge(u, v, w)

    start_vertex = graph.vertices[0]
    result = bellman_ford(graph, start_vertex)
    print("No negative weight cycle detected" if result else "Graph contains a negative weight cycle")
