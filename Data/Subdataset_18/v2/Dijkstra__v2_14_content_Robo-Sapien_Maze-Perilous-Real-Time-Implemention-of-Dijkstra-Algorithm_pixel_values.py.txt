import cv2
import numpy as np

# Load the image
img = cv2.imread('newa4.jpg')

# Get the dimensions of the image
height, width, channels = img.shape
print(f"Image dimensions: {img.shape}")

# Define the font to use for text
font = cv2.FONT_HERSHEY_SIMPLEX

# Iterate through the image with a step of 100 pixels
for y in range(0, height, 100):
    for x in range(0, width, 100):
        # Get the BGR color value at the current position
        color = img[y, x]
        # Convert the BGR value to a string
        color_str = str(tuple(color))
        # Place the color value as text on the image
        cv2.putText(img, color_str, (x, y), font, 0.4, (255, 255, 255), 1)
        # Draw a small circle at the current position
        cv2.circle(img, (x, y), 3, (255, 255, 255), 1)

# Display the image with the color values and circles
cv2.imshow('Annotated Image', img)
cv2.waitKey(0)
cv2.destroyAllWindows()
