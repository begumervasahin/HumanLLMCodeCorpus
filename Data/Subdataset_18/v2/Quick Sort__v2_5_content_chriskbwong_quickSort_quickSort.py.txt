def partition(arr, low, high):
    """
    Partition the array on the basis of pivot element.
    
    :param arr: List of elements to be sorted.
    :param low: Starting index of the array slice.
    :param high: Ending index of the array slice.
    :return: Index position of the pivot element after partition.
    """
    i = low - 1  # Index of the smaller element
    pivot = arr[high]  # Pivot element

    for j in range(low, high):
        if arr[j] <= pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]

    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    return i + 1

def quick_sort(arr, low, high):
    """
    Perform non-recursive QuickSort on the array.
    
    :param arr: List of elements to be sorted.
    :param low: Starting index of the array slice.
    :param high: Ending index of the array slice.
    """
    size = high - low + 1
    stack = [0] * size  # Initialize stack

    top = -1
    top += 1
    stack[top] = low
    top += 1
    stack[top] = high

    while top >= 0:
        high = stack[top]
        top -= 1
        low = stack[top]
        top -= 1

        p = partition(arr, low, high)

        if p - 1 > low:
            top += 1
            stack[top] = low
            top += 1
            stack[top] = p - 1

        if p + 1 < high:
            top += 1
            stack[top] = p + 1
            top += 1
            stack[top] = high

    array = [4, 3, 5, 2, 1, 3, 2, 3]
    n = len(array)
    quick_sort(array, 0, n - 1)
