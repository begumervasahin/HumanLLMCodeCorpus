import random

def generate_random_numbers(count=100, upper_limit=100000):
    """
    Generates a list of unique random numbers within the specified range.

    Args:
        count (int): The number of random numbers to generate. Default is 100.
        upper_limit (int): The upper limit for the random numbers. Default is 100000.

    Returns:
        list: A list of unique random numbers.
    """
    return random.sample(range(upper_limit + 1), count)

def selection_sort(arr):
    """
    Sorts the input list using the selection sort algorithm.

    Args:
        arr (list): The list of integers to be sorted.
    
    Returns:
        list: The sorted list.
    """
    n = len(arr)
    for i in range(n):
        # Find the index of the minimum element in the remaining unsorted part of the list
        min_index = i
        for j in range(i + 1, n):
            if arr[j] < arr[min_index]:
                min_index = j
        # Swap the found minimum element with the first element of the unsorted part
        arr[i], arr[min_index] = arr[min_index], arr[i]
    return arr

def main():
    """
    Main function to generate random numbers and sort them using selection sort.
    """
    random_list = generate_random_numbers()
    print("Original list:")
    print(random_list)
    
    sorted_list = selection_sort(random_list)
    print("Sorted list:")
    print(sorted_list)

if __name__ == "__main__":
    main()
