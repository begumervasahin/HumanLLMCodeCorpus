import numpy as np
from kernels import Kernels

class Ktsne:
    def __init__(self, X, f_opts=None):
        """
        Initializes the Ktsne class with input data and options.

        :param X: Input data (numpy array).
        :param f_opts: Dictionary of options including n_dims, perplexity, eta, gamma, p_dims, p_degree, and ker.
        """
        if f_opts is None:
            f_opts = {}

        # Initialization of parameters
        self._dim = f_opts.get("n_dims", 2)
        self._perp = f_opts.get("perplexity", 30)
        self._eta = f_opts.get("eta", 200)
        self.f_opts = f_opts
        self._momentum = 0.4
        self._iter = 0

        # Copy input data and initialize other variables
        self.X = X.copy()
        self.P = None
        self.Q = None
        self.dY = None

        # Initialize Y and Y_t for the t-SNE process
        n, d = self.X.shape
        self.Y = np.random.randn(n, self._dim) * np.sqrt(6. / sum((n, self._dim)))
        self.Y_t = np.array([np.zeros_like(self.Y), np.zeros_like(self.Y)])

        # Suppress warnings for division and invalid operations
        np.seterr(divide='ignore', invalid='ignore')

    def reduce_X(self):
        """
        Reduces the dimensionality of X using the specified kernel method.
        """
        kernel_options = {
            "kernel": self.f_opts.get("ker", "pca"),
            "gamma": self.f_opts.get("gamma", 0.5),
            "degree": self.f_opts.get("p_degree", 1),
            "p_dims": self.f_opts.get("p_dims", 4)
        }
        kn = Kernels(self.X, k_opts=kernel_options)
        self.X = kn.process_data()

    @staticmethod
    def L1(X):
        """
        Computes the L1 (Manhattan) distance between rows of X.

        :param X: Input data matrix.
        :return: L1 distance matrix.
        """
        return np.sum(np.abs(X[:, None] - X[None, :]), axis=-1)

    @staticmethod
    def L2(X):
        """
        Computes the L2 (Euclidean) distance between rows of X.

        :param X: Input data matrix.
        :return: L2 distance matrix.
        """
        return np.sum((X[:, None] - X[None, :])**2, axis=-1)

    def bin_search(self, d_row, target, tol=1e-2, niter=1000, low=1e-10, up=1e3):
        """
        Performs binary search to find the optimal value that matches the target entropy.

        :param d_row: Distance row.
        :param target: Target entropy.
        :param tol: Tolerance for convergence.
        :param niter: Maximum number of iterations.
        :param low: Lower bound for search.
        :param up: Upper bound for search.
        :return: Optimized probability row and corresponding sigma.
        """
        for _ in range(niter):
            estimated = (low + up) / 2.0
            p_row = np.exp(-d_row / estimated)
            p_row /= np.sum(p_row)

            # Calculate the entropy and the difference from the target
            val = self.entropy(p_row)
            dif = np.abs(val - target)

            if dif <= tol:
                break

            # Adjust bounds based on whether the entropy is greater or less than the target
            if val > target:
                up = estimated
            else:
                low = estimated

        return p_row, estimated

    @staticmethod
    def entropy(p_row):
        """
        Computes the Shannon entropy of a probability distribution.

        :param p_row: Probability row.
        :return: Entropy value.
        """
        return -np.sum(p_row * np.log2(p_row))

    def compute_P(self):
        """
        Computes the joint probability matrix P based on the input data X.

        :return: Probability matrix P.
        """
        n, d = self.X.shape
        P = np.zeros((n, n))
        sigmas = np.ones((n, 1))

        # Compute pairwise distances
        D = self.L2(self.X)
        target_entropy = np.log2(self._perp)

        # Calculate P matrix row by row using binary search for each point
        for i in range(n):
            d_row = D[i, np.hstack((np.arange(0, i), np.arange(i + 1, n)))]
            p_row, sigma = self.bin_search(d_row, target_entropy)
            P[i, np.hstack((np.arange(0, i), np.arange(i + 1, n)))] = p_row
            sigmas[i] = sigma

        # Normalize and symmetrize P matrix
        P = (P + P.T) / (2 * n)
        P = P / np.sum(P)
        P = np.maximum(P, 1e-12)  # Avoid zero probabilities

        # Print the mean value of sigma for information
        msig = np.mean(np.sqrt(1 / sigmas))
        print("Mean value of sigma:", msig)

        return P

    def compute_Q(self):
        """
        Computes the joint probability matrix Q based on the low-dimensional representation Y.

        :return: Probability matrix Q.
        """
        D = self.L2(self.Y)
        q = 1 / (1 + D)
        np.fill_diagonal(q, 0)
        self.Q = q / np.sum(q)
        self.Q = np.maximum(self.Q, 1e-12)  # Avoid zero probabilities
        return q

    def gradient(self, q):
        """
        Computes the gradient of the Kullback-Leibler divergence with respect to Y.

        :param q: Joint probability matrix Q.
        """
        PQ = self.P - self.Q
        M = PQ * q
        MD = 4 * (np.diag(np.sum(M, axis=1)) - M)
        self.dY = np.dot(MD, self.Y)

    def get_solution(self, steps=500):
        """
        Runs the t-SNE optimization and returns the final low-dimensional representation.

        :param steps: Number of iterations.
        :return: Low-dimensional representation Y.
        """
        self.reduce_X()
        self.P = self.compute_P() * 10.0  # Amplify P initially

        for i in range(steps):
            cost = self.step()
            if i % 500 == 0:
                print(f"Iteration {i}: cost is {cost}")

        print("Final cost:", cost)
        self.Y -= np.mean(self.Y, axis=0)
        return self.Y

    def step(self):
        """
        Performs a single optimization step.

        :return: Kullback-Leibler divergence cost.
        """
        q = self.compute_Q()
        self.gradient(q)

        # Adjust parameters at specific iterations
        if self._iter == 100:
            self.P /= 10.0
        if self._iter == 25:
            self._momentum = 0.8

        # Update Y and perform momentum-based optimization
        self.Y -= self._eta * self.dY
        self.Y += self._momentum * np.diff(self.Y_t, axis=0)[0]
        self.Y_t[1] = self.Y_t[0].copy()
        self.Y_t[0] = self.Y

        # Compute the current cost using Kullback-Leibler divergence
        C = np.sum(self.P * np.log(self.P / self.Q))
        self._iter += 1

        return C

    # Generate random data for testing
    data = np.random.rand(100, 10)

    # Initialize Ktsne with specific options
    f_opts = {
        "n_dims": 2,
        "perplexity": 30,
        "eta": 200,
        "ker": "rbf",
        "gamma": 0.5,
        "p_dims": 50,
        "p_degree": 3
    }
    tsne = Ktsne(data, f_opts)

    # Perform t-SNE and get the reduced data
    reduced_data = tsne.get_solution(steps=1000)

    print("Reduced data shape:", reduced_data.shape)
