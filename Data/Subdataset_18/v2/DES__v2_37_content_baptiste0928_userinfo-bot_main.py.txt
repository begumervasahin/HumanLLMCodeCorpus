import json
import discord
from discord.ext import commands
import glob
import os

# Load the configuration file
with open('config.json', 'r') as f:
    config = json.load(f)

# Initialize the bot with the specified prefix
bot = commands.AutoShardedBot(command_prefix=commands.when_mentioned_or(config.get("prefix", ".")))

# Remove the default help command
bot.remove_command("help")

# Add the config to the bot instance
bot.config = config

# Discover and load all cogs from the cogs directory
def load_cogs():
    cogs = [file.replace(os.path.sep, '.').replace('.py', '') for file in glob.glob("cogs/*.py")]
    for cog in cogs:
        try:
            bot.load_extension(cog)
            print(f"Successfully loaded extension {cog}")
        except Exception as e:
            print(f"Failed to load extension {cog}: {e}")

if __name__ == '__main__':
    load_cogs()

# Define the on_ready event
@bot.event
async def on_ready():
    await bot.change_presence(
        activity=discord.Activity(
            status=discord.Status.idle,
            name=config["activity"],
            type=discord.ActivityType.watching
        )
    )
    print(f"Connected as {bot.user}")
    print("Ready!")

# Run the bot with the token from the config
bot.run(config.get("token"))
