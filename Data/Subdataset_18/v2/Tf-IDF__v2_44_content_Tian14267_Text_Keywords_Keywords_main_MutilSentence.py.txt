import numpy as np
import time
import Keywords_Algorithm as KA
import keywords_tools as KT

def pred_model():
    """
    Initialize and return the necessary models and tools for keyword extraction.
    """
    tools = KT.Tools()
    tf_idf = KA.TFIDF(tools)
    tr = KA.TextRank(3, 0.85, 700, tools)
    return tf_idf, tr, tools

def mix_keywords(tf_idf, tr, tools, text):
    """
    Perform keyword extraction using TF-IDF and TextRank, then combine results.

    Args:
        tf_idf: TF-IDF model.
        tr: TextRank model.
        tools: Utility tools for keyword processing.
        text: The input text to process.
    """
    # Extract keywords using TF-IDF
    result, result_sort = tf_idf.Do_keywords(text)
    
    # Extract keywords using TextRank
    result_tr, result_tr_sort = tr.calculate_textrank(text)
    
    # Combine the results using a weighted cross method
    method_num = "Weight_Cross"
    mix_result = tools.Mix_algorithm(method_num, result, result_tr)
    return mix_result

if __name__ == '__main__':
    # Define the path to the input file
    path = "./data/cnews.test.txt"
    
    # Record the start time of the processing
    start_time = time.time()

    # Read the file and preprocess the lines
    with open(path, 'r', encoding='utf-8') as f:
        lines = f.readlines()
    
    new_lines = [line.strip().replace('\n', '') for line in lines]

    # Initialize the total character counter
    all_characters = 0
    
    # Initialize the models and tools
    tf_idf, tr, tools = pred_model()

    # Process each line of text
    for i, one_line in enumerate(new_lines):
        all_characters += len(one_line)
        mix_keywords(tf_idf, tr, tools, one_line)
        print(f"Completed: {i}")

    # Record the end time of the processing
    end_time = time.time()
    
    # Print the total number of characters processed and the total time taken
    print(f"Total characters: {all_characters}")
    print(f"Total time: {end_time - start_time} seconds")
