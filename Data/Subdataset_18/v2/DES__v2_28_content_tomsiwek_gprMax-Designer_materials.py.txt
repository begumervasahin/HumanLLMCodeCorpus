class TMaterial:
    """
    Class representing a material with its properties.

    Attributes:
        epsilon_r (float): Relative permittivity.
        sigma (float): Conductivity.
        mu_r (float): Relative permeability.
        sigma_mag (float): Magnetic loss.
        name (str): Material identifier.
    """

    def __init__(self, epsilon_r=0.0, sigma=0.0, mu_r=0.0, sigma_mag=0.0, name=""):     
        """
        Initialize the properties of the material.

        Args:
            epsilon_r (float): Relative permittivity.
            sigma (float): Conductivity.
            mu_r (float): Relative permeability.
            sigma_mag (float): Magnetic loss.
            name (str): Material identifier.
        """
        self.epsilon_r = epsilon_r
        self.sigma = sigma
        self.mu_r = mu_r
        self.sigma_mag = sigma_mag
        self.name = name

    def velocity(self):
        """
        Calculate the electromagnetic wave velocity in the material.

        Returns:
            float: The velocity of the electromagnetic wave in the material.
        """
        c = 299792458  # Speed of light in vacuum (in meters per second)
        return c / (self.epsilon_r ** 0.5)

def main():
    """
    Main function to demonstrate the usage of the TMaterial class.
    """
    # Create an instance of TMaterial with sample properties
    material = TMaterial(epsilon_r=4.0, sigma=0.01, mu_r=1.0, sigma_mag=0.0, name="SampleMaterial")

    # Calculate the velocity of the electromagnetic wave in the material
    velocity = material.velocity()

    # Print the result
    print(f"The velocity of the electromagnetic wave in {material.name} is {velocity:.2f} m/s")

if __name__ == "__main__":
    main()
