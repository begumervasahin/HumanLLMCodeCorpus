# Compute the initial value
val = (5**30) % 2017
print(f"S1 {val}\n")

# Initialize a list to store values
values = []

# Iterate to generate and classify values
for _ in range(1, 10):
    # Compute the next value in the sequence
    val2 = (val**2) % 2017
    
    # Classify and print val2
    if 1 <= val2 <= 672:
        print(f"S0 {val2}\n")
    elif 673 <= val2 <= 1345:
        print(f"S1 {val2}\n")
    elif 1346 <= val2 <= 2016:
        print(f"S2 {val2}\n")
    else:
        print("oops\n")
    
    # Append current values to the list
    values.append(val)
    values.append(val2)
    
    # Update val for the next iteration
    val = (1736 * val2) % 2017
    
    # Classify and print the updated val
    if 1 <= val <= 672:
        print(f"S0 {val}\n")
    elif 673 <= val <= 1345:
        print(f"S1 {val}\n")
    elif 1346 <= val <= 2016:
        print(f"S2 {val}\n")
    else:
        print("oops\n")

# Print the final list of values
print(values)

# Create a dictionary to map each value to its indices in the list
value_indices = {}
for i, value in enumerate(values):
    if value not in value_indices:
        value_indices[value] = []
    value_indices[value].append(i)

# Print the dictionary of values and their indices
print(value_indices)
