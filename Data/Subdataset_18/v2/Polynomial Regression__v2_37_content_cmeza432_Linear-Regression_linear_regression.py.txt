import numpy as np
import sys

def get_bigphi(data, rows, cols, degree):
    """
    Constructs the design matrix (big_phi) with polynomial features up to the given degree.
    Also extracts target values from the last column of the data.

    Parameters:
    data (array): Input data
    rows (int): Number of rows in the data
    cols (int): Number of columns in the data
    degree (int): Degree of polynomial features

    Returns:
    tuple: (design matrix, target values)
    """
    result = []
    t_values = []
    for i in range(rows):
        row = [1]  # Start with the bias term (intercept)
        for k in range(cols):
            if k == (cols - 1):
                t_values.append(data[i][k])
            else:
                for deg in range(degree):
                    row.append(data[i][k] ** (deg + 1))
        result.append(row)
    return np.array(result), np.array(t_values)

def get_weights(big_phi, t_values, lamb):
    """
    Computes the weights for regularized linear regression.

    Parameters:
    big_phi (array): Design matrix
    t_values (array): Target values
    lamb (float): Regularization parameter

    Returns:
    array: Computed weights
    """
    phi_t = np.transpose(big_phi)
    identity_value = big_phi.shape[1]
    identity = np.identity(identity_value)
    lamb_m = lamb * identity
    first = np.add(np.matmul(phi_t, big_phi), lamb_m)
    inverse = np.linalg.pinv(first)
    second = np.matmul(inverse, phi_t)
    weights = np.matmul(second, t_values)
    return weights

def get_prediction(test_phi, weights):
    """
    Predicts target values for the test set using the computed weights.

    Parameters:
    test_phi (array): Design matrix for the test set
    weights (array): Computed weights

    Returns:
    array: Predicted target values
    """
    return np.dot(test_phi, weights)

def print_weights(weights):
    """
    Prints the computed weights.

    Parameters:
    weights (array): Computed weights
    """
    for i, w in enumerate(weights):
        print(f"w{i}={w:.4f}")

def print_test(predictions, target_values):
    """
    Prints the prediction results and squared errors for the test set.

    Parameters:
    predictions (array): Predicted target values
    target_values (array): Actual target values
    """
    for i, (pred, target) in enumerate(zip(predictions, target_values)):
        error = (pred - target) ** 2
        print(f"ID={i+1:5d}, output={pred:.4f}, target value={target:.4f}, squared error={error:.4f}")

def linear_regression(training, degree, lamb, test):
    """
    Performs linear regression with polynomial features and regularization.

    Parameters:
    training (array): Training data
    degree (int): Degree of polynomial features
    lamb (float): Regularization parameter
    test (array): Test data
    """
    training_rows, training_cols = training.shape
    test_rows, test_cols = test.shape

    big_phi, t_values = get_bigphi(training, training_rows, training_cols, degree)
    weights = get_weights(big_phi, t_values, lamb)
    
    print_weights(weights)
    
    test_phi, target_values = get_bigphi(test, test_rows, test_cols, degree)
    predictions = get_prediction(test_phi, weights)
    
    print_test(predictions, target_values)

if len(sys.argv) != 5:
    print("Error, not enough arguments given!")
else:
    training_file = sys.argv[1]
    degree = int(sys.argv[2])
    lamb = float(sys.argv[3])
    test_file = sys.argv[4]

    with open(training_file) as f:
        training_data = np.array([list(map(float, line.split())) for line in f])

    with open(test_file) as f:
        test_data = np.array([list(map(float, line.split())) for line in f])

    linear_regression(training_data, degree, lamb, test_data)
