def insertion_sort(array):
    """
    Perform an insertion sort on a list of integers.
    
    The function iteratively takes an element from the list and 
    places it into its correct position relative to the already sorted part of the list.
    """
    for i in range(1, len(array)):
        current_value = array[i]
        position = i - 1

        # Shift elements of the sorted portion of the list to the right
        while position >= 0 and array[position] > current_value:
            array[position + 1] = array[position]
            position -= 1
        
        # Insert the current value into its correct position
        array[position + 1] = current_value

def read_data_from_file(filename):
    """
    Read a file containing multiple lines of integers and return a list of lists.
    
    Each line represents a list of integers. The first value of each line is ignored, 
    as it is assumed to be a placeholder.
    """
    all_lists = []
    
    with open(filename, "r") as file:
        for line in file:
            # Convert the line into a list of integers, skipping the first value
            numbers = list(map(int, line.split()[1:]))
            all_lists.append(numbers)
    
    return all_lists

def write_data_to_file(filename, data):
    """
    Write a list of lists of integers to a file, with each inner list written on a new line.
    
    Integers within each list are separated by spaces.
    """
    with open(filename, "w") as file:
        for number_list in data:
            file.write(" ".join(map(str, number_list)) + "\n")

def main():
    # Step 1: Read unsorted lists from 'data.txt'
    unsorted_data = read_data_from_file("data.txt")
    
    # Step 2: Sort each list using the insertion sort algorithm
    for sublist in unsorted_data:
        insertion_sort(sublist)
    
    # Step 3: Write the sorted lists to 'insertionsort.txt'
    write_data_to_file("insertionsort.txt", unsorted_data)

if __name__ == "__main__":
    main()
