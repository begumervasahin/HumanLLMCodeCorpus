class Node:
    def __init__(self, char=None):
        self.char = char if char is not None else b'*'
        self.weight = 0
        self.level = 0
        self.parent = None
        self.left = None
        self.right = None

    def set_left(self, child_node):
        self.left = child_node
        child_node.parent = self
        child_node.update_level()

    def set_right(self, child_node):
        self.right = child_node
        child_node.parent = self
        child_node.update_level()

    def replace_child(self, current_child, new_child):
        if self.left == current_child:
            self.set_left(new_child)
        elif self.right == current_child:
            self.set_right(new_child)

    def update_level(self):
        self.level = self.parent.level + 1 if self.parent else 0
        if self.left:
            self.left.update_level()
        if self.right:
            self.right.update_level()

    def has_no_children(self):
        return self.left is None and self.right is None

    def is_ancestor_of(self, node):
        ancestor = self.parent
        while ancestor:
            if ancestor == node:
                return True
            ancestor = ancestor.parent
        return False

    def swap_with(self, node):
        if self == node or node.is_ancestor_of(self) or self.is_ancestor_of(node):
            return
        
        self_parent = self.parent
        node_parent = node.parent
        
        node_parent.replace_child(node, self)
        self_parent.replace_child(self, node)

# Example Usage
if __name__ == "__main__":
    # Create some nodes
    node_a = Node('A')
    node_b = Node('B')
    node_c = Node('C')

    # Build a simple tree
    node_a.set_left(node_b)
    node_a.set_right(node_c)

    # Swap nodes
    node_b.swap_with(node_c)

    # Check tree structure after swap
    print(f"Node A's left child: {node_a.left.char}")  # Should be 'C'
    print(f"Node A's right child: {node_a.right.char}")  # Should be 'B'
