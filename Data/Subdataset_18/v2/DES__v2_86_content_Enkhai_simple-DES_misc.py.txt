import binascii

def text_to_bits(text, encoding='utf-8', errors='surrogatepass'):
    """
    Converts a text string to its binary representation.

    :param text: The input text to convert.
    :param encoding: The encoding to use for text conversion.
    :param errors: The error handling scheme to use for encoding errors.
    :return: The binary representation of the input text.
    """
    bits = bin(int(binascii.hexlify(text.encode(encoding, errors)), 16))[2:]
    return bits.zfill(8 * ((len(bits) + 7) // 8))

def text_from_bits(bits, encoding='utf-8', errors='surrogatepass'):
    """
    Converts a binary string back to its text representation.

    :param bits: The binary string to convert.
    :param encoding: The encoding to use for text conversion.
    :param errors: The error handling scheme to use for decoding errors.
    :return: The text representation of the input binary string.
    """
    n = int(bits, 2)
    return int2bytes(n).decode(encoding, errors)

def int2bytes(i):
    """
    Converts an integer to its byte representation.

    :param i: The integer to convert.
    :return: The byte representation of the integer.
    """
    hex_string = '%x' % i
    n = len(hex_string)
    return binascii.unhexlify(hex_string.zfill(n + (n & 1)))

def hex_to_bits(text):
    """
    Converts a hexadecimal string to its binary representation.

    :param text: The hexadecimal string to convert.
    :return: The binary representation of the input hexadecimal string.
    """
    bits = bin(int(text, 16))[2:]
    return bits.zfill(8 * ((len(bits) + 7) // 8))

def binary_addition(value1, value2, num_of_bits):
    """
    Performs binary addition of two binary strings and returns the result.

    :param value1: The first binary string.
    :param value2: The second binary string.
    :param num_of_bits: The number of bits for the result.
    :return: The binary result of the addition.
    """
    bits = bin(int(value1, 2) + int(value2, 2))[2:]
    return bits.zfill(num_of_bits)[-num_of_bits:]

def XOR(value1, value2):
    """
    Performs bitwise XOR on two binary strings and returns the result.

    :param value1: The first binary string.
    :param value2: The second binary string.
    :return: The binary result of the XOR operation.
    """
    bits = bin(int(value1, 2) ^ int(value2, 2))[2:]
    return bits.zfill(8 * ((len(bits) + 7) // 8))

def left_shift(text, d):
    """
    Performs a left circular shift on a binary string.

    :param text: The binary string to shift.
    :param d: The number of positions to shift.
    :return: The shifted binary string.
    """
    return text[d:] + text[:d]

def main():
    """
    Main function to demonstrate the usage of the utility functions.
    """
    text = "Hello, World!"
    key = "AABB09182736CCDD"
    
    # Convert text to binary
    binary_text = text_to_bits(text)
    print(f"Binary representation of '{text}': {binary_text}")
    
    # Convert binary back to text
    restored_text = text_from_bits(binary_text)
    print(f"Restored text from binary: {restored_text}")
    
    # Convert hexadecimal to binary
    binary_key = hex_to_bits(key)
    print(f"Binary representation of key '{key}': {binary_key}")
    
    # Perform binary addition
    added_binary = binary_addition('1101', '1011', 4)
    print(f"Binary addition result: {added_binary}")
    
    # Perform XOR operation
    xor_result = XOR('1101', '1011')
    print(f"XOR result: {xor_result}")
    
    # Perform left shift
    shifted_binary = left_shift('1101', 2)
    print(f"Left shift result: {shifted_binary}")

if __name__ == "__main__":
    main()
