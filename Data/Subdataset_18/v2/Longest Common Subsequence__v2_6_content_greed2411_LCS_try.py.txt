import re

def verify(sequence):
    # Verify that the sequence contains only A, T, C, G, and spaces
    return bool(re.match("^[ATCG ]*$", sequence))

def lcs(X, Y, m, n):
    # Initialize the matrix for storing the lengths of longest common subsequences
    L = [[0] * (n + 1) for _ in range(m + 1)]

    # Fill the matrix in a bottom-up manner
    for i in range(m + 1):
        for j in range(n + 1):
            if i == 0 or j == 0:
                L[i][j] = 0
            elif X[i - 1] == Y[j - 1]:
                L[i][j] = L[i - 1][j - 1] + 1
            else:
                L[i][j] = max(L[i - 1][j], L[i][j - 1])

    # The length of the longest common subsequence
    index = L[m][n]

    # Create an array to store the LCS
    lcs = [""] * (index + 1)
    lcs[index] = "\0"

    # Backtrack to find the LCS
    i, j = m, n
    while i > 0 and j > 0:
        if X[i - 1] == Y[j - 1]:
            lcs[index - 1] = X[i - 1]
            i -= 1
            j -= 1
            index -= 1
        elif L[i - 1][j] > L[i][j - 1]:
            i -= 1
        else:
            j -= 1

    # Convert the LCS array to a string
    lcs_result = "".join(lcs).rstrip("\0")

    # Print the LCS and its length
    print(f"\n\nLCS of {X} and {Y} is {lcs_result} \nand length of it is: {len(lcs_result)}")

def main():
    # Get the sequences from the user
    s1 = input("Enter the First Sequence\n").upper()
    s2 = input("\nEnter the Querying Sequence\n").upper()

    # Verify the sequences and compute the LCS if valid
    if verify(s1) and verify(s2):
        if len(s1) >= len(s2):
            m = len(s1)
            n = len(s2)
            lcs(s1, s2, m, n)
        else:
            print("Querying Sequence should be smaller than First Sequence")
    else:
        print("The Sequences should contain 'A', 'T', 'C', 'G', ' ' only")

if __name__ == "__main__":
    main()
