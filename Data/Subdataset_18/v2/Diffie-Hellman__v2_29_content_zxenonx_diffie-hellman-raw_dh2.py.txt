import random
import hashlib

# Define the public parameters for the key exchange
g = 9  # Shared base
p = 1001  # Prime number

# Generate random private secrets for Alice and Bob
alice_secret = random.randint(5, 10)
bob_secret = random.randint(10, 20)

# Compute Alice's and Bob's public values
alice_public = pow(g, alice_secret, p)
bob_public = pow(g, bob_secret, p)

# Display the public parameters and calculated values
print(f"Shared parameters:")
print(f"  Base (g): {g}")
print(f"  Prime (p): {p}")

print("\nAlice's calculations:")
print(f"  Alice's secret (a): {alice_secret}")
print(f"  Alice's public value (A): {alice_public} (g^a mod p)")

print("\nBob's calculations:")
print(f"  Bob's secret (b): {bob_secret}")
print(f"  Bob's public value (B): {bob_public} (g^b mod p)")

# Calculate and display the shared key for Alice and Bob
print("\nAlice computes the shared key:")
alice_shared_key = pow(bob_public, alice_secret, p)
print(f"  Shared key (computed by Alice): {alice_shared_key}")
print(f"  SHA-256 hash of Alice's shared key: {hashlib.sha256(str(alice_shared_key).encode()).hexdigest()}")

print("\nBob computes the shared key:")
bob_shared_key = pow(alice_public, bob_secret, p)
print(f"  Shared key (computed by Bob): {bob_shared_key}")
print(f"  SHA-256 hash of Bob's shared key: {hashlib.sha256(str(bob_shared_key).encode()).hexdigest()}")
