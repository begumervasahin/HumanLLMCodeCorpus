"""
Pattern name - Singleton
Pattern type - Creational Design Pattern
"""

class SingletonMeta(type):
    """
    A metaclass for implementing the Singleton pattern.
    Ensures that only one instance of the class is created.
    """
    _instances = {}

    def __call__(cls, *args, **kwargs):
        if cls not in cls._instances:
            cls._instances[cls] = super().__call__(*args, **kwargs)
        return cls._instances[cls]

class DBConnector(metaclass=SingletonMeta):
    """
    A database connector class that follows the Singleton pattern.
    Only one instance of this class can be created.
    """
    def __init__(self):
        self.status = "Not Connected"

    def disconnect(self):
        """
        Sets the status to 'Disconnected'.
        """
        self.status = "Disconnected"

    def connect(self):
        """
        Sets the status to 'Connected'.
        """
        self.status = "Connected"

def main():
    """
    The main function to demonstrate the Singleton pattern using the DBConnector class.
    """
    # Creating the first client instance
    client1 = DBConnector()
    print("Client 1:", client1)
    print("Client 1 status:", client1.status)

    # Creating the second client instance (will return the same instance as client1)
    client2 = DBConnector()
    print("Client 2:", client2)

    # Connecting with the second client instance
    client2.connect()
    print("Client 1 status after client2 connects:", client1.status)

    # Disconnecting with the first client instance
    client1.disconnect()
    print("Client 2 status after client1 disconnects:", client2.status)

if __name__ == "__main__":
    main()
