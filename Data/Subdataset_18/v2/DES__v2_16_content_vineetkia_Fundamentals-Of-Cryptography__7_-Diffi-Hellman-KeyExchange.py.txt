def power(base, exponent, modulus):
    """
    Compute (base^exponent) % modulus.

    Parameters
    ----------
    base : int
        The base number.
    exponent : int
        The exponent.
    modulus : int
        The modulus.

    Returns
    -------
    int
        The result of (base^exponent) % modulus.
    """
    return pow(base, exponent, modulus)

# Given values
G = 9   # Base number
a = 4   # Private key A
b = 3   # Private key B
P = 29  # Modulus

# Display given values
print(f"P Value: {P}\n")
print(f"G Value: {G}\n")
print(f"A Private KEY: {a}\n")

# Calculate public keys
x = power(G, a, P)
print(f"The private key B: {b}\n")
y = power(G, b, P)

# Calculate secret keys
ka = power(y, a, P)
kb = power(x, b, P)

# Display secret keys
print(f"A Secret Key: {ka}\n")
print(f"B Secret Key: {kb}\n")
