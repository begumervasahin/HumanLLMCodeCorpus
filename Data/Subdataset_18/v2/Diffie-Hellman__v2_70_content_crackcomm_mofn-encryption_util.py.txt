import base64
from cryptography.hazmat.primitives import serialization
from cryptography.hazmat.primitives.asymmetric.x25519 import X25519PrivateKey

def private_bytes(private_key):
    """
    Serialize the private key to bytes.

    Parameters:
    private_key (X25519PrivateKey): The private key to serialize.

    Returns:
    bytes: The serialized private key.
    """
    return private_key.private_bytes(
        encoding=serialization.Encoding.Raw,
        format=serialization.PrivateFormat.Raw,
        encryption_algorithm=serialization.NoEncryption()
    )

def public_bytes(private_key):
    """
    Serialize the public key derived from the private key to bytes.

    Parameters:
    private_key (X25519PrivateKey): The private key to derive the public key from.

    Returns:
    bytes: The serialized public key.
    """
    return private_key.public_key().public_bytes(
        encoding=serialization.Encoding.Raw,
        format=serialization.PublicFormat.Raw
    )

def exchange_key(private_key_bytes, public_key_bytes):
    """
    Perform a key exchange using the private key and the public key.

    Parameters:
    private_key_bytes (bytes): The private key bytes.
    public_key_bytes (bytes): The public key bytes.

    Returns:
    bytes: The shared secret derived from the key exchange.
    """
    private_key = X25519PrivateKey.from_private_bytes(private_key_bytes)
    public_key = X25519PrivateKey.from_private_bytes(public_key_bytes).public_key()
    return private_key.exchange(public_key)

def print_key(private_key):
    """
    Print the base64 encoded private and public keys.

    Parameters:
    private_key (X25519PrivateKey): The private key to print.
    """
    private_key_encoded = base64.b64encode(private_bytes(private_key)).decode('utf-8')
    public_key_encoded = base64.b64encode(public_bytes(private_key)).decode('utf-8')
    print(f'Private Key: {private_key_encoded}')
    print(f'Public Key: {public_key_encoded}')

def main():
    # Generate a new private key
    private_key = X25519PrivateKey.generate()

    # Print the private and public keys
    print_key(private_key)

    # Generate another private key for the key exchange demonstration
    another_private_key = X25519PrivateKey.generate()

    # Perform a key exchange
    shared_secret = exchange_key(private_bytes(private_key), public_bytes(another_private_key))
    shared_secret_encoded = base64.b64encode(shared_secret).decode('utf-8')
    print(f'Shared Secret: {shared_secret_encoded}')

if __name__ == '__main__':
    main()
