import random

def generate_array(size):
    """
    Generate an array of given size with random integers ranging from -size to size.
    
    Parameters:
    size (int): The size of the array to generate.
    
    Returns:
    list: An array of random integers.
    """
    return [random.randint(-size, size) for _ in range(size)]

def generate_payload():
    """
    Generate a list of arrays with predefined sizes.
    
    Returns:
    list: A list containing arrays of various sizes.
    """
    sizes = [10, 100, 1000, 10000, 100000, 1000000, 10000000]
    return [generate_array(size) for size in sizes]

# Generate the payload
payload = generate_payload()

# Print a summary of the sizes of the generated arrays
for index, array in enumerate(payload):
    print(f"Array {index + 1} size: {len(array)}")
