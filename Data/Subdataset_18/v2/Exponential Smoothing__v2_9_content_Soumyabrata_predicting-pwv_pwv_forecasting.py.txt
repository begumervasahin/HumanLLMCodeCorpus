import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from matplotlib.dates import DateFormatter
from statsmodels.tsa.api import ExponentialSmoothing
from read_matfile import read_matfile

def import_data(matlab_file):
    """
    Imports data from a MATLAB file and returns a DataFrame.

    Parameters:
    matlab_file (str): Path to the MATLAB file.

    Returns:
    pd.DataFrame: DataFrame containing the imported data.
    """
    timestamp, pwv = read_matfile(matlab_file)
    data = np.column_stack((timestamp, pwv))
    df = pd.DataFrame(data, columns=['timestamps', 'pwv'])
    df.set_index('timestamps', inplace=True)
    return df

def train_model(train_data, seasonal_periods=288):
    """
    Trains an Exponential Smoothing model.

    Parameters:
    train_data (pd.Series): Training data for the model.
    seasonal_periods (int): Number of seasonal periods.

    Returns:
    ExponentialSmoothing: Fitted model.
    """
    model = ExponentialSmoothing(
        train_data,
        seasonal_periods=seasonal_periods,
        trend='add',
        seasonal='add'
    ).fit()
    return model

def plot_results(train, test, predictions, save_path):
    """
    Plots the training data, test data, and predictions.

    Parameters:
    train (pd.Series): Training data.
    test (pd.Series): Test data.
    predictions (pd.Series): Predicted data.
    save_path (str): Path to save the plot.
    """
    fig, ax = plt.subplots(figsize=(10, 5))
    ax.plot(train[-150:], 'b:', label='Train')
    ax.plot(test, 'r--', label='Test')
    ax.plot(predictions, 'k-', label='Predicted')
    ax.legend(loc='best', fontsize=18)
    fig.autofmt_xdate()
    formatter = DateFormatter('%d-%m-%y %H:%M')
    ax.xaxis.set_major_formatter(formatter)
    ax.grid(True)
    ax.set_xlabel('Timestamps', fontsize=14)
    ax.set_ylabel('Precipitable Water Vapor (in mm)', fontsize=14)
    fig.tight_layout()
    fig.savefig(save_path)
    plt.show()

def main():
    # Define the path to the MATLAB file
    matlab_file = './data/PWV_2010from_WS_2_withGradient.mat'

    # Import data from the MATLAB file
    df = import_data(matlab_file)

    # Define the start index and create training and test sets
    start_index = 73513
    train = df.iloc[start_index:start_index + 10000]
    test = df.iloc[start_index + 10000:start_index + 10000 + 50]

    # Train the model
    print('Computation started')
    model = train_model(train['pwv'])
    predictions = model.forecast(len(test))
    print('Computation completed')

    # Plot the results
    plot_results(train['pwv'], test['pwv'], predictions, './results/73513-example.pdf')

if __name__ == '__main__':
    main()
