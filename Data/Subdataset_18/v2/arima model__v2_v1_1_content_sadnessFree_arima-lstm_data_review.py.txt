import math
import matplotlib.pyplot as plt
from pandas import read_csv, to_datetime

def parser(x):
    """Parse date string to datetime object."""
    return to_datetime(x, format='%Y-%m-%d %H:%M:%S')

def mean_absolute_percentage_error(y_true, y_pred):
    """
    Calculate Mean Absolute Percentage Error (MAPE).

    Parameters:
    y_true (list): List of true values.
    y_pred (list): List of predicted values.

    Returns:
    float: MAPE as a percentage.
    """
    sum_error = 0
    for true, pred in zip(y_true, y_pred):
        sum_error += abs((pred - true) / true)
    return (sum_error / len(y_true)) * 100  # Multiply by 100 to get percentage

# Path to the CSV file
csv_file_path = '/Users/daihanru/Desktop/arima-lstm/DataSet/FEB15.csv'

# Read CSV file, parse dates, and set the first column as the index
series = read_csv(csv_file_path, usecols=[2, 3], header=0, parse_dates=[0], index_col=0, squeeze=True, date_parser=parser)

# Extract the first 1000 values for plotting
X = series.values[:1000]

# Plot the time series data
plt.plot(X, '-', label="Real Flow")
plt.legend(loc='upper right')
plt.xlabel("Period (15-minute intervals)")
plt.ylabel("Volume (vehicle/period)")
plt.title("Vehicle Volume Over Time")
plt.show()

# Example data for MAPE calculation
# Replace y_true and y_pred with actual data as needed
y_true = [1, 2, 3, 4, 5]
y_pred = [0.9, 2.1, 3.2, 4.3, 5.1]

# Calculate and print MAPE
mape = mean_absolute_percentage_error(y_true, y_pred)
print(f"Mean Absolute Percentage Error (MAPE): {mape:.2f}%")
