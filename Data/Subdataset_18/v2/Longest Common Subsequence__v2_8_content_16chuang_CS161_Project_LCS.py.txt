import sys
import numpy as np

# Initialize a 2D array to store intermediate results
arr = np.zeros((2048, 2048), dtype=int)

def LCS(A, B):
    """
    Compute the length of the Longest Common Subsequence (LCS) of strings A and B.
    """
    m = len(A)
    n = len(B)
    
    # Fill the array using dynamic programming
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if A[i - 1] == B[j - 1]:
                arr[i][j] = arr[i - 1][j - 1] + 1
            else:
                arr[i][j] = max(arr[i - 1][j], arr[i][j - 1])
    
    return arr[m][n]

def main():
    """
    Main function to read input from stdin and print the LCS length for each pair of strings.
    """
    # Ensure the script is used correctly
    if len(sys.argv) != 1:
        sys.exit('Usage: python LCS.py < input')
    
    # Read lines from standard input
    for line in sys.stdin:
        A, B = line.strip().split()
        # Print the length of the Longest Common Subsequence
        print(LCS(A, B))

if __name__ == '__main__':
    main()
