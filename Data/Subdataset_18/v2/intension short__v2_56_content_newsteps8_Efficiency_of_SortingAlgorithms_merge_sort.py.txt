import time

# Initialize a global variable to track the number of comparisons
comparison_count = 0

def merge_sort(arr):
    """Recursively divides and merges the array, counting the number of comparisons."""
    global comparison_count
    
    if len(arr) > 1:
        mid = len(arr) // 2  # Find the middle point to divide the array
        left_half = arr[:mid]  # Create the left half
        right_half = arr[mid:]  # Create the right half

        # Recursively sort both halves
        merge_sort(left_half)
        merge_sort(right_half)

        # Initialize pointers for left_half, right_half, and merged array
        i = j = k = 0

        # Merge the sorted halves
        while i < len(left_half) and j < len(right_half):
            comparison_count += 1  # Increment the comparison count
            if left_half[i] < right_half[j]:
                arr[k] = left_half[i]
                i += 1
            else:
                arr[k] = right_half[j]
                j += 1
            k += 1

        # Check if any element is left in the left_half
        while i < len(left_half):
            comparison_count += 1  # Increment the comparison count
            arr[k] = left_half[i]
            i += 1
            k += 1

        # Check if any element is left in the right_half
        while j < len(right_half):
            comparison_count += 1  # Increment the comparison count
            arr[k] = right_half[j]
            j += 1
            k += 1

def print_list(arr):
    """Prints the elements of the array in a single line."""
    print(" ".join(map(str, arr)))

def main():
    # Start the timer
    start_time = time.time()

    # Read the input file "case"
    with open("case", "r") as file:
        arr = file.read().split()

    # Convert the read strings to integers
    array_to_sort = [int(num) for num in arr]

    # Sort the array using merge sort
    merge_sort(array_to_sort)

    # Print the sorted array
    print("Sorted array is:")
    print_list(array_to_sort)

    # Print the total number of comparisons made
    print(f"Number of comparisons: {comparison_count}")

    # Stop the timer and print the elapsed time
    end_time = time.time()
    print(f"Time taken: {end_time - start_time:.6f} seconds")

if __name__ == "__main__":
    main()
