import datetime
import functools
import operator
import decimal

def factorial(n):
    """
    Computes the factorial of n using reduce and operator.mul.
    
    Parameters:
    n (int): The number to compute the factorial of.
    
    Returns:
    int: The factorial of the input number.
    """
    return functools.reduce(operator.mul, range(1, n + 1), 1)

def log_time_and_result(start_time, end_time, number, result):
    """
    Logs the start time, end time, and result of the factorial computation to a file.
    
    Parameters:
    start_time (datetime): The start time of the computation.
    end_time (datetime): The end time of the computation.
    number (int): The number whose factorial was computed.
    result (decimal.Decimal): The result of the factorial computation.
    """
    time_difference = (end_time - start_time).total_seconds()
    with open("Factorial.txt", "a") as file:
        file.write(f"Start time: {start_time.strftime('%H:%M:%S.%f')}\n")
        file.write(f"End time: {end_time.strftime('%H:%M:%S.%f')}\n")
        file.write(f"Factorial of {number} is {format(result, '.6e')}\n")
        file.write(f"The time difference is {time_difference:.6f} seconds\n")

def main():
    """
    Main function to get inputs, compute factorials, and log results.
    """
    max_number = int(input('What will the maximum count be?: '))
    number_increase = int(input('What will be the number added per round?: '))

    number = 0
    while number < max_number:
        number += number_increase
        start_time = datetime.datetime.now()

        result = factorial(number)

        end_time = datetime.datetime.now()

        log_time_and_result(start_time, end_time, number, decimal.Decimal(result))

if __name__ == "__main__":
    main()
