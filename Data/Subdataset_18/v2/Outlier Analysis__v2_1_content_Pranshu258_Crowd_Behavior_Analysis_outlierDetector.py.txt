import numpy as np
import math
import matplotlib.pyplot as plt
from copy import deepcopy
import csv

def standard_deviation(distances):
    distances_array = np.array(distances)
    std = np.std(distances_array)
    print(f"Mean Distance: {np.mean(distances_array)}")
    print(f"Minimum Distance: {np.min(distances_array)}")
    print(f"Maximum Distance: {np.max(distances_array)}")
    print(f"Standard Deviation of Distances: {std}")
    return std

def projPointOnLine(point, line_2pt):
    x0, y0 = point
    x1, y1, x2, y2 = line_2pt[0], line_2pt[1], line_2pt[2], line_2pt[3]
    vx, vy = x2 - x1, y2 - y1
    dx, dy = x0 - x1, y0 - y1
    t = (dx * vx + dy * vy) / (vx * vx + vy * vy)
    return [x1 + t * vx, y1 + t * vy]

def tp_distance(L1, L2):
    w1, w2, w3, w4 = 1.0, 1.0, 1.0, 1.0
    Dspeed = abs((length(L1) / L1[0][2]) - (length(L2) / L2[0][2]))
    
    if length(L1) > length(L2):
        L1, L2 = deepcopy(L2), deepcopy(L1)

    proj1 = projPointOnLine([L2[0][0], L2[0][1]], [L1[0][0], L1[0][1], L1[1][0], L1[1][1]])
    proj2 = projPointOnLine([L1[0][0], L1[0][1]], [L2[0][0], L2[0][1], L2[1][0], L2[1][1]])
    
    lper1 = np.linalg.norm([L2[0][1] - proj1[1], L2[0][0] - proj1[0]])
    lper2 = np.linalg.norm([L1[0][1] - proj2[1], L1[0][0] - proj2[0]])
    Dper = (math.pow(lper1, 2) + math.pow(lper2, 2)) / (lper1 + lper2)
    
    lpar1 = min(np.linalg.norm([L1[0][1] - proj1[1], L1[0][0] - proj1[0]]), np.linalg.norm([L1[1][1] - proj1[1], L1[1][0] - proj1[0]]))
    lpar2 = min(np.linalg.norm([L1[0][1] - proj2[1], L1[0][0] - proj2[0]]), np.linalg.norm([L1[1][1] - proj2[1], L1[1][0] - proj2[0]]))
    Dpar = min(lpar1, lpar2)
    
    x1, y1 = L2[1][0] - L2[0][0], L2[1][1] - L2[0][1]
    x2, y2 = L1[1][0] - L1[0][0], L1[1][1] - L1[0][1]
    costheta = np.clip((x1 * x2 + y1 * y2) / (np.linalg.norm([x1, y1]) * np.linalg.norm([x2, y2])), -1, 1)
    angle = math.acos(costheta)
    
    Dang = math.sin(angle) * length(L2) if angle < (math.pi / 2) else length(L2)
    
    return w1 * Dper + w2 * Dpar + w3 * Dang + w4 * Dspeed

def length(segment):
    return np.linalg.norm([segment[1][0] - segment[0][0], segment[1][1] - segment[0][1]])

def mdl_par(t, s, e):
    pfactor = 2
    ld = length([t[s], t[e]])
    Dx, Dy = t[e][0] - t[s][0], t[e][1] - t[s][1]
    D = np.linalg.norm([Dx, Dy])
    d = sum(math.fabs((Dy * t[i][0] - Dx * t[i][1] + t[e][0] * t[s][1] - t[e][1] * t[s][0]) / D) for i in range(s, e))
    ldh = d / pfactor
    return ld + ldh

def mdl_nopar(t, s, e):
    return sum(length([t[i], t[i + 1]]) for i in range(s, e))

def mdl_partition(t):
    cp = [t[0]]
    si, l = 1, 1
    while si + l <= len(t):
        ci = si + l
        if mdl_par(t, si, ci) > mdl_nopar(t, si, ci):
            cp.append(t[ci - 1])
            si, l = ci - 1, 1
        else:
            l += 1
    cp.append(t[-1])
    return cp

def partition(T):
    return [[segment, p, 0] for p in T for i in range(len(T[p]) - 1) for segment in [[T[p][i], T[p][i + 1]]]]

def TotalDensity(T, L, D):
    distances = []
    for Li in L:
        for p in T:
            if p != Li[1]:
                for segment in [[T[p][i], T[p][i + 1]] for i in range(len(T[p]) - 1)]:
                    dist = tp_distance(Li[0], segment)
                    if not math.isnan(dist):
                        distances.append(dist)
    sd = standard_deviation(distances)
    tdensity = sum(len([d for d in [tp_distance(Li[0], [T[p][i], T[p][i + 1]]) for i in range(len(T[p]) - 1)] if d <= sd]) + 1 for Li in L for p in T if p != Li[1])
    return tdensity, sd

def detect(T, L, D, P):
    totaldensity, sd = TotalDensity(T, L, D)
    outlier_count = 0
    for Li in L:
        CTR_count = sum(1 for p in T if p != Li[1] and sum(length(segment) for segment in [[T[p][i], T[p][i + 1]] for i in range(len(T[p]) - 1)] if tp_distance(Li[0], segment) < D) > length(Li[0]))
        density = (len([d for d in [tp_distance(Li[0], [T[p][i], T[p][i + 1]]) for i in range(len(T[p]) - 1)] if d <= sd]) + 1) * len(L)
        if (CTR_count * totaldensity) / density < P * len(T):
            Li[2] = 1
            outlier_count += 1
    return L, outlier_count

def mark(T, L, F):
    return [p for p in T if sum(length(seg) for seg in [Li[0] for Li in L if Li[2] == 1 and Li[1] == p]) / sum(length([T[p][i], T[p][i + 1]]) for i in range(len(T[p]) - 1)) > F]

def traod(T, D, P, F):
    print("Partition Phase Begins ...")
    L = partition(T)
    print("Partition Done!")
    print("Total Number of t-partitions: ", len(L))
    print("Outlying t-partition Detection Phase Begins ...")
    L, outlier_count = detect(T, L, D, P)
    print("Outlying t-partition Detection Done!")
    print("Number of Outlying t-partitions: ", outlier_count, " of ", len(L))
    print("Outlying Trajectory Detection Phase Begins ...")
    otraj = mark(T, L, F)
    print("Outlying Trajectory Detection Phase Done!")
    print("Number of Outlying Trajectories: ", len(otraj), " of ", len(T))
    return otraj

def gettime(tr):
    return ((int(tr[:2]) * 3600) + (int(tr[3:5]) * 60) + int(tr[6:8])) / 86400.0

def trajectory(filename, N):
    trajectory = {}
    with open(filename, newline='') as csvfile:
        data = csv.reader(csvfile, delimiter=' ', quotechar='|')
        t1, p1, P = -1, -1, 0
        for row in data:
            r = ', '.join(row).split(";")
            t2 = gettime(r[0].split('T')[1][:-4])
            x2, y2, p2 = int(int(r[2]) / 67), int(int(r[3]) / 67), int(r[4])
            if p1 == p2 and t1 != t2:
                if trajectory[P][-1][0] != x2 or trajectory[P][-1][1] != y2:
                    trajectory[P].append([x2, y2, t2])
            if p1 != p2:
                P += 1
                if P > N:
                    break
                trajectory[P] = [[x2, y2, t2]]
            t1, p1 = t2, p2
    return trajectory

def plot_trajectory(traj, p):
    x, y = [point[0] for point in traj], [point[1] for point in traj]
    fig, ax = plt.subplots()
    ax.plot(x, y)
    ax.set_title(f'TRAJECTORY ID: {p}')
    ax.set_xlabel('x')
    ax.set_ylabel('y')
    plt.show()
    fig.savefig(f'Results/{p}.png')
