import heapq

class PriorityQueue:
    """
    A priority queue implementation using a binary heap.
    
    This class provides a way to manage items with associated priorities. 
    Items with lower priority values are given higher precedence.

    Attributes:
        elements (list): A list that stores the elements of the priority queue as tuples (priority, item).
    """

    def __init__(self):
        """
        Initializes an empty priority queue.
        """
        self.elements = []

    def empty(self):
        """
        Checks if the priority queue is empty.

        Returns:
            bool: True if the priority queue has no elements, False otherwise.
        """
        return len(self.elements) == 0

    def put(self, item, priority):
        """
        Adds an item to the priority queue with a specified priority.

        Args:
            item (any): The item to add to the queue.
            priority (int or float): The priority of the item; lower values denote higher priority.
        """
        heapq.heappush(self.elements, (priority, item))

    def get(self):
        """
        Retrieves and removes the item with the highest priority from the queue.

        Returns:
            any: The item with the highest priority (i.e., the lowest priority value).
        """
        if self.empty():
            raise IndexError("get from an empty priority queue")
        return heapq.heappop(self.elements)[1]

# Example usage
if __name__ == "__main__":
    # Create a new priority queue
    pq = PriorityQueue()
    
    # Add some items with various priorities
    pq.put("task1", priority=5)
    pq.put("task2", priority=1)
    pq.put("task3", priority=3)
    
    # Retrieve and print items according to their priority
    while not pq.empty():
        item = pq.get()
        print(item)
