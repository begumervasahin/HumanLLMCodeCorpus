class Vertex:
    def __init__(self, id):
        """
        Initialize a Vertex with an identifier.

        Parameters:
        id (int or str): The identifier for the vertex.
        """
        self.id = id
        self._adjacent_vertices = {}

    def add_adjacent_vertex(self, vertex, weight):
        """
        Add an adjacent vertex with a given weight.

        Parameters:
        vertex (Vertex): The adjacent vertex to add.
        weight (float): The weight of the edge connecting this vertex to the adjacent vertex.
        """
        self._adjacent_vertices[vertex] = weight

    def __repr__(self):
        """
        Return a string representation of the Vertex.

        Returns:
        str: A string representing the Vertex.
        """
        return f"Vertex(id={self.id})"
