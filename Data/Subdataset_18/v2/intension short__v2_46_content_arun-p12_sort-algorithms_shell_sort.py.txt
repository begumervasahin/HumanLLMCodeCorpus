def shell_sort(A, verbose=0, desc=0):
    def insertion_sort_with_gap(A, gap):
        """Perform insertion sort on elements with a given gap."""
        for i in range(gap, len(A)):
            current_value = A[i]
            position = i

            # Shift earlier gap-sorted elements until the correct location is found
            while position >= gap and A[position - gap] > current_value:
                A[position] = A[position - gap]
                position -= gap

            A[position] = current_value

    n = len(A)
    gap = n // 2  # Start with a large gap, then reduce the gap size

    # Continue reducing the gap until it becomes zero
    while gap > 0:
        if verbose:
            print(f"Current gap: {gap}")

        # Perform a gap-based insertion sort
        insertion_sort_with_gap(A, gap)

        # Reduce the gap for the next iteration
        gap //= 2

    if desc:
        A.reverse()

    return A

# Example usage:
if __name__ == "__main__":
    sample_array = [19, 2, 31, 45, 6, 11, 121, 27]
    sorted_array = shell_sort(sample_array, verbose=1)
    print("Sorted array:", sorted_array)
