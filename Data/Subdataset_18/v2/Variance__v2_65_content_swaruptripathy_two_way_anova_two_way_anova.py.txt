import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error

# Set plot size
plt.rcParams['figure.figsize'] = (20.0, 10.0)

# Load the data
data = pd.read_csv('/Users/swaruptripathy/Desktop/Data Science/headbrain.csv')
print(f'Data Shape: {data.shape}')
print(data.head())

# Extract variables
X = data['Head Size(cm^3)'].values
Y = data['Brain Weight(grams)'].values

# Calculate mean values
mean_x = np.mean(X)
mean_y = np.mean(Y)

# Calculate the coefficients
numer = np.sum((X - mean_x) * (Y - mean_y))
denom = np.sum((X - mean_x) ** 2)
b1 = numer / denom
b0 = mean_y - (b1 * mean_x)
print(f'Coefficients: b1 = {b1}, b0 = {b0}')

# Plot the regression line
max_x = np.max(X) + 100
min_x = np.min(X) - 100
x = np.linspace(min_x, max_x, 1000)
y = b0 + b1 * x

plt.plot(x, y, color='#58b970', label='Regression Line')
plt.scatter(X, Y, c='#ef5423', label='Scatter Plot')
plt.xlabel('Head Size in cm^3')
plt.ylabel('Brain Weight in grams')
plt.legend()
plt.show()

# Calculate RMSE
y_pred = b0 + b1 * X
rmse = np.sqrt(np.mean((Y - y_pred) ** 2))
print(f'RMSE: {rmse}')

# Calculate R2 score
ss_t = np.sum((Y - mean_y) ** 2)
ss_r = np.sum((Y - y_pred) ** 2)
r2 = 1 - (ss_r / ss_t)
print(f'R2 Score: {r2}')

# Using sklearn for linear regression
X = X.reshape((len(X), 1))
reg = LinearRegression().fit(X, Y)
Y_pred = reg.predict(X)
mse = mean_squared_error(Y, Y_pred)
rmse_sklearn = np.sqrt(mse)
r2_score_sklearn = reg.score(X, Y)

print(f'Sklearn RMSE: {rmse_sklearn}')
print(f'Sklearn R2 Score: {r2_score_sklearn}')
