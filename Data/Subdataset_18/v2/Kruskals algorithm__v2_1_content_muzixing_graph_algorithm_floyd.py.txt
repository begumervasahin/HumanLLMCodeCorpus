def floyd(graph):
    """
    Floyd-Warshall algorithm for finding the shortest paths in a graph represented as an adjacency matrix.

    Parameters:
    - graph (list of lists): Adjacency matrix representing the graph, where graph[i][j] is the weight of the edge from node i to node j.

    Returns:
    - tuple: A tuple containing the modified graph with shortest paths and the path reconstruction dictionary.
    """
    length = len(graph)
    path = {}
    
    for i in range(length):
        path[i] = {}
        for j in range(length):
            if i == j:
                continue
            path[i][j] = [i, j]
            for k in range(length):
                if k == j:
                    continue
                new_distance = graph[i][k] + graph[k][j]
                if graph[i][j] > new_distance:
                    graph[i][j] = new_distance
                    path[i][j] = [i, k, j]
    
    return graph, path

def floyd_dict(graph):
    """
    Floyd-Warshall algorithm for finding the shortest paths in a graph represented as a dictionary.

    Parameters:
    - graph (dict): Dictionary representing the graph, where graph[src][dst] is the weight of the edge from node src to node dst.

    Returns:
    - tuple: A tuple containing the modified graph with shortest paths and the path reconstruction dictionary.
    """
    path = {}
    
    for src in graph:
        path[src] = {}
        for dst in graph[src]:
            if src == dst:
                continue
            path[src][dst] = [src, dst]
            for mid in graph:
                if mid == dst:
                    continue
                new_distance = graph[src][mid] + graph[mid][dst]
                if graph[src][dst] > new_distance:
                    graph[src][dst] = new_distance
                    path[src][dst] = [src, mid, dst]
    
    return graph, path
