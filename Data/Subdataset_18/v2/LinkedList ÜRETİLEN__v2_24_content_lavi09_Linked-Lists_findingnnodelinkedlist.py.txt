class Node:
    def __init__(self, data):
        """
        Initialize a node with given data and set the next pointer to None.
        
        Args:
            data: The data to be stored in the node.
        """
        self.data = data
        self.next = None

class LinkedList:
    def __init__(self):
        """Initialize the linked list with head set to None."""
        self.head = None

    def push(self, new_data):
        """
        Push a new node with the given data to the front of the list.
        
        Args:
            new_data: The data to be stored in the new node.
        """
        new_node = Node(new_data)
        new_node.next = self.head
        self.head = new_node

    def print_list(self):
        """Print all the elements in the linked list."""
        temp = self.head
        while temp:
            print(temp.data)
            temp = temp.next

    def find_nth_node_from_end(self, n):
        """
        Find and print the nth node from the end of the linked list.
        
        Args:
            n: The position from the end (1-based index).
        """
        main_ptr = self.head
        ref_ptr = self.head
        count = 0

        # Move ref_ptr n nodes ahead
        while count < n:
            if ref_ptr is None:
                print(f"The list has fewer than {n} elements.")
                return
            ref_ptr = ref_ptr.next
            count += 1

        # Move both pointers until ref_ptr reaches the end
        while ref_ptr is not None:
            main_ptr = main_ptr.next
            ref_ptr = ref_ptr.next

        if main_ptr is not None:
            print(main_ptr.data)
        else:
            print(f"The list has fewer than {n} elements.")

if __name__ == "__main__":
    llist = LinkedList()
    # Push elements to the linked list
    llist.push(6)
    llist.push(5)
    llist.push(4)
    llist.push(3)
    llist.push(2)
    llist.push(1)

    print("The linked list is:")
    llist.print_list()

    print("\nThe 2nd node from the end is:")
    llist.find_nth_node_from_end(2)
