class Edge:
    def __init__(self, vertex1, vertex2, weight):
        self.vertex1 = vertex1
        self.vertex2 = vertex2
        self.weight = weight
        self.selected = False

def return_weight(edge):
    return edge.weight

def run_kruskal(edgelist, node_count):
    # Sort edges based on their weight
    edgelist.sort(key=return_weight)
    
    # Initialize data structures
    count = 0
    sets = []
    edgelist[count].selected = True
    count += 1
    
    sets.append(set())
    sets[0].add(edgelist[0])
    set_count = 1
    
    while count < len(edgelist):
        found_first = False
        found_second = False
        discard = False
        set_index_first = -1
        set_index_second = -1
        
        for j in range(set_count):
            for edge in sets[j]:
                if edge.vertex2 == edgelist[count].vertex1:
                    found_first = True
                    set_index_first = j
                if edge.vertex2 == edgelist[count].vertex2:
                    found_second = True
                    set_index_second = j
                if edge.vertex1 == edgelist[count].vertex1:
                    found_first = True
                    set_index_first = j
                if edge.vertex1 == edgelist[count].vertex2:
                    found_second = True
                    set_index_second = j
        
        if found_first and found_second and set_index_first == set_index_second:
            discard = True
        else:
            if found_first and found_second:
                if set_index_first < set_index_second:
                    sets[set_index_first].update(sets[set_index_second])
                    sets[set_index_first].add(edgelist[count])
                    sets[set_index_second].clear()
                    edgelist[count].selected = True
                else:
                    sets[set_index_second].update(sets[set_index_first])
                    sets[set_index_first].clear()
                    sets[set_index_second].add(edgelist[count])
                    edgelist[count].selected = True
            else:
                if found_first and not found_second:
                    sets[set_index_first].add(edgelist[count])
                    edgelist[count].selected = True
                if not found_first and found_second:
                    sets[set_index_second].add(edgelist[count])
                    edgelist[count].selected = True
        
        if not found_first and not found_second:
            edgelist[count].selected = True
            new_set = set()
            new_set.add(edgelist[count])
            sets.append(new_set)
            set_count += 1
        
        count += 1
    
    return edgelist

# Example usage
edges = [
    Edge(1, 2, 1),
    Edge(2, 3, 2),
    Edge(3, 4, 3),
    Edge(1, 4, 4)
]

node_count = 4

selected_edges = run_kruskal(edges, node_count)

for edge in selected_edges:
    if edge.selected:
        print(f"Edge ({edge.vertex1}, {edge.vertex2}) with weight {edge.weight} is selected")
