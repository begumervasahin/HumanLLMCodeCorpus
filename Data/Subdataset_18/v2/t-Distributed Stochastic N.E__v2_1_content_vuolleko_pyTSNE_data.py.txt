import numpy as np
import struct

def read_MNIST(image_file, label_file):
    """
    Load the MNIST dataset from IDX format files for images and labels.

    Args:
        image_file (str): Path to the file containing the MNIST images in IDX format.
        label_file (str): Path to the file containing the MNIST labels in IDX format.

    Returns:
        tuple: A tuple containing:
            - np.ndarray: Array of images with shape (n_samples, n_rows * n_cols).
            - np.ndarray: Array of labels with shape (n_samples,).
    """
    images = _load_images(image_file)
    labels = _load_labels(label_file)
    return images, labels

def _load_images(image_file):
    """
    Load images from an IDX format file.

    Args:
        image_file (str): Path to the file containing the MNIST images in IDX format.

    Returns:
        np.ndarray: Array of images with shape (n_samples, n_rows * n_cols).
    """
    with open(image_file, 'rb') as f:
        # Skip the magic number (first 4 bytes)
        f.seek(4)

        # Read the number of images, rows, and columns
        n_samples = struct.unpack('>I', f.read(4))[0]
        n_rows = struct.unpack('>I', f.read(4))[0]
        n_cols = struct.unpack('>I', f.read(4))[0]
        n_pixels = n_rows * n_cols

        # Read image data
        images = np.fromfile(f, dtype=np.uint8).reshape(n_samples, n_pixels)

    return images

def _load_labels(label_file):
    """
    Load labels from an IDX format file.

    Args:
        label_file (str): Path to the file containing the MNIST labels in IDX format.

    Returns:
        np.ndarray: Array of labels with shape (n_samples,).
    """
    with open(label_file, 'rb') as f:
        # Skip the magic number and number of items (first 8 bytes)
        f.seek(8)

        # Read label data
        labels = np.fromfile(f, dtype=np.uint8)

    return labels
