class Node:
    def __init__(self, xloc, yloc, kval, adj_list):
        """
        Initializes a Node object with coordinates, a value, and an adjacency list.

        Parameters:
        - xloc (float or int): The x-coordinate of the node.
        - yloc (float or int): The y-coordinate of the node.
        - kval (int): A value associated with the node.
        - adj_list (list): A list of adjacent nodes or relevant data.
        """
        self.xloc = xloc  # x-coordinate of the node
        self.yloc = yloc  # y-coordinate of the node
        self.kval = kval  # A specific value associated with the node
        self.adj_list = adj_list  # List of adjacent nodes or other relevant data

    def __eq__(self, other):
        """
        Determines equality between two Node objects based on their coordinates and value.

        Parameters:
        - other (Node): The other Node object to compare with.

        Returns:
        - bool: True if both nodes have the same xloc, yloc, and kval; False otherwise.
        """
        return (
            self.xloc == other.xloc and 
            self.yloc == other.yloc and 
            self.kval == other.kval
        )

# Example usage
if __name__ == "__main__":
    # Create two nodes with identical attributes
    node1 = Node(1, 2, 3, [])
    node2 = Node(1, 2, 3, [])

    # Check if node1 and node2 are equal
    print(node1 == node2)  # Expected output: True

    # Create a node with different attributes
    node3 = Node(1, 2, 4, [])

    # Check if node1 and node3 are equal
    print(node1 == node3)  # Expected output: False
