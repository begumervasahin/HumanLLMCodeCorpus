def main() -> None:
    """
    Main function to demonstrate bitwise XOR operation between characters.
    """
    char_a = 'A'
    char_space = ' '

    # Print the Unicode code point in hexadecimal for 'A' and 'space'
    print(f"ord('A') = {hex(ord(char_a))}")
    print(f"ord(' ') = {hex(ord(char_space))}")
    
    # Perform XOR operation between 'A' and 'space', then convert back to character
    xor_result = chr(ord(char_a) ^ ord(char_space))
    print(f"chr(ord('A') ^ ord(' ')) = {xor_result}")

if __name__ == "__main__":
    main()
