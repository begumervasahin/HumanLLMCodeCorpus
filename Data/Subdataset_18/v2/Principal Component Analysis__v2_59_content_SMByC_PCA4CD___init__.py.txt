import os
import site
import pkg_resources

def pre_init_plugin_libs_inside():
    """
    This function adds the extra libraries needed by the plugin to the Python path.
    It ensures that any dependencies located in the 'extlibs' directory are available.
    """
    # Determine the path to the 'extlibs' directory
    extra_libs_path = os.path.abspath(os.path.join(os.path.dirname(__file__), "extlibs"))
    
    # If the directory exists, add it to the Python site-packages and pkg_resources
    if os.path.isdir(extra_libs_path):
        site.addsitedir(extra_libs_path)
        pkg_resources.working_set.add_entry(extra_libs_path)

def classFactory(iface):
    """
    This function loads the PCA4CD class, initializing the plugin within QGIS.
    
    :param iface: A QGIS interface instance.
    :type iface: QgsInterface
    """
    # Initialize any additional libraries required by the plugin
    pre_init_plugin_libs_inside()
    
    # Import the PCA4CD class from the 'pca4cd' module
    from .pca4cd import PCA4CD
    
    # Return an instance of the PCA4CD class
    return PCA4CD(iface)
