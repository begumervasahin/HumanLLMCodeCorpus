def is_valid_day(day):
    """
    Checks if the given day is within the valid range (1-31).

    :param day: The day to check.
    :return: True if valid, False otherwise.
    """
    return 1 <= day <= 31

def is_valid_month(month):
    """
    Checks if the given month is within the valid range (1-12).

    :param month: The month to check.
    :return: True if valid, False otherwise.
    """
    return 1 <= month <= 12

def is_valid_year(year):
    """
    Checks if the given year is valid (1582 and later).

    :param year: The year to check.
    :return: True if valid, False otherwise.
    """
    return year >= 1582

def is_leap_year(year):
    """
    Determines if the given year is a leap year.

    :param year: The year to check.
    :return: True if leap year, False otherwise.
    """
    return (year % 400 == 0) or ((year % 4 == 0) and (year % 100 != 0))

def is_valid_date(day, month, year):
    """
    Validates if the provided date is correct.

    :param day: The day of the month.
    :param month: The month of the year.
    :param year: The year.
    :return: None
    """
    if not (is_valid_day(day) and is_valid_month(month) and is_valid_year(year)):
        print("La fecha digitada es inválida.")
        return

    # Number of days in each month
    days_in_month = {
        1: 31, 2: 29 if is_leap_year(year) else 28, 3: 31, 4: 30,
        5: 31, 6: 30, 7: 31, 8: 31, 9: 30, 10: 31, 11: 30, 12: 31
    }
    
    # Check if the day is valid for the given month
    if day > days_in_month.get(month, 0):
        print("La fecha digitada es inválida.")
    else:
        print("La fecha digitada es válida.")

def main():
    """
    Main function to input a date and validate it.
    """
    day = int(input("Digite el día: "))
    month = int(input("Digite el mes: "))
    year = int(input("Digite el año: "))
    is_valid_date(day, month, year)

if __name__ == "__main__":
    main()
