def insertion_sort(arr, verbose=0, descending=False):
    """
    Sorts a list in ascending order using the insertion sort algorithm.
    
    Parameters:
    arr (list): The list to be sorted.
    verbose (int): Level of detail in printed output (0 = none, 1 = iterations, 2 = detailed steps).
    descending (bool): If True, sorts in descending order.
    
    Returns:
    list: The sorted list.
    """
    for i in range(1, len(arr)):
        current_value = arr[i]
        position = i - 1

        # Shift larger elements to the right
        while position >= 0 and arr[position] > current_value:
            arr[position + 1] = arr[position]
            position -= 1
            if verbose == 2:
                print(f"  Moved element at index {position+1}: {arr}")
        
        # Insert the current value in its correct position
        arr[position + 1] = current_value
        
        if verbose >= 1:
            print(f"After iteration {i}: {arr}")
    
    # If descending is True, reverse the sorted list
    if descending:
        arr.reverse()
    
    return arr

# Example usage
if __name__ == "__main__":
    data = [29, 10, 14, 37, 13]
    sorted_data = insertion_sort(data, verbose=1)
    print("Final sorted list:", sorted_data)
