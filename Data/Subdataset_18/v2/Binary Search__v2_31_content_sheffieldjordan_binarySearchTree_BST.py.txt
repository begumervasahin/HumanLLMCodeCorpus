class Node:
    def __init__(self, word):
        self.word = word       # Word stored in the node
        self.left = None       # Pointer to the left child node
        self.right = None      # Pointer to the right child node
        self.count = 1         # Count of occurrences of the word

class BSTree:
    def __init__(self, root=None):
        self.root = root       # Initialize the root of the BST

    def add(self, word):
        """Add a word to the BST."""
        if not self.root:
            self.root = Node(word)
        else:
            self._add(self.root, word)

    def find(self, word):
        """Find the number of occurrences of a word in the BST."""
        return self._find(self.root, word)

    def size(self):
        """Return the number of nodes in the BST."""
        return self._size(self.root)

    def height(self):
        """Return the height of the BST."""
        return self._height(self.root)

    def in_order_print(self):
        """Print the words in the BST in in-order traversal."""
        self._in_order_print(self.root)

    def _add(self, root, word):
        """Helper method to add a word to the BST."""
        if root.word == word:
            root.count += 1  # Increment count if word already exists
        elif word < root.word:
            if root.left is None:
                root.left = Node(word)  # Add a new node to the left
            else:
                self._add(root.left, word)
        else:
            if root.right is None:
                root.right = Node(word)  # Add a new node to the right
            else:
                self._add(root.right, word)

    def _find(self, root, word):
        """Helper method to find a word in the BST."""
        if root is None:
            return 0  # Word not found
        if root.word == word:
            return root.count
        elif word < root.word:
            return self._find(root.left, word)
        else:
            return self._find(root.right, word)

    def _size(self, root):
        """Helper method to calculate the size of the BST."""
        if root is None:
            return 0
        return 1 + self._size(root.left) + self._size(root.right)

    def _height(self, root):
        """Helper method to calculate the height of the BST."""
        if root is None:
            return 0
        left_height = self._height(root.left)
        right_height = self._height(root.right)
        return 1 + max(left_height, right_height)

    def _in_order_print(self, root):
        """Helper method for in-order traversal and printing of the BST."""
        if root is None:
            return
        self._in_order_print(root.left)
        print(f'{root.word}: {root.count}')
        self._in_order_print(root.right)

# Example usage
if __name__ == "__main__":
    bst = BSTree()
    
    words = ["apple", "banana", "apple", "orange", "banana", "apple", "orange", "orange"]
    
    for word in words:
        bst.add(word)
    
    print("In-order traversal of the BST:")
    bst.in_order_print()

    print("\nSize of the BST:", bst.size())
    print("Height of the BST:", bst.height())

    word_to_find = "apple"
    print(f"\nOccurrences of '{word_to_find}':", bst.find(word_to_find))
