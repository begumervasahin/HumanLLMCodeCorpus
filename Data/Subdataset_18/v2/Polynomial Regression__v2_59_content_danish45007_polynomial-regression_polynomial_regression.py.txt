import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import PolynomialFeatures

# Load the dataset
dataset = pd.read_csv('position_salaries.csv')
X = dataset.iloc[:, 1:2].values  # Extracting the position levels
y = dataset.iloc[:, 2].values    # Extracting the salaries

# Fitting a Linear Regression model to the dataset
lin_reg = LinearRegression()
lin_reg.fit(X, y)

# Fitting a Polynomial Regression model to the dataset with degree 4
poly_reg = PolynomialFeatures(degree = 4)
X_poly = poly_reg.fit_transform(X)
lin_reg_2 = LinearRegression()
lin_reg_2.fit(X_poly, y)

# Visualizing the results of the Linear Regression model
plt.figure(figsize=(10, 6))
plt.scatter(X, y, color = 'red')
plt.plot(X, lin_reg.predict(X), color = 'blue')
plt.title('Truth or Bluff (Linear Regression)')
plt.xlabel('Position level')
plt.ylabel('Salary')
plt.show()

# Visualizing the results of the Polynomial Regression model
plt.figure(figsize=(10, 6))
plt.scatter(X, y, color = 'red')
plt.plot(X, lin_reg_2.predict(poly_reg.fit_transform(X)), color = 'blue')
plt.title('Truth or Bluff (Polynomial Regression)')
plt.xlabel('Position level')
plt.ylabel('Salary')
plt.show()

# Generating a smoother curve for the Polynomial Regression model
X_grid = np.arange(min(X), max(X), 0.1)  # Creating a range of values with a step of 0.1
X_grid = X_grid.reshape((len(X_grid), 1))  # Reshaping to the correct format
plt.figure(figsize=(10, 6))
plt.scatter(X, y, color = 'red')
plt.plot(X_grid, lin_reg_2.predict(poly_reg.fit_transform(X_grid)), color = 'blue')
plt.title('Truth or Bluff (Polynomial Regression)')
plt.xlabel('Position level')
plt.ylabel('Salary')
plt.show()

# Save the final plot to a file
plt.savefig('position_salary.png')
