class Node:
    def __init__(self, val=-1, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

class ReturnType:
    def __init__(self, size, head, mins, maxs):
        self.size = size  # Size of the largest BST
        self.head = head  # Root of the largest BST
        self.mins = mins  # Minimum value in the subtree
        self.maxs = maxs  # Maximum value in the subtree

def process(head):
    if head is None:
        return ReturnType(0, None, float('inf'), float('-inf'))
    
    leftSubTreeInfo = process(head.left)
    rightSubTreeInfo = process(head.right)
    
    includeSelf = 0
    if (leftSubTreeInfo.head == head.left and 
        rightSubTreeInfo.head == head.right and 
        head.val > leftSubTreeInfo.maxs and 
        head.val < rightSubTreeInfo.mins):
        includeSelf = leftSubTreeInfo.size + 1 + rightSubTreeInfo.size
    
    maxsize = max(leftSubTreeInfo.size, rightSubTreeInfo.size, includeSelf)
    
    if leftSubTreeInfo.size > rightSubTreeInfo.size:
        maxhead = leftSubTreeInfo.head
    else:
        maxhead = rightSubTreeInfo.head
    
    if maxsize == includeSelf:
        maxhead = head
    
    return ReturnType(
        maxsize, 
        maxhead, 
        min(leftSubTreeInfo.mins, rightSubTreeInfo.mins, head.val), 
        max(leftSubTreeInfo.maxs, rightSubTreeInfo.maxs, head.val)
    )

def printTree(root):
    print("Binary Tree:")
    printInOrder(root, 0, 'H', 17)

def printInOrder(root, height, s, length):
    if root is None:
        return
    printInOrder(root.right, height + 1, 'v', length)
    val = s + str(root.val) + s
    lenM = len(val)
    lenL = (length - lenM) // 2
    lenR = length - lenM - lenL
    val = getSpace(lenL) + val + getSpace(lenR)
    print(getSpace(height * length) + val)
    printInOrder(root.left, height + 1, '^', length)

def getSpace(num):
    return ' ' * num
