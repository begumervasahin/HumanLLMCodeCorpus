# Define the file paths for input and output
input_file_path = "tvshow_list_unsorted.txt"
output_file_path = "tvshow_list_sorted.txt"

# Initialize an empty list to store TV show tuples
tv_shows = []

# Read the unsorted list of TV shows from the input file
with open(input_file_path, "r") as input_file:
    for line in input_file:
        # Split each line by the underscore and strip newline characters
        show_name, show_detail = line.strip().split('_')
        # Append the tuple (show_name, show_detail) to the list
        tv_shows.append((show_name, show_detail))

# Sort the list first by the second part (show_detail) and then by the first part (show_name) of the tuple
tv_shows.sort(key=lambda show: (show[1], show[0]))

# Write the sorted list to the output file
with open(output_file_path, 'w') as output_file:
    for show in tv_shows:
        output_file.write(f"{show[0]}_{show[1]}\n")
