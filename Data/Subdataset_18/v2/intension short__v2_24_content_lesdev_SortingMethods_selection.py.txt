import tkinter as tk
from tkinter import simpledialog

def select_numbers():
    try:
        # Retrieve the number of elements to input
        num_elements = int(entry_numeros.get())
        numbers = []

        # Prompt user to enter the specified number of elements
        for _ in range(num_elements):
            num = simpledialog.askinteger("Input", "Ingresa el número", parent=app)
            if num is not None:
                numbers.append(num)

        # Display the list of numbers entered
        Label2.config(text=f"Números ingresados: {numbers}")

        # Perform selection sort on the list
        for i in range(len(numbers) - 1):
            min_index = i
            for j in range(i + 1, len(numbers)):
                if numbers[j] < numbers[min_index]:
                    min_index = j
            # Swap the found minimum element with the first element
            numbers[i], numbers[min_index] = numbers[min_index], numbers[i]

        # Display the sorted list
        label3.config(text=f"Números ordenados: {numbers}")

    except ValueError:
        Label2.config(text="Por favor ingresa un número válido.")

# Initialize the main application window
app = tk.Tk()
app.title("Selection Sort")
app.configure(bg='SkyBlue2')

# Label to ask for the number of elements
Label1 = tk.Label(app, text="¿Cuántos números vas a ingresar?", font='Helvetica 12', bg='SkyBlue2')
Label1.pack()

# Entry widget to input the number of elements
entry_numeros = tk.Entry(app, width=8)
entry_numeros.pack()

# Button to trigger the selection process
button1 = tk.Button(app, text="Ok!", command=select_numbers)
button1.pack(pady=(10, 0))

# Labels to display the entered and sorted numbers
Label2 = tk.Label(app, text="", bg='SkyBlue2')
Label2.pack(pady=(10, 0))

label3 = tk.Label(app, text="", bg='SkyBlue2')
label3.pack(pady=(10, 0))

# Run the application loop
app.mainloop()
