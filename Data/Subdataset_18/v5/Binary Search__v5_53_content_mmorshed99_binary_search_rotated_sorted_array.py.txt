class Solution:
    def search(self, A, B):
        def find_break_point(start, end, A):
            """Finds the index of the smallest element (break point) in a rotated sorted array."""
            while start <= end:
                # If the array is already sorted, return the first element
                if A[start] <= A[end]:
                    return start
                
                mid = start + (end - start) // 2

                # Check if mid is the break point
                if A[mid] > A[mid + 1]:
                    return mid + 1
                if A[mid] < A[mid - 1]:
                    return mid
                
                # Decide which side to search next
                if A[mid] > A[end]:
                    start = mid + 1
                else:
                    end = mid - 1
            
            return -1  # Should never be reached if the array is properly rotated

        def binary_search(start, end, target, A):
            """Performs binary search for the target in the subarray A[start:end+1]."""
            while start <= end:
                mid = start + (end - start) // 2
                
                # Check if the target is found
                if A[mid] == target:
                    return mid
                
                # Decide which side to search next
                if A[mid] < target:
                    start = mid + 1
                else:
                    end = mid - 1
            
            return -1  # Target not found

        # Find the break point in the rotated sorted array
        break_point = find_break_point(0, len(A) - 1, A)
        
        # If no break point is found, the array is not rotated; perform binary search on the entire array
        if break_point == -1:
            return binary_search(0, len(A) - 1, B, A)
        
        # Determine which subarray to search
        if B >= A[0] and B <= A[break_point - 1]:
            return binary_search(0, break_point - 1, B, A)
        else:
            return binary_search(break_point, len(A) - 1, B, A)
