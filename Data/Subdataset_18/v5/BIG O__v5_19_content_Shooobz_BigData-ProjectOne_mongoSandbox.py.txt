import pymongo
import csv

# Database setup
client = pymongo.MongoClient()
db = client["projectOne"]
users = db["Users"]
organizations = db["Organizations"]
interests = db["Interests"]
projects = db["Projects"]
skills = db["Skills"]

def main():
    while True:
        val = input("\nWould you like to search for a user (y/n): ").strip().lower()
        if val == 'y':
            user_id = input("Enter user ID: ").strip()
            look_up_user(user_id)
        else:
            break

def look_up_user(user_id):
    """
    Look up and display user information based on the user ID.

    Args:
        user_id (str): The user ID to look up.
    """
    if users.count_documents({"User_id": user_id}) == 0:
        print("This user does not exist. Please try another user ID.")
        return

    user_results = users.find({"User_id": user_id})
    organization_results = organizations.find({"User_id": user_id})
    interest_results = interests.find({"User_id": user_id})
    skill_results = skills.find({"User_id": user_id})
    project_results = projects.find({"User_id": user_id})

    for user in user_results:
        print(f"\tName: {user['First name']} {user['Last name']}")

    for org in organization_results:
        print(f"\tWorks at: {org['organization']}")

    interest_collection = gather_interest_or_skill(interest_results, "Interest", "Interest level")
    print(f"\tInterests: {interest_collection}")

    skill_collection = gather_interest_or_skill(skill_results, "Skill", "Skill level")
    print(f"\tSkills: {skill_collection}")

    project_collection = gather_projects(project_results)
    print(f"\tWorks on: {project_collection}")

def gather_interest_or_skill(results, key, level_key):
    """
    Helper function to gather interests or skills from results.

    Args:
        results: The results from the database query.
        key: The key for interest or skill.
        level_key: The key for the level of interest or skill.

    Returns:
        str: A formatted string of interests or skills.
    """
    collection = ""
    for item in results:
        if isinstance(item[key], list):
            for i in range(len(item[key])):
                collection += f"{item[key][i]} ({item[level_key][i]}), "
        else:
            collection += f"{item[key]} ({item[level_key]})"
    return collection.strip(', ')

def gather_projects(results):
    """
    Helper function to gather projects from results.

    Args:
        results: The results from the database query.

    Returns:
        str: A formatted string of projects.
    """
    collection = ""
    for project in results:
        if isinstance(project["Project"], list):
            for p in project["Project"]:
                collection += f"{p}, "
        else:
            collection += project["Project"]
    return collection.strip(', ')

def read_file(file_name):
    """
    Read a CSV file and return a list of dictionaries representing the data.

    Args:
        file_name (str): The name of the CSV file to read.

    Returns:
        list: A list of dictionaries containing the data.
    """
    with open(file_name, mode='r') as file:
        csv_file = csv.reader(file)
        headers = next(csv_file)
        data_collection = [{headers[i]: row[i] for i in range(len(headers))} for row in csv_file]
    return data_collection

def insert_data(collection, entries):
    """
    Insert data into a MongoDB collection.

    Args:
        collection (pymongo.collection.Collection): The MongoDB collection.
        entries (list): A list of dictionaries to be inserted into the collection.
    """
    collection.create_index("User_id", unique=True)
    for data in entries:
        try:
            collection.insert_one(data)
        except pymongo.errors.DuplicateKeyError:
            print(f"Duplicate entry found for User_id: {data['User_id']}")

def load_files(csv_files):
    """
    Load data from CSV files into the respective MongoDB collections.

    Args:
        csv_files (list): A list of CSV file paths.
    """
    insert_data(users, read_file(csv_files[0]))
    insert_data(organizations, read_file(csv_files[1]))
    insert_data(projects, read_int_or_skill(csv_files[2], "project"))
    insert_data(interests, read_int_or_skill(csv_files[3], "interest"))
    insert_data(skills, read_int_or_skill(csv_files[4], "skill"))

def read_int_or_skill(file_name, tag):
    """
    Read a CSV file for interests or skills and return a list of dictionaries representing the data.

    Args:
        file_name (str): The name of the CSV file to read.
        tag (str): The tag indicating whether the data is 'interest' or 'skill'.

    Returns:
        list: A list of dictionaries containing the data.
    """
    with open(file_name, mode='r') as file:
        csv_file = csv.reader(file)
        headers = next(csv_file)
        data_collection = []
        index = [("User_id", pymongo.ASCENDING), (tag.capitalize(), pymongo.DESCENDING)]
        if tag == "interest":
            interests.create_index(index, unique=True)
        elif tag == "skill":
            skills.create_index(index, unique=True)

        for row in csv_file:
            data = {headers[i]: row[i] for i in range(len(headers))}
            if find_matching_id(data, data_collection, tag):
                data_collection.append(data)
    return data_collection

def find_matching_id(info, data_list, tag):
    """
    Find matching user ID and update the list accordingly for interests, skills, or projects.

    Args:
        info (dict): The current data entry.
        data_list (list): The list of existing data entries.
        tag (str): The tag indicating the type of data ('interest', 'skill', or 'project').

    Returns:
        bool: True if the user ID was not found and added as new entry, False if updated an existing entry.
    """
    for element in data_list:
        if element["User_id"] == info["User_id"]:
            key = tag.capitalize()
            if not isinstance(element[key], list):
                element[key] = [element[key], info[key]]
                element[f"{key} level"] = [element[f"{key} level"], info[f"{key} level"]]
            else:
                if info[key] not in element[key]:
                    element[key].append(info[key])
                    element[f"{key} level"].append(info[f"{key} level"])
            return False
    return True

def drop_db():
    """
    Drop the 'projectOne' database.
    """
    client.drop_database("projectOne")

if __name__ == "__main__":
    main()
