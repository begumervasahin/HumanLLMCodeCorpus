# Utility functions
def get_cell_size(DEM):
    """Extracts and returns the cell size from the DEM name."""
    return int(DEM.split('_')[-1])

def get_sample_area(DEM):
    """Determines and returns the sample area based on the DEM name."""
    return "SampleArea" if DEM.startswith("sa_") else DEM.split('_')[0]

def get_input_mxd_path(SA):
    """Returns the path to the MXD file for the given sample area."""
    return os.path.join(workspace_dirs['imageDataWS'], f"{SA}.mxd")

def get_source_dem(DEM):
    """Determines and returns the source DEM type based on the DEM name."""
    code_source_dem = DEM.split('_')[1]
    source_dem_map = {
        "zm": "ZM50",
        "lls": "LLS",
        "d4": "DMR4G",
        "d5": "DMR5G",
        "d": "DMU25",
        "as": "ASTER",
        "sr": "SRTM"
    }
    return source_dem_map.get(code_source_dem, code_source_dem)

def get_buffer_size_cluster():
    """Returns the minimum and maximum buffer sizes for clustering."""
    return [100, 200]

def get_rotation_angle(DEM):
    """Extracts and returns the rotation angle from the DEM name."""
    rotation_part = DEM.split('_r')[-1]
    return int(rotation_part.split('_')[0])

def get_hs_azimuth(shp_name):
    """Extracts and returns the HS azimuth from the shapefile name."""
    return shp_name.split('_')[1]
