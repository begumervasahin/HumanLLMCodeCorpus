import random
import fileOp

def generate_random_n_bit_number(n):
    """
    Generate a random integer with n bits.
    """
    return random.randrange(2**(n-1) + 1, 2**n)

def generate_prime_candidate(n):
    """
    Generate a candidate for an n-bit prime number.
    """
    try:
        first_primes = fileOp.read_list("FfirstPrimes")
    except Exception as e:
        raise RuntimeError("Could not read first primes from file") from e
    
    while True:
        candidate = generate_random_n_bit_number(n)
        if is_prime(candidate, first_primes):
            return candidate

def is_prime(candidate, first_primes):
    """
    Check if the candidate is a prime number using trial division by small primes.
    """
    for prime in first_primes:
        if candidate % prime == 0:
            return False
        if prime > candidate ** 0.5:
            return True
    return True

def miller_rabin_primality_test(n, k=20):
    """
    Perform the Miller-Rabin primality test with k iterations.
    """
    if n in {0, 1, 4, 6, 8, 9}:
        return False
    if n in {2, 3, 5, 7}:
        return True

    s, d = 0, n - 1
    while d % 2 == 0:
        d //= 2
        s += 1

    def is_composite(a):
        if pow(a, d, n) == 1:
            return False
        for i in range(s):
            if pow(a, 2**i * d, n) == n - 1:
                return False
        return True

    for _ in range(k):
        a = random.randrange(2, n)
        if is_composite(a):
            return False
    
    return True

def generate_prime(n):
    """
    Generate a prime number of n bits using both trial division and Miller-Rabin test.
    """
    while True:
        candidate = generate_prime_candidate(n)
        if miller_rabin_primality_test(candidate):
            return candidate

def fermat_primality_test(p, base=2):
    """
    Perform the Fermat primality test with the given base (default is 2).
    """
    return pow(base, p - 1, p) == 1
