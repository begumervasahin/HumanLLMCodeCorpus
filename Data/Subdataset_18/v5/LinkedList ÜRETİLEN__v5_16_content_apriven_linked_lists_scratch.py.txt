class Hedgehog:
    def __init__(self, val):
        self.value = val
        self.next = None


class HedgehogList:
    def __init__(self):
        self.head = None

    def add_first(self, number):
        """Add a new Hedgehog at the beginning of the list."""
        new_hedgehog = Hedgehog(number)
        new_hedgehog.next = self.head
        self.head = new_hedgehog

    def add_last(self, number):
        """Add a new Hedgehog at the end of the list."""
        new_hedgehog = Hedgehog(number)
        if not self.head:
            self.head = new_hedgehog
        else:
            current = self.head
            while current.next:
                current = current.next
            current.next = new_hedgehog

    def get_length(self):
        """Return the number of Hedgehogs in the list."""
        current = self.head
        length = 0
        while current:
            length += 1
            current = current.next
        return length

    def is_empty(self):
        """Check if the list is empty."""
        return self.head is None

    def find_max(self):
        """Find the maximum value in the list."""
        if not self.head:
            return None
        max_value = self.head.value
        current = self.head.next
        while current:
            if current.value > max_value:
                max_value = current.value
            current = current.next
        return max_value

    def remove(self, number):
        """Remove the first occurrence of the specified value from the list."""
        if not self.head:
            return None
        if self.head.value == number:
            removed_hedgehog = self.head
            self.head = self.head.next
            return removed_hedgehog
        current = self.head
        while current.next and current.next.value != number:
            current = current.next
        if current.next:
            removed_hedgehog = current.next
            current.next = current.next.next
            return removed_hedgehog
        return None


def test_list(actual, expected):
    """Test if the list matches the expected values."""
    actual_item = actual
    for index, expected_number in enumerate(expected, start=1):
        actual_value = actual_item.value if actual_item else None
        assert actual_item and actual_value == expected_number, (
            f"Element number {index} should have the value of {expected_number}, got {actual_value}"
        )
        actual_item = actual_item.next


def test_length(linked_list, expected_length):
    """Test if the length of the list matches the expected length."""
    assert linked_list.get_length() == expected_length, (
        f"Length should be {expected_length}, got {linked_list.get_length()}"
    )


def test_value(actual, expected, msg=""):
    """Test if the actual value matches the expected value."""
    assert actual == expected, f"{msg} should be {expected}, got {actual}"


def test_max(linked_list, expected):
    """Test if the maximum value in the list matches the expected value."""
    max_value = linked_list.find_max()
    test_value(max_value, expected, "Wrong maximum")


def create_list(numbers):
    """Create a list from the provided numbers."""
    linked_list = HedgehogList()
    for number in numbers:
        linked_list.add_last(number)
    return linked_list
