import sys
import os

# Define stop words and create a list from the comma-separated string
STOP_WORDS = (
    "a,able,about,across,after,all,almost,also,am,among,an,and,any,are,as,at,"
    "be,because,been,but,by,can,cannot,could,dear,did,do,does,either,else,ever,"
    "every,for,from,get,got,had,has,have,he,her,hers,him,his,how,however,i,if,"
    "in,into,is,it,its,just,least,let,like,likely,may,me,might,most,must,my,"
    "neither,no,nor,not,of,off,often,on,only,or,other,our,own,rather,said,say,"
    "says,she,should,since,so,some,than,that,the,their,them,then,there,these,"
    "they,this,tis,to,too,twas,us,wants,was,we,were,what,when,where,which,while,"
    "who,whom,why,will,with,would,yet,you,your"
)
STOP_WORDS_LIST = set(STOP_WORDS.split(","))  # Use set for faster lookup

def get_file_name_from_path(file_path):
    """Extract the file name from the given file path."""
    _, file_name = os.path.split(file_path)
    return file_name

def is_not_stop_word(word, stop_words):
    """Check if a word is not in the stop words list."""
    return word.lower() not in stop_words

def process_input(file_name):
    """Process each line from standard input and print non-stop words."""
    for line in sys.stdin:
        words = line.strip().split()
        for word in words:
            if is_not_stop_word(word, STOP_WORDS_LIST):
                print(f'{word}&{file_name}\t1')

def main():
    """Main function to execute the script."""
    file_path = os.getenv('map_input_file')
    if not file_path:
        print("Error: 'map_input_file' environment variable not set.", file=sys.stderr)
        sys.exit(1)

    file_name = get_file_name_from_path(file_path)
    process_input(file_name)

if __name__ == "__main__":
    main()
