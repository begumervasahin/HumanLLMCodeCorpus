def longest_subsequence_Onlogn(arr):
    """
    Find the longest increasing subsequence in O(n log n) time complexity.
    
    Args:
        arr (list): A list of integers.
        
    Returns:
        list: The longest increasing subsequence.
    """
    if len(arr) < 2:
        return arr
    
    M = [None] * len(arr)
    P = [None] * len(arr)
    L = 1
    M[0] = 0
    
    for i in range(1, len(arr)):
        if arr[i] > arr[M[L - 1]]:
            j = L
        else:
            lo, hi = 0, L - 1
            while lo + 1 < hi:
                mid = (lo + hi) // 2
                if arr[i] > arr[M[mid]]:
                    lo = mid + 1
                else:
                    hi = mid
            j = lo
        
        P[i] = M[j - 1]
        M[j] = i
        L = max(L, j + 1)
    
    results = []
    k = M[L - 1]
    for _ in range(L):
        results.append(arr[k])
        k = P[k]
    
    return results[::-1]

def longest_subsequence_On2(arr):
    """
    Find the longest increasing subsequence in O(n^2) time complexity.
    
    Args:
        arr (list): A list of integers.
        
    Returns:
        list: The longest increasing subsequence.
    """
    if not arr:
        return []
    
    mem = [[] for _ in range(len(arr))]
    best = []
    
    for i in range(len(arr)):
        for j in range(i):
            if arr[i] > arr[j]:
                if len(mem[j]) + 1 > len(mem[i]):
                    mem[i] = mem[j] + [arr[i]]
                    best = max(mem[i], best, key=len)
        if not mem[i]:
            mem[i] = [arr[i]]
            best = max(mem[i], best, key=len)
    
    return best

    print("Longest increasing subsequence (O(n log n)):", longest_subsequence_Onlogn(arr1))
