def quicksort(alist):
    """
    Perform an in-place quicksort on the given list.

    Parameters:
    alist (list): The list of elements to be sorted.
    """
    quicksort_helper(alist, 0, len(alist) - 1)

def quicksort_helper(alist, first, last):
    """
    Helper function to perform quicksort recursively.

    Parameters:
    alist (list): The list of elements to be sorted.
    first (int): The starting index of the sublist to be sorted.
    last (int): The ending index of the sublist to be sorted.
    """
    if first < last:
        split_point = partition(alist, first, last)
        quicksort_helper(alist, first, split_point - 1)
        quicksort_helper(alist, split_point + 1, last)

def partition(alist, first, last):
    """
    Partition the list around a pivot element.

    Parameters:
    alist (list): The list of elements to be sorted.
    first (int): The starting index of the sublist to be partitioned.
    last (int): The ending index of the sublist to be partitioned.

    Returns:
    int: The index of the pivot element after partitioning.
    """
    pivot_value = alist[first]
    left_mark = first + 1
    right_mark = last

    while True:
        # Move left_mark to the right as long as alist[left_mark] is less than or equal to the pivot value
        while left_mark <= right_mark and alist[left_mark] <= pivot_value:
            left_mark += 1

        # Move right_mark to the left as long as alist[right_mark] is greater than or equal to the pivot value
        while right_mark >= left_mark and alist[right_mark] >= pivot_value:
            right_mark -= 1

        # If the left_mark crosses the right_mark, we are done partitioning
        if right_mark < left_mark:
            break
        else:
            # Swap the elements at left_mark and right_mark
            alist[left_mark], alist[right_mark] = alist[right_mark], alist[left_mark]

    # Swap the pivot element with the element at right_mark
    alist[first], alist[right_mark] = alist[right_mark], alist[first]

    return right_mark
