import codecs

def create_crc_table():
    """
    Create the CRC table for efficient computation of CRC checksums.
    :return: The CRC table as a list of integers.
    """
    crc_table = []
    polynomial = 0xEDB88320

    for byte in range(256):
        remainder = byte
        for _ in range(8):
            if remainder & 1:
                remainder = (remainder >> 1) ^ polynomial
            else:
                remainder >>= 1
        crc_table.append(remainder)

    return crc_table

def calculate_crc_checksum(data, crc_table):
    """
    Calculate the CRC checksum for the given data.
    :param data: The data for which the CRC checksum is to be computed.
    :param crc_table: The CRC table.
    :return: The computed CRC checksum.
    """
    crc = 0xFFFFFFFF
    for byte in data:
        crc = (crc >> 8) ^ crc_table[(crc & 0xFF) ^ byte]
    return crc ^ 0xFFFFFFFF

def hex_to_binary_string(hex_string):
    """
    Convert a hexadecimal string to a binary string without the '0b' prefix.
    :param hex_string: The hexadecimal string.
    :return: The binary string.
    """
    return bin(int(hex_string, 16))[2:]

def main():
    # Define the message in hexadecimal
    message_hex = 'FAFAFA'
    
    # Convert the message from hexadecimal to binary (without the '0b' prefix)
    message_bin = hex_to_binary_string(message_hex)
    print(f"Input message: {message_bin} (0x{message_hex})")

    # Create the CRC table
    crc_table = create_crc_table()
    
    # Decode the message from hexadecimal to bytes
    message_bytes = codecs.decode(message_hex, 'hex')
    
    # Calculate the CRC checksum
    checksum = calculate_crc_checksum(message_bytes, crc_table)
    
    # Convert the checksum to binary (without the '0b' prefix) and ensure 32-bit representation
    checksum_bin = bin(checksum)[2:].zfill(32)
    print(f"Checksum: {checksum_bin} ({hex(checksum)})")

    # Combine the message and the checksum in binary form
    send_message = message_bin + checksum_bin
    print("Message to be sent:")
    print(send_message)

if __name__ == "__main__":
    main()
