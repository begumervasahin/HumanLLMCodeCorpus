import os
import sys
from gui.app import Application

def setup_environment():
    """
    Set up the environment by adding the script's directory to the system path.
    
    Returns:
        tuple: A tuple containing the script's absolute path and directory.
    """
    script_path = os.path.abspath(__file__)
    script_directory = os.path.dirname(script_path)
    sys.path.append(script_directory)
    return script_path, script_directory

def print_initialization_info(script_path, script_directory):
    """
    Print initialization information to the console.
    
    Args:
        script_path (str): The absolute path of the main script.
        script_directory (str): The directory containing the main script.
    """
    print("[ START PROGRAM ]")
    print(f"Start main file: {script_path}")
    print(f"System path appended: {script_directory}")

def main():
    """
    Main function to initialize and run the application.
    """
    # Set up the environment and retrieve paths
    script_path, script_directory = setup_environment()
    
    # Print initialization messages
    print_initialization_info(script_path, script_directory)
    
    # Initialize and run the application
    app = Application(script_directory)
    app.mainloop()

if __name__ == "__main__":
    main()
