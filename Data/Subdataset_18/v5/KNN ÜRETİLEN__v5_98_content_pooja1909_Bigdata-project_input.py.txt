import csv
from collections import defaultdict

def load_csv(input_filename, output_filename='output.csv'):
    """
    Load data from an input CSV file, process specific columns, and write to an output CSV file.

    Parameters:
    input_filename (str): The name of the input CSV file.
    output_filename (str): The name of the output CSV file. Default is 'output.csv'.
    """
    # Open the output file for writing
    with open(output_filename, 'w', newline='') as clusterfile:
        writer = csv.writer(clusterfile)

        # Open the input file for reading
        with open(input_filename, 'r', newline='') as f:
            reader = csv.reader(f)
            headers = next(reader)

            # Write the required columns headers to the output file
            required_columns = ["Lon", "Lat", "Text_General_Code"]
            writer.writerow([headers[12], headers[13], headers[9]])

            # Initialize a dictionary to store columns
            columns = defaultdict(list)

            # Read the input file as a dictionary and populate the columns dictionary
            reader = csv.DictReader(f, fieldnames=headers)
            for row in reader:
                for k, v in row.items():
                    columns[k].append(v)

            # Write the specific columns to the output file
            for i in range(len(columns["Lon"])):
                writer.writerow([columns["Lon"][i], columns["Lat"][i], columns["Text_General_Code"][i]])

def main():
    input_filename = 'input.csv'
    load_csv(input_filename)

if __name__ == "__main__":
    main()
