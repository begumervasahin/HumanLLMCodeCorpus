import random

def quicksort(lst):
    """
    Sorts a list using the quicksort algorithm.
    
    Parameters:
    lst (list): The list of elements to be sorted.
    
    Returns:
    list: A new list containing the sorted elements.
    """
    # Base case: if the list is empty or contains a single element, it is already sorted
    if len(lst) <= 1:
        return lst
    
    # Select a random pivot element from the list
    pivot_index = random.randint(0, len(lst) - 1)
    pivot = lst[pivot_index]
    
    # Print the current pivot and list for debugging
    print(f"Pivoting {pivot} in {lst}")
    
    # Partition the list into elements less than, greater than, and equal to the pivot
    left = []
    right = []
    pivot_clones = []
    
    for i, value in enumerate(lst):
        if i == pivot_index:
            pivot_clones.append(value)
        elif value < pivot:
            left.append(value)
        elif value > pivot:
            right.append(value)
        else:
            pivot_clones.append(value)
    
    # Print the partitions for debugging
    print(f"Left: {left}; Right: {right}; Pivot clones: {pivot_clones}")
    
    # Recursively sort the left and right partitions and combine with pivot clones
    return quicksort(left) + pivot_clones + quicksort(right)

# Example usage
lst = [3, 6, 8, 10, 1, 2, 1]
sorted_lst = quicksort(lst)
print("Sorted result:", sorted_lst)
