import numpy as np

def load_data(filename: str) -> np.array:
    """
    Loads the dataset from a file and returns it as a NumPy array.
    
    Parameters:
    - filename (str): The name of the file containing the dataset.

    Returns:
    - np.array: A NumPy array with the dataset, with an additional column for cluster assignment initialized to 0.
    """
    data = []
    with open(filename, "r") as file:
        lines = file.readlines()
        for line in lines:
            # Convert each line into a list of floats and append a zero for cluster assignment
            data.append([float(value) for value in line.strip().split('\t')] + [0])
    return np.asarray(data)

def compute_error(data: np.array, centroids: np.array) -> float:
    """
    Computes the average error of the clustering.
    
    Parameters:
    - data (np.array): The dataset with cluster assignments.
    - centroids (np.array): The array of centroids.

    Returns:
    - float: The average error of the clustering.
    """
    total_error = 0.0
    for point in data:
        centroid_idx = int(point[-1])
        total_error += np.linalg.norm(point[:-1] - centroids[centroid_idx])
    return total_error / data.shape[0]

def calculate_mean(data: np.array, centroids: np.array) -> np.array:
    """
    Recalculates the centroids based on the current cluster assignments.
    
    Parameters:
    - data (np.array): The dataset with cluster assignments.
    - centroids (np.array): The array of current centroids.

    Returns:
    - np.array: The updated centroids.
    """
    new_centroids = np.zeros_like(centroids)
    counts = np.zeros(centroids.shape[0])

    for point in data:
        cluster_id = int(point[-1])
        new_centroids[cluster_id] += point[:-1]
        counts[cluster_id] += 1

    for i in range(centroids.shape[0]):
        if counts[i] > 0:
            new_centroids[i] /= counts[i]  # Avoid division by zero

    return new_centroids

def group_data(data: np.array, centroids: np.array) -> np.array:
    """
    Assigns each data point in the dataset to the nearest cluster based on Euclidean distance.
    
    Parameters:
    - data (np.array): The dataset.
    - centroids (np.array): The array of centroids.

    Returns:
    - np.array: The dataset with updated cluster assignments.
    """
    for i in range(data.shape[0]):
        point = data[i][:-1]
        distances = np.linalg.norm(point - centroids, axis=1)
        data[i][-1] = np.argmin(distances)  # Assign the point to the closest centroid

    return data
