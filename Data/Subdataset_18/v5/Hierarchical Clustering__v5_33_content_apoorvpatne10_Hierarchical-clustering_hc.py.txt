import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
import scipy.cluster.hierarchy as sch
from sklearn.cluster import AgglomerativeClustering

def load_dataset(file_path):
    """
    Load the dataset and extract the relevant features for clustering.

    Args:
        file_path (str): The path to the CSV file.

    Returns:
        np.ndarray: Extracted features (Annual Income and Spending Score).
    """
    dataset = pd.read_csv(file_path)
    features = dataset.iloc[:, [3, 4]].values  # Extract Annual Income and Spending Score
    return features

def plot_dendrogram(X):
    """
    Plot the dendrogram to help determine the optimal number of clusters.

    Args:
        X (np.ndarray): The data to be used for plotting the dendrogram.
    """
    plt.figure(figsize=(10, 7))
    dendrogram = sch.dendrogram(sch.linkage(X, method='ward'))
    plt.title('Dendrogram')
    plt.xlabel('Customers')
    plt.ylabel('Euclidean Distance')
    plt.show()

def perform_clustering(X, n_clusters=5):
    """
    Perform Agglomerative Clustering on the dataset.

    Args:
        X (np.ndarray): The data to be clustered.
        n_clusters (int): The number of clusters to form.

    Returns:
        np.ndarray: Cluster labels for each data point.
    """
    clustering_model = AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage='ward')
    cluster_labels = clustering_model.fit_predict(X)
    return cluster_labels

def visualize_clusters(X, y_hc):
    """
    Visualize the clusters formed by Agglomerative Clustering.

    Args:
        X (np.ndarray): The data used for clustering.
        y_hc (np.ndarray): Cluster labels for the data points.
    """
    plt.figure(figsize=(10, 7))
    cluster_colors = ['red', 'blue', 'green', 'cyan', 'magenta']
    cluster_labels = ['Careful', 'Standard', 'Targets', 'Careless', 'Sensible']

    for i, color in enumerate(cluster_colors):
        plt.scatter(X[y_hc == i, 0], X[y_hc == i, 1], s=100, c=color, label=cluster_labels[i])

    plt.title('Clusters of Clients')
    plt.xlabel('Annual Income (k$)')
    plt.ylabel('Spending Score (1-100)')
    plt.legend()
    plt.show()

def main():
    # File path to the dataset
    file_path = 'Mall_Customers.csv'

    # Load the dataset and extract features
    features = load_dataset(file_path)

    # Plot the dendrogram to determine the optimal number of clusters
    plot_dendrogram(features)

    # Perform Agglomerative Clustering
    cluster_labels = perform_clustering(features, n_clusters=5)

    # Visualize the clusters
    visualize_clusters(features, cluster_labels)

if __name__ == "__main__":
    main()
