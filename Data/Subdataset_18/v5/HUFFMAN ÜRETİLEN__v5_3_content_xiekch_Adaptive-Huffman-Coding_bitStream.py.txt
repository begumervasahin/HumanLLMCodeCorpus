class BitStream:
    def __init__(self, file_name, mode):
        """Initializes the BitStream with the specified file and mode."""
        self.file = open(file_name, mode)
        self.word = 0
        self.pos = 7 if 'w' in mode else -1
        self.mode = mode

    def read(self, size):
        """Reads a specified number of bits from the file."""
        bits = []
        for _ in range(size):
            if self.pos == -1:
                byte = self.file.read(1)
                if not byte:
                    return ''.join(bits)
                self.word = ord(byte)
                self.pos = 7
            bits.append('1' if self.word & (1 << self.pos) else '0')
            self.pos -= 1
        return ''.join(bits)

    def write(self, bit_string):
        """Writes a string of bits to the file."""
        for bit in bit_string:
            self.word = (self.word << 1) | int(bit)
            self.pos -= 1
            if self.pos == -1:
                self.flush()

    def flush(self):
        """Flushes any remaining bits in the buffer to the file."""
        if self.pos != 7:
            self.word <<= self.pos + 1
            self.file.write(self.word.to_bytes(1, byteorder='little'))
            self.word = 0
            self.pos = 7

    def close(self):
        """Closes the file, flushing any remaining bits if in write mode."""
        if 'w' in self.mode:
            self.flush()
        self.file.close()


def main():
    input_file = './test.txt'
    output_file = './temp.txt'

    input_stream = BitStream(input_file, 'rb')
    output_stream = BitStream(output_file, 'wb')

    i = 0
    while True:
        bit = input_stream.read(1)
        if not bit:
            break
        print(bit, end='')
        if i % 8 == 7:
            print()
        output_stream.write(bit)
        i += 1

    input_stream.close()
    output_stream.close()


if __name__ == '__main__':
    main()
