from collections import defaultdict
import operator

class Driver:
    def __init__(self, modules):
        self._modules = modules
        self._intervals = []
        self._values = []
        self.print_modules()

    def print_modules(self):
        """
        Print the header for the modules and the winner column.
        """
        print('Modules\t|\t', end='')
        for module in self._modules:
            print(f'{module}\t', end='')
        print('Winner')

    def set_module(self, index, value=None, interval=None):
        """
        Set the last value or interval for a specific module.

        Args:
            index (int): The index of the module to set.
            value: The value to set for the module.
            interval: The interval to set for the module.
        """
        if index < len(self._modules):
            if value is not None:
                self._modules[index]._lastValue = value
            if interval is not None:
                self._modules[index]._lastInterval = interval

    def test(self, intervals):
        """
        Test the modules with given intervals and print the results.

        Args:
            intervals (list): A list of intervals to test.
        """
        for interval in intervals:
            self.run(interval)

    def run(self, interval):
        """
        Run a test with a single interval, calculate results, and give feedback.

        Args:
            interval: The interval to test.
        """
        results = [module.getResult(interval) for module in self._modules]
        votes = self.count_votes(results)
        winner = self.get_winner(votes)
        self.print_results(interval, results, winner)
        self._values.append(winner)
        self._intervals.append(interval)
        self.give_feedback(results, winner)

    def count_votes(self, results):
        """
        Count votes for each result from the modules.

        Args:
            results (list): A list of results from the modules.

        Returns:
            dict: A dictionary with results as keys and their vote counts as values.
        """
        votes = defaultdict(int)
        for res in results:
            votes[res] += 1
        return votes

    def get_winner(self, votes):
        """
        Determine the winner based on the votes.

        Args:
            votes (dict): A dictionary with results as keys and their vote counts as values.

        Returns:
            The result with the highest votes.
        """
        return max(votes.items(), key=operator.itemgetter(1))[0]

    def print_results(self, interval, results, winner):
        """
        Print the results for a given interval.

        Args:
            interval: The interval tested.
            results (list): A list of results from the modules.
            winner: The winning result.
        """
        print(f'{interval}\t|\t', end='')
        for res in results:
            print(f'{res}\t', end='')
        print(winner)

    def give_feedback(self, results, winner):
        """
        Give feedback to the modules based on the results and the winner.

        Args:
            results (list): A list of results from the modules.
            winner: The winning result.
        """
        for i, module in enumerate(self._modules):
            feedback = {'status': 'ok'} if results[i] == winner else {'status': 'error', 'goodValue': winner}
            module.receiveFeedback(feedback)
