def counting_sort(arr, verbose=0, desc=0):
    """
    Performs counting sort on the given array.

    Parameters:
    arr (list of int): The array to be sorted.
    verbose (int, optional): Verbosity level for debugging output. Default is 0.
    desc (int, optional): If set to 1, sorts the array in descending order. Default is 0.

    Returns:
    list of int: The sorted array.
    """

    if not arr:
        return []  # Return empty list if input is empty

    # Identify the range of the data
    smallest = min(arr)
    largest = max(arr)

    # Normalize the array if the smallest number is negative
    normalized_arr = [x - smallest for x in arr]

    if verbose:
        print("Normalized array:", normalized_arr)

    # Create a count array and populate it with the frequency of each element
    range_size = largest - smallest + 1
    count = [0] * range_size

    for num in normalized_arr:
        count[num] += 1

    # Convert the count array to a cumulative count array
    for i in range(1, range_size):
        count[i] += count[i - 1]

        if verbose == 2:
            print(f"Cumulative count at index {i}: {count[i]}")

    # Create the output array and place elements in their correct positions
    output = [0] * len(arr)

    for num in reversed(normalized_arr):
        count[num] -= 1
        output[count[num]] = num

    if verbose:
        print("Sorted output (before denormalization):", output)

    # Denormalize the output array to restore original values
    denormalized_output = [x + smallest for x in output]

    # Reverse the output array if descending order is requested
    if desc:
        denormalized_output.reverse()

    return denormalized_output
