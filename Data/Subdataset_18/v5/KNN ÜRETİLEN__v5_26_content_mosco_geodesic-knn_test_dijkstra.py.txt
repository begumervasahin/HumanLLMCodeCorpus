import numpy as np
import scipy.sparse as sparse
import scipy.sparse.csgraph as csgraph
import dijkstra

def generate_random_weighted_graph(size, threshold=1.0):
    """
    Generate a random symmetric weighted graph and convert it to CSR format.
    
    Parameters:
        size (int): The size of the graph (number of nodes).
        threshold (float): Threshold below which weights are set to infinity.
        
    Returns:
        scipy.sparse.csr_matrix: The generated graph in CSR format.
    """
    W = np.random.random((size, size))
    W = (W + W.T) / 2  # Make the graph symmetric
    W[W < threshold] = np.inf  # Set weights less than threshold to infinity to represent no path
    return sparse.csr_matrix(W)  # Convert to CSR format for efficient processing

def test_dijkstra_against_scipy(graph_size=100, seeds=[0, 1, 2]):
    """
    Test the custom Dijkstra implementation against SciPy's implementation.
    
    Parameters:
        graph_size (int): The size of the graph (number of nodes).
        seeds (list of int): List of seed values to test.
    """
    # Generate a random weighted graph
    W = generate_random_weighted_graph(graph_size)

    # Test for different seed values
    for seed in seeds:
        result_custom = dijkstra.dijkstra(W, seed)
        result_scipy = csgraph.dijkstra(W, indices=seed)[0]
        
        # Ensure the results from both implementations are the same
        assert np.allclose(result_custom, result_scipy), f"Mismatch found for seed {seed}"
        print(f"Test passed for seed {seed}")

if __name__ == "__main__":
    test_dijkstra_against_scipy()
