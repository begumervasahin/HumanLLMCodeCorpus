# Apriori Algorithm for Market Basket Optimization

# Importing the necessary libraries
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from apyori import apriori

# Data Preprocessing
def load_data(file_path):
    return pd.read_csv(file_path, header=None)

def prepare_transactions(dataset):
    transactions = []
    for i in range(len(dataset)):
        transaction = [str(dataset.values[i, j]) for j in range(dataset.shape[1]) if str(dataset.values[i, j]) != 'nan']
        transactions.append(transaction)
    return transactions

# Training the Apriori algorithm on the dataset
def apply_apriori(transactions, min_support, min_confidence, min_lift, min_length):
    return apriori(transactions, min_support=min_support, min_confidence=min_confidence, min_lift=min_lift, min_length=min_length)

# Visualizing the results
def visualize_results(results):
    for result in results:
        print(result)

def main():
    dataset = load_data('Market_Basket_Optimisation.csv')
    transactions = prepare_transactions(dataset)
    rules = apply_apriori(transactions, min_support=0.003, min_confidence=0.2, min_lift=3, min_length=2)
    results = list(rules)
    visualize_results(results)

if __name__ == "__main__":
    main()
