import pandas as pd

def load_data(info_path, reply_path):
    """Load the information and reply data from CSV files."""
    info_df = pd.read_csv(info_path)
    reply_df = pd.read_csv(reply_path).drop(columns=['Unnamed: 0'])
    return info_df, reply_df

def get_low_star_data(info_df):
    """Get low star point data, sorted and filtered."""
    sorted_info_df = info_df.sort_values(by='star_point')
    low_star_df = sorted_info_df.head(1000).drop_duplicates(subset=["titleId", "nom"])
    return low_star_df

def get_high_star_data(info_df):
    """Get high star point data, sampled and filtered."""
    high_star_df = info_df[info_df['star_point'] == 9.99].sample(n=1000).drop(columns=['Unnamed: 0'])
    high_star_df = high_star_df.drop_duplicates(subset=["titleId", "nom"])
    return high_star_df

def extract_comments(df, reply_df):
    """Extract comments from the reply data and add to the dataframe."""
    title_ids = df['titleId'].tolist()
    noms = df['nom'].tolist()
    reply_list = [[] for _ in range(10)]
    
    for title_id, nom in zip(title_ids, noms):
        filtered_replies = reply_df[(reply_df['titleId'] == title_id) & (reply_df['nom'] == nom)]
        for i in range(10):
            try:
                reply_list[i].append(str(filtered_replies[f'comment{i+1}'].values[0]))
            except IndexError:
                reply_list[i].append('')

    for i in range(10):
        df[f'comment{i+1}'] = reply_list[i]
    
    df = df[df['comment1'] != '']
    return df.head(500)

def save_data(df, file_path):
    """Save the dataframe to a CSV file."""
    df.to_csv(file_path, encoding="utf_8_sig", mode='w')

def main():
    # Paths to the CSV files
    info_path = 'path/to/weebtoon_star_info.csv'
    reply_path = 'path/to/weebtoon_reply_info.csv'
    
    # Load the data
    info_df, reply_df = load_data(info_path, reply_path)
    
    # Process low star data
    low_star_df = get_low_star_data(info_df)
    print(low_star_df)
    low_star_df = extract_comments(low_star_df, reply_df)
    print(low_star_df)
    save_data(low_star_df, "./lowstar_reply.csv")
    
    # Process high star data
    high_star_df = get_high_star_data(info_df)
    high_star_df = extract_comments(high_star_df, reply_df)
    print(high_star_df)
    save_data(high_star_df, "./highstar_reply.csv")

if __name__ == "__main__":
    main()
