import time
import random

def selection_sort(array):
    """Sort the array using selection sort algorithm and measure the time taken."""
    start_time = time.time()  # Record the start time
    
    # Traverse through all elements except the last one
    for i in range(len(array) - 1):
        # Assume the minimum element is the first unsorted element
        min_idx = i
        
        # Find the smallest element in the remaining unsorted portion
        for j in range(i + 1, len(array)):
            if array[min_idx] > array[j]:
                min_idx = j
        
        # Swap the found minimum element with the first unsorted element
        array[i], array[min_idx] = array[min_idx], array[i]
    
    end_time = time.time()  # Record the end time
    print(f"Sorting completed in {end_time - start_time:.6f} seconds")
    
    return array

# Example usage
def main():
    array = [random.randint(0, 100) for _ in range(10)]
    print("Original array:", array)
    sorted_array = selection_sort(array)
    print("Sorted array:", sorted_array)

if __name__ == "__main__":
    main()
