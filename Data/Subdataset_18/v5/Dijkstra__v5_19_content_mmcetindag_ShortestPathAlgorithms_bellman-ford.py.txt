class Graph:
    def __init__(self, vertices):
        """
        Initialize the graph with a given number of vertices.

        Args:
        - vertices (int): The number of vertices in the graph.
        """
        self.num_vertices = vertices
        self.edges = []

    def add_edge(self, start, end, weight):
        """
        Add an edge to the graph.

        Args:
        - start (int): The start vertex of the edge.
        - end (int): The end vertex of the edge.
        - weight (int): The weight of the edge.
        """
        self.edges.append((start, end, weight))

    def display_distances(self, distances):
        """
        Print the shortest distances from the source to all vertices.

        Args:
        - distances (list): A list of shortest distances from the source.
        """
        print("Vertex   Distance from Source")
        for vertex, distance in enumerate(distances):
            print(f"{vertex} \t\t {distance}")

    def bellman_ford(self, source):
        """
        Run the Bellman-Ford algorithm to find the shortest paths from the source vertex.

        Args:
        - source (int): The source vertex for the shortest path calculation.
        """
        # Step 1: Initialize distances from the source to all other vertices as INFINITY
        distances = [float("inf")] * self.num_vertices
        distances[source] = 0

        # Step 2: Relax all edges |V| - 1 times
        for _ in range(self.num_vertices - 1):
            for start, end, weight in self.edges:
                if distances[start] != float("inf") and distances[start] + weight < distances[end]:
                    distances[end] = distances[start] + weight

        # Step 3: Check for negative weight cycles
        for start, end, weight in self.edges:
            if distances[start] != float("inf") and distances[start] + weight < distances[end]:
                print("Graph contains a negative weight cycle")
                return

        # Print the shortest distances
        self.display_distances(distances)
