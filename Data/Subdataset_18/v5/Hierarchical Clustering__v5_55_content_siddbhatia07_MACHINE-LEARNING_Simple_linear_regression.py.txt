import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression

def load_data(filepath):
    """
    Load dataset from a CSV file.

    Parameters:
    - filepath (str): The path to the CSV file.

    Returns:
    - X (ndarray): The features matrix.
    - y (ndarray): The target variable array.
    """
    dataset = pd.read_csv(filepath)
    X = dataset.iloc[:, :-1].values  # Features: all columns except the last one
    y = dataset.iloc[:, 1].values    # Target: the second column
    return X, y

def split_data(X, y, test_size=1/3, random_state=0):
    """
    Split the dataset into training and test sets.

    Parameters:
    - X (ndarray): The features matrix.
    - y (ndarray): The target variable array.
    - test_size (float): The proportion of the dataset to include in the test split.
    - random_state (int): The seed used by the random number generator.

    Returns:
    - X_train, X_test, y_train, y_test: The split datasets.
    """
    return train_test_split(X, y, test_size=test_size, random_state=random_state)

def train_model(X_train, y_train):
    """
    Train a Linear Regression model.

    Parameters:
    - X_train (ndarray): The training features.
    - y_train (ndarray): The training target variable.

    Returns:
    - regressor (LinearRegression): The trained Linear Regression model.
    """
    regressor = LinearRegression()
    regressor.fit(X_train, y_train)
    return regressor

def visualize_results(X, y, model, title, xlabel, ylabel):
    """
    Visualize the results of the Linear Regression model.

    Parameters:
    - X (ndarray): The features.
    - y (ndarray): The target variable.
    - model (LinearRegression): The trained Linear Regression model.
    - title (str): The title of the plot.
    - xlabel (str): The label for the x-axis.
    - ylabel (str): The label for the y-axis.
    """
    plt.figure(figsize=(10, 6))
    plt.scatter(X, y, color='red', label='Actual Data')
    plt.plot(X, model.predict(X), color='blue', label='Regression Line')
    plt.title(title)
    plt.xlabel(xlabel)
    plt.ylabel(ylabel)
    plt.legend()
    plt.show()

def main():
    # Load the dataset
    X, y = load_data('Data.csv')

    # Split the dataset into training and test sets
    X_train, X_test, y_train, y_test = split_data(X, y)

    # Train the Linear Regression model on the training set
    model = train_model(X_train, y_train)

    # Visualize the training set results
    visualize_results(X_train, y_train, model, 'Salary vs Experience (Training Set)', 'Years of Experience', 'Salary')

    # Visualize the test set results
    visualize_results(X_test, y_test, model, 'Salary vs Experience (Test Set)', 'Years of Experience', 'Salary')

if __name__ == "__main__":
    main()
