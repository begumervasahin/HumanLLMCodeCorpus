import unittest
from StackImplUsingArray import Stack

class TestStackImplUsingArray(unittest.TestCase):
    
    def setUp(self):
        """Set up the stack for each test."""
        self.stack_capacity = 10
        self.myStack = Stack(self.stack_capacity)

    def test_push(self):
        """Test pushing an element onto the stack."""
        self.assertEqual(self.myStack.push(5), "success")

    def test_stack_overflow(self):
        """Test stack overflow when pushing more elements than its capacity."""
        small_stack = Stack(3)
        small_stack.push(5)
        small_stack.push(5)
        small_stack.push(5)
        self.assertEqual(small_stack.push(5), "Stack full")

    def test_top_element(self):
        """Test retrieving the top element of the stack."""
        self.myStack.push(10)
        self.assertEqual(self.myStack.topElement(), 10)

    def test_top_element_stack_empty(self):
        """Test retrieving the top element when the stack is empty."""
        self.assertEqual(self.myStack.topElement(), "Stack empty")

    def test_pop(self):
        """Test popping an element from the stack."""
        self.myStack.push(10)
        self.assertEqual(self.myStack.pop(), 10)

    def test_stack_underflow(self):
        """Test stack underflow when popping an element from an empty stack."""
        self.assertEqual(self.myStack.pop(), "Stack empty")

    def test_is_empty(self):
        """Test checking if the stack is empty."""
        self.assertTrue(self.myStack.isEmpty())

    def test_is_not_empty(self):
        """Test checking if the stack is not empty."""
        self.myStack.push(10)
        self.assertFalse(self.myStack.isEmpty())

    def test_print_empty_stack(self):
        """Test printing an empty stack."""
        # To verify the output manually, use self.myStack.print() and check the console output
        # Here we assume print() method works correctly if no exceptions are raised
        try:
            self.myStack.print()
            result = "No exception"
        except Exception as e:
            result = str(e)
        self.assertEqual(result, "No exception")

    def test_print(self):
        """Test printing the elements of the stack."""
        # To verify the output manually, use self.myStack.print() and check the console output
        # Here we assume print() method works correctly if no exceptions are raised
        self.myStack.push(10)
        self.myStack.push(20)
        try:
            self.myStack.print()
            result = "No exception"
        except Exception as e:
            result = str(e)
        self.assertEqual(result, "No exception")

if __name__ == '__main__':
    unittest.main()
