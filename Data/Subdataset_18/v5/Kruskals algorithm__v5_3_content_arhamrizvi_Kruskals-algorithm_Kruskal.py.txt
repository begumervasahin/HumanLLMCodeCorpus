def swap(array, i, j):
    """Swap elements at indices i and j in the array."""
    array[i], array[j] = array[j], array[i]

def quicksort(array):
    """Sort the array using the Quicksort algorithm."""
    _quicksort(array, 0, len(array) - 1)

def _quicksort(array, start, end):
    """Auxiliary recursive function for Quicksort."""
    if start < end:
        boundary = _partition(array, start, end)
        _quicksort(array, start, boundary)
        _quicksort(array, boundary + 1, end)

def _partition(array, start, end):
    """Partition the array and return the pivot index."""
    mid = (start + end) // 2
    pivot = array[mid][2]
    swap(array, start, mid)
    index = start
    for k in range(start + 1, end + 1):
        if array[k][2] < pivot:
            index += 1
            swap(array, k, index)
    swap(array, start, index)
    return index

def find(x):
    """Find the representative of the set that x belongs to."""
    if parent[x] != x:
        parent[x] = find(parent[x])  # Path compression
    return parent[x]

def union(x, y):
    """Union the sets that x and y belong to."""
    root_x = find(x)
    root_y = find(y)
    if root_x != root_y:
        if rank[root_x] > rank[root_y]:
            parent[root_y] = root_x
        else:
            parent[root_x] = root_y
            if rank[root_x] == rank[root_y]:
                rank[root_y] += 1

def read_graph(filename):
    """Read a graph from a file and return it as a list of edges and the number of vertices."""
    edges = []
    max_vertex = 0
    with open(filename, "r") as file:
        for line in file:
            u, v, w = map(int, line.split())
            edges.append([u, v, w])
            max_vertex = max(max_vertex, u, v)
    return edges, max_vertex + 1

def kruskal(filename, output_filename):
    """
    Run Kruskal's algorithm on the graph from filename and write the MST to output_filename.
    
    Args:
        filename (str): The input file containing the graph.
        output_filename (str): The output file to write the MST.
    """
    edges, num_vertices = read_graph(filename)
    
    global parent, rank
    parent = list(range(num_vertices))
    rank = [0] * num_vertices
    
    quicksort(edges)  # Sort edges by weight
    
    mst = []
    for u, v, weight in edges:
        if find(u) != find(v):
            mst.append([u, v, weight])
            union(u, v)
    
    # Write the MST to the output file
    with open(output_filename, "w") as file:
        for u, v, w in mst:
            file.write(f"{u} {v} {w}\n")
