# Import necessary libraries
from itertools import combinations

def load_dataset(filename):
    """
    Load the dataset from a text file.

    Parameters:
    filename (str): Path to the dataset file.

    Returns:
    list: List of transactions.
    """
    with open(filename, 'r') as file:
        return file.read().splitlines()

def generate_C1(dataset):
    """
    Generate candidate itemsets of size 1.

    Parameters:
    dataset (list): List of transactions.

    Returns:
    dict: Dictionary of candidate 1-itemsets with their support count.
    """
    C1 = {}
    for line in dataset:
        items = line.split()
        for item in items:
            C1[item] = C1.get(item, 0) + 1
    return C1

def generate_L1(C1, min_support):
    """
    Generate frequent itemsets of size 1.

    Parameters:
    C1 (dict): Dictionary of candidate 1-itemsets with their support count.
    min_support (int): Minimum support threshold.

    Returns:
    dict: Dictionary of frequent 1-itemsets with their support count.
    """
    return {item: count for item, count in C1.items() if count >= min_support}

def generate_C2(L1):
    """
    Generate candidate itemsets of size 2 from L1.

    Parameters:
    L1 (dict): Dictionary of frequent 1-itemsets with their support count.

    Returns:
    dict: Dictionary of candidate 2-itemsets initialized with support count 0.
    """
    C2 = {}
    items = list(L1.keys())
    for i in range(len(items)):
        for j in range(i + 1, len(items)):
            C2[(items[i], items[j])] = 0
    return C2

def count_support_C2(C2, dataset):
    """
    Count support for candidate itemsets of size 2.

    Parameters:
    C2 (dict): Dictionary of candidate 2-itemsets.
    dataset (list): List of transactions.

    Returns:
    dict: Dictionary of candidate 2-itemsets with their support count.
    """
    for line in dataset:
        items = set(line.split())
        for pair in C2.keys():
            if set(pair).issubset(items):
                C2[pair] += 1
    return C2

def generate_L2(C2, min_support):
    """
    Generate frequent itemsets of size 2.

    Parameters:
    C2 (dict): Dictionary of candidate 2-itemsets with their support count.
    min_support (int): Minimum support threshold.

    Returns:
    dict: Dictionary of frequent 2-itemsets with their support count.
    """
    return {pair: count for pair, count in C2.items() if count >= min_support}

def apriori(dataset, min_support):
    """
    Run the Apriori algorithm to find frequent itemsets.

    Parameters:
    dataset (list): List of transactions.
    min_support (int): Minimum support threshold.
    """
    # Generate candidate itemsets of size 1
    C1 = generate_C1(dataset)
    # Generate frequent itemsets of size 1
    L1 = generate_L1(C1, min_support)
    print(f"Frequent 1-itemsets: {L1}")

    # Generate candidate itemsets of size 2
    C2 = generate_C2(L1)
    # Count support for candidate itemsets of size 2
    C2 = count_support_C2(C2, dataset)
    # Generate frequent itemsets of size 2
    L2 = generate_L2(C2, min_support)
    print(f"Frequent 2-itemsets: {L2}")

if __name__ == "__main__":

    # Minimum support threshold

    # Run the Apriori algorithm
