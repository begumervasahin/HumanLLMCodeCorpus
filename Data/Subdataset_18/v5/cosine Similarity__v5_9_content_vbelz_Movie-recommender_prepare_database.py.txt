import os
import re
import pandas as pd
from sqlalchemy import create_engine

def load_data(data_folder):
    """
    Load movies and ratings data from CSV files.

    Args:
        data_folder (str): Path to the folder containing the CSV files.

    Returns:
        pd.DataFrame, pd.DataFrame: DataFrames containing movies and ratings data.
    """
    df_movies = pd.read_csv(os.path.join(data_folder, 'movies.csv'))
    df_ratings = pd.read_csv(os.path.join(data_folder, 'ratings.csv'))
    return df_movies, df_ratings

def merge_data(df_movies, df_ratings):
    """
    Merge movies and ratings DataFrames on the movieId column.

    Args:
        df_movies (pd.DataFrame): DataFrame containing movies data.
        df_ratings (pd.DataFrame): DataFrame containing ratings data.

    Returns:
        pd.DataFrame: Merged DataFrame.
    """
    return pd.merge(df_ratings, df_movies, on='movieId', how='outer')

def extract_year(title):
    """
    Extract the year from the movie title.

    Args:
        title (str): The movie title.

    Returns:
        str: The extracted year, or '0' if not found.
    """
    match = re.search(r"\(([0-9]{4})\)", title)
    return match.group(1) if match else '0'

def clean_data(df):
    """
    Clean the merged DataFrame by extracting the year from the title and selecting relevant columns.

    Args:
        df (pd.DataFrame): Merged DataFrame.

    Returns:
        pd.DataFrame: Cleaned DataFrame.
    """
    df['year'] = df['title'].apply(extract_year)
    df['title'] = df['title'].str.replace(r"\(([0-9]{4})\)", "", regex=True).str.strip()
    columns = ['userId', 'movieId', 'rating', 'title', 'genres', 'year', 'timestamp']
    df = df[columns]
    df.sort_values(by='movieId', inplace=True)
    df.dropna(subset=['userId'], inplace=True)
    return df

def save_to_sqlite(df, db_path):
    """
    Save DataFrame to a SQLite database.

    Args:
        df (pd.DataFrame): DataFrame to be saved.
        db_path (str): Path to the SQLite database file.
    """
    engine = create_engine(f'sqlite:///{db_path}')
    df.to_sql('movie_ratings', engine, if_exists='replace', index=False)

def save_to_postgres(df, connection_string):
    """
    Save DataFrame to a PostgreSQL database.

    Args:
        df (pd.DataFrame): DataFrame to be saved.
        connection_string (str): Connection string for the PostgreSQL database.
    """
    engine = create_engine(connection_string)
    df.to_sql('movies_ratings', engine, if_exists='replace', index=False)

def main():
    data_folder = './data/'
    movies_file = os.path.join(data_folder, 'movies.csv')
    ratings_file = os.path.join(data_folder, 'ratings.csv')

    # Load the data
    df_movies, df_ratings = load_data(data_folder)

    # Merge and clean the data
    df_merged = merge_data(df_movies, df_ratings)
    df_cleaned = clean_data(df_merged)

    # Save the cleaned DataFrame to the desired database
    SAVE_TO_POSTGRES = False
    SAVE_TO_SQLITE = True

    if SAVE_TO_SQLITE:
        save_to_sqlite(df_cleaned, os.path.join(data_folder, 'movielens.db'))

    if SAVE_TO_POSTGRES:
        HOST = 'localhost'
        PORT = '5432'
        DBNAME = 'movielens'
        connection_string = f'postgresql://{HOST}:{PORT}/{DBNAME}'
        save_to_postgres(df_cleaned, connection_string)

    # Remove the original CSV files
    os.remove(movies_file)
    os.remove(ratings_file)

if __name__ == '__main__':
    main()
