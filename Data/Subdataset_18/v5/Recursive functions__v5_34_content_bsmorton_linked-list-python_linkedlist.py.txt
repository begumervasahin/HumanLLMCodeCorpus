class LN:
    def __init__(self, value, next=None):
        self.value = value
        self.next = next

def len_ll(ll):
    """Calculate the length of the linked list iteratively."""
    count = 0
    while ll:
        count += 1
        ll = ll.next
    return count

def len_ll_r(ll):
    """Calculate the length of the linked list recursively."""
    return 0 if ll is None else 1 + len_ll_r(ll.next)

def sum_ll(ll):
    """Calculate the sum of values in the linked list iteratively."""
    total = 0
    while ll:
        total += ll.value
        ll = ll.next
    return total

def sum_ll_r(ll):
    """Calculate the sum of values in the linked list recursively."""
    return 0 if ll is None else ll.value + sum_ll_r(ll.next)

def str_ll(ll):
    """Create a string representation of the linked list iteratively."""
    result = ''
    while ll:
        result += f"{ll.value}->"
        ll = ll.next
    return result + 'None'

def str_ll_r(ll):
    """Create a string representation of the linked list recursively."""
    return 'None' if ll is None else f"{ll.value}->" + str_ll_r(ll.next)

def list_to_ll(lst):
    """Convert a list to a linked list iteratively."""
    if not lst:
        return None
    front = rear = LN(lst[0])
    for value in lst[1:]:
        rear.next = LN(value)
        rear = rear.next
    return front

def list_to_ll_r(lst):
    """Convert a list to a linked list recursively."""
    return None if not lst else LN(lst[0], list_to_ll_r(lst[1:]))

def find_ll(ll, value):
    """Find the first node with the given value in the linked list iteratively."""
    while ll:
        if ll.value == value:
            return ll
        ll = ll.next
    return None

def find_ll_r(ll, value):
    """Find the first node with the given value in the linked list recursively."""
    if ll is None or ll.value == value:
        return ll
    return find_ll_r(ll.next, value)

def copy_ll(ll):
    """Create a copy of the linked list iteratively."""
    if not ll:
        return None
    front = rear = LN(ll.value)
    while ll.next:
        ll = ll.next
        rear.next = LN(ll.value)
        rear = rear.next
    return front

def copy_ll_r(ll):
    """Create a copy of the linked list recursively."""
    return None if ll is None else LN(ll.value, copy_ll_r(ll.next))

def iterator(ll):
    """Iterate over the values in the linked list."""
    while ll:
        yield ll.value
        ll = ll.next

def append_ll(ll, value):
    """Append a value to the end of the linked list iteratively."""
    if not ll:
        return LN(value)
    front = ll
    while ll.next:
        ll = ll.next
    ll.next = LN(value)
    return front

def append_ll_r(ll, value):
    """Append a value to the end of the linked list recursively."""
    return LN(value) if ll is None else (ll.next := append_ll_r(ll.next, value)) or ll

def add_after_ll(ll, value):
    """Add a new node with the given value after the current node."""
    ll.next = LN(value, ll.next)

def remove_after_ll(ll):
    """Remove the node immediately following the current node."""
    if ll.next:
        ll.next = ll.next.next
