def selection_sort(array):
    """Sort the array using the selection sort algorithm."""
    n = len(array)
    
    for i in range(n):
        # Assume the minimum element is the first unsorted element
        min_idx = i
        
        # Find the smallest element in the remaining unsorted portion
        for j in range(i + 1, n):
            if array[min_idx] > array[j]:
                min_idx = j
        
        # Swap the found minimum element with the first unsorted element
        array[i], array[min_idx] = array[min_idx], array[i]
    
    return array

def main():
    # Define the array to be sorted
    my_arr = [8, 5, 2, 6, 9, 3, 1, 4, 0, 7]
    print("Original array:", my_arr)

    # Sort the array
    sorted_array = selection_sort(my_arr)
    print("Sorted array:", sorted_array)

if __name__ == "__main__":
    main()
