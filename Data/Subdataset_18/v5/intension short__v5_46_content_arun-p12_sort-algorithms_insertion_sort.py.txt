def insertion_sort(arr, verbose=0, desc=False):
    """
    Sort a list using the insertion sort algorithm.

    Parameters:
        arr (list): The list of elements to be sorted.
        verbose (int, optional): Level of verbosity for logging progress. 
                                 0 = no output, 1 = output after each iteration, 2 = detailed output during shifts.
        desc (bool, optional): If True, the sorted list will be returned in descending order. Defaults to False.

    Returns:
        list: The sorted list.
    """
    for i in range(1, len(arr)):
        key = arr[i]
        j = i - 1
        
        # Shift elements of arr[0..i-1] that are greater than key
        while j >= 0 and arr[j] > key:
            arr[j + 1] = arr[j]
            j -= 1
            if verbose == 2:
                print(f"  Subarray after shifting: {arr}")
        
        arr[j + 1] = key
        
        if verbose >= 1:
            print(f"Iteration {i}: {arr}")
    
    if desc:
        arr.reverse()
    
    return arr
