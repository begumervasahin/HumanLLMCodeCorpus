import random

def generate_conf(file_prefix, num_samples, sample_size, algoritmo):
    """
    Generates configuration samples and writes them to a file.

    Parameters:
    file_prefix (str): The prefix for the output file name.
    num_samples (int): The number of samples to generate.
    sample_size (int): The size of each sample.
    algoritmo (str): The algorithm name to be included in the file name.

    Returns:
    list: A list of generated samples.
    """
    config = []
    for n in range(num_samples):
        sample = random.sample(range(1000000000, 9999999999), sample_size)
        config.append(sample)
        print(f'Sample {n + 1} generated.')
    
    with open(f'{file_prefix}{algoritmo}.txt', 'a') as file:
        for sample in config:
            file.write(f'{sample}\n')
    
    return config

def confA(algoritmo):
    """
    Generates configuration A samples.

    Parameters:
    algoritmo (str): The algorithm name to be included in the file name.

    Returns:
    list: A list of generated samples for configuration A.
    """
    return generate_conf('confAInicial', 100000, 10, algoritmo)
