class Queue:
    """A simple implementation of a queue with state tracking."""
    
    def __init__(self):
        self.items = []
        self.state = set()
        
    def enqueue(self, item):
        """Add item to the queue."""
        self.items.append(item)
        self.state.add(tuple(item.state))
        
    def dequeue(self):
        """Remove and return the first item from the queue (FIFO)."""
        if not self.is_empty():
            item = self.items.pop(0)
            self.state.remove(tuple(item.state))
            return item
        return None
        
    def is_empty(self):
        """Check if the queue is empty."""
        return len(self.items) == 0

class PuzzleState:
    """Example PuzzleState class to be used with Queue."""
    
    def __init__(self, state_list):
        self.state = state_list
        
    def __repr__(self):
        return f"PuzzleState({self.state})"
    
if __name__ == "__main__":
    # Create a Queue instance
    queue = Queue()
    
    # Enqueue some PuzzleState instances
    queue.enqueue(PuzzleState([1, 2, 3, 4, 5, 6, 7, 8, 0]))
    queue.enqueue(PuzzleState([1, 2, 3, 4, 5, 6, 7, 0, 8]))
    
    # Check if the queue is empty
    print(f"Queue is empty: {queue.is_empty()}")
    
    # Dequeue an item and print it
    dequeued_item = queue.dequeue()
    print(f"Dequeued: {dequeued_item}")
    
    # Check if the queue is empty again
    print(f"Queue is empty: {queue.is_empty()}")
    
    # Dequeue another item and print it
    dequeued_item = queue.dequeue()
    print(f"Dequeued: {dequeued_item}")
    
    # Check if the queue is empty one last time
    print(f"Queue is empty: {queue.is_empty()}")
