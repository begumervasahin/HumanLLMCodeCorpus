from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from time import sleep

def initialize_webdriver():
    """
    Initializes the Selenium WebDriver for Firefox.
    
    Returns:
        webdriver.Firefox: Configured WebDriver instance.
    """
    driver_path = 'D:/geckodriver/geckodriver.exe'
    return webdriver.Firefox(executable_path=driver_path)

def input_date(driver, date):
    """
    Inputs a date into the date field on the webpage.

    Args:
        driver (webdriver.Firefox): The WebDriver instance.
        date (str): The date in 'YYYY-MM-DD' format to input.
    """
    date_field = driver.find_element(By.ID, 'fecha_salida')
    date_field.click()
    date_field.clear()  # Clear any pre-existing value
    date_field.send_keys(date)

def fetch_and_print_dropdown_options(driver, dropdown_id):
    """
    Fetches and prints the options from a dropdown menu.

    Args:
        driver (webdriver.Firefox): The WebDriver instance.
        dropdown_id (str): The ID of the dropdown menu.
    """
    dropdown = WebDriverWait(driver, 10).until(
        EC.presence_of_element_located((By.ID, dropdown_id))
    )
    options = dropdown.find_elements(By.TAG_NAME, 'option')
    for option in options:
        print(option.get_attribute('value'))
        sleep(1)  # Pause for readability

def process_flight_options(driver):
    """
    Processes flight options and prints related hotel options.
    
    Args:
        driver (webdriver.Firefox): The WebDriver instance.
    """
    flight_dropdown = WebDriverWait(driver, 10).until(
        EC.presence_of_element_located((By.ID, 'flight'))
    )
    flights = flight_dropdown.find_elements(By.TAG_NAME, 'option')
    for flight in flights:
        flight.click()
        print(f"Selected flight: {flight.get_attribute('value')}")
        
        # Fetch and print hotel options for the selected flight
        print("Hotel options:")
        fetch_and_print_dropdown_options(driver, 'hoteles')
        sleep(1)  # Pause between flight selections for readability

def main():
    """
    Main function to perform the web interactions and data extraction.
    """
    driver = initialize_webdriver()
    try:
        driver.get("http://checkyourflight.info")

        # Set the departure date
        input_date(driver, '2019-08-14')
        sleep(1)  # Allow time for page updates
        
        # Print flight options
        print("Flight options:")
        fetch_and_print_dropdown_options(driver, 'flight')
        
        # Process each flight option and print associated hotel options
        print("Processing flight options and related hotels...")
        process_flight_options(driver)

        # Pause to observe results before closing
        sleep(100)
    finally:
        driver.quit()  # Ensure the browser closes properly

if __name__ == "__main__":
    main()
