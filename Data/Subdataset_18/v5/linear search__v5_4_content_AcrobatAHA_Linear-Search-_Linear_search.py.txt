def linear_search(my_list, search_num):
    """
    This function performs a linear search on a list.
    
    Parameters:
    my_list (list): The list to search through.
    search_num (int): The number to search for.
    
    Returns:
    tuple: A tuple containing a boolean indicating if the number was found,
           and the index of the found number or -1 if not found.
    """
    for index, num in enumerate(my_list):
        if num == search_num:
            return True, index
    return False, -1

def get_user_input_list():
    """
    This function prompts the user to enter a list of numbers.
    
    Returns:
    list: A list of integers entered by the user.
    """
    user_input = input('Enter the numbers for your list, separated by spaces: ')
    return [int(num) for num in user_input.split()]

def main():
    """
    This is the main function that controls the program flow.
    """
    while True:
        run_program = input('Enter ** YES ** to run the searching program: ')
        if run_program.strip().upper() == 'YES':
            user_list = get_user_input_list()
            print('\n\n\t **********************************************************\n')
            print('Your list is:', user_list)
            
            search_num = int(input('\n\nSearch for a number: '))
            found, index = linear_search(user_list, search_num)
            
            if found:
                print(f'\n\t{search_num} is found at index {index}.')
            else:
                print(f'\n\t\t{search_num} is not in the list. Better luck next time.')
                print('\t(If you want to search another number, please search again. Otherwise, type 0 for exit)')
                
            continue_search = input('Do you want to search another number? (YES/NO): ')
            if continue_search.strip().upper() != 'YES':
                break
        else:
            break

if __name__ == '__main__':
    main()
