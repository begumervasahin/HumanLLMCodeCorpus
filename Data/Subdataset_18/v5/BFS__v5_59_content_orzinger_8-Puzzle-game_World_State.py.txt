class WorldState:
    """
    Represents the state of the 8-puzzle world.

    Attributes:
        state (list of list): The current configuration of the puzzle.
        operator (str): The move that led to this state.
        parent (WorldState): The parent state from which this state was derived.
        depth (int): The depth of this state in the search tree.
        f_value (float): The sum of g(x) + h(x) for the A* algorithm, where g(x) is the cost to reach this state
                         and h(x) is the heuristic estimate of the cost to reach the goal.

    Methods:
        __eq__: Compares two states for equality.
        __lt__: Compares two states to determine which has a lower f_value.
    """
    def __init__(self, state, operator=None, parent=None, depth=0, f_value=0.0):
        self.state = state
        self.operator = operator
        self.parent = parent
        self.depth = depth
        self.f_value = f_value
    
    def __eq__(self, other):
        """
        Compare two WorldState instances for equality.

        :param other: Another WorldState instance.
        :return: True if the states are equal, False otherwise.
        """
        return self.state == other.state
    
    def __lt__(self, other):
        """
        Compare two WorldState instances to determine which has a lower f_value.

        :param other: Another WorldState instance.
        :return: True if this instance has a lower f_value than the other, False otherwise.
        """
        return self.f_value < other.f_value

    def __repr__(self):
        """
        Provide a string representation of the WorldState instance for debugging purposes.

        :return: A string representation of the WorldState instance.
        """
        return f"WorldState(state={self.state}, operator={self.operator}, depth={self.depth}, f_value={self.f_value})"
