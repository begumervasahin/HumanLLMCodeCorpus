from mpi4py import MPI
import numpy as np
import time
from operator import itemgetter

def bubble_sort(nums):
    """
    Sorts a list of numbers using the bubble sort algorithm.

    Args:
        nums (list): List of numbers to be sorted.
    """
    swapped = True
    while swapped:
        swapped = False
        for i in range(len(nums) - 1):
            if nums[i] > nums[i + 1]:
                nums[i], nums[i + 1] = nums[i + 1], nums[i]
                swapped = True

def main():
    # Initialize MPI
    comm = MPI.COMM_WORLD
    size = comm.Get_size()
    rank = comm.Get_rank()

    if rank == 0:
        # Input array size from user
        array_size = int(input("Please enter array size: "))
        
        # Generate and shuffle the array
        numbers = np.arange(array_size)
        np.random.shuffle(numbers)
        print(f"Generated list of size {array_size} is: {numbers}")
        
        # Split the array into chunks for each process
        chunks = np.array_split(numbers, size)
    else:
        chunks = None

    # Scatter the chunks to all processes
    start_time = time.time()
    chunk = comm.scatter(chunks, root=0)
    print(f"Process {rank} has this chunk of data: {chunk}")

    # Sort each chunk using bubble sort
    bubble_sort(chunk)

    # Gather the sorted chunks back at the root process
    sorted_arrays = comm.gather(chunk, root=0)

    if rank == 0:
        # Merge the sorted chunks into a single sorted array
        sorted_array = merge_sorted_chunks(sorted_arrays, array_size, size)
        
        # Print the sorted array and execution time
        print(f"\n\nSorted Array: {sorted_array}")
        print(f"\n\nExecution Time: {time.time() - start_time} seconds")

def merge_sorted_chunks(sorted_arrays, array_size, size):
    """
    Merges sorted chunks from different processes into a single sorted array.

    Args:
        sorted_arrays (list): List of sorted arrays from different processes.
        array_size (int): Total size of the final sorted array.
        size (int): Number of processes.

    Returns:
        list: The merged and sorted array.
    """
    iterator_indices = np.zeros(size, dtype=int)
    sorted_array = []

    for _ in range(array_size):
        current_elements = [
            (i, (float('inf') if iterator_indices[i] >= len(sorted_arrays[i]) else sorted_arrays[i][iterator_indices[i]]))
            for i in range(size)
        ]
        min_element = min(current_elements, key=itemgetter(1))
        iterator_indices[min_element[0]] += 1
        sorted_array.append(min_element[1])

    return sorted_array

if __name__ == "__main__":
    main()
