def is_perfect_number(num):
    """
    Check if a number is a perfect number.
    
    Parameters:
    num (int): The number to check.
    
    Returns:
    bool: True if the number is perfect, False otherwise.
    """
    sum_of_divisors = sum(divisor for divisor in range(1, num) if num % divisor == 0)
    return sum_of_divisors == num

def find_perfect_numbers(limit):
    """
    Find and print all perfect numbers up to a given limit.
    
    Parameters:
    limit (int): The upper limit (exclusive) to search for perfect numbers.
    """
    for num in range(2, limit):
        if is_perfect_number(num):
            print(num)

def main():
    """
    Main function to prompt user input and find perfect numbers up to the specified limit.
    
    Returns:
    None
    """
    try:
        limit = int(input("Input the range number: "))
        find_perfect_numbers(limit)
    except ValueError:
        print("Please enter a valid integer.")

if __name__ == "__main__":
    main()
