import random

def is_prime(num, iterations):
    """
    Checks if a number is prime using the Miller-Rabin primality test.

    Args:
        num (int): The number to check.
        iterations (int): The number of iterations for the primality test.

    Returns:
        bool: True if the number is likely prime, False otherwise.
    """
    if num <= 1:
        return False
    if iterations >= num:
        iterations = num - 1

    for _ in range(iterations):
        a = random.randint(1, num - 1)
        if pow(a, num - 1, num) != 1:
            return False
    return True

def generate_large_prime(bit_length):
    """
    Generates a large prime number with the specified bit length.

    Args:
        bit_length (int): The number of bits.

    Returns:
        int: A large prime number.
    """
    while True:
        candidate = random.randint(2**(bit_length-1), 2**bit_length - 1)
        if is_prime(candidate, 1000):
            return candidate

def generate_private_key(bit_length):
    """
    Generates a private key with the specified bit length.

    Args:
        bit_length (int): The number of bits.

    Returns:
        int: A private key.
    """
    return random.randint(2**(bit_length-1), 2**bit_length - 1)

def diffie_hellman_key_exchange():
    """
    Performs Diffie-Hellman key exchange and generates shared secret keys for Alice and Bob.
    """
    prime_bit_length = 10  # Bit length for prime numbers
    private_key_bit_length = 15  # Bit length for private keys
    
    # Generate two large prime numbers
    prime1 = generate_large_prime(prime_bit_length)
    print("Prime 1:", prime1)
    
    prime2 = generate_large_prime(prime_bit_length)
    print("Prime 2:", prime2)

    # Generate private keys for Alice and Bob
    alice_private_key = generate_private_key(private_key_bit_length)
    bob_private_key = generate_private_key(private_key_bit_length)

    # Generate public keys
    alice_public_key = pow(prime2, alice_private_key, prime1)
    bob_public_key = pow(prime2, bob_private_key, prime1)

    # Generate shared secret keys
    alice_shared_key = pow(bob_public_key, alice_private_key, prime1)
    bob_shared_key = pow(alice_public_key, bob_private_key, prime1)

    # Print the shared secret keys
    print("Alice's Shared Secret Key:", alice_shared_key)
    print("Bob's Shared Secret Key:", bob_shared_key)

if __name__ == "__main__":
    diffie_hellman_key_exchange()
