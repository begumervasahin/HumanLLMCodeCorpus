def update_cell(row, col, board):
    """
    Return a list of possible values that can be placed in the cell at (row, col) without violating Sudoku rules.
    
    Args:
        row (int): The row index of the cell.
        col (int): The column index of the cell.
        board (list of list of str): The Sudoku board.
    
    Returns:
        list of str: A list of valid values for the cell at (row, col).
    """
    choices = []
    for k in [str(v) for v in range(1, 10)]:
        if is_valid_choice(row, col, k, board):
            choices.append(k)
    return choices

def is_valid_choice(row, col, value, board):
    """
    Check if placing a value in the cell at (row, col) is valid according to Sudoku rules.
    
    Args:
        row (int): The row index of the cell.
        col (int): The column index of the cell.
        value (str): The value to be placed in the cell.
        board (list of list of str): The Sudoku board.
    
    Returns:
        bool: True if the value can be placed in the cell, False otherwise.
    """
    for y in range(9):
        if (board[row][y] == value or 
            board[y][col] == value or 
            board[row - row % 3 + y // 3][col - col % 3 + y % 3] == value):
            return False
    return True

def print_board(board):
    """
    Print the Sudoku board.
    
    Args:
        board (list of list of str): The Sudoku board.
    """
    for row in board:
        print(" ".join(row))
    print()

def sudoku_solve(board):
    """
    Solve the Sudoku puzzle using backtracking.
    
    Args:
        board (list of list of str): The Sudoku board.
    
    Returns:
        bool: True if the puzzle is solved, False otherwise.
    """
    row, col, choices = find_best_cell(board)
    
    if choices is None:
        print_board(board)
        return True

    for value in choices:
        board[row][col] = value
        if sudoku_solve(board):
            return True
        board[row][col] = '.'

    return False

def find_best_cell(board):
    """
    Find the best cell to fill next based on the fewest possible choices.
    
    Args:
        board (list of list of str): The Sudoku board.
    
    Returns:
        tuple: A tuple containing the row index, column index, and a list of possible values for the best cell to fill.
    """
    choices = None
    row, col = -1, -1
    
    for i in range(9):
        for j in range(9):
            if board[i][j] == '.':
                temp_choices = update_cell(i, j, board)
                if choices is None or (len(temp_choices) > 0 and len(temp_choices) < len(choices)):
                    choices = temp_choices
                    row, col = i, j
    
    return row, col, choices
