from flask import Flask, render_template, request
from forms import SearchForm
import math
from tfidf import search as Searching

# Initialize the Flask application
app = Flask(__name__)
app.secret_key = 'development'

def get_pagination_bounds(page, pagination_size, data_length):
    """Calculate the start and end indices for pagination."""
    start = (page - 1) * pagination_size
    end = min(page * pagination_size, data_length)
    return start, end

def calculate_total_pages(data_length, pagination_size):
    """Calculate the total number of pages for the given data length."""
    return math.ceil(data_length / pagination_size)

@app.route("/")
def index():
    """Render the index page with the search form."""
    form = SearchForm()
    return render_template("index.html", form=form)

@app.route("/search")
def search():
    """Handle the search functionality and render the search results page."""
    form = SearchForm()
    query = request.args.get('keyword')
    page = int(request.args.get('page', 1))  # Default to page 1 if not specified

    # Perform the search using the provided query
    data = Searching.search_for(query)
    data_length = len(data)

    # Pagination settings
    pagination_size = 5
    start, end = get_pagination_bounds(page, pagination_size, data_length)
    total_pages = calculate_total_pages(data_length, pagination_size)

    # Render the search results page with pagination
    return render_template(
        "search.html",
        form=form,
        data=data[start:end],
        keyword=query,
        page=page,
        start=start,
        end=end,
        total=total_pages
    )

if __name__ == '__main__':
    app.run(debug=True)
