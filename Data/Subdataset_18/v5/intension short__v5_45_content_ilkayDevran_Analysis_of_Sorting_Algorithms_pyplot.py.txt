import selection_sort
import insertion_sort
import MERGE
import allInOne
import time
import random


def main():
    """Main function to select and execute the sorting algorithm or draw runtime graphs."""
    display_menu()
    
    choice = get_user_choice()
    if choice is None:
        return

    if choice == 4:
        allInOne.drawAllGraph()
    else:
        array_size, array_range = get_array_parameters()
        array = generate_random_array(array_size, array_range)
        execute_sort(choice, array)


def display_menu():
    """Displays the menu options for sorting and graph drawing."""
    print("""
    Choose the type of sort to get runtime with specific variables:
    1. Insertion Sort
    2. Selection Sort
    3. Merge Sort
    4. Draw runtime graphs for all sorts with 10**k where k = 1, 2, 3, 4, 5
    """)


def get_user_choice():
    """Gets the user's choice of sorting algorithm or graph drawing."""
    try:
        return int(input('>>> '))
    except ValueError:
        print("Invalid input. Please enter a number between 1 and 4.")
        return None


def get_array_parameters():
    """Prompts the user for array size and range parameters."""
    size_exponent = get_input('Enter the array size exponent (10**size): ', default=0)
    range_exponent = get_input('Enter the array range exponent (0 to 10**range): ', default=0)
    print(f'\nArray Size: {10 ** size_exponent}, Range: 0 - {10 ** range_exponent}')
    return 10 ** size_exponent, 10 ** range_exponent


def get_input(prompt, default):
    """Gets integer input from the user, returning a default value if input is invalid."""
    try:
        return int(input(prompt))
    except ValueError:
        print(f"Invalid input. Using default value: {default}")
        return default


def generate_random_array(size, rng):
    """Generates a random array of specified size and range."""
    return [random.randint(0, rng) for _ in range(size)]


def execute_sort(choice, data):
    """Executes the chosen sorting algorithm and measures its runtime."""
    sort_functions = {
        1: (insertion_sort.insertionSort, "Insertion Sort"),
        2: (selection_sort.selectionSort, "Selection Sort"),
        3: (MERGE.mergeSort, "Merge Sort")
    }
    
    if choice in sort_functions:
        sort_function, sort_name = sort_functions[choice]
        measure_runtime(sort_function, data, sort_name)
    else:
        print("Invalid choice. Please select a valid sorting algorithm.")


def measure_runtime(sort_function, data, sort_name):
    """Measures and prints the runtime of the given sorting function."""
    start_time = time.time()
    sort_function(data)
    elapsed_time = time.time() - start_time
    print(f"\n{sort_name} --- {elapsed_time:.6f} seconds ---\n")


if __name__ == '__main__':
    main()
