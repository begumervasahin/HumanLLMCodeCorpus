def zigzag(n):
    """
    Generates a dictionary with coordinates as keys and zigzag order as values.

    Args:
    n (int): The size of the n x n grid.

    Returns:
    dict: A dictionary where keys are (x, y) tuples and values are the order in the zigzag pattern.
    """
    index_order = sorted(
        ((x, y) for x in range(n) for y in range(n)),
        key=lambda xy: (xy[0] + xy[1], -xy[1] if (xy[0] + xy[1]) % 2 else xy[1])
    )
    return {index: num for num, index in enumerate(index_order)}

def print_zz(zigzag_dict):
    """
    Prints the zigzag ordered grid in a formatted way.

    Args:
    zigzag_dict (dict): The dictionary containing the zigzag order.
    """
    n = int(len(zigzag_dict) ** 0.5 + 0.5)
    for x in range(n):
        for y in range(n):
            print(f"{zigzag_dict[(x, y)]:2}", end=' ')
        print()

# Example usage
if __name__ == "__main__":
    print_zz(zigzag(6))
