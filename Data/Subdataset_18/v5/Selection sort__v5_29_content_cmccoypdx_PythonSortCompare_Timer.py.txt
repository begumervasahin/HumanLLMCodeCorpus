import timeit

def wrapper(func, *args, **kwargs):
    """
    Wrap the function and its arguments in another function
    to delay its execution.

    Parameters:
    func (function): The function to be wrapped.
    *args: Variable length argument list for the function.
    **kwargs: Arbitrary keyword arguments for the function.

    Returns:
    function: A wrapped function that includes the provided arguments.
    """
    def wrapped():
        return func(*args, **kwargs)
    return wrapped

def time_function(func, *args, **kwargs):
    """
    Measure the execution time of a function with the provided arguments.

    Parameters:
    func (function): The function to be timed.
    *args: Variable length argument list for the function.
    **kwargs: Arbitrary keyword arguments for the function.

    Returns:
    float: The time taken to execute the function once.
    """
    wrapped = wrapper(func, *args, **kwargs)
    return timeit.timeit(wrapped, number=1)

# Example usage
if __name__ == "__main__":
    def sample_function(x, y):
        """
        Sample function for demonstration purposes.

        Parameters:
        x (int): The first integer to add.
        y (int): The second integer to add.

        Returns:
        int: The sum of x and y.
        """
        return x + y

    # Time the sample function with arguments 10 and 20
    elapsed_time = time_function(sample_function, 10, 20)
    print(f"Elapsed time: {elapsed_time:.6f} seconds")
