import unittest

def check_pesel(pesel_nr):
    """
    Validates a PESEL number using the standard PESEL validation algorithm.

    Parameters:
    - pesel_nr (str): The PESEL number to be validated.

    Returns:
    - bool: True if the PESEL number is valid, False otherwise.
    """
    # Validate PESEL number length and content
    if len(pesel_nr) != 11 or not pesel_nr.isdigit():
        return False

    # Weights for each digit position in the PESEL number
    weights = [1, 3, 7, 9, 1, 3, 7, 9, 1, 3]
    
    # Calculate the weighted sum of the first 10 digits
    weighted_sum = sum(int(pesel_nr[i]) * weights[i] for i in range(10))

    # Compute the expected check digit
    expected_check_digit = (10 - (weighted_sum % 10)) % 10

    # Compare the computed check digit with the last digit of the PESEL number
    return expected_check_digit == int(pesel_nr[-1])

class TestPeselValidator(unittest.TestCase):
    def test_valid_pesel(self):
        """Test with known valid PESEL numbers."""
        valid_pesels = ["92121510595", "99810235403"]
        for pesel in valid_pesels:
            with self.subTest(pesel=pesel):
                self.assertTrue(check_pesel(pesel))

    def test_invalid_pesel(self):
        """Test with known invalid PESEL numbers."""
        invalid_pesels = [
            "44051401358",  # Incorrect check digit
            "51401358",     # Too short
            "12345678901"   # Incorrect check digit
        ]
        for pesel in invalid_pesels:
            with self.subTest(pesel=pesel):
                self.assertFalse(check_pesel(pesel))

if __name__ == "__main__":
    unittest.main()
