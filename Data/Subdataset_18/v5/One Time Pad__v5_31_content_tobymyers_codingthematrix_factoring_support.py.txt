import operator
from functools import reduce

def gcd(x, y):
    """Calculate the greatest common divisor of x and y."""
    while y:
        x, y = y, x % y
    return x

def factorize(x, primeset):
    """Factorize x over a given set of primes.

    Args:
        x (int): The number to factorize.
        primeset (set): A set of prime numbers.

    Returns:
        list: A list of tuples (prime, exponent) where prime is a prime factor and exponent is its exponent.
              Returns an empty list if x cannot be fully factorized over the primeset.
    """
    factors = []
    for prime in primeset:
        exponent = 0
        while x % prime == 0:
            exponent += 1
            x //= prime
        if exponent > 0:
            factors.append((prime, exponent))
    return factors if x == 1 else []

def generate_primes(limit):
    """Generate a set of prime numbers up to a given limit using the Sieve of Eratosthenes.

    Args:
        limit (int): The upper limit for generating prime numbers.

    Returns:
        set: A set of prime numbers up to the given limit.
    """
    sieve = [True] * limit
    sieve[0] = sieve[1] = False  # 0 and 1 are not primes
    primeset = set()
    
    for num in range(2, limit):
        if sieve[num]:
            primeset.add(num)
            for multiple in range(num * num, limit, num):
                sieve[multiple] = False
    return primeset

def int_sqrt(x):
    """Calculate the integer square root of x.

    Args:
        x (int): The number to calculate the square root of.

    Returns:
        int: The integer square root of x.
    """
    low, high = 1, x
    while high - low > 1:
        mid = (low + high) // 2
        if mid * mid <= x:
            low = mid
        else:
            high = mid
    return low if low * low == x else high

def product(factors):
    """Calculate the product of a list of numbers.

    Args:
        factors (list): A list of numbers to multiply.

    Returns:
        int: The product of the numbers in the list.
    """
    return reduce(operator.mul, factors, 1)
