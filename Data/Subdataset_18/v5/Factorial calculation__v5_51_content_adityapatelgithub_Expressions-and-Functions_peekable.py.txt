class Peekable:
    """
    An iterator that allows peeking at the next value without advancing the iterator.
    """
    def __init__(self, iterator):
        """
        Initializes the Peekable iterator with an existing iterator.

        :param iterator: An existing iterator to enhance with peek functionality.
        """
        self._iterator = iterator
        self._peeked = None

    def __iter__(self):
        """
        Returns the iterator object itself.
        """
        return self

    def __next__(self):
        """
        Returns the next element from the iterator. If an element has been peeked at,
        it returns the peeked value and resets the peeked value.

        :return: The next element from the iterator.
        :raises StopIteration: If the iterator is exhausted.
        """
        if self._peeked is None:
            self._peeked = next(self._iterator)
        result = self._peeked
        self._peeked = None
        return result

    def peek(self):
        """
        Returns the next element without advancing the iterator. If the next element
        has not been peeked at yet, it retrieves it from the iterator.

        :return: The next element from the iterator.
        """
        if self._peeked is None:
            self._peeked = next(self._iterator)
        return self._peeked

def peek(iterator):
    """
    Returns the next element of a Peekable iterator without advancing it.

    :param iterator: A Peekable iterator instance.
    :return: The next element from the iterator.
    """
    return iterator.peek()

# Example usage
if __name__ == "__main__":
    # Create a Peekable iterator with a list of numbers
    peekable_iter = Peekable(iter([1, 2, 3, 4, 5]))

    # Demonstrate peek and next functionality
    print(peek(peekable_iter))  # Output: 1
    print(peek(peekable_iter))  # Output: 1
    print(next(peekable_iter))  # Output: 1
    print(next(peekable_iter))  # Output: 2
    print(next(peekable_iter))  # Output: 3
    print(peek(peekable_iter))  # Output: 4
    print(next(peekable_iter))  # Output: 4
    print(list(Peekable(iter([1, 2, 3, 4, 5]))))  # Output: [1, 2, 3, 4, 5]
