import os

hindi_dic_file = './datasets/hi_hdtb-ud-dev.conllu'
hindi_out_file = './outputs/train_data.txt'

def are_lemmas_equal(lemma1, lemma2):
    """
    Compare two lemmas to check if they are equal based on their root and tags.
    
    Args:
        lemma1 (dict): First lemma with keys 'root', 'tag1', and 'tag2'.
        lemma2 (dict): Second lemma with keys 'root', 'tag1', and 'tag2'.
        
    Returns:
        bool: True if all keys in both lemmas are equal, otherwise False.
    """
    return all(lemma1[key] == lemma2[key] for key in ["root", "tag1", "tag2"])

def serialize_lemma(lemma):
    """
    Convert a lemma dictionary to a formatted string.
    
    Args:
        lemma (dict): Lemma with keys 'root', 'tag1', and 'tag2'.
        
    Returns:
        str: Serialized string representation of the lemma.
    """
    return '[{}, {}, {}]'.format(lemma["root"], lemma["tag1"], lemma["tag2"])

def read_training_data(input_training_file, dest_file=None):
    """
    Read and process the training data from a file, optionally writing the processed data to a destination file.
    
    Args:
        input_training_file (str): Path to the input training data file.
        dest_file (str, optional): Path to the destination file to write processed data. Defaults to None.
        
    Returns:
        dict: Processed dataset with words as keys and lists of unique lemmas as values.
    """
    dataset = {}

    with open(input_training_file, 'r') as hindi_file:
        for line in hindi_file:
            # Skip lines that are too short or empty
            if len(line) < 5 or line.startswith('\n'):
                continue

            # Extract relevant fields
            fields = line.split("\t")
            if len(fields) < 5:
                continue
            word, root, tag1, tag2 = fields[1:5]

            # Initialize the word entry in the dataset if it doesn't exist
            if word not in dataset:
                dataset[word] = []

            # Create a new lemma dictionary
            new_lemma = {"root": root, "tag1": tag1, "tag2": tag2}

            # Add the new lemma if it is unique for the word
            if not any(are_lemmas_equal(existing_lemma, new_lemma) for existing_lemma in dataset[word]):
                dataset[word].append(new_lemma)

    # Optionally write the processed dataset to the destination file
    if dest_file:
        with open(dest_file, 'w') as out_file:
            for word, lemmas in dataset.items():
                serialized_lemmas = ', '.join(serialize_lemma(lemma) for lemma in lemmas)
                out_file.write(f'{word}:{serialized_lemmas}\n')

    return dataset

def supervised_learn():
    """
    Main function to read training data and process each word with its lemmas.
    """
    training_data = read_training_data(hindi_dic_file, hindi_out_file)

    for word, lemmas in training_data.items():
        print(f'Processing word: {word} with {len(lemmas)} lemmas')

# Ensure the output directory exists
os.makedirs(os.path.dirname(hindi_out_file), exist_ok=True)

# Execute the main function
supervised_learn()
