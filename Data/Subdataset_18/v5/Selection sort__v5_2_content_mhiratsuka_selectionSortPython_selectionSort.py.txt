def selection_sort_descending(arr):
    """
    Sorts a list of numbers in descending order using the selection sort algorithm.
    
    Args:
        arr (list): The list of numbers to be sorted.
    
    Returns:
        list: The sorted list in descending order.
    """
    n = len(arr)
    for i in range(n - 1):
        # Assume the current position holds the largest number
        max_index = i
        # Check the rest of the array for a larger number
        for j in range(i + 1, n):
            if arr[j] > arr[max_index]:
                max_index = j
        # Swap the found maximum element with the current element
        if max_index != i:
            arr[i], arr[max_index] = arr[max_index], arr[i]
    return arr

def test_selection_sort_descending():
    """
    Test the selection_sort_descending function to ensure it sorts the list correctly in descending order.
    """
    test_numbers = [0, 2, 100, 0, 120, 11]
    sorted_numbers = selection_sort_descending(test_numbers[:])
    expected_result = [120, 100, 11, 2, 0, 0]
    assert sorted_numbers == expected_result, f"Test failed: {sorted_numbers}"
    print("Test passed!")

if __name__ == "__main__":
    numbers = [0, 2, 100, 0, 120, 11]

    print("Sorted list:", sorted_numbers)  # Output: [120, 100, 11, 2, 0, 0]
