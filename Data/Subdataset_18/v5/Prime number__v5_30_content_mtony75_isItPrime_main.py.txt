def is_valid_integer(value):
    """
    Check if the provided value can be converted to an integer.
    
    Args:
        value (str): The value to check.
        
    Returns:
        bool: True if the value is a valid integer, False otherwise.
    """
    try:
        int(value)
        return True
    except ValueError:
        print("User input is not a valid integer.")
        return False

def is_prime(number):
    """
    Check if the given number is a prime number.
    
    Args:
        number (int): The number to check.
        
    Returns:
        bool: True if the number is prime, False otherwise.
    """
    if number <= 1:
        print(f"{number} is not a prime number.")
        return False
    
    for divisor in range(2, int(number ** 0.5) + 1):
        if number % divisor == 0:
            print(f"{number} is not a prime number.")
            return False
    
    print(f"{number} is a prime number.")
    return True

def main():
    """
    Main loop to get user input and check if it's a valid integer and if it's prime.
    """
    while True:
        user_input = input("Enter a number (or type 'stop' to exit): ").strip()
        
        if user_input.lower() == 'stop':
            break
        
        if is_valid_integer(user_input):
            is_prime(int(user_input))

if __name__ == "__main__":
    main()
