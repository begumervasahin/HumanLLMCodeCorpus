import numpy as np

class KMeans:
    """Implementation of the k-Means clustering algorithm."""

    def __init__(self, k, data):
        """
        Initializes the KMeans instance.

        Parameters:
        k (int): The number of clusters.
        data (np.ndarray): The input data for clustering.
        """
        self.k = k
        self.n_data, self.n_dim = data.shape
        self.centres = None

    def train(self, data, max_iterations=10):
        """
        Trains the k-Means model on the given data.

        Parameters:
        data (np.ndarray): The input data for clustering.
        max_iterations (int): The maximum number of iterations to run the algorithm.

        Returns:
        np.ndarray: The coordinates of the cluster centers.
        """
        minima = data.min(axis=0)
        maxima = data.max(axis=0)
        
        # Initialize the cluster centers randomly within the range of data
        self.centres = np.random.rand(self.k, self.n_dim) * (maxima - minima) + minima
        old_centres = np.zeros_like(self.centres)
        
        count = 0

        while not np.allclose(self.centres, old_centres) and count < max_iterations:
            old_centres = self.centres.copy()
            count += 1
            
            # Calculate distances from data points to cluster centers
            distances = np.linalg.norm(data[:, np.newaxis] - self.centres, axis=2)
            cluster_labels = np.argmin(distances, axis=1)
            
            # Update cluster centers based on the mean of points assigned to each cluster
            for i in range(self.k):
                points_in_cluster = data[cluster_labels == i]
                if points_in_cluster.size > 0:
                    self.centres[i] = points_in_cluster.mean(axis=0)

        return self.centres

    def predict(self, data):
        """
        Predicts the cluster for each data point.

        Parameters:
        data (np.ndarray): The input data for prediction.

        Returns:
        np.ndarray: The cluster labels for each data point.
        """
        distances = np.linalg.norm(data[:, np.newaxis] - self.centres, axis=2)
        cluster_labels = np.argmin(distances, axis=1)
        return cluster_labels
