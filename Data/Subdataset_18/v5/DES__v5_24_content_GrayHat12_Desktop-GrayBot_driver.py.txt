import GrayAPI
import colorama as clr

def fetch_and_display_news(topic):
    """
    Fetch and display news articles for a given topic using GrayAPI.

    Parameters
    ----------
    topic : str
        The topic to fetch news articles on.
    """
    # Create a GrayNews object for the specified topic
    news = GrayAPI.GrayNews(topic)

    # Make the request and print the status
    print(f'MAKE REQUEST: {news.makeRequest()}')
    print(f'REQ ERR: {news.getRequestError()}')
    print(f'SOURCE: {news.getSource()}\n')

    # Get the list of articles
    articles = news.getArticles()
    print(f'LENGTH: {len(articles)}\n')

    # Print each article's details
    for article in articles:
        print_article_details(article)

def print_article_details(article):
    """
    Print the details of a news article.

    Parameters
    ----------
    article : GrayAPI.Article
        The article to print details for.
    """
    details = [
        ('TITLE', article.getTitle()),
        ('AUTHOR', article.getAuthor()),
        ('URL', article.getUrl()),
        ('DESCRIPTION', article.getShortDesc()),
        ('IMAGE', article.getImage()),
        ('PUBLISHED', article.getPublishedAt())
    ]

    for label, value in details:
        print(f"{clr.Fore.RED}{label}: {clr.Fore.GREEN}{value}")
    print('\n')

def main():
    """
    Main function to initialize colorama and fetch/display news based on user input.
    """
    # Initialize Colorama
    clr.init(autoreset=True)

    # Get user input for the news topic
    topic = input('Input the Topic to get News On: ')

    # Fetch and display news for the given topic
    fetch_and_display_news(topic)

if __name__ == "__main__":
    main()
