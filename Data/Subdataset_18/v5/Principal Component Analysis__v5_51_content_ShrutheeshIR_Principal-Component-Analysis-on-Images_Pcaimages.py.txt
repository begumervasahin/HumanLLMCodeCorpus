import cv2
import os
import numpy as np
import scipy.linalg as sl
import matplotlib.pyplot as plt

def read_images(directory, max_images=3000, image_size=(64, 64)):
    """
    Reads and resizes images from a specified directory.
    
    Parameters:
        directory (str): Directory containing images.
        max_images (int): Maximum number of images to read.
        image_size (tuple): Size to resize images to.
        
    Returns:
        list: List of resized images.
    """
    images = []
    for i, file in enumerate(os.listdir(directory)):
        if i < max_images:
            img = cv2.imread(os.path.join(directory, file), 0)  # Read image in grayscale
            if img is not None:
                img = cv2.resize(img, image_size)  # Resize image
                images.append(img)
    return images

def create_data_matrix(images):
    """
    Creates a data matrix from a list of images.
    
    Parameters:
        images (list): List of images.
        
    Returns:
        np.ndarray: Data matrix where each row is a flattened image.
    """
    print("Creating data matrix", end=" ... ")
    
    num_images = len(images)
    img_shape = images[0].shape
    data_matrix = np.zeros((num_images, img_shape[0] * img_shape[1]), dtype=np.float32)
    
    for i in range(num_images):
        image = images[i].flatten()
        data_matrix[i, :] = image
    
    print("DONE")
    return data_matrix

def compute_eigenfaces(data_matrix):
    """
    Computes eigenfaces from the data matrix.
    
    Parameters:
        data_matrix (np.ndarray): Data matrix where each row is a flattened image.
        
    Returns:
        np.ndarray: Eigenvalues sorted in descending order.
        np.ndarray: Corresponding eigenvectors.
    """
    covariance_matrix = np.cov(data_matrix, rowvar=False)
    print(f"Covariance matrix shape: {covariance_matrix.shape}")
    
    eigen_values, eigen_vectors = sl.eig(covariance_matrix)
    
    sorted_indices = eigen_values.argsort()[::-1]
    eigen_values = eigen_values[sorted_indices]
    eigen_vectors = eigen_vectors[:, sorted_indices]
    
    return eigen_values, eigen_vectors

def reconstruct_image(eigen_vectors, image_vector, num_components=75):
    """
    Reconstructs an image using a specified number of eigenfaces.
    
    Parameters:
        eigen_vectors (np.ndarray): Eigenvectors sorted by eigenvalues.
        image_vector (np.ndarray): Flattened image vector to reconstruct.
        num_components (int): Number of eigenfaces to use.
        
    Returns:
        np.ndarray: Reconstructed image.
    """
    top_eigen_vectors = eigen_vectors[:, :num_components].real
    projected_image = top_eigen_vectors.T @ image_vector
    reconstructed_image = top_eigen_vectors @ projected_image
    return reconstructed_image

def display_images(original_image, reconstructed_image):
    """
    Displays the original and reconstructed images.
    
    Parameters:
        original_image (np.ndarray): Original image.
        reconstructed_image (np.ndarray): Reconstructed image.
    """
    plt.figure(figsize=(10, 5))
    
    plt.subplot(1, 2, 1)
    plt.imshow(original_image, cmap='gray')
    plt.title("Original Image")
    
    plt.subplot(1, 2, 2)
    plt.imshow(reconstructed_image, cmap='gray')
    plt.title("Reconstructed Image")
    
    plt.show()
