import re
from urllib import robotparser
from urllib.parse import urljoin
from datetime import timedelta
from downloader_requests_cache import Downloader
import requests_cache

def get_robots_parser(robots_url):
    """Return the robots.txt parser object using the given robots_url."""
    rp = robotparser.RobotFileParser()
    rp.set_url(robots_url)
    rp.read()
    return rp

def get_links(html):
    """Return a list of links from the HTML content using simple regex matching."""
    link_pattern = re.compile(r'<a[^>]+href=["\'](.*?)["\']', re.IGNORECASE)
    return link_pattern.findall(html)

def link_crawler(start_url, link_regex, robots_url=None, user_agent='wswp',
                 proxies=None, delay=3, max_depth=4, num_retries=2, expires=timedelta(days=30)):
    """
    Crawl from the given start URL following links matched by link_regex.
    
    Args:
        start_url (str): The website to start crawling from.
        link_regex (str): Regex pattern to match the links.
        
    Kwargs:
        robots_url (str): URL of the site's robots.txt (default: start_url + '/robots.txt').
        user_agent (str): User agent string (default: 'wswp').
        proxies (list of dicts): List of proxy dicts for HTTP/HTTPS proxies.
        delay (int): Seconds to wait between requests to the same domain (default: 3).
        max_depth (int): Maximum depth for crawling to avoid traps (default: 4).
        num_retries (int): Number of retry attempts for failed requests.
        expires (timedelta): Cache expiration timedelta (default: 30 days).
    """
    crawl_queue = [start_url]
    seen_urls = {}
    
    # Install cache with expiration time
    requests_cache.install_cache(backend='redis', expire_after=expires)
    
    # Determine robots.txt URL if not provided
    if not robots_url:
        robots_url = urljoin(start_url, '/robots.txt')
    
    # Initialize the robots.txt parser
    robots_parser = get_robots_parser(robots_url)
    
    # Initialize the downloader
    downloader = Downloader(delay=delay, user_agent=user_agent, proxies=proxies)
    
    while crawl_queue:
        url = crawl_queue.pop()
        
        # Check if the URL is allowed by robots.txt
        if robots_parser.can_fetch(user_agent, url):
            current_depth = seen_urls.get(url, 0)
            
            if current_depth == max_depth:
                print(f'Skipping {url} due to reaching max depth limit.')
                continue
            
            html_content = downloader(url, num_retries=num_retries)
            
            if not html_content:
                continue
            
            # Extract and enqueue links
            for link in get_links(html_content):
                if re.match(link_regex, link):
                    absolute_link = urljoin(start_url, link)
                    
                    if absolute_link not in seen_urls:
                        seen_urls[absolute_link] = current_depth + 1
                        crawl_queue.append(absolute_link)
        else:
            print(f'Blocked by robots.txt: {url}')
