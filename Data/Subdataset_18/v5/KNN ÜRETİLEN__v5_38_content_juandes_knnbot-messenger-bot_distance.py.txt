import math

def euclidean_distance(data1, data2):
    """
    Calculate the Euclidean distance between two data points.
    
    Parameters:
    data1 (list): The first data point.
    data2 (list): The second data point.
    
    Returns:
    float: The Euclidean distance between the two data points.
    """
    # Ensure both data points have the same length
    if len(data1) != len(data2):
        raise ValueError("Data points must have the same number of dimensions.")
    
    # Calculate the Euclidean distance
    sum_squared_diff = sum((a - b) ** 2 for a, b in zip(data1, data2))
    return math.sqrt(sum_squared_diff)

# Example usage
point1 = [2, 3, 4]
point2 = [5, 6, 7]
print("Distance:", euclidean_distance(point1, point2))
