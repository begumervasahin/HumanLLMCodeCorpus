import nltk
from nltk.corpus import stopwords
from nltk import word_tokenize, ngrams
from sklearn.feature_extraction.text import CountVectorizer, TfidfTransformer
import glob
import os

# Ensure necessary NLTK resources are downloaded
nltk.download('punkt')
nltk.download('stopwords')

def extract_kgrams(text, k):
    """
    Extract k-grams from the given text.

    Parameters:
    - text (str): The input text from which k-grams are to be extracted.
    - k (int): The length of the k-grams.

    Returns:
    - set: A set containing all unique k-grams.
    """
    tokens = word_tokenize(text)
    kgrams = {' '.join(gram) for gram in ngrams(tokens, k)}
    return kgrams

def vectorize_article(text, ngram_range=(3, 3)):
    """
    Vectorize the article using TF-IDF.

    Parameters:
    - text (str): The input text to be vectorized.
    - ngram_range (tuple): The range of n-grams to consider.

    Returns:
    - numpy.ndarray: The TF-IDF vectorized representation of the text.
    """
    tokens = word_tokenize(text)
    stop_words = set(stopwords.words("english"))
    filtered_tokens = [word for word in tokens if word.lower() not in stop_words]

    count_vectorizer = CountVectorizer(ngram_range=ngram_range)
    bag_of_words = count_vectorizer.fit_transform([' '.join(filtered_tokens)])
    
    tfidf_transformer = TfidfTransformer()
    tfidf_matrix = tfidf_transformer.fit_transform(bag_of_words)
    
    return tfidf_matrix.toarray()

def save_kgrams(kgrams, article_name):
    """
    Save the k-grams to a file.

    Parameters:
    - kgrams (set): The set of k-grams to save.
    - article_name (str): The name of the article (file) to save the k-grams for.
    """
    output_filename = f'{article_name}_kgrams.txt'
    with open(output_filename, "w") as f:
        f.write(",".join(kgrams))

def process_text_files(file_pattern, k=3):
    """
    Process all text files matching the given pattern, vectorize their content, 
    and save the k-grams.

    Parameters:
    - file_pattern (str): The pattern to match text files.
    - k (int): The length of the k-grams.
    """
    text_files = glob.glob(file_pattern)
    
    for filepath in text_files:
        with open(filepath, 'r', encoding='utf-8') as file:
            text = file.read()
            
            # Extract k-grams and vectorize the article
            kgrams = extract_kgrams(text, k)
            tfidf_vector = vectorize_article(text, ngram_range=(k, k))
            
            # Save the k-grams to a file
            base_filename = os.path.splitext(os.path.basename(filepath))[0]
            save_kgrams(kgrams, base_filename)

if __name__ == "__main__":
    # Replace '/somePath/*.txt' with the actual path pattern to your text files
    process_text_files('/somePath/*.txt', k=3)
