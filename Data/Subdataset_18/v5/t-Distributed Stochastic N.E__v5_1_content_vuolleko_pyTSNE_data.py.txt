import numpy as np
import struct

def read_MNIST(image_file, label_file):
    """
    Load the MNIST dataset from IDX format files for images and labels.

    Args:
        image_file (str): Path to the file containing the MNIST images in IDX format.
        label_file (str): Path to the file containing the MNIST labels in IDX format.

    Returns:
        tuple: A tuple containing:
            - np.ndarray: Array of images with shape (n_samples, n_pixels).
            - np.ndarray: Array of labels with shape (n_samples,).
    """
    images = _load_images(image_file)
    labels = _load_labels(label_file)
    return images, labels

def _load_images(image_file):
    """
    Load images from an IDX format file.

    Args:
        image_file (str): Path to the file containing the MNIST images in IDX format.

    Returns:
        np.ndarray: Array of images with shape (n_samples, n_pixels).
    """
    with open(image_file, 'rb') as f:
        _skip_bytes(f, 4)  # Skip the magic number
        
        # Read number of samples, rows, and columns
        n_samples = _read_int(f)
        n_rows = _read_int(f)
        n_cols = _read_int(f)
        n_pixels = n_rows * n_cols

        # Read and reshape image data
        images = np.fromfile(f, dtype=np.uint8).reshape(n_samples, n_pixels)

    return images

def _load_labels(label_file):
    """
    Load labels from an IDX format file.

    Args:
        label_file (str): Path to the file containing the MNIST labels in IDX format.

    Returns:
        np.ndarray: Array of labels with shape (n_samples,).
    """
    with open(label_file, 'rb') as f:
        _skip_bytes(f, 8)  # Skip the magic number and number of labels

        # Read label data
        labels = np.fromfile(f, dtype=np.uint8)

    return labels

def _skip_bytes(file, num_bytes):
    """
    Skip a specified number of bytes in a file.

    Args:
        file (file object): The file object to skip bytes in.
        num_bytes (int): The number of bytes to skip.
    """
    file.seek(num_bytes)

def _read_int(file):
    """
    Read a 4-byte integer from a file in big-endian format.

    Args:
        file (file object): The file object to read from.

    Returns:
        int: The integer value read from the file.
    """
    return struct.unpack('>I', file.read(4))[0]
