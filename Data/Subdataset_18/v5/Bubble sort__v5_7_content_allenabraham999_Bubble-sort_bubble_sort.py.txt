import random

def bubble_sort(arr):
    """
    Sorts an array using the bubble sort algorithm and prints the array after each pass.
    
    Parameters:
    arr (list): The list of elements to be sorted.
    """
    length = len(arr)
    for j in range(length):
        for i in range(1, length):
            if arr[i - 1] > arr[i]:
                # Swap the elements
                arr[i - 1], arr[i] = arr[i], arr[i - 1]
        print(f"Array after {j + 1} pass:")
        print_array(arr)

def print_array(arr):
    """
    Prints the elements of the array in a single line.
    
    Parameters:
    arr (list): The list of elements to be printed.
    """
    for elem in arr:
        print(elem, end=" ")
    print()

def fill_array(arr, size=8, lower_bound=1, upper_bound=9):
    """
    Fills an array with random integers within a specified range.
    
    Parameters:
    arr (list): The list to be filled with random integers.
    size (int): The number of elements to be added to the list.
    lower_bound (int): The lower bound of the random integers.
    upper_bound (int): The upper bound of the random integers.
    """
    for _ in range(size):
        arr.append(random.randint(lower_bound, upper_bound))
    print("The initial array is:")
    print_array(arr)

def main():
    """
    Main function to demonstrate filling an array with random integers and sorting it using bubble sort.
    """
    array = []
    fill_array(array)
    bubble_sort(array)
    print("\nThe final sorted array is:")
    print_array(array)

if __name__ == "__main__":
    main()
