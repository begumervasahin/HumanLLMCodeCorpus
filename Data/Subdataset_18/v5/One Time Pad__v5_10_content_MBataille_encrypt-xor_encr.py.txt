import os
import argparse

def generate_key(length):
    """Generate a random key of specified length."""
    key = ''.join(chr(65 + os.urandom(1)[0] % 26) for _ in range(length))
    return key

def str_xor(a, b):
    """XOR two strings and return the result."""
    return ''.join(chr(65 + (ord(a[i]) ^ ord(b[i]))) for i in range(len(a)))

def clean_text(text):
    """Clean the text by replacing specific characters with designated substitutes."""
    replacements = {" ": "a", ".": "b", ",": "c", "'": "d"}
    return text.upper().translate(str.maketrans(replacements))

def unclean_text(text):
    """Revert cleaned text back to its original form."""
    replacements = {"a": " ", "b": ".", "c": ",", "d": "'"}
    return text.translate(str.maketrans(replacements))

def main():
    parser = argparse.ArgumentParser(description='Encrypt or Decrypt data using One-Time Pad')
    parser.add_argument('-d', '--decrypt', action='store_true', help='Decrypt data (default is to encrypt)')
    args = parser.parse_args()

    if args.decrypt:
        key = input("Enter key: ")
        coded_text = input("Enter coded text: ")
        decrypted_text = unclean_text(str_xor(key, coded_text))
        print("Decrypted text:", decrypted_text)
    else:
        message = input("Enter message: ")
        cleaned_message = clean_text(message)
        key = generate_key(len(cleaned_message))
        encrypted_text = str_xor(cleaned_message, key)
        print("Generated key:", key)
        print("Encrypted text:", encrypted_text)

if __name__ == "__main__":
    main()
