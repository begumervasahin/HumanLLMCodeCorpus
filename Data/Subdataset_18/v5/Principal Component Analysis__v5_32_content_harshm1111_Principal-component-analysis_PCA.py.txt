import numpy as np
from scipy.spatial import distance
from sklearn.decomposition import PCA

def get_distance(distance_type, a, b):
    """
    Compute distance between two points based on the specified distance type.
    
    Parameters:
    distance_type (int): The type of distance metric to use.
    a (array-like): The first point.
    b (array-like): The second point.
    
    Returns:
    float: The computed distance between points a and b.
    """
    if distance_type == 1:
        return distance.minkowski(a, b, p=1)
    elif distance_type == 2:
        return distance.minkowski(a, b, p=2)
    elif distance_type == 3:
        return distance.minkowski(a, b, p=float('inf'))
    else:
        return distance.cosine(a, b)

def apply_pca(n_components, P_data, other_patient_data):
    """
    Apply PCA transformation to the data if required.
    
    Parameters:
    n_components (int): Number of principal components to use.
    P_data (list): Data of the primary patient.
    other_patient_data (list of lists): Data of other patients.
    
    Returns:
    tuple: Transformed P_data, transformed other_patient_data, explained variance ratio.
    """
    data = np.array([P_data] + other_patient_data)
    pca = PCA(n_components=n_components)
    pca.fit(data)
    P_data_transformed = pca.transform([P_data])[0]
    other_patient_data_transformed = pca.transform(other_patient_data)
    return P_data_transformed, other_patient_data_transformed, pca.explained_variance_

def find_closest_patients(P_data, other_patient_data, distance_type, X):
    """
    Compute and print the indices of the 5 closest patients, and PCA variance if applied.
    
    Parameters:
    P_data (list): Data of the primary patient.
    other_patient_data (list of lists): Data of other patients.
    distance_type (int): The type of distance metric to use.
    X (int): Number of principal components for PCA. If -1, PCA is not applied.
    """
    if X != -1:
        P_data, other_patient_data, variance = apply_pca(X, P_data, other_patient_data)
    
    distances = [
        (i, get_distance(distance_type, P_data, other_patient_data[i])) 
        for i in range(len(other_patient_data))
    ]
    closest_patients = sorted(distances, key=lambda x: x[1])[:5]

    for patient in closest_patients:
        print(patient[0] + 1)
    
    if X != -1:
        print(sum(variance))

def main():
    """
    Main function to read inputs and invoke the processing functions.
    """
    D = int(input("Enter the value for D: "))
    N = int(input("Enter the value for N: "))
    distance_type = int(input("Enter the distance type: "))
    X = int(input("Enter the value for X: "))

    P_data = list(map(int, input("Enter the P_data values: ").split()))
    other_patient_data = [
        list(map(int, input(f"Enter the data for patient {i + 1}: ").split())) 
        for i in range(N)
    ]

    find_closest_patients(P_data, other_patient_data, distance_type, X)

if __name__ == "__main__":
    main()
