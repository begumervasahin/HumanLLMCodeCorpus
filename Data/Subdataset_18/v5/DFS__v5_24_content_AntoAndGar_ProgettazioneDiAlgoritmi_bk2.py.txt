def generate_binary_strings(n, k):
    """
    Generate all binary strings of length n with at most k ones.
    :param n: int - The length of the binary strings.
    :param k: int - The maximum number of ones in the binary strings.
    """
    def backtrack(index, ones_count, current_string):
        """
        Helper function to perform the backtracking.
        :param index: int - The current index in the binary string.
        :param ones_count: int - The current count of ones in the binary string.
        :param current_string: List[str] - The current binary string being constructed.
        """
        if index == n:
            print(''.join(current_string))
            return

        # Add '0' and proceed to the next position
        current_string.append('0')
        backtrack(index + 1, ones_count, current_string)
        current_string.pop()
        
        # Add '1' and proceed to the next position if ones_count is less than k
        if ones_count < k:
            current_string.append('1')
            backtrack(index + 1, ones_count + 1, current_string)
            current_string.pop()

    backtrack(0, 0, [])

# Example usage:
generate_binary_strings(4, 1)
