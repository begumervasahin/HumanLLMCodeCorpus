Implementation of an edge priority queue using a d-ary heap.
"""

from edge import Edge  # Assuming Edge is a class with a 'weight' attribute

class DaryHeap:
    def __init__(self, d):
        """
        Initialize the d-ary heap.

        Args:
            d (int): The degree of the heap (number of children per node).
        """
        self.data = []
        self.d = d

    def make_heap(self, lst):
        """
        Convert a list into a heap.

        Args:
            lst (list of Edge): The list of edges to be turned into a heap.
        """
        self.data = lst
        for i in range(self.size() - 1, -1, -1):
            self.sift_down(i)

    def find_min(self):
        """
        Return the minimum element in the heap.

        Returns:
            Edge: The edge with the minimum weight, or None if the heap is empty.
        """
        return self.data[0] if self.data else None

    def delete(self, index):
        """
        Delete the element at a specified index in the heap.

        Args:
            index (int): The index of the element to be deleted.
        """
        if index == self.size() - 1:
            self.data.pop()
        else:
            last_item = self.data.pop()
            self.data[index] = last_item
            if last_item.weight < self.data[index].weight:
                self.sift_up(index)
            else:
                self.sift_down(index)

    def delete_min(self):
        """Delete the minimum element (root) from the heap."""
        if self.size() > 0:
            self.delete(0)

    def size(self):
        """
        Return the size of the heap.

        Returns:
            int: The number of elements in the heap.
        """
        return len(self.data)

    def sift_up(self, index):
        """
        Perform sift-up operation to maintain the heap property.

        Args:
            index (int): The index of the element to sift up.
        """
        while index > 0:
            parent_index = (index - 1) // self.d
            if self.data[parent_index].weight <= self.data[index].weight:
                break
            self.data[index], self.data[parent_index] = self.data[parent_index], self.data[index]
            index = parent_index

    def sift_down(self, index):
        """
        Perform sift-down operation to maintain the heap property.

        Args:
            index (int): The index of the element to sift down.
        """
        while True:
            min_child_index = self.find_min_child(index)
            if min_child_index is None or self.data[index].weight <= self.data[min_child_index].weight:
                break
            self.data[index], self.data[min_child_index] = self.data[min_child_index], self.data[index]
            index = min_child_index

    def find_min_child(self, index):
        """
        Return the index of the smallest child of the given index, or None if no children exist.

        Args:
            index (int): The index of the parent element.

        Returns:
            int: The index of the smallest child, or None if no children exist.
        """
        start_child = self.d * index + 1
        end_child = min(self.d * index + self.d, self.size() - 1)
        if start_child >= self.size():
            return None

        min_child_index = start_child
        for i in range(start_child + 1, end_child + 1):
            if self.data[i].weight < self.data[min_child_index].weight:
                min_child_index = i

        return min_child_index

    def __repr__(self):
        """Return a string representation of the heap."""
        return f"DaryHeap({self.d}): " + str(self.data)
