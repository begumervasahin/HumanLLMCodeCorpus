import glob
import time
import networkx as nx
from statistics import mean
import Abnormal_Scenario_Generation as ASG

def analyze_files(file_paths, detection_function):
    """
    Analyze the given GML files using the provided detection function.

    Args:
        file_paths (list of str): List of file paths to analyze.
        detection_function (callable): Function to detect outliers in the network.

    Returns:
        list of float: List of analysis times for each file.
    """
    analysis_times = []

    for file_path in file_paths:
        start_time = time.time()
        print(f'Analyzing file: {file_path}')
        network = nx.read_gml(file_path)
        score_set, score_com = detection_function(network)
        analysis_times.append(time.time() - start_time)

    return analysis_times

def print_analysis_statistics(analysis_times):
    """
    Print statistics about the analysis times.

    Args:
        analysis_times (list of float): List of analysis times for each file.
    """
    min_time = min(analysis_times)
    max_time = max(analysis_times)
    avg_time = mean(analysis_times)
    sum_time = sum(analysis_times)

    print(f'Minimum analysis time: {min_time:.2f} seconds')
    print(f'Average analysis time: {avg_time:.2f} seconds')
    print(f'Maximum analysis time: {max_time:.2f} seconds')
    print(f'Total analysis time: {sum_time:.2f} seconds')

def main(normal_path, abnormal_path):
    """
    Main entry point for analyzing normal and abnormal GML files.

    Args:
        normal_path (str): Path to the directory containing normal GML files.
        abnormal_path (str): Path to the directory containing abnormal GML files.
    """
    normal_files = glob.glob(f'{normal_path}/*.gml')
    abnormal_files = glob.glob(f'{abnormal_path}/*.gml')

    normal_analysis_times = analyze_files(normal_files, ASG.Detection_Outliers)

    print_analysis_statistics(normal_analysis_times)

if __name__ == "__main__":
    normal_path = 'Normal_Example'
    abnormal_path = 'Abnormal_Example'
    main(normal_path, abnormal_path)
