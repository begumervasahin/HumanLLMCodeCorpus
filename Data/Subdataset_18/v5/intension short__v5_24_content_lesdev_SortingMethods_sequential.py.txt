import sys
from tkinter import *
import tkSimpleDialog

def busqueda():
    """Handles the number input and performs the search operation."""
    # Retrieve the number of elements to input
    try:
        num_elements = int(entry_num_elements.get())
    except ValueError:
        label3.config(text="Please enter a valid number.")
        return

    # Ask the user for the number to search
    search_number = tkSimpleDialog.askinteger("Input", "Ingresa el número a buscar")

    if search_number is None:
        label3.config(text="Search operation canceled.")
        return

    # Collect the list of numbers
    numbers = get_numbers(num_elements)

    if numbers is None:
        label3.config(text="Operation canceled.")
        return

    # Display the list of numbers
    Label2.config(text=f"Lista de números: {numbers}")

    # Perform the search and display the result
    search_result = secuencial(numbers, search_number)
    label3.config(text=search_result)

def get_numbers(num_elements):
    """Prompts the user to enter a specific number of integers."""
    numbers = []
    for _ in range(num_elements):
        number = tkSimpleDialog.askinteger("Input", "Ingresa un número:")
        if number is None:
            return None  # Return None if the user cancels the operation
        numbers.append(number)
    return numbers

def secuencial(numbers, search_number):
    """Performs a sequential search for the specified number."""
    for i, number in enumerate(numbers):
        if number == search_number:
            return f"Encontrado en la posición: {i}"
    return "Número no encontrado"

# Main application setup
app = Tk()
app.title("Sequential Search")
app.geometry('300x250')
app.configure(bg='SkyBlue2')

# UI elements
Label1 = Label(app, text="¿Cuántos números vas a ingresar?", font='Helvetica 12', bg='SkyBlue2')
Label1.pack(pady=10)

entry_num_elements = Entry(app, width=8)
entry_num_elements.pack(pady=5)

button1 = Button(app, text="Ok!", command=busqueda)
button1.pack(pady=10)

Label2 = Label(app, text="", bg='SkyBlue2')
Label2.pack(pady=10)

label3 = Label(app, text="", bg='SkyBlue2')
label3.pack(pady=10)

# Start the Tkinter main loop
app.mainloop()
