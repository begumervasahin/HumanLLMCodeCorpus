it marks each composite once and only once.

import db
import itertools
import time

def wsieve(N):
    """
    Perform a sieve to find all prime numbers up to N.
    This implementation marks each composite number once and only once.
    """
    # Initialize the state array: -1 for the first two indices and 0 for the rest
    state = [-1, -1] + [0] * (N - 1)
    primes = [2]
    prime_index = 0
    lcm = 1
    current_prime = 2

    while current_prime**2 < N:
        # Extend the list of primes with new primes up to current_prime^2
        start_range = primes[prime_index - 1]**2 if prime_index else current_prime + 1
        for i in range(start_range, current_prime**2):
            if state[i] == 0:
                primes.append(i)

        # Generate the sequence U based on the current lcm
        if lcm < current_prime**2:
            U = itertools.chain([1], (i for i in range(current_prime, min(lcm, N // current_prime) + 1) if state[i] == 0))
        else:
            U = itertools.chain([1], primes[prime_index:], (i for i in range(current_prime**2, min(lcm, N // current_prime) + 1) if state[i] == 0))

        next_lcm = current_prime * lcm

        # Mark multiples of the current prime
        for sortaprime in U:
            for i in range(current_prime * sortaprime, N + 1, next_lcm):
                state[i] = current_prime

        lcm = next_lcm
        prime_index += 1
        current_prime = primes[prime_index]

    return primes

def go():
    """
    Execute the wsieve function and measure its execution time.
    Ensures the resulting primes are stored in the database.
    """
    N = 10**6
    start_time = time.time()
    primes = wsieve(N)
    end_time = time.time()
    duration = end_time - start_time
    print(f"Finding primes up to {N} took {duration:.4f} seconds")
    
    db.ensure(primes)

if __name__ == "__main__":
    go()
