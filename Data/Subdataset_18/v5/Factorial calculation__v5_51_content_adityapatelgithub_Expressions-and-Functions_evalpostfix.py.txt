from linkedlist import LinkedList
from vartree import binarySearchTree
from infixtotree import to_expr_tree

# Initialize the variable tree
var_tree = binarySearchTree()

def eval_postfix(postfix_expr):
    """
    Evaluates a postfix expression using a stack and a binary search tree for variable storage.

    :param postfix_expr: Postfix expression as a list of tokens
    :return: Evaluation result
    """
    stack = LinkedList()

    for token in postfix_expr:
        if token[0].isdigit() or token[0].isalpha():
            stack.push(token)
        elif token == '=':
            handle_assignment(stack)
        else:
            handle_operator(stack, token)

    return get_final_result(stack)

def handle_assignment(stack):
    """
    Handles the assignment operation in the postfix expression.

    :param stack: Stack used for evaluation
    """
    top = stack.pop()
    bottom = stack.pop()
    if top[0].isalpha():
        top = var_tree.lookup(top)
    var_tree.assign(bottom, top)
    stack.push(bottom)

def handle_operator(stack, operator):
    """
    Handles the operator in the postfix expression.

    :param stack: Stack used for evaluation
    :param operator: Operator to be applied
    """
    top = stack.pop()
    bottom = stack.pop()
    if top[0].isalpha():
        top = var_tree.lookup(top)
    if bottom[0].isalpha():
        bottom = var_tree.lookup(bottom)
    result = eval(f"{bottom} {operator} {top}")
    stack.push(str(result))

def get_final_result(stack):
    """
    Retrieves the final result from the stack.

    :param stack: Stack used for evaluation
    :return: Final result of the evaluation
    """
    final_result = stack.top()
    if final_result[0].isdigit():
        return float(final_result)
    return var_tree.lookup(final_result)

# Example usage
if __name__ == "__main__":
    # Example postfix expression: 12 3 * 5 + a =
    postfix_expr = ['12', '3', '*', '5', '+', 'a', '=']
    result = eval_postfix(postfix_expr)
    print("Result:", result)
