def insertion_sort(lst):
    """Sorts a list of integers in place using the insertion sort algorithm."""
    for i in range(1, len(lst)):
        current_value = lst[i]
        j = i - 1

        # Shift elements of lst[0..i-1], that are greater than current_value,
        # to one position ahead of their current position
        while j >= 0 and lst[j] > current_value:
            lst[j + 1] = lst[j]
            j -= 1

        lst[j + 1] = current_value

def read_data(file_path):
    """
    Reads a file where each line represents a list of integers.
    The first value of each line is skipped, and the remaining integers are stored as lists.
    
    Args:
        file_path (str): Path to the input file.

    Returns:
        list of list of int: A list containing lists of integers.
    """
    with open(file_path, "r") as file:
        return [list(map(int, line.split()[1:])) for line in file]

def write_data(file_path, sorted_lists):
    """
    Writes the sorted lists of integers to a file.
    
    Args:
        file_path (str): Path to the output file.
        sorted_lists (list of list of int): A list of sorted integer lists to write to the file.
    """
    with open(file_path, "w") as file:
        for lst in sorted_lists:
            file.write(" ".join(map(str, lst)) + "\n")

def main(input_file, output_file):
    """Main function to read, sort, and write the data."""
    unsorted_lists = read_data(input_file)

    for lst in unsorted_lists:
        insertion_sort(lst)

    write_data(output_file, unsorted_lists)

if __name__ == "__main__":
    main("data.txt", "insertionsort.txt")
