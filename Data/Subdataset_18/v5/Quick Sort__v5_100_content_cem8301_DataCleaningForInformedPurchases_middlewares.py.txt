from scrapy import signals

class CraigslistSpiderMiddleware:
    @classmethod
    def from_crawler(cls, crawler):
        """Initialize the middleware and connect the spider_opened signal."""
        instance = cls()
        crawler.signals.connect(instance.spider_opened, signal=signals.spider_opened)
        return instance

    def process_spider_input(self, response, spider):
        """Process each response before passing it to the spider."""
        return None

    def process_spider_output(self, response, result, spider):
        """Process the results returned from the spider."""
        for item in result:
            yield item

    def process_spider_exception(self, response, exception, spider):
        """Handle exceptions raised by the spider or process_spider_input."""
        pass

    def process_start_requests(self, start_requests, spider):
        """Process start requests before passing them to the spider."""
        for request in start_requests:
            yield request

    def spider_opened(self, spider):
        """Log when the spider is opened."""
        spider.logger.info(f'Spider opened: {spider.name}')


class CraigslistDownloaderMiddleware:
    @classmethod
    def from_crawler(cls, crawler):
        """Initialize the middleware and connect the spider_opened signal."""
        instance = cls()
        crawler.signals.connect(instance.spider_opened, signal=signals.spider_opened)
        return instance

    def process_request(self, request, spider):
        """Process each request before sending it to the downloader."""
        return None

    def process_response(self, request, response, spider):
        """Process each response before passing it to the spider."""
        return response

    def process_exception(self, request, exception, spider):
        """Handle exceptions raised during the download process."""
        pass

    def spider_opened(self, spider):
        """Log when the spider is opened."""
        spider.logger.info(f'Spider opened: {spider.name}')
