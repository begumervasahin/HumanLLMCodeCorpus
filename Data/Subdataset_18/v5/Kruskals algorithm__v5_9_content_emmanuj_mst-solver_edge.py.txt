class Edge:
    def __init__(self, start_vertex, end_vertex, weight):
        """
        Initializes an Edge object.

        :param start_vertex: The starting vertex of the edge.
        :param end_vertex: The ending vertex of the edge.
        :param weight: The weight of the edge.
        """
        self.start_vertex = start_vertex  # Starting vertex of the edge
        self.end_vertex = end_vertex      # Ending vertex of the edge
        self.weight = weight              # Weight of the edge

    def __repr__(self):
        """
        Provides a string representation of the Edge object.

        :return: A formatted string representing the edge's start, end, and weight.
        """
        return f"Edge from {self.start_vertex} to {self.end_vertex} with weight {self.weight}"
