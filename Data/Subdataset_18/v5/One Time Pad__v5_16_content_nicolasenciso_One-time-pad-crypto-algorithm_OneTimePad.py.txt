# Dictionary to map letters to numbers and vice versa
LETTERS_TO_NUMBERS = {
    'A': 0, 'B': 1, 'C': 2, 'D': 3, 'E': 4, 'F': 5, 'G': 6, 'H': 7, 'I': 8, 'J': 9,
    'K': 10, 'L': 11, 'M': 12, 'N': 13, 'O': 14, 'P': 15, 'Q': 16, 'R': 17, 'S': 18, 'T': 19,
    'U': 20, 'V': 21, 'W': 22, 'X': 23, 'Y': 24, 'Z': 25
}

NUMBERS_TO_LETTERS = {v: k for k, v in LETTERS_TO_NUMBERS.items()}

def cipher(key, message):
    """
    Encrypts the message using the provided key with a simple substitution cipher.
    
    Args:
        key (str): The key used for encryption.
        message (str): The message to be encrypted.
        
    Returns:
        list: The encrypted message as a list of characters.
    """
    if len(key) != len(message):
        raise ValueError("Key and message must be of the same length")
    
    encrypted_message = []
    
    for key_char, message_char in zip(key, message):
        key_value = LETTERS_TO_NUMBERS[key_char]
        message_value = LETTERS_TO_NUMBERS[message_char]
        encrypted_value = (key_value + message_value) % 26
        encrypted_message.append(NUMBERS_TO_LETTERS[encrypted_value])
    
    return encrypted_message

def num_cipher(key, message):
    """
    Encrypts the message using the provided numerical key with simple modular arithmetic.
    
    Args:
        key (str): The numerical key used for encryption.
        message (str): The numerical message to be encrypted.
        
    Returns:
        list: The encrypted numerical message.
    """
    if len(key) != len(message):
        raise ValueError("Key and message must be of the same length")
    
    encrypted_numbers = []
    
    for key_char, message_char in zip(key, message):
        encrypted_number = (int(key_char) + int(message_char)) % 10
        encrypted_numbers.append(encrypted_number)
    
    return encrypted_numbers

# Example usage
encrypted_message = cipher("EVXLL", "MONTH")
encrypted_numbers = num_cipher("12345", "67890")

print("Encrypted message:", encrypted_message)
print("Encrypted numbers:", encrypted_numbers)
