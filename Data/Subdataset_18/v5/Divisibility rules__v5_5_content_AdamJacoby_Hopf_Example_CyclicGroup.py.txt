import numpy as np
import scipy.sparse as sps
from itertools import product
from ExampleGroup_Functions import Group_Comult_Matrix, Group_Counit, Group_Integral
from HopfClass import HopfAlgebra

def generate_mult_matrix(dim):
    """
    Generate the multiplication matrix for a cyclic group of given dimension.

    Parameters:
    dim (int): Dimension of the cyclic group.

    Returns:
    scipy.sparse.csr_matrix: Multiplication matrix in CSR format.
    """
    mult = np.zeros((dim, dim**2), dtype=complex)
    for i, j in product(range(dim), repeat=2):
        mult[(i + j) % dim, i * dim + j] = 1
    return sps.csr_matrix(mult, dtype=complex)

def generate_antipode_matrix(dim):
    """
    Generate the antipode matrix for a cyclic group of given dimension.

    Parameters:
    dim (int): Dimension of the cyclic group.

    Returns:
    scipy.sparse.csr_matrix: Antipode matrix in CSR format.
    """
    antipode = np.zeros((dim, dim), dtype=complex)
    for i in range(dim):
        antipode[i, (-i) % dim] = 1
    return sps.csr_matrix(antipode, dtype=complex)

def generate_element_names(dim, base_name):
    """
    Generate the element names for a cyclic group of given dimension.

    Parameters:
    dim (int): Dimension of the cyclic group.
    base_name (str): Base name for the elements.

    Returns:
    list: List of element names.
    """
    return [f"{base_name}^{i}" for i in range(dim)]

def create_cyclic_group(dim, element_name):
    """
    Create a cyclic group Hopf algebra of given dimension and element name.

    Parameters:
    dim (int): Dimension of the cyclic group.
    element_name (str): Base name for the elements.

    Returns:
    HopfAlgebra: Hopf algebra representing the cyclic group.
    """
    # Generate necessary matrices for the cyclic group
    mult_matrix = generate_mult_matrix(dim)
    comult_matrix = Group_Comult_Matrix(dim)
    counit_matrix = Group_Counit(dim)
    integral_matrix = Group_Integral(dim)
    antipode_matrix = generate_antipode_matrix(dim)

    # Define the name and element names of the cyclic group
    name = f"C_{dim}"
    element_names = generate_element_names(dim, element_name)

    # Create the Hopf algebra for the cyclic group
    cyclic_group = HopfAlgebra(name, element_names, mult_matrix, comult_matrix, counit_matrix, antipode_matrix)
    cyclic_group.Input_Integral(integral_matrix)

    return cyclic_group
