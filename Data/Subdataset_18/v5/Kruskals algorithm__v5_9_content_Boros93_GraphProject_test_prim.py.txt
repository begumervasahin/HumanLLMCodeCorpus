import networkx as nx
import matplotlib.pyplot as plt
from fibonacci_heap import FibonacciHeap
from prim_algorithm import prim

def build_heap(vertices, keys, heap):
    """
    Builds a Fibonacci heap from the given vertices and their corresponding keys.

    Parameters:
    - vertices (iterable): The vertices of the graph.
    - keys (dict): A dictionary mapping each vertex to its key value.
    - heap (FibonacciHeap): The Fibonacci heap where the vertices will be inserted.

    Returns:
    - dict: A dictionary mapping each vertex to its corresponding node in the heap.
    """
    nodes_in_heap = {}
    for vertex in vertices:
        nodes_in_heap[vertex] = heap.insert(keys[vertex], vertex)
    return nodes_in_heap

# Function to initialize the graph with nodes and edges
def initialize_graph():
    G = nx.Graph()

    # Define nodes with their positions
    nodes = {
        'a': (0, 1),
        'b': (2, 1),
        'c': (1, 0),
        'd': (1, 2)
    }

    # Add nodes to the graph with positions and initial keys
    for node, position in nodes.items():
        G.add_node(node, pos=position, key=0)

    # Define edges with weights
    edges = [
        ('a', 'b', 2),
        ('a', 'c', 30),
        ('b', 'c', 90),
        ('d', 'b', 1),
        ('d', 'a', 3)
    ]

    # Add edges to the graph
    for v1, v2, weight in edges:
        G.add_edge(v1, v2, weight=weight)

    return G

# Function to plot the original graph and the Minimum Spanning Tree (MST)
def plot_graphs(G, MST, node_positions):
    plt.figure(figsize=(12, 6))

    # Plot the original graph with edge weights
    plt.subplot(121)
    nx.draw(G, pos=node_positions, with_labels=True)
    edge_weights = nx.get_edge_attributes(G, 'weight')
    nx.draw_networkx_edge_labels(G, pos=node_positions, edge_labels=edge_weights)
    plt.title("Original Graph")

    # Plot the MST with edge weights
    plt.subplot(122)
    nx.draw(MST, pos=node_positions, with_labels=True)
    mst_edge_weights = nx.get_edge_attributes(MST, 'weight')
    nx.draw_networkx_edge_labels(MST, pos=node_positions, edge_labels=mst_edge_weights)
    plt.title("Minimum Spanning Tree (MST)")

    # Display the plots
    plt.show()

def main():
    # Initialize the graph
    G = initialize_graph()

    # Extract node positions for plotting
    node_positions = nx.get_node_attributes(G, 'pos')

    # Set the starting node for Prim's algorithm
    start_node = 'a'

    # Compute the Minimum Spanning Tree (MST) using Prim's algorithm
    node_set = set(G.nodes())
    edge_weights = nx.get_edge_attributes(G, 'weight')
    MST = prim(G, edge_weights, start_node, node_set)

    # Plot the original graph and the MST
    plot_graphs(G, MST, node_positions)

if __name__ == "__main__":
    main()
