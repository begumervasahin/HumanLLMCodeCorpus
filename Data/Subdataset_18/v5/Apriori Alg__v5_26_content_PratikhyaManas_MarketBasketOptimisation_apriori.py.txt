# Apriori Algorithm for Market Basket Optimization

# Importing necessary libraries
import numpy as np
import pandas as pd
from apyori import apriori

def load_dataset(file_path):
    """
    Load dataset from a CSV file.

    Parameters:
    file_path (str): Path to the CSV file containing transaction data.

    Returns:
    pd.DataFrame: Loaded dataset.
    """
    return pd.read_csv(file_path, header=None)

def preprocess_data(dataset):
    """
    Preprocess the dataset into a list of transactions.

    Parameters:
    dataset (pd.DataFrame): DataFrame containing the dataset.

    Returns:
    list: List of transactions where each transaction is a list of items.
    """
    transactions = []
    for i in range(len(dataset)):
        transaction = [str(dataset.values[i, j]) for j in range(dataset.shape[1])]
        transactions.append(transaction)
    return transactions

def train_apriori(transactions, min_support, min_confidence, min_lift, min_length):
    """
    Train the Apriori algorithm on the given transactions.

    Parameters:
    transactions (list): List of transactions.
    min_support (float): Minimum support value.
    min_confidence (float): Minimum confidence value.
    min_lift (float): Minimum lift value.
    min_length (int): Minimum length of the rule.

    Returns:
    list: List of rules generated by the Apriori algorithm.
    """
    rules = apriori(transactions, min_support=min_support, min_confidence=min_confidence, min_lift=min_lift, min_length=min_length)
    return list(rules)

def visualize_rules(rules):
    """
    Visualize the results of the Apriori algorithm.

    Parameters:
    rules (list): List of rules generated by the Apriori algorithm.
    """
    for rule in rules:
        items = [x for x in rule.items]
        print(f"Rule: {items[0]} -> {items[1]}")
        print(f"Support: {rule.support}")
        print(f"Confidence: {rule.ordered_statistics[0].confidence}")
        print(f"Lift: {rule.ordered_statistics[0].lift}")
        print("=====================================")

def main():
    """
    Main function to execute the Apriori algorithm workflow.
    """
    # File path to the dataset
    file_path = 'Market_Basket_Optimisation.csv'
    
    # Load dataset
    dataset = load_dataset(file_path)
    
    # Preprocess data
    transactions = preprocess_data(dataset)
    
    # Set parameters for the Apriori algorithm
    min_support = 0.003
    min_confidence = 0.2
    min_lift = 3
    min_length = 2
    
    # Train Apriori on the dataset
    rules = train_apriori(transactions, min_support, min_confidence, min_lift, min_length)
    
    # Visualize the results
    visualize_rules(rules)

# Execute the main function
if __name__ == "__main__":
    main()
