def selection_sort(arr):
    """
    Sorts an array in ascending order using the selection sort algorithm.
    Prints the array after each swap to show the sorting progress.

    Parameters:
    arr (list): The list of elements to be sorted.
    """
    n = len(arr)
    
    # Traverse through all array elements
    for i in range(n):
        # Find the minimum element in the remaining unsorted array
        min_index = i
        for j in range(i + 1, n):
            if arr[j] < arr[min_index]:
                min_index = j

        # Swap the found minimum element with the first element of the unsorted part
        arr[i], arr[min_index] = arr[min_index], arr[i]
        
        # Print the array after each swap to show the sorting progress
        print(f"Array after swapping index {i} and {min_index}: {arr}")

def main():
    """
    Main function to initialize the array and perform selection sort.
    """
    arr = [100, 48, 70, 29, 2, 17]  # Initialize the array
    
    # Perform selection sort on the array
    selection_sort(arr)

if __name__ == "__main__":
    main()
