def get_db_connection(config):
    """Establish and return a database connection."""
    return pymysql.connect(
        host=config['host'],
        user=config['user'],
        password=config['password'],
        db=config['db']
    )

def word_freq(sentences):
    """Calculate word frequencies from a list of sentences."""
    word_freq_dict = {}
    encountered_words = set()

    for sentence in sentences:
        sentence = str(sentence)
        
        # Remove specific unwanted text
        if "(Read more about how we rate cars.)" in sentence:
            sentence = sentence.replace("(Read more about how we rate cars.)", "")
        
        # Normalize to lowercase and remove punctuation
        sentence = sentence.lower()
        punctuation = r'[,，。.!?:"@]'
        sentence = re.sub(punctuation, '', sentence)
        
        # Update word frequency dictionary
        for word in sentence.split():
            if word in encountered_words:
                word_freq_dict[word] += 1
            else:
                word_freq_dict[word] = 1
                encountered_words.add(word)
    
    return word_freq_dict

def fetch_data_from_db(query, db_config):
    """Fetch data from the database based on the provided query."""
    conn = get_db_connection(db_config)
    cursor = conn.cursor()
    cursor.execute(query)
    results = cursor.fetchall()
    cursor.close()
    conn.close()
    return results

def save_word_frequencies(word_freq_dict, filename):
    """Save word frequencies to a file."""
    sorted_word_freq = sorted(word_freq_dict.items(), key=lambda x: x[1], reverse=True)
    with open(filename, 'w') as f:
        for index, (word, freq) in enumerate(sorted_word_freq, start=1):
            f.write(f"{index} : {word} : {freq}\n")

def save_translations(translations, filename):
    """Save translations to a file."""
    with open(filename, 'w') as f:
        for index, translation in enumerate(translations, start=1):
            if index % 10000 == 0:
                print(index)
            f.write(translation[0] + '\n')

def en_freq():
    """Fetch English data from the database and write word frequencies to a file."""
    query = "SELECT data FROM RAW_DATA"
    results = fetch_data_from_db(query, db_config)
    sentences = [result[0] for result in results]
    word_freq_dict = word_freq(sentences)
    save_word_frequencies(word_freq_dict, './word_freq_en_v2.txt')

def cn_freq():
    """Fetch Chinese translations from the database and write them to a file."""
    query = "SELECT translate FROM RAW_DATA"
    results = fetch_data_from_db(query, db_config)
    save_translations(results, './cn_translations.txt')

def process_cn_translations():
    """Process Chinese translations and save word frequencies to a file."""
    with open('./cn_translations.txt', 'r', encoding='utf8') as f:
        word_list = get_words(f.read())
    
    with open('./word_freq_cn_v1.txt', 'w') as f:
        for word in word_list:
            f.write(word + '\n')
