def load(filename="input.txt"):
    # Open the file and read the contents
    with open(filename, 'r') as file:
        # Split the file contents into a list of strings
        data = file.read().split()
    
    # Convert the first three elements of the list to integers
    fizz = int(data[0])
    buzz = int(data[1])
    end = int(data[2])
    
    # Iterate through the range from 1 to end (inclusive)
    for i in range(1, end + 1):
        # Check if the number is divisible by both fizz and buzz
        if i % fizz == 0 and i % buzz == 0:
            print("FB")
        # Check if the number is divisible by fizz only
        elif i % fizz == 0:
            print("F")
        # Check if the number is divisible by buzz only
        elif i % buzz == 0:
            print("B")
        # If none of the above conditions are met, print the number itself
        else:
            print(i)

# Example usage
load("input.txt")
