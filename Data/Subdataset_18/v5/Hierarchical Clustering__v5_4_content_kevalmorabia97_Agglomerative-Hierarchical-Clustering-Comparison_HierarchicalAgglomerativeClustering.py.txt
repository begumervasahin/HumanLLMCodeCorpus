from sklearn.cluster import AgglomerativeClustering
from sklearn.decomposition import PCA
from sklearn.preprocessing import normalize
import matplotlib.pyplot as plt

def load_data(file_path):
    """
    Load the dataset from the specified file and extract features and labels.

    Args:
        file_path (str): Path to the data file.

    Returns:
        tuple: A tuple containing the feature data and the corresponding labels.
    """
    data, labels = [], []

    with open(file_path, 'r') as file:
        for line in file:
            values = list(map(float, line.strip().split(',')))
            data.append(values[:-1])
            labels.append(int(values[-1]))

    return data, labels

def preprocess_data(data):
    """
    Normalize the data and reduce its dimensionality using PCA.

    Args:
        data (list): The input data.

    Returns:
        np.ndarray: The processed data with reduced dimensions.
    """
    normalized_data = normalize(data, axis=0)
    pca = PCA(n_components=2)
    return pca.fit_transform(normalized_data)

def plot_clusters(X, labels, linkage_methods, n_clusters=3):
    """
    Plot clusters generated by different linkage methods and compare with actual data.

    Args:
        X (np.ndarray): The processed data for clustering.
        labels (list): The actual labels for the data.
        linkage_methods (list): List of linkage methods to compare.
        n_clusters (int): The number of clusters to form.
    """
    num_plots = len(linkage_methods)
    plt.figure(figsize=(15, 5))

    for i, linkage in enumerate(linkage_methods, start=1):
        if linkage == 'actual_data':
            cluster_labels = labels
        else:
            model = AgglomerativeClustering(linkage=linkage, n_clusters=n_clusters)
            cluster_labels = model.fit_predict(X)

        plt.subplot(1, num_plots, i)
        plt.scatter(X[:, 0], X[:, 1], c=cluster_labels, cmap='viridis')
        plt.title(f'Linkage: {linkage}')

    plt.tight_layout()
    plt.show()

def main():
    """
    Main function to load data, preprocess it, and visualize clusters.
    """
    file_path = 'data/random_3_clusters.txt'

    # Load and preprocess the data
    data, correct_labels = load_data(file_path)
    X = preprocess_data(data)

    # Define the linkage methods to compare
    linkage_methods = ['complete', 'average', 'ward', 'actual_data']

    # Plot and compare clusters generated by different methods
    plot_clusters(X, correct_labels, linkage_methods)

if __name__ == "__main__":
    main()
