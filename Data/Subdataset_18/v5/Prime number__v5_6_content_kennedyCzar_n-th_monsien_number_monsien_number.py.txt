import random

def is_prime(number):
    """
    Check if the given number is a prime number.
    
    Args:
        number (int): The number to check.
        
    Returns:
        bool: True if the number is prime, False otherwise.
    """
    if number <= 1:
        return False
    for i in range(2, int(number ** 0.5) + 1):
        if number % i == 0:
            return False
    return True

def find_monisen_number(n):
    """
    Find the nth Monisen number.
    
    Args:
        n (int): The position of the Monisen number to find.
        
    Returns:
        int: The nth Monisen number.
    """
    count = 0
    p = 2
    while count < n:
        if is_prime(p):
            m = 2 ** p - 1
            if is_prime(m):
                count += 1
                if count == n:
                    return m
        p += 1

def convert_and_print(num, base):
    """
    Convert a number to the given base and print the result.
    
    Args:
        num (int): The number to convert.
        base (int): The base to convert to.
    """
    if num >= base:
        convert_and_print(num // base, base)
    print(num % base, end=' ')

def main():
    """
    Main function to handle user input and output Monisen number and base conversion.
    """
    # Get the nth Monisen number
    n = int(input("Enter the nth Monisen number: "))
    nth_monisen_number = find_monisen_number(n)
    print(f"The {n}-th Monisen number is: {nth_monisen_number}")

    # Save the nth Monisen number to a file
    with open("monisen_number.txt", "w") as file:
        file.write(str(nth_monisen_number))

    # Convert and print a number in a given base
    numA = int(input("Enter a number: "))
    numB = int(input("Enter the base: "))
    convert_and_print(numA, numB)
    print()  # For a new line after convert_and_print output

if __name__ == "__main__":
    main()
