import random
import timeit

def print_array(arr):
    """Prints the array elements separated by spaces."""
    print(' '.join(map(str, arr)))

def insertion_sort(arr):
    """Sorts the array using the insertion sort algorithm."""
    for i in range(1, len(arr)):
        temp = arr[i]
        j = i - 1
        while j >= 0 and arr[j] > temp:
            arr[j + 1] = arr[j]
            j -= 1
        arr[j + 1] = temp

def merge_sort(arr):
    """Sorts the array using the merge sort algorithm."""
    if len(arr) > 1:
        mid = len(arr) // 2
        left_half = arr[:mid]
        right_half = arr[mid:]

        merge_sort(left_half)
        merge_sort(right_half)

        i = j = k = 0

        # Merging the left and right halves
        while i < len(left_half) and j < len(right_half):
            if left_half[i] < right_half[j]:
                arr[k] = left_half[i]
                i += 1
            else:
                arr[k] = right_half[j]
                j += 1
            k += 1

        # Copying any remaining elements from the left half
        while i < len(left_half):
            arr[k] = left_half[i]
            i += 1
            k += 1

        # Copying any remaining elements from the right half
        while j < len(right_half):
            arr[k] = right_half[j]
            j += 1
            k += 1

def unh_sort(arr):
    """Sorts the array using a custom sorting algorithm (similar to bubble sort)."""
    for i in range(len(arr) - 1, 0, -1):
        for j in range(i):
            if arr[j] > arr[j + 1]:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]

def measure_sorting_time(sort_func, arr):
    """Measures the time taken by the given sorting function to sort the array."""
    start_time = timeit.default_timer()
    sort_func(arr)
    return timeit.default_timer() - start_time

# Generate a random array
array_size = 10000
initial_array = random.sample(range(array_size), array_size)

# Make copies for each sorting algorithm
insertion_randoms = initial_array.copy()
merge_randoms = initial_array.copy()
unh_randoms = initial_array.copy()

print("Initial Array:")
print_array(initial_array)

# Measure sorting times for each algorithm
insertion_sort_time = measure_sorting_time(insertion_sort, insertion_randoms)
merge_sort_time = measure_sorting_time(merge_sort, merge_randoms)
unh_sort_time = measure_sorting_time(unh_sort, unh_randoms)

# Print the results
print(f"Insertion Sort Time: {insertion_sort_time:.6f} seconds")
print(f"Merge Sort Time: {merge_sort_time:.6f} seconds")
print(f"UNH Sort Time: {unh_sort_time:.6f} seconds")
