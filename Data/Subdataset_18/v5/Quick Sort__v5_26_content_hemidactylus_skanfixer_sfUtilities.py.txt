import os

# Define the allowed image suffixes
PIC_SUFFIXES = ['jpg', 'jpeg', 'bmp', 'tiff', 'gif', 'png']

def safe_build_file_name(directory, body, extension):
    """
    Combines a directory, a filename body, and an extension,
    adding a numeric suffix to the body if the file already exists.
    
    Args:
        directory (str): The directory where the file will be saved.
        body (str): The base name of the file.
        extension (str): The file extension.

    Returns:
        str: A unique file path with the given directory, body, and extension.
    """
    new_body = body
    iteration = 0
    while True:
        file_name = os.path.join(directory, f'{new_body}.{extension}')
        if not os.path.isfile(file_name):
            return file_name
        iteration += 1
        new_body = f'{body}_r{iteration:03d}'

def ensure_directory_exists(dir_name):
    """
    Ensures that the specified directory exists, creating it if necessary.
    
    Args:
        dir_name (str): The name of the directory to check/create.
    """
    if not os.path.isdir(dir_name):
        os.mkdir(dir_name)

def right_clip_text(text, length):
    """
    Clips text from the right, adding '...' if it exceeds the specified length.
    
    Args:
        text (str): The text to be clipped.
        length (int): The maximum length of the returned text.

    Returns:
        str: The clipped text.
    """
    return '...' + text[-length:] if len(text) > length else text

def pop_item(q_list, q_item):
    """
    Removes an item from a list.
    
    Args:
        q_list (list): The list from which to remove the item.
        q_item: The item to be removed.
    """
    q_list.pop(q_list.index(q_item))

def list_image_files(directory):
    """
    Builds a sorted list of image files for a given directory.
    
    Args:
        directory (str): The directory to search for image files.

    Returns:
        list: A sorted list of image file names.
    """
    return sorted([file_name for file_name in os.listdir(directory) if is_picture(file_name)])

def is_picture(filename):
    """
    Determines if a file is an image based on its suffix.
    
    Args:
        filename (str): The name of the file.

    Returns:
        bool: True if the file is an image, False otherwise.
    """
    return any(filename.lower().endswith(suffix) for suffix in PIC_SUFFIXES)

def find_rescale_factor(img_size, allowed_size, allow_zoom=True):
    """
    Calculates the rescale factor given an image size and the size of the container.
    
    Args:
        img_size (tuple): The width and height of the image.
        allowed_size (tuple): The width and height of the container.
        allow_zoom (bool): Whether to allow zooming in.

    Returns:
        float: The rescale factor.
    """
    max_factor = max(float(img_dim) / float(allowed_dim) for img_dim, allowed_dim in zip(img_size, allowed_size))
    if not allow_zoom and max_factor < 1:
        max_factor = 1
    print(f'Rescale Factor = {max_factor:.3f}')
    return max_factor

# Define the allowed characters for filenames
FILE_NAME_ALLOWED_CHARS = 'qwertyuiopasdfghjklzxcvbnmQWERTYUIOPASDFGHJKLZXCVBNM1234567890-_~.,'

def normalize_string(text):
    """
    Discards all characters not in the allowed set for filenames.
    
    Args:
        text (str): The text to be normalized.

    Returns:
        str: The normalized text containing only allowed characters.
    """
    return ''.join(char for char in text if char in FILE_NAME_ALLOWED_CHARS)
