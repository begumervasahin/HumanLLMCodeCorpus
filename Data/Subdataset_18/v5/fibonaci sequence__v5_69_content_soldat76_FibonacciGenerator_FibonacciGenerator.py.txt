def generate_fibonacci_sequence(length):
    """
    Generate a Fibonacci sequence of the desired length.

    Args:
        length (int): The number of Fibonacci numbers to generate.

    Returns:
        list or str: A list containing the first `length` Fibonacci numbers or a message for invalid input.
    """
    if length <= 0:
        return "The sequence must contain some positive number."
    elif length == 1:
        return [1]
    elif length == 2:
        return [1, 1]

    fibonacci_sequence = [1, 1]
    while len(fibonacci_sequence) < length:
        next_number = fibonacci_sequence[-2] + fibonacci_sequence[-1]
        fibonacci_sequence.append(next_number)
    
    return fibonacci_sequence

def print_sequence(length, fibonacci_sequence):
    """
    Print the Fibonacci sequence with the desired length.

    Args:
        length (int): The length of the Fibonacci sequence.
        fibonacci_sequence (list): The Fibonacci sequence to print.
    """
    print(f"\nHere is your Fibonacci sequence of {length} number(s):\n\t{fibonacci_sequence}")

def main():
    """
    Main function to interact with the user, generate, and print Fibonacci sequences.
    """
    print("\nThis program will print out a Fibonacci sequence of your desired length!")

    while True:
        try:
            length = int(input("\nEnter your desired Fibonacci sequence length: "))
            if length <= 0:
                print("The sequence must contain some positive number.")
                continue

            fibonacci_sequence = generate_fibonacci_sequence(length)
            if isinstance(fibonacci_sequence, str):
                print(fibonacci_sequence)
            else:
                print_sequence(length, fibonacci_sequence)
        except ValueError:
            print("Please enter a valid number.")

        continue_prompt = input("\nDo you want to generate another sequence? (yes/no): ").strip().lower()
        if continue_prompt != 'yes':
            break

if __name__ == "__main__":
    main()
