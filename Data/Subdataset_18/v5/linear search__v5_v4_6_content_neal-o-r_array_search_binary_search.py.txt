import numpy as np
import timeit
import matplotlib.pyplot as plt

# Use ggplot style for plots
plt.style.use('ggplot')

def linear_search(arr):
    comparisons = 0
    previous_element = -np.inf

    for current_element in arr:
        if current_element < previous_element:
            return comparisons  # Early exit if the array is not sorted
        comparisons += 1
        previous_element = current_element

    return comparisons

def binary_search(arr, initial_index=0):
    pivot = arr[len(arr) // 2]

    if pivot == arr[0]:
        return initial_index + 1

    if pivot < arr[0]:
        return binary_search(arr[:len(arr) // 2], initial_index)
    else:
        initial_index += len(arr) // 2
        return binary_search(arr[len(arr) // 2:], initial_index)

def time_wrapper(func, *args, **kwargs):
    def wrapped():
        return func(*args, **kwargs)
    return wrapped

def measure_search_times(ns, iterations=100, num_timings=10):
    bins = []
    lins = []

    for n in ns:
        binary_search_time_total = 0
        linear_search_time_total = 0

        for _ in range(iterations):
            arr = np.roll(np.arange(n), np.random.randint(n))

            t_linear_search = time_wrapper(linear_search, arr)
            t_binary_search = time_wrapper(binary_search, arr)

            binary_search_time_total += timeit.timeit(t_binary_search, number=num_timings)
            linear_search_time_total += timeit.timeit(t_linear_search, number=num_timings)

        bins.append(binary_search_time_total / iterations)
        lins.append(linear_search_time_total / iterations)

    return bins, lins

def plot_search_performance(ns, lins, bins):
    plt.plot(ns, np.array(lins) / max(lins), '-o', label='Linear Search')
    plt.plot(ns, np.array(bins) / max(lins), '-o', label='Binary Search')
    plt.plot(ns, np.array(ns) / max(ns), '--k', label='Linear')
    plt.plot(ns, np.log2(np.array(ns)) / max(np.log2(np.array(ns))), '--r', label='Logarithmic')

    plt.xlabel('Array Size (n)')
    plt.ylabel('Normalized Time')
    plt.legend()
    plt.title('Performance Comparison of Search Algorithms')
    plt.show()

if __name__ == '__main__':
    ns = [10, 50, 100, 200, 500, 1000, 5000, 10000]
    bins, lins = measure_search_times(ns)
    plot_search_performance(ns, lins, bins)
