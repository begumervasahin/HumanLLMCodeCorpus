import numpy as np
import matplotlib.pyplot as plt
from matplotlib.backends.backend_pdf import PdfPages
import sys

def main():
    # Check for correct number of arguments
    if len(sys.argv) != 5:
        print(f"{sys.argv[0]} takes 4 arguments. Not {len(sys.argv) - 1}")
        sys.exit()

    # Assign command line arguments to variables
    first, second, third, four = sys.argv[1:5]

    # Load data from the provided files
    Xt = np.genfromtxt(first, delimiter=',')
    y = np.genfromtxt(second, autostrip=True)

    # Perform Linear Discriminant Analysis (LDA)
    V_r, result = perform_lda(Xt, y)

    # Plot the results
    plot_lda_result(result)

    # Save the projection matrix and the projected data
    np.savetxt(third, V_r.T, delimiter=',')
    np.savetxt(four, result, delimiter=',')

def perform_lda(Xt, y):
    """Perform Linear Discriminant Analysis (LDA)"""
    m, n = Xt.shape

    # Identify unique labels in y
    unique_labels = set(y)

    # Calculate the global mean
    mu_global = np.mean(Xt, axis=0)

    # Initialize within-class scatter and between-class scatter matrices
    W_class = np.zeros((n, n))
    B_class = np.zeros((n, n))

    # Calculate scatter matrices for each class
    for c in unique_labels:
        C1, BT1, mu1 = calculate_scatter_matrices(Xt, y, c, mu_global)
        W_class += C1
        B_class += BT1

    # Solve the generalized eigenvalue problem for the matrix
    x = np.dot(np.linalg.inv(W_class), B_class)
    evals, evecs = np.linalg.eig(x)

    # Select the top 2 eigenvectors
    r = 2
    V_r = evecs[:, :r]

    # Project the data onto the new subspace
    result = np.dot(Xt, V_r)

    return V_r, result

def calculate_scatter_matrices(Xt, y, label, mu_global):
    """Calculate scatter matrices for a given label"""
    n = Xt.shape[1]
    C1 = np.zeros((n, n))
    BT1 = np.zeros((n, n))
    s1 = np.zeros((1, n))
    m1 = 0

    # Calculate class-specific means and scatter matrices
    for i, xt in enumerate(Xt):
        if y[i] == label:
            s1 += xt
            m1 += 1

    mu1 = s1 / m1

    for i, xt in enumerate(Xt):
        if y[i] == label:
            ct = xt - mu1
            bt = mu1 - mu_global
            C1 += np.dot(ct.T, ct)
            BT1 += np.dot(bt.T, bt)

    return C1, BT1, mu1

def plot_lda_result(result):
    """Plot the LDA results"""
    x1 = result[:, 0]
    x2 = result[:, 1]

    fig, ax = plt.subplots()
    ax.scatter(x1, x2)
    plt.title("Maximize the ratio of between-class scatter and within-class scatter")
    plt.xlabel("PC1")
    plt.ylabel("PC2")

    # Save the plot as a PDF
    plotname = 'scatter_3_plot'
    pdf = PdfPages(plotname + '.pdf')
    pdf.savefig(fig)
    pdf.close()

if __name__ == "__main__":
    main()
