class Node:
    def __init__(self, val=-1, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

class ReturnType:
    def __init__(self, size, head, mins, maxs):
        self.size = size
        self.head = head
        self.mins = mins
        self.maxs = maxs

def process(node):
    """Process the binary tree to find the largest BST subtree."""
    if node is None:
        return ReturnType(0, None, float('inf'), float('-inf'))
    
    left_info = process(node.left)
    right_info = process(node.right)
    
    include_self = 0
    if (left_info.head == node.left and
        right_info.head == node.right and
        node.val > left_info.maxs and
        node.val < right_info.mins):
        include_self = left_info.size + 1 + right_info.size
    
    max_size = max(max(left_info.size, right_info.size), include_self)
    
    if left_info.size > right_info.size:
        max_head = left_info.head
    else:
        max_head = right_info.head
    
    if max_size == include_self:
        max_head = node
    
    return ReturnType(
        max_size,
        max_head,
        min(min(left_info.mins, right_info.mins), node.val),
        max(max(left_info.maxs, right_info.maxs), node.val)
    )

def print_tree(root):
    """Print the entire binary tree."""
    print("Binary Tree:")
    print_in_order(root, 0, 'H', 17)

def print_in_order(node, height, label, length):
    """Print the binary tree in an in-order manner with formatting."""
    if node is None:
        return
    
    print_in_order(node.right, height + 1, 'v', length)
    val = f"{label}{node.val}{label}"
    len_val = len(val)
    len_left = (length - len_val) // 2
    len_right = length - len_val - len_left
    val = get_spaces(len_left) + val + get_spaces(len_right)
    print(get_spaces(height * length) + val)
    print_in_order(node.left, height + 1, '^', length)

def get_spaces(num):
    """Return a string of spaces of length num."""
    return ' ' * num

if __name__ == '__main__':
    # Create the binary tree
    root = Node(6)
    root.left = Node(1)
    root.left.left = Node(0)
    root.left.right = Node(3)
    root.right = Node(12)
    root.right.left = Node(10)
    root.right.left.left = Node(4)
    root.right.left.left.left = Node(2)
    root.right.left.left.right = Node(5)
    root.right.left.right = Node(14)
    root.right.left.right.left = Node(11)
    root.right.left.right.right = Node(15)
    root.right.right = Node(13)
    root.right.right.left = Node(20)
    root.right.right.right = Node(16)
    
    # Find and print the root value of the largest BST subtree
    largest_bst = process(root)
    if largest_bst.head:
        print("Root of the largest BST subtree:", largest_bst.head.val)
    else:
        print("No BST subtree found.")
    
    # Print the original tree
    print_tree(root)
    
    # Print the largest BST subtree
    print_tree(largest_bst.head)
