from decimal import Decimal, getcontext

def nth_root(n, A, precision):
    """
    Calculate the nth root of A with a specified precision using the Newton-Raphson method.
    
    Parameters:
    n (int): The degree of the root.
    A (Decimal or float): The number to find the root of.
    precision (int): The desired precision of the result.
    
    Returns:
    Decimal: The nth root of A with the specified precision.
    """
    getcontext().prec = precision  # Set the precision for Decimal operations
    n = Decimal(n)
    A = Decimal(A)
    x_prev = A / n  # Initial guess
    x_next = Decimal(1)
    
    while True:
        x_next = (1 / n) * ((n - 1) * x_prev + (A / (x_prev ** (n - 1))))
        if x_prev == x_next:
            return x_next
        x_prev = x_next

def main():
    """
    Main function to test the nth_root function with example inputs.
    
    Returns:
    None
    """
    degree = 5
    number = 34
    precision = 10
    root = nth_root(degree, number, precision)
    print(f"The {degree}th root of {number} with precision {precision} is {root}")

# Run the main function
if __name__ == "__main__":
    main()
