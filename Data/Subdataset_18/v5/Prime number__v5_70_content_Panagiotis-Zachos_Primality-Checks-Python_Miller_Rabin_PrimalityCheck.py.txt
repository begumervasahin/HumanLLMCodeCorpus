import math
import random
import time

def power_of_two(n):
    """
    Decompose n-1 into 2^r * d where d is odd.
    
    Args:
        n (int): The number to decompose.
    
    Returns:
        tuple: (r, d) where n-1 = 2^r * d and d is odd.
    """
    r = 0
    d = n
    while d % 2 == 0:
        r += 1
        d >>= 1
    assert 2**r * d == n
    return r, d

def miller_rabin_test(n, k):
    """
    Miller-Rabin primality test.
    
    Args:
        n (int): The number to test for primality.
        k (int): The number of iterations for accuracy.
    
    Returns:
        bool: False if n is certainly not prime, True if n is very likely a prime.
    """
    if n <= 3:
        return n == 2 or n == 3
    
    r, d = power_of_two(n - 1)
    for _ in range(k):
        rand_int = random.randint(2, n - 2)
        x = pow(rand_int, d, n)
        if x == 1 or x == n - 1:
            continue
        for _ in range(r - 1):
            x = pow(x, 2, n)
            if x == n - 1:
                break
        else:
            return False
    return True

def is_prime(n):
    """
    Simpler primality check.
    
    Args:
        n (int): The number to check.
    
    Returns:
        bool: True if n is prime, False otherwise.
    """
    if n <= 1:
        return False
    elif n <= 3:
        return True
    elif n % 2 == 0 or n % 3 == 0:
        return False
    
    i = 5
    while i * i <= n:
        if n % i == 0 or n % (i + 2) == 0:
            return False
        i += 6
    return True

def main():
    """
    Main function to test primality using Miller-Rabin and simple methods, and measure their execution time.
    """
    n = 200000015717
    
    start_time = time.time()
    miller_rabin_result = miller_rabin_test(n, 8)
    end_time = time.time()
    print("Miller-Rabin Test Result:", miller_rabin_result)
    print("Miller-Rabin Test Time: {:.6f} seconds".format(end_time - start_time))
    
    start_time = time.time()
    simple_prime_result = is_prime(n)
    end_time = time.time()
    print("Simple Primality Test Result:", simple_prime_result)
    print("Simple Primality Test Time: {:.6f} seconds".format(end_time - start_time))

if __name__ == "__main__":
    main()
