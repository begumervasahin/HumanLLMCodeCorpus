from Repository import Repository
from Controller import Controller
from UI import UI

def main():
    """
    The main function initializes the application's components and starts the user interface.
    """
    # Initialize the repository
    repository = Repository()

    # Initialize the controller with the repository
    controller = Controller(repository)

    # Initialize the user interface with the controller
    user_interface = UI(controller)

    # Start the user interface
    user_interface.start()

if __name__ == "__main__":
    main()
