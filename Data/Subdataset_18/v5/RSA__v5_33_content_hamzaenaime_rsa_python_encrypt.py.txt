import math
import random
import sys

def gcd(a, b):
    """Calculate the Greatest Common Divisor of a and b."""
    while b:
        a, b = b, a % b
    return a

def is_prime(n):
    """Check if a number is prime."""
    if n <= 1:
        return False
    for i in range(2, int(math.sqrt(n)) + 1):
        if n % i == 0:
            return False
    return True

def get_primes(limit=50):
    """Generate a list of prime numbers up to a given limit."""
    return [i for i in range(2, limit) if is_prime(i)]

def get_random_prime(primes):
    """Select a random prime number from a list of primes."""
    return random.choice(primes)

def get_phy_n(primes):
    """Generate n and phi(n) using two distinct random prime numbers."""
    p = get_random_prime(primes)
    q = get_random_prime(primes)
    while p == q:
        q = get_random_prime(primes)
    n = p * q
    phi = (p - 1) * (q - 1)
    return phi, n

def get_d(e, phi):
    """Compute the modular multiplicative inverse of e modulo phi."""
    for d in range(1, phi):
        if (e * d) % phi == 1:
            return d

def get_e(phi):
    """Generate a random number e such that 1 < e < phi and gcd(e, phi) == 1."""
    e = random.randint(2, phi - 1)
    while gcd(phi, e) != 1:
        e = random.randint(2, phi - 1)
    return e

def get_ascii(message):
    """Convert a message into a list of ASCII values."""
    return [ord(char) for char in message]

def encrypt(ascii_codes, n, e):
    """Encrypt a list of ASCII codes using the public key (n, e)."""
    return [(code ** e) % n for code in ascii_codes]

if __name__ == "__main__":
    msg = str(sys.argv[1])
    
    primes = get_primes()
    phi, n = get_phy_n(primes)
    e = get_e(phi)
    d = get_d(e, phi)
    
    print(f"Public key: ({n}, {e})")
    print(f"Private key: ({n}, {d})")
    
    ascii_codes = get_ascii(msg)
    encrypted_message = encrypt(ascii_codes, n, e)
    
    print("Encrypted message:", encrypted_message)
    sys.stdout.flush()
