def sort_array(arr):
    """
    Sorts an array in ascending order using a simple sorting algorithm.

    Parameters:
    arr (list): The list of elements to be sorted.

    Returns:
    list: The sorted list.
    """
    n = len(arr)  # Get the length of the array
    
    # Perform a simple sort by swapping elements
    for i in range(n - 1):
        for j in range(i + 1, n):
            if arr[j] < arr[i]:
                # Swap the elements if arr[j] is smaller than arr[i]
                arr[i], arr[j] = arr[j], arr[i]
    
    return arr

def main():
    """
    Main function to initialize the array, sort it, and print the sorted array.
    """
    arr = [1, 3, 5, 7, 9, 2, 4, 6, 8, 0]  # Initialize the array
    
    sorted_arr = sort_array(arr)  # Sort the array
    
    # Print the sorted array
    print("Sorted array:")
    print(sorted_arr)

if __name__ == "__main__":
    main()
