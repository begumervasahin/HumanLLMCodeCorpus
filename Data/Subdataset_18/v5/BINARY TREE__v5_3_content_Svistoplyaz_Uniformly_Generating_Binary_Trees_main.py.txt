from collections import namedtuple, defaultdict
import random
import bisect

TreeNode = namedtuple("TreeNode", "value left_child right_child")

catalan_numbers = [
    1, 1, 2, 5, 14, 42, 132, 429, 1430, 4862, 16796, 58786, 208012, 742900, 2674440, 9694845,
    35357670, 129644790, 477638700, 1767263190, 6564120420, 24466267020, 91482563640, 343059613650,
    1289904147324, 4861946401452
]

max_n = len(catalan_numbers) - 1

average_max_height = [0, 0] + [-1] * (max_n - 1)
average_leaves = [0, 1] + [-1] * (max_n - 1)
average_height = [0, 0] + [-1] * (max_n - 1)

prob_distr = [{0: 1}, {0: 1}] + [{} for _ in range(max_n - 1)]
prob_distr_leaves = [{0: 1}, {1: 1}] + [{} for _ in range(max_n - 1)]
prob_distr_av_height = [{0: 1}, {0: 1}] + [{} for _ in range(max_n - 1)]

cur_value = 0

def generate_binary_tree(n):
    """
    Generate a random binary tree with `n` nodes.
    """
    global cur_value
    if n == 0 or n > max_n:
        return None

    root = TreeNode(value=cur_value, left_child=None, right_child=None)
    cur_value += 1

    if n == 1:
        return root

    chance = [catalan_numbers[0] * catalan_numbers[n - 1]]
    for i in range(1, n):
        chance.append(chance[i - 1] + catalan_numbers[i] * catalan_numbers[n - 1 - i])

    max_number = chance[-1]
    chosen_chance = random.randint(1, max_number)
    index = bisect.bisect_left(chance, chosen_chance)

    root = root._replace(left_child=generate_binary_tree(index))
    root = root._replace(right_child=generate_binary_tree(n - 1 - index))

    return root

def compute_average_max_height(n):
    """
    Compute the average maximum height of a binary tree with `n` nodes.
    """
    if n > max_n:
        return None
    if average_max_height[n] != -1:
        return average_max_height[n]

    total_height = 0
    coef_sum = 0
    for i in range(n):
        coef = catalan_numbers[i] * catalan_numbers[n - 1 - i]
        coef_sum += coef
        total_height += coef * max(compute_average_max_height(i), compute_average_max_height(n - 1 - i))

    average_max_height[n] = 1 + total_height / coef_sum
    return average_max_height[n]

def compute_prob_distr_max_height(n):
    """
    Compute the probability distribution of the maximum height of binary trees with `n` nodes.
    """
    if n > max_n:
        return None
    if prob_distr[n]:
        return prob_distr[n]

    for i in range(n):
        left_distr = compute_prob_distr_max_height(i)
        right_distr = compute_prob_distr_max_height(n - 1 - i)
        coef = catalan_numbers[i] * catalan_numbers[n - 1 - i] / catalan_numbers[n]
        for left_key, left_prob in left_distr.items():
            for right_key, right_prob in right_distr.items():
                new_key = 1 + max(left_key, right_key)
                prob_distr[n][new_key] = prob_distr[n].get(new_key, 0) + left_prob * right_prob * coef

    return prob_distr[n]

def compute_average_leaves(n):
    """
    Compute the average number of leaves in a binary tree with `n` nodes.
    """
    if n > max_n:
        return None
    if average_leaves[n] != -1:
        return average_leaves[n]

    total_leaves = 0
    coef_sum = 0
    for i in range(n):
        coef = catalan_numbers[i] * catalan_numbers[n - 1 - i]
        coef_sum += coef
        total_leaves += coef * (compute_average_leaves(i) + compute_average_leaves(n - 1 - i))

    average_leaves[n] = total_leaves / coef_sum
    return average_leaves[n]

def compute_prob_distr_leaves(n):
    """
    Compute the probability distribution of the number of leaves in binary trees with `n` nodes.
    """
    if n > max_n:
        return None
    if prob_distr_leaves[n]:
        return prob_distr_leaves[n]

    for i in range(n):
        left_distr = compute_prob_distr_leaves(i)
        right_distr = compute_prob_distr_leaves(n - 1 - i)
        coef = catalan_numbers[i] * catalan_numbers[n - 1 - i] / catalan_numbers[n]
        for left_key, left_prob in left_distr.items():
            for right_key, right_prob in right_distr.items():
                new_key = left_key + right_key
                prob_distr_leaves[n][new_key] = prob_distr_leaves[n].get(new_key, 0) + left_prob * right_prob * coef

    return prob_distr_leaves[n]

def sub_func_average_height(n):
    """
    Compute the average height of binary trees with `n` nodes.
    """
    return compute_average_height(n) / n

def compute_average_height(n):
    """
    Compute the total height of binary trees with `n` nodes.
    """
    if n > max_n:
        return None
    if average_height[n] != -1:
        return average_height[n]

    total_height = 0
    coef_sum = 0
    for i in range(n):
        coef = catalan_numbers[i] * catalan_numbers[n - 1 - i]
        coef_sum += coef
        total_height += coef * ((n - 1) + compute_average_height(i) + compute_average_height(n - 1 - i))

    average_height[n] = total_height / coef_sum
    return average_height[n]

def sub_function_distr_av_height(n):
    """
    Compute the probability distribution of the average height of binary trees with `n` nodes.
    """
    distr = compute_prob_distr_av_height(n)
    return {key / n: val for key, val in distr.items()}

def compute_prob_distr_av_height(n):
    """
    Compute the probability distribution of the height of binary trees with `n` nodes.
    """
    if n > max_n:
        return None
    if prob_distr_av_height[n]:
        return prob_distr_av_height[n]

    for i in range(n):
        left_distr = compute_prob_distr_av_height(i)
        right_distr = compute_prob_distr_av_height(n - 1 - i)
        coef = catalan_numbers[i] * catalan_numbers[n - 1 - i] / catalan_numbers[n]
        for left_key, left_prob in left_distr.items():
            for right_key, right_prob in right_distr.items():
                new_key = left_key + right_key + (n - 1)
                prob_distr_av_height[n][new_key] = prob_distr_av_height[n].get(new_key, 0) + left_prob * right_prob * coef

    return prob_distr_av_height[n]
