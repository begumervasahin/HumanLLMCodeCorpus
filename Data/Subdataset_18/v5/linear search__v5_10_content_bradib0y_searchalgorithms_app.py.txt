import time
import math

def recursive_binary_search(lst, target):
    """
    Perform a recursive binary search on a sorted list.
    Return True if the target is found, else return False.
    """
    if not lst:
        return False
    
    midpoint = len(lst) // 2
    if lst[midpoint] == target:
        return True
    elif lst[midpoint] < target:
        return recursive_binary_search(lst[midpoint + 1:], target)
    else:
        return recursive_binary_search(lst[:midpoint], target)

def binary_search(lst, target):
    """
    Perform an iterative binary search on a sorted list.
    Return the index of the target if found, else return None.
    """
    first = 0
    last = len(lst) - 1
    
    while first <= last:
        midpoint = (first + last) // 2
        if lst[midpoint] == target:
            return midpoint
        elif lst[midpoint] < target:
            first = midpoint + 1
        else:
            last = midpoint - 1
    
    return None

def linear_search(lst, target):
    """
    Perform a linear search on a list.
    Return the index of the target if found, else return None.
    """
    for i, value in enumerate(lst):
        if value == target:
            return i
    return None

def log_search_times():
    """
    Test and log the execution times of binary search and recursive binary search
    for lists of increasing sizes.
    """
    for i in range(10, 19):
        numbers = list(range(int(math.pow(10, i))))
        target = int(math.pow(10, i)) - 3
        data_string = f"{len(numbers)};"

        print(f"\n-> Quantity: {target + 3} (10 to the power of {i})")

        # Linear search timing (commented out as it's not being timed)
        print("Linear search: null")
        data_string += "null;"

        # Binary search timing
        start_time = time.time()
        binary_search(numbers, target)
        binary_search_time = time.time() - start_time
        print(f"Binary search: {binary_search_time:.6f} seconds")
        data_string += f"{binary_search_time:.6f};"

        # Recursive binary search timing
        start_time = time.time()
        recursive_binary_search(numbers, target)
        recursive_binary_search_time = time.time() - start_time
        print(f"Recursive binary search: {recursive_binary_search_time:.6f} seconds")
        data_string += f"{recursive_binary_search_time:.6f}"

        with open("data.csv", "a") as file:
            file.write(data_string + "\n")

if __name__ == "__main__":
    log_search_times()
