class Vertex:
    def __init__(self, name):
        """
        Initializes a Vertex object.

        :param name: The name of the vertex.
        """
        self.name = name          # Name of the vertex
        self.neighbors = []       # List to store neighboring vertices

    def add_neighbor(self, neighbor):
        """
        Adds a neighboring vertex to this vertex.

        :param neighbor: A Vertex object representing the neighbor to add.
        :return: False if the neighbor is not a Vertex instance, otherwise None.
        """
        if not isinstance(neighbor, Vertex):
            return False  # Return False if the neighbor is not a Vertex instance

        # Add the neighbor if it's not already present
        if neighbor.name not in self.neighbors:
            self.neighbors.append(neighbor.name)
            neighbor.neighbors.append(self.name)

            # Keep the neighbors lists sorted for both vertices
            self.neighbors.sort()
            neighbor.neighbors.sort()

    def __repr__(self):
        """
        Provides a string representation of the Vertex object.

        :return: A formatted string showing the vertex's name and its neighbors.
        """
        return f"Vertex({self.name}): Neighbors -> {self.neighbors}"
