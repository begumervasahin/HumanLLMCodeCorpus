# Color definitions with their corresponding RGB values
COLORS = {
    "or": (255, 210, 0),
    "argent": (192, 192, 192),
    "azure": (0, 0, 139),
    "gules": (139, 0, 0),
    "purpure": (75, 0, 130),
    "sable": (30, 30, 10),
    "vert": (0, 139, 0),
}

def get_message(dimension, delta):
    """
    Retrieve a message based on the dimension and delta value.
    
    Args:
        dimension (int): The dimension index to retrieve the message from.
        delta (int): The delta value to determine the direction of the message.
        
    Returns:
        str: The corresponding message, or None if not found.
    """
    return MESSAGES[dimension].get(delta)

def get_color(color_name):
    """
    Retrieve the RGB color tuple based on the color name.
    
    Args:
        color_name (str): The name of the color.
        
    Returns:
        tuple: The RGB color tuple, or None if not found.
    """
    return COLORS.get(color_name)
