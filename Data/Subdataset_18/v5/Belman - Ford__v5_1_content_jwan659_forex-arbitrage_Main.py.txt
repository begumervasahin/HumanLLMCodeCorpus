from BellmanFord import BellmanFord
from Edges import Edge
from Vert import Vertex
from forex_python.converter import CurrencyRates
import numpy as np

def main():
    """
    Main function to detect arbitrage opportunities in currency exchange rates
    using the Bellman-Ford algorithm.
    """
    # Initialize the CurrencyRates object
    currency_rates = CurrencyRates()
    
    # Input a list of currencies
    input_string = input('Enter a list of currencies separated by spaces: ')
    currencies = input_string.split()
    
    # Create a list of Vertex objects for each currency
    vertices = create_vertices(currencies)
    
    # Create a list of Edge objects representing exchange rates between currencies
    edges = create_edges(vertices, currency_rates)
    
    # Initialize and run the Bellman-Ford algorithm
    detect_arbitrage(vertices, edges)

def create_vertices(currencies):
    """
    Create a list of Vertex objects for each currency.

    :param currencies: A list of currency codes.
    :return: A list of Vertex objects.
    """
    return [Vertex(currency) for currency in currencies]

def create_edges(vertices, currency_rates):
    """
    Create a list of Edge objects representing exchange rates between currencies.

    :param vertices: A list of Vertex objects.
    :param currency_rates: An instance of CurrencyRates to fetch exchange rates.
    :return: A list of Edge objects.
    """
    edges = []
    for start_vertex in vertices:
        for end_vertex in vertices:
            if start_vertex != end_vertex:
                try:
                    rate = currency_rates.get_rate(str(start_vertex), str(end_vertex))
                    weight = -np.log(rate)
                    edges.append(Edge(start_vertex, end_vertex, weight))
                except Exception as e:
                    print(f"Error fetching rate from {start_vertex} to {end_vertex}: {e}")
    return edges

def detect_arbitrage(vertices, edges):
    """
    Initialize and run the Bellman-Ford algorithm to detect arbitrage opportunities.

    :param vertices: A list of Vertex objects.
    :param edges: A list of Edge objects.
    """
    bellman_ford = BellmanFord(vertices, edges)
    bellman_ford.bellman_ford(vertices[0])
    bellman_ford.print_cycle()

if __name__ == "__main__":
    main()
