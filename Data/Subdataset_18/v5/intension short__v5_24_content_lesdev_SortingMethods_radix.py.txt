import sys
from tkinter import *
from tkinter import simpledialog

def get_numbers_from_user(num_count):
    """Prompts the user to input the specified number of integers."""
    numbers = []
    for _ in range(num_count):
        number = simpledialog.askinteger("Input", "Enter a number:")
        if number is not None:
            numbers.append(number)
    return numbers

def counting_sort(arr, exp):
    """Performs counting sort on arr[] according to the digit represented by exp."""
    n = len(arr)
    output = [0] * n  # Output array that will hold the sorted values
    count = [0] * 10  # Count array to store occurrences of digits

    # Store the count of occurrences in count[]
    for i in range(n):
        index = arr[i] // exp
        count[index % 10] += 1

    # Change count[i] so that it now contains the actual position of this digit in output[]
    for i in range(1, 10):
        count[i] += count[i - 1]

    # Build the output array
    i = n - 1
    while i >= 0:
        index = arr[i] // exp
        output[count[index % 10] - 1] = arr[i]
        count[index % 10] -= 1
        i -= 1

    # Copy the output array to arr[], so that arr[] now contains sorted numbers
    for i in range(n):
        arr[i] = output[i]

def radix_sort(arr):
    """Main function to implement radix sort."""
    max_num = max(arr)
    exp = 1
    while max_num // exp > 0:
        counting_sort(arr, exp)
        exp *= 10

def sorting():
    """Handles the process of collecting user input, sorting it, and displaying the results."""
    try:
        num_count = int(entry_numbers.get())  # Get the number of integers to be inputted
        numbers = get_numbers_from_user(num_count)  # Collect the numbers from the user

        label_original.config(text=f"Original list: {numbers}")  # Display the original list

        radix_sort(numbers)  # Perform the radix sort

        label_sorted.config(text=f"Sorted list: {numbers}")  # Display the sorted list
    except ValueError:
        label_original.config(text="Invalid input! Please enter a valid number.")

# Setup the main application window
app = Tk()
app.title("Radix Sort")
app.geometry('300x200')
app.configure(bg='SkyBlue2')

# Widgets
label_prompt = Label(app, text="How many numbers will you enter?", font='Helvetica 12', bg='SkyBlue2')
label_prompt.pack(pady=10)

entry_numbers = Entry(app, width=8)
entry_numbers.pack()

button_sort = Button(app, text="Sort", command=sorting)
button_sort.pack(pady=10)

label_original = Label(app, text="", bg='SkyBlue2')
label_original.pack(pady=10)

label_sorted = Label(app, text="", bg='SkyBlue2')
label_sorted.pack(pady=10)

# Start the main loop
app.mainloop()
