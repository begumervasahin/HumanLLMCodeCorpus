# Importing necessary libraries
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans

# Define column names for the dataset
column_names = [
    'area', 'perimeter', 'compactness', 'length_of_kernel', 
    'width_of_kernel', 'asymmetry_coefficient', 'length_of_kernel_groove', 'classifier'
]

# Read the dataset into a pandas DataFrame
data = pd.read_table("seeds_dataset.txt", names=column_names, header=None, delimiter='\s+')

# Separate features (X) and target (y) from the dataset
X = data.drop(columns=['classifier'])
y = data['classifier']

# Perform Principal Component Analysis (PCA) to reduce dimensionality to 2 components
pca = PCA(n_components=2)
X_pca = pca.fit_transform(X)

# Scale the features using StandardScaler
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X_pca)

# Determine the optimal number of clusters using the Elbow method
cluster_range = range(1, 7)
cluster_errors = []

for n_clusters in cluster_range:
    # Initialize and fit the KMeans clustering algorithm
    kmeans = KMeans(n_clusters=n_clusters)
    kmeans.fit(X_scaled)
    # Append the inertia (within-cluster sum of squares) to the cluster_errors list
    cluster_errors.append(kmeans.inertia_)

# Create a DataFrame to store the cluster errors for different numbers of clusters
clusters_df = pd.DataFrame({"num_clusters": cluster_range, "cluster_errors": cluster_errors})

# Print the DataFrame to observe the results
print(clusters_df)

# Plot the Elbow method graph to visualize the optimal number of clusters
plt.figure(figsize=(12, 6))
plt.plot(clusters_df.num_clusters, clusters_df.cluster_errors, marker="o")
plt.title('Elbow Method for Optimal k')
plt.xlabel('Number of Clusters (k)')
plt.ylabel('Cluster Error (Inertia)')
plt.grid(True)
plt.show()
