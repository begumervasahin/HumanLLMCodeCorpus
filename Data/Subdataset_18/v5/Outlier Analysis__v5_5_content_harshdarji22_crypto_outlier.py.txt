def get_percentage_change(values):
    """
    Calculate the percentage change for the last 100 days.

    :param values: List of values
    :return: List of percentage changes
    """
    changes = []
    for i in range(-1, -100, -1):
        changes.append(((values[i] - values[i - 1]) / values[i - 1]) * 100)
    return changes

def get_opposite_trend_percentage(base_changes, comparison_changes):
    """
    Calculate the percentage of days where the trends are opposite.

    :param base_changes: List of percentage changes for the base currency
    :param comparison_changes: List of percentage changes for the comparison currency
    :return: Percentage of opposite trends
    """
    opposite_trend_count = 0
    for i in range(len(base_changes)):
        if base_changes[i] * comparison_changes[i] < 0:
            opposite_trend_count += 1
    return (opposite_trend_count / len(base_changes)) * 100
