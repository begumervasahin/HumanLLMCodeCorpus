import csv
import sys

def convert_dat_to_csv(source_file, destination_file):
    """
    Convert a .dat file to a .csv file.

    Parameters:
    source_file (str): Path to the source .dat file.
    destination_file (str): Path to the destination .csv file.
    """
    with open(source_file, 'r') as input_file:
        # Read all lines from the source .dat file
        lines = input_file.readlines()
        # Split each line into a list of values
        new_lines = [line.strip().split() for line in lines]

    with open(destination_file, 'w', newline='') as output_file:
        # Write the list of values to the destination .csv file
        file_writer = csv.writer(output_file)
        file_writer.writerows(new_lines)

def main():
    """
    Main function to handle command line arguments and initiate the conversion.
    """
    if len(sys.argv) != 3:
        print("Usage: python dat2csv.py <source_file> <destination_file>")
        sys.exit(1)
    
    source_file = sys.argv[1]
    destination_file = sys.argv[2]
    
    # Convert the .dat file to .csv file
    convert_dat_to_csv(source_file, destination_file)

if __name__ == "__main__":
    main()
