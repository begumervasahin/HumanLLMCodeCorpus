"""
Precluster a signal by determining the threshold, getting spike times and spike waveforms,
and determining principal components.

.. module:: precluster.py
.. moduleauthor:: Bonne Habekost <b.habekost1@ncl.ac.uk>
.. modulemodified:: July 27, 2014
"""

import numpy as np
import princomp
import thresh
import spktimes
import get_stimtrigavs
import spkwaveform
import filter_spikes

def get(signal, Fs, dt_swf):
    """
    Preclusters a signal by filtering, detecting spikes, and extracting principal components.

    Args:
        signal (array): The input signal.
        Fs (float): The sampling frequency.
        dt_swf (float): Time window for spike waveforms.

    Returns:
        tuple: threshold, spike times, spike heights, spike waveforms, principal components.
    """
    
    # Make a copy of the original signal
    signal_copy = signal.copy()
    
    # Apply high-pass filter to the signal
    signal = filter_spikes.high_pass(signal, Fs, 1000, 3)
    
    # Apply low-pass filter to the signal
    signal = filter_spikes.low_pass(signal, Fs, Fs / 2.0, 1)
    
    # Detect threshold manually
    threshold = thresh.detect(signal, 'manual')
    
    # Detect spike times
    is_spike = spktimes.detect(signal, threshold)
    spike_indices = np.where(is_spike)[0]
    spike_times = spike_indices / float(Fs)
    
    # Get spike-triggered average waveforms
    spike_waveforms = get_stimtrigavs.get_avs(signal, Fs, spike_times, dt_swf, 'zeros')
    
    # Align spike waveforms
    spike_times, spike_waveforms = spkwaveform.align(spike_times, spike_waveforms, dt_swf, Fs)
    
    # Detect spikes within a specified window
    spike_times, spike_waveforms = spkwaveform.window_detect(spike_times, spike_waveforms, [0.0005, 0.001], Fs * 10, 'manual')
    
    # Compute principal components of spike waveforms
    PCs = compute_principal_components(spike_waveforms)
    
    # Recompute spike waveforms using the original signal copy
    spike_waveforms = get_stimtrigavs.get_avs(signal_copy, Fs, spike_times, dt_swf, 'zeros')
    spike_heights = spike_waveforms.max(axis=1)
    
    # Align the recomputed spike waveforms
    spike_times, spike_waveforms = spkwaveform.align(spike_times, spike_waveforms, dt_swf, Fs)
    
    return threshold, spike_times, spike_heights, spike_waveforms, PCs

def compute_principal_components(spike_waveforms):
    """
    Compute the principal components of the spike waveforms.

    Args:
        spike_waveforms (array): Array of spike waveforms.

    Returns:
        dict: Dictionary containing principal components coefficients, scores, and latent values.
    """
    coeff, score, latent = princomp.get(spike_waveforms)
    PCs = {
        'coeff': coeff,
        'score': score,
        'latent': latent
    }
    return PCs
