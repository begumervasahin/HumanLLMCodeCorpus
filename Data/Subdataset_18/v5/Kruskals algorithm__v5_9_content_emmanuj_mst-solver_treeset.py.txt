from node import Node

class TreeSet:
    """
    A class to represent a set of disjoint d-heaps using a TreeSet structure.
    Each element in the set is represented by a Node, which keeps track of its parent and rank.
    """

    def __init__(self):
        """Initialize the TreeSet with an empty dictionary to store nodes."""
        self.nodes = {}

    def make_set(self, value):
        """
        Create a new set with a single element.

        Parameters:
            value: The value of the element to be added to the set.
        """
        if value not in self.nodes:
            self.nodes[value] = Node(value)

    def find(self, value):
        """
        Find the representative (root) of the set containing the given element using path compression.

        Parameters:
            value: The value of the element whose set representative is to be found.

        Returns:
            The value of the representative element.
        """
        node = self.nodes[value]
        if node.parent != node:
            node.parent = self.nodes[self.find(node.parent.value)]  # Path compression
        return node.parent.value

    def union(self, value_u, value_v):
        """
        Unite the sets containing the elements value_u and value_v.

        Parameters:
            value_u: The value of the first element.
            value_v: The value of the second element.

        Returns:
            The representative node of the newly united set.
        """
        root_u = self.nodes[self.find(value_u)]
        root_v = self.nodes[self.find(value_v)]

        if root_u != root_v:  # Only unite if they are in different sets
            if root_u.rank > root_v.rank:
                root_v.parent = root_u
            elif root_u.rank < root_v.rank:
                root_u.parent = root_v
            else:
                root_v.parent = root_u
                root_u.rank += 1

    def print_data(self):
        """Prints the internal data of the TreeSet."""
        for value, node in self.nodes.items():
            parent_value = node.parent.value if node.parent else None
            print(f"Node: {value}, Parent: {parent_value}, Rank: {node.rank}")
