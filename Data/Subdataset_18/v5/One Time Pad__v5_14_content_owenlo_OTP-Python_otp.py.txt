import random

CHARSET = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"

def main():
    """
    Demonstrates the usage of the encrypt and decrypt functions.
    """
    test_vector = "Once more into the fray."
    one_time_pad, encrypted_message = encrypt(test_vector)
    decrypted_message = decrypt(one_time_pad, encrypted_message)
    
    print(f"Test Vector: {test_vector}")
    print(f"One-Time Pad: {one_time_pad}")
    print(f"Encrypted Message: {encrypted_message}")
    print(f"Decrypted Message: {decrypted_message}")

def encrypt(plaintext):
    """
    Encrypts the given plaintext using a randomly generated one-time pad.
    
    Args:
        plaintext (str): The plaintext value to encrypt.
        
    Returns:
        tuple: A tuple containing the one-time pad and the encrypted message.
    """
    # Generate a one-time pad by shuffling the charset
    one_time_pad = "".join(random.sample(CHARSET, len(CHARSET)))
    # Encrypt the message using the one-time pad
    encrypted_message = "".join(one_time_pad[CHARSET.find(c)] for c in plaintext.upper() if c in CHARSET)
    return one_time_pad, encrypted_message

def decrypt(one_time_pad, encrypted_message):
    """
    Decrypts the given secret using the provided one-time pad.
    
    Args:
        one_time_pad (str): The one-time pad used when the secret value was encrypted.
        encrypted_message (str): The value to be decrypted.
        
    Returns:
        str: The decrypted message.
    """
    # Decrypt the message using the one-time pad
    decrypted_message = "".join(CHARSET[one_time_pad.find(c)] for c in encrypted_message.upper() if c in one_time_pad)
    return decrypted_message

if __name__ == "__main__":
    main()
