import random
import time
from bubbleSort import bubbleSort
from selectionSort import selectionSort
from insertionSort import insertionSort
from heapSort import heapSort
from quickSort import quickSort

def yes_no(question):
    """
    Prompt the user with a yes/no question and return True for 'y' and False for 'n'.
    
    Args:
        question (str): The question to prompt the user.
    
    Returns:
        bool: True if the user answers 'y', False if the user answers 'n'.
    """
    while True:
        reply = input(question + " (y/n): ").strip().lower()
        if reply.startswith('y'):
            return True
        elif reply.startswith('n'):
            return False
        else:
            print("Please enter 'y' or 'n'.")

def main():
    """
    Main function to run the sorting scripts.
    """
    print("\nPYTHON SORTING SCRIPTS")
    print("----------------------\n")

    while True:
        # Input validation for the number of elements in the array
        while True:
            n = int(input("How many numbers (between 2 and 50,000; could take a couple minutes) do you want in the array? "))
            if 2 <= n <= 50000:
                break
            else:
                print("Invalid number. Please try again.")

        # Generate a random array
        array = random.sample(range(1, n + 1), n)

        # Optionally display the array
        if yes_no("Do you want to see the array?"):
            print(array)

        # List of sorting functions and their names
        sorting_algorithms = [
            (bubbleSort, "Bubble Sort"),
            (selectionSort, "Selection Sort"),
            (insertionSort, "Insertion Sort"),
            (heapSort, "Heap Sort"),
            (quickSort, "Quick Sort")
        ]

        # Perform each sorting algorithm and measure the time taken
        for sort_function, name in sorting_algorithms:
            print(f"Performing {name}...")
            start = time.time()
            sorted_array = sort_function(array.copy())
            elapsed_time = time.time() - start
            print(f"{name} completed in {elapsed_time:.6f} seconds.\n")

        # Ask the user if they want to try again with another array
        if not yes_no("Try again with another array?"):
            break

    input("Press any key to quit...")

if __name__ == "__main__":
    main()
