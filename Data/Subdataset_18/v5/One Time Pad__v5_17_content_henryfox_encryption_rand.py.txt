import random

def generate_random_number(digits):
    """
    Generate a random number with the specified number of digits.
    
    Parameters:
    digits (int): The number of digits for the random number.
    
    Returns:
    str: A string representing the random number.
    """
    return ''.join(str(random.randint(0, 9)) for _ in range(digits))

def main():
    """
    Main function to demonstrate the usage of generate_random_number.
    
    Returns:
    None
    """
    num_digits = 5
    random_number = generate_random_number(num_digits)
    print(f"Generated random number with {num_digits} digits: {random_number}")

if __name__ == "__main__":
    main()
