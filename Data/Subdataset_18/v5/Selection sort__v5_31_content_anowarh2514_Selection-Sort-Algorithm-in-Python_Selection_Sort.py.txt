def selection_sort(nums):
    """
    Sorts a list of numbers in ascending order using the selection sort algorithm.
    
    Args:
        nums (list): The list of numbers to be sorted.
    """
    n = len(nums)
    for i in range(n - 1):
        # Assume the current position holds the smallest number
        minpos = i
        # Check the rest of the array for a smaller number
        for j in range(i + 1, n):
            if nums[j] < nums[minpos]:
                minpos = j
        # Swap the found minimum element with the current element
        nums[i], nums[minpos] = nums[minpos], nums[i]
        print(f"Step {i + 1}: {nums}")

# Example usage
if __name__ == "__main__":
    nums = [5, 3, 7, 2, 4, 1, 11, 8, 10, 9, 6]
    print("Original list:", nums)
    selection_sort(nums)
    print("Sorted list:", nums)
