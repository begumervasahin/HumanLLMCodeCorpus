import numpy as np
import itertools
from statistics import mean
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d import Axes3D

def get_data(seq):
    return [float(v) for v in seq.split(',')]

def get_power(dim, order):
    p = (order + 1) ** dim
    base = [np.base_repr(i, 3) for i in range(p)]
    base = ["{:02d}".format(int(v)) for v in base]
    for i, v in enumerate(base):
        ii, iii = list(v)
        base[i] = (int(ii), int(iii))
    return base[1:]

def coef_of_det(y, f):
    f_bar = mean(f)
    SSR = sum((fi - f_bar) ** 2 for fi in f)
    SST = sum((yi - y_bar) ** 2 for yi in y)
    SSE = sum((yi - fi) ** 2 for yi, fi in zip(y, f))
    r2 = SSR / SST
    print((SSR + SSE) / SST)
    return r2

def read_input_data(file_path):
    with open(file_path, 'r') as f:
        A = get_data(f.readline())
        B = get_data(f.readline())
        Y = [get_data(line) for line in f]
    Y = np.array(Y).T.flatten()
    return A, B, Y

def construct_linear_equations(A, B, Y, dim, order):
    p, n = (order + 1) ** dim - 1, len(A) * len(B)
    S = np.zeros((p, p))
    M = np.zeros(p)
    x = np.zeros((n, p))
    y = np.array(Y)
    params = np.array([(a, b) for a, b in itertools.product(A, B)])
    powers = get_power(dim, order)

    for i in range(n):
        for j in range(p):
            ii, iii = powers[j]
            x[i][j] = (params[i][0] ** ii) * (params[i][1] ** iii)

    x_bar = np.mean(x, axis=0)
    y_bar = mean(y)

    for j, k in itertools.product(range(p), repeat=2):
        S[j][k] = sum((x[i][j] - x_bar[j]) * (x[i][k] - x_bar[k]) for i in range(n))

    for j in range(p):
        M[j] = sum((x[i][j] - x_bar[j]) * (y[i] - y_bar) for i in range(n))

    a = np.dot(np.linalg.inv(S), M)
    a0 = y_bar - np.dot(a, x_bar)

    return a, a0, powers, x, y, x_bar, y_bar

def polynomial_function(x, a, a0, powers):
    X = [(x[0] ** powers[i][0]) * (x[1] ** powers[i][1]) for i in range(len(a))]
    return np.dot(X, a) + a0

def plot_response_surface(a, a0, powers, y):
    x1 = np.linspace(14500., 15500.)
    x2 = np.linspace(28., 33.)
    z = np.array([[polynomial_function([x1i, x2i], a, a0, powers) for x1i in x1] for x2i in x2])
    
    fig = plt.figure()
    ax = fig.gca(projection='3d')
    x1, x2 = np.meshgrid(x1, x2)
    ax.plot_surface(x1, x2, z, rstride=1, cstride=1, cmap='viridis', linewidth=0.3)
    
    x1 = np.linspace(14500, 15500, 6)
    x2 = np.linspace(28, 33, 6)
    x1, x2 = np.meshgrid(x1, x2)
    ax.scatter(x1, x2, y, c='r')
    plt.show()

    print('Reading input data')
    A, B, Y = read_input_data('aaa.csv')

    print('Constructing Simultaneous Linear Equations')
    dim, order = 2, 2
    a, a0, powers, x, y, x_bar, y_bar = construct_linear_equations(A, B, Y, dim, order)

    print(f'a0: {a0:>10e}')

    print('Computing Coefficient of Determination')
    x1 = np.linspace(14500., 15500.)

    z = np.array([[polynomial_function([x1i, x2i], a, a0, powers) for x1i in x1] for x2i in x2])
    r2 = coef_of_det(y, z.flatten())
    print(f'R2: {r2}')
