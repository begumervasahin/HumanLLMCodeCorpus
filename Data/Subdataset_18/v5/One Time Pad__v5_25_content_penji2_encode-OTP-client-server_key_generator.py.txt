import random
import string

def generate_keys(num_keys, key_file, binary_key_file):
    """
    Generate random keys and their binary representations, writing them to files.
    
    Parameters:
    num_keys (int): The number of keys to generate.
    key_file (str): The file to write the random keys to.
    binary_key_file (str): The file to write the binary representations of the keys to.
    """
    with open(key_file, mode="w") as key_f, open(binary_key_file, mode="w") as bin_key_f:
        for _ in range(num_keys):
            char = random.choice(string.ascii_letters)
            key_f.write(char + "\n")
            binary_representation = f'{ord(char):08b}'
            bin_key_f.write(binary_representation + "\n")

def main():
    """
    Main function to generate keys and write them to files.
    """
    num_keys = 1_000_000
    key_file = "key.dat"
    binary_key_file = "key1.dat"
    
    generate_keys(num_keys, key_file, binary_key_file)

if __name__ == "__main__":
    main()
