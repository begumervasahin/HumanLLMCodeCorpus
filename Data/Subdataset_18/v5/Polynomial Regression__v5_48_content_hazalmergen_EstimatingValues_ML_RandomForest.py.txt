import pandas as pd
import matplotlib.pyplot as plt
from sklearn.model_selection import KFold
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import explained_variance_score, mean_absolute_error, mean_squared_error, r2_score
import numpy as np

# Load the dataset
dataset = pd.read_csv('datas.csv')

# Split the data into features (X) and target (y)
X = dataset.iloc[0:100, 1:6].values
y = dataset.iloc[0:100, 6].values
X_Pred = dataset.iloc[100:120, 1:6].values

# Initialize KFold with 10 splits
kf = KFold(n_splits=10, shuffle=False, random_state=None)

# Initialize lists to store predictions and true values
y_preds = []
y_tests = []

# Initialize array to store final predictions
Y_final_pred = []

# Perform K-Fold cross-validation
for train_index, test_index in kf.split(X):
    # Split the data into training and testing sets
    X_train, X_test = X[train_index], X[test_index]
    y_train, y_test = y[train_index], y[test_index]
    
    # Initialize and fit the Random Forest Regressor
    regressor = RandomForestRegressor(n_estimators=10, random_state=0)
    regressor.fit(X_train, y_train)
    
    # Predict on the test set
    y_pred = regressor.predict(X_test)
    y_preds.extend(y_pred)
    y_tests.extend(y_test)
    
    # Predict on X_Pred and store predictions
    y_pred_X_Pred = regressor.predict(X_Pred)
    Y_final_pred.append(y_pred_X_Pred)

# Convert list of predictions to a numpy array
Y_final_pred = np.array(Y_final_pred)

# Calculate the mean of the final predictions across all folds
Y_final_pred_mean = np.mean(Y_final_pred, axis=0)

# Plot true vs predicted values
plt.scatter(y_tests, y_preds)
plt.plot(y_tests, y_tests, color="black")
plt.xlabel("True Values")
plt.ylabel("Predictions")
plt.title("True vs Predicted Values")
plt.show()

# Calculate and print regression metrics
ex_var_score = explained_variance_score(y_tests, y_preds)
m_absolute_error = mean_absolute_error(y_tests, y_preds)
m_squared_error = mean_squared_error(y_tests, y_preds)
r_2_score = r2_score(y_tests, y_preds)

print("Explained Variance Score:", ex_var_score)
print("Mean Absolute Error:", m_absolute_error)
print("Mean Squared Error:", m_squared_error)
print("R Squared Error:", r_2_score)

# Print final averaged predictions
print("Final Averaged Predictions:", Y_final_pred_mean)
