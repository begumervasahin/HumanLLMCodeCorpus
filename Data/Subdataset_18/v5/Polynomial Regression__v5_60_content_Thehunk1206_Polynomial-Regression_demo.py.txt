import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import time

def load_and_clean_data(file_path):
    """Load and clean the data from the given CSV file."""
    data = pd.read_csv(file_path)
    data.drop(['Unnamed: 0'], axis=1, inplace=True)
    drop_indices = [0, 1, 5, 9, 73, 74, 75, 710, 711, 712, 713, 714, 715, 716, 717, 718, 719]
    data.drop(data.index[drop_indices], inplace=True)
    return data

def polynomial_fit(x, y, degree):
    """Perform polynomial fitting and return the coefficients."""
    return np.polyfit(x, y, degree)

def calculate_error(actual, predicted):
    """Calculate the Mean Squared Error (MSE)."""
    return np.mean((np.array(actual) - np.array(predicted)) ** 2)

def main():
    # Load and clean the data
    data = load_and_clean_data('CleanData.csv')

    # Display data info and statistics
    print(data.info())
    print(data.describe())

    # Extract data for plotting and model training/testing
    x = data['Decimal Date'].tolist()
    y = data['Carbon Dioxide (ppm)'].tolist()
    a = data['Seasonally Adjusted CO2 (ppm)'].tolist()

    x_train = x[:600]
    a_train = a[:600]
    x_test = x[601:703]
    a_actual = a[601:703]

    degree = 2

    # Polynomial fitting
    start_time = time.time()
    coef = polynomial_fit(x_train, a_train, degree)
    print("Polynomial Coefficients:", coef.tolist())

    # Generate training curve
    training_curve = [np.polyval(coef, xi) for xi in x_train]

    # Plot the training data and regression line
    plt.scatter(x_train, a_train, color='red', label='Seasonally Adjusted CO2 (ppm)')
    plt.plot(x_train, training_curve, color='blue', label='Regression Line')
    plt.legend(loc='upper left')

    # Calculate training error
    training_error = calculate_error(a_train, training_curve)
    print("==================Error During Training=======================")
    print("MSE =", training_error)

    # Predict on the test data
    predicted_list = [np.polyval(coef, xi) for xi in x_test]

    # Display predictions
    print("==================Prediction===========================")
    for year, prediction in zip(x_test, predicted_list):
        print(f"Predicted value of CO2 (in ppm) for Year {year}: {prediction}")

    # Calculate prediction error
    prediction_error = calculate_error(a_actual, predicted_list)
    print("==================Prediction Error=======================")
    print("MSE for Prediction =", prediction_error)

    # Display execution time
    end_time = time.time()
    print("Approx execution time:", end_time - start_time, "seconds")

    # Show plot
    plt.show()

if __name__ == "__main__":
    main()
