import random
from collections import defaultdict

def create_grid_and_edges(grid_size=9):
    """
    Creates a grid of nodes and generates edges between adjacent nodes (excluding diagonals).
    
    Args:
        grid_size (int): The size of the grid (default is 9x9).
    
    Returns:
        tuple: A dictionary representing the tree structure and a list of edges.
    """
    tree = defaultdict(list)
    edges = []
    node_id = 0

    for x in range(grid_size):
        for y in range(grid_size):
            # Add the current node to the tree
            tree[node_id].append((x, y))
            node_id += 1

            # Generate edges to adjacent nodes, excluding diagonals
            for dx in range(-1, 2):
                for dy in range(-1, 2):
                    if abs(dx) == abs(dy):
                        continue

                    neighbor = (x + dx, y + dy)
                    if 0 <= neighbor[0] < grid_size and 0 <= neighbor[1] < grid_size:
                        edge = tuple(sorted([(x, y), neighbor]))
                        if edge not in edges:
                            edges.append(edge)

    return tree, edges

def find_tree_containing_node(tree, node):
    """
    Finds and returns the tree ID that contains the given node.

    Args:
        tree (dict): A dictionary representing the tree structure.
        node (tuple): The node to search for in the tree.
    
    Returns:
        tuple: The tree ID and the list of nodes in the tree containing the node.
    """
    for tree_id, nodes in tree.items():
        if node in nodes:
            return tree_id, nodes
    return None, None

def merge_trees(tree, tree1_id, tree2_id):
    """
    Merges two trees into one by extending the first tree with the second tree's nodes.
    
    Args:
        tree (dict): The tree structure.
        tree1_id (int): The ID of the first tree.
        tree2_id (int): The ID of the second tree to merge into the first tree.
    """
    tree[tree1_id].extend(tree[tree2_id])
    tree.pop(tree2_id)

def main():
    # Initialize the grid and edges
    tree, edges = create_grid_and_edges(grid_size=9)

    # Randomly connect nodes until only one tree remains
    while len(tree) > 1:
        random_index = random.randint(0, len(edges) - 1)
        selected_edge = edges[random_index]

        # Identify the trees containing the edge's endpoints
        tree1_id, tree1_nodes = find_tree_containing_node(tree, selected_edge[0])
        tree2_id, tree2_nodes = find_tree_containing_node(tree, selected_edge[1])

        # Skip if both endpoints are already in the same tree
        if tree1_id == tree2_id:
            continue

        # Merge the two trees
        print(f"Connecting edge: {selected_edge}")
        print(f"Tree {tree1_id} before merge: {tree1_nodes}")
        print(f"Tree {tree2_id} before merge: {tree2_nodes}")

        merge_trees(tree, tree1_id, tree2_id)

        # Remove the selected edge
        edges.pop(random_index)

        # Display the current state of the trees
        for tree_nodes in tree.values():
            print(tree_nodes)

        print(f"Remaining edges: {len(edges)}, Remaining trees: {len(tree)}")
        print()

    # Display the final tree structure
    for tree_id, nodes in tree.items():
        print(f"Tree {tree_id}: {nodes}")

if __name__ == "__main__":
    main()
