from PIL import Image

def create_and_display_image():
    """
    Create a 100x100 image with a repeating color pattern and display it.
    """
    # Define the color pattern
    color_pattern = [
        (0, 0, 0),       # Black
        (0, 0, 0),       # Black
        (128, 128, 128), # Gray
        (255, 255, 255), # White
        (255, 255, 255)  # White
    ]
    
    # Calculate the number of pixels in the image
    image_size = (100, 100)
    num_pixels = image_size[0] * image_size[1]
    
    # Repeat the color pattern to fill the entire image
    pattern_length = len(color_pattern)
    repeated_pattern = color_pattern * (num_pixels // pattern_length)
    
    # Create a new image with RGB mode and size 100x100
    image = Image.new("RGB", image_size, "white")
    
    # Apply the repeated color pattern to the image
    image.putdata(repeated_pattern)
    
    # Display the image
    image.show()

if __name__ == "__main__":
    create_and_display_image()
