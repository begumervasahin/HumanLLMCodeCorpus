def binary_search(alist, item):
    first = 0
    last = len(alist) - 1

    while first <= last:
        midpoint = (first + last) // 2
        
        if alist[midpoint] == item:
            return True
        elif item < alist[midpoint]:
            last = midpoint - 1
        else:
            first = midpoint + 1

    return False

# Test the binary_search function
test_list = [0, 1, 2, 8, 13, 17, 19, 32, 42]
print(binary_search(test_list, 3))   # Output: False
print(binary_search(test_list, 13))  # Output: True
