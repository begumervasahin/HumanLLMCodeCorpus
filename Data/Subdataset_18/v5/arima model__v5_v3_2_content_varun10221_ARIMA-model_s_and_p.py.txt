import os
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.tsa.seasonal import seasonal_decompose
import seaborn as sns

def read_stock_data(file_path):
    """
    Read stock data from a CSV file, parse dates, sort by date, and set the date as the index.
    
    Parameters:
    file_path (str): The path to the CSV file containing stock data.
    
    Returns:
    pd.DataFrame: A DataFrame with the stock data.
    """
    stock_data = pd.read_csv(file_path)
    stock_data['Date'] = pd.to_datetime(stock_data['Date'])
    stock_data.sort_values(by='Date', inplace=True)
    stock_data.set_index('Date', inplace=True)
    return stock_data

def calculate_statistics(stock_data):
    """
    Calculate various statistics for the stock data, including differences, log values, and variances.
    
    Parameters:
    stock_data (pd.DataFrame): The DataFrame containing stock data.
    
    Returns:
    pd.DataFrame: The DataFrame with additional calculated statistics.
    """
    stock_data['First Difference'] = stock_data['Close'] - stock_data['Close'].shift()
    stock_data['Natural Log'] = np.log(stock_data['Close'])
    stock_data['Original Variance'] = stock_data['Close'].rolling(window=30).var()
    stock_data['Log Variance'] = stock_data['Natural Log'].rolling(window=30).var()
    stock_data['Logged First Difference'] = stock_data['Natural Log'].diff()
    stock_data['Lag 20'] = stock_data['Logged First Difference'].shift(20)
    return stock_data

def visualize_data(stock_data):
    """
    Visualize the stock data, including decomposing the time series and plotting forecasts.
    
    Parameters:
    stock_data (pd.DataFrame): The DataFrame containing stock data with calculated statistics.
    """
    # Decompose the time series
    decomposition = seasonal_decompose(stock_data['Natural Log'], model='additive', period=30)
    
    # Fit ARIMA model
    model = sm.tsa.ARIMA(stock_data['Logged First Difference'].iloc[1:], order=(0, 0, 1))
    results = model.fit(disp=-1)
    stock_data['Forecast'] = results.fittedvalues

    # Plot the data
    plt.figure(figsize=(16, 12))
    plt.plot(stock_data['Logged First Difference'], label='Logged First Difference')
    plt.plot(stock_data['Forecast'], label='Forecast')
    plt.legend()
    plt.show()

def main():
    """
    Main function to execute the data reading, calculation, and visualization pipeline.
    """
    sns.set_style('darkgrid')

    # Read stock data
    file_path = os.path.join(os.getcwd(), 'q_table.csv')
    stock_data = read_stock_data(file_path)

    # Calculate statistics
    stock_data = calculate_statistics(stock_data)

    # Visualize data
    visualize_data(stock_data)

if __name__ == "__main__":
    main()
