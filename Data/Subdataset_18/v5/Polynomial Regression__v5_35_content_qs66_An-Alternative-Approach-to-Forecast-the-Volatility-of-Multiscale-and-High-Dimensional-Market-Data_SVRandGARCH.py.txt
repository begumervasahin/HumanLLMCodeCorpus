# Import necessary libraries
import numpy as np
import pandas as pd
from sklearn.svm import SVR
import matplotlib.pyplot as plt

# Load and preprocess the data
def load_and_preprocess_data(file_path):
    data = pd.read_csv(file_path)
    data = data[data['DEXCHUS'] != '.']
    data['DEXCHUS'] = data['DEXCHUS'].astype(float)
    data['returns'] = 100 * data['DEXCHUS'].pct_change().dropna()
    data['log_ret'] = np.log(data['DEXCHUS']) - np.log(data['DEXCHUS'].shift(1))
    data = data[data['log_ret'] != 0]
    data['vol_squared'] = data['log_ret']**2
    data['log_vol_squared'] = np.log(data['vol_squared'])
    data = data[1:].reset_index(drop=True)
    return data

# Plot the data
def plot_data(data):
    plt.figure(figsize=(12, 6))
    data['DEXCHUS'].plot(title='DEXCHUS', label='DEXCHUS')
    plt.legend()
    plt.show()

    plt.figure(figsize=(12, 6))
    data['returns'].plot(title='Returns', label='Returns')
    plt.legend()
    plt.show()

    plt.figure(figsize=(12, 6))
    data['vol_squared'].plot(title='Volatility Squared', label='Volatility Squared')
    plt.legend()
    plt.show()

# Prepare the data for SVR
def prepare_data(data, n):
    X = np.zeros((data.shape[0] - n, n))
    for i in range(n, data.shape[0]):
        X[i - n] = data['log_ret'][i - n:i]
    y = data['log_vol_squared'][n:]
    return X, y

# Train SVR models and make predictions
def train_and_predict_svr(X, y):
    models = {
        'rbf': SVR(kernel='rbf', C=1e3, gamma=250),
        'linear': SVR(kernel='linear', C=1e3, gamma=250),
        'poly': SVR(kernel='poly', C=1e3, gamma=250)
    }
    
    predictions = {}
    for name, model in models.items():
        predictions[name] = np.exp(model.fit(X, y).predict(X))
    
    return predictions

# Plot the real vs predicted volatility
def plot_predictions(real_vol, predicted_vol, n, title_suffix=''):
    xaxis = range(len(real_vol) - n)
    plt.figure(figsize=(12, 6))
    plt.plot(xaxis, real_vol[-len(xaxis):], color='r', label='Real Volatility')
    plt.plot(xaxis, predicted_vol['rbf'][-len(xaxis):], color='b', label='SVR RBF')
    plt.plot(xaxis, predicted_vol['linear'][-len(xaxis):], color='g', label='SVR Linear')
    plt.plot(xaxis, predicted_vol['poly'][-len(xaxis):], color='orange', label='SVR Poly')
    plt.title(f'SVR Volatility Forecasting {title_suffix}')
    plt.legend()
    plt.show()

# Main function to run the SVR models
def main():
    data = load_and_preprocess_data('DEXCHUS.csv')
    plot_data(data)
    
    n = 50
    X, y = prepare_data(data, n)
    predicted_vol = train_and_predict_svr(X, y)
    real_vol = data['vol_squared'][n:]
    plot_predictions(real_vol, predicted_vol, n)

    # Run SVR with moving windows of different lengths
    for window in range(10, 60, 10):
        X, y = prepare_data(data, window)
        predicted_vol = train_and_predict_svr(X, y)
        plot_predictions(data['vol_squared'][window:], predicted_vol, window, f'window length: {window} days')

# Execute the main function
if __name__ == "__main__":
    main()
