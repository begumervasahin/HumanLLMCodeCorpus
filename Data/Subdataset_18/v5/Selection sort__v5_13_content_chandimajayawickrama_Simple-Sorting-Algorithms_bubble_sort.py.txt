import time

def bubble_sort(array):
    """
    Perform a bubble sort on the given array in descending order.

    Args:
    array (list): The list of integers to be sorted.

    Returns:
    list: The sorted list.
    """
    start_time = time.time()
    n = len(array)
    
    if n == 0:
        return array

    for i in range(n):
        swapped = False
        for j in range(0, n - i - 1):
            if array[j] < array[j + 1]:
                array[j], array[j + 1] = array[j + 1], array[j]
                swapped = True
        if not swapped:
            break
    
    end_time = time.time()
    print(f"Sorting completed in {end_time - start_time:.5f} seconds")
    return array

def get_input_list():
    """
    Read a list of integers from user input.

    Returns:
    list: The list of integers entered by the user.
    """
    input_str = input("Enter a list of integers separated by space: ")
    return [int(x) for x in input_str.split()]

def main():
    """
    Main function to read input, perform sorting, and handle exceptions.
    """
    try:
        input_list = get_input_list()
        sorted_list = bubble_sort(input_list)
        print("Sorted list:", sorted_list)
    except ValueError:
        print("Invalid input. Please enter a list of integers separated by spaces.")

if __name__ == "__main__":
    main()
