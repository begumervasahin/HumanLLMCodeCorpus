from flask import Flask, render_template, flash, request
from flask_bootstrap import Bootstrap
from flask_wtf import FlaskForm
from flask_wtf.file import FileField
from wtforms import SubmitField, ValidationError
from werkzeug.utils import secure_filename

class SimpleForm(FlaskForm):
    file = FileField('Input file')
    submit_button = SubmitField('Run Code')

    def validate_file(self, field):
        """Validate that a file has been uploaded."""
        if not field.data:
            raise ValidationError('No file uploaded.')

def calculate_valid_multiples(n, x, y):
    """
    Calculate numbers that are multiples of x but not multiples of y up to n.

    Args:
        n (int): Upper limit.
        x (int): Multiples of this number.
        y (int): Not multiples of this number.

    Returns:
        list: List of numbers meeting the criteria.
    """
    result = []
    for i in range(max(x, 1), n):
        if i % x == 0 and i % y != 0:
            result.append(i)
    return result

def process_file(file):
    """
    Process the uploaded file and return results or error messages.

    Args:
        file (FileStorage): Uploaded file.

    Returns:
        list: List of results or error messages.
    """
    try:
        file_lines = file.readlines()
        elems = int(file_lines[0].strip())
        
        if elems >= 100:
            return ["The number of elements exceeds the limit."]

        results = []
        for line in file_lines[1:]:
            try:
                n, x, y = map(int, line.strip().split())
                result = calculate_valid_multiples(n, x, y)
                results.append(' '.join(map(str, result)))
            except ValueError:
                return ["Each line after the first should contain three integers."]

        return results
    except Exception as e:
        return [f"An error occurred: {str(e)}"]
