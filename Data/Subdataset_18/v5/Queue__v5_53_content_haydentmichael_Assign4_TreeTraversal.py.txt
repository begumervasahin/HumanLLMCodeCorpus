class TreeNode:
    """
    A class to represent a node in a binary tree.

    Attributes:
    val : int or None
        The value of the node.
    left : TreeNode or None
        The left child node.
    right : TreeNode or None
        The right child node.
    """
    def __init__(self, value):
        self.val = value
        self.left = None
        self.right = None

def inorder_traversal(node, result=None):
    """
    Perform in-order traversal on a binary tree.

    Parameters:
    node : TreeNode
        The root node of the binary tree.
    result : list, optional
        A list to store the traversal result. Defaults to an empty list.

    Returns:
    list
        The in-order traversal result.
    """
    if result is None:
        result = []
    if node:
        inorder_traversal(node.left, result)
        if node.val is not None:
            result.append(node.val)
        inorder_traversal(node.right, result)
    return result

def preorder_traversal(node, result=None):
    """
    Perform pre-order traversal on a binary tree.

    Parameters:
    node : TreeNode
        The root node of the binary tree.
    result : list, optional
        A list to store the traversal result. Defaults to an empty list.

    Returns:
    list
        The pre-order traversal result.
    """
    if result is None:
        result = []
    if node:
        if node.val is not None:
            result.append(node.val)
        preorder_traversal(node.left, result)
        preorder_traversal(node.right, result)
    return result

def populate_tree(values, index=0):
    """
    Populate a binary tree from a list of values.

    Parameters:
    values : list
        A list of values to populate the binary tree. None values indicate absent nodes.
    index : int, optional
        The current index in the list of values. Defaults to 0.

    Returns:
    TreeNode
        The root node of the populated binary tree.
    """
    if index < len(values) and values[index] is not None:
        node = TreeNode(values[index])
        node.left = populate_tree(values, 2 * index + 1)
        node.right = populate_tree(values, 2 * index + 2)
        return node
    return None

def print_traversal_result(title, result):
    """
    Print the result of a tree traversal.

    Parameters:
    title : str
        The title to print before the result.
    result : list
        The list of traversal results to print.
    """
    print(title)
    for value in result:
        print(value)
    print(" ")

# Test with different arrays
test_arrays = [
    [1, None, 2, None, None, 3],
    [1, 2, 3, 4, 5, 6, None],
    [1, 2, 3]
]

for arr in test_arrays:
    root = populate_tree(arr)

    inorder_result = inorder_traversal(root)
    print_traversal_result("In-order Traversal", inorder_result)

    preorder_result = preorder_traversal(root)
    print_traversal_result("Pre-order Traversal", preorder_result)
