"""
The :mod:`autostopwordgen.generator` module implements a research paper published by Olatunde et al.,
https://www.researchgate.net/publication/318969652_AN_AUTO-GENERATED_APPROACH_OF_STOP_WORDS_USING_AGGREGATED_ANALYSIS
Created on Sat Feb 2 20:50:34 2019
@author: Nagaraju Budigam
"""

import pandas as pd
import numpy as np
from sklearn.feature_extraction.text import CountVectorizer, TfidfVectorizer
import logging

logging.basicConfig(format='%(asctime)s : %(levelname)s : %(message)s', level=logging.INFO)


class AutoStopWordsGen:
    def __init__(self, corpus):
        """
        Initialize the AutoStopWordsGen with a given corpus.

        :param corpus: List of documents as strings.
        """
        logging.info('Initializing the AutoStopwordsGen started')

        # Vectorize the corpus using CountVectorizer and TfidfVectorizer
        self.cv = CountVectorizer()
        self.cvft = self.cv.fit_transform(corpus)
        self.tfidfcv = TfidfVectorizer()
        self.tfidfcvft = self.tfidfcv.fit_transform(corpus)

        logging.info('Initializing the AutoStopwordsGen completed')

    def get_stopwords(self, top_n=0.95, last_n=0.1):
        """
        Generate stopwords based on term frequency, entropy, and variance probability.

        :param top_n: Top n percent threshold.
        :param last_n: Last n percent threshold.
        :return: List of stopwords.
        """
        logging.info('Generating stopwords started')

        # Create a DataFrame with word frequencies and IDF values
        word_freq_df = pd.DataFrame({
            'word': self.tfidfcv.get_feature_names_out(),
            'frequency': np.asarray(self.cvft.sum(axis=0)).ravel(),
            'idf': self.tfidfcv.idf_
        })

        # Sort words by frequency in descending order
        word_freq_df.sort_values(by='frequency', ascending=False, inplace=True)

        # Calculate probability, entropy, and variance probability for each word
        word_freq_df['prob'] = word_freq_df['frequency'] / word_freq_df['frequency'].sum()
        word_freq_df['entropy'] = word_freq_df['prob'].apply(lambda x: x * np.log(1 / x) if x > 0 else 0)
        word_freq_df['vp'] = np.power(word_freq_df['prob'] - word_freq_df['prob'].mean(), 2)

        # Initialize stopwords dictionary
        stopwords = {'frequency': [], 'idf': [], 'entropy': [], 'vp': []}
        metrics = ['frequency', 'entropy', 'vp']

        # Determine stopwords based on frequency, entropy, and variance probability thresholds
        for metric in metrics:
            if metric == 'frequency':
                threshold = word_freq_df[metric].quantile(top_n)
                stopwords[metric] = word_freq_df[word_freq_df[metric] >= threshold]['word'].tolist()
            else:
                threshold = word_freq_df[metric].quantile(last_n)
                stopwords[metric] = word_freq_df[word_freq_df[metric] <= threshold]['word'].tolist()

        # Convert lists to sets for intersection operation
        for key in stopwords:
            stopwords[key] = set(stopwords[key])

        # Find words that have very high aggregation
        very_high_aggregation = set(word_freq_df[word_freq_df['frequency'] < 2]['word'].tolist())
        very_high_aggregation.update(
            stopwords['frequency'].intersection(stopwords['entropy']).intersection(stopwords['vp'])
        )

        logging.info('Generating stopwords completed')

        return list(very_high_aggregation)
