"""

The Rabin-Karp algorithm searches for a substring within a text.
If text has length b, and the substring has length s, the Rabin-Karp algorithm
can perform the search in O(b+s) time on average and O(bs) time in the
worst case. This is better than O(s(b-s)) if we search the substring
iteratively in text.

d: number of characters in the alphabet
q: a prime number
h: d^(m-1)
hash(txt[n...n+m)]) = (d*(hash(txt[n...n+m-1]) - h*txt[n]) + txt[n+m]) % q
"""

def rabin_karp_algo(text, substring):
    """
    Search for the substring in the text using the Rabin-Karp algorithm.

    Args:
        text (str): The text in which to search for the substring.
        substring (str): The substring to search for.

    Returns:
        list: A list of starting indices where the substring is found in the text.
    """
    found = []
    n = len(text)
    m = len(substring)
    q = 101  # A prime number
    d = 128  # Number of characters in the alphabet

    # Calculate h = d^(m-1) % q
    h = 1
    for i in range(m-1):
        h = (h * d) % q

    # Calculate the hash values for the substring and the first window of the text
    substring_hash = 0
    text_hash = 0
    for i in range(m):
        substring_hash = (d * substring_hash + ord(substring[i])) % q
        text_hash = (d * text_hash + ord(text[i])) % q

    # Slide the pattern over the text one by one
    for i in range(n - m + 1):
        # Check if the current window's hash matches the substring's hash
        if substring_hash == text_hash:
            # Check the characters one by one for a match
            if text[i:i+m] == substring:
                found.append(i)
                print(f"Substring found at index {i}")

        # Calculate the hash value for the next window of text
        if i < n - m:
            text_hash = (d * (text_hash - ord(text[i]) * h) + ord(text[i + m])) % q
            # We might get a negative value of text_hash, convert it to positive
            if text_hash < 0:
                text_hash += q

    return found

# Example usage
if __name__ == "__main__":
    result = rabin_karp_algo('Get bit; set bit; clear bit; update bit', 'bit')
    print(f"Substring found at indices: {result}")
