def edit_distance(s1, s2):
    """
    Calculates the Levenshtein distance (edit distance) between two strings s1 and s2.

    Parameters:
    s1 (str): The first string.
    s2 (str): The second string.

    Returns:
    int: The edit distance between s1 and s2.
    """
    # Ensure s1 is the shorter string to optimize space complexity
    if len(s1) > len(s2):
        s1, s2 = s2, s1

    # Initialize the distance matrix for the first row
    previous_row = list(range(len(s1) + 1))

    # Compute the edit distance row by row
    for i2, c2 in enumerate(s2):
        # Start the new row with the incremented index
        current_row = [i2 + 1]  

        for i1, c1 in enumerate(s1):
            # Check if characters match or compute the minimum edit distance
            if c1 == c2:
                current_row.append(previous_row[i1])
            else:
                current_row.append(1 + min(
                    previous_row[i1],      # Substitution
                    previous_row[i1 + 1],  # Insertion
                    current_row[-1]        # Deletion
                ))

        # Update previous_row for the next iteration
        previous_row = current_row

    # The last element in the last row is the edit distance
    return previous_row[-1]  


def alternative_distance(s1, s2):
    """
    An alternative implementation for calculating the Levenshtein distance between two strings s1 and s2.

    Parameters:
    s1 (str): The first string.
    s2 (str): The second string.

    Returns:
    int: The edit distance between s1 and s2.
    """
    # Ensure s1 is the longer string for the recursive case handling
    if len(s1) < len(s2):
        return alternative_distance(s2, s1)

    # Base case: If the second string is empty, return the length of the first string
    if len(s2) == 0:
        return len(s1)

    # Initialize the first row of the matrix
    previous_row = list(range(len(s2) + 1))

    # Compute the edit distance row by row
    for i, c1 in enumerate(s1):
        # Start the row with the incremented index
        current_row = [i + 1]  

        for j, c2 in enumerate(s2):
            # Compute the cost of insertions, deletions, and substitutions
            insertions = previous_row[j + 1] + 1
            deletions = current_row[j] + 1
            substitutions = previous_row[j] + (c1 != c2)

            # Append the minimum cost to the current row
            current_row.append(min(insertions, deletions, substitutions))

        # Update previous_row for the next iteration
        previous_row = current_row

    # The last element contains the edit distance
    return previous_row[-1]
