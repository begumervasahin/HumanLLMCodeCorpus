import csv
import string
from typing import List, Dict

def get_player_list() -> List[Dict[str, str]]:
    """
    Prompts the user to enter a filename, reads the CSV file, and returns a list of player dictionaries.
    
    Returns:
        List[Dict[str, str]]: A list of dictionaries, each representing a player.
    """
    filename = input('Enter filename (without .csv extension): ') + '.csv'
    player_list = []

    try:
        with open(filename, newline='') as file:
            reader = csv.reader(file)
            # Skip the header row
            next(reader)
            
            for row in reader:
                if len(row) >= 12:  # Ensure there are enough columns
                    player = {
                        'FIRST': row[2],
                        'LAST': row[4],
                        'POS': row[1],
                        'PRICE': row[6],
                        'TEAM': row[9],
                        'OPP': row[10],
                        'INJ': row[11]
                    }
                    player_list.append(player)
                else:
                    print(f"Warning: Skipping row with insufficient columns: {row}")
                    
    except FileNotFoundError:
        print(f"Error: The file '{filename}' was not found.")
    except Exception as e:
        print(f"An error occurred: {e}")
    
    return player_list

def trim_names(player_list: List[Dict[str, str]]) -> List[Dict[str, str]]:
    """
    Removes punctuation from the first and last names of players in the provided list.
    
    Args:
        player_list (List[Dict[str, str]]): A list of dictionaries, each representing a player.
    
    Returns:
        List[Dict[str, str]]: The modified list with trimmed player names.
    """
    translator = str.maketrans('', '', string.punctuation)
    for player in player_list:
        player['FIRST'] = player['FIRST'].translate(translator)
        player['LAST'] = player['LAST'].translate(translator)
    
    return player_list

def main():
    players = get_player_list()
    if players:  # Only proceed if the player list is not empty
        players = trim_names(players)
        for player in players:
            print(player)

if __name__ == "__main__":
    main()
