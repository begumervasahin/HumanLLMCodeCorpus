import os
import sys
from euclidean import *
from numberPrime import *
from modular_arithmetic import *

class RSA:
    def __init__(self):
        """
        Initialize the RSA object by generating the parameters n, d, and e.
        """
        self.n, self.d, self.e = GenParms(n_bits=100, DEBUG=True)

    def encrypt(self, message):
        """
        Encrypt the message using RSA encryption.
        
        Parameters:
        message (list of int): The message to encrypt as a list of integers.
        
        Returns:
        list of int: The encrypted message as a list of integers.
        """
        return [modExp(i, self.e, self.n) for i in message]

    def decrypt(self, message):
        """
        Decrypt the message using RSA decryption.
        
        Parameters:
        message (list of int): The encrypted message to decrypt as a list of integers.
        
        Returns:
        list of int: The decrypted message as a list of integers.
        """
        return [modExp(i, self.d, self.n) for i in message]

    def get_public_key(self):
        """
        Get the public key (n, e).
        
        Returns:
        tuple: The public key.
        """
        return self.n, self.e

    def get_private_key(self):
        """
        Get the private key (n, d).
        
        Returns:
        tuple: The private key.
        """
        return self.n, self.d

def main():
    """
    Main function to test the RSA implementation.
    """
    print("Testing RSA Implementation:")
    rsa = RSA()
    
    print('\nPublic Key:', rsa.get_public_key())
    print('Private Key:', rsa.get_private_key())

    # Example usage
    message = [42, 12, 89]  # Example message to encrypt and decrypt
    print('\nOriginal Message:', message)

    encrypted_message = rsa.encrypt(message)
    print('Encrypted Message:', encrypted_message)

    decrypted_message = rsa.decrypt(encrypted_message)
    print('Decrypted Message:', decrypted_message)

    print('\nFinal Decrypted Message:', decrypted_message)

if __name__ == '__main__':
    main()
