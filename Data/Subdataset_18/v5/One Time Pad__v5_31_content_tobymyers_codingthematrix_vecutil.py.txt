from vec import Vec

def list2vec(L):
    """
    Given a list L of field elements, return a Vec with domain {0, ..., len(L)-1}
    whose entry i is L[i].

    Args:
        L (list): A list of field elements.

    Returns:
        Vec: A vector with domain {0, ..., len(L)-1} and corresponding entries from L.

    Examples:
        >>> list2vec([10, 20, 30])
        Vec({0, 1, 2}, {0: 10, 1: 20, 2: 30})
    """
    domain = set(range(len(L)))
    function = {k: L[k] for k in range(len(L))}
    return Vec(domain, function)

def zero_vec(D):
    """
    Returns a zero vector with the given domain.

    Args:
        D (set): The domain of the vector.

    Returns:
        Vec: A vector with the given domain and all zero entries.

    Examples:
        >>> zero_vec({0, 1, 2})
        Vec({0, 1, 2}, {0: 0, 1: 0, 2: 0})
    """
    function = {d: 0 for d in D}
    return Vec(D, function)
