import numpy as np
import csv

# Constants
TEST_SIZE = 7000
STEP = 1
DATA_FILENAME = 'distributions.csv'

def write_data(values, filename):
    """
    Writes a list of dictionaries to a CSV file.

    Parameters:
    values (list of dict): The data to write to the file.
    filename (str): The name of the file to write to.
    """
    fieldnames = ['uniform', 'binomial', 'poisson', 'normal']
    
    print('Writing data to file...')
    with open(filename, 'w', newline='') as output_file:
        csv_writer = csv.DictWriter(output_file, fieldnames=fieldnames)
        csv_writer.writeheader()
        csv_writer.writerows(values)
    print(f'Data successfully written to {filename}.')

def generate_distributions(test_size, step):
    """
    Generates multiple random distributions.

    Parameters:
    test_size (int): The number of samples for each distribution.
    step (int): The step size for iterating through the generated distributions.

    Returns:
    list of dict: A list of dictionaries, each containing one sample from each distribution.
    """
    uniform_dist = np.random.uniform(-1, 1, test_size)
    binomial_dist = np.random.binomial(1000, 0.5, test_size)
    poisson_dist = np.random.poisson(1000, test_size)
    normal_dist = np.random.normal(0, 0.1, test_size)
    
    distributions_list = [
        {
            'uniform': uniform_dist[i],
            'binomial': binomial_dist[i],
            'poisson': poisson_dist[i],
            'normal': normal_dist[i]
        }
        for i in range(0, test_size, step)
    ]
    
    return distributions_list

if __name__ == "__main__":
    distributions = generate_distributions(TEST_SIZE, STEP)
    write_data(distributions, DATA_FILENAME)
