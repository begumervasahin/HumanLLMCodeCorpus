def load_frequency_dictionary(file_path):
    """
    Loads the frequency dictionary from a file.

    Parameters:
        file_path (str): The path to the frequency file.

    Returns:
        dict: A dictionary with words as keys and their frequencies as values.
    """
    dictionary = {}
    with open(file_path, 'r', encoding='utf-8') as file:
        for line in file:
            # Skip lines that do not start with '0' or '1'
            if line[0] not in ['0', '1']:
                continue
            freq, word = line.strip().split()
            dictionary[word] = int(freq)
    return dictionary

def is_stop_word(word, dictionary):
    """
    Checks if the given word is in the dictionary.

    Parameters:
        word (str): The word to check.
        dictionary (dict): The frequency dictionary.

    Returns:
        bool: True if the word is in the dictionary, False otherwise.
    """
    return word in dictionary

def get_word_freq(word, dictionary):
    """
    Returns the frequency of the given word.

    Parameters:
        word (str): The word to check.
        dictionary (dict): The frequency dictionary.

    Returns:
        int: The frequency of the word if it is in the dictionary, 0 otherwise.
    """
    return dictionary.get(word, 0)

def check_stop_words(stop_words_file, dictionary):
    """
    Checks if stop words are in the dictionary and prints results.

    Parameters:
        stop_words_file (str): The path to the stop words file.
        dictionary (dict): The frequency dictionary.
    """
    with open(stop_words_file, 'r', encoding='utf-8') as file:
        for word in file:
            word = word.strip()
            if not is_stop_word(word, dictionary):
                print(f'{word} is not in the dictionary')

if __name__ == '__main__':
    # Load the frequency dictionary
    frequency_dict = load_frequency_dictionary("dosyalar/derlemtr2016-10000.txt")

    # Check the stop words
    check_stop_words("dosyalar/turkce-stop-words", frequency_dict)
