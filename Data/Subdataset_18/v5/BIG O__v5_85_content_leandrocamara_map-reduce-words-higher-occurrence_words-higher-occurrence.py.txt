import csv
import glob
import sys
import mincemeat

# Define the path to the text files
PATH = '/home/leandro/Documents/puc/06-solutions/activities/map-reduce-words-higher-occurrence/'

# Get the list of text files in the data directory
text_files = glob.glob(PATH + 'data/*')

def file_contents(file_name):
    """Read the contents of a file."""
    with open(file_name, 'r') as f:
        return f.read()

def mapfn(k, v):
    """Map function to process each line of the input files."""
    from stopwords import allStopWords

    for line in v.splitlines():
        fields = line.split(':::')
        authors = fields[1]
        title = fields[2]

        for word in title.split():
            word = word.strip('.,:?!()""')
            if word not in allStopWords:
                for author in authors.split('::'):
                    if author in ['Grzegorz Rozenberg', 'Philip S. Yu']:
                        yield word, author

def reducefn(k, v):
    """Reduce function to count the occurrences of authors."""
    total_philip = v.count('Philip S. Yu')
    total_rozenberg = v.count('Grzegorz Rozenberg')

    if total_philip > 5 or total_rozenberg > 5:
        authors = []
        if total_rozenberg > 0:
            authors.append(f'Grzegorz Rozenberg: {total_rozenberg}')
        if total_philip > 0:
            authors.append(f'Philip S. Yu: {total_philip}')
        return authors
    return None

def write_results_to_csv(results, path):
    """Write the results to a CSV file."""
    with open(path + 'result.csv', 'w', newline='') as csvfile:
        writer = csv.writer(csvfile)
        for k, v in results.items():
            if v is not None:
                writer.writerow([k, v])

def main():
    """Main function to set up and run the map-reduce job."""
    # Create a dictionary with file names as keys and file contents as values
    source = {file_name: file_contents(file_name) for file_name in text_files}

    # Set up the mincemeat server
    s = mincemeat.Server()
    s.datasource = source
    s.mapfn = mapfn
    s.reducefn = reducefn

    # Run the server and get the results
    results = s.run_server(password="mapreduce")

    # Write the results to a CSV file
    write_results_to_csv(results, PATH)

if __name__ == "__main__":
    main()
