import time
import bst

def get_student_details():
    """Collects and returns student details from the user."""
    roll_num = int(input("Enter the roll number: "))
    name = input("Enter the name: ")
    address = input("Enter the address: ")
    mobile = int(input("Enter the mobile number: "))
    course = input("Enter the course: ")

    print("Enter the marks in three subjects:")
    marks = [int(input(f"Subject {i + 1}: ")) for i in range(3)]

    total_marks = sum(marks)
    average_marks = total_marks / 3

    print(f"Total Marks: {total_marks}")
    print(f"Average Marks: {average_marks:.2f}")

    return {
        "roll_num": roll_num,
        "name": name,
        "address": address,
        "mobile": mobile,
        "course": course,
        "total_marks": total_marks,
        "average_marks": average_marks
    }

def insert_students_into_bst(bst_tree, num_students):
    """Inserts student details into the BST."""
    for i in range(num_students):
        print(f"\nEnter the details of Student {i + 1}:")
        student_details = get_student_details()
        bst_tree.insert(student_details["roll_num"])  # Use roll number as the key

def delete_keys_from_bst(bst_tree):
    """Handles deletion of keys from the BST."""
    while True:
        key_to_delete = int(input("Enter the roll number to delete: "))
        bst_tree.remove(key_to_delete)
        if input("Do you want to delete another key? (y/n): ").lower() != 'y':
            break
    print("\nAfter deletion, BST In-Order Traversal:\n")
    bst_tree.inorder()

def search_keys_in_bst(bst_tree):
    """Handles searching for keys in the BST."""
    while True:
        key_to_search = int(input("Enter the roll number to search: "))
        bst_tree.search(key_to_search)
        if input("Do you want to search another key? (y/n): ").lower() != 'y':
            break

def main():
    # Initialize the BST
    bst_tree = bst.BST()

    # Input the number of students
    num_students = int(input("Enter the number of students: "))

    # Insert student details into the BST
    insert_students_into_bst(bst_tree, num_students)

    # Display the BST in-order traversal
    print("\nBST In-Order Traversal:\n")
    bst_tree.inorder()

    # Handle deletion of keys from the BST
    if input("Do you want to delete any keys in the tree? (y/n): ").lower() == 'y':
        delete_keys_from_bst(bst_tree)

    # Handle searching for keys in the BST
    if input("Do you want to search any keys in the tree? (y/n): ").lower() == 'y':
        search_keys_in_bst(bst_tree)

    print("\n************************* END OF BST OPERATIONS *******************************\n")

if __name__ == '__main__':
    main()
