def partition(arr, start, end):
    """
    Partition the array around the pivot element.

    Parameters:
    arr (list): The array to be partitioned.
    start (int): The starting index of the array segment to partition.
    end (int): The ending index of the array segment to partition.

    Returns:
    int: The partition index where the pivot element is placed.
    """
    pivot = arr[start]
    left = start + 1
    right = end
    done = False

    while not done:
        while left <= right and arr[left] <= pivot:
            left += 1
        while right >= left and arr[right] >= pivot:
            right -= 1
        if right < left:
            done = True
        else:
            arr[left], arr[right] = arr[right], arr[left]

    arr[start], arr[right] = arr[right], arr[start]
    return right

def quicksort(arr, start, end):
    """
    Perform QuickSort on the array.

    Parameters:
    arr (list): The array to be sorted.
    start (int): The starting index of the array segment to sort.
    end (int): The ending index of the array segment to sort.

    Returns:
    list: The sorted array.
    """
    global total_operations
    if start < end:
        total_operations += end - start
        split_point = partition(arr, start, end)
        total_operations += abs(start - (split_point - 1))
        quicksort(arr, start, split_point - 1)
        total_operations += abs((split_point + 1) - end)
        quicksort(arr, split_point + 1, end)
    return arr

def read_list_from_file(file_path):
    """
    Read a list of integers from a file.

    Parameters:
    file_path (str): The path to the file containing the list of integers.

    Returns:
    list: The list of integers.
    """
    with open(file_path, "r") as file:
        return [int(line.strip()) for line in file]

def main():
    """
    Main function to execute the QuickSort algorithm on a list read from a file.
    """
    file_path = "list.txt"
    my_list = read_list_from_file(file_path)
    
    sorted_list = quicksort(my_list, 0, len(my_list) - 1)
    print("Sorted List:", sorted_list)
    print("Total Operations:", total_operations)

if __name__ == "__main__":
    total_operations = 0
    main()
