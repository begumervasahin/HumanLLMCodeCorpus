This script demonstrates the use of the Central Limit Theorem (CLT) based algorithm to mine frequent patterns and 
form association rules from a transactional dataset.
"""

import pyclt  # Import the Central Limit Theorem based algorithm package
import fp2ar  # Import the package to form association rules from frequent patterns
import re  # Import the regular expression package for splitting strings and extracting items from transactions

def read_dataset(file_name):
    """
    Read the transactional dataset from a file.

    Parameters:
    file_name (str): The name of the file containing the dataset.

    Returns:
    list: A list of transactions, each transaction is a list of items.
    """
    transactions = []
    with open(file_name, 'r') as file:
        content = file.readlines()
        for line in content:
            items = re.split(r' |\n', line)
            transaction = [int(item) for item in items if item.isdigit()]
            if transaction:
                transactions.append(transaction)
    return transactions

def main():
    """
    Main function to read the dataset, mine frequent patterns, and form association rules.
    """
    # Read the transactional dataset
    # The dataset kosarak.dat can be downloaded from http://fimi.ua.ac.be/data/
    dataset = read_dataset("kosarak.dat")

    # Parameters for the CLT-based algorithm
    min_support = 0.03
    support_deviation_range = 0.005
    sample_probability = 0.01

    # Mine the dataset for frequent patterns using the CLT-based algorithm
    frequent_patterns, sample_size = pyclt.getFP(dataset, min_support, support_deviation_range, sample_probability)

    # Parameters for forming association rules
    min_confidence = 0.75
    min_lift = 1

    # Form association rules from frequent patterns
    association_rules = fp2ar.getAR(frequent_patterns, sample_size, min_confidence, min_lift)

    # Output the frequent patterns and association rules
    print("Frequent Patterns:")
    for pattern in frequent_patterns:
        print(pattern)
    
    print("\nAssociation Rules:")
    for rule in association_rules:
        print(rule)

if __name__ == "__main__":
    main()
