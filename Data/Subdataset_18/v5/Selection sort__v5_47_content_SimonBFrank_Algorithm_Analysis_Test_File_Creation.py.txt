import random

def generate_descending_list(n):
    """Generate a descending list of numbers from n to 1."""
    return list(range(n, 0, -1))

def write_list_to_file(filename, number_list):
    """Write each number in the list to a file, each on a new line."""
    with open(filename, "w") as f:
        for number in number_list:
            f.write(f"{number}\n")

# Define the number of elements and the filename
n = 100000
filename = "Descending_100000"

# Generate the descending list
number_list = generate_descending_list(n)

# Write the list to the file
write_list_to_file(filename, number_list)
