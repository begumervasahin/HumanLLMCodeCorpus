import numpy as np
import time

def selection_sort(arr):
    """Sorts an array in place using the selection sort algorithm."""
    n = len(arr)
    for i in range(n):
        min_index = i
        for j in range(i + 1, n):
            if arr[j] < arr[min_index]:
                min_index = j
        # Swap the found minimum element with the first element
        arr[i], arr[min_index] = arr[min_index], arr[i]

def measure_execution_time(arr, description):
    """Measures and prints the execution time of the selection sort algorithm."""
    start_time = time.perf_counter()
    selection_sort(arr)
    end_time = time.perf_counter()
    
    execution_time = end_time - start_time
    print(f"Execution time for selection sort on {description} array of size {len(arr)}: {execution_time:.6f} seconds")
    print(f"Sorted array: {arr}\n")

def generate_test_arrays(size):
    """Generates test arrays for different scenarios."""
    random_arr = np.random.randint(0, 99999, size)
    ascending_arr = np.sort(random_arr)
    reverse_arr = ascending_arr[::-1]
    return random_arr, ascending_arr, reverse_arr

def main():
    """Main function to execute selection sort on different types of arrays."""
    arr_size = 40
    random_arr, ascending_arr, reverse_arr = generate_test_arrays(arr_size)

    measure_execution_time(random_arr.copy(), "random order")
    measure_execution_time(ascending_arr.copy(), "ascending order")
    measure_execution_time(reverse_arr.copy(), "reverse order")

if __name__ == "__main__":
    main()
