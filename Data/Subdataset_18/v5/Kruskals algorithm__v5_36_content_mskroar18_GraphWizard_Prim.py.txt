class Graph:
    def run_prim(self, edges, node_count):
        """
        Runs Prim's algorithm to find the Minimum Spanning Tree (MST) of a graph.

        Parameters:
        edges (list): A list of edges where each edge is an object with 'weight', 'vertex1', and 'vertex2'.
        node_count (int): The total number of nodes in the graph.

        Returns:
        set: A set containing the selected edges that form the MST.
        """
        # Sort edges by weight
        edges.sort(key=lambda edge: edge.weight)

        # Initialize the MST and the set of connected vertices
        mst_edges = set()
        connected_vertices = set()

        # Start with the smallest edge
        smallest_edge = edges[0]
        mst_edges.add(smallest_edge)
        connected_vertices.update({smallest_edge.vertex1, smallest_edge.vertex2})

        # Keep adding edges until we have the MST
        while len(mst_edges) < node_count - 1:
            for edge in edges:
                # Check if one vertex is in the connected set and the other is not
                if (edge.vertex1 in connected_vertices) != (edge.vertex2 in connected_vertices):
                    mst_edges.add(edge)
                    connected_vertices.update({edge.vertex1, edge.vertex2})
                    break  # Exit the loop once an edge is added

        return mst_edges

    @staticmethod
    def return_weight(edge):
        """Returns the weight of an edge."""
        return edge.weight
