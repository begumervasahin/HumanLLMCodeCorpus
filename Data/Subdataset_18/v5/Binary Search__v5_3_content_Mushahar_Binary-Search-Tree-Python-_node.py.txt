class Node:
    """
    Node class for a binary search tree (BST).
    This class is designed to be used internally within the BST.
    """

    def __init__(self, val):
        """
        Initialize the Node with a given value.
        """
        self.value = val
        self.left = None
        self.right = None

    def insert(self, val):
        """
        Insert a value into the BST.
        Returns True if the insertion is successful, False if the value already exists.
        """
        if val == self.value:
            return False
        elif val < self.value:
            if self.left:
                return self.left.insert(val)
            else:
                self.left = Node(val)
                return True
        else:  # val > self.value
            if self.right:
                return self.right.insert(val)
            else:
                self.right = Node(val)
                return True

    def find(self, val):
        """
        Search for a value in the BST.
        Returns True if the value is found, False otherwise.
        """
        if val == self.value:
            return True
        elif val < self.value:
            return self.left.find(val) if self.left else False
        else:  # val > self.value
            return self.right.find(val) if self.right else False

    def preorder(self):
        """
        Perform a pre-order traversal of the BST.
        Prints the values of the nodes.
        """
        print(self.value)
        if self.left:
            self.left.preorder()
        if self.right:
            self.right.preorder()

    def inorder(self):
        """
        Perform an in-order traversal of the BST.
        Prints the values of the nodes.
        """
        if self.left:
            self.left.inorder()
        print(self.value)
        if self.right:
            self.right.inorder()

    def postorder(self):
        """
        Perform a post-order traversal of the BST.
        Prints the values of the nodes.
        """
        if self.left:
            self.left.postorder()
        if self.right:
            self.right.postorder()
        print(self.value)

    def delete(self, val):
        """
        Delete a node with the specified value from the BST and rearrange the tree nodes accordingly.
        Returns the root of the modified subtree.
        """
        if val == self.value:
            # Node with two children
            if self.left and self.right:
                successor = self.right.find_min()
                self.value = successor.value
                self.right = self.right.delete(successor.value)
            # Node with only one child or no children
            else:
                return self.left if self.left else self.right
        elif val < self.value:
            if self.left:
                self.left = self.left.delete(val)
        else:  # val > self.value
            if self.right:
                self.right = self.right.delete(val)
        
        return self

        Find the minimum value node in the subtree rooted at the current node.
        The minimum value node is the leftmost node.
