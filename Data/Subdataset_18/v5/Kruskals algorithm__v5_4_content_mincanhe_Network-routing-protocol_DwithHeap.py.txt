from collections import defaultdict
from heapq import heappop, heappush

def dijkstra(edges, start, target):
    """
    Implements Dijkstra's algorithm to find the shortest path between two nodes.

    Parameters:
        edges (list of tuples): A list of tuples representing the graph's edges, where each tuple is (from_node, to_node, cost).
        start (str): The starting node for the path.
        target (str): The target node for the path.

    Returns:
        tuple: A tuple containing the total cost of the shortest path and the path itself as a linked tuple structure.
               If no path exists, returns (infinity, None).
    """
    # Create a graph as an adjacency list
    graph = defaultdict(list)
    for from_node, to_node, cost in edges:
        graph[from_node].append((cost, to_node))

    # Priority queue to store (cost, current_node, path)
    priority_queue = [(0, start, ())]

    # Set to keep track of visited nodes
    visited = set()

    while priority_queue:
        # Pop the node with the smallest cost
        current_cost, current_node, path = heappop(priority_queue)

        if current_node not in visited:
            visited.add(current_node)
            path = (current_node, path)  # Update the path

            # If the target node is reached, return the cost and path
            if current_node == target:
                return current_cost, path

            # Explore neighbors
            for neighbor_cost, neighbor in graph[current_node]:
                if neighbor not in visited:
                    heappush(priority_queue, (current_cost + neighbor_cost, neighbor, path))

    # If the target node is not reachable, return infinity and None for the path
    return float("inf"), None

def extract_path(path):
    """
    Helper function to extract the path from the linked tuple structure.

    Parameters:
        path (tuple): The linked tuple structure representing the path.

    Returns:
        list: A list of nodes representing the path from start to target.
    """
    result = []
    while path:
        node, path = path
        result.append(node)
    return result[::-1]  # Reverse the list to get the correct order
