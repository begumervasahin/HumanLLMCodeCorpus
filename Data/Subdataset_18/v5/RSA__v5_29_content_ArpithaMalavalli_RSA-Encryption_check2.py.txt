import random

# Function to perform modular exponentiation
def exponent_mod(base, exp, mod):
    if base == 0:
        return 0
    if exp == 0:
        return 1
    if exp % 2 == 0:
        half_exp = exponent_mod(base, exp // 2, mod)
        return (half_exp * half_exp) % mod
    else:
        return (base * exponent_mod(base, exp - 1, mod)) % mod

# Function to check if a number is prime
def is_prime(n):
    if n <= 1:
        return False
    for i in range(2, int(n ** 0.5) + 1):
        if n % i == 0:
            return False
    return True

# Function to calculate the greatest common divisor
def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

# Function to generate the public key
def generate_public_key(n, phi):
    candidates = [e for e in range(2, phi) if gcd(e, phi) == 1]
    e = random.choice(candidates)
    return n, e

# Function to generate the private key
def generate_private_key(n, phi, e):
    k = 1
    while (k * phi + 1) % e != 0:
        k += 1
    d = (k * phi + 1) // e
    return n, d

# Generate a list of prime numbers in a given range
primes = [i for i in range(10, 100) if is_prime(i)]

# Select two distinct prime numbers
p = random.choice(primes)
primes.remove(p)
q = random.choice(primes)

# Calculate n and phi
n = p * q
phi = (p - 1) * (q - 1)

# Generate public and private keys
public_key = generate_public_key(n, phi)
private_key = generate_private_key(n, phi, public_key[1])

# Display the generated keys
print("Public key (n, e):", public_key)
print("Private key (n, d):", private_key)

# Original message to be encrypted
original_message = 1099
print("Original message:", original_message)

# Encrypt and decrypt the message
encrypted_message = exponent_mod(original_message, public_key[1], n)
decrypted_message = exponent_mod(encrypted_message, private_key[1], n)

# Display the encrypted and decrypted messages
print("Encrypted message:", encrypted_message)
print("Decrypted message:", decrypted_message)
