def find_min_distance_node(distance_dict, nodes):
    """
    Finds the node with the minimum distance in the given list of nodes.

    Parameters:
    distance_dict (dict): Dictionary of nodes and their respective distances.
    nodes (list): List of nodes to search through.

    Returns:
    int: The node with the minimum distance.
    """
    min_distance = float('inf')
    min_node = -1
    for node in nodes:
        if distance_dict[node] < min_distance:
            min_distance = distance_dict[node]
            min_node = node
    return min_node

def dijkstra(matrix, origin):
    """
    Implements Dijkstra's algorithm to find the shortest paths from the origin node to all other nodes.

    Parameters:
    matrix (list): Adjacency matrix representing the graph.
    origin (int): The starting node.

    Returns:
    dict: Dictionary of nodes and their minimum distances from the origin.
    list: List of previous nodes in the shortest path.
    """
    inf = float('inf')
    num_nodes = len(matrix)
    nodes = list(range(num_nodes))
    visited = []

    # Initialize distance and previous node information
    min_distance = {i: inf for i in range(num_nodes)}
    min_distance[origin] = 0
    previous_nodes = [-1] * num_nodes

    while nodes:
        # Select the unvisited node with the smallest distance
        current_node = find_min_distance_node(min_distance, nodes)
        
        if current_node == -1:
            print(f"Cannot reach remaining nodes: {nodes}")
            break
        
        # Process the neighbors of the current node
        for neighbor in range(num_nodes):
            if matrix[current_node][neighbor] < inf:
                new_distance = min_distance[current_node] + matrix[current_node][neighbor]
                if new_distance < min_distance[neighbor]:
                    min_distance[neighbor] = new_distance
                    previous_nodes[neighbor] = current_node
        
        # Mark the current node as visited and remove it from the unvisited nodes
        visited.append(current_node)
        nodes.remove(current_node)
        
        # Debugging output to trace the computation
        print(f"{previous_nodes[current_node]} -> {current_node}")

    return min_distance, previous_nodes

def main():
    inf = float('inf')
    matrix = [
        [0, 1, inf, 2, inf, inf],
        [inf, 0, 3, 4, inf, inf],
        [inf, inf, 0, 5, 1, inf],
        [inf, 4, inf, 0, inf, inf],
        [inf, inf, 2, 3, 0, inf],
        [inf, inf, 2, inf, 2, 0]
    ]
    origin = 1

    min_distance, previous_nodes = dijkstra(matrix, origin)
    print("\nMinimum distances from origin:", min_distance)

if __name__ == "__main__":
    main()
