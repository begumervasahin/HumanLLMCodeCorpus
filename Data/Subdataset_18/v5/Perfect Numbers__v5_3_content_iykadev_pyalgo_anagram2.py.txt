def is_anagram(s1, s2):
    """
    Check if two strings are anagrams of each other.
    
    Parameters:
    s1 (str): The first string.
    s2 (str): The second string.
    
    Returns:
    bool: True if the strings are anagrams, False otherwise.
    """
    def clean_string(s):
        """
        Remove spaces and convert to lowercase for accurate comparison.
        
        Parameters:
        s (str): The string to be cleaned.
        
        Returns:
        str: The cleaned string.
        """
        return s.replace(" ", "").lower()
    
    cleaned_s1 = clean_string(s1)
    cleaned_s2 = clean_string(s2)
    
    return sorted(cleaned_s1) == sorted(cleaned_s2)

def main():
    """
    Main function to test the is_anagram function with example inputs.
    
    Returns:
    None
    """
    s1 = 'ey edip'
    s2 = 'pide ye'
    result = is_anagram(s1, s2)
    print(f"'{s1}' and '{s2}' are anagrams: {result}")

# Run the main function
if __name__ == "__main__":
    main()
