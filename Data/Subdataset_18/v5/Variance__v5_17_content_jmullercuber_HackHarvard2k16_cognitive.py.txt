import json
import httplib
import time

# Headers for the API requests
headers = {
    'Content-Type': 'application/json',
    'Ocp-Apim-Subscription-Key': 'b3f5d9f8d81046598dedc07a7541e2c9',
}

def documents_to_sentiments(docs):
    """Analyze the sentiments of a list of documents."""
    
    # Prepare the request body
    c_body = {
        'documents': [
            {'id': str(i), 'text': doc}
            for i, doc in enumerate(docs) if doc
        ]
    }

    # Initialize the connection
    conn = httplib.HTTPSConnection('westus.api.cognitive.microsoft.com')
    
    # Make the API request
    conn.request("POST", "/text/analytics/v2.0/sentiment", json.dumps(c_body), headers)
    response = conn.getresponse()
    data = response.read()
    conn.close()

    # Parse and return the response
    result = json.loads(data)
    return {'body': c_body, 'result': result}

def documents_to_topics(docs):
    """Extract topics from a list of documents."""
    
    # Prepare the request body
    c_body = {
        'stop_words': [],
        'topicsToExclude': [],
        'documents': [
            {'id': str(i), 'text': doc}
            for i, doc in enumerate(docs)
        ]
    }

    # Initialize the connection
    conn = httplib.HTTPSConnection('westus.api.cognitive.microsoft.com')
    
    # Make the API request
    conn.request("POST", "/text/analytics/v2.0/topics", json.dumps(c_body), headers)
    response = conn.getresponse()
    op_loc = response.getheader('operation-location')

    print(response.getheaders())
    print(op_loc)

    # Adjust operation location URL
    if '.com' in op_loc:
        op_loc = op_loc[op_loc.index('.com') + 4:]

    print(op_loc)

    res = {'body': c_body}

    # Poll for the result until it's ready
    while True:
        time.sleep(5)
        conn2 = httplib.HTTPSConnection('westus.api.cognitive.microsoft.com')
        conn2.request("GET", op_loc, '', headers)
        response = conn2.getresponse()
        data = response.read()
        data = json.loads(data)

        if data.get('status') == 'Succeeded':
            res['result'] = data
            break

        conn2.close()

    conn.close()
    return res
