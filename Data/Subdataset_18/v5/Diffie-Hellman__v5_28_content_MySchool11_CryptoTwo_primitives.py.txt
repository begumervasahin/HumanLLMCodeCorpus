from random import choice

def is_primitive_root(candidate, prime):
    """
    Determines if a candidate is a primitive root of a given prime number.

    A primitive root of a prime number is an integer that generates all the numbers from 1 to (prime - 1) when raised to successive powers modulo the prime.

    Args:
        candidate (int): The integer to check.
        prime (int): The prime number to test against.

    Returns:
        bool: True if the candidate is a primitive root, False otherwise.
    """
    residues = set()
    
    for exponent in range(1, prime):
        residue = pow(candidate, exponent, prime)
        residues.add(residue)
    
    # A primitive root will generate all residues from 1 to (prime - 1)
    return len(residues) == prime - 1

def find_random_primitive_root(prime):
    """
    Finds and returns a random primitive root of the given prime number.

    Args:
        prime (int): The prime number for which to find a primitive root.

    Returns:
        int: A randomly selected primitive root, or None if no primitive roots are found.
    """
    print("Calculating public key and base...")
    
    primitive_roots = [
        candidate 
        for candidate in range(1, prime) 
        if is_primitive_root(candidate, prime)
    ]
    
    # Randomly select one primitive root, if available
    return choice(primitive_roots) if primitive_roots else None

    primitive_root = find_random_primitive_root(prime_number)
