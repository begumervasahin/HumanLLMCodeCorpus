class NodeRB:
    """
    Node class for Red-Black Tree (RBT). 
    Each node has a key, color, and pointers to its parent and children.
    """

    def __init__(self, key, color="Red"):
        """
        Initialize a new node with a given key and default color red.
        """
        self.key = key
        self.left = None
        self.right = None
        self.parent = None
        self.color = color

    def get(self):
        """ Return the key of the node. """
        return self.key

    def set(self, key):
        """ Set the key of the node. """
        self.key = key

    def set_color(self, color):
        """ Set the color of the node. """
        self.color = color

    def get_color(self):
        """ Return the color of the node. """
        return self.color

    def get_children(self):
        """ Return the list of the node's children. """
        children = []
        if self.left is not None:
            children.append(self.left)
        if self.right is not None:
            children.append(self.right)
        return children


class RBT:
    """
    Red-Black Tree (RBT) data structure implementation.
    """

    def __init__(self):
        """
        Initialize an empty Red-Black Tree with a sentinel nil node.
        """
        self.nil = NodeRB(None, color="Black")
        self.root = self.nil

    def set_root(self, key):
        """
        Set the root of the Red-Black Tree with a given key.
        The root is always colored black.
        """
        self.root = NodeRB(key, color="Black")
        self.root.parent = self.nil
        self.root.left = self.nil
        self.root.right = self.nil

    def left_rotate(self, x):
        """
        Perform a left rotation around the given node x.
        """
        y = x.right
        x.right = y.left
        if y.left != self.nil:
            y.left.parent = x
        y.parent = x.parent
        if x.parent == self.nil:
            self.root = y
        elif x == x.parent.left:
            x.parent.left = y
        else:
            x.parent.right = y
        y.left = x
        x.parent = y

    def right_rotate(self, x):
        """
        Perform a right rotation around the given node x.
        """
        y = x.left
        x.left = y.right
        if y.right != self.nil:
            y.right.parent = x
        y.parent = x.parent
        if x.parent == self.nil:
            self.root = y
        elif x == x.parent.left:
            x.parent.left = y
        else:
            x.parent.right = y
        y.right = x
        x.parent = y

    def insert(self, key):
        """
        Insert a new node with the given key into the Red-Black Tree.
        """
        new_node = NodeRB(key)
        self._insert_node(new_node)

    def _insert_node(self, z):
        """
        Insert the node z into the Red-Black Tree and then fix up the tree.
        """
        y = self.nil
        current_node = self.root
        while current_node != self.nil:
            y = current_node
            if z.key < current_node.key:
                current_node = current_node.left
            else:
                current_node = current_node.right

        z.parent = y
        if y == self.nil:
            self.root = z
        elif z.key < y.key:
            y.left = z
        else:
            y.right = z

        z.left = self.nil
        z.right = self.nil
        z.color = "Red"
        self._insert_fixup(z)

    def _insert_fixup(self, z):
        """
        Restore the Red-Black Tree properties after inserting a node.
        """
        while z.parent.color == "Red":
            if z.parent == z.parent.parent.left:
                y = z.parent.parent.right
                if y.color == "Red":
                    z.parent.color = "Black"
                    y.color = "Black"
                    z.parent.parent.color = "Red"
                    z = z.parent.parent
                else:
                    if z == z.parent.right:
                        z = z.parent
                        self.left_rotate(z)
                    z.parent.color = "Black"
                    z.parent.parent.color = "Red"
                    self.right_rotate(z.parent.parent)
            else:
                y = z.parent.parent.left
                if y.color == "Red":
                    z.parent.color = "Black"
                    y.color = "Black"
                    z.parent.parent.color = "Red"
                    z = z.parent.parent
                else:
                    if z == z.parent.left:
                        z = z.parent
                        self.right_rotate(z)
                    z.parent.color = "Black"
                    z.parent.parent.color = "Red"
                    self.left_rotate(z.parent.parent)
        self.root.color = "Black"

    def find(self, key):
        """
        Search for a node with the given key in the Red-Black Tree.
        Returns True if the key is found, False otherwise.
        """
        return self._find_node(self.root, key)

    def _find_node(self, node, key):
        """
        Helper method to recursively search for a key in the Red-Black Tree.
        """
        if node == self.nil or key == node.key:
            return node != self.nil
        elif key < node.key:
            return self._find_node(node.left, key)
        else:
            return self._find_node(node.right, key)

    def inorder(self):
        """
        Perform an in-order traversal of the Red-Black Tree and print each node's key and color.
        """
        def _inorder(node):
            if node != self.nil:
                _inorder(node.left)
                print(f"{node.key} {node.color}")
                _inorder(node.right)

        _inorder(self.root)
