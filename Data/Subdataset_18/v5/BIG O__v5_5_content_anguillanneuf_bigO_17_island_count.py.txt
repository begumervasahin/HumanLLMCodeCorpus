def search(mat, i, j, rows, cols):
    """
    Recursive function to traverse the matrix and mark visited cells as 0.
    
    Args:
        mat (list of list of int): The binary matrix.
        i (int): The current row index.
        j (int): The current column index.
        rows (int): The total number of rows in the matrix.
        cols (int): The total number of columns in the matrix.
    """
    if 0 <= i < rows and 0 <= j < cols and mat[i][j] == 1:
        mat[i][j] = 0  # Mark the cell as visited
        search(mat, i + 1, j, rows, cols)
        search(mat, i, j + 1, rows, cols)
        search(mat, i - 1, j, rows, cols)
        search(mat, i, j - 1, rows, cols)

def get_number_of_islands(binaryMatrix):
    """
    Recursive approach to count the number of islands in a binary matrix.
    
    Args:
        binaryMatrix (list of list of int): The binary matrix.
    
    Returns:
        int: The number of islands.
    """
    rows, cols = len(binaryMatrix), len(binaryMatrix[0])
    count = 0
    for i in range(rows):
        for j in range(cols):
            if binaryMatrix[i][j] == 1:
                search(binaryMatrix, i, j, rows, cols)
                count += 1
    return count

print("Number of islands (recursive):", get_number_of_islands(binaryMatrix1))

def dfs_helper(stack, binaryMatrix, i, j, rows, cols):
    """
    Helper function for depth-first search to push valid positions onto the stack.
    
    Args:
        stack (list of tuple of int): The stack used for DFS.
        binaryMatrix (list of list of int): The binary matrix.
        i (int): The current row index.
        j (int): The current column index.
        rows (int): The total number of rows in the matrix.
        cols (int): The total number of columns in the matrix.
    """
    if 0 <= i < rows and 0 <= j < cols and binaryMatrix[i][j] == 1:
        stack.append((i, j))

def depth_first_search(binaryMatrix, i, j, rows, cols):
    """
    Iterative depth-first search to traverse the matrix and mark visited cells as 0.
    
    Args:
        binaryMatrix (list of list of int): The binary matrix.
        i (int): The starting row index.
        j (int): The starting column index.
        rows (int): The total number of rows in the matrix.
        cols (int): The total number of columns in the matrix.
    """
    stack = [(i, j)]
    while stack:
        i, j = stack.pop()
        binaryMatrix[i][j] = 0  # Mark the cell as visited
        dfs_helper(stack, binaryMatrix, i + 1, j, rows, cols)
        dfs_helper(stack, binaryMatrix, i - 1, j, rows, cols)
        dfs_helper(stack, binaryMatrix, i, j + 1, rows, cols)
        dfs_helper(stack, binaryMatrix, i, j - 1, rows, cols)

def get_number_of_islands_iterative(binaryMatrix):
    """
    Iterative approach to count the number of islands in a binary matrix.
    
    Args:
        binaryMatrix (list of list of int): The binary matrix.
    
    Returns:
        int: The number of islands.
    """
    count = 0
    rows, cols = len(binaryMatrix), len(binaryMatrix[0])
    for i in range(rows):
        for j in range(cols):
            if binaryMatrix[i][j] == 1:
                count += 1
                depth_first_search(binaryMatrix, i, j, rows, cols)
    return count
