import pandas as pd
from format import clean_str, shuffle_both

def read_train_data(file_name, x_col, y_col, separator='\t', with_cleaning=True, with_header=False, with_shuffle=True):
    """
    Reads training data from a file, optionally cleans and shuffles it.
    
    Parameters:
    - file_name: str, the name of the file to read
    - x_col: int, the column index for input data
    - y_col: int, the column index for labels
    - separator: str, the separator used in the file (default is tab)
    - with_cleaning: bool, whether to clean the data (default is True)
    - with_header: bool, whether the file contains a header row (default is False)
    - with_shuffle: bool, whether to shuffle the data (default is True)
    
    Returns:
    - A tuple of lists (x_text, y_text) with the input data and labels
    """
    # Read the CSV file
    header_option = None if with_header else 'infer'
    data = pd.read_csv(file_name, sep=separator, header=header_option)

    # Extract and optionally clean the text data
    x_text = _process_lines(data[x_col].tolist(), with_cleaning)
    y_text = data[y_col].tolist()

    # Shuffle the data if required
    if with_shuffle:
        x_text, y_text = shuffle_both(x_text, y_text)

    return x_text, y_text

def read_test_data(file_name, x_col, with_cleaning=True):
    """
    Reads test data from a file and optionally cleans it.
    
    Parameters:
    - file_name: str, the name of the file to read
    - x_col: int, the column index for input data
    - with_cleaning: bool, whether to clean the data (default is True)
    
    Returns:
    - A list of cleaned input data
    """
    # Read the CSV file
    data = pd.read_csv(file_name, header=None)

    # Extract and optionally clean the text data
    return _process_lines(data[x_col].tolist(), with_cleaning)

def _process_lines(lines, with_cleaning=True):
    """
    Decodes and optionally cleans each line.
    
    Parameters:
    - lines: list of str, the lines to process
    - with_cleaning: bool, whether to clean the data (default is True)
    
    Returns:
    - A list of processed lines
    """
    if with_cleaning:
        return [clean_str(line.decode('utf-8')) for line in lines]
    else:
        return [line.decode('utf-8') for line in lines]
