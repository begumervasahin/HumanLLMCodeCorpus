"""
This file contains source code for reading and extracting data from PDFs.
@author:
"""
import fitz
from storage import enumerate_filenames

def extract_text_from_pdf(pdf_path):
    """
    Extracts text from a single PDF file.
    
    Args:
        pdf_path (str): The path to the PDF file.
    
    Returns:
        List[Tuple[str, str]]: A list of tuples containing the PDF name and the extracted text.
    """
    pdf_texts = []
    with fitz.open(pdf_path) as infile:
        for page in infile:
            pdf_texts.append((pdf_path, page.get_text()))
    return pdf_texts

def read_all_pdfs():
    """
    Reads all PDF files from the storage and returns the text from all in a list along with the file names.
    
    Returns:
        List[Tuple[str, str]]: A list of tuples containing the PDF name and the extracted text.
    """
    all_pdfs_texts = []
    for pdf in enumerate_filenames():
        all_pdfs_texts.extend(extract_text_from_pdf(pdf))
    return all_pdfs_texts

def read_pdf(pdf_path):
    """
    Reads a single PDF file from the storage and returns the text in a list along with the file name.
    
    Args:
        pdf_path (str): The path to the PDF file.
    
    Returns:
        List[Tuple[str, str]]: A list of tuples containing the PDF name and the extracted text.
    """
    return extract_text_from_pdf(pdf_path)
