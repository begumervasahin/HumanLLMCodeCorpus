from graph import Graph

def display_graph_info(graph):
    """
    Display the adjacency list and DFS path of the graph.

    Args:
        graph (Graph): The graph instance whose information is to be displayed.
    """
    # Display the adjacency list of the graph
    print("Adjacency List:")
    for vertex, neighbors in graph.adj_list.items():
        print(f"Vertex {vertex}: {neighbors}")

    # Perform Depth-First Search (DFS) traversal
    graph.dfs()
    
    # Display the DFS traversal path
    print("DFS Path:", graph.path)

def main():
    """
    Main function to create a graph, display its information, and perform DFS.
    """
    # Initialize a graph with 10 vertices
    graph = Graph(10)
    
    # Display the graph's information
    display_graph_info(graph)

if __name__ == "__main__":
    main()
