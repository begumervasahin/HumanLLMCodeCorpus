import os
import collections
import pandas as pd
import csv

# Define paths for data files
DATA_DIR = "data"
MBTI_RAW_CSV_PATH = os.path.join(DATA_DIR, "mbti_1.csv")
MBTI_CLEAN_CSV_PATH = os.path.join(DATA_DIR, "mbti_clean.csv")
MBTI_UNCLEAN_CSV_PATH = os.path.join(DATA_DIR, "mbti_unclean.csv")

# Define MBTI type frequencies
MBTI_TO_FREQUENCY_DICT = {
    "ISTJ": 0.11,
    "ISFJ": 0.09,
    "INFJ": 0.04,
    "INTJ": 0.05,
    "ISTP": 0.05,
    "ISFP": 0.05,
    "INFP": 0.06,
    "INTP": 0.06,
    "ESTP": 0.04,
    "ESFP": 0.04,
    "ENFP": 0.08,
    "ENTP": 0.06,
    "ESTJ": 0.08,
    "ESFJ": 0.09,
    "ENFJ": 0.05,
    "ENTJ": 0.05,
}

# Read the raw MBTI data
df = pd.read_csv(MBTI_RAW_CSV_PATH)

# Count occurrences of each MBTI type
mbti_counts = df["type"].value_counts().to_dict()

# Identify the limiting MBTI type based on minimum size ratio
min_size = float("infinity")
limiting_type = None

for mbti, count in mbti_counts.items():
    size_ratio = count / MBTI_TO_FREQUENCY_DICT[mbti]
    if size_ratio < min_size:
        min_size = size_ratio
        limiting_type = mbti

# Group rows by MBTI type
grouped_rows = df.groupby("type").apply(lambda x: x.to_dict(orient="records")).to_dict()

# Write cleaned and uncleaned data to separate CSV files
clean_data = []
unclean_data = []

for mbti, rows in grouped_rows.items():
    sample_size = int(round(min_size * MBTI_TO_FREQUENCY_DICT[mbti]))
    clean_data.extend(rows[:sample_size])
    unclean_data.extend(rows[sample_size:])

# Write clean data to CSV
with open(MBTI_CLEAN_CSV_PATH, "w", newline='') as clean_file:
    writer = csv.DictWriter(clean_file, fieldnames=["type", "posts"])
    writer.writeheader()
    writer.writerows(clean_data)

# Write unclean data to CSV
with open(MBTI_UNCLEAN_CSV_PATH, "w", newline='') as unclean_file:
    writer = csv.DictWriter(unclean_file, fieldnames=["type", "posts"])
    writer.writeheader()
    writer.writerows(unclean_data)
