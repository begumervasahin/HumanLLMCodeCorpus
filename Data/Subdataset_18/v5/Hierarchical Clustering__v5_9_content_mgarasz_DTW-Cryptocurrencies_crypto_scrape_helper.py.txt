import pandas as pd

# Base URL template for fetching cryptocurrency data from Poloniex
URL_TEMPLATE = 'https://poloniex.com/public?command=returnChartData&currencyPair={}&start=1439020800&end=1511202699&period=14400'

# List of cryptocurrencies to monitor against BTC

    'ETH', 'VTC', 'STR', 'XRP', 'DASH', 'LTC', 'ETC', 'XMR', 'ZEC',
    'STRAT', 'XEM', 'VIA', 'SYS', 'LSK', 'FCT', 'CVC', 'BTS', 'DOGE',
    'DGB', 'DCR', 'SC', 'EXP', 'GNO', 'XCP', 'MAID', 'NXC', 'GNT', 
    'STEEM', 'ZRX', 'GAME', 'FLO', 'NAV', 'CLAM', 'LBC', 'HUC', 'OMNI',

    'RIC', 'PASC', 'NMC', 'PPC', 'PINK', 'XPM', 'SBD', 'GRC', 'BELA', 
    'BTM'
]

def fetch_coin_data(coin_symbol):
    """
    Fetch historical data for a specific cryptocurrency from Poloniex.

    Args:
        coin_symbol (str): The symbol of the cryptocurrency (e.g., 'ETH').

    Returns:
        pd.DataFrame: A DataFrame containing the historical price data for the specified cryptocurrency.
    """
    # Construct the currency pair string (e.g., 'BTC_ETH')
    currency_pair = f'BTC_{coin_symbol}'
    
    # Create the complete URL by inserting the currency pair into the template
    url = URL_TEMPLATE.format(currency_pair)
    
    # Fetch the data and load it into a DataFrame
    data = pd.read_json(url)
    
    return data

def main():
    """
    Main function to demonstrate fetching data for a cryptocurrency.
    """
    # Example usage: Fetch data for Ethereum (ETH)
    eth_data = fetch_coin_data('ETH')
    
    # Display the first few rows of the data
    print(eth_data.head())

if __name__ == "__main__":
    main()
