import sys
from time import time
import pickle

def process_pubtator_file(input_file_name):
    # Open the file and read lines
    with open(input_file_name, "r") as file:
        lines = file.readlines()

    # Dictionary to store PMID-disease mappings
    pmid_disease_dict = {}

    # Start time
    start_time = time()

    # Iterate through each line in the file
    for line in lines:
        # Remove leading and trailing whitespaces, split the line into tokens
        tokens = line.strip().split("\t")
        
        # Check if the line has all required fields
        if len(tokens) == 4:
            # Extract PMID, Mesh ID, diseases_mentioned, and disease_source
            pmid, _, dis_mens, _ = tokens
            
            # Split diseases_mentioned into individual diseases   
            diseases = dis_mens.split("|")
            
            # Update the PMID-disease dictionary
            pmid_disease_dict.setdefault(pmid, []).extend(diseases)

    # Print the time taken
    print("Processing done in {:.4f} seconds".format(time() - start_time))
    print()

    # Return the PMID-disease dictionary
    return pmid_disease_dict

def run():
    # Get the input file name from command-line arguments
    input_file_name = sys.argv[1]
    
    # Process the PubTator file
    pmid_disease_dict = process_pubtator_file(input_file_name)
    
    # Define the file name for storing the dictionary
    file_name = './pubtator_dumps/pmid_disease_dict.pickle'
    
    # Serialize and save the dictionary using pickle
    with open(file_name, 'wb') as handle:
        pickle.dump(pmid_disease_dict, handle, protocol=pickle.HIGHEST_PROTOCOL)

if __name__ == '__main__':
    # Call the run function
    run()
