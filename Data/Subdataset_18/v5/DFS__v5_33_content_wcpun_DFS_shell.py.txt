import cmd

class CLI(cmd.Cmd):
    """A simple command line interface."""

    def __init__(self):
        """Initialize the CLI with a custom prompt and current path."""
        super().__init__()
        self.current_path = ''
        self.prompt = "FYP >>> "  # Custom command prompt

    def do_hi(self, arg):
        """Print a greeting message."""
        print("Hello World.")

def main():
    """Main function to start the command line interface."""
    cli = CLI()
    cli.cmdloop()  # Start the command loop

if __name__ == '__main__':
    main()  # Run the main function if this script is executed
