from nltk.tokenize import word_tokenize
import gensim

def load_word2vec_model(model_path):
    """
    Load a pre-trained Word2Vec model from a file.

    Args:
    - model_path (str): Path to the Word2Vec model file.

    Returns:
    - gensim.models.Word2Vec: Loaded Word2Vec model.
    """
    return gensim.models.Word2Vec.load(model_path, mmap='r')

def prepare_inputs_for_scoring(seq1, seq2, model, tail=100, head=10):
    """
    Prepare inputs for scoring similarity between two sequences using a Word2Vec model.

    Args:
    - seq1 (str): First sequence of words.
    - seq2 (str): Second sequence of words.
    - model (gensim.models.Word2Vec): Pre-trained Word2Vec model.
    - tail (int): Number of words to consider from the end of seq1 (default: 100).
    - head (int): Number of words to consider from the beginning of seq2 (default: 10).

    Returns:
    - float: Average maximum similarity score between seq1 and seq2.
    """
    seq1_words = word_tokenize(seq1.strip().lower())[-tail:]
    seq2_words = word_tokenize(seq2.strip().lower())[:head]
    return calculate_similarity_score(seq1_words, seq2_words, model)

def calculate_similarity_score(words1, words2, model):
    """
    Calculate the average maximum similarity score between two lists of words using a Word2Vec model.

    Args:
    - words1 (list of str): List of words from the first sequence.
    - words2 (list of str): List of words from the second sequence.
    - model (gensim.models.Word2Vec): Pre-trained Word2Vec model.

    Returns:
    - float: Average maximum similarity score between words1 and words2.
    """
    max_similarities = []
    for word1 in words1:
        max_similarity = 0
        for word2 in words2:
            if word1 == word2:
                similarity = 1
                max_similarity = similarity
            elif word1 in model.vocab and word2 in model.vocab:
                similarity = model.similarity(word1, word2)
                if similarity > max_similarity:
                    max_similarity = similarity
        if max_similarity != 0:
            max_similarities.append(max_similarity)
    
    if not max_similarities:
        return 0.0
    
    return sum(max_similarities) / len(max_similarities)

# Example usage:
model_path = 'vectors'  # Replace with your Word2Vec model path
word2vec_model = load_word2vec_model(model_path)

sequence1 = "This is a sample sequence for scoring similarity."
sequence2 = "Sample sequence scoring similarity using Word2Vec model."

similarity_score = prepare_inputs_for_scoring(sequence1, sequence2, word2vec_model)
print(f"Similarity Score: {similarity_score}")
