def max_heapify(heap, index):
    """
    Maintains the max-heap property for the heap subtree rooted at the given index.

    Parameters:
        heap (list): List representing the heap.
        index (int): Index of the root of the subtree to be heapified.
    """
    n = len(heap)
    left_child = 2 * index + 1  # Left child index (zero-based indexing)
    right_child = 2 * index + 2  # Right child index (zero-based indexing)
    largest = index

    # Compare left child with root
    if left_child < n and heap[left_child] > heap[largest]:
        largest = left_child

    # Compare right child with the current largest
    if right_child < n and heap[right_child] > heap[largest]:
        largest = right_child

    # If the largest is not the root, swap and continue heapifying
    if largest != index:
        heap[index], heap[largest] = heap[largest], heap[index]  # Swap
        max_heapify(heap, largest)  # Recursively heapify the affected subtree

def build_max_heap(heap):
    """
    Converts an unsorted array into a max-heap.

    Parameters:
        heap (list): List to be converted into a max-heap.
    """
    n = len(heap)
    # Start from the last non-leaf node and heapify each node
    for i in range(n // 2 - 1, -1, -1):
        max_heapify(heap, i)

def heapsort(heap):
    """
    Sorts an array in ascending order using the heapsort algorithm.

    Parameters:
        heap (list): List to be sorted.
    """
    build_max_heap(heap)
    n = len(heap)
    # Extract elements one by one from the heap
    for i in range(n - 1, 0, -1):
        heap[0], heap[i] = heap[i], heap[0]  # Swap the root (max element) with the last element
        max_heapify(heap[:i], 0)  # Heapify the reduced heap

def heap_maximum(heap):
    """
    Returns the maximum element in the heap.

    Parameters:
        heap (list): List representing the heap.

    Returns:
        int: The maximum element of the heap.
    """
    return heap[0]

def heap_insert(heap, key):
    """
    Inserts a new key into the heap.

    Parameters:
        heap (list): List representing the heap.
        key (int): The new element to be inserted.
    """
    heap.append(key)  # Add the new element to the end of the heap
    i = len(heap) - 1
    # Fix the max-heap property if it is violated
    while i > 0 and heap[(i - 1) // 2] < heap[i]:
        heap[i], heap[(i - 1) // 2] = heap[(i - 1) // 2], heap[i]
        i = (i - 1) // 2

def heap_delete(heap, index):
    """
    Deletes an element from the heap at a given index.

    Parameters:
        heap (list): List representing the heap.
        index (int): Index of the element to be deleted.
    """
    heap[index] = heap[-1]  # Replace the element to be deleted with the last element
    heap.pop()  # Remove the last element
    if index < len(heap):
        max_heapify(heap, index)  # Heapify the affected subtree
