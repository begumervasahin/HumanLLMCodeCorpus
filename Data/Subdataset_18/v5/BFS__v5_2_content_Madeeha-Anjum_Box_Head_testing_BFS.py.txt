import collections

def bfs(grid, start):
    """
    Perform Breadth-First Search (BFS) on a grid.

    Args:
        grid (list of list of str): The grid representing the environment.
        start (tuple of int): The starting coordinates (x, y).

    Returns:
        list of tuple of int: The path from start to the goal if found, otherwise None.
    """
    queue = collections.deque([[start]])
    seen = set([start])
    rows, columns = len(grid), len(grid[0])

    while queue:
        path = queue.popleft()
        x, y = path[-1]

        if grid[y][x] == goal:
            return path  # Return the path to the goal

        for x2, y2 in [(x + 1, y), (x - 1, y), (x, y + 1), (x, y - 1)]:
            if 0 <= x2 < columns and 0 <= y2 < rows and grid[y2][x2] != wall and (x2, y2) not in seen:
                queue.append(path + [(x2, y2)])
                seen.add((x2, y2))

    return None  # Return None if no path is found

# Constants representing different types of cells in the grid
wall, clear, goal = "#", ".", "*"

# Define the grid
grid = [
    "..........",
    "...#*..##.",
    "..##...#..",
    ".....###..",
    ".........."
]

# Convert grid from list of strings to list of lists for easier manipulation
grid = [list(row) for row in grid]

# Example of how to use the bfs function
start_position = (0, 0)
path = bfs(grid, start_position)

if path:
    print("Path to goal:", path)
else:
    print("No path found to the goal.")
