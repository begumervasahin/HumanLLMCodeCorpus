class Node:
    def __init__(self, key):
        """
        Initialize a new Node.

        :param key: The key or value of the node.
        """
        self.key = key
        self.left = None
        self.right = None
        self.parent = None
        self.height = 0
        self.real_height = 0
        self.color = 1  # 1 for red, 0 for black (assuming this is for a Red-Black Tree)

    def __repr__(self):
        """
        Return a string representation of the node.
        """
        return (f"Node(key={self.key}, left={self.left}, right={self.right}, "
                f"parent={self.parent}, height={self.height}, real_height={self.real_height}, color={self.color})")
