def selection_sort(arr):
    """
    Perform selection sort on the given list.
    """
    n = len(arr)
    for i in range(n):
        # Assume the current position is the smallest
        smallest_index = i
        # Find the index of the smallest element in the remaining unsorted array
        for j in range(i + 1, n):
            if arr[j] < arr[smallest_index]:
                smallest_index = j
        # Swap the found smallest element with the first element of the unsorted part
        arr[i], arr[smallest_index] = arr[smallest_index], arr[i]
    return arr

def get_user_input():
    """
    Get a list of integers from the user.
    """
    # Prompt user for the number of elements
    num_elements = int(input("Enter how many elements you want: "))
    arr = []

    # Collect the elements from the user
    print("Enter numbers in array: ")
    for _ in range(num_elements):
        num = int(input("num: "))
        arr.append(num)
    
    return arr

def main():
    # Get user input
    arr = get_user_input()

    # Print the original array
    print("ARRAY:", arr)

    # Sort the array using selection sort
    sorted_arr = selection_sort(arr)

    # Print the sorted array
    print("SORTED ARRAY:", sorted_arr)

if __name__ == "__main__":
    main()
