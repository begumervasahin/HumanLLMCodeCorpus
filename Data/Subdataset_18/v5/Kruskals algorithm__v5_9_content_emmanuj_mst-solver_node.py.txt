Description: Encapsulates node data for use in data structures like union-find.
"""

class Node:
    def __init__(self, value):
        """
        Initializes a Node object.

        :param value: The value stored in the node.
        """
        self.value = value       # The value stored in the node
        self.parent = self       # The parent node (initially the node itself)
        self.rank = 0            # The rank of the node (used in union-find algorithms)

    def __repr__(self):
        """
        Provides a string representation of the Node object.

        :return: A formatted string representing the node's value, rank, and parent value.
        """
        return (
            f"Node(value: {self.value}, "
            f"rank: {self.rank}, "
            f"parent: {self.parent.value})"
        )
