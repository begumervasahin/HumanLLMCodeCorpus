def insertion_sort(arr):
    """
    Sorts a list of numbers in ascending order using the insertion sort algorithm.
    
    Args:
        arr (list): The list of numbers to be sorted.
    """
    for i in range(1, len(arr)):
        current_value = arr[i]
        position = i - 1

        # Move elements of arr[0..i-1], that are greater than current_value, to one position ahead of their current position
        while position >= 0 and arr[position] > current_value:
            arr[position + 1] = arr[position]
            position -= 1
        arr[position + 1] = current_value

# Example usage
if __name__ == "__main__":
    arr = [32, 13, 14, 15, 98]
    print("Original array:", arr)
    insertion_sort(arr)
    print("Sorted array is:", arr)
