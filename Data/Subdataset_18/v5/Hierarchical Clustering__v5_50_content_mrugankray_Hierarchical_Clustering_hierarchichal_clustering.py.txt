import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
import scipy.cluster.hierarchy as sch
from sklearn.cluster import AgglomerativeClustering

def load_data(filename):
    """Load the dataset from a CSV file and extract relevant features."""
    dataset = pd.read_csv(filename)
    return dataset.iloc[:, [3, 4]].values

def plot_dendrogram(X):
    """Plot a dendrogram to help determine the optimal number of clusters."""
    plt.figure(figsize=(10, 7))
    plt.title('Dendrogram')
    plt.xlabel('Customers')
    plt.ylabel('Distance Between Clusters / Dissimilarities')
    sch.dendrogram(sch.linkage(X, method='ward'))
    plt.show()

def apply_clustering(X, n_clusters=5):
    """Apply Agglomerative Clustering to the dataset and return cluster labels."""
    hc = AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage='ward')
    return hc.fit_predict(X)

def plot_clusters(X, y_hc, colors, labels):
    """Visualize the resulting clusters."""
    plt.figure(figsize=(10, 7))
    for i, color, label in zip(range(len(labels)), colors, labels):
        plt.scatter(X[y_hc == i, 0], X[y_hc == i, 1], c=color, label=label)
    
    plt.title('Hierarchical Clustering')
    plt.xlabel('Annual Income (k$)')
    plt.ylabel('Spending Score (0-100)')
    plt.legend()
    plt.show()

def main():
    """Main function to execute the clustering analysis."""
    # Load the dataset
    X = load_data('Mall_Customers.csv')
    
    # Plot the dendrogram
    plot_dendrogram(X)
    
    # Apply clustering
    y_hc = apply_clustering(X)
    
    # Define cluster colors and labels
    colors = ['red', 'blue', 'magenta', 'cyan', 'green']
    labels = ['Careful', 'Standard', 'Target', 'Careless', 'Sensible']
    
    # Plot the clusters
    plot_clusters(X, y_hc, colors, labels)

if __name__ == "__main__":
    main()
