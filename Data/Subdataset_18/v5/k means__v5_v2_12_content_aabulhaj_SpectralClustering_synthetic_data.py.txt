import numpy as np
from sklearn.datasets import make_blobs
import matplotlib.pyplot as plt

def create_circles():
    """
    Generates noisy circular data for four concentric circles.
    
    Returns:
        np.ndarray: Array containing the generated circular data points.
    """
    # Define angles for generating circles
    angles = np.arange(0, 2 * np.pi, 0.03)
    num_points = len(angles)
    
    # Helper function to generate a circle with noise
    def generate_circle(radius):
        return np.array([radius * np.cos(angles) + 0.1 * np.random.randn(num_points),
                         radius * np.sin(angles) + 0.1 * np.random.randn(num_points)]).T
    
    # Generate noisy circular data for four circles with different radii
    circle1 = generate_circle(1)
    circle2 = generate_circle(2)
    circle3 = generate_circle(3)
    circle4 = generate_circle(4)
    
    # Concatenate all circles into one dataset
    circles_data = np.concatenate((circle1, circle2, circle3, circle4), axis=0)
    
    return circles_data

def create_blobs():
    """
    Generates blob-shaped data using predefined centers.
    
    Returns:
        np.ndarray: Array containing the generated blob data points.
    """
    # Define centers for blobs
    centers = [(-8, -8), (0, 0), (8, 8), (20, 20), (-20, -20)]
    # Generate blob-shaped data with specified centers
    blobs_data, _ = make_blobs(n_samples=3000, n_features=2, cluster_std=1.0, centers=centers, shuffle=False, random_state=99)
    
    return blobs_data

def plot_data(circles_data, blobs_data):
    """
    Plots the circular and blob-shaped data.
    
    Args:
        circles_data (np.ndarray): Array containing the circular data points.
        blobs_data (np.ndarray): Array containing the blob data points.
    """
    plt.figure(figsize=(16, 8))
    
    # Plot the circular data
    plt.subplot(1, 2, 1)
    plt.title('Circles')
    plt.scatter(circles_data[:, 0], circles_data[:, 1], s=5)
    plt.xlabel('X-axis')
    plt.ylabel('Y-axis')
    
    # Plot the blob-shaped data
    plt.subplot(1, 2, 2)
    plt.title('Blobs')
    plt.scatter(blobs_data[:, 0], blobs_data[:, 1], s=5)
    plt.xlabel('X-axis')
    plt.ylabel('Y-axis')
    
    # Adjust layout and show the plots
    plt.tight_layout()
    plt.show()

# Generate circular and blob-shaped data
circles_data = create_circles()
blobs_data = create_blobs()

# Plot the generated data
plot_data(circles_data, blobs_data)
