import unittest
from SentimentPredictor import SentimentPredictor
from driver_3 import imdb_data_preprocess

# Reporter base class to handle message writing
class Reporter:
    def write(self, msg):
        pass

# A Reporter implementation that does nothing
class NullReporter(Reporter):
    def write(self, msg):
        print()  # No-op

# Function to generate a test corpus
def generate_corpus():
    return []

# Unit tests for the SentimentPredictor class
class TestSentimentPredictor(unittest.TestCase):

    def test_can_create_from_default(self):
        """Test if the SentimentPredictor can be created with default settings."""
        predictor = SentimentPredictor().build()
        self.assertIsNotNone(predictor)

    def test_can_inject_console_reporter(self):
        """Test if the SentimentPredictor can accept a custom reporter."""
        predictor = SentimentPredictor().with_reporter(NullReporter()).build()
        self.assertIsNotNone(predictor)

    def test_can_run_on_default_test_data(self):
        """Test if the SentimentPredictor can run on the default test data."""
        predictor = SentimentPredictor().with_reporter(NullReporter()).build()
        corpus = generate_corpus()
        predictor.fit()
        results = [predictor.predict(text) for text in corpus]
        self.assertIsNotNone(results)

# Unit tests for the imdb_data_preprocess function
class TestImdbDataPreprocess(unittest.TestCase):

    def test_run(self):
        """Test running the imdb_data_preprocess function."""
        imdb_data_preprocess('./')

    def test_run_mixed(self):
        """Test running the imdb_data_preprocess function with the mix option."""
        imdb_data_preprocess('./', mix=True)

if __name__ == '__main__':
    unittest.main()
