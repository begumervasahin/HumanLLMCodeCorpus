from __future__ import print_function

# Public parameters for the key exchange
prime = 23
base = 5

# Private secrets for Alice and Bob
alice_secret = 6
bob_secret = 15

# Display public parameters
print("Publicly Shared Parameters:")
print(f"    Prime: {prime}")
print(f"    Base:  {base}")

# Compute and display Alice's public value
alice_public_value = pow(base, alice_secret, prime)
print(f"\n  Alice's Public Value (sent to Bob): {alice_public_value}")

# Compute and display Bob's public value
bob_public_value = pow(base, bob_secret, prime)
print(f"  Bob's Public Value (sent to Alice): {bob_public_value}")

print("\n------------\n")

# Calculate and display shared secrets privately
print("Privately Calculated Shared Secrets:")

# Alice calculates the shared secret using Bob's public value
alice_shared_secret = pow(bob_public_value, alice_secret, prime)
print(f"    Alice's Shared Secret: {alice_shared_secret}")

# Bob calculates the shared secret using Alice's public value
bob_shared_secret = pow(alice_public_value, bob_secret, prime)
print(f"    Bob's Shared Secret: {bob_shared_secret}")
