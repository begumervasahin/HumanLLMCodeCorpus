def sieve_of_eratosthenes(n):
    """Generate all prime numbers up to n using the Sieve of Eratosthenes."""
    primes = [True] * (n + 1)
    p = 2
    
    while p * p <= n:
        if primes[p]:
            for multiple in range(p * p, n + 1, p):
                primes[multiple] = False
        p += 1
    
    prime_numbers = [num for num in range(2, n + 1) if primes[num]]
    return prime_numbers

def main():
    print("Welcome! This program will print all the prime numbers from 2 to n.")
    try:
        n = int(input("Enter n: "))
        if n < 2:
            raise ValueError("Please enter a number greater than or equal to 2.")
        
        prime_numbers = sieve_of_eratosthenes(n)
        print(f"Prime numbers up to {n} are:")
        for prime in prime_numbers:
            print(prime)
    except ValueError as e:
        print(f"Invalid input: {e}")

if __name__ == "__main__":
    main()
