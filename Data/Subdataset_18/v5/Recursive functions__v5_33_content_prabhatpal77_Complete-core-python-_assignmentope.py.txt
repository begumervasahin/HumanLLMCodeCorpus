def main():
    """
    Main function to demonstrate basic arithmetic and bitwise operations on variables.
    """
    # Initializing variables
    x = 1
    y = 2
    z = 3

    # Performing arithmetic and bitwise operations
    x += 5  # Increment x by 5
    y *= 10  # Multiply y by 10
    z >>= 6  # Right shift z by 6 bits

    # Printing the results
    print(f"x: {x}")
    print(f"y: {y}")
    print(f"z: {z}")

if __name__ == "__main__":
    main()
