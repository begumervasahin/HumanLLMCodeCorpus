import numpy as np
from numpy.linalg import eig

def compute_Z(X, centering=True, scaling=True):
    """
    Standardizes the input matrix X by centering and scaling.
    
    Parameters:
    X (numpy.ndarray): The input matrix.
    centering (bool): If True, center the data by subtracting the mean.
    scaling (bool): If True, scale the data to unit variance.
    
    Returns:
    numpy.ndarray: The standardized matrix Z.
    """
    Z = X.astype(float)

    if centering:
        means = Z.mean(axis=0)
        Z -= means

    if scaling:
        std_devs = Z.std(axis=0, ddof=1)
        Z /= std_devs

    return Z

def compute_covariance_matrix(Z):
    """
    Computes the covariance matrix of the standardized matrix Z.
    
    Parameters:
    Z (numpy.ndarray): The standardized matrix.
    
    Returns:
    numpy.ndarray: The covariance matrix.
    """
    return np.cov(Z, rowvar=False)

def find_pcs(COV):
    """
    Finds the principal components of the covariance matrix.
    
    Parameters:
    COV (numpy.ndarray): The covariance matrix.
    
    Returns:
    tuple: Eigenvalues and eigenvectors sorted by descending eigenvalues.
    """
    eigenvalues, eigenvectors = eig(COV)
    sorted_indices = eigenvalues.argsort()[::-1]
    eigenvalues = eigenvalues[sorted_indices]
    eigenvectors = eigenvectors[:, sorted_indices]
    return eigenvalues, eigenvectors

def project_data(Z, PCS, L, k=0, var=0.0):
    """
    Projects the data onto the principal components.
    
    Parameters:
    Z (numpy.ndarray): The standardized data matrix.
    PCS (numpy.ndarray): The principal components (eigenvectors).
    L (numpy.ndarray): The eigenvalues.
    k (int): The number of principal components to keep. If 0, use variance to determine components.
    var (float): The amount of variance to cover if k is 0.
    
    Returns:
    numpy.ndarray: The projected data matrix.
    """
    if k > 0:
        component_matrix = PCS[:, :k]
    else:
        total_variance = 0
        num_components = 0
        for eigenvalue in L:
            total_variance += eigenvalue / L.sum()
            num_components += 1
            if total_variance >= var:
                break
        component_matrix = PCS[:, :num_components]

    Z_star = Z.dot(component_matrix)
    return Z_star
