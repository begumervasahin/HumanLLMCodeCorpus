import nltk
nltk.download('punkt')
from textblob import TextBlob as tb
import json

class TfIdf:
    def __init__(self, corpus_paths):
        self.corpus_paths = corpus_paths
        self.blob_list = []
        self.word_df_dict = {}

    def setup(self):
        self._load_corpus()
        self._calculate_word_frequencies()

    def _load_corpus(self):
        for corpus_path in self.corpus_paths:
            with open(corpus_path, 'r') as file:
                documents = json.load(file)
                self._build_blob_list(documents)

    def _build_blob_list(self, documents):
        for document in documents:
            content = '. '.join(document['content']).replace('..', '.')
            self.blob_list.append(tb(content))

    def _calculate_word_frequencies(self):
        for blob in self.blob_list:
            unique_words = set(blob.words)
            for word in unique_words:
                if word not in self.word_df_dict:
                    self.word_df_dict[word] = 0
                self.word_df_dict[word] += 1

    def write_to_file(self, output_filename="stop-words.txt"):
        sorted_words = sorted(self.word_df_dict.items(), key=lambda x: x[1], reverse=True)[:102]
        with open(output_filename, 'w') as out_file:
            for word, _ in sorted_words:
                out_file.write(f"{word}\n")
