def bubble_sort(arr):
    """
    Sorts an array using the bubble sort algorithm.
    
    Parameters:
    arr (list): The list of elements to be sorted.
    
    Returns:
    list: The sorted list.
    """
    n = len(arr)
    for i in range(n):
        for j in range(0, n - i - 1):
            if arr[j] > arr[j + 1]:
                # Swap the elements
                arr[j], arr[j + 1] = arr[j + 1], arr[j]
    return arr

def merge_sort(arr):
    """
    Sorts an array using the merge sort algorithm.
    
    Parameters:
    arr (list): The list of elements to be sorted.
    """
    if len(arr) > 1:
        mid = len(arr) // 2
        left = arr[:mid]
        right = arr[mid:]

        merge_sort(left)
        merge_sort(right)

        i = j = k = 0

        print("Left:", left, "Right:", right)

        # Merge the sorted halves
        while i < len(left) and j < len(right):
            if left[i] < right[j]:
                arr[k] = left[i]
                i += 1
            else:
                arr[k] = right[j]
                j += 1
            k += 1

        # Copy the remaining elements of left[], if any
        while i < len(left):
            arr[k] = left[i]
            i += 1
            k += 1

        # Copy the remaining elements of right[], if any
        while j < len(right):
            arr[k] = right[j]
            j += 1
            k += 1

        print("Merged:", arr)

def main():
    """
    Main function to demonstrate bubble sort and merge sort.
    """
    array = [34, 7, 23, 32, 5, 62]
    print("Original array:", array)
    
    sorted_array = bubble_sort(array.copy())
    print("Bubble sorted array:", sorted_array)
    
    merge_sort(array)
    print("Merge sorted array:", array)

if __name__ == "__main__":
    main()
