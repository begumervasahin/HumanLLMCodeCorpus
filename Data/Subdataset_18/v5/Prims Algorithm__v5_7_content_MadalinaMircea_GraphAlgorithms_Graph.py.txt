class Graph:
    def __init__(self, num_vertices):
        """
        Initializes a graph with num_vertices vertices (numbered from 0 to num_vertices-1) and no edges.
        
        :param num_vertices: Number of vertices in the graph
        """
        self.num_vertices = num_vertices
        self.num_edges = 0
        self.adjacency_list = {i: [] for i in range(num_vertices)}

    def is_vertex(self, vertex):
        """
        Checks if a vertex is in the graph.
        
        :param vertex: Vertex to check
        :return: True if the vertex is in the graph, False otherwise
        """
        return vertex in self.adjacency_list

    def get_all_vertices(self):
        """
        Returns a list of all vertices in the graph.
        
        :return: List of vertices
        """
        return list(self.adjacency_list.keys())

    def get_neighbours(self, vertex):
        """
        Returns a list of neighbors of a given vertex.
        
        :param vertex: Vertex whose neighbors are to be returned
        :return: List of neighbors if the vertex is in the graph, otherwise False
        """
        if self.is_vertex(vertex):
            return self.adjacency_list[vertex]
        return False

    def is_edge(self, vertex1, vertex2):
        """
        Checks if there is an edge between two vertices.
        
        :param vertex1: Starting vertex
        :param vertex2: Ending vertex
        :return: True if there is an edge between vertex1 and vertex2, False otherwise
        """
        if self.is_vertex(vertex1):
            return vertex2 in self.adjacency_list[vertex1]
        return False

    def add_edge(self, vertex1, vertex2):
        """
        Adds an edge between two vertices.
        
        :param vertex1: Starting vertex
        :param vertex2: Ending vertex
        :return: True if the edge was added, False otherwise
        """
        if not self.is_edge(vertex1, vertex2):
            self.adjacency_list[vertex1].append(vertex2)
            self.adjacency_list[vertex2].append(vertex1)
            self.num_edges += 1
            return True
        return False

    def get_num_vertices(self):
        """
        Returns the number of vertices in the graph.
        
        :return: Number of vertices
        """
        return self.num_vertices

    def get_num_edges(self):
        """
        Returns the number of edges in the graph.
        
        :return: Number of edges
        """
        return self.num_edges

    def get_degree(self, vertex):
        """
        Returns the degree of a given vertex.
        
        :param vertex: Vertex whose degree is to be returned
        :return: Degree of the vertex if it exists, otherwise False
        """
        if self.is_vertex(vertex):
            return len(self.adjacency_list[vertex])
        return False
