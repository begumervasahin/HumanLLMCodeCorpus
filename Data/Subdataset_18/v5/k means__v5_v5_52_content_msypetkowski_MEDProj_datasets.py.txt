import pandas as pd

def load_iris():
    """
    Loads the Iris dataset into a pandas DataFrame.

    Returns:
        pd.DataFrame: The Iris dataset.
    """
    iris_url = 'https://raw.githubusercontent.com/mwaskom/seaborn-data/master/iris.csv'
    iris_data = pd.read_csv(iris_url)
    return iris_data


def load_adult():
    """
    Loads the Adult dataset into a pandas DataFrame.

    Returns:
        pd.DataFrame: The Adult dataset.
    """
    adult_url = 'https://archive.ics.uci.edu/ml/machine-learning-databases/adult/adult.data'
    adult_columns = [
        "Age", "Workclass", "fnlwgt", "Education", "Education-Num", "Marital Status",
        "Occupation", "Relationship", "Race", "Sex", "Capital Gain", "Capital Loss",
        "Hours per week", "Country", "Target"
    ]
    adult_data = pd.read_csv(
        adult_url,
        names=adult_columns,
        sep=r'\s*,\s*',
        engine='python',
        na_values="?"
    )
    return adult_data


def load_ctg():
    """
    Loads the Cardiotocography dataset into a pandas DataFrame.

    Returns:
        pd.DataFrame: The Cardiotocography dataset.
    """
    ctg_url = 'http://staff.ii.pw.edu.pl/~gprotazi/dydaktyka/dane/cardioto_all_corr.csv'
    ctg_data = pd.read_csv(ctg_url)
    ctg_data_cleaned = ctg_data.drop('Unnamed: 0', axis=1)
    return ctg_data_cleaned


def load_cars():
    """
    Loads the Car Evaluation dataset into a pandas DataFrame.

    Returns:
        pd.DataFrame: The Car Evaluation dataset.
    """
    cars_url = 'https://archive.ics.uci.edu/ml/machine-learning-databases/car/car.data'
    cars_columns = ["class", "buying", "maint", "doors", "persons", "lug_boot", "safety"]
    cars_data = pd.read_csv(cars_url, names=cars_columns)
    return cars_data
