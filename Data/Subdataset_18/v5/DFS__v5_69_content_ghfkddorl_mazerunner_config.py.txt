# Configuration variables
DEBUG_MODE = False

COLOR_BLOCK = 'gray'
COLOR_BLOCK_OUTLINE = 'black'
COLOR_EMPTY = 'white'
COLOR_GRAPH = 'green'
COLOR_PATH = 'orange'  # Corrected the typo from 'organe' to 'orange'

def get_configurations():
    """
    Retrieve the configuration settings.

    Returns:
        dict: A dictionary of configuration settings.
    """
    return {
        "Debug Mode": DEBUG_MODE,
        "Color Block": COLOR_BLOCK,
        "Color Block Outline": COLOR_BLOCK_OUTLINE,
        "Color Empty": COLOR_EMPTY,
        "Color Graph": COLOR_GRAPH,
        "Color Path": COLOR_PATH,
    }

def display_config(config):
    """
    Display the configuration settings.

    Args:
        config (dict): A dictionary of configuration settings.
    """
    for key, value in config.items():
        print(f"{key}: {value}")

def main():
    """
    Main function to run the script.
    """
    config = get_configurations()
    display_config(config)

if __name__ == "__main__":
    main()
