import random

def hcfnaive(a, b):
    """Calculate the greatest common divisor using Euclidean algorithm."""
    return a if b == 0 else hcfnaive(b, a % b)

def egcd(a, b):
    """Extended Euclidean Algorithm to find the multiplicative inverse."""
    if a == 0:
        return b, 0, 1
    else:
        g, x, y = egcd(b % a, a)
        return g, y - (b // a) * x, x

def is_prime(n, k=1):
    """Test if a number is prime using Miller-Rabin primality test."""
    if n == 2:
        return True
    if n % 2 == 0 or n < 2:
        return False

    r, s = 0, n - 1
    while s % 2 == 0:
        r += 1
        s //= 2

    for _ in range(k):
        a = random.randrange(2, n - 1)
        x = pow(a, s, n)
        if x == 1 or x == n - 1:
            continue

        for _ in range(r - 1):
            x = pow(x, 2, n)
            if x == n - 1:
                break
        else:
            return False

    return True

def gcd(a, b):
    """Compute the greatest common divisor of a and b."""
    while b != 0:
        a, b = b, a % b
    return a

def multiplicative_inverse(e, phi):
    """Find the multiplicative inverse of e modulo phi."""
    g, x, y = egcd(e, phi)
    if g != 1:
        raise Exception('Multiplicative inverse does not exist')
    else:
        return x % phi

def generate_prime(leng):
    """Generate a prime number with a specified bit length."""
    prime = rand_bin(leng)
    prime = two_com(prime, leng)
    while not is_prime(binary2dec(prime)):
        prime = rand_bin(leng)
        prime = two_com(prime, leng)
    return binary2dec(prime)

def generate_keys(mlen, rlen, eleng):
    """Generate RSA keys."""
    m = generate_prime(mlen)
    r = generate_prime(rlen)
    phi = (m - 1) * (r - 1)
    e = generate_prime(eleng)
    while gcd(e, phi) != 1:
        e = generate_prime(eleng)
    d = multiplicative_inverse(e, phi)
    return phi, e, d

def main():
    """Main function to generate and display RSA keys."""
    mlen = 512
    rlen = 512
    eleng = 1024

    phi, e, d = generate_keys(mlen, rlen, eleng)

    print(f'n (phi) = {phi}')
    print(f'Public Key (e) = {e}')
    print(f'Private Key (d) = {d}')

if __name__ == "__main__":
    main()
