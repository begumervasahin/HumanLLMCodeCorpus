from sklearn.metrics import confusion_matrix, accuracy_score, average_precision_score, roc_auc_score
from sklearn.naive_bayes import MultinomialNB

def MultinomialNBAlgo(x_train_vft, y_train, x_test_vft, y_test, vec):
    """
    Trains and evaluates a Multinomial Naive Bayes model.

    Parameters:
    x_train_vft : array-like, shape (n_samples, n_features)
        Training vectors, where n_samples is the number of samples and
        n_features is the number of features.

    y_train : array-like, shape (n_samples,)
        Target values for the training set.

    x_test_vft : array-like, shape (n_samples, n_features)
        Test vectors, where n_samples is the number of samples and
        n_features is the number of features.

    y_test : array-like, shape (n_samples,)
        True labels for the test set.

    vec : array-like, shape (1, n_features)
        A single vector to predict the class label.

    Returns:
    str : "Positive" or "Negative"
        The prediction for the provided vector `vec`.
    """
    print("Multinomial Naive Bayes")

    # Initialize the Multinomial Naive Bayes model
    mnb = MultinomialNB()

    # Fit the model to the training data
    mnb.fit(x_train_vft, y_train)

    # Predict the class labels for the test set
    y_predict_class = mnb.predict(x_test_vft)

    # Print evaluation metrics
    print("Confusion Matrix:")
    print(confusion_matrix(y_test, y_predict_class))
    print('Accuracy Score:', accuracy_score(y_test, y_predict_class))
    print('ROC AUC Score:', roc_auc_score(y_test, y_predict_class))
    print('Average Precision Score:', average_precision_score(y_test, y_predict_class))

    # Predict the class for the provided vector
    prediction = mnb.predict(vec)

    return "Positive" if prediction == [1] else "Negative"
