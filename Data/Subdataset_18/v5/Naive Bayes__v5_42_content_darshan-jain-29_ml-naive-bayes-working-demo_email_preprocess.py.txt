import pickle
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.feature_selection import SelectPercentile, f_classif

def preprocess(words_file="../tools/word_data.pkl", authors_file="../tools/email_authors.pkl"):
    """
    This function takes a pre-made list of email texts (default: word_data.pkl)
    and the corresponding authors (default: email_authors.pkl) and performs
    several preprocessing steps:
        -- splits into training/testing sets (10% testing)
        -- vectorizes into TF-IDF matrix
        -- selects/keeps the most helpful features
    After this, the features and labels are converted into numpy arrays, 
    which work well with sklearn functions.
    
    Returns:
        -- features_train_transformed (np.ndarray): Training features
        -- features_test_transformed (np.ndarray): Testing features
        -- labels_train (np.ndarray): Training labels
        -- labels_test (np.ndarray): Testing labels
    """
    # Load the authors and word data
    with open(authors_file, "rb") as af:
        authors = pickle.load(af)
    
    with open(words_file, "rb") as wf:
        word_data = pickle.load(wf)

    # Split data into training and testing sets
    features_train, features_test, labels_train, labels_test = train_test_split(
        word_data, authors, test_size=0.1, random_state=42
    )

    # Vectorize the data into a TF-IDF matrix
    vectorizer = TfidfVectorizer(sublinear_tf=True, max_df=0.5, stop_words='english')
    features_train_transformed = vectorizer.fit_transform(features_train)
    features_test_transformed = vectorizer.transform(features_test)

    # Select the most helpful features
    selector = SelectPercentile(f_classif, percentile=10)
    selector.fit(features_train_transformed, labels_train)
    features_train_transformed = selector.transform(features_train_transformed).toarray()
    features_test_transformed = selector.transform(features_test_transformed).toarray()

    # Print some statistics about the training data
    print(f"No. of Chris training emails: {sum(labels_train)}")
    print(f"No. of Sara training emails: {len(labels_train) - sum(labels_train)}")

    return features_train_transformed, features_test_transformed, labels_train, labels_test
