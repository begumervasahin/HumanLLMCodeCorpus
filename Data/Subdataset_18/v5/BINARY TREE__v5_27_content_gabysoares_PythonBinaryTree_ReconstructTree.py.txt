class ListBinaryTree:
    """A binary tree class with nodes as lists."""
    
    DATA = 0
    LEFT = 1
    RIGHT = 2
    
    def __init__(self, root_value, left=None, right=None):
        """Create a binary tree with a given root value and optional left and right subtrees."""
        self.node = [root_value, left, right]
    
    def create_tree(self, a_list):
        """Create a binary tree from a list."""
        return ListBinaryTree(a_list[self.DATA], a_list[self.LEFT], a_list[self.RIGHT])
    
    def insert_value_left(self, value):
        """Insert value to the left of this node, pushing any existing left subtree down as the left child of the new node."""
        self.node[self.LEFT] = ListBinaryTree(value, self.node[self.LEFT], None)
    
    def insert_value_right(self, value):
        """Insert value to the right of this node, pushing any existing right subtree down as the right child of the new node."""
        self.node[self.RIGHT] = ListBinaryTree(value, None, self.node[self.RIGHT])
    
    def insert_tree_left(self, tree):
        """Insert a new left subtree of the current node."""
        self.node[self.LEFT] = tree
    
    def insert_tree_right(self, tree):
        """Insert a new right subtree of the current node."""
        self.node[self.RIGHT] = tree
    
    def set_value(self, new_value):
        """Set the value of the node."""
        self.node[self.DATA] = new_value
    
    def get_value(self):
        """Get the value of the node."""
        return self.node[self.DATA]
    
    def get_left_subtree(self):
        """Get the left subtree of the node."""
        return self.node[self.LEFT]
    
    def get_right_subtree(self):
        """Get the right subtree of the node."""
        return self.node[self.RIGHT]
    
    def __str__(self):
        return f'[{str(self.node[self.DATA])}, {str(self.node[self.LEFT])}, {str(self.node[self.RIGHT])}]'

def construct_tree(inorder, preorder, start, end):
    """Constructs a binary tree from inorder and preorder traversal sequences."""
    if start > end:
        return None
    
    # Create a new tree node with the current preorder element
    tree = ListBinaryTree(preorder[construct_tree.preIndex])
    construct_tree.preIndex += 1
    
    # If this node has no children, return it
    if start == end:
        return tree
    
    # Find the index of this node in inorder traversal
    inorder_node_index = inorder.index(tree.get_value())
    
    # Recursively construct the left and right subtrees
    tree.insert_tree_left(construct_tree(inorder, preorder, start, inorder_node_index - 1))
    tree.insert_tree_right(construct_tree(inorder, preorder, inorder_node_index + 1, end))
    
    return tree

# Initialize the static variable for preorder index
construct_tree.preIndex = 0

def main():
    print("Binary Tree reconstructed by gsoa420:")
    
    # Get inorder and preorder sequences from user input
    inorder = input("Please enter the inorder sequence: ").split()
    preorder = input("Please enter the preorder sequence: ").split()
    
    # Construct the tree
    tree = construct_tree(inorder, preorder, 0, len(preorder) - 1)
    
    # Print the reconstructed tree
    print(tree)

    main()
