import numpy as np
import matplotlib.pyplot as plt

class LeastSquares:
    def __init__(self, datafile):
        self.x, self.y, self.sigma = np.loadtxt(datafile, unpack=True)
    
    def expectation(self, nsteps, min_val, max_val):
        a = np.linspace(min_val, max_val, num=nsteps)
        b = np.linspace(min_val, max_val, num=nsteps)
        da = (max_val - min_val) / nsteps
        db = (max_val - min_val) / nsteps
        
        self.expa = 0
        self.expb = 0
        self.prob = np.zeros((nsteps, nsteps))
        
        for i, ai in enumerate(a):
            for j, bj in enumerate(b):
                probfxn = 1
                for xi, yi, si in zip(self.x, self.y, self.sigma):
                    probfxn *= (
                        (1 / (np.sqrt(2 * np.pi) * si)) *
                        np.exp(-((yi - (ai * xi + bj)) ** 2) / (2 * si ** 2))
                    )
                self.prob[i, j] = probfxn
        
        self.vol = np.sum(self.prob) * da * db
        
        self.expa = np.sum(a[:, None] * self.prob * da * db / self.vol)
        self.expb = np.sum(b[None, :] * self.prob * da * db / self.vol)
        
        siga_sq = np.sum((a[:, None] - self.expa) ** 2 * self.prob * da * db / self.vol)
        sigb_sq = np.sum((b[None, :] - self.expb) ** 2 * self.prob * da * db / self.vol)
        
        self.siga = np.sqrt(siga_sq)
        self.sigb = np.sqrt(sigb_sq)
        
        covab = np.sum((a[:, None] - self.expa) * (b[None, :] - self.expb) * self.prob * da * db / self.vol)
        self.covab = covab
        self.pco = self.covab / (self.siga * self.sigb)
    
    def plot_model(self):
        plt.plot(self.x, self.y, 'o', label='Data')
        xvar = np.linspace(0, 10, 100)
        model = self.expa * xvar + self.expb
        plt.plot(xvar, model, label='Model')
        plt.xlabel('X')
        plt.ylabel('Y')
        plt.title('Least Squares Model Fit')
        plt.legend()
        plt.show()
