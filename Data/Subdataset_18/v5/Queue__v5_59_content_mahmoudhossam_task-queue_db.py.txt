import os
from sqlalchemy import create_engine, Column, String, Integer
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker
from sqlalchemy_utils import database_exists, create_database

# Define the base class for declarative models
Base = declarative_base()

class DB:
    """
    A class to manage database connection and operations.
    """

    def __init__(self):
        """
        Connects to the database and sets up the session.
        """
        self.engine = create_engine(os.getenv("DATABASE_URL"))
        Session = sessionmaker(bind=self.engine)
        self.session = Session()

    def add_user(self, user):
        """
        Performs an upsert for the given user. If a user with the same email exists,
        updates the user's name. Otherwise, inserts a new user.

        Parameters:
        user (User): The user object to be added or updated.
        """
        existing_user = self.session.query(User).filter(User.email == user.email).first()
        if existing_user:
            existing_user.name = user.name
            self.session.add(existing_user)
        else:
            self.session.add(user)
        self.session.commit()

class User(Base):
    """
    A class representing the user table.
    """
    __tablename__ = "users"
    
    id = Column(Integer, primary_key=True)
    name = Column(String(50))
    email = Column(String(50), unique=True)

def initialize_database():
    """
    Creates the database and the tables if they do not exist.
    """
    db = DB()
    if not database_exists(db.engine.url):
        create_database(db.engine.url)
    Base.metadata.create_all(db.engine)

if __name__ == "__main__":
    initialize_database()
