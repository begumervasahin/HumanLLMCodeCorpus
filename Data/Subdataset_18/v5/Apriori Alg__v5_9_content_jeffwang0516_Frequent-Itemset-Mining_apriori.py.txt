import sys
import time
from association_rules.frequent_itemset_mining import Apriori

def parse_arguments():
    """Parse and validate command line arguments."""
    if len(sys.argv) != 4:
        print("Usage: python apriori.py <input_file> <min_support_ratio> <output_file>")
        sys.exit(1)

    input_file = sys.argv[1]
    try:
        min_support_ratio = float(sys.argv[2])
    except ValueError:
        print("Error: min_support_ratio must be a float.")
        sys.exit(1)
    
    output_file = sys.argv[3]
    return input_file, min_support_ratio, output_file

def run_apriori(input_file, min_support_ratio, output_file):
    """Run the Apriori algorithm and save the results."""
    apriori = Apriori()

    # Preprocess data
    apriori.process_input_data(input_file)

    print(f"[Apriori] Finding frequent itemsets with minimum support > {min_support_ratio} --")
    start_time = time.time()

    apriori.find_supersets_k(min_support_ratio)
    apriori.save_output(output_file)

    end_time = time.time()
    elapsed_time = end_time - start_time

    print(f"Elapsed time = {elapsed_time:.2f} seconds")
    print('----------\n')

def main():
    input_file, min_support_ratio, output_file = parse_arguments()
    run_apriori(input_file, min_support_ratio, output_file)

if __name__ == '__main__':
    main()
