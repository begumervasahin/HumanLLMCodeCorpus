import matplotlib.pyplot as plt
import networkx as nx
import sys

def display_graph(mst_list, vert_list):
    G = nx.Graph()
    
    # Add edges to the graph
    for edge in mst_list:
        G.add_edge(vert_list[edge[0]], vert_list[edge[1]], weight=edge[2])
    
    pos = nx.spring_layout(G, k=20, iterations=150)
    weights = nx.get_edge_attributes(G, 'weight')
    
    # Draw the graph
    nx.draw(G, pos, with_labels=True, node_size=700, node_color='lightblue', font_size=10)
    nx.draw_networkx_edge_labels(G, pos, edge_labels=weights)
    plt.show()

def sort_edges(wc_graph):
    # Sort edges by their weight
    return sorted(wc_graph, key=lambda x: x[2])

def find(parent, node):
    if parent[node] != node:
        parent[node] = find(parent, parent[node])
    return parent[node]

def union(parent, rank, root1, root2):
    if rank[root1] > rank[root2]:
        parent[root2] = root1
    elif rank[root1] < rank[root2]:
        parent[root1] = root2
    else:
        parent[root2] = root1
        rank[root1] += 1

def kruskal(wc_graph, vert_count):
    mst_list = []
    sorted_graph = sort_edges(wc_graph)
    parent = list(range(vert_count))
    rank = [0] * vert_count
    edge_count = 0
    index = 0
    
    while edge_count < vert_count - 1:
        v, u, weight = sorted_graph[index]
        index += 1
        
        root_v = find(parent, v)
        root_u = find(parent, u)
        
        if root_v != root_u:
            edge_count += 1
            mst_list.append([v, u, weight])
            union(parent, rank, root_v, root_u)
    
    return mst_list

def main():
    if len(sys.argv) != 2:
        print("Usage: python3 prim.py city-pairs.txt")
        return
    
    file_name = sys.argv[1]
    vert_set = set()
    
    # Read the vertices from the file
    with open(file_name) as file:
        for line in file:
            v1, v2, _ = line.strip().split()
            vert_set.add(v1)
            vert_set.add(v2)
    
    vert_list = list(vert_set)
    vert_count = len(vert_list)
    wc_graph = []
    
    # Read the weighted edges from the file
    with open(file_name) as file:
        for line in file:
            v1, v2, weight = line.strip().split()
            wc_graph.append([vert_list.index(v1), vert_list.index(v2), int(weight)])
    
    # Find the Minimum Spanning Tree using Kruskal's algorithm
    mst_list = kruskal(wc_graph, vert_count)
    
    # Print the MST and its total weight
    total_weight = sum(edge[2] for edge in mst_list)
    print("The minimum spanning tree is as follows:")
    for edge in mst_list:
        print(f"{vert_list[edge[0]]} to {vert_list[edge[1]]} = {edge[2]} miles")
    print(f"Total weight: {total_weight} miles.")
    
    # Display the graph
    display_graph(mst_list, vert_list)

if __name__ == "__main__":
    main()
