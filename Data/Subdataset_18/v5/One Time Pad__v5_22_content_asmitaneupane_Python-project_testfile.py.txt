import sys
from PyQt5.QtWidgets import QApplication, QMainWindow, QPushButton, QDialog, QVBoxLayout, QStatusBar
from PyQt5.QtCore import pyqtSlot

class App(QMainWindow):
    """
    Main application window.
    """
    def __init__(self):
        super().__init__()
        self.title = 'PyQt5 Status Bar Example'
        self.left = 10
        self.top = 10
        self.width = 640
        self.height = 480
        self.init_ui()

    def init_ui(self):
        """
        Initializes the user interface.
        """
        self.setWindowTitle(self.title)
        self.setGeometry(self.left, self.top, self.width, self.height)
        self.statusBar().showMessage('Message in status bar.')
        
        open_dialog_button = QPushButton('Open Dialog', self)
        open_dialog_button.setToolTip('This button opens a dialog')
        open_dialog_button.move(100, 70)
        open_dialog_button.clicked.connect(self.open_dialog)
        
        self.show()

    def open_dialog(self):
        """
        Opens the dialog window.
        """
        dialog = Dialog(self)
        dialog.exec_()

class Dialog(QDialog):
    """
    Dialog window.
    """
    def __init__(self, parent=None):
        super(Dialog, self).__init__(parent)
        self.init_ui()

    def init_ui(self):
        """
        Initializes the user interface of the dialog.
        """
        self.setWindowTitle("Dialog")
        self.setGeometry(100, 100, 400, 300)
        
        layout = QVBoxLayout()
        
        self.status_bar = QStatusBar()
        self.status_bar.showMessage('Message in status bar.')
        layout.addWidget(self.status_bar)
        
        example_button = QPushButton('PyQt5 Button', self)
        example_button.setToolTip('This is an example button')
        example_button.clicked.connect(self.slot_method)
        layout.addWidget(example_button)
        
        self.setLayout(layout)
        
    @pyqtSlot()
    def slot_method(self):
        """
        Slot method called when the button is clicked.
        """
        print('Slot method is called.')

if __name__ == '__main__':
    app = QApplication(sys.argv)
    main_app = App()
    sys.exit(app.exec_())
