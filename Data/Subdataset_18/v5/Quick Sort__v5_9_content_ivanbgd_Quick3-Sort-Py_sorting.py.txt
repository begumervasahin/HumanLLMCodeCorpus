import random

def partition3(arr, left, right):
    """
    Partition the array around the pivot element using 3-way partitioning.

    Parameters:
    arr (list): The list to be partitioned.
    left (int): The starting index of the segment to partition.
    right (int): The ending index of the segment to partition.

    Returns:
    tuple: Indices where the pivot elements are placed.
    """
    pivot = arr[left]
    j, o = left, left

    for i in range(left + 1, right + 1):
        if arr[i] < pivot:
            o += 1
            arr[i], arr[o] = arr[o], arr[i]
            arr[j], arr[o] = arr[o], arr[j]
            j += 1
        elif arr[i] == pivot:
            o += 1
            arr[i], arr[o] = arr[o], arr[i]

    arr[left], arr[j] = arr[j], arr[left]
    
    return j, o

def randomized_quick_sort3(arr, left, right):
    """
    Perform 3-way randomized QuickSort on the given list.

    Parameters:
    arr (list): The list to be sorted.
    left (int): The starting index of the segment to sort.
    right (int): The ending index of the segment to sort.
    """
    if left >= right:
        return
    
    k = random.randint(left, right)
    arr[left], arr[k] = arr[k], arr[left]
    m1, m2 = partition3(arr, left, right)
    randomized_quick_sort3(arr, left, m1 - 1)
    randomized_quick_sort3(arr, m2 + 1, right)

def parse_input(input_data):
    """
    Parse the input data to extract the list of integers.

    Parameters:
    input_data (str): The input data string.

    Returns:
    list: The list of integers extracted from the input data.
    """
    input_list = list(map(int, input_data.split()))
    return input_list[1:]
