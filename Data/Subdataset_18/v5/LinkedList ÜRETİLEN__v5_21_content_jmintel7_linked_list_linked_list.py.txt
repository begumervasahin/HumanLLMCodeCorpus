class Node:
    def __init__(self, data=None):
        """
        Initialize a new node with the given data.
        
        Args:
            data: The data to store in the node. Default is None.
        """
        self.data = data
        self.next = None

class LinkedList:
    def __init__(self):
        """
        Initialize an empty linked list.
        """
        self.head = Node()

    def append(self, data):
        """
        Append a new node with the given data to the end of the list.
        
        Args:
            data: The data to store in the new node.
        """
        new_node = Node(data)
        current = self.head
        while current.next is not None:
            current = current.next
        current.next = new_node

    def length(self):
        """
        Return the length of the linked list.
        
        Returns:
            int: The number of nodes in the linked list.
        """
        current = self.head
        total = 0
        while current.next is not None:
            current = current.next
            total += 1
        return total

    def display_images(self):
        """
        Return a list of images from the linked list.
        
        Returns:
            list: A list of images stored in the linked list.
        """
        current = self.head
        images = []
        while current.next is not None:
            current = current.next
            if current.data:
                images.append(current.data[0])  # Assuming data[0] is an image
        return images

    def get_by_rollno(self, rollno):
        """
        Return the data of the node with the given roll number.
        
        Args:
            rollno: The roll number to search for.
        
        Returns:
            The data of the node with the given roll number, or None if not found.
        """
        current = self.head
        while current.next is not None:
            current = current.next
            data = current.data
            if data and data.rollno == rollno:
                return data
        print('Student with the roll number does not exist')
        return None

    def erase_by_rollno(self, rollno):
        """
        Erase the node with the given roll number.
        
        Args:
            rollno: The roll number of the node to erase.
        """
        current = self.head
        while current.next is not None:
            last_node = current
            current = current.next
            data = current.data
            if data and data.rollno == rollno:
                last_node.next = current.next
                print('Record erased')
                return
        print('Student with the roll number does not exist')

    def get_image_by_index(self, index):
        """
        Return the data of the node at the given index.
        
        Args:
            index (int): The index of the node to retrieve.
        
        Returns:
            The data of the node at the given index, or None if index is out of range.
        """
        if index >= self.length():
            print('ERROR: Index out of range')
            return None
        current = self.head
        for _ in range(index + 1):
            current = current.next
        return current.data

    def erase_image_by_index(self, index):
        """
        Erase the node at the given index.
        
        Args:
            index (int): The index of the node to erase.
        """
        if index >= self.length():
            print('ERROR: Index out of range')
            return None
        current = self.head
        for _ in range(index + 1):
            last_node = current
            current = current.next
        last_node.next = current.next
