import numpy as np

# Example matrix to compute determinant for
test_matrix = np.matrix('1 2 3 4; 1 0 2 0; 0 1 2 3; 2 3 0 0')

# Debug flag to enable verbose output
DEBUG = False

def calculate_2x2_determinant(matrix):
    """
    Calculates the determinant of a 2x2 matrix.

    :param matrix: A 2x2 numpy matrix.
    :return: Determinant of the 2x2 matrix or None if not 2x2.
    """
    if matrix.shape == (2, 2):
        a, b, c, d = matrix[0, 0], matrix[0, 1], matrix[1, 0], matrix[1, 1]
        return a * d - b * c
    return None

def is_square_matrix(matrix):
    """
    Checks if a matrix is square (n x n).

    :param matrix: A numpy matrix.
    :return: True if the matrix is square, False otherwise.
    """
    rows, cols = matrix.shape
    return rows == cols

def get_submatrix(matrix, col_index):
    """
    Creates a submatrix by removing the first row and the specified column.

    :param matrix: A numpy matrix.
    :param col_index: Index of the column to be removed.
    :return: Submatrix after removing the specified column and the first row.
    """
    reduced_matrix = np.delete(matrix, col_index, axis=1)
    return reduced_matrix[1:, :]

def calculate_determinant(matrix):
    """
    Recursively calculates the determinant of an nxn matrix using Laplace expansion.

    :param matrix: A numpy matrix.
    :return: Determinant of the matrix.
    """
    if DEBUG:
        print("Current Matrix:\n", matrix)
    
    if not is_square_matrix(matrix):
        print("Error: The matrix must be square. Current shape:", matrix.shape)
        return None

    # Directly compute the determinant for 2x2 matrices
    determinant = calculate_2x2_determinant(matrix)
    if determinant is not None:
        return determinant

    # Compute determinant for larger matrices
    result = 0
    for col_index in range(matrix.shape[1]):
        # Calculate sign factor and submatrix determinant
        sign_factor = (-1) ** col_index
        submatrix = get_submatrix(matrix, col_index)
        submatrix_determinant = calculate_determinant(submatrix)
        result += sign_factor * matrix[0, col_index] * submatrix_determinant
        
        if DEBUG:
            print(f"Intermediate result after processing column {col_index}: {result}")

    return result

# Compute and display the determinant of the test matrix
print("Determinant:", calculate_determinant(test_matrix))
