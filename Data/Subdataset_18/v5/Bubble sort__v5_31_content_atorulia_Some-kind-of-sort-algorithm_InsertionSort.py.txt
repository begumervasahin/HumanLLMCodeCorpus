import random

def generate_random_list(size=10, upper_limit=100):
    """
    Generate a list of random numbers.
    
    Args:
        size (int): Number of random numbers to generate.
        upper_limit (int): The upper limit for the random numbers.
    
    Returns:
        list: A list of random numbers.
    """
    return [random.randrange(upper_limit) for _ in range(size)]

def is_sorted(array) -> bool:
    """
    Check if the array is sorted in ascending order.
    
    Args:
        array (list): The array to check.
    
    Returns:
        bool: True if the array is sorted, False otherwise.
    """
    return all(array[i] <= array[i + 1] for i in range(len(array) - 1))

def insertion_sort(array):
    """
    Perform Insertion Sort on an array.
    
    Args:
        array (list): The array to sort.
    
    Returns:
        list: The sorted array.
    """
    for i in range(1, len(array)):
        key = array[i]
        j = i - 1
        # Move elements of array[0..i-1] that are greater than key to one position ahead of their current position
        while j >= 0 and array[j] > key:
            array[j + 1] = array[j]
            j -= 1
        array[j + 1] = key
    return array

# Generate a list of 10 random numbers between 0 and 99
arr = generate_random_list()

# Print the original array
print("Original array:", arr)

# Sort the array using Insertion Sort and print the sorted array
sorted_arr = insertion_sort(arr)
print("Sorted array:", sorted_arr)

# Check if the array is sorted
print("Is the array sorted?", is_sorted(sorted_arr))
