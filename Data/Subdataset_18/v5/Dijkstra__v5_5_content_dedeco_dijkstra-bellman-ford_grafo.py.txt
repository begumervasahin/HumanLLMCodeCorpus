from vertex import Vertex

class Graph:
    def __init__(self, directed=False):
        """
        Initialize a Graph instance.

        Parameters:
        directed (bool): If True, the graph is directed. Otherwise, it's undirected.
        """
        self._vertices = {}
        self.directed = directed

    def add_vertex(self, id):
        """
        Add a vertex to the graph.

        Parameters:
        id (int or str): The unique identifier for the vertex.

        Returns:
        Vertex: The created Vertex instance.
        """
        if id not in self._vertices:
            self._vertices[id] = Vertex(id)
        return self._vertices[id]

    def add_edge(self, from_id, to_id, weight=0):
        """
        Add an edge between two vertices with a specified weight.

        Parameters:
        from_id (int or str): The identifier of the start vertex.
        to_id (int or str): The identifier of the end vertex.
        weight (float): The weight of the edge.
        """
        from_vertex = self.add_vertex(from_id)
        to_vertex = self.add_vertex(to_id)

        from_vertex.add_adjacent_vertex(to_vertex, weight)

        if not self.directed:
            to_vertex.add_adjacent_vertex(from_vertex, weight)

    def get_vertices(self):
        """
        Retrieve all vertices in the graph.

        Returns:
        list: A list of Vertex instances.
        """
        return list(self._vertices.values())

    def get_vertex(self, id):
        """
        Retrieve a specific vertex by its identifier.

        Parameters:
        id (int or str): The identifier of the vertex.

        Returns:
        Vertex or None: The Vertex instance if found, otherwise None.
        """
        return self._vertices.get(id)

    def get_edges(self):
        """
        Retrieve all edges in the graph.

        Returns:
        set: A set of tuples representing edges in the form (Vertex, Vertex).
        """
        edges = set()
        for vertex in self._vertices.values():
            for adjacent_vertex in vertex.get_adjacent_vertices():
                edges.add((vertex, adjacent_vertex))
        return edges

    def __iter__(self):
        """
        Return an iterator over the vertices of the graph.

        Returns:
        iterator: An iterator over the Vertex instances in the graph.
        """
        return iter(self._vertices.values())
