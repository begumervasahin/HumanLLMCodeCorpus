import complib
import itertools

def main():
    # Define the functions f and g using complib
    f = complib.Function.from_graph(
        name="f",
        graph={(0, 1, 3), (1, 7, 1)}
    )
    g = complib.Function.from_graph(
        name="g",
        graph={(0, 5, 1)}
    )

    # Display the graphs of the functions
    print("Graph of function f:")
    f.pp_graph()
    print("Graph of function g:")
    g.pp_graph()
    print("Graph of function K0:")
    complib.K0.pp_graph()
    print("Graph of function ident:")
    complib.ident.pp_graph()
    print("Graph of function undef:")
    complib.undef.pp_graph()

    # Check if the functions halt in 1 step for input 0
    print("Function f(0) halts in 1 step:", complib.T3(f, 0, 1))
    print("Function g(0) halts in 1 step:", complib.T3(g, 0, 1))
    print("Function K0(0) halts in 1 step:", complib.T3(complib.K0, 0, 1))
    print("Function ident(0) halts in 1 step:", complib.T3(complib.ident, 0, 1))
    print("Function undef(0) halts in 1 step:", complib.T3(complib.undef, 0, 1))

    # Display the graphs of the functions for a given range
    print("Graph of function f:", f.graph(0, 3))
    print("Graph of function g:", g.graph(0, 3))
    print("Graph of function K0:", complib.K0.graph(0, 3))
    print("Graph of function ident:", complib.ident.graph(0, 3))
    print("Graph of function undef:", complib.undef.graph(0, 3))

    # Evaluate the functions at input 0
    print("f(0) =", f(0))
    print("g(0) =", g(0))
    print("K0(0) =", complib.K0(0))
    print("ident(0) =", complib.ident(0))
    print("undef(0) =", end=" ")
    complib.undef(0, on_undef="continue")

    # Define and test functions for dovetailing
    def sd1_1(f):
        for t in itertools.count(1):
            for k in range(t):
                print(f"Running function {f.name} on input {k} for {t} steps.")
                output = f(k, steps=t)
                if output is not None and output == 7:
                    print(f"Function {f.name} returns 7 on input {k} in {t} steps.")
                    return True

    def sd1_2(f):
        return complib.dovetailing(f, lambda i, o, t: o == 7)

    def sd2_1(f):
        results = set()
        for t in itertools.count(1):
            for k in range(t):
                print(f"Running function {f.name} on input {k} for {t} steps.")
                output = f(k, steps=t)
                if output is not None:
                    for input_val, output_val in results:
                        if input_val != k and output_val == output:
                            print(f"Function {f.name} returns {output} on inputs {input_val} and {k}!")
                            return True
                    print(f"Recording that {f.name}({k}) = {output}.")
                    results.add((k, output))
                else:
                    print(f"{f.name}({k}) did not terminate in {t} steps.")

    def sd2_2(f):
        return complib.dovetailing_on_pairs(f, lambda i, oi, j, oj, t: i != j and oi == oj)

    # Test the dovetailing functions with f and ident
    sd1_1(f)
    sd1_2(f)
    sd1_1(complib.ident)
    sd1_2(complib.ident)

    # Test dovetailing with a function generated by complib.K
    k12_3_steps = complib.K(12, 3)
    sd2_1(k12_3_steps)
    sd2_2(k12_3_steps)

    # Test encoding and decoding of pairs as natural numbers
    print("Encoding and decoding of pairs as natural numbers:")
    for i in range(10):
        pair = complib.number2pair(i)
        number = complib.pair2number(*pair)
        print(f"{i} -> {pair} -> {number}")

if __name__ == "__main__":
    main()
