import matplotlib.pyplot as plt
from DynammicProgramming import *

def plot_route_with_info(recommended_points, coordinates_db, total_distance, total_time, x_median, y_median):
    """
    Plots the route on a graph and adds distance and time information.

    Parameters:
        recommended_points (list): List of recommended points (indices) for the route.
        coordinates_db (dict or list): Dictionary or list containing coordinates (latitude, longitude) for each point.
        total_distance (float): Total distance of the route.
        total_time (float): Total time taken for the route.
        x_median (float): Median value of x-coordinates (used for plotting vertical line).
        y_median (float): Median value of y-coordinates (used for plotting horizontal line).
    """
    x_coords = [coordinates_db[point[0]][0] for point in recommended_points]
    y_coords = [coordinates_db[point[0]][1] for point in recommended_points]

    # Close the loop by appending the first point at the end
    x_coords.append(x_coords[0])
    y_coords.append(y_coords[0])

    # Plot the route
    plt.plot(x_coords, y_coords, label="Route")

    # Add distance and time information as text on the plot
    plt.text(-84, -17.5, f"Distancia: {total_distance}")
    plt.text(-84, -19.5, f"Tiempo: {total_time}")

    # Plot points and mark the starting point with a red 'x'
    plt.plot(x_coords, y_coords, 'o', label="Points")
    plt.plot(x_coords[0], y_coords[0], 'rx', label="Start Point")

    # Add median lines for reference
    plt.axhline(y_median, color='black', linestyle='--', label="Y Median")
    plt.axvline(x_median, color='black', linestyle='--', label="X Median")

    # Add title and labels for clarity
    plt.title("Route Visualization with Distance and Time Information")
    plt.xlabel("Longitude")
    plt.ylabel("Latitude")

    # Add a legend for clarity
    plt.legend()

    # Display the plot
    plt.show()

# Assuming recommended_points, coordinates_db, total_distance, total_time, x_median, y_median are defined elsewhere
plot_route_with_info(recommended_points, coordinates_db, total_distance, total_time, x_median, y_median)
