import numpy as np

class EstimatePerceptualMixingTime:
    def __init__(self, RoomDims):
        self.RoomDims = RoomDims
        self.mixing_time_estimate = {}
    
    def model_based(self):
        print('----------------------------------')
        print('MODEL-BASED MIXING TIME PREDICTION')
        print('----------------------------------')
        print('ROOM PROPERTIES:')
        
        x_d, y_d, z_d = self.RoomDims
        
        # Validate room dimensions
        self._validate_dimensions(x_d, y_d, z_d)
        
        # Print room dimensions
        print(f'Room dimensions: height = {x_d}m, length = {y_d}m, width = {z_d}m')
        
        # Calculate and print volume and surface area
        volume, surface_area = self._calculate_room_properties(x_d, y_d, z_d)
        print(f'Volume: {volume} m³')
        print(f'Surface area: {surface_area} m²')
        
        # Calculate and print perceptual mixing times
        tmp50, tmp95, rootvol, MFP = self._calculate_mixing_times(volume, surface_area)
        print(f'tmp50: {tmp50} ms')
        print(f'tmp95: {tmp95} ms')
        
        # Update mixing time estimates
        self._update_mixing_time_estimate(tmp50, tmp95, rootvol, MFP)
        
        return self
    
    def _validate_dimensions(self, x_d, y_d, z_d):
        if x_d <= 0 or y_d <= 0 or z_d <= 0:
            raise ValueError('RoomDims must contain positive, numeric values only.')
    
    def _calculate_room_properties(self, x_d, y_d, z_d):
        volume = x_d * y_d * z_d
        surface_area = 2 * (x_d * y_d + x_d * z_d + y_d * z_d)
        return volume, surface_area
    
    def _calculate_mixing_times(self, volume, surface_area):
        rootvol = np.sqrt(volume)
        MFP = 47 * volume / surface_area
        tmp50 = 20.08 * volume / surface_area + 12
        tmp95 = 0.0117 * volume + 50.1
        return tmp50, tmp95, rootvol, MFP
    
    def _update_mixing_time_estimate(self, tmp50, tmp95, rootvol, MFP):
        self.mixing_time_estimate['model'] = {
            'tmp50': tmp50,
            'tmp95': tmp95,
            'rootV': rootvol,
            'mfp_based': MFP
        }
