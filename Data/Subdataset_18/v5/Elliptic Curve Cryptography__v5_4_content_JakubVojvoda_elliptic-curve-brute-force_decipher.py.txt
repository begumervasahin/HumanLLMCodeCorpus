class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y

    def equals(self, other):
        return self.x == other.x and self.y == other.y

    def __repr__(self):
        return f"Point(x={self.x}, y={self.y})"


class ECurve:
    def __init__(self, a, b, p):
        self.a = a
        self.b = b
        self.p = p

    def check(self, point):
        """Check if a point is on the elliptic curve."""
        left = self.mod(point.y ** 2)
        right = self.mod(point.x ** 3 + self.a * point.x + self.b)
        discriminant = 4 * self.a ** 3 + 27 * self.b ** 2
        return left == right and discriminant != 0

    def mod(self, x):
        """Perform a modulo operation."""
        return x % self.p

    def invmod(self, x):
        """Calculate the modular inverse."""
        s0, s1 = 0, 1
        r0, r1 = self.p, x
        while r0 != 0:
            q = r1 // r0
            r1, r0 = r0, r1 - q * r0
            s1, s0 = s0, s1 - q * s0
        return s1 % self.p

    def add(self, p, q):
        """Add two points on the elliptic curve."""
        if p.equals(Point(0, 0)):
            return q
        if q.equals(Point(0, 0)):
            return p
        if p.equals(q):
            if p.y == 0:
                return Point(0, 0)
            l = self.mod((3 * p.x ** 2 + self.a) * self.invmod(2 * p.y))
        else:
            if p.x == q.x:
                return Point(0, 0)
            l = self.mod((q.y - p.y) * self.invmod(q.x - p.x))

        rx = self.mod(l ** 2 - p.x - q.x)
        ry = self.mod(l * (p.x - rx) - p.y)
        return Point(rx, ry)

    def double(self, p):
        """Double a point on the elliptic curve."""
        return self.add(p, p)

    def multiply(self, p, n):
        """Multiply a point by an integer using the double-and-add algorithm."""
        result = Point(0, 0)
        current = p

        while n > 0:
            if n % 2 == 1:
                result = self.add(result, current)
            current = self.double(current)
            n //= 2

        return result


class Decipher:
    def __init__(self, curve, p, q, n):
        self.curve = curve
        self.p = p
        self.q = q
        self.n = n

    def run(self):
        """Run the discrete logarithm problem solver."""
        current_point = Point(0, 0)
        for d in range(1, self.n):
            current_point = self.curve.add(self.p, current_point)
            if not self.curve.check(current_point):
                return 0
            if current_point.equals(self.q):
                return d
        return 0
