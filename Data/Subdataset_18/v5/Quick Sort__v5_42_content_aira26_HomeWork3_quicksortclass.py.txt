import random
import time
import matplotlib.pyplot as plt

class Quicksort:
    def __init__(self):
        self.mylist = []

    def add(self, value):
        self.mylist.append(value)

    def quick_sort(self):
        if len(self.mylist) <= 1:
            return self.mylist
        pivot = self.mylist[0]
        left = Quicksort()
        right = Quicksort()
        for i in range(1, len(self.mylist)):
            if self.mylist[i] < pivot:
                left.add(self.mylist[i])
            else:
                right.add(self.mylist[i])
        return left.quick_sort() + [pivot] + right.quick_sort()

    def get_min(self):
        sorted_list = self.quick_sort()
        return sorted_list[0]

    def get_max(self):
        sorted_list = self.quick_sort()
        return sorted_list[-1]

    def get_list(self):
        return self.mylist

def measure_time(quicksort_instance, values_list):
    total_time_add = 0
    total_time_min = 0
    total_time_max = 0

    for num in values_list:
        start = time.time()
        quicksort_instance.add(num)
        total_time_add += time.time() - start

        start = time.time()
        quicksort_instance.get_min()
        total_time_min += time.time() - start

        start = time.time()
        quicksort_instance.get_max()
        total_time_max += time.time() - start

    return total_time_add, total_time_min, total_time_max

    repetitions = 5

    values_add, values_min, values_max = [], [], []

    for rounds in range(step, max_operations, step):
