class BinarySearchTree:
    class Node:
        def __init__(self, value):
            """Initialize a node with a given value."""
            self.value = value
            self.left = None
            self.right = None

    def __init__(self):
        """Initialize an empty binary search tree."""
        self.root = None

    def pretty_print(self):
        """Pretty print the binary search tree."""
        self._pretty_print_aux(self.root, 0)

    def _pretty_print_aux(self, node, indent_level):
        """Recursive helper function for pretty printing the tree."""
        if node is None:
            return

        if node.right:
            self._pretty_print_aux(node.right, indent_level + 1)

        print("    " * indent_level + str(node.value))

        if node.left:
            self._pretty_print_aux(node.left, indent_level + 1)

    def find(self, target):
        """Find and return the node with the given target value."""
        return self._find_aux(self.root, target)

    def _find_aux(self, node, target):
        """Recursive helper function to find a node."""
        if node is None:
            return None
        if target == node.value:
            return node
        elif target < node.value:
            return self._find_aux(node.left, target)
        else:
            return self._find_aux(node.right, target)

    def insert(self, value):
        """Insert a new value into the tree in its correct position."""
        if self.root is None:
            self.root = self.Node(value)
        else:
            self._insert_aux(self.root, value)

    def _insert_aux(self, node, value):
        """Recursive helper function to insert a new value."""
        if value < node.value:
            if node.left is None:
                node.left = self.Node(value)
            else:
                self._insert_aux(node.left, value)
        elif value > node.value:
            if node.right is None:
                node.right = self.Node(value)
            else:
                self._insert_aux(node.right, value)

    def flip(self):
        """Flip (mirror) the binary search tree."""
        self._flip_aux(self.root)

    def _flip_aux(self, node):
        """Recursive helper function to flip the tree."""
        if node:
            node.left, node.right = node.right, node.left
            self._flip_aux(node.left)
            self._flip_aux(node.right)

def main():
    """Main function to demonstrate the binary search tree functionality."""
    tree = BinarySearchTree()

    # Insert nodes into the tree
    values = ["man", "dog", "zebra", "ape", "elephant", "yak", "zorse", "fly"]
    for value in values:
        tree.insert(value)

    # Pretty print the tree
    print("Original tree:")
    tree.pretty_print()

    # Flip the tree and print again
    tree.flip()
    print("\nFlipped tree:")
    tree.pretty_print()

if __name__ == "__main__":
    main()
