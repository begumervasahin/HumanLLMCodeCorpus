def check_divisibility_by_7(number):
    """
    Check if a given number, after repeatedly applying a specific transformation, is divisible by 7.

    Parameters:
    number (int): The initial number to check.

    Returns:
    str: Message indicating whether the final number is divisible by 7 or not.
    """
    current_number = number

    while len(str(current_number)) > 2 and current_number > 0:
        print(current_number)
        prefix = str(current_number)[:-1]  # All digits except the last one
        last_digit = int(str(current_number)[-1])  # The last digit
        current_number = int(prefix) - (2 * last_digit)

    # Check if the resulting number is divisible by 7
    if current_number % 7 == 0:
        return f"Number {current_number} is divisible by 7!"
    else:
        return f"Number {current_number} is not divisible by 7."

# Example usage
if __name__ == "__main__":
    initial_number = 1946
    result = check_divisibility_by_7(initial_number)
    print(result)
