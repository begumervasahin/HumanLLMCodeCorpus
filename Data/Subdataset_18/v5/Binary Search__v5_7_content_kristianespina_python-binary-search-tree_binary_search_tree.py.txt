from random import randint

class Node:
    def __init__(self, value):
        self.value = value
        self.left = None
        self.right = None

    def insert(self, value):
        """Inserts a value into the binary search tree."""
        if value < self.value:
            if self.left is None:
                self.left = Node(value)
            else:
                self.left.insert(value)
        elif value > self.value:
            if self.right is None:
                self.right = Node(value)
            else:
                self.right.insert(value)
        # Duplicates are ignored as BSTs typically do not allow them

    def to_array(self, node, array):
        """Performs a Depth-First Search (DFS) to convert the tree into an array."""
        if node:
            array.append(node.value)
            self.to_array(node.left, array)
            self.to_array(node.right, array)
        return array

class BinaryTree:
    def __init__(self):
        self.root = None

    def insert(self, value):
        """Inserts a value into the binary tree."""
        if self.root is None:
            self.root = Node(value)
        else:
            self.root.insert(value)

    def find(self, value):
        """Finds a value in the binary tree."""
        return self._find_node(self.root, value)

    def _find_node(self, node, value):
        """Recursively finds a node with the given value."""
        if node is None:
            return False
        if value == node.value:
            return True
        elif value < node.value:
            return self._find_node(node.left, value)
        else:
            return self._find_node(node.right, value)

    def to_array(self):
        """Converts the binary tree to a DFS array representation."""
        if self.root is None:
            return []
        return self.root.to_array(self.root, [])

    def rearrange(self):
        """
        Re-arranges the binary tree so that the median of the values becomes the root.
        Returns the binary tree in DFS array form.
        """
        nodes = self.to_array()
        nodes.sort()

        # Rebuild the tree with the median as the root
        median_index = len(nodes) // 2
        self.root = Node(nodes.pop(median_index))

        for value in nodes:
            self.insert(value)

        return self.to_array()

# Example usage
if __name__ == "__main__":
    tree = BinaryTree()
    
    # Insert random values into the tree
    for _ in range(50):
        tree.insert(randint(0, 100))

    print("Original Tree (DFS array form):", tree.to_array())
    print("Rearranged Tree (Median as root):", tree.rearrange())
    print("Finding 55 in the tree:", tree.find(55))
