import heapq
from collections import Counter
from functools import partial
from io import StringIO

class Huffman:
    """Class representing the Huffman encoding and decoding process."""

    def __init__(self, data, width):
        if not data or width <= 0:
            raise ValueError("Invalid data or width.")
        
        self.original_data = data
        self.width = width
        self.data_chunks = [chunk for chunk in iter(partial(StringIO(data).read, width), '')]
        self._dictionary = {}

    def encode_huffman(self):
        """Encode the original data using Huffman coding."""
        freq_map = self._create_frequency_map()
        huffman_tree = self._build_huffman_tree(freq_map)
        self._generate_huffman_codes(huffman_tree)
        return self._encode_data()

    def decode_huffman(self, encoded_data):
        """Decode the encoded data using the Huffman coding scheme."""
        if not encoded_data:
            raise ValueError("Encoded data cannot be empty.")
        
        decoded_data = ''
        current_bits = ''
        reverse_dict = self._get_reverse_dictionary()

        for bit in encoded_data:
            current_bits += bit
            if current_bits in reverse_dict:
                decoded_data += reverse_dict[current_bits]
                current_bits = ''
        
        return decoded_data

    def get_compression_ratio(self, encoded_data):
        """Calculate and return the compression ratio."""
        if not encoded_data:
            raise ValueError("Encoded data cannot be empty.")
        
        original_bits = len(self.original_data) * 8
        compressed_bits = len(encoded_data)
        return 1 - (compressed_bits / float(original_bits))

    @property
    def dictionary(self):
        return self._dictionary

    @dictionary.setter
    def dictionary(self, value):
        self._dictionary = value

    def _create_frequency_map(self):
        """Create a frequency map of the data chunks."""
        return [(freq, chunk) for chunk, freq in Counter(self.data_chunks).items()]

    def _build_huffman_tree(self, freq_map):
        """Build the Huffman tree based on the frequency map."""
        if not freq_map:
            raise ValueError("Frequency map cannot be empty.")
        
        heapq.heapify(freq_map)
        while len(freq_map) > 1:
            left = heapq.heappop(freq_map)
            right = heapq.heappop(freq_map)
            combined_node = (left[0] + right[0], left, right)
            heapq.heappush(freq_map, combined_node)
        
        return freq_map[0]

    def _generate_huffman_codes(self, tree):
        """Generate Huffman codes based on the Huffman tree."""
        stack = [(tree, '')]
        while stack:
            node, code = stack.pop()
            if len(node) == 2:  # Leaf node
                self._dictionary[node[1]] = code
            else:  # Internal node
                stack.append((node[1], code + '0'))
                stack.append((node[2], code + '1'))

    def _encode_data(self):
        """Encode the original data using the generated Huffman codes."""
        if not self._dictionary:
            raise ValueError("Huffman encoding dictionary is empty.")
        
        return ''.join(self._dictionary[chunk] for chunk in self.data_chunks if chunk in self._dictionary)

    def _get_reverse_dictionary(self):
        """Get the reverse of the Huffman encoding dictionary."""
        return {code: char for char, code in self._dictionary.items()}

def test_encoding_decoding(data):
    """Test the Huffman encoding and decoding process."""
    print("************** Test Encoding **************")
    
    huffman = Huffman(data, 4)
    encoded_data = huffman.encode_huffman()
    decoded_data = huffman.decode_huffman(encoded_data)

    print(f"Original data: {data}")
    print(f"Encoded data: {encoded_data}")
    print(f"Decoded data: {decoded_data}")
    print(f"Original and decoded data are the same: {data == decoded_data}")
    print(f"Compression Ratio: {huffman.get_compression_ratio(encoded_data):.4f}")
    return decoded_data
