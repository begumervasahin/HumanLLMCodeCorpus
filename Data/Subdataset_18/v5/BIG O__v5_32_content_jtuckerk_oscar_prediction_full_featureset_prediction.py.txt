import numpy as np
from sklearn import linear_model
from sklearn.metrics import roc_auc_score
from sklearn.naive_bayes import GaussianNB
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import StratifiedKFold
from sklearn import svm
import codecs
import re

def load():
    return load_batting(), load_allstars()

def get_input():
    OMDB = True
    IMDB_experience = True
    IMDB_oscar_prior = True
    PLOT = False
    year_range_min = 1990
    year_range_max = 2015

    omdb_file = './data/omdb_data.txt'
    omdb_feats = load_omdb_features(omdb_file, year_range_min, year_range_max)
    actor_oscar_feats = load_imdb_features('./data/actor_oscar_prior.txt', 10, "actor oscar related features")
    actor_ex_feats = load_imdb_features('./data/actor_experience.txt', 10, "actor experience related features")
    plot_embedding_feats = load_plot_embeddings('./data/plot_embeddings.txt', 128) if PLOT else {}

    feats = combine_features(omdb_feats, actor_ex_feats, actor_oscar_feats, plot_embedding_feats, OMDB, IMDB_experience, IMDB_oscar_prior, PLOT)
    
    return feats

def load_omdb_features(omdb_file, year_range_min, year_range_max):
    omdb_feats = {}
    print("Getting OMDB features")
    
    with codecs.open(omdb_file, 'r', encoding='utf-8') as file:
        for line in file.readlines():
            line_data = eval(line)
            assert len(line_data) == 14
            
            critic_score = float(line_data[3])
            audience_score = float(line_data[4]) if line_data[4] != 'N/A' else 0
            duration = float(line_data[6].split()[0]) if line_data[6] != 'N/A' else 0
            metascore = float(line_data[7]) if line_data[7] != 'N/A' else 0
            imdb_rating = float(line_data[10].replace(",", '')) if line_data[10] != 'N/A' else 0
            imdb_votes = float(line_data[11].replace(",", '')) if line_data[11] != 'N/A' else 0
            year = int(re.findall(r'\d{4}', line_data[2]).pop())
            
            if year < year_range_min or year > year_range_max:
                continue

            features = [critic_score, audience_score, duration, metascore, imdb_rating, imdb_votes]
            title_year = (line_data[1], str(year))
            omdb_feats[title_year] = features

    return omdb_feats

def load_imdb_features(file_path, feature_size, feature_name):
    feats = {}
    print(f"Getting IMDB {feature_name}")
    
    with codecs.open(file_path, 'r', encoding='utf-8') as file:
        for line in file.readlines():
            line_data = eval(line)
            assert len(line_data[1]) == feature_size
            feats[line_data[0]] = line_data[1]

    return feats

def load_plot_embeddings(file_path, embedding_size):
    feats = {}
    print("Getting plot vector representations")
    
    with codecs.open(file_path, 'r', encoding='utf-8') as file:
        for line in file.readlines():
            line_data = eval(line)
            assert len(line_data[1]) == embedding_size
            title, year = line_data[0]
            feats[(title, str(year))] = line_data[1]

    return feats

def combine_features(omdb_feats, actor_ex_feats, actor_oscar_feats, plot_embedding_feats, OMDB, IMDB_experience, IMDB_oscar_prior, PLOT):
    feats = {}
    
    for k in omdb_feats.keys():
        a = omdb_feats[k] if OMDB else []
        b = actor_ex_feats.get(k, [0] * len(next(iter(actor_ex_feats.values())))) if IMDB_experience else []
        c = actor_oscar_feats.get(k, [0] * len(next(iter(actor_oscar_feats.values())))) if IMDB_oscar_prior else []
        d = plot_embedding_feats.get(k, [0] * len(next(iter(plot_embedding_feats.values())))) if PLOT else []
        
        feats[k] = np.array(a + b + c + d)
    
    return feats

def determine_y(encoding):
    return 1 if encoding >= 1 else 0

def get_output():
    oscars = {}
    oscar_file = './data/oscar_y_full.txt'
    year_data = {}
    
    with codecs.open(oscar_file, 'r', encoding='utf-8') as file:
        for line in file.readlines():
            line_data = eval(line)
            y = determine_y(line_data[1])
            year_key = str(line_data[0][1])
            
            if year_key not in year_data:
                year_data[year_key] = {"nom": 0, "not": 0}
            
            if y:
                year_data[year_key]["nom"] += 1
                oscars[(line_data[0][0], year_key)] = y
            else:
                year_data[year_key]["not"] += 1
    
    return oscars

def get_X_y(input_dict, output_dict):
    X = [input_dict.get(k, np.zeros(20)) for k in input_dict.keys()]
    y = [output_dict.get(k, 0) for k in input_dict.keys()]
    
    return np.array(X), np.array(y)

def test_classifier(clf, X, Y):
    folds = StratifiedKFold(n_splits=5)
    aucs = []
    
    for train, test in folds.split(X, Y):
        clf.fit(X[train], Y[train])
        prediction = clf.predict_proba(X[test])
        aucs.append(roc_auc_score(Y[test], prediction[:, 1]))
    
    print(f"{clf.__class__.__name__} AUCs: {aucs}, Mean AUC: {np.mean(aucs)}")

def main():
    X, Y = get_X_y(get_input(), get_output())
    
    classifiers = [
        linear_model.SGDClassifier(loss='log'),
        GaussianNB(),
        RandomForestClassifier(n_estimators=10, max_depth=10)
    ]
    
    for clf in classifiers:
        test_classifier(clf, X, Y)

if __name__ == '__main__':
    main()
