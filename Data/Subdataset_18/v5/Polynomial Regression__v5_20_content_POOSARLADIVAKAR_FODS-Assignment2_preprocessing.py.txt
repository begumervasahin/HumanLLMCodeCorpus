import numpy as np

class NormalScaler:
    """
    Transforms data to a normal distribution with mean 0 and standard deviation 1.
    """
    def fit(self, arr):
        """
        Computes the mean and standard deviation of the input data.

        Parameters:
        arr (numpy array): The data to compute mean and standard deviation from.
        """
        self.mean = np.mean(arr)
        self.std = np.std(arr)

    def transform(self, arr):
        """
        Transforms the input data using the computed mean and standard deviation.

        Parameters:
        arr (numpy array): The data to be transformed.

        Returns:
        numpy array: The normalized data.
        """
        return (arr - self.mean) / self.std


class MinMaxScaler:
    """
    Scales data to be within the range [0, 1].
    """
    def fit(self, arr):
        """
        Computes the minimum and maximum values of the input data.

        Parameters:
        arr (numpy array): The data to compute minimum and maximum values from.
        """
        self.min = np.min(arr)
        self.max = np.max(arr)

    def transform(self, arr):
        """
        Transforms the input data to be within the range [0, 1].

        Parameters:
        arr (numpy array): The data to be transformed.

        Returns:
        numpy array: The scaled data.
        """
        return (arr - self.min) / (self.max - self.min)

    def inverse_transform(self, arr):
        """
        Applies the inverse transformation to return the data to its original range.

        Parameters:
        arr (numpy array): The scaled data to be reverted.

        Returns:
        numpy array: The original data.
        """
        return arr * (self.max - self.min) + self.min


def main():
    data = np.array([1, 2, 3, 4, 5])

    # Example usage of NormalScaler
    normal_scaler = NormalScaler()
    normal_scaler.fit(data)
    normalized_data = normal_scaler.transform(data)
    print("Normalized data:", normalized_data)

    # Example usage of MinMaxScaler
    minmax_scaler = MinMaxScaler()
    minmax_scaler.fit(data)
    scaled_data = minmax_scaler.transform(data)
    print("Min-Max scaled data:", scaled_data)
    original_data = minmax_scaler.inverse_transform(scaled_data)
    print("Inverse transformed data:", original_data)


if __name__ == "__main__":
    main()
