from __future__ import print_function
import sys
import pandora.utils
from pandora.tagger import Tagger

def load_data(data_dir, params):
    return pandora.utils.load_annotated_dir(
        data_dir,
        format='tab',
        extension='.tsv',
        include_pos=params['include_pos'],
        include_lemma=params['include_lemma'],
        include_morph=params['include_morph'],
        nb_instances=None
    )

def main():
    print('::: started :::')

    # Load configuration file path from command line arguments
    config_file_path = sys.argv[1]
    
    # Load parameters from configuration file
    params = pandora.utils.get_param_dict(config_file_path)
    params['config_path'] = config_file_path

    # Load training and development data
    train_data = load_data('data/medieval_latin/train', params)
    dev_data = load_data('data/medieval_latin/test', params)

    # Initialize the tagger with the loaded parameters
    tagger = Tagger(**params)
    
    # Setup the tagger for training with the training and development data
    tagger.setup_to_train(train_data=train_data, dev_data=dev_data)

    # Train the tagger for the specified number of epochs
    num_epochs = int(params['nb_epochs'])
    for epoch in range(num_epochs):
        tagger.epoch()
        tagger.save()

    # Save the final model
    tagger.save()
    
    print('::: ended :::')

if __name__ == '__main__':
    main()
