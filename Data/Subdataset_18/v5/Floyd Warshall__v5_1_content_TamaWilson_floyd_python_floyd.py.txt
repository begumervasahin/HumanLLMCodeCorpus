# Define infinity as a large enough value to represent the absence of an edge between nodes
INFINITY = float("inf")

# Define the graph as an adjacency matrix
graph = [
    [0, 5, INFINITY, 10],
    [INFINITY, 0, 3, INFINITY],
    [INFINITY, INFINITY, 0, 1],
    [INFINITY, INFINITY, INFINITY, 0]
]

def floyd_warshall(graph):
    """
    Perform the Floyd-Warshall algorithm to find the shortest paths between all pairs of nodes.
    
    Parameters:
    graph (list of list of float): A 2D list representing the graph as an adjacency matrix.
                                  Each element graph[i][j] represents the weight of the edge from node i to node j.
                                  A value of INFINITY indicates no direct edge between the nodes.
    """
    num_vertices = len(graph)
    
    # Initialize the distance matrix with the given graph (deep copy)
    distance_matrix = [row[:] for row in graph]

    # Run the Floyd-Warshall algorithm
    for k in range(num_vertices):
        for i in range(num_vertices):
            for j in range(num_vertices):
                # Update the distance matrix with the shortest path found so far
                distance_matrix[i][j] = min(distance_matrix[i][j], distance_matrix[i][k] + distance_matrix[k][j])
    
    # Print the resulting distance matrix
    print_distance_matrix(distance_matrix)

def print_distance_matrix(distance_matrix):
    """
    Print the distance matrix.
    
    Parameters:
    distance_matrix (list of list of float): The 2D list representing the shortest path distances.
                                             Each element distance_matrix[i][j] represents the shortest distance from node i to node j.
                                             A value of INFINITY indicates no path between the nodes.
    """
    for row in distance_matrix:
        formatted_row = "\t".join(f"{distance if distance != INFINITY else 'INF'}" for distance in row)
        print(formatted_row)

if __name__ == '__main__':
    floyd_warshall(graph)
