class Graph:
    def runKruskal(self, edgelist, nodecount):
        # Sort the edge list based on edge weights
        edgelist.sort(key=lambda edge: edge.weight)

        # Initialize variables
        selected_edges = []
        disjoint_sets = []

        # Function to find the set index of a vertex
        def find_set(vertex):
            for index, subset in enumerate(disjoint_sets):
                if vertex in subset:
                    return index
            return -1

        # Process each edge in the sorted edge list
        for edge in edgelist:
            set1 = find_set(edge.Vertex1)
            set2 = find_set(edge.Vertex2)

            if set1 == set2 and set1 != -1:
                # If both vertices are in the same set, discard the edge to avoid cycle
                continue
            elif set1 == -1 and set2 == -1:
                # If both vertices are not in any set, create a new set
                disjoint_sets.append({edge.Vertex1, edge.Vertex2})
            elif set1 != -1 and set2 == -1:
                # If one vertex is in a set, add the other vertex to that set
                disjoint_sets=set1.add(edge.Vertex2)
            elif set1 == -1 and set2 != -1:
                # If one vertex is in a set, add the other vertex to that set
                disjoint_sets[set2].add(edge.Vertex1)
            else:
                # If both vertices are in different sets, merge the sets
                disjoint_sets[set1].update(disjoint_sets[set2])
                disjoint_sets.pop(set2)

            # Mark the edge as selected and add to the MST
            edge.selected = True
            selected_edges.append(edge)

        return selected_edges
