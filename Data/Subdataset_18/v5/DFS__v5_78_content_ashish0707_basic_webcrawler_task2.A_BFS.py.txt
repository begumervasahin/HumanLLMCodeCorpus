from Webcrawler import Crawler  # Import the Crawler class from the Webcrawler module
from bfs import BFS  # Import the BFS class from the bfs module

def setup_crawler(url, searcher):
    """
    Set up the crawler with specified configurations.
    
    Args:
        url (str): The URL to be crawled.
        searcher (Searcher): An instance of a searcher class.
    
    Returns:
        Crawler: Configured Crawler instance.
    """
    crawler = Crawler(url, searcher)
    crawler.add_to_mandatory_list('/wiki/')
    crawler.add_to_disallowed_list(':')
    crawler.set_focused_text('solar')
    return crawler

def main():
    """
    Main function to initialize and run the web crawler.
    """
    # Define the URL for crawling
    target_url = "https://en.wikipedia.org/wiki/Sustainable_energy"
    
    # Create an instance of the BFS searcher
    bfs_searcher = BFS()
    
    # Set up the crawler with the target URL and searcher
    crawler = setup_crawler(target_url, bfs_searcher)
    
    # Start the crawling process
    crawler.start()

if __name__ == "__main__":
    main()
