import numpy as np
import pandas as pd
from selenium import webdriver
from selenium.common.exceptions import NoSuchElementException
import time

class CourseScraper:
    def __init__(self, driver_path: str = "./chromedriver"):
        """Initialize the web driver and open the timetable page."""
        self.driver = webdriver.Chrome(driver_path)
        self.driver.get("https://timetable.iit.artsci.utoronto.ca/")
        
    def close(self):
        """Close the web driver."""
        self.driver.close()
        
    def retrieve_courses(self, department: str) -> pd.DataFrame:
        """Retrieve course information for a specific department."""
        department = department.upper()
        search_box = self.driver.find_element_by_id("courseCode")
        search_button = self.driver.find_element_by_id("searchButton")
        
        search_box.clear()
        search_box.click()
        search_box.send_keys(department)
        search_button.click()
        
        course_elements = []
        while not course_elements:
            time.sleep(1.0)
            course_elements = self.driver.find_elements_by_class_name("perCourse")
        
        course_info_list = []
        print(f"Total courses found: {len(course_elements)}")
        
        for course_element in course_elements:
            code, title = course_element.find_element_by_class_name("courseTitle").text.split("   ")
            print(f"{code}\t{title}")
            
            meeting_elements = course_element.find_elements_by_class_name("perMeeting")
            for meeting_element in meeting_elements:
                meeting_code = meeting_element.find_element_by_class_name("colCode").text
                print(f"\t{meeting_code}")
                
                try:
                    meeting_info = meeting_element.find_element_by_class_name(
                        "secLec" if meeting_code.startswith("LEC") else "secTut"
                    ).text     
                except NoSuchElementException:
                    meeting_info = meeting_element.find_element_by_class_name("secPra").text       
                
                course_info_list.append([code, title, meeting_code, meeting_info])
        
        course_info_df = pd.DataFrame(
            course_info_list,
            columns=["Code", "Title", "Session", "Details"]
        )
        return course_info_df
    
    def download_all_courses(self, save_path: str) -> None:
        """Download all courses information and save to a CSV file."""
        department_elements = self.driver.find_elements_by_class_name("option")
        department_list = [dept.text for dept in department_elements]
        
        code_list = [
            code[1:-1]
            for department in department_list
            for code in department.split(" ")
            if code.startswith("(") and code.endswith(")")
        ]
        
        all_courses = []
        for code in code_list:
            courses_df = self.retrieve_courses(code)
            all_courses.append(courses_df)
        
        final_df = pd.concat(all_courses, axis=0)
        print(final_df.head())
        print(final_df.shape)
        final_df.to_csv(save_path, index=False)

if __name__ == "__main__":
    scraper = CourseScraper()
    scraper.download_all_courses(save_path="./results.csv")
    scraper.close()
