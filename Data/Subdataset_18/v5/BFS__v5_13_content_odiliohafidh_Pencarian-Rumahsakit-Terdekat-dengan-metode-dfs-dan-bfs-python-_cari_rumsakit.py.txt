# Graph representation of the map
graph = {
    'A': {'RS1', 'C', 'E'},
    'B': {'E', 'D'},
    'C': {'A', 'B', 'RS2'},
    'D': {'B', 'F', 'K', 'RS3'},
    'E': {'A', 'B'},
    'F': {'D', 'G'},
    'G': {'H', 'F', 'I'},
    'H': {'G'},
    'I': {'G', 'J'},
    'J': {'I'},
    'K': {'D', 'L', 'M'},
    'L': {'RS2', 'K'},
    'M': {'K'},
    'RS1': {'A'},
    'RS2': {'C', 'L'},
    'RS3': {'D'}
}

def bfs(graph, start):
    """
    Perform Breadth-First Search (BFS) to find the shortest path to any hospital (RS1, RS2, RS3).

    Args:
        graph (dict): The graph representation of the map.
        start (str): The starting node.

    Returns:
        list: The shortest path to the nearest hospital, or None if no path is found.
    """
    queue = [[start]]
    goals = {'RS1', 'RS2', 'RS3'}
    visited = set()

    while queue:
        path = queue.pop(0)
        current_node = path[-1]

        if current_node in goals:
            return path

        if current_node not in visited:
            visited.add(current_node)
            for neighbor in graph.get(current_node, []):
                new_path = path + [neighbor]
                queue.append(new_path)

    print("No path found")
    return None

def dfs(graph, start):
    """
    Perform Depth-First Search (DFS) to find a path to any hospital (RS1, RS2, RS3).

    Args:
        graph (dict): The graph representation of the map.
        start (str): The starting node.

    Returns:
        list: The path to the nearest hospital, or None if no path is found.
    """
    stack = [[start]]
    goals = {'RS1', 'RS2', 'RS3'}
    visited = set()

    while stack:
        path = stack.pop()
        current_node = path[-1]

        if current_node in goals:
            return path

        if current_node not in visited:
            visited.add(current_node)
            for neighbor in graph.get(current_node, []):
                new_path = path + [neighbor]
                stack.append(new_path)

    print("No path found")
    return None

# Example usage
if __name__ == "__main__":
    start_node = 'A'
    print("BFS path to nearest hospital:", bfs(graph, start_node))
    print("DFS path to nearest hospital:", dfs(graph, start_node))
