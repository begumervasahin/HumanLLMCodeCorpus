def initialize_api():
    """Initialize the Tweepy API with OAuth credentials."""
    auth = tweepy.OAuthHandler(CONSUMER_KEY, CONSUMER_SECRET)
    auth.set_access_token(ACCESS_TOKEN, ACCESS_SECRET)
    return tweepy.API(auth, wait_on_rate_limit=True, wait_on_rate_limit_notify=True)

def check_api_connection(api):
    """Check the API connection and print rate limit status."""
    if not api:
        print("Problem connecting to API")
    else:
        print(api.rate_limit_status()['resources']['search'])

def get_place_ids(api, locations):
    """Fetch place IDs for given locations using the Tweepy API."""
    place_id_list = []

    for city, lat, long in locations:
        try:
            places = api.geo_search(lat=lat, long=long, granularity="neighborhood")
            place_id = places[0].id
            place_id_list.append((city, place_id))
            print(f'{city} ID is: {place_id}')
        except Exception as e:
            print(f"Error fetching data for {city}: {e}")
    
    return place_id_list

def write_place_ids_to_csv(filename, place_id_list):
    """Write the place IDs to a CSV file."""
    with open(filename, 'w', newline='') as file:
        writer = csv.writer(file, delimiter=',')
        writer.writerows(place_id_list)

def main():
    api = initialize_api()
    check_api_connection(api)

    locations = [
        ("Rochester", "43.155708", "-77.612545"),
        ("Manhattan", "40.753250", "-74.003807"),
        ("Buffalo", "42.887691", "-78.879372"),
        ("Syracuse", "43.047939", "-76.147453"),

        ("Brentwood", "40.781212", "-73.246147"),
        ("Schenectady", "42.814220", "-73.944099"),
        ("Utica", "43.102039", "-75.230003"),
        ("Niagara Falls", "43.094460", "-79.056427")
    ]

    place_id_list = get_place_ids(api, locations)
    write_place_ids_to_csv('place_id_list.csv', place_id_list)

if __name__ == "__main__":
    main()
