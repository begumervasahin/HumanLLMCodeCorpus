class Set:
    """
    Union-Find Data Structure.
    Heuristics: path compression, union by rank.
    """

    def __init__(self, value):
        """
        Initialize a set containing the given value.
        """
        self.value = value
        self.parent = self
        self.next = self
        self.rank = 0

    def findSet(self):
        """
        Returns the representative element of the set and applies path compression.
        """
        if self.parent != self:
            self.parent = self.parent.findSet()
        return self.parent

    def union(self, other):
        """
        Merges two sets using union by rank.
        """
        root1 = self.findSet()
        root2 = other.findSet()

        if root1 != root2:
            if root1.rank < root2.rank:
                root1.parent = root2
            elif root1.rank > root2.rank:
                root2.parent = root1
            else:
                root2.parent = root1
                root1.rank += 1

            # Maintain the next pointers
            root1.next, root2.next = root2.next, root1.next

    def add(self, value):
        """
        Inserts a new value into the set.
        """
        new_set = Set(value)
        self.union(new_set)

    def getElements(self):
        """
        Generator for the elements in the set.
        """
        current = self
        while True:
            yield current.value
            current = current.next
            if current == self:
                break

    def __str__(self):
        """
        Returns a string representation of the set.
        """
        return "{{{}}}".format(", ".join(map(str, self.getElements())))

    def __repr__(self):
        """
        Returns the official string representation of the set.
        """
        return self.__str__()
