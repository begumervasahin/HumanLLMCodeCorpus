import json
import sys
import MapReduce

class RachnaRajputPhase2:
    def __init__(self):
        self.mr = MapReduce.MapReduce()
        self.results_dict = {}

    def mapper(self, record):
        """Map function to process each record."""
        with open(sys.argv[1]) as f:
            for line in f:
                chunk_data = json.loads(line)
                count = sum(1 for chunk_line in chunk_data if set(chunk_line) >= set(record))
                record_tuple = tuple(record)
                self.results_dict[record_tuple] = [count, len(chunk_data)]
                self.mr.emit_intermediate(record_tuple, [count, len(chunk_data)])

    def reducer(self, key, values):
        """Reduce function to aggregate results."""
        total_count = sum(value[0] for value in values)
        self.mr.emit([key, total_count])

    def execute(self, input_data_path):
        """Execute the MapReduce process."""
        with open(input_data_path) as input_data:
            self.mr.execute(input_data, self.mapper, self.reducer)

def main():
    if len(sys.argv) != 3:
        print("Usage: python script.py <chunk_file> <input_file>")
        sys.exit(1)

    chunk_file = sys.argv[1]
    input_file = sys.argv[2]

    phase2 = RachnaRajputPhase2()
    phase2.execute(input_file)

if __name__ == '__main__':
    main()
