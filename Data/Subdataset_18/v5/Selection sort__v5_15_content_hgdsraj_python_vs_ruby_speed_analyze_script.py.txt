# Open the file and read all lines
with open('pyt_res.txt') as file:
    lines = file.readlines()

# Initialize an empty list to store the extracted times
times = []

# Loop through each line and extract the time if the line contains "real"
for line in lines:
    if "real" in line:
        # Extract the time value, convert it to float, and divide by 1000
        time_str = line.split()[1][:-1]
        time_value = float(time_str) / 1000
        times.append(time_value)

# Calculate the average of the extracted times
average_time = sum(times) / len(times) if times else 0

# Print the average time
print("The average time was")
print(f"{average_time:.3f} seconds")
