def write_file(file_name: str, data: str) -> None:
    """
    Writes the given string data to a file in binary mode using UTF-8 encoding.

    Args:
        file_name (str): The name of the file to write to.
        data (str): The string data to write to the file.
    """
    with open(file_name, 'wb') as file:
        file.write(data.encode('utf-8'))  # Encode the entire string to bytes using UTF-8


def read_file(file_name: str) -> None:
    """
    Reads and prints the contents of a file in binary mode using UTF-8 decoding.

    Args:
        file_name (str): The name of the file to read from.
    """
    with open(file_name, 'rb') as file:
        content = file.read()
        print(content.decode('utf-8', errors='replace'))  # Decode the entire file content and print

    file_name = 'test.txt'

    write_file(file_name, data)
    read_file(file_name)
