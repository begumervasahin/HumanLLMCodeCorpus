def selection_sort(items):
    """
    Sorts a list of items in ascending order using the selection sort algorithm.

    Args:
    items (list): The list of items to be sorted.

    Returns:
    None: The list is sorted in place.
    """
    n = len(items)
    for step in range(n):
        # Find the index of the smallest item in the unsorted portion of the list
        min_index = step
        for i in range(step + 1, n):
            if items[i] < items[min_index]:
                min_index = i
        
        # Swap the smallest item with the first item in the unsorted portion
        items[step], items[min_index] = items[min_index], items[step]

    print("Sorted items:", items)
    print("Total number of items:", n)

# Example usage
items = [64, 25, 12, 22, 11]
selection_sort(items)
