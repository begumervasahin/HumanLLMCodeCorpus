from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives import serialization, hashes
from cryptography.hazmat.primitives.asymmetric import padding

# Define the text to encrypt
TEXT_TO_ENCRYPT = (
    'This is a “Hazardous Materials” module. This module is full of land mines, '
    'dragons, and dinosaurs with laser guns.'
)

# Function to load a public key from a PEM file
def load_public_key(pem_file_path):
    with open(pem_file_path, "rb") as key_file:
        public_key = serialization.load_pem_public_key(
            key_file.read(),
            backend=default_backend()
        )
    return public_key

# Function to serialize a public key to PEM format
def serialize_public_key_to_pem(public_key):
    public_pem = public_key.public_bytes(
        encoding=serialization.Encoding.PEM,
        format=serialization.PublicFormat.SubjectPublicKeyInfo
    )
    return public_pem.decode()

# Function to encrypt text using a public key with OAEP padding and SHA-256
def encrypt_text(public_key, text):
    ciphertext = public_key.encrypt(
        text.encode(),
        padding.OAEP(
            mgf=padding.MGF1(algorithm=hashes.SHA256()),
            algorithm=hashes.SHA256(),
            label=None
        )
    )
    return ciphertext

# Function to save the ciphertext to a file
def save_ciphertext_to_file(ciphertext, file_path):
    with open(file_path, 'w') as f:
        f.write(ciphertext.hex())

# Load the public key
public_key = load_public_key("public_key.pem")

# Serialize and print the public key in PEM format
public_pem = serialize_public_key_to_pem(public_key)
print("Public Key in PEM Format:")
print(public_pem)

# Encrypt the text
ciphertext = encrypt_text(public_key, TEXT_TO_ENCRYPT)

# Print the encoded text
ciphertext_string = ciphertext.hex()
print('Encoded text:')
print(ciphertext_string)

# Save the encoded text to a file
save_ciphertext_to_file(ciphertext, 'encoded_text.txt')
