class Edge:
    def __init__(self, start, target, weight):
        """
        Initialize an Edge instance.

        :param start: The starting node of the edge.
        :param target: The target node of the edge.
        :param weight: The weight of the edge.
        """
        self._start = start
        self._target = target
        self._weight = weight

    @property
    def weight(self):
        """
        Get the weight of the edge.

        :return: The weight of the edge.
        """
        return self._weight

    @weight.setter
    def weight(self, value):
        """
        Set a new weight for the edge.

        :param value: The new weight of the edge.
        """
        self._weight = value

    @property
    def start(self):
        """
        Get the starting node of the edge.

        :return: The starting node.
        """
        return self._start

    @start.setter
    def start(self, value):
        """
        Set a new starting node for the edge.

        :param value: The new starting node.
        """
        self._start = value

    @property
    def target(self):
        """
        Get the target node of the edge.

        :return: The target node.
        """
        return self._target

    @target.setter
    def target(self, value):
        """
        Set a new target node for the edge.

        :param value: The new target node.
        """
        self._target = value

# Example usage:
if __name__ == "__main__":
    # Create an edge from node 'A' to node 'B' with a weight of 5
    edge = Edge("A", "B", 5)

    # Print the initial values
    print(f"Start: {edge.start}")   # Output: Start: A
    print(f"Target: {edge.target}") # Output: Target: B
    print(f"Weight: {edge.weight}") # Output: Weight: 5

    # Update the edge's properties
    edge.start = "C"
    edge.target = "D"
    edge.weight = 10

    # Print the updated values
    print(f"Updated Start: {edge.start}")   # Output: Updated Start: C
    print(f"Updated Target: {edge.target}") # Output: Updated Target: D
    print(f"Updated Weight: {edge.weight}") # Output: Updated Weight: 10
