def main(path_v1):
    from map_distance import main as process_file

    edges, num_of_dest = process_file(path_v1)
    edges_corrected = [edge for edge in edges if edge[2] < 900]
    print("Corrected edges:", edges_corrected)

    class Graph:
        def __init__(self):
            self.vertices = {}

        def add_vertex(self, key):
            """Add a vertex with the given key to the graph."""
            self.vertices[key] = Vertex(key)

        def get_vertex(self, key):
            """Return vertex object with the corresponding key."""
            return self.vertices.get(key)

        def add_edge(self, src_key, dest_key, weight=1):
            """Add an edge from src_key to dest_key with the given weight."""
            self.vertices[src_key].add_neighbour(self.vertices[dest_key], weight)

        def __contains__(self, key):
            return key in self.vertices

        def __len__(self):
            return len(self.vertices)

        def __iter__(self):
            return iter(self.vertices.values())

    class Vertex:
        def __init__(self, key):
            self.key = key
            self.points_to = {}

        def get_key(self):
            """Return key corresponding to this vertex object."""
            return self.key

        def add_neighbour(self, dest, weight):
            """Make this vertex point to dest with the given edge weight."""
            self.points_to[dest] = weight

        def get_neighbours(self):
            """Return all vertices pointed to by this vertex."""
            return self.points_to.keys()

        def get_weight(self, dest):
            """Get weight of edge from this vertex to dest."""
            return self.points_to[dest]

        def does_it_point_to(self, dest):
            """Return True if this vertex points to dest."""
            return dest in self.points_to

    def floyd_warshall(graph):
        """Compute shortest paths between all pairs of vertices in the graph."""
        distance = {v: {w: float('inf') for w in graph} for v in graph}
        next_v = {v: {w: None for w in graph} for v in graph}

        for v in graph:
            for n in v.get_neighbours():
                distance[v][n] = v.get_weight(n)
                next_v[v][n] = n

        for v in graph:
            distance[v][v] = 0
            next_v[v][v] = None

        for k in graph:
            for i in graph:
                for j in graph:
                    if distance[i][j] > distance[i][k] + distance[k][j]:
                        distance[i][j] = distance[i][k] + distance[k][j]
                        next_v[i][j] = next_v[i][k]

        return distance, next_v

    def print_path(next_v, start, end):
        """Print the shortest path from start to end vertex."""
        path = []
        while next_v[start][end]:
            path.append(start.get_key())
            start = next_v[start][end]
        path.append(end.get_key())
        return path

    graph = Graph()
    for i in range(num_of_dest):
        graph.add_vertex(i + 1)

    for src, dest, weight in edges_corrected:
        graph.add_edge(src, dest, weight)

    distance, next_v = floyd_warshall(graph)
    list_for_sketch = []

    for start in graph:
        for end in graph:
            if next_v[start][end]:
                path = print_path(next_v, start, end)
                list_for_sketch.append([[start.get_key(), end.get_key()], path, [distance[start][end]]])

    return list_for_sketch

if __name__ == "__main__":
    path_v1 = 'path_to_your_file.txt'  # Replace with your actual file path
    list_for_sketch = main(path_v1)
    print("List for sketch:", list_for_sketch)
