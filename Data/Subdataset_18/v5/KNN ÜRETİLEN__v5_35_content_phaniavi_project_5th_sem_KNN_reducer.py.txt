import sys
from operator import itemgetter

def main():
    distances = []
    k = 5

    # Read input from standard input
    for line in sys.stdin:
        line = line.strip()
        elements = line.split()
        distances.append(elements)

    # Sort distances based on the first element (assuming it represents a numerical value)
    distances.sort(key=itemgetter(0))

    # Calculate the estimated value as the average of the top k estimates
    estimate_sum = 0.0
    count = 0

    for i in range(k):
        try:
            estimate_sum += float(distances[i][1])
            count += 1
        except (ValueError, IndexError):
            continue

    # Print the average of the top k estimates
    if count > 0:
        print(estimate_sum / count)
    else:
        print("No valid estimates found in the input.")

if __name__ == "__main__":
    main()
