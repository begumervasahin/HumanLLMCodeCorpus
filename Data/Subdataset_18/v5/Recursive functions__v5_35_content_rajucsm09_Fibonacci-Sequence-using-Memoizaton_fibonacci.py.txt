# Cache dictionary for storing and searching Fibonacci calculations
fibonacci_cache = {}

def fibonacci(num):
    """
    Calculate the Fibonacci number using memoization to cache results.

    Args:
        num (int): The position in the Fibonacci sequence.

    Returns:
        int: The Fibonacci number at the specified position.
        bool: False if the input is not an integer.
    """
    if not isinstance(num, int):
        return False

    # Check if the value is already in the cache
    if num in fibonacci_cache:
        return fibonacci_cache[num]

    # Base cases
    if num == 0:
        value = 0
    elif num == 1:
        value = 1
    else:
        # Recursive calculation with memoization
        value = fibonacci(num - 1) + fibonacci(num - 2)

    # Cache the computed value
    fibonacci_cache[num] = value
    return value

def print_fibonacci_sequence(n):
    """
    Print the first n Fibonacci numbers.

    Args:
        n (int): The number of Fibonacci numbers to print.
    """
    for num in range(n):
        print(f"{num}: {fibonacci(num)}")

if __name__ == "__main__":
    # Print the first 1000 Fibonacci numbers and observe system behavior
    print_fibonacci_sequence(1000)
