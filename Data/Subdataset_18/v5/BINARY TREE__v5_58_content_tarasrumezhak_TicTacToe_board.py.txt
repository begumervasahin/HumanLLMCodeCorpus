class Board:
    """Board representation class for a Tic-Tac-Toe game."""

    def __init__(self):
        self.board = [[None] * 3 for _ in range(3)]
        self.last_sign = None
        self.last_pos = None
        self.tree = None
        self.user_sign = "O"
        self.computer_sign = "X"

    def check_state(self):
        """
        Checks the current game state.
        :return: None, 'O', 'X', or 'Draw!'
        """
        def all_same(lst):
            """
            Returns True if all elements in the list are the same and not None.
            :param lst: list
            :return: bool
            """
            return lst[0] is not None and all(x == lst[0] for x in lst)

        # Check rows and columns
        for i in range(3):
            if all_same(self.board[i]) or all_same([self.board[j][i] for j in range(3)]):
                return self.board[i][0] if all_same(self.board[i]) else self.board[0][i]

        # Check diagonals
        if all_same([self.board[i][i] for i in range(3)]):
            return self.board[0][0]
        if all_same([self.board[i][2 - i] for i in range(3)]):
            return self.board[0][2]

        # Check for draw
        if not self.is_free_cell():
            return 'Draw!'

        return None

    def __str__(self):
        """
        Returns a string representation of the board.
        :return: str
        """
        rows = ["| " + " | ".join(self.board[i][j] if self.board[i][j] else " " for j in range(3)) + " |" for i in range(3)]
        return "------------- y:\n" + "\n".join(rows) + "\n-------------\n" + "x: 1   2   3"

    def put(self, sign, x, y):
        """
        Puts the sign at the position (x, y).
        :param sign: str
        :param x: int
        :param y: int
        :return: None
        """
        if not (1 <= x <= 3 and 1 <= y <= 3):
            raise IndexError("Out of bounds")
        if self.board[x - 1][y - 1] is not None:
            raise IndexError("The position is already used")
        self.board[x - 1][y - 1] = sign

    def check_cell(self, coords):
        """
        Checks if the cell at the given coordinates is empty.
        :param coords: tuple (row, col)
        :return: bool
        """
        row, col = coords
        return self.board[row][col] is None

    def find_empty(self):
        """
        Returns a list of all empty cells.
        :return: list of tuples
        """
        return [(row + 1, col + 1) for row in range(3) for col in range(3) if self.check_cell((row, col))]

    def is_free_cell(self):
        """
        Checks if there is any free cell on the board.
        :return: bool
        """
        return any(self.check_cell((row, col)) for row in range(3) for col in range(3))
