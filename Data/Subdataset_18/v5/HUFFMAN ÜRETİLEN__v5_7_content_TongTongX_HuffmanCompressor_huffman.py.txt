from collections import Counter
from minheap import MinHeap

class TreeLeafEndMessage:
    """Special marker to indicate the end of the Huffman Tree."""
    pass

class TreeLeaf:
    """Represents a leaf node in the Huffman Tree."""
    def __init__(self, value):
        self.value = value

class TreeBranch:
    """Represents a branch node in the Huffman Tree, holding left and right children."""
    def __init__(self, left, right):
        self.left = left
        self.right = right

def make_tree(freq_table):
    """Constructs a Huffman Tree from a frequency table."""
    trees = MinHeap()
    
    # Add a special end message to the heap with a frequency of 1
    trees.add(1, TreeLeafEndMessage())
    
    # Add each symbol as a leaf node with its frequency
    for symbol, freq in freq_table.items():
        trees.add(freq, TreeLeaf(symbol))
    
    # Combine trees until only one tree remains
    while len(trees) > 1:
        rfreq, right = trees.pop_min()
        lfreq, left = trees.pop_min()
        trees.add(lfreq + rfreq, TreeBranch(left, right))
    
    # The final remaining tree is the complete Huffman Tree
    _, huffman_tree = trees.pop_min()
    return huffman_tree

def decode(tree, bitreader):
    """Decodes a stream of bits using the Huffman Tree."""
    while True:
        if isinstance(tree, TreeLeafEndMessage):
            return None
        elif isinstance(tree, TreeLeaf):
            return tree.value
        elif isinstance(tree, TreeBranch):
            # Traverse left or right based on the next bit
            tree = tree.left if bitreader.readbit() == 0 else tree.right
        else:
            raise TypeError(f'Invalid tree type: {type(tree)}')

def make_encoding_table(huffman_tree):
    """Generates an encoding table from the Huffman Tree."""
    encoding_table = {}

    def recurse(tree, path):
        if isinstance(tree, TreeLeafEndMessage):
            encoding_table[None] = path
        elif isinstance(tree, TreeLeaf):
            encoding_table[tree.value] = path
        elif isinstance(tree, TreeBranch):
            # Recursively encode left as False and right as True
            recurse(tree.left, path + (False,))
            recurse(tree.right, path + (True,))
        else:
            raise TypeError(f'Invalid tree type: {type(tree)}')

    recurse(huffman_tree, ())
    return encoding_table

def make_freq_table(stream):
    """Creates a frequency table from a binary stream."""
    freqs = Counter()
    buffer = bytearray(512)
    
    while True:
        count = stream.readinto(buffer)
        freqs.update(buffer[:count])
        if count < len(buffer):
            break
    
    return freqs
