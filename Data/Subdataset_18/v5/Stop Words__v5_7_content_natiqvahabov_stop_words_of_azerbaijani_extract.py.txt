import urllib.request as urllib2
from bs4 import BeautifulSoup
from nltk.tokenize import RegexpTokenizer
from collections import Counter

def fetch_wikipedia_urls(page_url):
    """Fetches Wikipedia article URLs and titles from the given page URL."""
    urls = []
    titles = []

    # Open and parse the Wikipedia page
    response = urllib2.urlopen(page_url)
    soup = BeautifulSoup(response, 'html.parser')

    # Find the table containing the article links
    table_row = soup.find('div', class_='mw-parser-output').find('tr')
    table_cells = table_row.find_next('tr').find_all('td')

    # Extract URLs and titles from the table cells
    for cell in table_cells:
        links = cell.find_all('a')
        for link in links:
            title = link.get('title')
            if title:
                href = link['href']
                urls.append(f'https://az.wikipedia.org{href}')
                titles.append(title)

    return urls, titles

def tokenize_wikipedia_articles(urls):
    """Fetches and tokenizes text from a list of Wikipedia article URLs."""
    tokenizer = RegexpTokenizer(r'\w+')
    all_tokens = []

    for index, url in enumerate(urls, start=1):
        print(f"Processing article {index} from {url}")

        # Open and parse the Wikipedia article page
        response = urllib2.urlopen(url)
        soup = BeautifulSoup(response, 'html.parser')
        content = soup.find('div', class_='mw-parser-output').get_text()

        # Tokenize the content
        tokens = tokenizer.tokenize(content)
        all_tokens.extend(tokens)

    return all_tokens

def save_word_counts_to_file(tokens, filename='stopwordsAZE.txt'):
    """Counts word frequencies and saves the results to a file."""
    word_counts = Counter(tokens)
    with open(filename, 'a') as file:
        file.write(str(word_counts))

def main():
    wiki_page_url = "https://az.wikipedia.org/wiki/Vikipediya:Seçilmiş_məqalələr"
    wiki_urls, wiki_titles = fetch_wikipedia_urls(wiki_page_url)
    
    # For demonstration purposes, you can use a subset of URLs
    demo_urls = [
        'https://az.wikipedia.org/wiki/GRB_970508',
        'https://az.wikipedia.org/wiki/Günəş_sistemi'
    ]
    
    # Get tokenized words from the articles
    tokens = tokenize_wikipedia_articles(demo_urls)

    # Save the word counts to a file
    save_word_counts_to_file(tokens)

if __name__ == "__main__":
    main()
