import collections

class Graph:
    def __init__(self):
        self.nodes = set()  # Set to store the graph's nodes
        self.edges = collections.defaultdict(list)  # Dictionary to store adjacency lists
        self.distances = {}  # Dictionary to store edge distances

    def add_node(self, value):
        """Adds a node to the graph."""
        self.nodes.add(value)

    def add_edge(self, from_node, to_node, distance):
        """Adds an undirected edge between two nodes with a given distance."""
        self.edges[from_node].append(to_node)
        self.edges[to_node].append(from_node)
        self.distances[(from_node, to_node)] = distance
        self.distances[(to_node, from_node)] = distance

def dijkstra(graph, start_node):
    """
    Implements Dijkstra's algorithm to find the shortest paths from the start_node to all other nodes.

    Parameters:
        graph (Graph): The Graph object containing the nodes, edges, and distances.
        start_node (str): The starting node for Dijkstra's algorithm.

    Returns:
        dict: A dictionary mapping each node to its shortest distance from the start_node.
        dict: A dictionary mapping each node to its predecessor on the shortest path.
    """
    visited = {start_node: 0}  # Shortest distance to each node from the start_node
    path = {}  # Shortest path tree mapping each node to its predecessor
    unvisited_nodes = set(graph.nodes)  # Set of all unvisited nodes

    while unvisited_nodes:
        current_node = None

        # Find the unvisited node with the smallest known distance from the start_node
        for node in unvisited_nodes:
            if node in visited:
                if current_node is None or visited[node] < visited[current_node]:
                    current_node = node

        if current_node is None:
            break  # All remaining unvisited nodes are inaccessible from start_node

        unvisited_nodes.remove(current_node)
        current_distance = visited[current_node]

        # Update the distances to the neighboring nodes
        for neighbor in graph.edges[current_node]:
            distance = current_distance + graph.distances[(current_node, neighbor)]
            if neighbor not in visited or distance < visited[neighbor]:
                visited[neighbor] = distance
                path[neighbor] = current_node

    return visited, path
