def generate_prime_numbers(n):
    """
    Generate a list of prime numbers up to n (inclusive).
    """
    if n is None:
        return "Please enter a positive integer as the limit!"
    
    if not isinstance(n, int):
        return "That is not an integer. Please enter a number without a decimal."
    
    if n < 2:
        return "Please enter a positive integer greater than or equal to 2."
    
    if n == 2:
        return [2]

    sieve = list(range(3, n + 1, 2))
    mroot = int(n ** 0.5)
    half = (n + 1) // 2 - 1

    for i in range(half):
        if sieve[i]:
            prime = 2 * i + 3
            if prime > mroot:
                break
            for j in range((prime * prime - 3) // 2, half, prime):
                sieve[j] = 0

    return [2] + [x for x in sieve if x]

def main():
    # Example usage
    n = 30
    primes = generate_prime_numbers(n)
    if isinstance(primes, list):
        print(f"Prime numbers up to {n}: {primes}")
    else:
        print(primes)

if __name__ == "__main__":
    main()
