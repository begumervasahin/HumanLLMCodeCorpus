import time
import gc
import numpy as np
import knn

def main():
    experiment_start_time = time.time()
    
    train_dir = './digits/trainingDigits'
    test_dir = './digits/testDigits'
    result_file_path = 'result.txt'
    result_csv_path = 'result_table.csv'
    
    # Load training and testing samples
    train_samples, train_labels = knn.load_sample_set(train_dir)
    test_samples, ground_labels = knn.load_sample_set(test_dir)
    
    ks = [3, 5, 7, 9]
    knn_functions = [
        (knn.get_knn_e_dist, 'get_knn_e_dist'),
        (knn.get_knn_e_dist_with_kdtree, 'get_knn_e_dist_with_kdtree'),
        (knn.get_knn_m_dist, 'get_knn_m_dist')
    ]
    label_functions = [
        (knn.get_label_by_knn, 'get_label_by_knn'),
        (knn.get_label_by_wknn, 'get_label_by_wknn')
    ]
    pca_parameters = [0, 4, 8, 16, 32, 64, 128, 256]
    
    results = [
        ['k', 'knn_function', 'label_function', 'pca_parameter', 'accuracy', 
         'macro_precision', 'macro_recall', 'macro_F1', 'execution_time', 'dimension']
    ]
    
    with open(result_file_path, 'w') as result_file:
        for k in ks:
            for knn_func, knn_func_str in knn_functions:
                for label_func, label_func_str in label_functions:
                    for pca_parameter in pca_parameters:
                        single_result = [k, knn_func_str, label_func_str, pca_parameter]
                        
                        # Log the experiment conditions
                        print_experiment_conditions(result_file, k, knn_func_str, label_func_str, pca_parameter)
                        
                        # Prepare datasets
                        train_set, test_set, train_labels_copy, ground_labels_copy = prepare_datasets(
                            train_samples, train_labels, test_samples, ground_labels
                        )
                        
                        # Measure the execution time
                        start_time = time.time()
                        result_labels, dimension = knn.get_test_samples_labels(
                            k, train_set, train_labels_copy, test_set, knn_func, label_func, pca_parameter
                        )
                        elapsed_time = time.time() - start_time
                        
                        # Evaluate results
                        metrics = knn.result_evaluate(ground_labels_copy, result_labels)
                        accuracy, macro_precision, macro_recall, macro_F1 = metrics[:4]
                        
                        # Log the results
                        log_results(result_file, accuracy, metrics, elapsed_time, dimension)
                        
                        single_result.extend([accuracy, macro_precision, macro_recall, macro_F1, elapsed_time, dimension])
                        results.append(single_result)
                        
                        gc.collect()
    
    total_time = time.time() - experiment_start_time
    
    # Log the total experiment time
    with open(result_file_path, 'a') as result_file:
        print('***********************', file=result_file)
        print(f'Total Time = {total_time}', file=result_file)
        print('***********************', file=result_file)
    
    # Write the results to a CSV file
    write_results_to_csv(result_csv_path, results)

def print_experiment_conditions(result_file, k, knn_func_str, label_func_str, pca_parameter):
    print('\n******** Conditions ******', file=result_file)
    print(f'k = {k}', file=result_file)
    print(f'knn_function = {knn_func_str}', file=result_file)
    print(f'label_function = {label_func_str}', file=result_file)
    print(f'pca_parameter = {pca_parameter}', file=result_file)
    print('**************************', file=result_file)

def prepare_datasets(train_samples, train_labels, test_samples, ground_labels):
    train_set = np.copy(train_samples)
    test_set = np.copy(test_samples)
    train_labels_copy = np.copy(train_labels)
    ground_labels_copy = np.copy(ground_labels)
    return train_set, test_set, train_labels_copy, ground_labels_copy

def log_results(result_file, accuracy, metrics, elapsed_time, dimension):
    precision, recall, F1, macro_precision, macro_recall, macro_F1 = metrics[1:]
    
    print(f'accuracy = {accuracy}', file=result_file)
    print(f'precision = {precision}', file=result_file)
    print(f'recall = {recall}', file=result_file)
    print(f'F1 = {F1}', file=result_file)
    print(f'macro_precision = {macro_precision}', file=result_file)
    print(f'macro_recall = {macro_recall}', file=result_file)
    print(f'macro_F1 = {macro_F1}', file=result_file)
    print(f'execution time = {elapsed_time}', file=result_file)
    print(f'dimension = {dimension}', file=result_file)
    print(file=result_file)

def write_results_to_csv(result_csv_path, results):
    with open(result_csv_path, 'w') as csv_file:
        for result in results:
            csv_file.write(', '.join(map(str, result)) + '\n')

if __name__ == '__main__':
    main()
