import random
import time

'''Utility Functions'''
def check_different_time(num1, num2):
    """Calculate the percentage difference between two numbers."""
    return (num2 / num1 * 100) if num1 > num2 else (num1 / num2 * 100)

def print_array(arr, name):
    """Print the sorted array with a given name."""
    print(f"Sorted array from {name} Sort is:")
    print(' '.join(map(str, arr)))

def check_time(name_func, start_time):
    """Calculate and print the elapsed time for a sorting function."""
    elapsed_time = time.time() - start_time
    print(f'Function [{name_func.__name__}] finished in {elapsed_time:.3f} sec')
    return elapsed_time

def generate_random_array(size, lower=1, upper=5000):
    """Generate an array of random integers."""
    return [random.randint(lower, upper) for _ in range(size)]

'''Sorting Algorithms'''
def heapify(arr, n, i):
    """Helper function to maintain heap property."""
    largest = i
    left = 2 * i + 1
    right = 2 * i + 2
    if left < n and arr[i] < arr[left]:
        largest = left
    if right < n and arr[largest] < arr[right]:
        largest = right
    if largest != i:
        arr[i], arr[largest] = arr[largest], arr[i]
        heapify(arr, n, largest)

def heap_sort(arr):
    """Perform heap sort on the array."""
    n = len(arr)
    for i in range(n // 2 - 1, -1, -1):
        heapify(arr, n, i)
    for i in range(n - 1, 0, -1):
        arr[i], arr[0] = arr[0], arr[i]
        heapify(arr, i, 0)

def merge_sort(arr):
    """Perform merge sort on the array."""
    if len(arr) > 1:
        mid = len(arr) // 2
        left_half = arr[:mid]
        right_half = arr[mid:]
        merge_sort(left_half)
        merge_sort(right_half)
        i = j = k = 0
        while i < len(left_half) and j < len(right_half):
            if left_half[i] < right_half[j]:
                arr[k] = left_half[i]
                i += 1
            else:
                arr[k] = right_half[j]
                j += 1
            k += 1
        while i < len(left_half):
            arr[k] = left_half[i]
            i += 1
            k += 1
        while j < len(right_half):
            arr[k] = right_half[j]
            j += 1
            k += 1

def partition(arr, low, high):
    """Helper function for quick sort to partition the array."""
    pivot = arr[high]
    i = low - 1
    for j in range(low, high):
        if arr[j] <= pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    return i + 1

def quick_sort_helper(arr, low, high):
    """Helper function for quick sort."""
    if low < high:
        pi = partition(arr, low, high)
        quick_sort_helper(arr, low, pi - 1)
        quick_sort_helper(arr, pi + 1, high)

def quick_sort(arr):
    """Perform quick sort on the array."""
    quick_sort_helper(arr, 0, len(arr) - 1)

def selection_sort(arr):
    """Perform selection sort on the array."""
    n = len(arr)
    for i in range(n):
        min_index = i
        for j in range(i + 1, n):
            if arr[min_index] > arr[j]:
                min_index = j
        arr[i], arr[min_index] = arr[min_index], arr[i]

def insertion_sort(arr):
    """Perform insertion sort on the array."""
    for i in range(1, len(arr)):
        key = arr[i]
        j = i - 1
        while j >= 0 and key < arr[j]:
            arr[j + 1] = arr[j]
            j -= 1
        arr[j + 1] = key

def bubble_sort(arr):
    """Perform bubble sort on the array."""
    n = len(arr)
    for i in range(n):
        for j in range(0, n - i - 1):
            if arr[j] > arr[j + 1]:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]

if __name__ == "__main__":
    array_sizes = [10000] * 6
    arrays = [generate_random_array(size) for size in array_sizes]
    array_names = ['Bubble', 'Insertion', 'Selection', 'Quick', 'Merge', 'Heap']

    print("Generated Arrays:")
    for name, arr in zip(array_names, arrays):
        print(f"Array for {name} Sort:")
        print(' '.join(map(str, arr)))
        print("")

    sorting_algorithms = {
        'Bubble': bubble_sort,
        'Insertion': insertion_sort,
        'Selection': selection_sort,
        'Quick': quick_sort,
        'Merge': merge_sort,
        'Heap': heap_sort
    }

    elapsed_times = {}

    for name, arr in zip(array_names, arrays):
        start_time = time.time()
        sorting_algorithms[name](arr)
        elapsed_time = check_time(sorting_algorithms[name], start_time)
        elapsed_times[name] = elapsed_time
        print("")

    for name, arr in zip(array_names, arrays):
        print_array(arr, name)

    print("Elapsed Times:")
    for name, elapsed_time in elapsed_times.items():
        print(f"{name} Sort took {elapsed_time:.3f} seconds")
