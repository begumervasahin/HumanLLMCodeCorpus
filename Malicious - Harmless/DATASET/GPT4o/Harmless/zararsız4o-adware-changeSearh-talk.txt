import json
import os

def list_files_in_directory(directory):
    try:
        # Get the list of files in the directory
        files = os.listdir(directory)
        
        # Filter out directories, keeping only files
        files = [f for f in files if os.path.isfile(os.path.join(directory, f))]
        
        return files
    except Exception as e:
        print(f"Error: {e}")
        return []

def save_files_to_json(file_list, json_file):
    try:
        # Save the list of files to a JSON file
        with open(json_file, 'w') as f:
            json.dump(file_list, f, indent=4)
        print(f"File list saved to {json_file}")
    except Exception as e:
        print(f"Error: {e}")

if __name__ == "__main__":
    # Specify the directory to list files from
    directory = "your_directory_path_here"
    
    # Specify the JSON file to save the list
    json_file = "file_list.json"
    
    # List the files in the directory
    files = list_files_in_directory(directory)
    
    # Save the list of files to a JSON file
    save_files_to_json(files, json_file)
