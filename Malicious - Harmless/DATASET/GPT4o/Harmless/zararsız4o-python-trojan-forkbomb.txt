import subprocess
import os

def list_files_and_count_lines(directory):
    try:
        # List all files in the given directory
        files = os.listdir(directory)
        print(f"Files in directory '{directory}':")
        for file in files:
            print(f"- {file}")
        
        # Count the number of lines in each file
        for file in files:
            file_path = os.path.join(directory, file)
            if os.path.isfile(file_path):
                result = subprocess.run(['wc', '-l', file_path], capture_output=True, text=True)
                line_count = result.stdout.split()[0]
                print(f"File '{file}' has {line_count} lines.")
            else:
                print(f"'{file}' is not a file, skipping line count.")
    except Exception as e:
        print(f"An error occurred: {e}")

# Replace 'your_directory_path' with the path to the directory you want to analyze
directory_path = 'your_directory_path'
list_files_and_count_lines(directory_path)
