import ctypes
import time

# Define the GetTickCount64 function from the kernel32 library (Windows)
GetTickCount64 = ctypes.windll.kernel32.GetTickCount64
GetTickCount64.restype = ctypes.c_ulonglong

def get_system_uptime():
    # Get the uptime in milliseconds
    milliseconds = GetTickCount64()
    
    # Convert milliseconds to seconds
    seconds = milliseconds // 1000
    
    # Get the days, hours, minutes, and seconds
    days = seconds // 86400
    hours = (seconds % 86400) // 3600
    minutes = (seconds % 3600) // 60
    seconds = seconds % 60
    
    return days, hours, minutes, seconds

def display_uptime():
    days, hours, minutes, seconds = get_system_uptime()
    print(f"System Uptime: {days} days, {hours} hours, {minutes} minutes, {seconds} seconds")

if __name__ == "__main__":
    while True:
        display_uptime()
        time.sleep(60)  # Update every 60 seconds
