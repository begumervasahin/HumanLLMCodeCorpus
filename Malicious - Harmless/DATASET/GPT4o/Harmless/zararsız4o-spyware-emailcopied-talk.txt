import pyperclip
import re
import time

# Regular expression pattern to identify URLs
url_pattern = re.compile(r'(https?://[^\s]+)')

def extract_urls(text):
    """Extracts URLs from the given text."""
    return re.findall(url_pattern, text)

def main():
    recent_text = ""
    print("Monitoring clipboard for URLs... Press Ctrl+C to stop.")

    try:
        while True:
            current_text = pyperclip.paste()
            if current_text != recent_text:
                recent_text = current_text
                urls = extract_urls(recent_text)
                if urls:
                    print("URLs found:")
                    for url in urls:
                        print(url)
            time.sleep(1)
    except KeyboardInterrupt:
        print("\nStopped monitoring clipboard.")

if __name__ == "__main__":
    main()
