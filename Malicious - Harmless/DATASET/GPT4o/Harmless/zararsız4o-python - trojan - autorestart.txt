import subprocess
import sys
import os

def check_python_version():
    """Check the current Python version."""
    print("Checking Python version...")
    version = sys.version
    print(f"Current Python version: {version}")

def create_directory(directory_name):
    """Create a directory if it does not exist."""
    if not os.path.exists(directory_name):
        os.makedirs(directory_name)
        print(f"Directory '{directory_name}' created.")
    else:
        print(f"Directory '{directory_name}' already exists.")

def list_directory_contents(directory_name):
    """List the contents of the directory."""
    print(f"Listing contents of directory '{directory_name}':")
    contents = os.listdir(directory_name)
    for item in contents:
        print(item)

def main():
    check_python_version()
    
    directory_name = "test_directory"
    create_directory(directory_name)
    
    # Create a sample file in the directory
    sample_file_path = os.path.join(directory_name, "sample_file.txt")
    with open(sample_file_path, 'w') as sample_file:
        sample_file.write("This is a sample file.")
    
    list_directory_contents(directory_name)
    
    # Open the directory in the file explorer (cross-platform)
    if sys.platform == "win32":
        subprocess.Popen(f'explorer "{os.path.abspath(directory_name)}"')
    elif sys.platform == "darwin":
        subprocess.Popen(["open", os.path.abspath(directory_name)])
    else:
        subprocess.Popen(["xdg-open", os.path.abspath(directory_name)])

if __name__ == "__main__":
    main()
