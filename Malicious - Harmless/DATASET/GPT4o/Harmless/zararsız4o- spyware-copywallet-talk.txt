import pyperclip
import time
import re

# Initialize variables
clipboard_history = []

def clean_text(text):
    # Remove extra whitespace and newlines
    text = re.sub(r'\s+', ' ', text).strip()
    return text

def main():
    print("Starting clipboard manager...")
    last_text = ""

    while True:
        # Get the current clipboard text
        current_text = pyperclip.paste()

        # If the text is new and non-empty
        if current_text != last_text and current_text:
            cleaned_text = clean_text(current_text)

            # Print the cleaned text
            print(f"New clipboard text: {cleaned_text}")

            # Add to history
            clipboard_history.append(cleaned_text)

            # Update last_text
            last_text = current_text

        # Wait a bit before checking again
        time.sleep(1)

if __name__ == "__main__":
    try:
        main()
    except KeyboardInterrupt:
        print("\nClipboard manager stopped.")
        print("Clipboard history:")
        for idx, text in enumerate(clipboard_history):
            print(f"{idx + 1}: {text}")
