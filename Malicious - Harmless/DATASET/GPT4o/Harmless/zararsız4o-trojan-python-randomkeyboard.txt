import keyboard
import random
import string
import pyperclip  # Additional library to handle clipboard operations

# Function to generate a random password
def generate_password(length=12):
    characters = string.ascii_letters + string.digits + string.punctuation
    password = ''.join(random.choice(characters) for i in range(length))
    return password

# Function to copy the generated password to the clipboard
def copy_to_clipboard(password):
    pyperclip.copy(password)
    print(f"Generated Password: {password}")
    print("Password copied to clipboard!")

# Function to handle the key press event
def on_key_event(e):
    if e.name == 'p':  # Change 'p' to any key you prefer
        password = generate_password()
        copy_to_clipboard(password)

# Register the key press event
keyboard.on_press(on_key_event)

print("Press 'p' to generate a random password and copy it to the clipboard.")
keyboard.wait('esc')  # Program will run until 'esc' key is pressed
