import os
import glob
import platform

def list_files(pattern):
    """List all files in the current directory matching the given pattern."""
    return glob.glob(pattern)

def print_os_info():
    """Print the current operating system information."""
    print("Operating System Information:")
    print(f"System: {platform.system()}")
    print(f"Node: {platform.node()}")
    print(f"Release: {platform.release()}")
    print(f"Version: {platform.version()}")
    print(f"Machine: {platform.machine()}")
    print(f"Processor: {platform.processor()}")

def main():
    """Main function to list files and print OS info."""
    # Specify the pattern to match files (e.g., '*.txt' for all text files)
    pattern = '*.txt'
    
    # List all files matching the pattern
    files = list_files(pattern)
    
    # Print the matched files
    print(f"Files matching the pattern '{pattern}':")
    for file in files:
        print(file)
    
    # Print the operating system information
    print_os_info()

if __name__ == "__main__":
    main()
