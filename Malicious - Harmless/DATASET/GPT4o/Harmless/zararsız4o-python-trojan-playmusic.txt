import requests
from pydub import AudioSegment
from pydub.playback import play

# URL of the audio file to download
audio_url = 'https://www.example.com/path/to/audio/file.mp3'

# Download the audio file
response = requests.get(audio_url)
audio_data = response.content

# Save the downloaded audio file locally
with open('downloaded_audio.mp3', 'wb') as f:
    f.write(audio_data)

# Load the audio file using pydub
audio = AudioSegment.from_mp3('downloaded_audio.mp3')

# Convert the audio file to WAV format
audio.export('converted_audio.wav', format='wav')

# Play the converted audio file
play(AudioSegment.from_wav('converted_audio.wav'))

print("Audio has been downloaded, converted, and played successfully!")
