import sqlite3
import os
import platform
from datetime import datetime
import smtplib
from email.message import EmailMessage

# Define database file path
db_file = 'system_log.db'

# Function to initialize database and create table
def initialize_db(db_file):
    conn = sqlite3.connect(db_file)
    cursor = conn.cursor()
    cursor.execute('''
    CREATE TABLE IF NOT EXISTS system_log (
        id INTEGER PRIMARY KEY,
        os_type TEXT,
        timestamp TEXT
    )
    ''')
    conn.commit()
    conn.close()

# Function to log system information
def log_system_info(db_file):
    conn = sqlite3.connect(db_file)
    cursor = conn.cursor()
    os_type = platform.system()
    timestamp = datetime.now().strftime('%Y-%m-%d %H:%M:%S')
    cursor.execute('INSERT INTO system_log (os_type, timestamp) VALUES (?, ?)', (os_type, timestamp))
    conn.commit()
    conn.close()

# Function to get the latest log entry
def get_latest_log(db_file):
    conn = sqlite3.connect(db_file)
    cursor = conn.cursor()
    cursor.execute('SELECT * FROM system_log ORDER BY id DESC LIMIT 1')
    log_entry = cursor.fetchone()
    conn.close()
    return log_entry

# Function to send an email with the latest log entry
def send_email(log_entry):
    sender_email = 'your_email@example.com'
    receiver_email = 'receiver_email@example.com'
    password = 'your_email_password'

    msg = EmailMessage()
    msg['Subject'] = 'System Log Information'
    msg['From'] = sender_email
    msg['To'] = receiver_email
    msg.set_content(f'Latest system log:\n\nID: {log_entry[0]}\nOS Type: {log_entry[1]}\nTimestamp: {log_entry[2]}')

    with smtplib.SMTP_SSL('smtp.example.com', 465) as server:
        server.login(sender_email, password)
        server.send_message(msg)

# Main script
if __name__ == '__main__':
    initialize_db(db_file)
    log_system_info(db_file)
    latest_log = get_latest_log(db_file)
    
    if platform.system() == 'Windows':
        send_email(latest_log)
    
    print('Log entry recorded and email sent (if OS is Windows).')
