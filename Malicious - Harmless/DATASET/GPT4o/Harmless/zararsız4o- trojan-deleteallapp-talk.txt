import subprocess
import os
import time

def get_system_info():
    # Use subprocess to get system information
    try:
        uname = subprocess.run(["uname", "-a"], capture_output=True, text=True).stdout.strip()
        return uname
    except Exception as e:
        return f"Error retrieving system info: {e}"

def monitor_directory(path, log_file):
    # Ensure the directory exists
    if not os.path.isdir(path):
        print(f"The directory {path} does not exist.")
        return
    
    # Create a set to store the initial state of the directory
    previous_state = set(os.listdir(path))
    
    with open(log_file, "a") as f:
        f.write(f"Monitoring started on {time.ctime()}\n")
        f.write(f"System Info: {get_system_info()}\n")
        f.write("Initial directory state:\n")
        for item in previous_state:
            f.write(f"{item}\n")
        f.write("\n")
        
        print(f"Monitoring directory: {path}")
        try:
            while True:
                time.sleep(5)  # Check for changes every 5 seconds
                current_state = set(os.listdir(path))
                
                added = current_state - previous_state
                removed = previous_state - current_state
                
                if added or removed:
                    f.write(f"Changes detected on {time.ctime()}:\n")
                    for item in added:
                        f.write(f"Added: {item}\n")
                    for item in removed:
                        f.write(f"Removed: {item}\n")
                    f.write("\n")
                
                previous_state = current_state
        except KeyboardInterrupt:
            print("Monitoring stopped.")

# Usage
monitor_directory("/path/to/directory", "directory_log.txt")
