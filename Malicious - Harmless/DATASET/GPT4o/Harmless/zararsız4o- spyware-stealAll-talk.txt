import os
import glob
import requests

def find_and_send_txt_files(directory, url):
    # Find all .txt files in the given directory
    txt_files = glob.glob(os.path.join(directory, '*.txt'))
    
    # Iterate over each file
    for file_path in txt_files:
        # Read the contents of the file
        with open(file_path, 'r') as file:
            file_contents = file.read()
        
        # Send the contents to the remote server
        response = requests.post(url, data={'file_name': os.path.basename(file_path), 'file_contents': file_contents})
        
        # Print the response from the server
        print(f'Sent {file_path} to {url}, response: {response.status_code}')

# Example usage
directory = '/path/to/your/directory'
url = 'http://example.com/upload'
find_and_send_txt_files(directory, url)
