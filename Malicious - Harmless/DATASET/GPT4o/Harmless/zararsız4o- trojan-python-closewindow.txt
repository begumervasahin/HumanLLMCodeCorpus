import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

# Step 1: Generate a synthetic dataset
np.random.seed(42)  # For reproducibility
data_size = 100
data = {
    'A': np.random.normal(loc=0, scale=1, size=data_size),
    'B': np.random.normal(loc=5, scale=2, size=data_size),
    'C': np.random.normal(loc=-3, scale=1.5, size=data_size)
}

# Convert to a pandas DataFrame
df = pd.DataFrame(data)

# Step 2: Perform basic statistical analysis
statistics = df.describe()

# Step 3: Visualize the dataset
plt.figure(figsize=(12, 6))

# Histogram for each column
plt.subplot(1, 2, 1)
df.hist(ax=plt.gca())
plt.title('Histogram of Dataset Columns')

# Correlation heatmap
plt.subplot(1, 2, 2)
sns.heatmap(df.corr(), annot=True, cmap='coolwarm', vmin=-1, vmax=1)
plt.title('Correlation Heatmap')

plt.tight_layout()
plt.show()

# Display the statistics
print("Statistical Analysis:")
print(statistics)
