import os
import uuid
import requests
import ctypes
import subprocess

# Generate a unique identifier
unique_id = uuid.uuid4()
print(f"Generated UUID: {unique_id}")

# Make a request to a public API (for example, getting a random joke)
response = requests.get('https://official-joke-api.appspot.com/random_joke')
joke = response.json()
print(f"Fetched joke: {joke['setup']} - {joke['punchline']}")

# Save the joke to a file
file_path = os.path.join(os.getcwd(), f"joke_{unique_id}.txt")
with open(file_path, 'w') as file:
    file.write(f"{joke['setup']} - {joke['punchline']}")
print(f"Joke saved to {file_path}")

# Display a message box with the joke
ctypes.windll.user32.MessageBoxW(0, f"{joke['setup']} - {joke['punchline']}", "Random Joke", 1)

# Open the file in the default text editor
subprocess.run(['notepad.exe', file_path])
