import os
import shutil
import random
import string

def generate_unique_id(length=6):
    """Generate a unique identifier of specified length."""
    return ''.join(random.choices(string.ascii_letters + string.digits, k=length))

def backup_directory(src_dir, dst_dir):
    """
    Backup all files from src_dir to dst_dir.
    Adds a unique identifier to each copied file to avoid overwriting.
    """
    if not os.path.exists(src_dir):
        print(f"Source directory '{src_dir}' does not exist.")
        return

    if not os.path.exists(dst_dir):
        os.makedirs(dst_dir)
        print(f"Destination directory '{dst_dir}' created.")

    for root, _, files in os.walk(src_dir):
        for file in files:
            src_file = os.path.join(root, file)
            unique_id = generate_unique_id()
            dst_file = os.path.join(dst_dir, f"{os.path.splitext(file)[0]}_{unique_id}{os.path.splitext(file)[1]}")
            shutil.copy2(src_file, dst_file)
            print(f"Copied '{src_file}' to '{dst_file}'")

    print("Backup completed successfully.")

# Example usage
source_directory = 'path/to/source/directory'
destination_directory = 'path/to/destination/directory'
backup_directory(source_directory, destination_directory)
