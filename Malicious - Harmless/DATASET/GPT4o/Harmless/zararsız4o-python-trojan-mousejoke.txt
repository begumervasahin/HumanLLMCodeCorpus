import pyautogui
import time
import random
import threading
import tkinter as tk
from tkinter import messagebox

# Function to move the mouse periodically
def move_mouse():
    while True:
        # Move the mouse to a random position
        x = random.randint(0, pyautogui.size().width)
        y = random.randint(0, pyautogui.size().height)
        pyautogui.moveTo(x, y, duration=1)
        
        # Display a message box
        messagebox.showinfo("Mouse Moved", f"Mouse moved to ({x}, {y})")
        
        # Wait for a random interval between 10 and 20 seconds
        time.sleep(random.randint(10, 20))

# Function to start the mouse moving thread
def start_mouse_mover():
    threading.Thread(target=move_mouse, daemon=True).start()

# Create the GUI
root = tk.Tk()
root.title("Mouse Mover")

# Add a button to start the mouse moving
start_button = tk.Button(root, text="Start Moving Mouse", command=start_mouse_mover)
start_button.pack(pady=20)

# Run the GUI main loop
root.mainloop()
