import os
from moviepy.editor import VideoFileClip, concatenate_videoclips

# Function to create a video montage from all video files in a directory
def create_video_montage(input_directory, output_file):
    # List to hold all video clips
    video_clips = []

    # Loop through all files in the input directory
    for filename in os.listdir(input_directory):
        if filename.endswith(".mp4") or filename.endswith(".mov"):
            # Create a VideoFileClip object for each video file
            video_path = os.path.join(input_directory, filename)
            video_clips.append(VideoFileClip(video_path))

    # Concatenate all video clips into one
    final_clip = concatenate_videoclips(video_clips, method="compose")

    # Write the final clip to a file
    final_clip.write_videofile(output_file, codec="libx264")

# Specify the input directory and output file
input_directory = "path/to/your/video/files"
output_file = "path/to/save/final_video.mp4"

# Call the function to create the video montage
create_video_montage(input_directory, output_file)
