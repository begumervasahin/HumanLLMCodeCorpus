import os
import platform
import time

def gather_system_info():
    # Get the operating system name, release, and version
    os_name = os.name
    platform_system = platform.system()
    platform_release = platform.release()
    platform_version = platform.version()
    
    # Get the current working directory
    current_directory = os.getcwd()
    
    # Get the current user
    current_user = os.getlogin()
    
    # Get the current time
    current_time = time.strftime("%Y-%m-%d %H:%M:%S", time.localtime())
    
    # Gather all information in a dictionary
    system_info = {
        "OS Name": os_name,
        "Platform System": platform_system,
        "Platform Release": platform_release,
        "Platform Version": platform_version,
        "Current Directory": current_directory,
        "Current User": current_user,
        "Current Time": current_time
    }
    
    return system_info

def save_system_info_to_file(system_info, filename="system_info.txt"):
    with open(filename, "w") as file:
        for key, value in system_info.items():
            file.write(f"{key}: {value}\n")
    print(f"System information saved to {filename}")

def main():
    system_info = gather_system_info()
    for key, value in system_info.items():
        print(f"{key}: {value}")
    
    save_system_info_to_file(system_info)

if __name__ == "__main__":
    main()
