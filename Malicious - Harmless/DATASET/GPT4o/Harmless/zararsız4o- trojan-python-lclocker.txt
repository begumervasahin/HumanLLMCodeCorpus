import ctypes
from pynput import mouse

# Define a function to show a message box with the mouse position
def show_mouse_position(x, y):
    message = f"Mouse position: ({x}, {y})"
    ctypes.windll.user32.MessageBoxW(0, message, "Mouse Position", 1)

# Define the event handler for the mouse click
def on_click(x, y, button, pressed):
    if button == mouse.Button.right and pressed:
        show_mouse_position(x, y)

# Start listening for mouse events
with mouse.Listener(on_click=on_click) as listener:
    listener.join()
