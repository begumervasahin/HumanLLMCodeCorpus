import screen_brightness_control as sbc
import time

def gradual_brightness_change(start_brightness, end_brightness, duration, steps=100):
    step_duration = duration / steps
    brightness_step = (end_brightness - start_brightness) / steps
    for step in range(steps + 1):
        current_brightness = start_brightness + (brightness_step * step)
        sbc.set_brightness(current_brightness)
        time.sleep(step_duration)

def main():
    evening_duration = 3600  # 1 hour
    morning_duration = 3600  # 1 hour
    evening_start_brightness = 100  # Maximum brightness
    evening_end_brightness = 20  # Dimmed brightness
    morning_start_brightness = 20  # Dimmed brightness
    morning_end_brightness = 100  # Maximum brightness

    # Gradually dim the brightness in the evening
    print("Dimming the brightness for the evening...")
    gradual_brightness_change(evening_start_brightness, evening_end_brightness, evening_duration)

    # Wait until morning (let's assume 8 hours of sleep)
    time.sleep(8 * 3600)

    # Gradually increase the brightness in the morning
    print("Increasing the brightness for the morning...")
    gradual_brightness_change(morning_start_brightness, morning_end_brightness, morning_duration)

if __name__ == "__main__":
    main()
