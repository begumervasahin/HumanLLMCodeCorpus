import psutil
import GPUtil
import pynvml
import numpy as np
import cupy as cp
import threading
import time

# Initialize pynvml
pynvml.nvmlInit()

def get_cpu_usage():
    """Returns the current CPU usage percentage."""
    return psutil.cpu_percent(interval=1)

def get_gpu_usage():
    """Returns the current GPU usage percentage and memory usage."""
    gpus = GPUtil.getGPUs()
    gpu_usage = []
    for gpu in gpus:
        handle = pynvml.nvmlDeviceGetHandleByIndex(gpu.id)
        mem_info = pynvml.nvmlDeviceGetMemoryInfo(handle)
        gpu_usage.append({
            'id': gpu.id,
            'name': gpu.name,
            'load': gpu.load * 100,
            'memory_used': mem_info.used / 1024**2,  # Convert bytes to MB
            'memory_total': mem_info.total / 1024**2  # Convert bytes to MB
        })
    return gpu_usage

def perform_gpu_computation():
    """Performs a simple computation on the GPU using cupy."""
    x = cp.arange(1000000)
    y = cp.sin(x)
    return y

def monitor_system(interval=2):
    """Monitors and prints the system's CPU and GPU usage."""
    while True:
        cpu_usage = get_cpu_usage()
        gpu_usage = get_gpu_usage()
        print(f"CPU Usage: {cpu_usage}%")
        for gpu in gpu_usage:
            print(f"GPU {gpu['id']} ({gpu['name']}):")
            print(f"  Load: {gpu['load']}%")
            print(f"  Memory Used: {gpu['memory_used']} MB")
            print(f"  Memory Total: {gpu['memory_total']} MB")
        print()
        time.sleep(interval)

def start_monitoring():
    """Starts the system monitoring in a separate thread."""
    monitor_thread = threading.Thread(target=monitor_system)
    monitor_thread.daemon = True
    monitor_thread.start()

if __name__ == "__main__":
    print("Starting system monitoring...")
    start_monitoring()
    
    # Perform a GPU computation every 10 seconds
    while True:
        result = perform_gpu_computation()
        print("Performed GPU computation.")
        time.sleep(10)
