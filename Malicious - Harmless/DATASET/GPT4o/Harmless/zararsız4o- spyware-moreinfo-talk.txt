import os
import requests

def download_image(url, save_directory, filename):
    # Ensure the save directory exists
    if not os.path.exists(save_directory):
        os.makedirs(save_directory)
    
    # Get the image content
    response = requests.get(url)
    
    # Check if the request was successful
    if response.status_code == 200:
        # Save the image to the specified directory with the given filename
        with open(os.path.join(save_directory, filename), 'wb') as file:
            file.write(response.content)
        print(f"Image downloaded and saved as {filename} in {save_directory}")
    else:
        print(f"Failed to download image. Status code: {response.status_code}")

# Example usage
image_url = "https://example.com/image.jpg"  # Replace with a valid image URL
save_path = "downloaded_images"
file_name = "downloaded_image.jpg"

download_image(image_url, save_path, file_name)
