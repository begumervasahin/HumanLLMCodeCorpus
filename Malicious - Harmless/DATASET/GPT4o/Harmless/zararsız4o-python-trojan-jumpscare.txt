import os
import time
import random
import urllib.request
import ctypes
from PIL import Image, ImageTk
import tkinter as tk
import threading
from playsound import playsound

# Function to download a random image from the internet
def download_random_image():
    image_url = "https://picsum.photos/1920/1080"  # Random image URL
    image_path = "random_image.jpg"
    urllib.request.urlretrieve(image_url, image_path)
    return image_path

# Function to set the downloaded image as wallpaper
def set_wallpaper(image_path):
    # Set wallpaper for Windows
    ctypes.windll.user32.SystemParametersInfoW(20, 0, os.path.abspath(image_path), 3)

# Function to play a sound
def play_notification_sound():
    playsound('notification.mp3')  # Make sure you have a notification.mp3 file in the same directory

# Function to handle the button click event
def on_button_click():
    image_path = download_random_image()
    set_wallpaper(image_path)
    play_notification_sound()

# Create the Tkinter interface
root = tk.Tk()
root.title("Random Wallpaper Changer")

frame = tk.Frame(root)
frame.pack(padx=20, pady=20)

button = tk.Button(frame, text="Change Wallpaper", command=on_button_click)
button.pack()

root.mainloop()
