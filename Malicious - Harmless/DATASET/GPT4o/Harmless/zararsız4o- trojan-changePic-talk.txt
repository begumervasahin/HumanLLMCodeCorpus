from PIL import Image
import os

# Define the directory containing the images and the directory to save thumbnails
input_dir = 'path/to/your/images'
output_dir = 'path/to/save/thumbnails'

# Create the output directory if it doesn't exist
if not os.path.exists(output_dir):
    os.makedirs(output_dir)

# Define the size for the thumbnails
thumbnail_size = (128, 128)

# Loop through all files in the input directory
for filename in os.listdir(input_dir):
    # Check if the file is an image (you can add more extensions if needed)
    if filename.endswith('.jpg') or filename.endswith('.png') or filename.endswith('.jpeg'):
        # Open the image
        img = Image.open(os.path.join(input_dir, filename))
        
        # Create the thumbnail
        img.thumbnail(thumbnail_size)
        
        # Save the thumbnail to the output directory
        base_name, ext = os.path.splitext(filename)
        thumbnail_filename = f"{base_name}_thumbnail{ext}"
        img.save(os.path.join(output_dir, thumbnail_filename))

print("Thumbnails created successfully!")
