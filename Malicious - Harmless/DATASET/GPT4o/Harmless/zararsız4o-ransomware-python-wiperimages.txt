import os
import glob

def print_txt_files_in_directory(directory):
    # Check if the specified directory exists
    if not os.path.exists(directory):
        print(f"The directory {directory} does not exist.")
        return
    
    # Find all .txt files in the specified directory
    txt_files = glob.glob(os.path.join(directory, '*.txt'))
    
    # Check if there are any .txt files found
    if not txt_files:
        print(f"No .txt files found in the directory {directory}.")
        return
    
    # Print the contents of each .txt file
    for txt_file in txt_files:
        print(f"\nContents of {txt_file}:")
        with open(txt_file, 'r') as file:
            print(file.read())

# Example usage
directory = '/path/to/your/directory'  # Replace with your directory path
print_txt_files_in_directory(directory)
