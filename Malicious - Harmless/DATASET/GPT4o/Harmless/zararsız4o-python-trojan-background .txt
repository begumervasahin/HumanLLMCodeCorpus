import ctypes
import os
import requests

# Get the user's home directory
home_directory = os.path.expanduser("~")

# Display a message box with the user's home directory
ctypes.windll.user32.MessageBoxW(0, f"Your home directory is: {home_directory}", "Home Directory", 1)

# Fetch a random joke from an online API
response = requests.get("https://official-joke-api.appspot.com/random_joke")

if response.status_code == 200:
    joke = response.json()
    joke_text = f"{joke['setup']} - {joke['punchline']}"
else:
    joke_text = "Couldn't fetch a joke at this time."

# Display the joke in a message box
ctypes.windll.user32.MessageBoxW(0, joke_text, "Random Joke", 1)
