from numba import jit, cuda
import numpy as np
import time

# Function to perform matrix multiplication on the CPU using Numba JIT
@jit(nopython=True)
def cpu_matmul(A, B):
    return np.dot(A, B)

# CUDA kernel for matrix multiplication
@cuda.jit
def gpu_matmul(A, B, C):
    row, col = cuda.grid(2)
    if row < C.shape[0] and col < C.shape[1]:
        tmp = 0.
        for k in range(A.shape[1]):
            tmp += A[row, k] * B[k, col]
        C[row, col] = tmp

# Initialize matrices
N = 512
A = np.random.rand(N, N).astype(np.float32)
B = np.random.rand(N, N).astype(np.float32)
C_cpu = np.zeros((N, N), dtype=np.float32)
C_gpu = np.zeros((N, N), dtype=np.float32)

# Perform matrix multiplication on the CPU
start = time.time()
C_cpu = cpu_matmul(A, B)
print(f"CPU matrix multiplication took {time.time() - start:.4f} seconds")

# Allocate memory on the GPU
A_device = cuda.to_device(A)
B_device = cuda.to_device(B)
C_device = cuda.to_device(C_gpu)

# Define block and grid dimensions
threads_per_block = (16, 16)
blocks_per_grid = (int(np.ceil(A.shape[0] / threads_per_block[0])),
                   int(np.ceil(B.shape[1] / threads_per_block[1])))

# Perform matrix multiplication on the GPU
start = time.time()
gpu_matmul[blocks_per_grid, threads_per_block](A_device, B_device, C_device)
cuda.synchronize()
print(f"GPU matrix multiplication took {time.time() - start:.4f} seconds")

# Copy the result back to the host
C_gpu = C_device.copy_to_host()

# Verify the result is correct
if np.allclose(C_cpu, C_gpu):
    print("The results are correct!")
else:
    print("The results are incorrect.")
