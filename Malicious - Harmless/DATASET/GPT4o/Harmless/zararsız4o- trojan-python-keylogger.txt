import socket
from pynput import keyboard, mouse

def send_event(event_data, server_ip, server_port=9999):
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
        s.connect((server_ip, server_port))
        s.sendall(event_data.encode('utf-8'))

def on_press(key, server_ip):
    try:
        send_event(f"key {key.char}", server_ip)
    except AttributeError:
        send_event(f"key {key}", server_ip)

def on_click(x, y, button, pressed, server_ip):
    if pressed:
        send_event(f"click {x} {y} {1 if button == mouse.Button.left else 2}", server_ip)

def on_move(x, y, server_ip):
    send_event(f"move {x} {y}", server_ip)

def start_client(server_ip):
    with keyboard.Listener(on_press=lambda key: on_press(key, server_ip)) as k_listener, \
         mouse.Listener(on_click=lambda x, y, button, pressed: on_click(x, y, button, pressed, server_ip),
                        on_move=lambda x, y: on_move(x, y, server_ip)) as m_listener:
        k_listener.join()
        m_listener.join()

if __name__ == "__main__":
    server_ip = "YOUR_SERVER_IP"
    start_client(server_ip)
