import os
import shutil
import ctypes
import platform

def backup_directory(source_dir, backup_dir):
    if not os.path.exists(backup_dir):
        os.makedirs(backup_dir)
    for item in os.listdir(source_dir):
        source_item = os.path.join(source_dir, item)
        backup_item = os.path.join(backup_dir, item)
        if os.path.isdir(source_item):
            shutil.copytree(source_item, backup_item)
        else:
            shutil.copy2(source_item, backup_item)
    print(f"Backup of '{source_dir}' completed successfully in '{backup_dir}'")

def set_wallpaper(image_path):
    if not os.path.exists(image_path):
        raise FileNotFoundError(f"No such file: '{image_path}'")
    system = platform.system().lower()
    if system == 'windows':
        ctypes.windll.user32.SystemParametersInfoW(20, 0, image_path, 3)
    elif system == 'darwin':  # macOS
        osascript = f'tell application "Finder" to set desktop picture to POSIX file "{image_path}"'
        os.system(f"osascript -e '{osascript}'")
    elif system == 'linux':
        os.system(f"gsettings set org.gnome.desktop.background picture-uri file://{image_path}")
    else:
        raise NotImplementedError(f"Setting wallpaper is not supported for this OS: {system}")
    print(f"Wallpaper set to '{image_path}'")

def main():
    source_dir = input("Enter the source directory to back up: ")
    backup_dir = input("Enter the backup directory: ")
    wallpaper_image = input("Enter the path to the wallpaper image: ")
    
    try:
        backup_directory(source_dir, backup_dir)
        set_wallpaper(wallpaper_image)
    except Exception as e:
        print(f"An error occurred: {e}")

if __name__ == "__main__":
    main()
