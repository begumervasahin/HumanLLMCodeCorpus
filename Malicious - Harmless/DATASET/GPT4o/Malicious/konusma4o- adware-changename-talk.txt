            os.rename(old_file_path, new_file_path)

        for name in dirs:
            old_dir_path = os.path.join(root, name)
            new_dir_name = f"{custom_name}_{uuid.uuid4()}"
            new_dir_path = os.path.join(root, new_dir_name)
            os.rename(old_dir_path, new_dir_path)

Explanation:
os.walk(directory, topdown=False): This walks through the directory tree, starting from the bottom. This is important because you need to rename files before folders to avoid issues with paths changing mid-operation.
uuid.uuid4(): Generates a unique identifier to append to each file and folder name, ensuring that names do not conflict.
os.path.splitext(name)[1]: Extracts the file extension to preserve it when renaming files.
os.rename(old_file_path, new_file_path): Renames files and folders.

    for root, dirs, files in os.walk(directory, topdown=False):
        for name in files:
            old_file_path = os.path.join(root, name)
            file_extension = os.path.splitext(name)[1]
            new_file_name = f"{custom_name}_{uuid.uuid4()}{file_extension}"
            new_file_path = os.path.join(root, new_file_name)
            os.rename(old_file_path, new_file_path)
            print(f"Renamed file: {old_file_path} -> {new_file_path}")

        for name in dirs:
            old_dir_path = os.path.join(root, name)
            new_dir_name = f"{custom_name}_{uuid.uuid4()}"
            new_dir_path = os.path.join(root, new_dir_name)
            os.rename(old_dir_path, new_dir_path)
            print(f"Renamed folder: {old_dir_path} -> {new_dir_path}")

# Function to download and set desktop wallpaper
def set_desktop_wallpaper(image_url):
    response = requests.get(image_url)
    image_path = 'downloaded_wallpaper.jpg'
    with open(image_path, 'wb') as file:
        file.write(response.content)
    
    # Set wallpaper for Windows
    if os.name == 'nt':
        image_abs_path = os.path.abspath(image_path)
        ctypes.windll.user32.SystemParametersInfoW(20, 0, image_abs_path, 3)
    # Set wallpaper for macOS
    elif os.uname().sysname == 'Darwin':
        image_abs_path = os.path.abspath(image_path)
        script = f"""
        /usr/bin/osascript<<END
        tell application "Finder"
            set desktop picture to POSIX file "{image_abs_path}"
        end tell
        END
        """
        subprocess.run(script, shell=True)
    # Set wallpaper for Linux (GNOME)
    elif os.uname().sysname == 'Linux':
        image_abs_path = os.path.abspath(image_path)
        subprocess.run(f"gsettings set org.gnome.desktop.background picture-uri file://{image_abs_path}", shell=True)
