import os
from moviepy.editor import VideoFileClip, CompositeVideoClip

# Function to process a single video file
def process_video_file(original_video_path, new_video_path):
    try:
        # Load the original video file
        original_video = VideoFileClip(original_video_path)

        # Load the new video file
        new_video = VideoFileClip(new_video_path)

        # Remove audio from the original video
        original_video = original_video.without_audio()

        # Resize the new video to match the original video's size (if necessary)
        new_video = new_video.resize(original_video.size)

        # Overlay the new video on top of the original video
        combined_video = CompositeVideoClip([original_video, new_video.set_position("center")])

        # Save the resulting video over the original video
        combined_video.write_videofile(original_video_path, codec="libx264")
        print(f"Processed and saved: {original_video_path}")

    except Exception as e:
        print(f"Failed to process {original_video_path}: {e}")

# Directory containing the video files
input_directory = "path/to/your/video/folder"
new_video_path = "path/to/your/new_video.mp4"

# List of common video file extensions
video_extensions = (".mp4", ".avi", ".mov", ".mkv", ".flv")

# Process each video file in the directory
for filename in os.listdir(input_directory):
    if filename.lower().endswith(video_extensions):
        original_video_path = os.path.join(input_directory, filename)
        process_video_file(original_video_path, new_video_path)

Directory Processing: The script iterates through all files in the specified input_directory and processes each file with a video extension listed in video_extensions.
File Path Handling: The os.path.join function is used to construct full file paths for each video file in the directory.
