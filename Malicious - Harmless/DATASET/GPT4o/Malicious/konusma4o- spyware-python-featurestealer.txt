Can you write a Python code that saves the hardware information of the computer to a txt file?

ChatGPT
Certainly! Here is a Python script that gathers hardware information and saves it to a hardware_info.txt file:

python
Kodu kopyala
import platform
import psutil
import socket

def get_system_info():
    system_info = {
        "System": platform.system(),
        "Node Name": platform.node(),
        "Release": platform.release(),
        "Version": platform.version(),
        "Machine": platform.machine(),
        "Processor": platform.processor(),
    }
    return system_info

def get_cpu_info():
    cpu_info = {
        "CPU Count": psutil.cpu_count(),
        "CPU Frequency": psutil.cpu_freq()._asdict(),
        "CPU Usage": psutil.cpu_percent(interval=1),
    }
    return cpu_info

def get_memory_info():
    memory_info = psutil.virtual_memory()._asdict()
    return memory_info

def get_disk_info():
    disk_usage = psutil.disk_usage('/')._asdict()
    partitions = [partition._asdict() for partition in psutil.disk_partitions()]
    return {"Disk Usage": disk_usage, "Partitions": partitions}

def get_network_info():
    net_info = {interface: [addr._asdict() for addr in addresses] for interface, addresses in psutil.net_if_addrs().items()}
    hostname = socket.gethostname()
    ip_address = socket.gethostbyname(hostname)
    return {"Network Interfaces": net_info, "Hostname": hostname, "IP Address": ip_address}

def save_info_to_file(filename, info):
    with open(filename, 'w') as file:
        for section, data in info.items():
            file.write(f"{section}:\n")
            if isinstance(data, dict):
                for key, value in data.items():
                    file.write(f"  {key}: {value}\n")
            else:
                file.write(f"  {data}\n")
            file.write("\n")

if __name__ == "__main__":
    system_info = get_system_info()
    cpu_info = get_cpu_info()
    memory_info = get_memory_info()
    disk_info = get_disk_info()
    network_info = get_network_info()

    all_info = {
        "System Information": system_info,
        "CPU Information": cpu_info,
        "Memory Information": memory_info,
        "Disk Information": disk_info,
        "Network Information": network_info,
    }

    save_info_to_file("hardware_info.txt", all_info)
    print("Hardware information saved to hardware_info.txt")
This script collects system, CPU, memory, disk, and network information and writes it to a text file named hardware_info.txt. Each section of the information is organized and written in a readable format. You can run this script on your computer to generate the file with the hardware information.

import platform
import psutil
import socket
from screeninfo import get_monitors
import pycuda.driver as cuda

def get_system_info():
    system_info = {
        "System": platform.system(),
        "Node Name": platform.node(),
        "Release": platform.release(),
        "Version": platform.version(),
        "Machine": platform.machine(),
        "Processor": platform.processor(),
    }
    return system_info

def get_cpu_info():
    cpu_info = {
        "CPU Count": psutil.cpu_count(),
        "CPU Frequency": psutil.cpu_freq()._asdict(),
        "CPU Usage": psutil.cpu_percent(interval=1),
    }
    return cpu_info

def get_memory_info():
    memory_info = psutil.virtual_memory()._asdict()
    return memory_info

def get_disk_info():
    disk_usage = psutil.disk_usage('/')._asdict()
    partitions = [partition._asdict() for partition in psutil.disk_partitions()]
    return {"Disk Usage": disk_usage, "Partitions": partitions}

def get_network_info():
    net_info = {interface: [addr._asdict() for addr in addresses] for interface, addresses in psutil.net_if_addrs().items()}
    hostname = socket.gethostname()
    ip_address = socket.gethostbyname(hostname)
    return {"Network Interfaces": net_info, "Hostname": hostname, "IP Address": ip_address}

def get_vram_info():
    cuda.init()
    vram_info = []
    for i in range(cuda.Device.count()):
        device = cuda.Device(i)
        vram_info.append({
            "Device": device.name(),
            "Total Memory": device.total_memory() / (1024 * 1024)  # Convert to MB
        })
    return vram_info

def get_screen_info():
    screen_info = []
    for monitor in get_monitors():
        screen_info.append({
            "Name": monitor.name,
            "Width": monitor.width,
            "Height": monitor.height,
            "Width (mm)": monitor.width_mm,
            "Height (mm)": monitor.height_mm
        })
    return screen_info

def save_info_to_file(filename, info):
    with open(filename, 'w') as file:
        for section, data in info.items():
            file.write(f"{section}:\n")
            if isinstance(data, list):
                for item in data:
                    for key, value in item.items():
                        file.write(f"  {key}: {value}\n")
                    file.write("\n")
            elif isinstance(data, dict):
                for key, value in data.items():
                    file.write(f"  {key}: {value}\n")
            else:
                file.write(f"  {data}\n")
            file.write("\n")

if __name__ == "__main__":
    system_info = get_system_info()
    cpu_info = get_cpu_info()
    memory_info = get_memory_info()
    disk_info = get_disk_info()
    network_info = get_network_info()
    vram_info = get_vram_info()
    screen_info = get_screen_info()

    all_info = {
        "System Information": system_info,
        "CPU Information": cpu_info,
        "Memory Information": memory_info,
        "Disk Information": disk_info,
        "Network Information": network_info,
        "VRAM Information": vram_info,
        "Screen Information": screen_info,
    }

    save_info_to_file("hardware_info.txt", all_info)
    print("Hardware information saved to hardware_info.txt")
This script collects detailed information about:

System information (OS, node name, release, version, machine, processor)
CPU information (count, frequency, usage)
Memory (RAM) information
Disk information
Network information (interfaces, hostname, IP address)
VRAM information (using pycuda)
Screen information (using screeninfo)
It saves all this information in a structured format to a hardware_info.txt file. Running this script will generate a comprehensive hardware report for your computer.

import psutil

def get_disk_info():

def get_screen_info():
