Sure, I can help you write a Python script that will search for all .txt files on your computer and move them to specified directories on disk C and D. This script will use the os and shutil modules to achieve this.

Here's an example script that searches for .txt files and moves them to specified directories:

python
Kodu kopyala
import os
import shutil

def find_txt_files(root_directory):
    txt_files = []
    for dirpath, _, filenames in os.walk(root_directory):
        for filename in filenames:
            if filename.endswith('.txt'):
                txt_files.append(os.path.join(dirpath, filename))
    return txt_files

def move_files(files, destination_directory):
    if not os.path.exists(destination_directory):
        os.makedirs(destination_directory)
    for file in files:
        shutil.move(file, destination_directory)

def main():
    # Directories to search for .txt files
    search_directories = ['C:\\', 'D:\\']
    
    # Destination directories for C and D drives
    destination_directory_C = 'C:\\destination_directory'
    destination_directory_D = 'D:\\destination_directory'
    
    for directory in search_directories:
        txt_files = find_txt_files(directory)
        
        if directory.startswith('C:'):
            move_files(txt_files, destination_directory_C)
        elif directory.startswith('D:'):
            move_files(txt_files, destination_directory_D)

if __name__ == "__main__":
    main()
Explanation:
find_txt_files(root_directory): This function recursively searches through the provided root directory to find all .txt files and returns a list of their paths.

move_files(files, destination_directory): This function moves the provided list of files to the specified destination directory. It creates the destination directory if it doesn't exist.

main(): The main function defines the directories to search and the destination directories. It then calls the find_txt_files function for each search directory and moves the found files to the respective destination directory.

Notes:
The script assumes that you want to search the entire C: and D: drives. Be cautious with this approach as it can take a significant amount of time and resources if there are many files and directories.
Ensure you have the necessary permissions to read and write files in the specified directories.
Modify the destination_directory_C and destination_directory_D variables to specify the exact paths where you want to move the .txt files.

import os
import shutil
import re
from nltk.corpus import words

import nltk

english_words = set(words.words())

def find_txt_files(root_directory):
    txt_files = []

    try:
        with open(file_path, 'r', encoding='utf-8') as file:

            words_in_file = re.findall(r'\b\w+\b', content)

            return meaningful_words_count >= 12
    except Exception as e:
        print(f"Error reading file {file_path}: {e}")
        return False

                elif directory.startswith('D:'):

if __name__ == "__main__":
    main()
