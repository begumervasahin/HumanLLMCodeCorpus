To achieve real-time audio streaming from one computer to another, you can use Python along with libraries like pyaudio for capturing and playing audio and socket for network communication. Here’s a basic example of how you can set this up.

Server (Main Computer) Code
This code captures audio from the microphone and sends it to the client computer.

python
Kodu kopyala
import socket
import pyaudio

# Server configuration
HOST = '0.0.0.0'  # Listen on all available interfaces
PORT = 50007

# Audio configuration
CHUNK = 1024
FORMAT = pyaudio.paInt16
CHANNELS = 1
RATE = 44100

# Initialize pyaudio
p = pyaudio.PyAudio()

# Open audio stream
stream = p.open(format=FORMAT,
                channels=CHANNELS,
                rate=RATE,
                input=True,
                frames_per_buffer=CHUNK)

# Initialize socket
s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
s.bind((HOST, PORT))
s.listen(1)
print("Server listening on port", PORT)

conn, addr = s.accept()
print('Connected by', addr)

# Capture and send audio
try:
    while True:
        data = stream.read(CHUNK)
        conn.sendall(data)
except KeyboardInterrupt:
    pass
finally:
    stream.stop_stream()
    stream.close()
    p.terminate()
    conn.close()
    s.close()

To add the functionality of starting and stopping sound transmission using the 'b' and 's' keys, you can use the keyboard library in Python. Here's how you can integrate this feature into the server and client code:

Server (Main Computer) Code with Key Control
This code captures audio from the microphone and sends it to the client computer when 'b' is pressed and stops when 's' is pressed.

python

import socket
import pyaudio
import threading
import keyboard

# Server configuration
HOST = '0.0.0.0'  # Listen on all available interfaces
PORT = 50007

# Audio configuration
CHUNK = 1024
FORMAT = pyaudio.paInt16
CHANNELS = 1
RATE = 44100

# Initialize pyaudio
p = pyaudio.PyAudio()

# Open audio stream
stream = p.open(format=FORMAT,
                channels=CHANNELS,
                rate=RATE,
                input=True,
                frames_per_buffer=CHUNK)

# Initialize socket
s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
s.bind((HOST, PORT))
s.listen(1)
print("Server listening on port", PORT)

conn, addr = s.accept()
print('Connected by', addr)

transmitting = False

def send_audio():
    global transmitting
    while True:
        if transmitting:
            data = stream.read(CHUNK)
            conn.sendall(data)

def start_transmission():
    global transmitting
    transmitting = True

def stop_transmission():
    global transmitting
    transmitting = False

keyboard.add_hotkey('b', start_transmission)
keyboard.add_hotkey('s', stop_transmission)

send_thread = threading.Thread(target=send_audio)
send_thread.start()

try:
    while True:
        pass  # Keep the main thread running
except KeyboardInterrupt:
    pass
finally:
    stream.stop_stream()
    stream.close()
    p.terminate()
    conn.close()
    s.close()
