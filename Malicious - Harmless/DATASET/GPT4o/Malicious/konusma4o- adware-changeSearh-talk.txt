Yes, you can change the search engine used by a browser programmatically using Python. This can be done by manipulating browser settings or preferences files, depending on the browser you are using. Below are some examples of how you can do this for different browsers:

For Google Chrome
Google Chrome stores user preferences in a JSON file called Preferences. You can modify this file to change the default search engine.

python
Kodu kopyala
import json
import os

# Path to the Chrome User Data
chrome_user_data_path = os.path.expanduser("~/.config/google-chrome/Default/Preferences")

# Load the preferences file
with open(chrome_user_data_path, 'r') as file:
    preferences = json.load(file)

# Change the default search engine
preferences['default_search_provider_data'] = {
    "template_url_data": {
        "short_name": "DuckDuckGo",
        "keyword": "duckduckgo.com",
        "url": "https://duckduckgo.com/?q={searchTerms}"
    }
}

# Save the modified preferences file
with open(chrome_user_data_path, 'w') as file:
    json.dump(preferences, file)
For Mozilla Firefox
Firefox preferences can be changed by modifying the prefs.js file located in the Firefox profile directory.

python

python
Kodu kopyala
import json
import os

def change_chrome_search_engine(new_engine_name, new_engine_url):
    # Path to the Chrome User Data
    chrome_user_data_path = os.path.expanduser("~/.config/google-chrome/Default/Preferences")

    # Check if the preferences file exists
    if not os.path.exists(chrome_user_data_path):
        print("Preferences file not found.")
        return

    # Load the preferences file
    with open(chrome_user_data_path, 'r') as file:
        preferences = json.load(file)

    # Change the default search engine
    preferences['default_search_provider_data'] = {
        "template_url_data": {
            "short_name": new_engine_name,
            "keyword": new_engine_name.lower(),
            "url": new_engine_url
        }
    }

    # Save the modified preferences file
    with open(chrome_user_data_path, 'w') as file:
        json.dump(preferences, file, indent=4)

    print(f"Default search engine changed to {new_engine_name}")

# Example usage
change_chrome_search_engine("DuckDuckGo", "https://duckduckgo.com/?q={searchTerms}")

python
Kodu kopyala
import json
import os

def change_chrome_search_engine(new_engine_name, new_engine_url):
    chrome_paths = [
        os.path.expanduser("~/.config/google-chrome/Default/Preferences"),  # Linux
        os.path.expanduser("~/Library/Application Support/Google/Chrome/Default/Preferences"),  # macOS
        os.path.expanduser("~\\AppData\\Local\\Google\\Chrome\\User Data\\Default\\Preferences")  # Windows
    ]
    
    for path in chrome_paths:
        if os.path.exists(path):
            try:
                with open(path, 'r') as file:
                    preferences = json.load(file)

                preferences['default_search_provider_data'] = {
                    "template_url_data": {
                        "short_name": new_engine_name,
                        "keyword": new_engine_name.lower(),
                        "url": new_engine_url
                    }
                }

                with open(path, 'w') as file:
                    json.dump(preferences, file, indent=4)

                print(f"Changed Chrome search engine to {new_engine_name} at {path}")
            except Exception as e:
                print(f"Failed to change Chrome search engine at {path}: {e}")

def change_firefox_search_engine(new_engine_name):
    firefox_paths = [
        os.path.expanduser("~/.mozilla/firefox"),  # Linux
        os.path.expanduser("~/Library/Application Support/Firefox/Profiles"),  # macOS
        os.path.expanduser("~\\AppData\\Roaming\\Mozilla\\Firefox\\Profiles")  # Windows
    ]
    
    for base_path in firefox_paths:
        if os.path.exists(base_path):
            for profile in os.listdir(base_path):
                prefs_path = os.path.join(base_path, profile, "prefs.js")
                if os.path.exists(prefs_path):
                    try:
                        search_engine_preference = f'user_pref("browser.search.defaultenginename", "{new_engine_name}");\n'
                        with open(prefs_path, 'a') as file:
                            file.write(search_engine_preference)

                        print(f"Changed Firefox search engine to {new_engine_name} at {prefs_path}")
                    except Exception as e:
                        print(f"Failed to change Firefox search engine at {prefs_path}: {e}")
