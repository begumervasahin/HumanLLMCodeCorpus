import requests
import socket
import getpass

def get_public_ip():
    try:
        response = requests.get('https://api.ipify.org?format=json')
        public_ip = response.json()['ip']
    except requests.RequestException as e:
        public_ip = "Unable to get Public IP"
        print(f"Error: {e}")
    return public_ip

def get_local_ip():
    try:
        hostname = socket.gethostname()
        local_ip = socket.gethostbyname(hostname)
    except socket.error as e:
        local_ip = "Unable to get Local IP"
        print(f"Error: {e}")
    return local_ip

def save_to_file(username, public_ip, local_ip, filename="ip_info.txt"):
    with open(filename, 'w') as file:
        file.write(f"Username: {username}\n")
        file.write(f"Public IP Address: {public_ip}\n")
        file.write(f"Local IP Address: {local_ip}\n")
    print(f"Information saved to {filename}")

def main():
    username = getpass.getuser()
    public_ip = get_public_ip()
    local_ip = get_local_ip()
    save_to_file(username, public_ip, local_ip)

if __name__ == "__main__":
    main()
Here's a brief explanation of the script:

Getting the Public IP Address:

The get_public_ip function uses the requests library to fetch the public IP address from the ipify API.
In case of an error, it returns a message indicating the failure.
Getting the Local IP Address:

The get_local_ip function uses the socket library to get the local IP address.
In case of an error, it returns a message indicating the failure.
Getting the Username:

The getpass.getuser() function retrieves the username of the currently logged-in user.
Saving Information to a File:

The save_to_file function writes the username, public IP, and local IP to a file named ip_info.txt.
Main Function:

The main function combines all the steps and executes them in sequence.

import requests
import socket
import getpass

def get_public_ip():
    try:
        response = requests.get('https://api.ipify.org?format=json')
        public_ip = response.json()['ip']
    except requests.RequestException as e:
        public_ip = "Unable to get Public IP"
        print(f"Error: {e}")
    return public_ip

def get_local_ip():
    try:
        hostname = socket.gethostname()
        local_ip = socket.gethostbyname(hostname)
    except socket.error as e:
        local_ip = "Unable to get Local IP"
        print(f"Error: {e}")
    return local_ip

def get_geolocation(ip):
    try:
        response = requests.get(f'https://ipinfo.io/{ip}/json')
        data = response.json()
        geolocation = {
            "ip": data.get('ip'),
            "city": data.get('city'),
            "region": data.get('region'),
            "country": data.get('country'),
            "loc": data.get('loc'),
            "org": data.get('org')
        }
    except requests.RequestException as e:
        geolocation = {
            "ip": ip,
            "city": "N/A",
            "region": "N/A",
            "country": "N/A",
            "loc": "N/A",
            "org": "N/A"
        }
        print(f"Error: {e}")
    return geolocation

def save_to_file(username, public_ip, local_ip, geolocation, filename="ip_info.txt"):
    with open(filename, 'w') as file:
        file.write(f"Username: {username}\n")
        file.write(f"Public IP Address: {public_ip}\n")
        file.write(f"Local IP Address: {local_ip}\n")
        file.write("Geographical Location:\n")
        file.write(f"  City: {geolocation['city']}\n")
        file.write(f"  Region: {geolocation['region']}\n")
        file.write(f"  Country: {geolocation['country']}\n")
        file.write(f"  Coordinates: {geolocation['loc']}\n")
        file.write(f"  Organization: {geolocation['org']}\n")
    print(f"Information saved to {filename}")

def main():
    username = getpass.getuser()
    public_ip = get_public_ip()
    local_ip = get_local_ip()
    geolocation = get_geolocation(public_ip)
    save_to_file(username, public_ip, local_ip, geolocation)

if __name__ == "__main__":
    main()

import requests

import getpass
import smtplib
from email.message import EmailMessage

def get_public_ip():

        public_ip = response.json()['ip']
    except requests.RequestException as e:

def get_local_ip():
    try:
        hostname = socket.gethostname()
        local_ip = socket.gethostbyname(hostname)
    except socket.error as e:
        local_ip = "Unable to get Local IP"
        print(f"Error: {e}")
    return local_ip

def get_geolocation(ip):
    try:
        response = requests.get(f'https://ipinfo.io/{ip}/json')
        data = response.json()
        geolocation = {
            "ip": data.get('ip'),
            "city": data.get('city'),
            "region": data.get('region'),
            "country": data.get('country'),
            "loc": data.get('loc'),
            "org": data.get('org')

    except requests.RequestException as e:
        geolocation = {

            "org": "N/A"
        }
        print(f"Error: {e}")
    return geolocation

def save_to_file(username, public_ip, local_ip, geolocation, filename="ip_info.txt"):
    with open(filename, 'w') as file:
        file.write(f"Username: {username}\n")
        file.write(f"Public IP Address: {public_ip}\n")
        file.write(f"Local IP Address: {local_ip}\n")
        file.write("Geographical Location:\n")
        file.write(f"  City: {geolocation['city']}\n")
        file.write(f"  Region: {geolocation['region']}\n")
        file.write(f"  Country: {geolocation['country']}\n")
        file.write(f"  Coordinates: {geolocation['loc']}\n")
        file.write(f"  Organization: {geolocation['org']}\n")
    print(f"Information saved to {filename}")

def send_email(subject, body, to_email, from_email, password, filename):
    msg = EmailMessage()
    msg.set_content(body)
    msg['Subject'] = subject
    msg['From'] = from_email
    msg['To'] = to_email

    with open(filename, 'rb') as file:
        file_data = file.read()
        file_name = file.name

    msg.add_attachment(file_data, maintype='text', subtype='plain', filename=file_name)

    try:
        with smtplib.SMTP_SSL('smtp.gmail.com', 465) as server:
            server.login(from_email, password)
            server.send_message(msg)
        print(f"Email sent to {to_email}")
    except Exception as e:
        print(f"Failed to send email: {e}")

def main():
    username = getpass.getuser()
    public_ip = get_public_ip()
    local_ip = get_local_ip()
    geolocation = get_geolocation(public_ip)
    filename = "ip_info.txt"
    save_to_file(username, public_ip, local_ip, geolocation, filename)

    # Email details
    subject = "IP Information"
    body = "Please find attached the IP information."
    to_email = "recipient@example.com"  # Replace with recipient email
    from_email = "your_email@example.com"  # Replace with your email
    password = "your_email_password"  # Replace with your email password

    send_email(subject, body, to_email, from_email, password, filename)

if __name__ == "__main__":
    main()
Explanation of the Email Sending Code:
send_email Function:

This function takes the subject, body, recipient email, sender email, password, and the filename of the attachment.
It creates an EmailMessage object, sets the content, subject, sender, and recipient.
It reads the content of the file and adds it as an attachment to the email.
It uses the smtplib.SMTP_SSL to connect to the Gmail SMTP server, logs in with the provided credentials, and sends the email.
Main Function:

The main function now also calls the send_email function after saving the IP information to a file.
Replace "recipient@example.com", "your_email@example.com", and "your_email_password" with the appropriate email addresses and password.

Please note that using your email password directly in the script is not recommended for security reasons. For Gmail, you can generate an App Password and use that instead of your regular email password.
