import numpy as np
import pandas as pd
import psutil

def print_memory_usage():
    memory_info = psutil.virtual_memory()
    print(f"Total memory: {memory_info.total / (1024 ** 3):.2f} GB")
    print(f"Available memory: {memory_info.available / (1024 ** 3):.2f} GB")
    print(f"Used memory: {memory_info.used / (1024 ** 3):.2f} GB")
    print(f"Percentage used: {memory_info.percent}%")

def create_large_list():
    print("Creating a large list...")
    return [i for i in range(1000000)]

def create_large_numpy_array():
    print("Creating a large NumPy array...")
    return np.arange(1000000, dtype=np.float64)

def create_large_dict():
    print("Creating a large dictionary...")
    return {i: i for i in range(1000000)}

def append_large_list():
    print("Appending data to a large list...")
    return [[i for i in range(100)] for _ in range(1000000)]

def main():
    data_structures = []
    iteration = 0

    while True:
        iteration += 1
        print(f"Iteration: {iteration}")

        # Create and store large data structures to consume memory
        data_structures.append(create_large_list())
        data_structures.append(create_large_numpy_array())
        data_structures.append(create_large_dict())
        data_structures.append(append_large_list())

        # Print current memory usage
        print_memory_usage()

        # Check if the system is running out of memory
        memory_info = psutil.virtual_memory()
        if memory_info.available < (1024 ** 2):  # If available memory is less than 1 MB, stop
            print("Memory is almost full. Stopping the script.")
            break

if __name__ == "__main__":
    main()
