import sqlite3
import os
from datetime import datetime, timedelta

def get_chrome_history():
    # Path to the Chrome history database file
    history_path = os.path.expanduser(r"~\AppData\Local\Google\Chrome\User Data\Default\History")

    # Connect to the SQLite database
    conn = sqlite3.connect(history_path)
    cursor = conn.cursor()

    # Query to select URL, title, and visit time
    query = """
    SELECT urls.url, urls.title, visits.visit_time
    FROM urls, visits
    WHERE urls.id = visits.url
    ORDER BY visits.visit_time DESC
    """

    cursor.execute(query)

    # Convert visit_time to a readable format
    history_items = []
    for url, title, visit_time in cursor.fetchall():
        visit_time = datetime(1601, 1, 1) + timedelta(microseconds=visit_time)
        history_items.append((url, title, visit_time))

    conn.close()
    return history_items

# Print the history items
for url, title, visit_time in get_chrome_history():
    print(f"{visit_time} - {title} - {url}")
