import os
import shutil
import subprocess

# Define paths
script_name = "shutdown.py"
executable_name = "shutdown.exe"
nssm_path = "C:\\path\\to\\nssm.exe"  # Replace with the actual path to nssm.exe
service_name = "ShutdownService"

# Step 1: Create the shutdown script
shutdown_script = """
import os

os.system("shutdown /s /t 1")
"""

with open(script_name, "w") as file:
    file.write(shutdown_script)

# Step 2: Convert the script to an executable
subprocess.run(["pyinstaller", "--onefile", script_name], check=True)

# Step 3: Move the executable to a known location
dist_dir = os.path.join(os.getcwd(), "dist")
shutil.move(os.path.join(dist_dir, executable_name), os.path.join(os.getcwd(), executable_name))

# Step 4: Create a Windows service using nssm
subprocess.run([nssm_path, "install", service_name, os.path.join(os.getcwd(), executable_name)], check=True)
subprocess.run([nssm_path, "set", service_name, "Start", "SERVICE_AUTO_START"], check=True)
subprocess.run([nssm_path, "start", service_name], check=True)

print(f"Service '{service_name}' created and started successfully.")

    # Step 3: Move the executable to a known location
    dist_dir = os.path.join(os.getcwd(), "dist")
