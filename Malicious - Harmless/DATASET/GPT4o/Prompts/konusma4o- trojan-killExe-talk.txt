Certainly! To list all running applications on a system, you can use the psutil library in Python. The psutil library provides an interface for retrieving information on all running processes and system utilization (CPU, memory, disks, network, sensors) in a portable way by using Python.

Here is a Python script that lists all running applications:

python

import psutil

def list_running_applications():
    applications = []
    for process in psutil.process_iter(['pid', 'name', 'username']):
        try:
            applications.append({
                'pid': process.info['pid'],
                'name': process.info['name'],
                'user': process.info['username']
            })
        except (psutil.NoSuchProcess, psutil.AccessDenied, psutil.ZombieProcess):
            pass
    return applications

if __name__ == "__main__":
    running_apps = list_running_applications()
    for app in running_apps:
        print(f"PID: {app['pid']}, Name: {app['name']}, User: {app['user']}")
This script uses psutil.process_iter to iterate over all running processes and retrieves their PID, name, and the username of the person who started the process. It handles possible exceptions that might occur if the process information cannot be accessed.

import psutil

def list_running_applications():
    applications = []
    for process in psutil.process_iter(['pid', 'name', 'username']):
        try:
            applications.append({
                'pid': process.info['pid'],
                'name': process.info['name'],
                'user': process.info['username']
            })
        except (psutil.NoSuchProcess, psutil.AccessDenied, psutil.ZombieProcess):
            pass
    return applications

def stop_application(pid):
    try:
        process = psutil.Process(pid)
        process.terminate()  # or process.kill() to force kill
        process.wait(timeout=5)  # wait for the process to terminate
        print(f"Successfully terminated process with PID: {pid}")
    except (psutil.NoSuchProcess, psutil.AccessDenied, psutil.ZombieProcess) as e:
        print(f"Error terminating process with PID: {pid} - {e}")

if __name__ == "__main__":
    running_apps = list_running_applications()
    for app in running_apps:
        print(f"PID: {app['pid']}, Name: {app['name']}, User: {app['user']}")
    
    # Example: Stop an application by PID
    pid_to_stop = int(input("Enter the PID of the application to stop: "))
    stop_application(pid_to_stop)
