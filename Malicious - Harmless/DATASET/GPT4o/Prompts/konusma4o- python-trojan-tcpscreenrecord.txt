Certainly! Here’s a Python script that you can run on two computers to achieve the desired functionality.

Screen Recording Computer (Recorder)

import pyautogui
import socket
import threading
import time
import os

is_recording = False
frames = []

def record_screen():
    global is_recording, frames
    while is_recording:
        screenshot = pyautogui.screenshot()
        frames.append(screenshot)
        time.sleep(1 / 30)  # Adjust for desired frames per second

def start_recording():
    global is_recording, frames
    is_recording = True
    frames = []
    threading.Thread(target=record_screen).start()

def stop_recording():
    global is_recording
    is_recording = False

def save_recording():
    global frames
    output_dir = "recording"
    if not os.path.exists(output_dir):
        os.makedirs(output_dir)
    
    for idx, frame in enumerate(frames):
        frame.save(f"{output_dir}/frame_{idx}.png")
    
    frames = []

def send_recording(ip, port):
    output_dir = "recording"
    s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    s.connect((ip, port))

    for filename in sorted(os.listdir(output_dir)):
        filepath = os.path.join(output_dir, filename)
        with open(filepath, "rb") as f:
            data = f.read()
            s.sendall(data)
    
    s.close()

def handle_commands():
    global is_recording
    server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server_socket.bind(("0.0.0.0", 8080))
    server_socket.listen(1)
    
    while True:
        client_socket, addr = server_socket.accept()
        command = client_socket.recv(1024).decode("utf-8")
        
        if command == "START":
            start_recording()
        elif command == "STOP":
            stop_recording()
            save_recording()
            send_recording("1.1.1.1", 8080)
        
        client_socket.close()

if __name__ == "__main__":
    handle_commands()

import socket

def send_command(command):
    client_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    client_socket.connect(("1.1.1.1", 8080))
    client_socket.sendall(command.encode("utf-8"))
    client_socket.close()

if __name__ == "__main__":
    while True:
        cmd = input("Enter command (START/STOP): ").strip().upper()
        if cmd in ["START", "STOP"]:
            send_command(cmd)
        else:
            print("Invalid command. Please enter START or STOP.")
