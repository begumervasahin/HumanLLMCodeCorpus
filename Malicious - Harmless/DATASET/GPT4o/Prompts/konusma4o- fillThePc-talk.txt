import psutil
import GPUtil
import pynvml

def get_cpu_usage():
    cpu_usage = psutil.cpu_percent(interval=1)
    return cpu_usage

def get_ram_usage():
    memory_info = psutil.virtual_memory()
    ram_usage = memory_info.percent
    return ram_usage

def get_gpu_usage():
    GPUtil.showUtilization()

def get_nvidia_vram_usage():
    pynvml.nvmlInit()
    handle = pynvml.nvmlDeviceGetHandleByIndex(0)
    memory_info = pynvml.nvmlDeviceGetMemoryInfo(handle)
    total_memory = memory_info.total / 1024**2
    used_memory = memory_info.used / 1024**2
    free_memory = memory_info.free / 1024**2
    pynvml.nvmlShutdown()
    return total_memory, used_memory, free_memory

def main():
    # CPU Usage
    cpu_usage = get_cpu_usage()
    print(f"CPU Usage: {cpu_usage}%")

    # RAM Usage
    ram_usage = get_ram_usage()
    print(f"RAM Usage: {ram_usage}%")

    # GPU Usage
    print("GPU Usage:")
    get_gpu_usage()

    # NVIDIA VRAM Usage
    total_memory, used_memory, free_memory = get_nvidia_vram_usage()
    print(f"Total GPU Memory: {total_memory} MB")
    print(f"Used GPU Memory: {used_memory} MB")
    print(f"Free GPU Memory: {free_memory} MB")

if __name__ == "__main__":
    main()
