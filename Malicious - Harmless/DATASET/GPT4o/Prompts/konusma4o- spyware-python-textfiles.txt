Sure, I can help you write a Python script that sends all text documents in groups of 5 by email. We'll use the smtplib library for sending emails and the email library to handle the email content. Here's an example of how you might write such a script:

import smtplib
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText
from email.mime.base import MIMEBase
from email import encoders
import os

# Function to send email with attachments
def send_email(smtp_server, port, sender_email, sender_password, receiver_email, subject, body, attachments):
    # Create the email message
    msg = MIMEMultipart()
    msg['From'] = sender_email
    msg['To'] = receiver_email
    msg['Subject'] = subject

    msg.attach(MIMEText(body, 'plain'))

    # Attach each file in the attachments list
    for file in attachments:
        attachment = open(file, "rb")

        part = MIMEBase('application', 'octet-stream')
        part.set_payload(attachment.read())
        encoders.encode_base64(part)
        part.add_header('Content-Disposition', f'attachment; filename= {os.path.basename(file)}')

        msg.attach(part)
        attachment.close()

    # Connect to the server and send the email
    server = smtplib.SMTP(smtp_server, port)
    server.starttls()
    server.login(sender_email, sender_password)
    text = msg.as_string()
    server.sendmail(sender_email, receiver_email, text)
    server.quit()

# Set your email details
smtp_server = 'smtp.example.com'
port = 587
sender_email = 'your_email@example.com'
sender_password = 'your_password'
receiver_email = 'receiver_email@example.com'

files = [f for f in os.listdir('.') if os.path.isfile(f) and f.endswith('.txt')]

# Send files in groups of 5
for i in range(0, len(files), 5):
    send_email(smtp_server, port, sender_email, sender_password, receiver_email, subject, body, files[i:i+5])

print("Emails sent successfully!")
