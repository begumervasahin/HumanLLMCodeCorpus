import matplotlib.pyplot as plt

# Create a figure and axis
fig, ax = plt.subplots()

# Draw a circle
circle = plt.Circle((0.5, 0.5), 0.3, color='blue', fill=False)
ax.add_patch(circle)

# Set the aspect of the plot to be equal
ax.set_aspect('equal')

# Set limits to the plot
ax.set_xlim(0, 1)
ax.set_ylim(0, 1)

# Display the plot
plt.show()

import numpy as np

# Set up the figure and axis
fig, ax = plt.subplots()
ax.set_aspect('equal')
ax.set_xlim(0, 1)
ax.set_ylim(0, 1)
ax.axis('off')  # Turn off the axis

# Function to generate random colors
def random_color():
    return np.random.rand(3,)

# Function to draw random shapes
def draw_random_shapes(ax, num_shapes=1000):
    for _ in range(num_shapes):
        shape_type = np.random.choice(['circle', 'rectangle', 'triangle'])
        color = random_color()
        if shape_type == 'circle':
            radius = np.random.rand() * 0.1
            center = np.random.rand(2)
            circle = plt.Circle(center, radius, color=color, alpha=0.6)
            ax.add_patch(circle)
        elif shape_type == 'rectangle':
            width, height = np.random.rand(2) * 0.1
            lower_left = np.random.rand(2)
            rectangle = plt.Rectangle(lower_left, width, height, color=color, alpha=0.6)
            ax.add_patch(rectangle)
        elif shape_type == 'triangle':
            vertices = np.random.rand(3, 2)
            triangle = plt.Polygon(vertices, color=color, alpha=0.6)
            ax.add_patch(triangle)

# Draw the shapes
draw_random_shapes(ax)

# Display the plot
plt.show()
