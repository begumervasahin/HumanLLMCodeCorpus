import pyautogui
import time

# Function to lock the screen
def lock_screen():
    # Simulate key press to lock the screen
    pyautogui.hotkey('win', 'l')

# Main function to run indefinitely
def main():
    while True:
        # Check if computer is just turned on (you can use some condition here)
        # For demonstration, let's just wait for 60 seconds before locking
        time.sleep(60)
        
        # Lock the screen
        lock_screen()

if __name__ == "__main__":
    main()
