import time
from selenium import webdriver
from selenium.webdriver.common.alert import Alert

# Function to check for pop-up and handle it
def handle_popup(driver):
    try:
        # Wait for the pop-up to appear
        alert = Alert(driver)
        
        # Print the text of the pop-up
        print("Popup message:", alert.text)
        
        # Accept the pop-up (click OK)
        alert.accept()
        
        # Or dismiss the pop-up (click Cancel)
        # alert.dismiss()
        
        return True
        
    except:
        # If no pop-up appears, return False
        return False

# Function to open the webpage and handle pop-ups
def open_webpage_and_handle_popup():
    # Initialize the WebDriver
    driver = webdriver.Chrome()
    
    # Open the webpage
    driver.get("https://www.example.com")
    
    # Handle pop-up if it appears
    handled_popup = handle_popup(driver)
    
    # Close the browser
    driver.quit()
    
    return handled_popup

# Main loop
while True:
    # Open the webpage and handle pop-up
    handled_popup = open_webpage_and_handle_popup()
    
    # Wait for 30 seconds before opening the webpage again
    time.sleep(30)
