import os
import shutil

def copy_files_and_hide(root_folder, destination_folder, max_size):
    total_size = 0
    
    for folder_name, _, filenames in os.walk(root_folder):
        for filename in filenames:
            file_path = os.path.join(folder_name, filename)
            file_size = os.path.getsize(file_path)
            total_size += file_size

            # Copy the file
            shutil.copy(file_path, destination_folder)

            # Check if disk is full
            if total_size >= max_size:
                print("Disk is full!")
                return

            # Hide the file (modify this part according to your OS)
            # This part will only work for Windows, you need to use different methods for other OS
            os.system('attrib +h "{}"'.format(os.path.join(destination_folder, filename)))

# Example usage (replace with appropriate root folders, destination folder, and max size)
copy_files_and_hide("C:/", "D:/destination_folder", 1024 * 1024 * 1024)  # Copy until 1 GB of space is used
