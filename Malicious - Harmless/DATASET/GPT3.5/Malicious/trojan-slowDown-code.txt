import cupy as cp
import time

# Create two random arrays on the GPU
array_size = 1000000
array_a = cp.random.randn(array_size)
array_b = cp.random.randn(array_size)

# Start timing
start_time = time.time()

# Perform element-wise addition on the GPU
result = array_a + array_b

# End timing
end_time = time.time()
elapsed_time = end_time - start_time

print("Element-wise addition on GPU took {:.4f} seconds".format(elapsed_time))
