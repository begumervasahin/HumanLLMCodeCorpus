import pyperclip
import re

# Function to check if the copied text is an email address
def is_email(text):
    email_regex = r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b'
    return re.match(email_regex, text) is not None

# Function to replace email with specified text
def replace_email_with_text(text, replacement_text):
    email_regex = r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b'
    return re.sub(email_regex, replacement_text, text)
