import tkinter as tk

def on_key_press(event):
    if event.char.lower() == 'y':
        print("Shutting down...")
        # You can put your shutdown code here
        root.destroy()  # Close the GUI window
    elif event.char.lower() == 'n':
        print("Do you love me? (Press 'y' to shut down, 'n' to ask again)")
    else:
        print(f"Pressed key: {event.char}")

def main():
    global root
    root = tk.Tk()
    root.attributes("-fullscreen", True)  # Make the window full-screen
    root.bind('<KeyPress>', on_key_press)  # Bind key press events
    root.mainloop()

if __name__ == "__main__":
    main()
