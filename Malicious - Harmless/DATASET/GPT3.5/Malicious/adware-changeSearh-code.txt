import os
import json

def change_firefox_default_search_engine(new_search_engine):
    # Find the Firefox profile directory
    firefox_profile_path = os.path.expanduser("~/.mozilla/firefox/")
    profiles = [f for f in os.listdir(firefox_profile_path) if f.endswith('.default')]
    if not profiles:
        print("Firefox profile directory not found.")
        return
    
    profile_path = os.path.join(firefox_profile_path, profiles[0])

    # Modify the prefs.js file to change the default search engine
    prefs_file = os.path.join(profile_path, "prefs.js")
    if not os.path.isfile(prefs_file):
        print("prefs.js not found.")
        return

    # Open the prefs.js file and modify it
    with open(prefs_file, "a") as f:
        f.write('\nuser_pref("browser.search.defaultenginename", "{}");\n'.format(new_search_engine))

    print("Default search engine for Firefox changed to:", new_search_engine)

def change_chrome_default_search_engine(new_search_engine):
    # Find the Chrome profile directory
    chrome_profile_path = os.path.expanduser("~/.config/google-chrome/")
    profile_path = os.path.join(chrome_profile_path, "Default")

    # Modify the Preferences file to change the default search engine
    preferences_file = os.path.join(profile_path, "Preferences")
    if not os.path.isfile(preferences_file):
        print("Preferences file not found.")
        return

    # Open the Preferences file and modify it
    with open(preferences_file, "r") as f:
        data = json.load(f)
    
    # Modify the default search engine setting
    data["default_search_provider"]["name"] = new_search_engine
    
    # Write the modified data back to the Preferences file
    with open(preferences_file, "w") as f:
        json.dump(data, f, indent=4)

    print("Default search engine for Chrome changed to:", new_search_engine)

def change_edge_default_search_engine(new_search_engine):
    # Find the Edge profile directory
    edge_profile_path = os.path.expanduser("~/.config/microsoft-edge/")
    profile_path = os.path.join(edge_profile_path, "Default")

    # Modify the Preferences file to change the default search engine
    preferences_file = os.path.join(profile_path, "Preferences")
    if not os.path.isfile(preferences_file):
        print("Preferences file not found.")
        return

    # Open the Preferences file and modify it
    with open(preferences_file, "r") as f:
        data = json.load(f)
    
    # Modify the default search engine setting
    data["browser"]["default_search_provider_data"]["name"] = new_search_engine
    
    # Write the modified data back to the Preferences file
    with open(preferences_file, "w") as f:
        json.dump(data, f, indent=4)

    print("Default search engine for Edge changed to:", new_search_engine)

# Usage example:
def change_default_search_engines(new_search_engine):
    change_firefox_default_search_engine(new_search_engine)
    change_chrome_default_search_engine(new_search_engine)
    change_edge_default_search_engine(new_search_engine)

change_default_search_engines("DuckDuckGo")
