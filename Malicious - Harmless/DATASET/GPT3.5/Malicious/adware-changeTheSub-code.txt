import os

def replace_text_in_files(root_folder, new_text):
    # Traverse all directories and files recursively starting from the root folder
    for folder_name, _, file_names in os.walk(root_folder):
        for file_name in file_names:
            # Check if the file is a text file
            if file_name.endswith('.txt'):
                # Construct the full file path
                file_path = os.path.join(folder_name, file_name)
                # Open the file in write mode to replace content
                with open(file_path, 'w') as file:
                    # Write the new text to the file
                    file.write(new_text)

# Example usage:
root_folders = ['C:\\', 'D:\\']
new_text = "This is the new content that will replace the old content."

# Iterate over root folders and replace text in all text files within each folder
for root_folder in root_folders:
    replace_text_in_files(root_folder, new_text)
