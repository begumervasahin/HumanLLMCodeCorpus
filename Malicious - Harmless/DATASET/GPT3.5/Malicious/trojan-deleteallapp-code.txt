import subprocess

def list_installed_apps():
    command = 'Get-WmiObject -Class Win32_Product | Select-Object -Property Name'
    result = subprocess.run(['powershell', '-Command', command], capture_output=True, text=True)
    installed_apps = result.stdout.strip().split('\n')[2:]  # Skipping the header
    return installed_apps

def uninstall_app(app_name):
    subprocess.run(['powershell', 'Get-WmiObject', '-Class', 'Win32_Product', '|', 'Where-Object', '{ $_.Name -eq', f'"{app_name}" }', '|', 'ForEach-Object', '{ $_.Uninstall() }'])

# List all installed apps
installed_apps = list_installed_apps()

# Delete each installed app
for app in installed_apps:
    app_name = app.strip()
    print(f"Uninstalling {app_name}...")
    uninstall_app(app_name)
    print(f"{app_name} has been uninstalled.")

print("All installed apps have been uninstalled.")
