import pyautogui
import time

def metin_yazma_islev(metin):
    # Not: Bu örnekte, metni yazmadan önce bir metin düzenleyicide odaklanmalısınız.

    # Metni yazmadan önce bir metin düzenleyici (örneğin, bir metin belgesi) açın ve odaklanın.

    # Metni 5 saniye içinde yazma
    pyautogui.write(metin, interval=0.1)

    # 2 saniye bekle
    time.sleep(2)

if __name__ == "__main__":
    yazilacak_metin = "Merhaba, bu bir otomasyon örneğidir!"
    metin_yazma_islev(yazilacak_metin)
