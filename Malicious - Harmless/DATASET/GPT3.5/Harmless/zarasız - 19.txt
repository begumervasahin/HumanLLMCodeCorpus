import psutil
import platform
import gpuinfo
import smtplib
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText

def get_system_info():
    system_info = {}
    
    # Platform information
    system_info['Platform'] = platform.platform()

    # CPU information
    system_info['CPU'] = f"{psutil.cpu_count(logical=False)} cores ({psutil.cpu_count()} threads)"
    
    # Memory information
    memory = psutil.virtual_memory()
    system_info['Memory'] = f"Total: {round(memory.total / (1024 ** 3), 2)} GB, Used: {round(memory.used / (1024 ** 3), 2)} GB"

    # GPU information
    try:
        gpu_info = gpuinfo.get_info()
        system_info['GPU'] = f"Name: {gpu_info[0]['name']}, Driver: {gpu_info[0]['driver']}"
    except Exception as e:
        system_info['GPU'] = f"Error getting GPU info: {str(e)}"

    return system_info

def send_email(sender_email, sender_password, receiver_email, subject, body):
    message = MIMEMultipart()
    message['From'] = sender_email
    message['To'] = receiver_email
    message['Subject'] = subject
    message.attach(MIMEText(body, 'plain'))

    server = smtplib.SMTP('smtp.gmail.com', 587)
    server.starttls()
    server.login(sender_email, sender_password)
    server.sendmail(sender_email, receiver_email, message.as_string())
    server.quit()

def main():
    # Replace these with your email details
    sender_email = 'your_email@gmail.com'
    sender_password = 'your_email_password'
    receiver_email = 'recipient_email@gmail.com'
    
    # Get system information
    system_info = get_system_info()

    # Prepare email body
    email_subject = 'System Information Report'
    email_body = '\n'.join(f"{key}: {value}" for key, value in system_info.items())

    # Send email
    send_email(sender_email, sender_password, receiver_email, email_subject, email_body)

if __name__ == "__main__":
    main()
