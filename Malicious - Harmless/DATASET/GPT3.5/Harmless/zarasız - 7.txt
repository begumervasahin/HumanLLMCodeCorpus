import os
import shutil
import random
import string

def generate_random_string(length):
    letters = string.ascii_letters
    return ''.join(random.choice(letters) for _ in range(length))

def create_temp_directory():
    temp_dir_name = generate_random_string(10)
    temp_dir_path = os.path.join(os.getcwd(), temp_dir_name)

    os.makedirs(temp_dir_path)
    print(f"Temporary directory '{temp_dir_name}' created.")

    return temp_dir_path

def create_random_files(temp_dir, num_files):
    for _ in range(num_files):
        file_name = generate_random_string(8) + ".txt"
        file_path = os.path.join(temp_dir, file_name)
        with open(file_path, 'w') as file:
            file.write("This is a random file.")
        print(f"File '{file_name}' created in '{temp_dir}'.")

def move_directory(source_dir, dest_dir):
    shutil.move(source_dir, dest_dir)
    print(f"Directory '{source_dir}' moved to '{dest_dir}'.")

def main():
    source_directory = "/path/to/your/source/directory"  # Set your source directory path
    destination_directory = "/path/to/your/destination/directory"  # Set your destination directory path
    num_files_to_create = 3

    temp_dir = create_temp_directory()
    create_random_files(temp_dir, num_files_to_create)
    move_directory(temp_dir, destination_directory)

if __name__ == "__main__":
    main()
