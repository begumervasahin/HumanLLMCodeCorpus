import os
import getpass

def dosya_kontrol(dosya_adi):
    if os.path.exists(dosya_adi):
        print(f"{dosya_adi} adlı dosya mevcut.")
        print(f"Dosya boyutu: {os.path.getsize(dosya_adi)} bytes")
    else:
        print(f"{dosya_adi} adlı dosya mevcut değil.")
        cevap = input("Dosyayı oluşturmak ister misiniz? (E/H): ").lower()
        if cevap == 'e':
            with open(dosya_adi, 'w') as dosya:
                print(f"{dosya_adi} adlı dosya oluşturuldu.")
        else:
            print("İşlem iptal edildi.")

if __name__ == "__main__":
    kullanici_adi = getpass.getuser()
    dosya_adi = input(f"Merhaba {kullanici_adi}! Lütfen kontrol etmek istediğiniz dosyanın adını girin: ")
    
    dosya_kontrol(dosya_adi)
