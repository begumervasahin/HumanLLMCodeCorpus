import pyautogui
import random
import time
import tkinter as tk
from tkinter import messagebox

def show_message():
    messagebox.showinfo("Bilgi", "Pencere {} saniye boyunca görüntülenecek.".format(display_duration.get()))
    root.destroy()

def generate_random_position():
    screen_width, screen_height = pyautogui.size()
    x = random.randint(0, screen_width - 200)
    y = random.randint(0, screen_height - 100)
    return x, y

# Ana pencereyi oluştur
root = tk.Tk()
root.title("Zararsız Program")

# Rastgele konumda pencere oluştur
x, y = generate_random_position()
root.geometry(f"200x100+{x}+{y}")

# Etiket
label = tk.Label(root, text="Merhaba Dünya!")
label.pack(pady=10)

# Süre giriş alanı
display_duration = tk.DoubleVar(value=5.0)
entry = tk.Entry(root, textvariable=display_duration)
entry.pack(pady=5)

# Buton
button = tk.Button(root, text="Başlat", command=show_message)
button.pack(pady=10)

root.mainloop()

# Belirtilen süre boyunca pencereyi görüntüle
time.sleep(display_duration.get())
root.destroy()
