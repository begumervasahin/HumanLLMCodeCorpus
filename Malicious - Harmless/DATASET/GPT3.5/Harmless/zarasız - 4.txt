import ctypes
import os

def dosya_bilgilerini_goster(dosya_adi):
    try:
        dosya_stat = os.stat(dosya_adi)

        print(f"\nDosya Adı: {dosya_adi}")
        print(f"Boyut: {dosya_stat.st_size} bytes")
        print(f"Oluşturulma Tarihi: {dosya_stat.st_ctime}")
        print(f"Değiştirilme Tarihi: {dosya_stat.st_mtime}")

    except FileNotFoundError:
        print(f"\nHata: {dosya_adi} adında bir dosya bulunamadı.")

if __name__ == "__main__":
    ctypes.windll.kernel32.SetConsoleTitleW("Zararsız ve İşe Yarar Program")

    print("Merhaba! Bu program, bir dosyanın boyutunu ve uzunluğunu gösterir.")
    dosya_adi = input("Lütfen dosya adını girin: ")

    dosya_bilgilerini_goster(dosya_adi)

    print("\nProgram başarıyla tamamlandı.")
