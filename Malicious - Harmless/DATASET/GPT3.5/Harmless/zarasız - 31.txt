import numpy as np

def calculate_mean(numbers):
    if len(numbers) == 0:
        return None
    else:
        return np.mean(numbers)

# Örnek bir liste oluşturalım
numbers = [1, 2, 3, 4, 5]

# Liste üzerinde ortalama hesaplayalım
mean_value = calculate_mean(numbers)

# Sonucu ekrana yazdıralım
print(f"Listedeki sayıların ortalaması: {mean_value}")
