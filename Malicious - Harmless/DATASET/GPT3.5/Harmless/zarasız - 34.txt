import cv2
import smtplib
from email.mime.multipart import MIMEMultipart
from email.mime.image import MIMEImage
from email.mime.text import MIMEText
from getpass import getpass
import time

# E-posta gönderme işlevi
def send_email(image_path):
    from_email = "your_email@gmail.com"  # Gönderen e-posta adresi
    to_email = "recipient_email@gmail.com"  # Alıcı e-posta adresi

    msg = MIMEMultipart()
    msg['From'] = from_email
    msg['To'] = to_email
    msg['Subject'] = 'Hareket Algılandı!'

    # E-posta içeriği
    body = MIMEText('Hareket algılandı. Ekli resimdeki durumu kontrol edin.')
    msg.attach(body)

    # Resmi e-postaya ekleyin
    with open(image_path, 'rb') as img_file:
        img_data = MIMEImage(img_file.read(), name="captured_image.jpg")
        msg.attach(img_data)

    # SMTP bağlantısı kurma
    smtp_server = "smtp.gmail.com"
    smtp_port = 587
    username = from_email
    password = getpass("E-posta şifresini girin: ")

    server = smtplib.SMTP(smtp_server, smtp_port)
    server.starttls()
    server.login(username, password)

    # E-postayı gönderme
    server.sendmail(from_email, to_email, msg.as_string())

    # Bağlantıyı kapatma
    server.quit()

# Hareket algılaması için video akışı başlatma
video_capture = cv2.VideoCapture(0)  # 0, varsayılan kamera

# İlk kareyi al
_, prev_frame = video_capture.read()

while True:
    # Şu anki kareyi al
    _, frame = video_capture.read()

    # Görüntü farkını hesapla
    diff = cv2.absdiff(prev_frame, frame)
    gray = cv2.cvtColor(diff, cv2.COLOR_BGR2GRAY)
    _, threshold = cv2.threshold(gray, 30, 255, cv2.THRESH_BINARY)

    # Hareket algılandı mı kontrol et
    if cv2.countNonZero(threshold) > 500:
        print("Hareket algılandı!")
        
        # Görüntüyü kaydet
        image_path = "captured_image.jpg"
        cv2.imwrite(image_path, frame)

        # E-posta gönderme
        send_email(image_path)

        # Bekleme süresi
        time.sleep(10)  # İstenirse bu süreyi ayarlayabilirsiniz

    # Önceki kareyi güncelle
    prev_frame = frame

    # Çıkış için 'q' tuşuna basma
    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

# Kamerayı serbest bırakma ve pencereyi kapatma
video_capture.release()
cv2.destroyAllWindows()
