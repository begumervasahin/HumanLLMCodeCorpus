import tensorflow as tf
import numpy as np

# Veri setini oluştur
np.random.seed(42)
X_train = np.random.rand(100, 1)
y_train = 3 * X_train + 2 + 0.1 * np.random.randn(100, 1)

# Modeli oluştur
model = tf.keras.Sequential([
    tf.keras.layers.Dense(1, input_shape=(1,), activation='linear', kernel_initializer='random_normal', bias_initializer='zeros')
])

# Modeli derle
model.compile(optimizer='sgd', loss='mean_squared_error')

# Modeli eğit
model.fit(X_train, y_train, epochs=100, verbose=2)

# Tahmin yap
X_test = np.array([[0.2], [0.5]])
predictions = model.predict(X_test)

print("Tahminler:")
for i in range(len(X_test)):
    print(f"Giriş: {X_test[i][0]}, Tahmin: {predictions[i][0]}")
