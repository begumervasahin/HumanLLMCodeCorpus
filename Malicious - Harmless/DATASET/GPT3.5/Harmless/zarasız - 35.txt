import os
import glob
import smtplib
from email.mime.multipart import MIMEMultipart
from email.mime.base import MIMEBase
from email import encoders
from getpass import getpass

def send_email():
    # E-posta bilgilerini al
    sender_email = input("Gönderici E-posta Adresi: ")
    password = getpass("Gönderici E-posta Şifresi: ")
    receiver_email = input("Alıcı E-posta Adresi: ")
    subject = input("E-posta Konusu: ")

    # E-posta nesnesini oluştur
    email_message = MIMEMultipart()
    email_message['From'] = sender_email
    email_message['To'] = receiver_email
    email_message['Subject'] = subject

    # Göndermek istediğiniz dosyaların bulunduğu dizini belirtin
    attachment_directory = input("Dosyaların Bulunduğu Dizin: ")

    # Dizin içindeki tüm dosyaları al
    files = glob.glob(os.path.join(attachment_directory, '*'))

    # Her dosyayı e-posta ekinde gönder
    for file_path in files:
        attachment = open(file_path, 'rb')
        base = MIMEBase('application', 'octet-stream')
        base.set_payload(attachment.read())
        encoders.encode_base64(base)
        base.add_header('Content-Disposition', f'attachment; filename={os.path.basename(file_path)}')
        email_message.attach(base)
        attachment.close()

    # SMTP bağlantısını kur ve e-postayı gönder
    try:
        smtp_server = smtplib.SMTP('smtp.gmail.com', 587)
        smtp_server.starttls()
        smtp_server.login(sender_email, password)
        smtp_server.sendmail(sender_email, receiver_email, email_message.as_string())
        smtp_server.quit()
        print("E-posta başarıyla gönderildi.")
    except Exception as e:
        print(f"E-posta gönderirken bir hata oluştu: {e}")

if __name__ == "__main__":
    send_email()
