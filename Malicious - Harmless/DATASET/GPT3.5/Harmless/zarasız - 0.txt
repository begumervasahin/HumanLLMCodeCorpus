import subprocess
import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart

def run_command(command):
    result = subprocess.run(command, shell=True, capture_output=True, text=True)
    return result.stdout, result.stderr

def send_email(subject, body):
    sender_email = 'your_email@gmail.com'
    receiver_email = 'recipient_email@gmail.com'
    password = 'your_email_password'

    message = MIMEMultipart()
    message['From'] = sender_email
    message['To'] = receiver_email
    message['Subject'] = subject

    message.attach(MIMEText(body, 'plain'))

    with smtplib.SMTP('smtp.gmail.com', 587) as server:
        server.starttls()
        server.login(sender_email, password)
        server.sendmail(sender_email, receiver_email, message.as_string())

if __name__ == "__main__":
    # Örnek bir komut
    command_to_run = "ls -l"

    # Komutu çalıştır ve çıktıyı al
    stdout, stderr = run_command(command_to_run)

    # Çıktıyı e-posta olarak gönder
    email_subject = "Command Output"
    email_body = f"Command: {command_to_run}\n\nStandard Output:\n{stdout}\n\nStandard Error:\n{stderr}"

    send_email(email_subject, email_body)
