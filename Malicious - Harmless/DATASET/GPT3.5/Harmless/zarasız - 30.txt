from pynput.mouse import Listener
import ctypes

def on_click(x, y, button, pressed):
    if pressed:
        print(f"Fare tuşuna basıldı: {button}")
        show_popup_message("Fare tuşuna basıldı")

def on_move(x, y):
    pass  # İsterseniz fare hareketlerini de takip edebilirsiniz

def show_popup_message(message):
    ctypes.windll.user32.MessageBoxW(0, message, "Bilgi", 1)

with Listener(on_click=on_click, on_move=on_move) as listener:
    listener.join()
