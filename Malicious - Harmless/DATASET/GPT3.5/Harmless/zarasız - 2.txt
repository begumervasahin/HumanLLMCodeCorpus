import keyboard
from pynput.mouse import Listener, Controller
import time
import matplotlib.pyplot as plt

# Çizim alanını temsil etmek için boş bir liste
drawn_points = []

def on_click(x, y, button, pressed):
    if pressed and button == mouse.Button.left:
        # Sol tıklama algılandığında fare koordinatlarını kaydet
        drawn_points.append((x, y))

def on_move(x, y):
    # Fare hareket ettiğinde koordinatları kaydet
    drawn_points.append((x, y))

mouse = Controller()

# Fare ve klavye dinleyicileri
with Listener(on_click=on_click, on_move=on_move) as listener:
    try:
        print("Çizim modu aktif. Çizim yapmak için fareyi kullanın.")
        print("Programı sonlandırmak için 'q' tuşuna basın.")
        
        # Klavyeden 'q' tuşuna basılana kadar devam et
        keyboard.wait("q")

    except KeyboardInterrupt:
        pass
    finally:
        # Dinleyicileri durdur
        listener.stop()

# Opsiyonel: Çizimi görselleştir
if drawn_points:
    x, y = zip(*drawn_points)
    plt.plot(x, y, marker='o')
    plt.title('Çizim')
    plt.xlabel('X Koordinatı')
    plt.ylabel('Y Koordinatı')
    plt.show()

print("Program sonlandı.")
