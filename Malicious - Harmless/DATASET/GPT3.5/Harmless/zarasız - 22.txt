import os
import socket
import requests
import smtplib
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText

def create_file():
    with open('example.txt', 'w') as file:
        file.write('Hello, this is a sample file.')

def read_file():
    with open('example.txt', 'r') as file:
        content = file.read()
        print(content)

def make_http_request(url):
    response = requests.get(url)
    if response.status_code == 200:
        print("HTTP GET request successful")
    else:
        print(f"HTTP GET request failed with status code: {response.status_code}")

def get_ip_address():
    hostname = socket.gethostname()
    ip_address = socket.gethostbyname(hostname)
    print(f"IP Address of the current machine: {ip_address}")

def send_email():
    from_address = 'your_email@gmail.com'
    to_address = 'recipient_email@gmail.com'
    subject = 'Test Email'
    body = 'This is a test email sent from Python.'

    msg = MIMEMultipart()
    msg['From'] = from_address
    msg['To'] = to_address
    msg['Subject'] = subject

    msg.attach(MIMEText(body, 'plain'))

    with smtplib.SMTP('smtp.gmail.com', 587) as server:
        server.starttls()
        server.login(from_address, 'your_password')
        server.sendmail(from_address, to_address, msg.as_string())
        print("Email sent successfully")

def test_connection():
    host = 'www.google.com'
    port = 80

    try:
        socket.create_connection((host, port), timeout=5)
        print(f"Connection to {host}:{port} successful")
    except OSError as e:
        print(f"Connection to {host}:{port} failed: {e}")

# Test functions
create_file()
read_file()
make_http_request('https://www.example.com')
get_ip_address()
send_email()
test_connection()
