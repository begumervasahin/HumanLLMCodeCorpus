import socket
from PIL import ImageGrab
import io

# Bağlanılacak IP adresi ve portu ayarlayın
server_ip = "localhost"  # Sunucu IP adresi
server_port = 12345  # Sunucu port numarası

# Soket oluşturun
client_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)

# Belirtilen IP ve port'a bağlanın
client_socket.connect((server_ip, server_port))

# Ekran görüntüsü alın
screenshot = ImageGrab.grab()

# Görüntüyü bir byte dizisine dönüştürün
image_byte_array = io.BytesIO()
screenshot.save(image_byte_array, format='PNG')
image_data = image_byte_array.getvalue()

# Veriyi gönderin
client_socket.send(image_data)

# Soketi kapatın
client_socket.close()
