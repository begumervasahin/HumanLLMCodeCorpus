import os
import shutil

def dosya_tasima(dizin, hedef_dizin, dosya_turu):
    try:
        # Dizin içindeki dosyaları listele
        dosyalar = os.listdir(dizin)

        # Hedef dizin kontrolü, eğer yoksa oluştur
        if not os.path.exists(hedef_dizin):
            os.makedirs(hedef_dizin)

        for dosya in dosyalar:
            # Belirli dosya türlerini kontrol et
            if dosya.endswith(dosya_turu):
                dosya_yolu = os.path.join(dizin, dosya)
                hedef_yol = os.path.join(hedef_dizin, dosya)

                # Dosyayı hedef dizine taşı
                shutil.move(dosya_yolu, hedef_yol)
                print(f"{dosya} taşındı.")

        print("Tüm dosyalar başarıyla taşındı.")
    except Exception as e:
        print(f"Hata oluştu: {e}")

# Kullanım örneği
dosya_tasima("kaynak_dizin", "hedef_dizin", ".txt")
