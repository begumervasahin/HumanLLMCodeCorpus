import os
from selenium import webdriver

def main():
    # Tarayıcı sürücüsünün yolu
    driver_path = "/path/to/chromedriver"  # chromedriver veya farklı bir tarayıcı sürücüsü seçebilirsiniz

    # Tarayıcı sürücüsünü başlatma
    driver = webdriver.Chrome(executable_path=driver_path)

    try:
        # Web sitesine gidin (Örnek olarak OpenAI'nin web sitesini kullanıyoruz)
        url = "https://www.openai.com"
        driver.get(url)

        # Sayfanın başlığını alın
        page_title = driver.title
        print(f"Current page title: {page_title}")

    finally:
        # Tarayıcıyı kapatma
        driver.quit()

if __name__ == "__main__":
    main()
