import pyautogui
import time

def main():
    try:
        print("Bu program, fare imlecini belirli bir konuma taşıyacak ve ardından bir metin girişi yapacaktır.")
        
        # Fare imlecini belirli bir konuma taşı
        target_position = (500, 500)
        pyautogui.moveTo(target_position, duration=1)
        print(f"Fare imleci {target_position} konumuna taşındı.")
        
        # Kullanıcıya bilgilendirme yap
        print("Lütfen birkaç saniye bekleyin...")

        # Bekleme süresi
        time.sleep(2)

        # Metin girişi yap
        input_text = "Merhaba, GPT-3.5!"
        pyautogui.typewrite(input_text)
        print(f"Metin girişi yapıldı: '{input_text}'")

        # İşlem tamamlandı mesajı
        print("İşlem tamamlandı!")

    except Exception as e:
        print(f"Hata oluştu: {str(e)}")

if __name__ == "__main__":
    main()
