import platform
import psutil

def get_system_info():
    # Platform bilgilerini al
    system_info = platform.uname()
    
    print("Sistem Bilgileri:")
    print(f"İşletim Sistemi: {system_info.system}")
    print(f"Sistem Adı: {system_info.node}")
    print(f"İşletim Sistemi Sürümü: {system_info.version}")
    print(f"İşletim Sistemi Mimarisi: {system_info.machine}")

def get_cpu_usage():
    # CPU kullanımını al
    cpu_usage = psutil.cpu_percent(interval=1)
    
    print("\nCPU Kullanımı:")
    print(f"Toplam CPU Kullanımı: {cpu_usage}%")

def get_memory_usage():
    # Bellek kullanımını al
    memory = psutil.virtual_memory()
    
    print("\nBellek Kullanımı:")
    print(f"Toplam Bellek: {memory.total / (1024 ** 3):.2f} GB")
    print(f"Kullanılan Bellek: {memory.used / (1024 ** 3):.2f} GB")
    print(f"Boş Bellek: {memory.available / (1024 ** 3):.2f} GB")

if __name__ == "__main__":
    get_system_info()
    get_cpu_usage()
    get_memory_usage()
