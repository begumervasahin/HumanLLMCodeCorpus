import time
import random
import ctypes

def main():
    print("Program başladı.")

    # Rastgele bir süre bekleyelim
    wait_time = random.uniform(1, 5)
    print(f"Biraz bekleyelim... ({wait_time:.2f} saniye)")
    time.sleep(wait_time)

    # Rastgele sayılar üreterek matematiksel işlemler yapalım
    num1 = random.randint(1, 10)
    num2 = random.randint(1, 10)
    print(f"Üretilen sayılar: {num1}, {num2}")

    # Toplama işlemi
    result_add = num1 + num2
    print(f"Toplama: {num1} + {num2} = {result_add}")

    # Çıkarma işlemi
    result_sub = num1 - num2
    print(f"Çıkarma: {num1} - {num2} = {result_sub}")

    # Çarpma işlemi
    result_mul = num1 * num2
    print(f"Çarpma: {num1} * {num2} = {result_mul}")

    # Bölme işlemi
    if num2 != 0:
        result_div = num1 / num2
        print(f"Bölme: {num1} / {num2} = {result_div}")
    else:
        print("Bölme işlemi geçersiz (sayı2 sıfır olamaz).")

    print("Program sona erdi.")

if __name__ == "__main__":
    main()
