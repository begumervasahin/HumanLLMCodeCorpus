import pandas as pd

# Read the ratings data
ratings_cols = ['user_id', 'movie_id', 'rating']
ratings = pd.read_csv('u.data', sep='\t', names=ratings_cols, usecols=range(3))

# Read the movies data
movies_cols = ['movie_id', 'title']
movies = pd.read_csv('u.item', sep='|', names=movies_cols, usecols=range(2))

# Merge the ratings and movies data
ratings = pd.merge(movies, ratings)

# Create a pivot table of user ratings
user_ratings = ratings.pivot_table(index='user_id', columns='title', values='rating')

# Calculate the correlation matrix of movie ratings
corr_matrix = user_ratings.corr(method='pearson', min_periods=100)

# Define the user ID and retrieve their ratings
user_id = 2
my_ratings = user_ratings.loc[user_id].dropna()

# Find similar movies based on ratings
sim_candidates = pd.Series()
for i in range(len(my_ratings.index)):
    similar_movies = corr_matrix[my_ratings.index[i]].dropna()
    similar_movies = similar_movies.map(lambda x: x * my_ratings[i])
    sim_candidates = sim_candidates.append(similar_movies)

# Aggregate the similar candidates
sim_candidates.sort_values(inplace=True, ascending=False)
sim_candidates = sim_candidates.groupby(sim_candidates.index).sum()
sim_candidates.sort_values(inplace=True, ascending=False)

# Filter out movies that the user has already rated
filtered_sims = sim_candidates.drop(my_ratings.index, errors='ignore')

# Print and save the filtered similarities
print(filtered_sims.head())
filtered_sims.to_csv("filteredSims.csv")
