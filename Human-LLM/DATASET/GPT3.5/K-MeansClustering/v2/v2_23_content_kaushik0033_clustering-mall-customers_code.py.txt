import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from sklearn.cluster import KMeans
from sklearn.preprocessing import LabelEncoder
import scipy.cluster.hierarchy as sch

# Load the dataset
data = pd.read_csv("path_to_your_file.csv")  # Replace "path_to_your_file.csv" with the actual path to your CSV file
data.columns = ['cid', 'gender', 'age', 'income', 'spend']

# Select the features for clustering
X = data[['income', 'spend']]

# KMeans clustering
num_clusters = 6
km = KMeans(n_clusters=num_clusters, random_state=0)
km.fit(X)

# Print cluster centers and number of labels
print("Cluster Centers:")
print(km.cluster_centers_)
print("Number of Labels:", len(km.labels_))

# Plot the cluster centers
plt.scatter(km.cluster_centers_[:, 0], km.cluster_centers_[:, 1], color='black')        
plt.title('Cluster Centers')
plt.xlabel('Income')
plt.ylabel('Spend')
plt.show()

# Encode categorical variable 'gender'
label_encoder = LabelEncoder()
data['gender'] = label_encoder.fit_transform(data['gender'])

# Hierarchical clustering
linked = sch.linkage(X, 'single')

# Plot the dendrogram
plt.figure(figsize=(10, 10))
sch.dendrogram(linked)
plt.title('Dendrogram')
plt.xlabel('Customer')
plt.ylabel('Euclidean Distance')
plt.show()
