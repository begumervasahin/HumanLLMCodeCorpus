import sys
import spotipy
import spotipy.util as util

def get_spotify_access_token(scope):
    """
    Get Spotify access token for the specified scope using user credentials.
    
    Parameters:
    - scope (str): The scope of access to the Spotify account.
    
    Returns:
    - username (str): The username of the Spotify account.
    - token (str): The access token for accessing the Spotify API.
    """
    if len(sys.argv) > 1:
        username = sys.argv[1]
    else:
        print("Please provide the username as an argument.")
        sys.exit()
        
    token = util.prompt_for_user_token(username, scope)
    return username, token

def get_liked_songs(sp):
    """
    Retrieve a list of liked songs from the user's Spotify account.
    
    Parameters:
    - sp (Spotify instance): The Spotify instance authenticated with an access token.
    
    Returns:
    - liked_songs (list): A list of dictionaries representing liked songs.
    """
    # Get initial batch of liked songs
    results = sp.current_user_saved_tracks(limit=50)
    liked_songs = results['items']
    
    # Paginate through results to retrieve all liked songs
    while results['next']:
        results = sp.next(results)
        liked_songs.extend(results['items'])
    
    return liked_songs

if __name__ == "__main__":
    # Set the scope for accessing Spotify (e.g., 'user-library-read' for reading user's library).
    scope = 'user-library-read'
    
    # Retrieve Spotify access token
    username, token = get_spotify_access_token(scope)
    
    if token:
        # Initialize Spotify client
        spotify_client = spotipy.Spotify(auth=token)
        
        # Get liked songs
        liked_songs = get_liked_songs(spotify_client)
        
        # Display liked songs
        print("Liked songs:")
        for idx, song in enumerate(liked_songs, 1):
            track_info = song['track']
            artists = ', '.join(artist['name'] for artist in track_info['artists'])
            print(f"{idx}. {track_info['name']} by {artists}")
    else:
        print("Failed to obtain access token for", username)
