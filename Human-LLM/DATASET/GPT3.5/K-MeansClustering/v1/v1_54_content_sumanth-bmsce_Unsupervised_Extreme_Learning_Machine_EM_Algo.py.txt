from sklearn import mixture
import matplotlib.pyplot as plt
import numpy as np
from itertools import groupby

def em(input_array, no_of_clusters):
    # Initialize Gaussian Mixture Model
    model = mixture.GaussianMixture(n_components=no_of_clusters, covariance_type='diag')
    
    # Fit the model to the input data
    model.fit(input_array)
    
    # Predict the cluster labels
    em = model.predict(input_array)
    
    # Print means and weights of the Gaussian components
    print("Means:", model.means_)
    print("Weights:", model.weights_)
    
    # Separate cluster labels for different classes
    l1 = em[:59]
    l1.sort()
    print("Cluster labels for class 1:", l1)
    print("Lengths of clusters for class 1:", [len(list(group)) for key, group in groupby(l1)])
    
    l2 = em[59:130]
    l2.sort()
    print("Cluster labels for class 2:", l2)
    print("Lengths of clusters for class 2:", [len(list(group)) for key, group in groupby(l2)])
    
    l3 = em[130:178]
    l3.sort()
    print("Cluster labels for class 3:", l3)
    print("Lengths of clusters for class 3:", [len(list(group)) for key, group in groupby(l3)])

try:
    input_data = np.genfromtxt("C:\\Users\\SUMANTH C\\Desktop\\Deep Learning\\Datasets\\wine_sort.csv", delimiter=',')
except:
    print("Could not open file")

# Select only the first 13 columns of the input data
input_data = input_data[:, :13]
print("Input data shape:", input_data.shape)

# Call the em function
em(input_data, 3)
