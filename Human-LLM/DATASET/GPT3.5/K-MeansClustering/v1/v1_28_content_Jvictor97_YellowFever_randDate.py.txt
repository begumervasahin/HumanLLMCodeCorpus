import random
import time
from datetime import timedelta, datetime

def strTimeProp(start, end, format, prop):
    """Get a time at a proportion of a range of two formatted times.
    start and end should be strings specifying times formatted in the
    given format (strftime-style), giving an interval [start, end].
    prop specifies how a proportion of the interval to be taken after
    start.  The returned time will be in the specified format.
    """
    stime = time.mktime(time.strptime(start, format))
    etime = time.mktime(time.strptime(end, format))
    ptime = stime + prop * (etime - stime)
    return time.strftime(format, time.localtime(ptime))

def randomDateHot():
    """Generate a random date between January 1 and June 30."""
    return strTimeProp("2018-01-01", "2018-06-30", '%Y-%m-%d', random.random())

def randomDateCold():
    """Generate a random date between July 1 and December 31."""
    return strTimeProp("2018-07-01", "2018-12-31", '%Y-%m-%d', random.random())

def randomDateP(start, delta):
    """Generate a random date starting from the given start date
    and adding a random number of days up to the specified delta."""
    start_date = datetime.strptime(start, '%Y-%m-%d').date()
    return strTimeProp(start, str(start_date + timedelta(days=delta)), '%Y-%m-%d', random.random())

# Example usage
print("Random date between January 1 and June 30:", randomDateHot())
print("Random date between July 1 and December 31:", randomDateCold())
print("Random date with custom start date and delta:", randomDateP("2022-01-01", 100))
