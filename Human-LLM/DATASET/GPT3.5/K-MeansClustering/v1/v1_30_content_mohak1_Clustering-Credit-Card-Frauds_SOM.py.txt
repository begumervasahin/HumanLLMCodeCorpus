import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.preprocessing import MinMaxScaler
from minisom import MiniSom

# Read the dataset
ds = pd.read_csv('G:\Credit_Card_Applications.csv')

# Prepare the features and target variable
x = ds.iloc[:, :-1].values
y = ds.iloc[:, -1].values

# Scale the features
sc = MinMaxScaler(feature_range=(0, 1))
x = sc.fit_transform(x)

# Initialize and train the SOM
som = MiniSom(x=10, y=10, input_len=15, sigma=1, learning_rate=0.5)
som.random_weights_init(x)
som.train_random(x, 100)

# Visualize the SOM
plt.figure(figsize=(8, 8))
plt.bone()
plt.pcolor(som.distance_map().T)
plt.colorbar()

markers = ['o', 's']
colors = ['r', 'g']
for i, j in enumerate(x):
    w = som.winner(j)
    plt.plot(w[0] + 0.5, w[1] + 0.5, markers[y[i]], markeredgecolor=colors[y[i]], markerfacecolor='None', markersize=10, markeredgewidth=2)

plt.title('Self Organizing Map')
plt.show()

# Find potential frauds
mappings = som.win_map(x)
frauds = np.concatenate((mappings[(5, 2)], mappings[(5, 7)], mappings[(5, 5)]), axis=0)
frauds = sc.inverse_transform(frauds)
print("Potential frauds:", frauds)
