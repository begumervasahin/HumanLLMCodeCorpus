import sys
import numpy as np

# Initialize variables
outputMatrix = []
sumForCentroid = []
newCluster = -2
lenMatrix = -1

# Read input from standard input (stdin) line by line
for line in sys.stdin:
    # Remove leading/trailing whitespace from the line
    line = line.strip()

    # Split the line into cluster, index, and matrix string
    cluster, index, matrixString = line.split('\t')

    # Convert matrix string to a list of float values
    matrix = [float(x) for x in matrixString.split(',')]
    matrix = np.array(matrix)  # Convert the list to a numpy array

    # Update the length of the matrix
    lenMatrix = len(matrix)

    # Convert cluster and index to integer values
    cluster = int(cluster)
    index = int(index)

    # Check if a new cluster is encountered
    if newCluster != cluster:
        # If it's not the first cluster, calculate and print the centroid of the previous cluster
        if newCluster != -2:
            newCentroid = sumForCentroid / len(outputMatrix)
            print('%s\t%s\t%s' % (newCluster, ','.join(map(str, outputMatrix)), ','.join(map(str, newCentroid))))

        # Reset outputMatrix and sumForCentroid for the new cluster
        outputMatrix = []
        sumForCentroid = np.zeros(lenMatrix)
        newCluster = cluster

    # Add index to outputMatrix and update sumForCentroid
    outputMatrix.append(index)
    sumForCentroid += matrix

# Calculate and print the centroid of the last cluster
newCentroid = sumForCentroid / len(outputMatrix)
print('%s\t%s\t%s' % (newCluster, ','.join(map(str, outputMatrix)), ','.join(map(str, newCentroid))))
