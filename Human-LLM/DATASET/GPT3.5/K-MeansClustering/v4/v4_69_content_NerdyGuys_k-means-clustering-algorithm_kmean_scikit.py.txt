from sklearn.cluster import KMeans
import numpy as np
import matplotlib.pyplot as plt

def kmeans_display(X, label):
    """
    Display the clustered data points.

    Args:
        X (numpy.ndarray): Data points.
        label (numpy.ndarray): Cluster labels.
    """
    # Determine the number of clusters
    K = np.amax(label) + 1

    # Separate data points into clusters
    clusters = [X[label == k] for k in range(K)]
    colors = ['b', 'g', 'r']  # Blue, Green, Red
    markers = ['^', 'o', 's']  # Triangle, Circle, Square

    # Plot each cluster with different color and marker
    for k, cluster in enumerate(clusters):
        plt.plot(cluster[:, 0], cluster[:, 1], color=colors[k], marker=markers[k], markersize=4, alpha=0.8)

    plt.axis('equal')
    plt.show()

# Generate synthetic data
means = [[2, 2], [8, 3], [3, 6]]
cov = [[1, 0], [0, 1]]
N = 500
X0 = np.random.multivariate_normal(means[0], cov, N)
X1 = np.random.multivariate_normal(means[1], cov, N)
X2 = np.random.multivariate_normal(means[2], cov, N)
X = np.concatenate((X0, X1, X2), axis=0)

# Perform KMeans clustering
K = 3
kmeans = KMeans(n_clusters=K, random_state=0).fit(X)
print('Centers found by scikit-learn:')
print(kmeans.cluster_centers_)

# Predict cluster labels
pred_label = kmeans.predict(X)

# Display clustered data points
kmeans_display(X, pred_label)
