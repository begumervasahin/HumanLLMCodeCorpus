import numpy as np
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
import pandas as pd

# Read dataset
dataset = pd.read_csv("Churn_Modelling.csv")

# Select features for clustering
X = dataset.iloc[:, [8, 12]].values

# Calculate WCSS (Within-Cluster Sum of Squares) for different numbers of clusters
wcss = []
for num_clusters in range(1, 16):
    kmeans = KMeans(n_clusters=num_clusters, init='k-means++', random_state=0)
    kmeans.fit(X)
    wcss.append(kmeans.inertia_)

# Plot the Elbow Method graph to determine the optimal number of clusters
plt.plot(range(1, 16), wcss)
plt.title('Elbow Method')
plt.xlabel('Number Of Clusters')
plt.ylabel('WCSS')
plt.show()

# Apply KMeans clustering with the optimal number of clusters
kmeans = KMeans(n_clusters=4, init='k-means++', random_state=0)
cluster_labels = kmeans.fit_predict(X)

# Plot the clusters
plt.scatter(X[cluster_labels == 0, 0], X[cluster_labels == 0, 1], s=25, c='red', label='Cluster 1')
plt.scatter(X[cluster_labels == 1, 0], X[cluster_labels == 1, 1], s=25, c='blue', label='Cluster 2')
plt.scatter(X[cluster_labels == 2, 0], X[cluster_labels == 2, 1], s=25, c='green', label='Cluster 3')
plt.scatter(X[cluster_labels == 3, 0], X[cluster_labels == 3, 1], s=25, c='pink', label='Cluster 4')
plt.scatter(kmeans.cluster_centers_[:, 0], kmeans.cluster_centers_[:, 1], s=25, c='yellow', label='Centroids')

plt.title('KMeans Clustering')
plt.xlabel('Balance')
plt.ylabel('Estimated Salary in $')
plt.legend()
plt.show()
