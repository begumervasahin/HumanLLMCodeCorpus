import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
from sklearn import preprocessing
from sklearn.metrics import silhouette_score
from settings import DATASET_DIR_PATH

# Set random seed for reproducibility
np.random.seed(42)

# Load the dataset
data = pd.read_csv(DATASET_DIR_PATH + '/3d_dataset_clustering.csv')

# Normalize the data
normalizer = preprocessing.Normalizer()
data_normalized = pd.DataFrame(normalizer.fit_transform(data))

# Calculate SSE for different values of k
sse = {}
for k in range(2, 11):
    kmeans = KMeans(n_clusters=k, max_iter=1000).fit(data_normalized)
    sse[k] = kmeans.inertia_

# Plot the Elbow Criterion
plt.figure()
plt.plot(list(sse.keys()), list(sse.values()))
plt.xlabel("Number of Clusters")
plt.ylabel("SSE")
plt.title('Elbow Criterion')
plt.show()

# Evaluate Silhouette Score for different values of k
for n_cluster in range(2, 11):
    kmeans = KMeans(n_clusters=n_cluster).fit(data_normalized)
    labels = kmeans.labels_
    sil_coeff = silhouette_score(data_normalized, labels, metric='euclidean', sample_size=1000)     
    print(f"For n_clusters={n_cluster}, The Silhouette Coefficient is {sil_coeff}")
