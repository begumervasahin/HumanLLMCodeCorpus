from sklearn import mixture
import matplotlib.pyplot as plt
import numpy as np
from itertools import groupby

def em(input_array, no_of_clusters):
    # Initialize Gaussian Mixture Model
    model = mixture.GaussianMixture(n_components=no_of_clusters, covariance_type='diag')
    
    # Fit the model to the input data
    model.fit(input_array)
    
    # Print means and weights of the Gaussian components
    print("Gaussian Means:", model.means_)
    print("Gaussian Weights:", model.weights_)
    
    # Predict cluster labels
    cluster_labels = model.predict(input_array)
    
    # Print cluster labels and lengths for each class
    print_cluster_stats(cluster_labels[:59], "Class 1")
    print_cluster_stats(cluster_labels[59:130], "Class 2")
    print_cluster_stats(cluster_labels[130:], "Class 3")

def print_cluster_stats(cluster_labels, class_name):
    # Sort cluster labels
    cluster_labels.sort()
    
    # Print sorted cluster labels
    print("Cluster labels for", class_name, ":", cluster_labels)
    
    # Calculate and print lengths of clusters
    cluster_lengths = [len(list(group)) for _, group in groupby(cluster_labels)]
    print("Lengths of clusters for", class_name, ":", cluster_lengths)
