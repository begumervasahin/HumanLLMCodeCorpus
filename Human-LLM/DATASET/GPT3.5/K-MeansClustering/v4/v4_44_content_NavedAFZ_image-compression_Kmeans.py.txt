import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from skimage import io
from sklearn.cluster import KMeans
import imageio

# Load the image
image = io.imread('nass.jpg')

# Display the original image
io.imshow(image)
io.show()

# Get the dimensions of the image
rows, cols, _ = image.shape

# Normalize the image
image = image / 255

# Reshape the image for KMeans clustering
X = image.reshape(image.shape[0] * image.shape[1], 3)

# Fit KMeans clustering with 15 clusters
kmeans = KMeans(n_clusters=15, init='k-means++')
y_kmeans = kmeans.fit_predict(X)

# Get cluster centers and assign each pixel to its cluster center
y_pred = kmeans.predict(X)
cluster_centers = kmeans.cluster_centers_
X_recovered = cluster_centers[y_pred]

# Reshape the compressed image
X_recovered = np.reshape(X_recovered, (rows, cols, 3))

# Save the compressed image
imageio.imwrite('nas1.jpg', X_recovered)

# Display the compressed image
image_compressed = io.imread('nas1.jpg')
io.imshow(image_compressed)
io.show()
