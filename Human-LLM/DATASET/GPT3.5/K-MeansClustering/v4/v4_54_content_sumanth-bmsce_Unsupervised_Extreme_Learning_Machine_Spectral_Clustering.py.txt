from sklearn.cluster import SpectralClustering
import numpy as np
from itertools import groupby

def spectral_clustering(data, num_clusters):
    # Convert data to a numpy array
    data = np.array(data)
    
    # Perform spectral clustering
    spectral = SpectralClustering(
        n_clusters=num_clusters,
        eigen_solver='arpack',
        affinity="nearest_neighbors"
    ).fit_predict(data)
    
    # Print cluster labels and lengths for each class
    print_cluster_stats(spectral[:59], "Class 1")
    print_cluster_stats(spectral[59:130], "Class 2")
    print_cluster_stats(spectral[130:178], "Class 3")

def print_cluster_stats(cluster_labels, class_name):
    # Sort cluster labels
    cluster_labels.sort()
    
    # Print sorted cluster labels
    print("Cluster labels for", class_name, ":", cluster_labels)
    
    # Calculate and print lengths of clusters
    cluster_lengths = [len(list(group)) for _, group in groupby(cluster_labels)]        
    print("Lengths of clusters for", class_name, ":", cluster_lengths)
