import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from skimage import io
from sklearn.cluster import KMeans
import imageio

def load_image(file_path):
    """Load an image from the given file path."""
    return io.imread(file_path)

def display_image(image):
    """Display the given image."""
    io.imshow(image)
    io.show()

def normalize_image(image):
    """Normalize the pixel values of the given image."""
    return image / 255

def reshape_image(image):
    """Reshape the given image for KMeans clustering."""
    return image.reshape(image.shape[0] * image.shape[1], 3)

def kmeans_clustering(X, num_clusters=15):
    """Perform KMeans clustering on the given data."""
    kmeans = KMeans(n_clusters=num_clusters, init='k-means++')
    return kmeans.fit_predict(X), kmeans.cluster_centers_

def compress_image(X, cluster_labels, cluster_centers, rows, cols):
    """Compress the image using the KMeans cluster centers."""
    compressed_pixels = cluster_centers[cluster_labels]
    return np.reshape(compressed_pixels, (rows, cols, 3))

def save_image(image, file_path):
    """Save the given image to the specified file path."""
    imageio.imwrite(file_path, image)

def main():
    # Load the original image
    original_image = load_image('nass.jpg')
    display_image(original_image)

    # Get the dimensions of the image
    rows, cols, _ = original_image.shape

    # Normalize the image
    normalized_image = normalize_image(original_image)

    # Reshape the image for KMeans clustering
    X = reshape_image(normalized_image)

    # Perform KMeans clustering
    cluster_labels, cluster_centers = kmeans_clustering(X)

    # Compress the image using the cluster centers
    compressed_image = compress_image(X, cluster_labels, cluster_centers, rows, cols)

    # Save and display the compressed image
    save_image(compressed_image, 'nas1.jpg')
    compressed_image = load_image('nas1.jpg')
    display_image(compressed_image)

if __name__ == "__main__":
    main()
