from sklearn.cluster import KMeans
import numpy as np
import matplotlib.pyplot as plt

def plot_clusters(X, labels):
    """
    Display clustered data points.

    Args:
        X (numpy.ndarray): Data points.
        labels (numpy.ndarray): Cluster labels.
    """
    # Determine the number of clusters
    num_clusters = np.amax(labels) + 1

    # Define colors and markers for each cluster
    colors = ['b', 'g', 'r']  # Blue, Green, Red
    markers = ['^', 'o', 's']  # Triangle, Circle, Square

    # Plot each cluster with a different color and marker
    for k in range(num_clusters):
        cluster_points = X[labels == k]
        plt.plot(cluster_points[:, 0], cluster_points[:, 1], color=colors[k], marker=markers[k], markersize=4, alpha=0.8)

    plt.axis('equal')
    plt.show()

# Generate synthetic data
means = [[2, 2], [8, 3], [3, 6]]
cov = [[1, 0], [0, 1]]
N = 500
X0 = np.random.multivariate_normal(means[0], cov, N)
X1 = np.random.multivariate_normal(means[1], cov, N)
X2 = np.random.multivariate_normal(means[2], cov, N)
X = np.concatenate((X0, X1, X2), axis=0)

# Perform KMeans clustering
num_clusters = 3
kmeans = KMeans(n_clusters=num_clusters, random_state=0).fit(X)
print('Centers found by scikit-learn:')
print(kmeans.cluster_centers_)

# Predict cluster labels
predicted_labels = kmeans.predict(X)

# Display clustered data points
plot_clusters(X, predicted_labels)
