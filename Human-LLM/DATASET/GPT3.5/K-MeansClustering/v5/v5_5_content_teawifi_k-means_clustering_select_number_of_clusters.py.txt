import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
from sklearn import preprocessing
from sklearn.metrics import silhouette_score
from settings import DATASET_DIR_PATH

def load_dataset(file_path):
    """Load dataset from the specified file path."""
    return pd.read_csv(file_path)

def normalize_data(data):
    """Normalize the dataset."""
    normalizer = preprocessing.Normalizer()
    return pd.DataFrame(normalizer.fit_transform(data))

def calculate_sse(data, max_clusters=10):
    """Calculate Sum of Squared Errors (SSE) for different values of k."""
    sse = {}
    for k in range(2, max_clusters + 1):
        kmeans = KMeans(n_clusters=k, max_iter=1000).fit(data)
        sse[k] = kmeans.inertia_
    return sse

def plot_elbow_criterion(sse):
    """Plot the Elbow Criterion based on SSE."""
    plt.figure()
    plt.plot(list(sse.keys()), list(sse.values()))
    plt.xlabel("Number of Clusters")
    plt.ylabel("SSE")
    plt.title('Elbow Criterion')
    plt.show()

def evaluate_silhouette_score(data, max_clusters=10):
    """Evaluate Silhouette Score for different values of k."""
    for n_cluster in range(2, max_clusters + 1):
        kmeans = KMeans(n_clusters=n_cluster).fit(data)
        labels = kmeans.labels_
        sil_coeff = silhouette_score(data, labels, metric='euclidean', sample_size=1000)
        print(f"For n_clusters={n_cluster}, The Silhouette Coefficient is {sil_coeff}")

if __name__ == "__main__":
    # Set random seed for reproducibility
    np.random.seed(42)
    
    # Load and normalize the dataset
    data = load_dataset(DATASET_DIR_PATH + '/3d_dataset_clustering.csv')
    data_normalized = normalize_data(data)
    
    # Calculate and plot SSE for different values of k
    sse = calculate_sse(data_normalized)
    plot_elbow_criterion(sse)
    
    # Evaluate Silhouette Score for different values of k
    evaluate_silhouette_score(data_normalized)
