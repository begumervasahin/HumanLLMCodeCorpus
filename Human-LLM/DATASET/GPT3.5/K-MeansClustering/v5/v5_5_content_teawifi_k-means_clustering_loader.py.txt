import os
import re
import pandas as pd
from settings import ENCODING

def load_data(directory_path):
    """
    Load data from text files in the specified directory.

    Args:
        directory_path (str): Path to the directory containing text files.

    Returns:
        list: List of filtered data.
    """
    data = []

    # Get list of files in the directory
    file_paths = [os.path.join(directory_path, file) for file in os.listdir(directory_path)
                  if os.path.isfile(os.path.join(directory_path, file)) and file.endswith(".txt")]

    # Read each file and extract its content
    for file_path in file_paths:
        with open(file_path, "r", encoding=ENCODING) as file:        
            content = file.read().split('\n')
            data.extend(content)

    # Filter and process the data
    filtered_data = [filter_data(item) for item in data]
    return filtered_data

def filter_data(text):
    """
    Filter numeric data from a string and return the first three numbers.

    Args:
        text (str): Input string containing numeric data.

    Returns:
        list: List of the first three numbers found in the input string.
    """
    pattern = "[0-9]+"
    numbers = re.findall(pattern, text)
    numbers = [float(num) for num in numbers]
    return numbers[:3]

def save_to_csv(data, file_path):
    """
    Save filtered data to a CSV file.

    Args:
        data (list): List of filtered data.
        file_path (str): Path to save the CSV file.
    """
    # Create a DataFrame from the data and save to CSV
    df = pd.DataFrame(data, columns=['x', 'y', 'z'])
    df.to_csv(file_path, index=False, encoding=ENCODING, header=['x', 'y', 'z'])

# Example usage:
if __name__ == "__main__":
    directory_path = "path/to/directory"  # Replace with the actual directory path      
    output_file_path = "output.csv"  # Specify the output CSV file path
    filtered_data = load_data(directory_path)
    save_to_csv(filtered_data, output_file_path)
