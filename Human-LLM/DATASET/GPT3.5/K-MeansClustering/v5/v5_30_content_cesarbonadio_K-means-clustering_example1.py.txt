import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans

# Define the data points
data = {
    'x': [1, 2, 4, 7, 5],
    'y': [1, 1, 5, 7, 7]
}

# Create a DataFrame
df = pd.DataFrame(data)

# Perform KMeans clustering with 2 clusters
kmeans = KMeans(n_clusters=2)
kmeans.fit(df)

# Get the cluster labels and centroids
labels = kmeans.predict(df)
centroids = kmeans.cluster_centers_
print("Centroids:", centroids)

# Plot the clusters and centroids
fig = plt.figure(figsize=(5, 5))
# Define colors for each cluster
colors = {0: 'red', 1: 'green'}
# Assign colors to each data point based on cluster labels
point_colors = [colors[label] for label in labels]
plt.scatter(df['x'], df['y'], color=point_colors, alpha=0.3)  # Plot data points
# Plot centroids with distinct markers
for idx, centroid in enumerate(centroids):
    plt.scatter(*centroid, color=colors[idx], marker='x', label=f'Cluster {idx + 1}')  # Plot centroids
plt.xlim(0, 10)
plt.ylim(0, 10)
plt.xlabel('x')
plt.ylabel('y')
plt.title('KMeans Clustering')
plt.legend()
plt.show()
