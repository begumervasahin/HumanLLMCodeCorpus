import random
import time
from datetime import timedelta, datetime

def get_time_prop(start, end, format_str, prop):
    """Get a time at a proportion of a range of two formatted times.

    Args:
        start (str): Start time formatted as specified in format_str.
        end (str): End time formatted as specified in format_str.
        format_str (str): Format of the start and end times (strftime-style).
        prop (float): Proportion of the interval to be taken after start.

    Returns:
        str: A formatted time at the specified proportion of the interval.
    """
    start_time = time.mktime(time.strptime(start, format_str))
    end_time = time.mktime(time.strptime(end, format_str))
    prop_time = start_time + prop * (end_time - start_time)
    return time.strftime(format_str, time.localtime(prop_time))

def random_date_hot():
    """Generate a random date between January 1 and June 30."""
    return get_time_prop("2018-01-01", "2018-06-30", '%Y-%m-%d', random.random())

def random_date_cold():
    """Generate a random date between July 1 and December 31."""
    return get_time_prop("2018-07-01", "2018-12-31", '%Y-%m-%d', random.random())

def random_date_plus_delta(start, delta):
    """Generate a random date starting from the given start date
    and adding a random number of days up to the specified delta."""
    start_date = datetime.strptime(start, '%Y-%m-%d').date()
    end_date = start_date + timedelta(days=delta)
    return get_time_prop(start, str(end_date), '%Y-%m-%d', random.random())
