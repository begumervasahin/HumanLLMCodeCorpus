import numpy as np
from Bio import SeqIO
from Bio import pairwise2 as pw
import random

# Load DNA sequences from a FASTA file
sequences = list(SeqIO.parse("DNASequences.fasta", "fasta"))

# Define alignment parameters
penalty = -1
reward = 1

# Define a function to calculate the score for pairwise alignment
def calculate_score(n1, n2):
    return reward if n1 == n2 else penalty

# Choose k random sequences as cluster representatives
num_clusters = 4
cluster_indices = random.sample(range(len(sequences)), num_clusters)
clusters = np.full(shape=len(sequences), fill_value=-1, dtype=int)
for i, idx in enumerate(cluster_indices):
    clusters[idx] = i

# Perform pairwise alignment between all pairs of sequences and calculate distances
num_sequences = len(sequences)
distances_matrix = np.zeros(shape=(num_sequences, num_sequences))
for i in range(num_sequences):
    for j in range(i + 1, num_sequences):
        alignment_results = pw.align.globalxx(sequences[i].seq, sequences[j].seq)
        alignment_score = alignment_results[0][2]
        distances_matrix[i, j] = distances_matrix[j, i] = alignment_score

# Save distances matrix to a binary file
with open("temp.bin", 'wb') as file:
    np.save(file, distances_matrix)
