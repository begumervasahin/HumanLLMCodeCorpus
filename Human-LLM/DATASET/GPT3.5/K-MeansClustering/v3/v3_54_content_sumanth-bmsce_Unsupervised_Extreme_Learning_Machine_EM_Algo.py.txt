from sklearn import mixture
import matplotlib.pyplot as plt
import numpy as np
from itertools import groupby

def em(input_data, num_clusters):
    # Initialize Gaussian Mixture Model
    model = mixture.GaussianMixture(n_components=num_clusters, covariance_type='diag')

    # Fit the model to the input data
    model.fit(input_data)

    # Predict the cluster labels
    cluster_labels = model.predict(input_data)

    # Print means and weights of the Gaussian components
    print("Gaussian Means:", model.means_)
    print("Gaussian Weights:", model.weights_)

    # Define data ranges for different classes
    class_ranges = [(0, 59), (59, 130), (130, 178)]

    # Iterate over classes
    for class_idx, (start_idx, end_idx) in enumerate(class_ranges, 1):
        class_cluster_labels = cluster_labels[start_idx:end_idx]
        class_cluster_labels.sort()

        # Print cluster labels and lengths for each class
        print(f"Cluster labels for class {class_idx}:", class_cluster_labels)
        cluster_lengths = [len(list(group)) for key, group in groupby(class_cluster_labels)]
        print(f"Lengths of clusters for class {class_idx}:", cluster_lengths)
