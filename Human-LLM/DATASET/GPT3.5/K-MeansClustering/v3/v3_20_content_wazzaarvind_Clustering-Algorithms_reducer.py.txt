import numpy as np
import sys

def process_input(line):
    """
    Process a single line of input and extract cluster, index, and matrix.

    Args:
    - line (str): A single line of input.

    Returns:
    - cluster (int): The cluster number.
    - index (int): The index.
    - matrix (numpy.ndarray): The matrix as a NumPy array.
    """
    cluster, index, matrix_string = line.strip().split('\t')
    matrix = np.array([float(x) for x in matrix_string.split(',')])
    return int(cluster), int(index), matrix

def print_cluster_data(cluster, output_matrix, centroid):
    """
    Print cluster data including cluster number, output matrix, and centroid.

    Args:
    - cluster (int): The cluster number.
    - output_matrix (list): List of indices in the cluster.
    - centroid (numpy.ndarray): The centroid of the cluster.
    """
    print('%s\t%s\t%s' % (cluster, ','.join(map(str, output_matrix)), ','.join(map(str, centroid))))

def main():
    # Initialize variables
    output_matrix = []
    sum_for_centroid = np.zeros(len_matrix)
    current_cluster = -2
    len_matrix = -1

    # Process input from standard input (stdin) line by line
    for line in sys.stdin:
        cluster, index, matrix = process_input(line)
        
        # Update the length of the matrix if necessary
        if len_matrix == -1:
            len_matrix = len(matrix)

        # Check if a new cluster is encountered
        if current_cluster != cluster:
            if current_cluster != -2:
                # Print data for previous cluster
                centroid = sum_for_centroid / len(output_matrix)
                print_cluster_data(current_cluster, output_matrix, centroid)

            # Reset variables for the new cluster
            output_matrix = []
            sum_for_centroid = np.zeros(len_matrix)
            current_cluster = cluster

        # Add index to output matrix and update sum for centroid
        output_matrix.append(index)
        sum_for_centroid += matrix

    # Print data for the last cluster
    centroid = sum_for_centroid / len(output_matrix)
    print_cluster_data(current_cluster, output_matrix, centroid)

if __name__ == "__main__":
    main()
