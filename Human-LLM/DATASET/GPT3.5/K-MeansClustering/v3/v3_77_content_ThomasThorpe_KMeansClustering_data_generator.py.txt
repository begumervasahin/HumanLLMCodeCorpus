import csv
import random

class DataGenerator:
    """
    Class to generate random data points and write them to a CSV file.
    """
    def __init__(self, dimensions, num_rows, lower_bound, upper_bound, output_file):
        """
        Initialize the DataGenerator object.

        Args:
            dimensions (int): Number of dimensions for each data point.
            num_rows (int): Number of rows (data points) to generate.
            lower_bound (float): Lower bound for generating data points.
            upper_bound (float): Upper bound for generating data points.
            output_file (str): File name for the output CSV file.
        """
        self.dimensions = dimensions
        self.num_rows = num_rows
        self.lower_bound = lower_bound
        self.upper_bound = upper_bound
        self.output_file = output_file

    def write_data(self):
        """
        Generate random data points and write them to a CSV file.
        """
        with open(self.output_file, 'w', newline='\n') as csvfile:
            writer = csv.writer(csvfile, delimiter=',')
            for _ in range(self.num_rows):
                data_point = [random.uniform(self.lower_bound, self.upper_bound) for _ in range(self.dimensions)]
                writer.writerow(data_point)

if __name__ == "__main__":
    # Generate random parameters
    dimensions = random.randint(2, 5)
    num_rows = random.randint(10_000, 100_000)
    lower_bound = -10
    upper_bound = 10
    output_file = "data.csv"

    # Create a DataGenerator object and write data to file
    data_generator = DataGenerator(dimensions, num_rows, lower_bound, upper_bound, output_file)
    data_generator.write_data()
