import random
import time
from datetime import timedelta, datetime

def get_random_date(start_date, end_date, proportion):
    """Generate a random date between start_date and end_date."""
    start_time = time.mktime(time.strptime(start_date, '%Y-%m-%d'))
    end_time = time.mktime(time.strptime(end_date, '%Y-%m-%d'))
    random_time = start_time + proportion * (end_time - start_time)
    return time.strftime('%Y-%m-%d', time.localtime(random_time))

def random_date_between_jan_and_june():
    """Generate a random date between January 1 and June 30."""
    return get_random_date("2018-01-01", "2018-06-30", random.random())

def random_date_between_july_and_dec():
    """Generate a random date between July 1 and December 31."""
    return get_random_date("2018-07-01", "2018-12-31", random.random())

def random_date_from_start_with_delta(start_date, delta_days):
    """Generate a random date starting from the given start_date
    and adding a random number of days up to the specified delta_days."""
    start_date_obj = datetime.strptime(start_date, '%Y-%m-%d').date()
    end_date_obj = start_date_obj + timedelta(days=delta_days)
    return get_random_date(start_date, str(end_date_obj), random.random())

# Example usage
print("Random date between January 1 and June 30:", random_date_between_jan_and_june()) 
print("Random date between July 1 and December 31:", random_date_between_july_and_dec())
print("Random date with custom start date and delta:", random_date_from_start_with_delta("2022-01-01", 100))
