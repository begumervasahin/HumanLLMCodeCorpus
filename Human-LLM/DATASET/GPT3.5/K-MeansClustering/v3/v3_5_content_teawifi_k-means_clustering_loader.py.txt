import os
import re
import pandas as pd
from settings import ENCODING

def load_data(path_to_dir):
    """
    Load data from text files in the specified directory.

    Args:
        path_to_dir (str): Path to the directory containing text files

    Returns:
        list: List of filtered data
    """
    data = []

    # Get list of text files in the directory
    files = [file for file in os.listdir(path_to_dir)
             if os.path.isfile(os.path.join(path_to_dir, file)) and file.endswith(".txt")]

    # Read each file and extract its content
    for file in files:
        with open(os.path.join(path_to_dir, file), "r", encoding=ENCODING) as f:
            content = f.read().split('\n')
            data.extend(content)

    # Filter and process the data
    filtered_data = [filter_data(item) for item in data]
    return filtered_data

def filter_data(data):
    """
    Filter numeric data from a string and return the first three numbers.

    Args:
        data (str): Input string

    Returns:
        list: List of the first three numbers found in the input string
    """
    pattern = "[0-9]+"
    strings = re.findall(pattern, data)
    numbers = [float(i) for i in strings]
    return numbers[:3]

def save_to_csv(data, file_path):
    """
    Save filtered data to a CSV file.

    Args:
        data (list): List of filtered data
        file_path (str): Path to save the CSV file
    """
    # Create a DataFrame from the data
    df = pd.DataFrame(data, columns=['x', 'y', 'z'])

    # Save the DataFrame to a CSV file
    df.to_csv(file_path, index=False, encoding=ENCODING)

# Example usage:
if __name__ == "__main__":
    directory_path = "path/to/directory"  # Replace with the actual directory path      
    output_file_path = "output.csv"  # Specify the output CSV file path
    filtered_data = load_data(directory_path)
    save_to_csv(filtered_data, output_file_path)
