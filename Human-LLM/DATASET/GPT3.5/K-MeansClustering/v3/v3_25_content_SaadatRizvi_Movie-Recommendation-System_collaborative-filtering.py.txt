import pandas as pd

# Read the ratings and movies data
ratings = pd.read_csv('u.data', sep='\t', names=['user_id', 'movie_id', 'rating'])
movies = pd.read_csv('u.item', sep='|', names=['movie_id', 'title'])

# Merge ratings and movies data
merged_data = pd.merge(movies, ratings)

# Create a pivot table of user ratings
user_ratings = merged_data.pivot_table(index='user_id', columns='title', values='rating')

# Calculate the correlation matrix of movie ratings
corr_matrix = user_ratings.corr(method='pearson', min_periods=100)

# Define the user ID and retrieve their ratings
user_id = 2
user_ratings = user_ratings.loc[user_id].dropna()

# Find similar movies based on ratings
similar_candidates = pd.Series()
for index, rating in user_ratings.iteritems():
    similar_movies = corr_matrix[index].dropna()
    similar_movies *= rating
    similar_candidates = similar_candidates.append(similar_movies)

# Aggregate similar candidates
similar_candidates.sort_values(ascending=False, inplace=True)
similar_candidates = similar_candidates.groupby(similar_candidates.index).sum()
similar_candidates.sort_values(ascending=False, inplace=True)

# Filter out movies that the user has already rated
filtered_similarities = similar_candidates.drop(user_ratings.index, errors='ignore')

# Print and save the filtered similarities
print(filtered_similarities.head())
filtered_similarities.to_csv("filteredSims.csv")
