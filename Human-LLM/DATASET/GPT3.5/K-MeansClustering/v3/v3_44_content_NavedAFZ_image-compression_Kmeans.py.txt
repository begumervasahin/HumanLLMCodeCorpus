import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from skimage import io
from sklearn.cluster import KMeans
import imageio

def load_image(file_path):
    """Load an image from the given file path."""
    return io.imread(file_path)

def display_image(image, title):
    """Display the given image with the specified title."""
    plt.imshow(image)
    plt.axis('off')
    plt.title(title)
    plt.show()

def normalize_image(image):
    """Normalize the pixel values of the image."""
    return image / 255

def reshape_image(image):
    """Reshape the image for KMeans clustering."""
    num_rows, num_cols, num_channels = image.shape
    return image.reshape(num_rows * num_cols, num_channels)

def compress_image(image, num_clusters):
    """Compress the image using KMeans clustering."""
    kmeans = KMeans(n_clusters=num_clusters, init='k-means++')
    cluster_labels = kmeans.fit_predict(image)
    cluster_centers = kmeans.cluster_centers_
    compressed_image = cluster_centers[cluster_labels]
    return np.reshape(compressed_image, image.shape)

def save_image(image, file_path):
    """Save the given image to the specified file path."""
    imageio.imwrite(file_path, image)

# Load the original image
original_image = load_image('nass.jpg')
display_image(original_image, 'Original Image')

# Normalize the pixel values of the image
normalized_image = normalize_image(original_image)

# Reshape the image for KMeans clustering
reshaped_image = reshape_image(normalized_image)

# Fit KMeans clustering to compress the image
num_clusters = 15
compressed_image = compress_image(reshaped_image, num_clusters)

# Save and display the compressed image
output_file = 'compressed_image.jpg'
save_image(compressed_image, output_file)
display_image(io.imread(output_file), 'Compressed Image')
