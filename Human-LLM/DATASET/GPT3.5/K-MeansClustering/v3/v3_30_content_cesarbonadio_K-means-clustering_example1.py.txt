import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans

# Create DataFrame with data points
df = pd.DataFrame({
    'x': [1, 2, 4, 7, 5],
    'y': [1, 1, 5, 7, 7]
})

# Perform KMeans clustering with 2 clusters
kmeans = KMeans(n_clusters=2)
kmeans.fit(df)

# Get cluster labels and centroids
labels = kmeans.predict(df)
centroids = kmeans.cluster_centers_
print("Centroids:", centroids)

# Define colors for clusters
cluster_colors = {0: 'r', 1: 'g'}
point_colors = [cluster_colors[label] for label in labels]

# Plot clusters and centroids
fig = plt.figure(figsize=(5, 5))
plt.scatter(df['x'], df['y'], color=point_colors, alpha=0.3)  # Plot data points
for idx, centroid in enumerate(centroids):
    plt.scatter(*centroid, color=cluster_colors[idx], marker='x', s=100)  # Plot centroids
plt.xlim(0, 10)
plt.ylim(0, 10)
plt.xlabel('x')
plt.ylabel('y')
plt.title('KMeans Clustering')
plt.show()
