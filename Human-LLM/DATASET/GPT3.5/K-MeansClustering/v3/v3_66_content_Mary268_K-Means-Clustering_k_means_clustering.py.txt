from sklearn.cluster import KMeans
from sklearn.metrics import accuracy_score
from sklearn.datasets import load_digits
from scipy.stats import mode
import numpy as np
import matplotlib.pyplot as plt
import sklearn

# Import train_test_split from appropriate module based on scikit-learn version   
if int((sklearn.__version__).split(".")[1]) < 18:
    from sklearn.cross_validation import train_test_split
else:
    from sklearn.model_selection import train_test_split

def determine_sklearn_module():
    """Determine the appropriate module for train_test_split based on scikit-learn version."""
    if int((sklearn.__version__).split(".")[1]) < 18:
        return __import__("sklearn.cross_validation")
    else:
        return __import__("sklearn.model_selection")

def load_digits_dataset():
    """Load the digits dataset."""
    return load_digits()

def perform_kmeans_clustering(data, n_clusters=10, random_state=0):
    """Perform KMeans clustering."""
    kmeans = KMeans(n_clusters=n_clusters, random_state=random_state)
    clusters = kmeans.fit_predict(data.data)
    return kmeans, clusters

def plot_cluster_centers(kmeans):
    """Plot the cluster centers."""
    fig, ax = plt.subplots(2, 5, figsize=(8, 3))
    centers = kmeans.cluster_centers_.reshape(10, 8, 8)
    for axi, center in zip(ax.flat, centers):
        axi.set(xticks=[], yticks=[])
        axi.imshow(center, interpolation='nearest', cmap=plt.cm.binary)

def assign_cluster_labels(clusters, target):
    """Assign labels to each cluster based on the most common digit in that cluster."""
    labels = np.zeros_like(clusters)
    for i in range(10):
        mask = (clusters == i)
        labels[mask] = mode(target[mask])[0]
    return labels

def calculate_accuracy_score(target, labels):
    """Calculate the accuracy score."""
    accuracy = accuracy_score(target, labels)
    return accuracy

if __name__ == "__main__":
    # Load the digits dataset
    data = load_digits_dataset()

    # Perform KMeans clustering
    kmeans, clusters = perform_kmeans_clustering(data)

    # Plot the cluster centers
    plot_cluster_centers(kmeans)

    # Assign labels to each cluster based on the most common digit in that cluster
    labels = assign_cluster_labels(clusters, data.target)

    # Calculate the accuracy score
    accuracy = calculate_accuracy_score(data.target, labels)
    print("Accuracy Score:", accuracy)
