# Index file
INDEX_FILE = "INDEX_FILE.INDEX"

def remove_stopwords(wordlist):
    """Remove English stopwords from the given wordlist."""
    return [word for word in wordlist if word not in ENGLISH_STOPWORDS]

def set_list_intersection(set_list):
    """Find the intersection of multiple sets."""
    if not set_list:
        return set()
    result = set_list[0]
    for s in set_list[1:]:
        result &= s
    return result

# Load the index dictionary
with open(INDEX_FILE) as index_file:
    index_dictionary = json.load(index_file)

# Initialize lemmatizer
lemmatizer = WordNetLemmatizer()

# Process query terms
query_terms = sys.argv[1:]
normalized_query = []
result_sets = []

print('_' * 50)
print("Query Result")
print('_' * 50)

# Lemmatize and filter stopwords from query terms
for term in query_terms:
    if term.lower() not in ENGLISH_STOPWORDS:
        lem_term = lemmatizer.lemmatize(term)
        try:
            result_sets.append(set(index_dictionary[lem_term]['postings']))
        except KeyError:
            print("No documents found matching query:", ' '.join(query_terms))
            break

# Find intersection of result sets
query_result = set_list_intersection(result_sets)

# Display query result
if query_result:
    print('\n'.join(query_result))
