import os
import csv
import pickle
import numpy as np
from nltk import word_tokenize
from nltk.stem import WordNetLemmatizer
from nltk.corpus import stopwords
from keras.preprocessing import sequence
from keras.models import load_model

# Directory paths
MODELS_DIR = "models"
DATA_DIR = "data"
TRUMP_TWEETS_PATH = os.path.join(DATA_DIR, "trumptweets.csv")

# Model parameters
DIMENSIONS = ["IE", "NS", "FT", "PJ"]
MODEL_BATCH_SIZE = 128
TOP_WORDS = 2500
MAX_POST_LENGTH = 40
EMBEDDING_VECTOR_LENGTH = 20

# Load Trump tweets
def load_trump_tweets(file_path):
    tweets = []
    with open(file_path, "r", encoding="ISO-8859-1") as file:
        reader = csv.reader(file)
        for row in reader:
            tweets.append(row[0])
    return tweets

# Lemmatize text
def lemmatize_text(text, types, lemmatizer, stop_words):
    lemmatized = []
    for post in text:
        temp = post.lower()
        for type_ in types:
            temp = temp.replace(" " + type_, "")
        temp = " ".join([lemmatizer.lemmatize(word) for word in temp.split(" ") if word not in stop_words])
        lemmatized.append(temp)
    return np.array(lemmatized)

# Preprocess data
def preprocess_data(data, tokenizer, max_length):
    tokenized = tokenizer.texts_to_sequences(data)
    return sequence.pad_sequences(tokenized, maxlen=max_length)

# Load and preprocess Trump tweets
x_test = load_trump_tweets(TRUMP_TWEETS_PATH)
lemmatizer = WordNetLemmatizer()
stop_words = set(stopwords.words("english"))
types = [x.lower() for x in DIMENSIONS]

# Iterate over dimensions
final_prediction = ""
for dimension in DIMENSIONS:
    # Load model and tokenizer
    model = load_model(os.path.join(MODELS_DIR, f"rnn_model_{dimension}.h5"))
    with open(os.path.join(MODELS_DIR, f"rnn_tokenizer_{dimension}.pkl"), "rb") as f:
        tokenizer = pickle.load(f)

    # Preprocess data
    lemmatized_tweets = lemmatize_text(x_test, types, lemmatizer, stop_words)
    x_processed = preprocess_data(lemmatized_tweets, tokenizer, MAX_POST_LENGTH)

    # Make predictions
    predictions = model.predict(x_processed)
    average_prediction = np.mean(predictions)
    print(f"{dimension}: {average_prediction}")

    # Determine final prediction
    final_prediction += dimension[1] if average_prediction >= 0.5 else dimension[0]

# Print final prediction
print("\nFinal prediction:", final_prediction)
