import os
import csv
import collections
import pandas as pd

# Constants
DATA_DIR = "data"
MBTI_RAW_CSV_PATH = os.path.join(DATA_DIR, "mbti_1.csv")
MBTI_CLEAN_CSV_PATH = os.path.join(DATA_DIR, "mbti_clean.csv")
MBTI_UNCLEAN_CSV_PATH = os.path.join(DATA_DIR, "mbti_unclean.csv")

# Define the expected frequency of each MBTI type
MBTI_TO_FREQUENCY_DICT = {
    "ISTJ": 0.11, "ISFJ": 0.09, "INFJ": 0.04, "INTJ": 0.05,
    "ISTP": 0.05, "ISFP": 0.05, "INFP": 0.06, "INTP": 0.06,
    "ESTP": 0.04, "ESFP": 0.04, "ENFP": 0.08, "ENTP": 0.06,
    "ESTJ": 0.08, "ESFJ": 0.09, "ENFJ": 0.05, "ENTJ": 0.05,
}

def read_mbti_data(csv_path):
    """Read MBTI data from CSV file."""
    return pd.read_csv(csv_path)

def write_to_csv(file_path, data, header):
    """Write data to CSV file."""
    with open(file_path, "w") as csv_file:
        writer = csv.writer(csv_file)
        writer.writerow(header)
        writer.writerows(data)

# Read the raw MBTI data
df = read_mbti_data(MBTI_RAW_CSV_PATH)

# Count occurrences of each MBTI type
mbti_counts = collections.Counter(df["type"])

# Find the limiting MBTI type based on the smallest size relative to its expected frequency
min_size = float("inf")
limiting_type = None
for mbti, count in mbti_counts.items():
    size = count / MBTI_TO_FREQUENCY_DICT[mbti]
    if size < min_size:
        min_size = size
        limiting_type = mbti

# Separate data by MBTI type
mbti_data = collections.defaultdict(list)
for _, row in df.iterrows():
    mbti_data[row["type"]].append(row)

# Write clean and unclean data to CSV files
unclean_data = []
for mbti, data_list in mbti_data.items():
    sample_size = int(round(min_size * MBTI_TO_FREQUENCY_DICT[mbti]))
    write_to_csv(MBTI_CLEAN_CSV_PATH, data_list[:sample_size], ["type", "posts"])
    unclean_data.extend(data_list[sample_size:])

write_to_csv(MBTI_UNCLEAN_CSV_PATH, unclean_data, ["type", "posts"])
