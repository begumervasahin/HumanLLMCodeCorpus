import os
import time
import string
import operator
import numpy as np
import nltk
from nltk.corpus import stopwords
from nltk.stem.wordnet import WordNetLemmatizer

# Function to clean up text
def text_cleanup(text):
    # Remove punctuation
    text_without_punctuation = ''.join([c for c in text if c not in string.punctuation])
    # Remove stopwords and lemmatize words
    lemmatizer = WordNetLemmatizer()
    cleaned_text = [lemmatizer.lemmatize(word.lower()) for word in text_without_punctuation.split() if word.lower() not in stopwords.words('english')]
    return cleaned_text

start_time = time.time()

# Initialize count dictionary
count = {}

# Directory containing emails
directory_path = "emails/"

# Iterate over files in the directory
for filename in os.listdir(directory_path):
    filename = filename.decode("utf-8")  # Decode filename
    file_path = os.path.join(directory_path, filename)  # Create file path
    with open(file_path, "r", encoding="utf-8", errors="ignore") as file:
        words = text_cleanup(file.read())  # Clean up text 
        # Count occurrences of words
        for word in words:
            if not word.isdigit() and len(word) > 2:
                count[word] = count.get(word, 0) + 1

# Sort count dictionary by values in descending order
sorted_count = dict(sorted(count.items(), key=operator.itemgetter(1), reverse=True))

# Write top words to CSV file
with open("wordslist.csv", "w+") as csv_file:
    csv_file.write("word,count\n")
    for word, times in sorted_count.items():
        if times < 100:
            break
        csv_file.write(f"{word},{times}\n")

# Print execution time
print("Time taken to preprocess the emails:", round(time.time() - start_time, 2), "seconds")
