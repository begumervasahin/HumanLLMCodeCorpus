import pandas as pd
import sys
import string
import nltk
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from nltk.stem.snowball import SnowballStemmer
import numpy as np

# Download NLTK resources
nltk.download('stopwords')
nltk.download('punkt')

# Define city names with underscores
CITIES = ["los_angel", "san_francisco", "san_diego", "houston", "chicago", "philadelphia", "toronto",
          "atlanta", "boston", "orlando", "washington", "manhattan"]

# Define city names
CITY_NAMES = ['Los_Angeles,_CA', 'San_Francisco,_CA', 'San_Diego,_CA', 'Houston,_TX', 'Chicago,_IL',
              'Philadelphia,_PA', 'Toronto,_Ontario', 'Atlanta,_GA', 'Boston,_MA', 'Orlando,_FL',
              'Washington,_DC', 'Manhattan,_NY']

# Function to parse the input file
def parse_file(file_path):
    with open(file_path, 'r') as datafile:
        content = datafile.readlines()

    words = [word_tokenize(line) for line in content]
    stop = stopwords.words('english')
    punct_stopwords = list(string.punctuation) + ['i', '\x89', '_CA', '_TX', '_IL', '_NY', '_PA', '_GA',
                                                  '_Ontario', '_MA', '_FL', '_DC', '__', '___'] + stop
    stemmer = SnowballStemmer("english")

    # Stem and filter words
    words = [[stemmer.stem(w) for w in item if w not in punct_stopwords] for item in words if len(item) > 1]
    return [word for word in words if len(word) > 1 and word[0] in CITIES]

# Function to create training DataFrame
def create_train_dataframe(train_data):
    word_list = set([w for word in train_data for w in word])      
    data = pd.DataFrame(0, index=word_list, columns=CITIES)

    for word in train_data[:]:
        current_city = word[0]
        for w in word[1:]:
            data.at[w, current_city] += 1

    return data

# Function to parse test data
def parse_test_data(file_path):
    with open(file_path, 'r') as datafile:
        content = datafile.readlines()

    test_dict = {}
    stop = stopwords.words('english')
    punct_stopwords = list(string.punctuation) + ['i', '\x89', '_CA', '_TX', '_IL', '_NY', '_PA', '_GA',
                                                  '_Ontario', '_MA', '_FL', '_DC', '__', '___'] + stop
    stemmer = SnowballStemmer("english")

    for line in content:
        words = word_tokenize(line)
        test_dict[line] = [stemmer.stem(w) for w in words if w not in punct_stopwords]
        if test_dict[line]:
            test_dict[line].pop(0)

    return test_dict

# Function to calculate city probabilities
def calculate_city_probabilities(train_data):
    city_prob = {}
    for word in train_data:
        city_prob[word[0]] = city_prob.get(word[0], 0) + 1

    total = sum(city_prob.values())
    for city_name in city_prob:
        city_prob[city_name] = city_prob[city_name] / total

    return city_prob

# Function to perform Naive Bayes classification
def naive_bayes_classification(train_df, test_data, output_file_path, city_prob):
    with open(output_file_path, 'w') as f:
        for tweet in test_data:
            label = ''
            words = test_data[tweet]

            if not words:
                continue
            else:
                wpost = [[(train_df.at[w, city[i]] / sum(train_df.loc[:, city[i]])) * city_prob[city[i]]
                          if w in train_df.index else 0.00001 * city_prob[city[i]] for i in range(0, 12)] for w in
                         words]

            p = np.prod(wpost, axis=0)
            label = CITY_NAMES[np.argmax(p)]
            f.write(label + " " + tweet)

    print("Output written to", output_file_path)

# Main function
def main():
    f1, f2, f3 = sys.argv[1], sys.argv[2], sys.argv[3]

    # Parse training data
    train_data = parse_file(f1)
    train_df_data = create_train_dataframe(train_data)

    # Parse test data
    test_data = parse_test_data(f2)

    # Calculate city probabilities
    city_probabilities = calculate_city_probabilities(train_data)

    # Perform Naive Bayes classification
    naive_bayes_classification(train_df_data, test_data, f3, city_probabilities)

    print('Top five words for each city:')
    for city_name in CITIES:
        print(CITY_NAMES[CITIES.index(city_name)], train_df_data.nlargest(5, city_name).index.tolist())

if __name__ == "__main__":
    main()
