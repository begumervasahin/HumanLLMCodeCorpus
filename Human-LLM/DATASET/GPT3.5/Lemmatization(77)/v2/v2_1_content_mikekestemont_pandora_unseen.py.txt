from __future__ import print_function
import sys
import os
import codecs
import pandora.utils
from pandora.tagger import Tagger

def main():
    print('::: Annotation Started :::')

    # Load the tagger
    tagger = Tagger(load=True, model_dir='models/new_cg')
    print('Tagger loaded, now annotating...')

    # Define paths for the original and tagged data
    orig_path = 'data/limborch/orig/'
    new_path = 'data/limborch/tagged/'

    # Annotate each file in the original directory
    for filename in os.listdir(orig_path):
        if not filename.endswith('.txt'):
            continue
        print('Processing:', filename)

        # Load unannotated tokens from the original file
        unseen_tokens = pandora.utils.load_unannotated_file(orig_path + filename, nb_instances=None, tokenized_input=False)

        # Annotate the tokens using the tagger
        annotations = tagger.annotate(unseen_tokens)

        # Write the annotations to the new file
        with codecs.open(new_path + filename, 'w', 'utf8') as f:
            for token, lemma, pos in zip(annotations['tokens'], annotations['lemmas'], annotations['pos']):
                annotated_line = '\t'.join((token.lower(), lemma.lower(), pos.lower())) + '\n'
                f.write(annotated_line)

    print('::: Annotation Ended :::')

if __name__ == '__main__':
    main()
