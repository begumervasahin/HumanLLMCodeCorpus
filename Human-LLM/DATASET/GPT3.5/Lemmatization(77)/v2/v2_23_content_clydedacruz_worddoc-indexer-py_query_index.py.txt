import json
from nltk.stem import WordNetLemmatizer

# Define the English stopwords set
ENGLISH_STOPWORDS = {
    "i", "me", "my", "myself", "we", "our", "ours", "ourselves", "you", "your", "yours", "yourself", "yourselves",
    "he", "him", "his", "himself", "she", "her", "hers", "herself", "it", "its", "itself", "they", "them", "their",
    "theirs", "themselves", "what", "which", "who", "whom", "this", "that", "these", "those", "am", "is", "are", "was",
    "were", "be", "been", "being", "have", "has", "had", "having", "do", "does", "did", "doing", "a", "an", "the", "and",
    "but", "if", "or", "because", "as", "until", "while", "of", "at", "by", "for", "with", "about", "against", "between",
    "into", "through", "during", "before", "after", "above", "below", "to", "from", "up", "down", "in", "out", "on", "off",
    "over", "under", "again", "further", "then", "once", "here", "there", "when", "where", "why", "how", "all", "any", "both",
    "each", "few", "more", "most", "other", "some", "such", "no", "nor", "not", "only", "own", "same", "so", "than", "too",
    "very", "s", "t", "can", "will", "just", "don", "should", "now"
}

# Define the index file
INDEX_FILE = "INDEX_FILE.INDEX"

def remove_stopwords(wordlist):
    """Remove English stopwords from the given wordlist."""
    return [word for word in wordlist if word.lower() not in ENGLISH_STOPWORDS]

def set_list_intersection(set_list):
    """Find the intersection of a list of sets."""
    if not set_list:
        return set()
    result = set_list[0]
    for s in set_list[1:]:
        result &= s
    return result

def main():
    # Check if query terms are provided
    if len(sys.argv) < 2:
        print("Usage: python script.py <query_terms>")
        return

    query_terms = sys.argv[1:]

    # Load the index dictionary
    with open(INDEX_FILE) as index_file:
        index_dictionary = json.load(index_file)

    lemmatizer = WordNetLemmatizer()
    
    # Normalize and lemmatize the query terms
    normalized_query = [lemmatizer.lemmatize(word) for word in remove_stopwords(query_terms)] 
    result_sets = []

    print('_' * 50 + "\nQuery Result \n" + '_' * 50)

    # Retrieve the postings for each query term
    for word in normalized_query:
        try:
            result_sets.append(set(index_dictionary[word]['postings']))
        except KeyError:
            print("No documents found matching query:", ' '.join(query_terms))
            return

    # Find the intersection of the result sets
    query_result = set_list_intersection(result_sets)

    # Display the query result
    if query_result:
        print('\n'.join(list(query_result)))

if __name__ == "__main__":
    main()
