import logging
import conll18_ud_eval
import typer
from pathlib import Path
from datasets import gold_path

# Constants for column indices
FORM = 1
LEMMA = 2

def main():
    setup_logging()

    resultdir, errordir, predictionsdir = setup_directories()

    process_model_directories(resultdir, errordir, predictionsdir)

def setup_logging():
    logging.basicConfig(format='%(levelname)s: %(message)s', level=logging.INFO)

def setup_directories():
    resultdir = Path('results')
    errordir = resultdir / 'errors'
    errordir.mkdir(parents=True, exist_ok=True)
    predictionsdir = resultdir / 'predictions'
    return resultdir, errordir, predictionsdir

def process_model_directories(resultdir, errordir, predictionsdir):
    for model_result_dir in predictionsdir.glob('*'):
        model_name = model_result_dir.stem
        process_prediction_files(model_result_dir, model_name, errordir)

def process_prediction_files(model_result_dir, model_name, errordir):
    for predictions_file in model_result_dir.glob('*.conllu'):
        testset_name = predictions_file.stem
        gold_ud = conll18_ud_eval.load_conllu_file(gold_path(testset_name))
        system_ud = conll18_ud_eval.load_conllu_file(predictions_file)
        output_file = errordir / f'lemma_errors_{model_name}_{testset_name}.txt'
        print(f'Writing errors to: {output_file}')
        write_errors(gold_ud, system_ud, output_file)

def write_errors(gold_ud, system_ud, output_file):
    with open(output_file, 'w') as f:
        for system_sentence, gold_sentence in zip(system_ud.sentences, gold_ud.sentences):
            system_words = [w for w in system_ud.words if is_within_sentence(w, system_sentence)]
            system_lemmas = [x.columns[LEMMA].lower() for x in system_words if not x.is_multiword]
            gold_words = [w for w in gold_ud.words if is_within_sentence(w, gold_sentence)]
            gold_lemmas = [x.columns[LEMMA].lower() for x in gold_words if not x.is_multiword]
            gold_forms = [x.columns[FORM].lower() for x in gold_words if not x.is_multiword]
            if system_lemmas != gold_lemmas:
                f.write(' '.join(gold_forms) + '\n')
                f.write(' '.join(gold_lemmas) + '\n')
                f.write(' '.join(system_lemmas) + '\n')
                f.write('\n')

def is_within_sentence(word, sentence):
    return word.span.start >= sentence.start and word.span.end <= sentence.end

if __name__ == '__main__':
    typer.run(main)
