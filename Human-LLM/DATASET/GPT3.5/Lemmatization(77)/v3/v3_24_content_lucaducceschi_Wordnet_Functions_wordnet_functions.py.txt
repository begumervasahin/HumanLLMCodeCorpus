import nltk
from nltk.corpus import wordnet as wn

def get_hyponyms(noun):
    """
    Get hyponyms for the given noun.

    Parameters:
        noun (str): The noun to find hyponyms for.

    Returns:
        list of lists: A list of lists containing hyponyms for each synset of the noun.
    """
    # Get synsets for the noun
    synsets = wn.synsets(noun)

    # Filter synsets containing the noun
    filtered_synsets = [synset for synset in synsets if noun in synset.lemmas()[0].name()]

    # Get hyponyms for each filtered synset
    hyponyms = [synset.hyponyms() for synset in filtered_synsets]

    return hyponyms

def get_hypernyms(noun):
    """
    Get hypernyms for the given noun.

    Parameters:
        noun (str): The noun to find hypernyms for.

    Returns:
        tuple: A tuple containing the filtered synsets and their hypernyms.
    """
    # Get synsets for the noun
    synsets = wn.synsets(noun)

    # Filter synsets containing the noun
    filtered_synsets = [synset.lemmas()[0].name() for synset in synsets if noun in synset.lemmas()[0].name()]

    # Get hypernyms for each filtered synset
    hypernyms = [wn.synset(synset).hypernyms() for synset in filtered_synsets]

    return filtered_synsets, hypernyms

def univ_pos_changer(pos_tag):
    """
    Convert universal POS tags to WordNet POS tags.

    Parameters:
        pos_tag (str): The universal POS tag.

    Returns:
        str: The corresponding WordNet POS tag.
    """
    pos_map = {"NOUN": "n", "VERB": "v", "ADJ": "a", "ADV": "r"}
    return pos_map.get(pos_tag, None)

def lemmatize_word(word, pos_tag):
    """
    Lemmatize the word based on its POS tag.

    Parameters:
        word (str): The word to lemmatize.
        pos_tag (str): The POS tag of the word.

    Returns:
        tuple: A tuple containing the lemmatized word and its POS tag.
    """
    lemma = wn.morphy(word, univ_pos_changer(pos_tag))
    return (lemma, pos_tag) if lemma is not None else (word, pos_tag)

def lemmatize_corpus(corpus):
    """
    Lemmatize words in the corpus.

    Parameters:
        corpus (list): A list of lists containing tokenized sentences.

    Returns:
        list: A list of lists containing lemmatized words with their POS tags.
    """
    return [[lemmatize_word(word, pos_tag) for word, pos_tag in line] for line in corpus]

# Example usage
noun = 'dog'
hyponyms = get_hyponyms(noun)
hypernyms = get_hypernyms(noun)
print("Hyponyms:")
print(hyponyms)
print("\nHypernyms:")
print(hypernyms)
