import os
import time
import string
import operator
import numpy as np
import nltk
from nltk.corpus import stopwords
from nltk.stem.wordnet import WordNetLemmatizer

# Function to clean up text
def clean_text(text):
    # Remove punctuation
    no_punctuation = ''.join([char for char in text if char not in string.punctuation])
    # Remove stopwords and lemmatize words
    lemmatizer = WordNetLemmatizer()
    cleaned = [lemmatizer.lemmatize(word.lower()) for word in no_punctuation.split() if word.lower() not in stopwords.words('english')]
    return cleaned

start_time = time.time()

# Initialize count dictionary
word_count = {}

# Directory containing emails
directory = "emails/"

# Iterate over files in the directory
for file_name in os.listdir(directory):
    file_path = os.path.join(directory, file_name)  # Create file path
    with open(file_path, "r", encoding="utf-8", errors="ignore") as file:
        words = clean_text(file.read())  # Clean up text
        # Count occurrences of words
        for word in words:
            if not word.isdigit() and len(word) > 2:
                word_count[word] = word_count.get(word, 0) + 1

# Sort word_count dictionary by values in descending order
sorted_word_count = dict(sorted(word_count.items(), key=operator.itemgetter(1), reverse=True))

# Write top words to CSV file
with open("wordslist.csv", "w+") as csv_file:
    csv_file.write("word,count\n")
    for word, count in sorted_word_count.items():
        if count < 100:
            break
        csv_file.write(f"{word},{count}\n")

# Print execution time
print("Time taken to preprocess the emails:", round(time.time() - start_time, 2), "seconds")
