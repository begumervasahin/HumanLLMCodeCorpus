import sys

# Load Lithuanian lemmas from the dictionary
with open('dazninis.utf8.txt', 'r', encoding='utf-8') as dictionary_file:
    lt_lemmas = [line.split('\t')[0] for line in dictionary_file]

# Define input and output files
input_file = sys.argv[1]
output_file = sys.argv[2]

# Process the input file and correct lemmas
with open(input_file, 'r', encoding='utf-8') as input_file, open(output_file, 'w', encoding='utf-8') as output_file:
    for line in input_file:
        line_parts = line.split('\t')
        if len(line_parts) == 10:
            lemma = line_parts[2]

            # Check if the word is a verb with specific prefixes
            if line_parts[3] == 'VERB':
                if lemma.startswith('nebe'):
                    if lemma.startswith('nebesi'):
                        if 'Reflex=Yes' in line_parts[5] and lemma[6:] + 's' in lt_lemmas:     
                            line_parts[2] = lemma[6:] + 's'
                    elif lemma[4:] in lt_lemmas:
                        line_parts[2] = lemma[4:]
                    elif lemma[2:] in lt_lemmas:
                        line_parts[2] = lemma[2:]
                elif lemma.startswith('ne'):
                    if 'Reflex=Yes' in line_parts[5]:
                        line_parts[2] = lemma[4:] + 's'
                    elif lemma[2:] in lt_lemmas:
                        line_parts[2] = lemma[2:]
                elif lemma.startswith('tebe'):
                    if 'Reflex=Yes' in line_parts[5]:
                        line_parts[2] = lemma[6:] + 's'
                    elif lemma[2:] in lt_lemmas:
                        line_parts[2] = lemma[2:]
                    else:
                        line_parts[2] = lemma[4:]
                elif lemma.startswith('be'):
                    if 'Reflex=Yes' in line_parts[5]:
                        line_parts[2] = lemma[4:] + 's'
                    elif lemma[2:] in lt_lemmas:
                        line_parts[2] = lemma[2:]

            # Write the corrected line to the output file
            corrected_line = '\t'.join(line_parts)
            output_file.write(corrected_line)
        else:
            # Write non-verb lines as they are
            output_file.write(line)
