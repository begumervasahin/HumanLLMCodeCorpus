import nltk
from nltk.corpus import wordnet as wn

def get_hyponyms(noun):
    """
    Get hyponyms for the given noun.

    Parameters:
        noun (str): The noun to find hyponyms for.

    Returns:
        list: A list of lists containing hyponyms for each synset of the noun.
    """
    # Get synsets for the noun
    list_of_synsets = nltk.corpus.wordnet.synsets(noun)
    
    # Filter synsets containing the noun
    filtered_synsets = [synset for synset in list_of_synsets if noun in synset.unicode_repr()[8:-2]]

    # Get hyponyms for each filtered synset
    hyponyms = [list(synset.closure(lambda s: s.hyponyms(), depth=1)) for synset in filtered_synsets]
    
    # Print filtered synsets and corresponding hyponyms
    print("Filtered Synsets:")
    for synset in filtered_synsets:
        print(synset)
    
    print("\nHyponyms:")
    for hyponym_list in hyponyms:
        for hyponym in hyponym_list:
            print(hyponym)

    return hyponyms

def get_hypernyms(noun):
    """
    Get hypernyms for the given noun.

    Parameters:
        noun (str): The noun to find hypernyms for.

    Returns:
        tuple: A tuple containing the filtered synsets and their hypernyms.
    """
    # Get synsets for the noun
    list_of_synsets = nltk.corpus.wordnet.synsets(noun)
    
    # Extract strings representing synsets containing the noun
    strings_of_synsets = [i.unicode_repr()[8:-2] for i in list_of_synsets if noun in i.unicode_repr()[8:-2]]
    
    # Define lambda function for getting hypernyms
    hyper = lambda s: s.hypernyms()
    
    # Get hypernyms for each synset string
    var_out = [list(nltk.corpus.wordnet.synset(i).closure(hyper, depth=8)) for i in strings_of_synsets]
    
    return strings_of_synsets, var_out

def univ_pos_changer(stringa):
    """
    Convert universal POS tags to WordNet POS tags.

    Parameters:
        stringa (str): The universal POS tag.

    Returns:
        str: The corresponding WordNet POS tag.
    """
    if stringa == "NOUN":
        return "n"
    elif stringa == "VERB":
        return "v"
    elif stringa == "ADJ":
        return "a"
    elif stringa == "ADV":
        return "r"
    else:
        return None

def lemmatizer(tupla):
    """
    Lemmatize the word based on its POS tag.

    Parameters:
        tupla (tuple): A tuple containing the word and its POS tag.

    Returns:
        tuple: A tuple containing the lemmatized word and its POS tag.
    """
    lemma = wn.morphy(tupla[0], univ_pos_changer(tupla[1]))
    if lemma is None:
        return tupla[0], tupla[1]
    else:
        return lemma, tupla[1]

def tag_me(corpus):
    """
    Lemmatize words in the corpus.

    Parameters:
        corpus (list): A list of lists containing tokenized sentences.

    Returns:
        list: A list of lists containing lemmatized words with their POS tags.
    """
    return [[lemmatizer(x) for x in linea] for linea in corpus]
