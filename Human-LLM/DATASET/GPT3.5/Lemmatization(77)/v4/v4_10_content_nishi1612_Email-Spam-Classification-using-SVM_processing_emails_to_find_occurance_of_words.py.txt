import os
import string
import numpy as np
import pandas as pd
from time import time
from nltk.corpus import stopwords
from nltk.stem.wordnet import WordNetLemmatizer

# Start time
start_time = time()

# Read the words list CSV file
words_df = pd.read_csv('wordslist.csv', header=0)
words = words_df['word']

# Initialize WordNet Lemmatizer
lmtzr = WordNetLemmatizer()

# Define directory paths
emails_directory = "emails/"
output_file_path = "frequency.csv"

# Create or overwrite the output file
with open(output_file_path, "w+") as output_file:
    # Write the header row
    output_file.write(','.join(words) + ',output\n')

# Count the number of files processed
files_processed = 0

# Iterate over files in the directory
for file in os.listdir(emails_directory):
    # Decode file name if necessary
    file = file.decode("utf-8") if isinstance(file, bytes) else file

    # Construct full file path
    file_path = os.path.join(emails_directory, file)

    # Skip directories
    if os.path.isdir(file_path):
        continue

    # Increment file counter
    files_processed += 1

    # Open and read the file
    with open(file_path, "r", encoding='utf-8', errors='ignore') as file_reading:
        # Initialize array to store word counts
        word_count_array = np.zeros(words.size)

        # Process each word in the file
        for word in file_reading.read().split():
            # Normalize and lemmatize the word
            word = lmtzr.lemmatize(word.lower())

            # Check if word should be skipped
            if (word in stopwords.words('english') or 
                word in string.punctuation or 
                len(word) <= 2 or 
                word.isdigit()):
                continue

            # Increment count for matching words in the words list
            for idx, w in enumerate(words):
                if w == word:
                    word_count_array[idx] += 1
                    break

        # Write word frequencies to the output file
        with open(output_file_path, "a") as output_file:
            output_file.write(','.join(map(str, word_count_array)) + ',' +
                              ("-1" if len(file) == 68 else "1") + '\n')

    # Print progress message for every 100 files processed
    if files_processed % 100 == 0:
        print("Processed", files_processed, "files")

# Calculate and print execution time
execution_time = round(time() - start_time, 2)
print("Time (in seconds) to segregate entire dataset to form input vector:", execution_time)
