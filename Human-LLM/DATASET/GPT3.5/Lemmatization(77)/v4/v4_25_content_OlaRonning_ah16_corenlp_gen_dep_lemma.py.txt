'''
To use the script:
1) Run coreNLP server in a separate process
   > python2 coreNLP.py &
2) Run the client script (i.e., this script)
   > python gen_dep_lemma.py <Corpus File>
'''
from __future__ import print_function
import json
from jsonrpc import ServerProxy, JsonRpc20, TransportTcpIp
import sys
import itertools

class StanfordNLP:
    def __init__(self):
        # Initialize the StanfordNLP server
        self.server = ServerProxy(JsonRpc20(), TransportTcpIp(addr=("127.0.0.1", 8080)))

    def parse(self, text):
        # Parse the input text using the StanfordNLP server
        return json.loads(self.server.parse(text))

def flatten(*args):
    '''Flattens arguments into a single list'''
    if len(args) == 1:
        l = itertools.chain.from_iterable(*args)
    else:
        l = itertools.chain(*args)
    return list(l)

def adjust_dependency_index(dependency):
    '''Fixes off-by-one error in dependency suffix index'''
    def adjust(index):
        return flatten([index, [str(int(index[-1]) - 1)]])

    rel, w1, w2 = dependency
    w1 = adjust(w1.split('-'))
    w2 = adjust(w2.split('-'))
    return [rel, w1, w2]

def extract_article(example):
    '''Extracts the match and previous sentence from the example'''
    before = [sent['string'] for sent in example['before']]
    match = example['match']['sentence']['string']
    candidates = before + [match]
    return candidates[-2:]

if __name__ == '__main__':
    # Initialize StanfordNLP object
    nlp = StanfordNLP()

    # Read annotation file path from command line argument
    annotation_file = sys.argv[1]

    # Dictionary to store parsed annotation data
    annotation_data = {}

    # Parse annotation data
    with open(annotation_file, 'r') as infile:
        for example in infile:
            data = json.loads(example)
            try:
                sluice_id = "{0[file]}_{0[line]}_{0[treeNode]}".format(data['metadata'])
                article = extract_article(data)
            except Exception as e:
                print(f'Error processing example: {e}', file=sys.stderr)
                continue
            annotation_data[sluice_id] = article

    # Write parsed data to output file
    with open('dep_lemma.jsons', 'w') as outfile:
        for sluice_id, article in annotation_data.items():
            for sentence in article:
                try:
                    result = nlp.parse(sentence)
                except Exception as e:
                    print(f'Error in parsing: {e}', file=sys.stderr)
                    print(f'Sluice ID: {sluice_id}', file=sys.stderr)
                    continue
                if len(result['sentences']) != 1:
                    print('Multiple sentences', file=sys.stderr)
                    print(f'Sluice ID: {sluice_id}', file=sys.stderr)
                    continue
                lemmas = [word[-1]['Lemma'] for word in result['sentences'][-1]['words']]
                try:
                    output_data = {'sluiceId': sluice_id, 'deps': list(map(adjust_dependency_index, result['sentences'][-1]['dependencies'])), 'string': sentence, 'lemmas': lemmas}
                except Exception as e:
                    print(f'Error in creating output data: {e}', file=sys.stderr)
                    print(f'Sluice ID: {sluice_id}', file=sys.stderr)
                    continue
                print(json.dumps(output_data), file=outfile)
