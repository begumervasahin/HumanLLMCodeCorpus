import requests
from bs4 import BeautifulSoup
import os
import constants as const

def month_to_digit(month):
    """Convert month name to its corresponding digit."""
    month_dict = {'January': '01', 'February': '02', 'March': '03',
                  'April': '04', 'May': '05', 'June': '06', 'July': '07',
                  'August': '08', 'September': '09', 'October': '10',
                  'November': '11', 'December': '12'}
    return month_dict.get(month, '00')

def conv_to_fname(s):
    """Convert date string to filename."""
    flist = s.split()
    m = month_to_digit(flist[2])
    day = flist[1].zfill(2)
    year = flist[3][2:]
    filename = f"1{year}{m}{day}"
    return filename, flist[3]

def get_data(i):
    """Retrieve data from the website."""
    url = const.URL_PREFIX + str(i)
    resp = requests.get(url)
    if resp.status_code == 200:
        soup = BeautifulSoup(resp.text, 'html.parser')
        links = soup.find_all("p", {"class": "text-center"})
        if len(links) != 1:
            title = soup.find("h1").text
            date = soup.find_all("div", {"class": "myd"})[1].text
            content = soup.find_all("p", {"class": ""})[0].text
            fname, year = conv_to_fname(date)
            path = "DataSet/"
            filename = f"{fname}{i}.utf8"
            filename_with_path = os.path.join(path, filename)
            with open(filename_with_path, 'w+') as f:
                f.write(f"<DOC>\n<DOCNO>{filename}</DOCNO>\n<TEXT>\n{title}\n{date}\n{content}\n</TEXT>\n</DOC>")
            print(f"Writing file {filename} to {path}")
        else:
            print("File missing at " + url)
    else:
        print("Error")
