import sys

# Load Lithuanian lemmas from the dictionary
with open('dazninis.utf8.txt', 'r', encoding='utf-8') as dic:
    lt_lemmas = [line.split('\t')[0] for line in dic]

# Define input and output files
input_file = sys.argv[1]
output_file = sys.argv[2]

# Process the input file and correct lemmas
with open(input_file, 'r', encoding='utf-8') as file:
    with open(output_file, 'w', encoding='utf-8') as doc:
        for line in file:
            line = line.split('\t')
            if len(line) == 10:
                lemma = line[2]

                # Check if the word is a verb with specific prefixes
                if line[3] == 'VERB':
                    if lemma.startswith('nebe'):
                        if lemma.startswith('nebesi'):
                            if 'Reflex=Yes' in line[5] and lemma[6:] + 's' in lt_lemmas:
                                line[2] = lemma[6:] + 's'
                        elif lemma[4:] in lt_lemmas:
                            line[2] = lemma[4:]
                        elif lemma[2:] in lt_lemmas:
                            line[2] = lemma[2:]
                    elif lemma.startswith('ne'):
                        if 'Reflex=Yes' in line[5]:
                            line[2] = lemma[4:] + 's'
                        elif lemma[2:] in lt_lemmas:
                            line[2] = lemma[2:]
                    elif lemma.startswith('tebe'):
                        if 'Reflex=Yes' in line[5]:
                            line[2] = lemma[6:] + 's'
                        elif lemma[2:] in lt_lemmas:
                            line[2] = lemma[2:]
                        else:
                            line[2] = lemma[4:]
                    elif lemma.startswith('be'):
                        if 'Reflex=Yes' in line[5]:
                            line[2] = lemma[4:] + 's'
                        elif lemma[2:] in lt_lemmas:
                            line[2] = lemma[2:]

                # Write the corrected line to the output file
                line1 = '\t'.join(line)
                doc.write(line1)
            else:
                # Write non-verb lines as they are
                line1 = '\t'.join(line)
                doc.write(line1)
