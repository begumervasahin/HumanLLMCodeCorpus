import os
import collections
import re

DATA_DIR = "data"
DIMENSIONS = ["IE", "NS", "FT", "PJ"]
WORDS_TO_REMOVE = [
    "intj", "intp", "infj", "infp", "istj", "istp", "isfj", "isfp",
    "entj", "entp", "enfj", "enfp", "estj", "estp", "esfj", "esfp",
    "si", "ni", "ti", "fi", "se", "ne", "te", "fe", "nt", "nf",
    "sxsp", "spsx", "spso", "sxso", "sosp", "sosx", "sp", "sx",
    "sj", "sf", "st", "le", "socionic", "socionics", "enneagram",
    "d", "w", "mbti",
]

for dimension in DIMENSIONS:
    wordcount_a = collections.Counter()
    wordcount_b = collections.Counter()

    # Count words in file for dimension[0]
    with open(os.path.join(DATA_DIR, f"extreme_examples_{dimension[0]}.txt"), "r") as f:
        wordcount_a.update(f.read().split())

    # Count words in file for dimension[1]
    with open(os.path.join(DATA_DIR, f"extreme_examples_{dimension[1]}.txt"), "r") as f:
        wordcount_b.update(f.read().split())

    # Combine unique keys
    cache = set(wordcount_a.keys()) | set(wordcount_b.keys())

    # Calculate differences in word counts
    a = {}
    b = {}
    for key in cache:
        if key in wordcount_a:
            if key in wordcount_b:
                if wordcount_a[key] > wordcount_b[key]:
                    a[key] = wordcount_a[key] - wordcount_b[key]
                elif wordcount_a[key] < wordcount_b[key]:
                    b[key] = wordcount_b[key] - wordcount_a[key]
            else:
                a[key] = wordcount_a[key]
        elif key in wordcount_b:
            b[key] = wordcount_b[key]

    # Write special words to files
    regex = re.compile("[^a-zA-Z]")
    with open(os.path.join(DATA_DIR, f"special_words_{dimension[0]}.txt"), "w") as f:
        for key, count in a.items():
            mod = regex.sub("", str(key))
            if mod not in WORDS_TO_REMOVE and count > 2:
                f.write(f"{mod}\n" * count)

    with open(os.path.join(DATA_DIR, f"special_words_{dimension[1]}.txt"), "w") as f:
        for key, count in b.items():
            mod = regex.sub("", str(key))
            if mod not in WORDS_TO_REMOVE and count > 2:
                f.write(f"{mod}\n" * count)
