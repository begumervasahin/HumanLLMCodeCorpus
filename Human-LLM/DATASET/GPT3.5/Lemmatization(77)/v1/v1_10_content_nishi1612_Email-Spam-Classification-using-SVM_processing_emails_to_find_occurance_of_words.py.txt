import os
import string
import numpy as np
import pandas as pd
from time import time
from nltk.corpus import stopwords
from nltk.stem.wordnet import WordNetLemmatizer

# Start time
start_time = time()

# Read the words list CSV file
df = pd.read_csv('wordslist.csv', header=0)
words = df['word']

# Initialize WordNet Lemmatizer
lmtzr = WordNetLemmatizer()

# Directory containing emails
directory_in_str = "emails/"
directory = os.fsencode(directory_in_str)

# Open the frequency CSV file for writing
with open("frequency.csv", "w+") as f:
    # Write the header
    for i in words:
        f.write(str(i) + ',')
    f.write('output\n')

# Iterate over files in the directory
k = 0
for file in os.listdir(directory):
    file = file.decode("utf-8")
    file_name = str(os.getcwd()) + '/emails/' + file.replace("b'", "").replace("'", "")
    k += 1
    
    # Open the file for reading
    with open(file_name, "r", encoding='utf-8', errors='ignore') as file_reading:
        words_list_array = np.zeros(words.size)
        # Iterate over words in the file
        for word in file_reading.read().split():
            word = lmtzr.lemmatize(word.lower())
            if (word in stopwords.words('english') or word in string.punctuation or len(word) <= 2 or word.isdigit()):
                continue
            # Increment word count in words_list_array
            for i, w in enumerate(words):
                if w == word:
                    words_list_array[i] += 1
                    break
        # Write word frequency to CSV file
        with open("frequency.csv", "a") as f:
            for i in range(words.size):
                f.write(str(int(words_list_array[i])) + ',')
            # Write output label
            if len(file_name) == 68:
                f.write("-1\n")
            elif len(file_name) == 71:
                f.write("1\n")
    if k % 100 == 0:
        print("Done", k)

# Print execution time
print("Time (in seconds) to segregate entire dataset to form input vector:", round(time() - start_time, 2))
