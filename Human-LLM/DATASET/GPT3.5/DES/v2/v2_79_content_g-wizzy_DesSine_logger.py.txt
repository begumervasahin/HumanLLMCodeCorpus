class bcolors:
    HEADER = '\033[95m'
    OKBLUE = '\033[94m'
    OKGREEN = '\033[92m'
    WARNING = '\033[93m'
    FAIL = '\033[91m'
    ENDC = '\033[0m'
    BOLD = '\033[1m'
    UNDERLINE = '\033[4m'

def error(prefix, line, message):
    """
    Print an error message with specific formatting.

    Args:
        prefix (str): Prefix for the log entry.
        line (int): Line number where the error occurred.
        message (str): Message to display.
    """
    print(f"{bcolors.FAIL}[{prefix}] {bcolors.ENDC}{bcolors.BOLD}Line {line}{bcolors.ENDC} : {message}")

def warning(prefix, message):
    """
    Print a warning message with specific formatting.

    Args:
        prefix (str): Prefix for the log entry.
        message (str): Message to display.
    """
    print(f"{bcolors.WARNING}[{prefix}] {bcolors.ENDC}{message}")

def info(prefix, message):
    """
    Print an information message with specific formatting.

    Args:
        prefix (str): Prefix for the log entry.
        message (str): Message to display.
    """
    print(f"{bcolors.OKGREEN}[{prefix}] {bcolors.ENDC}{message}")

def debug(variables):
    """
    Print debugging information with specific formatting.

    Args:
        variables (list): List of printable objects.
    """
    print(f"{bcolors.OKBLUE}[DesSine Debug] {bcolors.ENDC}{', '.join(map(lambda var: str(var), variables)))}")

# Example usage
if __name__ == "__main__":
    error("ERROR", 42, "This is an error message")
    warning("WARNING", "This is a warning message")
    info("INFO", "This is an information message")
    debug(["Debugging", "information"])
