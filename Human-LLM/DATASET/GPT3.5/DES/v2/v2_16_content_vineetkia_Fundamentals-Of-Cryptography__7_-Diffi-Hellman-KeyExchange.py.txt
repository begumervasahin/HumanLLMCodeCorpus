def power(base, exponent, modulus):
    """
    Calculate the power modulo modulus.

    Parameters:
        base (int): The base.
        exponent (int): The exponent.
        modulus (int): The modulus.

    Returns:
        int: The result of (base^exponent) % modulus.
    """
    if exponent == 1:
        return base
    else:
        return pow(base, exponent, modulus)
