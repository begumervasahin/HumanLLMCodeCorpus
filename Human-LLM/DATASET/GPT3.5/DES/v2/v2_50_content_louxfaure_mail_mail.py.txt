import os
import smtplib
import logging
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText

class Mail:
    def __init__(self, maillogin, mailpwd, smtpserver, smtpport):
        self.logger = logging.getLogger()
        if maillogin is None:
            raise ValueError("Please provide a valid mail login")
        if mailpwd is None:
            raise ValueError("Please provide a valid mail password")
        if smtpserver is None:
            raise ValueError("Please provide a valid SMTP server")
        if smtpport is None:
            raise ValueError("Please provide a valid SMTP port")
        self.maillogin = maillogin
        self.mailpwd = mailpwd
        self.smtpserver = smtpserver
        self.smtpport = smtpport

    def send_email(self, mailfrom, mailto, subject, body):
        msg = MIMEMultipart()
        msg['From'] = mailfrom
        msg['To'] = mailto
        msg['Subject'] = subject
        msg.attach(MIMEText(body))

        try:
            with smtplib.SMTP_SSL(self.smtpserver, self.smtpport) as mailserver:
                mailserver.login(self.maillogin, self.mailpwd)
                mailserver.sendmail(mailfrom, mailto, msg.as_string())
            return "Email sent successfully"
        except Exception as e:
            return f"Failed to send email: {e}"

# Example usage
mail = Mail(
    os.getenv('MAIL_LOGIN'),
    os.getenv('MAIL_PWD'),
    os.getenv('SMTP_SERVER'),
    os.getenv('SMTP_PORT')
)

result = mail.send_email('sender@example.com', 'recipient@example.com', 'Test Subject', 'This is a test email.')
print(result)
