import sys
from collections import OrderedDict

def create_header_and_wordset(string):
    """
    Function to create a header and unique word set from a string.
    Returns header_bytes and set_palavras.
    """
    char_list = list(string)
    word_hold = ""
    word_list = []
    len_string = len(string)

    for index, char in enumerate(char_list):
        if char.isalnum():
            word_hold += char
            if not len_string - 1 == index:
                if not char_list[index + 1].isalnum():
                    if len(word_hold) > 3:
                        word_list.append(word_hold)
                    word_hold = ""
            else:
                word_list.append(word_hold)
                word_hold = ""

    unique_words = list(OrderedDict.fromkeys(word_list))
    header_bytes = len(unique_words).to_bytes(2, byteorder='big')
    return header_bytes, unique_words

def list_words(unique_words):
    """
    Function to convert a set of words into bytes, separated by commas.
    """
    if not unique_words:
        return b','

    word_bytes = ','.join(unique_words).encode()
    return word_bytes

def create_dict(unique_words):
    """
    Function to create a dictionary with word addresses based on their positions.
    """
    word_dict = {}
    first_byte = 0
    second_byte = 0

    for word in unique_words:
        word_dict[word] = bytes([255, second_byte, first_byte])
        first_byte += 1
        if first_byte == 256:
            first_byte = 0
            second_byte += 1

    return word_dict

def compress_string(word_dict, string_to_compress):
    """
    Function to compress a string using a dictionary.
    """
    compressed_string = b''
    len_string = len(string_to_compress)
    word_buffer = ""

    for index, char in enumerate(string_to_compress):
        if char.isalnum():
            word_buffer += char
            if not len_string - 1 == index:
                if not string_to_compress[index + 1].isalnum():
                    compressed_string += word_dict.get(word_buffer, word_buffer.encode())
                    word_buffer = ""
            else:
                compressed_string += word_dict.get(word_buffer, word_buffer.encode())
                word_buffer = ""
        if not char.isalnum():
            compressed_string += char.encode()

    return compressed_string

if len(sys.argv) != 3:
    print("Usage: python compressor.py -c/-d/-h input_file")
    sys.exit(1)

if sys.argv[1] == '-c':
    with open(sys.argv[2], 'r') as file:
        input_text = file.read()
        header, unique_word_set = create_header_and_wordset(input_text)
        word_bytes = list_words(unique_word_set)
        word_dictionary = create_dict(unique_word_set)
        compressed_data = compress_string(word_dictionary, input_text)
        compressed_output = header + word_bytes + compressed_data

        with open(f"{sys.argv[2]}.cmp", 'wb') as compressed_file:
            compressed_file.write(compressed_output)

        print(f"Successfully compressed text into {sys.argv[2]}.cmp.")

elif sys.argv[1] == '-d':
    with open(sys.argv[2], 'rb') as file:
        header_bytes = int.from_bytes(file.read(2), byteorder='big')
        compressed_data = file.read()
        word_map = []
        reference = 0

        while header_bytes > 0:
            reference = compressed_data.find(b',')
            word_map.append(compressed_data[:reference])
            compressed_data = compressed_data[reference + 1:]
            header_bytes -= 1

        decompressed_text = compressed_data
        index_num = 0

        while index_num < len(word_map):
            decompressed_text = decompressed_text.replace(
                bytes([255]) + (index_num).to_bytes(2, 'big'), word_map[index_num]
            )
            index_num += 1

        output_filename = sys.argv[2][:-4]
        
        with open(f"{output_filename}", 'w') as decompressed_file:
            decompressed_file.write(decompressed_text.decode('utf-8'))

        print(f"Decompressed data saved to {output_filename}")

elif sys.argv[1] == '-h':
    with open(sys.argv[2], 'rb') as file:
        print(file.read())
        print("End of text")
else:
    print("Usage: python compressor.py -c/-d/-h input_file")
