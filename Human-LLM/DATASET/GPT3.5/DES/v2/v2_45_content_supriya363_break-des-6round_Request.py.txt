import requests
import json
import warnings

# Suppress SSL warnings
warnings.filterwarnings('ignore')

# Define the URL and headers for the request
url = "https://172.27.26.163:9999/des"
headers = {
    'Content-type': 'application/json',
    'Orgin': 'https://172.27.26.163:9999',
    'Referer': 'https://172.27.26.163:9999/game/caves.swf'
}

# Open the input file to read plaintext passwords
with open('input.txt', 'r') as f:
    plaintexts = f.readlines()

# Open the output file to write ciphertext passwords
with open('response.txt', 'w') as f2:
    # Define the data payload for the request
    data = {"password": "555acfd4d3c1b7a398d71e4c1538ad4d", "teamname": "Amalgam"}

    # Iterate through each plaintext password in the input file
    for plaintext in plaintexts:
        # Update the plaintext in the data dictionary
        data["plaintext"] = plaintext.split(',')[0].strip()

        # Send a POST request to the server
        r = requests.post(url, json=data, headers=headers, verify=False)

        # Check if the request was successful (status code 200)
        if r.status_code == 200:
            # Parse the response JSON
            response = json.loads(r.text)
            # Write the ciphertext to the output file
            f2.write(f"{response['ciphertext']}\n")
        else:
            # Print a message if failed to retrieve ciphertext
            print("Failed to retrieve ciphertext")

# Print a completion message
print("Task completed.")
