# Placeholder classes and methods for missing imports
class Config:
    pass

class story_code:
    @staticmethod
    def start_point():
        pass

    @staticmethod
    def story_points():
        pass

    @staticmethod
    def tasks():
        pass

if __name__ == "__main__":
    from app.story.story import start_point, story_points, tasks

    def validate_story():
        validate_references()
        validate_lookups()

    def validate_references():
        missing = {
            "points": set(),
            "tasks": set(),
        }
        existing_story_points = story_points.keys()
        referenced_story_points = [start_point] + [dest for point in story_points.values() for dest in point["paths"].values()]
        missing["points"] = {point for point in referenced_story_points if point not in existing_story_points}

        existing_tasks = tasks.keys()
        referenced_tasks = [task for point in story_points.values() for task in point.get("tasks", [])]
        missing["tasks"] = {task for task in referenced_tasks if task not in existing_tasks}

        if missing["points"]:
            raise KeyError(f"There are referenced, but unknown storypoints: {missing['points']}")
        if missing["tasks"]:
            raise KeyError(f"There are referenced, but unknown tasks: {missing['tasks']}")
        print("All references in story.json found!")

    def validate_lookups():
        missing_validation_methods = {
            task["validation_method"] for task in tasks.values()
            if task["validation_method"] not in dir(story_code)
        }
        if missing_validation_methods:
            raise KeyError(
                f"There are referenced, but unknown task validation methods: {missing_validation_methods}"
            )
        print("All validation methods defined!")

        missing_action_methods = {
            action for point in story_points.values() for action in point.get("actions", [])
            if action not in dir(story_code)
        }
        if missing_action_methods:
            raise KeyError(
                f"There are referenced, but unknown server action methods: {missing_action_methods}"
            )
        print("All server action methods defined!")

        missing_placeholders = {
            ph for point in story_points.values()
            for ph in re.findall(r"{(.*?)}", ''.join(point["description"]) + ''.join(point["paths"].keys()))
            if ph not in story_code.placeholder_getters
        }
        if missing_placeholders:
            raise KeyError(
                f"There are referenced, but unknown placeholders: {missing_placeholders}"
            )
        print("All placeholder references found!")

    validate_story()
