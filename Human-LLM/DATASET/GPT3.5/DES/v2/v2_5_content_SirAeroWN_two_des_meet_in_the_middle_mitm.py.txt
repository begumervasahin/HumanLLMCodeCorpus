from multiprocessing import Pool
from functools import partial
from pyDes import des
import array

PLAIN_TEXT = '89504E470D0A1A0A'
CIPHER_TEXT = '89A3F4E3A99337A4'

def encrypt(key, plain_text):
    cipher = des(key)
    return cipher.encrypt(plain_text)

def decrypt(key, cipher_text):
    cipher = des(key)
    return cipher.decrypt(cipher_text)

def generate_keys(num):
    for n in range(num):
        key_bytes = bytearray([0] * 8)
        powers = [7, 6, 5, 4, 3, 2, 1, 0]
        for i, p in enumerate(powers):
            key_bytes[i] = n // (128 ** p)
            n = n - (key_bytes[i] * (128 ** p)) if n >= (128 ** p) else n
        yield bytes(key_bytes)

def convert_string_to_bytes(s):
    if len(s) % 2 != 0:
        raise ValueError("String length must be even")
    else:
        byte_list = [s[i:i + 2] for i in range(0, len(s), 2)]
        return bytes([int(c, 16) for c in byte_list])

def mitm(nkeys, plain_text, cipher_text, pool=None):
    def composed_encrypt(plain_text, key):
        return (encrypt(key, plain_text), key)

    if pool is None:
        table = {}
        for k in generate_keys(nkeys):
            c = encrypt(k, plain_text)
            table[c] = k
    else:
        table = dict(pool.map(partial(composed_encrypt, plain_text), generate_keys(nkeys)))

    for k in generate_keys(nkeys):
        p = decrypt(k, cipher_text)
        if p in table:
            k1 = table[p].hex()
            k2 = k.hex()
            print(f'Found keys: (k1: {k1}, k2: {k2})')
            print('Apply k2 then k1 to decrypt.')
            print('Use a tool such as this one (http://des.online-domain-tools.com/) to decrypt the whole file with the keys.')
            return (k1, k2)

    print('Did not find keys.')
