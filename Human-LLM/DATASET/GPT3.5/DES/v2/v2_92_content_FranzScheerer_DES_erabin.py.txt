# Spritz PRNG Functions
def spritz_update():
    global a, i, j, w, s
    i = (i + w) % 256
    j = s[(j + s[i]) % 256]
    s[i], s[j] = s[j], s[i]

def spritz_output():
    global a, i, j, w, s
    spritz_update()
    return s[j]

def spritz_shuffle():
    global a, i, j, w, s
    for _ in range(256):
        spritz_update()
    w = (w + 2) % 256
    a = 0

def spritz_absorb_nibble(x):
    global a, i, j, w, s
    if a == 240:
        spritz_shuffle()
    s[a], s[240 + x] = s[240 + x], s[a]
    a += 1

def spritz_absorb_byte(b):
    spritz_absorb_nibble(b % 16)
    spritz_absorb_nibble(b // 16)

def spritz_squeeze(out, outlen):
    global a, i, j, w, s
    if a != 0:
        spritz_shuffle()
    for _ in range(outlen):
        out.append(spritz_output())

# Helper Functions
def hash_value(x):
    global a, i, j, w, s
    j = i = a = 0
    w = 1
    s = list(range(256))
    for c in x:
        spritz_absorb_byte(ord(c))
    res = []
    spritz_squeeze(res, 128)
    out = 0
    for bx in res:
        out = (out << 8) + bx
    return out % (2 ** 1000)

def code_to_number(x):
    res = 0
    for c in x:
        if ord(c) >= 48 and ord(c) < 58:
            res = (res << 6) + ord(c) - 48
        if ord(c) >= 65 and ord(c) < 91:
            res = (res << 6) + ord(c) - 55
        if ord(c) >= 97 and ord(c) < 123:
            res = (res << 6) + ord(c) - 61
        if c == "'":
            res = (res << 6) + 62
        if c == '/':
            res = (res << 6) + 63
    return res

def number_to_code(x):
    res = ''
    while x > 0:
        y = x % 64
        if y < 10:
            res = chr(y + 48) + res
        elif y < 36:
            res = chr(y + 55) + res
        elif y < 62:
            res = chr(y + 61) + res
        elif y == 62:
            res = "'" + res
        elif y == 63:
            res = '/' + res
        x //= 64
    return res

def gcd(a, b):
    while b > 0:
        a, b = b, a % b
    return a

def next_prime(p):
    while p % 4 != 3:
        p += 1
    return next_prime_helper(p)

def next_prime_helper(p):
    m_ = 3 * 5 * 7 * 11 * 13 * 17 * 19 * 23 * 29
    while gcd(p, m_) != 1:
        p += 4
    if pow(2, p - 1, p) != 1:
        return next_prime_helper(p + 4)
    if pow(3, p - 1, p) != 1:
        return next_prime_helper(p + 4)
    if pow(5, p - 1, p) != 1:
        return next_prime_helper(p + 4)
    if pow(17, p - 1, p) != 1:
        return next_prime_helper(p + 4)
    return p

def verify_signature(s, fnam):
    p = read_number_from_file("p_")
    q = read_number_from_file("q_")
    a = AFactor
    b = BFactor
    h0 = hash_value(fnam)
    ha = (a * h0) % RABIN_CONSTANT_NUMBER
    hb = (b * h0) % RABIN_CONSTANT_NUMBER
    hab = (a * b * h0) % RABIN_CONSTANT_NUMBER
    sq = (s * s) % RABIN_CONSTANT_NUMBER
    return (h0 == sq) or (ha == sq) or (hb == sq) or (hab == sq)

def write_number_to_file(number, fnam):
    f = open(fnam, 'wb')
    n = number
    while n > 0:
        byte = n % 256
        n //= 256
        f.write(chr(byte))
    f.close()

def read_number_from_file(fnam):
    f = open(fnam, 'rb')
    n = 0
    snum = f.read()
    for i in range(len(snum)):
        n = (n << 8) ^ ord(snum[len(snum) - i - 1])
    f.close()
    return n
