# Function to read a number from a file
def read_number(file_name):
    with open(file_name, 'rb') as file:
        num = 0
        byte_str = file.read()
        for byte in byte_str:
            num = (num << 8) ^ byte
    return num

# Convert hexadecimal text to number
def hex_text_to_num(text):
    result = 0
    for char in text:
        if '0' <= char <= '9':
            result = (result << 4) + ord(char) - 48
        elif 'a' <= char <= 'f':
            result = (result << 4) + ord(char) - 87
        elif 'A' <= char <= 'F':
            result = (result << 4) + ord(char) - 55
    return result

# Extended Euclidean Algorithm to find multiplicative inverse
def inv(b, m):
    s, t, a = 0, 1, m
    while b != 1:
        q = a // b
        a, b = b, a % b
        s, t = t, s - q * t
    if s < 0:
        s += m
    return s

# Hash function
def h(x):
    digest = hashlib.sha256(x.encode()).digest()
    result = 0
    for byte in digest:
        result = (result << 8) ^ byte
    return result % n4

# Generate a point on the elliptic curve
def gen_point(x, a, b):
    while pow(x ** 3 + a * x + b, (prime - 1) // 2, prime) != 1:
        x += 1
    y = pow(x ** 3 + a * x + b, (prime + 1) // 4, prime)
    return [x % prime, y % prime]

# Add two points on the elliptic curve
def add_point(P, Q):
    x1, x2 = P[0], Q[0]
    y1, y2 = P[1], Q[1]
    while x1 < x2:
        x1 += prime
    if x1 == x2:
        s = ((3 * (x1 ** 2) + a) * inv(2 * y1, prime)) % prime
    else:
        s = ((y1 - y2) * inv(x1 - x2, prime)) % prime
    xr = (s ** 2 - x1 - x2) % prime
    yr = (s * (x1 - xr) - y1) % prime
    return [xr, yr]

# Multiply a point by a scalar
def mul_point(P, n):
    result = P[:]
    if n < 0:
        result[1] = prime - result[1]
        n = -n
    for _ in range(1, n):
        result = add_point(result, P)
    return result

# Verify a signature
def verify_signature(G, s, Y, e, m):
    return e == h(str(add_point(mul_point(G, s), mul_point(Y, e))[0]) + m)

# Verify an ECDSA signature
def ecdsa_verify(G, m, S, Y):
    si = inv(S[0], n4)
    hh = h(m + str(S[1]))
    u1 = (si * hh) % n4
    u2 = (si * S[1]) % n4
    return add_point(mul_point(G, u1), mul_point(Y, u2))[0] == S[1]
