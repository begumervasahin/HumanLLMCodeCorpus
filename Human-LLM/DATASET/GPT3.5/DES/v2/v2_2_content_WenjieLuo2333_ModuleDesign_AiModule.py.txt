import random

class AiModule:
    def __init__(self):
        # Initialize lists to store input data
        self.blood_oxygen = []
        self.blood_pressure = []
        self.pulse = []

    def input_check(self, blood_oxygen, blood_pressure, pulse):
        """
        Check the input type from the database to ensure it's of type float.
        """
        for bo, bp, p in zip(blood_oxygen, blood_pressure, pulse):
            if not all(isinstance(val, float) for val in (bo, bp, p)):
                raise TypeError("Input data must be of type float")
            self.blood_oxygen.append(bo)
            self.blood_pressure.append(bp)
            self.pulse.append(p)

    def predict(self):
        """
        Predict blood oxygen, blood pressure, and pulse based on available data.
        """
        if not self.blood_oxygen or not self.blood_pressure or not self.pulse:
            raise ValueError("No data available for prediction")

        rand_index = random.randint(1, len(self.blood_oxygen))
        pred_bo = sum(self.blood_oxygen[:rand_index]) / rand_index
        pred_bp = sum(self.blood_pressure[:rand_index]) / rand_index
        pred_pulse = sum(self.pulse[:rand_index]) / rand_index

        print('Predicted blood oxygen:', pred_bo)
        print('Predicted blood pressure:', pred_bp)
        print('Predicted pulse:', pred_pulse)

        return pred_bo, pred_bp, pred_pulse

# Example usage
ai = AiModule()
blood_oxygen_data = [98.5, 99.2, 97.8, 100.0]
blood_pressure_data = [120.5, 122.1, 118.9, 121.6]
pulse_data = [75.2, 76.5, 74.8, 77.3]

try:
    ai.input_check(blood_oxygen_data, blood_pressure_data, pulse_data)
    ai.predict()
except (TypeError, ValueError) as e:
    print(f"Error: {e}")
