# Function to compute the greatest common divisor
def gcd(a, b):
    while b > 0:
        a, b = b, a % b
    return a

# Function to find the next prime number
def nextPrime(p):
    while p % 6 != 5:
        p += 1
    return nextPrime_odd(p)

# Function to find the next odd prime number
def nextPrime_odd(p):

# Function to read a number from a file
def readNumber(fnam):
    f = open(fnam, 'rb')
    n = 0
    snum = f.read()
    for i in range(len(snum)):
        n = (n << 8) ^ ord(snum[len(snum)-i-1])
    f.close()
    return n

# Function to convert hexadecimal text to a number
def hextxt2num(x):
    res = 0
    for c in x:
        if ord(c) < 58 and ord(c) >= 48:
            res = (res << 4) + ord(c) - 48
        elif ord(c) <= ord('f') and ord(c) >= ord('a'):
            res = (res << 4) + ord(c) - 87
        elif ord(c) <= ord('F') and ord(c) >= ord('A'):
            res = (res << 4) + ord(c) - 55
    return res

# Function to compute the multiplicative inverse modulo m
def inv(b, m):
    s = 0
    t = 1
    a = m
    while b != 1:
        q = a / b
        aa = b
        b = a % b
        a = aa
        ss = t
        t = s - q * t
        s = ss
    if t < 0:
        t += m
    return t

# Function to compute the hash value of a string using SHA-256
def h(x):
    dx1 = hashlib.sha256(x.encode()).digest()
    res = 0
    for cx in dx1:
        res = (res << 8) ^ cx
    return res % hsize

    xr = s**2 - x1 - x2
    yr = s * (x1 - xr) - y1
    return [xr % prime, yr % prime]

# Function to multiply a point on an elliptic curve by a scalar
def mulP(P, n):
    isFirst = True
    resP = P
    if n < 0:
        resP[1] = prime - resP[1]
        n = -n
    PP = resP
    while n > 0:
        if n % 2 != 0:
            if isFirst:
                resP = PP
                isFirst = False
            else:
                resP = addP(resP, PP)
        PP = addP(PP, PP)
        n //= 2
    return resP

# Function to verify a signature
def verify(G, s, Y, e, m):
    return e == h(str(addP(mulP(G, s), mulP(Y, e))[0]) + m)

# Function to verify an ECDSA signature
def ecdsa_v(G, m, S, Y):
    si = inv(S[0], n4)
    hh = h(m + str(S[1]))
    u1 = (si * hh) % n4
    u2 = (si * S[1]) % n4
    return addP(mulP(G, u1), mulP(Y, u2))[0] == S[1]

# Function to write a number to a file
def writeNumber(number, fnam):
    f = open(fnam, 'wb')
    n = number
    while n > 0:
        byte = n % 256
        n //= 256
        f.write(chr(byte))
    f.close()

# Function to generate a Schnorr signature
def signSchnorr(G, m, x):
    k = h(m + 'kk1')
    R = mulP(G, k)
    e = h(str(R[0]) + m)
    return [(k - x * e) % n4, e]

# Function to generate an ECDSA signature
def ecdsa(G, m, x):
    k = h(m + 'ecdsa')
    R = mulP(G, k)
    hh = h(m + str(R[0]))
    s = (inv(k, n4) * (hh + R[0] * x)) % n4
    return [s, R[0]]
