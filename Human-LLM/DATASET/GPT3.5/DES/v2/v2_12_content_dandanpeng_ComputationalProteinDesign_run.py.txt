# Import necessary classes and functions from function.py and seqpred.py
from function import Protein, Matches, genetic_algorithm
from seqpred import *

# List of protein IDs
protein_ids = ['1aba', '1bxv', '1by2', '1hyp', '1opc', '1tmy', '2acy', '2mcm', '3e0e', '3k63']

# Loop through each protein ID and perform genetic algorithm
for protein_id in protein_ids:
    # Create Protein object
    protein = Protein(protein_id)
    # Create Matches object
    match = Matches(protein.match)
    # Perform genetic algorithm with specified parameters
    genetic_algorithm(protein, match, population_size=100, generations=50, tournament_size=2, mutation_rate=0.03, elitism=2)
