import os
import re
import folium
import pandas as pd
import geopandas as gpd

# Function to determine color based on candidate
def get_color(row):
    max_vote = max(row['plq'], row['pq'], row['caq'], row['qs'])
    if row['plq'] == max_vote:
        return 'red'
    elif row['pq'] == max_vote:
        return 'blue'
    elif row['caq'] == max_vote:
        return 'lightblue'
    elif row['qs'] == max_vote:
        return 'orange'
    else:
        return 'white'

# List of coverage areas
coverage_areas = ['104', '110', '116', '120', '126', '132', '138', '144', '150', '204', '206', '210', '212', '216', '218', '220', '224', '226', '230', '232', '236', '238', '240', '244', '246', '250', '252', '256', '258', '260', '264', '502', '508', '514', '542', '548', '554', '560', '566', '554', '594', '570', '576', '660', '666', '670', '676', '708', '714', '720', '742', '754', '760', '802', '806', '810', '814', '818', '822', '826', '834', '838', '842', '850', '854', '858', '902', '906', '914', '918', '922', '926', '930', '938']

# Regular expressions to match candidate names
re_caq = re.compile('.*C\.A\.Q\.-É\.F\.L\.$')
re_qs = re.compile('.*Q\.S\.$')
re_plq = re.compile('.*P\.L\.Q\./Q\.L\.P\.$')
re_pq = re.compile('.* P\.Q\.$')
re_pvq = re.compile('.*P\.V\.Q\./G\.P\.Q\.$')
re_pmlq = re.compile('.* P\.M\.L\.Q\.$')
re_pcq = re.compile('.* P.C.Q./C.P.Q.$')
re_npdq = re.compile('.* N\.P\.D\.Q.$')
re_bp = re.compile('.* B\.P\.$')
re_ea = re.compile('.* É\.A\.$')
re_pn = re.compile('.* P\.N\.$')

# Load sections data
sections_path = os.path.join('data', 'sherbrooke.json')
sections = gpd.read_file(sections_path)

# Filter sections based on coverage areas
sections = sections[sections['CO_CEP'].isin(coverage_areas)]

# Load and process results data
cumulative_data = pd.DataFrame()
for coverage_area in coverage_areas:
    results_path = os.path.join('data', 'resultats-section-vote', f"{coverage_area}.csv")
    data = pd.read_csv(results_path, sep=';', encoding='latin1', dtype={'S.V.': 'str', 'Code': 'str'})
    data.dropna(subset=['S.V.'], inplace=True)
    data.rename(columns={"Code": "CO_CEP", 'Nom des Municipalités': 'NM_MUNI'}, inplace=True)

    # Rename columns based on candidate names
    candidates = data.columns.tolist()[7:-3]
    for candidate in candidates:
        if re_caq.match(candidate):
            data.rename(columns={candidate: 'caq'}, inplace=True)
        elif re_qs.match(candidate):
            data.rename(columns={candidate: 'qs'}, inplace=True)
        elif re_plq.match(candidate):
            data.rename(columns={candidate: 'plq'}, inplace=True)
        elif re_pq.match(candidate):
            data.rename(columns={candidate: 'pq'}, inplace=True)
        elif re_pvq.match(candidate):
            data.rename(columns={candidate: 'pvq'}, inplace=True)
        elif re_pmlq.match(candidate):
            data.rename(columns={candidate: 'pmlq'}, inplace=True)
        elif re_pcq.match(candidate):
            data.rename(columns={candidate: 'pcq'}, inplace=True)
        elif re_npdq.match(candidate):
            data.rename(columns={candidate: 'npdq'}, inplace=True)
        elif re_bp.match(candidate):
            data.rename(columns={candidate: 'bp'}, inplace=True)
        elif re_ea.match(candidate):
            data.rename(columns={candidate: 'ea'}, inplace=True)
        elif re_pn.match(candidate):
            data.rename(columns={candidate: 'pn'}, inplace=True)
        else:
            print("WARNING: candidate:{} not matched".format(candidate))
            data.drop(labels=candidate, axis=1, inplace=True)

    data.drop(labels=['Circonscription', 'Date scrutin', 'Étendue', 'Secteur', 'Regroupement', 'É.I.'], axis=1, inplace=True)
    if cumulative_data.empty:
        cumulative_data = data
    else:
        cumulative_data = cumulative_data.append(data, sort=False)

# Select relevant columns for the map
selected_columns = cumulative_data[['plq', 'pq', 'caq', 'qs', 'NM_MUNI']]
sections = sections.join(selected_columns)

# Apply getColor function to determine section color
sections['couleur'] = sections.apply(get_color, axis=1)

# Generate Folium map
m = folium.Map([45.5, -72], tiles='stamentoner', zoom_start=8)

# Add GeoJson layer to the map
folium.GeoJson(
    sections,
    tooltip=folium.features.GeoJsonTooltip(fields=['NM_CEP', 'S.V.', 'couleur', 'NM_MUNI'], labels=True),
    style_function=lambda dd: {
        'fillColor': dd['properties']['couleur'],
        'color': 'green',
        'weight': 0.25,
        'fillOpacity': 0.5,
    },
).add_to(m)

# Save the map as HTML file
region_sherbrooke_html = os.path.join('docs', 'region_sherbrooke.html')
m.save(region_sherbrooke_html)
