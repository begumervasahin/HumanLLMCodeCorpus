import math
import numpy as np
import matplotlib.pyplot as plt
from Bio import Align
from Bio.PDB.PDBParser import PDBParser
from Bio.SubsMat.MatrixInfo import blosum62
from Bio.PDB.Polypeptide import PPBuilder
from classProtein import Protein, Match
from collections import Counter
from random import random

def cuckoo_search_new(protein, match, pa):
    aligner = Align.PairwiseAligner()
    aligner.open_gap_score = -10
    aligner.extend_gap_score = -0.5
    aligner.substitution_matrix = blosum62

    lower_bound = [0] * len(protein.terms)
    upper_bound = [match.frags_count[i] for i in match.frags_count]

    def create_individual():
        individual = []
        for term in protein.terms:
            rand = random()
            if rand <= 0.75:
                individual.append(round(random() * match.frags_count[term]))
            else:
                individual.append(match.frags_count[term])
        return individual

    def initial_population(pop_size):
        population = np.zeros((pop_size, len(protein.terms)), dtype=int)
        for i in range(pop_size):
            population[i] = create_individual()
        return population

    def compare_aa(individual):
        frag_num = dict(zip(protein.terms, individual))
        frag_seq = {}
        for term in protein.terms:
            frag_seq[term] = match.select_frag(term, frag_num[term])
        score = 0
        for edge in protein.graph.edges:
            if type(frag_seq[edge[0]]) == np.ndarray and type(frag_seq[edge[1]]) == np.ndarray:
                for pos in protein.graph.edges[edge]['sameAA']:
                    u_aa = frag_seq[edge[0]][pos[0]]
                    v_aa = frag_seq[edge[1]][pos[1]]
                    score += aligner.score(u_aa, v_aa)
        return score / len(protein.terms)

    def term_count(individual):
        sel_frag = dict(zip(protein.terms, individual))
        is_null = [sel_frag[term] < match.frags_count[term] for term in protein.terms]
        return np.sum(is_null) / len(protein.terms)

    def energy(individual):
        return compare_aa(individual) - 10 * term_count(individual)

    def simple_bounds(s, lb, ub):
        for i in range(len(s)):
            if s[i] < lb[i]:
                s[i] = lb[i]
            if s[i] > ub[i]:
                s[i] = ub[i]
        s = s.astype(np.int)
        return s

    def get_cuckoos(nest, best, lb, ub):
        beta = 1.5
        sigma_u = (math.gamma(1 + beta) * math.sin(math.pi * beta / 2) /
                   (math.gamma((1 + beta) / 2) * beta * (2 ** ((beta - 1) / 2)))) ** (1 / beta)
        sigma_v = 1
        for i in range(len(nest)):
            s = nest[i]
            u = np.random.normal(0, sigma_u, 1)
            v = np.random.normal(0, sigma_v, 1)
            step = u / ((abs(v)) ** (1 / beta))
            stepsize = 0.01 * step * (s - best)
            s = s + stepsize * np.random.randn(len(s))
            nest[i] = simple_bounds(s, lb, ub)
        return nest

    def get_best_nest(nest, new_nest, fitness):
        for j in range(len(nest)):
            fnew = energy(new_nest[j])
            if fnew >= fitness[j]:
                fitness[j] = fnew
                nest[j] = new_nest[j]
        fmax = max(fitness.values())
        best_nest = max(fitness, key=fitness.get)
        return fmax, best_nest, nest, fitness

    def empty_nests(nest, lb, ub, pa):
        K = np.random.uniform(0, 1, size=nest.shape) > pa
        nest1 = nest.copy()
        nest2 = nest.copy()
        np.random.shuffle(nest1)
        np.random.shuffle(nest2)
        step_size = np.random.uniform(0, 1) * (nest1 - nest2)
        new_nest = nest + (step_size * K).astype(int)
        for i in range(len(new_nest)):
            new_nest[i] = simple_bounds(new_nest[i], lb, ub)
        return new_nest

    def restore_seq(individual):
        possible_aa = [[] for _ in protein.terms]
        candidate = dict(zip(protein.terms, possible_aa))
        frag_num = dict(zip(protein.terms, individual))
        frag_seq = {}
        for term in protein.terms:
            frag_seq[term] = match.select_frag(term, frag_num[term])
        for pos in protein.inverse:
            for term in protein.inverse[pos]:
                if type(frag_seq[term]) == np.ndarray:
                    index = protein.neighbors[term].index(pos)
                    candidate[pos].append(frag_seq[term][index])
        possible_seq = ''
        for term in protein.terms:
            if candidate[term]:
                possible_seq += Counter(candidate[term]).most_common(1)[0][0]
            else:
                possible_seq += '-'
        return possible_seq

    def real_seq():
        structure = PDBParser().get_structure(protein.protein_id, protein.protein_id + '.pdb')
        ppb = PPBuilder()
        seq = ''
        for pp in ppb.build_peptides(structure):
            seq += pp.get_sequence()
        return seq

    original_seq = real_seq()
    nest = initial_population(100)
    fitness = {}
    for i in range(len(nest)):
        fitness[i] = energy(nest[i])
    max_fitness, best_nest, nest, fitness = get_best_nest(nest, nest, fitness)
    fitness_scores = []
    alignment_scores = []
    for i in range(50):
        nest_a = nest.copy()
        new_nest = get_cuckoos(nest_a, best_nest, lower_bound, upper_bound)
        max_fitness, best_nest, nest, fitness = get_best_nest(nest, new_nest, fitness)
        nest_b = nest.copy()
        new_nest = empty_nests(nest_b, lower_bound, upper_bound, pa)
        max_fitness, best_nest, nest, fitness = get_best_nest(nest, new_nest, fitness)
        fitness_scores.append(max_fitness)
        prediction = restore_seq(nest[best_nest])
        alignment_scores.append(aligner.score(prediction, original_seq))
    plt.plot(fitness_scores)
    plt.savefig(protein.protein_id + '_fitness.jpg')
    plt.close()
    plt.plot(alignment_scores)
    plt.savefig(protein.protein_id + '_align.jpg')
    plt.close()

# Example usage
# Replace the following lines with appropriate input for protein, match, and pa parameters
protein = Protein()  # Define your Protein object
match = Match()      # Define your Match object
pa = 0.5             # Define your pa parameter

# Call the function
cuckoo_search_new(protein, match, pa)
