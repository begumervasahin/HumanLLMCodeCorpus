import os
import re

def concat_files(readme):
    """
    Concatenate files referenced in README.md using include and end tags.

    Parameters:
    - readme: str, path to the README.md file
    """
    result_doc = ""
    matched = False

    with open(readme, "r+", encoding="utf-8") as f:
        for line in f:
            if not matched:
                result_doc += line

            include_match = re.search(r"<!--\s*include\s+(.+?)\s*-->", line)
            if include_match and not matched:
                matched = True
                included_file = include_match.group(1)
                with open(included_file, "r") as inc:
                    for included_line in inc:
                        result_doc += included_line

            elif re.search(r"<!--\s*end\s*-->", line):
                matched = False
                result_doc += "\n" + line
                continue

    with open(readme, "w") as f:
        f.write(result_doc)

if __name__ == '__main__':
    concat_files("README.md")
