from nltk.corpus import stopwords
from nltk.corpus import wordnet
import difflib as dl
import os
import string

# Importing necessary functions from difflib
get_close_matches = dl.get_close_matches

# Function to strip punctuation from words
def strip_punctuation(word):
    return word.translate(str.maketrans('', '', string.punctuation))

# Function to read a text file and remove stopwords
def process_text_file(file_path):
    with open(file_path, 'r') as file:
        lines = file.read()
        word_tokens = lines.split()
        stop_words = set(stopwords.words('english'))
        filtered_sentence = [strip_punctuation(w) for w in word_tokens if w.lower() not in stop_words]
    return filtered_sentence

# Performing OCR on the image and saving text to a file
os.system("tesseract new_img.png st_sheet.txt")

# Reading and processing the reference text file
ref_lines = process_text_file("new_text.txt")

# Reading and processing the OCR text file
ocr_lines = process_text_file("st_sheet.txt")

# Creating a dictionary to store synonyms for each word in the reference text
synonyms = {word: [] for word in ref_lines}
for word, synonyms_list in synonyms.items():
    for synset in wordnet.synsets(word):
        for lemma in synset.lemmas():
            synonyms_list.append(lemma.name())

# Initializing counters
num_synonyms_matched = 0
num_words_matched = 0

# Counting the number of matches between the reference and OCR text
for word in ref_lines:
    if get_close_matches(word, ocr_lines):
        num_synonyms_matched += 1
for word in ref_lines:
    if get_close_matches(word, ocr_lines):
        num_words_matched += 1

# Calculating the similarity ratio
similarity_ratio = float(num_words_matched + num_synonyms_matched) / float(len(ref_lines) + num_synonyms_matched)

# Printing the result
print("The Student is awarded")
print('%d%% Marks' % int(similarity_ratio * 100))
