# Global variables for Rabin Signature
MODULUS = 0

# Function to update Rabin signature parameters
def update_state():
    global index_i, index_j, index_a, step_w, state_S
    index_i = (index_i + step_w) % 256
    index_j = state_S[(index_j + state_S[index_i]) % 256]
    state_S[index_i], state_S[index_j] = state_S[index_j], state_S[index_i]

# Function to output Rabin signature
def output_state():
    global index_i, index_j, index_a, step_w, state_S
    update_state()
    return state_S[index_j]

# Function to shuffle Rabin signature parameters
def shuffle_state():
    global index_i, index_j, index_a, step_w, state_S
    for _ in range(256):
        update_state()
    step_w = (step_w + 2) % 256
    index_a = 0

# Function to absorb a nibble for Rabin signature
def absorb_nibble(x):
    global index_i, index_j, index_a, step_w, state_S
    if index_a == 240:
        shuffle_state()
    state_S[index_a], state_S[240 + x] = state_S[240 + x], state_S[index_a]
    index_a += 1

# Function to absorb a byte for Rabin signature
def absorb_byte(byte):
    absorb_nibble(byte % 16)
    absorb_nibble(byte // 16)

# Function to squeeze Rabin signature output
def squeeze(out, outlen):
    global index_i, index_j, index_a, step_w, state_S
    if index_a != 0:
        shuffle_state()
    for _ in range(outlen):
        out.append(output_state())

# Function to compute hash function for Rabin signature
def hash_function(data):
    global index_i, index_j, index_a, step_w, state_S, MODULUS
    index_j = index_i = index_a = 0
    step_w = 1
    state_S = list(range(256))
    for char in data:
        absorb_byte(ord(char))
    result = []
    squeeze(result, 128)
    out = 0
    for byte in result:
        out = (out << 8) + byte
    return out % MODULUS

# Function to convert code to number
def code_to_number(code):
    result = 0
    for char in code:
        if '0' <= char <= '9':
            result = (result << 6) + ord(char) - 48
        elif 'A' <= char <= 'Z':
            result = (result << 6) + ord(char) - 55
        elif 'a' <= char <= 'z':
            result = (result << 6) + ord(char) - 61
        elif char == '.':
            result = (result << 6) + 62
        elif char == '/':
            result = (result << 6) + 63
    return result

# Function to convert number to code
def number_to_code(number):
    result = ''
    while number > 0:
        remainder = number % 64
        if remainder < 10:
            result = chr(remainder + 48) + result
        elif remainder < 36:
            result = chr(remainder + 55) + result
        elif remainder < 62:
            result = chr(remainder + 61) + result
        elif remainder == 62:
            result = '.' + result
        elif remainder == 63:
            result = '/' + result
        number //= 64
    return result

# Function to find greatest common divisor
def find_gcd(a, b):
    while b > 0:
        a, b = b, a % b
    return a

# Function to find the next prime number
def next_prime(candidate):
    while candidate % 4 != 3:
        candidate += 1
    return find_next_prime(candidate)

def find_next_prime(candidate):

# Function to compute Rabin Signature
def compute_signature(p, q):
    a_factor = 3
    b_factor = 3
    while pow(a_factor, (p-1)//2, p) == 1 or pow(a_factor, (q-1)//2, q) != 1:
        a_factor += 1
    while pow(b_factor, (p-1)//2, p) != 1 or pow(b_factor, (q-1)//2, q) == 1:
        b_factor += 1
    return a_factor, b_factor

# Function to compute square root mod pq
def square_root_mod_pq(message_hash, p, q):
    global MODULUS
    x = hash_function(message_hash)
    a, b = compute_signature(p, q)
    MODULUS = p * q
    if pow(x, (p-1)//2, p) > 1:
        x *= a
    if pow(x, (q-1)//2, q) > 1:
        x *= b
    if pow(x, (q-1)//2, q) != 1 or pow(x, (p-1)//2, p) != 1:
        print('ERROR')
        return -1
    return (pow(p, q-2, q) * p * pow(x, (q+1)//4, q) + pow(q, p-2, p) * q * pow(x, (p+1)//4, p)) % MODULUS

# Function to write number to file
def write_to_file(number, filename):
    with open(filename, 'wb') as f:
        while number > 0:
            byte = number % 256
            number //= 256
            f.write(chr(byte))

# Function to read number from file
def read_from_file(filename):
    with open(filename, 'rb') as f:
        num = 0
        byte_string = f.read()
        for i in range(len(byte_string)):
            num = (num << 8) ^ ord(byte_string[len(byte_string)-i-1])
    return num

# Function to compute hash of a file
def hash_file(filename):
    with open(filename, 'r') as f:
        return hash_function(f.read())
