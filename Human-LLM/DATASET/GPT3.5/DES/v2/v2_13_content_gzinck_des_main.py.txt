# Import necessary modules
from cli.menus.main_menu import main_menu
import global_settings
from tempfile import TemporaryDirectory

# Function to initialize global settings
def initialize_settings():
    """
    Initialize global settings required for the application.
    """
    global_settings.initialize()

# Function to run the DES application on the command line
def run_application():
    """
    Start the DES application and navigate through different screens.
    """
    # Initialize screens and automata
    screens = [main_menu]
    automata = []

    # Use TemporaryDirectory for temporary file operations
    with TemporaryDirectory() as temp_dir:
        # Loop until all screens are processed
        while len(screens) > 0:
            # Get the next screen to be processed
            next_screen = screens[-1]
            # Execute the next screen function
            next_screen(screens, automata, temp_dir)

# Main function to start the application
if __name__ == "__main__":
    # Initialize global settings
    initialize_settings()
    # Run the DES application
    run_application()
