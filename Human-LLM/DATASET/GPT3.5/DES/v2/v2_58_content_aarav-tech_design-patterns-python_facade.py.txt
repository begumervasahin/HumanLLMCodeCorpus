# Define a Cook class that helps in preparing a dish
class Cook:
    # Method to prepare the dish
    def prepareDish(self):
        # Instantiate a Cutter object to cut vegetables
        cutter = Cutter()
        cutter.cutVegetables()  # Call the cutVegetables method of Cutter
        # Instantiate a Boiler object to boil vegetables
        boiler = Boiler()
        boiler.boilVegetables()  # Call the boilVegetables method of Boiler
        # Instantiate a Frier object to fry vegetables
        frier = Frier()
        frier.fry()  # Call the fry method of Frier

# Define a Cutter class that provides the feature of cutting vegetables
class Cutter:
    # Method to cut vegetables
    def cutVegetables(self):
        print("All vegetables are cut")  # Print a message indicating that vegetables are cut

# Define a Boiler class that provides the feature of boiling vegetables
class Boiler:
    # Method to boil vegetables
    def boilVegetables(self):
        print("All vegetables are boiled")  # Print a message indicating that vegetables are boiled

# Define a Frier class that provides the feature of frying vegetables
class Frier:
    # Method to fry vegetables
    def fry(self):
        print("All vegetables are mixed and fried.")  # Print a message indicating that vegetables are fried

# Check if the script is being run directly
if __name__ == "__main__":
    cook = Cook()  # Instantiate a Cook object
    cook.prepareDish()  # Call the prepareDish method of Cook to prepare the dish
