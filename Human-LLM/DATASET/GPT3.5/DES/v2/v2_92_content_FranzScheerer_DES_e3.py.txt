# Function to write a number to a file
def write_number(number, filename):
    with open(filename, 'wb') as file:
        n = number
        while n > 0:
            byte = n % 256
            n = n // 256
            file.write(chr(byte))

# Function to convert binary to a number
def binary_to_number(binary):
    result = 0
    for char in binary:
        result = (result << 8) ^ ord(char)
    return result

# Function to convert a number to binary
def number_to_binary(number):
    result = ''
    while number > 0:
        result = chr(number % 256) + result
        number //= 256
    return result

# Function to calculate the greatest common divisor
def greatest_common_divisor(a, b):
    while b > 0:
        a, b = b, a % b
    return a

# Function to find the next prime number
def next_prime(p):
    while p % 3 != 2 or p % 2 != 1:
        p += 1
    return next_prime_odd(p)

# Function to find the next odd prime number
def next_prime_odd(p):
    m_ = 5 * 7 * 11 * 13 * 17 * 19 * 23 * 29
    while True:
        while greatest_common_divisor(p, m_) != 1:
            p += 6
        q = (p + 1) // 4
        if (pow(2, p - 1, p) != 1 or pow(2, q - 1, q) != 1) or \
           (pow(3, p - 1, p) != 1 or pow(3, q - 1, q) != 1) or \
           (pow(5, p - 1, p) != 1 or pow(5, q - 1, q) != 1) or \
           (pow(17, p - 1, p) != 1 or pow(17, q - 1, q) != 1):
            p += 6
            continue
        break
    return p

# Function to calculate the modular inverse
def modular_inverse(b, m):
    s, t = 0, 1
    a = m
    while b != 1:
        q = a // b
        a, b = b, a % b
        aa = b
        b = a % b
        a = aa
        ss = t
        t = s - q * t
        s = ss
    if t < 0:
        t += m
    return t

# Function to perform elliptic curve encryption
def elliptic_curve_encryption(m, key, prime):
    return pow(m, key, prime)

# Function to perform elliptic curve decryption
def elliptic_curve_decryption(c, key, prime):
    key = modular_inverse(key, prime - 1)
    return pow(c, key, prime)

# Function to convert hexadecimal text to a number
def hex_text_to_number(text):
    result = 0
    for char in text:
        if 48 <= ord(char) < 58:
            result = (result << 4) + ord(char) - 48
        elif 97 <= ord(char) < 123:
            result = (result << 4) + ord(char) - 87
    return result

# Function to convert code to a number
def code_to_number(text):
    result = 0
    for char in text:
        if 48 <= ord(char) < 58:
            result = (result << 6) + ord(char) - 48
        elif 65 <= ord(char) < 91:
            result = (result << 6) + ord(char) - 55
        elif 97 <= ord(char) < 123:
            result = (result << 6) + ord(char) - 61
        elif char == "'":
            result = (result << 6) + 62
        elif char == '/':
            result = (result << 6) + 63
    return result

# Function to convert a number to a code
def number_to_code(number):
    result = ''
    while number > 0:
        y = number % 64
        if y < 10:
            result = chr(y + 48) + result
        elif 10 <= y < 36:
            result = chr(y + 55) + result
        elif 36 <= y < 62:
            result = chr(y + 61) + result
        elif y == 62:
            result = "'" + result
        elif y == 63:
            result = '/' + result
        number //= 64
    return result

# Function to calculate the hash value
def calculate_hash(x):
    dx1 = hashlib.sha256(x).digest()
    result = 0
    for char in dx1:
        result = (result << 8) ^ ord(char)
    return result % n4

# Function to generate a random 256-bit number
def random_256(m):
    md = hashlib.sha256("***RANDOM-SEED_X***")
    md.update('large key value for generation of random number')
    md.update(m)
    result = 0
    largestr = md.digest()
    for i in range(len(largestr)):
        result = (result << 8) ^ ord(largestr[i])
    return result

# Function to generate a random number
def random_x(m):
    md = hashlib.sha256("***RANDOM-SEED_X***")
    md.update('large key value for generation of random number')
    md.update(m)
    md.update(str(time.gmtime().tm_year + time.gmtime().tm_mday))
    result = 0
    largestr = md.digest()
    for i in range(len(largestr)):
        result = (result << 8) ^ ord(largestr[i])
    return result

# Function to add two points on the elliptic curve

    else:

    xr = s ** 2 - x1 - x2
    yr = s * (x1 - xr) - y1
    return [xr % prime, yr % prime]

# Function to multiply a point on the elliptic curve by a scalar
def multiply_point(P, n):
    is_first = True
    res_P = P
    if n < 0:
        res_P[1] = prime - res_P[1]
        n = (-1) * n
    PP = res_P
    while n > 0:
        if n & 1:
            if is_first:
                res_P = PP
                is_first = False
            else:
                res_P = add_points(res_P, PP)
        PP = add_points(PP, PP)
        n >>= 1
    return res_P

# Function to sign a message using Schnorr signature
def sign_schnorr(G, m, x):
    k = random_x(m)
    R = multiply_point(G, k)
    e = calculate_hash(str(R[0]) + m)
    return [(k - x * e) % n4, e]

# Function to perform ECDSA (Elliptic Curve Digital Signature Algorithm)
def ecdsa(G, m, x):
    k = random_x(m)
    R = multiply_point(G, k)
    hh = calculate_hash(m + str(R[0]))
    s = (modular_inverse(k, n4) * (hh + R[0] * x)) % n4
    return [s, R[0]]

# Function to verify ECDSA signature
def ecdsa_v(G, m, S, Y):
    si = modular_inverse(S[0], n4)
    hh = calculate_hash(m + str(S[1]))
    u1 = (si * hh) % n4
    u2 = (si * S[1]) % n4
    return add_points(multiply_point(G, u1), multiply_point(Y, u2))[0] == S[1]
