def read_roulement(file_path):
    """
    Reads the 'roulement.csv' file and returns the data as a list of lists.
    Each inner list represents a row from the CSV file.
    """
    data = []  # List to store the CSV data
    with open(file_path, 'r') as csv_file:
        # Read the header and convert it into a list of column names
        header = csv_file.readline().strip().split('\t')

        # Read each line of the CSV file and append it to the data list
        for line in csv_file:
            row = line.strip().split('\t')  # Split the line into columns
            data.append(row)  # Append the row to the data list
    return data

def find_indices_by_date_and_kh(date, kh, data):
    """
    Returns the indices of columns in the CSV data where the given date is found in the first column,
    and the given 'kh' value is found in subsequent columns.
    """
    indices = []  # List to store the indices of matching columns
    for i, row in enumerate(data):
        if date in row[0]:  # Check if the date is found in the first column of the current row
            for j, value in enumerate(row):
                if str(kh) in value:  # Check if the 'kh' value is found in any column of the row
                    indices.append(j)  # Append the index of the matching column
    return indices

# Example usage
file_path = "resources/roulement.csv"
csv_data = read_roulement(file_path)

# Assuming 'date' and 'kh' are provided
date = "2024-04-06"
kh = 12345
matching_indices = find_indices_by_date_and_kh(date, kh, csv_data)

if not matching_indices:
    print("No matching indices found.")
else:
    for index in matching_indices:
        print(csv_data[0][index])  # Assuming the header is in the first row of csv_data
