import GrayAPI
import colorama as clr

# Initialize colorama for colored output
clr.init(autoreset=True)

# Prompt user to input the topic to get news on
topic = input('Input the topic to get news on: ')

# Create a GrayNews object for the specified topic
news_object = GrayAPI.GrayNews(topic)

# Make a request to fetch news related to the specified topic
print('Making request: ', news_object.makeRequest())

# Print any request error, if occurred
print('Request Error: ', news_object.getRequestError())

# Print the source of the news
print('Source: ', news_object.getSource(), end='\n\n')

# Get the list of articles
articles = news_object.getArticles()

# Print the number of articles fetched
print('Number of Articles: ', len(articles), end='\n\n')

# Iterate through each article and print its details with colored output
for article in articles:
    print(clr.Fore.RED + 'Title: ', clr.Fore.GREEN + str(article.getTitle()))
    print(clr.Fore.RED + 'Author: ', clr.Fore.GREEN + str(article.getAuthor()))
    print(clr.Fore.RED + 'URL: ', clr.Fore.GREEN + str(article.getUrl()))
    print(clr.Fore.RED + 'Description: ', clr.Fore.GREEN + str(article.getShortDesc()))
    print(clr.Fore.RED + 'Image: ', clr.Fore.GREEN + str(article.getImage()))
    print(clr.Fore.RED + 'Published: ', clr.Fore.GREEN + str(article.getPublishedAt()))
    print('\n\n')
