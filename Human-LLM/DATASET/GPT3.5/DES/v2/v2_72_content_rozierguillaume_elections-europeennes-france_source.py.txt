import csv
import matplotlib.pyplot as plt
import numpy as np
import time
import datetime
import scipy
import scipy.interpolate
import json

# Define colors for plotting
colors = ['#1f77b4', '#ff7f0e', '#2ca02c', '#d62728', '#9467bd', '#8c564b', '#e377c2', '#7f7f7f', '#bcbd22', '#17becf']

# Total number of seats
nb_sieges = 74

def read_surveys(path):
    """
    Read survey data from a CSV file and return relevant information.
    """
    surveys = []
    surveys_dict = {}

    # Read CSV file
    with open(path) as csv_file:
        csv_reader = csv.reader(csv_file, delimiter=",")
        line_count = -1

        # Process each row in the CSV
        for row in csv_reader:
            surveys.append(row)

            # Skip header row
            if line_count == -1:
                parties = row[2:]
            else:
                survey_str = np.array([row[i] for i in range(2, 19)])
                survey_str[survey_str == ''] = float("nan")
                surveys_dict[line_count] = {
                    "date": row[0],
                    "results": survey_str.astype(float)
                }
            line_count += 1

    # Extract dates and survey data
    dates = np.array([x[0] for x in surveys[1:]])
    data = np.array([np.array(d[2:]) for d in np.asarray(surveys[1:])])
    data = replace_empty_with_nan(data)

    return dates, data, parties, surveys_dict

def replace_empty_with_nan(data):
    """
    Replace empty strings in the data with NaN values.
    """
    data[data == ''] = float("nan")
    return data.astype(float)

def plot_surveys(dates, data, colors, export=False):
    """
    Plot survey data using Matplotlib.
    """
    fig, ax = plt.subplots(figsize=(15, 8))
    fig.suptitle('Intention de vote aux élections européennes en France\n(hypothèses liste gilets jaunes et UK dans UE)', fontsize=15)
    ax.set_xlabel('Dates')
    ax.set_ylabel('Intention de vote (%)')

    # Convert dates to timestamps
    dates_timestamp = [float(time.mktime(datetime.datetime.strptime(s, "%Y-%m-%d").timetuple())) for s in dates]

    # Plot data
    for i in range(len(data[0])):
        d = np.array([data[j][i] for j in range(len(data))]).astype(float)
        x, y = zip(*sorted((xVal, np.mean([yVal for a, yVal in zip(dates_timestamp, d) if xVal == a])) for xVal in set(dates_timestamp)))
        interp = scipy.interpolate.interp1d(x, y)
        ax.plot(dates, d, 'o', color=colors[i])
        ax.plot(dates, [interp(x) for x in dates_timestamp], color=colors[i], label=parties[i], linewidth=1.5)
        ax.legend(bbox_to_anchor=(0.75, 0.35, 0.5, 0.5))

    ax.grid(True, which='major', linestyle='-', alpha=0.6)
    plt.minorticks_on()
    ax.grid(True, which='minor', linestyle=':', alpha=0.2)

    # Save plot if export is True
    if export:
        fig.savefig('export/sondages/' + max(dates), dpi=200)

def calculate_obtained_seats(parties, surveys_dict, nb_seats, colors, export=False):
    """
    Calculate obtained seats based on survey results.
    """
    for id in surveys_dict:
        # Initialize variables
        nb_seats_remaining = nb_sieges
        s = 0
        index = 0

        for res in surveys_dict[id]['results']:
            if res >= 50:
                s = nb_sieges // 2 + 1 if nb_sieges > 4 else nb_sieges // 2
                nb_seats_remaining = nb_sieges - s
                index = surveys_dict[id]['results'].tolist().index(res)

        # Calculate seats
        seats = (surveys_dict[id]['results'] * nb_seats_remaining / 100).astype(int)

        # Adjust seats based on conditions
        for i, val in enumerate(surveys_dict[id]['results']):
            if (val < 5) or not (val >= 0):
                surveys_dict[id]['results'][i] = 0
                seats[i] = 0

        remaining = nb_sieges - sum(seats) - s

        while remaining > 0:
            calc = surveys_dict[id]['results'] / (seats + 1)
            index_remaining_seat = np.argmax(calc)
            seats[index_remaining_seat] += 1
            remaining = nb_sieges - sum(seats) - s

        seats[index] += s

        # Update surveys dictionary
        surveys_dict[id]['results'] = surveys_dict[id]['results'].tolist()
        surveys_dict[id]['seats'] = seats.tolist()

    # Export data if specified
    if export:
        with open('export/data/sondages_avec_sieges.json', 'w') as file:
            json.dump(surveys_dict, file, indent=4)

    return parties, surveys_dict

def plot_seats_pie_chart(parties, obtained_seats, colors, export=False):
    """
    Plot seats distribution as a pie chart.
    """
    fig, ax = plt.subplots(figsize=(8, 8))

    # Remove parties with 0 seats
    parties_filtered = [party for i, party in enumerate(parties) if obtained_seats[i] != 0]
    seats_filtered = [seats for seats in obtained_seats if seats != 0]
    colors_filtered = [colors[i] for i, seats in enumerate(obtained_seats) if seats != 0]

    wedges, _, _ = ax.pie(seats_filtered, labels=parties_filtered, autopct=lambda p: '{:.0f} sièges\n({:.1f} %)'.format(sum(seats_filtered) * p / 100, p), startangle=90, colors=colors_filtered, textprops=dict(color="w", fontsize=13))
    ax.axis('equal')
    ax.legend(wedges, parties_filtered, loc="center right", bbox_to_anchor=(1.1, 0))
    fig.suptitle('Projection nombre de sièges, dernier sondage\n(hypothèses liste gilets jaunes et UK dans UE)', fontsize=15)

    # Save plot if export is True
    if export:
        fig.savefig('export/pie_chart_sieges/' + max(dates), dpi=200)

parties, surveys_dict = calculate_obtained_seats(parties, surveys_dict, nb_sieges, colors, export=True)
