import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# Set figure size for plots
plt.rcParams['figure.figsize'] = (5.0, 5.0)

# Read data from CSV file
data = pd.read_csv('data.csv')

# Extract X and Y values
X = data.iloc[:, 0]
Y = data.iloc[:, 1]

# Calculate mean of X and Y
mean_X = np.mean(X)
mean_Y = np.mean(Y)

# Calculate deviations from mean
deviation_X = X - mean_X
deviation_Y = Y - mean_Y

# Calculate squared deviations
squared_deviation_X = deviation_X * deviation_X
squared_deviation_Y = deviation_Y * deviation_Y

# Calculate slope (m) and intercept (c) using the method of least squares
m = np.sum(deviation_X * deviation_Y) / np.sum(squared_deviation_X)
c = mean_Y - m * mean_X

# Plot original data and regression line using least squares
plt.scatter(X, Y)
plt.plot(X, m*X + c, color='red')
plt.title('Linear Regression (Least Squares)')
plt.xlabel('X')
plt.ylabel('Y')
plt.show()

# Initialize parameters for gradient descent
m = 0
c = 0
learning_rate = 0.0001
epochs = 1000
n = float(len(X))

# Perform gradient descent
for i in range(epochs):
    Y_pred = m * X + c
    gradient_m = (-2/n) * sum(X * (Y - Y_pred))
    gradient_c = (-2/n) * sum(Y - Y_pred)
    m = m - learning_rate * gradient_m
    c = c - learning_rate * gradient_c

# Plot original data and regression line obtained from gradient descent
plt.scatter(X, Y)
plt.plot([min(X), max(X)], [min(m*X + c), max(m*X + c)], color='red')
plt.title('Linear Regression (Gradient Descent)')
plt.xlabel('X')
plt.ylabel('Y')
plt.show()

# Print slope (m) and intercept (c) obtained from gradient descent
print("Slope (m):", m)
print("Intercept (c):", c)
