from abc import ABC, abstractmethod

class BaseCommand(ABC):
    '''
    Abstract base command class.
    '''
    @abstractmethod
    def execute(self):
        '''
        Abstract method to execute a command.
        '''
        raise NotImplementedError("Please implement in a subclass.")

class EmailCommand(BaseCommand):
    '''
    Email command class.
    '''
    def __init__(self, receiver, data):
        '''
        Initializes the EmailCommand with a receiver and email data.
        '''
        self.receiver = receiver
        self.data = data

    def execute(self):
        '''
        Executes the email command by sending an email through the receiver.
        '''
        self.receiver.send_email(self.data)

class SMSCommand(BaseCommand):
    '''
    SMS command class.
    '''
    def __init__(self, receiver, data):
        '''
        Initializes the SMSCommand with a receiver and SMS data.
        '''
        self.receiver = receiver
        self.data = data

    def execute(self):
        '''
        Executes the SMS command by sending an SMS through the receiver.
        '''
        self.receiver.send_sms(self.data)

class NotificationService:
    '''
    Notification service class acting as a receiver for commands.
    '''
    def send_email(self, data):
        '''
        Sends an email with the given data.
        '''
        print("Sending email:", data)

    def send_sms(self, data):
        '''
        Sends an SMS with the given data.
        '''
        print("Sending SMS:", data)

class NotificationInvoker:
    '''
    Invoker class responsible for invoking commands and maintaining history.
    '''
    def __init__(self):
        '''
        Initializes the NotificationInvoker with an empty notification history.
        '''
        self.notification_history = []

    def invoke(self, command):
        '''
        Invokes a command by executing it and appending it to the notification history.
        '''
        self.notification_history.append(command)
        command.execute()

if __name__ == "__main__":
    # Example usage
    invoker = NotificationInvoker()
    receiver = NotificationService()

    # Create email and SMS commands
    email_command = EmailCommand(receiver, {"subject": "Test Email"})
    sms_command = SMSCommand(receiver, {"subject": "Test SMS"})

    # Invoke the commands
    invoker.invoke(email_command)
    invoker.invoke(sms_command)

    # Print the notification history
    print("Notification History:", invoker.notification_history)
