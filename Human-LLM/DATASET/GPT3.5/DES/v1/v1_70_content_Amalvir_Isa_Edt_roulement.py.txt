def read_roulement(file_path):
    """Reads the 'roulement.csv' file and returns the data as a list of lists."""
    with open(file_path, 'r') as roulement:
        Lroul = []
        entete = roulement.readline().strip().split('\t')
        for ligne in roulement:
            Lroul.append(ligne.strip().split('\t'))
        for i in range(len(Lroul)):
            print(str(i) + " " + str(Lroul[i][0]))
            for j in range(len(Lroul[i])):
                Lroul[i][j] = Lroul[i][j].split(',')
        return Lroul

def rlmnt(date, kh, Lroul):
    """Returns the indices of the header."""
    j = -1
    liste = []
    for i in range(len(Lroul)):
        if date in Lroul[i][0]:
            j = i
    if j < 0:
        print("Date introuvable !")
        return None
    else:
        for k in range(len(Lroul[j])):
            try:
                Lroul[j][k].index(str(kh))
                liste.append(k)
            except ValueError:
                pass
    return liste

# Example usage
file_path = "resources/roulement.csv"
Lroul = read_roulement(file_path)

# Assuming 'date' and 'kh' are provided
