class Alert:
    def __init__(self):
        self.fine_turn = 0
        self.bo_flag = 0
        self.bp_flag = 0
        self.pul_flag = 0

    def get_bo_data(self, data):
        if self.fine_turn > 20:
            self.bo_flag = 0
        if 0 <= data <= 0.3:
            self.bo_flag += 1
        else:
            self.fine_turn += 1

    def get_bp_data(self, data):
        if self.fine_turn > 20:
            self.bp_flag = 0
        if 60 <= data <= 90:
            self.bp_flag += 1
        else:
            self.fine_turn += 1

    def get_pul_data(self, data):
        if self.fine_turn > 20:
            self.pul_flag = 0
        if 90 <= data <= 120:
            self.pul_flag += 1
        else:
            self.fine_turn += 1

    def alert_output(self):
        """
        Compare data with certain threshold
        send flags to user interface module.
        """
        if self.bo_flag != 0:
            return 1
        elif self.bp_flag != 0:
            return 2
        elif self.pul_flag != 0:
            return 3
        else:
            return 0

# Example usage:
alert_instance = Alert()

# Simulating data input
alert_instance.get_bo_data(0.2)
alert_instance.get_bp_data(80)
alert_instance.get_pul_data(100)

# Getting alert output
alert_output = alert_instance.alert_output()

# Printing alert output
print("Alert Output:", alert_output)
