def write_number(number, filename):
    with open(filename, 'wb') as f:
        n = number
        while n > 0:
            byte = n % 256
            n //= 256
            f.write(chr(byte))

def bin_to_num(x):
    res = 0
    for c in x:
        res = (res << 8) ^ ord(c)
    return res

def num_to_bin(x):
    res = ''
    while x > 0:
        res = chr(x % 256) + res
        x //= 256
    return res

def gcd(a, b):
    while b > 0:
        a, b = b, a % b
    return a

def next_prime(p):
    while p % 8 != 3:
        p += 1
    return next_prime_odd(p)

def next_prime_odd(p):
    m_ = 3 * 5 * 7 * 11 * 13 * 17 * 19 * 23 * 29
    while True:
        while gcd(p, m_) != 1:
            p += 8
        q = (p + 1) // 4
        if (pow(2, p - 1, p) != 1 or pow(2, q - 1, q) != 1 or 
            pow(3, p - 1, p) != 1 or pow(3, q - 1, q) != 1 or 
            pow(5, p - 1, p) != 1 or pow(5, q - 1, q) != 1 or 
            pow(17, p - 1, p) != 1 or pow(17, q - 1, q) != 1):
            p += 8
            continue
        break
    return p

def inv(b, m):
    s = 0
    t = 1
    a = m
    while b != 1:
        q = a // b
        aa = b
        b = a % b
        a = aa
        ss = t
        t = s - q * t
        s = ss
    if t < 0:
        t += m
    return t

def D(c, key, prime):
    key = inv(key, prime - 1)
    return pow(c, key, prime)

def hex_txt_to_num(x):
    res = 0
    for c in x:
        if ord(c) < 58 and ord(c) >= 48:
            res = (res << 4) + ord(c) - 48
        elif ord(c) <= ord('f') and ord(c) >= ord('a'):
            res = (res << 4) + ord(c) - 87

def code_to_num(x):
    res = 0
    for c in x:
        if ord(c) >= 48 and ord(c) < 58:
            res = (res << 6) + ord(c) - 48
        if ord(c) >= 65 and ord(c) < 91:
            res = (res << 6) + ord(c) - 55
        if ord(c) >= 97 and ord(c) < 123:
            res = (res << 6) + ord(c) - 61
        if c == "'":
            res = (res << 6) + 62
        if c == '/':
            res = (res << 6) + 63
    return res

def num_to_code(x):
    res = ''
    while x > 0:
        y = x % 64
        if y < 10:
            res = chr(y + 48) + res
        elif y < 36:
            res = chr(y + 55) + res
        elif y < 62:
            res = chr(y + 61) + res
        elif y == 62:
            res = "'" + res
        elif y == 63:
            res = '/' + res
        x //= 64
    return res

def h(x):
    dx1 = hashlib.sha256(x.encode()).digest()
    res = 0
    for cx in dx1:
        res = (res << 8) ^ cx
    return res % n4

def random256(m):
    md = hashlib.sha256("***RANDOM-SEED_X***".encode())
    md.update('large key value for generation of random number'.encode())
    md.update(m.encode())
    result = 0
    largestr = md.digest()
    for i in range(len(largestr)):
        result = (result << 8) ^ ord(largestr[i])
    return result

def randomX(m):
    md = hashlib.sha256("***RANDOM-SEED_X***".encode())
    md.update('large key value for generation of random number'.encode())
    md.update(m.encode())
    md.update(str(time.gmtime().tm_year + time.gmtime().tm_mday).encode())
    result = 0
    largestr = md.digest()
    for i in range(len(largestr)):
        result = (result << 8) ^ ord(largestr[i])
    return result

def genP(x, a, b):
    if (4 * a**3 + 27 * b**2) % prime == 0:
        b += 1
    while pow(c * x**3 + a * x + b, (prime - 1) / 2, prime) != 1:
        x += 1
    y = pow(c * x**3 + a * x + b, (prime + 1) / 4, prime)
    return [x % prime, y % prime]

def dpoint(P):
    x = P[0]
    y = P[1]
    s = ((3 * c * (x**2) + a) * inv(2 * y, prime)) % prime
    xr = (cinv * s**2 - 2 * x) % prime
    yr = (-y + s * (x - xr)) % prime
    return [xr , yr]

def addP(P, Q):
    if P == Q:
        return dpoint(P)
    x1 = P[0]
    x2 = Q[0]
    y1 = P[1]
    y2 = Q[1]
    while x1 < x2:
        x1 += prime
    while y1 < y2:
        y1 += prime
    s = ((y1 - y2) * inv(x1 - x2, prime)) % prime
    xr = (cinv * s**2 - x1 - x2) % prime
    yr = (s * (x1 - xr) - y1) % prime
    return [xr % prime, yr % prime]

def mulP(P, n):
    isFirst = True
    resP = P
    if n < 0:
        resP[1] = prime - resP[1]
        n = (-1) * n
    bsize = 20
    while 2**bsize < n:
        bsize += 1
    PP = resP
    for b in range(bsize + 1):
        if (n & (1 << b) != 0):
            if isFirst:
                resP = PP
                isFirst = False
            else:
                resP = addP(resP, PP)
        PP = dpoint(PP)
    return resP

def sign_schnorr(G, m, x):
    k = randomX(m)
    R = mulP(G, k)
    e = h(str(R[0]) + m)
    return [(k - x * e) % n4, e]

def ecdsa(G, m, x):
    hh = h(m)
    k = 2 * randomX(m) + 1
    R = mulP(G, k)
    s = (inv(k, n4) * (hh + R[0] * x)) % n4
    return [s, R[0]]

def ecdsa_verify(G, m, S, Y):
    si = inv(S[0], n4)
    hh = h(m)
    u1 = (si * h(m)) % n4
    u2 = (si * S[1]) % n4
    return addP(mulP(G, u1), mulP(Y, u2))[0] == S[1]
