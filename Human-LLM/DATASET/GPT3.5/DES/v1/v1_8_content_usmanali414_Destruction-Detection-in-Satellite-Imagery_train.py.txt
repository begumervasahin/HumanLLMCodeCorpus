import utility
from sklearn.utils import shuffle
from keras.models import Model
from keras.layers import Input, Dense, Multiply, Dropout, Lambda
from keras.optimizers import Adam
import numpy as np
from optparse import OptionParser

def custom_loss(layer):
    """
    Define a custom loss function for the model.
    """
    def loss(y_true, y_pred):
        # Compute attention weights and the attention-weighted loss
        attention_weights = K.squeeze(layer, axis=2)
        attention_sum = K.sum(attention_weights, axis=-1)
        attention_sum_reshaped = K.reshape(attention_sum, (-1, 1))
        attention_normalized = Lambda(lambda x: x[0] / x[1])([attention_weights, attention_sum_reshaped])
        return K.binary_crossentropy(y_true, y_pred) + (0.3 * K.sum(attention_normalized, axis=-1))
    return loss

def model_attention():
    """
    Define the model architecture.
    """
    # Define input layer
    model_input = Input(shape=(None, 1024))
    
    # Hidden layers
    x = Dense(512, kernel_initializer='glorot_uniform', bias_initializer='zeros')(model_input)
    x = Dropout(0.8)(x)
    x = Dense(1, kernel_initializer='glorot_uniform', bias_initializer='zeros')(x)
    att_weights = Dense(1, activation='sigmoid')(x)
    att_weights = Lambda(lambda x: x / K.sum(x, axis=1, keepdims=True))(att_weights)
    att_mull = Multiply()([model_input, att_weights])
    y = Dense(512, kernel_initializer='glorot_uniform', bias_initializer='zeros')(att_mull)
    y = Dropout(0.7)(y)
    y = Dense(1, activation='sigmoid', kernel_initializer='glorot_uniform', bias_initializer='zeros')(y)

    # Create the model
    model = Model(model_input, y)

    # Compile the model
    adam = Adam(lr=0.001, beta_1=0.9, beta_2=0.999, epsilon=None, decay=0.0, amsgrad=True)
    model.compile(optimizer=adam, loss=custom_loss(att_weights), metrics=['accuracy'])

    return model

def train_model(model, train_dest, train_non_dest, val_dest, val_non_dest, batch=32, epochs=500):
    batch_size = batch
    half_batch = int(batch_size / 2)
    iterations = 100
    epoch_loss = []
    epoch_acc = []
    val_loss_list = []
    val_acc_list = []
    val_label_1 = np.zeros(len(val_non_dest))
    val_label_2 = np.ones(len(val_dest))
    val_data = np.concatenate((val_non_dest, val_dest), axis=0)
    val_label = np.concatenate((val_label_1, val_label_2), axis=0)
    
    for epoch in range(epochs):
        print("Training Epoch:", epoch)
        iter_loss = []
        iter_acc = []
        start_1, start_2 = 0, 0
        end_1, end_2 = half_batch, half_batch
        total_2 = len(train_dest)
        
        for _ in range(iterations):
            pre_1 = train_non_dest[start_1:end_1]
            if (total_2 - end_2) < 8:
                start_2 = 0
                end_2 = half_batch
            post_1 = train_dest[start_2:end_2]
            data = np.concatenate((pre_1, post_1), axis=0)
            label_1 = np.zeros(half_batch)
            label_2 = np.ones(half_batch)
            labels = np.concatenate((label_1, label_2), axis=0)
            loss, acc = model.train_on_batch(data, labels)
            iter_loss.append(loss)
            iter_acc.append(acc)
            start_1 = end_1
            end_1 = start_1 + half_batch
            start_2 = end_2
            end_2 = start_2 + half_batch
        
        avg_loss = sum(iter_loss) / len(iter_loss)
        avg_acc = sum(iter_acc) / len(iter_acc)
        epoch_loss.append(avg_loss)
        epoch_acc.append(avg_acc)
        
        val_loss, val_acc = model.evaluate(val_data, val_label)
        val_loss_list.append(val_loss)
        val_acc_list.append(val_acc)
        
        print("Training Loss and Accuracy: Loss: {:.4f}, Accuracy: {:.4f}".format(avg_loss, avg_acc))
        print("Validation Loss and Accuracy: Loss: {:.4f}, Accuracy: {:.4f}".format(val_loss, val_acc))
        
        if epoch % 100 == 0:
            model.save('models/Model1_AttentionNetwork_{}.h5'.format(epoch))
    
    return model, val_loss_list, val_acc_list

def load_features(filename):
    """
    Load features from file.
    """
    train_destruct_features, train_non_destruct_features = utility.loadtrainfeatures(filename)
    train_destruct_features = np.array(train_destruct_features)
    train_destruct_features = np.reshape(train_destruct_features, (-1, 9, 1024))
    train_non_destruct_features = np.array(train_non_destruct_features)
    train_non_destruct_features = np.reshape(train_non_destruct_features, (-1, 9, 1024))
    
    return train_destruct_features, train_non_destruct_features

if __name__ == "__main__":
    parser = OptionParser()
    parser.add_option("--trainfeatures_filename", dest="trainfeatures_filename", help="Path to train features file.")
    parser.add_option("--epochs", dest="epochs", type="int", help="Number of epochs to train model")
    (options, args) = parser.parse_args()

    if not options.trainfeatures_filename:
        parser.error('Error: Features file name must be specified. Pass --trainfeatures_filename to command line')
    
    print("Loading Features...")
    train_dest_features, train_non_dest_features = load_features(options.trainfeatures_filename)
    
    print("Splitting data into training and testing....")
    train_dest_features = shuffle(train_dest_features, random_state=2)
    train_non_dest_features = shuffle(train_non_dest_features, random_state=2)
    
    print("Loading attention model....")
    model = model_attention()
    
    print("Starting Training..!!")
    model, val_loss, val_acc = train_model(model, train_dest_features, train_non_dest_features, val_dest, val_non_dest, batch=32, epochs=options.epochs)
    
    print("Training Done!!!")
    print("Saving model..!!")
    model.save("models/Model1_AttentionNetwork_{}.h5".format(options.epochs))
