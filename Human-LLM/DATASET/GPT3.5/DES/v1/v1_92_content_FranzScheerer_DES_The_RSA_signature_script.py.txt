def code_to_num(x):
    res = 0
    for c in x:
        if '0' <= c <= '9':
            res = (res << 6) + ord(c) - ord('0')
        elif 'A' <= c <= 'Z':
            res = (res << 6) + ord(c) - ord('A') + 10
        elif 'a' <= c <= 'z':
            res = (res << 6) + ord(c) - ord('a') + 36
        elif c == '+':
            res = (res << 6) + 62
        elif c == '/':
            res = (res << 6) + 63
    return res

def num_to_code(x):
    res = ''
    while x > 0:
        y = x % 64
        if y < 10:
            res = chr(y + ord('0')) + res
        elif y < 36:
            res = chr(y + ord('A') - 10) + res
        elif y < 62:
            res = chr(y + ord('a') - 36) + res
        elif y == 62:
            res = '+' + res
        elif y == 63:
            res = '/' + res
        x //= 64
    return res

def bin_to_num(x):
    res = 0
    for c in x:
        res = (res << 8) ^ ord(c)
    return res

def num_to_bin(x):
    res = ''
    while x > 0:
        res = chr(x % 256) + res
        x //= 256
    return res

def digital_to_num(x):
    res = 0
    for c in x:
        if '0' <= c <= '9':
            res = (res * 10) + ord(c) - ord('0')
    return res

def h(x):
    dx1 = hashlib.sha512(x.encode()).digest()
    dx2 = hashlib.sha512(dx1 + x.encode()).digest()
    dx3 = hashlib.sha512(x.encode() + dx2).digest()
    dx4 = hashlib.sha512(x.encode() + dx3).digest()
    dx5 = hashlib.sha512(x.encode() + dx4).digest()
    res = 0
    for cx in (dx1 + dx2 + dx3 + dx4 + dx5):
        res = (res << 8) ^ cx
    return res % N_RSA

def inv(b, m):
    s = 0
    t = 1
    a = m
    while b != 1:
        q = a // b
        aa = b
        b = a % b
        a = aa
        ss = t
        t = s - q * t
        s = ss
    if t < 0:
        t = t + m
    return t

def next_prime(p):
    if p % 2 == 0:
        p = p + 1
    return next_prime_odd(p)

def next_prime_odd(p):
    m_ = 3 * 5 * 7 * 11 * 13 * 17 * 19 * 23 * 29
    while gcd(p, m_) != 1:
        p = p + 2
    if (pow(2, p - 1, p) != 1) or (pow(3, p - 1, p) != 1) or (pow(5, p - 1, p) != 1) or (pow(17, p - 1, p) != 1):
        return next_prime_odd(p + 2)
    return p

def write_number(number, fnam):
    with open(fnam, 'wb') as f:
        n = number
        while n > 0:
            byte = n % 256
            n = n // 256
            f.write(bytes([byte]))

def read_number(fnam):
    with open(fnam, 'rb') as f:
        n = 0
        snum = f.read()
        for i in range(len(snum)):
            n = (n << 8) ^ ord(snum[len(snum) - i - 1])
    return n

def random512():
    md = hashlib.sha512("RANDOM-SEED".encode())
    md.update('large key value for generation of random number'.encode())
    md.update(str(random.random()).encode())
    md.update(str(random.random()).encode())
    result = 0
    largestr = md.digest()
    for cx in largestr:
        result = (result << 8) ^ cx
    return result

def random1024():
    return random512() * random512()
