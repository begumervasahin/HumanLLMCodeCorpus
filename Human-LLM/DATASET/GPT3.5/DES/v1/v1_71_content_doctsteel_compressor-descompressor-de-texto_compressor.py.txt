import sys
from collections import OrderedDict

def create_header_and_wordset(string):
    """
    Function to create a header and unique word set from a string.
    Returns header_bytes and set_palavras.
    """
    char_list = list(string)
    word_hold = ""
    word_list = []
    len_frase = len(string)
    for index, char in enumerate(char_list):
        if char.isalnum():
            word_hold += char
            if not len_frase - 1 == index:
                if not char_list[index + 1].isalnum():
                    if len(word_hold) > 3:
                        word_list.append(word_hold)
                    word_hold = ""
            else:
                word_list.append(word_hold)
                word_hold = ""
    somelist = [x for x in word_list if len(x) > 3]
    set_palavras = list(OrderedDict.fromkeys(somelist))
    len_cabecalho = len(set_palavras)
    header_bytes = (len_cabecalho).to_bytes(2, byteorder='big')
    return header_bytes, set_palavras

def list_words(set_palavras):
    """
    Function to convert a set of words into bytes, separated by commas.
    """
    parte_2 = ""
    if not set_palavras:
        parte_2 = ','
    for palavra in set_palavras:
        parte_2 = parte_2 + palavra + ','
    return str.encode(parte_2)

def create_dict(set_palavras):
    """
    Function to create a dictionary with word addresses based on their positions.
    """
    parte_3_dict = {}
    contador_first_byte = 0
    contador_second_byte = 0
    for palavra in set_palavras:
        parte_3_dict[palavra] = bytes([255, contador_second_byte, contador_first_byte])
        contador_first_byte += 1
        if contador_first_byte == 256:
            contador_first_byte = 0
            contador_second_byte += 1
    return parte_3_dict

def compress_string(stringdict, stringmaltratada):
    """
    Function to compress a string using a dictionary.
    """
    aux_word = ""
    final = b''
    len_frase = len(stringmaltratada)
    for index, char in enumerate(stringmaltratada):
        if char.isalnum():
            aux_word += char
            if not len_frase - 1 == index:
                if not stringmaltratada[index + 1].isalnum():
                    final += stringdict.get(aux_word, str.encode(aux_word))
                    aux_word = ""
            else:
                final += stringdict.get(aux_word, str.encode(aux_word))
                aux_word = ""
        if not char.isalnum():
            final += str.encode(char)
    return final

if len(sys.argv) != 3:
    print("Usage: python compressor.py -c/-d/-h input_file")
    sys.exit(1)

if sys.argv[1] == '-c':
    with open(sys.argv[2], 'r') as arquivo:
        frase = arquivo.read()
        header, word_set = create_header_and_wordset(frase)
        part_2 = list_words(word_set)
        word_dict = create_dict(word_set)
        compressed_string_list = compress_string(word_dict, frase)
        final = header + part_2 + compressed_string_list
        with open(f"{sys.argv[2]}.cmp", 'wb') as file_comprimido:
            file_comprimido.write(final)
        print(f"Texto comprimido com sucesso no arquivo {sys.argv[2]}.cmp.")

elif sys.argv[1] == '-d':
    with open(sys.argv[2], 'rb') as arquivo:
        headaer = int.from_bytes(arquivo.read(2), byteorder='big')
        d = arquivo.read()
        wordmap = []
        reference = 0
        while headaer > 0:
            reference = d.find(b',')
            wordmap.append(d[:reference])
            d = d[reference + 1:]
            headaer -= 1
        frase_lista_bytes = []
        index_num = 0
        while index_num < len(d):
            if d[index_num] == 255:
                frase_lista_bytes.append(d[index_num:index_num + 3])
            index_num += 1
        frase_dict_enderecos = {}
        index_num = 0
        while index_num < len(wordmap):
            d = d.replace(b'\xff' + (index_num).to_bytes(2, 'big'), wordmap[index_num])
            index_num += 1
        nome_de_arquivo = sys.argv[2]
        nome_de_arquivo = nome_de_arquivo[:-4]
        with open(f"{nome_de_arquivo}", 'w') as descomprimido:
            descomprimido.write(d.decode('utf-8'))
        print(f"Descomprimido no arquivo {nome_de_arquivo}")

elif sys.argv[1] == '-h':
    with open(sys.argv[2], 'rb') as arquivo:
        print(arquivo.read())
        print("Texto")
else:
    print("Usage: python compressor.py -c/-d/-h input_file")
