import numpy as np
import math
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.linear_model import Lasso, LassoCV
from sklearn.metrics import mean_squared_error, mean_absolute_error

# Function to calculate standard error
def calculate_standard_error(sample):
    std = np.std(sample, ddof=0)
    standard_error = std / math.sqrt(len(sample))
    return standard_error

# Read data
data = pd.read_table(r'D:\360MoveData\Users\HP\Desktop\prostate.data.txt')

# Define factors and response variable
factors = ['lcavol', 'lweight', 'age', 'lbph', 'svi', 'lcp', 'gleason', 'pgg45']
X = data[factors]
Y = data['lpsa']

# Split data into train and test sets
X_train, X_test, y_train, y_test = train_test_split(X, Y, test_size=0.2, random_state=1)

# Define lambdas for LassoCV
Lambdas = np.logspace(-5, 5, 200)

# Perform Lasso cross-validation
lasso_cv = LassoCV(alphas=Lambdas, normalize=True, cv=10)
lasso_cv.fit(X_train, y_train)
alpha = round(lasso_cv.alpha_, 4)

# Fit Lasso model
lasso = Lasso(alpha=lasso_cv.alpha_)
lasso.fit(X_train, y_train)

# Make predictions
y_test_predict = lasso.predict(X_test)

# Calculate evaluation metrics
mse_predict = round(mean_squared_error(y_test, y_test_predict), 4)
mae_predict = round(mean_absolute_error(y_test, y_test_predict), 4)
std_error = round(calculate_standard_error(y_test_predict), 4)

# Print results
intercept = round(lasso.intercept_, 4)
coef = [(factors[i], round(lasso.coef_[i], 4)) for i in range(len(factors))]

print('Alpha is:', alpha)
print('Intercept is:', intercept)
print('Estimated coefficients are:', coef)
print('Std Error is:', std_error)
print('MSE is:', mse_predict)
print('MAE is:', mae_predict)
