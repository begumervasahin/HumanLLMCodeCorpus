def power(a, b, P):
    """
    Calculate the power modulo P.

    Parameters:
        a (int): The base.
        b (int): The exponent.
        P (int): The modulus.

    Returns:
        int: The result of (a^b) % P.
    """
    if b == 1:
        return a
    else:
        return pow(a, b, P)
