# Global variables
reverse_map = {}
input_set = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ1234567890"

# Function to create reverse map
def create_reverse_map():
    global reverse_map
    k = 0
    for i in bit_set:
        reverse_map[i] = input_set[k]
        k = (k + 1) % 16

# Function to convert bit string to string using reverse map
def convert_bitstr_to_string(cipheredBits):
    mapBittoString = ""
    for i in range(0, 16):
        fourBitString = cipheredBits[(i * 4): ((i * 4) + 4)]
        mapBittoString += reverse_map[fourBitString]
    return mapBittoString

# Function to convert string to bit string using alphabet map
def convert_string_to_bitstring(ciphertext):
    create_alphabet_map(0)
    cipher_str = [alphabet_map[ch] for ch in ciphertext]
    cipher_bitstring = ''.join(cipher_str)
    return cipher_bitstring

# Function to get round keys
def get_round_keys(key):

    key_list.append(round4Key()[key])
    key_list.append(round3Key()[key])
    key_list.append(round2Key()[key])

    key_list = convert_to_str(key_list)

# Function to perform DES encryption round
def round(right, left, key):
    func_output = func(left, key)
    next_left = xor(func_output, right)
    next_right = left
    return (next_right, next_left)

# Function to get left part of ciphertext
def getleft(cipher):
    return cipher[32:]

# Function to get right part of ciphertext
def getright(cipher):
    return cipher[:32]

# Function to perform DES decryption
def decryption(ciphertext, key):
    cipher_bit_str = convert_string_to_bitstring(ciphertext)
    round_keys = get_round_keys(key)
    cipher = initial_permutation(list(cipher_bit_str))
    left = list(getleft(''.join(cipher)))
    right = list(getright(''.join(cipher)))
    for round_no in range(6):
        (right, left) = round(right, left, round_keys[round_no])
    plaintext = final_permutation(left + right)
    return convert_bitstr_to_string(''.join(plaintext))
