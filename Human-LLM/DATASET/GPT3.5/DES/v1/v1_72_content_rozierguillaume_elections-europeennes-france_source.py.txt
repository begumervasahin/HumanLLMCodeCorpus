import csv
import matplotlib.pyplot as plt
import numpy as np
import time
import datetime
import scipy
import scipy.interpolate
import json

colors = ['#1f77b4', '#ff7f0e', '#2ca02c', '#d62728', '#9467bd', '#8c564b', '#e377c2', '#7f7f7f', '#bcbd22', '#17becf']

nb_sieges = 74

def read_surveys(path):
    surveys = []
    surveys_dict = {}
    with open(path) as csv_file:
        csv_reader = csv.reader(csv_file, delimiter=",")
        line_count = -1
        for row in csv_reader:
            surveys = surveys + [row]
            if line_count == -1:
                parties = row[2:]
            else:
                survey_str = np.array([row[i] for i in range(2, 19)])
                survey_str[survey_str == ''] = float("nan")
                surveys_dict[line_count] = {}
                surveys_dict[line_count] = {"date": row[0], "results": survey_str.astype(float)}
            line_count += 1

    def replace_x_by_y(x, y, obj):
        for a in range(0, len(obj)):
            for i in range(0, len(obj[a])):
                if obj[a][i] == x:
                    obj[a][i] = y
        return obj

    dates = np.array([x[0] for x in surveys[1:]])
    data = np.array([np.array(d[2:]) for d in np.asarray(surveys[1:])])
    data = replace_x_by_y('', float("nan"), data)
    data = [data[i].astype(float) for i in range(0, len(data))]

    return dates, data, parties, surveys_dict

def plot_surveys(dates, data, colors, export):
    grid = plt.GridSpec(10, 10, wspace=0.4, hspace=0.3)
    fig = plt.figure(figsize=(15, 8))
    ax = fig.add_subplot(grid[:9, :9])
    fig.suptitle('Intention de vote aux élections européennes en France\n(hypothèses liste gilets jaunes et UK dans UE)', fontsize=15)
    plt.xlabel('Dates')
    plt.ylabel('Intention de vote (%)')
    dates_timestamp = [float(time.mktime(datetime.datetime.strptime(s, "%Y-%m-%d").timetuple())) for s in dates]
    for i in range(len(data[0])):
        d = np.array([data[j][i] for j in range(0, len(data))]).astype(float)
        x, y = zip(*sorted((xVal, np.mean([yVal for a, yVal in zip(dates_timestamp, d) if xVal == a])) for xVal in set(dates_timestamp)))
        interp = scipy.interpolate.interp1d(x, y)
        ax.plot(dates, d, 'o', color=colors[i])
        ax.plot(dates, [interp(x) for x in dates_timestamp], color=colors[i], label=parties[i], linewidth=1.5)
        ax.legend(bbox_to_anchor=(0.75, 0.35, 0.5, 0.5))
    plt.xticks(dates, dates, rotation=30, horizontalalignment='right')
    ax.grid(True, which='major', linestyle='-', alpha=0.6)
    plt.minorticks_on()
    ax.grid(True, which='minor', linestyle=':', alpha=0.2)
    if export:
        fig.savefig('export/sondages/' + max(dates), dpi=200)

def calculate_obtained_seats(parties, surveys_dict, nb_seats, colors, export):
    n = len(surveys_dict)
    nb_seats_remaining = [nb_seats] * n
    s = [0] * n
    id_s = [0] * n
    index = [0] * n
    for id in surveys_dict:
        for res in surveys_dict[id]['results']:
            if res >= 50:
                if nb_seats > 4:
                    s[id] = int(nb_seats / 2) + 1
                else:
                    s[id] = int(nb_seats / 2)
                nb_seats_remaining[id] = nb_seats - s[id]
                index[id] = (surveys_dict[id]['results'].tolist().index(res))
    for id in surveys_dict:
        surveys_dict[id]['results'] = surveys_dict[id]['results'][:]
        surveys_dict[id]['seats'] = (surveys_dict[id]['results'] * nb_seats_remaining[id] / 100).astype(int)
        cpt = 0
        surveys_dict_without_nan = surveys_dict.copy()
        for val in surveys_dict[id]['results']:
            if (val < 5) or not (val >= 0):
                surveys_dict_without_nan[id]['results'][cpt] = 0
                surveys_dict[id]['seats'][cpt] = 0
            cpt += 1
        remaining = nb_seats - sum(surveys_dict[id]['seats']) - s[id]
        while remaining > 0:
            calc = surveys_dict_without_nan[id]['results'] / (surveys_dict[id]['seats'] + 1)
            index_remaining_seat = np.argmax(calc)
            surveys_dict[id]['seats'][index_remaining_seat] += 1
            remaining = nb_seats - sum(surveys_dict[id]['seats']) - s[id]
        surveys_dict[id]['seats'][index[id]] += s[id]
    for id in surveys_dict:
        surveys_dict[id]['results'] = surveys_dict[id]['results'].tolist()
        surveys_dict[id]['seats'] = surveys_dict[id]['seats'].tolist()
    if export:
        with open('export/data/sondages_avec_sieges.json', 'w') as file:
            json.dump(surveys_dict, file, indent=4)
    return parties, surveys_dict

def export_seats(parties, surveys_dict):
    for id in surveys_dict:
        calculate_obtained_seats(parties, surveys_dict[surveys_dict]['results'], nb_sieges, colors, True)

def plot_seats_pie_chart(parties, obtained_seats, colors, export):
    fig2, ax2 = plt.subplots(figsize=(8, 8))
    parties2 = parties
    remove_count = 0
    for i in range(0, len(obtained_seats)):
        if obtained_seats[i - remove_count] == 0:
            obtained_seats = np.delete(obtained_seats, i - remove_count)
            parties.pop(i - remove_count)
            colors.pop(i - remove_count)
            remove_count += 1
    wedges, texts, autotexts = ax2.pie(obtained_seats, labels=parties, autopct=lambda p: '{:.0f} sièges\n({:.1f} %)'.format(sum(obtained_seats) * p / 100, p), startangle=90, colors=colors, textprops=dict(color="w", fontsize=13))
    ax2.axis('equal')
    ax2.legend(wedges, parties, loc="center right", bbox_to_anchor=(1.1, 0))
    fig2.suptitle('Projection nombre de sièges, dernier sondage\n(hypothèses liste gilets jaunes et UK dans UE)', fontsize=15)
    if export:
        fig2.savefig('export/pie_chart_sieges/' + max(dates), dpi=200)
