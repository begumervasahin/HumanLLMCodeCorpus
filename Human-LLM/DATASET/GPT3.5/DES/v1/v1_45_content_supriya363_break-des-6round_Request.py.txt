import requests
import json
import warnings

warnings.filterwarnings('ignore')

url = "https://172.27.26.163:9999/des"
headers = {
    'Content-type': 'application/json',
    'Orgin': 'https://172.27.26.163:9999',
    'Referer': 'https://172.27.26.163:9999/game/caves.swf'
}

# Open input file for reading plaintext passwords
with open('input.txt', 'r') as f:
    plaintexts = f.readlines()

# Open output file for writing ciphertext passwords
with open('response.txt', 'w') as f2:
    data = {"password": "555acfd4d3c1b7a398d71e4c1538ad4d", "teamname": "Amalgam"}
    
    for plaintext in plaintexts:
        # Update plaintext in data dictionary
        data["plaintext"] = plaintext.split(',')[0].strip()
        
        # Make POST request to the server
        r = requests.post(url, json=data, headers=headers, verify=False)
        
        # Check if the request was successful (status code 200)
        if r.status_code == 200:
            response = json.loads(r.text)
            # Write ciphertext to output file
            f2.write(f"{response['ciphertext']}\n")
        else:
            print("Failed to retrieve ciphertext")

print("Task completed.")
