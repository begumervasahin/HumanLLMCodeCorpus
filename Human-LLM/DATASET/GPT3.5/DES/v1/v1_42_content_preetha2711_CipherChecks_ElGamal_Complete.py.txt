def miller_rabin(number):
    """
    Miller-Rabin primality test.
    """
    B = False
    num_1 = number - 1
    k = 0
    while num_1 % 2 == 1:
        num_1 = num_1 // 2
        k += 1
    m = num_1 // (2 ** k)
    a = random.randint(0, number - 1)
    b = pow(a, m, number)
    if b == 1:
        B = True
    else:
        for i in range(k):
            if pow(b, 2, number) == -1 % number:
                B = True
            else:
                b = pow(b, 2, number)
    return B

def generate_prime(length=300):
    """
    Generate a prime number using Miller-Rabin primality test.
    """
    flag = False
    while flag != True:
        random_num = random.getrandbits(length)
        random_num |= (1 << length - 1) | 1
        flag = miller_rabin(random_num)
        prime_number = random_num
        return prime_number

def get_two_primes():
    """
    Generate two prime numbers p and q such that q = 2p + 1.
    """
    flag = False
    while flag != True:
        p = generate_prime()
        q = 2 * p + 1
        flag = miller_rabin(q)
    return p, q

def lucas_test(integer):
    """
    Perform Lucas primality test.
    """
    p = integer
    Q = []
    for i in range(2, p):
        if (p - 1) % i == 0:
            Q.append(i)
    prim_roots = []
    for g in range(2, p):
        flag = True
        for q in Q:
            if (g ** int((p - 1) / q)) % p == 1:
                flag = False
                break
        if flag == True:
            prim_roots.append(g)
    return prim_roots
