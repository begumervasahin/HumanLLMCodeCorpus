import os
import getpass
import subprocess

def install_library(library_name):
    try:
        subprocess.check_call(["pip", "install", library_name])
        print(f"{library_name} installed successfully!")
    except subprocess.CalledProcessError:
        print(f"Failed to install {library_name}")

response = getpass.getpass("Please enter a password: ")
file_input = input("Enter a file to encrypt. (If blank, uses provided input.txt): ")
if file_input == "":
    file_input = 'input.txt'

print("Checking for Python bitstring library...")
install_library("bitstring")

print("Checking for Python Cryptodome library...")
install_library("pycryptodomex")

print('Libraries installed, proceeding...')
print("Generating keyfile with 3Des keys at keyfile.txt...")
subprocess.run(['python', '3Des.py', 'genkey', response, 'keyfile.txt'])

print("Proceeding with encryption:")
print("Encrypting input with 3DES in ECB Mode and outputting to cipher_ecb.txt...")
subprocess.run(['python', '3Des.py', 'encrypt', file_input, 'keyfile.txt', 'cipher_ecb.txt', 'ECB'])

print("Encrypting input with 3DES in CBC Mode and outputting to cipher_cbc.txt...")
subprocess.run(['python', '3Des.py', 'encrypt', file_input, 'keyfile.txt', 'cipher_cbc.txt', 'CBC'])

print("Encrypting input with 3DES in OFB Mode and outputting to cipher_ofb.txt...")
subprocess.run(['python', '3Des.py', 'encrypt', file_input, 'keyfile.txt', 'cipher_ofb.txt', 'OFB'])

print("Encryption complete, proceeding to decryption:")
print("Decrypting cipher_ecb.txt and outputting to clear_ecb.txt...")
subprocess.run(['python', '3Des.py', 'decrypt', 'cipher_ecb.txt', 'keyfile.txt', 'clear_ecb.txt', 'ECB'])

print("Decrypting cipher_cbc.txt and outputting to clear_cbc.txt...")
subprocess.run(['python', '3Des.py', 'decrypt', 'cipher_cbc.txt', 'keyfile.txt', 'clear_cbc.txt', 'CBC'])

print("Decrypting cipher_ofb.txt and outputting to clear_ofb.txt...")
subprocess.run(['python', '3Des.py', 'decrypt', 'cipher_ofb.txt', 'keyfile.txt', 'clear_ofb.txt', 'OFB'])

print("Done!")
