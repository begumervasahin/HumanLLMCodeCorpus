# Global variables for Rabin Signature
nrabin = 0

# Function to update Rabin signature parameters
def update_spz():
    global aSPZ, iSPZ, jSPZ, wSPZ, sSPZ
    iSPZ = (iSPZ + wSPZ) % 256
    jSPZ = sSPZ[(jSPZ + sSPZ[iSPZ]) % 256]
    sSPZ[iSPZ], sSPZ[jSPZ] = sSPZ[jSPZ], sSPZ[iSPZ]

# Function to output Rabin signature
def output_spz():
    global aSPZ, iSPZ, jSPZ, wSPZ, sSPZ
    update_spz()
    return sSPZ[jSPZ]

# Function to shuffle Rabin signature parameters
def shuffle_spz():
    global aSPZ, iSPZ, jSPZ, wSPZ, sSPZ
    for v in range(256):
        update_spz()
    wSPZ = (wSPZ + 2) % 256
    aSPZ = 0

# Function to absorb a nibble for Rabin signature
def absorb_nibble_spz(x):
    global aSPZ, iSPZ, jSPZ, wSPZ, sSPZ
    if aSPZ == 240:
        shuffle_spz()
    sSPZ[aSPZ], sSPZ[240 + x] = sSPZ[240 + x], sSPZ[aSPZ]
    aSPZ += 1

# Function to absorb a byte for Rabin signature
def absorb_byte_spz(b):
    absorb_nibble_spz(b % 16)
    absorb_nibble_spz(b // 16)

# Function to squeeze Rabin signature output
def squeeze_spz(out, outlen):
    global aSPZ, iSPZ, jSPZ, wSPZ, sSPZ
    if aSPZ != 0:
        shuffle_spz()
    for v in range(outlen):
        out.append(output_spz())

# Function to compute hash function for Rabin signature
def h(x):
    global aSPZ, iSPZ, jSPZ, wSPZ, sSPZ, nrabin
    jSPZ = iSPZ = aSPZ = 0
    wSPZ = 1
    sSPZ = list(range(256))
    for c in x:
        absorb_byte_spz(ord(c))
    res = []
    squeeze_spz(res, 128)
    out = 0
    for bx in res:
        out = (out << 8) + bx
    return out % nrabin

# Function to convert code to number
def code2num(x):
    res = 0
    for c in x:
        if '0' <= c <= '9':
            res = (res << 6) + ord(c) - 48
        elif 'A' <= c <= 'Z':
            res = (res << 6) + ord(c) - 55
        elif 'a' <= c <= 'z':
            res = (res << 6) + ord(c) - 61
        elif c == '.':
            res = (res << 6) + 62
        elif c == '/':
            res = (res << 6) + 63
    return res

# Function to convert number to code
def num2code(x):
    res = ''
    while x > 0:
        y = x % 64
        if y < 10:
            res = chr(y + 48) + res
        elif y < 36:
            res = chr(y + 55) + res
        elif y < 62:
            res = chr(y + 61) + res
        elif y == 62:
            res = '.' + res
        elif y == 63:
            res = '/' + res
        x //= 64
    return res

# Function to find greatest common divisor
def gcd(a, b):
    while b > 0:
        a, b = b, a % b
    return a

# Function to find the next prime number
def next_prime(p):
    while p % 4 != 3:
        p += 1
    return next_prime_(p)

def next_prime_(p):
    m_ = 3 * 5 * 7 * 11 * 13 * 17 * 19 * 23 * 29
    while gcd(p, m_) != 1:
        p += 4
    if (pow(2, p-1, p) != 1):
        return next_prime_(p + 4)
    if (pow(3, p-1, p) != 1):
        return next_prime_(p + 4)
    if (pow(5, p-1, p) != 1):
        return next_prime_(p + 4)
    if (pow(17, p-1, p) != 1):
        return next_prime_(p + 4)
    return p

# Function to compute Rabin Signature
def f_ab(p, q):
    res = [3, 3]
    while pow(res[0], (p-1)//2, p) == 1 or pow(res[0], (q-1)//2, q) != 1:
        res[0] += 1
    while pow(res[1], (p-1)//2, p) != 1 or pow(res[1], (q-1)//2, q) == 1:
        res[1] += 1
    return res

# Function to compute square root mod pq
def root(m, p, q):
    global nrabin
    x = h(m)
    ab = f_ab(p, q)
    a = ab[0]
    b = ab[1]
    nrabin = p * q
    if pow(x, (p-1)//2, p) > 1:
        x *= a
    if pow(x, (q-1)//2, q) > 1:
        x *= b
    if pow(x, (q-1)//2, q) != 1 or pow(x, (p-1)//2, p) != 1:
        print('ERROR')
        return -1
    return (pow(p, q-2, q) * p * pow(x, (q+1)//4, q) + pow(q, p-2, p) * q * pow(x, (p+1)//4, p)) % nrabin

# Function to write number to file
def write_number(number, fnam):
    with open(fnam, 'wb') as f:
        n = number
        while n > 0:
            byte = n % 256
            n //= 256
            f.write(chr(byte))

# Function to read number from file
def read_number(fnam):
    with open(fnam, 'rb') as f:
        num = 0
        snum = f.read()
        for i in range(len(snum)):
            num = (num << 8) ^ ord(snum[len(snum)-i-1])
    return num

# Function to compute hash of a file
def hF(fnam):
    with open(fnam, 'r') as f:
        return h(f.read())
