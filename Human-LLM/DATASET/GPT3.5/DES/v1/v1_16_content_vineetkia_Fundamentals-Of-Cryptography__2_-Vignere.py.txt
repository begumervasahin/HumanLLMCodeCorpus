def encrypt(plain_text, key):
    """
    Encrypts a plain text using the Vigenère cipher with the specified key.

    Parameters:
        plain_text (str): The text to be encrypted.
        key (str): The key for encryption.

    Returns:
        str: The encrypted text.
    """
    result = ''

    for i in range(len(plain_text)):
        j = i % len(key)
        if plain_text[i].isupper():
            result += chr(((ord(plain_text[i]) + ord(key[j]) - 65 * 2) % 26) + 65)
        else:
            result += chr(((ord(plain_text[i]) + ord(key[j]) - 97 * 2) % 26) + 97)

    return result
