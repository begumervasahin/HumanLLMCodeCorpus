import gplaylist as gp
import os

# Help message
help_message = '''
show : to see the list of your favorite artists..
add : to add a name to your favorite artist list
rem : to remove a name from your favorite artist list
quit : to quit
clr : to clear screen
help : for this menu
'''
print(help_message)

# Function to clear the screen
def clear_screen():
    if name == 'nt':
        _ = os.system('cls')
    elif name == 'posix':
        _ = os.system('clear')
    else:
        print('Sorry, Clear Screen is not supported for your OS:', name)

# Main loop
while True:
    inp = input('..> ').lower()

    if inp == 'show':
        # Show favorite artists
        lst = gp.GetPlayList().getNames()
        for name in lst:
            print(name)

    elif inp == 'add':
        # Add a favorite artist
        name = input('Name of Artist to add: ')
        ob = gp.AddPlayList(name)
        if ob.hasErr():
            print(ob.getError())
        else:
            print('Successfully added the artist')

    elif inp == 'rem':
        # Remove a favorite artist
        name = input('Name of Artist to remove: ')
        ob = gp.RemoveArtist(name)
        if ob.hasErr():
            print(ob.getError())
        elif ob.found:
            print('Successfully removed the artist')
        else:
            print('Artist not found in the List')

    elif inp == 'quit':
        # Quit the program
        break

    elif inp == 'clr':
        # Clear the screen
        clear_screen()

    elif inp == 'help':
        # Show help message
        print(help_message)
