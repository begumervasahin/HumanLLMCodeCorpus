import requests
import zipfile
import os
import pickle
import json
import sqlite3

def get_manifest():
    manifest_url = 'https://www.bungie.net/Platform/Destiny2/Manifest/'
    r = requests.get(manifest_url)
    manifest = r.json()
    mani_url = 'https://www.bungie.net' + \
        manifest['Response']['mobileWorldContentPaths']['en']
    r = requests.get(mani_url)
    with open("MANZIP", "wb") as zip_file:
        zip_file.write(r.content)
    print("Download Complete!")
    with zipfile.ZipFile('MANZIP') as zip_file:
        name = zip_file.namelist()
        zip_file.extractall()
    os.rename(name[0], 'Manifest.content')
    print('Unzipped!')

def build_dict(hash_dict):
    con = sqlite3.connect('Manifest.content')
    print('Connected')
    cur = con.cursor()
    all_data = {}
    for table_name in hash_dict.keys():
        cur.execute('SELECT json from '+table_name)
        print('Generating '+table_name+' dictionary....')
        items = cur.fetchall()
        item_jsons = [json.loads(item[0]) for item in items]
        item_dict = {}
        hash_key = hash_dict[table_name]
        for item in item_jsons:
            item_dict[item[hash_key]] = item
        all_data[table_name] = item_dict
    print('Dictionary Generated!')
    return all_data

def init_manifest():
    if not os.path.isfile(r'Manifest.content'):
        get_manifest()
        manifest = build_dict(hashes)
        with open('manifest.pickle', 'wb') as data_file:
            pickle.dump(manifest, data_file)
            print("'manifest.pickle' created!\nDONE!")
    else:
        print('Pickle Exists')
    with open('manifest.pickle', 'rb') as data_file:
        manifest = pickle.load(data_file)
    return manifest

hashes = {
    'DestinyClassDefinition': 'hash',
    'DestinyInventoryItemDefinition': 'hash',
    'DestinyTalentGridDefinition': 'hash',
    'DestinyHistoricalStatsDefinition': 'statId',
    'DestinyStatDefinition': 'hash',
    'DestinySandboxPerkDefinition': 'hash',
    'DestinyStatGroupDefinition': 'hash',
    'DestinyInventoryBucketDefinition': 'hash'
}

# Initialize the manifest
manifest_data = init_manifest()
