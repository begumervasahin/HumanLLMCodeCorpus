import requests
import click
import textwrap

# Function to print wrapped text with optional indentation
def wrap_print(content, indent=0):
    global width
    print(
        textwrap.indent(
            textwrap.fill(content, (width if width else 80) - indent), " " * indent
        )
    )

@click.command()
def main():
    global width
    width = click.get_terminal_size()[0]
    while True:
        click.clear()
        print("Welcome to ParteiDuell \033[31mK\033[0m\033[33ml\033[0m\033[93m\033[32mi\033[0m\033[34me\033[0m\033[35mh\033[0m")
        print("")
        r = requests.get("https://solver.cloud:440/list")
        json_data = r.json()
        wrap_print(json_data[0]["these"])
        print()
        wrap_print(json_data[0]["statement"])
        print()
        keys = list(json_data[0]["possibleAnswers"])
        print(", ".join(f"{i}: {answer}" for i, answer in enumerate(keys)))
        party = input("> ")
        try:
            party = int(party)
            party = keys[party]
        except ValueError as error:
            print(error)
        print()
        print()
        if json_data[0]["answer"] == party:
            print("Correct!")
        else:
            print(f"Wrong, this statement was from {json_data[0]['answer']}")
            print()
            print(f"The party {party} made the following statement:\n")
            wrap_print(json_data[0]["possibleAnswers"][party], indent=4)
        print()
        click.pause()

if __name__ == "__main__":
    main()
