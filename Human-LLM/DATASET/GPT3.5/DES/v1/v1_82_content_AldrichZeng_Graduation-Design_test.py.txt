import torch
import torch.nn as nn
import torch.optim as optim
import torchvision.transforms as transforms
import torchvision.datasets as datasets
import resnet
import vgg
import os
from datetime import datetime
from prune import select_and_prune_filter
import numpy as np
import config as conf

# Define device
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

# Define transforms
transform = transforms.Compose([
    transforms.ToTensor(),
    transforms.Normalize((0.5, 0.5, 0.5), (0.5, 0.5, 0.5))
])

# Load validation dataset
validation_dataset = datasets.CIFAR10(root='./data', train=False, download=True, transform=transform)
validation_loader = torch.utils.data.DataLoader(validation_dataset, batch_size=conf.batch_size, shuffle=False)

# Define the neural network (replace with your model)
net = resnet.ResNet50()  # Example, replace with your model
net = net.to(device)

# Load checkpoint if available
if os.path.exists(conf.checkpoint_path):
    net.load_state_dict(torch.load(conf.checkpoint_path))
    print("Checkpoint loaded")

print("{} Start validation".format(datetime.now()))
print("{} global step = {}".format(datetime.now(), conf.global_step))

# Set network to evaluation mode
net.eval()

# Perform validation
with torch.no_grad():
    correct = 0
    total = 0
    for val_data in validation_loader:
        images, labels = val_data
        images, labels = images.to(device), labels.to(device)
        outputs = net(images)
        _, predicted = torch.max(outputs.data, 1)
        total += labels.size(0)
        correct += (predicted == labels).sum().item()

    accuracy = correct / total
    print("{} Validation Accuracy = {:.4f}".format(datetime.now(), accuracy))

    # Save model if accuracy is higher
    if accuracy > conf.highest_accuracy:
        conf.highest_accuracy = accuracy
        print("{} Saving model...".format(datetime.now()))
        torch.save(net.state_dict(), conf.checkpoint_path)
        print("{} Model saved ".format(datetime.now()))
        with open(conf.highest_accuracy_path, 'w') as f:
            f.write(str(conf.highest_accuracy))
        with open(conf.global_step_path, 'w') as f:
            f.write(str(conf.global_step))
        print("{} model saved at global step = {}".format(datetime.now(), conf.global_step))
