class Cook:
    def prepareDish(self):
        cutter = Cutter()
        cutter.cutVegetables()
        boiler = Boiler()
        boiler.boilVegetables()
        frier = Frier()
        frier.fry()

class Cutter:
    def cutVegetables(self):
        print("All vegetables are cut")

class Boiler:
    def boilVegetables(self):
        print("All vegetables are boiled")

class Frier:
    def fry(self):
        print("All vegetables are mixed and fried.")

if __name__ == "__main__":
    cook = Cook()
    cook.prepareDish()
