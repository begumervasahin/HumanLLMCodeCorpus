# Import required modules
import misc
import permutations

# Define the function to generate subkeys from a primary key
def primary_key_to_subkeys(key):
    subkeys = []
    
    # Convert hexadecimal key to binary
    binary_prim_key = misc.hex_to_bits(key)
    
    # Group binary key into 7-bit blocks
    binary_prim_key = ''.join([binary_prim_key[i:i + 7] for i in range(0, len(binary_prim_key), 8)])
    
    # Split binary key into left and right parts
    left_part, right_part = binary_prim_key[:int(len(binary_prim_key) / 2)], binary_prim_key[int(len(binary_prim_key) / 2):]
    
    # Generate subkeys
    for i in range(16):
        if i == 0 or i == 1 or i == 8 or i == 15:
            left_part = misc.left_shift(left_part, 1)
            right_part = misc.left_shift(right_part, 1)
        else:
            left_part = misc.left_shift(left_part, 2)
            right_part = misc.left_shift(right_part, 2)
        
        # Combine left and right parts and apply permutation
        subkeys.append(permutations.permutation(left_part + right_part, "compression"))
    
    return subkeys

# Test the function
if __name__ == "__main__":
    primary_key = "0123456789ABCDEF"  # Example primary key
    subkeys = primary_key_to_subkeys(primary_key)
    print("Subkeys:")
    for i, subkey in enumerate(subkeys, 1):
        print(f"Subkey {i}: {subkey}")
