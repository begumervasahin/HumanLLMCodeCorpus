import random

class AiModule:
    def __init__(self):
        self.bo = []
        self.bp = []
        self.pulse = []

    def input_check(self, bloodOxygen, bloodPressure, pulse):
        """
        Check the input type from the database in case there are some mistaken data.
        """
        for i, j, k in zip(bloodOxygen, bloodPressure, pulse):
            if (type(i) != float) or (type(j) != float) or (type(k) != float):
                raise TypeError("Input data must be of type float")
            else:
                self.bo.append(i)
                self.bp.append(j)
                self.pulse.append(k)

    def predict(self):
        """
        Predict blood oxygen, blood pressure, pulse for each type from the database.
        """
        if not self.bo or not self.bp or not self.pulse:
            raise ValueError("No data available for prediction")
        
        rand = random.randint(1, len(self.bo))
        predBloodOxygen = sum(self.bo[:rand]) / rand
        predBloodPressure = sum(self.bp[:rand]) / rand
        predPulse = sum(self.pulse[:rand]) / rand

        print('Predicted blood oxygen:', predBloodOxygen)
        print('Predicted blood pressure:', predBloodPressure)
        print('Predicted pulse:', predPulse)

        return predBloodOxygen, predBloodPressure, predPulse

# Example usage
ai = AiModule()
blood_oxygen_data = [98.5, 99.2, 97.8, 100.0]
blood_pressure_data = [120.5, 122.1, 118.9, 121.6]
pulse_data = [75.2, 76.5, 74.8, 77.3]

try:
    ai.input_check(blood_oxygen_data, blood_pressure_data, pulse_data)
    ai.predict()
except (TypeError, ValueError) as e:
    print(f"Error: {e}")
