import json

# Mock function for displaying error messages
def show_error(message):
    print("Error:", message)

settings_file = "settings.ini"
defaults = {
    "graphviz_file_type": "pdf",
    "graphviz_auto_vis": True
}
settings = None

def initialize():
    """Initializes the program's settings"""
    global settings
    settings = defaults
    try:
        try:
            with open(settings_file, 'r') as f:
                settings = json.load(f)
        except FileNotFoundError:
            with open(settings_file, 'w') as f:
                settings = defaults
                json.dump(settings, f)
    except Exception:
        show_error("Error reading/writing settings file.\n" +
                   "Reverting to defaults...")
    for k, v in defaults.items():
        if k not in settings:
            settings[k] = v

def update(setting, value):
    """Updates the setting with a new value"""
    settings[setting] = value
    try:
        with open(settings_file, 'w') as f:
            json.dump(settings, f)
    except FileNotFoundError:
        show_error("Error writing settings file.\n" +
                   "Changes are only saved for this session.")

# Test the functionality
initialize()
print("Initial settings:", settings)

# Update a setting
update("graphviz_auto_vis", False)
print("Updated settings:", settings)
