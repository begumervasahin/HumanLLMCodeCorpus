import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# Set figure size for plots
plt.rcParams['figure.figsize'] = (5.0, 5.0)

# Read data from CSV file
data = pd.read_csv('data.csv')

# Extract X and Y values
X = data.iloc[:, 0]
Y = data.iloc[:, 1]

# Calculate mean of X and Y
xi = np.mean(X)
yi = np.mean(Y)

# Calculate deviations from mean
x_xi = X - xi
y_yi = Y - yi

# Calculate squared deviations
x_xi2 = x_xi * x_xi
y_yi2 = y_yi * y_yi

# Calculate slope (m) and intercept (c) using the method of least squares
m = np.sum(x_xi * y_yi) / np.sum(x_xi2)
c = yi - m * xi

# Plot original data and regression line
plt.scatter(X, Y)
plt.plot(X, m*X + c, color='red')
plt.title('Linear Regression')
plt.xlabel('X')
plt.ylabel('Y')
plt.show()

# Initialize parameters for gradient descent
m = 0
c = 0
learning_rate = 0.0001
epochs = 1000
n = float(len(X))

# Perform gradient descent
for i in range(epochs):
    Y_pred = m * X + c
    D_m = (-2/n) * sum(X * (Y - Y_pred))
    D_c = (-2/n) * sum(Y - Y_pred)
    m = m - learning_rate * D_m
    c = c - learning_rate * D_c

# Plot original data and regression line obtained from gradient descent
plt.scatter(X, Y)
plt.plot([min(X), max(X)], [min(m*X + c), max(m*X + c)], color='red')
plt.title('Linear Regression using Gradient Descent')
plt.xlabel('X')
plt.ylabel('Y')
plt.show()

# Print slope (m) and intercept (c) obtained from gradient descent
print("Slope (m):", m)
print("Intercept (c):", c)
