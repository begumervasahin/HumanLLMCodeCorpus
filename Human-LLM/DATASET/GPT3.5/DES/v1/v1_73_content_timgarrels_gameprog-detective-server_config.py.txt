import os
import sys
import requests

class Config():
    """Config Object for Flask App"""
    def __init__(self):
        basedir = os.path.abspath(os.path.dirname(__file__))
        self.SQLALCHEMY_DATABASE_URI = os.environ.get('DATABASE_URL') or \
            'sqlite:///' + os.path.join(basedir, 'app/app.db')
        self.SQLALCHEMY_TRACK_MODIFICATIONS = False
        self.BOT_TOKEN_ENV_VAR = "GP_TELEGRAM_BOT_TOKEN"
        try:
            self.BOT_TOKEN = os.environ[self.BOT_TOKEN_ENV_VAR]
        except KeyError:
            print("You did not set {}".format(self.BOT_TOKEN_ENV_VAR))
            sys.exit()
        self.FIREBASE_URL_ENV_VAR = "GP_FIREBASE_URL"
        try:
            self.FIREBASE_URL = os.environ[self.FIREBASE_URL_ENV_VAR]
        except KeyError:
            self.FIREBASE_URL = None
            if not self.FIREBASE_URL:
                print("You did not set {}".format(self.FIREBASE_URL_ENV_VAR), file=sys.stderr)       
                print("Communication with the app will not work!", file=sys.stderr)
        try:
            resp = requests.get("http://api.telegram.org/bot{token}/getMe".format(token=self.BOT_TOKEN))
            self.BOT_NAME = resp.json()["result"]["username"]
        except requests.exceptions.ConnectionError:
            print("No connection, bot name will be set to default <AndyAbbot>")
            self.BOT_NAME = "AndyAbbot"
        except KeyError:
            sys.exit("Bot Name could not be parsed from telegram api. Maybe the bot token is out of date?")
        self.IMAGE_UPLOAD_FOLDER = os.path.join(basedir, "app/image_upload")
        self.ALLOWED_IMAGE_EXTENSIONS = {"jpg", "jpeg", "png"}
        self.STORY_FILE = os.path.join(basedir, "app/story/story.json")

# Example usage:
config = Config()
print(config.SQLALCHEMY_DATABASE_URI)
print(config.BOT_TOKEN)
print(config.FIREBASE_URL)
print(config.BOT_NAME)
print(config.IMAGE_UPLOAD_FOLDER)
print(config.ALLOWED_IMAGE_EXTENSIONS)
print(config.STORY_FILE)
