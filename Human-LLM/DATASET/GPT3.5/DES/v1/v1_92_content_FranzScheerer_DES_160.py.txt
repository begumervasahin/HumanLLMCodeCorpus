def readNumber(fnam):
    with open(fnam, 'rb') as f:
        n = 0
        snum = f.read()
        for i in range(len(snum)):
            n = (n << 8) ^ snum[len(snum) - i - 1]
    return n

def hextxt2num(x):
    res = 0
    for c in x:
        if 48 <= ord(c) < 58:
            res = (res << 4) + ord(c) - 48
        elif 97 <= ord(c) < 102:
            res = (res << 4) + ord(c) - 87
        elif 65 <= ord(c) < 70:
            res = (res << 4) + ord(c) - 55
    return res

def inv(b, m):
    s = 0
    t = 1
    a = m
    while b != 1:
        q = a // b
        aa = b
        b = a % b
        a = aa
        ss = t
        t = s - q * t
        s = ss
    if t < 0:
        t = t + m
    return t

def h(x):
    dx1 = hashlib.sha256(x.encode()).digest()
    res = 0
    for cx in dx1:
        res = (res << 8) ^ cx
    return res % n4

def genP(x, a, b):
    while pow(x ** 3 + a * x + b, (prime - 1) // 2, prime) != 1:
        x = x + 1
    y = pow(x ** 3 + a * x + b, (prime + 1) // 4, prime)
    return [x % prime, y % prime]

def addP(P, Q):
    x1, x2 = P[0], Q[0]
    y1, y2 = P[1], Q[1]
    while x1 < x2:
        x1 = x1 + prime
    if x1 == x2:
        s = ((3 * (x1 ** 2) + a) * inv(2 * y1, prime)) % prime
    else:
        s = ((y1 - y2) * inv(x1 - x2, prime)) % prime
    xr = (s ** 2 - x1 - x2) % prime
    yr = (s * (x1 - xr) - y1) % prime
    return [xr, yr]

def mulP(P, n):
    isFirst = True
    resP = P[:]
    if n < 0:
        resP[1] = prime - resP[1]
        n = (-1) * n
    PP = resP
    while n > 0:
        if n % 2 != 0:
            if isFirst:
                resP = PP
                isFirst = False
            else:
                resP = addP(resP, PP)
        PP = addP(PP, PP)
        n = n // 2
    return resP

def verify(G, s, Y, e, m):
    return e == h(str(addP(mulP(G, s), mulP(Y, e))[0]) + m)

def ecdsa_v(G, m, S, Y):
    si = inv(S[0], n4)
    hh = h(m + str(S[1]))
    u1 = (si * hh) % n4
    u2 = (si * S[1]) % n4
    return addP(mulP(G, u1), mulP(Y, u2))[0] == S[1]

def ecdsa(G, m, x):
    k = h(m + 'ecdsa')
    R = mulP(G, k)
    hh = h(m + str(R[0]))
    s = (inv(k, n4) * (hh + R[0] * x)) % n4
    return [s, R[0]]

def writeNumber(number, fnam):
    with open(fnam, 'wb') as f:
        n = number
        while n > 0:
            byte = n % 256
            n = n // 256
            f.write(chr(byte).encode())

def signSchnorr(G, m, x):
    k = h(m + 'kk2')
    R = mulP(G, k)
    e = h(str(R[0]) + m)
    return [(k - x * e) % n4, e]
