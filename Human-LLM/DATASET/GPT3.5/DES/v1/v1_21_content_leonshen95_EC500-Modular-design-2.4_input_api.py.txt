class InputAPI:
    def __init__(self, user_id, age, gender, heartrate, systolic_bp, diastolic_bp, blood_oxygen, temperature, time):
        self.user_id = user_id
        self.age = age
        self.gender = gender
        self.heartrate = heartrate
        self.systolic_bp = systolic_bp
        self.diastolic_bp = diastolic_bp
        self.blood_oxygen = blood_oxygen
        self.temperature = temperature
        self.time = time
        self.data_dict = {
            "gender": gender,
            "heartrate": heartrate,
            "diastolic_bp": diastolic_bp,
            "systolic_bp": systolic_bp,
            "blood_oxygen": blood_oxygen,
            "temperature": temperature,
            "time": time
        }

    @staticmethod
    def filter_data(data):
        wrong_flag = -1
        noise = 500
        if data > noise:
            data = wrong_flag
        return data

    def implement_filter(self):
        for key in self.data_dict:
            if key not in ["user_id", "age", "gender", "time"]:
                self.data_dict[key] = self.filter_data(self.data_dict[key])

    def return_request(self, wire):
        alert = 1
        data_db = 2
        if wire == alert:
            user_data_dict = {
                "heartrate": self.heartrate,
                "diastolic_bp": self.diastolic_bp,
                "systolic_bp": self.systolic_bp,
                "blood_oxygen": self.blood_oxygen,
                "temperature": self.temperature,
                "time": self.time
            }
            return user_data_dict
        elif wire == data_db:
            return self.user_id, self.data_dict


# Example usage of the InputAPI class
user_input = InputAPI(
    user_id="12345",
    age=30,
    gender="Male",
    heartrate=120,
    systolic_bp=120,
    diastolic_bp=80,
    blood_oxygen=98,
    temperature=37.5,
    time="2024-04-05 14:30:00"
)

user_input.implement_filter()
print(user_input.return_request(1))  # Returns user data for alert
print(user_input.return_request(2))  # Returns user ID and filtered data dictionary for database
