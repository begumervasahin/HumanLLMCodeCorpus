def read_binary_file(filename_path):
    """
    Reads a binary file and returns its content as a binary string.

    Parameters:
    - filename_path: str, path to the binary file

    Returns:
    - string_binary: str, binary representation of the file content
    """
    string_binary = ''
    with open(filename_path, mode='rb') as file:
        file_bytes = file.read()
    for i in range(len(file_bytes)):
        tmp = file_bytes[i]
        uj = bin(tmp)
        if len(uj) > 8:
            uj = uj[2:]
        zero_repeat_count = 8 - len(uj)
        for j in range(zero_repeat_count):
            string_binary += '0'
        string_binary += uj
    return string_binary.replace('b', '0')

def write_binary_file(binary, filename_path):
    """
    Writes a binary string to a file.

    Parameters:
    - binary: str, binary string to write to the file
    - filename_path: str, path to the output file
    """
    byts = bytearray()
    for i in range(0, len(binary), 8):
        bin_number = binary[i:i+8]
        byts.append(int(bin_number, 2))
    with open(filename_path, mode='wb') as file:
        file.write(byts)

def hex_key_to_binary(hex_key):
    """
    Converts a hexadecimal key to its binary representation.

    Parameters:
    - hex_key: str, hexadecimal key

    Returns:
    - binary_key: str, binary representation of the hexadecimal key
    """
    binary_key = ""
    for char in hex_key:
        binary_char = "{:04b}".format(int(char, 16))
        binary_key += binary_char
    return binary_key

# Example usage
if __name__ == "__main__":
    # Example file paths
    input_file = "input.bin"
    output_file = "output.bin"
    
    # Read binary file
    binary_content = read_binary_file(input_file)
    print("Binary content read from file:", binary_content)

    # Write binary content to a file
    write_binary_file(binary_content, output_file)
    print("Binary content written to file:", output_file)

    # Example hexadecimal key
    hex_key = "1A3F"
    binary_key = hex_key_to_binary(hex_key)
    print("Binary representation of hexadecimal key {}: {}".format(hex_key, binary_key))
