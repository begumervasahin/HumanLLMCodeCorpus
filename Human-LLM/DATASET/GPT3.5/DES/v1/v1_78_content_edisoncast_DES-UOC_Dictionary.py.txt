import crypt

def generate_hash(filename, outputfilename, salt):
    with open(outputfilename, "w") as out:
        with open(filename, 'r') as f:
            for line in f:
                hashed = crypt.crypt(line.strip(), salt)
                result = f"{hashed} {line}"
                print(result)
                out.write(result + "\n")

def des_dictionary(filename, outputfilename):
    input_text(filename, outputfilename)

def input_text(filename, outputfilename):
    with open(outputfilename, "w") as out:
        with open(filename, 'r') as f:
            for line in f:
                if len(line.strip()) == 8:
                    print(line)
                    out.write(line)
