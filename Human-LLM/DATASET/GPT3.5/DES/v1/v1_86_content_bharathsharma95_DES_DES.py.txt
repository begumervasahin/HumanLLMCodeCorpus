import time
import readFile
import encrypt
import decrypt
import os

def write_bytes_to_file(data, filename):
    with open(filename, 'w+b') as file:
        byte_data = bytearray(data)
        file.write(byte_data)

if __name__ == "__main__":
    start_time = str(format(time.ctime()))
    print("Script Started at:", start_time)
    print("")

    key_bytes = readFile.main("key.txt")
    print("key_bytes data:", key_bytes)
    print("")

    file_size_in_bytes = os.path.getsize("input.txt")
    print("File size in bytes:", file_size_in_bytes)

    input_data_bytes = readFile.main("input.txt")
    print("Input data bytes:", input_data_bytes)
    print("")

    cipher = encrypt.DESencryption(input_data_bytes, key_bytes)
    print("Cipher:", cipher)
    write_bytes_to_file(cipher, "cipher.txt")

    cipher_bytes = readFile.main("cipher.txt")
    decypher = decrypt.DESdecryption(cipher_bytes, key_bytes)
    write_bytes_to_file(decypher, "decypher.txt")
