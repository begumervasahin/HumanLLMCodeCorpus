from bs4 import BeautifulSoup
import requests
import re
import os
import pymongo

class FilmList:
    def __init__(self, year):
        self.year = year

    def download_html(self):
        for month in range(1, 13):
            url = f'https://www.imdb.com/movies-coming-soon/{self.year:04d}-{month:02d}'
            r = requests.get(url)
            if r.status_code != 200:
                continue
            soup = BeautifulSoup(r.content, 'html.parser')
            file_path = f'data/film_lists/{self.year:04d}-{month:02d}.html'
            with open(file_path, 'w') as f:
                f.write(str(soup))

    def get_films_ids(self):
        film_ids = []
        for month in range(1, 13):
            file_path = f'data/film_lists/{self.year:04d}-{month:02d}.html'
            if not os.path.exists(file_path):
                continue
            with open(file_path, 'r') as f:
                soup = BeautifulSoup(f, 'html.parser')
                for elem in soup.find_all('td', attrs={"class": "overview-top"}):
                    href = elem.h4.a.get('href')
                    id = href.split('/')[2]
                    film_ids.append(id)
        return film_ids

class Film:
    def __init__(self, id):
        self.url = 'https://www.imdb.com/title/' + id
        self.id = id

    def download_html(self):
        r = requests.get(self.url)
        if r.status_code != 200:
            return
        soup = BeautifulSoup(r.content, 'html.parser')
        file_path = f'data/films/{self.id}.html'
        with open(file_path, 'w') as f:
            f.write(str(soup))

    def download_actors_html(self):
        r = requests.get(self.get_actors_url())
        if r.status_code != 200:
            return
        soup = BeautifulSoup(r.content, 'html.parser')
        file_path = f'data/actors/{self.id}.html'
        with open(file_path, 'w') as f:
            f.write(str(soup))

    def get_actors_url(self):
        return f'{self.url}/fullcredits'

    def scrap(self):
        file_path = f'data/films/{self.id}.html'
        if not os.path.exists(file_path):
            return
        with open(file_path, 'r') as f:
            soup = BeautifulSoup(f, 'html.parser')
            self.title = soup.find('h1').text
            self.grossUSA = self.get_amount(soup, 'Gross USA')
            self.grossWW = self.get_amount(soup, 'Cumulative Worldwide Gross')
            self.actors = self.get_actors()

    def get_actors(self):
        actors = []
        file_path = f'data/actors/{self.id}.html'
        if not os.path.exists(file_path):
            return actors
        with open(file_path, 'r') as f:
            soup = BeautifulSoup(f, 'html.parser')
            table = soup.find('table', attrs={"class": "cast_list"})
            if table is None:
                return actors
            odd_trs = table.find_all('tr', attrs={"class": "odd"})
            for tr in odd_trs:
                td = tr.find_all('td')[1]
                actors.append(td.a.text)
            even_trs = table.find_all('tr', attrs={"class": "even"})
            for tr in even_trs:
                td = tr.find_all('td')[1]
                actors.append(td.a.text)
        return actors

    def get_amount(self, soup, amount_type):
        amount = None
        for div in soup.find_all('div', attrs={"class": "txt-block"}):
            h4 = div.find('h4', attrs={"class": "inline"})
            if h4 is not None and h4.text == amount_type + ':':
                amount = div.text.split('\n')[1]
                pattern = '^' + amount_type + ': \$((\d+,?)+)(,.+?)?$'
                x = re.match(pattern, amount)
                amount = x.group(1)
        return amount

    def load(self, db):
        film_dict = {"title": self.title, "url": self.url, "imdb_id": self.id,
                     "grossUSA": self.grossUSA, "grossWW": self.grossWW, "actors": self.actors}
        film_coll = db.films
        film_coll.insert_one(film_dict)

    def visu(self):
        pass

# Example usage
if __name__ == "__main__":
    year = 2022
    film_list = FilmList(year)
    film_list.download_html()
    film_ids = film_list.get_films_ids()

    client = pymongo.MongoClient("mongodb://localhost:27017/")
    db = client["imdb_database"]

    for film_id in film_ids:
        film = Film(film_id)
        film.download_html()
        film.download_actors_html()
        film.scrap()
        film.load(db)
