from socket import *
import threading

class FollowerServer:
    def __init__(self, selfport, leaderport):
        self.selfport = selfport
        self.leaderport = leaderport
        self.lock_map = []
        self.clients = []
        self.connections = []
        self.followerid = None

    def _connect_with_leaderserver(self, host, port):
        print("Start connecting with leader_server")
        leader_socket = socket(AF_INET, SOCK_STREAM)
        leader_socket.connect((host, port))
        senddata = "NewFollower"
        leader_socket.sendall(senddata.encode())
        print("Follower sends data")
        data = leader_socket.recv(1024).decode('utf-8')
        msg = data.split(":")
        if msg[0] == "FollowerId":
            self.followerid = int(msg[1])
        return leader_socket

    def _forward_down(self, leader_socket):
        """
        Forward messages from leader_server to clients.
        """
        while True:
            data = leader_socket.recv(1024).decode()
            if not data:
                continue
            msg = data.split(":")
            if msg[0] == 'UpdateLockmap':
                self.lock_map.append({'name': msg[1], 'client': msg[2]})
                print("UpdateLockmap")
            elif msg[0] == 'RemoveLockmap':
                self.lock_map.remove({'name': msg[1], 'client': msg[2]})
                print("RemoveLockmap")
            elif msg[0] in ['PreemptLock Success', 'PreemptLock Failed', 'ReleaseLock Success', 'ReleaseLock Failed']:
                client_id = msg[1]
                for client in self.clients:
                    if client['client_id'] == int(client_id):
                        client['socket'].sendall(msg[0].encode())

    def _new_client(self, c_socket):
        client_id = self.followerid * 1000 + len(self.clients) + 1
        self.clients.append({'client_id': client_id, 'socket': c_socket})
        return client_id

    def _forward_up(self, leader_socket, c_socket):
        """
        Forward messages from clients to leader_server.
        """
        while True:
            data = c_socket.recv(1024).decode()
            if not data:
                continue
            if data == 'NewClient':
                client_id = self._new_client(c_socket)
                c_socket.sendall(("ClientId:%d" % client_id).encode())
                continue
            msg = data.split(":")
            if msg[0] == "PreemptLock" or msg[0] == "ReleaseLock":
                leader_socket.sendall((data).encode())
            elif msg[0] == "CheckLock":
                if self.lock_map:
                    for lock in self.lock_map:
                        if lock['name'] == msg[1]:
                            client_id = lock['client']
                            break
                    c_socket.sendall(("CheckLock:Lock:%d" % int(client_id)).encode())
                else:
                    c_socket.sendall(("There are no locks in the server").encode())

    def run(self):
        host = '127.0.0.1'
        leader_socket = self._connect_with_leaderserver(host, self.leaderport)
        print("Leader socket and follower_id:", leader_socket, self.followerid)

        forward_down = threading.Thread(target=self._forward_down, args=(leader_socket,))
        forward_down.setDaemon(True)
        forward_down.start()

        s = socket(AF_INET, SOCK_STREAM)
        s.bind((host, self.selfport))
        s.listen(3)
        print(s, "listening")

        while True:
            c_soc, addr = s.accept()
            print("Connected with", addr)
            forward_up = threading.Thread(target=self._forward_up, args=(leader_socket, c_soc))
            forward_up.setDaemon(True)
            forward_up.start()

if __name__ == "__main__":
    follower_server = FollowerServer(selfport=5000, leaderport=6000)
    follower_server.run()
