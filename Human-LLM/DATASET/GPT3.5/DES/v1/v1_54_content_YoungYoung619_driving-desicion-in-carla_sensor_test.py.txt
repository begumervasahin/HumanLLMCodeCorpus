import glob
import os
import sys
import time
from threading import Thread
import cv2
import carla
from carla_utils.world_ops import *
from carla_utils.sensor_ops import *

# Suppress SSL warnings
import warnings
warnings.filterwarnings('ignore')

# Define logging setup (assuming logger is defined somewhere else)
logger = ...

def check_whether_respawn_actors(world, vehicles):
    """Check whether to respawn the static actors in a frequency."""
    while True:
        if carla_actors_static(vehicles, bigger_than=0.8):
            respawn_static_actors(world, vehicles)
        time.sleep(20)

if __name__ == '__main__':
    # Connect to CARLA server
    client = carla.Client('127.0.0.1', 2000)
    client.set_timeout(10.0)
    logger.info('Connected successfully...')
    
    # Get the CARLA world
    world = client.get_world()
    
    # Spawn vehicles
    spawn_vehicles(world, number=5)
    
    # Get actor list and filter vehicles
    actor_list = world.get_actors()
    vehicles = list(actor_list.filter('vehicle*'))
    
    # Configure RGB camera sensor
    bgr_camera_config = {
        'data_type': 'sensor.camera.rgb',
        'image_size_x': 418,
        'image_size_y': 278,
        'fov': 110,
        'sensor_tick': 0.05,
        'transform': carla.Transform(carla.Location(x=0.8, z=1.7)),
        'attach_to': vehicles[0]
    }
    bgr_sensor = bgr_camera(world, bgr_camera_config)
    
    # Configure depth camera sensor
    depth_camera_config = {
        'data_type': 'sensor.camera.depth',
        'image_size_x': 418,
        'image_size_y': 278,
        'fov': 110,
        'sensor_tick': 0.02,
        'transform': carla.Transform(carla.Location(x=0.8, z=1.7)),
        'attach_to': vehicles[0]
    }
    depth_sensor = depth_camera(world, depth_camera_config)
    
    # Configure semantic segmentation camera sensor
    semantic_camera_config = {
        'data_type': 'sensor.camera.semantic_segmentation',
        'image_size_x': 418,
        'image_size_y': 278,
        'fov': 110,
        'sensor_tick': 0.02,
        'transform': carla.Transform(carla.Location(x=0.8, z=1.7)),
        'attach_to': vehicles[0]
    }
    semantic_sensor = semantic_camera(world, semantic_camera_config)
    
    # Start a thread to check whether to respawn actors
    respawn_thread = Thread(target=check_whether_respawn_actors, args=(world, vehicles))
    respawn_thread.daemon = True
    respawn_thread.start()
    
    logger.info('Press key "a" to stop...')
    
    # Display sensor data until "a" key is pressed
    while True:
        bgr = bgr_sensor.get()
        depth = depth_sensor.get()
        semantic = semantic_sensor.get()
        
        cv2.imshow('Color', bgr)
        cv2.imshow('Depth', depth)
        cv2.imshow('Semantic', semantic)
        
        key = cv2.waitKey(10)
        if key == 97:  # "a" key
            cv2.destroyAllWindows()
            break
    
    # Destroy all vehicles
    vehicles = list(actor_list.filter('vehicle*'))
    for vehicle in vehicles:
        vehicle.destroy()
    logger.info('Destroyed all vehicles...')
    
    # Destroy all sensors
    sensors = list(actor_list.filter('sensor*'))
    for sensor in sensors:
        sensor.destroy()
    logger.info('Destroyed all sensors...')
