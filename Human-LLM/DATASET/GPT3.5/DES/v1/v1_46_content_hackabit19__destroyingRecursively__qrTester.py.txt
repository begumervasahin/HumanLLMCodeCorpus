from qr import QR
from picamera.array import PiRGBArray
from picamera import PiCamera
from time import sleep
import cv2

# Initialize PiCamera and QR scanner
camera = PiCamera()
qr = QR()

# Continuously capture images and scan for QR codes
while True:
    # Pause for 0.5 seconds
    sleep(0.5)
    
    # Capture an image
    rawCapture = PiRGBArray(camera)
    camera.capture(rawCapture, "bgr")
    image = rawCapture.array
    
    # Scan the captured image for QR codes
    for f_x, f_y, data in qr.scan(image):
        print(f"Found QR code at ({f_x}, {f_y}): {data}")
