import numpy as np
import pandas as pd
import math
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error, mean_absolute_error

# Function to load data from a file
def load_data(file_path):
    return pd.read_table(file_path)

# Function to calculate the standard error of a sample
def standard_error(sample):
    std = np.std(sample, ddof=0)
    return std / math.sqrt(len(sample))

# Function to calculate the objective function for Lasso regression
def objective_function(A, x, y, alpha):
    residual_error = np.dot(A, x) - y
    lasso_penalty = alpha * np.sum(np.abs(x))
    return 0.5 * (residual_error.T.dot(residual_error)) + lasso_penalty

# Function to calculate the loss function for Lasso regression
def loss_function(A, x, y):
    residual_error = np.dot(A, x) - y
    return 0.5 * (residual_error.T.dot(residual_error))

# Function to calculate the gradient of the loss function
def gradient(A, x, y):
    return A.T.dot(A.dot(x) - y)

# Function to calculate the value of the approximation function
def approximate_function(x, xk, A, y, gamma_k):
    inner_product = gradient(A, xk, y).T.dot(x - xk)
    x_diff = x - xk
    return loss_function(A, xk, y) + inner_product + (1.0 / (2.0 * gamma_k)) * x_diff.T.dot(x_diff)

# Function to perform shrinkage operation for Lasso regression
def shrinkage(x, kappa):
    return np.maximum(0., x - kappa) - np.maximum(0., -x - kappa)

# Function to perform Lasso regression using proximal gradient descent
def lasso_proximal(X, y, alpha=1, max_iter=2000, beta=0.99, gamma_k=0.05):
    X = np.array(X)
    y = np.array([[i] for i in y])
    initial_coefficients = np.array([[0.48], [0.50], [-0.01], [0.11], [0.81], [-0.03], [0.13], [0.003]])
    
    # Perform proximal gradient descent
    coefficients = initial_coefficients
    for _ in range(max_iter):
        while True:
            next_coefficients = coefficients - gamma_k * gradient(X, coefficients, y)
            if loss_function(X, next_coefficients, y) <= approximate_function(next_coefficients, coefficients, X, y, gamma_k):
                break
            else:
                gamma_k = beta * gamma_k
        
        # Apply L1 regularization (proximal operator)
        next_coefficients = shrinkage(next_coefficients, gamma_k * alpha)
        
        # Check convergence
        if np.linalg.norm(objective_function(X, next_coefficients, y, alpha) - objective_function(X, coefficients, y, alpha)) < 0.01:
            break
        
        coefficients = next_coefficients
    
    return coefficients

# Load data
data = load_data(r'D:\360MoveData\Users\HP\Desktop\prostate.data.txt')

# Prepare data
factors = ['lcavol', 'lweight', 'age', 'lbph', 'svi', 'lcp', 'gleason', 'pgg45']        
X = data[factors]
Y = data['lpsa']
X_train, X_test, y_train, y_test = train_test_split(X, Y, test_size=0.2, random_state=1)

# Perform Lasso regression
coefficients = lasso_proximal(X_train, y_train, alpha=0.0058)

# Predict on test set
y_test_predict = X_test.dot(coefficients)

# Calculate metrics
mse_predict = round(mean_squared_error(y_test, y_test_predict), 4)
mae_predict = round(mean_absolute_error(y_test, y_test_predict), 4)
std_error = round(standard_error(y_test_predict), 4)

# Format coefficients
coef = [(factors[i], round(coefficients[i][0], 4)) for i in range(8)]

# Display results
print('Estimated coefficients are:', coef)
print('Std Error is:', std_error)
print('MSE is:', mse_predict)
print('MAE is:', mae_predict)
