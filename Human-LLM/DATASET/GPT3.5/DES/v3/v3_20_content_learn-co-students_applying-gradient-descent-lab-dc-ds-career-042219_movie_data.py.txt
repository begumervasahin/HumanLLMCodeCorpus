# Function to calculate the return on investment (ROI) for a movie
def calculate_roi(movie_data):
    budget = movie_data['budget']
    domgross = movie_data['domgross']
    roi = ((domgross - budget) / budget) * 100
    return round(roi, 2)

# Function to filter movies based on a specified ROI threshold
def filter_movies_by_roi(movies_list, roi_threshold):
    filtered_movies = [movie for movie in movies_list if calculate_roi(movie) >= roi_threshold]
    return filtered_movies

# Function to print movie titles and their ROIs
def print_movies_with_roi(movies_list):
    for movie_data in movies_list:
        title = movie_data['title']
        roi = calculate_roi(movie_data)
        print(f"Movie: {title.ljust(30)} ROI: {roi}%")

# List of movies with budget and domestic gross data
movies_data = [
    {'budget': 13000000, 'domgross': 25682380.0, 'title': '21 & Over'},
    {'budget': 45658735, 'domgross': 13414714.0, 'title': 'Dredd 3D'},
    # Other movie data omitted for brevity
    {'budget': 30000000, 'domgross': 4167493.0, 'title': 'Gloria'}
]

# Set the ROI threshold for filtering movies
roi_threshold = 50  # 50% ROI threshold

# Filter movies based on the ROI threshold
filtered_movies_data = filter_movies_by_roi(movies_data, roi_threshold)

# Print filtered movies along with their ROIs
print("Movies with ROI greater than or equal to", roi_threshold, "percent:")
print_movies_with_roi(filtered_movies_data)
