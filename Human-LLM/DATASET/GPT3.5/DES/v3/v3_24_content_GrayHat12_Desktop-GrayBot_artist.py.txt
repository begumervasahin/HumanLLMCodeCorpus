import gplaylist as gp
import os

# Help message
HELP_MESSAGE = '''
show : See the list of your favorite artists
add : Add a name to your favorite artist list
rem : Remove a name from your favorite artist list
quit : Quit the program
clr : Clear the screen
help : Show this menu
'''
print(HELP_MESSAGE)

# Function to clear the screen
def clear_screen():
    if os.name == 'nt':
        os.system('cls')
    elif os.name == 'posix':
        os.system('clear')
    else:
        print('Sorry, Clear Screen is not supported for your OS:', os.name)

# Main loop
while True:
    user_input = input('..> ').lower()

    if user_input == 'show':
        # Show favorite artists
        favorite_artists = gp.GetPlayList().getNames()
        for artist in favorite_artists:
            print(artist)

    elif user_input == 'add':
        # Add a favorite artist
        name = input('Name of Artist to add: ')
        playlist = gp.AddPlayList(name)
        if playlist.hasErr():
            print(playlist.getError())
        else:
            print('Successfully added the artist')

    elif user_input == 'rem':
        # Remove a favorite artist
        name = input('Name of Artist to remove: ')
        playlist = gp.RemoveArtist(name)
        if playlist.hasErr():
            print(playlist.getError())
        elif playlist.found:
            print('Successfully removed the artist')
        else:
            print('Artist not found in the List')

    elif user_input == 'quit':
        # Quit the program
        break

    elif user_input == 'clr':
        # Clear the screen
        clear_screen()

    elif user_input == 'help':
        # Show help message
        print(HELP_MESSAGE)
