import sys

# Rabin constants
CRABIN = 'rP9hpgejqp8v9YLvJ6b1QArhFOTm'
A_FACTOR = 7
B_FACTOR = 11
NRABIN = 162200874457722185484714045548885175084

# Rabin hash functions
A_H = I_H = J_H = W_H = 0
S_H = []

def update_h():
    global A_H, I_H, J_H, W_H, S_H
    I_H += W_H
    if I_H > 255:
        I_H -= 256
    J_H += S_H[I_H]
    if J_H > 255:
        J_H = S_H[J_H - 256]
    else:
        J_H = S_H[J_H]
    t = S_H[I_H] + S_H[J_H]
    S_H[J_H] = t - S_H[J_H]
    S_H[I_H] = t - S_H[I_H]

def output_h():
    global A_H, I_H, J_H, W_H, S_H
    update_h()
    return S_H[J_H]

def shuffle_h():
    global A_H, I_H, J_H, W_H, S_H
    for v in range(256):
        update_h()
    W_H = (W_H + 2) % 256
    A_H = 0

def absorb_nibble_h(x):
    global A_H, I_H, J_H, W_H, S_H
    if A_H == 241:
        shuffle_h()
    S_H[A_H], S_H[240 + x] = S_H[240 + x], S_H[A_H]
    A_H += 1

def absorb_byte_h(b):
    absorb_nibble_h(b % 16)
    absorb_nibble_h(b >> 4)

def squeeze_h(out, outlen):
    global A_H, I_H, J_H, W_H, S_H
    shuffle_h()
    for v in range(outlen):
        out.append(output_h())

def h(x):
    global A_H, I_H, J_H, W_H, S_H
    A_H = I_H = J_H = 0
    W_H = 1
    S_H = [ix for ix in range(256)]
    for c in x:
        absorb_byte_h(ord(c))
    res = []
    squeeze_h(res, 128)
    out = 0
    for bx in res:
        out = (out << 8) + bx
    return out % NRABIN

def hb(x):
    global A_H, I_H, J_H, W_H, S_H
    A_H = I_H = J_H = 0
    W_H = 1
    S_H = [ix for ix in range(256)]
    for c in x:
        absorb_byte_h(c)
    res = []
    squeeze_h(res, 128)
    out = 0
    for bx in res:
        out = (out << 8) + bx
    return out % NRABIN

# Conversion functions
def num2code(x):
    res = ''
    while x > 0:
        y = x % 64
        if y < 10:
            res = chr(y + 48) + res
        elif y < 36:
            res = chr(y + 55) + res
        elif y < 62:
            res = chr(y + 61) + res
        elif y == 62:
            res = '@' + res
        elif y == 63:
            res = '/' + res
        x >>= 6
    return res

def code2num(x):
    res = 0
    for c in x:
        if 48 <= ord(c) < 58:
            res = (res << 6) + ord(c) - 48
        elif 65 <= ord(c) < 91:
            res = (res << 6) + ord(c) - 55
        elif 97 <= ord(c) < 123:
            res = (res << 6) + ord(c) - 61
        elif c == '@':
            res = (res << 6) + 62
        elif c == '/':
            res = (res << 6) + 63
    return res

# File I/O functions
def write_number(n, fnam):
    with open(fnam, 'wb') as f:
        while n > 0:
            b = n & 0xFF
            n >>= 8
            f.write(bytes([b]))

def read_number(fnam):
    with open(fnam, 'rb') as f:
        n = 0
        for c in reversed(f.read()):
            n = (n << 8) ^ c
    return n

def hF(fnam):
    with open(fnam, 'rb') as f:
        return hb(f.read())

def vF(s, fnam):
    a = A_FACTOR
    b = B_FACTOR
    h0 = hF(fnam)
    ha = (a * h0) % NRABIN
    hb = (b * h0) % NRABIN
    hab = (a * b * h0) % NRABIN
    sq = (s * s) % NRABIN
    return (h0 == sq) or (ha == sq) or (hb == sq) or (hab == sq)
