import socket
from PIL import ImageGrab
import io
from threading import Thread
import sys
from PyQt5.QtWidgets import QMainWindow, QApplication, QPushButton, QLineEdit
from PyQt5.QtCore import QRect
from random import randint

class DesktopClient(QMainWindow):
    def __init__(self):
        super().__init__()
        self.init_ui()

    def start_thread(self):
        self.start.start()

    def change_image(self):
        try:
            if self.ip.text() and self.port.text():
                sock = socket.socket()
                sock.connect((self.ip.text(), int(self.port.text())))
                while True:
                    img = ImageGrab.grab()
                    img_bytes = io.BytesIO()
                    img.save(img_bytes, format='PNG')
                    sock.send(img_bytes.getvalue())
                sock.close()
        except Exception as e:
            print("Error:", e)

    def init_ui(self):
        self.setWindowTitle(f"[CLIENT] Remote Desktop: {randint(99999, 999999)}")
        self.setFixedSize(400, 90)

        self.start = Thread(target=self.change_image, daemon=True)

        self.btn = QPushButton("Start Demo", self)
        self.btn.setGeometry(5, 55, 390, 30)
        self.btn.clicked.connect(self.start_thread)

        self.ip = QLineEdit("IP", self)
        self.ip.setGeometry(5, 5, 390, 20)

        self.port = QLineEdit("PORT", self)
        self.port.setGeometry(5, 30, 390, 20)

if __name__ == '__main__':
    app = QApplication(sys.argv)
    ex = DesktopClient()
    ex.show()
    sys.exit(app.exec_())
