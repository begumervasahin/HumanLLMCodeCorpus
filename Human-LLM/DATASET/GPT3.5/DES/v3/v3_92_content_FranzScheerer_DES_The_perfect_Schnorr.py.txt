def write_number(number, filename):
    """
    Write a number to a file in binary format.
    """
    with open(filename, 'wb') as f:
        while number > 0:
            byte = number % 256
            number //= 256
            f.write(chr(byte))

def bin_to_num(binary_string):
    """
    Convert a binary string to a number.
    """
    result = 0
    for char in binary_string:
        result = (result << 8) ^ ord(char)
    return result

def num_to_bin(number):
    """
    Convert a number to a binary string.
    """
    binary_string = ''
    while number > 0:
        binary_string = chr(number % 256) + binary_string
        number //= 256
    return binary_string

def gcd(a, b):
    """
    Compute the greatest common divisor of two numbers.
    """
    while b > 0:
        a, b = b, a % b
    return a

def next_prime(p):
    """
    Find the next prime number greater than p.
    """
    while p % 8 != 3:
        p += 1
    return next_prime_odd(p)

def next_prime_odd(p):
    """
    Find the next prime number greater than p that is congruent to 3 modulo 8.
    """
    mod_product = 3 * 5 * 7 * 11 * 13 * 17 * 19 * 23 * 29
    while True:
        while gcd(p, mod_product) != 1:
            p += 8
        q = (p + 1) // 4
        if (pow(2, p - 1, p) != 1 or pow(2, q - 1, q) != 1 or
            pow(3, p - 1, p) != 1 or pow(3, q - 1, q) != 1 or
            pow(5, p - 1, p) != 1 or pow(5, q - 1, q) != 1 or
            pow(17, p - 1, p) != 1 or pow(17, q - 1, q) != 1):
            p += 8
            continue
        break
    return p

def inv(b, m):
    """
    Compute the modular multiplicative inverse of b modulo m.
    """
    s, t = 0, 1
    while b != 1:
        q = a // b
        a, b = b, a % b
        s, t = t, s - q * t
    if t < 0:
        t += m
    return t

def E(m, key, prime):
    """
    Encrypt the message m using the key and prime modulus.
    """
    return pow(m, key, prime)

def D(c, key, prime):
    """
    Decrypt the ciphertext c using the key and prime modulus.
    """
    key = inv(key, prime - 1)
    return pow(c, key, prime)

def hex_text_to_num(hex_text):
    """
    Convert a hexadecimal string to a number.
    """
    result = 0
    for char in hex_text:
        if ord(char) < 58 and ord(char) >= 48:
            result = (result << 4) + ord(char) - 48
        elif ord(char) <= ord('f') and ord(char) >= ord('a'):
            result = (result << 4) + ord(char) - 87
    return result

def code_to_num(code):
    """
    Convert a custom code to a number.
    """
    result = 0
    for char in code:
        if ord(char) >= 48 and ord(char) < 58:
            result = (result << 6) + ord(char) - 48
        if ord(char) >= 65 and ord(char) < 91:
            result = (result << 6) + ord(char) - 55
        if ord(char) >= 97 and ord(char) < 123:
            result = (result << 6) + ord(char) - 61
        if char == "'":
            result = (result << 6) + 62
        if char == '/':
            result = (result << 6) + 63
    return result

def num_to_code(number):
    """
    Convert a number to a custom code.
    """
    code = ''
    while number > 0:
        remainder = number % 64
        if remainder < 10:
            code = chr(remainder + 48) + code
        elif remainder < 36:
            code = chr(remainder + 55) + code
        elif remainder < 62:
            code = chr(remainder + 61) + code
        elif remainder == 62:
            code = "'" + code
        elif remainder == 63:
            code = '/' + code
        number //= 64
    return code

def h(x):
    """
    Compute the hash of a string x modulo n4.
    """
    digest = hashlib.sha256(x.encode()).digest()
    result = 0
    for byte in digest:
        result = (result << 8) ^ byte
    return result % n4

def random_256(m):
    """
    Generate a random number from a seed using 256-bit hash function.
    """
    md = hashlib.sha256("***RANDOM-SEED_X***".encode())
    md.update('large key value for generation of random number'.encode())
    md.update(m.encode())
    result = 0
    largestr = md.digest()
    for i in range(len(largestr)):
        result = (result << 8) ^ ord(largestr[i])
    return result

def random_X(m):
    """
    Generate a random number from a seed using 256-bit hash function.
    """
    md = hashlib.sha256("***RANDOM-SEED_X***".encode())
    md.update('large key value for generation of random number'.encode())
    md.update(m.encode())
    md.update(str(time.gmtime().tm_year + time.gmtime().tm_mday).encode())
    result = 0
    largestr = md.digest()
    for i in range(len(largestr)):
        result = (result << 8) ^ ord(largestr[i])
    return result

def gen_P(x, a, b):
    """
    Generate a point on an elliptic curve.
    """
    if (4 * a**3 + 27 * b**2) % prime == 0:
        b += 1
    while pow(c * x**3 + a * x + b, (prime - 1) / 2, prime) != 1:
        x += 1
    y = pow(c * x**3 + a * x + b, (prime + 1) / 4, prime)
    return [x % prime, y % prime]

def d_point(P):
    """
    Compute the negation of a point on an elliptic curve.
    """
    x, y = P
    s = ((3 * c * (x**2) + a) * inv(2 * y, prime)) % prime
    xr = (cinv * s**2 - 2 * x) % prime
    yr = (-y + s * (x - xr)) % prime
    return [xr, yr]

def add_P(P, Q):
    """
    Add two points on an elliptic curve.
    """
    if P == Q:
        return d_point(P)
    x1, x2 = P[0], Q[0]
    y1, y2 = P[1], Q[1]
    while x1 < x2:
        x1 += prime
    while y1 < y2:
        y1 += prime
    s = ((y1 - y2) * inv(x1 - x2, prime)) % prime
    xr = (cinv * s**2 - x1 - x2) % prime
    yr = (s * (x1 - xr) - y1) % prime
    return [xr % prime, yr % prime]

def mul_P(P, n):
    """
    Multiply a point on an elliptic curve by a scalar.
    """
    is_first = True
    res_P = P
    if n < 0:
        res_P[1] = prime - res_P[1]
        n = (-1) * n
    bit_size = 20
    while 2**bit_size < n:
        bit_size += 1
    PP = res_P
    for bit in range(bit_size + 1):
        if (n & (1 << bit) != 0):
            if is_first:
                res_P = PP
                is_first = False
            else:
                res_P = add_P(res_P, PP)
        PP = d_point(PP)
    return res_P

def sign_schnorr(G, m, x):
    """
    Generate a Schnorr signature for a message.
    """
    k = random_X(m)
    R = mul_P(G, k)
    e = h(str(R[0]) + m)
    return [(k - x * e) % n4, e]

def ecdsa(G, m, x):
    """
    Generate an ECDSA signature for a message.
    """
    hh = h(m)
    k = 2 * random_X(m) + 1
    R = mul_P(G, k)
    s = (inv(k, n4) * (hh + R[0] * x)) % n4
    return [s, R[0]]

def ecdsa_verify(G, m, S, Y):
    """
    Verify an ECDSA signature for a message.
    """
    si = inv(S[0], n4)
    hh = h(m)
    u1 = (si * h(m)) % n4
    u2 = (si * S[1]) % n4
    return add_P(mul_P(G, u1), mul_P(Y, u2))[0] == S[1]
