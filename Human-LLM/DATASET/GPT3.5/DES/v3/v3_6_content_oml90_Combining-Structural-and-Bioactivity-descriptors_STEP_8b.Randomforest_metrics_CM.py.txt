import time
import pandas as pd
from sklearn import metrics

def calculate_metrics(assay, fp, index, file_path, output_dir):
    # Read data from the file
    df = pd.read_csv(file_path.format(assay, assay, fp), sep='\t').sort_values('PredProb(A)', ascending=False).reset_index(drop=True)

    # Calculate ROC AUC
    fpr, tpr, thresholds = metrics.roc_curve(df['label'], df['PredProb(A)'], pos_label='A')
    roc_auc = round(metrics.auc(fpr, tpr), 3)

    # Calculate precision, recall, F1 score, and support
    precision, recall, f1_score, support = metrics.precision_recall_fscore_support(df['label'], df['pred'])

    # Calculate confusion matrix
    confusion_matrix = metrics.confusion_matrix(df['label'], df['pred'])

    # Calculate Cohen's Kappa
    kappa = metrics.cohen_kappa_score(df['label'], df['pred'])

    # Calculate Matthews correlation coefficient
    mcc = metrics.matthews_corrcoef(df['label'], df['pred'])

    # Extract individual values from precision, recall, and F1 score
    tp, fn, fp, tn = confusion_matrix.flatten()
    tp, fn, fp, tn = int(tp), int(fn), int(fp), int(tn)

    # Write the calculated metrics to the output file
    with open(output_dir, 'a') as outfile:
        outfile.write('{}'.format(index))
        for val in [fp, assay, roc_auc, mcc, kappa, precision[0], recall[0], f1_score[0], support[0], support[1], tp, fn, fp, tn]:
            outfile.write('\t{}'.format(val))
        outfile.write('\n')
