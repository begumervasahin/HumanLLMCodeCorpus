import hashlib
import sys

# Function to read a number from a file
def readNumberFromFile(filename):
    with open(filename, 'rb') as file:
        number = 0
        file_content = file.read()
        for i in range(len(file_content)):
            number = (number << 8) ^ ord(file_content[len(file_content) - i - 1])
    return number

# Constants
prime = (2**160 * 5 * 23) + 86427
n_ = prime + 1
a = prime - 31
c = 17

# Function to compute the hash of input x using MD5
def computeHash(x):
    md5_hash = hashlib.md5(x).digest()
    res = 0
    for character in md5_hash:
        res = (res << 8) ^ ord(character)
    return res

# Function to generate a point on the elliptic curve
def generatePoint(x, a, b):
    if (4*a*a*a + 27*b*b) % prime == 0:
        b = b + 1
    while pow(c*x**3 + a*x + b, (prime - 1)/2, prime) != 1:
        x = x + 1
    y = pow(c*x**3 + a*x + b, (prime + 1)/4, prime)
    return [x % prime, (y) % prime]

# Function to add two points on the elliptic curve
def addPoints(P, Q):
    x1, x2 = P[0], Q[0]
    y1, y2 = P[1], Q[1]
    while x1 < x2:
        x1 = x1 + prime
    if x1 == x2:
        s = ((3*c*(x1**2) + a) * pow(2*y1, prime - 2, prime)) % prime
    else:
        s = ((y1 - y2) * pow(x1 - x2, prime - 2, prime)) % prime
    xr = cinv*s**2 - x1 - x2
    yr = s * (x1 - xr) - y1
    return [xr % prime, yr % prime]

# Function to multiply a point on the elliptic curve by a scalar
def multiplyPoint(P, n):
    isFirst = True
    result = P
    if n < 0:
        result[1] = prime - result[1]
        n = (-1) * n
    PP = result
    while n > 0:
        if n % 2 != 0:
            if isFirst:
                result = PP
                isFirst = False
            else:
                result = addPoints(result, PP)
        PP = addPoints(PP, PP)
        n = n // 2
    return result

# Function to verify the signature of a message
def verifySignature(G, s, Y, e, message):
    return e == computeHash(str(addPoints(multiplyPoint(G, s), multiplyPoint(Y, e))[0]) + message)
