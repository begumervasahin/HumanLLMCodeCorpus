import random as r

# Constants
NUM_BOULE_MIN = 1
NUM_BOULE_MAX_NORMAL = 49
NUM_BOULE_MAX_COMP = 10
NB_BOULES_NORMALES = 6
JACKPOT = 5000000
MISE = 2

def update_win_history(player_grid, official_draw, win_history, total_winnings):
    correct_numbers = 0
    bonus_number_matched = False

    for player_num in player_grid:
        for draw_num in official_draw:
            if player_num == draw_num:
                correct_numbers += 1

    if player_grid[NB_BOULES_NORMALES] == official_draw[NB_BOULES_NORMALES]:
        bonus_number_matched = True

    if correct_numbers == 5 and bonus_number_matched:
        win_history[0][1] += 1
        total_winnings += win_history[0][0]
    elif correct_numbers == 5:
        win_history[1][1] += 1
        total_winnings += win_history[1][0]
    elif correct_numbers == 4 and bonus_number_matched:
        win_history[2][1] += 1
        total_winnings += win_history[2][0]
    elif correct_numbers == 4:
        win_history[3][1] += 1
        total_winnings += win_history[3][0]
    elif correct_numbers == 3 and bonus_number_matched:
        win_history[4][1] += 1
        total_winnings += win_history[4][0]
    elif correct_numbers == 3:
        win_history[5][1] += 1
        total_winnings += win_history[5][0]
    elif correct_numbers == 2 and bonus_number_matched:
        win_history[6][1] += 1
        total_winnings += win_history[6][0]
    elif correct_numbers == 2:
        win_history[7][1] += 1
        total_winnings += win_history[7][0]
    elif correct_numbers == 1 and bonus_number_matched:
        win_history[8][1] += 1
        total_winnings += win_history[8][0]
    elif bonus_number_matched:
        win_history[8][1] += 1
        total_winnings += win_history[8][0]

    return win_history, total_winnings

def create_win_history():
    win_history = [
        [JACKPOT, 0],
        [100000, 0],
        [1000, 0],
        [500, 0],
        [50, 0],
        [20, 0],
        [10, 0],
        [5, 0],
        [MISE, 0],
    ]
    return win_history

win_history = create_win_history()
