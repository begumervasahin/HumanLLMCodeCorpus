import tensorly as tl
import torch
import torch.nn as nn

# Alias for parafac from tensorly
parafac = tl.decomposition.parafac

def cp_decomposition_conv_layer(layer, rank):
    """ Decompose a convolutional layer using CP decomposition and return the decomposed layers """

    # Perform CP decomposition on the layer's weight data
    last, first, vertical, horizontal = parafac(layer.weight.data, rank=rank, init='svd')

    # Create pointwise convolution layers
    pointwise_s_to_r_layer = nn.Conv2d(
        in_channels=first.shape[0],
        out_channels=first.shape[1],
        kernel_size=1,
        stride=1,
        padding=0,
        dilation=layer.dilation,
        bias=False
    )

    depthwise_vertical_layer = nn.Conv2d(
        in_channels=vertical.shape[1],
        out_channels=vertical.shape[1],
        kernel_size=(vertical.shape[0], 1),
        stride=1,
        padding=(layer.padding[0], 0),
        dilation=layer.dilation,
        groups=vertical.shape[1],
        bias=False
    )

    depthwise_horizontal_layer = nn.Conv2d(
        in_channels=horizontal.shape[1],
        out_channels=horizontal.shape[1],
        kernel_size=(1, horizontal.shape[0]),
        stride=layer.stride,
        padding=(0, layer.padding[0]),
        dilation=layer.dilation,
        groups=horizontal.shape[1],
        bias=False
    )

    pointwise_r_to_t_layer = nn.Conv2d(
        in_channels=last.shape[1],
        out_channels=last.shape[0],
        kernel_size=1,
        stride=1,
        padding=0,
        dilation=layer.dilation,
        bias=True
    )

    # Set bias for the pointwise_r_to_t_layer
    pointwise_r_to_t_layer.bias.data = layer.bias.data

    # Transpose and set weights for the decomposed layers
    depthwise_horizontal_layer.weight.data = torch.transpose(horizontal, 1, 0).unsqueeze(1).unsqueeze(1)
    depthwise_vertical_layer.weight.data = torch.transpose(vertical, 1, 0).unsqueeze(1).unsqueeze(-1)
    pointwise_s_to_r_layer.weight.data = torch.transpose(first, 1, 0).unsqueeze(-1).unsqueeze(-1)
    pointwise_r_to_t_layer.weight.data = last.unsqueeze(-1).unsqueeze(-1)

    # Create a sequential model with the decomposed layers
    decomposed_layers = [
        pointwise_s_to_r_layer,
        depthwise_vertical_layer,
        depthwise_horizontal_layer,
        pointwise_r_to_t_layer
    ]

    return nn.Sequential(*decomposed_layers)
