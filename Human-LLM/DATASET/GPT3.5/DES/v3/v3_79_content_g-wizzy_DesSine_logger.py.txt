class TextColors:
    HEADER = '\033[95m'
    OKBLUE = '\033[94m'
    OKGREEN = '\033[92m'
    WARNING = '\033[93m'
    FAIL = '\033[91m'
    ENDC = '\033[0m'
    BOLD = '\033[1m'
    UNDERLINE = '\033[4m'

def print_error(prefix, line, message):
    """
    Print an error message with specific formatting.

    Args:
        prefix (str): Prefix for the log entry.
        line (int): Line number where the error occurred.
        message (str): Message to display.
    """
    print(f"{TextColors.FAIL}[{prefix}] {TextColors.ENDC}{TextColors.BOLD}Line {line}{TextColors.ENDC} : {message}")

def print_warning(prefix, message):
    """
    Print a warning message with specific formatting.

    Args:
        prefix (str): Prefix for the log entry.
        message (str): Message to display.
    """
    print(f"{TextColors.WARNING}[{prefix}] {TextColors.ENDC}{message}")

def print_info(prefix, message):
    """
    Print an information message with specific formatting.

    Args:
        prefix (str): Prefix for the log entry.
        message (str): Message to display.
    """
    print(f"{TextColors.OKGREEN}[{prefix}] {TextColors.ENDC}{message}")

def print_debug(variables):
    """
    Print debugging information with specific formatting.

    Args:
        variables (list): List of printable objects.
    """
    print(f"{TextColors.OKBLUE}[DesSine Debug] {TextColors.ENDC}{', '.join(map(str, variables)))")

# Example usage
if __name__ == "__main__":
    print_error("ERROR", 42, "This is an error message")
    print_warning("WARNING", "This is a warning message")
    print_info("INFO", "This is an information message")
    print_debug(["Debugging", "information"])
