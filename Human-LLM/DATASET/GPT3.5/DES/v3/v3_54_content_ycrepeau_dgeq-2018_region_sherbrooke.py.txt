import os
import re
import folium
import pandas as pd
import geopandas as gpd

# Function to determine section color based on candidate votes
def determine_section_color(row):
    max_vote = max(row['plq'], row['pq'], row['caq'], row['qs'])
    if row['plq'] == max_vote:
        return 'red'
    elif row['pq'] == max_vote:
        return 'blue'
    elif row['caq'] == max_vote:
        return 'lightblue'
    elif row['qs'] == max_vote:
        return 'orange'
    else:
        return 'white'

# List of coverage areas
coverage_areas = ['104', '110', '116', '120', '126', '132', '138', '144', '150', '204', '206', '210', '212', '216', '218', '220', '224', '226', '230', '232', '236', '238', '240', '244', '246', '250', '252', '256', '258', '260', '264', '502', '508', '514', '542', '548', '554', '560', '566', '554', '594', '570', '576', '660', '666', '670', '676', '708', '714', '720', '742', '754', '760', '802', '806', '810', '814', '818', '822', '826', '834', '838', '842', '850', '854', '858', '902', '906', '914', '918', '922', '926', '930', '938']

# Regular expressions to match candidate names
candidate_regex = {
    'caq': re.compile('.*C\.A\.Q\.-É\.F\.L\.$'),
    'qs': re.compile('.*Q\.S\.$'),
    'plq': re.compile('.*P\.L\.Q\./Q\.L\.P\.$'),
    'pq': re.compile('.* P\.Q\.$'),
    'pvq': re.compile('.*P\.V\.Q\./G\.P\.Q\.$'),
    'pmlq': re.compile('.* P\.M\.L\.Q\.$'),
    'pcq': re.compile('.* P.C.Q./C.P.Q.$'),
    'npdq': re.compile('.* N\.P\.D\.Q.$'),
    'bp': re.compile('.* B\.P\.$'),
    'ea': re.compile('.* É\.A\.$'),
    'pn': re.compile('.* P\.N\.$')
}

# Load sections data
sections_path = os.path.join('data', 'sherbrooke.json')
sections_data = gpd.read_file(sections_path)

# Filter sections based on coverage areas
sections_data = sections_data[sections_data['CO_CEP'].isin(coverage_areas)]

# Load and process results data
results_data = pd.DataFrame()
for area in coverage_areas:
    results_path = os.path.join('data', 'resultats-section-vote', f"{area}.csv")
    try:
        data = pd.read_csv(results_path, sep=';', encoding='latin1', dtype={'S.V.': 'str', 'Code': 'str'})
        data.dropna(subset=['S.V.'], inplace=True)
        data.rename(columns={"Code": "CO_CEP", 'Nom des Municipalités': 'NM_MUNI'}, inplace=True)

        # Rename columns based on candidate names using regex
        for candidate, regex in candidate_regex.items():
            matching_columns = [col for col in data.columns if regex.match(col)]
            if matching_columns:
                data.rename(columns={matching_columns[0]: candidate}, inplace=True)

        data.drop(labels=['Circonscription', 'Date scrutin', 'Étendue', 'Secteur', 'Regroupement', 'É.I.'], axis=1, inplace=True)
        results_data = pd.concat([results_data, data], ignore_index=True, sort=False)
    except Exception as e:
        print(f"Error processing data for coverage area {area}: {e}")

# Merge sections data with results data
merged_data = sections_data.merge(results_data, on=['CO_CEP', 'S.V.'], how='left')

# Apply color determination function to create 'couleur' column
merged_data['couleur'] = merged_data.apply(determine_section_color, axis=1)

# Generate Folium map
map_obj = folium.Map([45.5, -72], tiles='stamentoner', zoom_start=8)

# Add GeoJson layer to the map
folium.GeoJson(
    merged_data,
    tooltip=folium.features.GeoJsonTooltip(fields=['NM_CEP', 'S.V.', 'couleur', 'NM_MUNI'], labels=True),
    style_function=lambda feature: {
        'fillColor': feature['properties']['couleur'],
        'color': 'green',
        'weight': 0.25,
        'fillOpacity': 0.5,
    },
).add_to(map_obj)

# Save the map as HTML file
output_html_path = os.path.join('docs', 'region_sherbrooke.html')
map_obj.save(output_html_path)
