import GrayAPI
import colorama as clr

# Initialize colorama for colored output
clr.init(autoreset=True)

def get_user_input(prompt):
    """Prompt the user for input."""
    return input(prompt)

def print_colored_text(color, text):
    """Print text in the specified color."""
    print(color + text)

# Prompt user to input the topic to get news on
topic = get_user_input('Input the topic to get news on: ')

# Create a GrayNews object for the specified topic
news_object = GrayAPI.GrayNews(topic)

# Make a request to fetch news related to the specified topic
print_colored_text(clr.Fore.WHITE, 'Making request: ')
print_colored_text(clr.Fore.YELLOW, news_object.makeRequest())

# Print any request error, if occurred
print_colored_text(clr.Fore.WHITE, 'Request Error: ')
print_colored_text(clr.Fore.YELLOW, news_object.getRequestError())

# Print the source of the news
print_colored_text(clr.Fore.WHITE, 'Source: ')
print_colored_text(clr.Fore.YELLOW, news_object.getSource())

print()

# Get the list of articles
articles = news_object.getArticles()

# Print the number of articles fetched
print_colored_text(clr.Fore.WHITE, 'Number of Articles: ')
print_colored_text(clr.Fore.YELLOW, str(len(articles)))
print()

# Iterate through each article and print its details with colored output
for article in articles:
    print_colored_text(clr.Fore.RED, 'Title: ')
    print_colored_text(clr.Fore.GREEN, str(article.getTitle()))

    print_colored_text(clr.Fore.RED, 'Author: ')
    print_colored_text(clr.Fore.GREEN, str(article.getAuthor()))

    print_colored_text(clr.Fore.RED, 'URL: ')
    print_colored_text(clr.Fore.GREEN, str(article.getUrl()))

    print_colored_text(clr.Fore.RED, 'Description: ')
    print_colored_text(clr.Fore.GREEN, str(article.getShortDesc()))

    print_colored_text(clr.Fore.RED, 'Image: ')
    print_colored_text(clr.Fore.GREEN, str(article.getImage()))

    print_colored_text(clr.Fore.RED, 'Published: ')
    print_colored_text(clr.Fore.GREEN, str(article.getPublishedAt()))

    print()
