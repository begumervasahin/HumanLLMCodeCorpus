import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
import math
import numpy as np
import random
import os
import argparse
from data_utils import Preprocessing  # Assuming data_utils is a module containing Preprocessing class

# Define command-line arguments
parser = argparse.ArgumentParser(description='Train TransE model for knowledge graph embeddings')
parser.add_argument('--datadir', required=True, help="Path to the directory containing the data files")
args = parser.parse_args()

# Load data using Preprocessing class
all_data_file = os.path.join(args.datadir, 'data10K.csv')
preprocessor = Preprocessing()
pos_triples, entity2id, rel2id = preprocessor.get_entity_relations(all_data_file)       

# Define model parameters
embedding_size = 100
num_entity = len(entity2id)
num_rel = len(rel2id)
batch_size = 100
learning_rate = 0.001
epochs = 1000
USE_CUDA = torch.cuda.is_available()
device = torch.device("cuda" if USE_CUDA else "cpu")

# Define the TransE model
class TransE(nn.Module):
    def __init__(self, num_entity, num_rel, embedding_size):
        super(TransE, self).__init__()
        self.entity_embedding = nn.Embedding(num_entity, embedding_size)
        self.relation_embedding = nn.Embedding(num_rel, embedding_size)
        self.margin = 1.0

    def forward(self, pos_h, pos_t, pos_r, neg_h, neg_t, neg_r):     
        pos_h_emb = self.entity_embedding(pos_h)
        pos_t_emb = self.entity_embedding(pos_t)
        pos_r_emb = self.relation_embedding(pos_r)
        neg_h_emb = self.entity_embedding(neg_h)
        neg_t_emb = self.entity_embedding(neg_t)
        neg_r_emb = self.relation_embedding(neg_r)

        pos_score = torch.norm(pos_h_emb + pos_r_emb - pos_t_emb, p=2, dim=1)
        neg_score = torch.norm(neg_h_emb + neg_r_emb - neg_t_emb, p=2, dim=1)

        loss = torch.sum(F.relu(pos_score - neg_score + self.margin))
        return loss

# Prepare batch data
def prepare_batch(batch_triples):
    batch_pos_h, batch_pos_t, batch_pos_r = [], [], []
    batch_neg_h, batch_neg_t, batch_neg_r = [], [], []

    for triple in batch_triples:
        h, r, t = triple

        # Append entity and relation IDs for positive triple
        batch_pos_h.append(entity2id[h])
        batch_pos_t.append(entity2id[t])
        batch_pos_r.append(rel2id[r])

        # Randomly sample negative triple
        if np.random.random() < 0.5:
            while True:
                neg_t = random.choice(list(entity2id.values()))
                if (h, r, neg_t) not in pos_triples:
                    break
            batch_neg_h.append(entity2id[h])
            batch_neg_t.append(neg_t)
            batch_neg_r.append(rel2id[r])
        else:
            while True:
                neg_h = random.choice(list(entity2id.values()))
                if (neg_h, r, t) not in pos_triples:
                    break
            batch_neg_h.append(neg_h)
            batch_neg_t.append(entity2id[t])
            batch_neg_r.append(rel2id[r])

    # Pad batches if necessary
    pad_size = batch_size - len(batch_pos_h)
    batch_pos_h += [0] * pad_size
    batch_pos_t += [0] * pad_size
    batch_pos_r += [0] * pad_size
    batch_neg_h += [0] * pad_size
    batch_neg_t += [0] * pad_size
    batch_neg_r += [0] * pad_size

    # Convert lists to tensors
    return torch.tensor(batch_pos_h), torch.tensor(batch_pos_t), torch.tensor(batch_pos_r), \
           torch.tensor(batch_neg_h), torch.tensor(batch_neg_t), torch.tensor(batch_neg_r)

# Training function
def train(model, optimizer):
    training_loss = 0.0
    for epoch in range(epochs):
        random.shuffle(pos_triples)
        for i in range(0, len(pos_triples), batch_size):
            batch_triples = pos_triples[i:i+batch_size]
            pos_h, pos_t, pos_r, neg_h, neg_t, neg_r = prepare_batch(batch_triples)
            optimizer.zero_grad()
            loss = model(pos_h, pos_t, pos_r, neg_h, neg_t, neg_r)
            loss.backward()
            optimizer.step()
            training_loss += loss.item()
        print(f"Epoch {epoch+1}, Loss: {training_loss:.4f}")
        training_loss = 0.0

# Initialize TransE model
model = TransE(num_entity, num_rel, embedding_size).to(device)
optimizer = optim.Adam(model.parameters(), lr=learning_rate)

# Train the model
train(model, optimizer)

# Save the trained model
torch.save(model.state_dict(), 'TransE.pth')
