# Define classes for different cooking operations
class Cutter:
    def cut_vegetables(self):
        print("All vegetables are cut")

class Boiler:
    def boil_vegetables(self):
        print("All vegetables are boiled")

class Frier:
    def fry(self):
        print("All vegetables are mixed and fried.")

# Define a Cook class that orchestrates the cooking process
class Cook:
    def prepare_dish(self):
        cutter = Cutter()
        cutter.cut_vegetables()
        
        boiler = Boiler()
        boiler.boil_vegetables()
        
        frier = Frier()
        frier.fry()

if __name__ == "__main__":
    cook = Cook()
    cook.prepare_dish()
