from Sticker import Sticker
from picamera.array import PiRGBArray
from picamera import PiCamera
from time import sleep

def main():
    # Initialize PiCamera and the sticker detector
    camera = PiCamera()
    sticker_detector = Sticker()

    try:
        # Continuously capture images and detect stickers
        while True:
            # Pause for 0.5 seconds
            sleep(0.5)

            # Capture an image
            raw_capture = PiRGBArray(camera)
            camera.capture(raw_capture, format="bgr")
            image = raw_capture.array

            # Find stickers in the captured image
            sticker_found, x_position, y_position = sticker_detector.find_sticker(image, show_output=True)

            # Print the result
            print(f"Sticker Found: {sticker_found}")

    except KeyboardInterrupt:
        print("Execution stopped by the user.")

if __name__ == "__main__":
    main()
