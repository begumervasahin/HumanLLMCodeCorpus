# Function to write a number to a file
def write_number(number, filename):
    with open(filename, 'wb') as file:
        while number > 0:
            byte = number % 256
            number //= 256
            file.write(chr(byte))

# Function to convert binary to numerical representation
def binary_to_number(binary_str):
    result = 0
    for char in binary_str:
        result = (result << 8) ^ ord(char)
    return result

# Function to convert numerical to binary representation
def number_to_binary(number):
    binary_str = ''
    while number > 0:
        binary_str = chr(number % 256) + binary_str
        number //= 256
    return binary_str

# Function to find greatest common divisor
def gcd(a, b):
    while b > 0:
        a, b = b, a % b
    return a

# Function to find the next prime number
def next_prime(p):
    while p % 8 != 3 or p % 3 != 2:
        p += 1
    return next_prime_odd(p)

# Function to find the next prime number (odd)
def next_prime_odd(p):
    m_ = 3 * 5 * 7 * 11 * 13 * 17 * 19 * 23 * 29
    while True:
        while gcd(p, m_) != 1:
            p += 24
        q = (p + 1) / 4 / 3
        if pow(2, p - 1, p) != 1 or pow(3, p - 1, p) != 1 or pow(5, p - 1, p) != 1 or pow(17, p - 1, p) != 1:
            p += 24
            continue
        break
    return p

# Function to find multiplicative inverse
def inverse(b, m):
    s = 0
    t = 1
    a = m
    while b != 1:
        q = a // b
        aa = b
        b = a % b
        a = aa
        ss = t
        t = s - q * t
        s = ss
    if t < 0:
        t += m
    return t

# Function for encryption
def encrypt(m, key, prime):
    return pow(m, key, prime)

# Function for decryption
def decrypt(c, key, prime):
    key = inverse(key, prime - 1)
    return pow(c, key, prime)

# Function to convert hexadecimal/text to number
def hex_text_to_number(text):
    result = 0
    for char in text:
        if '0' <= char <= '9':
            result = (result << 4) + ord(char) - 48
        elif 'a' <= char <= 'f':
            result = (result << 4) + ord(char) - 87
    return result

# Function to convert code to number
def code_to_number(code):
    result = 0
    for char in code:
        if '0' <= char <= '9':
            result = (result << 6) + ord(char) - 48
        elif 'A' <= char <= 'Z':
            result = (result << 6) + ord(char) - 55
        elif 'a' <= char <= 'z':
            result = (result << 6) + ord(char) - 61
        elif char == "'":
            result = (result << 6) + 62
        elif char == '/':
            result = (result << 6) + 63
    return result

# Function to convert number to code
def number_to_code(number):
    result = ''
    while number > 0:
        digit = number % 64
        if digit < 10:
            result = chr(digit + 48) + result
        elif digit < 36:
            result = chr(digit + 55) + result
        elif digit < 62:
            result = chr(digit + 61) + result
        elif digit == 62:
            result = "'" + result
        elif digit == 63:
            result = '/' + result
        number //= 64
    return result

# Hash function
def hash_function(data):
    hash_digest = hashlib.sha256(data).digest()
    result = 0
    for char in hash_digest:
        result = (result << 8) ^ ord(char)
    return result % n4

# Function to generate random number
def random_number_256(seed):
    md = hashlib.sha256("***RANDOM-SEED_X***")
    md.update('large key value for generation of random number')
    md.update(seed)
    result = 0
    largestr = md.digest()
    for i in range(len(largestr)):
        result = (result << 8) ^ ord(largestr[i])
    return result

# Function to generate random number
def random_number_X(seed):
    md = hashlib.sha256("***RANDOM-SEED_X***")
    md.update('large key value for generation of random number')
    md.update(seed)
    md.update(str(time.gmtime().tm_year + time.gmtime().tm_mday))
    result = 0
    largestr = md.digest()
    for i in range(len(largestr)):
        result = (result << 8) ^ ord(largestr[i])
    return result

# Function to generate point on elliptic curve
def generate_point(x, a, b):
    while pow(x**3 + a * x + b, (prime - 1) // 2, prime) != 1:
        x += 1
    y = pow(x**3 + a * x + b, (prime + 1) // 4, prime)
    return [(x) % prime, (y) % prime]

# Function to add two points on elliptic curve
def add_points(P, Q):
    x1, x2 = P[0], Q[0]
    y1, y2 = P[1], Q[1]
    if x2 > x1:
        x2 -= prime
    if x1 == x2:
        s = ((3 * (x1**2) + a) * inverse(2 * y1, prime)) % prime
    else:
        s = ((y1 - y2) * inverse(x1 - x2, prime)) % prime
    xr = s**2 - x1 - x2
    yr = s * (x1 - xr) - y1
    return [xr % prime, yr % prime]

# Function to multiply point by scalar
def multiply_point(P, n):
    result = P
    is_first = True
    if n < 0:
        result[1] = prime - result[1]
        n = -n
    PP = result
    while n > 0:
        if n & 1:
            if is_first:
                result = PP
                is_first = False
            else:
                result = add_points(result, PP)
        PP = add_points(PP, PP)
        n >>= 1
    return result

# Function to sign message using Schnorr signature
def sign_schnorr(G, message, x):
    k = random_number_X(message)
    R = multiply_point(G, k)
    e = hash_function(str(R[0]) + message)
    return [(k - x * e) % n4, e]

# Function to sign message using ECDSA signature
def sign_ecdsa(G, message, x):
    k = random_number_X(message)
    R = multiply_point(G, k)
    hh = hash_function(message + str(R[0] % n4))
    s = (inverse(k, n4) * (hh + (R[0] % n4) * x)) % n4
    return [s, R[0] % n4]

# Function to verify ECDSA signature
def verify_ecdsa(G, message, signature, Y):
    si = inverse(signature[0], n4)
    hh = hash_function(message + str(signature[1]))
    u1 = (si * hh) % n4
    u2 = (si * signature[1]) % n4
    return add_points(multiply_point(G, u1), multiply_point(Y, u2))[0] % n4 == signature[1]
