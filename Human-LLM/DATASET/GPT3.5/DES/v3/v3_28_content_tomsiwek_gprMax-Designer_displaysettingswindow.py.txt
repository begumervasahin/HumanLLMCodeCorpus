import tkinter as tk
from tkinter import simpledialog, Label, Entry, messagebox, Frame
from tkinter.ttk import Combobox
from settings import TColours

class TDisplaySettingsWindow(simpledialog.Dialog):
    """
    Represents a pop-up window used to tweak display options.
    """

    def __init__(self, master, tickintx=1.0, tickinty=1.0, round_digits=2,
                 min_model_x=0.0, min_model_y=0.0, max_model_x=10.0,
                 max_model_y=10.0, fit_to_window=True, label_int=1.0):
        """
        Initializes object variables and calls the parent class constructor.
        """
        self.tickintx = tickintx
        self.tickinty = tickinty
        self.round_digits = round_digits
        self.min_model_x = min_model_x
        self.min_model_y = min_model_y
        self.max_model_x = max_model_x
        self.max_model_y = max_model_y
        self.fit_to_window = fit_to_window
        self.label_int = label_int
        super().__init__(master)

    def body(self, master):
        """
        Initializes widgets.
        """
        self.body_frame = Frame(self)
        self.create_label("x tick interval:", 0)
        self.create_label("y tick interval:", 1)
        self.create_label("Label decimal precision:", 2)
        self.create_label("Shapes fill", 3, column=0)
        self.create_label("Tick labels interval", 4, column=0)

        self.e1 = self.create_entry(self.tickintx, 0)
        self.e2 = self.create_entry(self.tickinty, 1)
        self.e3 = self.create_entry(self.round_digits, 2)

        fill_values = ["colour", "none"]
        self.toggle_fill = self.create_combobox(fill_values, TColours.FILL, 3)
        self.label_int_e = self.create_entry(self.label_int, 4)

        self.body_frame.pack()
        return self.e1

    def create_label(self, text, row, column=1):
        """
        Creates a label widget and places it in the body frame.
        """
        label = Label(self.body_frame, text=text)
        label.grid(row=row, column=column, sticky="ew")
        return label

    def create_entry(self, value, row):
        """
        Creates an entry widget with a default value and places it in the body frame.
        """
        entry = Entry(self.body_frame)
        entry.insert(0, str(value))
        entry.grid(row=row, column=1, sticky="ew", padx=5)
        return entry

    def create_combobox(self, values, default_value, row):
        """
        Creates a combobox widget with specified values and default selection.
        """
        combobox = Combobox(self.body_frame, values=values)
        combobox.set(default_value)
        combobox.grid(row=row, column=1, sticky="ew", padx=5)
        return combobox

    def apply(self):
        """
        Returns requested inputs.
        """
        try:
            tickintx = float(self.e1.get())
            tickinty = float(self.e2.get())
            round_digits = int(self.e3.get())
            fill = self.toggle_fill.get()
            label_int = float(self.label_int_e.get())
            self.result = tickintx, tickinty, round_digits, fill, label_int
        except Exception as message:
            messagebox.showerror("Error while changing settings!", message)

# Example usage
root = tk.Tk()
app = TDisplaySettingsWindow(root)
root.mainloop()
