import sys

# Constants and parameters
RABIN_SIGNATURE = '658ak9OxOEUcONnbDo9BfSREHPsSCCMETh15Mh6jrMwIAaJ6WkM4wP2' \
                  'IoOaa6s7NrXf1bJgcHk7Ai44bNJ2y5moKvJIhowkzO6GvydQ6AQBkR5goZP'
AF_FACTOR = 41
BF_FACTOR = 3

# Spritz hash functions
STATE_SIZE = 256
a_spritz = i_spritz = j_spritz = w_spritz = 0
s_spritz = []

# Update Spritz state
def update_spritz():
    global a_spritz, i_spritz, j_spritz, w_spritz, s_spritz
    i_spritz += w_spritz
    if i_spritz > 255:
        i_spritz -= 256
    j_spritz += s_spritz[i_spritz]
    if j_spritz > 255:
        j_spritz = s_spritz[j_spritz - 256]
    else:
        j_spritz = s_spritz[j_spritz]
    tsum = s_spritz[i_spritz] + s_spritz[j_spritz]
    s_spritz[i_spritz] = tsum - s_spritz[i_spritz]
    s_spritz[j_spritz] = tsum - s_spritz[j_spritz]

# Generate Spritz output byte
def output_spritz():
    global a_spritz, i_spritz, j_spritz, w_spritz, s_spritz
    update_spritz()
    return s_spritz[j_spritz]

# Shuffle Spritz state
def shuffle_spritz():
    global a_spritz, i_spritz, j_spritz, w_spritz, s_spritz
    for _ in range(STATE_SIZE):
        update_spritz()
    w_spritz = (w_spritz + 2) % STATE_SIZE
    a_spritz = 0

# Absorb a nibble into Spritz state
def absorb_nibble_spritz(x):
    global a_spritz, s_spritz
    if a_spritz == 240:
        shuffle_spritz()
    s_spritz[a_spritz], s_spritz[240 + x] = s_spritz[240 + x], s_spritz[a_spritz]       
    a_spritz += 1

# Absorb a byte into Spritz state
def absorb_byte_spritz(b):
    absorb_nibble_spritz(b % 16)
    absorb_nibble_spritz(b // 16)

# Squeeze bytes from Spritz state
def squeeze_spritz(out, outlen):
    global a_spritz
    if a_spritz != 0:
        shuffle_spritz()
    for _ in range(outlen):
        out.append(output_spritz())

# Compute hash using Spritz
def hash_with_spritz(input_data):
    global a_spritz, i_spritz, j_spritz, w_spritz, s_spritz
    j_spritz = i_spritz = a_spritz = 0
    w_spritz = 1
    s_spritz = list(range(STATE_SIZE))
    for byte in input_data:
        absorb_byte_spritz(ord(byte))
    hash_result = []
    squeeze_spritz(hash_result, 128)
    out = 0
    for byte in hash_result:
        out = (out << 8) + byte
    return out % RABIN_SIGNATURE

# Convert binary string to number
def binary_string_to_number(binary_string):
    result = 0
    for char in binary_string:
        result = (result << 8) ^ ord(char)
    return result

# Convert number to binary string
def number_to_binary_string(number):
    result = ''
    while number > 0:
        result = chr(number % 256) + result
        number //= 256
    return result

# Convert code string to number
def code_string_to_number(code_string):
    result = 0
    for char in code_string:
        if '0' <= char <= '9':
            result = (result << 6) + ord(char) - ord('0')
        elif 'A' <= char <= 'Z':
            result = (result << 6) + ord(char) - ord('A') + 10
        elif 'a' <= char <= 'z':
            result = (result << 6) + ord(char) - ord('a') + 36
        elif char == '@':
            result = (result << 6) + 62
        elif char == '/':
            result = (result << 6) + 63
    return result

# Pre-computed Rabin constant
RABIN_CONSTANT = code_string_to_number(RABIN_SIGNATURE)

# Read number from file
def read_number_from_file(filename):
    with open(filename, 'rb') as file:
        number = 0
        byte_data = file.read()
        for i in range(len(byte_data)):
            number = (number << 8) ^ ord(byte_data[len(byte_data) - i - 1])
    return number

# Compute hash from file content
def hash_from_file_content(filename):
    with open(filename, 'r') as file:
        return hash_with_spritz(file.read())

# Verify hash against signature
def verify_hash(signature, filename):
    a_factor = AF_FACTOR
    b_factor = BF_FACTOR
    hash_value = hash_from_file_content(filename)
    h0 = hash_value
    ha = (a_factor * h0) % RABIN_CONSTANT
    hb = (b_factor * h0) % RABIN_CONSTANT
    hab = (a_factor * b_factor * h0) % RABIN_CONSTANT
    squared_hash = (signature * signature) % RABIN_CONSTANT
    return (h0 == squared_hash) or (ha == squared_hash) or (hb == squared_hash) or (hab == squared_hash)
