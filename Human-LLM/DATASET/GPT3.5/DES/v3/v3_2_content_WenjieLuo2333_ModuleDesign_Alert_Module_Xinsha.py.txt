class Alert:
    def __init__(self):
        self.fine_turn = 0
        self.bo_flag = 0
        self.bp_flag = 0
        self.pul_flag = 0

    def update_flag(self, data, threshold_lower, threshold_upper, flag):
        """
        Update flag based on data and given threshold range.
        """
        if self.fine_turn > 20:
            flag = 0
        if threshold_lower <= data <= threshold_upper:
            flag += 1
        else:
            self.fine_turn += 1
        return flag

    def get_bo_data(self, data):
        """
        Update blood oxygen flag based on data.
        """
        self.bo_flag = self.update_flag(data, 0, 0.3, self.bo_flag)

    def get_bp_data(self, data):
        """
        Update blood pressure flag based on data.
        """
        self.bp_flag = self.update_flag(data, 60, 90, self.bp_flag)

    def get_pul_data(self, data):
        """
        Update pulse flag based on data.
        """
        self.pul_flag = self.update_flag(data, 90, 120, self.pul_flag)

    def alert_output(self):
        """
        Determine which flag is raised and return corresponding alert code.
        """
        if self.bo_flag != 0:
            return 1  # Alert for blood oxygen
        elif self.bp_flag != 0:
            return 2  # Alert for blood pressure
        elif self.pul_flag != 0:
            return 3  # Alert for pulse
        else:
            return 0  # No alert

# Example usage:
alert_instance = Alert()

# Simulate data input for blood oxygen, blood pressure, and pulse
alert_instance.get_bo_data(0.2)
alert_instance.get_bp_data(80)
alert_instance.get_pul_data(100)

# Get alert output based on the collected data
alert_output = alert_instance.alert_output()

# Print the obtained alert output
print("Alert Output:", alert_output)
