import tkinter as tk
from tkinter import simpledialog, Label, Entry, W, EW

class ModelSettingsWindow(simpledialog.Dialog):
    """
    Represents a popup window for entering model size (domain).
    """

    def __init__(self, master, maxx=0.0, maxy=0.0, dx=0.01, dy=0.01):
        """
        Initializes object variables and calls the parent class constructor.
        """
        self.maxx = maxx
        self.maxy = maxy
        self.dx = dx
        self.dy = dy
        super().__init__(master)

    def body(self, master):
        """
        Initializes widgets.
        """
        labels = ["x [m]:", "y [m]:", "\u2206x [m]:", "\u2206y [m]:"]
        default_values = [str(self.maxx), str(self.maxy), str(self.dx), str(self.dy)]

        for index, (label, default_value) in enumerate(zip(labels, default_values)):
            Label(master, text=label, anchor=W).grid(row=index, sticky=EW)

        self.entries = []
        for default_value in default_values:
            entry = Entry(master)
            entry.insert(0, default_value)
            self.entries.append(entry)

        for row, entry in enumerate(self.entries, start=0):
            entry.grid(row=row, column=1)

        return self.entries[0]

    def apply(self):
        """
        Returns the entered model size and discretization values.
        """
        modelsizex = float(self.entries[0].get())
        modelsizey = float(self.entries[1].get())
        dx = float(self.entries[2].get())
        dy = float(self.entries[3].get())
        self.result = modelsizex, modelsizey, dx, dy

# Example usage
def run_model_settings_window():
    root = tk.Tk()
    model_settings_window = ModelSettingsWindow(root)
    root.mainloop()

if __name__ == "__main__":
    run_model_settings_window()
