from odoo import models, fields

class ResRegion(models.Model):
    _name = 'res.region'
    _description = 'Region'

    # Define the name field for the region
    name = fields.Char(string='Region Name')

class ResPartner(models.Model):
    _inherit = 'res.partner'

    # Define the region field for the partner
    region = fields.Many2one('res.region', string='Region of the Client')
