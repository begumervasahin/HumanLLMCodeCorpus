# Import necessary classes and functions
from function import Protein, Matches, genetic_algorithm
from seqpred import ProteinMatcher

def perform_genetic_algorithm(protein_id, population_size=100, generations=50, tournament_size=2, mutation_rate=0.03, elitism=2):
    """
    Perform genetic algorithm for a given protein ID with specified parameters.
    """
    protein = Protein(protein_id)
    matcher = Matches(protein.match)
    genetic_algorithm(protein, matcher, population_size, generations, tournament_size, mutation_rate, elitism)

# List of protein IDs
protein_ids = ['1aba', '1bxv', '1by2', '1hyp', '1opc', '1tmy', '2acy', '2mcm', '3e0e', '3k63']

# Loop through each protein ID and perform genetic algorithm
for protein_id in protein_ids:
    perform_genetic_algorithm(protein_id)
