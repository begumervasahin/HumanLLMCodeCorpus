# Helper function to extract embedded numbers from a string
def extract_numbers(s):
    re_digits = re.compile(r'(\d+)')
    pieces = re_digits.split(s)
    pieces[1::2] = map(int, pieces[1::2])
    return pieces

# Helper function to sort a list of strings containing embedded numbers
def sort_strings(lst):
    return sorted(lst, key=extract_numbers)

# Helper function to find overlap positions between two terms
def find_overlap_positions(term1, term2):
    positions = []
    for i in set(term1) & set(term2):
        positions.append((term1.index(i), term2.index(i)))
    return positions

# Dictionary mapping non-standard amino acid codes to standard ones
TO_COMMON_AA = {
    'ASX': 'ASN', 'CSO': 'CYS', 'GLX': 'GLU', 'HIP': 'HIS', 'HSC': 'HIS',     
    'HSD': 'HIS', 'HSE': 'HIS', 'HSP': 'HIS', 'MSE': 'MET', 'SEC': 'CYS',     
    'SEP': 'SER', 'TPO': 'THR', 'PTR': 'TYR', 'XLE': 'LEU'
}

class Protein:
    def __init__(self, protein_id):
        self.protein_id = protein_id
        self.neighbors = None
        self.terms = None
        self.inverse = None
        self.overlap = None
        self.match = None
        self.graph = None
        self._get_neighbors()
        self._get_term_names()
        self._compute_inverse_and_overlap()
        self._load_match_data()
        self._build_graph()

    def _get_neighbors(self):
        # Parse PDB file to extract neighbors for each residue
        protein = parsePDB(self.protein_id + '.pdb').select('protein').copy()
        neighbors = {}
        for res in protein.iterResidues():
            cid, resnum = res.getChid(), res.getResnum()
            neighbors[cid + str(resnum)] = []
            term_frag = PATH + '/' + self.protein_id + '_' + cid + str(resnum) + '.pdb' 
            f_struct = parsePDB(term_frag)
            for fres in f_struct.iterResidues():
                fcid, fresnum = fres.getChid(), fres.getResnum()
                neighbors[cid + str(resnum)].append(fcid + str(fresnum))
        self.neighbors = neighbors

    def _get_term_names(self):
        # Extract term names from neighbor dictionary
        self.terms = list(self.neighbors.keys())

    def _compute_inverse_and_overlap(self):
        # Compute inverse and overlap dictionaries based on neighbor relationships
        inverse = {term: set() for term in self.terms}
        overlap = self.neighbors.copy()
        for i in self.neighbors:
            for j in self.neighbors[i]:
                inverse[j].add(i)
        for i in inverse:
            for j in inverse[i]:
                for k in inverse[i] - {j}:
                    overlap[j].add(k)
        self.inverse = inverse
        self.overlap = overlap

    def _load_match_data(self):
        # Load sequence matches from files and preprocess them
        match = {}
        for term in self.terms:
            file = MATCH_PATH + self.protein_id + '_' + term + '.seq'
            match[term] = np.loadtxt(file, dtype=np.str)
            if len(match[term].shape) > 1:
                match[term] = match[term][:, 1:]
            else:
                match[term] = match[term][1:]
            for code in TO_COMMON_AA:
                match[term] = np.where(match[term] == code, TO_COMMON_AA[code], match[term])
            match[term] = np.vectorize(three_to_one)(match[term])
        self.match = match

    def _build_graph(self):
        # Build a graph representing neighbor relationships and overlap positions
        G = nx.Graph()
        edges = []
        for i in self.overlap:
            for j in self.overlap[i]:
                edges.append((i, j))
        G.add_edges_from(edges)
        G.remove_edges_from(G.selfloop_edges())
        for i in G.edges():
            G.add_edge(i[0], i[1], sameAA=find_overlap_positions(self.neighbors[i[0]], self.neighbors[i[1]]))
        self.graph = G

class Matches:
    def __init__(self, seq):
        self.seq = seq
        self.frags_count = None
        self._count_frags()

    def _count_frags(self):
        # Count the number of fragments for each term
        self.frags_count = {term: len(self.seq[term]) if len(self.seq[term].shape) > 1 else 1 for term in self.seq}

    def select_frag(self, term, number):
        # Select a fragment by term and number
        if number < self.frags_count[term]:
            return self.seq[term][number] if len(self.seq[term].shape) > 1 else self.seq[term]
        else:
            return None
