import keras
from keras.layers import TimeDistributed, Input
from keras.models import Model
import matplotlib.pyplot as plt

from src.data_loading_tools.image_dataset_loaders import image_data_handler
from src.cnn_models import cnn_models_collection
from src.lstm_models import lstm_models_collection
from src.utils import model_saving_funcs

# Define dataset directory names
dataset_dir_name = "scenario_1"
train_dataset_dir_name = "training"
val_dataset_dir_name = "validation"
test_dataset_dir_name = "test"

# Initialize image data handler
img_data_src = image_data_handler.ImageDataHandler(dataset_dir_name)

# Define frame properties
frame_width = 25
frame_height = 247
channels = 1
img_colour_format = "gray_scale"

# Load dataset images and labels
train_images, train_labels = img_data_src.get_dataset_images_and_labels(train_dataset_dir_name, img_colour_format)
val_images, val_labels = img_data_src.get_dataset_images_and_labels(val_dataset_dir_name, img_colour_format)
test_images, test_labels = img_data_src.get_dataset_images_and_labels(test_dataset_dir_name, img_colour_format)

# Generate labelled frames batches
X_train, y_train = img_data_src.gen_labelled_frames_batches(train_images, train_labels, frame_width)
X_val, y_val = img_data_src.gen_labelled_frames_batches(val_images, val_labels, frame_width)
X_test, y_test = img_data_src.gen_labelled_frames_batches(test_images, test_labels, frame_width)

# Get number of classes and other parameters
num_classes = img_data_src.num_class_labels
num_frames = X_train.shape[1]
cnn_model_input_shape = (frame_height, frame_width, channels)

# Build CNN model
cnn_model = cnn_models_collection.build_simple_cnn_feature_extractor_seq_model(cnn_model_input_shape)

# Define time distributed video input tensor shape
td_video_input_shape = (num_frames, frame_height, frame_width, channels)
td_video_input_tensor = Input(shape=td_video_input_shape)

# Connect CNN model with time distributed layer
td_model_output = TimeDistributed(cnn_model)(td_video_input_tensor)

# Build LSTM model
fc_lstm_model_output = lstm_models_collection.get_simple_single_layer_fc_lstm_func_api_model_output(td_model_output, num_classes)
conv_lstm_model = Model(inputs=td_video_input_tensor, outputs=fc_lstm_model_output)     

# Compile the model
learning_rate = 0.001
optimizer = keras.optimizers.RMSprop(learning_rate=learning_rate)
conv_lstm_model.compile(loss='categorical_crossentropy', metrics=['accuracy'], optimizer=optimizer)
print(conv_lstm_model.summary())

# Train the model
batch_size = 4
num_epochs = 20
history = conv_lstm_model.fit(X_train, y_train, epochs=num_epochs, verbose=1, batch_size=batch_size, validation_data=(X_val, y_val))

# Plot training and validation accuracy
plt.plot(history.history['accuracy'], 'g', label="Training Accuracy")
plt.plot(history.history['val_accuracy'], 'r', label="Validation Accuracy")
plt.title('Convolutional LSTM Model Accuracy')
plt.xlabel('Epoch')
plt.ylabel('Accuracy')
plt.legend(loc='upper left')
plt.show()

# Plot training and validation loss
plt.plot(history.history['loss'], 'g', label="Training Loss")
plt.plot(history.history['val_loss'], 'r', label="Validation Loss")
plt.title('Convolutional LSTM Model Loss')
plt.xlabel('Epoch')
plt.ylabel('Loss')
plt.legend(loc='upper left')
plt.show()

# Save model architecture plot
model_saving_funcs.save_model_arch_plot(conv_lstm_model, "conv_lstm_model")
