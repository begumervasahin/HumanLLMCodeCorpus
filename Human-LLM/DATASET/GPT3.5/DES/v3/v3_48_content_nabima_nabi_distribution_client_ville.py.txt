from openerp import models, fields

class ResVille(models.Model):
    _name = 'res.ville'
    _description = 'Ville'

    # Field to store the name of the city
    name = fields.Char(string='City Name')

class ResPartner(models.Model):
    _inherit = 'res.partner'

    # Field to link partner to a city
    city_id = fields.Many2one('res.ville', string="City of the Client")
