import multiprocessing
import logging
import sys
from order_bot import order_bot_processing
from castle_bot import castle_bot_processing
from castle_files.bin.telethon_script import auth

# Configure logging
console_handler = logging.StreamHandler()
console_handler.setLevel(logging.INFO)

file_handler = logging.FileHandler(filename='error.log', mode='a')
file_handler.setLevel(logging.ERROR)

logging.basicConfig(
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    level=logging.INFO,
    handlers=[file_handler, console_handler]
)

def start_telethon_auth():
    """Start telethon authentication."""
    logging.info("Starting telethon auth")
    auth()
    sys.exit(0)

def start_bot_process(target):
    """Create and start a bot process."""
    bot_process = multiprocessing.Process(target=target)
    bot_process.start()
    return bot_process

if __name__ == "__main__":
    # Check for command-line arguments
    if "--auth" in sys.argv:
        start_telethon_auth()

    # Start separate processes for order bot and castle bot
    order_bot_process = start_bot_process(order_bot_processing)
    castle_bot_process = start_bot_process(castle_bot_processing)

    # Wait for the processes to complete
    try:
        order_bot_process.join()
    except KeyboardInterrupt:
        pass

    # Print a message when the processes end
    print("ended")
