import requests
from bs4 import BeautifulSoup
import json

def fetch_page(url):
    """Fetches and returns the HTML content of a webpage."""
    headers = {'Access-Control-Allow-Origin': '*'}
    response = requests.get(url, headers=headers)
    return BeautifulSoup(response.text, 'html.parser')

def extract_car_info(car):
    """Extracts and returns information of a single car."""
    car_id = car['data-key'].strip()
    car_name = car.select_one('.title').get_text(strip=True)
    car_price = car.select_one('.price').get_text(strip=True)
    car_link = f"https://www.automobile.tn{car.select_one('a')['href'].strip()}"
    car_image = car.select_one('img')['src'].strip()

    return {
        'ID': car_id,
        'Voiture': car_name,
        'Image': car_image,
        'Prix': car_price,
        'Lien': car_link
    }

def scrape_cars_info():
    """Scrapes information of cars from the website."""
    base_url = 'https://www.automobile.tn/fr/neuf/recherche'

    # Fetch the HTML content of the main page
    main_page = fetch_page(base_url)

    # Get the total number of cars
    total_count = main_page.select('.total')[0].get_text(strip=True).split()[0]

    # Define the number of cars per page and calculate the number of pages
    cars_per_page = 15
    total_pages = (int(total_count) // cars_per_page) + (1 if int(total_count) % cars_per_page != 0 else 0)

    # Initialize an empty list to store car information
    cars_info = []

    # Iterate through each page
    for page_number in range(1, total_pages + 1):
        # Construct the URL for the current page
        page_url = f'{base_url}/s=sort%21price?sort=price&page={page_number}'

        # Fetch the HTML content of the current page
        current_page = fetch_page(page_url)

        # Extract information for each car on the page
        cars = current_page.select('.vehicle')
        for car in cars:
            cars_info.append(extract_car_info(car))

    # Truncate the list to match the total count
    cars_info = cars_info[:int(total_count)]

    return cars_info

if __name__ == "__main__":
    # Scrape car information and print in JSON format
    cars_info = scrape_cars_info()
    print(json.dumps(cars_info, ensure_ascii=False, indent=2))
