def DES_decryption(data_bytes, key_bytes):
    """
    Perform DES decryption on the given data bytes using the provided key bytes.
    
    Args:
        data_bytes (list): List of bytes representing the encrypted data.
        key_bytes (list): List of bytes representing the decryption key.
    
    Returns:
        list: List of decrypted data bytes.
    """
    print("Decrypting file ...")
    print("")
    print("Key bytes: " + str(key_bytes))
    print("")
    print("Cipher data bytes: " + str(data_bytes))
    print("")

    out_data = data_bytes

    # Iterate over the DES rounds
    for rounds in range(8):
        # Iterate over the data bytes in pairs
        for i in range(0, len(data_bytes), 2):
            # Perform decryption on every other pair of data bytes
            if i % 2 == 0:
                data_bytes_slice = data_bytes[i:i+2]
                key_byte = key_bytes[7 - rounds]
                out_data[i:i+2] = iteration(data_bytes_slice, key_byte)

    return out_data

def iteration(data_bytes, key_byte):
    """
    Perform a single iteration of DES decryption.
    
    Args:
        data_bytes (list): List of bytes representing the data block.
        key_byte (int): Byte representing the decryption key for the iteration.
    
    Returns:
        list: List of decrypted data bytes.
    """
    data_L = data_bytes[0]
    data_R = data_bytes[1]
    data_L_dash = data_R ^ key_byte
    data_R_dash = data_L
    out_data = [data_L_dash, data_R_dash]
    return out_data

# Example usage:
data_bytes = [0x12, 0x34, 0x56, 0x78, 0x9a, 0xbc, 0xde, 0xf0]
key_bytes = [0x01, 0x23, 0x45, 0x67, 0x89, 0xab, 0xcd, 0xef]
decrypted_data = DES_decryption(data_bytes, key_bytes)
print("Decrypted data:", decrypted_data)
