# Global variables
bit_string_to_char = {}
char_set = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ1234567890"

# Function to create a mapping for converting bit strings to characters
def create_bit_string_to_char_map():
    global bit_string_to_char
    char_index = 0
    for bit_string in bit_set:
        bit_string_to_char[bit_string] = char_set[char_index]
        char_index = (char_index + 1) % 16

# Function to convert a bit string to a string using the mapping
def bit_string_to_string(bit_string):
    string_result = ""
    for i in range(0, 16):
        four_bit_string = bit_string[(i * 4): ((i * 4) + 4)]
        string_result += bit_string_to_char[four_bit_string]
    return string_result

# Function to convert a string to a bit string using the alphabet map
def string_to_bit_string(input_string):
    create_alphabet_map(0)
    bit_str = [alphabet_map[char] for char in input_string]
    return ''.join(bit_str)

# Function to get the round keys based on the provided key
def get_round_keys(key):
    key_list = [round6Key()[key], round5Key()[key], round4Key()[key],
                round3Key()[key], round2Key()[key], round1Key()[key]]
    return convert_to_str(key_list)

# Function to perform a single round of DES decryption
def des_round(right, left, key):
    func_output = func(left, key)
    next_left = xor(func_output, right)
    next_right = left
    return (next_right, next_left)

# Function to extract the left part of the ciphertext
def get_left_part(cipher):
    return cipher[32:]

# Function to extract the right part of the ciphertext
def get_right_part(cipher):
    return cipher[:32]

# Function to perform DES decryption
def decrypt(ciphertext, key):
    bit_str = string_to_bit_string(ciphertext)
    round_keys = get_round_keys(key)
    cipher = initial_permutation(list(bit_str))
    left = list(get_left_part(''.join(cipher)))
    right = list(get_right_part(''.join(cipher)))
    for round_no in range(6):
        (right, left) = des_round(right, left, round_keys[round_no])
    plaintext = final_permutation(left + right)
    return bit_string_to_string(''.join(plaintext))
