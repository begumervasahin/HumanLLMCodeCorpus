    def validate_references():
        missing_references = {
            "points": set(),
            "tasks": set(),
        }

        existing_points = story_points.keys()
        referenced_points = [start_point] + [dest for point in story_points.values() for dest in point["paths"].values()]
        missing_references["points"] = {point for point in referenced_points if point not in existing_points}

        existing_tasks = tasks.keys()
        referenced_tasks = [task for point in story_points.values() for task in point.get("tasks", [])]
        missing_references["tasks"] = {task for task in referenced_tasks if task not in existing_tasks}

        if missing_references["points"]:
            log_error(f"There are referenced, but unknown storypoints: {missing_references['points']}")
        if missing_references["tasks"]:
            log_error(f"There are referenced, but unknown tasks: {missing_references['tasks']}")
        else:
            print("All references in story.json found!")

    def validate_lookups():
        missing_validation_methods = {
            task["validation_method"] for task in tasks.values()

        }
        if missing_validation_methods:
