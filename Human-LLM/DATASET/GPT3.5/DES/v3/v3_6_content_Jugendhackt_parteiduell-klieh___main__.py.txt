import requests
import click
import textwrap

# Function to print wrapped text with optional indentation
def wrap_print(content, indent=0):
    global terminal_width
    print(
        textwrap.indent(
            textwrap.fill(content, (terminal_width if terminal_width else 80) - indent), " " * indent
        )
    )

@click.command()
def main():
    global terminal_width
    terminal_width = click.get_terminal_size()[0]
    
    while True:
        click.clear()
        print("Welcome to ParteiDuell \033[31mK\033[0m\033[33ml\033[0m\033[93m\033[32mi\033[0m\033[34me\033[0m\033[35mh\033[0m")
        print("")

        try:
            response = requests.get("https://solver.cloud:440/list")
            response.raise_for_status()  # Raise an exception for HTTP errors
            data = response.json()
        except requests.RequestException as e:
            print(f"Error fetching data: {e}")
            continue

        wrap_print(data[0]["these"])
        print()
        wrap_print(data[0]["statement"])
        print()
        
        possible_answers = data[0]["possibleAnswers"]
        options = ", ".join(f"{i}: {answer}" for i, answer in enumerate(possible_answers))
        print(options)
        
        user_input = input("> ")

        try:
            user_input = int(user_input)
            selected_answer = possible_answers[user_input]
        except (ValueError, IndexError):
            print("Invalid input. Please enter a valid option.")
            continue
        
        print()

        if data[0]["answer"] == selected_answer:
            print("Correct!")
        else:
            print(f"Wrong, this statement was from {data[0]['answer']}")
            print()
            print(f"The party {selected_answer} made the following statement:\n")
            wrap_print(possible_answers[selected_answer], indent=4)

        print()
        click.pause()

if __name__ == "__main__":
    main()
