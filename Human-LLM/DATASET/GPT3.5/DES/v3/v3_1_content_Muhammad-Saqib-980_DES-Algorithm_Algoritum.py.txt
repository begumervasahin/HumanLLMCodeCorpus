# Function to convert an integer to a binary list with a minimum length
def int_to_binary_list(num, min_len=8):
    binary_list = [int(digit) for digit in format(num, '08b')]
    if len(binary_list) < min_len:
        binary_list = [0] * (min_len - len(binary_list)) + binary_list
    return binary_list

# Function to convert a hexadecimal string to a binary list
def hex_to_binary_list(hex_str):
    binary_list = []
    byte_array = bytes.fromhex(hex_str)
    for byte in byte_array:
        binary_list.extend(int_to_binary_list(byte))
    return binary_list

# Function to convert a text string to a binary list
def text_to_binary_list(text_str):
    binary_list = []
    byte_array = bytearray(text_str, encoding='utf-8')
    for byte in byte_array:
        binary_list.extend(int_to_binary_list(byte))
    return binary_list
