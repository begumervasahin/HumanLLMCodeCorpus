import plotly
from plotly.offline import iplot, init_notebook_mode

# Initialize plotly notebook mode
init_notebook_mode(connected=True)

def trace(data, mode='markers', name="data"):
    """
    Create a trace from data.

    Parameters:
        data (list of dicts): List of dictionaries containing 'x' and 'y' values.
        mode (str): Mode of the trace, defaults to 'markers'.
        name (str): Name of the trace, defaults to "data".

    Returns:
        dict: Trace dictionary containing 'x', 'y', 'mode', and 'name' keys.
    """
    x_values = [point['x'] for point in data]
    y_values = [point['y'] for point in data]
    return {'x': x_values, 'y': y_values, 'mode': mode, 'name': name}

def line_function_trace(line_function, x_values, mode='line', name='line function'):    
    """
    Create a trace for a line function.

    Parameters:
        line_function (function): Line function.
        x_values (list): List of x values.
        mode (str): Mode of the trace, defaults to 'line'.
        name (str): Name of the trace, defaults to 'line function'.

    Returns:
        dict: Trace dictionary containing 'x', 'y', 'mode', and 'name' keys.
    """
    values = line_function_data(line_function, x_values)
    values.update({'mode': mode, 'name': name})
    return values

def line_function_data(line_function, x_values):
    """
    Generate data for a line function.

    Parameters:
        line_function (function): Line function.
        x_values (list): List of x values.

    Returns:
        dict: Dictionary containing 'x' and 'y' values.
    """
    y_values = [line_function(x) for x in x_values]
    return {'x': x_values, 'y': y_values}

def m_b_data(m, b, x_values):
    """
    Generate data for a linear function (y = mx + b).

    Parameters:
        m (float): Slope.
        b (float): Intercept.
        x_values (list): List of x values.

    Returns:
        dict: Dictionary containing 'x' and 'y' values.
    """
    y_values = [m*x + b for x in x_values]
    return {'x': x_values, 'y': y_values}

def m_b_trace(m, b, x_values, mode='line', name='line function'):
    """
    Create a trace for a linear function (y = mx + b).

    Parameters:
        m (float): Slope.
        b (float): Intercept.
        x_values (list): List of x values.
        mode (str): Mode of the trace, defaults to 'line'.
        name (str): Name of the trace, defaults to 'line function'.

    Returns:
        dict: Trace dictionary containing 'x', 'y', 'mode', and 'name' keys.
    """
    values = m_b_data(m, b, x_values)
    values.update({'mode': mode, 'name': name})
    return values

def plot(traces, layout={}):
    """
    Plot traces.

    Parameters:
        traces (list): List of traces.
        layout (dict): Layout options.

    Raises:
        TypeError: If traces is not a list.
    """
    if not isinstance(traces, list):
        raise TypeError('First argument must be a list. Instead is', traces)
    plotly.offline.iplot({'data': traces, 'layout': layout})

def build_layout(x_range=None, y_range=None, options={}):
    """
    Build layout options.

    Parameters:
        x_range (list): List containing x-axis range.
        y_range (list): List containing y-axis range.
        options (dict): Additional layout options.

    Returns:
        dict: Layout options.
    """
    layout = {}
    if isinstance(x_range, list):
        layout.update({'xaxis': {'range': x_range}})
    if isinstance(y_range, list):
        layout.update({'yaxis': {'range': y_range}})
    layout.update(options)
    return layout

def trace_values(x_values, y_values, mode='markers', name="data", text=[]):
    """
    Create a trace from x and y values.

    Parameters:
        x_values (list): List of x values.
        y_values (list): List of y values.
        mode (str): Mode of the trace, defaults to 'markers'.
        name (str): Name of the trace, defaults to 'data'.
        text (list): List of text values, defaults to empty list.

    Returns:
        dict: Trace dictionary containing 'x', 'y', 'mode', 'name', and 'text' keys.
    """
    return {'x': x_values, 'y': y_values, 'mode': mode, 'name': name, 'text': text}
