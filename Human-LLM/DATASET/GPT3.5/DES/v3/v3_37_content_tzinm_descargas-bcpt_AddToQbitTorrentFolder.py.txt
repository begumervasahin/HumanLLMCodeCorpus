import logging
import os
import urllib.request
import zipfile
from telegram.ext import Updater, CommandHandler, MessageHandler, Filters

# Configure logging
logging.basicConfig(format='%(asctime)s - %(name)s - %(levelname)s - %(message)s', level=logging.INFO)
logger = logging.getLogger(__name__)

# Function to download files
def download_file(url, path, filename):
    opener = urllib.request.build_opener()
    opener.addheaders = [('User-agent', 'Mozilla/5.0')]
    urllib.request.install_opener(opener)
    urllib.request.urlretrieve(url, os.path.join(path, filename))

# Function to list files in a directory
def list_files(path=os.getcwd()):
    return [arch.name for arch in os.scandir(path) if arch.is_file()]

# Function to handle downloading and processing of files
def handle_files(update, context):
    allowed_members = []

    # Load allowed members from environment variables
    for i in range(1, 100):  # Assuming a maximum of 100 allowed members
        try:
            member_id = int(os.environ[f'usuario{i}'])
            allowed_members.append(member_id)
        except KeyError:
            break

    for i in range(1, 100):
        try:
            member_id = int(os.environ[f'grupo{i}'])
            allowed_members.append(member_id)
        except KeyError:
            break

    if update.message.chat_id not in allowed_members:
        update.message.reply_text("No tienes permisos suficientes para utilizar el bot.")
        return

    ruta = '/home/descargas/'
    tmp = '/zip/'

    if update.message.document:
        file_name = update.message.document.file_name
        file_id = update.message.document.file_id

        archivo = context.bot.get_file(file_id)

        if file_name.endswith('.zip'):
            download_file(archivo.file_path, tmp, file_name)
            with zipfile.ZipFile(tmp + file_name, "r") as zf:
                for torrents in zf.namelist():
                    if os.path.dirname(torrents) == '' and torrents.endswith('.torrent'):
                        zf.extract(torrents, ruta)
            os.remove(tmp + file_name)
            update.message.reply_text(f"Los torrents en el archivo <b>{file_name}</b> han sido descomprimidos y agregados al directorio correspondiente.", parse_mode="HTML")   

        elif file_name.endswith('.torrent'):
            download_file(archivo.file_path, ruta, file_name)
            update.message.reply_text(f"El archivo <b>{file_name}</b> ha sido agregado correctamente al directorio correspondiente.", parse_mode="HTML")

# Function to handle errors
def handle_error(update, context):
    logger.warning('Update "%s" caused error "%s"' % (update, context.error))

# Main function to start the bot
def main():
    # Initialize the bot updater with the provided token
    updater = Updater(os.environ['TOKEN'], use_context=True)
    dp = updater.dispatcher

    # Register message handler for documents
    dp.add_handler(MessageHandler(Filters.document, handle_files))

    # Register error handler
    dp.add_error_handler(handle_error)

    # Start polling for updates
    updater.start_polling(clean=True)
    updater.idle()

# Run the main function if the script is executed directly
if __name__ == '__main__':
    main()
