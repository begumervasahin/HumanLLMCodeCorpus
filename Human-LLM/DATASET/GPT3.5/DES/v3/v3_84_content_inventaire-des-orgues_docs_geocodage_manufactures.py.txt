import json
import requests
import geojson

def geocode_address(address):
    """
    Geocode an address using the api-adresse.data.gouv.fr API.

    Args:
        address (str): The address to geocode.

    Returns:
        tuple: A tuple containing latitude and longitude if available, otherwise (None, None).
    """
    base_url = "https://api-adresse.data.gouv.fr/search/?q="
    formatted_address = address.replace(" ", "+")
    response = requests.get(base_url + formatted_address)
    feature_collection = geojson.loads(response.text)

    if 'features' in feature_collection and feature_collection['features']:
        coordinates = feature_collection['features'][0]['geometry']['coordinates']
        latitude, longitude = coordinates[1], coordinates[0]  # Latitude, Longitude
        return latitude, longitude
    else:
        return None, None

# Load data from datas.json
with open('datas.json') as json_file:
    data_dict = json.load(json_file)

# Geocode each address and update data_dict with latitude and longitude
for facteur in data_dict:
    address = facteur['adresse']
    latitude, longitude = geocode_address(address)
    facteur['latitude'] = latitude
    facteur['longitude'] = longitude

# Write the updated data to facteurs_localisation.json
with open('facteurs_localisation.json', 'w') as output_file:
    json.dump(data_dict, output_file)
