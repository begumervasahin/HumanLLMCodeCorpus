    Lroul = []

for i in range(len(Lroul)):

def rlmnt(date, kh):
    """Retourne les indices de entete"""
    j = -1
    liste = []
    for i in range(len(Lroul)):
        if date in Lroul[i][0]:
            j = i
    if j < 0:
        print("Date introuvable !")
        return None
    else:
        for k in range(len(Lroul[j])):
            try:
                Lroul[j][k].index(str(kh))
                liste.append(k)
            except ValueError:
                pass
    return liste
