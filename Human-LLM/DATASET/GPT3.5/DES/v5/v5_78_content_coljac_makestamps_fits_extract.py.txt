import os
import sys
import h5py as h5
import astropy.io.fits as pyfits

def read_file_lines(filename, skip_comments=True):
    """Read lines from a file, optionally skipping comments."""
    with open(filename, "r") as f:
        lines = f.readlines()
    if skip_comments:
        return [l.strip() for l in lines if not l.startswith("#")]
    else:
        return [l.strip() for l in lines]

def progress_bar(current, total, width=40, show=True, message=None, stderr=False):   
    """Display a progress bar."""
    percent = float(current) / float(total)
    length = int(width * percent)
    if show:
        count = f" ({current}/{total})    "
    else:
        count = ""
    if message:
        count += message
    outstream = sys.stderr if stderr else sys.stdout
    outstream.write(("\r[" + ("#" * length) + (" " * (width - length)) + f"] {percent * 100:.0f}%" + count))
    outstream.flush()

def extract_objects(datastore, objids, outdir):
    """Extract objects from the datastore."""
    if objids is None:
        extract_all(datastore, outdir)
    else:
        extract_some(objids, datastore, outdir)

def extract_some(ids, datastore, outdir):
    """Extract specific objects from the datastore."""
    total = len(ids)
    done = 0
    progress_bar(done, total)
    with h5.File(datastore, "r") as f:
        ds = f['/stamps/']
        for tile in ds:
            data = ds[f"{tile}/data"]
            headers = ds[f"{tile}/header"]
            objids = ds[f"{tile}/catalog"]
            N = data.shape[0]
            for i in range(N):
                objid = objids[i].strip()
                if objid in ids:
                    done += 1
                    progress_bar(done, total)
                    heads = headers[i]
                    for b in range(5):
                        band_name = "grizY"[b]
                        d = data[i, :, :, b]
                        head = pyfits.Header.fromstring(heads[b].strip())
                        filename = os.path.join(outdir, f"{objid}_{band_name}.fits") 
                        outfits = pyfits.PrimaryHDU(data=d, header=head)
                        outfits.writeto(filename, overwrite=True)

def extract_all(datastore, outdir):
    """Extract all objects from the datastore."""
    with h5.File(datastore, "r") as f:
        ds = f['/stamps/']
        for tile in ds:
            data = ds[f"{tile}/data"]
            headers = ds[f"{tile}/header"]
            objids = ds[f"{tile}/catalog"]
            N = data.shape[0]
            for i in range(N):
                objid = objids[i].strip()
                heads = headers[i]
                for b in range(5):
                    band_name = "grizY"[b]
                    d = data[i, :, :, b]
                    head = pyfits.Header.fromstring(heads[b].strip())
                    filename = os.path.join(outdir, f"{objid}_{band_name}.fits")     
                    outfits = pyfits.PrimaryHDU(data=d, header=head)
                    outfits.writeto(filename, overwrite=True)

if __name__ == "__main__":
    if len(sys.argv) < 3:
        print("Usage: python script.py <datastore> <dest_dir> [objids]")
        sys.exit(0)

    datastore = sys.argv[1]
    outdir = sys.argv[2]
    inputs = sys.argv[3] if len(sys.argv) == 4 else None

    if not os.path.exists(outdir):
        print("Output directory doesn't exist.")
        sys.exit(0)

    if inputs and os.path.exists(inputs):
        objids = read_file_lines(inputs)
    elif inputs:
        objids = inputs.split(",")
    else:
        objids = None

    extract_objects(datastore, objids, outdir)
