def read_csv(file_path):
    """Reads the CSV file and returns a list of rows."""
    rows = []  # List to store the rows from the CSV file
    with open(file_path, 'r') as csv_file:
        # Read the header line and split it into columns
        header = csv_file.readline().strip().split('\t')

        # Read each line of the CSV file, split it into columns, and append it to 'rows'
        for line in csv_file:
            row = line.strip().split('\t')
            rows.append(row)
    return header, rows

def print_row_indices(rows):
    """Prints the indices of each row along with the first element."""
    for i, row in enumerate(rows):
        print(f"{i} {row[0]}")
        # Split each element of the row by commas
        for j in range(len(row)):
            row[j] = row[j].split(',')

def find_indices_by_date_and_kh(date, kh, rows):
    """Returns the indices of columns where 'date' and 'kh' are found in the rows."""
    indices = []  # List to store the indices of matching columns
    for i, row in enumerate(rows):
        if date in row[0]:  # Check if 'date' is found in the first column of the row
            for k, value in enumerate(row):
                if str(kh) in value:  # Check if 'kh' is found in any column of the row
                    indices.append(k)  # Append the index of the matching column
    return indices

# Example usage
file_path = "resources/roulement.csv"
header, csv_data = read_csv(file_path)
print_row_indices(csv_data)

# Assuming 'date' and 'kh' are provided
date = "2024-04-06"
kh = 12345
matching_indices = find_indices_by_date_and_kh(date, kh, csv_data)

if not matching_indices:
    print("No matching indices found.")
else:
    for index in matching_indices:
        print(header[index])  # Assuming the header is in the first row of csv_data
