import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# Set figure size for plots
plt.rcParams['figure.figsize'] = (5.0, 5.0)

def plot_scatter(X, Y):
    plt.scatter(X, Y)

def calculate_least_squares_regression(X, Y):
    mean_X = np.mean(X)
    mean_Y = np.mean(Y)
    deviation_X = X - mean_X
    deviation_Y = Y - mean_Y
    squared_deviation_X = deviation_X * deviation_X
    m = np.sum(deviation_X * deviation_Y) / np.sum(squared_deviation_X)
    c = mean_Y - m * mean_X
    return m, c

def plot_least_squares_regression(X, m, c):
    plt.plot(X, m*X + c, color='red')

def calculate_gradient_descent_regression(X, Y, learning_rate=0.0001, epochs=1000):
    m = 0
    c = 0
    n = float(len(X))
    for i in range(epochs):
        Y_pred = m*X + c
        D_m = (-2/n) * sum(X * (Y - Y_pred))
        D_c = (-2/n) * sum(Y - Y_pred)
        m -= learning_rate * D_m
        c -= learning_rate * D_c
    return m, c

def plot_gradient_descent_regression(X, m, c):
    plt.plot([min(X), max(X)], [min(m*X + c), max(m*X + c)], color='red')

def show_plot():
    plt.show()

def print_regression_parameters(m, c):
    print("Slope (m):", m)
    print("Intercept (c):", c)

# Read data from CSV file
data = pd.read_csv('data.csv')

# Extract X and Y values
X = data.iloc[:, 0]
Y = data.iloc[:, 1]

# Scatter plot of the data
plot_scatter(X, Y)

# Perform least squares regression
m_ls, c_ls = calculate_least_squares_regression(X, Y)
plot_least_squares_regression(X, m_ls, c_ls)

# Perform gradient descent regression
m_gd, c_gd = calculate_gradient_descent_regression(X, Y)
plot_gradient_descent_regression(X, m_gd, c_gd)

# Show plots
show_plot()

# Print regression parameters obtained from gradient descent
print_regression_parameters(m_gd, c_gd)
