import gplaylist as gp
import os

# Display the help message
help_message = '''
show : See the list of your favorite artists
add : Add a name to your favorite artist list
rem : Remove a name from your favorite artist list
quit : Quit the program
clr : Clear the screen
help : Show this menu
'''
print(help_message)

# Main loop
while True:
    # Get user command
    command = input('..> ').lower()

    # Execute commands based on user input
    if command == 'show':
        # Show the list of favorite artists
        artists = gp.GetPlayList().getNames()
        for artist in artists:
            print(artist)

    elif command == 'add':
        # Add a new favorite artist
        name = input('Name of Artist to add: ')
        result = gp.AddPlayList(name)
        if result.hasErr():
            print(result.getError())
        else:
            print('Successfully added the artist')

    elif command == 'rem':
        # Remove an artist from the favorite list
        name = input('Name of Artist to remove: ')
        result = gp.RemoveArtist(name)
        if result.hasErr():
            print(result.getError())
        elif result.found:
            print('Successfully removed the artist')
        else:
            print('Artist not found in the list')

    elif command == 'quit':
        # Exit the program
        break

    elif command == 'clr':
        # Clear the screen
        if os.name == 'nt':
            os.system('cls')
        elif os.name == 'posix':
            os.system('clear')
        else:
            print('Sorry, Clearing the screen is not supported for your OS')

    elif command == 'help':
        # Display the help message
        print(help_message)
