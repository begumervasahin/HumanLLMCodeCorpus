# Import necessary modules
from fbs.cmdline import command
from os.path import dirname
import subprocess
import fbs.cmdline

# Get the directory of the current script
project_dir = dirname(__file__)

# Define a custom command to build UI files
@command
def build_ui_files():
    """
    Command to build UI files.
    """
    # Execute the shell script to build UI files
    subprocess.call([project_dir + "/src/scripts/build_ui_files.sh"])

# Main entry point of the script
if __name__ == '__main__':
    # Execute the fbs command line tool with the project directory
    fbs.cmdline.main(project_dir)
