from des import *

from generate_textstrings import alphabet_map, create_alphabet_map, bit_set

# Global variables
reverse_map = {}
input_set = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ1234567890"

# Create a reverse map for converting bit strings to characters
def create_reverse_map():
    global reverse_map
    index = 0
    for bit_str in bit_set:
        reverse_map[bit_str] = input_set[index]
        index = (index + 1) % 16

# Convert a bit string to a string using the reverse map
def convert_bitstr_to_string(ciphered_bits):
    result_string = ""
    for i in range(0, 16):
        four_bit_str = ciphered_bits[(i * 4): ((i * 4) + 4)]
        result_string += reverse_map[four_bit_str]
    return result_string

# Convert a string to a bit string using the alphabet map
def convert_string_to_bitstring(ciphertext):
    create_alphabet_map(0)
    bit_str_list = [alphabet_map[char] for char in ciphertext]
    return ''.join(bit_str_list)

# Get the round keys based on the provided key and round number
def get_round_keys(key, round_no):
    key_list = [
        round6Key()[round_no],
        round5Key()[round_no],
        round4Key()[round_no],
        round3Key()[round_no],
        round2Key()[round_no],
        round1Key()[round_no]
    ]
    return convert_to_str(key_list)

# Perform one round of DES decryption
def perform_des_round(right, left, key):
    expanded_output = expansion(left)
    sbox_input = xor(key, expanded_output)
    sbox_output = substitute(sbox_input)
    permuted_output = permute(sbox_output)
    next_left = xor(permuted_output, right)
    next_right = left
    return (next_right, next_left)

# Extract the left part of the ciphertext
def get_left_part(cipher):
    return cipher[32:]

# Extract the right part of the ciphertext
def get_right_part(cipher):
    return cipher[:32]

# Perform DES decryption
def decryption(ciphertext, key):
    cipher_bit_str = convert_string_to_bitstring(ciphertext)
    create_reverse_map()
    round_keys = get_round_keys(key)
    cipher = initial_permutation(list(cipher_bit_str))
    left = list(get_left_part(''.join(cipher)))
    right = list(get_right_part(''.join(cipher)))
    for round_no in range(6):
        (right, left) = perform_des_round(right, left, round_keys[round_no])
    plaintext = final_permutation(left + right)
    return convert_bitstr_to_string(''.join(plaintext))
