import pygame
import numpy as np
import cv2
import os
from configparser import ConfigParser
import gui

def find_available_cameras():
    available_cams = []
    for cam_index in range(10):
        cap = cv2.VideoCapture(cam_index)
        if cap is not None and cap.isOpened():
            available_cams.append(cam_index)
            cap.release()
    return available_cams

def apply_transparency(frame):
    sensitivity = config.getint('main', 'sensitivity')
    green_color = np.array([0, 255, 0])
    mask = np.sum((frame - green_color) ** 2, axis=-1) < sensitivity ** 2
    frame[mask] = green_color

def update_settings():
    global camera
    selected_index = int(menu_obj[0].get())
    if selected_index != config.getint('main', 'DeviceIndex'):
        config.set('main', 'DeviceIndex', str(selected_index))
        camera.release()
        camera = cv2.VideoCapture(selected_index)
    x_offset = menu_obj[1].get()
    y_offset = menu_obj[2].get()
    screen_width, screen_height, video_width, video_height = geometry
    x_pos = screen_width // 3 * 2 + x_offset
    y_pos = screen_height - video_height + y_offset
    config.set('main', 'X_AxisOffset', str(x_offset))
    config.set('main', 'Y_AxisOffset', str(y_offset))
    gui.update_location(hwnd, (x_pos, y_pos), geometry)
    sensitivity = menu_obj[3].get()
    config.set('main', 'sensitivity', str(sensitivity))
    save_config(config)

def read_configuration():
    config_dir = os.path.expandvars(r'%LOCALAPPDATA%\FacefigDesktopWidget')
    config_path = os.path.join(config_dir, 'config.ini')
    config = ConfigParser()
    if not os.path.exists(config_dir):
        os.makedirs(config_dir)
    if not os.path.exists(config_path):
        config['main'] = {
            'DeviceIndex': '1',
            'X_AxisOffset': '0',
            'Y_AxisOffset': '0',
            'sensitivity': '150'
        }
        save_config(config)
    else:
        config.read(config_path)
    return config

def save_config(config):
    config_path = os.path.join(os.path.expandvars(r'%LOCALAPPDATA%\FacefigDesktopWidget'), 'config.ini')
    with open(config_path, 'w') as f:
        config.write(f)

def on_close():
    global running
    running = False
    pygame.quit()

if __name__ == '__main__':
    camera_list = find_available_cameras()
    config = read_configuration()
    camera = cv2.VideoCapture(config.getint('main', 'DeviceIndex'))
    geometry = gui.get_geometry(camera)
    hwnd, display = gui.main_window_setup(config, geometry)
    menu, menu_obj = gui.menu_setup(config, camera_list, update_settings, on_close)     
    running = True
    while running:
        _, frame = camera.read()
        frame = cv2.rotate(frame, cv2.ROTATE_90_COUNTERCLOCKWISE)
        rgb_frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
        apply_transparency(rgb_frame)
        try:
            pygame.surfarray.blit_array(display, rgb_frame)
        except ValueError:
            pass
        pygame.display.flip()
        menu.update()
        cv2.waitKey(1)
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                pygame.quit()
                camera.release()
                running = False
