# Utility Functions
def base64_to_num(base64_str):
    return int.from_bytes(hashlib.sha512(base64_str.encode()).digest(), 'big')

def num_to_base64(number):
    return hashlib.sha512(number.to_bytes((number.bit_length() + 7) // 8, 'big')).digest().decode()

def sha512_hash(data):
    return hashlib.sha512(data.encode('utf-8')).hexdigest()

def generate_random_number():
    seed = "RANDOM-SEED" + str(random.random())
    return int.from_bytes(hashlib.sha512(seed.encode('utf-8')).digest(), 'big')

# RSA Encryption/Decryption
def encrypt(message):
    message_num = int.from_bytes(message.encode('utf-8'), 'big')
    return pow(message_num, PUBLIC_EXPONENT, N_RSA)

def decrypt(ciphertext):
    plaintext_num = pow(ciphertext, PRIVATE_EXPONENT, N_RSA)
    return plaintext_num.to_bytes((plaintext_num.bit_length() + 7) // 8, 'big').decode('utf-8')

# Digital Signature
def sign_file(filename):
    with open(filename, 'r') as file:
        file_content = file.read()
    hash_num = int.from_bytes(sha512_hash(file_content).encode('utf-8'), 'big')
    signature = pow(hash_num, PRIVATE_EXPONENT, N_RSA)
    return num_to_base64(signature)

def verify_signature(filename, signature):
    with open(filename, 'r') as file:
        file_content = file.read()
    hash_num = int.from_bytes(sha512_hash(file_content).encode('utf-8'), 'big')
    signature_num = base64_to_num(signature)
    verification_result = pow(signature_num, PUBLIC_EXPONENT, N_RSA)
    return hash_num == verification_result
