# Helper Functions

def write_number_to_file(number, filename):
    """Write a number to a binary file."""
    with open(filename, 'wb') as f:
        while number > 0:
            byte = number % 256
            number //= 256
            f.write(chr(byte))

def binary_to_number(x):
    """Convert binary string to a number."""
    res = 0
    for c in x:
        res = (res << 8) ^ ord(c)
    return res

def number_to_binary(x):
    """Convert a number to binary string."""
    res = ''
    while x > 0:
        res = chr(x % 256) + res
        x //= 256
    return res

def extended_gcd(a, b):
    """Extended Euclidean algorithm."""
    s, old_s = 0, 1
    t, old_t = 1, 0
    while b != 0:
        quotient, remainder = divmod(a, b)
        a, b = b, remainder
        old_s, s = s, old_s - quotient * s
        old_t, t = t, old_t - quotient * t
    return old_s, old_t

def mod_inverse(b, m):
    """Calculate the modular multiplicative inverse."""
    inv, _ = extended_gcd(b, m)
    return (inv + m) % m

def hash_to_integer(x):
    """Hash a string and convert it to an integer."""
    dx1 = hashlib.sha256(x).digest()
    res = 0
    for cx in dx1:
        res = (res << 8) ^ cx
    return res

def generate_random_integer(m):
    """Generate a random integer using a seed."""
    md = hashlib.sha256("***RANDOM-SEED_X***")
    md.update('large key value for generation of random number')
    md.update(m)
    result = 0
    largestr = md.digest()
    for i in range(len(largestr)):
        result = (result << 8) ^ ord(largestr[i])
    return result

def generate_random_integer_with_time(m):
    """Generate a random integer using a seed with current time."""
    md = hashlib.sha256("***RANDOM-SEED_X***")
    md.update('large key value for generation of random number')
    md.update(m)
    md.update(str(time.gmtime().tm_year + time.gmtime().tm_mday))
    result = 0
    largestr = md.digest()
    for i in range(len(largestr)):
        result = (result << 8) ^ ord(largestr[i])
    return result

# Elliptic Curve Functions

def next_prime(p):
    """Find the next prime number."""
    while p % 8 != 3:
        p += 1
    return next_prime_odd(p)

def next_prime_odd(p):
    """Find the next odd prime number."""
    m_ = 3 * 5 * 7 * 11 * 13 * 17 * 19 * 23 * 29
    while True:
        while gcd(p, m_) != 1:
            p += 8
        q = (p + 1) // 4
        if (pow(2, p - 1, p) != 1 or pow(2, q - 1, q) != 1 or
            pow(3, p - 1, p) != 1 or pow(3, q - 1, q) != 1 or
            pow(5, p - 1, p) != 1 or pow(5, q - 1, q) != 1 or
            pow(17, p - 1, p) != 1 or pow(17, q - 1, q) != 1):
            p += 8
            continue
        break
    return p

def generate_base_point(x, a, b):
    """Generate a base point on the elliptic curve."""
    if (4 * a * a * a + 27 * b * b) % prime == 0:
        b += 1
    while pow(c * x ** 3 + a * x + b, (prime - 1) // 2, prime) != 1:
        x += 1
    y = pow(c * x ** 3 + a * x + b, (prime + 1) // 4, prime)
    return [x % prime, y % prime]

def double_point(P):
    """Calculate the double of a point on the elliptic curve."""
    x, y = P[0], P[1]
    s = ((3 * c * (x ** 2) + a) * mod_inverse(2 * y, prime)) % prime
    xr = (c * s ** 2 - 2 * x) % prime
    yr = (-y + s * (x - xr)) % prime
    return [xr, yr]

def add_points(P, Q):
    """Add two points on the elliptic curve."""
    if P == Q:
        return double_point(P)
    x1, x2 = P[0], Q[0]
    y1, y2 = P[1], Q[1]
    while x1 < x2:
        x1 += prime
    while y1 < y2:
        y1 += prime
    s = ((y1 - y2) * mod_inverse(x1 - x2, prime)) % prime
    xr = cinv * s ** 2 - x1 - x2
    yr = s * (x1 - xr) - y1
    return [xr % prime, yr % prime]

def scalar_multiply(P, n):
    """Multiply a point on the elliptic curve by a scalar."""
    result_point = P
    if n < 0:
        result_point[1] = prime - result_point[1]
        n = -n
    bit_size = 20
    while 2 ** bit_size < n:
        bit_size += 1
    base_point = result_point
    for bit in range(bit_size + 1):
        if n & (1 << bit) != 0:
            if result_point == base_point:
                result_point = base_point
            else:
                result_point = add_points(result_point, base_point)
        base_point = double_point(base_point)
    return result_point

def schnorr_signature(G, m, x):
    """Generate a Schnorr signature."""
    k = generate_random_integer_with_time(m)
    R = scalar_multiply(G, k)
    e = hash_to_integer(str(R[0]) + m)
    return [(k - x * e) % n4, e]

def ecdsa_signature(G, m, x):
    """Generate an ECDSA signature."""
    hh = hash_to_integer(m)
    k = 2 * generate_random_integer_with_time(m) + 1
    R = scalar_multiply(G, k)
    s = (mod_inverse(k, n4) * (hh + R[0] * x)) % n4
    return [s, R[0]]

def ecdsa_verification(G, m, S, Y):
    """Verify an ECDSA signature."""
    si = mod_inverse(S[0], n4)
    hh = hash_to_integer(m)
    u1 = (si * hash_to_integer(m)) % n4
    u2 = (si * S[1]) % n4
    return add_points(scalar_multiply(G, u1), scalar_multiply(Y, u2))[0] == S[1]

# Constants
prime = 2 ** 256 - 2 ** 224 + 2 ** 192 + 2 ** 96 - 1
n4 = (prime + 1) // 4
a = prime - 31
c = 17
cinv = mod_inverse(c, prime)

# Elliptic Curve Base Point
base_point_x = a + 17
base_point = generate_base_point(base_point_x, a, 0)
base_point = scalar_multiply(base_point, 4)

# Read the message from a file
with open(sys.argv[1], 'r') as f:
    message = f.read()

# Generate public key and signature
private_key = 2 * generate_random_integer(sys.argv[1]) + 1
public_key = scalar_multiply(base_point, private_key)
write_number_to_file(public_key[0], 'y0')
write_number_to_file(public_key[1], 'y1')
signature = schnorr_signature(base_point, message, private_key)
write_number_to_file(signature[0], 's0')
write_number_to_file(signature[1], 's1')

# Verify the signature
verification_result = (hash_to_integer(str(add_points(scalar_multiply(base_point, signature[0]),
                                                      scalar_multiply(public_key, signature[1]))[0]) + message)
                       == signature[1])
print("Test Schnorr ", verification_result)
