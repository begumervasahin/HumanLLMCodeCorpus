import os
import re

def concat_files(readme_path):
    """
    Concatenate files specified in README.md using include and end tags.

    Parameters:
    - readme_path: str, path to the README.md file
    """
    result_document = ""
    is_include_section = False

    with open(readme_path, "r+", encoding="utf-8") as readme_file:
        for line in readme_file:
            if not is_include_section:
                result_document += line

            include_match = re.search(r"<!--\s*include\s+(.+?)\s*-->", line)
            if include_match and not is_include_section:
                is_include_section = True
                included_file_path = include_match.group(1)
                with open(included_file_path, "r") as included_file:
                    for included_line in included_file:
                        result_document += included_line

            elif re.search(r"<!--\s*end\s*-->", line):
                is_include_section = False
                result_document += "\n" + line
                continue

    with open(readme_path, "w") as readme_file:
        readme_file.write(result_document)

if __name__ == '__main__':
    concat_files("README.md")
