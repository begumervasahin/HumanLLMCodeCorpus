import os
import getpass

# Function to prompt the user to enter a password securely
def get_password(prompt="Please enter a password: "):
    return getpass.getpass(prompt)

# Function to prompt the user to enter a file name
def get_file_to_encrypt():
    file_name = input("Enter a file to encrypt. (If blank, uses provided input.txt): ")
    return file_name if file_name else 'input.txt'

# Function to install Python libraries using pip
def install_libraries():
    libraries = ['bitstring', 'pycryptodomex']
    for lib in libraries:
        os.system(f'pip install {lib}')
    print('Libraries installed successfully, proceeding...')

# Function to generate a keyfile with 3DES keys
def generate_keyfile(password):
    os.system(f'python 3Des.py genkey {password} keyfile.txt')
    print("Generated keyfile with 3DES keys at keyfile.txt...")

# Function to perform encryption or decryption based on mode
def process(mode, input_file, output_file):
    os.system(f'python 3Des.py {mode} {input_file} keyfile.txt {output_file} {mode}')
    print(f"{mode.capitalize()}ing {input_file} and outputting to {output_file}...")

# Main function to execute the encryption and decryption process
def main():
    # Prompt user for password and file to encrypt
    password = get_password()
    file_to_encrypt = get_file_to_encrypt()

    # Check for and install required Python libraries
    print("Checking for required Python libraries...")
    install_libraries()

    # Generate keyfile with 3DES keys
    generate_keyfile(password)

    # Proceed with encryption
    print("Proceeding with encryption:")
    encryption_modes = ['ECB', 'CBC', 'OFB']
    for mode in encryption_modes:
        output_file = f'cipher_{mode.lower()}.txt'
        process('encrypt', file_to_encrypt, output_file)

    # Proceed with decryption
    print("Encryption complete, proceeding to decryption:")
    for mode in encryption_modes:
        input_file = f'cipher_{mode.lower()}.txt'
        output_file = f'clear_{mode.lower()}.txt'
        process('decrypt', input_file, output_file)

    print("Done!")

if __name__ == "__main__":
    main()
