import tkinter as tk
from tkinter import simpledialog, Label, Entry

class ModelSizeWindow(simpledialog.Dialog):
    """
    Represents a popup window for entering model size (domain).
    """

    def body(self, master):
        """
        Initializes widgets.
        """
        # Labels for x and y inputs
        Label(master, text="x:").grid(row=0)
        Label(master, text="y:").grid(row=1)

        # Entry widgets for x and y inputs
        self.entry_x = Entry(master)
        self.entry_y = Entry(master)

        # Grid placement for entry widgets
        self.entry_x.grid(row=0, column=1)
        self.entry_y.grid(row=1, column=1)

    def apply(self):
        """
        Returns requested inputs.
        """
        # Get values from entry widgets
        x_value = self.entry_x.get()
        y_value = self.entry_y.get()

        # Set result with entered values
        self.result = x_value, y_value

# Example usage
def run_model_size_window():
    # Create Tkinter root window
    root = tk.Tk()

    # Initialize and run ModelSizeWindow
    model_size_window = ModelSizeWindow(root)
    root.mainloop()

# Run the example
if __name__ == "__main__":
    run_model_size_window()
