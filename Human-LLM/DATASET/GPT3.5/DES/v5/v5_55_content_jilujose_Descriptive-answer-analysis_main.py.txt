from nltk.corpus import stopwords
from nltk.corpus import wordnet
import difflib as dl
import os
import string
import new

# Import necessary functions from difflib
get_close_matches = dl.get_close_matches

# OCR image and save text to a file
os.system("tesseract {} st_sheet".format(new.img))

# Read text files and remove stopwords and punctuation
def process_text(file_path):
    with open(file_path, 'r') as file:
        lines = file.read()
        word_tokens = lines.split()
        stop_words = set(stopwords.words('english'))
        filtered_sentence = [word.strip(string.punctuation) for word in word_tokens if word not in stop_words]
    return filtered_sentence

# Process reference text
ref_lines = process_text(str(new.tea))

# Process OCR text
with open("st_sheet.txt", 'r') as file:
    ocr_lines = process_text("st_sheet.txt")

# Create a dictionary to store synonyms for each word in the reference text
synonyms = {word: [] for word in ref_lines}
for word, synonym_list in synonyms.items():
    for synset in wordnet.synsets(word):
        for lemma in synset.lemmas():
            synonym_list.append(lemma.name())

# Count matches between reference and OCR text
num_synonyms_matched = sum(1 for word in ref_lines if get_close_matches(word, ocr_lines))
num_words_matched = sum(1 for word in ref_lines if get_close_matches(word, ocr_lines))

# Calculate similarity ratio
similarity_ratio = float(num_words_matched + num_synonyms_matched) / float(len(ref_lines) + num_synonyms_matched)

# Print the result
print("The Student is awarded")
print('{}% Marks'.format(int(similarity_ratio * 100)))
