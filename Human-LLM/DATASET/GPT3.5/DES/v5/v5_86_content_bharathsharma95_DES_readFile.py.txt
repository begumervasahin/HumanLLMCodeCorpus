from __future__ import with_statement

def read_file_bytes(file_path):
    """
    Read the specified file and return its content as a list of bytes.

    Parameters:
        file_path (str): The path to the file to be read.

    Returns:
        list: A list containing the bytes read from the file.
    """
    with open(file_path, 'rb') as file:
        file_bytes = []
        # Read bytes from the file one by one
        byte = file.read(1)
        while byte:
            file_bytes.append(byte)
            byte = file.read(1)
    return file_bytes

if __name__ == "__main__":
    file_path = "example.txt"  # Replace with the path to your file
    file_bytes = read_file_bytes(file_path)
    print(file_bytes)
