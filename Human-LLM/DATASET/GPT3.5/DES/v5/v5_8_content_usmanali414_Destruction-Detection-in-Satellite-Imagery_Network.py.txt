import keras
from keras import backend as K
from keras.models import Model
from keras.layers import Input, Dense, Multiply, Dropout, Lambda, Activation
from keras.optimizers import Adam

def custom_loss(layer):
    """
    Define a custom loss function for the model.
    """
    def loss(y_true, y_pred):
        # Compute attention weights and the attention-weighted loss
        attention_weights = K.squeeze(layer, axis=2)
        attention_sum = K.sum(attention_weights, axis=-1)
        attention_sum_reshaped = K.reshape(attention_sum, (-1, 1))
        attention_normalized = Lambda(lambda x: x[0] / x[1])([attention_weights, attention_sum_reshaped])
        return K.binary_crossentropy(y_true, y_pred) + (0.3 * K.sum(attention_normalized, axis=-1))
    return loss

def model_attention():
    """
    Define the model architecture.
    """
    # Define input layer
    model_input = Input(shape=(None, 1024))
    
    # Hidden layers
    x = Dense(512, kernel_initializer='glorot_uniform', bias_initializer='zeros')(model_input)
    x = Activation('sigmoid')(x)
    x = Dropout(0.8)(x)
    
    # Attention layer
    x = Dense(1, kernel_initializer='glorot_uniform', bias_initializer='zeros')(x)
    attention_weights = Activation('sigmoid')(x)
    attention_multiply = Multiply()([model_input, attention_weights])
    mean = Lambda(lambda x: K.mean(x, axis=1))(attention_multiply)
    
    # Second set of hidden layers
    y = Dense(512, kernel_initializer='glorot_uniform', bias_initializer='zeros')(mean)
    y = Activation('sigmoid')(y)
    y = Dropout(0.7)(y)
    
    # Output layer
    output = Dense(1, activation='sigmoid', kernel_initializer='glorot_uniform', bias_initializer='zeros')(y)

    # Create the model
    model = Model(model_input, output)

    # Compile the model
    adam = Adam(lr=0.001, beta_1=0.9, beta_2=0.999, epsilon=None, decay=0.0, amsgrad=True)
    for i, layer in enumerate(model.layers):
        layer.name = 'layer_' + str(i)
    model.compile(optimizer=adam, loss=custom_loss(attention_weights), metrics=['accuracy'])

    return model
