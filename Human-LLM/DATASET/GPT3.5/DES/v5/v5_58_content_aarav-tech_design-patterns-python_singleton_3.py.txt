# Singleton design pattern using a decorator

# SingletonDecorator acts as a decorator to enforce the Singleton pattern
class SingletonDecorator:
    def __init__(self, klass):
        self.klass = klass  # Store the class being decorated
        self.instance = None  # Initialize the instance attribute

    def __call__(self, *args, **kwargs):
        if self.instance is None:  # Check if instance is not created
            self.instance = self.klass(*args, **kwargs)  # Create a new instance of the class
        return self.instance  # Return the instance

# Decorate the Logger class with the SingletonDecorator
@SingletonDecorator
class Logger:
    def __init__(self):
        self.start = None  # Initialize the start attribute

    def write(self, message):
        if self.start:
            print(f"{self.start} {message}")  # Print with a prefix if start attribute is set
        else:
            print(message)  # Print the message directly

# Create instances of the Logger class
logger1 = Logger()  # Create the first instance
logger1.start = "<"  # Set the start attribute
print("Logger 1:", logger1)  # Print the first instance
logger1.write("Logger 1 object is created.")  # Call the write method

logger2 = Logger()  # Try to create a second instance
logger2.start = ">"  # Set the start attribute of the second instance
print("Logger 2:", logger2)  # Print the second instance, which is the same as the first
logger2.write("Logger 2 object is created.")  # Call the write method of the second instance
