import socket
import threading
import re

server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)

SERVER_RUNNING = True
SERVER_IP = str(socket.gethostbyname(socket.gethostname()))
SERVER_PORT = 1234

clients = {}
client_public_keys = {}

server_socket.bind((SERVER_IP, SERVER_PORT))
server_socket.listen()
print('Server IP Address:', SERVER_IP)

def handle_client_connection(client_socket, client_username):
    client_connected = True
    while client_connected:
        try:
            message = client_socket.recv(1024).decode('ascii')
            if not message:
                break

            if '**quit' in message:
                handle_quit_message(client_socket, client_username)
                client_connected = False
            elif '**get' in message:
                handle_get_message(client_socket, client_username)
            elif '@' in message:
                handle_private_message(client_socket, client_username, message)
            else:
                broadcast_message(client_socket, client_username, message)
        except Exception as e:
            handle_client_error(client_socket, client_username, e)
            client_connected = False

def handle_quit_message(client_socket, client_username):
    response = 'Goodbye!'
    client_socket.send(response.encode('ascii'))
    clients.pop(client_username)
    print(f'{client_username} logged out from the server')

def handle_get_message(client_socket, client_username):
    for client_name, public_key in client_public_keys.items():
        if client_username != client_name:
            response = '!!' + public_key
            client_socket.send(response.encode('ascii'))

def handle_private_message(client_socket, client_username, message):
    for client_name, socket_obj in clients.items():
        if client_username != client_name:
            response = '@' + message
            socket_obj.send(response.encode('ascii'))

def broadcast_message(client_socket, client_username, message):
    for name, socket_obj in clients.items():
        if client_username != name:
            formatted_msg = format_message(client_username, message)
            socket_obj.send(formatted_msg.encode('ascii'))

def format_message(sender, message):
    prefix = sender + '>>'
    message_without_prefix = message.replace(prefix, '')
    hex_chars = re.findall('..', message_without_prefix)
    decoded_message = ''.join(chr(int(char, 16)) for char in hex_chars)
    return prefix + decoded_message

def handle_client_error(client_socket, client_username, error):
    clients.pop(client_username)
    print(f'{client_username} logged out from the server due to error:', error)

while SERVER_RUNNING:
    client_socket, address = server_socket.accept()
    client_username, public_key = [str(i) for i in client_socket.recv(1024).decode('ascii').split('\n')]
    print(f'{client_username} connected to the server with public key {public_key}')
    client_socket.send('\nHello! Start chatting with your conversation partner!\n'.encode('ascii'))

    if client_socket not in clients:
        clients[client_username] = client_socket
        client_public_keys[client_username] = public_key
        threading.Thread(target=handle_client_connection, args=(client_socket, client_username)).start()
