from des import *
import os

def reverse_permute(input_file='cipher.txt', output_file='finalcipher.txt'):
    # Check if the input file exists
    if not os.path.isfile(input_file):
        print("No input pair file found.")
        raise SystemExit

    # Open the input and output files
    with open(input_file, 'r') as f_in, open(output_file, 'w') as f_out:
        # Process each line in the input file
        for line in f_in:
            # Split the line into two halves
            c1, c2 = line[:64], line[65:-1]
            # Apply initial permutation to each half
            c1_rev = ''.join(initial_permutation(list(c1)))
            c2_rev = ''.join(initial_permutation(list(c2)))
            # Write the reversed halves to the output file
            f_out.write(f"{c1_rev} {c2_rev}\n")

def compute_sbox_xor(input_file='finalcipher.txt', output_file='cipherxor.txt',
                     expansion_file='expandr5.txt', sbox_input_file='inputxor.txt',     
                     sbox_output_file='outputxor.txt'):
    # Check if the input file exists
    if not os.path.isfile(input_file):
        print("No input file found.")
        raise SystemExit

    # Open all necessary files for writing
    with open(input_file, 'r') as f_in, \
         open(output_file, 'w') as f_out, \
         open(expansion_file, 'w') as f_exp, \
         open(sbox_input_file, 'w') as f_sbox_inp, \
         open(sbox_output_file, 'w') as f_sbox_out:
        
        # Process each line in the input file
        for line in f_in:
            # Split the line into two halves
            c1, c2 = line[:64], line[65:-1]
            c1_l6, c1_r6 = c1[:32], c1[32:]
            c2_l6, c2_r6 = c2[:32], c2[32:]
            
            # Expand the right halves
            c1_exp = ''.join(expansion(list(c1_r6)))
            c2_exp = ''.join(expansion(list(c2_r6)))
            
            # XOR the left and right halves
            l6_xor = ''.join(xor(c1_l6, c2_l6))
            r6_xor = ''.join(xor(c1_r6, c2_r6))
            
            # Perform inverse permutation on the XOR result
            permutation_out = list(l6_xor)
            sbox_output_xor = ''.join(inverse_permute(permutation_out))
            
            # XOR the expanded halves
            sbox_input_xor = ''.join(xor(c1_exp, c2_exp))
            
            # Write the results to respective output files
            f_out.write(f"{l6_xor} {r6_xor}\n")
            f_exp.write(f"{c1_exp} {c2_exp}\n")
            f_sbox_inp.write(f"{sbox_input_xor}\n")
            f_sbox_out.write(f"{sbox_output_xor}\n")

# Call functions to execute
reverse_permute()
compute_sbox_xor()
