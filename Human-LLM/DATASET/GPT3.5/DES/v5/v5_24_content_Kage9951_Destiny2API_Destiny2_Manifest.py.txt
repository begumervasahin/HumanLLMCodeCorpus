import requests
import zipfile
import os
import pickle
import json
import sqlite3

# Function to download the Destiny 2 manifest
def download_manifest():
    manifest_url = 'https://www.bungie.net/Platform/Destiny2/Manifest/'
    response = requests.get(manifest_url)
    manifest_data = response.json()

    manifest_content_url = 'https://www.bungie.net' + manifest_data['Response']['mobileWorldContentPaths']['en']
    response = requests.get(manifest_content_url)

    with open("MANIFEST.ZIP", "wb") as zip_file:
        zip_file.write(response.content)
    print("Manifest download complete!")

    with zipfile.ZipFile('MANIFEST.ZIP') as zip_file:
        files = zip_file.namelist()
        zip_file.extractall()
    os.rename(files[0], 'Manifest.content')
    print('Manifest unzipped!')

# Function to build dictionaries from the manifest data
def build_manifest_dictionary(hashes):
    connection = sqlite3.connect('Manifest.content')
    print('Connected to the manifest database')
    cursor = connection.cursor()
    all_data = {}

    for table_name, hash_key in hashes.items():
        cursor.execute(f'SELECT json FROM {table_name}')
        print(f'Generating dictionary for {table_name}...')
        items = cursor.fetchall()
        item_jsons = [json.loads(item[0]) for item in items]
        item_dict = {}

        for item in item_jsons:
            item_dict[item[hash_key]] = item
        all_data[table_name] = item_dict

    print('Dictionary generated!')
    return all_data

# Function to initialize the manifest
def init_manifest():
    if not os.path.isfile('Manifest.content'):
        download_manifest()
        manifest = build_manifest_dictionary(hashes)
        with open('manifest.pickle', 'wb') as pickle_file:
            pickle.dump(manifest, pickle_file)
            print("'manifest.pickle' created!\nInitialization complete!")
    else:
        print('Manifest content already exists')

    with open('manifest.pickle', 'rb') as pickle_file:
        manifest = pickle.load(pickle_file)
    return manifest

# Define hash keys for each table in the manifest
hashes = {
    'DestinyClassDefinition': 'hash',
    'DestinyInventoryItemDefinition': 'hash',
    'DestinyTalentGridDefinition': 'hash',
    'DestinyHistoricalStatsDefinition': 'statId',
    'DestinyStatDefinition': 'hash',
    'DestinySandboxPerkDefinition': 'hash',
    'DestinyStatGroupDefinition': 'hash',
    'DestinyInventoryBucketDefinition': 'hash'
}

# Initialize the manifest
manifest_data = init_manifest()
