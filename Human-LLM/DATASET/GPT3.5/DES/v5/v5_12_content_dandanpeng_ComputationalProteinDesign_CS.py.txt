import math
import numpy as np
import matplotlib.pyplot as plt
from Bio import Align
from Bio.PDB.PDBParser import PDBParser
from Bio.SubsMat.MatrixInfo import blosum62
from Bio.PDB.Polypeptide import PPBuilder
from classProtein import Protein, Match
from collections import Counter
from random import random

def cuckoo_search_new(protein, match, pa):
    aligner = Align.PairwiseAligner()
    aligner.open_gap_score = -10
    aligner.extend_gap_score = -0.5
    aligner.substitution_matrix = blosum62

    lower_bounds = [0] * len(protein.terms)
    upper_bounds = [match.frags_count[term] for term in match.frags_count]

    def create_individual():
        individual = [random_frag_count(term) for term in protein.terms]
        return individual

    def random_frag_count(term):
        if random() <= 0.75:
            return round(random() * match.frags_count[term])
        else:
            return match.frags_count[term]

    def initial_population(pop_size):
        population = np.array([create_individual() for _ in range(pop_size)], dtype=int)
        return population

    def compare_aa(individual):
        frag_numbers = {term: count for term, count in zip(protein.terms, individual)}
        frag_sequences = {term: match.select_frag(term, frag_numbers[term]) for term in protein.terms}
        score = sum(aligner.score(frag_sequences[edge[0]][pos[0]], frag_sequences[edge[1]][pos[1]])
                    for edge in protein.graph.edges
                    if isinstance(frag_sequences[edge[0]], np.ndarray) and isinstance(frag_sequences[edge[1]], np.ndarray)
                    for pos in protein.graph.edges[edge]['sameAA'])
        return score / len(protein.terms)

    def term_count(individual):
        selected_frags = {term: count for term, count in zip(protein.terms, individual)}
        return sum(selected_frags[term] < match.frags_count[term] for term in protein.terms) / len(protein.terms)

    def energy(individual):
        return compare_aa(individual) - 10 * term_count(individual)

    def enforce_bounds(s):
        return np.clip(s, lower_bounds, upper_bounds).astype(np.int)

    def get_cuckoos(nest, best):
        beta = 1.5
        sigma_u = (math.gamma(1 + beta) * math.sin(math.pi * beta / 2) /
                   (math.gamma((1 + beta) / 2) * beta * (2 ** ((beta - 1) / 2)))) ** (1 / beta)
        sigma_v = 1
        for i, s in enumerate(nest):
            u = np.random.normal(0, sigma_u)
            v = np.random.normal(0, sigma_v)
            step = u / abs(v) ** (1 / beta)
            stepsize = 0.01 * step * (s - best)
            nest[i] = enforce_bounds(s + stepsize * np.random.randn(len(s)))
        return nest

    def get_best_nest(nest, new_nest, fitness):
        for i, f_new in enumerate(map(energy, new_nest)):
            if f_new >= fitness[i]:
                fitness[i] = f_new
                nest[i] = new_nest[i]
        max_fitness = max(fitness)
        best_nest = np.argmax(fitness)
        return max_fitness, best_nest, nest, fitness

    def empty_nests(nest):
        K = np.random.uniform(0, 1, size=nest.shape) > pa
        nest1 = nest[np.random.permutation(len(nest))]
        nest2 = nest[np.random.permutation(len(nest))]
        step_size = np.random.uniform(0, 1) * (nest1 - nest2)
        new_nest = nest + (step_size * K).astype(int)
        return np.array([enforce_bounds(s) for s in new_nest])

    def restore_seq(individual):
        possible_aa = [[] for _ in protein.terms]
        candidate = {term: aa_list for term, aa_list in zip(protein.terms, possible_aa)}
        frag_numbers = {term: count for term, count in zip(protein.terms, individual)}
        frag_sequences = {term: match.select_frag(term, frag_numbers[term]) for term in protein.terms}
        for pos in protein.inverse:
            for term in protein.inverse[pos]:
                if isinstance(frag_sequences[term], np.ndarray):
                    index = protein.neighbors[term].index(pos)
                    candidate[pos].append(frag_sequences[term][index])
        possible_seq = ''.join([Counter(candidate[term]).most_common(1)[0][0] if candidate[term] else '-' for term in protein.terms])
        return possible_seq

    def real_seq():
        structure = PDBParser().get_structure(protein.protein_id, protein.protein_id + '.pdb')
        ppb = PPBuilder()
        seq = ''.join([pp.get_sequence() for pp in ppb.build_peptides(structure)])
        return seq

    original_seq = real_seq()
    nest = initial_population(100)
    fitness = [energy(nest[i]) for i in range(len(nest))]
    max_fitness, best_nest, nest, fitness = get_best_nest(nest, nest, fitness)
    fitness_scores = []
    alignment_scores = []
    for _ in range(50):
        nest_a = nest.copy()
        new_nest = get_cuckoos(nest_a, nest[best_nest])
        max_fitness, best_nest, nest, fitness = get_best_nest(nest, new_nest, fitness)
        nest_b = nest.copy()
        new_nest = empty_nests(nest_b)
        max_fitness, best_nest, nest, fitness = get_best_nest(nest, new_nest, fitness)
        fitness_scores.append(max_fitness)
        prediction = restore_seq(nest[best_nest])
        alignment_scores.append(aligner.score(prediction, original_seq))
    plt.plot(fitness_scores)
    plt.savefig(protein.protein_id + '_fitness.jpg')
    plt.close()
    plt.plot(alignment_scores)
    plt.savefig(protein.protein_id + '_align.jpg')
    plt.close()

# Example usage
# Replace the following lines with appropriate input for protein, match, and pa parameters
protein = Protein()  # Define your Protein object
match = Match()      # Define your Match object
pa = 0.5             # Define your pa parameter

# Call the function
cuckoo_search_new(protein, match, pa)
