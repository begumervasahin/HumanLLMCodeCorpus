from tkinter import *
import time

# Function to handle closing the application
def close(event):
    global running
    running = False

# Create the Tkinter root window
root = Tk()
root.overrideredirect(True)
root.wm_attributes("-transparentcolor", "gray99")

# Get screen dimensions
screen_width = root.winfo_screenwidth()
screen_height = root.winfo_screenheight()

# Create a frame to hold the time and date
timeframe = Frame(root, width=screen_width, height=screen_height, bg="gray99") 
timeframe.grid(row=0, column=0)

# Label to display the current time
time_label = Label(
    timeframe,
    textvariable=tkintertime,
    fg="white",
    bg="gray99",
    font=("NovaMono", 40)
)
time_label.place(y=screen_height/2 - 60, x=screen_width/2, anchor="center")     

# Label to display the current date
date_label = Label(
    timeframe,
    textvariable=tkinterdate,
    fg="white",
    bg="gray99",
    font=("Bahnschrift", 15)
)
date_label.place(y=screen_height/2 + 60, x=screen_width/2, anchor="center")     

# Bind the 'Escape' key to close the application
root.bind('<Escape>', close)

# Main loop to update time and date continuously
running = True
while running:
    current_time = time.strftime("%H:%M:%S")
    current_date = time.strftime("%A, %e %B")
    tkintertime.set(value=current_time)
    tkinterdate.set(value=current_date)
    root.update_idletasks()
    root.update()
    time.sleep(1)
