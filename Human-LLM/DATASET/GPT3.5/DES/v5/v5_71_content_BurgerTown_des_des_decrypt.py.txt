class DESDecrypt:
    def __init__(self):
        pass

    def decode(self, string, key, key_len, string_len):
        output = ""
        code_string = self._char_to_binary(string, string_len)
        code_key = self._generate_key(key, key_len)
        real_len = (key_len // 16) + 1 if key_len % 16 != 0 else key_len // 16
        turn_len = string_len * 4

        num = 0
        for i in range(0, turn_len, 64):
            run_code = code_string[i:i + 64]
            run_key = code_key[num % real_len]
            run_code = self._initial_permutation(run_code)

            for j in range(16):
                code_r = run_code[32:64]
                code_l = run_code[0:32]
                run_code = code_r
                code_r = self._expansion_permutation(code_r)
                key_y = run_key[15 - j]
                code_r = self._xor(code_r, key_y)
                code_r = self._substitution_boxes(code_r)
                code_r = self._permutation(code_r)
                code_r = self._xor(code_l, code_r)
                run_code += code_r

            num += 1
            code_r = run_code[32:64]
            code_l = run_code[0:32]
            run_code = code_r + code_l
            output += self._final_permutation(run_code)

        return output

    @staticmethod
    def _generate_key(key, key_len):
        code_key = DESDecrypt._char_to_binary(key, key_len)
        real_len = (key_len // 16) * 16 + 16 if key_len % 16 != 0 else key_len
        b = [[''] * 16 for _ in range(real_len // 16)]
        num = 0
        turn_len = 4 * key_len

        for i in range(0, turn_len, 64):
            run_key = code_key[i:i + 64]
            run_key = DESDecrypt._key_permutation(run_key)

            for j in range(16):
                key_l = run_key[0:28]
                key_r = run_key[28:56]
                key_l = key_l[KEY_SHIFT[j]:28] + key_l[0:KEY_SHIFT[j]]
                key_r = key_r[KEY_SHIFT[j]:28] + key_r[0:KEY_SHIFT[j]]
                run_key = key_l + key_r
                key_y = DESDecrypt._final_key_permutation(run_key)
                b[num][j] = key_y[:]

            num += 1

        return b

    @staticmethod
    def _xor(code, key):
        code_len = len(key)
        return_list = ''
        for i in range(code_len):
            if code[i] == key[i]:
                return_list += '0'
            else:
                return_list += '1'
        return return_list

    @staticmethod
    def _initial_permutation(code):
        changed_code = ''
        for i in range(64):
            changed_code += code[INITIAL_PERMUTATION[i] - 1]
        return changed_code

    @staticmethod
    def _final_permutation(code):
        return_list = ''
        for i in range(16):
            temp_list = ''
            for j in range(4):
                temp_list += code[FINAL_PERMUTATION[i * 4 + j] - 1]
            return_list += "%x" % int(temp_list, 2)
        return return_list

    @staticmethod
    def _expansion_permutation(code):
        return_list = ''
        for i in range(48):
            return_list += code[EXPANSION_PERMUTATION[i] - 1]
        return return_list

    @staticmethod
    def _substitution_boxes(key):
        return_list = ''
        for i in range(8):
            row = int(str(key[i * 6]) + str(key[i * 6 + 5]), 2)
            raw = int(str(key[i * 6 + 1]) + str(key[i * 6 + 2]) +
                      str(key[i * 6 + 3]) + str(key[i * 6 + 4]), 2)
            return_list += DESDecrypt._int_to_binary(SUBSTITUTION_BOXES[i][row][raw], 4)
        return return_list

    @staticmethod
    def _permutation(code):
        return_list = ''
        for i in range(32):
            return_list += code[PERMUTATION[i] - 1]
        return return_list

    @staticmethod
    def _char_to_binary(code, length):
        return_code = ''
        length = length % 16
        for key in code:
            code_ord = int(key, 16)
            return_code += DESDecrypt._int_to_binary(code_ord, 4)
        if length != 0:
            return_code += '0' * (16 - length) * 4
        return return_code

    @staticmethod
    def _int_to_binary(o, length):
        return_code = ''
        for i in range(length):
            return_code = str(o >> i & 1) + return_code
        return return_code

    @staticmethod
    def _key_permutation(key):
        changed_key = ''
        for i in range(56):
            changed_key += key[KEY_PERMUTATION[i] - 1]
        return changed_key

    @staticmethod
    def _final_key_permutation(key):
        return_list = ''
        for i in range(48):
            return_list += key[FINAL_KEY_PERMUTATION[i] - 1]
        return return_list
