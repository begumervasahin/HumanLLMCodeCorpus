import numpy as np
import pandas as pd
import math
from sklearn.model_selection import train_test_split
from sklearn.linear_model import Ridge
from sklearn.linear_model import RidgeCV
from sklearn.metrics import mean_squared_error, mean_absolute_error

# Load data
data_path = r'D:\360MoveData\Users\HP\Desktop\prostate.data.txt'
data = pd.read_table(data_path)

# Define factors and target variable
factors = ['lcavol', 'lweight', 'age', 'lbph', 'svi', 'lcp', 'gleason', 'pgg45']        
X = data[factors]
Y = data['lpsa']

# Function to calculate the standard error of a sample
def calculate_standard_error(sample):
    std = np.std(sample, ddof=0)
    return std / math.sqrt(len(sample))

# Split the data into training and testing sets
X_train, X_test, y_train, y_test = train_test_split(X, Y, test_size=0.2, random_state=1)

# Initialize an array of alpha values for RidgeCV
alphas = np.logspace(-5, 2, 200)

# Perform Ridge regression with cross-validation to find the optimal alpha
ridge_cv = RidgeCV(alphas=alphas, normalize=True, scoring='neg_mean_squared_error', cv=10)
ridge_cv.fit(X_train, y_train)

# Print the optimal alpha value
print('Optimal Alpha:', round(ridge_cv.alpha_, 4))

# Train the Ridge regression model using the optimal alpha
ridge = Ridge(alpha=ridge_cv.alpha_)
ridge.fit(X_train, y_train)

# Make predictions on the entire dataset and the test set
y_train_predict = ridge.predict(X_train)
y_test_predict = ridge.predict(X_test)

# Calculate evaluation metrics
mse_train = mean_squared_error(y_train, y_train_predict)
mse_test = mean_squared_error(y_test, y_test_predict)
mae_train = mean_absolute_error(y_train, y_train_predict)
mae_test = mean_absolute_error(y_test, y_test_predict)

# Calculate standard error for the test set predictions
std_error = calculate_standard_error(y_test_predict)

# Format coefficients
coefficients = [(factor, round(coef, 4)) for factor, coef in zip(factors, ridge.coef_)]

# Display results
print('Intercept:', round(ridge.intercept_, 4))
print('Estimated coefficients:', coefficients)
print('Standard Error:', std_error)
print('MSE Train:', round(mse_train, 4))
print('MSE Test:', round(mse_test, 4))
print('MAE Train:', round(mae_train, 4))
print('MAE Test:', round(mae_test, 4))
