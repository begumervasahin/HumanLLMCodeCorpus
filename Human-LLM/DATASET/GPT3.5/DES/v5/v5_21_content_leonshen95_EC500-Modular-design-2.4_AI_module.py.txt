import numpy as np
from Database_Module import DataBaseModule

class AI_module:
    def __init__(self, data_dict):
        self.data_dict = data_dict

    def query_data_from_database(self, patient_id, info_db):
        blood_oxygen = []
        blood_pressure = []
        systolic = []
        diastolic = []

        # Search for patient data in the database
        patient_data = DataBaseModule.search(patient_id)

        # Extract relevant information if patient ID exists in the infoDB
        if patient_id in info_db:
            pressure = patient_data.get('blood_pressure')
            oxygen = patient_data.get('blood_oxygen')
            diastolic_BP = patient_data.get('Diastolic_BP')
            systolic_BP = patient_data.get('Systolic_BP')

            blood_pressure.append(pressure)
            blood_oxygen.append(oxygen)
            systolic.append(systolic_BP)
            diastolic.append(diastolic_BP)

        return blood_oxygen, blood_pressure, systolic, diastolic

    def ai_module(self, blood_oxygen, blood_pressure, systolic, diastolic):
        oxygen = np.array(blood_oxygen)
        pressure = np.array(blood_pressure)
        diastolic = np.array(diastolic)
        systolic = np.array(systolic)

        pressure_prediction = np.mean(pressure)
        oxygen_prediction = np.mean(oxygen)
        diastolic_prediction = np.mean(diastolic)
        systolic_prediction = np.mean(systolic)

        return pressure_prediction, oxygen_prediction, diastolic_prediction, systolic_prediction

    def feedback(self, pressure_prediction, oxygen_prediction, diastolic_prediction, systolic_prediction):
        lower_BP = 80
        upper_BP = 120
        lower_BO = 80
        upper_BO = 120

        # Initialize alert flags
        BO_alert = False
        BP_alert = False
        Pulse_alert = False

        # Check if values are outside of normal range
        if oxygen_prediction < lower_BO or oxygen_prediction > upper_BO:
            BO_alert = True
        if pressure_prediction < lower_BP or pressure_prediction > upper_BP:
            BP_alert = True
        if (systolic_prediction < 90 or diastolic_prediction < 60 or 
            systolic_prediction > 140 or diastolic_prediction > 90):
            Pulse_alert = True

        return BO_alert, BP_alert, Pulse_alert
