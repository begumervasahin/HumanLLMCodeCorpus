def execute_design_commands(args, be_input):
    """
    Execute base editor design commands based on the provided arguments and input data.
    """
    for index, row in be_input.iterrows():
        print(row[0])  # Print BE type or parameters

        # Define output prefix
        output_prefix = args.output_prefix + '_' + row[0]

        # Construct command based on BE type or parameters
        if args.be_type == 'be':
            cmd = (
                f'python base_editing_guide_designs.py --input-file {args.input_file} --input-type {args.ip_type}'
                f' --be-type {row[0]} --output-name {output_prefix}'
            )
        else:
            pam = row['PAM']
            window = row['Window']
            sglen = row['Sgrna length']
            edit = row['Edit']
            intron_buffer = row['Intron buffer']
            filter_gc = row['Filter GC']
            cmd = (
                f'python base_editing_guide_designs.py --input-file {args.input_file} --input-type {args.ip_type}'
                f' --pam {pam} --edit-window {window} --sg-len {sglen} --edit {edit} --intron-buffer {intron_buffer}'
                f' --filter-gc {filter_gc} --output-name {output_prefix}'
            )

        # Execute command
        os.system(cmd)
