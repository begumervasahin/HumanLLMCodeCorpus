def calculate_power_modulo(base, exponent, modulus):
    """
    Calculate the power of a number modulo a given modulus.

    Parameters:
        base (int): The base.
        exponent (int): The exponent.
        modulus (int): The modulus.

    Returns:
        int: The result of (base^exponent) % modulus.
    """
    if exponent == 1:
        # Base case: If exponent is 1, return the base itself.
        return base
    else:
        # Calculate the power modulo the given modulus.
        return pow(base, exponent, modulus)
