from tkinter import simpledialog, Label, Checkbutton, IntVar, Frame, W

class TraceWindow(simpledialog.Dialog):
    """
    Represents a popup window for selecting trace components and FFT.
    """

    def __init__(self, master, *, show_fft=True):
        """
        Initializes object variables and calls the parent class constructor.
        """
        self.show_fft = show_fft
        super().__init__(master)

    def body(self, master):
        """
        Initializes widgets.
        """
        Label(self, text="Choose component(s) to plot:").pack()
        button_width = 3
        self.button_frame = Frame(self)

        # Trace component checkboxes
        trace_components = [
            ("Ex", self.ex_en), ("Ey", self.ey_en), ("Ez", self.ez_en),
            ("Hx", self.hx_en), ("Hy", self.hy_en), ("Hz", self.hz_en)
        ]
        for idx, (text, var) in enumerate(trace_components):
            var.set(1)
            chk_btn = Checkbutton(self.button_frame, text=text, variable=var,
                                  width=button_width, anchor=W)
            chk_btn.grid(row=0, column=idx)
        
        self.button_frame.pack()

        # FFT checkbox
        self.fft_en = IntVar()
        if self.show_fft:
            self.fft_button = Checkbutton(self, text="Show FFT", variable=self.fft_en)
            self.fft_button.pack()

    def apply(self):
        """
        Returns requested inputs.
        """
        selected_components = [var.cget('text') for var in [self.ex_en, self.ey_en, self.ez_en, 
                                                            self.hx_en, self.hy_en, self.hz_en] 
                               if var.get()]
        res_str = ' '.join(selected_components)

        if self.fft_en.get() == 1:
            res_str += " -fft"
        self.result = res_str.strip()
