# Function to write a number to a file in binary format
def write_number(number, fname):
    with open(fname, 'wb') as f:
        while number > 0:
            byte = number % 256
            number //= 256
            f.write(chr(byte))

# Function to hash a message using SHA256 and return the result modulo n4
def h(x):
    dx1 = hashlib.sha256(x).digest()
    res = 0
    for cx in dx1:
        res = (res << 8) ^ ord(cx)
    return res % n4

# Function to generate a random seed using SHA256
def random_seed(m):
    md = hashlib.sha256("***RANDOM-SEED_X***".encode())
    md.update('large key value for generation of random number'.encode())
    md.update(m.encode())
    result = 0
    largest_r = md.digest()
    for i in range(len(largest_r)):
        result = (result << 8) ^ ord(largest_r[i])
    return result
