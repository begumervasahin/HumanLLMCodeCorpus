# Import specific classes and functions from the function.py and seqpred.py modules
from function import Protein, Matches, genetic_algorithm
from seqpred import ProteinMatcher

# Define a list of protein IDs
protein_ids = [
    '1aba', '1bxv', '1by2', '1hyp', '1opc',
    '1tmy', '2acy', '2mcm', '3e0e', '3k63'
]

# Initialize a variable to keep track of the score
score = 0

# Iterate over each protein ID
for protein_id in protein_ids:
    # Create a Protein object for the current protein ID
    protein = Protein(protein_id)
    
    # Create a Matches object for the matching sequence of the protein
    match = Matches(protein.match)
    
    # Perform genetic algorithm with specified parameters
    genetic_algorithm(
        protein, match,
        population_size=100, generations=50,
        tournament_size=2, mutation_rate=0.03, elitism=2
    )
