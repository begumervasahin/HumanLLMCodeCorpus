# Import necessary modules
from cli.menus.main_menu import main_menu
import global_settings
from tempfile import TemporaryDirectory

# Initialize global settings
global_settings.initialize()

# Initialize the list of screens with the main menu
screens_to_process = [main_menu]

# Initialize the list of automata
automata = []

# Use TemporaryDirectory for temporary file operations
with TemporaryDirectory() as temp_dir:
    # Continue processing screens until the list is empty
    while screens_to_process:
        # Get the next screen to be processed
        current_screen = screens_to_process[-1]
        # Execute the function associated with the current screen
        current_screen(screens_to_process, automata, temp_dir)
