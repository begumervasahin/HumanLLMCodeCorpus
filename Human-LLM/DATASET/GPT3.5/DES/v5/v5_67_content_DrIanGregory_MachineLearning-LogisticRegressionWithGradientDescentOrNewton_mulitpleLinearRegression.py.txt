import numpy as np
import pandas as pd
from random import seed, normalvariate

def root_mean_squared_error(Y, Y_pred):
    return np.sqrt(np.mean((Y - Y_pred) ** 2))

def r_squared_score(Y, Y_pred):
    mean_y = np.mean(Y)
    ss_tot = np.sum((Y - mean_y) ** 2)
    ss_res = np.sum((Y - Y_pred) ** 2)
    r2 = 1 - (ss_res / ss_tot)
    return r2

def cost_function(X, Y, W):
    N = len(Y)
    cost = np.sum((X.dot(W) - Y) ** 2) / (2 * N)
    return cost

def gradient_descent(X, Y, W, alpha, max_iterations=10000):
    N = len(Y)
    cost_history = []
    weight_history = []
    iteration = 0

    while iteration < max_iterations:
        predicted_values = X.dot(W)
        loss = predicted_values - Y
        gradient = X.T.dot(loss) / N
        W = W - alpha * gradient
        cost = cost_function(X, Y, W)
        cost_history.append(cost)
        weight_history.append(W)
        iteration += 1

    return W, cost_history, weight_history

def show_results(X, Y, initial_weights, final_weights, cost_history, max_iterations):
    initial_cost = cost_function(X, Y, initial_weights)
    predicted_values = X.dot(final_weights)
    dash_line = '=' * 80

    print(dash_line)
    print("MULTI LINEAR REGRESSION USING GRADIENT DESCENT TERMINATION RESULTS")
    print(dash_line)
    print(f"Initial Weights:    {initial_weights[0]:>12.1f}, {initial_weights[1]:>2.1f}, {initial_weights[2]:>2.1f}.")
    print(f"Initial Cost:        {initial_cost:>12,.1f}")
    print()
    print(f"Final Weights:       w0:{final_weights[0]:>+0.2f}, w1:{final_weights[1]:>+3.2f}, w2:{final_weights[2]:>+3.3f}")
    print(f"Final Cost:         {cost_history[-1]:>+12.1f}")
    print(f"RMSE:              {root_mean_squared_error(Y, predicted_values):>+12.1f}, R-Squared: {r_squared_score(Y, predicted_values):>+12.1f}")
    print(dash_line)

def program_body(data, learning_rate, max_iterations):
    num_columns = data.shape[1] - 1
    initial_weights = np.zeros(num_columns + 1)
    X_bias = np.ones(data.shape[0])
    X = np.column_stack((X_bias, data.iloc[:, 1:(num_columns + 1)].values))
    Y = np.array(data.iloc[:, 0])
    final_weights, cost_history, weight_history = gradient_descent(X, Y, initial_weights, learning_rate, max_iterations)     
    show_results(X, Y, initial_weights, final_weights, cost_history, max_iterations)

def run():
    learning_rate = 0.0001
    max_iterations = 2500000
    np.random.seed(1234)
    num_data_points = 500
    means = [70, 70]
    stds = [9, 9]
    correlation = 0.8
    covs = [[stds[0] ** 2, stds[0] * stds[1] * correlation], [stds[0] * stds[1] * correlation, stds[1] ** 2]]
    data1 = np.random.multivariate_normal(means, covs, num_data_points).T
    job_probabilities = (data1[0] + data1[1]) / 2.5 + normalvariate(25, 4)
    data = np.vstack((job_probabilities, data1)).T
    data = pd.DataFrame({"JobPotential": data[:, 0], "AI": data[:, 1], "MachineLearning": data[:, 2]})
    program_body(data, learning_rate, max_iterations)
    print("Finished")

if __name__ == '__main__':
    run()
