def read_binary_file(filename_path):
    """
    Read binary content from a file and return it as a binary string.

    Parameters:
    - filename_path: str, path to the binary file

    Returns:
    - binary_content: str, binary representation of the file content
    """
    binary_content = ''
    with open(filename_path, mode='rb') as file:
        file_bytes = file.read()
        binary_content = ''.join(format(byte, '08b') for byte in file_bytes)
    return binary_content

def write_binary_file(binary_content, filename_path):
    """
    Write binary content to a file.

    Parameters:
    - binary_content: str, binary string to write to the file
    - filename_path: str, path to the output file
    """
    byte_array = bytearray()
    for i in range(0, len(binary_content), 8):
        byte_str = binary_content[i:i+8]
        byte_int = int(byte_str, 2)  # Convert binary string to integer
        byte_array.append(byte_int)
    with open(filename_path, mode='wb') as file:
        file.write(byte_array)

def hex_key_to_binary(hex_key):
    """
    Convert a hexadecimal key to its binary representation.

    Parameters:
    - hex_key: str, hexadecimal key

    Returns:
    - binary_key: str, binary representation of the hexadecimal key
    """
    binary_key = ''.join("{:04b}".format(int(char, 16)) for char in hex_key)
    return binary_key
