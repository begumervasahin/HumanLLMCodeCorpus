import torch
import torch.nn as nn
import torch.optim as optim
import torch.nn.functional as F
import math
import numpy as np
import random
import os
import argparse
from data_utils import Preprocessing

# Define command-line arguments
parser = argparse.ArgumentParser(description='Train TransE model for knowledge graph embeddings')
parser.add_argument('--datadir', required=True, help="Path to the directory containing the data files")
args = parser.parse_args()

# Load data
all_data_file = os.path.join(args.datadir, 'data10K.csv')
preprocessor = Preprocessing()
pos_triples, entity2id, rel2id = preprocessor.get_entity_relations(all_data_file)

# Model parameters
embedding_size = 100
num_entity = len(entity2id)
num_rel = len(rel2id)
batch_size = 100
learning_rate = 0.001
epochs = 1000
USE_CUDA = torch.cuda.is_available()

# CUDA configuration
device = torch.device("cuda" if USE_CUDA else "cpu")

# Define the TransE model
class TransE(nn.Module):
    def __init__(self, num_entity, num_rel, embedding_size):
        super(TransE, self).__init__()
        self.embedding_size = embedding_size
        self.entity_embeddings = nn.Embedding(num_entity, embedding_size)
        self.rel_embeddings = nn.Embedding(num_rel, embedding_size)
        self.initialize_weights()

    def initialize_weights(self):
        nn.init.uniform_(self.entity_embeddings.weight, a=-6/math.sqrt(self.embedding_size), b=6/math.sqrt(self.embedding_size))
        nn.init.uniform_(self.rel_embeddings.weight, a=-6/math.sqrt(self.embedding_size), b=6/math.sqrt(self.embedding_size))
        self.entity_embeddings.weight = nn.Parameter(F.normalize(self.entity_embeddings.weight, p=2, dim=1))
        self.rel_embeddings.weight = nn.Parameter(F.normalize(self.rel_embeddings.weight, p=2, dim=1))

    def forward(self, pos_h, pos_t, pos_r, neg_h, neg_t):
        pos_h_emb = self.entity_embeddings(pos_h)
        pos_t_emb = self.entity_embeddings(pos_t)
        pos_r_emb = self.rel_embeddings(pos_r)
        neg_h_emb = self.entity_embeddings(neg_h)
        neg_t_emb = self.entity_embeddings(neg_t)
        margin = torch.ones(batch_size, device=device)
        zeros = torch.zeros(batch_size, device=device)
        pos_distance = torch.norm(pos_h_emb + pos_r_emb - pos_t_emb, p=2, dim=1)
        neg_distance = torch.norm(neg_h_emb + pos_r_emb - neg_t_emb, p=2, dim=1)
        loss = torch.sum(torch.max(zeros, pos_distance + margin - neg_distance))
        return loss

# Prepare batch data
def prepare_batch(batch_triples):
    batch_pos_h, batch_pos_t, batch_r = [], [], []
    batch_neg_h, batch_neg_t = [], []

    for triple in batch_triples:
        h, r, t = triple
        batch_pos_h.append(entity2id[h])
        batch_pos_t.append(entity2id[t])
        batch_r.append(rel2id[r])

        if np.random.random() < 0.5:
            while True:
                neg_t = random.choice(list(entity2id.values()))
                if (h, r, neg_t) not in pos_triples:
                    break
            batch_neg_h.append(entity2id[h])
            batch_neg_t.append(neg_t)
        else:
            while True:
                neg_h = random.choice(list(entity2id.values()))
                if (neg_h, r, t) not in pos_triples:
                    break
            batch_neg_h.append(neg_h)
            batch_neg_t.append(entity2id[t])

    pad_size = batch_size - len(batch_pos_h)
    batch_pos_h += [0] * pad_size
    batch_pos_t += [0] * pad_size
    batch_r += [0] * pad_size
    batch_neg_h += [0] * pad_size
    batch_neg_t += [0] * pad_size

    return torch.tensor(batch_pos_h, dtype=torch.long, device=device), \
           torch.tensor(batch_pos_t, dtype=torch.long, device=device), \
           torch.tensor(batch_r, dtype=torch.long, device=device), \
           torch.tensor(batch_neg_h, dtype=torch.long, device=device), \
           torch.tensor(batch_neg_t, dtype=torch.long, device=device)

# Training function
def train(model, optimizer):
    for epoch in range(epochs):
        random.shuffle(pos_triples)
        total_loss = 0.0
        for i in range(0, len(pos_triples), batch_size):
            batch_triples = pos_triples[i:i+batch_size]
            pos_h, pos_t, pos_r, neg_h, neg_t = prepare_batch(batch_triples)
            optimizer.zero_grad()
            loss = model(pos_h, pos_t, pos_r, neg_h, neg_t)
            loss.backward()
            optimizer.step()
            total_loss += loss.item()
        print(f"Epoch {epoch+1}, Loss: {total_loss:.4f}")

# Initialize TransE model
model = TransE(num_entity, num_rel, embedding_size).to(device)
optimizer = optim.Adam(model.parameters(), lr=learning_rate)

# Train the model
train(model, optimizer)

# Save the trained model
torch.save(model.state_dict(), 'TransE.pth')
