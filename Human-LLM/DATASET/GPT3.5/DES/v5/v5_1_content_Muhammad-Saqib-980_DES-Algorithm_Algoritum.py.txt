def int_to_binary(num, min_len=8):
    # Convert integer to binary representation with a minimum length
    binary = bin(num)[2:]  # Remove '0b' prefix
    binary = '0' * (min_len - len(binary)) + binary if len(binary) < min_len else binary
    return [int(bit) for bit in binary]

def hex_to_binary(hex_str):
    # Convert hexadecimal string to binary representation
    byte_array = bytes.fromhex(hex_str)
    return [bit for byte in byte_array for bit in int_to_binary(byte)]

def text_to_binary(text):
    # Convert text string to binary representation
    byte_array = bytearray(text, encoding='utf-8')
    return [bit for byte in byte_array for bit in int_to_binary(byte)]

def permute(binary, table):
    # Perform permutation operation based on the specified table
    return [binary[i - 1] for i in table]

def round1(text, key):
    # Perform round 1 of the encryption algorithm
    # Initial permutation of text
    text_binary = permute(hex_to_binary(text), IP_TABLE)
    left = text_binary[:32]
    right = text_binary[32:]
    left2 = right.copy()

    # Generate subkey
    key_binary = text_to_binary(key)
    if len(key_binary) == 64:
        key_binary = permute(key_binary, PC1_TABLE)
        key_binary = shift_left(key_binary)
    key_binary = permute(key_binary, PC2_TABLE)

    # Expansion permutation of right half
    right = permute(right, E_TABLE)

    # XOR operation with subkey
    right = xor(right, key_binary)

    # S-box substitution
    right = s_box(right)

    # Permutation
    right = permute(right, P_TABLE)

    # XOR operation with left half
    right2 = xor(right, left)

    return text_binary, left2, right2

def xor(a, b):
    # Perform XOR operation on two binary lists
    return [bit_a ^ bit_b for bit_a, bit_b in zip(a, b)]

def shift_left(binary):
    # Perform circular left shift on a binary list
    return binary[1:] + [binary[0]]

def s_box(binary):
    # Apply S-box substitution on the binary list
    s_boxes = [S1, S2, S3, S4, S5, S6, S7, S8]
    result = []
    for i in range(8):
        start = i * 6
        end = start + 6
        row = binary_to_int([binary[start], binary[end]])
        col = binary_to_int(binary[start + 1:end - 1])
        val = s_boxes[i][row][col]
        result.extend(int_to_binary(val, min_len=4))
    return result

def binary_to_int(binary):
    # Convert binary list to integer
    return int(''.join(map(str, binary)), 2)

def display(binary, end_line=8):
    # Display binary list with a specified number of bits per line
    for i in range(len(binary)):
        if i % end_line == 0:
            print()
        print(binary[i], end=' ')
