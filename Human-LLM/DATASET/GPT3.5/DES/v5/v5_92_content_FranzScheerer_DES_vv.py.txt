import hashlib
import sys

def read_number(file_name):
    with open(file_name, 'rb') as f:
        snum = f.read()
        n = 0
        for i in range(len(snum)):
            n = (n << 8) ^ snum[len(snum)-i-1]
    return n

def inverse_modulo(b, m):
    s, t = 0, 1
    a = m
    while b != 1:
        q = a // b
        a, b = b, a % b
        s, t = t, s - q * t
    if t < 0:
        t += m
    return t

def hash_message(x):
    dx1 = hashlib.sha256(x.encode()).digest()
    res = 0
    for cx in dx1:
        res = (res << 8) ^ cx
    return res

def generate_point(y, b, prime):
    yy = (y * y - b) % prime
    x = pow(yy, inverse_modulo(3, prime-1), prime)
    return [x % prime, y % prime]

def add_points(P, Q, prime):
    x1, x2 = P[0], Q[0]
    y1, y2 = P[1], Q[1]
    while x1 < x2:
        x1 += prime
    if x1 == x2:
        s = ((3 * x1 * x1 + 0) * inverse_modulo(2 * y1, prime)) % prime
    else:
        s = ((y1 - y2) * inverse_modulo(x1 - x2, prime)) % prime
    xr = s * s - x1 - x2
    yr = s * (x1 - xr) - y1
    return [xr % prime, yr % prime]

def multiply_point(P, n, prime):
    resP = P[:]
    if n < 0:
        resP[1] = prime - resP[1]
        n = -n
    PP = resP
    isFirst = True
    while n > 0:
        if n % 2 != 0:
            resP = add_points(resP, PP, prime) if isFirst else add_points(resP, PP, prime)
            isFirst = False
        PP = add_points(PP, PP, prime)
        n //= 2
    return resP

def verify_signature(G, s, Y, e, m, prime):
    return e == hash_message(str(add_points(multiply_point(G, s, prime), multiply_point(Y, e, prime), prime)[0]) + m)

def ecdsa_verification(G, m, S, Y, prime):
    si = inverse_modulo(S[0], prime)
    hh = hash_message(m + str(S[1]))
    u1 = (si * hh) % prime
    u2 = (si * S[1]) % prime
    return add_points(multiply_point(G, u1, prime), multiply_point(Y, u2, prime), prime)[0] == S[1]
