import hashlib
import sys

def read_number(filename):
    """Reads a number from a file."""
    with open(filename, 'rb') as file:
        number = 0
        data = file.read()
        for i in range(len(data)):
            number = (number << 8) ^ ord(data[len(data) - i - 1])
    return number

# Constants
PRIME = (2 ** 160 * 5 * 23) + 86427
N_PLUS = PRIME + 1
A_COEFF = PRIME - 31
C_COEFF = 17
B_COEFF = 0

def hash_md5(x):
    """Computes the MD5 hash of input x."""
    txt = hashlib.md5(x).digest()
    res = 0
    for cx in txt:
        res = (res << 8) ^ ord(cx)
    return res

def generate_point(x, a, b):
    """Generates a point on the elliptic curve."""
    if (4 * a * a * a + 27 * b * b) % PRIME == 0:
        b += 1
    while pow(C_COEFF * x ** 3 + a * x + b, (PRIME - 1) / 2, PRIME) != 1:
        x += 1
    y = pow(C_COEFF * x ** 3 + a * x + b, (PRIME + 1) / 4, PRIME)
    return [x % PRIME, (y) % PRIME]

def add_points(P, Q):
    """Adds two points on the elliptic curve."""
    x1, x2 = P[0], Q[0]
    y1, y2 = P[1], Q[1]
    while x1 < x2:
        x1 += PRIME
    if x1 == x2:
        s = ((3 * C_COEFF * (x1 ** 2) + A_COEFF) * pow(2 * y1, PRIME - 2, PRIME)) % PRIME
    else:
        s = ((y1 - y2) * pow(x1 - x2, PRIME - 2, PRIME)) % PRIME
    xr = C_INV * s ** 2 - x1 - x2
    yr = s * (x1 - xr) - y1
    return [xr % PRIME, yr % PRIME]

def multiply_point(P, n):
    """Multiplies a point on the elliptic curve by a scalar."""
    is_first = True
    resP = P
    if n < 0:
        resP[1] = PRIME - resP[1]
        n = (-1) * n
    PP = resP
    while n > 0:
        if (n % 2 != 0):
            if is_first:
                resP = PP
                is_first = False
            else:
                resP = add_points(resP, PP)
        PP = add_points(PP, PP)
        n = n / 2
    return resP

def verify_signature(G, s, Y, e, m):
    """Verifies the signature of a message."""
    return e == hash_md5(str(add_points(multiply_point(G, s), multiply_point(Y, e))[0]) + m)
