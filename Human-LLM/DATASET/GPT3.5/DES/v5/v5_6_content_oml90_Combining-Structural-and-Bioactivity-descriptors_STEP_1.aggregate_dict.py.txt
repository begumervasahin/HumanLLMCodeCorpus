import time
import sys
import os

def print_usage():
    print('Usage: python {} [Raw hts data] [Output folder]'.format(sys.argv[0]))
    quit()

# Check if the correct number of arguments is provided
if len(sys.argv) != 3:
    print_usage()

# Get the output folder path from command line arguments
output_folder = sys.argv[2]

# Check if the output folder exists
if not os.path.isdir(output_folder):
    print("ERROR: Can't find output folder at '{}'".format(output_folder), file=sys.stderr)
    quit()

# Add a trailing slash to the output folder path if it's missing
output_folder = output_folder.rstrip('/') + '/'

header = "CID\tflag\n"
assay_list = os.listdir(sys.argv[1])
start_time = time.time()

sys.stderr.write('Assays found: {}\n'.format(len(assay_list)))

processed_assays = 0
for assay_file in assay_list:
    assay_path = os.path.join(sys.argv[1], assay_file)
    file_extension = assay_file.split('.')[-1]

    if file_extension != 'tsv':
        print("Skipping '{}': Not a .tsv file".format(assay_file))
        continue

    processed_assays += 1
    sys.stderr.write('Processing {} ({}/{})\r'.format(assay_file, processed_assays, len(assay_list)))

    if not os.path.isfile(assay_path):
        sys.stderr.write("ERROR: Can't find file '{}'\n".format(assay_file))
        continue

    compound_data = {}
    start_assay_time = time.time()

    with open(assay_path, encoding='latin-1') as assay_content:
        assay_lines = assay_content.read().splitlines()

    total_compounds = len(assay_lines)
    interesting_compounds = 0
    first_line_encountered = False
    head_rows_count = 0

    for line in assay_lines:
        if line.startswith('1'):
            first_line_encountered = True

        if line.startswith('1\t'):
            print('\nNumber of head rows: {}'.format(head_rows_count))
