from uncompyle6.main import decompile
import marshal
import sys
import os
import marcode

def decompile_python3():
    # Prompt user for output file name
    output_file = input("[*] Enter the output file name for decompiled code (Python 3.7.X):\n[?] Output file: ")
    
    # Load compiled code and decompile it
    compiled_code = marshal.loads(marcode.py3)
    decompiled_code = decompile(3.7, compiled_code, sys.stdout)
    
    # Save decompiled code to file
    save_decompiled_code(output_file, decompiled_code)

def decompile_python2():
    # Prompt user for output file name
    output_file = input("[*] Enter the output file name for decompiled code (Python 2.7.X):\n[?] Output file: ")
    
    # Load compiled code and decompile it
    compiled_code = marshal.loads(marcode.py2)
    decompiled_code = decompile(2.7, compiled_code, sys.stdout)
    
    # Save decompiled code to file
    save_decompiled_code(output_file, decompiled_code)

def save_decompiled_code(output_file, decompiled_code):
    # Save decompiled code to file
    with open(output_file + ".py", "w") as f:
        f.write(decompiled_code)
    print(f"\n\n[result] Saved as \033[95m{output_file}.py")

try:
    # Clear the console screen
    os.system('clear')
    
    # Display header
    print("""
        ;;;;;;;;;;;;;;;;;;;;;
        ; Decompile Marshal ;
        ;   By KANG-NEWBIE  ;
        ;;;;;;;;;;;;;;;;;;;;;
        """)
    
    # Check Python version and decompile accordingly
    if sys.version[0] == '3':
        decompile_python3()
    elif sys.version[0] == '2':
        decompile_python2()

except Exception as e:
    # Handle any exceptions and display error message
    print(f"Error: {e}")
