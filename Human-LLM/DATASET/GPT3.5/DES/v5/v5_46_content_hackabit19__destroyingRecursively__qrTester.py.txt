from qr import QR
from picamera.array import PiRGBArray
from picamera import PiCamera
from time import sleep

# Initialize PiCamera and QR scanner
camera = PiCamera()
qr_scanner = QR()

# Continuously capture images and scan for QR codes
while True:
    # Pause for 0.5 seconds
    sleep(0.5)

    # Capture an image
    with PiRGBArray(camera) as raw_capture:
        camera.capture(raw_capture, "bgr")
        image = raw_capture.array

    # Scan the captured image for QR codes
    qr_codes = qr_scanner.scan(image)
    for qr_code in qr_codes:
        position_x, position_y, data = qr_code
        print(f"QR Code found at position ({position_x}, {position_y}): {data}")
