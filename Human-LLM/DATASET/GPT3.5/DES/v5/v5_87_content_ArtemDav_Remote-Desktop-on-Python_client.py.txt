import socket
from PIL import ImageGrab
import io
import sys
from PyQt5.QtWidgets import QMainWindow, QApplication, QPushButton, QLineEdit
from PyQt5.QtCore import QRect
from random import randint
from threading import Thread

class DesktopClient(QMainWindow):
    def __init__(self):
        super().__init__()
        self.init_ui()

    def start_thread(self):
        self.image_thread.start()

    def grab_and_send_image(self):
        try:
            if len(self.ip.text()) != 0 and len(self.port.text()):
                sock = socket.socket()
                sock.connect((self.ip.text(), int(self.port.text())))
                while True:
                    img = ImageGrab.grab()
                    img_bytes = io.BytesIO()
                    img.save(img_bytes, format='PNG')
                    sock.send(img_bytes.getvalue())
                sock.close()
        except:
            print("DISCONNECTED")

    def init_ui(self):
        self.setWindowTitle("[CLIENT] Remote Desktop: " + str(randint(99999, 999999)))  
        self.setFixedSize(400, 90)

        self.image_thread = Thread(target=self.grab_and_send_image, daemon=True)

        self.btn = QPushButton("Start Demo", self)
        self.btn.move(5, 55)
        self.btn.resize(390, 30)
        self.btn.clicked.connect(self.start_thread)

        self.ip = QLineEdit("IP", self)
        self.ip.move(5, 5)
        self.ip.resize(390, 20)
        self.ip.setPlaceholderText("IP")

        self.port = QLineEdit("PORT", self)
        self.port.move(5, 30)
        self.port.resize(390, 20)
        self.port.setPlaceholderText("PORT")

if __name__ == '__main__':
    app = QApplication(sys.argv)
    ex = DesktopClient()
    ex.show()
    sys.exit(app.exec_())
