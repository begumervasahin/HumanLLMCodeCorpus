from config import Config
import app.story.story as story_code
from app.story.story import start_point, story_points, tasks
import re

def validate_story():
    """Validates the consistency of story.json"""
    validate_references()
    validate_lookups()

def validate_references():
    """Checks if all referenced story points and tasks in story.json are defined"""
    missing_points = set()
    missing_tasks = set()

    existing_story_points = set(story_points.keys())
    referenced_story_points = set([start_point] + [dest for point in story_points.values() for dest in point["paths"].values()])
    missing_points = referenced_story_points - existing_story_points

    existing_tasks = set(tasks.keys())
    referenced_tasks = set([task for point in story_points.values() for task in point.get("tasks", [])])
    missing_tasks = referenced_tasks - existing_tasks

    if missing_points:
        raise KeyError(f"There are referenced, but unknown story points: {missing_points}")
    if missing_tasks:
        raise KeyError(f"There are referenced, but unknown tasks: {missing_tasks}")

    print("All references in story.json found!")

def validate_lookups():
    """Checks if all task validation methods and placeholder methods referenced in story.json are defined"""
    missing_validation_methods = {
        task["validation_method"] for task in tasks.values()
        if task["validation_method"] not in dir(story_code)
    }
    if missing_validation_methods:
        raise KeyError(f"There are referenced, but unknown task validation methods: {missing_validation_methods}")
    print("All validation methods defined!")

    missing_action_methods = {
        action for point in story_points.values() for action in point.get("actions", [])
        if action not in dir(story_code)
    }
    if missing_action_methods:
        raise KeyError(f"There are referenced, but unknown server action methods: {missing_action_methods}")
    print("All server action methods defined!")

    missing_placeholders = {
        ph for point in story_points.values()
        for ph in re.findall(r"{(.*?)}", ''.join(point["description"]) + ''.join(point["paths"].keys()))
        if ph not in story_code.placeholder_getters
    }
    if missing_placeholders:
        raise KeyError(f"There are referenced, but unknown placeholders: {missing_placeholders}")
    print("All placeholder references found!")

# Entry point of the script
if __name__ == "__main__":
    validate_story()
