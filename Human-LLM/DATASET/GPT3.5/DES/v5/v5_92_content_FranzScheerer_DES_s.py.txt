def write_number(number, filename):
    with open(filename, 'wb') as file:
        while number > 0:
            byte = number % 256
            number //= 256
            file.write(chr(byte))

def binary_to_number(binary):
    result = 0
    for char in binary:
        result = (result << 8) ^ ord(char)
    return result

def number_to_binary(number):
    binary = ''
    while number > 0:
        binary = chr(number % 256) + binary
        number //= 256
    return binary

def gcd(a, b):
    while b > 0:
        a, b = b, a % b
    return a

def next_prime(p):
    while p % 8 != 3 or p % 3 != 2:
        p += 1
    return next_prime_odd(p)

def next_prime_odd(p):
    small_primes_product = 3 * 5 * 7 * 11 * 13 * 17 * 19 * 23 * 29
    while True:
        while gcd(p, small_primes_product) != 1:
            p += 24
        q = (p + 1) / 4 / 3
        if (pow(2, p - 1, p) != 1) or (pow(3, p - 1, p) != 1) or (pow(5, p - 1, p) != 1) or (pow(17, p - 1, p) != 1):
            p += 24
            continue
        break
    return p

def inverse_mod(b, m):
    s = 0
    t = 1
    a = m
    while b != 1:
        q = a // b
        a, b = b, a % b
        ss = t
        t = s - q * t
        s = ss
    if t < 0:
        t += m
    return t

def encrypt(m, key, prime):
    return pow(m, key, prime)

def decrypt(c, key, prime):
    key = inverse_mod(key, prime - 1)
    return pow(c, key, prime)

def hex_text_to_number(hex_text):
    result = 0
    for char in hex_text:
        if '0' <= char <= '9':
            result = (result << 4) + ord(char) - 48
        elif 'a' <= char <= 'f':
            result = (result << 4) + ord(char) - 87
    return result

def code_to_number(code):
    result = 0
    for char in code:
        if '0' <= char <= '9':
            result = (result << 6) + ord(char) - 48
        elif 'A' <= char <= 'Z':
            result = (result << 6) + ord(char) - 55
        elif 'a' <= char <= 'z':
            result = (result << 6) + ord(char) - 61
        elif char == "'":
            result = (result << 6) + 62
        elif char == '/':
            result = (result << 6) + 63
    return result

def number_to_code(number):
    code = ''
    while number > 0:
        remainder = number % 64
        if remainder < 10:
            code = chr(remainder + 48) + code
        elif remainder < 36:
            code = chr(remainder + 55) + code
        elif remainder < 62:
            code = chr(remainder + 61) + code
        elif remainder == 62:
            code = "'" + code
        elif remainder == 63:
            code = '/' + code
        number //= 64
    return code

def hash_function(x):
    dx1 = hashlib.sha256(x.encode()).digest()
    result = 0
    for char in dx1:
        result = (result << 8) ^ char
    return result % n4

def random_number_256(m):
    md = hashlib.sha256("***RANDOM-SEED_X***".encode())
    md.update('large key value for generation of random number'.encode())
    md.update(m.encode())
    result = 0
    largestr = md.digest()
    for i in range(len(largestr)):
        result = (result << 8) ^ ord(largestr[i])
    return result

def random_number_X(m):
    md = hashlib.sha256("***RANDOM-SEED_X***".encode())
    md.update('large key value for generation of random number'.encode())
    md.update(m.encode())
    md.update(str(time.gmtime().tm_year + time.gmtime().tm_mday).encode())
    result = 0
    largestr = md.digest()
    for i in range(len(largestr)):
        result = (result << 8) ^ ord(largestr[i])
    return result

def generate_point(x, a, b):
    while pow(x**3 + a * x + b, (prime - 1) // 2, prime) != 1:
        x += 1
    y = pow(x**3 + a * x + b, (prime + 1) // 4, prime)
    return [(x) % prime, (y) % prime]

def add_points(P, Q):
    x1, x2 = P[0], Q[0]
    y1, y2 = P[1], Q[1]
    if x2 > x1:
        x2 -= prime
    if x1 == x2:
        s = ((3 * (x1**2) + a) * inverse_mod(2 * y1, prime)) % prime
    else:
        s = ((y1 - y2) * inverse_mod(x1 - x2, prime)) % prime
    xr = s**2 - x1 - x2
    yr = s * (x1 - xr) - y1
    return [xr % prime, yr % prime]

def multiply_point(P, n):
    is_first = True
    res_P = P
    if n < 0:
        res_P[1] = prime - res_P[1]
        n = (-1) * n
    PP = res_P
    while n > 0:
        if (n & 1) != 0:
            if is_first:
                res_P = PP
                is_first = False
            else:
                res_P = add_points(res_P, PP)
        PP = add_points(PP, PP)
        n >>= 1
    return res_P

def sign_schnorr(G, m, x):
    k = random_number_X(m)
    R = multiply_point(G, k)
    e = hash_function(str(R[0]) + m)
    return [(k - x * e) % n4, e]

def sign_ecdsa(G, m, x):
    k = random_number_X(m)
    R = multiply_point(G, k)
    hh = hash_function(m + str(R[0] % n4))
    s = (inverse_mod(k, n4) * (hh + (R[0] % n4) * x)) % n4
    return [s, R[0] % n4]

def verify_ecdsa(G, m, signature, Y):
    si = inverse_mod(signature[0], n4)
    hh = hash_function(m + str(signature[1]))
    u1 = (si * hh) % n4
    u2 = (si * signature[1]) % n4
    return add_points(multiply_point(G, u1), multiply_point(Y, u2))[0] % n4 == signature[1]
