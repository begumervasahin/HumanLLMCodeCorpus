    @staticmethod
    def add_bit(bit_arr, bit):
        """Append a bit to a BitArray."""
        bit_arr.append(1)
        bit_arr[-1] = bit

    def add_padding(self, text):
        """Add padding to the plaintext."""
        temp = BitArray(text)
        temp.append(1)
        temp[-1] = 1
        pad_len = 64 - (temp.len % 64)
        if pad_len != 0:
            temp.append(pad_len)
        return temp

    def remove_padding(self, text):
        """Remove padding from the decrypted text."""
        last1 = 0
        for i, bit in enumerate(text):
            if bit == 1:
                last1 = i
        return text[:last1]

    def permute(self, block, table):
        """Permute the block using the given table."""
        temp = BitArray()
        for index in table:
            if block[index - 1] == 1:
                DES.add_bit(temp, 1)
            else:
                DES.add_bit(temp, 0)
        return temp

    def expand(self, block, table):
        """Expand the block using the given table."""
        temp = BitArray()
        for index in table:
            if block[index - 1] == 1:
                DES.add_bit(temp, 1)
            else:
                DES.add_bit(temp, 0)
        return temp

    def left_shift(self, block, n):
        """Left shift the block by n positions."""
        return block[n:] + block[:n]
