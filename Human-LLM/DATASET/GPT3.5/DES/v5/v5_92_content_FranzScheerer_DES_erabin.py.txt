import sys

# Constants
crabin = '4zmx8OD7vbXzBnvA10kQyOM1BfTckS8ZxUpmn/'
afactor = 3
bfactor = 7

# Spritz PRNG Functions
def update_spz():
    global aSPZ, iSPZ, jSPZ, wSPZ, sSPZ
    iSPZ = (iSPZ + wSPZ) % 256
    jSPZ = sSPZ[(jSPZ + sSPZ[iSPZ]) % 256]
    sSPZ[iSPZ], sSPZ[jSPZ] = sSPZ[jSPZ], sSPZ[iSPZ]

def output_spz():
    global aSPZ, iSPZ, jSPZ, wSPZ, sSPZ
    update_spz()
    return sSPZ[jSPZ]

def shuffle_spz():
    global aSPZ, iSPZ, jSPZ, wSPZ, sSPZ
    for _ in range(256):
        update_spz()
    wSPZ = (wSPZ + 2) % 256
    aSPZ = 0

def absorb_nibble_spz(x):
    global aSPZ, iSPZ, jSPZ, wSPZ, sSPZ
    if aSPZ == 240:
        shuffle_spz()
    sSPZ[aSPZ], sSPZ[240 + x] = sSPZ[240 + x], sSPZ[aSPZ]
    aSPZ += 1

def absorb_byte_spz(b):
    absorb_nibble_spz(b % 16)
    absorb_nibble_spz(b // 16)

def squeeze_spz(out, outlen):
    global aSPZ, iSPZ, jSPZ, wSPZ, sSPZ
    if aSPZ != 0:
        shuffle_spz()
    for _ in range(outlen):
        out.append(output_spz())

# Helper Functions
def hg(x):
    global aSPZ, iSPZ, jSPZ, wSPZ, sSPZ
    jSPZ = iSPZ = aSPZ = 0
    wSPZ = 1
    sSPZ = list(range(256))
    for c in x:
        absorb_byte_spz(ord(c))
    res = []
    squeeze_spz(res, 128)
    out = 0
    for bx in res:
        out = (out << 8) + bx
    return out % (2 ** 1000)

def h(x):
    global aSPZ, iSPZ, jSPZ, wSPZ, sSPZ
    jSPZ = iSPZ = aSPZ = 0
    wSPZ = 1
    sSPZ = list(range(256))
    for c in x:
        absorb_byte_spz(ord(c))
    res = []
    squeeze_spz(res, 128)
    out = 0
    for bx in res:
        out = (out << 8) + bx
    return out % (nrabin)

def code_to_num(x):
    res = 0
    for c in x:
        if '0' <= c <= '9':
            res = (res << 6) + ord(c) - 48
        elif 'A' <= c <= 'Z':
            res = (res << 6) + ord(c) - 55
        elif 'a' <= c <= 'z':
            res = (res << 6) + ord(c) - 61
        elif c == "'":
            res = (res << 6) + 62
        elif c == '/':
            res = (res << 6) + 63
    return res

def num_to_code(x):
    res = ''
    while x > 0:
        y = x % 64
        if y < 10:
            res = chr(y + 48) + res
        elif y < 36:
            res = chr(y + 55) + res
        elif y < 62:
            res = chr(y + 61) + res
        elif y == 62:
            res = "'" + res
        elif y == 63:
            res = '/' + res
        x //= 64
    return res

def gcd(a, b):
    while b > 0:
        a, b = b, a % b
    return a

def next_prime(p):
    while p % 4 != 3:
        p += 1
    return next_prime_helper(p)

def next_prime_helper(p):
    m_ = 3 * 5 * 7 * 11 * 13 * 17 * 19 * 23 * 29
    while gcd(p, m_) != 1:
        p += 4
    if pow(2, p - 1, p) != 1 or pow(3, p - 1, p) != 1 or pow(5, p - 1, p) != 1 or pow(17, p - 1, p) != 1:
        return next_prime_helper(p + 4)
    return p

def root(m, p, q):
    x = h(m)
    a = afactor
    b = bfactor
    if pow(x, (p - 1) // 2, p) > 1:
        x *= a
    if pow(x, (q - 1) // 2, q) > 1:
        x *= b
    if pow(x, (q - 1) // 2, q) != 1 or pow(x, (p - 1) // 2, p) != 1:
        print('ERROR')
        return -1
    return (pow(p, q - 2, q) * p * pow(x, (q + 1) // 4, q) + pow(q, p - 2, p) * q * pow(x, (p + 1) // 4, p)) % nrabin

def decrypt(c, p, q):
    m1 = p * (pow(p, q - 2, q) * pow(c, (q + 1) // 4, q))
    m2 = q * (pow(q, p - 2, p) * pow(c, (p + 1) // 4, p))
    ma = (m1 + m2) % nrabin
    if m1 > m2:
        mb = (m1 - m2) % nrabin
    else:
        mb = (nrabin + (m1 - m2)) % nrabin
    s_ = 2 ** 500
    return [num_to_code(ma // s_), num_to_code(mb // s_), num_to_code((nrabin - ma) // s_), num_to_code((nrabin - mb) // s_)]

def write_number(number, fnam):
    with open(fnam, 'wb') as f:
        n = number
        while n > 0:
            byte = n % 256
            n //= 256
            f.write(chr(byte))

def read_number(fnam):
    with open(fnam, 'rb') as f:
        n = 0
        snum = f.read()
        for i in range(len(snum)):
            n = (n << 8) ^ ord(snum[len(snum) - i - 1])
    return n

def hF(fnam):
    with open(fnam, 'r') as f:
        return h(f.read())

def sF(fnam):
    p = read_number("p_")
    q = read_number("q_")
    with open(fnam, 'r') as f:
        s = root(f.read(), p, q)
    return s

def f_ab(p, q):
    res = [3, 3]
    while pow(res[0], (p - 1) // 2, p) == 1 or pow(res[0], (q - 1) // 2, q) != 1:       
        res[0] += 1
    while pow(res[1], (p - 1) // 2, p) != 1 or pow(res[1], (q - 1) // 2, q) == 1:       
        res[1] += 1
    return res

def vF(s, fnam):
    a = afactor
    b = bfactor
    h0 = hF(fnam)
    ha = (a * h0) % nrabin
    hb = (b * h0) % nrabin
    hab = (a * b * h0) % nrabin
    sq = (s * s) % nrabin
    return (h0 == sq) or (ha == sq) or (hb == sq) or (hab == sq)
