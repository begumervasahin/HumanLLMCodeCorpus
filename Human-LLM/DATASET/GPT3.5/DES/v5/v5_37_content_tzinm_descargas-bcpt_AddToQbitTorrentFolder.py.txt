import logging
import os
import urllib.request
import zipfile
from telegram.ext import Updater, MessageHandler, Filters

# Configure logging
logging.basicConfig(format='%(asctime)s - %(name)s - %(levelname)s - %(message)s', level=logging.INFO)
logger = logging.getLogger(__name__)

# Function to download files
def download_file(url, path, filename):
    opener = urllib.request.build_opener()
    opener.addheaders = [('User-agent', 'Mozilla/5.0')]
    urllib.request.install_opener(opener)
    urllib.request.urlretrieve(url, os.path.join(path, filename))

# Function to list files in a directory
def list_files(directory=os.getcwd()):
    return [entry.name for entry in os.scandir(directory) if entry.is_file()]

# Function to count total members based on environment variables
def count_members(member_type):
    total_members = 0
    counter = 1
    while True:
        try:
            if os.environ[member_type + str(counter)]:
                total_members += 1
                counter += 1
        except KeyError:
            break
    return total_members

# Function to handle file downloading and processing
def handle_files(update, context):
    allowed_members = []

    # Load allowed members from environment variables
    allowed_members.extend(int(os.environ[member + str(i)]) for member in ["usuario", "grupo"] for i in range(1, count_members(member) + 1))

    if update.message.chat_id not in allowed_members:
        update.message.reply_text("No tienes permisos suficientes para utilizar el bot.")
        return

    download_dir = '/home/descargas/'
    temp_dir = '/zip/'
    message = update.message

    if message.document:
        filename = message.document.file_name
        document = context.bot.getFile(message.document.file_id)

        if filename.endswith('.zip'):
            download_file(document.file_path, temp_dir, filename)
            with zipfile.ZipFile(temp_dir + filename, "r") as zip_file:
                for file_entry in zip_file.namelist():
                    if os.path.dirname(file_entry) == '' and file_entry.endswith('.torrent'):
                        zip_file.extract(file_entry, download_dir)
            os.remove(temp_dir + filename)
            update.message.reply_text(f"Los torrents en el archivo <b>{filename}</b> han sido descomprimidos y agregados al directorio correspondiente.", parse_mode="HTML")

        elif filename.endswith('.torrent'):
            download_file(document.file_path, download_dir, filename)
            update.message.reply_text(f"El archivo <b>{filename}</b> ha sido agregado correctamente al directorio correspondiente.", parse_mode="HTML")

# Function to handle errors
def handle_error(update, context):
    logger.warning(f'Update "{update}" caused error "{context.error}"')

# Main function to start the bot
def main():
    updater = Updater(os.environ['TOKEN'], use_context=True)
    dispatcher = updater.dispatcher
    dispatcher.add_handler(MessageHandler(Filters.document, handle_files))
    dispatcher.add_error_handler(handle_error)
    updater.start_polling(clean=True)
    updater.idle()

if __name__ == '__main__':
    main()
