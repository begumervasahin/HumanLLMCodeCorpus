class TOperation:
    """
    Represents a single operation performed in the program.

    Attributes:
        operation_type (str): The type of operation.
        shape (object): The shape on which the operation was performed.        
        shape_index (int): The index of the shape in the shape list.
    """

    def __init__(self, operation_type=None, **kwargs):
        """
        Initializes the TOperation object.

        Args:
            operation_type (str): The type of operation.
            **kwargs: Additional keyword arguments.
        """
        self.set_operation(operation_type, **kwargs)

    def set_operation(self, operation_type=None, **kwargs):
        """
        Set the parameters of an operation.

        Args:
            operation_type (str): The type of operation.
            **kwargs: Additional keyword arguments.
                shape (object): The shape on which the operation was performed.
                shape_index (int): The index of the shape in the shape list.
        """
        self.operation_type = operation_type
        self.shape = kwargs.pop('shape', None)
        self.shape_index = kwargs.pop('shape_index', None)
