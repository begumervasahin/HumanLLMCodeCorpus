def DES_decryption(data_bytes, key_bytes):
    """
    Decrypts the file using the DES algorithm.

    Args:
        data_bytes (list): The list of encrypted data bytes.
        key_bytes (list): The list of key bytes used for decryption.

    Returns:
        list: The list of decrypted data bytes.
    """
    print("Decrypting file ...")
    print("Key bytes:", key_bytes)
    print("Cipher data bytes:", data_bytes)
    print("")

    out_data = data_bytes

    # Iterate over DES rounds
    for rounds in range(8):
        # Iterate over data bytes in pairs
        for i in range(0, len(data_bytes), 2):
            # Perform decryption on every other pair of data bytes
            if i % 2 == 0:
                data_bytes_slice = data_bytes[i:i + 2]
                key_byte = key_bytes[7 - rounds]
                out_data[i:i + 2] = iteration(data_bytes_slice, key_byte)

    return out_data

def iteration(data_bytes, key_byte):
    """
    Performs a single iteration of DES decryption.

    Args:
        data_bytes (list): The list of data bytes for the iteration.
        key_byte (int): The key byte used for the iteration.

    Returns:
        list: The list of decrypted data bytes.
    """
    data_L, data_R = data_bytes
    data_L_dash = data_R ^ key_byte
    data_R_dash = data_L
    return [data_L_dash, data_R_dash]
