from Crypto.Cipher import Blowfish
from Crypto.Util.Padding import pad
from Crypto.Random import get_random_bytes

# Define the encryption key and plaintext
key = b'An arbitrarily long key'
plaintext = b'docendo discimus'

# Initialize the Blowfish cipher in CBC mode with a random IV
cipher = Blowfish.new(key, Blowfish.MODE_CBC, iv=get_random_bytes(Blowfish.block_size))

# Pad the plaintext to make its length a multiple of the block size
padded_plaintext = pad(plaintext, Blowfish.block_size)

# Encrypt the padded plaintext
encrypted_message = cipher.encrypt(padded_plaintext)

# Output the initialization vector (IV) followed by the encrypted message
output_message = cipher.iv + encrypted_message

print(output_message)
