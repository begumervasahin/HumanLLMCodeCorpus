import sys
import getpass

# Global variables for Rabin Signature
nrabin = 0

# Rabin Signature related functions

def update_SPZ():
    """Update the SPZ variables."""
    global aSPZ, iSPZ, jSPZ, wSPZ, sSPZ
    iSPZ = (iSPZ + wSPZ) % 256
    jSPZ = sSPZ[(jSPZ + sSPZ[iSPZ]) % 256]
    sSPZ[iSPZ], sSPZ[jSPZ] = sSPZ[jSPZ], sSPZ[iSPZ]

def output_SPZ():
    """Output the SPZ."""
    global aSPZ, iSPZ, jSPZ, wSPZ, sSPZ
    update_SPZ()
    return sSPZ[jSPZ]

def shuffle_SPZ():
    """Shuffle the SPZ."""
    global aSPZ, iSPZ, jSPZ, wSPZ, sSPZ
    for _ in range(256):
        update_SPZ()
    wSPZ = (wSPZ + 2) % 256
    aSPZ = 0

def absorb_nibble_SPZ(x):
    """Absorb a nibble in the SPZ."""
    global aSPZ, iSPZ, jSPZ, wSPZ, sSPZ
    if aSPZ == 240:
        shuffle_SPZ()
    sSPZ[aSPZ], sSPZ[240 + x] = sSPZ[240 + x], sSPZ[aSPZ]
    aSPZ += 1

def absorb_byte_SPZ(b):
    """Absorb a byte in the SPZ."""
    absorb_nibble_SPZ(b % 16)
    absorb_nibble_SPZ(b // 16)

def squeeze_SPZ(out, outlen):
    """Squeeze the SPZ."""
    global aSPZ, iSPZ, jSPZ, wSPZ, sSPZ
    if aSPZ != 0:
        shuffle_SPZ()
    for _ in range(outlen):
        out.append(output_SPZ())

def h_global(x):
    """Global hash function."""
    global aSPZ, iSPZ, jSPZ, wSPZ, sSPZ, nrabin
    jSPZ = iSPZ = aSPZ = 0
    wSPZ = 1
    sSPZ = list(range(256))
    for c in x:
        absorb_byte_SPZ(ord(c))
    res = []
    squeeze_SPZ(res, 128)
    out = 0
    for bx in res:
        out = (out << 8) + bx
    return out % (2 ** 1000)

def h(x):
    """Hash function."""
    global aSPZ, iSPZ, jSPZ, wSPZ, sSPZ, nrabin
    jSPZ = iSPZ = aSPZ = 0
    wSPZ = 1
    sSPZ = list(range(256))
    for c in x:
        absorb_byte_SPZ(ord(c))
    res = []
    squeeze_SPZ(res, 128)
    out = 0
    for bx in res:
        out = (out << 8) + bx
    return out % nrabin

def code_to_num(x):
    """Convert code to number."""
    res = 0
    for c in x:
        if 48 <= ord(c) < 58:
            res = (res << 6) + ord(c) - 48
        if 65 <= ord(c) < 91:
            res = (res << 6) + ord(c) - 55
        if 97 <= ord(c) < 123:
            res = (res << 6) + ord(c) - 61
        if c == "'":
            res = (res << 6) + 62
        if c == '/':
            res = (res << 6) + 63
    return res

def num_to_code(x):
    """Convert number to code."""
    res = ''
    while x > 0:
        y = x % 64
        if y < 10:
            res = chr(y + 48) + res
        elif y < 36:
            res = chr(y + 55) + res
        elif y < 62:
            res = chr(y + 61) + res
        elif y == 62:
            res = "'" + res
        elif y == 63:
            res = '/' + res
        x //= 64
    return res

def gcd(a, b):
    """Calculate the greatest common divisor."""
    while b > 0:
        a, b = b, a % b
    return a

def next_prime(p):
    """Find the next prime number."""
    while p % 4 != 3:
        p += 1
    return next_prime_(p)

def next_prime_(p):
    """Find the next prime number starting from p."""
    m_ = 3 * 5 * 7 * 11 * 13 * 17 * 19 * 23 * 29
    while gcd(p, m_) != 1:
        p += 4
    if pow(2, p - 1, p) != 1:
        return next_prime_(p + 4)
    if pow(3, p - 1, p) != 1:
        return next_prime_(p + 4)
    if pow(5, p - 1, p) != 1:
        return next_prime_(p + 4)
    if pow(17, p - 1, p) != 1:
        return next_prime_(p + 4)
    return p

def f_ab(p, q):
    """Calculate factors a and b."""
    res = [3, 3]
    while pow(res[0], (p - 1) / 2, p) == 1 or pow(res[0], (q - 1) / 2, q) != 1:
        res[0] += 1
    while pow(res[1], (p - 1) / 2, p) != 1 or pow(res[1], (q - 1) / 2, q) == 1:
        res[1] += 1
    return res

def root(m, p, q):
    """Calculate the root."""
    global nrabin
    x = h(m)
    ab = f_ab(p, q)
    a = ab[0]
    b = ab[1]
    nrabin = p * q
    if pow(x, (p - 1) / 2, p) > 1:
        x *= a
    if pow(x, (q - 1) / 2, q) > 1:
        x *= b
    if pow(x, (q - 1) / 2, q) != 1 or pow(x, (p - 1) / 2, p) != 1:
        print('ERROR')
        return -1
    return (pow(p, q - 2, q) * p * pow(x, (q + 1) / 4, q) + pow(q, p - 2, p) * q * pow(x, (p + 1) / 4, p)) % nrabin

def write_number(number, fnam):
    """Write the number to a file."""
    f = open(fnam, 'wb')
    n = number
    while n > 0:
        byte = n % 256
        n //= 256
        f.write(chr(byte))
    f.close()

def read_number(fnam):
    """Read the number from a file."""
    f = open(fnam, 'rb')
    n = 0
    snum = f.read()
    for i in range(len(snum)):
        n = (n << 8) ^ ord(snum[len(snum) - i - 1])
    f.close()
    return n

def hash_file(fnam):
    """Hash the contents of a file."""
    f = open(fnam, 'r')
    return h_global(f.read())

def sign_file(fnam, pw):
    """Sign a file."""
    global nrabin
    print("Generating primes...")
    p = next_prime(h_global(pw) % (2 ** 501 + 1))
    q = next_prime(h_global(pw + '0') % (2 ** 501 + 1))
    nrabin = p * q
    f = open(fnam, 'r')
    s = root(f.read(), p, q)
    f.close()
    return s

    p = next_prime(h_global(sys.argv[2]) % (2 ** 501 + 1))
    q = next_prime(h_global(sys.argv[2] + '0') % (2 ** 501 + 1))
    write_number(p, 'p_')
    write_number(q, 'q_')
    print("Modulus:", p * q)
    print("\nModulus code:\n", num_to_code(p * q))
    print("A factor:", f_ab(p, q)[0])
