class Alert:
    def __init__(self):
        self.fine_turn = 0
        self.bo_flag = 0
        self.bp_flag = 0
        self.pul_flag = 0

    def update_flag(self, flag, threshold_lower, threshold_upper, data):
        """
        Update flag based on data and threshold.
        """
        if self.fine_turn > 20:
            flag = 0
        if threshold_lower <= data <= threshold_upper:
            flag += 1
        else:
            self.fine_turn += 1
        return flag

    def get_bo_data(self, data):
        """
        Update blood oxygen flag based on data.
        """
        self.bo_flag = self.update_flag(self.bo_flag, 0, 0.3, data)

    def get_bp_data(self, data):
        """
        Update blood pressure flag based on data.
        """
        self.bp_flag = self.update_flag(self.bp_flag, 60, 90, data)

    def get_pul_data(self, data):
        """
        Update pulse flag based on data.
        """
        self.pul_flag = self.update_flag(self.pul_flag, 90, 120, data)

    def alert_output(self):
        """
        Determine which flag is raised and return corresponding alert code.
        """
        if self.bo_flag != 0:
            return 1  # Alert for blood oxygen
        elif self.bp_flag != 0:
            return 2  # Alert for blood pressure
        elif self.pul_flag != 0:
            return 3  # Alert for pulse
        else:
            return 0  # No alert
