from Crypto.Cipher import Blowfish
from struct import pack

# Define the block size of the Blowfish cipher
block_size = Blowfish.block_size

# Define the encryption key
key = b'An arbitrarily long key'

# Initialize the Blowfish cipher in CBC mode with the given key
cipher = Blowfish.new(key, Blowfish.MODE_CBC)

# Define the plaintext to be encrypted
plaintext = b'docendo discimus '

# Calculate the padding length required to make the plaintext length a multiple of the block size
padding_length = block_size - len(plaintext) % block_size

# Create padding bytes with the calculated length
padding = bytes([padding_length] * padding_length)

# Pack the padding bytes into a string
padding = pack('b' * padding_length, *padding)

# Encrypt the plaintext along with the padding using CBC mode
encrypted_message = cipher.iv + cipher.encrypt(plaintext + padding)
