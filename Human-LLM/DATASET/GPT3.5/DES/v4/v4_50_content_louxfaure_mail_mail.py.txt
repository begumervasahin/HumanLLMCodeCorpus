import os
import smtplib
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText

# Fetch environment variables
mail_login = os.getenv('MAIL_LOGIN')
mail_pwd = os.getenv('MAIL_PWD')
smtp_server = os.getenv('SMTP_SERVER')
smtp_port = os.getenv('SMTP_PORT')

class EmailSender:
    def __init__(self, mail_login=mail_login, mail_pwd=mail_pwd, smtp_server=smtp_server, smtp_port=smtp_port):
        if not mail_login:
            raise ValueError("Please provide a valid email login")
        if not mail_pwd:
            raise ValueError("Please provide a valid email password")
        if not smtp_server:
            raise ValueError("Please provide a valid SMTP server")
        if not smtp_port:
            raise ValueError("Please provide a valid SMTP port")

        self.mail_login = mail_login
        self.mail_pwd = mail_pwd
        self.smtp_server = smtp_server
        self.smtp_port = smtp_port

    def send_email(self, sender_email, recipient_email, subject, body):
        msg = MIMEMultipart()
        msg['From'] = sender_email
        msg['To'] = recipient_email
        msg['Subject'] = subject
        msg.attach(MIMEText(body))

        try:
            with smtplib.SMTP_SSL(self.smtp_server, self.smtp_port) as server:
                server.login(self.mail_login, self.mail_pwd)
                server.sendmail(sender_email, recipient_email, msg.as_string())
            return "Email sent successfully"
        except Exception as e:
            return f"Failed to send email: {e}"

# Example usage
email_sender = EmailSender()

result = email_sender.send_email('sender@example.com', 'recipient@example.com', 'Test Subject', 'This is a test email.')
print(result)
