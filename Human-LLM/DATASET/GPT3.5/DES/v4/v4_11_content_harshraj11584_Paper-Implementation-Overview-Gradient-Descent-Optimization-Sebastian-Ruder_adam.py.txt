import numpy as np
from sympy import *

# Set up pretty printing and initialize SymPy
init_printing()
print('\n\nInitial Setup Complete')

def adam(c_f, f):
    # Display the input cost function and its derivative
    print("Cost function f(x) =", c_f)
    f_dash = diff(c_f, x)
    print("Derivative df(x)/dx =", f_dash)

    # Get user input for initial parameters
    a = float(input("Enter initial approximation: "))
    x0 = a
    err = float(input("Enter error tolerance: "))
    n = float(input("Enter learning rate: "))

    # Print initial values and start Adam optimization
    print("Starting Adam")
    print("Initial approximation x0 =", a)
    print("f(x0) =", f(a))

    iter_count = 0
    xk = x0
    x_prev = 0.0
    m0 = 0.0
    mk = 0.0
    v0 = 0.0
    vk = 0.0
    b1 = 0.9
    b2 = 0.999
    epsilon = 10**-8

    while True:
        iter_count += 1
        x_prev = x0
        x0 = xk
        m0 = mk
        v0 = vk

        fk_dash = (lambdify(x, f_dash, "numpy"))(xk)
        gt = fk_dash
        mk = b1 * m0 + (1 - b1) * gt
        vk = b2 * v0 + (1 - b2) * (gt**2)
        mc_k = mk / (1 - b1**iter_count)
        vc_k = vk / (1 - b2**iter_count)

        xk = xk - n * mc_k / ((vc_k**0.5) + epsilon)

        # Check convergence criteria
        if abs(N(xk - x0)) < float(err) or abs(N(xk - x_prev)) < 0.1 * float(err):
            break

    # Print optimization results
    print("Number of Iterations =", iter_count)
    print("Minima is at =", xk)
    print("Minimum value of Cost Function =", f(xk))

# Define the symbol x
x = Symbol('x')

# Print syntax constraints for the cost function
print("""

- Function must be an algebraic combination of one or more of the following:
    p(x)      Polynomials
    exp(x)    Mathematical constant e (2.71828...) raised to power x
    pi        Mathematical constant 3.14159...

    acos(x)   Arc cosine of x
    asin(x)   Arc sine of x
    atan(x)   Arc tangent of x
    cos(x)    Cosine of x
    sin(x)    Sine of x
    tan(x)    Tangent of x
    acosh(x)  Inverse hyperbolic cosine of x
    asinh(x)  Inverse hyperbolic sine of x
    atanh(x)  Inverse hyperbolic tangent of x
    cosh(x)   Hyperbolic cosine of x
    sinh(x)   Hyperbolic cosine of x
    tanh(x)   Hyperbolic tangent of x
""")

# Get user input for the cost function and convert it to a symbolic expression
cost_function = input("Enter cost function f(x): ").strip()
c_f = sympify(cost_function)

# Convert the symbolic expression to a function that can be evaluated numerically
f = lambdify(x, c_f, "numpy")

# Call the Adam optimization function
adam(c_f, f)
