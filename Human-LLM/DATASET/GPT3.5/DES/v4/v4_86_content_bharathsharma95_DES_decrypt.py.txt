def DES_decryption(data_bytes, key_bytes):
    """
    Decrypts the file using DES algorithm.

    Args:
        data_bytes (list): The encrypted data bytes.
        key_bytes (list): The key bytes used for decryption.

    Returns:
        list: The decrypted data bytes.
    """
    print("Decrypting file ...")
    print("")
    print("Key bytes: " + str(key_bytes))
    print("")
    print("Cipher data bytes: " + str(data_bytes))
    print("")

    out_data = data_bytes

    for rounds in range(8):
        for i in range(len(data_bytes)):
            if i % 2 == 0:
                data_bytes_slice = [data_bytes[i], data_bytes[i+1]]
                key_byte = key_bytes[7 - rounds]
                out_data[i:i+2] = iteration(data_bytes_slice, key_byte)

    return out_data

def iteration(data_bytes, key_byte):
    """
    Performs a single iteration of DES decryption.

    Args:
        data_bytes (list): The data bytes for the iteration.
        key_byte (int): The key byte used for the iteration.

    Returns:
        list: The decrypted data bytes.
    """
    data_L = data_bytes[0]
    data_R = data_bytes[1]
    data_L_dash = data_R ^ key_byte
    data_R_dash = data_L
    out_data = [data_L_dash, data_R_dash]
    return out_data
