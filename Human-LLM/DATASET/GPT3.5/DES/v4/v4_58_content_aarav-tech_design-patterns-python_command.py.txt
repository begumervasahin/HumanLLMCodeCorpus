from abc import ABC, abstractmethod

# Base command class
class BaseCommand(ABC):
    @abstractmethod
    def execute(self):
        """Abstract method to execute the command."""
        raise NotImplementedError("Please implement in subclass")

# Email command class
class EmailCommand(BaseCommand):
    def __init__(self, receiver, data):
        """Initialize the EmailCommand with a receiver and email data."""
        self.receiver = receiver
        self.data = data

    def execute(self):
        """Execute the email command by sending an email through the receiver."""
        self.receiver.send_email(self.data)

# SMS command class
class SMSCommand(BaseCommand):
    def __init__(self, receiver, data):
        """Initialize the SMSCommand with a receiver and SMS data."""
        self.receiver = receiver
        self.data = data

    def execute(self):
        """Execute the SMS command by sending an SMS through the receiver."""
        self.receiver.send_sms(self.data)

# Receiver class
class NotificationService:
    def send_email(self, data):
        """Send an email with the given data."""
        print("Sending email:", data)

    def send_sms(self, data):
        """Send an SMS with the given data."""
        print("Sending SMS:", data)

# Invoker class
class NotificationInvoker:
    def __init__(self):
        """Initialize the NotificationInvoker with an empty notification history."""
        self.notification_history = []

    def invoke(self, command):
        """Invoke a command by executing it and appending it to the notification history."""
        self.notification_history.append(command)
        command.execute()

if __name__ == "__main__":
    # Example usage
    invoker = NotificationInvoker()
    receiver = NotificationService()

    # Create email and SMS commands and invoke them
    invoker.invoke(EmailCommand(receiver, {"subject": "Test Email"}))
    invoker.invoke(SMSCommand(receiver, {"subject": "Test SMS"}))

    # Print the notification history
    print("Notification History:", invoker.notification_history)
