import multiprocessing
import logging
import sys
from order_bot import order_bot_processing
from castle_bot import castle_bot_processing
from castle_files.bin.telethon_script import auth

# Configure logging
console_handler = logging.StreamHandler()
console_handler.setLevel(logging.INFO)

log_file_handler = logging.FileHandler(filename='error.log', mode='a')
log_file_handler.setLevel(logging.ERROR)

logging.basicConfig(
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    level=logging.INFO,
    handlers=[log_file_handler, console_handler]
)

def start_telethon_auth():
    """Start telethon authentication."""
    logging.info("Starting telethon auth")
    auth()
    sys.exit(0)

if __name__ == "__main__":
    # Check for command-line arguments
    if "--auth" in sys.argv:
        start_telethon_auth()

    # Start bot processes
    order_bot_process = multiprocessing.Process(target=order_bot_processing)
    castle_bot_process = multiprocessing.Process(target=castle_bot_processing)

    order_bot_process.start()
    castle_bot_process.start()

    # Wait for the processes to complete
    try:
        order_bot_process.join()
    except KeyboardInterrupt:
        pass

    print("ended")
