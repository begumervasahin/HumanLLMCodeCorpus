class DESDecrypt:
    def __init__(self):
        pass

    def decode(self, string, key, key_len, string_len):
        output = ""
        code_string = self._function_char_to_a(string, string_len)
        code_key = self._get_key(key, key_len)
        real_len = (key_len // 16) + 1 if key_len % 16 != 0 else key_len // 16
        turn_len = string_len * 4

        num = 0
        for i in range(0, turn_len, 64):
            run_code = code_string[i:i + 64]
            run_key = code_key[num % real_len]
            run_code = self._code_first_change(run_code)

            for j in range(16):
                code_r = run_code[32:64]
                code_l = run_code[0:32]
                run_code = code_r
                code_r = self._function_e(code_r)
                key_y = run_key[15 - j]
                code_r = self._code_xor(code_r, key_y)
                code_r = self._function_s(code_r)
                code_r = self._function_p(code_r)
                code_r = self._code_xor(code_l, code_r)
                run_code += code_r

            num += 1
            code_r = run_code[32:64]
            code_l = run_code[0:32]
            run_code = code_r + code_l
            output += self._function_code_change(run_code)

        return output

    @staticmethod
    def _get_key(key, key_len):
        code_key = DESDecrypt._function_char_to_a(key, key_len)
        real_len = (key_len // 16) * 16 + 16 if key_len % 16 != 0 else key_len
        b = [[''] * 16 for _ in range(real_len // 16)]
        num = 0
        turn_len = 4 * key_len

        for i in range(0, turn_len, 64):
            run_key = code_key[i:i + 64]
            run_key = DESDecrypt._key_first_change(run_key)

            for j in range(16):
                key_l = run_key[0:28]
                key_r = run_key[28:56]
                key_l = key_l[d[j]:28] + key_l[0:d[j]]
                key_r = key_r[d[j]:28] + key_r[0:d[j]]
                run_key = key_l + key_r
                key_y = DESDecrypt._function_key_second_change(run_key)
                b[num][j] = key_y[:]

            num += 1

        return b

    @staticmethod
    def _code_xor(code, key):
        code_len = len(key)
        return_list = ''
        for i in range(code_len):
            if code[i] == key[i]:
                return_list += '0'
            else:
                return_list += '1'
        return return_list

    @staticmethod
    def _code_first_change(code):
        changed_code = ''
        for i in range(64):
            changed_code += code[ip[i] - 1]
        return changed_code

    @staticmethod
    def _function_code_change(code):
        return_list = ''
        for i in range(16):
            temp_list = ''
            for j in range(4):
                temp_list += code[ip_1[i * 4 + j] - 1]
            return_list += "%x" % int(temp_list, 2)
        return return_list

    @staticmethod
    def _function_e(code):
        return_list = ''
        for i in range(48):
            return_list += code[e[i] - 1]
        return return_list

    @staticmethod
    def _function_p(code):
        return_list = ''
        for i in range(32):
            return_list += code[p[i] - 1]
        return return_list

    @staticmethod
    def _function_s(key):
        return_list = ''
        for i in range(8):
            row = int(str(key[i * 6]) + str(key[i * 6 + 5]), 2)
            raw = int(str(key[i * 6 + 1]) + str(key[i * 6 + 2]) +
                      str(key[i * 6 + 3]) + str(key[i * 6 + 4]), 2)
            return_list += DESDecrypt._function_to_s(s[i][row][raw], 4)
        return return_list

    @staticmethod
    def _function_key_second_change(key):
        return_list = ''
        for i in range(48):
            return_list += key[pc2[i] - 1]
        return return_list

    @staticmethod
    def _function_char_to_a(code, lens):
        return_code = ''
        lens = lens % 16
        for key in code:
            code_ord = int(key, 16)
            return_code += DESDecrypt._function_to_s(code_ord, 4)
        if lens != 0:
            return_code += '0' * (16 - lens) * 4
        return return_code

    @staticmethod
    def _function_to_s(o, lens):
        return_code = ''
        for i in range(lens):
            return_code = str(o >> i & 1) + return_code
        return return_code
