# Open the 'roulement.csv' file for reading
with open("resources/roulement.csv", 'r') as roulement_file:
    # Initialize a list to store the data from the CSV file
    data = []

    # Read the header line, split it into columns, and store it in 'entete'
    header_line = roulement_file.readline().strip()
    entete = header_line.split('\t')

    # Read each line of the CSV file, split it into columns, and append it to 'data'
    for line in roulement_file:
        row = line.strip().split('\t')
        data.append(row)

# Print the indices of each row in the 'data' list along with the first element of each row
for i, row in enumerate(data):
    print(str(i) + " " + str(row[0]))

    # Split each element of the row by commas
    for j in range(len(row)):
        row[j] = row[j].split(',')

# Define a function 'rlmnt' that takes 'date' and 'kh' as parameters
def rlmnt(date, kh):
    """Returns the indices of columns in the 'entete' list."""
    j = -1
    liste = []

    # Iterate through each row in 'data' to find the row containing 'date'
    for i, row in enumerate(data):
        if date in row[0]:
            j = i

    # If 'date' is found in 'data', search for 'kh' in the corresponding row
    if j < 0:
        print("Date not found!")
        return None
    else:
        for k in range(len(data[j])):
            try:
                data[j][k].index(str(kh))
                liste.append(k)
            except ValueError:
                pass
    return liste
