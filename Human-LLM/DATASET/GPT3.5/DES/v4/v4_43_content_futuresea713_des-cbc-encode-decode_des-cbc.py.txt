import pyDes
import base64

# Define the data to be encrypted
data = "abc123"

# Create a DES encryption object with the specified key and mode
key = 'g9G16nTs'
des_cipher = pyDes.des(key, pyDes.CBC, key, pad=None, padmode=pyDes.PAD_PKCS5)     

# Encrypt the data
encrypted_data = des_cipher.encrypt(data)

# Encode the encrypted data using base64
encoded_encrypted_data = base64.b64encode(encrypted_data)

# Decode the base64 encoded encrypted data
decoded_encrypted_data = base64.b64decode(encoded_encrypted_data)

# Decrypt the decoded encrypted data
decrypted_data = des_cipher.decrypt(decoded_encrypted_data)

# Print the encrypted and decrypted data
print("Encrypted:", encoded_encrypted_data)
print("Decrypted:", decrypted_data.decode())
