import requests
import json
import warnings

# Suppress SSL warnings
warnings.filterwarnings('ignore')

# Define the URL, headers, and data payload for the request
url = "https://172.27.26.163:9999/des"
headers = {
    'Content-type': 'application/json',
    'Origin': 'https://172.27.26.163:9999',
    'Referer': 'https://172.27.26.163:9999/game/caves.swf'
}
data = {
    "password": "555acfd4d3c1b7a398d71e4c1538ad4d",
    "teamname": "Amalgam",
    "plaintext": "password"
}

# Open input file for reading plaintext passwords
with open('input.txt') as input_file:
    plaintexts = input_file.readlines()

# Open output file for writing ciphertext passwords
with open('response.txt', 'w+') as output_file:
    # Parse data payload from JSON string
    data = json.loads(json.dumps(data))

    # Iterate through each line in the input file
    for line in plaintexts:
        plaintext = line.split(',')[0].strip()  # Extract plaintext from the line
        data["plaintext"] = plaintext  # Update the plaintext in the data payload

        # Send POST request to the server
        response = requests.post(url, json=data, headers=headers, verify=False)

        # Check if the request was successful (status code 200)
        if response.status_code == 200:
            ciphertext = json.loads(response.text)["ciphertext"]  # Extract ciphertext from the response
            output_file.write(f"{ciphertext}\n")  # Write the ciphertext to the output file
        else:
            print("Failed to retrieve ciphertext")

# Close input and output files
input_file.close()
output_file.close()
