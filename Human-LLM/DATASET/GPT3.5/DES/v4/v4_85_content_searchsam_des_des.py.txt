import os
import sys
import subprocess

class ProjectCreator:
    def __init__(self):
        self.project_types = ["laravel", "vue", "composer"]
        self.current_file = os.path.realpath(__file__)
        self.current_directory = os.getcwd()

    def check_params(self):
        """Check if command-line arguments are provided."""
        return len(sys.argv) <= 1

    def call(self, project_type, num_arg=None):
        """Call the appropriate project creation method."""
        return getattr(self, "create_" + str(project_type))(num_arg)

    def create_laravel(self, num_arg=None):
        """Create a Laravel project."""
        if num_arg is None:
            num_arg = 1
        os.chdir(self.current_directory)
        print("Creating Laravel project...")
        self.execute_command([self.project_types[0], "new", "api.{}".format(sys.argv[num_arg])])
        os.chdir(os.path.join(self.current_directory, "api.{}".format(sys.argv[num_arg])))
        print("Installing Composer dependencies...")
        self.execute_command([self.project_types[2], "require", "nuwave/lighthouse", "mll-lab/laravel-graphql-playground"])
        print("Publishing Lighthouse configuration...")
        self.execute_command(["php", "artisan", "vendor:publish", '--provider="Nuwave\\Lighthouse\\LighthouseServiceProvider"'])
        print("Publishing GraphQL Playground configuration...")
        self.execute_command(["php", "artisan", "vendor:publish", '--provider="MLL\\GraphQLPlayground\\GraphQLPlaygroundServiceProvider"'])

    def create_vue(self, num_arg=None):
        """Create a Vue project."""
        if num_arg is None:
            num_arg = -1
        os.chdir(self.current_directory)
        print("Creating Vue project...")
        os.system("{0} create {1}".format(self.project_types[1], sys.argv[num_arg]))    
        os.chdir(os.path.join(self.current_directory, sys.argv[num_arg]))
        print("Adding Apollo to Vue project...")
        os.system("{} add apollo".format(self.project_types[1]))

    def execute_command(self, command):
        """Execute a shell command."""
        stdout, stderr = subprocess.Popen(command, stdout=subprocess.PIPE, stderr=subprocess.STDOUT).communicate()
        if stderr is not None:
            print("The command {} does not exist or was executed incorrectly.".format(command[0]))    
            sys.exit(1)
        return True

    def scaffold_projects(self):
        """Create scaffold projects based on command-line arguments."""
        if self.check_params():
            print("Invalid or missing arguments.")
            sys.exit(1)
        if len(sys.argv) == 3 and sys.argv[2] not in self.project_types:
            self.call(self.project_types[0])
            self.call(self.project_types[1])
            return True
        if len(sys.argv) >= 4 and sys.argv[-1] not in self.project_types:
            for project_type in [self.project_types[0], self.project_types[1]]:
                if project_type in sys.argv:
                    index = sys.argv.index(project_type) + 1
                    if sys.argv[index] not in self.project_types:
                        self.call(project_type, index)
                    else:
                        self.call(project_type)

    def run_docker(self):
        """Run Docker."""
        print("Running Docker...")

    def run_podman(self):
        """Run Podman."""
        print("Running Podman...")

if __name__ == "__main__":
    project_creator = ProjectCreator()
    {
        "scaffold": project_creator.scaffold,
        "docker": project_creator.run_docker,
        "podman": project_creator.run_podman
    }[sys.argv[1]]()
