from openerp.osv import osv, fields

class ResRegion(osv.osv):
    _name = 'res.region'
    
    _columns = {
        'name': fields.Char('Region Name'),
    }

class ResPartner(osv.osv):
    _inherit = 'res.partner'
    
    _columns = {
        'region': fields.Many2one('res.region', string="Client Region"),
    }
