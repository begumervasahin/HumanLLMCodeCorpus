from tkinter import simpledialog, Label, Checkbutton, IntVar, Frame, W

class TraceWindow(simpledialog.Dialog):
    """
    Represents a popup window used to choose trace components and FFT.
    """

    def __init__(self, master, *, show_fft=True):
        """
        Initialize object variables and call the parent class constructor.
        """
        self.show_fft = show_fft
        super().__init__(master)

    def body(self, master):
        """
        Initialize widgets.
        """
        Label(self, text="Choose component(s) to plot:").pack()
        button_width = 3
        self.button_frame = Frame(self)

        # Trace component checkboxes
        self.ex_en = IntVar()
        self.ex = Checkbutton(self.button_frame, text="Ex", variable=self.ex_en,
                              width=button_width, anchor=W)
        self.ex.select()
        self.ex.grid(row=0, column=0)

        self.ey_en = IntVar()
        self.ey = Checkbutton(self.button_frame, text="Ey", variable=self.ey_en,
                              width=button_width, anchor=W)
        self.ey.select()
        self.ey.grid(row=0, column=1)

        self.ez_en = IntVar()
        self.ez = Checkbutton(self.button_frame, text="Ez", variable=self.ez_en,
                              width=button_width, anchor=W)
        self.ez.select()
        self.ez.grid(row=0, column=2)

        self.hx_en = IntVar()
        self.hx = Checkbutton(self.button_frame, text="Hx", variable=self.hx_en,
                              width=button_width, anchor=W)
        self.hx.select()
        self.hx.grid(row=1, column=0)

        self.hy_en = IntVar()
        self.hy = Checkbutton(self.button_frame, text="Hy", variable=self.hy_en,
                              width=button_width, anchor=W)
        self.hy.select()
        self.hy.grid(row=1, column=1)

        self.hz_en = IntVar()
        self.hz = Checkbutton(self.button_frame, text="Hz", variable=self.hz_en,
                              width=button_width, anchor=W)
        self.hz.select()
        self.hz.grid(row=1, column=2)

        self.button_frame.pack()

        # FFT checkbox
        self.fft_en = IntVar()
        if self.show_fft:
            self.fft_button = Checkbutton(self, text="Show FFT", variable=self.fft_en)
            self.fft_button.pack()

    def apply(self):
        """
        Return requested inputs.
        """
        res_str = ""
        trace_components = [self.ex_en, self.ey_en, self.ez_en, self.hx_en, self.hy_en, self.hz_en]
        for component in trace_components:
            if component.get() == 1:
                res_str += f" {component.cget('text')}"

        if self.fft_en.get() == 1:
            res_str += " -fft"
        self.result = res_str.strip()
