# Read the BE input file
be_input = pd.read_table(args.be_file)

# Iterate over rows in the BE input file
for i, b in be_input.iterrows():
    print(b[0])  # Print BE type or parameters

    # Define output prefix
    output_prefix = args.output_prefix + '_' + b[0]

    # Construct command based on BE type or parameters
    if args.be_type == 'be':
        cmd = (
            f'python base_editing_guide_designs.py --input-file {args.input_file} --input-type {args.ip_type}'
            f' --be-type {b[0]} --output-name {output_prefix}'
        )
    else:
        pam = b['PAM']
        window = b['Window']
        sglen = b['Sgrna length']
        edit = b['Edit']
        intron_buffer = b['Intron buffer']
        filter_gc = b['Filter GC']
        cmd = (
            f'python base_editing_guide_designs.py --input-file {args.input_file} --input-type {args.ip_type}'
            f' --pam {pam} --edit-window {window} --sg-len {sglen} --edit {edit} --intron-buffer {intron_buffer}'
            f' --filter-gc {filter_gc} --output-name {output_prefix}'
        )

    # Execute command
    os.system(cmd)
