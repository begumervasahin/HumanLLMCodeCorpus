def default_ambig_solver(list):
    if not isinstance(list, list):
        raise ValueError("Input must be a list")
    else:
        if len(list) == 0:
            return None
        else:
            return list[0]

class NamelistMap:
    def __init__(self, inverse_ambiguity_solver=default_ambig_solver):
        self.itemlistmap = {}  # Dictionary to store mappings from name to item list
        self.item_to_nm_map = {}  # Dictionary to store mappings from item to name list
        self.nm_ambig_solver = inverse_ambiguity_solver  # Ambiguity resolver function for item-to-name mappings

    def set(self, name, item_list):
        # Ensure unique items in the list
        unique_item_list = list(set(item_list))
        self.itemlistmap[name] = unique_item_list
        # Update item-to-name mapping
        for item in unique_item_list:
            if item in self.item_to_nm_map:
                if name not in self.item_to_nm_map[item]:
                    self.item_to_nm_map[item].append(name)
            else:
                self.item_to_nm_map[item] = [name]

    def add(self, name, item):
        if self.get(name) is None:
            self.set(name, [item])
        else:
            self.set(name, self.itemlistmap[name] + [item])

    def empty(self):
        return not bool(self.itemlistmap)

    def get(self, name):
        return self.itemlistmap.get(name)

    def get_all_keys(self):
        return self.itemlistmap.keys()

    def get_key_for(self, item):
        if item in self.item_to_nm_map:
            return self.nm_ambig_solver(self.item_to_nm_map[item])
        else:
            return None

    def get_key_list_for(self, item):
        return self.item_to_nm_map.get(item, [])

    def get_key_list_for_matching_items(self, item_matching_fun=lambda *x: False, item_matching_arglist=[]):
        matched_key_list = []
        for name in self.get_all_keys():
            if item_matching_fun(self.itemlistmap[name], *item_matching_arglist):
                matched_key_list.append(name)
        return matched_key_list

    def remove(self, name=None, item=None):
        if name is not None:
            assert item is None
            for item in self.itemlistmap[name]:
                assert name in self.item_to_nm_map[item]
                self.item_to_nm_map[item].remove(name)
            self.itemlistmap.pop(name)
        elif item is not None:
            if item in self.item_to_nm_map.keys():
                aux_nm_list = []
                for nm in self.item_to_nm_map[item]:
                    self.itemlistmap[nm].remove(item)
                    aux_nm_list.append(nm)
                for n in aux_nm_list:
                    if not self.itemlistmap[n]:
                        self.itemlistmap.pop(n)
                self.item_to_nm_map.pop(item)

    def clear(self):
        self.itemlistmap.clear()
        self.item_to_nm_map.clear()

    def set_ambig_solver(self, ambg):
        self.nm_ambig_solver = ambg

    def __str__(self):
        output = '=== Direct Map ===\n'
        for name, items in self.itemlistmap.items():
            output += f'{name} --> {items}\n'
        output += '=== Reverse Map ===\n'
        all_items = set()
        for items in self.itemlistmap.values():
            all_items.update(items)
        for item in all_items:
            output += f'{item} --> {self.item_to_nm_map.get(item, [])}\n'
        output += '====='
        return output

    def __repr__(self):
        return f'<NamelistMap: nm_ambig_solver={self.nm_ambig_solver}, itemlistmap={self.itemlistmap}, reverse_map={self.item_to_nm_map}>'

    def keys_for_item(self, item=None):
        if item is not None:
            return self.item_to_nm_map.get(item, [])
        else:
            return []
