import random
import math
import sys

# Increase recursion limit to handle large numbers in Miller-Rabin test
sys.setrecursionlimit(100000)

def miller_rabin_test(number):
    is_prime = False
    num_minus_1 = number - 1
    k = 0
    
    # Calculate k and m such that number - 1 = 2^k * m
    while num_minus_1 % 2 == 1:
        num_minus_1 = num_minus_1 // 2
        k += 1
    m = num_minus_1 // (2 ** k)
    
    # Randomly choose 'a' for the Miller-Rabin test
    a = random.randint(2, number - 2)
    
    # Compute b = a^m mod number
    b = pow(a, m, number)
    
    if b == 1 or b == number - 1:
        is_prime = True
    else:
        for _ in range(k):
            b = pow(b, 2, number)
            if b == number - 1:
                is_prime = True
                break
                
    return is_prime

def generate_prime(length=300):
    prime_found = False
    
    while not prime_found:
        random_num = random.getrandbits(length)
        random_num |= (1 << length - 1) | 1  # Ensure the number is odd and has the desired length
        prime_found = miller_rabin_test(random_num)
    
    return random_num

def get_two_primes():
    primes_found = False
    
    while not primes_found:
        p = generate_prime()
        q = 2 * p + 1  # Ensure q is also prime for Lucas test
        primes_found = miller_rabin_test(q)
    
    return p, q

def lucas_test(integer):
    p = integer
    factors_of_p_minus_1 = []
    
    # Find factors of p - 1 and store them in factors_of_p_minus_1
    for i in range(2, p):
        if (p - 1) % i == 0:
            factors_of_p_minus_1.append(i)
    
    primitive_roots = []
    
    # Find primitive roots modulo p
    for g in range(2, p):
        is_primitive_root = True
        for q in factors_of_p_minus_1:
            if pow(g, (p - 1) // q, p) == 1:
                is_primitive_root = False
                break
        if is_primitive_root:
            primitive_roots.append(g)
    
    return primitive_roots
