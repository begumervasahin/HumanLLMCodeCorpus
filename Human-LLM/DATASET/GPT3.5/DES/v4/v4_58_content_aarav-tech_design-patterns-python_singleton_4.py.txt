# Define a metaclass for implementing the Singleton pattern
class SingletonMeta(type):
    _instances = {}  # Dictionary to hold the instances of the classes

    # Override the __call__ method to control the instance creation
    def __call__(cls, *args, **kwargs):
        if cls not in cls._instances:
            # Create a new instance if it does not exist
            cls._instances[cls] = super().__call__(*args, **kwargs)
        return cls._instances[cls]

# Define the DBConnector class with the Singleton metaclass
class DBConnector(metaclass=SingletonMeta):
    def __init__(self):
        self.status = "Not Connected"

    def disconnect(self):
        self.status = "Disconnected"

    def connect(self):
        self.status = "Connected"

# Create instances of the DBConnector class
client1 = DBConnector()
print("Client 1:", client1)
print(client1.status)

client2 = DBConnector()
print("Client 2:", client2)

# Modify the status of client2
client2.connect()

# Check the status of client1 after modifying client2
print("Status of Client 1 after modifying Client 2:", client1.status)

# Modify the status of client1
client1.disconnect()

# Check the status of client2 after modifying client1
print("Status of Client 2 after modifying Client 1:", client2.status)
