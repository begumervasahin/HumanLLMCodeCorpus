import sys
import copy
from PyQt5.QtWidgets import (
    QApplication, QDialog, QWidget, QVBoxLayout,
    QCheckBox, QPushButton
)
from PyQt5.QtCore import Qt, pyqtSignal

height = 30

class FileList(QDialog):
    ListSignal = pyqtSignal(list)

    def __init__(self, file_list, pos):
        super().__init__()
        self.file_list = file_list
        self.auxiliary_list = copy.copy(self.file_list)

        self.setup_ui()
        self.init_list()
        self.init_connect()

    def setup_ui(self):
        self.setWindowTitle("File List")
        self.setWindowFlags(self.windowFlags() | Qt.FramelessWindowHint)

        self.scroll_bar = QWidget()
        self.scroll_bar.setMinimumSize(250, len(self.file_list) * height + 30)
        self.scroll_area = QWidget()
        self.scroll_area.setWidget(self.scroll_bar)

        self.select_all = QCheckBox("Select All")
        self.btn_confirm = QPushButton("Confirm")

        layout = QVBoxLayout(self)
        layout.addWidget(self.select_all)
        layout.addWidget(self.scroll_area)
        layout.addWidget(self.btn_confirm)

        self.init_list()

    def init_list(self):
        for number, video_name in enumerate(self.file_list):
            checkbox = QCheckBox(self.scroll_bar)
            checkbox.move(10, (number + 1) * height)
            checkbox.setText(self.auxiliary_list[number])
            self.file_list[number] = checkbox

    def init_connect(self):
        self.select_all.toggled.connect(self.handle_select_all)
        self.btn_confirm.clicked.connect(self.handle_confirm)

    def handle_select_all(self):
        checked = self.sender().isChecked()
        for btn in self.file_list:
            btn.setChecked(checked)

    def handle_confirm(self):
        videos_chosen = [btn.text() for btn in self.file_list if btn.isChecked()]
        self.ListSignal.emit(videos_chosen)
        self.close()

if __name__ == '__main__':
    app = QApplication(sys.argv)
    file_list = ['a', 'b', 'c', 'ddd', 'a', 'b', 'c', 'ddd', 'a', 'b', 'c', 'ddd']
    pos = 123
    MyList = FileList(file_list, pos)
    MyList.show()
    sys.exit(app.exec_())
