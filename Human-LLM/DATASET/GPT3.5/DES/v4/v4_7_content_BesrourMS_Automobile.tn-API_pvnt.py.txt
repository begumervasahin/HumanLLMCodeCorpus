import requests
from bs4 import BeautifulSoup
import json

# Set headers for the request
headers = {'Access-Control-Allow-Origin': '*'}

# Define the URL to scrape
url = 'https://www.automobile.tn/fr/neuf/recherche'

# Send a GET request to the URL and parse the HTML content
response = requests.get(url, headers=headers)
page = BeautifulSoup(response.text, 'html.parser')

# Extract total number of cars
total = page.select('.total')[0].get_text(strip=True)
total_count = int(total.split()[0])

# Calculate the number of cars per page
limit = 0
for _ in page.select(''):
    limit += 1

# Calculate the number of pages based on the limit
pager = (total_count // limit) + 1 if total_count % limit != 0 else total_count // limit

# Initialize an empty list to store car information
arrayclassement = []

# Loop through each page
for j in range(1, pager + 1):
    # Construct the URL for the current page
    page_url = f'https://www.automobile.tn/fr/neuf/recherche/s=sort%21price?sort=price&page={j}'

    # Send a GET request to the current page and parse the HTML content
    response = requests.get(page_url, headers=headers)
    html = BeautifulSoup(response.text, 'html.parser')

    # Loop through each car on the page
    for i in range(1, limit + 1):
        # Extract car information
        car_id = html.select(f'')[0]['data-key'].strip() if html.select(f'') else ''
        car_name = html.select(f'')[0].get_text(strip=True) if html.select(f'') else ''
        car_price = html.select(f'')[0].get_text(strip=True) if html.select(f'') else ''
        car_link = 'https://www.automobile.tn' + html.select(f'')[0]['href'].strip() if html.select(f'') else ''
        car_image = html.select(f'')[0]['src'].strip() if html.select(f'') else ''

        # Append car information to the list
        arrayclassement.append({
            'ID': car_id,
            'Voiture': car_name,
            'Image': car_image,
            'Prix': car_price,
            'Lien': car_link
        })

# Truncate the list to match the total count
arrayclassement = arrayclassement[:total_count]

# Print the list of car information in JSON format
print(json.dumps(arrayclassement, ensure_ascii=False))
