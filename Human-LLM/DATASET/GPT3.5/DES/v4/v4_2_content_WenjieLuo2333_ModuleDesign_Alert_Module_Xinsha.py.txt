class Alert:
    def __init__(self):
        # Initialize flags and fine turn count
        self.fine_turn = 0
        self.bo_flag = 0
        self.bp_flag = 0
        self.pul_flag = 0

    def get_bo_data(self, data):
        # Update blood oxygen flag based on data
        if self.fine_turn > 20:
            self.bo_flag = 0
        if 0 <= data <= 0.3:
            self.bo_flag += 1
        else:
            self.fine_turn += 1

    def get_bp_data(self, data):
        # Update blood pressure flag based on data
        if self.fine_turn > 20:
            self.bp_flag = 0
        if 60 <= data <= 90:
            self.bp_flag += 1
        else:
            self.fine_turn += 1

    def get_pul_data(self, data):
        # Update pulse flag based on data
        if self.fine_turn > 20:
            self.pul_flag = 0
        if 90 <= data <= 120:
            self.pul_flag += 1
        else:
            self.fine_turn += 1

    def alert_output(self):
        """
        Compare data with certain thresholds
        and send flags to the user interface module.
        """
        # Determine which flag is raised and return corresponding alert code
        if self.bo_flag != 0:
            return 1  # Alert for blood oxygen
        elif self.bp_flag != 0:
            return 2  # Alert for blood pressure
        elif self.pul_flag != 0:
            return 3  # Alert for pulse
        else:
            return 0  # No alert
