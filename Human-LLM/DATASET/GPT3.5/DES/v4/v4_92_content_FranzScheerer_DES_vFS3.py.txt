import sys

# Rabin constants
CRABIN = 'rP9hpgejqp8v9YLvJ6b1QArhFOTm'
A_FACTOR = 7
B_FACTOR = 11

# Rabin hash functions
a_h = i_h = j_h = w_h = 0
s_h = []

def update_h():
    global a_h, i_h, j_h, w_h, s_h
    i_h += w_h
    if i_h > 255:
        i_h -= 256
    j_h += s_h[i_h]
    if j_h > 255:
        j_h = s_h[j_h - 256]
    else:
        j_h = s_h[j_h]
    t = s_h[i_h] + s_h[j_h]
    s_h[j_h] = t - s_h[j_h]
    s_h[i_h] = t - s_h[i_h]

def output_h():
    global a_h, i_h, j_h, w_h, s_h
    update_h()
    return s_h[j_h]

def shuffle_h():
    global a_h, i_h, j_h, w_h, s_h
    for v in range(256):
        update_h()
    w_h = (w_h + 2) % 256
    a_h = 0

def absorb_nibble_h(x):
    global a_h, i_h, j_h, w_h, s_h
    if a_h == 241:
        shuffle_h()
    s_h[a_h], s_h[240 + x] = s_h[240 + x], s_h[a_h]
    a_h += 1

def absorb_byte_h(b):
    absorb_nibble_h(b % 16)
    absorb_nibble_h(b >> 4)

def squeeze_h(out, outlen):
    global a_h, i_h, j_h, w_h, s_h
    shuffle_h()
    for v in range(outlen):
        out.append(output_h())

def h(x):
    global a_h, i_h, j_h, w_h, s_h
    j_h = i_h = a_h = 0
    w_h = 1
    s_h = [ix for ix in range(256)]
    for c in x:
        absorb_byte_h(ord(c))
    res = []
    squeeze_h(res, 128)
    out = 0
    for bx in res:
        out = (out << 8) + bx
    return out % nrabin

def hb(x):
    global a_h, i_h, j_h, w_h, s_h
    j_h = i_h = a_h = 0
    w_h = 1
    s_h = [ix for ix in range(256)]
    for c in x:
        absorb_byte_h(c)
    res = []
    squeeze_h(res, 128)
    out = 0
    for bx in res:
        out = (out << 8) + bx
    return out % nrabin

# Conversion functions
def num2code(x):
    res = ''
    while x > 0:
        y = x % 64
        if y < 10:
            res = chr(y + 48) + res
        elif y < 36:
            res = chr(y + 55) + res
        elif y < 62:
            res = chr(y + 61) + res
        elif y == 62:
            res = '@' + res
        elif y == 63:
            res = '/' + res
        x >>= 6
    return res

def code2num(x):
    res = 0
    for c in x:
        if 48 <= ord(c) < 58:
            res = (res << 6) + ord(c) - 48
        elif 65 <= ord(c) < 91:
            res = (res << 6) + ord(c) - 55
        elif 97 <= ord(c) < 123:
            res = (res << 6) + ord(c) - 61
        elif c == '@':
            res = (res << 6) + 62
        elif c == '/':
            res = (res << 6) + 63
    return res

# Pre-computed Rabin constant
nrabin = code2num(CRABIN)

# File I/O functions
def write_number(n, fnam):
    with open(fnam, 'wb') as f:
        while n > 0:
            b = n & 0xFF
            n >>= 8
            f.write(bytes([b]))

def read_number(fnam):
    with open(fnam, 'rb') as f:
        n = 0
        for c in reversed(f.read()):
            n = (n << 8) ^ c
    return n

def hF(fnam):
    with open(fnam, 'rb') as f:
        return hb(f.read())

def vF(s, fnam):
    a = A_FACTOR
    b = B_FACTOR
    h0 = hF(fnam)
    ha = (a * h0) % nrabin
    hb = (b * h0) % nrabin
    hab = (a * b * h0) % nrabin
    sq = (s * s) % nrabin
    return (h0 == sq) or (ha == sq) or (hb == sq) or (hab == sq)
