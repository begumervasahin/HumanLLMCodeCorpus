import numpy as np
from sklearn import metrics
from sklearn.utils.linear_assignment_ import linear_assignment

def cluster_accuracy(true_labels, predicted_labels):
    """
    Calculate the unsupervised clustering accuracy.

    Parameters:
        true_labels: numpy array containing the true labels with shape (n_samples,)
        predicted_labels: numpy array containing the predicted labels with shape (n_samples,)

    Returns:
        Accuracy in the range [0, 1]
    """
    true_labels = true_labels.astype(np.int64)
    assert predicted_labels.size == true_labels.size

    num_clusters = max(predicted_labels.max(), true_labels.max()) + 1
    confusion_matrix = np.zeros((num_clusters, num_clusters), dtype=np.int64)

    for i in range(predicted_labels.size):
        confusion_matrix[predicted_labels[i], true_labels[i]] += 1

    row_ind, col_ind = linear_assignment(confusion_matrix.max() - confusion_matrix)
    accuracy = sum([confusion_matrix[i, j] for i, j in zip(row_ind, col_ind)]) / predicted_labels.size

    return accuracy

def cluster_purity(true_labels, predicted_labels):
    """
    Calculate the clustering purity.

    Parameters:
        true_labels: numpy array containing the true labels with shape (n_samples,)
        predicted_labels: numpy array containing the predicted labels with shape (n_samples,)

    Returns:
        Purity in the range [0, 1]
    """
    true_labels = true_labels.astype(np.int64)
    assert predicted_labels.size == true_labels.size

    num_clusters = max(predicted_labels.max(), true_labels.max()) + 1
    confusion_matrix = np.zeros((num_clusters, num_clusters), dtype=np.int64)

    for i in range(predicted_labels.size):
        confusion_matrix[predicted_labels[i], true_labels[i]] += 1

    label_mapping = confusion_matrix.argmax(axis=1)
    predicted_labels_voted = label_mapping[predicted_labels]

    purity = metrics.accuracy_score(predicted_labels_voted, true_labels)
    return purity

def quantization_error(distances):
    """
    Calculate the k-means quantization error.

    Parameters:
        distances: numpy array of distances with shape (n_samples, n_clusters)

    Returns:
        Quantization error
    """
    return distances.min(axis=1).mean()

def topographic_error(distances, map_size):
    """
    Calculate the SOM topographic error.

    Parameters:
        distances: numpy array of distances with shape (n_samples, n_units)
        map_size: tuple containing the dimensions of the map, e.g., (10, 10)

    Returns:
        Topographic error
    """
    height, width = map_size

    def is_adjacent(unit1, unit2):
        """
        Check if two units are adjacent on the SOM map.
        """
        return (abs(unit1 // width - unit2 // width) == 1 and abs(unit1 % width - unit2 % width) == 0) or \
               (abs(unit1 // width - unit2 // width) == 0 and abs(unit1 % width - unit2 % width) == 1)

    best_matching_units = np.argsort(distances, axis=1)[:, :2]
    topographic_error = 1.0 - np.mean([is_adjacent(unit1, unit2) for unit1, unit2 in best_matching_units])

    return topographic_error
