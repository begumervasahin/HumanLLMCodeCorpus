import subprocess
import datetime
import requests
import time
import RPi.GPIO as GPIO

# Define global variables
ACCESSKEY = "{YOUR ACCESS KEY}"
API_URL = f"https://www.divera247.com/api/last-alarm?accesskey={ACCESSKEY}"
SENSOR_PIN = 23
SCREEN_ACTIVE = True
SCREENSAVER_ACTIVE = False
DUTY_TIME = False
ALARM_ACTIVE = False

# Function to control monitor
def control_monitor(command):
    script_path = ". .divera_commands.sh"
    if command == "on":
        subprocess.Popen(['bash', '-c', f'{script_path}; monitor on'])
    elif command == "off":
        subprocess.Popen(['bash', '-c', f'{script_path}; monitor off'])

# Function to control screen
def control_screen(command):
    global SCREEN_ACTIVE
    if command == "on":
        print("Turning the screen on.")
        if ALARM_ACTIVE:
            control_screen_saver("off")
        else:
            initialize_screen_saver()
        SCREEN_ACTIVE = True
        subprocess.Popen(['bash', '-c', '. .divera_commands.sh; screen on']).wait()
    elif command == "off":
        print("Turning the screen off.")
        control_screen_saver("off")
        SCREEN_ACTIVE = False
        subprocess.Popen(['bash', '-c', '. .divera_commands.sh; screen off']).wait()

# Function to control screensaver
def control_screen_saver(command):
    global SCREENSAVER_ACTIVE
    if command == "on":
        print("Activating the screensaver.")
        SCREENSAVER_ACTIVE = True
        subprocess.Popen(['xscreensaver-command', '-activate']).wait()
    elif command == "off":
        print("Deactivating the screensaver.")
        SCREENSAVER_ACTIVE = False
        subprocess.Popen(['xscreensaver-command', '-deactivate']).wait()

# Function to handle motion detection
def handle_motion(channel):
    if GPIO.input(channel):
        print('Motion detected!')
        if SCREEN_ACTIVE and SCREENSAVER_ACTIVE:
            print("Turning off the screensaver.")
            control_screen_saver("off")
        elif not SCREEN_ACTIVE:
            print("Turning the screen on.")
            control_screen("on")
    else:
        print('No motion detected.')
        if DUTY_TIME and not ALARM_ACTIVE:
            control_screen_saver("on")
        elif not DUTY_TIME and not ALARM_ACTIVE:
            control_screen("off")

# Function to initialize screensaver
def initialize_screen_saver():
    control_screen_saver("off") if GPIO.input(SENSOR_PIN) else control_screen_saver("on")

# Initial setup
control_monitor("on")
GPIO.setmode(GPIO.BCM)
GPIO.setup(SENSOR_PIN, GPIO.IN)
GPIO.add_event_detect(SENSOR_PIN, GPIO.BOTH, callback=handle_motion)
initialize_screen_saver()

# Main loop
while True:
    try:
        result = requests.get(API_URL).content.decode()
        ALARM_ACTIVE = "\"success\":true" in result and "Probealarm" not in result
    except:
        ALARM_ACTIVE = True

    now = datetime.datetime.now()
    day_of_week = now.weekday() + 1
    hour = now.hour
    minutes = now.minute

    if day_of_week == 3 and hour >= 17:
        DUTY_TIME = True
    elif day_of_week == 4 and hour < 1:
        DUTY_TIME = True
    elif day_of_week == 6 and 7 <= hour <= 19:
        DUTY_TIME = True
    else:
        DUTY_TIME = False

    if not SCREEN_ACTIVE:
        if ALARM_ACTIVE or DUTY_TIME:
            print("Turning the display on.")
            control_screen("on")
        elif not ALARM_ACTIVE and hour == 3 and minutes == 5:
            print("Updating and restarting the Raspberry Pi.")
            time.sleep(45)
            subprocess.Popen(['sudo', 'apt', 'update']).wait()
            subprocess.Popen(['sudo', 'apt', '--yes', '--force-yes', 'upgrade']).wait()
            subprocess.Popen(['sudo', 'reboot'])
    else:
        if not ALARM_ACTIVE and not DUTY_TIME and SCREENSAVER_ACTIVE:
            print("Turning the display off.")
            control_screen("off")

    time.sleep(30)
