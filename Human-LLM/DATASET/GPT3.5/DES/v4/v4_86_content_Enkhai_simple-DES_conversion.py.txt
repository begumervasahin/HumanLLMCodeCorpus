import misc
import permutations

def primary_key_to_subkeys(key):
    # Initialize an empty list to store the generated subkeys
    subkeys = []

    # Convert hexadecimal key to binary
    binary_prim_key = misc.hex_to_bits(key)

    # Group binary key into 7-bit blocks
    binary_prim_key = ''.join([binary_prim_key[i:i + 7] for i in range(0, len(binary_prim_key), 8)])

    # Split binary key into left and right parts
    left_part = binary_prim_key[:len(binary_prim_key) // 2]
    right_part = binary_prim_key[len(binary_prim_key) // 2:]

    # Generate subkeys
    for i in range(16):
        # Determine the shift amount based on the round
        if i in {0, 1, 8, 15}:
            shift_amount = 1
        else:
            shift_amount = 2

        # Apply left shift to left and right parts accordingly
        left_part = misc.left_shift(left_part, shift_amount)
        right_part = misc.left_shift(right_part, shift_amount)

        # Concatenate left and right parts and apply permutation
        subkey = permutations.permutation(left_part + right_part, "compression")
        
        # Append the generated subkey to the list
        subkeys.append(subkey)

    return subkeys
