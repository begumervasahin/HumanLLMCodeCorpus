import random

class AiModule:
    def __init__(self):
        # Initialize empty lists to store input data
        self.blood_oxygen = []
        self.blood_pressure = []
        self.pulse = []

    def input_check(self, blood_oxygen, blood_pressure, pulse):
        """
        Check the input type from the database for correctness.
        """
        try:
            for bo, bp, p in zip(blood_oxygen, blood_pressure, pulse):
                if not all(isinstance(val, float) for val in (bo, bp, p)):
                    raise AttributeError
                else:
                    self.blood_oxygen.append(bo)
                    self.blood_pressure.append(bp)
                    self.pulse.append(p)
        except AttributeError:
            print('Input data type is incorrect')

    def predict(self):
        """
        Predict blood oxygen, blood pressure, and pulse for each type from the database.
        """
        if not self.blood_oxygen or not self.blood_pressure or not self.pulse:
            print('No data available for prediction')
            return

        rand = random.randint(1, len(self.blood_oxygen))
        pred_blood_oxygen = 0
        pred_blood_pressure = 0
        pred_pulse = 0

        for i in range(rand):
            pred_blood_oxygen += self.blood_oxygen[i] / rand
            pred_blood_pressure += self.blood_pressure[i] / rand
            pred_pulse += self.pulse[i] / rand

            print(f'Predicted blood oxygen: {pred_blood_oxygen}')
            print(f'Predicted blood pressure: {pred_blood_pressure}')
            print(f'Predicted pulse: {pred_pulse}')

        return pred_blood_oxygen, pred_blood_pressure, pred_pulse

# Example usage
if __name__ == "__main__":
    ai_module = AiModule()
    blood_oxygen_data = [98.5, 99.2, 97.8, 100.0]
    blood_pressure_data = [120.5, 122.1, 118.9, 121.6]
    pulse_data = [75.2, 76.5, 74.8, 77.3]

    ai_module.input_check(blood_oxygen_data, blood_pressure_data, pulse_data)
    ai_module.predict()
