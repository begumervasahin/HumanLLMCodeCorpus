import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
import math
import numpy as np
import random
import os
import argparse
from data_utils import Preprocessing

# Define command-line arguments
parser = argparse.ArgumentParser(description='Train TransE model for knowledge graph embeddings')
parser.add_argument('--datadir', required=True, help="Path to the directory containing the data files")
args = parser.parse_args()

# Load data
all_data_file = os.path.join(args.datadir, 'data10K.csv')
preprocessor = Preprocessing()
pos_triples, entity2id, rel2id = preprocessor.get_entity_relations(all_data_file)

# Model parameters
embedding_size = 100
num_entity = len(entity2id)
num_rel = len(rel2id)
batch_size = 100
learning_rate = 0.001
epochs = 1000
USE_CUDA = torch.cuda.is_available()

# CUDA configuration
if USE_CUDA:
    FloatTensor = torch.cuda.FloatTensor
    LongTensor = torch.cuda.LongTensor
else:
    FloatTensor = torch.FloatTensor
    LongTensor = torch.LongTensor

# Define the TransE model
class TransE(nn.Module):
    def __init__(self):
        super(TransE, self).__init__()
        # Initialize entity and relation embeddings
        ent_weight = FloatTensor(num_entity, embedding_size)
        torch.nn.init.uniform_(ent_weight, a=-6/math.sqrt(embedding_size), b=6/math.sqrt(embedding_size))
        rel_weight = FloatTensor(num_rel, embedding_size)
        torch.nn.init.uniform_(rel_weight, a=-6/math.sqrt(embedding_size), b=6/math.sqrt(embedding_size))
        # Normalize embeddings
        normalized_ent_weight = F.normalize(ent_weight, p=2, dim=1)
        normalized_rel_weight = F.normalize(rel_weight, p=2, dim=1)
        # Define embedding layers
        self.ent_embeddings = nn.Embedding(num_entity, embedding_size)
        self.rel_embeddings = nn.Embedding(num_rel, embedding_size)
        # Set weights for embedding layers
        self.ent_embeddings.weight = nn.Parameter(normalized_ent_weight)
        self.rel_embeddings.weight = nn.Parameter(normalized_rel_weight)

    def forward(self, batch_pos_h, batch_pos_t, batch_r, batch_neg_h, batch_neg_t):     
        # Lookup embeddings for positive and negative triples
        batch_pos_h_emb = self.ent_embeddings(batch_pos_h)
        batch_pos_t_emb = self.ent_embeddings(batch_pos_t)
        batch_r_emb = self.rel_embeddings(batch_r)
        batch_neg_h_emb = self.ent_embeddings(batch_neg_h)
        batch_neg_t_emb = self.ent_embeddings(batch_neg_t)
        # Margin for margin-based ranking loss
        margin = Variable(FloatTensor([1.0 for _ in range(batch_size)]), requires_grad=True)
        if USE_CUDA: margin = margin.cuda()
        # Zeros tensor for comparison
        zeros = Variable(FloatTensor([0.0 for _ in range(batch_size)]), requires_grad=True)
        if USE_CUDA: zeros = zeros.cuda()
        # Compute distances for positive and negative triples
        pos_distance = torch.norm(batch_pos_h_emb + batch_r_emb - batch_pos_t_emb, p=2, dim=1)
        neg_distance = torch.norm(batch_neg_h_emb + batch_r_emb - batch_neg_t_emb, p=2, dim=1)
        # Margin-based ranking loss
        loss = torch.sum(torch.max(zeros, pos_distance + margin - neg_distance))        
        return loss

# Prepare batch data
def prepare_batch(batch_triples):
    batch_pos_h, batch_pos_t, batch_r = [], [], []
    batch_neg_h, batch_neg_t = [], []

    for triple in batch_triples:
        h, r, t = triple
        # Append entity and relation IDs for positive triple
        batch_pos_h.append(entity2id[h])
        batch_pos_t.append(entity2id[t])
        batch_r.append(rel2id[r])

        # Randomly sample negative triple
        if np.random.random() < 0.5:
            while True:
                neg_t = random.choice(list(entity2id.values()))
                if (h, r, neg_t) not in pos_triples:
                    break
            batch_neg_h.append(entity2id[h])
            batch_neg_t.append(neg_t)
        else:
            while True:
                neg_h = random.choice(list(entity2id.values()))
                if (neg_h, r, t) not in pos_triples:
                    break
            batch_neg_h.append(neg_h)
            batch_neg_t.append(entity2id[t])

    # Pad batches if necessary
    pad_size = batch_size - len(batch_pos_h)
    batch_pos_h += [0] * pad_size
    batch_pos_t += [0] * pad_size
    batch_r += [0] * pad_size
    batch_neg_h += [0] * pad_size
    batch_neg_t += [0] * pad_size

    # Convert lists to tensors
    return LongTensor(batch_pos_h), LongTensor(batch_pos_t), LongTensor(batch_r), \
           LongTensor(batch_neg_h), LongTensor(batch_neg_t)

# Training function
def train(model, optimizer):
    training_loss = 0.0
    for epoch in range(epochs):
        random.shuffle(pos_triples)
        for i in range(0, len(pos_triples), batch_size):
            batch_triples = pos_triples[i:i+batch_size]
            pos_h, pos_t, pos_r, neg_h, neg_t = prepare_batch(batch_triples)
            optimizer.zero_grad()
            loss = model(pos_h, pos_t, pos_r, neg_h, neg_t)
            loss.backward()
            optimizer.step()
            training_loss += loss.item()
        print(f"Epoch {epoch+1}, Loss: {training_loss:.4f}")
        training_loss = 0.0

# Initialize TransE model
model = TransE()
if USE_CUDA:
    model = model.cuda()
optimizer = optim.Adam(model.parameters(), lr=learning_rate)

# Train the model
train(model, optimizer)

# Save the trained model
torch.save(model.state_dict(), 'TransE.pth')
