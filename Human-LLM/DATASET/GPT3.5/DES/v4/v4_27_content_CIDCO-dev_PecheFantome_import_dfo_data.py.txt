import xlrd
import mysql.connector
from datetime import datetime
import re
import sys
import os

def display_help_message():
    os.system('cls' if os.name == 'nt' else 'clear')
    print("Help")
    print("")
    print("Syntax: python import_dfo_data.py [options]")
    print("")
    print("Options:")
    print("")
    print("help or h          Print Help Page")
    print("[Host]             Hostname or IP of the Database Server")
    print("[User]             User name to access to the Database")
    print("[Password]         Password to access to the Database")
    print("[DB_name]          Name of the Database")
    print("[filename]         File name")
    print("")
    print("Command line example:")
    print("python3 import_dfo_data.py -help")
    print("python3 import_dfo_data.py server_hostname user_name user_pass database_name file_name")
    print("python3 import_dfo_data.py 192.168.1.100 user_test pass1234 data_test file.xlsx")
    print("python3 import_dfo_data.py test.com user_test pass1234 data_test file.xlsx") 
    print("")
    print("")
    print("")

def main():
    if len(sys.argv) != 6:
        display_help_message()
    else:
        db_host = sys.argv[1]
        db_user = sys.argv[2]
        db_password = sys.argv[3]
        db_database_name = sys.argv[4]
        input_filename = sys.argv[5]

        db = mysql.connector.connect(
            host=db_host,
            user=db_user,
            password=db_password,
            database=db_database_name
        )

        wb = xlrd.open_workbook(input_filename)
        sheet = wb.sheet_by_index(1)
        cursor = db.cursor()

        create_table_query = """
        CREATE TABLE IF NOT EXISTS `dfo_recuperes` (
          `id` BIGINT(20) NOT NULL AUTO_INCREMENT PRIMARY KEY,
          `retrieved` DATETIME NOT NULL,
          `type` VARCHAR(255) NOT NULL,
          `quantity` BIGINT(20) NOT NULL,
          `net_length` BIGINT(20) NOT NULL,
          `rope_length` BIGINT(20) NOT NULL,
          `LATITUDE` DOUBLE NOT NULL,
          `LONGITUDE` DOUBLE NOT NULL,
          `position` GEOMETRY NOT NULL
        ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT ='';
        """
        cursor.execute(create_table_query)

        insert_query = """
        INSERT INTO dfo_recuperes (retrieved, type, quantity, net_length, rope_length, LATITUDE, LONGITUDE, position)
        VALUES (%s, %s, %s, %s, %s, %s, %s, POINT(%s, %s))
        """

        DMS_pattern = r"[0-9]{2}°[0-9]{2}'"
        DDM_pattern = r"[0-9]{2}[\s°][0-9]{2}[\.,][0-9]{1,4}"

        for r in range(1, sheet.nrows):
            try:
                retrieved = None
                try:
                    retrieved_date = sheet.cell(r, 0).value
                    if re.match(r"\d{4}-(0[1-9]|1[0-2])-(0[1-9]|[12][0-9]|3[01])$", retrieved_date):
                        retrieved = datetime.strptime(retrieved_date, "%Y-%m-%d")
                except:
                    pass

                type_val = sheet.cell(r, 3).value
                quantity = sheet.cell(r, 4).value
                net_length = sheet.cell(r, 5).value
                rope_length = sheet.cell(r, 6).value

                LATITUDE, LONGITUDE = None, None
                lat_value, lon_value = sheet.cell(r, 8).value, sheet.cell(r, 9).value

                if re.match(DMS_pattern, lat_value):
                    lat_deg, lat_min = map(int, re.findall(r"\d+", lat_value))
                    LATITUDE = lat_deg + lat_min / 60
                    lon_deg, lon_min = map(int, re.findall(r"\d+", lon_value))
                    LONGITUDE = -(lon_deg + lon_min / 60)
                elif re.match(DDM_pattern, lat_value):
                    lat_deg, lat_min, lat_sec = map(float, re.findall(r"\d+\.\d+|\d+", lat_value))
                    LATITUDE = lat_deg + lat_min / 60 + lat_sec / 3600
                    lon_deg, lon_min, lon_sec = map(float, re.findall(r"\d+\.\d+|\d+", lon_value))
                    LONGITUDE = -(lon_deg + lon_min / 60 + lon_sec / 3600)

                values = (retrieved, type_val, quantity, net_length, rope_length, LATITUDE, LONGITUDE, LONGITUDE, LATITUDE)
                cursor.execute(insert_query, values)
            except Exception as e:
                print(f"Error processing row {r}: {e}")
                continue

        cursor.close()
        db.commit()
        db.close()
        print("Completed")

if __name__ == "__main__":
    main()
