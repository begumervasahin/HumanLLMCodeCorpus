# Facade design pattern to simplify cooking process

# Facade class for easy dish preparation
class Cook:
    # Provides an easy interface to prepare a dish
    def prepare_dish(self):
        cutter = Cutter()
        cutter.cut_vegetables()

        boiler = Boiler()
        boiler.boil_vegetables()

        frier = Frier()
        frier.fry()

# Class for cutting vegetables
class Cutter:
    # Provides the feature of cutting vegetables
    def cut_vegetables(self):
        print("All vegetables are cut")

# Class for boiling vegetables
class Boiler:
    # Provides the feature of boiling vegetables
    def boil_vegetables(self):
        print("All vegetables are boiled")

# Class for frying vegetables
class Frier:
    # Provides the feature of frying vegetables
    def fry(self):
        print("All vegetables are mixed and fried.")

# Main function to create Cook object and prepare a dish
if __name__ == "__main__":
    cook = Cook()
    cook.prepare_dish()
