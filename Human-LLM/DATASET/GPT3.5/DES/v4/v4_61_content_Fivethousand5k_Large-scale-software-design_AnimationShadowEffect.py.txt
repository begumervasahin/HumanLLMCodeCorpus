from PyQt5.QtCore import QPropertyAnimation, pyqtProperty
from PyQt5.QtWidgets import QGraphicsDropShadowEffect

class AnimationShadowEffect(QGraphicsDropShadowEffect):
    """
    A QGraphicsDropShadowEffect subclass that adds animation to the shadow radius.
    """
    def __init__(self, color, *args, **kwargs):
        """
        Initialize the AnimationShadowEffect.

        Args:
            color: The color of the shadow.
            *args, **kwargs: Additional arguments for QGraphicsDropShadowEffect.
        """
        super().__init__(*args, **kwargs)
        self.setColor(color)
        self.setOffset(0, 0)
        self.setBlurRadius(0)
        self._radius = 0

        # Configure animation
        self.animation = QPropertyAnimation(self)
        self.animation.setTargetObject(self)
        self.animation.setDuration(2000)
        self.animation.setLoopCount(-1)
        self.animation.setPropertyName(b'radius')
        self.animation.setKeyValueAt(0, 1)
        self.animation.setKeyValueAt(0.5, 30)
        self.animation.setKeyValueAt(1, 1)

    def start_animation(self):
        """Start the shadow radius animation."""
        self.animation.start()

    def stop_animation(self, radius=0):
        """
        Stop the shadow radius animation and set the radius to a specified value.

        Args:
            radius: The radius value to set after stopping the animation.
        """
        self.animation.stop()
        self.setRadius(radius)

    @pyqtProperty(int)
    def radius(self):
        """Get the current radius of the shadow."""
        return self._radius

    @radius.setter
    def radius(self, r):
        """
        Set the radius of the shadow.

        Args:
            r: The radius value to set.
        """
        self._radius = r
        self.setBlurRadius(r)
