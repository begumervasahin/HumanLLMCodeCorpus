import Destiny2_API
import Destiny2_Manifest

# Initialize the Destiny 2 manifest
manifest = Destiny2_Manifest.init_manifest()

# Search for a Destiny 2 player by display name
player_display_name = 'Kage%2312572'
membership_type = 4
destiny_player = Destiny2_API.call_API('SearchDestinyPlayer', membershipType=membership_type, displayName=player_display_name)[0]

# Retrieve the player's profile
profile_membership_type = destiny_player['membershipType']
profile_membership_id = destiny_player['membershipId']
profile_components = 'Profiles'
profile = Destiny2_API.call_API('GetProfile', membershipType=profile_membership_type, destinyMembershipId=profile_membership_id, components=profile_components)

# Retrieve information about the player's characters
characters = []
for character_id in profile['profile']['data']['characterIds']:
    character_info = Destiny2_API.call_API('GetCharacter',
                                           membershipType=profile['profile']['data']['userInfo']['membershipType'],
                                           destinyMembershipId=profile['profile']['data']['userInfo']['membershipId'],
                                           characterId=character_id,
                                           components='Characters,CharacterInventories,CharacterEquipment')
    characters.append(character_info)

# Display character information
for character in characters:
    class_hash = character['character']['data']['classHash']
    class_manifest = manifest['DestinyClassDefinition'][class_hash]
    class_name = class_manifest["displayProperties"]["name"]
    print(f'\n{class_name}:')
    
    for item in character['equipment']['data']['items']:
        bucket_hash = item['bucketHash']
        bucket_manifest = manifest['DestinyInventoryBucketDefinition'][bucket_hash]
        
        item_hash = item['itemHash']
        item_manifest = manifest['DestinyInventoryItemDefinition'][item_hash]
        
        bucket_name = bucket_manifest["displayProperties"]["name"]
        item_name = item_manifest["displayProperties"]["name"]
        
        print(f' {bucket_name}: {item_name}')
