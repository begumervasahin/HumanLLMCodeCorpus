import numpy as np
import math
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.linear_model import Ridge
from sklearn.linear_model import RidgeCV
from sklearn.metrics import mean_squared_error, mean_absolute_error

# Load data
data = pd.read_table(r'D:\360MoveData\Users\HP\Desktop\prostate.data.txt')

# Define factors and target variable
factors = ['lcavol', 'lweight', 'age', 'lbph', 'svi', 'lcp', 'gleason', 'pgg45']        
X = data[factors]
Y = data['lpsa']

# Function to calculate the standard error of a sample
def calculate_standard_error(sample):
    std = np.std(sample, ddof=0)
    return std / math.sqrt(len(sample))

# Split the data into training and testing sets
X_train, X_test, y_train, y_test = train_test_split(X, Y, test_size=0.2, random_state=1)

# Initialize an array of alpha values for RidgeCV
Lambdas = np.logspace(-5, 2, 200)

# Perform Ridge regression with cross-validation to find the optimal alpha
ridge_cv = RidgeCV(alphas=Lambdas, normalize=True, scoring='neg_mean_squared_error', cv=10)
ridge_cv.fit(X_train, y_train)

# Print the optimal alpha value
print('Alpha is:', round(ridge_cv.alpha_, 4))

# Train the Ridge regression model using the optimal alpha
ridge = Ridge(alpha=ridge_cv.alpha_)
ridge.fit(X_train, y_train)

# Make predictions on the entire dataset and the test set
y_predict = ridge.predict(X)
y_test_predict = ridge.predict(X_test)

# Calculate evaluation metrics
mse_predict = round(mean_squared_error(y_test, y_test_predict), 4)
mae_predict = round(mean_absolute_error(y_test, y_test_predict), 4)
std_error = round(calculate_standard_error(y_test_predict), 4)

# Format coefficients
coefficients = [(factor, round(coef, 4)) for factor, coef in zip(factors, ridge.coef_)]

# Display results
print('Intercept is:', round(ridge.intercept_, 4))
print('Estimated coefficients are:', coefficients)
print('Std Error is:', std_error)
print('MSE is:', mse_predict)
print('MAE is:', mae_predict)
