def base64_to_num(base64_str):
    """Converts a base64 string to a number."""
    result = 0
    for char in base64_str:
        if '0' <= char <= '9':
            result = (result << 6) + ord(char) - 48
        elif 'A' <= char <= 'Z':
            result = (result << 6) + ord(char) - 55
        elif 'a' <= char <= 'z':
            result = (result << 6) + ord(char) - 61
        elif char == '+':
            result = (result << 6) + 62
        elif char == '/':
            result = (result << 6) + 63
    return result

def num_to_base64(number):
    """Converts a number to a base64 string."""
    base64_chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
    result = ""
    while number > 0:
        result = base64_chars[number % 64] + result
        number //= 64
    return result

def bytes_to_num(bytes_str):
    """Converts a bytes string to a number."""
    result = 0
    for byte in bytes_str:
        result = (result << 8) + ord(byte)
    return result

def num_to_bytes(number):
    """Converts a number to a bytes string."""
    result = ""
    while number > 0:
        result = chr(number % 256) + result
        number //= 256
    return result

def sha512_hash(data):
    """Returns a SHA-512 hash of the given data."""
    return hashlib.sha512(data.encode('utf-8')).hexdigest()

def generate_random_number():
    """Generates a random number using SHA-512."""
    seed = "RANDOM-SEED" + str(random.random())
    return bytes_to_num(hashlib.sha512(seed.encode('utf-8')).digest())

def sign_file(filename):
    """Generates a digital signature for the given file."""
    with open(filename, 'r') as file:
        file_content = file.read()
    hash_num = bytes_to_num(sha512_hash(file_content).encode('utf-8'))
    signature = pow(hash_num, d, N_RSA)
    return num_to_base64(signature)

def verify_signature(filename, signature):
    """Verifies the digital signature of the given file."""
    with open(filename, 'r') as file:
        file_content = file.read()
    hash_num = bytes_to_num(sha512_hash(file_content).encode('utf-8'))
    signature_num = base64_to_num(signature)
    verification_result = pow(signature_num, e, N_RSA)
    return hash_num == verification_result
