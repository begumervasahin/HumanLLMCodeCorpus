# Function to write a number to a file
def writeNumber(n, filename):
    with open(filename, 'wb') as file:
        while n > 0:
            b = n & 0xFF
            n >>= 8
            file.write(bytes([b]))

# Function to read a number from a file
def readNumber(filename):
    with open(filename, 'rb') as file:
        n = 0
        for c in reversed(file.read()):
            n = (n << 8) ^ c
    return n

# Convert hexadecimal text to integer
def hexTextToNum(x):
    res = 0
    for c in x:
        if '0' <= c <= '9':
            res = (res << 4) + ord(c) - ord('0')
        elif 'a' <= c <= 'f':
            res = (res << 4) + ord(c) - ord('a') + 10
        elif 'A' <= c <= 'F':
            res = (res << 4) + ord(c) - ord('A') + 10
    return res

# Calculate modular inverse
def inv(b, m):
    s, t, a = 0, 1, m
    while b != 1:
        q = a // b
        aa = b
        b = a % b
        a = aa
        ss = t
        t = s - q * t
        s = ss
    if t < 0:
        t += m
    return t

# Add two points on elliptic curve
def addP(P, Q):
    x1, x2, y1, y2 = P[0], Q[0], P[1], Q[1]
    while x1 < x2:
        x1 += prime
    if x1 == x2:
        s = ((3*(x1**2) + a) * inv(2*y1, prime)) % prime
    else:
        s = ((y1-y2) * inv(x1-x2, prime)) % prime
    xr = s**2 - x1 - x2
    yr = s * (x1-xr) - y1
    return [xr % prime, yr % prime]

# Multiply a point on elliptic curve by a scalar
def mulP(P, n):
    isFirst = True
    resP = P
    if n < 0:
        resP[1] = prime - resP[1]
        n = (-1)*n
    PP = resP
    while n > 0:
        if (n % 2 != 0):
            if isFirst:
                resP = PP
                isFirst = False
            else:
                resP = addP(resP, PP)
        PP = addP(PP, PP)
        n = n // 2
    return resP

# Verify signature using ECDSA
def verify(G, s, Y, e, m):
    return e == h(str(addP(mulP(G, s), mulP(Y, e))[0]) + m)

# Verify ECDSA signature
def ecdsa_v(G, m, S, Y):
    si = inv(S[0], n4)
    hh = h(m + str(S[1]))
    u1 = (si * hh) % n4
    u2 = (si * S[1]) % n4
    return addP(mulP(G, u1), mulP(Y, u2))[0] == S[1]
