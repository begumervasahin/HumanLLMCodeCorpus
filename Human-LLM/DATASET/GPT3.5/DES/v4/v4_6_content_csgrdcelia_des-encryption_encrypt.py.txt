# Import necessary functions from modules
from utils.alpha_binary_conversion import conv_bin, nib_vnoc
from utils.data_manipulation import (
    permutation, get_subkey_array, packet, list_to_string,
    initial_permutation, cut_off, pse_xor_subkey, xor,
    inverse_initial_permutation
)

# Define the encryption function
def encrypt(message, key):
    # Generate the subkey array using the key
    key = permutation(key, des_constants["CP_1"][0], True)
    subkey_array = get_subkey_array(key, des_constants["CP_2"][0])

    # Convert the message to binary packets
    packets = packet(conv_bin(message))

    # Encrypt each packet and concatenate the results
    encrypted_message = ""
    for packet_data in packets:
        encrypted_message += list_to_string(encrypt_packet(packet_data, subkey_array))

    # Print the encrypted message after applying nibble-to-vector notation
    print(nib_vnoc(encrypted_message))

# Define the packet encryption function
def encrypt_packet(message_part, subkey_array):
    # Perform initial permutation on the message part
    message_part = initial_permutation(message_part)

    # Divide the message part into left and right halves
    left, right = cut_off(message_part)

    # Perform the 16 rounds of encryption
    left, right = perform_rounds(left, right, subkey_array)

    # Combine the left and right halves and apply inverse initial permutation
    message_part = left + right
    return inverse_initial_permutation(message_part)

# Define the round function for encryption
def round_function(left, right, subkey):
    # Apply permutation and XOR operations
    temp_right = pse_xor_subkey(right, subkey)
    temp_right = xor(temp_right, left)
    return right, temp_right

# Define the function to perform the 16 rounds of encryption
def perform_rounds(left, right, subkey_array):
    for index in range(16):
        left, right = round_function(left, right, subkey_array[index])
    return left, right
