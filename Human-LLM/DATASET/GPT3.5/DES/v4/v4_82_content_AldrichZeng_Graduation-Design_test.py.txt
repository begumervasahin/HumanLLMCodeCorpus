import torch
from datetime import datetime
import numpy as np
import config as conf

# Print start time and global step
print(f"{datetime.now()} Start validation")
print(f"{datetime.now()} Global step = {conf.global_step}")

# Evaluate the model on the validation dataset
with torch.no_grad():
    correct = 0
    total = 0
    for val_data in validation_loader:
        net.eval()
        images, labels = val_data
        images, labels = images.to(device), labels.to(device)
        outputs = net(images)
        _, predicted = torch.max(outputs.data, 1)
        total += labels.size(0)
        correct += (predicted == labels).sum()

    correct = float(correct.cpu().numpy().tolist())
    accuracy = correct / total
    print(f"{datetime.now()} Validation Accuracy = {accuracy:.4f}")

    # Save the model if accuracy is higher
    if accuracy > highest_accuracy:
        highest_accuracy = accuracy
        print(f"{datetime.now()} Saving model...")
        torch.save(net.state_dict(), f"{checkpoint_path}/global_step={global_step}.pth")
        print(f"{datetime.now()} Model saved")

        # Update highest accuracy in a file
        with open(highest_accuracy_path, 'w') as f:
            f.write(str(highest_accuracy))
        
        # Update global step in a file
        with open(global_step_path, 'w') as f:
            f.write(str(global_step))
        
        print(f"{datetime.now()} Model saved at global step = {global_step}")
