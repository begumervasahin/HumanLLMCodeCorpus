import os
import re

def concat_files(readme):
    """
    Concatenate files specified in README.md using include and end tags.

    Parameters:
    - readme: str, path to the README.md file
    """
    result_doc = ""
    is_include_section = False

    with open(readme, "r+", encoding="utf-8") as readme_file:
        for line in readme_file.readlines():
            if not is_include_section:
                result_doc += line

            include_match = re.search(r"<!--\s*include\s+(.+?)\s*-->", line)
            if include_match and not is_include_section:
                is_include_section = True
                include_file_path = include_match.group(1)
                with open(include_file_path, "r") as include_file:
                    for include_line in include_file:
                        result_doc += include_line

            elif re.search(r"<!--\s*end\s*-->", line):
                is_include_section = False
                result_doc += "\n" + line
                continue

    with open(readme, "w") as readme_file:
        readme_file.write(result_doc)

if __name__ == '__main__':
    concat_files("README.md")
