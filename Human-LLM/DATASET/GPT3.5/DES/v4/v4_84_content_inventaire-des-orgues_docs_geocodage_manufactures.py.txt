import json
import requests
import geojson

# API URL for geocoding addresses
geocoding_api_url = "https://api-adresse.data.gouv.fr/search/?q="

# Load data from datas.json
with open('datas.json') as json_file:
    data_dict = json.load(json_file)

# Iterate over each address and geocode it
for facteur in data_dict:
    # Extract address and replace spaces with '+'
    address = facteur['adresse'].replace(" ", "+")
    
    # Send request to the geocoding API
    response = requests.get(geocoding_api_url + address)
    
    # Parse the response as GeoJSON
    feature_collection = geojson.loads(response.text)
    
    # Extract latitude and longitude from the response
    coordinates = feature_collection['features'][0]['geometry']['coordinates']
    latitude, longitude = coordinates[1], coordinates[0]
    
    # Update the dictionary with latitude and longitude
    facteur['latitude'] = latitude
    facteur['longitude'] = longitude

# Write the updated data to facteurs_localisation.json
with open('facteurs_localisation.json', 'w') as output_file:
    json.dump(data_dict, output_file)
