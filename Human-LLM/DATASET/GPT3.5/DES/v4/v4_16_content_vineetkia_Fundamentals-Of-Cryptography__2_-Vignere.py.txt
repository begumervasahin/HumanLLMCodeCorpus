def encrypt(plain_text, key):
    """
    Encrypts a plain text using a custom encryption algorithm with the given key.

    Parameters:
        plain_text (str): The text to be encrypted.
        key (str): The encryption key.

    Returns:
        str: The encrypted text.
    """
    result = ''

    for i, char in enumerate(plain_text):
        key_char = key[i % len(key)]
        shift = ord(key_char)

        if char.isupper():
            result += chr(((ord(char) + shift) % 26) + 65)
        else:
            result += chr(((ord(char) + shift) % 26) + 97)

    return result
