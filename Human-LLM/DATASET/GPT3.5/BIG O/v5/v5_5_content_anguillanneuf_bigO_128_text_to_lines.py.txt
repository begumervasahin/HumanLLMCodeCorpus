from itertools import product, combinations

class LineBreaker:
    def __init__(self, raw_text="", width=80):
        self.raw_text = raw_text
        self.width = width
        self.borders = []
        self.linebreaks = []

    def find_borders(self):
        """Find positions where the text should be broken into lines."""
        for i in range(len(self.raw_text)):
            if not self.raw_text[i].isspace():
                if i == 0 or self.raw_text[i - 1].isspace():
                    self.borders.append(i)
                if self.borders and i - self.borders[-1] > self.width:
                    print(f"\"{self.raw_text[self.borders[-1]:i + 1]}\" is longer than the allowed line width {self.width}.")
                    print("Please consider adjusting the line width.")
                    self.borders = []
        self.borders.append(len(self.raw_text))

    def find_next_linebreak(self, position, reversed=False):
        """Find the next line break."""
        if not reversed:
            if position == len(self.borders) - 1:
                return None
            start_index = position
            end_index = len(self.borders) - 1
            if self.borders[end_index] - self.borders[start_index] <= self.width:
                return end_index
            sign = 1
        else:
            if position == 0:
                return None
            start_index = 0
            end_index = position
            if self.borders[end_index] - self.borders[start_index] <= self.width:
                return start_index
            sign = -1

        while start_index < end_index:
            middle_index = (start_index + end_index) // 2
            if self.borders[middle_index - 1] <= self.borders[position] + sign * self.width <= self.borders[middle_index]:
                position = middle_index - 1 * (not reversed)
                break
            elif self.borders[middle_index] > self.borders[position] + sign * self.width:
                end_index = middle_index
            else:
                start_index = middle_index
        return position

    def find_linebreaks(self):
        """Find optimal line breaks."""
        front_push = []
        position = 0
        while position is not None and position <= len(self.borders):
            front_push.append((position, self.borders[position]))
            position = self.find_next_linebreak(position)

        back_push = []
        position = len(self.borders) - 1
        while position is not None and position >= 0:
            back_push.append((position, self.borders[position]))
            position = self.find_next_linebreak(position, reversed=True)

        min_cost = float('inf')
        possible_ranges = []
        for back, front in zip(back_push[::-1], front_push):
            possible_ranges.append(range(back[0], front[0] + 1))

        count = 0
        for scenario in product(*possible_ranges):
            count += 1
            cost = 0
            ok = True
            for index in range(1, len(scenario)):
                difference = self.borders[scenario[index]] - self.borders[scenario[index - 1]]
                if difference > self.width:
                    cost = float('inf')
                    ok = False
                    break
                else:
                    cost += (self.width - difference) ** 2
            if ok:
                self.linebreaks = [self.borders[index] for index in scenario]

        print(f"My solution runs {count} times.")
        print(f"{self.linebreaks} has least cost of {min_cost}")

    def find_linebreaks_brute_force(self):
        """Find line breaks using a brute-force approach."""
        borders = self.borders[1:len(self.borders) - 1]
        n = len(self.raw_text) // self.width
        min_cost = float('inf')
        count = 0
        for scenario in combinations(borders, n):
            ok = True
            cost = 0
            scenario = [0] + list(scenario) + [len(self.raw_text)]
            count += 1
            for index in range(1, len(scenario)):
                difference = scenario[index] - scenario[index - 1]
                if difference > self.width:
                    cost = float('inf')
                    ok = False
                    break
                else:
                    cost += (self.width - difference) ** 2
            if cost < min_cost:
                min_cost = cost
                self.linebreaks = scenario

        print(f"Brute force solution runs {count} times.")
        print(f"{self.linebreaks} has least cost of {min_cost}")

    def break_text_into_lines(self):
        """Break text into lines based on line breaks found."""
        lines = ['']
        position = 0
        offset = 0
        for border in self.linebreaks:
            if border - offset < self.width:
                lines[-1] += self.raw_text[position:border]
            else:
                lines.append(self.raw_text[position:border])
                offset = position
            position = border
        return lines

    raw_text = "Try this: Given a string of English text and a paragraph width, design an algorithm to break the " \
               "texts into lines not exceeding the paragraph width, and not too jagged."
    lines = LineBreaker(raw_text, 80)
    lines.find_borders()
    lines.find_linebreaks_brute_force()
    lines.find_linebreaks()
    print(lines.break_text_into_lines())
