def update_cell(row, col, board):
    """
    Update the choices available for a cell in the Sudoku board.
    """
    choices = []

    for k in [str(v) for v in range(1, 10)]:
        collision = False

        for y in range(9):
            if (
                board[row][y] == k
                or board[y][col] == k
                or board[row - row % 3 + y // 3][col - col % 3 + y % 3] == k
            ):
                collision = True
                break

        if not collision:
            choices.append(k)

    return choices


def print_board(board):
    """
    Print the Sudoku board.
    """
    for row in board:
        print(" ".join(row))


def sudoku_solve(board):
    """
    Solve the Sudoku puzzle recursively.
    """
    choices = None
    row = -1
    col = -1

    # Find the first empty cell
    for i in range(9):
        for j in range(9):
            if board[i][j] == ".":
                temp = update_cell(i, j, board)
                if choices is None or (temp and len(temp) < len(choices)):
                    choices = temp
                    row = i
                    col = j

    # Base case: if no empty cell found, the puzzle is solved
    if choices is None:
        print_board(board)
        return True

    # Try each choice for the empty cell recursively
    for v in choices:
        board[row][col] = v
        if sudoku_solve(board):
            return True
        board[row][col] = "."

    return False
