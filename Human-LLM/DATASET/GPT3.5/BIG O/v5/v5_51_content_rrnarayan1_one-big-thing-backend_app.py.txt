# Route for the home page
@app.route('/')
def hello():
    return "Hello World"

# Route for getting game details
@app.route('/game')
def game():
    season_id = request.args.get('seasonId')
    team_id = request.args.get('teamId')
    game_id = request.args.get('gameId')

    if not season_id or not team_id or not game_id:
        return abort(400)

    game_details = get_game_details(season_id, team_id, game_id)

    if not game_details:
        return abort(404)

    return jsonify(game_details)

# Route for getting score details
@app.route('/score')
def score():
    season_id = request.args.get('seasonId')
    team_id = request.args.get('teamId')
    game_id = request.args.get('gameId')
    data = request.args.get('data')
    portion = request.args.get('portion')

    if not season_id or not team_id or not game_id:
        return abort(400)

    score_details = get_score_details(season_id, team_id, game_id, data, portion)

    if not score_details:
        return abort(404)

    return jsonify(score_details)

# Route for getting game information
@app.route('/games')
def games():
    season_id = request.args.get('seasonId')
    team_id = request.args.get('teamId')
    with_obt = request.args.get('withOBT')
    with_summary = request.args.get('withSummary')

    if not season_id or not team_id:
        return abort(400)

    games_info = get_games_info(season_id, team_id, with_obt, with_summary)

    return jsonify(games_info)

# Helper function to retrieve game details
def get_game_details(season_id, team_id, game_id):
    game = get_game(db, season_id, team_id, game_id)
    if not game:
        return None
    return game

# Helper function to retrieve score details
def get_score_details(season_id, team_id, game_id, data, portion):
    game = get_game(db, season_id, team_id, game_id)
    opp_game = get_opp_game(db, season_id, team_id, game_id)
    if not game or not opp_game:
        return None

    team = get_team_by_id(db, game[u"TEAM_ID"])
    opp_team = get_team_by_id(db, opp_game[u"TEAM_ID"])
    if not team or not opp_team:
        return None

    if not data:
        data_season_id = int(game[u"SEASON_ID"])
        data, portion = _get_default_data(data_season_id, game)

    if portion == "before" and game[u"SEASON_ID"] != data:
        return None

    list_data_games = get_games_stats(db, data, portion, team_id, game)
    if not list_data_games:
        return None

    response = _get_scores(list_data_games, game)
    response["team"] = team
    response["opp_team"] = opp_team
    return response

# Helper function to retrieve game information
def get_games_info(season_id, team_id, with_obt, with_summary):
    games = get_games_info_by_team_and_season(db, season_id, team_id, with_obt, with_summary)
    return games
