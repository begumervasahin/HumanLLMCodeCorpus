from time import sleep
import os

# Function to clear the console screen
def clear():
    os.system("cls" if os.name == "nt" else "clear")

# Main function to handle user input and logic
def main():
    clear()
    
    # Get content input from the user
    content = input("What is the content? e.g: He, N or D: ").lower().replace(" ", "")
    clear()

    # Logic based on content input
    if content == "he" or content == "n" or content == "d":
        # Get nozzle input
        nozzle = input("How many nozzles?: ")
        clear()
        
        # Logic based on nozzle input
        if nozzle in ["1", "2", "3"]:
            clear()
            print("Engine")
            sleep(3)
        else:
            clear()
            print("I can't understand that")
            sleep(3)
    else:
        clear()
        print("Something went wrong! Try again")
        sleep(3)

if __name__ == "__main__":
    main()
