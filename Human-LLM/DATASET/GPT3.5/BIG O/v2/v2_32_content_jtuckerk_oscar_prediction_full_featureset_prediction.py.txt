# Function to prepare input data (X) and output labels (Y) for classification
def get_X_y(input_dict, output_dict):
    X = []
    y = []

    for k in input_dict.keys():
        X.append(input_dict.get(k, np.zeros(20)))
        y.append(output_dict.get(k, 0))

    return np.array(X), np.array(y)

# Function to perform cross-validation and evaluate classifiers
def test_classifier(clf, X, Y):
    folds = StratifiedKFold(n_splits=5)
    aucs = []

    for train, test in folds.split(X, Y):
        clf.fit(X[train], Y[train])
        prediction = clf.predict_proba(X[test])
        aucs.append(roc_auc_score(Y[test], prediction[:, 1]))

    print(clf.__class__.__name__, aucs, np.mean(aucs))

# Main function to execute the classification process
def main():
    X, Y = get_X_y(get_input(), get_output())

    # Test SGD Classifier
    clf = SGDClassifier(loss='log')
    test_classifier(clf, X, Y)

    # Test Gaussian Naive Bayes Classifier
    clf = GaussianNB()
    test_classifier(clf, X, Y)

    # Test Random Forest Classifier
    clf = RandomForestClassifier(n_estimators=10, max_depth=10)
    test_classifier(clf, X, Y)

if __name__ == '__main__':
    main()
