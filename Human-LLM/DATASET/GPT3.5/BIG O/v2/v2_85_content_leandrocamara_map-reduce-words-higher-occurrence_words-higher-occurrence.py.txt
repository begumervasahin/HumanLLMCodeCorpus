import csv
import glob
import mincemeat
import os

# Define the path to the directory containing text files
path = '/home/leandro/Documents/puc/06-solutions/activities/map-reduce-words-higher-occurrence/'

# Get a list of all text files in the directory
text_files = glob.glob(os.path.join(path, 'data', '*'))

# Function to read the contents of a file
def file_contents(file_name):
    with open(file_name, 'r') as f:
        return f.read()

# Map function to process each line in a file
def mapfn(k, v):
    # Import the stopwords
    from stopwords import allStopWords
    
    # Iterate over each line in the file
    for line in v.splitlines():
        # Split the line into fields
        fields = line.split(':::')
        authors = fields[1]
        title = fields[2]
        valid_book = ''
        
        # Process each word in the title
        for word in title.split():
            # Remove punctuation and check if the word is not a stopword
            word = word.replace('.', '').replace(',', '').replace(':', '').replace('?', '').replace('(', '').replace(')', '').replace('"', '').replace("''", '')
            if word not in allStopWords:
                # Check if the author is Grzegorz Rozenberg or Philip S. Yu
                for author in authors.split('::'):
                    if author == 'Grzegorz Rozenberg' or author == 'Philip S. Yu':
                        # If the book is valid, yield the word and author
                        valid_book = 'author: ' + author + '; title: ' + title
                        yield word, author
        if valid_book != '':
            print(valid_book)

# Reduce function to aggregate word occurrences per author
def reducefn(k, v):
    authors = ''
    total_philip = total_rozenberg = 0
    has_philip = has_rozenberg = False
    
    # Count occurrences of each author
    for item in v:
        if item == 'Grzegorz Rozenberg':
            has_rozenberg = True
            total_rozenberg += 1
        if item == 'Philip S. Yu':
            has_philip = True
            total_philip += 1
    
    # Create a string with author counts if the counts are greater than 5
    if total_philip > 5 or total_rozenberg > 5:
        L = list()
        authors += 'Grzegorz Rozenberg: ' + str(total_rozenberg) + '; ' if has_rozenberg else ''
        authors += 'Philip S. Yu: ' + str(total_philip) if has_philip else ''
        L.append(authors)
        return L
    else:
        return None

# Create a dictionary with file contents for each text file
source = {file_name: file_contents(file_name) for file_name in text_files}

# Set up the mincemeat server
s = mincemeat.Server()
s.datasource = source
s.mapfn = mapfn
s.reducefn = reducefn

# Run the server to process the data
results = s.run_server(password="mapreduce")

# Write the results to a CSV file
with open(os.path.join(path, 'result.csv'), 'w') as csvfile:
    writer = csv.writer(csvfile)
    for k, v in results.items():
        if v is not None:
            writer.writerow([k, v])
