import sys

class User:
    def __init__(self, username, is_moderator=False):
        self.username = username
        self.is_moderator = is_moderator

class DatabaseManager:
    @staticmethod
    def drop_all_tables():
        print("Dropping all tables")

    @staticmethod
    def create_all_tables():
        print("Creating all tables")

class ApplicationContext:
    @staticmethod
    def get_context():
        return None

class UserQuery:
    @staticmethod
    def find_by_username(username):
        return None

    @staticmethod
    def add(user):
        pass

class ModeratorService:
    @staticmethod
    def make_moderator(username):
        user = UserQuery.find_by_username(username)
        if not user:
            return f"{username} is not a recognized user"
        if user.is_moderator:
            return f"{username} is already a moderator"
        user.is_moderator = True
        UserQuery.add(user)
        return f"{username} is now a moderator"

    @staticmethod
    def remove_moderator(username):
        user = UserQuery.find_by_username(username)
        if not user:
            return f"{username} is not a recognized user"
        if not user.is_moderator:
            return f"{username} is not a current moderator"
        user.is_moderator = False
        UserQuery.add(user)
        return f"{username} is no longer a moderator"

class DatabaseManagementService:
    @staticmethod
    def replace_database():
        DatabaseManager.drop_all_tables()
        DatabaseManager.create_all_tables()
        return "Dropped all tables. Created all tables."

if __name__ == '__main__':
    method = sys.argv[1]
    if method == 'mod':
        print(ModeratorService.make_moderator(sys.argv[2]))
    elif method == 'rmmod':
        print(ModeratorService.remove_moderator(sys.argv[2]))
    elif method == 'newdb':
        wipe = input('This will delete all data. Are you sure? (Y/n): ')
        if wipe.upper() == 'Y':
            print(DatabaseManagementService.replace_database())
