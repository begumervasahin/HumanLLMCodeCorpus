import sys
import csv
import operator

def read_input():
    """Reads input data from standard input."""
    storage = {}
    for line in sys.stdin:
        value, key = line.split()
        storage[key] = int(value)
    return storage

def sort_storage(storage):
    """Sorts the storage dictionary by values in descending order."""
    return sorted(storage.items(), key=operator.itemgetter(1), reverse=True)

def print_sorted_data(sorted_storage):
    """Prints the sorted key-value pairs."""
    for key, value in sorted_storage:
        print(f"{key},{value}")

def main():
    """Main function to orchestrate the process."""
    storage = read_input()
    sorted_storage = sort_storage(storage)
    print_sorted_data(sorted_storage)

if __name__ == "__main__":
    main()
