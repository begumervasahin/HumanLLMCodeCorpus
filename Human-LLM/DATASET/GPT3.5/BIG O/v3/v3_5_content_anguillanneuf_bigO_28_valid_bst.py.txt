class BinaryTreeNode:
    """
    Represents a node in a binary tree.
    """

    def __init__(self, value):
        """
        Initialize a binary tree node with a given value.

        Args:
            value: The value of the node.
        """
        self.value = value
        self.left = None
        self.right = None

    def insert_left(self, value):
        """
        Insert a left child node with the given value.

        Args:
            value: The value of the left child node.
        """
        self.left = BinaryTreeNode(value)
        return self.left

    def insert_right(self, value):
        """
        Insert a right child node with the given value.

        Args:
            value: The value of the right child node.
        """
        self.right = BinaryTreeNode(value)
        return self.right

    def check_binary_search_tree_iterative(self):
        """
        Check if the binary tree is a binary search tree (BST) iteratively.
        Returns True if the tree is a BST, False otherwise.
        """
        stack = []
        stack.append((self, float('-inf'), float('inf')))
        while stack:
            node, lower, upper = stack.pop()
            if not lower < node.value < upper:
                return False
            if node.left:
                stack.append((node.left, lower, node.value))
            if node.right:
                stack.append((node.right, node.value, upper))
        return True

    def check_bst_recursive(self, node=None, lower=float('-inf'), upper=float('inf')):
        """
        Check if the binary tree is a binary search tree (BST) recursively.
        Returns True if the tree is a BST, False otherwise.

        Args:
            node (BinaryTreeNode, optional): The current node being checked.
            lower (int, optional): The lower bound for node values.
            upper (int, optional): The upper bound for node values.

        Returns:
            bool: True if the tree is a BST, False otherwise.
        """
        if node is None:
            node = self
        if not node:
            return True
        if not lower < node.value < upper:
            return False
        return self.check_bst_recursive(node.left, lower, node.value) and \
               self.check_bst_recursive(node.right, node.value, upper)

# Create a binary tree and test the methods
bt = BinaryTreeNode(5)
bt.insert_left(3)
bt.insert_right(7)
bt.left.insert_left(1)
bt.left.insert_right(4)

# Check if the tree is a binary search tree
print("Iterative Check:", bt.check_binary_search_tree_iterative())  # Expected output: True
print("Recursive Check:", bt.check_bst_recursive())  # Expected output: True
