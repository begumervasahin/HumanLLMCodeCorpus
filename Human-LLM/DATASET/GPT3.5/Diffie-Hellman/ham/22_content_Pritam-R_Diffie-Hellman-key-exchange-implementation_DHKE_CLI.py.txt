import socket
from random import randint
import re
import random
class DHE:
	def __init__(self,root,prime,recv_key):
		self.root=root
		self.prime=prime
		self.recv_key=recv_key
	def key_gen(self):
		range_start = 10**(7-1)
		range_end = (10**7)-1
		self.key = randint(range_start, range_end)
	def shared_key(self):
		mod=(self.root**self.key) % self.prime
		return str(mod)
	def secret_key(self):
		sec_key=(self.recv_key**self.key) % self.prime
		return sec_key
