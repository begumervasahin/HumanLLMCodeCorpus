        if (hexLines == ''):

    def splitPoint(hexLines):
        gen=hexLines.replace(' ','').replace(':','')[2:]
        gl=len(gen)
        return (int(gen[:gl/2],16), int(gen[gl/2:], 16))

def modinv(a,n=p):
    lm, hm = 1,0
    low, high = a%n,n
    while low > 1:
        ratio = high/low
        nm, new = hm-lm*ratio, high-low*ratio
        lm, low, hm, high = nm, new, lm, low
    return lm % n
def ECadd(a,b):
    LamAdd = ((b[1]-a[1]) * modinv(b[0]-a[0],p)) % p
    x = (LamAdd*LamAdd-a[0]-b[0]) % p
    y = (LamAdd*(a[0]-x)-a[1]) % p
    return (x,y)
def ECdouble(a):
    Lam = ((3*a[0]*a[0]+A) * modinv((2*a[1]),p)) % p
    x = (Lam*Lam-2*a[0]) % p
    y = (Lam*(a[0]-x)-a[1]) % p
    return (x,y)

def encrypt(key, plaintext, associated_data):
    iv = os.urandom(12)
    encryptor = Cipher(
        algorithms.AES(key),
        modes.GCM(iv),
        backend=default_backend()
    ).encryptor()
    encryptor.authenticate_additional_data(associated_data)
    ciphertext = encryptor.update(plaintext) + encryptor.finalize()
    return (iv, ciphertext, encryptor.tag)
def decrypt(key, associated_data, iv, ciphertext, tag):
    decryptor = Cipher(
        algorithms.AES(key),
        modes.GCM(iv, tag),
        backend=default_backend()
    ).decryptor()
    decryptor.authenticate_additional_data(associated_data)
    return decryptor.update(ciphertext) + decryptor.finalize()
