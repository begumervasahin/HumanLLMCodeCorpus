import socket
import random
from Crypto import Random
from Crypto.Cipher import AES
from Crypto.PublicKey import RSA
import hashlib

# Create a socket
server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
print("Socket created")

# Set socket options and bind
server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
server.bind(('localhost', 8080))
print("Socket bind complete")

# Listen for connections
server.listen(5)
print("Server is now listening...")

# Accept incoming connections
connection, address = server.accept()
print("Connected with", address[0])

# Receive ID from client
data = connection.recv(1024)
print("Received ID:", data)

# Send challenge to client
challenge = Random.new().read(16)
connection.sendall(challenge)

# Receive response from client
response = connection.recv(1024)

# Decrypt response and check if it matches the challenge
password = "Abc1234"
key = hashlib.sha256(password.encode()).digest()
IV = 16 * b'\x00'
decryptor = AES.new(key, AES.MODE_CFB, IV)
plain = decryptor.decrypt(response)
if plain == challenge:
    connection.sendall("Welcome!")
    print("Client logged in")
else:
    connection.sendall("Incorrect password!")
    exit()

# Receive g, p, g^a mod(p), sign(A) from client
g = connection.recv(1024)
p = connection.recv(1024)
A = connection.recv(1024)
signA = connection.recv(1024)
print("Received g, p, g^a mod(p), sign(A) from Client")
print("g:", g)
print("p:", p)
print("A:", A)
print("signA:", signA)

# Verify the signature
with open("privatekey1.txt", "r") as file:
    RSAkey1 = RSA.importKey(file.read())
checkHash = RSAkey1.decrypt(signA)
hashA = hashlib.sha256(A).digest()

with open("publickey2.txt", "r") as file:
    RSAkey2 = RSA.importKey(file.read())
print("hashA:", hashA)
print("CheckHashA:", checkHash)

if hashA == checkHash:
    print("CheckHash Success!")
    b = 15
    B = (int(g) ** b) % int(p)
    print("Secret value b:", b)
    hashB = hashlib.sha256(str(B).encode()).digest()
    signB = RSAkey2.encrypt(hashB, 32)
    connection.sendall(str(B))
    connection.sendall(signB[0])
    print("B:", B)
    print("signB:", signB[0])
    print("Send g^b mod(p), sign(B) to Client")
    sessionKey = str((int(A) ** b) % int(p))
    print("Generated session key:", sessionKey)
else:
    print("CheckHash failed!")
    exit()

# Send encrypted EMR to client
with open("EMR.txt", "r") as file:
    EMRfile = file.read()
sessionKey = hashlib.sha256(sessionKey.encode()).digest()
encryptor = AES.new(sessionKey, AES.MODE_CFB, IV)
encryptedEMR = encryptor.encrypt(EMRfile)
connection.send(encryptedEMR)
print("Send encrypted EMR to Client")
print("Encrypted file:", encryptedEMR)

# Close the connection and server
connection.close()
server.close()
