from rsa import RSA
from convertN2T import *

def encrypt_message(msg, d, n):
    """Encrypts the message using RSA."""
    encrypted_msg = [modExp(char, d, n) for char in msg]
    return encrypted_msg

def create_signature(msg, e, n):
    """Creates a digital signature for the message using RSA."""
    signature = [modExp(char, e, n) for char in msg]
    return signature

def decrypt_verify(msg, na, ea, nb, db):
    """Decrypts and verifies the digital signature of the received message."""
    decrypted_msg_a = [modExp(char, ea, na) for char in msg]
    decrypted_msg_b = [modExp(char, db, nb) for char in decrypted_msg_a]
    return decrypted_msg_b

def verify(msg, original_msg):
    """Verifies the received message against the original message."""
    print("Received signed message:\n\n", msg, "\n")
    print("Original message sent:\n\n", original_msg, "\n")

if __name__ == '__main__':
    block_size = 15
    rsa_b = RSA()
    rsa_a = RSA()

    with open("original_msg") as msg_file:
        original_msg = msg_file.read()
    
    num_list = str2num(original_msg)
    blocks = num2block(num_list, block_size)

    (Nb, Eb) = rsa_b.publicKey()
    (Nb, Db) = rsa_b.privateKey()
    (Na, Ea) = rsa_a.publicKey()
    (Na, Da) = rsa_a.privateKey()

    y_signed = create_signature(blocks, Eb, Nb)
    y_encrypted = encrypt_message(y_signed, Da, Na)
    x_verified_signature = decrypt_verify(y_signed, Na, Ea, Nb, Db)

    num_list = block2num(x_verified_signature, block_size)
    verified_msg = num2str(num_list)

    verify(verified_msg, original_msg)
