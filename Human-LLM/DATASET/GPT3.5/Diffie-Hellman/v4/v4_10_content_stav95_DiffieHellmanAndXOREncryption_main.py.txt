from flask import Flask, request
from threading import Thread
from argparse import ArgumentParser
import random
import requests
import time

app = Flask(__name__)

def main():
    """Starts the Flask application based on the provided command-line arguments."""
    parser = ArgumentParser()
    parser.add_argument('--porta', type=int, help="Port number to run the Flask application on")
    parser.add_argument('--portb', type=int, help="Port number for the peer")
    args = parser.parse_args()

    app.port_b = args.portb
    app.run(host='localhost', port=args.porta, debug=True)

def diffie_hellman_algo():
    """Performs the Diffie-Hellman key exchange algorithm."""
    app.prime = generate_prime(100)
    app.g = find_primitive_root(app.prime)
    app.n = random.randint(2, app.prime)

def generate_prime(range_max):
    """Generates a prime number within the specified range."""
    primes = [x for x in range(3, range_max) if all(x % y != 0 for y in range(2, x))]
    return random.choice(primes)

def find_primitive_root(prime):
    """Finds a primitive root modulo of a prime number."""
    found_g = [a for a in range(1, prime) if len(set(pow(a, x) % prime for x in range(1, prime))) == prime - 1]
    return random.choice(found_g)

def send_request(url, is_get, data):
    """Sends a GET or POST request to the specified URL."""
    time.sleep(0.01)
    if is_get:
        requests.get(url=url)
    else:
        requests.post(url=url, data=data)

@app.route('/init', methods=['GET'])
def initiate():
    """Initiates the Diffie-Hellman key exchange process."""
    diffie_hellman_algo()
    a = pow(app.g, app.n) % app.prime
    url = f'http://127.0.0.1:{app.port_b}/exchange/1?p={app.prime}&g={app.g}&A={a}'
    Thread(target=send_request, args=(url, True, None)).start()
    return ''

@app.route('/exchange/1', methods=['GET'])
def exchange1():
    """Handles the first exchange in the Diffie-Hellman key exchange."""
    prime = int(request.args.get('p'))
    g = int(request.args.get('g'))
    n = random.randint(2, prime)
    app.s_key = pow(int(request.args.get('A')), n) % prime
    b = pow(g, n) % prime
    url = f'http://127.0.0.1:{app.port_b}/exchange/2?B={b}'
    Thread(target=send_request, args=(url, True, None)).start()
    print('Secret key for Bob:', app.s_key, '(Keep it secret)')
    return ''

@app.route('/exchange/2', methods=['GET'])
def exchange2():
    """Handles the second exchange in the Diffie-Hellman key exchange."""
    app.s_key = pow(int(request.args.get('B')), app.n) % app.prime
    print('Secret key for Alice:', app.s_key, '(Keep it secret)')
    return ''

@app.route('/send_plain_message', methods=['POST'])
def send_plain_message():
    """Sends a plain message using the established shared key."""
    url = f'http://127.0.0.1:{app.port_b}/receive_secure_message'
    cipher_data = encrypt_message(request.data, app.s_key)
    response = str(requests.post(url=url, data=cipher_data).content)
    print('Encrypted message:', response)
    print('Decrypted message:', decrypt_message(response, app.s_key))
    return ''

@app.route('/receive_secure_message', methods=['POST'])
def receive_secure_message():
    """Receives and processes a secure message using the shared key."""
    plain_text = decrypt_message(request.data, app.s_key)
    print('Encrypted message received:', request.data)
    print('Decrypted message:', plain_text)
    plain_text += 'ack'
    return encrypt_message(plain_text, app.s_key)

def encrypt_message(message, secret_key):
    """Encrypts a message using XOR with the shared key."""
    return ''.join(chr(ord(char) ^ secret_key) for char in message)

def decrypt_message(cipher_text, secret_key):
    """Decrypts a message using XOR with the shared key."""
    return encrypt_message(cipher_text, secret_key)

if __name__ == '__main__':
    main()
