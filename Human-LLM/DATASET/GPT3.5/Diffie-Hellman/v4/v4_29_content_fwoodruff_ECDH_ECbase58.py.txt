import TypeConversion as TC
import ModularOperations as mod
import ECRing

def point_to_base58(P, curve):
    """
    Convert a point to a base-58 string representation.

    Args:
        P (ECRing.Point): The point to convert.
        curve (ECRing.Curve): The elliptic curve parameters.

    Returns:
        str: The base-58 string representation of the point.
    """
    key58 = TC.int_to_base58(P.x)
    if P.y > curve.p / 2:
        return '03' + key58
    else:
        return '02' + key58

def base58_to_point(P58, curve):
    """
    Convert a base-58 string representation to a point.

    Args:
        P58 (str): The base-58 string representation of the point.
        curve (ECRing.Curve): The elliptic curve parameters.

    Returns:
        ECRing.Point: The point represented by the base-58 string.
    """
    a, p, b = curve.a, curve.p, curve.b
    key_x = TC.b58_to_int(P58[2:])
    key_y_sq = (key_x ** 3 + a * key_x + b) % p
    key_y = mod.modular_sqrt(key_y_sq, p)
    if P58[0] == '0' and P58[1] == '3':
        if key_y > curve.p / 2:
            return ECRing.Point(key_x, key_y)
        else:
            return ECRing.Point(key_x, p - key_y)
    else:
        if key_y > curve.p / 2:
            return ECRing.Point(key_x, p - key_y)
        else:
            return ECRing.Point(key_x, key_y)
