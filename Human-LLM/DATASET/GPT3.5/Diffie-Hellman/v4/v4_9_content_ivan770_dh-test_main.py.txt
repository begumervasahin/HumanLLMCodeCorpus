import random

# Generate a random number for the key exchange
random_number = random.randint(1, 999999999)
print(f"Using {random_number} as a random number")

# Generate random private keys for two users
user1_private_key = random.randint(1, 999999999)
user2_private_key = random.randint(1, 999999999)
private_user1 = user1_private_key + random_number
private_user2 = user2_private_key + random_number
print(f"Using {private_user1} as user1's private key")
print(f"Using {private_user2} as user2's private key")

# Generate public keys for both users
public_user1 = private_user1 % 256
public_user2 = private_user2 % 256
print(f"Using {public_user1} as user1's public key")
print(f"Using {public_user2} as user2's public key")

# Calculate intermediate values for shared key generation
pre_user1 = public_user2 + user1_private_key
pre_user2 = public_user1 + user2_private_key
print(f"Using {pre_user1} for generating user1's shared key")
print(f"Using {pre_user2} for generating user2's shared key")

# Generate shared keys for both users
shared_user1 = pre_user1 % 256
shared_user2 = pre_user2 % 256
print(f"Using {shared_user1} as the result of user1's key generation")
print(f"Using {shared_user2} as the result of user2's key generation")

# Check if the shared keys are equal
print("Checking equality...")
if shared_user1 == shared_user2:
    print("Correct! Key exchange completed successfully!")
else:
    print("Oops! There seems to be an error in the key exchange process.")

input("Press Enter to exit...")
