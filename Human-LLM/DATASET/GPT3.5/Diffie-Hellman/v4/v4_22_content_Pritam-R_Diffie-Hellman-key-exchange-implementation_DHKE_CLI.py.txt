import socket
from random import randint
import re

class DiffieHellmanExchange:
    def __init__(self, root, prime, recv_key):
        self.root = root
        self.prime = prime
        self.recv_key = recv_key
    
    def generate_key(self):
        """Generate a random key."""
        range_start = 10 ** (7 - 1)
        range_end = (10 ** 7) - 1
        self.key = randint(range_start, range_end)
    
    def calculate_shared_key(self):
        """Calculate the shared key."""
        mod = (self.root ** self.key) % self.prime
        return str(mod)
    
    def calculate_secret_key(self):
        """Calculate the secret key."""
        sec_key = (self.recv_key ** self.key) % self.prime
        return sec_key

            data = sock.recv(4096)

                if pattern:
                    root = int(pattern.group(1))
                    prime = int(pattern.group(2))
                    recv_key = int(pattern.group(3))

                    dho.generate_key()

                    message1 = f"SHARED_KEY-{IP}-{shared_key}"
