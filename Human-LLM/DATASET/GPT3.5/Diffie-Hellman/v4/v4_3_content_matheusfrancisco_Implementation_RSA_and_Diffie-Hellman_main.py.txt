from rsa import RSA
from convertN2T import *

def main():
    print('================== Criptografia RSA ======================')
    block_size = 15
    rsa_instance = RSA()
    
    # Read the original message from the file
    with open("original_msg", "r") as msg_file:
        original_message = msg_file.read()
    
    print('================== Mensagem Original ======================')
    print(original_message)
    print(' ')

    # Convert the original message to a list of ASCII values
    ascii_list = string_to_ascii_list(original_message)
    
    print('================== Lista de Valores ASCII ======================')
    print(ascii_list)
    print(' ')
    
    # Convert the ASCII values to blocks of a specified size
    blocks = num_to_blocks(ascii_list, block_size)
    
    # Encrypt the blocks using RSA encryption
    encrypted_blocks = rsa_instance.encryption(blocks)
    
    print("================== Bloco Criptografado ======================")
    print(encrypted_blocks)
    print(' ')
    
    # Decrypt the encrypted blocks using RSA decryption
    decrypted_blocks = rsa_instance.decryption(encrypted_blocks)
    
    # Convert the decrypted blocks back to ASCII values
    decrypted_ascii_list = blocks_to_num(decrypted_blocks, block_size)
    
    print('================== Lista de Valores ASCII Decodificada ======================')
    print(decrypted_ascii_list)
    print(' ')
    
    # Convert the ASCII values back to the decrypted message
    decrypted_message = num_to_str(decrypted_ascii_list)
    
    print('================== Mensagem Decodificada ======================')
    print(decrypted_message)

if __name__ == '__main__':
    main()
