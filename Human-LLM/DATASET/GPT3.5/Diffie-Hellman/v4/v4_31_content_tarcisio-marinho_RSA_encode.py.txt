import random

def totient(number):
    """
    Calculates the totient of a prime number.
    """
    if is_prime(number):
        return number - 1
    else:
        return False

def is_prime(n):
    """
    Checks if a number is prime.
    """
    if n <= 1:
        return False
    if n <= 3:
        return True
    if n % 2 == 0 or n % 3 == 0:
        return False
    i = 5
    while i * i <= n:
        if n % i == 0 or n % (i + 2) == 0:
            return False
        i += 6
    return True

def generate_E(num):
    """
    Generates a random number 'E' satisfying the conditions.
    """
    def gcd(n1, n2):
        """
        Calculates the greatest common divisor of two numbers.
        """
        while n2 != 0:
            remainder = n1 % n2
            n1 = n2
            n2 = remainder
        return n1
    
    while True:
        e = random.randrange(2, num)
        if gcd(num, e) == 1:
            return e

def generate_prime():
    """
    Generates a random prime number.
    """
    while True:
        x = random.randint(1, 100)
        if is_prime(x):
            return x

def mod(a, b):
    """
    Modular function between two numbers.
    """
    if a < b:
        return a
    else:
        c = a % b
        return c

def decipher(ciphered_text, n, d):
    """
    Decrypts an encrypted message.
    """
    decrypted_text = []
    for ciphertext in ciphered_text:
        result = pow(ciphertext, d, n)
        char_code = mod(result, n)
        char = chr(char_code)
        decrypted_text.append(char)
    return decrypted_text

def calculate_private_key(totient, e):
    """
    Calculates the private key.
    """
    d = 0
    while (d * e) % totient != 1:
        d += 1
    return d
