from time import sleep
import random
import socket

def is_prime(n):
    """Check if a number is prime."""
    for i in range(2, n):
        if n % i == 0:
            return False
    return True

def generate_random_prime(limit=2000):
    """Generate a random prime number."""
    primes_list = [p for p in range(2, limit + 1) if is_prime(p)]
    return random.choice(primes_list)

def alice_send_message(num):
    """Send a number to Bob and receive a message."""
    from_bob = ""
    host = "localhost"
    port = 42428
    backlog = 5
    size = 1024

    # Establish a socket for Alice
    s_alice = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    s_alice.bind((host, port))
    s_alice.listen(backlog)

    # Accept the connection from Bob
    conn, clientaddress = s_alice.accept()
    data = conn.recv(size)
    if data:
        datastring = data.decode("utf-8")
        from_bob = datastring
    conn.close()
    s_alice.close()
    sleep(0.1)

    # Create a new socket to send data to Bob
    c_alice = socket.create_connection((host, port))
    snum = str(num)
    buf = snum.encode("utf-8")
    c_alice.sendall(buf)
    c_alice.close()

    return from_bob
