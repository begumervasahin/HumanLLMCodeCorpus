import binascii

# Define the plaintext and keys
plain_text = 'informationsecurity'
key1 = 'vlaksjdhfgqodzmxncb'

# Print the original plaintext and keys
print('The plain text is:', plain_text)
print('The first key is:', key1, '\n')

# Convert the plaintext and keys to binary
plain_text_binary = bin(int(binascii.hexlify(plain_text.encode('utf-8')), 16))
key_binary = bin(int(binascii.hexlify(key1.encode('utf-8')), 16))

# Print the plaintext and keys in binary format
print('Plain text in bits:', plain_text_binary, '\n')
print('Key in bits:', key_binary, '\n')

def xor(m, k):
    """
    Perform XOR operation between two binary strings.
    
    Args:
        m (str): Binary string representing the message.
        k (str): Binary string representing the key.
    
    Returns:
        str: XOR result of the two binary strings.
    """
    result = []
    for i, j in zip(m, k):
        if i.isalpha() and j.isalpha():
            result.append(i)
        else:
            result.append(str(int(i) ^ int(j)))
    return "".join(result)

# Perform XOR encryption with the normal key
encryption_result = xor(plain_text_binary, key_binary)
encrypted_text = binascii.unhexlify('%x' % int(encryption_result, 2))

# Print the encryption result
print('XOR result for the encryption with the normal key:', encryption_result, '\n')
print('Encrypted text:', encrypted_text)

# Perform decryption with the normal key
decryption_result = xor(encryption_result, key_binary)
decrypted_text = binascii.unhexlify('%x' % int(decryption_result, 2))

# Print the decryption result
print('\nDecryption result:', decryption_result, '\n')
print('Decrypted text with the normal key:', decrypted_text, '\n')

# Define the alternative key
alternative_key = 'tlftrffwmixor|{xbch'

# Convert the alternative key to binary
alternative_key_binary = bin(int(binascii.hexlify(alternative_key.encode('utf-8')), 16))

# Perform decryption with the alternative key
decryption_result_alternative = xor(encryption_result, alternative_key_binary)
decrypted_text_alternative = binascii.unhexlify('%x' % int(decryption_result_alternative, 2))

# Print the decryption result with the alternative key
print('Decryption with the alternative key \n')
print('Alternative key in bits:', alternative_key_binary, '\n')
print('Decrypted text with the alternative key:', decrypted_text_alternative)
