def exponential_modulation(a, n, z):
    '''Exponential function used by the algorithm.'''
    a = int(a)
    n = int(n)
    z = int(z)
    result = 1
    
    if n == 0:
        return 1
    else:
        if n % 2 != 0:
            result = exponential_modulation(a, n - 1, z)
            return ((a * result) % z)
        else:
            result = exponential_modulation(a, n / 2, z)
            return ((result * result) % z)
