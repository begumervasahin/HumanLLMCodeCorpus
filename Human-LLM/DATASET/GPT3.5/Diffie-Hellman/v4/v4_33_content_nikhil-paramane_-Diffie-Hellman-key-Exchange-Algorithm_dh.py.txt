from random import randint

def is_prime(num, count):
    """
    Checks if a number is prime using the Miller-Rabin primality test.

    Args:
        num (int): The number to be checked for primality.
        count (int): The number of iterations for the primality test.

    Returns:
        bool: True if the number is prime, False otherwise.
    """
    if num == 1:
        return False
    if count >= num:
        count = num - 1
    for _ in range(count):
        a = randint(1, num - 1)
        if pow(a, num - 1, num) != 1:
            return False
    return True

def generate_big_prime(n_bits):
    """
    Generates a large prime number with the specified number of bits.

    Args:
        n_bits (int): The number of bits for the prime number.

    Returns:
        int: A large prime number.
    """
    found_prime = False
    while not found_prime:
        p = randint(2 ** (n_bits - 1), 2 ** n_bits)
        if is_prime(p, 1000):
            return p

# Generate two large prime numbers
prime1 = generate_big_prime(2 ** 10)
prime2 = generate_big_prime(2 ** 10)

# Generate Alice and Bob's private keys
alice_private_key = randint(2 ** 10, 2 ** 15)
bob_private_key = randint(2 ** 10, 2 ** 15)

# Calculate public keys for Alice and Bob
a = pow(prime2, alice_private_key, prime1)
b = pow(prime2, bob_private_key, prime1)

# Calculate shared secret keys for Alice and Bob
alice_secret_key = pow(b, alice_private_key, prime1)
bob_secret_key = pow(a, bob_private_key, prime1)

# Print the results
print("Prime 1:", prime1)
print("Prime 2:", prime2)
print("Alice's Secret Key:", alice_secret_key)
print("Bob's Secret Key:", bob_secret_key)
