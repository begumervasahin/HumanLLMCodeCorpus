from random import sample

def calculate_prime_primitive(prime):
    # Initialize the number to check for primitive roots
    number_to_check = 0
    print("Calculating primitive roots...")
    
    # List to store primitive roots
    primitive_roots = []

    # Iterate through numbers to find primitive roots
    for candidate_root in range(1, prime):
        number_to_check += 1
        candidate_prime_roots = []

        # Calculate residues for the candidate root
        for exponent in range(1, prime):
            modulus = (number_to_check ** exponent) % prime
            candidate_prime_roots.append(modulus)

            # Check if all residues are unique
            cleaned_up_candidate_prime_roots = set(candidate_prime_roots)
            if len(cleaned_up_candidate_prime_roots) == prime - 1:
                primitive_roots.append(number_to_check)

    # Randomly select a primitive root from the list
    return sample(primitive_roots, 1)[0]

# Example usage
prime_number = int(input("Enter the prime number: "))
primitive_root = calculate_prime_primitive(prime_number)
print(f"The primitive root for the prime number {prime_number} is: {primitive_root}")
