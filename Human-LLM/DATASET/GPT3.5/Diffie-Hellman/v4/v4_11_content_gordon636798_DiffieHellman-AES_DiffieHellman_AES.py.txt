from Crypto.Cipher import AES
import hashlib

a = mpz(11)
q = mpz(1040699159)
print('a:', a, ', q:', q)
data = 'I love ncyucsie 噗嘻~~(≧Ꙫ≦)'
print('Plain:', data, '\n')


class User:
    def __init__(self, name, a, q):
        self.name = name
        self.a = a
        self.q = q
        temp = sum(ord(i) for i in name)
        self.__X = mpz_random(random_state(temp), q)
        self.Y = powmod(self.a, self.__X, self.q)
        self.__key = {}
        print(self.name + '\'s master key:', self.__X)

    def common_key(self, other_user, Yb):
        key = powmod(Yb, self.__X, self.q)
        key_bytes = int(key).to_bytes((key.bit_length() + 7) // 8, byteorder='big')
        key_digest = hashlib.sha256(key_bytes).digest()
        print('\nCommon key for ' + self.name + ' and ' + other_user.name + ':\n' + key_digest.hex())
        self.__key[other_user.name] = key_digest

    def AES_encrypt(self, other_user, data):
        data = data.encode('utf-8')
        padding = 16 - len(data) % 16
        if padding == 0:
            padding = 16
        padding_bytes = bytes([padding])
        data += padding * padding_bytes
        cryptor = AES.new(self.__key[other_user.name], AES.MODE_CBC, 16 * b'\x00')
        cipher_text = cryptor.encrypt(data)
        print('\ncipher:\n' + cipher_text.hex())
        return [cipher_text, self.name]

    def AES_decrypt(self, cip):
        data, name = cip[0], cip[1]
        cryptor = AES.new(self.__key[name], AES.MODE_CBC, 16 * b'\x00')
        plain_text = cryptor.decrypt(data)
        padding_length = plain_text[-1]
        plain_text = plain_text[:-padding_length]
        plain_text = plain_text.decode('utf-8')
        print('\nplain from ' + name + ':\n' + plain_text)
        return plain_text


Alice = User('Alice', a, q)
Bob = User('Bob', a, q)

Alice.common_key(Bob, Bob.Y)
Bob.common_key(Alice, Alice.Y)

cip = Alice.AES_encrypt(Bob, data)
plain = Bob.AES_decrypt(cip)
print('Decrypted message from Bob:', plain)
