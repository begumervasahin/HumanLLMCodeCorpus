import math
import random
import sys
from fractions import gcd

def generate_primes():
    """Generate a list of prime numbers between 1000 and 10000."""
    primes = []
    for num in range(1000, 10001):
        if all(num % i != 0 for i in range(2, int(math.sqrt(num)) + 1)):
            primes.append(num)
    return primes

def pick_random_prime(primes):
    """Select a random prime number from the given list."""
    secure_random = random.SystemRandom()
    return secure_random.choice(primes)

def main():
    # Generate a list of prime numbers
    primes_list = generate_primes()
    # Pick a random prime number
    modulus = pick_random_prime(primes_list)
    print("Modulus Number:", modulus)

    # Generate random base, exponent A, and exponent B
    secure_random = random.SystemRandom()
    base_number = secure_random.choice(range(1, 10001))
    exponent_a = secure_random.choice(range(1, 10001))
    exponent_b = secure_random.choice(range(1, 10001))

    # Check if base and exponents are relatively prime
    if gcd(base_number, exponent_a) != 1:
        print(f"N({base_number}) and A({exponent_a}) must be relatively prime.")
        sys.exit(1)

    if gcd(base_number, exponent_b) != 1:
        print(f"N({base_number}) and B({exponent_b}) must be relatively prime.")
        sys.exit(1)

    # Print the generated numbers and their computational results
    print("Base Number:", base_number)
    print("Exponent A:", exponent_a)
    print("Exponent B:", exponent_b)

    result_a = base_number ** exponent_a % modulus
    result_b = base_number ** exponent_b % modulus

    print("A computational numbers:", result_a)
    print("B computational numbers:", result_b)

    # Check if computed values match
    if result_b ** exponent_a % modulus == result_a ** exponent_b % modulus:
        print("Computation successful! The computed values match.")
    else:
        print("Computation failed! The computed values do not match.")

if __name__ == "__main__":
    main()
