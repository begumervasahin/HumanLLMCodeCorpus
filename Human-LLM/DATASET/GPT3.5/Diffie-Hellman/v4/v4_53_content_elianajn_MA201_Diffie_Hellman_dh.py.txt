import random
import time

def compute_shared(g, p, k):
    """Compute the shared secret using modular exponentiation."""
    return pow(g, k, p)

def compute_random():
    """Generate a random number between 0 and 1000."""
    return random.randint(0, 1000)

def generate_sophie_germain_prime(base_prime, iterations):
    """Generate a Sophie Germain prime number."""
    prime_candidate = base_prime
    for _ in range(iterations):
        prime_candidate = (2 * prime_candidate) + 1
    return prime_candidate

    bob_public = compute_shared(g, p, bob_secret)
