import Client
import Server
from argparse import ArgumentParser

if __name__ == '__main__':
    # Setting up command line argument parser
    parser = ArgumentParser()
    parser.add_argument("-m", "--mode", dest="mode", type=str, required=True,
                        help="Specify CLIENT to start a client or SERVER to start a server"
                        )
    parser.add_argument("-d", "--debug", dest="debug", required=False,
                        help="Enable this option to print debug messages",
                        action="store_true"
                        )
    
    # Parsing command line arguments
    args = parser.parse_args()
    
    # Printing debug messages if debug flag is set
    if args.debug:
        print(args)
    
    # Starting client or server based on mode
    if args.mode.lower() == "client":
        # Configuring server address
        server = "127.0.0.2"
        
        # Creating client socket instance
        client = Client.ClientSocket(args.debug)
        
        # Starting client
        client.start_client(server)
    elif args.mode.lower() == "server":
        # Starting server
        Server.start_server(args.debug)
