import sys
import os
import subprocess
import re
import csv

# Open output CSV file in write mode
outfile = open('output.csv', 'w', newline='')
wr = csv.writer(outfile)

# Open Alexa top 1 million CSV file
with open('alexa_top1mil.csv', 'r') as csvfile:
    reader = csv.reader(csvfile, delimiter=',')
    for row in reader:
        server = row[0]
        servername = row[1]
        
        try:
            # Run OpenSSL command to trace SSL handshake
            cmd = subprocess.check_output([
                os.path.join(os.path.dirname(sys.argv[0]), "openssl-trace"),
                "s_client", "-trace",
                "-cipher", "DHE",
                "-connect", f"{server}:443"
            ],
            stdin=subprocess.DEVNULL, stderr=subprocess.DEVNULL, timeout=1)

            # Extract prime number if present in the output
            for line in cmd.decode("ISO-8859-1").splitlines():
                if 'dh_p' in line:
                    prime = int(re.sub(".*: ", "", line), 16)
                    # Write server information and prime number to CSV
                    wr.writerow([server, servername, prime])
        
        except (subprocess.CalledProcessError, subprocess.TimeoutExpired):
            pass

# Close the output file
outfile.close()
