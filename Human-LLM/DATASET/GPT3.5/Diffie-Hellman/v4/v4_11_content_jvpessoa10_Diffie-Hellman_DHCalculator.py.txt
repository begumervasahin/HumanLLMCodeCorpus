from math import sqrt
from itertools import count, islice
import random

class DHCalculator:
    def __init__(self):
        # Private key 'a', public key 'B', and shared prime 'P'
        self.private_key_a = 0
        self.public_key_B = 0
        self.shared_prime_P = 0

    def is_prime(self, n):
        """Check if a number is prime."""
        return n > 1 and all(n % i for i in islice(count(2), int(sqrt(n) - 1)))

    def generate_shared_prime(self):
        """Generate a shared prime number within a range."""
        min_range = 1000
        max_range = 10000
        primes = [p for p in range(min_range, max_range) if self.is_prime(p)]
        self.shared_prime_P = random.choice([p for p in primes if min_range < p < max_range])

    def set_private_key(self, private_key):
        """Set the private key 'a'."""
        self.private_key_a = private_key

    def calculate_A(self, G, P):
        """Calculate public key 'A' based on generator 'G' and shared prime 'P'."""
        self.public_key_B = G ** self.private_key_a % P

    def calculate_shared_key(self, other_public_key_B):
        """Calculate shared secret key based on other party's public key 'B' and shared prime 'P'."""
        return other_public_key_B ** self.private_key_a % self.shared_prime_P
