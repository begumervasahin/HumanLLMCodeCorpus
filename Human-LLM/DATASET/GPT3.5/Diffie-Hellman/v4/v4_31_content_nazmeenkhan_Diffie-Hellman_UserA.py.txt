import socket
import random

def generate_prime():
    """
    Generate a random prime number between 2 and 99.
    """
    prime = 0
    while prime == 0:
        num = random.randint(2, 99)
        for i in range(2, (num // 2)):
            if (num % i) == 0:
                prime = 0
                break
        else:
            prime = num
            return prime

def gcd(a, b):
    """
    Calculate the greatest common divisor of two numbers.
    """
    while b != 0:
        a, b = b, a % b
    return a

def find_primitive_root():
    """
    Find a primitive root 'alpha' modulo 'q'.
    """
    while True:
        q = generate_prime()
        roots = []
        required_set = set(num for num in range(1, q) if gcd(num, q) == 1)
        for g in range(1, q):
            actual_set = set(pow(g, powers) % q for powers in range(1, q))
            if required_set == actual_set:
                roots.append(g)
                if len(roots) > 10:
                    break
        if len(roots) > 0:
            return q, random.choice(list(roots))

def main():
    # Find 'q' and 'alpha'
    q, alpha = find_primitive_root()
    print("Prime number (q):", q)
    print("Primitive root (alpha):", alpha)

    # Generate private key 'Xa'
    Xa = random.randint(1, q)
    print("Private key (Xa):", Xa)

    # Establish connection with the client
    server = socket.socket()
    port = 5050
    server.bind(('127.0.0.1', port))
    server.listen(10)
    conn, address = server.accept()

    # Send 'q' and 'alpha' to the client
    conn.send(str(q).encode())
    conn.send(str(alpha).encode())

    # Calculate and send 'Ya' to the client
    Ya = pow(alpha, Xa, q)
    print("Public key (Ya):", Ya)
    conn.send(str(Ya).encode())

    # Receive 'Yb' from the client
    Yb = int(conn.recv(1024))

    # Calculate the symmetric key 'Ka'
    Ka = pow(Yb, Xa, q)
    print("Symmetric Key (Ka):", Ka)

    # Close the connection
    conn.close()

if __name__ == '__main__':
    main()
