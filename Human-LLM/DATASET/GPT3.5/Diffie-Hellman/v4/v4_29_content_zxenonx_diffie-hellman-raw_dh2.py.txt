import random
import hashlib

# Define the prime number (p) and generator (g)
prime = 1001
generator = 9

# Generate random private keys for Alice and Bob
private_key_alice = random.randint(5, 10)
private_key_bob = random.randint(10, 20)

# Calculate public keys for Alice and Bob
public_key_alice = pow(generator, private_key_alice, prime)
public_key_bob = pow(generator, private_key_bob, prime)

# Print the initial values
print('g:', generator, '(a shared value), p:', prime, '(a prime number)')
print('\nAlice calculates:')
print('a (Alice random):', private_key_alice)
print('Alice value (A):', public_key_alice, '(g^a) mod p')
print('\nBob calculates:')
print('b (Bob random):', private_key_bob)
print('Bob value (B):', public_key_bob, '(g^b) mod p')

# Calculate the shared keys using Diffie-Hellman key exchange
shared_key_alice = pow(public_key_bob, private_key_alice, prime)
shared_key_bob = pow(public_key_alice, private_key_bob, prime)

# Hash the shared keys using SHA-256 for added security
hashed_shared_key_alice = hashlib.sha256(str(shared_key_alice).encode()).hexdigest()
hashed_shared_key_bob = hashlib.sha256(str(shared_key_bob).encode()).hexdigest()

# Print the shared keys (hashed) for both Alice and Bob
print('\nAlice calculates the shared key (hashed):', hashed_shared_key_alice)
print('Bob calculates the shared key (hashed):', hashed_shared_key_bob)
