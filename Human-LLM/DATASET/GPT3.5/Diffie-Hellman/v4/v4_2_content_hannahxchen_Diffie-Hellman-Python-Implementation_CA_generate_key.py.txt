from Crypto import Random
from Crypto.PublicKey import RSA

# Generate random number generator instances
rng1 = Random.new().read
rng2 = Random.new().read

# Generate RSA key pairs for Alice and Bob
RSAkey1 = RSA.generate(1024, rng1)
privatekey1 = RSAkey1
publickey1 = RSAkey1.publickey()

RSAkey2 = RSA.generate(1024, rng2)
privatekey2 = RSAkey2
publickey2 = RSAkey2.publickey()

# Write the private and public keys to text files
with open("privatekey1.txt", "w") as file1:
    file1.write(privatekey1.exportKey().decode())

with open("publickey1.txt", "w") as file2:
    file2.write(publickey1.exportKey().decode())

with open("privatekey2.txt", "w") as file3:
    file3.write(privatekey2.exportKey().decode())

with open("publickey2.txt", "w") as file4:
    file4.write(publickey2.exportKey().decode())

print("Finished generating key pairs.")
