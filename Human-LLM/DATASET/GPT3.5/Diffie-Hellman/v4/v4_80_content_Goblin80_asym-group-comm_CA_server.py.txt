from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives.asymmetric import rsa, ec, padding
from cryptography.hazmat.primitives import serialization, hashes
from flask import Flask, jsonify
import requests
from lib import rsa as RSA
from lib import ecc as ECC
from lib.dh import DH_Wrapper
from lib.fernet import FRNET_Wrapper
from lib.user import User

app = Flask(__name__)

def generate_rsa_pair(dh_key):
    """Generate a pair of RSA keys and encrypt the private key."""
    fernet_wrapper = FRNET_Wrapper(dh_key)
    rsa_private_key = RSA.Private()
    ca_private_key = ECC.Private(ECC.load_CA_private())
    rsa_public_key = RSA.Public(rsa_private_key.deduce_public())
    rsa_private_encrypted = fernet_wrapper.encrypt(rsa_private_key.export())
    return rsa_private_encrypted, rsa_public_key, ca_private_key.sign(rsa_private_key.export())

user_list = []

@app.route("/rsa/request/<int:peer_y>/<int:modulus>/<name>/<int:port>", methods=['GET'])
def generate_rsa_private_encrypted(peer_y, modulus, name, port):
    """Generate an encrypted RSA private key and update the user list."""
    dh_wrapper = DH_Wrapper(modulus)
    shared_secret = dh_wrapper.calc_shared_key(peer_y)
    rsa_key_encrypted, rsa_public_key, signature = generate_rsa_pair(shared_secret)
    user_list.append({
        'name': name,
        'host': 'localhost',
        'port': port,
        'public': rsa_public_key.export().decode(),
        'signature': RSA.encode(signature)
    })
    notify_users()
    return jsonify({'peer_y': dh_wrapper.y, 'rsa': RSA.encode(rsa_key_encrypted), 'signature': RSA.encode(signature)})

def notify_users():
    """Notify users about the updated user list."""
    for user in user_list:
        requests.post(
            f"http://{user['host']}:{user['port']}/update/users", json={'users': user_list})

@app.route("/registry", methods=['GET'])
def view_registry():
    """View the user registry."""
    return jsonify({'users': user_list})

if __name__ == "__main__":
    app.run(debug=True)
