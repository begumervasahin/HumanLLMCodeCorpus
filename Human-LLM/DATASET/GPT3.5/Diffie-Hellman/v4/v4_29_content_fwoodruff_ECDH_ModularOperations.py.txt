# Function to calculate the Greatest Common Divisor (GCD) of two numbers
def egcd(a, b):
    '''
    Finds the Greatest Common Divisor (GCD) of numbers a and b
    '''
    if a == 0:
        return (b, 0, 1)
    else:
        gcd, y, x = egcd(b % a, a)
        return (gcd, x - (b // a) * y, y)

# Function to find the modular inverse of 'a' modulo 'm'
def modinv(a, m):
    '''
    Calculates the inverse of 'a' modulo 'm'
    '''
    gcd, x, y = egcd(a, m)
    if gcd != 1:
        return m
    else:
        return x % m

# Function to find the modular square root of 'a' modulo 'p'
def modular_sqrt(a, p):
    if a == p:
        return a

    # Function to calculate the Legendre symbol
    def legendre_symbol(a, p):
        ls = pow(a, (p - 1) // 2, p)
        return -1 if ls == p - 1 else ls

    # Checking for special cases
    if legendre_symbol(a, p) != 1:
        return 0
    elif a == 0:
        return 0
    elif p == 2:
        return p
    elif p % 4 == 3:
        return pow(a, (p + 1) // 4, p)

    # Calculating the modular square root using Tonelli-Shanks algorithm
    s = p - 1
    e = 0
    while s % 2 == 0:
        s //= 2
        e += 1

    n = 2
    while legendre_symbol(n, p) != -1:
        n += 1

    x = pow(a, (s + 1) // 2, p)
    b = pow(a, s, p)
    g = pow(n, s, p)
    r = e

    while True:
        t = b
        m = 0
        for m in range(r):
            if t == 1:
                break
            t = pow(t, 2, p)

        if m == 0:
            return x

        gs = pow(g, 2 ** (r - m - 1), p)
        g = (gs * gs) % p
        x = (x * gs) % p
        b = (b * g) % p
        r = m
