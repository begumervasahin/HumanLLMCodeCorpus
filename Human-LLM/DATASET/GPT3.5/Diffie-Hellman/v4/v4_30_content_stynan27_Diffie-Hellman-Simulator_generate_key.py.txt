import secrets
import random

def generate_large_prime():
    """
    Generates a large prime number suitable for use in public key cryptography.
    """
    lower_bound = 10 ** 1229
    
    # Generate a random number with 4096 bits
    prime_candidate = secrets.randbits(4096)
    
    # Ensure the generated number is greater than the lower bound and passes the primality test
    while prime_candidate < lower_bound or not is_prime(prime_candidate):
        prime_candidate = secrets.randbits(4096)
    
    return prime_candidate

def is_prime(n, k=8):
    """
    Miller-Rabin primality test.
    Returns True if n is very likely prime, False otherwise.
    """
    # Base cases
    if n < 2:
        return False
    if n in {2, 3}:
        return True
    if n % 2 == 0:
        return False

    # Write n - 1 as 2^r * d
    d, r = n - 1, 0
    while d % 2 == 0:
        d //= 2
        r += 1

    # Witness loop
    for _ in range(k):
        a = random.randint(2, n - 2)
        x = pow(a, d, n)
        if x == 1 or x == n - 1:
            continue
        for _ in range(r - 1):
            x = pow(x, 2, n)
            if x == n - 1:
                break
        else:
            return False
    return True
