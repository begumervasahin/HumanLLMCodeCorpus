from math import floor
from temporary_client import Client

def split_length(list_length):
    """
    Splits the length of a list into two halves.
    """
    return floor(list_length / 2)

def secret_comparison(clients):
    """
    Compares the shared secrets of clients.
    """
    prev = clients[0]
    for client in clients:
        if client.shared_secret != prev.shared_secret:
            return False
        prev = client
    return True

class KeyGen:
    """
    KeyGen class to generate new keys upon entry of a new client to the server.
    """
    def __init__(self, large_prime):
        """
        Initializes the KeyGen instance with a large prime number.
        """
        self.p = large_prime

    def generate_keys(self, current_list, list_length, generator):
        """
        Generates keys for clients recursively.
        """
        if list_length == 1:
            # If only one client, calculate its key
            current_list[0].calculate_key(generator, self.p)
            return
        else:
            # Split the list of clients into two halves
            left_length = split_length(list_length)
            right_length = list_length - left_length
            left = current_list[:left_length]
            right = current_list[left_length:]
            left_key = generator
            right_key = generator
            # Calculate keys for left and right halves
            for client in left:
                left_key = client.calculate_key(left_key, self.p)
            for client in right:
                right_key = client.calculate_key(right_key, self.p)
            # Recursively generate keys for subgroups
            self.generate_keys(left, left_length, right_key)
            self.generate_keys(right, right_length, left_key)

if __name__ == '__main__':
    """
    For testing purposes
    """
    # Create a list to store connected clients
    connected_clients = []
    for i in range(2):
        connected_clients.append(Client())
    
    # Create a KeyGen instance with a large prime number
    k = KeyGen(71)
    
    # Generate keys for the connected clients
    k.generate_keys(connected_clients, len(connected_clients), 3)
    
    # Check if all clients have the same shared secret
    if secret_comparison(connected_clients):
        print("The shared secret {}, was correctly calculated for all clients!".format(connected_clients[0].shared_secret))
    else:
        print("The shared secret wasn't correctly calculated")
