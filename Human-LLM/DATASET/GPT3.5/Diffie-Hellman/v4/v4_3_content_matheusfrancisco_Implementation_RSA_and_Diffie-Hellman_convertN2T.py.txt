import binascii

def str_to_num(input_str):
    """Converts a string to a list of ASCII values."""
    return [num for num in input_str.encode()]

def num_to_str(msg):
    """Converts a list of integers to a string."""
    return bytes([min(num, 127) for num in msg]).decode()

def num_to_blocks(input_list, block_size):
    """Converts a list of integers to blocks of a specified size."""
    split_list = [input_list[i:i+block_size] for i in range(0, len(input_list), block_size)]
    if len(split_list[-1]) < block_size:
        split_list[-1] += [0] * (block_size - len(split_list[-1]))
    return [int(binascii.hexlify(bytes(block)), 16) for block in split_list]

def blocks_to_num(blocks, block_size):
    """Converts blocks to a list of integers."""
    num_list = []
    for block in blocks:
        hex_num = hex(block)[2:]
        if len(hex_num) % 2 != 0:
            hex_num = '0' + hex_num
        num_list += [num for num in bytes.fromhex(hex_num)]
    return num_list
