import unittest
import diffie_hellman

class TestDiffieHellman(unittest.TestCase):
    """
    Unit tests for the Diffie-Hellman key exchange algorithm.
    Refer to https://en.wikipedia.org/wiki/Diffie%E2%80%93Hellman_key_exchange for more information.
    """

    def test_encrypt_integer(self):
        """Test encryption of integers using the Diffie-Hellman algorithm."""
        self.assertEqual(diffie_hellman.encrypt_integer(5, 23, 3), 10)
        self.assertEqual(diffie_hellman.encrypt_integer(5, 23, 4), 4)

    def test_shared_secret_is_equal(self):
        """Test generation of shared secret keys using Diffie-Hellman."""
        secret_key_1 = diffie_hellman.encrypt_integer(5, 23, 4)
        secret_key_2 = diffie_hellman.encrypt_integer(5, 23, 3)

        self.assertEqual(
            diffie_hellman.encrypt_integer(secret_key_1, 23, 3),
            diffie_hellman.encrypt_integer(secret_key_2, 23, 4),
            "Shared secrets should be equal"
        )

        secret_key_3 = diffie_hellman.encrypt_integer(5, 23, 14)
        secret_key_4 = diffie_hellman.encrypt_integer(5, 23, 17)

        self.assertEqual(
            diffie_hellman.encrypt_integer(secret_key_3, 23, 17),
            diffie_hellman.encrypt_integer(secret_key_4, 23, 14),
            "Shared secrets should be equal"
        )

    def test_brute_force_secret_is_equal(self):
        """Test brute force checking of shared secret keys."""
        g = 5
        p = 23
        for i in range(p):
            for j in range(p):
                shared1 = diffie_hellman.encrypt_integer(
                    diffie_hellman.encrypt_integer(g, p, i), p, j
                )
                shared2 = diffie_hellman.encrypt_integer(
                    diffie_hellman.encrypt_integer(g, p, j), p, i
                )
                self.assertEqual(shared1, shared2)

if __name__ == '__main__':
    unittest.main()
