from tkinter import *
import tkinter.messagebox

# Create the main window
root = Tk()
root.title('Math 243 Project')
root.geometry('275x210')

# Create labels and entry fields for user input
label_prime1 = Label(root, text="What is your first prime number?")
entry_prime1 = Entry(root, bd=1)

label_prime2 = Label(root, text="What is your second prime number?")
entry_prime2 = Entry(root, bd=1)

label_secret1 = Label(root, text="What is your first secret number?")
entry_secret1 = Entry(root, bd=1)

label_secret2 = Label(root, text="What is your second secret number?")
entry_secret2 = Entry(root, bd=1)

# Function to check if a number is prime
def is_prime(n):
    if n < 2:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

# Function to handle button click event
def calculate_secret_key():
    try:
        p1 = int(entry_prime1.get())
        p2 = int(entry_prime2.get())
        a1 = int(entry_secret1.get())
        b1 = int(entry_secret2.get())
    except ValueError:
        tkinter.messagebox.showinfo('Error', 'Please enter integers for prime numbers and secret numbers')
        return

    if not is_prime(p1):
        tkinter.messagebox.showinfo('Error', 'First input is not a prime number')
    elif not is_prime(p2):
        tkinter.messagebox.showinfo('Error', 'Second input is not a prime number')
    else:
        a2 = (p1 ** a1) % p2
        b2 = (p1 ** b1) % p2
        secret_key = (b2 ** a1) % p2
        secret_key_check = (a2 ** b1) % p2
        if secret_key != secret_key_check:
            tkinter.messagebox.showinfo('Error', 'Unknown Error')
        else:
            tkinter.messagebox.showinfo('Key', 'Your Secret Key is: ' + str(secret_key))

# Create the submit button
submit_button = Button(root, text="Submit", command=calculate_secret_key)

# Place the widgets in the window using the pack geometry manager
label_prime1.pack()
entry_prime1.pack()

label_prime2.pack()
entry_prime2.pack()

label_secret1.pack()
entry_secret1.pack()

label_secret2.pack()
entry_secret2.pack()

submit_button.pack(side=BOTTOM)

# Start the main event loop
root.mainloop()
