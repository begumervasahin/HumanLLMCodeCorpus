import random
import time

class Device:
    # Initialize Device with random values for n and g
    n = random.randint(1000, 5000)
    g = 15319

    def __init__(self):
        # Generate a random private key and calculate the public key
        self.private_key = random.randint(1, Device.n)
        self.public_key = (Device.g ** self.private_key) % Device.n
        self.key = None

    @staticmethod
    def key_exchange(d1, d2):
        # Perform key exchange using Diffie-Hellman algorithm
        d1.key = (d2.public_key ** d1.private_key) % Device.n
        d2.key = (d1.public_key ** d2.private_key) % Device.n

    def encrypt(self, msg):
        # Encrypt the message using the shared key
        encrypted = [ord(letter) + self.key for letter in msg]
        return encrypted

    def decrypt(self, encrypted):
        # Decrypt the message using the shared key
        decrypted = ''.join([chr(byte - self.key) for byte in encrypted])
        return decrypted

def main():
    # Initialize two devices
    device1 = Device()
    device2 = Device()

    print('\nPerforming key exchange...\n')
    Device.key_exchange(device1, device2)

    message = 'Hello, how are you?'
    print(f'[Device1] encrypting message: {message}')
    encrypted_msg = device1.encrypt(message)
    print('[Device1] sending message...\n')
    time.sleep(0.5)

    decrypted_msg = device2.decrypt(encrypted_msg)
    print('[Device2] received message')
    print(f'[Device2] decrypted message as: {decrypted_msg}\n')

if __name__ == "__main__":
    main()
