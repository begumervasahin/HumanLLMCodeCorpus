import random
import sys

def miller_rabin_pass(a, s, d, n):
    """
    Check if n passes the Miller-Rabin test for a given base a.
    """
    result = pow(a, d, n)
    if result == 1 or result == n - 1:
        return True
    for _ in range(s - 1):
        result = pow(result, 2, n)
        if result == n - 1:
            return True
        elif result == 1:
            return False
    return False

def miller_rabin(n):
    """
    Apply the Miller-Rabin Test to determine if n is probably prime.
    """
    if n == 2 or n == 3:
        return True
    if n == 1 or n % 2 == 0:
        return False
    
    d = n - 1
    s = 0
    while d % 2 == 0:
        d >>= 1
        s += 1
    
    for _ in range(20):  # Number of iterations
        a = random.randint(2, n - 2)
        if not miller_rabin_pass(a, s, d, n):
            return False
    return True

def generate_prime(nbits):
    """
    Generate a prime number with nbits using the Miller-Rabin test.
    """
    while True:
        p = random.getrandbits(nbits)
        p |= 2 ** (nbits - 1) | 1  # Ensure the number has the correct bit length and is odd
        if miller_rabin(p):
            return p

def generate_prime_range(start, stop):
    """
    Generate a prime number within the given range using the Miller-Rabin test.
    """
    while True:
        p = random.randrange(start, stop, 2)  # Ensure odd number generation
        if miller_rabin(p):
            return p

if __name__ == "__main__":
    if len(sys.argv) < 3:
        print("Usage: python script.py [test/genprime] [number]")
        sys.exit(1)
    
    command = sys.argv[1]
    number = int(sys.argv[2])
    
    if command == "test":
        print("PRIME" if miller_rabin(number) else "COMPOSITE")
    elif command == "genprime":
        print(generate_prime(number))
