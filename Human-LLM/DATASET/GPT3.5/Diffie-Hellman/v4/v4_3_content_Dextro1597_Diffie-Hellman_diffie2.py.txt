def calculate_shared_secret(shared_prime, shared_base, alice_secret, bob_secret):
    """Calculate the shared secret key exchange between Alice and Bob."""
    print('Shared Variables:')
    print('Shared Prime (p):', shared_prime)
    print('Shared Base (q):', shared_base)

    # Calculate Alice's public key (A) and send it over the insecure channel
    A = (shared_base ** alice_secret) % shared_prime
    print('Alice sends value over insecure channel:', A)

    # Calculate Bob's public key (B) and send it over the insecure channel
    B = (shared_base ** bob_secret) % shared_prime
    print('Bob sends value over insecure channel:', B)

    # Calculate the final shared secret key for Alice
    A_final = (B ** alice_secret) % shared_prime
    print('Value received by Alice:', A_final)

    # Calculate the final shared secret key for Bob
    B_final = (A ** bob_secret) % shared_prime
    print('Value received by Bob:', B_final)

if __name__ == "__main__":
    # Get input values from the user
    shared_prime = int(input('Enter the shared prime (p): '))
    shared_base = int(input('Enter the shared base (q): '))
    alice_secret = int(input('Enter Alice\'s secret key (a): '))
    bob_secret = int(input('Enter Bob\'s secret key (b): '))

    # Calculate and display the shared secret exchange
    calculate_shared_secret(shared_prime, shared_base, alice_secret, bob_secret)
