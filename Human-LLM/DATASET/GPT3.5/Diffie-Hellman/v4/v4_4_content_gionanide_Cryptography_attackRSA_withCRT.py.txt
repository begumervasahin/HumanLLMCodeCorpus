def chinese_remainder(n, a):
    sum_ = 0
    prod = math.prod(n)
    for n_i, a_i in zip(n, a):
        p = prod // n_i
        sum_ += a_i * mul_inv(p, n_i) * p
    return sum_ % prod

def mul_inv(a, b):
    b0 = b
    x0, x1 = 0, 1
    if b == 1:
        return 1
    while a > 1:
        try:
            q = a // b
            a, b = b, a % b
            x0, x1 = x1 - q * x0, x0
        except:
            print("Bad N values (check no common factors in N vals)")
            return 0
    if x1 < 0:
        x1 += b0
    return x1

def GCD(a, b):
    a = abs(a)
    b = abs(b)
    while a:
        a, b = b % a, a
    return b

def GCD_List(lst):
    return math.prod([abs(x) for x in lst])
