def rational_to_contfrac(x, y):
    """
    Convert a rational fraction x/y into a continued fraction [a0, ..., an].
    """
    partial_quotients = [x // y]
    while x * partial_quotients[-1] != y:
        x, y = y, x - y * partial_quotients[-1]
        partial_quotients.append(x // y)
    return partial_quotients

def convergents_from_contfrac(frac):
    """
    Compute the list of convergents from the continued fraction.
    """
    convergents = []
    for i in range(len(frac)):
        convergents.append(contfrac_to_rational(frac[0:i + 1]))
    return convergents

def contfrac_to_rational(frac):
    """
    Convert a finite continued fraction [a0, ..., an] to a rational number x/y.
    """
    if len(frac) == 0:
        return (0, 1)
    num, denom = frac[-1], 1
    for i in range(len(frac) - 2, -1, -1):
        num, denom = frac[i] * num + denom, num
    return (num, denom)

def test_continued_fraction_functions():
    """
    Test the functionality of the continued fraction manipulation functions.
    """
    test_nums = [(1, 1), (1, 2), (5, 15), (27, 73), (73, 27)]
    for num, denom in test_nums:
        print('Rational number:')
        print((num, denom))
        cont_frac = rational_to_contfrac(num, denom)
        print('Continued fraction:')
        print(cont_frac)
        print('Convergents:')
        print(convergents_from_contfrac(cont_frac))
        print('***********************************')

if __name__ == "__main__":
    test_continued_fraction_functions()
