import base64
from cryptography.hazmat.primitives import serialization
from cryptography.hazmat.primitives.asymmetric.x25519 import X25519PrivateKey

def private_key_to_bytes(private_key):
    """Converts a private key object to bytes."""
    return private_key.private_bytes(
        encoding=serialization.Encoding.Raw,
        format=serialization.PrivateFormat.Raw,
        encryption_algorithm=serialization.NoEncryption()
    )

def public_key_to_bytes(private_key):
    """Converts a private key object to its corresponding public key bytes."""
    return private_key.public_key().public_bytes(
        encoding=serialization.Encoding.Raw,
        format=serialization.PublicFormat.Raw
    )

def perform_key_exchange(private_key, public_key):
    """Performs key exchange using the private and public keys."""
    private_key_object = X25519PrivateKey.from_private_bytes(private_key)
    return private_key_object.exchange(public_key)

def print_key_pair(private_key):
    """Prints the base64 encoded private and public keys."""
    private_key_bytes = private_key_to_bytes(private_key)
    public_key_bytes = public_key_to_bytes(private_key)
    
    private_key_encoded = base64.b64encode(private_key_bytes)
    public_key_encoded = base64.b64encode(public_key_bytes)
    
    print('Private Key:', private_key_encoded.decode())
    print('Public Key:', public_key_encoded.decode())

# Example usage:
private_key = X25519PrivateKey.generate()
print_key_pair(private_key)
