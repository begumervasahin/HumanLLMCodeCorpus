import random

def generate_prime():
    """
    Generate a prime number between 2 and 99.
    
    Returns:
        int: A prime number.
    """
    prime = 0
    while prime == 0:
        num = random.randint(2, 99)
        for i in range(2, (num // 2)):
            if (num % i) == 0:
                 prime = 0
                 break
        else:
            prime = num
            return prime

def gcd(a, b):
    """
    Calculate the greatest common divisor (GCD) of two numbers.
    
    Args:
        a (int): First number.
        b (int): Second number.
    
    Returns:
        int: The GCD of the two numbers.
    """
    while b != 0:
        a, b = b, a % b
    return a

def find_primitive_roots(modulo):
    """
    Find primitive roots modulo a given number.
    
    Args:
        modulo (int): The modulus value.
    
    Returns:
        list: List of primitive roots.
    """
    roots = []
    required_set = set(num for num in range(1, modulo) if gcd(num, modulo) == 1)
    for g in range(1, modulo):
        actual_set = set(pow(g, powers) % modulo for powers in range(1, modulo))
        if required_set == actual_set:
            roots.append(g)
            if len(roots) > 3:
                break
    return roots

def find_prime_and_primitive_root():
    """
    Find a prime number and its corresponding primitive root.
    
    Returns:
        tuple: A tuple containing the prime number and primitive root.
    """
    while True:
        q = generate_prime()
        primitive_roots = find_primitive_roots(q)
        if len(primitive_roots) == 0:
            continue
        else:
            break
    return (q, random.choice(list(primitive_roots)))

def calculate_X_and_Y(q, alpha):
    """
    Calculate X and Y values for Diffie-Hellman key exchange.
    
    Args:
        q (int): Prime modulus.
        alpha (int): Primitive root modulo q.
    
    Returns:
        tuple: Tuple containing X and Y values.
    """
    x = random.randint(1, q)
    y = pow(alpha, x, q)
    return x, y

def calculate_shared_key(y, x, q):
    """
    Calculate the shared key for Diffie-Hellman key exchange.
    
    Args:
        y (int): Partner's public key.
        x (int): Private key.
        q (int): Prime modulus.
    
    Returns:
        int: Shared key.
    """
    return pow(y, x, q)
