import math
from collections import defaultdict
from argparse import ArgumentParser
from gmpy2 import powmod, divm

def meet_in_the_middle(h, g, p, R):
    """
    Perform a meet-in-the-middle attack to find private keys and shared secret key.
    
    Args:
        h (int): Public key of the target user.
        g (int): Base value.
        p (int): Prime modulus value.
        R (int): Range value for the attack.
    
    Returns:
        int: Private key of the target user, or -1 if not found.
    """
    mappings = defaultdict(lambda: -1)
    
    # Precompute values for the first half of the attack
    for i in range(0, R):
        x0 = powmod(g, i * R, p)
        mappings[x0] = i
    
    # Perform the second half of the attack
    for i in range(0, R):
        x1 = divm(h, powmod(g, i, p), p)
        if mappings[x1] != -1:
            return i + mappings[x1] * R
    
    return -1

def main():
    parser = ArgumentParser(description="Meet-in-the-Middle Attack")
    parser.add_argument("-A", "--alice", required=True, help="Alice's public key")
    parser.add_argument("-B", "--bob", required=True, help="Bob's public key")
    parser.add_argument("-g", "--base", required=True, help="Base value")
    parser.add_argument("-p", "--prime", required=True, help="Prime modulus value")
    parser.add_argument("-r", "--range", required=True, help="Range value")
    args = parser.parse_args()
    
    A = int(args.alice)
    B = int(args.bob)
    g = int(args.base)
    p = int(args.prime)
    R = int(2 ** (int(args.range) / 2))
    
    print("Calculating Alice's private key..")
    a = meet_in_the_middle(A, g, p, R)
    if a != -1:
        print("Alice's private key:", a)
    else:
        print("Could not find Alice's private key :(")
    
    print("Calculating Bob's private key..")
    b = meet_in_the_middle(B, g, p, R)
    if b != -1:
        print("Bob's private key:", b)
    else:
        print("Could not find Bob's private key :(")
    
    if a != -1 and b != -1:
        K = powmod(g, a * b, p)
        print("Shared secret key:", K)
    else:
        print("Could not find the shared key :(")

if __name__ == "__main__":
    main()
