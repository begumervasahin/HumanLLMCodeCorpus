from euclidean import euclidean_extended_gcd, gcd
from numberPrime import GenParms
from modular_arithmetic import modExp

class RSA:
    def __init__(self):
        self.n, self.d, self.e = self.generate_rsa_parameters(n_bits=100, debug=True)

    def encryption(self, msg):
        """Encrypts the message using RSA."""
        encrypted_msg = [modExp(char, self.e, self.n) for char in msg]
        print('Encrypted Message:', encrypted_msg)
        return encrypted_msg

    def decryption(self, msg):
        """Decrypts the message using RSA."""
        decrypted_msg = [modExp(char, self.d, self.n) for char in msg]
        print('Decrypted Message:', decrypted_msg)
        return decrypted_msg

    def public_key(self):
        """Returns the public key (n, e)."""
        return self.n, self.e

    def private_key(self):
        """Returns the private key (n, d)."""
        return self.n, self.d

    def generate_rsa_parameters(self, n_bits, debug=False):
        """Generates RSA parameters n, d, and e."""
        n, d, e = GenParms(n_bits=n_bits, DEBUG=debug)
        return n, d, e

    print("RSA Encryption and Decryption Test:")
