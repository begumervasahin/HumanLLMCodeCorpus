import binascii

def xor(m, k):
    r = []
    for i, j in zip(m, k):
        if i.isalpha() and j.isalpha():
            r.append(i)
        else:
            r.append(str(int(i) ^ int(j)))
    return "".join(r)

def main():
    plainText = 'informationsecurity'
    key1 = 'vlaksjdhfgqodzmxncb'
    
    print('The plain text is:', plainText)
    print('The first key is:', key1, '\n')
    
    plainTextBits = bin(int(binascii.hexlify(plainText.encode('utf-8')), 16))
    keyBits = bin(int(binascii.hexlify(key1.encode('utf-8')), 16))
    
    print('Plain text in bits:', plainTextBits, '\n')
    print('Key in bits:', keyBits, '\n')
    
    result = xor(plainTextBits, keyBits)
    n = int(result, 2)
    encrypted_text = binascii.unhexlify('%x' % n)
    
    print('XOR result for the encryption with the normal key:', result, '\n')
    print('Encrypted text:', encrypted_text)
    
    decryption = xor(result, keyBits)
    l = int(decryption, 2)
    
    print('\nDecryption result:', decryption, '\n')
    print('Decrypted text with the normal key:', binascii.unhexlify('%x' % l), '\n')
    
    print('Decryption with the alternative key \n')
    alternativeKey = 'tlftrffwmixor|{xbch'
    alternativeKeyBits = bin(int(binascii.hexlify(alternativeKey.encode('utf-8')), 16))
    
    print('Alternative key in bits:', alternativeKeyBits, '\n')
    
    result1 = xor(result, alternativeKeyBits)
    k = int(result1, 2)
    
    print('Decrypted text with the alternative key:', binascii.unhexlify('%x' % k))

if __name__ == "__main__":
    main()
