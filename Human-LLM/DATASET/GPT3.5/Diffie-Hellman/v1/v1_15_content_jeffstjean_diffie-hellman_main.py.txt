import random
import time

class Device:
    n = random.randint(1000, 5000)
    g = 15319

    def __init__(self):
        self.private = random.randint(1, Device.n)
        self.public = pow(Device.g, self.private, Device.n)
        self.key = None

    @staticmethod
    def keygen(d1, d2):
        d1.key = pow(d2.public, d1.private, Device.n)
        d2.key = pow(d1.public, d2.private, Device.n)

    def encrypt(self, msg):
        encrypted = [ord(letter) + self.key for letter in msg]
        return encrypted

    def decrypt(self, encrypted):
        unencrypted = ''.join([chr(byte - self.key) for byte in encrypted])
        return unencrypted

def main():
    msg = 'Hello, how are you?'
    d1 = Device()
    d2 = Device()

    print('\nPerforming key exchange...\n')
    Device.keygen(d1, d2)

    encrypted = d1.encrypt(msg)
    print('[Device1] encrypting message:', msg)
    print('[Device1] sending message...\n')
    time.sleep(0.5)

    decrypted = d2.decrypt(encrypted)
    print('[Device2] received message')
    print('[Device2] decrypted message as:', decrypted, '\n')

if __name__ == "__main__":
    main()
