def calculate_shared_secret(shared_prime, shared_base, alice_secret, bob_secret):
    """
    Calculates the shared secret key exchange between Alice and Bob.
    """
    print('Shared Variables:')
    print('Shared Prime no.:', shared_prime)
    print('Shared Base no.:', shared_base)

    A = (shared_base ** alice_secret) % shared_prime
    print('Alice sends value over insecure channel:', A)

    B = (shared_base ** bob_secret) % shared_prime
    print('Bob sends value over insecure channel:', B)

    A1 = (B ** alice_secret) % shared_prime
    print('Value got by Alice:', A1)

    B1 = (A ** bob_secret) % shared_prime
    print('Value got by Bob:', B1)

if __name__ == "__main__":
    shared_prime = int(input('Enter the value of p: '))
    shared_base = int(input('Enter the value of q: '))
    alice_secret = int(input('Enter Alice\'s secret key value (a): '))
    bob_secret = int(input('Enter Bob\'s secret key value (b): '))

    calculate_shared_secret(shared_prime, shared_base, alice_secret, bob_secret)
