import binascii

def str_to_num(input_str):
    """Converts a string to a list of ASCII values."""
    return [ord(char) for char in input_str]

def num_to_str(input_num):
    """Converts a list of ASCII values to a string."""
    return ''.join([chr(num) for num in input_num])

def blocks_to_num(blocks, block_size):
    """Converts blocks to a list of integers."""
    num_list = []
    for block in blocks:
        hex_num = hex(block)[2:]
        if len(hex_num) % 2 != 0:
            hex_num = '0' + hex_num
        num_list += [num for num in bytes.fromhex(hex_num)]
