import base64
from cryptography.hazmat.primitives import serialization
from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.asymmetric.x25519 import X25519PrivateKey
from cryptography.hazmat.primitives.kdf.hkdf import HKDF

# Utility functions
def public_bytes(key):
    return key.public_key().public_bytes(
        encoding=serialization.Encoding.Raw,
        format=serialization.PublicFormat.Raw
    )

def exchange_key(private_key, peer_public_key):
    return private_key.exchange(peer_public_key)

def print_key(key):
    print(key)

# Generate private keys
secret_alice = X25519PrivateKey.generate()
secret_bob = X25519PrivateKey.generate()
secret_joe = X25519PrivateKey.generate()
ephemeral_key = X25519PrivateKey.generate()

# Generate shared keys
shared_alice = secret_alice.exchange(ephemeral_key.public_key())
shared_bob = secret_bob.exchange(ephemeral_key.public_key())
shared_joe = secret_joe.exchange(ephemeral_key.public_key())

# Perform key exchange
shared_alice_joe = shared_alice.exchange(shared_joe.public_key())
shared_alice_bob = shared_alice.exchange(shared_bob.public_key())
shared_bob_joe = shared_joe.exchange(shared_bob.public_key())

# Perform ephemeral key exchange
ephemeral_alice_joe = exchange_key(shared_alice_joe, ephemeral_key.public_key())
ephemeral_alice_bob = exchange_key(shared_alice_bob, ephemeral_key.public_key())
ephemeral_bob_joe = exchange_key(shared_bob_joe, ephemeral_key.public_key())

# Main function to derive encryption key
def main():
    backend = default_backend()
    hkdf = HKDF(
        algorithm=hashes.BLAKE2s(32),
        length=32,
        salt=public_bytes(ephemeral_key),
        info=b'MofN-Encryption-demo',
        backend=backend
    )
    key = hkdf.derive(ephemeral_alice_joe + b'\x01' +
                      ephemeral_alice_bob + b'\x02' + ephemeral_bob_joe)
    print(base64.b64encode(key))

if __name__ == "__main__":
    main()
