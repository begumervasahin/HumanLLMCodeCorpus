import math
import random
import getpass

def prime_generator():
    primes = []
    count = 3
    while len(primes) < 1000:
        is_prime = True
        for x in range(2, int(math.sqrt(count) + 1)):
            if count % x == 0:
                is_prime = False
                break
        if is_prime:
            primes.append(count)
        count += 1
    return primes

def is_integer(s):
    try:
        return int(s)
    except ValueError:
        return is_integer(getpass.getpass("Not an integer, Try again: "))

if __name__ == "__main__":
    primes = prime_generator()
    g = random.choice(primes)
    primes.remove(g)
    p = random.choice(primes)
    print("Shared g value:", g)
    print("Shared p value:", p)
    a = is_integer(getpass.getpass("Hidden \"a\" value: "))
    b = is_integer(getpass.getpass("Hidden \"b\" value: "))
    A = (g ** a) % p
    B = (g ** b) % p

    print("\nCalculation of shared key with \033[0;31m\"a\"\033[0m:\n")
    print("User assigned value of \033[0;31m\"a\"\033[0m is \033[0;31m", a, "\033[0m")
    print("Formula: A = g^a mod p")
    print(f"{g}^{a} mod {p} =", A, "= A")
    print("Received \"B\" value")
    print("shared_key_a = B^a mod p")
    shared_key_a = (B ** a) % p
    print(f"{B}^{a} mod {p} =", shared_key_a)
    print("Shared Key Calculated from \"a\" is:", shared_key_a)

    print("\nCalculation of shared key with \033[0;34m\"b\"\033[0m:\n")
    print("User assigned value of \033[0;34m\"b\"\033[0m is \033[0;34m", b, "\033[0m")
    print("Formula: B = g^b mod p")
    print(f"{g}^{b} mod {p} =", B, "= B")
    print("Received \"A\" value")
    print("shared_key_b = A^b mod p")
    shared_key_b = (A ** b) % p
    print(f"{A}^{b} mod {p} =", shared_key_b)
    print("Shared Key Calculated from \"b\" is:", shared_key_b)
