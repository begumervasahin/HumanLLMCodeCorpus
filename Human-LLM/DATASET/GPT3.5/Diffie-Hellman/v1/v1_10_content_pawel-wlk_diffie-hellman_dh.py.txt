import math

def is_prime(n):
    for i in range(2, math.ceil(math.sqrt(n)) + 1):
        if n % i == 0:
            return False
    return True

def fast_modulo_power(n, power, mod):
    result = 1
    binary_power = bin(power)[2:]
    for bit in binary_power:
        result = (result * result) % mod
        if bit == '1':
            result = (result * n) % mod
    return result

def factorize(n):
    result = []
    k = 2
    while k * k <= n:
        if n % k == 0:
            result.append(k)
            n //= k
        else:
            k += 1
    if n > 1:
        result.append(n)
    return result

def find_generator(n):
    for m in range(2, n):
        for i in factorize(n-1):
            if fast_modulo_power(m, i, n) == 1:
                break
        else:
            return m
    return None

number = 1000003

if is_prime(number):
    print("{} is prime\n".format(number))
    generator = find_generator(number)
    if generator is not None:
        print("Smallest generator is: {}\n".format(generator))
        print("Diffie-Hellman protocol:\n")
        try:
            a = int(input("Pick private number for person A: "))
            b = int(input("Pick private number for person B: "))
            A = fast_modulo_power(generator, a, number)
            B = fast_modulo_power(generator, b, number)
            A_s = fast_modulo_power(B, a, number)
            B_s = fast_modulo_power(A, b, number)
            print("")
            print("Private key received by A is {}".format(A_s))
            print("Private key received by B is {}".format(B_s))
        except ValueError:
            print("Wrong input: please enter integers for private numbers.")
    else:
        print("No generator found for {}".format(number))
else:
    print("{} is not prime".format(number))
