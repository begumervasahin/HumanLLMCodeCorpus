def initialize_PuK():
    a = int(input('Give me the a factor: '))
    b = int(input('Give me the b factor: '))
    p = int(input('Give me the p prime: '))
    x, y = map(int, input('Give me the initial point with this format xy: '))
    print('\n')
    print('The curve\'s equation as a PUBLIC KEY: (y**2 = x**3 + {}x + {}) mod {}'.format(a, b, p))
    print('And the initial point: ({},{}) \n'.format(x, y))
    return a, b, p, x, y

def initialize_PrK():
    m = int(input('Give the multiplier: '))
    print('The multiplier as a PRIVATE KEY: {}\n'.format(m))
    return m

def multiplicative_inverse(n):
    x = 1
    while not (n * x % 17 == 1):
        x += 1
    return x

def calculate_P(x1, y1, x2, y2, a, p, flag, i):
    if x1 == x2:
        if y1 == y2:
            if flag:
                divide = multiplicative_inverse(2 * y1)
                s = (((3 * x1 ** 2 + a) % p) * divide) % p
                flag = False
        else:
            s = 0
    else:
        divide1 = multiplicative_inverse(x1 - x2)
        s = ((y1 - y2) % p * divide1) % p
    x3 = ((s ** 2) % p - (x1 + x2) % p) % p
    y3 = (((s * (x1 - x3)) % p) % p - y1) % p
    print('Round : {} ({},{})'.format(i, x3, y3))
    return x2, y2, x3, y3

def calculate_send_message(x1, y1, x2, y2, m, a, p, flag):
    variables = calculate_P(x1, y1, x2, y2, a, p, flag, 1)
    if m - 1 <= 1:
        return variables[2], variables[3]
    else:
        for x in range(1, m):
            variables = calculate_P(variables[2], variables[3], variables[0], variables[1], a, p, flag, x + 1)
        return variables[2], variables[3]
