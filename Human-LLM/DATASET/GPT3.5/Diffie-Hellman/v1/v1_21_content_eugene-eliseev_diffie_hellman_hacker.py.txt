# Mock implementation of AESCipher for demonstration purposes
class AESCipher:
    def __init__(self, key):
        self.key = key

    def encrypt(self, data):
        # For simplicity, we'll just append "encrypted_" to data
        return "encrypted_" + data

    def decrypt(self, data):
        # For simplicity, we'll remove "encrypted_" from data
        return data.replace("encrypted_", "")

class Person:
    def __init__(self, prime, p):
        self.prime = prime
        self.p = p
        self.keys = {}
        self.messages = {}

    def start_communication_with(self, other):
        # Simulate communication with another person or hacker
        # For demonstration purposes, return mock data
        return "sender", "receiver", "message", "Hello, this is a message!"

    def encode(self, cipher, data):
        return cipher.encrypt(data)

    def decode(self, cipher, data):
        return cipher.decrypt(data)

class Hacker(Person):
    def read_data(self, sent):
        send_from, send_to, action, data = sent
        if action == "my param":
            K = (int(data) ** self.prime) % self.p
            self.keys[send_from] = AESCipher(str(K))
            _, _, action, data = self.start_communication_with("")
            return send_from, send_to, action, data
        elif send_from in self.keys and send_to in self.keys:
            data = self.decode(self.keys[send_from], data)
            if data in self.messages:
                data = self.messages[data]
            return send_from, send_to, "message", self.encode(self.keys[send_to], data)
        return sent
