import socket
from nacl.secret import SecretBox
from nacl.utils import random
from binascii import hexlify
from Crypto.Cipher import AES
import hashlib

class DiffieHellman:
    def __init__(self):
        self.p = 23
        self.g = 5
        self.a = random(32)

    def gen_key(self, b_pub_key):
        self.shared_key = pow(b_pub_key, self.a, self.p)

    @property
    def pub_key(self):
        return pow(self.g, self.a, self.p)

    def get_key(self):
        return hashlib.sha256(str(self.shared_key).encode()).digest()

# Initialize Diffie-Hellman key exchange
client = DiffieHellman()
server_address = ('localhost', 5555)

# Connect to the server
sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
sock.connect(server_address)
print("Connected to %s on port %s" % server_address)

# Send client's public key to the server
sock.sendall(str(client.pub_key).encode())

# Receive server's public key and generate shared secret
data = sock.recv(1024)
server_pub_key = int(data.decode())
client.gen_key(server_pub_key)
print("Shared secret key:", hexlify(client.get_key()))

# Initialize encryption box
box = SecretBox(client.get_key())

# Start communication loop
while True:
    print('Enter a message or "exit" to close the program:')
    message = input()
    if message != 'exit':
        # Encrypt and send the message
        encrypted_message = box.encrypt(message.encode())
        sock.sendall(encrypted_message)
        
        # Receive and decrypt the response
        data = sock.recv(1024)
        decrypted_data = box.decrypt(data)
        print('Received data:', decrypted_data.decode())
    else:
        print('Closing socket.')
        sock.close()
        break
