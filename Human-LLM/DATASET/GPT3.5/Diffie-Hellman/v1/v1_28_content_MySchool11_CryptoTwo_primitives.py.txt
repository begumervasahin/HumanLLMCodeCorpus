from random import sample

def calculate_prime_primitives(prime):
    print("Working out public key and public shared base...")
    primitive_roots = []
    for number_to_check in range(1, prime):
        candidate_prime_roots = set()
        for i in range(1, prime):
            modulus = (number_to_check ** i) % prime
            candidate_prime_roots.add(modulus)
            if len(candidate_prime_roots) == prime - 1:
                primitive_roots.append(number_to_check)
                break
    return sample(primitive_roots, 1)[0]

def main():
    prime = int(input("Enter the prime number: "))
    primitive_root = calculate_prime_primitives(prime)
    print(f"The primitive root for {prime} is: {primitive_root}")

if __name__ == "__main__":
    main()
