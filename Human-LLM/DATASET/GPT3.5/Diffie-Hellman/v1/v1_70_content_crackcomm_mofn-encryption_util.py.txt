import base64
from cryptography.hazmat.primitives import serialization
from cryptography.hazmat.primitives.asymmetric.x25519 import X25519PrivateKey

def private_bytes(private_key):
    """Converts a private key to bytes."""
    return private_key.private_bytes(
        encoding=serialization.Encoding.Raw,
        format=serialization.PrivateFormat.Raw,
        encryption_algorithm=serialization.NoEncryption()
    )

def public_bytes(private_key):
    """Converts a private key to its corresponding public key bytes."""
    return private_key.public_key().public_bytes(
        encoding=serialization.Encoding.Raw,
        format=serialization.PublicFormat.Raw
    )

def exchange_key(private_key, public_key):
    """Performs key exchange using the private and public keys."""
    return X25519PrivateKey.from_private_bytes(private_key).exchange(public_key)

def print_key(private_key):
    """Prints the private and public keys."""
    private_bytes_encoded = base64.b64encode(private_bytes(private_key))
    public_bytes_encoded = base64.b64encode(public_bytes(private_key))
    
    print('Private Key:', private_bytes_encoded.decode())
    print('Public Key:', public_bytes_encoded.decode())

# Example usage:
private_key = X25519PrivateKey.generate()
print_key(private_key)
