from rsa import RSA
from convertN2T import *

def main():
    print('================== - Criptografia RSA - ======================')
    block_size = 15
    rsa = RSA()
    
    with open("original_msg", "r") as msg_file:
        original_message = msg_file.read()
    
    print('================== -------------------- ======================')
    num_list = str_to_num(original_message)
    
    print(' ')
    print('Mensagem original')
    print(original_message)
    print(' ')
    print('---------------------------')
    print(num_list)
    
    blocks = num_to_blocks(num_list, block_size)
    encrypted_blocks = rsa.encryption(blocks)
    
    print("--------------------------")
    
    decrypted_blocks = rsa.decryption(encrypted_blocks)
    num_list = blocks_to_num(decrypted_blocks, block_size)
    
    print('---------- Voltando numero para string----------------')
    print(' ')
    decrypted_message = num_to_str(num_list)
    print(decrypted_message)

if __name__ == '__main__':
    main()
