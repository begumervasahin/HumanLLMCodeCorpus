import random

def totient(number):
    """
    Calculates the totient of a prime number.
    """
    if prime(number):
        return number - 1
    else:
        return False

def prime(n):
    """
    Checks if a number is prime.
    """
    if n <= 1:
        return False
    if n <= 3:
        return True
    if n % 2 == 0 or n % 3 == 0:
        return False
    i = 5
    while i * i <= n:
        if n % i == 0 or n % (i + 2) == 0:
            return False
        i += 6
    return True

def generate_E(num):
    """
    Generates a random number 'E' satisfying the conditions.
    """
    def mdc(n1, n2):
        rest = 1
        while n2 != 0:
            rest = n1 % n2
            n1 = n2
            n2 = rest
        return n1
    while True:
        e = random.randrange(2, num)
        if mdc(num, e) == 1:
            return e

def generate_prime():
    """
    Generates a random prime number.
    """
    while True:
        x = random.randrange(1, 100)
        if prime(x):
            return x

def mod(a, b):
    """
    Modular function between two numbers.
    """
    if a < b:
        return a
    else:
        c = a % b
        return c

def cipher(words, e, n):
    """
    Encrypts a message.
    """
    tam = len(words)
    i = 0
    lista = []
    while i < tam:
        letter = words[i]
        k = ord(letter)
        k = k ** e
        d = mod(k, n)
        lista.append(d)
        i += 1
    return lista

def descifra(cifra, n, d):
    """
    Decrypts an encrypted message.
    """
    lista = []
    i = 0
    tamanho = len(cifra)
    while i < tamanho:
        result = cifra[i] ** d
        texto = mod(result, n)
        letra = chr(texto)
        lista.append(letra)
        i += 1
    return lista

def calculate_private_key(toti, e):
    """
    Calculates the private key.
    """
    d = 0
    while mod(d * e, toti) != 1:
        d += 1
    return d

if __name__ == '__main__':
    text = input("Insert message: ")
    p = generate_prime()
    q = generate_prime()
    n = p * q
    y = totient(p)
    x = totient(q)
    totient_de_N = x * y
    e = generate_E(totient_de_N)
    public_key = (n, e)
    print('Your public key:', public_key)
    text_cipher = cipher(text, e, n)
    print('Your encrypted message:', text_cipher)
    d = calculate_private_key(totient_de_N, e)
    print('Your private key is:', d)
    original_text = descifra(text_cipher, n, d)
    print('Your original message:', ''.join(original_text))
