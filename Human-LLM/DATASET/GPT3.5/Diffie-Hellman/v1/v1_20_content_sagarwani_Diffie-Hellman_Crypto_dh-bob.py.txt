import sys
import random
import re

def pow_x(g_base, a, p_mod):
    """Efficiently computes modular exponentiation."""
    x = 1
    bits = "{0:b}".format(a)
    for bit in bits:
        if bit == '1':
            x = (((x ** 2) * g_base) % p_mod)
        elif bit == '0':
            x = ((x ** 2) % p_mod)
    return x % p_mod

def check_prime(n, k=128):
    """Checks if a number is prime using the Miller-Rabin primality test."""
    if n == 2 or n == 3:
        return True
    if n <= 1 or n % 2 == 0:
        return False
    s = 0
    r = n - 1
    while r & 1 == 0:
        s += 1
        r //= 2
    for _ in range(k):
        a = random.randint(2, n - 1)
        x = pow_x(a, r, n)
        if x != 1 and x != n - 1:
            j = 1
            while j < s and x != n - 1:
                x = pow_x(x, 2, n)
                if x == 1:
                    return False
                j += 1
            if x != n - 1:
                return False
    return True

if __name__ == '__main__':
    if len(sys.argv) != 3:
        print("Usage: python script.py <input_file> <output_file>")
        sys.exit(1)

    with open(sys.argv[1], 'r') as input_file:
        content = input_file.readline()
        big_primep = int(re.findall('\d+', content.split(',')[0])[0])
        generator = int(re.findall('\d+', content.split(',')[1])[0])
        ga = int(re.findall('\d+', content.split(',')[2])[0])
        
    big_primeq = (big_primep - 1) // 2
    while not check_prime(big_primeq):
        big_primeq -= 1
    
    b = random.randint(2, big_primeq - 1)
    gb = pow_x(generator, b, big_primep)
    
    with open(sys.argv[2], 'w') as output_file:
        output_file.write(str(gb))
    
    gab = pow_x(ga, b, big_primep)
    print(gab)
