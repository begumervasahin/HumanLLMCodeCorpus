import random

def encrypt_block(public_key, message, debug=False):
    k = random.randint(1, public_key['p'] - 1)
    if debug:
        print("DEBUG c1: %d c2: %d" % (pow(public_key['g'], k, public_key['p']), (pow(public_key['e2'], k, public_key['p']) * message) % public_key['p']))
    return {
        'c1': pow(public_key['g'], k, public_key['p']),
        'c2': (pow(public_key['e2'], k, public_key['p']) * message) % public_key['p']
    }

def make_message(string):
    num = sum((ord(c) << (i * 8)) for i, c in enumerate(string[::-1]))
    return num

def make_public_key_from_file(public_key_file):
    public_key = {}
    with open(public_key_file, 'r') as pkf:
        public_key_list = pkf.readlines()
    public_key['p'] = int(public_key_list[0])
    public_key['g'] = int(public_key_list[1])
    public_key['e2'] = int(public_key_list[2])
    return public_key

def encrypt_file(read_path="test/testText.txt", write_path="test/testEncrypted.txt", public_key_path='test/testpubkey.txt', debug=False):
    public_key = make_public_key_from_file(public_key_path)
    with open(read_path, 'r', encoding='utf-8') as rf:
        all_text = rf.read()
    with open(write_path, 'w') as wf:
        for c in all_text:
            cipher = encrypt_block(public_key, ord(c), debug)
            wf.write(str(cipher['c1']) + " " + str(cipher['c2']) + "\n")

# Example usage:
encrypt_file(read_path="test/testText.txt", write_path="test/testEncrypted.txt", public_key_path='test/testpubkey.txt', debug=True)
