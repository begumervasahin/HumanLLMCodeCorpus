import Encryption
import os

def convert_to_int(key):
    """Converts a random key to an integer."""
    return int.from_bytes(key, byteorder='big')

# Generate random keys
random_key1 = os.urandom(32)
random_key2 = os.urandom(32)

random_key1f = convert_to_int(random_key1)
random_key2f = convert_to_int(random_key2)

print(" -=keys=- ")
print(random_key1f)
print(random_key2f)

# Initialize AESCipher instances with the random keys
part1 = Encryption.AESCipher(random_key1f)
part2 = Encryption.AESCipher(random_key2f)

print("\n-=public keys=-")
print(part1.getPubKey())
print(part2.getPubKey())

# Perform Diffie-Hellman key exchange
part1.DHEC(part2.getPubKey())
part2.DHEC(part1.getPubKey())

print("\n-=TEST ROUNDS=-")
# Encrypt and decrypt test messages
ent1 = part1.encrypt("howdy" + " " + 128 * ",")
print(part2.decrypt(ent1))

ent2 = part2.encrypt("howdy2")
print(part1.decrypt(ent2))
