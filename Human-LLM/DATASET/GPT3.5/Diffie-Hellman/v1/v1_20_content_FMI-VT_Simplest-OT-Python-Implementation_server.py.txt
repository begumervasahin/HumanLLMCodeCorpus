from Crypto.Cipher import AES, DES, DES3, Blowfish, ChaCha20
import hashlib
import pickle
import random
import socket
import time
import argparse
from ecc import getcurvebyname

# Constants for host and port
host = "127.0.0.1"
port = 4446

# Define command-line arguments
parser = argparse.ArgumentParser()
parser.add_argument("e", type=int, choices=[0, 1, 2, 3, 4],
                    help="Symmetric encryption algorithm [0 - AES, 1 - DES, 2 - 3DES, 3 - Blowfish, 4 - ChaCha20]")
parser.add_argument("-d", "--hash", type=int, choices=[0, 1, 2],
                    help="Hash function [0 - md5, 1 - blake2s, 2 - SHA256]")
args = parser.parse_args()

def decrypt_data(secret, msg):
    "Decrypts the data based on the used cipher"
    msg_nonce = msg[:8]
    ciphertext = msg[8:]
    cipher = ChaCha20.new(key=secret, nonce=msg_nonce)
    decrypted_data = cipher.decrypt(ciphertext)
    return decrypted_data

def get_cipher(key):
    "Returns the Cipher based on the preferred algorithm - AES, DES, 3DES, Blowfish, or ChaCha20"
    if args.e == 0:
        cipher = AES.new(key, AES.MODE_ECB)
    elif args.e == 1:
        cipher = DES.new(key, DES.MODE_ECB)
    elif args.e == 2:
        cipher = DES3.new(key, DES3.MODE_ECB)
    elif args.e == 3:
        cipher = Blowfish.new(key, Blowfish.MODE_ECB)
    elif args.e == 4:
        cipher = ChaCha20.new(key=key)
    return cipher

def get_key(str_value):
    "Returns the key based on the chosen symmetric function"
    if args.e == 0:
        if args.hash == 0:
            temp_key = hashlib.md5()
        elif args.hash == 1:
            temp_key = hashlib.blake2s(digest_size=16)
        else:
            temp_key = hashlib.md5()
    elif args.e == 1:
        temp_key = hashlib.blake2s(digest_size=8)
    elif args.e == 2:
        temp_key = hashlib.blake2s(digest_size=16)
    elif args.e == 3:
        temp_key = hashlib.blake2s()
    elif args.e == 4:
        temp_key = hashlib.blake2s(digest_size=32)
    temp_key.update(str_value)
    return temp_key.digest()

def read_from_client(q):
    "Reads data from the client and performs decryption"
    c = random.randint(0, 1)
    curve = getcurvebyname("ed25519")
    g = curve.G
    b = random.randint(1, 2 ** 255 - 19)
    alice = pickle.loads(q.recv(4096))

    if c == 0:
        bob = g.__mul__(b)
    else:
        bob = (alice.__mul__(c)).__add__(g.__mul__(b))

    q.send(pickle.dumps(bob, pickle.HIGHEST_PROTOCOL))
    k = get_key(str(alice.__mul__(b)).encode())
    cipher1 = get_cipher(k)
    messages = [1] * 2

    for i in range(2):
        en = q.recv(1024)
        if args.e == 4:
            messages[i] = decrypt_data(k, en)
        else:
            messages[i] = cipher1.decrypt(en)
        print('Message [', i, ']', messages[i].decode('iso-8859-15'))

# Set up socket and listen for connections
s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
s.bind((host, port))
s.listen(1)
print("Listening for connections...")

# Print encryption algorithm
encryption_algorithms = ["AES", "DES", "3DES", "Blowfish", "ChaCha20"]
print(f"{encryption_algorithms[args.e]} encryption")

# Accept connections and process data
q, addr = s.accept()
start_time = time.time()
for i in range(10):
    start_time = time.time()
    read_from_client(q)
    print("----- %s seconds ----" % (time.time() - start_time))

s.close()
