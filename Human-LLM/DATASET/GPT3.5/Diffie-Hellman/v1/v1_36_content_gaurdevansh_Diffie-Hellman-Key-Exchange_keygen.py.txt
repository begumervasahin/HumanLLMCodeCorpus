import random

# Prime number and generator
prime = 26994308385016394749558484505346578147056894665639
g = 2

# Function to generate a private key and save it to a file
def gen_private(filename):
    private_key = str(random.randint(1000, 10000))
    with open(filename, 'w') as outfile:
        outfile.write(private_key)

# Function to generate a public key from a private key and save it to a file
def gen_public(private_filename, public_filename):
    with open(private_filename, 'r') as infile:
        private_key = int(infile.read())
    public_key = (g ** private_key) % prime
    with open(public_filename, 'w') as outfile:
        outfile.write(str(public_key))

# Function to generate the shared secret key using private and public keys
def gen_secret(private_file, public_file):
    with open(private_file, 'r') as infile:
        private_key = int(infile.read())
    with open(public_file, 'r') as infile:
        public_key = int(infile.read())
    secret_key = (public_key ** private_key) % prime
    return secret_key

# Example usage
if __name__ == "__main__":
    # Generate and save private keys
    gen_private("private_key_Alice.txt")
    gen_private("private_key_Bob.txt")

    # Generate public keys and save them
    gen_public("private_key_Alice.txt", "public_key_Alice.txt")
    gen_public("private_key_Bob.txt", "public_key_Bob.txt")

    # Generate and print shared secret keys
    secret_Alice = gen_secret("private_key_Alice.txt", "public_key_Bob.txt")
    secret_Bob = gen_secret("private_key_Bob.txt", "public_key_Alice.txt")
    print("Shared secret key for Alice:", secret_Alice)
    print("Shared secret key for Bob:", secret_Bob)
