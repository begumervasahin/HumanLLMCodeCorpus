def gcd(n1, n2):
    """Calculate the greatest common divisor (GCD) of two numbers."""
    while n2 != 0:
        n1, n2 = n2, n1 % n2
    return n1

def lcm(n1, n2):
    """Calculate the least common multiple (LCM) of two numbers."""
    return (n1 * n2) // gcd(n1, n2)

def euclidean_extended_gcd(n1, n2):
    """Calculate the extended Euclidean algorithm for two numbers."""
    u, v, s, t = 1, 0, 0, 1
    while n2 != 0:
        q = n1 // n2
        n1, n2 = n2, n1 - q * n2
        u, s = s, u - q * s
        v, t = t, v - q * t
    return n1, u, v
