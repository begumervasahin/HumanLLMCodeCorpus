from random import randint, randrange

def is_prime(number):
    if number <= 3:
        return number > 1
    if not number % 2 or not number % 3:
        return False
    for i in range(5, int(number ** 0.5) + 1, 6):
        if not number % i or not number % (i + 2):
            return False
    return True

def generate_prime(n, m):
    prime = 0
    while not is_prime(prime):
        prime = randint(n, m)
    return prime

def gcd(a, b):
    while b != 0:
        a, b = b, a % b
    return a

def extended_gcd(a, b):
    x, xx, y, yy = 1, 0, 0, 1
    while b != 0:
        q = a // b
        a, b = b, a % b
        xx, x = x - q * xx, xx
        yy, y = y - q * yy, yy
    return a, x, y

def mod_inverse(a, n):
    c, u, v = extended_gcd(a, n)
    if c != 1:
        return 0
    else:
        return u % n

print("---------------------------------\n----------ALGORITHME RSA---------\n---------------------------------\n")

p = generate_prime(100000000, 999999999)
q = generate_prime(100000000, 999999999)

print("\nLes nombres premiers choisis sont:\np = {}, q = {}\n".format(p, q))

n = p * q
print("\nn = p * q = {}\n".format(n))

phi = (p - 1) * (q - 1)
print("fonction de Euler [phi(n) = (p-1)(q-1)] : {}\n".format(phi))

e = 2
g = gcd(phi, e)

while g != 1:
    e = randrange(1, phi)
    g = gcd(phi, e)

d = mod_inverse(e, phi)

print("\nNotre clé publique est la paire des nombres (e = {}, n = {}).".format(e, n))
