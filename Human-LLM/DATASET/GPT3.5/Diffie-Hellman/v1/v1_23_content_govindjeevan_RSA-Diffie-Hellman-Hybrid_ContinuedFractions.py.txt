def rational_to_contfrac(x, y):
    """
    Converts a rational x/y fraction into a list of partial quotients [a0, ..., an]
    """
    a = x // y
    partial_quotients = [a]
    while a * y != x:
        x, y = y, x - a * y
        a = x // y
        partial_quotients.append(a)
    return partial_quotients

def convergents_from_contfrac(frac):
    """
    Computes the list of convergents using the list of partial quotients
    """
    convergents = []
    for i in range(len(frac)):
        convergents.append(contfrac_to_rational(frac[0:i + 1]))
    return convergents

def contfrac_to_rational(frac):
    """
    Converts a finite continued fraction [a0, ..., an] to an x/y rational.
    """
    if len(frac) == 0:
        return (0, 1)
    num = frac[-1]
    denom = 1
    for i in range(len(frac) - 2, -1, -1):
        num, denom = frac[i] * num + denom, num
    return (num, denom)

def test1():
    """
    Verify that the basic continued-fraction manipulation stuff works.
    """
    test_numbers = [(1, 1), (1, 2), (5, 15), (27, 73), (73, 27)]
    for r in test_numbers:
        num, denom = r
        print('Rational number:')
        print(r)
        contfrac = rational_to_contfrac(num, denom)
        print('Continued fraction:')
        print(contfrac)
        print('Convergents:')
        print(convergents_from_contfrac(contfrac))
        print('***********************************')

if __name__ == "__main__":
    test1()
