import tkinter as tk
from tkinter import messagebox

def is_prime(n):
    if n < 2:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

def calculate_secret_key():
    try:
        p1 = int(entry1.get())
        p2 = int(entry2.get())
        a1 = int(entry3.get())
        b1 = int(entry4.get())
    except ValueError:
        messagebox.showinfo('Error', 'Please enter integers for prime numbers and secret numbers')
        return

    if not is_prime(p1):
        messagebox.showinfo('Error', 'First input is not prime')
    elif not is_prime(p2):
        messagebox.showinfo('Error', 'Second input is not prime')
    else:
        a2 = (p1 ** a1) % p2
        b2 = (p1 ** b1) % p2
        secret_key = (b2 ** a1) % p2
        secret_key_check = (a2 ** b1) % p2

        if secret_key != secret_key_check:
            messagebox.showinfo('Error', 'Unknown Error')
        else:
            messagebox.showinfo('Key', 'Your Secret Key is: ' + str(secret_key))

root = tk.Tk()
root.title('Math 243 Project')
root.geometry('275x210')

label1 = tk.Label(root, text="What is your first prime number? ")
entry1 = tk.Entry(root, bd=1)

label2 = tk.Label(root, text="What is your second prime number? ")
entry2 = tk.Entry(root, bd=1)

label3 = tk.Label(root, text="What is your first secret number? ")
entry3 = tk.Entry(root, bd=1)

label4 = tk.Label(root, text="What is your second secret number? ")
entry4 = tk.Entry(root, bd=1)

submit = tk.Button(root, text="Submit", command=calculate_secret_key)

label1.pack()
entry1.pack()

label2.pack()
entry2.pack()

label3.pack()
entry3.pack()

label4.pack()
entry4.pack()

submit.pack(side=tk.BOTTOM)

root.mainloop()
