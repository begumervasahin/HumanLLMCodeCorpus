import math
import random
import sys
from fractions import gcd

def give_me_primes():
    """Generate a list of prime numbers between 1000 and 10000."""
    prime_list = []
    for num in range(1000, 10001):
        if all(num % i != 0 for i in range(2, int(math.sqrt(num)) + 1)):
            prime_list.append(num)
    return prime_list

def pick_random_prime(primes):
    """Pick a random prime number from the given list of primes."""
    secure_random = random.SystemRandom()
    return secure_random.choice(primes)

def main():
    primes_list = give_me_primes()
    P = pick_random_prime(primes_list)
    print(f"Modulus Number: {P}")

    secure_random = random.SystemRandom()
    N = secure_random.choice(range(1, 10001))
    print(f"Base Number: {N}")

    A = secure_random.choice(range(1, 10001))
    B = secure_random.choice(range(1, 10001))

    if gcd(N, A) != 1:
        print(f"N({N}) and A({A}) can't be relatively prime")
        sys.exit(1)

    if gcd(N, B) != 1:
        print(f"N({N}) and B({B}) can't be relatively prime")
        sys.exit(1)

    print(f"A is {A}")
    print(f"B is {B}")

    J = N ** A % P
    K = N ** B % P

    print(f"A computational numbers: {J}")
    print(f"B computational numbers: {K}")

    if K ** A % P == J ** B % P:
        print("Computation successful! The computed values match.")
    else:
        print("Computation failed! The computed values do not match.")

if __name__ == "__main__":
    main()
