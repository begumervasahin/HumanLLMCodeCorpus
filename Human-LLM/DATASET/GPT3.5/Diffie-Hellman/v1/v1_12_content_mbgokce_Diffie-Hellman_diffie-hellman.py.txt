import random
from Crypto.Util import number

# Generate a prime number with 9 digits
p = number.getPrime(9, None)
g = random.randint(5, 200)
x = random.randint(0, p - 1)
y = random.randint(0, p - 1)

aliceKey = pow(g, x, p)
bobKey = pow(g, y, p)

print("g:", g)
print("p:", p)
print("Alice's public key:", aliceKey)
print("Bob's public key:", bobKey)

alicePriv = pow(bobKey, x, p)
bobPriv = pow(aliceKey, y, p)

if bobPriv != alicePriv:
    print("Woah! Something is terribly wrong.")
else:
    print("Alice's and Bob's shared secret:", alicePriv)

print("\n---Brute force---")
for X in range(0, p - 1):
    alicePublic = pow(g, X, p)
    if alicePublic == aliceKey:
        break

secret = pow(bobKey, X, p)
