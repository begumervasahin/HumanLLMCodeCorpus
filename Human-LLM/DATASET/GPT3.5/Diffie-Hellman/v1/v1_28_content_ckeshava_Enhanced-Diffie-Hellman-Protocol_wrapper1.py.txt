import os
import random
import paderson_commitment
import constants
from datetime import datetime
from threading import Thread

class NodeA(Thread):
    def run(self):
        os.system('sleep 0.1; python3 node-A.py')

class NodeB(Thread):
    def run(self):
        os.system('python3 node-B.py')

def main():
    A = list(range(10, 100, 5))
    P = [100, 140, 160, 180, 320, 384, 512]
    
    with open('100.csv', 'a') as c:
        for i in P:
            p, g, q = paderson_commitment.generate_p_g_q()
            with open('constants.py', 'w') as f:
                m = 'PORT = {}\nSERVER_IP_ADDR = "127.0.0.1"\nSHARED_PRIME = {}\nSHARED_BASE = 2\n'.format(
                    random.randint(13000, 20000), p)
                for j in A:
                    m += 'RANDOM_STRING_LENGTH = {}\nRAND_NUMBER_LOWER_BOUND = 100\nRAND_NUMBER_UPPER_BOUND = 1000\nLENGTH_OF_PRIME = {}\n'.format(
                        j, i)
                    f.write(m)
                    os.system('sleep 1')
                    b = NodeB()
                    a = NodeA()
                    start = datetime.now()
                    b.start()
                    a.start()
                    a.join()
                    b.join()
                    end = datetime.now()
                    print("Prime = {}, String Length = {}".format(i, j))
                    t = (end - start).microseconds
                    s = '{},{},{}\n'.format(i, j, t)
                    c.write(s)
                    c.flush()

if __name__ == "__main__":
    main()
