import socket

def powmod(p, e, n):
    result = 1
    while e != 0:
        result *= p % n
        e -= 1
    return result % n

# Create a socket object
s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)

# Get the hostname
host = socket.gethostname()

print("Server will start on host", host)

# Set the port number
port = 1024

# Bind the socket to the host and port
s.bind((host, port))

print("Server done binding of host and port successfully")
print("Server is waiting for connection")

# Listen for incoming connections
s.listen(1)

# Accept a connection
conn, addr = s.accept()

print(addr, "is connected to the server and is online now...")

# Get the server's private key from user input
prikey = int(input("Enter Server private key: "))

# Calculate and send the encrypted message to the client
message = str(powmod(17, prikey, 23))
message = message.encode()
conn.send(message)

print("Message has been sent")

# Receive the encrypted message from the client
income_message = conn.recv(1024)
income_message = income_message.decode()

print("Client:", income_message)

income_message = int(income_message)

# Calculate and print the shared secret key
shared_secret = powmod(income_message, prikey, 23)
print("Shared secret key:", shared_secret)

# Close the connection
conn.close()
