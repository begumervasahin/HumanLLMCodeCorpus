import ModularOperations as mod

class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y

def add_points(P, Q, curve):
    """
    Adds points P and Q for a specified elliptic curve.
    """
    p = curve.p
    a = curve.a
    R = Point(p, p)
    
    if P.x == p:
        return Q
    if Q.x == p:
        return P
    
    if P.x == Q.x and P.y == Q.y:
        modinv_2P = mod.modinv((2 * P.y) % p, p)
        if modinv_2P == p:
            return Point(p, p)
        else:
            s = ((3 * (P.x * P.x) + a) * modinv_2P) % p
    else:
        modinv_PQ = mod.modinv((P.x - Q.x) % p, p)
        if modinv_PQ == p:
            return Point(p, p)
        else:
            s = ((P.y - Q.y) * modinv_PQ) % p
    
    R.x = (s * s - P.x - Q.x) % p
    R.y = (-s * s * s + s * (P.x + Q.x) - P.y + s * P.x) % p
    
    return R

def multiply(k, P, curve):
    """
    Multiplies point P by an integer k for a specified elliptic curve.
    """
    p = curve.p
    ls = []
    
    if k >= 0 or P.x == p:
        Q = P
    else:
        Q = Point(P.x, p - P.y)
    
    k_bits = "{0:b}".format(abs(k))
    ls.append(Q)
    
    for i in range(len(k_bits) - 1):
        Q = add_points(Q, Q, curve)
        ls.append(Q)
    
    R = Point(p, p)
    for i, i_k in enumerate(k_bits[::-1]):
        if i_k == '1':
            R = add_points(R, ls[i], curve)
    
    return R

# Example usage:
# Define an elliptic curve and points P and Q
class EllipticCurve:
    def __init__(self, p, a, b):
        self.p = p
        self.a = a
        self.b = b

curve_p = 23
curve_a = 1
curve_b = 1
curve = EllipticCurve(curve_p, curve_a, curve_b)

P = Point(5, 1)
Q = Point(6, 3)

# Add points P and Q
result_add = add_points(P, Q, curve)
print("Result of adding points P and Q:", result_add.x, result_add.y)

# Multiply point P by an integer k
k = 7
result_multiply = multiply(k, P, curve)
print("Result of multiplying point P by", k, ":", result_multiply.x, result_multiply.y)
