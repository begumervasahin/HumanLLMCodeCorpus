from math import sqrt
from itertools import count, islice
import random

class DHCalculator:
    def __init__(self):
        self.a = 0
        self.B = 0
        self.P = 0

    def isPrime(self, n):
        return n > 1 and all(n % i for i in islice(count(2), int(sqrt(n) - 1)))

    def generateGP(self):
        minN = 1000
        maxN = 10000
        cached_primes = [i for i in range(minN, maxN) if self.isPrime(i)]
        return random.choice([i for i in cached_primes if minN < i < maxN])

    def setA(self, a):
        self.a = a

    def calcularA(self, G, P):
        return G ** self.a % P

    def calcularKey(self, B, P):
        return B ** self.a % P


# Example usage
if __name__ == "__main__":
    dh = DHCalculator()
    
    # Generate random G and P
    G = dh.generateGP()
    P = dh.generateGP()
    print(f"Generated G: {G}, P: {P}")
    
    # Set a random value for a
    a = random.randint(1, 100)
    dh.setA(a)
    print(f"Set a: {a}")
    
    # Calculate A and B
    A = dh.calcularA(G, P)
    B = dh.calcularA(G, P)  # Note: Using calcularA instead of calcularKey as per the provided code
    print(f"Calculated A: {A}, B: {B}")
