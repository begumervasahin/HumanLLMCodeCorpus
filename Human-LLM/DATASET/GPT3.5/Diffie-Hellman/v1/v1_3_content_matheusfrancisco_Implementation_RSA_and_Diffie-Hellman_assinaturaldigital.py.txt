from rsa import RSA
from convertN2T import *

def encryption(msg, d, n):
    """Encrypts the message using RSA."""
    encrypted_msg = []
    for char in msg:
        encrypted_msg.append(modExp(char, d, n))
    return encrypted_msg

def sign_message(msg, e, n):
    """Creates a digital signature for the message using RSA."""
    signature = []
    for char in msg:
        signature.append(modExp(char, e, n))
    return signature

def verify_signature(signature, msg_original, ea, na, eb, db):
    """Verifies the digital signature of the received message."""
    decrypted_signature = []
    for char in signature:
        decrypted_signature.append(modExp(char, ea, na))
    decrypted_msg = []
    for char in signature:
        decrypted_msg.append(modExp(char, db, eb))
    return decrypted_msg
