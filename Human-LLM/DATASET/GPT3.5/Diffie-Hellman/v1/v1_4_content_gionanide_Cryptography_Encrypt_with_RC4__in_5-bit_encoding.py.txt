def sxor(s1, s2):
    """Perform XOR operation on two strings."""
    return ''.join(str(int(x) ^ int(y)) for x, y in zip(s1, s2))

def text_enc(text):
    """Encode text using a predefined dictionary."""
    aDict = dict(zip('abcdefghijklmnopqrstuvwxyz.!?()-ABCDEFGHIJKLMNOPQRSTUVWXYZ',
                     ['00000', '00001', '00010', '00011', '00100',
                      '00101', '00110', '00111', '01000', '01001',
                      '01010', '01011', '01100', '01101', '01110',
                      '01111', '10000', '10001', '10010', '10011',
                      '10100', '10101', '10110', '10111', '11000',
                      '11001', '11010', '11011', '11100', '11101',
                      '11110', '11111', '00000', '00001', '00010',
                      '00011', '00100', '00101', '00110', '00111',
                      '01000', '01001', '01010', '01011', '01100',
                      '01101', '01110', '01111', '10000', '10001',
                      '10010', '10011', '10100', '10101', '10110',
                      '10111', '11000', '11001']))
    text = text[::-1]  # Reverse the text
    coded_text = ''
    for char in text:
        coded_text = aDict[char] + coded_text
    return coded_text.lower()

def text_dec(binary_string):
    """Decode a binary string using a predefined dictionary."""
    aDict = dict(zip('abcdefghijklmnopqrstuvwxyz.!?()-ABCDEFGHIJKLMNOPQRSTUVWXYZ',
                     ['00000', '00001', '00010', '00011', '00100',
                      '00101', '00110', '00111', '01000', '01001',
                      '01010', '01011', '01100', '01101', '01110',
                      '01111', '10000', '10001', '10010', '10011',
                      '10100', '10101', '10110', '10111', '11000',
                      '11001', '11010', '11011', '11100', '11101',
                      '11110', '11111', '00000', '00001', '00010',
                      '00011', '00100', '00101', '00110', '00111',
                      '01000', '01001', '01010', '01011', '01100',
                      '01101', '01110', '01111', '10000', '10001',
                      '10010', '10011', '10100', '10101', '10110',
                      '10111', '11000', '11001']))
    inv_map = {v: k for k, v in aDict.items()}
    length = len(binary_string)
    decoded_text = ''
    for i in range(0, length, 5):
        decoded_text = inv_map[binary_string[i:i + 5]] + decoded_text
    decoded_text = decoded_text[::-1]  # Reverse the decoded text back
    return decoded_text.lower()

def ex2():
    """Encrypt text using a stream cipher."""
    a = 'WE ALL MAKE MISTAKES AND WE ALL PAY A PRICE'
    key = 'HOUSE'
    text = a.replace(" ", "")  # Remove spaces from text
    print('Text:', text)
    print('Key:', key)
    keyByte = text_enc(key)
    textByte = text_enc(text)
    print('Key in five-bit encoding:', keyByte)
    print('Message in five-bit encoding:', textByte)

    # Initialize variables
    keyLength = len(keyByte)
    textLength = len(textByte)
    S = list(range(256))
    keyBytes = [int(char) for char in keyByte]
    textBytes = [int(char) for char in textByte]
    i = j = 0

    keystreamList = []
    cipherList = []

    # Key-scheduling algorithm
    for i in range(256):
        j = (j + S[i] + keyBytes[i % keyLength]) % 256
        S[i], S[j] = S[j], S[i]

    # Generate keystream and encrypt message
    for m in range(textLength):
        i = (i + 1) % 256
        j = (j + S[i]) % 256
        S[i], S[j] = S[j], S[i]
        k = S[(S[i] + S[j]) % 256]
        keystreamList.append(k)
        cipherList.append(sxor(str(k), str(textBytes[m])))

    # Decrypt the encrypted message
    decrypted_text = ''.join([str(sxor(str(c), str(k))) for c, k in zip(cipherList, keystreamList)])
    print('Decrypted text:', text_dec(decrypted_text))


# Run the example function
ex2()
