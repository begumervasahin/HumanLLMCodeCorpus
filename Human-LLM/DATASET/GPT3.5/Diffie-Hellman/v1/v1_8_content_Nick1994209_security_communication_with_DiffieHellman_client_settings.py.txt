import sys
from PyQt5.QtWidgets import QApplication, QWidget, QLineEdit, QLabel, QPushButton, QSizePolicy
from PyQt5.QtCore import QCoreApplication
import random


class ClientSetSocketSettingsMixin:
    def __init__(self):
        self.app = QApplication(sys.argv)
        self.window = QWidget()
        self.init_ui()

    def init_ui(self):
        self.window.setWindowTitle('Client')
        self.window.setGeometry(600, 300, 400, 200)

        self.input_port_client = QLineEdit(self.window)
        self.input_port_client.move(40, 10)
        label_port = QLabel("port", self.window)
        label_port.setSizePolicy(QSizePolicy.Expanding, QSizePolicy.Expanding)
        label_port.setStyleSheet("QLabel {color: white;}")
        label_port.move(0, 12)

        self.input_ip_client = QLineEdit(self.window)
        self.input_ip_client.move(200, 10)
        self.input_ip_client.setText('localhost')
        label_ip = QLabel("ip", self.window)
        label_ip.setSizePolicy(QSizePolicy.Expanding, QSizePolicy.Expanding)
        label_ip.setStyleSheet("QLabel {color: white;}")
        label_ip.move(180, 12)

        self.input_secret_key_client = QLineEdit(self.window)
        self.input_secret_key_client.move(40, 50)
        label_secret_key = QLabel("key", self.window)
        label_secret_key.setSizePolicy(QSizePolicy.Expanding, QSizePolicy.Expanding)
        label_secret_key.setStyleSheet("QLabel {color: white;}")
        label_secret_key.move(0, 50)

        btn_random_secret_key = QPushButton('random_secret_key', self.window)
        btn_random_secret_key.clicked.connect(self.random_secret_key)
        btn_random_secret_key.resize(btn_random_secret_key.sizeHint())
        btn_random_secret_key.move(200, 50)

        btn_next = QPushButton('next', self.window)
        btn_next.clicked.connect(self.next_client_window)
        btn_next.resize(btn_next.sizeHint())
        btn_next.move(180, 170)

        btn_exit = QPushButton('Quit', self.window)
        btn_exit.clicked.connect(QCoreApplication.instance().quit)
        btn_exit.resize(btn_exit.sizeHint())
        btn_exit.move(300, 170)

        btn_widget_choice = QPushButton('ago', self.window)
        btn_widget_choice.clicked.connect(self.gui_choice_server_client)
        btn_widget_choice.resize(btn_exit.sizeHint())
        btn_widget_choice.move(30, 170)

        self.objects_to_hide = (
            btn_exit, btn_widget_choice, btn_next, btn_random_secret_key,
            label_port, label_ip, label_secret_key,
            self.input_port_client, self.input_ip_client, self.input_secret_key_client
        )

        self.show_current_objects()

    def show_current_objects(self):
        for obj in self.objects_to_hide:
            obj.show()

    def hide_last_objects(self):
        for obj in self.objects_to_hide:
            obj.hide()

    def random_secret_key(self):
        self.input_secret_key_client.setText(str(random.randint(0, 10000)))

    def next_client_window(self):
        if self.input_port_client.text() and self.input_ip_client.text() and self.input_secret_key_client.text():
            if self.input_port_client.text().isdigit():
                self.client_communication()

    def client_communication(self):
        # Your client communication logic goes here
        pass

    def gui_choice_server_client(self):
        # Your GUI choice logic goes here
        pass

    def run(self):
        self.window.show()
        sys.exit(self.app.exec_())


if __name__ == '__main__':
    client = ClientSetSocketSettingsMixin()
    client.run()
