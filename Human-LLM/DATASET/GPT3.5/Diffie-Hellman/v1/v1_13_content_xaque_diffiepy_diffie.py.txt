from random import SystemRandom
import subprocess

def mod_exp(x, y, N):
    if N == 1:
        return 0
    result = 1
    x = x % N
    while y > 0:
        if y % 2 == 1:
            result = (result * x) % N
        y >>= 1
        x = (x * x) % N
    return result

def generate_p():
    rounds = 0
    prime_found = False
    while not prime_found:
        rounds += 1
        cmd = 'openssl prime -generate -bits 512'
        proc = subprocess.Popen(cmd, shell=True, stdin=subprocess.PIPE, stdout=subprocess.PIPE, close_fds=True)
        prime = proc.stdout.read().decode("utf-8")
        prime = int(prime[:-1])
        p = int((prime * 2) + 1)
        is_prime = 0
        cmd = 'openssl prime -checks 20 ' + str(p)
        proc = subprocess.Popen(cmd, shell=True, stdin=subprocess.PIPE, stdout=subprocess.PIPE, close_fds=True)
        output = proc.stdout.read().decode("utf-8")
        output = output[:-1]
        words = output.split(" ")
        is_prime = words[-2] != 'not'
        if is_prime:
            prime_found = True
            return p

def generate_a():
    return int(SystemRandom().random() * (2 ** 512))
