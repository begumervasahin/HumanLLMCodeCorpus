import random

def checklist(pri):
    duplicate = pri[:]
    check = 1
    for i in range(len(pri)):
        for j in range(i + 1, len(pri)):
            if pri[i] == duplicate[j]:
                check = 0
                break
            else:
                check = 1
                continue
        break
    return check

def main():
    q = int(input("Enter the prime number: "))
    n = int(input("Enter the number of users: "))
    keys = []
    for i in range(n):
        key = int(input("Enter secret key for user {}: ".format(i + 1)))
        keys.append(key)

    if n > 2:
        print("Enter the users between whom you want to communicate:")
        temp1 = int(input("Enter the user 1: "))
        xa = keys[temp1 - 1]
        temp2 = int(input("Enter the user 2: "))
        xb = keys[temp2 - 1]
    else:
        xa = keys[0]
        xb = keys[1]

    p = 0
    a = []
    pri = []
    alpha = 0

    for i in range(1, q):
        for j in range(1, q):
            pri.append((i**j) % q)
        if checklist(pri) == 1:
            p += 1
            a.append(i)
        pri.clear()

    alpha = random.choice(a)
    print("alpha:", alpha)

    ya = (alpha**xa) % q
    print("Public key of A:", ya)

    yb = (alpha**xb) % q
    print("Public key of B:", yb)

    Kab1 = (yb**xa) % q
    print("Shared session key with respect to user A:", Kab1)

    Kab2 = (ya**xb) % q
    print("Shared session key with respect to user B:", Kab2)

if __name__ == "__main__":
    main()
