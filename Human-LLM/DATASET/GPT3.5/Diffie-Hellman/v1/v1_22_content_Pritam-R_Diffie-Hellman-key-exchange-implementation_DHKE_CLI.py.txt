import socket
import re
import base64
import random

class DiffieHellman:
    def __init__(self, root, prime, recv_key):
        self.root = root
        self.prime = prime
        self.recv_key = recv_key
    
    def key_gen(self):
        range_start = 10**(7-1)
        range_end = (10**7)-1
        self.key = random.randint(range_start, range_end)
    
    def shared_key(self):
        mod = (self.root ** self.key) % self.prime
        return str(mod)
    
    def secret_key(self):
        sec_key = (self.recv_key ** self.key) % self.prime
        return sec_key

    port = port_no
    IP = client_ip

            data = sock.recv(4096)

                data_decode = base64.b64decode(data).decode('utf-8')

                    prime = int(pattern.group(2))
                    recv_key = int(pattern.group(3))
