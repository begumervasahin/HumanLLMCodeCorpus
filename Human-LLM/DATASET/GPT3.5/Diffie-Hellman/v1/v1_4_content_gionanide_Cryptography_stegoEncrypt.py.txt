from PIL import Image
import hashlib

def secret_message():
    bits_list = []
    secret = input('Enter the message you want to hide: ')
    for x in secret:
        bits_list.append("{0:08b}".format(ord(x)))
    print(''.join(bits_list))
    return ''.join(bits_list)

def key_schedule(height, width):
    key = input('Enter the key: ')
    first_pass = hashlib.sha256(key.encode()).hexdigest()
    second_pass = hashlib.sha256(first_pass.encode()).hexdigest()
    third_pass = hashlib.sha256(second_pass.encode()).hexdigest()
    secret_key = first_pass + second_pass + third_pass
    signal = create_signal(secret_key, height, width)
    print('Private key:', secret_key, ', Signal:', signal)
    return signal

def create_signal(key, height, width):
    result = 0
    for x in range(len(key) - 4):
        result += xor(key[x:x + 2], key[x + 2:x + 4])
    return result % (height - len(key))

def xor(a, b):
    c = int(a, 16)
    d = int(b, 16)
    e = c ^ d
    return e

def transform_image():
    image = Image.open('walknut.jpg')
    width, height = image.size
    rgb_im = image.convert('RGB')
    return width, height, image

def replace_bit(color, bit):
    color = "{0:08b}".format(color)
    color = color[::-1].replace(color[len(color) - 1], bit, 1)
    return color[::-1]

def lsb(height, width, image, message_list, signal):
    y = signal
    k = 0
    for x in range(signal, signal + len(message_list) - 1, 3):
        r, g, b = image.getpixel((y, y))
        r = replace_bit(r, message_list[k])
        g = replace_bit(g, message_list[k + 1])
        b = replace_bit(b, message_list[k + 2])
        image.putpixel((y, y), (int(r, 2), int(g, 2), int(b, 2)))
        k += 3
        y += 1
    r, g, b = image.getpixel((y + 1, y))
    r = replace_bit(r, message_list[k])
    image.putpixel((y, y), (int(r, 2), g, b))
    print('Stego image in stego.jpg')
    image.save('encrypted.jpg')
    return signal

def main_procedure():
    width, height, image = transform_image()
    message_bit_list = secret_message()
    signal = key_schedule(height, width)
    signal = lsb(height, width, image, message_bit_list, signal)
    print('Signal:', signal)

if __name__ == "__main__":
    main_procedure()
