def ETF(n):
    """Euler's Totient Function."""
    count = 0
    for i in range(1, n):
        if gcd(i, n) == 1:
            count += 1
    return count

def gcd(a, b):
    """Calculate the Greatest Common Divisor."""
    while a % b != 0:
        a, b = b, (a % b)
    return b

def Primitive_Root(n):
    """Find the primitive root of a number."""
    phi = ETF(n)
    primitive_root = []
    prime_List = []
    for i in range(1, n):
        if gcd(i, n) == 1:
            prime_List.append(i)
    for j in prime_List:
        listt = []
        for k in range(1, phi + 1):
            listt.append(j**k % n)
        listt.sort()
        if listt == prime_List:
            primitive_root.append(j)
    if len(primitive_root) != 0:
        return n, primitive_root

def discrete_log(a, p, result):
    """Find the discrete logarithm."""
    ans = 0
    while (a**ans) % p != result:
        ans += 1
        continue
    return ans
