import random

# Generate a random number
rand1 = random.randint(1, 999999999)
print("Using", rand1, "as a random number")

# Generate random private keys for two users
user1_rand2 = random.randint(1, 999999999)
user2_rand2 = random.randint(1, 999999999)
private_user1 = user1_rand2 + rand1
private_user2 = user2_rand2 + rand1
print("Using", private_user1, "as a user1 private key")
print("Using", private_user2, "as a user2 private key")

# Generate public keys for both users
public_user1 = private_user1 % 256
public_user2 = private_user2 % 256
print("Using", public_user1, "as a user1 public key")
print("Using", public_user2, "as a user2 public key")

# Calculate intermediate values for shared key generation
pre_user1 = public_user2 + user1_rand2
pre_user2 = public_user1 + user2_rand2
print("Using", pre_user1, "for generating shared key user1")
print("Using", pre_user2, "for generating shared key user2")

# Generate shared keys for both users
shared_user1 = pre_user1 % 256
shared_user2 = pre_user2 % 256
print("Using", shared_user1, "as the result of user1 key gen")
print("Using", shared_user2, "as the result of user2 key gen")

# Check if the shared keys are equal
print("Checking equality...")
if shared_user1 == shared_user2:
    print("Correct! Key exchange completed!")
else:
    print("Looks like math went wrong...")
