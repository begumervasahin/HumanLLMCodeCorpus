import random
import MillerRabin
import Arithmetic

def get_prime_pair(bits=512):
    '''
    Generate a pair of primes (p, q) with:
        p of nbits and
        p < q < 2p
    '''
    assert bits % 4 == 0
    p = MillerRabin.gen_prime(bits)
    q = MillerRabin.gen_prime_range(p + 1, 2 * p)
    return p, q

def generate_keys(nbits=1024):
    '''
    Generates a key pair:
        public = (e, n)
        private = d
    such that:
        n is nbits long
        (e, n) is vulnerable to the Wiener Continued Fraction Attack
    '''
    assert nbits % 4 == 0
    p, q = get_prime_pair(nbits // 2)
    n = p * q
    phi = Arithmetic.totient(p, q)
    good_d = False
    while not good_d:
        d = random.getrandbits(nbits // 4)
        if Arithmetic.gcd(d, phi) == 1 and 36 * pow(d, 4) < n:
            good_d = True
    e = Arithmetic.mod_inverse(d, phi)
    return e, n, d
