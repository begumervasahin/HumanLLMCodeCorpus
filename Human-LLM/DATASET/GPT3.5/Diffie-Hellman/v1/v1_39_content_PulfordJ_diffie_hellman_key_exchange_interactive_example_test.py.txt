def encrypt_integer(base, prime, exponent):
    """
    Encrypts an integer using modular exponentiation.
    """
    return (base ** exponent) % prime

def test_encrypt_integer():
    """
    Test cases for the encrypt_integer function.
    """
    assert encrypt_integer(5, 23, 3) == 10
    assert encrypt_integer(5, 23, 4) == 4

def test_shared_secret_is_equal():
    """
    Test cases for verifying shared secrets.
    """
    assert encrypt_integer(encrypt_integer(5, 23, 4), 23, 3) == encrypt_integer(encrypt_integer(5, 23, 3), 23, 4)
    assert encrypt_integer(encrypt_integer(5, 23, 14), 23, 17) == encrypt_integer(encrypt_integer(5, 23, 17), 23, 14)

def test_brute_force_secret_is_equal():
    """
    Brute force test to check all possible combinations.
    """
    assert encrypt_integer(5, 23, 3) == 10
    assert encrypt_integer(5, 23, 4) == 4
    g = 5
    p = 23
    for i in range(0, p):
        for j in range(0, p):
            shared1 = encrypt_integer(encrypt_integer(g, p, i), p, j)
            shared2 = encrypt_integer(encrypt_integer(g, p, j), p, i)
            assert shared1 == shared2

if __name__ == "__main__":
    test_encrypt_integer()
    test_shared_secret_is_equal()
    test_brute_force_secret_is_equal()
    print("All tests passed successfully!")
