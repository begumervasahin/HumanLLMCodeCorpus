import constants
import generatePrivateSecret as getSecret
import generateRandomString as getString
from uuid import getnode as get_mac

def generate_message():
    mac_addr = get_mac()
    random_string_length = constants.RANDOM_STRING_LENGTH
    
    # Generate random strings
    random_string1 = getString.generateRandomString(random_string_length)
    random_string2 = getString.generateRandomString(random_string_length)
    
    # Concatenate MAC address and random string
    mac_addr_str = str(mac_addr) + random_string1
    
    # Generate private secret and Xi
    private_secret, Xi = getSecret.generatePrivateSecret()
    
    # Concatenate message components
    message = mac_addr_str + str(private_secret) + random_string2
    
    print("MAC Address:", mac_addr_str)
    print("Authentication String:", random_string2)
    print("Private Secret:", private_secret)
    
    return message, private_secret, Xi

if __name__ == "__main__":
    message, private_secret, Xi = generate_message()
    print("Generated Message:", message)
