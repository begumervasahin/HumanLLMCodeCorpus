# Input prime and generator
p = int(input("Confirm the prime: "))
g = int(input("Confirm the generator: "))

print()

# Input public keys exchanged by A and B
A = int(input("A sent: "))
B = int(input("B sent: "))

print()

# Initialize variables to store the calculated private keys
a = None
b = None

# Perform cryptanalysis to find private keys
for x in range(1, p):
    if (g ** x) % p == A:
        a = x
    if (g ** x) % p == B:
        b = x

# Output the found private keys
print("Password for A:", a)
print("Password for B:", b)
print()

# Calculate the shared secret key using the private keys
k_a = (B ** a) % p
k_b = (A ** b) % p
k = k_a  # Both keys are the same in this example, so we choose k_a

# Output the shared secret key
print("Their secret common key:", k)
print()
