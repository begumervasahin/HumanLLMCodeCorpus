def euclid(a, b):
    """Computes the greatest common divisor using Euclidean algorithm."""
    while b != 0:
        a, b = b, a % b
    return a

def extended_euclid(a, b):
    """Extended Euclidean algorithm to find modular inverse."""
    x, xx = 1, 0
    y, yy = 0, 1
    while b != 0:
        q = a // b
        a, b = b, a % b
        xx, x = x - q * xx, xx
        yy, y = y - q * yy, yy
    return a, x, y

def modular_inverse(a, n):
    """Computes the modular inverse of 'a' modulo 'n'."""
    gcd, u, v = extended_euclid(a, n)
    if gcd != 1:
        return None  # Modular inverse does not exist
    else:
        return u % n

def rsa_private_key(p, q, e):
    """Computes the RSA private key given primes 'p', 'q', and public exponent 'e'."""
    n = p * q
    phi = (p - 1) * (q - 1)
    d = modular_inverse(e, phi)
    return n, d

def rsa_encrypt(message, n, e):
    """Encrypts a message using RSA."""
    encrypted = [pow(ord(char), e, n) for char in message]
    return encrypted

def rsa_decrypt(encrypted, n, d):
    """Decrypts an RSA-encrypted message."""
    decrypted = [chr(pow(char, d, n)) for char in encrypted]
    return ''.join(decrypted)

def is_prime(number):
    """Checks if a number is prime."""
    if number <= 1:
        return False
    for divisor in range(2, int(number**0.5) + 1):
        if number % divisor == 0:
            return False
    return True

# Generate random prime numbers for p and q
p = random.randint(100000000, 999999999)
while not is_prime(p):
    p = random.randint(100000000, 999999999)

q = random.randint(100000000, 999999999)
while not is_prime(q):
    q = random.randint(100000000, 999999999)

# Choose a public exponent 'e' such that gcd(phi, e) = 1
phi = (p - 1) * (q - 1)
e = random.randint(2, phi - 1)
while euclid(phi, e) != 1:
    e = random.randint(2, phi - 1)

# Compute RSA private and public keys
public_key = (p * q, e)
private_key = rsa_private_key(p, q, e)

# Message to encrypt and sign
message_to_send = input("Enter the message to send to Michel: ")

# Encrypt the message and hash using RSA
encrypted_message = rsa_encrypt(message_to_send, *public_key)

# Decrypt the received message and signature
decrypted_message = rsa_decrypt(encrypted_message, *private_key)
decrypted_hash = rsa_decrypt(encrypted_hash, *public_key)
