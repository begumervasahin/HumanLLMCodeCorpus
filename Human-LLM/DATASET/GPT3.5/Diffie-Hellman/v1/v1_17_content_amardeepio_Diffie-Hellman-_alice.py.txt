import os
import json
from socket import *
import random
import time

print("")
print("***********************")
print("*    Diffie Hellman   *")
print("*     Key Exchange    *")
print("***********************")

def is_prime(n):
    c = 0
    for i in range(1, n):
        if n % i == 0:
            c += 1
    if c == 1:
        return True

prime = 1
while True:
    prime = random.randint(100, 1000)
    if is_prime(prime):
        break

generator = 2
while True:
    generator = random.randint(2, prime - 2)
    if is_prime(generator):
        break

print("prime =", prime)
print("generator =", generator)

a = random.randint(1, 35535)

tcp = socket(AF_INET, SOCK_STREAM)
tcp.setsockopt(SOL_SOCKET, SO_REUSEADDR, 1)
tcp.bind(('localhost', 2222))
tcp.listen(5)

temp = {'p': prime, 'base': generator}
foo = ''
bar = ''

conn, addr = tcp.accept()

while True:
    msg = conn.recv(1024)
    if b"NEGOCIATION" in msg:
        conn.send(json.dumps(temp).encode())
        print("Sending base and prime number...")
        break

A = pow(generator, a, prime)
print("The shared secret of Alice is:", A)

msg = conn.recv(1024)
foo = json.loads(msg.decode())
B = foo['B']
s = pow(B, a, prime)

temp = {'A': A}
conn.send(json.dumps(temp).encode())

print("Waiting for calculations to be completed... <<<<<>>>>>>")
time.sleep(3)

print("The secret is:", s)

msg = conn.recv(1024)
bar = json.loads(msg.decode())
secret2 = bar['S']

if s == secret2:
    print("Keys are verified successfully!")
