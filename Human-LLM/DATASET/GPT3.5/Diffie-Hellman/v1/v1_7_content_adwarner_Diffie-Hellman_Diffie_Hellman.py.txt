import math

def sieve_of_eratosthenes(n):
    """Generate primes up to n using Sieve of Eratosthenes"""
    primes = []
    sieve = [True] * (n + 1)
    for p in range(2, int(math.sqrt(n)) + 1):
        if sieve[p]:
            primes.append(p)
            for i in range(p * p, n + 1, p):
                sieve[i] = False
    return primes

def first_primitive_root(p):
    """Find the first primitive root modulo p"""
    primes = sieve_of_eratosthenes(int(round(math.sqrt(p))))
    totient = p - 1
    divisors = [d for d in primes if totient % d == 0]
    more = [totient // d for d in divisors]
    for i in range(2, p):
        prim_check = [(i ** j) % p for j in more]
        if 1 not in prim_check:
            return i

def is_prime(n):
    """Check if a number is prime"""
    return all(n % j for j in range(2, int(n ** 0.5) + 1)) and n > 1

def all_primitive_roots(p):
    """Find all primitive roots modulo p"""
    if is_prime(p):
        roots = [first_primitive_root(p)]
        a = first_primitive_root(p)
        for i in range(2, p):
            if math.gcd(i, p - 1) == 1:
                roots.append(a ** i % p)
        return roots
    else:
        return "Use a prime please"

def inverse_modulo(a, p):
    """Calculate the modular multiplicative inverse of a modulo p"""
    for d in range(1, p):
        r = (d * a) % p
        if r == 1:
            break
    else:
        raise ValueError(f'{a} has no inverse mod {p}')
    return d

def baby_step_giant_step(h, g, p):
    """Perform the baby-step giant-step algorithm to find the discrete logarithm"""
    m = math.ceil(math.sqrt(p))
    group = [(g ** i) % p for i in range(m + 1)]
    z = pow(inverse_modulo(g, p), m, p)
    store = []
    more = 0
    k = 0
    q = m
    for i in range(q):
        if any(group_elem in store for group_elem in group):
            more += i - 1
            break
        else:
            store.append((h * pow(z, i, p)) % p)
    for j, group_elem in enumerate(group):
        if group_elem == store[more]:
            k = k + j
    final_number = more * m + k
    return final_number

def diffie_hellman(p, g, a, b):
    """Perform the Diffie-Hellman key exchange"""
    A = pow(g, a, p)
    B = pow(g, b, p)
    s_a = pow(B, a, p)
    s_b = pow(A, b, p)
    return {'secret': "The common secret is " + str(s_a), 'A': A, 'B': B, 'p': p, 'g': g}
