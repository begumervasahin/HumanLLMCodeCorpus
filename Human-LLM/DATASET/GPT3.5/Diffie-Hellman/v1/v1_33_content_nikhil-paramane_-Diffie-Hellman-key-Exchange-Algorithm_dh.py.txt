from random import randint
import math

def is_prime(num, count):
    """
    Checks if a number is prime using the Miller-Rabin primality test.
    
    Args:
        num (int): The number to be checked for primality.
        count (int): The number of iterations for the primality test.
    
    Returns:
        bool: True if the number is prime, False otherwise.
    """
    if num == 1:
        return False
    if count >= num:
        count = num - 1
    for x in range(count):
        a = randint(1, num - 1)
        if pow(a, num-1, num) != 1:
            return False
    return True

def generate_big_prime(n):
    """
    Generates a big prime number of n bits using the Miller-Rabin primality test.
    
    Args:
        n (int): The number of bits for the prime number.
    
    Returns:
        int: The generated prime number.
    """
    found_prime = False
    while not found_prime:
        p = randint(2**(n-1), 2**n)
        if is_prime(p, 1000):
            return p

# Generate two large prime numbers
prime1 = generate_big_prime(2**10)
