from math import sqrt
from itertools import count, islice
import random

class DHCalculator:
    def __init__(self):
        self.private_key = 0  # Alice's private key
        self.public_key = 0  # Alice's public key
        self.shared_prime = 0  # Shared prime modulus

    def is_prime(self, n):
        """Check if a number is prime."""
        return n > 1 and all(n % i for i in islice(count(2), int(sqrt(n) - 1)))

    def generate_shared_prime(self):
        """Generate a shared prime number within a specified range."""
        min_prime = 1000
        max_prime = 10000
        primes = [p for p in range(min_prime, max_prime) if self.is_prime(p)]
        self.shared_prime = random.choice(primes)

    def set_private_key(self, private_key):
        """Set Alice's private key."""
        self.private_key = private_key

    def calculate_public_key(self, generator, shared_prime):
        """Calculate Alice's public key using the generator and shared prime."""
        self.public_key = generator ** self.private_key % shared_prime

    def calculate_shared_secret_key(self, other_public_key):
        """Calculate the shared secret key based on other party's public key."""
        return other_public_key ** self.private_key % self.shared_prime
