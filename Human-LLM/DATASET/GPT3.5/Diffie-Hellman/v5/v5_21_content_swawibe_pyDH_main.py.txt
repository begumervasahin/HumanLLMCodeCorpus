import Client
import Server
from argparse import ArgumentParser

def parse_command_line_arguments():
    """Parse command line arguments."""
    parser = ArgumentParser()
    parser.add_argument("-m", "--mode", dest="mode", type=str, required=True,
                        help="Specify CLIENT to start a client or SERVER to start a server")
    parser.add_argument("-d", "--debug", dest="debug", required=False,
                        help="Enable this option to print debug messages",
                        action="store_true")
    return parser.parse_args()

def print_debug_messages(args):
    """Print debug messages if debug flag is set."""
    if args.debug:
        print(args)

def start_client_or_server(args):
    """Start client or server based on mode."""
    mode = args.mode.lower()
    if mode == "client":
        start_client(args)
    elif mode == "server":
        start_server(args)
    else:
        print("Invalid mode. Please specify CLIENT or SERVER.")

def start_client(args):
    """Start client."""
    server_address = "127.0.0.2"
    client = Client.ClientSocket(args.debug)
    client.start_client(server_address)

def start_server(args):
    """Start server."""
    Server.start_server(args.debug)

if __name__ == '__main__':
    args = parse_command_line_arguments()
    print_debug_messages(args)
    start_client_or_server(args)
