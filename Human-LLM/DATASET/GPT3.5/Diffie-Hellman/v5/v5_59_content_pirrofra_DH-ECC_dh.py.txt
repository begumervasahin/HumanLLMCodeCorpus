import os
import sys
import xml.etree.ElementTree as ET
from curve import EllipticCurve, Point
from wnaf import WNAF

class ECC_CDH:
    def __init__(self, path, wNAF=0, w=4):
        self.load_curve_from_xml(path, wNAF, w)

    def load_curve_from_xml(self, path, wNAF, w):
        tree = ET.parse(path)
        curve = tree.getroot()
        p = int(curve.find("p").text, 10)
        a = int(curve.find("a").text, 10)
        b = int(curve.find("b").text, 10)
        n = int(curve.find("n").text, 10)
        G = curve.find("G")
        x = int(G.find("x").text, 10)
        y = int(G.find("y").text, 10)
        
        self.curve = EllipticCurve(a, b, p) if wNAF == 0 else WNAF(a, b, p, w)
        self.n = n
        self.G = Point(x, y)

    def key_pair_generator(self):
        L = self.n.bit_length() + 64
        returned_bits = os.urandom(L)
        c = int.from_bytes(returned_bits, byteorder=sys.byteorder, signed=False)
        d = (c % (self.n - 1)) + 1
        Q = self.curve.scalar_multiply(d, self.G)
        
        if Q.x == 0 and Q.y == 0:
            raise ValueError("Public Key is zero")
        elif not self.curve.is_on_curve(Q):
            raise ValueError("Invalid Point")
        
        return d, Q

    def shared_key_generator(self, d, Q):
        P = self.curve.scalar_multiply(d, Q)
        
        if P.x == 0 and P.y == 0:
            raise ValueError("P is zero")
        
        return P.x
