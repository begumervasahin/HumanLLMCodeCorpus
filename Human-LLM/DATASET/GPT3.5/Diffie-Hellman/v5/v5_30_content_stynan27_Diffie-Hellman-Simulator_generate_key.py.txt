import secrets
import random

def generate_large_prime():
    """
    Generates a large prime number suitable for use in public key cryptography.
    
    Returns:
        int: A large prime number.
    """
    # Define the lower bound for the generated prime number
    lower_bound = 10 ** 1229
    
    # Generate a random prime candidate with 4096 bits
    prime_candidate = secrets.randbits(4096)
    
    # Check if the prime candidate is greater than the lower bound and is a prime number
    while prime_candidate < lower_bound or not is_prime(prime_candidate):
        prime_candidate = secrets.randbits(4096)
    
    return prime_candidate

def is_prime(n, k=8):
    """
    Performs the Miller-Rabin primality test to determine if a number is prime.
    
    Args:
        n (int): The number to test for primality.
        k (int): The number of iterations for the Miller-Rabin test. Higher values increase accuracy.
        
    Returns:
        bool: True if the number is very likely prime, False otherwise.
    """
    # Base cases
    if n < 2:
        return False
    if n in {2, 3}:
        return True
    if n % 2 == 0:
        return False

    # Write n - 1 as 2^r * d
    d, r = n - 1, 0
    while d % 2 == 0:
        d //= 2
        r += 1

    # Witness loop
    for _ in range(k):
        a = random.randint(2, n - 2)
        x = pow(a, d, n)
        if x == 1 or x == n - 1:
            continue
        for _ in range(r - 1):
            x = pow(x, 2, n)
            if x == n - 1:
                break
        else:
            return False
    return True
