from math import floor
from temporary_client import Client


def split_length(list_length):
    """
    Splits the length of a list into two halves.
    """
    return floor(list_length / 2)


def compare_shared_secrets(clients):
    """
    Compares the shared secrets of clients.
    """
    prev_client = clients[0]
    for client in clients:
        if client.shared_secret != prev_client.shared_secret:
            return False
        prev_client = client
    return True


class KeyGenerator:
    """
    KeyGenerator class to generate new keys for clients.
    """

    def __init__(self, large_prime):
        """
        Initializes the KeyGenerator instance with a large prime number.
        """
        self.prime_modulus = large_prime

    def generate_keys(self, client_list, list_length, generator):
        """
        Generates keys for clients recursively.
        """
        if list_length == 1:
            # If there's only one client, calculate its key
            client_list[0].calculate_key(generator, self.prime_modulus)
            return
        else:
            # Split the list of clients into two halves
            left_length = split_length(list_length)
            right_length = list_length - left_length
            left_clients = client_list[:left_length]
            right_clients = client_list[left_length:]
            left_generator = generator
            right_generator = generator
            # Calculate keys for left and right halves
            for client in left_clients:
                left_generator = client.calculate_key(left_generator, self.prime_modulus)
            for client in right_clients:
                right_generator = client.calculate_key(right_generator, self.prime_modulus)
            # Recursively generate keys for subgroups
            self.generate_keys(left_clients, left_length, right_generator)
            self.generate_keys(right_clients, right_length, left_generator)


if __name__ == '__main__':
    """
    For testing purposes
    """
    # Create a list to store connected clients
    connected_clients = [Client() for _ in range(2)]
    
    # Create a KeyGenerator instance with a large prime number
    key_generator = KeyGenerator(71)
    
    # Generate keys for the connected clients
    key_generator.generate_keys(connected_clients, len(connected_clients), 3)
    
    # Check if all clients have the same shared secret
    if compare_shared_secrets(connected_clients):
        print("The shared secret {}, was correctly calculated for all clients!".format(connected_clients[0].shared_secret))
    else:
        print("The shared secret wasn't correctly calculated")
