import numpy as np
from numpy.polynomial import polynomial as poly
from math import floor, sqrt, pi
import sys

# Helper Functions
def cross_round(x, q):
    return np.floor((4 * x) / q) % 2

def mod_round(x, q):
    return np.around((2 * x) / q) % 2

def dbl(x, n, q):
    e = np.random.choice([-1, 0, 1], n, p=[0.25, 0.5, 0.25])
    dbl_x = (2 * x) % (2 * q)
    dbl_x = poly.polysub(dbl_x, e) % (2 * q)
    return dbl_x

def rec_I_0(w, q, E_left, E_right):
    I_end = round(q / 2) - 1
    bound_left = (0 + E_left) % (2 * q)
    bound_right = (I_end + E_right) % (2 * q)
    if bound_right < bound_left:
        bound_left, bound_right = bound_right, bound_left
    return 0 if bound_left <= w < bound_right else 1

def rec_I_1(w, q, E_left, E_right):
    I_start = -floor(q/2)
    bound_left = (I_start + E_left) % (2 * q)
    bound_right = (-1 + E_right) % (2 * q)
    if bound_right < bound_left:
        bound_left, bound_right = bound_right, bound_left
    return 0 if bound_left <= w < bound_right else 1

def invert(k):
    return [1 - val for val in k]

# Parameters and Random Data Generation
n = 1024
q = 2**32 - 1
sigma = 8 / sqrt(2 * pi)

A = np.floor(np.random.random(size=n) * q) % q
sA = np.floor(np.random.normal(scale=sigma, size=n))
eA = np.floor(np.random.normal(scale=sigma, size=n))
bA = poly.polymul(A, sA) % q
bA = poly.polyadd(bA, eA) % q

sB = np.floor(np.random.normal(scale=sigma, size=n))
eB = np.floor(np.random.normal(scale=sigma, size=n))
bB = poly.polymul(A, sB) % q
bB = poly.polyadd(bB, eB) % q

eeB = np.floor(np.random.normal(scale=sigma, size=n))
v = poly.polymul(bA, sB) % q
v = poly.polyadd(v, eeB) % q

# Key Generation and Processing
dbl_v = dbl(v, n, q)
c = cross_round(dbl_v, 2*q)
kB = mod_round(dbl_v, 2*q)
recA = (2 * bB) % (2 * q)
recA = poly.polymul(recA, sA) % (2 * q)
kA = [rec_I_0(recA[i], q, -q / 4, q / 4) if c[i] == 0 else rec_I_1(recA[i], q, -q / 4, q / 4) for i in range(n)]
kA = invert(kA)

# Output and Comparison
print("\n-Params---")
print(f" n: {n}")
print(f" q: {q}")
print(f" A: {A}")
print("\n-Alice---")
print(f" s: {sA}")
print(f" e: {eA}")
print(f" b: {bA}")
print("\n-Bob---")
print(f" s': {sB}")
print(f" e': {eB}")
print(f" b': {bB}")
print(f" e'': {eeB}")
print("\n-Computed data---")
print(f" v: {v}")
print(f" dbl_v: {dbl_v}")
print(f" c: {c}")
print(f" recA: {recA}")
print("\n-Generated Keys---")
print(f" Ka: {kA}")
print(f" Kb: {kB}")

# Calculate Equality Percentage
result = sum(kA[i] == kB[i] for i in range(n)) * 100 / n
print(f" Equals: {result}%")

sys.exit(0)
