from curve import EllipticCurve, Point

class WNAF(EllipticCurve):
    def __init__(self, a, b, n, w):
        super().__init__(a, b, n)
        self.window_size = w
        self.window_exp = pow(2, w)

    def mod_exp_window(self, d):
        """
        Calculate the windowed modular exponentiation.
        """
        mod_result = d % self.window_exp
        if mod_result >= self.window_exp // 2:
            return mod_result - self.window_exp
        else:
            return mod_result

    def compute_naf_representation(self, d):
        """
        Compute the Non-Adjacent Form (NAF) representation of an integer.
        """
        naf_representation = []
        while d > 0:
            if d % 2 == 1:
                d_i = self.mod_exp_window(d)
                naf_representation.append(d_i)
                d = d - d_i
            else:
                naf_representation.append(0)
            d = d // 2
        return naf_representation

    def precalculate_points(self, base_point):
        """
        Precompute points for the WNAF multiplication.
        """
        precomputed_points = {}
        i = 1
        double_base_point = self.double(base_point)
        current_point = base_point
        while i <= self.window_exp - 1:
            precomputed_points[i] = current_point
            precomputed_points[-i] = Point(current_point.x, -current_point.y)
            current_point = self.sum(current_point, double_base_point)
            i += 2
        return precomputed_points

    def wnaf_multiply(self, scalar, base_point):
        """
        Perform point multiplication using Windowed Non-Adjacent Form (WNAF).
        """
        naf_exp, naf_representation = self.compute_naf_representation(scalar)
        precomputed_points = self.precalculate_points(base_point)
        result_point = Point(0, 0)
        for exp_index in reversed(range(naf_exp)):
            result_point = self.double(result_point)
            naf_digit = naf_representation[exp_index]
            if naf_digit != 0:
                result_point = self.sum(result_point, precomputed_points[naf_digit])
        return result_point

    def multiply(self, scalar, base_point):
        """
        Wrapper method for WNAF point multiplication.
        """
        return self.wnaf_multiply(scalar, base_point)
