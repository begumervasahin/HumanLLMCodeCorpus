from __future__ import print_function

# Define shared prime and base
shared_prime = 23
shared_base = 5

# Define Alice and Bob's secrets
alice_secret = 6
bob_secret = 15

# Display publicly shared variables
print("Publicly Shared Variables:")
print("    Publicly Shared Prime:", shared_prime)
print("    Publicly Shared Base:", shared_base)

# Calculate and display Alice's public value
A = pow(shared_base, alice_secret, shared_prime)
print("\n  Alice Sends Over Public Channel:", A)

# Calculate and display Bob's public value
B = pow(shared_base, bob_secret, shared_prime)
print("Bob Sends Over Public Channel:", B)

print("\n------------\n")

# Calculate the shared secret privately by both Alice and Bob
print("Privately Calculated Shared Secret:")
alice_shared_secret = pow(B, alice_secret, shared_prime)
print("    Alice Shared Secret:", alice_shared_secret)

bob_shared_secret = pow(A, bob_secret, shared_prime)
print("    Bob Shared Secret:", bob_shared_secret)
