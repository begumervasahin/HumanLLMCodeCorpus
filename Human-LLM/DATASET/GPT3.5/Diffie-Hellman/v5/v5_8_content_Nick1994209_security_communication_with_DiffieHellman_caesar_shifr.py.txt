class CaesarCipher:
    def __init__(self, cipher_key):
        """Initialize the CaesarCipher object with the given cipher key."""
        self.cipher_key = str(cipher_key)

    def encrypt(self, phrase):
        """Encrypt the input phrase using the Caesar cipher."""
        encrypted_phrase = []
        key_index = 0

        for symbol in str(phrase):
            if key_index >= len(self.cipher_key):
                key_index = 0

            encrypted_symbol = ord(symbol) + int(self.cipher_key[key_index])
            encrypted_phrase.append(encrypted_symbol)
            key_index += 1

        return encrypted_phrase

    def decrypt(self, encrypted_phrase):
        """Decrypt the encrypted phrase using the Caesar cipher."""
        decryption_phrase = ''
        key_index = 0

        for cipher_code in encrypted_phrase:
            if key_index >= len(self.cipher_key):
                key_index = 0

            decrypted_symbol = chr(cipher_code - int(self.cipher_key[key_index]))
            decryption_phrase += decrypted_symbol
            key_index += 1

        return decryption_phrase

if __name__ == '__main__':
    # Create an instance of CaesarCipher with a main key
    main_key = 32142
    caesar_cipher = CaesarCipher(main_key)

    # Define a phrase to be encrypted
    phrase = "Hi, I'm Caesar 123"

    # Encrypt the phrase and print the encrypted result
    encrypted_phrase = caesar_cipher.encrypt(phrase)
    print("Encrypted Phrase:", encrypted_phrase)

    # Decrypt the encrypted phrase and print the decrypted result
    decrypted_phrase = caesar_cipher.decrypt(encrypted_phrase)
    print("Decrypted Phrase:", decrypted_phrase)

    # Assert that the decrypted phrase matches the original input phrase
    assert phrase == decrypted_phrase
