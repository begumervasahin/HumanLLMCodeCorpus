from rsa import RSA
from convertN2T import *

def encrypt_message(msg, private_key, modulus):
    """Encrypts the message using RSA."""
    encrypted_msg = [modExp(char, private_key, modulus) for char in msg]
    return encrypted_msg

def create_signature(msg, public_key, modulus):
    """Creates a digital signature for the message using RSA."""
    signature = [modExp(char, public_key, modulus) for char in msg]
    return signature

def decrypt_verify(msg, sender_private_key, sender_modulus, receiver_public_key, receiver_modulus):
    """Decrypts and verifies the digital signature of the received message."""
    decrypted_msg_a = [modExp(char, sender_private_key, sender_modulus) for char in msg]
    decrypted_msg_b = [modExp(char, receiver_public_key, receiver_modulus) for char in decrypted_msg_a]
    return decrypted_msg_b

def verify(msg, original_msg):
    """Verifies the received message against the original message."""
    print("Received signed message:\n\n", msg, "\n")
    print("Original message sent:\n\n", original_msg, "\n")
