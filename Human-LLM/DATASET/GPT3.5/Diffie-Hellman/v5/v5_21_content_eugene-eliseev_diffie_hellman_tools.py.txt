import random
import base64
import hashlib
import os

from Cryptodome import Random as CryptoRandom
from Cryptodome.Cipher import AES as CryptoAES
from Cryptodome.Util.Padding import pad as CryptoPad, unpad as CryptoUnpad

class PrimeGenerator:
    def __init__(self, lower_limit, upper_limit):
        # Generate a list of prime numbers within the specified range
        self.primes = [num for num in range(lower_limit, upper_limit) if self.is_prime(num)]

    def get_random_prime(self):
        # Return a random prime number from the list
        return random.choice(self.primes)

    @staticmethod
    def is_prime(num):
        # Check if a number is prime
        divisor = 2
        while divisor * divisor <= num and num % divisor != 0:
            divisor += 1
        return divisor * divisor > num

class AESCipher:
    def __init__(self, key):
        # Initialize AES cipher with a hashed key
        self.block_size = CryptoAES.block_size
        self.key = hashlib.sha256(key.encode()).digest()

    def encrypt(self, plaintext):
        # Encrypt plaintext using AES in CBC mode with PKCS#7 padding
        padded_plaintext = CryptoPad(plaintext.encode(), self.block_size)
        iv = CryptoRandom.new().read(self.block_size)
        cipher = CryptoAES.new(self.key, CryptoAES.MODE_CBC, iv)
        encrypted_data = cipher.encrypt(padded_plaintext)
        return base64.b64encode(iv + encrypted_data).decode()

    def decrypt(self, ciphertext):
        # Decrypt ciphertext encrypted with AES
        ciphertext = base64.b64decode(ciphertext)
        iv = ciphertext[:self.block_size]
        cipher = CryptoAES.new(self.key, CryptoAES.MODE_CBC, iv)
        decrypted_data = cipher.decrypt(ciphertext[self.block_size:])
        return CryptoUnpad(decrypted_data, self.block_size).decode('utf-8')

# Example usage
secret_key = 'my_secret_key'
aes_cipher = AESCipher(secret_key)

# Encrypting and decrypting a message
original_message = 'Hello, AES!'
encrypted_message = aes_cipher.encrypt(original_message)
print('Encrypted:', encrypted_message)
decrypted_message = aes_cipher.decrypt(encrypted_message)
print('Decrypted:', decrypted_message)
