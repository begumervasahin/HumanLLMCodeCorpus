import socket
import json
import threading
from diffiehellman.diffiehellman import DiffieHellman

# Initialize Diffie-Hellman for Alice
alice_dh = DiffieHellman()

# Server configuration
SERVER_IP = '192.168.168.184'
SERVER_PORT = 8201

def handle_client(connection, address):
    # Generate Alice's public key
    alice_dh.generate_public_key()

    # Initialize total data buffer
    total_data = ""

    try:
        # Set timeout for receiving data
        connection.settimeout(1)

        # Receive data from the client
        while True:
            message = connection.recv(100)
            if not message:
                break
            total_data += message.decode("utf-8")

        # Send Alice's public key to the client
        alice_public_key_message = {"alice_public_key": alice_dh.public_key}
        connection.send(json.dumps(alice_public_key_message).encode("utf-8"))

        # Receive Bob's public key from the client
        received_data = json.loads(total_data)
        bob_public_key = received_data.get("bob_public_key", None)

        # Generate the shared secret key
        if bob_public_key:
            alice_dh.generate_shared_secret(bob_public_key, echo_return_key=True)
            print("Shared Secret Key (Alice):", alice_dh.shared_key)

    except socket.error as ex:
        print("Socket error:", ex)

    finally:
        # Close the client connection
        connection.close()
        print("Client connection closed.")

# Set up the server socket
server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
server_socket.bind((SERVER_IP, SERVER_PORT))
server_socket.listen()

print(f"Server is listening on {SERVER_IP}:{SERVER_PORT}")

try:
    while True:
        # Accept client connections
        client_socket, client_address = server_socket.accept()
        print("Client connected from:", client_address)

        # Start a new thread to handle the client connection
        client_thread = threading.Thread(
            target=handle_client, args=(client_socket, client_address)
        )
        client_thread.start()

except socket.error as e:
    print("Socket error:", e)

finally:
    # Close the server socket
    server_socket.close()
    print("Server socket closed.")
