from flask import Flask, jsonify, render_template, request
import requests
from lib import rsa as RSA
from lib import ecc as ECC
from lib.dh import DH_Wrapper
from lib.fernet import FRNET_Wrapper
from lib.user import User
from lib.serial import encode, decode

# Initialize Flask app
app = Flask(__name__, template_folder='UI', static_url_path='/static', static_folder='UI/static')

# Constants for Certificate Authority host and port
CA_HOST, CA_PORT = 'localhost', 5000

# Initialize current user, mailbox, and users map
currentUser = User('', 0)
mailbox = []
usersMap = {}

# Route to receive encrypted message
@app.route("/mailbox/<ciphertext>")
def receive_message(ciphertext):
    print(ciphertext)
    mailbox.append(currentUser.decrypt(ciphertext))
    return jsonify({'status': 'Message received'})

# Route to broadcast message to all users
@app.route("/broadcast/<message>")
def broadcast_message(message):
    for user_id, user in usersMap.items():
        try:
            currentUser.send(currentUser.name + ": " + message, user)
        except Exception as e:
            print(f"Could not send to user {user.name}: {e}")
    return jsonify({'status': 'Message broadcasted'})

# Route to request RSA key securely
@app.route("/request/<name>/<int:port>")
def request_rsa_key_securely(name, port):
    currentUser.name, currentUser.port = name, port
    dh_wrapper = DH_Wrapper()
    response = requests.get(
        f'http://{CA_HOST}:{CA_PORT}/rsa/request/{dh_wrapper.y}/{dh_wrapper.modulus}/{currentUser.name}/{currentUser.port}').json()
    peer_y = int(response['peer_y'])
    shared_key = dh_wrapper.calc_shared_key(peer_y)
    fernet_wrapper = FRNET_Wrapper(shared_key)
    rsa_key_encrypted, signature = response['rsa'], response['signature']
    ca_public_key = ECC.Public(ECC.load_CA_public())
    rsa_key = fernet_wrapper.key_decrypt(rsa_key_encrypted)
    if ca_public_key.verify(decode(signature), rsa_key):
        print("Key has been verified")
    else:
        print("Key smells fishy")
    currentUser.private = RSA.Private(rsa_key)
    currentUser.public = RSA.Public(currentUser.private.deduce_public())
    return jsonify({'status': 'Keys generated'})

# Route to update users map
@app.route("/update/users", methods=['POST'])
def update_users_map():
    user_data_list = request.get_json().get('users', [])
    for user_data in user_data_list:
        usersMap[user_data['name']] = User(user_data['name'], user_data['port'], user_data['public'].encode())
    return jsonify({'status': 'Users have been updated successfully'})

# Route to fetch mailbox messages
@app.route("/fetch/mailbox")
def view_mailbox_messages():
    if request.remote_addr == '127.0.0.1':
        return jsonify({'messages': mailbox})
    else:
        print(f"--- Unauthorized access attempt --- {request.remote_addr}")
        return jsonify({'error': 'UNAUTHORIZED ACCESS'})

# Route for login page
@app.route("/")
def show_login_page():
    return render_template('login.html')

# Route for home page after login
@app.route("/home", methods=['POST'])
def show_home_page():
    name = request.form['name']
    port = request.form['port']
    request_rsa_key_securely(name, port)
    return render_template('client.html', name=currentUser.name, port=currentUser.port)

# Run the Flask app
if __name__ == "__main__":
    app.run(debug=True)
