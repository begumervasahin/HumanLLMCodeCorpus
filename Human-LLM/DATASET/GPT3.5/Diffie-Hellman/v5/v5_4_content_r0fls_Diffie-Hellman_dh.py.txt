from decimal import Decimal, getcontext
import random
import base64

# Set the precision for Decimal calculations
getcontext().prec = 100000

class KeyExchange:
    def __init__(self):
        # Define prime modulus and base for key exchange
        self.n = Decimal('1340185579782030309029142285845485748073406778702270938755484147318382420338087834406828955714187005654640257038495796545155402280055987076251704557994637589726712709889312042801858044039590155407650471667907995888292123909278046563998441725881316702608454953284969473141146885140822683049274853701491')
        self.base = (Decimal('14759984361802021245410475928101669395348791811705709117374129427051861355011151') * Decimal('5915587277')) % self.n
        
        # Generate a random secret and compute public key
        self.secret = self.generate_random_secret()
        self.public = self.modpow(self.base, self.secret, self.n)
        
        # Dictionary to store shared secrets with peers
        self.peers = dict()

    def generate_random_secret(self):
        # Generate a random secret within the range [11, base)
        return Decimal(random.randrange(11, self.base))

    def add_peer(self, peer_public):
        # Compute shared secret with a new peer and store it
        shared_secret = self.modpow(peer_public, self.secret, self.n)
        self.peers[peer_public] = shared_secret

    @staticmethod
    def modpow(base, exponent, modulus):
        # Modular exponentiation
        result = 1
        base %= modulus
        while exponent > 0:
            if exponent % 2 == 1:
                result = (result * base) % modulus
            exponent //= 2
            base = (base * base) % modulus
        return result

if __name__ == "__main__":
    # Initialize Alice and Bob for key exchange
    alice = KeyExchange()
    bob = KeyExchange()
    
    # Save Alice's and Bob's private keys to files
    with open("aliceprivate.pem", 'w') as f:
        f.write(base64.b64encode(str(alice.secret).encode()).decode())
    with open("bobprivate.pem", 'w') as f:
        f.write(base64.b64encode(str(bob.secret).encode()).decode())
    
    # Display Alice's and Bob's public keys
    print("Alice's public key:", base64.b64encode(str(alice.public).encode()).decode())
    print("Bob's public key:", base64.b64encode(str(bob.public).encode()).decode())
    
    # Perform key exchange between Alice and Bob
    bob.add_peer(alice.public)
    alice.add_peer(bob.public)
    
    # Display the shared secrets
    print("Shared secrets:")
    print("Alice:", list(alice.peers.values()))
    print("Bob:", list(bob.peers.values()))
