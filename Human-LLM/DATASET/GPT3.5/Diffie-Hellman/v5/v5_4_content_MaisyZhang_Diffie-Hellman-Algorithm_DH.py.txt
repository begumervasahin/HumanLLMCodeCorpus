from argparse import ArgumentParser

def calculate_public_key(private_key, base, prime):
    """
    Calculate the public key using the Diffie-Hellman key exchange algorithm.
    
    Args:
        private_key (int): Private key of the party.
        base (int): Base value.
        prime (int): Prime modulus value.
    
    Returns:
        int: Public key.
    """
    return pow(base, private_key, prime)

def main():
    # Parse command line arguments
    parser = ArgumentParser(description="Diffie-Hellman Key Exchange")
    parser.add_argument("-A", "--A_private_key", required=True, help="Private key of party A")
    parser.add_argument("-B", "--B_private_key", required=True, help="Private key of party B")
    parser.add_argument("-g", "--base", required=True, help="Base value")
    parser.add_argument("-p", "--prime", required=True, help="Prime modulus value")
    args = parser.parse_args()
    
    # Extract command line arguments
    A_private_key = int(args.A_private_key)
    B_private_key = int(args.B_private_key)
    base = int(args.base)
    prime = int(args.prime)
    
    # Calculate public keys for both parties
    A_public_key = calculate_public_key(A_private_key, base, prime)
    B_public_key = calculate_public_key(B_private_key, base, prime)
    
    # Calculate shared secret keys for both parties
    A_shared_key = calculate_public_key(A_private_key, B_public_key, prime)
    B_shared_key = calculate_public_key(B_private_key, A_public_key, prime)
    
    # Print the shared secret key
    print('Shared secret: {}'.format(A_shared_key))

if __name__ == '__main__':
    main()
