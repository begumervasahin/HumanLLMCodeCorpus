import binascii

def str_to_num_list(input_str):
    """Converts a string to a list of ASCII values."""
    return [num for num in input_str.encode()]

def num_list_to_str(input_list):
    """Converts a list of ASCII values to a string."""
    return bytes([min(num, 127) for num in input_list]).decode()

def num_list_to_blocks(input_list, block_size):
    """Converts a list of integers to blocks of a specified size."""
    num_blocks = (len(input_list) + block_size - 1) // block_size
    padded_list = input_list + [0] * (num_blocks * block_size - len(input_list))
    return [int.from_bytes(padded_list[i:i+block_size], 'big') for i in range(0, len(padded_list), block_size)]

def blocks_to_num_list(blocks, block_size):
    """Converts blocks to a list of integers."""
    num_list = []
    for block in blocks:
        hex_num = hex(block)[2:].rjust(block_size * 2, '0')  # Ensure consistent length
        num_list += [num for num in bytes.fromhex(hex_num)]
    return num_list
