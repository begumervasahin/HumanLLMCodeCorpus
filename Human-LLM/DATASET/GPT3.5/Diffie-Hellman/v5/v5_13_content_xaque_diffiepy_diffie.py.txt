from random import SystemRandom
import subprocess

def modular_exponentiation(base, exponent, modulus):
    if modulus == 1:
        return 0
    result = 1
    base %= modulus
    while exponent > 0:
        if exponent % 2 == 1:
            result = (result * base) % modulus
        exponent >>= 1
        base = (base * base) % modulus
    return result

def generate_prime():
    prime_found = False
    while not prime_found:
        prime = SystemRandom().randint(2 ** 511, 2 ** 512 - 1)  # Generate a random 512-bit number
        p = (prime * 2) + 1
        is_prime = subprocess.run(['openssl', 'prime', '-checks', '20', str(p)], capture_output=True, text=True)
        if is_prime.stdout.strip() == f'{p}: prime':
            prime_found = True
            return p

def generate_random_number():
    return SystemRandom().randrange(2 ** 512)  # Generate a random number in the range [0, 2^512)
