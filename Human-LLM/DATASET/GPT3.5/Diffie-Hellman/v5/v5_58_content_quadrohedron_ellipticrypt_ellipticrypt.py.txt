import random

class EllipticCurve:
    def __init__(self, p, a, b):
        """Initialize an elliptic curve with parameters p, a, and b."""
        self.p = p
        self.a = a
        self.b = b
        self.gen = None  # Generator point

    def extended_euclidean_algorithm(self, a, b):
        """Extended Euclidean algorithm to find the greatest common divisor."""
        if not b:
            return (a, 1, 0)
        d1, x1, y1 = self.extended_euclidean_algorithm(b, a % b)
        return (d1, y1, x1 - (a // b) * y1)

    def modular_inverse(self, n):
        """Calculate the modular inverse of n modulo p."""
        return (self.extended_euclidean_algorithm(n, self.p)[1]) % self.p

    def point_check(self, point):
        """Check if a point is on the curve."""
        if point == (-1, -1):
            return True  # Special case for infinity
        x, y = point
        if not ((y * y) % self.p - (x * x + self.a) % self.p * x - self.b) % self.p:    
            return True
        return False

    def point_addition(self, point1, point2):
        """Add two points on the curve."""
        x1, y1 = point1
        x2, y2 = point2
        if x1 == x2:
            return (-1, -1)  # Point at infinity
        elif x1 < 0:
            return point2
        elif x2 < 0:
            return point1
        slope = ((y2 - y1) * self.modular_inverse((x2 - x1) % self.p)) % self.p
        x3 = ((slope * slope) % self.p - x1 - x2) % self.p
        y3 = (slope * (x1 - x3) - y1) % self.p
        return (x3, y3)

    def point_double(self, point):
        """Double a point on the curve."""
        x1, y1 = point
        if x1 < 0 or y1 == 0:
            return (-1, -1)  # Point at infinity
        slope = (((3 * x1 * x1) % self.p + self.a) % self.p * self.modular_inverse((2 * y1) % self.p)) % self.p
        x2 = ((slope * slope) % self.p - 2 * x1) % self.p
        y2 = ((slope * (x1 - x2)) % self.p - y1) % self.p
        return (x2, y2)

    def scalar_multiplication(self, n, point):
        """Multiply a point on the curve by a scalar."""
        if n:
            result = point
            exp = []
            factor = 1
            while factor <= n:
                exp.append(bool(factor & n))
                factor <<= 1
            exp.reverse()
            for factor in exp[1:]:
                result = self.point_double(result)
                if factor:
                    result = self.point_addition(result, point)
            return result
        else:
            return (-1, -1)  # Point at infinity

    def generate_key_pair(self, dot=None, point=None):
        """Generate a key pair using elliptic curve cryptography."""
        if not point:
            point = self.gen
        if not dot:
            dot = random.randint(3, self.p - 1)
        return self.scalar_multiplication(dot, point)
