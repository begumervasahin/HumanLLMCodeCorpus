from Crypto.Cipher import AES

def recover_private_key(p, g, public_key):
    """Recover a private key based on a public key."""
    exponent = 1
    while pow(g, exponent, p) != public_key:
        exponent += 1
    return exponent

def decrypt_message(private_key1, private_key2, g, p, encrypted_message):
    """Decrypt a message using private keys."""
    shared_secret = pow(g, (private_key1 * private_key2), p)
    shared_secret_hex = hex(shared_secret)[2:].zfill(64)
    cipher = AES.new(bytes.fromhex(shared_secret_hex))
    decrypted_message = cipher.decrypt(encrypted_message).decode('utf-8').replace('\x00', '')
    return decrypted_message
