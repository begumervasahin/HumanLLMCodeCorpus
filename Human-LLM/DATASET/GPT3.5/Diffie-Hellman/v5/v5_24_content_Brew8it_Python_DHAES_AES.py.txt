from base64 import b64decode, b64encode
from Crypto import Random
from Crypto.Cipher import AES

class AESCipher:
    """
    AESCipher class for encryption and decryption using AES CBC mode.

    Usage:
        cipher = AESCipher('password')
        encrypted = cipher.encrypt('message')
        decrypted = cipher.decrypt(encrypted)
    """

    def __init__(self, key):
        """
        Initialize AESCipher object with the provided key.
        """
        self.key = key.encode('utf-8')

    def encrypt(self, plaintext):
        """
        Encrypt the input plaintext using AES CBC mode.
        """
        padded_plaintext = self._pad(plaintext)
        iv = Random.new().read(AES.block_size)
        cipher = AES.new(self.key, AES.MODE_CBC, iv)
        ciphertext = cipher.encrypt(padded_plaintext)
        return b64encode(iv + ciphertext)

    def decrypt(self, ciphertext):
        """
        Decrypt the input ciphertext using AES CBC mode.
        """
        decoded_ciphertext = b64decode(ciphertext)
        iv = decoded_ciphertext[:AES.block_size]
        cipher = AES.new(self.key, AES.MODE_CBC, iv)
        decrypted_plaintext = cipher.decrypt(decoded_ciphertext[AES.block_size:])
        return self._unpad(decrypted_plaintext).decode('utf-8')

    @staticmethod
    def _pad(s):
        """
        Pad the input string to make its length a multiple of the block size.
        """
        pad_length = AES.block_size - len(s) % AES.block_size
        return s + pad_length * chr(pad_length)

    @staticmethod
    def _unpad(s):
        """
        Unpad the input string by removing the padding.
        """
        return s[:-s[-1]]
