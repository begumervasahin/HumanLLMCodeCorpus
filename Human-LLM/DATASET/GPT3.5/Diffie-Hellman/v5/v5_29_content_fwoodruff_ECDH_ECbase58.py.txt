import TypeConversion as TC
import ModularOperations as mod
import ECRing

def point_to_base58(point, curve):
    """
    Convert a point to a base-58 string representation.
    """
    key58 = TC.int_to_base58(point.x)
    if point.y > curve.p / 2:
        return '03' + key58
    return '02' + key58

def base58_to_point(base58_str, curve):
    """
    Convert a base-58 string representation to a point.
    """
    a, p, b = curve.a, curve.p, curve.b
    key_x = TC.b58_to_int(base58_str[2:])
    key_y_sq = (key_x ** 3 + a * key_x + b) % p
    key_y = mod.modular_sqrt(key_y_sq, p)
    if base58_str[0] == '0' and base58_str[1] == '3':
        return ECRing.Point(key_x, p - key_y) if key_y <= curve.p / 2 else ECRing.Point(key_x, key_y)
    return ECRing.Point(key_x, key_y) if key_y > curve.p / 2 else ECRing.Point(key_x, p - key_y)
