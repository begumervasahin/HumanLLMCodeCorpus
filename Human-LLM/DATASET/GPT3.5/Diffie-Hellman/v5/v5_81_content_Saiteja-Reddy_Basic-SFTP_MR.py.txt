from random import randrange, getrandbits
import math

def is_prime(n, k=128):
    """Check if a number is prime using the Miller-Rabin primality test."""
    if n in {2, 3}:
        return True
    if n <= 1 or n % 2 == 0:
        return False
    
    # Write n as 2^r * d + 1
    r = 0
    d = n - 1
    while d % 2 == 0:
        r += 1
        d //= 2
    
    # Perform Miller-Rabin primality test
    for _ in range(k):
        a = randrange(2, n - 1)
        x = pow(a, d, n)
        
        if x != 1 and x != n - 1:
            for _ in range(r - 1):
                x = pow(x, 2, n)
                if x == 1:
                    return False
                if x == n - 1:
                    break
            else:
                return False
    return True

def generate_prime_candidate(length):
    """Generate a probable prime candidate."""
    p = getrandbits(length)
    p |= (1 << length - 1) | 1
    return p

def generate_prime_number(length=26):
    """Generate a prime number with a specified bit length."""
    candidate = 4
    while not is_prime(candidate, 128):
        candidate = generate_prime_candidate(length)
    return candidate

def find_prime_factors(num):
    """Find the prime factors of a number."""
    factors = set()
    while num % 2 == 0:
        factors.add(2)
        num //= 2
    
    for i in range(3, int(math.sqrt(num)) + 1):
        while num % i == 0:
            factors.add(i)
            num //= i
    
    if num > 2:
        factors.add(num)
    
    return factors

def get_prime_and_primitive_root():
    """Generate a prime number p and its primitive root g."""
    prime = generate_prime_number()
    phi = prime - 1
    factors_phi = find_prime_factors(phi)
    
    for i in range(2, phi + 1):
        flag = False
        for factor in factors_phi:
            if pow(i, phi // factor, prime) == 1:
                flag = True
                break
        if not flag:
            return prime, i
    
    return get_prime_and_primitive_root()
