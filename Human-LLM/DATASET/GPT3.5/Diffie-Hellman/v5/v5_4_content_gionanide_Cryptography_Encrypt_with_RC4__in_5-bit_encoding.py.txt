# Constants for encoding and decoding
ENCODING_DICT = dict(zip(
    'abcdefghijklmnopqrstuvwxyz.!?()-ABCDEFGHIJKLMNOPQRSTUVWXYZ',
    ['00000', '00001', '00010', '00011', '00100',

     '10000', '10001', '10010', '10011',
     '10100', '10101', '10110', '10111',
     '11000', '11001']
))

def xor_bits(s1, s2):
    """Perform XOR operation on two binary strings."""
    return ''.join(str(int(bit1) ^ int(bit2)) for bit1, bit2 in zip(s1, s2))

def text_encode(text):
    """Encode text using the encoding dictionary."""
    text = text[::-1]  # Reverse the text
    return ''.join(ENCODING_DICT[char] for char in text).lower()

def text_decode(binary_string):
    """Decode a binary string using the encoding dictionary."""
    inv_map = {v: k for k, v in ENCODING_DICT.items()}
    length = len(binary_string)
    decoded_text = ''
    for i in range(0, length, 5):
        decoded_text = inv_map[binary_string[i:i + 5]] + decoded_text
    return decoded_text[::-1].lower()

    # Key-scheduling algorithm (KSA)
    for i in range(256):
        j = (j + S[i] + key_bytes[i % key_length]) % 256
        S[i], S[j] = S[j], S[i]

    # Generate keystream and encrypt message
    for m in range(text_length):
        i = (i + 1) % 256
        j = (j + S[i]) % 256
        S[i], S[j] = S[j], S[i]
        k = S[(S[i] + S[j]) % 256]
        keystream_list.append(k)
        cipher_list.append(pow(k, text_bytes[m]))

    final_array = [int(xor_bits(str(c), str(k)), 2) for c, k in zip(cipher_list, keystream_list)]
