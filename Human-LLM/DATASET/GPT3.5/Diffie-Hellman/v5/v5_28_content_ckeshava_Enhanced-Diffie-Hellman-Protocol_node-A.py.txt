import socket
import constants
import generateMessage as getMsg
import getVerificationString as getS
import generatePrivateSecret as getSecret
import getPrivateSecretFromMessage as getP
import generateSharedKey as genK
import paderson_commitment
import json

def establish_connection():
    """Establishes a connection to the server."""
    socket_connection = socket.socket()
    server_port = constants.PORT
    server_ip = constants.SERVER_IP_ADDR
    socket_connection.connect((server_ip, server_port))
    print("Connection established")
    return socket_connection

def generate_message_and_commit(socket_connection):
    """Generates a message, computes commitment, and sends it to DEVICE B."""
    message, private_secret, Xa = getMsg.generateMessage()
    full_message = '0' + message
    print("Message:", full_message)

    r = paderson_commitment.generate_r((constants.SHARED_PRIME - 1) / 2)
    commitment = str(paderson_commitment.commit(int(full_message, base=36), constants.SHARED_BASE, private_secret, r, constants.SHARED_PRIME))
    print("COMMITMENT =", commitment)

    print("\nSending COMMITMENT to DEVICE B")
    socket_connection.send(commitment.encode())
    return full_message, private_secret, Xa, r

def receive_message_and_verify(socket_connection, full_message, private_secret, Xa, r):
    """Receives a message from DEVICE B, verifies authenticity, and computes the shared key."""
    private_message_B = socket_connection.recv(2048).decode()
    print("\nMESSAGE received from DEVICE B")

    print("\nExtracting PRIVATE SECRET(g^Xb) and AUTHENTICATION STRING(Nb) of DEVICE B")
    private_secret_from_B, N_b = getP.getPrivateSecretFromMessage(private_message_B)

    print("\nSending COMMIT-OPEN message to DEVICE B")
    commit_open_message = json.dumps({'message': full_message, 'r': r, 'h': private_secret})
    socket_connection.send(commit_open_message.encode())

    print("\nVerifying the AUTHENTICATION STRINGS Na and Nb")
    S = getS.getVerificationString(full_message[(-1)*constants.RANDOM_STRING_LENGTH:], N_b)
    socket_connection.send(S.encode())
    S_from_B = socket_connection.recv(2048).decode()

    if S == S_from_B:
        print("Authenticity of DEVICE B is VERIFIED\n")
        print("The SHARED KEY is:")
        print(genK.generateSharedKey(private_secret_from_B, Xa))
    else:
        print("Authenticity of DEVICE B is NOT VERIFIED\n")

def main():
    """Main function to orchestrate the communication process."""
    socket_connection = establish_connection()
    full_message, private_secret, Xa, r = generate_message_and_commit(socket_connection)
    receive_message_and_verify(socket_connection, full_message, private_secret, Xa, r)
    socket_connection.close()

if __name__ == "__main__":
    main()
