def exponential_modulo(base, exponent, modulus):
    '''Performs modular exponentiation.'''
    base = int(base)
    exponent = int(exponent)
    modulus = int(modulus)
    result = 1
    
    if exponent == 0:
        return 1
    else:
        if exponent % 2 != 0:
            result = exponential_modulo(base, exponent - 1, modulus)
            return ((base * result) % modulus)
        else:
            result = exponential_modulo(base, exponent / 2, modulus)
            return ((result * result) % modulus)
