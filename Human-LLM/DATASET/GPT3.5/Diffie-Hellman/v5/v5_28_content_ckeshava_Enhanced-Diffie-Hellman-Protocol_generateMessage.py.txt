import constants
import generatePrivateSecret as getSecret
import generateRandomString as getString
from uuid import getnode as get_mac

def generate_message():
    # Get MAC address
    mac_address = get_mac()
    
    # Obtain the length of random strings
    random_string_length = constants.RANDOM_STRING_LENGTH
    
    # Generate random strings for MAC address and authentication
    mac_address_string = str(mac_address) + getString.generateRandomString(random_string_length)
    authentication_string = getString.generateRandomString(random_string_length)
    
    # Generate private secret and Xi
    private_secret, Xi = getSecret.generatePrivateSecret()
    
    # Construct the message by concatenating components
    message = mac_address_string + str(private_secret) + authentication_string
    
    # Print message components for reference
    print("MAC Address:", mac_address_string)
    print("Authentication String:", authentication_string)
    print("Private Secret:", private_secret)
    
    return message, private_secret, Xi

if __name__ == "__main__":
    # Generate and display the message
    message, private_secret, Xi = generate_message()
    print("Generated Message:", message)
