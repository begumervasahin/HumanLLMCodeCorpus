import sys
import os
import subprocess
import re
import time
import itertools
import csv
from multiprocessing import Process, Manager, Queue

def worker(in_queue, out_list):
    """
    Worker function to process tasks from the input queue.
    """
    while True:
        line = in_queue.get()
        if line is None:
            break
        server, servername = line[1].split(',')
        servername = servername.strip('\n')
        
        try:
            # Run OpenSSL command to check DHE cipher support
            openssl_path = os.path.join(os.path.dirname(sys.argv[0]), "openssl-trace")
            cmd = subprocess.check_output([openssl_path, "s_client", "-trace", "-cipher", "DHE", "-connect", f"{server}:443"],
                                          stdin=subprocess.DEVNULL, stderr=subprocess.DEVNULL, timeout=1)
            for output_line in cmd.decode("ISO-8859-1").splitlines():
                if 'dh_p' in output_line:
                    prime = int(re.sub(".*: ", "", output_line), 16)
                    out_list.append([f"{server} {servername} {prime}"])
        except subprocess.CalledProcessError:
            out_list.append([f"{server} {servername} No_DHE"])
        except subprocess.TimeoutExpired:
            out_list.append([f"{server} {servername} Can't_connect"])
        except Exception as e:
            out_list.append([f"{server} {servername} Error"])
        
        # Add a delay to avoid excessive CPU usage
        time.sleep(1)

if __name__ == "__main__":
    num_workers = 100
    manager = Manager()
    results = manager.list()
    work = Queue(num_workers)
    pool = []
    
    with open('output.csv', 'w') as outfile:
        wr = csv.writer(outfile)
        
        # Create worker processes
        for _ in range(num_workers):
            p = Process(target=worker, args=(work, results))
            p.start()
            pool.append(p)
        
        # Read input from file and put tasks into the input queue
        with open("test") as f:
            iters = itertools.chain(f, (None,) * num_workers)
            for line in enumerate(iters):
                work.put(line)
        
        # Join worker processes
        for p in pool:
            p.join()
        
        # Write results to CSV file
        wr.writerows(results)
