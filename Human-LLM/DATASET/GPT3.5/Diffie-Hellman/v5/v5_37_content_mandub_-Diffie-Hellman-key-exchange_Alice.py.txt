from time import sleep
import random
import socket

def is_prime(n):
    """Check if a number is prime."""
    for i in range(2, n):
        if n % i == 0:
            return False
    return True

def generate_random_prime(limit=2000):
    """Generate a random prime number."""
    primes_list = [p for p in range(2, limit + 1) if is_prime(p)]
    return random.choice(primes_list)

def communicate_with_bob(num):
    """Communicate with Bob by sending a number and receiving a message."""
    from_bob = ""
    host = "localhost"
    port = 42428
    backlog = 5
    size = 1024

    # Send a number to Bob
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s_alice:
        s_alice.bind((host, port))
        s_alice.listen(backlog)

        # Accept the connection from Bob
        conn, clientaddress = s_alice.accept()
        data = conn.recv(size)
        if data:
            datastring = data.decode("utf-8")
            from_bob = datastring
        conn.close()

    sleep(0.1)  # Adding a short delay
    # Create a new socket to send data to Bob
    with socket.create_connection((host, port)) as c_alice:
        snum = str(num)
        buf = snum.encode("utf-8")
        c_alice.sendall(buf)

    return from_bob
