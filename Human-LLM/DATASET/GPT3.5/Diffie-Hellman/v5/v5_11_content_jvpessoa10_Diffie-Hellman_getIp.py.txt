import os
import socket

# Check if the operating system is not Windows
if os.name != "nt":
    # Import specific modules for non-Windows systems
    import fcntl
    import struct

    # Function to get the IP address associated with a network interface
    def get_interface_ip(ifname):
        # Create a socket object
        s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        # Use ioctl to get the IP address of the specified interface
        return socket.inet_ntoa(
            fcntl.ioctl(
                s.fileno(),
                0x8915,  # SIOCGIFADDR flag
                struct.pack('256s', ifname[:15].encode('utf-8'))  # Ensure proper encoding
            )[20:24]
        )

# Function to get the local IP address
def get_local_ip():
    # Get the IP address associated with the hostname
    ip = socket.gethostbyname(socket.gethostname())
    # If the IP address starts with "127." (loopback address) and not on Windows
    if ip.startswith("127.") and os.name != "nt":
        # List of common network interfaces to check
        interfaces = [
            "eth0",
            "eth1",
            "eth2",
            "wlan0",
            "wlan1",
            "wifi0",
            "ath0",
            "ath1",
            "ppp0",
        ]
        # Iterate through the interfaces to find a valid IP address
        for ifname in interfaces:
            try:
                ip = get_interface_ip(ifname)
                break  # Stop iterating if a valid IP is found
            except IOError:
                pass  # Ignore IOError and continue to the next interface
    return ip  # Return the obtained IP address

# Print the local IP address
print(get_local_ip())
