def modinv(a, n=p):
    """Calculate the modular inverse."""
    lm, hm = 1, 0
    low, high = a % n, n
    while low > 1:
        ratio = high // low
        nm, new = hm - lm * ratio, high - low * ratio
        lm, low, hm, high = nm, new, lm, low
    return lm % n

def ECadd(a, b):
    """Perform ECC point addition."""
    LamAdd = ((b[1] - a[1]) * modinv(b[0] - a[0], p)) % p
    x = (LamAdd * LamAdd - a[0] - b[0]) % p
    y = (LamAdd * (a[0] - x) - a[1]) % p
    return (x, y)

def EccMultiply(GenPoint, ScalarHex):
    """Perform ECC point multiplication."""
    if ScalarHex == 0 or ScalarHex >= n:
        raise Exception("Invalid Scalar/Private Key")
    ScalarBin = bin(ScalarHex)[2:]
    Q = GenPoint
    for bit in ScalarBin[1:]:
        Q = ECdouble(Q)
        if bit == "1":
            Q = ECadd(Q, GenPoint)
    return Q

def encrypt(key, plaintext, associated_data):
    """Encrypt data using AES-GCM."""
    iv = os.urandom(12)
    encryptor = Cipher(
        algorithms.AES(key),
        modes.GCM(iv),
        backend=default_backend()
    ).encryptor()
    encryptor.authenticate_additional_data(associated_data)
    ciphertext = encryptor.update(plaintext) + encryptor.finalize()
    return (iv, ciphertext, encryptor.tag)

def decrypt(key, associated_data, iv, ciphertext, tag):
    """Decrypt data using AES-GCM."""
    decryptor = Cipher(
        algorithms.AES(key),
        modes.GCM(iv, tag),
        backend=default_backend()
    ).decryptor()
    decryptor.authenticate_additional_data(associated_data)
    return decryptor.update(ciphertext) + decryptor.finalize()
