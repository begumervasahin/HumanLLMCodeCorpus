import random

def euler_totient_function(n):
    """Calculate Euler's Totient Function."""
    return sum(1 for i in range(1, n) if greatest_common_divisor(i, n) == 1)

def greatest_common_divisor(a, b):
    """Calculate the Greatest Common Divisor."""
    while b:
        a, b = b, a % b
    return a

def find_primitive_root(n):
    """Find the primitive root of a number."""
    phi = euler_totient_function(n)
    prime_list = [i for i in range(1, n) if greatest_common_divisor(i, n) == 1]
    primitive_roots = [j for j in prime_list if all(j**k % n != 1 for k in range(1, phi))]
    if primitive_roots:
        return n, primitive_roots

def discrete_logarithm(a, p, result):
    """Find the discrete logarithm."""
    ans = 0
    while pow(a, ans, p) != result:
        ans += 1
    return ans
