import socket
import ssl

def create_tcp_socket():
    """Create and open a TCP socket for the server."""
    tcp_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    tcp_socket.bind(('', 50001))
    tcp_socket.listen(1)
    return tcp_socket

def create_tls_context():
    """Create a TLS context for secure communication."""
    cert_file = "server.pem"
    key_file = "server.key"
    dh_params_file = "dh.pem"
    ciphers = "AES128-SHA256:DHE-RSA-AES128-SHA256"
    
    context = ssl.SSLContext(ssl.PROTOCOL_TLSv1_2)
    context.load_cert_chain(keyfile=key_file, certfile=cert_file)
    context.set_ciphers(ciphers)
    context.load_dh_params(dh_params_file)
    context.options = ssl.OP_SINGLE_DH_USE
    return context

def accept_connection(server_socket):
    """Accept incoming connections and perform TLS handshake."""
    print("Server listening...")
    try:
        client_socket, client_address = server_socket.accept()
    except Exception as e:
        print(f"Connection failed: {e}")
        return -1
    
    try:
        print("Connection succeeded, attempting handshake...")
        tls_context = create_tls_context()
        tls_socket = tls_context.wrap_socket(client_socket, server_side=True)
    except Exception as e:
        print(f"Handshake failed: {e}")
        return -1
    
    print(f"Handshake succeeded. Chosen cipher is {tls_socket.cipher()[0]}.")
    return tls_socket

# Create a TCP socket for the server
server_socket = create_tcp_socket()

# Continuously accept incoming connections and handle them
while True:
    tls_socket = accept_connection(server_socket)
    if tls_socket == -1:
        continue
    
    # Receive data from the client over the TLS connection
    data = tls_socket.recv(1024)
    print(f"Server receives: {data}")
