import random

def generate_random_key():
    return random.randint(1, 999999999)

def print_message(message, value):
    print(f"{message}: {value}")

def generate_private_key(random_key):
    return random_key + generate_random_key()

def calculate_public_key(private_key):
    return private_key % 256

def calculate_shared_key(public_key, private_key):
    return (public_key + private_key) % 256

def main():
    # Generate a random number for the key exchange
    random_key = generate_random_key()
    print_message("Using random number", random_key)

    # Generate random private keys for two users
    user1_private_key = generate_private_key(random_key)
    user2_private_key = generate_private_key(random_key)
    print_message("Using user1's private key", user1_private_key)
    print_message("Using user2's private key", user2_private_key)

    # Generate public keys for both users
    user1_public_key = calculate_public_key(user1_private_key)
    user2_public_key = calculate_public_key(user2_private_key)
    print_message("Using user1's public key", user1_public_key)
    print_message("Using user2's public key", user2_public_key)

    # Calculate intermediate values for shared key generation
    pre_user1 = calculate_shared_key(user2_public_key, user1_private_key)
    pre_user2 = calculate_shared_key(user1_public_key, user2_private_key)
    print_message("Using for generating user1's shared key", pre_user1)
    print_message("Using for generating user2's shared key", pre_user2)

    # Generate shared keys for both users
    shared_user1 = calculate_shared_key(pre_user1, user1_private_key)
    shared_user2 = calculate_shared_key(pre_user2, user2_private_key)
    print_message("Result of user1's key generation", shared_user1)
    print_message("Result of user2's key generation", shared_user2)

    # Check if the shared keys are equal
    print("Checking equality...")
    if shared_user1 == shared_user2:
        print("Correct! Key exchange completed successfully!")
    else:
        print("Oops! There seems to be an error in the key exchange process.")

    input("Press Enter to exit...")

if __name__ == "__main__":
    main()
