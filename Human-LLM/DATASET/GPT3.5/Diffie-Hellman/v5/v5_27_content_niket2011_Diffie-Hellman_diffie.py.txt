import random

def has_duplicates(lst):
    """Check if a list has duplicate elements."""
    return len(lst) != len(set(lst))

def main():
    # Input prime number and number of users
    prime_number = int(input("Enter the prime number: "))
    num_users = int(input("Enter the number of users: "))

    # Input secret keys for each user
    secret_keys = [int(input(f"Enter secret key for user {i + 1}: ")) for i in range(num_users)]

    # Determine which users want to communicate
    if num_users > 2:
        print("Enter the users between whom you want to communicate:")
        user1_index = int(input("Enter the user 1: "))
        xa = secret_keys[user1_index - 1]
        user2_index = int(input("Enter the user 2: "))
        xb = secret_keys[user2_index - 1]
    else:
        xa, xb = secret_keys[0], secret_keys[1]

    # Find possible values of alpha
    possible_alphas = []
    for i in range(1, prime_number):
        possible_values = [(i**j) % prime_number for j in range(1, prime_number)]
        if not has_duplicates(possible_values):
            possible_alphas.append(i)

    # Select random alpha from possible alphas
    alpha = random.choice(possible_alphas)
    print("alpha:", alpha)

    # Calculate public keys for users A and B
    ya = (alpha**xa) % prime_number
    yb = (alpha**xb) % prime_number
    print("Public key of user A:", ya)
    print("Public key of user B:", yb)

    # Calculate shared session keys for users A and B
    Kab1 = (yb**xa) % prime_number
    Kab2 = (ya**xb) % prime_number
    print("Shared session key for user A:", Kab1)
    print("Shared session key for user B:", Kab2)

if __name__ == "__main__":
    main()
