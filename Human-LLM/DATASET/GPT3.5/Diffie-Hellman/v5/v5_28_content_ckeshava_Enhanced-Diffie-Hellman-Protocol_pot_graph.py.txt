import numpy as np
import matplotlib.pyplot as plt
import csv

# Define the range of x-axis values
x_axis = list(range(10, 100, 5))

# List of CSV files containing data
data_files = ['100.csv', '140.csv', '160.csv', '180.csv', '320.csv', '384.csv', '512.csv']

# Initialize lists to store scaled data for each file
scaled_data = [[] for _ in range(len(data_files))]

# Read and process data from each CSV file
for i, file in enumerate(data_files):
    with open(file) as csv_file:
        csv_reader = csv.reader(csv_file, delimiter=',')
        # Iterate through each row in the CSV file
        for row in csv_reader:
            # Scale and append data to corresponding list
            scaled_value = (float(row[2]) - 0.1) / 1000000
            scaled_data[i].append(scaled_value)

# Plot selected data
plt.plot(x_axis, scaled_data[4], 'r--', label='320.csv')
plt.plot(x_axis, scaled_data[5], 'b--', label='384.csv')
plt.plot(x_axis, scaled_data[6], 'g--', label='512.csv')

# Add labels and legend
plt.xlabel('X Axis Label')
plt.ylabel('Y Axis Label')
plt.title('Plot Title')
plt.legend()

# Display the plot
plt.show()
