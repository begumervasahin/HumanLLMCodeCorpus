import ModularOperations as mod

class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y

def add_points(P, Q, curve):
    """
    Adds two points P and Q on an elliptic curve.
    """
    p = curve.p
    a = curve.a
    result = Point(p, p)
    
    # Check for points at infinity
    if P.x == p:
        return Q
    if Q.x == p:
        return P
    
    # Check if points P and Q are the same
    if P.x == Q.x and P.y == Q.y:
        modinv_2P = mod.modinv((2 * P.y) % p, p)
        if modinv_2P == p:
            return Point(p, p)
        else:
            s = ((3 * (P.x * P.x) + a) * modinv_2P) % p
    else:
        modinv_PQ = mod.modinv((P.x - Q.x) % p, p)
        if modinv_PQ == p:
            return Point(p, p)
        else:
            s = ((P.y - Q.y) * modinv_PQ) % p
    
    # Calculate the result of point addition
    result.x = (s * s - P.x - Q.x) % p
    result.y = (-s * s * s + s * (P.x + Q.x) - P.y + s * P.x) % p
    
    return result

def multiply(k, P, curve):
    """
    Multiplies a point P on an elliptic curve by an integer k.
    """
    p = curve.p
    temp_points = []
    
    # Handle special cases
    if k >= 0 or P.x == p:
        Q = P
    else:
        Q = Point(P.x, p - P.y)
    
    # Convert integer k to binary string
    k_binary = "{0:b}".format(abs(k))
    temp_points.append(Q)
    
    # Double the point for each bit in the binary representation of k
    for i in range(len(k_binary) - 1):
        Q = add_points(Q, Q, curve)
        temp_points.append(Q)
    
    # Compute the result using binary exponentiation
    result = Point(p, p)
    for i, bit in enumerate(k_binary[::-1]):
        if bit == '1':
            result = add_points(result, temp_points[i], curve)
    
    return result
