import random
import crypto_utils as cutils
from crypto.decryption import Decryption
from crypto.encryption import Encryption
from crypto.signature import Signature

# Initialize encryption, decryption, and signature objects
enc = Encryption()
dec = Decryption()
sign = Signature()

# Constants for permutation keys
KEY_PERMS = [56, 48, 40, 32, 24, 16, 8,
             0, 57, 49, 41, 33, 25, 17,
             9, 1, 58, 50, 42, 34, 26,
             18, 10, 2, 59, 51, 43, 35,
             62, 54, 46, 38, 30, 22, 14,
             6, 61, 53, 45, 37, 29, 21,
             13, 5, 60, 52, 44, 36, 28,
             20, 12, 4, 27, 19, 11, 3]
LEFT_PERM = [13, 16, 10, 23, 0, 4,
             2, 27, 14, 5, 20, 9,
             22, 18, 11, 3, 25, 7,
             15, 6, 26, 19, 12, 1]
RIGHT_PERM = [40, 51, 30, 36, 46, 54,
              29, 39, 50, 44, 32, 47,
              43, 48, 38, 55, 33, 52,
              45, 41, 49, 35, 28, 31]

class Utilities:
    # Dictionary mapping encryption algorithms to their respective methods
    ENC_ALGOS = {
        "rsa": enc.rsa,
        "des": enc.des,
        "otp": enc.one_time_pad,
        "el_gamal": enc.el_gamal
    }
    # Dictionary mapping decryption algorithms to their respective methods
    DEC_ALGOS = {
        "rsa": dec.rsa,
        "des": dec.des,
        "otp": dec.one_time_pad,
        "el_gamal": dec.el_gamal
    }
    # Dictionary mapping signature algorithms to their respective methods
    SIGN_ALGOS = {
        "rsa": sign.rsa,
        "el_gamal": sign.el_gamal,
        "dsa": sign.dsa
    }
    # Default host and server port
    HOST = '127.0.0.1'
    SERVER_PORT = 8080

    def __init__(self):
        pass

    @staticmethod
    def generate_keys(algo="el_gamal", prime_length=31, key=""):
        """Generate encryption keys for different algorithms."""
        if algo == "el_gamal":
            p = cutils.find_large_prime(prime_length)
            alpha = cutils.randroot(p, 2, 100)
            a = random.randint(3, p - 1)
            beta = pow(alpha, a, p)
            return {'p': p, 'alpha': alpha, 'beta': beta, 'a': a}
        elif algo == "rsa":
            e = 3
            while True:
                p = q = cutils.find_large_prime(prime_length // 2 + 1)
                while p == q:
                    p = cutils.find_large_prime(prime_length // 2 + 1)
                phi = (p - 1) * (q - 1)
                try:
                    d = cutils.modinv(e, phi)
                except:
                    e += 1
                    continue
                return {'d': d, 'e': e, 'p': p, 'q': q}
        elif algo == "dsa":
            q = cutils.find_large_prime(prime_length)
            p = 2 * q + 1
            while (p - 1) % q != 0 or not cutils.isPrime(p):
                p = p + q
                while not cutils.isPrime(p):
                    p = p + q
            g = cutils.randroot(p)
            alpha = pow(g, (p - 1) / q, p)
            while pow(alpha, q, p) != 1:
                g = cutils.randroot(p, 2, q - 1)
                alpha = pow(g, (p - 1) / q, p)
            a = random.randint(3, q - 1)
            beta = pow(alpha, a, p)
            return {'p': p, 'q': q, 'alpha': alpha, 'beta': beta, 'a': a}
        elif algo == "des":
            round_keys = []
            binary_key = bin(int(key.encode('hex'), 16))[2:]
            while len(binary_key) % 64 != 0:
                binary_key = "0" + binary_key
            perm_key = [0] * len(KEY_PERMS)
            for i in range(len(KEY_PERMS)):
                perm_key[i] = binary_key[KEY_PERMS[i]]
            c = perm_key[:28]
            d = perm_key[28:]
            for i in range(16):
                c.append(c.pop(0))
                d.append(d.pop(0))
                if i not in [0, 1, 8, 15]:
                    c.append(c.pop(0))
                    d.append(d.pop(0))
                pre_perm_key = c + d
                left_side = ""
                right_side = ""
                for i in range(len(LEFT_PERM)):
                    left_side += pre_perm_key[LEFT_PERM[i]]
                for i in range(len(RIGHT_PERM)):
                    right_side += pre_perm_key[RIGHT_PERM[i]]
                round_key = left_side + right_side
                round_keys.append(round_key)
            return {'k': round_keys}
        elif algo == 'otp':
            binary_key = ""
            for _ in range(len(key)):
                bit = random.getrandbits(1)
                binary_key += str(bit)
            return {'key': binary_key}
        else:
            return {}
