import random

def is_prime(number):
    """
    Check if a number is prime.
    
    Args:
        number (int): Number to check for primality.
    
    Returns:
        bool: True if the number is prime, False otherwise.
    """
    divisor_count = sum(1 for i in range(1, number + 1) if number % i == 0)
    return divisor_count == 2

def generate_random_prime(start, end):
    """
    Generate a random prime number in the range [start, end).
    
    Args:
        start (int): Lower bound of the range.
        end (int): Upper bound of the range.
    
    Returns:
        int: Random prime number.
    """
    primes = [number for number in range(start, end) if is_prime(number)]
    return random.choice(primes)

class Host:
    def __init__(self, base, prime, name):
        self.base = base
        self.prime = prime
        self.name = name
        self.random_sender = generate_random_prime(1, 15)
        self.key = 0
        self.partial_key = 0
        print(f"{self.name} -> Secret random prime: {self.random_sender}")
    
    def receive_calculated_key(self, partner_key):
        self.key = (partner_key ** self.random_sender) % self.prime
    
    def create_partial_key(self):
        self.partial_key = (self.base ** self.random_sender) % self.prime
        print(f"{self.name} -> Partial key: {self.partial_key}")

# Parameters
base_value = 5
prime_modulus = 23

# Host initialization
Alice = Host(base_value, prime_modulus, "Alice")
Bob = Host(base_value, prime_modulus, "Bob")

# Partial key creation
print("\nCreating partial keys...")
Alice.create_partial_key()
Bob.create_partial_key()

# Key exchange
print("\nPerforming key exchange...")
Alice.receive_calculated_key(Bob.partial_key)
Bob.receive_calculated_key(Alice.partial_key)

# Displaying secret keys
print("\nSecret keys for communication:")
print("Alice:", Alice.key)
print("Bob:", Bob.key)
