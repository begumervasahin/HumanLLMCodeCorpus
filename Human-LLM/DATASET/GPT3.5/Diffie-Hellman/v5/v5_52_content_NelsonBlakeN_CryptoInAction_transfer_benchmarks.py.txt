import time
import utilities as utils

message = "h"

def benchmark_algorithm(algo_name, min_key_size=3, max_key_size=15):
    """
    Benchmark the specified encryption algorithm.

    Args:
    - algo_name (str): Name of the encryption algorithm.
    - min_key_size (int): Minimum key size to test.
    - max_key_size (int): Maximum key size to test.
    """
    for key_size in range(min_key_size, max_key_size + 1):
        start_time = time.time()
        # Generate keys based on the specified algorithm
        keys = utils.Utilities.generate_keys(algo=algo_name, prime_length=key_size)
        
        # Encrypt the message
        ciphertext = utils.encryption_functions[algo_name](message, keys)
        
        # Decrypt the ciphertext
        plaintext = utils.decryption_functions[algo_name](ciphertext, keys)
        end_time = time.time()
        
        # Calculate and print the execution time
        execution_time = end_time - start_time
        print(f"Key size: {key_size}, Execution time: {execution_time:.6f} seconds")

def run_benchmark():
    """Run benchmark for multiple encryption algorithms."""
    print("Running benchmark...")
    algorithms = ['rsa', 'el_gamal', 'des', 'otp']
    for algo in algorithms:
        print(f"\nBenchmarking {algo} algorithm:")
        benchmark_algorithm(algo, min_key_size=3, max_key_size=20)

# Run the benchmark
run_benchmark()
