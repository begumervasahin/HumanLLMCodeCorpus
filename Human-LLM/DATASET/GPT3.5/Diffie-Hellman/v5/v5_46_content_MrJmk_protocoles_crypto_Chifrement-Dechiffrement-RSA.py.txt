import random

def sha1(data):
    """
    Returns the SHA1 sum as a 40-character hexadecimal string.
    """
    # Initialize SHA1 constants
    h0 = 0x67452301
    h1 = 0xEFCDAB89
    h2 = 0x98BADCFE
    h3 = 0x10325476
    h4 = 0xC3D2E1F0
    
    # Convert input data to binary string
    binary_str = ''.join(format(ord(char), '08b') for char in data)
    
    # Append '1' bit and padding
    bits = binary_str + '1'
    padded_bits = bits + '0' * ((448 - len(bits) % 512) % 512) + format(len(bits), '064b')
    
    # Split into 512-bit chunks
    chunks = [padded_bits[i:i+512] for i in range(0, len(padded_bits), 512)]
    
    # Process each chunk
    for chunk in chunks:
        words = [int(chunk[i:i+32], 2) for i in range(0, 512, 32)]
        
        # Extend the chunk to 80 words
        for i in range(16, 80):
            words.append(((words[i-3] ^ words[i-8] ^ words[i-14] ^ words[i-16]) << 1 | (words[i-3] ^ words[i-8] ^ words[i-14] ^ words[i-16]) >> 31) & 0xFFFFFFFF)
        
        a, b, c, d, e = h0, h1, h2, h3, h4
        
        # Main loop
        for i in range(80):
            if 0 <= i <= 19:
                f = (b & c) | ((~b) & d)
                k = 0x5A827999
            elif 20 <= i <= 39:
                f = b ^ c ^ d
                k = 0x6ED9EBA1
            elif 40 <= i <= 59:
                f = (b & c) | (b & d) | (c & d)
                k = 0x8F1BBCDC
            else:
                f = b ^ c ^ d
                k = 0xCA62C1D6
            
            temp = ((a << 5 | a >> 27) + f + e + k + words[i]) & 0xFFFFFFFF
            e, d, c, b, a = d, c, (b << 30 | b >> 2), a, temp
        
        h0 = (h0 + a) & 0xFFFFFFFF
        h1 = (h1 + b) & 0xFFFFFFFF
        h2 = (h2 + c) & 0xFFFFFFFF
        h3 = (h3 + d) & 0xFFFFFFFF
        h4 = (h4 + e) & 0xFFFFFFFF
    
    return '{:08x}{:08x}{:08x}{:08x}{:08x}'.format(h0, h1, h2, h3, h4)


def euclidean_algorithm(a, b):
    while b != 0:
        a, b = b, a % b
    return a


def extended_euclidean_algorithm(a, b):
    x, xx, y, yy = 1, 0, 0, 1
    
    while b != 0:
        q = a // b
        a, b = b, a % b
        x, xx = xx, x - q * xx
        y, yy = yy, y - q * yy
    
    return a, x, y


def modular_inverse(a, n):
    c, u, v = extended_euclidean_algorithm(a, n)
    
    if c != 1:
        return 0
    else:
        return u % n


def power(x, k, n):
    result = 1
    
    while k > 0:
        if k % 2 != 0:
            result = (result * x) % n
        x = (x * x) % n
        k //= 2
    
    return result


def generate_prime():
    while True:
        num = random.randint(100000000, 999999999)
        if is_prime(num):
            return num


def generate_keys():
    p, q = generate_prime(), generate_prime()
    n = p * q
    phi = (p - 1) * (q - 1)
    e = random.randint(2, phi - 1)
    while euclidean_algorithm(phi, e) != 1:
        e = random.randint(2, phi - 1)
    d = modular_inverse(e, phi)
    return (n, e), (n, d)


def is_prime(number):
    if number <= 3:
        return number > 1
    if number % 2 == 0 or number % 3 == 0:
        return False
    i = 5
    while i * i <= number:
        if number % i == 0 or number % (i + 2) == 0:
            return False
        i += 6
    return True


def encrypt_message(message, key):
    n, e = key
    return [power(ord(char), e, n) for char in message]


def decrypt_message(encrypted_message, key):
    n, d = key
    return ''.join(chr(power(char, d, n)) for char in encrypted_message)


# Key generation
public_key, private_key = generate_keys()
print('\nPublic key:', public_key)
print('Private key:', private_key)

# Message encryption
message_to_encrypt = input('\nEnter the message you want to send to Michel: ')
encrypted_message = encrypt_message(message_to_encrypt, public_key)
print("\nThe encrypted message to send is:", encrypted_message)

# Message decryption
decrypted_message = decrypt_message(encrypted_message, private_key)
print("\nMichel received and decrypted the message:", decrypted_message)
