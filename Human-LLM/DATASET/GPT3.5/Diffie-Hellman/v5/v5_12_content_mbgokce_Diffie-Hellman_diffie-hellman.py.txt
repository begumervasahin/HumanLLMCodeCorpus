import random
from Crypto.Util import number

# Generate a 9-digit prime number
prime = number.getPrime(9, None)

# Generate random values for g, x, and y
base = random.randint(5, 200)
alice_private = random.randint(0, prime - 1)
bob_private = random.randint(0, prime - 1)

# Calculate public keys for Alice and Bob
alice_public = pow(base, alice_private, prime)
bob_public = pow(base, bob_private, prime)

# Display parameters and public keys
print("Base (g):", base)
print("Prime (p):", prime)
print("Alice's public key:", alice_public)
print("Bob's public key:", bob_public)

# Calculate shared secrets for Alice and Bob
alice_shared_secret = pow(bob_public, alice_private, prime)
bob_shared_secret = pow(alice_public, bob_private, prime)

# Check if shared secrets match
if alice_shared_secret != bob_shared_secret:
    print("Error: Shared secrets don't match.")
else:
    print("Alice's and Bob's shared secret:", alice_shared_secret)

print("\n---Brute force---")
# Brute force to find the shared secret (for demonstration purposes)
for trial_secret in range(0, prime - 1):
    trial_public = pow(base, trial_secret, prime)
    if trial_public == alice_public:
        break

secret_found = pow(bob_public, trial_secret, prime)
print("Found shared secret using brute force:", secret_found)
