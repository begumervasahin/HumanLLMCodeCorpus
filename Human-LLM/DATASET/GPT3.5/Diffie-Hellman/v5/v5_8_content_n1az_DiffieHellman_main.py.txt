import primRoot
import sympy
import random

class PrimitiveRootFinder:
    """Finds a primitive root modulo n using the PrimRoot module."""
    def __init__(self, n):
        self.result = primRoot.primRoots(n)[0]

class DiffieHellman:
    """Implements the Diffie-Hellman key exchange."""
    def __init__(self, range_limit):
        self.range_limit = range_limit
        self.prime_number = 0
        self.generator = 0
        self.private_key_alice = 0
        self.private_key_bob = 0
        self.public_key_alice = 0
        self.public_key_bob = 0
        self.shared_secret_alice = 0
        self.shared_secret_bob = 0

    def generate_random_prime(self):
        """Generates a random prime number within the specified range."""
        self.prime_number = sympy.randprime(0, self.range_limit)
        return self.prime_number

    def initialize(self, prime):
        """Initializes the Diffie-Hellman parameters."""
        prim_root_obj = PrimitiveRootFinder(prime)
        self.generator = prim_root_obj.result
        self.private_key_alice = random.randint(0, self.prime_number)
        self.private_key_bob = random.randint(0, self.prime_number)
        self.public_key_alice = pow(self.generator, self.private_key_alice, self.prime_number)
        self.public_key_bob = pow(self.generator, self.private_key_bob, self.prime_number)
        print('Generated prime number (p):', self.prime_number)
        print('Primitive root (g):', self.generator)
        print('Alice\'s private key:', self.private_key_alice)
        print('Bob\'s private key:', self.private_key_bob)
        print('Alice\'s public key:', self.public_key_alice)
        print('Bob\'s public key:', self.public_key_bob)

    def verify_key_exchange(self):
        """Verifies the key exchange and calculates the shared secret."""
        self.shared_secret_alice = pow(self.public_key_bob, self.private_key_alice, self.prime_number)
        self.shared_secret_bob = pow(self.public_key_alice, self.private_key_bob, self.prime_number)
        if self.shared_secret_alice == self.shared_secret_bob:
            print('Key exchange verified. Shared secret:', self.shared_secret_alice)
        else:
            print('Key exchange verification failed.')
