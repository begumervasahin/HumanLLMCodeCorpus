import os
import json
from socket import *
import random
import time

# Print header
print("")

print("*     Key Exchange    *")
print("***********************")

def is_prime(n):
    """Check if a number is prime."""
    factors = [i for i in range(1, n + 1) if n % i == 0]
    return len(factors) == 2

def generate_random_prime():
    """Generate a random prime number."""
    while True:
        prime = random.randint(100, 1000)
        if is_prime(prime):
            return prime

def generate_random_generator(prime):
    """Generate a random generator."""
    while True:
        generator = random.randint(2, prime - 2)
        if is_prime(generator):
            return generator

def generate_random_secret_key():
    """Generate a random secret key."""
    return random.randint(1, 35535)

def run_server():
    """Run the server side of the key exchange."""
    prime = generate_random_prime()
    generator = generate_random_generator(prime)
    print("Prime number:", prime)
    print("Generator:", generator)

    tcp = socket(AF_INET, SOCK_STREAM)
    tcp.setsockopt(SOL_SOCKET, SO_REUSEADDR, 1)
    tcp.bind(('localhost', 2222))
    tcp.listen(5)

    conn, addr = tcp.accept()

    temp = {'p': prime, 'base': generator}
    conn.send(json.dumps(temp).encode())
    print("Sent base and prime number to Bob.")

    a = generate_random_secret_key()
    A = pow(generator, a, prime)
    print("Alice's secret key:", a)
    print("Alice's shared secret:", A)

    foo = json.loads(conn.recv(1024).decode())
    B = foo['B']

    s = pow(B, a, prime)

    temp = {'A': A}
    conn.send(json.dumps(temp).encode())
    print("Sent Alice's public key to Bob.")

    print("Waiting for calculations to be completed...")

    time.sleep(3)

    bar = json.loads(conn.recv(1024).decode())
    secret2 = bar['S']

    if s == secret2:
        print("Keys are verified successfully!")

def run_client():
    """Run the client side of the key exchange."""
    tcp = socket(AF_INET, SOCK_STREAM)
    tcp.connect(('localhost', 2222))

    foo = json.loads(tcp.recv(1024).decode())
    prime = foo['p']
    generator = foo['base']
    print("Received base and prime number from Alice.")
    print("Prime number:", prime)
    print("Generator:", generator)

    b = generate_random_secret_key()
    B = pow(generator, b, prime)
    print("Bob's secret key:", b)
    print("Bob's shared secret:", B)

    temp = {'B': B}
    tcp.send(json.dumps(temp).encode())
    print("Sent Bob's public key to Alice.")

    foo = json.loads(tcp.recv(1024).decode())
    A = foo['A']

    s = pow(A, b, prime)

    temp = {'S': s}
    tcp.send(json.dumps(temp).encode())
    print("Sent Bob's shared secret to Alice.")

if __name__ == "__main__":
    run_server()
