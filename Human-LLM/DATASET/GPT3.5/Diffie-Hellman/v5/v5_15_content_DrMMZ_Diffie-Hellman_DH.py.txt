import getpass
import re

def find_generators(p):
    """
    Find the generators of the multiplicative group Z mod p.

    Parameters:
    - p (int): The prime number.

    Prints:
    - The residues of each integer x.
    - The generators of Z mod p.
    """
    generators = set()
    for x in range(1, p):
        residues = set()
        for i in range(1, p):
            y = (x ** i) % p
            residues.add(y)
        print("Residues for <", x, ">:", residues)
        if len(residues) == p - 1:
            generators.add(x)
    print("\nGenerators of Z mod", p, "are", generators)

q = p - 1

    print("Choose an integer in [1, {}]".format(q))
    a = getpass.getpass(prompt="Enter your integer: ")
    if re.findall("[a-zA-Z]", a) or int(a) <= 0 or int(a) > q:
        print("Invalid format or value!")
    else:
        a = int(a)
        break
