import socket
from random import randint
import re

class DiffieHellmanExchange:
    def __init__(self, root, prime, recv_key):
        self.root = root
        self.prime = prime
        self.recv_key = recv_key
    
    def generate_key(self):
        """Generate a random key."""
        min_range = 10 ** 6
        max_range = (10 ** 7) - 1
        self.key = randint(min_range, max_range)
    
    def calculate_shared_key(self):
        """Calculate the shared key."""
        shared_key = (self.root ** self.key) % self.prime
        return str(shared_key)
    
    def calculate_secret_key(self):
        """Calculate the secret key."""
        secret_key = (self.recv_key ** self.key) % self.prime
        return secret_key
