import random

def generate_random_bit_string(bits):
    # Generate a random bit string with the specified number of bits
    bit_string = '1'  # Start with '1' to ensure the number is odd
    for _ in range(bits - 2):  # Generate the middle bits
        random_bit = random.SystemRandom().randrange(2)  # Generate either '0' or '1'
        bit_string += str(random_bit)  # Append the random bit to the bit string
    bit_string += '1'  # End with '1' to ensure the number is odd and greater than 1
    return bit_string

def is_prime_miller_rabin(prime_candidate):
    # Perform the Miller-Rabin primality test
    exponent = prime_candidate - 1
    shift = 0

    # Calculate the exponent and shift values
    while exponent % 2 == 0:
        exponent //= 2
        shift += 1

            return False  # Number is not prime

    return True  # Number is likely prime

def find_random_prime(bits):
    # Generate a random prime number with the specified number of bits
    while True:
        random_bit_string = generate_random_bit_string(bits)
        prime_candidate = int(random_bit_string, 2)  # Convert the bit string to an integer
        if is_prime_miller_rabin(prime_candidate):  # Check if the candidate passes the primality test
            return prime_candidate  # Return the prime number
