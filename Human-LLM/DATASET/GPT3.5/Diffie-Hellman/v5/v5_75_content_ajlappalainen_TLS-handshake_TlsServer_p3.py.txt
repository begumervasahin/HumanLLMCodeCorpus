import socket
import ssl

def open_tcp_socket(port=50001):
    """
    Open a TCP socket and start listening on the specified port.

    Args:
    - port (int): The port number to listen on.

    Returns:
    - server_socket (socket.socket): The created TCP socket.
    """
    server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server_socket.bind(('', port))
    server_socket.listen(1)
    return server_socket

def create_tls_context(cert_path, key_path, dh_path, ciphers):
    """
    Create an SSL context for TLS communication.

    Args:
    - cert_path (str): The path to the server certificate file.
    - key_path (str): The path to the server private key file.
    - dh_path (str): The path to the Diffie-Hellman parameters file.
    - ciphers (str): The list of ciphers to be used.

    Returns:
    - context (ssl.SSLContext): The SSL context object.
    """
    context = ssl.SSLContext(ssl.PROTOCOL_TLSv1_2)
    context.load_cert_chain(certfile=cert_path, keyfile=key_path)
    context.set_ciphers(ciphers)
    context.load_dh_params(dh_path)
    context.options |= ssl.OP_SINGLE_DH_USE
    context.options |= ssl.OP_NO_TICKET
    return context

def accept_connections(server_socket):
    """
    Accept incoming connections and perform TLS handshake.

    Args:
    - server_socket (socket.socket): The TCP server socket.

    Returns:
    - tls_socket (ssl.SSLSocket): The TLS socket for the accepted connection.
    """
    print("Server listening...")
    try:
        client_socket, client_address = server_socket.accept()
    except Exception as e:
        print(f"Connection failed: {e}")
        return None
    
    try:
        print("Connection succeeded, attempting handshake...")
        tls_context = create_tls_context(cert_path="server.pem", key_path="server.key", dh_path="dh.pem", ciphers="AES128-SHA256:DHE-RSA-AES128-SHA256")
        tls_socket = tls_context.wrap_socket(client_socket, server_side=True)
    except Exception as e:
        print(f"Handshake failed: {e}")
        return None
    
    print(f"Handshake succeeded. Chosen cipher is {tls_socket.cipher()[0]}.")
    return tls_socket

def main():
    """
    Main function to run the server and receive data over TLS.
    """
    server_socket = open_tcp_socket()

    while True:
        tls_socket = accept_connections(server_socket)
        if tls_socket:
            try:
                data = tls_socket.recv(1024).decode()
                print(f"Server receives: {data}")
            except Exception as e:
                print(f"Error receiving data: {e}")
            finally:
                tls_socket.close()

if __name__ == "__main__":
    main()
