import Seed_Generator
from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives.asymmetric import dh

def generate_access_key():
    # Generate Diffie-Hellman (DH) parameters
    parameters = dh.generate_parameters(generator=5, key_size=1024, backend=default_backend())
    
    # Generate private and public keys for party A
    a_private_key = parameters.generate_private_key()
    a_public_key = a_private_key.public_key()
    
    # Generate private and public keys for party B
    b_private_key = parameters.generate_private_key()
    b_public_key = b_private_key.public_key()
    
    # Perform key exchange between parties A and B
    a_shared_key = a_private_key.exchange(b_public_key)
    b_shared_key = b_private_key.exchange(a_public_key)
    
    # Check if the shared keys match
    if a_shared_key == b_shared_key:
        # If the shared keys match, generate seed using Seed_Generator module
        Seed_Generator.generate_seed(a_private_key, b_private_key, a_public_key, b_public_key)
        return a_shared_key
    else:
        # If the shared keys do not match, return False
        return False

# Generate the access key
generate_access_key()
