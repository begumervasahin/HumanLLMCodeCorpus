from random import randrange
from modular_arithmetic import modExp

def is_prime(number, num_loops=8):
    if number == 2:
        return True
    if number < 2 or number % 2 == 0:
        return False
    
    # Calculate (number - 1) / 2 and count the number of factors of 2
    remainder = (number - 1) >> 1
    while remainder % 2 == 0:
        remainder >>= 1
    
    # Perform Miller-Rabin primality test with specified number of loops
    for _ in range(num_loops):
        if is_composite(remainder, number):
            return False
    
    return True

def is_composite(remainder, number):
    # Generate a random witness in the range [1, number)
    witness = randrange(1, number)
    # Check if the witness is not a strong witness
    if modExp(witness, remainder, number) == 1:
        return False
    
    # Double the exponent until it reaches number - 1
    exponent = remainder
    while exponent < number - 1:
        if modExp(witness, exponent, number) == number - 1:
            return False
        exponent <<= 1
    
    return True
