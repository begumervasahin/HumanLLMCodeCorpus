import socket
import ssl
import sys

# Constants for key exchange modes
RSA_MODE = "RSA"
DHE_MODE = "DHE"

def create_tls_context(key_exchange_mode):
    """Create an SSL context based on the key exchange mode."""
    cert_path = "client.pem"
    key_path = "client.key"
    ciphers = "DHE-RSA-AES128-SHA256" if key_exchange_mode == DHE_MODE else "AES128-SHA256"
    
    context = ssl.SSLContext(ssl.PROTOCOL_TLSv1_2)
    context.load_cert_chain(keyfile=key_path, certfile=cert_path)
    context.set_ciphers(ciphers)
    return context

def establish_connection(server_address, server_port, key_exchange_mode):
    """Establish a TLS connection with the server using the specified key exchange mode."""
    try:
        client_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        context = create_tls_context(key_exchange_mode)
        tls_sock = context.wrap_socket(client_socket, do_handshake_on_connect=False)
        tls_sock.connect((server_address, server_port))
    except Exception as e:
        print(f"Connection failed: {e}")
        return None
    
    try:
        print("Connection succeeded, attempting handshake...")
        tls_sock.do_handshake()
    except Exception as e:
        print(f"Handshake failed: {e}")
        return None
    
    print(f"Handshake succeeded. Chosen cipher is {tls_sock.cipher()[0]}")
    return tls_sock

def get_key_exchange_mode(args):
    """Get the key exchange mode from command-line arguments."""
    if len(args) != 2:
        raise IndexError(f"Require 1 argument but {len(args)-1} were given. "
                         f"Only accept the following argument: <key_exchange_mode> = {RSA_MODE} or {DHE_MODE}")
    
    key_exchange_mode = args[1].upper()
    if key_exchange_mode not in (RSA_MODE, DHE_MODE):
        raise ValueError(f"key_exchange_mode must be either {DHE_MODE} or {RSA_MODE}, not {key_exchange_mode}")
    
    return key_exchange_mode

def main():
    """Main function to connect to the server and send a test message."""
    server_address, server_port = '127.0.0.1', 50001
    key_exchange_mode = get_key_exchange_mode(sys.argv)
    tls_socket = establish_connection(server_address, server_port, key_exchange_mode)
    
    if tls_socket:
        data = 'test_message'
        print(f'Client sends: {data}')
        tls_socket.send(data.encode())
        tls_socket.close()

if __name__ == "__main__":
    main()
