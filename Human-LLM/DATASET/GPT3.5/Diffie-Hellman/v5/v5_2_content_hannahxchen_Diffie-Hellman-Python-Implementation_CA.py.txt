import socket
from Crypto import Random
from Crypto.PublicKey import RSA

def generate_rsa_key():
    """Generate a new RSA key pair."""
    rng = Random.new().read
    return RSA.generate(1024, rng)

def create_keys():
    """Create RSA keys for the client and server."""
    client_key = generate_rsa_key()
    server_key = generate_rsa_key()
    return client_key, server_key

def export_keys(client_key, server_key):
    """Export RSA keys."""
    client_private_key = client_key.exportKey()
    client_public_key = client_key.publickey().exportKey()
    server_private_key = server_key.exportKey()
    server_public_key = server_key.publickey().exportKey()
    return {'Client': client_private_key, 'Server': server_public_key}, {'Client': client_public_key, 'Server': server_private_key}

def create_certificate_authority_socket():
    """Create a socket for the Certificate Authority (CA)."""
    ca_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    print("Socket created")
    return ca_socket

def bind_socket(socket, host, port):
    """Bind the socket to a port."""
    socket.bind((host, port))
    print("Socket bind complete")

def start_listening(socket, backlog):
    """Listen for incoming connections."""
    socket.listen(backlog)
    print("Server is now listening...")

def handle_connection(connection, client_keys, server_keys):
    """Handle an incoming connection."""
    address = connection.getpeername()[0]
    print("Connected with " + address)

    # Receive the user name
    user = connection.recv(1024)

    # Send the appropriate key based on the user name
    if user in client_keys:
        connection.sendall(client_keys[user])
        print("Sent client's key to the client")
    elif user in server_keys:
        connection.sendall(server_keys[user])
        print("Sent server's key to the client")
    else:
        print("Access denied!")

    # Close the connection
    connection.close()

def main():
    # Generate RSA keys for the client and server
    client_key, server_key = create_keys()

    # Export RSA keys
    client_keys, server_keys = export_keys(client_key, server_key)

    # Create and bind the CA socket
    ca_socket = create_certificate_authority_socket()
    bind_socket(ca_socket, 'localhost', 9999)

    # Listen for incoming connections
    start_listening(ca_socket, 5)

    # Handle incoming connections indefinitely
    while True:
        # Accept connections
        connection, _ = ca_socket.accept()
        handle_connection(connection, client_keys, server_keys)

    # Close the CA socket
    ca_socket.close()

if __name__ == "__main__":
    main()
