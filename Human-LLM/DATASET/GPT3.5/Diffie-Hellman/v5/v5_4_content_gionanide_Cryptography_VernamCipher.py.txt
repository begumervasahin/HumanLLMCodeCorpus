import binascii

def main():
    # Define the plaintext and keys
    plain_text = 'informationsecurity'
    key = 'vlaksjdhfgqodzmxncb'
    alternative_key = 'tlftrffwmixor|{xbch'

    # Print the original plaintext and keys
    print('The plain text is:', plain_text)
    print('The first key is:', key, '\n')

    # Convert the plaintext and keys to binary
    plain_text_binary = to_binary(plain_text)
    key_binary = to_binary(key)

    # Print the plaintext and keys in binary format
    print('Plain text in bits:', plain_text_binary, '\n')
    print('Key in bits:', key_binary, '\n')

    # Encrypt and decrypt with the normal key
    encrypted_text, decrypted_text = encrypt_decrypt(plain_text_binary, key_binary)
    print('XOR result for the encryption with the normal key:', encrypted_text, '\n')
    print('Encrypted text:', decrypted_text)
    print('\nDecryption result:', encrypted_text, '\n')
    print('Decrypted text with the normal key:', decrypted_text, '\n')

    # Encrypt and decrypt with the alternative key
    alternative_key_binary = to_binary(alternative_key)
    decrypted_text_alternative = xor(encrypted_text, alternative_key_binary)
    print('Decryption with the alternative key \n')
    print('Alternative key in bits:', alternative_key_binary, '\n')
    print('Decrypted text with the alternative key:', decrypted_text_alternative)

def to_binary(text):
    """
    Convert a text to its binary representation.
    
    Args:
        text (str): Input text.
    
    Returns:
        str: Binary representation of the input text.
    """
    return bin(int(binascii.hexlify(text.encode('utf-8')), 16))

def xor(m, k):
    """
    Perform XOR operation between two binary strings.
    
    Args:
        m (str): Binary string representing the message.
        k (str): Binary string representing the key.
    
    Returns:
        str: XOR result of the two binary strings.
    """
    result = []
    for i, j in zip(m, k):
        if i.isalpha() and j.isalpha():
            result.append(i)
        else:
            result.append(str(int(i) ^ int(j)))
    return "".join(result)

def encrypt_decrypt(plain_text_binary, key_binary):
    """
    Encrypt and decrypt a message using XOR operation with a given key.
    
    Args:
        plain_text_binary (str): Binary string representing the plaintext.
        key_binary (str): Binary string representing the key.
    
    Returns:
        tuple: Encrypted and decrypted text.
    """
    encryption_result = xor(plain_text_binary, key_binary)
    decrypted_result = xor(encryption_result, key_binary)
    return encryption_result, decrypted_result

if __name__ == "__main__":
    main()
