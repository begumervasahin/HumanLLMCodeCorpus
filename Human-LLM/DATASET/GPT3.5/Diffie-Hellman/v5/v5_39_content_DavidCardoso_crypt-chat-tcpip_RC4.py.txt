class RC4:
    def __init__(self, key='160'):
        self.key = key
        self.S = []  # State variable

    def initialize_state(self):
        self.S = list(range(256))  # Initialize S as a list of integers from 0 to 255
        j = 0
        key_int = self._convert_to_int(self.key)
        key_length = len(key_int)
        for i in range(256):
            j = (j + self.S[i] + key_int[i % key_length]) % 256
            self.S[i], self.S[j] = self.S[j], self.S[i]  # Swap values in S based on key scheduling algorithm

    def _generate_keystream_byte(self):
        i = (i + 1) % 256
        j = (j + self.S[i]) % 256
        self.S[i], self.S[j] = self.S[j], self.S[i]  # Swap values in S based on pseudo-random generation algorithm
        return self.S[(self.S[i] + self.S[j]) % 256]

    def _convert_to_int(self, text):
        return [ord(char) for char in text]  # Convert characters to ASCII integers

    def _convert_to_text(self, nums):
        return ''.join([chr(num) for num in nums])  # Convert ASCII integers to characters

    def encrypt(self, plaintext):
        self.initialize_state()  # Initialize RC4 state using the key
        plaintext_int = self._convert_to_int(plaintext)  # Convert plaintext to ASCII integers
        keystream = [self._generate_keystream_byte() for _ in range(len(plaintext_int))]  # Generate keystream
        encrypted_int = [plaintext_int[i] ^ keystream[i] for i in range(len(plaintext_int))]  # XOR plaintext and keystream
        return self._convert_to_text(encrypted_int)  # Convert encrypted ASCII integers back to characters

    def decrypt(self, ciphertext):
        self.initialize_state()  # Reinitialize RC4 state using the key
        ciphertext_int = self._convert_to_int(ciphertext)  # Convert ciphertext to ASCII integers
        keystream = [self._generate_keystream_byte() for _ in range(len(ciphertext_int))]  # Generate keystream
        decrypted_int = [ciphertext_int[i] ^ keystream[i] for i in range(len(ciphertext_int))]  # XOR ciphertext and keystream
        return self._convert_to_text(decrypted_int)  # Convert decrypted ASCII integers back to characters
