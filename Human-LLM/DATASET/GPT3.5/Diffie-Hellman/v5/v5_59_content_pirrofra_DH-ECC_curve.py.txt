class Point:
    """Represents a point on an elliptic curve."""

    def __init__(self, x, y):
        """Initialize the coordinates of the point."""
        self.x = x
        self.y = y


class EllipticCurve:
    """Represents an elliptic curve in the form y^2 = x^3 + ax + b (mod p)."""

    def __init__(self, a, b, p):
        """Initialize the parameters of the elliptic curve."""
        self.a = a
        self.b = b
        self.p = p

    def multiplicative_inverse(self, x):
        """Compute the multiplicative inverse using the extended Euclidean algorithm."""
        return pow(x, -1, self.p)

    def double_and_add(self, scalar, P):
        """Perform scalar multiplication using the double-and-add algorithm."""
        result = Point(0, 0)
        while scalar != 0:
            if scalar & 1:  # Check if the least significant bit is set
                result = self.point_add(result, P)
            P = self.double_point(P)
            scalar >>= 1  # Right shift to divide by 2
        return result

    def double_point(self, p):
        """Double a point on the elliptic curve."""
        if p.x == 0 and p.y == 0:
            return Point(0, 0)
        lmb_numerator = (3 * p.x ** 2 + self.a) % self.p
        lmb_denominator = (2 * p.y) % self.p
        lmb = (lmb_numerator * self.multiplicative_inverse(lmb_denominator)) % self.p
        xr = (lmb ** 2 - 2 * p.x) % self.p
        yr = (lmb * (p.x - xr) - p.y) % self.p
        return Point(xr, yr)

    def point_add(self, p, q):
        """Add two points on the elliptic curve."""
        if p.x == q.x and p.y == q.y:
            return self.double_point(p)
        if p.x == 0 and p.y == 0:
            return q
        if q.x == 0 and q.y == 0:
            return p
        lmb_numerator = (q.y - p.y) % self.p
        lmb_denominator = (q.x - p.x) % self.p
        if lmb_denominator == 0:
            return Point(0, 0)
        lmb = (lmb_numerator * self.multiplicative_inverse(lmb_denominator)) % self.p
        xr = (lmb ** 2 - p.x - q.x) % self.p
        yr = (lmb * (p.x - xr) - p.y) % self.p
        return Point(xr, yr)

    def is_on_curve(self, P):
        """Check if a point is on the elliptic curve."""
        return (P.y ** 2) % self.p == ((P.x ** 3 + self.a * P.x + self.b) % self.p)

    def scalar_multiply(self, scalar, P):
        """Perform scalar multiplication of a point on the elliptic curve."""
        return self.double_and_add(scalar, P)
