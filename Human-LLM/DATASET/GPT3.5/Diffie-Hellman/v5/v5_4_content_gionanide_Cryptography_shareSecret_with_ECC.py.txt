def main():
    alice_public_key = get_public_key()
    alice_private_key = get_private_key()
    
    print('Send message procedure begins')
    
    flag = True
    
    results = send_message(int(alice_public_key[3]), int(alice_public_key[4]), int(alice_public_key[3]),

def get_public_key():
    a = input('Enter the "a" factor: ')
    b = input('Enter the "b" factor: ')
    p = input('Enter the prime number "p": ')
    x, y = map(int, input('Enter the initial point in the format "xy": ').split())

    print('\n')
    print('The curve equation as a PUBLIC KEY: (y^2 = x^3 + {}x + {}) mod {}'.format(a, b, p))
    print('Initial point: ({},{}) \n'.format(x, y))

    return a, b, p, x, y


def get_private_key():
    multiplier = input('Enter the multiplier (private key): ')
    print('Multiplier (private key): {}\n'.format(multiplier))

    return multiplier


def multiplicative_inverse(n):
    x = 1
    while not (n * x % 17 == 1):
        x += 1
    return x


def calculate_point(x1, y1, x2, y2, a, p, flag, i):
    if x1 == x2:
        if y1 == y2:
            if flag:
                divide = multiplicative_inverse(2 * y1)
                s = (((3 * x1 ** 2 + a) % p) * divide) % p
                flag = False
        else:
            s = 0
    else:
        divide1 = multiplicative_inverse(x1 - x2)
        s = ((y1 - y2) % p * divide1) % p

    x3 = ((s ** 2) % p - (x1 + x2) % p) % p
    y3 = (((s * (x1 - x3)) % p) % p - y1) % p

    print('Round {}: ({},{})'.format(i, x3, y3))

    return x2, y2, x3, y3


def send_message(x1, y1, x2, y2, m, a, p, flag):
    variables = calculate_point(x1, y1, x2, y2, a, p, flag, 1)
    if m - 1 <= 1:
        return variables[2], variables[3]
    else:
        for i in range(1, m):
            variables = calculate_point(variables[2], variables[3], variables[0], variables[1], a, p, flag, i + 1)
        return variables[2], variables[3]

if __name__ == "__main__":
    main()
