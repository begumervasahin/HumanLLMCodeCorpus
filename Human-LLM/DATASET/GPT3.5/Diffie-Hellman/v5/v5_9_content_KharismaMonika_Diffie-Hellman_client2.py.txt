import socket
from desv4 import *
from desv4c import *

# Define keys and parameters
KEY_A = 3
KEY_Q = 353

def generate_key(base, exponent, modulus):
    return (base ** exponent) % modulus

def receive_server_key(socket):
    return socket.recv(10)

def adjust_key_length(key, target_length):
    key_str = str(key)
    if len(key_str) < target_length:
        key_str += ''.join(str(i) for i in range(len(key_str), target_length))
    return key_str

def exchange_keys(socket, random_key):
    server_key = receive_server_key(socket)
    print("Received server key:", server_key)
    
    computed_key = generate_key(int(server_key), random_key, KEY_Q)
    print("Computed key:", computed_key)
    
    computed_key_str = adjust_key_length(computed_key, 8)
    print("Adjusted key:", computed_key_str)
    
    with open('key.txt', 'wb') as f:
        f.write(computed_key_str.encode())

def send_message(socket, message):
    with open('plain.txt', 'wb') as f:
        f.write(message.encode())
        
    exec(open("desv4.py").read())
    with open('hasil.txt', 'rb') as file:
        encrypted_data = file.read()
    
    print('Sending message:', message)
    socket.sendall(encrypted_data)

def receive_message(socket):
    data = socket.recv(150)
    amount_received = 0
    amount_expected = len(data)
    
    while amount_received < amount_expected:
        print('Received encrypted data:', data)
        with open('terima2.txt', 'wb') as f:
            f.write(data)
        
        exec(open("desv4c.py").read())
        with open('dekrip2.txt', 'rb') as file:
            decrypted_data = file.read()
        
        amount_received += len(decrypted_data)
        print('Decrypted message:', decrypted_data.decode())

def main():
    # Take input for random key
    random_key = input('Enter a random key: ')
    key_result = generate_key(KEY_A, random_key, KEY_Q)
    print("Generated key:", key_result)

    checkpoint = 0

    while True:
        checkpoint += 1
        try:
            # Create a socket
            with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as sock:
                server_address = ('localhost', 11000)
                
                # Connect to the server
                sock.connect(server_address)
                
                # Exchange keys with the server
                if checkpoint == 1:
                    exchange_keys(sock, random_key)
                
                # Get user input message
                message = input('Faishal : ')
                
                # Send message to the server
                send_message(sock, message)
                
                # Receive encrypted data from the server
                receive_message(sock)
        except Exception as e:
            print("An error occurred:", e)

if __name__ == "__main__":
    main()
