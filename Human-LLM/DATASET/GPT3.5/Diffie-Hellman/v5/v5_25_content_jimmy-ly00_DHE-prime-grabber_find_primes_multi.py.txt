import sys
import os
import subprocess
import re
import csv
from multiprocessing import Pool

# Constants
FILE = "alexa_top1mil"
WORKERS = 80

def process_line(line):
    """
    Process a line from the input file to extract Diffie-Hellman parameters.
    """
    row = line.split(',')
    server = row[0]
    servername = row[1].strip('\n')

    try:
        # Execute openssl-trace command to check Diffie-Hellman parameters
        cmd = subprocess.check_output([
            os.path.join(os.path.dirname(sys.argv[0]), "openssl-trace"),
            "s_client", "-trace",
            "-cipher", "DHE",
            "-connect", f"{server}:443"
        ], stdin=subprocess.DEVNULL, stderr=subprocess.DEVNULL, timeout=1)

        # Process the output to extract Diffie-Hellman prime parameter
        for line in cmd.decode("ISO-8859-1").splitlines():
            if 'dh_p' in line:
                prime = int(re.sub(".*: ", "", line), 16)
                return f'{server} {servername} {prime}'

    except subprocess.CalledProcessError:
        return f'{server} {servername} No_DHE'

    except subprocess.TimeoutExpired:
        return f'{server} {servername} Can\'t_connect'

    except Exception as e:
        return f'{server} {servername} Error: {str(e)}'

if __name__ == "__main__":
    # Open the output CSV file for writing
    with open('output.csv', 'w', newline='') as outfile:
        wr = csv.writer(outfile)

        # Use multiprocessing to process lines in parallel
        pool = Pool(WORKERS)
        with open(FILE) as f:
            result = pool.map(process_line, f)

        # Write the processed results to the output CSV file
        for item in result:
            wr.writerow([item])
