import random
import string

class AESCipher:
    """Mock implementation of AESCipher for demonstration purposes."""
    def __init__(self, key):
        self.key = key

    def encrypt(self, data):
        """Encrypt data by prepending 'encrypted_'."""
        return "encrypted_" + data

    def decrypt(self, data):
        """Decrypt data by removing 'encrypted_'."""
        return data.replace("encrypted_", "")

class Person:
    """Represents a person with encryption capabilities."""
    def __init__(self, prime_number, modulus):
        self.prime_number = prime_number
        self.modulus = modulus
        self.keys = {}  # Stores encryption keys
        self.messages = {}  # Stores decrypted messages

    def start_communication_with(self, other):
        """Simulates communication with another person or entity."""
        return "sender", "receiver", "message", "Hello, this is a message!"

    def encode_message(self, cipher, message):
        """Encrypts a message using the given cipher."""
        return cipher.encrypt(message)

    def decode_message(self, cipher, encoded_message):
        """Decrypts an encoded message using the given cipher."""
        return cipher.decrypt(encoded_message)

class Hacker(Person):
    """Represents a hacker inheriting from Person."""
    def read_data(self, data_tuple):
        """Reads and processes incoming data."""
        sender, receiver, action, data = data_tuple
        if action == "generate_key":
            # Generate a key based on received data
            generated_key = (int(data) ** self.prime_number) % self.modulus
            self.keys[sender] = AESCipher(str(generated_key))

            # Simulate communication and return result
            _, _, action, data = self.start_communication_with("")
            return sender, receiver, action, data
        elif sender in self.keys and receiver in self.keys:
            # Decrypt data if both sender and receiver have keys
            decrypted_data = self.decode_message(self.keys[sender], data)

            # Retrieve stored message if decrypted data matches
            if decrypted_data in self.messages:
                decrypted_data = self.messages[decrypted_data]

            # Encode decrypted data using receiver's key
            encoded_data = self.encode_message(self.keys[receiver], decrypted_data)
            return sender, receiver, "message", encoded_data

        # Return original data tuple if no specific action is taken
        return data_tuple
