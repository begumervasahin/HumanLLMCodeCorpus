    "generator": 2

class DiffieHellman:
    """Class to represent the Diffie-Hellman key exchange protocol."""

    def __init__(self):
        self.p = PRIMES["prime"]
        self.g = PRIMES["generator"]
        self.private_key = int(binascii.hexlify(os.urandom(32)), base=16)

    def get_private_key(self):
        """Get the private key."""
        return self.private_key

    def generate_public_key(self):
        """Generate the public key."""
        return pow(self.g, self.private_key, self.p)

    def generate_shared_key(self, other_party_public_key):
        """Generate the shared key."""
        try:
            shared_secret = pow(other_party_public_key, self.private_key, self.p)
            return hashlib.sha256(str(shared_secret).encode()).hexdigest()
        except Exception as e:
            print("Error during shared key generation:", e)
