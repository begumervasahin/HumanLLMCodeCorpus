import string

def is_prime(n):
    """Check if a number is prime."""
    if n <= 1:
        return False
    if n <= 3:
        return True
    if n % 2 == 0 or n % 3 == 0:
        return False
    
    # Check divisibility with numbers of the form 6k ± 1 up to sqrt(n)
    for i in range(5, int(n**0.5) + 1, 6):
        if n % i == 0 or n % (i + 2) == 0:
            return False
    return True

def get_shared_prime():
    """Prompt user to input a large prime number."""
    while True:
        shared_prime = int(input('Enter a large prime number to share: '))
        if is_prime(shared_prime):
            return shared_prime
        print(shared_prime, 'is not a prime number')

def get_shared_generator(shared_prime):
    """Prompt user to input a shared generator."""
    while True:
        shared_generator = int(input(f'Enter a shared generator less than {shared_prime}: '))
        if shared_generator < shared_prime:
            return shared_generator
        print(f'{shared_generator} is larger than {shared_prime}')

def get_secret_key(person):
    """Prompt user to input a secret key for a person."""
    return int(input(f'Enter a secret key for person {person}: '))

def calculate_shared_secret(secret, other_party, shared_prime):
    """Calculate the shared secret."""
    return pow(other_party, secret, shared_prime)

def encrypt_message(plaintext, shared_secret):
    """Encrypt the message using the shared secret."""
    alphabet = string.ascii_uppercase
    encrypted_message = ''
    for char in plaintext:
        if char in alphabet:
            char_index = alphabet.index(char)
            encrypted_index = (char_index + shared_secret) % 26
            encrypted_message += alphabet[encrypted_index]
    return encrypted_message

def main():
    """Main function to execute the encryption process."""
    plaintext = input('Enter the message you want to encrypt: ').replace(' ', '').upper()
    shared_prime = get_shared_prime()
    shared_generator = get_shared_generator(shared_prime)
    secret_A = get_secret_key('A')
    secret_B = get_secret_key('B')
    
    A = pow(shared_generator, secret_A, shared_prime)
    B = pow(shared_generator, secret_B, shared_prime)
    
    print('Person A sends over public channel:', A)
    print('Person B sends over public channel:', B)
    
    shared_secret_A = calculate_shared_secret(secret_A, B, shared_prime)
    shared_secret_B = calculate_shared_secret(secret_B, A, shared_prime)
    
    print('Person A\'s Shared Secret:', shared_secret_A)
    print('Person B\'s Shared Secret:', shared_secret_B)
    
    print('\n``````````The Key has been chosen```````````\n')
    
    encrypted_message = encrypt_message(plaintext, shared_secret_A)
    print('Encrypted Message:', encrypted_message)

if __name__ == "__main__":
    main()
