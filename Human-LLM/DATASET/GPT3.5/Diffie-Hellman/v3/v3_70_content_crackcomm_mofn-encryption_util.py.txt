import base64
from cryptography.hazmat.primitives import serialization
from cryptography.hazmat.primitives.asymmetric.x25519 import X25519PrivateKey

def private_bytes_to_raw(private_key):
    """Converts a private key object to raw bytes."""
    return private_key.private_bytes(
        encoding=serialization.Encoding.Raw,
        format=serialization.PrivateFormat.Raw,
        encryption_algorithm=serialization.NoEncryption()
    )

def public_bytes_to_raw(private_key):
    """Converts a private key object to raw public key bytes."""
    return private_key.public_key().public_bytes(
        encoding=serialization.Encoding.Raw,
        format=serialization.PublicFormat.Raw
    )

def perform_key_exchange(private_key, public_key):
    """Performs key exchange using the private and public keys."""
    private_key_obj = X25519PrivateKey.from_private_bytes(private_key)
    return private_key_obj.exchange(public_key)

def print_keys(private_key):
    """Prints the base64 encoded private and public keys."""
    private_key_raw = private_bytes_to_raw(private_key)
    public_key_raw = public_bytes_to_raw(private_key)
    
    private_key_encoded = base64.b64encode(private_key_raw).decode()
    public_key_encoded = base64.b64encode(public_key_raw).decode()
    
    print('Private Key:', private_key_encoded)
    print('Public Key:', public_key_encoded)

# Example usage:
private_key = X25519PrivateKey.generate()
print_keys(private_key)
