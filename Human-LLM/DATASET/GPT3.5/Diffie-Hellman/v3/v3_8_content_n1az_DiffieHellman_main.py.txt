import sympy
import random

class PrimitiveRoot:
    """Find a primitive root modulo n."""
    def __init__(self, n):
        self.result = sympy.primitive_root(n)

class DiffieHellman:
    """Implement the Diffie-Hellman key exchange."""
    def __init__(self, p_range):
        self.p_range = p_range
        self.prime_n = 0
        self.generator = 0
        self.private_key_A = 0
        self.private_key_B = 0
        self.public_key_A = 0
        self.public_key_B = 0
        self.shared_secret_A = 0
        self.shared_secret_B = 0
    
    def generate_random_prime(self):
        """Generate a random prime number."""
        self.prime_n = sympy.randprime(0, self.p_range)
        return self.prime_n
    
    def initialize(self, prime):
        """Initialize the Diffie-Hellman parameters."""
        prim_root_obj = PrimitiveRoot(prime)
        self.generator = prim_root_obj.result
        self.private_key_A = random.randint(0, self.prime_n)
        self.private_key_B = random.randint(0, self.prime_n)
        self.public_key_A = pow(self.generator, self.private_key_A, self.prime_n)
        self.public_key_B = pow(self.generator, self.private_key_B, self.prime_n)
        print('Generated prime number (p):', self.prime_n)
        print('Primitive root (g):', self.generator)
        print('Alice\'s private key (a):', self.private_key_A)
        print('Bob\'s private key (b):', self.private_key_B)
        print('Alice\'s public key (A):', self.public_key_A)
        print('Bob\'s public key (B):', self.public_key_B)
    
    def calculate_shared_secret(self):
        """Calculate the shared secret."""
        self.shared_secret_A = pow(self.public_key_B, self.private_key_A, self.prime_n)
        self.shared_secret_B = pow(self.public_key_A, self.private_key_B, self.prime_n)
    
    def verify_key_exchange(self):
        """Verify the key exchange."""
        if self.shared_secret_A == self.shared_secret_B:
            print('Key exchange verified. Shared secret:', self.shared_secret_A)
        else:
            print('Key exchange verification failed.')
