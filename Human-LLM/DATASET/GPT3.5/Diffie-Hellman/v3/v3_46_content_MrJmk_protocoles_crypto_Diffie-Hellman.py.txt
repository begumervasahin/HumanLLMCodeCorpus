import random

def is_prime(number):
    """
    Check if a number is prime.

    Args:
        number (int): The number to check.

    Returns:
        bool: True if the number is prime, False otherwise.
    """
    if number <= 3:
        return number > 1
    if number % 2 == 0 or number % 3 == 0:
        return False
    for i in range(5, int(number ** 0.5) + 1, 6):
        if number % i == 0 or number % (i + 2) == 0:
            return False
    return True

def generate_prime(lower_bound, upper_bound):
    """
    Generate a prime number between n and m.

    Args:
        lower_bound (int): The lower bound of the range.
        upper_bound (int): The upper bound of the range.

    Returns:
        int: The generated prime number.
    """
    prime = 0
    while not is_prime(prime):
        prime = random.randint(lower_bound, upper_bound)
    return prime
