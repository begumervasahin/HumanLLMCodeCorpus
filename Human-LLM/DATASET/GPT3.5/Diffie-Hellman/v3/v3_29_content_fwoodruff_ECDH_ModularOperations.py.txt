# Function to calculate the Greatest Common Divisor (GCD) of two numbers
def gcd(a, b):
    '''
    Calculates the Greatest Common Divisor of numbers a and b
    '''
    if a == 0:
        return b, 0, 1
    else:
        gcd_val, x, y = gcd(b % a, a)
        return gcd_val, x - (b // a) * y, y

# Function to find the modular inverse of 'a' modulo 'm'
def mod_inverse(a, m):
    '''
    Calculates the inverse of 'a' modulo 'm'
    '''
    gcd_val, x, y = gcd(a, m)
    if gcd_val != 1:
        return None  # Modular inverse does not exist
    else:
        return x % m

# Function to find the modular square root of 'a' modulo 'p'
def mod_sqrt(a, p):
    if a == p:
        return a

    # Function to calculate the Legendre symbol
    def legendre_symbol(a, p):
        ls = pow(a, (p - 1) // 2, p)
        return -1 if ls == p - 1 else ls

    # Handling special cases
    if legendre_symbol(a, p) != 1:
        return 0
    elif a == 0:
        return 0
    elif p == 2:
        return p
    elif p % 4 == 3:
        return pow(a, (p + 1) // 4, p)

    s = p - 1
    e = 0
    while s % 2 == 0:
        s //= 2
        e += 1

    n = 2
    while legendre_symbol(n, p) != -1:
        n += 1

    x = pow(a, (s + 1) // 2, p)
    b = pow(a, s, p)
    g = pow(n, s, p)
    r = e

    while True:
        t = b
        m = 0
        for m in range(r):
            if t == 1:
                break
            t = pow(t, 2, p)

        if m == 0:
            return x

        gs = pow(g, 2 ** (r - m - 1), p)
        g = (gs * gs) % p
        x = (x * gs) % p
        b = (b * g) % p
        r = m

# Example usage:
a = 10
p = 13
print("Modular Square Root of", a, "modulo", p, "is:", mod_sqrt(a, p))
