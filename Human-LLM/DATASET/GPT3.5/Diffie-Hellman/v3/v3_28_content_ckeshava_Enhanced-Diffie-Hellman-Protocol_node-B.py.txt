import socket
import constants
import generateMessage as getMsg
import getVerificationString as getS
import getPrivateMessageFromNodeA as f
import getPrivateSecretFromMessage as getP
import generateSharedKey as genK

def main():
    # Create a server socket
    server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    print("Server socket successfully created")

    # Bind the server socket to the specified port
    port = constants.PORT
    server_socket.bind(('', port))
    print("Server socket bound to port", port)

    # Listen for incoming connections
    server_socket.listen(5)
    print("Server socket is listening")

    # Accept connection from NODE A
    client_socket, addr = server_socket.accept()
    print('Got connection from', addr)

    # Generate message, private secret, and Xb
    msg, private_secret, Xb = getMsg.generateMessage()
    full_message = '1' + msg
    print("Message:", full_message)

    # Receive COMMITMENT message from NODE A
    print("\nCOMMITMENT message received from DEVICE A")
    commit_value_from_A = int(client_socket.recv(2048).decode())

    # Send MESSAGE to NODE A
    print("\nMESSAGE sent to DEVICE A")
    client_socket.send(full_message.encode())

    # Receive OPEN value from NODE A
    print("\nVerifying COMMITMENT and OPENING it")
    open_value_from_A = client_socket.recv(2048).decode()

    # Extract private message from NODE A
    private_message_A = f.getPrivateMessageFromNodeA(commit_value_from_A, open_value_from_A)
    print("COMMITMENT successfully VERIFIED")

    # Extract private secret and authentication string from NODE A's message
    print("\nExtracting PRIVATE SECRET(g^Xa) and AUTHENTICATION STRING(Na) of DEVICE A")
    private_secret_from_A, N_a = getP.getPrivateSecretFromMessage(private_message_A)

    # Verify authentication strings Na and Nb
    print("\nVerifying the AUTHENTICATION STRINGS Na and Nb")
    S = getS.getVerificationString(full_message[(-1)*constants.RANDOM_STRING_LENGTH:], N_a)
    client_socket.send(S.encode())
    S_from_A = client_socket.recv(2048).decode()

    if S == S_from_A:
        print("Authenticity of DEVICE A is VERIFIED\n")
        print("The SHARED KEY is:")
        print(genK.generateSharedKey(private_secret_from_A, Xb))
    else:
        print("Authenticity of DEVICE A is NOT VERIFIED\n")

    # Close the connection with NODE A
    client_socket.close()

    # Close the server socket
    server_socket.close()

if __name__ == "__main__":
    main()
