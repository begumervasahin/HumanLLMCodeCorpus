import Seed_Generator
from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives.asymmetric import dh

def generate_dh_parameters():
    """
    Generate Diffie-Hellman (DH) parameters.
    """
    return dh.generate_parameters(generator=5, key_size=1024, backend=default_backend())

def generate_key_pair(parameters):
    """
    Generate private and public key pair using DH parameters.
    """
    private_key = parameters.generate_private_key()
    public_key = private_key.public_key()
    return private_key, public_key

def perform_key_exchange(private_key, peer_public_key):
    """
    Perform key exchange between two parties.
    """
    return private_key.exchange(peer_public_key)

def generate_access_key():
    """
    Generate an access key using Diffie-Hellman key exchange.
    """
    # Generate DH parameters
    parameters = generate_dh_parameters()
    
    # Generate key pairs for party A and party B
    a_private_key, a_peer_public_key = generate_key_pair(parameters)
    b_private_key, b_peer_public_key = generate_key_pair(parameters)
    
    # Perform key exchange between parties A and B
    a_shared_key = perform_key_exchange(a_private_key, b_peer_public_key)
    b_shared_key = perform_key_exchange(b_private_key, a_peer_public_key)
    
    # Check if the shared keys match
    if a_shared_key == b_shared_key:
        # If the shared keys match, generate seed using Seed_Generator module
        Seed_Generator.generate_seed(a_private_key, b_private_key, a_peer_public_key, b_peer_public_key)
        return a_shared_key
    else:
        # If the shared keys do not match, return False
        return False

if __name__ == "__main__":
    # Call the function to generate access key
    generate_access_key()
