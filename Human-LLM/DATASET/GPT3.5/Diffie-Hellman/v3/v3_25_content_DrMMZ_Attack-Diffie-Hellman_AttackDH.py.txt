def perform_diffie_hellman_cryptanalysis():
    print("Cryptanalysis of the Diffie-Hellman public key protocol")
    print("-------------------------------------------------------")
    print()

    # Input prime number and generator
    prime = int(input("Enter the prime number (p): "))
    generator = int(input("Enter the generator (g): "))

    print()

    # Input public keys exchanged by A and B
    public_key_A = int(input("Public key sent by A (A): "))
    public_key_B = int(input("Public key sent by B (B): "))

    print()

    # Initialize variables for private keys
    private_key_A = None
    private_key_B = None

    # Perform cryptanalysis to find private keys
    for x in range(1, prime):
        if (generator ** x) % prime == public_key_A:
            private_key_A = x
        if (generator ** x) % prime == public_key_B:
            private_key_B = x

    # Output the discovered private keys
    print("Private key for A:", private_key_A)
    print("Private key for B:", private_key_B)
    print()

    # Calculate the shared secret key using the private keys
    shared_key_A = (public_key_B ** private_key_A) % prime
    shared_key_B = (public_key_A ** private_key_B) % prime

    # Both shared keys should be the same, so we choose shared_key_A
    shared_secret_key = shared_key_A

    # Output the shared secret key
    print("Shared secret key:", shared_secret_key)
    print()

# Run the cryptanalysis function
perform_diffie_hellman_cryptanalysis()
