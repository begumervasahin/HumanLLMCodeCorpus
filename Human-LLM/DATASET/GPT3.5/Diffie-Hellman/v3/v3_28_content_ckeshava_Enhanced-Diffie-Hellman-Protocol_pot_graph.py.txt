import numpy as np
import matplotlib.pyplot as plt
import csv

# List of CSV files containing data
data_files = ['100.csv', '140.csv', '160.csv', '180.csv', '320.csv', '384.csv', '512.csv']

# Initialize a list to store data from each CSV file
data = [[] for _ in range(len(data_files))]

# Read data from each CSV file and scale it
for i, filename in enumerate(data_files):
    with open(filename) as file:
        reader = csv.reader(file, delimiter=',')
        for row in reader:
            # Scale the third column data and append to the corresponding data list
            scaled_value = (float(row[2]) - 0.1) / 1000000
            data[i].append(scaled_value)

# Generate x-axis values
x_values = list(range(10, 100, 5))

# Plot data from each CSV file
for i, filename in enumerate(data_files):
    plt.plot(x_values, data[i], label=filename)

# Add labels and legend
plt.xlabel('X Axis Label')
plt.ylabel('Y Axis Label')
plt.title('Plot Title')
plt.legend()

# Display the plot
plt.show()
