import random
import time
import matplotlib.pyplot as plt

_mrpt_num_trials = 5

def generate_prime(a, b):
    """
    Generate a prime number within the specified range [a, b].
    """
    while True:
        p = random.randint(a, b)
        if is_probable_prime(p):
            return p

def is_probable_prime(n):
    """
    Check if a number is probably prime using the Miller-Rabin primality test.
    """
    assert n >= 2
    if n == 2:
        return True
    if n % 2 == 0:
        return False
    
    s = 0
    d = n - 1
    while d % 2 == 0:
        s += 1
        d //= 2
    
    def try_composite(a):
        if pow(a, d, n) == 1:
            return False
        for i in range(s):
            if pow(a, 2 ** i * d, n) == n - 1:
                return False
        return True
    
    for _ in range(_mrpt_num_trials):
        a = random.randrange(2, n)
        if try_composite(a):
            return False
    return True

def generate_shared_secrets(n_bits):
    """
    Generate shared secrets between Alice and Bob for different number of bits.
    """
    time_taken = []
    num_bits = []

    for n in range(1, n_bits + 1):
        start_time = time.time()
        
        # Generate a prime number within the specified range
        p = generate_prime(2 ** n, 2 ** (n + 1))
        
        # Generate random values for base and exponents
        b, A, B = random.randint(500, 5000), random.randint(500, 5000), random.randint(500, 5000)
        
        print("Shared prime:", p, "Shared base:", b)
        
        # Calculate values exchanged between Alice and Bob
        X = pow(b, A, p)
        Y = pow(b, B, p)
        
        print("Alice sends:", X, "Bob sends:", Y)
        
        # Calculate shared secrets
        m = pow(Y, A, p)
        n = pow(X, B, p)
        
        print("Alice shared secret:", m, "Bob shared secret:", n)
        
        # Measure and store time taken
        elapsed_time = time.time() - start_time
        print("--- %s seconds ---" % elapsed_time)
        print("n:", n)
        num_bits.append(n)
        time_taken.append(elapsed_time)
    
    return num_bits, time_taken

def plot_time_vs_bits(num_bits, time_taken):
    """
    Plot the time taken vs. the number of bits in the prime number.
    """
    plt.plot(num_bits, time_taken)
    plt.xlabel('Number of bits')
    plt.ylabel('Time taken (seconds)')
    plt.title('Bits in prime number vs. Time taken')
    plt.show()

if __name__ == '__main__':
    num_bits, time_taken = generate_shared_secrets(14)
    plot_time_vs_bits(num_bits, time_taken)
