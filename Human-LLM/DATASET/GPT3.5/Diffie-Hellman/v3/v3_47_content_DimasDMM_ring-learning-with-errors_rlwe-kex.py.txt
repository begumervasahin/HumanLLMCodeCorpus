import numpy as np
from numpy.polynomial import polynomial as poly
from math import floor, sqrt, pi
import sys

# Functions for rounding operations and key generation
def cross_round(x, q):
    return np.floor((4 * x) / q) % 2

def mod_round(x, q):
    return np.around((2 * x) / q) % 2

def dbl(x, n, q):
    dist = [-1, 0, 1]
    prob = [0.25, 0.5, 0.25]
    e = np.random.choice(dist, n, p=prob)
    dbl_x = poly.polysub((2 * x) % (2 * q), e) % (2 * q)
    return dbl_x

def rec(w, b, q):
    n = len(w)
    r = np.zeros(n)
    E_left, E_right = -q / 4, q / 4
    for i in range(n):
        if b[i] == 0:
            r[i] = rec_I_0(w[i], q, E_left, E_right)
        else:
            r[i] = rec_I_1(w[i], q, E_left, E_right)
    return r

def rec_I_0(w, q, E_left, E_right):
    I_start, I_end = 0, round(q / 2) - 1
    bound_left, bound_right = (I_start + E_left) % (2 * q), (I_end + E_right) % (2 * q)
    if bound_right < bound_left:
        bound_left, bound_right = bound_right, bound_left
    return 0 if bound_left <= w < bound_right else 1

def rec_I_1(w, q, E_left, E_right):
    I_start, I_end = -floor(q / 2), -1
    bound_left, bound_right = (I_start + E_left) % (2 * q), (I_end + E_right) % (2 * q)
    if bound_right < bound_left:
        bound_left, bound_right = bound_right, bound_left
    return 0 if bound_left <= w < bound_right else 1

def invert(k):
    return [1 - val for val in k]

# Constants and key size
n = 1024
q = 2 ** 32 - 1
sigma = 8 / sqrt(2 * pi)

# Key generation process
A = np.floor(np.random.random(size=n) * q) % q
sA = np.floor(np.random.normal(0, scale=sigma, size=n))
eA = np.floor(np.random.normal(0, scale=sigma, size=n))
bA = poly.polyadd(poly.polymul(A, sA) % q, eA) % q

sB = np.floor(np.random.normal(0, scale=sigma, size=n))
eB = np.floor(np.random.normal(0, scale=sigma, size=n))
bB = poly.polyadd(poly.polymul(A, sB) % q, eB) % q

eeB = np.floor(np.random.normal(0, scale=sigma, size=n))
v = poly.polyadd(poly.polymul(bA, sB) % q, eeB) % q

dbl_v = dbl(v, n, q)
c = cross_round(dbl_v, 2 * q)
kB = mod_round(dbl_v, 2 * q)

recA = poly.polyadd(poly.polymul((2 * bB) % (2 * q), sA) % (2 * q), eA) % (2 * q)
kA = invert(rec(recA, c, q))

# Output and comparison
print("\n-Params---")
print(f" n: {n}\n q: {q}\n A: {A}")

print("\n-Alice---")
print(f" s: {sA}\n e: {eA}\n b: {bA}")

print("\n-Bob---")
print(f" s': {sB}\n e': {eB}\n b': {bB}\n e'': {eeB}")

print("\n-Computed data---")
print(f" v: {v}\n dbl_v: {dbl_v}\n c: {c}\n recA: {recA}")

print("\n-Generated Keys---")
print(f" Ka: {kA}\n Kb: {kB}")

# Compare keys and calculate equality percentage
equality_percentage = sum(kA[i] == kB[i] for i in range(len(kA))) * 100 / len(kA)
print(f" Equals: {equality_percentage}%")

sys.exit(0)
