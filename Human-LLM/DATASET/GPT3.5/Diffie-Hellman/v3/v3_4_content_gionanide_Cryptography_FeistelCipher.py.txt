import hashlib

def get_password():
    """
    Prompt the user to enter a password, hash it twice using SHA-256, and return the concatenated hashes.
    """
    password = input('Enter your password: ')
    first_pass_hash = hashlib.sha256(password.encode()).hexdigest()
    second_pass_hash = hashlib.sha256(first_pass_hash.encode()).hexdigest()
    return first_pass_hash + second_pass_hash

def xor_hex_strings(op1, op2):
    """
    Perform bitwise XOR operation on two hexadecimal strings.
    """
    a = int(op1, 16)
    b = int(op2, 16)
    return hex(a ^ b)[2:]

def feistel_cipher_round(left, right, key):
    """
    Perform a round of Feistel cipher encryption.
    """
    temp_right = right
    right = xor_hex_strings(left, f(key, right))
    left = temp_right
    return left, right

def encipher_block(plaintext_block, keys):
    """
    Encipher a block of plaintext using Feistel cipher encryption.
    """
    plaintext_hex = str_to_hex(plaintext_block)
    left_half = plaintext_hex[:len(plaintext_hex) // 2]
    right_half = plaintext_hex[len(plaintext_hex) // 2:]
    for key in keys:
        left_half, right_half = feistel_cipher_round(left_half, right_half, key)
    return right_half + left_half

def feistel_cipher_decrypt_round(left, right, key):
    """
    Perform a round of Feistel cipher decryption.
    """
    temp_left = left
    left = xor_hex_strings(right, f(key, left))
    right = temp_left
    return left, right

def decipher_block(cipher_block, keys):
    """
    Decipher a block of ciphertext using Feistel cipher decryption.
    """
    left_half = cipher_block[:len(cipher_block) // 2]
    right_half = cipher_block[len(cipher_block) // 2:]
    for key in reversed(keys):
        left_half, right_half = feistel_cipher_decrypt_round(left_half, right_half, key)
    return left_half + right_half

def blockify_text(plaintext, block_size):
    """
    Divide the plaintext into blocks of the specified size.
    """
    length = len(plaintext)
    padding = block_size - (length % block_size)
    plaintext += padding * " "
    return [plaintext[i:i + block_size] for i in range(0, len(plaintext), block_size)]

def str_to_hex(string):
    """
    Convert a string to its hexadecimal representation.
    """
    return ''.join(hex(ord(char))[2:] for char in string)

def hex_to_str(hex_string):
    """
    Convert a hexadecimal string to its corresponding string.
    """
    hex_string = '0' + hex_string if len(hex_string) % 2 != 0 else hex_string
    return bytes.fromhex(hex_string).decode('utf-8')

def f(key, oper):
    """
    Dummy function for Feistel cipher.
    """
    return key

def build_key_stream(password):
    """
    Build the key stream from the password.
    """
    return ["0x" + password[i:i + 8] for i in range(0, len(password), 8)]

def main():
    """
    Main routine for Feistel cipher encryption and decryption.
    """
    plaintext = "Hello, World!"
    password = get_password()
    key_stream = build_key_stream(password)
    
    # Encryption
    ciphertext = ""
    for block in blockify_text(plaintext, 8):
        ciphertext += encipher_block(block, key_stream)
    print("Cipher:", ciphertext)
    
    # Decryption
    decrypted_text = ""
    for block in reversed(blockify_text(ciphertext, 8)):
        decrypted_text = hex_to_str(decipher_block(block, key_stream)) + decrypted_text
    print("Decryption:", decrypted_text)

if __name__ == "__main__":
    main()
