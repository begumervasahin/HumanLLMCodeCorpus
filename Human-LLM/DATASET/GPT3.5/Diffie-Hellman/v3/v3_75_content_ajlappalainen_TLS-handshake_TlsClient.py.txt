import socket
import ssl
import sys

def create_tls_context(key_exchange):
    """Create a TLS context based on the key exchange mode."""
    cert_path = "client.pem"
    key_path = "client.key"
    ciphers = "DHE-RSA-AES128-SHA256" if key_exchange == "DHE" else "AES128-SHA256"
    context = ssl.SSLContext(ssl.PROTOCOL_TLSv1_2)
    context.load_cert_chain(keyfile=key_path, certfile=cert_path)
    context.set_ciphers(ciphers)
    return context

def connect(server_address, server_port, key_exchange):
    """Connect to the server using TLS with the specified key exchange mode."""
    try:
        client_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        context = create_tls_context(key_exchange)
        tls_sock = context.wrap_socket(client_socket, do_handshake_on_connect=False)
        tls_sock.connect((server_address, server_port))
    except Exception as e:
        print(f"Connection failed: {e}")
        return None
    
    try:
        print("Connection succeeded, attempting handshake...")
        tls_sock.do_handshake()
    except Exception as e:
        print(f"Handshake failed: {e}")
        return None
    
    print(f"Handshake succeeded. Chosen cipher is {tls_sock.cipher()[0]}")
    return tls_sock

def get_key_mode(args):
    """Get the key exchange mode from command-line arguments."""
    if len(args) != 2:
        raise IndexError("Requires 1 argument but {len(args)-1} were given. "
                         "Only accept the following argument: <key_exchange_mode> = DHE or RSA")
    
    key_mode = args[1].upper()
    if key_mode not in ("RSA", "DHE"):
        raise ValueError(f"key_exchange_mode must be either DHE or RSA, not {key_mode}")
    
    return key_mode

def main():
    """Main function to connect to the server and send a test message."""
    server_address, server_port = '127.0.0.1', 50001
    key_exchange_mode = get_key_mode(sys.argv)
    tls_socket = connect(server_address, server_port, key_exchange_mode)
    
    if tls_socket:
        data = 'test_message'
        print('Client sends:', data)
        tls_socket.send(data.encode())
        tls_socket.close()

if __name__ == "__main__":
    main()
