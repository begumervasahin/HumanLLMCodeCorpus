import base64
from cryptography.hazmat.primitives import serialization, hashes
from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives.asymmetric.x25519 import X25519PrivateKey
from cryptography.hazmat.primitives.kdf.hkdf import HKDF

# Utility functions
def get_public_bytes(private_key):
    return private_key.public_key().public_bytes(
        encoding=serialization.Encoding.Raw,
        format=serialization.PublicFormat.Raw
    )

def perform_key_exchange(private_key, peer_public_key):
    return private_key.exchange(peer_public_key)

def generate_key_pair():
    return X25519PrivateKey.generate()

def derive_encryption_key(salt, *keys):
    backend = default_backend()
    hkdf = HKDF(
        algorithm=hashes.BLAKE2s(32),
        length=32,
        salt=salt,
        info=b'MofN-Encryption-demo',
        backend=backend
    )
    combined_keys = b'\x01'.join(keys)
    return hkdf.derive(combined_keys)

def main():
    # Generate private keys
    secret_alice = generate_key_pair()
    secret_bob = generate_key_pair()
    secret_joe = generate_key_pair()
    ephemeral_key = generate_key_pair()

    # Generate shared keys
    shared_alice = perform_key_exchange(secret_alice, ephemeral_key.public_key())
    shared_bob = perform_key_exchange(secret_bob, ephemeral_key.public_key())
    shared_joe = perform_key_exchange(secret_joe, ephemeral_key.public_key())

    # Perform key exchanges
    shared_alice_joe = perform_key_exchange(shared_alice, shared_joe.public_key())
    shared_alice_bob = perform_key_exchange(shared_alice, shared_bob.public_key())
    shared_bob_joe = perform_key_exchange(shared_bob, shared_joe.public_key())

    # Perform ephemeral key exchanges
    ephemeral_alice_joe = perform_key_exchange(shared_alice_joe, ephemeral_key.public_key())
    ephemeral_alice_bob = perform_key_exchange(shared_alice_bob, ephemeral_key.public_key())
    ephemeral_bob_joe = perform_key_exchange(shared_bob_joe, ephemeral_key.public_key())

    # Derive encryption key using HKDF
    keys_to_combine = [ephemeral_alice_joe, ephemeral_alice_bob, ephemeral_bob_joe]
    combined_key = derive_encryption_key(get_public_bytes(ephemeral_key), *keys_to_combine)
    print(base64.b64encode(combined_key))

if __name__ == "__main__":
    main()
