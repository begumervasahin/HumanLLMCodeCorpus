import socket
import select
import threading
import queue
import os
import sys

# Constants
TIMEOUT_E = 1000
TIMEOUT_T = 2.0
SUCCESS = 'Ok!'
FAILURE = 'Closed!'
ENDL = '\n'
NOTIFY_DH = '[PUBKEY]'
GET_PUBKEY = '[ENTRAR]'
ALTER_PRIMO = '[TROCARPRIMO]'
ALTER_ALFA = '[TROCARALFA]'
READ_ONLY = select.POLLIN | select.POLLPRI | select.POLLHUP | select.POLLERR
READ_WRITE = READ_ONLY | select.POLLOUT

# Initialize dictionaries and lists
message_queues = {}
fd_to_socket = {}

# Function to clear console
def clear_console():
    os.system('cls' if os.name == 'nt' else 'reset')

# Function to send feedback to client
def send_feedback(client_socket, msg):
    client_socket.send(msg.encode())

# Function to broadcast message to all clients except the sender
def broadcast(client_socket, msg):
    for client_fd, client in fd_to_socket.items():
        if client is not server_socket and client is not client_socket:
            try:
                client.send(msg.encode())
            except:
                print('(Server) Error while sending message to client!')

# Function to handle client threads
def handle_client(func, client_socket, msg):
    handler = threading.Thread(target=func, args=(client_socket, msg))
    handler.daemon = True
    handler.start()
    handler.join(TIMEOUT_T)
    handler.isAlive()

if __name__ == "__main__":
    if len(sys.argv) != 3:
        print('Usage: python server.py <host_ip> <host_port>')
        sys.exit()

    host_ip = sys.argv[1]
    host_port = int(sys.argv[2])

    # Initialize server socket
    server_address = (host_ip, host_port)
    server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server_socket.setblocking(0)
    server_socket.bind(server_address)
    server_socket.listen(5)

    # Initialize poller
    poller = select.poll()
    poller.register(server_socket, READ_ONLY)
    fd_to_socket[server_socket.fileno()] = server_socket

    # Clear console and print server info
    clear_console()
    print("=> Chat via TCP/IP | Server")
    print("By David and Genilson")
    print("Enjoy it! xD\n\n")
    print(f"Server active: {host_ip} port {host_port}\n\n")
    print("[Waiting for events...]\n\n")

    while True:
        events = poller.poll(TIMEOUT_E)
        for fd, flag in events:
            current_socket = fd_to_socket[fd]
            if flag & (select.POLLIN | select.POLLPRI):
                if current_socket is server_socket:
                    connection, client_address = current_socket.accept()
                    connection.setblocking(0)
                    print(f"Connection accepted from {client_address}\n\n")
                    poller.register(connection, READ_ONLY)
                    fd_to_socket[connection.fileno()] = connection
                    message_queues[connection] = queue.Queue()
                else:
                    data = current_socket.recv(1024)
                    if data:
                        if data != 'sair':
                            poller.modify(current_socket, READ_WRITE)
                            message_queues[current_socket].put(data)
                        else:
                            print(f"Closing connection with {current_socket.getpeername()}\n\n")
                            handle_client(send_feedback, current_socket, FAILURE)
                            poller.modify(current_socket, select.POLLERR)
                            del message_queues[current_socket]
                            poller.unregister(current_socket)
                            del fd_to_socket[current_socket.fileno()]
                            current_socket.close()
            elif flag & select.POLLOUT:
                try:
                    next_msg = message_queues[current_socket].get_nowait()
                except queue.Empty:
                    print(f"Message queue is empty for {current_socket.getpeername()}\n\n")
                    poller.modify(current_socket, READ_ONLY)
                else:
                    if not next_msg.startswith((NOTIFY_DH, GET_PUBKEY, ALTER_PRIMO, ALTER_ALFA)):
                        handle_client(send_feedback, current_socket, SUCCESS)
                    broadcast_msg = f"\n{str(current_socket.getpeername()).ljust(28, ' ')}{next_msg}"
                    handle_client(broadcast, current_socket, broadcast_msg)
                    print(f"Broadcasted message to {len(fd_to_socket) - 2} clients!\n\n")
            elif flag & select.POLLERR:
                print(f"Error occurred with {current_socket.getpeername()}. Closing connection.\n\n")
                handle_client(send_feedback, current_socket, FAILURE)
                del message_queues[current_socket]
                poller.unregister(current_socket)
                del fd_to_socket[current_socket.fileno()]
                current_socket.close()
