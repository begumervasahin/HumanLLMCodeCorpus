import tkinter as tk
from tkinter import filedialog
from keygen import generate_private_key, generate_public_key, generate_secret_key
from encrypt_decrypt import encrypt_file, decrypt_file

class EncryptorDecryptorApp(tk.Tk):
    def __init__(self):
        super().__init__()
        self.title('Encryptor-Decryptor')
        self.geometry('600x400')
        self.resizable(False, False)

        self.private_file = ''
        self.public_file = ''
        self.file_name = ''

        self.setup_ui()

    def setup_ui(self):
        self.private_key_label = tk.Label(self, text='Private Key:')
        self.private_key_label.grid(row=0, column=0)
        self.private_key_name_entry = tk.Entry(self)
        self.private_key_name_entry.insert(tk.END, 'Enter Private Key Name')
        self.private_key_name_entry.grid(row=0, column=1)
        self.browse_private_key_button = tk.Button(self, text='Browse', command=self.browse_private_key)
        self.browse_private_key_button.grid(row=0, column=2)
        self.generate_private_key_button = tk.Button(self, text='Generate Private Key', command=self.generate_private_key)
        self.generate_private_key_button.grid(row=0, column=3)

        self.public_key_label = tk.Label(self, text='Public Key:')
        self.public_key_label.grid(row=1, column=0)
        self.public_key_name_entry = tk.Entry(self)
        self.public_key_name_entry.insert(tk.END, 'Enter Public Key Name')
        self.public_key_name_entry.grid(row=1, column=1)
        self.browse_public_key_button = tk.Button(self, text='Browse', command=self.browse_public_key)
        self.browse_public_key_button.grid(row=1, column=2)
        self.generate_public_key_button = tk.Button(self, text='Generate Public Key', command=self.generate_public_key)
        self.generate_public_key_button.grid(row=1, column=3)

        self.file_label = tk.Label(self, text='File:')
        self.file_label.grid(row=2, column=0)
        self.file_entry = tk.Entry(self)
        self.file_entry.insert(tk.END, 'Select File')
        self.file_entry.grid(row=2, column=1)
        self.browse_file_button = tk.Button(self, text='Browse', command=self.browse_file)
        self.browse_file_button.grid(row=2, column=2)

        self.encrypt_button = tk.Button(self, text='Encrypt', command=self.encrypt_file)
        self.encrypt_button.grid(row=3, column=1)
        self.decrypt_button = tk.Button(self, text='Decrypt', command=self.decrypt_file)
        self.decrypt_button.grid(row=3, column=2)

    def browse_private_key(self):
        filename = filedialog.askopenfilename(title="Select Private Key", filetypes=(("All files", "*.*"), ("Text files", "*.txt")))
        if filename:
            self.private_file = filename.split('/')[-1]
            self.private_key_name_entry.delete(0, tk.END)
            self.private_key_name_entry.insert(0, self.private_file)
