import socket
from DiffieHellman import DiffieHellman
import json

class ClientSocket:
    def __init__(self, debug_flag):
        self.diffie_hellman = DiffieHellman()
        self.debug_flag = debug_flag

    def initialize_diffie_hellman(self, sock):
        sock.send("connected".encode())
        step1 = sock.recv(2048)
        if self.debug_flag:
            print(step1)

        json_data = json.loads(step1.decode())
        dh_params = json_data["dh-keyexchange"]      
        self.diffie_hellman.base = int(dh_params["base"])
        self.diffie_hellman.shared_prime = int(dh_params["prime"])

        public_secret = int(dh_params["publicSecret"])
        calculated_public_secret = str(self.diffie_hellman.calc_public_secret())

        step2 = {
            "dh-keyexchange": {
                "step": 2,
                "publicSecret": calculated_public_secret   
            }
        }
        sock.send(json.dumps(step2).encode())

        self.diffie_hellman.calc_shared_secret(public_secret)

    def start_client(self, ip_address):
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        try:
            sock.connect((ip_address, 20000))
            self.initialize_diffie_hellman(sock)
            print("The secret key is {}".format(self.diffie_hellman.key))
        finally:
            sock.close()

# Example usage:
if __name__ == "__main__":
    client = ClientSocket(debug_flag=True)  # Set debug_flag to True to enable debug messages
    client.start_client("127.0.0.1")  # Replace "127.0.0.1" with the server IP address
