from Crypto.Cipher import AES, DES3, DES, Blowfish, ChaCha20
import hashlib
import random
import socket
import pickle
import argparse
from ecc import getcurvebyname

# Encryption functions

def encrypt_data(cipher, plaintext):
    """Encrypts data using the specified cipher."""
    if args.encryption_algorithm == 4:
        encrypted_value = cipher.nonce + cipher.encrypt(plaintext)
    else:
        encrypted_value = cipher.encrypt(plaintext)
    return encrypted_value

def get_cipher(key):
    """Returns the cipher based on the chosen symmetric algorithm."""
    if args.encryption_algorithm == 0:
        cipher = AES.new(key, AES.MODE_ECB)
    elif args.encryption_algorithm == 1:
        cipher = DES.new(key, DES.MODE_ECB)
    elif args.encryption_algorithm == 2:
        cipher = DES3.new(key, DES3.MODE_ECB)
    elif args.encryption_algorithm == 3:
        cipher = Blowfish.new(key, Blowfish.MODE_ECB)
    elif args.encryption_algorithm == 4:
        cipher = ChaCha20.new(key=key)
    return cipher

def get_key(str_value):
    """Generates a key based on the chosen hash function."""
    if args.encryption_algorithm == 0:
        if args.hash_function == 0:
            key = hashlib.md5()
        elif args.hash_function == 1:
            key = hashlib.blake2s(digest_size=16)
        else:
            key = hashlib.md5()
    elif args.encryption_algorithm in [1, 2, 3, 4]:
        key = hashlib.blake2s(digest_size=8 * (args.encryption_algorithm + 1))
    key.update(str_value)
    return key.digest()

# Communication function

def send_to_server(socket):
    """Establishes a secure communication with the server."""
    curve = getcurvebyname("ed25519")
    g = curve.G
    a = random.randint(1, 2 ** 255 - 19)
    alice = g.__mul__(a)
    socket.send(pickle.dumps(alice, pickle.HIGHEST_PROTOCOL))
    bob = pickle.loads(socket.recv(4096))
    messages = ['Jived fox nymph grabs quick waltz.', 'Glib jocks quiz nymph to vex dwarf.']
    for i, message in enumerate(messages):
        padded_message = message.ljust(64)
        key = get_key(str((bob.__add__(alice.__neg__().__mul__(i))).__mul__(a)).encode())
        cipher = get_cipher(key)
        encrypted_message = encrypt_data(cipher, padded_message.encode())
        socket.send(encrypted_message)

# Constants and arguments

host = "127.0.0.1"
port = 4446

parser = argparse.ArgumentParser()
parser.add_argument("encryption_algorithm", type=int, choices=[0, 1, 2, 3, 4],
                    help="symmetric encryption algorithm [0 - AES, 1 - DES, 2 - 3DES, 3 - Blowfish, 4 - ChaCha20]")
parser.add_argument("-d", "--hash_function", type=int, choices=[0, 1, 2],
                    help="hash function [0 - md5, 1 - blake2s, 2 - SHA256]")
args = parser.parse_args()

# Establish connection and perform communication

socket_client = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
socket_client.connect((host, port))

for _ in range(10):
    send_to_server(socket_client)

socket_client.close()
