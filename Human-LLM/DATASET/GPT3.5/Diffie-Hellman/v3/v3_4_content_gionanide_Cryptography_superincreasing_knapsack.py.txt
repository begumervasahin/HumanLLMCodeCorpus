import random

CHARS_PER_BLOCK = 2
BLOCK_SIZE = 8 * CHARS_PER_BLOCK

def generate_superincreasing_knapsack():
    SIK = [0] * BLOCK_SIZE
    for i in range(BLOCK_SIZE):
        sum_ij = sum(SIK[:i])
        SIK[i] = sum_ij + random.randint(1, 5)
    return SIK

def check_superincreasing_knapsack(SIK):
    for i in range(len(SIK) - 1):
        superi = sum(SIK[:i+1])
        if superi > SIK[i + 1]:
            return False
    return True

def choose_m_n(SIK):
    sum_i = sum(SIK)
    m = random.randint(41, 100)
    n = 0
    while n < sum_i or not coprime(m, sum_i + 1):
        m = random.randint(41, 100)
        n = sum_i + 1
    return m, n

def coprime(a, b):
    while b != 0:
        a, b = b, a % b
    return a == 1

def build_general_knapsack(m, n, SIK):
    GK = [(ai * m) % n for ai in SIK]
    return GK

    msg_blocks = [message[i:i + CHARS_PER_BLOCK].ljust(CHARS_PER_BLOCK) for i in range(0, len(message), CHARS_PER_BLOCK)]
