from math import floor

class Client:
    """
    Represents a temporary client to simulate clients.
    """
    def __init__(self):
        self.private_key = None
        self.shared_secret = None
    
    def calculate_key(self, generator, large_prime):
        """
        Simulates key calculation for the client.
        """
        # Simulate key calculation
        self.shared_secret = generator % large_prime
        return generator * self.shared_secret

def split_list_length(list_length):
    """
    Splits the length of a list into two halves.
    """
    return floor(list_length / 2)

def compare_shared_secrets(clients):
    """
    Compares the shared secrets of clients.
    """
    prev = clients[0]
    for client in clients:
        if client.shared_secret != prev.shared_secret:
            return False
        prev = client
    return True

class KeyGen:
    """
    KeyGen class to generate new keys upon entry of a new client to the server.
    """
    def __init__(self, large_prime):
        self.prime = large_prime
    
    def generate_keys(self, client_list, list_length, generator):
        """
        Recursively generates keys for clients.
        """
        if list_length == 1:
            client_list[0].calculate_key(generator, self.prime)
            return
        else:
            left_length = split_list_length(list_length)
            right_length = list_length - left_length
            left_clients = client_list[:left_length]
            right_clients = client_list[left_length:]
            left_key = generator
            right_key = generator
            for client in left_clients:
                left_key = client.calculate_key(left_key, self.prime)
            for client in right_clients:
                right_key = client.calculate_key(right_key, self.prime)
            self.generate_keys(left_clients, left_length, right_key)
            self.generate_keys(right_clients, right_length, left_key)

if __name__ == '__main__':
    """
    For testing purposes
    """
    # Create a list of connected clients
    connected_clients = [Client() for _ in range(2)]
    
    # Initialize the KeyGen with a large prime
    key_generator = KeyGen(71)
    
    # Generate keys for the connected clients
    key_generator.generate_keys(connected_clients, len(connected_clients), 3)
    
    # Check if all clients have the same shared secret
    if compare_shared_secrets(connected_clients):
        print("The shared secret {}, was correctly calculated for all clients!".format(connected_clients[0].shared_secret))
    else:
        print("The shared secret wasn't correctly calculated")
