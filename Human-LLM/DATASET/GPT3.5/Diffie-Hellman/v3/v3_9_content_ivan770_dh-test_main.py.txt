import random

# Function to generate a random integer
def generate_random_int():
    return random.randint(1, 999999999)

# Function to generate a user's private key
def generate_private_key(random_number):
    return random.randint(1, 999999999) + random_number

# Function to generate a user's public key
def generate_public_key(combined_private):
    return combined_private % 256

# Function to generate a user's shared key
def generate_shared_key(public_key, private_key):
    return (public_key + private_key) % 256

# Generate a random number for the key exchange
random_number = generate_random_int()
print("Using", random_number, "as a random number")

# Generate random private keys for two users
user1_combined_private = generate_private_key(random_number)
user2_combined_private = generate_private_key(random_number)
print("Using", user1_combined_private, "as user1's private key")
print("Using", user2_combined_private, "as user2's private key")

# Generate public keys for both users
user1_public_key = generate_public_key(user1_combined_private)
user2_public_key = generate_public_key(user2_combined_private)
print("Using", user1_public_key, "as user1's public key")
print("Using", user2_public_key, "as user2's public key")

# Generate shared keys for both users
user1_shared_key = generate_shared_key(user2_public_key, user1_combined_private)
user2_shared_key = generate_shared_key(user1_public_key, user2_combined_private)
print("Using", user1_shared_key, "as user1's shared key")
print("Using", user2_shared_key, "as user2's shared key")

# Check if the shared keys are equal
print("Checking equality of shared keys...")
if user1_shared_key == user2_shared_key:
    print("Key exchange successful! Shared keys match.")
else:
    print("Key exchange failed! Shared keys do not match.")
