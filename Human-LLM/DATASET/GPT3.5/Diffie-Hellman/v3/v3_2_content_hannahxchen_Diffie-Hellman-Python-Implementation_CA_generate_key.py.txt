from Crypto import Random
from Crypto.PublicKey import RSA

# Generate random number generator instances
rng_alice = Random.new().read
rng_bob = Random.new().read

# Generate RSA key pairs for Alice and Bob
rsa_key_pair_alice = RSA.generate(1024, rng_alice)
private_key_alice = rsa_key_pair_alice
public_key_alice = rsa_key_pair_alice.publickey()

rsa_key_pair_bob = RSA.generate(1024, rng_bob)
private_key_bob = rsa_key_pair_bob
public_key_bob = rsa_key_pair_bob.publickey()

# Write the private and public keys to text files for Alice
with open("privatekey_Alice.txt", "w") as file_alice_private:
    file_alice_private.write(private_key_alice.exportKey().decode())

with open("publickey_Alice.txt", "w") as file_alice_public:
    file_alice_public.write(public_key_alice.exportKey().decode())

# Write the private and public keys to text files for Bob
with open("privatekey_Bob.txt", "w") as file_bob_private:
    file_bob_private.write(private_key_bob.exportKey().decode())

with open("publickey_Bob.txt", "w") as file_bob_public:
    file_bob_public.write(public_key_bob.exportKey().decode())

print("RSA key pairs for Alice and Bob have been generated and saved to files.")
