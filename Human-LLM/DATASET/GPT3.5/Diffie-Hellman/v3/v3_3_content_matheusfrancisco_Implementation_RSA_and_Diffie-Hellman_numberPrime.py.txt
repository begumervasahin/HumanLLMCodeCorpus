import random
from euclidean import euclidean_extended_gcd, gcd
from primality_tester import primality
from random import getrandbits

def is_prime(n):
    """Check if a number is prime."""
    if n in (2, 3):
        return True
    if n < 2 or n % 3 == 0:
        return False
    return primality(n)

def generate_prime_numbers(n_bits):
    """Generate two distinct prime numbers with the specified number of bits."""
    while True:
        p = getrandbits(n_bits)
        q = getrandbits(n_bits)
        if p != q and is_prime(p) and is_prime(q):
            return p, q

def generate_rsa_parameters(phi_n):
    """Generate parameters d and e for RSA encryption."""
    while True:
        e = random.randint(2, phi_n - 1)
        if gcd(e, phi_n) == 1:
            _, d, _ = euclidean_extended_gcd(e, phi_n)
            if d < 0:
                d += phi_n
            return d, e

def generate_rsa_keys(n_bits, debug=False):
    """Generate RSA public and private keys."""
    p, q = generate_prime_numbers(n_bits)
    n = p * q
    phi_n = (p - 1) * (q - 1)
    d, e = generate_rsa_parameters(phi_n)
    if debug:
        print("Primes Generated:", p, q)
        print("N:", n)
        print("Phi(n):", phi_n)
        print("Public Key (e, n):", e, n)
        print("Private Key (d):", d)
    return n, d, e

if __name__ == '__main__':
    bits = 512  # Number of bits for prime number generation
    n, d, e = generate_rsa_keys(bits, debug=True)
    print("Generated RSA Keys:")
    print("Public Key (e, n):", e, n)
    print("Private Key (d):", d)
