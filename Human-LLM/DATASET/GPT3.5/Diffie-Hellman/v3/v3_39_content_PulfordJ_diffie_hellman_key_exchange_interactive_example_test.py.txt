def calculate_encrypted_value(base, prime, exponent):
    """Calculate the encrypted value using modular exponentiation."""
    return (base ** exponent) % prime

def test_calculate_encrypted_value():
    """Test the calculate_encrypted_value function."""
    assert calculate_encrypted_value(5, 23, 3) == 10
    assert calculate_encrypted_value(5, 23, 4) == 4

def test_shared_secret_is_equal():
    """Test for equality of shared secrets between two parties."""
    secret1 = calculate_encrypted_value(5, 23, 4)
    secret2 = calculate_encrypted_value(5, 23, 3)
    assert calculate_encrypted_value(secret1, 23, 3) == calculate_encrypted_value(secret2, 23, 4)
    
    secret3 = calculate_encrypted_value(5, 23, 14)
    secret4 = calculate_encrypted_value(5, 23, 17)
    assert calculate_encrypted_value(secret3, 23, 17) == calculate_encrypted_value(secret4, 23, 14)

def test_brute_force_secret_is_equal():
    """Test brute force combinations of keys for equality."""
    base = 5
    prime = 23
    for i in range(prime):
        for j in range(prime):
            shared1 = calculate_encrypted_value(calculate_encrypted_value(base, prime, i), prime, j)
            shared2 = calculate_encrypted_value(calculate_encrypted_value(base, prime, j), prime, i)
            assert shared1 == shared2

if __name__ == "__main__":
    test_calculate_encrypted_value()
    test_shared_secret_is_equal()
    test_brute_force_secret_is_equal()
    print("All tests passed successfully!")
