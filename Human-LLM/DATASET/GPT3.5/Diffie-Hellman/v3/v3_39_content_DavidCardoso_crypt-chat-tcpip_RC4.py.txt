class RC4:
    def __init__(self, key='160'):
        self.key = key

    def initialize_state(self):
        global S
        S = list(range(256))
        j = 0
        key_int = self.convert_to_int(self.key)
        key_length = len(key_int)
        for i in range(256):
            j = (j + S[i] + key_int[i % key_length]) % 256
            S[i], S[j] = S[j], S[i]

    def generate_keystream(self, length):
        global S
        i = j = 0
        keystream = []
        for _ in range(length):
            i = (i + 1) % 256
            j = (j + S[i]) % 256
            S[i], S[j] = S[j], S[i]
            keystream_byte = S[(S[i] + S[j]) % 256]
            keystream.append(keystream_byte)
        return keystream

    def convert_to_int(self, text):
        return [ord(char) for char in text]

    def convert_to_text(self, nums):
        return ''.join([chr(num) for num in nums])

    def encrypt(self, plaintext):
        self.initialize_state()
        plaintext_int = self.convert_to_int(plaintext)
        keystream = self.generate_keystream(len(plaintext_int))
        encrypted_int = [plaintext_int[i] ^ keystream[i] for i in range(len(plaintext_int))]
        return self.convert_to_text(encrypted_int)

    def decrypt(self, ciphertext):
        self.initialize_state()
        ciphertext_int = self.convert_to_int(ciphertext)
        keystream = self.generate_keystream(len(ciphertext_int))
        decrypted_int = [ciphertext_int[i] ^ keystream[i] for i in range(len(ciphertext_int))]
        return self.convert_to_text(decrypted_int)

# Example usage
rc4 = RC4()
plaintext = "Hello, RC4 Encryption!"
encrypted_text = rc4.encrypt(plaintext)
print(f"Encrypted: {encrypted_text}")
decrypted_text = rc4.decrypt(encrypted_text)
print(f"Decrypted: {decrypted_text}")
