import math
import random
import sys
from fractions import gcd

def generate_prime_numbers():
    """Generates a list of prime numbers between 1000 and 10000."""
    primes = []
    for num in range(1000, 10001):
        if all(num % i != 0 for i in range(2, int(math.sqrt(num)) + 1)):
            primes.append(num)
    return primes

def pick_random_prime_number(prime_numbers):
    """Selects a random prime number from the given list."""
    secure_random = random.SystemRandom()
    return secure_random.choice(prime_numbers)

def main():
    prime_numbers = generate_prime_numbers()
    modulus = pick_random_prime_number(prime_numbers)
    print(f"Modulus Number: {modulus}")

    secure_random = random.SystemRandom()
    base_number = secure_random.choice(range(1, 10001))
    print(f"Base Number: {base_number}")

    exponent_a = secure_random.choice(range(1, 10001))
    exponent_b = secure_random.choice(range(1, 10001))

    if gcd(base_number, exponent_a) != 1:
        print(f"Error: Base Number ({base_number}) and Exponent A ({exponent_a}) must be relatively prime.")
        sys.exit(1)

    if gcd(base_number, exponent_b) != 1:
        print(f"Error: Base Number ({base_number}) and Exponent B ({exponent_b}) must be relatively prime.")
        sys.exit(1)

    print(f"Exponent A: {exponent_a}")
    print(f"Exponent B: {exponent_b}")

    result_a = base_number ** exponent_a % modulus
    result_b = base_number ** exponent_b % modulus

    print(f"A Computational Numbers: {result_a}")
    print(f"B Computational Numbers: {result_b}")

    if result_b ** exponent_a % modulus == result_a ** exponent_b % modulus:
        print("Computation successful! The computed values match.")
    else:
        print("Computation failed! The computed values do not match.")

if __name__ == "__main__":
    main()
