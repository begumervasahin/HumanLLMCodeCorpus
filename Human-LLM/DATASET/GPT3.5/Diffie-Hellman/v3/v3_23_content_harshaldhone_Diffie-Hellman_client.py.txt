import socket

def powmod(p, e, n):
    """
    Computes (p^e) mod n using the modular exponentiation algorithm.
    """
    result = 1
    while e != 0:
        result = (result * p) % n
        e -= 1
    return result

def connect_to_server(host, port):
    """
    Connects to the server using a TCP socket.
    """
    s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    s.connect((host, port))
    return s

def receive_message(socket):
    """
    Receives a message from the server.
    """
    return socket.recv(1024).decode()

def send_message(socket, message):
    """
    Sends a message to the server.
    """
    socket.send(message.encode())

def main():
    """
    Main function to execute the client-side logic.
    """
    # Get the hostname of the server from the user
    host = input("Please enter the hostname of the server: ")
    port = 1024  # Define the port number

    # Connect to the server
    s = connect_to_server(host, port)
    print("Connected to the server\n")

    # Get the client's private key from the user
    prikey = int(input("Enter the client's private key: "))

    # Compute the message to be sent
    message = str(powmod(17, prikey, 23))
    print("")

    # Receive a message from the server
    income_message = receive_message(s)
    print("Server:", income_message)

    # Encode the message and send it to the server
    send_message(s, message)
    print("Message has been sent\n")

    # Convert the income_message to an integer
    income_message = int(income_message)

    # Compute the shared secret key
    shared_secret = powmod(income_message, prikey, 23)
    print("Secret key shared:", shared_secret)

if __name__ == "__main__":
    main()
