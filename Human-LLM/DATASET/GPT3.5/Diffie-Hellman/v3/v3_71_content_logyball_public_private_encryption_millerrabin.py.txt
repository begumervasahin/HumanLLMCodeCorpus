import random

def is_prime(n):
    """
    Miller-Rabin primality test.
    
    Determines if a given integer is likely to be prime.
    
    Args:
        n (int): The number to test for primality.
        
    Returns:
        bool: True if n is likely to be prime, False otherwise.
    """
    # Check if n is not an integer
    if n != int(n):
        return False
    
    n = int(n)
    
    # Check for specific cases
    if n in {0, 1, 4, 6, 8, 9}:
        return False
    if n in {2, 3, 5, 7}:
        return True
    
    # Decompose n - 1 to the form 2^s * d
    s = 0
    d = n - 1
    while d % 2 == 0:
        d >>= 1
        s += 1
    
    # Apply Miller-Rabin test
    for _ in range(8):
        a = random.randrange(2, n)
        if pow(a, d, n) == 1:
            continue
        composite = True
        for _ in range(s):
            if pow(a, 2 ** _, d, n) == n - 1:
                composite = False
                break
        if composite:
            return False
    
    return True
