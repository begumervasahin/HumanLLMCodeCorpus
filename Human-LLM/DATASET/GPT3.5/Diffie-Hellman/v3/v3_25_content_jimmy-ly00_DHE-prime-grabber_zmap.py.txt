import sys
import os
import subprocess
import re
import csv
from multiprocessing import Pool

# Constants
INPUT_FILE = "zmap_results"
NUM_WORKERS = 80

def process_line(line):
    """
    Process a line from the input file to extract Diffie-Hellman prime parameters.
    """
    server_address = line.strip('\n')
    try:
        # Execute openssl-trace command to check Diffie-Hellman parameters
        cmd_output = subprocess.check_output([
            os.path.join(os.path.dirname(sys.argv[0]), "openssl-trace"),
            "s_client", "-trace",
            "-cipher", "DHE",
            "-connect", f"{server_address}:443"
        ], stdin=subprocess.DEVNULL, stderr=subprocess.DEVNULL, timeout=1)

        # Process the output to extract Diffie-Hellman prime parameter
        for line in cmd_output.decode("ISO-8859-1").splitlines():
            if 'dh_p' in line:
                prime_param = int(re.sub(".*: ", "", line), 16)
                return f'{server_address} {prime_param}'

    except subprocess.CalledProcessError:
        return f'{server_address} No_DHE'

    except subprocess.TimeoutExpired:
        return f'{server_address} Can\'t_connect'

    except Exception:
        return f'{server_address} Error'

if __name__ == "__main__":
    # Open the output CSV file for writing
    with open('output.csv', 'w', newline='') as outfile:
        csv_writer = csv.writer(outfile)

        # Use multiprocessing to process lines in parallel
        with Pool(NUM_WORKERS) as pool:
            with open(INPUT_FILE) as file:
                result = pool.map(process_line, file)

        # Write the processed results to the output CSV file
        for item in result:
            csv_writer.writerow([item])
