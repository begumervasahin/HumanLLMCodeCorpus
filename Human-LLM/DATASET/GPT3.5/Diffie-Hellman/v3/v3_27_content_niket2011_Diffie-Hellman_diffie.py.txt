import random

def check_for_duplicates(pri):
    unique_elements = set(pri)
    return len(unique_elements) == len(pri)

def get_prime_number():
    return int(input("Enter the prime number: "))

def get_num_users():
    return int(input("Enter the number of users: "))

def get_secret_keys(num_users):
    secret_keys = []
    for i in range(num_users):
        key = int(input("Enter secret key for user {}: ".format(i + 1)))
        secret_keys.append(key)
    return secret_keys

def get_user_indices(num_users):
    if num_users > 2:
        print("Enter the users between whom you want to communicate:")
        user1_index = int(input("Enter the user 1: "))
        user2_index = int(input("Enter the user 2: "))
        return user1_index, user2_index
    else:
        return 1, 2

def generate_unique_exponents(prime_number):
    unique_exponents = []
    for i in range(1, prime_number):
        exponents = [(i**j) % prime_number for j in range(1, prime_number)]
        if check_for_duplicates(exponents):
            unique_exponents.append(i)
    return unique_exponents

def generate_shared_secret(prime_number, alpha, xa, xb):
    ya = (alpha**xa) % prime_number
    yb = (alpha**xb) % prime_number
    Kab1 = (yb**xa) % prime_number
    Kab2 = (ya**xb) % prime_number
    return ya, yb, Kab1, Kab2

def main():
    prime_number = get_prime_number()
    num_users = get_num_users()
    secret_keys = get_secret_keys(num_users)
    user1_index, user2_index = get_user_indices(num_users)
    xa = secret_keys[user1_index - 1]
    xb = secret_keys[user2_index - 1]
    unique_exponents = generate_unique_exponents(prime_number)
    alpha = random.choice(unique_exponents)
    print("alpha:", alpha)
    ya, yb, Kab1, Kab2 = generate_shared_secret(prime_number, alpha, xa, xb)
    
    print("Public key of A:", ya)
    print("Public key of B:", yb)
    print("Shared session key with respect to user A:", Kab1)
    print("Shared session key with respect to user B:", Kab2)

if __name__ == "__main__":
    main()
