from math import gcd as bltin_gcd

def is_prime(number):
    """Check if a number is prime."""
    if number <= 1:
        return False
    if number <= 3:
        return True
    if number % 2 == 0 or number % 3 == 0:
        return False

    divisor = 5
    while divisor * divisor <= number:
        if number % divisor == 0 or number % (divisor + 2) == 0:
            return False
        divisor += 6
    return True

def find_primitive_roots(prime_number):
    """Find the primitive roots of a prime number."""
    required_set = {num for num in range(1, prime_number) if bltin_gcd(num, prime_number)}
    return [root for root in range(1, prime_number) if required_set == {pow(root, power, prime_number)
            for power in range(1, prime_number)}]

if __name__ == '__main__':
    print('Enter a prime number:')
    user_input = int(input())
    checked_message = 'Checked: Number is a PRIME'
    warning_message = 'Warning: Number is not a PRIME'
    
    if is_prime(user_input):
        print(checked_message)
        print('Primitive root/roots:')
        print(find_primitive_roots(user_input))
    else:
        print(warning_message)
