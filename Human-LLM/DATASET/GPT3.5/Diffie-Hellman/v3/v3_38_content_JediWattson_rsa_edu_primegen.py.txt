import random

def generate_random_bit_string(bits):
    # Generate a random bit string of specified length
    bit_string = '1'  # Start with '1' as prime numbers are odd
    for _ in range(bits - 2):  # Generate middle bits
        random_bit = random.SystemRandom().randrange(2)  # Randomly choose '0' or '1'
        bit_string += str(random_bit)  # Append the bit to the bit string
    bit_string += '1'  # Ensure the number is odd and greater than 1
    return bit_string

def is_prime_miller_rabin(prime_candidate):
    # Perform Miller-Rabin primality test
    exponent = prime_candidate - 1
    shift = 0

    while exponent % 2 == 0:  # Calculate exponent and shift values
        exponent //= 2
        shift += 1

            continue
        else:
            return False  # Not a prime
    return True  # Likely a prime

def find_random_prime(bits):
    # Generate a random prime number of specified bit length
    while True:
        random_bit_string = generate_random_bit_string(bits)
        prime_candidate = int(random_bit_string, 2)  # Convert bit string to integer
        if is_prime_miller_rabin(prime_candidate):
            return prime_candidate
