import random

def generate_prime_number():
    """
    Generate a random prime number in the range [2, 99].
    
    Returns:
        int: A prime number.
    """
    while True:
        num = random.randint(2, 99)
        if is_prime(num):
            return num

def is_prime(num):
    """
    Check if a number is prime.
    
    Args:
        num (int): Number to check.
    
    Returns:
        bool: True if the number is prime, False otherwise.
    """
    if num < 2:
        return False
    for i in range(2, int(num ** 0.5) + 1):
        if num % i == 0:
            return False
    return True

def find_primitive_roots(modulo):
    """
    Find primitive roots modulo a given number.
    
    Args:
        modulo (int): The modulus value.
    
    Returns:
        list: List of primitive roots.
    """
    roots = []
    for g in range(1, modulo):
        if gcd(g, modulo) == 1:
            actual_set = {pow(g, powers, modulo) for powers in range(1, modulo)}
            if set(range(1, modulo)) == actual_set:
                roots.append(g)
                if len(roots) > 3:
                    break
    return roots

def gcd(a, b):
    """
    Calculate the greatest common divisor (GCD) of two numbers.
    
    Args:
        a (int): First number.
        b (int): Second number.
    
    Returns:
        int: The GCD of the two numbers.
    """
    while b != 0:
        a, b = b, a % b
    return a

def find_prime_and_primitive_root():
    """
    Find a prime number and its corresponding primitive root.
    
    Returns:
        tuple: A tuple containing the prime number and primitive root.
    """
    while True:
        prime_number = generate_prime_number()
        primitive_roots = find_primitive_roots(prime_number)
        if primitive_roots:
            return prime_number, random.choice(primitive_roots)

def calculate_X_and_Y(q, alpha):
    """
    Calculate X and Y values for Diffie-Hellman key exchange.
    
    Args:
        q (int): Prime modulus.
        alpha (int): Primitive root modulo q.
    
    Returns:
        tuple: Tuple containing X and Y values.
    """
    x = random.randint(1, q)
    y = pow(alpha, x, q)
    return x, y

def calculate_shared_key(y, x, q):
    """
    Calculate the shared key for Diffie-Hellman key exchange.
    
    Args:
        y (int): Partner's public key.
        x (int): Private key.
        q (int): Prime modulus.
    
    Returns:
        int: Shared key.
    """
    return pow(y, x, q)

# Generate prime number and primitive root
q, alpha = find_prime_and_primitive_root()

# Calculate X and Y values for Alice and Bob
print("Calculating X and Y values for Alice...")
x_a, y_a = calculate_X_and_Y(q, alpha)
print("Calculating X and Y values for Bob...")
x_b, y_b = calculate_X_and_Y(q, alpha)

# Calculate shared keys
shared_key_a = calculate_shared_key(y_b, x_a, q)
shared_key_b = calculate_shared_key(y_a, x_b, q)
