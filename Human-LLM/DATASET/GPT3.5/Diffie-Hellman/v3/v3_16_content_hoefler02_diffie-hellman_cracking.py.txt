from Crypto.Cipher import AES

def recover_private_keys(p, g, pk1, pk2):
    """Recover the private keys based on public keys."""
    private_key1, private_key2 = 0, 0
    exponent = 1
    count = 0
    while True:
        if pow(g, exponent, p) == pk1:
            private_key1 = exponent
            count += 1
            if count == 2:
                break
        if pow(g, exponent, p) == pk2:
            private_key2 = exponent
            count += 1
            if count == 2:
                break
        exponent += 1
    return private_key1, private_key2

def decrypt_messages(private_key1, private_key2, g, p, encrypted_message1, encrypted_message2):
    """Decrypt messages using the private keys."""
    shared_secret = pow(g, (private_key1 * private_key2), p)
    shared_secret_hex = hex(shared_secret)[2:]
    if len(shared_secret_hex) % 2 != 0:
        shared_secret_hex = '0' + shared_secret_hex
    cipher = AES.new(bytes.fromhex(shared_secret_hex))
    plaintext1 = cipher.decrypt(encrypted_message1).decode('utf-8').replace('\x00', '')
    plaintext2 = cipher.decrypt(encrypted_message2).decode('utf-8').replace('\x00', '')
    return plaintext1, plaintext2
