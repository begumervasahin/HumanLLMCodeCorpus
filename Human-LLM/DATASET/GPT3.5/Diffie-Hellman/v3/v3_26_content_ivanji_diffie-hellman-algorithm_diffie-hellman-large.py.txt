import random
import math

# Miller-Rabin primality test
def rabin_miller(n):
    s = n - 1
    t = 0
    
    # Factor out powers of 2
    while s & 1 == 0:
        s //= 2
        t += 1
    
    # Perform the test for 128 random values of 'a'
    for _ in range(128):
        a = random.randrange(2, n - 1)
        v = pow(a, s, n)
        if v != 1:
            i = 0
            while v != (n - 1):
                if i == t - 1:
                    return False
                else:
                    i += 1
                    v = (v ** 2) % n
    return True

# Generate a large prime number
def generate_large_prime(k):
    max_attempts = 100 * (math.log(k, 2) + 1)
    attempts = max_attempts
    while attempts > 0:
        n = random.randrange(2 ** (k - 1), 2 ** k)
        attempts -= 1
        if is_prime(n):
            return n
    return "Failure after " + str(max_attempts) + " tries."
