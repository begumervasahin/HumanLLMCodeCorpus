import socket
from Crypto import Random
from Crypto.PublicKey import RSA

def generate_rsa_key_pair():
    """Generate a RSA key pair."""
    return RSA.generate(1024, Random.new().read)

def create_key_dicts(client_key, server_key):
    """Create dictionaries to store RSA keys."""
    keys_for_client = {'Client': client_key.exportKey(), 'Server': server_key.publickey().exportKey()}
    keys_for_server = {'Client': server_key.publickey().exportKey(), 'Server': client_key.exportKey()}
    return keys_for_client, keys_for_server

def create_ca_socket():
    """Create a socket for the Certificate Authority (CA)."""
    ca_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    print("Certificate Authority socket created")
    return ca_socket

def bind_ca_socket(ca_socket, address, port):
    """Bind the socket to a port."""
    ca_socket.bind((address, port))
    print("Certificate Authority socket bind complete")

def listen_for_connections(ca_socket, backlog):
    """Listen for incoming connections."""
    ca_socket.listen(backlog)
    print("Certificate Authority server is now listening...")

def handle_connection(connection, keys_for_client, keys_for_server):
    """Handle an incoming connection."""
    address = connection.getpeername()[0]
    print("Connected with", address)

    # Receive the user name
    user = connection.recv(1024)

    # Send the appropriate key based on the user name
    if user == 'Client':
        connection.sendall(keys_for_client['Client'])
        print("Sent Client's private key to the client")
    elif user == 'Server':
        connection.sendall(keys_for_server['Server'])
        print("Sent Server's public key to the client")
    else:
        print("Access denied!")

    # Close the connection
    connection.close()

def main():
    # Generate RSA keys for the client and server
    client_key = generate_rsa_key_pair()
    server_key = generate_rsa_key_pair()

    # Create dictionaries to store keys
    keys_for_client, keys_for_server = create_key_dicts(client_key, server_key)

    # Create and bind the CA socket
    ca_socket = create_ca_socket()
    bind_ca_socket(ca_socket, 'localhost', 9999)

    # Listen for incoming connections
    listen_for_connections(ca_socket, 5)

    # Handle incoming connections indefinitely
    while True:
        # Accept connections
        connection, _ = ca_socket.accept()
        handle_connection(connection, keys_for_client, keys_for_server)

    # Close the CA socket
    ca_socket.close()

if __name__ == "__main__":
    main()
