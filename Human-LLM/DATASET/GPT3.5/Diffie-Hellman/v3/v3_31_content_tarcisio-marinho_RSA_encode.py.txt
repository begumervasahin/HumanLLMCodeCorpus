import random

def totient(n):
    """
    Calculates the totient of a number.
    """
    if is_prime(n):
        return n - 1
    else:
        return False

def is_prime(n):
    """
    Checks if a number is prime.
    """
    if n <= 1:
        return False
    if n <= 3:
        return True
    if n % 2 == 0 or n % 3 == 0:
        return False
    i = 5
    while i * i <= n:
        if n % i == 0 or n % (i + 2) == 0:
            return False
        i += 6
    return True

def generate_E(totient_val):
    """
    Generates a random number 'E' satisfying the conditions.
    """
    while True:
        e = random.randrange(2, totient_val)
        if gcd(totient_val, e) == 1:
            return e

def generate_prime():
    """
    Generates a random prime number.
    """
    while True:
        x = random.randint(1, 100)
        if is_prime(x):
            return x

def gcd(a, b):
    """
    Calculates the greatest common divisor of two numbers.
    """
    while b != 0:
        a, b = b, a % b
    return a

def decrypt_message(ciphertext, n, d):
    """
    Decrypts an encrypted message.
    """
    decrypted_text = []
    for char_code in ciphertext:
        decrypted_char_code = pow(char_code, d, n)
        decrypted_char = chr(decrypted_char_code)
        decrypted_text.append(decrypted_char)
    return decrypted_text

def calculate_private_key(totient_val, e):
    """
    Calculates the private key.
    """
    d = 0
    while (d * e) % totient_val != 1:
        d += 1
    return d
