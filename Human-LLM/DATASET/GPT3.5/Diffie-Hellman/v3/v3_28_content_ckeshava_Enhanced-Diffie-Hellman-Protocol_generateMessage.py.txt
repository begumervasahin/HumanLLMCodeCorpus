import constants
import generatePrivateSecret as getSecret
import generateRandomString as getString
from uuid import getnode as get_mac

def generate_message():
    # Obtain MAC address
    mac_address = get_mac()

    # Define the length of random strings
    random_string_length = constants.RANDOM_STRING_LENGTH
    
    # Generate random strings for authentication and concatenation
    auth_random_string = getString.generateRandomString(random_string_length)
    concatenation_random_string = getString.generateRandomString(random_string_length)
    
    # Concatenate MAC address and random string for authentication
    mac_address_string = str(mac_address) + auth_random_string
    
    # Generate private secret and corresponding Xi value
    private_secret, Xi = getSecret.generatePrivateSecret()
    
    # Create the message by concatenating components
    message = mac_address_string + str(private_secret) + concatenation_random_string
    
    # Print the components for reference
    print("MAC Address:", mac_address_string)
    print("Authentication String:", concatenation_random_string)
    print("Private Secret:", private_secret)
    
    return message, private_secret, Xi

if __name__ == "__main__":
    # Generate and display the message
    message, private_secret, Xi = generate_message()
    print("Generated Message:", message)
