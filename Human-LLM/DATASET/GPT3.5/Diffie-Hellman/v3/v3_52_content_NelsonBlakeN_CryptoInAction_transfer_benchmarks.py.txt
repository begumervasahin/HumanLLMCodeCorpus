import time
import utilities as utils

def run_encryption_decryption_benchmark(algorithm_name, min_key_size, max_key_size, message):
    print(f"\n{algorithm_name} Encryption/Decryption Benchmark:")
    for key_size in range(min_key_size, max_key_size + 1):
        start_time = time.time()
        keys = utils.Utilities.generate_keys(algo=algorithm_name, prime_length=key_size)
        ciphertext = getattr(utils.enc, algorithm_name)(message, keys)
        plaintext = getattr(utils.dec, algorithm_name)(ciphertext, keys)
        end_time = time.time()
        if plaintext == message:
            print(f"Key size: {key_size}, Time taken: {end_time - start_time:.6f} seconds")
        else:
            raise Exception("Decryption failed")

def rsa_benchmark(min_key_size=3, max_key_size=15, message="h"):
    run_encryption_decryption_benchmark("rsa", min_key_size, max_key_size, message)

def elgamal_benchmark(min_key_size=3, max_key_size=15, message="h"):
    run_encryption_decryption_benchmark("el_gamal", min_key_size, max_key_size, message)

def des_benchmark(min_key_size=3, max_key_size=15, message="h"):
    run_encryption_decryption_benchmark("des", min_key_size, max_key_size, message)

def otp_benchmark(min_key_size=3, max_key_size=15, message="h"):
    binary_msg = bin(int(message.encode('hex'), 16))[2:]
    run_encryption_decryption_benchmark("one_time_pad", min_key_size, max_key_size, binary_msg)

if __name__ == "__main__":
    print("Running benchmark...")
    rsa_benchmark(min_key_size=3, max_key_size=20)
