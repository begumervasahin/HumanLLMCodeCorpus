from random import randrange, getrandbits
import random
import sys

def modular_exponentiation(g_base, exponent, modulo):
    """Computes modular exponentiation: g_base^exponent mod modulo."""
    result = 1
    binary_exponent = "{0:b}".format(exponent)
    
    for bit in binary_exponent:
        result = (result ** 2) % modulo
        
        if bit == '1':
            result = (result * g_base) % modulo

    return result

def miller_rabin_primality_test(number, iterations=128):
    """Performs the Miller-Rabin primality test."""
    if number == 2 or number == 3:
        return True
    if number <= 1 or number % 2 == 0:
        return False

    s = 0
    r = number - 1
    while r & 1 == 0:
        s += 1
        r //= 2

    for _ in range(iterations):
        witness = randrange(2, number - 1)
        x = modular_exponentiation(witness, r, number)
        
        if x != 1 and x != number - 1:
            j = 1
            while j < s and x != number - 1:
                x = modular_exponentiation(x, 2, number)
                if x == 1:
                    return False
                j += 1
            if x != number - 1:
                return False

    return True

def generate_prime_candidate(bits):
    """Generates a random prime candidate with the specified number of bits."""
    candidate = getrandbits(bits)
    candidate |= (1 << bits - 1) | 1
    return candidate

def generate_prime_number(bits=1023):
    """Generates a random prime number with the specified number of bits."""
    prime = 4
    while not miller_rabin_primality_test(prime, 128):
        prime = generate_prime_candidate(bits)
    return prime

if __name__ == '__main__':
    # Generate a random big prime q
    big_prime_q = generate_prime_number()

    # Choose a factor j and calculate big prime p = j * q + 1
    j = 2
    big_prime_p = j * big_prime_q + 1
    while not miller_rabin_primality_test(big_prime_p):
        big_prime_p += 2

    # Find a generator for the group
    generator = 1
    while generator == 1:
        h = random.randint(2, big_prime_p - 1)
        generator = modular_exponentiation(h, j, big_prime_p)

    # Choose a random exponent a
    a = random.randint(2, big_prime_q - 1)

    # Calculate g^a mod p
    ga = modular_exponentiation(generator, a, big_prime_p)

    # Write the generated parameters to files specified by command-line arguments
    with open(sys.argv[1], 'w') as file_p:
        file_p.write('( %d, %d, %d )' % (big_prime_p, generator, ga))

    with open(sys.argv[2], 'w') as file_a:
        file_a.write('( %d, %d, %d )' % (big_prime_p, generator, a))
