import random

def is_prime(n):
    """
    Check if a number is prime.
    
    Args:
        n (int): Number to check for primality.
    
    Returns:
        bool: True if the number is prime, False otherwise.
    """
    if n < 2:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

def generate_random_prime(start, end):
    """
    Generate a random prime number in the range [start, end).
    
    Args:
        start (int): Lower bound of the range.
        end (int): Upper bound of the range.
    
    Returns:
        int: Random prime number.
    """
    primes = [i for i in range(start, end) if is_prime(i)]
    return random.choice(primes)
