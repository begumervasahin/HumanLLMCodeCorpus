import binascii

def xor(m, k):
    """
    Perform XOR operation between two binary strings.
    
    Args:
        m (str): Binary string representing the message.
        k (str): Binary string representing the key.
    
    Returns:
        str: XOR result of the two binary strings.
    """
    result = []
    for i, j in zip(m, k):
        if i.isalpha() and j.isalpha():
            result.append(i)
        else:
            result.append(str(int(i) ^ int(j)))
    return "".join(result)

def convert_to_binary(string):
    """
    Convert a string to its binary representation.
    
    Args:
        string (str): Input string.
    
    Returns:
        str: Binary representation of the input string.
    """
    return bin(int(binascii.hexlify(string.encode('utf-8')), 16))

def main():
    # Define plaintext and keys
    plain_text = 'informationsecurity'
    key1 = 'vlaksjdhfgqodzmxncb'
    alternative_key = 'tlftrffwmixor|{xbch'
    
    # Print the original inputs
    print('The plain text is:', plain_text)
    print('The first key is:', key1, '\n')
    
    # Convert plaintext and keys to binary
    plain_text_bits = convert_to_binary(plain_text)
    key_bits = convert_to_binary(key1)
    
    # Print plaintext and keys in binary format
    print('Plain text in bits:', plain_text_bits, '\n')
    print('Key in bits:', key_bits, '\n')
    
    # Perform XOR operation for encryption with the first key
    encryption_result = xor(plain_text_bits, key_bits)
    encrypted_text = binascii.unhexlify('%x' % int(encryption_result, 2))
    
    # Print encryption result
    print('XOR result for the encryption with the normal key:', encryption_result, '\n')
    print('Encrypted text:', encrypted_text)
    
    # Perform decryption with the first key
    decryption = xor(encryption_result, key_bits)
    decrypted_text = binascii.unhexlify('%x' % int(decryption, 2))
    
    # Print decryption result
    print('\nDecryption result:', decryption, '\n')
    print('Decrypted text with the normal key:', decrypted_text, '\n')
    
    # Convert alternative key to binary
    alternative_key_bits = convert_to_binary(alternative_key)
    
    # Perform decryption with the alternative key
    decryption_with_alternative_key = xor(encryption_result, alternative_key_bits)
    decrypted_text_alternative_key = binascii.unhexlify('%x' % int(decryption_with_alternative_key, 2))
    
    # Print decryption result with the alternative key
    print('Decryption with the alternative key \n')
    print('Alternative key in bits:', alternative_key_bits, '\n')
    print('Decrypted text with the alternative key:', decrypted_text_alternative_key)

if __name__ == "__main__":
    main()
