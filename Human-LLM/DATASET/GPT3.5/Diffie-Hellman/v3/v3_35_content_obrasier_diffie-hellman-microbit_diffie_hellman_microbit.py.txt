# Constants
SENDER = 'Alice'
RECEIVER = 'Bob'
SECRET_MESSAGE = 'secret message'
SECRET_KEY = 6
PRIME_NUMBER = 541
BASE_NUMBER = 10
INITIAL_KEY = '00000:09000:90999:09009:00000'
UNLOCKED_PATTERN = '00900:09090:09000:09990:09990'
LOCKED_PATTERN = '00900:09090:09090:09990:09990'

# Function to create a message
def create_message(sender, receiver, payload):
    '''Create a message to be sent to the receiver.'''
    return '{}:{}:{}'.format(sender, receiver, payload)

# Function to calculate the Diffie-Hellman shared secret
def calculate_shared_secret(partner_calculated_key):
    '''Calculate the Diffie-Hellman shared secret.'''
    return (partner_calculated_key ** SECRET_KEY) % PRIME_NUMBER

# Function to rotate letters
def rotate_letter(letter, key):
    '''Rotate a lowercase or uppercase letter by the key.'''
    if letter.islower():
        code = (ord(letter) - ord('a') + key) % 26 + ord('a')
        return chr(code)
    elif letter.isupper():
        code = (ord(letter) - ord('A') + key) % 26 + ord('A')
        return chr(code)
    return letter

# Function to encrypt a message using a Caesar cipher
def encrypt_message(plaintext, key):
    '''Encrypt a message using a Caesar cipher.'''
    ciphertext = ''
    for letter in plaintext:
        ciphertext += rotate_letter(letter, key)
    return ciphertext

# Function to decrypt a message
def decrypt_message(ciphertext, key):
    '''Decrypt a message.'''
    return encrypt_message(ciphertext, -key)

# Initialization
encrypted = False
key_sent = False

# Sending Diffie-Hellman calculated key to Bob
print("Sending Diffie-Hellman calculated key to Bob...")
calculated_key = (BASE_NUMBER ** SECRET_KEY) % PRIME_NUMBER
message_to_send = create_message(SENDER, RECEIVER, calculated_key)
print("Message Sent:", message_to_send)

print("\nWaiting for response from Bob...\n")

# Waiting for response from Bob and decrypting the message
while not encrypted:
    print("No response received yet...")
    encrypted_text_received = input("Enter the encrypted message received from Bob: ")
    print("Decrypting message...")
    shared_secret_key = calculate_shared_secret(int(calculated_key))
    offset = shared_secret_key % 26
    decrypted_message = decrypt_message(encrypted_text_received, offset)
    print("Decrypted Message:", decrypted_message)
    encrypted = True

# Communication encrypted
print("\nCommunication encrypted.")
print("Sending encrypted message to Bob...")
encrypted_message_to_send = create_message(SENDER, RECEIVER, encrypted_text_received)
print("Message Sent:", encrypted_message_to_send)
print("\nMessage sent successfully. Exiting...")
