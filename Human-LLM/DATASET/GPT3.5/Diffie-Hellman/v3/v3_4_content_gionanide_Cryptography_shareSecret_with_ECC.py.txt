    alice_public_key = initialize_public_key()

def initialize_public_key():
    # Get parameters for the public key from the user
    a = int(input('Enter the "a" factor: '))
    b = int(input('Enter the "b" factor: '))
    p = int(input('Enter the prime number "p": '))
    x, y = map(int, input('Enter the initial point in the format "xy": ').split())

    # Print the public key parameters
    print('\n')
    print('The curve equation as a PUBLIC KEY: (y^2 = x^3 + {}x + {}) mod {}'.format(a, b, p))
    print('Initial point: ({},{}) \n'.format(x, y))

    return {'a': a, 'b': b, 'p': p, 'x': x, 'y': y}


def initialize_private_key():
    # Get the multiplier (private key) from the user
    multiplier = int(input('Enter the multiplier (private key): '))
    print('Multiplier (private key): {}\n'.format(multiplier))

    return multiplier


def multiplicative_inverse(n):
    # Find the multiplicative inverse of a number modulo 17
    x = 1
    while not (n * x % 17 == 1):
        x += 1
    return x


def calculate_point(x1, y1, x2, y2, a, p, flag, i):
    # Calculate the next point on the elliptic curve
    if x1 == x2:
        if y1 == y2:
            if flag:
                # Calculate the slope of the tangent line
                divide = multiplicative_inverse(2 * y1)
                s = (((3 * x1 ** 2 + a) % p) * divide) % p
                flag = False
        else:
            s = 0
    else:
        divide1 = multiplicative_inverse(x1 - x2)
        s = ((y1 - y2) % p * divide1) % p

    # Calculate the new point
    x3 = ((s ** 2) % p - (x1 + x2) % p) % p
    y3 = (((s * (x1 - x3)) % p) % p - y1) % p

    # Print the result of this calculation
    print('Round {}: ({},{})'.format(i, x3, y3))

    return x2, y2, x3, y3


def calculate_send_message(public_key, private_key, flag):
    # Extract parameters from the public key
    a, b, p, x, y = public_key['a'], public_key['b'], public_key['p'], public_key['x'], public_key['y']

    # Calculate the result of sending the message
    variables = calculate_point(x, y, x, y, a, p, flag, 1)
    if private_key - 1 <= 1:
        return variables[2], variables[3]
    else:
        for i in range(1, private_key):
            variables = calculate_point(variables[2], variables[3], variables[0], variables[1], a, p, flag, i + 1)
        return variables[2], variables[3]
