from random import randrange
from modular_arithmetic import modExp

def is_prime(number, n_loops=8):
    """Check if a number is prime using the Miller-Rabin primality test."""
    if number == 2:
        return True
    if number < 2 or number % 2 == 0:
        return False
    
    # Compute (number - 1) // 2 and remove factors of 2
    rest = (number - 1) >> 1
    while rest % 2 == 0:
        rest >>= 1
    
    # Perform primality testing using Miller-Rabin algorithm
    for _ in range(n_loops):
        if is_composite(rest, number):
            return False
    
    return True

def is_composite(rest, number):
    """Check if a number is composite using the Miller-Rabin primality test."""
    # Choose a random base 'a' in the range [1, number)
    a = randrange(1, number)
    
    # Check if 'a' is a witness for the compositeness of 'number'
    if modExp(a, rest, number) == 1:
        return False
    
    # Double 'exponent' until reaching 'number - 1'
    exponent = rest
    while exponent < number - 1:
        if modExp(a, exponent, number) == number - 1:
            return False
        exponent <<= 1
    
    return True
