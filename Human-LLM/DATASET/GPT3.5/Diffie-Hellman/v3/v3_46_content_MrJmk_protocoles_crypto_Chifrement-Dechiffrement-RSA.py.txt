# Compute greatest common divisor using Euclidean algorithm
def euclidean_gcd(a, b):
    while b != 0:
        a, b = b, a % b
    return a

# Extended Euclidean algorithm to find modular inverse
def extended_euclidean(a, b):
    x, xx = 1, 0
    y, yy = 0, 1
    while b != 0:
        q = a // b
        a, b = b, a % b
        xx, x = x - q * xx, xx
        yy, y = y - q * yy, yy
    return a, x, y

# Compute modular inverse
def modular_inverse(a, n):
    gcd, u, v = extended_euclidean(a, n)
    if gcd != 1:
        return None  # Modular inverse does not exist
    else:
        return u % n

# Compute RSA private key
def rsa_private_key(p, q, e):
    n = p * q
    phi = (p - 1) * (q - 1)
    d = modular_inverse(e, phi)
    return n, d

# Encrypt message using RSA
def rsa_encrypt(message, n, e):
    encrypted = [pow(ord(char), e, n) for char in message]
    return encrypted

# Decrypt RSA-encrypted message
def rsa_decrypt(encrypted, n, d):
    decrypted = [chr(pow(char, d, n)) for char in encrypted]
    return ''.join(decrypted)

# Check if a number is prime
def is_prime(number):
    if number <= 1:
        return False
    for divisor in range(2, int(number**0.5) + 1):
        if number % divisor == 0:
            return False
    return True

# Generate random prime numbers for p and q
