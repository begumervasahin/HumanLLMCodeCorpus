def calculate_public_key(base, private_key, prime_modulus):
    '''Calculates the public key.'''
    return exp_mod(base, private_key, prime_modulus)

def calculate_shared_secret_key(public_key, private_key, prime_modulus):
    '''Calculates the shared secret key.'''
    return exp_mod(public_key, private_key, prime_modulus)

def exp_mod(base, exponent, modulus):
    '''Exponential function used for modular exponentiation.'''
    base = int(base)
    exponent = int(exponent)
    modulus = int(modulus)
    
    result = 1
    if exponent == 0:
        return 1
    else:
        while exponent > 0:
            if exponent % 2 == 1:
                result = (result * base) % modulus
            base = (base * base) % modulus
            exponent //= 2
        return result
