from random import randint, randrange

def is_prime(number):
    if number <= 3:
        return number > 1
    if not number % 2 or not number % 3:
        return False
    for i in range(5, int(number ** 0.5) + 1, 6):
        if not number % i or not number % (i + 2):
            return False
    return True

def generate_prime(start, end):
    """
    Generate a prime number in the given range.
    """
    prime = 0
    while not is_prime(prime):
        prime = randint(start, end)
    return prime

def gcd(a, b):
    """
    Calculate the greatest common divisor (GCD) of two numbers.
    """
    while b != 0:
        a, b = b, a % b
    return a

def extended_gcd(a, b):
    """
    Extended Euclidean Algorithm to find modular inverse.
    """
    x, xx, y, yy = 1, 0, 0, 1
    while b != 0:
        q = a // b
        a, b = b, a % b
        xx, x = x - q * xx, xx
        yy, y = y - q * yy, yy
    return a, x, y

def modular_inverse(a, n):
    """
    Calculate the modular inverse of 'a' modulo 'n'.
    """
    c, u, v = extended_gcd(a, n)
    if c != 1:
        return 0
    else:
        return u % n

def generate_key_pair():
    """
    Generate a pair of public and private RSA keys.
    """
    start, end = 100000000, 999999999
    p = generate_prime(start, end)
    q = generate_prime(start, end)
    n = p * q
    phi = (p - 1) * (q - 1)
    
    e = 2
    g = gcd(phi, e)
    while g != 1:
        e = randrange(1, phi)
        g = gcd(phi, e)
    
    d = modular_inverse(e, phi)
    
    return (e, n), (d, n)

print("---------------------------------")
print("----------ALGORITHME RSA---------")
print("---------------------------------\n")

public_key, private_key = generate_key_pair()

print("\nLa paire de clés publiques est : (e, n) =", public_key)
print("La paire de clés privées est : (d, n) =", private_key)
