import tkinter as tk
from tkinter import messagebox

def is_prime(n):
    """Check if a number is prime."""
    if n < 2:
        return False
    for divisor in range(2, int(n ** 0.5) + 1):
        if n % divisor == 0:
            return False
    return True

def calculate_secret_key():
    """Calculate the secret key based on user input."""
    try:
        prime1 = int(entry_prime1.get())
        prime2 = int(entry_prime2.get())
        secret1 = int(entry_secret1.get())
        secret2 = int(entry_secret2.get())
    except ValueError:
        messagebox.showinfo('Error', 'Please enter integers for prime numbers and secret numbers')
        return

    if not is_prime(prime1):
        messagebox.showinfo('Error', 'First input is not a prime number')
    elif not is_prime(prime2):
        messagebox.showinfo('Error', 'Second input is not a prime number')
    else:
        a_raised_to_b_mod_p = (prime1 ** secret1) % prime2
        b_raised_to_a_mod_p = (prime1 ** secret2) % prime2
        secret_key = (b_raised_to_a_mod_p ** secret1) % prime2
        secret_key_check = (a_raised_to_b_mod_p ** secret2) % prime2

        if secret_key != secret_key_check:
            messagebox.showinfo('Error', 'Unknown Error')
        else:
            messagebox.showinfo('Key', 'Your Secret Key is: ' + str(secret_key))

# Create the main window
root = tk.Tk()
root.title('Secret Key Calculator')
root.geometry('275x210')

# Create labels and entry fields for user input
label_prime1 = tk.Label(root, text="Enter the first prime number:")
entry_prime1 = tk.Entry(root, bd=1)

label_prime2 = tk.Label(root, text="Enter the second prime number:")
entry_prime2 = tk.Entry(root, bd=1)

label_secret1 = tk.Label(root, text="Enter the first secret number:")
entry_secret1 = tk.Entry(root, bd=1)

label_secret2 = tk.Label(root, text="Enter the second secret number:")
entry_secret2 = tk.Entry(root, bd=1)

# Create the submit button
submit_button = tk.Button(root, text="Calculate Secret Key", command=calculate_secret_key)

# Place the widgets in the window using the pack geometry manager
label_prime1.pack()
entry_prime1.pack()

label_prime2.pack()
entry_prime2.pack()

label_secret1.pack()
entry_secret1.pack()

label_secret2.pack()
entry_secret2.pack()

submit_button.pack(side=tk.BOTTOM)

# Start the main event loop
root.mainloop()
