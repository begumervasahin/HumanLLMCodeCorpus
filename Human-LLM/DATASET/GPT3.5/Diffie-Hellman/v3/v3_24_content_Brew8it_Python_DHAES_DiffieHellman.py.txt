    def generate_random(self, bits):
        bytes_count = bits // 8 + 1
        rand_bytes = os.urandom(bytes_count)
        rand_int = int.from_bytes(rand_bytes, 'big')
        return rand_int

        return self.generate_random(self.key_length)

        return pow(self.generator, self.private_int, self.prime)

    def generate_shared_secret(self, private_int, public_key):
        return pow(public_key, private_int, self.prime)

    def generate_key(self, public_key):
        shared_secret = self.generate_shared_secret(self.private_int, public_key)
        shared_secret_bytes = shared_secret.to_bytes((shared_secret.bit_length() + 7) // 8, 'big')
        hash_obj = hashlib.sha256()
        hash_obj.update(shared_secret_bytes)
        self.key = hash_obj.digest()

    def get_key(self):
        return self.key
