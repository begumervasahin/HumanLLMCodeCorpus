def rational_to_contfrac(x, y):
    """
    Convert a rational fraction x/y into a continued fraction [a0, ..., an].
    """
    partial_quotients = []
    while y != 0:
        a = x // y
        partial_quotients.append(a)
        x, y = y, x - a * y
    return partial_quotients

def convergents_from_contfrac(frac):
    """
    Compute the list of convergents from the continued fraction.
    """
    convergents = []
    for i in range(len(frac)):
        convergents.append(contfrac_to_rational(frac[0:i + 1]))
    return convergents

def contfrac_to_rational(frac):
    """
    Convert a continued fraction [a0, ..., an] into a rational number x/y.
    """
    num = frac[-1]
    denom = 1
    for a in reversed(frac[:-1]):
        num, denom = a * num + denom, num
    return (num, denom)

def test_continued_fraction_functions():
    """
    Test the functionality of the continued fraction manipulation functions.
    """
    test_numbers = [(1, 1), (1, 2), (5, 15), (27, 73), (73, 27)]
    for num, denom in test_numbers:
        print('Rational number:', (num, denom))
        contfrac = rational_to_contfrac(num, denom)
        print('Continued fraction:', contfrac)
        print('Convergents:', convergents_from_contfrac(contfrac))
        print('***********************************')

if __name__ == "__main__":
    test_continued_fraction_functions()
