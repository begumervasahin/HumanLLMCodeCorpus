def exp_mod(a, n, z):
    '''Exponential function used by the algorithm.'''
    a = int(a)
    n = int(n)
    z = int(z)
    r = 1
    
    if n == 0:
        return 1
    else:
        if n % 2 != 0:
            r = exp_mod(a, n - 1, z)
            return ((a * r) % z)
        else:
            r = exp_mod(a, n / 2, z)
            return ((r * r) % z)
