def exponentiate(y, b, mod):
    result = 1
    while b > 1:
        if b % 2 == 0:
            y = (y ** 2) % mod
            b = b / 2
        if b % 2 == 1:
            b -= 1
            result = (result * y) % mod
    return result

def calculate_mod(alfa, xa, xb, xc, modulo):
    ya = exponentiate(alfa, xa, modulo)
    yb = exponentiate(alfa, xb, modulo)
    yc = exponentiate(alfa, xc, modulo)
    zac = exponentiate(yc, xa, modulo)
    zba = exponentiate(ya, xb, modulo)
    zcb = exponentiate(yb, xc, modulo)
    k1 = exponentiate(zcb, xa, modulo)
    k2 = exponentiate(zac, xb, modulo)
    k3 = exponentiate(zba, xc, modulo)
    if k1 == k2 == k3:
        print(
            f"p = {modulo}, a = {alfa}, xa = {xa}, xb = {xb}, ya = {ya}, yb = {yb}, k = {k1}"
        )
    else:
        print("Not equal")

def order_y_vector(y_vector, user, count):
    aux_vector = y_vector[:]
    z = 0
    for i in range(count, user):
        y_vector[i] = aux_vector[z]
        z = z + 1
    p = 0
    for i in range(p, count - 1):
        y_vector[i] = aux_vector[z]
        z += 1
    return y_vector
