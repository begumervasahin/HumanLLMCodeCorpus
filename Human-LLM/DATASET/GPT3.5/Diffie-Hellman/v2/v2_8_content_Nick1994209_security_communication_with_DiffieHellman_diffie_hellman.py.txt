class DiffieHellman:
    def __init__(self, shared_prime, shared_base, secret_key):
        # Convert the shared prime, shared base, and secret key to numbers
        self.shared_prime = self.convert_key_to_numbers(shared_prime)
        self.shared_base = self.convert_key_to_numbers(shared_base)
        self.secret_key = self.convert_key_to_numbers(secret_key)
        
        # Print the converted values for debugging
        print("Shared base:", self.shared_base)
        print("Shared prime:", self.shared_prime)
        print("Secret key:", self.secret_key)
    
    def convert_key_to_numbers(self, key):
        # Convert each character in the key to a number and concatenate them
        converted_key = ''
        for symbol in key:
            number_for_symbol = ord(symbol) % 10  # Get the ASCII value and take modulo 10
            converted_key += str(number_for_symbol)
        return int(converted_key)  # Convert the concatenated string to an integer
    
    def compute_medium_key(self):
        print('Computing medium key')
        # Compute the medium key using modular exponentiation
        medium_key = pow(self.shared_base, self.secret_key, self.shared_prime)
        print("Medium key:", medium_key)
        return medium_key
    
    def compute_general_secret_key(self, mixed_key):
        # Compute the general secret key using modular exponentiation
        general_secret_key = pow(mixed_key, self.secret_key, self.shared_prime)
        print("General secret key:", general_secret_key)
        return general_secret_key

    # Define shared prime, shared base, and secret keys for Alice and Bob
    shared_prime, shared_base = 'prime_123', 'base_456'
    a_secret_key = 'a_123'
    b_secret_key = 'b_key'
    
    # Initialize Diffie-Hellman instances for Alice and Bob
    a_diffie = DiffieHellman(shared_prime=shared_prime, shared_base=shared_base, secret_key=a_secret_key)
    b_diffie = DiffieHellman(shared_prime=shared_prime, shared_base=shared_base, secret_key=b_secret_key)
    
    # Compute medium keys for Alice and Bob
    a_medium_key = a_diffie.compute_medium_key()
    b_medium_key = b_diffie.compute_medium_key()
    
    # Verify key exchange by computing general secret keys
    assert a_diffie.compute_general_secret_key(b_medium_key) == b_diffie.compute_general_secret_key(a_medium_key)
