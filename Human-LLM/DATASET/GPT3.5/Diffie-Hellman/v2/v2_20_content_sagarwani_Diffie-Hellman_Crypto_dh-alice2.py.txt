import sys

def modular_exponentiation(base, exponent, modulus):
    """Computes modular exponentiation efficiently."""
    result = 1
    bits = "{0:b}".format(exponent)
    for bit in bits:
        if bit == '1':
            result = (((result ** 2) * base) % modulus)
        elif bit == '0':
            result = ((result ** 2) % modulus)
    return result % modulus
