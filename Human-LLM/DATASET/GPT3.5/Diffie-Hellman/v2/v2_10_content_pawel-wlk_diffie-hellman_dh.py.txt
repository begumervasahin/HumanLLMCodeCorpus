import math

# Function to check if a number is prime
def is_prime(number):
    for divisor in range(2, math.ceil(math.sqrt(number)) + 1):
        if number % divisor == 0:
            return False
    return True

# Function to compute modular exponentiation efficiently
def fast_modulo_power(base, exponent, modulus):
    result = 1
    binary_exponent = bin(exponent)[2:]  # Convert exponent to binary
    for bit in binary_exponent:
        result = (result * result) % modulus
        if bit == '1':
            result = (result * base) % modulus
    return result

# Function to factorize a number
def factorize(number):
    factors = []
    divisor = 2
    while divisor * divisor <= number:
        if number % divisor == 0:
            factors.append(divisor)
            number //= divisor
        else:
            divisor += 1
    if number > 1:
        factors.append(number)
    return factors

# Function to find a generator for a prime number
def find_generator(prime_number):
    for candidate in range(2, prime_number):
        for factor in factorize(prime_number - 1):  # Factorize p-1
            if fast_modulo_power(candidate, factor, prime_number) == 1:
                break
        else:
            return candidate
    return None

# Main code
number_to_check = 1000003

if is_prime(number_to_check):
    print("{} is a prime number.".format(number_to_check))
    generator = find_generator(number_to_check)
    if generator is not None:
        print("The smallest generator is: {}".format(generator))
        print("Simulating Diffie-Hellman key exchange:")
        try:
            private_a = int(input("Enter private number for person A: "))
            private_b = int(input("Enter private number for person B: "))
            public_a = fast_modulo_power(generator, private_a, number_to_check)
            public_b = fast_modulo_power(generator, private_b, number_to_check)
            shared_key_a = fast_modulo_power(public_b, private_a, number_to_check)
            shared_key_b = fast_modulo_power(public_a, private_b, number_to_check)
            print("Shared key computed by person A: {}".format(shared_key_a))
            print("Shared key computed by person B: {}".format(shared_key_b))
        except ValueError:
            print("Invalid input. Please enter integers for private numbers.")
    else:
        print("No generator found for {}.".format(number_to_check))
else:
    print("{} is not a prime number.".format(number_to_check))
