import socket
from threading import Thread
import random
from aes import decrypt, encrypt
from ec import generate_curve, gen_curve

# Initialize socket and connection-related variables
sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
connections = []
keys = {}

# Bind the socket to an address and start listening
sock.bind(('0.0.0.0', 12345))
sock.listen(10)

def accept_connect():
    """Accept incoming connections and start handling them in separate threads."""
    while True:
        client, client_address = sock.accept()
        connections.append(client)
        Thread(target=handle_message, args=(client, client_address)).start()

def handle_message(client, client_addr):
    """Handle messages from clients."""
    # Exchange public keys
    public_key = client.recv(1024).decode("utf-8")
    E, G = generate_curve()
    nServer = random.random() * 100000
    pServer = nServer * G
    Px = pServer.x()
    Py = pServer.y()
    keyServer = f"{Px} {Py}"
    client.send(bytes(keyServer, "utf-8"))

    # Receive exchanged key and derive AES key
    exchanged_key = client.recv(1024).decode("utf-8").split(" ")
    F, P = gen_curve(int(exchanged_key[0]), int(exchanged_key[1]))
    key = nServer * P
    aes_key = str(key.x() * key.y())
    ex_key = bytes(aes_key[-16:], "utf-8")  # Ensure AES key length is 16 bytes

    # Store client's AES key and handle messages
    print(f"Welcome {client_addr} to the chat\n")
    keys[client] = ex_key
    while True:
        data = decrypt(client.recv(1024).decode("utf-8"), ex_key)
        if not data:
            break
        if data == b"quit":
            connections.remove(client)
            del keys[client]
        print(data)
        try:
            broadcast(bytes(data, "utf-8"))
        except:
            print("Error")

def broadcast(msg):
    """Broadcast a message to all connected clients."""
    for connection in connections:
        enc_msg = encrypt(msg, keys[connection])
        connection.send(bytes(enc_msg, "utf-8"))

# Start accepting connections
accept_connect()
sock.close()
