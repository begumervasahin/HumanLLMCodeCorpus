class Point:
    def __init__(self, x, y):
        # Initialize the coordinates of the point
        self.x = x
        self.y = y

class EllipticCurve:
    def __init__(self, a, b, p):
        # Initialize the parameters of the elliptic curve
        self.a = a
        self.b = b
        self.p = p

    def multiplicative_inverse(self, x):
        # Compute the multiplicative inverse using Extended Euclidean Algorithm
        t = 0
        r = self.p
        new_t = 1
        new_r = x
        while new_r != 0:
            q = r // new_r
            t, new_t = new_t, t - q * new_t
            r, new_r = new_r, r - q * new_r
        if t < 0:
            return t + self.p
        else:
            return t

    def point_double(self, p):
        # Double a point on the elliptic curve
        if p.x == 0 and p.y == 0:
            return Point(0, 0)
        numerator = (3 * p.x ** 2) + self.a
        denominator = 2 * p.y % self.p
        if denominator == 0:
            return Point(0, 0)
        denominator_inv = self.multiplicative_inverse(denominator)
        lmb = (numerator * denominator_inv) % self.p
        xr = (lmb ** 2 - 2 * p.x) % self.p
        yr = (lmb * (p.x - xr) - p.y) % self.p
        return Point(xr, yr)

    def point_addition(self, p, q):
        # Add two points on the elliptic curve
        if p.x == 0 and p.y == 0:
            return q
        if q.x == 0 and q.y == 0:
            return p
        if p.x == q.x and p.y == q.y:
            return self.point_double(p)
        lmb_numerator = q.y - p.y
        lmb_denominator = q.x - p.x % self.p
        if lmb_denominator == 0:
            return Point(0, 0)
        lmd_denominator_inv = self.multiplicative_inverse(lmb_denominator)
        lmb = (lmb_numerator * lmd_denominator_inv) % self.p
        xr = (lmb ** 2 - p.x - q.x) % self.p
        yr = (lmb * (p.x - xr) - p.y) % self.p
        return Point(xr, yr)

    def double_and_add(self, scalar, P):
        # Double-and-add algorithm for scalar multiplication
        result = Point(0, 0)
        while scalar != 0:
            if scalar % 2 == 1:
                result = self.point_addition(result, P)
            P = self.point_double(P)
            scalar = scalar // 2
        return result

    def is_on_curve(self, P):
        # Check if a point is on the elliptic curve
        arg1 = P.y ** 2 % self.p
        arg2 = (P.x ** 3 + self.a * P.x + self.b) % self.p
        return arg1 == arg2

    def scalar_multiply(self, scalar, P):
        # Scalar multiplication of a point on the elliptic curve
        return self.double_and_add(scalar, P)

# Example usage
if __name__ == "__main__":
    # Define parameters for the elliptic curve
    a_coefficient = 2
    b_coefficient = 3
    prime_modulus = 17

    # Create an elliptic curve object
    elliptic_curve = EllipticCurve(a_coefficient, b_coefficient, prime_modulus)

    # Define a point on the curve
    point_P = Point(5, 1)

    # Check if the point is on the curve
    print("Is P on the curve?", elliptic_curve.is_on_curve(point_P))

    # Multiply the point by a scalar
    scalar_value = 7
    result_point = elliptic_curve.scalar_multiply(scalar_value, point_P)
    print(f"{scalar_value} * P =", f"({result_point.x}, {result_point.y})")
