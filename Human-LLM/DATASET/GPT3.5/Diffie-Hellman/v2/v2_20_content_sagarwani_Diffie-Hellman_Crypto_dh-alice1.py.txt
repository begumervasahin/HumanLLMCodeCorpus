from random import randrange, getrandbits
import random
import sys

def pow_x(g_base, a, p_mod):
    """Modular exponentiation: g_base^a mod p_mod."""
    x = 1
    bits = "{0:b}".format(a)
    for i, bit in enumerate(bits):
        if bit == '1':
            x = (((x ** 2) * g_base) % p_mod)
        elif bit == '0':
            x = ((x ** 2) % p_mod)
    return x % p_mod

def check_prime(n, k=128):
    """Miller-Rabin primality test."""
    if n == 2 or n == 3:
        return True
    if n <= 1 or n % 2 == 0:
        return False

    s = 0
    r = n - 1
    while r & 1 == 0:
        s += 1
        r //= 2

    for _ in range(k):
        a = randrange(2, n - 1)
        x = pow_x(a, r, n)
        if x != 1 and x != n - 1:
            j = 1
            while j < s and x != n - 1:
                x = pow_x(x, 2, n)
                if x == 1:
                    return False
                j += 1
            if x != n - 1:
                return False

    return True

def generate_prime_candidate(l):
    """Generate a random prime candidate with l bits."""
    p = getrandbits(l)
    p |= (1 << l - 1) | 1
    return p

def generate_prime_number(l=1023):
    """Generate a random prime number with l bits."""
    p = 4
    while not check_prime(p, 128):
        p = generate_prime_candidate(l)
    return p

if __name__ == '__main__':
    # Generate a random big prime q
    big_prime_q = generate_prime_number()

    # Choose a factor j and calculate big prime p = j * q + 1
    j = 2
    big_prime_p = j * big_prime_q + 1
    while not check_prime(big_prime_p):
        big_prime_p += 2

    # Find a generator for the group
    generator = 1
    while generator == 1:
        h = random.randint(2, big_prime_p - 1)
        generator = pow_x(h, j, big_prime_p)

    # Choose a random exponent a
    a = random.randint(2, big_prime_q - 1)

    # Calculate g^a mod p
    ga = pow_x(generator, a, big_prime_p)

    # Write the generated parameters to files specified by command-line arguments
    with open(sys.argv[1], 'w') as file_p:
        file_p.write('( %d, %d, %d )' % (big_prime_p, generator, ga))

    with open(sys.argv[2], 'w') as file_a:
        file_a.write('( %d, %d, %d )' % (big_prime_p, generator, a))
