import random
from euclidean import gcd, euclidean_extended_gcd
from numberPrime import is_prime, generate_prime_numbers
from modular_arithmetic import modExp

class RSA:
    def __init__(self, n_bits=100, debug=False):
        self.n, self.d, self.e = self.generate_keys(n_bits, debug)

    def generate_keys(self, n_bits, debug=False):
        p, q = generate_prime_numbers(n_bits)
        self.n = p * q
        phi_n = (p - 1) * (q - 1)
        self.d, self.e = self.get_params(phi_n)
        if debug:
            self.print_keys_info(p, q, phi_n)

    def get_params(self, phi_n):
        while True:
            e = random.randint(2, phi_n - 1)
            if gcd(e, phi_n) == 1:
                d, _, _ = euclidean_extended_gcd(e, phi_n)
                if d < 0:
                    d += phi_n
                return d, e

    def encryption(self, msg):
        """Encrypts the message using RSA."""
        encrypted_msg = [modExp(char, self.e, self.n) for char in msg]
        print('Encrypted Message:', encrypted_msg)
        return encrypted_msg

    def decryption(self, msg):
        """Decrypts the message using RSA."""
        decrypted_msg = [modExp(char, self.d, self.n) for char in msg]
        print('Decrypted Message:', decrypted_msg)
        return decrypted_msg

    def public_key(self):
        """Returns the public key (e, n)."""
        return self.e, self.n

    def private_key(self):
        """Returns the private key (d, n)."""
        return self.d, self.n

    def print_keys_info(self, p, q, phi_n):
        """Prints generated RSA keys information."""
        print("Primes Generated:", p, q)
        print("N:", self.n)
        print("Phi(n):", phi_n)
        print("Public Key (e, n):", self.e, self.n)
        print("Private Key (d):", self.d)

if __name__ == '__main__':
    print("RSA Encryption and Decryption Test:")
    rsa = RSA(n_bits=128, debug=True)
    msg = [65, 66, 67]  # Example message to encrypt
    encrypted = rsa.encryption(msg)
    decrypted = rsa.decryption(encrypted)
    print("Original Message:", msg)
    print("Decrypted Message Matches Original:", decrypted == msg)
