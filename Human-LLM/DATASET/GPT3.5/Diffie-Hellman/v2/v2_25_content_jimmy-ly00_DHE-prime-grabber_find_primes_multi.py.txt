import sys
import os
import subprocess
import re
import csv
from multiprocessing import Pool

FILE = "alexa_top1mil"
WORKERS = 80

def process_line(line):
    row = line.split(',')
    server = row[0]
    servername = row[1].strip('\n')
    
    try:
        # Execute openssl-trace command to check Diffie-Hellman parameters
        cmd = subprocess.check_output([
            os.path.dirname(sys.argv[0]) + "/openssl-trace",
            "s_client", "-trace",
            "-cipher", "DHE",
            "-connect", server + ":443"
        ], stdin=subprocess.DEVNULL, stderr=subprocess.DEVNULL, timeout=1)
        
        # Process the output to extract Diffie-Hellman prime parameter
        for line in cmd.decode("ISO-8859-1").splitlines():
            if 'dh_p' in line:
                prime = int(re.sub(".*: ", "", line), 16)
                return '{} {} {}'.format(server, servername, prime)
    
    except subprocess.CalledProcessError:
        return '{} {} {}'.format(server, servername, "No_DHE")
    
    except subprocess.TimeoutExpired:
        return '{} {} {}'.format(server, servername, "Can't_connect")
    
    except Exception as e:
        return '{} {} {}'.format(server, servername, "Error: " + str(e))

if __name__ == "__main__":
    # Open the output CSV file for writing
    with open('output.csv', 'w') as outfile:
        wr = csv.writer(outfile)
        pool = Pool(WORKERS)
        
        # Process each line in parallel using multiprocessing
        with open(FILE) as f:
            result = pool.map(process_line, f)
        
        # Write the processed results to the output CSV file
        for item in result:
            wr.writerow([item])
