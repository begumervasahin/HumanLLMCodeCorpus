import random
from math import log

def generate_random_bit_string(bits):
    bit_string = '1'  # Start the bit string with '1' as all prime numbers are odd
    for _ in range(bits - 2):  # Generate the middle bits of the bit string
        random_bit = random.SystemRandom().randrange(2)  # Generate either '0' or '1'
        bit_string += str(random_bit)  # Append the random bit to the bit string
    bit_string += '1'  # End the bit string with '1' to ensure the number is odd and greater than 1
    return bit_string

def convert_bit_string_to_prime(bit_string):
    prime_number = int(bit_string, 2)  # Convert the bit string to an integer
    while True:

        is_prime = True
        exponent = prime_number - 1

        # Perform the Miller-Rabin primality test
        while exponent % 2 == 0:  # Calculate the exponent and shift values
            exponent //= 2
            shift += 1

            else:
