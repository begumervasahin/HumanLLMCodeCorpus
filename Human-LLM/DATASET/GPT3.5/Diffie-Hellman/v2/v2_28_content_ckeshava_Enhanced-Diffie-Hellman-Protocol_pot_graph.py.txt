import numpy as np
import matplotlib.pyplot as plt
import csv

# Prepare a list of CSV files to read data from
data_files = ['100.csv', '140.csv', '160.csv', '180.csv', '320.csv', '384.csv', '512.csv']

# Initialize an empty list to store data from each CSV file
data = [[] for _ in range(len(data_files))]

# Read data from each CSV file
for i, file in enumerate(data_files):
    with open(file) as csv_file:
        csv_reader = csv.reader(csv_file, delimiter=',')
        # Iterate through each row in the CSV file
        for row in csv_reader:
            # Convert the third column data to float and scale it
            scaled_value = (float(row[2]) - 0.1) / 1000000
            # Append the scaled value to the corresponding data list
            data[i].append(scaled_value)

# Generate x-axis values
x_axis = list(range(10, 100, 5))

# Plot the data from each CSV file
for i, file in enumerate(data_files):
    plt.plot(x_axis, data[i], label=file)

# Add labels and legend to the plot
plt.xlabel('X Axis Label')
plt.ylabel('Y Axis Label')
plt.title('Plot Title')
plt.legend()

# Display the plot
plt.show()
