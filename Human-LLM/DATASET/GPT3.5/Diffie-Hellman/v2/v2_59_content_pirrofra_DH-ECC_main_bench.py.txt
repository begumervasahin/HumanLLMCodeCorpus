import timeit
from dh import ECC_CDH

def key_test(dh):
    try:
        d1, Q1 = dh.key_pair_generator()
        d2, Q2 = dh.key_pair_generator()
        z1 = dh.shared_key_generator(d1, Q2)
        z2 = dh.shared_key_generator(d2, Q1)
    except:
        print("*** An error occurred while generating the shared key ***")
        raise ValueError("Error during key generation")
    
    if z1 == z2:
        pass
    else:
        print("*** Two different secret keys were generated ***")
        raise ValueError("Inconsistent key generation")

def bench(curve, times):
    print("Curve Type: " + curve)
    
    setup_code = '''
from __main__ import key_test
from dh import ECC_CDH
dh = ECC_CDH("''' + curve + '''")'''

    key_test_code = '''key_test(dh)'''
    double_and_add_time = timeit.timeit(setup=setup_code, stmt=key_test_code, number=times)
    print("*** All keys generated successfully ***")
    print(str(times) + " keys generated using Double-and-Add in " + str(double_and_add_time) + " seconds")

    setup_code_wNAF = '''
from __main__ import key_test
from dh import ECC_CDH
dh = ECC_CDH("''' + curve + '''", wNAF=1)'''

    key_test_code_wNAF = '''key_test(dh)'''
    wNAF_time = timeit.timeit(setup=setup_code_wNAF, stmt=key_test_code_wNAF, number=times)
    print("*** All keys generated successfully ***")
    print(str(times) + " keys generated using wNAF in " + str(wNAF_time) + " seconds")

    if wNAF_time < double_and_add_time:
        print("*** WNAF is faster ***")
    else:
        print("*** Double-and-Add is faster ***")
    
    print("----------------------------------------------------")

def multiple_bench(curve, key_times, num_bench):
    for _ in range(num_bench):
        bench(curve, key_times)
    
    print("----------------------------------------------------")

# Example usage
multiple_bench("curves/p192.xml", 1000, 10)
multiple_bench("curves/p224.xml", 800, 10)
multiple_bench("curves/p256.xml", 600, 10)
multiple_bench("curves/p384.xml", 400, 10)
multiple_bench("curves/p521.xml", 200, 10)
