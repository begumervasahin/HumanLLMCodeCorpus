from rsa import RSA
from convertN2T import *

def encrypt_message(message, private_key, modulus):
    """Encrypts the message using RSA."""
    encrypted_message = []
    for char in message:
        encrypted_char = modExp(char, private_key, modulus)
        encrypted_message.append(encrypted_char)
    return encrypted_message

def create_signature(message, public_key, modulus):
    """Creates a digital signature for the message using RSA."""
    signature = []
    for char in message:
        signed_char = modExp(char, public_key, modulus)
        signature.append(signed_char)
    return signature

def verify_signature(signature, original_message, sender_public_key, sender_modulus, receiver_public_key, receiver_modulus):
    """Verifies the digital signature of the received message."""
    decrypted_signature = []
    for char in signature:
        decrypted_char = modExp(char, sender_public_key, sender_modulus)
        decrypted_signature.append(decrypted_char)
    
    decrypted_message = []
    for char in decrypted_signature:
        decrypted_char = modExp(char, receiver_public_key, receiver_modulus)
        decrypted_message.append(decrypted_char)
    
    return decrypted_message
