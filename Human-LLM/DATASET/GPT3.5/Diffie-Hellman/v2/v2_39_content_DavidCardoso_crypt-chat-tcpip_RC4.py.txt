class RC4(object):
    def __init__(self, key='160'):
        self.key = key

    def key_scheduling_algorithm(self):
        j = 0
        if isinstance(self.key, str):
            key_int = self.convert_to_int(self.key)
        key_length = len(key_int)
        global S
        S = list(range(256))
        for i in range(256):
            j = (j + S[i] + key_int[i % key_length]) % 256
            S[i], S[j] = S[j], S[i]

    def pseudo_random_generation_algorithm(self, msg):
        i = 0
        j = 0
        if isinstance(msg, str):
            msg = self.convert_to_int(msg)
        msg_length = len(msg)
        result = []
        for k in range(msg_length):
            i = (i + 1) % 256
            j = (j + S[i]) % 256
            S[i], S[j] = S[j], S[i]
            result.append(S[(S[i] + S[j]) % 256] ^ msg[k])
        return result

    def convert_to_int(self, text):
        return [ord(char) for char in text]

    def convert_to_text(self, nums):
        return ''.join([chr(num) for num in nums])

    def encrypt(self, plaintext):
        self.key_scheduling_algorithm()
        result = self.pseudo_random_generation_algorithm(plaintext)
        return self.convert_to_text(result)

    def decrypt(self, ciphertext):
        self.key_scheduling_algorithm()
        result = self.pseudo_random_generation_algorithm(ciphertext)
        return self.convert_to_text(result)

# Example usage
rc4 = RC4()
plaintext = "Hello, RC4 Encryption!"
encrypted_text = rc4.encrypt(plaintext)
print(f"Encrypted: {encrypted_text}")
decrypted_text = rc4.decrypt(encrypted_text)
print(f"Decrypted: {decrypted_text}")
