import hashlib

def get_password():
    """
    Prompt the user to enter a password, hash it twice using SHA-256, and return the concatenated hashes.
    """
    password = input('Enter your password: ')
    first_pass = hashlib.sha256(password.encode()).hexdigest()
    second_pass = hashlib.sha256(first_pass.encode()).hexdigest()
    return first_pass + second_pass

def XOR(op1, op2):
    """
    Perform bitwise XOR operation on two hexadecimal strings.
    """
    a = int(op1, 16)
    b = int(op2, 16)
    return hex(a ^ b)[2:]

def do_round(left, right, key):
    """
    Perform a round of Feistel cipher encryption.
    """
    tmp = right
    right = XOR(left, f(key, right))
    left = tmp
    return left, right

def encipher_block(plaintext_block, ks):
    """
    Encipher a block of plaintext using Feistel cipher encryption.
    """
    plaintext_block = str2hex(plaintext_block)
    left = plaintext_block[:len(plaintext_block) // 2]
    right = plaintext_block[len(plaintext_block) // 2:]
    for key in ks:
        left, right = do_round(left, right, key)
    return right + left

def do_round_decr(left, right, key):
    """
    Perform a round of Feistel cipher decryption.
    """
    tmp = left
    left = XOR(right, f(key, left))
    right = tmp
    return left, right

def decipher_block(cipher_block, ks):
    """
    Decipher a block of ciphertext using Feistel cipher decryption.
    """
    left = cipher_block[:len(cipher_block) // 2]
    right = cipher_block[len(cipher_block) // 2:]
    for key in reversed(ks):
        left, right = do_round_decr(left, right, key)
    return left + right

def blockify(plaintext, blocksize):
    """
    Divide the plaintext into blocks of the specified size.
    """
    length = len(plaintext)
    padding = 0
    if length % blocksize != 0:
        padding = blocksize - (length % blocksize)
    plaintext += padding * " "
    return [plaintext[i:i + blocksize] for i in range(0, len(plaintext), blocksize)]

def str2hex(string):
    """
    Convert a string to its hexadecimal representation.
    """
    hexa = ""
    for char in string:
        hexa += hex(ord(char))[2:]
    return hexa

def hex2str(hex_string):
    """
    Convert a hexadecimal string to its corresponding string.
    """
    if len(hex_string) % 2 == 0:
        pass
    else:
        hex_string = "0" + hex_string
    return bytes.fromhex(hex_string).decode('utf-8')

def f(key, oper):
    """
    Dummy function for Feistel cipher.
    """
    return key

def build_keys_stream(password):
    """
    Build the key stream from the password.
    """
    keys_stream = [password[i:i + 8] for i in range(0, len(password), 8)]
    return ["0x" + key for key in keys_stream]

def main_routine(plaintext):
    """
    Main routine for Feistel cipher encryption and decryption.
    """
    password = get_password()
    keys_stream = build_keys_stream(password)
    
    # Encryption
    ciphertext = ""
    for block in blockify(plaintext, 8):
        ciphertext += encipher_block(block, keys_stream)
    print("Cipher:", ciphertext)
    
    # Decryption
    plaintext_decrypted = ""
    for block in reversed(blockify(ciphertext, 8)):
        plaintext_decrypted = hex2str(decipher_block(block, keys_stream)) + plaintext_decrypted
    print("Decryption:", plaintext_decrypted)

# Example usage
plaintext = "Hello, World!"
main_routine(plaintext)
