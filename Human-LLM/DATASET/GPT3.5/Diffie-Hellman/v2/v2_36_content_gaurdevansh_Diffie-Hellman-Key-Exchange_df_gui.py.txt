from tkinter import *
from tkinter import filedialog
from keygen import generate_private_key, generate_public_key, generate_secret_key
from encrypt_decrypt import encrypt_file, decrypt_file

private_file = ''
public_file = ''
file_name = ''

def browse_private_key():
    global private_file
    filename = filedialog.askopenfilename(title="Select Private Key", filetypes=(("All files", "*.*"), ("Text files", "*.txt")))
    if filename:
        private_file = filename.split('/')[-1]
        private_key_entry.delete(0, END)
        private_key_entry.insert(0, private_file)

def browse_public_key():
    global public_file
    filename = filedialog.askopenfilename(title="Select Public Key", filetypes=(("All files", "*.*"), ("Text files", "*.txt")))
    if filename:
        public_file = filename.split('/')[-1]
        public_key_entry.delete(0, END)
        public_key_entry.insert(0, public_file)

def browse_file():
    global file_name
    filename = filedialog.askopenfilename(title="Select File", filetypes=(("All files", "*.*"), ("Text files", "*.txt")))
    if filename:
        file_name = filename.split('/')[-1]
        file_entry.delete(0, END)
        file_entry.insert(0, file_name)

def encrypt_file_func():
    secret_key = str(generate_secret_key(private_file, public_file))
    encrypt_file(secret_key, file_name)

def decrypt_file_func():
    secret_key = str(generate_secret_key(private_file, public_file))
    decrypt_file(secret_key, file_name)

def generate_private_key_func():
    filename = private_key_name_entry.get()
    generate_private_key(filename)

def generate_public_key_func():
    private_key_filename = private_key_name_entry.get()
    public_key_filename = public_key_name_entry.get()
    generate_public_key(private_key_filename, public_key_filename)

root = Tk()
root.title('Encryptor-Decryptor')
root.geometry('600x400')
root.resizable(False, False)

private_key_label = Label(root, text='Private Key:')
private_key_label.grid(row=0, column=0)
private_key_name_entry = Entry(root)
private_key_name_entry.insert(END, 'Enter Private Key Name')
private_key_name_entry.grid(row=0, column=1)
browse_private_key_button = Button(root, text='Browse', command=browse_private_key)
browse_private_key_button.grid(row=0, column=2)
generate_private_key_button = Button(root, text='Generate Private Key', command=generate_private_key_func)
generate_private_key_button.grid(row=0, column=3)

public_key_label = Label(root, text='Public Key:')
public_key_label.grid(row=1, column=0)
public_key_name_entry = Entry(root)
public_key_name_entry.insert(END, 'Enter Public Key Name')
public_key_name_entry.grid(row=1, column=1)
browse_public_key_button = Button(root, text='Browse', command=browse_public_key)
browse_public_key_button.grid(row=1, column=2)
generate_public_key_button = Button(root, text='Generate Public Key', command=generate_public_key_func)
generate_public_key_button.grid(row=1, column=3)

file_label = Label(root, text='File:')
file_label.grid(row=2, column=0)
file_entry = Entry(root)
file_entry.insert(END, 'Select File')
file_entry.grid(row=2, column=1)
browse_file_button = Button(root, text='Browse', command=browse_file)
browse_file_button.grid(row=2, column=2)

encrypt_button = Button(root, text='Encrypt', command=encrypt_file_func)
encrypt_button.grid(row=3, column=1)
decrypt_button = Button(root, text='Decrypt', command=decrypt_file_func)
decrypt_button.grid(row=3, column=2)

root.mainloop()
