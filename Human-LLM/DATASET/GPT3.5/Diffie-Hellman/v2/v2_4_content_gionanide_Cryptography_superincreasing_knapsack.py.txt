from random import randint

CHARS_PER_BLOCK = 2
BLOCK_SIZE = 8 * CHARS_PER_BLOCK

def build_superincreasing_knapsack():
    SIK = [0] * BLOCK_SIZE
    for i in range(BLOCK_SIZE):
        sum_ij = sum(SIK[:i])
        SIK[i] = sum_ij + randint(1, 5)
    superi = 0
    for i in range(len(SIK) - 1):
        superi += SIK[i]
        if superi > SIK[i + 1]:
            print("NON SIK")
    return SIK

SIK = [1, 4, 6, 12, 25, 50, 103, 205, 409, 820, 1639, 3276, 6554, 13106, 26212, 52425]
m, n = 89, 104848
mmi = 87177
GK = [89, 356, 534, 1068, 2225, 4450, 9167, 18245, 36401, 72980, 41023, 81868, 59066, 13106, 26212, 52513]

def gcd(a, b):
    while b != 0:
        a, b = b, a % b
    return a

def coprime(a, b):
    return gcd(a, b) == 1

def choose_m_n():
    sum_i = sum(SIK)
    m = randint(41, 100)
    n = 0
    while n < sum_i:
        if coprime(m, sum_i + 1):
            n = sum_i + 1
    return m, n

def build_general_knapsack(m, n, SIK):
    GK = [(ai * m) % n for ai in SIK]
    return GK

def modulo_multiplicative_inverse(A, M):
    gcd, x, y = extended_euclid_gcd(A, M)
    if x < 0:
        x += M
    return x

def extended_euclid_gcd(a, b):
    s, old_s = 0, 1
    t, old_t = 1, 0
    r, old_r = b, a
    while r != 0:
        quotient = old_r // r
        old_r, r = r, old_r - quotient * r
        old_s, s = s, old_s - quotient * s
        old_t, t = t, old_t - quotient * t
    return old_r, old_s, old_t

def str2ascii(x):
    ascii_val = ord(x)
    binary = "{0:b}".format(ascii_val).zfill(8)
    return binary

def encrypt_N_char(chars):
    bin_chars = "".join(str2ascii(char)[::-1] for char in chars)
    return sum(int(bit) * val for bit, val in zip(bin_chars, GK))

def decrypt_N_char(number):
    d = (number * mmi) % n
    res = ""
    for sik_i in reversed(SIK):
        if d >= sik_i:
            res = "1" + res
            d -= sik_i
        else:
            res = "0" + res
    res = res[::-1]
    char1, char2 = chr(bit2Str(res[:8])), chr(bit2Str(res[8:]))
    return char1 + char2

def bit2Str(bitstring):
    out = sum(int(bit) * (1 << i) for i, bit in enumerate(reversed(bitstring)))
    return out

def encrypt_mh(message):
    msg_blocks = [message[i:i + CHARS_PER_BLOCK] for i in range(0, len(message), CHARS_PER_BLOCK)]
    encrypted = [encrypt_N_char(block.ljust(CHARS_PER_BLOCK)) for block in msg_blocks]
    return encrypted

def decrypt_mh(crypto_list):
    plaintext = "".join(decrypt_N_char(num) for num in crypto_list)
    return plaintext
