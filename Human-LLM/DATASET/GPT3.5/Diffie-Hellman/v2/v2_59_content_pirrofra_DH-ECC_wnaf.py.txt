class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y

class EllipticCurve:
    def __init__(self, a, b, n):
        self.a = a
        self.b = b
        self.n = n

    def multiplicative_inverse(self, x):
        return pow(x, -1, self.n)

    def double_point(self, p):
        if p.x == 0 and p.y == 0:
            return Point(0, 0)

        lmb_numerator = (3 * p.x ** 2 + self.a) % self.n
        lmb_denominator = (2 * p.y) % self.n
        lmb = (lmb_numerator * self.multiplicative_inverse(lmb_denominator)) % self.n
        xr = (lmb ** 2 - 2 * p.x) % self.n
        yr = (lmb * (p.x - xr) - p.y) % self.n

        return Point(xr, yr)

    def point_add(self, p, q):
        if p.x == q.x and p.y == q.y:
            return self.double_point(p)
        if p.x == 0 and p.y == 0:
            return q
        if q.x == 0 and q.y == 0:
            return p

        lmb_numerator = (q.y - p.y) % self.n
        lmb_denominator = (q.x - p.x) % self.n

        if lmb_denominator == 0:
            return Point(0, 0)

        lmb = (lmb_numerator * self.multiplicative_inverse(lmb_denominator)) % self.n
        xr = (lmb ** 2 - p.x - q.x) % self.n
        yr = (lmb * (p.x - xr) - p.y) % self.n

        return Point(xr, yr)

    def scalar_multiply(self, scalar, p):
        result = Point(0, 0)
        while scalar != 0:
            if scalar & 1:
                result = self.point_add(result, p)
            p = self.double_point(p)
            scalar >>= 1
        return result

class WNAF(EllipticCurve):
    def __init__(self, a, b, n, w):
        super().__init__(a, b, n)
        self.__w = w
        self.__expw = pow(2, w)

    def __mod(self, d):
        mod = d % self.__expw
        if mod >= self.__expw // 2:
            return mod - self.__expw
        else:
            return mod

    def __NAF(self, d):
        i = 0
        naf = []
        while d > 0:
            if d % 2 == 1:
                d_i = self.__mod(d)
                naf.append(d_i)
                d = d - d_i
            else:
                naf.append(0)
            d = d // 2
            i += 1
        return (i - 1, naf)

    def __precalc(self, P):
        precalculated = {}
        i = 1
        double_p = self.double_point(P)
        Q = P
        while i <= self.__expw - 1:
            precalculated[i] = Q
            precalculated[-i] = Point(Q.x, 0 - Q.y)
            Q = self.point_add(Q, double_p)
            i += 2
        return precalculated

    def __wNAF(self, d, P):
        i, naf = self.__NAF(d)
        precalculated = self.__precalc(P)
        Q = Point(0, 0)
        while i >= 0:
            Q = self.double_point(Q)
            di = naf[i]
            if di != 0:
                Q = self.point_add(Q, precalculated[di])
            i -= 1
        return Q

    def mul(self, d, P):
        return self.__wNAF(d, P)

# Example usage
a = 2
b = 3
n = 23
w = 4
curve = WNAF(a, b, n, w)
d = 5
P = Point(6, 7)
result = curve.mul(d, P)
print("Result:", result.x, result.y)
