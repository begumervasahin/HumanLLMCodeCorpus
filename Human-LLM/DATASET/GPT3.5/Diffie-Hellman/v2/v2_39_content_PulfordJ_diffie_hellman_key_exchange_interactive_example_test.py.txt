def calculate_encrypted_value(base, prime, exponent):
    """
    Calculates the encrypted value using modular exponentiation.
    """
    return (base ** exponent) % prime

def test_encrypt_integer():
    """
    Tests the calculate_encrypted_value function.
    """
    assert calculate_encrypted_value(5, 23, 3) == 10
    assert calculate_encrypted_value(5, 23, 4) == 4

def test_shared_secret_is_equal():
    """
    Tests for equality of shared secrets between two parties.
    """
    assert calculate_encrypted_value(calculate_encrypted_value(5, 23, 4), 23, 3) == \
           calculate_encrypted_value(calculate_encrypted_value(5, 23, 3), 23, 4)
    
    assert calculate_encrypted_value(calculate_encrypted_value(5, 23, 14), 23, 17) == \
           calculate_encrypted_value(calculate_encrypted_value(5, 23, 17), 23, 14)

def test_brute_force_secret_is_equal():
    """
    Tests brute force combinations of keys for equality.
    """
    assert calculate_encrypted_value(5, 23, 3) == 10
    assert calculate_encrypted_value(5, 23, 4) == 4
    
    base = 5
    prime = 23
    for i in range(0, prime):
        for j in range(0, prime):
            shared1 = calculate_encrypted_value(calculate_encrypted_value(base, prime, i), prime, j)
            shared2 = calculate_encrypted_value(calculate_encrypted_value(base, prime, j), prime, i)
            assert shared1 == shared2

if __name__ == "__main__":
    test_encrypt_integer()
    test_shared_secret_is_equal()
    test_brute_force_secret_is_equal()
    print("All tests passed successfully!")
