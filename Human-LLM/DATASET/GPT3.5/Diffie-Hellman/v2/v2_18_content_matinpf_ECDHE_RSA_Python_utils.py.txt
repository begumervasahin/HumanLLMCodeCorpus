        return int(ensure_bytes(b)[::-1].encode('hex'), 16)

        return ('%x' % b).zfill(sz * 2).decode('hex')[::-1]

# Ensure bytes type
def ensure_bytes(s):
    if type(s) is Str:
        return s.encode('utf-8')
    elif type(s) in (Bytes, bytearray):
        return Bytes(s)
    else:
        raise TypeError

# Ensure string type
def ensure_str(s):
    if type(s) is Str:
        return s
    elif type(s) in (Bytes, bytearray):
        return Bytes(s).decode('utf-8')
    else:
        raise TypeError

# Extended greatest common divisor
def egcd(a, b):
    l, r = abs(a), abs(b)
    x, lx, y, ly = 0, 1, 1, 0
    while r:
        l, (q, r) = r, divmod(l, r)
        x, lx = lx - q*x, x
        y, ly = ly - q*y, y
    return l, -lx if a < 0 else lx, -ly if b < 0 else ly

# Modular inverse
def modinv(a, m):
    g, x, y = egcd(a, m)
    assert g == a * x + m * y
    if g != 1:
        raise ValueError
    return x % m

# Profile function execution time
def profile(f, args=None):
    t = time.time()
    if not args:
        args = []
    r = f(*args)
    e = time.time()
    return e-t, r

# Convert bytes to hexadecimal
def enhex(s):
    return binascii.hexlify(s)

# Convert hexadecimal to bytes
def unhex(s):
    return binascii.unhexlify(s)
