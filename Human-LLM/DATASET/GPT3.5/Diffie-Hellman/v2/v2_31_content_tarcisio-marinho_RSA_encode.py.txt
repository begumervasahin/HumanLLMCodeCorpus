import random

def totient(number):
    """
    Calculates the totient of a prime number.
    """
    if is_prime(number):
        return number - 1
    else:
        return False

def is_prime(n):
    """
    Checks if a number is prime.
    """
    if n <= 1:
        return False
    if n <= 3:
        return True
    if n % 2 == 0 or n % 3 == 0:
        return False
    i = 5
    while i * i <= n:
        if n % i == 0 or n % (i + 2) == 0:
            return False
        i += 6
    return True

def generate_E(totient):
    """
    Generates a random number 'E' satisfying the conditions.
    """
    while True:
        e = random.randrange(2, totient)
        if gcd(totient, e) == 1:
            return e

def generate_prime():
    """
    Generates a random prime number.
    """
    while True:
        x = random.randint(1, 100)
        if is_prime(x):
            return x

def gcd(a, b):
    """
    Calculates the greatest common divisor of two numbers.
    """
    while b != 0:
        a, b = b, a % b
    return a

def cipher(words, e, n):
    """
    Encrypts a message.
    """
    ciphered_text = []
    for letter in words:
        k = ord(letter)
        k = pow(k, e, n)
        ciphered_text.append(k)
    return ciphered_text

def decipher(ciphered_text, n, d):
    """
    Decrypts an encrypted message.
    """
    deciphered_text = []
    for ciphertext in ciphered_text:
        result = pow(ciphertext, d, n)
        deciphered_text.append(chr(result))
    return deciphered_text

def calculate_private_key(totient, e):
    """
    Calculates the private key.
    """
    d = 0
    while (d * e) % totient != 1:
        d += 1
    return d
