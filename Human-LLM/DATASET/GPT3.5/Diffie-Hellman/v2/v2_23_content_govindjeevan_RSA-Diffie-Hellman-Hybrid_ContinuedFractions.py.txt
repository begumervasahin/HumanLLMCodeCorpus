def rational_to_contfrac(x, y):
    """
    Converts a rational fraction x/y into a continued fraction [a0, ..., an].
    """
    a = x // y
    partial_quotients = [a]
    while a * y != x:
        x, y = y, x - a * y
        a = x // y
        partial_quotients.append(a)
    return partial_quotients

def convergents_from_contfrac(frac):
    """
    Computes the list of convergents from the continued fraction.
    """
    convergents = []
    for i in range(len(frac)):
        convergents.append(contfrac_to_rational(frac[0:i + 1]))
    return convergents

def contfrac_to_rational(frac):
    """
    Converts a continued fraction [a0, ..., an] into a rational number x/y.
    """
    if len(frac) == 0:
        return (0, 1)
    num = frac[-1]
    denom = 1
    for i in range(len(frac) - 2, -1, -1):
        num, denom = frac[i] * num + denom, num
    return (num, denom)

def test_continued_fraction_functions():
    """
    Tests the functionality of the continued fraction manipulation functions.
    """
    test_numbers = [(1, 1), (1, 2), (5, 15), (27, 73), (73, 27)]
    for r in test_numbers:
        num, denom = r
        print('Rational number:', r)
        contfrac = rational_to_contfrac(num, denom)
        print('Continued fraction:', contfrac)
        print('Convergents:', convergents_from_contfrac(contfrac))
        print('***********************************')

if __name__ == "__main__":
    test_continued_fraction_functions()
