import socket
from Crypto import Random
from Crypto.PublicKey import RSA

# Generate RSA keys for the client and server
client_key = RSA.generate(1024, Random.new().read)
server_key = RSA.generate(1024, Random.new().read)

# Create dictionaries to store keys
keys_for_client = {'Client': client_key.exportKey(), 'Server': server_key.publickey().exportKey()}
keys_for_server = {'Client': server_key.publickey().exportKey(), 'Server': client_key.exportKey()}

# Create a socket for the Certificate Authority (CA)
ca_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
print("Certificate Authority socket created")

# Bind the socket to a port
ca_socket.bind(('localhost', 9999))
print("Certificate Authority socket bind complete")

# Listen for incoming connections
ca_socket.listen(5)
print("Certificate Authority server is now listening...")

while True:
    # Accept connections
    connection, address = ca_socket.accept()
    print("Connected with " + address[0])

    # Receive the user name
    user = connection.recv(1024)

    # Send the appropriate key based on the user name
    if user == 'Client':
        connection.sendall(keys_for_client['Client'])
        print("Sent Client's private key to the client")
    elif user == 'Server':
        connection.sendall(keys_for_server['Server'])
        print("Sent Server's public key to the client")
    else:
        print("Access denied!")

    # Close the connection
    connection.close()

# Close the CA socket
ca_socket.close()
