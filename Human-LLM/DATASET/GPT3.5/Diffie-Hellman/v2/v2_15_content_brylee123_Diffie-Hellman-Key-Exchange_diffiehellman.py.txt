import math
import random
import getpass

def prime_generator():
    primes = []
    count = 3
    while len(primes) < 1000:
        is_prime = True
        for x in range(2, int(math.sqrt(count)) + 1):
            if count % x == 0:
                is_prime = False
                break
        if is_prime:
            primes.append(count)
        count += 1
    return primes

def get_integer_input(prompt):
    while True:
        try:
            return int(input(prompt))
        except ValueError:
            print("Not an integer. Please try again.")

if __name__ == "__main__":
    primes = prime_generator()
    g = random.choice(primes)
    primes.remove(g)
    p = random.choice(primes)

    print("Shared g value:", g)
    print("Shared p value:", p)

    a = get_integer_input("Enter hidden \"a\" value: ")
    b = get_integer_input("Enter hidden \"b\" value: ")

    A = pow(g, a, p)
    B = pow(g, b, p)

    print("\nCalculation of shared key with \"a\":\n")
    print("User assigned value of \"a\" is", a)
    print("Formula: A = g^a mod p")
    print(f"{g}^{a} mod {p} =", A, "= A")
    print("Received \"B\" value")
    print("shared_key_a = B^a mod p")
    shared_key_a = pow(B, a, p)
    print(f"{B}^{a} mod {p} =", shared_key_a)
    print("Shared Key Calculated from \"a\" is:", shared_key_a)

    print("\nCalculation of shared key with \"b\":\n")
    print("User assigned value of \"b\" is", b)
    print("Formula: B = g^b mod p")
    print(f"{g}^{b} mod {p} =", B, "= B")
    print("Received \"A\" value")
    print("shared_key_b = A^b mod p")
    shared_key_b = pow(A, b, p)
    print(f"{A}^{b} mod {p} =", shared_key_b)
    print("Shared Key Calculated from \"b\" is:", shared_key_b)

    if shared_key_a == shared_key_b:
        print("The shared keys calculated from \"a\" and \"b\" are equal.")
