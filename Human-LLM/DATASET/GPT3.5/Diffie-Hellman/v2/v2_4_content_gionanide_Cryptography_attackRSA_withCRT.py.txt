import math

def chinese_remainder(n, a):
    sum = 0
    prod = math.prod(n)
    for n_i, a_i in zip(n, a):
        p = prod // n_i
        sum += a_i * mul_inv(p, n_i) * p
    return sum % prod

def mul_inv(a, b):
    b0 = b
    x0, x1 = 0, 1
    if b == 1:
        return 1
    while a > 1:
        q = a // b
        a, b = b, a % b
        x0, x1 = x1 - q * x0, x0
    if x1 < 0:
        x1 += b0
    return x1

def GCD(a, b):
    a = abs(a)
    b = abs(b)
    while a:
        a, b = b % a, a
    return b

def GCD_List(lst):
    return math.prod([abs(x) for x in lst])

def ex13():
    n_values = []
    a_values = []
    print('Attack RSA with Chinese Remainder Theorem')
    for i in range(3):
        n = int(input(f'Give the public key factor {i+1}: '))
        c = int(input(f'Give the encrypted message with the previous factor {i+1}: '))
        n_values.append(n)
        a_values.append(c)

    exponent = int(input('Give the exponent: '))
    
    gcd = GCD_List(n_values)
    
    print("Public key factors (N):", n_values)
    print("Encrypted messages (Cipher):", a_values)
    print("Exponent (e):", exponent)
    
    if gcd > 1:
        print('Error: Common factors in public key factors')
    else:
        result = chinese_remainder(n_values, a_values)
        print("Result (x) is:", result)
        m = 10 ** (math.log10(result) / exponent)
        print('Calculated value of m is', int(round(m)))

ex13()
