from random import randint, randrange

def is_prime(number):
    if number <= 3:
        return number > 1
    if not number % 2 or not number % 3:
        return False
    for i in range(5, int(number ** 0.5) + 1, 6):
        if not number % i or not number % (i + 2):
            return False
    return True

def generate_prime_in_range(start, end):
    prime = 0
    while not is_prime(prime):
        prime = randint(start, end)
    return prime

def gcd(a, b):
    while b != 0:
        a, b = b, a % b
    return a

def extended_gcd(a, b):
    x, xx, y, yy = 1, 0, 0, 1
    while b != 0:
        q = a // b
        a, b = b, a % b
        xx, x = x - q * xx, xx
        yy, y = y - q * yy, yy
    return a, x, y

def modular_inverse(a, n):
    c, u, v = extended_gcd(a, n)
    if c != 1:
        return 0
    else:
        return u % n

print("---------------------------------")
print("----------ALGORITHME RSA---------")
print("---------------------------------\n")

# Generate two prime numbers
p = generate_prime_in_range(100000000, 999999999)
q = generate_prime_in_range(100000000, 999999999)

print("\nLes nombres premiers choisis sont:")
print("p =", p)
print("q =", q, "\n")

# Calculate n and Euler's totient function phi(n)
n = p * q
print("n = p * q =", n, "\n")
phi = (p - 1) * (q - 1)
print("fonction de Euler [phi(n) = (p-1)(q-1)] :", phi, "\n")

# Choose public key 'e'
e = 2
g = gcd(phi, e)
while g != 1:
    e = randrange(1, phi)
    g = gcd(phi, e)

# Calculate private key 'd'
d = modular_inverse(e, phi)

print("Notre clé publique est la paire des nombres (e = {}, n = {}).".format(e, n))
print("Notre clé privée est la paire des nombres (d = {}, n = {}).".format(d, n))
