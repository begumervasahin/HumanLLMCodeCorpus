import asyncio
import os
import json
import random
from socket import *
from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives.asymmetric import dh
from cryptography.hazmat.primitives.ciphers.aead import AESCCM
from Crypto.Random import get_random_bytes

class Client:
    def __init__(self, name=''):
        self.name = name

    def init_handshake(self):
        return b'PHello'

    def init_message(self):
        return f"Hello from {self.name}!".encode()

    def respond(self, msg):
        print('Received:', msg.decode())
        new_msg = input().encode()
        return new_msg

    def generate_private_key(self, parameters):
        private_key = parameters.generate_private_key()
        return private_key

    def generate_public_key(self, private_key):
        public_key = private_key.public_key()
        return public_key

    def get_public_value(self, public_key):
        key = public_key.public_numbers().y
        return key

    def generate_shared_key(self, private_key, peer_public_key):
        shared_key = private_key.exchange(peer_public_key)
        return shared_key

    def decompose_peer_key(self, pub_bob_key, pn):
        peer_public_numbers = dh.DHPublicNumbers(pub_bob_key, pn)
        peer_public_key_bob = peer_public_numbers.public_key(default_backend())
        return peer_public_key_bob

    def encrypt_message(self, key, msg):
        aesccm = AESCCM(key)
        nonce = get_random_bytes(13)
        ct = aesccm.encrypt(nonce, msg, None)
        return (nonce, ct)

    def decrypt_message(self, key, msg, nonce):
        aesccm = AESCCM(key)
        dt = aesccm.decrypt(nonce, msg, None)
        return dt

@asyncio.coroutine
def tcp_echo_client(loop=None):
    if loop is None:
        loop = asyncio.get_event_loop()
    reader, writer = yield from asyncio.open_connection('127.0.0.1', 9999, loop=loop)
    data = b'P'
    client = Client("Client 1")
    msg = client.init_handshake()
    while len(data) > 0:
        if msg:
            writer.write(msg)
            if msg[:1] == b'E':
                break
            data = yield from reader.read(1000)
            if len(data) > 0:
                if data[:1] == b'P':
                    g = int(data[1:2].decode())
                    p = int(data[2:].decode())
                    pn = dh.DHParameterNumbers(p, g)
                    parameters = pn.parameters(default_backend())
                    private_key = client.generate_private_key(parameters)
                    public_key = client.generate_public_key(private_key)
                    public_value = client.get_public_value(public_key)
                    msg = 'K' + str(public_value)
                    msg = msg.encode()
                elif data[:1] == b'K':
                    server_key = data[1:].decode()
                    peer_public_key = client.decompose_peer_key(int(data[1:].decode()), pn)
                    shared_key = client.generate_shared_key(private_key, peer_public_key)
                    msg = client.init_message()
                    nonce, msg = client.encrypt_message(shared_key[:32], msg)
                    msg = b'S' + nonce + msg
                elif data[:1] == b'S':
                    msg = client.decrypt_message(shared_key[:32], data[14:], data[1:14])
                    msg = client.respond(msg)
                    nonce, msg = client.encrypt_message(shared_key[:32], msg)
                    msg = b'S' + nonce + msg
            else:
                writer.write(b'E')
                break
        else:
            writer.write(b'E')
            break
    print('Socket closed!')
    writer.close()

loop = asyncio.get_event_loop()
loop.run_until_complete(tcp_echo_client())
