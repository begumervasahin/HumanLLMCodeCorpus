from Crypto.Cipher import AES

def recover_private_keys(p, g, pk1, pk2):
    """Recover the private keys based on public keys."""
    pr1, pr2 = 0, 0
    e = 1
    b = 0
    while True:
        if pow(g, e, p) == pk1:
            pr1 = e
            b += 1
            if b == 2:
                break
        if pow(g, e, p) == pk2:
            pr2 = e
            b += 1
            if b == 2:
                break
        e += 1
    return pr1, pr2

def decrypt_messages(pr1, pr2, g, p, m1, m2):
    """Decrypt messages using the private keys."""
    s = hex(pow(g, (pr1 * pr2), p))[2:]
    if len(s) % 2 != 0:
        s = '0' + s
    cipher = AES.new(bytes.fromhex(s))
    p1 = cipher.decrypt(m1).decode('utf-8').replace('\x00', '')
    p2 = cipher.decrypt(m2).decode('utf-8').replace('\x00', '')
    return p1, p2
