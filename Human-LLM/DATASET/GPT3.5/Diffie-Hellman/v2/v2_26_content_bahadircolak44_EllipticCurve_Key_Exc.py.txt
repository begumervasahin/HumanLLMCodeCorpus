import time
from EllipticCurve import EllipticCurve
from PointAddition import PointAddition

def point_doubling(a, b, x, y, f):
    """
    Perform point doubling on an elliptic curve.
    """
    alice_curve = EllipticCurve(a, b)
    x, y = alice_curve.sum_ee(x, y, f)
    return x, y

def point_addition(a, b, x1, y1, x2, y2, f, i):
    """
    Perform point addition on an elliptic curve.
    """
    alice_curve = PointAddition(a, b)
    x3, y3 = alice_curve.sum_ee(x1, y1, x2, y2, f)
    return x3, y3

def key_exchange(k, verbose=False):
    """
    Perform key exchange between parties.
    """
    a, b = 0, 1
    x1, y1 = 7, 5
    x2, y2 = 1, 1
    f = 11
    x2, y2 = point_doubling(a, b, x1, y1, f)
    if verbose:
        print("  P:", (x1, y1))
    for i in range(3, k + 1):
        if verbose:
            print(f"{i-1}P", (x2, y2))
        if x1 > x2:
            x2, y2 = point_addition(a, b, x2, y2, x1, y1, f, i)
        elif x1 < x2:
            x2, y2 = point_addition(a, b, x1, y1, x2, y2, f, i)
    if verbose:
        print("-------------------------------------------")
    return x2, y2

def diffie_hellman(g, k):
    """
    Perform the Diffie-Hellman key exchange.
    """
    a, b = 0, 1
    x1, y1 = g
    f = 11
    x2, y2 = point_doubling(a, b, x1, y1, f)
    for i in range(1, k - 1):
        if x1 > x2:
            x2, y2 = point_addition(a, b, x2, y2, x1, y1, f, i)
        elif x1 < x2:
            x2, y2 = point_addition(a, b, x1, y1, x2, y2, f, i)
        print(x2, y2)
    return x2, y2
