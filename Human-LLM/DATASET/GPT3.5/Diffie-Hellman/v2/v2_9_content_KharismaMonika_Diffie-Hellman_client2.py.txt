import socket
import sys
from desv4 import *
from desv4c import *

# Define keys and parameters
key_a = 3
key_q = 353

# Take input for random key
random_key = input('Enter a random key: ')
key_result = (key_a ** random_key) % key_q
key_result_str = str(key_result)
print("Generated key:", key_result)

checkpoint = 0

while True:
    checkpoint += 1
    try:
        # Create a socket
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        server_address = ('localhost', 11000)
        
        # Connect to the server
        sock.connect(server_address)
        
        # Exchange keys with the server
        if checkpoint == 1:
            server_key = sock.recv(10)
            print("Received server key:", server_key)
            sock.sendall(key_result_str.encode())
            computed_key = (int(server_key) ** random_key) % key_q
            print("Computed key:", computed_key)
            computed_key_str = str(computed_key)
            if len(computed_key_str) < 8:
                computed_key_str += ''.join(str(i) for i in range(len(computed_key_str), 8))
            print("Adjusted key:", computed_key_str)
            with open('key.txt', 'wb') as f:
                f.write(computed_key_str.encode())
        
        # Send message to the server
        message = input('Faishal : ')
        with open('plain.txt', 'wb') as f:
            f.write(message.encode())
        exec(open("desv4.py").read())
        with open('hasil.txt', 'rb') as file:
            encrypted_data = file.read()
        print('Sending message:', message)
        sock.sendall(encrypted_data)
        
        # Receive encrypted data from the server
        data = sock.recv(150)
        amount_received = 0
        amount_expected = len(data)
        while amount_received < amount_expected:
            print('Received encrypted data:', data)
            with open('terima2.txt', 'wb') as f:
                f.write(data)
            exec(open("desv4c.py").read())
            with open('dekrip2.txt', 'rb') as file:
                decrypted_data = file.read()
            amount_received += len(decrypted_data)
            print('Decrypted message:', decrypted_data.decode())
    finally:
        sock.close()
