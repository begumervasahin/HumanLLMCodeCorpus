# S-boxes
s_box = [
    0x9, 0x4, 0xa, 0xb,
    0xd, 0x1, 0x8, 0x5,
    0x6, 0x2, 0x0, 0x3,
    0xc, 0xe, 0xf, 0x7
]
s_box_inverse = [
    0xa, 0x5, 0x9, 0xb,
    0x1, 0x7, 0x8, 0xf,
    0x6, 0x0, 0x2, 0x3,
    0xc, 0x4, 0xd, 0xe
]

# Round keys
round_keys = [None] * 6

def multiply(p1, p2):
    """Multiply two polynomials in GF(2^4)/x^4 + x + 1"""
    product = 0
    while p2:
        if p2 & 0b1:
            product ^= p1
        p1 <<= 1
        if p1 & 0b10000:
            p1 ^= 0b11
        p2 >>= 1
    return product & 0b1111

def int_to_vec(n):
    """Convert a 2-byte integer into a 4-element vector"""
    return [
        n >> 12,
        (n >> 4) & 0xf,
        (n >> 8) & 0xf,
        n & 0xf
    ]

def vec_to_int(m):
    """Convert a 4-element vector into 2-byte integer"""
    return (m[0] << 12) + (m[2] << 8) + (m[1] << 4) + m[3]

def add_key(s1, s2):
    """Add two keys in GF(2^4)"""
    return [i ^ j for i, j in zip(s1, s2)]

def substitute_nibbles(s_box, s):
    """Nibble substitution function"""
    return [s_box[e] for e in s]

def shift_rows(s):
    """ShiftRow function"""
    return [s[0], s[1], s[3], s[2]]

def key_expansion(key):
    """Generate the three round keys"""
    def substitute_nibbles_swap(b):
        """Swap each nibble and substitute it using s_box"""
        return s_box[b >> 4] + (s_box[b & 0x0f] << 4)

    rcon1, rcon2 = 0b10000000, 0b00110000
    round_keys[0] = (key & 0xff00) >> 8
    round_keys[1] = key & 0x00ff
    round_keys[2] = round_keys[0] ^ rcon1 ^ substitute_nibbles_swap(round_keys[1])
    round_keys[3] = round_keys[2] ^ round_keys[1]
    round_keys[4] = round_keys[2] ^ rcon2 ^ substitute_nibbles_swap(round_keys[3])
    round_keys[5] = round_keys[4] ^ round_keys[3]

def encrypt(plaintext, key):
    """Encrypt plaintext block"""
    def mix_columns(s):
        return [
            s[0] ^ multiply(4, s[2]),
            s[1] ^ multiply(4, s[3]),
            s[2] ^ multiply(4, s[0]),
            s[3] ^ multiply(4, s[1])
        ]

    state = int_to_vec(((round_keys[0] << 8) + round_keys[1]) ^ plaintext)
    state = mix_columns(shift_rows(substitute_nibbles(s_box, state)))
    state = add_key(int_to_vec((round_keys[2] << 8) + round_keys[3]), state)
    state = shift_rows(substitute_nibbles(s_box, state))
    return vec_to_int(add_key(int_to_vec((round_keys[4] << 8) + round_keys[5]), state))

def decrypt(ciphertext, key):
    """Decrypt ciphertext block"""
    def inverse_mix_columns(s):
        return [
            multiply(9, s[0]) ^ multiply(2, s[2]),
            multiply(9, s[1]) ^ multiply(2, s[3]),
            multiply(9, s[2]) ^ multiply(2, s[0]),
            multiply(9, s[3]) ^ multiply(2, s[1])
        ]

    state = int_to_vec(((round_keys[4] << 8) + round_keys[5]) ^ ciphertext)
    state = substitute_nibbles(s_box_inverse, shift_rows(state))
    state = inverse_mix_columns(add_key(int_to_vec((round_keys[2] << 8) + round_keys[3]), state))
    state = substitute_nibbles(s_box_inverse, shift_rows(state))
    return vec_to_int(add_key(int_to_vec((round_keys[0] << 8) + round_keys[1]), state))

def main():
    key = 0x5467  # Example key
    plaintext = 0x1234  # Example plaintext

    # Generate round keys
    key_expansion(key)

    # Encrypt plaintext
    ciphertext = encrypt(plaintext, key)
    print("Encrypted ciphertext:", hex(ciphertext))

    # Decrypt ciphertext
    decrypted_plaintext = decrypt(ciphertext, key)
    print("Decrypted plaintext:", hex(decrypted_plaintext))

if __name__ == "__main__":
    main()
