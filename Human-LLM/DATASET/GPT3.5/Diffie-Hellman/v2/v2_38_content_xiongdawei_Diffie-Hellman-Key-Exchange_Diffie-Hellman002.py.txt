import random

def euler_totient_function(n):
    """Calculate Euler's Totient Function."""
    count = 0
    for i in range(1, n):
        if greatest_common_divisor(i, n) == 1:
            count += 1
    return count

def greatest_common_divisor(a, b):
    """Calculate the Greatest Common Divisor."""
    while a % b != 0:
        a, b = b, (a % b)
    return b

def find_primitive_root(n):
    """Find the primitive root of a number."""
    phi = euler_totient_function(n)
    primitive_roots = []
    for i in range(1, n):
        if greatest_common_divisor(i, n) == 1:
            prime_list = []
            for k in range(1, phi + 1):
                prime_list.append(i**k % n)
            prime_list.sort()
            if prime_list == list(range(1, phi + 1)):
                primitive_roots.append(i)
    if len(primitive_roots) != 0:
        return n, primitive_roots

def discrete_logarithm(a, p, result):
    """Find the discrete logarithm."""
    ans = 0
    while (a**ans) % p != result:
        ans += 1
    return ans

    Solve the key using brute force.
    """
    ansa, ansb = 1, 1
    while (primitive_root**ansa) % prime != keya:
        ansa += 1
    while (primitive_root**ansb) % prime != keyb:
        ansb += 1
    shared_key = (keya**ansb) % prime
    return ansa, ansb, shared_key
