import socket
import random

# Establish a socket connection
client_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
HOST = '127.0.0.1'
PORT = 5050
client_socket.connect((HOST, PORT))

# Receive prime number (q) and primitive root (alpha) from the server
q = int(client_socket.recv(1024))
alpha = int(client_socket.recv(1024))

# Generate a random private key (Xb)
Xb = random.randint(1, q)

# Calculate public key (Yb) using the Diffie-Hellman key exchange formula
Yb = pow(alpha, Xb, q)
print("Generated Public Key (Yb):", Yb)

# Send Yb to the server
client_socket.send(str(Yb).encode())

# Receive Ya from the server
Ya = int(client_socket.recv(1024))

# Calculate the symmetric key (Kb) using the received Ya and Xb
Kb = pow(Ya, Xb, q)
print("Shared Symmetric Key (Kb):", Kb)

# Close the connection
client_socket.close()
