import random
import time

def compute_shared(g, p, k):
    """Compute the shared secret using Diffie-Hellman key exchange."""
    return (g ** k) % p

def compute_random():
    """Generate a random number between 0 and 999."""
    return random.randint(0, 999)

def sophie_germain(p, n):
    """Generate a Sophie Germain prime number."""
    for i in range(n):
        p = (2 * p) + 1
    return p
