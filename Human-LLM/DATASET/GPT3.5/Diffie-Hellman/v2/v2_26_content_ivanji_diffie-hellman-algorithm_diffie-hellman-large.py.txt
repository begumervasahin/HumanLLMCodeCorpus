import random
import math

# Miller-Rabin primality test
def rabinMiller(n):
    s = n - 1
    t = 0
    
    # Factor out powers of 2
    while s & 1 == 0:
        s = s / 2
        t += 1
    
    # Perform the test for 128 random values of 'a'
    for _ in range(128):
        a = random.randrange(2, n - 1)
        v = pow(a, s, n)
        if v != 1:
            i = 0
            while v != (n - 1):
                if i == t - 1:
                    return False
                else:
                    i += 1
                    v = (v ** 2) % n
    return True

# Generate a large prime number
def generateLargePrime(k):
    r = 100 * (math.log(k, 2) + 1)
    r_ = r
    while r > 0:
        n = random.randrange(2 ** (k - 1), 2 ** k)
        r -= 1
        if isPrime(n):
            return n
    return "Failure after " + str(r_) + " tries."
