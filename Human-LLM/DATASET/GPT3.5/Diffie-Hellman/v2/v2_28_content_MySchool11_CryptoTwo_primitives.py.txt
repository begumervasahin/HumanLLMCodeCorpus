from random import sample

def calculate_prime_primitives(prime):
    print("Calculating the primitive roots...")
    primitive_roots = []

    # Iterate through numbers to check for primitive roots
    for candidate_root in range(1, prime):
        residues = set()

        # Calculate residues for each candidate root
        for exponent in range(1, prime):
            residue = (candidate_root ** exponent) % prime
            residues.add(residue)

            # If all residues are unique, it's a primitive root
            if len(residues) == prime - 1:
                primitive_roots.append(candidate_root)
                break

    # Randomly select one primitive root from the list
    return sample(primitive_roots, 1)[0]

def main():
    prime = int(input("Enter the prime number: "))
    primitive_root = calculate_prime_primitives(prime)
    print(f"The primitive root for the prime number {prime} is: {primitive_root}")

if __name__ == "__main__":
    main()
