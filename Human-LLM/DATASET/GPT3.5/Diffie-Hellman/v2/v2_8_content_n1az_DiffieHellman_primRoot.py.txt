from math import gcd as bltin_gcd

def is_prime(n):
    """Check if a number is prime."""
    if n <= 1:
        return False
    if n <= 3:
        return True
    if n % 2 == 0 or n % 3 == 0:
        return False

    i = 5
    while i * i <= n:
        if n % i == 0 or n % (i + 2) == 0:
            return False
        i += 6
    return True

def primitive_roots(in_prime):
    """Find the primitive roots of a prime number."""
    required_set = {num for num in range(1, in_prime) if bltin_gcd(num, in_prime)}
    return [g for g in range(1, in_prime) if required_set == {pow(g, powers, in_prime)
            for powers in range(1, in_prime)}]

if __name__ == '__main__':
    print('Enter a prime number:')
    x = int(input())
    checked_message = 'Checked: Number is a PRIME'
    warning_message = 'Warning: Number is not a PRIME'
    
    if is_prime(x):
        print(checked_message)
        print('Primitive root/roots:')
        print(primitive_roots(x))
    else:
        print(warning_message)
