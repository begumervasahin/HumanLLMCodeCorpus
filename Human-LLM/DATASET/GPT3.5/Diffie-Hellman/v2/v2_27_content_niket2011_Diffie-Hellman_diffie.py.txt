import random

def check_for_duplicates(pri):
    duplicate = pri[:]
    check = 1
    for i in range(len(pri)):
        for j in range(i + 1, len(pri)):
            if pri[i] == duplicate[j]:
                check = 0
                break
            else:
                check = 1
                continue
        break
    return check

def main():
    prime_number = int(input("Enter the prime number: "))
    num_users = int(input("Enter the number of users: "))
    secret_keys = []

    for i in range(num_users):
        key = int(input("Enter secret key for user {}: ".format(i + 1)))
        secret_keys.append(key)

    if num_users > 2:
        print("Enter the users between whom you want to communicate:")
        user1_index = int(input("Enter the user 1: "))
        user2_index = int(input("Enter the user 2: "))
        xa = secret_keys[user1_index - 1]
        xb = secret_keys[user2_index - 1]
    else:
        xa = secret_keys[0]
        xb = secret_keys[1]

    unique_exponents = []
    for i in range(1, prime_number):
        exponents = []
        for j in range(1, prime_number):
            exponents.append((i**j) % prime_number)
        if check_for_duplicates(exponents) == 1:
            unique_exponents.append(i)

    alpha = random.choice(unique_exponents)
    print("alpha:", alpha)

    ya = (alpha**xa) % prime_number
    print("Public key of A:", ya)

    yb = (alpha**xb) % prime_number
    print("Public key of B:", yb)

    Kab1 = (yb**xa) % prime_number
    print("Shared session key with respect to user A:", Kab1)

    Kab2 = (ya**xb) % prime_number
    print("Shared session key with respect to user B:", Kab2)

if __name__ == "__main__":
    main()
