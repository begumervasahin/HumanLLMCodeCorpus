class KeyHandler:
    def __init__(self, private_key, public_key, n):
        """
        Initializes a KeyHandler object with private key, public key, and n value.

        Args:
            private_key (int): The private key.
            public_key (int): The public key.
            n (int): The verification circle value.
        """
        self.private_key = private_key
        self.public_key = public_key
        self.n = n

    def compute_combined_key(self):
        """
        Computes the combined key using the private and public keys.

        Returns:
            int: The computed combined key.
        """
        return pow(self.public_key, self.private_key, self.n)

    def verify_combined_key(self, combined_key):
        """
        Verifies the combined key using the private key.

        Args:
            combined_key (int): The combined key to be verified.

        Returns:
            int: The verification result.
        """
        return pow(combined_key, self.private_key, self.n)

def main():
    # Define parameters
    n = 100000
    private_key_a = 3426158395
    private_key_b = 3479502846
    public_key_g = 6748390294

    # Print parameter values
    print('Public Key:', public_key_g)
    print('Private Key a:', private_key_a)
    print('Private Key b:', private_key_b)
    print('Verification Circle n:', n)

    # Create KeyHandler instances
    key_handler_a = KeyHandler(private_key_a, public_key_g, n)
    key_handler_b = KeyHandler(private_key_b, public_key_g, n)

    # Compute combined keys
    combined_key_a = key_handler_a.compute_combined_key()
    combined_key_b = key_handler_b.compute_combined_key()

    # Print computed combined keys
    print('Combined Key of Private a and Public:', combined_key_a)
    print('Combined Key of Private b and Public:', combined_key_b)

    # Verify combined keys
    verification_result_agb = key_handler_a.verify_combined_key(combined_key_b)
    verification_result_bga = key_handler_b.verify_combined_key(combined_key_a)

    # Print verification results
    print('Verification Result for Combined Key agb:', verification_result_agb)
    print('Verification Result for Combined Key bga:', verification_result_bga)

if __name__ == '__main__':
    main()
