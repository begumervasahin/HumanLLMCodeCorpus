import unittest

def remove_spaces(s):
    """
    Removes spaces from a given string recursively.
    """
    if s is None:
        return None
    elif s == "":
        return ""
    elif s[0] == " ":
        return remove_spaces(s[1:])
    else:
        return s[0] + remove_spaces(s[1:])

def is_palindrome(s):
    """
    Checks if a string is a palindrome recursively.
    """
    if s is None:
        return False
    elif s == "":
        return True
    elif len(s) == 1:
        return True
    elif s[0] == s[-1]:
        return is_palindrome(s[1:-1])
    else:
        return False

        self.assertEqual(remove_spaces(""), "")
