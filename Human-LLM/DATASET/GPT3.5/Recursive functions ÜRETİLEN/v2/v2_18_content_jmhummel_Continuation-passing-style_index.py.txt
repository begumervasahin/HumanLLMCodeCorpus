def fact_rec(n):
    """Recursive factorial calculation."""
    if n == 0:
        return 1
    else:
        return n * fact_rec(n - 1)

def fact_cps(n, cont):
    """Factorial calculation using continuation-passing style."""
    if n == 0:
        return cont(1)
    else:
        return fact_cps(n - 1, lambda value: cont(n * value))

def end_cont(n):
    """Continuation function for the end of a calculation."""
    return n

def trampoline(f, *args):
    """Trampoline function to handle tail-recursive calls."""
    v = f(*args)
    while callable(v):
        v = v()
    return v

def fact_cps_thunked(n, cont):
    """Factorial calculation using continuation-passing style with thunks."""
    if n == 0:
        return cont(1)
    else:
        return lambda: fact_cps_thunked(
            n - 1,
            lambda value: lambda: cont(n * value)
        )

def fib_rec(n):
    """Recursive Fibonacci calculation."""
    if n <= 2:
        return 1
    else:
        return fib_rec(n - 1) + fib_rec(n - 2)

def fib_cps(n, cont):
    """Fibonacci calculation using continuation-passing style."""
    if n <= 2:
        return cont(1)
    else:
        return fib_cps(
            n - 1,
            lambda value1: fib_cps(
                n - 2,
                lambda value2: cont(value1 + value2)
            )
        )

def fib_cps_thunked(n, cont):
    """Fibonacci calculation using continuation-passing style with thunks."""
    if n <= 2:
        return cont(1)
    else:
        return lambda: fib_cps_thunked(
            n - 1,
            lambda value1: fib_cps_thunked(
                n - 2,
                lambda value2: cont(value1 + value2)
            )
        )

def main():
    """Main function to execute factorial and Fibonacci calculations."""
    try:
        print(fact_rec(1000))
    except RecursionError as e:
        print(f'RecursionError: {e}')
    try:
        print(fact_cps(1000, end_cont))
    except RecursionError as e:
        print(f'RecursionError: {e}')
    try:
        print(trampoline(fact_cps_thunked(1000, end_cont)))
    except RecursionError as e:
        print(f'RecursionError: {e}')
    try:
        print(fib_cps(43, end_cont))
    except RecursionError as e:
        print(f'RecursionError: {e}')
    try:
        print(trampoline(fib_cps_thunked(43, end_cont)))
    except RecursionError as e:
        print(f'RecursionError: {e}')

if __name__ == '__main__':
    main()
