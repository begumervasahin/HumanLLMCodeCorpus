import numpy as np
from scipy.special import spherical_jn
from scipy import integrate

def integrand(x, l, n):
    """Define the integrand function."""
    return x**n * spherical_jn(l, x)

x_low = 2
x_high = 4
l = 4
n = 3

# Numerical integration
val, err = integrate.quad(integrand, x_low, x_high, args=(l, n))

# Custom integration (dummy implementation)
def custom_integrand(dummy_obj, x, params):
    """Custom integrand function."""
    return x**params[1] * spherical_jn(params[0], x)

y_low = custom_integrand(None, x_low, (l, n))
y_high = custom_integrand(None, x_high, (l, n))

print("Testing Integration")
print("X in [{}, {}]".format(x_low, x_high))
print("l =", l)
print("n =", n)
print("Recursive Integral: ", y_high - y_low)
print("Numerical Integral: ", val, " +/- ", err)
