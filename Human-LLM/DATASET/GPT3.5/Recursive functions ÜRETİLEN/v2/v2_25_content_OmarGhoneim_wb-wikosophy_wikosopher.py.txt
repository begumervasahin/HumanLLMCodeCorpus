import requests
from bs4 import BeautifulSoup
import regex
import time

def get_to_philosophy(url, cache=[]):
    # Append the current URL to the cache
    cache.append(url)
    
    # Define excluded classes and IDs to filter out from the HTML content
    excluded_classes = ["new", "extiw", 'nowrap', 'IPA', 'reference', 'mw-selflink selflink']
    excluded_ids = ["coordinates"]
    
    # Add a delay to avoid overwhelming the server with requests
    time.sleep(0.5)
    
    # Send a GET request to the URL
    response = requests.get(url)
    
    # Extract the text content
    html_text = response.text
    
    # Parse the HTML using BeautifulSoup
    soup = BeautifulSoup(html_text, 'lxml')
    
    # Flag to check if there are no wikilinks in the article
    no_wikilinks = True
    
    # Iterate through paragraphs in the main content area
    for paragraph in soup.find(class_='mw-parser-output').find_all('p', recursive=False):
        # Remove elements with excluded classes
        for element in paragraph.find_all(class_=excluded_classes):
            element.extract()
        
        # Remove elements with excluded IDs
        for element in paragraph.find_all(id=excluded_ids):
            element.extract()
        
        # Remove italic text
        paragraph = BeautifulSoup(regex.sub(r'(?<!_)(?<rec>\((?:[^()]++|(?&rec))*\))', '', str(paragraph)), features="lxml")
        for italic in paragraph.find_all('i'):
            italic.extract()
        
        # Check if the paragraph contains a link
        try:
            link_suffix = paragraph.a.get('href')
        except:
            continue
        
        # If a link is found, set the flag to False and break the loop
        no_wikilinks = False
        break

    # If no wikilinks are found, return the current URL
    if no_wikilinks:
        print("No outgoing wikilinks in the article:", url[30:])
        return url
    else:
        link = "https://en.wikipedia.org" + link_suffix
        print(link)

    # If the link leads to the Philosophy page, print a message and return the link
    if link == "https://en.wikipedia.org/wiki/Philosophy":
        print("Wikosopher found it, Just as expected !")
        return link
    # If the link is already in the cache, print a message and return the link
    elif link in cache:
        print("Stuck in a loop !")
        return link
    # Otherwise, recursively call the function with the new link
    else:
        return get_to_philosophy(link, cache)

# Define a random Wikipedia URL
random_url = "https://en.wikipedia.org/wiki/Special:Random"

# Call the function with the random URL
get_to_philosophy(random_url)
