import socket
import sys

def client():
    # Extract command line arguments
    if len(sys.argv) != 3:
        print("Usage: python client.py <server_host> <file_name>")
        return

    rs_host = sys.argv[1]
    file_name = sys.argv[2]

    # Create a socket
    try:
        client_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    except socket.error as err:
        print("Socket open error:", err)
        return

    # Open the file to read hostnames
    try:
        with open(file_name, "r") as file:
            hostnames = file.readlines()
    except IOError as err:
        print("File open error:", err)
        print("Please ensure the desired file to reverse exists in the source folder")
        client_socket.close()
        return

    # Resolve the IP address of the remote server
    try:
        server_ip = socket.gethostbyname(rs_host)
        port = 50008
    except socket.error as err:
        print("Socket error:", err)
        client_socket.close()
        return

    # Connect to the server
    try:
        server_address = (server_ip, port)
        client_socket.connect(server_address)
    except socket.error as err:
        print("Connect error:", err)
        client_socket.close()
        return

    # Open a file to write the resolved hostnames
    with open("RESOLVED.txt", "w") as output_file:
        # Send each hostname to the server and write the resolved IP address to the file
        for hostname in hostnames:
            hostname = hostname.strip().encode('utf-8')
            client_socket.send(hostname)
            data_from_server = client_socket.recv(100).decode('utf-8')
            if not data_from_server:
                break
            output_file.write(data_from_server + '\n')

    # Close the socket
    client_socket.close()

# Run the client function
client()
