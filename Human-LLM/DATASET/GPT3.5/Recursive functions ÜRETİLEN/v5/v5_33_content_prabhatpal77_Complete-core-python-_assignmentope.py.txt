# Initialize variables
initial_x = 1
initial_y = 2
initial_z = 3

# Perform operations
x = initial_x + 5  # Increment x by 5
y = initial_y * 10  # Multiply y by 10
z = initial_z >> 6  # Right shift z by 6 bits

# Print results
print("x after incrementing by 5:", x)
print("y after multiplying by 10:", y)
print("z after right shifting by 6 bits:", z)
