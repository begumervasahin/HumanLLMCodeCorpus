import datetime

fib_cache = {}

def fib(x):
    start_time = datetime.datetime.now()
    
    if x == 1:
        value = 1
    elif x == 2:
        value = 1
    elif x > 2:
        if x in fib_cache:
            value = fib_cache[x]
        else:
            value = fib(x - 1) + fib(x - 2)
            fib_cache[x] = value
            
    return value

def main():
    x = int(input("Enter the number of Fibonacci terms: "))
    for i in range(1, x + 1):
        print(i, ":", fib(i))
        
    stop_time = datetime.datetime.now()
    dt = stop_time - stop_time
    print("Execution time:", dt)

if __name__ == "__main__":
    main()
