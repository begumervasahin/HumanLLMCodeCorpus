# Declare a Python dictionary to work as a cache for storing and searching Fibonacci calculations
fibonacci_cache = {}

# Define the Fibonacci function
def fibonacci(num):
    # Check if the input is an integer
    if type(num) != int:
        return False
    
    # Check if the Fibonacci number is already in the cache
    if num in fibonacci_cache:
        return fibonacci_cache[num]
    
    # Base cases for Fibonacci sequence
    if num == 0:
        return 0
    elif num == 1:
        return 1
    else:
        # Calculate Fibonacci recursively and store the result in the cache
        value = fibonacci(num - 1) + fibonacci(num - 2)
        fibonacci_cache[num] = value
        return value

# Print the first 1000 Fibonacci numbers
for num in range(1000):
    print(num, ":", fibonacci(num))
