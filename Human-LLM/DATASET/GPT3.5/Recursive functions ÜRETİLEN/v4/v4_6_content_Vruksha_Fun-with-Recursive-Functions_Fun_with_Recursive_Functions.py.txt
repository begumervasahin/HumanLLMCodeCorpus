# Function to replace sep1 with sep2 in a string
def replace_separator(my_string, sep1, sep2):
    if not my_string:
        return ""
    elif my_string[:len(sep1)] == sep1:
        return sep2 + replace_separator(my_string[len(sep1):], sep1, sep2)
    else:
        return my_string[0] + replace_separator(my_string[1:], sep1, sep2)

# Function to count the total number of siblings
def count_siblings(number_elves):
    if number_elves == 0:
        return 0
    if number_elves % 2 == 1:
        return 3 + count_siblings(number_elves - 1)
    if number_elves % 2 == 0:
        return 1 + count_siblings(number_elves - 1)

# Function to extract substring enclosed within brackets
def extract_substring(string):
    if string[:1] != "(":
        return extract_substring(string[1:])
    elif string[len(string)-1:] != ")":
        new_string = string[:string[len(string)-1]]
        return extract_substring(new_string)
    else:
        return string

# Function to check if a sequence is a palindrome
def is_palindrome(sequence):
    sequence = sequence.split()
    sequence = ''.join(sequence)
    if len(sequence) < 2:
        return True
    if sequence[0] != sequence[-1]:
            return False
    return is_palindrome(sequence[1:-1])
