nMax = -1

# Keep asking for input until a positive integer is entered
while nMax <= 0:
    s = input('Enter a positive integer: ')
    
    try:
        nMax = int(s)
    except ValueError:
        print('Invalid input')
    
    print()

f1 = 1
f2 = 2
n = 1

# Print Fibonacci sequence up to nMax terms
while n <= nMax:
    print('%4d: %d' % (n, f1))
    temp = f1 + f2
    f1 = f2
    f2 = temp
    n += 1
