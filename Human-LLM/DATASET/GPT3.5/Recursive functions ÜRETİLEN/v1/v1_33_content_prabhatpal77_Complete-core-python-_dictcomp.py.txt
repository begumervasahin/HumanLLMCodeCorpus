# Dictionary comprehension to create dictionary x where keys are p and values are p*p for p in range(10)
x = {p: p*p for p in range(10)}
print(x)

# Dictionary comprehension to create dictionary y where keys are q and values are q**2 for q in range(10, 20) if q is even
y = {q: q**2 for q in range(10, 20) if q % 2 == 0}
print(y)

# Dictionary comprehension to create dictionary z where keys are r and values are r*2 for r in range(20, 30) if r is odd
z = {r: r*2 for r in range(20, 30) if r % 2 != 0}
print(z)
