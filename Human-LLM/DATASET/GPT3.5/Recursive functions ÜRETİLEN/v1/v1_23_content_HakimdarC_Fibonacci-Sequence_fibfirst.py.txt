import datetime

fib_cache = {}

def fib(x):
    start_time = datetime.datetime.now()
    if x == 1:
        value = 1
    if x == 2:
        value = 1
    elif x > 2:
        value = fib(x - 1) + fib(x - 2)
        fib_cache[x] = value
    return value

x = int(input("Give the number of Fibonacci terms: "))
for i in range(1, x + 1):
    print(i, ":", fib(i))

stop_time = datetime.datetime.now()
dt = stop_time - start_time
print(dt)
