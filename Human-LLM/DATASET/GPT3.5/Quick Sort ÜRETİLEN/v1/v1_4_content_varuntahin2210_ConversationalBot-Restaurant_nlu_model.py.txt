from rasa_nlu.training_data import load_data
from rasa_nlu.config import RasaNLUModelConfig
from rasa_nlu.model import Trainer
from rasa_nlu.model import Metadata, Interpreter
from rasa_nlu import config
from rasa_nlu.components import ComponentBuilder

builder = ComponentBuilder(use_cache=True)

def train_nlu(data, config_file, model_dir):
    # Load the training data from the specified file
    training_data = load_data(data)
    
    # Initialize a Trainer with the specified configuration file and builder
    trainer = Trainer(config.load(config_file), builder)
    
    # Train the NLU model using the loaded training data
    trainer.train(training_data)
    
    # Persist the trained model to the specified model directory
    model_directory = trainer.persist(model_dir, fixed_model_name='restaurantnlu')

def run_nlu():
    # Load the trained model from the specified directory
    interpreter = Interpreter.load('./models/nlu/default/restaurantnlu', builder)
    
    # Parse a user input using the loaded model and print the result
    print(interpreter.parse("can you please suggest food"))

if __name__ == '__main__':
    # Train the NLU model using the specified training data and configuration file
    train_nlu('./data/data.json', 'config_spacy.json', './models/nlu')
    
    # Use the trained model to parse a user input
    run_nlu()
