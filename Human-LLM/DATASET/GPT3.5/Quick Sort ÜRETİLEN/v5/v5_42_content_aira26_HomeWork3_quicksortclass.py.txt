import random
import time
import matplotlib.pyplot as plt

class Quicksort:
    def __init__(self):
        self.mylist1 = []
        self.size = 0

    def add(self, value):
        self.mylist1.append(value)
        self.size += 1

    def quick_sort(self, arr):
        if len(arr) > 1:
            pivot = arr[0]
            left = [x for x in arr[1:] if x < pivot]
            right = [x for x in arr[1:] if x >= pivot]
            return self.quick_sort(left) + [pivot] + self.quick_sort(right)
        else:
            return arr

    def get_min(self):
        new = self.quick_sort(self.mylist1)
        return new[0]

    def get_max(self):
        new = self.quick_sort(self.mylist1)
        return new[-1]

    def get_list(self):
        return self.mylist1

def measure_time(obj, num_list):
    tot_time_add, tot_time_min, tot_time_max = 0, 0, 0
    for num in num_list:
        start = time.time()
        obj.add(num)
        tot_time_add += (time.time() - start)

        start = time.time()
        min_val = obj.get_min()
        tot_time_min += (time.time() - start)

        start = time.time()
        max_val = obj.get_max()
        tot_time_max += (time.time() - start)

    return tot_time_add, tot_time_min, tot_time_max

    repetitions = 5
    max_operations = 500
    step = 100
    values_heap_add, values_heap_min, values_heap_max = [], [], []

    for rounds in range(step, max_operations, step):
        num_list = [random.randint(0, 500) for _ in range(rounds)]

        tot_time_add, tot_time_min, tot_time_max = 0, 0, 0

        for repetition in range(repetitions):
            obj = Quicksort()
            time_add, time_min, time_max = measure_time(obj, num_list)
            tot_time_add += time_add
            tot_time_min += time_min
            tot_time_max += time_max

        avg_time_add = tot_time_add / repetitions
        avg_time_min = tot_time_min / repetitions
        avg_time_max = tot_time_max / repetitions

        values_heap_add.append(avg_time_add * 1000)
        values_heap_min.append(avg_time_min * 1000)
        values_heap_max.append(avg_time_max * 1000)

    xlabels = range(step, max_operations, step)

    plt.plot(xlabels, values_heap_add, label='Add')
    plt.plot(xlabels, values_heap_min, label='Get Min')
    plt.plot(xlabels, values_heap_max, label='Get Max')

    plt.legend()
    plt.xlabel("Number of Operations")
