def partition(arr, start, end):
    """
    Partition the array based on the pivot element.

    Args:
        arr (list): The array to be partitioned.
        start (int): The starting index of the partition.
        end (int): The ending index of the partition.

    Returns:
        int: The index of the pivot element after partitioning.
    """
    pivot = arr[start]
    left = start + 1
    right = end

    while True:
        while left <= right and arr[left] <= pivot:
            left += 1
        while left <= right and arr[right] > pivot:
            right -= 1
        if left > right:
            break
        arr[left], arr[right] = arr[right], arr[left]

    arr[start], arr[right] = arr[right], pivot
    return right

def quick_sort(arr, start, end):
    """
    Sort the array using the Quick Sort algorithm.

    Args:
        arr (list): The array to be sorted.
        start (int): The starting index of the array.
        end (int): The ending index of the array.
    """
    if start < end:
        pivot_index = partition(arr, start, end)
        quick_sort(arr, start, pivot_index - 1)
        quick_sort(arr, pivot_index + 1, end)
