import datetime
import statistics

def quick_sort(arr):
    """
    Sorts a list using the quick sort algorithm.

    Args:
    - arr (list): The list to be sorted.

    Returns:
    - list: The sorted list.
    """
    # Base case: return the list if it has 1 or fewer elements
    if len(arr) <= 1:
        return arr

    # Select pivot element as the median of three random elements
    pivot = statistics.median([arr[0], arr[len(arr)//2], arr[len(arr)-1]])

    # Partition the array into three parts: left, middle, and right
    left = [x for x in arr if x < pivot]
    middle = [x for x in arr if x == pivot]
    right = [x for x in arr if x > pivot]

    # Recursively sort the left and right parts
    return quick_sort(left) + middle + quick_sort(right)

# Test case
test = [21, 4, 1, 3, 9, 20, 25, 6, 21, 14]
start = datetime.datetime.now()
sorted_test = quick_sort(test)
finish = datetime.datetime.now()

# Output sorted list and time taken for sorting
print("Sorted list:", sorted_test)
print("Time taken:", finish - start)
