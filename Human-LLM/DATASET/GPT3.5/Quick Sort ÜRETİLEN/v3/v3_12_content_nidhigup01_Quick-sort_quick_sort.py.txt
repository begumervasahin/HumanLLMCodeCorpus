import datetime
import statistics

def quick_sort(arr):
    # Base case: if the list has one or fewer elements, it's already sorted
    if len(arr) <= 1:
        return arr

    # Select pivot as the median of three random elements from the list
    pivot = choose_pivot(arr)

    # Partition the list into three parts: items less than, equal to, and greater than the pivot
    left, middle, right = partition(arr, pivot)

    # Recursively sort the left and right parts, then concatenate them with the middle part
    return quick_sort(left) + middle + quick_sort(right)

def choose_pivot(arr):
    # Select the pivot as the median of three random elements from the list
    items = [arr[0], arr[len(arr)//2], arr[-1]]
    return statistics.median(items)

def partition(arr, pivot):
    # Partition the list into three parts based on the pivot
    left = [x for x in arr if x < pivot]
    middle = [x for x in arr if x == pivot]
    right = [x for x in arr if x > pivot]
    return left, middle, right

def test_quick_sort():
    # Test the quick_sort function
    test = [21, 4, 1, 3, 9, 20, 25, 6, 21, 14]

    # Measure the time taken to sort the list
    start_time = datetime.datetime.now()
    sorted_list = quick_sort(test)
    end_time = datetime.datetime.now()

    # Print the sorted list and the time taken for sorting
    print("Sorted list:", sorted_list)
    print("Time taken:", end_time - start_time)

if __name__ == "__main__":
    test_quick_sort()
