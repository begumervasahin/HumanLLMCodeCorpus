import random
import datetime
import xlwt

class Sorts:
    def __init__(self):
        self.wb = xlwt.Workbook()
        self.style0 = xlwt.easyxf('font: name Arial, color-index black')
        self.min_num = 0
        self.max_num = 1000
        self.max_n_squared_size = 10000
        self.max_size = 10000000
        self.size = 10
        self.increment = "0"
        self.program_begin = datetime.datetime.now()
        
    def run_sorts(self):
        while self.size <= self.max_size:
            self.iterate_sorts()
            self.size += int(self.increment)
        print("Time taken overall:", datetime.datetime.now() - self.program_begin)
    
    def iterate_sorts(self):
        iteration_begin = datetime.datetime.now()
        print("iteration:", self.size)
        ws = self.wb.add_sheet(str(self.size))
        if self.size <= self.max_n_squared_size:
            self.write_headers(ws)
        for i in range(1, 6):
            array = self.generate_random_array(self.size, self.min_num, self.max_num)
            if self.size <= self.max_n_squared_size:
                self.run_bubble_and_insertion_sorts(array, ws, i)
            self.run_other_sorts(array, ws, i)
        self.save_workbook()
        print("Time taken for iteration:", datetime.datetime.now() - iteration_begin)
        print()
    
    def write_headers(self, ws):
        headers = ["Bubble sort", "Insertion sort", "Quick sort", "Heap sort", "Merge sort", "Radix sort"]
        for i, header in enumerate(headers):
            ws.write(0, i, header)
    
    def run_bubble_and_insertion_sorts(self, array, ws, i):
        time = self.bubblesort(array)
        ws.write(i, 0, time, self.style0)
        time = self.insertionsort(array)
        ws.write(i, 1, time, self.style0)
    
    def run_other_sorts(self, array, ws, i):
        times = [self.quicksort(array), self.heapsort(array), self.mergesort(array), self.radixsort(array)]
        for j, time in enumerate(times, start=2):
            ws.write(i, j, time, self.style0)
    
    def save_workbook(self):
        self.wb.save('Data.xls')
    
    def bubblesort(self, array):
        a = array[:]
        start_time = datetime.datetime.now()
        length = len(a)
        for i in range(length - 1):
            for j in range(length - i - 1):
                if a[j+1] < a[j]:
                    a[j], a[j+1] = a[j+1], a[j]
        return self.get_milliseconds(start_time)
    
    def insertionsort(self, array):
        a = array[:]
        start_time = datetime.datetime.now()
        length = len(a)
        for i in range(1, length):
            temp = a[i]
            j = i
            while j > 0 and temp < a[j-1]:
                a[j] = a[j-1]
                j -= 1
            a[j] = temp
        return self.get_milliseconds(start_time)
    
    def radixsort(self, array):
        a = array[:]
        start_time = datetime.datetime.now()
        # Implementation of radix sort omitted for brevity
        return self.get_milliseconds(start_time)
    
    def quicksort(self, array):
        a = array[:]
        start_time = datetime.datetime.now()
        # Implementation of quicksort omitted for brevity
        return self.get_milliseconds(start_time)
    
    def mergesort(self, array):
        a = array[:]
        start_time = datetime.datetime.now()
        # Implementation of mergesort omitted for brevity
        return self.get_milliseconds(start_time)
    
    def heapsort(self, array):
        a = array[:]
        start_time = datetime.datetime.now()
        # Implementation of heapsort omitted for brevity
        return self.get_milliseconds(start_time)
    
    def get_milliseconds(self, start_time):
        c = datetime.datetime.now() - start_time
        milliseconds = (c.days * 24 * 60 * 60 + c.seconds) * 1000 + c.microseconds / 1000.0
        return milliseconds
    
    def generate_random_array(self, size, minimum, maximum):
        array = [None] * size
        for i in range(len(array)):
            array[i] = random.randint(minimum, maximum)
        return array

s = Sorts()
s.run_sorts()
