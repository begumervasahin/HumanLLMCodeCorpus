from Student import Student

# Global variables to track comparison and swap times
comparison_times = 0
swap_times = 0

def bubble_sort(arr):
    """Sorts the array using Bubble Sort algorithm."""
    global comparison_times, swap_times
    swapped = False
    for i in range(len(arr) - 1):
        comparison_times += 1
        if int(arr[i].rating) < int(arr[i + 1].rating):
            arr[i], arr[i + 1] = arr[i + 1], arr[i]
            swap_times += 1
            swapped = True
    if swapped:
        bubble_sort(arr)
    else:
        print("BubbleSort  \nComparison times: ", comparison_times, "\nSwap times: ", swap_times)
        # Reset global variables
        comparison_times = 0
        swap_times = 0

def partition(pivot_index, left, right, arr):
    """Performs partitioning for Quick Sort."""
    global comparison_times, swap_times
    arr[pivot_index], arr[left] = arr[left], arr[pivot_index]
    if left + 1 == right:
        comparison_times += 1
        if int(arr[left].growth) > int(arr[right - 1].growth):
            arr[right], arr[left] = arr[left], arr[right]
            swap_times += 1
    if left < right:
        if pivot_index != left:
            quick_sort(left, left, arr)
        if left + 1 != right:
            quick_sort(left + 1, right, arr)

def quick_sort(start, end, arr):
    """Sorts the array using Quick Sort algorithm."""
    global swap_times, comparison_times
    pivot_index = start
    swap_index = start
    left = start
    for i in range(start, end):
        comparison_times += 1
        if int(arr[i].growth) < int(arr[pivot_index].growth):
            left = i
        else:
            right = i
            if swap_index == start:
                swap_index = right
        if left > right and right != start:
            arr[swap_index], arr[left] = arr[left], arr[swap_index]
            left = swap_index
            swap_index = left + 1
            swap_times += 1
    partition(pivot_index, left, end, arr)
