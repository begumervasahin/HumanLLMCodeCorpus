import math
import sys

def eratosthenes(n):
    print("Starting!")
    file = open("Eratosthenes_output.txt", "w")
    array = [True for _ in range(n)]
    for i in range(2, n):
        if i < math.sqrt(n):
            if array[i]:
                k = i
                while k + i < n:
                    array[k + i] = False
                    k += i
        if array[i]:
            print(i)
            file.write(str(i) + "\n")
    file.close()

if __name__ == "__main__":
    n = 0
    try:
        n = int(input("Find primes up to: "))
        print("Allocating memory...")
        eratosthenes(n)
        print("Finished generating all primes less than " + str(n) + "!")
    except ValueError:
        print("Enter a valid number.")
