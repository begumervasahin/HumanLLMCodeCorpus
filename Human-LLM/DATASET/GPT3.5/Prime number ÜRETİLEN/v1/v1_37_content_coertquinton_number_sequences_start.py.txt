class OutputPrinter:
    SIMPLE_NUMBERS = 1

    def __init__(self):
        self._printer_types = {
            self.SIMPLE_NUMBERS: "Simple Numbers"
        }

    def display_printer_options(self):
        print("Available printer options:")
        for key, value in self._printer_types.items():
            print(f"{key}: {value}")

    def validate_printer(self, printer_type):
        return printer_type in self._printer_types

    def select_printer(self, printer_type):
        return self._printer_types.get(printer_type, self._printer_types[self.SIMPLE_NUMBERS])

        try:
            if not menu_item.isdigit():
                print(error_message)
                return False
            validated_item = int(menu_item)
            if validated_item in self._tasks:
                return True
