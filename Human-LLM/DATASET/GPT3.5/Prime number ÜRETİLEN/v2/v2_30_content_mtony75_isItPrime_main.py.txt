# Function to check if the input is a valid integer
def is_valid_integer(number):
    try:
        int(number)  # Attempt to convert the input to an integer
        return True  # If successful, return True
    except ValueError:
        print("User Input Not An Integer")  # If conversion fails, print error message
        return False  # Return False indicating invalid input

# Function to check if a number is prime
def is_prime(number):
    for loop_number in range(2, number):
        if number % loop_number == 0:
            print(str(number) + ' is not a prime number')  # If number is divisible, it's not prime
            return False
    print(str(number) + ' is a prime number')  # If no divisors found, number is prime
    return True

user_input = ""
while user_input != "stop":
    user_input = input("Enter a Number: ")  # Prompt user for input
    if is_valid_integer(user_input):  # Check if input is a valid integer
        is_prime(int(user_input))  # Check if the integer is a prime number
