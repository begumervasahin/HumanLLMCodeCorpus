while True:
    is_prime = True  # Assume the number is prime initially
    n = input('> ')  # Prompt the user for input
    if n.lower() == 'q':  # Check if the user wants to quit
        break
    else:
        n = int(n)  # Convert input to an integer
        if n != 2:  # Check if the number is not 2
            # Check if the number is divisible by any integer from 2 to n-1
            for i in range(2, n):
                if n % i == 0:  # If the number is divisible by i, it's not prime
                    is_prime = False
                    break
    # Print whether the number is prime or not
    print(f'{n} is prime: {is_prime}')
