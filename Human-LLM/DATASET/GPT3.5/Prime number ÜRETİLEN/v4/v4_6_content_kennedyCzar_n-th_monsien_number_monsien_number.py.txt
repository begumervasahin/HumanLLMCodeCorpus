# Find the n-th Monisen number.
# A number M is a Monisen number if M = 2**P - 1
# and both M and P are prime numbers.
# For example, if P = 5, M = 2**P - 1 = 31,
# 5 and 31 are both prime numbers,
# so 31 is a Monisen number.
# Put the 6-th Monisen number into a
# single text file and submit online.

def find_monisen(n):
    number = 10
    p = 2
    while p < n:
        if all(number % i != 0 for i in range(2, number)):
            p += 1
        number += 1
        m = 2**p - 1
    return number - 1, m

def print_even_numbers():
    # Create a list of even numbers from 1 to 10
    d = [i + 1 for i in range(10) if i % 2 == 0]
    print(d)

def calculate_sum():
    # Calculate the sum of numbers until it exceeds 10
    sumA = 0
    i = 1
    while True:
        sumA += i
        i += 1
        if sumA > 10:
            break
    print('i = {}, sum = {}'.format(i, sumA))

def print_numbers_divisible_by_3():
    # Print numbers until reaching 10 or a number divisible by 3
    i = 1
    while(i % 3):
        print(i, end=' ')
        if (i >= 10):
            break
        i += 1

def convert_base(num, base):
    # Convert a number to the given base
    if(num >= base):
        convert_base(num // base, base)
    print(num % base, end=' ')
