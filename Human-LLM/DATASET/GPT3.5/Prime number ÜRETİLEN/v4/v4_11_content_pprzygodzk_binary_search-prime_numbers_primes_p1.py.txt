import math

def SieveofEratosthenes(n):
    """Determine prime numbers from the set of natural numbers from 2 to n."""
    primes = [i for i in range(2, n + 1)]
    for p in primes:
        for j in range(p * p, n + 1, p):
            if j in primes:
                primes.remove(j)
    return primes

def PrimesGenerator(n):
    """Generate n prime numbers using trial division."""
    primes = []
    p = 2
    i = 1
    k = 1
    r = 1
    while i <= n:
        if p == 2 or p == 3:
            primes.append(p)
            i += 1
            p = p + 1 if p == 2 else p + 2
            continue
        for d in range(2, int(math.sqrt(p)) + 1):
            if p % d == 0:
                break
            if p % d != 0:
                if d == int(math.sqrt(p)):
                    primes.append(p)
                    i += 1
                continue
        k = k + 1 if p == 6 * k + 1 else k
        r = 1 - r
        p = 6 * k + 1 if r == 0 else 6 * k - 1
    return primes

def PrimorialGenerator(n):
    """Generate n primorial numbers."""
    primes = PrimesGenerator(n - 1)
    primorials = [1]
    product = 1
    for prime in primes:
        product *= prime
        primorials.append(product)
    return primorials

def CheckPrimarility(n):
    """Check if the given number n is a prime number."""
    assert n >= 2, "The least prime number is 2!"
    if n % 2 == 0 and n != 2:
        return "composite"
    for d in range(3, int(n ** 0.5) + 1):
        if n % d == 0:
            return "composite"
        if n % d != 0:
            continue
    return "prime"

def Decompose(n):
    """Decompose the given number n into prime factors."""
    if n == 1 or CheckPrimarility(n) == "prime":
        return [n]
    factor_base = SieveofEratosthenes(math.ceil(n ** 0.5))
    i = 0
    d = factor_base[0]
    decomposition = []
    while d * d <= n:
        if n % d == 0:
            n //= d
            decomposition.append(d)
        else:
            i += 1
            d = factor_base[i]
    decomposition.append(n)
    return decomposition
