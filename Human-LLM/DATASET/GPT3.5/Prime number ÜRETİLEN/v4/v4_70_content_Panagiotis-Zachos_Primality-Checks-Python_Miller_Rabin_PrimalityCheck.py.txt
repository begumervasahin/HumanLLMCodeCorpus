import math
import random
import time

def power_of_two(n):
    """
    Returns the highest power of 2 that divides n, along with the remaining odd number.

    Args:
    - n (int): The number to be decomposed.

    Returns:
    - tuple: A tuple containing the highest power of 2 that divides n and the remaining odd number.
    """
    r = 0
    d = n
    while d % 2 == 0:
        r += 1
        d >>= 1
    assert 2**r * d == n
    return r, d

def miller_rabin_test(n, k):
    """
    Miller-Rabin primality test.
    A return value of False means n is certainly not prime. A return value of
    True means n is very likely a prime.

    Args:
    - n (int): The number to be tested for primality.
    - k (int): The number of iterations for the Miller-Rabin test.

    Returns:
    - bool: True if n is likely prime, False otherwise.
    """
    if n <= 3:
        return False
    r, d = power_of_two(n - 1)
    for _ in range(k):
        r_int = random.randint(2, n - 2)
        x = pow(r_int, d, n)
        if x == 1 or x == n - 1:
            continue
        flag = 0
        for _ in range(r - 1):
            x = pow(x, 2, n)
            if x == n - 1:
                flag = 1
                break
        if flag == 1:
            continue
        return False
    return True

def is_prime(n):
    """
    Simple primality check.

    Args:
    - n (int): The number to be tested for primality.

    Returns:
    - bool: True if n is prime, False otherwise.
    """
    if n <= 1:
        return False
    elif n <= 3:
        return True
    elif n % 2 == 0 or n % 3 == 0:
        return False
    i = 5
    while i * i <= n:
        if n % i == 0 or n % (i + 2) == 0:
            return False
        i += 6
    return True

# Test with a large number and measure time taken for both methods
n = 200000015717
t0 = time.time()
print(miller_rabin_test(n, 8))
t1 = time.time()
print("Miller-Rabin Primality Test Time: " + str(t1 - t0))
