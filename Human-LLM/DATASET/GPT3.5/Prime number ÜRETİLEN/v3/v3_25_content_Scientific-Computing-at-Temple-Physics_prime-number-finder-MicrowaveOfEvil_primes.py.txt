import math

def get_user_input():
    """
    Get user input for the range of numbers to check for primes.

    Returns:
        tuple: A tuple containing the smallest and largest numbers to check.
    """
    print("This program will find prime numbers. Choose two positive integers to find primes between: ")
    x1 = int(input('Smallest number to check: '))
    x2 = int(input('Largest number to check: '))
    return min(x1, x2), max(x1, x2)

def is_prime(number):
    """
    Check if a number is prime.

    Parameters:
        number (int): The number to check.

    Returns:
        bool: True if the number is prime, False otherwise.
    """
    if number < 2:
        return False
    for divisor in range(2, int(math.sqrt(number)) + 1):
        if number % divisor == 0:
            return False
    return True

def find_primes(x1, x2):
    """
    Find prime numbers within a given range.

    Parameters:
        x1 (int): The smallest number to check.
        x2 (int): The largest number to check.

    Returns:
        list: A list of prime numbers found within the given range.
    """
    primes_list = [number for number in range(x1, x2 + 1) if is_prime(number)]
    return primes_list

def display_primes(primes_list, x1, x2):
    """
    Display prime numbers found within the given range.

    Parameters:
        primes_list (list): A list of prime numbers found within the range.
        x1 (int): The smallest number in the range.
        x2 (int): The largest number in the range.
    """
    print(f"Attempting to find primes between {x1} and {x2}...")
    if primes_list:
        for prime in primes_list:
            print(prime)
    else:
        print("No primes found")

def main():
    x1, x2 = get_user_input()
    primes_list = find_primes(x1, x2)
    display_primes(primes_list, x1, x2)

    if x1 == x2:
        print("You put in the same number twice. There are zero primes between that number and itself.")
        if x1 > 1:
            print("However, that number is itself a prime.")
    else:
        print(f"There are {len(primes_list)} primes between {x1} and {x2}")

if __name__ == '__main__':
    main()
