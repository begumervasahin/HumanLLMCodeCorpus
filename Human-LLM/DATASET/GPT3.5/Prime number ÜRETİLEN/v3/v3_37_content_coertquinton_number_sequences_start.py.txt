class OutputPrinter:
    # Constants for printer types
    SIMPLE_NUMBERS = 1

    def __init__(self):
        # Initialize available printer types
        self._printer_types = {
            self.SIMPLE_NUMBERS: "Simple Numbers"
        }

    def display_printer_options(self):
        # Display available printer options to the user
        print("Available printer options:")
        for key, value in self._printer_types.items():
            print(f"{key}: {value}")

    def validate_printer(self, printer_type):
        # Validate if the selected printer type is valid
        return printer_type in self._printer_types

    def select_printer(self, printer_type):
        # Select and return the printer type based on the input
        return self._printer_types.get(printer_type, self._printer_types[self.SIMPLE_NUMBERS])


class CommandLine:
    def __init__(self):
        # Initialize CommandLine with OutputPrinter instance and menu tasks
        self._printer = OutputPrinter().select_printer(OutputPrinter.SIMPLE_NUMBERS)
        self._tasks = {
            0: 'Change printer. Default is simple numbers.',
            1: 'Exit',
            2: 'Print prime numbers',
            3: 'Print Fibonacci numbers',
            4: 'Print Collatz Conjecture',
            5: 'Print strong prime numbers',

    def process_menu(self):
        # Main function to process the menu options and execute corresponding tasks
        while True:
            self.display_menu()
            menu_item = input('Type menu item number to start math operation: ')
            if int(menu_item) == 0:
                self.set_printer()
                continue
            if self.validate_menu_item(menu_item):
                if int(menu_item) == 1:
                    break
                print(menu_item + ' has been selected')
                self.set_task(int(menu_item))
                self.process_task()

    def display_menu(self):
        # Display the menu options to the user
        print('---------------------------------------------------')
        print('Type the corresponding number to access a menu item')
        for key, value in self._tasks.items():
            print(f"{key}: {value}")

    def validate_menu_item(self, menu_item):
        # Validate if the input menu item is valid
        try:
            validated_item = int(menu_item)
            if validated_item in self._tasks:
                return True
            else:
                print('Invalid menu item chosen.')
                return False
        except ValueError:
            print(f"Menu Item {menu_item} is not a valid choice. Input must be an integer.")

    def set_task(self, menu_item):
        # Placeholder for setting the task based on the menu item selected
        pass

    def process_task(self):
        # Placeholder for processing the task selected by the user
        pass

    def set_printer(self):
        # Allow user to select the output printer type
        output_printer = OutputPrinter()
        output_printer.display_printer_options()
        printer_type = int(input("Type the number of the type of output you would like to see: "))
        if output_printer.validate_printer(printer_type):
            self._printer = output_printer.select_printer(printer_type)
            print(f"Printer set to {output_printer._printer_types[printer_type]}")
        else:
            self._printer = output_printer.select_printer(output_printer.SIMPLE_NUMBERS)
