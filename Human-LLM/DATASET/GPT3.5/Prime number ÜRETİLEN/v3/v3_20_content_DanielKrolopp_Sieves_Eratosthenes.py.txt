import math

def sieve_of_eratosthenes(n):
    """Generate prime numbers up to n using the Sieve of Eratosthenes algorithm."""
    print("Starting the Sieve of Eratosthenes!")
    try:
        with open("primes.txt", "w") as file:
            primes = []
            is_prime = [True] * (n + 1)
            is_prime[0] = is_prime[1] = False

            # Sieve of Eratosthenes algorithm
            for i in range(2, int(math.sqrt(n)) + 1):
                if is_prime[i]:
                    for j in range(i * i, n + 1, i):
                        is_prime[j] = False

            # Write prime numbers to file and print them
            for i in range(2, n + 1):
                if is_prime[i]:
                    primes.append(i)
                    file.write(str(i) + "\n")
                    print(i)

        print("Finished generating all prime numbers up to", n)
    except Exception as e:
        print("An error occurred:", e)

if __name__ == "__main__":
    try:
        n = int(input("Find prime numbers up to: "))
        if n < 2:
            print("Please enter a number greater than 1.")
        else:
            print("Allocating memory and generating primes...")
            sieve_of_eratosthenes(n)
    except ValueError:
        print("Please enter a valid integer.")
