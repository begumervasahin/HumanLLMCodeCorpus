import math

def sieve_of_eratosthenes(n):
    """Generate prime numbers from 2 to n using the Sieve of Eratosthenes algorithm."""
    primes = [True] * (n+1)
    primes[0] = primes[1] = False
    p = 2
    while p**2 <= n:
        if primes[p]:
            for i in range(p**2, n+1, p):
                primes[i] = False
        p += 1
    return [i for i in range(2, n+1) if primes[i]]

def generate_primes(n):
    """Generate n prime numbers using trial division."""
    primes = [2, 3]
    num = 5
    while len(primes) < n:
        is_prime = True
        for p in primes:
            if p * p > num:
                break
            if num % p == 0:
                is_prime = False
                break
        if is_prime:
            primes.append(num)
        num += 2 if num % 6 == 1 else 4
    return primes

def generate_primorials(n):
    """Generate the first n primorial numbers."""
    primes = generate_primes(n-1)
    primorials = [1]
    for prime in primes:
        primorials.append(primorials[-1] * prime)
    return primorials

def is_prime(n):
    """Check if a number n is prime."""
    if n < 2:
        return "The least prime number is 2!"
    if n == 2:
        return "prime"
    if n % 2 == 0:
        return "composite"
    for d in range(3, math.isqrt(n) + 1, 2):
        if n % d == 0:
            return "composite"
    return "prime"

def decompose_prime_factors(n):
    """Decompose a number into prime factors."""
    if n == 1:
        return [1]
    if is_prime(n) == "prime":
        return [n]
    factors = []
    d = 2
    while d * d <= n:
        if n % d == 0:
            factors.append(d)
            n //= d
        else:
            d += 1
    factors.append(n)
    return factors
