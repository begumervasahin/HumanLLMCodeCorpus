def is_valid_integer(input_str):
    """
    Check if the input string can be converted to an integer.

    Args:
        input_str (str): The input string to be checked.

    Returns:
        bool: True if the input is a valid integer, False otherwise.
    """
    try:
        int(input_str)
        return True
    except ValueError:
        print("Invalid input: Please enter an integer.")
        return False

def is_prime(number):
    """
    Check if a number is prime.

    Args:
        number (int): The number to be checked.

    Returns:
        bool: True if the number is prime, False otherwise.
    """
    if number <= 1:
        print(f"{number} is not a prime number.")
        return False
    for divisor in range(2, int(number**0.5) + 1):
        if number % divisor == 0:
            print(f"{number} is not a prime number.")
            return False
    print(f"{number} is a prime number.")
    return True

def main():
    """
    Main function to repeatedly prompt the user for input and check if the input is prime.
    """
    while True:
        user_input = input("Enter a Number (or 'stop' to quit): ")
        if user_input.lower() == "stop":
            print("Exiting the program.")
            break
        if is_valid_integer(user_input):
            is_prime(int(user_input))

if __name__ == "__main__":
    main()
