def gcd_iteration_test(higher, lower):
    """
    Perform GCD iteration test using modular division.

    Args:
        higher (int): Higher value for GCD test.
        lower (int): Lower value for GCD test.

    Returns:
        int: Remainder after GCD test.
    """
    remains, _, next_divisor = modular_division(higher, lower)
    while remains != 1:
        if remains == 0:
            break
        else:
            remains, _, next_divisor = modular_division(next_divisor, remains)
    return remains

def modular_division(suspect_prime_or_remainder, next_mapped_prime):
    """
    Perform modular division.

    Args:
        suspect_prime_or_remainder (int): Suspected prime or remainder.
        next_mapped_prime (int): Next mapped prime for division.

    Returns:
        tuple: Remainder, quotient, divisor.
    """
    divisor = next_mapped_prime
    dividend = suspect_prime_or_remainder
    quotient = dividend // divisor
    remainder = dividend % divisor
    return remainder, quotient, divisor
