import sys
import numpy as np
import matplotlib.pyplot as plt

# Default values for rows and columns
DEFAULT_ROWS = 24
DEFAULT_COLUMNS = 2

# Read command line arguments to adjust the number of columns and rows
if len(sys.argv) > 1:
    columns = int(sys.argv[1])
else:
    columns = DEFAULT_COLUMNS

if len(sys.argv) > 2:
    rows = int(sys.argv[2])
else:
    rows = DEFAULT_ROWS

N_MAX = columns * rows

# Function to generate prime numbers up to a given limit using the Sieve of Eratosthenes algorithm
def primes(n):
    if n == 2:
        return [2]
    elif n < 2:
        return []

    sieve = list(range(3, n + 1, 2))
    mroot = n ** 0.5
    half = (n + 1) // 2 - 1
    i = 0
    m = 3

    while m <= mroot:
        if sieve[i]:
            j = (m * m - 3) // 2
            sieve[j] = 0
            while j < half:
                sieve[j] = 0
                j += m
        i += 1
        m = 2 * i + 3

    return [2] + [x for x in sieve if x]

# Generate prime numbers
known_primes = primes(N_MAX)
max_primes = len(known_primes)

# Print known primes if the number of columns is less than 5
if columns < 5:
    print(known_primes)

# Initialize arrays to store numbers and primes
n_numbers = np.zeros(N_MAX)
r_numbers = np.zeros(N_MAX)
th_numbers = np.zeros(N_MAX)
n_primes = np.zeros(max_primes)
r_primes = np.zeros(max_primes)
th_primes = np.zeros(max_primes)

# Fill arrays with numbers and primes
n = 0
p = 0
for c in range(columns):
    for i in range(rows):
        n_numbers[n] = n + 1
        r_numbers[n] = c + 1
        th_numbers[n] = (2. * np.pi / float(rows)) * (i + 1)

        if (p < max_primes) and (n_numbers[n] == known_primes[p]):
            n_primes[p] = n_numbers[n]
            r_primes[p] = r_numbers[n]
            th_primes[p] = th_numbers[n]
            p += 1

        n += 1

# Create polar plot
fig = plt.figure(figsize=(10., 10.))
ax = fig.add_subplot(111, projection="polar")
points_size = 100
font_size = 12

# Adjust point size and font size based on the number of columns
if columns > 10:
    points_size = 50
    font_size = 6
if columns > 30:
    points_size = 10
    font_size = 0

# Plot all numbers
for x, y, n in zip(th_numbers, r_numbers, n_numbers):
    ax.scatter((x,), (y,), color="gray", s=points_size)
    if columns < 30:
        plt.text(x, y, '%i' % n, fontsize=font_size)

# Plot prime numbers
for x, y, n in zip(th_primes, r_primes, n_primes):
    ax.scatter((x,), (y,), color="red", s=points_size)
    if columns < 30:
        plt.text(x, y, '%i' % n, fontsize=font_size)

# Customize the plot
ax.set_rticks([])
dth = 360 / rows
ax.set_thetagrids([-dth, dth, 90 - dth, 90 + dth, 180 - dth, 180 + dth, 270 - dth, 270 + dth], labels=[''] * columns)
if columns > 1:
    ax.set_rgrids(np.arange(1, columns), labels=[''] * columns)
ax.grid(True)

# Save and display the plot
ext = "jpg"
plt.savefig("%s/primes_r%i_c%i.%s" % (ext, rows, columns, ext))
plt.show()
