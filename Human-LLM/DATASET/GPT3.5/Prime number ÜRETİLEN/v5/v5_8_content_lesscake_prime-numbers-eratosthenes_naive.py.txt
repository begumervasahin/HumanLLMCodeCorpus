def find_all_primes_up_to(n):
    """
    Find all prime numbers up to the given number n.

    Args:
    - n (int): The upper limit for prime number search.

    Returns:
    - list: A list containing all prime numbers found up to n.
    """
    primes = []
    for number in range(2, n + 1):
        if is_prime(number):
            primes.append(number)
    return primes

def is_prime(number):
    """
    Check if a given number is prime.

    Args:
    - number (int): The number to check.

    Returns:
    - bool: True if the number is prime, False otherwise.
    """
    if number < 2:
        return False
    for divisor in range(2, int(number ** 0.5) + 1):
        if number % divisor == 0:
            return False
    return True

if __name__ == '__main__':
    # Example usage: Find all primes up to 100
    print(find_all_primes_up_to(100))
