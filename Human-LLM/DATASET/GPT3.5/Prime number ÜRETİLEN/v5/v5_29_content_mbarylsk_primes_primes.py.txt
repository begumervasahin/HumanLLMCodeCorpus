import os
import numpy
import pycuda.driver as cuda
import pycuda.autoinit
from pycuda.compiler import SourceModule

class PrimeFinder:
    def __init__(self, cache_results):
        self.cache_results = cache_results
        self.primes = set()
        self.twin_primes = set()
        self.non_primes = set()
        self.primes_to_exclude = set()
        self.prime_cache = {}
        self.twin_prime_cache = {}

        self.kernel = SourceModule("""
            __global__ void first_factor(long long n, int *value, int *primes){
                int i = threadIdx.x;
                int p = primes[i];
                if(n % p == 0){
                    value[i] = p;
                }else{
                    value[i] = 0;
                }
            }
        """)

    def is_prime(self, n):
        if n < 2 or n in self.non_primes or n in self.primes_to_exclude:
            return False
        if n in self.primes or n in self.prime_cache:
            return True

        if n % 2 == 0 or n % 3 == 0:
            self.non_primes.add(n)
            return False

        i = 5
        while i * i <= n:
            if n % i == 0 or n % (i + 2) == 0:
                self.non_primes.add(n)
                return False
            i += 6

        if self.cache_results:
            self.prime_cache[n] = True
        return True

    def is_twin_prime(self, n):
        if n in self.twin_primes or n in self.non_primes:
            return False
        if n in self.twin_prime_cache:
            return True
        return self.is_prime(n) and (self.is_prime(n + 2) or self.is_prime(n - 2))

    def is_prime_cuda(self, n):
        def min2(list, bound=0):
            for item in list:
                if item > bound:
                    return item
            return None

        first_factor = self.kernel.get_function('first_factor')
        if n == 1:
            return False
        if n == 2 or n == 3:
            return True
        all_primes = [2, 3]
        num_primes = len(all_primes)
        num_threads = 384
        result = numpy.zeros(num_primes, numpy.int32)
        primes = numpy.copy(all_primes).astype(numpy.int32)
        while True:
            first_factor(numpy.int64(n), cuda.InOut(result), cuda.In(primes), block=(num_threads, 1, 1))
            prime = min2(result, 1)
            if prime is None:
                break
            return False
        return True

    def add_to_primes_to_exclude(self, n):
        self.primes_to_exclude.add(n)

    def add_to_primes(self, n):
        self.primes.add(n)

    def add_to_twin_primes(self, n):
        self.twin_primes.add(n)

    def add_to_non_primes(self, n):
        self.non_primes.add(n)

    def get_ith_prime(self, i):
        if not self.primes:
            n = 2
        else:
            n = max(self.primes)
        primes_to_find = i - len(self.primes)
        while primes_to_find > 0:
            n += 1
            if self.is_prime(n):
                primes_to_find -= 1
        return n

    def get_ith_twin_prime(self, i):
        if not self.twin_primes:
            n = 3
        else:
            n = max(self.twin_primes)
        twin_primes_to_find = i - len(self.twin_primes)
        while twin_primes_to_find > 0:
            n += 1
            if self.is_twin_prime(n):
                twin_primes_to_find -= 1
        return n

    def get_ith_composite(self, i):
        if not self.non_primes:
            n = 4
        else:
            n = max(self.non_primes)
        composites_to_find = i - len(self.non_primes)
        while composites_to_find > 0:
            n += 1
            if not self.is_prime(n):
                composites_to_find -= 1
        return n

    def is_symmetric_prime(self, n, i):
        k1 = n - i
        k2 = n + i
        if self.is_prime(k1) and self.is_prime(k2):
            return True, k1, k2
        else:
            return False, 0, 0

    def is_6km1(self, n):
        return self.is_prime(n) and n > 3 and n % 6 == 5

    def is_6kp1(self, n):
        return self.is_prime(n) and n > 3 and n % 6 == 1

    def find_unique_prime_in_sum(self, n):
        is_prime_lower_than_current_sum = True
        k = 1
        while is_prime_lower_than_current_sum:
            q = self.get_ith_prime(k)
            if 2 <= n - q and q not in self.list_of_primes_used:
                self.list_of_primes_used.append(q)
                self.find_unique_prime_in_sum(n - q)
            elif 2 <= n - q:
                k += 1
            else:
                self.list_of_primes_used.append(q)
                is_prime_lower_than_current_sum = False
