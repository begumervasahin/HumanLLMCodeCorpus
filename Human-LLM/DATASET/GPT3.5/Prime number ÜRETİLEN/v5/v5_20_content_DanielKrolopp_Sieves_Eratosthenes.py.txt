import math

def sieve_of_eratosthenes(n):
    """
    Generates prime numbers up to n using the Sieve of Eratosthenes algorithm.

    Args:
        n (int): The upper limit for prime number generation.

    Returns:
        list: List of prime numbers up to n.
    """
    print("Starting Sieve of Eratosthenes algorithm...")
    primes = []

    # Create a boolean array "array[0..n]" and initialize
    # all entries as true. A value in array[i] will finally
    # be false if i is not a prime, else true.
    sieve = [True] * (n + 1)
    sieve[0] = sieve[1] = False

    for i in range(2, int(math.sqrt(n)) + 1):
        if sieve[i]:
            # Mark multiples of i as non-prime
            for j in range(i * i, n + 1, i):
                sieve[j] = False

    # Print all prime numbers
    with open("Eratosthenes_output.txt", "w") as file:
        for i in range(2, n + 1):
            if sieve[i]:
                primes.append(i)
                print(i)
                file.write(str(i) + "\n")

    print("Finished generating all primes less than or equal to", n)
    return primes
