def is_prime(number):
    """
    Check if a given number is prime.

    Parameters:
    - number: int, the number to check for primality.

    Returns:
    - bool, True if the number is prime, False otherwise.
    """
    if number < 2:
        return False
    for factor in range(2, number):
        if number % factor == 0:
            return False
    return True

print(is_prime(21))
