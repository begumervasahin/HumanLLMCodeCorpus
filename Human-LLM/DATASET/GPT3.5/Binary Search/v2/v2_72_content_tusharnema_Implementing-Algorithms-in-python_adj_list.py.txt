from collections import defaultdict, deque

class Graph:
    def __init__(self, connections, directed=False):
        # Initialize a defaultdict to store the graph connections
        self.graph = defaultdict(set)
        self.directed = directed
        self.add_connections(connections)

    def add_connections(self, connections):
        # Add connections to the graph
        for node1, node2 in connections:
            self.add(node1, node2)

    def add(self, node1, node2):
        # Add an edge between two nodes
        self.graph[node1].add(node2)
        if not self.directed:
            self.graph[node2].add(node1)

    def remove(self, node):
        # Remove a node and its connections from the graph
        for n, cxns in self.graph.items():
            cxns.discard(node)
        self.graph.pop(node, None)

    def is_connected(self, node1, node2):
        # Check if two nodes are connected in the graph
        return node1 in self.graph and node2 in self.graph[node1]

    def find_path(self, node1, node2, path=[]):
        # Find a path between two nodes using DFS
        path = path + [node1]
        if node1 == node2:
            return path
        if node1 not in self.graph:
            return None
        for node in self.graph[node1]:
            if node not in path:
                new_path = self.find_path(node, node2, path)
                if new_path:
                    return new_path
        return None

    def BFS(self, start_node):
        # Perform Breadth-First Search (BFS) traversal
        visited = set()
        queue = deque([start_node])
        while queue:
            node = queue.popleft()
            if node not in visited:
                print(node, end="-->")
                visited.add(node)
                queue.extend(self.graph[node] - visited)

    def DFS(self, start_node):
        # Perform Depth-First Search (DFS) traversal
        visited = set()
        self._DFS_recursive(start_node, visited)

    def _DFS_recursive(self, node, visited):
        # Helper function for DFS traversal
        if node not in visited:
            print(node, end="-->")
            visited.add(node)
            for neighbor in self.graph[node]:
                self._DFS_recursive(neighbor, visited)

    def __str__(self):
        # Return a string representation of the graph
        return '{}({})'.format(self.__class__.__name__, dict(self.graph))

    def display_connections(self):
        # Display the connections in the graph
        for node, neighbors in self.graph.items():
            print(f"{node} ---> {neighbors}")

# Define the connections for the graph
connections = [(0, 1), (1, 2), (2, 3), (2, 4), (3, 4), (5, 6), (7, 3)]
# Create a graph object
g = Graph(connections)
# Add additional connections to the graph
g.add_connections([[6, 8], [8, 4], [9, 10]])

# Print the graph connections
print("Graph Connections:")
g.display_connections()

print("\nBreadth-First Search (BFS):")
g.BFS(0)

print("\nDepth-First Search (DFS):")
g.DFS(0)
