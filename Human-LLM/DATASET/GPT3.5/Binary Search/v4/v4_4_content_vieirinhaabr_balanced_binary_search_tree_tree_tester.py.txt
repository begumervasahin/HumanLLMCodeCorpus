# Import necessary modules
from balanced_binary_search_tree import Tree
import random as rdn
from chronometer import Chronometer

# Main code execution starts here
if __name__ == "__main__":
    # Create an instance of the Tree class
    tree_control = Tree()

    # Measure time for various operations using Chronometer
    with Chronometer() as time:
        i = 0
        count = 50

        # Measure time for inserting nodes into the tree
        with Chronometer() as t:
            while i < count:
                # Generate random data for the nodes
                info = rdn.randint(1, 100)
                tree_control.insert_node(info)

                # Store a random value for future reference
                if i == count / 2:
                    temp = info
                i += 1

        # Print the time taken for insertion
        print('To insert, it took {:.5f} seconds'.format(float(t)))

        # Measure time for in-order traversal of the tree
        with Chronometer() as t:
            print('\nIn-order traversal: ', end='')
            tree_control.in_order()

        # Print the time taken for traversal
        print('\n\nTo print in-order, it took {:.5f} seconds'.format(float(t)))

        # Print tree information
        print('\nTree Info: ')
        print('      root -> ', tree_control.root.info)
        print('      Left height -> ', tree_control.root.left_height)
        print('      Right height -> ', tree_control.root.right_height)

        # Measure time for searching a specific value in the tree
        with Chronometer() as t:
            print('\nTrying to find ', temp, '...')
            tree_control.search_node(temp)

        # Print the time taken for the search operation
        print('    Searching took {:.5f} seconds'.format(float(t)))
        print('\n')

        # Measure time for converting the tree to tuple format
        with Chronometer() as t:
            print('Getting Tree in tuple format...')
            tuple_tree = tree_control.print_tree_on_tuple()
            print(tuple_tree)

        # Print the time taken for converting to tuple format
        print('    Converting to tuple took {:.5f} seconds'.format(float(t)))
        print('\n')

        # Measure time for printing the tree structure
        with Chronometer() as t:
            tree_control.print_tree()

        # Print the time taken for printing the tree structure
        print('\nPrinting the tree took {:.5f} seconds'.format(float(t)))

    # Print the overall time taken for all operations
    print('\n\nTotal time taken: {:.5f} seconds'.format(float(time)))
