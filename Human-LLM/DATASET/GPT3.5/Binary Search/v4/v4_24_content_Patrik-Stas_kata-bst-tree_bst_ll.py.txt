class Node:
    """Node of a Tree"""
    def __init__(self, key, value, parent=None):
        self.key = key
        self.value = value
        self.left = None
        self.right = None
        self.parent = parent

    def remove_child(self, child_to_remove):
        """Remove the specified child node."""
        if child_to_remove == self.left:
            self.left = None
        elif child_to_remove == self.right:
            self.right = None

    def print_info(self):
        """Print information about the node."""
        left_str = "        " if self.left is None else f"_left: {self.left.value}"
        right_str = "         " if self.right is None else f"_right: {self.right.value}"
        parent_str = "   ROOT" if self.parent is None else f"   _parent: {self.parent.value}"
        print(f"Node({self.value})  {left_str}  {right_str}  {parent_str}")

    def is_leaf(self):
        """Check if the node is a leaf."""
        return self.left is None and self.right is None

    def is_root(self):
        """Check if the node is the root."""
        return self.parent is None

    def set_right(self, child_node):
        """Set the right child of the node."""
        self.right = child_node
        child_node.parent = self

    def set_left(self, child_node):
        """Set the left child of the node."""
        self.left = child_node
        child_node.parent = self

    def replace_child(self, old_child, new_child):
        """Replace a child node with a new node."""
        if old_child == self.left:
            self.set_left(new_child)
        elif old_child == self.right:
            self.set_right(new_child)
        else:
            raise ValueError("old_child node not found")


class TreePrinter:
    """Prints the binary search tree."""
    def __init__(self, bst):
        self.bst = bst

    def printbst(self):
        """Print 'foo'."""
        print("foo")

    def print_preorder(self, nice_print=False):
        """Print the tree in preorder traversal."""
        self.preorder(self.bst.root, nice_print)

    def preorder(self, traverse_node, nice_print):
        """Traverse the tree in preorder."""
        if traverse_node is None:
            raise ValueError("Passing None as node to preorder print is not expected.")
        if nice_print:
            traverse_node.print_info()
        else:
            print(str(traverse_node.value) + " ", end='')
        if traverse_node.left is not None:
            self.preorder(traverse_node.left, nice_print)
        if traverse_node.right is not None:
            self.preorder(traverse_node.right, nice_print)


class BstByLinkedList:
    """Binary Search Tree implemented using linked list."""
    def __init__(self):
        self.root = None

    def count_nodes(self):
        """Count the number of nodes in the tree."""
        return self.__count_nodes(self.root)

    def __count_nodes(self, subtree_root):
        """Recursively count the nodes in the subtree."""
        if subtree_root is not None:
            return 1 + self.__count_nodes(subtree_root.left) + self.__count_nodes(subtree_root.right)
        return 0

    def insert(self, key, value):
        """Insert a node into the tree."""
        if key is None or value is None:
            raise ValueError("Insert key or value is null.")
        elif self.root is None:
            self.root = Node(key, value, parent=None)
        else:
            self.__insert(self.root, key, value)

    def __insert(self, subtree_root, key, value):
        """Recursively insert a node into the subtree."""
        if subtree_root.key == key:
            raise ValueError("Duplicate key " + str(key))
        else:
            if key < subtree_root.key:
                if subtree_root.left is None:
                    subtree_root.set_left(Node(key, value))
                else:
                    self.__insert(subtree_root.left, key, value)
            else:
                if subtree_root.right is None:
                    subtree_root.set_right(Node(key, value))
                else:
                    self.__insert(subtree_root.right, key, value)

    def __search(self, subtree_root, search_key):
        """Search for a node with the given key in the subtree."""
        if subtree_root is None:
            return None
        if subtree_root.key == search_key:
            return subtree_root
        elif search_key < subtree_root.key:
            return self.__search(subtree_root.left, search_key)
        else:
            return self.__search(subtree_root.right, search_key)

    def search(self, search_key):
        """Search for a node with the given key."""
        node = self.__search(self.root, search_key)
        return node.value if node is not None else None

    def find_biggest_node_in_subtree(self, subtree_root):
        """Find the node with the maximum key in the subtree."""
        if subtree_root.right is None:
            return subtree_root
        else:
            return self.find_biggest_node_in_subtree(subtree_root.right)

    def delete(self, key_delete):
        """Delete a node with the given key."""
        node_to_delete = self.__search(self.root, key_delete)
        if node_to_delete is None:
            raise ValueError("Can't delete node with key " + str(key_delete) + " because it was not found in the tree.")
        deleted_value = node_to_delete.value
        if node_to_delete is None:
            raise ValueError("Value " + str(key_delete) + " not found in the tree.")
        if node_to_delete.is_leaf():
            node_to_delete.parent.remove_child(node_to_delete)
        elif node_to_delete.left is None:
            node_to_delete.parent.replace_child(node_to_delete, node_to_delete.right)
        else:
            replacement_node = self.find_biggest_node_in_subtree(node_to_delete.left)
            if replacement_node.left is not None:
                replacement_node.parent.replace_child(replacement_node, replacement_node.left)
            else:
                replacement_node.parent.remove_child(replacement_node)
            node_to_delete.key = replacement_node.key
            node_to_delete.value = replacement_node.value
        return deleted_value
