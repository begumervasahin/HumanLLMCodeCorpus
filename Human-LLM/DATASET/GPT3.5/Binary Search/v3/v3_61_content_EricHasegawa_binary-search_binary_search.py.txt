def input_integer(prompt):
    while True:
        try:
            return int(input(prompt))
        except ValueError:
            print("Invalid input. Please enter an integer.")

def read_and_sort_numbers(filename):
    with open(filename, 'r') as file:
        numbers = file.read().split(", ")
        return sorted(map(int, numbers))

def binary_search():
    target = input_integer("Enter an integer to search: ")
    sorted_numbers = read_and_sort_numbers('numberlist.txt')

    low, high = 0, len(sorted_numbers) - 1

    while high >= low:
        mid = (high + low) // 2
        if target == sorted_numbers[mid]:
            print("Target found.")
            return
        elif mid == low:
            print("Target not found.")
            return
        elif target > sorted_numbers[mid]:
            low = mid + 1
        elif target < sorted_numbers[mid]:
            high = mid - 1

if __name__ == "__main__":
    binary_search()
