class Node:
    """Node of a Binary Search Tree"""
    def __init__(self, key, value, parent=None):
        """
        Initialize a node with a key, value, and parent.
        
        Args:
            key: The key of the node.
            value: The value associated with the key.
            parent: The parent node of this node (default is None).
        """
        self.key = key
        self.value = value
        self.left = None
        self.right = None
        self.parent = parent

    def remove_child(self, child_node):
        """Remove a child node."""
        if child_node == self.left:
            self.left = None
        elif child_node == self.right:
            self.right = None

    def print_info(self):
        """Print information about the node."""
        left_str = "        " if self.left is None else f"_left: {self.left.value}"
        right_str = "         " if self.right is None else f"_right: {self.right.value}"
        parent_str = "   ROOT" if self.parent is None else f"   _parent: {self.parent.value}"
        print(f"Node({self.value})  {left_str}  {right_str}  {parent_str}")

    def is_leaf(self):
        """Check if the node is a leaf."""
        return self.left is None and self.right is None

    def is_root(self):
        """Check if the node is the root."""
        return self.parent is None

    def set_right(self, child_node):
        """Set the right child of the node."""
        self.right = child_node
        child_node.parent = self

    def set_left(self, child_node):
        """Set the left child of the node."""
        self.left = child_node
        child_node.parent = self

    def replace_child(self, old_child, new_child):
        """Replace a child node with a new node."""
        if old_child == self.left:
            self.set_left(new_child)
        elif old_child == self.right:
            self.set_right(new_child)
        else:
            raise ValueError("old_child node not found")


class BinarySearchTree:
    """Binary Search Tree implemented using linked list"""
    def __init__(self):
        """Initialize an empty Binary Search Tree."""
        self.root = None

    def count_nodes(self):
        """Count the number of nodes in the tree."""
        return self._count_nodes(self.root)

    def _count_nodes(self, subtree_root):
        """Recursively count the number of nodes in the subtree."""
        if subtree_root:
            return 1 + self._count_nodes(subtree_root.left) + self._count_nodes(subtree_root.right)
        return 0

    def insert(self, key, value):
        """Insert a new node into the tree."""
        if key is None or value is None:
            raise ValueError("Key or value cannot be None.")
        if not self.root:
            self.root = Node(key, value)
        else:
            self._insert(self.root, key, value)

    def _insert(self, subtree_root, key, value):
        """Recursively insert a new node into the subtree."""
        if key == subtree_root.key:
            raise ValueError("Duplicate key: " + str(key))
        elif key < subtree_root.key:
            if not subtree_root.left:
                subtree_root.set_left(Node(key, value))
            else:
                self._insert(subtree_root.left, key, value)
        else:
            if not subtree_root.right:
                subtree_root.set_right(Node(key, value))
            else:
                self._insert(subtree_root.right, key, value)

    def search(self, search_key):
        """Search for a node with the given key."""
        return self._search(self.root, search_key)

    def _search(self, subtree_root, search_key):
        """Recursively search for a node with the given key in the subtree."""
        if not subtree_root or subtree_root.key == search_key:
            return subtree_root
        if search_key < subtree_root.key:
            return self._search(subtree_root.left, search_key)
        return self._search(subtree_root.right, search_key)

    def find_max_node(self, subtree_root):
        """Find the node with the maximum key in the subtree."""
        while subtree_root.right:
            subtree_root = subtree_root.right
        return subtree_root

    def delete(self, key):
        """Delete a node with the given key from the tree."""
        node = self.search(key)
        if not node:
            raise ValueError(f"Node with key {key} not found.")
        if node.is_leaf():
            if node.parent:
                node.parent.remove_child(node)
            else:
                self.root = None
        elif not node.left:
            node.parent.replace_child(node, node.right)
        elif not node.right:
            node.parent.replace_child(node, node.left)
        else:
            max_left_node = self.find_max_node(node.left)
            node.key, node.value = max_left_node.key, max_left_node.value
            self.delete(max_left_node.key)


class TreePrinter:
    """Tree Printer for Binary Search Tree"""
    def __init__(self, bst):
        """Initialize the tree printer with the given Binary Search Tree."""
        self.bst = bst

    def print_preorder(self):
        """Print the tree in preorder traversal."""
        self._preorder(self.bst.root)

    def _preorder(self, node):
        """Recursively print the tree in preorder traversal."""
        if node:
            node.print_info()
            self._preorder(node.left)
            self._preorder(node.right)
