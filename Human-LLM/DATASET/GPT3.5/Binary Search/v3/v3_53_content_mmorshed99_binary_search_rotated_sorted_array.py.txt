class Solution:
    def search(self, nums, target):
        def find_break_point(start, end):
            mid = start + (end - start) // 2
            if end - start < 2:
                return start
            if nums[mid] < nums[mid - 1] and nums[mid] < nums[mid + 1]:
                return mid
            elif nums[mid] > nums[mid - 1] and nums[mid] > nums[mid + 1]:
                return mid
            elif nums[mid] > nums[end]:
                return find_break_point(mid, end)
            elif nums[mid] < nums[start]:
                return find_break_point(start, mid)
            else:
                return -1

        def binary_search(start, end):
            mid = start + (end - start) // 2
            if target == nums[mid]:
                return mid
            if end - start < 2:
                if target == nums[start]:
                    return start
                elif target == nums[end]:
                    return end
                else:
                    return -1
            elif target > nums[mid]:
                return binary_search(mid, end)
            else:
                return binary_search(start, mid)

        break_point = find_break_point(0, len(nums) - 1)
        if break_point == -1:
            return binary_search(0, len(nums) - 1)
        else:
            left_search = binary_search(0, break_point - 1)
            if left_search != -1:
                return left_search
            else:
                return binary_search(break_point, len(nums) - 1)

# Example usage
if __name__ == "__main__":
    sol = Solution()
    nums = [4, 5, 6, 7, 0, 1, 2]
    target = 0
    result = sol.search(nums, target)
    print("Index of the target element in the rotated sorted array:", result)
