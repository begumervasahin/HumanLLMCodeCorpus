import random

class Node:
    def __init__(self, value):
        self.value = value
        self.left = None
        self.right = None

    def insert(self, value):
        if value < self.value:
            if self.left is None:
                self.left = Node(value)
            else:
                self.left.insert(value)
        elif value > self.value:
            if self.right is None:
                self.right = Node(value)
            else:
                self.right.insert(value)
        # If value is equal, we don't insert it again as it's assumed to be a binary tree

    def to_array_dfs(self, node, array):
        '''
        Depth-First Search to convert binary tree to array
        '''
        if node:
            array.append(node.value)
            self.to_array_dfs(node.left, array)
            self.to_array_dfs(node.right, array)
        return array

class BinaryTree:
    def __init__(self):
        self.root = None

    def insert(self, data):
        if self.root is None:
            self.root = Node(data)
        else:
            self.root.insert(data)

    def find(self, data):
        return self._find_node(self.root, data)

    def _find_node(self, node, data):
        '''
        Search for a value in the binary tree
        '''
        if node is None:
            return False
        elif data == node.value:
            return True
        elif data < node.value:
            return self._find_node(node.left, data)
        else:
            return self._find_node(node.right, data)

    def to_array(self):
        '''
        Convert binary tree to array using Depth-First Search
        '''
        return self.root.to_array_dfs(self.root, [])

    def rearrange(self):
        '''
        Rearrange the binary tree with median as root
        @return Binary Tree in DFS Array Form
        '''
        nodes = self.to_array()
        nodes.sort()
        middle = len(nodes) // 2
        self.root = Node(nodes.pop(middle))
        for element in nodes:
            self.root.insert(element)
        return self.to_array()

# Example usage:
tree = BinaryTree()
for i in range(50):
    tree.insert(random.randint(0, 100))

print("Original Binary Tree:", tree.to_array())
print("Rearranged Binary Tree with Median as Root:", tree.rearrange())
print("Is 55 in the tree?", tree.find(55))
