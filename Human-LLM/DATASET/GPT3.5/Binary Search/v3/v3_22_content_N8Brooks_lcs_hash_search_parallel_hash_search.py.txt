from array import array
from zlib import crc32
import multiprocessing as mp

# Number of parallel processes
PARALLEL = 4

def hash_function(word, base):
    """Hash function using CRC32."""
    return crc32(word.encode(), base)

def found_common(set_a, set_b, m, a, b):
    """Check if there is a common substring of length m between set_a and set_b."""
    set_a_indices = {x: i for i, x in enumerate(set_a)}
    return any(x in set_a_indices and a[set_a_indices[x]:set_a_indices[x] + m] == b[i:i + m]
               for i, x in enumerate(set_b))

def lcs(a, b):
    """
    Compute the length of the Longest Common Substring (LCS) between strings a and b,
    and return a list of LCSs.

    Args:
        a (str): The first string.
        b (str): The second string.

    Returns:
        int: The length of the LCS.
        list: A list of longest common substrings.
    """
    # Initialize a pool of parallel processes
    pool = mp.Pool(PARALLEL)

    # Ensure len_a is not greater than len_b
    len_a, len_b = len(a) + 1, len(b) + 1
    if len_a > len_b:
        len_a, len_b = len_b, len_a
        a, b = b, a

    # Initialize variables for binary search
    l, r = 0, 1

    # Initialize arrays for rolling hashes
    L_a = array('L', [0] * len_a)
    R_a = array('L', pool.starmap(hash_function, zip(a, L_a[1:])))
    L_b = array('L', [0] * len_b)
    R_b = array('L', pool.starmap(hash_function, zip(b, L_b[1:])))

    # Perform a binary search to find the longest common substring
    while r < len_a and found_common(R_a, R_b, r, a, b):
        L_a, R_a = R_a, array('L', pool.starmap(hash_function, zip((a[i:i + r] for i in range(r, len(a))), R_a[:-r])))
        L_b, R_b = R_b, array('L', pool.starmap(hash_function, zip((b[i:i + r] for i in range(r, len(b))), R_b[:-r])))
        l, r = r, r * 2

    r = min(r, len(a))

    while l < r:
        m = (l + r) // 2
        d = m - l
        R_a = array('L', pool.starmap(hash_function, zip((a[i:i + d] for i in range(d, len(a))), L_a[:-d])))
        R_b = array('L', pool.starmap(hash_function, zip((b[i:i + d] for i in range(d, len(a))), L_b[:-d])))

        if found_common(R_a, R_b, m, a, b):
            l = m
            L_a, L_b = R_a, R_b
        else:
            r = m

    # Find the longest common substring
    set_a_indices = {x: i for i, x in enumerate(L_a)}
    unique = []
    for i, x in enumerate(L_b):
        loc = set_a_indices.get(x, None)
        if loc is not None and a[loc:loc + m] == b[i:i + m]:
            unique.append(a[loc:loc + m])

    return l, unique

# Example usage:
if __name__ == "__main__":
    a = "ABAB"
    b = "BABA"
    length, lcs_list = lcs(a, b)
    print(f"Length of LCS: {length}")
    print(f"Longest Common Substrings: {lcs_list}")
