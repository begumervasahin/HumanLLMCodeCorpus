import time

def binary_search(nums, item):
    """
    Perform binary search on a sorted list.
    
    Args:
    nums (list): The sorted list of numbers.
    item (int): The number to search for.
    
    Returns:
    bool: True if the item is found, False otherwise.
    float: The time taken to perform the search in microseconds.
    """
    first = 0
    last = len(nums) - 1
    found = False
    start_time = time.time()

    while first <= last and not found:
        middle = (first + last) // 2
        if nums[middle] == item:
            found = True
        else:
            if item < nums[middle]:
                last = middle - 1
            else:
                first = middle + 1

    elapsed_time = (time.time() - start_time) * 1_000_000  # Convert to microseconds
    return found, elapsed_time

def calculate_runtime(search_function, nums, item):
    """
    Calculate the runtime of a search function.
    
    Args:
    search_function (function): The search function to be timed.
    nums (list): The list of numbers to search in.
    item (int): The number to search for.
    
    Returns:
    bool: True if the item is found, False otherwise.
    float: The time taken to perform the search in microseconds.
    float: The total time including function overhead in seconds.
    """
    start_time = time.perf_counter()
    found, elapsed_time = search_function(nums, item)
    end_time = time.perf_counter()
    total_time = end_time - start_time

    return found, elapsed_time, total_time

    number_list = [-125, -50, -1, 0, 1, 5, 7, 10, 12, 20, 26, 30, 50, 127, 500, 833, 900, 1000, 12345, 123456]
    number_to_find = int(input("Enter the number you want to find: "))
