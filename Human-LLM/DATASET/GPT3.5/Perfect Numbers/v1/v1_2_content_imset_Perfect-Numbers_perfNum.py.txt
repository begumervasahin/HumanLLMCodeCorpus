import sys

def perf_check(arg1, arg2):
    """
    Check if a number is perfect or iterate through a range of numbers and check for perfection.
    
    Args:
        arg1 (str): The mode ('check' or 'iterate').
        arg2 (int): The number to check or the upper limit for iteration.
    """
    num_chk = 0
    if arg1.lower() == 'check':
        num_chk = int(arg2)
    elif arg1.lower() == 'iterate':
        num_chk = 2
    else:
        print("Argument", arg1, "is not valid.")
        return
    
    while num_chk <= int(arg2):
        factor_list = []
        factor_chk = 1
        while factor_chk < num_chk:
            if num_chk % factor_chk == 0:
                factor_list.append(factor_chk)
            factor_chk += 1
        factor_sum = sum(factor_list)
        if factor_chk == factor_sum:
            print(str(num_chk) + " is perfect!")
        else:
            print(str(num_chk) + " is not perfect!")
        if arg1.lower() == 'check':
            break
        else:
            num_chk += 1

if __name__ == '__main__':
    if len(sys.argv) != 3:
        print("Usage: python script_name.py [check | iterate] number")
    else:
        perf_check(sys.argv[1], sys.argv[2])
