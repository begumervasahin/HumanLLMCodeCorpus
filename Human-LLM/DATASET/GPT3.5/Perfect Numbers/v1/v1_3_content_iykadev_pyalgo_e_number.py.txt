import time

def e_number(start, end):
    for i in range(start, end):
        result = (1 + (1 / i)) ** i
        print(result)

def e_number_main():
    start_time = time.time()
    e_number(1, 100000)
    end_time = time.time()
    print("Execution time:", end_time - start_time, "seconds")

if __name__ == "__main__":
    e_number_main()
