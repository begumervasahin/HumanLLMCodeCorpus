def anagram(s1, s2):
    c1 = [0] * 26  # Initialize a list to count occurrences of each character in s1
    c2 = [0] * 26  # Initialize a list to count occurrences of each character in s2

    # Count occurrences of each character in s1
    for char in s1:
        pos = ord(char) - ord('a')  # Get the position of the character in the alphabet
        c1[pos] += 1

    # Count occurrences of each character in s2
    for char in s2:
        pos = ord(char) - ord('a')  # Get the position of the character in the alphabet
        c2[pos] += 1

    # Compare the counts of characters in both strings
    for j in range(26):
        if c1[j] != c2[j]:
            return False  # If counts don't match, strings are not anagrams

    return True  # If all counts match, strings are anagrams

# Test the function with example inputs
print(anagram('pide', 'epid'))  # Output: True
