import math

def find_sum_prop_factors(x):
    factors = []
    for i in range(1, math.ceil(x / 2) + 1):
        if x % i == 0:
            factors.append(i)
    return sum(factors)

abundants = []
for k in range(12, 28123):
    if k % 1000 == 0:
        print("trying", k, "...")
    if find_sum_prop_factors(k) > k:
        abundants.append(k)

sum2abundants = set()
for p in range(len(abundants)):
    if p % 100 == 0:
        print("passing through p being", p, "out of", len(abundants))
    for q in range(len(abundants)):
        if abundants[p] + abundants[q] < 28123:
            sum2abundants.add(abundants[p] + abundants[q])

output = 0
for z in range(28123):
    if z % 100 == 0:
        print("passing through z being", z, "out of 28123")
    if z not in sum2abundants:
        output += z

print("The answer to this question should be", output)
