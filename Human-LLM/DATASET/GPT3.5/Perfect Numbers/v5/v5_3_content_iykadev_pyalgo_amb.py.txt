import itertools

class Amb(object):
    def __init__(self):
        self._names2values = {}
        self._func = None
        self._value_iterator = None
        self._func_arg_names = None

    def __call__(self, arg=None):
        if hasattr(arg, '__code__'):
            globals_dict = arg.__globals__ if hasattr(arg, '__globals__') else arg.func_globals
            arg_names = arg.__code__.co_varnames[:arg.__code__.co_argcount]
            for name in arg_names:
                if name not in self._names2values:
                    assert name in globals_dict, f"Global name {name} not found in function globals"
                    self._names2values[name] = globals_dict[name]
            value_sets = [self._names2values[name] for name in arg_names]
            self._value_iterator = itertools.product(*value_sets)
            self._func = arg
            self._func_arg_names = arg_names
            return self
        elif arg is not None:
            arg = frozenset(arg)
            return arg
        else:
            return self._next_in_search()

    def _next_in_search(self):
        arg = self._func
        globals_dict = arg.__globals__
        arg_names = self._func_arg_names
        found = False
        for values in self._value_iterator:
            if arg(*values):
                found = True
                for name, value in zip(arg_names, values):
                    globals_dict[name] = value
                break
        if not found:
            raise StopIteration
        return values

    def __iter__(self):
        return self

    def __next__(self):
        return self()

    next = __next__
