def is_anagram(s1, s2):
    """
    Check if two strings are anagrams of each other.

    Parameters:
    - s1: str, first string
    - s2: str, second string

    Returns:
    - bool, True if s1 and s2 are anagrams, False otherwise
    """
    # Convert s2 to a list for easier manipulation
    s2_list = list(s2)

    # Iterate through each character in s1
    for char in s1:
        # Check if the character is in s2_list
        if char in s2_list:
            # If found, remove it from s2_list
            s2_list.remove(char)
        else:
            # If not found, strings are not anagrams
            return False

    # If s1 is exhausted and all characters found in s2_list, they are anagrams
    return not s2_list

# Test the function
print(is_anagram('ey edip', 'pide ye'))
