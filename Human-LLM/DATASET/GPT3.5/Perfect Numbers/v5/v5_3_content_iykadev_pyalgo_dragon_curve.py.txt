from turtle import *

def draw_dragon_curve(step, length):
    """
    Draw a dragon curve using turtle graphics.

    Args:
        step (int): The number of recursive steps.
        length (float): The initial length of each segment.
    """
    _draw_dragon_curve(step, length)

def _draw_dragon_curve(step, length):
    """
    Recursive function to draw a dragon curve.

    Args:
        step (int): The number of recursive steps remaining.
        length (float): The current length of each segment.
    """
    step -= 1
    length /= 1.41421

    if step > 0:
        _turn_right(45)
        _draw_dragon_curve(step, length)
        _turn_left(90)
        _draw_dragon_curve_reverse(step, length)
        _turn_right(45)
    else:
        _turn_right(45)
        _move_forward(length)
        _turn_left(90)
        _move_forward(length)
        _turn_right(45)

def _draw_dragon_curve_reverse(step, length):
    """
    Recursive function to draw the reverse part of a dragon curve.

    Args:
        step (int): The number of recursive steps remaining.
        length (float): The current length of each segment.
    """
    step -= 1
    length /= 1.41421

    if step > 0:
        _turn_left(45)
        _draw_dragon_curve(step, length)
        _turn_right(90)
        _draw_dragon_curve_reverse(step, length)
        _turn_left(45)
    else:
        _turn_left(45)
        _move_forward(length)
        _turn_right(90)
        _move_forward(length)
        _turn_left(45)

def _turn_left(angle):
    """
    Turn the turtle left by the specified angle.

    Args:
        angle (float): The angle to turn left.
    """
    left(angle)

def _turn_right(angle):
    """
    Turn the turtle right by the specified angle.

    Args:
        angle (float): The angle to turn right.
    """
    right(angle)

def _move_forward(distance):
    """
    Move the turtle forward by the specified distance.

    Args:
        distance (float): The distance to move forward.
    """
    forward(distance)
