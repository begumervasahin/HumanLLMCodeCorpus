def move_disk(source, destination):
    """Prints the move of a single disk from source rod to destination rod."""
    print(f"Move disk from {source} to {destination}")


def hanoi_tower_algorithm(n, source, auxiliary, destination):
    """
    Solves the Tower of Hanoi problem recursively.

    Args:
        n (int): The number of disks to be moved.
        source (str): The rod from which the disks are initially stacked.
        auxiliary (str): The auxiliary rod to facilitate the movement of disks.
        destination (str): The rod onto which the disks are to be moved.
    """
    if n == 1:
        move_disk(source, destination)
    else:
        hanoi_tower_algorithm(n - 1, source, destination, auxiliary)
        move_disk(source, destination)
        hanoi_tower_algorithm(n - 1, auxiliary, source, destination)


def hanoi_tower_algorithm_main(ndiscs):
    """
    Demonstrates the Tower of Hanoi algorithm for a given number of discs.

    Args:
        ndiscs (int): The number of discs in the Tower of Hanoi puzzle.
    """
    print(f"Solving Tower of Hanoi problem for {ndiscs} discs:")
    hanoi_tower_algorithm(ndiscs, "A", "B", "C")


# Example usage:
hanoi_tower_algorithm_main(3)
