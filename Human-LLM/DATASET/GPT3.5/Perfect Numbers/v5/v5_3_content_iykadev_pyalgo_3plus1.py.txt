def collatz_sequence(n):
    """
    Generate and print the Collatz sequence starting from a given number.

    Parameters:
    - n: int, the starting number for the Collatz sequence
    """
    step_count = 0
    while n > 1:
        step_count += 1
        if n % 2 == 1:  # Check if n is odd
            n = 3 * n + 1
        else:
            n = n // 2  # Use integer division for even numbers
        print(n)

def main_collatz_sequence(starting_number):
    """
    Main function to generate and print the Collatz sequence for a given starting number.

    Parameters:
    - starting_number: int, the starting number for the Collatz sequence
    """
    collatz_sequence(starting_number)

# Example usage:
main_collatz_sequence(15000)
