import cProfile

def compute_golden_ratio_sequence(n):
    """Compute and print the golden ratio sequence up to the nth term."""
    prev_term = 1
    curr_term = 1
    for _ in range(2, n):
        next_term = prev_term + curr_term
        prev_term = curr_term
        curr_term = next_term
        ratio = curr_term / prev_term
        print(ratio)

def main():
    """Main function to compute and print the golden ratio sequence."""
    num_terms = 1476  # Number of terms in the sequence
    compute_golden_ratio_sequence(num_terms)

if __name__ == "__main__":
    # Profile the main function and save results to a file
    cProfile.run("main()", filename="profile_results.txt")
