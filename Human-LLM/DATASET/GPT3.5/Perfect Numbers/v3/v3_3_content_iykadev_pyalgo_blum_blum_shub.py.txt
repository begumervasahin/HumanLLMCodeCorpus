import random

def generate_probable_prime(bits=256):
    """
    Generate a random probable prime with a given number of bits.
    """
    candidate = random.getrandbits(bits) | 1  # Ensure odd number
    while not is_probable_prime(candidate):
        candidate += 2  # Increment by 2 to maintain oddness
    return candidate

def is_probable_prime(n, num_bases=90):
    """
    Test whether a number is a probable prime using Miller-Rabin test.
    """
    if n <= 1:
        return False
    bases = [random.randrange(2, 50000) for _ in range(num_bases)]
    for b in bases:
        if n % b == 0:
            return False
    tests, s = 0, 0
    m = n - 1
    while not m & 1:
        m >>= 1
        s += 1
    for b in bases:
        tests += 1
        is_prob = miller_rabin(m, s, b, n)
        if not is_prob:
            return False
    return True

def miller_rabin(m, s, b, n):
    """
    Miller-Rabin primality test based on Algorithm P in 'Art of Computer Programming' by D. Knuth.
    """
    y = pow(b, m, n)
    for _ in range(s):
        if (y == 1 and s == 0) or (y == n - 1):
            return True
        y = pow(y, 2, n)
    return False

class BlumBlumShub:
    def __init__(self, bits):
        """
        Initialize Blum-Blum-Shub generator with the specified number of bits.
        """
        self.n = self._generate_n(bits)
        length = self._bit_length(self.n)
        seed = random.getrandbits(length)
        self._set_seed(seed)

    def _generate_prime(self, bits):
        """
        Generate an appropriate prime number for use in Blum-Blum-Shub.
        """
        while True:
            p = generate_probable_prime(bits)
            if p & 3 == 3:  # Check if p = 3 mod 4
                return p

    def _generate_n(self, bits):
        """
        Generate the 'n value' for Blum-Blum-Shub algorithm.
        """
        p = self._generate_prime(bits // 2)
        while True:
            q = self._generate_prime(bits // 2)
            if p != q:
                return p * q

    def _set_seed(self, seed):
        """
        Set or reset the seed value and internal state.
        """
        self.state = seed % self.n

    def _bit_length(self, x):
        """
        Get the bit length of a positive number.
        """
        assert x > 0
        length = 0
        while x:
            length += 1
            x >>= 1
        return length

    def next(self, num_bits):
        """
        Return up to num_bits random bits.
        """
        result = 0
        for _ in range(num_bits):
            self.state = (self.state ** 2) % self.n
            result = (result << 1) | (self.state & 1)
        return result
