import math

def find_proper_factors_sum(x):
    factors = [i for i in range(1, math.ceil(x / 2) + 1) if x % i == 0]
    return sum(factors)

def find_abundant_numbers(limit):
    abundant_numbers = []
    for num in range(12, limit):
        if find_proper_factors_sum(num) > num:
            abundant_numbers.append(num)
    return abundant_numbers

def find_sum_of_two_abundants(abundant_numbers):
    sum_of_two_abundants = set()
    for i in range(len(abundant_numbers)):
        for j in range(len(abundant_numbers)):
            if abundant_numbers[i] + abundant_numbers[j] < 28123:
                sum_of_two_abundants.add(abundant_numbers[i] + abundant_numbers[j])
    return sum_of_two_abundants

def find_non_abundant_sums(limit, sum_of_two_abundants):
    non_abundant_sums = sum(num for num in range(limit) if num not in sum_of_two_abundants)
    return non_abundant_sums

if __name__ == "__main__":
    limit = 28123
    
    print("Finding abundant numbers...")
    abundant_numbers = find_abundant_numbers(limit)

    print("Finding sum of two abundant numbers...")
    sum_of_two_abundants = find_sum_of_two_abundants(abundant_numbers)

    print("Finding sum of all positive integers that cannot be written as the sum of two abundant numbers...")
    non_abundant_sums = find_non_abundant_sums(limit, sum_of_two_abundants)

    print("The sum of all positive integers that cannot be written as the sum of two abundant numbers is:", non_abundant_sums)
