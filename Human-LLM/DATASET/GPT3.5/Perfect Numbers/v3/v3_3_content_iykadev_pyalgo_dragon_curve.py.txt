from turtle import *

# Function to draw a dragon curve
def draw_dragon_curve(step, length):
    _draw_dragon_curve_right(step, length)

# Helper function to draw the dragon curve to the right
def _draw_dragon_curve_right(step, length):
    step -= 1
    length /= 1.41421
    if step > 0:
        right(45)
        _draw_dragon_curve_right(step, length)
        left(90)
        _draw_dragon_curve_left(step, length)
        right(45)
    else:
        _draw_line(length, 45)

# Helper function to draw the dragon curve to the left
def _draw_dragon_curve_left(step, length):
    step -= 1
    length /= 1.41421
    if step > 0:
        left(45)
        _draw_dragon_curve_right(step, length)
        right(90)
        _draw_dragon_curve_left(step, length)
        left(45)
    else:
        _draw_line(length, -45)

# Helper function to draw a line
def _draw_line(length, angle):
    setheading(angle)
    forward(length)

# Test the dragon curve function
speed(0)
penup()
goto(-200, 0)
pendown()
draw_dragon_curve(12, 300)

# Keep the window open
done()
