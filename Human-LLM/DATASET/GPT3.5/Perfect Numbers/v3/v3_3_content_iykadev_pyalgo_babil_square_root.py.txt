def babylonian_square_root(number, epsilon):
    """
    Compute the square root of a number using the Babylonian method.

    Parameters:
        number (float): The number for which to compute the square root.
        epsilon (float): The desired precision of the result.

    Returns:
        float: The approximate square root of the number.
    """
    # Initial guess for the square root using Babylonian method
    x0 = (1 + number) / 2

    # Improved guess using Babylonian method
    x1 = (x0 + number / x0) * 0.5

    # Calculate the relative error
    relative_error = abs((x1 - x0) / x0)

    # Iteratively refine the estimate until the desired precision is reached
    while relative_error > epsilon:
        x0 = x1
        x1 = (x0 + number / x0) * 0.5
        relative_error = abs((x1 - x0) / x0)

    return x1

def babylonian_square_root_main():
    """
    Main function to demonstrate Babylonian square root calculation.
    """
    number = 5
    precision = 1e-7
    result = babylonian_square_root(number, precision)
    result_string = "The square root of {} is {:.10f}".format(number, result)
    print(result_string)

# Call the main function
if __name__ == "__main__":
    babylonian_square_root_main()
