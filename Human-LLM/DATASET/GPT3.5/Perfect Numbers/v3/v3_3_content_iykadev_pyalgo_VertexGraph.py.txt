class Vertex:
    def __init__(self, key):
        self.id = key
        self.connected_to = {}

    def add_neighbor(self, nbr, weight=0):
        self.connected_to[nbr] = weight

    def __str__(self):
        return f"Vertex {self.id} connected to: {list(self.connected_to.keys())}"

    def get_connections(self):
        return self.connected_to.keys()

    def get_id(self):
        return self.id

    def get_weight(self, nbr):
        return self.connected_to[nbr]


class Graph:
    def __init__(self):
        self.vertex_list = {}
        self.num_vertices = 0

    def add_vertex(self, key):
        self.num_vertices += 1
        new_vertex = Vertex(key)
        self.vertex_list[key] = new_vertex
        return new_vertex

    def get_vertex(self, n):
        return self.vertex_list.get(n)

    def __contains__(self, n):
        return n in self.vertex_list

    def add_edge(self, from_vertex, to_vertex, cost=0):
        if from_vertex not in self.vertex_list:
            self.add_vertex(from_vertex)
        if to_vertex not in self.vertex_list:
            self.add_vertex(to_vertex)
        self.vertex_list[from_vertex].add_neighbor(self.vertex_list[to_vertex], cost)

    def get_vertices(self):
        return self.vertex_list.keys()

    def __iter__(self):
        return iter(self.vertex_list.values())


def build_graph(word_file):
    buckets = {}
    graph = Graph()
    with open(word_file, 'r') as file:
        for line in file:
            word = line.strip()
            for i in range(len(word)):
                bucket = word[:i] + '_' + word[i + 1:]
                if bucket in buckets:
                    buckets[bucket].append(word)
                else:
                    buckets[bucket] = [word]
    for bucket in buckets.keys():
        for word1 in buckets[bucket]:
            for word2 in buckets[bucket]:
                if word1 != word2:
                    graph.add_edge(word1, word2)
    return graph


def knight_graph(board_size):
    kt_graph = Graph()
    for row in range(board_size):
        for col in range(board_size):
            node_id = pos_to_node_id(row, col, board_size)
            new_positions = gen_legal_moves(row, col, board_size)
            for e in new_positions:
                new_id = pos_to_node_id(e[0], e[1], board_size)
                kt_graph.add_edge(node_id, new_id)
    return kt_graph


def pos_to_node_id(row, column, board_size):
    return (row * board_size) + column


def gen_legal_moves(x, y, board_size):
    new_moves = []
    move_offsets = [(-1, -2), (-1, 2), (-2, -1), (-2, 1),
                    (1, -2), (1, 2), (2, -1), (2, 1)]
    for i in move_offsets:
        new_x = x + i[0]
        new_y = y + i[1]
        if legal_coord(new_x, board_size) and legal_coord(new_y, board_size):
            new_moves.append((new_x, new_y))
    return new_moves


def legal_coord(x, board_size):
    return 0 <= x < board_size


def knight_tour(n, path, current_vertex, limit):
    current_vertex.set_color('gray')
    path.append(current_vertex)
    if n < limit:
        neighbors = list(current_vertex.get_connections())
        i = 0
        done = False
        while i < len(neighbors) and not done:
            if neighbors[i].get_color() == 'white':
                done = knight_tour(n + 1, path, neighbors[i], limit)
            i += 1
        if not done:
            path.pop()
            current_vertex.set_color('white')
    else:
        done = True
    return done
