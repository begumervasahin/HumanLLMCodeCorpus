def three_plus_one_sequence(x):
    """
    Perform the 3n+1 sequence on the given number x until it reaches 1.
    
    Parameters:
        x (int): The initial number for the sequence.
    
    Returns:
        int: The number of iterations taken to reach 1.
    """
    iterations = 0  # Initialize a counter for the number of iterations
    while x > 1:
        iterations += 1  # Increment the iteration counter by 1
        if x % 2:  # If x is odd
            x = 3 * x + 1  # Multiply x by 3 and add 1
        else:
            x //= 2  # If x is even, divide it by 2 using integer division
        print(x)  # Print the current value of x
    return iterations

def main():
    """
    Main function to demonstrate the 3n+1 sequence with an example.
    """
    initial_value = 15000
    print(f"Starting 3n+1 sequence with initial value: {initial_value}")
    iterations_taken = three_plus_one_sequence(initial_value)
    print(f"Number of iterations taken: {iterations_taken}")

# Call the main function
if __name__ == "__main__":
    main()
