def is_anagram(s1, s2):
    # Convert s2 to a list for easier manipulation
    alist = list(s2)
    
    pos1 = 0
    still_ok = True  # Use snake_case for variable names
    
    # Iterate through each character in s1
    while pos1 < len(s1) and still_ok:
        pos2 = 0
        found = False
        
        # Search for the current character of s1 in alist (s2)
        while pos2 < len(alist) and not found:
            if s1[pos1] == alist[pos2]:
                found = True
            else:
                pos2 += 1
                
        # If the character is found, mark it as None in alist
        if found:
            alist[pos2] = None
        else:
            # If the character is not found, the strings are not anagrams
            still_ok = False
        
        pos1 += 1
        
    return still_ok

# Test the function with example inputs
print(is_anagram('ey edip', 'pide ye'))  # Output: True
