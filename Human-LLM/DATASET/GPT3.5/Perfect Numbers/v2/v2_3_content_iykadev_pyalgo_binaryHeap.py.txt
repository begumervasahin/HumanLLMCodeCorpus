class BinaryHeap:
    def __init__(self):
        self.heap_list = [0]  # Initial dummy element
        self.current_size = 0

    def percolate_up(self, i):
        while i // 2 > 0:
            if self.heap_list[i] < self.heap_list[i // 2]:
                # Swap parent and child if child is smaller
                self.heap_list[i], self.heap_list[i // 2] = self.heap_list[i // 2], self.heap_list[i]
            i = i // 2

    def insert(self, k):
        # Add new element to the end of the list
        self.heap_list.append(k)
        self.current_size += 1
        # Percolate up to maintain heap property
        self.percolate_up(self.current_size)

    def percolate_down(self, i):
        while (i * 2) <= self.current_size:
            min_child = self.min_child(i)
            if self.heap_list[i] > self.heap_list[min_child]:
                # Swap parent and minimum child if parent is greater
                self.heap_list[i], self.heap_list[min_child] = self.heap_list[min_child], self.heap_list[i]
            i = min_child

    def min_child(self, i):
        # Return index of the minimum child of a parent
        left_child = i * 2
        right_child = (i * 2) + 1
        if right_child > self.current_size:
            return left_child
        else:
            if self.heap_list[left_child] < self.heap_list[right_child]:
                return left_child
            else:
                return right_child

    def delete_min(self):
        if self.current_size == 0:
            return None
        min_value = self.heap_list[1]  # Minimum value is at index 1
        # Move the last element to the root position
        self.heap_list[1] = self.heap_list[self.current_size]
        self.current_size -= 1
        self.heap_list.pop()
        # Percolate down to maintain heap property
        self.percolate_down(1)
        return min_value

    def build_heap(self, alist):
        mid_point = len(alist) // 2
        self.current_size = len(alist)
        # Copy the list and treat it as the heap list
        self.heap_list = [0] + alist[:]
        # Perform percolate down on non-leaf nodes
        for i in range(mid_point, 0, -1):
            self.percolate_down(i)

# Example usage
bh = BinaryHeap()
bh.build_heap([9, 5, 6, 2, 3])
print(bh.delete_min())
print(bh.delete_min())
