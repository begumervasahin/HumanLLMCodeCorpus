def is_anagram(s1, s2):
    # Convert s1 and s2 to lists
    alist1 = list(s1)
    alist2 = list(s2)
    
    # Sort the lists of characters in ascending order
    alist1.sort()
    alist2.sort()
    
    pos = 0
    matches = True
    
    # Iterate through each character of s1 and compare with s2
    while pos < len(s1) and matches:
        if alist1[pos] == alist2[pos]:
            pos += 1
        else:
            # If characters at the same position don't match, the strings are not anagrams
            matches = False
            
    return matches

# Test the function with example inputs
print(is_anagram('ey edip', 'pide ye'))  # Output: True
