def bubble_sort(arr):
    """
    Sorts a list using the bubble sort algorithm.
    """
    # Iterate through the list from the end to the beginning
    for pass_num in range(len(arr) - 1, 0, -1):
        # Iterate through the list from the beginning to pass_num
        for i in range(pass_num):
            # Compare adjacent elements and swap them if they are in the wrong order
            if arr[i] > arr[i + 1]:
                # Swap elements
                arr[i], arr[i + 1] = arr[i + 1], arr[i]

# Test the bubble_sort function
test_list = [54, 26, 93, 17, 77, 31, 44, 55, 20]
bubble_sort(test_list)
print(test_list)  # Output: [17, 20, 26, 31, 44, 54, 55, 77, 93]
