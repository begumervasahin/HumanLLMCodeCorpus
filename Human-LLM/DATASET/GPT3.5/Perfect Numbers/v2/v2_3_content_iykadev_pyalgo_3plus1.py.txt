def three_plus_one(x):
    count_iterations = 0  # Initialize a counter for the number of iterations
    while x > 1:
        count_iterations += 1  # Increment the iteration counter by 1
        if x % 2:  # If x is odd
            x = 3 * x + 1  # Multiply x by 3 and add 1
        else:
            x = x / 2  # If x is even, divide it by 2
        print(x)  # Print the current value of x

def three_plus_one_main(x):
    # Call the function three_plus_one to perform the 3n+1 sequence
    three_plus_one(x)

# Call the main function with an initial value of x
three_plus_one_main(15000)
