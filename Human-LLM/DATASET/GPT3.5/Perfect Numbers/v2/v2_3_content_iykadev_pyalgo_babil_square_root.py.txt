def babylonian_square_root(number, epsilon):
    # Initial guess for the square root using Babylonian method
    x0 = (1 + number) / 2
    
    # Improved guess using Babylonian method
    x1 = (x0 + number / x0) * 0.5
    
    # Calculate the relative error
    relative_error = abs((x1 - x0) / x0)
    
    # Iteratively refine the estimate until the desired precision is reached
    while relative_error > epsilon:
        x0 = x1
        x1 = (x0 + number / x0) * 0.5
        relative_error = abs((x1 - x0) / x0)
    
    # Format the result string
    result_string = "The square root of {} is {}".format(number, x1)
    
    # Print the result
    print(result_string)

def babylonian_square_root_main():
    # Example usage with number 5 and precision 1e-7
    babylonian_square_root(5, 1e-7)

# Call the main function
babylonian_square_root_main()
