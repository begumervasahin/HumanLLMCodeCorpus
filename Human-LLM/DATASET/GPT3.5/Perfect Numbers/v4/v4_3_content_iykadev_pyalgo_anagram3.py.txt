def is_anagram(s1, s2):
    # Initialize character count lists for s1 and s2
    count_s1 = [0] * 26
    count_s2 = [0] * 26
    
    # Count occurrences of each character in s1
    for char in s1:
        pos = ord(char) - ord('a')
        count_s1[pos] += 1
    
    # Count occurrences of each character in s2
    for char in s2:
        pos = ord(char) - ord('a')
        count_s2[pos] += 1
    
    # Compare the counts of characters in both strings
    index = 0
    still_ok = True
    while index < 26 and still_ok:
        if count_s1[index] == count_s2[index]:
            index += 1
        else:
            still_ok = False
    
    # Return whether s1 and s2 are anagrams
    return still_ok

# Test the function
print(is_anagram('pide', 'epid'))
