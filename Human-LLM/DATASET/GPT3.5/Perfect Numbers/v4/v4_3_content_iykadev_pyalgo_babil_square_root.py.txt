def babylonian_square_root(number, epsilon):
    # Initial guess for square root using Babylonian method
    x0 = (1 + number) / 2
    x1 = (x0 + number / x0) * 0.5
    
    # Calculate the relative error
    error = (x1 - x0) / x0
    if error < 0:
        error = -error
    
    # Iterate until the error is within the desired epsilon
    while error > epsilon:
        x0 = x1
        x1 = (x0 + number / x0) * 0.5
        
        # Recalculate the relative error
        error = (x1 - x0) / x0
        if error < 0:
            error = -error
    
    # Format and print the result
    result_message = "The square root of %s is approximately %s" % (number, x1)
    print(result_message)

def babylonian_square_root_main():
    # Example usage
    babylonian_square_root(5, 0.00001)

# Call the main function
babylonian_square_root_main()
