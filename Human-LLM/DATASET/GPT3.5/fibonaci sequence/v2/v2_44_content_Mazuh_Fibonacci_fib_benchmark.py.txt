import csv
from timeit import timeit

# Open the CSV file for writing the benchmark results
with open('./fibonacci_benchmark.csv', 'w', newline='') as reportfile:
    writer = csv.writer(reportfile)

    # Write the header row with column names
    writer.writerow(['n_arg', 'recursive_seconds', 'iterative_seconds', 'explicit_seconds'])

    # Loop through Fibonacci numbers from 1 to 45 and benchmark the functions
    for n in range(1, 46):
        print(f'Calculating Fibonacci({n})...', end='')

        # Benchmark recursive Fibonacci function if n is less than 15, otherwise mark it as 'NA'
        if n < 15:
            recursive_time = timeit('fib.f_recursive({})'.format(n), setup='import fib')
        else:
            recursive_time = 'NA'
            print(' [Recursive skipped]', end='')

        # Benchmark iterative Fibonacci function
        iterative_time = timeit('fib.f_iterative({})'.format(n), setup='import fib')

        # Benchmark explicit Fibonacci function
        explicit_time = timeit('fib.f_explicit({})'.format(n), setup='import fib')

        print(' Done')

        # Write the benchmark results to the CSV file
        writer.writerow([n, recursive_time, iterative_time, explicit_time])
