def generate_fibonacci_sequence(start, end):
    """
    Generates a Fibonacci sequence between the specified start and end values.

    Args:
    - start (int): Starting value for the Fibonacci sequence.
    - end (int): Ending value for the Fibonacci sequence.

    Returns:
    - list: Fibonacci sequence within the specified range.
    """
    fibonacci_sequence = [0, start]
    i = 2
    while fibonacci_sequence[i - 1] + fibonacci_sequence[i - 2] <= end:
        fibonacci_sequence.append(fibonacci_sequence[i - 1] + fibonacci_sequence[i - 2])
        i += 1
    return fibonacci_sequence

def find_minimum_integer(org_seq, target):
    """
    Finds the minimum integer from the original Fibonacci-like sequence that divides the target.

    Args:
    - org_seq (list): Original Fibonacci-like sequence.
    - target (int): Target integer to be divided by the Fibonacci-like sequence members.

    Returns:
    - int: Minimum integer found that divides the target.
    """
    min_start_int = None
    for member in org_seq:
        if member > target:
            break
        if member in (0, 1):
            continue
        if target % member == 0:
            if min_start_int is None:
                min_start_int = target // member
            else:
                if target // member <= min_start_int:
                    min_start_int = target // member
    return min_start_int if min_start_int is not None else target

    original_sequence = generate_fibonacci_sequence(1, input_integer)
    min_integer = find_minimum_integer(original_sequence, input_integer)
