# Function to calculate Fibonacci number recursively
def fibo(n: int) -> int:
    if n < 1:
        return 0
    elif n == 1:
        return 1
    else:
        return fibo(n - 1) + fibo(n - 2)

# Global variable to keep track of the current Fibonacci number index
n = 0

# Function to display available routes
def home():
    return (
        f"Available Routes:<br/>"
        f"/previous<br/>"
        f"/current<br/>"
        f"/next<br/>"
    )

# Function to return the current Fibonacci number
def current_fib():
    global n
    return str(fibo(n))

# Function to return the next Fibonacci number
def next_fib():
    global n
    n += 1
    return str(fibo(n))

# Function to return the previous Fibonacci number
def prev_fib():
    global n
    n -= 1
    return str(fibo(n))

# Function to handle user requests based on endpoint
def handle_request(endpoint):
    if endpoint == "/":
        return home()
    elif endpoint == "/current":
        return current_fib()
    elif endpoint == "/next":
        return next_fib()
    elif endpoint == "/previous":
        return prev_fib()
    else:
        return "Invalid endpoint"

# Main program execution
if __name__ == '__main__':
    # Continuously prompt the user for input and handle requests
    while True:
        user_input = input("Enter endpoint (current/next/previous): ")
        print(handle_request(user_input))
