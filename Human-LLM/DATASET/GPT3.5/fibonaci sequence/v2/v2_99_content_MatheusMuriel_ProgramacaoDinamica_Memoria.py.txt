class Brain:
    def __init__(self, capacity, cleaning_rate):
        self.memory = {}         # Dictionary to store memories
        self.usage_count = {}   # Dictionary to store the usage count of each memory
        self.capacity = capacity  # Maximum capacity of the brain
        self.cleaning_rate = cleaning_rate  # Cleaning rate as a fraction of capacity

    def new_memory(self, name, value):
        """Add a new memory to the brain."""
        name = str(name)  # Ensure name is a string
        value = int(value)  # Ensure value is an integer
        if len(self.memory) >= self.capacity:
            # If the memory capacity is exceeded, clean up the memory
            self.clean_memory(name)
        # Add the new memory to the brain
        self.memory[name] = value
        self.usage_count[name] = 1  # Initialize the usage count for the new memory

    def recall_memory(self, name):
        """Retrieve a memory from the brain."""
        if name in self.memory:
            # If the memory exists, update its usage count and return its value
            self.usage_count[name] += 1
            return self.memory[name]
        else:
            # If the memory does not exist, return -1
            return -1

    def clean_memory(self, current_iteration):
        """Clean up the least used memories from the brain."""
        print("Cleaning called at iteration", current_iteration)
        # Calculate the number of memories to be cleaned based on the cleaning rate
        num_to_clean = int(self.capacity * self.cleaning_rate)
        # Remove the least used memories until the desired cleaning rate is achieved
        for _ in range(num_to_clean):
            least_used_entry = min(self.usage_count.items(), key=lambda x: x[1])
            self.memory.pop(least_used_entry[0])
            self.usage_count.pop(least_used_entry[0])

# Example usage
brain = Brain(capacity=100, cleaning_rate=0.5)
brain.new_memory("A", 10)
brain.new_memory("B", 20)
print(brain.recall_memory("A"))
print(brain.recall_memory("C"))
