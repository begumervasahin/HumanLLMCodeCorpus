def classic_euclid_recursive(a, b):
    # Base case: if either number is 0, return the other number
    if b == 0:
        return a
    if a == 0:
        return b
    
    # Recursive case: subtract smaller from larger until they're equal
    if a > b:
        return classic_euclid_recursive(a - b, b)
    else:
        return classic_euclid_recursive(a, b - a)

def modern_euclid_recursive(a, b):
    # Base case: if b is 0, return a
    if b == 0:
        return a
    
    # Recursive case: call the function with b and the remainder of a divided by b      
    return modern_euclid_recursive(b, a % b)

if __name__ == "__main__":
    while True:
        # Prompt the user to choose between classic and modern Euclid algorithms
        algorithm_choice = input("Choose Classic (c) or Modern (e) Euclid algorithm: ")
        
        while True:
            number_input = input("Enter 'exit' to choose another algorithm. Otherwise, enter a number: ")
            
            # Check if the input is numeric
            if number_input.isnumeric():
                a = int(number_input)
                b = int(input("Enter the second number: "))
                
                # Calculate GCD using the chosen algorithm
                if algorithm_choice.lower() == "c":
                    gcd = classic_euclid_recursive(a, b)
                else:
                    gcd = modern_euclid_recursive(a, b)
                
                print(f"The GCD of {a} and {b} is: {gcd}")
            
            else:
                # Exit the inner loop if input is not numeric
                break
