def generate_doubling_sequence(start, base):
    """Generate a doubling sequence starting from the given number and base."""
    while True:
        yield convert_decimal_to_base(start, base)
        start *= 2

def convert_decimal_to_base(number, base, converted=''):
    """Convert a decimal number to the specified base."""
    if number:
        if 2 <= base <= 36:
            return convert_decimal_to_base(int(number / base), base, CHARACTER_MAP[number % base] + converted)
        else:
            return convert_decimal_to_base(int(number / base), base, str(number % base) + ('.' + converted if converted else ''))
    return converted

def convert_base_to_decimal(number, base, converted=0):
    """Convert a number from the specified base to decimal."""
    number = str(number)
    if number:
        if 2 <= base <= 36:
            return convert_base_to_decimal(number[:-1], base, converted + CHARACTER_MAP[number[-1]] * (base ** len(number[:-1])))
        else:
            return convert_base_to_decimal('.'.join(number.split('.')[:-1]), base, converted + int(number.split('.')[-1]) * (base ** len(number.split('.')[:-1])))

    return converted

def calculate_digital_root(number, base):
    """Calculate the digital root of a number in the specified base."""
    decimal_value = convert_base_to_decimal(number, base)
    while decimal_value >= base:
        decimal_value = sum([CHARACTER_MAP[elem] for elem in str(decimal_value)])
    return decimal_value
