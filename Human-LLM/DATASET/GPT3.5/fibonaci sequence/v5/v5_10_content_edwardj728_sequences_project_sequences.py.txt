def is_arithmetic(sequence):
    first_difference = sequence[1] - sequence[0]
    for i in range(2, len(sequence)):
        if sequence[i] != sequence[i - 1] + first_difference:
            return False
    return True

def is_geometric(sequence):
    for i in range(1, len(sequence) - 1):
        if sequence[i + 1] != sequence[i] * 2:
            return False
    return True

def is_quadratic(sequence):
    if len(sequence) < 3:
        return False
    first_difference = sequence[1] - sequence[0]
    second_difference = sequence[2] - sequence[1]
    if first_difference == second_difference:
        return True
    return False

def is_cubic(sequence):
    if len(sequence) < 4:
        return False
    first_difference = sequence[1] - sequence[0]
    second_difference = sequence[2] - sequence[1]
    third_difference = sequence[3] - sequence[2]
    if third_difference - second_difference == second_difference - first_difference:
        return True
    return False

def is_fibonacci(sequence):
    if len(sequence) < 4:
        return False
    for i in range(2, len(sequence) - 1):
        if sequence[i] != sequence[i - 1] + sequence[i - 2]:
            return False
    return True

# Take user input
user_input = input("Type in 5 integers of any sequence separated by commas. Example: 1,2,3,4,5: ")
sequence = list(map(int, user_input.split(",")))

# Check sequences
if is_arithmetic(sequence):
    print("Arithmetic Sequence")
if is_geometric(sequence):
    print("Geometric Sequence")
if is_quadratic(sequence):
    print("Quadratic Sequence")
if is_cubic(sequence):
    print("Cubic Sequence")
if is_fibonacci(sequence):
    print("Fibonacci Sequence")
