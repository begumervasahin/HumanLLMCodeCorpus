import redis
import json

class Fibonacci:
    def __init__(self, host, db):
        self.redis = redis.StrictRedis(host=host, db=db)
        # Initialize Fibonacci sequence in Redis if not already present
        if self.get(1) is None:
            self.redis.set(1, 1)
            self.redis.set(2, 1)
            self.save()

    def find_nearest_left(self, n):
        # Find the nearest left Fibonacci number
        if self.get(n) is None:
            return self.find_nearest_left(n - 1)
        return n

    def get(self, k):
        # Get Fibonacci number at index k from Redis
        return self.redis.get(k)

    def set(self, k):
        # Set Fibonacci number at index k based on previous two Fibonacci numbers
        n1 = self.get(k - 1)
        n2 = self.get(k - 2)
        self.redis.set(k, int(n1) + int(n2))

    def save(self):
        # Save changes to Redis
        self.redis.bgsave()

    def calculate(self, n1, n2):
        # Calculate Fibonacci numbers for the range [n1, n2)
        for n in range(n1, n2):
            n_v = self.get(n)
            if n_v is None:
                self.set(n)

    def numbers(self, n1, n2):
        # Start calculating Fibonacci numbers from the nearest left Fibonacci number
        l_v = self.get(n1)
        if l_v is None:
            l_n = self.find_nearest_left(n1)
            self.numbers(l_n, n2)
        self.calculate(n1 + 1, n2)

    def control(self, n1, n2):
        # Control the Fibonacci sequence within the range [n1, n2)
        return ['<div>[{}] == {}</div>'.format(k, self.get(k)) for k in range(n1, n2)]  

def main():
    # Load configuration from JSON file
    with open('config.json') as file:
        conf = json.load(file)
    # Initialize Fibonacci object
    F = Fibonacci(conf['redis_host'], conf['redis_db'])
    # Control Fibonacci sequence within the range [1, 50)
    F.control(1, 50)
    # Calculate Fibonacci numbers for specific ranges
    F.numbers(5, 7)
    F.numbers(30, 40)
    F.numbers(25, 35)
    F.numbers(49, 50)
    # Control Fibonacci sequence within the range [1, 50) again
    F.control(1, 50)
    # Control Fibonacci sequence within the range [9, 13)
    print(F.control(9, 13))

if __name__ == '__main__':
    main()
