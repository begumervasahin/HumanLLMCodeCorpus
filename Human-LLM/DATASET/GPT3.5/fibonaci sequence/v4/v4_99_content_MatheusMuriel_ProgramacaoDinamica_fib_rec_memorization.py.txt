from Memoria import Cerebro

class Memorization:
    def __init__(self):
        # Set up memory with a limit and cleaning threshold
        self.memory = Cerebro(10000000, 0.7)

    def calc_fibonacci(self, n):
        # Base cases for Fibonacci sequence
        if n == 0:
            return 0
        if n == 1:
            return 1

        # Convert n-1 and n-2 to strings for memoization
        str_n1 = str(n - 1)
        str_n2 = str(n - 2)

        # Retrieve values from memory if available
        fib_n1 = self.memory.get_lembranca(str_n1)
        fib_n2 = self.memory.get_lembranca(str_n2)

        # Calculate Fibonacci numbers recursively if not in memory
        if fib_n1 == -1:
            fib_n1 = self.calc_fibonacci(n - 1)
            self.memory.nova_memoria(str_n1, fib_n1)

        if fib_n2 == -1:
            fib_n2 = self.calc_fibonacci(n - 2)
            self.memory.nova_memoria(str_n2, fib_n2)

        # Return the sum of Fibonacci numbers
        return fib_n1 + fib_n2

    def get_fibonacci_function(self):
        # Return a lambda function for calculating Fibonacci numbers
        return lambda n: self.calc_fibonacci(n)

    def __str__(self):
        return "Recursive Fibonacci with Memoization"

if __name__ == "__main__":
    # Create an instance of the Memorization class
    memorization = Memorization()

    # Input the number for which Fibonacci number is to be calculated
    n = int(input("Enter a number: "))

    # Calculate and print the Fibonacci number
    result = memorization.calc_fibonacci(n)
    print("Result:", result)
