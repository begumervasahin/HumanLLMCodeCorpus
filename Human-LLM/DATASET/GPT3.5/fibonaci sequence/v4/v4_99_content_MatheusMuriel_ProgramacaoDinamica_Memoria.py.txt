class Cerebro:
    def __init__(self, limite_memoria, taxa_limpeza):
        self.memoria = {}             # Dicionário para armazenar as memórias
        self.usos = {}                # Dicionário para armazenar a contagem de uso de cada memória
        self.limite_memoria = limite_memoria  # Capacidade máxima do cérebro
        self.taxa_limpeza = taxa_limpeza      # Taxa de limpeza como uma fração da capacidade

    def nova_memoria(self, nome, valor):
        """Adiciona uma nova memória ao cérebro."""
        nome = str(nome)  # Garante que o nome seja uma string
        valor = int(valor)  # Garante que o valor seja um número inteiro

        # Se a capacidade de memória for excedida, limpa a memória
        if len(self.memoria) >= self.limite_memoria:
            self.limpar_memoria()

        # Adiciona a nova memória ao cérebro
        self.memoria[nome] = valor
        self.usos[nome] = 1  # Inicializa a contagem de uso para a nova memória

    def get_lembranca(self, nome):
        """Recupera uma memória do cérebro."""
        if nome in self.memoria:
            # Se a memória existir, atualiza sua contagem de uso e retorna seu valor
            self.usos[nome] += 1
            return self.memoria[nome]
        else:
            # Se a memória não existir, retorna -1
            return -1

    def limpar_memoria(self):
        """Limpa as memórias menos utilizadas do cérebro."""
        print("Chamada de limpeza")
        # Remove as memórias menos utilizadas até atingir a taxa de limpeza desejada
        while len(self.memoria) > self.limite_memoria * self.taxa_limpeza:
            menor_registro = min(self.usos.items(), key=lambda x: x[1])
            self.memoria.pop(menor_registro[0])
            self.usos.pop(menor_registro[0])

# Exemplo de uso
cerebro = Cerebro(limite_memoria=100, taxa_limpeza=0.5)
cerebro.nova_memoria("A", 10)
cerebro.nova_memoria("B", 20)
print(cerebro.get_lembranca("A"))
print(cerebro.get_lembranca("C"))
