import sequences

def test_return_value_matches_expected_result():
    """
    Test if the return value of the Fibonacci function matches the expected result.     
    """
    expected_list = [1, 1, 2, 3, 5]
    ret_list = sequences.fibonacci(5)

    assert isinstance(ret_list, list), "Return value should be a list"
    assert len(ret_list) == len(expected_list), "Returned list length doesn't match the expected length"
    assert expected_list == ret_list, "Returned list does not match the expected list"
    print("Test 1 success")

def test_last_number_of_fibonacci_list():
    """
    Verify the last number of the Fibonacci list for a certain valid number.
    """
    ret_list = sequences.fibonacci(8)
    print("Last number of Fibonacci list:", ret_list[-1])
    assert ret_list[-1] == 21, "Last number of Fibonacci list is incorrect"
    print("Test 2 success")

def test_input_validation():
    """
    Test if the Fibonacci function can validate the input.
    """
    invalid_values = [-1, 2.2, 0]
    for value in invalid_values:
        assert not sequences.fibonacci(value), f"Input validation failed for value: {value}"
    print("Input validation success")

if __name__ == "__main__":
    print("Test 1 start")
    test_return_value_matches_expected_result()
    print("Test 2 start")
    test_last_number_of_fibonacci_list()
    print("Test 3 start")
    test_input_validation()
