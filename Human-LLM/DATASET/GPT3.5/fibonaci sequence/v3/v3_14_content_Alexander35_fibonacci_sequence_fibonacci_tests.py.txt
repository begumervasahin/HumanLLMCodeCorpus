import unittest
from fibonacci import Fibonacci  # Assuming Fibonacci class is defined in fibonacci.py  
import json

class TestFibonacciIntegration(unittest.TestCase):
    def setUp(self):
        """Load configuration and initialize Fibonacci object."""
        # Load configuration from config.json
        with open('config.json') as file:
            config_data = json.load(file)
        # Initialize Fibonacci object using configuration data
        self.fibonacci_instance = Fibonacci(config_data['redis_host'], config_data['redis_db'])

    def test_generate_numbers(self):
        """Test generating Fibonacci numbers in a specified range."""
        # Generate Fibonacci numbers in the specified range
        self.fibonacci_instance.numbers(9, 13)
        # Get the generated sequence
        generated_sequence = self.fibonacci_instance.control(9, 13)
        # Expected sequence based on the specified range
        expected_sequence = [
            "<div>[9] == b'34'</div>",
            "<div>[10] == b'55'</div>",
            "<div>[11] == b'89'</div>",
            "<div>[12] == b'144'</div>"
        ]
        # Assert that the generated sequence matches the expected sequence
        self.assertEqual(generated_sequence, expected_sequence)

    def test_starter_sequence(self):
        """Test the initial Fibonacci sequence for a specified range."""
        # Get the initial sequence for the specified range
        initial_sequence = self.fibonacci_instance.control(1, 3)
        # Expected initial sequence for the specified range
        expected_initial_sequence = [
            "<div>[1] == b'1'</div>",
            "<div>[2] == b'1'</div>"
        ]
        # Assert that the initial sequence matches the expected initial sequence        
        self.assertEqual(initial_sequence, expected_initial_sequence)

if __name__ == '__main__':
    # Run the unit tests
    unittest.main()
