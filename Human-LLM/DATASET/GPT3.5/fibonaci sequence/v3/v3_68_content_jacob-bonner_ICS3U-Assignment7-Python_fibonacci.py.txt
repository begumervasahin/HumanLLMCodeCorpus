def generate_fibonacci_list(limit):
    """
    Generate Fibonacci numbers up to the specified limit.

    Args:
        limit (int): The upper limit for the Fibonacci sequence.

    Returns:
        list: A list containing Fibonacci numbers.
    """
    fib_sequence = [1, 1]  # Starting Fibonacci sequence
    for _ in range(limit - 2):  # Loop to generate Fibonacci sequence up to the limit
        next_fib = fib_sequence[-1] + fib_sequence[-2]  # Calculate the next Fibonacci number
        fib_sequence.append(next_fib)  # Append the next Fibonacci number to the sequence
    return fib_sequence

def print_fibonacci_sequence(fib_sequence):
    """
    Print the Fibonacci sequence.

    Args:
        fib_sequence (list): A list containing Fibonacci numbers.
    """
    for i, fib_num in enumerate(fib_sequence):
        if i == 0:
            print(f"{fib_num} + 0 = {fib_num}")
        elif i == 1:
            print(f"0 + {fib_num} = {fib_num}")
        else:
            prev_fib = fib_sequence[i - 1]
            prev_prev_fib = fib_sequence[i - 2]
            print(f"{prev_prev_fib} + {prev_fib} = {fib_num}")

def main():
    """
    Main function to generate and print the Fibonacci sequence.
    """
    limit = 100
    fibonacci_sequence = generate_fibonacci_list(limit)
    print_fibonacci_sequence(fibonacci_sequence)

if __name__ == "__main__":
    main()
