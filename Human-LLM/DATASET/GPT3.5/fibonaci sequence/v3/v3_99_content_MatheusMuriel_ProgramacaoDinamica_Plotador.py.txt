import matplotlib.pyplot as plt
import numpy as np

def plot_simple(x_values, y_values, label):
    """
    Plot a simple line graph.

    Args:
        x_values (list): List of x-axis values.
        y_values (list): List of y-axis values.
        label (str): Label for the graph.
    """
    fig, ax = plt.subplots()
    ax.plot(x_values, y_values)

    # Set labels and title
    ax.set(xlabel='Size N', ylabel='Execution Time', title=label)

    # Enable grid
    ax.grid()

    # Show the plot
    plt.show()

def plot_double(start, end, y1_values, label1, y2_values, label2):
    """
    Plot two lines on the same graph.

    Args:
        start (int): Starting value for x-axis.
        end (int): Ending value for x-axis.
        y1_values (list): List of y-axis values for the first line.
        label1 (str): Label for the first line.
        y2_values (list): List of y-axis values for the second line.
        label2 (str): Label for the second line.
    """
    # Create a new figure and axis
    fig, ax = plt.subplots()

    # Plot the data
    ax.plot(range(start, end + 1), y1_values, label=label1)
    ax.plot(range(start, end + 1), y2_values, label=label2)

    # Set labels and title
    ax.set(xlabel='Size N', ylabel='Execution Time', title=f"{label1} vs {label2}")     

    # Add legend
    plt.legend()

    # Show the plot
    plt.show()

# Example data
x_values = [1, 2, 3, 4, 5]
y_values = [10, 15, 20, 25, 30]
y2_values = [8, 13, 18, 23, 28]

# Example usage
plot_simple(x_values, y_values, "Simple Plot")
plot_double(0, 5, y_values, "First Line", y2_values, "Second Line")
