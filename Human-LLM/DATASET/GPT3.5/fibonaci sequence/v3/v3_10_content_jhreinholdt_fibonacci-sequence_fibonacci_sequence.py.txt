def fibonacci_number(position):
    """Calculate the Fibonacci number at a given position."""
    if position == 0:
        return 0
    elif position == 1:
        return 1
    else:
        return fibonacci_number(position - 2) + fibonacci_number(position - 1)

def fibonacci_sequence(num_elements):
    """Generate and print the Fibonacci sequence up to a given number of elements."""
    if num_elements <= 0:
        print("Please enter a positive integer.")
        return
    
    print("The Fibonacci Sequence with", num_elements, "elements is:")
    for position in range(num_elements):
        print(fibonacci_number(position), end=" ")

def get_positive_integer(prompt):
    """Prompt the user to enter a positive integer."""
    while True:
        try:
            num = int(input(prompt))
            if num <= 0:
                print("Please enter a positive integer.")
            else:
                return num
        except ValueError:
            print("Please enter a valid positive integer.")

def main():
    """Interact with the user and control the program flow."""
    num_elements = get_positive_integer('\nEnter a positive number to generate the Fibonacci Sequence with that amount of elements: ')
    fibonacci_sequence(num_elements)

if __name__ == '__main__':
    main()
