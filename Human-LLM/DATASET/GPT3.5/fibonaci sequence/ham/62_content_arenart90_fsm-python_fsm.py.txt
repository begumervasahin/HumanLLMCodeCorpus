import time
from enum import Enum, auto
class States(Enum):
    ini = auto()
    fib = auto()
    off = auto()
def ini():
    global a
    global b
    a = b = 1
    print(a)
    print(b)
    return States.fib
def fib():
    global a
    global b
    c = a + b
    print(c)
    a = b
    b = c
    time.sleep(0.1)
    if a + b < 620:
        return States.fib
    else:
        return States.off
def off():
    pass
def fsm(st):
    switcher = {
        States.off: off,
        States.ini: ini,
        States.fib: fib
        }
    func = switcher.get(st)
    return func()
def fsm_loop(st):
    while 1:
        if st == States.off:
            fsm(st)
            break
        else:
            st = fsm(st)
fsm_loop(States.ini)
