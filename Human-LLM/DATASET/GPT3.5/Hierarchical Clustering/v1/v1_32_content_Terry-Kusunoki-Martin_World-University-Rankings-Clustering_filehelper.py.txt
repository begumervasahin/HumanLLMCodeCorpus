def process_times_data():
    # Open files for reading and writing
    with open("timesData.csv", 'r') as f, open("universities.points", 'w') as pts, \
         open("universities.labels", 'w') as lbls, open("metadata.txt", 'w') as meta:
        
        ct = 0
        max_num_students = 0
        max_stud_staff_ratio = 0.0
        
        # Process each line in the CSV file
        for line in f.readlines():
            csplit = line.rstrip().split(',')
            if csplit[0] == "world_rank":
                continue
            else:
                new_line = []
                skip_until = 0
                
                # Handle quotes in the CSV fields
                for i in range(len(csplit)):
                    if i <= skip_until:
                        continue
                    if '"' in csplit[i]:
                        new_elt = csplit[i]
                        j = i + 1
                        while '"' not in csplit[j]:
                            j += 1
                        new_elt = "".join(csplit[i:j + 1]).strip('"')
                        new_line.append(new_elt)
                        skip_until = j
                    else:
                        new_line.append(csplit[i])
                
                # Extract relevant data and update max values
                try:
                    num_students = int(new_line[-5])
                    student_staff_ratio = float(new_line[-4])
                    
                    if num_students > max_num_students:
                        max_num_students = num_students
                    if student_staff_ratio > max_stud_staff_ratio:
                        max_stud_staff_ratio = student_staff_ratio
                except:
                    continue
        
        # Write metadata to the metadata file
        meta.write("max_num_students: %d\n" % (max_num_students))
        meta.write("max_stud_staff_ratio: %f\n" % (max_stud_staff_ratio))
        
        # Open timesData.csv again for processing data
        f.seek(0)
        
        # Process each line in the CSV file again
        for line in f.readlines():
            csplit = line.rstrip().split(',')
            if csplit[0] == "world_rank":
                continue
            else:
                new_line = []
                skip_until = 0
                
                # Handle quotes in the CSV fields
                for i in range(len(csplit)):
                    if i <= skip_until:
                        continue
                    if '"' in csplit[i]:
                        new_elt = csplit[i]
                        j = i + 1
                        while '"' not in csplit[j]:
                            j += 1
                        new_elt = "".join(csplit[i:j + 1]).strip('"')
                        new_line.append(new_elt)
                        skip_until = j
                    else:
                        new_line.append(csplit[i])
                
                # Extract relevant data and write to points and labels files
                try:
                    num_students = float(new_line[-5]) / max_num_students
                    student_staff_ratio = 1.0 - (float(new_line[-4]) / max_stud_staff_ratio)
                    name = new_line[0]
                    teaching_score = float(new_line[2]) / 100
                    tot_score = new_line[-6]
                    lbls.write(name + "\n")
                    try:
                        tot_score = float(tot_score) / 100
                    except:
                        tot_score = teaching_score
                    pts.write(str(tot_score) + " " + str(num_students) + " " + str(student_staff_ratio) + "\n")
                except:
                    continue

if __name__ == "__main__":
    process_times_data()
