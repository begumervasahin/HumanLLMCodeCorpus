import gc
import warnings
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split, cross_val_score
from sklearn.preprocessing import LabelEncoder
from sklearn.tree import export_graphviz
from sklearn.ensemble import RandomForestClassifier
from sklearn.neighbors import KNeighborsClassifier
from sklearn.naive_bayes import GaussianNB
from sklearn.neural_network import MLPClassifier
from sklearn.metrics import f1_score, accuracy_score

# Suppress warnings
warnings.filterwarnings("ignore")

# Load dataset
df = pd.read_csv('10k.anon.csv')

# Shuffle dataset
df = df.sample(frac=1).reset_index(drop=True)

# Display sample data and dataset information
print("Sample Data:\n", df.head(10))
print("\nNumber of classes:", len(df['app_name'].unique()))
print("Number of samples:", len(df))

# Encode class labels
le = LabelEncoder()
df['app_name'] = le.fit_transform(df['app_name'])
print("Encoded classes:", list(le.classes_))

# Split dataset into train and test sets
X_train, X_test, y_train, y_test = train_test_split(df.iloc[:, :-1], df['app_name'], test_size=0.4)

# Plot class frequency
plt.figure(figsize=(8, 6))
plt.hist([y_train, y_test], bins=len(df['app_name'].unique()), label=['Train', 'Test'])
plt.title('Class Frequency')
plt.xlabel('Class (encoded)')
plt.ylabel('Frequency')
plt.legend()
plt.show()

def evaluate_classifier(classifier, X_train, y_train, X_test, y_test):
    """
    Evaluate classifier performance and print results.
    """
    classifier.fit(X_train, y_train)
    y_pred = classifier.predict(X_test)
    f_score = f1_score(y_test, y_pred, average='weighted')
    accuracy = accuracy_score(y_test, y_pred)
    print("F-score:", f_score)
    print("Accuracy:", accuracy)
    del classifier, y_pred
    gc.collect()
    return f_score, accuracy

# Evaluate classifiers
classifiers = {
    "KNN": KNeighborsClassifier(n_neighbors=5),
    "Naive Bayes (Gaussian)": GaussianNB(),
    "Multilayer Perceptron": MLPClassifier(hidden_layer_sizes=(100,), max_iter=200),
    "Random Forest": RandomForestClassifier(n_estimators=100, class_weight='balanced')
}

results = {}
for name, classifier in classifiers.items():
    print("\n***", name, "***")
    f_score, accuracy = evaluate_classifier(classifier, X_train, y_train, X_test, y_test)
    results[name] = {"F-score": f_score, "Accuracy": accuracy}

# Cross-validation
classifier = RandomForestClassifier(n_estimators=100, class_weight='balanced')
scores = cross_val_score(classifier, df.iloc[:, :-1], df['app_name'], cv=5, scoring='accuracy')
print("\nCross-validated accuracy:", scores)

# Plot actual vs predicted values
_, ax = plt.subplots(figsize=(10, 6))
ax.scatter(range(len(y_test)), y_test, c='blue', label='Actual', alpha=0.3)
ax.scatter(range(len(y_test)), y_pred, c='red', label='Predicted', alpha=0.3)
plt.title('Actual and Predicted Values')
plt.xlabel('Samples')
plt.ylabel('Classes')
plt.legend()
plt.show()
