import numpy as np
import scipy.cluster.hierarchy as sch
import matplotlib.pyplot as plt
import time

def find_minimum(x, y):
    """Find the minimum of two numbers."""
    return min(x, y)

def find_maximum(x, y):
    """Find the maximum of two numbers."""
    return max(x, y)

def agglomerative_clustering(distance_matrix):
    """Perform agglomerative clustering and return the linkage matrix."""
    size = len(distance_matrix)
    linkage_matrix = np.zeros(shape=(size - 1, 4))
    new_clusters = {i: -1 for i in range(size)}
    clusters_dict = {i: [i] for i in range(size)}

    start_time = time.time()
    for k in range(size - 1):
        distance_matrix = merge_clusters(distance_matrix, size, k, linkage_matrix, new_clusters, clusters_dict)

    print("Clustering done in", time.time() - start_time, "seconds")

    return linkage_matrix

def merge_clusters(distance_matrix, size, k, linkage_matrix, new_clusters, clusters_dict):
    """Merge clusters and update the distance matrix."""
    min_val = float('inf')
    m, n = -1, -1

    for i in range(size - 1):
        for j in range(i + 1, size):
            if distance_matrix[i][j] < min_val:
                min_val = distance_matrix[i][j]
                m, n = i, j

    if m < n:
        clusters_dict[m].extend(clusters_dict[n])
        del clusters_dict[n]

    if new_clusters[m] == -1 and new_clusters[n] == -1:
        linkage_matrix[k][0], linkage_matrix[k][1] = m, n
        x = find_minimum(m, n)
        new_clusters[x] = size + k
        linkage_matrix[k][3] = 2
    else:
        if new_clusters[m] != -1:
            m = new_clusters[m]
        if new_clusters[n] != -1:
            n = new_clusters[n]

        linkage_matrix[k][0], linkage_matrix[k][1] = m, n
        new_clusters[find_maximum(m, n)] = size + k
        linkage_matrix[k][3] = linkage_matrix[find_maximum(m, n) - size][3] + 1

    linkage_matrix[k][2] = min_val

    for j in range(size):
        if j != n:
            distance_matrix[j][m] = find_minimum(distance_matrix[j][m], distance_matrix[j][n])
        distance_matrix[m][j] = distance_matrix[j][m]
        distance_matrix[j][n] = float('inf')
        distance_matrix[n][j] = float('inf')

    return distance_matrix

def plot_dendrogram(linkage_matrix, labels):
    """Plot the dendrogram."""
    plt.figure(figsize=(25, 25))
    plt.title('Hierarchical Clustering Dendrogram (Agglomerative)')
    plt.xlabel('Sequence No.')
    plt.ylabel('Distance')
    sch.dendrogram(linkage_matrix, labels=labels, show_leaf_counts=True, p=25, truncate_mode='lastp')
    plt.show()

if __name__ == "__main__":
    distance_matrix = np.load('distance_matrix.npy')
    linkage_matrix = agglomerative_clustering(distance_matrix)
    labels = [i for i in range(len(distance_matrix))]
    plot_dendrogram(linkage_matrix, labels)
