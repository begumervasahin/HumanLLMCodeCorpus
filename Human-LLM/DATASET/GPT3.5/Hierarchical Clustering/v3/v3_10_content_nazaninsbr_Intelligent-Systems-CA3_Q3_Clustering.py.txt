import sys
import math
import random
import copy
import os
import numpy as np
import matplotlib.pyplot as plt
from helper import read_mat_file

ITERATION = 200
KNN_DATA_FILE = './HW'
KNN_LABELS_FILE = './HW'

def get_data():
    data = read_mat_file(KNN_DATA_FILE, 'data2')
    labels = read_mat_file(KNN_LABELS_FILE, 'labels')
    return data, labels

def euclidean_distance(instance, center):
    squared_distance = sum((x - y) ** 2 for x, y in zip(instance, center))
    return math.sqrt(squared_distance)

def calculate_cluster_mean(cluster):
    if len(cluster) == 0:
        return [0] * len(cluster[0][1])
    num_features = len(cluster[0][1])
    cluster_mean = [sum(instance[1][i] for instance in cluster) / len(cluster) for i in range(num_features)]
    return cluster_mean

def k_means_clustering(data, k_values):
    resulting_centers = {}
    resulting_clusters = {}

    for k in k_values:
        data_copy = copy.deepcopy(data)
        centers = {i: data_copy.pop(random.randint(0, len(data_copy) - 1)) for i in range(k)}
        clusters = {i: [] for i in range(k)}

        for idx, instance in enumerate(data_copy):
            min_dist = sys.maxsize
            closest_center = None
            for center_idx, center_instance in centers.items():
                dist = euclidean_distance(instance, center_instance)
                if dist < min_dist:
                    min_dist = dist
                    closest_center = center_idx
            clusters[closest_center].append([idx, instance])

        for _ in range(ITERATION):
            for center_idx, cluster_instances in clusters.items():
                cluster_mean = calculate_cluster_mean(cluster_instances)
                centers[center_idx] = cluster_mean

            for center_idx, cluster_instances in clusters.items():
                idx = 0
                while idx < len(cluster_instances):
                    min_dist = sys.maxsize
                    closest_center = None
                    for center_id, center_instance in centers.items():
                        dist = euclidean_distance(cluster_instances[idx][1], center_instance)
                        if dist < min_dist:
                            min_dist = dist
                            closest_center = center_id
                    clusters[closest_center].append(cluster_instances[idx])
                    del clusters[center_idx][idx]
                    idx += 1

            cost_func = sum(euclidean_distance(instance[1], centers[center_idx]) ** 2   
                            for center_idx, cluster_instances in clusters.items()       
                            for instance in cluster_instances) / len(data)

            plt.scatter(_, cost_func, color='black')

        print('Cluster Centers:')
        print(centers)
        resulting_centers[k] = centers
        resulting_clusters[k] = clusters

        inner_dist = sum(euclidean_distance(val[1], centers[center_id])
                         for center_id, cluster_instances in centers.items()
                         for val in cluster_instances) / len(data)

        outer_dist = sum(euclidean_distance(val[1], centers[center_id2])
                         for center_id, cluster_instances in centers.items()
                         for val in cluster_instances
                         for center_id2 in centers.keys() if center_id2 != center_id) / len(data)

        print("Inner dist:", inner_dist)
        print("Outer dist:", outer_dist)

        for cluster_num, cluster_instances in clusters.items():
            file_name = f'./Clusters/{k}_{cluster_num}_Kcluster.txt'
            try:
                os.remove(file_name)
            except OSError:
                pass
            with open(file_name, 'a') as file:
                for item in cluster_instances:
                    file.write(str(item))
                    file.write('\n')
        plt.show()

    return resulting_centers, resulting_clusters

def main():
    data, labels = get_data()
    my_clusters = k_means_clustering(data, [5])
    return my_clusters[5]

if __name__ == "__main__":
    main()
