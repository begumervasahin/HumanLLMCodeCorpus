import pandas as pd

# Load prediction data from the CSV file
prediction_data = pd.read_csv("y_prediction_itzy.csv").values
predictions = list(prediction_data)

# Initialize lists to store start and end times of segments
cut_start_data = []
cut_end_data = []

# Iterate through the prediction data
idx = 0
data_length = len(predictions) - 1
while idx != data_length:
    count = 0
    most_dyn_count = 0

    # Check if the prediction is 0.5 or 1
    if predictions[idx] == 0.5 or predictions[idx] == 1:
        start_count = idx

        # Count consecutive 0.5 or 1 predictions
        while predictions[idx] == 0.5 or predictions[idx] == 1:
            idx += 1
            count += 1

            # Count the number of 1 predictions
            if predictions[idx] == 1:
                most_dyn_count += 1

        last_count = idx

    # If the segment is longer than 3 and contains at least one 1 prediction
    if count > 3 and most_dyn_count > 0:
        print("SUCCESS")
        cut_start_data.append(start_count)
        cut_end_data.append(last_count)

    idx += 1

# Create DataFrames for start and end times
cut_start_data = pd.DataFrame(cut_start_data, columns=["Start Time"])
cut_end_data = pd.DataFrame(cut_end_data, columns=["End Time"])

# Concatenate start and end time DataFrames
time_data = pd.concat([cut_start_data, cut_end_data], axis=1)

# Write time data to a new CSV file
time_data.to_csv("timedata_itzy.csv", mode='w', index=False)
