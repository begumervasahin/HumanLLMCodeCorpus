import time

start_time = time.time()
count = 0

def merge_sort(arr):
    global count
    if len(arr) > 1:
        mid = len(arr) // 2
        L = arr[:mid]
        R = arr[mid:]
        merge_sort(L)
        merge_sort(R)
        i = j = k = 0
        while i < len(L) and j < len(R):
            count += 1
            if L[i] < R[j]:
                arr[k] = L[i]
                i += 1
            else:
                arr[k] = R[j]
                j += 1
            k += 1
        while i < len(L):
            count += 1
            arr[k] = L[i]
            i += 1
            k += 1
        while j < len(R):
            count += 1
            arr[k] = R[j]
            j += 1
            k += 1

def print_list(arr):
    for i in range(len(arr)):
        print(arr[i], end=" ")
    print()

if __name__ == '__main__':
    with open("case", "r") as fp:
        arr = fp.read().split(" ")
        list2 = [int(x) for x in arr]

    merge_sort(list2)
    print("Sorted array is:")
    print_list(list2)
    print("Total comparisons:", count)

end_time = time.time()
print("Execution time:", end_time - start_time)
