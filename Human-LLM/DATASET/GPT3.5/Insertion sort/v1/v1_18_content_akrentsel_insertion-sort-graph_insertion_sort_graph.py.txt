import matplotlib.pyplot as plt
import matplotlib.animation as anim
import random

def plot_cont(gen):
    # Get initial values
    y_vals, red_index = next(gen)
    xmax = len(y_vals) ** 2  # Set maximum number of frames
    fig = plt.figure()
    ax = fig.add_subplot(1, 1, 1)

    def update(i):
        # Get next values
        y_vals, red_index = next(gen)
        red_bar = [0] * len(y_vals)
        red_bar[red_index] = y_vals[red_index]
        x = range(len(y_vals))
        ax.clear()
        # Plot bars
        ax.bar(x, y_vals, width=1)
        ax.bar(x, red_bar, width=1, color="red")
        print('Iteration:', i)

    # Create animation
    a = anim.FuncAnimation(fig, update, frames=xmax, repeat=False)
    plt.show()

def insertion_sort_step_generator(ls):
    # Check list length
    assert len(ls) > 1, "Length of list must be greater than 1 to sort"
    curr = 1
    # Swap function
    def swap(pos1, pos2):
        nonlocal ls
        ls[pos1], ls[pos2] = ls[pos2], ls[pos1]

    # Sorting algorithm
    while curr < len(ls):
        temp = curr
        while curr > 0 and ls[curr] < ls[curr - 1]:
            swap(curr, curr - 1)
            curr = curr - 1
            yield ls, curr  # Yield current state
        if curr != temp:
            curr = temp + 1
        else:
            curr = curr + 1
            yield ls, curr  # Yield current state
    yield ls, len(ls) - 1  # Final state

number_of_elements_in_list = 25
rand_ls = [random.randrange(1, 200) for _ in range(number_of_elements_in_list)]
plot_cont(insertion_sort_step_generator(rand_ls))
