import random

def quicksort(arr):
    # Get the length of the list
    n = len(arr)

    # Check if the list contains more than one element
    if n > 1:
        # Choose a random pivot index
        pivot_index = random.randint(0, n - 1)
        pivot = arr[pivot_index]

        # Print the pivot element and the list before partitioning
        print("Pivoting", pivot, "in", arr)

        # Initialize lists for elements less than, greater than, and equal to the pivot
        left = []
        right = []
        pivot_clones = [pivot]

        # Partition the list around the pivot
        for i in range(n):
            # Skip the pivot index as it's already included in pivot_clones
            if i == pivot_index:
                continue
            elif arr[i] < pivot:
                left.append(arr[i])
            elif arr[i] > pivot:
                right.append(arr[i])
            else:
                pivot_clones.append(arr[i])

        # Print the left and right sublists after partitioning
        print("Left becomes", left, "; right becomes", right)

        # Recursively sort the left and right sublists
        sorted_left = quicksort(left)
        sorted_right = quicksort(right)

        # Concatenate the sorted sublists and pivot_clones
        return sorted_left + pivot_clones + sorted_right

    # If the list contains only one element or is empty, return it
    return arr

# Example usage
l = [3, 6, 8, 10, 1, 2, 1]
print("Original list:", l)
print("Sorted list:", quicksort(l))
