def shell_sort(arr, verbose=0, desc=0):
    """
    Sorts a list of integers using the shell sort algorithm.

    Args:
        arr (list): The unsorted list of integers.
        verbose (int, optional): Controls verbosity level. Defaults to 0.
        desc (int, optional): Controls whether the output is sorted in descending order. Defaults to 0.

    Returns:
        list: The sorted list.
    """
    def insertion_sort(gap, start_index=0):
        """
        Performs insertion sort with a specified gap.

        Args:
            gap (int): The gap between elements to compare.
            start_index (int, optional): The starting index. Defaults to 0.
        """
        j = start_index + gap
        n = len(arr)
        if start_index < 0 or j >= n:
            return
        if verbose == 2:
            print("Sub:", start_index, "-", j, "::", arr)
        if arr[start_index] > arr[j]:
            arr[start_index], arr[j] = arr[j], arr[start_index]
            insertion_sort(gap, start_index - gap)
        insertion_sort(gap, start_index + 1)

    # Initialize the gap as half the length of the list
    gap = len(arr) // 2

    # Reduce the gap in each iteration until it becomes 0
    while gap:
        if verbose:
            print("Gap:", gap)
        insertion_sort(gap)
        gap //= 2

    # Reverse the list if descending order is requested
    if desc:
        arr = arr[::-1]

    return arr
