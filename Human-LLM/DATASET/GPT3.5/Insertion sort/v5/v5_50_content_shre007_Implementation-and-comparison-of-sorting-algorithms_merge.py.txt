def merge_sort(arr):
    """
    Sorts the given list using merge sort algorithm.

    Args:
    - arr: The list to be sorted.

    Returns:
    - None (The list is sorted in-place).
    """
    # Base case: If the length of the list is 1 or less, it is already sorted
    if len(arr) <= 1:
        return

    # Divide the list into two halves
    mid = len(arr) // 2
    left_list = arr[:mid]
    right_list = arr[mid:]

    # Recursively sort the two halves
    merge_sort(left_list)
    merge_sort(right_list)

    # Merge the sorted halves
    merge(left_list, right_list, arr)

def merge(left_list, right_list, arr):
    """
    Merges two sorted lists into one sorted list.

    Args:
    - left_list: The left sorted list.
    - right_list: The right sorted list.
    - arr: The list to store the merged result.

    Returns:
    - None (The merged result is stored in arr).
    """
    i = j = k = 0

    # Merge the two lists while preserving the sorted order
    while i < len(left_list) and j < len(right_list):
        if left_list[i] < right_list[j]:
            arr[k] = left_list[i]
            i += 1
        else:
            arr[k] = right_list[j]
            j += 1
        k += 1

    # Copy the remaining elements of left_list
    while i < len(left_list):
        arr[k] = left_list[i]
        i += 1
        k += 1

    # Copy the remaining elements of right_list
    while j < len(right_list):
        arr[k] = right_list[j]
        j += 1
        k += 1

# Input
number_of_elements = int(input("How many elements do you want in this list: "))
given_list = [int(input("Enter element: ")) for _ in range(number_of_elements)]

# Sorting
merge_sort(given_list)

# Output
print("Sorted list:", given_list)
