def insertion_sort(arr, verbose=0, desc=0):
    """
    Sorts a list of integers using the insertion sort algorithm.

    Args:
        arr (list): The unsorted list of integers.
        verbose (int, optional): Controls verbosity level. Defaults to 0.
        desc (int, optional): Controls whether the output is sorted in descending order. Defaults to 0.

    Returns:
        list: The sorted list.
    """
    for i in range(1, len(arr)):
        key = arr[i]
        # Find the position for key in the sorted portion of the list
        j = i - 1
        while j >= 0 and arr[j] > key:
            arr[j + 1] = arr[j]  # Shift elements to the right
            j -= 1
            if verbose == 2:
                print("Shifting:", j, "::", arr)
        arr[j + 1] = key  # Insert key at its correct position
        if verbose:
            print("Iteration", i, ":", arr)
    # Optionally reverse the list if descending order is requested
    if desc:
        arr = arr[::-1]
    return arr
