import timeit
import random

# Function to copy elements from one array to another
def copy_array(src, src_pos, dest, dest_pos, length):
    for i in range(length):
        dest[i + dest_pos] = src[i + src_pos]

# Function to perform merge sort
def merge_sort(array):
    """
    Sorts the given array using the merge sort algorithm.
    """
    if len(array) <= 1:
        return array

    mid = len(array) // 2
    left = array[:mid]
    right = array[mid:]

    merge_sort(left)
    merge_sort(right)
    merge(left, right, array)

    return array

# Function to merge two sorted arrays
def merge(left, right, array):
    """
    Merges two sorted arrays into one sorted array.
    """
    r_index = 0
    l_index = 0
    merge_index = 0

    while l_index < len(left) and r_index < len(right):
        if left[l_index] > right[r_index]:
            array[merge_index] = right[r_index]
            r_index += 1
        else:
            array[merge_index] = left[l_index]
            l_index += 1
        merge_index += 1

    copy_array(left, l_index, array, merge_index, len(left) - l_index)
    copy_array(right, r_index, array, merge_index, len(right) - r_index)

# Generate a list of random numbers
numbers = [7, 4, 8, 3, 1, 12, 9, 14]
for _ in range(5000):
    rand = random.randint(1, 99999)
    numbers.append(rand)

# Measure the time taken to perform merge sort
start_time = timeit.default_timer()
merge_sort(numbers)
elapsed_time = timeit.default_timer() - start_time

# Print the elapsed time and the sorted list
print("Merge Sort Time:", elapsed_time)
print("Sorted List:", numbers)
