from is_140422H import InsertionSort

def test_insertion_sort(input_list):
    """
    Test the Insertion Sort algorithm and print the results.

    Parameters:
    input_list (list): The list to be sorted and tested.
    """
    print('Unsorted list:', input_list)
    sorted_list = InsertionSort(input_list)
    print('Sorted list:', sorted_list)

    is_sorted = all(sorted_list[i] <= sorted_list[i + 1] for i in range(len(sorted_list) - 1))
    if is_sorted:
        print("The list is correctly sorted.")
    else:
        print('Error: The list is not sorted correctly.')

# Define test lists
list1 = [20, 12, 8, 5, 7, 10, 14]
list2 = [20, -7, 10, 14]
list3 = []
list4 = [0, 0, 0]
big_list = [list1, list2, list3, list4]

# Test the Insertion Sort algorithm on each list
for test_list in big_list:
    test_insertion_sort(test_list)
