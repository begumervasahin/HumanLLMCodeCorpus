def quicksort_Spivot(qPArr,start,end):
    size = end - start + 1
    stack = [0]*(size)
    top = -1
    top = top + 1
    stack[top] = start
    top = top + 1
    stack[top] = end
    while top >= 0:
        end = stack[top]
        top = top - 1
        start = stack[top]
        top = top - 1
        i = ( start-1 )
        pv = qPArr[end]
        for j in range(start,end,1):
            if   qPArr[j] <= pv:
                i = i+1
                qPArr[i],qPArr[j] = qPArr[j],qPArr[i]
        qPArr[i+1],qPArr[end] = qPArr[end],qPArr[i+1]
        pv=i+1
        if abs(start-pv)<abs(pv-end):
            if pv+1 < end:
                top = top + 1
                stack[top] = pv + 1
                top = top + 1
                stack[top] = end
            if pv-1 > start:
                top = top + 1
                stack[top] = start
                top = top + 1
                stack[top] = pv - 1
        else:
            if pv-1 > start:
                top = top + 1
                stack[top] = start
                top = top + 1
                stack[top] = pv - 1
            if pv+1 < end:
                top = top + 1
                stack[top] = pv + 1
                top = top + 1
                stack[top] = end
