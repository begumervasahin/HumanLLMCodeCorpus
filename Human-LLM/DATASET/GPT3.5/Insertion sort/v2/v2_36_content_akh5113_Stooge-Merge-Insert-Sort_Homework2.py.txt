# Function to perform Stooge Sort
def stooge_sort(arr, start, end):
    num_elms = end - start + 1
    if num_elms == 2:
        # Swap elements if necessary
        if arr[start] > arr[end]:
            arr[start], arr[end] = arr[end], arr[start]
    elif num_elms > 2:
        # Divide the array into three parts
        third = num_elms // 3
        # Recursively sort the first 2/3 elements
        stooge_sort(arr, start, end - third)
        # Recursively sort the last 2/3 elements
        stooge_sort(arr, start + third, end)
        # Recursively sort the first 2/3 elements again
        stooge_sort(arr, start, end - third)

# Read data from file
nums = []
with open("data.txt", "r") as file:
    for line in file:
        # Split the line and convert elements to integers
        var = line.split(" ")
        nums.append(list(map(int, var[1:])))

print("Stooge Sort")
array_size = len(nums)

# Print unsorted arrays
print("The unsorted arrays are: ")
for arr in nums:
    print(arr)

# Sort arrays using Stooge Sort
for arr in nums:
    stooge_sort(arr, 0, len(arr) - 1)

# Print sorted arrays
print("The sorted arrays are: ")
for arr in nums:
    print(arr)

# Write sorted arrays to file
with open("stooge.out", "w") as f_out:
    for arr in nums:
        # Write each array to a line in the file
        f_out.write(" ".join(map(str, arr)) + "\n")
