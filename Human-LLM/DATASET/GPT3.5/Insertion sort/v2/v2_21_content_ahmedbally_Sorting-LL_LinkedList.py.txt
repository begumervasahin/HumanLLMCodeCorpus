class Node:
    def __init__(self, data):
        self.data = data
        self.next = None

class LinkedList:
    def __init__(self):
        self.head = None
        self.tail = None
        self.size = 0

    def is_empty(self):
        return self.head is None

    def append(self, item):
        new_node = Node(item)
        new_node.next = self.head
        if self.head is None:
            self.tail = new_node
        self.head = new_node
        self.size += 1

    def append_sort(self, item):
        new_node = Node(item)
        new_node.next = None
        current = self.tail
        previous = None

        while current is not None:
            if current.data > item:
                break
            else:
                previous = current
                current = current.next

        if previous is None:
            new_node.next = self.tail
            self.tail = new_node
        else:
            new_node.next = current
            previous.next = new_node

    def length(self):
        current = self.head
        count = 0
        while current is not None:
            count += 1
            current = current.next
        return count

    def search(self, item):
        current = self.head
        while current is not None:
            if current.data == item:
                return True
            current = current.next
        return False

    def remove(self, item):
        current = self.head
        previous = None
        found = False

        while not found:
            if current.data == item:
                found = True
            else:
                previous = current
                current = current.next

        if previous is None:
            self.head = current.next
        else:
            previous.next = current.next

        self.size -= 1

# Example usage:
if __name__ == "__main__":
    linked_list = LinkedList()
    linked_list.append(1)
    linked_list.append(3)
    linked_list.append(2)
    linked_list.append(5)
    linked_list.append_sort(4)

    print("Length:", linked_list.length())
    print("Is 2 in the list?", linked_list.search(2))
    print("Is 6 in the list?", linked_list.search(6))

    linked_list.remove(3)
    print("Length after removing:", linked_list.length())
