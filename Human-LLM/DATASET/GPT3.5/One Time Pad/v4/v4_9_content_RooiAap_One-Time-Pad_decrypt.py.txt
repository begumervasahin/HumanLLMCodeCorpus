# Import necessary functions from the 'algorithms' module
from modules import algorithms

# Get input from the user for ciphertext and pad
ciphertext = input("Enter ciphertext: ")
pad = input("Enter pad: ")

# Convert the input strings into lists of characters
ciphertext_list = list(ciphertext)
pad_list = list(pad)

# Convert the characters in the lists to their corresponding numerical values
ciphertext_num = algorithms.alph_pos(ciphertext_list)
pad_num = algorithms.alph_pos(pad_list)

# Initialize an empty list to store the numerical values of the plaintext
plaintext_num = []

# Iterate through each numerical value in the ciphertext and pad lists
for index in range(len(ciphertext_num)):
    # Calculate the inverse modulo 26 of the difference between the corresponding ciphertext and pad numerical values
    inverse_mod = algorithms.modinverse((ciphertext_num[index] - pad_num[index]), 26)
    # Append the calculated numerical value to the plaintext numerical list
    plaintext_num.append(inverse_mod)

# Convert the numerical values of the plaintext back to their corresponding characters
plaintext = algorithms.num_alph(plaintext_num)

# Print the resulting plaintext as a string
print("Decrypted plaintext:", "".join(plaintext))
