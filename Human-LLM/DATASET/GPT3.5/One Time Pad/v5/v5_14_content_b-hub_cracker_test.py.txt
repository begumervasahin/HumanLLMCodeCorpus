import sys

def convert_number_to_column_id(number):
    """
    Convert a number to a column ID in Excel format.

    Args:
    - number (int): The number to convert.

    Returns:
    - str: The column ID.
    """
    column_id = ""
    while number > 0:
        number -= 1
        column_id = chr(ord('A') + number % 26) + column_id
        number //= 26
    return column_id

if len(sys.argv) < 2:
    print("Usage: python script.py <input_file>")
    sys.exit(1)

input_file_path = sys.argv[1]

# Read the input file line by line
with open(input_file_path, 'r') as file:
    for line in file:
        test_number = int(line.strip())
        column_id = convert_number_to_column_id(test_number)
        print(f"{test_number}: {column_id}")
