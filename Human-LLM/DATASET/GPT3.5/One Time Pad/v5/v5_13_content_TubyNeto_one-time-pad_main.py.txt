import string
import random

def encode_letter(plain_letter, key_letter):
    """Encode a single letter using the Vigenère cipher."""
    alphabet = string.ascii_lowercase
    plain_num = alphabet.index(plain_letter.lower()) + 1
    key_num = alphabet.index(key_letter.lower()) + 1
    cipher_num = (plain_num + key_num) % 26
    return alphabet[cipher_num - 1]

def decode_letter(cipher_letter, key_letter):
    """Decode a single letter using the Vigenère cipher."""
    alphabet = string.ascii_lowercase
    cipher_num = alphabet.index(cipher_letter.lower()) + 1
    key_num = alphabet.index(key_letter.lower()) + 1
    plain_num = (cipher_num - key_num) % 26
    return alphabet[plain_num - 1]

def decrypt(cipher_text, key):
    """Decrypt a text using the Vigenère cipher."""
    plain_text = ''
    for i in range(len(cipher_text)):
        plain_text += decode_letter(cipher_text[i], key[i % len(key)])
    return plain_text

def encrypt(plain_text, key):
    """Encrypt a text using the Vigenère cipher."""
    cipher_text = ''
    for i in range(len(plain_text)):
        cipher_text += encode_letter(plain_text[i], key[i % len(key)])
    return cipher_text

def generate_key(text_length):
    """Generate a random key of specified length."""
    letters = string.ascii_lowercase
    return ''.join(random.choice(letters) for _ in range(text_length))

def main():
    """Main function to interact with the user."""
    while True:
        print("Menu:")
        print("1 - Encrypt")
        print("2 - Decrypt")
        print("3 - Quit")
        selected = input("Select an option: ")
        if selected == '1':
            plain_text = input("Enter the Plain Text: ")
            key = generate_key(len(plain_text))
            print('Cipher Text:', encrypt(plain_text, key))
        elif selected == '2':
            cipher_text = input("Enter the Cipher Text: ")
            key = input("Enter the Key: ")
            print('Plain Text:', decrypt(cipher_text, key))
        elif selected == '3':
            break
        else:
            print("Invalid option. Please choose 1, 2, or 3.")

if __name__ == "__main__":
    main()
