import random

def string_to_ascii(string, accept_range):
    """Converts a string to a list of ASCII values, filtering out characters not in the accept range."""
    ascii_list = []
    for char in string:
        num = ord(char)
        if num in accept_range:
            ascii_list.append(num)
    return ascii_list

def ascii_to_string(ascii_list):
    """Converts a list of ASCII values to a string."""
    string_out = ""
    for num in ascii_list:
        string_out += chr(num)
    return string_out

def encrypt_message():
    """Encrypts a message and writes it to a file along with the corresponding key."""
    accept_range = list(range(32, 127))
    print('Write your message. Only ASCII characters in the range 32-126 are accepted.')
    plaintext = input("> ")
    print('Name your ciphertext output file, excluding the .txt extension. For example: message1')
    print('The corresponding key will have "key" appended to the filename. For example: message1key')
    file_name = input("> ")
    plaintext_ascii = string_to_ascii(plaintext, accept_range)
    key = [random.choice(accept_range) for _ in range(len(plaintext_ascii))]
    ciphertext_ascii = [(plaintext_ascii[i] - accept_range[0] + key[i] - accept_range[0]) % len(accept_range) + accept_range[0] for i in range(len(plaintext_ascii))]
    ciphertext = ascii_to_string(ciphertext_ascii)
    key = ascii_to_string(key)
    new_file_name = file_name + ".txt"
    with open(new_file_name, "w") as file:
        file.write(ciphertext)
    key_file = file_name + "key.txt"
    with open(key_file, "w") as file:
        file.write(key)

def decrypt_message():
    """Decrypts a message using the provided key."""
    accept_range = list(range(32, 127))
    print('Enter the name of the ciphertext file, excluding the extension.')
    cipher_file = input('> ')
    print('Enter the name of the key file, excluding the extension.')
    key_file = input('> ')
    with open(cipher_file + ".txt", "r") as file:
        ciphertext = file.read()
    with open(key_file + ".txt", "r") as file:
        key = file.read()
    ciphertext_ascii = string_to_ascii(ciphertext, accept_range)
    key_ascii = string_to_ascii(key, accept_range)
    plaintext_ascii = [(ciphertext_ascii[i] - accept_range[0] - key_ascii[i] + accept_range[0]) % len(accept_range) + accept_range[0] for i in range(len(ciphertext_ascii))]
    plaintext = ascii_to_string(plaintext_ascii)
    plaintext_file_name = cipher_file + "plain.txt"
    with open(plaintext_file_name, "w") as file:
        file.write(plaintext)

while True:
    print('Choose a mode:\n (1) Encrypt a message\n (2) Decrypt files\n (3) Quit')
    mode = int(input('> '))
    if mode == 1:
        encrypt_message()
        print('Encryption successful.\n\n')
    elif mode == 2:
        decrypt_message()
        print('Decryption successful. Note: spaces may be removed!\n\n')
    else:
        print('Goodbye.')
        break
