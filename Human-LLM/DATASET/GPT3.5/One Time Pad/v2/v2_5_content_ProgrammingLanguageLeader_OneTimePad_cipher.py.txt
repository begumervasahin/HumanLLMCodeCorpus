import argparse

def encrypt_text(plain_text: bytes, key: bytes) -> bytes:
    """
    Encrypts the given plaintext using the one-time pad technique with the provided key.

    Args:
        plain_text (bytes): The plaintext to encrypt.
        key (bytes): The encryption key.

    Returns:
        bytes: The encrypted text.
    """
    encrypted_text = b''
    for plain_char, key_char in zip(plain_text, key):
        encrypted_text += (plain_char ^ key_char).to_bytes(1, 'big')
    return encrypted_text

def create_argument_parser() -> argparse.ArgumentParser:
    """
    Creates an argument parser for the one-time pad encoder/decoder.

    Returns:
        argparse.ArgumentParser: The argument parser object.
    """
    parser = argparse.ArgumentParser(description='One-time pad encoder/decoder')
    parser.add_argument('key', type=str, help='Path to the encryption key file')
    parser.add_argument('source', type=str, help='Path to the source file')
    parser.add_argument('output', type=str, help='Path to the output file')
    return parser

if __name__ == '__main__':
    parser = create_argument_parser()
    args = parser.parse_args()

    source_file_path = args.source
    output_file_path = args.output
    key_file_path = args.key

    block_size = 8192

    with open(key_file_path, 'rb') as key_file, \
            open(source_file_path, 'rb') as source_file, \
            open(output_file_path, 'wb') as output_file:
        for source_block in iter(lambda: source_file.read(block_size), b''):
            key_block = key_file.read(len(source_block))
            output_block = encrypt_text(source_block, key_block)
            output_file.write(output_block)

    print('Encryption/decryption completed.')
