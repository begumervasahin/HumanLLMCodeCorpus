from GF2 import zero, one
import mat
import random

def str_to_bits(input_str):
    """
    Convert a string into a list of bits.
    Each character is represented by its ASCII code, and each bit in the
    code is converted to a zero or one in the resulting list.
    """
    bits = [1 << i for i in range(8)]
    return [one if ord(char) & bit else zero for char in input_str for bit in bits]

def bits_to_str(bits):
    """
    Convert a list of bits into a string.
    Bits are grouped into bytes (8 bits), and each byte is converted to
    its corresponding ASCII character.
    """
    bits_per_byte = 8
    return ''.join(chr(sum(bitval if bit else 0 for bitval, bit in zip(bits[i:i + bits_per_byte], range(bits_per_byte)))) for i in range(0, len(bits), bits_per_byte))

def bits_to_matrix(bits, nrows=4, transpose=False):
    """
    Convert a list of bits into a matrix with specified number of rows.
    The matrix is filled column by column, and each bit represents an entry.
    """
    ncols = len(bits) // nrows
    entries = {(i, j): one for j in range(ncols) for i in range(nrows) if bits[nrows * j + i]}
    matrix = mat.Mat((set(range(nrows)), set(range(ncols))), entries)
    if transpose:
        matrix = mat.transpose(matrix)
    return matrix

def matrix_to_bits(matrix, transpose=False):
    """
    Convert a matrix into a list of bits.
    Entries in the matrix are read column by column, and each bit
    is appended to the resulting list.
    """
    if transpose:
        return [matrix[i, j] for i in sorted(matrix.D[0]) for j in sorted(matrix.D[1])]
    else:
        return [matrix[i, j] for j in sorted(matrix.D[1]) for i in sorted(matrix.D[0])]

def add_noise(matrix, frequency):
    """
    Generate a random noise matrix with the same domain as the input matrix.
    The probability of a one in any entry of the noise matrix is specified
    by the frequency parameter.
    """
    entries = {(i, j): one for i in matrix.D[0] for j in matrix.D[1] if random.random() < frequency}
    return mat.Mat(matrix.D, entries)
