def strxor(a, b):
    """Returns the XOR of two strings."""
    return "".join([chr(ord(x) ^ ord(y)) for (x, y) in zip(a, b)])

def intercept_in():
    """Reads input from the user and returns it."""
    return raw_input("Enter input: ")

def intercept_out(data):
    """Prints the intercepted data."""
    print("Intercepted data:", data)
