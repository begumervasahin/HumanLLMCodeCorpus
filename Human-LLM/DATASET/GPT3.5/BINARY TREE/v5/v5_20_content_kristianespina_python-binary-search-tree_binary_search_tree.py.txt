from random import randint

class Node:
    def __init__(self, value):
        self.value = value
        self.left = None
        self.right = None

    def insert(self, value):
        """
        Inserts a value into the binary tree.
        """
        if value < self.value:
            if self.left is None:
                self.left = Node(value)
            else:
                self.left.insert(value)
        elif value > self.value:
            if self.right is None:
                self.right = Node(value)
            else:
                self.right.insert(value)
        else:
            self.value = value

    def convert_to_array(self, node, array):
        """
        Converts the binary tree to an array using Depth-First Search (DFS).
        """
        if node:
            array.append(node.value)
            self.convert_to_array(node.left, array)
            self.convert_to_array(node.right, array)
        return array

class BinaryTree:
    def __init__(self):
        self.root = None

    def insert(self, data):
        """
        Inserts data into the binary tree.
        """
        if self.root is None:
            self.root = Node(data)
        else:
            self.root.insert(data)

    def find(self, data):
        """
        Searches for a value in the binary tree.
        """
        return self.find_node(self.root, data)

    def find_node(self, node, data):
        """
        Helper function to recursively find a value in the binary tree.
        """
        if node is None:
            return False
        elif data == node.value:
            return True
        elif data < node.value:
            return self.find_node(node.left, data)
        else:
            return self.find_node(node.right, data)

    def convert_to_array(self):
        """
        Converts the binary tree to an array using Depth-First Search (DFS).
        """
        return self.root.convert_to_array(self.root, [])

    def rearrange(self):
        '''
        Re-arrange the binary tree so that the median value becomes the root.
        Returns the binary tree in DFS Array Form.
        '''
        nodes = self.convert_to_array()
        nodes.sort()
        middle = len(nodes) // 2
        self.root = Node(nodes.pop(middle))
        for element in nodes:
            self.root.insert(element)
        return self.convert_to_array()

# Instantiate BinaryTree object
tree = BinaryTree()

# Insert random values into the tree
for i in range(50):
    tree.insert(randint(0, 100))

# Print original tree
print("Original tree:")
print(tree.convert_to_array())

# Rearrange the tree so that the median is the root
print("\nRearranged tree:")
print(tree.rearrange())

# Check if a value exists in the tree
print("\nDoes the tree contain 55?", tree.find(55))
