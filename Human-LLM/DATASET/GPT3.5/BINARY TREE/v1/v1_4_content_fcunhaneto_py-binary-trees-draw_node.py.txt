class Node:
    def __init__(self, key):
        self.key = key
        self.left = None
        self.right = None
        self.parent = None
        self.height = 0
        self.real_height = 0
        self.color = 1

# Example usage:
# Create nodes and set their attributes
node1 = Node(10)
node1.height = 3
node1.real_height = 4
node1.color = 1

node2 = Node(20)
node2.height = 2
node2.real_height = 3
node2.color = 0

# Print node attributes
print("Node 1:")
print("Key:", node1.key)
print("Left Child:", node1.left)
print("Right Child:", node1.right)
print("Parent:", node1.parent)
print("Height:", node1.height)
print("Real Height:", node1.real_height)
print("Color:", node1.color)

print("\nNode 2:")
print("Key:", node2.key)
print("Left Child:", node2.left)
print("Right Child:", node2.right)
print("Parent:", node2.parent)
print("Height:", node2.height)
print("Real Height:", node2.real_height)
print("Color:", node2.color)
