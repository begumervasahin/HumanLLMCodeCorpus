from PersistentBST import PersistentBST

print("\n\tARCHEIO")
Tree = PersistentBST()  # Initialize the Persistent Binary Search Tree
currentBranch = "master"
prevBranch = None
branches = ["master"]

while True:
    print("\nCurrent Branch:", currentBranch)
    print("\n1. List Files\n2. View File\n3. New File\n4. Delete File\n5. Edit File\n6. List Branches\n7. New Branch\n8. Switch Branch\n9. Exit\n")
    choice = input("$ ")
    print()

    if choice == "1":
        print("Files: ")
        # Use Tree.inFix() method to list files in the current branch
        Tree.inFix(currentBranch)
    elif choice == "2":
        name = input("Enter name of file to be opened: ")
        # Use Tree.search() method to search for the file in the current branch
        file = Tree.search(name, currentBranch)
        if file is None:
            print("File does not exist!")
        else:
            print(file.getData())
    elif choice == "3":
        name = input("Enter name of new file: ")
        # Use Tree.insert() method to insert a new file in the current branch
        Tree.insert(name, currentBranch)
    elif choice == "4":
        name = input("Enter name of file to be deleted: ")
        # Use Tree.delete() method to delete a file from the current branch
        Tree.delete(name, currentBranch)
    elif choice == "5":
        name = input("Enter name of file to be edited: ")
        # Use Tree.edit() method to edit a file in the current branch
        Tree.edit(name, currentBranch)
    elif choice == "6":
        print("Branches:")
        for branch in branches:
            print(branch)
    elif choice == "7":
        branchName = input("Enter name of new branch: ")
        if branchName in branches:
            print("Branch already exists")
            continue
        branches.append(branchName)
        prevBranch, currentBranch = currentBranch, branchName
        # Use Tree.newBranch() method to create a new branch
        Tree.newBranch(currentBranch, prevBranch)
    elif choice == "8":
        branchName = input("Enter name of branch to switch to: ")
        if branchName not in branches:
            print("Branch doesn't exist")
            continue
        prevBranch, currentBranch = currentBranch, branchName
    elif choice == "exit" or choice == "9":
        print("Exiting...")
        break
    elif choice == "debug":
        print()
    else:
        print("Invalid Input")
