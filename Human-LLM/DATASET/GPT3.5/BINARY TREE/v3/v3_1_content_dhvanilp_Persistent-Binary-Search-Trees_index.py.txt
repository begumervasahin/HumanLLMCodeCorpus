from PersistentBST import PersistentBST

def display_menu(current_branch):
    """
    Display menu options.
    """
    print("\nCurrent Branch:", current_branch)
    print("\n1. List Files\n2. View File\n3. New File\n4. Delete File\n5. Edit File\n6. List Branches\n7. New Branch\n8. Switch Branch\n9. Exit\n")

def list_files(tree, current_branch):
    """
    List files in the current branch.
    """
    print("Files: ")
    tree.inFix(current_branch)

def view_file(tree, current_branch):
    """
    View contents of a file.
    """
    name = input("Enter name of file to be opened: ")
    file = tree.search(name, current_branch)
    if file is None:
        print("File does not exist!")
    else:
        print(file.getData())

def new_file(tree, current_branch):
    """
    Insert a new file into the current branch.
    """
    name = input("Enter name of new file: ")
    tree.insert(name, current_branch)

def delete_file(tree, current_branch):
    """
    Delete a file from the current branch.
    """
    name = input("Enter name of file to be deleted: ")
    tree.delete(name, current_branch)

def edit_file(tree, current_branch):
    """
    Edit a file in the current branch.
    """
    name = input("Enter name of file to be edited: ")
    tree.edit(name, current_branch)

def list_branches(branches):
    """
    List all branches.
    """
    print("Branches:")
    for branch in branches:
        print(branch)

def new_branch(tree, branches, current_branch):
    """
    Create a new branch.
    """
    branch_name = input("Enter name of new branch: ")
    if branch_name in branches:
        print("Branch already exists")
    else:
        branches.append(branch_name)
        previous_branch, current_branch = current_branch, branch_name
        tree.newBranch(current_branch, previous_branch)

def switch_branch(branches, current_branch):
    """
    Switch to another branch.
    """
    branch_name = input("Enter name of branch to switch to: ")
    if branch_name not in branches:
        print("Branch doesn't exist")
    else:
        current_branch = branch_name
    return current_branch

def main():
    """
    Main function to drive the program.
    """
    print("\n\tARCHEIO")
    tree = PersistentBST()  # Initialize the Persistent Binary Search Tree
    current_branch = "master"
    branches = ["master"]

    while True:
        display_menu(current_branch)
        choice = input("$ ")
        print()

        if choice == "1":
            list_files(tree, current_branch)
        elif choice == "2":
            view_file(tree, current_branch)
        elif choice == "3":
            new_file(tree, current_branch)
        elif choice == "4":
            delete_file(tree, current_branch)
        elif choice == "5":
            edit_file(tree, current_branch)
        elif choice == "6":
            list_branches(branches)
        elif choice == "7":
            new_branch(tree, branches, current_branch)
        elif choice == "8":
            current_branch = switch_branch(branches, current_branch)
        elif choice.lower() in ("exit", "9"):
            print("Exiting...")
            break
        elif choice.lower() == "debug":
            print()
        else:
            print("Invalid Input")

if __name__ == "__main__":
    main()
