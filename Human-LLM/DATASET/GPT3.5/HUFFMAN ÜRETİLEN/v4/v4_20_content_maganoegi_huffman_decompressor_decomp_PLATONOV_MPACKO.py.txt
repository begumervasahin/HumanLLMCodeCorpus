import os
from tqdm import tqdm
from decomp_lib import *

# Clear the console screen
os.system('cls' if os.name == 'nt' else 'clear')

# Print a message from the Hepia team
hepia_print()
print("\n")

# Initialize progress bar
bar_format = '{l_bar}{bar}| {n_fmt}/{total_fmt}'
with tqdm(total=6, ncols=80, bar_format=bar_format) as pbar:
    # Read user input for filenames
    filenames = read_user_input()
    update_bar(pbar, 'Open File  ')

    # Assign filenames
    file_2_read = filenames[0]
    json_name = filenames[1]
    file_2_write = filenames[2]

    # Convert file contents to byte array
    byte_array = bytes_2_array(file_2_read)
    update_bar(pbar, 'File->Array')

    # Read JSON dictionary
    dictionary = json_2_dict(json_name)
    update_bar(pbar, 'Json Read  ')

    # Remove zero bytes from the byte array
    byte_string_no_zeros = strip_zeros(byte_array)
    update_bar(pbar, 'Strip Zeros')

    # Retranslate byte array using the dictionary
    retranslated_byte_array = retranslate(byte_string_no_zeros, dictionary)
    update_bar(pbar, 'Retranslate')

    # Write retranslated byte array to binary file
    write_2_binary(retranslated_byte_array, file_2_write)
    update_bar(pbar, 'Done       ')

    # Display decompression info
    time.sleep(0.1)
    pbar.close()
    decompressor_display(file_2_write)
