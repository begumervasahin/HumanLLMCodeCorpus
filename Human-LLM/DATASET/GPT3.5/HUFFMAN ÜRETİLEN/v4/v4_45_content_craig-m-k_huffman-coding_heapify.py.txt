class Heap:
    def __init__(self, heap_type):
        if heap_type not in ('max', 'min'):
            raise ValueError('Invalid heap type. Allowed values are "max" or "min".')
        
        self.queue = []
        self.heapsize = 0
        self.heap_type = heap_type
    
    def __len__(self):
        return len(self.queue)
    
    def _parent(self, i):
        return (i - 1) // 2
    
    def _left_child(self, i):
        return 2 * i + 1
    
    def _right_child(self, i):
        return 2 * i + 2
    
    def _heapify_down(self, index):
        left = self._left_child(index)
        right = self._right_child(index)
        
        if self.heap_type == 'min':
            smallest = index
            if left < self.heapsize and self.queue[left] < self.queue[index]:
                smallest = left
            if right < self.heapsize and self.queue[right] < self.queue[smallest]:
                smallest = right
            if smallest != index:
                self.queue[index], self.queue[smallest] = self.queue[smallest], self.queue[index]
                self._heapify_down(smallest)
        else:
            largest = index
            if left < self.heapsize and self.queue[left] > self.queue[index]:
                largest = left
            if right < self.heapsize and self.queue[right] > self.queue[largest]:
                largest = right
            if largest != index:
                self.queue[index], self.queue[largest] = self.queue[largest], self.queue[index]
                self._heapify_down(largest)
    
    def _heapify_up(self, index):
        parent = self._parent(index)
        if self.heap_type == 'min':
            while index > 0 and self.queue[parent] > self.queue[index]:
                self.queue[index], self.queue[parent] = self.queue[parent], self.queue[index]
                index = parent
                parent = self._parent(index)
        else:
            while index > 0 and self.queue[parent] < self.queue[index]:
                self.queue[index], self.queue[parent] = self.queue[parent], self.queue[index]
                index = parent
                parent = self._parent(index)
    
    def insert(self, key):
        self.queue.append(key)
        self.heapsize += 1
        self._heapify_up(self.heapsize - 1)
    
    def pop(self):
        if not self.queue:
            raise IndexError("Heap is empty")
        
        popped_key = self.queue[0]
        self.queue[0] = self.queue.pop()
        self.heapsize -= 1
        
        if self.heapsize > 0:
            self._heapify_down(0)
        
        return popped_key
    
    def build_heap(self):
        for i in range(self.heapsize // 2, -1, -1):
            self._heapify_down(i)
