class HuffmanHeap:
    """
    A priority queue implementation for Huffman trees.
    
    Attributes:
        old: A sorted list of old Huffman trees.
        new: A sorted list of new Huffman trees.
    """

    def __init__(self, old, new=None):
        """
        Initialize the HuffmanHeap with old and new lists.

        Args:
            old (list): The old list of Huffman trees, sorted.
            new (list): The new list of Huffman trees, optional and defaults to None.
        """
        self.old = old
        self.new = new

    def enqueue(self, item):
        """
        Add a Huffman tree to the new list.

        Args:
            item: A Huffman tree to be added.
        """
        self.new.append(item)

    def dequeue(self):
        """
        Remove and return the smallest item from the heap.

        Returns:
            The smallest item. If the heap is empty, returns None.
        """
        if len(self.old) == 0 and len(self.new) != 0:
            item = self.new.pop(0)
        elif len(self.new) == 0 and len(self.old) != 0:
            item = self.old.pop(0)
        elif len(self.new) != 0 and len(self.old) != 0:
            if self.old[0].get_freq() >= self.new[0].get_freq():
                item = self.new.pop(0)
            else:
                item = self.old.pop(0)
        else:
            print("Both old and new lists are empty!")
            return None

        return item
