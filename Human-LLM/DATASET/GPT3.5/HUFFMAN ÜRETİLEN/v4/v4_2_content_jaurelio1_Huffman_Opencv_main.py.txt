import cv2
from Heap import Heap
from Huffman import Huffman
from Node import Node
import zipfile

if __name__ == '__main__':
    # Read the image
    image = cv2.imread("img1.tif")
    
    # Resize the image
    scale = 60
    w = int(image.shape[1] * scale / 100)
    h = int(image.shape[0] * scale / 100)
    d = (w, h)
    resized = cv2.resize(image, d, interpolation=cv2.INTER_AREA)
    (h_resized, w_resized, d_resized) = resized.shape
    
    # Convert the resized image to grayscale
    image_copy = resized.copy()
    image_copy = cv2.cvtColor(image_copy, cv2.COLOR_RGB2GRAY)
    
    # Create a Node object and set frequencies for pixels
    node = Node()
    node.setFrequencePixels(image_copy, h_resized, w_resized)
    
    # Return an array of nodes
    tmp = node.returnArrayNode()
    
    # Create a heap using the array of nodes
    heap = Heap(tmp)
    
    # Return the minimum heap
    S = heap.returnHeapMinimum()
    
    # Create a Huffman tree
    huff = Huffman(S)
    
    # Return the Huffman code
    R = huff.returnHuff()
    
    # Traverse the Huffman tree to generate the dictionary
    D = huff.goThroughTree(R)
    
    # Compress the image using the generated dictionary
    T, index = huff.compressOp(image_copy, w_resized, h_resized, D)
    
    # Write the compressed data to a file
    file = open('test.txt', 'w')
    for i in range(index):
        file.write(str(T[i]) + ' ')
    file.close()
    
    # Read the compressed data from the file
    file1 = open('test.txt', 'r')
    
    # Decompress the data using the Huffman tree
    dc = huff.decompressOp(file1, R)
    print(dc)
    file1.close()
    
    # Wait for a key press
    cv2.waitKey(0)
