# Open the input file for reading
with open("input.txt", "r") as f:
    # Read the first line and convert it to lowercase
    s = f.readline().lower()

# Create a set of unique characters in the string
t = list(set(s))

# Display the original string and the character list
print("The String:", s)
print("The char list:", t)

# Initialize a list to store the frequency of each character
b = [0] * len(t)

# Populate the frequency table
for x in s:
    count = s.count(x)
    y = t.index(x)
    if b[y] == 0:
        b[y] = count

# Display the frequency table before sorting
print("Frequency Table before sorting:")
print(b)

# Initialize lists to store sorted frequencies and corresponding characters
new = []
newchar = []

# Sort the frequencies and corresponding characters
while b:
    max_freq = max(b)
    index = b.index(max_freq)
    new.append(max_freq)
    newchar.append(t[index])
    b.pop(index)
    t.pop(index)

# Display the frequency table after sorting
print("Frequency Table after sorting:")
print(new)
print(newchar)

# Calculate the number of nodes and leafs in the tree
k = sum(1 for x in new if x != 0)
j = 2 * k - 1
print("Number of nodes in the tree:", j)
print("Number of leafs in the tree:", k)

# Initialize encoder and binary tree lists
encoder = ["null"] * k
bintree = list(new)
strtree = list(newchar)

# Huffman encoding algorithm
while len(bintree) > 1:
    right = min(bintree)
    rightstr = strtree[bintree.index(right)]
    bintree.remove(right)
    strtree.remove(rightstr)

    left = min(bintree)
    leftstr = strtree[bintree.index(left)]
    bintree.remove(left)
    strtree.remove(leftstr)

    summation = right + left
    strsum = leftstr + rightstr
    strtree.append(strsum)
    bintree.append(summation)

    for char in rightstr:
        ci = newchar.index(char)
        encoder[ci] = "0" + encoder[ci]

    for char in leftstr:
        ci = newchar.index(char)
        encoder[ci] = "1" + encoder[ci]

# Display the final encoder
print("Encoder:", encoder)

# Generate the encoded string
encoded_string = ""
for char in s:
    index = newchar.index(char)
    encoded_string += encoder[index]

print("Encoded String:", encoded_string)

# Write the encoded string to output.txt
with open("output.txt", "w") as g:
    g.write(encoded_string)

# Write the dictionary to dictionary.txt
with open("dictionary.txt", "w") as p:
    for i in range(len(encoder)):
        q = newchar[i]
        t = encoder[i]
        w = q + "=" + t
        print(w)
        p.write(w + "\n")
