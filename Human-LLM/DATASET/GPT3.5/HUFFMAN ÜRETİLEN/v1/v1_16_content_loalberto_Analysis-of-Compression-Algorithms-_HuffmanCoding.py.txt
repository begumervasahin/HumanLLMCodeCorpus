from heapq import heappush, heappop, heapify
from collections import defaultdict
import string
import random
import time

def encode(values):
    """Huffman encode the given dict mapping symbols to weights"""
    count = 0
    heap = []
    for sym, wt in values.items():
        count += 1
        heap.append([wt, [sym, ""]])
    heapify(heap)
    while len(heap) > 1:
        count += 1
        lo = heappop(heap)
        hi = heappop(heap)
        for pair in lo[1:]:
            pair[1] = '0' + pair[1]
        for pair in hi[1:]:
            pair[1] = '1' + pair[1]
        heappush(heap, [lo[0] + hi[0]] + lo[1:] + hi[1:])
    return sorted(heappop(heap)[1:], key=lambda val: (len(val[-1]), val)), count

def random_string(string_length):
    """Generate a random string of fixed length """
    letters = string.ascii_lowercase
    return ''.join(random.choice(letters) for i in range(string_length))

itr = 10

op_matrix = {f'Iteration{i}': [] for i in range(1, itr + 1)}
time_matrix = {f'Iteration{i}': [] for i in range(1, itr + 1)}

for i in range(1, itr + 1):
    for n in range(1, 6):
        n = 10 ** n
        txt = random_string(n)
        huffman = defaultdict(int)
        first_count = 0
        for ch in txt:
            first_count += 1
            huffman[ch] += 1
        start_time = time.time()
        vals = encode(huffman)
        end_time = time.time()
        total_time = (end_time - start_time) * 1000000000
        total = vals[1] + first_count
        op_matrix[f'Iteration{i}'].append(total)
        time_matrix[f'Iteration{i}'].append(total_time)

res = {i: 0 for i in range(len(op_matrix['Iteration1']))}
for key in op_matrix:
    for idx, val in enumerate(op_matrix[key]):
        res[idx] += val

op_matrix['Average'] = [val / itr for val in res.values()]

res = {i: 0 for i in range(len(time_matrix['Iteration1']))}
for key in time_matrix:
    for idx, val in enumerate(time_matrix[key]):
        res[idx] += val

time_matrix['Average'] = [val / itr for val in res.values()]

print('\n\n------------This is for the operation measurement------------\n\n')
for key, value in op_matrix.items():
    print(f"{key}: {value}")

print('\n\n------------This is for the time measurement------------\n\n')
for key, value in time_matrix.items():
    print(f"{key}: {value}")
