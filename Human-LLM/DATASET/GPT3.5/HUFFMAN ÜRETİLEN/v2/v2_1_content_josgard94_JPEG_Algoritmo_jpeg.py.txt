# Function to perform Discrete Cosine Transform (DCT) on a matrix
def dct2(a):
    new_matrix = np.zeros((8, 8))
    b = np.zeros((8, 8))
    for i in range(8):
        for j in range(8):
            b[i][j] = a[i][j] - 128
    c = scipy.fftpack.dct(scipy.fftpack.dct(b, axis=0, norm='ortho'), axis=1, norm='ortho')
    for i in range(8):
        for j in range(8):
            new_matrix[i][j] = np.fix(c[i][j] / Z[i][j])
    return new_matrix

# Function to perform Zigzag scanning on a matrix
def zigzag(matrix):
    global g
    con = 0
    matrix = np.array(matrix)
    rows, columns = 8, 8
    aux = np.zeros((1, 64))
    x = np.zeros((1, 64))
    y = np.zeros((1, 64))
    solution = [[] for i in range(rows + columns - 1)]
    for i in range(rows):
        for j in range(columns):
            sum = i + j
            if(sum % 2 == 0):
                solution[sum].insert(0, matrix[i][j])
            else:
                solution[sum].append(matrix[i][j])
    solution.reverse()
    for i in solution:
        for j in i:
            if j == -0.0:
                j = abs(j)
                aux[0, con] = j
            else:
                aux[0, con] = j
            con = con + 1
    indice = 0
    for i in range(0, 64):
        if(aux[0, i] != 0):
            indice = i
            break
    for i in range(indice, 64):
        temporal[g] = aux[0, i]
        if aux[0, i] in frecuencia:
            pass
        else:
            frecuencia[aux[0, i]] = aux[0, i]
        g = g + 1

im2 = np.zeros((256, 256))
for i in range(0, alto, 8):

        im2[i:(i+8), j:(j+8)] = dct2(a[i:(i+8), j:(j+8)])

# Perform Zigzag scanning on the DCT matrix
for i in range(0, alto, 8):
    for j in range(0, ancho, 8):
        zigzag(im2[i:(i+8), j:(j+8)])
