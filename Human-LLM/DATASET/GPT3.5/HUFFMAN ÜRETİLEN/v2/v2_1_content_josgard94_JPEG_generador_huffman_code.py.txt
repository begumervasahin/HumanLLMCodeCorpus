from decimal import Decimal
import sys
import os
import operator

# File paths and names
result_probabilities_file = "result.txt"
input_file = sys.argv[2]
compressed_file = "compressed.dat"
codes_file = "codes.txt"

def huffman_compressor():
    # Initialize dictionaries to store probabilities and Huffman codes
    probabilities = {}
    codes_table = {}

    # Read probabilities from the result file
    with open(result_probabilities_file, 'r') as file:
        for line in file:
            tokens = line.split("\t")
            if tokens[0] in ["space", "salto"]:
                continue
            probabilities[tokens[0]] = float(tokens[1].rstrip())

    # Generate Huffman codes
    codes_table = generate_huffman_codes(probabilities)
    
    # Save Huffman codes to a file
    save_codes(codes_table)

    # Read input text file, encode it using Huffman codes, and compress it
    with open(input_file, 'r') as text_file, open(compressed_file, 'wb') as output:
        text = text_file.read().rstrip().lower()
        encoded_text = encode_text(codes_table, text)
        padded_encoded = pad_encoded_text(encoded_text)
        bit_array = generate_bit_array(padded_encoded)
        output.write(bytes(bit_array))

def sort_probabilities(dictionary):
    sorted_probabilities = sorted(dictionary.items(), key=operator.itemgetter(1), reverse=False)
    return sorted_probabilities[0][0], sorted_probabilities[1][0]

def generate_huffman_codes(dictionary):
    if len(dictionary) == 2:
        return dict(zip(dictionary.keys(), ['0', '1']))

    probabilities_copy = dictionary.copy()
    key1, key2 = sort_probabilities(dictionary)
    prob1, prob2 = probabilities_copy.pop(key1), probabilities_copy.pop(key2)
    probabilities_copy[key1 + key2] = prob1 + prob2
    codes = generate_huffman_codes(probabilities_copy)
    combined_code = codes.pop(key1 + key2)
    codes[key1], codes[key2] = combined_code + '0', combined_code + '1'
    return codes

def save_codes(dictionary):
    with open(codes_file, "w") as file:
        for key in dictionary:
            if key == '\n':
                file.write("salto" + "\t" + dictionary.get(key, key) + "\n")
            else:
                file.write(dictionary.get(0, key) + "\t" + dictionary.get(key, key) + "\n")

def encode_text(codes, text):
    encoded_text = ""
    for char in text.split():
        if char in codes:
            encoded_text += codes[char]
    return encoded_text

def pad_encoded_text(encoded):
    padding = 8 - len(encoded) % 8
    for _ in range(padding):
        encoded += "0"
    padded_info = "{0:08b}".format(padding)
    return padded_info + encoded

def generate_bit_array(binary_string):
    if len(binary_string) % 8 != 0:
        exit(0)
    bit_array = bytearray()
    for i in range(0, len(binary_string), 8):
        byte = binary_string[i:i+8]
        bit_array.append(int(byte, 2))
    return bit_array

# Compress the input file using Huffman coding
huffman_compressor()

# Display compression statistics
print("\n\n")
original_size = os.path.getsize(sys.argv[2]) / (1024 * 1024.0)  # Convert bytes to MB
compressed_size = os.path.getsize(compressed_file) / (1024 * 1024.0)  # Convert bytes to MB
compression_ratio = ((compressed_size / original_size) * 100)

print("Original Text: " + str(sys.argv[2]) + " Size: " + str(original_size) + " MB")
print("Compressed File: " + compressed_file + " Size: " + str(compressed_size) + " MB")
print("File " + sys.argv[2] + " compressed by " + str(round(compression_ratio)) + "%")
print("Text file compressed successfully! :) !!\n\n")
