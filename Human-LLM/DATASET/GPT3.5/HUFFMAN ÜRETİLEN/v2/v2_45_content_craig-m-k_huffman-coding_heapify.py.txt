class Heap:
    def __init__(self, h_type):
        self.queue = []  # Initialize the heap as an empty list
        self.heapsize = 0  # Initialize heap size
        if h_type not in ('max', 'min'):  # Check if the heap type is valid
            raise ValueError('Invalid heap type. Use "max" or "min".')
        self.h_type = h_type  # Set the heap type

    def __len__(self):
        return len(self.queue)

    def perc_up(self, i):
        parent = (i + 1) // 2 - 1  # Calculate the index of the parent node
        while i > 0 and ((self.h_type == 'max' and self.queue[parent] < self.queue[i]) or
                         (self.h_type == 'min' and self.queue[parent] > self.queue[i])):
            # Swap the child with its parent if the heap property is violated
            self.queue[i], self.queue[parent] = self.queue[parent], self.queue[i]
            i = parent  # Move up the tree
            parent = (i + 1) // 2 - 1  # Update the index of the parent node

    def insert(self, key):
        self.queue.append(key)  # Add the new key to the end of the list
        self.heapsize += 1  # Increment heap size
        self.perc_up(self.heapsize - 1)  # Restore the heap property

    def pop(self):
        last_idx = self.heapsize - 1  # Get the index of the last element
        self.queue[last_idx], self.queue[0] = self.queue[0], self.queue[last_idx]  # Swap the root with the last element
        popped_key = self.queue.pop()  # Remove and return the last element (former root)
        self.heapsize -= 1  # Decrement heap size
        self.heapify(0)  # Restore the heap property starting from the root
        return popped_key

    def heapify(self, index):
        left = 2 * index + 1  # Calculate the index of the left child
        right = 2 * index + 2  # Calculate the index of the right child
        n = self.heapsize  # Get the current heap size

        if self.h_type == 'max':
            largest = index
            if left < n and self.queue[left] > self.queue[largest]:  # Compare with left child
                largest = left
            if right < n and self.queue[right] > self.queue[largest]:  # Compare with right child
                largest = right
            if largest != index:  # Swap if necessary
                self.queue[index], self.queue[largest] = self.queue[largest], self.queue[index]
                self.heapify(largest)  # Recursively heapify the subtree
        else:
            smallest = index
            if left < n and self.queue[left] < self.queue[smallest]:  # Compare with left child
                smallest = left
            if right < n and self.queue[right] < self.queue[smallest]:  # Compare with right child
                smallest = right
            if smallest != index:  # Swap if necessary
                self.queue[index], self.queue[smallest] = self.queue[smallest], self.queue[index]
                self.heapify(smallest)  # Recursively heapify the subtree

    def build_heap(self):
        n = self.heapsize  # Get the current heap size
        for i in range(n // 2, -1, -1):  # Iterate from the last parent node to the root
            self.heapify(i)  # Heapify each subtree rooted at index i

# Example usage:
max_heap = Heap('max')
max_heap.insert(4)
max_heap.insert(1)
max_heap.insert(7)
max_heap.insert(3)
print("Max Heap:", max_heap.queue)

min_heap.insert(4)
min_heap.insert(1)
