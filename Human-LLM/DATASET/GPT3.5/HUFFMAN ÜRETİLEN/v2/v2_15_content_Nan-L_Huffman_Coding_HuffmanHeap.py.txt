class HuffmanTree:
    def __init__(self, value, freq):
        """
        Initialize a Huffman Tree node with a value and its frequency.
        :param value: The value of the node.
        :param freq: The frequency of occurrence of the value.
        """
        self.value = value
        self.freq = freq

    def get_freq(self):
        """
        Get the frequency of the node.
        :return: Frequency of the node.
        """
        return self.freq


class HuffmanHeap:
    """
    HuffmanHeap class represents a priority queue for Huffman trees.
    """

    def __init__(self, old, new=None):
        """
        Initialize the HuffmanHeap with old and new lists of Huffman trees.
        :param old: Old list of Huffman trees, sorted.
        :param new: New list of Huffman trees, initially empty.
        """
        self.old = old
        self.new = new if new is not None else []

    def enqueue(self, item):
        """
        Enqueue a Huffman tree into the new list.
        :param item: A Huffman tree to be enqueued.
        :return: None
        """
        self.new.append(item)

    def dequeue(self):
        """
        Dequeue the smallest item from the heap.
        :return: The smallest item; if the heap is empty, return None.
        """
        if len(self.old) == 0 and len(self.new) != 0:
            item = self.new.pop(0)
        elif len(self.new) == 0 and len(self.old) != 0:
            item = self.old.pop(0)
        elif len(self.new) != 0 and len(self.old) != 0:
            if self.old[0].get_freq() >= self.new[0].get_freq():
                item = self.new.pop(0)
            else:
                item = self.old.pop(0)
        else:
            print("Both old and new lists are empty!")
            return None
        return item


# Example usage
if __name__ == "__main__":
    # Initialize old and new lists of Huffman trees
    old_huffman_trees = [HuffmanTree('a', 5), HuffmanTree('b', 3)]
    new_huffman_trees = [HuffmanTree('c', 2), HuffmanTree('d', 4)]

    # Create a Huffman heap
    huffman_heap = HuffmanHeap(old_huffman_trees, new_huffman_trees)

    # Enqueue a new Huffman tree
    huffman_heap.enqueue(HuffmanTree('e', 1))

    # Dequeue the smallest item
    smallest_tree = huffman_heap.dequeue()
    if smallest_tree:
        print("Smallest tree dequeued:", smallest_tree.value, smallest_tree.freq)
