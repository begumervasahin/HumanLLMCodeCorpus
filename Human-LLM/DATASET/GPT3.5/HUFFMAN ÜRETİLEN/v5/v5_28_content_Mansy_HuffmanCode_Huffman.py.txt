import heapq
from collections import Counter

class Node:
    """
    Represents a node in the Huffman tree.
    """

    def __init__(self, left, right):
        """
        Initializes a Node object with left and right child nodes.

        Args:
            left (Node): The left child node.
            right (Node): The right child node.
        """
        self.left = left
        self.right = right

    def walk(self, code, acc):
        """
        Traverses the Huffman tree to assign binary codes to characters.

        Args:
            code (dict): The dictionary to store the binary codes.
            acc (str): The accumulated binary code.
        """
        self.left.walk(code, acc + '0')
        self.right.walk(code, acc + '1')

class Leaf:
    """
    Represents a leaf node in the Huffman tree.
    """

    def __init__(self, char):
        """
        Initializes a Leaf object with a character.

        Args:
            char (str): The character associated with this leaf node.
        """
        self.char = char

    def walk(self, code, acc):
        """
        Assigns a binary code to the character.

        Args:
            code (dict): The dictionary to store the binary codes.
            acc (str): The accumulated binary code.
        """
        code[self.char] = acc or '0'

def huffman_encode(string):
    """
    Generates Huffman codes for characters in a string.

    Args:
        string (str): The input string to encode.

    Returns:
        dict: A dictionary mapping characters to their Huffman codes.
    """
    frequencies = Counter(string)
    heap = [(freq, Leaf(char)) for char, freq in frequencies.items()]
    heapq.heapify(heap)

    while len(heap) > 1:
        freq1, left = heapq.heappop(heap)
        freq2, right = heapq.heappop(heap)
        heapq.heappush(heap, (freq1 + freq2, Node(left, right)))

    code = {}
    if heap:
        [(_freq, root)] = heap
        root.walk(code, '')

    return code

def huffman_decode(encoded, code):
    """
    Decodes Huffman-encoded string using provided Huffman codes.

    Args:
        encoded (str): The Huffman-encoded string.
        code (dict): The Huffman codes for characters.

    Returns:
        str: The decoded string.
    """
    decode = ""
    current_code = ""
    for bit in encoded:
        current_code += bit
        for char, huff_code in code.items():
            if huff_code == current_code:
                decode += char
                current_code = ""
                break
    return decode

    code = huffman_encode(string)
    encoded = ''.join(code[ch] for ch in string)

        file.write(bytes(int(encoded[i:i + 8], 2) for i in range(0, len(encoded), 8)))

    print('\nEncoded string in binary:', int(encoded, 2))

    decoded = huffman_decode(encoded, code)
