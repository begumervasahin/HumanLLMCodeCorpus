# Import necessary libraries
import numpy as np
from PIL import Image
from scipy.fftpack import dct

# Global constants
Z = np.array([
    [16, 11, 10, 16, 24, 40, 51, 61],
    [12, 12, 14, 19, 26, 58, 60, 55],
    [14, 13, 16, 24, 40, 57, 69, 56],
    [14, 17, 22, 29, 51, 87, 80, 62],
    [18, 22, 37, 56, 68, 109, 103, 77],
    [24, 35, 55, 64, 81, 104, 113, 92],
    [49, 64, 78, 87, 103, 121, 120, 101],
    [72, 92, 95, 98, 112, 100, 103, 99]
])

# Global variables for temporary storage
temporal = {}
frecuencia = {}

# Function to perform 2D Discrete Cosine Transform (DCT)
def dct2(a):
    b = a - 128
    c = dct(dct(b, axis=0, norm='ortho'), axis=1, norm='ortho')
    return np.fix(c / Z)

# Function to perform zigzag scanning
def zigzag(matrix):
    global temporal
    con = 0
    rows, columns = 8, 8
    aux = np.zeros((1, 64))

    solution = [[] for _ in range(rows + columns - 1)]
    for i in range(rows):
        for j in range(columns):
            sum_ij = i + j
            if sum_ij % 2 == 0:
                solution[sum_ij].insert(0, matrix[i][j])
            else:
                solution[sum_ij].append(matrix[i][j])

    solution.reverse()

    for i in solution:
        for j in i:
            if j == -0.0:
                j = abs(j)
                aux[0, con] = j
            else:
                aux[0, con] = j
            con += 1

    indice = next((i for i, value in enumerate(aux[0]) if value != 0), None)

    for i in range(indice, 64):
        temporal[con] = aux[0, i]
        frecuencia[aux[0, i]] = aux[0, i]
        con += 1

# Load the image
image = Image.open("lena.jpg").convert('L')
image.show()

# Convert the image to numpy array
image_array = np.asarray(image, dtype=np.float32)

# Subtract 128 from the image array
image_array_subtracted = image_array - 128
Image.fromarray(image_array_subtracted.astype(np.uint8)).save("restada.jpg")
I = Image.open("restada.jpg")
I.show()

# Initialize a matrix of zeros of size 256x256
im2 = np.zeros((256, 256))

# Perform 2D Discrete Cosine Transform (DCT) and zigzag scanning on 8x8 blocks
for i in range(0, image_array.shape[0], 8):
    for j in range(0, image_array.shape[1], 8):
        im2[i:(i + 8), j:(j + 8)] = dct2(image_array[i:(i + 8), j:(j + 8)])
        zigzag(im2[i:(i + 8), j:(j + 8)])


# Save the matrix after DCT processing to a text file
np.savetxt("dct.txt", im2)

# Compute probabilities and save to a file
elementos = list(temporal.values())
probabilidades = {i: float(elementos.count(i)) / len(elementos) for i in frecuencia}

with open("result.txt", "w") as file:
    for key in frecuencia:
        file.write(f"{key}\t{probabilidades.get(key, key)}\n")

print("Image processing and probability computation completed successfully.")
print("A file of probabilities has been generated.")
print("The text file containing the result of JPEG processing has been generated.")
