class Wing:
    """
    Represents a wing with a flying ratio.
    """

    def __init__(self, ratio):
        """
        Initializes a Wing object with a flying ratio.

        Args:
            ratio (float): The flying ratio.
        """
        self.ratio = ratio

    def fly(self):
        """
        Simulates flying based on the wing's flying ratio.
        """
        if self.ratio > 1:
            print("Weee, this is fun!")
        elif self.ratio == 1:
            print("This is hard work, but I'm flying")
        else:
            print("I think I'll just walk")


class Duck:
    """
    Represents a generic duck.
    """

    def __init__(self):
        """
        Initializes a Duck object with default attributes.
        """
        self._wing = Wing(1.8)

    def walk(self):
        """
        Simulates walking for a duck.
        """
        print("Waddle waddle waddle")

    def swim(self):
        """
        Simulates swimming for a duck.
        """
        print("Come on in the water is lovely")

    def quack(self):
        """
        Simulates quacking for a duck.
        """
        print("Quack Quack")

    def fly(self):
        """
        Simulates flying for a duck.
        """
        self._wing.fly()

class Flock:
    """
    Represents a flock of ducks.
    """

    def __init__(self):
        """
        Initializes a Flock object with an empty list of ducks.
        """
        self.flock = []

    def add_duck(self, duck: Duck) -> None:
        """
        Adds a duck to the flock.

        Args:
            duck (Duck): The duck to add to the flock.

        Raises:
            TypeError: If the object passed is not a Duck.
        """
        if isinstance(duck, Duck):
            self.flock.append(duck)
        else:
            raise TypeError(f"Cannot add duck, are you sure it is not a {type(duck).__name__}?")

    def migrate(self):
        """
        Simulates the migration of the flock of ducks.

        Raises:
            AttributeError: If a duck encounters an attribute error during flight.
        """
        for duck in self.flock:
            try:
                duck.fly()
                raise AttributeError("Testing exception handling in migrate")
            except AttributeError:
                print("One duck down")
                raise
