import numpy as np
from scipy.misc import imread, imresize
import queue

class Node:
    def __init__(self):
        self.prob = None
        self.code = None
        self.data = None
        self.left = None
        self.right = None

def rgb_to_gray(img):
    # Convert RGB image to grayscale
    gray_img = np.rint(img[:, :, 0] * 0.2989 + img[:, :, 1] * 0.5870 + img[:, :, 2] * 0.1140)
    gray_img = gray_img.astype(int)
    return gray_img

def get_two_smallest(data):
    # Find indices and values of two smallest elements in the data array
    smallest_idx = second_smallest_idx = 0
    smallest_val = second_smallest_val = 1
    for idx, element in enumerate(data):
        if element < smallest_val:
            second_smallest_val = smallest_val
            second_smallest_idx = smallest_idx
            smallest_val = element
            smallest_idx = idx
        elif element < second_smallest_val and element != smallest_val:
            second_smallest_val = element
            second_smallest_idx = idx
    return smallest_idx, smallest_val, second_smallest_idx, second_smallest_val

def build_huffman_tree(probabilities):
    # Build Huffman tree using priority queue
    priority_queue = queue.PriorityQueue()
    for color, probability in enumerate(probabilities):
        leaf = Node()
        leaf.data = color
        leaf.prob = probability
        priority_queue.put(leaf)

    while priority_queue.qsize() > 1:
        new_node = Node()
        left_child = priority_queue.get()
        right_child = priority_queue.get()
        new_node.left = left_child
        new_node.right = right_child
        new_probability = left_child.prob + right_child.prob
        new_node.prob = new_probability
        priority_queue.put(new_node)

    return priority_queue.get()

def traverse_huffman_tree(root_node, tmp_array, output_file):
    # Traverse the Huffman tree to generate codes and write them to the output file
    if root_node.left is not None:
        tmp_array[traverse_huffman_tree.count] = 1
        traverse_huffman_tree.count += 1
        traverse_huffman_tree(root_node.left, tmp_array, output_file)
        traverse_huffman_tree.count -= 1

    if root_node.right is not None:
        tmp_array[traverse_huffman_tree.count] = 0
        traverse_huffman_tree.count += 1
        traverse_huffman_tree(root_node.right, tmp_array, output_file)
        traverse_huffman_tree.count -= 1

    else:
        huffman_code_length = traverse_huffman_tree.count
        huffman_code = ''.join(str(bit) for bit in tmp_array[1:huffman_code_length])
        color = str(root_node.data)
        output_line = f"{color} {huffman_code}\n"
        output_file.write(output_line)

    return

# Read and preprocess the image
img = imread('tiger.bmp')
img = imresize(img, 10)
gray_img = rgb_to_gray(img)

# Compute histogram and probabilities
histogram = np.bincount(gray_img.ravel(), minlength=256)
probabilities = histogram / np.sum(histogram)

# Build Huffman tree
root_node = build_huffman_tree(probabilities)

# Traverse Huffman tree to generate codes and write them to the output file
tmp_array = np.ones([64], dtype=int)
traverse_huffman_tree.count = 0
output_file = open('codes.txt', 'w')
traverse_huffman_tree(root_node, tmp_array, output_file)

# Calculate compression ratio
input_bits = img.shape[0] * img.shape[1] * 8
compression_ratio = (1 - np.sum(traverse_huffman_tree.output_bits * histogram) / input_bits) * 100
print('Compression ratio:', compression_ratio, 'percent')

# Close the output file
output_file.close()
