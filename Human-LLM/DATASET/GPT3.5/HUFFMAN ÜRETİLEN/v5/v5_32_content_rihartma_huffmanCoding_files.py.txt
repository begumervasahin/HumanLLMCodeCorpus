class Files:
    def load_file(self, name):
        """
        Load text from a file.

        Args:
            name (str): The name of the file to load.

        Returns:
            str: The text read from the file.
        """
        with open(name, 'r') as file:
            return file.read()

    def save_table(self, text, name):
        """
        Save text to a file.

        Args:
            text (str): The text to save.
            name (str): The name of the file to save to.
        """
        with open(name, 'w') as file:
            file.write(text)

    def save_file(self, name, text, codes):
        """
        Save encoded text to a file.

        Args:
            name (str): The name of the file to save to.
            text (str): The original text to encode and save.
            codes (dict): A dictionary containing character codes.

        Note:
            Characters in the text must have corresponding codes in the codes dictionary.
        """
        binary_text = ''.join(codes[char] for char in text)

        file_text = ""
        for i in range(0, len(binary_text) - 8, 8):
            file_text += chr(int(binary_text[i:i + 8], 2))

        if len(binary_text) % 8 != 0:
            file_text += "~~~" + binary_text[len(binary_text) - 1 - len(binary_text) % 8: len(binary_text)]

        with open(name, 'w') as file:
            file.write(file_text)

    def load_table(self, name):
        """
        Load character codes from a file.

        Args:
            name (str): The name of the file containing character codes.

        Returns:
            dict: A dictionary containing character codes.
        """
        with open(name, "r") as file:
            text = file.read()

        codes = {}
        pairs = text.split(",")
        pairs = [pair.strip() for pair in pairs if pair.strip()]  # Remove empty pairs and whitespace

        for pair in pairs:
            char, code = pair.split("-")
            codes[code] = char

        return codes
