class HuffmanNode:
    """
    Represents a node in a Huffman tree.

    Attributes:
    - symbol: The symbol associated with the node.
    - left: The left child node.
    - right: The right child node.
    - number: A unique number assigned to the node.
    """

    def __init__(self, symbol=None, left=None, right=None) -> None:
        self.symbol = symbol
        self.left = left
        self.right = right
        self.number = None

    def __eq__(self, other: object) -> bool:
        """
        Check if two Huffman nodes are equal.

        Returns:
        - True if the two nodes are equal, False otherwise.
        """
        return (
            type(self) == type(other)
            and self.symbol == other.symbol
            and self.left == other.left
            and self.right == other.right
        )

    def __lt__(self, other: object) -> bool:
        """
        Compare two Huffman nodes.

        Always returns False for this implementation.

        Returns:
        - False always.
        """
        return False

    def __repr__(self) -> str:
        """
        Return a string representation of the HuffmanNode object.
        """
        return f'HuffmanNode({self.symbol}, {self.left}, {self.right})'

    def is_leaf(self) -> bool:
        """
        Check if the node is a leaf node.

        Returns:
        - True if the node is a leaf node, False otherwise.
        """
        return not self.left and not self.right


class ReadNode:
    """
    Represents a node used for reading data.

    Attributes:
    - l_type: The type of data in the left child node.
    - l_data: The data stored in the left child node.
    - r_type: The type of data in the right child node.
    - r_data: The data stored in the right child node.
    """

    def __init__(self, l_type: str, l_data: object, r_type: str, r_data: object) -> None:
        self.l_type = l_type
        self.l_data = l_data
        self.r_type = r_type
        self.r_data = r_data

    def __repr__(self) -> str:
        """
        Return a string representation of the ReadNode object.
        """
        return f'ReadNode({self.l_type}, {self.l_data}, {self.r_type}, {self.r_data})'
