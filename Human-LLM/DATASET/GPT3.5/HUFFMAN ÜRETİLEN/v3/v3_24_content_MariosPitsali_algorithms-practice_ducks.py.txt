class Wing:
    """
    Represents the wing of a bird.
    """

    def __init__(self, aspect_ratio):
        """
        Initializes a Wing object with the given aspect ratio.
        """
        self.aspect_ratio = aspect_ratio

    def fly(self):
        """
        Simulates flying based on the wing aspect ratio.
        """
        if self.aspect_ratio > 1:
            print("Weee, this is fun!")
        elif self.aspect_ratio == 1:
            print("This is hard work, but I'm flying")
        else:
            print("I think I'll just walk")


class Duck:
    """
    Represents a duck.
    """

    def __init__(self, wing_aspect_ratio=1.8):
        """
        Initializes a Duck object with a Wing instance.
        """
        self.wing = Wing(wing_aspect_ratio)

    def walk(self):
        """
        Simulates walking.
        """
        print("Waddle waddle waddle")

    def swim(self):
        """
        Simulates swimming.
        """
        print("Come on in the water is lovely")

    def quack(self):
        """
        Simulates quacking.
        """
        print("Quack Quack")

    def fly(self):
        """
        Simulates flying by calling the fly method of the Wing instance.
        """
        self.wing.fly()

class Flock:
    """
    Represents a flock of ducks.
    """

    def __init__(self):
        """
        Initializes an empty list to store ducks in the flock.
        """
        self.ducks = []

    def add_duck(self, duck: Duck) -> None:
        """
        Adds a duck to the flock.
        """
        if isinstance(duck, Duck):
            self.ducks.append(duck)
        else:
            raise TypeError(f"Cannot add duck, expected Duck instance, got {type(duck).__name__}")

    def migrate(self):
        """
        Simulates migration by having each duck in the flock attempt to fly.
        """
        for duck in self.ducks:
            duck.fly()
            print("One duck down")
