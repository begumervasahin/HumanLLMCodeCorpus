class HuffmanTree:
    def __init__(self, value, freq):
        """
        Initialize a Huffman Tree node with a value and its frequency.
        :param value: The value of the node.
        :param freq: The frequency of occurrence of the value.
        """
        self.value = value
        self.freq = freq

    def get_freq(self):
        """
        Get the frequency of the node.
        :return: Frequency of the node.
        """
        return self.freq


class HuffmanHeap:
    """
    HuffmanHeap class represents a priority queue for Huffman trees.
    """

    def __init__(self, old_trees, new_trees=None):
        """
        Initialize the HuffmanHeap with old and new lists of Huffman trees.
        :param old_trees: Old list of Huffman trees, sorted.
        :param new_trees: New list of Huffman trees, initially empty.
        """
        self.old_trees = old_trees
        self.new_trees = new_trees if new_trees is not None else []

    def enqueue(self, tree):
        """
        Enqueue a Huffman tree into the new list.
        :param tree: A Huffman tree to be enqueued.
        :return: None
        """
        self.new_trees.append(tree)

    def dequeue(self):
        """
        Dequeue the smallest item from the heap.
        :return: The smallest item; if the heap is empty, return None.
        """
        if not self.old_trees and self.new_trees:
            smallest_tree = self.new_trees.pop(0)
        elif not self.new_trees and self.old_trees:
            smallest_tree = self.old_trees.pop(0)
        elif self.old_trees and self.new_trees:
            if self.old_trees[0].get_freq() >= self.new_trees[0].get_freq():
                smallest_tree = self.new_trees.pop(0)
            else:
                smallest_tree = self.old_trees.pop(0)
        else:
            print("Both old and new lists are empty!")
            return None
        return smallest_tree


# Example usage
if __name__ == "__main__":
    # Initialize old and new lists of Huffman trees
    old_trees = [HuffmanTree('a', 5), HuffmanTree('b', 3)]
    new_trees = [HuffmanTree('c', 2), HuffmanTree('d', 4)]

    # Create a Huffman heap
    huffman_heap = HuffmanHeap(old_trees, new_trees)

    # Enqueue a new Huffman tree
    huffman_heap.enqueue(HuffmanTree('e', 1))

    # Dequeue the smallest item
    smallest_tree = huffman_heap.dequeue()
    if smallest_tree:
        print("Smallest tree dequeued:", smallest_tree.value, smallest_tree.freq)
