class Node:
    def __init__(self, value, probability):
        self.val = value
        self.probability = probability
        self.code = ''

    def __str__(self):
        return f"({self.val})"

    def set_code(self, code):
        self.code = code

    def get_code(self):
        return self.code

    def get_prob(self):
        return self.probability

    def get_val(self):
        return self.val

class NodeJoin:
    def __init__(self, node1, node2):
        self.node1 = node1
        self.node2 = node2
        self.probability = node1.get_prob() + node2.get_prob()
        self.code = ''

    def __str__(self):
        return f"({self.node1},{self.node2})"

    def set_code(self, code):
        self.code = code

    def get_code(self):
        return self.code

    def get_prob(self):
        return self.probability

    def get_val(self):
        return self.node1.get_val() + self.node2.get_val()

# Function to split nodes recursively and set codes
def split_nodes(node):
    code = node.get_code()
    node1, node2 = node.node1, node.node2
    node1.set_code(code + "1")
    node2.set_code(code + "0")
    if isinstance(node1, NodeJoin):
        split_nodes(node1)
    else:
        final_codes.append(node1)
    if isinstance(node2, NodeJoin):
        split_nodes(node2)
    else:
        final_codes.append(node2)

# Input and processing of nodes
def input_nodes():
    nodes = []
    while True:
        n = input("Enter node (value,probability): ")
        if not n:
            break
        if "," not in n:
            print("Invalid input format. Please provide value and probability separated by comma.")
            continue
        value, probability = n.split(",")
        probability = float(probability)
        nodes.append(Node(value, probability))
    return nodes

def sort_nodes(nodes):
    sorted_nodes = sorted(nodes, key=lambda x: x.get_prob(), reverse=True)
    if sorted_nodes != nodes:
        print("Nodes not sorted! Sorting...")
        print("New nodes:")
        for node in sorted_nodes:
            print(f"  {node}")
    return sorted_nodes

def create_node_joins(nodes):
    while len(nodes) > 1:
        new_nodes = nodes[:-2]
        new_node = NodeJoin(nodes[-1], nodes[-2])
        for i, obj in enumerate(new_nodes):
            if new_node.get_prob() >= obj.get_prob():
                new_nodes.insert(i, new_node)
                break
        else:
            new_nodes.append(new_node)
        nodes = new_nodes
    return nodes[0]

def print_final_codes(final_codes):
    print("\nFinal Codes:")
    for node in final_codes:
        print(f"Node {node} has code {node.get_code()}")

if __name__ == "__main__":
    nodes = input_nodes()
    if not nodes:
        print("No nodes provided. Exiting.")
    else:
        nodes = sort_nodes(nodes)
        final_node = create_node_joins(nodes)
        final_codes = []
        final_node.set_code("")
        if isinstance(final_node, Node):
            final_codes.append(final_node)
        else:
            split_nodes(final_node)
        final_codes.sort(key=lambda x: x.get_val())
        print_final_codes(final_codes)
