from collections import Counter

# Constants
TRAIN_SIZE = 6000
TEST_SIZE = 15000
TRAIN_FILE_NAME = "news_train.txt"
TEST_FILE_NAME = "news_test.txt"

class TextClassifier:
    def __init__(self):
        self.categories = []
        self.contents = []
        self.names = []
        self.total_words = {}
        self.vocabulary = []

    def read_training_data(self, file_name):
        """Read training data from a file."""
        with open(file_name, encoding='utf-8') as f:
            articles = f.readlines()
            for article in articles:
                category, text = article.split("\t", 1)
                self.categories.append(category)
                self.contents.append(text.strip())

    def get_unique_categories(self):
        """Get unique category names."""
        self.names = list(set(self.categories))

    def index_select(self, category_name):
        """Select indices of articles with a given category."""
        return [i for i, cat in enumerate(self.categories) if cat == category_name][:TRAIN_SIZE]

    def common_words(self, category, test_dict):
        """Count common words between test data and training data of a category."""
        test_counts = Counter(test_dict)
        test_counts.update({word: 0 for word in test_counts.keys()})  # Initialize counts for unseen words
        for i in self.index_select(category):
            train_dict = Counter(self.contents[i].split())
            for key in train_dict:
                if key in test_counts:
                    test_counts[key] += train_dict[key]
        return test_counts

    def prior_probability(self, category):
        """Calculate prior probability of a category."""
        return len(self.index_select(category)) / len(self.categories)

    def words_and_vocabulary(self, category):
        """Count total words and vocabulary of a category."""
        total_words = 0
        vocabulary = set()
        for i in self.index_select(category):
            train_dict = Counter(self.contents[i].split())
            total_words += len(train_dict)
            vocabulary.update(train_dict.keys())
        return total_words, list(vocabulary)

    def train(self):
        """Train the text classifier."""
        self.read_training_data(TRAIN_FILE_NAME)
        self.get_unique_categories()
        for category_name in self.names:
            total_words, vocabulary = self.words_and_vocabulary(category_name)
            self.total_words[category_name] = total_words
            self.vocabulary.extend(vocabulary)
        self.vocabulary = list(set(self.vocabulary))

    def classify(self, test_data):
        """Classify test data."""
        predictions = []
        for test_article in test_data:
            test_dict = Counter(test_article.split())
            prediction_probs = []
            for category_name in self.names:
                test_counts = self.common_words(category_name, test_dict)
                conditional_probs = [10000 * (test_counts[word] + 1) / (self.total_words[category_name] + len(self.vocabulary))
