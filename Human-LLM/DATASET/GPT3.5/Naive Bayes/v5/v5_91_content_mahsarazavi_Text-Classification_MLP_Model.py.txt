import keras
from keras.datasets import reuters
from keras.preprocessing.text import Tokenizer
from keras.models import Sequential
from keras.layers import Dense, Dropout, Activation

# Load the Reuters dataset
(x_train, y_train), (x_test, y_test) = reuters.load_data(num_words=None, test_split=0.2)

# Get the word index
word_index = reuters.get_word_index(path="reuters_word_index.json")

# Define the number of classes and create index_to_word dictionary
num_classes = max(y_train) + 1
index_to_word = {value: key for key, value in word_index.items()}

# Print an example to see the original text and its label
print(' '.join([index_to_word.get(x - 3, '?') for x in x_train[0]]))
print('Label:', y_train[0])

# Tokenize the data
max_words = 10000
tokenizer = Tokenizer(num_words=max_words)
x_train = tokenizer.sequences_to_matrix(x_train, mode='binary')
x_test = tokenizer.sequences_to_matrix(x_test, mode='binary')

# Convert labels to one-hot encoding
y_train = keras.utils.to_categorical(y_train, num_classes)
y_test = keras.utils.to_categorical(y_test, num_classes)

# Build the model
model = Sequential()
model.add(Dense(512, input_shape=(max_words,)))
model.add(Activation('relu'))
model.add(Dropout(0.5))
model.add(Dense(num_classes))
model.add(Activation('softmax'))

# Compile the model
model.compile(loss='categorical_crossentropy', optimizer='adam', metrics=['accuracy'])

# Print the model metrics names
print(model.metrics_names)

# Train the model
batch_size = 32
epochs = 3
history = model.fit(x_train, y_train, batch_size=batch_size, epochs=epochs, verbose=1, validation_split=0.1)

# Evaluate the model on the test data
score = model.evaluate(x_test, y_test, batch_size=batch_size, verbose=1)
print('Test loss:', score[0])
print('Test accuracy:', score[1])
