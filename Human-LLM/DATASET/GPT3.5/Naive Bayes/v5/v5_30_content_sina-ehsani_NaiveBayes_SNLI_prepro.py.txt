import json
import re
import numpy as np
from nltk.stem import PorterStemmer
from nltk.tokenize import word_tokenize
from nltk.stem import WordNetLemmatizer
from data_utils import savedata

# Initialize lemmatizer and stemmer
lemmatizer = WordNetLemmatizer()
ps = PorterStemmer()

def extract_tokens_from_binary_parse(parse):
    """Extract tokens from a binary parse"""
    return parse.replace('(', ' ').replace(')', ' ').replace('-LRB-', '(').replace('-RRB-', ')').split()

def yield_examples(file_path, skip_no_majority=True, limit=None):
    """Yield examples from a file"""
    with open(file_path) as f:
        for i, line in enumerate(f):
            if limit and i > limit:
                break
            data = json.loads(line)
            label = data['gold_label']
            s1 = ' '.join(extract_tokens_from_binary_parse(data['sentence1_binary_parse']))
            s2 = ' '.join(extract_tokens_from_binary_parse(data['sentence2_binary_parse']))
            if skip_no_majority and label == '-':
                continue
            yield (label, s1, s2)

def clean_data(sentence_list):
    """Clean sentences by removing punctuation, stop-words, lower-casing, and stemming"""
    clean_sentences = []
    for sentence in sentence_list:
        sentence = re.sub(r"[^a-zA-Z]", " ", sentence)
        words = sentence.lower().split()
        words = [ps.stem(w) for w in words]
        clean_sentences.append(" ".join(words))
    return clean_sentences

def preprocess_file(file_path, limit=None):
    """Preprocess a file containing SNLI data"""
    raw_data = list(yield_examples(file_path, limit=limit))
    left = [s1 for _, s1, s2 in raw_data]
    right = [s2 for _, s1, s2 in raw_data]
    left = clean_data(left)
    right = clean_data(right)
    LABELS = {'contradiction': 0, 'neutral': 1, 'entailment': 2}
    Y = np.array([LABELS[l] for l, _, _ in raw_data])
    return left, right, Y

def main():
    # Preprocess train, test, and dev datasets
    train_data = preprocess_file('snli_1.0/snli_1.0_train.jsonl')
    test_data = preprocess_file('snli_1.0/snli_1.0_test.jsonl')
    dev_data = preprocess_file('snli_1.0/snli_1.0_dev.jsonl')

    # Save preprocessed data
    savedata(test_data, "test_stem")
    savedata(train_data, "train_stem")
    savedata(dev_data, "dev_stem")

if __name__ == '__main__':
    main()
