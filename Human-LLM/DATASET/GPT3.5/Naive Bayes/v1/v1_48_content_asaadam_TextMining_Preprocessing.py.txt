import string
import re
from Sastrawi.Stemmer.StemmerFactory import StemmerFactory
import collections

class Preprocessing:
    stopword = open("stopword-list.txt", "r").read().split('\n')

    @staticmethod
    def cleaning(text):
        # Remove links
        remove_link = re.sub('[\S]+\.(net|com|org|info|edu|gov|uk|de|ca|jp|fr|au|us|ru|ch|it|nel|se|no|es|mil)[\S]*',' ', text)
        # Replace punctuation with spaces
        replace_punctuation = remove_link.maketrans(string.punctuation, ' ' * len(string.punctuation))
        text = remove_link.translate(replace_punctuation)
        # Remove numbers
        cleaning = text.translate(str.maketrans('', '', '1234567890'))
        # Replace multiple spaces with a single space
        return re.sub('\s+', ' ', cleaning).strip()

    @staticmethod
    def case_folding(text):
        # Convert text to lowercase
        return str.casefold(text)

    @staticmethod
    def tokenisasi(text):
        # Tokenize text
        return text.split()

    @staticmethod
    def filtering(text):
        # Filter out stop words
        return [text for text in text if text not in Preprocessing.stopword]

    @staticmethod
    def type(text):
        # Remove duplicate words while preserving order
        return list(collections.OrderedDict([(word, None) for word in text]))

    @staticmethod
    def stemming(text):
        # Perform stemming using Sastrawi stemmer
        factory = StemmerFactory()
        stemmer = factory.create_stemmer()
        return [stemmer.stem(x) for x in text]

    @staticmethod
    def all_in_one(text):
        # Perform all preprocessing steps
        return Preprocessing.filtering(
                Preprocessing.stemming(
                    Preprocessing.type(
                        Preprocessing.tokenisasi(
                            Preprocessing.case_folding(
                                Preprocessing.cleaning(text)
                            )
                        )
                    )
                )
            )

    @staticmethod
    def all_in_one_without_type(text):
        # Perform preprocessing without removing duplicate words
        return Preprocessing.filtering(
                Preprocessing.stemming(
                    Preprocessing.tokenisasi(
                        Preprocessing.case_folding(
                            Preprocessing.cleaning(text)
                        )
                    )
                )
            )

# Example usage:
text = "This is an example text with some numbers 123 and links http://example.com."
processed_text = Preprocessing.all_in_one(text)
print("Processed text with type removal:", processed_text)

processed_text_without_type = Preprocessing.all_in_one_without_type(text)
print("Processed text without type removal:", processed_text_without_type)
