from sklearn.metrics import confusion_matrix, accuracy_score, average_precision_score, roc_auc_score
from sklearn.naive_bayes import MultinomialNB

def multinomial_naive_bayes(x_train_vectorized, y_train, x_test_vectorized, y_test, new_data_vectorized):
    """
    Perform Multinomial Naive Bayes classification.

    Parameters:
    - x_train_vectorized: Vectorized training features.
    - y_train: Training labels.
    - x_test_vectorized: Vectorized test features.
    - y_test: Test labels.
    - new_data_vectorized: Vectorized new data for prediction.

    Returns:
    - prediction: Prediction result for new data.
    """
    print("Multinomial Naive Bayes")

    # Initialize Multinomial Naive Bayes classifier
    mnb = MultinomialNB()

    # Train the classifier
    mnb.fit(x_train_vectorized, y_train)

    # Predict test data
    y_pred_class = mnb.predict(x_test_vectorized)

    # Evaluate the model
    print("Confusion Matrix:")
    print(confusion_matrix(y_test, y_pred_class))
    print('Accuracy Score:', accuracy_score(y_test, y_pred_class))
    print('ROC and AUC:', roc_auc_score(y_test, y_pred_class))
    print('Average Precision Score:', average_precision_score(y_test, y_pred_class))

    # Predict new data
    prediction = "Positive" if mnb.predict(new_data_vectorized) == [1] else "Negative"
    return prediction
