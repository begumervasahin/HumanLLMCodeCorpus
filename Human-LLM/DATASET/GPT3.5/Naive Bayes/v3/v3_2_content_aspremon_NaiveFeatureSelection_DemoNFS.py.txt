import numpy as np
from sklearn.svm import LinearSVC
from sklearn.datasets import fetch_20newsgroups
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn import metrics
from naive_feature_selection import NaiveFeatureSelection
from sklearn.pipeline import Pipeline
from sklearn.model_selection import GridSearchCV

def load_data(categories, remove):
    print("Loading 20 newsgroups dataset for categories:")
    print(categories if categories else "all")
    print()

    data_train = fetch_20newsgroups(subset='train', categories=categories, shuffle=True, random_state=42, remove=remove)
    data_test = fetch_20newsgroups(subset='test', categories=categories, shuffle=True, random_state=42, remove=remove)
    y_train, y_test = data_train.target, data_test.target

    return data_train, data_test, y_train, y_test

def extract_features(vectorizer, data):
    features = vectorizer.transform(data)
    print("n_samples: %d, n_features: %d" % features.shape)
    print()
    return features

def print_selected_features(nfs, clfsv, feature_names):
    print('Space features:')
    print([feature_names[np.nonzero(nfs.mask_)[0][i]] for i in range(nfs.k) if clfsv.coef_[0][i] >= 0])
    print()

    print('Med features:')
    print([feature_names[np.nonzero(nfs.mask_)[0][i]] for i in range(nfs.k) if clfsv.coef_[0][i] < 0])
    print()

def main():
    print("Testing Naive Feature Selection (NFS) ...")

    # Define categories and data subsets
    categories = ['sci.med', 'sci.space']
    remove = ('headers', 'footers', 'quotes')

    data_train, data_test, y_train, y_test = load_data(categories, remove)

    # Extract features from the training data using TfidfVectorizer
    print("Extracting features from the training data using a sparse vectorizer")
    vectorizer = TfidfVectorizer(sublinear_tf=True, max_df=0.5, stop_words='english')
    X_train = extract_features(vectorizer, data_train.data)

    # Extract features from the test data using the same vectorizer
    print("Extracting features from the test data using the same vectorizer")
    X_test = extract_features(vectorizer, data_test.data)

    # Initialize and fit NaiveFeatureSelection
    feature_names = vectorizer.get_feature_names()
    kv = 100
    nfs = NaiveFeatureSelection(k=kv, alpha=1e-4)
    X_new = nfs.fit_transform(X_train, y_train)
    clfsv = LinearSVC(random_state=0, tol=1e-5)
    clfsv.fit(X_new, y_train == 1)
    X_testnew = nfs.transform(X_test)
    y_pred_NFS = clfsv.predict(X_testnew)
    score_nfs = metrics.accuracy_score(y_test == 1, y_pred_NFS)
    print("NFS accuracy:\t%0.3f" % score_nfs)
    print()

    print_selected_features(nfs, clfsv, feature_names)

    # Build a pipeline with NaiveFeatureSelection and LinearSVC
    clf = Pipeline([
        ('feature_selection', NaiveFeatureSelection(k=100)),
        ('classification', LinearSVC())
    ])
    clf.fit(X_train, y_train)
    y_pred_pp = clf.predict(X_test)
    score_pp = metrics.accuracy_score(y_test == 1, y_pred_pp)
    print("Pipeline accuracy:\t%0.3f" % score_pp)
    print()

    # Perform GridSearchCV to find the best parameters for NaiveFeatureSelection
    parameters = {'feature_selection__k': [10, 100, 500],
                  'feature_selection__alpha': [1e-10, 1e-5, 1e-3, 1e-2, 1e-1, 1, 10, 100]}
    svcp = Pipeline([
        ('feature_selection', NaiveFeatureSelection()),
        ('classification', LinearSVC())
    ])
    clf = GridSearchCV(svcp, parameters, cv=5)
    clf.fit(X_train, y_train)
    print("Best cross validated k: " + str(clf.best_params_['feature_selection__k']))
    print("Best cross validated alpha: " + str(clf.best_params_['feature_selection__alpha']))

if __name__ == "__main__":
    main()
