plt.close('all')

# Data paths
DATA = "DATA_001_House"
path_data = "data1_filtered_wifis/" + DATA + "/"
path_parts = ["data2_divided_parts/" + DATA + "/part{:02d}/".format(i) for i in range(1, 11)]

# Read data and shuffle
def read_and_shuffle_data(path):
    df = pd.read_csv(path, sep=" ")
    df = df.sample(frac=1)  # Shuffle the data
    return df

# Divide data into 10 parts and save
def divide_and_save_data(data, parts_paths):
    num_intervals = int(0.1 * len(data))
    for i, part_path in enumerate(parts_paths, start=1):
        part_data = data.iloc[(i - 1) * num_intervals: i * num_intervals].reset_index(drop=True)
        part_data.to_csv(part_path + "R%.3d.csv" % i, sep=' ', index=False)
        print("Saving:", part_path + "R%.3d.csv" % i)

# Read and process data for each region
for file_path in glob.glob(path_data + "*.csv"):
    region_data = read_and_shuffle_data(file_path)
    divide_and_save_data(region_data, path_parts)
