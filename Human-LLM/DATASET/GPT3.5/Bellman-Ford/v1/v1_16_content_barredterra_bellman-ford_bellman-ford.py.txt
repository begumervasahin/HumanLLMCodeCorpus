import csv

def bellman_ford(edges, distances):
    log = []
    n = len(distances)
    for i in range(1, n - 1):
        for edge in edges:
            if distances[edge[0]] + edge[2] < distances[edge[1]]:
                distances[edge[1]] = distances[edge[0]] + edge[2]
            log.append(get_state(edge, distances))
        if i == 2:
            return distances, log

def get_state(edge, distances):
    edge_str = f'({edge[0]}-{edge[1]})'
    state = [edge_str] + [distances[vertex] for vertex in sorted(distances.keys())]
    return state

def save_csv(logs, vertices):
    with open('bf.csv', 'w', newline='') as csvfile:
        logwriter = csv.writer(csvfile)
        logwriter.writerow(['edge'] + vertices)
        logwriter.writerows(logs)
        print('\nlog saved as ' + csvfile.name)

def print_log(log):
    print('Log:\n')
    for l in log:
        print(l)

def main():
    edges = [('a', 'b', 4), ('a', 'c', 1), ('a', 'f', 9), ('b', 'd', 1),
             ('c', 'b', 2), ('c', 'e', 3), ('d', 'e', 1), ('d', 'f', 4),
             ('e', 'b', -2), ('e', 'f', 1)]
    distances = {'a': 0, 'b': 999, 'c': 999, 'd': 999, 'e': 999, 'f': 999}
    result, log = bellman_ford(edges, distances)
    print_log(log)
    vertices = list(distances.keys())
    save_csv(log, vertices)

if __name__ == '__main__':
    main()
