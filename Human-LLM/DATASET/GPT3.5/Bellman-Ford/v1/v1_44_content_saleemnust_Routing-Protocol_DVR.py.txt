import threading
import time
import sys
import json
from socket import socket, AF_INET, SOCK_DGRAM

class Router:
    def __init__(self, router_name, server_ip, server_port, config_file):
        self.router_name = router_name
        self.server_ip = server_ip
        self.server_port = server_port
        self.config_file = config_file
        self.my_neighbors = {}
        self.update_vector = {}
        self.rcv_update = {}
        self.flag = 0

    def Bellmanford(self, received_update):
        for i in received_update:
            for j in received_update[i]:
                if j in self.update_vector.keys():
                    if self.update_vector[j][0] > received_update[i][j][0] + self.update_vector[i][0]:
                        self.update_vector[j][0] = received_update[i][j][0] + self.update_vector[i][0]
                        self.update_vector[j][1] = self.update_vector[i][1] + received_update[i][j][1]
                else:
                    if j == self.router_name:
                        self.update_vector[j] = [0, self.router_name + self.router_name]
                    else:
                        self.update_vector[j] = [received_update[i][j][0] + self.update_vector[i][0], self.update_vector[i][1] + received_update[i][j][1]]

    def client_side(self):
        client_socket = socket(AF_INET, SOCK_DGRAM)
        while True:
            message = json.dumps(self.update_vector)
            data = [self.router_name, message]
            message = json.dumps(data)
            client_socket.sendto(message, (self.server_ip, self.server_port))
            modified_message, server_address = client_socket.recvfrom(2048)
            if modified_message == "Received":
                client_socket.close()
                break

    def active_host(self, port):
        status = -1
        flag_socket = socket(AF_INET, SOCK_DGRAM)
        try:
            data = "hello"
            flag_socket.sendto(data, (self.server_ip, port))
            message, server_address = flag_socket.recvfrom(2048)
            status = 1
        except:
            status = 0
        flag_socket.close()
        return status

    def client_status(self):
        while True:
            for key, val in self.my_neighbors.items():
                response = self.active_host(int(val[0]))
                if response == 1:
                    self.client_side()
            time.sleep(3)

    def router_status(self):
        sorted_update_vector = sorted(self.update_vector.items(), key=lambda t: t[0])
        print("I am Router", self.router_name)
        for i in range(len(sorted_update_vector)):
            if sorted_update_vector[i][0] != self.router_name:
                path = sorted_update_vector[i][1][1]
                new_path = ''.join([path[x] for x in range(len(path) - 1) if path[x] != path[x + 1]]) + path[-1]
                print("Least cost path to router", sorted_update_vector[i][0], ":", new_path, "and the cost:", sorted_update_vector[i][1][0])

    def server_side(self):
        server_socket = socket(AF_INET, SOCK_DGRAM)
        server_socket.bind((self.server_ip, self.server_port))
        while True:
            message, client_address = server_socket.recvfrom(2048)
            if message == "hello":
                modified_message = "Received"
                server_socket.sendto(modified_message, client_address)
            else:
                data = json.loads(message)
                modified_message = "Received"
                server_socket.sendto(modified_message, client_address)
                neighbor, dist_vector = data[0], json.loads(data[1])
                if neighbor not in self.rcv_update.keys():
                    self.rcv_update[neighbor] = dist_vector.copy()
                    self.Bellmanford(self.rcv_update)
                    self.flag = 0
                elif dist_vector != self.rcv_update[neighbor]:
                    self.rcv_update[neighbor] = dist_vector.copy()
                    self.Bellmanford(self.rcv_update)
                    self.flag = 0
                else:
                    self.flag += 1
                    if self.flag == 5:
                        self.router_status()

def main():
    if len(sys.argv) != 4:
        print("Usage: python script.py <router_name> <router_port> <config_file>")
        return

    router_name = sys.argv[1]
    router_port = int(sys.argv[2])
    config_file = sys.argv[3]

    with open(config_file, 'r') as f:
        lines = f.readlines()

    server_ip = "127.0.0.1"
    my_neighbors = {}
    update_vector = {}
    rcv_update = {}

    i = 1
    while i < len(lines):
        router, cost, neighbor_port = lines[i], float(lines[i + 1]), int(lines[i + 2])
        my_neighbors[router] = [neighbor_port]
        update_vector[router] = [cost, router + router]
        i += 3

    router_instance = Router(router_name, server_ip, router_port, config_file)

    thread1 = threading.Thread(target=router_instance.server_side)
    thread2 = threading.Thread(target=router_instance.client_status)

    thread1.start()
    thread2.start()

if __name__ == "__main__":
    main()
