def initialize(graph, source):
    d = {}
    p = {}
    for node in graph:
        d[node] = float('inf')
        p[node] = None
    d[source] = 0
    return d, p

def bellman_ford(graph, source):
    countloop = 0
    d, p = initialize(graph, source)
    for i in range(len(graph) - 1):
        for u in graph:
            for v in graph[u]:
                if d[v] > d[u] + graph[u][v]:
                    d[v] = d[u] + graph[u][v]
                    p[v] = u
                countloop += 1
    for u in graph:
        for v in graph[u]:
            assert d[v] <= d[u] + graph[u][v]
    return d, p, countloop

def widest_bellman_ford(graph, source):
    countloop = 0
    d, p = initialize(graph, source)
    for i in range(len(graph) - 1):
        for u in graph:
            for v in graph[u]:
                if d[v] < min(d[u], graph[u][v]):
                    d[v] = min(d[u], graph[u][v])
                    p[v] = u
                countloop += 1
    return d, p, countloop

def initialize_dijkstra(graph, source):
    d = {}
    p = {}
    for node in graph:
        d[node] = float('inf')
        p[node] = None
    d[source] = 0
    return d, p

def dijkstra(graph, source):
    nodes = graph.keys()
    unvisited = {node: float('inf') for node in nodes}
    visited = {}
    current = source
    current_cost = 0
    unvisited[current] = current_cost
    p = {}
    for node in graph:
        p[node] = None
    countloop = 0
    while True:
        for neighbour, cost in graph[current].items():
            countloop += 1
            if neighbour not in unvisited:
                continue
            new_cost = current_cost + cost
            if unvisited[neighbour] > new_cost:
                unvisited[neighbour] = new_cost
                p[neighbour] = current
        visited[current] = current_cost
        del unvisited[current]
        if not unvisited:
            break
        candidates = [node for node in unvisited.items() if node[1]]
        current, current_cost = sorted(candidates, key=lambda x: x[1])[0]
    return visited, p, countloop

def widest_dijkstra(graph, source):
    nodes = graph.keys()
    unvisited = {node: float('-inf') for node in nodes}
    visited = {}
    current = source
    current_cost = float('inf')
    unvisited[current] = current_cost
    p = {}
    for node in graph:
        p[node] = None
    countloop = 0
    while True:
        for neighbour, cost in graph[current].items():
            countloop += 1
            if neighbour not in unvisited:
                continue
            new_cost = min(current_cost, cost)
            if unvisited[neighbour] < new_cost:
                unvisited[neighbour] = new_cost
                p[neighbour] = current
        visited[current] = current_cost
        del unvisited[current]
        if not unvisited:
            break
        candidates = [node for node in unvisited.items() if node[1]]
        current, current_cost = sorted(candidates, key=lambda x: x[1])[0]
    return visited, p, countloop

def measure_algorithm_performance(graph, source):
    d, p, countloop = bellman_ford(graph, source)
    print("Bellman-Ford:")
    print("Number of loop iterations:", countloop)
    print("______________________________________________________")

    d, p, countloop = widest_bellman_ford(graph, source)
    print("Widest Bellman-Ford:")

    d, p, countloop = dijkstra(graph, source)

    d, p, countloop = widest_dijkstra(graph, source)
