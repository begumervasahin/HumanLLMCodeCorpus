import random

class Action:
    def __init__(self, aKey, result, result_chances=1.0, create_counter_table=False):
        '''
        Preconditions:
            Required:
                aKey - comparable object
                    The key for this action, so it can be searched if needed.
                result - object or list
                    An object or a list storing possible results for performing this action
            Unrequired:
                result_chance - float or list
                    Default Value is 1 or 100%
                    Used to calculate the chance that an action was successful.
                    Stores the chances that a result can be reached.
                    **If result was a list, then this has to be a list of equal size.** 
                create_counter_table - Boolean
                    Default value is False
                    Determines whether we need to count up the individual states reached by
                    performing this action.
        Postconditions:
            This function returns an Action object that can be used.
        Errors:
            ValueError - Raised if total num result_chances does not equal total num results
        '''
        self.action_key = aKey
        if isinstance(result, list):
            self.__results = result
            self.numResults = len(result)
        else:
            self.__results = [result]
            self.numResults = 1
        
        self.results_len = len(self.__results)
        
        if isinstance(result_chances, list):
            if len(result_chances) == self.numResults:
                self.__results_chances = result_chances
            else:
                raise ValueError("Total number of result chances does not equal total number of results")
        else:
            if self.numResults != 1:
                raise ValueError("Result chances should be a list of equal size if result is a list")
            else:
                self.__results_chances = [result_chances]
                
        self.countResults = create_counter_table
        
        if create_counter_table:
            self.counter_list = {res.getKey(): 0 for res in self.__results}
    
    def testAction(self):
        '''
        Preconditions:
            None
        Postconditions:
            Returns the result of performing an action based on a probability
            generated by randomly generating a number between 0 and 1 inclusive.        
            It returns -1 though if the action had failed.
        '''
        rand_num = random.randint(1, 100000) / 100000
        prev_value = 0
        
        for i in range(self.results_len):
            curr_value = prev_value + self.__results_chances[i]
            
            if prev_value < rand_num < curr_value:
                if self.countResults:
                    self.counter_list[self.__results[i].getKey()] += 1
                return self.__results[i]
            
            prev_value = curr_value
        
        return -1
    
    def getKey(self):
        '''
        Preconditions:
            None
        Postconditions:
            Returns the search key for this action.
        '''
        return self.action_key
    
    def getCount(self):
        '''
        Preconditions:
            None
        Postconditions:
            Returns the counting table for each state reached by performing this action.
        Errors:
            NumberError - if counting is not implemented for this action.
        '''
        if self.countResults:
            return self.counter_list
        else:
            raise ValueError("Counting is not implemented for this action")
    
    def getNumResults(self):
        '''
        Preconditions:
            None
        Postconditions:
            Returns the number of possible results that this action could reach.
        '''
        return self.results_len

# Example usage of the Action class
if __name__ == '__main__':
    # Create an Action instance
    action_key = 'example_action'
    results = ['result1', 'result2', 'result3']
    result_chances = [0.3, 0.5, 0.2]
    create_counter_table = True
    
    action_instance = Action(action_key, results, result_chances, create_counter_table)
    
    # Test the action
    print("Testing the action...")
    result = action_instance.testAction()
    print("Result of the action:", result)
    
    # Get the counting table
    try:
        count_table = action_instance.getCount()
        print("Counting table:", count_table)
    except ValueError as e:
        print("Error:", e)
    
    # Get the number of possible results
    num_results = action_instance.getNumResults()
    print("Number of possible results:", num_results)
