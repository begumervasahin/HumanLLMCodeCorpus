import sys

class DirectedGraph:
    def __init__(self, v):
        self.noOfVerticies = v
        self.adjacencyListArray = [{} for _ in range(v)]
        self.shortestPathArray = [sys.maxsize] * v
        self.visitedArray = [False] * v
        self.graphQueue = []

    def printGraph(self):
        for i in range(self.noOfVerticies):
            print("For", i, "=", self.adjacencyListArray[i])

    def printShortestPath(self):
        for i in range(self.noOfVerticies):
            print("Bellman-Ford", i, "=", self.shortestPathArray[i])

def addEdge(graph, src, dest, weight):
    graph.adjacencyListArray[src][dest] = weight

def BellmanFordFunction(graph):
    while graph.graphQueue:
        BellmanFordFunctionAlgorithm(graph, graph.graphQueue.pop())

def BellmanFordFunctionAlgorithm(graph, source):
    if graph.visitedArray[source]:
        return
    tempArray = []
    leastIndex = 0
    minValue = sys.maxsize
    for dest, weight in graph.adjacencyListArray[source].items():
        tempAdd = graph.shortestPathArray[source] + weight
        tempArray.append(tempAdd)
        if graph.shortestPathArray[dest] > tempAdd:
            graph.shortestPathArray[dest] = tempAdd
            graph.graphQueue.insert(0, dest)
            graph.visitedArray[dest] = False
    for dest, _ in graph.adjacencyListArray[source].items():
        if not graph.visitedArray[dest] and dest not in graph.graphQueue:
            graph.graphQueue.insert(0, dest)
    graph.visitedArray[source] = True

def createDirectedGraph(graph, source):
    graph.shortestPathArray[source] = 0
    graph.graphQueue.insert(0, source)
    addEdge(graph, 0, 1, 4)
    addEdge(graph, 0, 2, 2)
    addEdge(graph, 1, 2, 3)
    addEdge(graph, 1, 3, 2)
    addEdge(graph, 1, 4, 3)
    addEdge(graph, 2, 1, 1)
    addEdge(graph, 2, 3, 4)
    addEdge(graph, 2, 4, 5)
    addEdge(graph, 4, 3, -5)

graph = DirectedGraph(5)
