import socket
import hashlib
import time
from threading import Thread
from collections import defaultdict

class Server(Thread):
    def __init__(self):
        Thread.__init__(self)

    def run(self):
        port = 60000
        s = socket.socket()
        host = ""
        s.bind((host, port))
        s.listen(5)
        for i in range(1, N):
            for u in range(1, N + 1):
                tosend = ""
                for v in range(1, N + 1):
                    tosend = tosend + str(v) + ' ' + str(d[u][v]) + "\n"
                forwd_table_u = tosend
                hashed = hashlib.md5(forwd_table_u.encode()).hexdigest()
                for v in graph[u]:
                    conn, addr = s.accept()
                    conn.send(forwd_table_u.encode())
                    conn, addr = s.accept()
                    conn.send(str(v).encode())
                    conn, addr = s.accept()
                    conn.send(str(u).encode())
                    conn, addr = s.accept()
                    conn.send(hashed.encode())
        conn, addr = s.accept()
        conn.send("Bye".encode())
        conn.close()

class Receiver(Thread):
    def __init__(self):
        Thread.__init__(self)

    def run(self):
        distance = defaultdict()
        ii = 0
        while True:
            ii += 1
            s = socket.socket()
            host = ""
            port = 60000
            s.connect((host, port))
            data = s.recv(1024).decode()
            if ii % 4 == 1:
                dist = data
            elif ii % 4 == 2:
                node = int(data)
            elif ii % 4 == 3:
                sender = int(data)
            else:
                hashVal = data
                ch = hashlib.md5(dist.encode()).hexdigest()
                if ch == hashVal:
                    dist = dist.split('\n')
                    for i in range(len(dist) - 1):
                        dist[i] = dist[i].split()
                        dist[i][0], dist[i][1] = int(dist[i][0]), int(dist[i][1])
                        distance[dist[i][0]] = dist[i][1]
                    for j in graph:
                        if d[node][j] > distance[j] + graph[sender][node]:
                            d[node][j] = distance[j] + graph[sender][node]
                else:
                    print('Some error occurred! Try again.')
            if data == 'Bye':
                break
        s.close()
