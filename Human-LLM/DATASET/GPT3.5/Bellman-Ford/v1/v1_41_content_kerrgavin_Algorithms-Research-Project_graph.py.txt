class Graph(object):
    """Class representing a graph."""
    def __init__(self, V=None, E=None, directed=False):
        self.V = V if V is not None else []
        self.E = E if E is not None else []
        self.directed = directed
        self.adj = {}
        
    def get_adj(self, v):
        """Get the adjacent vertices of a vertex."""
        return self.adj[v.value]
    
    def add_vertex(self, value):
        """Add a new vertex to the graph."""
        u = Vertex(value=value)
        self.V.append(u)
        self.adj[u.value] = []
        
    def add_edge(self, u, v, weight):
        """Add a new edge to the graph."""
        e = Edge(u, v, weight)
        self.adj[u.value].append(e)
        self.E.append(e)
        if not self.directed:
            f = Edge(v, u, weight)
            if v.value not in self.adj:
                self.adj[v.value] = []
            self.adj[v.value].append(f)
            
class Vertex(object):
    """Class representing a vertex."""
    def __init__(self, value=None, d=None, pre=None):
        self.value = value
        self.d = d
        self.pre = pre
        
class Edge(object):
    """Class representing an edge."""
    def __init__(self, u=None, v=None, weight=None):
        self.u = u
        self.v = v
        self.weight = weight
        
    def equals(self, other):
        """Check if two edges are equal."""
        return self.u == other.u and self.v == other.v

# Example usage:
if __name__ == "__main__":
    # Create a new graph
    graph = Graph(directed=True)
    
    # Add vertices
    graph.add_vertex(1)
    graph.add_vertex(2)
    
    # Add edges
    graph.add_edge(Vertex(1), Vertex(2), 5)
    graph.add_edge(Vertex(2), Vertex(1), 3)
