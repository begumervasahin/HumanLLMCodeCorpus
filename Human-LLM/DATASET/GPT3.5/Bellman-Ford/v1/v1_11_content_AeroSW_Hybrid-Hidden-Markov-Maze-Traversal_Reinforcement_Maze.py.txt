import random
import math

class State_Controller:
    def __init__(self):
        self.states = []

    def add_state(self, state):
        self.states.append(state)

    def get_num_states(self):
        return len(self.states)

    def switch_state(self, state_key):
        for state in self.states:
            if state.get_key() == state_key:
                self.current_state = state
                return

class State:
    def __init__(self, state_key, num_actions, reward, terminal=False):
        self.state_key = state_key
        self.num_actions = num_actions
        self.reward = reward
        self.terminal = terminal

    def get_key(self):
        return self.state_key

    def get_num_actions(self):
        return self.num_actions

    def get_reward(self):
        return self.reward

    def is_terminal(self):
        return self.terminal

class Reinforcement_Maze:
    def __init__(self, gamma, state_controller):
        self.state_controller = state_controller
        self.num_states = state_controller.get_num_states()
        self.gamma = gamma
        self.utility_list = {state.get_key(): 0 for state in state_controller.states}
        self.policy_list = {state.get_key(): 0 for state in state_controller.states}

    def create_utilities(self, tol, averages):
        other_u_list = self.__create_adjusted_utility_list(1)
        for state_key in range(1, self.num_states + 1):
            last_utility_for_s = self.utility_list[state_key]
            curr_utility_for_s = other_u_list[state_key]
            util_diff = curr_utility_for_s - last_utility_for_s
            util_diff = math.fabs(util_diff)
            while util_diff > tol:
                last_utility_for_s = curr_utility_for_s
                temp = self.__value_iteration(state_key, averages, self.utility_list)
                self.utility_list[state_key] = temp[0]
                self.policy_list[state_key] = temp[1]
                curr_utility_for_s = self.utility_list[state_key]
                util_diff = curr_utility_for_s - last_utility_for_s
                util_diff = math.fabs(util_diff)
        return self.utility_list.copy(), self.policy_list.copy()

    def __create_adjusted_utility_list(self, val=1.0):
        other_u_list = {}
        for state_key in range(1, self.num_states + 1):
            other_u_list[state_key] = self.utility_list[state_key] + val
        return other_u_list

    def __value_iteration(self, curr_state_key, averages, u_list):
        self.state_controller.switch_state(curr_state_key)
        if self.state_controller.current_state.is_terminal():
            return self.state_controller.current_state.get_reward(), 0
        policy = 0
        action_dictionaries = averages[curr_state_key]
        up_dictionaries = action_dictionaries[1]
        left_dictionaries = action_dictionaries[2]
        down_dictionaries = action_dictionaries[3]
        right_dictionaries = action_dictionaries[4]
        up_summ = sum(u_list[i] * up_dictionaries[i] for i in up_dictionaries.keys())
        left_summ = sum(u_list[i] * left_dictionaries[i] for i in left_dictionaries.keys())
        down_summ = sum(u_list[i] * down_dictionaries[i] for i in down_dictionaries.keys())
        right_summ = sum(u_list[i] * right_dictionaries[i] for i in right_dictionaries.keys())
        randomAU = random.randint(1, 10000)
        highest = None
        policy = None
        if randomAU > 1500:
            highest = up_summ
            policy = 1
            if left_summ > highest:
                highest = left_summ
                policy = 2
            if down_summ > highest:
                highest = down_summ
                policy = 3
            if right_summ > highest:
                highest = right_summ
                policy = 4
        else:
            if randomAU <= 375:
                highest = up_summ
                policy = 1
            elif 375 < randomAU <= 750:
                highest = left_summ
                policy = 2
            elif 750 < randomAU <= 1125:
                highest = down_summ
                policy = 3
            else:
                highest = right_summ
                policy = 4
        cost = self.gamma * highest
        reward_s = self.state_controller.current_state.get_reward()
        return reward_s + cost, policy
