import csv

def parse_graph_file(input_file):
    Adjacency_List = {}
    W = {}
    vertices = 0
    with open(input_file + ".gr") as inputfile:
        for line in inputfile:
            l = line.split()
            if l[0] == 'p':
                vertices = int(l[2])
            elif l[0] == 'a':
                W[int(l[1]), int(l[2])] = int(l[3])
                T = int(l[1])
                H = int(l[2])
                if T not in Adjacency_List:
                    Adjacency_List[T] = []
                Adjacency_List[T].append(H)
        for v in range(1, vertices + 1):
            if v not in Adjacency_List:
                Adjacency_List[v] = []
    return Adjacency_List, W, vertices

def Bellman_Ford(G, W, vertices, s):
    dist = {}
    dist[s] = 0
    pred = {}
    pred[s] = 0
    N = list(range(1, vertices + 1))
    N.remove(s)
    for i in N:
        dist[i] = int(vertices * max(W.values()))
        pred[i] = -1
    for i in range(1, vertices + 1):
        for j in G[i]:
            if dist[j] > dist[i] + W[i, j]:
                dist[j] = dist[i] + W[i, j]
                pred[j] = i
    return dist, pred

def write_csv(distances, predecessors, output_file):
    with open(output_file, 'w', newline='') as csvfile:
        fieldnames = ['Vertex', 'Distance', 'Predecessor']
        writer = csv.DictWriter(csvfile, fieldnames=fieldnames)
        writer.writeheader()
        for key in distances:
            writer.writerow({'Vertex': key, 'Distance': distances[key], 'Predecessor': predecessors[key]})

files = ['matrix_100_1_3_1', 'matrix_100_1_3_2']
for f in files:
    Adjacency_List, W, vertices = parse_graph_file(f)
    s = 1
    D, P = Bellman_Ford(Adjacency_List, W, vertices, s)
    out_file = f + ".csv"
    write_csv(D, P, out_file)
