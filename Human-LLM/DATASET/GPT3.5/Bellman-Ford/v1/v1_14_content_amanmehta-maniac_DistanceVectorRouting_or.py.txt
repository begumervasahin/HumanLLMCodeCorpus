from collections import defaultdict
import time

def initialize(graph, source):
    distance = {node: float('inf') for node in graph}
    distance[source] = 0
    return distance

def bellman_ford(graph, source):
    distance = initialize(graph, source)
    
    # Relaxation process
    for _ in range(len(graph) - 1):
        for u in graph:
            for v in graph[u]:
                if distance[u] + graph[u][v] < distance[v]:
                    distance[v] = distance[u] + graph[u][v]
    
    # Check for negative-weight cycles
    for u in graph:
        for v in graph[u]:
            assert distance[v] <= distance[u] + graph[u][v], "Graph contains a negative-weight cycle"
    
    return distance

def test(graph, source):
    print("Graph size:", len(graph))
    distance = bellman_ford(graph, source)
    print("Shortest distances from source", source, ":", distance)

if __name__ == '__main__':
    graph = defaultdict(dict)
    source = None
    
    # Example graph input handling
    with open('graph_input.txt') as f:
        lines = f.readlines()
        for i, line in enumerate(lines):
            if i == 0:
                source = int(line.strip())
                continue
            parts = line.split()
            u = int(parts[0])
            for j in range(1, len(parts), 2):
                v = int(parts[j])
                w = int(parts[j + 1])
                graph[u][v] = w
    
    test(graph, source)
