edges2 = {
    "v_start": {"1_start": 0, "2_start": 0},
    "1_start": {"1_finish": 40},
    "2_start": {"2_finish": 36},
    "1_finish": {"v_finish": 0},
    "2_finish": {"v_finish": 0, "1_start": 0}
}

def connections(vertex):
    """
    Get connections for a given vertex.
    """
    return edges2.get(vertex)

def weight(start, finish):
    """
    Get the weight of the edge from start to finish.
    """
    return edges2.get(start, {}).get(finish, 0)

def longest_path(edges, paths):
    """
    Find the longest path in the graph.
    """
    for path in paths:
        for vertex in connections(path):
            paths.append([vertex])
    print(paths)

def shortest_path(edges):
    """
    Find the shortest path in the graph.
    """
    paths = []
    weights = []
    for vertex, weight in edges['v_start'].items():
        paths.append([vertex])
        weights.append(weight)

    for path in paths:
        end_vertex = path[-1]
        if len(edges[end_vertex]) == 1:
            path.extend(edges[end_vertex].keys())
            weights.extend(edges[end_vertex].values())
        elif len(edges[end_vertex]) > 1:
            for next_vertex in edges[end_vertex].keys():
                paths.append(list(path) + [next_vertex])
        for vertex in edges[end_vertex]:
            paths.append([])

    return paths, weights
