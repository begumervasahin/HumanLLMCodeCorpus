import socket
import time
from threading import Thread
from collections import defaultdict

class Server(Thread):
    def __init__(self):
        super().__init__()

    def run(self):
        graph = defaultdict(dict)
        with open('test') as f:
            lines = f.readlines()
            for i in range(1, 1 + int(lines[0])):
                gf = lines[i].split()
                if i != 0:
                    k = 1
                    for _ in range(int(gf[0])):
                        graph[i][int(gf[k])] = int(gf[k + 1])
                        k += 2

        d = defaultdict(dict)
        for node in graph:
            for other_node in graph:
                if node != other_node:
                    d[node][other_node] = float('Inf')
                else:
                    d[node][other_node] = 0

        port = 60000
        server_socket = socket.socket()
        host = ""
        server_socket.bind((host, port))
        server_socket.listen(5)

        for _ in range(len(graph) - 1):
            print("Server listening....")
            conn, addr = server_socket.accept()
            time.sleep(0.1)
            for u in graph:
                print("Sending data for node", u)
                to_send = ""
                for v in graph:
                    to_send += f"{v} {d[u][v]}\n"
                forward_table_u = f"{u} {to_send}"
                conn.send(forward_table_u.encode())
            print("Data sent. Closing connection.")
            conn.close()

        print("Out of loop")
        time.sleep(0.1)
        conn, addr = server_socket.accept()
        conn.send("Bye".encode())
        print("Bye sent")
        conn.close()

class Receiver(Thread):
    def __init__(self):
        super().__init__()

    def run(self):
        while True:
            s = socket.socket()
            host = ""
            port = 60000
            s.connect((host, port))
            print("Waiting to receive...")
            data = s.recv(1024).decode()
            print("Received data:", data)
            if data == 'Bye':
                break
            s.close()

if __name__ == '__main__':
    rec = Receiver()
    ser = Server()
    ser.start()
    time.sleep(3)
    rec.start()
    ser.join()
    rec.join()
