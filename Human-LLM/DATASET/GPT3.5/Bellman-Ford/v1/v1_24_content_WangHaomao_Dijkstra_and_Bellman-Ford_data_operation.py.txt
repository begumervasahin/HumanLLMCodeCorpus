from bs4 import BeautifulSoup
import urllib3

class CurrencyScraper:
    def __init__(self):
        self.exchange_rates = {}
        self.currency_names = {}
        self.currency_keys = {}
        self.current_key = 0
        self.scrape_website()

    def try_generate_key(self, currency_name):
        """
        Generate a unique key for a given currency if it doesn't exist already.
        """
        if currency_name not in self.currency_names:
            self.currency_names[currency_name] = self.current_key
            self.currency_keys[self.current_key] = currency_name
            self.current_key += 1

    def scrape_website(self):
        """
        Scrape currency exchange rates from x-rates website and store them.
        """
        http = urllib3.PoolManager()
        base_url = "http://www.x-rates.com"
        response_main = http.request('GET', base_url)
        soup_main = BeautifulSoup(response_main.data, "lxml")
        currency_list = soup_main.find("ul", class_="currencyList ratestable")

        for currency_link in currency_list.find_all("a"):
            currency_name = currency_link.text
            self.try_generate_key(currency_name)
            currency_page_url = base_url + currency_link['href'] + "&amount=1"
            response_page = http.request('GET', currency_page_url)
            soup_page = BeautifulSoup(response_page.data, "lxml")
            tables = soup_page.find_all("table")

            for table in tables:
                table_body = table.find("tbody")
                rows = table_body.find_all("tr")

                for row in rows:
                    columns = [item.text.strip() for item in row.find_all("td")]
                    self.try_generate_key(columns[0])
                    forward_exchange = (self.currency_names[currency_name], self.currency_names[columns[0]])

                    if forward_exchange not in self.exchange_rates:
                        self.exchange_rates[forward_exchange] = float(columns[1])
                        self.exchange_rates[(forward_exchange[1], forward_exchange[0])] = float(columns[2])

def arrange_edges(edge):
    u,v,w = edge.split(' ')
    if (float(w) < 0):
        return (u, [v + ":" + w]), (v, []), (-1, [])
    return (u,[v+":"+w]),(v,[])

def gr_arrange_edges(edge):
    split_list = edge.split(' ')
    if(split_list[0] !='a'): return []
    else:
        u, v, w = split_list[1],split_list[2],split_list[3]     
        if(int(w)<0):
            return (u, [v + ":" + w]), (v, []),(-1,[])
        return (u, [v + ":" + w]), (v, [])

def get_negative_flag(it):
    for i in it:
        if(i[0]==-1):return i
    return (None,None)

def merge_nodes(node1,node2):
    if(node1 == None):
        return node2
    else:
        return node1

def get_graph(filePath,sc):
    data = sc.textFile(filePath,8)
    if('.gr' in filePath):
        nodesMapper = data.flatMap(gr_arrange_edges)
    else:
        nodesMapper = data.flatMap(arrange_edges)
    flag_exist_negative = True if -1 in nodesMapper.mapPartitions(get_negative_flag).collect() else False
    graph = nodesMapper.\
        reduceByKey(lambda x, y: x + y).\
        filter(lambda x:x[0]!=-1).sortByKey()
    return graph,flag_exist_negative
