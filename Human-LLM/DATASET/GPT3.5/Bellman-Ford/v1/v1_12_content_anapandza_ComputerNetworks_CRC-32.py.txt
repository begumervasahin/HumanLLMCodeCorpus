import codecs

def create_crc32_table():
    """
    Create a table for the CRC-32 polynomial.
    
    Returns:
    - list: A list of CRC remainders for all possible byte values (0-255).
    """
    table = []
    for i in range(256):
        remainder = i
        for _ in range(8):
            if remainder & 1:
                remainder >>= 1
                remainder ^= 0xEDB88320
            else:
                remainder >>= 1
        table.append(remainder)
    return table

def crc32_update(buf, crc_table):
    """
    Update CRC32 checksum based on input buffer.
    
    Parameters:
    - buf (bytearray): Input data buffer.
    - crc_table (list): Precomputed CRC32 table.
    
    Returns:
    - int: Updated CRC32 checksum.
    """
    crc = 0xFFFFFFFF
    for byte in buf:
        crc = (crc >> 8) ^ crc_table[(crc & 0xFF) ^ byte]
    return crc ^ 0xFFFFFFFF

def main():
    # Create the CRC32 lookup table.
    crc_table = create_crc32_table()
    
    # Input hexadecimal message.
    message_hex = 'FAFAFA'
    
    # Convert the hexadecimal message to a binary string for display.
    message_bin = bin(int(message_hex, 16))[2:]
    
    print(f"Input message is: 0b{message_bin} (0x{message_hex})")
    
    # Compute the CRC checksum of the hexadecimal message.
    checksum = crc32_update(codecs.decode(message_hex, 'hex'), crc_table)
    
    # Format the checksum as binary and hexadecimal strings.
    checksum_bin = bin(checksum)[2:]
    checksum_hex = hex(checksum)
    
    print(f"Checksum is: 0b{checksum_bin} ({checksum_hex})")
    
    # Construct the message that combines the original binary message and its checksum.
    combined_message = f"0b{message_bin}{checksum_bin}"
    
    print("Message to be sent is:")
    print(combined_message)

if __name__ == "__main__":
    main()
