from graph import *
import pickle
import random
import argparse
import os

PATH = "graphs\\"

def pickle_graph_list(graph, file_name):
    with open(file_name, 'ab') as file:
        pickle.dump(graph, file)

def generate_graph(num_vertices, edge_capacity, weight_max, weight_min=1):
    G = Graph(V=[], E=[], directed=True)
    
    # Add vertices to the graph
    for i in range(num_vertices):
        G.addVertex(i)
    
    # Create edges with random weights
    for i in range(len(G.V)):
        if i < len(G.V) - 1:
            u = G.V[i]
            v = G.V[i + 1]
        else:
            u = G.V[-1]
            v = G.V[0]
        G.addEdge(u, v, random.randint(weight_min, weight_max))
    
    # Add additional edges based on edge capacity
    for u in G.V:
        for _ in range(edge_capacity - 1):
            v = G.V[random.randint(0, len(G.V) - 1)]
            if u != v:
                edge = Edge(u, v, random.randint(weight_min, weight_max))
                G.addEdge(edge.u, edge.v, weight=edge.weight)
    
    return G

def parse_arguments():
    parser = argparse.ArgumentParser()
    parser.add_argument('vertices', action='store', type=int, help='Number of vertices in the generated graph')
    parser.add_argument('max_weight', action='store', type=int, help='Maximum weight for a given edge')
    parser.add_argument('--min_weight', dest='min_weight', type=int, action='store', default=1,
                        help='Minimum value for a given edge (default = 1)')
    parser.add_argument('file_name', action='store', type=str, help='File name')
    return parser.parse_args()

def main():
    args = parse_arguments()
    print("Generating data set " + args.file_name + "...")
    
    # Generate graphs with varying edge capacities
    for i in range(1, args.vertices):
        print("Generating graph with edge capacity " + str(i) + "...")
        graph = generate_graph(args.vertices, i, args.max_weight, args.min_weight)
        print("Generation complete.")
        
        # Save the graph to a pickle file
        file_path = os.path.join(PATH, args.file_name)
        pickle_graph_list(graph, file_path)
        print("File saved.")
    
    print("Data set generated and saved.")

if __name__ == "__main__":
    main()
