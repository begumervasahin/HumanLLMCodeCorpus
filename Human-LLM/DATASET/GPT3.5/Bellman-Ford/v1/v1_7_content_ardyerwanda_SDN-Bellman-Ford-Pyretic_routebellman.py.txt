def build_dic(sip, dip, path):
    dic = {}
    dic[sip] = {dip: path}
    return dic

class BuildPath:
    def __init__(self, topology, switch, dst_switch, dst_port, wG=None):
        self.topology = topology
        self.src_switch = switch
        self.dst_switch = dst_switch
        self.dst_port = dst_port
        self.wG = wG
        print("<<<>>>")
        print(self.topology)
        print(self.src_switch)
        print(self.dst_switch)
        print(self.dst_port)
        print(self.wG)
        print("<<<>>>")

    def bellman_ford_routing(self):
        pathlist = []
        pred, total_cost = self.bellman_ford_alg(self.topology, str(self.src_switch), self.wG)  
        if pred:
            v = self.dst_switch
            print("Asal  : Node ", self.src_switch)
            print("Tujuan: Node ", self.dst_switch)
            while v != self.src_switch:
                pathlist.append(Location(pred[v], self.topology[pred[v]][v][pred[v]]))   
                v = pred[v]
            print("pathlist <<switch[port]>>: ")
            for a in reversed(pathlist):
                print(a, "->", end="")
            print(Location(self.dst_switch, self.dst_port))
            pathlist.append(Location(self.dst_switch, self.dst_port))
            print("Total cost: ", total_cost)
        return pathlist

    def bellman_ford_alg(self, G, source, wG=None, huge=1e30000):
        the_clock0 = time.time()
        pred = {}
        pred_indic = {}
        route_table = {}
        route_table2 = {}
        route = {}

        if wG is None:
            RT = 0
            for RT in G.edge:
                route_table[RT] = {}
                for u in G.edge:
                    route_table[RT][u] = {}
                    route[u] = {}
                    for v in G.edge:
                        if u == RT and G.has_edge(u, v):
                            route[u][v] = 1
                            route[u][u] = 0
                        elif u == v and v == RT:
                            route[u][v] = 0
                        else:
                            route[u][v] = huge
                        route_table[RT][u][v] = route[u][v]
                print("table node", RT, ": ", route_table[RT])
                print(" ")
            route_table2 = copy.deepcopy(route_table)
        else:
            route_table = copy.deepcopy(wG)
            route_table2 = copy.deepcopy(wG)
            for RT in G.edge:
                print("table node", RT, ": ", route_table[RT])

        for k in G.edge:
            pred_indic[k] = huge

        same_table = False
        while not same_table:
            same_indic = 0
            for x in G.edge:
                for y in G.edge:
                    for z in G.edge:
                        if G.has_edge(x, y) and (route_table[x][x][z] < route_table[y][x][z]):
                            route_table[y][x][z] = copy.deepcopy(route_table[x][x][z])    
                            route_table[y][y][z] = min(route_table[y][y][z], route_table[y][y][x] + route_table[y][x][z])
                            same_indic += 1
            print("<<<<<<UPDATE>>>>>>")
            for RT in G.edge:
                print("table node", RT, ": ")
                for RT2 in G.edge:
                    print(RT2, " to ", route_table[RT][RT2])
                    print(RT2, " ke ", route_table2[RT][RT2])
                    for RT3 in G.edge:
                        if RT3 == self.src_switch and (route_table[RT][RT2][RT3] + route_table2[RT][RT][RT2]) < pred_indic[RT] and (RT != RT2):
                            pred[RT] = RT2
                            pred_indic[RT] = copy.deepcopy(route_table[RT][RT2][RT3] + route_table2[RT][RT][RT2])
                        if RT == self.src_switch and RT3 == self.dst_switch and RT == RT2:
                            total_cost = route_table[RT][RT2][RT3]
                print(" ")

            if same_indic == 0:
                same_table = True

        print(pred)
        print("Waktu Eksekusi: ", time.time() - the_clock0)
        return pred, total_cost

class BuildFlowTable(DynamicPolicy):
    def __init__(self, dic, weight=None):
        self.weight = weight
        self.switch = None
        self.port = None
        self.forward = drop
        self.dic = dic
        self.topology = drop
        super(BuildFlowTable, self).__init__()

    def set_network(self, network):
        if network is not None:
            for sip in self.dic.keys():
                for dip in self.dic[sip].keys():
                    for loc in self.dic[sip][dip]:
                        self.forward = if_(match(switch=loc.switch, srcip=sip, dstip=dip), fwd(loc.port_no), self.forward)
            self.policy = self.forward
