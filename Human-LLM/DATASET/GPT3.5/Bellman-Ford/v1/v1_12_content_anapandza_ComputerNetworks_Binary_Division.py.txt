def xor(a, b):
    # Perform XOR operation between two binary strings
    result = []
    # Ensure we iterate over the length of the shorter string
    for i in range(len(a)):
        # XOR operation for binary is equivalent to != (not equal) for boolean
        if a[i] == b[i]:
            result.append('0')
        else:
            result.append('1')
    return ''.join(result)

def mod2div(dividend, divisor):
    # Perform binary division (mod 2) of the dividend by the divisor
    pick = len(divisor)
    # Take a portion from the dividend initially to start the division
    tmp = dividend[0: pick]
    
    result = ""  # To store the result of each division step

    while pick < len(dividend):
        if tmp[0] == '1':
            # Perform XOR if the leftmost bit is 1
            tmp = xor(divisor, tmp) + dividend[pick]
        else:
            # If the leftmost bit is 0, perform XOR with a string of 0s
            tmp = xor('0'*pick, tmp) + dividend[pick]
        # Append '1' or '0' to the result based on whether we used the divisor or 0s string for XOR
        result += '1' if tmp[0] == '1' else '0'
        pick += 1

    # Perform the final XOR operation for the remaining bits
    if tmp[0] == '1':
        tmp = xor(divisor, tmp)
    else:
        tmp = xor('0'*pick, tmp)

    # The final bits of tmp now represent the remainder (or CRC)
    print("Result (quotient) is: ", result)
    print("Remainder is: ", tmp)

# Example usage
data = "10011101"  # Corrected to include an additional bit for a proper example
key = "1011"
mod2div(data, key)
