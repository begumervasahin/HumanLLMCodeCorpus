import itertools

graphRE = re.compile(r"(\d+)\s(\d+)")
edgeRE = re.compile(r"(\d+)\s(\d+)\s(-?\d+)")

def BellmanFord(G):

    print("\nEntered BellmanFord\n")
    V = G[0]
    E = G[1]
    inf = float('inf')
    dist = [inf] * len(V)
    dist[0] = 0
    
    print("dist before: ", dist)
    
    x = list(itertools.permutations(V, 2))
    
    for j in range(1, len(V)):
        for i in range(len(x)):
            u = x[i][0]
            v = x[i][1]
            
            if dist[v] > dist[u] + float(E[u][v]):
                dist[v] = dist[u] + float(E[u][v])
    
    print("\ndist after: ", dist)
    
    print("\nChecking if negative cycle exists: ")
    negCycle = False
    for i in range(len(x)):
        u = x[i][0]
        v = x[i][1]
        if dist[v] > dist[u] + float(E[u][v]):
            negCycle = True

        print("A negative cycle exists")

def readFile(filename):
    vertices = []
    edges = []
    with open(filename, 'r') as inFile:
        line1 = inFile.readline()
        graphMatch = graphRE.match(line1)
        if not graphMatch:
            print(line1 + " not properly formatted")
            quit(1)
        
        num_vertices = int(graphMatch.group(1))
        vertices = list(range(num_vertices))
        
        for _ in range(num_vertices):
            row = []
            for _ in range(num_vertices):
                row.append(float("inf"))
            edges.append(row)
        
        for line in inFile.readlines():
            line = line.strip()
            edgeMatch = edgeRE.match(line)
            if edgeMatch:
                source = int(edgeMatch.group(1))
                sink = int(edgeMatch.group(2))
                weight = int(edgeMatch.group(3))
                edges[source][sink] = weight
    
    print("\nvertex    edges")
    for i in range(len(edges)):
        print(str(vertices[i]) + "         " + str(edges[i]))
    
    return (vertices, edges)

def main(filename, algorithm):
    algorithm = algorithm[1:]
    G = readFile(filename)
    print("\nalgorithm: ", algorithm)
    
    if algorithm == 'b' or algorithm == 'B':
        print("\nabout to call BellmanFord")
        BellmanFord(G)
        print("\ncalled BellmanFord\n")
    elif algorithm == 'f' or algorithm == 'F':
        FloydWarshall(G)
    elif algorithm == "both":
        start = time.process_time()
        BellmanFord(G)
        end = time.process_time()
        BFTime = end - start
        
        start = time.process_time()
        FloydWarshall(G)
        end = time.process_time()
        FWTime = end - start
        
        print("Bellman-Ford timing: " + str(BFTime))
        print("Floyd-Warshall timing: " + str(FWTime))
