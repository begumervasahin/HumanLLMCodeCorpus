import socket
import sys
import pickle

def main(port):
    # Create a UDP socket
    recvsock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    
    # Set socket option to allow reusing address
    recvsock.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    
    # Bind the socket to a specific address and port
    recvsock.bind(("", port))
    
    # Receive data from sender (up to 4096 bytes)
    data, sender_address = recvsock.recvfrom(4096)
    
    # Deserialize the received data using pickle
    recovered = pickle.loads(data)
    
    # Print sender address and recovered entry
    print("Sender:", sender_address)
    print("Recovered Entry:", recovered)

if __name__ == "__main__":
    if len(sys.argv) != 2:
        print("Usage: python script_name.py <port>")
        sys.exit(1)
    
    port = int(sys.argv[1])
    main(port)
