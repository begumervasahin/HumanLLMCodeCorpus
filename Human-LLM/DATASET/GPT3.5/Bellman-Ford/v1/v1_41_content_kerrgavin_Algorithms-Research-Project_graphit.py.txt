import matplotlib.pyplot as plt
import json
import argparse

INPUT_PATH = "data\\"
OUTPUT_PATH = "visualization\\"

def json_to_dict(file_name):
    with open(INPUT_PATH + file_name, "r") as input_file:
        data = json.load(input_file)
    return data

def parse_arguments():
    parser = argparse.ArgumentParser()
    parser.add_argument('input_file', action='store', type=str, help='Input file name')
    parser.add_argument('output_file', action='store', type=str, help='Output file name')
    return parser.parse_args()

def main():
    args = parse_arguments()
    print("Reading data from", args.input_file, "...")
    data = json_to_dict(args.input_file)
    print("Data read.")

    x1 = data["edges"]
    y1 = data["dijkstra"]
    y2 = data["bellmanford"]
    y3 = data["yen"]
    y4 = data["fibonacciheap"]

    name_prefix = args.output_file.split(".")[0]

    def plot_and_save(x, y1, y2, y3, y4, name_suffix, label1, label2, label3, label4, color1, color2, color3, color4):
        name = name_prefix + "-" + name_suffix
        print("Plotting graph for", name_suffix, "...")
        plt.plot(x, y1, label=label1, color=color1)
        plt.plot(x, y2, label=label2, color=color2)
        plt.plot(x, y3, label=label3, color=color3)
        plt.plot(x, y4, label=label4, color=color4)
        plt.legend(loc='best')
        plt.xlabel("Edges")
        plt.ylabel("Times (ms)")
        plt.title(name.upper())
        print("Generating graph labels...")
        plt.savefig(OUTPUT_PATH + name + ".png", bbox_inches="tight")
        print("Graph saved for", name_suffix)
        plt.close()

    plot_and_save(x1, y1, y2, y3, y4, "all", "Dijkstra", "Bellman-Ford", "FibonacciHeap", "Yen", "b", "orange", "g", "r")
    plot_and_save(x1, y1, y3, y4, [], "df", "Dijkstra", "FibonacciHeap", "Yen", "", "b", "g", "r", "")
    plot_and_save(x1, [], y2, y4, "by", "", "Bellman-Ford", "", "Yen", "orange", "", "r")
    plot_and_save(x1, y1, y3, y4, "dfy", "Dijkstra", "FibonacciHeap", "Yen", "b", "g", "r")

if __name__ == "__main__":
    main()
