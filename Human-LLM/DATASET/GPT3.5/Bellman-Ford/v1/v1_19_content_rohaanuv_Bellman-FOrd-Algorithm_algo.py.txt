class Graph:
    def __init__(self, vertices):
        """
        Initializes a new instance of the Graph class.

        :param vertices: Number of vertices in the graph.
        """
        self.V = vertices  # Total number of vertices in the graph
        self.graph = []  # List to store graph edges and weights

    def add_edge(self, u, v, w):
        """
        Adds an edge to the graph.

        :param u: Starting vertex of the edge.
        :param v: Ending vertex of the edge.
        :param w: Weight of the edge.
        """
        self.graph.append([u, v, w])

    def print_arr(self, dist):
        """
        Prints the distance of all vertices from the source.

        :param dist: List containing the distance of each vertex from the source.
        """
        print("Vertex \tDistance from Source")
        for i in range(self.V):
            print(f"{i} \t\t {dist[i]}")

    def bellman_ford(self, src):
        """
        The main function that finds the shortest distances from src to all other vertices
        using Bellman-Ford algorithm. It also detects negative weight cycle.

        :param src: Source vertex
        """
        # Initialize distance from src to all other vertices as INFINITE
        dist = [float("Inf")] * self.V
        dist[src] = 0

        # Relax all edges |V| - 1 times. A simple shortest path from src
        # to any other vertex can have at-most |V| - 1 edges
        for _ in range(self.V - 1):
            for u, v, w in self.graph:
                if dist[u] != float("Inf") and dist[u] + w < dist[v]:
                    dist[v] = dist[u] + w

        # Check for negative-weight cycles. The above step guarantees
        # shortest distances if graph doesn't contain negative weight cycle.
        # If we get a shorter path, then there is a cycle.
        for u, v, w in self.graph:
            if dist[u] != float("Inf") and dist[u] + w < dist[v]:
                print("Graph contains negative weight cycle")
                return

        # Print all distance
        self.print_arr(dist)
