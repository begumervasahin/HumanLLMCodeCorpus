from collections import defaultdict
import socket
import time
from threading import Thread

class Server(Thread):
    def __init__(self):
        super().__init__()

    def run(self):
        port = 60000
        server_socket = socket.socket()
        host = ""
        server_socket.bind((host, port))
        server_socket.listen(5)

        conn, addr = server_socket.accept()
        for i in range(1, N):
            for u in range(1, N + 1):
                to_send = ""
                for v in range(1, N + 1):
                    to_send = to_send + str(v) + ' ' + str(d[u][v]) + "\n"
                forward_table_u = to_send
                for v in graph[u]:
                    conn.send(forward_table_u.encode())
                    time.sleep(0.01)
                    conn.send(str(v).encode())
                    time.sleep(0.01)
                    conn.send(str(u).encode())

        conn.send("Bye".encode())
        conn.close()

class Receiver(Thread):
    def __init__(self):
        super().__init__()

    def run(self):
        distance = defaultdict()
        ii = 0
        while True:
            ii += 1
            s = socket.socket()
            host = ""
            port = 60000
            s.connect((host, port))
            data = s.recv(1024)
            if ii % 3 == 1:
                dist = data
            elif ii % 3 == 2:
                node = int(data)
            else:
                sender = int(data)
                dist = dist.split('\n')
                for i in range(len(dist) - 1):
                    dist[i] = dist[i].split()
                    dist[i][0], dist[i][1] = int(dist[i][0]), int(dist[i][1])
                    distance[dist[i][0]] = dist[i][1]
                for j in graph:
                    if d[node][j] > distance[j] + graph[sender][node]:
                        d[node][j] = distance[j] + graph[sender][node]
            if data == b'Bye':
                break
        s.close()
