import time
import pickle
import json
import argparse
import graph  # Make sure to import the correct graph module
import variants  # Make sure to import the correct variants module

INPUT_PATH = "graphs\\"
OUTPUT_PATH = "data\\"

def read_graph_list(filename):
    with open(INPUT_PATH + filename, 'rb') as file:
        G = pickle.load(file)
    return G

def write_data(data, filename):
    with open(OUTPUT_PATH + filename, "w") as output:
        json.dump(data, output, indent=4)

def parse_arguments():
    parser = argparse.ArgumentParser()
    parser.add_argument('input_file', action='store', type=str, help='Input file name')        
    parser.add_argument('output_file', action='store', type=str, help='Output file name')       
    return parser.parse_args()

def get_time_dijkstra(G):
    avg = 0
    for _ in range(10):
        start = round(time.clock() * 10000)
        variants.dijkstra(G, G.V[0])  # Adjust function call based on your implementation
        end = round(time.clock() * 10000)
        avg += (end - start)
    return avg / 10

def main():
    data = {"vertices": 0, "dijkstra": [], "bellmanford": [], "fibonacciheap": [], "yen": [], "edges": []}
    args = parse_arguments()
    print("Beginning to calculate run times...")
    with open(INPUT_PATH + args.input_file, "rb") as inputdata:
        index = 1
        while True:
            try:
                print("Loading graph data...")
                G = pickle.load(inputdata)
                print("Loading Complete.")
                data["vertices"] = len(G.V)
                data["edges"].append(len(G.E))
                print(f"Graph {index} out of {len(data['vertices']) - 1} info:")
                print("\tDirected:", G.directed)
                print("\tVertex Count:", len(G.V))
                print("\tEdge Count:", len(G.E))
                print("Calculating average time for shortest path...")
                avg = get_time_dijkstra(G)
                data["dijkstra"].append(avg)
                print("Dijkstra Run Time:", data["dijkstra"][-1])
                avg = get_time_fib(G)
                data["fibonacciheap"].append(avg)
                print("Fibonacci Heap Run Time:", data["fibonacciheap"][-1])
                avg = get_time_bellman_ford(G)
                data["bellmanford"].append(avg)
                print("Bellman-Ford Run Time:", data["bellmanford"][-1])
                avg = get_time_yen(G)
                data["yen"].append(avg)
                print("Yen Run Time:", data["yen"][-1])
                G = None
                index += 1
            except EOFError:
                break
    print("Finished calculations.")
    print("Writing data to", args.output_file, "...")
    write_data(data, args.output_file)
    print("Data written.")

if __name__ == "__main__":
    main()
