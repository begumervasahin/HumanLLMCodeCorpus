def initialize_graph():
    """Initializes the graph data."""
    vertices = {1: 0, 2: 1, 3: 2, 4: 3, 5: 4, 6: 5}
    edges = [[1, 2], [1, 3], [1, 4], [3, 4], [4, 5], [4, 6], [5, 6]]
    weights = [1, 2, 4, 3, 5, 2, 4]
    return vertices, edges, weights

def dijkstra(vertices, edges, weights):
    """Implements Dijkstra's algorithm to find the shortest paths."""
    inf = float("inf")
    distance = [inf] * len(vertices)  # Initial distances are set to infinity.
    previous = [0] * len(vertices)  # To track the previous node in the optimal path.
    distance[0] = 0  # Distance from the source to itself is always 0.

    # Update distances based on the edges and their weights.
    for v, i in vertices.items():
        print(f"\nIteration: {i}\nResult Table:")
        for e, edge in enumerate(edges):
            if edge[0] == v:  # Edge emanating from the current vertex.
                adjacent = vertices[edge[1]]  # Index of the adjacent vertex.
                weight = weights[e]  # Weight of the edge.
                # Check for a shorter path to the adjacent vertex.
                if distance[i] + weight < distance[adjacent]:
                    # Update the distance and the path.
                    distance[adjacent] = distance[i] + weight
                    previous[adjacent] = v
                    # Print the current state of the algorithm.
                    print(f"Edge: {edge}\nVertices: {vertices}\nDistance(cost): {distance}\nPrevious: {previous}")
    return distance, previous

def print_path(destination, vertices, previous):
    """Recursively prints the shortest path to the destination."""
    if previous[vertices[destination]] != 0:
        print_path(list(vertices.keys())[list(vertices.values()).index(previous[vertices[destination]])], vertices, previous)
    print(destination)

def main():
    vertices, edges, weights = initialize_graph()
    _, previous = dijkstra(vertices, edges, weights)
    print("\nList of vertices that make up the shortest path, with the least cost:")
    print_path(6, vertices, previous)

if __name__ == "__main__":
    main()
