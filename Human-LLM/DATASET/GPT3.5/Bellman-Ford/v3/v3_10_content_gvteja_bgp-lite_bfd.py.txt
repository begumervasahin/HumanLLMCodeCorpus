import sys
import json
import socket
import pickle
from multiprocessing import Process
from select import select
from datetime import datetime

# Get the current datetime
current_time = datetime.now

# Check if the required arguments are provided
if len(sys.argv) < 3:
    print('Please pass the 2 arguments: node name and periodic refresh value')
    sys.exit()

# Extract node name and periodic refresh value from command line arguments
node_name = sys.argv[1]
periodic_refresh = int(sys.argv[2])
neighbour_weights_file = node_name

# Initialize variables
last_read_time = current_time()
last_update_time = current_time()

# Function to read neighbors from file
def read_neighbors(file_name):
    with open(file_name, 'r') as file:
        neighbors_txt = file.read().split('\n')

    neighbors = {}
    ips = {}
    links = {}

    for neighbor_info in neighbors_txt:
        src_ip, name, dst_ip, cost = neighbor_info.strip().split(':')
        cost = int(cost)
        neighbors[name] = cost
        ips[dst_ip] = name
        links[src_ip] = dst_ip

    global last_read_time
    last_read_time = current_time()
    return neighbors, ips, links

# Initialize neighbors and routing tables
neighbors, ip_to_name, links = read_neighbors(neighbour_weights_file)
routing_table = {name: cost for name, cost in neighbors.items()}
max_distance = 100000
next_hops = {}
distance_vectors = {}
connected_sockets = {}
accepted_sockets = {}
port = 9999

# Function to connect to neighbors
def connect_to_neighbors(neighbors):
    for ip, name in neighbors.items():
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.connect((ip, port))
        connected_sockets[name] = sock

# Function to listen for incoming connections from neighbors
def listen_to_neighbor(ip, neighbor):
    server_sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server_sock.bind((ip, port))
    server_sock.listen(1)
    sock, addr = server_sock.accept()
    accepted_sockets[neighbor] = sock
    print(f'Accepted connection from {addr}')

# Create processes for listening and connecting to neighbors
threads = []
for src_ip, dst_ip in links.items():
    neighbor = ip_to_name[dst_ip]
    thread = Process(target=listen_to_neighbor, args=(src_ip, neighbor,))
    threads.append(thread)

threads.append(Process(target=connect_to_neighbors, args=(ip_to_name,)))

# Start and join all threads
for thread in threads:
    thread.start()

for thread in threads:
    thread.join()

# Prepare lists for socket operations
read_sockets = []
read_names = []
for name, sock in accepted_sockets.items():
    read_sockets.append(sock)
    read_names.append(name)

write_sockets = connected_sockets.values()
read_timeout = 0

# Main loop for communication and routing updates
while True:
    ready_to_read, _, _ = select(read_sockets, [], read_sockets, read_timeout)       
    update_required = False

    # Process incoming data from neighbors
    for sock in ready_to_read:
        data = sock.recv(100000)
        neighbor = read_names[read_sockets.index(sock)]
        neighbor_distance_vector = json.loads(data)
        distance_vectors[neighbor] = neighbor_distance_vector
        cost = neighbors[neighbor]

        for destination, estimated_cost in neighbor_distance_vector.items():
            if estimated_cost + cost < routing_table.get(destination, max_distance):
                routing_table[destination] = estimated_cost + cost
                next_hops[destination] = neighbor
                update_required = True

    # Check for periodic updates or changes in neighbor configuration
    if (not update_required) and ((current_time() - last_update_time).seconds >= periodic_refresh):
        update_required = True

    if (current_time() - last_read_time).seconds >= periodic_refresh:
        updated_neighbors, ip_to_name, links = read_neighbors(neighbour_weights_file)

        if neighbors != updated_neighbors:
            neighbors = updated_neighbors
            new_routing_table = {name: cost for name, cost in neighbors.items()}

            for neighbor, neighbor_dv in distance_vectors.items():
                cost = neighbors[neighbor]

                for destination, estimated_cost in neighbor_dv.items():
                    if cost + estimated_cost < new_routing_table.get(destination, max_distance):
                        new_routing_table[destination] = cost + estimated_cost
                        next_hops[destination] = neighbor

            if new_routing_table != routing_table:
                routing_table = new_routing_table
                update_required = True

    # Perform updates if required
    if update_required:
        data = json.dumps(routing_table)

        for sock in write_sockets:
            sock.sendall(data)

        last_update_time = current_time()
        pickle.dump(distance_vectors, open(f'{node_name}_dv.p', 'wb'))
        pickle.dump(next_hops, open(f'{node_name}_p.p', 'wb'))
        pickle.dump(routing_table, open(f'{node_name}_d.p', 'wb'))

    read_timeout = periodic_refresh - (current_time() - max(last_update_time, last_read_time)).seconds
