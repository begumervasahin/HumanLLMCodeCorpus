import numpy as np

def bellman_ford(start, end, weights):
    n = len(weights)
    distances = [float('inf')] * n
    predecessors = [None] * n
    distances[start] = 0

    for _ in range(n - 1):
        for u in range(n):
            for v in range(n):
                if weights[u][v] != 0 and distances[u] + weights[u][v] < distances[v]:
                    distances[v] = distances[u] + weights[u][v]
                    predecessors[v] = u

    path = [end]
    while predecessors[end] is not None:
        path.append(predecessors[end])
        end = predecessors[end]
    return distances, path[::-1]

def calculate_weights(start_nodes, end_nodes, durations):
    nodes = max(max(start_nodes), max(end_nodes)) + 1
    weights = np.zeros((nodes, nodes))
    for s, f, d in zip(start_nodes, end_nodes, durations):
        weights[s][f] = d
    return weights
