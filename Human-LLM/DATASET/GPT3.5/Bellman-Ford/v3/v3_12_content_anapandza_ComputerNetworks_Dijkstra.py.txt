# Define a function to extract the minimum value from a priority queue
def popmin(pqueue):
    min_value = float('inf')
    min_key = None
    for key, value in pqueue.items():
        if value < min_value:
            min_value = value
            min_key = key
    del pqueue[min_key]
    return min_key

# Implement Dijkstra's algorithm
def dijkstra(graph, start):
    # Initialize priority queue, distances, and predecessors    
    pqueue = {}
    distances = {}
    predecessors = {}
    for vertex in graph:
        distances[vertex] = float('inf')
        predecessors[vertex] = None
    distances[start] = 0

    # Fill priority queue with initial distances
    for vertex in graph:
       pqueue[vertex] = distances[vertex]

    # Main loop of Dijkstra's algorithm
    while pqueue:
        # Extract the node with the minimum distance
        current_vertex = popmin(pqueue)
        for neighbor, weight in graph[current_vertex].items():
            new_distance = distances[current_vertex] + weight
            # Update distances and predecessors if a shorter path is found
            if new_distance < distances[neighbor]:
                pqueue[neighbor] = new_distance
                distances[neighbor] = new_distance
                predecessors[neighbor] = current_vertex

    return distances, predecessors

# Define the graph
graph = {
    0: {1: 6, 2: 1, 3: 4},
    1: {4: 3},
    2: {1: -3, 3: 2},
    3: {4: -1},
    4: {2: 5}
}

# Run Dijkstra's algorithm on the graph
distances, predecessors = dijkstra(graph, 0)

# Print the results
print("Shortest distance from start node 0 to all other nodes:")
for vertex, distance in distances.items():
    print(vertex, '=', distance)

print("Paths from start node to all other nodes (predecessors of nodes):")
for vertex, predecessor in predecessors.items():
    print(vertex, '=', predecessor)
