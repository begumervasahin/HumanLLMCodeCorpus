import socket

import time
from threading import Thread
from collections import defaultdict

class Server(Thread):
    def __init__(self):
        super().__init__()

    def run(self):
        port = 60000
        server_socket = socket.socket()
        host = ""
        server_socket.bind((host, port))
        server_socket.listen(5)
        conn, addr = server_socket.accept()

        for _ in range(1, N):
            for u in range(1, N + 1):
                to_send = "\n".join([f"{v} {d[u][v]}" for v in range(1, N + 1)])
                hashed_data = hashlib.md5(to_send.encode()).hexdigest()

                for v in graph[u]:
                    self.send_data(conn, to_send)
                    self.send_data(conn, str(v))
                    self.send_data(conn, str(u))
                    self.send_data(conn, hashed_data)

        self.send_data(conn, "Bye")
        conn.close()

    def send_data(self, conn, data):
        conn.send(data.encode())
        time.sleep(0.01)

class Receiver(Thread):
    def __init__(self):
        super().__init__()

    def run(self):
        distance = defaultdict()
        ii = 0
        s = socket.socket()
        host = ""
        port = 60000
        s.connect((host, port))
        s.send('Hello server!')

        while True:
            ii += 1
            data = s.recv(1024)

            if ii % 4 == 1:
                dist = data
            elif ii % 4 == 2:
                node = int(data)
            elif ii % 4 == 3:
                sender = int(data)
            else:
                hash_val = data
                computed_hash = hashlib.md5(dist).hexdigest()

                if computed_hash == hash_val:
                    self.process_data(dist, sender)
                else:
                    print('Some error occurred! Try again.')

            if data == 'Bye':
                break

        s.close()

    def process_data(self, dist, sender):
        distance = defaultdict()
        dist = dist.split('\n')
        for i in range(len(dist) - 1):
            dist[i] = dist[i].split()
            dist[i][0], dist[i][1] = int(dist[i][0]), int(dist[i][1])
            distance[dist[i][0]] = dist[i][1]

        for j in graph:
            if d[node][j] > distance[j] + graph[sender][node]:
                d[node][j] = distance[j] + graph[sender][node]
