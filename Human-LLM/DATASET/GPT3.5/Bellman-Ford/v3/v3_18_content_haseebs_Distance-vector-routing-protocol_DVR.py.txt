    def run(self):
        global TIMEOUT_PERIOD
        while True:
            for router_id, timer in self.timers.items():
                if self.is_active[router_id]:
                    self.timers[router_id] = time.time()
                    self.is_active[router_id] = False
                    self.is_down[router_id] = False
                elif not self.is_down[router_id] and time.time() - timer >= TIMEOUT_PERIOD and not self.is_active[router_id]:
                    self.timers[router_id] = time.time()
                    self.is_down[router_id] = True
                    self.set_router_down(router_id)
