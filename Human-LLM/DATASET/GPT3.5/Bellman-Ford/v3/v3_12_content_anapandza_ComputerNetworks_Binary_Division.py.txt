def xor_operation(binary_string1, binary_string2):
    """
    Computes the XOR result of two binary strings.

    Args:
        binary_string1 (str): The first binary string.
        binary_string2 (str): The second binary string.

    Returns:
        str: A binary string representing the XOR operation result.
    """
    # Initialize a list to store the XOR result for each bit.
    xor_result = []

    # Iterate through the bits of the binary strings.
    for i in range(len(binary_string1)):
        # Perform XOR operation: if bits are the same, result is '0'; if different, '1'.
        if binary_string1[i] == binary_string2[i]:
            xor_result.append('0')
        else:
            xor_result.append('1')

    # Join the list of bits to form the resulting binary string.
    return ''.join(xor_result)

def perform_binary_division(dividend, divisor):
    """
    Performs binary division (modulo 2 division) of the given dividend by the divisor,
    printing the remainder of the division.

    Args:
        dividend (str): The binary number to be divided.
        divisor (str): The binary number to divide by.
    """
    # Determine the number of bits to process based on the divisor's length.
    pick_length = len(divisor)
    # Take the initial segment from the dividend to begin division.
    temp_segment = dividend[:pick_length]

    # Iterate over the dividend bits, updating the temporary segment for division.
    for current_bit in dividend[pick_length:]:
        if temp_segment[0] == '1':
            # If leading bit is '1', perform XOR with the divisor.
            temp_segment = xor_operation(divisor, temp_segment[1:] + current_bit)
        else:
            # If leading bit is '0', simulate division by '0's (equivalent to shifting).
            temp_segment = xor_operation('0' * len(divisor), temp_segment[1:] + current_bit)

    # Final XOR to get the remainder if the leading bit is '1'.
    if temp_segment[0] == '1':
        temp_segment = xor_operation(divisor, temp_segment)
    # No need for another XOR if leading bit is '0'; it's already in the correct form.

    # Output the remainder of the binary division.
    print("Remainder is:", temp_segment)

# Example usage of the function with a test case.
test_data = "10011101"
test_key = "1011"
perform_binary_division(test_data, test_key)
