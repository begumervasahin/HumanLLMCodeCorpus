import socket
import sys
from datetime import datetime

# Constants for command types
COMMAND_CLOSE    = "close"
COMMAND_LINKDOWN = "linkdown"
COMMAND_LINKUP   = "linkup"
COMMAND_RTUPDATE = "rtupdate"
COMMAND_SHOWRT   = "showrt"
INFINITY_DISTANCE = float('inf')
BUFFER_SIZE = 4096
LOCALHOST = socket.gethostbyname(socket.gethostname())

# Custom Exceptions
class NoNodeError(Exception):
    pass

class NoInputCmdError(Exception):
    pass

class NotUserCmdError(Exception):
    def __init__(self, cmd):
        self.cmd = cmd

class NoParamsForCmdError(Exception):
    def __init__(self, cmd):
        self.cmd = cmd

class NotEnoughParamsForCmdError(Exception):
    pass

# Utility functions
def get_current_time():
    return datetime.now().strftime("%Y-%b-%d %I:%M:%S %p")

def key_to_address(key):
    ip, port = key.split(':')
    return ip, int(port)

def address_to_key(ip, port):
    return f"{ip}:{port}"

def initialize_socket(ip, port):
    sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    try:
        sock.bind((ip, int(port)))
        print(f"Start PyRoute BF-Client on {ip} with port {port} ...\n")
        return sock
    except socket.error as err_msg:
        print(f"Oops! Something wrong with socket binding!\nError code: {err_msg[0]}, Error message: {err_msg[1]}\n")
        sys.exit(1)

def get_ip_address(ip):
    return LOCALHOST if ip == "localhost" else ip

def encode_node_info(ip, port, cost):
    return f"{ip}:{port},{cost}"

def decode_node_info(node_info_str):
    addr_key, cost = node_info_str.split(',')
    return addr_key, float(cost)

# Command line argument parser
def parse_command_line_args(argv):
    argv = argv[1:]
    route_config = {}
    try:
        port = int(argv.pop(0))
        timeout = float(argv.pop(0))
    except ValueError as err_msg:
        print(f"Error code: {err_msg}, Error message: {err_msg}\n")
        sys.exit(1)
    route_config["port"] = port
    route_config["timeout"] = timeout
    while argv:
        try:
            neighbor_ip = get_ip_address(argv.pop(0))
            neighbor_port = int(argv.pop(0))
            neighbor_cost = float(argv.pop(0))
            neighbor_info = encode_node_info(neighbor_ip, neighbor_port, neighbor_cost) 
            route_config.setdefault("neighbors", []).append(neighbor_info)
        except (ValueError, IndexError) as err_msg:
            print(f"Error: {err_msg}\n")
            sys.exit(1)
    return route_config

# User command parser
def parse_user_command(input_cmd, builtin_cmds):
    update_dict = {'cmd': '', 'addr': (), 'payload': {}}
    input_cmd_list = input_cmd.lower().split()
    if not input_cmd_list:
        raise NoInputCmdError
    cmd = input_cmd_list.pop(0)
    if cmd not in builtin_cmds:
        raise NotUserCmdError(cmd)
    if cmd in [COMMAND_LINKUP, COMMAND_LINKDOWN]:
        if not input_cmd_list:
            raise NoParamsForCmdError(cmd)
        ip = input_cmd_list.pop(0)
        if not input_cmd_list:
            raise NotEnoughParamsForCmdError
        port = input_cmd_list.pop(0)
        update_dict['addr'] = (get_ip_address(ip), int(port))
    update_dict['cmd'] = cmd
    return update_dict

# Main function
def main(argv):
    if len(argv) < 3:
        print("Usage: python script.py <port> <timeout> [<neighbor_ip> <neighbor_port> <neighbor_cost>]...")
        sys.exit(1)

    # Parse command line arguments
    route_config = parse_command_line_args(argv)

    # Initialize socket
    sock = initialize_socket(LOCALHOST, route_config['port'])

    # Built-in commands
    builtin_cmds = [COMMAND_CLOSE, COMMAND_LINKDOWN, COMMAND_LINKUP, COMMAND_RTUPDATE, COMMAND_SHOWRT]

    while True:
        try:
            input_cmd = input("Enter a command: ")
            update_dict = parse_user_command(input_cmd, builtin_cmds)
            print("Parsed Command:", update_dict)
        except (NoInputCmdError, NotUserCmdError, NoParamsForCmdError, NotEnoughParamsForCmdError) as err:
            print("Error:", err)
            continue

        # Handle commands
        cmd = update_dict['cmd']
        if cmd == COMMAND_CLOSE:
            break
        elif cmd == COMMAND_LINKUP:
            addr = update_dict['addr']
            print("Link Up:", addr)
            # Implement your logic for handling Link Up command
        elif cmd == COMMAND_LINKDOWN:
            addr = update_dict['addr']
            print("Link Down:", addr)
            # Implement your logic for handling Link Down command
        elif cmd == COMMAND_RTUPDATE:
            print("Route Update")
            # Implement your logic for handling Route Update command
        elif cmd == COMMAND_SHOWRT:
            print("Show Routing Table")
            # Implement your logic for showing the Routing Table
        else:
            print("Unknown Command:", cmd)

    # Clean up
    sock.close()
