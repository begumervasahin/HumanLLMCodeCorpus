import copy
import json
import sys
import time
import socket
from collections import defaultdict
from datetime import datetime
from threading import Thread

# Constants
BUFFSIZE = 1024
INF = float('inf')
localhost = '127.0.0.1'

# Command constants
CLOSE = 'close'
LINKDOWN = 'linkdown'
LINKUP = 'linkup'
SHOWRT = 'showrt'
RTUPDATE = 'rtupdate'

# Error constants
class NoNodeError(Exception):
    pass

# Helper functions
def addr_to_key(ip, port):
    return f"{ip}:{port}"

def key_to_addr(addr_key):
    return addr_key.split(':')

def decode_node_info(info):
    parts = info.split(':')
    return parts[0], float(parts[1])

def now_time():
    return datetime.now().strftime("%Y-%m-%d %H:%M:%S")

def init_socket(ip, port):
    sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    sock.bind((ip, port))
    return sock

# Timer class
class CountDownTimer:
    def __init__(self, interval, function):
        self.interval = interval
        self.function = function
        self.running = False

    def _run(self):
        while self.running:
            time.sleep(self.interval)
            self.function()

    def start(self):
        if not self.running:
            self.running = True
            self._thread = Thread(target=self._run)
            self._thread.start()

    def stop(self):
        self.running = False
        self._thread.join()

# ResetTimer class
class ResetTimer(CountDownTimer):
    def reset(self):
        self.stop()
        self.start()

# BFClient class
class BFClient:
    def __init__(self):
        self.me_key = ""
        self.sock = None
        self.node_dict = {}
        self.timeout = 0
        self.running = False

        # Command handlers
        self.user_cmds = {
            CLOSE: self.close,
            LINKDOWN: self.link_down,
            LINKUP: self.link_up,
            SHOWRT: self.show_rt
        }
        self.update_cmds = {
            RTUPDATE: self.rt_update,
            LINKDOWN: self.link_down,
            LINKUP: self.link_up
        }

    def start_bfclient(self):
        self.running = True

    def close_bfclient(self):
        print(f"{self.sock.getsockname()} is leaving PyRoute...\n")
        self.sock.close()
        self.running = False
        sys.exit(0)

    def get_node(self, ip, port):
        addr_key = addr_to_key(ip, port)
        node = self.node_dict.get(addr_key)
        if not node:
            raise NoNodeError(f"Node at {ip}:{port} is not in the nodes")
        return node

    def link_up(self, ip, port, **kwargs):
        try:
            node = self.get_node(ip, port)
            node['direct_dist'] = node.pop('link_downed_dist')
            node['is_neighbor'] = True
            self.calculate_costs()
        except NoNodeError as e:
            print(e)

    def link_down(self, ip, port, **kwargs):
        try:
            node = self.get_node(ip, port)
            node['link_downed_dist'] = node.pop('direct_dist')
            node['direct_dist'] = INF
            node['is_neighbor'] = False
            node['watch_dog'].stop()
            self.calculate_costs()
        except NoNodeError as e:
            print(e)

    def close(self):
        self.close_bfclient()

    def show_rt(self):
        print(f"{now_time()}  Distance vector list is:")
        for addr_key, node in self.node_dict.items():
            if addr_key == self.me_key:
                continue
            print(f"Destination = {addr_key}, Cost = {node['cost']}, Link = ({node['link']})")
        print()

    def init_node(self):
        return {"cost": INF, "is_neighbor": False, "link": ""}

    def node_generator(self, cost, is_neighbor, costs=None, direct_dist=INF, addr_key=""):
        if costs is None:
            costs = defaultdict(lambda: INF)
        node = self.init_node()
        node.update({
            "cost": cost,
            "is_neighbor": is_neighbor,
            "costs": costs,
            "direct_dist": direct_dist
        })
        if is_neighbor:
            node['link'] = addr_key
            monitor = ResetTimer(interval=self.timeout, func_ptr=self.link_down, args=list(key_to_addr(addr_key)))
            monitor.start()
            node['watch_dog'] = monitor
        return node

    def get_neighbors(self):
        return {addr_key: node_info for addr_key, node_info in self.node_dict.items() if node_info['is_neighbor']}

    def rt_update(self, ip, port, **kwargs):
        costs = kwargs['costs']
        addr_key = addr_to_key(ip, port)
        for node_addr_key in costs:
            if node_addr_key not in self.node_dict:
                self.node_dict[node_addr_key] = self.init_node()
        if self.node_dict[addr_key]['is_neighbor']:
            node = self.node_dict[addr_key]
            node['costs'] = costs
            node['watch_dog'].reset()
        else:
            print(f"Welcome new neighbor at {addr_key}!\n")
            del self.node_dict[addr_key]
            self.node_dict[addr_key] = self.node_generator(
                cost=self.node_dict[addr_key]['cost'],
                is_neighbor=True,
                direct_dist=kwargs['neighbor']['direct_dist'],
                costs=costs,
                addr_key=addr_key
            )
        self.calculate_costs()

    def broadcast_costs(self):
        costs = {addr_key: node_info['cost'] for addr_key, node_info in self.node_dict.items()}
        packet = {'cmd': RTUPDATE}
        for neighbor_key, neighbor in self.get_neighbors().items():
            updated_costs = copy.deepcopy(costs)
            for dest_addr, cost in costs.items():
                if dest_addr not in [self.me_key, neighbor_key] and self.node_dict[dest_addr]['link'] == neighbor_key:
                    updated_costs[dest_addr] = INF
            packet['payload'] = {'costs': updated_costs, 'neighbor': {'direct_dist': neighbor['direct_dist']}}
            send_packet = json.dumps(packet)
            self.sock.sendto(send_packet, key_to_addr(neighbor_key))

    def calculate_costs(self):
        for dest_addr, dest_node in self.node_dict.items():
            if dest_addr == self.me_key:
                continue
            min_cost, next_hop = INF, ""
            for neighbor_key, neighbor in self.get_neighbors().items():
                if dest_addr in neighbor['costs']:
                    curr_dist = neighbor['direct_dist'] + neighbor['costs'][dest_addr]
                    if curr_dist < min_cost:
                        min_cost = curr_dist
                        next_hop = neighbor_key
            dest_node['link'] = next_hop
            dest_node['cost'] = min_cost
