class BellmanFord:
    """
    Implementation of the Bellman-Ford algorithm to find shortest paths and detect negative cycles.
    """

    def __init__(self, vertices, edges, source):
        """
        Initializes the Bellman-Ford algorithm with the given vertices, edges, and source node.
        """
        self.vertices = vertices
        self.size = len(vertices)
        self.edges = edges
        self.source = source
        self.distance = {vertex: float('inf') for vertex in vertices}
        self.prev_node = {vertex: None for vertex in vertices}
        self.distance[source] = 0
        self.neg_cycles = []
        self.compute_shortest_paths()

    def compute_shortest_paths(self):
        """
        Computes the shortest paths using the Bellman-Ford algorithm.
        """
        for _ in range(self.size - 1):
            for u in self.edges:
                for v in self.edges[u]:
                    if self.distance[u] != float('inf') and self.distance[u] + self.edges[u][v] < self.distance[v]:
                        self.distance[v] = self.distance[u] + self.edges[u][v]
                        self.prev_node[v] = u

        self.print_shortest_paths()
        self.find_negative_cycles()

    def print_shortest_paths(self):
        """
        Prints the shortest paths found during the Bellman-Ford algorithm execution.
        """
        for node in self.prev_node:
            print(f"Shortest path to node {node}: {self.get_shortest_path(node)}")

    def get_shortest_path(self, node):
        """
        Returns the shortest path to the specified node.
        """
        path = []
        current = node
        while current is not None:
            path.insert(0, current)
            current = self.prev_node[current]
        return path

    def find_negative_cycles(self):
        """
        Detects negative cycles in the graph.
        """
        for u in self.edges:
            for v in self.edges[u]:
                if self.distance[u] != float('inf') and self.distance[u] + self.edges[u][v] < self.distance[v]:
                    cycle = self.retrieve_negative_cycle(v)
                    if cycle not in self.neg_cycles:
                        self.neg_cycles.append(cycle)

    def retrieve_negative_cycle(self, start_node):
        """
        Retrieves a negative cycle starting from the given node.
        """
        marked = {start_node: False}
        cycle = [start_node]
        current_node = start_node

        while not marked[current_node]:
            marked[current_node] = True
            current_node = self.prev_node[current_node]
            cycle.append(current_node)

        start_index = cycle.index(current_node)
        return cycle[start_index:]
