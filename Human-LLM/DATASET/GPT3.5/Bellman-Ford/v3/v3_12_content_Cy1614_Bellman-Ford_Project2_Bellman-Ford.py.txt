import numpy as np
import sys

def BellmanFord(start_node, end_node, weights):
    num_nodes = weights.shape[1]
    inf = sys.maxsize
    distances = np.full(num_nodes, np.inf, dtype=float)
    predecessors = np.full(num_nodes, inf, dtype=int)
    distances[start_node] = 0

    for _ in range(num_nodes - 1):
        for u in range(num_nodes):
            for v in range(num_nodes):
                weight = weights[u, v]
                if weight != 0:
                    if distances[u] + weight < distances[v]:
                        distances[v] = distances[u] + weight
                        predecessors[v] = u

    for u in range(num_nodes):
        for v in range(num_nodes):
            weight = weights[u, v]
            if weight != 0:
                if distances[u] + weight < distances[v]:
                    print('Graph contains a negative-weight cycle')

    shortest_path = [end_node]
    while predecessors[end_node] != start_node:
        shortest_path.append(predecessors[end_node])
        end_node = predecessors[end_node]
    shortest_path.append(start_node)

    return shortest_path[::-1]
