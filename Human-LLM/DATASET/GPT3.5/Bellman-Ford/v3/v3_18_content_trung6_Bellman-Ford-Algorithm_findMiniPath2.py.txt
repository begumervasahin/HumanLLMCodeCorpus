import numpy as np
from collections import defaultdict

class Node:
    def __init__(self, vertex_name, dist=float('inf'), trace=None):
        self.vertex_name = vertex_name
        self.dist = dist
        self.trace = trace

class Graph:
    def __init__(self, vertices=None, edges=None):
        self.vertices = vertices or []
        self.edges = edges or []

    def add_edge(self, src, dest, weight):
        self.edges.append((src, dest, weight))

    def bellman_ford(self, src, dest):
        for vertex in self.vertices:
            vertex.dist = float('inf')
        self.vertices[src - 1].dist = 0

        for _ in range(len(self.vertices) - 1):
            for u, v, w in self.edges:
                if self.vertices[v - 1].dist > self.vertices[u - 1].dist + w:
                    self.vertices[v - 1].dist = self.vertices[u - 1].dist + w
                    self.vertices[v - 1].trace = u

        minipath = []
        minipath.append(dest)
        while src not in minipath:
            minipath.append(self.vertices[minipath[-1] - 1].trace)

        return minipath[::-1]

def read_input_file(file_path):
    with open(file_path, 'r') as file:
        num_ver, num_edge, start_vertex, end_vertex = map(int, file.readline().split())
        graph = Graph([Node(i) for i in range(1, num_ver + 1)])

        for line in file:
            src, dest, weight = map(int, line.split())
            graph.add_edge(src, dest, weight)

    return graph, start_vertex, end_vertex

def write_output_file(file_path, minipath, graph):
    with open(file_path, 'w') as file:
        if not minipath:
            file.write(f"\nThere are no paths from {graph.vertices[start_vertex - 1].vertex_name} "
                       f"to {graph.vertices[end_vertex - 1].vertex_name}")
        else:
            file.write(f"Distance from {graph.vertices[start_vertex - 1].vertex_name} "
                       f"to {graph.vertices[end_vertex - 1].vertex_name} is: "
                       f"{graph.vertices[end_vertex - 1].dist}\n")
            file.write(' '.join(str(graph.vertices[v - 1].vertex_name) for v in minipath))

# Read input from file
graph, start_vertex, end_vertex = read_input_file('minpath_in.txt')

# Run Bellman-Ford algorithm
minipath = graph.bellman_ford(start_vertex, end_vertex)

# Write output to file
write_output_file('minpath_out.txt', minipath, graph)
