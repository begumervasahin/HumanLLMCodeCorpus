import socket

# Create a UDP socket
udp_socket = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)

# Allow reusing the address/port combination
udp_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)

# Bind the socket to localhost and port 8179
udp_socket.bind(("localhost", 8179))

# Define the destination address and port
destination_address = ('localhost', 8181)

# Send a message "Hello" to the destination
udp_socket.sendto(b"Hello", destination_address)

# Receive a response from the server (up to 4096 bytes)
response_data, _ = udp_socket.recvfrom(4096)

# Print the received data after decoding it
print(response_data.decode())

# Close the socket
udp_socket.close()
