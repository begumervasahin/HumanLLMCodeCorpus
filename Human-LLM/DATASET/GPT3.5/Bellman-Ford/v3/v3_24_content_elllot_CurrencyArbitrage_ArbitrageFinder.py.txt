from math import exp
import argparse
from CurrencyDigraph import CurrencyDigraph
import asyncio

class ArbitrageFinder:
    def __init__(self, starting_amount):
        self.starting_amount = starting_amount
        self.currency_graph = CurrencyDigraph()
        self.source_currency = self.select_starting_currency()
        self.currency_graph.set_source(self.source_currency)
        self.find_arbitrage()

    def select_starting_currency(self):
        """
        Method to select the starting currency with user input.
        """
        print(chr(27) + "[2J")  # Clear the console
        self.currency_graph.print_currency_list()
        while True:
            user_input = input('Select a currency (pick an ID): ')
            if user_input.isdigit() and int(user_input) in self.currency_graph.scraper.keys:
                selected_currency = int(user_input)
                print('You picked {0}'.format(self.currency_graph.scraper.keys[selected_currency]))
                return selected_currency
            else:
                print('Please select a valid ID...')

    def print_arbitrage_opportunity(self, cycle, original_amount, result_amount):
        """
        Print details of the arbitrage opportunity and profit.
        """
        trade_sequence = [self.currency_graph.scraper.keys[currency] for currency in cycle]
        profit_percent = round(((result_amount - original_amount) / float(original_amount)) * 100, 5)
        print('->'.join(trade_sequence))
        print("Started with: {0} | Ended With: {1} | {2}% profit".format(original_amount, result_amount, profit_percent))

    def find_arbitrage(self):
        """
        Detect negative weight cycles using the Bellman-Ford algorithm.
        """
        for _ in range(len(self.currency_graph.graph) - 1):
            for edge in self.currency_graph.weights:
                source, destination = edge
                if self.currency_graph.distances[source] != float("inf"):
                    temp_distance = self.currency_graph.distances[source] + self.currency_graph.weights[edge]
                    if temp_distance < self.currency_graph.distances[destination]:
                        self.currency_graph.distances[destination] = temp_distance
                        self.currency_graph.parents[destination] = source
                        if destination == self.source_currency:
                            cycle = self.currency_graph.get_cycle(source)
                            self.print_arbitrage_opportunity(cycle, self.starting_amount, exp(-temp_distance) * self.starting_amount)
                            return
        for edge in self.currency_graph.weights:
            source, destination = edge
            temp_distance = self.currency_graph.distances[source] + self.currency_graph.weights[edge]
            if destination == self.source_currency:
                if temp_distance < self.currency_graph.distances[destination]:
                    cycle = self.currency_graph.get_cycle(source)
                    self.print_arbitrage_opportunity(cycle, self.starting_amount, exp(-temp_distance) * self.starting_amount)
                    return
        print("No Arbitrage")

def entry_point(run_count):
    """
    Entry point of the script.
    """
    parser = argparse.ArgumentParser()
    parser.add_argument('-s', '--starting_amount')
    args = parser.parse_args()
    af = ArbitrageFinder(args.starting_amount)
    loop = asyncio.get_event_loop()
    loop.run_until_complete(af.find_arbitrage())
    try:
        loop.run_forever()
    except InterruptedError:
        asyncio.asyncgens.shut_down()

if __name__ == '__main__':
    entry_point(1000)
