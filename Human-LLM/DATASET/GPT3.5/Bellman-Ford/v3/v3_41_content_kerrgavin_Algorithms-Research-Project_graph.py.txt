class Graph:
    """Class representing a graph."""

    def __init__(self, vertices=None, edges=None, directed=False):
        self.vertices = vertices if vertices is not None else []
        self.edges = edges if edges is not None else []
        self.directed = directed
        self.adjacency_list = {}

    def get_adjacent_vertices(self, vertex):
        """Get the adjacent vertices of a given vertex."""
        return self.adjacency_list.get(vertex.value, [])

    def add_vertex(self, value):
        """Add a new vertex to the graph."""
        new_vertex = Vertex(value=value)
        self.vertices.append(new_vertex)
        self.adjacency_list[new_vertex.value] = []

    def add_edge(self, source_vertex, target_vertex, weight):
        """Add a new edge to the graph."""
        new_edge = Edge(source_vertex, target_vertex, weight)
        self.adjacency_list[source_vertex.value].append(new_edge)
        self.edges.append(new_edge)

        if not self.directed:
            reverse_edge = Edge(target_vertex, source_vertex, weight)
            if target_vertex.value not in self.adjacency_list:
                self.adjacency_list[target_vertex.value] = []
            self.adjacency_list[target_vertex.value].append(reverse_edge)


class Vertex:
    """Class representing a vertex."""

    def __init__(self, value=None, distance=None, predecessor=None):
        self.value = value
        self.distance = distance
        self.predecessor = predecessor


class Edge:
    """Class representing an edge."""

    def __init__(self, source=None, target=None, weight=None):
        self.source = source
        self.target = target
        self.weight = weight


if __name__ == "__main__":
    # Example usage
    graph = Graph(directed=True)

    # Add vertices
    graph.add_vertex(1)
    graph.add_vertex(2)

    # Add edges
    vertex1 = Vertex(1)
    vertex2 = Vertex(2)
    graph.add_edge(vertex1, vertex2, 5)
    graph.add_edge(vertex2, vertex1, 3)
