import sys
import time
import csv
import numpy as np

def dijkstra(ist, isp, weights):
    # Initialize variables
    n = len(weights)
    inf = sys.maxsize
    unvisited = np.ones(n, dtype=int)
    cost = np.ones(n) * 1.e6
    parent = -np.ones(n, dtype=int) * inf
    current_node = ist
    
    # Set initial cost for starting node
    cost[current_node] = 0
    unvisited[current_node] = 0
    
    # Main loop
    while np.sum(unvisited) != 0:
        # Find neighbors of current node
        neighbors = np.nonzero(unvisited * weights[current_node, :])[0]
        
        # Update costs of neighbors
        for neighbor in neighbors:
            new_cost = cost[current_node] + weights[current_node, neighbor]
            if new_cost < cost[neighbor]:
                cost[neighbor] = new_cost
                parent[neighbor] = current_node
        
        # Mark current node as visited
        unvisited[current_node] = 0
        
        # Select next node with minimum cost
        unvisited_nodes = np.nonzero(unvisited)[0]
        current_node = unvisited_nodes[np.argmin(cost[unvisited_nodes])]
    
    # Reconstruct shortest path
    shortest_path = [isp]
    while parent[isp] != ist:
        shortest_path.append(parent[isp])
        isp = parent[isp]
    shortest_path.append(ist)
    
    return shortest_path[::-1]

def load_vertices(file_path):
    # Load vertex coordinates from file
    x_coords, y_coords = [], []
    with open(file_path, 'r') as file:
        reader = csv.reader(file)
        for row in reader:
            x_coords.append(float(row[1]))
            y_coords.append(float(row[2]))
    return np.array(x_coords), np.array(y_coords)

def load_edges(file_path):
    # Load edge information from file
    A, B, V = [], [], []
    with open(file_path, 'r') as file:
        reader = csv.reader(file)
        for row in reader:
            A.append(int(row[0]))
            B.append(int(row[1]))
            V.append(float(row[2]))
    return np.array(A), np.array(B), np.array(V)

def calculate_weights(RX, RY, RA, RB, RV):
    # Calculate weights based on edge distances and velocities
    n = len(RX)
    weights = np.zeros((n, n), dtype=float)
    for i in range(len(RA)):
        xa, ya = RX[RA[i] - 1], RY[RA[i] - 1]
        xb, yb = RX[RB[i] - 1], RY[RB[i] - 1]
        distance = np.sqrt((xb - xa) ** 2 + (yb - ya) ** 2)
        time_taken = distance / RV[i]
        weights[RA[i] - 1, RB[i] - 1] = time_taken
    return weights

if __name__ == '__main__':
    # Load vertex and edge information
    RomeX, RomeY = load_vertices('RomeVertices')
    RomeA, RomeB, RomeV = load_edges('RomeEdges')
    
    # Calculate edge weights
    weights = calculate_weights(RomeX, RomeY, RomeA, RomeB, RomeV)
    
    # Set start and end points
    ist, isp = 12, 51
    
    # Run Dijkstra's algorithm
    t0 = time.time()
    shortest_path = dijkstra(ist, isp, weights)
    t1 = time.time()
    
    # Output results
    print('Dijkstra:     ', ist + 1, ' -> ', isp + 1, ' is ', np.array(shortest_path) + 1, t1 - t0)
