from collections import defaultdict

def load_graph_from_file(filename):
    """
    Loads a graph from a given filename. The first line of the file contains
    the source node. The subsequent lines contain edges in the format:
    node1 node2 weight.
    
    Parameters:
    - filename: The name of the file containing the graph representation.
    
    Returns:
    - A tuple containing the graph (as a defaultdict of dicts) and the source node.
    """
    graph = defaultdict(dict)
    with open(filename, 'r') as file:
        lines = file.readlines()
        source = int(lines[0].strip())

        for line in lines[1:]:
            u, *edges = map(int, line.split())
            it = iter(edges)
            for v, w in zip(it, it):
                graph[u][v] = w

    return graph, source

def initialize_distance_map(graph, source):
    """
    Initializes the distance map for all nodes in the graph from the source node.
    
    Parameters:
    - graph: The graph represented as a defaultdict of dicts.
    - source: The source node from which distances are calculated.
    
    Returns:
    - A dict mapping each node to its distance from the source node.
    """
    return {node: float('inf') if node != source else 0 for node in graph}

def run_bellman_ford_algorithm(graph, source):
    """
    Runs the Bellman-Ford algorithm to find the shortest path from the source
    to all other nodes in the graph.
    
    Parameters:
    - graph: The graph represented as a defaultdict of dicts.
    - source: The source node from which shortest paths are calculated.
    
    Returns:
    - A dict mapping each node to its distance from the source node.
    
    Raises:
    - ValueError if a negative-weight cycle is detected in the graph.
    """
    distances = initialize_distance_map(graph, source)

    # Relax edges repeatedly
    for _ in range(len(graph) - 1):
        for u in graph:
            for v, weight in graph[u].items():
                if distances[u] + weight < distances[v]:
                    distances[v] = distances[u] + weight

    # Check for negative-weight cycles
    for u in graph:
        for v, weight in graph[u].items():
            if distances[u] + weight < distances[v]:
                raise ValueError("Graph contains a negative-weight cycle.")

    return distances

def display_shortest_path_distances(distances):
    """
    Prints the shortest path distances from the source to all nodes.
    
    Parameters:
    - distances: A dict mapping each node to its distance from the source node.
    """
    print("Shortest path distances from source:")
    for node, distance in distances.items():
        print(f"Node {node}: Distance = {distance}")

def main():
    """
    Main function to load the graph, run the Bellman-Ford algorithm, and
    display the shortest path distances.
    """
    graph_file = 'graph_input.txt'
    graph, source = load_graph_from_file(graph_file)

    try:
        distances = run_bellman_ford_algorithm(graph, source)
        display_shortest_path_distances(distances)
    except ValueError as error:
        print(error)

if __name__ == "__main__":
    main()
