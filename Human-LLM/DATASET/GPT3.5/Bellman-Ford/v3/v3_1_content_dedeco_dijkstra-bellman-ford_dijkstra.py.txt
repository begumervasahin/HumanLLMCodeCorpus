from utils import initialize_single_source, extract_min, relax

def dijkstra(graph, source):
    """
    Implements Dijkstra's algorithm to find the shortest paths from a source vertex.    

    Args:
    - graph: Graph object representing the graph structure
    - source: Starting vertex for Dijkstra's algorithm

    Returns:
    None
    """
    initialize_single_source(graph, source)

    visited_vertices = []  # List to store visited vertices
    unvisited_vertices = list(graph.get_vertices())  # List to store unvisited vertices

    while unvisited_vertices:
        current_vertex = extract_min(unvisited_vertices)  # Get the vertex with the minimum distance from unvisited vertices
        current_vertex.set_visited()  # Mark the current vertex as visited
        visited_vertices.append(current_vertex)  # Add the current vertex to the list of visited vertices

        # Relax edges going from the current vertex to adjacent vertices
        for adjacent_vertex in current_vertex.get_adjacent_vertices():
            if adjacent_vertex.is_visited():
                continue
            relax(current_vertex, adjacent_vertex)
