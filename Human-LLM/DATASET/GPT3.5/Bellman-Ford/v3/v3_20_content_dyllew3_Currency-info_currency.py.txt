import urllib.request
import json
import threading
import time

CURRENCIES = [
    "USD", "EUR", "JPY", "BGN", "CZK", "DKK", "GBP", "HUF", "LTL", "LVL",
    "PLN", "RON", "SEK", "CHF", "NOK", "HRK", "RUB", "TRY", "AUD", "BRL",
    "CAD", "CNY", "HKD", "ILS", "INR", "KRW", "MXN", "MYR", "NZD", "PHP",
    "SGD", "THB", "ZAR"
]

class CurrencyThread(threading.Thread):
    """
    Thread class to fetch currency exchange rates.
    """

    def __init__(self, start, end, currency_list):
        super().__init__()
        self.subset = currency_list[start:end]
        self.full_list = currency_list
        self.data = []
        self.base_url = "https://query.yahooapis.com/v1/public/yql?q=select%20*%20from%20yahoo.finance.xchange%20where%20pair%20in%20("
        self.base_url_end = ")&format=json&diagnostics=true&env=store%3A%2F%2Fdatatables.org%2Falltableswithkeys&callback="

    def run(self):
        self.data = self.fetch_currency_rates()

    def fetch_currency_rates(self):
        """
        Fetch currency exchange rates from Yahoo Finance API.
        """
        url = self.base_url + ",".join(['"{}{}"'.format(i, j) for i in self.subset for j in self.full_list if i != j]) + self.base_url_end
        try:
            with urllib.request.urlopen(url) as response:
                data = json.loads(response.read().decode())
                return data["query"]["results"]["rate"]
        except Exception as e:
            print("Error occurred while fetching data:", e)
            return None

    def __repr__(self):
        return f"Subset is {self.subset}"

    @staticmethod
    def run_all_threads(thread_list):
        """
        Start and join all threads.
        """
        for thread in thread_list:
            thread.start()
        for thread in thread_list:
            thread.join()


def create_threads(amount, currency_list=CURRENCIES):
    """
    Create threads to fetch currency exchange rates.
    """
    threads = []
    if amount > 0:
        step = len(currency_list) // amount
        for i in range(amount):
            start = i * step
            end = start + step if i < amount - 1 else len(currency_list)
            threads.append(CurrencyThread(start, end, currency_list))
    return threads


def main():
    """
    Main function to orchestrate currency rate fetching.
    """
    start_time = time.time()
    num_threads = 5
    threads = create_threads(num_threads, CURRENCIES)
    
    for thread in threads:
        print(thread)
        thread.start()
    
    count = 0
    for thread in threads:
        thread.join()
        for item in thread.data:
            print(item["id"])
            count += 1
    
    time_taken = time.time() - start_time
    print("Time taken is", time_taken, "seconds")
    print("Number of currency matches is", count)


if __name__ == "__main__":
    main()
