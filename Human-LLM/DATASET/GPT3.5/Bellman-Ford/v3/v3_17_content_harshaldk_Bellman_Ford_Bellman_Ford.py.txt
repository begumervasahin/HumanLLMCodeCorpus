import csv

def parse_graph_file(input_file):
    """
    Parse the graph file to extract adjacency list, weights, and number of vertices.
    """
    adjacency_list = {}
    weights = {}
    vertices = 0

    with open(input_file + ".gr") as inputfile:
        for line in inputfile:
            parts = line.split()
            if parts[0] == 'p':
                vertices = int(parts[2])  # Extract the number of vertices from the header line
            elif parts[0] == 'a':
                # Extract edge information and weight
                weights[int(parts[1]), int(parts[2])] = int(parts[3])
                tail = int(parts[1])
                head = int(parts[2])
                adjacency_list.setdefault(tail, []).append(head)

        # Ensure all vertices have an entry in the adjacency list
        for v in range(1, vertices + 1):
            adjacency_list.setdefault(v, [])

    return adjacency_list, weights, vertices

def bellman_ford(adjacency_list, weights, vertices, source):
    """
    Perform the Bellman-Ford algorithm to calculate distances and predecessors.
    """
    distances = {source: 0}
    predecessors = {source: 0}
    remaining_vertices = list(range(1, vertices + 1))
    remaining_vertices.remove(source)

    for v in remaining_vertices:
        distances[v] = vertices * max(weights.values())
        predecessors[v] = -1

    for _ in range(vertices):
        for u in adjacency_list:
            for v in adjacency_list[u]:
                if distances[v] > distances[u] + weights[u, v]:
                    distances[v] = distances[u] + weights[u, v]
                    predecessors[v] = u

    return distances, predecessors

def write_csv(distances, predecessors, output_file):
    """
    Write distances and predecessors to a CSV file.
    """
    with open(output_file, 'w', newline='') as csvfile:
        fieldnames = ['Vertex', 'Distance', 'Predecessor']
        writer = csv.DictWriter(csvfile, fieldnames=fieldnames)
        writer.writeheader()
        for vertex in distances:
            writer.writerow({'Vertex': vertex, 'Distance': distances[vertex], 'Predecessor': predecessors[vertex]})

# List of input files
files = ['matrix_100_1_3_1', 'matrix_100_1_3_2']
for filename in files:
    adjacency_list, weights, num_vertices = parse_graph_file(filename)
    source_vertex = 1
    distances, predecessors = bellman_ford(adjacency_list, weights, num_vertices, source_vertex)
    output_csv = filename + ".csv"
    write_csv(distances, predecessors, output_csv)
