def create_graph(duration_list, edges_from_finish):
    """
    Create a directed graph based on job durations and edges from finish nodes.
    """
    graph_size = 2 * len(duration_list) + 2
    graph = np.zeros((graph_size, graph_size))

    # Fill in the graph with negative durations for the main path
    for i in range(len(duration_list)):
        graph[i][i + graph_size // 2] += -duration_list[i]

    # Connect edges from finish nodes to subsequent nodes with a small negative value   
    for i in range(len(edges_from_finish)):
        for j in edges_from_finish[i]:
            graph[i + graph_size // 2][j] = -1e-4

    return graph

def bubble_sort(list1, list2):
    """
    Perform bubble sort on two lists simultaneously.
    """
    j = 0
    stop = 0
    while stop < len(list1) / 2:
        stop = 0
        for i in range(len(list1) - 1):
            if i % 2 == j:
                if list1[i][1] > list1[i + 1][1]:
                    list1[i], list1[i + 1] = list1[i + 1], list1[i]
                    list2[i], list2[i + 1] = list2[i + 1], list2[i]
                else:
                    stop += 1
        j = (j + 1) % 2
    return list1, list2
