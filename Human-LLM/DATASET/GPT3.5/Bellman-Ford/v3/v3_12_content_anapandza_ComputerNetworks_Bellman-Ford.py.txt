def initialize(graph, source):
    """
    Belirli bir grafik için mesafeler ve önceki düðümleri baþlatýr.
    
    Args:
        graph (dict): Grafik verisi, düðümlerin komþularýný ve aralarýndaki mesafeleri içerir.
        source (str): Kaynak düðüm.
        
    Returns:
        dict, dict: Mesafeler ve önceki düðümler için sözlükler.
    """
    distances = {}  # Düðümler arasýndaki mesafeler
    predecessors = {}  # Her düðümün önceki düðümleri
    for node in graph:
        distances[node] = float('inf')  # Baþlangýçta tüm mesafeler sonsuz
        predecessors[node] = None  # Baþlangýçta hiçbir önceki düðüm yok
    distances[source] = 0  # Kaynaktan kendine olan mesafe 0
    return distances, predecessors

def relax(node, neighbour, graph, distances, predecessors):
    """
    Bir düðümün komþularýyla iliþkili mesafeleri günceller.
    
    Args:
        node (str): Mevcut düðüm.
        neighbour (str): Mevcut düðümün bir komþusu.
        graph (dict): Grafik verisi, düðümlerin komþularýný ve aralarýndaki mesafeleri içerir.
        distances (dict): Her düðümün kaynaktan olan mesafeleri.
        predecessors (dict): Her düðümün önceki düðümleri.
    """
    # Eðer bir komþunun mesafesi, mevcut düðümün mesafesinden daha uzaksa, güncellenir
    if distances[neighbour] > distances[node] + graph[node][neighbour]:
        distances[neighbour] = distances[node] + graph[node][neighbour]
        predecessors[neighbour] = node

def bellman_ford(graph, source):
    """
    Bellman-Ford algoritmasýný kullanarak en kýsa yollarý bulur.
    
    Args:
        graph (dict): Grafik verisi, düðümlerin komþularýný ve aralarýndaki mesafeleri içerir.
        source (str): Kaynak düðüm.
        
    Returns:
        dict, dict: Tüm düðümlere olan en kýsa mesafeler ve önceki düðümler için sözlükler.
    """
    # Ýlk olarak tüm mesafeleri ve önceki düðümleri baþlangýç deðerlerine ayarla
    distances, predecessors = initialize(graph, source)
    
    # Grafikteki düðümlerin sayýsýný (n-1) kez gezin
    for i in range(len(graph) - 1):
        for u in graph:
            # Her düðümde komþularýný kontrol et
            for v in graph[u]:
                relax(u, v, graph, distances, predecessors)
    
    # Negatif döngü kontrolü
    for u in graph:
        for v in graph[u]:
            assert distances[v] <= distances[u] + graph[u][v]
    
    return distances, predecessors
