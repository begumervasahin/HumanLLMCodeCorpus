import copy
from decimal import Decimal
import numpy as np
from bellman_ford import bellman_ford  # Assuming bellman_ford function is available

class Portfolio:
    def __init__(self, currencies):
        self.currencies = currencies

    def increase_currency(self, name, amount):
        # Increase currency amount
        if amount < 0:
            raise ValueError('Amount must be non-negative')
        self.currencies[name] = self.currencies.get(name, 0) + amount

    def decrease_currency(self, name, amount):
        # Decrease currency amount
        if name not in self.currencies or self.currencies[name] < amount:
            raise RuntimeError('Insufficient funds')
        self.currencies[name] -= amount

def append_exchange_fee(rates, fee_percentage):
    # Append exchange fee to rates
    rates_with_fee = copy.deepcopy(rates)
    for from_currency, to_currencies in rates_with_fee.items():
        for to_currency in to_currencies:
            to_currencies[to_currency] *= (1 - fee_percentage)
    return rates_with_fee

def transform_rates(rates):
    # Transform rates for calculation
    transformed_rates = copy.deepcopy(rates)
    for from_currency, to_currencies in transformed_rates.items():
        for to_currency in to_currencies:
            to_currencies[to_currency] = -np.log(to_currencies[to_currency])
    return transformed_rates

def build_route(predecessor, source, target):
    # Build the route for the shortest path
    route = [target]
    while route[0] != source:
        target = predecessor[route[0]]
        route.insert(0, target)
    return route

def find_shortest_path(portfolio, target_currency, transformed_rates):
    # Find the shortest path for currency exchange
    source_currency = None
    final_predecessor = None
    shortest_distance = float('inf')
    for name, amount in portfolio.currencies.items():
        if name == target_currency:
            continue
        distance, predecessor = bellman_ford(transformed_rates, name)
        if distance[target_currency] < shortest_distance:
            shortest_distance = distance[target_currency]
            final_predecessor = predecessor
            source_currency = name
    if source_currency is None:
        raise RuntimeError(f'No route to {target_currency}')
    route = build_route(final_predecessor, source_currency, target_currency)
    return source_currency, route

def calculate_full_route_rate(route, rates):
    # Calculate the full route rate
    full_route_rate = Decimal(1)
    for idx in range(len(route) - 1):
        _from, _to = route[idx], route[idx + 1]
        full_route_rate *= Decimal(rates[_from][_to])
    return full_route_rate

def exchange_currencies(portfolio, rates, fee, target_currency, amount):
    # Perform currency exchange
    if not portfolio.currencies:
        raise RuntimeError('You have no currency')

    rates_with_fee = append_exchange_fee(rates, fee)
    transformed_rates = transform_rates(rates_with_fee)
    source_currency, route = find_shortest_path(portfolio, target_currency, transformed_rates)
    full_route_rate = calculate_full_route_rate(route, rates_with_fee)

    max_exchange_amount = Decimal(portfolio.currencies[source_currency]) * full_route_rate
    currency_cost = min(Decimal(amount), max_exchange_amount)

    for idx in range(len(route) - 1):
        _from, _to = route[idx], route[idx + 1]
        _exchange_amount = currency_cost * Decimal(rates_with_fee[_from][_to])
        portfolio.decrease_currency(_from, currency_cost)
        portfolio.increase_currency(_to, _exchange_amount)
        print(f'Exchanged {currency_cost} {_from} to {_exchange_amount} {_to}')
        currency_cost = _exchange_amount

# Example usage:
initial_currencies = {'USD': 1000, 'EUR': 500, 'GBP': 300}
portfolio = Portfolio(initial_currencies)

exchange_rates = {
    'USD': {'USD': 1.0, 'EUR': 0.9, 'GBP': 0.75},
    'EUR': {'USD': 1.1, 'EUR': 1.0, 'GBP': 0.85},
    'GBP': {'USD': 1.33, 'EUR': 1.18, 'GBP': 1.0}
}

exchange_fee = 0.01
target_currency = 'GBP'
exchange_amount = 200

exchange_currencies(portfolio, exchange_rates, exchange_fee, target_currency, exchange_amount)
