from Action import Action
import random

class State:
    def __init__(self, reward: float, is_terminal: bool, state_key: any, action_list=None, count_actions: bool = False):
        """
        Initialize a State object.

        Parameters:
            reward (float): The reward received for being in this state at timestep t.
            is_terminal (bool): True if this state is terminal, False otherwise.
            state_key (any): Search key for this state, can be any value.
            action_list (list, optional): List of actions this state can perform. Defaults to None.
            count_actions (bool, optional): If True, actions will be counted. Defaults to False.
        """
        self.reward = reward
        self.is_terminal = is_terminal
        self.state_key = state_key
        self.actions = action_list if action_list is not None else []
        self.num_actions = len(self.actions)
        self.count_actions = count_actions
        self.action_counter = {} if count_actions else None
        if self.count_actions:
            for action in self.actions:
                self.action_counter[action.get_key()] = 0

    def try_action(self, action_key: any) -> float or int:
        """
        Try to perform an action.

        Parameters:
            action_key (any): The search key for the action to attempt.

        Returns:
            float or int: The result of the action if successful, -1 otherwise.
        """
        for action in self.actions:
            if action.get_key() == action_key:
                if self.count_actions:
                    self.action_counter[action_key] += 1
                return action.test_action()
        return -1

    def get_key(self) -> any:
        """
        Get the search key for this state.

        Returns:
            any: The search key for this state.
        """
        return self.state_key

    def get_reward(self) -> float:
        """
        Get the reward for being in this state.

        Returns:
            float: The reward for being in this state.
        """
        return self.reward

    def is_terminal_state(self) -> bool:
        """
        Check if this state is terminal.

        Returns:
            bool: True if this state is terminal, False otherwise.
        """
        return self.is_terminal

    def add_action(self, action: Action):
        """
        Add an action to this state.

        Parameters:
            action (Action): The action object to add.

        Raises:
            NameError: If the action key already exists in the state's actions.
            TypeError: If the action is not an instance of the Action class.
        """
        if isinstance(action, Action):
            action_key = action.get_key()
            if action_key in [a.get_key() for a in self.actions]:
                raise NameError(f"Action key '{action_key}' already exists in the state's actions.")
            self.actions.append(action)
            self.num_actions += 1
            if self.count_actions:
                self.action_counter[action_key] = 0
        else:
            raise TypeError("Action must be an instance of the Action class.")

    def get_num_actions(self) -> int:
        """
        Get the number of actions this state can perform.

        Returns:
            int: The number of actions.
        """
        return self.num_actions

    def get_count(self) -> dict:
        """
        Get the count of actions performed and their results.

        Returns:
            dict: A dictionary containing the action counter and action information.
        """
        if not self.count_actions:
            raise ValueError("Counting is not enabled for this state.")

        action_info = {}
        for action in self.actions:
            action_info[action.get_key()] = action.get_count()

        return self.action_counter, action_info

# Usage example:
if __name__ == '__main__':
    # Create an example state
    state = State(reward=10.0, is_terminal=True, state_key='State1', count_actions=True)

    # Create example actions and add them to the state
    action1 = Action('Action1', lambda: random.uniform(0, 1))
    action2 = Action('Action2', lambda: random.uniform(0, 1))
    state.add_action(action1)
    state.add_action(action2)

    # Test try_action method
    print(state.try_action('Action1'))  # Output: random value between 0 and 1
    print(state.try_action('Action2'))  # Output: random value between 0 and 1

    # Get state key, reward, terminal status, and number of actions
    print(state.get_key())  # Output: 'State1'
    print(state.get_reward())  # Output: 10.0
    print(state.is_terminal_state())  # Output: True
    print(state.get_num_actions())  # Output: 2

    # Get action count information (assuming counting is enabled)
    try:
        counter, action_info = state.get_count()
        print(counter)  # Output: {'Action1': 0, 'Action2': 0}
        print(action_info)  # Output: {'Action1': 0, 'Action2': 0}
    except ValueError as e:
        print(e)  # Output: Counting is not enabled for this state.
