import socket
import time
from threading import Thread
from collections import defaultdict

class Server(Thread):
    def __init__(self):
        super().__init__()

    def run(self):
        graph = self.load_graph_data('test')
        distance_matrix = self.initialize_distance_matrix(graph)
        server_socket = self.setup_server_socket()

        self.send_data_to_clients(server_socket, graph, distance_matrix)

    def load_graph_data(self, filename):
        graph = defaultdict(dict)
        with open(filename) as file:
            lines = file.readlines()
            for i in range(1, int(lines[0]) + 1):
                elements = lines[i].split()
                node = int(elements[0])
                for j in range(1, len(elements), 2):
                    neighbor = int(elements[j])
                    weight = int(elements[j + 1])
                    graph[node][neighbor] = weight
        return graph

    def initialize_distance_matrix(self, graph):
        distance_matrix = defaultdict(dict)
        for node in graph:
            for neighbor in graph:
                distance_matrix[node][neighbor] = float('inf') if node != neighbor else 0
        return distance_matrix

    def setup_server_socket(self):
        port = 60000
        server_socket = socket.socket()
        host = ""
        server_socket.bind((host, port))
        server_socket.listen(5)
        return server_socket

    def send_data_to_clients(self, server_socket, graph, distance_matrix):
        for _ in range(len(graph) - 1):
            print("Server listening....")
            conn, _ = server_socket.accept()
            time.sleep(0.1)
            for node in graph:
                print("Sending data for node", node)
                message = self.construct_message(node, graph, distance_matrix)
                conn.send(message.encode())
            print("Data sent. Closing connection.")
            conn.close()

        print("Out of loop")
        time.sleep(0.1)
        conn, _ = server_socket.accept()
        conn.send("Bye".encode())
        print("Bye sent")
        conn.close()

    def construct_message(self, node, graph, distance_matrix):
        message = f"{node} "
        for neighbor, weight in graph[node].items():
            message += f"{neighbor} {weight}\n"
        return message

class Receiver(Thread):
    def __init__(self):
        super().__init__()

    def run(self):
        while True:
            s = socket.socket()
            host = ""
            port = 60000
            s.connect((host, port))
            print("Waiting to receive...")
            data = s.recv(1024).decode()
            print("Received data:", data)
            if data == 'Bye':
                break
            s.close()

if __name__ == '__main__':
    receiver_thread = Receiver()
    server_thread = Server()
    server_thread.start()
    time.sleep(3)
    receiver_thread.start()
    server_thread.join()
    receiver_thread.join()
