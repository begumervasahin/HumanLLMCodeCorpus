import argparse
import math
import graph_tool
import graph_tool.search
from CurrencyExchangeReader import CurrencyExchangeReader

class ArbitrageDetector:
    def __init__(self, data_file):
        self.graph = graph_tool.Graph()
        self.currency_vertices = {}
        self.vertex_labels = self.graph.new_vertex_property("string")
        self.edge_weights = self.graph.new_edge_property("float")
        self.currency_data_reader = CurrencyExchangeReader(data_file)

    def detect_arbitrage(self):
        currency_codes, exchange_rates = self.currency_data_reader.read_data()

        # Create vertices for each currency code
        for code in currency_codes:
            vertex = self.graph.add_vertex()
            self.currency_vertices[code] = vertex
            self.vertex_labels[vertex] = code

        # Create edges with corresponding exchange rates
        for exchange_rate in exchange_rates:
            if self.is_valid_exchange(exchange_rate):
                from_vertex = self.currency_vertices[exchange_rate.from_currency]
                to_vertex = self.currency_vertices[exchange_rate.to_currency]
                edge = self.graph.add_edge(from_vertex, to_vertex)
                self.edge_weights[edge] = -math.log(exchange_rate.exchange_rate)

        # Detect arbitrage using Bellman-Ford algorithm
        all_edges_minimized, _, _ = graph_tool.search.bellman_ford_search(
            self.graph,
            self.graph.vertex(0),
            self.edge_weights,
            infinity=200000000.0
        )
        return all_edges_minimized

    def is_valid_exchange(self, exchange_rate):
        return exchange_rate.from_currency != exchange_rate.to_currency and exchange_rate.exchange_rate > 0

def parse_command_line_args():
    parser = argparse.ArgumentParser()
    parser.add_argument("currency_exchange_file", type=str, help="File containing currency exchange rates")
    args = parser.parse_args()
    return args.currency_exchange_file

def main():
    currency_exchange_file = parse_command_line_args()
    arbitrage_detector = ArbitrageDetector(currency_exchange_file)
    arbitrage_detector.detect_arbitrage()

if __name__ == '__main__':
    main()
