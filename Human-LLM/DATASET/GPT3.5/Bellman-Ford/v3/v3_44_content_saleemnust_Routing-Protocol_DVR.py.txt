import threading
import time
import sys
import json
from socket import socket, AF_INET, SOCK_DGRAM

class Router:
    def __init__(self, router_name, server_ip, server_port, config_file):
        self.router_name = router_name
        self.server_ip = server_ip
        self.server_port = server_port
        self.config_file = config_file
        self.my_neighbors = {}
        self.update_vector = {}
        self.rcv_update = {}
        self.flag = 0

    def bellman_ford(self, received_update):
        for src_router, dst_routers in received_update.items():
            for dst_router, (cost, path) in dst_routers.items():
                if dst_router in self.update_vector:
                    new_cost = cost + self.update_vector[src_router][0]
                    if new_cost < self.update_vector[dst_router][0]:
                        self.update_vector[dst_router] = [new_cost, self.update_vector[src_router][1] + path]
                else:
                    self.update_vector[dst_router] = [cost + self.update_vector[src_router][0], self.update_vector[src_router][1] + path]

    def send_update_to_server(self):
        client_socket = socket(AF_INET, SOCK_DGRAM)
        message = json.dumps(self.update_vector)
        data = [self.router_name, message]
        message = json.dumps(data)
        client_socket.sendto(message, (self.server_ip, self.server_port))
        modified_message, _ = client_socket.recvfrom(2048)
        if modified_message == "Received":
            client_socket.close()

    def check_neighbor_status(self, port):
        status = -1
        flag_socket = socket(AF_INET, SOCK_DGRAM)
        try:
            flag_socket.sendto(b"hello", (self.server_ip, port))
            flag_socket.recvfrom(2048)
            status = 1
        except:
            status = 0
        finally:
            flag_socket.close()
        return status

    def update_neighbor_status(self):
        while True:
            for neighbor, (_, port) in self.my_neighbors.items():
                if self.check_neighbor_status(port) == 1:
                    self.send_update_to_server()
            time.sleep(3)

    def print_router_status(self):
        sorted_update_vector = sorted(self.update_vector.items())
        print(f"I am Router {self.router_name}")
        for dst_router, (cost, path) in sorted_update_vector:
            if dst_router != self.router_name:
                new_path = ''.join([path[i] for i in range(len(path) - 1) if path[i] != path[i + 1]]) + path[-1]
                print(f"Least cost path to router {dst_router}: {new_path} and the cost: {cost}")

    def receive_updates_from_neighbors(self):
        server_socket = socket(AF_INET, SOCK_DGRAM)
        server_socket.bind((self.server_ip, self.server_port))
        while True:
            message, client_address = server_socket.recvfrom(2048)
            if message == b"hello":
                server_socket.sendto(b"Received", client_address)
            else:
                neighbor, dist_vector = json.loads(message)
                server_socket.sendto(b"Received", client_address)
                if neighbor not in self.rcv_update or dist_vector != self.rcv_update[neighbor]:
                    self.rcv_update[neighbor] = dist_vector.copy()
                    self.bellman_ford(self.rcv_update)
                    self.flag = 0
                else:
                    self.flag += 1
                    if self.flag == 5:
                        self.print_router_status()

def main():
    if len(sys.argv) != 4:
        print("Usage: python script.py <router_name> <router_port> <config_file>")
        return

    router_name = sys.argv[1]
    router_port = int(sys.argv[2])
    config_file = sys.argv[3]

    with open(config_file, 'r') as f:
        lines = f.readlines()

    server_ip = "127.0.0.1"
    my_neighbors = {}
    update_vector = {}

    i = 1
    while i < len(lines):
        router, cost, neighbor_port = lines[i], float(lines[i + 1]), int(lines[i + 2])
        my_neighbors[router] = [cost, neighbor_port]
        update_vector[router] = [cost, router + router]
        i += 3

    router_instance = Router(router_name, server_ip, router_port, config_file)

    thread1 = threading.Thread(target=router_instance.receive_updates_from_neighbors)
    thread2 = threading.Thread(target=router_instance.update_neighbor_status)

    thread1.start()
    thread2.start()

if __name__ == "__main__":
    main()
