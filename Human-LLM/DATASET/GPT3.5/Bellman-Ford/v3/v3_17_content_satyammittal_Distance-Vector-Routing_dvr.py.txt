import socket
import json
import time
import ctypes
import numpy as np
from multiprocessing import Process, Array

# Constants and global variables
TERMINATE_TIME = 12
NUM_NODES = 3
NEARBY_NODES = []
SERVER_PORTS = []
ANSWER_MATRIX_BASE = 0
ANSWER_MATRIX = 0
UPDATE_MATRIX_BASE = 0
UPDATE_MATRIX = 0

class Node:
    def __init__(self, node_num, initial_distance_vector, nearby_nodes):
        self.node_num = node_num
        self.current_distance_vector_base = Array(ctypes.c_float, (NUM_NODES + 1))
        self.current_distance_vector = np.ctypeslib.as_array(self.current_distance_vector_base.get_obj())
        for i in range(0, NUM_NODES + 1):
            self.current_distance_vector[i] = initial_distance_vector[i]
        self.host = ""
        s1 = socket.socket()
        s1.bind(("", 0))
        self.server_port = s1.getsockname()[1]
        SERVER_PORTS.append(self.server_port)
        s1.close()
        self.nearby_nodes = nearby_nodes
        self.connected = {}
        if node_num == 1:
            UPDATE_MATRIX[self.node_num] = 1
        else:
            UPDATE_MATRIX[self.node_num] = 1

    def node_server(self):
        s = socket.socket()
        s.bind((self.host, self.server_port))
        s.listen(5)
        while True:
            if (time.time() - old_time) > TERMINATE_TIME:
                break
            fl = 0
            s.settimeout(1)
            conn = 0
            try:
                conn, addr = s.accept()
                data = conn.recv(10000)
            except socket.timeout:
                if conn:
                    conn.close()
                fl = 1
            except socket.error:
                if conn:
                    conn.close()
                fl = 1
            if fl == 1:
                break
            data = json.loads(data)
            self.update_distance_matrix(data["node"], data["arr"])
            conn.close()
        s.close()

    def node_client(self):
        while True:
            while UPDATE_MATRIX[self.node_num] == 0:
                if (time.time() - old_time) > TERMINATE_TIME:
                    break
                continue
            UPDATE_MATRIX[self.node_num] = 0
            data = json.dumps({"node": self.node_num, "arr": self.current_distance_vector.tolist()})
            try:
                self.try_connection(data)
            except Exception as e:
                self.try_connection(data)
            if (time.time() - old_time) > TERMINATE_TIME:
                break

    def try_connection(self, data):
        for ver in NEARBY_NODES[self.node_num - 1]:
            s = socket.socket()
            s.connect((self.host, SERVER_PORTS[int(ver) - 1]))
            s.sendall(data)
            s.close()

    def update_distance_matrix(self, ver, matrix):
        for k in range(1, NUM_NODES + 1):
            self.current_distance_vector[k] = min(self.current_distance_vector[k], self.current_distance_vector[ver] + matrix[k])
        UPDATE_MATRIX[self.node_num] = 1
        global ANSWER_MATRIX
        ANSWER_MATRIX[self.node_num - 1] = self.current_distance_vector


def read_file(name):
    global TERMINATE_TIME
    file = open(name, 'r')
    neighbor_data = file.read().split('\n')
    file.close()
    cost = []
    for idx, row in enumerate(neighbor_data):
        if len(row) == 0:
            continue
        arr = row.strip().split(" ")
        if idx == 0:
            val = int(arr[0])
            global NUM_NODES
            NUM_NODES = val
            for i in range(0, val + 1):
                new_row = []
                for j in range(0, val + 1):
                    new_row.append(float('inf'))
                cost.append(new_row)
        if idx > 0:
            val = int(arr[0])
            neighbors = []
            for x in range(0, val):
                v = int(arr[2 * x + 1])
                c = float(arr[2 * x + 2])
                neighbors.append(v)
                cost[idx][v] = c
            NEARBY_NODES.append(neighbors)
    return cost


def main(argv):
    file_name = argv[0]
    cost_matrix = read_file(file_name)
    global old_time
    old_time = time.time()
    global ANSWER_MATRIX
    global ANSWER_MATRIX_BASE
    global UPDATE_MATRIX
    global UPDATE_MATRIX_BASE
    ANSWER_MATRIX_BASE = Array(ctypes.c_float, (NUM_NODES + 1) * (NUM_NODES + 1))
    ANSWER_MATRIX = np.ctypeslib.as_array(ANSWER_MATRIX_BASE.get_obj())
    ANSWER_MATRIX = ANSWER_MATRIX.reshape((NUM_NODES + 1), (NUM_NODES + 1))
    for i in range(NUM_NODES + 1):
        for j in range(NUM_NODES + 1):
            ANSWER_MATRIX[i][j] = float(float('inf'))
    UPDATE_MATRIX_BASE = Array(ctypes.c_int, NUM_NODES + 1)
    UPDATE_MATRIX = np.ctypeslib.as_array(UPDATE_MATRIX_BASE.get_obj())
    for i in range(NUM_NODES):
        x = Node(i + 1, cost_matrix[i + 1], NEARBY_NODES[i])
        t = Process(target=x.node_server)
        t2 = Process(target=x.node_client)
        t.start()
        t2.start()
        t.join()
        t2.join()
    new_file = "output_" + file_name
    with open(new_file, 'w') as f:
        f.write(str(NUM_NODES) + '\n')
        for i in range(NUM_NODES):
            st = "" + str(NUM_NODES - 1) + " "
            for j in range(1, NUM_NODES + 1):
                if (i + 1) != j:
                    st += str(j) + " " + str(ANSWER_MATRIX[i][j]) + " "
            f.write(st + '\n')
