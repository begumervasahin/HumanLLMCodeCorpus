from PIL import Image
import numpy as np
import sys

def load_image(infilename):
    """Load an image file and convert it to grayscale."""
    img = Image.open(infilename)
    img_bw = img.convert('L')
    data = np.asarray(img_bw, dtype=np.float32)
    return data

def add_block(array, block_x=(0,0), block_y=(0,0)):
    """Add a block of high energy to the image array."""
    array[block_y[0]:block_y[1], block_x[0]:block_x[1]] = np.inf
    return array

def energy(I, plot=False, block=False):
    """Calculate the energy of the image."""
    I_row, I_col = I.shape
    C = np.zeros_like(I)

    for i in range(1, I_row-1):
        for j in range(1, I_col-1):
            C[i, j] = abs(I[i-1, j] - I[i, j]) + abs(I[i+1, j] - I[i, j]) + \
                       abs(I[i, j-1] - I[i, j]) + abs(I[i, j+1] - I[i, j])
            C[0, j] = C[1, j]
            C[-1, j] = C[-2, j]
        C[i, 0] = C[i, 1]
        C[i, -1] = C[i, -2]

    for j in range(I_col):
        C[0, j] = C[1, j]
        C[-1, j] = C[-2, j]

    if block:
        C = add_block(C, block_x=(730, 1000), block_y=(100, 350))
        C = add_block(C, block_x=(0, 370), block_y=(80, 350))

    if plot:
        img = Image.fromarray(C)
        img.show()

    E = np.zeros_like(I) + C

    for i in range(I_row-1):
        for j in range(I_col):
            if j == 0:
                E[i+1, j] += min(E[i, j], E[i, j+1])
            elif j == I_col-1:
                E[i+1, j] += min(E[i, j], E[i, j-1])
            else:
                E[i+1, j] += min(E[i, j-1], E[i, j], E[i, j+1])

    return E

def seams(E, reverse=False):
    """Find the optimal seam."""
    if reverse:
        col = len(E[0]) - np.argmin(E[0][::-1]) - 1
    else:
        col = np.argmin(E[0])

    Ppath = [col]

    for row in range(len(E)-1):
        if col == 0:
            l = min(E[row+1, col], E[row+1, col+1])
            if l == E[row+1, col]:
                Ppath.append(col)
            else:
                Ppath.append(col+1)
                col += 1
        elif col == len(E[0]) - 1:
            l = min(E[row+1, col], E[row+1, col-1])
            if l == E[row+1, col]:
                Ppath.append(col)
            else:
                Ppath.append(col-1)
                col -= 1
        else:
            l = min(E[row+1, col-1], E[row+1, col], E[row+1, col+1])
            if l == E[row+1, col]:
                Ppath.append(col)
            elif l == E[row+1, col-1]:
                Ppath.append(col-1)
                col -= 1
            else:
                Ppath.append(col+1)
                col += 1

    return Ppath

def delete(E, I, Ppath):
    """Delete the pixels along the seam."""
    for row in range(len(E)):
        np.delete(E[row], Ppath[row])
        np.delete(I[row], Ppath[row])

    return E, I
