import time
import argparse
import sys

def solve_knapsack(infinity=9999999):
    start_time = time.process_time()

    # Parse command-line arguments
    parser = argparse.ArgumentParser(description="Solves the 0/1 knapsack problem for a given set of items.")
    parser.add_argument("-d", help="display distances from the source edge", action="store_true")
    parser.add_argument("-p", help="display paths to each vertex from the source", action="store_true")
    parser.add_argument("-i", help="enable custom infinity value", action="store_true") 
    parser.add_argument("-t", help="display execution time", action="store_true")       
    parser.add_argument("filename", help="input file in .txt format")
    args = parser.parse_args()

    try:
        # Read data from the input file
        with open(args.filename, 'r') as file:
            raw_data = [list(map(int, line.split())) for line in file.readlines()]      
    except FileNotFoundError:
        print("Error: File not found.")
        sys.exit()

    # Get the source node from the user
    while True:
        source = int(input("Enter the source node: "))
        if 1 <= source <= raw_data[0][0]:
            source -= 1  # Adjust index for 0-based indexing
            break
        else:
            print("Invalid source node. Please enter a valid node number.")

    # Set the infinity value
    infinity_value = infinity if args.i else float('inf')

    # Extract data and initialize variables
    data = [(i[0], i[1], i[2]) for i in raw_data[1:]]
    temp_distances = [infinity_value if i != source else 0 for i in range(raw_data[0][0])]
    paths = [[source + 1] for _ in range(raw_data[0][0])]
    negative_cycle = False

    # Bellman-Ford algorithm for finding shortest paths
    for _ in range(len(temp_distances)):
        for edge in [(i[0], i[1]) for i in data]:
            weight = next(j[2] for j in data if {edge[0], edge[1]} == {j[0], j[1]})
            if not (weight < 0 and temp_distances[edge[0] - 1] == infinity_value):      
                if temp_distances[edge[0] - 1] + weight < temp_distances[edge[1] - 1]:  
                    temp_distances[edge[1] - 1] = temp_distances[edge[0] - 1] + weight  
                    paths[edge[1] - 1].extend(paths[edge[0] - 1])
                    paths[edge[1] - 1].append(edge[1])

    # Check for negative weight cycles
    for edge in [(i[0], i[1]) for i in data]:
        weight = next(j[2] for j in data if {edge[0], edge[1]} == {j[0], j[1]})
        if not (weight < 0 and temp_distances[edge[0] - 1] == infinity_value):
            if temp_distances[edge[1] - 1] > temp_distances[edge[0] - 1] + weight:      
                negative_cycle = True
                print("Error: The graph contains a negative weight cycle.")
                break

    # Display results based on command-line options
    if not negative_cycle:
        temp_paths = [paths[i][len(paths[i]) - paths[i][::-1].index(source + 1) - 1:] for i in range(len(paths))]
        final_paths = [tuple(temp_paths[i]) if temp_distances[i] != infinity_value else 'infinity' for i in range(len(temp_distances))]
        final_distances = [temp_distances[i] if temp_distances[i] != infinity_value else 'infinity' for i in range(len(temp_distances))]
        if args.d:
            print("Distances from the source edge: {}".format(final_distances))
        if args.p:
            print("Paths to each vertex from the source: {}".format(final_paths))       

    # Display execution time if requested
    if args.t:
        print("--- Execution time: {} seconds ---".format(time.process_time() - start_time))

if __name__ == "__main__":
    solve_knapsack()
