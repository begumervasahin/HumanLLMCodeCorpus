import argparse
import re
import sys
import time
import math

# Define command-line arguments
parser = argparse.ArgumentParser(description='Calculate the shortest path between all pairs of vertices in a graph')
parser.add_argument('--algorithm', default='a', help='Algorithm: Select the algorithm to run, default is all. (a)ll, (b)ellman-ford only or (f)loyd-warshall only')
parser.add_argument('-v', '--verbose', action='store_true')
parser.add_argument('--profile', action='store_true')
parser.add_argument('filename', metavar='<filename>', help='Input file containing graph')

# Regular expressions for parsing graph input
graph_pattern = re.compile(r'(\d+)\s(\d+)')
edge_pattern = re.compile(r'(\d+)\s(\d+)\s(-?\d+)')
vertices = []
edges = []

def bellman_ford(graph):
    path_pairs = []
    edges_list = []
    graph_vertices, graph_edges = graph
    num_vertices = len(graph_vertices)

    for i in range(num_vertices):
        for j in range(num_vertices):
            if math.isinf(float(graph_edges[i][j])):
                edges_list.append([(int(i), int(j)), float(graph_edges[i][j])])

    for k in range(num_vertices):
        dist = [float('inf')] * num_vertices
        dist[k] = 0

        for _ in range(num_vertices - 1):
            for edge in edges_list:
                if dist[edge[0][1]] > dist[edge[0][0]] + edge[1]:
                    dist[edge[0][1]] = dist[edge[0][0]] + edge[1]

        for edge in edges_list:
            if dist[edge[0][1]] > dist[edge[0][0]] + edge[1]:
                print("negative circle")
                return 0

        for x in range(num_vertices):
            path_pairs.append(((k, x), dist[x]))

    print(path_pairs)
    return path_pairs

def floyd_warshall(graph):
    path_pairs = []
    graph_vertices, graph_edges = graph
    num_vertices = len(graph_vertices)
    d = [[float('inf')] * num_vertices for _ in range(num_vertices)]

    for i in range(num_vertices):
        d[i][i] = 0
        for j in range(num_vertices):
            d[i][j] = float(graph_edges[i][j])

    for k in range(num_vertices):
        for i in range(num_vertices):
            for j in range(num_vertices):
                if d[i][j] > d[i][k] + d[k][j]:
                    d[i][j] = d[i][k] + d[k][j]

    for i in range(num_vertices):
        if d[i][i] < 0:
            print("negative circle")
            return 0

    for i in range(num_vertices):
        for j in range(num_vertices):
            path_pairs.append(((i, j), d[i][j]))

    print(path_pairs)
    return path_pairs

def read_graph(filename):
    global vertices, edges
    with open(filename, 'r') as file:
        line1 = file.readline()
        graph_match = graph_pattern.match(line1)
        if not graph_match:
            print(line1 + " not properly formatted")
            sys.exit(1)

        num_vertices = int(graph_match.group(1))
        vertices = list(range(num_vertices))
        edges = [[float('inf')] * num_vertices for _ in range(num_vertices)]

        for line in file.readlines():
            line = line.strip()
            edge_match = edge_pattern.match(line)
            if edge_match:
                source = int(edge_match.group(1)) - 1
                sink = int(edge_match.group(2)) - 1
                weight = float(edge_match.group(3))
                edges[source][sink] = weight

    return vertices, edges

def main(filename, algorithm):
    global vertices, edges
    start_time = time.time()
    graph = read_graph(filename)

    if algorithm == 'a':
        print('running both')
        bellman_ford_result = bellman_ford(graph)
        floyd_warshall_result = floyd_warshall(graph)
        print("--- %s seconds ---" % (time.time() - start_time))

if __name__ == '__main__':
    args = parser.parse_args()
    main(args.filename, args.algorithm)
