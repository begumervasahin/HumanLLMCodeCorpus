import sys

# Constants for simulation parameters
INFINITY = float('inf')
MAX_SEGMENT_SIZE = 4096  # Renamed for clarity

# Command Identifiers for network operations
COMMAND_LINK_DOWN = 'LINKDOWN'
COMMAND_LINK_UP = 'LINKUP'
COMMAND_CHANGE_COST = 'CHANGECOST'
COMMAND_SHOW_ROUTING_TABLE = 'SHOWRT'
COMMAND_CLOSE = 'CLOSE'
COMMAND_TRANSFER = 'TRANSFER'

class NetworkNode:
    """
    Simulates a network node with capabilities for modifying link states,
    changing link costs, displaying the routing table, and transferring files.
    """
    
    def __init__(self):
        self.routing_table = {}  # Placeholder for routing table data structure

    def link_down(self, link_id):
        """Simulate taking a network link down."""
        print(f"Link {link_id} down.")

    def link_up(self, link_id):
        """Simulate bringing a network link up."""
        print(f"Link {link_id} up.")

    def change_cost(self, link_id, new_cost):
        """Change the cost of a network link."""
        print(f"Changing cost of link {link_id} to {new_cost}.")

    def show_routing_table(self):
        """Display the current routing table."""
        print("Current routing table:")

    def close(self):
        """Close the network node and exit the simulation."""
        print("Closing node.")
        sys.exit()

    def transfer_file(self, file_name, destination_node):
        """Simulate transferring a file to another node."""
        print(f"Transferring file {file_name} to {destination_node}.")

    def handle_command(self, command, *args):
        """Route commands to their appropriate handler methods."""
        command_mapping = {
            COMMAND_LINK_DOWN: self.link_down,
            COMMAND_LINK_UP: self.link_up,
            COMMAND_CHANGE_COST: self.change_cost,
            COMMAND_SHOW_ROUTING_TABLE: self.show_routing_table,
            COMMAND_CLOSE: self.close,
            COMMAND_TRANSFER: self.transfer_file
        }

        try:
            command_mapping[command](*args)
        except KeyError:
            print("Unknown command:", command)
        except Exception as e:
            print(f"Error handling command {command}: {e}")

if __name__ == "__main__":
    node = NetworkNode()
    node.handle_command(COMMAND_LINK_UP, '1')
    node.handle_command(COMMAND_CHANGE_COST, '1', 10)
    node.handle_command(COMMAND_SHOW_ROUTING_TABLE)
    node.handle_command(COMMAND_TRANSFER, 'file.txt', 'NodeB')
