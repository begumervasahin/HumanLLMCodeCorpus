import sys
import re
import time
import itertools

# Regular expressions for parsing input
GRAPH_RE = re.compile(r"(\d+)\s(\d+)")
EDGE_RE = re.compile(r"(\d+)\s(\d+)\s(-?\d+)")

def bellman_ford(graph):
    path_pairs = []
    print("\nEntered Bellman-Ford Algorithm\n")

    vertices = graph[0]
    edges = graph[1]

    infinity = float('inf')
    distances = [infinity] * len(vertices)
    distances[0] = 0

    print("Initial distances:", distances)

    vertex_permutations = list(itertools.permutations(vertices, 2))

    for _ in range(1, len(vertices)):
        for u, v in vertex_permutations:
            if distances[v] > distances[u] + float(edges[u][v]):
                distances[v] = distances[u] + float(edges[u][v])

    print("\nDistances after Bellman-Ford:", distances)

    print("\nChecking if negative cycle exists:")
    negative_cycle = any(distances[v] > distances[u] + float(edges[u][v]) for u, v in vertex_permutations)

    if negative_cycle:
        print("A negative cycle exists")
    else:
        print("No negative cycles exist")

    path_pairs = [(('0', '0'), '0')]
    print(path_pairs)
    print("\nExiting Bellman-Ford Algorithm")
    return path_pairs

def read_file(filename):
    vertices = []
    edges = []

    with open(filename, 'r') as infile:
        line1 = infile.readline()
        graph_match = GRAPH_RE.match(line1)
        if not graph_match:
            print(line1 + " not properly formatted")
            quit(1)

        num_vertices = int(graph_match.group(1))
        vertices = list(range(num_vertices))

        for _ in range(num_vertices):
            row = [float("inf")] * num_vertices
            edges.append(row)

        for line in infile.readlines():
            line = line.strip()
            edge_match = EDGE_RE.match(line)
            if edge_match:
                source = int(edge_match.group(1))
                sink = int(edge_match.group(2))
                weight = int(edge_match.group(3))
                edges[source][sink] = weight

    print("\nVertices    Edges")
    for i in range(len(edges)):
        print(str(vertices[i]) + "         " + str(edges[i]))

    return (vertices, edges)

def main(filename, algorithm):
    algorithm = algorithm[1:]
    graph = read_file(filename)
    print("\nAlgorithm:", algorithm)

    if algorithm in ('b', 'B'):
        print("\nAbout to call Bellman-Ford Algorithm")
        bellman_ford(graph)
        print("\nCalled Bellman-Ford Algorithm\n")
    elif algorithm in ('f', 'F'):
        floyd_warshall(graph)
    elif algorithm == "both":
        start_time = time.process_time()
        bellman_ford(graph)
        end_time = time.process_time()
        bellman_ford_time = end_time - start_time

        start_time = time.process_time()
        floyd_warshall(graph)
        end_time = time.process_time()
        floyd_warshall_time = end_time - start_time

        print("Bellman-Ford Algorithm Timing:", bellman_ford_time)
        print("Floyd-Warshall Algorithm Timing:", floyd_warshall_time)

if __name__ == '__main__':
    if len(sys.argv) < 3:
        print("Usage: python bellman_ford.py -<f|b> <input_file>")
        quit(1)

    main(sys.argv[2], sys.argv[1])
