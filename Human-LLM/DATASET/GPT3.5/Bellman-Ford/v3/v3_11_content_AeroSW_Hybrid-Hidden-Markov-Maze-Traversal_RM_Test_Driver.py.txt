from Reinforcement_Maze import Reinforcement_Maze
from State_Controller import State_Controller
from State import State
from Action import Action
import sys
import time

class RM_Compare_Driver:
    @staticmethod
    def main(arguments):
        # Default values for tolerance and number of trials
        default_tol = 0.00001
        default_num_trials = 10000

        # Check and update values based on command line arguments
        tol, num_trials = RM_Compare_Driver.parse_arguments(arguments, default_tol, default_num_trials)

        # Read reinforcement information from file
        s_controller, gamma = RM_Compare_Driver.read_reinforcement_information(arguments[1])

        # Create lists of Reinforcement_Maze instances for comparison
        r_maze_list_1 = [Reinforcement_Maze(gamma, s_controller) for _ in range(8)]
        r_maze_list_2 = [Reinforcement_Maze(gamma, s_controller) for _ in range(8)]

        # Calculate utilities and store them in dictionaries
        util_dict_1 = {}
        util_dict_2 = {}

        # Calculate utilities for r_maze_list_1
        time_1_total = RM_Compare_Driver.calculate_utilities(r_maze_list_1, tol, num_trials, util_dict_1)

        # Calculate utilities for r_maze_list_2
        time_2_total = RM_Compare_Driver.calculate_utilities(r_maze_list_2, tol, num_trials, util_dict_2, adjusted=True)

        return time_1_total, time_2_total

    @staticmethod
    def parse_arguments(arguments, default_tol, default_num_trials):
        tol = default_tol
        num_trials = default_num_trials

        if len(arguments) == 4:
            tol = float(arguments[2])
            num_trials = int(arguments[3])
        elif len(arguments) != 2:
            raise IndexError

        return tol, num_trials

    @staticmethod
    def read_reinforcement_information(file_name):
        # Read and process reinforcement information from file
        with open(file_name, 'r') as my_file:
            num_actions = int(my_file.readline())
            num_states = int(my_file.readline())
            gamma = float(my_file.readline())

            s_controller = State_Controller(curr_state=None, state_list=None)
            file_list = []

            # Read state information
            for i in range(num_states):
                value = float(my_file.readline())
                terminality = int(my_file.readline())
                file_of_state = my_file.readline().rstrip()
                file_list.append(file_of_state)
                temp_state = State(value, terminality, i + 1, action_list=None, count_actions=True)
                added = s_controller.addState(temp_state)

            # Read action information for each state
            for i, file_name in enumerate(file_list):
                swapped = s_controller.switchState(i + 1)
                if s_controller.isTerminal():
                    continue
                with open(file_name, 'r') as state_file:
                    size = state_file.readline()
                    num_actions_state, num_states_state = map(int, size.split(' '))     
                    if num_actions_state != num_actions or num_states_state != num_states:
                        raise MemoryError

                    for count in range(num_actions):
                        line = state_file.readline()
                        myList = list(map(float, line.split(' ')))

                        if len(myList) != num_states:
                            raise MemoryError

                        t_list = []
                        for q in range(len(myList)):
                            s_controller.switchState(q + 1)
                            temp_state = s_controller.getState()
                            t_list.append(s_controller.getState())
                            myList[q] = float(myList[q])

                        s_controller.switchState(i + 1)
                        new_action = Action(count + 1, t_list, result_chances=myList, create_counter_table=True)
                        s_controller.addAction(new_action)

        return s_controller, gamma

    @staticmethod
    def calculate_utilities(r_maze_list, tol, num_trials, util_dict, adjusted=False):
        time_initial = time.time()

        for i, r_maze in enumerate(r_maze_list):
            action_type = i + (9 if adjusted else 1)
            if adjusted:
                util_dict[action_type] = r_maze.createListUtilitiesAdjusted(tol, num_trials, 200)
            else:
                util_dict[action_type] = r_maze.createListUtilities(tol, num_trials)

        time_final = time.time()
        time_total = time_final - time_initial

        return time_total

    time_1_totals = 0
