import matplotlib.pyplot as plt
import json
import argparse

INPUT_PATH = "data\\"
OUTPUT_PATH = "visualization\\"

def read_json_data(file_name):
    with open(INPUT_PATH + file_name, "r") as input_file:
        data = json.load(input_file)
    return data

def parse_command_line_arguments():
    parser = argparse.ArgumentParser()
    parser.add_argument('input_file', action='store', type=str, help='Input file name')
    parser.add_argument('output_file', action='store', type=str, help='Output file name')
    return parser.parse_args()

def main():
    args = parse_command_line_arguments()
    print(f"Reading data from {args.input_file}...")
    data = read_json_data(args.input_file)
    print("Data read.")

    x_values = data["edges"]
    y_dijkstra = data["dijkstra"]
    y_bellman_ford = data["bellmanford"]
    y_yen = data["yen"]
    y_fibonacci_heap = data["fibonacciheap"]

    output_name_prefix = args.output_file.split(".")[0]

    def plot_and_save_graph(x, y1, y2, y3, y4, name_suffix, label1, label2, label3, label4, color1, color2, color3, color4):
        output_name = f"{output_name_prefix}-{name_suffix}"
        print(f"Plotting graph for {name_suffix}...")
        plt.plot(x, y1, label=label1, color=color1)
        plt.plot(x, y2, label=label2, color=color2)
        plt.plot(x, y3, label=label3, color=color3)
        plt.plot(x, y4, label=label4, color=color4)
        plt.legend(loc='best')
        plt.xlabel("Edges")
        plt.ylabel("Times (ms)")
        plt.title(output_name.upper())
        print("Generating graph labels...")
        plt.savefig(f"{OUTPUT_PATH}{output_name}.png", bbox_inches="tight")
        print(f"Graph saved for {name_suffix}")
        plt.close()

    plot_and_save_graph(x_values, y_dijkstra, y_bellman_ford, y_yen, y_fibonacci_heap, "all",
                        "Dijkstra", "Bellman-Ford", "FibonacciHeap", "Yen", "b", "orange", "g", "r")
    plot_and_save_graph(x_values, y_dijkstra, y_fibonacci_heap, y_yen, [], "df",        
                        "Dijkstra", "FibonacciHeap", "Yen", "", "b", "g", "r", "")      
    plot_and_save_graph(x_values, [], y_bellman_ford, y_yen, y_fibonacci_heap, "by",    
                        "", "Bellman-Ford", "", "Yen", "orange", "", "r")
    plot_and_save_graph(x_values, y_dijkstra, y_fibonacci_heap, y_yen, [], "dfy",       
                        "Dijkstra", "FibonacciHeap", "Yen", "b", "g", "r")

if __name__ == "__main__":
    main()
