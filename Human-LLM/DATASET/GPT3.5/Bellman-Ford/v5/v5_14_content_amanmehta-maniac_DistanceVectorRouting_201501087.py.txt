import socket
import hashlib
import time
from threading import Thread
from collections import defaultdict

class Server(Thread):
    def __init__(self):
        super().__init__()

    def run(self):
        server_socket = socket.socket()
        port = 60000
        host = ""
        server_socket.bind((host, port))
        server_socket.listen(5)
        conn, addr = server_socket.accept()

        for i in range(1, N):
            for u in range(1, N + 1):
                message = self.construct_message(u)
                hashed_data = hashlib.md5(message.encode()).hexdigest()

                for v in graph[u]:
                    self.send_data(conn, message)
                    self.send_data(conn, str(v))
                    self.send_data(conn, str(u))
                    self.send_data(conn, hashed_data)

        self.send_data(conn, "Bye")
        conn.close()

    def construct_message(self, u):
        message = ""
        for v in range(1, N + 1):
            message += f"{v} {d[u][v]}\n"
        return message

    def send_data(self, conn, data):
        conn.send(data.encode())
        time.sleep(0.01)

class Receiver(Thread):
    def __init__(self):
        super().__init__()

    def run(self):
        distance = defaultdict()
        s = socket.socket()
        host = ""
        port = 60000
        s.connect((host, port))
        s.send('Hello server!')

        while True:
            data = s.recv(1024)
            if not data:
                break
            self.process_data(data)

        s.close()

    def process_data(self, data):
        global d
        data_str = data.decode()
        parts = data_str.split('\n')

        if len(parts) < 4:
            return

        dist = parts[0]
        node = int(parts[1])
        sender = int(parts[2])
        hash_val = parts[3]

        computed_hash = hashlib.md5(dist.encode()).hexdigest()

        if computed_hash == hash_val:
            dist_data = dist.split('\n')
            for i in range(len(dist_data) - 1):
                dist_data[i] = dist_data[i].split()
                dist_data[i][0], dist_data[i][1] = int(dist_data[i][0]), int(dist_data[i][1])
                d[dist_data[i][0]] = dist_data[i][1]

            for j in graph:
                if d[node][j] > distance[j] + graph[sender][node]:
                    d[node][j] = distance[j] + graph[sender][node]
        else:
            print('Some error occurred! Try again.')
