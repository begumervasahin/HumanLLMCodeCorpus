import matplotlib.pyplot as plt
import json
import argparse

INPUT_PATH = "data\\"
OUTPUT_PATH = "visualization\\"

def read_json_data(file_name):
    with open(INPUT_PATH + file_name, "r") as file:
        data = json.load(file)
    return data

def parse_arguments():
    parser = argparse.ArgumentParser()
    parser.add_argument('input_file', type=str, help='Input file name')
    parser.add_argument('output_file', type=str, help='Output file name')
    return parser.parse_args()

def main():
    args = parse_arguments()
    print(f"Reading data from {args.input_file}...")
    data = read_json_data(args.input_file)
    print("Data read.")

    x_values = data["edges"]
    y_dijkstra = data["dijkstra"]
    y_bellman_ford = data["bellmanford"]
    y_yen = data["yen"]
    y_fibonacci_heap = data["fibonacciheap"]

    def plot_graph(x, y1, y2, y3, y4, name_suffix, labels, colors):
        output_name = args.output_file.split(".")[0] + "-" + name_suffix
        print(f"Plotting graph for {name_suffix}...")
        plt.plot(x, y1, label=labels[0], color=colors[0])
        plt.plot(x, y2, label=labels[1], color=colors[1])
        plt.plot(x, y3, label=labels[2], color=colors[2])
        plt.plot(x, y4, label=labels[3], color=colors[3])
        plt.legend(loc='best')
        plt.xlabel("Edges")
        plt.ylabel("Times (ms)")
        plt.title(output_name.upper())
        print("Generating graph labels...")
        plt.savefig(OUTPUT_PATH + output_name + ".png", bbox_inches="tight")
        print(f"Graph saved for {name_suffix}")
        plt.close()

    plot_graph(x_values, y_dijkstra, y_bellman_ford, y_yen, y_fibonacci_heap, "all",
               ["Dijkstra", "Bellman-Ford", "FibonacciHeap", "Yen"], ["b", "orange", "g", "r"])
    plot_graph(x_values, y_dijkstra, y_fibonacci_heap, y_yen, [], "df",
               ["Dijkstra", "FibonacciHeap", "Yen", ""], ["b", "g", "r", ""])
    plot_graph(x_values, [], y_bellman_ford, y_yen, y_fibonacci_heap, "by",
               ["", "Bellman-Ford", "", "Yen"], ["orange", "", "r", ""])
    plot_graph(x_values, y_dijkstra, y_fibonacci_heap, y_yen, [], "dfy",
               ["Dijkstra", "FibonacciHeap", "Yen", ""], ["b", "g", "r", ""])

if __name__ == "__main__":
    main()
