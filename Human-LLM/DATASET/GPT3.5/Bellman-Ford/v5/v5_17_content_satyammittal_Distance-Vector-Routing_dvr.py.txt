import sys
import json
import socket
from multiprocessing import Process, Array
import ctypes
import numpy as np
import time

# Global variables and constants
terminate_time = 12
old_time = 0
num_nodes = 3
near = []
ser_port = []
ans_mat_base = 0
ans_mat = 0
upd_mat_base = 0
upd_mat = 0

class Node:
    def __init__(self, node_num, cur_dv, nar):
        self.node_num = node_num
        self.initialize_distance_vector(cur_dv)
        self.host = ""
        self.server_port = self.get_free_port()
        self.register_server_port()
        self.neigh_ar = nar
        self.connected = {}
        self.set_update_matrix_flag()

    def initialize_distance_vector(self, cur_dv):
        self.cur_dv_base = Array(ctypes.c_float, (num_nodes + 1))
        self.cur_dv = np.ctypeslib.as_array(self.cur_dv_base.get_obj())
        for r in range(num_nodes + 1):
            self.cur_dv[r] = cur_dv[r]

    def get_free_port(self):
        s = socket.socket()
        s.bind(("", 0))
        port = s.getsockname()[1]
        s.close()
        return port

    def register_server_port(self):
        ser_port.append(self.server_port)

    def set_update_matrix_flag(self):
        upd_mat[self.node_num] = 1 if self.node_num == 1 else 1

    def node_server(self):
        s = socket.socket()
        s.bind((self.host, self.server_port))
        s.listen(5)
        while True:
            if time.time() - old_time > 14:
                break
            fl = 0
            s.settimeout(1)
            conn = None
            try:
                conn, addr = s.accept()
                data = conn.recv(10000)
            except (socket.timeout, socket.error):
                fl = 1
            finally:
                if conn:
                    conn.close()
            if fl == 1:
                break
            data = json.loads(data)
            self.update_matrix(data["node"], data["arr"])
        s.close()

    def node_client(self):
        while True:
            if time.time() - old_time > 12:
                break
            while upd_mat[self.node_num] == 0:
                continue
            upd_mat[self.node_num] = 0
            data = json.dumps({"node": self.node_num, "arr": self.cur_dv.tolist()})
            try:
                self.try_conn(data)
            except Exception as e:
                self.try_conn(data)

    def try_conn(self, data):
        for ver in near[self.node_num - 1]:
            s = socket.socket()
            s.connect((self.host, ser_port[int(ver) - 1]))
            s.sendall(data)
            s.close()

    def update_matrix(self, ver, matrix):
        for k in range(1, num_nodes + 1):
            self.cur_dv[k] = min(self.cur_dv[k], self.cur_dv[ver] + matrix[k])
        upd_mat[self.node_num] = 1
        ans_mat[self.node_num - 1] = self.cur_dv


def file_read(name):
    global num_nodes
    with open(name, 'r') as file:
        lines = file.readlines()
    num_nodes = int(lines[0].strip())
    cost = [[float('inf')] * (num_nodes + 1) for _ in range(num_nodes + 1)]
    for line in lines[1:]:
        if line.strip():
            val, *data = map(float, line.strip().split())
            for x in range(0, int(val)):
                v, c = int(data[2 * x]), data[2 * x + 1]
                cost[int(val)][v] = c
                near.append([v])
    return cost


def main(argv):
    file_name = argv[0]
    cost = file_read(file_name)
    threads = []
    global old_time, ans_mat, ans_mat_base, upd_mat, upd_mat_base
    old_time = time.time()
    ans_mat_base = Array(ctypes.c_float, (num_nodes + 1) * (num_nodes + 1))
    ans_mat = np.ctypeslib.as_array(ans_mat_base.get_obj()).reshape((num_nodes + 1), (num_nodes + 1))
    ans_mat[:, :] = float('inf')
    upd_mat_base = Array(ctypes.c_int, num_nodes + 1)
    upd_mat = np.ctypeslib.as_array(upd_mat_base.get_obj())
    
    for i in range(num_nodes):
        x = Node(i + 1, cost[i + 1], near[i])
        threads.append(Process(target=x.node_server))
        threads.append(Process(target=x.node_client))
    
    for t in threads:
        t.start()
    for t in threads:
        t.join()
    
    new_file = "output_" + file_name
    with open(new_file, 'w') as f:
        f.write(f"{num_nodes}\n")
        for i in range(num_nodes):
            st = f"{num_nodes - 1} "
            st += " ".join(f"{j} {ans_mat[i][j]}" for j in range(1, num_nodes + 1) if i + 1 != j) + " \n"
            f.write(st)


if __name__ == "__main__":
    main(sys.argv[1:])
