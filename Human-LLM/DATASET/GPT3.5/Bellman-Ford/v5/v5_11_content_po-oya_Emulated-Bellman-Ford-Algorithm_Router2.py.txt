# Import necessary modules
from BellmanFord import BFA
import msvcrt
import re
import timeit

# Function to read router's name from user input
def get_router_name():
    return input("Welcome to Emulated Bellman-Ford Algorithm\n"
                 "Please enter your router's name:\n")

# Function to read port mapping file and populate dictionaries
def read_port_mapping():
    whichPort = {}
    adrToName = {}
    routerCount = 0
    with open("which_port.txt") as whichRouter:
        for line in whichRouter:
            port = int(line[0])
            router_id = int(line[2:6])
            whichPort[port] = router_id
            adrToName[router_id] = port
            routerCount += 1
    return whichPort, adrToName, routerCount

# Function to read initial cost for the router from the adjacency matrix file
def read_initial_cost(routerName):
    with open("adj_mat.txt") as adjMatrix:
        myLine = adjMatrix.readlines()[int(routerName) - 1]
        initialCost = myLine.rstrip('\n').split(" ")
    print('Initial Cost is {}\n'.format(initialCost))
    return initialCost

# Function to prompt user to start Bellman-Ford Algorithm
def prompt_start_algorithm():
    while True:
        start_prompt = "To start the BellmanFord Algorithm, Enter 's'\n"
        s = input(start_prompt)
        if s == 's':
            break
        else:
            print("Invalid input! Please enter 's' to start the algorithm.")

# Function to handle keyboard input
def handle_keyboard_input(routerName, myBf):
    hit = msvcrt.kbhit()
    if hit:
        key = ord(msvcrt.getch())
        if key == ord('u'):
            with open("adj_mat.txt") as adjMatrix:
                myLine = adjMatrix.readlines()[int(routerName) - 1]
                newCost = myLine.rstrip('\n').split(" ")
                print('New cost is {}\n'.format(newCost))
                myBf.check_cost(newCost)

# Main function
def main():
    # Initialize variables
    start = timeit.default_timer()

    # Get router's name
    routerName = get_router_name()

    # Read port mapping
    whichPort, adrToName, routerCount = read_port_mapping()

    # Read initial cost
    initialCost = read_initial_cost(routerName)

    # Create an instance of the Bellman-Ford Algorithm
    myBf = BFA(routerCount, initialCost, routerName, whichPort, adrToName)
    myBf.who_to_send()

    # Prompt user to start Bellman-Ford Algorithm
    prompt_start_algorithm()

    # Send initial data
    myBf.send()

    # Main loop
    while True:
        # Check if a key has been pressed
        elapsedTime = timeit.default_timer() - start

        # Send updates if more than 0.1 second has passed
        if elapsedTime > 0.1:
            myBf.send()
            start = timeit.default_timer()

        # Receive updates
        myBf.receive()

        # Handle keyboard input
        handle_keyboard_input(routerName, myBf)

# Run the main function
if __name__ == "__main__":
    main()
