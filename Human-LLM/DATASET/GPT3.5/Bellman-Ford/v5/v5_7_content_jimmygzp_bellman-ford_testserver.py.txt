import socket
import sys
import pickle

def receive_data(port):
    """Receive data from sender via UDP."""
    # Create a UDP socket
    recv_socket = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)

    # Allow reusing the address
    recv_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)

    # Bind the socket to the specified port
    recv_socket.bind(("", port))

    # Receive data from the sender (up to 4096 bytes)
    data, sender_address = recv_socket.recvfrom(4096)

    # Deserialize the received data using pickle
    recovered_data = pickle.loads(data)

    # Print sender address and recovered data
    print("Sender:", sender_address)
    print("Recovered Data:", recovered_data)

if __name__ == "__main__":
    # Ensure correct number of arguments provided
    if len(sys.argv) != 2:
        print("Usage: python script_name.py <port>")
        sys.exit(1)

    # Extract the port number from command-line arguments
    port = int(sys.argv[1])

    # Call the function to receive data
    receive_data(port)
