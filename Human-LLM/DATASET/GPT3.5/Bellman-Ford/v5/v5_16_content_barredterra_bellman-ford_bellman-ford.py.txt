import csv

def bellman_ford(edges, distances):
    """
    Computes shortest paths from a single source vertex to all other vertices in a weighted digraph
    using the Bellman-Ford algorithm.

    Args:
    - edges: List of tuples representing edges in the graph (source, destination, weight).
    - distances: Dictionary containing initial distances from the source vertex to other vertices.

    Returns:
    - A tuple containing the final distances after applying the algorithm and a log of states.
    """
    log = []
    num_vertices = len(distances)

    # Run Bellman-Ford algorithm
    for i in range(1, num_vertices - 1):
        for edge in edges:
            source, destination, weight = edge
            if distances[source] + weight < distances[destination]:
                distances[destination] = distances[source] + weight
            log.append(create_log_entry(edge, distances))
        if i == 2:
            return distances, log

def create_log_entry(edge, distances):
    """
    Creates one row for the log containing edge information and distances.

    Args:
    - edge: Tuple representing an edge (source, destination, weight).
    - distances: Dictionary containing distances from the source vertex to other vertices.

    Returns:
    - List representing a log entry with edge information and distances.
    """
    edge_str = f'({edge[0]}-{edge[1]})'
    state = [edge_str] + [distances[vertex] for vertex in sorted(distances.keys())]
    return state

def save_log_to_csv(logs, vertices):
    """
    Produces a CSV file containing the logs.

    Args:
    - logs: List of log entries to be saved.
    - vertices: List of vertices.

    Saves the log entries along with vertex names to a CSV file named 'bf_log.csv'.
    """
    with open('bf_log.csv', 'w', newline='') as csvfile:
        log_writer = csv.writer(csvfile, quoting=csv.QUOTE_MINIMAL)
        log_writer.writerow(['edge'] + vertices)
        log_writer.writerows(logs)
        print('\nLog saved as ' + csvfile.name)

def print_log(log):
    """
    Prints the log entries.

    Args:
    - log: List of log entries.

    Prints the log entries to the console.
    """
    print('Log:\n')
    for entry in log:
        print(entry)

def main():
    # Define the edges and initial distances
    edges = [('a', 'b', 4), ('a', 'c', 1), ('a', 'f', 9), ('b', 'd', 1),
             ('c', 'b', 2), ('c', 'e', 3), ('d', 'e', 1), ('d', 'f', 4),
             ('e', 'b', -2), ('e', 'f', 1)]
    distances = {'a': 0, 'b': 999, 'c': 999, 'd': 999, 'e': 999, 'f': 999}

    # Run Bellman-Ford algorithm and save the log
    result, log = bellman_ford(edges, distances)
    print_log(log)
    vertices = list(distances.keys())
    save_log_to_csv(log, vertices)

if __name__ == '__main__':
    main()
