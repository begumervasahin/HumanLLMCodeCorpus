import os
import re
import sys
import time

# Regular expressions for parsing input data
GRAPH_PATTERN = re.compile(r"(\d+)\s(\d+)")
EDGE_PATTERN = re.compile(r"(\d+)\s(\d+)\s(\d+)")

def bellman_ford(graph_data):
    """
    Perform Bellman-Ford algorithm on the given graph data.

    Args:
    - graph_data: Tuple containing vertices and edge weights of the graph.

    Returns:
    - List of shortest paths from each source vertex.
    """
    paths = []
    vertices, edges = graph_data
    for source in range(len(vertices)):
        distances = [float("inf")] * len(vertices)
        distances[source] = 0
        for _ in range(1, len(vertices) - 1):
            for from_node in range(len(vertices)):
                for to_node in range(len(vertices)):
                    weight = edges[from_node][to_node]
                    if float(weight) < float("inf"):
                        if distances[to_node] > distances[from_node] + float(weight):
                            distances[to_node] = int(distances[from_node]) + int(float(weight))
        paths.append(distances)
    return paths

def floyd_warshall(graph_data):
    """
    Perform Floyd-Warshall algorithm on the given graph data.

    Args:
    - graph_data: Tuple containing vertices and edge weights of the graph.

    Returns:
    - List of shortest paths between all pairs of vertices.
    """
    paths = []
    vertices, edges = graph_data
    distances = [[float("inf") for _ in range(len(vertices))] for _ in range(len(vertices))]
    for i in range(len(vertices)):
        distances[i][i] = 0
        for j in range(len(vertices)):
            if i != j and float(edges[i][j]) < float("inf"):
                distances[i][j] = int(float(edges[i][j]))
    for k in range(len(vertices)):
        for i in range(len(vertices)):
            for j in range(len(vertices)):
                if distances[i][j] > distances[i][k] + distances[k][j]:
                    distances[i][j] = distances[i][k] + distances[k][j]
    paths.append(distances)
    return paths

def read_graph(filename):
    """
    Read graph data from the given file.

    Args:
    - filename: Name of the input file containing graph data.

    Returns:
    - Tuple containing vertices and edge weights of the graph.
    """
    with open(filename, 'r') as infile:
        line1 = infile.readline()
        graph_match = GRAPH_PATTERN.match(line1)
        if not graph_match:
            print(line1 + " not properly formatted")
            sys.exit(1)
        vertices = list(range(int(graph_match.group(1))))
        edges = [[float("inf") for _ in range(len(vertices))] for _ in range(len(vertices))]
        for line in infile.readlines():
            line = line.strip()
            edge_match = EDGE_PATTERN.match(line)
            if edge_match:
                source = int(edge_match.group(1))
                sink = int(edge_match.group(2))
                if source >= len(vertices) or sink >= len(vertices):
                    print(f"Attempting to insert an edge between {source} and {sink} in a graph with {len(vertices)} vertices")
                    sys.exit(1)
                weight = edge_match.group(3)
                edges[source][sink] = weight
    return (vertices, edges)

def write_file(length_matrix, filename):
    """
    Write length matrix to the output file.

    Args:
    - length_matrix: List of shortest paths to write.
    - filename: Name of the output file.
    """
    base_name = os.path.splitext(os.path.split(filename)[1])[0]
    with open(f'output/{base_name}_output.txt', 'w+') as outfile:
        for vertex in length_matrix:
            for length in vertex:
                outfile.write(str(length) + ',')
            outfile.write('\n')

def main(filename, algorithm):
    """
    Main function to run Bellman-Ford or Floyd-Warshall algorithm based on user input.

    Args:
    - filename: Name of the input file.
    - algorithm: Choice of algorithm ('-b' for Bellman-Ford, '-f' for Floyd-Warshall, '-both' for both).

    Example usage: python bellman_ford.py -f input_file.txt
    """
    algorithm = algorithm[1:]
    graph_data = read_graph(filename)
    path_lengths = []
    if algorithm == 'b' or algorithm == 'B':
        path_lengths = bellman_ford(graph_data)
    elif algorithm == 'f' or algorithm == 'F':
        path_lengths = floyd_warshall(graph_data)
    elif algorithm == "both":
        start_time = time.process_time()
        bellman_ford(graph_data)
        bellman_ford_time = time.process_time() - start_time
        floyd_warshall(graph_data)
        floyd_warshall_time = time.process_time() - start_time
        print("Bellman-Ford timing:", bellman_ford_time)
        print("Floyd-Warshall timing:", floyd_warshall_time)
    write_file(path_lengths, filename)

if __name__ == '__main__':
    if len(sys.argv) < 3:
        print("Usage: python bellman_ford.py -<f|b|both> <input_file>")
        sys.exit(1)
    main(sys.argv[2], sys.argv[1])
