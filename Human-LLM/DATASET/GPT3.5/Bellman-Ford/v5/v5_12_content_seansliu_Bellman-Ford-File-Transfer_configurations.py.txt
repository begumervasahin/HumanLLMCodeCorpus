import sys
import logging

# Setup basic configuration for logging
logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')

# Constants for simulation parameters
INFINITY = float('inf')
MSS = 4096  # Maximum Segment Size for simulated TCP-like transmission

# Commands for network operations
COMMANDS = {
    'LINKDOWN': 'link_down',
    'LINKUP': 'link_up',
    'CHANGECOST': 'change_cost',
    'SHOWRT': 'show_routing_table',
    'CLOSE': 'close',
    'TRANSFER': 'transfer_file',
}

class NetworkNode:
    """
    Simulates a network node with capabilities for link management, cost adjustment,
    routing table display, and file transfer.
    """
    def __init__(self):
        """
        Initialize the network node with an empty routing table.
        """
        self.routing_table = {}  # Would contain more details in a full implementation.

    def link_down(self, link_id):
        """
        Simulate taking a network link down.

        :param link_id: Identifier for the link to take down.
        """
        logging.info(f"Link {link_id} down.")

    def link_up(self, link_id):
        """
        Simulate bringing a network link up.

        :param link_id: Identifier for the link to bring up.
        """
        logging.info(f"Link {link_id} up.")

    def change_cost(self, link_id, new_cost):
        """
        Change the cost of a network link.

        :param link_id: Identifier for the link to change cost.
        :param new_cost: New cost for the link.
        """
        logging.info(f"Changing cost of link {link_id} to {new_cost}.")

    def show_routing_table(self):
        """
        Display the current routing table.
        """
        logging.info("Current routing table:")

    def close(self):
        """
        Close the network node, exiting the simulation.
        """
        logging.info("Closing node.")
        sys.exit()

    def transfer_file(self, file_name, destination_node):
        """
        Simulate transferring a file to another node.

        :param file_name: Name of the file to transfer.
        :param destination_node: Node to transfer the file to.
        """
        logging.info(f"Transferring file {file_name} to {destination_node}.")

    def handle_command(self, command, *args):
        """
        Route commands to the appropriate method based on the command received.
        
        :param command: Command string indicating the action to perform.
        :param args: Arguments for the command.
        """
        method_name = COMMANDS.get(command.upper())
        if method_name and hasattr(self, method_name):
            getattr(self, method_name)(*args)
        else:
            logging.error(f"Unknown command: {command}")
