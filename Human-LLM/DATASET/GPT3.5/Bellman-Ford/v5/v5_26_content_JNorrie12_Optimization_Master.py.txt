from PIL import Image
import numpy as np
import sys

def load_image(filename):
    """
    Load an image file and convert it to grayscale.
    """
    img = Image.open(filename)
    img_bw = img.convert('L')
    data = np.asarray(img_bw, dtype=np.float32)
    return data

def add_block(array, block_x=(0, 0), block_y=(0, 0)):
    """
    Add a block of high energy to the image array.
    """
    array[range(*block_y), range(*block_x)] = np.inf
    return array

def calculate_energy(image):
    """
    Calculate the energy of the image.
    """
    rows, cols = image.shape
    energy_matrix = np.zeros_like(image)

    for i in range(1, rows - 1):
        for j in range(1, cols - 1):
            energy_matrix[i][j] = abs(image[i-1][j] - image[i][j]) + \
                                  abs(image[i+1][j] - image[i][j]) + \
                                  abs(image[i][j-1] - image[i][j]) + \
                                  abs(image[i][j+1] - image[i][j])

    for j in range(cols):
        energy_matrix[0][j] = energy_matrix[1][j]
        energy_matrix[-1][j] = energy_matrix[-2][j]

    return energy_matrix

def find_seam(energy_matrix, reverse=False):
    """
    Find the optimal seam.
    """
    if reverse:
        col = len(energy_matrix[0]) - np.argmin(energy_matrix[0][::-1]) - 1
    else:
        col = np.argmin(energy_matrix[0])

    seam_path = [col]

    for row in range(len(energy_matrix) - 1):
        if col == 0:
            l = min(energy_matrix[row+1][col], energy_matrix[row+1][col+1])
            if l == energy_matrix[row+1][col]:
                seam_path.append(col)
            else:
                seam_path.append(col + 1)
                col += 1
        elif col == len(energy_matrix[0]) - 1:
            l = min(energy_matrix[row+1][col], energy_matrix[row+1][col-1])
            if l == energy_matrix[row+1][col]:
                seam_path.append(col)
            else:
                seam_path.append(col - 1)
                col -= 1
        else:
            l = min(energy_matrix[row+1][col-1], energy_matrix[row+1][col], energy_matrix[row+1][col+1])
            if l == energy_matrix[row+1][col]:
                seam_path.append(col)
            elif l == energy_matrix[row+1][col-1]:
                seam_path.append(col - 1)
                col -= 1
            else:
                seam_path.append(col + 1)
                col += 1

    return seam_path

def remove_seam(energy_matrix, image, seam_path):
    """
    Remove the pixels along the seam.
    """
    energy_matrix = np.delete(energy_matrix, seam_path, axis=1)
    image = np.delete(image, seam_path, axis=1)
    return energy_matrix, image

if __name__ == '__main__':
    input_arg = sys.argv[1]

    if input_arg in ['1', '2', '4']:
        image = load_image("penguins.png")

        if input_arg == '4':
            energy_matrix = calculate_energy(image)
            energy_matrix = add_block(energy_matrix, block_x=(830, 1000), block_y=(599, 600))
            energy_matrix = add_block(energy_matrix, block_x=(0, 250), block_y=(599, 600))
        else:
            energy_matrix = calculate_energy(image)

        image = image.tolist()
        energy_matrix = energy_matrix.tolist()

        for _ in range(300):
            if input_arg in ['1', '2']:
                reverse = (input_arg == '2')
                seam_path = find_seam(energy_matrix, reverse=reverse)
                energy_matrix, image = remove_seam(energy_matrix, image, seam_path)

        image = np.array(image, dtype=np.float32)
        img = Image.fromarray(image)
        img.show()
