import argparse
import re
import math
import cProfile
import time

# Regular expressions for parsing input
GRAPH_REGEX = re.compile(r"(\d+)\s(\d+)")
EDGE_REGEX = re.compile(r"(\d+)\s(\d+)\s(-?\d+)")

def parse_arguments():
    """Parse command-line arguments."""
    parser = argparse.ArgumentParser(description='Calculate the shortest path between all pairs of vertices in a graph')
    parser.add_argument('--algorithm', default='a', help='Algorithm: Select the algorithm to run, default is all. (a)ll, (b)ellman-ford only or (f)loyd-warshall only')
    parser.add_argument('-v', '--verbose', action='store_true')
    parser.add_argument('--profile', action='store_true')
    parser.add_argument('filename', metavar='<filename>', help='Input file containing graph')
    return parser.parse_args()

def read_graph(filename):
    """Read the graph from the input file."""
    vertices = []
    edges = []
    with open(filename, 'r') as infile:
        line1 = infile.readline()
        graph_match = GRAPH_REGEX.match(line1)
        if not graph_match:
            print(f"Error: First line of {filename} is not properly formatted.")
            exit(1)
        num_vertices = int(graph_match.group(1))
        vertices = list(range(num_vertices))
        # Initialize edges with infinity
        edges = [[float("inf")] * num_vertices for _ in range(num_vertices)]
        # Read edges
        for line in infile.readlines():
            line = line.strip()
            edge_match = EDGE_REGEX.match(line)
            if edge_match:
                source, sink, weight = map(int, edge_match.groups())
                if source > num_vertices or sink > num_vertices:
                    print(f"Error: Attempting to insert an edge between {source} and {sink} in a graph with {num_vertices} vertices")
                    exit(1)
                edges[source - 1][sink - 1] = weight
    return vertices, edges

def bellman_ford(vertices, edges):
    """Implementation of the Bellman-Ford algorithm."""
    path_pairs = []
    num_vertices = len(vertices)
    for k in range(num_vertices):
        dist = [float("inf")] * num_vertices
        dist[k] = 0
        for _ in range(num_vertices - 1):
            for i in range(num_vertices):
                for j in range(num_vertices):
                    if dist[j] > dist[i] + edges[i][j]:
                        dist[j] = dist[i] + edges[i][j]
    # Check for negative cycles
    for i in range(num_vertices):
        for j in range(num_vertices):
            if dist[j] > dist[i] + edges[i][j]:
                print("Negative cycle detected")
                return 0
    # Construct path pairs
    for x in range(num_vertices):
        path_pairs.append(((k, x), dist[x]))
    return path_pairs

def floyd_warshall(vertices, edges):
    """Implementation of the Floyd-Warshall algorithm."""
    path_pairs = []
    num_vertices = len(vertices)
    d = [[edges[i][j] for j in range(num_vertices)] for i in range(num_vertices)]
    for k in range(num_vertices):
        for i in range(num_vertices):
            for j in range(num_vertices):
                if d[i][j] > d[i][k] + d[k][j]:
                    d[i][j] = d[i][k] + d[k][j]
    # Check for negative cycles
    for i in range(num_vertices):
        if d[i][i] < 0:
            print("Negative cycle detected")
            return 0
    # Construct path pairs
    for i in range(num_vertices):
        for j in range(num_vertices):
            path_pairs.append(((i, j), d[i][j]))
    return path_pairs

def main():
    """Main function."""
    args = parse_arguments()
    vertices, edges = read_graph(args.filename)
    if args.algorithm == 'a':
        print('Running both algorithms')
        start_time = time.time()
        bellman_ford_results = bellman_ford(vertices, edges)
        floyd_warshall_results = floyd_warshall(vertices, edges)
        print("--- %s seconds ---" % (time.time() - start_time))

if __name__ == '__main__':
    main()
