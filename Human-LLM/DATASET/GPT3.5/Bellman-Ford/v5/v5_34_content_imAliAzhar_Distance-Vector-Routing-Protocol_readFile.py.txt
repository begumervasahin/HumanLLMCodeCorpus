def parse_line(line):
    """
    Parse a line of input to extract neighbor, weight, and port information.
    """
    words = line.split()
    neighbor = words[0]
    weight = float(words[1])
    port = int(words[2])
    return neighbor, {'cost': weight, 'port': port}

def read_file(filename):
    """
    Read data from a file and store neighbor information in a dictionary.
    """
    # Construct the full file path
    filepath = "C:\\" + filename

    # Open the file in read mode
    with open(filepath, 'r') as in_file:
        neighbors = {}  # Dictionary to store neighbor information
        line_counter = 0  # Counter to track lines in the file

        # Iterate through each line in the file
        for line in in_file:
            if line_counter == 0:
                # First line contains the number of neighbors
                num_neighbors = int(line)
                line_counter += 1
            else:
                # Parse the line and store data in the neighbors dictionary
                neighbor, data = parse_line(line)
                neighbors[neighbor] = data

    return neighbors

# Example usage
filename = "example.txt"
neighbors_data = read_file(filename)
