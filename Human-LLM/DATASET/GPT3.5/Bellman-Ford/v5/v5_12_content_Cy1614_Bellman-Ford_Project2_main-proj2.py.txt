import numpy as np
import csv

def BellmanFord(start, end, weights):
    n = len(weights)
    dist = [float('inf')] * n
    pred = [None] * n
    dist[start] = 0

    for _ in range(n - 1):
        for u in range(n):
            for v in range(n):
                if weights[u][v] != 0 and dist[u] + weights[u][v] < dist[v]:
                    dist[v] = dist[u] + weights[u][v]
                    pred[v] = u

    path = [end]
    while pred[end] is not None:
        path.append(pred[end])
        end = pred[end]
    return dist, path[::-1]

def calculateWeights(start_nodes, end_nodes, durations):
    nodes = max(max(start_nodes), max(end_nodes)) + 1
    weights = np.zeros((nodes, nodes))
    for s, f, d in zip(start_nodes, end_nodes, durations):
        weights[s][f] = d
    return weights
