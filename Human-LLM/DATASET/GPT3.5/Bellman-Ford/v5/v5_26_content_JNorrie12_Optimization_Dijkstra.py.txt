import numpy as np
import csv
import time
import math
import sys

def dijkstra(ist, isp, wei):
    if ist == isp:
        return [ist]

    num_vertices = len(wei)
    infinity = sys.maxsize
    unvisited = np.ones(num_vertices, dtype=int)
    costs = np.ones(num_vertices) * float('inf')
    parents = -np.ones(num_vertices, dtype=int) * infinity

    current_node = ist
    costs[current_node] = 0
    unvisited[current_node] = 0
    neighbors = np.nonzero(unvisited * wei[current_node, :])[0]
    remaining_nodes = np.count_nonzero(unvisited)

    while remaining_nodes != 0:
        for neighbor in neighbors:
            new_cost = costs[current_node] + wei[current_node, neighbor]
            if new_cost < costs[neighbor]:
                costs[neighbor] = new_cost
                parents[neighbor] = current_node

        unvisited_nodes = np.nonzero(unvisited)[0]
        min_cost, min_index = costs[unvisited_nodes].min(), costs[unvisited_nodes].argmin()
        current_node = unvisited_nodes[min_index]
        unvisited[current_node] = 0
        neighbors = np.nonzero(unvisited * wei[current_node, :])[0]
        remaining_nodes = np.count_nonzero(unvisited)

    shortest_path = [isp]
    while parents[isp] != ist:
        shortest_path.append(parents[isp])
        isp = parents[isp]
    shortest_path.append(ist)
    return shortest_path[::-1]

def calculate_weights(RX, RY, RA, RB, RV):
    num_vertices = len(RX)
    weights = np.zeros((num_vertices, num_vertices), dtype=float)
    num_edges = len(RA)

    for i in range(num_edges):
        xa, ya = RX[RA[i]-1], RY[RA[i]-1]
        xb, yb = RX[RB[i]-1], RY[RB[i]-1]
        distance = math.sqrt((xb-xa)**2 + (yb-ya)**2)
        time_taken = distance / RV[i]
        weights[RA[i]-1, RB[i]-1] = time_taken

    return weights
