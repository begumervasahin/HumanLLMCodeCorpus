from State_Controller import State_Controller
import random
import math

class Reinforcement_Maze:
    def __init__(self, gamma, state_controller):
        self.state_controller = state_controller
        self.num_states = self.state_controller.getNumStates()
        self.gamma = gamma
        self.utility_list = {}
        self.policy_list = {}
        self.initialize_utility_and_policy_lists()

    def initialize_utility_and_policy_lists(self):
        for state_key in range(1, self.num_states + 1):
            self.utility_list[state_key] = 0
            self.policy_list[state_key] = 0

    def create_adjusted_utility_list(self, value=1.0):
        adjusted_utilities = {}
        for key, value in self.utility_list.items():
            adjusted_utilities[key] = value + value
        return adjusted_utilities

    def create_copy_utility_list(self):
        copied_utility_list = {}
        for key, value in self.utility_list.items():
            copied_utility_list[key] = value
        return copied_utility_list

    def create_utilities(self, tolerance, action_probabilities):
        adjusted_utilities = self.create_adjusted_utility_list(1)
        for state_key in range(1, self.num_states + 1):
            last_utility = self.utility_list[state_key]
            current_utility = adjusted_utilities[state_key]
            utility_difference = abs(current_utility - last_utility)
            while utility_difference > tolerance:
                last_utility = current_utility
                temp_result = self.value_iteration(state_key, action_probabilities, self.utility_list)
                self.utility_list[state_key] = temp_result[0]
                self.policy_list[state_key] = temp_result[1]
                current_utility = self.utility_list[state_key]
                utility_difference = abs(current_utility - last_utility)
        return self.utility_list.copy(), self.policy_list.copy()

    def value_iteration(self, state_key, action_probabilities, utility_list):
        self.state_controller.switch_state(state_key)
        if self.state_controller.current_state.is_terminal():
            return self.state_controller.current_state.get_reward(), 0

        action_dict = action_probabilities[state_key]
        up_sum = sum(utility_list[i] * action_dict[1][i] for i in action_dict[1])
        left_sum = sum(utility_list[i] * action_dict[2][i] for i in action_dict[2])
        down_sum = sum(utility_list[i] * action_dict[3][i] for i in action_dict[3])
        right_sum = sum(utility_list[i] * action_dict[4][i] for i in action_dict[4])

        random_value = random.randint(1, 10000)
        highest_utility = None
        policy = None

        if random_value > 1500:
            highest_utility = up_sum
            policy = 1
            if left_sum > highest_utility:
                highest_utility = left_sum
                policy = 2
            if down_sum > highest_utility:
                highest_utility = down_sum
                policy = 3
            if right_sum > highest_utility:
                highest_utility = right_sum
                policy = 4
        else:
            if random_value <= 375:
                highest_utility = up_sum
                policy = 1
            elif 375 < random_value <= 750:
                highest_utility = left_sum
                policy = 2
            elif 750 < random_value <= 1125:
                highest_utility = down_sum
                policy = 3
            else:
                highest_utility = right_sum
                policy = 4

        cost = self.gamma * highest_utility
        reward = self.state_controller.current_state.get_reward()
        return reward + cost, policy
