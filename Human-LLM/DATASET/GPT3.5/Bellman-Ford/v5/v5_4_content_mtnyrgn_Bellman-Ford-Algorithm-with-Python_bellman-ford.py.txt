class Graph:
    def __init__(self, vertices):
        """Initialize the graph with a given number of vertices."""
        self.num_vertices = vertices
        self.edges = []

    def add_edge(self, u, v, weight):
        """Add an edge to the graph."""
        self.edges.append((u, v, weight))

    def print_distances(self, distances):
        """Print the distances of vertices from the source."""
        print("Distances:")
        for i, distance in enumerate(distances):
            print(f"{i}\t\t{distance}")

    def bellman_ford(self, source):
        """
        Apply the Bellman-Ford algorithm to find shortest paths from a source vertex.

        Parameters:
        - source: The source vertex.
        """
        # Initialize distances with infinity
        distances = [float("inf")] * self.num_vertices
        distances[source] = 0

        # Relax edges repeatedly for V-1 iterations
        for _ in range(self.num_vertices - 1):
            for u, v, weight in self.edges:
                if distances[u] != float("inf") and distances[u] + weight < distances[v]:
                    distances[v] = distances[u] + weight

        # Check for negative cycles
        for u, v, weight in self.edges:
            if distances[u] != float("inf") and distances[u] + weight < distances[v]:
                print("Graph contains negative cycle!!")
                return

        # Print the distances of vertices from the source
        self.print_distances(distances)

# Get input from the user
num_vertices = int(input("Enter number of vertices: "))
num_edges = int(input("Enter number of edges: "))

# Create a graph object and add edges
graph = Graph(num_vertices)
for _ in range(num_edges):
    source_vertex = int(input("Source vertex: "))
    destination_vertex = int(input("Destination vertex: "))
    edge_weight = int(input("Weight of Edge: "))
    graph.add_edge(source_vertex, destination_vertex, edge_weight)

# Apply Bellman-Ford algorithm from source vertex 0
graph.bellman_ford(0)
