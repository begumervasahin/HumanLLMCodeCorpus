class Graph:
    def __init__(self, vertices):
        """
        Initializes a graph with a specified number of vertices.

        Parameters:
        vertices (int): The number of vertices in the graph.
        """
        self.vertices = vertices  # Total number of vertices in the graph
        self.graph = []  # Container for all edges in the graph (source, destination, weight)

    def add_edge(self, source, destination, weight):
        """
        Adds an edge to the graph.

        Parameters:
        source (int): The starting vertex of the edge.
        destination (int): The ending vertex of the edge.
        weight (int): The weight or cost of the edge.
        """
        # Edge is represented as a list containing the source, destination, and weight
        self.graph.append([source, destination, weight])

    def display_distances(self, distances):
        """
        Displays the distance from the source vertex to all other vertices in the graph.

        Parameters:
        distances (list): A list containing the distances of each vertex from the source.
        """
        print("Vertex\tDistance from Source")
        for vertex, distance in enumerate(distances):
            print(f"{vertex}\t\t{distance}")

    def bellman_ford(self, source):
        """
        Finds the shortest path from the source vertex to all other vertices using the Bellman-Ford algorithm.
        Also checks for the presence of negative-weight cycles.

        Parameters:
        source (int): The source vertex for which to find the shortest paths.
        """
        # Initialize distances from source to all vertices as infinity, and distance to source itself as 0
        distances = [float("Inf")] * self.vertices
        distances[source] = 0

        # Relax all edges 'vertices - 1' times to ensure shortest paths are found
        for _ in range(self.vertices - 1):
            for source, destination, weight in self.graph:
                if distances[source] != float("Inf") and distances[source] + weight < distances[destination]:
                    distances[destination] = distances[source] + weight

        # Check for negative-weight cycles in the graph
        for source, destination, weight in self.graph:
            if distances[source] != float("Inf") and distances[source] + weight < distances[destination]:
                print("Graph contains a negative weight cycle")
                return

        # Display the shortest distances from the source to all other vertices
        self.display_distances(distances)

# Demonstration of how to use the Graph class
if __name__ == "__main__":
    g = Graph(5)
    # Adding edges to the graph
    g.add_edge(0, 1, -1)
    g.add_edge(0, 2, 4)
    g.add_edge(1, 2, 3)
    g.add_edge(1, 3, 2)
    g.add_edge(1, 4, 2)
    g.add_edge(3, 2, 5)
    g.add_edge(3, 1, 1)
    g.add_edge(4, 3, -3)

    # Running Bellman-Ford algorithm from vertex 0
    g.bellman_ford(0)
