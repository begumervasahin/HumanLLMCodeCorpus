from Reinforcement_Maze import Reinforcement_Maze
from State_Controller import State_Controller
from State import State
from Action import Action
import sys
import time

class RM_Compare_Driver:
    @staticmethod
    def main(arguments):
        tol, num_trials = RM_Compare_Driver.parse_arguments(arguments)

        s_controller, gamma = RM_Compare_Driver.read_reinforcement_information(arguments[1])

        r_maze_list_1 = [Reinforcement_Maze(gamma, s_controller) for _ in range(16)]    
        r_maze_list_2 = [Reinforcement_Maze(gamma, s_controller) for _ in range(16)]    

        r_maze_control = Reinforcement_Maze(gamma, s_controller)
        averages = r_maze_control.countedAverages(num_trials)

        util_dict_1 = {}
        util_dict_2 = {}

        time_1_total = RM_Compare_Driver.calculate_utilities(r_maze_list_1, util_dict_1, tol, averages)
        time_2_total = RM_Compare_Driver.calculate_utilities(r_maze_list_2, util_dict_2, tol, averages, adjusted=True)

        return time_1_total, time_2_total

    @staticmethod
    def parse_arguments(arguments):
        tol = 0.00001
        num_trials = 10000

        if len(arguments) == 4:
            tol = float(arguments[2])
            num_trials = int(arguments[3])
        elif len(arguments) != 2:
            raise IndexError

        return tol, num_trials

    @staticmethod
    def read_reinforcement_information(file_name):
        with open(file_name, 'r') as my_file:
            num_actions = int(my_file.readline())
            num_states = int(my_file.readline())
            gamma = float(my_file.readline())

            s_controller = State_Controller(curr_state=None, state_list=None)
            file_list = []

            for _ in range(num_states):
                value = float(my_file.readline())
                terminality = int(my_file.readline())
                file_of_state = my_file.readline().rstrip()
                file_list.append(file_of_state)
                temp_state = State(value, terminality, len(file_list), action_list=None, count_actions=True)
                s_controller.addState(temp_state)

            for i, file_name in enumerate(file_list):
                s_controller.switchState(i + 1)
                if not s_controller.isTerminal():
                    with open(file_name, 'r') as state_file:
                        size = state_file.readline()
                        num_actions_state, num_states_state = map(int, size.split(' ')) 

                        if num_actions_state != num_actions or num_states_state != num_states:
                            raise MemoryError

                        for _ in range(num_actions):
                            line = state_file.readline()
                            myList = list(map(float, line.split(' ')))

                            if len(myList) != num_states:
                                raise MemoryError

                            t_list = []
                            for q in range(len(myList)):
                                s_controller.switchState(q + 1)
                                t_list.append(s_controller.getState())

                            s_controller.switchState(i + 1)
                            new_action = Action(len(util_dict_1) + len(util_dict_2) + 1, t_list, result_chances=myList, create_counter_table=True)
                            s_controller.addAction(new_action)

        return s_controller, gamma

    @staticmethod
    def calculate_utilities(r_maze_list, util_dict, tol, averages, adjusted=False):
        time_initial = time.time()
        for i, r_maze in enumerate(r_maze_list):
            index = i + 1 if not adjusted else i + 17
            util_dict[index] = r_maze.createListUtilitiesAdjusted(tol, averages, 200) if adjusted else r_maze.createListUtilities(tol, averages)
        time_final = time.time()

        return time_final - time_initial

if __name__ == '__main__':
    time_1_totals = 0
    time_2_totals = 0
    num_iterations = 30
    num_trials_total = 100

    for _ in range(num_iterations):
        time_totals = RM_Compare_Driver.main(sys.argv)
        time_1_totals += time_totals[0]
        time_2_totals += time_totals[1]

    print(time_1_totals / num_trials_total)
    print(time_2_totals / num_trials_total)
