import re
from socket import socket, AF_INET, SOCK_DGRAM

class BellmanFordAlgorithm:
    def __init__(self, router_count, initial_costs, my_name, port_mapping, address_to_name_mapping):
        self.router_count = router_count
        self.initial_costs = initial_costs
        self.my_name = my_name
        self.port_mapping = port_mapping
        self.address_to_name_mapping = address_to_name_mapping

        self.use_who = {}
        self.new_pm = None
        self.pm_address = None
        self.old_pm = {}
        self.permission = []

        self.table = []
        self.lie_table = []

        self.initialize_tables()
        self.display_routing_table()

        self.router_socket = socket(AF_INET, SOCK_DGRAM)
        self.router_socket.bind(('', self.port_mapping[my_name]))
        self.socket_ip = '127.0.0.1'
        self.router_socket.settimeout(0.5)

    def initialize_tables(self):
        self.table = [['99'] * self.router_count for _ in range(self.router_count)]
        self.lie_table = [self.initial_costs.copy() for _ in range(self.router_count)]

        for i, cost in enumerate(self.initial_costs):
            self.use_who[i + 1] = '' if cost == 'N' else str(i + 1)
            self.table[int(self.my_name) - 1][i] = cost

    def determine_who_to_send(self):
        self.permission = [cost != 'N' for cost in self.initial_costs]

    def send_update(self):
        message = '@'.join(['@'.join(row) for row in self.lie_table])
        for sending_router in range(self.router_count):
            if self.permission[sending_router] and sending_router + 1 != int(self.my_name):
                self.router_socket.sendto(bytes(message, 'UTF-8'), (self.socket_ip, self.port_mapping[str(sending_router + 1)]))

    def receive_update(self):
        try:
            message, client_address = self.router_socket.recvfrom(2048)
            self.new_pm = re.split("@", message.decode())
            self.new_pm.remove('')
            self.pm_address = client_address[1]

            if self.pm_address in self.old_pm and self.old_pm[self.pm_address] != self.new_pm:
                self.old_pm[self.pm_address] = self.new_pm
                self.table[self.address_to_name_mapping[self.pm_address] - 1] = self.new_pm
                print(f"Router {self.pm_address} sent a new update: {self.new_pm}")
                self.calculate_algorithm()
            elif self.pm_address not in self.old_pm:
                self.old_pm[self.pm_address] = self.new_pm
                self.table[self.address_to_name_mapping[self.pm_address] - 1] = self.new_pm
                print(f"Router {self.pm_address} sent a new update: {self.new_pm}")
                self.calculate_algorithm()
        except:
            pass

    def calculate_algorithm(self):
        for dest_router in range(self.router_count):
            if dest_router == int(self.my_name) - 1:
                continue

            temp = self.table[int(self.my_name) - 1][dest_router]
            temp = '99' if temp == 'N' else temp

            for conn_router in range(self.router_count):
                cost1 = 99 if self.initial_costs[conn_router] == 'N' else int(self.initial_costs[conn_router])
                cost2 = 99 if self.table[conn_router][dest_router] == 'N' else int(self.table[conn_router][dest_router])
                min_distance = cost1 + cost2

                if min_distance < int(temp):
                    self.table[int(self.my_name) - 1][dest_router] = str(min_distance)
                    self.lie_table[conn_router][dest_router] = str(min_distance)
                    min_through = self.lie_table[conn_router].index(str(min_distance))
                    self.use_who[dest_router + 1] = str(min_through + 1)
                    self.lie_table[min_through][dest_router] = '99'
                    self.send_update()

        self.display_routing_table()

    def display_routing_table(self):
        print(f"\nThe Cost table of router {self.my_name} is :")
        print("        ", end="")
        for i in range(1, self.router_count + 1):
            print(f"{i:<8}", end="")
        print("\n----------------------------------------------------")
        for row in range(self.router_count):
            print(f"{row + 1}|      ", end="")
            for col in range(self.router_count):
                if row == int(self.my_name) - 1:
                    print(f"{self.table[row][col]:<2}\{self.use_who[col + 1]:<5}", end="")
                else:
                    print(f"{self.table[row][col]:<8}", end="")
            print()
        print("----------------------------------------------------")

    def check_cost_change(self, new_costs):
        if self.initial_costs == new_costs:
            print("No Cost Change!")
        else:
            self.initial_costs = new_costs
            print(f"Link costs of router {self.my_name} have changed!")
            self.table[int(self.my_name) - 1] = new_costs
            self.lie_table = [new_costs.copy() for _ in range(self.router_count)]
            self.determine_who_to_send()
            self.calculate_algorithm()

# Example usage
if __name__ == '__main__':
    router_count = 5
    initial_costs = ['1', '2', 'N', '4', 'N']
    my_name = '1'
    port_mapping = {'1': 10001, '2': 10002, '3': 10003, '4': 10004, '5': 10005}
    address_to_name_mapping = {10001: 1, 10002: 2, 10003: 3, 10004: 4, 10005: 5}

    bfa = BellmanFordAlgorithm(router_count, initial_costs, my_name, port_mapping, address_to_name_mapping)
    bfa.determine_who_to_send()
    bfa.calculate_algorithm()
