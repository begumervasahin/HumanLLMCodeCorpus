import socket
import time
from threading import Thread
from collections import defaultdict

class Server(Thread):
    def __init__(self):
        super().__init__()

    def run(self):
        graph = self.load_graph_data('test')
        distances = self.initialize_distances(graph)

        server_socket = self.setup_server_socket()
        self.send_graph_data(server_socket, graph, distances)

    def load_graph_data(self, filename):
        graph = defaultdict(dict)
        with open(filename) as file:
            lines = file.readlines()
            for i in range(1, int(lines[0]) + 1):
                elements = lines[i].split()
                for j in range(1, len(elements), 2):
                    graph[i][int(elements[j])] = int(elements[j + 1])
        return graph

    def initialize_distances(self, graph):
        distances = defaultdict(dict)
        for node in graph:
            for other_node in graph:
                distances[node][other_node] = 0 if node == other_node else float('inf')
        return distances

    def setup_server_socket(self):
        server_socket = socket.socket()
        host = ""
        port = 60000
        server_socket.bind((host, port))
        server_socket.listen(5)
        return server_socket

    def send_graph_data(self, server_socket, graph, distances):
        conn, addr = server_socket.accept()
        for node in graph:
            data_to_send = self.construct_message(node, graph, distances)
            conn.send(data_to_send.encode())
        conn.close()

    def construct_message(self, node, graph, distances):
        message = f"{node} "
        for neighbor in graph[node]:
            message += f"{neighbor} {distances[node][neighbor]}\n"
        return message

class Receiver(Thread):
    def __init__(self):
        super().__init__()

    def run(self):
        client_socket = socket.socket()
        host = ""
        port = 60000
        client_socket.connect((host, port))

        while True:
            data = client_socket.recv(1024).decode()
            if data == 'Bye':
                break
        client_socket.close()

if __name__ == '__main__':
    receiver = Receiver()
    server = Server()
    server.start()
    time.sleep(3)
    receiver.start()
    server.join()
    receiver.join()
