import numpy as np

def update_routing(routes, num_routers):
    """
    Update routing information for all routers based on the current state.
    """
    temp = np.full((1, num_routers, num_routers, num_routers), 100)

    for i in range(num_routers):
        for j in range(num_routers):
            if i != j:
                if routes[-1, i, i, j] == 100:
                    temp[0, i, j, :] = 100
                else:
                    temp[0, i, j, :] = routes[-1, j, j, :]
            else:
                for t in range(num_routers):
                    if t == i:
                        temp[0, i, i, t] = 0
                    else:
                        temp[0, i, i, t] = 100
                        for u in range(num_routers):
                            current_route = routes[-1, i, i, u] + routes[-1, u, u, t]   
                            if current_route < temp[0, i, i, t]:
                                temp[0, i, i, t] = current_route

    if not np.array_equal(routes[-1, :, :, :], temp[0, :, :, :]):
        routes = np.concatenate((routes, temp))

    return routes

def get_input_matrix(num_routers):
    """
    Get user input for the routing matrix.
    """
    input_matrix = np.full((num_routers, num_routers), 100, dtype=int)
    for i in range(num_routers):
        row_input = input("Fill row number %d of the matrix: " % (i + 1))
        input_matrix[i] = np.fromstring(row_input, dtype=int, sep=' ')
    return input_matrix

def main():
    """
    Main function to run the router routing system.
    """
    print("Welcome to the router routing system!")

    num_routers = int(input("Please enter the number of routers: "))

    # Initialize the routing matrix and get user input
    routing_matrix = np.full((1, num_routers, num_routers, num_routers), 100, dtype=int)
    input_matrix = get_input_matrix(num_routers)

    # Set the diagonal of the routing matrix
    for j in range(num_routers):
        routing_matrix[0, j, j, :] = input_matrix[j, :]

    # Update the routing matrix until convergence
    while True:
        before_matrix_shape = routing_matrix.shape[0]
        routing_matrix = update_routing(routing_matrix, num_routers)
        after_matrix_shape = routing_matrix.shape[0]

        if after_matrix_shape == before_matrix_shape:
            break

    # Display routing information
    iterations = routing_matrix.shape[0]
    print("\nRouting finished successfully after %d iterations." % iterations)

    while True:
        print("\nThere are %d iterations and %d routers.\n" % (iterations, num_routers))
        print("-------------------------------------------------------------")
        iteration_pointer = input("Please enter the iteration level you want to take a look at: ")
        router_pointer = input("Enter router number or enter 'a' for an overall view: ")

        if router_pointer == 'a':
            print(routing_matrix[int(iteration_pointer) - 1])
        else:
            print(routing_matrix[int(iteration_pointer) - 1, int(router_pointer) - 1])  

        if input("Do you need more report? (Y/N)").lower() == 'n':
            break

if __name__ == "__main__":
    main()
