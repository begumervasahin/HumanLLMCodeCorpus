def initialize_graph():
    """Initialize the graph's vertices, edges, and weights."""
    vertices = {1: 0, 2: 1, 3: 2, 4: 3, 5: 4, 6: 5}
    edges = [(1, 2, 1), (1, 3, 2), (1, 4, 4), (3, 4, 3), (4, 5, 5), (4, 6, 2), (5, 6, 4)]
    return vertices, edges

def dijkstra(vertices, edges, start_vertex):
    """Compute shortest paths from the start_vertex using Dijkstra's algorithm."""
    inf = float("inf")
    distance = [inf for _ in vertices]
    previous = [None for _ in vertices]
    distance[vertices[start_vertex]] = 0

    for _ in vertices:
        for start, end, weight in edges:
            if distance[vertices[start]] + weight < distance[vertices[end]]:
                distance[vertices[end]] = distance[vertices[start]] + weight
                previous[vertices[end]] = start

    return distance, previous

def print_shortest_path(previous, vertices, destination):
    """Recursively print the shortest path to the destination."""
    def recursive_print(current_vertex):
        if previous[vertices[current_vertex]] is not None:
            recursive_print(previous[vertices[current_vertex]])
        print(current_vertex, end=' ')

    print("Shortest path to vertex", destination, ":", end=' ')
    recursive_print(destination)

def main():
    vertices, edges = initialize_graph()
    start_vertex = 1  # Assuming the path starts from vertex 1
    distance, previous = dijkstra(vertices, edges, start_vertex)
    print_shortest_path(previous, vertices, 6)  # Print the path to vertex 6

if __name__ == "__main__":
    main()
