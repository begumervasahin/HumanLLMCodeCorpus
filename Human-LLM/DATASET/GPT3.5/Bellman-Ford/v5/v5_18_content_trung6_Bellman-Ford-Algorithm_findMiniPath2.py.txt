import numpy as np

class Node:
    def __init__(self, vertex_name, dist=float('inf'), trace=None):
        self.vertex_name = vertex_name
        self.dist = dist
        self.trace = trace

class Graph:
    def __init__(self):
        self.vertices = []       # List of vertices
        self.edges = []          # List of edges
        self.shortest_path = []  # Shortest path
        self.num_vertices = 0    # Number of vertices
        self.num_edges = 0       # Number of edges

    def add_edge(self, src, dest, weight):
        self.edges.append([src, dest, weight])

    def bellman_ford(self, src, dest):
        # Initialize distances
        for vertex in self.vertices:
            vertex.dist = float("inf")
        self.vertices[src - 1].dist = 0

        # Relax edges num_vertices - 1 times
        for _ in range(self.num_vertices - 1):
            for u, v, w in self.edges:
                if self.vertices[v - 1].dist > self.vertices[u - 1].dist + w:
                    self.vertices[v - 1].dist = self.vertices[u - 1].dist + w
                    self.vertices[v - 1].trace = u

        # Check for negative cycle
        if self.vertices[dest - 1].dist == float('inf'):
            return

        # Reconstruct the shortest path
        self.shortest_path.append(dest)
        while src not in self.shortest_path:
            self.shortest_path.append(self.vertices[self.shortest_path[-1] - 1].trace)

    def read_input_from_file(self, file_path):
        with open(file_path, 'r') as file:
            first_line = file.readline().split()
            self.num_vertices = int(first_line[0])
            self.num_edges = int(first_line[1])
            start_vertex = int(first_line[2])
            end_vertex = int(first_line[3])

            for line in file:
                edge_info = line.split()
                self.add_edge(int(edge_info[0]), int(edge_info[1]), int(edge_info[2]))

        self.vertices = [Node(i) for i in range(1, self.num_vertices + 1)]

        return start_vertex, end_vertex

    def write_output_to_file(self, file_path, start_vertex, end_vertex):
        with open(file_path, 'w') as file:
            if start_vertex not in self.shortest_path:
                file.write("\nThere are no paths from %d to %d" % (self.vertices[start_vertex - 1].vertex_name,
                                                                    self.vertices[end_vertex - 1].vertex_name))
            else:
                file.write("Distance from %d to %d is: %d\n" % (self.vertices[start_vertex - 1].vertex_name,
                                                                 self.vertices[end_vertex - 1].vertex_name,
                                                                 self.vertices[end_vertex - 1].dist))     
                file.write('%d' % self.vertices[end_vertex - 1].vertex_name)
                for vertex in self.shortest_path:
                    if self.vertices[vertex - 1].trace is not None:
                        file.write(" <-- %d" % self.vertices[vertex - 1].trace)

# Main program
graph = Graph()
print("\nRead input from file")
start_vertex, end_vertex = graph.read_input_from_file('minpath_in.txt')
graph.bellman_ford(start_vertex, end_vertex)
print("\nWrite output into file")
graph.write_output_to_file('minpath_out.txt', start_vertex, end_vertex)
