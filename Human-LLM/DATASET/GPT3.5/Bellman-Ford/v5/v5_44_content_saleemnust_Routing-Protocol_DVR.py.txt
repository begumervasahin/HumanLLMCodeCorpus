import threading
import time
import sys
import json
from socket import socket, AF_INET, SOCK_DGRAM

# Global variables
server_ip = "127.0.0.1"
my_neighbors = {}
update_vector = {}
rcv_update = {}
flag = 0
router_name = ""

# Function to perform the Bellman-Ford algorithm
def bellman_ford(received_update):
    global update_vector
    for src_router, dst_routers in received_update.items():
        for dst_router, (cost, path) in dst_routers.items():
            if dst_router in update_vector:
                if update_vector[dst_router][0] > cost + update_vector[src_router][0]:
                    update_vector[dst_router][0] = cost + update_vector[src_router][0]
                    update_vector[dst_router][1] = update_vector[src_router][1] + path
            else:
                if dst_router == router_name:
                    update_vector[dst_router] = [0, router_name * 2]
                else:
                    update_vector[dst_router] = [cost + update_vector[src_router][0], update_vector[src_router][1] + path]

# Function to send updates to the server
def send_update_to_server(server_port):
    global server_ip, router_name, update_vector
    client_socket = socket(AF_INET, SOCK_DGRAM)
    while True:
        message = json.dumps(update_vector)
        data = [router_name, message]
        message = json.dumps(data)
        client_socket.sendto(message, (server_ip, server_port))
        modified_message, _ = client_socket.recvfrom(2048)
        if modified_message == "Received":
            client_socket.close()
            break

# Function to check the status of a neighbor
def check_neighbor_status(port):
    global server_ip
    status = -1
    flag_socket = socket(AF_INET, SOCK_DGRAM)
    try:
        flag_socket.sendto(b"hello", (server_ip, port))
        flag_socket.recvfrom(2048)
        status = 1
    except:
        status = 0
    finally:
        flag_socket.close()
    return status

# Function to manage client status
def client_status():
    global my_neighbors, flag
    while True:
        for key, val in my_neighbors.items():
            response = check_neighbor_status(int(val[0]))
            if response == 1:
                send_update_to_server(int(val[0]))
        time.sleep(3)

# Function to display router status
def router_status():
    global update_vector, router_name
    sorted_update_vector = sorted(update_vector.items(), key=lambda t: t[0])
    print(f"I am Router {router_name}")
    for dst_router, (cost, path) in sorted_update_vector:
        if dst_router != router_name:
            new_path = ''.join([path[i] for i in range(len(path) - 1) if path[i] != path[i + 1]]) + path[-1]
            print(f"Least cost path to router {dst_router}: {new_path} and the cost: {cost}")

# Function to handle server-side operations
def server_side(server_port):
    global server_ip, flag
    server_socket = socket(AF_INET, SOCK_DGRAM)
    server_socket.bind((server_ip, server_port))
    dist_vector = {}
    while True:
        message, client_address = server_socket.recvfrom(2048)
        if message == b"hello":
            modified_message = "Received"
            server_socket.sendto(modified_message.encode(), client_address)
        else:
            data = json.loads(message)
            modified_message = "Received"
            server_socket.sendto(modified_message.encode(), client_address)
            neighbor, dst_vector = data[0], json.loads(data[1])
            dist_vector.clear()
            for key, val in dst_vector.items():
                key = key.encode('ascii', 'ignore')
                val[1] = val[1].encode('ascii', 'ignore')
                dist_vector[key] = [val[0], val[1]]
            neighbor = neighbor.encode('ascii', 'ignore')
            if neighbor not in rcv_update.keys():
                rcv_update[neighbor] = dist_vector.copy()
                bellman_ford(rcv_update)
                flag = 0
            elif dist_vector != rcv_update[neighbor]:
                rcv_update[neighbor] = dist_vector.copy()
                bellman_ford(rcv_update)
                flag = 0
            else:
                flag += 1
                if flag == 5:
                    router_status()

# Main function to start the program
def main():
    global my_neighbors, router_name, update_vector
    router_name = sys.argv[1]
    my_port = int(sys.argv[2])
    config_file = sys.argv[3]

    with open(config_file, 'r') as f:
        lines = f.read().split()

    i = 1
    while i < len(lines):
        update_vector[lines[i]] = [float(lines[i + 1]), (router_name + lines[i])]
        my_neighbors[lines[i]] = [int(lines[i + 2])]
        i += 3

    thread = threading.Thread(target=client_status)
    thread.start()
    thread1 = threading.Thread(target=server_side, args=(my_port,))
    thread1.start()

if __name__ == "__main__":
    main()
