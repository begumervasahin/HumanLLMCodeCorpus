import sys
import re
import time
import itertools

# Regular expressions for parsing input
graph_re = re.compile(r"(\d+)\s(\d+)")
edge_re = re.compile(r"(\d+)\s(\d+)\s(-?\d+)")

vertices = []
edges = []

def BellmanFord(graph):
    path_pairs = []
    print("\nEntered BellmanFord")

    V, E = graph

    permutations = list(itertools.permutations(V, 2))
    inf = float('inf')

    for vertex in V:
        index = V.index(vertex)
        dist = [inf] * len(V)
        dist[index] = 0

        for _ in range(1, len(V)):
            for u, v in permutations:
                if dist[v] > dist[u] + float(E[u][v]):
                    dist[v] = dist[u] + float(E[u][v])

        neg_cycle = any(dist[v] > dist[u] + float(E[u][v]) for u, v in permutations)

        for i in range(len(V)):
            if dist[i] != float('inf'):
                dist[i] = int(dist[i])
            new_pair = (str(index), str(V[i])), str(dist[i])
            path_pairs.append(new_pair)

    print("\npathPairs")
    print(path_pairs)

    if neg_cycle:
        print("\nNote: A negative cycle exists")
    else:
        print("\nNote: No negative cycles exist")

    print("\nExiting BellmanFord\n")
    return path_pairs

def FloydWarshall(graph):
    path_pairs = []
    print("\nEntered FloydWarshall")

    V, E = graph
    inf = float('inf')
    dist = E

    for i in V:
        for j in V:
            dist[i][j] = float(dist[i][j])

    for v in V:
        dist[v][v] = 0

    for k in V:
        for i in V:
            for j in V:
                if dist[i][j] > dist[i][k] + dist[k][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]

    for i in V:
        for j in V:
            if dist[i][j] != float('inf'):
                dist[i][j] = int(dist[i][j])
            new_pair = (str(i), str(j)), str(dist[i][j])
            path_pairs.append(new_pair)

    print("\npathPairs")
    print(path_pairs)

    print("\nExiting FloydWarshall\n")
    return path_pairs

def read_graph(filename):
    global vertices
    global edges

    with open(filename, 'r') as infile:
        line1 = infile.readline()
        graph_match = graph_re.match(line1)

        if not graph_match:
            print(line1 + " not properly formatted")
            sys.exit(1)

        vertices = list(range(int(graph_match.group(1))))
        edges = [[float("inf")] * len(vertices) for _ in range(len(vertices))]

        for line in infile.readlines():
            line = line.strip()
            edge_match = edge_re.match(line)

            if edge_match:
                source, sink, weight = int(edge_match.group(1)), int(edge_match.group(2)), int(edge_match.group(3))

                if source >= len(vertices) or sink >= len(vertices):
                    print(f"Attempting to insert an edge between {source} and {sink} in a graph with {len(vertices)} vertices")
                    sys.exit(1)

                edges[source][sink] = weight

    print("\nVertices    Edges")
    for i in range(len(edges)):
        print(str(vertices[i]) + "         " + str(edges[i]))

    return vertices, edges

def main(filename, algorithm):
    algorithm = algorithm[1:]
    graph = read_graph(filename)
    print("\nAlgorithm:", algorithm)

    if algorithm in ('b', 'B'):
        start_time = time.time()
        BellmanFord(graph)
        end_time = time.time()
        bf_time = end_time - start_time
        print("\nBellman-Ford timing:", bf_time, "seconds\n")

    elif algorithm in ('f', 'F'):
        start_time = time.time()
        FloydWarshall(graph)
        end_time = time.time()
        fw_time = end_time - start_time
        print("Floyd-Warshall timing:", fw_time, "seconds")

    elif algorithm == "both":
        start_time = time.time()
        BellmanFord(graph)
        end_time = time.time()
        bf_time = end_time - start_time

        start_time = time.time()
        FloydWarshall(graph)
        end_time = time.time()
        fw_time = end_time - start_time

        print("\nBellman-Ford timing:", bf_time, "seconds")
        print("Floyd-Warshall timing:", fw_time, "seconds\n")

if __name__ == '__main__':
    if len(sys.argv) < 3:
        print("python bellman_ford.py -<f|b> <input_file>")
        sys.exit(1)

    main(sys.argv[2], sys.argv[1])
