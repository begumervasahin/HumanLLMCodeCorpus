from socket import *
import time
import pickle
import math
import sys

class VectorRouting:
    def __init__(self, port, num_routers, num_neighbors, routing_dict):
        self.port = port
        self.num_routers = num_routers
        self.num_neighbors = num_neighbors
        self.routing_dict = routing_dict

    def calculate_vector(self):
        router_socket = socket(AF_INET, SOCK_DGRAM)
        router_socket.bind(('', self.port))

        while True:
            print(self.routing_dict)
            time.sleep(1)

            encoded_msg = pickle.dumps(self.routing_dict, protocol=-1)

            for key, _ in self.routing_dict.items():
                try:
                    router_socket.sendto(encoded_msg, ('', key))
                except Exception as e:
                    print(f"Error sending message to router {key}: {e}")

            try:
                msg, address = router_socket.recvfrom(1024)
                decoded_msg = pickle.loads(msg)
                neighbor_addr = address[1]

                for key in decoded_msg:
                    if key not in self.routing_dict:
                        self.routing_dict[key] = math.inf

                self.routing_dict = self.bellman_ford(self.routing_dict, decoded_msg, neighbor_addr)
            except Exception as e:
                print(f"Error receiving message: {e}")
                continue

    def bellman_ford(self, dict1, dict2, addr):
        for key in dict2:
            if dict1[key] > dict1[addr] + dict2[key]:
                dict1[key] = dict1[addr] + dict2[key]
        return dict1

def main():
    if len(sys.argv) < 4:
        print("Usage: python script.py PORT NUM_ROUTERS NUM_NBRS [router_nbr cost] ...")      
        return

    port = int(sys.argv[1])
    num_routers = int(sys.argv[2])
    num_neighbors = int(sys.argv[3])
    routing_dict = {port: 0}

    for i in range(0, num_neighbors * 2, 2):
        neighbor = int(sys.argv[4 + i])
        cost = int(sys.argv[4 + i + 1])
        routing_dict[neighbor] = cost

    router = VectorRouting(port, num_routers, num_neighbors, routing_dict)
    router.calculate_vector()

if __name__ == '__main__':
    main()
