class FibonacciHeap:
    class Node:
        def __init__(self, key, value):
            self.key = key
            self.value = value
            self.parent = None
            self.child = None
            self.left = None
            self.right = None
            self.degree = 0
            self.mark = False

    def __init__(self):
        self.root_list = None  # Circular doubly linked list of roots
        self.min_node = None  # Pointer to the node with the minimum key
        self.total_nodes = 0  # Total number of nodes in the heap

    def find_min(self):
        return self.min_node

    def extract_min(self):
        min_node = self.min_node
        if min_node is not None:
            if min_node.child is not None:
                for child in self._iterate(min_node.child):
                    self._merge_with_root_list(child)
                    child.parent = None
            self._remove_from_root_list(min_node)
            if min_node == min_node.right:
                self.min_node = self.root_list = None
            else:
                self.min_node = min_node.right
                self._consolidate()
            self.total_nodes -= 1
        return min_node

    def insert(self, key, value=None):
        new_node = self.Node(key, value)
        new_node.left = new_node.right = new_node
        self._merge_with_root_list(new_node)
        if self.min_node is None or new_node.key <= self.min_node.key:
            self.min_node = new_node
        self.total_nodes += 1
        return new_node

    def decrease_key(self, node, new_key):
        if new_key > node.key:
            return None
        node.key = new_key
        parent = node.parent
        if parent is not None and node.key <= parent.key:
            self._cut(node, parent)
            self._cascading_cut(parent)
        if node.key <= self.min_node.key:
            self.min_node = node

    def merge(self, other_heap):
        merged_heap = FibonacciHeap()
        merged_heap.root_list = self.root_list
        merged_heap.min_node = self.min_node
        last = other_heap.root_list.left
        other_heap.root_list.left = merged_heap.root_list.left
        merged_heap.root_list.left.right = other_heap.root_list
        merged_heap.root_list.left = last
        merged_heap.root_list.left.right = merged_heap.root_list
        if other_heap.min_node.key <= merged_heap.min_node.key:
            merged_heap.min_node = other_heap.min_node
        merged_heap.total_nodes = self.total_nodes + other_heap.total_nodes
        return merged_heap

    def _cut(self, child, parent):
        self._remove_from_child_list(parent, child)
        parent.degree -= 1
        self._merge_with_root_list(child)
        child.parent = None
        child.mark = False

    def _cascading_cut(self, node):
        parent = node.parent
        if parent is not None:
            if node.mark is False:
                node.mark = True
            else:
                self._cut(node, parent)
                self._cascading_cut(parent)

    def _consolidate(self):
        degree_table = [None] * self.total_nodes
        nodes = [w for w in self._iterate(self.root_list)]
        for node in nodes:
            x = node
            d = x.degree
            while degree_table[d] is not None:
                y = degree_table[d]
                if x.key > y.key:
                    x, y = y, x
                self._heap_link(y, x)
                degree_table[d] = None
                d += 1
            degree_table[d] = x
        for i in range(len(degree_table)):
            if degree_table[i] is not None:
                if degree_table[i].key <= self.min_node.key:
                    self.min_node = degree_table[i]

    def _heap_link(self, child, parent):
        self._remove_from_root_list(child)
        child.left = child.right = child
        self._merge_with_child_list(parent, child)
        parent.degree += 1
        child.parent = parent
        child.mark = False

    def _merge_with_root_list(self, node):
        if self.root_list is None:
            self.root_list = node
        else:
            node.right = self.root_list.right
            node.left = self.root_list
            self.root_list.right.left = node
            self.root_list.right = node

    def _merge_with_child_list(self, parent, node):
        if parent.child is None:
            parent.child = node
        else:
            node.right = parent.child.right
            node.left = parent.child
            parent.child.right.left = node
            parent.child.right = node

    def _remove_from_root_list(self, node):
        if node == self.root_list:
            self.root_list = node.right
        node.left.right = node.right
        node.right.left = node.left

    def _remove_from_child_list(self, parent, node):
        if parent.child == parent.child.right:
            parent.child = None
        elif parent.child == node:
            parent.child = node.right
            node.right.parent = parent
        node.left.right = node.right
        node.right.left = node.left

    def _iterate(self, head):
        node = stop = head
        flag = False
        while True:
            if node == stop and flag is True:
                break
            elif node == stop:
                flag = True
            yield node
            node = node.right
