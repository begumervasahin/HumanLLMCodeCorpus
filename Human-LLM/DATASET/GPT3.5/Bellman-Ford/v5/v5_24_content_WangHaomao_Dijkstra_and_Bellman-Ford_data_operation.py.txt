from pyspark import SparkContext
from pprint import pprint

def arrange_edges(edge):
    """
    Process edges from input data.
    """
    u, v, w = edge.split(' ')
    if float(w) < 0:
        return (u, [f"{v}:{w}"]), (v, []), (-1, [])
    return (u, [f"{v}:{w}"]), (v, [])

def gr_arrange_edges(edge):
    """
    Process edges from input data in .gr format.
    """
    split_list = edge.split(' ')
    if split_list[0] != 'a':
        return []
    u, v, w = split_list[1:4]    
    if int(w) < 0:
        return (u, [f"{v}:{w}"]), (v, []), (-1, [])
    return (u, [f"{v}:{w}"]), (v, [])

def get_negative_flag(it):
    """
    Check if there are negative flags in the data.
    """
    for i in it:
        if i[0] == -1:
            return i
    return (None, None)

def merge_nodes(node1, node2):
    """
    Merge nodes with the same key.
    """
    return node2 if node1 is None else node1

def get_graph(file_path, sc):
    """
    Process input data to generate the graph.
    """
    data = sc.textFile(file_path, 8)
    nodes_mapper = data.flatMap(gr_arrange_edges) if '.gr' in file_path else data.flatMap(arrange_edges)
    flag_exist_negative = any(-1 in x for x in nodes_mapper.mapPartitions(get_negative_flag).collect())
    graph = nodes_mapper.reduceByKey(lambda x, y: x + y).filter(lambda x: x[0] != -1).sortByKey()
    return graph, flag_exist_negative

if __name__ == '__main__':
    sc = SparkContext.getOrCreate()
    graph, negative_flag = get_graph('data/input_slides.dat', sc)
    pprint(graph.collect())
