import codecs

def create_crc32_lookup_table():
    """
    Creates a lookup table for CRC-32 calculations using the polynomial 0xEDB88320.
    
    Returns:
        A list containing the CRC-32 result for all possible byte values (0-255).
    """
    table = []
    for byte_value in range(256):
        crc = byte_value
        for _ in range(8):
            crc = (crc >> 1) ^ 0xEDB88320 if (crc & 1) else crc >> 1
        table.append(crc)
    return table

def compute_crc32(data, table):
    """
    Computes the CRC-32 checksum for the given data.
    
    Args:
        data (bytes): The data over which to compute the CRC-32.
        table (list): The CRC-32 lookup table.
    
    Returns:
        The computed CRC-32 checksum.
    """
    crc = 0xFFFFFFFF
    for byte in data:
        index = (crc ^ byte) & 0xFF
        crc = table[index] ^ (crc >> 8)
    return ~crc & 0xFFFFFFFF

# Prepare the CRC-32 lookup table.
crc32_table = create_crc32_lookup_table()

# Convert the hexadecimal message to bytes and calculate its CRC-32 checksum.
hex_message = 'FAFAFA'
message_bytes = codecs.decode(hex_message, 'hex')
checksum = compute_crc32(message_bytes, crc32_table)

# Format and display the original message, checksum, and combined message for transmission.
formatted_message = f"{bin(int(hex_message, 16))[2:]} ({hex_message})"
formatted_checksum = f"{bin(checksum)[2:].zfill(32)} ({hex(checksum)})"
combined_message = f"{formatted_message}{formatted_checksum}"

print(f"Original Message: {formatted_message}")
print(f"Checksum: {formatted_checksum}")
print("Message to be sent (binary representation):")
print(combined_message)
