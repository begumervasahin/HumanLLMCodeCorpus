# Constants
TIMEOUT = 0

# Global variables
forwarding_table = {}

# Signal handler for graceful exit
def signal_handler(signum, frame):
    print('Closing Bellman-Ford Host...\n')
    shutdown()

# Gracefully shutdown the program
def shutdown():
    try:
        host_sock.close()
        send_sock.close()
    except Exception as e:
        print(f"Error while shutting down: {e}")
    finally:
        print('--------------------Bellman-Ford Host closed--------------------\n')
        sys.exit(1)

# Exit the program with an error message
def die(msg):
    print('ERROR:', msg)
    sys.exit()

# Create a UDP socket
def create_udp_socket():
    return socket.socket(socket.AF_INET, socket.SOCK_DGRAM)

# Convert address to string
def addr_to_str(addr):
    return f"{addr[0]}:{addr[1]}"

# Convert address from string
def addr_from_str(addr_str):
    parts = addr_str.split(':')
    return parts[0], int(parts[1])

# Check command line arguments
def check_commandline(args, msg):
    if args != len(sys.argv):
        die(msg)

# Initialize the host
def initialize_host():
    try:
        with open(sys.argv[1], 'r') as f:
            host_port, global TIMEOUT = map(float, f.readline().rstrip().split())
            for line in f:
                line = line.strip()
                if line:
                    neighbor_addr, cost = line.split()
                    cost = float(cost)
                    neighbors[neighbor_addr] = [cost, dt.datetime.now(), True]
                    forwarding_table[neighbor_addr] = [cost, neighbor_addr]
    except Exception as e:
        die(f"Failed to initialize host from file {sys.argv[1]}: {e}")

    global last_update
    last_update = dt.datetime.now()

    global host_addr
    host_addr = (socket.gethostbyname(socket.gethostname()), int(host_port))

    global host_sock
    host_sock = create_udp_socket()
    try:
        host_sock.bind(host_addr)
    except Exception as e:
        die(f"Failed to bind host UDP socket: {e}")

    global send_sock
    send_sock = create_udp_socket()

# Send a packet to a destination
def send_packet(packet, dest_str):
    if dest_str not in neighbors or not neighbors[dest_str][2] or forwarding_table[dest_str][0] == INFINITY:
        return False
    send_sock.sendto(packet, addr_from_str(dest_str))
    return True

# Update neighbors and check for their inactivity
def update_neighbors():
    limit = dt.timedelta(seconds=3 * TIMEOUT)
    timeout = dt.timedelta(seconds=TIMEOUT)
    while True:
        now = dt.datetime.now()
        if now - last_update > timeout:
            broadcast_vtable()
        for addr, n in neighbors.items():
            if n[2] and now - n[1] > limit:
                deactivate_link(addr)

# Broadcast the distance vector to active neighbors
def broadcast_vtable():
    vectors = []
    host_addr_str = addr_to_str(host_addr)
    addresses = neighbors.keys()
    dests = forwarding_table.keys()
    for address in addresses:
        if neighbors[address][2]:
            for dest in dests:
                vector = forwarding_table[dest]
                if vector[1] == address and dest != address:
                    v = (dest, 'inf', vector[1])
                else:
                    v = (dest, str(vector[0]), vector[1])
                vectors.append(','.join(v))
            v_table_packet = f"{H_ROUTE_UPDATE} {host_addr_str} {' '.join(vectors)}"
            send_packet(v_table_packet, address)
    global last_update
    last_update = dt.datetime.now()

# Process incoming packets from the inbox queue
def process_packets():
    while True:
        packet = inbox.get()
        p_contents = packet.split(' ', 2)
        p_type = p_contents[0]
        if p_type == H_ROUTE_UPDATE:
            update_table(p_contents[1], p_contents[2])
        elif p_type == H_LINK_UP:
            activate_link(p_contents[1])
        elif p_type == H_LINK_DOWN:
            deactivate_link(p_contents[1])
        elif p_type == H_CHGCOST:
            update_cost(p_contents[1], p_contents[2])
        elif p_type == H_FILE:
            handle_file(packet)
        else:
            print('WARNING: unrecognized packet received.\n')

# Update forwarding table with neighbors' information
def update_table_neighbors():
    changed = False
    for a, vector in forwarding_table.items():
        n = neighbors[a]
        if n[0] < vector[0] and n[2]:
            vector[0] = n[0]
            vector[1] = a
            changed = True
    return changed

# Update the forwarding table from a ROUTE UPDATE message
def update_table(source_str, vector_msg):
    if source_str in neighbors:
        neighbors[source_str][2] = True
        neighbors[source_str][1] = dt.datetime.now()
    else:
        neighbors[source_str] = [INFINITY, dt.datetime.now(), True]

    host_addr_str = addr_to_str(host_addr)
    vectors = vector_msg.split(' ')
    for vector in vectors:
        dest, cost, link = vector.split(',')
        if dest == host_addr_str or link == host_addr_str:
            continue
        cost = float(cost)
        if source_str in forwarding_table:
            entry = forwarding_table[dest]
            if link == host_addr_str and entry[1] == source_str:
                entry[0] = cost
                neighbors[source_str][0] = cost
        else:
            forwarding_table[dest] = [cost, link]
            neighbors[source_str][0] = cost

    update_table_neighbors()
