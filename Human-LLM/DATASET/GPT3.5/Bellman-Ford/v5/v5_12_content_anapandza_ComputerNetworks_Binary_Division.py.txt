def bitwise_xor(a, b):
    """
    Compute the bitwise XOR of two binary strings.

    Parameters:
    - a (str): Binary string.
    - b (str): Binary string of equal length to `a`.

    Returns:
    - str: Result of bitwise XOR operation.
    """
    # Use list comprehension for concise and efficient XOR operation.
    return ''.join('0' if bit_a == bit_b else '1' for bit_a, bit_b in zip(a, b))

def binary_mod2_division(dividend, divisor):
    """
    Perform a modulo-2 division of `dividend` by `divisor`, commonly used in error-checking algorithms.

    Parameters:
    - dividend (str): The binary number to be divided.
    - divisor (str): The binary divisor.

    Displays:
    - Quotient and remainder of the modulo-2 division.
    """
    # Initialize variables for operation.
    pick_len = len(divisor)
    tmp = dividend[:pick_len]
    quotient = []

    # Iteratively divide the dividend by the divisor, modulo 2.
    for bit in dividend[pick_len - 1:]:
        # Determine if the leftmost bit allows division.
        if tmp[0] == '1':
            tmp = bitwise_xor(divisor, tmp) + bit
        else:
            tmp = bitwise_xor('0'*pick_len, tmp) + bit
        # Append the appropriate quotient bit.
        quotient.append('1' if tmp[0] == '1' else '0')

        # Ensure `tmp` is always the correct length for the next iteration.
        tmp = tmp[-pick_len:]

    # Correct the quotient for leading zeros removed by the division process.
    corrected_quotient = ''.join(quotient).lstrip('0')

    # The last `tmp` value is the remainder after division.
    print(f"Quotient: {corrected_quotient or '0'}")
    print(f"Remainder: {tmp}")
