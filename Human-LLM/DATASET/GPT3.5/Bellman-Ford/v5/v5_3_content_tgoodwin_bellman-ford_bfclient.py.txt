import sys
import socket
import select
import json
import threading
import time
import copy
import signal

# Constants
RECV_BUFFER = 4096
INFINITY = float('inf')

# Global variables
self_id = ""
neighbors = {}
routing_table = {}
adjacent_links = {}
old_links = {}
active_hist = {}
dead_links = []

# Create a UDP socket
recv_sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
recv_sock.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)

def main():
    global self_id

    # Check command-line arguments
    if len(sys.argv) < 3:
        print("Usage: bfclient.py <localport> <timeout> <[id_addr1 port1 weight1...]>")
        sys.exit()

    local_port = int(sys.argv[1])
    recv_sock.bind(('', local_port))
    signal.signal(signal.SIGINT, close_handler)

    host = socket.gethostbyname(socket.gethostname())
    self_id = f"{host}:{sys.argv[1]}"
    timeout = int(sys.argv[2])

    # Parse neighbor information from command-line arguments
    parse_neighbors(sys.argv[3:])

    print(f"bfclient running at address [{host}] on port [{sys.argv[1]}]")

    # Start timers
    update_timer(timeout)
    node_timer(timeout)

    # Start main loop
    prompt()
    while True:
        handle_inputs()

    recv_sock.close()

def prompt():
    """Display the command prompt."""
    sys.stdout.write('> ')
    sys.stdout.flush()

def update_neighbor():
    """Update neighbors with routing information."""
    for neighbor in copy.deepcopy(neighbors):
        addr = parse_address(neighbor)
        send_routing_info(addr)

def update_timer(timeout_interval):
    """Set timer to update neighbors periodically."""
    update_neighbor()
    t = threading.Timer(timeout_interval, update_timer, [timeout_interval])
    t.setDaemon(True)
    t.start()

def node_timer(timeout_interval):
    """Set timer to monitor node activity and close inactive connections."""
    for neighbor in copy.deepcopy(neighbors):
        if neighbor in active_hist:
            check_node_activity(neighbor, timeout_interval)

        start_time = int(time.time())
        t = threading.Timer(3, node_timer, [timeout_interval])
        t.setDaemon(True)
        t.start()

def handle_inputs():
    """Handle user inputs and received data."""
    socket_list = [sys.stdin, recv_sock]

    try:
        read_sockets, _, _ = select.select(socket_list, [], [])
    except (select.error, socket.error) as e:
        sys.exit()

    for sock in read_sockets:
        if sock == recv_sock:
            handle_received_data()
        else:
            handle_user_input()

def handle_received_data():
    """Handle received data."""
    data, addr = recv_sock.recvfrom(RECV_BUFFER)
    if data:
        msg = json.loads(data)
        msg_handler(msg, addr)
    else:
        print("[Error] 0 bytes received.")

def handle_user_input():
    """Handle user input."""
    data = sys.stdin.readline().rstrip()
    if len(data) > 0:
        data_list = data.split()
        cmd_handler(data_list)
        prompt()
    else:
        sys.stdout.flush()
        prompt()

def cmd_handler(args):
    """Handle user commands."""
    command = args[0]

    if command == "LINKDOWN":
        handle_link_down_command(args)
    elif command == "LINKUP":
        handle_link_up_command(args)
    elif command == "SHOWRT":
        show_routing_table()
    elif command == "TWEET":
        tweet_content = ' '.join(args[1:])
        tweet(self_id, tweet_content)
    elif command == "CLOSE":
        close()

        handle_update_message(received_data, parse_address(tuple_addr))

def close_handler(signum, frame):
    """Handle SIGINT signal."""
    sys.exit(f"Signal {str(signum)} called, closing down.")

def handle_link_down_command(args):
    """Handle LINKDOWN command."""
    if len(args) == 3:
        link_down(args[1], args[2])
    else:
        print("[ERROR] Incorrect number of arguments for 'LINKDOWN' command.")

def handle_link_up_command(args):
    """Handle LINKUP command."""
    if len(args) == 3:
        link_up(args[1], args[2])
    else:
        print("[ERROR] Incorrect number of arguments for 'LINKUP' command.")

def link_down(ip_addr, port):
    """Handle LINKDOWN command."""
    node_address = f"{ip_addr}:{port}"

    if node_address not in neighbors:
        print(f"[ERROR] {node_address} is not a neighbor.")
    else:
        handle_link_failure(node_address)

def link_up(ip_addr, port):
    """Handle LINKUP command."""
    node_address = f"{ip_addr}:{port}"

    if node_address not in old_links:
        print("[Error] This link does not exist.")
    else:
        handle_link_recovery(node_address)

def tweet(sender_id, message):
    """Handle TWEET command."""
    timestamp = time.strftime('%H:%M:%S', time.localtime(time.time()))
    tweet_msg = f"[{timestamp}] @{sender_id}: {message}"
    send_tweet_message(tweet_msg)

def show_routing_table():
    """Display routing table."""
    timestamp = time.strftime('%H:%M:%S', time.localtime(time.time()))
    print(f"[{timestamp}] Distance vector list for [{self_id}] is:")

    for node, info in routing_table.items():
        link = info['link']
        print(f"Destination = ({node}), Cost = {info['cost']}, Link = ({link})")

def parse_address(addr):
    """Parse address tuple (ip, port) into string format."""
    return f"{addr[0]}:{addr[1]}"
