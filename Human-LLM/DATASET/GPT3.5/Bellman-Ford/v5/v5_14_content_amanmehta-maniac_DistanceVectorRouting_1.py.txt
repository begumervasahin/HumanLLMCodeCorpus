from collections import defaultdict
import socket
import hashlib
import time
from threading import Thread

class Server(Thread):
    def __init__(self):
        super().__init__()

    def run(self):
        port = 60000
        server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        host = ""
        server_socket.bind((host, port))
        server_socket.listen(5)

        for i in range(1, N):
            for u in range(1, N + 1):
                to_send = ""
                for v in range(1, N + 1):
                    to_send += f"{v} {d[u][v]}\n"

                hashed_data = hashlib.md5(to_send.encode()).hexdigest()

                for v in graph[u]:
                    conn, addr = server_socket.accept()
                    conn.send(to_send.encode())
                    conn, addr = server_socket.accept()
                    conn.send(str(v).encode())
                    conn, addr = server_socket.accept()
                    conn.send(str(u).encode())
                    conn, addr = server_socket.accept()
                    conn.send(hashed_data.encode())

        conn, addr = server_socket.accept()
        conn.send("Bye".encode())
        conn.close()

class Receiver(Thread):
    def __init__(self):
        super().__init__()

    def run(self):
        distance = defaultdict()
        ii = 0

        while True:
            ii += 1
            client_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            host = ""
            port = 60000
            client_socket.connect((host, port))

            data = client_socket.recv(1024).decode()

            if ii % 4 == 1:
                dist = data
            elif ii % 4 == 2:
                node = int(data)
            elif ii % 4 == 3:
                sender = int(data)
            else:
                hash_val = data
                computed_hash = hashlib.md5(dist.encode()).hexdigest()

                if computed_hash == hash_val:
                    dist = dist.split('\n')
                    for i in range(len(dist) - 1):
                        dist[i] = dist[i].split()
                        dist[i][0], dist[i][1] = int(dist[i][0]), int(dist[i][1])       
                        distance[dist[i][0]] = dist[i][1]

                    for j in graph:
                        if d[node][j] > distance[j] + graph[sender][node]:
                            d[node][j] = distance[j] + graph[sender][node]
                else:
                    print('Error occurred! Try again.')

            if data == 'Bye':
                break

            client_socket.close()
