class Graph:
    """A class representing a graph."""

    def __init__(self, vertices=[], edges=[], directed=False):
        """Initialize the graph with vertices, edges, and whether it's directed."""
        self.vertices = vertices
        self.edges = edges
        self.directed = directed
        self.adjacency_list = {}

    def get_adjacent_vertices(self, vertex):
        """Return the adjacent vertices of a given vertex."""
        return self.adjacency_list[vertex.value]

    def add_vertex(self, value):
        """Add a vertex to the graph."""
        new_vertex = Vertex(value=value)
        self.vertices.append(new_vertex)
        self.adjacency_list[new_vertex.value] = []

    def add_edge(self, u, v, weight):
        """Add an edge between two vertices."""
        new_edge = Edge(u, v, weight)
        self.adjacency_list[u.value].append(new_edge)
        self.edges.append(new_edge)
        if not self.directed:
            reverse_edge = Edge(v, u, weight)
            if v.value not in self.adjacency_list:
                self.adjacency_list[v.value] = []
            self.adjacency_list[v.value].append(reverse_edge)


class Vertex:
    """A class representing a vertex in a graph."""

    def __init__(self, value=None, distance=None, predecessor=None):
        """Initialize the vertex with a value, distance, and predecessor."""
        self.value = value
        self.distance = distance
        self.predecessor = predecessor


class Edge:
    """A class representing an edge in a graph."""

    def __init__(self, u=None, v=None, weight=None):
        """Initialize the edge with two vertices and a weight."""
        self.u = u
        self.v = v
        self.weight = weight

    def equals(self, other):
        """Check if two edges are equal."""
        if self.u == other.u and self.v == other.v:
            return True
        return False
