from graph import *
import pickle
import variants
import argparse
import time
import json

INPUT_PATH = "graphs\\"
OUTPUT_PATH = "data\\"

def read_graph_list(file_name):
    with open(INPUT_PATH + file_name, 'rb') as file:
        graph_data = pickle.load(file)
    return graph_data

def write_data(data, file_name):
    with open(OUTPUT_PATH + file_name, "w") as output:
        json.dump(data, output, indent=4)

def parse_arguments():
    parser = argparse.ArgumentParser()
    parser.add_argument('input_file', type=str, help='Input file name') 
    parser.add_argument('output_file', type=str, help='Output file name')
    return parser.parse_args()

def get_average_time(func, graph_data):
    total_time = 0
    for _ in range(10):
        start_time = round(time.clock() * 10000)
        func(graph_data, graph_data.V[0])
        end_time = round(time.clock() * 10000)
        total_time += (end_time - start_time)
    return total_time / 10

def main():
    data = {"vertices": 0, "dijkstra": [], "bellmanford": [], "fibonacciheap": [], "yen": [], "edges": []}
    args = parse_arguments()
    print("Beginning to calculate run times...")
    
    with open(INPUT_PATH + args.input_file, "rb") as input_file:
        index = 1
        while True:
            try:
                print("Loading graph data...")
                graph_data = pickle.load(input_file)
                print("Loading Complete.")
                
                data["vertices"] = len(graph_data.V)
                data["edges"].append(len(graph_data.E))
                print(f"Graph {index} out of {len(data['vertices']) - 1} info:")        
                print("\tDirected:", graph_data.directed)
                print("\tVertex Count:", len(graph_data.V))
                print("\tEdge Count:", len(graph_data.E))
                
                print("Calculating average time for shortest path algorithms...")
                avg_time_dijkstra = get_average_time(variants.dijkstra, graph_data)
                data["dijkstra"].append(avg_time_dijkstra)
                print("Dijkstra Run Time:", avg_time_dijkstra)
                
                avg_time_fib = get_average_time(variants.fibDijkstra, graph_data)
                data["fibonacciheap"].append(avg_time_fib)
                print("Fibonacci Heap Run Time:", avg_time_fib)
                
                avg_time_bellman_ford = get_average_time(variants.bellmanFord, graph_data)       
                data["bellmanford"].append(avg_time_bellman_ford)
                print("Bellman-Ford Run Time:", avg_time_bellman_ford)
                
                avg_time_yen = get_average_time(variants.Yen, graph_data)
                data["yen"].append(avg_time_yen)
                print("Yen Run Time:", avg_time_yen)
                
                graph_data = None
                index += 1
            except EOFError:
                break

    print("Finished calculations.")
    print("Writing data to", args.output_file, "...")
    write_data(data, args.output_file)
    print("Data written.")

if __name__ == "__main__":
    main()
