from collections import defaultdict
import socket
import time
from threading import Thread

class Server(Thread):
    def __init__(self):
        super().__init__()

    def run(self):
        port = 60000
        server_socket = socket.socket()
        host = ""
        server_socket.bind((host, port))
        server_socket.listen(5)

        conn, addr = server_socket.accept()
        for i in range(1, N):
            for u in range(1, N + 1):
                forward_table_u = self.construct_forward_table(u)
                self.send_data(conn, forward_table_u)

        self.send_data(conn, "Bye")
        conn.close()

    def construct_forward_table(self, u):
        forward_table = ""
        for v in range(1, N + 1):
            forward_table += f"{v} {d[u][v]}\n"
        return forward_table

    def send_data(self, conn, data):
        conn.send(data.encode())
        time.sleep(0.01)

class Receiver(Thread):
    def __init__(self):
        super().__init__()

    def run(self):
        distance = defaultdict(int)
        ii = 0
        while True:
            ii += 1
            s = socket.socket()
            host = ""
            port = 60000
            s.connect((host, port))
            data = s.recv(1024)
            data_type = self.identify_data_type(data)

            if data_type == 'Distance':
                distance = self.process_distance_data(data)
            elif data_type == 'Node':
                node = self.extract_node(data)
            else:
                sender = self.extract_sender(data)
                self.update_distance(node, sender, distance)

            if data == b'Bye':
                break

        s.close()

    def identify_data_type(self, data):
        data_str = data.decode()
        if '\n' not in data_str:
            return 'Node'
        elif data_str.split('\n')[0].isdigit():
            return 'Distance'
        else:
            return 'Sender'

    def process_distance_data(self, data):
        distances = defaultdict(int)
        distance_str = data.decode().split('\n')
        for item in distance_str[:-1]:
            node, dist = map(int, item.split())
            distances[node] = dist
        return distances

    def extract_node(self, data):
        return int(data.decode())

    def extract_sender(self, data):
        return int(data.decode())

    def update_distance(self, node, sender, distance):
        for j in graph:
            if d[node][j] > distance[j] + graph[sender][node]:
                d[node][j] = distance[j] + graph[sender][node]
