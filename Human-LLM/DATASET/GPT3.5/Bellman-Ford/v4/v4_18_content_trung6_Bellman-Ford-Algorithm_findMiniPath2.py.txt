import numpy as np
from collections import defaultdict

class Node:
    def __init__(self, vertex_name, dist=float('inf'), trace=None):
        self.vertex_name = vertex_name
        self.dist = dist
        self.trace = trace

class Graph:
    def __init__(self):
        self.V = []  # Vertices
        self.E = []  # Edges
        self.minipath = []  # Shortest path
        self.numVer = 0  # Number of vertices
        self.numEdge = 0  # Number of edges

    def add_edge(self, src, dest, weight):
        self.E.append([src, dest, weight])

    def bellman_ford(self, src, dest):
        # Initialize distances
        for v in self.V:
            v.dist = float("inf")
        self.V[src - 1].dist = 0

        # Relax edges |V| - 1 times
        for _ in range(self.numVer - 1):
            for u, v, w in self.E:
                if self.V[v - 1].dist > self.V[u - 1].dist + w:
                    self.V[v - 1].dist = self.V[u - 1].dist + w
                    self.V[v - 1].trace = u

        # Check for negative cycle
        if self.V[dest - 1].dist == float('inf'):
            return

        # Reconstruct the shortest path
        self.minipath.append(dest)
        while src not in self.minipath:
            self.minipath.append(self.V[self.minipath[-1] - 1].trace)

    def read_input_from_file(self, file_path):
        with open(file_path, 'r') as fp:
            line = fp.readline()
            coords = line.split()
            self.numVer = int(coords[0])
            self.numEdge = int(coords[1])
            s = int(coords[2])
            f = int(coords[3])

            line = fp.readline()
            while line:
                coords = line.split()
                self.add_edge(int(coords[0]), int(coords[1]), int(coords[2]))
                line = fp.readline()

        for i in range(1, self.numVer + 1):
            self.V.append(Node(i))

        return s, f

    def write_output_to_file(self, file_path, s, f):
        with open(file_path, 'w') as fp:
            if s not in self.minipath:
                fp.write("\nThere are no paths from %d to %d" % (self.V[s - 1].vertex_name, self.V[f - 1].vertex_name))
            else:
                fp.write("Distance from %d to %d is: %d\n" % (self.V[s - 1].vertex_name,
                                                               self.V[f - 1].vertex_name,
                                                               self.V[f - 1].dist))
                fp.write('%d' % self.V[f - 1].vertex_name)
                for v in self.minipath:
                    if self.V[v - 1].trace is not None:
                        fp.write(" <-- %d" % self.V[v - 1].trace)

# Main program
G = Graph()
print("\nRead input from file")
start_vertex, end_vertex = G.read_input_from_file('minpath_in.txt')
G.bellman_ford(start_vertex, end_vertex)
print("\nWrite output into file")
G.write_output_to_file('minpath_out.txt', start_vertex, end_vertex)
