import socket
import time
from threading import Thread
from collections import defaultdict

class Server(Thread):
    def __init__(self):
        super().__init__()

    def run(self):
        graph = self.load_graph_data('test')
        d = self.initialize_distance(graph)

        server_socket = socket.socket()
        host = ""
        port = 60000
        server_socket.bind((host, port))
        server_socket.listen(5)

        self.send_graph_data(server_socket, graph, d)

    def load_graph_data(self, filename):
        graph = defaultdict(dict)
        with open(filename) as file:
            lines = file.readlines()
            for i in range(1, int(lines[0]) + 1):
                elements = lines[i].split()
                for j in range(1, len(elements), 2):
                    graph[i][int(elements[j])] = int(elements[j + 1])
        return graph

    def initialize_distance(self, graph):
        d = defaultdict(dict)
        for node in graph:
            for other_node in graph:
                d[node][other_node] = 0 if node == other_node else float('inf')
        return d

    def send_graph_data(self, server_socket, graph, d):
        while True:
            conn, addr = server_socket.accept()
            for u in graph:
                data_to_send = self.construct_message(u, graph, d)
                conn.send(data_to_send.encode())
            conn.close()
            break

    def construct_message(self, u, graph, d):
        message = f"{u} "
        for v in graph[u]:
            message += f"{v} {d[u][v]}\n"
        return message

class Receiver(Thread):
    def __init__(self):
        super().__init__()

    def run(self):
        client_socket = socket.socket()
        host = ""
        port = 60000
        client_socket.connect((host, port))

        while True:
            data = client_socket.recv(1024).decode()
            if data == 'Bye':
                break
        client_socket.close()

if __name__ == '__main__':
    receiver = Receiver()
    server = Server()
    server.start()
    time.sleep(3)
    receiver.start()
    server.join()
    receiver.join()
