import socket

# Create a UDP socket
send_socket = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)

# Set socket option to allow reusing address
send_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)

# Bind the socket to a specific address and port (localhost, port 8179)
send_socket.bind(("", 8179))

# Define the destination address and port
destination_address = ('localhost', 8181)

# Send a message "Hello" to the destination
send_socket.sendto(b"Hello", destination_address)

# Receive a response from the server (up to 4096 bytes)
response_data, _ = send_socket.recvfrom(4096)

# Print the received data
print(response_data.decode())

# Close the socket
send_socket.close()
