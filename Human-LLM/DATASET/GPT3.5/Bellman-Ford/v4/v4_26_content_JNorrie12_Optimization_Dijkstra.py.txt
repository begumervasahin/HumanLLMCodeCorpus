import numpy as np
import sys
import csv
import time
import math

def dijkstra(ist, isp, wei):
    if ist == isp:
        return [ist]

    N = len(wei)
    Inf = sys.maxsize
    UnVisited = np.ones(N, int)
    cost = np.ones(N) * 1.e6
    par = -np.ones(N, int) * Inf
    jj = ist
    cost[jj] = 0
    UnVisited[jj] = 0
    tmp = UnVisited * wei[jj, :]
    ineigh = np.nonzero(tmp)[0]
    L = np.count_nonzero(UnVisited)

    while L != 0:
        for k in ineigh:
            newcost = cost[jj] + wei[jj, k]
            if newcost < cost[k]:
                cost[k] = newcost
                par[k] = jj

        icnsdr = np.nonzero(UnVisited)[0]
        cmin, icmin = cost[icnsdr].min(), cost[icnsdr].argmin()
        jj = icnsdr[icmin]
        UnVisited[jj] = 0
        tmp = UnVisited * wei[jj, :]
        ineigh = np.nonzero(tmp)[0]
        L = np.count_nonzero(UnVisited)

    shpath = [isp]
    while par[isp] != ist:
        shpath.append(par[isp])
        isp = par[isp]
    shpath.append(ist)
    return shpath[::-1]

def calculate_weights(RX, RY, RA, RB, RV):
    n = len(RX)
    wei = np.zeros((n, n), dtype=float)
    m = len(RA)

    for i in range(m):
        xa = RX[RA[i]-1]
        ya = RY[RA[i]-1]
        xb = RX[RB[i]-1]
        yb = RY[RB[i]-1]
        dd = math.sqrt((xb-xa)**2 + (yb-ya)**2)
        tt = dd / RV[i]
        wei[RA[i]-1, RB[i]-1] = tt

    return wei
