import sys
import os
import time
import msvcrt
import threading
import json
from socket import socket, AF_INET, SOCK_DGRAM
from readFile import read_file

send_lock = threading.Lock()

def send_cost_change(port, node, cost):
    with send_lock:
        sock = socket(AF_INET, SOCK_DGRAM)
        message = {
            'packettype': "change",
            'sendingNode': "CostEditor",
            'node': node,
            'cost': cost,
            'port': port
        }
        message_json = json.dumps(message)
        sock.sendto(message_json.encode(), ('localhost', port))

nodes = {}

# Populate nodes dictionary with data from command line arguments
for arg in sys.argv[1:]:
    node = arg[-5]
    nodes[node] = read_file(arg)

pointer = 0
user_input = 'n'
nodes_taken = False

while True:
    os.system("cls")  # Clear screen (Windows specific)
    print("\n\t         COST EDITOR\n")
    print("\t   Use Q and W for cursor\n\t   Use double E to select\n\n")
    print("=" * 44)

    done = []
    option_num = 0

    for node_a in nodes:
        for node_b in nodes[node_a]:
            if node_b not in done:
                if option_num == pointer:
                    print("  ->", end="")
                cost = "{0:.1f}".format(float(nodes[node_a][node_b]['cost']))
                print(f"\t Cost between {node_a} and {node_b} is {cost}")
                option_num += 1
        done.append(node_a)

    print("=" * 44)
    user_input = msvcrt.getch().decode("utf-8")

    if user_input in ['Q', 'q']:
        pointer -= 1
    elif user_input in ['W', 'w']:
        pointer += 1

    pointer = max(pointer, 0)
    pointer = min(pointer, option_num - 1)

    if nodes_taken:
        nodes_taken = False
        user_input = 'n'
        print(f"\n    Enter new cost for selected nodes: ", end="")
        new_cost = input()
        print()
        for node_a in nodes:
            for node_b in nodes[node_a]:
                if pointer == 0:
                    port_a_to_b = nodes[node_a][node_b]['port']
                    nodes[node_a][node_b]['cost'] = new_cost
                    threading.Thread(target=send_cost_change, kwargs={'port': port_a_to_b, 'node': node_a, 'cost': new_cost}).start()
                elif pointer == 1:
                    port_b_to_a = nodes[node_b][node_a]['port']
                    nodes[node_b][node_a]['cost'] = new_cost
                    threading.Thread(target=send_cost_change, kwargs={'port': port_b_to_a, 'node': node_b, 'cost': new_cost}).start()
        time.sleep(1)
        print("\n              Costs updated")
        time.sleep(2)
