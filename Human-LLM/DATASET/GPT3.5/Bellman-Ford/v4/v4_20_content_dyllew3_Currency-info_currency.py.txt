import urllib.request
import json
import threading
import time

CURRENCIES = [
    "USD", "EUR", "JPY", "BGN", "CZK", "DKK", "GBP", "HUF", "LTL", "LVL",
    "PLN", "RON", "SEK", "CHF", "NOK", "HRK", "RUB", "TRY", "AUD", "BRL",
    "CAD", "CNY", "HKD", "ILS", "INR", "KRW", "MXN", "MYR", "NZD", "PHP",
    "SGD", "THB", "ZAR"
]


def generate_thread_id():
    """
    Generate unique thread IDs.
    """
    id = 0
    while True:
        yield id
        id += 1


class CurrencyThread(threading.Thread):
    """
    Thread class for fetching currency exchange rates.
    """

    def __init__(self, start, end, currency_list):
        threading.Thread.__init__(self)
        self.subset = currency_list[start:end]
        self.full_list = currency_list
        self.base_url = "https://query.yahooapis.com/v1/public/yql?q=select%20*%20from%20yahoo.finance.xchange%20where%20pair%20in%20("
        self.base_url_end = ")&format=json&diagnostics=true&env=store%3A%2F%2Fdatatables.org%2Falltableswithkeys&callback="
        self.data = []

    def run(self):
        """
        Run method for executing the thread.
        """
        self.data = self.get_currency_rates()

    def get_currency_rates(self):
        """
        Fetch currency exchange rates from Yahoo Finance API.
        """
        query_params = []
        for i in self.subset:
            for j in self.full_list:
                if i != j:
                    query_params.append("%22" + i + j + "%22")
        query_url = self.base_url + "%2C".join(query_params) + self.base_url_end

        try:
            with urllib.request.urlopen(query_url) as url:
                data = json.loads(url.read().decode())
                return data["query"]["results"]["rate"]
        except Exception as e:
            print(query_url)
            return None

    def __repr__(self):
        return "Subset is " + str(self.subset)


def run_all_threads(thread_list):
    """
    Start and join all threads.
    """
    for thread in thread_list:
        thread.start()
    for thread in thread_list:
        thread.join()


def make_threads(amount, cur_list=CURRENCIES):
    """
    Create threads to fetch currency exchange rates.
    """
    threads = []
    if amount > 0:
        inc = len(cur_list) // amount
        for i in range(0, amount):
            start = inc * i
            end = inc * (i + 1) if i != amount - 1 else len(cur_list)
            threads.append(CurrencyThread(start, end, cur_list))
    return threads


def main():
    """
    Main function to orchestrate fetching currency exchange rates.
    """
    start_time = time.time()
    num_threads = 5
    threads = make_threads(num_threads, CURRENCIES)
    run_all_threads(threads)

    count = sum(len(thread.data) for thread in threads)
    time_taken = time.time() - start_time

    print("Time taken: {:.2f} seconds".format(time_taken))
    print("Number of currency matches:", count)


if __name__ == "__main__":
    main()
