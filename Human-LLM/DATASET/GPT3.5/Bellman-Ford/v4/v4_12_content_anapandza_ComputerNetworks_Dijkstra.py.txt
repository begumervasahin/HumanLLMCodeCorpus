# Function to extract the node with the minimum distance from the priority queue
def popmin(pqueue):
    lowest_distance = float('inf')
    node_with_lowest_distance = None
    for node, distance in pqueue.items():
        if distance < lowest_distance:
            lowest_distance = distance
            node_with_lowest_distance = node
    del pqueue[node_with_lowest_distance]
    return node_with_lowest_distance

# Implementation of Dijkstra's algorithm
def dijkstra(graph, start):
    # Initialize priority queue, distances, and predecessors    
    pqueue = {}
    distances = {}
    predecessors = {}
    for vertex in graph:
        distances[vertex] = float('inf')
        predecessors[vertex] = -1
    distances[start] = 0

    # Fill priority queue with initial distances
    for vertex in graph:
        pqueue[vertex] = distances[vertex]

    # Main loop of Dijkstra's algorithm
    while pqueue:
        # Extract the node with the minimum distance
        u = popmin(pqueue)
        for v, weight in graph[u].items():
            new_distance = distances[u] + weight
            # Update distances and predecessors if a shorter path is found
            if new_distance < distances[v]:
                pqueue[v] = new_distance
                distances[v] = new_distance
                predecessors[v] = u

    return distances, predecessors

# Define the graph
graph = {
    0: {1: 6, 2: 1, 3: 4},
    1: {4: 3},
    2: {1: -3, 3: 2},
    3: {4: -1},
    4: {2: 5}
}

# Run Dijkstra's algorithm on the graph
distances, predecessors = dijkstra(graph, 0)

# Print the results
print("Shortest distance from start node 0 to all other nodes:")
for vertex, distance in distances.items():
    print(vertex, '=', distance)

print("Paths from start node to all other nodes (predecessors of nodes):")
for vertex, predecessor in predecessors.items():
    print(vertex, '=', predecessor)
