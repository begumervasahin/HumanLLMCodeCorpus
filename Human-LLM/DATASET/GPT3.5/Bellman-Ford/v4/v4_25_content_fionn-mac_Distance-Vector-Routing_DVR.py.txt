import threading
import socket
import time
import json
import csv
import sys

class NodeThread(threading.Thread):
    def __init__(self, thread_id, routing_table, neighbors):
        threading.Thread.__init__(self)
        self.thread_id = thread_id
        self.routing_table = routing_table
        self.neighbors = neighbors
        self.port = 65505
        self.is_active = True

    def update_routing_table(self, new_row, vertex_number):
        new_vertex_id, new_distance = new_row
        if self.routing_table[new_vertex_id - 1][1] > self.routing_table[vertex_number - 1][1] + new_distance:
            self.routing_table[new_vertex_id - 1][1] = self.routing_table[vertex_number - 1][1] + new_distance
            self.routing_table[new_vertex_id - 1][2] = vertex_number

    def stop(self):
        self.is_active = False
        with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as udp_socket:
            host = ''
            address = (host, self.port + self.thread_id)
            udp_socket.sendto("End".encode(), address)

    def run(self):
        while self.is_active:
            with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as udp_socket:        
                host = ''
                udp_socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)        
                udp_socket.bind((host, self.port + self.thread_id))
                buffer_size = 1024
                recv_node, _ = udp_socket.recvfrom(buffer_size)
                if recv_node.decode() == "End":
                    return
                for _ in range(num_vertices):
                    message, _ = udp_socket.recvfrom(buffer_size)
                    self.update_routing_table(json.loads(message.decode()), int(recv_node.decode()))

    def send(self):
        with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as udp_socket:
            host = ''
            for i in range(1, 6):
                address = (host, self.port + i)
                if i in self.neighbors:
                    udp_socket.sendto(str(self.thread_id).encode(), address)
                    time.sleep(0.00001)
                    for j in range(num_vertices):
                        udp_socket.sendto(json.dumps(self.routing_table[j]).encode(), address)  
                        time.sleep(0.00001)

def read_input_file(file_path):
    input_data = []
    with open(file_path, 'r') as file:
        reader = csv.reader(file, delimiter=' ')
        for row in reader:
            input_data.append(row)
    return input_data

if __name__ == "__main__":
    input_data = read_input_file(sys.argv[1])
    num_vertices = int(input_data[0][0])
    routing_info = [[[_ + 1, 10000000, _ + 1] for _ in range(num_vertices)] for _ in range(num_vertices)]
    neighbors = [[] for _ in range(num_vertices)]

    for i in range(num_vertices):
        num_edges = int(input_data[i + 1][0])
        for j in range(num_edges):
            vertex_id = int(input_data[i + 1][2 * j + 1])
            routing_info[i][vertex_id - 1][0] = vertex_id
            routing_info[i][vertex_id - 1][1] = int(input_data[i + 1][2 * j + 2])       
            neighbors[vertex_id - 1].append(i + 1)

    threads = [NodeThread(i + 1, routing_info[i], neighbors[i]) for i in range(num_vertices)]

    for thread in threads:
        thread.start()

    for _ in range(num_vertices - 1):
        for thread in threads:
            thread.send()

    with open("out", 'w') as file:
        file.write(str(num_vertices) + "\n")

    for thread in threads:
        thread.print_routing_table(file)

    for thread in threads:
        thread.stop()
        thread.join()
