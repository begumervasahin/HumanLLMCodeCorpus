# Print welcome message
print("Emulated Bellman-Ford Algorithm")
print("Authors: PoOya Khandel, Mohammad Hossein Tavakoli Bina\n")

# Import necessary modules
from BellmanFord import BFA
import msvcrt
import re
import timeit

# Initialize variables
elapsedTime = 0
start = 0
whichPort = {}
adrToName = {}
routerCount = 0
hit = None

# Get router's name from user input
routerName = input("Welcome to Emulated Bellman-Ford Algorithm\n"
                   "Please enter your router's name:\n")

# Read port mapping file and populate dictionaries
with open("which_port.txt") as whichRouter:
    for line in whichRouter:
        port = int(line[0])
        router_id = int(line[2:6])
        whichPort[port] = router_id
        adrToName[router_id] = port
        routerCount += 1

# Start timer
start = timeit.default_timer()

# Read initial cost for the router from the adjacency matrix file
with open("adj_mat.txt") as adjMatrix:
    myLine = adjMatrix.readlines()[int(routerName) - 1]
    initialCost = myLine.rstrip('\n').split(" ")
    print('Initial Cost is {}\n'.format(initialCost))

# Create an instance of the Bellman-Ford Algorithm
myBf = BFA(routerCount, initialCost, routerName, whichPort, adrToName)
myBf.who_to_send()

# Prompt user to start Bellman-Ford Algorithm
while True:
    start_prompt = "To start the BellmanFord Algorithm, Enter 's'\n"
    s = input(start_prompt)
    if s == 's':
        break
    else:
        print("Invalid input! Please enter 's' to start the algorithm.")

# Send initial data
myBf.send()

# Main loop
while True:
    # Check if a key has been pressed
    hit = msvcrt.kbhit()
    elapsedTime = timeit.default_timer() - start

    # Send updates if more than 0.1 second has passed
    if elapsedTime > 0.1:
        myBf.send()
        start = timeit.default_timer()

    # Receive updates
    myBf.receive()

    # Check for keyboard input
    if hit:
        key = ord(msvcrt.getch())
        if key == ord('u'):
            # Read new cost from the adjacency matrix file
            with open("adj_mat.txt") as adjMatrix:
                myLine = adjMatrix.readlines()[int(routerName) - 1]
                newCost = myLine.rstrip('\n').split(" ")
                print('New cost is {}\n'.format(newCost))
                myBf.check_cost(newCost)
