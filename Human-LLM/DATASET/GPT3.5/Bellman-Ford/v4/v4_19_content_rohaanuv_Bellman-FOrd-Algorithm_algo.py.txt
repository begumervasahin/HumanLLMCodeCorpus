class Graph:
    def __init__(self, vertices):
        """
        Initialize a graph with a specified number of vertices.

        Parameters:
        vertices (int): The number of vertices in the graph.
        """
        self.vertices = vertices
        self.graph = []

    def add_edge(self, source, destination, weight):
        """
        Add an edge to the graph.

        Parameters:
        source (int): The source vertex of the edge.
        destination (int): The destination vertex of the edge.
        weight (int): The weight of the edge.
        """
        self.graph.append([source, destination, weight])

    def display_distances(self, distances):
        """
        Print the distances from the source vertex to all other vertices.

        Parameters:
        distances (list): A list containing distances from the source to each vertex.
        """
        print("Vertex\tDistance from Source")
        for index, distance in enumerate(distances):
            print(f"{index}\t\t{distance}")

    def bellman_ford(self, source):
        """
        Execute the Bellman-Ford algorithm to find the shortest path from the source to all other vertices.

        Parameters:
        source (int): The source vertex.
        """
        distances = [float("Inf")] * self.vertices
        distances[source] = 0

        # Relax all edges vertices-1 times
        for _ in range(self.vertices - 1):
            for u, v, w in self.graph:
                if distances[u] != float("Inf") and distances[u] + w < distances[v]:
                    distances[v] = distances[u] + w

        # Check for negative-weight cycles
        for u, v, w in self.graph:
            if distances[u] != float("Inf") and distances[u] + w < distances[v]:
                print("Graph contains a negative weight cycle")
                return

        # Display the result
        self.display_distances(distances)

# Example usage
if __name__ == "__main__":
    g = Graph(5)
    g.add_edge(0, 1, -1)
    g.add_edge(0, 2, 4)
    g.add_edge(1, 2, 3)
    g.add_edge(1, 3, 2)
    g.add_edge(1, 4, 2)
    g.add_edge(3, 2, 5)
    g.add_edge(3, 1, 1)
    g.add_edge(4, 3, -3)

    # Execute Bellman-Ford algorithm from vertex 0
    g.bellman_ford(0)
