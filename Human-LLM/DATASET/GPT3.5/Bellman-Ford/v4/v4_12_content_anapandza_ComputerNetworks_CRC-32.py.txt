import codecs

def generate_crc32_table():
    """
    Generate the CRC-32 table using the polynomial 0xEDB88320.

    Returns:
        A list of CRC-32 checksums for all 8-bit integers.
    """
    table = []
    for byte in range(256):
        crc = byte
        for bit in range(8):
            if crc & 1:
                crc = (crc >> 1) ^ 0xEDB88320
            else:
                crc >>= 1
        table.append(crc)
    return table

def calculate_crc32(data, crc_table):
    """
    Calculate the CRC-32 checksum for the given data.

    Args:
        data (bytes): Data for which the CRC-32 checksum is to be calculated.
        crc_table (list): The CRC-32 table.

    Returns:
        The CRC-32 checksum of the data.
    """
    crc = 0xFFFFFFFF
    for byte in data:
        crc = (crc >> 8) ^ crc_table[(crc ^ byte) & 0xFF]
    return ~crc & 0xFFFFFFFF

# Generate the CRC-32 table.
crc_table = generate_crc32_table()

# The input message in hexadecimal format.
hex_message = 'FAFAFA'

# Decode the hexadecimal message to bytes.
message_bytes = codecs.decode(hex_message, 'hex')

# Calculate the CRC-32 checksum of the message.
checksum = calculate_crc32(message_bytes, crc_table)

# Convert the message and checksum to binary strings for display.
message_bin = bin(int.from_bytes(message_bytes, 'big'))
checksum_bin = bin(checksum)[2:].zfill(32)  # Ensure the checksum is 32 bits.

print(f"Input message is: {message_bin} (0x{hex_message})")
print(f"Checksum is: {checksum_bin} ({hex(checksum)})")

# Prepare the full message with checksum for transmission.
full_message = message_bin + checksum_bin
print("Message to be sent is:")
print(full_message)
