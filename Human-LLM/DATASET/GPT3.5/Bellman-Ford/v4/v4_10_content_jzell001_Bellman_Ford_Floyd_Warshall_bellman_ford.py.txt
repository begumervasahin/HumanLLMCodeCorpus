import sys
import re
import time
import itertools

# Regular expressions for parsing input
graph_re = re.compile(r"(\d+)\s(\d+)")
edge_re = re.compile(r"(\d+)\s(\d+)\s(-?\d+)")

vertices = []
edges = []

def BellmanFord(G):
    path_pairs = []
    print("\nEntered BellmanFord")

    V, E = G[0], G[1]
    x = list(itertools.permutations(V, 2))
    inf = float('inf')

    for i in range(len(V)):
        dist = [inf] * len(V)
        index = V[i]
        dist[index] = 0

        for _ in range(1, len(V)):
            for i in range(len(x)):
                u, v = x[i][0], x[i][1]
                if dist[v] > dist[u] + float(E[u][v]):
                    dist[v] = dist[u] + float(E[u][v])

        negCycle = any(dist[v] > dist[u] + float(E[u][v]) for u, v in x)

        for i in range(len(V)):
            if dist[i] != float('inf'):
                dist[i] = int(dist[i])
            new = (str(index), str(V[i])), str(dist[i])
            path_pairs.append(new)

    print("\npathPairs")
    print(path_pairs)

    if negCycle:
        print("\nNote: A negative cycle exists")
    else:
        print("\nNote: No negative cycles exist")

    print("\nExiting BellmanFord\n")
    return path_pairs

def FloydWarshall(G):
    path_pairs = []
    print("\nEntered FloydWarshall")

    V, E = G[0], G[1]
    inf = float('inf')
    dist = E

    for i in V:
        for j in V:
            dist[i][j] = float(dist[i][j])

    for v in V:
        dist[v][v] = 0

    for k in V:
        for i in V:
            for j in V:
                if dist[i][j] > dist[i][k] + dist[k][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]

    for i in V:
        for j in V:
            if dist[i][j] != float('inf'):
                dist[i][j] = int(dist[i][j])
            new = (str(i), str(j)), str(dist[i][j])
            path_pairs.append(new)

    print("\npathPairs")
    print(path_pairs)

    print("\nExiting FloydWarshall\n")
    return path_pairs

def readFile(filename):
    global vertices
    global edges

    inFile = open(filename, 'r')
    line1 = inFile.readline()
    graph_match = graph_re.match(line1)

    if not graph_match:
        print(line1 + " not properly formatted")
        sys.exit(1)

    vertices = list(range(int(graph_match.group(1))))
    edges = [[float("inf")] * len(vertices) for _ in range(len(vertices))]

    for line in inFile.readlines():
        line = line.strip()
        edge_match = edge_re.match(line)

        if edge_match:
            source, sink, weight = int(edge_match.group(1)), int(edge_match.group(2)), int(edge_match.group(3))

            if source >= len(vertices) or sink >= len(vertices):
                print(f"Attempting to insert an edge between {source} and {sink} in a graph with {len(vertices)} vertices")
                sys.exit(1)

            edges[source][sink] = weight

    print("\nVertices    Edges")
    for i in range(len(edges)):
        print(str(vertices[i]) + "         " + str(edges[i]))

    return vertices, edges

def main(filename, algorithm):
    algorithm = algorithm[1:]
    G = readFile(filename)
    print("\nAlgorithm:", algorithm)

    if algorithm == 'b' or algorithm == 'B':
        start = time.time()
        BellmanFord(G)
        end = time.time()
        BFTime = end - start
        print("\nBellman-Ford timing:", BFTime, "seconds\n")

    if algorithm == 'f' or algorithm == 'F':
        start = time.time()
        FloydWarshall(G)
        end = time.time()
        FWTime = end - start
        print("Floyd-Warshall timing:", FWTime, "seconds")

    if algorithm == "both":
        start = time.time()
        BellmanFord(G)
        end = time.time()
        BFTime = end - start

        start = time.time()
        FloydWarshall(G)
        end = time.time()
        FWTime = end - start

        print("\nBellman-Ford timing:", BFTime, "seconds")
        print("Floyd-Warshall timing:", FWTime, "seconds\n")

if __name__ == '__main__':
    if len(sys.argv) < 3:
        print("python bellman_ford.py -<f|b> <input_file>")
        sys.exit(1)

    main(sys.argv[2], sys.argv[1])
