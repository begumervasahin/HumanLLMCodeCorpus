import sys
from data_operation import get_graph
import numpy as np
from pyspark import SparkContext
import datetime
from pprint import pprint

def initial(node, source):
    """
    Initialize a node with initial distance, path, and visit flag.
    """
    u, edges = node
    distance = np.inf
    path = str(u)
    flag = 1 if u == source else 0
    if u == source:
        distance = 0
    return (u, (edges, distance, path, flag))

def relax(value1, value2):
    """
    Relax paths and update distances.
    """
    neighbors1, distance1, path1, flag1 = value1
    neighbors2, distance2, path2, flag2 = value2

    if distance1 <= distance2:
        distance = distance1
        path = path1
    else:
        distance = distance2
        path = path2

    flag = flag1 | flag2
    return (neighbors1 or neighbors2, distance, path, flag)

def split_nodes_iterative(node, uid):
    """
    Split nodes after relaxation and update visit flags.
    """
    nid = node[0]
    distance = node[1][1]
    neighbors = node[1][0]
    path = node[1][2]

    elements = path.split('->')
    if elements[-1] != nid:
        path = path + '->' + nid

    flag = 1 if nid == uid else node[1][3]
    return (nid, (neighbors, distance, path, flag))

def format_result(node):
    """
    Format the result node.
    """
    nid = node[0]
    min_distance = node[1][1]
    path = node[1][2]
    return nid, min_distance, path

def generate_edges(u_node, neighbor):
    """
    Generate new edges for Dijkstra's algorithm.
    """
    old_path = u_node[1][2]
    old_dis = u_node[1][1]
    v_id, weight = neighbor.split(":")
    new_dis = old_dis + float(weight)
    new_path = old_path + '->' + v_id
    return (v_id, ('None', new_dis, new_path, 0))

def Dijkstra(source, graph, sc):
    """
    Implements Dijkstra's algorithm using Spark.
    """
    nodes = graph.map(lambda node: initial(node, source))
    nodes_count = nodes.count()

    for _ in range(nodes_count):
        candidate_nodes = nodes.filter(lambda node: node[1][3] == 0)
        min_node = candidate_nodes.min(lambda node: node[1][1])
        all_edges = sc.parallelize(map(
            lambda neighbor: generate_edges(min_node, neighbor), min_node[1][0]
        ))
        mapper = nodes.union(all_edges)
        reducer = mapper.reduceByKey(lambda x, y: relax(x, y))
        nodes = reducer.map(lambda node: split_nodes_iterative(node, min_node[0]))

    result = reducer.sortByKey().map(lambda node: format_result(node))
    return result
