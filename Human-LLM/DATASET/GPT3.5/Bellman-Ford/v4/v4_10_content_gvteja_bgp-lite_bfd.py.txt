import sys
import json
import socket
import pickle
from multiprocessing import Process
from select import select
from datetime import datetime

# Get the current datetime
now = datetime.now

# Check if the required arguments are provided
if len(sys.argv) < 3:
    print('Please pass the 2 args, node name and periodic refresh value')
    sys.exit()

# Extract node name and periodic refresh value from command line arguments
node_name = sys.argv[1]
timeout = int(sys.argv[2])
neighbour_wts_f = node_name

# Initialize variables for tracking time
last_read_time = now()
last_upd_time = now()

# Function to read neighbors from file
def get_neighbours(file_name):
    with open(file_name, 'r') as f:
        neighbours_txt = f.read().split('\n')

    neighbours = {}
    ips = {}
    links = {}

    for neighbour_info in neighbours_txt:
        src_ip, n, dst_ip, cost = neighbour_info.strip().split(':')
        cost = int(cost)
        neighbours[n] = cost
        ips[dst_ip] = n
        links[src_ip] = dst_ip

    global last_read_time
    last_read_time = now()
    return neighbours, ips, links

# Initialize neighbors and routing tables
neighbours, ips, links = get_neighbours(neighbour_wts_f)
distances = {neighbour: cost for neighbour, cost in neighbours.items()}
max_distance = 100000
previous_hops = {}
distance_vectors = {}
connected_sockets = {}
accepted_sockets = {}
port = 9999

# Function to connect to neighbors
def connect_to_neighbours(neighbours):
    for ip, name in neighbours.items():
        s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        s.connect((ip, port))
        connected_sockets[name] = s

# Function to listen for incoming connections from neighbors
def listen_neighbour_on(ip, neighbour):
    s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    s.bind((ip, port))
    s.listen(1)
    sock, addr = s.accept()
    accepted_sockets[neighbour] = sock
    print('Accepted connection from', addr)

# Create processes for listening and connecting to neighbors
threads = []
for src_ip, dst_ip in links.items():
    neighbour = ips[dst_ip]
    t = Process(target=listen_neighbour_on, args=(src_ip, neighbour,))
    threads.append(t)

threads.append(Process(target=connect_to_neighbours, args=(ips,)))

# Start and join all threads
for t in threads:
    t.start()

for t in threads:
    t.join()

# Prepare lists for socket operations
read_sockets = []
read_names = []
for name, sock in accepted_sockets.items():
    read_sockets.append(sock)
    read_names.append(name)

write_sockets = connected_sockets.values()
read_timeout = 0

# Main loop for communication and routing updates
while True:
    ready_to_read, _, _ = select(read_sockets, [], read_sockets, read_timeout)      
    update_needed = False

    # Process incoming data from neighbors
    for sock in ready_to_read:
        data = sock.recv(100000)
        neighbour = read_names[read_sockets.index(sock)]
        neighbour_dv = json.loads(data)
        distance_vectors[neighbour] = neighbour_dv
        cost = neighbours[neighbour]

        for destination, est_cost in neighbour_dv.items():
            if est_cost + cost < distances.get(destination, max_distance):
                distances[destination] = est_cost + cost
                previous_hops[destination] = neighbour
                update_needed = True

    # Check for periodic updates or changes in neighbor configuration
    if (not update_needed) and ((now() - last_upd_time).seconds >= timeout):
        update_needed = True

    if (now() - last_read_time).seconds >= timeout:
        updated_neighbours, ips, links = get_neighbours(neighbour_wts_f)

        if neighbours != updated_neighbours:
            neighbours = updated_neighbours
            new_distances = {neighbour: cost for neighbour, cost in neighbours.items()}

            for neighbour, neighbour_dv in distance_vectors.items():
                cost = neighbours[neighbour]

                for destination, est_cost in neighbour_dv.items():
                    if cost + est_cost < new_distances.get(destination, max_distance):
                        new_distances[destination] = cost + est_cost
                        previous_hops[destination] = neighbour

            if new_distances != distances:
                distances = new_distances
                update_needed = True

    # Perform updates if needed
    if update_needed:
        data = json.dumps(distances)

        for sock in write_sockets:
            sock.sendall(data)

        last_upd_time = now()
        pickle.dump(distance_vectors, open(f'{node_name}_dv.p', 'wb'))
        pickle.dump(previous_hops, open(f'{node_name}_p.p', 'wb'))
        pickle.dump(distances, open(f'{node_name}_d.p', 'wb'))

    read_timeout = timeout - (now() - max(last_upd_time, last_read_time)).seconds
