def xor_operation(a, b):
    """
    Perform a bitwise XOR operation between two binary strings of equal length.

    Args:
    - a (str): The first binary string.
    - b (str): The second binary string.

    Returns:
    - str: The result of the bitwise XOR operation.
    """
    # Initialize an empty string to store the XOR result.
    xor_result = ""
    
    # Iterate over the characters in the binary strings.
    for bit_a, bit_b in zip(a, b):
        # Perform XOR operation for each pair of bits and concatenate the result.
        xor_result += '0' if bit_a == bit_b else '1'
    
    return xor_result

def modulo_2_division(dividend, divisor):
    """
    Perform modulo-2 division on a binary string (dividend) by another binary string (divisor).

    Args:
    - dividend (str): The binary string to be divided.
    - divisor (str): The binary string divisor.

    Prints:
    - The quotient and remainder of the modulo-2 division.
    """
    # Determine the length of the divisor.
    pick = len(divisor)
    # Initialize the temporary variable with the initial portion of the dividend.
    tmp = dividend[:pick]
    
    # Use a list to keep track of which bits result in a '1' in the quotient.
    quotient_bits = []

    # Process each bit of the dividend.
    for bit in dividend[pick:]:
        if tmp[0] == '1':
            # Perform XOR between the divisor and the current temporary value.
            tmp = xor_operation(divisor, tmp) + bit
        else:
            # Perform XOR with a string of '0's (same length as the divisor).
            tmp = xor_operation('0'*pick, tmp) + bit
        # Record whether a '1' or '0' was added to the quotient.
        quotient_bits.append('1' if tmp[0] == '1' else '0')

    # Handle the last portion of the division process.
    if tmp[0] == '1':
        tmp = xor_operation(divisor, tmp)
    else:
        tmp = xor_operation('0'*pick, tmp)

    # The final value of tmp is the remainder. The quotient is formed from quotient_bits.
    print("Quotient is: ", ''.join(quotient_bits))
    print("Remainder is:", tmp)

# Example usage
data = "10011101"  # Correcting the dividend to have an extra bit for clarity
key = "1011"
modulo_2_division(data, key)
