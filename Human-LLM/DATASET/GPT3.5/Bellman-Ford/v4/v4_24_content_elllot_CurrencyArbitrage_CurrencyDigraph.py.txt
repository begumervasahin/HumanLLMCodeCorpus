from collections import defaultdict
from math import log
from Scraper import Scraper  # Assuming Scraper is a class or module that provides currency exchange rates

class CurrencyDigraph:
    def __init__(self, src=None):
        # Initialize graph, weights, Scraper instance, and other attributes
        self.graph = defaultdict(list)  # Stores the graph structure as a defaultdict
        self.weights = {}  # Stores the weights of edges in the graph
        self.scraper = Scraper()  # Instantiate the Scraper class to get currency rates
        self.parents = {}  # Stores parent vertices in a cycle
        self.distances = {vertex: float("inf") for vertex in self.graph}  # Initialize distances to infinity
        if src:
            self.set_source(src)  # Set the source currency if provided
        self.currency_list = self.get_currency_list()  # Generate a sorted list of currencies
        self.print_cache = []  # Cache for printing the currency list
        self.items_per_line = 4  # Number of items to display per line in the currency list

    def set_source(self, src):
        """Set the source currency and initialize its distance to 0."""
        self.distances[src] = 0.0

    def generate_graph(self):
        """Populate the graph and weights based on currency exchange rates."""
        for rate_pair in self.scraper.rates:
            source, target = rate_pair
            self.graph[source].append(target)  # Add target currency to the adjacency list of source currency
            self.weights[rate_pair] = -log(self.scraper.rates[rate_pair])  # Calculate and store the weight

    def find_cycle(self, v):
        """
        Find a cycle in the parent map using Depth-First Search (DFS) starting from vertex v.
        
        :param v: vertex in the cycle
        :returns: list representing the cycle
        """
        visited, cycle = set(), []
        while v not in visited:
            visited.add(v)
            cycle.append(v)
            v = self.parents[v]
        return cycle[cycle.index(v):][::-1]  # Return the cycle in reverse order

    def get_currency_list(self):
        """Convert currency rates dictionary to a sorted list for printing."""
        return sorted([[k, v] for k, v in self.scraper.strs.items()], key=lambda x: x[1])

    def print_currency_list(self):
        """
        Print the currency list with a specified number of items per line.
        Caches the previous result for efficient reprinting if necessary.
        """
        print('Currency List:')
        if self.print_cache:
            for line in self.print_cache:
                print(line, '\n')
        else:
            i = 0
            while i < len(self.currency_list):
                end = i + self.items_per_line
                self.print_cache.append(self.currency_list[i:min(len(self.currency_list), end)])
                print(self.print_cache[-1], '\n')
                i = end

# Example usage
currency_digraph = CurrencyDigraph()
currency_digraph.print_currency_list()
