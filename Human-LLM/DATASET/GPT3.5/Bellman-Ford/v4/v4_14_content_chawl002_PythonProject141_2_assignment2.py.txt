import os
import re
import sys
import time

# Regular expressions for parsing input data
graphRE = re.compile(r"(\d+)\s(\d+)")
edgeRE = re.compile(r"(\d+)\s(\d+)\s(\d+)")

def BellmanFord(graph_data):
    paths = []
    for source in range(len(graph_data[0])):
        distances = [float("inf")] * len(graph_data[0])
        distances[source] = 0
        for _ in range(1, len(graph_data[0])-1):
            for from_node in range(len(graph_data[0])):
                for to_node in range(len(graph_data[0])):
                    weight = graph_data[1][from_node][to_node]
                    if float(weight) < float("inf"):
                        if distances[to_node] > distances[from_node] + float(weight):   
                            distances[to_node] = int(distances[from_node]) + int(float(weight))
        paths.append(distances)
    return paths

def FloydWarshall(graph_data):
    paths = []
    distances = [[float("inf") for _ in range(len(graph_data[0]))] for _ in range(len(graph_data[0]))]
    for i in range(len(graph_data[0])):
        distances[i][i] = 0
        for j in range(len(graph_data[0])):
            if i != j and float(graph_data[1][i][j]) < float("inf"):
                distances[i][j] = int(float(graph_data[1][i][j]))
    for k in range(len(graph_data[0])):
        for i in range(len(graph_data[0])):
            for j in range(len(graph_data[0])):
                if distances[i][j] > distances[i][k] + distances[k][j]:
                    distances[i][j] = distances[i][k] + distances[k][j]
    paths.append(distances)
    return paths

def readGraph(filename):
    with open(filename, 'r') as inFile:
        line1 = inFile.readline()
        graphMatch = graphRE.match(line1)
        if not graphMatch:
            print(line1 + " not properly formatted")
            sys.exit(1)
        vertices = list(range(int(graphMatch.group(1))))
        edges = [[float("inf") for _ in range(len(vertices))] for _ in range(len(vertices))]
        for line in inFile.readlines():
            line = line.strip()
            edgeMatch = edgeRE.match(line)
            if edgeMatch:
                source = int(edgeMatch.group(1))
                sink = int(edgeMatch.group(2))
                if source >= len(vertices) or sink >= len(vertices):
                    print(f"Attempting to insert an edge between {source} and {sink} in a graph with {len(vertices)} vertices")
                    sys.exit(1)
                weight = edgeMatch.group(3)
                edges[source][sink] = weight
    return (vertices, edges)

def writeFile(lengthMatrix, filename):
    baseName = os.path.splitext(os.path.split(filename)[1])[0]
    with open(f'output/{baseName}_output.txt', 'w+') as outFile:
        for vertex in lengthMatrix:
            for length in vertex:
                outFile.write(str(length) + ',')
            outFile.write('\n')

def main(filename, algorithm):
    algorithm = algorithm[1:]
    graphData = readGraph(filename)
    pathLengths = []
    if algorithm == 'b' or algorithm == 'B':
        pathLengths = BellmanFord(graphData)
    elif algorithm == 'f' or algorithm == 'F':
        pathLengths = FloydWarshall(graphData)
    elif algorithm == "both":
        start = time.process_time()
        BellmanFord(graphData)
        BFTime = time.process_time() - start
        FloydWarshall(graphData)
        FWTime = time.process_time() - start
        print("Bellman-Ford timing:", BFTime)
        print("Floyd-Warshall timing:", FWTime)
    writeFile(pathLengths, filename)

if __name__ == '__main__':
    if len(sys.argv) < 3:
        print("Usage: python bellman_ford.py -<f|b> <input_file>")
        sys.exit(1)
    main(sys.argv[2], sys.argv[1])
