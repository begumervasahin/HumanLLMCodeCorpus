# Define the graph structure
vertices = {1: 0, 2: 1, 3: 2, 4: 3, 5: 4, 6: 5}
edges = [[1, 2], [1, 3], [1, 4], [3, 4], [4, 5], [4, 6], [5, 6]]
weights = [1, 2, 4, 3, 5, 2, 4]

# Initialize variables for the algorithm
inf = float("inf")
distance = [inf for _ in vertices]  # Set initial distances to infinity
previous = [None for _ in vertices]  # Track the previous vertex in the path
distance[0] = 0  # Distance from the source to itself is 0

# Main loop to update distances based on edges and weights
for v, i in vertices.items():
    print(f"\nIteration: {i}\nResult Table:")
    for e, (start, end) in enumerate(edges):
        if start == v:
            adjacent = vertices[end]
            weight = weights[e]
            if distance[i] + weight < distance[adjacent]:
                distance[adjacent] = distance[i] + weight
                previous[adjacent] = v
                print(f"Edge: {start, end}\nVertices: {vertices}\nDistance(cost): {distance}\nPrevious: {previous}")

# Function to recursively print the shortest path
def print_path(destination):
    if previous[vertices[destination]] is not None:
        print_path(previous[vertices[destination]])  # Recursively print the previous vertex
    print(destination, end=' ')

print("\nList of vertices that make up the shortest path, with the least cost:")
print_path(6)
