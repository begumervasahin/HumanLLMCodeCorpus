import re
from allPairsShortestPath import graphRE, edgeRE

def read_file(filename):
    """
    Read graph data from a file and construct the graph.
    """
    vertices = []
    edges = []

    with open(filename, 'r') as file:
        # Read the first line to determine the number of vertices
        line1 = file.readline().strip()
        graph_match = graphRE.match(line1)

        if not graph_match:
            print(f"Error: First line '{line1}' is not properly formatted.")
            quit(1)

        num_vertices = int(graph_match.group(1))
        vertices = list(range(num_vertices))

        # Initialize the adjacency matrix with infinity for all edges
        edges = [[float("inf")] * num_vertices for _ in range(num_vertices)]

        # Process the remaining lines to populate the adjacency matrix
        for line in file.readlines():
            process_edge_line(line.strip(), edges, vertices)

    return vertices, edges

def process_edge_line(line, edges, vertices):
    """
    Process a line containing edge information and update the adjacency matrix.
    """
    edge_match = edgeRE.match(line)
    if edge_match:
        source = int(edge_match.group(1)) - 1
        sink = int(edge_match.group(2)) - 1

        if source >= len(vertices) or sink >= len(vertices):
            print(f"Error: Attempting to insert an edge between {source} and {sink} "
                  f"in a graph with {len(vertices)} vertices.")
            quit(1)

        weight = float(edge_match.group(3))
        edges[source][sink] = weight

if __name__ == "__main__":
    filename = input("Enter the filename: ")
    vertices, edges = read_file(filename)
    print("Vertices:", vertices)
    print("Edges:", edges)
