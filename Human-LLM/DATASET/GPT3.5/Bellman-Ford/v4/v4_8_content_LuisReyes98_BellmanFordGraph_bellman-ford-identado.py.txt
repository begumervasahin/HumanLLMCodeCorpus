import tkinter as tk
from tkinter import messagebox, Toplevel, Scrollbar, END, Entry, Button, Label, BOTTOM, TOP, RIGHT, LEFT
import sys

# Global variables
num_nodes = 1
num_edges = 1
node_vector = []
creating_edge = False
canvas = None
text_panel = None

# Extend Canvas functionality to draw circles
def _create_circle(self, x, y, r, **kwargs):
    return self.create_oval(x-r, y-r, x+r, y+r, **kwargs)

tk.Canvas.create_circle = _create_circle

# Extend Canvas functionality to draw circle arcs
def _create_circle_arc(self, x, y, r, **kwargs):
    if "start" in kwargs and "end" in kwargs:
        kwargs["extent"] = kwargs["end"] - kwargs["start"]
        del kwargs["end"]
    return self.create_arc(x-r, y-r, x+r, y+r, **kwargs)

tk.Canvas.create_circle_arc = _create_circle_arc

# Initialize distances and predecessors
def initialize(graph, source):
    distances = {}
    predecessors = {}

    for node in graph:
        distances[node] = float('Inf')
        predecessors[node] = None

    distances[source] = 0
    return distances, predecessors

# Relaxation step in Bellman-Ford algorithm
def relax(node, neighbour, graph, distances, predecessors):
    if distances[neighbour] > distances[node] + graph[node][neighbour]:
        distances[neighbour] = distances[node] + graph[node][neighbour]
        predecessors[neighbour] = node

# Bellman-Ford algorithm
def bellman_ford(graph, source):
    global text_panel
    iteration = 0
    distances, predecessors = initialize(graph, source)

    for _ in range(len(graph) - 1):
        for u in graph:
            iteration += 1
            for v in graph[u]:
                relax(u, v, graph, distances, predecessors)
            
            # Display iteration details
            text_panel.insert(END, f'Iteration {iteration} \n')
            text_panel.insert(END, 'Distance Array \n')
            text_panel.insert(END, f'{distances} \n')
            text_panel.insert(END, 'Predecessor Array \n')
            text_panel.insert(END, f'{predecessors} \n')

    # Check for negative cycles
    for u in graph:
        for v in graph[u]:
            assert distances[v] <= distances[u] + graph[u][v]

    return distances, predecessors

# Entry point of the program
def main():
    graph = {
         'a': {'b': -1, 'c':  4},
         'b': {'c':  3, 'd':  2, 'e':  2},
         'c': {},
         'd': {'b':  1, 'c':  5},
         'e': {'d': -3}
    }
    distances, predecessors = bellman_ford(graph, 'a')
    print (distances)
    print (predecessors)
    mainWindow()

# Create the main window
def mainWindow():
    global canvas
    root = tk.Tk()
    root.title("Bellman-Ford Graphs")
    bottom_frame = tk.Frame(root)
    bottom_frame.pack(side=BOTTOM)
    top_frame = tk.Frame(root)
    top_frame.pack(side=TOP)
    w = 790
    h = 600
    ws = root.winfo_screenwidth()
    hs = root.winfo_screenheight()
    x = (ws/2) - (w/2)
    y = (hs/2) - (h/2)
    root.geometry('%dx%d+%d+%d' % (w, h, x, y))
    createTopPanel(top_frame, w, h)
    arial14 = ('Arial', 14, "bold")
    circle_button = tk.Button(bottom_frame, text="Add Node", command=lambda: createNode(canvas), font=arial14)
    evaluate_button = tk.Button(bottom_frame, text="Evaluate Graph", font=arial14, command=evaluateGraph)
    delete_button = tk.Button(bottom_frame, text="Delete All", font=arial14, command=resetAll)
    circle_button.pack(side=tk.LEFT)
    evaluate_button.pack(side=tk.LEFT)
    delete_button.pack(side=tk.LEFT)
    root.mainloop()
    sys.exit(0)

# Create the top panel with canvas and text panel
def createTopPanel(top_frame, w, h):
    global canvas, text_panel
    canvas = tk.Canvas(top_frame, width=585, height=h-50, borderwidth=0, highlightthickness=0, bg="white")
    text_panel = tk.Text(top_frame, height=300, width=50)
    scroll = Scrollbar(top_frame)
    canvas.pack(side=tk.LEFT)
    scroll.pack(side=RIGHT, fill=tk.Y)
    text_panel.pack(side=LEFT)
    scroll.config(command=text_panel.yview)
    text_panel.config(yscrollcommand=scroll.set)

# Reset all global variables
def resetAll():
    global canvas, num_nodes, num_edges, creating_edge, node_vector
    canvas.delete("all")
    num_nodes = 1
    num_edges = 1
    node_vector = []
    creating_edge = False

# Evaluate the graph and display results
def evaluateGraph():
    global node_vector, text_panel
    graph = {}
    aux_edge = {}

    for node in node_vector:
        for edge in node.edges:
            aux_edge[edge.node2.id] = int(edge.weight)

        graph[node.id] = aux_edge
        aux_edge = {}

    distances, predecessors = bellman_ford(graph, node_vector[0].id)

    text_panel.insert(END, f'{distances} \n')
    text_panel.insert(END, f'{predecessors} \n')

# Create a new node on canvas
def createNode(canvas):
    global node_vector
    node = Node(canvas, 100, 120, 30)
    node_vector.append(node)

# Define the Node class
class Node:
    def __init__(self, canvas, x, y, r):
        self.edges = []
        self.canvas = canvas
        self.x = x
        self.y = y
        self.r = r
        self.moving = True
        self.drawNode()
        self.createPopupMenu()

    def moveCircle(self, event):
        if self.moving:
            pos_x = event.x - self.x
            pos_y = event.y - self.y
            self.x = event.x
            self.y = event.y
            self.canvas.move(self.circle, pos_x, pos_y)
            self.canvas.move(self.canvas_num, pos_x, pos_y)

    def drawNode(self):
        global num_nodes
        self.id = num_nodes
        self.circle = self.canvas.create_circle(self.x, self.y, self.r, fill="blue")
        self.canvas_num = self.canvas.create_text(self.x, self.y, fill="white", text=f'{self.id}')
        num_nodes += 1
        self.canvas.tag_bind(self.circle, '<B1-Motion>', self.moveCircle)
        self.canvas.tag_bind(self.canvas
