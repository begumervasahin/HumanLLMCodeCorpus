import time
import sys
from currency import CURRENCIES, make_threads, CurrencyThread

# Constants
MAX_REQ_ALLOWED = 7
NUM_TESTS = 1
MIN_THREADS = 1
MAX_THREADS = len(CURRENCIES)


def calculate_optimal_threads(num_tests=NUM_TESTS, min_threads=MIN_THREADS, max_threads=MAX_THREADS):
    """
    Calculate the optimal number of threads for fetching currency exchange rates.
    """
    print("Max number of queries is 7 * 34 per thread")
    optimal_num_threads = MIN_THREADS
    min_time = float("inf")

    for num_threads in range(min_threads, max_threads + 1):
        if (len(CURRENCIES) - (len(CURRENCIES) // num_threads * (num_threads - 1))) > MAX_REQ_ALLOWED:
            continue

        total_time = 0
        if num_tests <= 0:
            num_tests = NUM_TESTS

        for _ in range(num_tests):
            start = time.time()
            threads = make_threads(num_threads, CURRENCIES)
            CurrencyThread.run_all_threads(threads)
            total_time += time.time() - start

        avg_time = total_time / num_tests

        if avg_time < min_time:
            min_time = avg_time
            optimal_num_threads = num_threads

        test_str = f"{num_tests} tests" if NUM_TESTS > 1 else " test"
        print(f"Average time for {num_threads} threads ({test_str}): {avg_time}")

    return optimal_num_threads


if __name__ == "__main__":
    num_tests = NUM_TESTS

    if len(sys.argv) > 1:
        num_tests = int(sys.argv[1])

    threads = []

    for num_threads in range(MIN_THREADS, MAX_THREADS + 1):
        if (len(CURRENCIES) - (len(CURRENCIES) // num_threads * (num_threads - 1))) > MAX_REQ_ALLOWED:
            continue

        if num_tests <= 0:
            break

        total_time = 0
        for _ in range(num_tests):
            start = time.time()
            threads = make_threads(num_threads, CURRENCIES)
            CurrencyThread.run_all_threads(threads)
            total_time += time.time() - start

        avg_time = total_time / num_tests
        test_str = f"{num_tests} tests" if NUM_TESTS > 1 else " test"
        print(f"Average time for {num_threads} threads ({test_str}): {avg_time}")

    count = sum(len(thread.data) for thread in threads)
    print(count)
