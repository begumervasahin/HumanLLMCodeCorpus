def initialize(graph, source):
    distances = {node: float('inf') for node in graph}
    predecessors = {node: None for node in graph}
    distances[source] = 0
    return distances, predecessors

def bellman_ford(graph, source):
    distances, predecessors = initialize(graph, source)
    count_loop = 0
    for _ in range(len(graph) - 1):
        for u in graph:
            for v in graph[u]:
                count_loop += 1
                if distances[v] > distances[u] + graph[u][v]:
                    distances[v] = distances[u] + graph[u][v]
                    predecessors[v] = u
    for u in graph:
        for v in graph[u]:
            assert distances[v] <= distances[u] + graph[u][v]
    return distances, predecessors, count_loop

def dijkstra(graph, source):
    distances, predecessors = initialize(graph, source)
    unvisited = {node: distances[node] for node in graph}
    visited = {}
    count_loop = 0
    while unvisited:
        count_loop += 1
        current_node = min(unvisited, key=unvisited.get)
        current_distance = unvisited.pop(current_node)
        visited[current_node] = current_distance
        for neighbor, weight in graph[current_node].items():
            if neighbor in unvisited:
                new_distance = current_distance + weight
                if new_distance < unvisited[neighbor]:
                    unvisited[neighbor] = new_distance
                    predecessors[neighbor] = current_node
    return visited, predecessors, count_loop

def widest_bellman_ford(graph, source):
    distances, predecessors = initialize(graph, source)
    count_loop = 0
    for _ in range(len(graph) - 1):
        for u in graph:
            for v in graph[u]:
                count_loop += 1
                if distances[v] < min(distances[u], graph[u][v]):
                    distances[v] = min(distances[u], graph[u][v])
                    predecessors[v] = u
    return distances, predecessors, count_loop

def widest_dijkstra(graph, source):
    distances, predecessors = initialize(graph, source)
    unvisited = {node: float('-inf') for node in graph}
    visited = {}
    count_loop = 0
    while unvisited:
        count_loop += 1
        current_node = max(unvisited, key=unvisited.get)
        current_distance = unvisited.pop(current_node)
        visited[current_node] = current_distance
        for neighbor, weight in graph[current_node].items():
            if neighbor in unvisited:
                new_distance = min(current_distance, weight)
                if new_distance > unvisited[neighbor]:
                    unvisited[neighbor] = new_distance
                    predecessors[neighbor] = current_node
    return visited, predecessors, count_loop

def measure_algorithm_performance(graph, source):
    print("Bellman-Ford:")
    distances, predecessors, count_loop = bellman_ford(graph, source)
    print("Number of loop iterations:", count_loop)
    print("______________________________________________________")

    print("Widest Bellman-Ford:")
    distances, predecessors, count_loop = widest_bellman_ford(graph, source)
    print("Number of loop iterations:", count_loop)
    print("______________________________________________________")

    print("Dijkstra:")
    distances, predecessors, count_loop = dijkstra(graph, source)
    print("Number of loop iterations:", count_loop)
    print("______________________________________________________")

    print("Widest Dijkstra:")
    distances, predecessors, count_loop = widest_dijkstra(graph, source)
    print("Number of loop iterations:", count_loop)
    print("______________________________________________________")
