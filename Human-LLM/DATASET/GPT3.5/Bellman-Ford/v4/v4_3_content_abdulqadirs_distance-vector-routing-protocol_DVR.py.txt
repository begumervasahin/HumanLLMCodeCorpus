import sys
import time
import socket
import pickle
import threading

# Global variables
start_time = time.clock()
graph = {'A': {}, 'B': {}, 'C': {}, 'D': {}, 'E': {}, 'F': {}}
distance = {}
predecessor = {}

def bellman_ford(distance_vector_graph):
    while True:
        initialize_distances()
        time.sleep(3)
        if time.clock() - start_time < 20:
            update_graph_with_distance_vectors(distance_vector_graph)
        relax_edges()
        print_distances()

def initialize_distances():
    for node in graph:
        distance[node], predecessor[node] = float('inf'), None
    distance[router_id] = 0

def update_graph_with_distance_vectors(distance_vector_graph):
    for key in graph:
        graph[key] = distance_vector_graph[key]
        for neighbor in distance_vector_graph[key]:
            graph[key][neighbor] = distance_vector_graph[key][neighbor]
            graph[neighbor][key] = distance_vector_graph[key][neighbor]

def relax_edges():
    for _ in range(len(graph) - 1):
        for u in graph:
            for v in graph[u]:
                if distance[v] >= distance[u] + graph[u][v]:
                    distance[v], predecessor[v] = distance[u] + graph[u][v], u

def print_distances():
    for node, dist in distance.items():
        print("Distance to {}: {}".format(node, dist))
    print()

            client_socket = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        except:
            print("Socket creation failed")
        i = 0
        while i < len(port_numbers):
            time.sleep(4)
            message = {router_id: graph[router_id]}
            client_socket.sendto(pickle.dumps(message), (server_address, int(port_numbers[i])))  
            i += 1

def update_distance_vector(message):
    main_key = 'updated'
    new_data = message[main_key]
    key_1 = list(new_data.keys())[0]
    distance_vector_graph[router_id][key_1] = new_data[key_1]

def update_graph(message, timer_dict):
    key = list(message.keys())[0]
    received_data = message[key]
    for neighbor, value in received_data.items():
        graph[neighbor][key] = value
        graph[key][neighbor] = value
    timer_dict[key] = time.clock()
    handle_expired_neighbors(timer_dict)

def handle_expired_neighbors(timer_dict):
    for neighbor in neighbor_list:
        if time.clock() - timer_dict[neighbor] > 15:
            change = neighbor
            graph[router_id][change] = float('inf')
            graph[change][router_id] = float('inf')
