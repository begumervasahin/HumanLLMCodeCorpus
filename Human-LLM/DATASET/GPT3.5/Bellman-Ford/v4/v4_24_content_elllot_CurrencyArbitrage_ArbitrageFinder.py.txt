from math import exp
import argparse
from CurrencyDigraph import CurrencyDigraph
import asyncio

class ArbitrageFinder:
    def __init__(self, starting_amount):
        self.starting_amount = starting_amount
        self.digraph = CurrencyDigraph()
        self.source_currency = self.select_starting_currency()
        self.digraph.set_source(self.source_currency)
        self.find_arbitrage()

    def select_starting_currency(self):
        """
        Method to handle user input for selecting a starting currency.
        """
        print(chr(27) + "[2J")  # Clear the console
        self.digraph.print_currency_list()
        while True:
            user_input = input('Select a currency (pick an ID): ')
            if user_input.isdigit() and int(user_input) in self.digraph.scraper.keys:
                selected_currency = int(user_input)
                print('You picked {0}'.format(self.digraph.scraper.keys[selected_currency]))
                return selected_currency
            else:
                print('Please select a valid ID...')

    def arbitrage_profit(self):
        """
        Print details of the arbitrage opportunity and profit.
        """
        cycle = self.digraph.find_cycle(self.source_currency)
        original_amount = result_amount = self.starting_amount
        trade_sequence = []

        for i in range(len(cycle)):
            trade_sequence.append(self.digraph.scraper.keys[cycle[i]])
            result_amount *= exp(-self.digraph.weights[(cycle[i], cycle[(i+1) % len(cycle)])])

        profit_percent = round((result_amount - original_amount) / float(original_amount) * 100, 5)
        print('->'.join(trade_sequence))
        print(f"Started with: {original_amount} | Ended With: {result_amount} | {profit_percent}% profit")

    def find_arbitrage(self):
        """
        Detect negative weight cycles using the Bellman-Ford algorithm.
        """
        for _ in range(len(self.digraph.graph) - 1):
            for edge in self.digraph.weights:
                source, destination = edge
                if self.digraph.distances[source] != float("inf"):
                    temp = self.digraph.distances[source] + self.digraph.weights[edge]
                    if temp < self.digraph.distances[destination]:
                        self.digraph.distances[destination] = temp
                        self.digraph.parents[destination] = source
                        if destination == self.source_currency:
                            self.arbitrage_profit()
                            return

        for edge in self.digraph.weights:
            source, destination = edge
            temp = self.digraph.distances[source] + self.digraph.weights[edge]
            if destination == self.source_currency:
                if temp < self.digraph.distances[destination]:
                    self.arbitrage_profit()
                    return

        print("No Arbitrage")

def entry_point(run_count):
    """
    Entry point of the script.
    """
    parser = argparse.ArgumentParser()
    parser.add_argument('-s', '--starting_amount')
    args = parser.parse_args()
    af = ArbitrageFinder(args.starting_amount)
    loop = asyncio.get_event_loop()
    loop.run_until_complete(af.find_arbitrage())
    try:
        loop.run_forever()
    except InterruptedError:
        asyncio.asyncgens.shut_down()

if __name__ == '__main__':
    entry_point(1000)
