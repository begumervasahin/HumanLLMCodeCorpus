import csv

def parse_input_file(filename):
    Adjacency_List = {}
    W = {}
    vertices = 0

    with open(filename + ".gr") as inputfile:
        for line in inputfile:
            parts = line.split()
            if parts[0] == 'p':
                vertices = int(parts[2])
            elif parts[0] == 'a':
                W[int(parts[1]), int(parts[2])] = int(parts[3])
                T = int(parts[1])
                H = int(parts[2])
                Adjacency_List.setdefault(T, []).append(H)

        for v in range(1, vertices + 1):
            Adjacency_List.setdefault(v, [])

    return Adjacency_List, W, vertices

def bellman_ford(Adjacency_List, W, vertices, source):
    dist = {}
    pred = {}
    dist[source] = 0
    pred[source] = 0
    N = list(range(1, vertices + 1))
    N.remove(source)

    for i in N:
        dist[i] = int(vertices * max(W.values()))
        pred[i] = -1

    for _ in range(vertices):
        for i in range(1, vertices + 1):
            for j in Adjacency_List[i]:
                if dist[j] > dist[i] + W[i, j]:
                    dist[j] = dist[i] + W[i, j]
                    pred[j] = i

    return dist, pred

files = ['matrix_100_1_3_1', 'matrix_100_1_3_2']
for filename in files:
    Adjacency_List, W, vertices = parse_input_file(filename)
    source = 1
    D, P = bellman_ford(Adjacency_List, W, vertices, source)
    out_file = filename + ".csv"

    with open(out_file, 'w', newline='') as outfile:
        writer = csv.writer(outfile)
        writer.writerow(['Vertex', 'Distance', 'Predecessor'])
        for key in D:
            writer.writerow([str(key), str(D[key]), str(P[key])])
