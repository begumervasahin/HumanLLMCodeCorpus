import sys
import tkinter as tk
from tkinter import Toplevel, Label, Button, Entry, BOTTOM, TOP, RIGHT, LEFT, Scrollbar, END, messagebox as tkMessageBox

# Global variables
num_nodes = 1
num_edges = 1
node_vector = []
creating_edge = False
edge_aux = None
canvas = None
text_panel = None

def initialize(graph, source):
    distances = {}
    predecessors = {}
    
    for node in graph:
        distances[node] = float('inf')
        predecessors[node] = None
    
    distances[source] = 0
    return distances, predecessors

def relax(node, neighbor, graph, distances, predecessors):
    if distances[neighbor] > distances[node] + graph[node][neighbor]:
        distances[neighbor] = distances[node] + graph[node][neighbor]
        predecessors[neighbor] = node

def bellman_ford(graph, source):
    global text_panel
    iteration = 0
    distances, predecessors = initialize(graph, source)
    
    for _ in range(len(graph) - 1):
        for node in graph:
            iteration += 1
            for neighbor in graph[node]:
                relax(node, neighbor, graph, distances, predecessors)
            
            text_panel.insert(END, f'Iteration {iteration} \n')
            text_panel.insert(END, 'Distance Array \n')
            text_panel.insert(END, f'{distances} \n')
            text_panel.insert(END, 'Predecessor Array \n')
            text_panel.insert(END, f'{predecessors} \n')
    
    for node in graph:
        for neighbor in graph[node]:
            assert distances[neighbor] <= distances[node] + graph[node][neighbor]
    
    return distances, predecessors

def main():
    graph = {
        'a': {'b': -1, 'c': 4},
        'b': {'c': 3, 'd': 2, 'e': 2},
        'c': {},
        'd': {'b': 1, 'c': 5},
        'e': {'d': -3}
    }
    
    distances, predecessors = bellman_ford(graph, 'a')
    print(distances)
    print(predecessors)
    create_main_window()

def create_main_window():
    global canvas, text_panel
    root = tk.Tk()
    root.title("Bellman-Ford Graphs")
    
    bottom_frame = tk.Frame(root)
    bottom_frame.pack(side=BOTTOM)
    
    top_frame = tk.Frame(root)
    top_frame.pack(side=TOP)
    
    w = 790
    h = 600
    ws = root.winfo_screenwidth()
    hs = root.winfo_screenheight()
    x = (ws / 2) - (w / 2)
    y = (hs / 2) - (h / 2)
    root.geometry('%dx%d+%d+%d' % (w, h, x, y))
    
    create_top_panel(top_frame, w, h)
    
    arial14 = ('Arial', 14, "bold")
    circle_button = tk.Button(bottom_frame, text="Add Node", command=lambda: create_node(canvas), font=arial14)
    evaluate_button = tk.Button(bottom_frame, text="Evaluate Graph", font=arial14, command=evaluate_graph)
    delete_button = tk.Button(bottom_frame, text="Delete All", font=arial14, command=reset_all)
    
    circle_button.pack(side=tk.LEFT)
    evaluate_button.pack(side=tk.LEFT)
    delete_button.pack(side=tk.LEFT)
    
    root.mainloop()
    sys.exit(0)

def create_top_panel(top_frame, w, h):
    global canvas, text_panel
    canvas = tk.Canvas(top_frame, width=585, height=h - 50, borderwidth=0, highlightthickness=0, bg="white")
    text_panel = tk.Text(top_frame, height=300, width=50)
    scroll = Scrollbar(top_frame)
    
    canvas.pack(side=tk.LEFT)
    scroll.pack(side=RIGHT, fill=tk.Y)
    text_panel.pack(side=tk.LEFT)
    
    scroll.config(command=text_panel.yview)
    text_panel.config(yscrollcommand=scroll.set)

def reset_all():
    global canvas, num_nodes, num_edges, creating_edge, edge_aux, node_vector
    canvas.delete("all")
    num_nodes = 1
    num_edges = 1
    node_vector = []
    creating_edge = False
    edge_aux = None

def evaluate_graph():
    global node_vector, text_panel
    graph = {}
    aux_edge = {}
    
    for node in node_vector:
        for edge in node.edges:
            aux_edge[edge.node2.id] = int(edge.weight)
        
        graph[node.id] = aux_edge
        aux_edge = {}
    
    distances, predecessors = bellman_ford(graph, node_vector[0].id)
    
    text_panel.insert(END, f'{distances} \n')
    text_panel.insert(END, f'{predecessors} \n')

def create_node(canvas):
    global node_vector
    node = Node(canvas, 100, 120, 30)
    node_vector.append(node)

class Node:
    def __init__(self, canvas, x, y, r):
        self.edges = []
        self.canvas = canvas
        self.x = x
        self.y = y
        self.r = r
        self.moving = True
        self.draw_node()
        self.create_popup_menu()

    def move_circle(self, event):
        if self.moving:
            pos_x = event.x - self.x
            pos_y = event.y - self.y
            self.x = event.x
            self.y = event.y
            self.canvas.move(self.circle, pos_x, pos_y)
            self.canvas.move(self.canvas_num, pos_x, pos_y)

    def draw_node(self):
        global num_nodes
        self.id = num_nodes
        self.circle = self.canvas.create_oval(self.x - self.r, self.y - self.r, self.x + self.r, self.y + self.r,
                                              fill="blue")
        self.canvas_num = self.canvas.create_text(self.x, self.y, fill="white", text=f'{self.id}')
        num_nodes += 1
        self.canvas.tag_bind(self.circle, '<B1-Motion>', self.move_circle)
        self.canvas.tag_bind(self.canvas_num, '<B1-Motion>', self.move_circle)
        self.canvas.tag_bind(self.circle, '<Button-3>', self.do_popup)
        self.canvas.tag_bind(self.canvas_num, '<Button-3>', self.do_popup)

    def create_popup_menu(self):
        self.popup_menu = tk.Menu(self.canvas, tearoff=0)
        self.popup_menu.add_command(label="Fix Node", command=self.popup_fix_node)
        self.popup_menu.add_command(label="Move Node", command=self.popup_move_node)

    def popup_fix_node(self):
        self.moving = False

    def popup_move_node(self):
        self.moving = True

    def do_popup(self, event):
        try:
            self.popup_menu.tk_popup(event.x_root + 55, event.y_root + 10, 0)
        finally:
            self.popup_menu.grab_release()
