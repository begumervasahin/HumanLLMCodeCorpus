import socket
import time
from threading import Thread
from collections import defaultdict

class Server(Thread):
    def __init__(self):
        super().__init__()

    def run(self):
        # Load graph data from file
        graph = self.load_graph_data('test')

        # Initialize distance dictionary
        d = self.initialize_distance(graph)

        # Set up server socket
        server_socket = self.setup_server_socket()

        # Accept connections and send data
        self.send_data_to_clients(server_socket, graph, d)

    def load_graph_data(self, filename):
        graph = defaultdict(dict)
        with open(filename) as file:
            lines = file.readlines()
            for i in range(1, 1 + int(lines[0])):
                elements = lines[i].split()
                if i != 0:
                    k = 1
                    for _ in range(int(elements[0])):
                        graph[i][int(elements[k])] = int(elements[k + 1])
                        k += 2
        return graph

    def initialize_distance(self, graph):
        d = defaultdict(dict)
        for node in graph:
            for other_node in graph:
                if node != other_node:
                    d[node][other_node] = float('inf')
                else:
                    d[node][other_node] = 0
        return d

    def setup_server_socket(self):
        port = 60000
        server_socket = socket.socket()
        host = ""
        server_socket.bind((host, port))
        server_socket.listen(5)
        return server_socket

    def send_data_to_clients(self, server_socket, graph, d):
        for _ in range(len(graph) - 1):
            print("Server listening....")
            conn, addr = server_socket.accept()
            time.sleep(0.1)
            for u in graph:
                print("Sending data for node", u)
                to_send = self.construct_message(u, graph, d)
                conn.send(to_send.encode())
            print("Data sent. Closing connection.")
            conn.close()

        print("Out of loop")
        time.sleep(0.1)
        conn, addr = server_socket.accept()
        conn.send("Bye".encode())
        print("Bye sent")
        conn.close()

    def construct_message(self, u, graph, d):
        to_send = f"{u} "
        for v in graph:
            to_send += f"{v} {d[u][v]}\n"
        return to_send

class Receiver(Thread):
    def __init__(self):
        super().__init__()

    def run(self):
        while True:
            s = socket.socket()
            host = ""
            port = 60000
            s.connect((host, port))
            print("Waiting to receive...")
            data = s.recv(1024).decode()
            print("Received data:", data)
            if data == 'Bye':
                break
            s.close()

if __name__ == '__main__':
    rec = Receiver()
    ser = Server()
    ser.start()
    time.sleep(3)
    rec.start()
    ser.join()
    rec.join()
