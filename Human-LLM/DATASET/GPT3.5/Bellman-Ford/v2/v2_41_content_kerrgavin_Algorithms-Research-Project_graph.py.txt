class Graph:
    """Class representing a graph."""
    
    def __init__(self, V=None, E=None, directed=False):
        self.V = V if V is not None else []  # List of vertices
        self.E = E if E is not None else []  # List of edges
        self.directed = directed  # Flag indicating whether the graph is directed
        self.adj = {}  # Adjacency list for storing edges
    
    def get_adjacent_vertices(self, v):
        """Get the adjacent vertices of a given vertex."""
        return self.adj.get(v.value, [])
    
    def add_vertex(self, value):
        """Add a new vertex to the graph."""
        vertex = Vertex(value=value)
        self.V.append(vertex)
        self.adj[vertex.value] = []
    
    def add_edge(self, u, v, weight):
        """Add a new edge to the graph."""
        edge = Edge(u, v, weight)
        self.adj[u.value].append(edge)
        self.E.append(edge)
        
        # If the graph is undirected, add the reverse edge
        if not self.directed:
            reverse_edge = Edge(v, u, weight)
            if v.value not in self.adj:
                self.adj[v.value] = []
            self.adj[v.value].append(reverse_edge)

class Vertex:
    """Class representing a vertex."""
    
    def __init__(self, value=None, d=None, pre=None):
        self.value = value  # Value of the vertex
        self.d = d  # Property used in graph algorithms (e.g., distance)
        self.pre = pre  # Property used in graph algorithms (e.g., predecessor)

class Edge:
    """Class representing an edge."""
    
    def __init__(self, u=None, v=None, weight=None):
        self.u = u  # Source vertex
        self.v = v  # Target vertex
        self.weight = weight  # Weight of the edge

    def equals(self, other):
        """Check if two edges are equal."""
        return self.u == other.u and self.v == other.v

# Example usage:
if __name__ == "__main__":
    # Create a new graph
    graph = Graph(directed=True)

    # Add vertices
    graph.add_vertex(1)
    graph.add_vertex(2)

    # Add edges
    graph.add_edge(Vertex(1), Vertex(2), 5)
    graph.add_edge(Vertex(2), Vertex(1), 3)
