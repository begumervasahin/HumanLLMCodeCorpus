import random

class Action:
    def __init__(self, action_key, result, result_chances=1.0, create_counter_table=False):
        '''
        Initialize an Action object.

        Args:
            action_key (object): A key to identify the action.
            result (object or list): A single object or a list of possible results from the action.
            result_chances (float or list, optional): Default is 1.0 (100% chance). 
                Specifies the probability of each result if the action is successful. 
                If `result` is a list, `result_chances` should be a list of equal size.
            create_counter_table (bool, optional): Default is False. 
                Determines if a counter table should be created to track result occurrences.

        Raises:
            ValueError: If the total number of result chances doesn't match the total number of results.

        Postconditions:
            An Action object is created.
        '''
        self.action_key = action_key

        if isinstance(result, list):
            self.results = result
            self.num_results = len(result)
        else:
            self.results = [result]
            self.num_results = 1

        if isinstance(result_chances, list):
            if len(result_chances) == self.num_results:
                self.result_chances = result_chances
            else:
                raise ValueError("Total number of result chances must match total number of results.")
        else:
            self.result_chances = [result_chances] * self.num_results

        self.create_counter_table = create_counter_table

        if create_counter_table:
            self.counter_table = {res.getKey(): 0 for res in self.results}

    def testAction(self):
        '''
        Perform the action and return a result based on probabilities.

        Returns:
            object or -1: The result of the action or -1 if the action failed.
        '''
        rand_num = random.randint(1, 100000) / 100000
        prev_value = 0

        for i in range(self.num_results):
            curr_value = prev_value + self.result_chances[i]

            if prev_value < rand_num < curr_value:
                if self.create_counter_table:
                    self.counter_table[self.results[i].getKey()] += 1
                return self.results[i]

            prev_value = curr_value

        return -1

    def getKey(self):
        '''Return the key associated with this action.'''
        return self.action_key

    def getCount(self):
        '''
        Return the counter table tracking result occurrences.

        Raises:
            ValueError: If counting is not implemented for this action.
        '''
        if self.create_counter_table:
            return self.counter_table
        else:
            raise ValueError("Counting is not implemented for this action.")

    def getNumResults(self):
        '''Return the number of possible results from this action.'''
        return self.num_results

# Example usage of the Action class
if __name__ == '__main__':
    # Create an Action instance
    action_key = 'example_action'
    results = ['result1', 'result2', 'result3']
    result_chances = [0.3, 0.5, 0.2]
    create_counter_table = True

    action_instance = Action(action_key, results, result_chances, create_counter_table)

    # Test the action
    print("Testing the action...")
    result = action_instance.testAction()
    print("Result of the action:", result)

    # Get the counting table
    try:
        count_table = action_instance.getCount()
        print("Counting table:", count_table)
    except ValueError as e:
        print("Error:", e)

    # Get the number of possible results
    num_results = action_instance.getNumResults()
    print("Number of possible results:", num_results)
