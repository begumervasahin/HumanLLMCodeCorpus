def xor_operation(binary1, binary2):
    """
    Perform bitwise XOR operation between two binary strings.
    
    Parameters:
    - binary1: The first binary string.
    - binary2: The second binary string.
    
    Returns:
    - A binary string that is the result of the XOR operation.
    """
    # Initialize an empty list to store the XOR result
    xor_result = []
    
    # Loop over each bit in the binary strings
    for bit_index in range(len(binary1)):
        # Compare bits at the same position; if they are the same, append '0', otherwise '1'
        if binary1[bit_index] == binary2[bit_index]:
            xor_result.append('0')
        else:
            xor_result.append('1')
    
    # Join the list of bits back into a string
    return ''.join(xor_result)

def binary_division(dividend, divisor):
    """
    Perform binary division (mod 2) and find the remainder.
    
    Parameters:
    - dividend: The binary number to be divided.
    - divisor: The binary number to divide by.
    
    Prints:
    - The quotient and remainder of the division.
    """
    # Length of the divisor to determine how many bits to consider at a time
    pick = len(divisor)
    
    # Take the initial segment from the dividend
    temp = dividend[:pick]
    
    while pick < len(dividend):
        if temp[0] == '1':
            # If the leading bit is '1', use the divisor for XOR
            temp = xor_operation(divisor, temp) + dividend[pick]
        else:
            # If the leading bit is '0', use a string of zeros for XOR
            temp = xor_operation('0'*pick, temp) + dividend[pick]
        
        # Move to the next bit
        pick += 1

    # Perform the final XOR operation for the remaining bits
    if temp[0] == '1':
        temp = xor_operation(divisor, temp)
    else:
        temp = xor_operation('0'*pick, temp)

    # Print the results
    print("Remainder is: ", temp)
