class Vertex:
    def __init__(self, id):
        self.id = id
        self.adjacent_vertices = {}  # Dictionary to store adjacent vertices and their weights

    def add_adjacent_vertex(self, adjacent_vertex, weight=0):
        """Adds an adjacent vertex with the given weight."""
        self.adjacent_vertices[adjacent_vertex] = weight

class Graph:
    def __init__(self, directed=False):
        self.vertices = {}  # Dictionary to store vertices
        self.directed = directed

    def add_vertex(self, id):
        """Adds a vertex with the given ID to the graph."""
        vertex = Vertex(id)
        self.vertices[id] = vertex
        return vertex

    def add_edge(self, from_vertex_id, to_vertex_id, weight=0):
        """Adds an edge from 'from_vertex_id' to 'to_vertex_id' with the given weight."""     
        if from_vertex_id not in self.vertices:
            self.add_vertex(from_vertex_id)
        if to_vertex_id not in self.vertices:
            self.add_vertex(to_vertex_id)
        
        self.vertices[from_vertex_id].add_adjacent_vertex(self.vertices[to_vertex_id], weight)
        if not self.directed:
            self.vertices[to_vertex_id].add_adjacent_vertex(self.vertices[from_vertex_id], weight)      

    def get_vertices(self):
        """Returns all vertices in the graph."""
        return list(self.vertices.values())

    def get_vertex(self, id):
        """Returns the vertex with the given ID."""
        return self.vertices.get(id)

    def get_edges(self):
        """Returns all edges in the graph."""
        edges = set()
        for vertex in self.vertices.values():
            for adjacent_vertex, weight in vertex.adjacent_vertices.items():
                edges.add((vertex, adjacent_vertex, weight))
        return edges

    def __iter__(self):
        """Allows iteration over the vertices in the graph."""
        return iter(self.vertices.values())
