import socket

# Create a UDP socket
sendsock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)

# Set socket option to allow reusing address
sendsock.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)

# Bind the socket to a specific address and port (localhost, port 8179)
sendsock.bind(("localhost", 8179))

# Send a message "Hello" to localhost on port 8181
sendsock.sendto(b"Hello", ('localhost', 8181))

# Receive a response from the server (up to 4096 bytes)
data, addr = sendsock.recvfrom(4096)

# Print the received data after decoding it
print(data.decode())

# Close the socket
sendsock.close()
