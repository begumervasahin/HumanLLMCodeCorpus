class FibonacciHeap:
    # Inner class to represent a node in the heap
    class Node:
        def __init__(self, key, value):
            # Node attributes
            self.key = key  # Key of the node
            self.value = value  # Value associated with the key
            self.parent = self.child = self.left = self.right = None  # References to related nodes
            self.degree = 0  # Degree of the node (number of children)
            self.mark = False  # Mark indicating whether the node has lost a child in the past

    # FibonacciHeap constructor
    def __init__(self):
        # Initialize root list and minimum node
        self.root_list = None  # Circular doubly linked list of roots
        self.min_node = None  # Pointer to the node with the minimum key
        self.total_nodes = 0  # Total number of nodes in the heap

    # Find the node with the minimum key
    def find_min(self):
        return self.min_node

    # Extract the node with the minimum key from the heap
    def extract_min(self):
        min_node = self.min_node
        if min_node is not None:
            # If the minimum node has children, merge them with the root list
            if min_node.child is not None:
                children = [x for x in self.iterate(min_node.child)]
                for child in children:
                    self.merge_with_root_list(child)
                    child.parent = None
            # Remove the minimum node from the root list
            self.remove_from_root_list(min_node)
            if min_node == min_node.right:
                self.min_node = self.root_list = None
            else:
                self.min_node = min_node.right
                self.consolidate()
            self.total_nodes -= 1
        return min_node

    # Insert a new node with the given key and value into the heap
    def insert(self, key, value=None):
        # Create a new node
        new_node = self.Node(key, value)
        # Initialize node's left and right pointers
        new_node.left = new_node.right = new_node
        # Merge the new node with the root list
        self.merge_with_root_list(new_node)
        # Update the minimum node if necessary
        if self.min_node is None or new_node.key <= self.min_node.key:
            self.min_node = new_node
        self.total_nodes += 1
        return new_node

    # Decrease the key of a node in the heap
    def decrease_key(self, node, new_key):
        if new_key > node.key:
            return None
        node.key = new_key
        parent = node.parent
        if parent is not None and node.key <= parent.key:
            self.cut(node, parent)
            self.cascading_cut(parent)
        if node.key <= self.min_node.key:
            self.min_node = node

    # Merge this heap with another heap
    def merge(self, other_heap):
        merged_heap = FibonacciHeap()
        # Copy root list and minimum node from both heaps
        merged_heap.root_list = self.root_list
        merged_heap.min_node = self.min_node
        # Merge the root lists of both heaps
        last = other_heap.root_list.left
        other_heap.root_list.left = merged_heap.root_list.left
        merged_heap.root_list.left.right = other_heap.root_list
        merged_heap.root_list.left = last
        merged_heap.root_list.left.right = merged_heap.root_list
        # Update the minimum node if necessary
        if other_heap.min_node.key <= merged_heap.min_node.key:
            merged_heap.min_node = other_heap.min_node
        # Update the total number of nodes
        merged_heap.total_nodes = self.total_nodes + other_heap.total_nodes
        return merged_heap

    # Helper function to perform a cut operation on a node
    def cut(self, child, parent):
        # Remove child from the child list of parent
        self.remove_from_child_list(parent, child)
        parent.degree -= 1
        # Merge child with the root list
        self.merge_with_root_list(child)
        # Reset parent pointer of child
        child.parent = None
        child.mark = False

    # Helper function to perform a cascading cut operation
    def cascading_cut(self, node):
        parent = node.parent
        if parent is not None:
            if node.mark is False:
                node.mark = True
            else:
                self.cut(node, parent)
                self.cascading_cut(parent)

    # Helper function to consolidate trees with the same degree
    def consolidate(self):
        # Initialize an array to store roots of trees with distinct degrees
        degree_table = [None] * self.total_nodes
        # Iterate over the root list and consolidate trees with the same degree
        nodes = [w for w in self.iterate(self.root_list)]
        for node in nodes:
            x = node
            d = x.degree
            while degree_table[d] is not None:
                y = degree_table[d]
                if x.key > y.key:
                    x, y = y, x
                self.heap_link(y, x)
                degree_table[d] = None
                d += 1
            degree_table[d] = x
        # Update the minimum node after consolidation
        for i in range(len(degree_table)):
            if degree_table[i] is not None:
                if degree_table[i].key <= self.min_node.key:
                    self.min_node = degree_table[i]

    # Helper function to perform a heap link operation
    def heap_link(self, child, parent):
        # Remove child from the root list
        self.remove_from_root_list(child)
        # Reset child's left and right pointers
        child.left = child.right = child
        # Merge child with the child list of parent
        self.merge_with_child_list(parent, child)
        parent.degree += 1
        # Update parent pointer of child and reset mark
        child.parent = parent
        child.mark = False

    # Helper function to merge a node with the root list
    def merge_with_root_list(self, node):
        if self.root_list is None:
            self.root_list = node
        else:
            node.right = self.root_list.right
            node.left = self.root_list
            self.root_list.right.left = node
            self.root_list.right = node

    # Helper function to merge a node with the child list of a parent
    def merge_with_child_list(self, parent, node):
        if parent.child is None:
            parent.child = node
        else:
            node.right = parent.child.right
            node.left = parent.child
            parent.child.right.left = node
            parent.child.right = node

    # Helper function to remove a node from the root list
    def remove_from_root_list(self, node):
        if node == self.root_list:
            self.root_list = node.right
        node.left.right = node.right
        node.right.left = node.left

    # Helper function to remove a node from the child list of a parent
    def remove_from_child_list(self, parent, node):
        if parent.child == parent.child.right:
            parent.child = None
        elif parent.child == node:
            parent.child = node.right
            node.right.parent = parent
        node.left.right = node.right
        node.right.left = node.left

    # Helper function to iterate over a circular doubly linked list
    def iterate(self, head):
        node = stop = head
        flag = False
        while True:
            if node == stop and flag is True:
                break
            elif node == stop:
                flag = True
            yield node
            node = node.right


# Example Usage:

# Initialize Fibonacci Heap
fib_heap = FibonacciHeap()

# Insert nodes
node1 = fib_heap.insert(5, "Node 1")
node2 = fib_heap.insert(3, "Node 2")
node3 = fib_heap.insert(8, "Node 3")

# Find minimum
min_node = fib_heap.find_min()
print("Minimum node:", min_node.key, min_node.value)

# Extract minimum
min_node = fib_heap.extract_min()
print("Extracted minimum node:", min_node.key, min_node.value)

# Decrease key
fib_heap.decrease_key(node3, 4)

# Merge heaps
new_heap = FibonacciHeap()
new_heap.insert(10)
merged_heap = fib_heap.merge(new_heap)
