class Vertex:
    def __init__(self, id):
        self.id = id
        self.distance = float('inf')  # Distance from the source vertex
        self.visited = False  # Whether the vertex has been visited

    def get_distance(self):
        """Returns the distance of this vertex from the source."""
        return self.distance

    def set_distance(self, distance):
        """Sets the distance of this vertex from the source."""
        self.distance = distance

    def visit(self):
        """Marks the vertex as visited."""
        self.visited = True

    def is_visited(self):
        """Checks if the vertex has been visited."""
        return self.visited

class Graph:
    def __init__(self):
        self.vertices = {}  # Dictionary to store vertices
        self.edges = []  # List to store edges

    def add_vertex(self, vertex_id):
        """Adds a vertex with the given ID to the graph."""
        if vertex_id not in self.vertices:
            self.vertices[vertex_id] = Vertex(vertex_id)

    def add_edge(self, source, destination, weight):
        """Adds an edge from source to destination with the given weight."""
        self.add_vertex(source)
        self.add_vertex(destination)
        self.edges.append((source, destination, weight))

    def get_vertices(self):
        """Returns all vertices in the graph."""
        return self.vertices.values()

    def get_edges(self):
        """Returns all edges in the graph, with vertex objects instead of IDs."""
        return [(self.vertices[u], self.vertices[v], weight) for u, v, weight in self.edges]

def initialize_single_source(graph, source):
    """Initializes the graph for the Bellman-Ford algorithm."""
    for vertex in graph.get_vertices():
        vertex.set_distance(float('inf'))
        vertex.visited = False
    graph.get_vertex(source).set_distance(0)

def relax(u, v, weight):
    """Relaxes the edge (u, v) with the given weight."""
    if v.get_distance() > u.get_distance() + weight:
        v.set_distance(u.get_distance() + weight)

def bellman_ford(graph, source):
    """Implements the Bellman-Ford algorithm."""
    initialize_single_source(graph, source)
    for _ in range(len(graph.get_vertices()) - 1):
        for u, v, weight in graph.get_edges():
            if not v.is_visited():
                relax(u, v, weight)
    for u, v, weight in graph.get_edges():
        if v.get_distance() > u.get_distance() + weight:
            return False  # Negative-weight cycle found
    return True  # No negative-weight cycles

# Example usage:
graph = Graph()
graph.add_edge('A', 'B', -1)
graph.add_edge('A', 'C', 4)
graph.add_edge('B', 'C', 3)
graph.add_edge('B', 'D', 2)
graph.add_edge('B', 'E', 2)
graph.add_edge('D', 'C', 5)
graph.add_edge('D', 'B', 1)
graph.add_edge('E', 'D', -3)

print("Graph contains negative-weight cycle:", not bellman_ford(graph, 'A'))
