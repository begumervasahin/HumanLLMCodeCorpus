import numpy as np
from collections import defaultdict

class Node:
    def __init__(self, vertex_name, dist=float('inf'), trace=None):
        self.vertex_name = vertex_name
        self.dist = dist
        self.trace = trace

class Graph:
    def __init__(self, vertices=[], edges=[], minipath=[], numVer=0, numEdge=0):
        self.vertices = vertices
        self.edges = edges
        self.minipath = minipath
        self.numVer = numVer
        self.numEdge = numEdge

    def add_edge(self, src, dest, weight):
        self.edges.append([src, dest, weight])

    def bellman_ford(self, src, dest):
        for vertex in self.vertices:
            vertex.dist = float("inf")
        self.vertices[src - 1].dist = 0

        for _ in range(self.numVer - 1):
            for u, v, w in self.edges:
                if self.vertices[v - 1].dist > self.vertices[u - 1].dist + w:
                    self.vertices[v - 1].dist = self.vertices[u - 1].dist + w
                    self.vertices[v - 1].trace = u

        if self.vertices[dest - 1].dist == float('inf'):
            return

        self.minipath.append(dest)
        while src not in self.minipath:
            self.minipath.append(self.vertices[self.minipath[-1] - 1].trace)

graph = Graph()
print("\nRead input from file")
with open('minpath_in.txt', 'r') as file:
    line = file.readline()
    coords = line.split()
    graph.numVer = int(coords[0])
    graph.numEdge = int(coords[1])
    start_vertex = int(coords[2])
    end_vertex = int(coords[3])

    line = file.readline()
    while line:
        coords = line.split()
        graph.add_edge(int(coords[0]), int(coords[1]), int(coords[2]))
        line = file.readline()

for i in range(1, graph.numVer + 1):
    graph.vertices.append(Node(i))
graph.bellman_ford(start_vertex, end_vertex)

print("\nWrite output into file")
with open('minpath_out.txt', 'w') as file:
    if start_vertex not in graph.minipath:
        file.write("\nThere are no paths from %d to %d" % (graph.vertices[start_vertex - 1].vertex_name, 
                                                           graph.vertices[end_vertex - 1].vertex_name))
    else:
        file.write("Distance from %d to %d is: %d\n" % (graph.vertices[start_vertex - 1].vertex_name, 
                                                         graph.vertices[end_vertex - 1].vertex_name, 
                                                         graph.vertices[end_vertex - 1].dist))
        file.write('%d' % graph.vertices[end_vertex - 1].vertex_name)
        for v in graph.minipath:
            if graph.vertices[v - 1].trace is not None:
                file.write(" <-- %d" % graph.vertices[v - 1].trace)
