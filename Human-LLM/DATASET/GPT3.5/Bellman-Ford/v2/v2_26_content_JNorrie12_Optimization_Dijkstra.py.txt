import sys
import time
import csv
import numpy as np

def dijkstra(ist, isp, wei):
    if ist == isp:
        return [ist]

    n = len(wei)
    inf = sys.maxsize
    unvisited = np.ones(n, dtype=int)
    cost = np.ones(n) * 1.e6
    parent = -np.ones(n, dtype=int) * inf
    current_node = ist
    cost[current_node] = 0
    unvisited[current_node] = 0
    tmp = unvisited * wei[current_node, :]
    neighbors = np.array(tmp.nonzero()).flatten()
    remaining_nodes = np.array(unvisited.nonzero()).flatten().size

    while remaining_nodes != 0:
        for neighbor in neighbors:
            new_cost = cost[current_node] + wei[current_node, neighbor]
            if new_cost < cost[neighbor]:
                cost[neighbor] = new_cost
                parent[neighbor] = current_node

        unvisited_nodes = np.array(unvisited.nonzero()).flatten()
        min_cost, min_index = cost[unvisited_nodes].min(), cost[unvisited_nodes].argmin()
        current_node = unvisited_nodes[min_index]
        unvisited[current_node] = 0
        tmp = unvisited * wei[current_node, :]
        neighbors = np.array(tmp.nonzero()).flatten()
        remaining_nodes = np.array(unvisited.nonzero()).flatten().size

    shortest_path = [isp]
    while parent[isp] != ist:
        shortest_path.append(parent[isp])
        isp = parent[isp]
    shortest_path.append(ist)
    return shortest_path[::-1]

def calculate_weights(RX, RY, RA, RB, RV):
    n = len(RX)
    weights = np.zeros((n, n), dtype=float)
    m = len(RA)

    for i in range(m):
        xa, ya = RX[RA[i] - 1], RY[RA[i] - 1]
        xb, yb = RX[RB[i] - 1], RY[RB[i] - 1]
        distance = np.sqrt((xb - xa) ** 2 + (yb - ya) ** 2)
        time_taken = distance / RV[i]
        weights[RA[i] - 1, RB[i] - 1] = time_taken

    return weights

if __name__ == '__main__':
    RomeX, RomeY = np.empty(0, dtype=float), np.empty(0, dtype=float)

    with open('RomeVertices', 'r') as file:
        reader = csv.reader(file)
        for row in reader:
            RomeX = np.concatenate((RomeX, [float(row[1])]))
            RomeY = np.concatenate((RomeY, [float(row[2])])))

    RomeA, RomeB, RomeV = np.empty(0, dtype=int), np.empty(0, dtype=int), np.empty(0, dtype=float)

    with open('RomeEdges', 'r') as file:
        reader = csv.reader(file)
        for row in reader:
            RomeA = np.concatenate((RomeA, [int(row[0])]))
            RomeB = np.concatenate((RomeB, [int(row[1])]))
            RomeV = np.concatenate((RomeV, [float(row[2])])))

    weights = calculate_weights(RomeX, RomeY, RomeA, RomeB, RomeV)
    ist, isp = 12, 51

    t0 = time.time()
    shortest_path = dijkstra(ist, isp, weights)
    t1 = time.time()

    print('Dijkstra:     ', ist + 1, ' -> ', isp + 1, ' is ', np.array(shortest_path) + 1, t1 - t0)
