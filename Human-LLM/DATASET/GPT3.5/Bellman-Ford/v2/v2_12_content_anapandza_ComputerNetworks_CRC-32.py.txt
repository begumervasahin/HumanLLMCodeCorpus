import codecs

def create_crc32_table():
    """
    Generates a table for the CRC-32 algorithm.

    Returns:
        A list of 256 integers, each representing the CRC-32 result for a single byte.
    """
    table = []
    for byte in range(256):
        crc = byte
        for _ in range(8):
            if crc & 1:
                crc = (crc >> 1) ^ 0xEDB88320
            else:
                crc >>= 1
        table.append(crc)
    return table

def calculate_crc32(data, crc_table):
    """
    Calculates the CRC-32 checksum for a given set of data.

    Args:
        data: The data to calculate the checksum for, as a bytes object.
        crc_table: The pre-generated CRC-32 table.

    Returns:
        The CRC-32 checksum, as an integer.
    """
    crc = 0xFFFFFFFF
    for byte in data:
        table_index = (crc ^ byte) & 0xFF
        crc = crc_table[table_index] ^ (crc >> 8)
    return ~crc & 0xFFFFFFFF

def main():
    # Prepare the CRC-32 lookup table.
    crc_table = create_crc32_table()

    # Example message in hexadecimal format.
    hex_message = 'FAFAFA'
    # Convert hex message to bytes for processing.
    byte_message = codecs.decode(hex_message, 'hex')

    # Calculate CRC-32 checksum.
    checksum = calculate_crc32(byte_message, crc_table)

    # Output the original message and its checksum in both binary and hexadecimal formats.
    print(f"Original message in hex: {hex_message}")
    print(f"Checksum in hex: {hex(checksum)}")

    # To send the message, concatenate the original hex message with the checksum.
    full_message_hex = hex_message + hex(checksum)[2:]
    print(f"Full message to send (hex): {full_message_hex}")

if __name__ == "__main__":
    main()
