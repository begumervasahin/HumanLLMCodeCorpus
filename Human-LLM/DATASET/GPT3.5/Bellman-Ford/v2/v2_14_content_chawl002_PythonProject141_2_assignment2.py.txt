import os
import re
import sys
import time

graph_pattern = re.compile(r"(\d+)\s(\d+)")
edge_pattern = re.compile(r"(\d+)\s(\d+)\s(\d+)")

def bellman_ford(graph):
    paths = []
    for source in range(len(graph[0])):
        distances = [float("inf")] * len(graph[0])
        distances[source] = 0
        for _ in range(1, len(graph[0])-1):
            for from_node in range(len(graph[0])):
                for to_node in range(len(graph[0])):
                    weight = graph[1][from_node][to_node]
                    if float(weight) < float("inf"):
                        if distances[to_node] > distances[from_node] + float(weight):
                            distances[to_node] = int(distances[from_node]) + int(float(weight))
        paths.append(distances)
    return paths

def floyd_warshall(graph):
    paths = []
    distances = [[float("inf") for _ in range(len(graph[0]))] for _ in range(len(graph[0]))]
    for i in range(len(graph[0])):
        distances[i][i] = 0
        for j in range(len(graph[0])):
            if i != j and float(graph[1][i][j]) < float("inf"):
                distances[i][j] = int(float(graph[1][i][j]))
    for k in range(len(graph[0])):
        for i in range(len(graph[0])):
            for j in range(len(graph[0])):
                if distances[i][j] > distances[i][k] + distances[k][j]:
                    distances[i][j] = distances[i][k] + distances[k][j]
    paths.append(distances)
    return paths

def read_graph(filename):
    with open(filename, 'r') as infile:
        first_line = infile.readline()
        graph_match = graph_pattern.match(first_line)
        if not graph_match:
            print(first_line + " not properly formatted")
            sys.exit(1)
        vertices = list(range(int(graph_match.group(1))))
        edges = [[float("inf") for _ in range(len(vertices))] for _ in range(len(vertices))]
        for line in infile.readlines():
            line = line.strip()
            edge_match = edge_pattern.match(line)
            if edge_match:
                source = int(edge_match.group(1))
                sink = int(edge_match.group(2))
                if source >= len(vertices) or sink >= len(vertices):
                    print(f"Attempting to insert an edge between {source} and {sink} in a graph with {len(vertices)} vertices")
                    sys.exit(1)
                weight = edge_match.group(3)
                edges[source][sink] = weight
    return (vertices, edges)

def write_file(length_matrix, filename):
    base_name = os.path.splitext(os.path.split(filename)[1])[0]
    with open(f'output/{base_name}_output.txt', 'w+') as outfile:
        for vertex in length_matrix:
            for length in vertex:
                outfile.write(str(length) + ',')
            outfile.write('\n')

def main(filename, algorithm):
    algorithm = algorithm[1:]
    graph_data = read_graph(filename)
    path_lengths = []
    if algorithm == 'b' or algorithm == 'B':
        path_lengths = bellman_ford(graph_data)
    elif algorithm == 'f' or algorithm == 'F':
        path_lengths = floyd_warshall(graph_data)
    elif algorithm == "both":
        start_time = time.process_time()
        bellman_ford(graph_data)
        bellman_ford_time = time.process_time() - start_time
        floyd_warshall(graph_data)
        floyd_warshall_time = time.process_time() - start_time
        print(f"Bellman-Ford timing: {bellman_ford_time}")
        print(f"Floyd-Warshall timing: {floyd_warshall_time}")
    write_file(path_lengths, filename)

if __name__ == '__main__':
    if len(sys.argv) < 3:
        print("Usage: python bellman_ford.py -<f|b> <input_file>")
        sys.exit(1)
    main(sys.argv[2], sys.argv[1])
