def read_file(filename):
    # Construct the full file path
    filepath = "C:\\" + filename

    # Open the file in read mode using a context manager
    with open(filepath, 'r') as in_file:
        neighbors = {}  # Dictionary to store neighbor information
        counter = 0

        # Iterate through each line in the file
        for line in in_file:
            if counter == 0:
                # The first line contains the number of neighbors
                num_neighbors = int(line)
                counter += 1
            else:
                # Parse the line to extract neighbor, weight, and port information
                words = line.split(" ")
                neighbor = words[0]
                weight = float(words[1])
                port = int(words[2])

                # Store the parsed data in the neighbors dictionary
                neighbors[neighbor] = {'cost': weight, 'port': port}

    return neighbors

# Example usage
filename = "example.txt"
neighbors_data = read_file(filename)
