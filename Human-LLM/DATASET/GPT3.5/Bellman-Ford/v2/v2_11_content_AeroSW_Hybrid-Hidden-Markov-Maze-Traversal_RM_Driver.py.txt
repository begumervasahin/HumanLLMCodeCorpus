from Reinforcement_Maze import Reinforcement_Maze
from State_Controller import State_Controller
from State import State
from Action import Action
import sys
import time

def read_reinforcement_information(file_name):
    # Open the file and read necessary information
    with open(file_name, 'r') as my_file:
        num_actions = int(my_file.readline())
        num_states = int(my_file.readline())
        gamma = float(my_file.readline())

        s_controller = State_Controller(curr_state=None, state_list=None)
        file_list = []

        # Read state information
        for i in range(num_states):
            value = float(my_file.readline())
            terminality = int(my_file.readline())
            file_of_state = my_file.readline().rstrip()
            file_list.append(file_of_state)
            temp_state = State(value, terminality, i + 1, action_list=None, count_actions=True)
            added = s_controller.addState(temp_state)

        # Read action information for each state
        for i in range(len(file_list)):
            swapped = s_controller.switchState(i + 1)
            if s_controller.isTerminal():
                continue
            with open(file_list[i], 'r') as state_file:
                size = state_file.readline()
                sizes = size.split(' ')
                if int(sizes[0]) != num_actions or int(sizes[1]) != num_states:
                    raise MemoryError

                for count in range(num_actions):
                    line = state_file.readline()
                    myList = line.split(' ')
                    temp_dict = dict()

                    if len(myList) != num_states:
                        raise MemoryError

                    t_list = []
                    count_i = 0

                    for q in range(len(myList)):
                        s_controller.switchState(q + 1)
                        temp_state = s_controller.getState()
                        t_list.append(s_controller.getState())
                        myList[q] = float(myList[q])

                    s_controller.switchState(i + 1)
                    new_action = Action(count + 1, t_list, result_chances=myList, create_counter_table=True)
                    s_controller.addAction(new_action)

    return s_controller, gamma

def main(arguments):
    tol = 0.00001
    num_trials = 10000

    # Check command line arguments
    if len(arguments) == 4:
        tol = float(arguments[2])
        num_trials = int(arguments[3])
    elif len(arguments) != 2:
        raise IndexError

    # Read reinforcement information from file
    result = read_reinforcement_information(arguments[1])
    r_maze_1 = Reinforcement_Maze(result[1], result[0])
    r_maze_2 = Reinforcement_Maze(result[1], result[0])

    # Calculate averaged utilities
    c_averages = r_maze_1.countedAverages(num_trials)
    c_averages_2 = r_maze_2.countedAverages(num_trials)

    # Print averaged action percentages
    for i in c_averages:
        act_averages = c_averages[i]
        for j in act_averages:
            to_state_avg = act_averages[j]
            for k in to_state_avg:
                state_str = f'STATE {i:02}'
                action_str = f"'S ACTION {j}->{k}"
                percentage_str = f'\t\t{to_state_avg[k]}% of the time' if k < 10 or i < 10 else f'\t{to_state_avg[k]}% of the time'
                print(state_str + action_str + percentage_str)
        print('\n')

    print("\n\nU\n\n")

    # Calculate utilities and print policies and utilities
    init_time = time.time() * 1000
    t_2 = r_maze_1.createListUtilities(tol, c_averages)
    fin_time = time.time() * 1000
    time_1 = fin_time - init_time

    policies = t_2[1]
    new_u_dict = t_2[0]

    for i in new_u_dict:
        state_str = f'STATE {i:02}'
        print(state_str + f': POLICY = {policies[i]}:\t UTILITY = {new_u_dict[i]}')

    print("\n\nU+\n\n")

    # Calculate adjusted utilities and print policies and utilities
    init_time = time.time() * 1000
    t_2 = r_maze_2.createListUtilitiesAdjusted(tol, c_averages_2, 200)
    fin_time = time.time() * 1000
    time_2 = fin_time - init_time

    policies = t_2[1]
    new_u_dict = t_2[0]

    for i in new_u_dict:
        state_str = f'STATE {i:02}'
        print(state_str + f': POLICY = {policies[i]}:\t UTILITY = {new_u_dict[i]}')

    print('\n\n\n\n')
    print("U's Time:  " + str(time_1))
    print("U+'s Time: " + str(time_2))

if __name__ == '__main__':
    main(sys.argv)
