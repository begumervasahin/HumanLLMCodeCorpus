import time
import pickle
import json
import argparse
import graph
import variants

INPUT_PATH = "graphs\\"
OUTPUT_PATH = "data\\"

def read_graph_list(filename):
    with open(INPUT_PATH + filename, 'rb') as file:
        G = pickle.load(file)
    return G

def write_data(data, filename):
    with open(OUTPUT_PATH + filename, "w") as output:
        json.dump(data, output, indent=4)

def parse_arguments():
    parser = argparse.ArgumentParser()
    parser.add_argument('input_file', action='store', type=str, help='Input file name') 
    parser.add_argument('output_file', action='store', type=str, help='Output file name')
    return parser.parse_args()

def get_average_time(func, G):
    total_time = 0
    for _ in range(10):
        start_time = round(time.clock() * 10000)
        func(G, G.V[0])
        end_time = round(time.clock() * 10000)
        total_time += (end_time - start_time)
    return total_time / 10

def main():
    data = {"vertices": 0, "dijkstra": [], "bellmanford": [], "fibonacciheap": [], "yen": [], "edges": []}
    args = parse_arguments()
    print("Beginning to calculate run times...")
    with open(INPUT_PATH + args.input_file, "rb") as inputdata:
        index = 1
        while True:
            try:
                print("Loading graph data...")
                G = pickle.load(inputdata)
                print("Loading Complete.")
                data["vertices"] = len(G.V)
                data["edges"].append(len(G.E))
                print(f"Graph {index} out of {len(data['vertices']) - 1} info:")        
                print("\tDirected:", G.directed)
                print("\tVertex Count:", len(G.V))
                print("\tEdge Count:", len(G.E))
                print("Calculating average time for shortest path...")
                avg_time_dijkstra = get_average_time(variants.dijkstra, G)
                data["dijkstra"].append(avg_time_dijkstra)
                print("Dijkstra Run Time:", avg_time_dijkstra)
                avg_time_fib = get_average_time(variants.fibDijkstra, G)
                data["fibonacciheap"].append(avg_time_fib)
                print("Fibonacci Heap Run Time:", avg_time_fib)
                avg_time_bellman_ford = get_average_time(variants.bellmanFord, G)
                data["bellmanford"].append(avg_time_bellman_ford)
                print("Bellman-Ford Run Time:", avg_time_bellman_ford)
                avg_time_yen = get_average_time(variants.Yen, G)
                data["yen"].append(avg_time_yen)
                print("Yen Run Time:", avg_time_yen)
                G = None
                index += 1
            except EOFError:
                break
    print("Finished calculations.")
    print("Writing data to", args.output_file, "...")
    write_data(data, args.output_file)
    print("Data written.")

if __name__ == "__main__":
    main()
