import argparse
import os
import pickle
import random
from graph import Graph

PATH = "graphs\\"

def pickle_graph_list(graph, file_name):
    with open(file_name, 'ab') as file:
        pickle.dump(graph, file)

def generate_graph(num_vertices, max_weight, min_weight=1):
    graph = Graph(V=[], E=[], directed=True)
    for vertex_id in range(num_vertices):
        graph.addVertex(vertex_id)
    for i in range(len(graph.V)):
        u = graph.V[i]
        v = graph.V[(i + 1) % len(graph.V)]  # Create circular edge connection
        graph.addEdge(u, v, random.randint(min_weight, max_weight))
    return graph

def parse_command_line_arguments():
    parser = argparse.ArgumentParser()
    parser.add_argument('vertices', type=int, help='Number of vertices in the generated graph')
    parser.add_argument('max_weight', type=int, help='Maximum weight for a given edge') 
    parser.add_argument('--min_weight', dest='min_weight', type=int, default=1,
                        help='Minimum value for a given edge (default = 1)')
    parser.add_argument('file_name', type=str, help='File name')
    return parser.parse_args()

def main():
    args = parse_command_line_arguments()
    print(f"Generating data set '{args.file_name}'...")
    for edge_capacity in range(1, args.vertices + 1):
        print(f"Generating graph with edge capacity {edge_capacity}...")
        graph = generate_graph(args.vertices, args.max_weight, args.min_weight)
        print("Generation complete.")
        print("Saving to file...")
        file_path = os.path.join(PATH, args.file_name)
        pickle_graph_list(graph, file_path)
        print("File saved.")
    print("Data set generated and saved.")

if __name__ == "__main__":
    main()
