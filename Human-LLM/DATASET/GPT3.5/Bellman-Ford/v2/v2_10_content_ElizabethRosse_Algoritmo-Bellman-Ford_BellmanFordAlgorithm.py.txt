# Define the graph structure
vertices = {1: 0, 2: 1, 3: 2, 4: 3, 5: 4, 6: 5}
edges = [[1, 2], [1, 3], [1, 4], [3, 4], [4, 5], [4, 6], [5, 6]]
weights = [1, 2, 4, 3, 5, 2, 4]

# Initialize variables for the algorithm
inf = float("inf")
distance = [inf] * len(vertices)  # Set initial distances to infinity
previous = [0] * len(vertices)  # Track the previous vertex in the path
distance[0] = 0  # Distance from the source to itself is 0

# Main loop to update distances based on edges and weights
for v, i in vertices.items():
    print(f"\nIteration: {i}\nResult Table:")
    for e, edge in enumerate(edges):
        if edge[0] == v:  # If the edge starts from the current vertex
            adjacent = vertices[edge[1]]  # Get the index of the adjacent vertex
            weight = weights[e]  # Get the weight of the edge
            # If a shorter path to the adjacent vertex is found
            if distance[i] + weight < distance[adjacent]:
                # Update the distance and the path
                distance[adjacent] = distance[i] + weight
                previous[adjacent] = v
                # Print the current state of the algorithm
                print(f"Edge: {edge}\nVertices: {vertices}\nDistance(cost): {distance}\nPrevious: {previous}")

# Function to recursively print the shortest path
def print_path(destination):
    # Base case: if the current vertex is not the start
    if previous[vertices[destination]] != 0:
        # Recursively print the path to the previous vertex
        print_path(list(vertices.keys())[list(vertices.values()).index(previous[vertices[destination]])])
    # Print the current vertex in the path
    print(destination)

# Print the shortest path from the start vertex to vertex 6
print("List of vertices that make up the shortest path, with the least cost:")
print_path(6)
