# Define a function to extract the minimum value from a priority queue   
def popmin(pqueue):
    lowest = float('inf')
    keylowest = None
    for key in pqueue:
        if pqueue[key] < lowest:
            lowest = pqueue[key]
            keylowest = key
    del pqueue[keylowest]
    return keylowest

# Implement Dijkstra's algorithm
def dijkstra(graph, start):
    # Initialize priority queue, distances, and predecessors    
    pqueue = {}
    dist = {}
    pred = {}
    for v in graph:
        dist[v] = float('inf')
        pred[v] = -1
    dist[start] = 0

    # Fill priority queue with initial distances
    for v in graph:
       pqueue[v] = dist[v]

    # Main loop of Dijkstra's algorithm
    while pqueue:
        # Extract the node with the minimum distance
        u = popmin(pqueue)
        for v in graph[u].keys():
            w = graph[u][v]
            newdist = dist[u] + w
            # Update distances and predecessors if a shorter path is found
            if newdist < dist[v]:
                pqueue[v] = newdist
                dist[v] = newdist
                pred[v] = u

    return dist, pred

# Define the graph
graph = {
    0: {1: 6, 2: 1, 3: 4},
    1: {4: 3},
    2: {1: -3, 3: 2},
    3: {4: -1},
    4: {2: 5}
}

# Run Dijkstra's algorithm on the graph
dist, pred = dijkstra(graph, 0)

# Print the results
print("Shortest distance from start node 0 to all other nodes:")
for v in dist:
    print(v, '=', dist[v])

print("Paths from start node to all other nodes (predecessors of nodes):")
for v in pred:
    print(v, '=', pred[v])
