from socket import *
import sys
import time
import _pickle as pickle
import math

class VectorRouting:
    def __init__(self, port, num_routers, num_neighbors, routing_table):
        self.port = port
        self.num_routers = num_routers
        self.num_neighbors = num_neighbors
        self.routing_table = routing_table

    def calculate_vector_routing(self):
        # Create a UDP socket for communication
        router_socket = socket(AF_INET, SOCK_DGRAM)
        router_socket.bind(('', self.port))

        while True:
            print(self.routing_table)  # Print the current routing table
            time.sleep(1)  # Wait for 1 second

            # Serialize the routing table using pickle
            encoded_msg = pickle.dumps(self.routing_table, -1)

            # Broadcast the routing table to all neighbors
            for neighbor, _ in self.routing_table.items():
                try:
                    router_socket.sendto(encoded_msg, ('', neighbor))
                except:
                    print("Nonexistent Router")

            try:
                # Receive and decode the updated routing table from neighbors
                msg, address = router_socket.recvfrom(1024)
                decoded_msg = pickle.loads(msg)
                neighbor_addr = address[1]

                # Update the routing table using the Bellman-Ford algorithm
                for router, cost in decoded_msg.items():
                    if router not in self.routing_table:
                        self.routing_table[router] = math.inf
                    if self.routing_table[router] > self.routing_table[neighbor_addr] + cost:
                        self.routing_table[router] = self.routing_table[neighbor_addr] + cost
            except:
                continue

def main():
    # Check if the command-line arguments are provided correctly
    if len(sys.argv) < 4:
        print("Usage: python script.py PORT NUM_ROUTERS NUM_NBRS [router_nbr cost] ...")
        return

    # Parse the command-line arguments
    port = int(sys.argv[1])
    num_routers = int(sys.argv[2])
    num_neighbors = int(sys.argv[3])
    routing_table = {port: 0}  # Initialize the routing table with the current router

    # Parse the neighbor information and their costs
    for i in range(0, num_neighbors * 2, 2):
        neighbor = int(sys.argv[4 + i])
        cost = int(sys.argv[4 + i + 1])
        routing_table[neighbor] = cost

    # Create an instance of VectorRouting and start calculating the vector routing
    router = VectorRouting(port, num_routers, num_neighbors, routing_table)
    router.calculate_vector_routing()

if __name__ == '__main__':
    main()
