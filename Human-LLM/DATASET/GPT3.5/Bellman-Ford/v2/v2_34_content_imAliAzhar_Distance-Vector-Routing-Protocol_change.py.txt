import os
import sys
import msvcrt
import threading
import json
import time
from socket import *

send_lock = threading.Lock()

def send_cost_change(port, node, cost):
    with send_lock:
        sock = socket(AF_INET, SOCK_DGRAM)
        message = {
            'packettype': "change",
            'sendingNode': "CostEditor",
            'node': node,
            'cost': cost,
            'port': port
        }
        message = json.dumps(message)
        sock.sendto(message.encode(), ('localhost', port))

def read_file(filename):
    # Placeholder for actual file reading logic
    return {'cost': 0, 'port': 0}  # Dummy data for demonstration

nodes = {}

for arg in sys.argv[1:]:
    node = arg[-5]
    nodes[node] = read_file(arg)

pointer = 0
user_input = 'n'
nodes_taken = False

while True:
    os.system("cls")  # Clear screen (Windows specific)
    print("\n\t         COST EDITOR\n")
    print("\t   Use Q and W for cursor\n\t   Use double E to select\n\n")
    print("=" * 44)

    done = []
    option_num = 0

    for node_a in nodes:
        for node_b in nodes[node_a]:
            if node_b not in done:
                if option_num == pointer:
                    print("  ->", end="")
                    if user_input in ['E', 'e']:
                        nodes_taken = True
                        node_a_changed = node_a
                        node_b_changed = node_b
                        user_input = 'n'
                print(f"\t Cost between {node_a} and {node_b} is {nodes[node_a][node_b]['cost']}")
                option_num += 1
        done.append(node_a)

    print("=" * 44)
    user_input = msvcrt.getch().decode("utf-8")

    if user_input in ['Q', 'q']:
        pointer -= 1
    elif user_input in ['W', 'w']:
        pointer += 1

    pointer = max(pointer, 0)
    pointer = min(pointer, option_num - 1)

    if nodes_taken:
        nodes_taken = False
        user_input = 'n'
        print(f"\n    Enter new cost for nodes {node_a_changed} and {node_b_changed} : ", end="")
        new_cost = input()
        print()
        print(f"  Sending control packet to {node_a_changed} at port {nodes[node_b_changed][node_a_changed]['port']}")
        threading.Thread(target=send_cost_change, kwargs={'port': nodes[node_b_changed][node_a_changed]['port'], 'node': node_b_changed, 'cost': new_cost}).start()
        print(f"  Sending control packet to {node_b_changed} at port {nodes[node_a_changed][node_b_changed]['port']}")
        threading.Thread(target=send_cost_change, kwargs={'port': nodes[node_a_changed][node_b_changed]['port'], 'node': node_a_changed, 'cost': new_cost}).start()
        nodes[node_a_changed][node_b_changed]['cost'] = new_cost
        nodes[node_b_changed][node_a_changed]['cost'] = new_cost
        time.sleep(1)
        print("\n              Costs updated")
        time.sleep(2)
