from flask import Flask, render_template, request
from graph.weighted_graph import weighted_graph
import time

# List of supported currencies
support_currs = ['EUR', 'USD', 'JPY', 'CHF', 'AUD', 'CAD', 'NZD', 'GBP', 'SEK', 'NOK', 'MXN', 'TRY', 'ZAR', 'CNH', 'XAU', 'XAG']
support_currs_string = ', '.join(support_currs)  # String representation of supported currencies

app = Flask(__name__)  # Initialize Flask app

@app.route('/', methods=['GET', 'POST'])
def currencies():
    currencies_string = ""  # Initialize variable to hold selected currencies
    opportunity_string = ""  # Initialize variable to hold arbitrage opportunity information
    
    # Process form submission
    if request.method == 'POST':
        # Retrieve selected currencies from form and format them
        currencies_string = request.form.get('currs', '').upper().replace(" ", "")
        curr_list = [currency for currency in currencies_string.split(',') if currency in support_currs]
        curr_list = list(set(curr_list))
        currencies_string = ', '.join(curr_list)

        # Check if at least two valid currencies are selected
        if len(curr_list) > 1:
            # Create weighted graph with selected currencies
            graph = weighted_graph(curr_list)
            opportunities = graph.show_arbitrage_opportunities()  # Find arbitrage opportunities
            time_epoch = graph.get_time()  # Get current time from graph
            time_string = time.strftime("%Z - %Y/%m/%d, %H:%M:%S", time.localtime(time_epoch))  # Format time
            if len(opportunities) == 0:
                opportunity_string = "No arbitrage opportunity (" + time_string + ")"  # No arbitrage opportunity found
            else:
                opp_string = ' -> '.join(opportunities[0])  # Format arbitrage opportunity
                opportunity_string = opp_string + ' at ' + time_string
        else:
            opportunity_string = "No arbitrage opportunity available with less than two currencies..."  # Less than two currencies selected
    else:
        opportunity_string = "No arbitrage opportunity available with less than two currencies..."  # No form submission

    # Render template with processed data
    return render_template('home.html', currencies=currencies_string, support_currs=support_currs_string, opportunity=opportunity_string)

if __name__ == '__main__':
    app.run(debug=True)  # Run Flask app in debug mode
