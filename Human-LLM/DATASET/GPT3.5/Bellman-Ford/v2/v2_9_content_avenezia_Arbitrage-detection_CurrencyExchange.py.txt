class CurrencyExchange:
    def __init__(self, from_currency, to_currency, exchange_rate):
        """Initialize a CurrencyExchange object with source currency, target currency, and exchange rate."""
        self.__from_currency = from_currency
        self.__to_currency = to_currency
        self.__exchange_rate = float(exchange_rate)

    def __str__(self):
        """Return a string representation of the CurrencyExchange object."""
        return f"{self.__from_currency} -> {self.__to_currency}: {self.__exchange_rate}"

    def __repr__(self):
        """Return a formal representation of the CurrencyExchange object."""
        return self.__str__()

    @property
    def from_currency(self):
        """Get the source currency."""
        return self.__from_currency

    @property
    def to_currency(self):
        """Get the target currency."""
        return self.__to_currency

    @property
    def exchange_rate(self):
        """Get the exchange rate."""
        return self.__exchange_rate

# Example usage:
if __name__ == "__main__":
    # Create a CurrencyExchange object
    exchange = CurrencyExchange("USD", "EUR", 0.85)

    # Print details of the exchange
    print("From:", exchange.from_currency)
    print("To:", exchange.to_currency)
    print("Exchange Rate:", exchange.exchange_rate)
