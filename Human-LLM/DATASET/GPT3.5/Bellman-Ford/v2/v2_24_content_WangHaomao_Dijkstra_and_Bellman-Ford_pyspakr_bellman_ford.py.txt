import datetime
from pprint import pprint
from pyspark import SparkContext
from pyspark_dijkstra import format_result
from data_operation import get_graph
import numpy as np

def initialize_node(node, source):
    """
    Initializes a node with initial distance and path.
    """
    node_id, edges = node
    distance = np.inf
    path = str(node_id)
    if node_id == source:
        distance = 0
    return (node_id, (edges, distance, path))

def generate_new_edges(old_path, old_distance, neighbor):
    """
    Generates new edges for relaxation.
    """
    neighbor_id, weight = neighbor.split(":")
    new_distance = old_distance + float(weight)
    new_path = old_path + '->' + neighbor_id
    return (neighbor_id, ('None', new_distance, new_path))

def relax_paths(path1, path2):
    """
    Relaxes paths and updates distances.
    """
    if path1[0] != 'None':
        neighbors = path1[0]
    else:
        neighbors = path2[0]
    distance1 = path1[1]
    distance2 = path2[1]
    if distance1 <= distance2:
        distance = distance1
        path = path1[2]
    else:
        distance = distance2
        path = path2[2]
    return (neighbors, distance, path)

def split_nodes(node):
    """
    Splits nodes after relaxation.
    """
    node_id, (neighbors, distance, path) = node
    if path.split('->')[-1] != node_id:
        path = path + '->' + node_id
    return (node_id, (neighbors, distance, path))

def bellman_ford(source, graph, sc):
    """
    Implements the Bellman-Ford algorithm.
    """
    nodes = graph.map(lambda node: initialize_node(node, source))
    total_nodes_number = nodes.count()
    nodes = nodes.sortByKey()

    for _ in range(total_nodes_number - 1):
        nodes_values = nodes.map(lambda node: node[1])
        probable_ways = nodes_values.filter(lambda data: data[0] is not None).map(
            lambda data: map(
                lambda neighbor: generate_new_edges(
                    data[2], data[1], neighbor
                ), data[0]
            )
        ).flatMap(lambda x: x)

        mapper = nodes.union(probable_ways)
        reducer = mapper.reduceByKey(lambda x, y: relax_paths(x, y))
        nodes = reducer.map(lambda node: split_nodes(node)).sortByKey()
        nodes.count()

    result = reducer.sortByKey().map(lambda node: format_result(node))
    return result
