class StackArray:
    """
    Stack implementation using an array.
    """

    def __init__(self, limit=10):
        """
        Initialize the StackArray with an optional limit.

        Args:
            limit (int): Maximum size of the stack. Default is 10.
        """
        self.limit = limit
        self.array = []

    def push(self, data):
        """
        Pushes an element onto the top of the stack.

        Args:
            data: The element to push onto the stack.

        Raises:
            ValueError: If the stack is full (Stack Overflow).
        """
        if len(self.array) >= self.limit:
            raise ValueError("Stack Overflow")
        self.array.append(data)
        print("Stack after push:", self.array)

    def is_empty(self):
        """
        Checks if the stack is empty.

        Returns:
            bool: True if the stack is empty, False otherwise.
        """
        return len(self.array) == 0

    def pop(self):
        """
        Removes and returns the top element from the stack.

        Returns:
            The top element of the stack.

        Raises:
            ValueError: If the stack is empty (Stack Underflow).
        """
        if self.is_empty():
            raise ValueError("Stack Underflow")
        return self.array.pop()

    def peek(self):
        """
        Returns the top element of the stack without removing it.

        Returns:
            The top element of the stack.

        Raises:
            ValueError: If the stack is empty (Stack Underflow).
        """
        if self.is_empty():
            raise ValueError("Stack Underflow")
        return self.array[-1]

    def size(self):
        """
        Returns the current size of the stack.

        Returns:
            int: The size of the stack.
        """
        return len(self.array)

    def print_stack(self):
        """
        Prints the elements of the stack.
        """
        print("Current stack:", self.array)


if __name__ == "__main__":
    # Test the StackArray class
    stack = StackArray(5)
    print("Is the stack empty?", stack.is_empty())
    stack.push(5)
    stack.push(6)

    print("Top element of the stack:", stack.peek())

    stack.print_stack()
