import unittest
import redis
from rq import Connection, Worker
from flask.cli import FlaskGroup
from project.worker import create_app

# Create a Flask app instance
app = create_app()

# Create a Flask CLI group
cli = FlaskGroup(create_app=create_app)

# Define a CLI command to run the worker
@cli.command('run_worker')
def run_worker_command():
    """Run the RQ worker."""
    # Get the Redis URL from the Flask app config
    redis_url = app.config['REDIS_URL']
    # Create a Redis connection
    redis_connection = redis.from_url(redis_url)
    # Establish a connection context
    with Connection(redis_connection):
        # Create an RQ worker with the specified queues
        worker = Worker(app.config['QUEUES'])
        # Start the worker
        worker.work()

# Entry point to execute CLI commands
if __name__ == '__main__':
    cli()
