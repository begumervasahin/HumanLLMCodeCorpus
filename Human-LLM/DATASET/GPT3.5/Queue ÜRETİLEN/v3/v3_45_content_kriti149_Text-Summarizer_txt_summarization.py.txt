import bs4 as bs
import urllib.request
import re
import nltk
from nltk.corpus import stopwords
from nltk.tokenize import word_tokenize, sent_tokenize
import heapq

# Download necessary NLTK resources
nltk.download('stopwords')
nltk.download('punkt')

# Retrieve content from the Wikipedia page
url = 'https://en.wikipedia.org/wiki/Global_warming'
source = urllib.request.urlopen(url).read()
soup = bs.BeautifulSoup(source, 'lxml')

# Extract text from paragraph tags
def extract_text(paragraphs):
    text = ""
    for paragraph in paragraphs:
        text += paragraph.text
    return text

# Preprocess the text
def preprocess_text(text):
    text = re.sub(r'\[[0-9]*\]', ' ', text)
    text = re.sub(r'\s+', ' ', text)
    clean_text = text.lower()
    clean_text = re.sub(r'\W', ' ', clean_text)
    clean_text = re.sub(r'\d', ' ', clean_text)
    clean_text = re.sub(r'\s+', ' ', clean_text)
    return clean_text

# Tokenize sentences
def tokenize_sentences(text):
    return sent_tokenize(text)

# Tokenize words and remove stopwords
def tokenize_words(clean_text):
    stop_words = set(stopwords.words('english'))
    word2count = {}
    for word in word_tokenize(clean_text):
        if word not in stop_words:
            if word not in word2count:
                word2count[word] = 1
            else:
                word2count[word] += 1
    return word2count

# Normalize word frequencies
def normalize_word_frequencies(word2count):
    max_freq = max(word2count.values())
    for word in word2count:
        word2count[word] = word2count[word] / max_freq
    return word2count

# Score sentences based on word frequencies
def score_sentences(sentences, word2count):
    sent2score = {}
    for sentence in sentences:
        for word in word_tokenize(sentence.lower()):
            if word in word2count and len(sentence.split(' ')) < 25:
                if sentence not in sent2score:
                    sent2score[sentence] = word2count[word]
                else:
                    sent2score[sentence] += word2count[word]
    return sent2score

# Get the top 5 sentences with the highest scores
def get_top_sentences(sent2score):
    return heapq.nlargest(5, sent2score, key=sent2score.get)

# Print the top sentences
def print_top_sentences(best_sentences):
    print('------------------------------------------------------------')
    for sentence in best_sentences:
        print(sentence)

# Main function to execute the text summarization process
def main():
    text = extract_text(soup.find_all('p'))
    clean_text = preprocess_text(text)
    sentences = tokenize_sentences(text)
    word2count = tokenize_words(clean_text)
    word2count = normalize_word_frequencies(word2count)
    sent2score = score_sentences(sentences, word2count)
    best_sentences = get_top_sentences(sent2score)
    print_top_sentences(best_sentences)

if __name__ == "__main__":
    main()
