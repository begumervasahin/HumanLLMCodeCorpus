import time
from PyQt5.QtCore import QThread, pyqtSignal, pyqtSlot
from PyQt5.QtWidgets import QApplication

class PyDexThread(QThread):
    """A template thread that continuously processes items
    from a FIFO queue."""

    # Signals
    stop_signal = pyqtSignal()

    def __init__(self):
        """Initialize the PyDexThread."""
        super().__init__()
        self.app = QApplication.instance()
        self.stop = False
        self.queue = []

    @pyqtSlot(object)
    def add_item(self, new_item):
        """Append a new item to the queue for processing."""
        self.queue.append(new_item)

    def process(self, item):
        """Process an item in the queue."""
        raise NotImplementedError

    def run(self):
        """Run the thread continuously processing items
        from the queue until the stop flag is set."""
        while True:
            self.app.processEvents()
            if self.check_stop():
                break
            elif self.queue:
                self.process(self.queue.pop(0))
            else:
                time.sleep(0.1)  # Add a small delay to avoid busy-waiting

    def check_stop(self):
        """Check if the thread should stop."""
        return self.stop

    def reset_stop(self):
        """Reset the stop flag."""
        self.stop = False

    def close(self):
        """Stop the thread safely."""
        self.stop_signal.emit()

        # Reset the stop flag once the thread has stopped
        self.finished.disconnect(self.reset_stop)
        self.finished.connect(self.reset_stop)

def reset_slot(signal, slot, reconnect=True):
    """Disconnect all instances of slot from signal.
    If reconnect=True, then reconnect slot to signal."""
    while True:
        try:
            signal.disconnect(slot)
        except TypeError:
            break
    if reconnect:
        signal.connect(slot)
