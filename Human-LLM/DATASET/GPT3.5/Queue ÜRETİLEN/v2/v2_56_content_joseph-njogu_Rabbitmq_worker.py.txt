import time
import pika

def process_message(channel, method, properties, body):
    """
    Process received messages.

    Args:
        channel: Channel object.
        method: Delivery information.
        properties: Message properties.
        body: Message body.
    """
    print(" [x] Received message:", body.decode())
    time.sleep(body.count(b'.'))
    print(" [x] Message processing complete")

def main():
    # Establish connection to RabbitMQ server
    connection = pika.BlockingConnection(pika.ConnectionParameters(host='localhost'))
    channel = connection.channel()

    # Declare the queue to consume messages from
    channel.queue_declare(queue='hello')

    # Set up consumer and start consuming messages
    print(' [*] Waiting for messages. To exit press CTRL+C')
    channel.basic_consume(queue='hello', on_message_callback=process_message, auto_ack=True)
    channel.start_consuming()

if __name__ == "__main__":
    main()
