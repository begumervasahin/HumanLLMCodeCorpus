import util.slurm as slurm
import datetime
import json
import os

# Get current timestamp
timestamp = datetime.datetime.now()
timestamp_str = timestamp.strftime('%Y-%m-%dT%H:%M:%S')

# Define output JSON files
output_json_file = "slurm.json"
output_json_file_tmp = "slurm.json.tmp"

# Define parameters
best_queue = "intellispace"
intellispace_cpus = 0
intellispace_cpus_limit = 160
partition_blacklist = [
    "data_mover",
    "cpu_dev",
    "gpu4_dev",
    "gpu8_dev",
    "fn_long"
]

# Initialize SLURM partitions and squeue
partitions = slurm.Partitions()
squeue = slurm.Squeue()

def get_best_queue(partitions, squeue, intellispace_cpus, intellispace_cpus_limit, partition_blacklist, timestamp_str, output_json_file, best_queue):  
    # Find most idle and most mixed nodes
    most_idle = partitions.most_idle_nodes(blacklist=partition_blacklist)
    most_mixed = partitions.most_mixed_nodes(blacklist=partition_blacklist)

    # Calculate total CPUs in the intellispace partition
    for entry in squeue.entries:
        if entry['PARTITION'] == 'intellispace':
            intellispace_cpus += int(entry['CPUS'])

    # Determine the best queue based on CPU availability
    if intellispace_cpus < intellispace_cpus_limit:
        best_queue = 'intellispace'
    elif most_idle:
        best_queue = most_idle
    elif most_mixed:
        best_queue = most_mixed

    # Prepare data for JSON output
    data = {
        'updated': timestamp_str,
        'most_idle': most_idle,
        'most_mixed': most_mixed,
        'intellispace_cpus': intellispace_cpus,
        'best_queue': best_queue
    }

    # Write data to temporary JSON file and rename it to the final output file
    with open(output_json_file_tmp, 'w') as f:
        json.dump(data, f, indent=4)
    os.rename(output_json_file_tmp, output_json_file)

# Check if SLURM commands returned valid exit status
if partitions.sinfo.returncode == 0 and squeue.returncode == 0:
    # Call function to determine the best queue
    get_best_queue(partitions, squeue, intellispace_cpus, intellispace_cpus_limit, partition_blacklist, timestamp_str, output_json_file, best_queue)   
else:
    # Print error message if SLURM commands returned invalid exit status
    print("[{timestamp}] error: SLURM command returned invalid exit status; sinfo code: {sinfo_code}, squeue code: {squeue_code}".format(
        timestamp=timestamp_str,
        sinfo_code=partitions.sinfo.returncode,
        squeue_code=squeue.returncode
    ))
