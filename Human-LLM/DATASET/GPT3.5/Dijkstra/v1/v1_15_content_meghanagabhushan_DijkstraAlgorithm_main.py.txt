from dijkstra import Graph

# Initialize an empty graph
graph = Graph()

def add_edge():
    n1 = input("Enter node 1: ")
    n2 = input("Enter node 2: ")
    cost = int(input("Enter cost: "))
    graph.add_edge(n1, n2, cost)

def remove_edge():
    n1 = input("Enter node 1: ")
    n2 = input("Enter node 2: ")
    graph.remove_edge(n1, n2)

def dijkstra():
    n1 = input("Enter source node: ")
    n2 = input("Enter destination node: ")
    shortest_path = graph.dijkstra(n1, n2)
    print("Shortest path from", n1, "to", n2, ":", shortest_path)

def view_graph():
    neighbors = graph.neighbours()
    print("Graph structure:")
    for node, neighbor in neighbors.items():
        print(node, "->", neighbor)

def switch_demo(argument):
    switcher = {
        "add": add_edge,
        "remove": remove_edge,
        "dijkstra": dijkstra,
        "view": view_graph,
    }
    func = switcher.get(argument)
    if func:
        func()
    else:
        print("Invalid argument")

options = {
    "add": add_edge,
    "remove": remove_edge,
    "dijkstra": dijkstra,
    "view": view_graph
}

arg = input("Enter one of the following: add, remove, dijkstra, view: ")
switch_demo(arg)
