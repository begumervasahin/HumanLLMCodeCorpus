import collections
import heapq

class PriorityQueue:
    """Priority queue implementation using heapq."""
    
    def __init__(self):
        """Initialize an empty priority queue."""
        self.elements = []
    
    def empty(self):
        """Check if the priority queue is empty."""
        return len(self.elements) == 0
    
    def put(self, item, priority):
        """Put an item into the priority queue with a given priority."""
        heapq.heappush(self.elements, (priority, item))
    
    def get(self):
        """Remove and return the item with the lowest priority from the priority queue."""
        return heapq.heappop(self.elements)[1]
    
    def __len__(self):
        """Return the number of items in the priority queue."""
        return len(self.elements)

class Queue:
    """FIFO queue implementation using collections.deque."""
    
    def __init__(self):
        """Initialize an empty queue."""
        self.elements = collections.deque()
    
    def empty(self):
        """Check if the queue is empty."""
        return len(self.elements) == 0
    
    def put(self, item):
        """Put an item into the end of the queue."""
        self.elements.append(item)
    
    def get(self):
        """Remove and return the first item from the queue."""
        return self.elements.popleft()
    
    def __len__(self):
        """Return the number of items in the queue."""
        return len(self.elements)

def bfs(start, end):
    """Breadth-first search algorithm."""
    frontier = Queue()
    frontier.put(start)
    came_from = {start: None}
    
    while not frontier.empty():
        current = frontier.get()
        current.visit()  # Assuming a 'visit' method exists in the Tile class
        if current == end:
            print("WE MADE IT!!!")
            break
        for next_tile in current.neighbours:
            if next_tile not in came_from:
                frontier.put(next_tile)
                came_from[next_tile] = current
    
    return came_from
