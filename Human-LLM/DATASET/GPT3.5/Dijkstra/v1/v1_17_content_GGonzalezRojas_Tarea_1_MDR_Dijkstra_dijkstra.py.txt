from collections import defaultdict
from heapq import *


def dijkstra(edges, f, t):
    """Find the shortest path from node 'f' to node 't' using Dijkstra's algorithm."""
    graph = defaultdict(list)
    for l, r, c in edges:
        graph[l].append((c, r))

    priority_queue = [(0, f, ())]
    visited = set()

    while priority_queue:
        (cost, v1, path) = heappop(priority_queue)
        if v1 not in visited:
            visited.add(v1)
            path = (v1, path)
            if v1 == t:
                return cost, path
            for c, v2 in graph.get(v1, ()):
                if v2 not in visited:
                    heappush(priority_queue, (cost + c, v2, path))
    return float("inf"), ()
