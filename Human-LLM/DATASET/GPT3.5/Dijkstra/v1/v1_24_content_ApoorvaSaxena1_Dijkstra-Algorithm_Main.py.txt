class Solution:
    @staticmethod
    def get_path(sender, start, sm, path_sum, ht_times):
        if start in ht_times:
            arr = ht_times[start]
            for x in arr:
                nwsm = sm + x[1]
                if path_sum[x[0] - 1] is None or nwsm < path_sum[x[0] - 1]:
                    path_sum[x[0] - 1] = nwsm
                    path_sum = Solution.get_path(start, x[0], nwsm, path_sum, ht_times)
        return path_sum

def max_time(N, K, times):
    ht_times = {}
    for i in times:
        if i[0] in ht_times:
            ht_times[i[0]].append([i[1], i[2]])
        else:
            ht_times[i[0]] = [[i[1], i[2]]]
    path_sum = [None] * N
    path_sum[K - 1] = 0
    path_sum = Solution.get_path(K, K, 0, path_sum, ht_times)

    # If there is a single None in the pathSum list, it means the particular node is unreachable by any other node
    if None not in path_sum:
        return max(path_sum)
    else:
        return -1
