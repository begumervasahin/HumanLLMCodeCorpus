import collections
import heapq

class Queue:
    """
    Last in first out queue. Implemented as a collections.deque (deck / stack).
    Used for BFS.
    """
    def __init__(self):
        self.elements = collections.deque()
    
    def empty(self):
        """Check if the queue is empty."""
        return len(self.elements) == 0
    
    def add(self, x):
        """Add an element to the queue."""
        self.elements.append(x)
    
    def pop(self):
        """Remove and return the first element from the queue."""
        return self.elements.popleft()
    
    def __len__(self):
        """Get the size of the queue."""
        return len(self.elements)

class PriorityQueue:
    """
    Priority queue. Implemented as a heapq from collections.
    Used in Dijkstra and A*.
    """
    def __init__(self):
        self.elements = []
    
    def empty(self):
        """Check if the priority queue is empty."""
        return len(self.elements) == 0
    
    def put(self, item, priority):
        """Put an item into the priority queue with a given priority."""
        heapq.heappush(self.elements, (priority, item))
    
    def pop(self):
        """Remove and return the item with the lowest priority from the priority queue."""
        return heapq.heappop(self.elements)[1]
    
    def __len__(self):
        """Get the size of the priority queue."""
        return len(self.elements)

# Example usage:
if __name__ == "__main__":
    # Create a queue and add elements
    q = Queue()
    q.add(1)
    q.add(2)
    q.add(3)
    
    # Pop elements from the queue
    print(q.pop())  # Output: 1
    print(q.pop())  # Output: 2
    print(q.pop())  # Output: 3
    
    # Create a priority queue and put elements
    pq = PriorityQueue()
    pq.put("task1", 3)
    pq.put("task2", 1)
    pq.put("task3", 2)
    
    # Pop elements from the priority queue
    print(pq.pop())  # Output: task2
    print(pq.pop())  # Output: task3
    print(pq.pop())  # Output: task1
