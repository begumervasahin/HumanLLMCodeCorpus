def find_point(adict, alist):
    min_distance = float('inf')
    k = -1
    for i in alist:
        if adict[i] < min_distance:
            min_distance = adict[i]
            k = i
    return k

inf = float('inf')
matrix = [
    [0, 1, inf, 2, inf, inf],
    [inf, 0, 3, 4, inf, inf],
    [inf, inf, 0, 5, 1, inf],
    [inf, 4, inf, 0, inf, inf],
    [inf, inf, 2, 3, 0, inf],
    [inf, inf, 2, inf, 2, 0]
]

origin = 1
nodes = list(range(len(matrix)))
visited = [origin]
nodes.remove(origin)
min_distance = {origin: 0}

for i in nodes:
    min_distance[i] = float('inf')

re_point = [0] * len(matrix)

for i in visited:
    for j in nodes:
        new_distance = min_distance[i] + matrix[i][j]
        if new_distance < min_distance[j]:
            min_distance[j] = new_distance
            re_point[j] = i
    new_point = find_point(min_distance, nodes)
    if new_point == -1:
        print("Unable to reach point {}".format(nodes))
        nodes.remove(nodes[0])
    else:
        visited.append(new_point)
        nodes.remove(new_point)
        print("%d -> %d" % (re_point[new_point], new_point))
    if not nodes:
        break

print(min_distance)
