class Edge:
    def __init__(self, vertex1, vertex2, weight):
        self.vertex1 = vertex1
        self.vertex2 = vertex2
        self.weight = weight
        self.selected = False
        self.extra = 0

def return_weight(value):
    return value.weight

def run_prim(edgelist, nodecount):
    edgelist.sort(key=return_weight)
    setlist = set()
    edgelist[0].selected = True
    setlist.add(edgelist[0])
    flag = 0
    edgelist[0].extra = 0
    while flag < nodecount - 1:
        count = 1
        not_set = False
        while count < len(edgelist) and not_set is False:
            found_first = False
            found_second = False
            discard = False
            if edgelist[count].extra == 1:
                for k in setlist:
                    if edgelist[count].vertex2 == k.vertex1:
                        found_second = True
                    if edgelist[count].vertex2 == k.vertex2:
                        found_second = True
                    if edgelist[count].vertex1 == k.vertex1:
                        found_first = True
                    if edgelist[count].vertex1 == k.vertex2:
                        found_first = True
                if found_first and found_second:
                    discard = True
                    edgelist[count].extra = 0
                elif found_second or found_first and not discard:
                    setlist.add(edgelist[count])
                    edgelist[count].selected = True
                    edgelist[count].extra = 0
                    not_set = True
            count += 1
        flag += 1
    return

# Example usage:
if __name__ == "__main__":
    # Create edges
    edges = [
        Edge(1, 2, 10),
        Edge(1, 3, 5),
        Edge(2, 3, 1),
        Edge(2, 4, 3),
        Edge(3, 4, 8)
    ]
    # Run Prim's algorithm
    run_prim(edges, 4)
