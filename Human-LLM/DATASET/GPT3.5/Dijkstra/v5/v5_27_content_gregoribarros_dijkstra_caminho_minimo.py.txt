from igraph import Graph

class Arrow:
    def __init__(self, origin='', target='', cost=''):
        self.origin = origin
        self.target = target
        self.cost = cost

    def print_arrow(self):
        print(f"Origin: {self.origin}, Cost: {self.cost}, Target: {self.target}")

def get_arrows(graph, vertex):
    arrows = []
    for edge in graph.incident(vertex):
        origin = graph.es[edge].tuple[0]
        cost = graph.es[edge]['peso']
        target = graph.es[edge].tuple[1]
        arrow = Arrow(origin, target, cost)
        arrows.append(arrow)
    return arrows

def compute_shortest_paths(graph):
    num_vertices = len(graph.vs)
    shortest_paths = [999] * num_vertices
    shortest_paths[0] = 0

    for vertex in graph.vs:
        for arrow in get_arrows(graph, vertex):
            if shortest_paths[arrow.origin] + arrow.cost < shortest_paths[arrow.target]:
                shortest_paths[arrow.target] = shortest_paths[arrow.origin] + arrow.cost

    return shortest_paths

def main():
    my_graph = Graph(directed=True)
    my_graph.add_vertices(5)
    my_graph.vs['nome'] = ['a', 'b', 'c', 'd', 'e']
    my_graph.add_edges([(0, 1), (0, 4), (1, 2), (1, 3), (1, 4), (2, 4)])
    my_graph.es['peso'] = [3, 11, 3, 2, 7, 2]

    shortest_paths = compute_shortest_paths(my_graph)
    print(shortest_paths)

if __name__ == "__main__":
    main()
