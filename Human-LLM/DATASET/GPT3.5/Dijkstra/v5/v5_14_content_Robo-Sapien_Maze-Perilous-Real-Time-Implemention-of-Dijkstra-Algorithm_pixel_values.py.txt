import cv2
import numpy as np

def display_color_info(image):
    # Constants
    FONT = cv2.FONT_HERSHEY_SIMPLEX
    STEP_SIZE = 100

    # Get image dimensions
    height, width, _ = image.shape
    print("Image shape:", image.shape)

    # Iterate through the image in steps of STEP_SIZE pixels
    for i in range(0, height, STEP_SIZE):
        for j in range(0, width, STEP_SIZE):
            # Extract color at the current pixel
            color = image[j, i]

            # Display color information on the image
            cv2.putText(image, str(color), (i, j), FONT, 0.4, (255, 255, 255), 2)

            # Draw a small circle at the current pixel
            cv2.circle(image, (i, j), 3, (255, 255, 255), 1)

    # Display the image
    cv2.imshow('img', image)
    cv2.waitKey(0) & 0xFF
    cv2.destroyAllWindows()

if __name__ == "__main__":
    # Load the image
    img = cv2.imread('newa4.jpg')
    # Process and display color information
    display_color_info(img)
