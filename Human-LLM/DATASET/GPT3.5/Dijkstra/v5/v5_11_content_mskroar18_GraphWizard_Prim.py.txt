def run_prim(edgelist, nodecount):
    # Sort the edge list by weight
    edgelist.sort(key=return_weight)
    
    # Initialize an empty set to store selected edges
    selected_edges = set()
    
    # Select the first edge and add it to the set
    edgelist[0].selected = True
    selected_edges.add(edgelist[0])
    
    flag = 0
    
    # Iterate until all nodes are covered
    while flag < nodecount - 1:
        not_set = False
        
        # Iterate through the edges
        for edge in edgelist[1:]:
            if edge.extra == 1:
                found_first = any(edge.vertex2 == k.vertex1 or edge.vertex2 == k.vertex2 for k in selected_edges)
                found_second = any(edge.vertex1 == k.vertex1 or edge.vertex1 == k.vertex2 for k in selected_edges)
                
                # Discard the edge if both vertices are already in the set
                if found_first and found_second:
                    edge.extra = 0
                # Add the edge to the set if one of the vertices is found
                elif found_first or found_second:
                    selected_edges.add(edge)
                    edge.selected = True
                    edge.extra = 0
                    not_set = True
                    break
            
        flag += 1

def return_weight(value):
    return value.weight
