import numpy as np
import cv2

def calculate_distance(point1, point2):
    """Calculate the Euclidean distance between two points."""
    return ((point2[0] - point1[0]) ** 2 + (point2[1] - point1[1]) ** 2) ** 0.5

def get_feasible_positions(image, current_position):
    """Get feasible positions for the next move."""
    height, width, _ = image.shape
    feasible_squares = [(current_position[0] + 20, current_position[1]),
                        (current_position[0] - 20, current_position[1]),
                        (current_position[0], current_position[1] + 20),
                        (current_position[0], current_position[1] - 20)]
    allowed_positions = []

    for square in feasible_squares:
        x, y = square
        if 0 <= x < height and 0 <= y < width:
            if image[y, x] == 255:  # Thresholded value
                allowed_positions.append(square)

    return allowed_positions

def find_next_position(image, current_position, goal_position):
    """Find the next position of the bot."""
    gray_image = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    _, threshold_image = cv2.threshold(gray_image, 75, 255, cv2.THRESH_BINARY_INV)

    allowed_positions = get_feasible_positions(threshold_image, current_position)

    min_distance = float('inf')
    next_position = current_position
    
    for position in allowed_positions:
        distance = calculate_distance(position, goal_position)
        if distance < min_distance:
            min_distance = distance
            next_position = position

    return next_position, next_position[0] - current_position[0], next_position[1] - current_position[1]
