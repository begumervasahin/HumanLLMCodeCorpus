import math
import matplotlib.pyplot as plt
from sympy import *

# Import Node class from Node.py
from Node import Node

def generate_points_on_circle(radius, num_points=100):
    """
    Generate points evenly distributed around a circumference.

    Args:
        radius (float): Radius of the circumference.
        num_points (int): Number of points to generate. Defaults to 100.

    Returns:
        list: List of (x, y) coordinates representing points on the circumference.
    """
    points = [(cos(2 * pi / num_points * x) * radius, sin(2 * pi / num_points * x) * radius) for x in range(num_points + 1)]
    return [(0, 0)] + points

def calculate_distance(node1, node2):
    """
    Calculate the Euclidean distance between two nodes.

    Args:
        node1 (Node): First node.
        node2 (Node): Second node.

    Returns:
        float: Euclidean distance between the two nodes.
    """
    return math.sqrt((node1.x - node2.x) ** 2 + (node1.y - node2.y) ** 2)

def generate_graph(radius=1.0, num_nodes=8):
    """
    Generate a graph with nodes distributed around a circumference.

    Args:
        radius (float): Radius of the circumference. Defaults to 1.0.
        num_nodes (int): Number of nodes to generate. Defaults to 8.

    Returns:
        tuple: Tuple containing points, nodes, and edges of the graph.
    """
    points = generate_points_on_circle(radius, num_nodes)
    nodes = [Node(x, y, i) for i, (x, y) in enumerate(points)]
    edges = [[0 if i == j else calculate_distance(nodes[i], nodes[j]) for j in range(len(nodes))] for i in range(len(nodes))]
    return points, nodes, edges

def plot_graph(points, edges, radius=1.0):
    """
    Plot the graph.

    Args:
        points (list): List of (x, y) coordinates representing points on the graph.
        edges (list): List of lists representing edges between nodes.
        radius (float): Radius of the circumference. Defaults to 1.0.
    """
    x = [p[0] for p in points]
    y = [p[1] for p in points]
    plt.plot(x, y, 'ro')

    for i in range(len(edges)):
        for j in range(len(edges[i])):
            if i != j:
                plt.plot([points[i][0], points[j][0]], [points[i][1], points[j][1]], 'b-', lw=1)

    size = radius * 2
    plt.axis([-size, size, -size, size])
    plt.grid(True)
    plt.gca().set_aspect('equal', adjustable='box')
    plt.show()

def main():
    circle_radius = 1.0
    num_nodes = 8
    points, _, edges = generate_graph(circle_radius, num_nodes)
    plot_graph(points, edges, circle_radius)

if __name__ == "__main__":
    main()
