from pprint import pprint
from collections import deque
import copy
import grid_helper as gh

def scan_grid(grid, start=(0, 0)):
    """
    Scan the entire grid to establish paths from the starting position to any other point.
    """
    queue = deque([start])
    came_from = {start: None}
    
    while queue:
        current_pos = queue.popleft()
        neighbors = gh.get_neighbors(grid, *current_pos)
        
        for neighbor in neighbors:
            if neighbor not in came_from:
                queue.append(neighbor)
                came_from[neighbor] = current_pos
                
    return came_from

def find_and_draw_path(start_pos, end_pos, grid, directions):
    """
    Find a path from the starting position to the end position and draw it on the grid.
    """
    path = gh.find_path(start_pos, end_pos, directions)
    grid_with_path = gh.draw_path(path, copy.deepcopy(grid))
    return path, grid_with_path

def display_path_and_steps(start_pos, end_pos, path, grid_with_path):
    """
    Display the path and the number of steps taken to reach the end position.
    """
    pprint(grid_with_path)
    print(f"Number of steps for Path to {end_pos}: {len(path)}")

def init():
    """
    Initialize the grid, find paths from a designated starting position, and display the paths.
    """
    # Generate the initial grid with obstacles
    initial_grid = gh.generate_grid_obstacle()
    
    # Define the starting position
    start_pos = (3, 0)
    
    # Scan the grid to establish paths
    directions = scan_grid(initial_grid, start_pos)
    
    # Find and display path to (8, 8)
    end_pos_1 = (8, 8)
    path1, grid_with_path1 = find_and_draw_path(start_pos, end_pos_1, initial_grid, directions)
    display_path_and_steps(start_pos, end_pos_1, path1, grid_with_path1)
    
    # Find and display path to (4, 7)
    end_pos_2 = (4, 7)
    path2, grid_with_path2 = find_and_draw_path(start_pos, end_pos_2, initial_grid, directions)
    display_path_and_steps(start_pos, end_pos_2, path2, grid_with_path2)

if __name__ == "__main__":
    init()
