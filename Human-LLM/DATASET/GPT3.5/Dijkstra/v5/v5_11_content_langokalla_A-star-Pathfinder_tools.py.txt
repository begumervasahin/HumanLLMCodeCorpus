import collections
import heapq

class PriorityQueue:
    """Implements a priority queue using heapq."""
    
    def __init__(self):
        """Initializes an empty priority queue."""
        self.elements = []

    def is_empty(self):
        """Checks if the priority queue is empty."""
        return len(self.elements) == 0

    def put(self, item, priority):
        """Adds an item to the priority queue with a given priority."""
        heapq.heappush(self.elements, (priority, item))

    def get(self):
        """Removes and returns the item with the lowest priority from the priority queue."""
        return heapq.heappop(self.elements)[1]

    def __len__(self):
        """Returns the number of items in the priority queue."""
        return len(self.elements)


class Queue:
    """Implements a FIFO queue using collections.deque."""
    
    def __init__(self):
        """Initializes an empty queue."""
        self.elements = collections.deque()

    def is_empty(self):
        """Checks if the queue is empty."""
        return len(self.elements) == 0

    def put(self, item):
        """Adds an item to the end of the queue."""
        self.elements.append(item)

    def get(self):
        """Removes and returns the first item from the queue."""
        return self.elements.popleft()

    def __len__(self):
        """Returns the number of items in the queue."""
        return len(self.elements)


def bfs(start, end):
    """Performs breadth-first search from start to end."""
    frontier = Queue()
    frontier.put(start)
    came_from = {start: None}
    
    while not frontier.is_empty():
        current = frontier.get()
        current.visit()  # Assuming a 'visit' method exists in the Tile class
        
        if current == end:
            print("WE MADE IT!!!")
            break
        
        for next_tile in current.neighbours:
            if next_tile not in came_from:
                frontier.put(next_tile)
                came_from[next_tile] = current
    
    return came_from
