def find_min_distance_node(distances, unvisited_nodes):
    """
    Finds the node with the minimum distance in the unvisited list.
    """
    min_distance = float('inf')
    min_node = -1
    for node in unvisited_nodes:
        if distances[node] < min_distance:
            min_distance = distances[node]
            min_node = node
    return min_node

# Define infinity as a large value
infinity = 10000

# Adjacency matrix representing the graph
adjacency_matrix = [
    [0, 1, infinity, 2, infinity, infinity],
    [infinity, 0, 3, 4, infinity, infinity],
    [infinity, infinity, 0, 5, 1, infinity],
    [infinity, 4, infinity, 0, infinity, infinity],
    [infinity, infinity, 2, 3, 0, infinity],
    [infinity, infinity, 2, infinity, 2, 0]
]

origin = 1
num_nodes = len(adjacency_matrix)
nodes = list(range(num_nodes))
visited = [origin]
nodes.remove(origin)

# Initialize minimum distance dictionary with origin node
min_distance = {origin: 0}
for node in nodes:
    min_distance[node] = infinity

# Array to keep track of the predecessor node on the shortest path
predecessor_node = [0] * num_nodes

# Main loop to find shortest paths
for node in visited:
    for j in nodes:
        new_distance = min_distance[node] + adjacency_matrix[node][j]
        if new_distance < min_distance[j]:
            min_distance[j] = new_distance
            predecessor_node[j] = node
    
    min_distance_node = find_min_distance_node(min_distance, nodes)
    if min_distance_node == -1:
        print("Unable to reach point {}".format(nodes))
        nodes.remove(nodes[0])
    else:
        visited.append(min_distance_node)
        nodes.remove(min_distance_node)
        print("%d -> %d" % (predecessor_node[min_distance_node], min_distance_node))
    
    if not nodes:
        break

print(min_distance)
