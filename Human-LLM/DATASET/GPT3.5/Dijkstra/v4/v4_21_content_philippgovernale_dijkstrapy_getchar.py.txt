class _Getch:
    """Gets a single character from standard input. Does not echo to the screen."""
    
    def __init__(self):
        # Initialize the appropriate implementation based on the platform
        try:
            self.impl = _GetchWindows()
        except ImportError:
            self.impl = _GetchUnix()
    
    def __call__(self):
        # Call the appropriate implementation to get a single character
        return self.impl()

class _GetchUnix:
    """Implementation for Unix-like systems."""
    
    def __init__(self):
        import tty, sys

    def __call__(self):
        import sys, tty, termios
        
        # Get the file descriptor for standard input
        fd = sys.stdin.fileno()
        
        # Save the current terminal settings
        old_settings = termios.tcgetattr(fd)
        
        try:
            # Set the terminal to raw mode to read a single character
            tty.setraw(sys.stdin.fileno())
            char = sys.stdin.read(1)
            
            # Convert backspace character to '\b' for consistency
            if char == '\x7f':
                char = '\b'
        finally:
            # Restore the terminal settings to their original state
            termios.tcsetattr(fd, termios.TCSADRAIN, old_settings)
        
        return char

class _GetchWindows:
    """Implementation for Windows systems."""
    
    def __init__(self):
        import msvcrt

    def __call__(self):
        import msvcrt
        
        # Use msvcrt.getch() to get a single character
        return msvcrt.getch()
