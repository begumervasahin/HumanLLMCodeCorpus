import math
import matplotlib.pyplot as plt
from sympy import *

# Import Node class from Node.py
from Node import Node

def points_in_circumference(radius, num_points=100):
    """
    Generates points evenly distributed around a circumference.

    Args:
        radius (float): Radius of the circumference.
        num_points (int): Number of points to generate. Defaults to 100.

    Returns:
        list: List of (x, y) coordinates representing points on the circumference.
    """
    return [(0, 0)] + [(cos(2 * pi / num_points * x) * radius, sin(2 * pi / num_points * x) * radius) for x in range(num_points + 1)]

def calculate_distance(node1, node2):
    """
    Calculates the Euclidean distance between two nodes.

    Args:
        node1 (Node): First node.
        node2 (Node): Second node.

    Returns:
        float: Euclidean distance between the two nodes.
    """
    return math.sqrt((node1.x - node2.x) ** 2 + (node1.y - node2.y) ** 2)

def generate_graph(radius=1.0, num_nodes=8):
    """
    Generates a graph with nodes distributed around a circumference.

    Args:
        radius (float): Radius of the circumference. Defaults to 1.0.
        num_nodes (int): Number of nodes to generate. Defaults to 8.

    Returns:
        tuple: Tuple containing points, nodes, and edges of the graph.
    """
    points = points_in_circumference(radius, num_nodes)
    nodes = [Node(points[i][0], points[i][1], i) for i in range(len(points))]
    edges = [[0 if i == j else calculate_distance(nodes[i], nodes[j]) for j in range(len(nodes))] for i in range(len(nodes))]
    return points, nodes, edges

def plot_graph(points, lines, radius=1.0):
    """
    Plots the graph.

    Args:
        points (list): List of (x, y) coordinates representing points on the graph.
        lines (list): List of lists representing edges between nodes.
        radius (float): Radius of the circumference. Defaults to 1.0.

    Returns:
        matplotlib.pyplot.figure: Figure object containing the plotted graph.
    """
    x = [p[0] for p in points]
    y = [p[1] for p in points]
    plt.plot(x, y, 'ro')

    for line in lines:
        print(line[0].idx, line[1].idx)
        plt.plot([line[0].x, line[1].x], [line[0].y, line[1].y], 'b-', lw=1)

    size = radius * 2
    plt.axis([-size, size, -size, size])
    plt.grid(True)
    plt.gca().set_aspect('equal', adjustable='box')
    plt.show()

# Example usage:
def main():
    circle_radius = 1.0
    num_nodes = 8
    points, nodes, edges = generate_graph(circle_radius, num_nodes)
    plot_graph(points, edges, circle_radius)

if __name__ == "__main__":
    main()
