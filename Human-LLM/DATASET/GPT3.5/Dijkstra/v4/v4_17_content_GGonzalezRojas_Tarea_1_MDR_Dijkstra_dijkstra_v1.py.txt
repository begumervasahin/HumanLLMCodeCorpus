from collections import defaultdict
from heapq import heappop, heappush

def dijkstra(graph, start, end):
    # Convert the list of edges to a dictionary representation
    graph_dict = defaultdict(list)
    for node, next_node, weight in graph:
        graph_dict[node].append((weight, next_node))

    # Initialize the priority queue with the starting node
    queue = [(0, start, ())]  # Tuple of (cost, current_node, path)
    visited = set()

    # Perform Dijkstra's algorithm
    while queue:
        cost, current_node, path = heappop(queue)
        if current_node not in visited:
            visited.add(current_node)
            path = (current_node, path)  # Update the path with the current node
            if current_node == end:
                return (cost, path)  # Return the cost and the path if the end node is reached
            for weight, next_node in graph_dict.get(current_node, ()):
                if next_node not in visited:
                    new_cost = cost + weight
                    heappush(queue, (new_cost, next_node, path))  # Update the queue with the new cost and node

    # Return infinity if the end node is unreachable
    return float("inf")
