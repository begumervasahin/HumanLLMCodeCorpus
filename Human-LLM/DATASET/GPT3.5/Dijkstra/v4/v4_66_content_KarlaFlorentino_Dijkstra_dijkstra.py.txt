import sys

def read_matrix(file_path):
    """
    Read the matrix from the given file.
    """
    with open(file_path, 'r') as file:
        content = file.readlines()

    matrix = []
    for line in content:
        row = [cell.strip() for cell in line.split()]
        matrix.append(row)

    return matrix

def generate_adjacent_nodes(matrix):
    """
    Generate adjacent nodes based on the matrix.
    """
    adjacent_nodes = []
    for i in range(len(matrix)):
        adj_row = []
        for j in range(len(matrix[i])):
            if matrix[i][j] != "1024":
                adj_row.append(f"{j} {matrix[i][j]}")
        adjacent_nodes.append(adj_row)
    return adjacent_nodes

def dijkstra(adjacent_nodes, start, end):
    """
    Apply Dijkstra's algorithm to find the shortest path.
    """
    visited_nodes = [0] * len(adjacent_nodes)
    solution = []
    local = [0] * len(adjacent_nodes)
    
    start = int(start)
    end = int(end)
    visited_nodes[start] = 1
    current_node = start
    current_distance = 0

    solution.append(f'- {start} 0')

    while visited_nodes[end] != 1:
        min_adj_node = adjacent_nodes[current_node][0].split()[0]
        min_distance = int(adjacent_nodes[current_node][0].split()[1]) + current_distance

        for i in range(1, len(adjacent_nodes[current_node])):
            adj_node, distance = map(int, adjacent_nodes[current_node][i].split())
            if not visited_nodes[adj_node]:
                if adj_node == end:
                    prev_node = current_node
                elif distance + current_distance < min_distance:
                    min_adj_node = adj_node
                    min_distance = distance + current_distance
                    prev_node = current_node
            elif prev_node == start or visited_nodes[min_adj_node]:
                min_adj_node = adj_node
                min_distance = distance + current_distance

        if not prev_node:
            prev_node = current_node

        prev_distance = int(solution[local[prev_node]][0].split()[2])

        if len(solution) > 2 and prev_node != start and min_adj_node != end:
            for i in range(len(adjacent_nodes[prev_node])):
                adj_node, distance = map(int, adjacent_nodes[prev_node][i].split())
                if not visited_nodes[adj_node]:
                    next_distance = distance + prev_distance
                    if next_distance < min_distance:
                        min_adj_node = adj_node
                        min_distance = next_distance

        solution.append(f'{prev_node} {min_adj_node} {min_distance}')
        visited_nodes[min_adj_node] = 1
        local[min_adj_node] = len(solution) - 1
        current_node = min_adj_node
        current_distance = min_distance

    return solution

def display_solution(solution, start, end):
    """
    Display the shortest path solution.
    """
    if start == end:
        print(start)
    else:
        path = ''
        path += solution[-1][0].split()[1] + ' '
        after = solution[-1][0].split()[0]

        while int(after) != int(start):
            for i in range(1, len(solution) - 1):
                if solution[i][0].split()[1] == after:
                    path += solution[i][0].split()[1] + ' '
                    after = solution[i][0].split()[0]

        path += solution[0][0].split()[1]
        print(path[::-1])

if __name__ == '__main__':
    file_path, start_node, end_node = sys.argv[1:]
    adjacent_nodes = generate_adjacent_nodes(read_matrix(file_path))
    solution = dijkstra(adjacent_nodes, start_node, end_node)
    display_solution(solution, start_node, end_node)
