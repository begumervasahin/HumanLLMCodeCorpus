import collections
import heapq

class Queue:
    """A FIFO queue implemented using collections.deque."""
    
    def __init__(self):
        """Initialize an empty queue."""
        self._elements = collections.deque()
    
    def empty(self):
        """Check if the queue is empty."""
        return len(self._elements) == 0
    
    def add(self, item):
        """Add an item to the end of the queue."""
        self._elements.append(item)
    
    def pop(self):
        """Remove and return the first item from the queue."""
        return self._elements.popleft()
    
    def __len__(self):
        """Return the number of items in the queue."""
        return len(self._elements)

class PriorityQueue:
    """A priority queue implemented using heapq."""
    
    def __init__(self):
        """Initialize an empty priority queue."""
        self._elements = []
    
    def empty(self):
        """Check if the priority queue is empty."""
        return len(self._elements) == 0
    
    def put(self, item, priority):
        """Put an item into the priority queue with a given priority."""
        heapq.heappush(self._elements, (priority, item))
    
    def pop(self):
        """Remove and return the item with the lowest priority from the priority queue."""
        return heapq.heappop(self._elements)[1]
    
    def __len__(self):
        """Return the number of items in the priority queue."""
        return len(self._elements)
