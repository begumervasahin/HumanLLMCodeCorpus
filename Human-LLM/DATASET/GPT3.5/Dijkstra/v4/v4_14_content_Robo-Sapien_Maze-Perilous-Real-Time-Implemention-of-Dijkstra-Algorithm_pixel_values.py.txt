# Importing necessary libraries
import cv2
import numpy as np

# Loading the image
img = cv2.imread('newa4.jpg')

# Getting dimensions of the image
height, width, channels = img.shape
print("Image shape:", img.shape)

# Defining font for text display
font = cv2.FONT_HERSHEY_SIMPLEX

# Iterating through the image in steps of 100 pixels
for i in range(0, height, 100):
    for j in range(0, width, 100):
        # Extracting color at the current pixel
        color = img[j, i]
        
        # Putting text on the image with the color information
        cv2.putText(img, str(color), (i, j), font, 0.4, (255, 255, 255), 2)
        
        # Drawing a small circle at the current pixel
        cv2.circle(img, (i, j), 3, (255, 255, 255), 1)

# Displaying the image
cv2.imshow('img', img)

# Waiting for a key press and then closing all windows
cv2.waitKey(0) & 0xFF
cv2.destroyAllWindows()
