from __future__ import print_function
from CPUtimer import CPUTimer
from numpy.polynomial import polynomial as P
from data3 import instance_iterator, print_solution

def solve(instance_path):
    # Create a CPU timer
    timer = CPUTimer()

    # Iterate through each instance
    for instance in instance_iterator(instance_path):
        instance_name, g, p1, p2 = instance
        
        # Reset and start the timer
        timer.reset()
        timer.start()
        
        # Perform polynomial multiplication
        for i in range(2):
            result = P.polymul(p1, p2)
            timer.lap()
        
        # Stop the timer
        timer.stop()
        
        # Print the solution
        print_solution(result, instance_name, '3b', timer)

def multiply_arrays_elementwise(g, x, y):
    # Initialize an empty list for the product
    product = []
    
    # Multiply corresponding elements of x and y arrays and store the result in product
    for i in range(g):
        product.append(karatsuba(int(x[i]), int(y[i])))
    
    return product

def karatsuba(x, y):
    # Perform Karatsuba multiplication
    if len(str(x)) == 1 or len(str(y)) == 1:
        return x * y
    else:
        n = max(len(str(x)), len(str(y)))
        nby2 = n // 2
        a = x // 10**nby2
        b = x % 10**nby2
        c = y // 10**nby2
        d = y % 10**nby2
        ac = karatsuba(a, c)
        bd = karatsuba(b, d)
        ad_plus_bc = karatsuba(a + b, c + d) - ac - bd
        prod = ac * 10**(2 * nby2) + ad_plus_bc * 10**nby2 + bd
        return prod
