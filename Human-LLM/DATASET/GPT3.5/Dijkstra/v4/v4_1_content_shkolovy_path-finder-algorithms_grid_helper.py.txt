def heuristic_distance(pos, end_pos, type="euclidean"):
    """
    Calculate the heuristic distance between two positions.

    Parameters:
    - pos: Current position.
    - end_pos: End position.
    - type: Type of distance calculation ("euclidean" or "manhattan").

    Returns:
    - The calculated heuristic distance.
    """
    dx = abs(pos[0] - end_pos[0])
    dy = abs(pos[1] - end_pos[1])
    if type == "manhattan":
        return dx + dy  # Manhattan distance
    return math.sqrt(dx * dx + dy * dy)  # Euclidean distance

def find_path(start, end, came_from):
    """
    Find the shortest path from start to end point based on the 'came_from' dictionary.

    Parameters:
    - start: Starting position.
    - end: Ending position.
    - came_from: Dictionary containing the path information.

    Returns:
    - The constructed path.
    """
    path = [end]
    current = end
    while current != start:
        current = came_from[current]
        path.append(current)
    path.reverse()
    return path

def get_cost(grid, pos):
    """
    Get the cost of a position in the grid.

    Parameters:
    - grid: The grid containing cell values.
    - pos: Position in the grid.

    Returns:
    - The cost of the cell.
    """
    col_val = grid[pos[0]][pos[1]]
    return int(col_val) if col_val.isdigit() else 1

def get_neighbors(grid, row, col):
    """
    Get valid neighboring cells in the grid.

    Parameters:
    - grid: The grid containing cell values.
    - row: Row index of the cell.
    - col: Column index of the cell.

    Returns:
    - List of valid neighboring cells.
    """
    height = len(grid)
    width = len(grid[0])
    neighbors = [(row + 1, col), (row, col - 1), (row - 1, col), (row, col + 1)]        
    if (row + col) % 2 == 0:
        neighbors.reverse()
    neighbors = filter(lambda t: (0 <= t[0] < height and 0 <= t[1] < width), neighbors) 
    neighbors = filter(lambda t: (grid[t[0]][t[1]] != OBSTACLE_COL), neighbors)
    return neighbors

def draw_path(path, grid):
    """
    Mark the path on the grid.

    Parameters:
    - path: The path to be marked.
    - grid: The grid to mark the path on.

    Returns:
    - The grid with the marked path.
    """
    for row, col in path:
        grid[row][col] = PATH_COL
    start_pos = path[0]
    end_pos = path[-1]
    grid[start_pos[0]][start_pos[1]] = START_COL
    grid[end_pos[0]][end_pos[1]] = END_COL
    return grid
