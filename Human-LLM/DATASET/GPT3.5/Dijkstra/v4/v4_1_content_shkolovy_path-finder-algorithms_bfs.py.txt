from pprint import pprint
from collections import deque
import copy
import grid_helper as gh

def scan_grid(grid, start=(0, 0)):
    """
    Scan the entire grid to establish paths from the starting position to any other point.
    """
    queue = deque()
    queue.append(start)
    came_from = {start: None}
    
    while queue:
        current_pos = queue.popleft()
        neighbors = gh.get_neighbors(grid, current_pos[0], current_pos[1])
        
        for neighbor in neighbors:
            if neighbor not in came_from:
                queue.append(neighbor)
                came_from[neighbor] = current_pos
                
    return came_from

def init():
    """
    Initialize the grid, find paths from a designated starting position, and display the paths.
    """
    # Generate the initial grid with obstacles
    initial_grid = gh.generate_grid_obstacle()
    
    # Define the starting position
    start_pos = (3, 0)
    
    # Scan the grid to establish paths
    directions = scan_grid(initial_grid, start_pos)
    
    # Find a path from the starting position to (8, 8)
    end_pos_1 = (8, 8)
    path1 = gh.find_path(start_pos, end_pos_1, directions)
    grid_with_path1 = gh.draw_path(path1, copy.deepcopy(initial_grid))
    pprint(grid_with_path1)
    print(f"Number of steps for Path 1: {len(path1)}")
    
    # Find a path from the starting position to (4, 7)
    end_pos_2 = (4, 7)
    path2 = gh.find_path(start_pos, end_pos_2, directions)
    grid_with_path2 = gh.draw_path(path2, copy.deepcopy(initial_grid))
    pprint(grid_with_path2)
    print(f"Number of steps for Path 2: {len(path2)}")

if __name__ == "__main__":
    init()
