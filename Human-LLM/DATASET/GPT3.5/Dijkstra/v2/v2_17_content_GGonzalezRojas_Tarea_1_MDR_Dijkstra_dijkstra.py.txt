from collections import defaultdict
from heapq import *


def dijkstra(edges, start, end):
    """
    Find the shortest path from node 'start' to node 'end' using Dijkstra's algorithm.

    Args:
    - edges (list): List of tuples representing edges in the graph. Each tuple contains
                    three elements: (source node, destination node, edge cost).
    - start (str): Starting node of the path.
    - end (str): Ending node of the path.

    Returns:
    - tuple: A tuple containing the cost of the shortest path and the path itself.
    """
    graph = defaultdict(list)
    for source, destination, cost in edges:
        graph[source].append((cost, destination))

    priority_queue = [(0, start, ())]
    visited = set()

    while priority_queue:
        cost, current_node, path = heappop(priority_queue)
        if current_node not in visited:
            visited.add(current_node)
            path = (current_node, path)
            if current_node == end:
                return cost, path
            for next_cost, next_node in graph.get(current_node, ()):
                if next_node not in visited:
                    heappush(priority_queue, (cost + next_cost, next_node, path))
    return float("inf"), ()
