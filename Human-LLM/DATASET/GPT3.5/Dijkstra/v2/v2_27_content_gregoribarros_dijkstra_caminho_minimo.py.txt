from igraph import *

class Arrow:
    def __init__(self):
        self.origin = ''
        self.target = ''
        self.cost = ''

    def print_arrow(self):
        print(f"Origin: {self.origin}, Cost: {self.cost}, Target: {self.target}")

def run_graph(graph, vertex):
    arrows = []
    for edge in graph.incident(vertex):
        arrow = Arrow()
        arrow.origin = graph.es[edge].tuple[0]
        arrow.cost = graph.es[edge]['peso']
        arrow.target = graph.es[edge].tuple[1]
        arrows.append(arrow)
    return arrows

my_graph = Graph(directed=True)
my_graph.add_vertices(5)
my_graph.vs['nome'] = ['a', 'b', 'c', 'd', 'e']
my_graph.add_edges([(0, 1), (0, 4), (1, 2), (1, 3), (1, 4), (2, 4)])
my_graph.es['peso'] = [3, 11, 3, 2, 7, 2]

cost_list = [999] * 5
cost_list[0] = 0

for vertex in my_graph.vs:
    for arrow in run_graph(my_graph, vertex):
        if cost_list[arrow.origin] + arrow.cost < cost_list[arrow.target]:
            cost_list[arrow.target] = cost_list[arrow.origin] + arrow.cost

print(cost_list)
