def bi_dijkstra_PQ(G, start, target):
    P_f = {start}
    P_b = {target}
    S_f = PriorityQueue()
    S_b = PriorityQueue()
    D_f = {n: 0 if n == start else G[start][n]["weight"] for n in G.nodes()}
    D_b = {n: 0 if n == target else G[target][n]["weight"] for n in G.nodes()}
    p_f = {n: start for n in G.nodes()}
    p_b = {n: target for n in G.nodes()}

    for n, dist in D_f.items():
        S_f.add_task(n, dist)
    for n, dist in D_b.items():
        S_b.add_task(n, dist)

    while True:
        u, Du = S_f.pop_task()
        if u in P_f:
            continue
        P_f.add(u)
        for v, Dv in S_f:
            if v in P_f:
                continue
            if G.has_edge(u, v):
                if D_f[v] > D_f[u] + G[u][v]["weight"]:
                    D_f[v] = D_f[u] + G[u][v]["weight"]
                    p_f[v] = u
                    S_f.add_task(v, D_f[v])
        if u in P_b:
            w = u
            break

        u, Du = S_b.pop_task()
        if u in P_b:
            continue
        P_b.add(u)
        for v, Dv in S_b:
            if v in P_b:
                continue
            if G.has_edge(u, v):
                if D_b[v] > D_b[u] + G[u][v]["weight"]:
                    D_b[v] = D_b[u] + G[u][v]["weight"]
                    p_b[v] = u
                    S_b.add_task(v, D_b[v])
        if u in P_f:
            w = u
            break

    min_dist = D_f[w] + D_b[w]
    SP_node = w
    for i in G.nodes():
        if D_f[i] + D_b[i] < min_dist:
            min_dist = D_f[i] + D_b[i]
            SP_node = i
            SP = deque()
        else:
            SP = deque([SP_node])
    next = SP_node
    while next != start:
        for i, k in p_f.items():
            if i == next:
                SP.appendleft(k)
                next = k
    next = SP_node
    while next != target:
        for i, k in p_b.items():
            if i == next:
                SP.append(k)
                next = k
    return list(SP)
