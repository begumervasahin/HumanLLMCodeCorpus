class Edge:
    def __init__(self, vertex1, vertex2, weight):
        self.vertex1 = vertex1
        self.vertex2 = vertex2
        self.weight = weight
        self.selected = False
        self.extra = 0

def return_weight(value):
    return value.weight

def run_prim(edgelist, nodecount):
    # Sort the edges based on weight
    edgelist.sort(key=return_weight)
    
    # Initialize an empty set to store selected edges
    selected_edges = set()
    
    # Select the first edge and add it to the set
    edgelist[0].selected = True
    selected_edges.add(edgelist[0])
    
    # Start iterating through the edges
    flag = 0
    edgelist[0].extra = 0
    while flag < nodecount - 1:
        count = 1
        not_set = False
        
        # Iterate through the edges
        while count < len(edgelist) and not_set is False:
            found_first = False
            found_second = False
            discard = False
            
            # Check if the edge is already selected
            if edgelist[count].extra == 1:
                # Check if the edge's vertices are already in the selected set
                for k in selected_edges:
                    if edgelist[count].vertex2 == k.vertex1:
                        found_second = True
                    if edgelist[count].vertex2 == k.vertex2:
                        found_second = True
                    if edgelist[count].vertex1 == k.vertex1:
                        found_first = True
                    if edgelist[count].vertex1 == k.vertex2:
                        found_first = True
                
                # Discard the edge if both vertices are found in the selected set
                if found_first and found_second:
                    discard = True
                    edgelist[count].extra = 0
                
                # Add the edge to the selected set if one of the vertices is found
                elif found_second or found_first and not discard:
                    selected_edges.add(edgelist[count])
                    edgelist[count].selected = True
                    edgelist[count].extra = 0
                    not_set = True
            
            count += 1
        flag += 1
    
    return

# Example usage:
if __name__ == "__main__":
    # Create edges
    edges = [
        Edge(1, 2, 10),
        Edge(1, 3, 5),
        Edge(2, 3, 1),
        Edge(2, 4, 3),
        Edge(3, 4, 8)
    ]
    
    # Run Prim's algorithm
    run_prim(edges, 4)
