import networkx as nx
import operator

def update_weight(client, student_weight, student_loss):
    """
    Update the weight of a student based on loss.

    Args:
        client: The client object.
        student_weight (float): Weight of the student.
        student_loss (int): Loss experienced by the student.

    Returns:
        float: Updated weight of the student.
    """
    epsilon = 0.2
    return student_weight * ((1 - epsilon) ** student_loss)

def find_bots(client, mst):
    """
    Find bots and their paths.

    Args:
        client: The client object.
        mst: Minimum spanning tree of the graph.

    Returns:
        dict: Paths of bots.
        int: Number of bots remoted home.
    """
    all_students = list(range(1, client.students + 1))
    non_home = list(range(1, client.home)) + list(range(client.home + 1, client.v + 1)) 
    losses = {student: 0 for student in all_students}
    student_weights = {student: 1 for student in all_students}
    paths = {}
    student_response = {}
    scores = {}
    bots_found = 0
    dont_remote = []

    for i in non_home:
        student_response[i] = client.scout(i, all_students)
        scores[i] = sum(student_response[i].values())

    bots_remoted_home = 0

    while bots_found < client.bots:
        if scores:
            max_vertex, _ = max(scores.items(), key=operator.itemgetter(1))
        else:
            break

        path = nx.dijkstra_path(mst, max_vertex, client.home)
        scores.pop(max_vertex)

        if path[0] not in dont_remote:
            num = client.remote(path[0], path[1])

            if num and path[1] == client.home:
                bots_remoted_home += num

            if num:
                paths[max_vertex] = path[1:]
                dont_remote.append(path[1])

            bots_found += num
            dont_remote.append(path[0])

        responses = student_response[max_vertex]

        for stud, resp in responses.items():
            if resp != num:
                losses[stud] += 1
                new_weight = update_weight(client, student_weights[stud], losses[stud]) 
                student_weights[stud] = new_weight if new_weight > 0.5 else 0

                if losses[stud] >= client.v / 2:
                    student_weights[stud] = 1

        total_weights = sum(student_weights.values())

        student_weights = {s: w / total_weights if total_weights != 0 else 0 for s, w in student_weights.items()}

        for v, resp in student_response.items():
            for stud, val in resp.items():
                if val == num and num != 0:
                    weight = student_weights[stud]
                    scores[v] += weight if val else 0

    return paths, bots_remoted_home

def solve(client):
    """
    Main function to solve the problem.

    Args:
        client: The client object.
    """
    client.end()
    client.start()
    mst = nx.minimum_spanning_tree(client.G)
    paths, bots_home = find_bots(client, client.graph)
    print("REMOTING HOME")
    remoted_on = {}
    path_lengths = {p: len(path) for p, path in paths.items()}

    while bots_home < client.bots:
        max_length = max(path_lengths.values())

        if max_length == 1:
            break

        for bot, bot_path in paths.items():
            if path_lengths[bot] == max_length:
                if bot_path[0] not in remoted_on:
                    num = client.remote(bot_path[0], bot_path[1])
                    remoted_on[bot_path[0]] = True
                    paths[bot] = bot_path[1:]
                    path_lengths[bot] -= 1

                    if num == 0:
                        break

                    if bot_path[1] == client.home:
                        bots_home += num

    print(bots_home)
    client.end()

# Assuming the existence of a 'client' object
solve(client)
