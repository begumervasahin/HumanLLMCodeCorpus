import numpy as np
import cv2

def distance(point1, point2):
    """Calculate the Euclidean distance between two points."""
    return ((point2[0] - point1[0]) ** 2 + (point2[1] - point1[1]) ** 2)

def next_position(image, bot_position, goal_position):
    """Determine the next position of the bot."""
    gray_image = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    cv2.imshow('gray', gray_image)
    
    _, threshold_image = cv2.threshold(gray_image, 75, 255, cv2.THRESH_BINARY_INV)
    height, width, _ = image.shape
    cv2.imshow('threshold', threshold_image)
    cv2.waitKey(1)

    bot_x, bot_y = bot_position
    goal_x, goal_y = goal_position
    
    feasible_squares = [(bot_x + 20, bot_y), (bot_x - 20, bot_y), (bot_x, bot_y + 20), (bot_x, bot_y - 20)]
    allowed_positions = []
    
    for square in feasible_squares:
        if 0 <= square[0] < height and 0 <= square[1] < width:
            if threshold_image[square[1], square[0]] == 255:
                allowed_positions.append(square)

    min_distance = float('inf')
    next_bot_position = bot_position
    
    for position in allowed_positions:
        d = distance(position, goal_position)
        if d < min_distance:
            min_distance = d
            next_bot_position = position

    return next_bot_position, next_bot_position[0] - bot_position[0], next_bot_position[1] - bot_position[1]
