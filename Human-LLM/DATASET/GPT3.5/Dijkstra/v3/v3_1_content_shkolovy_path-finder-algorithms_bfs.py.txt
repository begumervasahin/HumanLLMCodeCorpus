from pprint import pprint
from collections import deque
import copy
import grid_helper as gh  # Assuming grid_helper is a module with helper functions

def scan_grid(grid, start=(0, 0)):
    """
    Find all accessible points in the grid from the starting position.
    """
    visited = set()
    queue = deque([start])
    came_from = {start: None}
    
    while queue:
        current_pos = queue.popleft()
        visited.add(current_pos)
        
        for neighbor in gh.get_neighbors(grid, *current_pos):
            if neighbor not in visited:
                queue.append(neighbor)
                came_from[neighbor] = current_pos
                
    return came_from

def init():
    """
    Initialize the grid, find paths from a starting position, and display them.
    """
    # Generate the initial grid with obstacles
    initial_grid = gh.generate_grid_obstacle()
    
    # Define the starting position
    start_pos = (3, 0)
    
    # Scan the grid to find paths
    directions = scan_grid(initial_grid, start_pos)
    
    # Find a path from the start to (8, 8)
    end_pos_1 = (8, 8)
    path1 = gh.find_path(start_pos, end_pos_1, directions)
    grid_with_path1 = gh.draw_path(path1, copy.deepcopy(initial_grid))
    pprint(grid_with_path1)
    print(f"Number of steps for Path to {end_pos_1}: {len(path1)}")
    
    # Find a path from the start to (4, 7)
    end_pos_2 = (4, 7)
    path2 = gh.find_path(start_pos, end_pos_2, directions)
    grid_with_path2 = gh.draw_path(path2, copy.deepcopy(initial_grid))
    pprint(grid_with_path2)
    print(f"Number of steps for Path to {end_pos_2}: {len(path2)}")

if __name__ == "__main__":
    init()
