class Solution:
    @staticmethod
    def calculate_shortest_path(start_node, current_sum, path_sum, edge_times):
        """
        Recursively calculates the shortest path from the start node to all other nodes.
        """
        if start_node not in edge_times:
            return path_sum

        adjacent_nodes = edge_times[start_node]
        for adj_node, time in adjacent_nodes:
            new_sum = current_sum + time
            if path_sum[adj_node - 1] is None or new_sum < path_sum[adj_node - 1]:
                path_sum[adj_node - 1] = new_sum
                path_sum = Solution.calculate_shortest_path(adj_node, new_sum, path_sum, edge_times)

        return path_sum

def max_time(nodes_count, start_node, edges):
    """
    Finds the maximum time it takes to reach any node from the start node.
    """
    edge_times = {}

    # Populate the dictionary with edges and their corresponding times
    for edge in edges:
        node_from, node_to, time = edge
        if node_from not in edge_times:
            edge_times[node_from] = []
        edge_times[node_from].append((node_to, time))

    # Initialize the path sum list with None for unreachable nodes and 0 for the start node
    path_sum = [None] * nodes_count
    path_sum[start_node - 1] = 0

    # Calculate the shortest path to all nodes
    path_sum = Solution.calculate_shortest_path(start_node, 0, path_sum, edge_times)

    # If there is a single None in the path sum list, it means the particular node is unreachable by any other node
    if None not in path_sum:
        return max(path_sum)
    else:
        return -1
