import collections
import heapq

class Queue:
    """Implements a queue using collections.deque."""
    
    def __init__(self):
        """Initializes an empty queue."""
        self._elements = collections.deque()
    
    def empty(self):
        """Checks if the queue is empty."""
        return len(self._elements) == 0
    
    def add(self, item):
        """Adds an item to the queue."""
        self._elements.append(item)
    
    def pop(self):
        """Removes and returns the first item from the queue."""
        return self._elements.popleft()
    
    def __len__(self):
        """Returns the size of the queue."""
        return len(self._elements)

class PriorityQueue:
    """Implements a priority queue using heapq."""
    
    def __init__(self):
        """Initializes an empty priority queue."""
        self._elements = []
    
    def empty(self):
        """Checks if the priority queue is empty."""
        return len(self._elements) == 0
    
    def put(self, item, priority):
        """Adds an item to the priority queue with a specified priority."""
        heapq.heappush(self._elements, (priority, item))
    
    def pop(self):
        """Removes and returns the item with the lowest priority from the priority queue."""
        return heapq.heappop(self._elements)[1]
    
    def __len__(self):
        """Returns the size of the priority queue."""
        return len(self._elements)

# Example usage:
if __name__ == "__main__":
    # Create a queue and add elements
    queue = Queue()
    queue.add(1)
    queue.add(2)
    queue.add(3)
    
    # Pop elements from the queue
    print(queue.pop())  # Output: 1
    print(queue.pop())  # Output: 2
    print(queue.pop())  # Output: 3
    
    # Create a priority queue and put elements
    priority_queue = PriorityQueue()
    priority_queue.put("task1", 3)
    priority_queue.put("task2", 1)
    priority_queue.put("task3", 2)
    
    # Pop elements from the priority queue
    print(priority_queue.pop())  # Output: task2
    print(priority_queue.pop())  # Output: task3
    print(priority_queue.pop())  # Output: task1
