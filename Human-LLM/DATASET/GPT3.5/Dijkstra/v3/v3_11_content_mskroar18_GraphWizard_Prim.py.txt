class Edge:
    def __init__(self, vertex1, vertex2, weight):
        self.vertex1 = vertex1
        self.vertex2 = vertex2
        self.weight = weight
        self.selected = False
        self.extra = 0

def sort_edges_by_weight(edgelist):
    """Sorts the edges in the list by their weight."""
    return sorted(edgelist, key=lambda x: x.weight)

def run_prim(edgelist, nodecount):
    """Runs Prim's algorithm to find a minimum spanning tree."""
    # Sort the edges by weight
    edgelist = sort_edges_by_weight(edgelist)
    
    # Initialize an empty set to store selected edges
    selected_edges = set()
    
    # Select the first edge and add it to the set
    edgelist[0].selected = True
    selected_edges.add(edgelist[0])
    
    # Iterate through the remaining edges
    for edge in edgelist[1:]:
        if len(selected_edges) == nodecount - 1:
            break
        
        # Check if the edge's vertices are already in the selected set
        found_first = any(edge.vertex1 in (e.vertex1, e.vertex2) for e in selected_edges)
        found_second = any(edge.vertex2 in (e.vertex1, e.vertex2) for e in selected_edges)
        
        # Add the edge to the selected set if one of the vertices is found
        if not found_first or not found_second:
            selected_edges.add(edge)
            edge.selected = True
    
    return selected_edges
