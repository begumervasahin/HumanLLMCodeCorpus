from dijkstra import Graph

# Create an empty graph
graph = Graph()

def add_edge():
    """Add an edge to the graph."""
    node1 = input("Enter the first node: ")
    node2 = input("Enter the second node: ")
    cost = int(input("Enter the cost: "))
    graph.add_edge(node1, node2, cost)

def remove_edge():
    """Remove an edge from the graph."""
    node1 = input("Enter the first node: ")
    node2 = input("Enter the second node: ")
    graph.remove_edge(node1, node2)

def find_shortest_path():
    """Find the shortest path between two nodes using Dijkstra's algorithm."""
    source_node = input("Enter the source node: ")
    destination_node = input("Enter the destination node: ")
    shortest_path = graph.dijkstra(source_node, destination_node)
    print("Shortest path from", source_node, "to", destination_node, ":", shortest_path)

def view_graph_structure():
    """View the structure of the graph."""
    neighbors = graph.neighbours()
    print("Graph structure:")
    for node, neighbor in neighbors.items():
        print(node, "->", neighbor)

def process_user_input(choice):
    """Process user input and execute the corresponding action."""
    options = {
        "add": add_edge,
        "remove": remove_edge,
        "dijkstra": find_shortest_path,
        "view": view_graph_structure,
    }
    selected_function = options.get(choice)
    if selected_function:
        selected_function()
    else:
        print("Invalid input")

def main():
    """Main function to prompt the user for input and execute corresponding actions."""
    user_input = input("Enter one of the following options: add, remove, dijkstra, view: ")
    process_user_input(user_input)

if __name__ == "__main__":
    main()
