from queue import PriorityQueue
from pprint import pprint
import grid_helper as gh

def find_cheapest_path(grid, start, end):
    """
    Find the cheapest path from the start to the end using Dijkstra's algorithm.
    
    Args:
    - grid (list of lists): The grid representing the environment.
    - start (tuple): The starting position.
    - end (tuple): The ending position.
    
    Returns:
    - came_from (dict): A dictionary mapping positions to their predecessors in the cheapest path.
    """
    priority_queue = PriorityQueue()
    priority_queue.put((0, start))
    came_from = {start: None}
    costs = {start: 0}
    
    while not priority_queue.empty():
        current_cost, current_pos = priority_queue.get()
        
        if current_pos == end:
            break
        
        neighbors = gh.get_neighbors(grid, *current_pos)
        
        for neighbor in neighbors:
            new_cost = costs[current_pos] + gh.get_cost(grid, neighbor)
            if neighbor not in costs or new_cost < costs[neighbor]:
                costs[neighbor] = new_cost
                priority_queue.put((new_cost, neighbor))
                came_from[neighbor] = current_pos
    
    return came_from

def initialize_and_display():
    """
    Initialize the grid, find the cheapest path from start to end using Dijkstra's algorithm, 
    and display the grid with the path.
    """
    initial_grid = gh.generate_grid_weighted()
    start_position = (4, 0)
    end_position = (4, 6)
    
    came_from = find_cheapest_path(initial_grid, start_position, end_position)
    path = gh.find_path(start_position, end_position, came_from)
    grid_with_path = gh.draw_path(path, initial_grid)
    
    pprint(grid_with_path)

if __name__ == "__main__":
    initialize_and_display()
