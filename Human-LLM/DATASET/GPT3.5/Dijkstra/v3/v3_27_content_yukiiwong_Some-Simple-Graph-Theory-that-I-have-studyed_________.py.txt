def find_min_distance_node(distances, unvisited_nodes):
    """
    Finds the unvisited node with the minimum distance.
    """
    min_distance = float('inf')
    min_node = -1
    for node in unvisited_nodes:
        if distances[node] < min_distance:
            min_distance = distances[node]
            min_node = node
    return min_node

inf = float('inf')

# Adjacency matrix representing the graph
graph = [
    [0, 1, inf, 2, inf, inf],
    [inf, 0, 3, 4, inf, inf],
    [inf, inf, 0, 5, 1, inf],
    [inf, 4, inf, 0, inf, inf],
    [inf, inf, 2, 3, 0, inf],
    [inf, inf, 2, inf, 2, 0]
]

origin = 1
num_nodes = len(graph)
nodes = list(range(num_nodes))
visited = [origin]
nodes.remove(origin)

# Initialize distances dictionary with origin node
distances = {origin: 0}
for node in nodes:
    distances[node] = float('inf')

# Array to keep track of the predecessor node on the shortest path
predecessors = [0] * num_nodes

# Main loop to find shortest paths
for _ in visited:
    for node in nodes:
        new_distance = distances[_] + graph[_][node]
        if new_distance < distances[node]:
            distances[node] = new_distance
            predecessors[node] = _
    
    min_distance_node = find_min_distance_node(distances, nodes)
    if min_distance_node == -1:
        print("Unable to reach point {}".format(nodes))
        nodes.remove(nodes[0])
    else:
        visited.append(min_distance_node)
        nodes.remove(min_distance_node)
        print("%d -> %d" % (predecessors[min_distance_node], min_distance_node))
    
    if not nodes:
        break

print(distances)
