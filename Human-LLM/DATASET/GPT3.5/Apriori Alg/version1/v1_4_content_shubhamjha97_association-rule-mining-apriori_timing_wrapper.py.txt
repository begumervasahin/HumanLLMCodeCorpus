from time import time

def timeit(fn):
    def wrapper(*args, **kwargs):
        start = time()
        res = fn(*args, **kwargs)
        print(fn.__name__, "took", time() - start, "seconds.")
        return res
    return wrapper

@timeit
def some_function():
    # Placeholder function to demonstrate the timing functionality
    for i in range(1000000):
        pass

if __name__ == "__main__":
    # Call the function to test the timing functionality
    some_function()
