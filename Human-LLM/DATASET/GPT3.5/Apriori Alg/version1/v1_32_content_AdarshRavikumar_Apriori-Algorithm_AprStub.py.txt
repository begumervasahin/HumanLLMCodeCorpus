import re
from Apriori import Apriori

# Input minimum support, confidence, and length of rules
min_support = int(input("Minimum support: "))
min_confidence = float(input("Minimum Confidence: "))
min_length = int(input("Minimum Length of rules: "))

# Define alphabets for item encoding
alpha = [chr(i) for i in range(ord('a'), ord('z') + 1)]

# Read data from the file and preprocess
item = []
item1 = []
with open('datasetUCI.txt', 'r') as file:
    for line in file:
        line = line.strip()[:-1]
        line = re.sub("[?\s]", 'a', line)
        items = line.split(',')
        for item_str in items:
            if item_str in alpha:
                item1.append(item_str)
        item.append(item1)
        item1 = []

# Run Apriori algorithm
apriori_result = Apriori(item, min_support, min_confidence, min_length)
