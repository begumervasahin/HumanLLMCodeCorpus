import re
import pandas as pd

def preprocessing_frame(frame):
    """
    Preprocesses the skills data frame.
    """
    # List of specific skills to include
    specific_skills = ['c#', '.net', 'asp.net', 'c++']
    big_list = []

    for _, rows in frame.iteritems():
        # Remove unwanted characters
        rows = rows.replace('>', '').replace('<', '').replace('...', '')

        # Split the rows into individual skills
        rows_split = rows.split(',')
        list_skill = []

        # Regular expressions patterns for filtering
        pattern1 = re.compile(r'([a-z].*[0-9])')
        pattern2 = re.compile(r'([a-z].*[a-z])')

        for elements in rows_split:
            space_split = elements.split(' ')

            # Only include skills with less than or equal to 4 words
            if len(space_split) <= 4:
                elements = elements.lower().strip()

                # Filter out certain patterns and specific skills
                if (pattern1.match(elements) or pattern2.match(elements) or elements in specific_skills):
                    elements = elements.replace('?', '').replace('&amp;', '').replace('&', '').replace('#x', '').replace('#x22', '').replace('.', '')

                    # Remove trailing periods
                    if elements.endswith('.'):
                        elements = elements[:-1]

                    # Remove leading periods if it's not part of '.net'
                    if elements.startswith('.') and not elements.startswith('.net'):
                        elements = elements[1:]

                    # Add the skill to the list if it's not already present
                    if elements not in list_skill:
                        list_skill.append(elements)

        big_list.append(list_skill)

    return big_list

# Example usage:
if __name__ == "__main__":
    # Sample data frame
    data = {'skills': ['Java, Python, HTML', 'C#, C++, JavaScript, HTML']}
    df = pd.DataFrame(data)

    # Preprocess the data frame
    preprocessed_data = preprocessing_frame(df['skills'])

    # Display the preprocessed data
    print("Preprocessed Data:")
    for skills in preprocessed_data:
        print(skills)
