import time
import operator
from collections import defaultdict

def read_text(filename):
    """
    Read text from a file and return as a list of lists of words.
    """
    with open(filename) as f:
        txtlines = [line.rstrip("\n").split(" ") for line in f]
    return txtlines

def get_dict_of_freq_words(list_of_lines):
    """
    Get a dictionary of word frequencies from a list of lines.
    """
    full_word_list = defaultdict(int)
    for line in list_of_lines:
        for word in line:
            full_word_list[word] += 1
    return full_word_list

def get_patterns_given_line(line):
    """
    Get patterns for a given line.
    """
    line_len = len(line)
    patterns = set()
    for i in range(1, line_len):
        for j in range(0, line_len - i + 1):
            patterns.add(" ".join(line[j:j+i]))
    return patterns

def get_patterns_for_each_line(all_text):
    """
    Get patterns for each line in the text.
    """
    list_of_line_sets = []
    for line in all_text:
        list_of_line_sets.append(get_patterns_given_line(line))
    return list_of_line_sets

def get_phrase_freq(list_of_line_sets):
    """
    Get frequency of phrases from a list of line sets.
    """
    phrase_dict = defaultdict(int)
    for line_set in list_of_line_sets:
        if line_set is not None:
            for element in line_set:
                phrase_dict[element] += 1
    return phrase_dict

def filter_by_frequency(phrase_frequency_dict, min_sup):
    """
    Filter phrases by frequency.
    """
    return {k: v for k, v in phrase_frequency_dict.items() if v >= min_sup}

def filter_by_word_length(phrase_frequency_dict, min_length):
    """
    Filter phrases by word length.
    """
    return {k: v for k, v in phrase_frequency_dict.items() if len(k.split(" ")) >= min_length}

def print_return_file(input_dict, output_file):
    """
    Print phrases to a file sorted by frequency in descending order.
    """
    input_list = [(k, v) for k, v in input_dict.items()]
    input_list.sort(key=operator.itemgetter(1), reverse=True)
    with open(output_file, 'w') as the_file:
        for element in input_list:
            the_file.write(f"{element[1]}:{element[0].replace(' ', ';')}\n")

if __name__ == "__main__":
    times = time.time()
    txt_lines = read_text("rawdata/yelp_reviews.txt")
    word_dict = get_dict_of_freq_words(txt_lines)
    list_of_line_sets = get_patterns_for_each_line(txt_lines)
    phrases_freq = get_phrase_freq(list_of_line_sets)
    frequent_phrases = filter_by_frequency(phrases_freq, 100)
    frequent_phrases = filter_by_word_length(frequent_phrases, 2)
    print(f"Ran Algorithm for Yelp in {time.time() - times} seconds")
    print_return_file(frequent_phrases, "output/yelp_contiguous_patterns.txt")
