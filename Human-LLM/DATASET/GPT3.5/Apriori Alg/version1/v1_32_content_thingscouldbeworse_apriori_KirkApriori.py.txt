from itertools import combinations

def read_file(filename):
    results = []
    with open(filename, 'r') as fData:
        for line in fData:
            results.append(line.strip())
    return results

def single_candidates(dataset):
    results = []
    for transaction in dataset:
        items = transaction.split(',')
        for item in items:
            if item not in results:
                results.append(item)
    return results

def count_candidates(dataset, singles):
    candidate_count = {}
    for transaction in dataset:
        items = transaction.split(',')
        for item in items:
            if item in singles:
                candidate_count.setdefault(item, 0)
                candidate_count[item] += 1
    return candidate_count

def total_items(dataset):
    total = 0
    for transaction in dataset:
        items = transaction.split(',')
        total += len(items)
    return total

def support_check_count(item_count, candidate_count, support):
    result = []
    for candidate in candidate_count:
        if candidate_count[candidate] / item_count >= support:
            result.append(candidate)
    return result

def frequent_pairs(frequent_items, dataset, total, support):
    results = {}
    for transaction in dataset:
        items = transaction.split(',')
        for item in combinations(items, 2):
            results.setdefault(item, 0)
            results[item] += 1

    marked_for_deletion = []
    for key in results:
        if results[key] / total < support:
            marked_for_deletion.append(key)
        else:
            results[key] = results[key] / total

    for key in marked_for_deletion:
        results.pop(key, None)

    return results

    data_file = 'mushroom.data'
