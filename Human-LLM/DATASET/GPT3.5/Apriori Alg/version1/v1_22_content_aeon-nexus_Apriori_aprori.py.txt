import csv
import itertools

dataPath = '/home/aeon/Proj2/Apriori/INTEGRATED-DATASET2.csv'
configPath = '/home/aeon/Proj2/Apriori/Parameters2.config'

def main():
    dataSet = get_data()
    minSup = get_param()
    frequentSets = apriori(dataSet, minSup)
    print("\n Frequent Sets \n --------------")
    count = 1
    for freq in frequentSets:
        print("\n{} Frequnet Set :{}".format(count, freq))
        count += 1

def apriori(dataSet, minSup):
    frequentSet = []
    curFreqSet = []
    temp = []
    k = 1
    for transaction in dataSet:
        for item in transaction:
            count = 0
            if len(curFreqSet) == 0:
                curFreqSet.append([item, 1])
                continue
            for curItem in curFreqSet:
                if curItem[0] == item:
                    curItem[1] += 1
                    break
                else:
                    count += 1
                    if count == len(curFreqSet):
                        curFreqSet.append([item, 1])
                        break
    for item in curFreqSet:
        if item[1] >= minSup:
            temp.append(item)
    print("\n1-Frequent ItemSet:", temp)
    curFreqSet = temp
    frequentSet.append(curFreqSet)
    print("\nFrequentSet:", frequentSet)
    while curFreqSet:
        preFreqSet = list(curFreqSet)
        curFreqSet = []
        candidateSet = gen_candidates(k, preFreqSet, dataSet)
        for cand in candidateSet:
            if cand[k+1] >= minSup:
                curFreqSet.append(cand)
        if curFreqSet:
            frequentSet.append(curFreqSet)
            print("\n{} FrequentSet : {}".format(k+1, curFreqSet))
        else:
            print("\nFreqSet EMPTY")
        k += 1
    print("ALL FrequentSets :", frequentSet)
    return frequentSet

def gen_candidates(k, preFreqSet, dataSet):
    items = []
    combination = []
    preSets = []
    candidates = []
    for itemset in preFreqSet:
        sets = []
        for i in range(0, k):
            sets.append(itemset[i])
            if itemset[i] not in items:
                items.append(itemset[i])
        preSets.append(sets)
    print("\nPRESETS :", preSets)
    print("\nITEMS:", items)
    for tup in list(itertools.combinations(items, k+1)):
        combination.append(list(tup))
    print("\nCOMBINATIONS :", combination)
    for comb in combination:
        flag = 0
        for tup in list(itertools.combinations(comb, k)):
            if list(tup) not in preSets:
                flag = 1
                break
        if flag == 0:
            candidates.append(comb)
    print("\nCandidates :", candidates)
    for cand in candidates:
        supval = 0
        for transaction in dataSet:
            if set(cand).issubset(set(transaction)):
                supval += 1
        cand.append(supval)
    print("\nFinal CandidateSet :", candidates)
    return candidates

def get_data():
    Data = []
    with open(dataPath, 'r') as f:
        reader = csv.reader(f)
        for row in reader:
            Data.append(row)
    return Data

def get_param():
    with open(configPath, 'r') as f:
        minSup = int(f.readline())
    return minSup

if __name__ == "__main__":
    main()
