import itertools

# Global variables
phi = 0.0
MS = {}
temp = {}
MISSortDict = {}
IList = []
ICount = {}
TList = []
TCount = 0
L = []
FList = [[] for _ in range(10)]
CList = [[] for _ in range(10)]
CDict = {}
cannotBeTogether = []
mustHave = []
tailCount = {}


def main():
    read_data()
    init_pass()
    F1()
    Fi()
    cannot_be_together_constraint()
    if mustHave:
        must_have_constraint()
    output()


def read_data():
    global MS, TCount, IList, ICount, phi, TList, cannotBeTogether, mustHave
    with open('para1-2.txt', 'r') as file1:
        for line in file1:
            if 'SDC' in line:
                phi = float(line.replace(' ', '').rstrip().split('=')[1])

            if 'MIS' in line:
                temp = line.replace(' ', '').replace('MIS', '').replace('(', '').replace(')', '').rstrip().split('=')
                MS[int(temp[0])] = float(temp[1])

            if 'must-have' in line:
                mustHave = line.replace('must-have:', '').replace(' ', '').rstrip().split('or')
                if mustHave:
                    mustHave = list(map(int, mustHave))

            if 'cannot_be_together' in line:
                cannotBeTogether = line.replace(' ', '').replace('cannot_be_together:', '').replace('{',
                                                                                                       '[').replace('}',
                                                                                                                    ']').replace(
                    "'", 'a').rstrip().split('a')

                cannotBeTogether = cannotBeTogether[0].replace('],[', ' ').replace('[', '').replace(']', '').split()
                for i in range(len(cannotBeTogether)):
                    cannotBeTogether[i] = list(map(int, cannotBeTogether[i].split(',')))

    items = sorted(MS, key=MS.get)
    for item in items:
        IList.append(int(item))
        ICount[int(item)] = 0

    with open('data-1.txt', 'r') as file2:
        for line in file2:
            TList.append([])
            transaction = line.replace(' ', '').replace('{', '').replace('}', '').replace('<', '').replace('>',
                                                                                                              '').split(',')
            for t in transaction:
                TList[len(TList) - 1].append(int(t))
                if ICount.get(int(t)) is not None:
                    ICount[int(t)] = ICount.get(int(t)) + 1

    TCount = len(TList)


def init_pass():
    global L
    for i in range(len(IList)):
        if i == 0:
            L.append(IList[0])
        else:
            if (ICount.get(IList[i]) / TCount) >= MS.get(IList[0]):
                L.append(IList[i])

    for i in range(len(L)):
        MISSortDict[L[i]] = i


def level2_candidate_generation():
    global CList, L
    for l in range(0, len(L)):
        if (ICount[L[l]] / TCount) >= MS[L[l]]:
            for h in range(l + 1, len(L)):
                if (ICount[L[h]] / TCount) >= MS[L[l]] and abs(
                        (ICount[L[h]] / TCount) - (ICount[L[l]] / TCount)) <= phi:
                    CList[2].append([])
                    CList[2][-1].append(L[l])
                    CList[2][-1].append(L[h])
    CList[2].sort(key=lambda row: row[1])


def find_subsets(S, m):
    return list(set(itertools.combinations(S, m)))


def MS_candidate_generation(n):
    m = n - 1
    k = 0
    for i in range(0, len(FList[m])):
        for j in range(0, len(FList[m])):
            while k < m - 1 and FList[m][i][k] == FList[m][j][k]:
                k += 1

            if k == m - 1:
                if (MISSortDict[FList[m][i][k]] < MISSortDict[FList[m][j][k]]) and (
                        abs((ICount[FList[m][i][k]] / TCount) - (ICount[FList[m][j][k]] / TCount)) <= phi):
                    CList[m + 1].append(list(FList[m][i]))
                    CList[m + 1][-1].append(FList[m][j][k])
                    subset = find_subsets(CList[m + 1][-1], m)
                    for sub in range(0, len(subset)):
                        if not CList[m + 1]:
                            if bool(CList[m + 1][-1][0]) in subset[sub] or MS[
                                CList[m + 1][-1][1]] == MS[CList[m + 1][-1][0]]:
                                if bool(subset[sub]) not in FList[m]:
                                    CList[m + 1][-1].remove()

            k = 0


def F1():
    global FList
    if not FList[1]:
        for i in range(len(L)):
            if (ICount.get(L[i]) / TCount) >= MS.get(L[i]):
                FList[1].append([L[i]])


def Fi():
    global FList, CDict, tailCount
    k = 2
    while True:
        if not FList[k - 1]:
            break

        if k == 2:
            level2_candidate_generation()
        else:
            MS_candidate_generation(k)

        for t in TList:
            for c in CList[k]:
                if set(c).issubset(set(t)):
                    if CDict.get(tuple(c)) is None:
                        CDict[tuple(c)] = 1
                    else:
                        CDict[tuple(c)] = CDict.get(tuple(c)) + 1

                if set(c[1:]).issubset(set(t)):
                    if tailCount.get(tuple(c)) is None:
                        tailCount[tuple(c)] = 1
                    else:
                        tailCount[tuple(c)] = tailCount.get(tuple(c)) + 1

        for c in CList[k]:
            if CDict.get(tuple(c)) is not None:
                if CDict.get(tuple(c)) / TCount >= MS[c[0]]:
                    FList[k].append(c[:])
        k += 1


def cannot_be_together_constraint():
    global cannotBeTogether, FList
    for fListIndex in range(len(FList)):
        listIndex = len(FList[fListIndex]) - 1
        while listIndex > -1:
            subset = FList[fListIndex][listIndex]
            for i in range(len(cannotBeTogether)):
                if set(cannotBeTogether[i]) <= set(subset):
                    FList[fListIndex].pop(FList[fListIndex].index(subset))
                    break
            listIndex = listIndex - 1


def must_have_constraint():
    global mustHave, FList
    for fListIndex in range(len(FList)):
        listIndex = len(FList[fListIndex]) - 1
        while listIndex > -1:
            subset = FList[fListIndex][listIndex]
            if set(mustHave).isdisjoint(subset):
                FList[fListIndex].pop(FList[fListIndex].index(subset))
            listIndex = listIndex - 1


def output():
    global FList, tailCount, ICount, CDict
    FList = [x for x in FList if x != []]
    with open("result1-2.txt", "w") as file3:
        count = 0
        file3.write("\nFrequent 1-itemsets\n")
        for t in FList[0]:
            count += 1
            file3.write('\t' + str(ICount[t[0]]) + ' : { ' + str(t[0]) + ' }\n')
        file3.write("\tTotal number of frequent 1-itemsets = " + str(count) + "\n")

        for i in range(1, len(FList)):
            count = 0
            file3.write('\n\nFrequent ' + str(i + 1) + '-itemsets\n')
            for t in FList[i]:
                tCount = 0
                for j in range(len(TList)):
                    if set(TList[j]) >= set(t):
                        tCount += 1
                count += 1
                file3.write('\t' + str(tCount) + ' : {' + str(t).replace("[", "").replace("]", "") + '}\n')
                file3.write('Tail Count = ' + str(tailCount[tuple(t)]) + "\n")
            file3.write('\n\tTotal number of frequent ' + str(i + 1) + '-itemsets = ' + str(count) + "\n")


if __name__ == "__main__":
    main()
