import csv
from collections import defaultdict
from itertools import chain, combinations
import time

# Initialize the time
start_time = time.time()

# Dictionary to store frequent itemsets and their support
freqSet = defaultdict(int)

def data_generator(fname):
    """Function to read from the file and yield a generator"""
    with open(fname, 'r') as file:
        for line in file:
            record = frozenset(line.strip().rstrip(',').split(','))
            yield record

def transactions(data_iterator):
    """Generate transactions and unique items"""
    transaction_list = list()
    items = set()
    for record in data_iterator:
        transaction_list.append(record)
        for item in record:
            items.add(frozenset([item]))  # Generate 1-itemSets
    return items, transaction_list

def join_set(item_set, length):
    """Join sets to return n-length itemset where n=length"""
    return set([i.union(j) for i in item_set for j in item_set if len(i.union(j)) == length])

def min_support(item_list, transaction, min_supp):
    """Calculate the support for each itemset and filter based on min_support"""
    support_dict = defaultdict(int)
    item_set = set()
    for item in item_list:
        for t in transaction:
            if item.issubset(t):
                support_dict[item] += 1
    for item, v in support_dict.items():
        support = float(v) / len(transaction)
        if support >= min_supp:
            item_set.add(item)
            freqSet[item] = support
    return item_set

def subsets(items):
    """Return non-empty subsets of items"""
    return chain(*[combinations(items, i + 1) for i, a in enumerate(items)])

def run_apriori(data_iter, min_support, min_confidence):
    """Run the Apriori algorithm"""
    itemSet, transactionList = transactions(data_iter)
    k = 2
    largeSet = dict()
    # Generate frequent itemsets
    oneCSet = min_support(itemSet, transactionList, min_support)
    currentLSet = oneCSet
    while currentLSet != set([]):
        largeSet[k - 1] = currentLSet
        currentLSet = join_set(currentLSet, k)
        currentCSet = min_support(currentLSet, transactionList, min_support)
        print(k)
        print(currentCSet)
        currentLSet = currentCSet
        k = k + 1
    # Generate association rules
    for key, value in largeSet.items()[1:]:
        for item in value:
            _subsets = map(frozenset, [x for x in subsets(item)])
            for element in _subsets:
                remain = item.difference(element)
                if len(remain) > 0:
                    confidence = freqSet[item] / freqSet[element]
                    if confidence >= min_confidence:
                        print(element, "=>", remain, "Confidence:", confidence)

if __name__ == "__main__":
    minSupport = 0.01
    minConfidence = 0.01
    inFile = data_generator('T10I4D100k-copy.txt')
    run_apriori(inFile, minSupport, minConfidence)

    # Calculate and print the elapsed time
    elapsed_time = time.time() - start_time
    print("Execution time:", elapsed_time)
