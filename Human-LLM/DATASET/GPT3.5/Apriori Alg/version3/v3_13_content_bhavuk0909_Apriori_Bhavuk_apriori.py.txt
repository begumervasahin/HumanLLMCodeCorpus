# Importing necessary libraries
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from apyori import apriori

def preprocess_dataset(file_path):
    """
    Read the dataset from a CSV file and preprocess it.
    """
    # Read the dataset
    dataset = pd.read_csv(file_path, header=None)
    transactions = []

    # Iterate through each row in the dataset
    for i in range(len(dataset)):
        # Create a list of items in each transaction and convert them to strings
        transaction_items = [str(dataset.values[i, j]) for j in range(len(dataset.columns))]
        # Append the transaction to the list of transactions
        transactions.append(transaction_items)

    return transactions

def find_association_rules(transactions, min_support=0.003, min_confidence=0.2, min_lift=3, min_length=2):
    """
    Apply the Apriori algorithm to find association rules.
    """
    # Apply Apriori algorithm
    rules = apriori(transactions, min_support=min_support, min_confidence=min_confidence,
                    min_lift=min_lift, min_length=min_length)
    return list(rules)

def print_association_rules(rules):
    """
    Print out each discovered association rule.
    """
    for rule in rules:
        print(rule)

def main():
    # Specify the file path for the dataset
    file_path = 'Market_Basket_Optimisation.csv'
    
    # Preprocess the dataset
    transactions = preprocess_dataset(file_path)

    # Find association rules
    rules = find_association_rules(transactions)

    # Print out the discovered association rules
    print_association_rules(rules)

if __name__ == "__main__":
    main()
