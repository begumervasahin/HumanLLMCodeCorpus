# Importing the required libraries
import pandas as pd
from apyori import apriori

# Data Preprocessing
dataset = pd.read_csv('Market_Basket_Optimisation.csv', header=None)
transactions = []

# Converting the dataset into a list of transactions
for _, row in dataset.iterrows():
    transactions.append([str(item) for item in row])

# Training Apriori on the dataset
min_support = 0.003
min_confidence = 0.2
min_lift = 3
min_length = 2
rules = apriori(transactions, min_support=min_support, min_confidence=min_confidence, min_lift=min_lift, min_length=min_length)
results = list(rules)

# Printing the results of the Apriori algorithm
print("Apriori Algorithm Results:")
for result in results:
    print(result)
