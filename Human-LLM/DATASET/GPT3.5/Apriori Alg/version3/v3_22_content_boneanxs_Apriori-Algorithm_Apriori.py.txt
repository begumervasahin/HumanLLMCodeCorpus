import itertools

def get_subsets(item_set):
    """
    Generate all possible subsets of an item set.
    """
    subset_length = len(item_set) - 1
    return [list(subset) for subset in itertools.combinations(item_set, subset_length)]

def is_subset_exist(transaction_set, candidate_set):
    """
    Check if each candidate set is a subset of any transaction set.
    """
    transaction_set = [set(transaction[0]) for transaction in transaction_set]
    for candidate in candidate_set:
        is_in = any(candidate_set <= transaction for transaction in transaction_set)
        if not is_in:
            return False
    return True

def apriori(transactions, min_support, max_length):
    """
    Implement the Apriori algorithm for frequent item set mining.
    """
    # Count occurrences of each item
    item_frequencies = {}
    for transaction in transactions.values():
        for item in transaction:
            item_frequencies[item] = item_frequencies.get(item, 0) + 1

    # Calculate support for each item
    total_transactions = len(transactions)
    frequent_items = [[item, support / total_transactions] for item, support in item_frequencies.items() if support / total_transactions >= min_support]

    if max_length == 1:
        return frequent_items

    current_length = 2
    while current_length <= max_length:
        candidate_sets = []
        for i in range(len(frequent_items)):
            for j in range(i + 1, len(frequent_items)):
                new_set = list(set(frequent_items[i][0]) | set(frequent_items[j][0]))
                if len(new_set) == current_length and is_subset_exist(transactions.items(), get_subsets(new_set)):
                    candidate_sets.append(new_set)

        frequent_items = []
        for candidate_set in candidate_sets:
            support_count = sum(1 for transaction in transactions.values() if set(candidate_set).issubset(transaction))
            support = support_count / total_transactions
            if support >= min_support:
                frequent_items.append([candidate_set, support])

        current_length += 1

    return frequent_items
