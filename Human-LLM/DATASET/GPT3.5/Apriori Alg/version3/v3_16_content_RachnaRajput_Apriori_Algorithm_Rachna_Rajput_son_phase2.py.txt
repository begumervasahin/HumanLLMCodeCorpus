import json
import sys
from collections import defaultdict

class MapReduce:
    def __init__(self):
        self.intermediate = defaultdict(list)
        self.result = []

    def emit_intermediate(self, key, value):
        self.intermediate[key].append(value)

    def emit(self, value):
        self.result.append(value)

    def execute(self, data, mapper, reducer):
        # Mapping phase
        for line in data:
            record = json.loads(line)
            mapper(record)

        # Reducing phase
        for key in self.intermediate:
            reducer(key, self.intermediate[key])

        # Printing the results
        for item in self.result:
            print(item)

def mapper(record):
    global mr
    with open(sys.argv[1]) as f:
        mylist = []
        for line in f:
            chunkdata = json.loads(line)
            count = sum(1 for chunkline in chunkdata if set(chunkline) >= set(record))
            length = len(chunkdata)
            mylist.extend([count, length])
        recordtuple = tuple(record)
        mr.emit_intermediate(recordtuple, mylist)

def reducer(key, value):
    global mr
    count = sum(value[0][i] for i in range(0, len(value[0]), 2))
    mylist = [key, count]
    mr.emit(mylist)

if __name__ == '__main__':
    mr = MapReduce()
    with open(sys.argv[2]) as inputdata:
        output = mr.execute(inputdata, mapper, reducer)
