from pymongo import MongoClient
from geopy.geocoders import Nominatim
import pandas as pd

class RestaurantAnalyzer:
    def __init__(self, db_name='restaurantinfo'):
        self.db_name = db_name
        self.client = MongoClient('mongodb://localhost:27017/' + db_name)
        self.db = self.client[self.db_name]

        # Define field names for different types of restaurant data
        self.info_fields = ['restaurant name', 'latitude', 'longitude', 'street', 'city', 'country', 'rating', 'price', 
                            'review excellent count', 'review good count', 'review avg count', 'review poor count', 
                            'review terrible count', 'positive review count', 'negative review count', 'total reviews']
        self.cuisine_fields = ['restaurant name', 'cuisine']
        self.features_fields = ['restaurant name', 'feature']
        self.meals_fields = ['restaurant name', 'meal']

    def convert_address_to_coordinates(self, address):
        # Convert address to latitude and longitude coordinates
        geolocator = Nominatim(user_agent="restaurant-finder")
        location = geolocator.geocode(address, timeout=3)
        if location:
            return location.latitude, location.longitude
        else:
            return 0, 0

    def load_mongodb_to_pandas(self):
        # Initialize lists to store restaurant data
        restaurant_info = []
        restaurant_cuisines = []
        restaurant_features = []
        restaurant_meals = []

        # Extract data from MongoDB and process it
        for doc in self.db.restaurantreviews.find():
            street = doc['rest_street']
            city = doc['rest_city']
            country = doc['rest_country']
            lat, lon = self.convert_address_to_coordinates(f"{street}, {city}, {country}")

            if lat != 0 and lon != 0 and doc['rest_cuisines'] and doc['rest_features'] and doc['rest_meals']:
                # Calculate review counts
                positive_review_count = int(doc['review_excellent_count']) + int(doc['review_good_count'])
                negative_review_count = int(doc['review_avg_count']) + int(doc['review_poor_count']) + int(doc['review_terrible_count'])

                # Categorize price
                price = doc['rest_price'].replace(" - ", "")
                if len(price) == 1:
                    rest_price = "AFFORDABLE"
                elif len(price) == 4:
                    rest_price = "MODERATE"
                else:
                    rest_price = "EXPENSIVE"

                # Append restaurant info to the list
                restaurant_info.append([doc['rest_name'], float(lat), float(lon), doc['rest_street'], doc['rest_city'], 
                                        doc['rest_country'], doc['rest_rating'], rest_price, doc['review_excellent_count'], 
                                        doc['review_good_count'], doc['review_avg_count'], doc['review_poor_count'], 
                                        doc['review_terrible_count'], positive_review_count, negative_review_count, 
                                        doc['rest_total_reviews']])

                # Append restaurant cuisines to the list
                for cuisine in doc['rest_cuisines'].split(','):
                    restaurant_cuisines.append([doc['rest_name'], cuisine.strip()])

                # Append restaurant features to the list
                for feature in doc['rest_features'].split(','):
                    restaurant_features.append([doc['rest_name'], feature.strip()])

                # Append restaurant meals to the list
                for meal in doc['rest_meals'].split(','):
                    restaurant_meals.append([doc['rest_name'], meal.strip()])

        # Create DataFrames from the lists
        self.info_df = pd.DataFrame(restaurant_info, columns=self.info_fields)
        self.cuisines_df = pd.DataFrame(restaurant_cuisines, columns=self.cuisine_fields)
        self.features_df = pd.DataFrame(restaurant_features, columns=self.features_fields)
        self.meals_df = pd.DataFrame(restaurant_meals, columns=self.meals_fields)

        # Write DataFrames to CSV files
        self.info_df.to_csv('data/restaurant_info.csv', index=False)
        self.cuisines_df.to_csv('data/cuisines_info.csv', index=False)
        self.features_df.to_csv('data/features_info.csv', index=False)
        self.meals_df.to_csv('data/meals_info.csv', index=False)

if __name__ == '__main__':
    # Instantiate the RestaurantAnalyzer class and load MongoDB data to Pandas DataFrames
    analyzer = RestaurantAnalyzer()
    analyzer.load_mongodb_to_pandas()
