# Importing necessary libraries
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from apyori import apriori

# Data Preprocessing
def load_dataset(file_path):
    """Load dataset from CSV file."""
    return pd.read_csv(file_path, header=None)

def preprocess_data(dataset):
    """Convert dataset into a list of transactions."""
    transactions = []
    for _, row in dataset.iterrows():
        transaction = [str(item) for item in row if pd.notna(item)]
        transactions.append(transaction)
    return transactions

# Main functionality
def discover_association_rules(transactions, min_support=0.003, min_confidence=0.2, min_lift=3, min_length=2):
    """Discover association rules using Apriori algorithm."""
    rules = apriori(transactions, min_support=min_support, min_confidence=min_confidence, min_lift=min_lift, min_length=min_length)
    return list(rules)

def print_association_rules(rules):
    """Print the discovered association rules."""
    for rule in rules:
        print(rule)

if __name__ == "__main__":
    # Load dataset
    dataset = load_dataset('Market_Basket_Optimisation.csv')

    # Preprocess the data
    transactions = preprocess_data(dataset)

    # Discover association rules
    rules = discover_association_rules(transactions)

    # Print the discovered association rules
    print_association_rules(rules)
