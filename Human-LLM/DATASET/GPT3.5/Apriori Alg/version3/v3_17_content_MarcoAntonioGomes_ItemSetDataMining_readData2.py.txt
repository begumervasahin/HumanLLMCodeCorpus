import csv

# Define file paths
TRANSACTIONS_FILE_PATH = 'caixa.csv'
PRODUCTS_FILE_PATH = 'produtos.csv'
OUTPUT_FILE_PATH = 'dataCaixaAno2015-MesAgosto.dat'

# Open the output file for writing transactions
output_file = open(OUTPUT_FILE_PATH, 'w')

# Initialize lists to store data
transactions = []
products = []
transactions_with_products = []

# Function to load data from CSV files
def load_csv_data():
    # Read data from 'caixa.csv'
    with open(TRANSACTIONS_FILE_PATH, newline='') as transactions_file:
        transactions_reader = csv.reader(transactions_file, delimiter=';')
        next(transactions_reader)  # Skip header
        for line in transactions_reader:
            transactions.append([line[6]])  # Extract product codes

    # Read data from 'produtos.csv'
    with open(PRODUCTS_FILE_PATH, newline='') as products_file:
        products_reader = csv.reader(products_file, delimiter=';')
        next(products_reader)  # Skip header
        for line in products_reader:
            products.append((line[0], line[1]))  # Extract product codes and names

    # Match product codes to their names
    for transaction in transactions:
        for product_code in transaction[1:]:
            for product in products:
                if product_code == product[0]:
                    transactions_with_products.append(product[1])

    return transactions_with_products

# Load data and print transactions
transactions_with_products = load_csv_data()
print(transactions_with_products)

# Write transactions to the output file
for transaction in transactions_with_products:
    if len(transaction) > 2:
        transaction_line = ' '.join([product_name.replace(" ", "_") for product_name in transaction[1:]]) + '\n'
        output_file.write(transaction_line)

# Close the output file
output_file.close()
