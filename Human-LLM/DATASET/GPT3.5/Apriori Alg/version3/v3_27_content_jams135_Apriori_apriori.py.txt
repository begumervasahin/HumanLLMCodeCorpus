import sys
from itertools import combinations

# Function to count occurrences of itemsets in transactions
def count_itemsets(itemsets, transactions):
    counts = []
    for itemset in itemsets:
        count = sum(1 for transaction in transactions if set(itemset).issubset(set(transaction)))
        counts.append(count)
    return counts

# Function to prune infrequent itemsets
def prune_itemsets(itemsets, counts, min_support):
    pruned_itemsets = [itemset for itemset, count in zip(itemsets, counts) if count >= min_support]
    return pruned_itemsets

# Minimum support threshold
min_support = int(input("\nEnter minimum support: "))

# List of transactions
transactions = [
    ["Mango", "Onion", "Jar", "KeyChain", "Eggs", "Chocolates"],
    ["Nuts", "Onion", "Jar", "KeyChain", "Eggs", "Chocolates"],
    ["Mango", "Apple", "KeyChain", "Eggs"],
    ["Mango", "Toothbrush", "Corn", "KeyChain", "Chocolates"],
    ["Corn", "Onion", "Onion", "KeyChain", "Knife", "Eggs"]
]

print("\n\n=======================Transactions=======================\n")
for transaction in transactions:
    print(" ".join(transaction))

# Step 1: Find frequent itemsets of size 1
unique_items = list(set(item for transaction in transactions for item in transaction))

print("\n\n1 frequent itemset (Unique items):\n")
print(unique_items)

# Count occurrences of each item in the dataset
counts_1_itemset = [sum(1 for transaction in transactions if item in transaction) for item in unique_items]

print("\n\ncount for 1-itemset\n")
for item, count in zip(unique_items, counts_1_itemset):
    print(f"{item} \t\t {count}")

# Prune infrequent itemsets
frequent_1_itemset = prune_itemsets(unique_items, counts_1_itemset, min_support)

print("\n\nPrune 1 frequent itemset:")
print(f"\nAfter pruning, we get frequent 1-itemset: \n{frequent_1_itemset}")

# Step 2: Generate candidate pairs (size 2)
pairs_2_itemset = list(combinations(frequent_1_itemset, 2))

print("\n\n2 frequent itemset:\n")
print(pairs_2_itemset)

# Count occurrences of each pair in the dataset
counts_2_itemset = count_itemsets(pairs_2_itemset, transactions)

print("\n\ncount for 2-itemset\n")
for pair, count in zip(pairs_2_itemset, counts_2_itemset):
    print(f"{pair} \t\t {count}")

# Prune infrequent itemsets
frequent_2_itemset = prune_itemsets(pairs_2_itemset, counts_2_itemset, min_support)

print("\n\nPrune 2 frequent itemset:")
print(f"\nAfter pruning, we get frequent 2-itemset: \n{frequent_2_itemset}")
