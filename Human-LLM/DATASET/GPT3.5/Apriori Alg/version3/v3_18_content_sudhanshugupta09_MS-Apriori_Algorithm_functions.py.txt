from operator import itemgetter

# Function to calculate M (sorted list of MIS values)
def calculate_sorted_MIS(MIS):
    sorted_MIS = sorted(MIS.items(), key=itemgetter(1))  # Sort MIS values based on support count
    return sorted_MIS

# Function to create a list of items
def create_item_list(MIS):
    items = []
    for item, _ in MIS.items():
        items.append(item)
    return items

# Initial pass over the items list
def perform_initial_pass(MIS_sorted, MIS, support_count, min_mis_required):
    L = []
    min_mis_required_value = None
    for item, mis_value in MIS_sorted:
        if support_count[item] >= MIS[item] and min_mis_required_value is None:
            L.append(item)
            min_mis_required_value = MIS[item]
        elif min_mis_required_value is not None and support_count[item] >= min_mis_required_value:
            L.append(item)
    return L

# Check constraints for items that cannot be together
def remove_cannot_be_together_constraints(F, cannot_be_together):
    filtered_F = []
    for frequent_set in F:
        if not any(set(c).issubset(frequent_set) for c in cannot_be_together):
            filtered_F.append(frequent_set)
    return filtered_F

# Check if must-have items are included in frequent itemsets
def include_must_have_items(F, must_have):
    filtered_F = []
    for frequent_set in F:
        if any(item in frequent_set for item in must_have):
            filtered_F.append(frequent_set)
    return filtered_F

# Sample Transactional Data
transactions = [
    {20, 30, 80, 70, 50, 90},
    {20, 10, 80, 70},
    {10, 20, 80},
    {20, 30, 80},
    {20, 80},
    {20, 30, 80, 70, 50, 90, 100, 120, 140}
]

# Sample Parameter File
MIS_values = {
    10: 0.43,
    20: 0.30,
    30: 0.30,
    40: 0.40,
    50: 0.40,
    60: 0.30,
    70: 0.20,
    80: 0.20,
    90: 0.20,
    100: 0.10,
    120: 0.20,
    140: 0.15
}

# Other Parameters
SDC = 0.1
cannot_be_together_constraints = [{20, 40}, {70, 80}]
must_have_items = {20, 40, 50}

# Perform the functions
MIS_sorted = calculate_sorted_MIS(MIS_values)
I = create_item_list(MIS_values)
support_count = {item: sum(1 for t in transactions if item in t) for item in I}
L = perform_initial_pass(MIS_sorted, MIS_values, support_count, SDC)
F = [L]

F = remove_cannot_be_together_constraints(F, cannot_be_together_constraints)
F = include_must_have_items(F, must_have_items)

# Printing the frequent itemsets
for i, freq_itemset in enumerate(F):
    print(f"Frequent {i + 1}-itemsets")
    for itemset in freq_itemset:
        print(f"    {len([t for t in transactions if itemset.issubset(t)])} : {itemset}")
    print(f"Total number of frequent {i + 1}-itemsets = {len(freq_itemset)}\n")
