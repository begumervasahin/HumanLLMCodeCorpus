import re
from Apriori import Apriori

def get_user_input(prompt):
    """Get user input for minimum support, confidence, and length of rules."""
    return int(input(prompt))

def preprocess_data(file_path):
    """Read data from the file, preprocess, and return the list of transaction items."""
    alphabet = [chr(i) for i in range(ord('a'), ord('z') + 1)]
    items_list = []

    with open(file_path, 'r') as file:
        for line in file:
            line = line.strip()[:-1]  # Remove trailing comma
            line = re.sub("[?\s]", 'a', line)  # Replace special characters and spaces with 'a'
            items = line.split(',')  # Split the line into items
            transaction_items = [item_str for item_str in items if item_str in alphabet]
            items_list.append(transaction_items)

    return items_list

def run_apriori(min_support, min_confidence, min_length, data):
    """Run the Apriori algorithm with given parameters and data."""
    apriori_result = Apriori(data, min_support, min_confidence, min_length)
    return apriori_result

if __name__ == "__main__":
    # Get user input for minimum support, confidence, and length of rules
    min_support = get_user_input("Enter the minimum support: ")
    min_confidence = float(get_user_input("Enter the minimum confidence: "))
    min_length = get_user_input("Enter the minimum length of rules: ")

    # Preprocess the data
    data_path = 'datasetUCI.txt'
    items_list = preprocess_data(data_path)

    # Run Apriori algorithm
    apriori_result = run_apriori(min_support, min_confidence, min_length, items_list)
