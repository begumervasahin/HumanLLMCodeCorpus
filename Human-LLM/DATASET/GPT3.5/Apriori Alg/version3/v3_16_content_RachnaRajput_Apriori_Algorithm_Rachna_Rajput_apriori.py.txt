import json
import sys

class AprioriAlgorithm:
    def __init__(self):
        self.json_data = []
        self.threshold = 0

    def load_json_data(self, file_path):
        with open(file_path) as json_file:
            self.json_data = json.load(json_file)

    def apriori(self, show_output=False):
        if not self.json_data:
            print("No JSON data loaded.")
            return []

        self.threshold = 0.3 * len(self.json_data)
        phase1_list = []

        unique_items = self._generate_unique_items()
        if show_output:
            print("C1:", unique_items)

        item_counts = self._count_item_occurrences()
        frequent_items = self._generate_frequent_itemsets(unique_items, item_counts)
        phase1_list.extend(frequent_items)

        candidate_sets = self._generate_candidate_sets(frequent_items)
        while candidate_sets:
            if show_output:
                print(f"C{len(phase1_list) + 1}:", candidate_sets)

            item_counts = self._count_item_occurrences(candidate_sets)
            frequent_items = self._generate_frequent_itemsets(candidate_sets, item_counts)
            phase1_list.extend(frequent_items)

            candidate_sets = self._generate_candidate_sets(frequent_items)

        return phase1_list

    def _generate_unique_items(self):
        unique_items = []
        for item_set in self.json_data:
            unique_items = list(set(unique_items) | set(item_set))
        return [[item] for item in unique_items]

    def _count_item_occurrences(self, item_sets=None):
        if item_sets is None:
            item_sets = self.json_data

        item_counts = {}
        for item_set in item_sets:
            for item in item_set:
                if item not in item_counts:
                    item_counts[item] = 1
                else:
                    item_counts[item] += 1
        return item_counts

    def _generate_frequent_itemsets(self, candidate_sets, item_counts):
        frequent_items = []
        for candidate_set in candidate_sets:
            if item_counts[candidate_set[0]] >= self.threshold:
                frequent_items.append(candidate_set)
        return frequent_items

    def _generate_candidate_sets(self, frequent_items):
        candidate_sets = []
        for i in range(len(frequent_items)):
            for j in range(i + 1, len(frequent_items)):
                if frequent_items[i][0] == frequent_items[j][0]:
                    new_candidate_set = frequent_items[i][1:] + frequent_items[j][1:]
                    new_candidate_set.sort()
                    if new_candidate_set not in candidate_sets:
                        candidate_sets.append(new_candidate_set)
        return candidate_sets

if __name__ == '__main__':
    if len(sys.argv) < 2:
        print("Usage: python script_name.py <json_file_path>")
        sys.exit(1)

    apriori_instance = AprioriAlgorithm()
    apriori_instance.load_json_data(sys.argv[1])
    checker = True
    apriori_instance.apriori(checker)
