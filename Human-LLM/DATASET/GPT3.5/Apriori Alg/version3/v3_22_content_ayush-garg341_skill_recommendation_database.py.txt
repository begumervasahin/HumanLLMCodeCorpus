import csv
from itertools import combinations

class DataBase:
    def __init__(self, data_path):
        # Initialize the database with the path to the skills data file
        self.data_path = data_path
        self.skills_data = self.read_skills_data()

    def read_skills_data(self):
        """
        Read skills data from a CSV file and return it as a list of lists.
        Each inner list represents a transaction containing skills.
        """
        skills_data = []
        with open(self.data_path, 'r') as file:
            reader = csv.reader(file)
            for row in reader:
                skills_data.append(row)
        return skills_data

    def apriori(self, min_support, min_confidence):
        """
        Apply the Apriori algorithm to find frequent patterns and related skills.
        """
        frequent_patterns = self.get_frequent_patterns(min_support)
        association_rules = self.generate_association_rules(frequent_patterns, min_confidence)
        return association_rules

    def get_frequent_patterns(self, min_support):
        """
        Generate frequent patterns based on minimum support.
        """
        skill_counts = {}
        for transaction in self.skills_data:
            for skill in transaction:
                skill_counts[skill] = skill_counts.get(skill, 0) + 1

        frequent_patterns = {skill: support for skill, support in skill_counts.items() if support >= min_support}

        return frequent_patterns

    def generate_association_rules(self, frequent_patterns, min_confidence):
        """
        Generate association rules based on frequent patterns and minimum confidence.
        """
        association_rules = []
        pattern_combinations = list(combinations(frequent_patterns.keys(), 2))

        for pattern in pattern_combinations:
            antecedent, consequent = pattern
            antecedent_support = frequent_patterns[antecedent]
            combined_support = frequent_patterns.get(f"{antecedent},{consequent}", 0)

            if combined_support > 0:
                confidence = combined_support / antecedent_support
                if confidence >= min_confidence:
                    association_rules.append((antecedent, consequent, confidence))

        return association_rules

if __name__ == "__main__":
    # Initialize the database with the path to the skills data file
    db = DataBase('skills_data.csv')

    # Set minimum support and minimum confidence
    min_support = 50
    min_confidence = 0.7

    # Apply Apriori algorithm
    association_rules = db.apriori(min_support, min_confidence)

    # Display association rules
    print("Association Rules:")
    for rule in association_rules:
        print(f"{rule[0]} -> {rule[1]}: Confidence = {rule[2]}")
