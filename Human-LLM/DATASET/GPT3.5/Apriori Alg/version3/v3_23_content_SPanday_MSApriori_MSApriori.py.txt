import itertools

# Global variables
support_difference_constraint = 0.0
minimum_support = {}
item_list = []
item_counts = {}
transactions = []
transaction_count = 0
frequent_item_sets = [[] for _ in range(10)]
candidate_item_sets = [[] for _ in range(10)]
item_set_counts = {}
incompatible_sets = []
required_items = []
tail_item_counts = {}


def main():
    read_data()
    initialize_pass()
    find_frequent_1_item_sets()
    find_frequent_item_sets()
    apply_incompatible_sets_constraint()
    if required_items:
        apply_required_items_constraint()
    write_output()


def read_data():
    global support_difference_constraint, minimum_support, transactions, incompatible_sets, required_items
    global item_list, item_counts, transaction_count
    
    with open('para1-2.txt', 'r') as file1:
        for line in file1:
            if 'SDC' in line:
                support_difference_constraint = float(line.replace(' ', '').rstrip().split('=')[1])

            if 'MIS' in line:
                temp = line.replace(' ', '').replace('MIS', '').replace('(', '').replace(')', '').rstrip().split('=')
                minimum_support[int(temp[0])] = float(temp[1])

            if 'must-have' in line:
                required_items = line.replace('must-have:', '').replace(' ', '').rstrip().split('or')
                if required_items:
                    required_items = list(map(int, required_items))

            if 'cannot_be_together' in line:
                incompatible_sets = line.replace(' ', '').replace('cannot_be_together:', '').replace('{', '[').replace(
                    '}', ']').replace("'", 'a').rstrip().split('a')
                incompatible_sets = incompatible_sets[0].replace('],[', ' ').replace('[', '').replace(']', '').split()
                incompatible_sets = [list(map(int, subset.split(','))) for subset in incompatible_sets]

    items = sorted(minimum_support, key=minimum_support.get)
    for item in items:
        item_list.append(int(item))
        item_counts[int(item)] = 0

    with open('data-1.txt', 'r') as file2:
        for line in file2:
            transaction = line.replace(' ', '').replace('{', '').replace('}', '').replace('<', '').replace('>', '').split(',')
            transactions.append([int(t) for t in transaction])
            for t in transaction:
                item_counts[int(t)] = item_counts.get(int(t), 0) + 1

    transaction_count = len(transactions)


def initialize_pass():
    global item_list, transaction_count, item_counts, minimum_support, frequent_item_sets
    for i in range(len(item_list)):
        if i == 0:
            frequent_item_sets.append(item_list[0])
        elif item_counts[item_list[i]] / transaction_count >= minimum_support[item_list[0]]:
            frequent_item_sets.append(item_list[i])

    for i, item in enumerate(frequent_item_sets):
        MISSortDict[item] = i


def level2_candidate_generation():
    global candidate_item_sets, frequent_item_sets, item_counts, transaction_count
    for l in range(len(frequent_item_sets)):
        for h in range(l + 1, len(frequent_item_sets)):
            if (item_counts[frequent_item_sets[h]] / transaction_count >= minimum_support[frequent_item_sets[l]] and
                    abs((item_counts[frequent_item_sets[h]] / transaction_count) -
                        (item_counts[frequent_item_sets[l]] / transaction_count)) <= support_difference_constraint):
                candidate_item_sets[2].append([frequent_item_sets[l], frequent_item_sets[h]])

    candidate_item_sets[2].sort(key=lambda row: row[1])


def find_subsets(S, m):
    return list(set(itertools.combinations(S, m)))


def MS_candidate_generation(n):
    m = n - 1
    for i in range(len(frequent_item_sets[m])):
        for j in range(i + 1, len(frequent_item_sets[m])):
            k = 0
            while k < m - 1 and frequent_item_sets[m][i][k] == frequent_item_sets[m][j][k]:
                k += 1

            if k == m - 1 and MISSortDict[frequent_item_sets[m][i][k]] < MISSortDict[frequent_item_sets[m][j][k]] and \
                    abs((item_counts[frequent_item_sets[m][i][k]] / transaction_count) -
                        (item_counts[frequent_item_sets[m][j][k]] / transaction_count)) <= support_difference_constraint:
                candidate_item_sets[m + 1].append(frequent_item_sets[m][i] + [frequent_item_sets[m][j][k]])
                subset = find_subsets(candidate_item_sets[m + 1][-1], m)
                if not candidate_item_sets[m + 1]:
                    if (bool(candidate_item_sets[m + 1][-1][0]) in subset[sub] or
                            minimum_support[candidate_item_sets[m + 1][-1][1]] == minimum_support[
                                candidate_item_sets[m + 1][-1][0]]):
                        if bool(subset[sub]) not in frequent_item_sets[m]:
                            candidate_item_sets[m + 1][-1].remove()

                k = 0


def find_frequent_1_item_sets():
    global frequent_item_sets, item_counts, transaction_count
    if not frequent_item_sets[1]:
        for item in item_list:
            if item_counts.get(item, 0) / transaction_count >= minimum_support.get(item, 0):
                frequent_item_sets[1].append([item])


def find_frequent_item_sets():
    global frequent_item_sets, item_set_counts, tail_item_counts, transaction_count
    k = 2
    while frequent_item_sets[k - 1]:
        if k == 2:
            level2_candidate_generation()
        else:
            MS_candidate_generation(k)

        for t in transactions:
            for c in candidate_item_sets[k]:
                if set(c).issubset(set(t)):
                    item_set_counts[tuple(c)] = item_set_counts.get(tuple(c), 0) + 1

                if set(c[1:]).issubset(set(t)):
                    tail_item_counts[tuple(c)] = tail_item_counts.get(tuple(c), 0) + 1

        for c in candidate_item_sets[k]:
            if item_set_counts.get(tuple(c), 0) / transaction_count >= minimum_support[c[0]]:
                frequent_item_sets[k].append(c[:])
        k += 1


def apply_incompatible_sets_constraint():
    global incompatible_sets, frequent_item_sets
    for fListIndex in range(len(frequent_item_sets)):
        listIndex = len(frequent_item_sets[fListIndex]) - 1
        while listIndex > -1:
            subset = frequent_item_sets[fListIndex][listIndex]
            for incompatible_set in incompatible_sets:
                if set(incompatible_set) <= set(subset):
                    frequent_item_sets[fListIndex].pop(listIndex)
                    break
            listIndex -= 1


def apply_required_items_constraint():
    global required_items, frequent_item_sets
    for fListIndex in range(len(frequent_item_sets)):
        listIndex = len(frequent_item_sets[fListIndex]) - 1
        while listIndex > -1:
            subset = frequent_item_sets[fListIndex][listIndex]
            if set(required_items).isdisjoint(subset):
                frequent_item_sets[fListIndex].pop(listIndex)
            listIndex -= 1


def write_output():
    global frequent_item_sets, tail_item_counts, item_counts, transaction_count
    frequent_item_sets = [x for x in frequent_item_sets if x]
    with open("result1-2.txt", "w") as file3:
        count = 0
        file3.write("\nFrequent 1-itemsets\n")
        for t in frequent_item_sets[0]:
            count += 1
            file3.write('\t' + str(item_counts[t[0]]) + ' : { ' + str(t[0]) + ' }\n')
        file3.write("\tTotal number of frequent 1-itemsets = " + str(count) + "\n")

        for i in range(1, len(frequent_item_sets)):
            count = 0
            file3.write('\n\nFrequent ' + str(i + 1) + '-itemsets\n')
            for t in frequent_item_sets[i]:
                tCount = sum(1 for j in transactions if set(j) >= set(t))
                count += 1
                file3.write('\t' + str(tCount) + ' : {' + str(t).replace("[", "").replace("]", "") + '}\n')
                file3.write('Tail Count = ' + str(tail_item_counts.get(tuple(t), 0)) + "\n")
            file3.write('\n\tTotal number of frequent ' + str(i + 1) + '-itemsets = ' + str(count) + "\n")


if __name__ == "__main__":
    main()
