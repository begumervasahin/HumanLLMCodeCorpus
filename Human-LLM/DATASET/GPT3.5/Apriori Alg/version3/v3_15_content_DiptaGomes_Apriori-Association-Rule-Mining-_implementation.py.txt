# Importing the required libraries
import pandas as pd
from apyori import apriori

# Read the dataset from the CSV file
dataset = pd.read_csv('Nursery.csv', header=1)

# Create a list to store transactions
transactions = []

# Extract transactions from the dataset
for i in range(1, 9409):
    transaction = [str(dataset.values[i, j]) for j in range(0, 9)]
    transactions.append(transaction)

# Apply Apriori algorithm to discover association rules
min_support = 0.04
min_confidence = 0.90
min_lift = 3
min_length = 2
association_rules = apriori(transactions,
                            min_support=min_support,
                            min_confidence=min_confidence,
                            min_lift=min_lift,
                            min_length=min_length)

# Convert the association rules to a list for iteration
rules_list = list(association_rules)

# Iterate through the discovered association rules and print them
for rule in rules_list:
    # Extract the antecedent and consequent of the rule
    antecedent, consequent = list(rule[0]), list(rule[1])

    # Print the association rule
    print(f"Rule: {', '.join(antecedent)} -> {', '.join(consequent)}")

    # Print the support of the rule
    print(f"Support: {rule[1]}")

    # Print the confidence of the rule
    confidence = rule[2][0][2]
    print(f"Confidence: {confidence}")

    # Print the lift of the rule
    lift = rule[2][0][3]
    print(f"Lift: {lift}")

    # Print separator for clarity
    print("=====================================")
