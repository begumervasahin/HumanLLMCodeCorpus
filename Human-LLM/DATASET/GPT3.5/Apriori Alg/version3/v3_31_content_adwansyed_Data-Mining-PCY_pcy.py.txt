import time
import itertools

# Global variables
pairs_hashtable = {}
frequent_pairs = {}
num_buckets = 10
support_threshold = 0.2
bitmap = [0] * num_buckets

# Hashes a pair using PCY hash function
def hash_pair(n1, n2):
    return ((n1 * n1) + (n2 * n2)) % num_buckets

# PCY Algorithm
def pcy(baskets, support_threshold, start_time):
    singletons = {}

    # Count occurrences of single items
    for transaction in baskets:
        for item in transaction.split():
            singletons[item] = singletons.get(item, 0) + 1

    # Find frequent single items
    frequent_single_items = {item: count for item, count in singletons.items() if count > (len(baskets) * support_threshold)}
    frequent_singletons = list(frequent_single_items.keys())

    # Generate pairs and update pairs hashtable
    for pair in itertools.combinations(frequent_singletons, 2):
        hash_value = hash_pair(int(pair[0]), int(pair[1]))
        pairs_hashtable[hash_value] = pairs_hashtable.get(hash_value, 0) + 1

    # Update bitmap based on pairs hashtable
    for key in sorted(pairs_hashtable.keys()):
        if pairs_hashtable[key] >= 2:
            bitmap[key] = 1

    # Generate frequent pairs
    for i in range(len(frequent_singletons)):
        for j in range(i + 1, len(frequent_singletons)):
            hash_value = hash_pair(int(frequent_singletons[i]), int(frequent_singletons[j]))
            if bitmap[hash_value] > 0:
                candidate_pair = frequent_singletons[i] + ", " + frequent_singletons[j]
                frequent_pairs[candidate_pair] = frequent_pairs.get(candidate_pair, 0) + 1

    # Print frequent pairs and time taken
    print(frequent_pairs)
    print("\nTime taken:", time.time() - start_time, "seconds")

# Main function to run PCY algorithm
def run_pcy(support_threshold, num_baskets, file_name):
    with open(file_name) as file:
        lines = file.readlines()

    print('Number of lines:', len(lines))
    basket_size = len(lines) / num_baskets
    print('Basket size:', basket_size)

    # Split the lines into baskets
    list_of_baskets = [lines[i:i+int(basket_size)] for i in range(0, len(lines), int(basket_size))]
    print('Number of baskets:', len(list_of_baskets))

    total_time = 0
    for i, basket in enumerate(list_of_baskets):
        start_time = time.time()
        pcy(basket, support_threshold, start_time)
        end_time = time.time()
        print('Time taken in seconds for basket {}: {}'.format(i, end_time - start_time))
        total_time += (end_time - start_time)

    print('Total time taken to run:', total_time, 'seconds')

# Run the PCY algorithm
run_pcy(support_threshold, num_baskets, 'retail.dat')
