#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import numpy as np
import csv
from sklearn.manifold import TSNE
from sklearn.decomposition import TruncatedSVD

def clean_disease_names(names):
    """Strip whitespace from disease names."""
    return [name.strip() for name in names]

def process_file(filename, output_index):
    """Process a single file and perform dimensionality reduction techniques."""
    with open(filename) as file:
        lines = [line.split('\t') for line in file]

    input_data = np.asarray(lines)
    disease_name = clean_disease_names(input_data[:, -1])

    data = np.mat(input_data[:, :(input_data.shape[1] - 1)].astype(float))

    mean = np.mean(data, axis=0)
    centered_data = np.subtract(data, mean)

    cov_mat = np.cov(np.transpose(centered_data))

    eigen_values, eigen_vectors = np.linalg.eig(cov_mat)
    eigen_values = np.reshape(eigen_values, (1, data.shape[1]))
    sort_indices = eigen_values.argsort()
    result = np.append(eigen_vectors[:, sort_indices[0][-1]], eigen_vectors[:, sort_indices[0][-2]])
    result = np.reshape(result, (data.shape[1], 2), order='F')

    new_data_points = (centered_data * result).tolist()

    for j in range(len(new_data_points)):
        new_data_points[j].append(disease_name[j])

    output_file = f'pca_{output_index}.csv'
    with open(output_file, "w") as output:
        writer = csv.writer(output)
        writer.writerows(new_data_points)

    data_tsne = TSNE(n_components=2).fit_transform(data)
    tsne = np.c_[data_tsne, disease_name]
    tsne_output = f'tsne_{output_index}.csv'
    with open(tsne_output, "w") as output:
        writer = csv.writer(output)
        writer.writerows(tsne)

    svd = TruncatedSVD(n_components=2)
    data_svd = svd.fit_transform(centered_data)
    data_svd = np.c_[data_svd, disease_name]
    svd_output = f'svd_{output_index}.csv'
    with open(svd_output, "w") as output:
        writer = csv.writer(output)
        writer.writerows(data_svd)

def main():
    """Main function to prompt user for files and process them."""
    output_index = 1
    while True:
        filename = input("Enter file name with extension or write 'exit' to stop: ")

        if filename.upper() == 'EXIT':
            break

        process_file(filename, output_index)
        output_index += 1

if __name__ == "__main__":
    main()
