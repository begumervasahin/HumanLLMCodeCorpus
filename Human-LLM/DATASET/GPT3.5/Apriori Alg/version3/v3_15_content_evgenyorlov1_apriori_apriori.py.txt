import argparse
from itertools import combinations

def join_set(item_set, k):
    """Generate combinations of itemsets."""
    return set([i.union(j) for i in item_set for j in item_set if len(i.union(j)) == k])

def generate_subsets(item_set):
    """Generate subsets of itemsets."""
    return combinations(item_set, i + 1) for i, a in enumerate(item_set)

def generate_itemsets(data):
    """Generate itemsets from data."""
    item_set = set()
    transaction_list = []

    for row in data:
        transaction_list.append(frozenset(row))
        for item in row:
            if item:
                item_set.add(frozenset([item]))

    return item_set, transaction_list

def calculate_support(transaction_list, item_set, min_support=0):
    """Calculate support for itemsets."""
    len_transaction_list = len(transaction_list)
    support_list = [(item, float(sum(1 for row in transaction_list if item.issubset(row))) / len_transaction_list) for item in item_set]
    
    return dict([(item, support) for item, support in support_list if support >= min_support])

def apriori(data, min_support, min_confidence):
    """Apriori Algorithm."""
    item_set, transaction_list = generate_itemsets(data)
    frequent_itemset = generate_frequent_itemsets(transaction_list, item_set, min_support)

    rules = []

            for A in generate_subsets(item):
                B = item.difference(A)

                if B:
                    A = frozenset(A)
                    AB = A | B

                    if confidence >= min_confidence:
                        rules.append((A, B, confidence))

    return rules, frequent_itemset

def print_report(rules, frequent_itemset):
    """Print the report with frequent itemsets and rules."""
    print('--Frequent Itemset--')
    
    for item, support in sorted(frequent_itemset.items(), key=lambda x: x[1]):
        print('[I] {} : {}'.format(tuple(item), round(support, 4)))

    print('\n--Rules--')
    
    for A, B, confidence in sorted(rules, key=lambda x: x[2]):
        print('[R] {} => {} : {}'.format(tuple(A), tuple(B), round(confidence, 4)))

def data_from_csv(filename):
    """Read data from CSV file."""
    with open(filename, 'rU') as f:
        for l in f:
            row = list(map(str.strip, l.split(',')))
            yield row

def parse_options():
    """Parse command-line arguments."""
    optparser = argparse.ArgumentParser(description='Apriori Algorithm.')
    optparser.add_argument(
        '-f', '--input_file',
        dest='filename',
        help='filename containing csv',
        required=True
    )
    optparser.add_argument(
        '-s', '--min_support',
        dest='min_support',
        help='minimum support',
        default=0.25,
        type=float
    )
    optparser.add_argument(
        '-c', '--min_confidence',
        dest='min_confidence',
        help='minimum confidence',
        default=0.5,
        type=float
    )
    return optparser.parse_args()

def main():
    """Main function."""
    options = parse_options()
    data = data_from_csv(options.filename)
    rules, frequent_itemset = apriori(data, options.min_support, options.min_confidence)
    print_report(rules, frequent_itemset)

if __name__ == '__main__':
    main()
