from optparse import OptionParser
from apriori import Apriori

def parse_command_line():
    """Parse command-line arguments and return the options."""
    parser = OptionParser()
    parser.add_option('-f', '--file',
                      dest='filePath',
                      help='Input a CSV file path',
                      type='string',
                      default=None)  

    parser.add_option('-s', '--minSupp',
                      dest='minSupp',
                      help='Minimum support threshold',
                      type='float',
                      default=0.10)  

    parser.add_option('-c', '--minConf', dest='minConf',
                      help='Minimum confidence threshold',
                      type='float',
                      default=0.40)  

    parser.add_option('-r', '--rhs', dest='rhs',
                      help='Right-hand side item for association rules',
                      type='string',
                      default=None)  

    return parser.parse_args()[0]

def print_parameters(filePath, minSupp, minConf, rhs):
    """Print the parameters."""
    print("Parameters:")
    print(f" - File Path: {filePath}")
    print(f" - Minimum Support: {minSupp}")
    print(f" - Minimum Confidence: {minConf}")
    print(f" - Right-hand Side Item: {rhs}\n")

def print_frequent_item_sets(freqSet):
    """Print frequent item sets."""
    for key, value in freqSet.items():
        print(f'Frequent {key}-term set:')
        print('-' * 20)
        for itemset in value:
            print(list(itemset))
        print()

def print_association_rules(rules, rhs):
    """Print association rules."""
    print('-' * 20)
    print(f'Rules referring to {rhs}:')
    for key, value in rules.items():
        print(f'{list(key)} -> {rhs}: {value}')

if __name__ == '__main__':
    options = parse_command_line()

    filePath = options.filePath
    minSupp = options.minSupp
    minConf = options.minConf
    rhs = options.rhs

    print_parameters(filePath, minSupp, minConf, rhs)

    objApriori = Apriori(minSupp, minConf)
    itemCountDict, freqSet = objApriori.fit(filePath)

    print_frequent_item_sets(freqSet)

    if rhs:
        rules = objApriori.getSpecRules(rhs)
        print_association_rules(rules, rhs)
