import csv
import itertools

# Paths to data and configuration files
DATA_PATH = '/home/aeon/Proj2/Apriori/INTEGRATED-DATASET2.csv'
CONFIG_PATH = '/home/aeon/Proj2/Apriori/Parameters2.config'

def main():
    # Retrieve dataset and minimum support from files
    data_set = read_data()
    min_support = read_min_support()

    # Apply Apriori algorithm to find frequent item sets
    frequent_sets = apriori(data_set, min_support)

    # Display frequent item sets
    display_frequent_sets(frequent_sets)

def apriori(data_set, min_support):
    frequent_sets = []
    k = 1
    cur_freq_set = initialize_frequent_set(data_set, min_support)

    # Find frequent item sets iteratively
    while cur_freq_set:
        frequent_sets.append(cur_freq_set)
        print_frequent_set(k, cur_freq_set)

        # Generate candidate item sets
        cur_freq_set = generate_candidates(k, cur_freq_set, data_set, min_support)
        k += 1

    print("All Frequent Sets:", frequent_sets)
    return frequent_sets

def initialize_frequent_set(data_set, min_support):
    freq_set = []
    cur_freq_set = []

    # Initialize support counts for 1-item sets
    for transaction in data_set:
        for item in transaction:
            if [item, 1] not in cur_freq_set:
                cur_freq_set.append([item, 1])

    # Filter items based on minimum support
    for item in cur_freq_set:
        if item[1] >= min_support:
            freq_set.append(item)

    print("\n1-Frequent Item Set:", freq_set)
    return freq_set

def generate_candidates(k, prev_freq_set, data_set, min_support):
    items = []
    combination = []
    pre_sets = []

    # Extract unique items from previous frequent sets
    for item_set in prev_freq_set:
        sets = []
        for i in range(0, k):
            sets.append(item_set[i])
            if item_set[i] not in items:
                items.append(item_set[i])
        pre_sets.append(sets)

    # Generate combinations of items
    for tup in itertools.combinations(items, k + 1):
        combination.append(list(tup))

    # Find candidate item sets
    candidates = find_candidate_sets(combination, pre_sets, data_set, k, min_support)

    print("\n{}-Frequent Candidate Set: {}".format(k + 1, candidates))
    return [cand for cand in candidates if cand[k + 1] >= min_support]

def find_candidate_sets(combination, pre_sets, data_set, k, min_support):
    candidates = []

    for comb in combination:
        flag = 0
        for tup in itertools.combinations(comb, k):
            if list(tup) not in pre_sets:
                flag = 1
                break
        if flag == 0:
            candidates.append(comb)

    # Calculate support for candidate item sets
    for cand in candidates:
        cand.append(calculate_support(cand, data_set))

    return candidates

def calculate_support(candidate, data_set):
    sup_val = 0
    for transaction in data_set:
        if set(candidate).issubset(set(transaction)):
            sup_val += 1
    return sup_val

def read_data():
    data = []
    with open(DATA_PATH, 'r') as f:
        reader = csv.reader(f)
        for row in reader:
            data.append(row)
    return data

def read_min_support():
    with open(CONFIG_PATH, 'r') as f:
        min_support = int(f.readline())
    return min_support

def display_frequent_sets(frequent_sets):
    print("\nFrequent Sets\n--------------")
    count = 1
    for freq_set in frequent_sets:
        print("\n{} Frequent Set: {}".format(count, freq_set))
        count += 1

if __name__ == "__main__":
    main()
