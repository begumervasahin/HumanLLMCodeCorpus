class Node:
    def __init__(self, k, prime_k, leaf_load, is_internal, is_leaf, is_root, level):
        self.k = k
        self.prime_k = prime_k
        self.leaf_load = leaf_load
        self.is_root = is_root
        self.is_internal = is_internal
        self.is_leaf = is_leaf
        self.level = level
        self.values = set()
        self.children = {}
        self._initialize()

    def _initialize(self):
        """
        Initializes an internal node or a root with its leaves.
        """
        if not self.is_leaf:
            for i in range(self.prime_k):
                self.children[i] = Node(self.k, self.prime_k, self.leaf_load, False, True, False, self.level + 1)

    def add_element(self, value):
        """
        Adds an element to the node.
        """
        if self.is_leaf:
            self.values.add(value)
            if self.is_overloaded() and self.level < self.k:
                return self._divide_leaf_node()
            else:
                return None
        else:
            assert self.is_root or self.is_internal, "A leaf can never be neither leaf, root, nor the internal node"
            try:
                child_index = value[self.level] % self.prime_k
            except:
                print(value, self.level)
                child_index = value[self.level] % self.prime_k
            result = self.children.get(child_index).add_element(value)
            if result:
                self.children[child_index] = result
            return None

    def add_elements(self, values):
        """
        Adds a list of values to the Hash tree.
        """
        for value in values:
            self.add_element(value)

    def is_overloaded(self):
        """
        Checks whether a leaf node is overloaded.
        """
        assert self.is_leaf, "Only the leaves can use this function"
        return len(self.values) > self.leaf_load

    def _divide_leaf_node(self):
        """
        Divides the leaf node when overloaded.
        """
        new_node = Node(self.k, self.prime_k, self.leaf_load, True, False, False, self.level)
        new_node.add_elements(self.values)
        return new_node

    def search(self, element):
        """
        Searches for an element in the node.
        """
        if self.is_leaf:
            return element in self.values
        else:
            child_index = element[self.level] % self.prime_k
            return self.children.get(child_index).search(element)

    def visualize(self):
        """
        Visualizes the contents of the node.
        """
        if self.is_leaf:
            print("------------------------------------It is a leaf node and it contains-------------------------------------------")
            print(self.values)
            print("--------------------------------------------------------------------------------------------------------------")
        else:
            for child_index in self.children:
                print("Child_" + str(child_index))
                self.children[child_index].visualize()
