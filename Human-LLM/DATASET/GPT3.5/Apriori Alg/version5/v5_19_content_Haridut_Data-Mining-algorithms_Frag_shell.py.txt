import sys
import itertools

# Function to count occurrences of a subset in the list of transactions
def count_occurrences(subset, transactions):
    count = 0
    for transaction in transactions:
        if set(subset) < set(transaction):
            count += 1
    return count

if __name__ == "__main__":
    # Read input from standard input
    input_lines = sys.stdin.readlines()
    input_lines = [line.rstrip('\n') for line in input_lines]

    # Extract partitions and item lists
    partitions = int(input_lines[0])
    item_list = [line.split() for line in input_lines[1:]]

    # Generate dimensional lists
    dim1_list = [[item_list[j][i] for j in range(len(item_list))] for i in range(len(item_list[0]))]
    dim_list = [sorted(set(dim)) for dim in dim1_list]

    # Initialize list for partitions
    lista = [[] for _ in range(partitions)]

    # Partition the dimensional lists
    for j in range(len(item_list[0])):
        lista[j % partitions].append(j)

    # Count the indices in each partition
    ind_count = [len(partition) for partition in lista]

    # Initialize list of fragmented lists
    m = 0
    list_fragmented = []
    for i in range(partitions):
        listc = [dim_list[m] for _ in range(ind_count[i])]
        m += 1
        list_fragmented.append(listc)

    # Generate combinations and count occurrences
    for i in range(len(list_fragmented)):
        for j in range(len(list_fragmented[i])):
            for combination in itertools.combinations(range(len(list_fragmented[i])), j + 1):
                merged_list = [item for sublist_index in combination for item in list_fragmented[i][sublist_index]]
                for subset_combination in itertools.combinations(merged_list, len(combination)):
                    count = count_occurrences(subset_combination, item_list)
                    if count != 0:
                        print(' '.join(map(str, subset_combination)), end='')
                        print(': ' + str(count))

        print()
