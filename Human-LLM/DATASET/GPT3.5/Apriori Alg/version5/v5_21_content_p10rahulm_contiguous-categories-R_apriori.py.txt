import time
import itertools
import operator
from collections import defaultdict

def read_transactions(filename):
    """
    Read transactions from a file and return as a list of sets.
    """
    with open(filename, "r") as file:
        transactions = [set(line.strip().split(";")) for line in file]
    return transactions

def build_skus(transactions):
    """
    Build a dictionary of SKUs from transactions and return the total count.
    """
    skus = defaultdict(int)
    for transaction in transactions:
        for element in transaction:
            skus[element] += 1
    total_skus = sum(skus.values())
    return skus, total_skus

def get_level_0_frequent(skus, min_support):
    """
    Get level 0 frequent SKUs based on minimum support count.
    """
    return {sku: count for sku, count in skus.items() if count > min_support}

def print_to_file(skus_dict, output_file):
    """
    Print SKUs dictionary items to a file sorted by count in descending order.
    """
    sorted_skus = sorted(skus_dict, key=skus_dict.get, reverse=True)
    with open(output_file, 'w+') as file:
        for element in sorted_skus:
            file.write(f"{skus_dict[element]}:{element}\n")

def get_level_wise_frequent(skus, min_support, transactions):
    """
    Get level-wise frequent SKUs based on minimum support count and transactions.
    """
    level = 2
    answer_list = [(k, v) for k, v in skus.items()]
    last_level = {frozenset({k}): v for k, v in skus.items()}

    while last_level:
        this_level = dict()
        for sku_combo in itertools.combinations(last_level.keys(), level):
            set_union = set().union(*sku_combo)
            if len(set_union) == level:
                new_item = frozenset(set_union)
                this_level[new_item] = sum(1 for transaction in transactions if new_item.issubset(transaction))

        this_level = {k: v for k, v in this_level.items() if v > min_support}
        if this_level:
            answer_list.extend([(k, v) for k, v in this_level.items()])
        last_level = this_level
        level += 1

    return answer_list

def append_to_file(input_list, output_file):
    """
    Append items from a list to a file, sorted by count in descending order.
    """
    input_list.sort(key=operator.itemgetter(1), reverse=True)
    with open(output_file, 'w') as file:
        for element in input_list:
            if isinstance(element[0], frozenset):
                file.write(f"{element[1]}:{';'.join(map(str, element[0]))}\n")
            else:
                file.write(f"{element[1]}:{element[0]}\n")

if __name__ == "__main__":
    start_time = time.time()
    transactions = read_transactions("rawdata/categories.txt")
    skus, total_skus = build_skus(transactions)
    min_support = 771
    level_0_frequent = get_level_0_frequent(skus, min_support)
    print_to_file(level_0_frequent, "output/myfile.txt")
    level_wise_frequent = get_level_wise_frequent(level_0_frequent, min_support, transactions)
    append_to_file(level_wise_frequent, "output/myfile2.txt")
    time_taken = time.time() - start_time
    print(f"Time taken for {len(transactions)} transactions with {total_skus} skus is {time_taken} seconds")
