import itertools

def generate_next_size_list(current_size_list):
    """
    Generate the next size list based on the current size list.
    """
    next_size_list = []
    size = len(current_size_list)
    for i in range(size):
        for j in range(i + 1, size):
            if current_size_list[i][:-1] == current_size_list[j][:-1]:
                new_itemset = current_size_list[i][:]
                new_itemset.append(current_size_list[j][-1])
                next_size_list.append(new_itemset)
            else:
                break
    return next_size_list

def generate_one_less_subsets(itemset):
    """
    Generate all subsets by removing one element from the itemset.
    """
    all_subsets = []
    size = len(itemset)
    for i in range(size):
        subset = itemset[:i] + itemset[i+1:]
        all_subsets.append(subset)
    return all_subsets

def prune_freqs_trie(freqs_trie, current_list):
    """
    Prune the current list of itemsets based on the frequent trie structure.
    """
    pruned_list = []
    for itemset in current_list:
        all_subsets = generate_one_less_subsets(itemset)
        flag = all(freqs_trie.hasNode(subset) for subset in all_subsets)
        if flag:
            pruned_list.append(itemset)
    return pruned_list

def count_item_sets(input_file, current_list):
    """
    Count the occurrences of itemsets in the input file.
    """
    counts = [0] * len(current_list)
    with open(input_file, 'r') as infile:
        for line in infile:
            line = line.strip()
            item_list = line.split(",")
            curr_transaction = set(item_list)
            curr_size = len(curr_transaction)
            for i, itemset in enumerate(current_list):
                if curr_size < len(itemset):
                    continue
                item_set = set(itemset)
                if curr_transaction.issuperset(item_set):
                    counts[i] += 1
    return counts

def print_associate_rules(freqs_trie, mincon):
    """
    Print association rules based on the frequent trie structure and minimum confidence.
    """
    count = 0
    for itemset in freqs_trie.get_item_sets([]):
        for j in range(1, len(itemset)):
            for k in itertools.combinations(itemset, j):
                if float(freqs_trie.get_count(itemset)) / float(freqs_trie.get_count(list(k))) >= mincon:
                    print(','.join(k) + "=>" + ','.join(set(itemset) - set(k)))
                    count += 1
    return count
