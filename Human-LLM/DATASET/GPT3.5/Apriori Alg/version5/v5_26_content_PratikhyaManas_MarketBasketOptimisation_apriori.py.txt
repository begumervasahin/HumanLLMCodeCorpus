# Importing necessary libraries
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from apyori import apriori

# Data Preprocessing
def load_dataset(file_path):
    """Load dataset from CSV file."""
    return pd.read_csv(file_path, header=None)

def convert_to_transactions(dataset):
    """Convert dataset into a list of transactions."""
    transactions = []
    for i in range(len(dataset)):
        transaction = [str(dataset.values[i, j]) for j in range(len(dataset.columns))]
        transactions.append(transaction)
    return transactions

# Main functionality
def mine_association_rules(transactions, min_support=0.003, min_confidence=0.2, min_lift=3, min_length=2):
    """Mine association rules using Apriori algorithm."""
    rules = apriori(transactions, min_support=min_support, min_confidence=min_confidence, min_lift=min_lift, min_length=min_length)
    return list(rules)

if __name__ == "__main__":
    # Data Preprocessing
    # Reading the dataset
    dataset = load_dataset('Market_Basket_Optimisation.csv')

    # Converting dataset into transactions
    transactions = convert_to_transactions(dataset)

    # Mining association rules
    rules = mine_association_rules(transactions)

    # Extracting the results
    results = list(rules)
