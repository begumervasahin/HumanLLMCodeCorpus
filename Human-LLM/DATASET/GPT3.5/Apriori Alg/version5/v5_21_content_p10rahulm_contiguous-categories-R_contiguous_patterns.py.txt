import time
import operator
from collections import defaultdict

def read_text(filename):
    """
    Read text from a file and split it into a list of lists of words.
    """
    with open(filename) as file:
        lines = [line.rstrip("\n").split(" ") for line in file]
    return lines

def count_word_frequency(lines):
    """
    Count the frequency of words in a list of lines.
    """
    word_frequency = defaultdict(int)
    for line in lines:
        for word in line:
            word_frequency[word] += 1
    return word_frequency

def generate_patterns(line):
    """
    Generate patterns for a given line.
    """
    line_length = len(line)
    patterns = set()
    for i in range(1, line_length):
        for j in range(0, line_length - i + 1):
            patterns.add(" ".join(line[j:j+i]))
    return patterns

def generate_patterns_for_each_line(all_lines):
    """
    Generate patterns for each line in the text.
    """
    line_patterns = []
    for line in all_lines:
        line_patterns.append(generate_patterns(line))
    return line_patterns

def count_phrase_frequency(line_patterns):
    """
    Count the frequency of phrases in a list of sets.
    """
    phrase_frequency = defaultdict(int)
    for patterns in line_patterns:
        if patterns:
            for pattern in patterns:
                phrase_frequency[pattern] += 1
    return phrase_frequency

def filter_phrases_by_frequency(phrase_frequency, min_sup):
    """
    Filter phrases by frequency.
    """
    return {phrase: freq for phrase, freq in phrase_frequency.items() if freq >= min_sup}

def filter_phrases_by_word_length(phrase_frequency, min_length):
    """
    Filter phrases by word length.
    """
    return {phrase: freq for phrase, freq in phrase_frequency.items() if len(phrase.split(" ")) >= min_length}

def print_phrases_to_file(phrase_frequency, output_file):
    """
    Print phrases to a file sorted by frequency in descending order.
    """
    sorted_phrases = sorted(phrase_frequency.items(), key=operator.itemgetter(1), reverse=True)
    with open(output_file, 'w') as file:
        for phrase, freq in sorted_phrases:
            file.write(f"{freq}:{phrase.replace(' ', ';')}\n")

if __name__ == "__main__":
    start_time = time.time()
    txt_lines = read_text("rawdata/yelp_reviews.txt")
    word_frequency = count_word_frequency(txt_lines)
    line_patterns = generate_patterns_for_each_line(txt_lines)
    phrase_frequency = count_phrase_frequency(line_patterns)
    print("Number of phrases checked:", len(phrase_frequency))
    frequent_phrases = filter_phrases_by_frequency(phrase_frequency, 100)
    frequent_phrases = filter_phrases_by_word_length(frequent_phrases, 2)
    print("Algorithm for Yelp ran in", time.time() - start_time, "seconds")
    print_phrases_to_file(frequent_phrases, "output/yelp_contiguous_patterns.txt")
