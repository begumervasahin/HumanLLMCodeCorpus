class TrieNode:
    def __init__(self, count=1, name="root"):
        """
        Initialize TrieNode with count and name.
        """
        self.child = {}  # Dictionary to store child nodes
        self.count = count  # Count of occurrences of the itemset
        self.name = name  # Name of the node

    def insertNode(self, itemset, count):
        """
        Insert a node into the trie.
        """
        current_node = self.child
        for item in itemset:
            if item in current_node:
                current_node = current_node[item].child
            else:
                new_node = TrieNode(count, item)
                current_node[item] = new_node

    def hasNode(self, itemset):
        """
        Check if a node exists in the trie.
        """
        current_node = self.child
        for item in itemset:
            if item in current_node:
                current_node = current_node[item].child
            else:
                return False
        return True

    def insertAll(self, FreqItemSets, counts):
        """
        Insert multiple nodes into the trie.
        """
        for itemset, count in zip(FreqItemSets, counts):
            self.insertNode(itemset, count)

    def printAll(self, prevStr=""):
        """
        Print all nodes in the trie.
        """
        for item in self.child:
            temp_str = prevStr + item
            print(temp_str)
            self.child[item].printAll(temp_str + ",")

    def getItemSets(self, prevList=None):
        """
        Get item sets from the trie.
        """
        if prevList is None:
            prevList = []
        for item in self.child:
            temp_list = prevList + [item]
            yield temp_list
            for sub_item in self.child[item].getItemSets(temp_list):
                yield sub_item

    def getCount(self, itemSet):
        """
        Get the count of occurrences of an item set.
        """
        current_node = self.child
        for item in itemSet:
            current_node = current_node[item].child
        return current_node.count
