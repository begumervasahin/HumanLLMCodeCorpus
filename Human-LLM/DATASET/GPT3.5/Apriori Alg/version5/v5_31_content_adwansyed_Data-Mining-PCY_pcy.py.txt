import time
import itertools

# Global variables
pairs_hashtable = {}
frequent_pairs = {}
num_buckets = 10
num_baskets = 10
support_threshold = 0.2
bitmap = [0] * num_buckets

# Hash function for frequent pairs hash table
def hash_pair(n1, n2):
    return ((n1 * n1) + (n2 * n2)) % num_buckets

# PCY algorithm
def pcy(baskets, threshold, start_time):
    # Pass 1: Generate singletons with counts
    singletons = count_singletons(baskets)

    # Prune frequent single items from singletons
    frequent_single_items = prune_singletons(singletons, len(baskets), threshold)

    # Store frequent singletons for data manipulation on list
    frequent_singletons = list(frequent_single_items.keys())

    # Generate pairs and update pairs hashtable
    generate_pairs(frequent_singletons)

    # Between passes: Update bitmap based on pairs hashtable
    update_bitmap()

    # Pass 2: Generate frequent pairs
    generate_frequent_pairs(frequent_singletons)

    # Print frequent pairs and time taken
    print_results(start_time)

# Function to count singletons in baskets
def count_singletons(baskets):
    singletons = {}
    for transaction in baskets:
        for item in transaction.split():
            singletons[item] = singletons.get(item, 0) + 1
    return singletons

# Function to prune frequent single items from singletons
def prune_singletons(singletons, basket_count, threshold):
    return {key: count for key, count in singletons.items() if count > basket_count * threshold}

# Function to generate pairs and update pairs hashtable
def generate_pairs(frequent_singletons):
    for key in itertools.combinations(frequent_singletons, 2):
        hash_value = hash_pair(int(key[0]), int(key[1]))
        pairs_hashtable[hash_value] = pairs_hashtable.get(hash_value, 0) + 1

# Function to update bitmap based on pairs hashtable
def update_bitmap():
    for key in sorted(pairs_hashtable.keys()):
        if pairs_hashtable[key] >= 2:
            bitmap[key] = 1

# Function to generate frequent pairs
def generate_frequent_pairs(frequent_singletons):
    for i in range(len(frequent_singletons)):
        for j in range(i + 1, len(frequent_singletons)):
            hash_value = hash_pair(int(frequent_singletons[i]), int(frequent_singletons[j]))
            if bitmap[hash_value] > 0:
                candidate_pair = frequent_singletons[i] + ", " + frequent_singletons[j]
                frequent_pairs[candidate_pair] = frequent_pairs.get(candidate_pair, 0) + 1

# Function to print frequent pairs and time taken
def print_results(start_time):
    print(frequent_pairs)
    print("\nTime taken:", time.time() - start_time, "seconds")

# Function to run PCY on a file with designated baskets and record timestamps
def run(threshold, num_baskets, file_name):
    dataset_file = file_name
    num_lines = sum(1 for line in open(dataset_file))
    print("Number of lines: ", num_lines)

    basket_size = num_lines / num_baskets
    print("Basket size: ", basket_size)

    list_of_baskets = read_baskets(dataset_file, basket_size)

    total_time = 0
    for i, basket in enumerate(list_of_baskets):
        start_time = time.time()
        pcy(basket, threshold, start_time)
        end_time = time.time()
        print("Time taken in seconds for basket %d: %f" % (i, end_time - start_time))
        total_time += (end_time - start_time)
    print("Total time taken to run: ", total_time)

# Function to read baskets from file
def read_baskets(dataset_file, basket_size):
    list_of_baskets = []
    with open(dataset_file) as file:
        basket = []
        for line in file:
            basket.append(line)
            if len(basket) > basket_size:
                list_of_baskets.append(basket)
                basket = []
        list_of_baskets.append(basket)
    print("Number of baskets: ", len(list_of_baskets))
    return list_of_baskets

# Running PCY algorithm
run(support_threshold, num_baskets, "retail.dat")
