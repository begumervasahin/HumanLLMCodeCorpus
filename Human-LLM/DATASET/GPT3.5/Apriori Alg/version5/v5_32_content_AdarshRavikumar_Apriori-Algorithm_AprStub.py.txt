import re
from Apriori import Apriori

# Prompt the user to enter minimum support, confidence, and length of rules
min_support = int(input("Enter the minimum support: "))
min_confidence = float(input("Enter the minimum confidence: "))
min_length = int(input("Enter the minimum length of rules: "))

# Define alphabets for item encoding
alphabet = [chr(i) for i in range(ord('a'), ord('z') + 1)]

# Read and preprocess data from the file
items = []
with open('datasetUCI.txt', 'r') as file:
    for line in file:
        # Preprocess the line to replace special characters and spaces with 'a'
        line = line.strip()[:-1]
        line = re.sub("[?\s]", 'a', line)
        # Split the line into items and filter out items not in the alphabet
        transaction = line.split(',')
        transaction_items = [item for item in transaction if item in alphabet]
        items.append(transaction_items)

# Run Apriori algorithm
apriori_result = Apriori(items, min_support, min_confidence, min_length)
