class DataBase:
    """
    Represents a database for skill recommendation using the Apriori algorithm.
    """

    def __init__(self, skills_data=None):
        """
        Initializes the database with skills data.
        """
        self.skills_data = skills_data if skills_data else []

    def load_skills_data(self, data_path):
        """
        Loads skills data from a file specified by the given data path.
        """
        try:
            with open(data_path, 'r') as file:
                self.skills_data = [line.strip().split(',') for line in file]
            print("Skills data loaded successfully.")
        except FileNotFoundError:
            print("Error: File not found.")

    def display_skills_data(self):
        """
        Displays the loaded skills data.
        """
        if self.skills_data:
            print("Skills Data:")
            for transaction in self.skills_data:
                print(transaction)
        else:
            print("No skills data loaded.")

    def apriori_recommendation(self, min_support, min_confidence):
        """
        Recommends skills using the Apriori algorithm based on minimum support and confidence.
        """
        if not self.skills_data:
            print("Error: No skills data loaded.")

        print("Applying Apriori algorithm...")
        # Placeholder for generating recommendations
