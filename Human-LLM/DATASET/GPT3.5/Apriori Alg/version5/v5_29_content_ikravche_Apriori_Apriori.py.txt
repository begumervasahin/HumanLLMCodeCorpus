import numpy as np
import itertools as it

# Minimum support and confidence thresholds
min_supp = 3
min_confidence = 0.7

# Counter for combinations
combinations_counter = 1

# Dictionaries for results and stopping criteria
dict_for_result = {}
dict_for_stop = {}

# List of transactions
transaction_list = [
    ['a', 'b'],
    ['a', 'c', 'd', 'e'],
    ['a', 'b', 'c', 'e'],
    ['c', 'd'],
    ['b', 'c', 'd', 'e'],
    ['a', 'd', 'e'],
    ['c', 'd', 'e']
]

# Set of unique products in transactions
products_set = set([item for sublist in transaction_list for item in sublist])


def last_maxsupport_check():
    """
    Check if the maximum support for the last combination size is less than the minimum support threshold.
    """
    max_value = max([value for key, value in dict_for_result.items() if len(key) == combinations_counter - 1], default=0)
    return max_value >= min_supp or combinations_counter == 1


def make_combination():
    """
    Generate combinations of products and count their appearances.
    """
    if last_maxsupport_check():
        combinations = list(it.combinations(products_set, combinations_counter))
        count_appearances(combinations)
    else:
        print_results()


def count_appearances(combinations):
    """
    Count the appearances of combinations in the transaction list.
    """
    global combinations_counter
    for element in combinations:
        for row in transaction_list:
            check_for_appearances(element, row)
    combinations_counter += 1
    make_combination()


def check_for_appearances(element, row):
    """
    Check if the combination appears in the transaction row and update the result dictionary.
    """
    element = ''.join(sorted(element))
    row = ''.join(sorted(row))
    row = throw_useless(element, row)
    if element in row:
        dict_for_result[element] = dict_for_result.get(element, 0) + 1


def throw_useless(element, row):
    """
    Remove unnecessary characters from the row based on the current combination.
    """
    for row_char in row:
        if row_char not in element:
            row = row.replace(row_char, '')
    return row


def print_results():
    """
    Print the result dictionary with combinations that meet the minimum support threshold.
    """
    print(dict_for_result)
    for key, value in dict_for_result.items():
        if value >= min_supp:
            print(f'Elements: {key} support {value}')


# Start the combination generation and counting process
make_combination()
