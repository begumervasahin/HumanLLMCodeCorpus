import tkinter as tk
from tkinter import ttk
from tkinter import filedialog as fd
import AprioriAlgorithmV5 as apriori

# Function to open and display file content
def open_file():
    file = fd.askopenfile()
    filename_output.delete(1.0, tk.END)
    filename_output.insert(tk.END, file.name)

    filename = filename_output.get("1.0", tk.END).rstrip()
    print(filename)
    item_transaction_dic = read_transaction_data(filename)
    display_transaction_data(item_transaction_dic)

# Function to read transaction data from file
def read_transaction_data(filename):
    item_transaction_dic = {}
    with open(filename, "r") as f:
        for line in f:
            items = line.split()
            item_transaction_dic[items[0]] = items[1:]
    return item_transaction_dic

# Function to display transaction data
def display_transaction_data(item_transaction_dic):
    transaction_list_output.delete(1.0, tk.END)
    transaction_list_output.insert(tk.END, "TID:   Transactions\n\n")
    for key, value in item_transaction_dic.items():
        s = f"{key}: {value}\n"
        transaction_list_output.insert(tk.END, s)

# Function to get association rules and display them
def get_apriori():
    filename = filename_output.get("1.0", tk.END).rstrip()
    print(filename)
    item_transaction_dic = read_transaction_data(filename)

    min_support = float(support_label_input.get("1.0", tk.END))
    min_confid = float(confidence_label_input.get("1.0", tk.END))

    associations_rule = apriori.findAssociationRules(item_transaction_dic, min_support, min_confid)

    display_association_rules(associations_rule, min_support, min_confid)

# Function to display association rules
def display_association_rules(associations_rule, min_support, min_confid):
    aprior_output.delete(1.0, tk.END)
    aprior_output.insert(tk.END, f"The minimum support is: {min_support}\n")
    aprior_output.insert(tk.END, f"The minimum confidence is: {min_confid}\n\n")
    aprior_output.insert(tk.END, f"There are {len(associations_rule)} association rules:\n")
    for a in associations_rule:
        s = f"{a[0]} --> {a[1]} {{{a[2]}%, {a[3]}%}}\n"
        aprior_output.insert(tk.END, s)

# Main window
window = tk.Tk()
window.title("Apriori Algorithm")
window.configure(background="black")
window.geometry("1200x700")

# Title label
title_label = ttk.Label(window, text="Apriori Algorithm", foreground="white", background="black")
title_label.config(font=("Verdana", 20))
title_label.grid(row=0, column=2, padx=10, pady=10)

# File selection button and output
filename_button = ttk.Button(window, text="Click Here: Open File", width=30, command=open_file)
filename_button.grid(row=1, column=0, sticky=tk.W, padx=10, pady=10)
filename_output = tk.Text(window, width=50, height=2, wrap=tk.WORD, background="white")
filename_output.grid(row=1, column=1, columnspan=2, sticky=tk.E, pady=10)

# Minimum support label and input
support_label = ttk.Label(window, text="Minimum Support (0-1): ", foreground="white", background="black")
support_label.config(font=("Verdana", 12))
support_label.grid(row=2, column=0, columnspan=2, sticky=tk.W, padx=10, pady=10)
support_label_input = tk.Text(window, width=10, height=2, wrap=tk.WORD, background="white")
support_label_input.grid(row=2, column=1, columnspan=2, sticky=tk.W, pady=10)

# Minimum confidence label and input
confidence_label = ttk.Label(window, text="Minimum Confidence (0-1):  ", foreground="white", background="black")
confidence_label.config(font=("Verdana", 11))
confidence_label.grid(row=3, column=0, columnspan=2, sticky=tk.W, padx=10, pady=10)
confidence_label_input = tk.Text(window, width=10, height=2, wrap=tk.WORD, background="white")
confidence_label_input.grid(row=3, column=1, columnspan=2, sticky=tk.W, pady=10)

# Transaction list label and output
transaction_label = ttk.Label(window, text="Transactions: ", foreground="white", background="black")
transaction_label.config(font=("Verdana", 12))
transaction_label.grid(row=4, column=0, columnspan=2, padx=10, pady=10)
transaction_list_output = tk.Text(window, width=50, height=25, wrap=tk.WORD, background="white")
transaction_list_output.grid(row=10, column=0, columnspan=2, sticky=tk.W, padx=10)

# Association rules label and output
association_rule_label = ttk.Label(window, text="Association Rules: ", foreground="white", background="black")
association_rule_label.config(font=("Verdana", 12))
association_rule_label.grid(row=4, column=10, columnspan=2, padx=10, pady=10)
aprior_button = ttk.Button(window, text="Get Association Rules", width=30, command=get_apriori)
aprior_button.grid(row=10, column=2, sticky=tk.W, padx=10)
aprior_output = tk.Text(window, width=50, height=25, wrap=tk.WORD, background="white")
aprior_output.grid(row=10, column=10, columnspan=2, sticky=tk.E, padx=10)

# Run the main loop
window.mainloop()
