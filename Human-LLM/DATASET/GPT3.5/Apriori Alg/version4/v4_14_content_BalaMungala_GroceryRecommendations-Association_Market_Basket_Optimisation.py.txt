# Importing necessary libraries
import pandas as pd
from apyori import apriori
import matplotlib.pyplot as plt

# File path to the dataset
dataset_path = "F://ML//Machine Learning A-Z Template Folder\Part 5 - Association Rule Learning//Section 28 - Apriori//Apriori-Python//Apriori_Python//Market_Basket_Optimisation.csv"

# Reading the dataset
dataset = pd.read_csv(dataset_path, header=None)

# Data Preprocessing
transactions = []

# Converting dataset into a list of transactions
for i in range(0, len(dataset)):
    transactions.append([str(dataset.values[i, j]) for j in range(0, 20)])

# Training Apriori model
rules = apriori(transactions,
                min_support=0.003,
                min_confidence=0.2,
                min_length=2,
                min_lift=3)

# Converting the rules to a list
results = list(rules)

# Visualizing item frequency
plt.itemFrequency(results)
