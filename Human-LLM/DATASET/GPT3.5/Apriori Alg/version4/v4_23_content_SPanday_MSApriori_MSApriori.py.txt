import itertools

# Global variables
phi = 0.0
MS = {}
temp = {}
MISSortDict = {}
IList = []
ICount = {}
TList = []
TCount = 0
L = []
FList = [[] for _ in range(10)]
CList = [[] for _ in range(10)]
CDict = {}
cannotBeTogether = []
mustHave = []
tailCount = {}

def main():
    read_data()
    init_pass()
    find_frequent_1_item_sets()
    find_frequent_item_sets()
    apply_cannot_be_together_constraint()
    if mustHave:
        apply_must_have_constraint()
    output()


def read_data():
    global MS, TCount, IList, ICount, phi, TList, cannotBeTogether, mustHave
    with open('para1-2.txt') as file1:
        for line in file1:
            if 'SDC' in line:
                phi = float(line.replace(' ', '').rstrip().split('=')[1])

            if 'MIS' in line:
                temp = line.replace(' ', '').replace('MIS', '').replace('(', '').replace(')', '').rstrip().split('=')
                MS[int(temp[0])] = float(temp[1])

            if 'must-have' in line:
                mustHave = line.replace('must-have:', '').replace(' ', '').rstrip().split('or')
                if mustHave:
                    mustHave = list(map(int, mustHave))

            if 'cannot_be_together' in line:
                cannotBeTogether = line.replace(' ', '').replace('cannot_be_together:', '').replace('{', '[').replace('}', ']').replace("'", 'a').rstrip().split('a')
                cannotBeTogether = cannotBeTogether[0].replace('],[', ' ').replace('[', '').replace(']', '').split()
                cannotBeTogether = [list(map(int, subset.split(','))) for subset in cannotBeTogether]

    items = sorted(MS, key=MS.get)
    for item in items:
        IList.append(int(item))
        ICount[int(item)] = 0

    with open('data-1.txt') as file2:
        for line in file2:
            transaction = line.replace(' ', '').replace('{', '').replace('}', '').replace('<', '').replace('>', '').split(',')
            TList.append([int(t) for t in transaction])
            for t in transaction:
                ICount[int(t)] = ICount.get(int(t), 0) + 1

    TCount = len(TList)

def init_pass():
    global L
    for i in range(len(IList)):
        if i == 0:
            L.append(IList[0])
        elif (ICount.get(IList[i]) / TCount) >= MS.get(IList[0]):
            L.append(IList[i])

    for i, item in enumerate(L):
        MISSortDict[item] = i

def level2_candidate_generation():
    global CList, L
    for l in range(len(L)):
        if (ICount[L[l]] / TCount) >= MS[L[l]]:
            for h in range(l + 1, len(L)):
                if (ICount[L[h]] / TCount) >= MS[L[l]] and abs((ICount[L[h]] / TCount) - (ICount[L[l]] / TCount)) <= phi:
                    CList[2].append([L[l], L[h]])
    CList[2].sort(key=lambda row: row[1])

def find_subsets(S, m):
    return list(set(itertools.combinations(S, m)))

def MS_candidate_generation(n):
    m = n - 1
    for i in range(len(FList[m])):
        for j in range(len(FList[m])):
            k = 0
            while k < m - 1 and FList[m][i][k] == FList[m][j][k]:
                k += 1

            if k == m - 1:
                if MISSortDict[FList[m][i][k]] < MISSortDict[FList[m][j][k]] and abs((ICount[FList[m][i][k]] / TCount) - (ICount[FList[m][j][k]] / TCount)) <= phi:
                    CList[m+1].append(list(FList[m][i]))
                    CList[m+1][-1].append(FList[m][j][k])
                    subset = find_subsets(CList[m+1][-1], m)
                    if not CList[m+1]:
                        if bool(CList[m+1][-1][0]) in subset[sub] or MS[CList[m+1][-1][1]] == MS[CList[m+1][-1][0]]:
                            if bool(subset[sub]) not in FList[m]:
                                CList[m+1][-1].remove()

            k = 0

def find_frequent_1_item_sets():
    global FList
    if not FList[1]:
        for item in IList:
            if (ICount.get(item) / TCount) >= MS.get(item):
                FList[1].append([item])

def find_frequent_item_sets():
    global FList, CDict, tailCount
    k = 2
    while FList[k - 1]:
        if k == 2:
            level2_candidate_generation()
        else:
            MS_candidate_generation(k)

        for t in TList:
            for c in CList[k]:
                if set(c).issubset(set(t)):
                    CDict[tuple(c)] = CDict.get(tuple(c), 0) + 1

                if set(c[1:]).issubset(set(t)):
                    tailCount[tuple(c)] = tailCount.get(tuple(c), 0) + 1

        for c in CList[k]:
            if CDict.get(tuple(c)) is not None:
                if CDict.get(tuple(c)) / TCount >= MS[c[0]]:
                    FList[k].append(c[:])
        k += 1

def apply_cannot_be_together_constraint():
    global cannotBeTogether, FList
    for fListIndex in range(len(FList)):
        listIndex = len(FList[fListIndex]) - 1
        while listIndex > -1:
            subset = FList[fListIndex][listIndex]
            for incompatible_set in cannotBeTogether:
                if set(incompatible_set) <= set(subset):
                    FList[fListIndex].pop(listIndex)
                    break
            listIndex -= 1

def apply_must_have_constraint():
    global mustHave, FList
    for fListIndex in range(len(FList)):
        listIndex = len(FList[fListIndex]) - 1
        while listIndex > -1:
            subset = FList[fListIndex][listIndex]
            if set(mustHave).isdisjoint(subset):
                FList[fListIndex].pop(listIndex)
            listIndex -= 1

def output():
    global FList, tailCount, ICount, TList
    FList = [x for x in FList if x]
    with open("result1-2.txt", "w") as file3:
        count = 0
        file3.write("\nFrequent 1-itemsets\n")
        for t in FList[0]:
            count += 1
            file3.write('\t' + str(ICount[t[0]]) + ' : { ' + str(t[0]) + ' }\n')
        file3.write("\tTotal number of frequent 1-itemsets = " + str(count) + "\n")

        for i in range(1, len(FList)):
            count = 0
            file3.write('\n\nFrequent ' + str(i + 1) + '-itemsets\n')
            for t in FList[i]:
                tCount = sum(1 for j in TList if set(j) >= set(t))
                count += 1
                file3.write('\t' + str(tCount) + ' : {' + str(t).replace("[", "").replace("]", "") + '}\n')
                file3.write('Tail Count = ' + str(tailCount.get(tuple(t), 0)) + "\n")
            file3.write('\n\tTotal number of frequent ' + str(i + 1) + '-itemsets = ' + str(count) + "\n")


if __name__ == "__main__":
    main()
