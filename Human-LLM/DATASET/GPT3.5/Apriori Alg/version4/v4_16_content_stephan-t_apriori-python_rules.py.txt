import pandas as pd
from itertools import combinations
from apriori import candidate_gen

def generate_association_rules(T, frequent_itemsets, min_confidence):
    """
    Generate association rules based on frequent itemsets and minimum confidence.
    
    Parameters:
    - T: Total number of transactions
    - frequent_itemsets: Dictionary containing frequent itemsets
    - min_confidence: Minimum confidence threshold for generating association rules
    """
    for size in range(2, len(frequent_itemsets)):
        # Start from frequent 2-itemsets
        for j in range(len(frequent_itemsets[size])):
            # Extract frequent k-itemset and its count
            frequent_k_itemset = frequent_itemsets[size].loc[j, ['items', 'count']]
            # Generate 1-item consequents
            consequents = tuple(combinations(frequent_k_itemset[0], 1))

            for c in consequents:
                # Calculate antecedents for association rules
                antecedents = tuple(sorted(set(frequent_k_itemset[0]) - set(c)))
                antecedent_support_count = int(frequent_itemsets[len(antecedents)]['count'][frequent_itemsets[len(antecedents)]['items'] == antecedents])

                if frequent_k_itemset[1] / antecedent_support_count >= min_confidence:
                    # Output association rules meeting minimum confidence
                    print(f"{antecedents} -> {c} [sup = {frequent_k_itemset[1]}/{T}, conf = {frequent_k_itemset[1]}/{antecedent_support_count}]")

                    # Store 1-item consequents meeting minimum confidence
                    if 'H' not in locals():
                        H = {1: pd.DataFrame(pd.Series((c,)), columns=['items'])}
                    else:
                        H[1] = H[1].append(pd.Series((c,), index=['items']), ignore_index=True)

                    # Generate all rules from 1-item consequents
                    generate_all_rules(T, frequent_itemsets, frequent_k_itemset, H, 1, min_confidence)

            # Delete dataframe for next iteration
            if 'H' in locals():
                del H

def generate_all_rules(T, frequent_itemsets, frequent_k_itemset, H, m, min_confidence):
    """
    Generate all association rules from frequent itemsets.
    
    Parameters:
    - T: Total number of transactions
    - frequent_itemsets: Dictionary containing frequent itemsets
    - frequent_k_itemset: Frequent k-itemset
    - H: DataFrame containing 1-item consequents
    - m: Current itemset size
    - min_confidence: Minimum confidence threshold for generating association rules
    """
    if len(frequent_k_itemset[0]) > m + 1 and not H[m].empty:
        # Generate consequent supersets
        H = candidate_gen(H, m)

        for h in H[m + 1]['items']:
            # Calculate antecedents for association rules
            antecedents = tuple(sorted(set(frequent_k_itemset[0]) - set(h)))
            antecedent_support_count = int(frequent_itemsets[len(antecedents)]['count'][frequent_itemsets[len(antecedents)]['items'] == antecedents])
            confidence = frequent_k_itemset[1] / antecedent_support_count

            if confidence >= min_confidence:
                # Output association rules meeting minimum confidence
                print(f"{antecedents} -> {h} [sup = {frequent_k_itemset[1]}/{T}, conf = {frequent_k_itemset[1]}/{antecedent_support_count}]")
            else:
                # Delete consequents not meeting minimum confidence
                H[m + 1] = H[m + 1].drop(H[m + 1].index[-1])

        # Generate all rules from (m + 1)-item consequents
        generate_all_rules(T, frequent_itemsets, frequent_k_itemset, H, m + 1, min_confidence)
