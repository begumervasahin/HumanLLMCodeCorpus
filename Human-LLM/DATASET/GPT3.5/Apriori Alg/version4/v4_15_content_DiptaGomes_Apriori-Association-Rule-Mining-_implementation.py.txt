# Apriori Association Rule Mining

# Importing the necessary libraries
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from apyori import apriori

# Data Preprocessing
# Read the dataset from the CSV file with headers
dataset = pd.read_csv('Nursery.csv', header=1)

# Initialize an empty list to store transactions
transactions = []

# Extract transactions from the dataset
for i in range(1, 9409):
    # Convert each row of the dataset to a list of strings and append to transactions
    transactions.append([str(dataset.values[i, j]) for j in range(0, 9)])

# Applying Apriori algorithm on the dataset
# Set the parameters for minimum support, confidence, lift, and length of the rule
min_support = 0.04
min_confidence = 0.90
min_lift = 3
min_length = 2

# Run Apriori algorithm to discover association rules
rules = apriori(transactions,
                min_support=min_support,
                min_confidence=min_confidence,
                min_lift=min_lift,
                min_length=min_length)

# Convert the discovered rules into a list for further processing
results = list(rules)

# Output the discovered association rules
for item in results:
    # Extract the items in the rule
    pair = item[0]
    items = [x for x in pair]

    # Print the association rule
    print("Rule: " + items[0] + " -> " + items[1])

    # Print the support of the rule
    print("Support: " + str(item[1]))

    # Print the confidence of the rule
    print("Confidence: " + str(item[2][0][2]))

    # Print the lift of the rule
    print("Lift: " + str(item[2][0][3]))

    # Print separator for clarity
    print("=====================================")
