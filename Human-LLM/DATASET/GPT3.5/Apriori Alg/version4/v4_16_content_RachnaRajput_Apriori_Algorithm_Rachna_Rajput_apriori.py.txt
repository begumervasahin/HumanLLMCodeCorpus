import json
import sys

class AprioriAlgorithm:

    def apriori(self, json_data, show_output):
        array = json_data
        threshold = 0.3 * len(array)
        phase1_list = []

        # Finding unique items
        unique_items = set()
        for item in array:
            unique_items.update(item)

        # Generating C1
        c1 = [[elem] for elem in unique_items]
        if show_output:
            print("C1:", c1)

        # Counting occurrences for C1
        counts = {}
        for arr in array:
            for subarr in arr:
                counts[subarr] = counts.get(subarr, 0) + 1

        # Generating L1
        l1 = [[elem] for elem in unique_items if counts[elem] >= threshold]
        if show_output:
            print("L1:", l1)
        else:
            phase1_list.extend(l1)

        # Generating C2
        c2 = []
        k = 1
        for i in range(len(l1)):
            for j in range(i + 1, len(l1)):
                pair = [l1[i][0], l1[j][0]]
                c2.append(pair)
                k += 1
        if show_output:
            print("C2:", c2)

        # Counting occurrences for C2
        counts_c2 = {}
        for index, elem in enumerate(c2):
            count = sum(1 for arr in array if set(elem) <= set(arr))
            counts_c2[index + 1] = count

        # Generating L2
        l2 = [c2[i - 1] for i in range(1, len(counts_c2) + 1) if counts_c2[i] >= threshold]
        if show_output:
            print("L2: ", l2)
        else:
            phase1_list.extend(l2)

        # Generating frequent itemsets L3 and beyond
        y = 3
        while l2:
            c3 = []
            for i in range(len(l2) // 2 + 1):
                for j in range(i + 1, len(l2) // 2 + 1):
                    if l2[i][0] == l2[j][0]:
                        item_set = []
                        for x in range(1, len(l2[i])):
                            item_set.extend([l2[i][x], l2[j][x]])
                        item_set.sort()
                        for item in l2:
                            if set(item_set) == set(item):
                                item_set.append(l2[i][0])
                                item_set = list(set(item_set))
                                item_set.sort()
                                c3.append(item_set)

            if show_output:
                print(f"C{y}: {c3}")

            # Counting occurrences for C3
            counts_c3 = {}
            for index, elem in enumerate(c3):
                count = sum(1 for arr in array if set(elem) <= set(arr))
                counts_c3[index + 1] = count

            # Generating L3
            l3 = [c3[i - 1] for i in range(1, len(counts_c3) + 1) if counts_c3[i] >= threshold]
            if show_output:
                print(f"L{y}: {l3}")
            else:
                phase1_list.extend(l3)

            l2 = l3
            y += 1

        return phase1_list

if __name__ == '__main__':
    if len(sys.argv) < 2:
        print("Usage: python script_name.py <json_file_path>")
        sys.exit(1)

    apriori_instance = AprioriAlgorithm()

    with open(sys.argv[1]) as json_file:
        json_data = json.load(json_file)

    checker = True
    apriori_instance.apriori(json_data, checker)
