from itertools import combinations

def read_file(filename):
    """
    Read data from a file.
    """
    with open(filename, 'r') as fData:
        return [line.strip() for line in fData]

def find_single_candidates(dataset):
    """
    Find single candidates from the dataset.
    """
    singles = []
    for transaction in dataset:
        items = transaction.split(',')
        for item in items:
            if item not in singles:
                singles.append(item)
    return singles

def count_candidates(dataset, singles):
    """
    Count occurrences of single candidates in the dataset.
    """
    candidate_count = {}
    for transaction in dataset:
        items = transaction.split(',')
        for item in items:
            if item in singles:
                candidate_count.setdefault(item, 0)
                candidate_count[item] += 1
    return candidate_count

def total_items(dataset):
    """
    Count total number of items in the dataset.
    """
    total = 0
    for transaction in dataset:
        total += len(transaction.split(','))
    return total

def check_support(item_count, candidate_count, support):
    """
    Check the support count for candidate items.
    """
    support_numbers = {}
    for candidate in candidate_count:
        if candidate_count[candidate] / item_count >= support:
            support_numbers[candidate] = candidate_count[candidate] / item_count
    return support_numbers

def check_support_count(item_count, candidate_count, support):
    """
    Check the support count for candidate items.
    """
    result = []
    for candidate in candidate_count:
        if candidate_count[candidate] / item_count >= support:
            result.append(candidate)
    return result

def frequent_pairs(frequent_items, dataset, item_count, support):
    """
    Find frequent pairs based on the support count.
    """
    results = {}
    for transaction in dataset:
        for item in combinations(transaction.split(','), 2):
            results.setdefault(item, 0)
            results[item] += 1

    mark_for_deletion = []
    for key in results:
        if results[key] / item_count < support:
            mark_for_deletion.append(key)
        else:
            results[key] = results[key] / item_count

    for key in mark_for_deletion:
        results.pop(key, None)

    return results

    data_file = 'mushroom.data'
    support = 0.03

    data = read_file(data_file)

    candidates1 = find_single_candidates(data)
    print("Single Candidates:", candidates1)

    frequent_candidates = check_support(total, counts, support)

    support_counts = check_support_count(total, counts, support)
