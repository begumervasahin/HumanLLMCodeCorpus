import json
import sys
from MapReduce import MapReduce

# Initialize MapReduce instance
mr = MapReduce.MapReduce()
dictionary = {}

class Rachna_Rajput_son_phase2:
    def mapper(record):
        f = open(sys.argv[1])
        mylist = []

        # Process each line in the file
        for line in f:
            chunkdata = json.loads(line)
            count = 0
            length = len(chunkdata)

            # Count occurrences
            for chunkline in chunkdata:
                if set(chunkline) >= set(record):
                    count += 1

            mylist.extend([count, length])
            recordtuple = tuple(record)
            dictionary[recordtuple] = mylist

        # Emit intermediate results
        mr.emit_intermediate(recordtuple, mylist)

    def reducer(key, value):
        count = 0
        mylist = []

        # Sum up the counts
        i = 0
        while i < len(value[0]):
            count += value[0][i]
            i += 2

        mylist.append(key)
        mylist.append(count)
        mr.emit(mylist)

if __name__ == '__main__':
    # Open input data
    inputdata = open(sys.argv[2])

    # Execute MapReduce job
    output = mr.execute(inputdata, mapper, reducer)
