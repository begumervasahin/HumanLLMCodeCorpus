import itertools

def get_subsets(S):
    """
    Return a list of subsets of S, where each subset contains len(S) - 1 elements.
    """
    subset_length = len(S) - 1
    return [list(i) for i in itertools.combinations(S, subset_length)]

def is_subset_exist(tar, L):
    """
    Check if each value in L is a subset of tar.
    Returns 1 if all values in L are subsets of tar, otherwise returns 0.
    """
    tar_set = [set(i[0]) for i in tar]
    is_in = 1
    for i in L:
        for j in tar_set:
            if j < set(i):
                is_in = 0
                return is_in
    return is_in

def Apriori(tar, thr, le):
    """
    Implement the Apriori algorithm for frequent item set mining.
    """
    curFrequ = {}

    # Count occurrences of each item
    for key in tar.keys():
        items = tar[key]
        for item in items:
            if item not in curFrequ.keys():
                curFrequ[item] = 1
            else:
                curFrequ[item] += 1

    # Calculate support for each item
    tarLen = len(list(tar.keys()))
    keys = list(curFrequ.keys())
    for key in keys:
        curFrequ[key] = curFrequ[key] / tarLen
        if curFrequ[key] < thr:
            del curFrequ[key]

    curFrequ = [[key, curFrequ[key]] for key in curFrequ.keys()]

    if le == 1:
        return curFrequ

    tempLe = 2
    while tempLe <= le:
        tempList = []
        for i in range(len(curFrequ)):
            for j in range(i + 1, len(curFrequ)):
                temp = []
                if tempLe == 2:
                    temp = [curFrequ[i][0], curFrequ[j][0]]
                else:
                    temp = list(set(curFrequ[i][0]) | set(curFrequ[j][0]))

                if len(temp) != tempLe:
                    continue

                temp.sort()

                if is_subset_exist(curFrequ, get_subsets(temp)) == 1:
                    if temp not in tempList:
                        tempList.append(temp)

        curFrequ = []
        for item in tempList:
            tempLen = 0
            for key in tar.keys():
                if set(item) <= set(tar[key]):
                    tempLen += 1
            tempLen = tempLen / tarLen
            if tempLen >= thr:
                curFrequ.append([item, tempLen])

        tempLe += 1

    return curFrequ
