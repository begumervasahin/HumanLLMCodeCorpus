import csv
import sys

# Check if the correct number of arguments is provided
if len(sys.argv) != 3:
    print("Execution format: python dat2csv.py <source_file> <destination_file>")
else:
    # Read data from the source file, strip extra spaces, and split lines into lists of values
    with open(sys.argv[1], 'r') as input_file:
        lines = input_file.readlines()
        formatted_lines = [line.strip().split() for line in lines]

    # Write the formatted data into the destination file in CSV format
    with open(sys.argv[2], 'w') as output_file:
        csv_writer = csv.writer(output_file)
        csv_writer.writerows(formatted_lines)
