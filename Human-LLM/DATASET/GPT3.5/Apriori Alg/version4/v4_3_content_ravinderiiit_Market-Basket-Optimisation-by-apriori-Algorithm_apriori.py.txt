# Apriori Implementation for Market Basket Optimization

# Importing necessary libraries
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from apyori import apriori

# Data Preprocessing
# Read the dataset
dataset = pd.read_csv('Market_Basket_Optimisation.csv', header=None)

# Convert the dataset into a list of transactions
transactions = []
for i in range(0, 7501):
    transactions.append([str(dataset.values[i, j]) for j in range(0, 20)])

# Training Apriori on the dataset
# Set the minimum support, confidence, lift, and length for the association rules
min_support = 0.003
min_confidence = 0.2
min_lift = 3
min_length = 2

# Generate association rules using Apriori algorithm
rules = apriori(transactions, 
                min_support=min_support, 
                min_confidence=min_confidence, 
                min_lift=min_lift, 
                min_length=min_length)

# Convert the rules into a list for visualization
results = list(rules)
