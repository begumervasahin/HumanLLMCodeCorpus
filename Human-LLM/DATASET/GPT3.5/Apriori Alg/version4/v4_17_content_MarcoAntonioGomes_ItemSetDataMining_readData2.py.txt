import csv
import copy

# Open a file to write transactions
output_file = open('dataCaixaAno2015-MesAgosto.dat', 'w')

# Paths to CSV files
transactions_file_path = 'caixa.csv'
products_file_path = 'produtos.csv'

# Lists to store data
vendaProd = []
listProd = []
vendaProdMat = []

# Function to load data from CSV files
def load_data_from_csv():
    with open(transactions_file_path, newline='') as file:
        reader = csv.reader(file, delimiter=';')
        line_number = 0
        for line in reader:
            if line_number > 0:
                venda = [line[6]]
                vendaProd.append(venda)
            line_number += 1

    with open(products_file_path, newline='') as file:
        reader = csv.reader(file, delimiter=';')
        next(reader)
        for line in reader:
            listProd.append((line[0], line[1]))

    vendaProdMat = copy.deepcopy(vendaProd)

    with open(transactions_file_path, newline='') as file:
        reader = csv.reader(file, delimiter=';')
        for line in reader:
            for i in range(len(vendaProd)):
                if vendaProd[i][0] == line[6]:
                    vendaProd[i].append(line[4])

    with open(products_file_path, newline='') as file:
        reader = csv.reader(file, delimiter=';')
        next(reader)
        for line in reader:
            listProd.append((line[0], line[1]))

    for i in range(len(vendaProd)):
        for j in range(1, len(vendaProd[i])):
            for k in range(len(listProd)):
                aux = listProd[k]
                if vendaProd[i][j] == aux[0]:
                    vendaProdMat[i].append(aux[1])

    return vendaProdMat

# Load data and print transactions
vendaProdMat = load_data_from_csv()
print(vendaProdMat)

# Write transactions to the output file
for i in range(len(vendaProdMat)):
    if len(vendaProdMat[i]) > 2:
        value = ''
        for j in range(1, len(vendaProdMat[i])):
            value += vendaProdMat[i][j].replace(" ", "_") + " "
        value += "\n"
        output_file.write(value)

# Close the output file
output_file.close()
