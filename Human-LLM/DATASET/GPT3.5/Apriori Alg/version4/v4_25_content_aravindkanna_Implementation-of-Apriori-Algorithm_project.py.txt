from Trie import trieNode
from math import ceil
from otherMethods import generate, prune, itemSetsCount, printAssociateRules
import sys
import fileinput
import time

def main():
    start_time = time.time()

    # Read parameters from the configuration file
    with open('config.csv') as conf:
        params = {}
        for line in conf:
            key, value = line.strip().split(",")
            params[key] = value

    # Extract parameters from the configuration
    inFile = params["input"]
    outFile = params["output"]
    flag = int(params["flag"])
    minsup = float(params["support"])
    mincon = float(params["confidence"])

    # Redirect standard output to the output file
    sys.stdout = open(outFile, 'w')

    # Process the input file to count singletons and transactions
    singletons = {}
    noOfTransactions = 0
    with open(inFile) as infp:
        for line in infp:
            line = line.strip()
            items = line.split(",")
            noOfTransactions += 1
            for item in items:
                singletons[item] = singletons.get(item, 0) + 1

    # Find frequent singletons
    freqSingles = []
    freqSinglesCount = []
    print("FreqCount")
    for item, count in singletons.items():
        minTransactions = ceil(noOfTransactions * minsup)
        if count >= minTransactions:
            print(item)
            freqSingles.append([item])
            freqSinglesCount.append(count)

    freqSingles.sort()

    # Initialize Trie for frequent itemsets
    FreqsTrie = trieNode()
    FreqsTrie.insertAll(freqSingles, freqSinglesCount)

    FreqItemSets = freqSingles
    currSizeList = freqSingles
    allfreqs = len(currSizeList)

    # Generate frequent itemsets
    while True:
        nextSizeList = generate(currSizeList)
        prunedList = prune(FreqsTrie, nextSizeList)
        if len(prunedList) == 0:
            break

        counts = itemSetsCount(inFile, prunedList)
        FreqItems = []
        FreqCounts = []

        for i in range(len(prunedList)):
            minTransactions = ceil(noOfTransactions * minsup)
            if counts[i] >= minTransactions:
                print(",".join(prunedList[i]))
                FreqItems.append(prunedList[i])
                FreqCounts.append(counts[i])

        if len(FreqCounts) == 0:
            break

        FreqsTrie.insertAll(FreqItems, FreqCounts)
        currSizeList = FreqItems
        allfreqs += len(currSizeList)

    # Print association rules if required
    if flag == 1:
        print("RulesCount")
        noOfRules = printAssociateRules(FreqsTrie, mincon)

    sys.stdout.close()

    # Update output file with counts
    with fileinput.FileInput(outFile, inplace=True) as file:
        for line in file:
            if "RulesCount" in line and flag == 1:
                line = str(noOfRules) + "\n"
            elif "FreqCount" in line:
                line = str(allfreqs) + "\n"
            print(line, end='')

    print(time.time() - start_time)

if __name__ == "__main__":
    main()
