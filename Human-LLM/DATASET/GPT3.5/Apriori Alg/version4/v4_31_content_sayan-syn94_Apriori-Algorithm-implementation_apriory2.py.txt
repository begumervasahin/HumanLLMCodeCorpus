import sys
import csv
from collections import defaultdict
from itertools import chain, combinations
import time

# Initialize the start time
start_time = time.time()

# Dictionary to store frequent itemsets and their support
freq_set = defaultdict(int)

def read_data(file_name):
    """Reads data from the file and yields a generator."""
    with open(file_name, 'r') as file:
        for line in file:
            line = line.strip().rstrip(',')   # Remove trailing comma
            record = frozenset(line.split(','))
            yield record

def generate_transactions(data_iterator):
    """Generates transactions and unique items."""
    transaction_list = list()
    items = set()
    for record in data_iterator:
        transaction_list.append(record)
        for item in record:
            items.add(frozenset([item]))  # Generate 1-itemSets
    return items, transaction_list

def join_item_sets(item_set, length):
    """Joins sets to return n-length itemset where n=length."""
    return set([i.union(j) for i in item_set for j in item_set if len(i.union(j)) == length])

def calculate_support(item_list, transaction, min_supp):
    """Calculates the support for each itemset and filters based on min_support."""
    s = defaultdict(int)
    item_set = set()
    for i in item_list:
        for t in transaction:
            if i.issubset(t):
                s[i] += 1
    for i, v in s.items():
        support = float(v) / len(transaction)
        if support >= min_supp:
            item_set.add(i)
            freq_set[i] = support
    return item_set

def generate_subsets(items):
    """Returns non-empty subsets of items."""
    return chain(*[combinations(items, i + 1) for i, a in enumerate(items)])

def run_apriori(data_iter, min_support, min_confidence):
    """Runs the Apriori algorithm."""
    item_set, transaction_list = generate_transactions(data_iter)
    large_set = dict()
    # Generate frequent itemsets
    one_c_set = calculate_support(item_set, transaction_list, min_support)
    current_l_set = one_c_set
    k = 2
    while current_l_set:
        large_set[k - 1] = current_l_set
        current_l_set = join_item_sets(current_l_set, k)
        current_c_set = calculate_support(current_l_set, transaction_list, min_support)
        print(k)
        print(current_c_set)
        current_l_set = current_c_set
        k += 1
    # Generate association rules
    for key, value in list(large_set.items())[1:]:
        for item in value:
            _subsets = map(frozenset, [x for x in generate_subsets(item)])
            for element in _subsets:
                remain = item.difference(element)
                if len(remain) > 0:
                    confidence = freq_set[item] / freq_set[element]
                    if confidence >= min_confidence:
                        print(element, "=>", remain, "Confidence:", confidence)

if __name__ == "__main__":
    min_support = 0.01
    min_confidence = 0.01
    input_file = 'T10I4D100k-copy.txt'
    input_data = read_data(input_file)
    run_apriori(input_data, min_support, min_confidence)

    # Calculate and print the elapsed time
    elapsed_time = time.time() - start_time
    print("Execution time:", elapsed_time)
