import re
import pandas as pd

def preprocessing_frame(frame):
    """
    Preprocesses the skills data frame by cleaning and filtering the skills.

    Args:
    - frame: pandas DataFrame representing the skills data frame

    Returns:
    - preprocessed_data: list of lists containing preprocessed skills
    """
    # List of specific skills to include
    specific_skills = ['c#', '.net', 'asp.net', 'c++']
    preprocessed_data = []

    # Compile regular expression patterns for filtering
    pattern1 = re.compile(r'([a-z].*[0-9])')
    pattern2 = re.compile(r'([a-z].*[a-z])')

    # Loop through each row in the DataFrame
    for _, rows in frame.iteritems():
        # Initialize variables
        skills_list = []
        
        # Remove unwanted characters from the row
        rows = rows.replace('>', '').replace('<', '').replace('...', '')

        # Split the row into individual skills
        rows_split = rows.split(',')

        # Iterate over each skill in the row
        for element in rows_split:
            # Split the skill into words
            space_split = element.split(' ')

            # Only include skills with less than or equal to 4 words
            if len(space_split) <= 4:
                # Convert to lowercase and remove leading/trailing whitespace
                element = element.lower().strip()

                # Filter out certain patterns and specific skills
                if (pattern1.match(element) or pattern2.match(element) or element in specific_skills):
                    # Clean up the skill text
                    element = element.replace('?', '').replace('&amp;', '').replace('&', '').replace('#x', '').replace('#x22', '').replace('.', '')        

                    # Remove trailing periods
                    if element.endswith('.'):
                        element = element[:-1]

                    # Remove leading periods if it's not part of '.net'
                    if element.startswith('.') and not element.startswith('.net'):
                        element = element[1:]

                    # Add the skill to the list if it's not already present
                    if element not in skills_list:
                        skills_list.append(element)

        # Append the list of skills to the preprocessed data
        preprocessed_data.append(skills_list)

    return preprocessed_data
