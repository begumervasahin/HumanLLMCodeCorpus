import csv

class Rule:
    """
    Represents an association rule in the Apriori algorithm.
    """

    def __init__(self, antecedent, consequent, support, confidence, lift, leverage):
        """
        Initialize a Rule object with its attributes.
        """
        self.antecedent = antecedent
        self.consequent = consequent
        self.support = support
        self.confidence = confidence
        self.lift = lift
        self.leverage = leverage

    def __str__(self):
        """
        Return a string representation of the rule.
        """
        return (
            f"{list(self.antecedent)} -> {list(self.consequent)} - "
            f"support: {self.support:.2f}, confidence: {self.confidence:.2f}, "
            f"lift: {self.lift:.2f}, leverage: {self.leverage:.2f}"
        )

def apriori(transactions, min_support=0.01, min_confidence=0.1):
    """
    Implement the Apriori algorithm for generating association rules.
    """
    num_transactions = len(transactions)
    min_support_count = min_support * num_transactions
    num_items = max(item for transaction in transactions for item in transaction)
    L = {frozenset({}): 1}

    def support(itemset):
        """
        Calculate the support of an itemset.
        """
        return L[itemset]

    def confidence(antecedent, consequent):
        """
        Calculate the confidence of a rule.
        """
        return support(antecedent | consequent) / support(antecedent)

    def lift(antecedent, consequent):
        """
        Calculate the lift of a rule.
        """
        return confidence(antecedent, consequent) / support(consequent)

    def leverage(antecedent, consequent):
        """
        Calculate the leverage of a rule.
        """
        return support(antecedent | consequent) - support(antecedent) * support(consequent)

    # Count occurrences of one-item sets
    item_counts = [0] * num_items
    for transaction in transactions:
        for item in transaction:
            item_counts[item - 1] += 1

    # Find frequent one-item sets
    L = {frozenset({x}): n / num_transactions for x, n in zip(range(1, num_items + 1), item_counts) if n >= min_support_count}

    while len(L):
        C = {}
        for itemset1 in L:
            for itemset2 in L:
                if len(itemset1 - itemset2) == len(itemset2 - itemset1) == 1:
                    C[itemset1 | itemset2] = 0

        for transaction in transactions:
            for itemset in C:
                if itemset <= transaction:
                    C[itemset] += 1

        L = {itemset: count / num_transactions for itemset, count in C.items() if count >= min_support_count}

    # Generate association rules
    rules = [Rule(itemset, frozenset({}), support(itemset), 1., 1., 0.) for itemset in L]
    strong_rules = []

    while len(rules):
        strong_rules += rules
        next_rules = []

        for rule in rules:
            for item in rule.antecedent:
                antecedent = rule.antecedent - {item}
                consequent = rule.consequent | {item}
                if confidence(antecedent, consequent) >= min_confidence:
                    next_rules.append(Rule(antecedent, consequent, support(antecedent | consequent), 
                                            confidence(antecedent, consequent), lift(antecedent, consequent), 
                                            leverage(antecedent, consequent)))

        rules = next_rules

    return strong_rules

def load_transactions(path='retail.dat'):
    """
    Load transactions from the file located at the given path.
    """
    transactions = []
    with open(path, 'r') as file:
        reader = csv.reader(file, delimiter=' ')
        for row in reader:
            transactions.append({int(item) for item in row if item})
    return transactions

# Example usage:
if __name__ == "__main__":
    transactions = load_transactions('retail.dat')
    rules = apriori(transactions)
    for rule in rules:
        print(rule)
