import numpy as np
import pandas as pd
from apyori import apriori

# Importing the dataset
dataset = pd.read_csv('FYP.csv')

# Extracting relevant columns for Apriori algorithm
customer_transactions = dataset.iloc[:, [5, 10]].values

# Sorting by customer ID
customer_transactions = customer_transactions[customer_transactions[:, 0].argsort()]

# Prepare data for Apriori algorithm
transactions = []
current_transaction = []

for i in range(len(customer_transactions)):
    if i > 0 and customer_transactions[i][0] != customer_transactions[i - 1][0]:
        transactions.append(list(set(current_transaction)))
        current_transaction = []
    current_transaction.append(str(customer_transactions[i][1]))

transactions.append(list(set(current_transaction)))

# Applying Apriori algorithm
min_support = 0.02
min_confidence = 0.2
min_lift = 1.4
min_length = 2

rules = apriori(transactions, 
                min_support=min_support, 
                min_confidence=min_confidence, 
                min_lift=min_lift, 
                min_length=min_length)
results = list(rules)

# Function to inspect the results
def inspect_results(results):
    extracted_results = []
    for result in results:
        rhs = tuple(result.ordered_statistics[0].items_base)
        lhs = tuple(result.ordered_statistics[0].items_add)
        support = result.support
        confidence = result.ordered_statistics[0].confidence
        lift = result.ordered_statistics[0].lift
        extracted_results.append((rhs, lhs, support, confidence, lift))
    return extracted_results

# Extracting and inspecting the results
extracted_results = inspect_results(results)

# Extracting LHS and RHS values
final_lhs = [extracted_results[i][1][0] for i in range(len(extracted_results))]
final_rhs = [extracted_results[i][0] for i in range(len(extracted_results))]

# Printing the final LHS and RHS values
print("Final LHS Values:")
print(final_lhs)
print("Final RHS Values:")
print(final_rhs)
