import sys
import csv

class APRIORI:
    def __init__(self, datafile, min_sup, min_conf, data):
        self.datafile = datafile
        self.min_sup = float(min_sup)
        self.min_conf = float(min_conf)
        self.data = data
        self.records = []
        self.itemsets = []
        self.rules = {}

    def convert_into_csv(self):
        with open(self.datafile, 'r') as input_file:
            lines = input_file.readlines()
            transactions = [line.strip().split(" ") for line in lines]

        with open(self.data, 'w', newline='') as output_file:
            csv_writer = csv.writer(output_file)
            csv_writer.writerows(transactions)

    def load_data(self):
        C_1 = {}
        with open(self.data, 'r') as data_file:
            for line in data_file:
                row = [item for item in line.strip().split(',') if item]
                self.records.append(set(row))

                for item in row:
                    item_set = frozenset([item])
                    C_1[item_set] = C_1.get(item_set, 0) + 1

        with open('itemsets.txt', 'w') as item_writer:
            item_writer.write(f'1-itemsets C1: {str(C_1)}\n')

        self.itemsets.append(C_1)
        self.gen_freq_1_itemset()

    def gen_freq_1_itemset(self):
        num_records = len(self.records)
        L_1 = self.itemsets[0]
        L_1 = {item: sup_value / num_records for item, sup_value in L_1.items() if self.min_sup <= sup_value}
        self.itemsets[0] = L_1

    def gen_freq_k_itemsets(self):
        base_set = list(self.itemsets[0])
        while base_set:
            itemset_len = len(base_set[0]) + 1
            candidate_itemsets = {}

            for i in range(len(base_set)):
                for j in range(i + 1, len(base_set)):
                    currentset = base_set[i] | base_set[j]
                    if len(currentset) == itemset_len:
                        count = sum(1 for set in base_set if set <= currentset)
                        if count == len(currentset):
                            candidate_itemsets[currentset] = 0

            if candidate_itemsets:
                for record in self.records:
                    for cand_set in candidate_itemsets:
                        if cand_set <= record:
                            candidate_itemsets[cand_set] += 1

                for cand_set in list(candidate_itemsets):
                    support_value = candidate_itemsets[cand_set] / len(self.records)
                    if support_value < self.min_sup:
                        del candidate_itemsets[cand_set]
                    else:
                        candidate_itemsets[cand_set] = support_value

            if candidate_itemsets:
                self.itemsets.append(candidate_itemsets)
            base_set = list(candidate_itemsets)

    def truncate(self, val, n):
        s = '{}'.format(val)
        if 'e' in s or 'E' in s:
            return '{0:.{1}f}'.format(val, n)
        i, p, d = s.partition('.')
        return '.'.join([i, (d+'0'*n)[:n]])

    def generate_rules(self):
        self.gen_freq_k_itemsets()
        for itemsets in self.itemsets[1:]:
            for set in list(itemsets):
                for item in set:
                    left_items = set - frozenset([item])
                    item_conf = self.itemsets[len(set) - 1][set] / self.itemsets[len(left_items) - 1][left_items]
                    if self.min_conf <= item_conf:
                        rule = f"[{', '.join(list(left_items))}] => [{item}] (Confidence: {self.truncate(item_conf * 100, 5)}%, Support: {self.truncate((itemsets[set] * 100), 5)}%)"
                        self.rules[rule] = item_conf

    def apriori_output_to_file(self):
        with open('output.txt', 'w') as writer:
            writer.write(f'Dataset used : {self.datafile}\n')
            writer.write(f'Dataset converted to : {self.data}\n')
            writer.write(f'Minimum Support: {self.min_sup * 100}%\n')
            writer.write(f'Minimum Confidence: {self.min_conf * 100}%\n')

            all_itemsets = [itemset for itemsets in self.itemsets for itemset in itemsets]

            writer.write('\nFrequent itemsets and their support values\n')
            for itemset in all_itemsets:
                writer.write(f"{list(itemset)}, Support: {self.truncate(self.itemsets[len(itemset) - 1][itemset] * 100, 5)}%\n")

            writer.write('\nAssociation rules and their support and confidence values \n')
            for rule, confidence in self.rules.items():
                writer.write(f"{rule}\n")

            writer.write('\nMinimum support and minimum confidence are user defined values given as arguments with the command. All the values of support and confidence are truncated to 5 decimals while displaying output.')

if __name__ == '__main__':
    if len(sys.argv) != 4:
        print("Inputs required are: [DATASET] [min_sup] [min_conf]")
        sys.exit(2)

    apr = APRIORI(sys.argv[1], sys.argv[2], sys.argv[3], 'data.csv')
    apr.convert_into_csv()
    apr.load_data()
    apr.generate_rules()
    apr.apriori_output_to_file()

print("Done")
