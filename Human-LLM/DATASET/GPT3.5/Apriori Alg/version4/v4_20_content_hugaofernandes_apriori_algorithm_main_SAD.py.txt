# -*- coding: utf-8 -*-

from itertools import combinations
import numpy as np
import pandas as pd


def apriori(data, support_threshold, min_length, confidence_threshold):
    # Convert the data into one-hot encoded format
    one_hot_encoded_data = pd.get_dummies(data.unstack().dropna()).groupby(level=1).sum()
    num_columns, num_rows = one_hot_encoded_data.shape
    patterns = []
    iterations = 0

    # Generate frequent itemsets
    for num_cols in range(min_length, num_rows + 1):
        for cols in combinations(one_hot_encoded_data, num_cols):
            # Calculate support for the itemset
            support_count = one_hot_encoded_data[list(cols)].all(axis=1).sum()
            support = float(support_count) / num_columns

            # Calculate confidence for the itemset
            aux_cols = list(cols)
            del aux_cols[-1]
            confidence = 0
            denominator_count = one_hot_encoded_data[aux_cols].all(axis=1).sum()
            if denominator_count != 0:
                confidence = float(support_count) / denominator_count

            # Add the pattern to the list
            patterns.append([",".join(cols), support * 100, confidence * 100])
            iterations += 1

    # Create DataFrame for patterns and filter by support and confidence thresholds
    patterns_df = pd.DataFrame(patterns, columns=["Pattern", "Support", "Confidence"])
    results = patterns_df[(patterns_df.Support >= support_threshold) & (patterns_df.Confidence >= confidence_threshold)]

    print(results)
    print('Iterations:', iterations)


def legs_function(legs, n, label):
    if legs == n:
        return label
    return np.nan


# Read the zoo dataset
zoo_data = pd.read_csv('zooOriginal.csv', sep=',', header=None)
zoo_data.columns = ['name', 'hair', 'feathers', 'eggs', 'milk', 'airborne', 'aquatic', 'predator', 'toothed',
                    'backbone', 'breathes', 'venomous', 'fins', 'legs', 'tail', 'domestic', 'catsize', 'type']
zoo_data = zoo_data.drop(['name'], axis=1)
zoo_data = zoo_data.drop(['type'], axis=1)

# Process the legs feature
legs = zoo_data['legs']
zoo_data = zoo_data.drop(['legs'], axis=1)
for col in zoo_data.columns.values:
    zoo_data[col] = zoo_data[col].replace(1, col)
    zoo_data[col] = zoo_data[col].replace(0, np.nan)
zoo_data['No Legs'] = legs.apply(lambda x: legs_function(x, 0, 'No Legs'))
zoo_data['2 Legs'] = legs.apply(lambda x: legs_function(x, 2, '2 Legs'))
zoo_data['4 Legs'] = legs.apply(lambda x: legs_function(x, 4, '4 Legs'))
zoo_data['5 Legs'] = legs.apply(lambda x: legs_function(x, 5, '5 Legs'))
zoo_data['6 Legs'] = legs.apply(lambda x: legs_function(x, 6, '6 Legs'))
zoo_data['8 Legs'] = legs.apply(lambda x: legs_function(x, 8, '8 Legs'))

# Run the Apriori algorithm on the zoo dataset
apriori(zoo_data, 30, 4, 97)
