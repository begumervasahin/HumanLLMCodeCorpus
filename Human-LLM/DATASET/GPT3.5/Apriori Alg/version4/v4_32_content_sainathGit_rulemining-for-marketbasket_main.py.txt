import sys

# Get minimum support and minimum confidence from command-line arguments
min_support = float(sys.argv[2])
min_confidence = float(sys.argv[4])

# Read transactions from the input file
db = []
with open(sys.argv[5], "r") as tf:
    for line in tf:
        db.append([int(x) for x in line.split()])

# Calculate minimum support count based on the number of transactions
min_support = min_support * len(db)


def clear_infrequent_items(table):
    """
    Remove infrequent items from the itemset table.
    """
    infrequent_items = [k for k, v in table.items() if v < min_support]
    for item in infrequent_items:
        del table[item]


def get_key(l1, l2):
    """
    Combine two itemsets to generate a new key.
    """
    key = []
    last = len(l1) - 1
    for i in range(last):
        key.append(l1[i])
    key.append(l1[last])
    key.append(l2[last])
    return tuple(key)


def can_combine(l1, l2):
    """
    Check if two itemsets can be combined to generate a new itemset.
    """
    last = len(l1) - 1
    for i in range(last):
        if l1[i] != l2[i]:
            return False
    if l1[last] >= l2[last]:
        return False
    return True


def apriori_gen(table):
    """
    Generate candidate itemsets by combining frequent itemsets.
    """
    new_table = {}
    for l1 in table.keys():
        for l2 in table.keys():
            if can_combine(l1, l2):
                new_table[get_key(l1, l2)] = 0
    return new_table


def count_itemset_occurrences(table):
    """
    Count occurrences of candidate itemsets in transactions.
    """
    for k in table.keys():
        for t in db:
            if set(k).issubset(set(t)):
                table[k] += 1


def powerset(seq):
    """
    Returns all the subsets of this set. This is a generator.
    """
    if len(seq) <= 1:
        yield seq
        yield []
    else:
        for item in powerset(seq[1:]):
            yield [seq[0]] + item
            yield item


def print_rule(s, l_s, conf):
    """
    Print association rule.
    """
    print(s, " ==> ", l_s, "              ", conf)


def print_confs(item, L):
    """
    Print all association rules with confidence above the minimum confidence.
    """
    global count
    for s in powerset(list(item)):
        if s == [] or s == list(item):
            continue
        l_s = [x for x in item if x not in s]
        conf = L[item] / L[tuple(s)]
        if conf > min_confidence:
            print_rule(s, l_s, conf)
            count += 1


# Initialize itemset list and table
l = []
table = {}
for t in db:
    for x in t:
        if (x,) not in table.keys():
            table[(x,)] = 1
        else:
            table[(x,)] += 1

# Remove infrequent items from the initial table
clear_infrequent_items(table)
l.append(table)

# Apriori algorithm
k = 1
while len(l[k - 1]) != 0:
    table = apriori_gen(l[k - 1])
    count_itemset_occurrences(table)
    clear_infrequent_items(table)
    l.append(table)
    k += 1

l.pop()  # Remove the empty set

# Merge all itemsets into a single dictionary
L = {}
for k in range(len(l)):
    L.update(l[k])

# Print association rules
count = 0
for item in L.keys():
    if len(item) > 1:
        print_confs(item, L)

print("Mined file:", sys.argv[5])
print("Total association rules found:", count)
