import itertools

# Define a class to represent association rules
class RuleGeneration:
    def __init__(self):
        self.x = []    # Antecedent
        self.y = []    # Consequent
        self.xy = []   # Combined set of Antecedent and Consequent
        self.x_sup = 0.0   # Support for Antecedent
        self.xy_sup = 0.0  # Support for Combined set of Antecedent and Consequent
        self.conf = 0.0    # Confidence

# Function to generate all possible permutations of a list of elements
def get_all_permutations(elements):
    if len(elements) <= 1:
        yield elements
    else:
        for permutation in get_all_permutations(elements[1:]):
            for i in range(len(elements)):
                yield permutation[:i] + elements[0:1] + permutation[i:]

print("####################################### Implementation of Apriori Algorithm ##############################")
print("Select dataset to test on:")
print("1 - Apple Store")
print("2 - Cycle Store")
print("3 - IKEA")
print("4 - Pet Shop")
print("5 - Music Store")

# Get user's choice for the dataset
ch = input("Enter your choice(1-5): ")

# Map the user's choice to the corresponding dataset file
datasets = {
    1: "apple_store",
    2: "cycle_store",
    3: "ikea",
    4: "pet_shop",
    5: "music_store"
}

input_file = datasets.get(int(ch), "music_store")

# Read dataset from the selected file
with open(input_file) as f:
    data = f.read()

# Process the dataset
dataset1 = []
for line in data.split("\n"):
    if line:
        items = line.split()[1:]
        dataset1.append(items)

print("Dataset:", dataset1)

# Get user input for minimum support and minimum confidence
min_sup = float(input("Enter Minimum Support: "))
min_conf = float(input("Enter Minimum Confidence: "))

# Find unique items in the dataset
items = sorted(set(item for sublist in dataset1 for item in sublist))
print("Unique items in the dataset are:", items)

# Calculate support for 1-itemsets
support1 = []
C1 = {}
F1 = {}

# Count occurrences of each item in the dataset
for k in items:
    count = sum(1 for transaction in dataset1 if k in transaction)
    support = (count / len(dataset1)) * 100.0
    support1.append(support)

    if support >= min_sup:
        F1[k] = support

print("Frequent item set of 1-item and its support are:")
print(F1)

# Generate 2-itemsets
combs = []
for i in range(1, len(F1) + 1):
    els = [list(x) for x in itertools.combinations(F1, i)]
    combs.extend(els)

# Calculate support for 2-itemsets
C2 = {}
F2 = {}
for transaction in dataset1:
    for temp in range(len(combs)):
        if set(combs[temp]).issubset(set(transaction)):
            C2[temp] = C2.get(temp, 0) + 1

for key, count in C2.items():
    support = (count / len(dataset1)) * 100
    if support >= min_sup:
        F2[key] = support

# Generate association rules
dict = {}
comb2 = list(F2.keys())
arules = []

for element in comb2:
    if len(element) > 1:
        r = RuleGeneration()
        left = 1
        while left < len(element):
            r.x = []
            r.xy = []
            for j in range(1, len(element) + 1):
                if j <= left:
                    r.x.append(element[j - 1])
                    r.xy.append(element[j - 1])
                else:
                    if j - 1 == left:
                        pass
                    r.y.append(element[j - 1])
                    r.xy.append(element[j - 1])
            for each_list in comb2:
                if set(r.xy).issubset(set(each_list)):
                    for x in F2:
                        if list(x) == r.xy:
                            y = x
                            break
                    for x in F2:
                        if list(x) == r.x:
                            z = x
                            break

                    xys = F2.get(y, 0)
                    xs = F2.get(z, 0)

                    conf = (xys / xs) * 100.0
                    if conf >= min_conf:
                        r.x_sup = xs
                        r.xy_sup = xys
                        arules.append("\t" + str(r.x) + "---->" + str(r.y) + "|" + str(r.x_sup) + "|" + str(
                            r.xy_sup) + "|" + str(conf) + "|")

            left += 1

print("\tRULE|SUPPORT(X)|SUPPORT(XY)|CONFIDENCE|")
print(" ")
for e in list(set(arules)):
    print(e)
