def create_candidate_set(data):
    """
    Create the initial candidate set from the dataset.
    """
    candidates = []
    
    for row in data:
        for item in row:
            if [item] not in candidates:
                candidates.append([item])
    
    candidates.sort()
    return list(map(frozenset, candidates))

def scan_data(data, candidate_set, min_support):
    """
    Scan the dataset to find frequent itemsets.
    """
    subset_count = {}
    
    for cur_set in data:
        for candidate in candidate_set:
            if candidate.issubset(cur_set):
                subset_count[candidate] = subset_count.get(candidate, 0) + 1
    
    n = float(len(data))
    valid = [key for key, support in subset_count.items() if support >= min_support]
    
    return valid, subset_count

def generate_apriori(freq_sets, k):
    """
    Generate candidate itemsets of length k.
    """
    valid = []
    n_freq_sets = len(freq_sets)
    
    for i in range(n_freq_sets):
        for j in range(i + 1, n_freq_sets):
            list_cands1 = list(freq_sets[i])[:k - 2]
            list_cands2 = list(freq_sets[j])[:k - 2]
            list_cands1.sort()
            list_cands2.sort()
            
            if list_cands1 == list_cands2:
                valid.append(freq_sets[i] | freq_sets[j])
    
    return valid

def apriori(data, min_support):
    """
    Apriori algorithm to find frequent itemsets.
    """
    candidate_set = create_candidate_set(data)
    set_data = list(map(set, data))
    list_candidates, subset_counts = scan_data(set_data, candidate_set, min_support)
    list_candidates = [list_candidates]
    k = 2
    
    while len(list_candidates[k - 2]) > 0:
        candidate_set_x = generate_apriori(list_candidates[k - 2], k)
        list_candidates_x, subset_counts_x = scan_data(set_data, candidate_set_x, min_support)
        subset_counts.update(subset_counts_x)
        list_candidates.append(candidate_set_x)
        k += 1
    
    return list_candidates, subset_counts
