import csv
import sys

# Check if the correct number of arguments is provided
if len(sys.argv) != 3:
    print("Execution format: python dat2csv.py <source_file> <destination_file>")
else:
    # Open the source file and read its content
    with open(sys.argv[1], 'r') as input_file:
        lines = input_file.readlines()

        # Process each line in the source file
        new_lines = []
        for line in lines:
            # Strip extra spaces and split the line into a list of values
            formatted_line = line.strip().split()
            new_lines.append(formatted_line)

    # Write the formatted data into the destination file in CSV format
    with open(sys.argv[2], 'w', newline='') as output_file:
        file_writer = csv.writer(output_file)
        file_writer.writerows(new_lines)
