#!/usr/bin/env python

import os
import sys
import django

def main():
    # Set the Django settings module
    os.environ.setdefault("DJANGO_SETTINGS_MODULE", "fypProject.settings")
    
    try:
        # Import execute_from_command_line function from Django
        from django.core.management import execute_from_command_line
    except ImportError:
        try:
            # Check if Django is installed
            import django
        except ImportError:
            # If Django is not installed, raise an ImportError
            raise ImportError(
                "Django is not installed. Make sure it's available on your PYTHONPATH or activate a virtual environment."
            )
        # Re-raise the ImportError if Django import failed for other reasons
        raise
    
    # Execute Django management commands from the command line arguments
    execute_from_command_line(sys.argv)

if __name__ == "__main__":
    # Call the main function when the script is executed
    main()
