from itertools import combinations

# Read transactions from a file
def read_transactions(filename):
    transactions = []
    with open(filename, 'r') as file:
        for line in file:
            transactions.append(line.strip())
    return transactions

# Find single candidates from the dataset
def find_single_candidates(transactions):
    singles = []
    for transaction in transactions:
        items = transaction.split(',')
        for item in items:
            if item not in singles:
                singles.append(item)
    return singles

# Count occurrences of single candidates in the dataset
def count_single_candidates(transactions, singles):
    candidate_count = {}
    for transaction in transactions:
        items = transaction.split(',')
        for item in items:
            if item in singles:
                candidate_count.setdefault(item, 0)
                candidate_count[item] += 1
    return candidate_count

# Count total number of items in the dataset
def count_total_items(transactions):
    total_items = 0
    for transaction in transactions:
        items = transaction.split(',')
        total_items += len(items)
    return total_items

# Check the support count for candidate items
def check_support_count(total_items, candidate_count, support):
    frequent_items = []
    for candidate in candidate_count:
        if candidate_count[candidate] / total_items >= support:
            frequent_items.append(candidate)
    return frequent_items

# Find frequent pairs based on the support count
def find_frequent_pairs(frequent_items, transactions, total_items, support):
    frequent_pairs = {}
    for transaction in transactions:
        items = transaction.split(',')
        for pair in combinations(items, 2):
            frequent_pairs.setdefault(pair, 0)
            frequent_pairs[pair] += 1

    infrequent_pairs = []
    for pair in frequent_pairs:
        if frequent_pairs[pair] / total_items < support:
            infrequent_pairs.append(pair)
        else:
            frequent_pairs[pair] = frequent_pairs[pair] / total_items

    for pair in infrequent_pairs:
        frequent_pairs.pop(pair, None)

    return frequent_pairs

if __name__ == "__main__":
    data_file = 'mushroom.data'
    min_support = 0.03

    transactions = read_transactions(data_file)

    single_candidates = find_single_candidates(transactions)
    print("Single Candidates:", single_candidates)

    candidate_counts = count_single_candidates(transactions, single_candidates)
    print("Candidate Counts:", candidate_counts)

    total_item_count = count_total_items(transactions)
    print("Total Items:", total_item_count)

    frequent_items = check_support_count(total_item_count, candidate_counts, min_support)
    print("Frequent Items:", frequent_items)

    frequent_pairs = find_frequent_pairs(frequent_items, transactions, total_item_count, min_support)
    print("Frequent Pairs:", frequent_pairs)
