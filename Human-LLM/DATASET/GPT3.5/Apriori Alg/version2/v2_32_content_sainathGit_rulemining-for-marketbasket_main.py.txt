import sys

# Read minimum support and minimum confidence from command-line arguments
min_support = float(sys.argv[2])
min_confidence = float(sys.argv[4])

# Read transactions from the input file
transactions = []
with open(sys.argv[5], "r") as file:
    for line in file:
        transactions.append([int(x) for x in line.split()])

# Calculate the minimum support count based on the number of transactions
min_support_count = min_support * len(transactions)

# Remove infrequent items from the initial table
def remove_infrequent_items(table):
    to_remove = [k for k, v in table.items() if v < min_support_count]
    for item in to_remove:
        del table[item]

# Generate candidate itemsets by combining frequent itemsets
def generate_candidate_itemsets(table):
    new_table = {}
    for l1 in table.keys():
        for l2 in table.keys():
            if can_combine(l1, l2):
                new_table[combine_itemsets(l1, l2)] = 0
    return new_table

# Count occurrences of candidate itemsets in transactions
def count_itemset_occurrences(table):
    for itemset in table.keys():
        for transaction in transactions:
            if set(itemset).issubset(set(transaction)):
                table[itemset] += 1

# Check if two itemsets can be combined to generate a new itemset
def can_combine(l1, l2):
    last_index = len(l1) - 1
    for i in range(last_index):
        if l1[i] != l2[i]:
            return False
    if l1[last_index] >= l2[last_index]:
        return False
    return True

# Combine two itemsets to generate a new itemset
def combine_itemsets(l1, l2):
    combined_itemset = []
    last_index = len(l1) - 1
    for i in range(last_index):
        combined_itemset.append(l1[i])
    combined_itemset.append(l1[last_index])
    combined_itemset.append(l2[last_index])
    return tuple(combined_itemset)

# Generate all possible subsets of a set
def generate_subsets(seq):
    if len(seq) <= 1:
        yield seq
        yield []
    else:
        for item in generate_subsets(seq[1:]):
            yield [seq[0]] + item
            yield item

# Print association rules
def print_association_rule(s, l_s, conf):
    print(s, " ==> ", l_s, "              ", conf)

# Print all association rules with confidence above the minimum confidence
def print_confident_association_rules(item, L):
    global count
    for s in generate_subsets(list(item)):
        if s == [] or s == list(item):
            continue
        l_s = [x for x in item if x not in s]
        conf = L[item] / L[tuple(s)]
        if conf > min_confidence:
            print_association_rule(s, l_s, conf)
            count += 1

# Initialize the list of frequent itemsets
frequent_itemsets = []
itemset_table = {}
for transaction in transactions:
    for item in transaction:
        if (item,) not in itemset_table.keys():
            itemset_table[(item,)] = 1
        else:
            itemset_table[(item,)] += 1

# Remove infrequent items from the initial table
remove_infrequent_items(itemset_table)
frequent_itemsets.append(itemset_table)

# Main Apriori algorithm
k = 1
while len(frequent_itemsets[k - 1]) != 0:
    itemset_table = generate_candidate_itemsets(frequent_itemsets[k - 1])
    count_itemset_occurrences(itemset_table)
    remove_infrequent_items(itemset_table)
    frequent_itemsets.append(itemset_table)
    k += 1

frequent_itemsets.pop()  # Remove the empty set

# Merge all itemsets into a single dictionary
all_itemsets = {}
for i in range(len(frequent_itemsets)):
    all_itemsets.update(frequent_itemsets[i])

# Print association rules
count = 0
for item in all_itemsets.keys():
    if len(item) > 1:
        print_confident_association_rules(item, all_itemsets)

print("Mined file:", sys.argv[5])
print("Total association rules found:", count)
