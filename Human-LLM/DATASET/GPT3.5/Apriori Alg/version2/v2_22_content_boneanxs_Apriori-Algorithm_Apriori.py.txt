import itertools

def get_subsets(S):
    """
    Return the list of subsets of S, where each subset contains len(S) - 1 elements.
    """
    subset_length = len(S) - 1
    return [list(i) for i in itertools.combinations(S, subset_length)]

def is_subset_exist(tar, L):
    """
    Check if each value in L is a subset of tar.
    Returns 1 if all values in L are subsets of tar, otherwise returns 0.
    """
    tar_set = [set(i[0]) for i in tar]
    is_in = 1
    for i in L:
        for j in tar_set:
            if j < set(i):
                is_in = 0
                return is_in
    return is_in

def apriori(tar, threshold, length):
    """
    Implement the Apriori algorithm for frequent item set mining.
    """
    cur_frequencies = {}

    # Count occurrences of each item
    for key in tar.keys():
        items = tar[key]
        for item in items:
            if item not in cur_frequencies.keys():
                cur_frequencies[item] = 1
            else:
                cur_frequencies[item] += 1

    # Calculate support for each item
    total_transactions = len(list(tar.keys()))
    keys = list(cur_frequencies.keys())
    for key in keys:
        cur_frequencies[key] = cur_frequencies[key] / total_transactions
        if cur_frequencies[key] < threshold:
            del cur_frequencies[key]

    cur_frequencies = [[key, cur_frequencies[key]] for key in cur_frequencies.keys()]

    if length == 1:
        return cur_frequencies

    temp_length = 2
    while temp_length <= length:
        temp_list = []
        for i in range(len(cur_frequencies)):
            for j in range(i + 1, len(cur_frequencies)):
                temp = []
                if temp_length == 2:
                    temp = [cur_frequencies[i][0], cur_frequencies[j][0]]
                else:
                    temp = list(set(cur_frequencies[i][0]) | set(cur_frequencies[j][0]))

                if len(temp) != temp_length:
                    continue

                temp.sort()

                if is_subset_exist(cur_frequencies, get_subsets(temp)) == 1:
                    if temp not in temp_list:
                        temp_list.append(temp)

        cur_frequencies = []
        for item in temp_list:
            temp_len = 0
            for key in tar.keys():
                if set(item) <= set(tar[key]):
                    temp_len += 1
            temp_len = temp_len / total_transactions
            if temp_len >= threshold:
                cur_frequencies.append([item, temp_len])

        temp_length += 1

    return cur_frequencies
