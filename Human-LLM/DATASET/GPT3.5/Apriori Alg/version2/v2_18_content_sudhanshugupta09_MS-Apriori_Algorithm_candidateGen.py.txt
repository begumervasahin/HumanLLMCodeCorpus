class MS_APRIORI:
    def __init__(self, transactions, MIS_values, SDC, cannot_be_together, must_have):
        self.transactions = transactions
        self.MIS_values = MIS_values
        self.SDC = SDC
        self.cannot_be_together = cannot_be_together
        self.must_have = must_have

    def generate_level2_candidates(self):
        candidates = []
        for l in self.MIS_values:
            for h in self.MIS_values:
                if l != h and self.transactions.count(l) >= self.MIS_values[l]:
                    if self.transactions.count(h) >= self.MIS_values[l] and abs(self.transactions.count(h) - self.transactions.count(l)) <= self.SDC:
                        candidates.append((l, h))
        return candidates

    def generate_MSCandidates(self, F, k):
        candidates = []
        for f1 in F:
            for f2 in F:
                if f1 != f2:
                    if (f1[-1] < f2[-1]) and (f1[:-1] == f2[:-1]) and abs(self.transactions.count(f1[-1]) - self.transactions.count(f2[-1])) <= self.SDC:
                        candidates.append(f1 + [f2[-1]])
                        c = f1 + [f2[-1]]
                        for i in range(1, len(c) + 1):
                            s = c[:i-1] + c[i:]
                            if c[0] in s or self.MIS_values[c[1]] == self.MIS_values[c[0]]:
                                if s not in F:
                                    candidates.remove(c)
                                    break
        return candidates

    def run_MS_APRIORI(self):
        F = []
        L = []
        for item, MIS in self.MIS_values.items():
            if self.transactions.count(item) >= MIS:
                L.append(item)
        F.append(L)
        k = 2
        while F[k-2]:
            if k == 2:
                C = self.generate_level2_candidates()
            else:
                C = self.generate_MSCandidates(F[k-2], k)
            temp_F = []
            for c in C:
                count = sum(1 for t in self.transactions if set(c).issubset(set(t)))
                if count >= self.MIS_values[c[0]]:
                    temp_F.append(c)
            if temp_F:
                F.append(temp_F)
            k += 1
        return F


# Sample Transactional Data
transactions = [
    {20, 30, 80, 70, 50, 90},
    {20, 10, 80, 70},
    {10, 20, 80},
    {20, 30, 80},
    {20, 80},
    {20, 30, 80, 70, 50, 90, 100, 120, 140}
]

# Sample MIS values for all items
MIS_values = {
    10: 0.43,
    20: 0.30,
    30: 0.30,
    40: 0.40,
    50: 0.40,
    60: 0.30,
    70: 0.20,
    80: 0.20,
    90: 0.20,
    100: 0.10,
    120: 0.20,
    140: 0.15
}

# Other Parameters
SDC = 0.1
cannot_be_together = [{20, 40}, {70, 80}]
must_have = {20, 40, 50}

ms_apriori = MS_APRIORI(transactions, MIS_values, SDC, cannot_be_together, must_have)
F = ms_apriori.run_MS_APRIORI()

# Printing the frequent itemsets
for i, freq_itemset in enumerate(F):
    print(f"Frequent {i+1}-itemsets")
    for itemset in freq_itemset:
        print(f"    {transactions.count(itemset)} : {set(itemset)}")
    print(f"Total number of frequent {i+1}-itemsets = {len(freq_itemset)}\n")
