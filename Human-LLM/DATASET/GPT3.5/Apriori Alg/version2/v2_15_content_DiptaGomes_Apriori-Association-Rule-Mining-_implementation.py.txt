# Importing the required libraries
import pandas as pd
from apyori import apriori

# Data Preprocessing
# Read the dataset from the CSV file
dataset = pd.read_csv('Nursery.csv', header=1)

# Create a list to store transactions
transactions = []

# Iterate through the dataset to create transaction lists
for i in range(1, 9409):
    # Extract each transaction and convert values to strings
    transaction = [str(dataset.values[i, j]) for j in range(0, 9)]
    transactions.append(transaction)

# Training Apriori on the dataset
# Apply Apriori algorithm with specified parameters
rules = apriori(transactions, min_support=0.04, min_confidence=0.90, min_lift=3, min_length=2)

# Visualising the results
# Convert the rules to a list for iteration
results = list(rules)

# Iterate through the discovered association rules
for item in results:
    # Extract the items in the rule
    pair = item[0]
    items = [x for x in pair]

    # Print the association rule
    print("Rule: " + items[0] + " -> " + items[1])

    # Print the support of the rule
    print("Support: " + str(item[1]))

    # Print the confidence of the rule
    print("Confidence: " + str(item[2][0][2]))

    # Print the lift of the rule
    print("Lift: " + str(item[2][0][3]))

    # Print separator for clarity
    print("=====================================")
