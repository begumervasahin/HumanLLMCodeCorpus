import re
from Apriori import Apriori

# Input minimum support, confidence, and length of rules
min_support = int(input("Enter the minimum support: "))
min_confidence = float(input("Enter the minimum confidence: "))
min_length = int(input("Enter the minimum length of rules: "))

# Define alphabets for item encoding
alphabet = [chr(i) for i in range(ord('a'), ord('z') + 1)]

# Read data from the file and preprocess
items_list = []  # Initialize list to hold transaction items
temp_items = []   # Temporary list to store preprocessed items

with open('datasetUCI.txt', 'r') as file:
    for line in file:
        line = line.strip()[:-1]  # Remove trailing comma
        line = re.sub("[?\s]", 'a', line)  # Replace special characters and spaces with 'a'
        items = line.split(',')  # Split the line into items
        for item_str in items:
            if item_str in alphabet:
                temp_items.append(item_str)
        items_list.append(temp_items)
        temp_items = []  # Reset the temporary list for the next transaction

# Run Apriori algorithm
apriori_result = Apriori(items_list, min_support, min_confidence, min_length)
