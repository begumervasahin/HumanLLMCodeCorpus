import os
import re
from preprocess import Preprocess

def getNextTransaction(reader):
    """
    Reads transactions from a file character by character.
    """
    transaction = ""
    while True:
        char = reader.read(1)
        transaction += char
        if char == "}":
            yield transaction
            transaction = ""
        if not char:
            break

class FileOperator:
    def __init__(self, datapath="./", data="data", params="params", results="results"):
        """
        Initializes the FileOperator class with default or user-specified paths.
        """
        self._path = datapath
        self._data = {
            "data": data,
            "params": params,
            "results": results
        }
        self._patterns = {
            "data": r"(data)([0-9]*)-*(|[0-9]*)(.txt)",
            "params": r"(params)([0-9]*)-*(|[0-9]*)(.txt)"
        }

    def getConstraints(self, setnumber=""):
        """
        Retrieves constraints from parameter files based on set number.
        """
        try:
            preprocess = Preprocess()
            absolute_path = os.path.join(self._path, self._data['params'])
            count = []
            constraints = []
            for filename in os.listdir(absolute_path):
                match = re.match(self._patterns['params'], filename)
                if match and match.group(2) == setnumber:
                    count.append(match.group(3))
                    with open(os.path.join(absolute_path, filename), "r") as c:
                        constraints.append(preprocess.preprocessConstraints(c.read().split("\n")))
            if not constraints:
                return {"error": True, "message": "No constraints found."}
            return {"error": False, "constraints": constraints, "count": count}
        except FileNotFoundError:
            return {"error": True, "message": "Parameter files not found."}

    def getDatasetNumbers(self):
        """
        Retrieves dataset numbers from data filenames.
        """
        try:
            absolute_path = os.path.join(self._path, self._data['data'])
            setnumbers = []
            for filename in os.listdir(absolute_path):
                match = re.match(self._patterns['data'], filename)
                if match:
                    setnumbers.append(match.group(2))
            if not setnumbers:
                return {"error": True, "message": "No dataset numbers found."}
            return {"error": False, "setnumbers": set(setnumbers)}
        except FileNotFoundError:
            return {"error": True, "message": "Data files not found."}

    def getTransactions(self, setnumber=""):
        """
        Retrieves transactions from data files based on set number.
        """
        preprocess = Preprocess()
        try:
            absolute_path = os.path.join(self._path, self._data['data'])
            for filename in os.listdir(absolute_path):
                match = re.match(self._patterns['data'], filename)
                if match and match.group(2) == setnumber:
                    transaction = ""
                    with open(os.path.join(absolute_path, filename), "r") as t:
                        for transaction in getNextTransaction(t):
                            yield preprocess.preprocessTransaction(transaction.strip())
        except FileNotFoundError:
            raise Exception("Data files not found.")

    def writeFrequentItemset(self, frequent, tailcounts=None, k=0, datanum="", paramnum=None):
        """
        Writes frequent itemsets to result files.
        """
        if not frequent or k <= 1:
            return

        filename = f"results{datanum}"
        if paramnum:
            filename += f"-{paramnum}"
        filename += ".txt"

        results_directory = os.path.join(self._path, self._data['results'])
        if not os.path.exists(results_directory):
            os.makedirs(results_directory)

        absolute_path = os.path.join(results_directory, filename)
        write_mode = 'a' if os.path.exists(absolute_path) else 'w'

        with open(absolute_path, write_mode) as w:
            w.write(f"Frequent {k}-itemsets:\n")
            if k == 1:
                for f in frequent:
                    w.write(f"\t{f[1]} : {{ {f[0]} }}\n")
            else:
                for itemset, count in frequent:
                    w.write(f"\t{count} : {{ {', '.join(map(str, itemset))} }}\n")
                    if tailcounts:
                        w.write(f"Tailcount: {tailcounts.get(str(itemset), 0)}\n")
            w.write(f"\tTotal number of frequent {k}-itemsets = {len(frequent)}\n\n")

if __name__ == "__main__":
    # Example usage of the FileOperator class
    f = FileOperator(datapath="./", data="data", params="params", results="results")
    constraints = f.getConstraints(setnumber="1")
    print(constraints)
    dataset_numbers = f.getDatasetNumbers()
    print(dataset_numbers)
    transactions = f.getTransactions(setnumber="1")
    for t in transactions:
        print(t)
    frequent_itemsets = [[(1, 5), 10], [(2, 3, 4), 15]]  # Example frequent itemsets
    tail_counts = {"(1, 5)": 8, "(2, 3, 4)": 12}  # Example tail counts
    f.writeFrequentItemset(frequent=frequent_itemsets, tailcounts=tail_counts, k=3, datanum="1", paramnum=None)
