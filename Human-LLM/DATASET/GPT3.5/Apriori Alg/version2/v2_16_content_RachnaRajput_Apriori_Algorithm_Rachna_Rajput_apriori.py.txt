import json
import sys

class AprioriAlgorithm:
    def apriori(self, json_data, value):
        array = json_data
        threshold = 0.3 * len(array)
        phase1_list = []

        # Generating unique items
        unique_items = []
        for item in array:
            unique_items = list(set(unique_items) | set(item))

        # Generating candidate set C1
        c1 = [[unique_items[i]] for i in range(len(unique_items))]
        if value:
            print("C1:", c1)

        # Counting support for C1
        item_counts = {}
        for arr in array:
            for subarr in arr:
                if subarr not in item_counts:
                    item_counts[subarr] = 1
                else:
                    item_counts[subarr] += 1

        # Generating frequent itemset L1
        l1 = [[unique_items[i]] for i in range(len(unique_items)) if item_counts[unique_items[i]] >= threshold]
        if value:
            print("L1:", l1)
        else:
            for item in l1:
                phase1_list.append(item)

        # Generating candidate set C2
        c2 = []
        k = 1
        for i in range(1, len(l1) + 1):
            for j in range(i + 1, len(l1) + 1):
                item_pair = [i, j]
                c2.append(item_pair)
                k += 1
        if value:
            print("C2:", c2)

        # Counting support for C2
        support_counts_c2 = {}
        for index, elem in enumerate(c2):
            count = 0
            for arr in array:
                if set(elem) <= set(arr):
                    count += 1
            support_counts_c2[index + 1] = count

        # Generating frequent itemset L2
        l2 = [c2[i - 1] for i in range(1, len(support_counts_c2) + 1) if support_counts_c2[i] >= threshold]
        if value:
            print("L2: ", l2)
        else:
            for item in l2:
                phase1_list.append(item)

        # Generating frequent itemsets L3 and beyond
        y = 3
        while len(l2) != 0:
            c3 = []
            for i in range(0, len(l2) // 2 + 1):
                for j in range(i + 1, len(l2) // 2 + 1):
                    if l2[i][0] == l2[j][0]:
                        item_set = []
                        for x in range(1, len(l2[i])):
                            item_set.append(l2[i][x])
                            item_set.append(l2[j][x])
                        item_set.sort()
                        for item in l2:
                            if set(item_set) == set(item):
                                item_set.append(l2[i][0])
                                item_set.sort()
                                item_set = list(set(item_set))
                                c3.append(item_set)

            # Printing candidate sets and frequent itemsets
            candidate_set_label = "C" + str(y) + ": "
            frequent_itemset_label = "L" + str(y) + ": "

            if value:
                print(candidate_set_label, c3)

            support_counts_c3 = {}
            for index, elem in enumerate(c3):
                count = 0
                for arr in array:
                    if set(elem) <= set(arr):
                        count += 1
                support_counts_c3[index + 1] = count

            l3 = [c3[i - 1] for i in range(1, len(support_counts_c3) + 1) if support_counts_c3[i] >= threshold]

            if value:
                print(frequent_itemset_label, l3)
            else:
                for item in l3:
                    phase1_list.append(item)

            l2 = l3
            y += 1

        return phase1_list

if __name__ == '__main__':
    with open(sys.argv[1]) as json_file:
        json_data = json.load(json_file)

    checker = True
    apriori_instance = AprioriAlgorithm()
    apriori_instance.apriori(json_data, checker)
