import re

class Preprocess:
    _patterns = {
        "MIS": r"MIS\(([\w-]+)\)\s*[\=\:]\s*([\d.]+)",
        "others": r"(others)\s*[\=\:]\s*([\d.]+)",
        "SDC": r"(SDC)\s*[\=\:]\s*([\d.]+)",
        "not_together": r"(cannot_be_together)\s*:\s*([{\w\-,\s}]*})",
        "must_have": r"(must-have)\s*:\s*([\w\s(or)]+)",
        "not_together_elements": r"\{*'*([\w-]+),*\s*([\w-]+)'*\}*",
        "not_together_split": r"\}\s*,\s*\{",
        "transaction": r"([\w-]+)"
    }

    def preprocessConstraints(self, constraints):
        """
        Preprocesses constraints from input lines.
        """
        if not constraints:
            return constraints

        processed_constraints = {}
        for line in constraints:
            MIS_match = re.match(self._patterns['MIS'], line)
            if MIS_match:
                item = MIS_match.group(1).lower()
                value = float(MIS_match.group(2))
                processed_constraints[item] = value

            SDC_match = re.match(self._patterns['SDC'], line)
            if SDC_match:
                item = SDC_match.group(1).upper()
                value = float(SDC_match.group(2))
                processed_constraints[item] = value

            not_together_match = re.match(self._patterns['not_together'], line)
            if not_together_match:
                elements = not_together_match.group(2)
                not_together_elements = re.split(self._patterns['not_together_split'], elements)
                parsed_elements = []
                for elem in not_together_elements:
                    match = re.match(self._patterns['not_together_elements'], elem)
                    parsed_elements.append((match.group(1).strip().lower(), match.group(2).strip().lower()))
                processed_constraints['not_together'] = parsed_elements

            must_have_match = re.match(self._patterns['must_have'], line)
            if must_have_match:
                items = [i.strip().lower() for i in must_have_match.group(2).split("or")]
                processed_constraints['must_have'] = items

            others_match = re.match(self._patterns['others'], line)
            if others_match:
                item = others_match.group(1)
                value = float(others_match.group(2))
                processed_constraints[item] = value

        if 'not_together' not in processed_constraints:
            processed_constraints['not_together'] = ""
        if 'must_have' not in processed_constraints:
            processed_constraints['must_have'] = ""

        return processed_constraints

    def preprocessTransaction(self, transaction):
        """
        Preprocesses transactions to extract individual items.
        """
        if not transaction:
            return transaction

        processed_transaction = [i.strip().lower() for i in re.findall(self._patterns["transaction"], transaction)]
        return processed_transaction

# Example usage:
if __name__ == "__main__":
    preprocess = Preprocess()

    # Example constraints
    constraints = [
        "MIS(a) = 0.2",
        "SDC = 0.1",
        "cannot_be_together: {a, b}, {c, d}",
        "must-have: x or y",
        "others = 0.5"
    ]
    processed_constraints = preprocess.preprocessConstraints(constraints)
    print("Processed Constraints:", processed_constraints)

    # Example transaction
    transaction = "{a, b, c, d}"
    processed_transaction = preprocess.preprocessTransaction(transaction)
    print("Processed Transaction:", processed_transaction)
