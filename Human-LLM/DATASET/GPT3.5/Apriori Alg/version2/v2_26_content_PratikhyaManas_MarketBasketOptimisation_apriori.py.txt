# Importing the necessary libraries
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from apyori import apriori

# Data Preprocessing
# Reading the dataset
dataset = pd.read_csv('Market_Basket_Optimisation.csv', header=None)

# Creating a list to hold transaction records
transactions = []

# Iterating through each row in the dataset
for i in range(len(dataset)):
    # Creating a list for each transaction and converting elements to strings
    transaction = [str(dataset.values[i, j]) for j in range(len(dataset.columns))]
    # Appending the transaction to the list of transactions
    transactions.append(transaction)

# Training Apriori on the dataset
# Applying Apriori algorithm with specified parameters
rules = apriori(transactions, min_support=0.003, min_confidence=0.2, min_lift=3, min_length=2)

# Extracting the results
results = list(rules)

# Printing the association rules obtained from the dataset
for result in results:
    print(result)
