from operator import itemgetter

# Function to calculate M (sorted list of MIS values)
def calculate_M(MIS):
    M = sorted(MIS.items(), key=itemgetter(1))  # Sort MIS values
    return M

# Function to create a list of items
def create_list_of_items(I, MIS):
    for key, value in MIS.items():
        I.append(key)

# Initial pass over the items list
def init_pass(M, L, MIS, support_count):
    conf_flag = True
    for (i, mis_val) in M:
        if support_count[i] >= MIS[i] and conf_flag:
            L.append(i)
            min_mis_required = MIS[i]
            conf_flag = False
        elif not conf_flag:
            if support_count[i] >= min_mis_required:
                L.append(i)

# Check constraints for items that cannot be together
def check_cannot_be_together(F, cannot_be_together):
    temp = [[], [], [], [], []]
    frequent_val = 0

    for f in F:
        if len(f) > 0:
            for s in f:
                if frequent_val > 0:
                    for c in cannot_be_together:
                        if set(c).issubset(s):
                            pass
                        else:
                            temp[frequent_val].append(s)
                else:
                    temp[frequent_val].append(s)
            frequent_val += 1
    return temp

# Check if must-have items are included in frequent itemsets
def check_must_have(F, must_have):
    final_F = [[], [], [], [], []]
    frequent_val = 0

    for f in F:
        if len(f) > 0:
            temp = set()
            for s in f:
                for must in must_have:
                    if frequent_val > 1:
                        if must in s:
                            temp.add(s)
                            continue
                    else:
                        if s in must_have:
                            temp.add(s)
            final_F[frequent_val - 1].extend(list(temp))
        frequent_val += 1
    return final_F

# Sample Transactional Data
transactions = [
    {20, 30, 80, 70, 50, 90},
    {20, 10, 80, 70},
    {10, 20, 80},
    {20, 30, 80},
    {20, 80},
    {20, 30, 80, 70, 50, 90, 100, 120, 140}
]

# Sample Parameter File
MIS_values = {
    10: 0.43,
    20: 0.30,
    30: 0.30,
    40: 0.40,
    50: 0.40,
    60: 0.30,
    70: 0.20,
    80: 0.20,
    90: 0.20,
    100: 0.10,
    120: 0.20,
    140: 0.15
}

# Other Parameters
SDC = 0.1
cannot_be_together = [{20, 40}, {70, 80}]
must_have = {20, 40, 50}

# Perform the functions
M = calculate_M(MIS_values)
I = []
create_list_of_items(I, MIS_values)
L = []
support_count = {item: sum(1 for t in transactions if item in t) for item in I}
init_pass(M, L, MIS_values, support_count)
F = [L]

F = check_cannot_be_together(F, cannot_be_together)
F = check_must_have(F, must_have)

# Printing the frequent itemsets
for i, freq_itemset in enumerate(F):
    print(f"Frequent {i + 1}-itemsets")
    for itemset in freq_itemset:
        print(f"    {len([t for t in transactions if itemset.issubset(t)])} : {itemset}")
    print(f"Total number of frequent {i + 1}-itemsets = {len(freq_itemset)}\n")
