import re
import pandas as pd

def preprocessing_frame(frame):
    """
    Preprocesses the skills data frame by removing unwanted characters and filtering specific skills.
    
    Args:
    - frame: pandas Series representing the skills data frame
    
    Returns:
    - preprocessed_data: list of lists containing preprocessed skills
    """
    # List of specific skills to include
    specific_skills = ['c#', '.net', 'asp.net', 'c++']
    preprocessed_data = []

    # Define regular expressions patterns for filtering
    pattern1 = re.compile(r'([a-z].*[0-9])')
    pattern2 = re.compile(r'([a-z].*[a-z])')

    for _, rows in frame.iteritems():
        # Remove unwanted characters from each row
        rows = rows.replace('>', '').replace('<', '').replace('...', '')

        # Split the row into individual skills
        skills = rows.split(',')
        preprocessed_skills = []

        for skill in skills:
            # Split by space to count words in the skill
            space_split = skill.split(' ')

            # Only include skills with less than or equal to 4 words
            if len(space_split) <= 4:
                skill = skill.lower().strip()

                # Filter out certain patterns and specific skills
                if (pattern1.match(skill) or pattern2.match(skill) or skill in specific_skills):
                    skill = skill.replace('?', '').replace('&amp;', '').replace('&', '').replace('#x', '').replace('#x22', '').replace('.', '')  

                    # Remove trailing periods
                    if skill.endswith('.'):
                        skill = skill[:-1]

                    # Remove leading periods if it's not part of '.net'
                    if skill.startswith('.') and not skill.startswith('.net'):
                        skill = skill[1:]

                    # Add the skill to the list if it's not already present
                    if skill not in preprocessed_skills:
                        preprocessed_skills.append(skill)

        preprocessed_data.append(preprocessed_skills)

    return preprocessed_data

# Example usage:
if __name__ == "__main__":
    # Sample data frame
    data = {'skills': ['Java, Python, HTML', 'C#, C++, JavaScript, HTML']}
    df = pd.DataFrame(data)

    # Preprocess the data frame
    preprocessed_data = preprocessing_frame(df['skills'])

    # Display the preprocessed data
    print("Preprocessed Data:")
    for skills in preprocessed_data:
        print(skills)
