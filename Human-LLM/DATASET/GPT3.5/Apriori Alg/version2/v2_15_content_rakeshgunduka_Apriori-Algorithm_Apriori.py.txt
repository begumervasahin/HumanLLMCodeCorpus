from nltk import regexp_tokenize
import csv
import copy

# Initialize global variables
d = []
d_main = {}
dv = []
count = {}
allowed = {}
last_level = 0

# Function to join two itemsets
def join(itemset1, itemset2):
    if type(itemset1) == list and type(itemset2) == list:
        result = []
        for item in itemset2:
            temp = copy.copy(itemset1)
            if item not in itemset1:
                temp.append(item)
                result.append(temp)
        return result
    else:
        return [itemset1, itemset2]

# Generate pairs of datasets
def generate_pairs(itemset, level):
    itemset1 = itemset2 = itemset
    pairs = []

    for i in itemset1:
        for j in itemset2:
            if j > i:
                val = join(i, j)
                if level != 1:
                    for k in val:
                        if k == sorted(k) and k not in pairs:
                            pairs.append(k)
                else:
                    pairs.append(val)
    return pairs

# Check for support
def check_support(itemset, level, min_support):
    supported_itemsets = []
    for index, count_item in enumerate(count[level]):
        if count_item >= min_support:
            supported_itemsets.append(itemset[index])
    return sorted(supported_itemsets)

# Check if an itemset is present in the data
def is_itemset_present(check_itemset, dataset):
    if type(check_itemset) == list:
        for item in check_itemset:
            if item not in dataset:
                return False
        return True
    else:
        return check_itemset in dataset

# Count occurrences of itemsets
def count_occurrences(itemset, dataset):
    itemset_count = [0 for _ in range(len(itemset))]
    for itemset_val in itemset:
        for data in dataset:
            if is_itemset_present(itemset_val, data):
                itemset_count[itemset.index(itemset_val)] += 1
    return itemset_count

# Apriori algorithm implementation
def apriori(level, itemset, min_support):
    global last_level
    last_level = level - 1
    if level != 1:
        itemset = generate_pairs(allowed[level - 1], level - 1)

    print("\n", 30 * "*", "level", level, "*"*30)
    print("Dataset(itemset) :", itemset)
    print("Support(min_support) :", min_support)
    print("Pairs :", itemset)
    count[level] = count_occurrences(itemset, d)
    supported_itemset = check_support(itemset, level, min_support)

    if len(supported_itemset) != 0:
        allowed[level] = supported_itemset
    else:
        print("There is no support further, Stopping Here")
        return

    supported_counts = [i for i in count[level] if i >= min_support]
    count[level] = supported_counts

    print("Count for level", level, ":", count[level])
    print("Allowed :", allowed[level])

    level += 1
    apriori(level, itemset, min_support)

# Check for strong association rules
def check_association_rules(frequent_itemset, value):
    strong_associations = []
    threshold = 70
    for item in value:
        calculated_value = ((count[len(frequent_itemset)][allowed[len(frequent_itemset)].index(frequent_itemset)] /
                             count[len(item)][allowed[len(item)].index(item)]) * 100)
        if calculated_value > threshold:
            strong_associations.append(item)
    return strong_associations

# Find association rules for frequent itemsets
def association_rules(frequent_itemsets):
    candidate_sets = {}
    for index, itemset in enumerate(frequent_itemsets):
        values = generate_pairs(itemset, last_level)
        for item in itemset:
            values.append(item)
        candidate_sets[index] = check_association_rules(itemset, values)

    lengths = ([len(candidate_sets[index]) for index in candidate_sets])
    return frequent_itemsets[lengths.index(max(lengths))]

if __name__ == '__main__':
    print("\n", 23 * "*", "Apriori Algorithm", "*"*21)
    choice = int(input("Mode of input\n1: Manual Dataset\n2: CSV File\nChoice : "))
    if choice == 1:
        data_length = int(input("Length of Dataset : "))
        print("Enter Data followed by space(Eg. 1 5 3)\nID", "\t", "Itemsets")
        for index in range(data_length):
            print(index + 1, end=' ')
            d.append([x for x in input("\t").split()])
    else:
        file_name = input("Enter Filename(.csv) :")
        with open(file_name, 'r') as file:
            reader = csv.reader(file)
            for row in reader:
                d.append(row)
        del d[0]

    print("\n", 23 * "*", "Itemsets Provided", "*"*21)
    for index, data_list in enumerate(d):
        if choice == 2:
            for string in data_list:
                tokens = regexp_tokenize(string, r',', gaps=True)
            d[index] = tokens
        for item in d[index]:
            if item not in dv:
                dv.append(item)
                d_main[item] = 0
        print(d[index])

    min_support = 2
    level = 1
    dv = sorted(dv)
    apriori(level, dv, min_support)

    print("\n", 20 * "*", "Applying Association Rules", "*"*21)
    frequent_itemsets = [item for sublist in allowed.values() for item in sublist]
    print("Frequent Itemsets :", frequent_itemsets)

    most_frequent_set = association_rules(allowed[last_level])
    print("\n", 23 * "*", "Final Frequent Patterns", "*"*21)
    for key in allowed:
        for item in allowed[key]:
            print(item, " "*10, ":", count[len(item)][allowed[len(item)].index(item)])

    print("\n", 23 * "*", "Most Frequent Pattern", "*"*21)
    print(most_frequent_set)
