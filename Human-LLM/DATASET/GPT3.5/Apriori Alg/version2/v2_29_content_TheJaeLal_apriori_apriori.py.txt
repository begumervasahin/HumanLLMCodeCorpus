# Function to read transaction data from a file
def read_from_file(fname):
    file = open(fname, 'r')
    dataset = file.read().split('\n')
    # Parsing transaction data
    dataset = [line.split(':')[1].split(',') for line in dataset if not (line.startswith('#') or line.strip() == "")]
    file.close()
    return dataset

# Function to remove items/sets whose frequency/count/support is less than the threshold
def prune(item_set, threshold):
    return {item: count for item, count in item_set.items() if count >= threshold}

# Function to create combinations of sets/items
def combine_sets(frequent_items):
    new_item_set = []
    for i in range(len(frequent_items)):
        for j in range(i + 1, len(frequent_items)):
            new_set = frequent_items[i].union(frequent_items[j])
            new_item_set.append(new_set)
    return new_item_set

# Function to calculate the count/frequency of the new sets in the database
def get_count(new_item_set, database):
    item_set = {}
    for item in new_item_set:
        for entry in database:
            if item.issubset(entry):
                item_set[item] = item_set.get(item, 0) + 1
    return item_set

if __name__ == "__main__":
    # Input file containing transaction data
    input_file = 'input.dat'

    # Read transaction data from file
    database = read_from_file(input_file)

    # Convert database to a list of sets/frozensets
    database = [frozenset(entry) for entry in database]

    # Get initial frequency count for every item in the database
    item_set = {}
    for entry in database:
        for item in entry:
            item = frozenset([item])
            item_set[item] = item_set.get(item, 0) + 1

    print('***Initial Frequency Count***')
    for item, count in item_set.items():
        print(item, ':', count)

    # Set the support threshold
    print('Enter Support_Threshold:')
    print('>>> ', end='')
    threshold = int(input())
    print('threshold =', threshold)

    previous_item_set = item_set

    # Iteratively apply the Apriori algorithm
    while True:
        # Prune the item set based on the support threshold
        frequent_items = prune(item_set, threshold)

        print('\n****Frequent Items****')
        for f, count in frequent_items.items():
            print(f, ':', count)

        # Stop if there is only 1 set/item left in frequent items
        if len(frequent_items) < 1:
            frequent_items = previous_item_set
            break

        print("\n***New Iteration***")

        # Generate new item sets by combining frequent items
        new_item_set = combine_sets(frequent_items)

        previous_item_set = frequent_items

        # Calculate the count/frequency of new item sets in the database
        item_set = get_count(new_item_set, database)

        print('\n***Frequency Count***')
        for item, count in item_set.items():
            print(item, ':', count)

    print('\n***************************************************************')
    print('The most frequently associated items are:')
    for item in frequent_items:
        tup = ', '.join(item)
        print('{', '{}'.format(tup), '}')
    print('\n')
