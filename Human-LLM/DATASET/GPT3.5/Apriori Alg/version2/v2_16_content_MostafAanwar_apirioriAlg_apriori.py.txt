import pandas as pd
import itertools

# Read user input
percentage = int(input("How many % of records you want to read? "))
min_support = int(input("What is the minimum support? "))
min_confidence = int(input("What is the % of the minimum confidence? "))

# Read the file and calculate the number of records
file_path = "CarSales.txt"
num_lines = sum(1 for line in open(file_path))

# Convert the percentage to the number of records
records = int((percentage / 100) * num_lines)

# Read the specified percentage of records from the file
data_set = pd.read_csv(file_path, sep=' ', header=None, nrows=records).values

# Dictionary to count item occurrences
item_count = {}

# Count occurrences of combinations
for transaction in data_set:
    for i in range(1, 5):
        for comb in itertools.combinations(transaction, i):
            item_count[comb] = item_count.get(comb, 0) + 1

# Remove items with support below the minimum support threshold
frequent_items = {k: v for k, v in item_count.items() if v >= min_support}

# Identify the maximum size of frequent itemsets
max_itemset_size = max(len(item) for item in frequent_items)

# Extract keys and values for confidence calculation
confidence_keys = [k for k in frequent_items if len(k) == max_itemset_size]
confidence_values = [frequent_items[k] for k in confidence_keys]

# Extract unique items for confidence calculation
unique_items = list(set(itertools.chain.from_iterable(confidence_keys)))

# Generate combinations for confidence calculation
combinations = []
for itemset in confidence_keys:
    for i in range(1, 3):
        for combination in itertools.combinations(itemset, i):
            combinations.append(combination)

# Generate permutations for confidence calculation
permutations = list(itertools.permutations(unique_items, max_itemset_size))

# Generate test combinations for confidence calculation
test_combinations = []
for i in unique_items:
    list_num = [i]
    for j in unique_items:
        if i != j:
            list_num.append(j)
    test_combinations.append(list_num)
    test_combinations.append(list_num[::-1])

print("Apriori algorithm completed successfully.")
