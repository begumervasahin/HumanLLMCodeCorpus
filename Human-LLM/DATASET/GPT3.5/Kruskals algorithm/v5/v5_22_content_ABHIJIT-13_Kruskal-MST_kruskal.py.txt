def find_root(vertex):
    """
    Finds the root of the given vertex in the disjoint set.
    """
    if rootIds[vertex] == vertex:
        return vertex
    else:
        while rootIds[vertex] != vertex:
            # Path compression: update the parent of the current vertex
            # to the root of its parent
            rootIds[vertex] = rootIds[rootIds[vertex]]
            vertex = rootIds[vertex]
        return vertex

def union(u, v):
    """
    Performs union operation on two vertices.
    """
    u_root = find_root(u)
    v_root = find_root(v)
    rootIds[v_root] = rootIds[u_root]

def kruskal(GE):
    """
    Implements Kruskal's algorithm to find the minimum spanning tree.
    """
    global rootIds
    min_weight = 0
    for edge in GE:
        u, v, weight = edge
        if find_root(u) != find_root(v):
            min_weight += weight
            union(u, v)
    return min_weight

def main():
    # Input
    V, E = map(int, input().split())
    GE = [list(map(int, input().split())) for _ in range(E)]

    # Sort edges by weight
    GE = sorted(GE, key=lambda x: x[2])

    # Initialize rootIds for disjoint set
    global rootIds
    rootIds = list(range(V + 1))

    # Output the minimum spanning tree weight
    print(kruskal(GE)))

if __name__ == "__main__":
    main()
