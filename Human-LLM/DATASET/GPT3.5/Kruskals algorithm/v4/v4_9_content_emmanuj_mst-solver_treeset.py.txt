from node import Node

class TreeSet:
    """
    Represents a set of d-heaps using a tree structure.
    """

    def __init__(self):
        """
        Initializes an empty tree set.
        """
        self.data = {}

    def make_set(self, n):
        """
        Creates a new set with a single element.

        Parameters:
            n: The value of the element to be added to the set.
        """
        self.data[n] = Node(n)

    def find(self, n):
        """
        Finds the representative of the set containing the given element.

        Parameters:
            n: The element whose representative needs to be found.

        Returns:
            int: The representative of the set containing the element.
        """
        gp = self.data[n].parent.parent
        while gp != self.data[n].parent:
            self.data[n].parent = gp
            n = gp.value
            gp = self.data[n].parent.parent
        return gp.value

    def union(self, u, v):
        """
        Combines the sets that contain the elements u and v.

        Parameters:
            u: The first element.
            v: The second element.

        Returns:
            Node: The representative of the combined set.
        """
        ptr = None
        if self.data[u].rank > self.data[v].rank:
            self.data[v].parent = self.data[u]
            ptr = self.data[u]
        elif self.data[u].rank < self.data[v].rank:
            self.data[u].parent = self.data[v]
            ptr = self.data[v]
        else:
            self.data[u].parent = self.data[v]
            self.data[v].rank += 1
            ptr = self.data[v]
        return ptr

    def printd(self):
        """
        Prints the elements and their corresponding nodes in the tree set.
        """
        print(self.data)
