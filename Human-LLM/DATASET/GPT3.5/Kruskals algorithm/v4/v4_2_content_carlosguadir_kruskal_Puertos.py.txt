# Lists of country names and their corresponding port lists
country_names = ['EEUU', 'Canada', 'Mexico', 'Argentina', 'Brazil', 'Chile', 'Colombia', 'Ecuador', 'Peru',
                 'Costa Rica', 'Panama', 'Alemania', 'Espana', 'Francia', 'Gran Bretana', 'Holanda', 'Italia',
                 'Grecia', 'Rusia', 'China', 'India', 'Japon', 'Tailandia', 'Emiratos Arabes', 'Egipto',
                 'Marruecos', 'Sudafrica', 'Australia']

port_lists = [EEUU, Canada, Mexico, Argentina, Brazil, Chile, Colombia, Ecuador, Peru, Costa_Rica, Panama,
              Alemania, Espana, Francia, Gran_Bretana, Holanda, Italia, Grecia, Rusia, China, India, Japon,
              Tailandia, Emiratos_Arabes_Unidos, Egipto, Marruecos, Sudafrica, Australia]


def port_list(lista):
    """
    Creates a list of all ports from the provided list of country port lists.
    """
    for port_list in port_lists:
        for port in port_list:
            lista.append(port)
    lista.insert(0, '-Empty-')
    return lista


def country_for_port(port):
    """
    Returns the country name for a given port.
    """
    index = 0
    for port_list in port_lists:
        if port in port_list:
            return country_names[index]
        index += 1
