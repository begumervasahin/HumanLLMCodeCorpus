class MaxHeap:
    """A class representing a max heap data structure."""

    def __init__(self):
        """Initialize the max heap."""
        self.heap = [-1]  # Heap list with a dummy element at index 0
        self.values = [-1]  # Corresponding values list with a dummy element at index 0
        self.size = 0  # Current size of the heap

    def maximum(self):
        """Return the maximum value in the heap."""
        return self.heap[1]

    def insert(self, name, value):
        """Insert a new element into the heap."""
        self.size += 1
        self.heap.append(name)
        self.values.append(value)
        self.heapify_up(self.size)

    def update(self, name, value):
        """Update the value of an existing element in the heap."""
        print(value)
        print(self.size)

    def delete(self, index):
        """Delete an element from the heap."""
        if self.size == 1 or index == self.size:
            self.values.pop()
            self.heap.pop()
            self.size -= 1
        else:
            self.values[index] = self.values.pop()
            self.heap[index] = self.heap.pop()
            self.size -= 1
            self.heapify_down(index)

    def heapify_up(self, index):
        """Perform heapify operation moving up from the given index."""
        if index > 1 and self.values[index] > self.values[index // 2]:
            current_index = index
            while current_index > 1 and self.values[current_index] > self.values[current_index // 2]:
                self.swap(current_index, current_index // 2)
                current_index //= 2

    def heapify_down(self, index):
        """Perform heapify operation moving down from the given index."""
        if index <= self.size // 2 and ((2 * index + 1 <= self.size and self.values[index] < max(self.values[2 * index], self.values[2 * index + 1])) or (2 * index + 1 > self.size and self.values[index] < self.values[2 * index])):
            current_index = index
            while current_index <= self.size // 2 and (((2 * current_index + 1) <= self.size and self.values[current_index] < max(self.values[2 * current_index], self.values[2 * current_index + 1])) or (2 * current_index + 1 > self.size and self.values[current_index] < self.values[2 * current_index])):
                child_index = 2 * current_index
                if (2 * current_index + 1) <= self.size and self.values[2 * current_index] < self.values[2 * current_index + 1]:
                    child_index = 2 * current_index + 1
                self.swap(current_index, child_index)
                current_index = child_index

    def swap(self, index1, index2):
        """Swap values at two given indices."""
        self.values[index1], self.values[index2] = self.values[index2], self.values[index1]
        self.heap[index1], self.heap[index2] = self.heap[index2], self.heap[index1]

    def heap_sort(self):
        """Perform heap sort and return the sorted elements."""
        sorted_edges = []
        sorted_weights = []
        while self.size >= 1:
            sorted_edges.append(self.heap[1])
            sorted_weights.append(self.values[1])
            self.delete(1)
        return [sorted_weights, sorted_edges]
