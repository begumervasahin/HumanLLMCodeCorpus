from tqdm import tqdm
import os
import time
from huffman_module import *
from comp_lib import *

def main():
    # Clear the console
    os.system('cls' if os.name == 'nt' else 'clear')

    # Print university name
    hepia_print()
    print("\n")

    # Progress bar format
    bar_format = '{l_bar}{bar}| {n_fmt}/{total_fmt}'

    # Progress bar with total steps
    with tqdm(total=9, ncols=80, bar_format=bar_format) as pbar:
        # Read user input for filenames
        filenames = read_user_input()
        update_bar(pbar, 'Open File  ')

        # Set filenames for reading and writing
        file_to_read = filenames[0]
        file_to_write = filenames[1]

        # Convert file contents to byte array
        byte_array = bytes_2_array(file_to_read)
        update_bar(pbar, 'File->Array')

        # Compute hash of the original file
        hash_value = hashOriginal(file_to_read)
        update_bar(pbar, 'Hashing    ')

        # Determine frequencies of bytes in the file
        freq_dict = determine_frequencies(byte_array)
        update_bar(pbar, 'Frequencies')

        # Perform Huffman coding
        compressed_list = huffman(freq_dict)
        update_bar(pbar, 'Huffman    ')

        # Translate byte array using Huffman encoding
        byte_string = translate(byte_array, compressed_list)
        update_bar(pbar, 'Translation')

        # Finalize byte string
        final_byte_array = finalize(byte_string)
        update_bar(pbar, 'Array->File')

        # Write Huffman dictionary to JSON file
        write_2_json(compressed_list, "dict.json")
        update_bar(pbar, 'dict.json  ')

        # Write compressed binary data to file
        write_2_binary(final_byte_array, file_to_write)
        update_bar(pbar, 'Done       ')

        # Pause briefly for visual feedback
        time.sleep(0.1)

        # Close progress bar
        pbar.close()

        # Display compression information
        compressor_display(freq_dict, compressed_list, file_to_read, file_to_write, byte_array, hash_value)

if __name__ == "__main__":
    main()
