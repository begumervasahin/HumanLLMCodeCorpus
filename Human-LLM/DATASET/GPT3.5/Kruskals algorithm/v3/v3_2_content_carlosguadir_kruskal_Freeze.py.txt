from cx_Freeze import setup, Executable

# Define executable details
executable_name = "ventana"  # Name of the executable
executable_version = "0.1"  # Version of the executable
executable_description = "ventana"  # Description of the executable
python_script = "RutasMaritimas.py"  # Python script to be turned into executable

# Configuration for building the executable
setup(
    name=executable_name,
    version=executable_version,
    description=executable_description,
    executables=[Executable(python_script)]
)
