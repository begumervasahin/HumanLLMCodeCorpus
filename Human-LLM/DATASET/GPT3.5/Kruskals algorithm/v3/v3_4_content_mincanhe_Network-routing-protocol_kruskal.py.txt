# Initialize parent and rank dictionaries
parent = {}
rank = {}

def make_set(vertice):
    """
    Create a new set with a single element.

    Args:
        vertice: The vertice to create the set for.

    Returns:
        None
    """
    parent[vertice] = vertice
    rank[vertice] = 0

def find(vertice):
    """
    Find the set representative (root) of the given vertice.

    Args:
        vertice: The vertice to find the root for.

    Returns:
        The root of the set containing the vertice.
    """
    if parent[vertice] != vertice:
        parent[vertice] = find(parent[vertice])
    return parent[vertice]

def union(vertice1, vertice2):
    """
    Merge the sets containing two vertices into a single set.

    Args:
        vertice1: The first vertice.
        vertice2: The second vertice.

    Returns:
        None
    """
    root1 = find(vertice1)
    root2 = find(vertice2)
    if root1 != root2:
        if rank[root1] < rank[root2]:
            parent[root2] = root1
        else:
            parent[root1] = root2
            if rank[root1] == rank[root2]:
                rank[root2] += 1

def kruskal(graph):
    """
    Find the maximum bandwidth path in the graph using Kruskal's algorithm.

    Args:
        graph (dict): The graph containing vertices and edges.

    Returns:
        set: The set of edges forming the maximum bandwidth path.
    """
    # Initialize sets for each vertice
    for vertice in graph['vertices']:
        make_set(vertice)

    max_bandwidth_path = set()
    # Sort edges by weight
    edges = sorted(graph['edges'])

    # Apply Kruskal's algorithm
    for edge in edges:
        weight, vertice1, vertice2 = edge
        if find(vertice1) != find(vertice2):
            union(vertice1, vertice2)
            max_bandwidth_path.add(edge)

    return max_bandwidth_path

# Example usage
graph = {
    'vertices': ['A', 'B', 'C', 'D', 'E'],
    'edges': [(4, 'A', 'B'), (2, 'A', 'D'), (5, 'B', 'C'), (2, 'B', 'D'), (3, 'D', 'E'), (5, 'C', 'E')]
}

max_bandwidth_path = kruskal(graph)
print("Max Bandwidth Path:", max_bandwidth_path)
