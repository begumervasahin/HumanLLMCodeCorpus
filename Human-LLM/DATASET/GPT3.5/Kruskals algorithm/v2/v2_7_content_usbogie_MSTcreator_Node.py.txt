class Node:
    def __init__(self, x_coordinate, y_coordinate, key_value, adjacency_list):
        """
        Initialize a Node object with its properties.

        Parameters:
        x_coordinate (int): X-coordinate of the node.
        y_coordinate (int): Y-coordinate of the node.
        key_value (int): Key value of the node.
        adjacency_list (list): List of adjacent nodes.
        """
        self.x_coordinate = x_coordinate
        self.y_coordinate = y_coordinate
        self.key_value = key_value
        self.adjacency_list = adjacency_list

    def __eq__(self, other):
        """
        Check if two Node objects are equal.

        Parameters:
        other (Node): Another Node object to compare.

        Returns:
        bool: True if the nodes are equal, False otherwise.
        """
        return (self.x_coordinate, self.y_coordinate, self.key_value) == \
               (other.x_coordinate, other.y_coordinate, other.key_value)

# Example usage
node1 = Node(1, 2, 3, [4, 5, 6])
node2 = Node(1, 2, 3, [4, 5, 6])
node3 = Node(4, 5, 6, [7, 8, 9])

print("Are node1 and node2 equal?", node1 == node2)  # Should print True
print("Are node1 and node3 equal?", node1 == node3)  # Should print False
