import graphviz as gv
from graphvizual import *

class Edge:
    def __init__(self, node_0, node_1, capacity):
        self.node_0 = node_0
        self.node_1 = node_1
        self.capacity = capacity
        self.flow = 0

class Graph:
    def __init__(self):
        self.list_edges = []
        self.max_flow = 0

    def add_edge(self, start, end, capacity):
        self.list_edges.append(Edge(start, end, capacity))
        return self

    def list_nodes(self):
        nodes = []
        for edge in self.list_edges:
            if edge.node_0 not in nodes:
                nodes.append(edge.node_0)
            if edge.node_1 not in nodes:
                nodes.append(edge.node_1)
        return nodes

    def making_friends(self, node):
        friends = []
        for edge in self.list_edges:
            if edge.node_0 == node:
                friends.append(edge)
        return friends

    def print_list_edges(self):
        list_e = []
        for edge in self.list_edges:
            list_e.append([edge.node_0, edge.node_1, edge.flow, edge.capacity])
        return list_e

    def creating_antecedents(self):
        antecedents = {}
        for node in self.list_nodes():
            antecedents[str(node)] = 0
        return antecedents

    def find_path(self, node_start, node_end):
        antecedents = self.creating_antecedents()
        list_visited_nodes = [str(node_start)]
        list_visited_edges = []
        friends = []
        roar = 1
        while roar != 20:
            for k in list_visited_nodes:
                if roar == 20:
                    break
                friends_i = self.making_friends(k)
                for edge in friends_i:
                    if edge.capacity != edge.flow:
                        antecedents[edge.node_1] = edge.node_0
                for edge in friends_i:
                    if edge not in friends:
                        friends.append(edge)
                for edge in friends:
                    if edge not in list_visited_edges and edge.node_1 not in list_visited_nodes and edge.node_0 != node_end and edge.capacity != edge.flow:
                        list_visited_edges.append(edge)
                        if edge.node_0 not in list_visited_nodes:
                            list_visited_nodes.append(edge.node_0)
                            if edge.node_0 == node_end:
                                roar = 20
                                break
                        if edge.node_1 not in list_visited_nodes:
                            list_visited_nodes.append(edge.node_1)
                            if edge.node_1 == node_end:
                                roar = 20
                                break
        node = node_end
        path_d = []
        node_ant = antecedents[node]
        end = 0
        while node_ant != 0:
            for edge in self.list_edges:
                if edge.node_1 == node and edge.node_0 == node_ant:
                    path_d.insert(0, edge)
                    node = edge.node_0
                    node_ant = antecedents[node]
        max_flow = 0
        minimum = path_d[0]
        for edge in path_d:
            if (edge.capacity - edge.flow) < (minimum.capacity - minimum.flow):
                minimum = edge
        minn = minimum.capacity - minimum.flow
        max_flow = max_flow + minn
        final_path = []
        for edge in path_d:
            edge.flow = edge.flow + minn
            final_path.append([edge.node_0, edge.node_1, edge.flow, edge.capacity])
        list_visited_nodes = []
        list_visited_edges = []
        antecedents = self.creating_antecedents()
        end = 0
        path_d = []
        return [final_path, max_flow]
