import networkx as nx
import matplotlib.pyplot as plt
from fibonacci_heap import FibonacciHeap
from prim_algorithm import prim

def build_priority_queue(vertices, keys, heap):
    """
    Build a priority queue (heap) based on the keys of vertices.

    Parameters:
    vertices (list): List of vertices.
    keys (dict): Dictionary mapping vertices to their respective keys.
    heap (FibonacciHeap): Fibonacci Heap instance.

    Returns:
    dict: Dictionary mapping vertices to their corresponding nodes in the heap.
    """
    nodes_heap = {}
    for vertex in vertices:
        nodes_heap[vertex] = heap.insert(keys[vertex], vertex)
    return nodes_heap

# Create a graph
graph = nx.Graph()
node_set = ('a', 'b', 'c', 'd')
graph.add_node('a', pos=(0, 1), key=0)
graph.add_node('b', pos=(2, 1), key=0)
graph.add_node('c', pos=(1, 0), key=0)
graph.add_node('d', pos=(1, 2), key=0)
graph.add_edge('a', 'b', weight=2)
graph.add_edge('a', 'c', weight=30)
graph.add_edge('b', 'c', weight=90)
graph.add_edge('d', 'b', weight=1)
graph.add_edge('d', 'a', weight=3)
weights = nx.get_edge_attributes(graph, 'weight')
positions = nx.get_node_attributes(graph, 'pos')
edges = graph.edges
source_node = 'a'

# Apply Prim's algorithm to get MST
minimum_spanning_tree = prim(graph, weights, source_node, node_set)
mst_edges = minimum_spanning_tree.edges()
mst_weights = nx.get_edge_attributes(minimum_spanning_tree, 'weight')

# Plot original graph and MST
plt.subplot(121)
nx.draw(graph, positions, with_labels=True)
nx.draw_networkx_edge_labels(graph, positions, edge_labels=weights)
nx.draw(graph, positions, edges=edges)
plt.title('Original Graph')

plt.subplot(122)
nx.draw(minimum_spanning_tree, positions, with_labels=True)
nx.draw_networkx_edge_labels(minimum_spanning_tree, positions, edge_labels=mst_weights)
nx.draw(minimum_spanning_tree, positions, edges=mst_edges)
plt.title('Minimum Spanning Tree (MST)')

plt.tight_layout()
plt.show()
