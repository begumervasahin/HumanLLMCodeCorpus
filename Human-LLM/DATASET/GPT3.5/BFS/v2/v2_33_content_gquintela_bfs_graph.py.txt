
class Graph:
    def __init__(self):
        # Initialize the graph with an empty adjacency list and size
        self.adjacency_list = {}
        self.size = 0

    def get_vertices(self):
        # Return a list of all vertices in the graph
        return list(self.adjacency_list.keys())

    def get_neighbours(self, v):
        # Return a list of neighbours for a given vertex v
        return self.adjacency_list.get(v, [])

    def print_vertices(self):
        # Print all vertices in the graph
        print("Vertices:")
        for vertex in self.adjacency_list:
            print(vertex)
        print('\n')

    def print_neighbours(self, v):
        # Print neighbours of a given vertex v along with the weight of edges
        print("Neighbours of", v)
        neighbours = self.adjacency_list.get(v, [])
        for neighbour, weight in neighbours:
            print(neighbour + ", weight of edge:", weight)

    def add_vertex(self, v):
        # Add a vertex v to the graph
        if v not in self.adjacency_list:
            self.adjacency_list[v] = []

    def add_undirected_edge(self, u, v):
        # Add an undirected edge between vertices u and v
        if u not in self.adjacency_list:
            self.add_vertex(u)
        if v not in self.adjacency_list:
            self.add_vertex(v)
        # Add v to the neighbours of u and vice versa
        self.adjacency_list[u].append((v, 0))
        self.adjacency_list[v].append((u, 0))

    def add_directed_edge(self, u, v):
        # Add a directed edge from vertex u to vertex v
        self.add_undirected_edge(u, v)
        # Since this is an undirected graph, add an edge from v to u as well
        self.adjacency_list[v].append((u, 0))
