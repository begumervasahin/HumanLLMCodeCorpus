class Queue:
    def __init__(self):
        # Initialize the queue with an empty list
        self.items = []

    def enqueue(self, item):
        # Add an item to the front of the queue
        self.items.insert(0, item)

    def dequeue(self):
        # Remove and return the item at the rear of the queue
        return self.items.pop()

    def is_empty(self):
        # Check if the queue is empty
        return self.items == []

    def size(self):
        # Return the number of items in the queue
        return len(self.items)


def bfs_queue(graph, start, came_from):
    # Perform Breadth-First Search using a queue
    frontier = Queue()  # Initialize a queue for traversal
    frontier.enqueue(start)  # Add the starting vertex to the queue
    came_from[start] = None  # Mark the starting vertex as having no parent

    while not frontier.is_empty():
        # Continue traversing until the queue is empty
        current_vertex = frontier.dequeue()  # Get the next vertex from the queue
        if current_vertex == "GOAL":
            return current_vertex  # Return if goal is reached

        # Explore neighbors of the current vertex
        for next_vertex in graph[current_vertex]:
            if next_vertex not in came_from:
                # If the neighbor hasn't been visited, enqueue it and mark its parent
                frontier.enqueue(next_vertex)
                came_from[next_vertex] = current_vertex

    return None  # Return None if no path to goal is found


def main():
    # Example graph representing connections between vertices
    graph = {
        "A": ["B", "C"],
        "B": ["D", "E"],
        "C": ["F"],
        "D": [],
        "E": ["GOAL"],
        "F": ["GOAL"]
    }

    came_from = {}  # Dictionary to track the parent of each vertex during traversal

    # Perform BFS using a queue starting from vertex "A"
    result = bfs_queue(graph, "A", came_from)

    if result:
        print("Path found!")  # Print if a path to the goal is found
    else:
        print("No path found.")  # Print if no path to the goal is found

    print("Came from:", came_from)  # Print the dictionary containing traversal information


if __name__ == "__main__":
    main()  # Call the main function to execute the BFS traversal
