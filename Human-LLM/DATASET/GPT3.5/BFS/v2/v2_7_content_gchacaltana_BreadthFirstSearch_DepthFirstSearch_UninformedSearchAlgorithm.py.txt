from Node import Node

class UninformedSearchAlgorithm:
    """
    Represents an uninformed search algorithm.
    """

    def __init__(self, list_nodes, searched):
        """
        Initializes the uninformed search algorithm.

        Parameters:
        - list_nodes: A list of nodes in the graph.
        - searched: The name of the node being searched for.
        """
        self.list_nodes = list_nodes
        self.searched = searched
        self.create_queue()

    def create_queue(self):
        """Creates a queue for the search algorithm."""
        self.queue = []
        self.queue.append(self.list_nodes[0])

    def read_queue(self):
        """Reads and removes the first node from the queue."""
        return self.queue.pop(0)

    def get_queue_len(self):
        """Returns the length of the queue."""
        return len(self.queue)

    def get_node_from_list(self, name):
        """
        Returns a node with the given name from the list of nodes.

        Parameters:
        - name: The name of the node to retrieve.
        """
        for node in self.list_nodes:
            if node.name == name:
                return node

    def validate_queue_length(self):
        """Raises an exception if the queue is empty."""
        if self.get_queue_len() == 0:
            raise Exception("The queue is empty")

    def match_searched(self, node_name):
        """
        Raises an exception if the given node matches the searched node.

        Parameters:
        - node_name: The name of the node to check.
        """
        if node_name == self.searched:
            raise Exception("City found: %s" % node_name)

    def search(self):
        """Performs the search algorithm."""
        pass

    def insert_node_child_queue(self, node):
        """
        Inserts child nodes of the given node into the queue.

        Parameters:
        - node: The node whose children will be inserted into the queue.
        """
        children_nodes = node.get_children_nodes()
        for child in children_nodes:
            child_node = self.get_node_from_list(child.name)
            if isinstance(child_node, Node):
                self.add_queue(child_node)
