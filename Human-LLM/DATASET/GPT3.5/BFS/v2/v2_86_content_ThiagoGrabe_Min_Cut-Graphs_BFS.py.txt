class BFS:
    def bfs(self, graph, source, terminal, family):
        # Initialize a list to keep track of visited vertices
        seen = [False] * len(graph)
        # Initialize a FIFO queue for BFS
        fifo = []
        fifo.append(source)  # Start BFS from the source vertex
        seen[source] = True  # Mark the source vertex as visited

        # Perform BFS
        while fifo:
            vertex = fifo.pop(0)  # Get the next vertex from the queue
            # Iterate over adjacent vertices of the current vertex
            for idx, value in enumerate(graph[vertex]):
                # If there's a residual capacity and the vertex hasn't been visited yet
                if value > 0 and not seen[idx]:
                    fifo.append(idx)  # Add the vertex to the queue
                    seen[idx] = True  # Mark it as visited
                    family[idx] = vertex  # Update the parent vertex in the augmenting path
        # Check if the terminal vertex is reachable
        return seen[terminal]


class EdmondsKarp:
    def __init__(self, graph):
        self.graph = graph  # Initialize the graph

    def max_flow(self, source, terminal):
        flow = 0  # Initialize the maximum flow
        family = [-1] * len(self.graph)  # Initialize the augmenting path

        bfs = BFS()  # Initialize BFS for finding augmenting paths

        # Continue while there's an augmenting path from source to terminal
        while bfs.bfs(self.graph, source, terminal, family):
            path_flow = float("inf")  # Initialize path flow to infinity
            s = terminal  # Start from the terminal vertex

            # Find the minimum residual capacity along the augmenting path
            while s != source:
                path_flow = min(path_flow, self.graph[family[s]][s])  # Update path flow
                s = family[s]  # Move to the parent vertex in the augmenting path

            flow += path_flow  # Update the maximum flow

            # Update residual capacities in the graph along the augmenting path
            v = terminal
            while v != source:
                u = family[v]
                self.graph[u][v] -= path_flow  # Update forward edge
                self.graph[v][u] += path_flow  # Update backward edge
                v = family[v]  # Move to the parent vertex

        return flow  # Return the maximum flow


def parse_input(file_path):
    with open(file_path, 'r') as file:
        lines = file.readlines()
        num_vertices, num_edges = map(int, lines[0].split())  # Get number of vertices and edges
        graph = [[0] * num_vertices for _ in range(num_vertices)]  # Initialize graph matrix
        # Parse the edges and their weights
        for line in lines[1:]:
            u, v, weight = map(int, line.split())
            graph[u][v] = weight
            graph[v][u] = weight  # Considering undirected graph
        return graph  # Return the parsed graph


def write_output(output_file, cut_size, cut_set, max_flow):
    with open(output_file, 'w') as file:
        file.write(str(cut_size) + '\n')  # Write the size of the min-cut
        file.write(' '.join(map(str, cut_set)) + '\n')  # Write the vertices in the min-cut
        file.write(str(max_flow) + '\n')  # Write the maximum flow


def main(input_file, output_file):
    graph = parse_input(input_file)  # Parse the input file to get the graph
    source = 0  # Source vertex
    terminal = len(graph) - 1  # Terminal vertex
    edmonds_karp = EdmondsKarp(graph)  # Initialize Edmonds-Karp algorithm
    max_flow = edmonds_karp.max_flow(source, terminal)  # Compute maximum flow
    # Calculate the size and vertices of the min-cut
    cut_size = sum(1 for flow in graph[source] if flow > 0)
    cut_set = [i for i, flow in enumerate(graph[source]) if flow > 0]
    write_output(output_file, cut_size, cut_set, max_flow)  # Write the output
