# A matrix representation of a directed graph.
# This is intended for testing purposes.
# For detailed information, please refer to the assignment handout.

from searchproblem import SearchProblem

class DGraph(SearchProblem):
    def __init__(self, matrix, goal_indices, start_state=0):
        """
        Initializes a directed graph.

        Parameters:
        - matrix: The matrix representation of the directed graph.
        - goal_indices: A Python set of indices representing goal states.
        - start_state: The index of the start state. Default is 0.
        """
        self.matrix = matrix
        self.goal_indices = goal_indices
        self.start_state = start_state

    def get_start_state(self):
        """
        Returns the start state of the graph.
        """
        return self.start_state

    def is_goal_state(self, state):
        """
        Checks if the given state is a goal state.

        Parameters:
        - state: The state to be checked.

        Returns:
        - True if the state is a goal state, False otherwise.
        """
        return state in self.goal_indices

    def get_successors(self, state):
        """
        Returns successors of the given state along with their costs.

        Parameters:
        - state: The state whose successors are to be found.

        Returns:
        - A dictionary containing successors as keys and their costs as values.
        """
        row = self.matrix[state]
        successors = {}
        index = 0
        for cost in row:
            if cost is not None:
                successors[index] = cost
            index += 1
        return successors
