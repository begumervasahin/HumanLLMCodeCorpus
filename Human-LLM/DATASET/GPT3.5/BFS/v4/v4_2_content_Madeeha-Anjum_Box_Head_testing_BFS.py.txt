import collections

def bfs(grid, start):
    queue = collections.deque([[start]])
    seen = set([start])
    
    while queue:
        path = queue.popleft()
        x, y = path[-1]
        
        if grid[y][x] == goal:
            return path  # Return the final path if goal is reached
        
        for x2, y2 in ((x+1, y), (x-1, y), (x, y+1), (x, y-1)):
            if 0 <= x2 < collombs and 0 <= y2 < rows and grid[y2][x2] != wall and (x2, y2) not in seen:
                queue.append(path + [(x2, y2)])
                seen.add((x2, y2))

# Define symbols for wall, clear space, and goal
wall, clear, goal = "#", ".", "*"

# Define grid size
collombs, rows = 10, 5

# Define grid layout
grid = ["..........",
        "...#*..##.",
        "..##...#..",
        ".....###..",
        ".........."]
