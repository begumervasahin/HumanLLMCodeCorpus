import wikipedia

def search(root, target):
    """
    Function to find the shortest path between two Wikipedia pages using breadth-first search algorithm.

    Args:
        root (str): The starting Wikipedia page.
        target (str): The target Wikipedia page.

    Returns:
        None
    """
    # Validate and clean input
    root = wikipedia.check(root)
    target = wikipedia.check(target)

    if root and target:
        # Ensure root and target are different
        if root != target:
            visited = set([root])
            queue = [[root]]
            found = False

            print("Finding the shortest route from '{}' to '{}'...".format(root, target))

            # BFS algorithm
            while queue and not found:
                try:
                    # Get the current path
                    path = queue[0]
                    current = path[-1]
                    print(" > ".join(path))
                    queue = queue[1:]

                    # Get links of the current page
                    nodes = wikipedia.get_links(current)

                    # Check each link
                    for node in nodes:
                        if node not in visited:
                            if node == target:
                                result = path + [node]
                                found = True
                            visited.add(node)
                            new_path = path + [node]
                            queue.append(new_path)
                except KeyboardInterrupt:
                    exit("Keyboard Interrupt")

            # Print the shortest path if found
            if found:
                print("Shortest path: ")
                print(" > ".join(result))
            else:
                print("No possible route")
        else:
            print("Root and target are the same")

# Input the starting page
start = input("Page to start at: ")

# If input is '?', randomly select two pages
if start == "?":
    start, end = wikipedia.random(2)
else:
    end = input("Page to find: ")

# Call the search function
search(start, end)
