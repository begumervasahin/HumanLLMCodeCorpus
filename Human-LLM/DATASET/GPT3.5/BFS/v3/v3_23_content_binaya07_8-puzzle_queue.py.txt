class Queue:
    """
    A class representing a queue.
    """

    def __init__(self):
        """
        Initialize an empty queue.
        """
        self.items = []  # List to store queue elements
        self.visited_states = set()  # Set to keep track of visited states

    def enqueue(self, item):
        """
        Add an item to the end of the queue.

        Args:
            item: The item to enqueue.
        """
        self.items.append(item)  # Append item to the end of the list
        self.visited_states.add(tuple(item.list))  # Add the state to the visited set

    def dequeue(self):
        """
        Remove and return the first item from the queue.

        Returns:
            The first item in the queue.
        """
        if not self.is_empty():
            item = self.items.pop(0)  # Remove and return the first item
            self.visited_states.remove(tuple(item.list))  # Remove the state from the visited set
            return item

    def is_empty(self):
        """
        Check if the queue is empty.

        Returns:
            True if the queue is empty, False otherwise.
        """
        return len(self.items) == 0
