def binary_search(input_array, value):
    # Initialize upper and lower bounds for the search
    upper_bound = len(input_array) - 1
    lower_bound = 0

    # Continue searching until the bounds overlap
    while lower_bound <= upper_bound:
        # Calculate the middle index
        mid = (upper_bound + lower_bound) // 2

        # If the value is found at the middle index, return its position
        if input_array[mid] == value:
            return mid

        # If the value is smaller than the middle element, adjust the upper bound
        elif value < input_array[mid]:
            upper_bound = mid - 1

        # If the value is larger than the middle element, adjust the lower bound
        else:
            lower_bound = mid + 1

    # If the value is not found, return -1
    return -1

# Test cases
test_list = [1, 3, 9, 11, 15, 19, 29]
test_val1 = 25
test_val2 = 15
print(binary_search(test_list, test_val1))  # Output: -1 (not found)
print(binary_search(test_list, test_val2))  # Output: 4 (found at index 4)
