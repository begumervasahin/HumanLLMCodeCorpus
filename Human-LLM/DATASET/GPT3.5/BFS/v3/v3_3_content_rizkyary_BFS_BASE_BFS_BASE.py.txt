def bfs(graph, start, goal):
    # Initialize a queue with the start node as the initial path
    queue = [[start]]

    # Initialize a set to keep track of visited nodes
    visited = set()

    # Continue searching until the queue is empty
    while queue:
        # Dequeue the first path from the queue
        path = queue.pop(0)

        # Get the last node in the current path
        current_node = path[-1]

        # If the goal node is found, return the path
        if current_node == goal:
            return path

        # If the current node has not been visited yet
        elif current_node not in visited:
            # Explore all neighbors of the current node
            neighbors = graph.get(current_node, [])
            for neighbor in neighbors:
                # Create a new path by appending the neighbor to the current path
                new_path = path + [neighbor]

                # Add the new path to the queue
                queue.append(new_path)

            # Mark the current node as visited
            visited.add(current_node)

    # If the goal node is not found, return None
    return None
