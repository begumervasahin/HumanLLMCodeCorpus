import pytest
import networkx as nx
from max_flow import flow_with_demands


def divergence(flow):
    """
    Computes the net flow into each node based on the given flow dictionary.

    Args:
        flow: A dictionary representing flow between nodes.

    Returns:
        A dictionary containing the net flow into each node.
    """
    div = {node: 0 for node in flow}
    for node in flow:
        forward_flow = sum(flow[node].values())
        backward_flow = sum(flow[neighbor][node] for neighbor in flow[node])
        div[node] = backward_flow - forward_flow
    return div


def test_max_flow():
    # Creating a graph from the contiguous USA data
    G = nx.Graph()
    with open('contiguous-usa.dat') as usa:
        for line in usa:
            s1, s2 = line.strip().split()
            G.add_edge(s1, s2)

    # Setting demands for each state
    for state in G.nodes():
        if state != 'CA':
            G.nodes[state]['demand'] = 1
    G.nodes['CA']['demand'] = -48

    # Converting the graph to a directed graph
    G = nx.DiGraph(G)

    # Assigning uniform capacity to all edges
    uniform_capacity = 16
    for s1, s2 in G.edges():
        G.edges[s1, s2]['capacity'] = uniform_capacity

    # Computing flow with demands using the function to be tested
    flow = flow_with_demands(G)

    # Computing net flow into each node
    div = divergence(flow)

    # Assertion to check if the net flow satisfies the demands of each node
    assert all(div[node] == G.nodes[node]['demand'] for node in G.nodes())


# Allowing the test function to be run if this script is executed directly
if __name__ == '__main__':
    test_max_flow()
