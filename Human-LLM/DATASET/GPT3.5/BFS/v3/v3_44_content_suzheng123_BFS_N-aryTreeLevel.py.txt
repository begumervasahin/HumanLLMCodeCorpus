class TreeNode:
    def __init__(self, value, children=None):
        self.value = value
        self.children = children if children else []

class Solution:
    def level_order_traversal(self, root):
        """
        Perform level order traversal on the given tree.

        Args:
        root (TreeNode): The root node of the tree.

        Returns:
        list: A list of lists containing the values of the nodes at each level.
        """
        if not root:
            return []

        result = []  # Store the level order traversal
        queue = [root]  # Initialize a queue to store nodes for traversal
        
        while queue:
            level_values = []  # Store values at the current level
            next_level = []  # Store nodes for the next level
            
            for node in queue:
                level_values.append(node.value)
                next_level.extend(node.children)
                
            queue = next_level  # Move to the next level
            result.append(level_values)

        return result
