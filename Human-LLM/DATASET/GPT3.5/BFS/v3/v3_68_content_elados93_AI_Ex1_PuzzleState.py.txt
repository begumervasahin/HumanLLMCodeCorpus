class State:
    """
    Class representing a state of the puzzle.

    Attributes:
        board (list): The configuration of the puzzle board.
        move (object): The move that led to this state.
        came_from (State): The previous state.
    """

    def __init__(self, board=None, move=None, came_from=None, state=None):
        """
        Initialize a State object.

        Args:
            board (list): The configuration of the puzzle board.
            move (object): The move that led to this state.
            came_from (State): The previous state.
            state (State): Another State object to copy from.
        """
        if state is None:
            # Constructor with board and move
            self._board = board
            self._move = move
            self.came_from = came_from
        else:
            # Copy constructor
            self._board = state.board
            self._move = state.move
            self.came_from = state.came_from

    def __hash__(self):
        """
        Return the hash value of the State object.
        """
        return hash(self._board)

    def __eq__(self, other):
        """
        Check if two State objects are equal.
        """
        return self._board == other.board

    @property
    def board(self):
        """
        Return the board configuration of the State.
        """
        return self._board

    @property
    def move(self):
        """
        Return the move that led to this state.
        """
        return self._move


class HeuristicState(State):
    """
    Class representing a state of the puzzle with heuristic costs.

    Attributes:
        h_cost (int): The heuristic cost.
        g_cost (int): The cost to reach this state.
    """

    def __init__(self, board=None, move=None, came_from=None, state=None):
        """
        Initialize a HeuristicState object.

        Args:
            board (list): The configuration of the puzzle board.
            move (object): The move that led to this state.
            came_from (State): The previous state.
            state (State): Another State object to copy from.
        """
        if state is None:
            # Constructor using other parameters
            super().__init__(board, move, came_from)
        else:
            # Constructor using a given state
            super().__init__(state=state)
        self.h_cost = 0
        self.g_cost = 0

    def __lt__(self, other):
        """
        Compare two HeuristicState objects based on their f_cost.
        """
        return self.f_cost < other.f_cost

    @property
    def f_cost(self):
        """
        Return the total cost (f_cost) of the HeuristicState.
        """
        return self.h_cost + self.g_cost
