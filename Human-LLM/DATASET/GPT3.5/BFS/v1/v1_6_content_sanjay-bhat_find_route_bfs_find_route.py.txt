import sys

class RomaniaMap:
    """Represents the map of Romania with cities and distances."""

    def __init__(self):
        """Initialize the RomaniaMap object."""
        self.graph = {}

    def get_data_from_file(self, filename):
        """Read data from the given file and construct the graph."""
        with open(filename, 'r') as file:
            for line in file:
                city1, city2, distance = line.strip().split()
                self.add_edge(city1, city2, int(distance))
                self.add_edge(city2, city1, int(distance))

    def add_edge(self, city1, city2, distance):
        """Add an edge between two cities with the given distance."""
        if city1 not in self.graph:
            self.graph[city1] = []
        self.graph[city1].append((city2, distance))

    def get_shortest_path(self, source, destination):
        """Find the shortest path between two cities using BFS."""
        queue = [(source, [source])]
        while queue:
            (current_city, path) = queue.pop(0)
            if current_city == destination:
                return path
            for (next_city, _) in self.graph.get(current_city, []):
                if next_city not in path:
                    queue.append((next_city, path + [next_city]))
        return None

def main():
    if len(sys.argv) != 4:
        print("Usage: python find_route.py <input_file> <source_city> <destination_city>")
        return

    input_file = sys.argv[1]
    source_city = sys.argv[2]
    destination_city = sys.argv[3]

    romania_map = RomaniaMap()
    romania_map.get_data_from_file(input_file)
    shortest_path = romania_map.get_shortest_path(source_city, destination_city)

    if shortest_path:
        total_distance = 0
        path_str = "route:\n"
        for i in range(len(shortest_path) - 1):
            current_city = shortest_path[i]
            next_city = shortest_path[i + 1]
            for (neighbor, distance) in romania_map.graph[current_city]:
                if neighbor == next_city:
                    path_str += f"{current_city} to {next_city}, {distance} km\n"
                    total_distance += distance
                    break
        path_str = f"distance: {total_distance} km\n" + path_str
        print(path_str)
    else:
        print("No route found between the given cities.")

if __name__ == "__main__":
    main()
