import sys
from PIL import Image


class MazeSolver:
    """Class for solving mazes using the BFS algorithm."""

    def __init__(self, input_file, output_file):
        """Initialize the MazeSolver object."""
        self.colors = {
            "WHITE": (255, 255, 255),
            "RED": (255, 0, 0),
            "GREEN": (0, 255, 0),
        }
        self.output_file = output_file
        self.image = Image.open(input_file).convert('RGB')
        self.bitmap = self.image.load()
        self.finish, self.start = self.find_start_end()

    def find_start_end(self):
        """Find the start and end points in the maze."""
        width, height = self.image.size
        start, end = None, None
        for x in range(width):
            for y in range(height):
                pixel_color = self.bitmap[x, y]
                if pixel_color == self.colors["RED"]:
                    end = (x, y)
                elif pixel_color == self.colors["GREEN"]:
                    start = (x, y)
        if start is not None and end is not None:
            return end, start
        print("No start or end found")
        sys.exit(1)

    def run(self):
        """Run the BFS algorithm and draw the solution path."""
        solution = self.bfs(self.start, self.finish)
        if solution is None:
            print("No solution found")
            sys.exit(1)

        # Draw the path.
        for pos in solution:
            x, y = pos
            self.bitmap[x, y] = self.colors["RED"]
        self.image.save(self.output_file)
        print("Solution saved to", self.output_file)

    def is_valid_position(self, x, y):
        """Check if coordinates are inside the picture."""
        width, height = self.image.size
        return 0 <= x < width and 0 <= y < height

    def neighbor_pixels(self, position):
        """Get neighboring pixels."""
        x, y = position
        return [(x + 1, y), (x, y + 1), (x - 1, y), (x, y - 1)]

    def bfs(self, start, end):
        """Breadth-first search algorithm."""
        image = self.image.copy()
        pixels = image.load()

        options = [[start]]
        visited = set()

        while options:
            path = options.pop(0)
            pos = path[-1]
            visited.add(pos)

            if pos == end:
                return path

            for x, y in self.neighbor_pixels(pos):
                if (x, y) not in visited and self.is_valid_position(x, y) \
                        and (pixels[x, y] == self.colors["WHITE"] or pixels[x, y] == self.colors["RED"]):
                    pixels[x, y] = self.colors["GREEN"]
                    new_path = list(path)
                    new_path.append((x, y))
                    options.append(new_path)
        return None


if __name__ == '__main__':
    if len(sys.argv) == 3:
        solver = MazeSolver(input_file=sys.argv[1], output_file=sys.argv[2])
        solver.run()
    else:
        print("Usage: python maze_solve.py <input_file.png> <output_file.png>")
        sys.exit(1)
