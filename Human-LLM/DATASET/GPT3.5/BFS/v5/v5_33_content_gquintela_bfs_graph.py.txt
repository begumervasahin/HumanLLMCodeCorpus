class Graph:
    def __init__(self):
        """
        Initializes an empty graph.
        """
        self.adjacency_list = {}  # Dictionary to store the adjacency list representation of the graph
        self.size = 0  # Number of vertices in the graph

    def get_vertices(self):
        """
        Returns a list of all vertices in the graph.
        """
        return list(self.adjacency_list.keys())

    def get_neighbors(self, vertex):
        """
        Returns a list of neighbors of a given vertex.
        """
        return self.adjacency_list[vertex] if vertex in self.adjacency_list else []

    def print_vertices(self):
        """
        Prints all vertices in the graph.
        """
        print("Vertices:")
        for vertex in self.adjacency_list:
            print(vertex)
        print('\n')

    def print_neighbors(self, vertex):
        """
        Prints the neighbors of a given vertex.
        """
        print("Neighbors of " + vertex)
        for neighbor in self.adjacency_list.get(vertex, []):
            print(neighbor)

    def add_vertex(self, vertex):
        """
        Adds a vertex to the graph.
        """
        if vertex not in self.adjacency_list:
            self.adjacency_list[vertex] = []

    def add_undirected_edge(self, u, v):
        """
        Adds an undirected edge between vertices u and v.
        """
        if u not in self.adjacency_list:
            self.add_vertex(u)
        if v not in self.adjacency_list:
            self.add_vertex(v)
        self.adjacency_list[u].append(v)
        self.adjacency_list[v].append(u)

    def add_directed_edge(self, u, v):
        """
        Adds a directed edge from vertex u to vertex v.
        """
        if u not in self.adjacency_list:
            self.add_vertex(u)
        if v not in self.adjacency_list:
            self.add_vertex(v)
        self.adjacency_list[u].append(v)
