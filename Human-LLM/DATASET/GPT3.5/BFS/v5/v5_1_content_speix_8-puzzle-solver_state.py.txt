from typing import Any, Optional

class State:
    """
    Represents a state in a search algorithm.

    Attributes:
        state: The state data.
        parent: The parent state from which this state was reached.
        move: The action that led to this state.
        depth: The depth of this state in the search tree.
        cost: The cost associated with reaching this state.
        key: A unique key for this state.
        map_str: A string representation of the state.
    """
    def __init__(
        self,
        state: Any,
        parent: Optional['State'],
        move: Any,
        depth: int,
        cost: float,
        key: Any
    ) -> None:
        self.state = state
        self.parent = parent
        self.move = move
        self.depth = depth
        self.cost = cost
        self.key = key

        # Generate a string representation of the state
        if self.state is not None:
            self.map_str = ''.join(map(str, self.state))

    def __eq__(self, other: 'State') -> bool:
        """Check if two states are equal based on their string representations."""
        return self.map_str == other.map_str

    def __lt__(self, other: 'State') -> bool:
        """Compare two states based on their string representations."""
        return self.map_str < other.map_str
