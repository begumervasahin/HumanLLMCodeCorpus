import collections

def bfs(grid, start):
    """
    Perform Breadth-First Search (BFS) on a grid.

    Args:
        grid (list): The grid to search.
        start (tuple): The starting position.

    Returns:
        list or None: The path from start to goal, or None if goal is not reachable.
    """
    queue = collections.deque([[start]])
    seen = set([start])

    while queue:
        path = queue.popleft()
        x, y = path[-1]

        if grid[y][x] == goal:
            return path  # Return the final path if goal is reached

        # Explore neighboring cells
        for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nx, ny = x + dx, y + dy
            if 0 <= nx < columns and 0 <= ny < rows and grid[ny][nx] != wall and (nx, ny) not in seen:
                queue.append(path + [(nx, ny)])
                seen.add((nx, ny))

# Define symbols for wall, clear space, and goal
wall, clear, goal = "#", ".", "*"

# Define grid size
columns, rows = 10, 5

# Define grid layout
grid = ["..........",
        "...#*..##.",
        "..##...#..",
        ".....###..",
        ".........."]
