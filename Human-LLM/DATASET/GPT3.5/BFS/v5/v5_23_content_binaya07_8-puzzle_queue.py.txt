class Queue:
    """A class representing a queue."""

    def __init__(self):
        """Initialize an empty queue."""
        self.items = []  # List to store queue elements
        self.state = set()  # Set to keep track of visited states

    def enqueue(self, item):
        """Add an item to the end of the queue."""
        self.items.append(item)  # Append item to the end of the list
        self.state.add(tuple(item.list))  # Add the state to the visited set

    def dequeue(self):
        """Remove and return the first item from the queue."""
        if not self.is_empty():
            item = self.items.pop(0)  # Remove and return the first item
            self.state.remove(tuple(item.list))  # Remove the state from the visited set
            return item

    def is_empty(self):
        """Check if the queue is empty."""
        return len(self.items) == 0
