import base64
import hashlib
from Crypto import Random
from Crypto.Cipher import AES

BLOCK_SIZE = 16

def pad_message(message):
    """
    Pad the message to a multiple of the block size.

    Args:
        message (str): The message to be padded.

    Returns:
        str: The padded message.
    """
    padding_length = BLOCK_SIZE - len(message) % BLOCK_SIZE
    padding = chr(padding_length) * padding_length
    return message + padding

def unpad_message(padded_message):
    """
    Remove padding from the message.

    Args:
        padded_message (str): The padded message.

    Returns:
        str: The unpadded message.
    """
    padding_length = ord(padded_message[-1])
    return padded_message[:-padding_length]

class AESAlgorithm:
    def encrypt(self, message, symmetric_key):
        """
        Encrypt a message using AES algorithm.

        Args:
            message (str): The message to be encrypted.
            symmetric_key (str): The symmetric key used for encryption.

        Returns:
            bytes: The encrypted message.
        """
        padded_message = pad_message(message)
        hashed_key = hashlib.sha256(symmetric_key.encode('utf-8')).digest()
        iv = Random.new().read(AES.block_size)
        cipher = AES.new(hashed_key, AES.MODE_CBC, iv)
        encrypted_message = cipher.encrypt(padded_message)
        return base64.b64encode(iv + encrypted_message)

    def decrypt(self, encrypted_message, symmetric_key):
        """
        Decrypt an encrypted message using AES algorithm.

        Args:
            encrypted_message (bytes): The encrypted message.
            symmetric_key (str): The symmetric key used for decryption.

        Returns:
            str: The decrypted message.
        """
        encrypted_message = base64.b64decode(encrypted_message)
        iv = encrypted_message[:AES.block_size]
        hashed_key = hashlib.sha256(symmetric_key.encode('utf-8')).digest()
        cipher = AES.new(hashed_key, AES.MODE_CBC, iv)
        decrypted_message = cipher.decrypt(encrypted_message[AES.block_size:])
        return unpad_message(decrypted_message)

# Create an instance of AESAlgorithm
aes_algorithm = AESAlgorithm()

# Define the plaintext message and symmetric key
plain_text = "Hello World"
symmetric_key = "63757843836583645936548364836"

# Encrypt the plaintext message and print the result
encrypted_text = aes_algorithm.encrypt(plain_text, symmetric_key)
print("Encrypted Text:", encrypted_text.decode())

# Decrypt the encrypted message and print the result
decrypted_text = aes_algorithm.decrypt(encrypted_text, symmetric_key)
print("Decrypted Text:", decrypted_text.decode())
