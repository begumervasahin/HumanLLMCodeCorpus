class Point:
    def __init__(self, x_coord, y_coord, z_coord, modulus):
        """
        Initialize a point with its coordinates and modulus.

        :param x_coord: x-coordinate of the point
        :param y_coord: y-coordinate of the point
        :param z_coord: z-coordinate of the point
        :param modulus: modulus for arithmetic operations
        """
        self.x = x_coord
        self.y = y_coord
        self.z = z_coord
        self.p = modulus

    def __repr__(self):
        """
        Return a string representation of the point.

        :return: string representation of the point
        """
        return f"({self.x}, {self.y}, {self.z})"

    def __eq__(self, other_point):
        """
        Check if two points are equal based on their coordinates and modulus.

        :param other_point: other Point object to compare
        :return: True if equal, False otherwise
        """
        return (
            self.x == other_point.x
            and self.y == other_point.y
            and self.z == other_point.z
            and self.p == other_point.p
        )

    def negate(self):
        """
        Negate the y-coordinate of the point modulo p.

        :return: the point with negated y-coordinate
        """
        self.y = (-self.y) % self.p
        return self

    @staticmethod
    def make_point(x_coord, y_coord, z_coord, modulus):
        """
        Static method to create a new Point instance.

        :param x_coord: x-coordinate of the point
        :param y_coord: y-coordinate of the point
        :param z_coord: z-coordinate of the point
        :param modulus: modulus for arithmetic operations
        :return: new Point instance
        """
        return Point(x_coord, y_coord, z_coord, modulus)

# Example usage:
point1 = Point(1, 2, 3, 5)
print(point1)  # Output: (1, 2, 3)

point2 = Point.make_point(4, 5, 6, 7)
print(point2)  # Output: (4, 5, 6)

point3 = Point(1, 2, 3, 5)
print(point1 == point3)  # Output: True

point4 = Point(1, 2, 4, 5)
print(point1 == point4)  # Output: False

point1.negate()
print(point1)  # Output: (1, 3, 3)
