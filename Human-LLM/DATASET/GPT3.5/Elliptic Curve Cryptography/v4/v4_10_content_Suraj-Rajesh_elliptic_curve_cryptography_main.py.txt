from ecc.ecc import EllipticCurve
from ecc.galois_field.galois_field import GaloisField
from ecc.galois_field.generic import is_prime

def main():
    # Initialize lists to store points and their corresponding orders
    points = []
    orders = []
    top_k_points = []
    top_k_orders = []

    # Create a Galois Field with modulus 103
    gf_103 = GaloisField(103)

    # Iterate over all possible values of 'a' and 'b'
    for a in range(gf_103.p):
        for b in range(gf_103.p):
            # Create an elliptic curve with parameters 'a' and 'b' and modulus 103
            elliptic_curve = EllipticCurve(a, b, 103)
            # Calculate the order of the curve
            order = elliptic_curve.order_of_curve()
            # Check if the order is prime
            if is_prime(order):
                # Store the order and corresponding parameters 'a' and 'b'
                orders.append(order)
                points.append((a, b))

    # Select the top 10 points with the highest orders
    for _ in range(10):
        # Find the maximum order
        maximum = max(orders)
        maximums_index = orders.index(maximum)
        # Add the point and its order to the top-k lists        
        top_k_orders.append(maximum)
        top_k_points.append(points[maximums_index])
        # Remove the selected point and order from the lists    
        del orders[maximums_index]
        del points[maximums_index]

    # Print the top 10 points and their orders
    for i in range(10):
        print("{} : {}".format(top_k_points[i], top_k_orders[i]))

if __name__ == "__main__":
    main()
