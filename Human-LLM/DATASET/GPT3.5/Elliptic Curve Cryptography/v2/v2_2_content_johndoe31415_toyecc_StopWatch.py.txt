import time

class StopWatch:
    def __init__(self, component=None, noisy=False):
        # Initialize the stopwatch with an optional component name and noise flag
        self._component = component
        self._noisy = noisy
        self.reset()

    @property
    def finishtime(self):
        # Property to get the finish time
        return self._finishtime

    def stop(self):
        # Stop the stopwatch and calculate the elapsed time
        self._finishtime = time.time() - self._t
        return self.finishtime

    def finish(self):
        # Stop the stopwatch and print the elapsed time if noisy flag is set
        self._finishtime = self.stop()
        if self._noisy:
            print(f"{self._component} took {str(self)}")

    def reset(self):
        # Reset the stopwatch
        self._finishtime = None
        self._t = time.time()

    def __str__(self):
        # Format the elapsed time for human-readable output
        t = self.stop()
        if t < 1:
            return f"{round(1000 * t)} ms"
        elif t < 10:
            return f"{t:.1f} sec"
        else:
            tint = round(t)
            if tint < 60:
                return f"{tint} sec"
            elif tint < 3600:
                return f"{tint // 60}:{tint % 60:02d} m:s"
            elif tint < 86400:
                return f"{tint // 3600}:{tint % 3600 // 60:02d}:{tint % 3600 % 60:02d} h:m:s"
            else:
                return f"{tint // 86400}-{tint % 86400 // 3600}:{tint % 86400 % 3600 // 60:02d}:{tint % 86400 % 3600 % 60:02d} d-h:m:s"

    def __enter__(self):
        # Context manager entry point, reset the stopwatch
        self.reset()

    def __exit__(self, type, value, traceback):
        # Context manager exit point, finish the stopwatch
        self.finish()

if __name__ == "__main__":
    # Example usage
    x = StopWatch("foobar", True)
    time.sleep(0.1)
    x.finish()
    print("Finish time:", x.finishtime)

    with StopWatch("foobar2", True):
        time.sleep(0.1)
