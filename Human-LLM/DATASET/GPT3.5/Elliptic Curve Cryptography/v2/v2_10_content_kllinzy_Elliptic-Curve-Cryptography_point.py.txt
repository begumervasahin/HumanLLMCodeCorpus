class Point:
    def __init__(self, x, y, z, p):
        self.x = x  # x-coordinate of the point
        self.y = y  # y-coordinate of the point
        self.z = z  # z-coordinate of the point
        self.p = p  # modulus for arithmetic operations

    def __repr__(self):
        return f"({self.x}, {self.y}, {self.z})"  # String representation of the point

    def __eq__(self, other):
        # Check if two points are equal based on their coordinates and modulus
        return self.x == other.x and self.y == other.y and self.z == other.z and self.p == other.p

    def negate(self):
        # Negate the y-coordinate of the point modulo p
        self.y = (-self.y) % self.p
        return self

    @staticmethod
    def make_point(x, y, z, p):
        # Static method to create a new Point instance
        return Point(x, y, z, p)

# Example usage:
point1 = Point(1, 2, 3, 5)
print(point1)  # Output: (1, 2, 3)

point2 = Point.make_point(4, 5, 6, 7)
print(point2)  # Output: (4, 5, 6)

point3 = Point(1, 2, 3, 5)
print(point1 == point3)  # Output: True

point4 = Point(1, 2, 4, 5)
print(point1 == point4)  # Output: False

point1.negate()
print(point1)  # Output: (1, 3, 3)
