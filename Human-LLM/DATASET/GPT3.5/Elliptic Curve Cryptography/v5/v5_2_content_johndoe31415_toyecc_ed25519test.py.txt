import sys
from toyecc import getcurvebyname, ECPrivateKey
from StopWatch import StopWatch

def main():
    """Main function for generating keypair, signing, and verifying a message."""
    # Get the curve
    curve = getcurvebyname("ed25519")

    # Check if a keypair is provided as a command-line argument
    if len(sys.argv) < 2:
        # Generate a keypair if no argument is provided
        keypair = ECPrivateKey.eddsa_generate(curve)
        print("Generating keypair on the fly")
    else:
        # Load the keypair from the provided argument
        keypair = ECPrivateKey.loadkeypair(bytes.fromhex(sys.argv[1]))

    print("Keypair:", keypair)

    # Define the message
    msg = b"Foobar!"
    print("Message:", msg)

    # Sign the message
    signature = keypair.eddsa_sign(msg)
    print("Signature:", signature)

    # Verify the signature with the correct message
    print("Verify correct message: %s (should be True)" % (keypair.pubkey.eddsa_verify(msg, signature)))

    # Verify the signature with a forged message
    print("Verify forged message : %s (should be False)" % (keypair.pubkey.eddsa_verify(msg + b"x", signature)))

if __name__ == "__main__":
    main()
