from Crypto.PublicKey import RSA
from Crypto import Random
import ast

def generate_rsa_key_pair():
    """Generate an RSA key pair with a key size of 1024 bits."""
    random_generator = Random.new().read
    key_pair = RSA.generate(1024, random_generator)
    return key_pair

def encrypt_message(public_key, message):
    """Encrypt a plaintext message using the provided public key."""
    encrypted_text = public_key.encrypt(message.encode(), 32)
    return encrypted_text

def decrypt_message(key_pair, encrypted_text):
    """Decrypt an encrypted message using the provided key pair."""
    decrypted_text = key_pair.decrypt(ast.literal_eval(str(encrypted_text)))
    return decrypted_text.decode()

def main():
    # Generate RSA key pair
    key_pair = generate_rsa_key_pair()
    
    # Extract the public key from the key pair
    public_key = key_pair.publickey()
    
    # Define the plaintext message to be encrypted
    plain_text = "Hello World"
    
    # Encrypt the plaintext message using the public key
    encrypted_text = encrypt_message(public_key, plain_text)
    
    # Print the encrypted text
    print("Encrypted Text:", encrypted_text)
    
    # Decrypt the encrypted text using the private key
    decrypted_text = decrypt_message(key_pair, encrypted_text)
    
    # Print the decrypted text
    print("Decrypted Text:", decrypted_text)

if __name__ == "__main__":
    main()
