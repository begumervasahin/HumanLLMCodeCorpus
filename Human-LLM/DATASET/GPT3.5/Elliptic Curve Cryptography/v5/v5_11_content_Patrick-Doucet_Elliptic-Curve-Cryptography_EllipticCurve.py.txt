import matplotlib.pyplot as plt
import math

class EllipticCurve:
    def __init__(self, prime_modulus, coefficient_a, coefficient_b, base_point, order, cofactor):
        """
        Initialize an EllipticCurve object with the specified parameters.

        :param prime_modulus: Prime modulus for the curve
        :param coefficient_a: Coefficient 'a' in the curve equation
        :param coefficient_b: Coefficient 'b' in the curve equation
        :param base_point: Base point of the curve
        :param order: Order of the base point
        :param cofactor: Cofactor of the curve
        """
        self.p = prime_modulus
        self.a = coefficient_a
        self.b = coefficient_b
        self.G = base_point
        self.n = order
        self.h = cofactor

    def plot_curve(self, n_points):
        """
        Plot the EllipticCurve by generating 'n_points' points on the curve.

        :param n_points: Number of points to plot on the curve
        """
        x_values = []
        y_values = []
        for i in range(n_points):
            x_values.append(i)
            y_values.append(math.sqrt(i**3 + self.a*i + self.b) % self.p)
        plt.plot(x_values, y_values)
        plt.show()

class EllipticCurvePoint(EllipticCurve):
    def __init__(self, x_coord, y_coord, curve):
        """
        Initialize a point on the EllipticCurve with coordinates (x, y) and associated curve.

        :param x_coord: x-coordinate of the point
        :param y_coord: y-coordinate of the point
        :param curve: EllipticCurve object to which the point belongs
        """
        super().__init__(curve.p, curve.a, curve.b, curve.G, curve.n, curve.h)
        self.x = x_coord
        self.y = y_coord
        self.curve = curve

    def __add__(self, other):
        """
        Add two points on the EllipticCurve.

        :param other: Another EllipticCurvePoint object
        :return: Result of the addition operation as a new EllipticCurvePoint object
        """
        if self.x < other.x:
            a = self
            b = other
        else:
            a = other
            b = self

        if a.x == 0 and a.y == 0:
            return b
        elif b.x == 0 and b.y == 0:
            return a
        elif b == a.ec_inv():
            return EllipticCurvePoint(0, 0, a.curve)
        else:
            if a == b:
                m = (3 * (a.x**2) + a.curve.a) * a.inv_mod(2*a.y)
            else:
                m = (b.y - a.y) * a.inv_mod(b.x - a.x)

        point = EllipticCurvePoint(0, 0, a.curve)
        point.x = ((m**2 % a.curve.p) - a.x - b.x) % a.curve.p
        point.y = (m * (a.x - point.x) - a.y) % a.curve.p
        return point

    def inv_mod(self, x):
        """
        Compute the modular inverse of 'x' modulo the curve's prime modulus.

        :param x: Input value
        :return: Modular inverse of 'x' modulo the curve's prime modulus
        """
        return pow(x, self.curve.p-2, self.curve.p)

    def ec_inv(self):
        """
        Compute the inverse of the point on the elliptic curve.

        :return: Inverse point on the elliptic curve
        """
        if self.x == 0 and self.y == 0:
            return self
        return EllipticCurvePoint(self.x, (-self.y) % self.curve.p, self.curve)

    def __eq__(self, other):
        """
        Check if two EllipticCurvePoint objects are equal.

        :param other: Another EllipticCurvePoint object
        :return: True if equal, False otherwise
        """
        return self.x == other.x and self.y == other.y and self.curve == other.curve    

    def __ne__(self, other):
        """
        Check if two EllipticCurvePoint objects are not equal.

        :param other: Another EllipticCurvePoint object
        :return: True if not equal, False otherwise
        """
        return not self == other

    def multiply(self, scalar):
        """
        Multiply the point by a scalar value using the double-and-add algorithm.

        :param scalar: Scalar value for multiplication
        :return: Result of the multiplication as a new EllipticCurvePoint object
        """
        binary_string = bin(scalar)[2:]
        temp_point = EllipticCurvePoint(self.x, self.y, self.curve)
        p_point = EllipticCurvePoint(self.x, self.y, self.curve)
        binary_string.pop()
        binary_string.reverse()
        for bit in binary_string:
            if bit == '0':
                temp_point = temp_point + temp_point
                temp_point.x %= self.curve.p
                temp_point.y %= self.curve.p
            else:
                temp_point = temp_point + temp_point
                temp_point.x %= self.curve.p
                temp_point.y %= self.curve.p
                temp_point = temp_point + p_point
                temp_point.x %= self.curve.p
                temp_point.y %= self.curve.p
        return temp_point
