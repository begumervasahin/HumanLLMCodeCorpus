def euclidean_gcd(a, b):
    """
    Calculate the greatest common divisor (GCD) of two numbers using the Euclidean algorithm.

    Args:
        a (int): The first number.
        b (int): The second number.

    Returns:
        int: The GCD of a and b.
    """
    if a == 0:
        return b
    if b == 0:
        return a
    return euclidean_gcd(b, a % b)


def extended_euclidean(a, b):
    """
    Calculate the extended Euclidean algorithm to find the GCD of two numbers,
    and provides additional values x, y such that ax + by = gcd(a, b).

    Args:
        a (int): The first number.
        b (int): The second number.

    Returns:
        tuple: A tuple containing x, y, and the GCD of a and b.
    """
    x, y, gcd = _extended_euclidean_util(a, b, 1, 1)
    return x, y, gcd


def _extended_euclidean_util(a, b, x, y):
    """
    Utility function to calculate x, y, and GCD for ax + by = gcd(a, b).

    Args:
        a (int): The first number.
        b (int): The second number.
        x (int): Auxiliary variable.
        y (int): Auxiliary variable.

    Returns:
        tuple: A tuple containing x, y, and the GCD of a and b.
    """
    if a == 0:
        x = 0
        y = 1
        return x, y, b
    x1, y1, gcd = _extended_euclidean_util(b % a, a, x, y)
    x = y1 - (b // a) * x1
    y = x1
    return x, y, gcd


def inverse_brute_force(n, p):
    """
    Calculate the inverse in modular arithmetic using brute force method.
    Calculate the inverse of n in n (mod p).

    Args:
        n (int): The number for which inverse is to be calculated.
        p (int): The modulus.

    Returns:
        int: The inverse of n modulo p.
    """
    for i in range(p):
        if (n * i) % p == 1:
            return i


def inverse_extended_euclidean(n, p):
    """
    Calculate the inverse in modular arithmetic using the extended Euclidean method.    
    Calculate the inverse of n in n (mod p).

    Args:
        n (int): The number for which inverse is to be calculated.
        p (int): The modulus.

    Returns:
        int: The inverse of n modulo p.
    """
    n = n % p
    inv, _, _ = extended_euclidean(n, p)
    return inv % p
