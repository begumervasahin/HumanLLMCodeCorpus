import math

class EllipticCurve:
    def __init__(self, a, b, p, Gx, Gy):
        self.a = a
        self.b = b
        self.p = p
        self.Gx = Gx
        self.Gy = Gy

    def is_valid_curve(self):
        """
        Checks if the provided curve parameters satisfy the necessary conditions.       
        """
        return (4 * (self.a ** 3) + 27 * (self.b ** 2)) != 0

    def is_point_on_curve(self, x, y):
        """
        Checks if a point (x, y) lies on the elliptic curve.
        """
        rhs = (x ** 3 + self.a * x + self.b) % self.p
        lhs = (y ** 2) % self.p
        return lhs == rhs

    def point_addition(self, x1, y1, x2, y2):
        """
        Performs point addition on the elliptic curve.
        """
        sn = (y1 - y2) % self.p
        sd = x1 - x2
        sd_inverse = self.modulo_multiplicative_inverse(sd)
        s = (sn * sd_inverse) % self.p
        xr = (s ** 2 - (x1 + x2)) % self.p
        yr = (s * (x1 - xr) - y1) % self.p
        return xr, yr

    def point_doubling(self, x, y):
        """
        Performs point doubling on the elliptic curve.
        """
        sn = (3 * (x ** 2) + self.a) % self.p
        sd = 2 * y
        sd_inverse = self.modulo_multiplicative_inverse(sd)
        s = (sn * sd_inverse) % self.p
        xr = (s ** 2 - 2 * x) % self.p
        yr = (s * (x - xr) - y) % self.p
        return xr, yr

    def scalar_multiplication(self, x, y, n):
        """
        Performs scalar multiplication of a point (x, y) on the elliptic curve.
        """
        xr, yr = self.point_doubling(x, y)
        n -= 2
        while n != 0:
            xr, yr = self.point_addition(x, y, xr, yr)
            n -= 1
        return xr, yr

    def modulo_multiplicative_inverse(self, A):
        """
        Computes the modulo multiplicative inverse of a number A.
        """
        return self.fast_power(A, self.p - 2)

    def fast_power(self, base, power):
        """
        Computes the fast exponentiation of a number.
        """
        result = 1
        while power > 0:
            if power % 2 == 1:
                result = (result * base) % self.p
            power //= 2
            base = (base * base) % self.p
        return result

def perform_key_exchange(curve):
    """
    Performs key exchange between Alice and Bob using elliptic curve cryptography.      
    """
    bob_private_key = int(input("Enter Bob's private key: "))
    alice_private_key = int(input("Enter Alice's private key: "))

    pbx, pby = curve.scalar_multiplication(curve.Gx, curve.Gy, bob_private_key)
    pax, pay = curve.scalar_multiplication(curve.Gx, curve.Gy, alice_private_key)       

    shared_key_bob = curve.scalar_multiplication(pax, pay, bob_private_key)
    shared_key_alice = curve.scalar_multiplication(pbx, pby, alice_private_key)

    if shared_key_bob == shared_key_alice:
        print("Same keys generated.")
        print("Shared public key: ({}, {})".format(shared_key_bob[0], shared_key_bob[1]))
        print("Is point on curve?", curve.is_point_on_curve(shared_key_bob[0], shared_key_bob[1]))
        return shared_key_bob

def main():
    curve_params = EllipticCurve(0, 0, 0, 0, 0)

    while not curve_params.is_valid_curve():
        curve_params.a = int(input("Enter the curve parameter a: "))
        curve_params.b = int(input("Enter the curve parameter b: "))

        curve_params.p = int(input("Enter a large prime number p: "))

        curve_params.Gx = int(input("Enter the curve parameter Gx: "))
        curve_params.Gy = int(input("Enter the curve parameter Gy: "))

        if not curve_params.is_point_on_curve(curve_params.Gx, curve_params.Gy):        
            print("Point not on elliptic curve")

    perform_key_exchange(curve_params)

if __name__ == "__main__":
    main()
