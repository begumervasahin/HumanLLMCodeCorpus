class Point:
    """
    Represents a point in three-dimensional space with coordinates (x, y, z) and a modulus p.
    """

    def __init__(self, x: int, y: int, z: int, p: int) -> None:
        """
        Initialize a Point object with coordinates and a modulus.

        :param x: x-coordinate of the point
        :param y: y-coordinate of the point
        :param z: z-coordinate of the point
        :param p: modulus for arithmetic operations
        """
        self.x = x
        self.y = y
        self.z = z
        self.p = p

    def __repr__(self) -> str:
        """
        Return a string representation of the Point object.

        :return: string representation of the Point object
        """
        return f"({self.x}, {self.y}, {self.z})"

    def __eq__(self, other: 'Point') -> bool:
        """
        Check if two Point objects are equal.

        :param other: other Point object to compare
        :return: True if equal, False otherwise
        """
        return self.x == other.x and self.y == other.y and self.z == other.z and self.p == other.p

    @staticmethod
    def make_point(x: int, y: int, z: int, p: int) -> 'Point':
        """
        Static method to create a new Point object.

        :param x: x-coordinate of the point
        :param y: y-coordinate of the point
        :param z: z-coordinate of the point
        :param p: modulus for arithmetic operations
        :return: new Point object
        """
        return Point(x, y, z, p)

    def negate(self) -> 'Point':
        """
        Negate the y-coordinate of the Point object modulo p.

        :return: the Point object with negated y-coordinate
        """
        self.y = (-self.y) % self.p
