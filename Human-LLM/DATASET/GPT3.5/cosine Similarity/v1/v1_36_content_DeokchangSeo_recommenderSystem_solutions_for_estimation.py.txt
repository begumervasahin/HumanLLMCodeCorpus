from __future__ import (absolute_import, division, print_function, unicode_literals)    
from collections import defaultdict
from surprise import KNNBasic
from surprise import Dataset, Reader

# Function to load dataset from file
def load_dataset(dataset_path, line_format='user item rating', rating_scale=(1, 7), sep='\t'):
    name = None
    file_format = Reader(name, line_format, sep, rating_scale)
    dataset = Dataset.load_from_file(dataset_path, file_format)
    return dataset

# Function to create a full training set from dataset
def create_full_trainset(dataset_path, line_format=None, rating_scale=(1, 7), sep='\t'):
    data = load_dataset(dataset_path, line_format, rating_scale, sep)
    trainset = data.build_full_trainset()
    return trainset

# Function to convert testset ratings to binary values
def ground_truth(testset, threshold=3):
    modified_testset = []
    for x in testset:
        if x[2] > threshold:
            modified_testset.append((x[0], x[1], 1.0))
        else:
            modified_testset.append((x[0], x[1], 0.0))
    return modified_testset

# Function to get top-N recommendations for each user
def get_top_n(predictions, n=10):
    top_n = defaultdict(list)
    for uid, iid, true_r, est, _ in predictions:
        top_n[uid].append((iid, est))
    for uid, user_ratings in top_n.items():
        user_ratings.sort(key=lambda x: x[1], reverse=True)
        top_n[uid] = user_ratings[:n]
    return top_n

# Function to create dictionaries of recommended items and true positive items
def create_k_reco_posi(prediction, testset, threshold=3):
    k_recommend = {}
    for uid, recommendations in prediction.items():
        k_recommend[uid] = [iid for (iid, _) in recommendations]
    p_items = {}
    for uid, recommends in k_recommend.items():
        temp_A = [x[1] for x in testset if x[0] == uid and x[2] == 1]
        p_items[uid] = temp_A
    return k_recommend, p_items

# Function to calculate precision
def precision(prediction, testset, threshold=3):
    k_recommend, p_items = create_k_reco_posi(prediction, testset, threshold)
    precision_dict = {}
    print("\nUser\t k-recommended\t\t True positive items")
    for uid1, recommend in k_recommend.items():
        for uid2, positives in p_items.items():
            if uid1 == uid2:
                print(uid1, '\t', recommend, '\t', positives)
                precision_dict[uid1] = (len(set(recommend) & set(positives)) / len(recommend)) * 100
    return precision_dict

# Function to calculate recall
def recall(prediction, testset, threshold=3):
    k_recommend, p_items = create_k_reco_posi(prediction, testset, threshold)
    recall_dict = {}
    print("\nUser\t k-recommended\t\t True positive items")
    for uid1, recommend in k_recommend.items():
        for uid2, positives in p_items.items():
            if uid1 == uid2:
                print(uid1, '\t', recommend, '\t', positives)
                if len(positives) == 0:
                    recall_dict[uid1] = "Cannot be divided by zero"
                else:
                    recall_dict[uid1] = (len(set(recommend) & set(positives)) / len(positives)) * 100
    return recall_dict

# File path
file_path = 'ratings_for_question1.txt'

# Create trainset
trainset = create_full_trainset(file_path, line_format='user item rating', rating_scale=(1, 7), sep='\t')

# Create testset
testset = trainset.build_testset()

# Set threshold
threshold = 3

# Print ground truth
print(ground_truth(testset, threshold))

# Train algorithm
algo = KNNBasic()
algo.train(trainset)

# Get predictions
predictions = algo.test(testset)

# Get top-N recommendations
top_n = get_top_n(predictions, n=3)

# Print precision
print(precision(top_n, testset, threshold))

# Print recall
print(recall(top_n, testset, threshold))
