import math
import re
import operator

# Load stop words
stop_list = []
with open('stop_list.py') as stop_file:
    stop_list = stop_file.read().split()

# Function to preprocess text
def preprocess_text(text):
    words = re.findall(r'\w+', text.lower())
    return [word for word in words if word not in stop_list]

# Function to calculate TF-IDF
def calculate_tfidf(docs, num_docs):
    tfidf = []
    words = {}
    docs_containing_word = {}
    
    # Calculate term frequency (TF)
    for doc in docs:
        vector = {}
        for word in doc:
            vector[word] = vector.get(word, 0) + 1
            words[word] = words.get(word, 0) + 1
            if word not in docs_containing_word:
                docs_containing_word[word] = set()
            docs_containing_word[word].add(len(tfidf))
        tfidf.append(vector)
    
    # Calculate inverse document frequency (IDF)
    idf = {word: math.log(num_docs / freq) for word, freq in words.items()}
    
    # Calculate TF-IDF
    for i, vector in enumerate(tfidf):
        tfidf_vector = {word: freq * idf[word] / math.log(len(vector)) for word, freq in vector.items()}
        tfidf[i] = tfidf_vector
    
    return tfidf, idf, docs_containing_word

# Load queries and abstracts
with open('cran.qry') as qry_file:
    qry_text = qry_file.read()
    queries = [preprocess_text(qry) for qry in qry_text.split('.I')[1:]]

with open('cran.all.1400') as abstract_file:
    abstract_text = abstract_file.read()
    abstracts = [preprocess_text(abst) for abst in re.findall(r'\.I[\s\S]*?\.W', abstract_text)]

# Calculate TF-IDF for queries and abstracts
q_tfidf, q_idf, q_docs_containing_word = calculate_tfidf(queries, len(queries))
a_tfidf, a_idf, a_docs_containing_word = calculate_tfidf(abstracts, len(abstracts))

# Calculate scores
output = []
for q_vector in q_tfidf:
    scores = {}
    for i, a_vector in enumerate(a_tfidf):
        sum_of_products = sum(q_vector[word] * a_vector.get(word, 0) for word in q_vector)
        sum_of_queries = sum(q_vector[word] ** 2 for word in q_vector)
        sum_of_abstracts = sum(a_vector[word] ** 2 for word in a_vector)
        qa_product = sum_of_queries * sum_of_abstracts
        if sum_of_products > 0:
            scores[i] = sum_of_products / math.sqrt(qa_product)
        else:
            scores[i] = 0
    sorted_scores = sorted(scores.items(), key=operator.itemgetter(1), reverse=True)
    output.append(sorted_scores)

# Write output to file
with open('output.txt', 'w') as out_file:
    for i, query in enumerate(output):
        for doc, score in query:
            out_file.write(f"{i} {doc} {score}\n")
