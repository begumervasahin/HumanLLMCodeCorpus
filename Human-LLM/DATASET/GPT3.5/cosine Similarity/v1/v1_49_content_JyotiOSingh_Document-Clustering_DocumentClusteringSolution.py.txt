import urllib.request
import re
import math
from bs4 import BeautifulSoup

def read_web_page_content(url):
    """Read the webpage contents and strip the HTML tags."""
    html = urllib.request.urlopen(url).read()
    soup = BeautifulSoup(html, "lxml")
    for script in soup(["script", "style"]):
        script.extract()
    text = soup.get_text()
    return text.lower()

def convert_content_to_word_list(content):
    """Convert the content to a list of words."""
    return re.compile(r'\W+', re.UNICODE).split(content)

def strip_common_words(word_list):
    """Remove common stop words from the word list."""
    common_words = ["the", "for", "of", "if", "to", "are", "a", "is", "as", "at", "any", "in", "on", "can", "an",
                    "or", "not", "that", "than", "which", 'very', 'well', 'was', 'that', 'then', 'and', 'be',
                    'also', 'such', 'from', 'into']
    return [word for word in word_list if word not in common_words]

def calculate_term_frequency(term_list):
    """Calculate the term frequency in the given word list."""
    term_frequency = {term: term_list.count(term) for term in term_list}
    return term_frequency

def calc_cosine_similarity_num(dict1, dict2):
    """Calculate the numerator of cosine similarity."""
    sum_val = 0
    for key in dict1:
        if key in dict2:
            sum_val += dict1[key] * dict2[key]
    return sum_val

def summation_of_count(input_dict):
    """Calculate the summation of counts in the given dictionary."""
    return sum(val**2 for val in input_dict.values())

def calculate_cosine_similarity(url1, url2):
    """Calculate cosine similarity between two webpages."""
    web_content1 = read_web_page_content(url1)
    web_content2 = read_web_page_content(url2)
    word_list1 = strip_common_words(convert_content_to_word_list(web_content1))
    word_list2 = strip_common_words(convert_content_to_word_list(web_content2))
    term_freq_dict1 = calculate_term_frequency(word_list1)
    term_freq_dict2 = calculate_term_frequency(word_list2)
    cosine_numerator = calc_cosine_similarity_num(term_freq_dict1, term_freq_dict2)
    summation_str1 = summation_of_count(term_freq_dict1)
    summation_str2 = summation_of_count(term_freq_dict2)
    cosine_denom = math.sqrt(summation_str1) * math.sqrt(summation_str2)
    cosine_similarity = cosine_numerator / cosine_denom if cosine_denom != 0 else 0
    return cosine_similarity

def read_input_file(file_name):
    """Read URLs from the input file and store them in a dictionary."""
    url_dict = {}
    try:
        with open(file_name, "r") as file:
            for idx, line in enumerate(file, start=1):
                url_dict[idx] = line.strip()
    except IOError:
        print("Error encountered, unable to find the given file:", file_name)
    except Exception as e:
        print("Error while trying to open the given file:", e)
    return url_dict

def calc_cosine_similarity_using_input_file():
    """Calculate cosine similarity for URLs provided in an input file."""
    web_pages_dict = read_input_file("input.txt")
    with open("output.txt", "w") as output_file:
        output_file.write("Following webpages have been processed to find the cosine similarity:\n\n")
        for key, url in web_pages_dict.items():
            output_file.write(f"{key}: {url}\n\n")
        output_file.write("\nCosine similarity for the above webpages is:\n\n")
        for key1, url1 in web_pages_dict.items():
            similarity_values = []
            for key2, url2 in web_pages_dict.items():
                cosine_similarity = round(calculate_cosine_similarity(url1, url2), 2)
                similarity_values.append(cosine_similarity)
                print(f"Cosine Similarity between webpage {key1} and webpage {key2} is: {cosine_similarity}")
            output_file.write(f"{key1}: {similarity_values}\n")
        output_file.write("\nHighest matches for each webpage:\n\n")
        for key1, url1 in web_pages_dict.items():
            max_similarity = max(similarity_values)
            max_index = similarity_values.index(max_similarity) + 1
            output_file.write(f"{key1}: {web_pages_dict[max_index]}\n")

calc_cosine_similarity_using_input_file()
