# Similarity functions
def shot_percentage_similarity(prospect, nba_player):
    numerator = (
        prospect['FGP'] * nba_player['FGP'] +
        prospect['2PP'] * nba_player['2PP'] +
        prospect['FTP'] * nba_player['FTP']
    )
    prosp_denom = (
        prospect['FGP']**2 + prospect['2PP']**2 + prospect['FTP']**2
    )
    comp_denom = (
        nba_player['FGP']**2 + nba_player['2PP']**2 + nba_player['FTP']**2
    )
    if (nba_player['3PP'] == 0.000 or nba_player['3PP'] is None) and prospect['3PA'] <= 1.0:
        return numerator / (sqrt(prosp_denom) * sqrt(comp_denom))
    elif (nba_player['3PP'] == 0.000 or nba_player['3PP'] is None) or (nba_player['3PA'] < 1.0 and prospect['3PA'] > 1.0):
        return 0
    else:
        numerator += prospect['3PP'] * nba_player['3PP']
        comp_denom += nba_player['3PP']**2
        prosp_denom += prospect['3PP']**2
    denom = sqrt(prosp_denom) * sqrt(comp_denom)
    return numerator / denom
