import distsim
from collections import defaultdict

def main():
    # Load word vectors
    word_to_vec_dict = distsim.load_word2vec("nyt_word2vec.4k")

    # Dictionary to store task accuracies
    task_accuracies = defaultdict(list)

    # Open word-test file
    with open("word-test.v3.txt", "r") as file:
        for line in file:
            line = line.strip('\n')
            if line.startswith('//'):
                continue
            if line.startswith(":"):
                category = line[2:]
            else:
                words = line.split()
                w1_dict = word_to_vec_dict[words[0]]
                w2_dict = word_to_vec_dict[words[1]]
                w4_dict = word_to_vec_dict[words[3]]
                # Calculate the vector representation for the fourth word
                fourth_word_vec = w1_dict - w2_dict + w4_dict
                # Find nearest word to the calculated vector excluding words in the set
                result = distsim.show_nearest(word_to_vec_dict, fourth_word_vec, set([words[0], words[1], words[3]]), distsim.cossim_dense)
                # Check if the correct word is in the result
                match = False
                for i, (word, _) in enumerate(result, 1):
                    if word == words[2]:
                        match = True
                        task_accuracies[category].append(i)
                        break
                if not match:
                    # If the correct word is not found, append 0
                    task_accuracies[category].append(0)

    # Calculate and print accuracies
    for key, value in task_accuracies.items():
        top1 = sum(1 for pos in value if pos == 1)
        top5 = sum(1 for pos in value if pos <= 5)
        top10 = sum(1 for pos in value if pos <= 10)
        total = len(value)
        print(f"{key}:     {top1 / total:.2f}    {top5 / total:.2f}    {top10 / total:.2f}")

if __name__ == "__main__":
    main()
