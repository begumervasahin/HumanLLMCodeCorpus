import distsim

def print_nearest_words(word_to_ccdict, word, word_set, similarity_func):
    print(f"Word is::::{word}")
    for i, (nearest_word, score) in enumerate(distsim.show_nearest(word_to_ccdict, word_to_ccdict[word], set([word]), similarity_func), start=1):
        print(f"{i}: {nearest_word} ({score})")
    print("--------------------")

# Load word contexts
word_to_ccdict = distsim.load_contexts("nytcounts.4k")

# Print nearest words for each target word
print_nearest_words(word_to_ccdict, 'america', set(['america']), distsim.cossim_sparse)
print_nearest_words(word_to_ccdict, 'years', set(['years']), distsim.cossim_sparse)
print_nearest_words(word_to_ccdict, 'great', set(['great']), distsim.cossim_sparse)
print_nearest_words(word_to_ccdict, 'run', set(['run']), distsim.cossim_sparse)
print_nearest_words(word_to_ccdict, 'between', set(['between']), distsim.cossim_sparse)
print_nearest_words(word_to_ccdict, 'wife', set(['wife']), distsim.cossim_sparse)

# Further exploration: comparing 'big' and 'tiny' with 'great'
print("Further exploring:big from great")
print_nearest_words(word_to_ccdict, 'big', set(['big']), distsim.cossim_sparse)
print("--------------------")
print("Further exploring:tiny from great")
print_nearest_words(word_to_ccdict, 'tiny', set(['tiny']), distsim.cossim_sparse)
print("--------------------")
