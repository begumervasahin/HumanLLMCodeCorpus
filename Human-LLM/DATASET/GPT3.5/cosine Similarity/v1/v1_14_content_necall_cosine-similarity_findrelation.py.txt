import distsim

def validate_digit(num):
    mark = 0
    value = ''
    strnum = str(num)
    for s in range(len(strnum)):
        if mark == 1 and strnum[s] == '0':
            mark += 1
            if strnum[s+1] >= '5':
                value += '1'
            else:
                value += '0'
            return value
        else:
            value += strnum[s]
        if strnum[s] != '0' and strnum[s] != '.':
            mark += 1
        if mark == 2:
            if len(strnum) == 4:
                return value
            elif strnum[s+1] >= '5':
                a = str(int(value[-1]) + 1)
                return value[:-1] + a
            return value
    return value

# Load word vectors
word_to_vec_dict = distsim.load_word2vec("nyt_word2vec.4k")

# Read and process the input file
with open("word-test.v3.txt", "r") as f:
    lines = f.readlines()

for line in lines[1:]:
    parts = line.strip().split(":")
    title = parts[0]
    word_pairs = parts[1].split("\n")
    
    best1 = 0
    best5 = 0
    best10 = 0
    total_words = len(word_pairs) - 1
    
    for word_pair in word_pairs[1:]:
        words = word_pair.strip().split()
        if len(words) == 1:
            continue
        
        w1 = word_to_vec_dict[words[0]]
        w2 = word_to_vec_dict[words[1]]
        w4 = word_to_vec_dict[words[3]]
        
        ret = distsim.show_nearest(word_to_vec_dict, w1 - w2 + w4, 
                                   set([words[0], words[1], words[3]]), 
                                   distsim.cossim_dense)
        
        if words[2] == ret[0][0]:
            best1 += 1
        for r in range(5):
            if words[2] == ret[r][0]:
                best5 += 1
        for r in range(10):
            if words[2] == ret[r][0]:
                best10 += 1
    
    acc_best1 = validate_digit(float(best1) / total_words)
    acc_best5 = validate_digit(float(best5) / total_words)
    acc_best10 = validate_digit(float(best10) / total_words)
    
    print(f"{title}: acc1: {acc_best1}, acc5: {acc_best5}, acc10: {acc_best10}")
