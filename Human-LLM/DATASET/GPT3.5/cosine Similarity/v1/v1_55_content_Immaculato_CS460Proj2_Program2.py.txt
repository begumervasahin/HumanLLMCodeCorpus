import sys
import random
import copy


class KNearestNeighbor:
    def __init__(self, file_contents, debug=False):
        self.debug = debug
        self.file_contents = file_contents
        self.user_ratings = {}
        self.users = set()
        self.movie_indexes = set()

        # Parse file contents
        for line in file_contents:
            parsed_line = line.split('\t')
            user_id, movie_index, rating = int(parsed_line[0]), int(parsed_line[1]), float(parsed_line[2])

            # Add user ID to set of users
            self.users.add(user_id)

            # Add movie index to set of movie indexes
            self.movie_indexes.add(movie_index)

            # Store user ratings
            if user_id not in self.user_ratings:
                self.user_ratings[user_id] = {}
            self.user_ratings[user_id][movie_index] = rating

        # Fill missing ratings with default value (2.5)
        for user in self.users:
            for i in range(1, max(self.movie_indexes) + 1):
                if i not in self.user_ratings[user]:
                    self.user_ratings[user][i] = 2.5

    def cross_validate(self, k_values, folds):
        fold_list = []
        elements_per_bin = len(self.file_contents) // folds

        # Partition folds for cross-validation
        for i in range(folds):
            fold_list.append([])
            for j in range(elements_per_bin):
                index_add = random.choice(range(len(self.file_contents)))
                fold_list[i].append(self.file_contents[index_add])
                del self.file_contents[index_add]

        # Perform cross-validation
        for k in k_values:
            average_mean_squared_errors = {}
            total_mean_squared_error = 0.0

            for i in range(folds):
                # Prepare training data for current fold
                training_list = [item for sublist in fold_list[:i] + fold_list[i + 1:] for item in sublist]
                current_fold = KNearestNeighbor(training_list, debug=False)

                # Prepare test data for current fold
                test_ratings = {}
                actual_ratings = []
                for line in fold_list[i]:
                    parsed_line = line.split('\t')
                    user_id, movie_index, rating = int(parsed_line[0]), int(parsed_line[1]), float(parsed_line[2])
                    if user_id not in test_ratings:
                        test_ratings[user_id] = []
                    test_ratings[user_id].append(movie_index)
                    actual_ratings.append(rating)

                # Calculate mean squared error for current fold
                mean_squared_error = 0.0
                index = 0
                num_users = float(len(test_ratings))
                for user_id in test_ratings:
                    prediction = current_fold.k_nearest_neighbors_prediction(user_id, k, test_ratings[user_id])
                    for j in range(len(prediction)):
                        mean_squared_error += (prediction[j] - actual_ratings[index]) ** 2
                        index += 1
                    percent_done = round((float(index) / len(actual_ratings)) * 100, 2)
                    print('Progress: [%.2f%%] (Fold %d for K value %d)' % (percent_done, i, k))

                total_mean_squared_error += mean_squared_error / len(actual_ratings)

            # Calculate average mean squared error for current k value
            total_mean_squared_error /= folds
            average_mean_squared_errors[k] = total_mean_squared_error
            print('Average mean squared error for k =', k, total_mean_squared_error)

        # Determine best k value
        best_k = min(average_mean_squared_errors, key=average_mean_squared_errors.get)
        print('Best k value:', best_k)
        return best_k
