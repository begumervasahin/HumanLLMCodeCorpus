import nltk
from nltk.tokenize import word_tokenize, sent_tokenize
from nltk.corpus import stopwords
from nltk.stem import PorterStemmer
from collections import Counter
import string
import os

class QAEngine:
    def __init__(self):
        nltk.download('punkt')
        nltk.download('stopwords')
        self.stop_words = set(stopwords.words('english'))
        self.stemmer = PorterStemmer()
        self.story = self.load_story()
        self.questions = self.load_questions()

    def load_story(self):
        # Load story data (replace with your own data loading logic)
        story = {}
        story_file = open('./story.txt', 'r', encoding='utf-8')
        story['text'] = story_file.read()
        story_file.close()
        return story

    def load_questions(self):
        # Load questions data (replace with your own data loading logic)
        questions = []
        questions_file = open('./questions.txt', 'r', encoding='utf-8')
        for line in questions_file:
            qid, text = line.split('::')
            questions.append({'qid': qid.strip(), 'text': text.strip()})
        questions_file.close()
        return questions

    def preprocess_text(self, text):
        # Tokenize, remove punctuation, lowercase, remove stop words, and stem
        tokens = word_tokenize(text)
        tokens = [token.lower() for token in tokens if token not in string.punctuation]
        tokens = [token for token in tokens if token not in self.stop_words]
        tokens = [self.stemmer.stem(token) for token in tokens]
        return tokens

    def calculate_tf_idf(self, tokens):
        # Calculate TF-IDF for tokens in a document
        tf = Counter(tokens)
        tf_idf = {}
        for token, freq in tf.items():
            idf = self.calculate_idf(token)
            tf_idf[token] = freq * idf
        return tf_idf

    def calculate_idf(self, token):
        # Calculate IDF for a token in the story
        token_count = sum(token in self.preprocess_text(para) for para in self.story['text'].split('\n') if para.strip())
        idf = 1 + (token_count > 0)
        return idf

    def answer_questions(self):
        answers = []
        for question in self.questions:
            tokens = self.preprocess_text(question['text'])
            tf_idf = self.calculate_tf_idf(tokens)
            answer = self.find_most_similar_paragraph(tf_idf)
            answers.append({'qid': question['qid'], 'answer': answer})
        return answers

    def find_most_similar_paragraph(self, query_tf_idf):
        # Find the most similar paragraph using TF-IDF
        max_similarity = 0
        most_similar_para = ''
        for para in self.story['text'].split('\n'):
            if para.strip():
                para_tokens = self.preprocess_text(para)
                para_tf_idf = self.calculate_tf_idf(para_tokens)
                similarity = sum(query_tf_idf[token] * para_tf_idf.get(token, 0) for token in query_tf_idf)
                if similarity > max_similarity:
                    max_similarity = similarity
                    most_similar_para = para
        return most_similar_para

    def save_answers(self, answers):
        # Save answers to a file (replace with your own saving logic)
        with open('./answers.txt', 'w', encoding='utf-8') as answers_file:
            for answer in answers:
                answers_file.write(f"{answer['qid']}:: {answer['answer']}\n")

def main():
    qa_engine = QAEngine()
    answers = qa_engine.answer_questions()
    qa_engine.save_answers(answers)

if __name__ == "__main__":
    main()
