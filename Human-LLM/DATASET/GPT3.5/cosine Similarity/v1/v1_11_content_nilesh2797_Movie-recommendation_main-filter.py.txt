import csv
from math import sqrt

# Read movie ratings data
with open('movie-ratings.csv', 'r') as file1:
    reader1 = csv.reader(file1)
    data = list(reader1)

movie = data[0][1:]
data.pop(0)

ratings = []
critic_names = []

for item in data:
    critic_names.append(item[0])
    ratings.append({movie[i]: float(item[i + 1]) for i in range(len(movie))})

# Read user preferences data
with open('user_preference.csv', 'r') as file2:
    reader2 = csv.reader(file2)
    user_data = list(reader2)

user_dic = {user_data[0][i]: float(user_data[1][i]) for i in range(len(user_data[0]))}

# Calculate Pearson correlation coefficient
corr_coeff = []
for critic in ratings:
    xy = []
    x = []
    y = []
    x_square = []
    y_square = []
    for movie_name in user_dic.keys():
        if critic.get(movie_name) is not None and user_dic[movie_name] != -1:
            xy.append(critic[movie_name] * user_dic[movie_name])
            x.append(critic[movie_name])
            y.append(user_dic[movie_name])
            x_square.append(critic[movie_name] ** 2)
            y_square.append(user_dic[movie_name] ** 2)

    n = len(xy)
    if n * sum(x_square) - sum(x) * sum(x) != 0 and n * sum(y_square) - sum(y) * sum(y) != 0:
        corr_coeff.append((n * sum(xy) - sum(x) * sum(y)) / (sqrt(n * sum(x_square) - sum(x) * sum(x)) * sqrt(n * sum(y_square) - sum(y) * sum(y))))
    else:
        corr_coeff.append(0)

pearson_order = sorted(zip(critic_names, corr_coeff), key=lambda x: x[1], reverse=True)

# Calculate weighted average movie rating
movie_rating = {}
for movie_name in movie:
    weight_rating = []
    sum_corr_coeff = []
    for i in range(len(corr_coeff)):
        if ratings[i].get(movie_name) is not None:
            weight_rating.append(corr_coeff[i] * ratings[i][movie_name])
            sum_corr_coeff.append(abs(corr_coeff[i]))
    movie_rating[movie_name] = sum(weight_rating) / sum(sum_corr_coeff)

rating_order = sorted(movie_rating, key=movie_rating.get, reverse=True)

# Output top 3 recommended movies
i = 1
for movie_name in rating_order:
    if user_dic[movie_name] == -1 and i < 4:
        print(movie_name)
        i += 1
