import json
import jieba
import re
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

# Load data from JSON file
def load_data(filename):
    with open(filename, 'r', encoding='utf-8') as file:
        data = json.load(file)
    return data

# Tokenize text using Jieba and remove stop words
def tokenize_text(content, stopwordset):
    article = ''
    words = jieba.cut(content, cut_all=False)
    for word in words:
        m = re.match(r'^[\u4E00-\u9FFFa-zA-Z]+$', word)
        if m is not None:
            if word not in stopwordset:
                article += word
                article += ' '
    return article.strip()

# Calculate TF-IDF and extract top keywords
def calculate_tfidf(contents):
    tfIdfVectorizer = TfidfVectorizer()
    tfIdf = tfIdfVectorizer.fit_transform(contents)
    myWord = tfIdfVectorizer.get_feature_names()
    weight = tfIdf.toarray()
    textWeightList = []
    for i in range(len(weight)):
        textMining = {}
        for j in range(len(myWord)):
            textMining[myWord[j]] = weight[i][j]
        textWeightList.append(textMining)
    return textWeightList

# Extract top keywords and assign tags to articles
def extract_keywords(text_weight_list):
    tagList = []
    for oneArticle in text_weight_list:
        sorted_dict = sorted(oneArticle.items(), key=lambda d: d[1], reverse=True)
        tag = [item[0] for item in sorted_dict[:5]]
        tagList.append(tag)
    return tagList

# Load stopwords
def load_stopwords(filename):
    with open(filename, 'r', encoding='utf-8') as file:
        stopwords = set(line.strip('\n') for line in file)
    return stopwords

def main():
    allData = load_data('news.json')
    stopwordset = load_stopwords('stopwords.txt')
    totalContent = [tokenize_text(article['Content'], stopwordset) for article in allData[:50]]
    textWeightList = calculate_tfidf(totalContent)
    tagList = extract_keywords(textWeightList)
