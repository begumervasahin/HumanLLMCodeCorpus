import logging
from flask import Flask, jsonify, request
from flask_cors import CORS
from tv_show_predictor_main import scrape_data  # Assuming tv_show_predictor_main contains the scrape_data function
import os
from data_processor import DataProcessor  # Assuming DataProcessor is defined in data_processor.py
import pandas as pd
from fuzzywuzzy import fuzz
from fuzzywuzzy import process
import pickle

# Configure logging
logging.basicConfig(filename='logging/web_server.log', filemode='w', format='%(asctime)s: %(name)s - %(levelname)s - %(message)s', datefmt='%Y-%m-%d %H:%M:%S')
logging.getLogger().setLevel(logging.INFO)

# Initialize Flask app
app = Flask(__name__)
cors = CORS(app, resources={r"/predict": {"origins": "*"}})

# Remove existing cosine_model.pkl if exists
if os.path.exists("data/cosine_model.pkl"):
    os.remove("data/cosine_model.pkl")

# Load trained model
print('Training Model')
show_data_processor = DataProcessor()
df_trained = show_data_processor.load_model()

# Predict endpoint
@app.route('/predict', methods=['GET'])
def predict():
    title = request.args.get('title')
    logging.info("Received request for title: " + title)

    # Find similar title using fuzzy matching
    titles = df_trained['title'].tolist()
    similar_title = process.extractOne(title, titles)[0]
    if similar_title != title:
        logging.info('Found "' + similar_title + '", which was the closest match\n')     

    logging.info('Finding TV shows similar to ' + similar_title + '...\n')
    
    # Get similar TV shows and their similarity scores
    top_shows, sim_scores = show_data_processor.get_similar(df_trained, similar_title)   
    top_shows = show_data_processor.predict_score(top_shows, sim_scores)
    top_shows = top_shows.set_index('index')
    top_shows['similarTitle'] = similar_title

    logging.info('Found the following TV Shows:\n')
    logging.info(top_shows.head(10))

    # Return JSON response
    prediction = top_shows.head(15).to_json(orient='records')
    return prediction

# Run the Flask app
if __name__ == '__main__':
    app.run(debug=True)
