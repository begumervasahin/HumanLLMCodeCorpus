import PyPDF2
import glob
import re
import argparse
import sys
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import linear_kernel

def load_pdf_files(path):
    """Load PDF files and extract text."""
    corpus = []
    for pdf in glob.glob(path + "/*.pdf"):
        pdf_obj = open(pdf, 'rb')
        pdf_reader = PyPDF2.PdfFileReader(pdf_obj)
        num_pages = pdf_reader.numPages
        whole_text = ''
        for i in range(num_pages):
            page_obj = pdf_reader.getPage(i)
            text = page_obj.extractText()
            text = text.replace('\n', ' ').replace('\r', ' ').replace('.', '').replace(',', '')
            text = ''.join([char for char in text if not char.isdigit()])
            text = re.sub(r'[^\w]', ' ', text)
            text = " ".join(text.split())
            text = text.lower()
            whole_text += text + ' '
        corpus.append((pdf, whole_text))
    return corpus

def calc_tfidf_matrix(corpus):
    """Calculate TF-IDF matrix."""
    vectorizer = TfidfVectorizer()
    tfidf_matrix = vectorizer.fit_transform([content for pdf, content in corpus])
    return corpus, tfidf_matrix

def find_similar(tfidf_matrix, index, top_n):
    """Find similar documents based on TF-IDF matrix."""
    cosine_similarities = linear_kernel(tfidf_matrix[index:index+1], tfidf_matrix).flatten()
    related_docs_indices = [i for i in cosine_similarities.argsort()[::-1] if i != index]
    return [(index, cosine_similarities[index]) for index in related_docs_indices][:top_n]

def check_args(args=None):
    """Parse command line arguments."""
    parser = argparse.ArgumentParser(description='Calculate Similarity Index of Documents')
    parser.add_argument('-P', '--path', help='Path to corpus', required=True)
    parser.add_argument('-T', '--top', help='Define number of top n identical documents', type=int, default=10)
    parser.add_argument('-I', '--index', help='Define corpus index', type=int, required=True)
    result = parser.parse_args(args)
    return result.path, result.top, result.index

def main():
    path, top, index = check_args(sys.argv[1:])
    corpus = load_pdf_files(path)
    tfidf_matrix = calc_tfidf_matrix(corpus)[1]
    print('The selected document is ' + corpus[index][0] + '\n')
    print('======== Similarity Index Results ========')
    for idx, score in find_similar(tfidf_matrix, index, top):
        print(score, corpus[idx][0])

if __name__ == '__main__':
    main()
