import gensim
from nltk.corpus import brown

# Load the Word2Vec model trained on Brown corpus
model = gensim.models.Word2Vec(brown.sents())

# Define the words to find similarities for
word1 = "rebellion"
word2 = "slave"

# Find top 10 similar words for each word
similarities1 = model.most_similar(positive=[word1], topn=10)
similarities2 = model.most_similar(positive=[word2], topn=10)

# Extract similar words from the tuples
similar_words1 = [p[0] for p in similarities1]
similar_words2 = [p[0] for p in similarities2]

# Calculate cosine similarities for word1 and similar words
cosine_similarities1 = ""
for word in similar_words1:
    similarity = model.similarity(word1, word)
    cosine_similarities1 += f"\n{word1} - {word}: {similarity}"

# Calculate cosine similarities for word2 and similar words
cosine_similarities2 = ""
for word in similar_words2:
    similarity = model.similarity(word2, word)
    cosine_similarities2 += f"\n{word2} - {word}: {similarity}"

# Print results
print("Top ten Similar words: \n")
print(f"{word1} = {similar_words1}")
print(f"{word2} = {similar_words2}")
print("Cosine Similarities: \n")
print(cosine_similarities1)
print(cosine_similarities2)
