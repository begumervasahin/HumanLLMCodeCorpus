import json
import random
import logging
import spacy
from sklearn.metrics import classification_report, precision_recall_fscore_support, accuracy_score
from spacy.gold import GoldParse
from spacy.scorer import Scorer

def convert_to_spacy(JSON_FilePath):
    try:
        training_data = []
        with open(JSON_FilePath, 'r', encoding='UTF-8') as f:
            for line in f:
                data = json.loads(line)
                text = data['content']
                entities = []
                for annotation in data['annotation']:
                    point = annotation['points'][0]
                    labels = annotation['label']
                    if not isinstance(labels, list):
                        labels = [labels]
                    for label in labels:
                        entities.append((point['start'], point['end'] + 1, label))
                training_data.append((text, {"entities": entities}))
        return training_data
    except Exception as e:
        logging.exception("Unable to process " + JSON_FilePath + "\n" + "error = " + str(e))
        return None

def train_spacy():
    # Load training data
    TRAIN_DATA = convert_to_spacy("training_resumes.json")
    
    # Create a blank English model
    nlp = spacy.blank('en')
    
    # Add named entity recognizer to the pipeline
    if 'ner' not in nlp.pipe_names:
        ner = nlp.create_pipe('ner')
        nlp.add_pipe(ner, last=True)
    
    # Add labels to the named entity recognizer
    for _, annotations in TRAIN_DATA:
         for ent in annotations.get('entities'):
            ner.add_label(ent[2])
    
    # Disable other pipeline components during training
    other_pipes = [pipe for pipe in nlp.pipe_names if pipe != 'ner']
    with nlp.disable_pipes(*other_pipes):
        # Begin training
        optimizer = nlp.begin_training()
        for itn in range(10):
            print("Starting iteration " + str(itn))
            random.shuffle(TRAIN_DATA)
            losses = {}
            for text, annotations in TRAIN_DATA:
                nlp.update(
                    [text],
                    [annotations],
                    drop=0.2,
                    sgd=optimizer,
                    losses=losses)
            print(losses)
    
    # Load test data
    examples = convert_to_spacy("test_resumes.json")
    
    # Initialize variables to calculate evaluation metrics
    tp = 0
    tr = 0
    tf = 0
    ta = 0
    c = 0
    
    # Iterate over test examples
    for text, annot in examples:
        with open("resume" + str(c) + ".txt", "w", encoding='UTF-8') as f:
            # Process the document
            doc_to_test = nlp(text)
            
            # Extract entities and write them to a file
            d = {}
            for ent in doc_to_test.ents:
                d[ent.label_] = []
            for ent in doc_to_test.ents:
                d[ent.label_].append(ent.text)
            for i in set(d.keys()):
                f.write("\n\n")
                f.write(i + ":" + "\n")
                for j in set(d[i]):
                    f.write(j.replace('\n', '') + "\n")
            
            # Calculate evaluation metrics
            d = {}
            for ent in doc_to_test.ents:
                d[ent.label_] = [0, 0, 0, 0, 0, 0]
            for ent in doc_to_test.ents:
                doc_gold_text = nlp.make_doc(text)
                gold = GoldParse(doc_gold_text, entities=annot.get("entities"))
                y_true = [ent.label_ if ent.label_ in x else 'Not ' + ent.label_ for x in gold.ner]
                y_pred = [x.ent_type_ if x.ent_type_ == ent.label_ else 'Not ' + ent.label_ for x in doc_to_test]
                if(d[ent.label_][0] == 0):
                    (p, r, f, s) = precision_recall_fscore_support(y_true, y_pred, average='weighted')
                    a = accuracy_score(y_true, y_pred)
                    d[ent.label_][0] = 1
                    d[ent.label_][1] += p
                    d[ent.label_][2] += r
                    d[ent.label_][3] += f
                    d[ent.label_][4] += a
                    d[ent.label_][5] += 1
        c += 1
    
    # Print evaluation metrics
    for i in d:
        print("\n For Entity " + i + "\n")
        print("Accuracy : " + str((d[i][4] / d[i][5]) * 100) + "%")
        print("Precision : " + str(d[i][1] / d[i][5]))
        print("Recall : " + str(d[i][2] / d[i][5]))
        print("F-score : " + str(d[i][3] / d[i][5]))

# Train and evaluate the spaCy model
train_spacy()
