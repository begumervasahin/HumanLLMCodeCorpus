import chars2vec
import sklearn.decomposition
import matplotlib.pyplot as plt
from sklearn.metrics.pairwise import cosine_similarity

# Load words from file
wordsarray = []
with open('words.txt') as f:
    wordsarray = [word.strip() for word in f.read().split()]

print("List of words:")
print(wordsarray)

# Load chars2vec model
c2v_model = chars2vec.load_model('eng_50')

# Vectorize words
word_embeddings = c2v_model.vectorize_words(wordsarray)

# Get word from user
word_input = input("Enter a word: ")
words_input = [word_input]

# Vectorize user input word
word_embed1 = c2v_model.vectorize_words(words_input)

# Calculate cosine similarity
similarities = cosine_similarity(word_embeddings, word_embed1)
max_similarity = max(similarities)
max_index = similarities.tolist().index(max_similarity)
result_word = wordsarray[max_index]

print("Most similar word:")
print(result_word)
