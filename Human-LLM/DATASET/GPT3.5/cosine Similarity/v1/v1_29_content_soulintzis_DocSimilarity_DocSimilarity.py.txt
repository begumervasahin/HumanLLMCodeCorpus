import string
from numpy import linalg as la
import numpy as np
import math as m
from collections import OrderedDict

def binomial_coefficient(n):
    binom = int(m.factorial(n) / (m.factorial(2)*m.factorial(n-2)))
    k = int(input("Give K (must be smaller than " + str(binom) + "): "))
    if n == 2:
        return 1
    elif k < binom:
        return k
    else:
        print("K must be smaller than " + str(binom) + ".")
        print("Default K = 1.")
        return 1

similarity = dict()
dictionary = []
file_names = []
doc_count = []

number_of_docs = int(input("How many documents do you want to insert: "))
while number_of_docs < 2:
    number_of_docs = int(input("Give number of documents again (minimum 2): "))

for x in range(number_of_docs):
    print('Give the', x+1, 'document name:')
    name = input()
    if name.endswith('.txt'):
        file_names.append(name)
    else:
        print('You must insert document with \"txt\" extension!')
        exit()

K = binomial_coefficient(number_of_docs)

for x in range(number_of_docs):
    file = "Documents/" + file_names[x]
    with open(file, 'r') as f:
        for line in f:
            for word in line.split():
                word = word.translate(str.maketrans('', '', string.punctuation)).lower()
                if word not in dictionary:
                    dictionary.append(word)

for x in range(number_of_docs):
    count = [0] * len(dictionary)
    with open("Documents/" + file_names[x], 'r') as f:
        for line in f:
            for word in line.split():
                word = word.translate(str.maketrans('', '', string.punctuation)).lower()
                position = dictionary.index(word)
                count[position] += 1
    doc_count.append(count)

for i in range(0, number_of_docs):
    for j in range(i, number_of_docs):
        if i == j:
            continue
        doc1 = doc_count[i]
        doc2 = doc_count[j]
        norm1 = la.norm(doc1, ord=2)
        norm2 = la.norm(doc2, ord=2)
        dot_product = np.dot(doc1, doc2)
        result = dot_product / (norm1 * norm2)
        pair_of_docs = file_names[i] + " and " + file_names[j]
        similarity[pair_of_docs] = result

sortedDict = OrderedDict(sorted(similarity.items(), key=lambda x: x[1], reverse=True)[:K])

print("\nTop K Similar Documents listed below:")
for key, value in sortedDict.items():
    print(f"{key} {value:.2f}, {value*100:.2f}%")
