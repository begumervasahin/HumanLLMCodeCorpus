from nltk.tokenize import word_tokenize
import gensim

def load_model(model_path):
    """
    Load the distributed word vectors model.
    
    Args:
        model_path (str): Path to the Word2Vec model.
    
    Returns:
        gensim.models.Word2Vec: Loaded Word2Vec model.
    """
    model = gensim.models.Word2Vec.load(model_path, mmap='r')
    return model

def score(seq1, seq2, model, tail=100, head=10):
    """
    Prepare inputs for scoring and calculate similarity score.
    
    Args:
        seq1 (str): First sequence.
        seq2 (str): Second sequence.
        model (gensim.models.Word2Vec): Word2Vec model.
        tail (int): Number of words to consider from the end of seq1.
        head (int): Number of words to consider from the beginning of seq2.
    
    Returns:
        float: Similarity score between the sequences.
    """
    seq1_word_list = word_tokenize(seq1.strip().lower())[-tail:]
    seq2_word_list = word_tokenize(seq2.strip().lower())[:head]
    return sim_score(seq1_word_list, seq2_word_list, model)

def sim_score(wordlist1, wordlist2, model):
    """
    Calculate average maximum similarity between two phrase inputs.
    
    Args:
        wordlist1 (list of str): List of words from the first sequence.
        wordlist2 (list of str): List of words from the second sequence.
        model (gensim.models.Word2Vec): Word2Vec model.
    
    Returns:
        float: Average maximum similarity score.
    """
    maxes = []
    for word in wordlist1:
        cur_max = 0
        for word2 in wordlist2:
            if word == word2:
                sim = 1
                cur_max = sim
            elif word in model.wv.vocab and word2 in model.wv.vocab:
                sim = model.wv.similarity(word, word2)
                if sim > cur_max:
                    cur_max = sim
        if cur_max != 0:
            maxes.append(cur_max)
    if sum(maxes) == 0:
        return 0
    return float(sum(maxes)) / len(maxes)

# Example usage
model_path = 'vectors'
loaded_model = load_model(model_path)
seq1 = "This is an example sequence."
seq2 = "Another example sentence."
similarity_score = score(seq1, seq2, loaded_model)
print("Similarity Score:", similarity_score)
