import pandas as pd
from sklearn.feature_extraction.text import CountVectorizer
from sklearn.metrics.pairwise import cosine_similarity
import movie_predict_knn

# Function to get movie title based on index
def get_title(index):
    return df[df.index == index]["title"].values[0]

# Function to get movie index based on title
def get_index(title):
    return df[df.title == title]["index"].values[0]

# Load movie dataset from CSV file
df = pd.read_csv("D:/roops_backup/projects/data mining project/movie_recommender/movie_dataset.csv")

# Fill missing values in selected features with empty strings
features = ['keywords', 'cast', 'genres', 'director']
for feature in features:
    df[feature] = df[feature].fillna('')

# Combine selected features into a single column
def combine_features(row):
    return row['keywords'] + " " + row['cast'] + " " + row['genres'] + " " + row['director']
df["combined_features"] = df.apply(combine_features, axis=1)

# Initialize CountVectorizer and create count matrix
cv = CountVectorizer()
count_matrix = cv.fit_transform(df["combined_features"])

# Calculate cosine similarity between movies
cosine_sim = cosine_similarity(count_matrix)

# Define movie of interest
movie = "Avengers: Age of Ultron"
index = get_index(movie)

# Get top similar movies using cosine similarity
similar_movies = list(enumerate(cosine_sim[index]))
sorted_similar_movies = sorted(similar_movies, key=lambda x: x[1], reverse=True)[:6]

print("Movies similar to", movie, "based on cosine similarity:")
for movie_id, similarity in sorted_similar_movies:
    if movie_id != index:
        print(get_title(movie_id))

# Use KNN for movie prediction
print("\nPredicted movies using KNN:")
K = 5
predicted_movies = movie_predict_knn.main(index, K)
for movie_id in predicted_movies:
    print(get_title(movie_id))
