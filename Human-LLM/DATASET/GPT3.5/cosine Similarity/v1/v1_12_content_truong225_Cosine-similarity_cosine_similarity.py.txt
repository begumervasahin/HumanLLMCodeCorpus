import math

def calculate_cosine(vector1=[], vector2=[]):
    """Calculate the cosine similarity between two vectors."""
    if len(vector1) != len(vector2):
        return 0
    
    dot_product = sum(vector1[i] * vector2[i] for i in range(len(vector1)))
    length1 = math.sqrt(sum(val ** 2 for val in vector1))
    length2 = math.sqrt(sum(val ** 2 for val in vector2))
    
    if length1 == 0 or length2 == 0:
        return 0
    
    return dot_product / (length1 * length2)

# Read words from dict_new.csv
with open('dict_new.csv', 'r') as f:
    word_list = [line.strip().lower() for line in f]

# Read documents from doc.csv
with open('doc.csv', 'r') as fi:
    doc_list = [line.strip().lower() for line in fi]

# Read stopwords from stopword.txt
with open('stopword.txt', 'r') as fs:
    stop_words = [line.strip().lower() for line in fs]

# Filter out stopwords from word list
filtered_word_list = [word for word in word_list if word not in stop_words]

# Calculate cosine similarity for each document
with open('cos.csv', 'w') as output:
    for document in doc_list:
        vector = [1 if word in document else 0 for word in filtered_word_list]
        similarity = calculate_cosine(vector, [1] * len(filtered_word_list))
        output.write(f'{similarity}\n')
        print(document)  # Print the document for visualization
