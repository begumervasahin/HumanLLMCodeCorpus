class Mapping:
    def __init__(self, items):
        self.item_list = items
        self.items = {index: item for index, item in enumerate(items)}
        self.index = {item: index for index, item in enumerate(items)}

    def __contains__(self, item):
        return item in self.index

    def __getitem__(self, item):
        return self.index[item]

    def __iter__(self):
        return iter(self.index)

    def __len__(self):
        return len(self.index)

    def __repr__(self):
        return f"<{self.__class__.__name__} with {len(self)} entries>"

    def add(self, item):
        if item not in self:
            index = len(self)
            self.items[index] = item
            self.index[item] = index
        return item

    def get(self, item, default=None):
        return self.index.get(item, default)

    def item(self, index):
        return self.items[index]

    def keys(self):
        """Preserving original order"""
        return self.item_list

# Example usage
items = ['apple', 'banana', 'orange']
mapping = Mapping(items)

print(mapping)  # Output: <Mapping with 3 entries>
print('apple' in mapping)  # Output: True
print(mapping['banana'])  # Output: 1
print(len(mapping))  # Output: 3

mapping.add('grape')
print(mapping)  # Output: <Mapping with 4 entries>

print(mapping.get('orange'))  # Output: 2

print(mapping.item(3))  # Output: grape

print(mapping.keys())  # Output: ['apple', 'banana', 'orange', 'grape']
