import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load the data from file.tsv and Movie_Id_Titles.csv
column_names = ['user_id', 'item_id', 'rating', 'timestamp']
df = pd.read_csv('file.tsv', sep='\t', names=column_names)
movie_titles = pd.read_csv('Movie_Id_Titles.csv')

# Merge the dataframes on 'item_id'
data = pd.merge(df, movie_titles, on='item_id')

# Compute mean ratings and count of ratings for each movie
ratings = pd.DataFrame(data.groupby('title')['rating'].mean())
ratings['num of ratings'] = data.groupby('title')['rating'].count()

# Visualize rating count distribution
plt.figure(figsize=(10, 4))
ratings['num of ratings'].hist(bins=70)
plt.xlabel('Number of Ratings')
plt.ylabel('Frequency')
plt.title('Distribution of Ratings Count')
plt.show()

# Visualize rating distribution
plt.figure(figsize=(10, 4))
ratings['rating'].hist(bins=70)
plt.xlabel('Rating')
plt.ylabel('Frequency')
plt.title('Distribution of Ratings')
plt.show()

# Create a pivot table with user_id as index and movie titles as columns
moviemat = data.pivot_table(index='user_id', columns='title', values='rating')

# Compute correlation with Star Wars (1977) and Liar Liar (1997)
starwars_user_ratings = moviemat['Star Wars (1977)']
liarliar_user_ratings = moviemat['Liar Liar (1997)']
similar_to_starwars = moviemat.corrwith(starwars_user_ratings)
similar_to_liarliar = moviemat.corrwith(liarliar_user_ratings)

# Create correlation dataframes and filter movies with more than 100 ratings
corr_starwars = pd.DataFrame(similar_to_starwars, columns=['Correlation'])
corr_starwars.dropna(inplace=True)
corr_starwars = corr_starwars.join(ratings['num of ratings'])
corr_starwars_filtered = corr_starwars[corr_starwars['num of ratings'] > 100].sort_values('Correlation', ascending=False)

corr_liarliar = pd.DataFrame(similar_to_liarliar, columns=['Correlation'])
corr_liarliar.dropna(inplace=True)
corr_liarliar = corr_liarliar.join(ratings['num of ratings'])
corr_liarliar_filtered = corr_liarliar[corr_liarliar['num of ratings'] > 100].sort_values('Correlation', ascending=False)

# Print the recommended movies
print('Movies recommended that are similar to Star Wars (1977):')
print(corr_starwars_filtered.head())
print()
print('Movies recommended that are similar to Liar Liar (1997):')
print(corr_liarliar_filtered.head())
