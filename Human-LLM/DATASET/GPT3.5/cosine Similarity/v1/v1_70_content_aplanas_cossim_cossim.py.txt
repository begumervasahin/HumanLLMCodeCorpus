import os
import argparse
import itertools
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

def main(good_dir, bad_dir, sample_file):
    good_files = [os.path.join(good_dir, f) for f in os.listdir(good_dir)
                  if os.path.isfile(os.path.join(good_dir, f))]
    bad_files = [os.path.join(bad_dir, f) for f in os.listdir(bad_dir)
                 if os.path.isfile(os.path.join(bad_dir, f))]
    
    all_files = good_files + bad_files + [sample_file]
    
    tfidf_good = TfidfVectorizer().fit_transform(
        [open(f, errors='ignore').read() for f in good_files])
    tfidf_bad = TfidfVectorizer().fit_transform(
        [open(f, errors='ignore').read() for f in bad_files])
    tfidf_sample = TfidfVectorizer().fit_transform(
        [open(sample_file, errors='ignore').read()])
    
    sim_good = cosine_similarity(tfidf_sample, tfidf_good)
    sim_bad = cosine_similarity(tfidf_sample, tfidf_bad)
    
    index_good = sim_good.argsort()[0][-2]
    index_bad = sim_bad.argsort()[0][-2]
    
    print('Maximum similarity with good samples:', sim_good[0][index_good])
    print('Maximum similarity with bad samples:', sim_bad[0][index_bad])
    
    similar_document = good_files[index_good] if sim_good[0][index_good] >= sim_bad[0][index_bad] else bad_files[index_bad]
    print('Similar document:', similar_document)

if __name__ == '__main__':
    parser = argparse.ArgumentParser(description='Basic cosine similarity comparator.')
    parser.add_argument('--good', help='directory where the good samples are')
    parser.add_argument('--bad', help='directory where the bad samples are')
    parser.add_argument('sample', help='file to compare with')
    args = parser.parse_args()
    
    if not args.good or not args.bad:
        print('Provide directory paths for good and bad samples')
        exit(1)
    
    main(args.good, args.bad, args.sample)
