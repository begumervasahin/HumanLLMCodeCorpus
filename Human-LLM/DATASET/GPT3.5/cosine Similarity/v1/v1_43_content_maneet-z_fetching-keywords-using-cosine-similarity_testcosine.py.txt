import math
import os

def get_file_data(f_name):
    if not os.path.exists(f_name):
        print('File is not present in the directory')
        return None
    
    with open(f_name, 'r') as file:
        data = file.read()
    
    str_data = data.replace('?', '.').replace('!', '.')
    split_data_fullstop = str_data.split('.')
    return split_data_fullstop

def get_test_words(test_word_file):
    if not os.path.exists(test_word_file):
        print('File is not present in the directory')
        return None
    
    with open(test_word_file, 'r') as file:
        test_words = [line.rstrip() for line in file]
    
    if len(test_words) > 1:
        print('Please check the file you have uploaded')
        return None
    
    return test_words

def get_common_words(common_word_file):
    common_words = {}
    with open(common_word_file, 'r') as file:
        for line in file:
            common_words[line.rstrip()] = 0
    return common_words

def get_word_frequency(string_data_clean, test_words, common_words):
    word_count = {}
    for test_word in test_words:
        word_count[test_word] = {}
        for line in string_data_clean.split('.'):
            if test_word in line:
                words_in_line = line.split()
                for word in words_in_line:
                    if word not in common_words and word != test_word:
                        word_count[test_word][word] = word_count[test_word].get(word, 0) + 1
    return word_count

def clean_data(split_data_fullstop):
    punctuation = [',', '\'', '\"', ':', ';', '(', ')', '[', ']', '\n', '_']
    temporary_string = ''
    for line in split_data_fullstop:
        for char in line:
            if char == '-':
                if line[line.index(char) - 1] == '-' or line[line.index(char) + 1] == '-':
                    temporary_string += ' '
                else:
                    temporary_string += char
            elif char in punctuation:
                temporary_string += ' '
            else:
                temporary_string += char
        temporary_string += '.'
    temporary_string = ' '.join(temporary_string.split())  # Remove extra spaces
    return temporary_string

def profile_calculation(profile_count, test_words):
    final_data_profile = []
    for test_word in test_words[1:]:
        pq = 0
        count_denom_p = sum(value ** 2 for value in profile_count[test_words[0]].values())
        count_denom_q = sum(value ** 2 for value in profile_count[test_word].values())
        for word in profile_count[test_words[0]]:
            if word in profile_count[test_word]:
                pq += profile_count[test_words[0]][word] * profile_count[test_word][word]
        final_denom = math.sqrt(count_denom_p * count_denom_q)
        similarity = round(pq / final_denom, 3) if final_denom != 0 else 0
        final_data_profile.append([test_word, similarity])
    return sorted(final_data_profile, key=lambda l: l[1], reverse=True)

def main(f_name, test_word_file, common_word_file=None):
    if f_name == '' or test_word_file == '':
        print('Please insert the File to read data from ')
        return
    
    split_data_fullstop = get_file_data(f_name)
    if split_data_fullstop is None:
        print('File cannot be processed because of the incorrect file upload')
        return
    
    test_words = get_test_words(test_word_file)
    if test_words is None:
        print('File cannot be processed because of the incorrect file upload')
        return
    
    string_data_clean = clean_data(split_data_fullstop)
    common_words = get_common_words(common_word_file) if common_word_file else {}
    profile_count = get_word_frequency(string_data_clean, test_words, common_words)
    final_data = profile_calculation(profile_count, test_words)
    
    for data in final_data:
        print(data)

if __name__ == "__main__":
    main('TCGA-XC-AA0X.B22BD117-8DF0-47FF-BE25-9FE2DA916DB6.txt', 'profile.txt', 'common.txt')
