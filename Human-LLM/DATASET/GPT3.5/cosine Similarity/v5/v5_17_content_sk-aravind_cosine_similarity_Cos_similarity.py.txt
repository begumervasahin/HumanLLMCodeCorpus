from collections import Counter
import math

# Define the threshold for similarity
THRESHOLD = 0.4

def cosine_similarity(c1, c2):
    """
    Calculate cosine similarity between two Counters.

    Args:
        c1 (Counter): First Counter object.
        c2 (Counter): Second Counter object.

    Returns:
        float: Cosine similarity between c1 and c2.
    """
    terms = set(c1).union(c2)
    dot_product = sum(c1.get(k, 0) * c2.get(k, 0) for k in terms)
    mag_a = math.sqrt(sum(c1.get(k, 0) ** 2 for k in terms))
    mag_b = math.sqrt(sum(c2.get(k, 0) ** 2 for k in terms))
    return dot_product / (mag_a * mag_b)

def length_similarity(c1, c2):
    """
    Calculate length similarity between two Counters.

    Args:
        c1 (Counter): First Counter object.
        c2 (Counter): Second Counter object.

    Returns:
        float: Length similarity between c1 and c2.
    """
    len_c1 = sum(c1.values())
    len_c2 = sum(c2.values())
    return min(len_c1, len_c2) / max(len_c1, len_c2)

def calculate_similarity_score(words1, words2):
    """
    Calculate the similarity score between two lists of words.

    Args:
        words1 (list): First list of words.
        words2 (list): Second list of words.

    Returns:
        float: Similarity score between words1 and words2.
    """
    counter1, counter2 = Counter(words1), Counter(words2)
    score = length_similarity(counter1, counter2) * cosine_similarity(counter1, counter2)
    return score

def main():
    # Define the input model and current data
    input_model = "algorithms node current shortest path lol"
    input_current = "shortest node current"

    # Split the input strings into lists of words
    model_words = input_model.split()
    current_words = input_current.split()

    # Calculate and print the similarity score
    similarity_score = calculate_similarity_score(model_words, current_words)
    print("Similarity Score:", similarity_score)

    # Check if the similarity score is above the threshold
    if similarity_score > THRESHOLD:
        print("Still doing work")
    else:
        print("Do more work")

if __name__ == "__main__":
    main()
