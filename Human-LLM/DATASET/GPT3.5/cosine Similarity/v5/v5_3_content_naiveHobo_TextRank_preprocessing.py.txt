import re
import string
import unicodedata
from nltk import pos_tag
from nltk.corpus import wordnet
from nltk.corpus import stopwords
from nltk.stem.snowball import SnowballStemmer
from nltk.stem import WordNetLemmatizer
from syntactic_unit import SyntacticUnit

class TextProcessor:
    """
    A class for preprocessing text data to prepare for keyword extraction
    """

    def __init__(self):
        self.stopwords = self.load_stopwords(path="../stopwords.txt")
        self.lemmatizer = WordNetLemmatizer()
        self.stemmer = SnowballStemmer("english")
        self.punctuation_pattern = re.compile('([%s])+' % re.escape(string.punctuation), re.UNICODE)
        self.numeric_pattern = re.compile(r"[0-9]+", re.UNICODE)
        self.alphabetic_pattern = re.compile('(((?![\d])\w)+)', re.UNICODE)

    def remove_punctuation(self, text):
        """Remove punctuation from text"""
        return self.punctuation_pattern.sub(" ", text)

    def remove_numeric(self, text):
        """Remove numeric characters from text"""
        return self.numeric_pattern.sub("", text)

    def remove_stopwords(self, tokens):
        """Remove stopwords from tokens"""
        return [word for word in tokens if word not in self.stopwords]

    def stem_tokens(self, tokens):
        """Stem tokens using Snowball Stemmer"""
        return [self.stemmer.stem(word) for word in tokens]

    def lemmatize_tokens(self, tokens):
        """Lemmatize tokens using WordNet Lemmatizer"""
        if not tokens:
            return []
        if isinstance(tokens[0], str):
            pos_tags = pos_tag(tokens)
        else:
            pos_tags = tokens
        lemmatized_tokens = [
            self.lemmatizer.lemmatize(word[0]) if not self.get_wordnet_pos(word[1])     
            else self.lemmatizer.lemmatize(word[0], pos=self.get_wordnet_pos(word[1]))  
            for word in pos_tags
        ]
        return lemmatized_tokens

    def part_of_speech_tag(self, tokens):
        """Perform Part-of-Speech tagging"""
        if isinstance(tokens, str):
            tokens = self.tokenize(tokens)
        return pos_tag(tokens)

    @staticmethod
    def load_stopwords(path="stopwords.txt"):
        """Load stopwords from file"""
        return list(set(stopwords.words('english')))

    @staticmethod
    def get_wordnet_pos(treebank_tag):
        """Map treebank tags to WordNet part of speech names"""
        if treebank_tag.startswith('J'):
            return wordnet.ADJ
        elif treebank_tag.startswith('V'):
            return wordnet.VERB
        elif treebank_tag.startswith('N'):
            return wordnet.NOUN
        elif treebank_tag.startswith('R'):
            return wordnet.ADV
        else:
            return None

    @staticmethod
    def deaccent(text):
        """Remove accentuation from text"""
        norm = unicodedata.normalize("NFD", text)
        result = "".join(ch for ch in norm if unicodedata.category(ch) != 'Mn')
        return unicodedata.normalize("NFC", result)

    def clean_text(self, text, filters=None, stem=False):
        """
        Tokenize, filter, and lemmatize text
        Returns a dict of word -> SyntacticUnit
        """
        text = text.lower()
        text = self.deaccent(text)
        text = self.remove_numeric(text)
        text = self.remove_punctuation(text)
        original_words = [match.group() for match in self.alphabetic_pattern.finditer(text)]
        filtered_words = self.remove_stopwords(original_words)
        pos_tags = pos_tag(filtered_words)
        if stem:
            filtered_words = self.stem_tokens(filtered_words)
        else:
            filtered_words = self.lemmatize_tokens(pos_tags)
        units = [
            SyntacticUnit(word, word, tag)
            for i, (word, tag) in enumerate(pos_tags)
            if tag.startswith('N') and len(filtered_words[i]) >= 3
        ]
        return {unit.text: unit for unit in units}

    def tokenize(self, text):
        """Tokenize text"""
        text = text.lower()
        text = self.deaccent(text)
        return [match.group() for match in self.alphabetic_pattern.finditer(text)]      

    def clean_sentence(self, text):
        """Clean sentence for word2vec training"""
        text = text.lower()
        text = self.deaccent(text)
        text = self.remove_numeric(text)
        text = self.remove_punctuation(text)
        return text
