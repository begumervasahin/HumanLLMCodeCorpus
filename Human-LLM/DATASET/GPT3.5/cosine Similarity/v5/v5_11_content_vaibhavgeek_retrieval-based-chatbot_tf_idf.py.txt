from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity
import numpy as np
import pickle
import json
import random

def talk_to_chatbot(test_sentence, min_similarity, json_file, vectorizer_path, matrix_path):
    try:
        with open(vectorizer_path, 'rb') as f:
            vectorizer = pickle.load(f)
        with open(matrix_path, 'rb') as f:
            matrix_train = pickle.load(f)
    except FileNotFoundError:
        vectorizer, matrix_train = train_chatbot(json_file, vectorizer_path, matrix_path)

    matrix_test = vectorizer.transform([test_sentence])
    cosine_sim = cosine_similarity(matrix_test, matrix_train)
    cosine_sim = np.delete(cosine_sim, 0)  # Remove self-similarity
    max_sim = cosine_sim.max()

    if max_sim > min_similarity:
        new_max = max_sim - 0.01
        indices = np.where(cosine_sim > new_max)[0]
        response_index = random.choice(indices)
    else:
        return "live_chat", 0

    with open(json_file, "r") as sentences_file:
        sentences = json.load(sentences_file)
        for idx, row in enumerate(sentences):
            if idx == response_index:
                return row["response"], max_sim

def train_chatbot(json_file, vectorizer_path, matrix_path):
    sentences = ["No you.", "No you."]
    with open(json_file, "r") as sentences_file:
        sentences.extend(row["message"] for row in json.load(sentences_file))

    vectorizer = TfidfVectorizer()
    matrix_train = vectorizer.fit_transform(sentences)

    with open(vectorizer_path, 'wb') as f:
        pickle.dump(vectorizer, f)
    with open(matrix_path, 'wb') as f:
        pickle.dump(matrix_train, f)

    return vectorizer, matrix_train

def previous_chats(query):
    min_similarity = 0.7
    json_file = "data/previous_chats.json"
    vectorizer_path = "data/previous_vectorizer.pickle"
    matrix_path = "data/previous_matrix.pickle"

    query_response, similarity = talk_to_chatbot(query, min_similarity, json_file, vectorizer_path, matrix_path)
    return query_response

while True:
    user_input = input("User: ")
    print("Bot:", previous_chats(user_input))
