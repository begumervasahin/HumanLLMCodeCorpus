import re
import string
import nltk
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from nltk.stem.porter import PorterStemmer
from sklearn.feature_extraction.text import CountVectorizer
from sklearn.metrics.pairwise import cosine_similarity

# Sample data
cnn_scroll = ["Sample CNN article text 1.", "Sample CNN article text 2.", "Sample CNN article text 3."]
fox_scroll = ["Sample Fox article text 1.", "Sample Fox article text 2.", "Sample Fox article text 3."]

# Preprocess function
def preprocess(text):
    # Convert to lowercase
    text = text.lower()
    # Remove punctuation
    text = re.sub(r'\W+', ' ', text)
    # Tokenize words
    words = word_tokenize(text)
    # Remove stopwords
    stop_words = set(stopwords.words('english'))
    words = [word for word in words if word not in stop_words]
    # Stemming
    porter = PorterStemmer()
    words = [porter.stem(word) for word in words]
    return words

# Preprocess CNN scroll
cnn_scroll_words = [preprocess(text) for text in cnn_scroll]

# Preprocess Fox scroll
fox_scroll_words = [preprocess(text) for text in fox_scroll]

# Vectorization
vectorizer = CountVectorizer(analyzer='word', min_df=1, lowercase=False)
cnn_vectorized = vectorizer.fit_transform(cnn_scroll_words)
fox_vectorized = vectorizer.fit_transform(fox_scroll_words)

# Calculate cosine similarity
cosine_sim_score = cosine_similarity(cnn_vectorized, fox_vectorized)
mean_cosine_sim_score = cosine_sim_score.mean()

print("Mean Cosine Similarity Score:", mean_cosine_sim_score)
