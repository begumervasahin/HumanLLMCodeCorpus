# Load word vectors
words, word_to_vec_map = read_glove_vecs('data/glove.6B.50d.txt')

# Function to compute cosine similarity between two vectors
def cosine_similarity(u, v):
    dot_product = np.dot(u, v)
    norm_u = np.linalg.norm(u)
    norm_v = np.linalg.norm(v)
    similarity = dot_product / (norm_u * norm_v)
    return similarity

# Function to complete analogy
def complete_analogy(word_a, word_b, word_c, word_to_vec_map):
    word_a, word_b, word_c = word_a.lower(), word_b.lower(), word_c.lower()
    e_a, e_b, e_c = word_to_vec_map[word_a], word_to_vec_map[word_b], word_to_vec_map[word_c]
    max_cosine_sim = -np.inf
    best_word = None
    for word, vec in word_to_vec_map.items():
        if word in [word_a, word_b, word_c]:
            continue
        cosine_sim = cosine_similarity(e_b - e_a, vec - e_c)
        if cosine_sim > max_cosine_sim:
            max_cosine_sim = cosine_sim
            best_word = word
    return best_word

# Analogy triads to try
triads_to_try = [('italy', 'italian', 'spain'), ('india', 'delhi', 'japan'), ('man', 'woman', 'boy'), ('small', 'smaller', 'large')]
for triad in triads_to_try:
    print('{} -> {} :: {} -> {}'.format(*triad, complete_analogy(*triad, word_to_vec_map)))

# Neutralize bias in word vectors
def neutralize(word, gender_bias_axis, word_to_vec_map):
    e = word_to_vec_map[word]
    e_biascomponent = np.dot(e, gender_bias_axis) / np.linalg.norm(gender_bias_axis) ** 2 * gender_bias_axis
    e_debiased = e - e_biascomponent
    return e_debiased

receptionist_vec = "receptionist"
print("cosine similarity between receptionist and gender bias axis before neutralizing:", cosine_similarity(word_to_vec_map[receptionist_vec], gender_bias_axis))
receptionist_debiased = neutralize(receptionist_vec, gender_bias_axis, word_to_vec_map)
print("cosine similarity between receptionist and gender bias axis after neutralizing:", cosine_similarity(receptionist_debiased, gender_bias_axis))

# Equalize gender bias in word pairs
def equalize(pair, gender_bias_axis, word_to_vec_map):
    w1, w2 = pair
    e_w1, e_w2 = word_to_vec_map[w1], word_to_vec_map[w2]
    mu = (e_w1 + e_w2) / 2
    mu_B = np.dot(mu, gender_bias_axis) / np.linalg.norm(gender_bias_axis) ** 2 * gender_bias_axis 
    mu_orth = mu - mu_B
    e_w1B = np.dot(e_w1, gender_bias_axis) / np.linalg.norm(gender_bias_axis) ** 2 * gender_bias_axis
    e_w2B = np.dot(e_w2, gender_bias_axis) / np.linalg.norm(gender_bias_axis) ** 2 * gender_bias_axis
    corrected_e_w1B = np.sqrt(np.abs(1 - np.sum(mu_orth ** 2))) * (e_w1B - mu_B) / np.abs(e_w1 - mu_orth - mu_B)
    corrected_e_w2B = np.sqrt(np.abs(1 - np.sum(mu_orth ** 2))) * (e_w2B - mu_B) / np.abs(e_w2 - mu_orth - mu_B)
    e1 = mu_orth + corrected_e_w1B
    e2 = mu_orth + corrected_e_w2B
    return e1, e2

print("cosine similarities before equalizing:")

print("cosine_similarity(word_to_vec_map[\"woman\"], gender_bias_axis) =", cosine_similarity(word_to_vec_map["woman"], gender_bias_axis))
