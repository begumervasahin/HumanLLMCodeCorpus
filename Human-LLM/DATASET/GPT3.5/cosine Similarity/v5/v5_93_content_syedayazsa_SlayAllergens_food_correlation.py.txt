import pandas as pd
from pymongo import MongoClient

# Connect to MongoDB
client = MongoClient()
db = client.project
collection_filefood = db.filefood
collection_newingredient = db.newingredient

# Load data from MongoDB collections
df_filefood = pd.DataFrame(list(collection_filefood.find()))
df_newingredient = pd.DataFrame(list(collection_newingredient.find()))

# Clean data and rename columns
df_filefood.drop(columns='_id', inplace=True)
df_newingredient.drop(columns='_id', inplace=True)
df_filefood.columns = ['rating', 'item_id', 'user_id']

# Merge dataframes on 'item_id'
merged_data = pd.merge(df_filefood, df_newingredient, on='item_id')

# Convert data types to numeric
merged_data[['rating', 'item_id', 'user_id']] = merged_data[['rating', 'item_id', 'user_id']].apply(pd.to_numeric)

# Calculate mean rating and count of ratings for each title
ratings = merged_data.groupby('title')['rating'].agg(['mean', 'count'])
ratings.columns = ['mean_rating', 'num_ratings']

# Sort titles by number of ratings
top_rated_items = ratings.sort_values('num_ratings', ascending=False).head(10)

# Create a pivot table for collaborative filtering
pivot_table = merged_data.pivot_table(index='user_id', columns='title', values='rating')

# Get input from user for the food item
item_name = input("Enter the name of the food item: \n")

# Filter pivot table for the selected item
item_ratings = pivot_table[item_name]

# Calculate correlation with other items
similar_items = pivot_table.corrwith(item_ratings)

# Create dataframe for item correlations
corr_df = pd.DataFrame(similar_items, columns=['Correlation'])
corr_df.dropna(inplace=True)

# Merge with ratings dataframe and filter items with more than 100 ratings
corr_df = corr_df.join(ratings['num_ratings'])
popular_corr_items = corr_df[corr_df['num_ratings'] > 100]

# Sort items by correlation and print top recommendations
sorted_recommendations = popular_corr_items.sort_values('Correlation', ascending=False).head()
print(sorted_recommendations.iloc[:, 0].head(25))
