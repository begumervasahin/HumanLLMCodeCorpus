from math import sqrt, factorial
from numpy import dot
from os import scandir
from os.path import isfile, join
from sklearn.feature_extraction.text import CountVectorizer

def cosine_similarity(x, y):
    dot_product = dot(x, y.T)[0, 0]
    magnitude_x = dot(x, x.T)[0, 0]
    magnitude_y = dot(y, y.T)[0, 0]
    return dot_product / sqrt(magnitude_x * magnitude_y)

def read_documents(directory_path):
    file_paths = [f.path for f in scandir(directory_path) if isfile(join(directory_path, f))]
    if len(file_paths) == 0:
        print("Empty directory")
        return []
    documents = []
    for file_path in file_paths:
        with open(file_path, 'r') as file:
            documents.append(file.read())
    return documents

def main():
    directory_path = input("Enter the directory path containing your text files: ")
    documents = read_documents(directory_path)
    if not documents:
        print("No valid documents found in the directory. Exiting.")
        return

    num_documents = len(documents)
    count_vectorizer = CountVectorizer()
    count_matrix = count_vectorizer.fit_transform(documents)

    top_k = int(input("Enter the number of most similar texts you want to print: "))    

    max_possible_pairs = factorial(num_documents) // (factorial(2) * factorial(num_documents - 2))

    similarities = []
    for i in range(num_documents):
        for j in range(i + 1, num_documents):
            similarity = cosine_similarity(count_matrix[i], count_matrix[j])
            similarities.append((similarity, i, j))

    similarities.sort(key=lambda x: x[0], reverse=True)

    if top_k > max_possible_pairs:
        print("Cannot print more than the maximum possible pairs.")
        return

    print("Top", top_k, "most similar documents:")
    for i in range(min(top_k, len(similarities))):
        similarity, doc1_idx, doc2_idx = similarities[i]
        print(f"The top {i+1} most similar documents are: Documents {doc1_idx+1} and {doc2_idx+1}")

if __name__ == "__main__":
    main()
