from app import bot_response
from constants import raw_response_data

def create_response(text, quick_replies=None, buttons=None):
    """
    Create a response object with text, quick replies, and buttons.
    """
    response = {
        'text': text,
        'quick_replies': quick_replies,
        'buttons': buttons
    }
    return response

class TestBotResponses:
    """
    Test class for bot responses.
    """
    def test_intro_message_response(self):
        """
        Test method for the intro message response.
        """
        # Call the bot_response function with the input "Hello"
        reply = bot_response("Hello")

        # Create the expected response object
        expected_response = create_response(
            "Hello! I’m Sloan, your automated guide for advice on your student loans brought to you by TISLA. This is not legal advice but simple guidance to help you manage your student loan debt. Send 'restart' at any time to restart. Ready? Simply type your question and I’ll try and guide you to the best way to manage your student debt.",
            raw_response_data['replies']['intro']
        )

        # Assert that the returned response matches the expected response object
        assert reply.get('text') == expected_response.get('text')
        assert reply.get('buttons') == expected_response.get('buttons')
        assert reply.get('quick_replies') == expected_response.get('quick_replies')
