from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity
import scipy.spatial as sp

# Display import status
print("Import Successful")
print("Threshold for merging set at 0.5")
print("------------------------")

# Threshold for merging sentences
thresh_merge = 0.5

def split_comma_separated(sentence):
    """Split comma-separated sentence into a list of frames."""
    frames = sentence.split(',')
    return [frame.strip() for frame in frames]

def remove_newline(frame):
    """Remove newline character from a frame."""
    return frame.strip('\n')

def create_corpus(first_frames, second_frames):
    """Create a corpus from two lists of frames."""
    return first_frames + second_frames

def sentence_compactor(cosine_sim_matrix, first_sentence, second_sentence):
    """Compact similar sentences into one."""
    mapping = {}
    index = cosine_sim_matrix.argmax(axis=1)
    for i, idx in enumerate(index):
        if cosine_sim_matrix[i][idx] >= thresh_merge:
            mapping[first_sentence[i]] = second_sentence[idx]
        else:
            mapping[first_sentence[i]] = ''
            mapping[second_sentence[idx]] = ''
    for i, frame in enumerate(second_sentence):
        if i not in index:
            mapping[frame] = ''
    return mapping

def sentence_fusion(mapping):
    """Select the longest sentence from the mapping."""
    return [u if len(u) >= len(v) else v for u, v in mapping.items()]

def main_util(sentence1, sentence2):
    """Process input sentences and generate compacted sentences."""
    first_sentence = split_comma_separated(sentence1)
    second_sentence = split_comma_separated(sentence2)
    last_frame = remove_newline(first_sentence[-1])
    first_sentence[-1] = last_frame
    first_frames = first_sentence
    second_frames = second_sentence
    corpus = create_corpus(first_frames, second_frames)
    vectorizer = TfidfVectorizer()
    X = vectorizer.fit_transform(corpus).toarray()
    first_sen_vec = X[:len(first_frames)]
    second_sen_vec = X[len(first_frames):]
    cosine_sim_matrix = 1 - sp.distance.cdist(first_sen_vec, second_sen_vec, 'cosine')  
    mapping = sentence_compactor(cosine_sim_matrix, first_sentence, second_sentence)    
    print(sentence_fusion(mapping))
    print("------------------------")

def main():
    """Process input file and generate compacted sentences for each pair."""
    print("Please Enter the filename")
    filename = input()
    with open(filename, 'r') as file_handler:
        sentences = file_handler.readlines()
    num_of_pairs = len(sentences) // 2
    counter = 0
    for _ in range(num_of_pairs):
        i = counter
        j = i + 1
        print("Output for pair starting at line number ", counter + 1)
        main_util(sentences[i], sentences[j])
        counter += 2

# Check if the script is being run as the main program
if __name__ == "__main__":
    main()
