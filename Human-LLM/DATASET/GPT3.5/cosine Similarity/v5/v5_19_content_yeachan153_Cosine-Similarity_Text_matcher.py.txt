from collections import Counter
import json
import math
import os

# Function to split text into lowercase words
def split_text(text):
    return text.lower().split()

# Function to calculate word frequencies in a file
def get_file_freqs(filename):
    freqs = Counter()
    with open(filename, 'r', encoding='utf8') as file:
        for line in file:
            words = split_text(line)
            freqs.update(words)
    return freqs

# Load corpus data from JSON file
def load_corpus_from_json(corpus_path):
    with open(os.path.join(corpus_path, 'corpus/corpus-subset.json'), 'r') as file:     
        return json.load(file)

# Convert word frequencies to a vector representation
def freqs_to_vector(freqs, vocabulary):
    vector = [freqs[word] if word in freqs else 0 for word in vocabulary]
    return vector

# Calculate cosine similarity between two vectors
def cosine_similarity(query_vector, doc_vector):
    dot_product = sum(a * b for a, b in zip(query_vector, doc_vector))
    mag_query = math.sqrt(sum(a**2 for a in query_vector))
    mag_doc = math.sqrt(sum(b**2 for b in doc_vector))
    return dot_product / (mag_query * mag_doc) if mag_query * mag_doc != 0 else 0

# Calculate the Euclidean norm of a vector
def norm(vector):
    return math.sqrt(sum(number**2 for number in vector))

# Calculate similarity between two vectors
def calculate_similarity(query_vector, doc_vector):
    if len(query_vector) != len(doc_vector):
        print("Lengths of query vector and document vector don't match")
        return
    return sum(a * b for a, b in zip(query_vector, doc_vector)) / (norm(query_vector) * norm(doc_vector))

# Load and preprocess corpus data
def preprocess_corpus(corpus_path, voc_size=100):
    os.chdir(corpus_path)
    corpus = load_corpus_from_json(corpus_path)

    corpus_freqs = Counter()
    for doc_id, info in corpus.items():
        filename = info['filename']
        corpus_freqs.update(get_file_freqs(filename))

    vocabulary = [word for word, _ in corpus_freqs.most_common(voc_size)]

    for doc_id, info in corpus.items():
        filename = info['filename']
        info['freq_vect'] = freqs_to_vector(get_file_freqs(filename), vocabulary)

    return corpus, vocabulary
