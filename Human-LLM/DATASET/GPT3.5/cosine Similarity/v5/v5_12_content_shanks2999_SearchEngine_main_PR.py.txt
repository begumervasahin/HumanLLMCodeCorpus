import requests
from bs4 import BeautifulSoup
from collections import OrderedDict
from operator import itemgetter
import preprocess

# Global variables
my_list = []
link_set = set()
removed_links = set()
web_graph = OrderedDict()

def calculate_page_rank(web_graph_new):
    out_link_count = {}
    N = len(web_graph_new)
    B = 0.15 * (1 / N)
    new_scores = {node: 1/N for node in web_graph_new}
    old_scores = new_scores.copy()

    # Count outgoing links for each node
    for node in web_graph_new:
        for connected_node in web_graph_new[node]:
            out_link_count[connected_node] = out_link_count.get(connected_node, 0) + 1  

    # Perform PageRank iterations
    for _ in range(10):
        for node in web_graph_new:
            summation = sum(old_scores[connected_node] / out_link_count[connected_node] for connected_node in web_graph_new[node])
            new_scores[node] = (0.85 * summation) + B
        old_scores = new_scores.copy()

    # Sort and return page ranks
    return OrderedDict(sorted(old_scores.items(), key=itemgetter(1), reverse=True))

def populate_web_graph(threshold):
    index = 0
    count = 1

    while count <= threshold:
        try:
            page = requests.get(my_list[index])
            if page.status_code != 200:
                if my_list[index] in web_graph:
                    del web_graph[my_list[index]]
                    removed_links.add(my_list[index])
                index += 1
                continue
            soup = BeautifulSoup(page.text, 'html.parser')
        except:
            index += 1
            continue

        current_links = get_all_links(my_list[index], soup)

        for link in current_links:
            if link not in removed_links:
                if link not in web_graph:
                    web_graph[link] = set()
                web_graph[link].add(my_list[index])

            if link[link.find("//")+2:] not in link_set:
                link_set.add(link[link.find("//") + 2:])
                my_list.append(link)

        index += 1
        if count % 10 == 0:
            print(count)
        count += 1

def get_all_links(link, soup):
    links_list = []

    for a_tag in soup.find_all('a', href=True):
        url = a_tag['href']

        if is_valid_link(url):
            final_url = build_absolute_url(link, url)
            if final_url:
                links_list.append(final_url)

    return links_list

def is_valid_link(url):
    excluded_extensions = ('.pdf', '.doc', '.docx', '.xls', '.xlsx', '.jpg', '.png')
    return not (url.endswith(excluded_extensions) or url.rfind(":") > 6 or "@" in url)

def build_absolute_url(base_link, url):
    if url.startswith('http'):
        return url
    elif url.startswith('/'):
        return base_link + url
    else:
        return base_link + "/" + url

def main():
    initial_link = "https://www.cs.uic.edu"
    link_set.add(initial_link[initial_link.find("//")+2:])
    my_list.append(initial_link)

    populate_web_graph(3000)

    preprocess.export_web_graph(web_graph)
    web_graph_new = preprocess.import_web_graph()

    page_rank = calculate_page_rank(web_graph_new)
    preprocess.export_page_rank(page_rank)

    print("PageRank calculation completed.")

if __name__ == '__main__':
    main()
