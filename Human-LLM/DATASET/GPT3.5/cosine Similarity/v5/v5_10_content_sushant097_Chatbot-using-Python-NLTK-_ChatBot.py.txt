import nltk
import numpy as np
import random
import string
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

# Open and read the 'chatbot.txt' file
with open('chatbot.txt', 'r', errors='ignore') as file:
    raw_text = file.read().lower()

# Tokenize the raw text
sent_tokens = nltk.sent_tokenize(raw_text)
word_tokens = nltk.word_tokenize(raw_text)

# Initialize NLTK resources
nltk.download('punkt')
nltk.download('wordnet')
lemmatizer = nltk.stem.WordNetLemmatizer()

def lemmatize_tokens(tokens):
    return [lemmatizer.lemmatize(token) for token in tokens]

def normalize_text(text):
    # Tokenize, lowercase, remove punctuation, and lemmatize
    tokens = nltk.word_tokenize(text.lower().translate(str.maketrans('', '', string.punctuation)))
    return lemmatize_tokens(tokens)

# Define greeting inputs and responses
GREETING_INPUTS = ("hello", "hi", "greetings", "sup", "what's up", "hey")
GREETING_RESPONSES = ["hi", "hey", "*nods*", "hi there", "hello", "I am glad! You are talking to me"]

def detect_greeting(sentence):
    for word in sentence.split():
        if word.lower() in GREETING_INPUTS:
            return random.choice(GREETING_RESPONSES)

# Generate a response using TF-IDF and cosine similarity
def generate_response(user_input):
    robo_response = ''
    sent_tokens.append(user_input)

    tfidf_vectorizer = TfidfVectorizer(tokenizer=normalize_text, stop_words='english')  
    tfidf_matrix = tfidf_vectorizer.fit_transform(sent_tokens)

    similarity_scores = cosine_similarity(tfidf_matrix[-1], tfidf_matrix)
    most_similar_idx = similarity_scores.argsort()[0][-2]

    flat_scores = similarity_scores.flatten()
    flat_scores.sort()
    most_similar_score = flat_scores[-2]

    if most_similar_score == 0:
        robo_response = "I am sorry! I don't understand you"
    else:
        robo_response = sent_tokens[most_similar_idx]

    return robo_response

# Chat with the user
print("ROBO: My name is Robo. I will answer your queries about Chatbots. If you want to exit, type 'Bye'!")

flag = True
while flag:
    user_input = input().lower()

    if user_input != 'bye':
        if user_input in ['thanks', 'thank you']:
            flag = False
            print("ROBO: You are welcome..")
        else:
            greeting_response = detect_greeting(user_input)
            if greeting_response:
                print("ROBO: " + greeting_response)
            else:
                print("ROBO:", end=" ")
                print(generate_response(user_input))
                sent_tokens.remove(user_input)
    else:
        flag = False
        print("ROBO: Bye! Take care..")
