from tkinter import Tk, Frame, TOP, BOTH
from pageone import PageOne

LARGE_FONT = ("Verdana", 16)

class MainApp(Tk):
    def __init__(self, *args, **kwargs):
        # Call the superclass constructor
        Tk.__init__(self, *args, **kwargs)

        # Set up the main application window
        self.setup_window()

        # Load and display the PageOne frame
        self.show_frame(PageOne)

    def setup_window(self):
        # Set the window title
        self.title("KetokMagicHalal")

        # Create a container to hold frames
        self.container = Frame(self)
        self.container.pack(side="top", fill="both", expand=True)

        # Dictionary to store frames
        self.frames = {}

        # Create and store PageOne frame
        self.create_frame(PageOne)

    def create_frame(self, frame_class):
        frame = frame_class(self.container, self)
        self.frames[frame_class] = frame
        frame.pack_forget()

    def show_frame(self, frame_class):
        # Hide the current frame
        if hasattr(self, 'current_frame'):
            self.current_frame.pack_forget()

        # Show the specified frame
        self.current_frame = self.frames[frame_class]
        self.current_frame.pack(side=TOP, fill=BOTH, expand=True, padx=20, pady=20)

if __name__ == '__main__':
    # Create an instance of the MainApp class
    app = MainApp()

    # Set window size
    app.geometry("800x600")

    # Start the main event loop
    app.mainloop()
