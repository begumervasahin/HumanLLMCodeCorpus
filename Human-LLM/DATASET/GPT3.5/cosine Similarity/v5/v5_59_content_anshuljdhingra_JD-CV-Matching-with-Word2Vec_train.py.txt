import json
import random
import logging
import spacy
from sklearn.metrics import precision_recall_fscore_support, accuracy_score
from spacy.gold import GoldParse

def convert_to_spacy(JSON_FilePath):
    try:
        training_data = []
        with open(JSON_FilePath, 'r', encoding='UTF-8') as f:
            for line in f:
                data = json.loads(line)
                text = data['content']
                entities = [(annotation['points'][0]['start'], annotation['points'][0]['end'] + 1, label)
                            for annotation in data['annotation']
                            for label in (annotation['label'] if isinstance(annotation['label'], list) else [annotation['label']])]
                training_data.append((text, {"entities": entities}))
        return training_data
    except Exception as e:
        logging.exception(f"Unable to process {JSON_FilePath}\nerror = {str(e)}")
        return None

def train_spacy():
    # Load training data
    TRAIN_DATA = convert_to_spacy("training_resumes.json")

    # Create a blank English model
    nlp = spacy.blank('en')

    # Add named entity recognizer to the pipeline
    if 'ner' not in nlp.pipe_names:
        ner = nlp.create_pipe('ner')
        nlp.add_pipe(ner, last=True)

    # Add labels to the named entity recognizer
    for _, annotations in TRAIN_DATA:
        for ent in annotations.get('entities'):
            ner.add_label(ent[2])

    # Disable other pipeline components during training
    other_pipes = [pipe for pipe in nlp.pipe_names if pipe != 'ner']
    with nlp.disable_pipes(*other_pipes):
        # Begin training
        optimizer = nlp.begin_training()
        for itn in range(10):
            print(f"Starting iteration {itn}")
            random.shuffle(TRAIN_DATA)
            losses = {}
            for text, annotations in TRAIN_DATA:
                nlp.update(
                    [text],
                    [annotations],
                    drop=0.2,
                    sgd=optimizer,
                    losses=losses
                )
            print(losses)

    # Load test data
    examples = convert_to_spacy("test_resumes.json")

    # Initialize variables for evaluation metrics
    evaluation_metrics = {}

    # Iterate over test examples
    for idx, (text, annot) in enumerate(examples):
        with open(f"resume{idx}.txt", "w", encoding='UTF-8') as f:
            # Process the document
            doc_to_test = nlp(text)

            # Extract entities and write them to a file
            entities_per_label = {}
            for ent in doc_to_test.ents:
                entities_per_label.setdefault(ent.label_, set()).add(ent.text)
            for label, entities in entities_per_label.items():
                f.write(f"\n\n{label}:\n")
                for entity in entities:
                    f.write(entity.replace('\n', '') + "\n")

            # Calculate evaluation metrics
            for ent in doc_to_test.ents:
                if ent.label_ not in evaluation_metrics:
                    evaluation_metrics[ent.label_] = [0, 0, 0, 0, 0]
                if evaluation_metrics[ent.label_][0] == 0:
                    gold = GoldParse(doc_to_test, entities=annot.get("entities"))     
                    y_true = [ent.label_ if ent.label_ in x else 'Not ' + ent.label_ for x in gold.ner]
                    y_pred = [x.ent_type_ if x.ent_type_ == ent.label_ else 'Not ' + ent.label_ for x in doc_to_test]
                    (p, r, f, _) = precision_recall_fscore_support(y_true, y_pred, average='weighted')
                    a = accuracy_score(y_true, y_pred)
                    evaluation_metrics[ent.label_][0] = 1
                    evaluation_metrics[ent.label_][1] += p
                    evaluation_metrics[ent.label_][2] += r
                    evaluation_metrics[ent.label_][3] += f
                    evaluation_metrics[ent.label_][4] += a

    # Print evaluation metrics
    for label, metrics in evaluation_metrics.items():
        print(f"\nFor Entity {label}\n")
        print(f"Accuracy: {(metrics[4] / len(examples)) * 100:.2f}%")
        print(f"Precision: {metrics[1] / len(examples):.4f}")
        print(f"Recall: {metrics[2] / len(examples):.4f}")
        print(f"F-score: {metrics[3] / len(examples):.4f}")

# Train and evaluate the spaCy model
train_spacy()
