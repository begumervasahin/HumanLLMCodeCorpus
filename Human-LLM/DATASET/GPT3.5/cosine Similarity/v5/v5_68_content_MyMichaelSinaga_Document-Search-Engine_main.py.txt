import tkinter as tk
from tkinter import messagebox, Entry, Button, Frame, Label
import os
import numpy as np
import tokeniser as tkn
import spellcheck as sc
import webbrowser

def search_query(query, text_files, stats_dict, Number_of_docs, doc_list, idf_dict, list_of_words):
    processed_query = tkn.tokenise_normalise(query)
    final_query = sc.synonym_list(processed_query)
    final_query = tkn.stem(final_query)

    print(processed_query)

    final_query = [x for x in final_query if x in stats_dict]

    if final_query:
        query_dict = tkn.create_tf_dict_query(final_query)
        query_dict = tkn.find_tfidf_query(query_dict, idf_dict, Number_of_docs)

        vector_array = np.zeros((len(query_dict.keys()), Number_of_docs))
        i = 0
        j = 0
        query_vector = np.zeros((1, len(query_dict.keys())))

        for w in query_dict:
            query_vector[0][i] = query_dict[w]
            for d in doc_list:
                if d not in stats_dict[w]:
                    j += 1
                else:
                    vector_array[i][j] = stats_dict[w][d]
                    j += 1
            i += 1
            j = 0

        magnitude = np.linalg.norm(vector_array, axis=0)
        vector_array = np.divide(vector_array, magnitude)
        vector_array[np.isnan(vector_array)] = 0

        q_magnitude = np.linalg.norm(query_vector, axis=1)
        query_vector = np.divide(query_vector, q_magnitude)

        dot_product = np.dot(query_vector, vector_array)
        dot_product = dot_product.tolist()

        final_rank = list(zip(dot_product[0], doc_list))
        final_rank.sort(reverse=True)

        search_window = tk.Tk()
        search_window.wm_title('4 Sejoli Search')

        label1 = tk.Label(search_window, text=query+'\n\n', font=("ComicSansMS", 20))   
        label1.pack()

        def open_url(event):
            webbrowser.open_new(event.widget.cget("text"))

        for i in final_rank[:2]:
            label = tk.Label(search_window, text=i[1], font=("ComicSansMS", 12), justify=tk.LEFT, fg="blue", cursor="hand2")
            label.bind("<Button-1>", open_url)
            label.pack()
    else:
        messagebox.showinfo("Error Message", "Query not found")

def load_document_statistics(corpus_dir):
    text_files = [os.path.join(corpus_dir, f) for f in os.listdir(corpus_dir)]
    stats_dict = {}
    Number_of_docs = len(text_files)
    doc_list = []

    for f in text_files:
        with open(f, 'r') as doc:
            lines = [l.strip() for l in doc.readlines()]
            full_transcript = tkn.tokenise_normalise(lines)
            full_transcript = tkn.stem(full_transcript)
            stats_dict = tkn.create_tf_dict_doc(full_transcript, f, stats_dict)
            doc_list.append(f)

    stats_dict, idf_dict = tkn.find_tfidf_doc(stats_dict, Number_of_docs)
    list_of_words = sorted(stats_dict.keys())
    doc_list.sort()

    return text_files, stats_dict, Number_of_docs, doc_list, idf_dict, list_of_words

def create_gui():
    top = tk.Tk()
    top.wm_title("4 Sejoli Search")

    frame = Frame(top, width=600, height=350)
    frame.pack(fill=tk.X, expand=True)

    entry = Entry(top, bd=6, width=40)
    entry.insert(tk.END, '')
    entry.place(relx=0.5, rely=0.35, anchor=tk.CENTER)

    button_search = Button(top, text="Search", command=lambda: search_query(entry.get(), *document_data))
    button_search.place(relx=0.5, rely=0.5, anchor=tk.CENTER)

    return top
