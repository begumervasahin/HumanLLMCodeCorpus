from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import linear_kernel
import PyPDF2
import glob
import re
import argparse

def load_documents(path):
    """Load PDF documents from the specified path."""
    documents = []
    for pdf_file in glob.glob(path + "/*.pdf"):
        with open(pdf_file, 'rb') as file:
            reader = PyPDF2.PdfFileReader(file)
            text = extract_text_from_pdf(reader)
            documents.append((pdf_file, text))
    return documents

def extract_text_from_pdf(reader):
    """Extract text from a PDF reader object."""
    text = ''
    for page_num in range(reader.numPages):
        page = reader.getPage(page_num)
        page_text = page.extractText()
        page_text = preprocess_text(page_text)
        text += page_text + ' '
    return text

def preprocess_text(text):
    """Preprocess text by removing special characters and digits, and converting to lowercase."""
    text = text.replace('\n', ' ').replace('\r', ' ').replace('.', '').replace(',', '') 
    text = ''.join([char for char in text if not char.isdigit()])
    text = re.sub(r'[^\w]', ' ', text)
    text = " ".join(text.split())
    return text.lower()

def calculate_tfidf_matrix(documents):
    """Calculate TF-IDF matrix for the given documents."""
    vectorizer = TfidfVectorizer()
    tfidf_matrix = vectorizer.fit_transform([content for _, content in documents])      
    return documents, tfidf_matrix

def find_similar_documents(tfidf_matrix, index, top_n):
    """Find similar documents based on TF-IDF matrix."""
    cosine_similarities = linear_kernel(tfidf_matrix[index:index+1], tfidf_matrix).flatten()
    related_indices = sorted(range(len(cosine_similarities)), key=lambda i: cosine_similarities[i], reverse=True)
    related_indices = [i for i in related_indices if i != index]
    return [(idx, cosine_similarities[idx]) for idx in related_indices][:top_n]

def parse_arguments():
    """Parse command line arguments."""
    parser = argparse.ArgumentParser(description='Calculate Similarity Index of Documents')
    parser.add_argument('-P', '--path', help='Path to the document corpus', required=True)
    parser.add_argument('-T', '--top', help='Number of top similar documents to display', type=int, default=10)
    parser.add_argument('-I', '--index', help='Index of the document to find similarities for', type=int, required=True)
    return parser.parse_args()

def main():
    args = parse_arguments()
    documents = load_documents(args.path)
    corpus, tfidf_matrix = calculate_tfidf_matrix(documents)

    print('The selected document is:', corpus[args.index][0], '\n')
    print('======== Similarity Index Results ========')
    for idx, score in find_similar_documents(tfidf_matrix, args.index, args.top):       
        print(score, corpus[idx][0])

if __name__ == '__main__':
    main()
