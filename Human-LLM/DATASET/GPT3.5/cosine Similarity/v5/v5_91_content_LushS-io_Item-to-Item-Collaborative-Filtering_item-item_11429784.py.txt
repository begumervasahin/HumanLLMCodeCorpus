import pandas as pd
import numpy as np
import scipy.sparse as sps
from sklearn.metrics.pairwise import cosine_similarity
from sklearn.neighbors import NearestNeighbors
import time

# Sample data
user_data = {
    'user_1': [1, 0, 3, 0, 0, 5, 0, 0, 5, 0, 4, 0],
    'user_2': [0, 0, 5, 4, 0, 0, 4, 0, 0, 2, 1, 3],
    'user_3': [2, 4, 0, 1, 2, 0, 3, 0, 4, 3, 5, 0],
    'user_4': [0, 2, 4, 0, 5, 0, 0, 4, 0, 0, 2, 0],
    'user_5': [0, 0, 4, 3, 4, 2, 0, 0, 0, 0, 2, 5],
    'user_6': [1, 0, 3, 0, 3, 0, 0, 2, 0, 0, 4, 0]
}

# Create DataFrame and sparse matrix
users_df = pd.DataFrame(data=user_data)
sparse_matrix = sps.csr_matrix(users_df)

# Load additional data
links = pd.read_csv("./movie-lens-data/links.csv")
movies = pd.read_csv("./movie-lens-data/movies.csv")
ratings = pd.read_csv("./movie-lens-data/ratings.csv")
tags = pd.read_csv("./movie-lens-data/tags.csv")

# Merge and process data
merged_df = pd.merge(movies, ratings, on='movieId', how='outer')
ratings_df = merged_df[['rating', 'movieId', 'userId']].copy()
pivot_df = ratings_df.pivot(index='userId', columns='movieId', values='rating').fillna(0)

# Calculate average and center CSR matrix
csr_matrix = sps.csr_matrix(sparse_matrix)
rows, cols, data = sps.find(csr_matrix)
counts = np.bincount(rows)
weighted_sum = np.bincount(rows, weights=data)
csr_avg = weighted_sum / counts

centered_matrix = sps.csr_matrix((csr_matrix.data - np.repeat(csr_avg, np.diff(csr_matrix.indptr)),
                                  csr_matrix.indices, csr_matrix.indptr), shape=csr_matrix.shape)

# Calculate cosine similarity
start_time = time.time()
similarity_matrix = centered_matrix.dot(centered_matrix.T).toarray()
diagonal = similarity_matrix.diagonal()
inv_diagonal = np.where(diagonal != 0, 1 / diagonal, 0)
cosine_sim = similarity_matrix * np.sqrt(inv_diagonal).reshape(-1, 1)
cosine_sim = cosine_sim.T * np.sqrt(inv_diagonal)
end_time = time.time()
run_time = end_time - start_time

# Calculate cosine similarity using sklearn
start = time.time()
sklearn_cosine_sim = cosine_similarity(centered_matrix)
end = time.time()
total_time = end - start

# Create COO matrix and fit Nearest Neighbors model
coo_matrix = sps.coo_matrix(cosine_sim)
k_neighbors = 5
knn_model = NearestNeighbors(n_neighbors=k_neighbors)
knn_model.fit(coo_matrix)
