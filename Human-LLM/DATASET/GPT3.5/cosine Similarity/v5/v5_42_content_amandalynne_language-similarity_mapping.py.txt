class Mapping:
    def __init__(self, items):
        """
        Initialize the Mapping object with the given items.

        Parameters:
        - items (iterable): A collection of items to be indexed.
        """
        # Initialize the item list, items dictionary, and index dictionary
        self.item_list = items
        self.items = {index: item for index, item in enumerate(items)}
        self.index = {item: index for index, item in enumerate(items)}

    def __contains__(self, item):
        """
        Check if the Mapping contains the specified item.

        Parameters:
        - item: The item to check for existence.

        Returns:
        - bool: True if the item exists in the Mapping, False otherwise.
        """
        # Check if the item exists in the index dictionary
        return item in self.index

    def __getitem__(self, item):
        """
        Get the index of the specified item.

        Parameters:
        - item: The item to retrieve the index for.

        Returns:
        - int: The index of the item.
        """
        # Get the index of the item from the index dictionary
        return self.index[item]

    def __iter__(self):
        """Return an iterator over the items in the Mapping."""
        # Iterate through the index dictionary
        return iter(self.index)

    def __len__(self):
        """Return the number of items in the Mapping."""
        # Get the length of the index dictionary
        return len(self.index)

    def __repr__(self):
        """Return a string representation of the Mapping object."""
        # Provide a string representation of the Mapping object
        return f"<{self.__class__.__name__} with {len(self)} entries>"

    def add(self, item):
        """
        Add a new item to the Mapping object if it doesn't already exist.

        Parameters:
        - item: The item to add to the Mapping.

        Returns:
        - item: The item added to the Mapping.
        """
        # Add a new item to the Mapping object if it doesn't already exist
        if item not in self:
            index = len(self)
            self.items[index] = item
            self.index[item] = index
        return item

    def get_index(self, item, default=None):
        """
        Get the index of the specified item from the Mapping.

        Parameters:
        - item: The item to retrieve the index for.
        - default: The default value to return if the item is not found.

        Returns:
        - int: The index of the item, or the default value if not found.
        """
        # Get the index of the item from the index dictionary with a default value
        return self.index.get(item, default)

    def get_item(self, index):
        """
        Get the item corresponding to the specified index from the Mapping.

        Parameters:
        - index (int): The index of the item to retrieve.

        Returns:
        - item: The item corresponding to the specified index.
        """
        # Get the item corresponding to the given index
        return self.items[index]

    def keys(self):
        """Return the keys of the Mapping to preserve the original order."""
        # Return the item list to preserve the original order
        return self.item_list
