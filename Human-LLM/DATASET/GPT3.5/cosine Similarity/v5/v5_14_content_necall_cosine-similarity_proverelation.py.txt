import distsim

# Load the word vectors dictionary
word_to_vec_dict = distsim.load_word2vec("nyt_word2vec.4k")

# Define the two relation lists
relationA = [
    'act action add addition',
    'attend attention decide decision',
    'die death feed food',
    'see scene grow growth',
    'move movement advise advice',
    'choose choice challenge challenge'
]

relationB = [
    'buy bought eat ate',
    'become became begin began',
    'accept accepted add added',
    'think thought build built',
    'break broke wake woke',
    'win won see saw',
    'hold held hurt hurt',
    'keep kept know knew',
    'find found give gave',
    'say said happen happened',
    'offer offered open opened',
    'pass passed tell told'
]

def calculate_accuracy(relation):
    # Initialize variables for counting accuracy
    correct_top1 = 0
    correct_top5 = 0
    correct_top10 = 0

    # Iterate through each relation
    for rel in relation:
        words = rel.split(' ')
        word1_vec = word_to_vec_dict[words[0]]
        word2_vec = word_to_vec_dict[words[1]]
        word4_vec = word_to_vec_dict[words[3]]

        # Calculate the word closest to the expected word using vector arithmetic
        nearest_words = distsim.show_nearest(
            word_to_vec_dict,
            word1_vec - word2_vec + word4_vec,
            set([words[0], words[1], words[3]]),
            distsim.cossim_dense
        )

        # Check if the expected word is in the top 1, 5, and 10 nearest words
        if words[2] == nearest_words[0][0]:
            correct_top1 += 1
        for i in range(5):
            if words[2] == nearest_words[i][0]:
                correct_top5 += 1
        for i in range(10):
            if words[2] == nearest_words[i][0]:
                correct_top10 += 1

    total_rels = len(relation)
    acc_top1 = round(float(correct_top1) / total_rels, 2)
    acc_top5 = round(float(correct_top5) / total_rels, 2)
    acc_top10 = round(float(correct_top10) / total_rels, 2)

    return acc_top1, acc_top5, acc_top10

# Print the results for relationA and relationB
print("Relation A Accuracy:", calculate_accuracy(relationA))
print("Relation B Accuracy:", calculate_accuracy(relationB))
