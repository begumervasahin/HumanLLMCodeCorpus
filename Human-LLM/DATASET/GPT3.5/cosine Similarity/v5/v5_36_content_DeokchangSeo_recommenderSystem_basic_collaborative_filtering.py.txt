from surprise import similarities as sims
from surprise import Dataset, Reader
import heapq

def load_dataset(dataset_path, line_format='user item rating', rating_scale=(1, 7), sep=None):
    # Load dataset from a file using Surprise library
    reader = Reader(line_format=line_format, sep=sep, rating_scale=rating_scale)
    dataset = Dataset.load_from_file(dataset_path, reader)
    return dataset

def create_full_trainset(dataset_path, line_format='user item rating', rating_scale=(1, 7), sep=None):
    # Create a full training set from the dataset
    dataset = load_dataset(dataset_path, line_format=line_format, rating_scale=rating_scale, sep=sep)
    trainset = dataset.build_full_trainset()
    return trainset

def mean_centered_ratings(rating_matrix):
    # Compute mean-centered ratings for each user or item
    mean_centered = {}
    for id, ratings in rating_matrix.items():
        ratings_list = [rating for _, rating in ratings]
        mean_rating = sum(ratings_list) / len(ratings_list)
        centered_ratings = [(item_id, rating - mean_rating) for item_id, rating in ratings]
        mean_centered[id] = centered_ratings
    return mean_centered

def adjusted_cosine_similarity(num_rows, rating_matrix, min_support):
    # Compute adjusted cosine similarity between all pairs of users or items
    mean_centered = mean_centered_ratings(rating_matrix)
    similarity_matrix = {}
    for id, ratings in mean_centered.items():
        sim_row = {}
        for xi, ri in ratings:
            for xj, rj in ratings:
                if xi != xj:
                    sim_row[xi] = sim_row.get(xi, 0) + ri * rj
        similarity_matrix[id] = sim_row
    return similarity_matrix

def predict_rating(similarity_matrix, target_id, target_row, target_ratings, neighbor_size):
    # Predict rating based on collaborative filtering
    target_sim = [similarity_matrix[target_id].get(item_id, 0) for item_id, _ in target_ratings[target_row]]
    n_sims = heapq.nlargest(neighbor_size, target_sim)
    neighbors = {item_id: similarity_matrix[target_id][item_id] for item_id in range(neighbor_size) if similarity_matrix[target_id].get(item_id, 0) in n_sims}
    selected_ratings = {item_id: rating for item_id, rating in target_ratings[target_row] if item_id in neighbors}
    if len(selected_ratings) < neighbor_size:
        print('Not enough neighbors')
        return 0
    prediction = sum(selected_ratings[item_id] * neighbors[item_id] for item_id in neighbors)   
    return prediction / sum(neighbors.values())

def collaborative_filtering_prediction(trainset, item_id, user_id, neighbor_size, user_item, sim_method):
    # Perform collaborative filtering prediction
    item_idx = item_id - 1
    user_idx = user_id - 1
    min_support = 0
    num_users = trainset.n_users
    num_items = trainset.n_items
    user_ratings = trainset.ur
    item_ratings = trainset.ir
    target_id, target_row, target_ratings = num_users, user_idx, user_ratings
    if user_item != "user":
        target_id, target_row, target_ratings = num_items, item_idx, item_ratings
    args = [target_id, target_ratings, min_support]
    if sim_method == "cosine":
        similarity_matrix = sims.cosine(*args)
    elif sim_method == "pearson":
        similarity_matrix = sims.pearson(*args)
    else:
        similarity_matrix = adjusted_cosine_similarity(*args)
    print('\nSimilarity Matrix:\n', similarity_matrix)
    return predict_rating(similarity_matrix, target_id, target_row, target_ratings, neighbor_size)

# Example usage
file_path = 'ratings_for_question1.txt'
trainset = create_full_trainset(file_path, line_format='user item rating', rating_scale=(1, 7), sep='\t')

item_id = 1
user_id = 3
neighbor = 2
user_item = 'user'
similarity_method = 'pearson'
prediction = collaborative_filtering_prediction(trainset, item_id, user_id, neighbor, user_item, similarity_method)
print ('\nUsing {} based CF, the predicted rating for item {} by user {} is: {est:4.2f} \n'.format(user_item, item_id, user_id, est=prediction))
