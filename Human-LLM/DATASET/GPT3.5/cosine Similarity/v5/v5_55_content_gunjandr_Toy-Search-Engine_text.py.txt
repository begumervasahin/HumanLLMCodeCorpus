import os
import nltk
import collections
import operator
import math
from collections import Counter
from nltk.tokenize import RegexpTokenizer
from nltk.corpus import stopwords
from nltk.stem.porter import PorterStemmer

# Define the corpus root directory
corpus_root = './presidential_debates'

def tokenize(corpus_root):
    # Initialize lists to store tokens, term frequency, and file names
    tokens_list = []
    tf_list = []
    file_list = []

    # Iterate through each file in the corpus
    for filename in os.listdir(corpus_root):
        file_list.append(filename)
        with open(os.path.join(corpus_root, filename), "r", encoding='UTF-8') as file:  
            # Read the document and convert to lowercase
            document = file.read().lower()

        # Tokenize the document and remove stopwords
        tokenizer = RegexpTokenizer(r'[a-zA-Z]+')
        tokens = tokenizer.tokenize(document)
        tokens = [token for token in tokens if token not in stopwords.words("english")] 

        # Stem the tokens using Porter Stemmer
        stemmer = PorterStemmer()
        stemmed_tokens = [stemmer.stem(token) for token in tokens]

        # Append tokenized document, term frequency, and filename to respective lists
        tokens_list.append(stemmed_tokens)
        tf_list.append(dict(Counter(stemmed_tokens)))

    return tokens_list, tf_list, file_list

def calculate_tfwt(tf_list):
    # Calculate Term Frequency-Weighted Term (TFWT)
    tfwt_list = []
    for tf_dict in tf_list:
        tfwt_dict = {token: 1 + math.log10(frequency) for token, frequency in tf_dict.items()}
        tfwt_list.append(tfwt_dict)
    return tfwt_list

def calculate_idf(tf_list):
    # Calculate Inverse Document Frequency (IDF)
    unique_words = set(token for tf_dict in tf_list for token in tf_dict.keys())        
    idf_dict = {word: math.log10(len(tf_list) / sum(1 for tf_dict in tf_list if word in tf_dict)) for word in unique_words}
    return idf_dict

def calculate_tf_idf(tfwt_list, idf_dict):
    # Calculate Term Frequency-Inverse Document Frequency (TF-IDF)
    tf_idf_list = []
    for tfwt_dict in tfwt_list:
        tf_idf_dict = {token: tfwt * idf_dict[token] for token, tfwt in tfwt_dict.items() if token in idf_dict}
        tf_idf_list.append(tf_idf_dict)
    return tf_idf_list

def normalize(tf_idf_list):
    # Normalize TF-IDF values
    normalized_tf_idf = []
    for tf_idf_dict in tf_idf_list:
        norm_factor = math.sqrt(sum(tfidf ** 2 for tfidf in tf_idf_dict.values()))      
        normalized_tf_idf.append({token: tfidf / norm_factor for token, tfidf in tf_idf_dict.items()})
    return normalized_tf_idf

def query(qstring, tokenizer):
    # Process the query string
    q_tokens = tokenizer.tokenize(qstring.lower())
    q_tokens = [token for token in q_tokens if token not in stopwords.words("english")] 

    stemmer = PorterStemmer()
    q_stemmed_tokens = [stemmer.stem(token) for token in q_tokens]
    q_dict = dict(Counter(q_stemmed_tokens))
    qtf_dict = {token: 1 + math.log10(freq) for token, freq in q_dict.items()}

    sums_q = math.sqrt(sum(freq ** 2 for freq in qtf_dict.values()))
    normalized_query = {token: freq / sums_q for token, freq in qtf_dict.items()}       
    return normalized_query

def create_postings_list(file_list, tokens_list):
    # Create postings list
    postings_list = {}
    for token in set(token for tokens in tokens_list for token in tokens):
        token_dict = {}
        for i, tokens in enumerate(tokens_list):
            if token in tokens:
                token_dict[file_list[i]] = tokens.count(token)
        postings_list[token] = token_dict
    return postings_list

def get_top_k(postings_list, qtf_dict, k=10):
    # Get top k postings
    top_k = {}
    for token, tf in qtf_dict.items():
        if token in postings_list:
            sorted_postings = sorted(postings_list[token].items(), key=operator.itemgetter(1), reverse=True)
            top_k[token] = sorted_postings[:k]
    return top_k

def similarity(top_k, file_list, normalized_query):
    # Calculate similarity scores
    final_score_dict = {}
    for file in file_list:
        score = 0
        for token, tf in normalized_query.items():
            if token in top_k:
                for doc, tfidf in top_k[token]:
                    if doc == file:
                        score += tf * tfidf
                        break
                    else:
                        score += tf * top_k[token][-1][1]
        final_score_dict[file] = score
    return final_score_dict

    tokens_list, tf_list, file_list = tokenize(corpus_root)
    tfwt_list = calculate_tfwt(tf_list)
    idf_dict = calculate_idf(tf_list)
    tf_idf_list = calculate_tf_idf(tfwt_list, idf_dict)
    normalized_tf_idf = normalize(tf_idf_list)

    # Process query
    qstring = input("Enter your query: ")
    normalized_query = query(qstring, RegexpTokenizer(r'[a-zA-Z]+'))

    # Create postings list and get top k postings
    postings_list = create_postings_list(file_list, tokens_list)
    top_k = get_top_k(postings_list, normalized_query)

    # Calculate similarity scores
    final_scores = similarity(top_k, file_list, normalized_query)

    # Print results
    for file, score in sorted(final_scores.items(), key=operator.itemgetter(1), reverse=True):
        print(file, score)
