import pickle
import string
import nltk
from nltk.stem import PorterStemmer
import settings

def export_page_rank(page_rank_data):
    """Export page rank data to a pickle file."""
    with open('./data/page_rank.p', 'wb') as file:
        pickle.dump(page_rank_data, file, protocol=pickle.HIGHEST_PROTOCOL)

def import_page_rank():
    """Import page rank data from a pickle file."""
    with open('./data/page_rank.p', 'rb') as file:
        page_rank_data = pickle.load(file)
    return page_rank_data

def import_crawled_data(threshold):
    """Import crawled data and tokenize it."""
    for index in range(1, threshold + 1):
        try:
            with open(f'./data/{index}', 'r', encoding="utf-8") as file:
                data = file.read().splitlines()
                settings.web_dict[data[0]] = tokenize_data(data[1])
        except:
            settings.web_dict[data[0]] = []

def export_web_graph(web_graph_data):
    """Export web graph data to a pickle file."""
    with open('./data/web_graph.p', 'wb') as file:
        pickle.dump(web_graph_data, file, protocol=pickle.HIGHEST_PROTOCOL)

def import_web_graph():
    """Import web graph data from a pickle file."""
    with open('./data/web_graph.p', 'rb') as file:
        web_graph_data = pickle.load(file)
    return web_graph_data

def export_url_list(url_list):
    """Export URL list to a text file."""
    with open("./data/all_url", 'w', encoding="utf-8") as file:
        for url in url_list:
            file.write(url)
            file.write("\n")

def export_spider():
    """Export spider data to individual files."""
    count = 1
    for key, value in settings.web_dict.items():
        with open(f"./data/{count}", 'w', encoding="utf-8") as file:
            file.write(key)
            file.write("\n")
            file.write(" ".join(str(x) for x in value))
        count += 1

def export_spider_sequential(link, tokens, count):
    """Export spider data to individual files sequentially."""
    with open(f"./data/{count}", 'w', encoding="utf-8") as file:
        file.write(link)
        file.write("\n")
        file.write(" ".join(str(x) for x in tokens))

def remove_punctuation(text):
    """Remove punctuation from text."""
    table = str.maketrans(string.punctuation, ' ' * len(string.punctuation))
    return text.translate(table)

def remove_digits(text):
    """Remove digits from text."""
    table = str.maketrans('', '', string.digits)
    return text.translate(table)

def tokenize_data(text):
    """Tokenize text using NLTK."""
    return nltk.word_tokenize(text)

def remove_stop_words(tokens):
    """Remove stop words from tokens."""
    filtered_tokens = []
    for token in tokens:
        if token not in settings.stop_words and len(token) > 2:
            filtered_tokens.append(token)
    return filtered_tokens

def perform_stemming(tokens):
    """Perform stemming on tokens using Porter Stemmer."""
    ps = PorterStemmer()
    stemmed_tokens = [ps.stem(token) for token in tokens]
    return stemmed_tokens
