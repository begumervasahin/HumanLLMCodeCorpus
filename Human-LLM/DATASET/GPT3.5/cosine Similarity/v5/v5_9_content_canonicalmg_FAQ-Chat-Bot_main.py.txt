from similarity import find_most_similar
from corpus import CORPUS

class Bot:
    def __init__(self):
        self.event_stack = []
        self.settings = {
            "min_score": 0.2,
            "help_email": "fakeEmail@notArealEmail.com",
            "faq_page": "www.NotActuallyAnFAQ.com"
        }
        print("Welcome to the FAQ Bot!")
        print("Ask a question:")
        while True:
            self.process_question()

    def process_question(self):
        potential_event = self.event_stack.pop() if self.event_stack else None

        if potential_event:
            response = input("Response: ")
            potential_event.handle_response(response, self)
        else:
            question = input("Question: ")
            answer = self.check_prebuilt_responses(question)
            if not answer:
                answer = find_most_similar(question)
                self.answer_question(answer, question)

    def answer_question(self, answer, question):
        if answer['score'] > self.settings['min_score']:
            print(f"\nBest-fit question: {answer['question']} (Score: {answer['score']})\nAnswer: {answer['answer']}\n")
        else:
            print("Oops! I'm having trouble finding the answer to your question. "
                  "Would you like to see the list of questions that I am able to answer?\n")
            self.event_stack.append(Event("corpus_dump", question))

    def check_prebuilt_responses(self, question):
        prebuilt_responses = [
            {"Question": "Who made you?", "Answer": "I was created by TS-North.\n"},
            {"Question": "When were you born?", "Answer": "I first opened my eyes in alpha stage on February 9th, 2018.\n"},
            {"Question": "What is your purpose?", "Answer": "I assist user experience by providing an interactive FAQ chat.\n"},
            {"Question": "Thanks", "Answer": "Glad I could help!\n"},
            {"Question": "Thank you", "Answer": "Glad I could help!\n"}
        ]
        for response in prebuilt_responses:
            if response['Question'].lower() in question.lower():
                print(response['Answer'])
                return response
        return None

class Event:
    def __init__(self, kind, text):
        self.kind = kind
        self.CONFIRMATIONS = ["yes", "sure", "okay", "that would be nice", "yep"]       
        self.NEGATIONS = ["no", "don't", "dont", "nope"]
        self.original_text = text

    def handle_response(self, response, bot):
        if self.kind == "corpus_dump":
            self.dump_corpus(response, bot)

    def dump_corpus(self, response, bot):
        for confirmation in self.CONFIRMATIONS:
            if confirmation.lower() in response.lower():
                corpus = bot.dump_corpus()
                corpus = ["-" + question for question in corpus]
                print(f"\n{chr(10).join(corpus)}\n")
                return

        for negation in self.NEGATIONS:
            if negation.lower() in response.lower():
                print(f"Feel free to ask another question or send an email to {bot.settings['help_email']}.\n")
                bot.process_question()
                return

        print(f"I'm having trouble understanding your response. "
              f"For assistance, please refer to {bot.settings['faq_page']} or email {bot.settings['help_email']}. "
              f"A Google search link has been generated for your original question: \n" 
              f"{'https://www.google.com/search?q=' + '+'.join(self.original_text.split(' '))}\n")

Bot()
