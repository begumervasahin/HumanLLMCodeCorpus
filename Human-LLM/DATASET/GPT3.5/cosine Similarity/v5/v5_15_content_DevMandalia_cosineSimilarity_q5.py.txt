import distsim

# Load the word vectors dictionary
word_to_vec_dict = distsim.load_word2vec("nyt_word2vec.4k")

# Define a list of words to process along with their corresponding descriptions
words_to_process = [
    ("america", "Word 1"),
    ("years", "Word 2"),
    ("great", "Word 3"),
    ("run", "Word 4"),
    ("between", "Word 5"),
    ("wife", "Word 6")
]

# Iterate through the list of words and descriptions
for word, description in words_to_process:
    print(f"{description} is: {word}")
    # Retrieve the nearest words and scores for the current word
    nearest_words = distsim.show_nearest(word_to_vec_dict, word_to_vec_dict[word], set([word]), distsim.cossim_dense)
    # Display the nearest words along with their scores
    for rank, (nearest_word, score) in enumerate(nearest_words, start=1):
        print(f"  Rank {rank}: {nearest_word} (Score: {score})")
    print("--------------------")
