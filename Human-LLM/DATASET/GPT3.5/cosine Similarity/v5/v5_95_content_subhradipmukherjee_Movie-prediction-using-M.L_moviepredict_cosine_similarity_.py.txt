import pandas as pd
from sklearn.feature_extraction.text import CountVectorizer
from sklearn.metrics.pairwise import cosine_similarity
import movie_predict_knn

# Function to get the movie title based on its index in the DataFrame
def get_movie_title(index):
    return df.loc[index, "title"]

# Function to get the index of a movie based on its title in the DataFrame
def get_movie_index(title):
    return df[df["title"] == title].index[0]

# Load the movie dataset from a CSV file
df = pd.read_csv("D:/roops_backup/projects/data mining project/movie_recommender/movie_dataset.csv")

# Fill missing values in selected features with empty strings
features_to_fill = ['keywords', 'cast', 'genres', 'director']
df[features_to_fill] = df[features_to_fill].fillna('')

# Combine selected features into a single column named 'combined_features'
df["combined_features"] = df.apply(lambda row: f"{row['keywords']} {row['cast']} {row['genres']} {row['director']}", axis=1)

# Initialize CountVectorizer to convert text data into a numerical format
cv = CountVectorizer()
count_matrix = cv.fit_transform(df["combined_features"])

# Calculate cosine similarity between movies based on their combined features
cosine_sim = cosine_similarity(count_matrix)

# Define the movie of interest and get its index
target_movie = "Avengers: Age of Ultron"
target_movie_index = get_movie_index(target_movie)

# Get top similar movies using cosine similarity
similar_movies = sorted(enumerate(cosine_sim[target_movie_index]), key=lambda x: x[1], reverse=True)

print("Results with cosine similarity:")
for movie_id, similarity_score in similar_movies[1:6]:  # Exclude the movie itself
    movie_title = get_movie_title(movie_id)
    print(f"{movie_title} (Similarity Score: {similarity_score:.2f})")

# Use KNN for movie prediction
print("\nResults with KNN:")
K = 5
movie_predict_knn.main(target_movie_index, K)
