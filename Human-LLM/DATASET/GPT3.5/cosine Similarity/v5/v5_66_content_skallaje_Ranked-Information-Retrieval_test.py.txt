from collections import defaultdict

# Define the documents and index
documents = [['dog', 'deer', 'cat', 'mouse'], ['parrot', 'cat'], ['turtle', 'dog']]
index = {'dog': [0, 2], 'cat': [0, 1], 'parrot': [1], 'turtle': [2]}

def and_search_with_list_comprehension(documents, index, queries):
    # Initialize a set with document IDs matching the first query
    doc_ids = set(index[queries[0]])

    # Perform AND search for subsequent queries
    for query in queries[1:]:
        doc_ids &= set(index[query])

    # Return documents corresponding to the matching document IDs
    return [documents[doc_id] for doc_id in doc_ids]

def and_search_with_defaultdict(documents, index, queries):
    # Create a defaultdict to count occurrences of document IDs
    doc_id_counter = defaultdict(int)

    # Count occurrences of document IDs for each query
    for query in queries:
        doc_id_counter.update(index[query])

    # Get the most common document ID and return the corresponding document
    most_common_doc_id = doc_id_counter.most_common(1)[0][0]
    return [documents[most_common_doc_id]]

# Test the AND search functions
print("AND Search using List Comprehension:", and_search_with_list_comprehension(documents, index, ['cat', 'dog']))
print("AND Search using defaultdict:", and_search_with_defaultdict(documents, index, ['cat', 'dog']))
