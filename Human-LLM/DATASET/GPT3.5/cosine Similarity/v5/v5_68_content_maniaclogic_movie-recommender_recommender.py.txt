import pandas as pd
import numpy as np
from models import R, nmf, DF

def recommend(user_input_dict):
    # Step 1: Prepare user ratings DataFrame
    movie_ids = sorted(list(R.columns))
    ratings_dict = {movie_id: np.nan for movie_id in movie_ids}
    for movie_id, rating in user_input_dict.items():
        ratings_dict[movie_id] = rating
    user_ratings_df = pd.DataFrame([list(ratings_dict.values())], index=[0], columns=movie_ids)

    # Step 2: Fill missing values and transpose the DataFrame
    user_ratings_filled = user_ratings_df.fillna(3.0).transpose()

    # Step 3: Transform user profile using NMF
    hidden_profile = nmf.transform(user_ratings_filled)

    # Step 4: Predict ratings for all movies
    predicted_ratings = np.dot(hidden_profile, nmf.components_)
    predicted_ratings_df = pd.DataFrame(predicted_ratings, columns=R.columns)

    # Step 5: Get movies not watched by the user
    unrated_movies = predicted_ratings_df.transpose().iloc[0][np.isnan(user_ratings_filled.iloc[0])]
    sorted_unrated_movies = unrated_movies.sort_values(ascending=False)
    recommended_movie_ids = sorted_unrated_movies.index[:5]

    # Step 6: Get movie titles from DataFrame
    recommended_movies = [DF.loc[movie_id, 'title'] for movie_id in recommended_movie_ids]

    return recommended_movies

# Example usage
user_input_dict = {1: 5, 2: 4, 3: 3, 4: 2, 5: 1}  # User ratings for movies
recommended_movies = recommend(user_input_dict)

# Print recommended movies
print("Recommended Movies:")
for movie in recommended_movies:
    print(movie)
