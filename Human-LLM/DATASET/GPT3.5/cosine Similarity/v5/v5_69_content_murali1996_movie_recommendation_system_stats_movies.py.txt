import pandas as pd
import numpy as np
import pickle

def load_data(file_path):
    with open(file_path, "rb") as file:
        return pickle.load(file)

def extract_watched_movies(ratings, items_df, user_id):
    user_ratings = ratings[ratings['user_id'] == user_id]
    watched_df = pd.DataFrame(columns=['movie_id', 'movie_title'])
    for _, row in user_ratings.iterrows():
        movie_id = row['movie_id']
        movie_title = items_df.loc[movie_id - 1, 'movie_title']
        watched_df = watched_df.append({'movie_id': movie_id, 'movie_title': movie_title}, ignore_index=True)
    return watched_df

def movies_watched(user_id):
    ratings = load_data("ratings.pickle")
    items_df = load_data("u_item.pickle")
    watched_df = extract_watched_movies(ratings, items_df, user_id)
    return watched_df.to_json(orient='records')

def calculate_genre_frequencies(ratings, items_df, user_feat, item_feat, user_id, retrieve_top=20):
    genre_indices_for_mat = range(6, 24)
    round_off_to = 2

    user_ratings = ratings[ratings['user_id'] == user_id]
    user_genre_freq = np.zeros(len(genre_indices_for_mat), dtype='float64')
    for _, row in user_ratings.iterrows():
        movie_id = row['movie_id']
        if row['rating'] == 1 and items_df.loc[movie_id - 1, 'unknown'] != 1:
            user_genre_freq += items_df.loc[movie_id - 1, genre_indices_for_mat].values 

    user_genre_freq /= user_genre_freq.sum()
    user_genre_freq = np.round(user_genre_freq, round_off_to)

    genre_freq_df = pd.DataFrame(user_genre_freq, index=['Action', 'Adventure', 'Animation', "Children's", 'Comedy',
                                                        'Crime', 'Documentary', 'Drama', 'Fantasy', 'Film-Noir',
                                                        'Horror', 'Musical', 'Mystery', 'Romance', 'Sci-Fi',
                                                        'Thriller', 'War', 'Western'], columns=['user_genre_freq'])

    pred_genre_freq = np.zeros(len(genre_indices_for_mat), dtype='float64')
    for movie_id in user_ratings['movie_id']:
        if movie_id not in items_df.index:
            continue
        pred_rat = user_feat[user_id - 1].dot(item_feat[movie_id - 1])
        pred_genre_freq += items_df.loc[movie_id - 1, genre_indices_for_mat].values * np.abs(1 - pred_rat)

    pred_genre_freq /= pred_genre_freq.sum()
    pred_genre_freq = np.round(pred_genre_freq, round_off_to)

    pred_genre_freq_df = pd.DataFrame(pred_genre_freq, index=['Action', 'Adventure', 'Animation', "Children's", 'Comedy',
                                                              'Crime', 'Documentary', 'Drama', 'Fantasy', 'Film-Noir',
                                                              'Horror', 'Musical', 'Mystery', 'Romance', 'Sci-Fi',
                                                              'Thriller', 'War', 'Western'], columns=['pred_genre_freq'])

    result_df = pd.concat([genre_freq_df, pred_genre_freq_df], axis=1).transpose()
    result_df.to_json("genre_frequencies.json", orient='index', indent=4)

    return result_df

def give_genre_freq(user_id, retrieve_top=20):
    ratings = load_data("ratings.pickle")
    items_df = load_data("u_item.pickle")
    user_feat = load_data("user_feat.pickle")
    item_feat = load_data("item_feat.pickle")
    
    return calculate_genre_frequencies(ratings, items_df, user_feat, item_feat, user_id, retrieve_top)

if __name__ == "__main__":
    user_id = 1
    print("Movies watched by user", user_id, ":\n", movies_watched(user_id))
    print("Genre frequencies for user", user_id, ":\n", give_genre_freq(user_id))
