import json
import jieba
import re
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

def load_data(filename):
    """Load data from a JSON file."""
    with open(filename, 'r', encoding='utf-8') as file:
        return json.load(file)

def tokenize_text(content, stopwordset):
    """Tokenize text using Jieba and remove stop words."""
    tokens = jieba.cut(content, cut_all=False)
    filtered_tokens = [token for token in tokens if token not in stopwordset and re.match(r'^[\u4E00-\u9FFFa-zA-Z]+$', token)]
    return ' '.join(filtered_tokens)

def load_stopwords(filename):
    """Load stopwords from a file."""
    with open(filename, 'r', encoding='utf-8') as file:
        return {line.strip() for line in file}

def preprocess_data(all_data, stopwordset):
    """Preprocess data by tokenizing and filtering."""
    total_content = []
    for article_data in all_data[:50]:
        content = article_data['Content']
        filtered_content = tokenize_text(content, stopwordset)
        total_content.append(filtered_content)
    return total_content

def calculate_tfidf(total_content):
    """Calculate TF-IDF matrix."""
    tfidf_vectorizer = TfidfVectorizer()
    tfidf_matrix = tfidf_vectorizer.fit_transform(total_content)
    feature_names = tfidf_vectorizer.get_feature_names_out()
    tfidf_weight = tfidf_matrix.toarray()
    return tfidf_weight, feature_names

def extract_top_keywords(tfidf_weight, feature_names):
    """Extract top keywords and assign tags."""
    tag_list = []
    for weights in tfidf_weight:
        keywords = {feat: weight for feat, weight in zip(feature_names, weights)}
        sorted_keywords = sorted(keywords.items(), key=lambda x: x[1], reverse=True)
        tags = [keyword[0] for keyword in sorted_keywords[:5]]
        tag_list.append(tags)
    return tag_list

def main():
    # Load data
    all_data = load_data('news.json')

    # Set up Jieba and stopwords
    jieba.set_dictionary('dict.txt-v2.big')
    stopwordset = load_stopwords('stopwords.txt')

    # Preprocess data
    total_content = preprocess_data(all_data, stopwordset)

    # Calculate TF-IDF
    tfidf_weight, feature_names = calculate_tfidf(total_content)

    # Extract top keywords and assign tags
    tag_list = extract_top_keywords(tfidf_weight, feature_names)

    # Update data with tags
    for i, article_data in enumerate(all_data[:50]):
        article_data['Tag'] = tag_list[i]

    # Calculate cosine similarity and print results
    similarity_matrix = cosine_similarity(tfidf_weight, tfidf_weight)
    for row in similarity_matrix:
        print(row)

if __name__ == "__main__":
    main()
