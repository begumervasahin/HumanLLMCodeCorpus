from extract import extractFeatures
from vector import calcCosineSimilarity, calcEuclideanDistance
import os

def match_images(image_path, db, t_compare, use_cosine_similarity, controller):
    # Set the score name based on whether cosine similarity or euclidean distance is used
    controller.score_name = 'Similarity' if use_cosine_similarity else 'Distance'       

    if image_path is not None:
        # Hide buttons, display message while processing, and initialize matched images list
        controller.hide_buttons()
        controller.msg.set('Extracting features from {}...'.format(os.path.basename(image_path)))
        controller.matched_images = []

        # Extract features from the input image
        input_features = extractFeatures(image_path)

        # Compare input image features with each image in the database
        for compared_path in db:
            compared_image = db[compared_path]

            # Calculate similarity or distance based on the chosen method
            if use_cosine_similarity:
                similarity_score = calcCosineSimilarity(input_features, compared_image)
            else:
                similarity_score = calcEuclideanDistance(input_features, compared_image)

            # Store compared image path and similarity/distance score in a dictionary
            compared_data = {'path': compared_path, 'x': similarity_score}
            controller.matched_images.append(compared_data)

        # Sort matched images based on similarity or distance
        controller.matched_images.sort(key=lambda x: -x['x']) if use_cosine_similarity \
                                    else controller.matched_images.sort(key=lambda x: x['x'])

        # Keep only the top matching images
        controller.matched_images = controller.matched_images[:t_compare]

        # Show buttons, draw matches, and display best match
        controller.show_buttons()
        controller.draw_matches()
        best_match = controller.matched_images[0]
        controller.msg.set('Match with: {} ({})'.format(best_match['path'], best_match['x']))
    else:
        controller.msg.set('Open a file first.')
