import os
import numpy as np
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

def read_documents(parent_dir):
    """Reads documents from the specified directory."""
    doc_list = []
    name_list = []
    # Initialize lists to store document content and names
    for file_name in os.listdir(parent_dir):
        full_path = os.path.join(parent_dir, file_name)
        with open(full_path, 'r') as file_ptr:
            # Append the file name to the name list
            name_list.append(file_ptr.name)
            # Read the content of the file
            content = file_ptr.read()
            # Append the content to the document list
            doc_list.append(content)
    return doc_list, name_list

def calculate_similarity(doc_list):
    """Calculates cosine similarity between the query document and all other documents."""
    # Create a TF-IDF vectorizer
    vectorizer = TfidfVectorizer(stop_words='english')
    # Fit the vectorizer on the document list
    model = vectorizer.fit(doc_list)
    # Transform the document list into TF-IDF matrix
    matrix = vectorizer.transform(doc_list)
    # Calculate cosine similarity between the query document and all other documents
    similarity = cosine_similarity(matrix[0:1], matrix)
    return np.ravel(similarity)

def main():
    # Define the query
    query = 'printer'
    # Define the parent directory containing the documents
    parent_dir = "/home/ai55/newsgroups/"
    # Read documents from the specified directory
    doc_list, name_list = read_documents(parent_dir)
    # Append the query to the document list
    doc_list.insert(0, query)
    # Append the query to the name list
    name_list.insert(0, query)
    # Calculate cosine similarity
    similarity = calculate_similarity(doc_list)
    # Find indices of documents with similarity greater than 0
    result = np.argwhere(similarity > 0)
    # Print the similarity matrix
    print(similarity)
    # Print the names of documents with similarity greater than 0
    for idx in result:
        idx = np.asscalar(idx)
        print(name_list[idx])
        print('-------------------------------------------------------------')

if __name__ == "__main__":
    main()
