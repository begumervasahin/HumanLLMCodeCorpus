import sys
import getopt
from image_downloader import jsonloader, downloader
from resize_images import resizer

def main():
    # Initialize the variable to store the number of images to download
    n = None

    try:
        # Parse command-line arguments
        opts, args = getopt.getopt(sys.argv[1:], "n:h", ['help'])
    except getopt.GetoptError:
        # Print usage information in case of incorrect arguments
        print_usage()
        sys.exit(2)

    # Check if no options are provided
    if not opts:
        print("All images will be downloaded")
    else:
        # Loop through the provided options
        for opt, arg in opts:
            if opt in ('-n'):
                # Parse the number of images to download
                n = int(arg)
                # Check if the number of images is greater than 10
                if n <= 10:
                    print('Number of images should be greater than 10')
                    sys.exit(2)
            elif opt in ('-h', '--help'):
                # Print usage information
                print_usage()
                sys.exit()
            else:
                # Print a message for unrecognized arguments
                print("Check your arguments")
                sys.exit(2)

    # Call the image downloader and resizer functions
    downloader(n, jsonloader())
    resizer(n)

def print_usage():
    print("Usage: python <filename.py>")
    print("Optional argument -n for the number of images to be downloaded (should be greater than 10)")

if __name__ == "__main__":
    main()
