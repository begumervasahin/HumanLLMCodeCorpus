import csv
import os
import re
from collections import Counter
import nltk
import pandas as pd
from scipy import spatial

# Download NLTK data if not already downloaded
nltk.download('punkt')

# Set the attachment path where files are located
ATTACHMENT_PATH = "Attachments"

def read_csv(filepath, usecols):
    """Read a CSV file and return the dataset."""
    return pd.read_csv(filepath, usecols=usecols)

def get_matched_summaries():
    """Get matched summaries from CSV files."""
    summaries_file = os.path.join(ATTACHMENT_PATH, "Summaries.csv")
    summaries = read_csv(summaries_file, usecols=["adsh", "full_summary", "full_summary_len", "expense_summary", "expense_len"])
    total_summaries = summaries["adsh"].tolist()

    non_summaries_path = os.path.join(ATTACHMENT_PATH, "Non-Summaries")
    non_summaries_files = [os.path.splitext(os.path.basename(file))[0] for file in os.listdir(non_summaries_path)]

    mapping_file = os.path.join(ATTACHMENT_PATH, "Mapping.csv")
    mapping = read_csv(mapping_file, usecols=["accession_xbrl", "accession_not_xbrl"])

    new_mapping = mapping[(mapping['accession_not_xbrl'].isin(non_summaries_files)) &
                          (mapping['accession_xbrl'].isin(total_summaries))]
    return new_mapping, mapping, summaries

def tokenize_paragraph(paragraph):
    """Tokenize a paragraph into words and lines."""
    words = []
    lines = []
    sentences = re.split(r'(?<!\w\.\w.)(?<![A-Z][a-z]\.)(?<=\.|\?)\s', paragraph)
    for sentence in sentences:
        words.append(remove_articles_prepositions(sentence.strip()))
        lines.append(sentence.strip())
    return words, lines

def remove_articles_prepositions(sentence):
    """Remove articles and prepositions from a sentence."""
    tokens = nltk.word_tokenize(sentence)
    tokens = [re.sub(r'[^A-Za-z0-9]+', '', token) for token in tokens if token.lower() not in ['a', 'an', 'the']]
    tagged = nltk.pos_tag(tokens)
    stemmer = nltk.stem.SnowballStemmer('english')
    clean_words = [stemmer.stem(word) for word, pos in tagged if pos != 'IN']
    return clean_words

def read_text_file(filepath):
    """Read a text file and tokenize sentences."""
    with open(filepath, "r", encoding="utf8", errors='ignore') as file:
        content = file.read()
    sentences = nltk.sent_tokenize(content)
    sentences = [re.sub('\s+', ' ', sentence) for sentence in sentences if sentence[-1] in ['.', '?']]
    return sentences

def calculate_similarity(summary_word_vectors, non_summary_word_vectors):
    """Calculate cosine similarity between summary and non-summary word vectors."""
    total_scores = []
    for summary_vector in summary_word_vectors:
        cosine_scores = []
        for non_summary_vector in non_summary_word_vectors:
            combined_vector = list(set(summary_vector + non_summary_vector))
            summary_counts = count_words(summary_vector, combined_vector)
            non_summary_counts = count_words(non_summary_vector, combined_vector)
            cosine_scores.append(get_cosine_similarity(summary_counts, non_summary_counts))
        total_scores.append(max(cosine_scores))
    average_similarity = sum(total_scores) / len(total_scores)
    return average_similarity, total_scores

def count_words(word_list, target_list):
    """Count word occurrences in a list."""
    word_counts = Counter(word_list)
    for word in target_list:
        if word not in word_counts:
            word_counts[word] = 0
    return [word_counts[word] for word in target_list]

def get_cosine_similarity(vector1, vector2):
    """Calculate cosine similarity between two vectors."""
    cosine_value = 1 - spatial.distance.cosine(vector1, vector2)
    return 1 if cosine_value >= 0.90 else 0

def add_rows_to_output_csv(output_rows):
    """Add processed rows to output CSV."""
    summaries_file = os.path.join(ATTACHMENT_PATH, "Summaries.csv")
    with open(summaries_file, 'r') as input_file:
        with open('Output/summaries_output.csv', 'w') as output_file:
            writer = csv.writer(output_file, lineterminator='\n')
            reader = csv.reader(input_file)
            header = next(reader)
            header.extend(['number_of_sentences_in_summary', 'number_of_sentences_in_non_summary', 'similarity'])
            writer.writerow(header)
            for row in reader:
                for output_row in output_rows:
                    if row[0] == output_row["summary_identifier"]:
                        row.extend([output_row["number_of_sentences_in_summary"], 
                                    output_row["number_of_sentences_in_non_summary"], 
                                    output_row["similarity"]])
                        writer.writerow(row)

def main():
    output_rows = []
    new_mapping, mapping, summaries = get_matched_summaries()
    for index, row in new_mapping.iterrows():
        summary_id, non_summary_id = row['accession_xbrl'], row['accession_not_xbrl']
        full_summary = summaries.loc[summaries["adsh"] == summary_id, 'full_summary'].iloc[0]
        summary_word_vectors, summary_lines = tokenize_paragraph(full_summary)
        num_summary_lines = len(summary_word_vectors)
        non_summary_file = os.path.join(ATTACHMENT_PATH, "Non-Summaries", non_summary_id + '.txt')
        non_summary_lines = read_text_file(non_summary_file)
        num_non_summary_lines = len(non_summary_lines)
        non_summary_word_vectors = [remove_articles_prepositions(line) for line in non_summary_lines]
        similarity_score, line_wise_similarity = calculate_similarity(summary_word_vectors, non_summary_word_vectors)
        output_rows.append({
            "summary_identifier": summary_id,
            "similarity": similarity_score,
            "number_of_sentences_in_summary": num_summary_lines,
            "number_of_sentences_in_non_summary": num_non_summary_lines
        })
    add_rows_to_output_csv(output_rows)

if __name__ == "__main__":
    main()
