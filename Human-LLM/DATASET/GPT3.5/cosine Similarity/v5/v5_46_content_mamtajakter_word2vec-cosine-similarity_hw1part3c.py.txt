import gensim
from nltk.corpus import brown

# Load the Word2Vec model trained on the Brown corpus
model = gensim.models.Word2Vec(brown.sents())

# Define the words to find similarities for
words_to_compare = [
    {"word": "rebellion", "description": "Rebellion"},
    {"word": "slave", "description": "Slave"}
]

# Find top 10 similar words and calculate cosine similarities for each word
results = []
for word_info in words_to_compare:
    word = word_info["word"]
    description = word_info["description"]

    # Find top 10 similar words
    similar_words = [similar_word for similar_word, _ in model.most_similar(positive=[word], topn=10)]

    # Calculate cosine similarities for the current word and similar words
    cosine_similarities = []
    for similar_word in similar_words:
        similarity = model.similarity(word, similar_word)
        cosine_similarities.append(f"{description} - {similar_word}: {similarity}")

    # Append results
    results.append({"word": description, "similar_words": similar_words, "cosine_similarities": cosine_similarities})

# Print results
for result in results:
    print(f"Top ten similar words for '{result['word']}':")
    print(result['similar_words'])

    print("\nCosine similarities:")
    print("\n".join(result['cosine_similarities']))
    print()
