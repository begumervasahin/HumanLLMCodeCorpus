import math
from nltk.tokenize import RegexpTokenizer
from nltk.stem import PorterStemmer

def tokenize_normalize(raw_string):
    """
    Tokenizes and normalizes the input string.

    Args:
        raw_string (str): The input string to be tokenized and normalized.

    Returns:
        list: A list of normalized tokens.
    """
    tokenizer = RegexpTokenizer(r'\w+')
    tokenized_text = tokenizer.tokenize(str(raw_string))
    normalized_text = [word.lower() for word in tokenized_text]
    return normalized_text

def stem_words(normalized_text):
    """
    Stems the words in the input list.

    Args:
        normalized_text (list): A list of normalized tokens.

    Returns:
        list: A list of stemmed tokens.
    """
    stemmer = PorterStemmer()
    stemmed_text = [stemmer.stem(word) for word in normalized_text]
    return stemmed_text

def create_tf_dict_doc(processed_text, filename, stats_dict):
    """
    Creates a term frequency dictionary for a document.

    Args:
        processed_text (list): A list of processed tokens from the document.
        filename (str): The name of the document.
        stats_dict (dict): The dictionary to store the term frequency.

    Returns:
        dict: The updated term frequency dictionary.
    """
    for root in processed_text:
        if root not in stats_dict:
            stats_dict[root] = {}
            stats_dict[root][filename] = 1
        else:
            stats_dict[root][filename] = stats_dict[root].get(filename, 0) + 1
    return stats_dict

def create_tf_dict_query(processed_query):
    """
    Creates a term frequency dictionary for a query.

    Args:
        processed_query (list): A list of processed tokens from the query.

    Returns:
        dict: The term frequency dictionary for the query.
    """
    query_dict = {}
    for root in processed_query:
        query_dict[root] = query_dict.get(root, 0) + 1
    return query_dict

def find_tfidf_doc(stats_dict, num_docs):
    """
    Calculates TF-IDF scores for documents.

    Args:
        stats_dict (dict): The term frequency dictionary for documents.
        num_docs (int): The total number of documents.

    Returns:
        tuple: A tuple containing the updated term frequency dictionary and IDF scores.
    """
    idf_dict = {}
    for word in stats_dict:
        idf_dict[word] = len(stats_dict[word])
    for word in stats_dict:
        for doc in stats_dict[word]:
            stats_dict[word][doc] = (1 + math.log(stats_dict[word][doc])) * math.log(num_docs / idf_dict[word])
    return stats_dict, idf_dict

def find_tfidf_query(query_dict, idf_dict, num_docs):
    """
    Calculates TF-IDF scores for a query.

    Args:
        query_dict (dict): The term frequency dictionary for the query.
        idf_dict (dict): The IDF scores for terms in the document corpus.
        num_docs (int): The total number of documents.

    Returns:
        dict: The TF-IDF scores for the query.
    """
    for word in query_dict:
        query_dict[word] = (1 + math.log(query_dict[word])) * math.log(num_docs / idf_dict.get(word, 1))
    return query_dict
