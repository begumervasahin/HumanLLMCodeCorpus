import numpy as np
from scipy.spatial.distance import cosine
import pickle

class CosineSimilarity:
    def __init__(self):
        # Load matrices needed for calculating similarity
        self.tfidf, self.terms, self.sigma, self.VT = self._load_matrices()

    def calculate_term_similarity(self, term1, term2):
        # Check if both terms are present in the corpus
        if term1 in self.terms and term2 in self.terms:
            # Get the indices of the terms
            index0 = self.terms.index(term1)
            index1 = self.terms.index(term2)
        else:
            print("One or both terms not found in the corpus.")
            return 0

        # Get the term vectors for the given terms
        term_vector0 = np.transpose(self.tfidf[index0])
        term_vector1 = np.transpose(self.tfidf[index1])

        # Calculate the cosine similarity between the term vectors
        cosine_distance = self._calculate_cosine_similarity(term_vector0, term_vector1)
        similarity = 1 - cosine_distance

        # Print similarity score
        print(f"Similarity between '{term1}' and '{term2}': {similarity}")

        return similarity

    def _create_diagonal_matrix(self):
        # Create a diagonal matrix using sigma values
        diagonal_matrix = np.zeros((len(self.sigma), len(self.sigma)), int)
        np.fill_diagonal(diagonal_matrix, self.sigma)
        return diagonal_matrix

    def _calculate_cosine_similarity(self, term_vector0, term_vector1):
        # Create diagonal matrix using sigma
        diagonal_matrix = self._create_diagonal_matrix()
        sigma_inv = np.linalg.inv(diagonal_matrix)
        # Transform term vectors
        transformed_vector0 = np.matmul(np.matmul(sigma_inv, self.VT), term_vector0)
        transformed_vector1 = np.matmul(np.matmul(sigma_inv, self.VT), term_vector1)
        # Calculate cosine distance
        cosine_distance = cosine(transformed_vector0, transformed_vector1)
        return cosine_distance

    @staticmethod
    def _load_matrices():
        # Load matrices from pickle files
        with open('./pickle_files/sigma', 'rb') as f:
            sigma = pickle.load(f)
        with open('./pickle_files/vt', 'rb') as f:
            VT = pickle.load(f)
        with open('./pickle_files/tfidf', 'rb') as f:
            tfidf = pickle.load(f)
        with open('./pickle_files/terms', 'rb') as f:
            terms = pickle.load(f)
        return tfidf, terms, sigma, VT

if __name__ == "__main__":
    # Create an instance of CosineSimilarity class
    cosine_similarity = CosineSimilarity()
    # Take user input for two terms
    term1 = input("Enter the first term: ")
    term2 = input("Enter the second term: ")
    # Calculate and print similarity between the terms
    cosine_similarity.calculate_term_similarity(term1, term2)
