from nltk.tokenize import word_tokenize
import gensim

def load_word2vec_model(model_path):
    """
    Load the Word2Vec model from the specified path.

    Args:
        model_path (str): The path to the Word2Vec model file.

    Returns:
        gensim.models.Word2Vec: The loaded Word2Vec model.
    """
    return gensim.models.Word2Vec.load(model_path, mmap='r')

def calculate_similarity(seq1, seq2, model, tail=100, head=10):
    """
    Calculate the similarity score between two sequences using a Word2Vec model.

    Args:
        seq1 (str): The first sequence.
        seq2 (str): The second sequence.
        model (gensim.models.Word2Vec): The Word2Vec model.
        tail (int): The number of words to consider from the end of seq1.
        head (int): The number of words to consider from the beginning of seq2.

    Returns:
        float: The similarity score between the sequences.
    """
    seq1_words = word_tokenize(seq1.strip().lower())[-tail:]
    seq2_words = word_tokenize(seq2.strip().lower())[:head]
    return calculate_average_max_similarity(seq1_words, seq2_words, model)

def calculate_average_max_similarity(words1, words2, model):
    """
    Calculate the average maximum similarity between two lists of words.

    Args:
        words1 (list of str): The list of words from the first sequence.
        words2 (list of str): The list of words from the second sequence.
        model (gensim.models.Word2Vec): The Word2Vec model.

    Returns:
        float: The average maximum similarity score.
    """
    max_similarities = []
    for word1 in words1:
        max_similarity = 0
        for word2 in words2:
            if word1 == word2:
                max_similarity = 1
            elif word1 in model.vocab and word2 in model.vocab:
                similarity = model.similarity(word1, word2)
                max_similarity = max(max_similarity, similarity)
        if max_similarity != 0:
            max_similarities.append(max_similarity)
    if max_similarities:
        return sum(max_similarities) / len(max_similarities)
    else:
        return 0
