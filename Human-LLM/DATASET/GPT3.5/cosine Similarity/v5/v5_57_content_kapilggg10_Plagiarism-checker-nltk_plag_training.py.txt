from nltk.tokenize import word_tokenize
import os
import string
from nltk.corpus import stopwords
from nltk.stem.porter import PorterStemmer
from sklearn.feature_extraction.text import TfidfVectorizer

def tokenize_text(text):
    """Tokenize the input text."""
    tokens = word_tokenize(text)
    return tokens

def apply_stemming(tokens, stemmer):
    """Apply stemming to the input tokens using the provided stemmer."""
    stemmed_tokens = [stemmer.stem(token) for token in tokens]
    return stemmed_tokens

def preprocess_text(text):
    """Preprocess the input text by tokenizing and stemming."""
    tokens = tokenize_text(text)
    stemmer = PorterStemmer()
    stemmed_tokens = apply_stemming(tokens, stemmer)
    return stemmed_tokens

def train_tfidf_model():
    """Train TF-IDF model on text files in a specified directory."""
    file_list = []  # List to store file names
    token_dictionary = {}  # Dictionary to store preprocessed text for each file        

    # Define the path to the directory containing text files
    directory_path = 'C:\\Users\\Kapil Goyal\\Desktop\\nlp_assignment\\files'

    # Iterate over files in the directory
    for subdir, _, files in os.walk(directory_path):
        for file_name in files:
            file_path = os.path.join(subdir, file_name)
            with open(file_path, 'r') as file:
                # Read the file content
                content = file.read()

                # Preprocess the text
                preprocessed_text = preprocess_text(content)

                # Store preprocessed text in the token dictionary
                token_dictionary[file_name] = ' '.join(preprocessed_text)

                # Append file name to the file list
                file_list.append(file_name)

    # Initialize and fit the TF-IDF vectorizer
    tfidf_vectorizer = TfidfVectorizer(stop_words='english', tokenizer=preprocess_text)
    tfs_matrix = tfidf_vectorizer.fit_transform(token_dictionary.values())

    return tfidf_vectorizer, tfs_matrix, file_list

# Call the train_tfidf_model function
tfidf_model, tfs_matrix, file_list = train_tfidf_model()
