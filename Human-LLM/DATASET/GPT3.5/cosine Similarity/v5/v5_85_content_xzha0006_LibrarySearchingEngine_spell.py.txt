import re
from collections import Counter

# Read the text file and generate a list of words with their frequencies
def generate_word_freq(file_path):
    """Generate word frequencies from a text file."""
    with open(file_path, 'r') as file:
        text = file.read().lower()
        word_list = re.findall(r'\w+', text)
        return Counter(word_list)

# Calculate the probability of a word based on its frequency
def calculate_probability(word, word_freq):
    """Calculate the probability of a word."""
    total_words = sum(word_freq.values())
    return word_freq[word] / total_words if total_words else 0

# Correct a misspelled word by choosing the most probable correction
def correct(word, word_freq):
    """Correct a misspelled word."""
    return max(generate_candidates(word, word_freq), key=lambda w: calculate_probability(w, word_freq))

# Generate possible corrections for a misspelled word
def generate_candidates(word, word_freq):
    """Generate possible corrections for a misspelled word."""
    if word in word_freq:
        return [word]
    elif levenshtein_edits := edits_levenshtein(word):
        return known(levenshtein_edits, word_freq)
    elif levenshtein2_edits := edits_levenshtein2(word):
        return known(levenshtein2_edits, word_freq)
    else:
        return [word]

# Filter out known words from a list of words
def known(words, word_freq):
    """Filter out known words from a list of words."""
    return set(w for w in words if w in word_freq)

# Generate edits with one edit distance from the input word
def edits_levenshtein(word):
    """Generate edits with one edit distance from the input word."""
    alphabet = 'abcdefghijklmnopqrstuvwxyz'
    splits = [(word[:i], word[i:]) for i in range(len(word) + 1)]
    deletion = [left + right[1:] for left, right in splits if right]
    transposition = [left + right[1] + right[0] + right[2:] for left, right in splits if len(right) > 1]
    replacement = [left + char + right[1:] for left, right in splits if right for char in alphabet]
    insertion = [left + char + right for left, right in splits for char in alphabet]
    return set(deletion + transposition + replacement + insertion)

# Generate edits with two edit distances from the input word
def edits_levenshtein2(word):
    """Generate edits with two edit distances from the input word."""
    return (j for i in edits_levenshtein(word) for j in edits_levenshtein(i))

# Example usage
def main():
    file_path = 'LargeTextFile.txt'
    word_freq = generate_word_freq(file_path)
    
    input_word = input("Enter a word to correct: ").lower()
    corrected_word = correct(input_word, word_freq)
    print("Corrected word:", corrected_word)

if __name__ == "__main__":
    main()
