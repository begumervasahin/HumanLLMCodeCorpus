import distsim
from collections import defaultdict

def load_word_vectors():
    """Load word vectors."""
    return distsim.load_word2vec("nyt_word2vec.4k")

def process_word_test_file(word_to_vec_dict):
    """Process the word-test file and calculate task accuracies."""
    task_accuracies = defaultdict(list)
    
    with open("word-test.v3.txt", "r") as file:
        for line in file:
            line = line.strip('\n')
            
            # Skip comment lines
            if line.startswith('//'):
                continue
            
            # Parse category lines
            if line.startswith(":"):
                category = line[2:]
            else:
                words = line.split()
                w1_vec = word_to_vec_dict[words[0]]
                w2_vec = word_to_vec_dict[words[1]]
                w4_vec = word_to_vec_dict[words[3]]
                
                # Calculate the vector representation for the fourth word
                fourth_word_vec = w1_vec - w2_vec + w4_vec
                
                # Find the nearest word to the calculated vector, excluding words in the set
                result = distsim.show_nearest(word_to_vec_dict, fourth_word_vec, set([words[0], words[1], words[3]]), distsim.cossim_dense)
                
                # Check if the correct word is in the result
                match = False
                for i, (word, _) in enumerate(result, 1):
                    if word == words[2]:
                        match = True
                        task_accuracies[category].append(i)
                        break
                
                # If the correct word is not found, append 0
                if not match:
                    task_accuracies[category].append(0)
                    
    return task_accuracies

def calculate_accuracies(task_accuracies):
    """Calculate and print task accuracies."""
    for key, value in task_accuracies.items():
        total = len(value)
        top1 = sum(1 for pos in value if pos == 1)
        top5 = sum(1 for pos in value if pos <= 5)
        top10 = sum(1 for pos in value if pos <= 10)
        
        # Print category and corresponding accuracies
        print(f"{key}:     {round(top1 / total, 2)}    {round(top5 / total, 2)}    {round(top10 / total, 2)}")

def main():
    # Load word vectors
    word_to_vec_dict = load_word_vectors()

    # Process word-test file and calculate accuracies
    task_accuracies = process_word_test_file(word_to_vec_dict)

    # Calculate and print accuracies
    calculate_accuracies(task_accuracies)

if __name__ == "__main__":
    main()
