import settings
import preprocess
import compute
from collections import OrderedDict
import time

# Global variable to store URLs
url_list = []

def calculate_tf_idf(query):
    # Collect tokenized data and URLs from the web dictionary
    token_collection = [settings.web_dict[key] for key in settings.web_dict]
    url_list.extend(settings.web_dict.keys())

    # Preprocess the query
    queries = preprocess.removePunctuation(query.lower())
    queries = preprocess.removeDigits(queries)
    token_queries = preprocess.tokenizeData(queries)
    token_queries = preprocess.removeStopWords(token_queries)
    token_queries = preprocess.performStemming(token_queries)
    token_queries = preprocess.removeStopWords(token_queries)

    # Build inverted index, vocabulary, and document set
    inverted_index, vocabulary, doc_set = compute.buildInvertedIndex(token_collection)

    # Build vector space model for documents
    document_model = compute.buildVectorSpaceModel(token_collection, inverted_index, vocabulary)

    # Transform query into vector space model
    query_model = compute.transformIntoVectorSpaceModel(token_queries, inverted_index, vocabulary, len(token_collection))

    # Calculate cosine similarity between documents and query
    cosine_similarity, cosine_similarity_sorted = compute.getCosineSimilarity(document_model, query_model, vocabulary, token_queries, doc_set)

    return cosine_similarity, cosine_similarity_sorted

def write_results(cosine_similarity, cosine_similarity_sorted, page_rank):
    top_200_dict = OrderedDict()
    top_200_ordered = OrderedDict()

    # Populate top 200 dictionary with URLs and corresponding indices
    for index in range(len(cosine_similarity_sorted)):
        if cosine_similarity[cosine_similarity_sorted[index]] <= 0 or len(top_200_dict) >= 200:
            break
        else:
            top_200_dict[url_list[cosine_similarity_sorted[index]]] = cosine_similarity_sorted[index]

    # Order URLs based on page rank
    for key in page_rank:
        if key in top_200_dict:
            top_200_ordered[key] = top_200_dict[key]

    # Include remaining URLs not covered by page rank
    if len(top_200_ordered) < len(top_200_dict):
        for key in top_200_dict:
            if key not in top_200_ordered:
                top_200_ordered[key] = top_200_dict[key]

    # Write results to files
    with open("./results/top_200_pages", 'w', encoding="utf-8") as fobj:
        for key in top_200_ordered:
            fobj.write(key + "\n")

    with open("./results/top_200_SimilarityScore", 'w', encoding="utf-8") as fobj:
        for key in top_200_ordered:
            fobj.write(str(cosine_similarity[top_200_ordered[key]]) + "\n")

    with open("./results/top_200_PageRankScore", 'w', encoding="utf-8") as fobj:
        for key in top_200_ordered:
            fobj.write(str(page_rank[key]) + "\n")

def main(query):
    settings.init()
    start_time = time.time()

    # Import crawled data
    preprocess.import_crawled_data(3000)

    # Calculate TF-IDF
    cosine_similarity, cosine_similarity_sorted = calculate_tf_idf(query)

    # Import page rank data
    page_rank = preprocess.import_page_rank()

    # Write results
    write_results(cosine_similarity, cosine_similarity_sorted, page_rank)

    end_time = time.time()
    execution_time = end_time - start_time
    print("Execution Time:", execution_time)
    print("END")

if __name__ == '__main__':
    import sys
    main(sys.argv[1])
