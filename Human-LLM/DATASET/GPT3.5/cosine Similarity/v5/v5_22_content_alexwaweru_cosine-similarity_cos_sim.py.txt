import numpy as np

def cosine_similarity(vector_a, vector_b):
    """Calculate the cosine similarity between two vectors."""
    dot_product = np.dot(vector_a, vector_b)
    norm_a = np.linalg.norm(vector_a)
    norm_b = np.linalg.norm(vector_b)
    return dot_product / (norm_a * norm_b)

def vectorizer(*sentences):
    """Convert sentences into binary vectors based on a template of unique words."""    
    combined_sentence = ' '.join(sentences)
    sentences_split = [sentence.split(" ") for sentence in sentences]
    words = combined_sentence.split(" ")
    template = list(set(words))
    binary_vectors = [[] for _ in range(len(sentences))]

    for i, sentence_words in enumerate(sentences_split):
        for word in template:
            binary_vectors[i].append(int(word in sentence_words))

    return binary_vectors

def fully_vectorized_trial():
    # Define example sentences
    sentences = [
        "How many teams participated in the first world cup",
        "Who holds the record for top scorer in a single World Cup",
        "Should I sell my car myself or trade it in",
        "How important is car maintenance",
        "How much does AWS SAM cost to use",
        "Which languages does AWS SAM support"
    ]

    # Vectorize example sentences and calculate cosine similarity
    sentence_vectors = vectorizer(*sentences)

    for i in range(6):
        print('\n')
        for j in range(6):
            similarity = cosine_similarity(sentence_vectors[i], sentence_vectors[j])
            print('{0:.4f}'.format(similarity), end='\t', flush=True)
    print('\n')

# Execute the fully vectorized cosine similarity trial
fully_vectorized_trial()
