import nltk
import string
import warnings
from sklearn.feature_extraction.text import TfidfVectorizer

class DocumentSimilarity:
    def __init__(self):
        # Initialize stemmer and punctuation removal map
        self.stemmer = nltk.stem.porter.PorterStemmer()
        self.punctuation_map = dict((ord(char), None) for char in string.punctuation)

    def stem_tokens(self, tokens):
        """Stem tokens using Porter stemmer."""
        return [self.stemmer.stem(token) for token in tokens]

    def normalize(self, text):
        """Normalize text by tokenizing, converting to lowercase, and removing punctuation."""
        tokens = nltk.word_tokenize(text.lower().translate(self.punctuation_map))
        return self.stem_tokens(tokens)

    def cosine_similarity(self, text1, text2):
        """Compute cosine similarity between two text documents."""
        vectorizer = TfidfVectorizer(tokenizer=self.normalize, stop_words='english')
        tfidf = vectorizer.fit_transform([text1, text2])
        return ((tfidf * tfidf.T).A)[0, 1]

# Suppress warnings
warnings.filterwarnings("ignore")

# Download NLTK resources
nltk.download('punkt')

# Instantiate the DocumentSimilarity class
document_similarity = DocumentSimilarity()

# Test the cosine similarity method with sample texts
print("Text1:", document_similarity.cosine_similarity('a little bird', 'a little bird'))
print("Text2:", document_similarity.cosine_similarity('a little bird', 'a little bird chirps'))
print("Text3:", document_similarity.cosine_similarity('a little bird', 'a big dog barks'))
