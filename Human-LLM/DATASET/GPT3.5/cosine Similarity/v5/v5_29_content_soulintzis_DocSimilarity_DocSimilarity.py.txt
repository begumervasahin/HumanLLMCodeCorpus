import string
from numpy import linalg as la
import numpy as np
import math as m
from collections import OrderedDict

def calculate_binomial_coefficient(n):
    binom = int(m.factorial(n) / (m.factorial(2) * m.factorial(n - 2)))
    k = int(input(f"Enter K (must be smaller than {binom}): "))
    if n == 2:
        return 1
    elif k < binom:
        return k
    else:
        print(f"K must be smaller than {binom}. Default K = 1.")
        return 1

def get_document_names(num_docs):
    file_names = []
    for i in range(num_docs):
        name = input(f'Enter the name of document {i + 1}: ')
        if not name.endswith('.txt'):
            print('You must insert documents with a ".txt" extension!')
            exit()
        file_names.append(name)
    return file_names

def create_word_dictionary(file_names):
    word_dictionary = set()
    for file_name in file_names:
        with open("Documents/" + file_name, 'r') as f:
            for line in f:
                for word in line.split():
                    word = word.translate(str.maketrans('', '', string.punctuation)).lower()
                    word_dictionary.add(word)
    return sorted(word_dictionary)

def count_word_occurrences(file_names, word_dictionary):
    doc_count = []
    for file_name in file_names:
        count = [0] * len(word_dictionary)
        with open("Documents/" + file_name, 'r') as f:
            for line in f:
                for word in line.split():
                    word = word.translate(str.maketrans('', '', string.punctuation)).lower()
                    position = word_dictionary.index(word)
                    count[position] += 1
        doc_count.append(count)
    return doc_count

def calculate_similarity(doc_count, file_names):
    similarity = {}
    num_docs = len(file_names)
    for i in range(num_docs):
        for j in range(i + 1, num_docs):
            doc1 = doc_count[i]
            doc2 = doc_count[j]
            norm1 = la.norm(doc1, ord=2)
            norm2 = la.norm(doc2, ord=2)
            dot_product = np.dot(doc1, doc2)
            result = dot_product / (norm1 * norm2)
            pair_of_docs = file_names[i] + " and " + file_names[j]
            similarity[pair_of_docs] = result
    return similarity

def print_top_k_similarity(similarity, k):
    sorted_similarity = OrderedDict(sorted(similarity.items(), key=lambda x: x[1], reverse=True)[:k])
    print("\nTop K Similar Documents listed below:")
    for key, value in sorted_similarity.items():
        print(f"{key}: {value:.2f}, {value * 100:.2f}%")

if __name__ == "__main__":
    num_docs = int(input("How many documents do you want to insert: "))
    while num_docs < 2:
        num_docs = int(input("Please enter more than 1 document: "))

    file_names = get_document_names(num_docs)
    k = calculate_binomial_coefficient(num_docs)
    word_dictionary = create_word_dictionary(file_names)
    doc_count = count_word_occurrences(file_names, word_dictionary)
    similarity = calculate_similarity(doc_count, file_names)
    print_top_k_similarity(similarity, k)
