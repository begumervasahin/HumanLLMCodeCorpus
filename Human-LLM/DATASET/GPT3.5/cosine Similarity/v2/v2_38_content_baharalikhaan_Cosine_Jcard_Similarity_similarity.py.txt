from nltk.corpus import stopwords
from nltk.tokenize import word_tokenize
import numpy as np
import nltk
import csv
import os

# Function to calculate Jaccard similarity between two sets
def jaccard_similarity(x, y):
    intersection_cardinality = len(set.intersection(*[set(x), set(y)]))
    union_cardinality = len(set.union(*[set(x), set(y)]))
    return intersection_cardinality / float(union_cardinality)

# Function to calculate cosine similarity between two vectors
def cos_sim(a, b):
    dot_product = np.dot(a, b)
    norm_a = np.linalg.norm(a)
    norm_b = np.linalg.norm(b)
    return dot_product / (norm_a * norm_b)

# Function to calculate similarity between two dictionaries
def get_similarity(dict1, dict2):
    all_words_list = list(set(dict1.keys()) | set(dict2.keys()))
    all_words_list_size = len(all_words_list)
    v1 = np.zeros(all_words_list_size, dtype=np.int)
    v2 = np.zeros(all_words_list_size, dtype=np.int)
    for i, key in enumerate(all_words_list):
        v1[i] = dict1.get(key, 0)
        v2[i] = dict2.get(key, 0)
    return cos_sim(v1, v2)

# Function to process a file and generate a word frequency dictionary
def process(file_path):
    raw = open(file_path).read()
    tokens = word_tokenize(raw)
    words = [w.lower() for w in tokens]
    porter = nltk.PorterStemmer()
    stemmed_tokens = [porter.stem(t) for t in words]
    stop_words = set(stopwords.words('english'))
    filtered_tokens = [w for w in stemmed_tokens if w not in stop_words]
    count = nltk.defaultdict(int)
    for word in filtered_tokens:
        count[word] += 1
    return count

# Create CSV files for similarity results
with open('similarity_cosine.csv', mode='w', newline='') as cosine_file:
    cosine_writer = csv.writer(cosine_file)
    cosine_writer.writerow(['File 1', 'File 2', 'Similarity'])
    with open('similarity_jcard.csv', mode='w', newline='') as jaccard_file:
        jaccard_writer = csv.writer(jaccard_file)
        jaccard_writer.writerow(['File 1', 'File 2', 'Similarity'])

        # Process files and calculate similarities
        for filex in os.listdir('files'):
            dict1 = process(os.path.join('files', filex))
            for filey in os.listdir('files'):
                dict2 = process(os.path.join('files', filey))
                sim_cosine = get_similarity(dict1, dict2)
                sim_jaccard = jaccard_similarity(dict1.keys(), dict2.keys())
                print(f'Similarity between {filex} & {filey} (Cosine): {sim_cosine}')   
                print(f'Similarity between {filex} & {filey} (Jaccard): {sim_jaccard}')   
                cosine_writer.writerow([filex, filey, sim_cosine])
                jaccard_writer.writerow([filex, filey, sim_jaccard])
