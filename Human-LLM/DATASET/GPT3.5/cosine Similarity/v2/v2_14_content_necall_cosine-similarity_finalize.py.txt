import json
from collections import defaultdict

def read_counts(file_path):
    """
    Read counts from a file and return as a list of tuples.
    Each tuple contains word, count, and context counts.
    """
    with open(file_path, 'r') as file:
        return [(w, int(c), json.loads(cc)) for w, c, cc in (line.split("\t") for line in file)]

def all_words(tw_words, nyt_words):
    """
    Yield all words from two lists of words until the minimum length is reached.
    """
    for i in range(min(len(tw_words), len(nyt_words))):
        yield tw_words[i][0]
        yield nyt_words[i][0]

# Read counts from files
tw = read_counts("twcounts")
nyt = read_counts("nytcounts")

# Sort words by count
tw_words = sorted([(w, c) for w, c, _ in tw if c > 100], key=lambda x: -x[1])
nyt_words = sorted([(w, c) for w, c, _ in nyt if c > 100], key=lambda x: -x[1])

# Create dictionaries for counts
tw_counts = {w: c for w, c, _ in tw}
nyt_counts = {w: c for w, c, _ in nyt}

# Extract a vocabulary of 4000 words
vocab = set()
for w in all_words(tw_words, nyt_words):
    vocab.add(w)
    if len(vocab) == 4000:
        break

# Write the vocabulary to a file
with open("vocab", 'w') as f:
    counts = [(w, nyt_counts.get(w, 0)) for w in vocab]
    counts.sort(key=lambda x: -x[1])
    counts = [(w, c) for w, c in counts if c > 0]
    for w, c in counts:
        print(w, c, file=f)

def filter_counts(infile, outfile, vocab):
    """
    Filter counts in a file based on a given vocabulary and write to another file.
    """
    print("%s -> %s" % (infile, outfile))
    ctx_numwords = defaultdict(int)
    w_cc = {}
    w_count = {}
    with open(infile, 'r') as inf:
        for line in inf:
            w, c, cc = line.split("\t")
            if w not in vocab:
                continue
            cc = json.loads(cc)
            for ctx in cc:
                ctx_numwords[ctx] += 1
            w_cc[w] = cc
            w_count[w] = c

    with open(outfile, 'w') as out:
        for w, cc in w_cc.items():
            for ctx in list(cc.keys()):
                if ctx_numwords[ctx] <= 1:
                    del cc[ctx]
            print("%s\t%s\t%s" % (w, w_count[w], json.dumps(w_cc[w])), file=out)        

# Filter counts for trimmed files and files with specific words
filter_counts("twcounts", "twcounts.trimmed", vocab)
filter_counts("nytcounts", "nytcounts.trimmed", vocab)
filter_counts("twcounts", "twcounts.university_cat_dog", {"university", "cat", "dog"})  
filter_counts("nytcounts", "nytcounts.university_cat_dog", {"university", "cat", "dog"})
