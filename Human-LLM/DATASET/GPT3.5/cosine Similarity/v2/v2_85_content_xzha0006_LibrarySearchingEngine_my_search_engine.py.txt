import os
import math
import numpy as np
from stemming import PorterStemmer
import time
from utils import *
from mpi4py import MPI

# Initialize MPI
comm = MPI.COMM_WORLD
mpi_size = comm.Get_size()
rank = comm.Get_rank()

def command_index(index_dir, file_path):
    # Load files and distribute workload across processes
    file_content_list, file_names = load_files(file_path)
    file_size = len(file_content_list)
    interval = math.ceil(file_size / mpi_size)
    start_idx = rank * interval
    end_idx = min((rank + 1) * interval, file_size)

    # Process files in the assigned range
    tf_dict_list = []
    for i in range(start_idx, end_idx):
        term_list = tokenize(file_content_list[i])
        tf_dict = calculate_term_frequency(term_list)
        tf_dict_list.append(tf_dict)
        print(rank, i, len(tf_dict))

    # Send processed data to rank 0
    if rank != 0:
        comm.send(tf_dict_list, dest=0, tag=1)

    # Rank 0 aggregates data and writes index
    if rank == 0:
        for i in range(1, mpi_size):
            tf_dict_list.extend(comm.recv(source=i, tag=1))

        # Calculate document frequency
        df_dict = calculate_document_frequency(tf_dict_list)

        # Write index to file
        write_index_to_file(index_dir, file_names, tf_dict_list, df_dict)

def calculate_document_frequency(tf_dict_list):
    df_dict = {}
    for tf_dict in tf_dict_list:
        for term in tf_dict.keys():
            df_dict[term] = df_dict.get(term, 0) + 1
    return df_dict

def write_index_to_file(index_dir, file_names, tf_dict_list, df_dict):
    with open(os.path.join(index_dir, "index_parallel.txt"), 'w') as f:
        tf_vector_list = [[] for _ in range(len(file_names))]
        for term in df_dict.keys():
            idf = math.log(len(tf_dict_list) / df_dict[term])
            tf_record = ""
            for i, tf_dict in enumerate(tf_dict_list):
                tf = tf_dict.get(term, 0)
                if tf != 0:
                    tf_vector_list[i].append(tf * idf)
                    tf_record += ",%s,%d" % (file_names[i], tf)
            record = term + tf_record + "," + ("%.3f" % idf) + "\n"
            f.write(record)

        # Calculate and record vector lengths
        vector_length_dict = {}
        for j, vector in enumerate(tf_vector_list):
            np_vector = np.array(vector)
            vector_length = np.sqrt(np.sum(np_vector ** 2))
            vector_length_dict[file_names[j]] = vector_length

if __name__ == '__main__':
    start_time = time.time()
    command_index("../index/", "../largeFiles/collection/")
    end_time = time.time()
    print("Time taken:", end_time - start_time)
