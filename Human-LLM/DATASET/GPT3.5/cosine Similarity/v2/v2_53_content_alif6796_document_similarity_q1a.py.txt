import os
import numpy as np
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

# Define the query
query = 'printer'

# Initialize lists to store document content and names
docList = [query]
namList = [query]

# Define the parent directory containing the documents
parentDir = "/home/ai55/newsgroups/"

# Get the list of files in the directory
files = os.listdir(parentDir)

# Iterate through each file in the directory
for file in files:
    # Construct the full path of the file
    fullPath = os.path.join(parentDir, file)

    # Open the file for reading
    with open(fullPath, 'r') as filePtr:
        # Append the file name to the name list
        namList.append(filePtr.name)

        # Read the content of the file
        content = filePtr.read()

        # Append the content to the document list
        docList.append(content)

# Create a TF-IDF vectorizer
vectorizer = TfidfVectorizer(stop_words='english')

# Fit the vectorizer on the document list
model = vectorizer.fit(docList)

# Transform the document list into TF-IDF matrix
matrix = vectorizer.transform(docList)

# Calculate cosine similarity between the query document and all other documents
similarity = cosine_similarity(matrix[0:1], matrix)

# Flatten the similarity matrix
similarity = np.ravel(similarity)

# Find indices of documents with similarity greater than 0
result = np.argwhere(similarity > 0)

# Print the similarity matrix
print(similarity)

# Print the names of documents with similarity greater than 0
for idx in result:
    idx = np.asscalar(idx)
    print(namList[idx])
    print('-------------------------------------------------------------')
