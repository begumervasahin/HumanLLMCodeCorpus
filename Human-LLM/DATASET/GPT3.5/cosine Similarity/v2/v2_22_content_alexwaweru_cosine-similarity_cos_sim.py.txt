import numpy as np
from collections import Counter

# Function to calculate cosine similarity between two vectors
def cos_similarity(vector_a, vector_b):
    dot_product = np.dot(vector_a, vector_b)
    norm_a = np.linalg.norm(vector_a)
    norm_b = np.linalg.norm(vector_b)
    return dot_product / (norm_a * norm_b)

# Function to vectorize sentences by creating binary vectors for each word
def vectorize_sentences(*sentences):
    combined_sentence = ' '.join(sentences)
    sentences_split = [sentence.split(" ") for sentence in sentences]
    words = combined_sentence.split(" ")
    template = list(set(words))
    binary_vectors = [[] for _ in range(len(sentences))]

    for i in range(len(sentences_split)):
        for word in template:
            binary_vectors[i].append(int(word in sentences_split[i]))

    return binary_vectors

# Function to perform fully vectorized cosine similarity comparison
def fully_vectorized_cos_similarity():
    sentence_a = "How many teams participated in the first world cup"
    sentence_b = "Who holds the record for top scorer in a single World Cup"
    sentence_c = "Should I sell my car myself or trade it in"
    sentence_d = "How important is car maintenance"
    sentence_e = "How much does AWS SAM cost to use"
    sentence_f = "Which languages does AWS SAM support"

    vector_a, vector_b, vector_c, vector_d, vector_e, vector_f = vectorize_sentences(
        sentence_a, sentence_b, sentence_c, sentence_d, sentence_e, sentence_f)

    sentence_vectors = [vector_a, vector_b, vector_c, vector_d, vector_e, vector_f]

    for i in range(6):
        print('\n')
        for j in range(6):
            similarity = cos_similarity(sentence_vectors[i], sentence_vectors[j])
            print('{0:.4f}'.format(similarity), end='\t', flush=True)
    print('\n')

# Execute the fully vectorized cosine similarity trial
fully_vectorized_cos_similarity()
