import sys

def auc(labels, predicted_ctr):
    # Sort indices of predicted_ctr in descending order
    sorted_indices = sorted(range(len(predicted_ctr)), key=lambda i: predicted_ctr[i], reverse=True)

    auc_temp = 0.0
    tp = 0.0
    tp_prev = 0.0
    fp = 0.0
    fp_prev = 0.0

    last_value = predicted_ctr[sorted_indices[0]]

    for i in range(len(labels)):
        if labels[sorted_indices[i]] > 0:
            tp += 1
        else:
            fp += 1

        if last_value != predicted_ctr[sorted_indices[i]]:
            auc_temp += (tp + tp_prev) * (fp - fp_prev) / 2.0
            tp_prev = tp
            fp_prev = fp
            last_value = predicted_ctr[sorted_indices[i]]

    auc_temp += (tp + tp_prev) * (fp - fp_prev) / 2.0

    if tp * fp == 0:
        return 0.0
    else:
        return auc_temp / (tp * fp)

def evaluate(ids, true_values, predict_values):
    labels = [int(true_val) for true_val in true_values]
    predicted_ctr = [float(pred_val) for pred_val in predict_values]
    return auc(labels, predicted_ctr)

if __name__ == "__main__":
    if len(sys.argv) != 2:
        print("Usage: python script_name.py input_file")
        sys.exit(1)

    input_file = sys.argv[1]

    with open(input_file, "r") as f:
        ids, true_values, predict_values = zip(*(line.strip().split(",") for line in f))

    print(evaluate(ids, true_values, predict_values))
