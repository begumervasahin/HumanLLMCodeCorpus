import re
import operator
from gensim.models import KeyedVectors
from sklearn.metrics.pairwise import cosine_similarity

# Define analogy categories
categories = [': capital-world', ': currency', ': city-in-state', ': family', ': gram1-adjective-to-adverb',
              ': gram2-opposite', ': gram3-comparative', ': gram6-nationality-adjective', ': cellphones',
              ': national-animals']

# Initialize analogy dictionaries and variables
analogy = {}
analogy_vocab = {}

# Read analogies from file and populate analogy dictionaries
with open('analogies.txt', 'r') as file:
    lines = file.readlines()

    key = None
    for line in lines:
        line = line.strip()
        if line.startswith(':'):
            key = line
            analogy[key] = []
            analogy_vocab[key] = set()
        else:
            analogy[key].append(line.lower())
            words = line.split()
            analogy_vocab[key].update(words)

# Load GloVe word vectors
word_vectors = KeyedVectors.load_word2vec_format('glove.42B.300d.txt', binary=False)    

def get_compound(word1, word2, word3):
    # Calculate compound vector using GloVe word vectors
    if all(word in word_vectors.vocab for word in (word1, word2, word3)):
        return [word_vectors[word2][i] - word_vectors[word1][i] + word_vectors[word3][i]
                for i in range(len(word_vectors[word1]))]
    return None

def predict_word(word1, word2, word3, analogy_section):
    # Predict the word that completes the analogy
    cosine_sim = {}
    return_value = get_compound(word1, word2, word3)
    if return_value:
        for each_section in analogy_vocab[analogy_section]:
            if each_section in word_vectors.vocab:
                cosine_sim[each_section] = cosine_similarity([word_vectors[each_section]], [return_value])
            else:
                cosine_sim[each_section] = 0
    return max(cosine_sim.items(), key=operator.itemgetter(1))[0] if cosine_sim else None

def main():
    # Evaluate accuracy of analogy predictions using GloVe embeddings
    acc = 0
    total = 0
    for section in categories:
        for value in analogy[section]:
            splits = value.split()
            if len(splits) == 4:
                word1, word2, word3, expected_word = splits
                predicted_word = predict_word(word1, word2, word3, section)
                if predicted_word and predicted_word == expected_word:
                    acc += 1
                total += 1

    accuracy = (acc / total) * 100 if total > 0 else 0
    print("Accuracy using GloVe embeddings is:", accuracy)

if __name__ == '__main__':
    main()
