from collections import defaultdict

# Define the documents and index
documents = [['dog', 'deer', 'cat', 'mouse'], ['parrot', 'cat'], ['turtle', 'dog']]
index = {'dog': [0, 2], 'cat': [0, 1], 'parrot': [1], 'turtle': [2]}

# Function for performing AND search using list comprehension
def and_search_with_list_comprehension(documents, index, queries):
    # Get the document IDs for the first query
    doc_ids = set(index[queries[0]])
    
    # Perform AND search for the remaining queries
    for query in queries[1:]:
        doc_ids = doc_ids & set(index[query])
    
    # Return the documents corresponding to the matching document IDs
    return [documents[doc_id] for doc_id in doc_ids]

# Function for performing AND search using defaultdict and Counter
def and_search_with_defaultdict(documents, index, queries):
    # Initialize a defaultdict to count the occurrences of document IDs
    doc_id_counter = defaultdict(int)
    
    # Count the occurrences of document IDs for each query
    for query in queries:
        doc_id_counter.update(index[query])
    
    # Get the most common document ID and return the corresponding document
    most_common_doc_id = doc_id_counter.most_common()[0][0]
    return [documents[most_common_doc_id]]

# Testing the AND search functions
print(and_search_with_list_comprehension(documents, index, ['cat', 'dog']))
print(and_search_with_defaultdict(documents, index, ['cat', 'dog']))

# Sorting a list of tuples based on the second element
tuples_list = [(1, 2, 3), (4, 5, 6), (5, 1, 9)]
sorted_tuples = sorted(tuples_list, key=lambda x: x[1])
for tuple_item in sorted_tuples[:2]:
    print(tuple_item)

# Calculating the dot product of two lists
dot_product_sum = 0
for x, y in zip([1, 2, 3], [1, 2, 3]):
    dot_product_sum += x * y
print(dot_product_sum)
