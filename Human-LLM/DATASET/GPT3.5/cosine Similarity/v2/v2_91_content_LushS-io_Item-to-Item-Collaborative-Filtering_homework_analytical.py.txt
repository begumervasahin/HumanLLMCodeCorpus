import pandas as pd
import numpy as np
from sklearn.preprocessing import normalize
from sklearn.metrics.pairwise import cosine_similarity
from scipy.sparse import csr_matrix, csc_matrix
from scipy.sparse import find
from itertools import combinations
import time

def cosine_similar(mat):
    row_normed_mat = normalize(mat.tocsc(), axis=0)
    return row_normed_mat.T * row_normed_mat

def cos_sim(a, b):
    dot_product = np.dot(a, b)
    norm_a = np.linalg.norm(a)
    norm_b = np.linalg.norm(b)
    return dot_product / (norm_a * norm_b)

def combs_nd(a, r, axis=0):
    a = np.asarray(a)
    if axis < 0:
        axis += a.ndim
    indices = np.arange(a.shape[axis])
    dt = np.dtype([('', np.intp)] * r)
    indices = np.fromiter(combinations(indices, r), dt)
    indices = indices.view(np.intp).reshape(-1, r)
    return np.take(a, indices, axis=axis)

def import_hw2data_to_pd_df():
    user_1 = np.array([1, 0, 3, 0, 0, 5, 0, 0, 5, 0, 4, 0])
    user_2 = np.array([0, 0, 5, 4, 0, 0, 4, 0, 0, 2, 1, 3])
    user_3 = np.array([2, 4, 0, 1, 2, 0, 3, 0, 4, 3, 5, 0])
    test_1 = np.array([4, 5, 0, 5, 1, 0])
    test_2 = np.array([0, 3, 4, 3, 1, 2])
    test_3 = np.array([2, 0, 1, 3, 0, 4])
    users = pd.DataFrame(data=(test_1, test_2, test_3))
    return users

users = import_hw2data_to_pd_df()
np_users = users.to_numpy()
np_sparse = csr_matrix(np_users)

print('Dataframe before converting to sparse matrix:\n{}'.format(users))
print('Sparse Matrix Representation:\n{}'.format(np_sparse))

csc_mat = csc_matrix(np_users)
plp = cosine_similar(csc_mat)
print('Cosine Similarity Matrix (dense representation):\n{}'.format(plp.todense()))

(x, y, z) = find(np_sparse)
counts = np.bincount(x)
sums = np.bincount(x, weights=z)
np_sparse_avg = sums / counts
print('Average by Row for Sparse Matrix:\n{}'.format(np_sparse_avg))

X = np_sparse
nnz_per_row = np.diff(X.indptr)
Y = csr_matrix((X.data - np.repeat(np_sparse_avg, nnz_per_row), X.indices, X.indptr), shape=X.shape)
print('Centered Sparse Matrix:\n{}'.format(Y.todense()))
print('Transposed Centered Sparse Matrix:\n{}'.format(Y.T.todense()))

yao = cosine_similar(Y)
print('Cosine Similarity Matrix for Centered Sparse Matrix:\n{}'.format(yao.todense()))

start = time.time()
similarity = Y.dot(Y.T).toarray()
end = time.time()
print('Similarity Matrix (dense representation):\n{}'.format(similarity))
print('Time taken for similarity calculation: {} seconds'.format(end - start))

yah = Y.todense()
print('Cosine Similarity (User 1, User 2):', cos_sim(yah[0], yah[1].T))
print('Cosine Similarity (User 1, User 3):', cos_sim(yah[0], yah[2].T))
print('Cosine Similarity (User 2, User 3):', cos_sim(yah[1], yah[2].T))

dot_prod = np.dot(Y, Y.T)
print('Dot Product of Centered Sparse Matrix and its Transpose:\n{}'.format(dot_prod.todense()))

cosine_similarity(users)
