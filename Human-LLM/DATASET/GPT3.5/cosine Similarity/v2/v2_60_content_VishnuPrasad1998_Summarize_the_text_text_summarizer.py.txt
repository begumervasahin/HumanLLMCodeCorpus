import nltk
from nltk.corpus import stopwords
from nltk.cluster.util import cosine_distance
import numpy as np
import networkx as nx

def read_article(file_name):
    """
    Read the content of a text file and split it into sentences.
    """
    with open(file_name, "r") as file:
        filedata = file.read()
    # Split the text into sentences based on period (.)
    article = filedata.split(". ")
    # Split each sentence into words
    sentences = [sentence.split(" ") for sentence in article]
    # Remove the last empty sentence
    sentences.pop()
    return sentences

def sentence_similarity(sent1, sent2, stopwords=None):
    """
    Compute the cosine similarity between two sentences.
    """
    if stopwords is None:
        stopwords = []
    # Convert words to lowercase
    sent1 = [w.lower() for w in sent1]
    sent2 = [w.lower() for w in sent2]
    # Get unique words from both sentences
    all_words = list(set(sent1 + sent2))
    # Initialize vectors for sentence representation
    vector1 = [0] * len(all_words)
    vector2 = [0] * len(all_words)
    # Update vectors with word counts
    for w in sent1:
        if w in stopwords:
            continue
        vector1[all_words.index(w)] += 1
    for w in sent2:
        if w in stopwords:
            continue
        vector2[all_words.index(w)] += 1
    # Compute cosine distance between the two vectors
    return 1 - cosine_distance(vector1, vector2)

def build_similarity_matrix(sentences, stop_words):
    """
    Build a similarity matrix based on cosine similarity between sentences.
    """
    similarity_matrix = np.zeros((len(sentences), len(sentences)))
    for idx1 in range(len(sentences)):
        for idx2 in range(len(sentences)):
            if idx1 == idx2:
                continue
            similarity_matrix[idx1][idx2] = sentence_similarity(sentences[idx1], sentences[idx2], stop_words)
    return similarity_matrix

def generate_summary(file_name, top_n=5):
    """
    Generate a summary of the text file using the PageRank algorithm.
    """
    # Download stopwords if not available
    nltk.download("stopwords")
    # Get English stopwords
    stop_words = set(stopwords.words('english'))
    # Initialize list to store summary sentences
    summarize_text = []
    # Read sentences from the text file
    sentences = read_article(file_name)
    # Build similarity matrix
    sentence_similarity_matrix = build_similarity_matrix(sentences, stop_words)
    # Convert similarity matrix to a graph
    sentence_similarity_graph = nx.from_numpy_array(sentence_similarity_matrix)
    # Compute scores using PageRank algorithm
    scores = nx.pagerank(sentence_similarity_graph)
    # Sort sentences based on scores
    ranked_sentences = sorted(((scores[i], s) for i, s in enumerate(sentences)), reverse=True)
    # Print the indexes of top ranked sentences
    print("Indexes of top ranked sentences order are ", ranked_sentences)
    # Select top-ranked sentences for summary
    for i in range(top_n):
        summarize_text.append(" ".join(ranked_sentences[i][1]))
    # Write the summary to a file
    with open("summary.txt", "w") as file:
        file.write('\n'.join(summarize_text))

# Generate a summary for the provided text file with 2 sentences
generate_summary("test.txt", 2)
