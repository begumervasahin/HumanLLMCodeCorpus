from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity
import numpy as np
import pickle
import json
import random

def talk_to_chatbot(test_sentence, min_score, json_file, vectorizer_pickle, matrix_pickle):
    test_set = [test_sentence]

    try:
        with open(vectorizer_pickle, 'rb') as f:
            vectorizer = pickle.load(f)
        with open(matrix_pickle, 'rb') as f:
            matrix_train = pickle.load(f)
    except FileNotFoundError:
        vectorizer, matrix_train = train_chatbot(json_file, vectorizer_pickle, matrix_pickle)

    matrix_test = vectorizer.transform(test_set)
    cosine_sim = cosine_similarity(matrix_test, matrix_train)
    cosine_sim = np.delete(cosine_sim, 0)
    max_score = cosine_sim.max()

    if max_score > min_score:
        new_max = max_score - 0.01
        indices = np.where(cosine_sim > new_max)[0]
        response_index = random.choice(indices)
    else:
        return "live_chat", 0

    with open(json_file, "r") as sentences_file:
        reader = json.load(sentences_file)
        for idx, row in enumerate(reader):
            if idx == response_index:
                return row["response"], max_score

def train_chatbot(json_file, vectorizer_pickle, matrix_pickle):
    sentences = ["No you.", "No you."]
    with open(json_file, "r") as sentences_file:
        reader = json.load(sentences_file)
        for row in reader:
            sentences.append(row["message"])

    vectorizer = TfidfVectorizer()
    matrix_train = vectorizer.fit_transform(sentences)

    with open(vectorizer_pickle, 'wb') as f:
        pickle.dump(vectorizer, f)
    with open(matrix_pickle, 'wb') as f:
        pickle.dump(matrix_train, f)

    return vectorizer, matrix_train

def previous_chats(query):
    min_score = 0.7
    json_file = "data/previous_chats.json"
    vectorizer_pickle = "data/previous_vectorizer.pickle"
    matrix_pickle = "data/previous_matrix.pickle"

    query_response, score = talk_to_chatbot(query, min_score, json_file, vectorizer_pickle, matrix_pickle)
    return query_response

while True:
    user_input = input("User: ")
    print("Bot:", previous_chats(user_input))
