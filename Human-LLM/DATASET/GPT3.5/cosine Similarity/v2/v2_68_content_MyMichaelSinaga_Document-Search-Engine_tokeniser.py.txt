import math
from nltk.tokenize import RegexpTokenizer
from nltk.stem import PorterStemmer

# Function to tokenize and normalize text
def tokenise_normalise(raw_string):
    tokenizer = RegexpTokenizer(r'\w+')
    tokenised_text = tokenizer.tokenize(str(raw_string))
    token_normalise = [w.lower() for w in tokenised_text]
    return token_normalise

# Function to stem words
def stem(token_normalised_text):
    stemmer = PorterStemmer()
    processed_text = [stemmer.stem(w) for w in token_normalised_text]
    return processed_text

# Function to create a term frequency dictionary for a document
def create_tf_dict_doc(processed_text, doc_name, stats_dict):
    for root in processed_text:
        if root not in stats_dict:
            stats_dict[root] = {}
            stats_dict[root][doc_name] = 1
        else:
            stats_dict[root][doc_name] = stats_dict[root].get(doc_name, 0) + 1
    return stats_dict

# Function to create a term frequency dictionary for a query
def create_tf_dict_query(processed_query):
    query_dict = {}
    for root in processed_query:
        query_dict[root] = query_dict.get(root, 0) + 1
    return query_dict

# Function to calculate TF-IDF scores for documents
def find_tfidf_doc(stats_dict, num_docs):
    idf_dict = {}
    for word in stats_dict:
        idf_dict[word] = len(stats_dict[word])
    for word in stats_dict:
        for doc in stats_dict[word]:
            tf = 1 + math.log(stats_dict[word][doc])
            idf = math.log(num_docs / idf_dict[word])
            stats_dict[word][doc] = tf * idf
    return stats_dict, idf_dict

# Function to calculate TF-IDF scores for a query
def find_tfidf_query(query_dict, idf_dict, num_docs):
    for word in query_dict:
        tf = 1 + math.log(query_dict[word])
        idf = math.log(num_docs / idf_dict.get(word, 1))
        query_dict[word] = tf * idf
    return query_dict
