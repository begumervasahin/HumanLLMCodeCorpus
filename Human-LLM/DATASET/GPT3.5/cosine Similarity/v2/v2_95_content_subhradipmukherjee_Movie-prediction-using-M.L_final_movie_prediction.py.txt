import pandas as pd
import numpy as np
from sklearn.feature_extraction.text import CountVectorizer
from scipy import spatial
import math
import operator

# Load movie ratings data
ratings = pd.read_csv("D:/roops_backup/projects/data mining project/movie_recommender/movie_dataset.csv")

# Select relevant columns
ratings = ratings[['id', 'title', 'vote_average', 'popularity', 'vote_count', 'genres']]

# Normalize popularity and vote count
popularity_min = ratings['popularity'].min()
popularity_max = ratings['popularity'].max()
vote_count_min = ratings['vote_count'].min()
vote_count_max = ratings['vote_count'].max()

normalized_popularity = (ratings['popularity'] - popularity_min) / (popularity_max - popularity_min)
normalized_vote_count = (ratings['vote_count'] - vote_count_min) / (vote_count_max - vote_count_min)

# Extract genres and convert to count vectors
genres = ratings['genres'].fillna('')
cv = CountVectorizer()
genre_count = cv.fit_transform(genres)

# Create a dictionary to store movie data
moviedict = {}
for i in range(len(ratings)):
    movie_id = ratings.loc[i, 'id']
    title = ratings.loc[i, 'title']
    vote_average = ratings.loc[i, 'vote_average']
    popularity = normalized_popularity.loc[i]
    vote_count = normalized_vote_count.loc[i]
    genre_vector = genre_count.toarray()[i]
    moviedict[i] = (movie_id, title, vote_average, popularity, vote_count, genre_vector)

def compute_distance(movie1, movie2):
    popularity1 = movie1[3]
    genre_vector1 = movie1[5]
    popularity2 = movie2[3]
    genre_vector2 = movie2[5]
    
    genre_distance = spatial.distance.cosine(genre_vector1, genre_vector2)
    popularity_distance = (popularity2 - popularity1) ** 2
    total_distance = math.sqrt(popularity_distance + genre_distance ** 2)
    return total_distance

def get_neighbors(movie_id, k):
    distances = []
    for neighbor_id, neighbor_movie in moviedict.items():
        if neighbor_id != movie_id:
            distance = compute_distance(moviedict[movie_id], neighbor_movie)
            if distance <= 0.65:
                distances.append((neighbor_id, distance))
    
    distances.sort(key=operator.itemgetter(1))
    
    neighbors = []
    for i in range(k):
        neighbor_id = distances[i][0]
        neighbor_title = moviedict[neighbor_id][1]
        neighbor_distance = distances[i][1]
        neighbors.append((neighbor_title, neighbor_distance))
        print(neighbor_title)
    
    return neighbors

# Parameters
K = 15
movie_id = 94

# Find K nearest neighbors
get_neighbors(movie_id, K)
