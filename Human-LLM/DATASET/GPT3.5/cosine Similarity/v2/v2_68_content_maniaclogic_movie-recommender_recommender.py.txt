import pandas as pd
import numpy as np
from models import R, nmf, DF

def recommend(user_input_dict):
    # Initialize movie IDs
    movie_ids = sorted(list(R.columns))
    
    # Create a dictionary with movie IDs and empty ratings
    ratings_dict = {movie_id: np.nan for movie_id in movie_ids}

    # Update ratings_dict with user input ratings
    for movie_id, rating in user_input_dict.items():
        ratings_dict[movie_id] = rating

    # Convert ratings_dict to DataFrame
    profile = pd.DataFrame([list(ratings_dict.values())], columns=movie_ids)

    # Fill missing values and transpose the DataFrame
    profile_filled = profile.fillna(3.0)
    profile_transposed = profile_filled.transpose()

    # Transform user profile using NMF
    hidden_profile = nmf.transform(profile_transposed)

    # Predict ratings using hidden profile and NMF components
    prediction = np.dot(hidden_profile, nmf.components_)
    coeff_profile = pd.DataFrame(prediction, columns=R.columns)

    # Get movies not watched by the user
    movies_not_watched = coeff_profile.transpose().iloc[0][np.isnan(profile_transposed.iloc[0])]
    sorted_movies = movies_not_watched.sort_values(ascending=False)
    recommended_movie_ids = sorted_movies.index[:5]

    # Get movie titles from DataFrame
    recommendations = [DF.loc[movie_id, 'title'] for movie_id in recommended_movie_ids]

    return recommendations

# Example usage
user_input_ratings = {1: 5, 2: 4, 3: 3, 4: 2, 5: 1}  # User ratings for movies
recommended_movies = recommend(user_input_ratings)

# Print recommended movies
print("Recommended Movies:")
for movie in recommended_movies:
    print(movie)
