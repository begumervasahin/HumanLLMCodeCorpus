import nltk
import random
import string
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

# Read the text from the chatbot.txt file
with open('chatbot.txt', 'r', errors='ignore') as file:
    raw_text = file.read().lower()

# Tokenize the text into sentences and words
sent_tokens = nltk.sent_tokenize(raw_text)
word_tokens = nltk.word_tokenize(raw_text)

# Initialize the WordNet lemmatizer for text normalization
lemmatizer = nltk.stem.WordNetLemmatizer()

# Define functions for text normalization and lemmatization
def lemmatize_tokens(tokens):
    return [lemmatizer.lemmatize(token) for token in tokens]

punctuation_dict = dict((ord(punct), None) for punct in string.punctuation)

def normalize_text(text):
    return lemmatize_tokens(nltk.word_tokenize(text.lower().translate(punctuation_dict)))     

# Define greeting inputs and corresponding responses
greeting_inputs = ('hello', 'hi', 'whats up', 'hey')
greeting_responses = ['hi', 'hello', 'I am glad you are talking to me!']

def respond_to_greeting(sentence):
    for word in sentence.split():
        if word.lower() in greeting_inputs:
            return random.choice(greeting_responses)

# Define the chatbot's response function using TF-IDF and cosine similarity
def generate_response(user_response):
    robo_response = ''
    tfidf_vectorizer = TfidfVectorizer(tokenizer=normalize_text, stop_words='english')
    tfidf_matrix = tfidf_vectorizer.fit_transform(sent_tokens)
    similarity_values = cosine_similarity(tfidf_matrix[-1], tfidf_matrix)
    most_similar_idx = similarity_values.argsort()[0][-2]
    flat_similarity_values = similarity_values.flatten()
    flat_similarity_values.sort()
    required_tfidf = flat_similarity_values[-2]
    if required_tfidf == 0:
        robo_response = "I am sorry, but I cannot understand."
        return robo_response
    else:
        robo_response = robo_response + sent_tokens[most_similar_idx]
        return robo_response

# Initialize the chatbot
chatting_active = True
print("BOT: My name is Robo. I will answer your queries about Chatbots. If you want to exit, type Bye!")

# Chatting loop
while chatting_active:
    user_input = input("You: ").lower()
    if user_input != 'bye':
        if user_input in ('thanks', 'thank you'):
            chatting_active = False
            print("BOT: You are welcome.")
        else:
            greeting_response = respond_to_greeting(user_input)
            if greeting_response is not None:
                print("BOT:", greeting_response)
            else:
                sent_tokens.append(user_input)
                word_tokens += nltk.word_tokenize(user_input)
                final_words = list(set(word_tokens))
                print("BOT:", generate_response(user_input))
                sent_tokens.remove(user_input)
    else:
        chatting_active = False
        print("BOT: Goodbye!")
