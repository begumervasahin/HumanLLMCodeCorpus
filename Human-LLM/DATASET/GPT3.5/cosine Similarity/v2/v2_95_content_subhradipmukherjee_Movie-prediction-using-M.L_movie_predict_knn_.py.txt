import pandas as pd
import numpy as np
from sklearn.feature_extraction.text import CountVectorizer
from scipy import spatial
import operator

# Load movie ratings data from CSV file
ratings = pd.read_csv("D:/roops_backup/projects/data mining project/movie_recommender/movie_dataset.csv")

# Select relevant columns from the ratings dataframe
ratings_subset = ratings[['id', 'title', 'vote_average', 'vote_count', 'popularity', 'genres']]

# Normalize the popularity column
min_popularity = ratings_subset['popularity'].min()
max_popularity = ratings_subset['popularity'].max()
normalized_popularity = (ratings_subset['popularity'] - min_popularity) / (max_popularity - min_popularity)

# Fill missing values in the genres column and convert them to count vectors
genres = ratings_subset['genres'].fillna('')
vectorizer = CountVectorizer()
genre_counts = vectorizer.fit_transform(genres)

# Create a dictionary to store movie information
movie_dict = {}
for i in range(len(ratings_subset)):
    movie_id = ratings_subset.loc[i, 'id']
    title = ratings_subset.loc[i, 'title']
    vote_average = ratings_subset.loc[i, 'vote_average']
    vote_count = ratings_subset.loc[i, 'vote_count']
    popularity = normalized_popularity.loc[i]
    genre_vector = genre_counts.toarray()[i]
    movie_dict[i] = (movie_id, title, vote_average, vote_count, popularity, genre_vector)

# Function to compute the distance between two movies based on genre and popularity
def compute_distance(movie1, movie2):
    genre_distance = spatial.distance.cosine(movie1[5], movie2[5])
    popularity_distance = abs(movie1[4] - movie2[4])
    return genre_distance + popularity_distance

# Function to get the index of a movie based on its title
def get_index(title):
    return ratings[ratings.title == title].index[0]

# Function to get K nearest neighbors for a given movie
def get_neighbors(movie_id, K):
    distances = []
    for movie_index, movie_info in movie_dict.items():
        if movie_index != movie_id:
            dist = compute_distance(movie_dict[movie_id], movie_info)
            if dist <= 0.55:
                distances.append((movie_index, dist))
    distances.sort(key=operator.itemgetter(1))
    neighbors = [(movie_dict[dist[0]][1], dist[1]) for dist in distances[:K]]
    return neighbors

# Main function to find neighbors for a given movie and print the results
def main(movie_title, K):
    movie_index = get_index(movie_title)
    print("The given movie is:", movie_dict[movie_index][1])
    neighbors = get_neighbors(movie_index, K)
    for neighbor in neighbors:
        print(neighbor)

# Parameters
K = 10
movie_title = "Unbreakable"

# Call the main function
main(movie_title, K)
