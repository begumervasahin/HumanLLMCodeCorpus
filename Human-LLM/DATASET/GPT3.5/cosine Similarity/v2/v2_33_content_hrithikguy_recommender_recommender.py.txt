from scipy import spatial
import numpy as np
import pandas as pd

# Function to calculate cosine similarity between two arrays
def cosine_similarity(arr1, arr2):
    """
    Calculate the cosine similarity between two arrays.
    
    Parameters:
    arr1 (array): First array.
    arr2 (array): Second array.
    
    Returns:
    float: Cosine similarity between the two arrays.
    """
    return 1 - spatial.distance.cosine(arr1, arr2)

# Read data from files
padpick = pd.read_json("padpick.json", "r")
with open("description_vectors.txt", "r") as dvs_file:
    description_vectors = [line.strip().split(' ') for line in dvs_file]
with open("indices.txt", "r") as index_file:
    indices = [int(float(line.strip())) for line in index_file]
with open("recommender_weights.txt", "r") as weights_file:
    weight_values = [float(line.strip()) for line in weights_file]

# Preprocess description vectors
new_description_vectors = []
for description in description_vectors:
    cur = np.asarray(map(float, description))
    cur2 = [cur[k] * l for k, l in enumerate(cur)]
    new_description_vectors.append(np.array(cur2))
new_description_vectors = np.asarray(new_description_vectors)

# Read user input and calculate cosine distances
with open("recommender_input.txt", "r") as input_file:
    entries_liked = [int(line.strip()) for line in input_file]
user_index = int(input("Enter a number between 0 and 3128\n"))
user_input = new_description_vectors[user_index]
cosine_distances = []
for i, vector in enumerate(new_description_vectors):
    cosine_distances.append(cosine_similarity(map(float, user_input), map(float, vector)))
cosine_distances = np.asarray(cosine_distances)
order = cosine_distances.argsort()
results = []

# Get top recommended entries
print("\n\n")
results.append(np.asarray(padpick)[int(indices[user_index])])
for i in order[-11:][::-1][1:]:
    results.append(np.asarray(padpick)[int(indices[i])])

# Write results to a TSV file
with open(str(user_index) + ".tsv", "w") as output_file:
    output_file.write("Address\tBathrooms\tBedrooms\tBrokerage\tBuilder\tDetailedCharacteristics\tDisclaimer\tDiscloseAddress\tExpenses\tForeclosureStatus\tFullBathrooms\tHalfBathrooms\tLeadRoutingEmail\tListPrice\tListPriceLow\tListingCategory\tListingDate\tListingDescription\tListingKey\tListingParticipants\tListingStatus\tListingTitle\tListingURL\tLivingArea\tLocation\tLotSize\tMarketingInformation\tMlsId\tMlsName\tMlsNumber\tModificationTimestamp\tOffices\tOneQuarterBathrooms\tOpenHouses\tPartialBathrooms\tPhotos\tPropertySubType\tPropertyType\tProviderCategory\tProviderName\tProviderURL\tTaxes\tThreeQuarterBathrooms\tVirtualTours\tYearBuilt\n")
    for result in results:
        cur = [j.encode('utf-8') if isinstance(j, unicode) else str(j).encode('utf-8') for j in result]
        output_file.write('\t'.join(cur) + '\n')
