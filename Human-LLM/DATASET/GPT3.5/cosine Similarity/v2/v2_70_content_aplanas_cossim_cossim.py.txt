import os
import argparse
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

def load_files_from_directory(directory):
    files = [os.path.join(directory, f) for f in os.listdir(directory)
             if os.path.isfile(os.path.join(directory, f))]
    return files

def calculate_similarity(tfidf_sample, tfidf_corpus):
    similarity = cosine_similarity(tfidf_sample, tfidf_corpus)
    max_similarity_index = similarity.argsort()[0][-2]
    max_similarity_score = similarity[0][max_similarity_index]
    return max_similarity_index, max_similarity_score

def main(good_dir, bad_dir, sample_file):
    good_files = load_files_from_directory(good_dir)
    bad_files = load_files_from_directory(bad_dir)

    all_files = good_files + bad_files + [sample_file]

    tfidf_good = TfidfVectorizer().fit_transform(
        [open(f, errors='ignore').read() for f in good_files])
    tfidf_bad = TfidfVectorizer().fit_transform(
        [open(f, errors='ignore').read() for f in bad_files])
    tfidf_sample = TfidfVectorizer().fit_transform(
        [open(sample_file, errors='ignore').read()])

    index_good, similarity_good = calculate_similarity(tfidf_sample, tfidf_good)
    index_bad, similarity_bad = calculate_similarity(tfidf_sample, tfidf_bad)

    print('Maximum similarity with good samples:', similarity_good)
    print('Maximum similarity with bad samples:', similarity_bad)

    similar_document = good_files[index_good] if similarity_good >= similarity_bad else bad_files[index_bad]
    print('Similar document:', similar_document)

if __name__ == '__main__':
    parser = argparse.ArgumentParser(description='Basic cosine similarity comparator.') 
    parser.add_argument('--good', help='directory where the good samples are')
    parser.add_argument('--bad', help='directory where the bad samples are')
    parser.add_argument('sample', help='file to compare with')
    args = parser.parse_args()

    if not args.good or not args.bad:
        print('Provide directory paths for good and bad samples')
        exit(1)

    main(args.good, args.bad, args.sample)
