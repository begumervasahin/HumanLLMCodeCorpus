from __future__ import (absolute_import, division, print_function, unicode_literals)    
from surprise import Dataset, Reader
import numpy as np

def calculate_mean_centered_ratings(dataset_path):
    # Load dataset from file
    file_path = dataset_path
    reader = Reader(line_format='user item rating', rating_scale=(1, 7), sep='\t')      
    data = Dataset.load_from_file(file_path, reader)
    
    # Build full training set
    trainset = data.build_full_trainset()
    
    # Get the number of users and items
    num_users = trainset.n_users
    num_items = trainset.n_items
    
    # Initialize matrices
    user_item_ratings = np.zeros((num_users, num_items), dtype=float)
    user_total_ratings = np.zeros((num_users, 3), dtype=float)
    average_user_item_ratings = np.zeros((num_users, num_items), dtype=float)

    # Iterate over user-item ratings
    for user_id, ratings in trainset.ur.items():
        for item_id, rating in ratings:
            if rating != 0:
                user_item_ratings[user_id, item_id] += rating
                user_total_ratings[user_id, 0] += rating
                user_total_ratings[user_id, 1] += 1
    
    print('Original user ratings\n', user_item_ratings, '\n')

    # Calculate average ratings per user
    for user_id, ratings in trainset.ur.items():
        for item_id, rating in ratings:
            if rating != 0:
                average_user_item_ratings[user_id, item_id] += user_total_ratings[user_id, 0] / user_total_ratings[user_id, 1]

    # Calculate mean-centered ratings
    mean_centered_ratings = user_item_ratings - average_user_item_ratings
    return mean_centered_ratings

# Define the path to the dataset file
file_path = 'data/table2.1_ratings.txt'

# Calculate and print the mean-centered ratings
print('The mean centered ratings\n', calculate_mean_centered_ratings(file_path))
