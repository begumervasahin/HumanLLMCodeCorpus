import re
import math
import nltk
from nltk.stem import PorterStemmer
from nltk.corpus import stopwords
import string

# Initialize NLTK components
nltk.download('stopwords')
nltk.download('punkt')

# Initialize Porter Stemmer
ps = PorterStemmer()

# Load stopwords and query collection
stop_words = set(stopwords.words('english'))
query_collection_name = "cran.qry"

# Read and preprocess queries
with open(query_collection_name, 'r') as query_file:
    queries = query_file.read().replace("\n", " ")
    queries = queries.split(".I")
    queries = [phase for phase in queries if phase != ""]

# Compute IDF for query terms
IDF_query_collection = {}
for query in queries:
    query_tokens = nltk.word_tokenize(query)
    query_tokens = [word for word in query_tokens if word.lower() not in stop_words and word not in string.punctuation]
    non_repeating_query = list(set(query_tokens))
    for word in non_repeating_query:
        IDF_query_collection[word] = IDF_query_collection.get(word, 0) + 1

for word in IDF_query_collection:
    IDF_query_collection[word] = math.log((len(queries) / IDF_query_collection[word]) + 1)

# Create TF-IDF vectors for queries
vector_list_queries = []
for query in queries:
    query_tokens = nltk.word_tokenize(query)
    query_tokens = [word for word in query_tokens if word.lower() not in stop_words and word not in string.punctuation]
    query_vector = {}
    for word in query_tokens:
        query_vector[word] = math.log(query_tokens.count(word) + 1) * IDF_query_collection[word]
    vector_list_queries.append(query_vector)

# Read and preprocess abstracts
abstract_name = "cran.all.1400"
with open(abstract_name, 'r') as abstract_file:
    abstract = abstract_file.read().replace("\n", " ").split(".I")
    abstract = [document for document in abstract if document != ""]
    for i in range(len(abstract)):
        abstract[i] = nltk.word_tokenize(abstract[i])
        result = 0
        for j in range(len(abstract[i])):
            if abstract[i][j] == ".W":
                result = j
                break
        del(abstract[i][0:j+1])
    for i in range(len(abstract)):
        abstract[i] = [word for word in abstract[i] if word.lower() not in stop_words and word not in string.punctuation]

# Compute IDF for abstract terms
IDF_abstract_collection = {}
for document in abstract:
    for word in document:
        IDF_abstract_collection[word] = IDF_abstract_collection.get(word, 0) + 1

for word in IDF_abstract_collection:
    IDF_abstract_collection[word] = math.log((len(abstract) / IDF_abstract_collection[word]) + 1)

# Create TF-IDF vectors for documents
vector_list_documents = []
for document in abstract:
    document_tokens = [word for word in document if word.lower() not in stop_words and word not in string.punctuation]
    document_vector = {}
    for word in document_tokens:
        document_vector[word] = math.log(document_tokens.count(word) + 1) * IDF_abstract_collection[word]
    vector_list_documents.append(document_vector)

# Open output file to write rankings
output_file = open("output.txt", 'w')

# Calculate cosine similarity and rank documents for each query
for i in range(len(vector_list_queries)):
    query_vec = vector_list_queries[i]
    ranking = []
    ranking.clear()
    for j in range(len(vector_list_documents)):
        document_vec = vector_list_documents[j]
        for word in query_vec:
            if word not in document_vec:
                document_vec[word] = 0
        cosine_num = sum(query_vec[word] * document_vec[word] for word in query_vec)
        queries_square = sum(query_vec[word] ** 2 for word in query_vec)
        document_square = sum(document_vec[word] ** 2 for word in document_vec)
        cosine_den = math.sqrt(queries_square * document_square)
        if cosine_den == 0:
            cosine_ranking = 0
        else:
            cosine_ranking = cosine_num / cosine_den
        ranking.append((cosine_ranking, j))
    ranking.sort(key=lambda tup: tup[0], reverse=True)
    for ranks in ranking:
        output_file.write(str(i + 1) + " " + str(ranks[1] + 1) + " " + str(ranks[0]) + "\n")

output_file.close()
