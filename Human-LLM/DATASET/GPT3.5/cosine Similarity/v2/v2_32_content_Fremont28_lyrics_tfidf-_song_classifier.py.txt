import pandas as pd
from sklearn.feature_extraction.text import TfidfVectorizer, CountVectorizer, TfidfTransformer
from sklearn.model_selection import train_test_split
from sklearn.naive_bayes import MultinomialNB
from sklearn.cluster import KMeans

# Load the lyrics data from 'lyrics.csv'
lyrics = pd.read_csv("lyrics.csv", encoding="latin-1")

# Drop rows with missing values
lyrics_cleaned = lyrics.dropna()

# Filter data for Hip-Hop genre
hiphop_lyrics = lyrics_cleaned[lyrics_cleaned['genre'] == 'Hip-Hop']

# Generate unique artist IDs based on the 'artist' column
hiphop_lyrics['artist_id'] = hiphop_lyrics.groupby(['artist']).ngroup()

# Create a TF-IDF vectorizer for text processing
tfidf_vectorizer = TfidfVectorizer(sublinear_tf=True, norm='l2', encoding='latin-1', ngram_range=(1, 2), stop_words='english')

# Transform lyrics data into TF-IDF features
tfidf_features = tfidf_vectorizer.fit_transform(hiphop_lyrics['lyrics']).toarray()
labels = hiphop_lyrics['artist_id']

# Split the data into training and testing sets
X_train, X_test, y_train, y_test = train_test_split(hiphop_lyrics['lyrics'], hiphop_lyrics['artist_id'], random_state=0)

# Create a CountVectorizer to count words in text
count_vect = CountVectorizer()
X_train_counts = count_vect.fit_transform(X_train)

# Transform word counts into TF-IDF values
tfidf_transformer = TfidfTransformer()
X_train_tfidf = tfidf_transformer.fit_transform(X_train_counts)

# Train a Multinomial Naive Bayes classifier on TF-IDF transformed data
clf_nb = MultinomialNB().fit(X_train_tfidf, y_train)

# Example: Predict the artist ID for a new song
new_song_lyrics = "Your new song lyrics here"
predicted_artist_id = clf_nb.predict(count_vect.transform([new_song_lyrics]))

# Perform KMeans clustering on TF-IDF features
tfidf_matrix = tfidf_vectorizer.fit_transform(hiphop_lyrics['lyrics'])
num_clusters = 5
kmeans = KMeans(n_clusters=num_clusters)
kmeans.fit(tfidf_matrix)
clusters = kmeans.labels_.tolist()

# Create a DataFrame to store clustering results
cluster_results = pd.DataFrame(clusters, columns=['cluster'])
final_cluster = pd.concat([cluster_results, hiphop_lyrics[['lyrics', 'artist']]], axis=1)

# Display cluster information
print(final_cluster.head(10))
print(final_cluster.info())
print(final_cluster['cluster'].value_counts())

# Example: Get the counts of artists in cluster one
cluster_one = final_cluster[final_cluster['cluster'] == 0]
print(cluster_one['artist'].value_counts())

# Example: Get the counts of artists in cluster two
cluster_two = final_cluster[final_cluster['cluster'] == 1]
print(cluster_two['artist'].value_counts())
