from scipy.spatial.distance import cosine
import numpy as np
import tensorflow as tf
import spacy
from spacy.lang.fr import French
from sklearn.feature_extraction.text import TfidfVectorizer

# Load French language model for tokenization
nlp = spacy.load('fr')
tokenizer = French().Defaults.create_tokenizer(nlp)

# Tokenizer function using SpaCy tokenizer for French text
def word_tokenizer(text):
    return [token.text for token in tokenizer(text)]

# Sample texts
text1 = 'bonjour, qui êtes vous ?'
text2 = "bonjour comment ça va, qu'est-ce que tu deviens ?"
text3 = "qui est le président des états unis ?"
text4 = "qui est la bas ?"

# Tokenize text using the word_tokenizer function
tokens_text2 = word_tokenizer(text2)
print(tokens_text2)

# Initialize TF-IDF Vectorizer with custom tokenizer
tfidf = TfidfVectorizer(norm='l2', min_df=0, use_idf=True, smooth_idf=False, sublinear_tf=True, tokenizer=word_tokenizer)

# Fit and transform the texts using TF-IDF Vectorizer
sparse_tfidf_texts = tfidf.fit_transform([text1, text2, text3, text4])

# Get feature names (tokens)
feature_names = tfidf.get_feature_names()

# Print TF-IDF scores for each document
for doc in range(len([text1, text2, text3, text4])):
    feature_index = sparse_tfidf_texts[doc, :].nonzero()[1]
    tfidf_scores = zip(feature_index, [sparse_tfidf_texts[doc, x] for x in feature_index])
    print(f"TF-IDF scores for Document {doc+1}:")
    for feature_idx, score in [(feature_names[i], s) for (i, s) in tfidf_scores]:
        print(feature_idx, score)
