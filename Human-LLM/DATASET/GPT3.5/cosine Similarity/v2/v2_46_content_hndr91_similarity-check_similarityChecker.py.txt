import PyPDF2
import glob
import re
import argparse
import sys
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import linear_kernel

def load_pdf_files(path):
    """Load PDF files and extract text."""
    corpus = []
    for pdf_file in glob.glob(path + "/*.pdf"):
        with open(pdf_file, 'rb') as pdf_obj:
            pdf_reader = PyPDF2.PdfFileReader(pdf_obj)
            num_pages = pdf_reader.numPages
            text = ''
            for i in range(num_pages):
                page = pdf_reader.getPage(i)
                page_text = page.extractText()
                page_text = page_text.replace('\n', ' ').replace('\r', ' ').replace('.', '').replace(',', '')
                page_text = ''.join([char for char in page_text if not char.isdigit()])
                page_text = re.sub(r'[^\w]', ' ', page_text)
                page_text = " ".join(page_text.split())
                page_text = page_text.lower()
                text += page_text + ' '
            corpus.append((pdf_file, text))
    return corpus

def calc_tfidf_matrix(corpus):
    """Calculate TF-IDF matrix."""
    vectorizer = TfidfVectorizer()
    tfidf_matrix = vectorizer.fit_transform([content for _, content in corpus])
    return corpus, tfidf_matrix

def find_similar(tfidf_matrix, index, top_n):
    """Find similar documents based on TF-IDF matrix."""
    cosine_similarities = linear_kernel(tfidf_matrix[index:index+1], tfidf_matrix).flatten()
    related_docs_indices = sorted(range(len(cosine_similarities)), key=lambda i: cosine_similarities[i], reverse=True)
    related_docs_indices = [i for i in related_docs_indices if i != index]
    return [(idx, cosine_similarities[idx]) for idx in related_docs_indices][:top_n]

def parse_arguments():
    """Parse command line arguments."""
    parser = argparse.ArgumentParser(description='Calculate Similarity Index of Documents')
    parser.add_argument('-P', '--path', help='Path to corpus', required=True)
    parser.add_argument('-T', '--top', help='Define number of top n identical documents', type=int, default=10)
    parser.add_argument('-I', '--index', help='Define corpus index', type=int, required=True)
    return parser.parse_args()

def main():
    args = parse_arguments()
    corpus = load_pdf_files(args.path)
    tfidf_matrix = calc_tfidf_matrix(corpus)[1]
    print('The selected document is:', corpus[args.index][0], '\n')
    print('======== Similarity Index Results ========')
    for idx, score in find_similar(tfidf_matrix, args.index, args.top):
        print(score, corpus[idx][0])

if __name__ == '__main__':
    main()
