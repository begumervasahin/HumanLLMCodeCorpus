import math
import pickle
import pandas as pd
import numpy as np
from sklearn.model_selection import StratifiedKFold

def evaluate(ratings, max_users, max_items, n_features, gaus_mean, gaus_var, num_iter, alpha, lbd, error_threshold, continue_from_prev_model, save_model):
    from my_funcs import init_func, compute_cost, gradient_descent
    
    user_feat, item_feat, prev_error, curr_error = init_func(ratings, max_users, max_items, n_features, gaus_mean, gaus_var, continue_from_prev_model)
    print("User and Item feature matrices initialized successfully")
    
    for itr in range(num_iter):
        error = compute_cost(ratings, user_feat, item_feat)
        print(f"Iteration: {itr + 1} and the normalized loss: {error}")
        
        if prev_error == -1 and curr_error == -1:
            prev_error = curr_error = error
            user_feat, item_feat = gradient_descent(ratings, user_feat, item_feat, alpha, lbd)
        else:
            curr_error = error
            err_diff = abs(curr_error - prev_error)
            if err_diff <= error_threshold:
                break
            else:
                prev_error = curr_error
                user_feat, item_feat = gradient_descent(ratings, user_feat, item_feat, alpha, lbd)
                
    if save_model:
        with open("user_feat.pickle", "wb") as output_file:
            pickle.dump(user_feat, output_file)
        with open("item_feat.pickle", "wb") as output_file:
            pickle.dump(item_feat, output_file)
        print("Pickle Save Successful :)")
    else:
        print("Training Complete and Result Unsaved :(")
        
    return n_features, user_feat, item_feat

def kfold_evaluate(ratings, max_users, max_items, n_features, gaus_mean, gaus_var, num_iter, alpha, continue_from_prev_model, save_model):
    from my_funcs import init_func, compute_cost, gradient_descent
    
    n_splits = 5
    print(f"Evaluating {n_splits}-fold cross validation...")
    
    skf = StratifiedKFold(n_splits=n_splits)
    order_by = ratings.values[:, 0]
    skf.get_n_splits(ratings.values, order_by)
    fold_number = 0
    
    for train_index, test_index in skf.split(ratings.values, order_by):
        fold_number += 1
        print(f"Fold Number: {fold_number}")
        
        train_data, test_data = ratings.values[train_index], ratings.values[test_index]
        train_df = pd.DataFrame(train_data, columns=['user_id', 'movie_id', 'rating'])
        test_df = pd.DataFrame(test_data, columns=['user_id', 'movie_id', 'rating'])
        
        n_features, user_feat, item_feat = evaluate(train_df, max_users, max_items, n_features, gaus_mean, gaus_var, num_iter, alpha, continue_from_prev_model, save_model)
        
        mean_avg_precision = 0.0
        for user_id in range(1, ratings['user_id'].nunique() + 1):
            pred_array = np.empty((1, 3))
            compare2 = test_df.loc[test_df['user_id'] == user_id, 'movie_id'].values
            
            for item_id in range(1, max_items + 1):
                if (user_id, item_id) not in train_index or (user_id, item_id) in test_index:
                    pred_array = np.append(pred_array, [[user_id, item_id, np.dot(user_feat[user_id - 1], item_feat[item_id - 1])]], axis=0)
                    
            pred_array = np.delete(pred_array, 0, axis=0)
            pred_array = pred_array[np.argsort(-pred_array[:, 2])]
            compare1 = pred_array[:compare2.size, 1].astype(int)
            mean_avg_precision += float(len(set(compare1) & set(compare2))) / compare2.size
            print(f"User ID: {user_id} Precision: {float(len(set(compare1) & set(compare2))) / compare2.size}")
            
        mean_avg_precision /= ratings['user_id'].nunique()
        print(f"MEAN AVG PRECISION: {mean_avg_precision}")
        
        testError_rmse = 0.0
        testError_mae = 0.0
        for row in range(test_df.shape[0]):
            user_id = test_df.loc[row, 'user_id'] - 1
            item_id = test_df.loc[row, 'movie_id'] - 1
            pred = np.dot(user_feat[user_id], item_feat[item_id])
            actual = test_df.iloc[row, 2]
            error = actual - pred
            testError_rmse += error ** 2
            testError_mae += abs(error)
            
        testError_rmse /= test_df.shape[0]
        testError_mae /= test_df.shape[0]
        print(f"Fold Number: {fold_number}")
        print(f"TEST ERRORS: RMSE: {math.sqrt(testError_rmse)} MAE: {testError_mae}")
        print("********************************************\n")
