import sys
import gensim
import numpy as np
import math
from nltk.corpus import stopwords
from gensim.models import KeyedVectors

# Set default encoding to utf-8 (for Python 2 compatibility)
reload(sys)
sys.setdefaultencoding("utf-8")

# Load the pre-trained word2vec model
path_to_bin_vectors = 'GoogleNews-vectors-negative300.bin'
print("Loading the data file... Please wait...")
model = KeyedVectors.load_word2vec_format(path_to_bin_vectors, binary=True)
print("Successfully loaded the word2vec model!")

class PhraseVector:
    def __init__(self, phrase):
        self.vector = self.phrase_to_vec(phrase)

    def convert_vector_set_to_vec_average_based(self, vector_set, ignore=[]):
        if len(ignore) == 0:
            return np.mean(vector_set, axis=0)
        else:
            return np.dot(np.transpose(vector_set), ignore) / sum(ignore)

    def phrase_to_vec(self, phrase):
        cached_stop_words = stopwords.words("english")
        phrase = phrase.lower()
        words_in_phrase = [word for word in phrase.split() if word not in cached_stop_words]
        vector_set = []
        for a_word in words_in_phrase:
            try:
                word_vector = model[a_word]
                vector_set.append(word_vector)
            except KeyError:
                pass
        return self.convert_vector_set_to_vec_average_based(vector_set)

    def cosine_similarity(self, other_phrase_vec):
        dot_product = np.dot(self.vector, other_phrase_vec)
        norm_product = np.linalg.norm(self.vector) * np.linalg.norm(other_phrase_vec)
        cosine_similarity = dot_product / norm_product if norm_product != 0 else 0
        return cosine_similarity

if __name__ == "__main__":
    while True:
        user_input1 = input("Type the first phrase: ")
        user_input2 = input("Type the second phrase: ")
        phrase_vector1 = PhraseVector(user_input1)
        phrase_vector2 = PhraseVector(user_input2)
        similarity_score = phrase_vector1.cosine_similarity(phrase_vector2.vector)
        print("Similarity Score:", similarity_score)
