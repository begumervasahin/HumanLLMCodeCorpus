import json
import jieba
import re
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

# Function to load data from a JSON file
def load_data(filename):
    with open(filename, 'r', encoding='utf-8') as file:
        data = json.load(file)
    return data

# Function to tokenize text using Jieba and remove stop words
def tokenize_text(content, stopwordset):
    tokens = jieba.cut(content, cut_all=False)
    filtered_tokens = [token for token in tokens if token not in stopwordset and re.match(r'^[\u4E00-\u9FFFa-zA-Z]+$', token)]
    return ' '.join(filtered_tokens)

# Function to calculate TF-IDF and extract top keywords
def calculate_tfidf(contents):
    tfidf_vectorizer = TfidfVectorizer()
    tfidf_matrix = tfidf_vectorizer.fit_transform(contents)
    feature_names = tfidf_vectorizer.get_feature_names_out()
    tfidf_weight = tfidf_matrix.toarray()
    text_weight_list = [{feature_names[j]: tfidf_weight[i][j] for j in range(len(feature_names))} for i in range(len(tfidf_weight))]
    return text_weight_list

# Function to extract top keywords and assign tags to articles
def extract_keywords(text_weight_list):
    tag_list = []
    for one_article in text_weight_list:
        sorted_keywords = sorted(one_article.items(), key=lambda x: x[1], reverse=True)
        tags = [keyword[0] for keyword in sorted_keywords[:5]]
        tag_list.append(tags)
    return tag_list

# Function to load stopwords from a file
def load_stopwords(filename):
    with open(filename, 'r', encoding='utf-8') as file:
        stopwords = set(line.strip('\n') for line in file)
    return stopwords

def main():
    # Load data, stopwords, and tokenize text
    all_data = load_data('news.json')
    stopwords = load_stopwords('stopwords.txt')
    total_content = [tokenize_text(article['Content'], stopwords) for article in all_data[:50]]

    # Calculate TF-IDF and extract top keywords
    text_weight_list = calculate_tfidf(total_content)
    tag_list = extract_keywords(text_weight_list)

    # Assign tags to articles
    for i in range(len(tag_list)):
        all_data[i]['Tag'] = tag_list[i]

    # Calculate cosine similarity and print results
    tfidf_matrix = TfidfVectorizer().fit_transform(total_content)
    similarity_matrix = cosine_similarity(tfidf_matrix, tfidf_matrix)
    for row in similarity_matrix:
        print(row)

if __name__ == "__main__":
    main()
