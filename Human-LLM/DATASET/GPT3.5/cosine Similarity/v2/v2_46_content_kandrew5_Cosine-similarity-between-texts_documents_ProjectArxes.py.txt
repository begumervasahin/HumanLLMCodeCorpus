from scipy import spatial
import math
import string

def read_integer_input(prompt):
    while True:
        try:
            value = int(input(prompt))
            if value <= 0:
                raise ValueError
            return value
        except ValueError:
            print("Please enter a valid positive integer.")

def read_document_name(prompt):
    while True:
        try:
            document_name = input(prompt)
            open(document_name, "r")
            return document_name
        except IOError:
            print("Error: The specified file was not found. Please try again.")

def calculate_document_vector(document_words, unique_words):
    vector = [0] * len(unique_words)
    for word in document_words:
        if word in unique_words:
            vector[unique_words.index(word)] += 1
    return vector

def main():
    num_documents = read_integer_input("How many documents do you want to compare? ")   

    documents = []
    document_vectors = []
    for i in range(num_documents):
        print("Please insert the document name:")
        document_name = read_document_name()
        documents.append(document_name)
        with open(document_name, "r") as file:
            document_text = file.read()
            document_words = document_text.lower().translate(str.maketrans('', '', string.punctuation)).split()
            document_vectors.append(calculate_document_vector(document_words, documents))

    similarity_matrix = [[0] * num_documents for _ in range(num_documents)]
    for i in range(num_documents):
        for j in range(num_documents):
            if i < j:
                similarity = 1 - spatial.distance.cosine(document_vectors[i], document_vectors[j])
                similarity_matrix[i][j] = round(similarity, 4)

    num_pairs = read_integer_input("How many pairs do you want to print? ")
    max_pairs = math.factorial(num_documents) // (2 * math.factorial(num_documents - 2))
    if num_pairs > max_pairs:
        print("Sorry, there are not enough documents to generate that many pairs. Please choose a smaller number.")
        return

    for _ in range(num_pairs):
        max_similarity = 0
        max_indices = (0, 0)
        for i in range(num_documents):
            for j in range(num_documents):
                if i < j and similarity_matrix[i][j] > max_similarity:
                    max_similarity = similarity_matrix[i][j]
                    max_indices = (i, j)
        doc1, doc2 = max_indices
        print(f"Documents {documents[doc1]} and {documents[doc2]} have a similarity of {max_similarity}")
        similarity_matrix[doc1][doc2] = 0

if __name__ == "__main__":
    main()
