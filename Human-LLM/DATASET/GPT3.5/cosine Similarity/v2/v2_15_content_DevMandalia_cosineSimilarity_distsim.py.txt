from __future__ import division
import sys
import json
import math
import os
import numpy as np

def load_word2vec(filename):
    word2vec = {}
    with open(filename, "r") as file_in:
        for line in file_in:
            line_split = line.strip().split()
            word = line_split[0]
            vec = np.array([float(x) for x in line_split[1:]])
            word2vec[word] = vec
    return word2vec

def load_contexts(filename):
    data = {}
    with open(filename, "r") as file_in:
        for line in file_in:
            word, n, ccdict = line.strip().split("\t")
            n = int(n)
            ccdict = json.loads(ccdict)
            data[word] = ccdict
    print("File %s has contexts for %s words" % (filename, len(data)))
    return data

def cossim_sparse(v1, v2):
    numerator_sum = 0
    denominator1_sum = 0
    denominator2_sum = 0
    cossim = 0
    for word1 in v1:
        if word1 in v2:
            count1 = v1.get(word1)
            count2 = v2.get(word1)
            numerator_sum += (count1 * count2)
        denominator1_sum += (count1 * count1)
    for word2 in v2:
        count2 = v2.get(word2)
        denominator2_sum += (count2 * count2)
    cossim = numerator_sum / (math.sqrt(denominator1_sum) * math.sqrt(denominator2_sum))
    return cossim

def cossim_dense(v1, v2):
    numerator_sum = np.sum(np.multiply(v1, v2))
    denominator1_sum = np.sqrt(np.sum(np.square(v1)))
    denominator2_sum = np.sqrt(np.sum(np.square(v2)))
    return np.divide(numerator_sum, np.multiply(denominator1_sum, denominator2_sum))    

def show_nearest(word2vec, word_vector, exclude_words, similarity_metric):
    similar_words = []
    for word1 in word2vec:
        if word1 not in exclude_words:
            if similarity_metric.__name__ == "cossim_sparse":
                score = cossim_sparse(word_vector, word2vec.get(word1))
            elif similarity_metric.__name__ == "cossim_dense":
                score = cossim_dense(word_vector, word2vec.get(word1))
            else:
                raise ValueError("Invalid similarity metric")
            similar_words.append((word1, score))
    similar_words.sort(key=lambda elem: elem[1], reverse=True)
    similar_words = similar_words[:10]  # Limit to top 10 similar words
    return similar_words
