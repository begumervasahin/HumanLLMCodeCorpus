import operator
import numpy as np

class MSAI_NLP:
    def __init__(self):
        self.corpus = {}
        self.texts = []
        self.titles = []
        self.corpus_counts = []
        self.total_counts = {}
        self.lexicon_size = None
        self.mean_counts = {}
        self.presence_counts = {}
        self.tfidfs = []
        self.tfidf_sorted = None
        self.lexicon_index = []
        self.word_to_id = {}

    def clean_text(self, text):
        # Clean the text by converting to lowercase and removing special characters
        text = text.lower()
        text = text.replace("'", "' ")
        text = text.replace(':', '')
        text = text.replace('"', '')
        text = text.replace(';', '')
        text = text.replace(',', '')
        text = text.replace('.', '')
        text = text.replace('-', ' ')
        return text

    def split(self, text):
        # Split the text into words after cleaning
        text = self.clean_text(text)
        return text.split()

    def count_words(self, text):
        # Count the occurrences of each word in the given text
        counts = {}
        words = self.split(text)
        for word in words:
            counts[word] = counts.get(word, 0) + 1
        return counts

    def get_corpus_count(self):
        # Count words for each text in the corpus
        self.corpus_counts = [self.count_words(text) for text in self.texts]
        return self

    def get_total_counts(self):
        # Calculate the total counts of each word in the corpus
        self.total_counts = {}
        for text_counts in self.corpus_counts:
            for word, count in text_counts.items():
                self.total_counts[word] = self.total_counts.get(word, 0) + count
        # Sort the total counts in descending order
        self.total_counts_sorted = sorted(self.total_counts.items(), key=operator.itemgetter(1), reverse=True)
        self.lexicon_size = len(self.total_counts)
        return self

    def get_mean_counts(self):
        # Calculate the mean count of each word in the corpus
        self.mean_counts = {word: count / len(self.texts) for word, count in self.total_counts_sorted}
        return self

    def get_presence_counts(self):
        # Calculate the presence counts of each word in the corpus
        self.presence_counts = {}
        for text in self.texts:
            words = self.split(text)
            for word in words:
                self.presence_counts[word] = self.presence_counts.get(word, 0) + 1
        return self

    def get_tfidf(self):
        # Calculate TF-IDF values for each word in each text
        self.tfidfs = []
        for text_counts in self.corpus_counts:
            tfidf = {word: np.log(len(self.texts) / self.presence_counts[word]) for word in text_counts}
            self.tfidfs.append(tfidf)
        return self

    def build_lexicon_index(self):
        # Build lexicon index and word-to-id mapping
        self.lexicon_index = [(i, word) for i, word in enumerate(self.total_counts)]
        self.word_to_id = {word: i for i, (word, _) in enumerate(self.lexicon_index)}
        return self

    def build_df_vec(self):
        # Build document frequency vector
        self.df_vec = np.zeros((1, len(self.lexicon_index)))
        for i, (word, _) in enumerate(self.lexicon_index):
            self.df_vec[0, i] = self.total_counts[word]
        return self

    def build_tfs_mat(self):
        # Build TF-IDF matrix
        self.tfs_mat = np.zeros((len(self.texts), len(self.lexicon_index)))
        for j, tfidf in enumerate(self.tfidfs):
            for word, value in tfidf.items():
                id_col = self.word_to_id[word]
                self.tfs_mat[j, id_col] = value
        return self

    def text2vec(self, text, word_to_id=None):
        # Convert text to binary vector representation
        if word_to_id is None:
            word_to_id = self.word_to_id
        split = self.split(text)
        vec = np.zeros((1, len(word_to_id)))
        for word in split:
            if word in word_to_id:
                id_col = word_to_id[word]
                vec[0, id_col] = 1
            else:
                print('Word not in lexicon:', word)
        return vec

    def texts2matrix(self, texts):
        # Process texts to build TF-IDF matrix
        if not isinstance(texts, list):
            raise ValueError('texts must be a list')
        if not all(isinstance(text, str) for text in texts):
            raise ValueError('texts must be a list of strings')
        self.texts = texts
        self.get_corpus_count().get_total_counts().get_presence_counts()
        self.get_tfidf().build_lexicon_index().build_df_vec().build_tfs_mat()

    def sort_tfidf(self):
        # Sort TF-IDF values for each text
        self.tfidf_sorted = sorted(self.tfidfs[0].items(), key=operator.itemgetter(1), reverse=True)

# Example usage:
nlp = MSAI_NLP()
texts = ["This is a sample text for testing.", "Another sample text for testing purposes."]
nlp.texts2matrix(texts)
print(nlp.tfs_mat)
