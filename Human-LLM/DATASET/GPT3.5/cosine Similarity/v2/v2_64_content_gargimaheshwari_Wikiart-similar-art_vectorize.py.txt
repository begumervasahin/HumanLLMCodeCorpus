import sys
import os
import pickle
from PIL import Image
from tqdm import tqdm
from img2vec import Img2Vec

def img2vec_converter(model, cuda):
    # Initialize Img2Vec object
    img2vec = Img2Vec(cuda=cuda, model=model)
    vectors = {}

    print("Converting images to vectors:")
    # Iterate over images in the 'resized' directory
    for image in tqdm(os.listdir("resized")):
        img_path = os.path.join("resized", image)
        try:
            # Open the image
            img = Image.open(img_path)
            # Convert the image to a vector representation
            vec = img2vec.get_vec(img)
        except (RuntimeError, FileNotFoundError):
            # Handle errors during image processing
            print(f"Error processing image: {img_path}")
            continue
        else:
            # Check vector size and add to dictionary
            assert len(vec) == img2vec.layer_output_size
            vectors[image] = vec
            img.close()

    # Ensure all images have been processed
    assert len(vectors) == len(os.listdir("resized"))

    print("Saving image vectors...")
    # Create directories to save pickle files
    os.makedirs(os.path.join("pickles", model), exist_ok=True)
    # Save vectors to a pickle file
    with open(os.path.join('pickles', model, 'vectors.pkl'), 'wb') as file:
        pickle.dump(vectors, file)

    print("Image vectors saved successfully.")
    return vectors

if __name__ == "__main__":
    # Specify the model and CUDA availability
    model = "vgg16"  # Example model, replace with the desired model
    cuda = True  # Set to True if CUDA is available, False otherwise

    # Convert images to vectors
    img2vec_converter(model, cuda)
