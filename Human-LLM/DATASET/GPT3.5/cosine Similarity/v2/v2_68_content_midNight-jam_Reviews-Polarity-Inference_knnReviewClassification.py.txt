import pandas as pd
from sklearn.feature_extraction.text import CountVectorizer, TfidfTransformer
from nltk.corpus import stopwords
import re
from scipy import spatial
import random

# Define global variables
cachedStopWords = stopwords.words("english")
columns = ['label', 'document']
docIdLabel = {}

def replace_special_character(document):
    # Replace special characters with space and remove periods
    result = re.sub('[^a-zA-Z\n\.]', ' ', document).replace(".", "")
    # Remove extra spaces and newlines
    result = ' '.join(result.split())
    result = "".join(result.splitlines())
    # Remove words with less than 3 characters
    result = re.sub(r'\b\w{1,3}\b', '', result)
    return result.strip()

def remove_stopword(document):
    # Remove stopwords from the document
    text = ' '.join([word for word in document.lower().split() if word not in cachedStopWords])
    return text

def read_document(file_path):
    # Read and preprocess documents from file
    data = []
    with open(file_path) as fl:
        idx = 0
        for lines in fl:
            temp = [lines[:2], lines[3:].strip()]
            data.append(temp)
            docIdLabel[idx] = lines[:2]
            idx += 1
    df = pd.DataFrame(data, columns=columns)
    # Apply preprocessing steps to the document column
    df['document'] = df['document'].apply(lambda x: remove_stopword(x))
    df['document'] = df['document'].apply(lambda x: replace_special_character(x))
    # Vectorize and transform documents using CountVectorizer and TfidfTransformer
    vectorizer = CountVectorizer(min_df=1, max_features=2000)
    X = vectorizer.fit_transform(df['document'].values)
    transformer = TfidfTransformer(smooth_idf=False)
    tfidf = transformer.fit_transform(X)
    return tfidf

def read_test_document(location):
    # Read and preprocess test documents
    data = []
    with open(location) as fl:
        for lines in fl:
            temp = lines.strip()
            data.append(temp)
    df = pd.DataFrame(data, columns=['document'])
    df['document'] = df['document'].apply(lambda x: remove_stopword(x))
    df['document'] = df['document'].apply(lambda x: replace_special_character(x))
    vectorizer = CountVectorizer(min_df=1, max_features=2000)
    X = vectorizer.fit_transform(df['document'].values)
    transformer = TfidfTransformer(smooth_idf=False)
    tfidf = transformer.fit_transform(X)
    return tfidf

def cosine_similarity(d1, d2):
    # Compute cosine similarity between two vectors
    return 1 - spatial.distance.cosine(d1, d2)

def k_nearest_neighbours(train_vector, test_vector, k):
    # Find k nearest neighbors for a test vector in the training set
    all_distances = {}
    for t in range(len(train_vector)):
        dist = cosine_similarity(train_vector[t], test_vector)
        all_distances[t] = dist
    return [(k, all_distances[k]) for k in sorted(all_distances, key=lambda x: all_distances[x], reverse=True)][:k]

def get_predicted_label(near_docs):
    # Generate predicted label based on nearest neighbors
    pos = sum(1 for k, v in near_docs if docIdLabel[k] == '+1')
    neg = sum(1 for k, v in near_docs if docIdLabel[k] == '-1')
    if pos == neg:
        return random.choice(['+1', '-1'])
    else:
        return '+1' if pos > neg else '-1'

if __name__ == '__main__':
    # Read train and test documents
    final_train_tf = read_document("data/train.dat")
    print("Train documents read completed")
    final_test_tf = read_test_document("data/test.dat")
    print("Test documents read completed")
    final_train_tf_arr = final_train_tf.toarray()
    final_test_tf_arr = final_test_tf.toarray()
    k = 5
    # Classify test documents and write results to file
    with open('result.dat', 'w') as f:
        for test_vector in final_test_tf_arr:
            nearest = k_nearest_neighbours(final_train_tf_arr, test_vector, k)
            lab = get_predicted_label(nearest)
            f.write(lab + "\n")
            print("Calculated Label:", lab)
