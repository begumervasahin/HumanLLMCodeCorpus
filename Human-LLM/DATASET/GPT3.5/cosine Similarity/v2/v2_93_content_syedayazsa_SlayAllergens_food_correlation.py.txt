import pandas as pd
from pymongo import MongoClient

# Connect to MongoDB
client = MongoClient()
db = client.project
collection_filefood = db.filefood
collection_newingredient = db.newingredient

# Load data from MongoDB collections
df = pd.DataFrame(list(collection_filefood.find()))
del df['_id']
df.columns = ['rating', 'item_id', 'user_id']

food_titles = pd.DataFrame(list(collection_newingredient.find()))
del food_titles['_id']

# Merge dataframes
data = pd.merge(df, food_titles, on='item_id')
data['rating'] = pd.to_numeric(data['rating'])
data['item_id'] = pd.to_numeric(data['item_id'])
data['user_id'] = pd.to_numeric(data['user_id'])

# Calculate mean rating and count of ratings for each title
ratings = pd.DataFrame(data.groupby('title')['rating'].mean())
ratings['num of ratings'] = pd.DataFrame(data.groupby('title')['rating'].count())       

# Create a pivot table for collaborative filtering
foodmat = data.pivot_table(index='user_id', columns='title', values='rating')

# Get input from user for the food item
item = input("Enter the name of the food item: \n")
item_user_ratings = foodmat[item]

# Calculate correlation with other items
similar_to_item = foodmat.corrwith(item_user_ratings)
corr_item = pd.DataFrame(similar_to_item, columns=['Correlation'])
corr_item.dropna(inplace=True)

# Merge with ratings dataframe and filter items with more than 100 ratings
corr_item = corr_item.join(ratings['num of ratings'])
corr_item = corr_item[corr_item['num of ratings'] > 100]

# Sort items by correlation and print top recommendations
corr_item = corr_item.sort_values('Correlation', ascending=False)
print(corr_item.iloc[:, 0].head(25))
