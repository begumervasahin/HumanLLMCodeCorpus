import re
import math
from collections import Counter

# Regular expression pattern to find words
WORD = re.compile(r'\w+')


def get_cosine_similarity(vector1, vector2):
    """
    Calculate the cosine similarity between two vectors.
    
    Args:
    - vector1 (Counter): First vector
    - vector2 (Counter): Second vector
    
    Returns:
    - float: Cosine similarity value
    """
    intersection = set(vector1.keys()) & set(vector2.keys())
    numerator = sum(vector1[word] * vector2[word] for word in intersection)
    sum1 = sum(vector1[word] ** 2 for word in vector1.keys())
    sum2 = sum(vector2[word] ** 2 for word in vector2.keys())
    denominator = math.sqrt(sum1) * math.sqrt(sum2)
    
    if denominator == 0:
        return 0.0
    else:
        return float(numerator) / denominator


def text_to_vector(text):
    """
    Convert text to a vector representation.
    
    Args:
    - text (str): Input text
    
    Returns:
    - Counter: Vector representation of the text
    """
    words = WORD.findall(text)
    return Counter(words)


def calculate_angle_between_texts(text1, text2):
    """
    Calculate the angle (in degrees) between two texts.
    
    Args:
    - text1 (str): First text
    - text2 (str): Second text
    
    Returns:
    - float: Angle between the texts in degrees
    """
    vector1 = text_to_vector(text1)
    vector2 = text_to_vector(text2)
    cosine_similarity = get_cosine_similarity(vector1, vector2)
    angle_in_radians = math.acos(cosine_similarity)
    
    return math.degrees(angle_in_radians)


# Example texts
text1 = 'This is a foo bar sentence.'
text2 = 'This sentence is similar to a foo bar sentence.'
text3 = 'A string that should not be close to the others!'

# Calculate angles
angle1 = calculate_angle_between_texts(text1, text2)
angle2 = calculate_angle_between_texts(text1, text3)

# Print results
print("String 1: " + text1)
print("String 2: " + text2)
print("String 3: " + text3)
print("Cosine similarity angle between string 1 and 2: " + str(angle1))
print("Cosine similarity angle between string 1 and 3: " + str(angle2))
