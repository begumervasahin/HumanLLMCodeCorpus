from __future__ import division
import json
import math
import numpy as np

def load_word2vec(filename):
    word_vectors = {}
    with open(filename, "r") as file_in:
        for line in file_in:
            line_parts = line.strip().split()
            word = line_parts[0]
            vector = np.array([float(x) for x in line_parts[1:]])
            word_vectors[word] = vector
    return word_vectors

def load_contexts(filename):
    contexts_data = {}
    for word, context_dict in stream_contexts(filename):
        contexts_data[word] = context_dict
    print(f"File {filename} has contexts for {len(contexts_data)} words")
    return contexts_data

def stream_contexts(filename):
    with open(filename, "r") as file_in:
        for line in file_in:
            word, n, context_json = line.strip().split("\t")
            n = int(n)
            context_dict = json.loads(context_json)
            yield word, context_dict

def calculate_cosine_similarity_sparse(v1, v2):
    numerator = sum(v1[key] * v2[key] for key in v1 if key in v2)
    denominator_v1 = math.sqrt(sum(v1[key] ** 2 for key in v1))
    denominator_v2 = math.sqrt(sum(v2[key] ** 2 for key in v2))
    similarity = numerator / (denominator_v1 * denominator_v2)
    return similarity

def calculate_cosine_similarity_dense(v1, v2):
    V1 = np.array(v1)
    V2 = np.array(v2)
    numerator = sum(v for v in V1 * V2)
    denominator = np.linalg.norm(V1) * np.linalg.norm(V2)
    similarity = numerator / denominator
    return similarity

def show_nearest(word_vectors, target_vector, exclude_words, similarity_metric):
    similarities = {}
    for word, vector in word_vectors.items():
        if word not in exclude_words:
            similarity = similarity_metric(vector, target_vector)
            similarities[word] = similarity
    sorted_similarities = sorted(similarities.items(), key=lambda x: x[1], reverse=True)
    return sorted_similarities[:10]
