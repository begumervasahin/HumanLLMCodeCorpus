from math import sqrt

def cosine_similarity(v1, v2):
    # Initialize variables for numerator and vector sums
    numerator_sum = 0
    v1_sum = 0
    v2_sum = 0
    
    # Calculate the numerator and vector sums
    for index, value in enumerate(v1):
        numerator_sum += value * v2[index]
        v1_sum += value ** 2
        v2_sum += v2[index] ** 2
    
    # Calculate the denominator
    denominator_total = sqrt(v1_sum) * sqrt(v2_sum)
    
    # Compute and return the cosine similarity
    similarity = numerator_sum / denominator_total
    return similarity

def jaccard_similarity(v1, v2):
    # Initialize variables for numerator and denominator sums
    numerator_min_sum = 0
    denominator_max_sum = 0
    
    # Calculate the numerator and denominator sums
    for index, value in enumerate(v1):
        numerator_min_sum += min(value, v2[index])
        denominator_max_sum += max(value, v2[index])
    
    # Compute and return the Jaccard similarity
    similarity = numerator_min_sum / denominator_max_sum
    return similarity

def dice_similarity(v1, v2):
    # Initialize variables for numerator and denominator sums
    numerator_min_sum = 0
    denominator_sum = 0
    
    # Calculate the numerator and denominator sums
    for index, value in enumerate(v1):
        numerator_min_sum += min(value, v2[index])
        denominator_sum += value + v2[index]
    
    # Compute and return the Dice similarity
    similarity = (2 * numerator_min_sum) / denominator_sum
    return similarity
