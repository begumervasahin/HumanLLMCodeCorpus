from collections import defaultdict

# Define the documents and index
documents = [['dog', 'deer', 'cat', 'mouse'], ['parrot', 'cat'], ['turtle', 'dog']]
index = {'dog': [0, 2], 'cat': [0, 1], 'parrot': [1], 'turtle': [2]}

# Function for AND search using list comprehension
def and_search1(documents, index, queries):
    # Initialize a set with document IDs matching the first query
    doc_ids = set(index[queries[0]])

    # Perform AND search for subsequent queries
    for query in queries[1:]:
        doc_ids = doc_ids & set(index[query])

    # Return documents corresponding to the matching document IDs
    return [documents[doc_id] for doc_id in doc_ids]

# Function for AND search using Counter from collections module
def and_search2(documents, index, queries):
    # Create a Counter to count occurrences of document IDs
    c = defaultdict(int)
    for q in queries:
        c.update(index[q])

    # Get the most common document ID and return the corresponding document
    return [documents[c.most_common()[0][0]]]

# Test the AND search functions
print("AND Search 1:", and_search1(documents, index, ['cat', 'dog']))
print("AND Search 2:", and_search2(documents, index, ['cat', 'dog']))

# Calculating the dot product of two lists
dot_product_sum = 0

    dot_product_sum += x * y
