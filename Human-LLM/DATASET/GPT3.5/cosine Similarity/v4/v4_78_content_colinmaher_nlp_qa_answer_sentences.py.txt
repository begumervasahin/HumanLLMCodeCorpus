def choose_sentence(question, story):
    # Determine the difficulty level of the question
    diff = question['difficulty']

    # Get the best sentence using WordNet
    sentence = get_best_wordnet_sent(question, story)

    if diff == 'Discourse':
        print('Processing discourse question')

        # Get the sentences from the story based on availability of 'sch'
        if isinstance(story["sch"], str):
            sentences = story["sch"]
        else:
            sentences = story["text"]
        
        sentences = nltk.sent_tokenize(sentences)
        sent_ind = 0

        # Find the index of the selected sentence
        for i, sent in enumerate(sentences):
            if sent == sentence:
                sent_ind = i
                break

        print("sent_ind:", sent_ind)

        # Extract the discourse type from the question text
        discourse_type = ''
        qwords = nltk.word_tokenize(question['text'])
        lowered_qwords = [qword.lower() for qword in qwords]

        print("tokenized qwords:", lowered_qwords)

        if 'after' in lowered_qwords:
            discourse_type = 'after'
        elif 'first' in lowered_qwords:
            discourse_type = 'first'
        elif 'before' in lowered_qwords:
            discourse_type = 'before'

        print('discourse_type:', discourse_type)

        sent_words = nltk.word_tokenize(sentence)
        lowered_sent = [word.lower() for word in sent_words]

        # Check if discourse type is present in the sentence
        if discourse_type in sent_words:
            return sentence
        else:
            # Return sentences based on discourse type and selected sentence index
            if discourse_type == 'after' and sent_ind < len(sentences):
                print('returning sentence after wordnet matched sentence')
                return sentences[sent_ind:sent_ind+2]
            elif (discourse_type == 'first' or discourse_type == 'before') and sent_ind > 0:
                print('returning sentence before wordnet matched sentence')
                return sentences[sent_ind-1:sent_ind+1]
            elif sent_ind > 0:
                return sentences[sent_ind-1:sent_ind+1]

    return sentence

def baseline(qbow, sentences, stopwords):
    # Calculate overlap between question bag-of-words and sentence bag-of-words
    answers = [(len(qbow & get_bow(sent, stopwords)), sent) for sent in sentences]

    # Sort answers based on overlap count in descending order
    answers.sort(key=operator.itemgetter(0), reverse=True)

    # Get the best answer (sentence with highest overlap)
    best_answer = answers[0][1]

    return best_answer

def get_best_wordnet_sent(question, story, use_sch=True):
    stemmer = PorterStemmer()
    best_sent = None
    best_score = 0.0

    # Determine the source of sentences based on 'sch' availability
    if isinstance(story["sch"], str) and use_sch:
        sentences = story["sch"]
        print("using sch")
    else:
        sentences = story["text"]
        print("using text")

    sentences = nltk.sent_tokenize(sentences)
    qwords = nltk.word_tokenize(question['text'])
    qwords = get_bow(get_sentences(question['text'])[0], stopwords)

    for i, sent in enumerate(wn_story_dict[question['sid']]):
        sent_score = 0.0
        words_not_found = set(qwords)

        # Calculate the score for each word in the sentence
        for qword in qwords:
            for word in sent:
                if str(qword) == word:
                    sent_score += 1
                    words_not_found.remove(qword)

        # Check Word2Vec similarity for words not found in WordNet
        for qword in words_not_found:
            highest_sim = 0
            for word in sent:
                if word in model.vocab and qword in model.vocab:
                    sim = model.similarity(word, str(qword))
                    if sim > highest_sim:
                        highest_sim = sim

            if highest_sim > 0.3:
                sent_score += highest_sim

        # Update best sentence based on the score
        if sent_score > best_score:
            best_score = sent_score
            best_sent = sentences[i]

        print(sent)

    # Recursively call with 'sch' if no sentence found and 'use_sch' is True
    if best_sent is None and use_sch:
        return get_best_wordnet_sent(question, story, False)

    return best_sent
