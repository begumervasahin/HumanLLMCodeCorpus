import numpy as np

# Initialize lists and dictionaries
documents_name = []
discrete_words = []
documents_dictionary = {}
common_words = {}
counter_list = []

def cosine_similarity(document1, document2):
    """Calculate cosine similarity between two vectors."""
    norm_doc1 = np.linalg.norm(document1)
    norm_doc2 = np.linalg.norm(document2)
    dot_product = np.dot(document1, document2)
    vector_length_product = norm_doc1 * norm_doc2
    cos_sim = dot_product / vector_length_product
    return cos_sim

# Input the number of documents
while True:
    try:
        number_of_documents = int(input("Enter the number of documents: "))
        if number_of_documents < 2:
            number_of_documents = 2
        break
    except ValueError:
        print("Invalid input. Please enter an integer.")

# Input document names and content
for i in range(number_of_documents):
    while True:
        try:
            file_name = input(f"Enter the name of document {i + 1} (including filename extension): ")
            current_document = open(file_name, "r")
            documents_name.append(file_name)
            document_content = current_document.read().lower().split()
            for word in document_content:
                if word not in discrete_words:
                    discrete_words.append(word)
            documents_dictionary[file_name] = document_content
            current_document.close()
            break
        except FileNotFoundError:
            print("File not found. Please try again.")

# Calculate word counts for each document
for document in documents_dictionary:
    for word in discrete_words:
        counter_list.append(documents_dictionary[document].count(word))
    common_words[document] = counter_list
    counter_list = []

# Calculate cosine similarity between document pairs
cosine_result_dictionary = {}
for doc1 in range(len(documents_dictionary)):
    for doc2 in range(doc1 + 1, len(documents_dictionary)):
        cosine_result = cosine_similarity(common_words[documents_name[doc1]], common_words[documents_name[doc2]])
        key_str = f". <{documents_name[doc1]}> <{documents_name[doc2]}> : "
        cosine_result_dictionary[key_str] = cosine_result

# Prompt user for how many similar document pairs to view
max_k = len(cosine_result_dictionary)
k = int(input("Enter how many of the most similar document pairs you want to view: "))   
k = max(min(k, max_k), 1)

# Print the most similar document pairs
sorted_pairs = sorted(cosine_result_dictionary.items(), key=lambda x: x[1], reverse=True)
for i in range(k):
    key, value = sorted_pairs[i]
    print(f"{i + 1}. {key}{value}")
