from math import sqrt

def shot_percentage_similarity(prospect, nba_player):
    numerator = prospect['FGP'] * nba_player['FGP']
    numerator += prospect['2PP'] * nba_player['2PP']
    numerator += prospect['FTP'] * nba_player['FTP']

    prosp_denom = prospect['FGP']**2 + prospect['2PP']**2 + prospect['FTP']**2
    comp_denom = nba_player['FGP']**2 + nba_player['2PP']**2 + nba_player['FTP']**2

    if (nba_player['3PP'] == 0.000 or nba_player['3PP'] is None) and prospect['3PA'] <= 1.0:
        return numerator / (sqrt(prosp_denom) * sqrt(comp_denom))
    elif (nba_player['3PP'] == 0.000 or nba_player['3PP'] is None) or (nba_player['3PA'] < 1.0 and prospect['3PA'] > 1.0):
        return 0
    else:
        numerator += prospect['3PP'] * nba_player['3PP']
        comp_denom += nba_player['3PP']**2
        prosp_denom += prospect['3PP']**2

    denom = sqrt(prosp_denom) * sqrt(comp_denom)
    return numerator / denom

def per_game_similarity(prospect, nba_player):
    numerator = prospect['PTS'] * nba_player['PTS']
    numerator += prospect['AST'] * nba_player['AST']
    numerator += prospect['TRB'] * nba_player['TRB']
    numerator += prospect['STL'] * nba_player['STL']
    numerator += prospect['BLK'] * nba_player['BLK']
    numerator += prospect['FTA'] * nba_player['FTA']

    prosp_denom = (
        prospect['PTS']**2 + prospect['AST']**2 + prospect['TRB']**2 +
        prospect['STL']**2 + prospect['BLK']**2 + prospect['FTA']**2
    )
    comp_denom = (
        nba_player['PTS']**2 + nba_player['AST']**2 + nba_player['TRB']**2 +
        nba_player['STL']**2 + nba_player['BLK']**2 + nba_player['FTA']**2
    )

    if nba_player['3PA'] <= 1.0 and prospect['3PA'] <= 1.0:
        return numerator / (sqrt(prosp_denom) * sqrt(comp_denom))
    elif nba_player['3PA'] <= 1.0 and prospect['3PA'] > 1.0:
        return 0
    else:
        numerator += prospect['3PA'] * nba_player['3PA']
        prosp_denom += prospect['3PA']**2
        comp_denom += nba_player['3PA']**2

    denom = sqrt(prosp_denom) * sqrt(comp_denom)
    return numerator / denom

        nba_player['ORB']**2 + nba_player['DRB']**2 + nba_player['AST']**2 +
        nba_player['STL']**2 + nba_player['BLK']**2 + nba_player['TOV']**2 +
        nba_player['USG']**2
    )

    denom = sqrt(prosp_denom) * sqrt(comp_denom)
    return numerator / denom
