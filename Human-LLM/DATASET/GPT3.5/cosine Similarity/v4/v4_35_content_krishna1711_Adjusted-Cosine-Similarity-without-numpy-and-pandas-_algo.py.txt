import csv
import copy

# Open the CSV file and read its contents into exampleData list
exampleFile = open('sample.csv')
exampleReader = csv.reader(exampleFile)
exampleData = list(exampleReader)

# Calculate the number of rows and columns in the CSV data
num_rows = len(exampleData) - 1
num_cols = len(exampleData[0]) - 1

# Initialize lists and variables for calculations
Listnn = []
avg = []
S_items = [[0 for _ in range(num_rows)] for _ in range(num_rows)]

# Deep copy the original data for modification
Listcpy = copy.deepcopy(exampleData)

# Calculate row averages and subtract from non-zero elements
for i in range(1, num_rows + 1):
    row_sum = 0
    count = 0
    for j in range(1, num_cols + 1):
        value = float(Listcpy[i][j])
        if value != 0.0:
            row_sum += value
            count += 1
    avg.append(row_sum / count if count != 0 else 0.0)

    for j in range(1, num_cols + 1):
        if float(Listcpy[i][j]) != 0.0:
            Listcpy[i][j] = float(Listcpy[i][j]) - float(avg[i - 1])

# Prepare modified data for similarity matrix calculation
Listnn = [[float(Listcpy[i][j]) for j in range(1, num_cols + 1)] for i in range(1, num_rows + 1)]

# Calculate similarities between rows and populate the similarity matrix
for z in range(num_rows - 1):
    a = z
    while a < num_rows - 1:
        tmplst1 = Listnn[z][:]
        tmplst2 = Listnn[a + 1][:]
        num = sum(x * y for x, y in zip(tmplst1, tmplst2))
        den1 = sum(x ** 2 for x in tmplst1)
        den2 = sum(x ** 2 for x in tmplst2)

        if den1 != 0 and den2 != 0:
            Ssum = num / (den1 ** 0.5 * den2 ** 0.5)
        else:
            Ssum = -5.0

        S_items[z][a + 1] = S_items[a + 1][z] = Ssum
        for i in range(num_rows):
            S_items[i][i] = 1.0  # Set diagonal elements to 1.0

        a += 1

# Print the final similarity matrix
print('\nFinal Similarity Matrix\n\n', S_items)
