from nltk.tokenize import word_tokenize
import os
import string
from nltk.corpus import stopwords
from nltk.stem.porter import PorterStemmer
from sklearn.feature_extraction.text import TfidfVectorizer

def get_tokenized(text):
    """Tokenize the input text."""
    tokens = word_tokenize(text)
    return tokens

def stemming(tokens, stemmer):
    """Apply stemming to the input tokens using the provided stemmer."""
    stemmed = [stemmer.stem(item) for item in tokens]
    return stemmed

def preprocessing(text):
    """Preprocess the input text by tokenizing and stemming."""
    tokens = get_tokenized(text)
    stemmer = PorterStemmer()
    stem = stemming(tokens, stemmer)
    return stem

def training_tfs():
    """Train TF-IDF model on text files in a specified directory."""
    file_list = []  # List to store file names
    token_dictionary = {}  # Dictionary to store preprocessed text for each file

    # Define the path to the directory containing text files
    path = 'C:\\Users\\Kapil Goyal\\Desktop\\nlp_assignment\\files'

    # Iterate over files in the directory
    for subdir, dirs, files in os.walk(path):
        for file in files:
            file_path = os.path.join(subdir, file)
            with open(file_path, 'r') as f:
                # Read the file content
                content = f.read()

                # Convert to lowercase and remove punctuation
                lowers = content.lower()
                no_punctuation = lowers.translate(str.maketrans('', '', string.punctuation))

                # Store preprocessed text in the token dictionary
                token_dictionary[file] = no_punctuation

                # Append file name to the file list
                file_list.append(file)

    # Initialize and fit the TF-IDF vectorizer
    tfidf = TfidfVectorizer(tokenizer=preprocessing, stop_words='english')
    tfs = tfidf.fit_transform(token_dictionary.values())

    return tfidf, tfs, file_list

# Call the training_tfs function
tfidf_model, tfs_matrix, file_list = training_tfs()
