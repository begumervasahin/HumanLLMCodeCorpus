import pandas as pd
import numpy as np
from models import R, nmf, DF

def recommend(user_input_dict):
    # Get sorted list of movie IDs
    movie_ids = sorted(list(R.columns))

    # Initialize dictionary with movie IDs and empty ratings
    ratings_dict = {movie_id: np.nan for movie_id in movie_ids}

    # Update ratings dictionary with user input ratings
    for key, value in user_input_dict.items():
        ratings_dict[key] = value

    # Convert ratings dictionary to a DataFrame
    user_ratings = pd.DataFrame([list(ratings_dict.values())], index=[0], columns=movie_ids)

    # Fill missing values with 3.0 and transpose the DataFrame
    profile_filled = user_ratings.fillna(3.0).transpose()

    # Transform user profile using NMF
    hidden_profile = nmf.transform(profile_filled)

    # Predict ratings using hidden profile and NMF components
    prediction = np.dot(hidden_profile, nmf.components_)
    coeff_profile = pd.DataFrame(prediction, columns=R.columns)

    # Get movies not watched by the user
    movies_not_watched = coeff_profile.transpose().iloc[0][np.isnan(profile_filled.iloc[0])]
    sorted_movies = movies_not_watched.sort_values(ascending=False)
    recommended_movie_ids = sorted_movies.index[:5]

    # Get movie titles from DataFrame
    recommendations = [DF.loc[movie_id, 'title'] for movie_id in recommended_movie_ids]

    return recommendations

# Example usage
user_input_dict = {1: 5, 2: 4, 3: 3, 4: 2, 5: 1}  # User ratings for movies
recommended_movies = recommend(user_input_dict)

# Print recommended movies
print("Recommended Movies:")
for movie in recommended_movies:
    print(movie)
