import ujson as json
from collections import defaultdict

# Read Twitter and New York Times counts
with open("twcounts") as tw_file:
    tw = [(w, int(c)) for w, c, _ in (line.split("\t") for line in tw_file)]

with open("nytcounts") as nyt_file:
    nyt = [(w, int(c)) for w, c, _ in (line.split("\t") for line in nyt_file)]

# Sort words by count
tw_words = sorted([(w, c) for w, c in tw if c > 100], key=lambda x: -x[1])
nyt_words = sorted([(w, c) for w, c in nyt if c > 100], key=lambda x: -x[1])

# Create dictionaries for counts
tw_counts = dict(tw)
nyt_counts = dict(nyt)

# Create a vocabulary of 4000 words
def all_words():
    for i in range(min(len(tw_words), len(nyt_words))):
        yield tw_words[i][0]
        yield nyt_words[i][0]

vocab = set()
for w in all_words():
    vocab.add(w)
    if len(vocab) == 4000:
        break

# Write the vocabulary to a file
with open("vocab", 'w') as f:
    counts = [(w, nyt_counts.get(w, 0)) for w in vocab]
    counts.sort(key=lambda x: -x[1])
    counts = [(w, c) for w, c in counts if c > 0]
    for w, c in counts:
        print(w, c, file=f)

# Filter counts based on vocabulary
def filter_counts(infile, outfile, vocab):
    print("%s -> %s" % (infile, outfile))
    ctx_numwords = defaultdict(int)
    w_count = {}
    w_cc = {}
    with open(infile) as inf:
        for line in inf:
            w, c, cc = line.split("\t")
            if w not in vocab:
                continue
            cc = json.loads(cc)
            for ctx in cc:
                ctx_numwords[ctx] += 1
            w_cc[w] = cc
            w_count[w] = c
    with open(outfile, 'w') as out:
        for w, cc in w_cc.items():
            for ctx in list(cc.keys()):
                if ctx_numwords[ctx] <= 1:
                    del cc[ctx]
            print("%s\t%s\t%s" % (w, w_count[w], json.dumps(cc)), file=out)

# Filter counts for trimmed files and files with specific words
filter_counts("twcounts", "twcounts.trimmed", vocab)
filter_counts("nytcounts", "nytcounts.trimmed", vocab)
filter_counts("twcounts", "twcounts.university_cat_dog", {"university", "cat", "dog"})
filter_counts("nytcounts", "nytcounts.university_cat_dog", {"university", "cat", "dog"})
