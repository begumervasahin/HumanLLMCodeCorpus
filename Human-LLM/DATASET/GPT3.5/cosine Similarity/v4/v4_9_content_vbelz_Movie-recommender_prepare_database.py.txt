import os
import re
import pandas as pd
from sqlalchemy import create_engine

# Define the folder where data files are located
data_folder = './data/'

# Read movie and rating data from CSV files
df_movies = pd.read_csv(data_folder + 'movies.csv')
df_ratings = pd.read_csv(data_folder + 'ratings.csv')

# Choose whether to save data to PostgreSQL or SQLite
save_to_postgres = False
save_to_sqlite = True

# Merge movie and rating dataframes
df_merge = pd.merge(df_ratings, df_movies, how='outer', left_on='movieId', right_on='movieId')

# Function to extract year from movie titles
def extract_year_from_title(title):
    '''Extract year from title'''
    year_match = re.findall("\(([0-9]{4})\)", title)
    return year_match[0] if year_match else 0

# Apply the extract_year_from_title function to extract years and clean movie titles
df_merge['year'] = df_merge['title'].apply(extract_year_from_title)
df_merge['title'] = df_merge['title'].str.replace("\(([0-9]{4})\)", "")

# Select columns of interest and sort by movieId
selected_columns = ['userId', 'movieId', 'rating', 'title', 'genres', 'year', 'timestamp']
df_merge = df_merge[selected_columns]
df_merge.sort_values(by='movieId', inplace=True)

# Remove rows with missing userId values
df_merge.dropna(subset=['userId'], inplace=True)

# Save data to PostgreSQL if save_to_postgres is True
if save_to_postgres:
    db_connection = create_engine('postgres://localhost:5432/movielens')
    df_merge.to_sql('movies_ratings', db_connection, index=False)

# Save data to SQLite if save_to_sqlite is True
if save_to_sqlite:
    db_connection_lite = create_engine('sqlite:///data/movielens.db')
    df_merge.to_sql('movie_ratings', db_connection_lite, index=False)

# Remove the original CSV files after processing
os.remove(os.path.join(data_folder, 'movies.csv'))
os.remove(os.path.join(data_folder, 'ratings.csv'))
