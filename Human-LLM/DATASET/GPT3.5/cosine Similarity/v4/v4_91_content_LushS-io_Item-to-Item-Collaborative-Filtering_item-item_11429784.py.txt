import pandas as pd
import numpy as np
import scipy.sparse as sps
from sklearn.metrics.pairwise import cosine_similarity
from sklearn.neighbors import NearestNeighbors
import time

# Sample data
user_data = {
    'user_1': [1, 0, 3, 0, 0, 5, 0, 0, 5, 0, 4, 0],
    'user_2': [0, 0, 5, 4, 0, 0, 4, 0, 0, 2, 1, 3],
    'user_3': [2, 4, 0, 1, 2, 0, 3, 0, 4, 3, 5, 0],
    'user_4': [0, 2, 4, 0, 5, 0, 0, 4, 0, 0, 2, 0],
    'user_5': [0, 0, 4, 3, 4, 2, 0, 0, 0, 0, 2, 5],
    'user_6': [1, 0, 3, 0, 3, 0, 0, 2, 0, 0, 4, 0]
}

# Create DataFrame and sparse matrix
pd_df = pd.DataFrame(data=user_data)
np_mov_rat = sps.csr_matrix(pd_df)

# Load additional data
links = pd.read_csv("./movie-lens-data/links.csv")
movies = pd.read_csv("./movie-lens-data/movies.csv")
ratings = pd.read_csv("./movie-lens-data/ratings.csv")
tags = pd.read_csv("./movie-lens-data/tags.csv")

# Merge and process data
df = pd.merge(movies, ratings, on='movieId', how='outer')
df_mov_rat = df[['rating', 'movieId', 'userId']].copy()
df = df_mov_rat.pivot(index='userId', columns='movieId', values='rating').fillna(0)

# Calculate average and center CSR matrix
csr_mat = sps.csr_matrix(np_mov_rat)
(x, y, z) = sps.find(csr_mat)
counts = np.bincount(x)
sums = np.bincount(x, weights=z)
csr_avg = sums / counts

X = csr_mat
nnz_per_row = np.diff(X.indptr)
Y = sps.csr_matrix((X.data - np.repeat(csr_avg, nnz_per_row), X.indices, X.indptr), shape=X.shape)

# Calculate cosine similarity
start_time = time.time()
A = Y
similarity = A.dot(A.T).toarray()
square_mag = similarity.diagonal()
inv_square_mag = 1 / square_mag
inv_square_mag[np.isinf(inv_square_mag)] = 0
inv_mag = np.sqrt(inv_square_mag)
cosine = similarity * inv_mag
cosine = cosine.T * inv_mag
end_time = time.time()
cosine_mat = cosine.T
run_time = end_time - start_time

# Calculate cosine similarity using sklearn
start = time.time()
test = cosine_similarity(Y)
end = time.time()
total_time = end - start

# Create COO matrix and fit Nearest Neighbors model
coo_cos = sps.coo_matrix(cosine_mat)
knn = 5
knn_model = NearestNeighbors(n_neighbors=knn)
knn_model.fit(coo_cos)
