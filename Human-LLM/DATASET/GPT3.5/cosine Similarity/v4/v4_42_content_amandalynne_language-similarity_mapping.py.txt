class Mapping:
    def __init__(self, items):
        # Initialize the item list, items dictionary, and index dictionary
        self.item_list = items
        self.items = {index: item for index, item in enumerate(items)}
        self.index = {item: index for index, item in enumerate(items)}

    def __contains__(self, item):
        # Check if the item exists in the index dictionary
        return item in self.index

    def __getitem__(self, item):
        # Get the index of the item from the index dictionary
        return self.index[item]

    def __iter__(self):
        # Iterate through the index dictionary
        return iter(self.index)

    def __len__(self):
        # Get the length of the index dictionary
        return len(self.index)

    def __repr__(self):
        # Provide a string representation of the Mapping object
        return f"<{self.__class__.__name__} with {len(self)} entries>"

    def add(self, item):
        # Add a new item to the Mapping object if it doesn't already exist
        if item not in self:
            index = len(self)
            self.items[index] = item
            self.index[item] = index
        return item

    def get(self, item, default=None):
        # Get the index of the item from the index dictionary with a default value
        return self.index.get(item, default)

    def item(self, index):
        # Get the item corresponding to the given index
        return self.items[index]

    def keys(self):
        # Return the item list to preserve the original order
        return self.item_list
