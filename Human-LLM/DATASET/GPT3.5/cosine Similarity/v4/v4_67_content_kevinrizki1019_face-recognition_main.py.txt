from tkinter import Tk, Frame, TOP, BOTH

from pageone import PageOne

class MainApp(Tk):
    def __init__(self, *args, **kwargs):
        # Call the superclass constructor
        Tk.__init__(self, *args, **kwargs)
        
        # Set the window title
        self.title("KetokMagicHalal")
        
        # Create a container to hold frames
        container = Frame(self)
        container.pack(side="top", fill="both", expand=True)

        # Dictionary to store frames
        self.frames = {}

        # Create and store PageOne frame
        page_one_frame = PageOne(container, self)
        self.frames[PageOne] = page_one_frame
        page_one_frame.pack_forget()

        # Show the PageOne frame
        self.show_frame(PageOne)

        self.frames[PageOne].load_db(db_path)

    def show_frame(self, cont):

        if self.currentFrame is not None:
            self.currentFrame.pack_forget()

        self.currentFrame = self.frames[cont]
        self.currentFrame.pack(side=TOP, fill=BOTH, expand=True, padx=20, pady=20)
