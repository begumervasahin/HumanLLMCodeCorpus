from surprise import similarities as sims
from surprise import Dataset, Reader
import heapq

# Load dataset from a file
def load_dataset(dataset_path, line_format='user item rating', rating_scale=(1, 7), sep=None):
    name = None
    file_format = Reader(name, line_format, sep, rating_scale)
    dataset = Dataset.load_from_file(dataset_path, file_format)
    return dataset

# Create a full training set from the dataset
def create_full_trainset(dataset_path, line_format=None, rating_scale=(1, 7), sep=None):
    data = load_dataset(dataset_path, line_format, rating_scale, sep)
    trainset = data.build_full_trainset()
    return trainset

# Compute mean-centered ratings for each user or item
def mean_centered_ratings(rating_matrix):
    mean_centered = {}
    for x, x_ratings in rating_matrix.items():
        ratings = [rating for _, rating in x_ratings]
        mean_rating = sum(ratings) / len(x_ratings)
        centered_ratings = [(item_id, rating - mean_rating) for item_id, rating in x_ratings]
        mean_centered[x] = centered_ratings
    return mean_centered

# Compute adjusted cosine similarity between all pairs of users or items
def adjustedCosine(num_rows, rating_matrix, min_support):
    mean_centered = mean_centered_ratings(rating_matrix)
    similarity_matrix = {}
    for y, y_ratings in mean_centered.items():
        sim_row = {}
        for xi, ri in y_ratings:
            for xj, rj in y_ratings:
                if xi != xj:
                    sim_row[xi] = sim_row.get(xi, 0) + ri * rj
        similarity_matrix[y] = sim_row
    return similarity_matrix

# Predict rating based on collaborative filtering
def rating_prediction(sim, target, rTarget, tRatings, neighbor_size):
    target_sim = [sim[target].get(iID, 0) for iID, _ in tRatings[rTarget]]
    n_sims = heapq.nlargest(neighbor_size, target_sim)
    neighbours = {i: sim[target][i] for i in range(neighbor_size) if sim[target].get(i, 0) in n_sims}
    selected_ratings = {x: rating for x, rating in tRatings[rTarget] if x in neighbours}
    if len(selected_ratings) < neighbor_size:
        print('Not enough neighbors')
        return 0
    prediction_ratings = sum(selected_ratings[x] * neighbours[x] for x in neighbours)
    return prediction_ratings / sum(neighbours.values())

# Perform collaborative filtering prediction
def prediction_by_CF(trainset, item_rawId, user_rawId, neighbor_size, user_item, sim_method):
    itemId = item_rawId - 1
    userId = user_rawId - 1
    min_support = 0
    num_users = trainset.n_users
    num_items = trainset.n_items
    user_ratings = trainset.ur
    item_ratings = trainset.ir
    n_target, target, rTarget, ratings, tRatings = num_users, userId, itemId, user_ratings, item_ratings
    if user_item != "user":
        n_target, target, rTarget, ratings, tRatings = num_items, itemId, userId, item_ratings, user_ratings
    args = [n_target, tRatings, min_support]
    if sim_method == "cosine":
        sim = sims.cosine(*args)
    elif sim_method == "pearson":
        sim = sims.pearson(*args)
    else:
        sim = adjustedCosine(*args)
    print('\nSimilarity:\n', sim)
    return rating_prediction(sim, target, rTarget, tRatings, neighbor_size)

# Example usage
file_path = 'ratings_for_question1.txt'
trainset = create_full_trainset(file_path, line_format='user item rating', rating_scale=(1, 7), sep='\t')

item_rawId = 1
user_rawId = 3
neighbor = 2
user_item = 'user'
sim_method = 'pearson'
estimate = prediction_by_CF(trainset, item_rawId, user_rawId, neighbor, user_item, sim_method)
print ('\nUsing {} based CF, the predicted rating to item {} for user {} is: {est:4.2f} \n'.format(user_item, item_rawId, user_rawId, est=estimate))
