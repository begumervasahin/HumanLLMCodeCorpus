from surprise import Dataset, Reader
from numpy import zeros
from six import iteritems

def remove_zeros(element1, element2):
    # Remove elements where either element1 or element2 is zero
    indices_to_remove = [i for i in range(len(element1)) if element1[i] == 0 or element2[i] == 0]
    element1 = delete(element1, indices_to_remove)
    element2 = delete(element2, indices_to_remove)
    print('Sorted to calculate similarity:\n', element1, '\n', element2, '\n')
    return element1, element2

def remove_zeros_all(element1, element2):
    # Remove elements where both element1 and element2 are zero
    indices_to_remove_element1 = [i for i in range(len(element1)) if element1[i] == 0]  
    indices_to_remove_element2 = [i for i in range(len(element2)) if element2[i] == 0]  
    indices_to_remove = set(indices_to_remove_element1 + indices_to_remove_element2)    
    element1 = delete(element1, list(indices_to_remove))
    element2 = delete(element2, list(indices_to_remove))
    print('Sorted to calculate average:\n', element1, '\n', element2, '\n')
    return element1, element2

def calculate_cosine_similarity(element1, element2):
    """
    Calculate cosine similarity between two vectors.
    """
    element1, element2 = remove_zeros(element1, element2)
    assert len(element1) == len(element2) and len(element1) > 0
    n = len(element1)
    sumxx, sumxy, sumyy = 0, 0, 0
    for i in range(n):
        x = element1[i]
        y = element2[i]
        sumxx += x * x
        sumyy += y * y
        sumxy += x * y
    return sumxy / (sumxx**0.5 * sumyy**0.5)

def calculate_pearson_similarity(element1, element2):
    """
    Calculate Pearson correlation coefficient between two vectors.
    """
    element1, element2 = remove_zeros_all(element1, element2)
    assert len(element1) == len(element2) and len(element1) > 0
    n = len(element1)
    avg_x = sum(element1) / n
    avg_y = sum(element2) / n
    sumxy_avg, sumxx_avg, sumyy_avg = 0, 0, 0
    for i in range(n):
        adiff = element1[i] - avg_x
        bdiff = element2[i] - avg_y
        sumxy_avg += adiff * bdiff
        sumxx_avg += adiff * adiff
        sumyy_avg += bdiff * bdiff
    return sumxy_avg / (sumxx_avg**0.5 * sumyy_avg**0.5)

def similarity_for_pair(dataset_path, element1, element2, users, cosine):
    """
    Calculate similarity between two elements in the dataset.
    """
    file_path = dataset_path
    reader = Reader(line_format='user item rating', rating_scale=(1, 7), sep='\t')     
    data = Dataset.load_from_file(file_path, reader)
    trainset = data.build_full_trainset()
    number_users = trainset.n_users
    number_items = trainset.n_items
    user_ratings = trainset.ur
    item_ratings = trainset.ir
    if users:
        ratings = user_ratings
        user_item = "User"
        matrix = zeros((number_users, number_items), float)
    else:
        ratings = item_ratings
        user_item = "Item"
        matrix = zeros((number_items, number_users), float)
    for y, y_ratings in iteritems(ratings):
        for xi, ri in y_ratings:
            if ri != 0:
                matrix[y, xi] += ri
    print('Original table:\n', matrix, '\n')
    print('Picked:\n', matrix[element1], '\n', matrix[element2], '\n')
    if cosine:
        similarity = calculate_cosine_similarity(matrix[element1], matrix[element2])
        similarity_method = "Cosine"
    else:
        similarity = calculate_pearson_similarity(matrix[element1], matrix[element2])
        similarity_method = "Pearson"
    return '{} similarity between {} {} and {} is: {}'.format(similarity_method, user_item, element1, element2, similarity)
