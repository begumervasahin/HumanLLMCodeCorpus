from __future__ import absolute_import, division, print_function, unicode_literals
from surprise import Dataset, Reader
import numpy as np
from six import iteritems

def mean_centered_ratings(dataset_path):
    # Load dataset from file
    file_path = dataset_path
    reader = Reader(line_format='user item rating', rating_scale=(1, 7), sep='\t')
    data = Dataset.load_from_file(file_path, reader)
    
    # Build full training set
    trainset = data.build_full_trainset()
    
    # Extract necessary information
    num_users = trainset.n_users
    num_items = trainset.n_items
    user_ratings = trainset.ur
    
    # Initialize matrices to store user-item ratings and cumulative ratings
    user_item_ratings = np.zeros((num_users, num_items), dtype=float)
    cumulative_ratings = np.zeros((num_users, 3), dtype=float)  # [total_rating, count_non_zero, average]

    # Calculate cumulative ratings per user
    for user_id, ratings in iteritems(user_ratings):
        for item_id, rating in ratings:
            if rating != 0:
                user_item_ratings[user_id, item_id] += rating
                cumulative_ratings[user_id, 0] += rating
                cumulative_ratings[user_id, 1] += 1

    # Calculate average ratings per user and update cumulative ratings
    for user_id, _ in iteritems(user_ratings):
        if cumulative_ratings[user_id, 1] != 0:
            cumulative_ratings[user_id, 2] = cumulative_ratings[user_id, 0] / cumulative_ratings[user_id, 1]

    # Calculate mean-centered ratings
    mean_centered_ratings = user_item_ratings - np.tile(cumulative_ratings[:, 2], (num_items, 1)).T

    return mean_centered_ratings

# Define the path to the dataset file
file_path = 'data/table2.1_ratings.txt'

# Calculate and print the mean-centered ratings
print('The mean centered ratings\n', mean_centered_ratings(file_path))
