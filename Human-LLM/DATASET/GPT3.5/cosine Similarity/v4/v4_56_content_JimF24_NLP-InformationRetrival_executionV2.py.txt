import re
import math
import nltk
from nltk.stem import PorterStemmer
from nltk.corpus import stopwords

# Initialize NLTK components
nltk.download('stopwords')
nltk.download('punkt')

# Initialize Porter Stemmer
ps = PorterStemmer()

# Load stopwords
stop_words = set(stopwords.words('english'))

# File paths
query_collection_name = "cran.qry"
abstract_name = "cran.all.1400"

# Process Query File
IDF_query_collection = {}
vector_list_queries = []

with open(query_collection_name, 'r') as query_file:
    queries = query_file.read().replace("\n", " ")
    queries = queries.split(".I")
    queries = [phase for phase in queries if phase != ""]

    for query in queries:
        query_tokens = nltk.word_tokenize(query)
        query_tokens = [word for word in query_tokens if word.lower() not in stop_words and word not in string.punctuation]
        non_repeating_query = list(set(query_tokens))
        
        for word in non_repeating_query:
            IDF_query_collection[word] = IDF_query_collection.get(word, 0) + 1

        # Create TF-IDF vector for query
        query_vector = {}
        for word in query_tokens:
            query_vector[word] = math.log(query_tokens.count(word) + 1) * IDF_query_collection[word]
        vector_list_queries.append(query_vector)

# Process Abstract File
IDF_abstract_collection = {}
vector_list_documents = []

with open(abstract_name, 'r') as abstract_file:
    abstract = abstract_file.read().replace("\n", " ").split(".I")
    abstract = [document for document in abstract if document != ""]

    for document in abstract:
        document_tokens = nltk.word_tokenize(document)
        result = 0
        for i, word in enumerate(document_tokens):
            if word == ".W":
                result = i
                break
        del(document_tokens[0:result+1])

        document_tokens = [word for word in document_tokens if word.lower() not in stop_words and word not in string.punctuation]
        non_repeating_doc = list(set(document_tokens))
        
        for word in non_repeating_doc:
            IDF_abstract_collection[word] = IDF_abstract_collection.get(word, 0) + 1

        # Create TF-IDF vector for document
        doc_vector = {}
        for word in document_tokens:
            doc_vector[word] = math.log(document_tokens.count(word) + 1) * IDF_abstract_collection[word]
        vector_list_documents.append(doc_vector)

# Calculate Cosine Similarity and Rank Documents
output_file = open("output.txt", 'w')

for i, query_vector in enumerate(vector_list_queries):
    ranking = []

    for j, doc_vector in enumerate(vector_list_documents):
        for word in query_vector:
            if word not in doc_vector:
                doc_vector[word] = 0

        # Compute Cosine Similarity
        cosine_num = sum(query_vector[word] * doc_vector[word] for word in query_vector)
        queries_square = sum(query_vector[word] ** 2 for word in query_vector)
        document_square = sum(doc_vector[word] ** 2 for word in doc_vector)
        cosine_den = math.sqrt(queries_square * document_square)
        
        if cosine_den == 0:
            cosine_ranking = 0
        else:
            cosine_ranking = cosine_num / cosine_den
        ranking.append((cosine_ranking, j))
    
    ranking.sort(key=lambda tup: tup[0], reverse=True)
    
    for rank, (cosine_similarity, document_index) in enumerate(ranking):
        output_file.write(f"{i + 1} {document_index + 1} {cosine_similarity}\n")

output_file.close()
