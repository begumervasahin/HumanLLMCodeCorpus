import pandas as pd
import numpy as np
from sklearn.feature_extraction.text import CountVectorizer
from scipy import spatial
import math
import operator

# Load movie ratings data from CSV file
ratings = pd.read_csv("D:/roops_backup/projects/data mining project/movie_recommender/movie_dataset.csv")

# Select relevant columns from the ratings dataframe
ratings_subset = ratings[['id', 'title', 'vote_average', 'popularity', 'vote_count', 'genres']]

# Normalize popularity and vote count columns
normalized_popularity = (ratings_subset['popularity'] - ratings_subset['popularity'].min()) / (ratings_subset['popularity'].max() - ratings_subset['popularity'].min())
normalized_vote_count = (ratings_subset['vote_count'] - ratings_subset['vote_count'].min()) / (ratings_subset['vote_count'].max() - ratings_subset['vote_count'].min())

# Extract genres and convert them into count vectors
genres = ratings_subset['genres'].fillna('')
vectorizer = CountVectorizer()
genre_counts = vectorizer.fit_transform(genres)

# Create a dictionary to store movie information
moviedict = {}
for i in range(len(ratings_subset)):
    movie_id = ratings_subset.loc[i, 'id']
    title = ratings_subset.loc[i, 'title']
    vote_average = ratings_subset.loc[i, 'vote_average']
    popularity = normalized_popularity.loc[i]
    vote_count = normalized_vote_count.loc[i]
    genre_vector = genre_counts.toarray()[i]
    moviedict[i] = (movie_id, title, vote_average, popularity, vote_count, genre_vector)

# Function to calculate distance between two movies
def calculate_distance(movie1, movie2):
    popularity1, genre_vector1 = movie1[3], movie1[5]
    popularity2, genre_vector2 = movie2[3], movie2[5]

    genre_distance = spatial.distance.cosine(genre_vector1, genre_vector2)
    popularity_distance = (popularity2 - popularity1) ** 2
    total_distance = math.sqrt(popularity_distance + genre_distance ** 2)
    return total_distance

# Function to find K nearest neighbors for a given movie
def find_neighbors(movie_id, K):
    distances = []
    for neighbor_id, neighbor_movie in moviedict.items():
        if neighbor_id != movie_id:
            distance = calculate_distance(moviedict[movie_id], neighbor_movie)
            if distance <= 0.65:
                distances.append((neighbor_id, distance))

    distances.sort(key=operator.itemgetter(1))

    neighbors = []
    for i in range(K):
        neighbor_id = distances[i][0]
        neighbor_title = moviedict[neighbor_id][1]
        neighbor_distance = distances[i][1]
        neighbors.append((neighbor_title, neighbor_distance))
        print(neighbor_title)

    return neighbors

# Parameters
K = 15
movie_id = 94

# Find K nearest neighbors for the given movie
find_neighbors(movie_id, K)
