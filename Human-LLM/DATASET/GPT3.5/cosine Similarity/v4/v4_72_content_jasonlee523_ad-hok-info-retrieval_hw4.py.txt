import math
import re
import operator
from stop_list import closed_class_stop_words as stop_list

# Function to preprocess text
def preprocess_text(text):
    words = re.findall(r'\w+', text.lower())
    return [word for word in words if word not in stop_list]

# Function to calculate TF-IDF
def calculate_tfidf(documents, num_docs):
    tfidf_vectors = []
    term_freqs = []
    inverse_doc_freqs = {}
    words_in_docs = {}

    # Calculate term frequency (TF) and document frequency (DF)
    for doc in documents:
        term_freq = {}
        for word in doc:
            term_freq[word] = term_freq.get(word, 0) + 1
            words_in_docs.setdefault(word, set()).add(len(tfidf_vectors))
        term_freqs.append(term_freq)

    # Calculate inverse document frequency (IDF)
    for term, docs in words_in_docs.items():
        inverse_doc_freqs[term] = math.log(num_docs / len(docs))

    # Calculate TF-IDF
    for doc_index, term_freq in enumerate(term_freqs):
        tfidf_vector = {}
        for term, freq in term_freq.items():
            tfidf_vector[term] = (freq / len(term_freq)) * inverse_doc_freqs[term]
        tfidf_vectors.append(tfidf_vector)

    return tfidf_vectors, inverse_doc_freqs

# Load queries
with open('cran.qry') as query_file:
    query_text = query_file.read()
    queries = [preprocess_text(qry) for qry in query_text.split('.I')[1:]]

# Load abstracts
with open('cran.all.1400') as abstract_file:
    abstract_text = abstract_file.read()
    abstracts = [preprocess_text(abst) for abst in re.findall(r'\.I[\s\S]*?\.W', abstract_text)]

# Calculate TF-IDF for queries and abstracts
query_tfidf, query_idf = calculate_tfidf(queries, len(queries))
abstract_tfidf, abstract_idf = calculate_tfidf(abstracts, len(abstracts))

# Calculate scores
scores = []
for query_vector in query_tfidf:
    query_scores = {}
    for i, abstract_vector in enumerate(abstract_tfidf):
        dot_product = sum(query_vector.get(term, 0) * abstract_vector.get(term, 0) for term in query_vector)
        query_norm = sum(freq ** 2 for freq in query_vector.values()) ** 0.5
        abstract_norm = sum(freq ** 2 for freq in abstract_vector.values()) ** 0.5
        if dot_product > 0:
            query_scores[i] = dot_product / (query_norm * abstract_norm)
        else:
            query_scores[i] = 0
    sorted_query_scores = sorted(query_scores.items(), key=operator.itemgetter(1), reverse=True)
    scores.append(sorted_query_scores)

# Write output to file
with open('output.txt', 'w') as output_file:
    for i, query in enumerate(scores):
        for doc, score in query:
            output_file.write(f"{i} {doc} {score}\n")
