from scipy import spatial
import math
import string

def read_positive_integer(prompt):
    while True:
        try:
            value = int(input(prompt))
            if value <= 0:
                raise ValueError
            return value
        except ValueError:
            print("Please enter a valid positive integer.")

def read_document_name(prompt):
    while True:
        try:
            document_name = input(prompt)
            open(document_name, "r")
            return document_name
        except IOError:
            print("Error: The specified file was not found. Please try again.")

def main():
    # Input: number of documents
    num_documents = read_positive_integer("How many documents do you want to compare? ")

    # Read document names and contents
    documents = []
    document_words = {}
    for i in range(num_documents):
        while True:
            try:
                document_name = read_document_name("Please insert the document name: ")
                with open(document_name, "r") as file:
                    document_content = file.read()
                    document_words[i] = document_content.lower().translate(str.maketrans('', '', string.punctuation)).split()
                documents.append(document_name)
                break
            except IOError:
                print("Error: No such file was found. Try again.")

    # Extract unique words from all documents
    unique_words = list(set(word for words in document_words.values() for word in words))

    # Calculate document vectors
    document_vectors = []
    for words in document_words.values():
        vector = [words.count(word) for word in unique_words]
        document_vectors.append(vector)

    # Calculate pairwise similarity matrix
    similarity_matrix = [[0] * num_documents for _ in range(num_documents)]
    for i in range(num_documents):
        for j in range(num_documents):
            if i < j:
                similarity = 1 - spatial.distance.cosine(document_vectors[i], document_vectors[j])
                similarity_matrix[i][j] = round(similarity, 4)

    # Input: number of pairs to print
    num_pairs = read_positive_integer("How many pairs do you want to be printed? ")
    max_pairs = math.factorial(num_documents) // (2 * math.factorial(num_documents - 2))
    if num_pairs > max_pairs:
        print("Sorry, there are not enough documents to generate that many pairs. Please choose a smaller number.")
        return

    # Print pairs with highest similarity
    for _ in range(num_pairs):
        max_similarity = 0
        max_indices = (0, 0)
        for i in range(num_documents):
            for j in range(num_documents):
                if i < j and similarity_matrix[i][j] > max_similarity:
                    max_similarity = similarity_matrix[i][j]
                    max_indices = (i, j)
        doc1, doc2 = max_indices
        print(f"Documents {documents[doc1]} and {documents[doc2]} with similarity: {max_similarity}")
        similarity_matrix[doc1][doc2] = 0

if __name__ == "__main__":
    main()
