import re
from collections import Counter
import math

# Define the original and new documents
original_document = 'AI is our friend and it has been friendly'
new_document = 'AI and humans have always been friendly'

# Function to convert a document into a word frequency vector
def document_to_vector(document):
    word_list = re.compile(r'\w+')
    all_words = word_list.findall(document)
    return Counter(all_words)

# Convert the original and new documents into word frequency vectors
vector_1 = document_to_vector(original_document.lower())
vector_2 = document_to_vector(new_document.lower())

# Function to calculate cosine similarity between two word frequency vectors
def cosine_similarity(vector1, vector2):
    common_words = set(vector1.keys()) & set(vector2.keys())
    sum_of_products = sum([vector1[word] * vector2[word] for word in common_words])
    sum_of_sq1 = sum([vector1[word] ** 2 for word in vector1.keys()])
    sum_of_sq2 = sum([vector2[word] ** 2 for word in vector2.keys()])
    multiplier_of_sum = math.sqrt(sum_of_sq1) * math.sqrt(sum_of_sq2)
    return sum_of_products / multiplier_of_sum

# Calculate the plagiarism factor
plagiarism_factor = cosine_similarity(vector_1, vector_2)

# Print the plagiarism factor
print('Plagiarism factor:', plagiarism_factor)

# Check if the plagiarism factor exceeds the threshold
if plagiarism_factor > 0.40:
    print('Your document is plagiarized.')
else:
    print('Your document is not plagiarized.')
