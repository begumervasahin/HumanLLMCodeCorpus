import re
import string
import nltk
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from nltk.stem.porter import PorterStemmer
from sklearn.feature_extraction.text import CountVectorizer
from sklearn.metrics.pairwise import cosine_similarity

# Sample data
cnn_scroll = ["Sample CNN article text 1.", "Sample CNN article text 2.", "Sample CNN article text 3."]
fox_scroll = ["Sample Fox article text 1.", "Sample Fox article text 2.", "Sample Fox article text 3."]

# Preprocessing CNN scroll
cnn_scroll_text = ' '.join(cnn_scroll)
cnn_scroll_text = re.sub(r'\W+', ' ', cnn_scroll_text)
cnn_scroll_words = word_tokenize(cnn_scroll_text)
cnn_scroll_words = [word.lower() for word in cnn_scroll_words if word.isalpha()]
stop_words = set(stopwords.words('english'))
cnn_scroll_words = [word for word in cnn_scroll_words if word not in stop_words]
porter = PorterStemmer()
cnn_scroll_words = [porter.stem(word) for word in cnn_scroll_words]

# Preprocessing Fox scroll
fox_scroll_text = ' '.join(fox_scroll)
fox_scroll_text = re.sub(r'\W+', ' ', fox_scroll_text)
fox_scroll_words = word_tokenize(fox_scroll_text)
fox_scroll_words = [word.lower() for word in fox_scroll_words if word.isalpha()]
fox_scroll_words = [word for word in fox_scroll_words if word not in stop_words]
fox_scroll_words = [porter.stem(word) for word in fox_scroll_words]

# Vectorization
vectorizer = CountVectorizer(analyzer='word', min_df=1, stop_words='english', lowercase=True, token_pattern='[a-zA-Z0-9]{3,}')
cnn_vectorized = vectorizer.fit_transform(cnn_scroll_words)
fox_vectorized = vectorizer.fit_transform(fox_scroll_words)

# Calculate cosine similarity
cosine_sim_score = cosine_similarity(cnn_vectorized, fox_vectorized)
mean_cosine_sim_score = cosine_sim_score.mean()

print("Mean Cosine Similarity Score:", mean_cosine_sim_score)
