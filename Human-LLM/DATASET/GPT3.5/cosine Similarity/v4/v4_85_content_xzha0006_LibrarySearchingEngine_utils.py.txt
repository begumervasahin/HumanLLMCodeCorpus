from stemming import PorterStemmer
import numpy as np
import re
import math
import os
from os import listdir
from os.path import isfile, join

def tokenization(file_content, stopwords=[]):
    stemmer = PorterStemmer()
    term_list = []

    # Preprocess file content
    file_content = preprocess_text(file_content)

    # Extract emails and URLs
    email_list = re.findall(r"(\w+([-+.]\w+)*@\w+([-.]\w+)*\.\w+([-.]\w+)*)", file_content, flags=0)
    term_list.extend([i[0] for i in email_list])
    file_content = re.sub(r"(\w+([-+.]\w+)*@\w+([-.]\w+)*\.\w+([-.]\w+)*)", r"", file_content)
    url_list = re.findall(r"(((ht|f)tp(s?)\:\/\/)?[a-zA-Z0-9][-a-zA-Z0-9]{0,62}(\.[a-zA-Z0-9][-a-zA-Z0-9]{0,62})+)", file_content, flags=0)
    term_list.extend([i[0] for i in url_list])
    file_content = re.sub(r"(((ht|f)tp(s?)\:\/\/)?[a-zA-Z0-9][-a-zA-Z0-9]{0,62}(\.[a-zA-Z0-9][-a-zA-Z0-9]{0,62})+)", r"", file_content)

    # Extract other terms
    rest_terms = extract_terms(file_content)

    # Stemming and adding to term_list
    for term in rest_terms:
        if term not in stopwords:
            term = stemmer.stem(term, 0, len(term) - 1)
            term_list.append(term)

    return term_list

def preprocess_text(text):
    text = re.sub(r"\n\ {1,}", r"\n", text)
    text = re.sub(r"\n{2,}", r"\n", text)
    text = text.replace("\xa0", " ")
    text = text.replace("-\n", "").replace("\n", " ")
    text = text.lower()
    text = re.sub(r"[\-]{2,}", r"", text)
    return text

def extract_terms(text):
    rest_terms = []
    single_comma_content = re.findall(r"( '\w+( \w+){1,6}')", text)
    rest_terms.extend([i[0][2:-1] for i in single_comma_content])
    text = re.sub(r"( '\w+( \w+){1,6}')", r" ", text)
    capital_lower_terms = re.findall(r"([A-Z][a-z]+\ ([A-Z][a-z]+[\ \.]{1})+)", text)
    rest_terms.extend([i[0][:-1] for i in capital_lower_terms])
    text = re.sub(r"([A-Z][a-z]+\ ([A-Z][a-z]+[\ \.]{1})+)", r" ", text)
    full_capital_terms = re.findall(r"([A-Z]+\ ([A-Z]+[\ \.]{1})+)", text)      
    rest_terms.extend([i[0][:-1] for i in full_capital_terms])
    text = re.sub(r"([A-Z]+\ ([A-Z]+[\ \.]{1})+)", r" ", text)
    rest_terms.extend(re.findall(r"[^\[\]{,:;\"()?!}\ |*]", text))
    return rest_terms

def cal_tf(term_list):
    tf_dict = {}
    for term in term_list:
        if term in tf_dict:
            tf_dict[term] += 1
        else:
            tf_dict[term] = 1

    # Calculate tf using log normalization
    for key in tf_dict.keys():
        tf_dict[key] = 1 + math.log(tf_dict[key])

    return tf_dict

def load_files(file_path):
    file_contentList = []
    file_names = [os.path.join(file_path, f) for f in listdir(file_path) if isfile(join(file_path, f)) and f != '.DS_Store']
    for filename in file_names:
        with open(filename, 'r') as f:
            data = f.read()
            file_contentList.append(data)
    return file_contentList, [i.split('/')[-1].replace(",", " ") for i in file_names]
