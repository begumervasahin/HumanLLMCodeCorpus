# Importing necessary libraries
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import confusion_matrix, accuracy_score, f1_score

# Load the training and test datasets
train_data = pd.read_csv('dota2Train.csv')
test_data = pd.read_csv('dota2Test.csv')

# Extracting features and labels from the datasets
y_train = train_data.iloc[:, 0].values
X_train = train_data.iloc[:, 4:].values
y_test = test_data.iloc[:, 0].values
X_test = test_data.iloc[:, 4:].values

# Initializing and training the Random Forest Classifier
classifier = RandomForestClassifier(max_features='log2', n_estimators=100, criterion='entropy', random_state=0)
classifier.fit(X_train, y_train)

# Predicting labels for the test dataset
y_pred = classifier.predict(X_test)

# Calculating confusion matrix
conf_matrix = confusion_matrix(y_test, y_pred)
print("Confusion Matrix:")
print(conf_matrix)

# Calculating accuracy
accuracy = accuracy_score(y_test, y_pred)
print("Accuracy:", accuracy)

# Calculating F1 score
f1 = f1_score(y_test, y_pred, average='macro')
print("F1 Score:", f1)
