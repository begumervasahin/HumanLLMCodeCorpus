from DatumBox import DatumBox
import urllib.request
import re
import nltk
from sklearn.feature_extraction.text import TfidfVectorizer
from nltk.stem import PorterStemmer
from nltk.corpus import stopwords

# Initialize DatumBox API
datum_box = DatumBox("2a13913dda346761765020c1f66e34f8")

# Compile regex pattern to remove HTML tags
TAG_RE = re.compile(r"<[^>]+>")

# Download NLTK data
nltk.download('punkt')

# Initialize Stemmer and create punctuation removal mapping
stemmer = PorterStemmer()
remove_punctuation_map = dict((ord(char), None) for char in "!\"#$%&'()*+,-./:;<=>?@[\]^_`{|}~")

def fetch_page(url):
    """Fetch web page content given a URL."""
    headers = {
        'User-Agent': 'Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.11 (KHTML, like Gecko) Chrome/23.0.1271.64 Safari/537.11',
        'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8',
        'Accept-Charset': 'ISO-8859-1,utf-8;q=0.7,*;q=0.3',
        'Accept-Encoding': 'none',
        'Accept-Language': 'en-US,en;q=0.8',
        'Connection': 'keep-alive'
    }
    req = urllib.request.Request(url, headers=headers)
    try:
        with urllib.request.urlopen(req) as response:
            content = response.read().decode('utf-8')
            return content
    except urllib.error.HTTPError as e:
        print(e)
        return ''

def remove_tags(text):
    """Remove HTML tags from text."""
    return TAG_RE.sub('', text)

def stem_tokens(tokens):
    """Stem tokens using Porter Stemmer."""
    return [stemmer.stem(token) for token in tokens]

def normalize(text):
    """Normalize text by removing punctuation, lowercasing, and stemming."""
    tokens = nltk.word_tokenize(text.lower().translate(remove_punctuation_map))
    return stem_tokens(tokens)

# Initialize TF-IDF Vectorizer with custom tokenizer and stop words
vectorizer = TfidfVectorizer(tokenizer=normalize, stop_words=stopwords.words('english'))

def cosine_sim(text1, text2):
    """Calculate cosine similarity between two texts."""
    tfidf = vectorizer.fit_transform([text1, text2])
    return ((tfidf * tfidf.T).A)[0, 1]

# Fetch web page contents
print("Crawling the first page...")
page1_content = fetch_page("http://www.geeksforgeeks.org/fundamentals-of-algorithms/")
print("Crawling the second page...")
page2_content = fetch_page("http://www.geeksforgeeks.org/data-structures/")

# Extract text from pages using DatumBox API
if page1_content and page2_content:
    print("Extracting data from the first page...")
    extracted_text1 = datum_box.text_extract(page1_content)
    print("Extracting data from the second page...")
    extracted_text2 = datum_box.text_extract(page2_content)

    # Calculate and print cosine similarity between extracted texts
    similarity_score = cosine_sim(extracted_text1, extracted_text2)
    print(f"Cosine Similarity between the two pages: {similarity_score}")
else:
    print("Failed to fetch page content.")
