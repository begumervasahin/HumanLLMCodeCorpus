# Import all symbols from the 'app' module and the 'constants' module
from app import *
from constants import *

# Define a function to create a response object
def return_response_object(text, quick_replies=None, buttons=None):
    response_object = {
        'text': text,
        'quick_replies': quick_replies,
        'buttons': buttons
    }
    return response_object

# Define a test class for bot responses
class TestBotResponses:
    # Define a test method for the intro message response
    def test_intro_message_response(self):
        # Call the 'bot_response' function with the input "Hello"
        reply = bot_response("Hello")

        # Create the expected response object using the provided text and buttons from raw_response_data
        response_obj = return_response_object(
            "Hello! I’m Sloan, your automated guide for advice on your student loans brought to you by TISLA. This is not legal advice but simple guidance to help you manage your student loan debt. Send 'restart' at any time to restart. Ready? Simply type your question and I’ll try and guide you to the best way to manage your student debt!",
            raw_response_data['replies']['intro']
        )

        # Assert that the returned response matches the expected response object
        assert reply.get('text') == response_obj.get('text')
        assert reply.get('buttons') == response_obj.get('buttons')
        assert reply.get('quick_replies') == response_obj.get('quick_replies')
