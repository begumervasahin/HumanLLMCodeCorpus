import distsim

# Load word contexts from the provided file
word_to_ccdict = distsim.load_contexts("nytcounts.4k")

def print_nearest_words(target_word, target_set):
    """Prints the nearest words for a target word based on similarity."""
    print(f"Target word: {target_word}")
    for i, (nearest_word, score) in enumerate(distsim.show_nearest(word_to_ccdict, word_to_ccdict[target_word], target_set, distsim.cossim_sparse), start=1):
        print(f"{i}: {nearest_word} ({score})")
    print("--------------------")

# Print nearest words for each target word
print_nearest_words('america', {'america'})
print_nearest_words('years', {'years'})
print_nearest_words('great', {'great'})
print_nearest_words('run', {'run'})
print_nearest_words('between', {'between'})
print_nearest_words('wife', {'wife'})

# Further exploration: comparing 'big' and 'tiny' with 'great'
print("Further exploring: big from great")
print_nearest_words('big', {'big'})
print("--------------------")
print("Further exploring: tiny from great")
print_nearest_words('tiny', {'tiny'})
print("--------------------")
