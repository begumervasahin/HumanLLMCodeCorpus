import sys
import os
import pickle
from tqdm import tqdm
from PIL import Image
from img2vec import Img2Vec

def convert_images_to_vectors(model, cuda):
    img2vec = Img2Vec(cuda=cuda, model=model)
    vectors = {}

    print("Converting images to vectors:")
    for image_filename in tqdm(os.listdir("resized")):
        image_path = os.path.join("resized", image_filename)
        try:
            with Image.open(image_path) as img:
                vec = img2vec.get_vec(img)
        except RuntimeError:
            # Retry with RGB conversion if RuntimeError occurs
            with Image.open(image_path).convert('RGB') as img:
                vec = img2vec.get_vec(img)

        assert len(vec) == img2vec.layer_output_size
        vectors[image_filename] = vec

    assert len(vectors) == len(os.listdir("resized"))

    print("Saving image vectors...")
    os.makedirs(os.path.join("pickles", model), exist_ok=True)
    with open(os.path.join('pickles', model, 'vectors.pkl'), 'wb') as file:
        pickle.dump(vectors, file)

    print("Image vectors saved successfully.")
    return vectors

if __name__ == "__main__":
    model = "vgg16"  # Specify the model
    cuda = True  # Set to True if CUDA is available, False otherwise

    # Convert images to vectors
    convert_images_to_vectors(model, cuda)
