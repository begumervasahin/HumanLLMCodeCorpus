import nltk
import os

def remove_non_ascii(content):
    """Remove non-ASCII characters from the given content."""
    return ''.join(c for c in content if 32 <= ord(c) <= 126)

def collect_all_sentences(text_lines, sentence_splitter):
    """Collect all sentences from the text lines using NLTK's sentence splitter."""
    print("Number of lines:", len(text_lines))
    sentences_tok = []
    line_count = 0

    for content in text_lines:
        line_count += 1
        if line_count % 1000 == 0:
            print("Processed lines:", float(line_count) / len(text_lines))

        content_printable = remove_non_ascii(content)
        content_printable = content_printable.replace("Mr .", "Mr")  # Fixing Mr. abbreviation
        sentences_raw = sentence_splitter.sentences_from_text(content_printable)

        for sent in sentences_raw:
            sentences_tok.append(sent)

    return sentences_tok

# Path to the input data file
PATH_TO_DATA = "/Users/ken77921/Desktop/TA/2004,7-05_nyt_tok"

# Read text lines from the input file
with open(PATH_TO_DATA, 'r') as doc:
    text_lines = doc.readlines()

# Load NLTK's English sentence tokenizer
sentence_splitter = nltk.data.load('tokenizers/punkt/english.pickle')

# Collect all sentences
sentences_tok = collect_all_sentences(text_lines, sentence_splitter)

# Path to the output file
OUTPUT_FILE_PATH = "/Users/ken77921/Desktop/TA/2004,7-05_nyt_sent"

# Write collected sentences to the output file
with open(OUTPUT_FILE_PATH, "w") as f_out:
    for sent in sentences_tok:
        f_out.write(sent + "\n")

print("Sentences extracted and saved successfully!")
