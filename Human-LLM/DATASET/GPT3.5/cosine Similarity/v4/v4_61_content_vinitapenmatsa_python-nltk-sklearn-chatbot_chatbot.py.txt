import nltk
import random
import string
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

# Read the content of the file and preprocess it
with open('test.txt', 'r') as file:
    raw_text = file.read().lower()

# Download NLTK resources
nltk.download('punkt')
nltk.download('wordnet')

# Tokenize the text into sentences and words
sent_tokens = nltk.sent_tokenize(raw_text)
word_tokens = nltk.word_tokenize(raw_text)

# Initialize the WordNet Lemmatizer
lemmatizer = nltk.stem.WordNetLemmatizer()

def lemmatize_tokens(tokens):
    """Lemmatize tokens"""
    return [lemmatizer.lemmatize(token) for token in tokens]

def normalize_text(text):
    """Normalize text by lemmatizing tokens and removing punctuation"""
    for p in string.punctuation:
        text = text.replace(p, '')
    return lemmatize_tokens(nltk.word_tokenize(text))

# Define greetings and responses
GREETING_INPUTS = ("hello", "hi", "greetings", "sup", "what's up",
                   "hey", "hola", "whatsup", "good morning", "morning")
GREETING_RESPONSES = ["hi", "hey", "*nods*", "hi there", "hello",
                      "I am glad! You are talking to me",
                      "Hello, what do you want to ask me?"]

# Define conversation stoppers and responses
CONVERSATION_STOPPERS = ("ok", "fine", "great", "nice", "ah", "alright", "yes", "no")   
CONVERSATION_STOPPER_RESPONSES = ["anything else?", "would that be it?", "have more questions?", "Ok, go on.."]

def greet_user(sentence):
    """Generate a greeting response if the user's input is a greeting"""
    for word in sentence.split():
        if word.lower() in GREETING_INPUTS:
            return random.choice(GREETING_RESPONSES)

def generate_response(user_response):
    """Generate a response to the user's input"""
    robo_response = ''

    # Vectorize the sentences and calculate TF-IDF
    tfidf_vectorizer = TfidfVectorizer(tokenizer=normalize_text, stop_words='english')  
    tfidf_matrix = tfidf_vectorizer.fit_transform(sent_tokens)

    # Calculate cosine similarity between the user's input and existing sentences       
    similarity_scores = cosine_similarity(tfidf_matrix[-1], tfidf_matrix)
    most_similar_index = similarity_scores.argsort()[0][-2]
    sorted_scores = similarity_scores.flatten()
    sorted_scores.sort()
    max_similarity = sorted_scores[-2]

    if max_similarity == 0:
        robo_response = "I am sorry! I don't understand you"
    else:
        robo_response = sent_tokens[most_similar_index]

    return robo_response

# Start the conversation
print("VINITA: Hi, I'm Vinita Penmetsa. I'm glad you are interested in my profile. What do you want to know? If you want to exit, type Bye!")

conversation_active = True
while conversation_active:
    user_input = input().lower()

    if user_input != 'bye':
        if user_input == 'thanks' or user_input == 'thank you':
            conversation_active = False
            print("VINITA: You are welcome..")
        else:
            greeting_response = greet_user(user_input)
            if greeting_response is not None:
                print("VINITA: " + greeting_response)
            else:
                if user_input in CONVERSATION_STOPPERS:
                    print("VINITA: " + random.choice(CONVERSATION_STOPPER_RESPONSES))   
                else:
                    sent_tokens.append(user_input)
                    word_tokens.extend(nltk.word_tokenize(user_input))
                    final_words = list(set(word_tokens))
                    print("VINITA: ")
                    print(generate_response(user_input))
                    sent_tokens.remove(user_input)
    else:
        conversation_active = False
        print("VINITA: Bye! take care..")
