import csv
import math
import heapq

# Global variables
term_freq = {}
norm_termfreq = {}
observe = []
inv_docfreq = {}
tfidffreq = {}
cosin_sim = {}
modulus = {}
k = 2

def read_file():
    # Open and read data from the CSV file
    with open('sample.csv') as csvfile:
        csv_reader = csv.reader(csvfile)
        for row in csv_reader:
            calculate_term_frequency(row[1], row[2])

def calculate_term_frequency(name, text):
    # Calculate term frequency for each word in the document
    if name not in term_freq:
        term_freq[name] = {}
    words = text.split()
    for word in words:
        term_freq[name][word] = term_freq[name].get(word, 0) + 1

def calculate_normalized_frequency():
    # Calculate normalized term frequency
    for name in term_freq:
        norm_termfreq[name] = {word: freq / len(term_freq[name]) for word, freq in term_freq[name].items()}

def calculate_inverse_doc_freq():
    # Calculate inverse document frequency
    for name in term_freq:
        for word in term_freq[name]:
            if word not in observe:
                observe.append(word)
                count = sum(word in term_freq[other_name] for other_name in term_freq if other_name != name) + 1
                inv_docfreq[word] = math.log(len(term_freq) / (1 + count), 2)

def calculate_norm_tfidf_freq():
    # Calculate TF-IDF scores
    for name in term_freq:
        tfidffreq[name] = {word: norm_termfreq[name][word] * inv_docfreq[word] for word in term_freq[name]}

def calculate_mod():
    # Calculate modulus for each document
    for name in tfidffreq:
        modulus[name] = math.sqrt(sum(tfidffreq[name][word] ** 2 for word in tfidffreq[name]))

def calculate_cosine_sim():
    # Calculate cosine similarity between documents
    for name in tfidffreq:
        if name not in cosin_sim:
            cosin_sim[name] = {}
        for other_name in tfidffreq:
            if other_name != name:
                dot_product = sum(tfidffreq[name][word] * tfidffreq[other_name][word] for word in tfidffreq[name] if word in tfidffreq[other_name])
                cosin_sim[name][other_name] = dot_product / (modulus[name] * modulus[other_name])

def k_nearest():
    # Find k-nearest neighbors based on cosine similarity
    for name in cosin_sim:
        print(f"{name} : ")
        top_k = heapq.nlargest(k, cosin_sim[name], key=cosin_sim[name].get)
        for word in top_k:
            print(word)
        print("\n")

# Main program
read_file()
calculate_normalized_frequency()
calculate_inverse_doc_freq()
calculate_norm_tfidf_freq()
calculate_mod()
calculate_cosine_sim()
k_nearest()
