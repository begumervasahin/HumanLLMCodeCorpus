# Import necessary modules and functions
from extensions import connect_to_database
from similarity import (
    shot_percentage_similarity,
    per_game_similarity,
    adv_rate_similarity,
    adv_percent_similarity,
    adv_misc_similarity,
)
from misc import (
    initialize_output_files,
    initialize_unique_dicts,
    remove_output_files,
    determine_output_file,
    write_per40_stats,
    write_NBA_comp_per40,
    write_NBA_player_VORPs,
    write_adv_stats,
    write_NBA_comp_adv,
    intersect_similarities,
    output_unique_seasons,
    reset_unique_players,
)

def main():
    # Connect to the database
    db_connection = connect_to_database()

    # Initialize dictionaries and variables
    players = {}
    adv_players = {}
    unique_per40 = {}
    unique_adv = {}
    unique_players = {}

    # Query database tables to get necessary data
    (
        prosp_szn_results,
        prosp_per40_results,
        prosp_adv_results,
        comps_per40_results,
        comps_adv_results,
    ) = query_tables(db_connection)

    # Initialize unique dictionaries and output files
    unique_per40, unique_adv = initialize_unique_dicts(
        comps_per40_results, comps_adv_results
    )
    remove_output_files()
    divider = (
        "______________________________________________________________________________________________________________________________________\n"
    )
    half_divider = "\n___________________________________________________________________\n\n"
    out_file = initialize_output_files(divider)

    # Process each prospect in per40 results
    for prospect in prosp_per40_results:
        unique_players[prospect["Player"]] = []
        college_player = {}
        output_file = determine_output_file(prospect, prosp_szn_results)
        write_per40_stats(output_file, prospect)
        per40_comparables = {}
        adv_comparables = {}

        # Compare per40 stats with NBA players
        for nba_player in comps_per40_results:
            comp_key = (
                f"{nba_player['Player']}, {nba_player['Season']}, {nba_player['School']}"
            )
            players[comp_key] = nba_player
            per40_comparables[comp_key] = (
                shot_percentage_similarity(prospect, nba_player)
                * per_game_similarity(prospect, nba_player)
            )

        # Compare advanced stats with NBA players
        for nba_player in comps_adv_results:
            college_player = get_advanced_stats(prospect, prosp_adv_results)
            comp_key = (
                f"{nba_player['Player']}, {nba_player['Season']}, {nba_player['School']}"
            )
            adv_players[comp_key] = nba_player
            adv_comparables[comp_key] = (
                adv_rate_similarity(college_player, nba_player)
                * adv_percent_similarity(college_player, nba_player)
                * adv_misc_similarity(college_player, nba_player)
            )

        # Sort comparables and write to output file
        per40_comparables = sorted(
            per40_comparables.items(), key=lambda x: x[1], reverse=True
        )
        adv_comparables = sorted(
            adv_comparables.items(), key=lambda x: x[1], reverse=True
        )
        per40_score = {}
        adv_score = {}

        output_file.write("Per 40 minute comparisons:\n\n")
        vorp_comps = {}
        idx = 0

        for comp in per40_comparables:
            unique_per40[comp[0]].append(prospect["Player"])
            per40_score[comp[0]] = comp[1]
            vorp_comps[comp[0]] = players[comp[0]]["VORP"]
            write_NBA_comp_per40(output_file, comp, players)
            idx += 1
            if idx >= 30:
                break

        output_file.write(
            "\nThe Top 30 per 40 minute NBA comps sorted in descending order based on their career VORPs:\n\n"
        )
        write_NBA_player_VORPs(output_file, vorp_comps)
        write_adv_stats(output_file, college_player, half_divider)
        output_file.write("\nAdvanced stat comparisons:\n")
        vorp_comps = {}
        idx = 0

        for comp in adv_comparables:
            unique_adv[comp[0]].append(prospect["Player"])
            adv_score[comp[0]] = comp[1]
            vorp_comps[comp[0]] = players[comp[0]]["VORP"]
            write_NBA_comp_adv(output_file, comp, adv_players)
            idx += 1
            if idx >= 30:
                break

        output_file.write(
            "\nThe Top 30 advanced stats NBA comps sorted in descending order based on their career VORPs:\n\n"
        )
        write_NBA_player_VORPs(output_file, vorp_comps)
        intersect_similarities(output_file, per40_score, adv_score, divider)
        output_file.close()

    # Write unique seasons to output file
    out_file = open("unique.txt", "a")
    out_file.write("\n\nPER 40 MINUTE UNIQUE SEASONS:\n\n")
    output_unique_seasons(unique_per40, unique_players, players, out_file, True)
    unique_players = reset_unique_players(prosp_per40_results)
    out_file.write("\n\nADVANCED UNIQUE SEASONS:\n\n")
    output_unique_seasons(unique_adv, unique_players, players, out_file, False)
    out_file.close()

# Entry point of the script
if __name__ == "__main__":
    main()
