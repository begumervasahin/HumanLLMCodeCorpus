import nltk
import operator
from nltk.stem.porter import PorterStemmer
from utils import get_bow, get_sentences, stopwords, wn_story_dict, model

def choose_sentence(question, story):
    def get_best_wordnet_sentence(question, story, use_sch=True):
        stemmer = PorterStemmer()
        best_sentence = None
        best_score = 0.0

        # Determine the source of sentences based on 'sch' availability
        if isinstance(story["sch"], str) and use_sch:
            sentences = story["sch"]
        else:
            sentences = story["text"]

        sentences = nltk.sent_tokenize(sentences)
        qwords = get_bow(get_sentences(question['text'])[0], stopwords)

        # Calculate the score for each sentence using WordNet and Word2Vec
        for i, sent in enumerate(wn_story_dict[question['sid']]):
            sentence_score = 0.0
            words_not_found = set(qwords)

            for qword in qwords:
                for word in sent:
                    if str(qword) == word:
                        sentence_score += 1
                        words_not_found.remove(qword)

            for qword in words_not_found:
                highest_sim = 0
                for word in sent:
                    if word in model.vocab and qword in model.vocab:
                        similarity = model.similarity(word, str(qword))
                        if similarity > highest_sim:
                            highest_sim = similarity

                if highest_sim > 0.3:
                    sentence_score += highest_sim

            if sentence_score > best_score:
                best_score = sentence_score
                best_sentence = sentences[i]

        # Recursively call with 'sch' if no sentence found and 'use_sch' is True
        if best_sentence is None and use_sch:
            return get_best_wordnet_sentence(question, story, False)

        return best_sentence

    # Get the difficulty level of the question
    difficulty = question['difficulty']
    sentence = get_best_wordnet_sentence(question, story)

    if difficulty == 'Discourse':
        print('Processing discourse question')
        discourse_type = ''
        qwords = nltk.word_tokenize(question['text'])

        # Determine the discourse type based on keywords in the question
        if 'after' in qwords:
            discourse_type = 'after'
        elif 'first' in qwords:
            discourse_type = 'first'
        elif 'before' in qwords:
            discourse_type = 'before'

        sent_index = 0
        sentences = nltk.sent_tokenize(story.get("sch", story["text"]))

        # Find the index of the selected sentence
        for i, sent in enumerate(sentences):
            if sent == sentence:
                sent_index = i
                break

        # Return sentences based on discourse type and selected sentence index
        if discourse_type and discourse_type not in nltk.word_tokenize(sentence):       
            if discourse_type == 'after' and sent_index < len(sentences):
                return sentences[sent_index:sent_index+2]
            elif (discourse_type == 'first' or discourse_type == 'before') and sent_index > 0:
                return sentences[sent_index-1:sent_index+1]
            elif sent_index > 0:
                return sentences[sent_index-1:sent_index+1]

    return sentence

def baseline(qbow, sentences, stopwords):
    # Calculate overlap between question bag-of-words and sentence bag-of-words
    answers = [(len(qbow & get_bow(sent, stopwords)), sent) for sent in sentences]

    # Sort answers based on overlap count in descending order
    answers.sort(key=operator.itemgetter(0), reverse=True)

    # Get the best answer (sentence with highest overlap)
    best_answer = answers[0][1]

    return best_answer
