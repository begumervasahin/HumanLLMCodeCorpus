import numpy as np

def cosine_similarity(vector_a, vector_b):
    """
    Calculate the cosine similarity between two vectors.
    
    Args:
    - vector_a (numpy.ndarray): First vector
    - vector_b (numpy.ndarray): Second vector
    
    Returns:
    - float: Cosine similarity between the two vectors
    """
    dot_product = np.dot(vector_a, vector_b)
    norm_a = np.linalg.norm(vector_a)
    norm_b = np.linalg.norm(vector_b)
    return dot_product / (norm_a * norm_b)

def vectorize_sentences(*sentences):
    """
    Vectorize sentences by creating binary vectors for each word.
    
    Args:
    - sentences (tuple): Input sentences as strings
    
    Returns:
    - list: List of binary vectors representing the input sentences
    """
    combined_sentence = ' '.join(sentences)
    words = combined_sentence.split()
    unique_words = list(set(words))
    binary_vectors = []

    for sentence in sentences:
        sentence_words = sentence.split()
        binary_vector = [1 if word in sentence_words else 0 for word in unique_words]
        binary_vectors.append(binary_vector)

    return binary_vectors

def fully_vectorized_cosine_similarity():
    """
    Perform fully vectorized cosine similarity comparison for a set of sentences.
    """
    sentences = [
        "How many teams participated in the first world cup",
        "Who holds the record for top scorer in a single World Cup",
        "Should I sell my car myself or trade it in",
        "How important is car maintenance",
        "How much does AWS SAM cost to use",
        "Which languages does AWS SAM support"
    ]

    sentence_vectors = vectorize_sentences(*sentences)

    for i, vector_a in enumerate(sentence_vectors):
        print('\n')
        for j, vector_b in enumerate(sentence_vectors):
            similarity = cosine_similarity(vector_a, vector_b)
            print('{0:.4f}'.format(similarity), end='\t', flush=True)
    print('\n')

# Execute the fully vectorized cosine similarity trial
fully_vectorized_cosine_similarity()
