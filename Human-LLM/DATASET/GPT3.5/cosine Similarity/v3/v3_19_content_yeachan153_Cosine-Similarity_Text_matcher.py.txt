# Function to split text into lowercase words
def split_text(text):
    return text.lower().split()

# Function to convert word frequencies to a vector representation
def freqs_to_vector(freqs, vocabulary):
    vector = [freqs[word] if word in freqs else 0 for word in vocabulary]
    return vector

# Function to calculate the cosine similarity between two vectors
def cosine_similarity(vector_a, vector_b):
    dot_product = sum(a * b for a, b in zip(vector_a, vector_b))
    mag_a = math.sqrt(sum(a**2 for a in vector_a))
    mag_b = math.sqrt(sum(b**2 for b in vector_b))
    return dot_product / (mag_a * mag_b) if mag_a * mag_b != 0 else 0

# Function to rank documents based on a query vector
def rank_documents(query_vector, corpus, num=100):
    similarities = {}
    for doc_id, info in corpus.items():
        freq_vect = info['freq_vect']
        similarities[doc_id] = cosine_similarity(query_vector, freq_vect)
    ranked_ids = sorted(similarities, key=lambda i: similarities[i], reverse=True)[:num]
    ranked_sims = [similarities[id] for id in ranked_ids]
    return ranked_ids, ranked_sims
