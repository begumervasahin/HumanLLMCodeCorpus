from __future__ import division
import json
import math
import numpy as np

def load_word2vec(filename):
    word_vectors = {}
    with open(filename, "r") as file:
        for line in file:
            parts = line.strip().split()
            word = parts[0]
            vector = np.array([float(x) for x in parts[1:]])
            word_vectors[word] = vector
    return word_vectors

def load_contexts(filename):
    contexts = {}
    with open(filename, "r") as file:
        for line in file:
            word, count, context_json = line.strip().split("\t")
            count = int(count)
            context_dict = json.loads(context_json)
            contexts[word] = {"count": count, "context": context_dict}
    print(f"Loaded {len(contexts)} words from {filename}")
    return contexts

def calculate_cosine_similarity_sparse(v1, v2):
    numerator = sum(v1[key] * v2[key] for key in v1 if key in v2)
    denominator_v1 = math.sqrt(sum(v1[key] ** 2 for key in v1))
    denominator_v2 = math.sqrt(sum(v2[key] ** 2 for key in v2))
    similarity = numerator / (denominator_v1 * denominator_v2)
    return similarity

def calculate_cosine_similarity_dense(v1, v2):
    similarity = np.dot(v1, v2) / (np.linalg.norm(v1) * np.linalg.norm(v2))
    return similarity

def find_nearest_words(word_vectors, target_vector, exclude_words, similarity_metric):  
    similarities = {word: similarity_metric(vector, target_vector) for word, vector in word_vectors.items() if word not in exclude_words}
    sorted_similarities = sorted(similarities.items(), key=lambda x: x[1], reverse=True)
    return sorted_similarities[:10]

# Example usage
filename_word2vec = 'path/to/word2vec_file.txt'
word2vec = load_word2vec(filename_word2vec)

target_word = 'example_word'
target_vector = word2vec[target_word]
exclude_words = ['word1', 'word2']  # Words to exclude from similarity calculation      

nearest_words = find_nearest_words(word2vec, target_vector, exclude_words, calculate_cosine_similarity_sparse)
print(nearest_words)
