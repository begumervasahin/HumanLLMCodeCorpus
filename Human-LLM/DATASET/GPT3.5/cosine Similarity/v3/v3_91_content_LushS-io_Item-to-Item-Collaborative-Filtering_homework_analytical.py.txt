import pandas as pd
import numpy as np
from sklearn.preprocessing import normalize
from sklearn.metrics.pairwise import cosine_similarity
from scipy.sparse import csr_matrix, csc_matrix, find
from itertools import combinations
import time

def cosine_similarity_matrix(matrix):
    row_normed_matrix = normalize(matrix.tocsc(), axis=0)
    return row_normed_matrix.T * row_normed_matrix

def cosine_similarity(a, b):
    dot_product = np.dot(a, b)
    norm_a = np.linalg.norm(a)
    norm_b = np.linalg.norm(b)
    return dot_product / (norm_a * norm_b)

def generate_combinations_nd(array, r, axis=0):
    array = np.asarray(array)
    if axis < 0:
        axis += array.ndim
    indices = np.arange(array.shape[axis])
    dt = np.dtype([('', np.intp)] * r)
    indices = np.fromiter(combinations(indices, r), dt)
    indices = indices.view(np.intp).reshape(-1, r)
    return np.take(array, indices, axis=axis)

def import_data_to_dataframe():
    user_1 = np.array([1, 0, 3, 0, 0, 5, 0, 0, 5, 0, 4, 0])
    user_2 = np.array([0, 0, 5, 4, 0, 0, 4, 0, 0, 2, 1, 3])
    user_3 = np.array([2, 4, 0, 1, 2, 0, 3, 0, 4, 3, 5, 0])
    test_1 = np.array([4, 5, 0, 5, 1, 0])
    test_2 = np.array([0, 3, 4, 3, 1, 2])
    test_3 = np.array([2, 0, 1, 3, 0, 4])
    users_data = pd.DataFrame(data=(test_1, test_2, test_3))
    return users_data

def calculate_sparse_matrix_average(matrix):
    (row_indices, _, values) = find(matrix)
    counts = np.bincount(row_indices)
    sums = np.bincount(row_indices, weights=values)
    return sums / counts

users_df = import_data_to_dataframe()
users_np = users_df.to_numpy()
sparse_matrix = csr_matrix(users_np)

print('Dataframe before converting to a sparse matrix:\n{}'.format(users_df))
print('Sparse Matrix Representation:\n{}'.format(sparse_matrix))

cosine_similarity_mat = cosine_similarity_matrix(csc_matrix(users_np))
print('Cosine Similarity Matrix (dense representation):\n{}'.format(cosine_similarity_mat.todense()))

sparse_matrix_avg = calculate_sparse_matrix_average(sparse_matrix)
print('Average by Row for Sparse Matrix:\n{}'.format(sparse_matrix_avg))

centered_sparse_matrix = csr_matrix((sparse_matrix.data - np.repeat(sparse_matrix_avg, np.diff(sparse_matrix.indptr)),
                                    sparse_matrix.indices, sparse_matrix.indptr), shape=sparse_matrix.shape)
print('Centered Sparse Matrix:\n{}'.format(centered_sparse_matrix.todense()))
print('Transposed Centered Sparse Matrix:\n{}'.format(centered_sparse_matrix.T.todense()))

cosine_similarity_centered = cosine_similarity_matrix(centered_sparse_matrix)
print('Cosine Similarity Matrix for Centered Sparse Matrix:\n{}'.format(cosine_similarity_centered.todense()))

start_time = time.time()
similarity_matrix = centered_sparse_matrix.dot(centered_sparse_matrix.T).toarray()
end_time = time.time()
print('Similarity Matrix (dense representation):\n{}'.format(similarity_matrix))
print('Time taken for similarity calculation: {} seconds'.format(end_time - start_time))

centered_matrix_dense = centered_sparse_matrix.todense()
print('Cosine Similarity (User 1, User 2):', cosine_similarity(centered_matrix_dense[0], centered_matrix_dense[1].T))
print('Cosine Similarity (User 1, User 3):', cosine_similarity(centered_matrix_dense[0], centered_matrix_dense[2].T))
print('Cosine Similarity (User 2, User 3):', cosine_similarity(centered_matrix_dense[1], centered_matrix_dense[2].T))

dot_product_centered = np.dot(centered_sparse_matrix, centered_sparse_matrix.T)
print('Dot Product of Centered Sparse Matrix and its Transpose:\n{}'.format(dot_product_centered.todense()))

cosine_similarity(users_df)
