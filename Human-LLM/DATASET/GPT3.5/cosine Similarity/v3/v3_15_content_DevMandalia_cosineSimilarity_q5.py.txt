import distsim

# Load the word vectors dictionary
word_to_vec_dict = distsim.load_word2vec("nyt_word2vec.4k")

# List of words to process
words_to_process = ['america', 'years', 'great', 'run', 'between', 'wife']

# Function to display nearest words for a given word
def display_nearest_words(word):
    print("Word is::::", word)
    nearest_words = distsim.show_nearest(word_to_vec_dict, word_to_vec_dict[word], set([word]), distsim.cossim_dense)
    for i, (nearest_word, score) in enumerate(nearest_words):
        print(f"{i}: {nearest_word} ({score})")
    print("--------------------")

# Iterate through the list of words and display nearest words
for word in words_to_process:
    display_nearest_words(word)
