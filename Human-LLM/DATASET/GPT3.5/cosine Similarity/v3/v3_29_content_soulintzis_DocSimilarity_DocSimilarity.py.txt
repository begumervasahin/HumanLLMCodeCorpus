import string
import numpy as np
import math as m
from collections import OrderedDict

def binomial_coefficient(n):
    binom = int(m.factorial(n) / (m.factorial(2) * m.factorial(n - 2)))
    k = int(input(f"Enter K (must be smaller than {binom}): "))
    if n == 2:
        return 1
    elif k < binom:
        return k
    else:
        print(f"K must be smaller than {binom}. Default K = 1.")
        return 1

# Initialize variables
similarity = {}
dictionary = []
file_names = []
doc_count = []

# Input number of documents
number_of_docs = int(input("How many documents do you want to insert: "))
while number_of_docs < 2:
    number_of_docs = int(input("Enter at least 2 documents: "))

# Input document names
for i in range(number_of_docs):
    name = input(f'Enter the name of document {i + 1}: ')
    if not name.endswith('.txt'):
        print('Document must have a ".txt" extension!')
        exit()
    file_names.append(name)

# Calculate K
K = binomial_coefficient(number_of_docs)

# Create dictionary and count occurrences of words in each document
for name in file_names:
    with open("Documents/" + name, 'r') as file:
        for line in file:
            for word in line.split():
                word = word.translate(str.maketrans('', '', string.punctuation)).lower()
                if word not in dictionary:
                    dictionary.append(word)

# Count occurrences of words in each document
for name in file_names:
    count = [0] * len(dictionary)
    with open("Documents/" + name, 'r') as file:
        for line in file:
            for word in line.split():
                word = word.translate(str.maketrans('', '', string.punctuation)).lower()
                position = dictionary.index(word)
                count[position] += 1
    doc_count.append(count)

# Calculate similarity between pairs of documents
for i in range(number_of_docs):
    for j in range(i + 1, number_of_docs):
        doc1 = doc_count[i]
        doc2 = doc_count[j]
        norm1 = np.linalg.norm(doc1, ord=2)
        norm2 = np.linalg.norm(doc2, ord=2)
        dot_product = np.dot(doc1, doc2)
        result = dot_product / (norm1 * norm2)
        pair_of_docs = file_names[i] + " and " + file_names[j]
        similarity[pair_of_docs] = result

# Sort and print top K similar documents
sorted_dict = OrderedDict(sorted(similarity.items(), key=lambda x: x[1], reverse=True)[:K])

print("\nTop K Similar Documents listed below:")
for key, value in sorted_dict.items():
    print(f"{key}: {value:.2f}, {value * 100:.2f}%")
