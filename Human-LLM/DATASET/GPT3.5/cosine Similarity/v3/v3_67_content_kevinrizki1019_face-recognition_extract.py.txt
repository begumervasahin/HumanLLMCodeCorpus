import cv2
import numpy as np

def extract_features(img_path, vector_size=32):
    try:
        # Read the image
        image = cv2.imread(img_path)

        # Create KAZE detector
        kaze_detector = cv2.KAZE_create()

        # Detect keypoints and compute descriptors
        keypoints, descriptors = kaze_detector.detectAndCompute(image, None)

        # Select top vectorSize keypoints based on response
        keypoints = sorted(keypoints, key=lambda x: -x.response)[:vector_size]

        # Flatten descriptors and ensure correct size
        descriptors = flatten_descriptors(descriptors, vector_size)

    except cv2.error as e:
        print('Error while extracting image:', e)
        return None

    return descriptors

def flatten_descriptors(descriptors, vector_size):
    # Flatten descriptors
    descriptors = descriptors.flatten()

    # Ensure descriptor size is equal to vectorSize * 64
    dsc_size = vector_size * 64
    if descriptors.size < dsc_size:
        descriptors = np.concatenate([descriptors, np.zeros(dsc_size - descriptors.size)])
    elif descriptors.size > dsc_size:
        descriptors = descriptors[:dsc_size]

    return descriptors
