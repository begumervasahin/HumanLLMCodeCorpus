import sys
import re
import time
import numpy as np
from os import listdir
from os.path import isfile, join
from stemming import PorterStemmer

class SearchEngine:
    def __init__(self):
        pass

    def parse_command(self):
        if len(sys.argv) >= 4 and (sys.argv[1] == "search" or sys.argv[1] == "index"):
            if sys.argv[1] == "search":
                self.command_search(sys.argv[2], sys.argv[3], sys.argv[4:])
            elif sys.argv[1] == "index":
                self.command_index(sys.argv[2], sys.argv[3], sys.argv[4])
        else:
            print("Please enter a valid command.")
            print("Example:")
            print("python my_search_engine search index_dir num_docs keyword_list")
            print("python my_search_engine index collection_dir index_dir stopwords.txt")

    def command_index(self, file_path, index_dir, stopwords_path):
        stopwords = self.load_stopwords(stopwords_path)
        file_content_list, file_names = self.load_files(file_path)
        tf_dicts = [self.calculate_tf(self.tokenize(content, stopwords)) for content in file_content_list]
        df_dict = self.calculate_df(tf_dicts)
        self.save_index(index_dir, df_dict, tf_dicts, file_names)

    def command_search(self, index_dir, top_n, keywords, feedback=True):
        keywords_text = " ".join(keywords)
        stopwords = self.load_stopwords(join(index_dir, "stopwords.txt"))
        keyword_list = self.tokenize(keywords_text, stopwords)
        keyword_tf_dict = self.calculate_tf(keyword_list)
        query_tfidf = self.calculate_tfidf(keyword_tf_dict, index_dir)
        doc_tfidf_dict = self.load_doc_tfidf(index_dir)
        similarity_scores = self.calculate_similarity(query_tfidf, doc_tfidf_dict)
        self.display_results(similarity_scores, int(top_n))

    def load_stopwords(self, stopwords_path):
        with open(stopwords_path) as f:
            return set(f.read().splitlines())

    def load_files(self, file_path):
        file_content_list = []
        file_names = []
        for file_name in listdir(file_path):
            if isfile(join(file_path, file_name)) and file_name != '.DS_Store':
                with open(join(file_path, file_name), 'r') as f:
                    file_content_list.append(f.read())
                    file_names.append(file_name.replace(",", " "))
        return file_content_list, file_names

    def tokenize(self, text, stopwords):
        stemmer = PorterStemmer()
        terms = re.findall(r'\b\w+\b', text.lower())
        return [stemmer.stem(term) for term in terms if term not in stopwords]

    def calculate_tf(self, term_list):
        tf_dict = {}
        for term in term_list:
            tf_dict[term] = tf_dict.get(term, 0) + 1
        return tf_dict

    def calculate_df(self, tf_dicts):
        df_dict = {}
        for tf_dict in tf_dicts:
            for term in tf_dict:
                df_dict[term] = df_dict.get(term, 0) + 1
        return df_dict

    def save_index(self, index_dir, df_dict, tf_dicts, file_names):
        with open(join(index_dir, "index.txt"), 'w') as f:
            for term, df in df_dict.items():
                idf = np.log(len(tf_dicts) / df)
                doc_weights = [(file_names[i], tf_dict.get(term, 0) * idf) for i, tf_dict in enumerate(tf_dicts) if term in tf_dict]
                doc_info = ",".join(f",{file_name},{weight}" for file_name, weight in doc_weights)
                f.write(f"{term}{doc_info},{idf:.3f}\n")

    def calculate_tfidf(self, tf_dict, index_dir):
        idf_dict = {}
        with open(join(index_dir, "index.txt")) as f:
            for line in f:
                parts = line.split(',')
                term, idf = parts[-1].split()[-1], float(parts[-1])
                idf_dict[term] = idf
        return {term: tf * idf_dict.get(term, 0) for term, tf in tf_dict.items()}

    def load_doc_tfidf(self, index_dir):
        doc_tfidf_dict = {}
        with open(join(index_dir, "index.txt")) as f:
            for line in f:
                parts = line.split(',')
                term = parts[0]
                for i in range(1, len(parts) - 1, 2):
                    doc_tfidf_dict.setdefault(parts[i], {})[term] = float(parts[i + 1])
        return doc_tfidf_dict
