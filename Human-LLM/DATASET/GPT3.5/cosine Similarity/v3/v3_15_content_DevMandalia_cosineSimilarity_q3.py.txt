import distsim

def print_nearest_words(word_to_ccdict, target_word, target_set, similarity_func):
    """Prints the nearest words for a target word based on similarity."""
    print(f"Target word: {target_word}")
    for i, (nearest_word, score) in enumerate(distsim.show_nearest(word_to_ccdict, word_to_ccdict[target_word], target_set, similarity_func), start=1):
        print(f"{i}: {nearest_word} ({score})")
    print("--------------------")

# Load word contexts from the provided file
word_to_ccdict = distsim.load_contexts("nytcounts.4k")

# Define target words and sets
targets = ['america', 'years', 'great', 'run', 'between', 'wife']
target_sets = [{target} for target in targets]  # Using set comprehension

# Print nearest words for each target word
for target, target_set in zip(targets, target_sets):
    print_nearest_words(word_to_ccdict, target, target_set, distsim.cossim_sparse)

# Further exploration: comparing 'big' and 'tiny' with 'great'
print("Further exploration: big from great")
print_nearest_words(word_to_ccdict, 'big', {'big'}, distsim.cossim_sparse)
print("--------------------")
print("Further exploration: tiny from great")
print_nearest_words(word_to_ccdict, 'tiny', {'tiny'}, distsim.cossim_sparse)
print("--------------------")
