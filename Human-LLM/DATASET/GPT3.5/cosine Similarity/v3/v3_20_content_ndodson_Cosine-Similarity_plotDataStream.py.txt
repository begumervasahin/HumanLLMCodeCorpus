import json
import pandas as pd
import matplotlib.pyplot as plt

# Path to the Twitter data file
tweets_data_path = '/home/nick/projects/twitter_data.txt'

# Read data from the file and store it in a list of dictionaries
tweets_data = []
with open(tweets_data_path, "r") as tweets_file:
    for line in tweets_file:
        try:
            tweet = json.loads(line)
            tweets_data.append(tweet)
        except json.JSONDecodeError:
            continue

# Create a DataFrame to store the tweets data
tweets_df = pd.DataFrame(tweets_data)

# Extract relevant columns from the DataFrame
tweets_df = tweets_df[['text', 'lang', 'place']]

# Extract country information from the 'place' column
tweets_df['country'] = tweets_df['place'].apply(lambda place: place['country'] if place is not None else None)

# Count the number of tweets by language
tweets_by_lang = tweets_df['lang'].value_counts()

# Plot the top 5 languages
fig, ax = plt.subplots()
ax.tick_params(axis='x', labelsize=15)
ax.tick_params(axis='y', labelsize=10)
ax.set_xlabel('Languages', fontsize=15)
ax.set_ylabel('Number of Tweets', fontsize=15)
ax.set_title('Top 5 Languages', fontsize=15, fontweight='bold')
tweets_by_lang[:5].plot(ax=ax, kind='bar', color='red')

# Show the plot
plt.show()

print("Data visualization completed.")
