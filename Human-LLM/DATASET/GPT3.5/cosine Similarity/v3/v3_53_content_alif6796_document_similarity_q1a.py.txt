import os
import numpy as np
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

def load_documents(parent_dir):
    """Load documents from the specified directory."""
    doc_list = []
    name_list = []
    for file in os.listdir(parent_dir):
        full_path = os.path.join(parent_dir, file)
        with open(full_path, 'r') as file_ptr:
            name_list.append(file_ptr.name)
            doc_list.append(file_ptr.read())
    return doc_list, name_list

def calculate_similarity(query, doc_list):
    """Calculate cosine similarity between the query and documents."""
    vectorizer = TfidfVectorizer(stop_words='english')
    matrix = vectorizer.fit_transform([query] + doc_list)
    similarity = cosine_similarity(matrix[0:1], matrix)
    return np.ravel(similarity)

def print_similar_documents(similarity, name_list):
    """Print names of documents with similarity greater than 0."""
    result = np.argwhere(similarity > 0)
    print(similarity)
    for idx in result:
        idx = np.asscalar(idx)
        print(name_list[idx])
        print('-------------------------------------------------------------')

# Define the query
query = 'printer'

# Define the parent directory containing the documents
parent_dir = "/home/ai55/newsgroups/"

# Load documents
doc_list, name_list = load_documents(parent_dir)

# Calculate cosine similarity
similarity = calculate_similarity(query, doc_list)

# Print similar documents
print_similar_documents(similarity, name_list)
