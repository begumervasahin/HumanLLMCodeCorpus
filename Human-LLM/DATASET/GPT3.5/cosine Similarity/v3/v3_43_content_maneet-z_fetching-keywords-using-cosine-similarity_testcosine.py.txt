import math
import os

def get_file_data(file_name):
    if not os.path.exists(file_name):
        print('File is not present in the directory')
        return None

    with open(file_name, 'r') as file:
        data = file.read()

    cleaned_data = clean_text(data)
    split_data = cleaned_data.split('.')
    return split_data

def get_test_words(test_word_file):
    if not os.path.exists(test_word_file):
        print('File is not present in the directory')
        return None

    with open(test_word_file, 'r') as file:
        test_words = [line.strip() for line in file]

    if len(test_words) != 1:
        print('Please check the file you have uploaded')
        return None

    return test_words

def get_common_words(common_word_file):
    common_words = {}
    with open(common_word_file, 'r') as file:
        for line in file:
            common_words[line.strip()] = 0
    return common_words

def get_word_frequency(cleaned_data, test_words, common_words):
    word_count = {}
    for test_word in test_words:
        word_count[test_word] = {}
        for line in cleaned_data.split('.'):
            if test_word in line:
                words_in_line = line.split()
                for word in words_in_line:
                    if word not in common_words and word != test_word:
                        word_count[test_word][word] = word_count[test_word].get(word, 0) + 1
    return word_count

def clean_text(text):
    punctuation = [',', '\'', '\"', ':', ';', '(', ')', '[', ']', '\n', '_']
    cleaned_text = ''
    for char in text:
        if char == '-' and ('--' in text or '- ' in text or ' -' in text):
            cleaned_text += ' '
        elif char in punctuation:
            cleaned_text += ' '
        else:
            cleaned_text += char
    cleaned_text = ' '.join(cleaned_text.split())  # Remove extra spaces
    return cleaned_text

def calculate_similarity(profile_count, test_words):
    similarity_scores = []
    for test_word in test_words[1:]:
        pq = 0
        count_denom_p = sum(value ** 2 for value in profile_count[test_words[0]].values())
        count_denom_q = sum(value ** 2 for value in profile_count[test_word].values())  
        for word in profile_count[test_words[0]]:
            if word in profile_count[test_word]:
                pq += profile_count[test_words[0]][word] * profile_count[test_word][word]
        final_denom = math.sqrt(count_denom_p * count_denom_q)
        similarity = round(pq / final_denom, 3) if final_denom != 0 else 0
        similarity_scores.append([test_word, similarity])
    return sorted(similarity_scores, key=lambda l: l[1], reverse=True)

def main(file_name, test_word_file, common_word_file=None):
    if not file_name or not test_word_file:
        print('Please provide valid file names')
        return

    file_data = get_file_data(file_name)
    if file_data is None:
        print('Unable to process the file due to incorrect upload')
        return

    test_words = get_test_words(test_word_file)
    if test_words is None:
        print('Unable to process the test word file due to incorrect upload')
        return

    cleaned_data = clean_text(' '.join(file_data))
    common_words = get_common_words(common_word_file) if common_word_file else {}       
    word_frequency = get_word_frequency(cleaned_data, test_words, common_words)
    similarity_scores = calculate_similarity(word_frequency, test_words)

    for data in similarity_scores:
        print(data)

if __name__ == "__main__":
    main('TCGA-XC-AA0X.B22BD117-8DF0-47FF-BE25-9FE2DA916DB6.txt', 'profile.txt', 'common.txt')
