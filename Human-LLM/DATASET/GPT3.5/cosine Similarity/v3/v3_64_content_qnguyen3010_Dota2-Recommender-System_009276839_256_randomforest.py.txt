import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import confusion_matrix, accuracy_score, f1_score

def load_dataset(train_file, test_file):
    train_data = pd.read_csv(train_file)
    test_data = pd.read_csv(test_file)
    return train_data, test_data

def preprocess_data(train_data, test_data):
    y_train = train_data.iloc[:, 0].values
    X_train = train_data.iloc[:, 4:].values
    y_test = test_data.iloc[:, 0].values
    X_test = test_data.iloc[:, 4:].values
    return X_train, y_train, X_test, y_test

def train_random_forest(X_train, y_train):
    classifier = RandomForestClassifier(max_features='log2', n_estimators=100, criterion='entropy', random_state=0)
    classifier.fit(X_train, y_train)
    return classifier

def evaluate_model(classifier, X_test, y_test):
    y_pred = classifier.predict(X_test)
    conf_matrix = confusion_matrix(y_test, y_pred)
    accuracy = accuracy_score(y_test, y_pred)
    f1 = f1_score(y_test, y_pred, average='macro')
    return conf_matrix, accuracy, f1

def print_results(conf_matrix, accuracy, f1):
    print("Confusion Matrix:")
    print(conf_matrix)
    print("Accuracy:", accuracy)
    print("F1 Score:", f1)

if __name__ == "__main__":
    # Load dataset
    train_data, test_data = load_dataset('dota2Train.csv', 'dota2Test.csv')

    # Preprocess data
    X_train, y_train, X_test, y_test = preprocess_data(train_data, test_data)

    # Train the Random Forest model
    classifier = train_random_forest(X_train, y_train)

    # Evaluate the model
    conf_matrix, accuracy, f1 = evaluate_model(classifier, X_test, y_test)

    # Print evaluation results
    print_results(conf_matrix, accuracy, f1)
