import os
import pickle
import sys
from extract import extractFeatures

def generate_pickle_from_batch(pickle_path, images_path):
    # Get a list of image files
    image_files = [os.path.join(images_path, filename) for filename in sorted(os.listdir(images_path))]
    database = {}
    total_images = len(image_files)
    extracted_count = 0

    # Iterate over each image file
    for image_file in image_files:
        extracted_count += 1
        print(f'-> [Entry {extracted_count}/{total_images}] Extracting {image_file}               \r', end='')
        database[image_file] = extractFeatures(image_file)

    # Save the extracted features to a pickle file
    with open(pickle_path, 'wb') as dbfile:
        pickle.dump(database, dbfile)

    print('\nDone.')

def generate_database(database_path, pins_path):
    # Create the database directory if it doesn't exist
    if not os.path.exists(database_path):
        os.makedirs(database_path)

    if os.path.exists(pins_path):
        person_dirs = [os.path.join(pins_path, person) for person in sorted(os.listdir(pins_path))]     
        total_persons = len(person_dirs)
        print(f'Found {total_persons} directories on {pins_path}')

        # Iterate over each directory containing images
        for index, person_dir in enumerate(person_dirs, 1):
            print(f'[{index}/{total_persons}] Generating {os.path.basename(person_dir)}.db')
            generate_pickle_from_batch(os.path.join(database_path, os.path.basename(person_dir)) + '.db', person_dir)

        print(f'Extracted {total_persons} entries.')
    else:
        print('Directory not found')

def join_databases(directory_path, database_path):
    merged_db = {}
    print(f'Joining databases from {directory_path} to {database_path}')

    if os.path.exists(directory_path):
        database_files = [os.path.join(directory_path, filename) for filename in sorted(os.listdir(directory_path))]

        # Iterate over each database file and merge them
        for database_file in database_files:
            print('->', database_file)
            with open(database_file, 'rb') as tempdb:
                temp_database = pickle.load(tempdb)
                merged_db.update(temp_database)

        # Save the merged database to a single pickle file
        with open(database_path, 'wb') as dbfile:
            pickle.dump(merged_db, dbfile)

        print('Done.')
    else:
        print('Directory not found')

if __name__ == '__main__':
    if len(sys.argv) > 1:
        if os.path.exists(sys.argv[1]):
            generate_database('db', sys.argv[1])
            join_databases('db', 'pins.db')
        else:
            print('Fatal: not a directory')
    else:
        print('Please specify a file. It must contain folders which contain pictures.') 
        print('Usage: python3 generate-pickle-file.py <folder-name>')
