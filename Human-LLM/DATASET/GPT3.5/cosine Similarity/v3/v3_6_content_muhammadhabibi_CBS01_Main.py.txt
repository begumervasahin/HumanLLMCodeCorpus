import nltk
import yaml
import Preprocessing
import TF_IDF
import CosineSimilarity

# Alias imports for readability
pp = Preprocessing
cs = CosineSimilarity

# Load comments from file
with open('komentar.json', 'r') as file:
    komentar = yaml.safe_load(file)

# Load preprocessed comments
with open('komentar_bersih.txt', 'r') as file:
    data = file.read().split('\n')
    komen_bersih = [komen for komen in data]

# Generate TF-IDF vectors for comments
vector = [cs.TFIDF(comment) for comment in komen_bersih]

# Test sentences
kalimat = [
    "sebaiknya waktu kuliahnya jangan terlalu sore",
    "lebih keras lagi suara",
    "buku rujuk kurang jelas mau pakai mana"
]

# Process each test sentence
for test in kalimat:
    print('---------------------------------------------------------------------------------------')
    print('Data Testing:', test)

    # Generate TF-IDF vector for the test sentence
    input_vector = cs.TFIDF(nltk.Text(str(test)))

    # Compare similarity with each comment
    similarity_results = []
    for i, t in enumerate(vector):
        similarity_score = cs.similarity(input_vector, t)
        similarity_results.append((i, similarity_score, komen_bersih[i]))

    # Sort and print results
    most_similar = sorted(similarity_results, key=lambda x: x[1], reverse=True)[0]
    print(most_similar)
    print("------------------------------------------------------------------------")   
    for (index, cosine_similarity, comment) in similarity_results:
        if cosine_similarity >= 0.93:
            print(cosine_similarity, comment)
