import math

def calculate_cosine_similarity(vector1, vector2):
    """Calculate the cosine similarity between two vectors."""
    if len(vector1) != len(vector2):
        return 0

    dot_product = sum(val1 * val2 for val1, val2 in zip(vector1, vector2))
    length1 = math.sqrt(sum(val ** 2 for val in vector1))
    length2 = math.sqrt(sum(val ** 2 for val in vector2))

    if length1 == 0 or length2 == 0:
        return 0

    return dot_product / (length1 * length2)

def read_lines_from_file(filename):
    """Read lines from a file and return them as a list."""
    with open(filename, 'r') as file:
        return [line.strip().lower() for line in file]

# Read words from dict_new.csv, documents from doc.csv, and stopwords from stopword.txt
word_list = read_lines_from_file('dict_new.csv')
doc_list = read_lines_from_file('doc.csv')
stop_words = read_lines_from_file('stopword.txt')

# Filter out stopwords from word list
filtered_word_list = [word for word in word_list if word not in stop_words]

# Calculate cosine similarity for each document
with open('cos.csv', 'w') as output_file:
    for document in doc_list:
        vector = [1 if word in document else 0 for word in filtered_word_list]
        similarity = calculate_cosine_similarity(vector, [1] * len(filtered_word_list))
        output_file.write(f'{similarity}\n')
        print(document)  # Print the document for visualization
