import sys
import getopt
import os
import pickle
from pathlib import Path
from vectorize import img2vec_converter
from cos_sim import matrix_calculator
from top_similar import top10

def display_usage():
    """Display usage information."""
    print("Usage: python <filename.py> -c cuda -m model")
    print("cuda = (True, False)\nDefault CUDA status: False")
    print("To see available models use -h or --help")
    sys.exit(2)

def parse_arguments():
    """Parse command-line arguments."""
    try:
        opts, _ = getopt.getopt(sys.argv[1:], "c:m:h", ['cuda=', 'model=', 'help'])  
    except getopt.GetoptError:
        display_usage()

    # Set default values
    cuda = False
    model = "resnet18"

    if not opts:
        print("Using default options. Model: resnet18, CUDA: False")
    else:
        for opt, arg in opts:
            if opt in ('-c', '--cuda'):
                cuda = arg.lower() == 'true'
            elif opt in ('-m', '--model'):
                model = arg
            elif opt in ('-h', '--help'):
                print_usage()
            else:
                print("Invalid argument:", opt)
                sys.exit(2)

        print("CUDA:", cuda)
        print("Model:", model)

    return cuda, model

def main():
    """Main function."""
    # Parse command-line arguments
    cuda, model = parse_arguments()

    # Calculate cosine similarity matrix
    matrix = matrix_calculator(img2vec_converter(model, cuda))

    # Save cosine similarity matrix
    print("Saving cosine similarity matrix")
    output_path = Path("pickles") / model / "sim_matrix.pkl"
    with open(output_path, 'wb') as file:
        pickle.dump(matrix, file)

    # Find and display top 10 similar images
    top10(matrix, model)

if __name__ == "__main__":
    main()
