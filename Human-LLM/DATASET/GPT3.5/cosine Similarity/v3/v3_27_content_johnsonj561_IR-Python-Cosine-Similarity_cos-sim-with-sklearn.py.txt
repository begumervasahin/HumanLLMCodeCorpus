import re
import os
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from nltk.stem import PorterStemmer
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

# Print header information
print("""
Assignment 1: NLTK text preprocessing and cosine similarity
Information Retrieval: Florida Atlantic University, Fall 2017
Justin J, jjohn273
""")

# Define the directory containing text files
data_dir = 'hw1-dataset'
DATA_SET_DIR = os.path.abspath(data_dir)

# Display the directory path
print('\nGetting List of text files from ' + DATA_SET_DIR)

# Get a list of text files in the directory
files = os.listdir(DATA_SET_DIR)
files = filter(lambda x: re.match(r'.*\.txt$', x, re.I), files)

# Print the retrieved file list
print('\nFile list retrieved from ' + DATA_SET_DIR)

# Initialize Porter Stemmer and NLTK English stop words
ps = PorterStemmer()
nltk_stop_words = set(stopwords.words('english'))
corpus = []

# Preprocess each text file
for f in files:
    with open(os.path.join(DATA_SET_DIR, f), 'r') as strm:
        words = word_tokenize(strm.read())
        words = [w for w in words if w not in nltk_stop_words]
        words = [ps.stem(w) for w in words]
        corpus.append(' '.join(words))

# Create TF-IDF vectorizer and compute TF-IDF matrix
vectorizer = TfidfVectorizer()
tfidf_matrix = vectorizer.fit_transform(corpus)
terms = vectorizer.get_feature_names()

# Print TF-IDF values for each term in the corpus
print('\nPrinting tfidf values for each term:')
print('\t\t Doc1         Doc2        Doc3        Doc4')
for idx, row in enumerate(tfidf_matrix.toarray().transpose()):
    print('{:<10}'.format(terms[idx]) + '\t' + str(row))

# Calculate cosine similarity matrix based on TF-IDF matrix
cos_sim_matrix = cosine_similarity(tfidf_matrix)

# Print cosine similarity matrix
print('\nPrinting cosine similarity matrix: ')
print('\t Doc1         Doc2        Doc3        Doc4        Doc5')
for idx, row in enumerate(cos_sim_matrix):
    print('Doc' + str(idx + 1) + '\t' + str(row))

# Print completion message
print('\nComplete.')
