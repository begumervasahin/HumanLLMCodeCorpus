import os
import argparse
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

def load_files(directory):
    """Load files from the specified directory."""
    return [os.path.join(directory, filename) for filename in os.listdir(directory)
            if os.path.isfile(os.path.join(directory, filename))]

def calculate_max_similarity(tfidf_sample, tfidf_corpus):
    """Calculate the maximum similarity between the sample and the corpus."""
    similarity_matrix = cosine_similarity(tfidf_sample, tfidf_corpus)
    max_similarity_index = similarity_matrix.argsort()[0][-2]
    max_similarity_score = similarity_matrix[0][max_similarity_index]
    return max_similarity_index, max_similarity_score

def main(good_dir, bad_dir, sample_file):
    """Main function to compare the sample file with good and bad files."""
    # Load files from directories
    good_files = load_files(good_dir)
    bad_files = load_files(bad_dir)

    # Combine all files
    all_files = good_files + bad_files + [sample_file]

    # Compute TF-IDF vectors
    tfidf_vectorizer = TfidfVectorizer()
    tfidf_good = tfidf_vectorizer.fit_transform(open(file, 'r', errors='ignore').read() for file in good_files)
    tfidf_bad = tfidf_vectorizer.fit_transform(open(file, 'r', errors='ignore').read() for file in bad_files)
    tfidf_sample = tfidf_vectorizer.fit_transform([open(sample_file, 'r', errors='ignore').read()])

    # Calculate similarity scores
    index_good, similarity_good = calculate_max_similarity(tfidf_sample, tfidf_good)
    index_bad, similarity_bad = calculate_max_similarity(tfidf_sample, tfidf_bad)

    # Print results
    print('Maximum similarity with good samples:', similarity_good)
    print('Maximum similarity with bad samples:', similarity_bad)

    # Determine similar document
    similar_document = good_files[index_good] if similarity_good >= similarity_bad else bad_files[index_bad]
    print('Similar document:', similar_document)

if __name__ == '__main__':
    # Parse command-line arguments
    parser = argparse.ArgumentParser(description='Basic cosine similarity comparator.') 
    parser.add_argument('--good', help='directory where the good samples are')
    parser.add_argument('--bad', help='directory where the bad samples are')
    parser.add_argument('sample', help='file to compare with')
    args = parser.parse_args()

    # Check if directories are provided
    if not args.good or not args.bad:
        print('Provide directory paths for good and bad samples')
        exit(1)

    # Run the main function
    main(args.good, args.bad, args.sample)
