from sklearn.feature_extraction.text import TfidfVectorizer
from bs4 import BeautifulSoup, Comment
from nltk.stem import SnowballStemmer
from urllib.parse import urlparse
import pprint
import json
import time
import re
import os


class Indexer:
    def __init__(self):
        self.file = open("WEBPAGES_RAW/bookkeeping.json", "r")
        self.pp = pprint.PrettyPrinter(indent=4)
        self.book = json.load(self.file)
        self.forward_index = {}
        self.inverted_index = {}

    @staticmethod
    def build_path(dir_num, file_num):
        return f"WEBPAGES_RAW/{dir_num}/{file_num}"

    @staticmethod
    def strip_html(html):
        cleaned = " ".join(html.replace("\n", " ").strip().split())
        return cleaned

    @staticmethod
    def soup_clean(soup):
        elements_to_remove = ["script", "style", "link", "comment"]
        for element in soup(elements_to_remove):
            element.extract()

    @staticmethod
    def is_valid_url(url):
        if len(url) > 600:
            return False
        parsed_url = urlparse(url)
        return ".ics.uci.edu" in parsed_url.path and not re.match(
            r".*\.(css|js|bmp|gif|jpe?g|ico|png|tiff?|pdf|zip|rar|txt|py|java|cpp|doc|docx|xls|xlsx|bin)$",
            parsed_url.path.lower())

    def tokenize_html(self, html_str):
        soup = BeautifulSoup(html_str, "html.parser")
        self.soup_clean(soup)
        doc = soup.get_text().lower()
        return [tok for tok in re.split('[^a-z0-9]', doc) if len(tok) > 0]

    def get_documents(self):
        documents = {}
        invalid_count = 0
        for loc, url in self.book.items():
            if self.is_valid_url(url):
                dir_num, file_num = loc.split("/")
                html_str = self.get_html(dir_num, file_num)
                soup = BeautifulSoup(html_str, "html.parser")
                self.soup_clean(soup)
                documents[loc] = soup.get_text()
            else:
                invalid_count += 1
        print(f"> Number of invalid documents: {invalid_count}")
        return documents

    def get_html(self, dir_num, file_num):
        path = self.build_path(dir_num, file_num)
        with open(path, "r") as file:
            return file.read()

    def build_forward_index(self):
        for loc, url in self.book.items():
            if self.is_valid_url(url):
                print(f"Working on document {loc}")
                dir_num, file_num = loc.split("/")
                html_str = self.get_html(dir_num, file_num)
                tokens = self.tokenize_html(html_str)
                self.forward_index[loc] = tokens

    def generate_clean_webpages(self):
        try:
            os.makedirs("NEWEST_WP_CLEAN", exist_ok=True)
            for i in range(75):
                os.makedirs(f"NEWEST_WP_CLEAN/{i}", exist_ok=True)
            print("> Folders generated")
        except OSError:
            print("OSError")
        for loc, url in self.book.items():
            if self.is_valid_url(url):
                dir_num, file_num = loc.split("/")
                html_raw = self.get_html(dir_num, file_num)
                html_str = html_raw.encode("ascii", errors="ignore").decode()
                soup = BeautifulSoup(html_str, "html.parser")
                self.soup_clean(soup)
                path = f"NEWEST_WP_CLEAN/{dir_num}/{file_num}"
                with open(path, "w") as file:
                    clean_text = soup.get_text()
                    file.write(self.strip_html(clean_text))

    def build_inverted_index(self):
        print("> Building Inverted Index")
        start_time = time.time()
        documents = self.get_documents()
        num_docs = len(documents)
        print(f"> Number of documents: {num_docs}")
        print("> Gathering documents completed.")

        def tokenizer(doc):
            stemmer = SnowballStemmer('english')
            return [stemmer.stem(token) for token in re.split('[^a-z0-9]', doc) if len(token) > 0]

        tfidf = TfidfVectorizer(norm=None, use_idf=True, lowercase=True, smooth_idf=False,
                                sublinear_tf=True, tokenizer=tokenizer, stop_words='english')
        rep = tfidf.fit_transform(documents.values())
        tokens = tfidf.get_feature_names()
        idf = tfidf.idf_
        print("\n> TF-IDF Vectorization completed.")

        index = 0
        alphanum = "0123456789abcdefghijklmnopqrstuvwxyz"
        document_locs = list(documents.keys())
        rep_array = rep.toarray()
        for i, token in enumerate(tokens):
            if not token.startswith(alphanum[index]):
                self.write_to_json(alphanum[index])
                self.inverted_index = {}
                index += 1
            self.inverted_index[token] = [idf[i] - 1, {}]
            for k in range(num_docs):
                tfidf_value = rep_array[k, i]
                if tfidf_value != 0:
                    self.inverted_index[token][1][document_locs[k]] = tfidf_value
            self.inverted_index[token][1] = sorted(self.inverted_index[token][1].items(), key=lambda x: x[1], reverse=True)
        self.write_to_json(alphanum[index])
        end_time = time.time()
        print(f"> Total time for index construction: {end_time - start_time} seconds")
        print(f"\n> Number of unique terms: {len(self.inverted_index)}")

    def search_forward_index(self, token):
        return [document_loc for document_loc, tokens in self.forward_index.items() if token in tokens]

    def print_forward_index(self):
        self.pp.pprint(self.forward_index)

    def print_inverted_index(self):
        self.pp.pprint(self.inverted_index)

    def search_inverted_index(self, query, count):
        if query not in self.inverted_index:
            print("Cannot find search term")
            return None
        index = self.inverted_index[query][1]
        return list(index.items())[:count]

    def lookup_url(self, loc):
        return self.book[loc]

    def write_to_json(self, name):
        with open(f"{name.upper()}_inverted_index.json", "w") as json_index:
            json.dump(self.inverted_index, json_index)
        print(f"\n{name.upper()} Written to JSON!")
