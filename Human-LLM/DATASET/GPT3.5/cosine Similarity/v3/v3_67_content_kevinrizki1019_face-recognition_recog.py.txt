from extract import extractFeatures
from vector import calcCosineSimilarity, calcEuclideanDistance
import os

def match_images(image_path, database, top_compare, use_cosine_similarity, controller):
    # Set the score name based on the similarity metric used
    controller.score_name = 'Similarity' if use_cosine_similarity else 'Distance'

    # Check if an image is selected
    if image_path is None:
        controller.msg.set('Please open a file first.')
        return

    # Hide buttons and display message while processing
    controller.hide_buttons()
    controller.msg.set('Extracting features from {}...'.format(os.path.basename(image_path)))
    controller.matched_images = []

    # Extract features from the input image
    input_features = extractFeatures(image_path)

    # Compare the input image with images in the database
    for compared_path in database:
        controller.msg.set('\rMatching: {}'.format(compared_path))
        compared_data = {'path': compared_path}
        compared_image = database[compared_path]

        # Calculate similarity or distance based on the chosen method
        if use_cosine_similarity:
            compared_data['x'] = calcCosineSimilarity(input_features, compared_image)
        else:
            compared_data['x'] = calcEuclideanDistance(input_features, compared_image)

        controller.matched_images.append(compared_data)

    # Sort the matched images based on similarity or distance
    controller.matched_images = sorted(controller.matched_images, key=lambda x: -x['x']) if use_cosine_similarity \
                                else sorted(controller.matched_images, key=lambda x: x['x'])

    # Keep only the top matching images
    if len(controller.matched_images) > top_compare:
        controller.matched_images = controller.matched_images[:top_compare]

    # Show buttons, draw matches, and display best match
    controller.show_buttons()
    controller.draw_matches()
    best_match = controller.matched_images[0]
    controller.msg.set('Best match: {} ({})'.format(best_match['path'], best_match['x']))
