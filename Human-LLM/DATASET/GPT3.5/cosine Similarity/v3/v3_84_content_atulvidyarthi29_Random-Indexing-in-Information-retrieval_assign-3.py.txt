import os
import numpy as np
from nltk.corpus import stopwords
from nltk.tokenize import RegexpTokenizer

class RandomIndexing:
    def __init__(self, folderpath):
        self.documents, self.terms = self._preprocess_documents(folderpath)
        self.index_vector = self._create_index_vector(self.terms)
        print("Index Vector Created.\n")
        self.final_context_vector = self._create_context_vector(self.documents, self.index_vector)
        print("Context Vectors calculated.\n")
        print("Calculating the similarity values of each term and sorting them.\n")     
        similar_terms = self._sort_similarity(self.final_context_vector)
        x = int(input("Enter x:"))
        print("Displaying Results.\n")
        for key in similar_terms.keys():
            print(key, similar_terms[key][:x])

    def _preprocess_documents(self, folderpath):
        documents = {}
        terms = []
        os.chdir(folderpath)
        tokenizer = RegexpTokenizer(r'\w+')
        stop_words = set(stopwords.words('english'))
        for filename in os.listdir():
            vector = []
            with open(os.path.join(folderpath, filename), 'r') as f:
                text = f.read().lower()
                words = tokenizer.tokenize(text)
                for word in words:
                    if word not in stop_words:
                        vector.append(word)
                        terms.append(word)
            documents[filename] = np.array(vector)
        return documents, np.array(list(set(terms)))

    def _create_index_vector(self, terms):
        index_vectors = {}
        for term in terms:
            seed = ''.join(str(ord(a)) for a in term)
            np.random.seed(int(seed))
            vector_length = 1024
            iv = []
            for i in range(vector_length):
                if np.random.uniform(1, 101) < 10:
                    iv.append(np.random.choice([1, -1]))
                else:
                    iv.append(0)
            index_vectors[term] = np.array(iv)
        return index_vectors

    def _create_context_vector(self, documents, index_vector):
        for doc in documents.values():
            for i in range(len(doc)):
                if i == 0:
                    index_vector[doc[i]] += (index_vector[doc[i + 1]] * 0.5) + (index_vector[doc[i + 2]] * 0.25)
                elif i == 1:
                    index_vector[doc[i]] += (index_vector[doc[i - 1]] * 0.5) + index_vector[doc[i]] + (index_vector[doc[i + 1]] * 0.5) + (index_vector[doc[i + 2]] * 0.25)      
                elif i == len(doc) - 2:
                    index_vector[doc[i]] += (index_vector[doc[i + 1]] * 0.5) + index_vector[doc[i]] + (index_vector[doc[i - 1]] * 0.5) + (index_vector[doc[i - 2]] * 0.25)      
                elif i == len(doc) - 1:
                    index_vector[doc[i]] += (index_vector[doc[i - 1]] * 0.5) + (index_vector[doc[i - 2]] * 0.25)
                else:
                    index_vector[doc[i]] += (index_vector[doc[i + 1]] * 0.5) + (index_vector[doc[i + 2]] * 0.25) + index_vector[doc[i]] + (index_vector[doc[i - 1]] * 0.5) + (index_vector[doc[i - 2]] * 0.25)
        return index_vector

    def _sort_similarity(self, context_vector):
        sorted_terms = {}
        for term in context_vector.keys():
            mag_a = np.linalg.norm(context_vector[term])
            for term2 in context_vector.keys():
                mag_b = np.linalg.norm(context_vector[term2])
                score = np.sum(context_vector[term] * context_vector[term2]) / (mag_a * mag_b)
                try:
                    sorted_terms[term] += [(term2, score)]
                except:
                    sorted_terms[term] = [(term2, score)]
        for key in sorted_terms.keys():
            sorted_terms[key].sort(reverse=True, key=lambda x: x[1])
        return sorted_terms
