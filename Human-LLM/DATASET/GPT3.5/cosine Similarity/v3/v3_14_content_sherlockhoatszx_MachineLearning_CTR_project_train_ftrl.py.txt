import random
import math

# Hyperparameters
ALPHA = 0.5
L2 = 10
L1 = 0.1
BETA = 0.01
ITERATIONS = 3

def initialize_arrays(max_index):
    """Initialize weights, z, and n arrays."""
    weights = [random.uniform(-0.5, 0.5) for _ in range(max_index + 1)]
    z_values = [0] * (max_index + 1)
    n_values = [0] * (max_index + 1)
    return weights, z_values, n_values

def update_weights(file, weights, z_values, n_values):
    """Update weights based on training data."""
    for _ in range(ITERATIONS):
        file.seek(0)  # Reset file pointer to start
        for line in file:
            segments = line.strip().split("\t")
            label = int(segments[0])
            s = sum(weights[int(segment.split(":")[0])] for segment in segments[1:])
            p = 1.0 / (1 + math.exp(-s))
            q = p - label
            for segment in segments[1:]:
                index = int(segment.split(":")[0])
                sigma = (math.sqrt(n_values[index] + q * q) - math.sqrt(n_values[index])) / ALPHA     
                z_values[index] += q - sigma * weights[index]
                n_values[index] += q * q
                if abs(z_values[index]) < L1:
                    weights[index] = 0
                else:
                    if z_values[index] > 0:
                        weights[index] = -(z_values[index] - L1) / ((BETA + math.sqrt(n_values[index])) / ALPHA + L2)
                    else:
                        weights[index] = -(z_values[index] + L1) / ((BETA + math.sqrt(n_values[index])) / ALPHA + L2)

def predict_and_write_results(input_file, output_file, weights, max_index):
    """Predict probabilities and write results to the output file."""
    for line_number, line in enumerate(input_file, 1):
        segments = line.strip().split("\t")
        label = int(segments[0])
        s = sum(weights[int(segment.split(":")[0])] for segment in segments[1:])
        p = 1.0 / (1 + math.exp(-s))
        output_line = f"{line_number},{segments[0]},{p}\n"
        output_file.write(output_line)

# Load train features and compute max_index
with open("train_feature", "r") as train_file:
    max_index = max(int(segment.split(":")[0]) for line in train_file for segment in line.strip().split("\t")[1:])

# Initialize arrays and update weights
weights, z_values, n_values = initialize_arrays(max_index)
with open("train_feature", "r") as train_file:
    update_weights(train_file, weights, z_values, n_values)

# Predict probabilities for validation data and write results
with open("validate_feature", "r") as validate_file, open("pctr", "w+") as output_file: 
    predict_and_write_results(validate_file, output_file, weights, max_index)

# Predict probabilities for test data and write results
with open("test_feature", "r") as test_file, open("test_pctr", "w+") as output_file2, open("test_pctr.csv", "w+") as output_file3:
    predict_and_write_results(test_file, output_file2, weights, max_index)
    predict_and_write_results(test_file, output_file3, weights, max_index)
