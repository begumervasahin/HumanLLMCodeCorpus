import csv
import copy

def calculate_similarity_matrix(input_data):
    # Calculate the number of rows and columns in the input data
    num_rows = len(input_data) - 1
    num_cols = len(input_data[0]) - 1

    # Deep copy the input data to avoid modifying the original
    data_copy = copy.deepcopy(input_data)

    # Calculate row averages and modify data_copy accordingly
    row_avgs = []
    for i in range(1, num_rows + 1):
        row_sum, count = 0, 0
        for j in range(1, num_cols + 1):
            value = float(data_copy[i][j])
            if value != 0.0:
                row_sum += value
                count += 1
        avg = row_sum / count if count != 0 else 0.0
        row_avgs.append(avg)
        for j in range(1, num_cols + 1):
            if float(data_copy[i][j]) != 0.0:
                data_copy[i][j] -= avg

    # Prepare data for similarity matrix calculation
    modified_data = [[float(data_copy[i][j]) for j in range(1, num_cols + 1)] for i in range(1, num_rows + 1)]

    # Initialize the similarity matrix with zeros
    similarity_matrix = [[0 for _ in range(num_rows)] for _ in range(num_rows)]

    # Calculate similarities between rows and populate the matrix
    for i in range(num_rows - 1):
        for j in range(i + 1, num_rows):
            num = sum(x * y for x, y in zip(modified_data[i], modified_data[j]))
            den1 = sum(x ** 2 for x in modified_data[i])
            den2 = sum(x ** 2 for x in modified_data[j])
            if den1 != 0 and den2 != 0:
                similarity_matrix[i][j] = similarity_matrix[j][i] = num / (den1 ** 0.5 * den2 ** 0.5)
            else:
                similarity_matrix[i][j] = similarity_matrix[j][i] = -5.0

    # Set diagonal elements of similarity matrix to 1.0
    for i in range(num_rows):
        similarity_matrix[i][i] = 1.0

    return similarity_matrix

def read_csv_data(filename):
    # Read data from CSV file and return as a list of lists
    with open(filename, newline='') as csvfile:
        data_reader = csv.reader(csvfile)
        data = list(data_reader)
    return data

if __name__ == '__main__':
    filename = 'sample.csv'
    input_data = read_csv_data(filename)
    similarity_matrix = calculate_similarity_matrix(input_data)
    print('\nFinal Similarity Matrix:\n')
    for row in similarity_matrix:
        print(row)
