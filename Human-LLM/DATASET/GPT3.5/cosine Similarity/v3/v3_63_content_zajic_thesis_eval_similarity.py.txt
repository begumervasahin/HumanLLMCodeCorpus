import numpy as np
from scipy.spatial.distance import cosine
import pickle

class CosineSimilarity:
    def __init__(self):
        self.tfidf, self.terms, self.sigma, self.VT = self._load_matrices()

    def calculate_term_similarity(self, term1, term2):
        if not all(term in self.terms for term in (term1, term2)):
            print("One or both terms not found in the corpus.")
            return 0

        index1 = self.terms.index(term1)
        index2 = self.terms.index(term2)

        term_vector1 = np.transpose(self.tfidf[index1])
        term_vector2 = np.transpose(self.tfidf[index2])

        cosine_dist = self._calculate_cosine_similarity(term_vector1, term_vector2)
        similarity = 1 - cosine_dist

        print(f"Similarity between '{term1}' and '{term2}': {similarity}")
        return similarity

    def _create_diagonal_matrix(self):
        diag_matrix = np.zeros((len(self.sigma), len(self.sigma)), int)
        np.fill_diagonal(diag_matrix, 1)
        diag_matrix = diag_matrix * self.sigma
        return diag_matrix

    def _calculate_cosine_similarity(self, term_vector1, term_vector2):
        sigma_diagonal = self._create_diagonal_matrix()
        sigma_inv = np.linalg.inv(sigma_diagonal)
        transformed_vector1 = np.matmul(np.matmul(sigma_inv, self.VT), term_vector1)
        transformed_vector2 = np.matmul(np.matmul(sigma_inv, self.VT), term_vector2)
        cosine_dist = cosine(transformed_vector1, transformed_vector2)
        return cosine_dist

    @staticmethod
    def _load_matrices():
        with open('./pickle_files/sigma', 'rb') as fp:
            sigma = pickle.load(fp)
        with open('./pickle_files/vt', 'rb') as fp:
            VT = pickle.load(fp)
        with open('./pickle_files/tfidf', 'rb') as fp:
            tfidf = pickle.load(fp)
        with open('./pickle_files/terms', 'rb') as fp:
            terms = pickle.load(fp)
        return tfidf, terms, sigma, VT

if __name__ == "__main__":
    cosine_similarity = CosineSimilarity()
    term1 = input("Enter the first term: ")
    term2 = input("Enter the second term: ")
    cosine_similarity.calculate_term_similarity(term1, term2)
