import requests
from bs4 import BeautifulSoup
import urllib.request
import queue
import settings
import preprocess

# Global variables
visited_links = set()
valid_links = set()

def plot_web_graph(initial_link, threshold):
    # Initialize the queue and add the initial link
    link_queue = queue.Queue()
    link_queue.put(initial_link)
    visited_links.add(initial_link)
    
    # Write the initial link to the file
    with open("./data/all_url", 'a', encoding="utf-8") as file:
        file.write(initial_link + "\n")
    
    # Start fetching the link graph
    fetch_link_graph(link_queue, threshold)

def fetch_link_graph(link_queue, threshold):
    while not link_queue.empty():
        current_link = link_queue.get()
        
        # Fetch the page content
        page = requests.get(current_link)
        soup = BeautifulSoup(page.text, 'html.parser')
        
        # Extract links from the page
        link_list = extract_links(current_link, soup)
        
        # Add new valid links to the queue and set
        for link in link_list:
            if link not in visited_links:
                visited_links.add(link)
                link_queue.put(link)
                valid_links.add(link)
                
        # Check if the threshold is reached
        if len(valid_links) >= threshold:
            break

def web_spider(initial_link, threshold):
    # Initialize the queue and add the initial link
    link_queue = queue.Queue()
    link_queue.put(initial_link)
    visited_links.add(initial_link)
    
    # Start scraping
    scrape(link_queue, threshold)

def scrape(link_queue, threshold):
    while not link_queue.empty() and len(valid_links) < threshold:
        current_link = link_queue.get()
        
        # Fetch the page content
        page = requests.get(current_link)
        soup = BeautifulSoup(page.text, 'html.parser')
        
        # Process visible text and extract links
        visible_text = extract_visible_text(soup)
        processed_text = preprocess_text(visible_text)
        link_list = extract_links(current_link, soup)
        
        # Add new valid links to the queue and set
        for link in link_list:
            if link not in visited_links:
                visited_links.add(link)
                link_queue.put(link)
                valid_links.add(link)
        
        # Export processed data
        preprocess.export_data(current_link, processed_text)

def extract_visible_text(soup):
    # Extract visible text from the page
    texts = soup.findAll(text=True)
    visible_texts = filter(tag_visible, texts)
    return " ".join(t.strip() for t in visible_texts)

def preprocess_text(text):
    # Preprocess the visible text
    processed_text = preprocess.remove_punctuation(text)
    processed_text = preprocess.remove_digits(processed_text)
    processed_text = preprocess.tokenize_data(processed_text)
    processed_text = preprocess.remove_stop_words(processed_text)
    processed_text = preprocess.perform_stemming(processed_text)
    processed_text = preprocess.remove_stop_words(processed_text)
    return processed_text

def extract_links(current_link, soup):
    # Extract valid links from the page
    link_list = []
    for link_tag in soup.find_all('a', href=True):
        url = link_tag['href']
        if is_valid_link(url):
            link_list.append(get_full_url(current_link, url))
    return link_list

def get_full_url(base_url, relative_url):
    # Construct full URL from base and relative URLs
    if relative_url.startswith(('http://', 'https://')):
        return relative_url
    if relative_url.startswith('/'):
        return base_url + relative_url[1:]
    return base_url + '/' + relative_url

def is_valid_link(url):
    # Check if the URL is valid and not a non-HTML file
    return url and not url.endswith(('.pdf', '.doc', '.docx', '.xls', '.xlsx', '.jpg', '.png'))

def tag_visible(element):
    # Filter out non-visible elements
    if element.parent.name in ['style', 'script', 'head', 'title', 'meta', '[document]']:
        return False
    return not isinstance(element, Comment)
