from math import sqrt

def cosine_similarity(vector1, vector2):
    # Initialize variables to store sums
    numerator_sum = 0
    v_vector_sum = 0
    w_vector_sum = 0

    # Compute numerator, v vector sum, and w vector sum
    for index, value in enumerate(vector1):
        numerator_sum += value * vector2[index]
        v_vector_sum += value ** 2
        w_vector_sum += vector2[index] ** 2

    # Compute denominator
    denominator_total = sqrt(v_vector_sum) * sqrt(w_vector_sum)

    # Compute cosine similarity
    similarity = numerator_sum / denominator_total
    return similarity

def jaccard_similarity(vector1, vector2):
    # Initialize variables to store sums
    numerator_min_sum = 0
    denominator_max_sum = 0

    # Compute numerator and denominator
    for index, value in enumerate(vector1):
        numerator_min_sum += min(value, vector2[index])
        denominator_max_sum += max(value, vector2[index])

    # Compute Jaccard similarity
    similarity = numerator_min_sum / denominator_max_sum
    return similarity

def dice_similarity(vector1, vector2):
    # Initialize variables to store sums
    numerator_min_sum = 0
    denominator_sum = 0

    # Compute numerator and denominator
    for index, value in enumerate(vector1):
        numerator_min_sum += min(value, vector2[index])
        denominator_sum += value + vector2[index]

    # Compute Dice similarity
    similarity = (2 * numerator_min_sum) / denominator_sum
    return similarity
