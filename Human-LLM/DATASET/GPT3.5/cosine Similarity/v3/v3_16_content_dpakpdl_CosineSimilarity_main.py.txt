import csv
import os
import re
from collections import Counter
import nltk
from nltk.tokenize import sent_tokenize
from scipy import spatial
import pandas as pd

nltk.download('punkt')

ATTACHMENT_PATH = "Attachments"


def read_csv_file(filepath, usecols):
    # Read a CSV file and return the dataset using pandas
    return pd.read_csv(filepath, usecols=usecols)


def get_matched_summaries():
    # Read summaries, non-summaries, and mapping CSV files
    summaries = read_csv_file(os.path.join(ATTACHMENT_PATH, "Summaries.csv"),
                              usecols=["adsh", "full_summary", "full_summary_len", "expense_summary", "expense_len"])
    total_summaries = summaries["adsh"].tolist()

    non_summaries_path = os.path.join(ATTACHMENT_PATH, "Non-Summaries")
    non_summaries_file_list = [os.path.splitext(os.path.basename(file_path))[0] for file_path in
                               os.listdir(non_summaries_path)]

    mapping = read_csv_file(os.path.join(ATTACHMENT_PATH, "Mapping.csv"),
                            usecols=["accession_xbrl", "accession_not_xbrl"])

    # Filter mapping to get matched summaries
    new_mapping = mapping[(mapping['accession_not_xbrl'].isin(non_summaries_file_list)) &
                          (mapping['accession_xbrl'].isin(total_summaries))]
    return new_mapping, mapping, summaries


def get_lines_from_paragraph(paragraph):
    # Tokenize paragraph into lines
    words_output = []
    lines = []
    splitted_sentences = sent_tokenize(paragraph)
    for line in splitted_sentences:
        words_output.append(remove_articles_preposition(line.strip()))
        lines.append(line.strip())
    return words_output, lines


def remove_articles_preposition(sentence):
    # Tokenize, remove articles/prepositions, and stem words
    tokens = nltk.word_tokenize(sentence)
    tokens = [re.sub(r'[^A-Za-z0-9]+', '', token) for token in tokens if token.lower() not in ['a', 'an', 'the']]
    tagged = nltk.pos_tag(tokens)
    stemmer = nltk.stem.SnowballStemmer('english')
    return [stemmer.stem(word) for word, tag in tagged if tag != 'IN']


def read_file(filepath):
    # Read file content and tokenize sentences
    with open(filepath, "r", encoding="utf8", errors='ignore') as input_file:
        content = input_file.read()
    sentences = sent_tokenize(content)
    return [re.sub('\s+', ' ', sentence) for sentence in sentences if sentence[-1] in ['.', '?']]


def calculate_word_vector_count(summary_word_vector, non_summary_wordvector):
    # Calculate cosine similarity between word vectors
    total_score = []
    for line in summary_word_vector:
        cosine = [get_cosine_similarity(get_word_count(line, set(line2)), get_word_count(line2, set(line)))
                  for line2 in non_summary_wordvector]
        total_score.append(max(cosine))
    return get_average_similarity(total_score), total_score


def get_average_similarity(total):
    # Calculate average similarity
    return sum(total) / len(total)


def get_cosine_similarity(summary_count_vector, non_summary_count_vector):
    # Calculate cosine similarity
    cosine_value = 1 - spatial.distance.cosine(summary_count_vector, non_summary_count_vector)
    return int(cosine_value >= 0.90)


def get_word_count(to_count_list, word_list):
    # Count word occurrences in a list
    counts = Counter(to_count_list)
    counts.update({word: 0 for word in word_list if word not in counts})
    return [counts[key] for key in word_list]


def add_rows_to_output_csv(output_rows):
    # Add processed rows to output CSV
    with open(os.path.join(ATTACHMENT_PATH, "Summaries.csv"), 'r') as csvinput:
        with open('Output/summaries_output.csv', 'w') as csvoutput:
            writer = csv.writer(csvoutput, lineterminator='\n')
            reader = csv.reader(csvinput)
            header = next(reader) + ['number_of_sentences_in_summary', 'number_of_sentences_in_non_summary', 'similarity']
            writer.writerow(header)
            for row in reader:
                for output_row in output_rows:
                    if row[0] == output_row["summary_indentifier"]:
                        row += [output_row["number_of_sentences_in_summary"],
                                output_row["number_of_sentences_in_non_summary"],
                                output_row["similarity"]]
                        writer.writerow(row)


def make_raw_files(filename, non_summary_identity, line_wise_similarity, summary_lines):
    # Create raw CSV files
    header = ["adsh", "non_summary_file", "sentence", "similarity"]
    with open(os.path.join('Raw', f'{filename}.csv'), 'w', newline='') as myfile:
        wr = csv.writer(myfile)
        wr.writerow(header)
        wr.writerows([[filename, non_summary_identity, line, sim] for line, sim in zip(summary_lines, line_wise_similarity)])


def main():
    output_rows = []
    new_mapping, mapping, summaries = get_matched_summaries()
    for index, row in new_mapping.iterrows():
        summary_file_identity, non_summary_file_identity = row['accession_xbrl'], row['accession_not_xbrl']
        full_summary = summaries.loc[summaries["adsh"] == summary_file_identity, 'full_summary'].iloc[0]
        summary_word_vector, summary_lines = get_lines_from_paragraph(full_summary)
        non_summary_file_path = os.path.join(ATTACHMENT_PATH, "Non-Summaries", f"{non_summary_file_identity}.txt")
        non_summary_lines = read_file(non_summary_file_path)
        non_summary_word_vector = [remove_articles_preposition(line) for line in non_summary_lines]
        total_similarity, line_wise_similarity = calculate_word_vector_count(summary_word_vector, non_summary_word_vector)
        make_raw_files(summary_file_identity, non_summary_file_identity, line_wise_similarity, summary_lines)
        output_rows.append({
            "summary_indentifier": summary_file_identity,
            "similarity": total_similarity,
            "number_of_sentences_in_summary": len(summary_word_vector),
            "number_of_sentences_in_non_summary": len(non_summary_word_vector)
        })
    add_rows_to_output_csv(output_rows)


if __name__ == "__main__":
    main()
