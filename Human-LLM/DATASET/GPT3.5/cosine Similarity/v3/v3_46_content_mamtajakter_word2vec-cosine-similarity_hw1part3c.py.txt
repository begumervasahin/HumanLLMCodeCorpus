import gensim
from nltk.corpus import brown

# Load the Word2Vec model trained on the Brown corpus
model = gensim.models.Word2Vec(brown.sents())

# Define the words to find similarities for
word1 = "rebellion"
word2 = "slave"

# Find top 10 similar words for each word
similar_words1 = [word for word, _ in model.most_similar(positive=[word1], topn=10)]
similar_words2 = [word for word, _ in model.most_similar(positive=[word2], topn=10)]

# Calculate cosine similarities for word1 and similar words
cosine_similarities1 = [
    f"{word1} - {word}: {model.similarity(word1, word)}" for word in similar_words1
]

# Calculate cosine similarities for word2 and similar words
cosine_similarities2 = [
    f"{word2} - {word}: {model.similarity(word2, word)}" for word in similar_words2
]

# Print results
print("Top ten Similar words: \n")
print(f"{word1} = {similar_words1}")
print(f"{word2} = {similar_words2}")
print("Cosine Similarities: \n")
print("\n".join(cosine_similarities1))
print("\n".join(cosine_similarities2))
