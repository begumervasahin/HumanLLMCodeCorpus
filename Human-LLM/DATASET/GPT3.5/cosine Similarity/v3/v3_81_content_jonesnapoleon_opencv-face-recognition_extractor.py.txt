import cv2
import numpy as np
import os
import pickle

def extract_features(image_path, vector_size=32):
    """
    Extracts features from an image using the KAZE feature detector and descriptor.

    Args:
        image_path (str): Path to the image file.
        vector_size (int): Size of the feature vector.

    Returns:
        np.ndarray: Feature descriptor vector.
    """
    try:
        # Read the image
        image = cv2.imread(image_path)

        # Convert the image to grayscale
        gray_image = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)

        # Create a KAZE feature detector
        alg = cv2.KAZE_create()

        # Detect keypoints in the image
        kps = alg.detect(gray_image)

        # Select top keypoints based on response
        kps = sorted(kps, key=lambda x: -x.response)[:vector_size]

        # Compute descriptors for selected keypoints
        kps, dsc = alg.compute(gray_image, kps)

        # Flatten and normalize the descriptors
        dsc = dsc.flatten()
        needed_size = vector_size * 64
        if dsc.size < needed_size:
            dsc = np.concatenate([dsc, np.zeros(needed_size - dsc.size)])
    except cv2.error as e:
        print('Error:', e)
        return None

    return dsc

def batch_extractor(images_path, pickled_db_path="features.pkl"):
    """
    Extracts features from a batch of images and saves them to a pickled file.

    Args:
        images_path (str): Path to the directory containing images.
        pickled_db_path (str): Path to save the pickled feature database.
    """
    # Get the list of image files in the directory
    files = [os.path.join(images_path, p) for p in sorted(os.listdir(images_path))]

    # Dictionary to store extracted features
    result = {}

    # Extract features from each image
    for f in files:
        print('Extracting features from image:', f)
        name = os.path.basename(f).lower()
        result[name] = extract_features(f)

    # Save the extracted features to a pickled file
    with open(pickled_db_path, 'wb') as fp:
        pickle.dump(result, fp)

if __name__ == '__main__':
    # Provide the path to the directory containing images
    images_path = 'path/to/images'

    # Extract features from images and save to a pickled file
    batch_extractor(images_path)
