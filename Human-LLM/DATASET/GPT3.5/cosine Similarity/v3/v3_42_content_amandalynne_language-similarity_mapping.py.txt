class Mapping:
    def __init__(self, items):
        self.item_list = items
        self.index_by_item = {item: index for index, item in enumerate(items)}
        self.item_by_index = {index: item for index, item in enumerate(items)}

    def __contains__(self, item):
        return item in self.index_by_item

    def __getitem__(self, item):
        return self.index_by_item[item]

    def __iter__(self):
        return iter(self.index_by_item)

    def __len__(self):
        return len(self.index_by_item)

    def __repr__(self):
        return f"<{self.__class__.__name__} with {len(self)} entries>"

    def add(self, item):
        if item not in self:
            index = len(self)
            self.item_list.append(item)
            self.index_by_item[item] = index
            self.item_by_index[index] = item
        return item

    def get_index(self, item, default=None):
        return self.index_by_item.get(item, default)

    def get_item(self, index):
        return self.item_by_index.get(index)

    def get_keys(self):
        """Preserving original order"""
        return self.item_list

# Example usage
items = ['apple', 'banana', 'orange']
mapping = Mapping(items)

print(mapping)                 # Output: <Mapping with 3 entries>
print('apple' in mapping)      # Output: True
print(mapping['banana'])       # Output: 1
print(len(mapping))            # Output: 3

mapping.add('grape')
print(mapping)                 # Output: <Mapping with 4 entries>

print(mapping.get_index('orange'))  # Output: 2

print(mapping.get_item(3))          # Output: grape

print(mapping.get_keys())           # Output: ['apple', 'banana', 'orange', 'grape']
