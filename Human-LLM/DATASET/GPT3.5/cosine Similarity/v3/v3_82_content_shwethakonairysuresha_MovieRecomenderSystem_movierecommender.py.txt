import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load user-item rating data and movie titles data
user_item_ratings = pd.read_csv('file.tsv', sep='\t', names=['user_id', 'item_id', 'rating', 'timestamp'])
movie_titles = pd.read_csv('Movie_Id_Titles.csv')

# Merge the two datasets on 'item_id'
merged_data = pd.merge(user_item_ratings, movie_titles, on='item_id')

# Compute mean ratings and count of ratings for each movie
movie_ratings = merged_data.groupby('title')['rating'].agg(['mean', 'count'])
movie_ratings.columns = ['mean_rating', 'num_ratings']

# Visualize rating count distribution
plt.figure(figsize=(10, 4))
movie_ratings['num_ratings'].hist(bins=70)
plt.xlabel('Number of Ratings')
plt.ylabel('Frequency')
plt.title('Distribution of Ratings Count')
plt.show()

# Visualize rating distribution
plt.figure(figsize=(10, 4))
movie_ratings['mean_rating'].hist(bins=70)
plt.xlabel('Rating')
plt.ylabel('Frequency')
plt.title('Distribution of Ratings')
plt.show()

# Create a user-movie matrix with user_id as index and movie titles as columns
user_movie_matrix = merged_data.pivot_table(index='user_id', columns='title', values='rating')

# Compute correlation with Star Wars (1977) and Liar Liar (1997)
star_wars_ratings = user_movie_matrix['Star Wars (1977)']
liar_liar_ratings = user_movie_matrix['Liar Liar (1997)']
corr_with_star_wars = user_movie_matrix.corrwith(star_wars_ratings)
corr_with_liar_liar = user_movie_matrix.corrwith(liar_liar_ratings)

# Create correlation dataframes and filter movies with more than 100 ratings
corr_star_wars_df = pd.DataFrame({'Correlation': corr_with_star_wars, 'num_ratings': movie_ratings['num_ratings']})
corr_star_wars_filtered = corr_star_wars_df[corr_star_wars_df['num_ratings'] > 100].sort_values('Correlation', ascending=False)

corr_liar_liar_df = pd.DataFrame({'Correlation': corr_with_liar_liar, 'num_ratings': movie_ratings['num_ratings']})
corr_liar_liar_filtered = corr_liar_liar_df[corr_liar_liar_df['num_ratings'] > 100].sort_values('Correlation', ascending=False)

# Print the recommended movies
print('Movies recommended that are similar to Star Wars (1977):')
print(corr_star_wars_filtered.head())
print()
print('Movies recommended that are similar to Liar Liar (1997):')
print(corr_liar_liar_filtered.head())
