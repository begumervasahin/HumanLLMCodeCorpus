class Mode:

def build_users_pairs(ratings_by_movie):
    users_ratings = ratings_by_movie[1]
    output_values = []
    num_users = len(users_ratings)
    for i in range(num_users - 1):
        for j in range(i + 1, num_users):
            user_x, rating_x = users_ratings[i]
            user_y, rating_y = users_ratings[j]
            user_pair = (user_x, user_y) if user_x <= user_y else (user_y, user_x)
            output_values.append((user_pair, (rating_x, rating_y)))
    return output_values

def calculate_cosine_similarity(ratings):
    x, y = ratings
    numerator, norm_x, norm_y = 0, 0, 0
    num_ratings = len(x)
    if num_ratings < 2:
        return iter(())
    for i in range(num_ratings):
        rating_x, rating_y = int(x[i]), int(y[i])
        numerator += rating_x * rating_y
        norm_x += rating_x ** 2
        norm_y += rating_y ** 2
    cosine_sim = numerator / (math.sqrt(norm_x) * math.sqrt(norm_y)) if norm_x + norm_y > 0 else 0
    return (x, (y, cosine_sim)), (y, (x, cosine_sim))

def build_rdd(files_list):
    rdd = sc.emptyRDD()
    for rating_file in files_list:
        print('Adding {}'.format(rating_file))
        user_ratings = []
        if mode == Mode.GFS:
            file_content = os.popen("gsutil cat " + rating_file).read()
            lines_iterator = (line for line in file_content.split('\n') if line)
            user_ratings = [(pair[0], pair[1]) for pair in (line.rstrip().split(',') for line in lines_iterator)]
        elif mode == Mode.LOCAL:
            with open(rating_file, 'r') as rating_f:
                user_ratings = [(pair[0], pair[1]) for pair in (line.rstrip().split(',') for line in rating_f)]
        movie_number = ''.join(filter(str.isdigit, rating_file))
        rdd = rdd.union(sc.parallelize([[movie_number, user_ratings]]))
    return rdd
