import json
from collections import defaultdict

def read_counts(file_path):
    """
    Read counts from a file and return as a list of tuples.
    Each tuple contains word, count, and context counts.
    """
    with open(file_path, 'r') as file:
        return [(word, int(count), json.loads(context)) for word, count, context in (line.split("\t") for line in file)]

def all_words(words1, words2):
    """
    Yield all words from two lists of words until the minimum length is reached.
    """
    for i in range(min(len(words1), len(words2))):
        yield words1[i][0]
        yield words2[i][0]

# Read counts from files
twitter_counts = read_counts("twcounts")
nyt_counts = read_counts("nytcounts")

# Sort words by count
tw_words = sorted([(word, count) for word, count, _ in twitter_counts if count > 100], key=lambda x: -x[1])
nyt_words = sorted([(word, count) for word, count, _ in nyt_counts if count > 100], key=lambda x: -x[1])

# Create dictionaries for counts
twitter_word_counts = {word: count for word, count, _ in twitter_counts}
nyt_word_counts = {word: count for word, count, _ in nyt_counts}

# Extract a vocabulary of 4000 words
vocab = set()
for word in all_words(tw_words, nyt_words):
    vocab.add(word)
    if len(vocab) == 4000:
        break

# Write the vocabulary to a file
with open("vocab", 'w') as f:
    vocab_counts = [(word, nyt_word_counts.get(word, 0)) for word in vocab]
    vocab_counts.sort(key=lambda x: -x[1])
    vocab_counts = [(word, count) for word, count in vocab_counts if count > 0]
    for word, count in vocab_counts:
        print(word, count, file=f)

def filter_counts(infile, outfile, vocab):
    """
    Filter counts in a file based on a given vocabulary and write to another file.
    """
    print("%s -> %s" % (infile, outfile))
    ctx_numwords = defaultdict(int)
    word_contexts = {}
    word_counts = {}

    with open(infile, 'r') as inf:
        for line in inf:
            word, count, context = line.split("\t")
            if word not in vocab:
                continue
            context = json.loads(context)
            for ctx in context:
                ctx_numwords[ctx] += 1
            word_contexts[word] = context
            word_counts[word] = count

    with open(outfile, 'w') as out:
        for word, context in word_contexts.items():
            for ctx in list(context.keys()):
                if ctx_numwords[ctx] <= 1:
                    del context[ctx]
            print("%s\t%s\t%s" % (word, word_counts[word], json.dumps(context)), file=out)        

# Filter counts for trimmed files and files with specific words
filter_counts("twcounts", "twcounts.trimmed", vocab)
filter_counts("nytcounts", "nytcounts.trimmed", vocab)
filter_counts("twcounts", "twcounts.university_cat_dog", {"university", "cat", "dog"})  
filter_counts("nytcounts", "nytcounts.university_cat_dog", {"university", "cat", "dog"})
