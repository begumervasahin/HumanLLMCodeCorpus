import re
import math
import nltk
from nltk.stem import PorterStemmer
from nltk.corpus import stopwords
import string

# Initialize NLTK components
nltk.download('stopwords')
nltk.download('punkt')

# Initialize Porter Stemmer
ps = PorterStemmer()

# Load stopwords and query collection
stop_words = set(stopwords.words('english'))
query_collection_name = "cran.qry"
abstract_name = "cran.all.1400"

def preprocess_text(text):
    tokens = nltk.word_tokenize(text)
    tokens = [word for word in tokens if word.lower() not in stop_words and word not in string.punctuation]
    return tokens

def compute_idf(text_list):
    idf_collection = {}
    for text in text_list:
        for word in set(text):
            idf_collection[word] = idf_collection.get(word, 0) + 1
    for word in idf_collection:
        idf_collection[word] = math.log((len(text_list) / idf_collection[word]) + 1)
    return idf_collection

def create_tfidf_vectors(text_list, idf_collection):
    tfidf_vectors = []
    for text in text_list:
        tfidf_vector = {}
        for word in text:
            tfidf_vector[word] = math.log(text.count(word) + 1) * idf_collection[word]
        tfidf_vectors.append(tfidf_vector)
    return tfidf_vectors

def calculate_cosine_similarity(query_vector, document_vector):
    cosine_num = sum(query_vector[word] * document_vector[word] for word in query_vector)
    query_square = sum(query_vector[word] ** 2 for word in query_vector)
    document_square = sum(document_vector[word] ** 2 for word in document_vector)
    cosine_den = math.sqrt(query_square * document_square)
    if cosine_den == 0:
        return 0
    else:
        return cosine_num / cosine_den

# Read and preprocess queries
with open(query_collection_name, 'r') as query_file:
    query_text = query_file.read().replace("\n", " ")
    queries = [phase for phase in query_text.split(".I") if phase.strip() != ""]

queries = [preprocess_text(query) for query in queries]

# Compute IDF for query terms
IDF_query_collection = compute_idf(queries)

# Create TF-IDF vectors for queries
vector_list_queries = create_tfidf_vectors(queries, IDF_query_collection)

# Read and preprocess abstracts
with open(abstract_name, 'r') as abstract_file:
    abstract_text = abstract_file.read().replace("\n", " ")
    abstracts = [document for document in abstract_text.split(".I") if document.strip() != ""]

abstracts = [re.split(r'\s+\.W\s+', document)[1] for document in abstracts]
abstracts = [preprocess_text(document) for document in abstracts]

# Compute IDF for abstract terms
IDF_abstract_collection = compute_idf(abstracts)

# Create TF-IDF vectors for documents
vector_list_documents = create_tfidf_vectors(abstracts, IDF_abstract_collection)

# Open output file to write rankings
output_file = open("output.txt", 'w')

# Calculate cosine similarity and rank documents for each query
for i, query_vector in enumerate(vector_list_queries):
    ranking = []
    for j, document_vector in enumerate(vector_list_documents):
        for word in query_vector:
            if word not in document_vector:
                document_vector[word] = 0
        cosine_similarity = calculate_cosine_similarity(query_vector, document_vector)
        ranking.append((cosine_similarity, j))
    ranking.sort(key=lambda tup: tup[0], reverse=True)
    for rank, (cosine_similarity, document_index) in enumerate(ranking):
        output_file.write(f"{i + 1} {document_index + 1} {cosine_similarity}\n")

output_file.close()
