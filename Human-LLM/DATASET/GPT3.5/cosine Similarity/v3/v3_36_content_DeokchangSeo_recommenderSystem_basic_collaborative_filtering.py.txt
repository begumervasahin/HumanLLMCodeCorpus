from surprise import similarities as sims
from surprise import Dataset, Reader
from numpy import zeros, sqrt
import heapq

def load_dataset(dataset_path, line_format='user item rating', rating_scale=(1, 7), sep=None):
    """
    Load the dataset from a file using Surprise's Reader.
    """
    reader = Reader(line_format=line_format, sep=sep, rating_scale=rating_scale)
    dataset = Dataset.load_from_file(dataset_path, reader)
    return dataset

def create_full_trainset(dataset_path, line_format=None, rating_scale=(1, 7), sep=None):
    """
    Create a full training set from the dataset.
    """
    dataset = load_dataset(dataset_path, line_format, rating_scale, sep)
    trainset = dataset.build_full_trainset()
    return trainset

def mean_centered_ratings(rating_matrix):
    """
    Compute mean-centered ratings for each user or item.
    """
    mean_centered = {}
    for entity, entity_ratings in rating_matrix.items():
        ratings = [rating for _, rating in entity_ratings]
        mean_rating = sum(ratings) / len(entity_ratings)
        centered_ratings = [(item_id, rating - mean_rating) for item_id, rating in entity_ratings]
        mean_centered[entity] = centered_ratings
    return mean_centered

def adjusted_cosine_similarity(num_rows, rating_matrix, min_support):
    """
    Compute adjusted cosine similarity between all pairs of users or items.
    """
    mean_centered = mean_centered_ratings(rating_matrix)
    prods = zeros((num_rows, num_rows), dtype=float)
    freq = zeros((num_rows, num_rows), dtype=float)
    sqi = zeros((num_rows, num_rows), dtype=float)
    sqj = zeros((num_rows, num_rows), dtype=float)
    sim = zeros((num_rows, num_rows), dtype=float)

    for entity, entity_ratings in mean_centered.items():
        for item_i, rating_i in entity_ratings:
            for item_j, rating_j in entity_ratings:
                freq[item_i, item_j] += 1
                prods[item_i, item_j] += rating_i * rating_j
                sqi[item_i, item_j] += rating_i ** 2
                sqj[item_i, item_j] += rating_j ** 2

    for i in range(num_rows):
        sim[i, i] = 1
        for j in range(i + 1, num_rows):
            if freq[i, j] < min_support:
                sim[i, j] = 0
            else:
                denum = sqrt(sqi[i, j]) * sqrt(sqj[i, j])
                sim[i, j] = prods[i, j] / denum
            sim[j, i] = sim[i, j]

    return sim

def predict_rating(similarity_matrix, target_id, related_ratings, neighbor_size):
    """
    Predict rating based on collaborative filtering.
    """
    target_similarities = [similarity_matrix[target_id][item_id] for item_id, _ in related_ratings]
    top_similarities = heapq.nlargest(neighbor_size, target_similarities)

    neighbors = {}
    for i, similarity in enumerate(similarity_matrix[target_id]):
        if similarity in top_similarities and i != target_id:
            neighbors[i] = similarity

    selected_ratings = {}
    for neighbor_id in neighbors.keys():
        for item_id, rating in related_ratings:
            if neighbor_id == item_id:
                selected_ratings[neighbor_id] = rating

    if len(selected_ratings) < neighbor_size:
        print('Not enough neighbors')
        return 0

    prediction = sum(selected_ratings[neighbor_id] * neighbors[neighbor_id] for neighbor_id in neighbors)
    return prediction / sum(neighbors.values())

def collaborative_filtering_prediction(trainset, item_id, user_id, neighbor_size, user_item, sim_method):
    """
    Perform collaborative filtering prediction.
    """
    item_idx = item_id - 1
    user_idx = user_id - 1
    min_support = 0

    num_users = trainset.n_users
    num_items = trainset.n_items
    user_ratings = trainset.ur
    item_ratings = trainset.ir

    if user_item == "user":
        target_num, target_idx, related_idx, ratings, related_ratings = num_users, user_idx, item_idx, user_ratings, item_ratings
    else:
        target_num, target_idx, related_idx, ratings, related_ratings = num_items, item_idx, user_idx, item_ratings, user_ratings

    sim_args = [target_num, related_ratings, min_support]
    if sim_method == "cosine":
        similarity_matrix = sims.cosine(*sim_args)
    elif sim_method == "pearson":
        similarity_matrix = sims.pearson(*sim_args)
    else:
        similarity_matrix = adjusted_cosine_similarity(*sim_args)

    print('\nSimilarity:\n', similarity_matrix)
    return predict_rating(similarity_matrix, target_idx, related_ratings[related_idx], neighbor_size)

# Example usage
file_path = 'ratings_for_question1.txt'
trainset = create_full_trainset(file_path, line_format='user item rating', rating_scale=(1, 7), sep='\t')

item_id = 1
user_id = 3
neighbor_size = 2
user_item = 'user'
sim_method = 'pearson'

estimate = collaborative_filtering_prediction(trainset, item_id, user_id, neighbor_size, user_item, sim_method)
print(f'\nUsing {user_item} based CF, the predicted rating to item {item_id} for user {user_id} is: {estimate:.2f}\n')
