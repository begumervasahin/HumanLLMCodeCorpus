def process_prospect(db, prospect, prosp_szn_results, prosp_adv_results, comps_per40_results, comps_adv_results):
    players = {}
    adv_players = {}
    unique_per40 = {}
    unique_adv = {}

    out_file = determine_output_file(prospect, prosp_szn_results)
    write_per40_stats(out_file, prospect)

    per40_comparables = compare_per40_stats(prospect, comps_per40_results, players)
    adv_comparables = compare_adv_stats(prospect, prosp_adv_results, comps_adv_results, adv_players)

    out_file.close()

def compare_per40_stats(prospect, comps_per40_results, players):
    per40_comparables = {}
    for nba_player in comps_per40_results:
        comp_key = f"{nba_player['Player']}, {nba_player['Season']}, {nba_player['School']}"
        players[comp_key] = nba_player
        per40_comparables[comp_key] = (
            shot_percentage_similarity(prospect, nba_player) * per_game_similarity(prospect, nba_player)
        )
    return sorted(per40_comparables.items(), key=lambda x: x[1], reverse=True)[:30]

def compare_adv_stats(prospect, prosp_adv_results, comps_adv_results, adv_players):
    college_player = get_advanced_stats(prospect, prosp_adv_results)
    adv_comparables = {}
    for nba_player in comps_adv_results:
        comp_key = f"{nba_player['Player']}, {nba_player['Season']}, {nba_player['School']}"
        adv_players[comp_key] = nba_player
        adv_comparables[comp_key] = (
            adv_rate_similarity(college_player, nba_player)
            * adv_percent_similarity(college_player, nba_player)
            * adv_misc_similarity(college_player, nba_player)
        )
    return sorted(adv_comparables.items(), key=lambda x: x[1], reverse=True)[:30]
