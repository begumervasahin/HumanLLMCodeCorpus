import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# Constants
CONSOLES = {'A': 'Wii', 'B': 'PS3'}
GENRES = ['Sports', 'Platform', 'Racing', 'Role-Playing', 'Puzzle', 'Misc', 'Shooter',
          'Simulation', 'Action', 'Fighting', 'Adventure', 'Strategy']

# Read data
df = pd.read_csv('vgsales.csv')

# Introduction
print('This application analyzes the similarity among markets of different video game consoles')
print('Game genres in the data:', GENRES)

# Filter data for consoles A and B
data_A = df.loc[df['Platform'] == CONSOLES['A']]
data_B = df.loc[df['Platform'] == CONSOLES['B']]

# Print top 10 best-selling games for consoles A and B
def print_top_games(data, console):
    print(f'Top 10 best-selling games on {console}:')
    print(data.head(10), '\n')

print_top_games(data_A, CONSOLES['A'])
print_top_games(data_B, CONSOLES['B'])

# Function to build sales data by genre for a given console
def build_sales_data(dataset, console):
    sales_data = np.zeros(len(GENRES))
    for idx, genre in enumerate(GENRES):
        sales_total = dataset.loc[dataset['Genre'] == genre]['Global_Sales'].sum()
        sales_data[idx] = sales_total
    print(f'Sales data of {console} by genre:', sales_data, '\n')
    return sales_data

# Build sales data for consoles A and B
sales_data_A = build_sales_data(data_A, CONSOLES['A'])
sales_data_B = build_sales_data(data_B, CONSOLES['B'])

# Function to calculate cosine similarity
def cosine_similarity(x, y):
    numerator = sum(a * b for a, b in zip(x, y))
    denominator = round(np.sqrt(sum(a ** 2 for a in x)) * np.sqrt(sum(b ** 2 for b in y)), 4)
    similarity = round(numerator / float(denominator), 4)
    return similarity

# Calculate similarity between consoles A and B
similarity_score = cosine_similarity(sales_data_A, sales_data_B)
print(f'The similarity between {CONSOLES["A"]} and {CONSOLES["B"]} is {similarity_score}')

# Function to plot sales distribution as pie chart
def plot_sales_distribution(sales_data, console):
    plt.figure(figsize=(8, 6))
    plt.pie(sales_data, labels=GENRES, autopct='%1.1f%%', startangle=140)
    plt.title(f'Sales Distribution of {console}')
    plt.show()

# Plot sales distribution for consoles A and B
plot_sales_distribution(sales_data_A, CONSOLES['A'])
plot_sales_distribution(sales_data_B, CONSOLES['B'])
