# Constants
DISTANCE_THRESHOLD = 0.80
NEIGHBOR_COUNT = 3

# File paths
path = '.'

token_path = '../compling/tokenizer_tokens.txt'

def k_shortest_paths(G, source, target, k, weight=None):
    return list(itertools.islice(nx.shortest_simple_paths(G, source, target, weight='weight'), k))

                nearest = word_vectors.most_similar(sentence[i], topn=NEIGHBOR_COUNT)[j]
                if nearest[1] >= DISTANCE_THRESHOLD:
                    word_tag = nltk.pos_tag(nltk.word_tokenize(sentence[i]))    
                    neighbor_tag = nltk.pos_tag(nltk.word_tokenize(nearest[0]))
                    if word_tag[0][1] and neighbor_tag[0][1] in {'NN', 'NNS', 'RB', 'RBR', 'RBS',

                                                                 'VBZ', 'JJ', 'JJR', 'JJS'}:
                        neighbor_node = nearest[0] + '*' + str(j) + '*'     
                        words.append(neighbor_node)
                        graph.add_node(neighbor_node)
                        if i == 0:
                            graph.add_edge("START", neighbor_node, weight=-round(nearest[1], 5))

                            graph.add_edge(prev_node, neighbor_node, weight=-round(nearest[1], 5))

    for w in range(len(words)):
        graph.add_edge(words[w], "END", weight=-1)

def main():

    # Check if the Word2Vec model file exists, if not, create and save the model
    if not os.path.isfile(model_path):
        sentences = LineSentence(token_path)
        model = Word2Vec(sentences, size=300, window=4, min_count=10)
        model.wv.save_word2vec_format('model.txt', binary=False)
