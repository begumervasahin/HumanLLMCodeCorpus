import numpy as np
import pandas as pd
from sklearn.datasets import load_iris
from sklearn.model_selection import train_test_split
from knn import calculate_similar_vector, main
from collections import Counter

# Load Iris dataset
iris = load_iris()
X = pd.DataFrame(iris.data, columns=["a", "b", "c", "d"])
Y = iris.target

# Split dataset into training and testing sets
train_X, test_X, train_Y, test_Y = train_test_split(X, Y, test_size=0.3, random_state=0)
train_X = train_X.reset_index()
test_X = test_X.reset_index()

# Define test functions
def test_calculate_similar_vector_shape():
    # Choose a random index from test data
    idx = np.random.choice(test_X["index"].values)
    # Get the corresponding feature vector
    vector = test_X[test_X["index"] == idx].values[0, 1:]
    K = 10
    # Ensure the shape of the result matches expectations
    assert calculate_similar_vector(vector, train_X, train_Y, K).shape == (K, )

def test_calculate_similar_vector_value1():
    # Choose a random index from test data
    idx = np.random.choice(test_X["index"].values)
    # Get the corresponding feature vector
    vector = test_X[test_X["index"] == idx].values[0, 1:]
    K = 10
    # Calculate similar vectors and find the most common label
    count_arrays = calculate_similar_vector(vector, train_X, train_Y, K)
    max_key = Counter(count_arrays).most_common(1)[0][0]
    # Ensure the predicted label matches the true label
    assert max_key == test_Y[test_X["index"] == idx].values[0]

def test_main_shape():
    K = 10
    # Ensure the shape of predictions matches the test labels
    assert main(train_X, test_X, train_Y, K).shape == test_Y.shape

def test_main_value():
    K = 3
    # Calculate accuracy rate
    rate = np.sum(main(train_X, test_X, train_Y, K) == test_Y) / test_Y.shape[0]
    # Ensure the accuracy rate exceeds 90%
    assert rate > 0.9

def test_main_value2():
    K = 3
    # Calculate accuracy rate
    rate = np.sum(main(train_X, test_X, train_Y, K) == test_Y) / test_Y.shape[0]
    # Ensure the accuracy rate is less than 100%
    assert rate < 1

# Run tests
if __name__ == '__main__':
    test_calculate_similar_vector_shape()
    test_calculate_similar_vector_value1()
    test_main_shape()
    test_main_value()
    test_main_value2()
    print("All tests passed!")
