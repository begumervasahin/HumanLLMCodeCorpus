import math
from nltk.tokenize import RegexpTokenizer
from nltk.stem import PorterStemmer
import os

# Function to tokenize and normalize text
def tokenize_normalize(raw_string):
    tokenizer = RegexpTokenizer(r'\w+')
    tokenized_text = tokenizer.tokenize(str(raw_string))
    normalized_text = [word.lower() for word in tokenized_text]
    return normalized_text

# Function to stem words
def stem_words(normalized_text):
    stemmer = PorterStemmer()
    stemmed_text = [stemmer.stem(word) for word in normalized_text]
    return stemmed_text

# Function to create a term frequency dictionary for a document
def create_tf_dictionary_doc(processed_text, doc_name, stats_dict):
    for word in processed_text:
        if word not in stats_dict:
            stats_dict[word] = {}
            stats_dict[word][doc_name] = 1
        else:
            stats_dict[word][doc_name] = stats_dict[word].get(doc_name, 0) + 1
    return stats_dict

# Function to create a term frequency dictionary for a query
def create_tf_dictionary_query(processed_query):
    query_dict = {}
    for word in processed_query:
        query_dict[word] = query_dict.get(word, 0) + 1
    return query_dict

# Function to calculate TF-IDF scores for documents
def calculate_tfidf_documents(stats_dict, num_docs):
    idf_dict = {}
    for word in stats_dict:
        idf_dict[word] = len(stats_dict[word])
    for word in stats_dict:
        for doc in stats_dict[word]:
            tf = 1 + math.log(stats_dict[word][doc])
            idf = math.log(num_docs / idf_dict[word])
            stats_dict[word][doc] = tf * idf
    return stats_dict, idf_dict

# Function to calculate TF-IDF scores for a query
def calculate_tfidf_query(query_dict, idf_dict, num_docs):
    for word in query_dict:
        tf = 1 + math.log(query_dict[word])
        idf = math.log(num_docs / idf_dict.get(word, 1))
        query_dict[word] = tf * idf
    return query_dict

# Example usage:
corpus_dir = "Dokumen"
text_files = [os.path.join(corpus_dir, f) for f in os.listdir(corpus_dir)]
stats_dict = {}
num_docs = len(text_files)
doc_list = []

# Process each document in the corpus
for doc_file in text_files:
    with open(doc_file, 'r') as doc:
        lines = [line.strip() for line in doc.readlines()]
        full_transcript = tokenize_normalize(lines)
        full_transcript = stem_words(full_transcript)
        stats_dict = create_tf_dictionary_doc(full_transcript, doc_file, stats_dict)
        doc_list.append(doc_file)

# Compute TF-IDF scores for documents and queries
stats_dict, idf_dict = calculate_tfidf_documents(stats_dict, num_docs)
list_of_words = sorted(stats_dict.keys())
doc_list.sort()
