import urllib.request
import re
import nltk
import string
from sklearn.feature_extraction.text import TfidfVectorizer
from nltk.stem.porter import PorterStemmer
from nltk.corpus import stopwords
from DatumBox import DatumBox

# Initialize DatumBox API with your API key
datum_box = DatumBox("2a13913dda346761765020c1f66e34f8")

# Compile regex pattern to remove HTML tags
HTML_TAG_PATTERN = re.compile(r"<[^>]+>")

# Download NLTK data
nltk.download('punkt')
nltk.download('stopwords')

# Initialize Stemmer and create punctuation removal mapping
stemmer = PorterStemmer()
PUNCTUATION_REMOVAL_MAP = dict((ord(char), None) for char in string.punctuation)

# Function to fetch web page content
def fetch_page(url):
    headers = {
        'User-Agent': 'Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.11 (KHTML, like Gecko) Chrome/23.0.1271.64 Safari/537.11',
        'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8',    
        'Accept-Charset': 'ISO-8859-1,utf-8;q=0.7,*;q=0.3',
        'Accept-Encoding': 'none',
        'Accept-Language': 'en-US,en;q=0.8',
        'Connection': 'keep-alive'
    }
    req = urllib.request.Request(url, headers=headers)
    try:
        with urllib.request.urlopen(req) as page:
            content = page.read().decode('utf-8')
            return content
    except urllib.error.HTTPError as e:
        print(e)
        return ''

# Function to remove HTML tags from text
def remove_html_tags(text):
    return HTML_TAG_PATTERN.sub('', text)

# Function to normalize text (remove punctuation, lowercase, stem)
def normalize_text(text):
    tokens = nltk.word_tokenize(text.lower().translate(PUNCTUATION_REMOVAL_MAP))
    return [stemmer.stem(token) for token in tokens]

# Initialize TF-IDF Vectorizer with custom tokenizer and stop words
vectorizer = TfidfVectorizer(tokenizer=normalize_text, stop_words=stopwords.words('english'))

# Function to calculate cosine similarity between two texts
def calculate_cosine_similarity(text1, text2):
    tfidf = vectorizer.fit_transform([text1, text2])
    return ((tfidf * tfidf.T).A)[0, 1]

# URLs of the web pages to crawl and compare
page_url1 = "http://www.geeksforgeeks.org/fundamentals-of-algorithms/"
page_url2 = "http://www.geeksforgeeks.org/data-structures/"

# Fetch web page contents
print("Crawling first page...")
page_content1 = fetch_page(page_url1)
print("Crawling second page...")
page_content2 = fetch_page(page_url2)

# Extract text from pages using DatumBox API
if page_content1 and page_content2:
    print("Extracting data from page 1...")
    extracted_text1 = datum_box.text_extract(page_content1)
    print("Extracting data from page 2...")
    extracted_text2 = datum_box.text_extract(page_content2)

    # Calculate and print cosine similarity between extracted texts
    similarity_score = calculate_cosine_similarity(extracted_text1, extracted_text2)
    print(f"Cosine Similarity between the two pages: {similarity_score}")
else:
    print("Failed to fetch page content.")
