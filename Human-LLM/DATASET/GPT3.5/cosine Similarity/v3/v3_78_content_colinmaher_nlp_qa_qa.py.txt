import nltk
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from nltk.stem import PorterStemmer
from collections import Counter
import string

class QAEngine:
    def __init__(self):
        # Download NLTK resources if not already downloaded
        nltk.download('punkt')
        nltk.download('stopwords')

        # Initialize NLTK components
        self.stop_words = set(stopwords.words('english'))
        self.stemmer = PorterStemmer()

        # Load story and questions data
        self.story = self.load_story()
        self.questions = self.load_questions()

    def load_story(self):
        # Load story data from file
        story_data = {}
        with open('./story.txt', 'r', encoding='utf-8') as story_file:
            story_data['text'] = story_file.read()
        return story_data

    def load_questions(self):
        # Load questions data from file
        questions_data = []
        with open('./questions.txt', 'r', encoding='utf-8') as questions_file:
            for line in questions_file:
                qid, text = line.split('::')
                questions_data.append({'qid': qid.strip(), 'text': text.strip()})
        return questions_data

    def preprocess_text(self, text):
        # Tokenize, remove punctuation, lowercase, remove stop words, and stem words
        tokens = word_tokenize(text)
        tokens = [token.lower() for token in tokens if token not in string.punctuation]
        tokens = [token for token in tokens if token not in self.stop_words]
        stemmed_tokens = [self.stemmer.stem(token) for token in tokens]
        return stemmed_tokens

    def calculate_tf_idf(self, tokens):
        # Calculate TF-IDF for tokens in a document
        tf_counts = Counter(tokens)
        tf_idf = {}
        for token, freq in tf_counts.items():
            idf = self.calculate_idf(token)
            tf_idf[token] = freq * idf
        return tf_idf

    def calculate_idf(self, token):
        # Calculate IDF for a token in the story
        token_count = sum(token in self.preprocess_text(para) for para in self.story['text'].split('\n') if para.strip())
        idf = 1 + (token_count > 0)
        return idf

    def answer_questions(self):
        # Answer each question using TF-IDF and find most similar paragraph
        answers = []
        for question_data in self.questions:
            tokens = self.preprocess_text(question_data['text'])
            tf_idf = self.calculate_tf_idf(tokens)
            answer = self.find_most_similar_paragraph(tf_idf)
            answers.append({'qid': question_data['qid'], 'answer': answer})
        return answers

    def find_most_similar_paragraph(self, query_tf_idf):
        # Find the most similar paragraph using TF-IDF
        max_similarity = 0
        most_similar_paragraph = ''
        for paragraph in self.story['text'].split('\n'):
            if paragraph.strip():
                para_tokens = self.preprocess_text(paragraph)
                para_tf_idf = self.calculate_tf_idf(para_tokens)
                similarity = sum(query_tf_idf[token] * para_tf_idf.get(token, 0) for token in query_tf_idf)
                if similarity > max_similarity:
                    max_similarity = similarity
                    most_similar_paragraph = paragraph
        return most_similar_paragraph

    def save_answers(self, answers):
        # Save answers to a file
        with open('./answers.txt', 'w', encoding='utf-8') as answers_file:
            for answer in answers:
                answers_file.write(f"{answer['qid']}:: {answer['answer']}\n")

def main():
    # Initialize QA engine, answer questions, and save answers
    qa_engine = QAEngine()
    answers = qa_engine.answer_questions()
    qa_engine.save_answers(answers)

if __name__ == "__main__":
    main()
