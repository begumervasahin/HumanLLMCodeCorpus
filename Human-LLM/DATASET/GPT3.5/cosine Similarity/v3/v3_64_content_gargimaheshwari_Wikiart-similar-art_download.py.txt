import sys
import getopt
from image_downloader import jsonloader, downloader
from resize_images import resizer

def parse_command_line_args(argv):
    """Parse the command-line arguments."""
    try:
        opts, _ = getopt.getopt(argv, "n:h", ['help'])
    except getopt.GetoptError:
        print_usage()
        sys.exit(2)

    n = None
    if opts:
        for opt, arg in opts:
            if opt == '-n':
                n = parse_num_images(arg)
            elif opt in ('-h', '--help'):
                print_usage()
                sys.exit()
            else:
                print("Unrecognized argument:", opt)
                print_usage()
                sys.exit(2)
    return n

def parse_num_images(num):
    """Parse the number of images to download."""
    try:
        num_images = int(num)
        if num_images <= 10:
            print('Number of images should be greater than 10')
            sys.exit(2)
        return num_images
    except ValueError:
        print("Invalid argument for -n option. Please provide an integer value.")
        sys.exit(2)

def print_usage():
    """Print usage information."""
    print("Usage: python <filename.py>")
    print("Optional argument -n for the number of images to be downloaded (should be greater than 10)")

def main(argv):
    """Main function."""
    num_images = parse_command_line_args(argv)

    if num_images is None:
        print("All images will be downloaded")
    else:
        downloader(num_images, jsonloader())
        resizer(num_images)

if __name__ == "__main__":
    main(sys.argv[1:])
