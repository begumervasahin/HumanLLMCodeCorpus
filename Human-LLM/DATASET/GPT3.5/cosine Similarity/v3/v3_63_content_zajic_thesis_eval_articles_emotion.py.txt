import pickle
import numpy as np
import matplotlib.pyplot as plt
from sklearn.decomposition import PCA
from sklearn.manifold import MDS
from WNAffect_modules import wnaffect
from collections import Counter, OrderedDict
from copy import deepcopy
import operator

class ArticleEmotionEvaluator:
    def __init__(self, filename):
        self.wordnet16_dir = "./wordnet/wordnet-1.6"
        self.wn_domains_dir = "./wordnet/wn-domains-3.2"
        self.wna = wnaffect.WNAffect(self.wordnet16_dir, self.wn_domains_dir)
        self.articles = self._load_articles(filename)
        self.em_dict = self._load_emotion_dictionary()

    def _load_articles(self, filename):
        with open(f'./pickle_files/{filename}', 'rb') as fp:
            return pickle.load(fp)

    def _load_emotion_dictionary(self):
        with open('./pickle_files/emotion_dictionary_bnc_baby_avg', 'rb') as fp:
            return pickle.load(fp)

    def eval_articles_emotions(self):
        for art in self.articles:
            art['emotion_score'] = {}
            art['emotion_score_sorted'] = self.tag_emotions_alt(art['body'])
        coordinates = [list(art['emotion_score'].values()) for art in self.articles]
        return np.array(coordinates)

    def pca_2D_visualization(self):
        labels = [f"{art['title']} ({art['source']})" for art in self.articles]
        scaled_coordinates = self._pca(self.coordinates, 2)
        self._visualize_2D_data(scaled_coordinates, labels, type='pca')

    def pca_3D_visualization(self):
        labels = [f"{art['title']} ({art['source']})" for art in self.articles]
        scaled_coordinates = self._pca(self.coordinates, 3)
        self._visualize_3D_data(scaled_coordinates, labels)

    def mds_2d_visualization(self):
        labels = [f"{art['title']} ({art['source']})" for art in self.articles]
        scaled_coordinates = self._multidimensional_scaling(self.coordinates, 2)
        self._visualize_2D_data(scaled_coordinates, labels, type="mds")

    def compare_pca_mds_rainbow(self):
        sc_coordinates = self._pca(self.coordinates, 2)
        colors = self._get_rainbow_color_codes(sc_coordinates)
        scaled_coordinates = self._multidimensional_scaling(self.coordinates, 2)
        self._visualize_2D_with_color_coding(scaled_coordinates, colors, type='mds metric')

    def tag_emotions_alt(self, article):
        bag_of_emotion_words = {}
        for word in article:
            if word in self.em_dict:
                bag_of_emotion_words[word] = deepcopy(self.em_dict[word])

        for word in bag_of_emotion_words:
            for key, value in bag_of_emotion_words[word].items():
                if value < 0.2:
                    bag_of_emotion_words[word][key] = 0

        counts = Counter(article)
        for word in bag_of_emotion_words:
            if counts[word] > 1:
                for key in bag_of_emotion_words[word]:
                    bag_of_emotion_words[word][key] *= counts[word]

        summed_em = {}
        emotions = [
            'apathy', 'neutral-unconcern', 'thing', 'pensiveness', 'gravity',
            'ambiguous-fear', 'ambiguous-expectation', 'ambiguous-agitation',
            'surprise', 'gratitude', 'levity', 'positive-fear', 'fearlessness',
            'positive-expectation', 'self-pride', 'affection', 'enthusiasm',
            'positive-hope', 'calmness', 'love', 'joy', 'liking', 'humility',
            'compassion', 'despair', 'shame', 'anxiety', 'negative-fear',
            'general-dislike', 'sadness', 'daze', 'ingratitude'
        ]
        for em in emotions:
            summed_em[em] = sum(bag_of_emotion_words.get(word, {}).get(em, 0) for word in article) / len(article)

        sorted_x = sorted(summed_em.items(), key=operator.itemgetter(1), reverse=True)
        return summed_em, sorted_x

    @staticmethod
    def _pca(coordinates, no_of_components):
        pca = PCA(n_components=no_of_components, random_state=123)
        pca.fit(coordinates)
        X = pca.transform(coordinates)
        print("explained variance ratio:")
        print(pca.explained_variance_ratio_)
        return X

    @staticmethod
    def _multidimensional_scaling(coordinates, no_components):
        print("Applying multidimensional scaling...")
        mds = MDS(
            n_components=no_components, metric=True, n_init=4, max_iter=300, verbose=0,
            eps=0.001, n_jobs=1, random_state=123, dissimilarity='euclidean'
        )
        return mds.fit_transform(coordinates)

    def _visualize_3D_data(coordinates, labels):
        pass

    def _get_rainbow_color_codes(coordinates):
        pass
