import pandas as pd
from sklearn.feature_extraction.text import CountVectorizer
from sklearn.metrics.pairwise import cosine_similarity
import movie_predict_knn

# Load movie dataset from a CSV file
df = pd.read_csv("D:/roops_backup/projects/data mining project/movie_recommender/movie_dataset.csv")

# Preprocess data: fill missing values and combine features
features_to_fill = ['keywords', 'cast', 'genres', 'director']
df[features_to_fill] = df[features_to_fill].fillna('')
df["combined_features"] = df.apply(lambda row: ' '.join(row[features_to_fill]), axis=1)

# Initialize CountVectorizer to convert text data into a numerical format
cv = CountVectorizer()
count_matrix = cv.fit_transform(df["combined_features"])

# Calculate cosine similarity between movies based on their combined features
cosine_sim = cosine_similarity(count_matrix)

# Define the movie of interest and get its index
target_movie = "Avengers: Age of Ultron"
target_movie_index = df[df['title'] == target_movie].index[0]

# Get top similar movies using cosine similarity
similar_movies = list(enumerate(cosine_sim[target_movie_index]))
sorted_similar_movies = sorted(similar_movies, key=lambda x: x[1], reverse=True)[1:7]

print(f"Movies similar to '{target_movie}' based on cosine similarity:")
for movie_id, similarity_score in sorted_similar_movies:
    movie_title = df.loc[movie_id, "title"]
    print(f"{movie_title} (Similarity Score: {similarity_score:.2f})")

# Use KNN for movie prediction
print("\nPredicted movies using KNN:")
num_predictions = 5
predicted_movies = movie_predict_knn.main(target_movie_index, num_predictions)
for movie_id in predicted_movies:
    movie_title = df.loc[movie_id, "title"]
    print(movie_title)
