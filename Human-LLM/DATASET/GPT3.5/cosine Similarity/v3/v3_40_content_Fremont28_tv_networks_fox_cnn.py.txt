import re
import string
import nltk
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from nltk.stem.porter import PorterStemmer
from sklearn.feature_extraction.text import CountVectorizer
from sklearn.metrics.pairwise import cosine_similarity

def preprocess_text(text):
    """Preprocess the given text."""
    words = word_tokenize(text)
    words = [w for w in words if w.isalpha()]
    stop_words = set(stopwords.words('english'))
    words = [w for w in words if w not in stop_words]
    porter = PorterStemmer()
    words = [porter.stem(w) for w in words]
    return words

def calculate_cosine_similarity(text1, text2):
    """Calculate the cosine similarity between two texts."""
    vectorizer = CountVectorizer(analyzer='word', min_df=1, stop_words='english', lowercase=True, token_pattern='[a-zA-Z0-9]{3,}')
    vectorized_text1 = vectorizer.fit_transform(text1)
    vectorized_text2 = vectorizer.fit_transform(text2)
    cosine_sim_scores = cosine_similarity(vectorized_text1, vectorized_text2)
    mean_cosine_sim_score = cosine_sim_scores.mean()
    return mean_cosine_sim_score

# Sample data
cnn_scroll = ["Sample CNN article text 1.", "Sample CNN article text 2.", "Sample CNN article text 3."]
fox_scroll = ["Sample Fox article text 1.", "Sample Fox article text 2.", "Sample Fox article text 3."]

# Preprocess CNN and Fox scrolls
cnn_words = [preprocess_text(text) for text in cnn_scroll]
fox_words = [preprocess_text(text) for text in fox_scroll]

# Calculate cosine similarity
mean_cosine_sim_score = calculate_cosine_similarity(cnn_words, fox_words)
print("Mean Cosine Similarity Score:", mean_cosine_sim_score)
