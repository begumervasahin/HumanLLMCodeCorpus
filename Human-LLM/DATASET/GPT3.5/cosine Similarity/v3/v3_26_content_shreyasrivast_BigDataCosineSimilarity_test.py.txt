import csv
import math
import heapq

# Global variables
term_freq = {}
norm_term_freq = {}
observe = []
inv_doc_freq = {}
tfidf_freq = {}
cosine_sim = {}
modulus = {}
k = 2

def read_file():
    # Read data from a CSV file
    with open('sample.csv', newline='') as csvfile:
        csv_reader = csv.reader(csvfile)
        for row in csv_reader:
            calculate_term_frequency(row[1], row[2])

def calculate_term_frequency(name, text):
    # Calculate term frequencies for each document
    if name not in term_freq:
        term_freq[name] = {}
    words = text.split(" ")
    for word in words:
        term_freq[name][word] = term_freq[name].get(word, 0) + 1

def calculate_normalized_frequency():
    # Calculate normalized term frequencies
    for name in term_freq:
        norm_term_freq[name] = {word: freq / len(term_freq[name]) for word, freq in term_freq[name].items()}

def calculate_inverse_doc_freq():
    # Calculate inverse document frequencies
    for name in term_freq:
        for word in term_freq[name]:
            if word not in observe:
                observe.append(word)
                count = sum(word in term_freq[other_name] for other_name in term_freq if other_name != name) + 1
                inv_doc_freq[word] = math.log(len(term_freq) / (1 + count), 2)

def calculate_norm_tfidf_freq():
    # Calculate TF-IDF scores
    for name in term_freq:
        tfidf_freq[name] = {word: norm_term_freq[name][word] * inv_doc_freq[word] for word in term_freq[name]}

def calculate_modulus():
    # Calculate the modulus for each document
    for name in tfidf_freq:
        modulus[name] = math.sqrt(sum(tfidf_freq[name][word] ** 2 for word in tfidf_freq[name]))

def calculate_cosine_similarity():
    # Calculate cosine similarity between documents
    for name in tfidf_freq:
        cosine_sim[name] = {}
        for other_name in tfidf_freq:
            if other_name != name:
                dot_product = sum(tfidf_freq[name][word] * tfidf_freq[other_name][word] for word in tfidf_freq[name] if word in tfidf_freq[other_name])
                cosine_sim[name][other_name] = dot_product / (modulus[name] * modulus[other_name])

def k_nearest_neighbors():
    # Find k-nearest neighbors for each document based on cosine similarity
    for name in cosine_sim:
        print(f"{name} : ")
        top_k = heapq.nlargest(k, cosine_sim[name], key=cosine_sim[name].get)
        for neighbor in top_k:
            print(neighbor)
        print("\n")

# Main program flow
read_file()
calculate_normalized_frequency()
calculate_inverse_doc_freq()
calculate_norm_tfidf_freq()
calculate_modulus()
calculate_cosine_similarity()
k_nearest_neighbors()
