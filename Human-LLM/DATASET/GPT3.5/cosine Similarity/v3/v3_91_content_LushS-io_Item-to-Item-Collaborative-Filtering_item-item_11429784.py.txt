def adj_to_neighbor_dict(adjacency_set):
    neighbor_dict = collections.defaultdict(lambda: set())
    for i, j in adjacency_set:
        if i != j:
            neighbor_dict[i].add(j)
            neighbor_dict[j].add(i)
    return neighbor_dict

def get_2d_neighbors(matrix):
    adjacency_set = set(
        (matrix[i, j], matrix[i+1, j])
        for i in range(matrix.shape[0]-1)
        for j in range(matrix.shape[1]-1)
    ) | set(
        (matrix[i, j], matrix[i, j+1])
        for i in range(matrix.shape[0]-1)
        for j in range(matrix.shape[1]-1)
    )
    return adj_to_neighbor_dict(adjacency_set)

# Calculate average and center CSR matrix
csr_matrix = sps.csr_matrix(sparse_matrix)
row_indices, col_indices, data_values = sps.find(csr_matrix)
row_counts = np.bincount(row_indices)
row_sums = np.bincount(row_indices, weights=data_values)
row_averages = row_sums / row_counts
centered_csr_matrix = sps.csr_matrix(
    (data_values - np.repeat(row_averages, np.diff(csr_matrix.indptr)), 
     col_indices, csr_matrix.indptr), 
    shape=csr_matrix.shape
)

# Calculate cosine similarity
cosine_similarity_matrix = centered_csr_matrix.dot(centered_csr_matrix.T).toarray()
diagonal_values = cosine_similarity_matrix.diagonal()
inv_diagonal_values = 1 / diagonal_values
inv_diagonal_values[np.isinf(inv_diagonal_values)] = 0
inv_magnitudes = np.sqrt(inv_diagonal_values)
normalized_cosine_similarity = cosine_similarity_matrix * inv_magnitudes
normalized_cosine_similarity = normalized_cosine_similarity.T * inv_magnitudes

# Calculate cosine similarity using sklearn
sklearn_cosine_similarity = cosine_similarity(centered_csr_matrix)

# Create COO matrix and fit Nearest Neighbors model
coo_matrix = sps.coo_matrix(normalized_cosine_similarity)
k_neighbors = 5
knn_model = NearestNeighbors(n_neighbors=k_neighbors)
knn_model.fit(coo_matrix)
