if __name__ == '__main__':

    song_id = '346576'
    text = ''

    # Retrieving comments
    for page in range(10):

        comments = comments['comments']
        for comment in comments:
            print(comment['content'])
            text += comment['content']

    # Extracting keywords
    keywords = jieba.analyse.extract_tags(text, topK=50)
    text = " ".join(keywords)

    # Generating word cloud
    background_image = plt.imread('tt.jpg')
    word_cloud = WordCloud(background_color="white",
                           width=1200, height=900,
                           mask=background_image,
                           font_path="simhei.ttf",
                           max_font_size=200,
                           random_state=50
                           )
    word_cloud_image = word_cloud.generate(text)

    # Displaying word cloud
    plt.imshow(word_cloud_image)
    plt.axis("off")
    plt.show()

    # Saving word cloud image
    file_path = 'image/' + str("aita") + '.png'
    word_cloud.to_file(file_path)
