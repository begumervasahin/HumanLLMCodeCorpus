from datetime import datetime
from utils import tools as tl
from synoptic.section1 import Group_iRixhVV

def fetch_synoptic_reports():
    """
    Generates the URL for fetching synoptic reports and extracts reports for a specific station.
    Returns:
        A list of synoptic reports.
    """
    # Retrieve today's date and six hours prior for URL formatting.
    today, today_minus_6h = tl.fecha_para_synops()

    base_url_es = ('http://ogimet.com/display_synops2.php?lugar=78764+78762+78774+78767&tipo=ALL&ord=DIR&nil=SI'
                   '&fmt=txt&ano={}&mes={}&day={}&hora={}&anof={}&mesf={}&dayf={}&horaf={}&enviar=Ver')

    formatted_url = base_url_es.format(today_minus_6h[0], today_minus_6h[1].zfill(2), today_minus_6h[2].zfill(2),
                                       today_minus_6h[3].zfill(2), today[0], today[1].zfill(2),
                                       today[2].zfill(2), today[3].zfill(2))
    print(formatted_url)

    # Assuming 'texto_web.txt' correctly formatted and contains the synoptic reports.
    synoptic_reports = tl.extraer_synops("texto_web.txt", 78774)
    return synoptic_reports

def evaluate_reports(reports, log_file):
    """
    Evaluates each synoptic report for errors.
    Args:
        reports: A list of synoptic reports.
        log_file: An open file object for logging the evaluations.
    """
    for report in reports:
        print(report)
        report_parts = report.split(' ')
        # Evaluate the specific group within the report using its designated class.
        group_evaluator = Group_iRixhVV(report_parts[4], log_file)
        group_evaluator.evaluate_group()

def main():
    with open("log.txt", "w") as log_file:
        synoptic_reports = fetch_synoptic_reports()
        evaluate_reports(synoptic_reports, log_file)

if __name__ == "__main__":
    main()
