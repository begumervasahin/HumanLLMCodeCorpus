import base64
import os
from cryptography.hazmat.backends import default_backend        
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC
from cryptography.fernet import Fernet

def generate_key(password):
    # Encode the password
    password_bytes = password.encode()
    # Generate a salt
    salt = b'salt_'
    # Derive a key from the password using PBKDF2HMAC
    kdf = PBKDF2HMAC(
        algorithm=hashes.SHA256(),
        length=32,
        salt=salt,
        iterations=100000,
        backend=default_backend()
    )
    key = base64.urlsafe_b64encode(kdf.derive(password_bytes))
    return key

def encrypt(key, input_file):
    # Define the output file name
    output_file = 'encrypted_' + input_file
    # Read the input file
    with open(input_file, 'rb') as file:
        data = file.read()
    # Create a Fernet cipher using the key
    fernet = Fernet(key)
    # Encrypt the data
    encrypted_data = fernet.encrypt(data)
    # Write the encrypted data to the output file
    with open(output_file, 'wb') as file:
        file.write(encrypted_data)

def decrypt(key, input_file):
    # Define the output file name
    output_file = 'decrypted_' + input_file
    # Read the input file
    with open(input_file, 'rb') as file:
        data = file.read()
    # Create a Fernet cipher using the key
    fernet = Fernet(key)
    # Decrypt the data
    decrypted_data = fernet.decrypt(data)
    # Write the decrypted data to the output file
    with open(output_file, 'wb') as file:
        file.write(decrypted_data)

def binary_encode(input_file):
    # Define the output file name
    output_file = 'binary_' + input_file
    # Read the input file
    with open(input_file, 'r') as file:
        data = file.read()
    # Convert each character to binary and join them together
    binary_data = ''.join([bin(ord(char))[2:].rjust(8, '0') for char in data])
    # Write the binary data to the output file
    with open(output_file, 'w') as file:
        file.write(binary_data)

def binary_decode(input_file):
    # Define the output file name
    output_file = 'debinary_' + input_file
    # Read the input file
    with open(input_file, 'r') as file:
        data = file.read()
    # Convert binary data to integer
    integer_data = int(data, 2)
    # Convert integer to bytes and decode
    debinary_data = integer_data.to_bytes((integer_data.bit_length() + 7) // 8, 'big').decode()
    # Write the decoded data to the output file
    with open(output_file, 'w') as file:
        file.write(debinary_data)

def main():
    # Prompt the user for an option
    option = input("\n'1' Encrypt a file\n'2' Decrypt a file\n'3' Convert to binary\n'4' Convert from binary\n>> ")
    if option == '1':
        password = input("Enter password >> ")
        name = input("\nEnter file name in current directory >> ")
        encrypt(generate_key(password), name)
    elif option == '2':
        password = input("Enter password >> ")
        name = input("\nEnter file name in current directory >> ")
        decrypt(generate_key(password), name)
    elif option == '3':
        name = input("\nEnter file name in current directory >> ")
        binary_encode(name)
    elif option == '4':
        name = input("\nEnter file name in current directory >> ")
        binary_decode(name)
    else:
        print("Sorry...")

if __name__ == "__main__":
    main()
