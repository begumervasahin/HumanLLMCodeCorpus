import unittest
from aes import AES

class TestAES(unittest.TestCase):
    def setUp(self):
        self.plaintext = bytes.fromhex('00112233445566778899aabbccddeeff')
        self.key = bytes.fromhex('000102030405060708090a0b0c0d0e0f')
        self.ciphertext = bytes.fromhex('69c4e0d86a7b0430d8cdb78070b4c55a')
        self.plaintext2 = bytes.fromhex('54776f204f6e65204e696e652054776f')
        self.key2 = bytes.fromhex('5468617473206d79204b756e67204675')
        self.ciphertext2 = bytes.fromhex('29c3505f571420f6402299b31a02d73a')
        self.aes = AES()

    def test_cipher(self):
        # Test first plaintext and key
        ciphertext = self.aes.cipher(self.plaintext, self.key)  
        self.assertEqual(ciphertext.hex(), self.ciphertext.hex())

        # Test second plaintext and key
        ciphertext2 = self.aes.cipher(self.plaintext2, self.key2)
        self.assertEqual(ciphertext2.hex(), self.ciphertext2.hex())

    def test_decipher(self):
        # Test deciphering of first ciphertext with its original key
        plaintext = self.aes.decipher(self.ciphertext, self.key)
        self.assertEqual(plaintext.hex(), self.plaintext.hex()) 

        # Test deciphering of second ciphertext with its original key
        plaintext2 = self.aes.decipher(self.ciphertext2, self.key2)
        self.assertEqual(plaintext2.hex(), self.plaintext2.hex())

    def test_sub_bytes(self):
        # Test substitution bytes
        state = self.aes.sub_bytes(bytes.fromhex('00102030405060708090a0b0c0d0e0f0'))
        self.assertEqual(state.hex(), '63cab7040953d051cd60e0e7ba70e18c')

        state2 = self.aes.sub_bytes(bytes.fromhex('001f0e543c4e08596e221b0b4774311a'))
        self.assertEqual(state2.hex(), '63c0ab20eb2f30cb9f93af2ba092c7a2')

    # Add other test methods here...

if __name__ == '__main__':
    unittest.main()
