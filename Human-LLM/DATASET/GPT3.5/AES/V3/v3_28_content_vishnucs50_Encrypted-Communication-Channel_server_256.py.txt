import random
import socket
from time import time
from Crypto.Cipher import AES

# Constants
HOST = '127.0.0.1'
PORT = 56739
BUFFER_SIZE = 4096
X_BOB = 253

class AESCipher:
    def __init__(self, key):
        self.key = key

    def encrypt(self, message):
        plaintext = self._pad(message)
        cipher = AES.new(self.key, AES.MODE_ECB)
        return cipher.encrypt(plaintext)

    def decrypt(self, ciphertext):
        cipher = AES.new(self.key, AES.MODE_ECB)
        plaintext = cipher.decrypt(ciphertext).decode('utf-8')
        return plaintext.rstrip('\x00')

    def _pad(self, message):
        length = len(message)
        padding_length = 16 - (length % 16)
        return message + chr(padding_length) * padding_length

def receive_data_and_respond(server_socket, sec_bob):
    while True:
        data, addr = server_socket.recvfrom(BUFFER_SIZE)
        print("Received message:", data.decode())
        server_socket.sendto(b"Hey! Got your Message", addr)

        data = server_socket.recvfrom(BUFFER_SIZE)
        print("Received parameters for Diffie-Hellman key exchange:", data.decode())
        string = data[0].decode()
        a, q, y_alice = map(int, (string[0], string[2:7], string[8:12]))

        y_bob = calculate_dh_y_bob(sec_bob, q, a)
        print("Computed Y_bob:", y_bob)
        server_socket.sendto(str(y_bob).encode(), addr)

        alice_key = calculate_aes_key(y_alice, sec_bob, q, a)
        key = bin(3750)[2:].zfill(32)
        start_time = time()

        data = server_socket.recvfrom(BUFFER_SIZE)[0]
        bob_cipher = AESCipher(key)
        bob_challenge = int(bob_cipher.decrypt(data))
        print("Initial challenge received from Alice:", bob_challenge)
        nonce = generate_random_number()
        bob_challenge -= 1
        message = f"{nonce},{bob_challenge}"
        cipher_bob = bob_cipher.encrypt(message)
        print("First challenge message sent:", nonce, cipher_bob)
        server_socket.sendto(cipher_bob, addr)

        data = server_socket.recvfrom(BUFFER_SIZE)[0]
        fin_challenge = int(bob_cipher.decrypt(data))
        print("Final challenge received:", fin_challenge)

        if fin_challenge == (nonce - 1):
            print("Authentication successful")
            message = "authentication successful" * 6
            cipher_bob = bob_cipher.encrypt(message)
            end_time = time()
            print("Duration of AES 192 encryption:", end_time - start_time)
            server_socket.sendto(cipher_bob, addr)
        else:
            print("Authentication failure")
            server_socket.sendto(b"Authentication failure", addr)
            break

def calculate_dh_y_bob(sec_bob, q, a):
    return pow(a, sec_bob, q)

def calculate_aes_key(y_alice, sec_bob, q, a):
    calculate_dh_y_bob(X_BOB, q, a)
    k = pow(y_alice, sec_bob, q)
    return bin(k)[2:].zfill(32)

def generate_random_number():
    return random.getrandbits(32)

def main():
    sec_bob = 102  # Assuming sec_bob is a predefined secret value
    with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as server_socket:
        server_socket.bind((HOST, PORT))
        receive_data_and_respond(server_socket, sec_bob)

if __name__ == '__main__':
    main()
