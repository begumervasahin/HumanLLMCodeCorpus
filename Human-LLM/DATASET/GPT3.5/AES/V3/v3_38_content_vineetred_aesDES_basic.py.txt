def column_hello(state):
    columns = []
    for j in range(4):
        column = []
        for i in range(4):
            column.append(state[i * 8 + 2 * j: i * 8 + 2 * j + 2])
        columns.append(column)
    return columns

def add_round_key(state_array, key):
    result = []
    key = column_hello(key)
    for i in range(4):
        row = []
        for j in range(4):
            value = hex(int(state_array[i][j], 16) ^ int(key[i][j], 16))[2:].zfill(2)
            row.append(value)
        result.append(row)
    return result

def sub_bytes(state_array):
    result = []
    for i in range(4):
        row = []
        for j in range(4):
            byte = hex(SBOX[int(state_array[i][j], 16)])
            row.append(byte)
        result.append(row)
    return result

def substitute_bytes(state_array):
    result = ""
    for i in range(0, 8, 2):
        byte = str(hex(SBOX[int(state_array[i:i+2], 16)]))[2:].zfill(2)
        result += byte
    return result

def shift_rows(state_array):
    columns = column_hello(state_array)
    for i in range(1, 4):
        columns[i] = columns[i][i:] + columns[i][:i]
    return columns

def xor(var1, var2):
    result = str(int(var1, 16) ^ int(var2, 16))
    result = hex(int(result))[2:].zfill(32)
    return result

def word_break(word):
    words = [word[i * 8:(i + 1) * 8] for i in range(4)]
    return words

def get_key_schedule(key):
    words = [key[i * 8:(i + 1) * 8] for i in range(4)]
    for i in range(4, 44):
        if i % 4 == 0:
            intermed = words[i - 1][2:] + words[i - 1][:2]
            sub_W = substitute_bytes(intermed)
            g_w = RC[i // 4 - 1] ^ int(sub_W, 16)
            hello = hex(int(words[i - 4], 16) ^ g_w)[2:]
            words.append(hello)
        else:
            hello = hex(int(words[i - 1], 16) ^ int(words[i - 4], 16))[2:]
            words.append(hello)
    return ["".join(words[i * 8:(i + 1) * 8]) for i in range(45)]

def encrypt(plaintext, key_schedule):
    arrXOR = xor(plaintext, key_schedule[0])
    state_array = word_break(arrXOR)
    for i in range(1, 11):
        state_array = column_hello(state_array)
        state_array = sub_bytes(state_array)
        state_array = shift_rows(state_array)
        state_array = add_round_key(state_array, key_schedule[i])
        print("Round", i, "State Array:", state_array)
