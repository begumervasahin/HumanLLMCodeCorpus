class GridNode:
    def __init__(self, x, y, grid_size=9):
        self.x = x
        self.y = y
        self.data = None
        self.grid_size = grid_size

    def __eq__(self, other):
        return self.x == other.x and self.y == other.y

    def set_f_value(self, f):
        self.f = f

    def get_f_value(self):
        return self.f

    def get_x(self):
        return self.x

    def get_y(self):
        return self.y

    def get_xy(self):
        return (self.x, self.y)

    def set_data(self, data):
        self.data = data

    def get_data(self):
        return self.data

    def get_neighbors(self, grid):
        neighbors = []
        for a in range(self.x - 1, self.x + 2):
            for b in range(self.y - 1, self.y + 2):
                if (a == self.x and b == self.y):
                    continue
                if (self.is_valid(a, b)):
                    node = GridNode(a, b)
                    node.set_data(grid[a][b])
                    neighbors.append(node)
        return neighbors

    def is_valid(self, x, y):
        return 0 <= x < self.grid_size and 0 <= y < self.grid_size

    def closed_neighbors_to_open(self, grid):
        neighbors = []
        closed = []
        neighbors = self.get_neighbors(grid)
        for cell in neighbors:
            if cell.get_data() == ' ':
                closed.append(cell)
        return closed

    def check_cell(self, grid):
        if self.data == 'F' or self.data == ' ':
            return 0
        bombs = 0
        neighbors = self.get_neighbors(grid)
        closed = self.closed_neighbors_to_open(grid)
        closed_count = len(closed)
        for cell in neighbors:
            if cell not in closed:
                if cell.get_data() == 'F':
                    bombs += 1
        if closed_count != 0 and grid[self.x][self.y] != ' ':
            return (int(grid[self.x][self.y]) - bombs) / closed_count
        else:
            return 0

# Example usage:
if __name__ == "__main__":
    grid = [
        ['1', 'F', ' ', '2', ' '],
        [' ', '2', 'F', ' ', ' '],
        ['1', ' ', ' ', '2', 'F'],
        [' ', '1', '1', ' ', '1'],
        [' ', ' ', ' ', '2', ' ']
    ]

    node = GridNode(1, 1)
    node.set_data(grid[node.get_x()][node.get_y()])
    result = node.check_cell(grid)
    print("Result:", result)
