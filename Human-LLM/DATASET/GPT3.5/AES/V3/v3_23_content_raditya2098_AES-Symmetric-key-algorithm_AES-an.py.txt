def sbox_substitute(input_string):
    """
    Substitute each character in the input string using the AES S-box.

    Args:
        input_string (str): The string to be processed.

    Returns:
        str: A new string with each character substituted via the S-box.
    """
    # Convert input string to hexadecimal, perform S-box substitution, and return the new string
    substituted_chars = []
    for char in input_string:
        # Convert character to ASCII value, then to corresponding S-box value
        ascii_val = ord(char)
        sbox_val = Sbox[ascii_val]
        
        # Convert the S-box value back to a character
        substituted_char = chr(sbox_val)
        substituted_chars.append(substituted_char)
    
    # Join all substituted characters back into a single string
    substituted_string = ''.join(substituted_chars)
