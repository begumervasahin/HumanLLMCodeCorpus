from Crypto.Cipher import AES
from Crypto import Random

def xor(a, b):
    """
    Perform a bitwise XOR operation between two binary strings. 
    """
    result = ''
    for bit1, bit2 in zip(a, b):
        result += str(int(bit1) ^ int(bit2))
    return result

def encrypt_credit_card_number(card_number, rounds, produce_keys):
    """
    Encrypt the credit card number using AES encryption with the specified number of rounds.
    """
    encoded_number = '{0:054b}'.format(int(card_number))        
    left_half = encoded_number[:27]
    right_half = encoded_number[27:]
    round_keys = []

    if produce_keys:
        round_keys = [Random.new().read(16).hex() for _ in range(rounds)]

    for i, key in enumerate(round_keys):
        cipher = AES.new(bytes.fromhex(key))
        hex_right_half = '{:08x}'.format(int(right_half + '0'*5, 2))
        block = hex_right_half + '0'*23 + str(i+1)
        encryption_result = cipher.encrypt(bytes.fromhex(block))[:4].hex() + '0'
        right_half = xor(encryption_result, left_half + '0'*5)  
        left_half = right_half

    return left_half + right_half

def decrypt_credit_card_number(cipher, rounds, round_keys):     
    """
    Decrypt the cipher using AES decryption with the specified number of rounds and given round keys.
    """
    encoded_number = '{0:054b}'.format(int(cipher))
    left_half = encoded_number[:27]
    right_half = encoded_number[27:]

    for i in range(rounds, 0, -1):
        key = round_keys[i-1]
        cipher = AES.new(bytes.fromhex(key))
        hex_left_half = '{:08x}'.format(int(left_half + '0'*5, 2))
        block = hex_left_half + '0'*23 + str(i)
        encryption_result = cipher.encrypt(bytes.fromhex(block))[:4].hex() + '0'
        left_half = xor(encryption_result, right_half + '0'*5)  
        right_half = left_half

    return left_half + right_half

credit_card_number = "4532294977918448"
print('Number to encrypt = ' + credit_card_number)

round_keys = []
encrypted_number = encrypt_credit_card_number(credit_card_number, 6, True)
print('Encrypted number = ' + encrypted_number)
