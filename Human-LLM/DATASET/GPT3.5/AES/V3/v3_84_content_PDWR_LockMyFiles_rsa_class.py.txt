import json
from cryptography.hazmat.primitives.asymmetric import rsa
from cryptography.hazmat.primitives import serialization
from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives.asymmetric import padding
from cryptography.hazmat.primitives import hashes

class RSA_Class:
    @staticmethod
    def generate_private_key():
        """Generate a private key."""
        return rsa.generate_private_key(
            public_exponent=65537,
            key_size=2048,
            backend=default_backend()
        )

    @staticmethod
    def generate_public_key(private_key):
        """Generate a public key from the given private key."""
        return private_key.public_key()

    @staticmethod
    def save_private_key(file_path, private_key):
        """Save the private key to a file."""
        with open(file_path, "wb") as f:
            f.write(private_key.private_bytes(
                encoding=serialization.Encoding.PEM,
                format=serialization.PrivateFormat.TraditionalOpenSSL,
                encryption_algorithm=serialization.NoEncryption(),
            ))

    @staticmethod
    def save_public_key(file_path, public_key):
        """Save the public key to a file."""
        with open(file_path, "wb") as f:
            f.write(public_key.public_bytes(
                encoding=serialization.Encoding.PEM,
                format=serialization.PublicFormat.SubjectPublicKeyInfo,
            ))

    @staticmethod
    def load_private_key(file_path):
        """Load the private key from a file."""
        return serialization.load_pem_private_key(
            open(file_path, 'rb').read(), password=None, backend=default_backend())

    @staticmethod
    def load_public_key(file_path):
        """Load the public key from a file."""
        return serialization.load_pem_public_key(
            open(file_path, 'rb').read(), backend=default_backend())

    @staticmethod
    def encrypt(public_key, plaintext):
        """Encrypt plaintext using the given public key."""
        return public_key.encrypt(
            plaintext,
            padding.OAEP(
                mgf=padding.MGF1(algorithm=hashes.SHA256()),
                algorithm=hashes.SHA256(),
                label=None
            )
        )

    @staticmethod
    def decrypt(private_key, ciphertext):
        """Decrypt ciphertext using the given private key."""
        return private_key.decrypt(
            ciphertext,
            padding.OAEP(
                mgf=padding.MGF1(algorithm=hashes.SHA256()),
                algorithm=hashes.SHA256(),
                label=None
            )
        )

    @staticmethod
    def save_data(file_path, data):
        """Save data to a file."""
        with open(file_path, 'w') as f:
            json.dump(data, f)

    @staticmethod
    def load_data(file_path):
        """Load data from a file."""
        with open(file_path, "r") as json_file:
            return json.load(json_file)

# Example usage:
if __name__ == "__main__":
    # Generate and save private key
    private_key = RSA_Class.generate_private_key()
    RSA_Class.save_private_key("private_key.pem", private_key)

    # Generate and save public key
    public_key = RSA_Class.generate_public_key(private_key)
    RSA_Class.save_public_key("public_key.pem", public_key)

    # Load private and public keys
    loaded_private_key = RSA_Class.load_private_key("private_key.pem")
    loaded_public_key = RSA_Class.load_public_key("public_key.pem")

    # Encrypt and decrypt data
    plaintext = b"Hello, World!"
    ciphertext = RSA_Class.encrypt(loaded_public_key, plaintext)
    decrypted_plaintext = RSA_Class.decrypt(loaded_private_key, ciphertext)        

    print("Original plaintext:", plaintext)
    print("Decrypted plaintext:", decrypted_plaintext.decode())
