import sys
import getopt
import os
from Crypto.Hash import SHA256, MD5, SHA
from Crypto.PublicKey import RSA
from Crypto.Signature import PKCS1_v1_5

# Define the block size for reading files
BLOCK_SIZE = 1024 * 1024

def verify_signature(hash_algo, input_file, signature_file, public_key_file):
    print("Hash mode:", hash_algo)

    # Check if the provided hash algorithm is valid
    valid_hash_algorithms = ["SHA256", "MD5", "SHA", "SHA1", "SHA-1"]
    if hash_algo not in valid_hash_algorithms:
        print("Available hash modes:", ", ".join(valid_hash_algorithms))
        print("Please choose a valid hash mode.")
        sys.exit(1)

    # Initialize the hash object based on the chosen algorithm      
    if hash_algo == "SHA256":
        hash_obj = SHA256.new()
    elif hash_algo == "MD5":
        hash_obj = MD5.new()
    elif hash_algo in ["SHA", "SHA1", "SHA-1"]:
        hash_obj = SHA.new()

    # Calculate the hash of the input file
    with open(input_file, "rb") as fin:
        while True:
            block = fin.read(BLOCK_SIZE)
            if len(block) == 0:
                break
            hash_obj.update(block)

    # Read the signature from the signature file
    with open(signature_file, "rb") as fin2:
        signature = fin2.read()
        print("Signature retrieved:", signature)

    # Read the public key from the public key file
    if not os.path.isfile(public_key_file):
        print("Public key does not exist. Task failed.")
        return False

    with open(public_key_file, "r") as fin3:
        public_key = RSA.importKey(fin3.read())

    # Verify the signature using the public key
    verifier = PKCS1_v1_5.new(public_key)
    result = verifier.verify(hash_obj, signature)
    print("Verification result:", result)
    return result

def main(argv):
    try:
        opts, args = getopt.getopt(argv, "h:")
    except getopt.GetoptError:
        print("Usage: very_sign.py -h <hash> <input_file> <signature_file> <public_key_file>")
        sys.exit(2)

    hash_algo = None
    for opt, arg in opts:
        if opt == "-h":
            hash_algo = arg.upper()

    if len(args) != 3 or not hash_algo:
        print("Usage: very_sign.py -h <hash> <input_file> <signature_file> <public_key_file>")
        sys.exit(2)

    input_file, signature_file, public_key_file = args
    verification_result = verify_signature(hash_algo, input_file, signature_file, public_key_file)
    print("Verification result:", verification_result)
    return verification_result

if __name__ == "__main__":
    main(sys.argv[1:])
