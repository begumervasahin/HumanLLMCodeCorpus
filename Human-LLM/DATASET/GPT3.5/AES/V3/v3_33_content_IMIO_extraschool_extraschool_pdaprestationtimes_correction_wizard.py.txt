from odoo import models, fields, api

class ExtraschoolPrestationTimesCorrectionWizard(models.TransientModel):
    """
    A wizard for correcting prestation times in the 'extraschool' module.
    Allows users to update the type and time of prestations in bulk.
    """
    _name = 'extraschool.pdaprestationtimes_correction_wizard'
    
    # Define a selection field for prestation type with options for 'In' and 'Out'.
    prestation_type = fields.Selection(
        [('E', 'In'), ('S', 'Out')],
        string='Prestation Type',
        required=True
    )
    
    # Define a float field for specifying the new prestation time.
    prestation_time = fields.Float(
        string='Prestation Time',
        required=True
    )

    @api.multi
    def apply_correction(self):
        """
        Applies the specified corrections (type and time) to selected prestation records.
        After updating, it resets and checks each prestation's day aggregate.
        """
        # Retrieve the active_ids from the context, representing selected prestation records.
        active_ids = self._context.get('active_ids')
        
        if active_ids:
            # Browse the prestation records by active_ids.
            prestations_to_correct = self.env['extraschool.pdaprestationtimes'].browse(active_ids)
            
            # Write the new type and time to the selected prestations.
            prestations_to_correct.write({
                'es': self.prestation_type,
                'prestation_time': self.prestation_time
            })
            
            # For each prestation, reset and check its day aggregate.
            for prestation in prestations_to_correct:
                day_aggregate = prestation.prestation_times_of_the_day_id
                day_aggregate.reset()
                day_aggregate.check()

        return True
