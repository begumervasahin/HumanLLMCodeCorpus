import string
from nltk.corpus import stopwords
from nltk.tokenize import word_tokenize

def clean_presenters_list(presenters):
    for presenter in presenters:
        clean_affiliation_names(presenter)
        clean_affiliation_locations(presenter)
    return presenters

def clean_affiliation_names(presenter):
    affiliations = presenter['affiliation_names']
    for idx, name in enumerate(affiliations):
        if name == "Queen Mary University London":
            affiliations[idx] = "Queen Mary University of London"
        elif name == "Centre for Digital Music (C4DM)":
            affiliations[idx] = "Queen Mary University of London"
        elif name == "Dolby Laboratories":
            affiliations[idx] = "Dolby Laboratories, Inc."
        elif name in ["Meyer Sound Labs", "Meyer Sound"]:
            affiliations[idx] = "Meyer Sound Laboratories"
        elif name == "BBC R&D":
            affiliations[idx] = "BBC Research and Development"
        elif name == "The Centre for Interdisciplinary Research in Music Media and Technology":
            affiliations[idx] = "Centre for Interdisciplinary Research in Music Media and Technology"

def clean_affiliation_locations(presenter):
    locations = presenter['affiliation_locations']
    for idx, location in enumerate(locations):
        country = location.split(',')[-1].strip()
        if country == "NY USA" or country == "CA USA":
            locations[idx] = "USA"

def clean_affiliation(affiliation):
    if affiliation == "Queen Mary University London":
        return "Queen Mary University of London"
    elif affiliation == "Centre for Digital Music (C4DM)":
        return "Queen Mary University of London"
    elif affiliation == "Dolby Laboratories":
        return "Dolby Laboratories, Inc."
    elif affiliation in ["Meyer Sound Labs", "Meyer Sound"]:
        return "Meyer Sound Laboratories"
    elif affiliation == "BBC R&D":
        return "BBC Research and Development"
    elif affiliation == "The Centre for Interdisciplinary Research in Music Media and Technology":
        return "Centre for Interdisciplinary Research in Music Media and Technology"
    return affiliation

def remove_stopwords(words):
    custom_stop_words = ["’", 'paper', 'using', 'used', 'based', 'different']
    stop_words = set(stopwords.words('english') + list(string.punctuation) + custom_stop_words)
    word_tokens = word_tokenize(words)
    filtered_words = [word for word in word_tokens if word.lower() not in stop_words]
    return filtered_words

# Example usage:
if __name__ == "__main__":
    # Example data
    presenters = [
        {'affiliation_names': ["Queen Mary University London", "Dolby Laboratories"], 'affiliation_locations': ["London, UK", "CA USA"]},
        {'affiliation_names': ["Centre for Digital Music (C4DM)", "Meyer Sound Labs"], 'affiliation_locations': ["London, UK", "NY USA"]}
    ]
    affiliation = "Queen Mary University London"
    words = "This is a sample text used for testing purposes."  

    # Clean presenters list
    cleaned_presenters = clean_presenters_list(presenters)      
    print("Cleaned Presenters:", cleaned_presenters)

    # Clean affiliation
    cleaned_affiliation = clean_affiliation(affiliation)        
    print("Cleaned Affiliation:", cleaned_affiliation)

    # Remove stopwords from words
    filtered_words = remove_stopwords(words)
    print("Filtered Words:", filtered_words)
