import struct

# Function to perform left rotation on a 32-bit integer
def left_rotate(i, n):
    return ((i << n) & 0xffffffff) | (i >> (32 - n))

# MD4 round functions
def F(x, y, z):
    return (x & y) | (~x & z)

def G(x, y, z):
    return (x & y) | (x & z) | (y & z)

def H(x, y, z):
    return x ^ y ^ z

# MD4 hashing algorithm implementation
class MD4(object):
    def __init__(self):
        self.remainder = ""
        self.count = 0
        self.h = [
            0x67452301,
            0xefcdab89,
            0x98badcfe,
            0x10325476
        ]

    def _add_chunk(self, chunk):
        self.count += 1
        X = list(struct.unpack("<16I", chunk) + (None,) * (80 - 16))
        h = self.h[:]  # Create a copy of the initial hash values
        # Constants for left rotations
        S = ((3, 7, 11, 19), (3, 5, 9, 13), (3, 9, 11, 15))
        K = ((0, 1, 2, 3), (0, 4, 8, 12), (0, 2, 1, 3))
        T = (0x00000000, 0x5a827999, 0x6ed9eba1)
        
        for i in range(3):
            s = S[i]
            k = K[i]
            t = T[i]
            for r in range(16):
                idx = (16 - r) % 4
                h[idx] = left_rotate((h[idx] + globals()[chr(72 + i)](h[(idx + 1) % 4], h[(idx + 2) % 4], h[(idx + 3) % 4]) + X[k[r]] + t) % 2 ** 32, s[r % 4])

        for i, v in enumerate(h):
            self.h[i] = (v + self.h[i]) % 2 ** 32

    def update(self, data):
        message = self.remainder + data
        remainder_length = len(message) % 64
        if remainder_length != 0:
            self.remainder = message[-remainder_length:]
        else:
            self.remainder = ""
        
        for chunk_start in range(0, len(message) - remainder_length, 64):
            self._add_chunk(message[chunk_start:chunk_start + 64])
        
        return self

    def finish(self):
        # Compute the length of the message in bits and pad the message
        message_length = len(self.remainder) + 64 * self.count
        padding_length = (55 - message_length) % 64
        self.update("\x80" + "\x00" * padding_length + struct.pack("<Q", message_length * 8))
        # Pack the hash values into bytes and reset the hash state
        result = struct.pack("<4I", *self.h)
        self.__init__()
        # Return the hexadecimal representation of the hash     
        return result.encode("hex")

# Example usage:
md4 = MD4()
md4.update("Hello, world!")
print(md4.finish())
