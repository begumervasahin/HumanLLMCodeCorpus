# Define node data containing the adjacency list representation of the graph
nodes = {
    'a': [2, 'B', 36, 'C', 61],
    'b': [2, 'A', 36, 'D', 31],
    'c': [4, 'A', 61, 'D', 32, 'F', 31, 'L', 80],
    'd': [3, 'B', 31, 'C', 32, 'E', 52],
    'e': [2, 'D', 52, 'G', 43],
    'f': [3, 'C', 31, 'K', 112, 'J', 122],
    'g': [2, 'E', 43, 'H', 40],
    'h': [2, 'G', 40, 'I', 40],
    'i': [2, 'H', 40, 'J', 45],
    'j': [3, 'I', 45, 'F', 122, 'K', 36],
    'k': [3, 'J', 36, 'F', 112, 'M', 32],
    'l': [2, 'C', 80, 'M', 102]
}

# Define distances between nodes
distances = {
    'A': 223, 'B': 222, 'C': 166, 'D': 192,
    'E': 165, 'F': 136, 'G': 122, 'H': 111,
    'I': 100, 'J': 60, 'K': 32, 'L': 102
}

# Initialize variables
path = []
current_node = input("Enter the starting node: ").lower()
path.append(current_node)
final_node = 'm'

# Define a function to calculate the total cost of reaching a node
def calculate_total_cost(node):
    total_cost = 0
    for i in range(0, len(nodes[node]), 2):
        if i + 1 < len(nodes[node]):
            total_cost += nodes[node][i + 1] + distances[nodes[node][i]]
    return total_cost

# Traverse nodes until reaching the final node 'M'
while current_node.upper() != final_node.upper():
    min_cost = float('inf')
    next_node = ''

    # Iterate over adjacent nodes to find the one with minimum cost
    for i in range(1, len(nodes[current_node]), 2):
        adjacent_node = nodes[current_node][i]
        cost = calculate_total_cost(adjacent_node)
        if cost < min_cost:
            min_cost = cost
            next_node = adjacent_node

    # Append the next node to the path and update the current node
    path.append(next_node)
    current_node = next_node

# Print the path
print("The shortest path to reach node 'M' from the starting node is:", path)
