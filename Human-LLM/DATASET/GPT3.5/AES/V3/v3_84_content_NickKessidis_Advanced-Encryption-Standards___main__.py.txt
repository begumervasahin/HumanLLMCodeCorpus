import random

if __name__ == "__main__":
    # Generate a random number and key
    random_number = random.randrange(2**127, (2**128)-1)
    random_key = random.randrange(2**127, (2**128)-1)

    # Display the result of XOR and the plain text
    xor_result = random_number ^ random_key
    print("Result of XOR:", bin(xor_result))
    print("Plain text:", bin(random_number))

    # Initialize the state array
    state_array = ["0" * 8 for _ in range(16)]
    for i in range(16):
        state_array[i] = bin(random_number)[8*i+2:8*i+10]

    # Display the state array
    print("\nState Array:")
    for row in state_array:
        print(row)

    # Generate a random S-box
    s_box = [random.randrange(2**7, (2**8)-1) for _ in range(256)]

    # Perform substitution process on the state array
    for i in range(16):
        current_row = state_array[i]
        row_index = int(current_row[:4], 2)
        col_index = int(current_row[4:], 2)
        state_array[i] = s_box[(col_index + row_index) % 16]

    # Display the state array after the substitution process
    print("\nState Array after the Substitution process:")
    for row in state_array:
        print(row)
