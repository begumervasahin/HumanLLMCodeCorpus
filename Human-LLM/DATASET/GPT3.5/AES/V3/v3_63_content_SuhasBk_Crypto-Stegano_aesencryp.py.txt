import os
import smtplib
from getpass import getpass
from Crypto.Cipher import AES
from email.message import EmailMessage

def get_email_credentials():
    email = os.environ.get('EMAIL')
    passwd = os.environ.get('PASSWORD')
    return email, passwd

def generate_key_and_iv():
    key = os.urandom(16)
    iv = os.urandom(16)
    return key, iv

def get_original_data():
    original_data = getpass("Enter the '16-byte' message to be encrypted:\n")        
    original_data = original_data[:16].ljust(16)
    return str.encode(original_data)

def read_dummy_data(image_path):
    with open(image_path, 'rb') as f:
        dummy_data = f.read()
    return dummy_data

def encrypt_data(key, iv, original_data):
    aes_enc = AES.new(key, AES.MODE_CBC, iv)
    enc = aes_enc.encrypt(original_data)
    return enc

def compose_email_message(email, original_data, dummy_data, image_name):
    msg = EmailMessage()
    msg['To'] = input("Enter the email-ID of the receiver:\n")
    msg['Subject'] = 'Cryptography Demo'
    msg['From'] = email
    msg.set_content(input("Type any extra message or instructions to be sent:\n"))
    msg.add_attachment(dummy_data + original_data + iv + key,
                       maintype='image',
                       subtype='jpeg',
                       filename=image_name + '.jpg')
    return msg

def send_email(email, passwd, msg):
    with smtplib.SMTP_SSL('smtp.gmail.com', 465) as smtp:
        smtp.login(email, passwd)
        smtp.send_message(msg)

def main():
    email, passwd = get_email_credentials()
    key, iv = generate_key_and_iv()
    original_data = get_original_data()
    image_path = input("Enter the path to an image file:\n")
    dummy_data = read_dummy_data(image_path)
    enc = encrypt_data(key, iv, original_data)
    image_name = input("Enter the name of the image to be sent (without the extension):\n")
    msg = compose_email_message(email, enc, dummy_data, image_name)
    send_email(email, passwd, msg)
    print('The message has been successfully encrypted, embedded, and sent!')

if __name__ == "__main__":
    main()
