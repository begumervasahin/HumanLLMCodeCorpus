import pyAesCrypt
import extUtils
import glob
from os import remove, path

# Constants
BUFFER_SIZE = 64 * 1024

def encrypt_file(input_file, output_file, password):
    """Encrypts a single file."""
    pyAesCrypt.encryptFile(input_file, output_file, password, BUFFER_SIZE)
    # Remove the original plaintext file after encryption       
    remove(input_file)
    return output_file

def decrypt_file(input_file, output_file, password):
    """Decrypts a single file."""
    pyAesCrypt.decryptFile(input_file, output_file, password, BUFFER_SIZE)
    # Remove the encrypted file after decryption
    remove(input_file)
    return output_file

def encrypt(input_path, password):
    """Encrypts either a file or a folder."""
    if path.isfile(input_path):
        # Encrypt a single file
        return encrypt_single_file(input_path, password)
    else:
        # Encrypt all files in a folder
        return encrypt_folder_contents(input_path, password)

def decrypt(input_path, password):
    """Decrypts either a file or a folder."""
    if path.isfile(input_path):
        # Decrypt a single file
        return decrypt_single_file(input_path, password)
    else:
        # Decrypt all files in a folder
        return decrypt_folder_contents(input_path, password)

def encrypt_single_file(file_path, password):
    """Encrypts a single file."""
    encrypted_file_path = extUtils.add_ext(file_path)
    return encrypt_file(file_path, encrypted_file_path, password)

def decrypt_single_file(file_path, password):
    """Decrypts a single file."""
    decrypted_file_path = extUtils.remove_ext(file_path)
    return decrypt_file(file_path, decrypted_file_path, password)

def encrypt_folder_contents(folder_path, password):
    """Encrypts all files in a folder."""
    files = glob.glob(path.join(folder_path, "**/*.*"), recursive=True)
    return [encrypt_single_file(file, password) for file in files]

def decrypt_folder_contents(folder_path, password):
    """Decrypts all files in a folder."""
    files = glob.glob(path.join(folder_path, "**/*.*"), recursive=True)
    return [decrypt_single_file(file, password) for file in files]
