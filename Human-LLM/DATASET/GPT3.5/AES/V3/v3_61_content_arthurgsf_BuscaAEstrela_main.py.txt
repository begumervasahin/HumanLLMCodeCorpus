import pandas as pd
import numpy as np

class TreeNode:
    def __init__(self, data):
        self.data = data
        self.children = []
        self.parent = None

    def add_child(self, child):
        child.parent = self
        self.children.append(child)
        return child

    def add_children(self, children):
        self.children.extend(children)
        for child in children:
            child.parent = self
        return self.children

    def key(self):
        return self.data["key"]

    def cost(self):
        return self.data["cost"]

class Heap:
    def __init__(self):
        self.heap = []

    def append(self, node):
        self.heap.append(node)
        self.heap.sort(key=lambda x: x.data["cost"])

    def pop(self):
        return self.heap.pop(0)

class Map:
    def __init__(self, graph):
        self.graph = graph

    def get_heuristic(self, start, end):
        if self.graph[start][end] == -1:
            return self.graph[end][start]
        elif self.graph[end][start] == -1:
            return self.graph[start][end]
        return min(self.graph[start][end], self.graph[end][start])

    def get_distance(self, start, end):
        if self.graph[start][end] == -1:
            return -1
        elif self.graph[end][start] == -1:
            return -1
        return max(self.graph[start][end], self.graph[end][start])

    def get_neighbors(self, city):
        triangle_inf = np.tril(np.ones(self.graph.shape)).astype(np.bool)     
        neighbors = self.graph.where(triangle_inf, -1)
        return list(neighbors[city][neighbors[city] > 0].index) + list(neighbors.loc[city][neighbors.loc[city] > 0].index)

def astar_search(map, start, end):
    start_node = TreeNode({"key": start, "cost": map.get_heuristic(start, end)}) 

    heap = Heap()
    current_cost = 0
    while start_node.data["key"] != end:
        neighbors = map.get_neighbors(start_node.data["key"])
        neighbors = [
            {
                "key": v,
                "cost": current_cost + map.get_heuristic(v, end) + map.get_distance(start_node.key(), v)
            }
            for v in neighbors
        ]
        neighbors = start_node.add_children(neighbors)
        heap.append(neighbors)
        new_node = heap.pop()
        current_cost += map.get_distance(start_node.key(), new_node.key())
        start_node = new_node
    path = [start_node.key()]
    while start_node.parent is not None:
        start_node = start_node.parent
        path.append(start_node.key())
    path.reverse()
    return path

    map = Map(pd.read_csv("grafo.csv", index_col=0).fillna(-1).astype(np.float64))
    start_city = input("Starting City: ")
    end_city = input("Destination City: ")
    path = astar_search(map, start_city, end_city)
