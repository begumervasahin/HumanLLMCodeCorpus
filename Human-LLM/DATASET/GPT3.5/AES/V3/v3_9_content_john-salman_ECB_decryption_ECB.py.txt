import sys

def read_file_contents(filename):
    """Read the contents of the file."""
    try:
        with open(filename, 'r') as file:
            return file.read().splitlines()
    except FileNotFoundError:
        print(f"Error: File '{filename}' not found.")
        return None

def extract_flags_and_outputs(lines):
    """Extract flags and outputs from each line."""
    flags = []
    outputs = []

    for line in lines:
        flag_index = line.find(":")  # Find the position of ':'
        if flag_index == -1:
            print(f"Warning: ':' not found in line '{line}'. Skipping...")
            continue

        flags.append(line[:flag_index].strip())  # Extract flag
        outputs.append(line[flag_index + 1:].strip())  # Extract output

    return flags, outputs

def print_outputs(outputs):
    """Print the list of outputs."""
    print(outputs)

def main():
    # Check if the input file is provided
    if len(sys.argv) < 2:
        print("Usage: python script.py <input_file>")
        return

    filename = sys.argv[1]  # Get the filename from command-line arguments

    # Read the contents of the file
    lines = read_file_contents(filename)
    if lines is None:
        return

    # Extract flags and outputs from each line
    flags, outputs = extract_flags_and_outputs(lines)

    # Print the list of outputs
    print_outputs(outputs)

if __name__ == "__main__":
    main()
