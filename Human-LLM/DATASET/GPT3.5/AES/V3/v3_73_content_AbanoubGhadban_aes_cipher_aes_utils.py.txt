# Import necessary modules
import aes_constants as const
from data_utils import *

# Substitute bytes using S-box
def sub_bytes(bytes: list) -> list:
    return [const.S_BOX[b] for b in bytes]

# Invert substitute bytes using inverse S-box
def inv_sub_bytes(bytes: list) -> list:
    return [const.S_BOX_INV[b] for b in bytes]

# Perform XOR operation on two matrices
def xor_matrix(m1: list, m2: list) -> list:
    return [i1 ^ i2 for i1, i2 in zip(m1, m2)]

# Perform shift rows operation
def shift_rows(state: list) -> list:
    rows = [state[i::4] for i in range(4)]
    return [rows[r][c] for c in range(4) for r in range(4)]

# Perform inverse shift rows operation
def inv_shift_rows(state: list) -> list:
    rows = [state[i::4] for i in range(4)]
    return [rows[r][-i:] + rows[r][:-i] for c in range(4) for r in range(4)]

# Mix columns
def mix_columns(state: list) -> list:
    new_state = []
    for i in range(16):
        r = const.MIX_COLUMNS_MTX[(i % 4) * 4:(i % 4) * 4 + 4]
        c = state[i - i % 4:i - i % 4 + 4]
        result = 0
        for j in range(4):
            result ^= c[j] if r[j] == 1 else const.GALOIS_MULTIP[r[j]][c[j]]
        new_state.append(result)
    return new_state

# Perform inverse mix columns operation
def inv_mix_columns(state: list) -> list:
    new_state = []
    for i in range(16):
        r = const.INV_MIX_COLUMNS_MTX[(i % 4) * 4:(i % 4) * 4 + 4]
        c = state[i - i % 4:i - i % 4 + 4]
        result = 0
        for j in range(4):
            result ^= c[j] if r[j] == 1 else const.GALOIS_MULTIP[r[j]][c[j]]
        new_state.append(result)
    return new_state

# Generate round keys
def generate_round_keys(hexa_key: str) -> list:
    assert len(hexa_key) == 32, "Cipher key must be 128 bytes length"
    aes_key = hex_to_dec(hexa_key)
    keys = [aes_key]
    for i in range(10):
        prev_key = keys[i]
        cur_key = []
        rcon = [const.RCON[i], 0, 0, 0]
        row0 = sub_bytes(prev_key[12:] + prev_key[:12][1:])
        cur_key.extend(xor_matrix(xor_matrix(prev_key[:4], row0), rcon))
        for j in range(3):
            cur_row = cur_key[j * 4:j * 4 + 4]
            prev_row = prev_key[j * 4 + 4:j * 4 + 8]
            cur_key.extend(xor_matrix(prev_row, cur_row))
        keys.append(cur_key)
    return keys

# Encrypt a block
def encrypt_block(hexa_block: str, hexa_key: str) -> str:
    keys = generate_round_keys(hexa_key)
    block = hex_to_dec(hexa_block)
    block = xor_matrix(block, keys[0])
    for r in range(10):
        block = sub_bytes(block)
        block = shift_rows(block)
        if r != 9:
            block = mix_columns(block)
        block = xor_matrix(block, keys[r + 1])
    return dec_to_hex(block)

# Decrypt a block
def decrypt_block(hexa_block: str, hexa_key: str) -> str:
    keys = generate_round_keys(hexa_key)
    keys.reverse()
    block = hex_to_dec(hexa_block)
    block = xor_matrix(block, keys[0])
    for r in range(10):
        block = inv_sub_bytes(block)
        block = inv_shift_rows(block)
        if r != 9:
            block = inv_mix_columns(block)
            block = xor_matrix(block, inv_mix_columns(keys[r + 1]))
        else:
            block = xor_matrix(block, keys[r + 1])
    return dec_to_hex(block)
