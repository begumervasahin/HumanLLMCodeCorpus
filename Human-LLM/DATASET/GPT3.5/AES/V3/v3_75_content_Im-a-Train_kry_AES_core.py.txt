# Importing the keyExpand function from the key_schedule module
from src.key_schedule import keyExpand

# Defining the main function
def main(key):
    # Generating the expanded key
    expanded_key = keyExpand(key, key_size=4, block_size=4, rounds=10)

    # Printing the expanded key
    for i, word in enumerate(expanded_key):
        print(f"Round {i+1}: {hex(word)}")

# Checking if the script is being run as the main program
if __name__ == '__main__':
    # Providing a sample key value and calling the main function
    main(0x2b7e151628aed2a6abf7158809cf4f3c)
