import torch
from torch.utils.data import DataLoader
import torchvision.transforms as transforms
import os
from torch.autograd import Variable
import torch.nn as nn
import torch.optim as optim
from datetime import datetime
from tensorboardX import SummaryWriter
from torchvision import models
from data.dataset import DogCat  # Assuming you have a custom dataset class
from utils.utils import validate, show_confMat  # Assuming you have these utility functions

# Define classes and parameters
classes_name = ['high', 'low']
batch_size = 128
lr_init = 0.001
max_epoch = 20
root = '../../Smith/Data/AVA/images/'
result_dir = 'results/logs/'

# Define data loaders
train_data = DogCat(root, train=True)
train_dataloader = DataLoader(train_data, batch_size, shuffle=True, num_workers=4)
val_data = DogCat(root, train=False)
val_dataloader = DataLoader(val_data, batch_size, shuffle=False, num_workers=4)
test_data = DogCat(root, test=True)
test_dataloader = DataLoader(test_data, batch_size, shuffle=False, num_workers=4)

# Create result directory
now_time = datetime.now()
time_str = datetime.strftime(now_time, '%m-%d_%H-%M-%S')        
log_dir = os.path.join(result_dir, time_str)
os.makedirs(log_dir, exist_ok=True)
writer = SummaryWriter(log_dir=log_dir)

# Load pre-trained ResNet model
net = models.resnet34(pretrained=True)
net.fc = nn.Linear(512, 2)  # Modify the fully connected layer for binary classification
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
net.to(device)

# Define loss function, optimizer, and learning rate scheduler  
criterion = nn.CrossEntropyLoss()
optimizer = optim.SGD(net.parameters(), lr=lr_init, momentum=0.9, dampening=0.1)
scheduler = torch.optim.lr_scheduler.StepLR(optimizer, step_size=50, gamma=0.1)

# Training loop
for epoch in range(max_epoch):
    loss_sigma = 0.0
    correct = 0.0
    total = 0.0
    scheduler.step()

    # Training phase
    net.train()
    for i, (inputs, labels) in enumerate(train_dataloader):
        inputs, labels = inputs.to(device), labels.to(device)   
        optimizer.zero_grad()
        outputs = net(inputs)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()

        _, predicted = torch.max(outputs.data, 1)
        total += labels.size(0)
        correct += (predicted == labels).sum().item()
        loss_sigma += loss.item()

        if i % 10 == 9:
            loss_avg = loss_sigma / 10
            loss_sigma = 0.0
            print(f"Training: Epoch[{epoch + 1:0>3}/{max_epoch:0>3}] "
                  f"Iteration[{i + 1:0>3}/{len(train_dataloader):0>3}] "
                  f"Loss: {loss_avg:.4f} Acc: {correct / total:.2%}")
            writer.add_scalars('Loss_group', {'train_loss': loss_avg}, epoch)
            writer.add_scalar('learning rate', scheduler.get_lr()[0], epoch)
            writer.add_scalars('Accuracy_group', {'train_acc': correct / total}, epoch)

    # Validation phase
    if epoch % 2 == 0:
        loss_sigma = 0.0
        conf_mat = torch.zeros(len(classes_name), len(classes_name))
        net.eval()
        with torch.no_grad():
            for images, labels in val_dataloader:
                images, labels = images.to(device), labels.to(device)
                outputs = net(images)
                loss = criterion(outputs, labels)
                loss_sigma += loss.item()
                _, predicted = torch.max(outputs, 1)
                for label, pred in zip(labels, predicted):
                    conf_mat[label, pred] += 1

        accuracy = conf_mat.diag().sum() / conf_mat.sum()
        print(f'Validation set Accuracy: {accuracy:.2%}')
        writer.add_scalars('Loss_group', {'valid_loss': loss_sigma / len(val_dataloader)}, epoch)
        writer.add_scalars('Accuracy_group', {'valid_acc': accuracy}, epoch)

# Testing phase
net.eval()
with torch.no_grad():
    for i, (inputs, labels) in enumerate(test_dataloader):
        inputs, labels = inputs.to(device), labels.to(device)
        outputs = net(inputs)
        loss = criterion(outputs, labels)
        _, predicted = torch.max(outputs, 1)
        total += labels.size(0)
        correct += (predicted == labels).sum().item()
        loss_sigma += loss.item()

        if i % 10 == 9:
            loss_avg = loss_sigma / 10
            print(f"Testing: Iteration[{i + 1:0>3}/{len(test_dataloader):0>3}] "
                  f"Loss: {loss_avg:.4f} Acc: {correct / total:.2%}")
            writer.add_scalars('Loss_group', {'test_loss': loss_avg}, i)
            writer.add_scalars('Accuracy_group', {'test_acc': correct / total}, i)

# Save the trained model
net_save_path = os.path.join(log_dir, 'net_params.pkl')
torch.save(net.state_dict(), net_save_path)

conf_mat_valid, valid_acc = validate(net, val_dataloader, 'valid', classes_name)
