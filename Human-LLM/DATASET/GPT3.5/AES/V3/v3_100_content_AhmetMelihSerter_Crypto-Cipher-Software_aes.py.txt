import base64
import hashlib
from Crypto import Random
from Crypto.Cipher import AES

class AESCipher:
    def __init__(self, key):
        # Initialization of AESCipher object
        self.block_size = 32  # Block size for AES encryption
        self.key = hashlib.sha256(key.encode()).digest()  # Hashing the key for security

    def encrypt(self, message):
        # Encrypts a message using AES encryption
        encoded_message = self._encode_message(message)  # Encodes the message
        padded_message = self._pad_message(encoded_message)  # Pads the message
        initialization_vector = self._generate_initialization_vector()  # Generates a random initialization vector
        cipher = AES.new(self.key, AES.MODE_CBC, initialization_vector)  # Creates AES cipher object
        encrypted_message = self._encrypt_message(cipher, padded_message, initialization_vector)  # Encrypts the message
        return encrypted_message

    def decrypt(self, encrypted_message):
        # Decrypts an encrypted message using AES decryption
        decoded_message = base64.b64decode(encrypted_message)  # Decodes the encrypted message
        initialization_vector = decoded_message[:AES.block_size]  # Retrieves the initialization vector
        cipher = AES.new(self.key, AES.MODE_CBC, initialization_vector)  # Creates AES cipher object
        decrypted_message = self._decrypt_message(cipher, decoded_message, initialization_vector)  # Decrypts the message
        return decrypted_message

    def _encode_message(self, message):
        # Encodes the message using CP857 encoding and base64
        encoded_message = base64.b64encode(message.encode("cp857"))
        return encoded_message

    def _pad_message(self, message):
        # Pads the message to make its length a multiple of the block size
        padding_length = self.block_size - len(message) % self.block_size
        padded_message = message + padding_length * chr(padding_length)
        return padded_message

    def _generate_initialization_vector(self):
        # Generates a random initialization vector
        initialization_vector = Random.new().read(AES.block_size)
        return initialization_vector

    def _encrypt_message(self, cipher, message, initialization_vector):
        # Encrypts the message using the cipher object
        encrypted_message = cipher.encrypt(message)
        return base64.b64encode(initialization_vector + encrypted_message).decode("cp857")

    def _decrypt_message(self, cipher, message, initialization_vector):
        # Decrypts the message using the cipher object
        decrypted_message = cipher.decrypt(message[AES.block_size:])
        return base64.b64decode(decrypted_message).decode("cp857").rstrip(chr(decrypted_message[-1]))

    message = "Hello, world!"   # Message to encrypt

    encrypted_message = cipher.encrypt(message)
    print("Encrypted message:", encrypted_message)

    decrypted_message = cipher.decrypt(encrypted_message)
    print("Decrypted message:", decrypted_message)
