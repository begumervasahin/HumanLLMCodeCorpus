from Crypto.Cipher import AES
from Crypto.Util.Padding import pad, unpad
import base64

class AESCipher:
    def __init__(self, key):
        """Initialize the cipher with a key."""
        # The key is encoded to bytes if it is not already and ensured to meet AES key size requirements.
        self.key = self._ensure_key_length(key.encode('utf-8'))

    def encrypt(self, message):
        """Encrypt the message."""
        # The message is encoded to bytes.
        message_bytes = message.encode('utf-8')

        # AES cipher is created in CBC mode with a new random IV.
        cipher = AES.new(self.key, AES.MODE_CBC)

        # The message is encrypted and then both the IV and ciphertext are encoded in base64 for safe transport.
        encrypted_bytes = cipher.encrypt(pad(message_bytes, AES.block_size))
        iv_encoded = base64.b64encode(cipher.iv).decode('utf-8')
        encrypted_encoded = base64.b64encode(encrypted_bytes).decode('utf-8')

        # The IV and encrypted message are concatenated and returned.
        return iv_encoded + encrypted_encoded

    def decrypt(self, encrypted_message):
        """Decrypt an encrypted message."""
        # The IV and ciphertext are extracted and decoded from base64.
        iv = base64.b64decode(encrypted_message[:24])
        encrypted_bytes = base64.b64decode(encrypted_message[24:])

        # A new cipher is created with the extracted IV for decryption.
        cipher = AES.new(self.key, AES.MODE_CBC, iv)

        # The ciphertext is decrypted, unpadded, and decoded to a string.
        decrypted_bytes = unpad(cipher.decrypt(encrypted_bytes), AES.block_size)
        return decrypted_bytes.decode('utf-8')

    @staticmethod
    def _ensure_key_length(key, desired_length=16):
        """Ensure the key meets AES requirements by padding or truncating."""
        # Adjust the key to the desired length for AES-128.
        return pad(key, desired_length)[:desired_length]

# Example usage demonstrating the process.
def main():
    key = "Thats my Kung Fu"
    message = "Two One Nine Two"

    aes_cipher = AESCipher(key)
    encrypted_message = aes_cipher.encrypt(message)

    print("Message:\t", message)
    print("Encrypted:\t", encrypted_message)

    # Demonstrate decryption as well.
    decrypted_message = aes_cipher.decrypt(encrypted_message)
    print("Decrypted:\t", decrypted_message)

if __name__ == "__main__":
    main()
