import os
import torch
from torch.utils.data import DataLoader
from random import randint
from rdkit import Chem, RDLogger
from src.datareader import SMILESReader, OnehotDecoder
from train import get_args, load_autoencoder

def prepare_data_loaders(args, use_cuda):
    """Prepare the DataLoader for both training and validation datasets."""
    base_path = os.path.dirname(__file__)
    dataset_file = "prior_trainingset_DRD2_actives_removed.smi"
    alphabet_file = "alphabet.json"
    
    dataset_path = os.path.join(base_path, "data", dataset_file)
    alphabet_path = os.path.join(base_path, "data", alphabet_file)

    # Split dataset into training and validation subsets
    training_data = SMILESReader(dataset_path, alphabet_path, subset=(0, 1200001))
    validation_data = SMILESReader(dataset_path, alphabet_path, subset=(1200001, None))

    # Setup DataLoader for both subsets
    train_loader = DataLoader(training_data, batch_size=args.batch_size, shuffle=True,
                              num_workers=4, drop_last=True, pin_memory=use_cuda)
    val_loader = DataLoader(validation_data, batch_size=args.batch_size, shuffle=False,
                            num_workers=4, drop_last=True, pin_memory=use_cuda)
    return train_loader, val_loader

def validate(encoded_mol, decoder, autoencoder, reference_smi=None, ground_truth=None, repeat=100, batch_size=1):
    """Validate a molecule's encoding by attempting to decode it back to the original SMILES."""
    # Replicate the encoded molecule for decoding
    encoded_batch = torch.stack([encoded_mol.squeeze() for _ in range(batch_size)])
    ground_truth_batch = torch.stack([ground_truth.squeeze() for _ in range(batch_size)]) if ground_truth is not None else None

    invalid_count, identical_count = 0, 0
    unique_smiles = set()
    RDLogger.DisableLog('rdApp.*')  # Suppress RDKit warnings
    
    for _ in range(repeat):
        decoded = autoencoder.decode(encoded_batch, ground_truth_batch) if ground_truth is not None else autoencoder.decode(encoded_batch)
        smiles_list = decoder.decode(decoded.data)
        
        for smi in smiles_list:
            mol = Chem.MolFromSmiles(smi)
            if not mol:
                invalid_count += 1
                continue
            
            canonical_smi = Chem.MolToSmiles(mol)
            if reference_smi and canonical_smi == reference_smi:
                identical_count += 1
            elif not reference_smi:
                unique_smiles.add(canonical_smi)

    return invalid_count, identical_count if reference_smi else unique_smiles

def run_validation(autoencoder, decoder, dataset, use_cuda):
    """Perform validation on a randomly selected molecule from the dataset."""
    idx = randint(0, len(dataset) - 1)
    x, _ = dataset[idx]
    if use_cuda:
        x = x.cuda()
    
    original_smi = decoder.decode(x.unsqueeze(0))[0]
    encoded_smi = autoencoder.encode(x.unsqueeze(0))
    noise = torch.randn_like(encoded_smi)

    # Validate reconstruction
    print(f"Validating reconstruction for: {original_smi}")
    invalid, identical = validate(encoded_smi, decoder, autoencoder, reference_smi=original_smi, ground_truth=dataset.onehot_encoder(original_smi), repeat=100, batch_size=1)
    print(f"Identical: {identical}, Invalid: {invalid}")

    # Generate new molecules from encoded representation
    print("Generating new molecules from encoded representation")
    invalid, unique_mols = validate(encoded_smi, decoder, autoencoder, repeat=100, batch_size=1)
    print(f"Unique molecules generated: {len(unique_mols)}, Invalid: {invalid}")

    # Generate molecules from random noise
    print("Generating molecules from random noise")
    invalid, unique_mols_noise = validate(noise, decoder, autoencoder, repeat=100, batch_size=1)
    print(f"Unique molecules generated from noise: {len(unique_mols_noise)}, Invalid: {invalid}")

def main():
    args, use_cuda = get_args()
    train_loader, val_loader = prepare_data_loaders(args, use_cuda)
    autoencoder = load_autoencoder(args)
    decoder = OnehotDecoder(os.path.join(os.path.dirname(__file__), "data", "alphabet.json"))

    # Validate a random molecule from the validation dataset
    run_validation(autoencoder, decoder, val_loader.dataset, use_cuda)

if __name__ == "__main__":
    main()
