import argparse
import decimal
import timeit
from PIL import Image, ImageDraw, ImageFont
import numpy as np

def parse_command_line_arguments():
    """Set up and parse command-line arguments."""
    parser = argparse.ArgumentParser(description="Applies text and ghosting effects to images.")
    parser.add_argument('input_file', help='Path to the input image file')
    parser.add_argument('output_file', help='Path for the output image file')
    parser.add_argument('-t', '--text', help="Text to add to the image, formatted as 'Text[x,y]'")
    parser.add_argument('-s', '--text_size', type=int, default=50, help="Font size for the added text")
    parser.add_argument('-g', '--ghost_width', type=float, default=1.0, help="Width of the ghost effect as a percentage of the image width")
    parser.add_argument('-y', '--chrominance_variation', type=int, default=7, help="Maximum chrominance variation in YUV color space")
    parser.add_argument('-v', '--luminance_variation', type=int, default=11, help="Maximum luminance variation in YUV color space")
    parser.add_argument('-r', '--rescale', type=decimal.Decimal, help="Scale factor to rescale the image, expressed as a decimal")

    effect_method = parser.add_mutually_exclusive_group()
    effect_method.add_argument('--ghost-inline', action='store_true', help='Apply ghost effect using Python loops (slower)')
    effect_method.add_argument('--ghost-numpy', action='store_true', default=True, help='Apply ghost effect using NumPy (faster)')

    return parser.parse_args()

class ExecutionTimer:
    """Context manager for measuring the execution time of code blocks."""
    def __init__(self, description):
        self.description = description

    def __enter__(self):
        self.start_time = timeit.default_timer()

    def __exit__(self, *args):
        elapsed_time = timeit.default_timer() - self.start_time
        print(f"{self.description} took {elapsed_time:.2f} seconds.")

def add_text(image, text, position, font_size):
    """Draw text on an image at a specified position."""
    draw = ImageDraw.Draw(image)
    font_path = "arial.ttf"  # Modify as needed
    font = ImageFont.truetype(font_path, font_size)
    draw.text(position, text, fill=(255, 255, 255), font=font)

def main():
    args = parse_command_line_arguments()

    with ExecutionTimer("Image loading"):
        image = Image.open(args.input_file).convert("RGB")

    if args.rescale:
        with ExecutionTimer("Image rescaling"):
            scaled_size = [int(dim * args.rescale) for dim in image.size]
            image = image.resize(scaled_size)

    if args.text:
        with ExecutionTimer("Text adding"):
            # Extract position and actual text from the argument
            position = (100, 100)  # Default position, adjust based on user input if needed
            add_text(image, args.text, position, args.text_size)

    with ExecutionTimer("Ghost effect application"):
        method = 'numpy' if args.ghost_numpy else 'inline'
        apply_ghost_effect(image, method=method, ghost_width=args.ghost_width)

    with ExecutionTimer("Image saving"):
        image.save(args.output_file)

if __name__ == "__main__":
    main()
