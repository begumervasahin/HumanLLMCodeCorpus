import os
import numpy as np
import binascii
from Crypto.Cipher import AES

def pad_text(text):
    """
    Pad the text to make its length a multiple of 16 bytes.
    """
    while len(text) % 16 != 0:
        text += '\0'
    return text

def encrypt_file(key, file_name):
    """
    Encrypt a file using AES encryption with the provided key.
    """
    aes = AES.new(key, AES.MODE_ECB)
    with open(file_name, 'rb') as fr, open(file_name + '.en', 'wb') as fw:       
        block_size = 16  # 16 bytes for AES block size
        while True:
            buffer = fr.read(block_size)
            if not buffer:
                break
            buffer = buffer.hex()  # Convert binary to hex
            buffer = pad_text(buffer)
            buffer = aes.encrypt(buffer)
            fw.write(buffer)

def decrypt_file(key, file_name):
    """
    Decrypt a file using AES decryption with the provided key.
    """
    aes = AES.new(key, AES.MODE_ECB)
    with open(file_name, 'rb') as fr:
        file_name_decrypted = file_name.split(".en")[0] + '.de'
        with open(file_name_decrypted, 'wb') as fw:
            block_size = 16  # 16 bytes for AES block size
            while True:
                buffer = fr.read(block_size)
                if not buffer:
                    break
                buffer = aes.decrypt(buffer).decode()
                buffer = buffer.rstrip('\0')
                buffer = binascii.a2b_hex(buffer)  # Convert hex to binary       
                fw.write(buffer)

def save_key_to_file(key, file_name):
    """
    Save the encryption key to a file.
    """
    key_hex = key.hex()
    with open('keybox.' + file_name + '.key', 'w') as f:
        f.write(key_hex)

def main():
    """
    Main function to interactively encrypt or decrypt a file.
    """
    mode = input('Encrypt or Decrypt? (Enter "e" or "d")\n')
    if mode == 'e':
        file_name = input('Enter the file name:\n')
        key = np.random.bytes(16)  # Generate a random 16-byte key
        try:
            encrypt_file(key, file_name)
            save_key_to_file(key, file_name)
            print('Successfully encrypted!')
        except Exception as e:
            print('An error has occurred:', e)
            input('Press Enter to exit...')
    elif mode == 'd':
        file_name = input('Enter the file name:\n')
        key_file_list = [entry.name for entry in os.scandir(os.getcwd()) if entry.name.endswith('.key')]
        if key_file_list:
            key_file = key_file_list[0]
            with open(key_file, 'r') as f:
                key = binascii.a2b_hex(f.read().strip())
                if len(key) == 16:
                    try:
                        decrypt_file(key, file_name)
                        print('Successfully decrypted!')
                    except Exception as e:
                        print('An error has occurred:', e)
                    finally:
                        f.close()
                else:
                    print('A 16-byte key is necessary!')
        else:
            print('No key file found!')
    else:
        print('Incorrect input!')
    input('Press Enter to exit...')

if __name__ == '__main__':
    main()
