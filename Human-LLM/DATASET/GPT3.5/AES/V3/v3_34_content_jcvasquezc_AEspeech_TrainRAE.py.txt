import time
import torch
import numpy as np
import sys
import os
from SpecDatset import SpecDataset
from RAE import RAEn

def standardize(tensor, minval, maxval):
    return (tensor - minval) / (maxval - minval)

def destandardize(tensor, minval, maxval):
    return tensor * (maxval - minval) + minval

if __name__ == "__main__":
    if len(sys.argv) != 2:
        print("Usage: python TrainRAE.py <bottleneck_size>")
        sys.exit()

    # Paths and constants
    PATH = os.path.dirname(os.path.abspath(__file__))
    PATH_TRAIN = os.path.join(PATH, "../data/CIEMPIESS_train_melspec/")
    PATH_TEST = os.path.join(PATH, "../data/CIEMPIESS_test_melspec/")
    BATCH_SIZE = 16
    NUM_WORKERS = 0
    BOTTLE_SIZE = int(sys.argv[1])
    LR = 0.001
    N_EPOCHS = 50
    MIN_SCALER = -50.527256
    MAX_SCALER = 6.8561997

    # Dataset and DataLoader setup
    train_dataset = SpecDataset(PATH_TRAIN)
    test_dataset = SpecDataset(PATH_TEST)

    train_loader = torch.utils.data.DataLoader(train_dataset, batch_size=BATCH_SIZE, drop_last=True, num_workers=NUM_WORKERS)
    test_loader = torch.utils.data.DataLoader(test_dataset, batch_size=BATCH_SIZE, drop_last=True, num_workers=NUM_WORKERS)

    # Model, criterion, and optimizer setup
    model = RAEn(BOTTLE_SIZE)
    criterion = torch.nn.MSELoss()
    optimizer = torch.optim.Adam(model.parameters(), lr=LR)

    # CUDA setup
    device = torch.device("cuda") if torch.cuda.is_available() else torch.device("cpu")
    model.to(device)

    # Training loop
    valid_loss_min = np.Inf
    for epoch in range(N_EPOCHS):
        start_time = time.time()
        train_loss = 0.0
        valid_loss = 0.0

        model.train()
        for data in train_loader:
            optimizer.zero_grad()
            data = standardize(data, MIN_SCALER, MAX_SCALER).float().to(device)
            data_out, _ = model(data)
            loss = criterion(data_out, data)
            loss.backward()
            optimizer.step()
            train_loss += loss.item() * data.size(0)

        model.eval()
        for data_val in test_loader:
            data_val = standardize(data_val, MIN_SCALER, MAX_SCALER).float().to(device)
            data_val_out, _ = model(data_val)
            loss = criterion(data_val_out, data_val)
            valid_loss += loss.item() * data.size(0)

        train_loss = train_loss / len(train_loader.dataset)
        valid_loss = valid_loss / len(test_loader.dataset)

        print(f'Epoch: {epoch+1} \tTraining Loss: {train_loss:.6f} \tValidation Loss: {valid_loss:.6f} \tTime: {time.time() - start_time:.6f}')

        if valid_loss <= valid_loss_min:
            print(f'Validation loss decreased ({valid_loss_min:.6f} --> {valid_loss:.6f}). Saving model ...')
            torch.save(model.state_dict(), os.path.join(PATH, f"{BOTTLE_SIZE}_RAE.pt"))
            valid_loss_min = valid_loss

        with open(os.path.join(PATH, f"loss_{BOTTLE_SIZE}_RAE.csv"), "a") as f:
            f.write(f"{train_loss}, {valid_loss}\n")
