import os
import pytest

# Set up environment variables before starting the test session
def configure_environment_variables():
    os.environ.setdefault("AESARA_FLAGS", "")
    aesara_flags = os.environ["AESARA_FLAGS"]
    aesara_flags += ",warn__ignore_bug_before=all,on_opt_error=raise,on_shape_error=raise,cmodule__warn_no_version=True"
    os.environ["AESARA_FLAGS"] = aesara_flags
    os.environ["NUMBA_BOUNDSCHECK"] = "1"

# Add a command-line option to pytest for running slow tests
def add_slow_test_option(parser):
    parser.addoption(
        "--runslow", 
        action="store_true", 
        default=False, 
        help="Run tests marked as slow"
    )

# Configure pytest markers
def configure_pytest_markers(config):
    config.addinivalue_line("markers", "slow: Mark a test as slow to run")

# Modify test collection items based on command-line options
def modify_test_collection(config, items):
    if not config.getoption("--runslow"):
        mark_slow_tests_to_skip(items)

# Mark slow tests to be skipped if --runslow option is not provided
def mark_slow_tests_to_skip(items):
    for item in items:
        if "slow" in item.keywords:
            skip_slow_test(item)

# Helper function to skip slow test
def skip_slow_test(item):
    skip_reason = "Use --runslow option to run this test"
    item.add_marker(pytest.mark.skip(reason=skip_reason))

# Define an example test function with the slow marker
@pytest.mark.slow
def test_example_slow():
    assert True

# Define an example test function without the slow marker
def test_example_fast():
    assert True

if __name__ == "__main__":
    configure_environment_variables()
    add_slow_test_option(pytest)
    configure_pytest_markers(pytest.config)
    modify_test_collection(pytest.config, pytest.config.getoption("--runslow"), pytest.config.getoption("--runslow"))
    pytest.main()
