import numpy as np
from textwrap import wrap

# AES polynomial constant
POLY_CONSTANT = 0b100011011

# Mix Columns matrix and its inverse
MIX_COLUMNS_MATRIX = np.array([[2, 3, 1, 1],
                               [1, 2, 3, 1],
                               [1, 1, 2, 3],
                               [3, 1, 1, 2]])

INVERSE_MIX_COLUMNS_MATRIX = np.array([[14, 11, 13, 9],
                                       [9, 14, 11, 13],
                                       [13, 9, 14, 11],
                                       [11, 13, 9, 14]])

def hex_string_to_int(hex_string):
    return int(hex_string, 16)

def int_to_hex_string(integer):
    return format(integer, '02x')

class AESMatrix:
    """
    Represents a 4x4 matrix used in AES encryption and decryption operations.
    """
    def __init__(self, data: str):
        """
        Initializes the matrix from a 16-byte input string.
        """
        self.matrix = np.empty([4, 4], dtype='U2')
        data_bytes = wrap(data, 2)
        for i, byte in enumerate(data_bytes):
            self.matrix[i % 4][i // 4] = byte

    def shift_rows(self, inverse=False):
        """
        Performs ShiftRows operation on the matrix.
        """
        for i, row in enumerate(self.matrix):
            shift_amount = i if inverse else -i
            self.matrix[i] = np.roll(row, shift_amount)

    def mix_columns(self, inverse=False):
        """
        Performs MixColumns operation on the matrix.
        """
        mix_matrix = INVERSE_MIX_COLUMNS_MATRIX if inverse else MIX_COLUMNS_MATRIX
        matrix_as_num = np.array([[hex_string_to_int(x) for x in row] for row in self.matrix])
        result_matrix = np.zeros([4, 4])
        for i in range(4):
            for j in range(4):
                result_matrix[i, j] = self._multiply_and_mod(mix_matrix[i, j], matrix_as_num[:, j])
        self.matrix = np.array([[int_to_hex_string(int(x)) for x in row] for row in result_matrix])

    def flatten_cols(self):
        """
        Flattens the matrix column-wise into a string.
        """
        return ''.join(self.matrix.flatten())

    def flatten_rows(self):
        """
        Flattens the matrix row-wise into a string.
        """
        return ''.join(self.matrix.flatten(order='F'))

    def _multiply_and_mod(self, multiplier, column):
        """
        Multiplies each element in the column by the multiplier and performs modular reduction.
        """
        result = 0
        for elem in column:
            result ^= self._multiply(multiplier, elem)
        return self._mod_p(result)

    @staticmethod
    def _multiply(a, b):
        """
        Multiplies two numbers in GF(2^8).
        """
        if b == 1:
            return a
        elif b == 2:
            return a << 1
        elif b == 3:
            return (a << 1) ^ a
        elif b == 9:
            return (a << 3) ^ a
        elif b == 11:
            return (a << 3) ^ ((a << 1) ^ a)
        elif b == 13:
            return (a << 3) ^ (a << 2) ^ a
        elif b == 14:
            return (a << 3) ^ (a << 2) ^ (a << 1)
        else:
            raise ValueError("Invalid multiplier")

    @staticmethod
    def _mod_p(number):
        """
        Computes the modulus of a number with the AES polynomial constant.
        """
        while number.bit_length() > 8:
            number ^= POLY_CONSTANT << (number.bit_length() - 9)
        return number

# Example usage:
data = "a1b3d6..."
matrix = AESMatrix(data)
print("Original Matrix:")
print(matrix)

matrix.mix_columns()
print("\nAfter Mix Columns:")
print(matrix)

matrix.shift_rows()
print("\nAfter Shift Rows:")
print(matrix)

matrix_inv = AESMatrix(data)
matrix_inv.shift_rows(inverse=True)
print("\nAfter Inverse Shift Rows:")
print(matrix_inv)

matrix_inv.mix_columns(inverse=True)
print("\nAfter Inverse Mix Columns:")
print(matrix_inv)
