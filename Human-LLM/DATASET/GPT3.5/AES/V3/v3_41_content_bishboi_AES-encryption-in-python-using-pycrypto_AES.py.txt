import os
import base64
from Crypto.Cipher import AES
from Crypto import Random
from Crypto.Random import get_random_bytes
from Crypto.Util import Counter

BLOCK_SIZE = 16

def pad(s):
    padding = BLOCK_SIZE - len(s) % BLOCK_SIZE
    return s + bytes([padding]) * padding

def unpad(s):
    padding = s[-1]
    return s[:-padding]

class CipherAES:
    def __init__(self, key):
        self.key = key

    def encrypt(self, plaintext, iv=None):
        plaintext = pad(plaintext)
        cipher = AES.new(self.key, AES.MODE_CBC, iv)
        encrypted = cipher.encrypt(plaintext)
        if iv:
            return iv + encrypted
        return encrypted

    def decrypt(self, ciphertext, iv=None):
        if iv:
            ciphertext = ciphertext[len(iv):]
        cipher = AES.new(self.key, AES.MODE_CBC, iv)
        decrypted = cipher.decrypt(ciphertext)
        return unpad(decrypted)

def generate_key():
    return os.urandom(BLOCK_SIZE)

def generate_iv():
    return Random.new().read(BLOCK_SIZE)

def generate_plaintext():
    return get_random_bytes(BLOCK_SIZE)

def generate_counter():
    return Counter.new(128)

def main():
    plaintext = generate_plaintext().hex()
    key = generate_key()
    iv = generate_iv()

    print('plaintext:', plaintext)
    print('iv:', iv.hex())
    print('key:', key.hex())

    cbc_cipher = CipherAES(key)
    ciphertext = cbc_cipher.encrypt(plaintext, iv)
    print('encrypted in CBC:', base64.b64encode(ciphertext).decode())        
    decrypted_text = cbc_cipher.decrypt(ciphertext, iv)
    print('decrypted in CBC:', decrypted_text)

    ecb_cipher = CipherAES(key)
    ciphertext = ecb_cipher.encrypt(plaintext)
    print('encrypted in ECB:', base64.b64encode(ciphertext).decode())        
    decrypted_text = ecb_cipher.decrypt(ciphertext)
    print('decrypted in ECB:', decrypted_text)

    ofb_cipher = CipherAES(key)
    ciphertext = ofb_cipher.encrypt(plaintext, iv)
    print('encrypted in OFB:', base64.b64encode(ciphertext).decode())        
    decrypted_text = ofb_cipher.decrypt(ciphertext, iv)
    print('decrypted in OFB:', decrypted_text)

    ctr_cipher = CipherAES(key)
    ciphertext = ctr_cipher.encrypt(plaintext, iv)
    print('encrypted in CTR:', base64.b64encode(ciphertext).decode())        
    decrypted_text = ctr_cipher.decrypt(ciphertext, iv)
    print('decrypted in CTR:', decrypted_text)

    print('CTR:', generate_counter())

if __name__ == "__main__":
    main()
