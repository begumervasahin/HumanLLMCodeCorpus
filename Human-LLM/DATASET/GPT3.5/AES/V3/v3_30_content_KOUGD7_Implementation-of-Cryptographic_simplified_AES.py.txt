import torch
from torch.utils.data import DataLoader
import torchvision.transforms as transforms
from torchvision import models
from torch.autograd import Variable
import os
from datetime import datetime
from tensorboardX import SummaryWriter
from data.dataset import DogCat  # Custom dataset class
from utils.utils import validate, show_confMat  # Utility functions

# Define classes and parameters
classes_name = ['high', 'low']
batch_size = 128
lr_init = 0.001
max_epoch = 20
root = '../../Smith/Data/AVA/images/'
result_dir = 'results/logs/'

# Data transforms
data_transform = transforms.Compose([
    transforms.Resize((224, 224)),
    transforms.ToTensor(),
])

# Data loaders
train_data = DogCat(root, train=True, transform=data_transform)
val_data = DogCat(root, train=False, transform=data_transform)
test_data = DogCat(root, test=True, transform=data_transform)

train_loader = DataLoader(train_data, batch_size=batch_size, shuffle=True, num_workers=4)
val_loader = DataLoader(val_data, batch_size=batch_size, shuffle=False, num_workers=4)
test_loader = DataLoader(test_data, batch_size=batch_size, shuffle=False, num_workers=4)

# Create result directory and summary writer
log_dir = os.path.join(result_dir, datetime.now().strftime('%m-%d_%H-%M-%S'))
os.makedirs(log_dir, exist_ok=True)
writer = SummaryWriter(log_dir=log_dir)

# Load pre-trained ResNet model
net = models.resnet34(pretrained=True)
net.fc = torch.nn.Linear(512, len(classes_name))  # Modify fully connected layer for binary classification
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
net.to(device)

# Loss function, optimizer, and scheduler
criterion = torch.nn.CrossEntropyLoss()
optimizer = torch.optim.SGD(net.parameters(), lr=lr_init, momentum=0.9, dampening=0.1)
scheduler = torch.optim.lr_scheduler.StepLR(optimizer, step_size=50, gamma=0.1)

# Training loop
for epoch in range(max_epoch):
    # Variables to track loss and accuracy
    running_loss = 0.0
    correct = 0
    total = 0
    scheduler.step()  # Adjust learning rate

    # Training phase
    net.train()  # Set model to train mode
    for i, (inputs, labels) in enumerate(train_loader):
        inputs, labels = inputs.to(device), labels.to(device)  # Move data to device

        optimizer.zero_grad()  # Clear gradients
        outputs = net(inputs)  # Forward pass
        loss = criterion(outputs, labels)  # Calculate loss
        loss.backward()  # Backward pass
        optimizer.step()  # Update weights

        # Compute statistics
        _, predicted = torch.max(outputs, 1)
        total += labels.size(0)
        correct += (predicted == labels).sum().item()
        running_loss += loss.item()

        # Print and log statistics every 10 iterations
        if i % 10 == 9:
            avg_loss = running_loss / 10
            running_loss = 0.0
            print(f"Epoch [{epoch + 1}/{max_epoch}], Iteration [{i + 1}/{len(train_loader)}], "
                  f"Loss: {avg_loss:.4f}, Accuracy: {correct / total:.2%}")
            # Log training loss and accuracy
            writer.add_scalar('Training Loss', avg_loss, epoch * len(train_loader) + i)
            writer.add_scalar('Training Accuracy', correct / total, epoch * len(train_loader) + i)

    # Validation phase (every 2 epochs)
    if epoch % 2 == 0:
        val_loss = 0.0
        val_correct = 0
        val_total = 0
        net.eval()  # Set model to evaluation mode
        with torch.no_grad():
            for images, labels in val_loader:
                images, labels = images.to(device), labels.to(device)  # Move data to device
                outputs = net(images)  # Forward pass
                val_loss += criterion(outputs, labels).item()
                _, predicted = torch.max(outputs, 1)
                val_total += labels.size(0)
                val_correct += (predicted == labels).sum().item()

        val_accuracy = val_correct / val_total
        print(f'Validation Accuracy: {val_accuracy:.2%}')
        # Log validation loss and accuracy
        writer.add_scalar('Validation Loss', val_loss / len(val_loader), epoch)
        writer.add_scalar('Validation Accuracy', val_accuracy, epoch)

# Testing phase
test_loss = 0.0
test_correct = 0
test_total = 0
net.eval()  # Set model to evaluation mode
with torch.no_grad():
    for inputs, labels in test_loader:
        inputs, labels = inputs.to(device), labels.to(device)  # Move data to device
        outputs = net(inputs)  # Forward pass
        test_loss += criterion(outputs, labels).item()
        _, predicted = torch.max(outputs, 1)
        test_total += labels.size(0)
        test_correct += (predicted == labels).sum().item()

test_accuracy = test_correct / test_total
print(f'Test Accuracy: {test_accuracy:.2%}')
# Log test loss and accuracy
writer.add_scalar('Test Loss', test_loss / len(test_loader), 0)
writer.add_scalar('Test Accuracy', test_accuracy, 0)

# Save the trained model
net_save_path = os.path.join(log_dir, 'net_params.pkl')
torch.save(net.state_dict(), net_save_path)

# Validate on train, validation, and test sets
conf_mat_train, train_acc = validate(net, train_loader, 'train', classes_name)
conf_mat_valid, valid_acc = validate(net, val_loader, 'valid', classes_name)
conf_mat_test, test_acc = validate(net, test_loader, 'test', classes_name)

# Show confusion matrices
show_confMat(conf_mat_train, classes_name, 'train', log_dir)
show_confMat(conf_mat_valid, classes_name, 'valid', log_dir)
show_confMat(conf_mat_test, classes_name, 'test', log_dir)

print('Finished Training and Evaluation')
