def substitute_bytes(state, sbox):
    """
    Substitutes each byte in the state matrix with its corresponding value in the S-box.
    """
    for i in range(4):
        for j in range(4):
            row = state[i][j] >> 4
            col = state[i][j] & 0x0f
            state[i][j] = sbox[row][col]
    return state

def shift_rows(state):
    """
    Performs row shifting on the state matrix according to AES algorithm.
    """
    for i in range(1, 4):
        state[i] = state[i][i:] + state[i][:i]
    return state

def mix_columns(state, matrix):
    """
    Mixes the columns of the state matrix using matrix multiplication in Galois Field.
    """
    for j in range(4):
        temp_column = []
        for i in range(4):
            temp = 0
            for k in range(4):
                temp ^= galois_multiplication(matrix[i][k], state[k][j])
            temp_column.append(temp)
        for i in range(4):
            state[i][j] = temp_column[i]
    return state

def add_round_key(state, round_key):
    """
    XORs the state matrix with the round key.
    """
    for i in range(4):
        for j in range(4):
            state[i][j] ^= round_key[i][j]
    return state

def key_schedule(core_key, round, rcon):
    """
    Generates the round keys from the core key for a specific round.
    """
    round_constant = rcon[round]
    new_key = [[0 for _ in range(4)] for _ in range(4)]
    new_key[0] = [
        core_key[0][i] ^ substitute_bytes(rot_word(core_key[3], i), SBOX) ^ round_constant[i] for i in range(4)
    ]
    for i in range(1, 4):
        new_key[i] = [new_key[i - 1][j] ^ core_key[i][j] for j in range(4)]
    return new_key

def rot_word(word, shifts):
    """
    Performs circular byte left shift on a word.
    """
    return word[shifts:] + word[:shifts]

def aes_encrypt(plain_text, cipher_key, sbox, mix_matrix, round_constants):
    """
    Encrypts the given plaintext using AES algorithm.
    """
    state = plain_text
    state = add_round_key(state, cipher_key)
    for i in range(1, 10):
        state = substitute_bytes(state, sbox)
        state = shift_rows(state)
        state = mix_columns(state, mix_matrix)
        cipher_key = key_schedule(cipher_key, i, round_constants)
        state = add_round_key(state, cipher_key)
    state = substitute_bytes(state, sbox)
    state = shift_rows(state)
    cipher_key = key_schedule(cipher_key, 10, round_constants)
    state = add_round_key(state, cipher_key)
    return state

def aes_decrypt(cipher_text, cipher_key, sbox, mix_matrix, round_constants):
    """
    Decrypts the given ciphertext using AES algorithm.
    """
    state = cipher_text
    cipher_key = key_schedule(cipher_key, 10, round_constants)
    state = add_round_key(state, cipher_key)
    for i in range(9, 0, -1):
        state = shift_rows(state)
        state = substitute_bytes(state, sbox)
        cipher_key = key_schedule(cipher_key, i, round_constants)
        state = add_round_key(state, cipher_key)
        state = mix_columns(state, mix_matrix)
    state = shift_rows(state)
    state = substitute_bytes(state, sbox)
    state = add_round_key(state, cipher_key)
    return state

def main():
    plain_text = [
        [0x32, 0x88, 0x31, 0xe0],
        [0x43, 0x5a, 0x31, 0x37],
        [0xf6, 0x30, 0x98, 0x07],
        [0xa8, 0x8d, 0xa2, 0x34],
    ]

    cipher_key = [
        [0x2b, 0x28, 0xab, 0x09],
        [0x7e, 0xae, 0xf7, 0xcf],
        [0x15, 0xd2, 0x15, 0x4f],
        [0x16, 0xa6, 0x88, 0x3c],
    ]

    encrypted = aes_encrypt(plain_text, cipher_key, SBOX, MIX_MATRIX, ROUND_CONSTANTS)
    decrypted = aes_decrypt(encrypted, cipher_key, INVERSE_SBOX, MIX_MATRIX, ROUND_CONSTANTS)
