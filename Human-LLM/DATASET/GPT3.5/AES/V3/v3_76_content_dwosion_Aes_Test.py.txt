import Aes

def get_valid_input(prompt, validator):
    """
    Helper function to repeatedly prompt for input until it satisfies the provided validator function.
    """
    while True:
        user_input = input(prompt)
        if validator(user_input):
            return user_input
        else:
            print("Invalid input. Please try again.")

def validate_key_length(key_length):
    """
    Validator function to check if the provided key length is valid.
    """
    return key_length in ["16", "24", "32"]

def validate_key_length_and_match_length(key_length, key):
    """
    Validator function to check if the provided key length is valid and if the key matches the specified length.
    """
    if not validate_key_length(key_length):
        return False
    return len(key) == int(key_length)

def validate_text_length(text):
    """
    Validator function to check if the provided text length is 16 bytes.
    """
    return len(text) == 16

# Main loop for choosing between encryption and decryption
while True:
    operation = get_valid_input('Please choose cipher (c) or decipher (d): ', lambda x: x in ['c', 'd'])

    # Encryption operation
    if operation == 'c':
        plain_text = get_valid_input('Enter the plaintext string: ', validate_text_length)
        key_length = get_valid_input('Enter the key length (16, 24, 32): ', validate_key_length)
        cipher_key = get_valid_input('Enter the cipher key: ', lambda x: validate_key_length_and_match_length(key_length, x))

        # Perform encryption
        aes_cipher = Aes.AesCore(key_length, list(plain_text), list(cipher_key))
        res_cipher_text = aes_cipher.Cipher()
        aes_cipher.FormatPrint('Cipher text', res_cipher_text)

    # Decryption operation
    elif operation == 'd':
        cipher_text_string = get_valid_input('Enter the cipher text string: ', validate_text_length)
        key_length = get_valid_input('Enter the key length (16, 24, 32): ', validate_key_length)
        cipher_key = get_valid_input('Enter the cipher key: ', lambda x: validate_key_length_and_match_length(key_length, x))

        # Perform decryption
        aes_decipher = Aes.AesCore(key_length, cipher_text_string.split(), list(cipher_key))
        res_plain_text = aes_decipher.InvCipher()
        print("Plain text: {}".format(res_plain_text))

    else:
        print("Please input 'c' or 'd'!")
