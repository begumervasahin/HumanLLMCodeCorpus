import os
import platform
import sys
import subprocess
from base64 import b64encode, b64decode
from datetime import datetime
from Crypto.Cipher import AES
import binascii
import hashlib

class Cipher:
    os_platform = platform.system().lower()
    aes_modes = {"OFB": AES.MODE_OFB}
    pic_formats = ["jpg", "png", "tif"]
    actions = {'e': 'Encrypting', 'd': 'Decrypting', 't': 'Testing'}

    def v_bytes(self, string):
        return bytes(string, "utf-8") if sys.version_info[0] == 3 else string

    def v_input(self, string):
        return input(string) if sys.version_info[0] == 3 else raw_input(string)

    def clear_console(self):
        if self.os_platform.startswith("win"):
            os.system("cls")
        elif self.os_platform.startswith("linux"):
            os.system("history -c")
        else:
            print("Make sure to clear your command line history after running the script!!!")

    def terminal_size(self):
        if self.os_platform.startswith("linux"):
            os.system("resize -s 40 120")
            return 120
        elif self.os_platform.startswith("win"):
            os.system("mode con: cols=120 lines=40")
            return 120
        else:
            return 80

    def start(self, argv):
        if len(argv) == 1:
            print(self.usage)
            sys.exit()
        self.parse_args(argv)
        self.prepare_args()
        if self.AES_args["-a"] == "e":
            self.encrypt_file()
        elif self.AES_args["-a"] == "d":
            self.decrypt_file()
        elif self.AES_args["-a"] == "t":
            self.test_decryption()

    def parse_args(self, strings):
        self.AES_args = {}
        self.required_args = ["-a", "-f", "-u", "-k"]
        missing_args = set(self.required_args) - set(strings)
        if len(strings) < 9:
            print("\nMissing arguments!")
        if len(missing_args) > 0:
            for arg in missing_args:
                print("   You forgot to define %s" % (arg))
            sys.exit()
        for arg in self.required_args:
            try:
                self.AES_args[arg] = strings[strings.index(arg) + 1]
            except:
                print("   You have to define the value of %s" % (arg))
        if self.AES_args["-a"] == "e" and (len(self.AES_args["-u"]) < 5 or len(self.AES_args["-k"]) < 12):
            print("\n   The key must consist of 12 characters and the username must consists of 5 characters at least")
            sys.exit()
        if self.AES_args["-a"] == "e" and (len(self.AES_args["-k"]) < 12):
            print("\n   It is recommended to use a key and user name with 24 characters in total at least")
            go_on = self.v_input("\n   Are you sure that you want to continue? Type y/n for yes/no: ")
            if go_on == "n":
                sys.exit()
        self.clear_console()
        print("\n%s %s ..." % (self.actions[self.AES_args["-a"]], self.AES_args["-f"]))
        return self.AES_args

    def prepare_args(self):
        if self.AES_args["-a"] not in ["e", "d", "t"]:
            print("\n-a must be e for encryption or d for decryption")
            sys.exit()
        self.start_crypto = datetime.now()
        self.AES_args["-f"] = self.AES_args["-f"].replace("\\", "/")
        self.AES_args["-k"] = self.build_key(self.AES_args["-k"], 251)
        self.AES_args["-u"] = self.build_key(self.AES_args["-u"], 239)[11:27]       

    def build_key(self, key, modulus):
        salt, index = self.v_bytes(""), 0
        for i in 4*key:
            index += 1
            hex_position = hex(int(binascii.hexlify(self.v_bytes(i)), 16)**index % modulus)[2:4].replace("L", "")
            hex_position = "0" + hex_position if len(hex_position) == 1 else hex_position
            salt += binascii.unhexlify(hex_position)
        new_key = hashlib.pbkdf2_hmac("sha512", self.v_bytes(key), salt, 2746398)   

        return new_key[17:49]

    def encrypt_file(self):
        with open(self.AES_args["-f"], "rb") as read_file:
            plaintext = b64encode(read_file.read())
            plaintext += (16-len(plaintext)%16)*b'\x00'
        obj = AES.new(self.AES_args["-k"], self.aes_modes["OFB"], self.AES_args["-u"])
        ciphertext = obj.encrypt(plaintext)
        self.write_file(ciphertext, "_encrypted.")

    def decrypt_file(self):
        with open(self.AES_args["-f"], "rb") as read_file:
            ciphertext = read_file.read()
            if self.AES_args["-f"].split(".").pop() not in self.pic_formats:       
                hextext = binascii.hexlify(ciphertext).replace(b"0d0a", b"0a")     
                ciphertext = binascii.unhexlify(hextext)
        obj = AES.new(self.AES_args["-k"], self.aes_modes["OFB"], self.AES_args["-u"])
        try:
            decrypted_text = obj.decrypt(ciphertext)
            decrypted_text = b64decode(decrypted_text)
            self.write_file(decrypted_text, "_decrypted.")
        except:
            print("\n   Are you sure you chose the right action for the file? \n   %s seems not to be encrypted with this script.\n" % (self.AES_args["-f"]))

    def test_decryption(self):
        print("\ntesting the encryption-decryption-process without saving encryption to a file")
        with open(self.AES_args["-f"], "rb") as read_file:
            plaintext = b64encode(read_file.read())
            plaintext += (16-len(plaintext)%16)*b'\x00'
        obj = AES.new(self.AES_args["-k"], self.aes_modes["OFB"], self.AES_args["-u"])
        ciphertext = obj.encrypt(plaintext)
        obj2 = AES.new(self.AES_args["-k"], self.aes_modes["OFB"], self.AES_args["-u"])
        decrypted_text = obj2.decrypt(ciphertext)
        decoded_text = b64decode(decrypted_text)
        self.write_file(decoded_text, "_tested.")

    def write_file(self, text, direction):
        self.end_crypto = datetime.now()
        path_parts = self.AES_args["-f"].split(".")
        file_path = path_parts[0] + direction + path_parts[1]
        file_name = file_path.split("/").pop()
        changed_name = self.v_input("\nThe new file name will be %s. If you want another name, type it, else just press enter: " % (file_name))
        print(changed_name)
        new_file_name = file_name if changed_name == "" else changed_name.split(".")[0] + "." + file_name.split(".").pop()
        new_file_path = file_path.replace(file_name, new_file_name)
        with open(new_file_path, "wb") as new_file:
            new_file.write(text)
        self.clean_up(new_file_name, new_file_path)

    def clean_up(self, new_file_name, new_file_path):
        width = self.terminal_size()
        print("\n%s" % (width * "-"))
        print("%s saved in %s" % (new_file_name, new_file_path.replace(new_file_name, "")))
        print("\nComputation time: %s" % (str(self.end_crypto - self.start_crypto)[2:]))
        print("%s" % (width * "-"))
        delete = self.v_input("\nDo you want to delete the original file? Type y/n for yes/no: ")
        if delete == "y":
            os.remove(self.AES_args["-f"])
            print("Deleted %s\n" % (self.AES_args["-f"]))
        if self.os_platform.startswith("linux"):
            print("type 'history -c' to clear the current command line history")   

if __name__ == '__main__':
    Cipher().start(sys.argv)
