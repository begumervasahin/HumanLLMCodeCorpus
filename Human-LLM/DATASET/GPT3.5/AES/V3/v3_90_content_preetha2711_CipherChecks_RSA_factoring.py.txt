import fractions
import random

def calculate_s_and_t(e, d):
    s = 0
    t = e * d - 1
    while t % 2 == 0:
        s += 1
        t = t / 2
    return s, t

def generate_z_star_n(n):
    return [i for i in range(2, n + 1) if fractions.gcd(i, n) == 1]

def find_b(s, t, z_star_n, n):
    b = 0
    flag = True
    while flag:
        a = random.choice(z_star_n)
        b = pow(a, t, n)
        while pow(b, 2, n) != 1:
            b = (b * b) % n
        if (b % n != 1) and ((b + 1) % n != 0):
            flag = False
    return b

def factor(n, e, d):
    s, t = calculate_s_and_t(e, d)
    z_star_n = generate_z_star_n(n)
    b = find_b(s, t, z_star_n, n)
    p = fractions.gcd(b - 1, n)
    q = n // p
    return p, q
