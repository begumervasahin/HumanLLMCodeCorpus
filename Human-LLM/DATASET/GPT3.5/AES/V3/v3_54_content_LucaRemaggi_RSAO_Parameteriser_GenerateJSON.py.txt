import numpy as np
import json

class JSONGenerator:
    def __init__(self, param_early, param_late, name, max_early, filename, obj_type):
        self.param_early = param_early
        self.param_late = param_late
        self.name = name
        self.max_early = max_early
        self.filename = filename
        self.obj_type = obj_type
        self.library_entry = []

    def generate_object_vector_room_library(self):
        self.library_entry = {
            'name': self.name,
            'type': self.obj_type,
            'id': 0,
            'channels': 0,
            'priority': 0,
            'level': 1.0,
            'position': {
                'az': '{:.2f}'.format(self.param_early['Direct_sound']['doa'][0]),
                'el': '{:.2f}'.format(self.param_early['Direct_sound']['doa'][1]),
                'radius': '1.00'
            },
            'room': {'ereflect': []}
        }

        for idx_refl in range(1, self.max_early + 1):
            reflect_data = self.param_early['Reflection' + str(idx_refl)]
            reflect_entry = {
                'level': '{:.3e}'.format(reflect_data['level']),
                'delay': '{:.3e}'.format(reflect_data['toa']),
                'position': {
                    'az': '{:.1f}'.format(reflect_data['doa'][0]),
                    'el': '{:.1f}'.format(reflect_data['doa'][1]),
                    'refdist': '1.0'
                },
                'biquadsos': []
            }

            for filter_params in reflect_data['filtersos']:
                biquad = {
                    'b0': '{:.3e}'.format(filter_params[0]),
                    'b1': '{:.3e}'.format(filter_params[1]),
                    'b2': '{:.3e}'.format(filter_params[2]),
                    'a0': '{:.3e}'.format(filter_params[3]),
                    'a1': '{:.3e}'.format(filter_params[4]),
                    'a2': '{:.3e}'.format(filter_params[5])
                }
                reflect_entry['biquadsos'].append(biquad)

            self.library_entry['room']['ereflect'].append(reflect_entry)

        self.library_entry['room'].update({
            'lreverb': {
                'delay': '{:.3e}'.format(self.param_late['Late']['toa']),
                'level': self.format_list(self.param_late['Late']['level']),
                'attacktime': self.format_list(self.param_late['Late']['attacktimes']),
                'decayconst': self.format_list(self.param_late['Late']['expdecays'])
            }
        })

        return self

    def save_to_json(self):
        with open(self.filename, 'w') as outfile:
            json.dump(self.library_entry, outfile)

    @staticmethod
    def format_list(list_val):
        return ', '.join([f'{item:.2e}' for item in list_val.values()])

# Example usage:
if __name__ == "__main__":
    # Define parameters
    param_early = {...}
    param_late = {...}
    name = "Object1"
    max_early = 3
    filename = "output.json"
    obj_type = "Type1"

    # Create instance and generate JSON
    generator = JSONGenerator(param_early, param_late, name, max_early, filename, obj_type)
    generator.generate_object_vector_room_library().save_to_json()
