def finite_field_add(a, b):
    """Performs bitwise XOR (finite field addition) of two bytes."""
    return a ^ b

def multiply_by_x(byte):
    """Multiplies a byte by x (0x02) in GF(2^8), applying modulo operation."""
    return ((byte << 1) & 0xFF) ^ (0x1B if (byte & 0x80) else 0)

def finite_field_multiply(a, b):
    """Multiplies two bytes in the finite field GF(2^8)."""
    result = 0
    for _ in range(8):  # Process 8 bits
        if b & 1:  # If the lowest bit of b is set
            result ^= a
        is_high_bit_set = a & 0x80
        a = (a << 1) & 0xFF  # Left shift and keep byte size
        if is_high_bit_set:
            a ^= 0x1B  # Apply x^8 + x^4 + x^3 + x + 1 polynomial
        b >>= 1
    return result

# Example AES-128 encryption and decryption
cipher_key_128 = bytearray([0x2B, 0x7E, 0x15, 0x16, 0x28, 0xAE, 0xD2, 0xA6,
                            0xAB, 0xF7, 0x15, 0x88, 0x09, 0xCF, 0x4F, 0x3C])
input_bytes_128 = bytearray([0x32, 0x88, 0x31, 0xE0, 0x43, 0x5A, 0x31, 0x37,
                             0xF6, 0x30, 0x98, 0x07, 0xA8, 0x8D, 0xA2, 0x34])

# Perform key expansion (placeholder)
round_keys_128 = aes_key_expansion(cipher_key_128)

# Encrypt and decrypt
encrypted_128 = aes_cipher(input_bytes_128, round_keys_128, 10)
decrypted_128 = aes_inv_cipher(encrypted_128, round_keys_128, 10)
