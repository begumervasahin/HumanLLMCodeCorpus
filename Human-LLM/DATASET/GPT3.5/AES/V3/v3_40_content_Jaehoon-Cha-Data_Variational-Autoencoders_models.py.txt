import tensorflow as tf
from tensorflow.contrib import slim

class Autoencoder(object):
    def __init__(self, n_input, n_output, architecture):
        self.n_input = n_input
        self.n_output = n_output
        self.architecture = architecture
        self.x = tf.placeholder(tf.float32, shape=(None, 784), name="input")

    def encoder(self, inputs, name):
        with tf.name_scope(name):
            layer_1 = slim.layers.linear(inputs, num_outputs=self.architecture['h1'], activation_fn=tf.nn.softplus, scope='encoder1')
            layer_2 = slim.layers.linear(layer_1, num_outputs=self.architecture['h2'], activation_fn=tf.nn.softplus, scope='encoder2')
            return layer_2

    def to_latent(self, inputs, name):
        with tf.name_scope(name):
            encoder_out = slim.layers.linear(inputs, num_outputs=self.architecture['latent'])
            return encoder_out

    def decoder(self, inputs, name):
        with tf.name_scope(name):
            layer_1 = slim.layers.linear(inputs, num_outputs=self.architecture['h2'], activation_fn=tf.nn.softplus, scope='decoder1')
            layer_2 = slim.layers.linear(layer_1, num_outputs=self.architecture['h1'], activation_fn=tf.nn.softplus, scope='decoder2')
            decoder_out = slim.layers.linear(layer_2, num_outputs=self.n_output, activation_fn=tf.nn.sigmoid, scope='decoder_out')
            return decoder_out

    def forward(self):
        self.encoder_out = self.encoder(self.x, 'encoder')
        self.z = self.to_latent(self.encoder_out, 'latent')
        self.reconstr = self.decoder(self.z, 'decoder')
        with tf.name_scope('loss'):
            self.loss = tf.reduce_mean(-tf.reduce_sum(
                self.x * tf.math.log(tf.clip_by_value(self.reconstr, 1e-10, 1.0))
                + (1 - self.x) * tf.math.log(tf.clip_by_value(1 - self.reconstr, 1e-10, 1.0)), axis=1))
        tf.summary.scalar('loss', self.loss)
        return self.z, self.reconstr, self.loss

class VariationalAutoencoder(Autoencoder):
    def __init__(self, n_input, n_output, architecture):
        super().__init__(n_input, n_output, architecture)
        self.prior = DiagonalGaussian(mu=tf.zeros(1, 1), logvar=tf.zeros(1, 1))

    def to_latent(self, inputs, name):
        with tf.name_scope(name):
            mu = slim.layers.linear(inputs, num_outputs=self.architecture['latent'], scope='mean')
            logvar = slim.layers.linear(inputs, num_outputs=self.architecture['latent'], scope='logvar')
            return DiagonalGaussian(mu, logvar)

    def forward(self):
        self.encoder_out = self.encoder(self.x, 'encoder')
        self.encode_dist = self.to_latent(self.encoder_out, 'encoder_dist')
        with tf.name_scope('latent'):
            self.z = self.encode_dist.sample()
        self.reconstr = self.decoder(self.z, 'decoder')
        with tf.name_scope('loss'):
            self.reconst_loss = tf.reduce_mean(-tf.reduce_sum(
                self.x * tf.math.log(tf.clip_by_value(self.reconstr, 1e-10, 1.0))
                + (1 - self.x) * tf.math.log(tf.clip_by_value(1 - self.reconstr, 1e-10, 1.0)), axis=1))
            self.kl_loss = tf.reduce_mean(-self.prior.log_probability(self.z))
            self.lat_loss = tf.reduce_mean(self.encode_dist.log_probability(self.z))
            self.loss = tf.reduce_mean(self.reconst_loss + self.kl_loss + self.lat_loss)
        tf.summary.scalar('reconst_loss', self.reconst_loss)
        tf.summary.scalar('kl_loss', self.kl_loss)
        tf.summary.scalar('lat_loss', self.lat_loss)
        tf.summary.scalar('loss', self.loss)
        return self.z, self.reconstr, self.loss

class DiagonalGaussian(object):
    def __init__(self, mu, logvar):
        self.mu = mu
        self.logvar = logvar

    def sample(self):
        epsilon = tf.random_normal(tf.shape(self.mu))
        return self.mu + tf.exp(0.5 * self.logvar) * epsilon

    def log_probability(self, z):
        return -0.5 * tf.reduce_sum(self.logvar + tf.square(z - self.mu) / tf.exp(self.logvar), axis=1)
