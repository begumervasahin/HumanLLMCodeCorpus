from flask import Flask, request
from xor import xor
from SHA1 import sha1
from hashlib import sha256
from time import sleep

app = Flask(__name__)

BLOCK_SIZE = 64
SHA1_OUTPUT_SIZE = 20
SHA256_OUTPUT_SIZE = 32

def HMAC_sha1(key, message):
    key = prepare_key(key, BLOCK_SIZE)
    o_key_pad = xor(key, (b"\x5c" * BLOCK_SIZE))
    i_key_pad = xor(key, (b"\x36" * BLOCK_SIZE))
    inner_hash = sha1(i_key_pad + message.encode()).digest()
    outer_hash = sha1(o_key_pad + inner_hash).hexdigest()
    return outer_hash[:SHA1_OUTPUT_SIZE * 2]

def HMAC_sha256(key, message):
    key = prepare_key(key, BLOCK_SIZE)
    o_key_pad = xor(bytes.fromhex(key), (b"\x5c" * BLOCK_SIZE))
    i_key_pad = xor(bytes.fromhex(key), (b"\x36" * BLOCK_SIZE))
    inner_hash = sha256(i_key_pad + message.encode()).hexdigest().encode()
    outer_hash = sha256(o_key_pad + inner_hash).hexdigest()
    return outer_hash[:SHA256_OUTPUT_SIZE * 2]

def prepare_key(key, block_size):
    if len(key) > block_size:
        key = sha1(key.encode()).digest()
    elif len(key) < block_size:
        key += b'\x00' * (block_size - len(key))
    return key

def compare_good(s1, s2):
    return s1 == s2

def compare_bad(s1, s2):
    length = len(s1)
    s2 = s2 + "\x00" * (len(s1) - len(s2))
    for i in range(length):
        sleep(0.002)
        if s1[i] != s2[i]:
            return False
    return True

@app.route('/get', methods=['GET'])
def get():
    file = str(request.args.get('file'))
    return HMAC_sha1("myKeyIsL337", file) + "\n"

@app.route('/retrieve', methods=['GET'])
def check():
    file = str(request.args.get('file'))
    signature = str(request.args.get('signature'))
    hmac_good = HMAC_sha1("myKeyIsL337", file)

    if compare_bad(hmac_good, signature):
        return "Signature authenticated!\n", 200
    else:
        return "Message was modified, WARNING!\n", 500

if __name__ == "__main__":
    app.run()
