import gf

def do(state):
    """
    Apply the MixColumns operation to the AES state matrix.

    Args:
        state (bytes): The state matrix.

    Returns:
        bytes: The state matrix after the MixColumns operation.
    """
    columns = [state[i:i + 4] for i in range(0, len(state), 4)]
    mixed_columns = [mix_column(column) for column in columns]
    return bytes(b''.join(mixed_columns))

def inverse(m):
    """
    Compute the inverse of the MixColumns operation.

    Args:
        m (bytes): The input matrix.

    Returns:
        bytes: The inverse matrix.
    """
    columns = [m[i:i + 4] for i in range(0, len(m), 4)]
    inverse_columns = [inv(column) for column in columns]
    return bytes(b''.join(inverse_columns))

def mix_column(column):
    """
    Mix a single column of the AES state matrix.

    Args:
        column (bytes): The input column.

    Returns:
        bytes: The mixed column.
    """
    fixed_matrix = bytes([0x02, 0x03, 0x01, 0x01])
    mix_column_matrix = bytearray()

    for i in range(4):
        result = gf.add(gf.add(gf.add(
            gf.mul(fixed_matrix[0], column[0]),
            gf.mul(fixed_matrix[1], column[1])),
            gf.mul(fixed_matrix[2], column[2])),
            gf.mul(fixed_matrix[3], column[3]))

        mix_column_matrix.append(result)
        fixed_matrix = gf.circrotateright(fixed_matrix)

    return mix_column_matrix

def inv(column):
    """
    Compute the inverse of a single column.

    Args:
        column (bytes): The input column.

    Returns:
        bytes: The inverse column.
    """
    fixed_matrix = bytes([0x0e, 0x0b, 0x0d, 0x09])
    inverse_column = bytearray()

    for i in range(4):
        result = gf.add(gf.add(gf.add(
            gf.mul(fixed_matrix[0], column[0]),
            gf.mul(fixed_matrix[1], column[1])),
            gf.mul(fixed_matrix[2], column[2])),
            gf.mul(fixed_matrix[3], column[3]))

        inverse_column.append(result)
        fixed_matrix = gf.circrotateright(fixed_matrix)

    return inverse_column
