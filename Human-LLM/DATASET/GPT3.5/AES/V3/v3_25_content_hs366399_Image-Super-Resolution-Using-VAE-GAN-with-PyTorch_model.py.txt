import torch
import torch.nn as nn
import torchvision.transforms as transforms
from PIL import Image

class BasicBlock(nn.Module):
    def __init__(self, channels=128, stride=1, padding=1):
        super(BasicBlock, self).__init__()
        self.channels = channels
        self.stride = stride
        self.padding = padding

        self.conv1 = nn.Conv2d(channels, channels, kernel_size=3, stride=stride, padding=padding)
        self.bn1 = nn.BatchNorm2d(channels)
        self.activation1 = nn.PReLU()

        self.conv2 = nn.Conv2d(channels, channels, kernel_size=3, stride=stride, padding=padding)
        self.bn2 = nn.BatchNorm2d(channels)
        self.activation2 = nn.PReLU()

    def forward(self, x):
        identity = x
        x = self.activation1(self.bn1(self.conv1(x)))
        x = self.bn2(self.conv2(x)) + identity
        return self.activation2(x)

class PreEncoder(nn.Module):
    def __init__(self, channels=128):
        super(PreEncoder, self).__init__()
        self.channels = channels
        self.layers = nn.Sequential(
            nn.Conv2d(3, 32, kernel_size=3, stride=1, padding=1),
            nn.BatchNorm2d(32),
            nn.PReLU(),
            nn.Conv2d(32, 32, kernel_size=3, stride=1, padding=1),
            nn.BatchNorm2d(32),
            nn.PReLU(),
            nn.Conv2d(32, 64, kernel_size=3, stride=1, padding=1),
            nn.BatchNorm2d(64),
            nn.PReLU()
        )

    def forward(self, x):
        return self.layers(x)

class Encoder(nn.Module):
    def __init__(self, block):
        super(Encoder, self).__init__()
        self.block = block
        self.input_conv = nn.Sequential(
            nn.Conv2d(64, 128, kernel_size=3, padding=1),
            nn.BatchNorm2d(128),
            nn.PReLU(),
            nn.Conv2d(128, 128, kernel_size=3, padding=1),
            nn.BatchNorm2d(128),
            nn.PReLU()
        )
        self.layers = self._make_layers(3)
        self.downsample_convs = nn.ModuleList([
            nn.Sequential(nn.Conv2d(128, 128, kernel_size=5, stride=2), nn.BatchNorm2d(128), nn.PReLU())
            for _ in range(3)
        ])

    def _make_layers(self, num_layers):
        return nn.Sequential(*[self.block() for _ in range(num_layers)])

    def forward(self, x):
        x = self.input_conv(x)
        for layer, downsample_conv in zip(self.layers, self.downsample_convs):
            x = layer(x)
            x = downsample_conv(x)
        return x

class Decoder(nn.Module):
    def __init__(self, block):
        super(Decoder, self).__init__()
        self.block = block
        self.layers = self._make_layers(3)
        self.trans_convs = nn.ModuleList([
            nn.Sequential(nn.ConvTranspose2d(128, 128, kernel_size=4, stride=2), nn.BatchNorm2d(128), nn.PReLU())
            for _ in range(5)
        ])
        self.output_conv = nn.Sequential(
            nn.Conv2d(128, 64, kernel_size=2),
            nn.BatchNorm2d(64),
            nn.PReLU(),
            nn.Conv2d(64, 32, kernel_size=3, padding=1),
            nn.BatchNorm2d(32),
            nn.PReLU(),
            nn.Conv2d(32, 3, kernel_size=3, padding=1),
        )

    def _make_layers(self, num_layers):
        return nn.Sequential(*[self.block() for _ in range(num_layers)])

    def forward(self, x):
        for layer, trans_conv in zip(self.layers, self.trans_convs):
            x = trans_conv(x)
            x = layer(x)
        return self.output_conv(x)

class Discriminator(nn.Module):
    def __init__(self, block):
        super(Discriminator, self).__init__()
        self.block = block
        self.layers = self._make_layers(3)
        self.main_convs = nn.Sequential(
            nn.Conv2d(3, 32, kernel_size=3, stride=2),
            nn.BatchNorm2d(32),
            nn.PReLU(),
            nn.Conv2d(32, 128, kernel_size=3, stride=2),
            nn.BatchNorm2d(128),
            nn.PReLU(),
            nn.Conv2d(128, 128, kernel_size=5, stride=2, padding=2),
            nn.BatchNorm2d(128),
            nn.PReLU(),
            nn.Conv2d(128, 128, kernel_size=3, stride=2),
            nn.BatchNorm2d(128),
            nn.PReLU(),
            nn.Conv2d(128, 64, kernel_size=4, stride=2),
            nn.BatchNorm2d(64),
            nn.PReLU(),
            nn.Conv2d(64, 32, kernel_size=3, stride=2),
            nn.BatchNorm2d(32),
            nn.PReLU(),
            nn.Conv2d(32, 16, kernel_size=3, stride=1),
            nn.BatchNorm2d(16),
            nn.PReLU()
        )
        self.linear_layers = nn.Sequential(
            nn.Linear(256, 128),
            nn.BatchNorm1d(128),
            nn.PReLU(),
            nn.Linear(128, 10),
            nn.BatchNorm1d(10),
            nn.PReLU(),
            nn.Linear(10, 2),
        )

    def _make_layers(self, num_layers):
        return nn.Sequential(*[self.block() for _ in range(num_layers)])

    def forward(self, x):
        x = self.main_convs(x)
        x = x.view(x.size(0), -1)
        return self.linear_layers(x)

# Instantiate the models
pre_encoder = PreEncoder()
encoder = Encoder(BasicBlock)
decoder = Decoder(BasicBlock)
discriminator = Discriminator(BasicBlock)

# Print the models
print("Pre-Encoder:")
print(pre_encoder)

print("\nEncoder:")
print(encoder)

print("\nDecoder:")
print(decoder)

print("\nDiscriminator:")
print(discriminator)
