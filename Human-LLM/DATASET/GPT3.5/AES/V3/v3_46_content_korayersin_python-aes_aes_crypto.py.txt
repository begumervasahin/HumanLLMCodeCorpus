from Crypto import Random
from Crypto.Cipher import AES
import base64

class AESCrypto:
    def __init__(self, key, value_of):
        self.value_of = value_of[-6:]
        self.key = self._generate_key(key)
        self.BLOCK_SIZE = 16

    def _generate_key(self, key):
        def change_name(name, index, rate):
            while len(name) <= index:
                index -= len(name)
            prefix = name[:index]
            char = name[index]
            total = char * rate
            return prefix + total + name[index:]

        for i in range(3):
            defined = self.value_of[i * 2:i * 2 + 2]
            defined_index = int(defined[:1])
            defined_rate = int(defined[1:])
            key = change_name(key, defined_index, defined_rate) 

        if len(key) > 16:
            key = key[:16]
        elif len(key) < 16:
            key += "_" * (16 - len(key))
        return key

    def _pad(self, data):
        pad_length = self.BLOCK_SIZE - len(data) % self.BLOCK_SIZE
        return data + chr(pad_length) * pad_length

    def _unpad(self, data):
        return data[:-data[-1]]

    def encrypt(self, message):
        message = self._pad(message)
        iv = Random.new().read(AES.block_size)
        cipher = AES.new(self.key.encode(), AES.MODE_CBC, iv)   
        encrypted = cipher.encrypt(message.encode())
        return base64.b64encode(iv + encrypted).decode()        

    def decrypt(self, encrypted):
        encrypted = base64.b64decode(encrypted)
        iv = encrypted[:self.BLOCK_SIZE]
        cipher = AES.new(self.key.encode(), AES.MODE_CBC, iv)   
        decrypted = cipher.decrypt(encrypted[self.BLOCK_SIZE:]) 
        return self._unpad(decrypted).decode()

def main():
    key = "YourSecretKey"  # Replace with your secret key       
    value_of = "YourValueOf"  # Replace with your value_of      
    aes_crypto = AESCrypto(key, value_of)

    # Example usage
    message = "Hello, World!"
    encrypted = aes_crypto.encrypt(message)
    decrypted = aes_crypto.decrypt(encrypted)

    print("Original Message:", message)
    print("Encrypted Message:", encrypted)
    print("Decrypted Message:", decrypted)

if __name__ == "__main__":
    main()
