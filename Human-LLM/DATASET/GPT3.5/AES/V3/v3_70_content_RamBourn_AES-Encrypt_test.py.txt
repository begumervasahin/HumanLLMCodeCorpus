from Crypto.Cipher import AES
from binascii import b2a_hex, a2b_hex
import tkinter as tk
from tkinter import filedialog

# Function to open a file
def open_file():
    filename = filedialog.askopenfilename()
    if filename:
        file_in.delete('1.0', tk.END)
        file_in.insert(tk.END, filename)
        with open(filename, 'r') as file:
            input_in.delete('1.0', tk.END)
            input_in.insert(tk.END, file.read())

# Function to pad text to a multiple of 16 bytes
def pad_to_multiple_of_16(text):
    padding_length = 16 - (len(text.encode('utf-8')) % 16)
    if padding_length != 16:
        text += '\0' * padding_length
    return text.encode('utf-8')

# Function to encrypt text
def encrypt_text():
    key = key_entry.get().encode('utf-8')
    mode = AES.MODE_CBC
    iv = b'qqqqqqqqqqqqqqqq'
    text = input_in.get('1.0', tk.END).strip('\n')
    padded_text = pad_to_multiple_of_16(text)
    cipher = AES.new(key, mode, iv)
    cipher_text = cipher.encrypt(padded_text)
    encrypted_text = b2a_hex(cipher_text).decode('utf-8')

    encrypt_out.delete('1.0', tk.END)
    encrypt_out.insert(tk.END, encrypted_text)

    with open('encrypted.txt', 'w') as file:
        file.write(encrypted_text)

# Function to decrypt text
def decrypt_text():
    key = key_entry.get().encode('utf-8')
    mode = AES.MODE_CBC
    iv = b'qqqqqqqqqqqqqqqq'
    cipher_text = encrypt_out.get('1.0', tk.END).strip('\n')
    cipher = AES.new(key, mode, iv)
    decrypted_text = cipher.decrypt(a2b_hex(cipher_text)).decode('utf-8').rstrip('\0')

    decrypt_out.delete('1.0', tk.END)
    decrypt_out.insert(tk.END, decrypted_text)

# Create the main window
win = tk.Tk()
win.title('AES Encryption Tool')

# Labels
key_label = tk.Label(win, text='Key')
input_label = tk.Label(win, text='Input Text')
encrypted_label = tk.Label(win, text='Encrypted Text')
decrypted_label = tk.Label(win, text='Decrypted Text')

# Buttons
file_button = tk.Button(win, text='Open File', command=open_file)
encrypt_button = tk.Button(win, text='Encrypt', command=encrypt_text)
decrypt_button = tk.Button(win, text='Decrypt', command=decrypt_text)

# Text areas and entry
input_in = tk.Text(win, height=10)
file_in = tk.Text(win, height=1)
key_entry = tk.Entry(win)
encrypt_out = tk.Text(win, height=15)
decrypt_out = tk.Text(win, height=15)

# Layout
file_button.grid(row=0, column=0)
file_in.grid(row=0, column=1)

key_label.grid(row=1, column=0)
key_entry.grid(row=1, column=1, sticky=tk.E+tk.W)

input_label.grid(row=2, column=0, sticky=tk.N)
input_in.grid(row=2, column=1)

encrypt_button.grid(row=3, column=0, sticky=tk.N)
encrypt_out.grid(row=3, column=1)

decrypt_button.grid(row=4, column=0, sticky=tk.N)
decrypt_out.grid(row=4, column=1)

win.mainloop()
