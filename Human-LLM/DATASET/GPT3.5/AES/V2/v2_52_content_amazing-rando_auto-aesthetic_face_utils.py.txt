from PIL import Image
import numpy as np
import cv2
from random import choice, randint
from glob import glob

def calculate_bounds(rectangle, offset):
    """Calculate the upper-left and lower-right coordinates of a rectangle with an applied offset."""
    upper_left = (rectangle[0] + offset[0], rectangle[1] + offset[1])
    lower_right = (rectangle[0] + rectangle[2] + offset[2], rectangle[1] + rectangle[3] + offset[3])
    return upper_left, lower_right

def find_feature(clf, gray_image, image_size, offset, sensitivity=10):
    """Detect a feature within an image using a classifier and return its coordinates."""
    detected_features = clf.detectMultiScale(gray_image, 1.5, sensitivity)
    for feature in detected_features:
        upper_left, lower_right = calculate_bounds(feature, offset)
        if upper_left[0] < 0 or lower_right[0] < image_size // 2 or (lower_right[1] - upper_left[1]) > 300:
            continue
        return [upper_left[0], upper_left[1], lower_right[0] - upper_left[0], lower_right[1] - upper_left[1]]
    return None

def apply_drag_effect(image, rectangle):
    """Apply a 'dragging' distortion effect to a specified area of an image."""
    orientation = choice(["horizontal", "vertical"])
    if orientation == "vertical":
        vertical_line = rectangle[1] + rectangle[3] // 2 - 90
        image[:vertical_line, rectangle[0]:rectangle[0] + rectangle[2] - 90] = image[vertical_line, rectangle[0]:rectangle[0] + rectangle[2] - 90]
    else:
        horizontal_line = randint(rectangle[0] + 50, rectangle[0] + rectangle[2] - 160)
        direction = choice(["left", "right"])
        if direction == "left":
            image[rectangle[1]:rectangle[1] + rectangle[3], :horizontal_line] = image[rectangle[1]:rectangle[1] + rectangle[3], horizontal_line:horizontal_line + 1]
        else:
            image[rectangle[1]:rectangle[1] + rectangle[3], horizontal_line:] = image[rectangle[1]:rectangle[1] + rectangle[3], horizontal_line:horizontal_line + 1]
    return image

def apply_stagger_effect(image, rectangle):
    """Apply a 'staggering' distortion effect to simulate a glitch within a specified area."""
    stagger_amount = randint(1, 50)
    for strip_start in range(rectangle[1], rectangle[1] + rectangle[3], stagger_amount):
        strip_end = min(strip_start + stagger_amount, rectangle[1] + rectangle[3])
        offset = randint(-50, 50)
        image[strip_start:strip_end, rectangle[0] + offset:rectangle[0] + rectangle[2] + offset] = \
            image[strip_start:strip_end, rectangle[0]:rectangle[0] + rectangle[2]]
    return image

def apply_censor_bar(image, censor_bar_path, rectangle):
    """Overlay a censor bar image on a specified area of an image."""
    with Image.open(censor_bar_path) as censor_bar:
        censor_bar = censor_bar.resize((rectangle[2], rectangle[3]))
        image_pil = Image.fromarray(image).convert("RGBA")
        image_pil.paste(censor_bar, (rectangle[0], rectangle[1]), censor_bar)
    return np.array(image_pil)

def modify_image(image, rectangle, modification_type, censor_bar_dir=None):
    """Modify an image according to the specified type within a given rectangle area."""
    if rectangle:
        if modification_type == "cover" and censor_bar_dir:
            censor_bar_path = choice(glob(f"{censor_bar_dir}/*"))
            image = apply_censor_bar(image, censor_bar_path, rectangle)
        elif modification_type == "stagger":
            image = apply_stagger_effect(image, rectangle)
        elif modification_type == "drag":
            image = apply_drag_effect(image, rectangle)
    return image

# Example usage
if __name__ == "__main__":
    # Load the image and convert it to grayscale
    image_path = "path/to/your/image.jpg"
    image = cv2.imread(image_path)
    gray_image = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)

    # Load the classifier for feature detection
    classifier_path = 'path/to/your/haarcascade_frontalface_default.xml'
    classifier = cv2.CascadeClassifier(classifier_path)

    # Detect the feature and apply modifications
    rectangle = find_feature(classifier, gray_image, max(image.shape[:2]), [0, 0, 0, 0])
    modified_image = modify_image(image, rectangle, "drag", "path/to/censor_bars")

    # Save the modified image
    cv2.imwrite("modified_image.jpg", modified_image)
