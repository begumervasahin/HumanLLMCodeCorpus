import fractions
import random

def factor(n, e, d):
    # Calculate s and t
    s = 0
    t = e * d - 1
    while t % 2 == 0:
        s += 1
        t = t / 2

    # Generate Z_n*
    z_star_n = [i for i in range(2, n + 1) if fractions.gcd(i, n) == 1]

    # Find a suitable value for b
    b = 0
    flag = True
    while flag:
        a = random.choice(z_star_n)
        b = pow(a, t, n)

        # Check if b^2 mod n is not equal to 1
        while pow(b, 2, n) != 1:
            b = (b * b) % n

        # Check if b is not congruent to 1 or -1 mod n
        if (b % n != 1) and ((b + 1) % n != 0):
            flag = False

    # Find the factors p and q
    p = fractions.gcd(b - 1, n)
    q = n // p

    return p, q

# Test the factorization function
