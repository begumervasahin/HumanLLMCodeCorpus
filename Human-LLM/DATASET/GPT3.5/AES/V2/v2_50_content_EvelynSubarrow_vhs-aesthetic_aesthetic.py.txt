import argparse
import decimal
import timeit
from PIL import Image, ImageDraw, ImageFont
import numpy as np

def parse_args():
    """Parse command-line arguments."""
    parser = argparse.ArgumentParser(description="Applies text and ghosting effects to an image.")
    parser.add_argument('file', help='Input file path')
    parser.add_argument('outfile', help='Output file path')
    parser.add_argument("-t", "--text", help="Text to add to the image, formatted as '[x,y]Text'")
    parser.add_argument("-s", "--text-size", type=int, default=50, help="Font size for the added text")
    parser.add_argument("-g", "--ghost-width", type=float, default=1.0, help="Width of the ghost effect as a percentage of the image width")
    parser.add_argument("-y", "--chrominance-static", type=int, default=7, help="Maximum variation for chrominance in YUV color space")
    parser.add_argument("-v", "--luminance-static", type=int, default=11, help="Maximum variation for luminance in YUV color space")
    parser.add_argument("-r", "--rescale", type=decimal.Decimal, help="Rescale the image (as a decimal fraction of the original size)")

    action = parser.add_mutually_exclusive_group()
    action.add_argument('--ghost-il', action='store_true', help='Apply ghost effect using inline Python (slower)')
    action.add_argument('--ghost-np', action='store_true', default=True, help='Apply ghost effect using NumPy (faster)')

    return parser.parse_args()

class PrintTimer:
    """A simple timer class for profiling code sections."""
    def __init__(self, activity):
        self.activity = activity

    def __enter__(self):
        self.start = timeit.default_timer()

    def __exit__(self, *args):
        elapsed = timeit.default_timer() - self.start
        print(f"{self.activity} took {elapsed:.2f} seconds")

def add_text_to_image(image, text, position, font_size):
    """Adds specified text to the given image at the specified position."""
    draw = ImageDraw.Draw(image)
    font = ImageFont.truetype("arial.ttf", font_size)  # Consider changing the font path as necessary
    draw.text(position, text, font=font, fill=(255, 255, 255))

def main():
    args = parse_args()

    with PrintTimer("Loading image"):
        image = Image.open(args.file).convert("RGB")

    if args.rescale:
        with PrintTimer("Rescaling image"):
            new_size = tuple(int(dim * args.rescale) for dim in image.size)
            image = image.resize(new_size)

    if args.text:
        with PrintTimer("Adding text"):
            # Extract text and position from args.text
            # Placeholder: Assumes text format is directly usable, adjust parsing as needed
            add_text_to_image(image, args.text, (100, 100), args.text_size)  # Adjust position as needed

    with PrintTimer("Applying ghost effect"):
        method = 'np' if args.ghost_np else 'il'
        apply_ghost_effect(image, method=method, ghost_width=args.ghost_width)

    with PrintTimer("Saving image"):
        image.save(args.outfile)

if __name__ == "__main__":
    main()
