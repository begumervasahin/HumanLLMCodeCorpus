from Crypto.Cipher import AES
import binascii
import struct
import sys
from argparse import ArgumentParser, RawDescriptionHelpFormatter

class AESCipher:
    SECTOR_SIZE = 512

    def __init__(self, key):
        self.key = key

    def decrypt_file(self, input_file, output_file):
        self._transfer_data(input_file, output_file, self._decrypt_sector)

    def encrypt_file(self, input_file, output_file):
        self._transfer_data(input_file, output_file, self._encrypt_sector)

    def _transfer_data(self, input_file, output_file, transfer_func):
        sector_id = 0
        with open(input_file, "rb") as file_in:
            with open(output_file, "wb") as file_out:
                while True:
                    sector_data = file_in.read(AESCipher.SECTOR_SIZE)
                    if len(sector_data) == 0:
                        break
                    sector_data = sector_data.ljust(AESCipher.SECTOR_SIZE, b'\0')
                    file_out.write(transfer_func(sector_data, sector_id))
                    sector_id += 1

    def _decrypt_sector(self, sector_data, sector_id):
        iv = struct.pack("<I", sector_id) + b'\0' * 12
        cipher = AES.new(self.key, AES.MODE_CBC, iv)
        return cipher.decrypt(sector_data)

    def _encrypt_sector(self, sector_data, sector_id):
        iv = struct.pack("<I", sector_id) + b'\0' * 12
        cipher = AES.new(self.key, AES.MODE_CBC, iv)
        return cipher.encrypt(sector_data)

def main(argv=None):
    if argv is None:
        argv = sys.argv[1:]

    program_description = 'A Python implementation of cryptoloop using AES CBC'
    try:
        parser = ArgumentParser(description=program_description, formatter_class=RawDescriptionHelpFormatter)
        parser.add_argument("-e", "--encrypt", dest="encrypt", action="store_true",
                            help="Encrypt a file", default=False) 
        parser.add_argument("-d", "--decrypt", dest="decrypt", action="store_true",
                            help="Decrypt a file", default=False) 
        parser.add_argument(dest="key", help="AES 128-bit key in hexadecimal format", metavar="key", nargs=1)
        parser.add_argument(dest="input_file", help="Input file path", metavar="input", nargs=1)
        parser.add_argument(dest="output_file", help="Output file path", metavar="output", nargs=1)
        args = parser.parse_args(argv)

        encrypt = args.encrypt
        decrypt = args.decrypt
        key = args.key[0]
        input_file = args.input_file[0]
        output_file = args.output_file[0]

        if encrypt and decrypt:
            raise Exception('Choose either encrypt or decrypt but not both')

        aes = AESCipher(binascii.unhexlify(key))

        if decrypt:
            aes.decrypt_file(input_file, output_file)
        elif encrypt:
            aes.encrypt_file(input_file, output_file)

        return 0

    except KeyboardInterrupt:
        return 0
    except Exception as e:
        indent = len('cryptoloop') * " "
        sys.stderr.write('cryptoloop' + ": " + str(e) + "\n")   
        sys.stderr.write(indent + "  For help, use --help")      
        return 2

if __name__ == "__main__":
    sys.exit(main())
