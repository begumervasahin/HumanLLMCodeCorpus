from flying_toaster.toaster import Toaster
from flying_toaster.super_toaster import SuperToaster

def print_toaster(toaster):
    print('Model:', type(toaster).__name__)
    if isinstance(toaster, SuperToaster):
        print('Alias:', toaster.alias)
    print('Colour:', toaster.colour)
    print('Number of slots:', len(toaster.slots))

def get_toaster(toaster_type):
    try:
        slots = int(input('\nHow many slots does your toaster have?: '))
        colour = input('\nWhat colour is your toaster?: ')
        toaster = None
        if toaster_type == '1':
            toaster = Toaster(slots, colour)
        elif toaster_type == '2':
            alias = input('\nWhat is the alias of your super toaster?: ')
            toaster = SuperToaster(slots, alias, colour)        
        print('\nHere is your toaster information:\n')
        print_toaster(toaster)
        toasts = int(input('\nHow many toasts would you like to make?: '))
        bread = input('\nWhat type of bread would you like to use?: ')
        toaster.add_toasts(toasts, bread)
        time = int(input('\nHow many seconds do you want to toast for?: '))
        toaster.set_timer(time)
        toaster.toast()
    except Exception as e:
        print(e)

def main():
    while True:
        print('\nPlease select a type of toaster:\n')
        print('1. Toaster')
        print('2. Super Toaster')
        toaster_type = input('\nEnter your choice: ')
        if toaster_type in ['1', '2']:
            get_toaster(toaster_type)
        else:
            print('\nInvalid input! Please select either 1 or 2.')
        end_program = input('\nDo you want to exit the program? (yes/no): ')
        if end_program.lower() == 'yes':
            break

if __name__ == '__main__':
    main()
