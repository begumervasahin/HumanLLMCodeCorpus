import pandas as pd
import numpy as np

class TreeNode:
    def __init__(self, data):
        self.data = data
        self.children = []
        self.father = None

    def add_child(self, child):
        child.father = self
        self.children.append(child)
        return child

    def add_children(self, children):
        for child in children:
            self.add_child(TreeNode(child))
        return self.children

    def key(self):
        return self.data["key"]

    def cost(self):
        return self.data["cost"]

class Heap:
    def __init__(self):
        self.heap = []

    def append(self, node):
        self.heap.append(node)
        self.heap.sort(key=lambda x: x.data["cost"])

    def pop(self, key_func):
        return self.heap.pop(0)

class Mapa:
    def __init__(self, grafo):
        self.__grafo = grafo

    def get_heuristic(self, partida, destino):
        if self.__grafo[partida][destino] == -1:
            return self.__grafo[destino][partida]
        elif self.__grafo[destino][partida] == -1:
            return self.__grafo[partida][destino]
        return min(self.__grafo[partida][destino], self.__grafo[destino][partida])

    def get_distance(self, partida, destino):
        if self.__grafo[partida][destino] == -1:
            return -1
        elif self.__grafo[destino][partida] == -1:
            return -1
        return max(self.__grafo[partida][destino], self.__grafo[destino][partida])

    def get_neighbors(self, cidade):
        triangulo_inf = np.tril(np.ones(self.__grafo.shape)).astype(np.bool)     
        vizinhos = self.__grafo.where(triangulo_inf, -1)
        return list(vizinhos[cidade][vizinhos[cidade] > 0].index) + list(vizinhos.loc[cidade][vizinhos.loc[cidade] > 0].index)

def busca_a_estrela(mapa, partida, destino):
    t = TreeNode({"key": partida, "cost": mapa.get_heuristic(partida, destino)}) 

    lista = Heap()
    custo_atual = 0
    while t.data["key"] != destino:
        vizinhos = mapa.get_neighbors(t.data["key"])
        vizinhos = [
            {
                "key": v,
                "cost": custo_atual + mapa.get_heuristic(v, destino) + mapa.get_distance(t.key(), v)
            }
            for v in vizinhos
        ]
        vizinhos = t.add_children(vizinhos)
        lista.append(vizinhos)
        new_t = lista.pop(lambda x: x.cost())
        custo_atual += mapa.get_distance(t.key(), new_t.key())
        t = new_t
    caminho = [t.key()]
    while t.father is not None:
        t = t.father
        caminho.append(t.key())
    caminho.reverse()
    return caminho

if __name__ == "__main__":
    mapa = Mapa(pd.read_csv("grafo.csv", index_col=0).fillna(-1).astype(np.float64))
    partida = input("Cidade de Partida: ")
    destino = input("Cidade de Destino: ")
    caminho = busca_a_estrela(mapa, partida, destino)
    print("->".join(caminho))
