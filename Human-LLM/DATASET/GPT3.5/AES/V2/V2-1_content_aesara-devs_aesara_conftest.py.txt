import os
import pytest

# Set up environment variables before starting the test session
def pytest_sessionstart(session):
    # Configure Aesara flags to raise warnings/errors for optimization and shape issues
    os.environ.setdefault("AESARA_FLAGS", "")
    os.environ["AESARA_FLAGS"] += ",warn__ignore_bug_before=all,on_opt_error=raise,on_shape_error=raise,cmodule__warn_no_version=True"
    # Enable bounds checking for Numba
    os.environ["NUMBA_BOUNDSCHECK"] = "1"

# Add a command-line option to pytest for running slow tests
def pytest_addoption(parser):
    parser.addoption(
        "--runslow", 
        action="store_true", 
        default=False, 
        help="Run tests marked as slow"
    )

# Configure pytest markers
def pytest_configure(config):
    config.addinivalue_line("markers", "slow: Mark a test as slow to run")

# Modify test collection items based on command-line options
def pytest_collection_modifyitems(config, items):
    # If --runslow option is provided, don't skip any tests
    if config.getoption("--runslow"):
        return
    # If --runslow option is not provided, mark slow tests to be skipped
    skip_slow = pytest.mark.skip(reason="Use --runslow option to run this test")
    for item in items:
        if "slow" in item.keywords:
            item.add_marker(skip_slow)

# Define an example test function with the slow marker
@pytest.mark.slow
def test_example_slow():
    assert True

# Define an example test function without the slow marker
def test_example_fast():
    assert True

if __name__ == "__main__":
    pytest.main()
