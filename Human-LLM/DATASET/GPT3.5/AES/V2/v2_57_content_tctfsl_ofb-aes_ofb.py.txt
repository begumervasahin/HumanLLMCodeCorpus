import random
import aes

# Function to divide a string into fixed-size chunks
def chunks(string, blockSize):
    for i in range(0, len(string), blockSize):
        yield string[i:i + blockSize]

# Function to generate a random initialization vector (IV) of a given length     
def generate_random_IV(length):
    IV = "".join(str(random.randint(0, 1)) for _ in range(length))
    return IV

# Function to generate a dummy initialization vector (IV)
def generate_dummy_IV():
    return [0, 1, 0, 1, 0, 0, 0, 0]

# Function to convert a value to its binary representation of a specified size   
def binvalue(val, bitsize):
    binval = bin(val)[2:] if isinstance(val, int) else bin(ord(val))[2:]
    if len(binval) > bitsize:
        raise ValueError("Binary value larger than the expected size")
    return binval.zfill(bitsize)

# Function to split a string into smaller strings of length n
def split_string(text, n):
    return [text[i:i + n] for i in range(0, len(text), n)]

# Function to convert a string to a bit array
def string_to_bit_array(text):
    return [int(bit) for char in text for bit in binvalue(ord(char), 8)]

# Function to convert a bit array to a string
def bit_array_to_string(array):
    return "".join([chr(int("".join(map(str, array[i:i+8])), 2)) for i in range(0, len(array), 8)])

# Function to encrypt plaintext blocks
def encrypt_block(blockSize, plaintextBlockList, shiftRegister):
    key = "11001001110010011100100111001001110010011100100111001001110010011100100111001001110010011100100111001001110010011100100111001001"
    ciphertextBlockList = []
    for plaintextBlock in plaintextBlockList:
        tempRegister = aes.encryption(shiftRegister, key)
        ciphertextBlock = [tempRegister[m] ^ plaintextBlock[m] for m in range(blockSize)]
        shiftRegister = tempRegister[:]
        ciphertextBlockList.append(ciphertextBlock)
    return ciphertextBlockList

# Function to encrypt plaintext blocks with a one-bit error introduced in the second block
def encrypt_block_with_error(blockSize, plaintextBlockList, shiftRegister):
    key = "11001001110010011100100111001001110010011100100111001001110010011100100111001001110010011100100111001001110010011100100111001001"
    ciphertextBlockList = []
    for i, plaintextBlock in enumerate(plaintextBlockList):
        tempRegister = aes.encryption(shiftRegister, key)
        ciphertextBlock = [tempRegister[m] ^ (plaintextBlock[m] if not (i == 1 and m == 1) else 1 - plaintextBlock[m]) for m in range(blockSize)]
        shiftRegister = tempRegister[:]
        ciphertextBlockList.append(ciphertextBlock)
    return ciphertextBlockList

# Function to decrypt ciphertext blocks
def decrypt_blocks(decrypted_blocks):
    decrypted_text = [bit for block in decrypted_blocks for bit in block]
    return bit_array_to_string(decrypted_text)

# Prompting user for input
plaintext = input('Enter plaintext:')
blockSize = int(input('Block size:'))

# Converting plaintext to bit array
plaintextBitArray = string_to_bit_array(plaintext)
print("PlaintextArrayLength:", len(plaintextBitArray))

# Adjusting plaintext to fit block size
padding = blockSize - len(plaintextBitArray) % blockSize
plaintextBitArray += [0] * padding

# Dividing plaintext into blocks
plaintextBlockList = list(chunks(plaintextBitArray, blockSize))

# Generating an initialization vector (IV)
IVsize = 128
shiftRegister = generate_random_IV(IVsize)
shiftRegister2 = shiftRegister

# Encrypting plaintext blocks
ciphertextBlockList = encrypt_block(blockSize, plaintextBlockList, shiftRegister)
wrongCiphertextBlockList = encrypt_block_with_error(blockSize, plaintextBlockList, shiftRegister)

# Decrypting ciphertext blocks
decryptedBlocks = encrypt_block(blockSize, ciphertextBlockList, shiftRegister2)
decryptedTextString = decrypt_blocks(decryptedBlocks)

# Encrypting ciphertext blocks with a one-bit error introduced in the second block
wrongBlocks = encrypt_block_with_error(blockSize, wrongCiphertextBlockList, shiftRegister2)
wrongDecryptedTextString = decrypt_blocks(wrongBlocks)

# Displaying results
print("\nCipherTextBlockList               : ", ciphertextBlockList)
print("CipherTextBlockList (1 bit error) : ", wrongCiphertextBlockList)
print("\nDecrypted BlockList (no error)    : ", decryptedBlocks)
print("Decrypted BlockList (1 bit error) : ", wrongBlocks)
print("\nDecrypted string          : ", decryptedTextString)
print("Incorrect decrypted string: ", wrongDecryptedTextString)
