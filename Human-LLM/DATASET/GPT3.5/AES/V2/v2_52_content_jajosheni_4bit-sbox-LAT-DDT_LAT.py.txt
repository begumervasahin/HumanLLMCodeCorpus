import sys

def read_sbox_from_file(file_path):
    """Reads S-box values from a file."""
    with open(file_path, "r") as file:
        contents = file.read().strip()
    return [int(contents[i], 16) for i in range(16)]

def input_sbox_values():
    """Allows the user to manually input S-box values."""
    sbox = []
    for i in range(16):
        while True:
            val = input(f'Enter the sBox value for {hex(i)} (0-9, a-f): ')
            if val.isdigit() and len(val) == 1 or val in 'abcdef':
                sbox.append(int(val, 16))
                break
            else:
                print("Invalid input. Please enter a single hex digit.")
    return sbox

def calculate_lat(sbox):
    """Calculates the Linear Approximation Table for the given S-box."""
    lat = []
    for input_mask in range(16):
        lat_row = []
        for output_mask in range(16):
            total = 0
            for value in range(16):
                # XOR parity of input and output masks with S-box values.
                input_bit = bin(input_mask & value).count('1') % 2
                output_bit = bin(output_mask & sbox[value]).count('1') % 2
                if input_bit == output_bit:
                    total += 1
            # Subtract 8 to adjust for bias (half the total number of entries).
            lat_row.append(total - 8)
        lat.append(lat_row)
    return lat

def print_lat(lat):
    """Prints the Linear Approximation Table."""
    print('    ', end='')
    for hex_no in range(16):
        print(f' {hex_no:X:>4} ', end='')
    print('\n    ' + '-----' * 16)
    for i, row in enumerate(lat):
        print(f'{i:X} |', end='')
        for val in row:
            print(f'{val:4} |', end='')
        print('\n    ' + '-----' * 16)

def calculate_non_linearity(lat):
    """Calculates the non-linearity measure based on the LAT."""
    max_diff = max(max(abs(x) for x in row) for row in lat)
    nlm = 2**(4-1) - max_diff
    nlms = 2**(4-1) - 2**(4/2 - 1)
    return (nlm / nlms) * 100

if __name__ == '__main__':
    print('\nWelcome to the S-box Linear Approximation Table (LAT) Calculator\n')
    if len(sys.argv) > 1:
        # S-box values provided through a file.
        sbox = read_sbox_from_file(sys.argv[1])
    else:
        # Manual S-box value entry.
        print('Please enter the S-box values:')
        sbox = input_sbox_values()

    lat = calculate_lat(sbox)
    print_lat(lat)

    non_linearity = calculate_non_linearity(lat)
    print(f'Non-linear Measure: {non_linearity:.2f}%')
