from datetime import datetime
# Assuming 'utils.py' and 'tools' module, along with 'Group_iRixhVV' class, are correctly implemented.
from utils import tools as tl
from synoptic.section1 import Group_iRixhVV

def main():
    # Start by creating or resetting the log file to track the evaluation process.
    with open("log.txt", "w") as log_file:
        # Retrieve today's date and time for synops evaluation.
        today, today_minus_6h = tl.fecha_para_synops()

        # URLs for the synoptic data in Spanish (ES) and English (EN).
        # These URLs are formatted with date and time parameters for fetching reports.
        base_url_es = 'http://ogimet.com/display_synops2.php?lugar=78764+78762+78774+78767&tipo=ALL&ord=DIR&nil=SI&fmt=txt&ano={}&mes={}&day={}&hora={}&anof={}&mesf={}&dayf={}&horaf={}&enviar=Ver'
        base_url_en = 'https://www.ogimet.com/display_synops2.php?lang=en&lugar=78764+78762+78774+78767&tipo=ALL&ord=DIR&nil=SI&fmt=txt&ano={}&mes={}&day={}&hora={}&anof={}&mesf={}&dayf={}s&horaf={}&send=send'

        # Fill in the URL with the correct date and time for today and six hours ago.
        formatted_url = base_url_es.format(today_minus_6h[0], str(today_minus_6h[1]).zfill(2), str(today_minus_6h[2]).zfill(2), str(today_minus_6h[3]).zfill(2), today[0], str(today[1]).zfill(2), str(today[2]).zfill(2), str(today[3]).zfill(2))
        print(formatted_url)

        # Assume 'texto_web.txt' contains synoptic reports in a correct format for extraction.
        # Here, reports for station 78774 are extracted.
        synoptic_reports = tl.extraer_synops("texto_web.txt", 78774)

        # Process each synoptic report.
        for report in synoptic_reports:
            print(report)
            report_parts = report.split(' ')
            # The fifth part (index 4) of each report contains the data for Group_iRixhVV evaluation.
            group_iRixhVV = Group_iRixhVV(report_parts[4], log_file)
            # Evaluate the specific group within the report.
            group_iRixhVV.evaluate_group()

# Ensures that this script runs only when executed directly, not when imported as a module.
if __name__ == "__main__":
    main()
