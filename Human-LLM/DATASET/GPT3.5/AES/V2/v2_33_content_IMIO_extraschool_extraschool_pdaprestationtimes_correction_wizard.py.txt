from odoo import models, fields, api

class ExtraschoolPdaPrestationTimesCorrectionWizard(models.TransientModel):
    _name = 'extraschool.pdaprestationtimes_correction_wizard'      

    es = fields.Selection([('E', 'In'), ('S', 'Out')], string='Prestation Type', required=True)
    prestation_time = fields.Float(string='Prestation Time', required=True)

    @api.multi
    def correction(self):
        active_ids = self._context.get('active_ids')
        if active_ids:
            prestations = self.env['extraschool.pdaprestationtimes'].browse(active_ids)
            prestations.write({'es': self.es, 'prestation_time': self.prestation_time})
            for prestation in prestations:
                prestation.prestation_times_of_the_day_id.reset()       
                prestation.prestation_times_of_the_day_id.check()       
        return True
