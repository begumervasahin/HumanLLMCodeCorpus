"""
This script provides functionality for encrypting and decrypting messages using AES encryption.
"""

from Crypto.Cipher import AES
from Crypto.Random import get_random_bytes

class Message:
    """
    A class to handle AES encryption and decryption of messages.
    """
    def __init__(self, key, plaintext=None, ciphertext=None):
        """
        Initializes the Message object with a key and optionally plaintext or ciphertext.

        Args:
            key (bytes): The encryption/decryption key.
            plaintext (str, optional): The message to encrypt. Defaults to None.
            ciphertext (bytes, optional): The message to decrypt. Defaults to None.
        
        Raises:
            InvalidMessage: If neither plaintext nor ciphertext is provided.
        """
        self.key = key
        self.iv = get_random_bytes(AES.block_size)  # Initialization vector

        if plaintext:
            self.plaintext = plaintext
            self.ciphertext = self.encrypt()
        elif ciphertext:
            self.ciphertext = ciphertext
            self.plaintext = self.decrypt()
        else:
            raise InvalidMessage("Please provide either plaintext or ciphertext.")

    def encrypt(self):
        """
        Encrypts the plaintext.

        Returns:
            bytes: The encrypted message.
        """
        cipher = AES.new(self.key, AES.MODE_CBC, self.iv)
        padding_length = AES.block_size - len(self.plaintext) % AES.block_size
        padded_plaintext = self.plaintext + ' ' * padding_length
        return cipher.encrypt(padded_plaintext.encode())

    def decrypt(self):
        """
        Decrypts the ciphertext.

        Returns:
            str: The decrypted message.
        """
        cipher = AES.new(self.key, AES.MODE_CBC, self.iv)
        decrypted_data = cipher.decrypt(self.ciphertext)
        return decrypted_data.rstrip(b' ').decode('utf-8')

    def pack(self):
        """
        Combines the IV and ciphertext into a single bytes object.

        Returns:
            bytes: The combined IV and encrypted message.
        """
        return self.iv + self.ciphertext

class InvalidMessage(Exception):
    """
    An exception for invalid message inputs.
    """
    def __init__(self, message):
        super().__init__(message)
        self.message = message

# Demonstrating the usage of the Message class
if __name__ == '__main__':
    key = get_random_bytes(16)  # Choose an appropriate key size
    plaintext = "Hello, AES Encryption!"

    # Encrypt the message
    message = Message(key, plaintext=plaintext)
    encrypted_message = message.pack()
    print(f"Encrypted: {encrypted_message}")

    # Decrypt the message
    decrypted_message = Message(key, ciphertext=encrypted_message[AES.block_size:])
    print(f"Decrypted: {decrypted_message.plaintext}")
