import sys

def main():
    # Check if the input file is provided
    if len(sys.argv) < 2:
        print("Usage: python script.py <input_file>")        
        return

    # Get the filename from command-line arguments
    filename = sys.argv[1]

    try:
        # Open the file and read its contents
        with open(filename, 'r') as file:
            lines = file.read().splitlines()
    except FileNotFoundError:
        # Handle the case where the file is not found
        print(f"Error: File '{filename}' not found.")        
        return

    # Initialize lists to store flags and outputs
    flags = []
    outputs = []

    # Iterate over each line in the file
    for line in lines:
        i = 5  # Start index for flag
        end = len(line)  # End index for output
        flag = ""
        output = ""

        # Find the position of ':' to separate flag and output
        while line[i] != ":":
            i += 1

        # Extract flag and output from the line
        flags.append(line[:i])
        i += 1
        outputs.append(line[i:end])

    # Print the list of outputs
    print(outputs)

if __name__ == "__main__":
    main()
