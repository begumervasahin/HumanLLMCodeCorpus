        q = int(string[2:7])
        y_alice = int(string[8:12])

        # Compute Y_bob for Diffie-Hellman key exchange
        y_bob = calculate_dh_y_bob(sec_bob, q, a)
        print("Computed Y_bob:", y_bob)
        server_socket.sendto(str(y_bob).encode(), addr)

        # Compute shared key for AES encryption
        alice_key = calculate_aes_key(y_alice, sec_bob, q, a)
        key = bin(3750)[2:].zfill(32)
        start_time = time()

        # Receive initial challenge from Alice and respond
        data = server_socket.recvfrom(BUFFER_SIZE)[0]
        bob_cipher = AESCipher(key)
        bob_challenge = bob_cipher.decrypt(data)

        nonce = generate_random_number()
        bob_challenge -= 1

        cipher_bob = bob_cipher.encrypt(message)

def calculate_dh_y_bob(sec_bob, q, a):
    y_bob = (a ** sec_bob) % q
    return y_bob

def calculate_aes_key(y_alice, sec_bob, q, a):
    calculate_dh_y_bob(X_BOB, q, a)
    k = (y_alice ** sec_bob) % q
    key_value = bin(k)[2:].zfill(32)
    return key_value

def generate_random_number():
    return random.getrandbits(32)

class AESCipher:
    def __init__(self, key):
        self.key = key

    def encrypt(self, message):
        if len(message) % 16 == 0:
            plaintext = message.encode('utf-8')
            cipher = AES.new(self.key, AES.MODE_ECB)
            msg = cipher.encrypt(plaintext)
        else:
            length = len(message)
            plaintext = message + ((16 - length % 16) * str(0))
            plaintext = plaintext.encode('utf-8')
            cipher = AES.new(self.key, AES.MODE_ECB)
            msg = cipher.encrypt(plaintext)
        return msg

    def decrypt(self, msg):
        decipher = AES.new(self.key, AES.MODE_ECB)
        plaintext = decipher.decrypt(msg)
        plaintext = plaintext.decode('utf-8')
        plaintext = plaintext.rstrip('0')
        return plaintext

def main():
    sec_bob = 102  # Assuming sec_bob is a predefined secret value
    server_socket = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    server_socket.bind((HOST, PORT))
    receive_data_and_respond(server_socket, sec_bob)
    server_socket.close()

if __name__ == '__main__':
    main()
