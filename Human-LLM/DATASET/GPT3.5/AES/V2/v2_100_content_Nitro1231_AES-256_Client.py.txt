import socket
from AES256 import AES256

def main():
    # Server configuration
    server_ip = '127.0.0.1'
    server_port = 5555

    # Create a socket object
    server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)

    # Connect to the server
    server_socket.connect((server_ip, server_port))

    # Create an instance of AES256 for encryption and decryption
    aes = AES256("YourSecretKeyHere")

    try:
        # Communication loop
        while True:
            # Get user input
            message = input("Enter message: ")

            # Encrypt the message
            encrypted_message = aes.encrypt(message)

            # Send the encrypted message to the server
            server_socket.send(encrypted_message)

            # Receive a response from the server
            encrypted_response = server_socket.recv(1024)

            # Decrypt and display the response
            decrypted_response = aes.decrypt(encrypted_response)
            print("Server response:", decrypted_response)

    except KeyboardInterrupt:
        print("\nClosing connection...")
    finally:
        # Close the connection
        server_socket.close()

if __name__ == "__main__":
    main()
