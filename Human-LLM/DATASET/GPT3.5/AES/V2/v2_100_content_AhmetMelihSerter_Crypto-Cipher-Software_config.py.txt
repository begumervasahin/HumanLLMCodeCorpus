class SuperNesne:
    def __init__(self, title, size):
        # Initialize the SuperNesne object
        self.window = Tk()
        self.window.title(title)
        self.set_theme()
        self.set_window_size(size)
        self.center_window()

    def set_theme(self):
        # Set the theme for the window
        theme = ThemedStyle(self.window)
        theme.set_theme("radiance")

    def set_window_size(self, size):
        # Set the size of the window
        self.window.geometry(size)
        self.window.resizable(False, False)

    def center_window(self):
        # Center the window on the screen
        self.window.update_idletasks()
        width = self.window.winfo_width()
        frm_width = self.window.winfo_rootx() - self.window.winfo_x()
        win_width = width + 2 * frm_width
        height = self.window.winfo_height()
        titlebar_height = self.window.winfo_rooty() - self.window.winfo_y()
        win_height = height + titlebar_height + frm_width
        x = self.window.winfo_screenwidth() // 2 - win_width // 2
        y = self.window.winfo_screenheight() // 2 - win_height // 2
        self.window.geometry('{}x{}+{}+{}'.format(width, height, x, y))
        self.window.deiconify()

    def listbox_config(self, listbox):
        # Configure the listbox
        listbox['selectbackground'] = ""

    def combobox_config(self, combobox):
        # Configure the combobox
        combobox.option_add("*TCombobox*Listbox*selectBackground", "")

    def window_freeze(self, window, window2=False):
        # Freeze or unfreeze the window
        if window2 == False:
            window.wm_attributes("-disabled", True)
        else:
            window.wm_attributes("-disabled", False)
            window2.destroy()

    def max_length(self, text, limit, password_and_login=False):
        # Check if the length of text exceeds the limit
        # If password_and_login is False, check if the text contains alphabetic characters
        if len(text) > int(limit) or (password_and_login == False and not text.isalpha()):
            return False
        return True

    def state_active(self, btn, btn1=False):
        # Activate the button and change its background color if btn1 is provided
        btn["state"] = "normal"
        if btn1:
            btn["bg"] = "white"

    def state_disabled(self, btn=False, btn1=False):
        # Disable the button or set it to readonly if btn1 is provided
        if btn == False and btn1:
            btn1["state"] = "readonly"
        elif btn and btn1:
            btn["bg"] = ""
            btn["state"] = "disabled"
        else:
            btn["state"] = "disabled"
