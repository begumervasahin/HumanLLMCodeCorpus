import os
import glob
import cv2
import random
import numpy as np
from scipy.spatial import distance

# Custom module for saving results
import ypoften as of

def clear_screen():
    """Clears the terminal screen."""
    os.system("clear")

def edge_detection(img_path, tf_folder, blur_first=True, blur_size=3,
                   threshold1=100, threshold2=250, adaptive_threshold=True,
                   ratio1=0.4, ratio2=0.8, save_tf=True,
                   select_random=True, n_random=1000):
    """Perform edge detection on an image."""
    
    # Read the image
    img = cv2.imread(img_path, 0)

    # Apply Gaussian blur if needed
    if blur_first:
        img = cv2.GaussianBlur(img, (blur_size, blur_size), 0)  

    # Adjust thresholds if adaptive thresholding is enabled     
    if adaptive_threshold:
        threshold1 = min(100, np.quantile(img, q=ratio1))       
        threshold2 = max(200, np.quantile(img, q=ratio2))       
        print("Edge detection thresholds:", threshold1, threshold2)

    # Perform edge detection
    edge = cv2.Canny(img, threshold1=threshold1, threshold2=threshold2)

    # Save the transformed image if required
    if save_tf:
        img_name = os.path.basename(img_path)
        img_save_path = os.path.join(tf_folder, "edge_canny", os.path.splitext(img_name)[0] + '.png')
        of.create_path(img_save_path)
        cv2.imwrite(img_save_path, edge)

    # Extract edge points
    epoints_twodims = np.nonzero(edge)
    epoints = np.transpose(epoints_twodims)
    h, w = edge.shape
    dia = (h**2 + w**2)**0.5
    etotal = len(epoints)
    edensity = etotal / (h * w)

    # Randomly select edge points if required
    if etotal > 0 and select_random:
        random.seed(42)
        epoints = random.sample(list(epoints), min(n_random, etotal))

    # Calculate average distance between edge points
    if etotal > 0:
        dists = distance.pdist(epoints, 'euclidean')
        edist = np.mean(dists) / dia
    else:
        edist = -99999

    edge_data = [edensity, edist]
    print("Edge density & distribution:", edge_data)

    return edge_data

def process_images():
    """Process images in the specified folder."""
    img_folder = os.path.join('img_all', '')
    tf_folder = os.path.join("img_transform", '')
    result_path = os.path.join('img_result', 'edge.txt')
    img_paths = glob.glob(img_folder + '*')

    for j, img_path in enumerate(img_paths[:]):
        print("-" * 100)
        img_name = os.path.basename(img_path)
        print(j, img_name)
        edge_data = [img_name] + edge_detection(img_path, tf_folder)        
        of.save_list_to_txt(edge_data, result_path)

    print("DONE" * 50)

if __name__ == "__main__":
    clear_screen()
    process_images()
