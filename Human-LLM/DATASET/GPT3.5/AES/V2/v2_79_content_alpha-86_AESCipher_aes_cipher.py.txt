import sys
from Crypto.Cipher import AES

class AESCipher:
    block_size = AES.block_size
    zero_padding = 1
    PKCS5Padding = 2
    PKCS7Padding = 3

    def __init__(self, key, mode=AES.MODE_ECB):
        if isinstance(key, str):
            key = key.encode('utf-8')
        self.key = key
        self.aes = AES.new(self.key, mode)

    @classmethod
    def padding(cls, data, padding_type):
        def PKCS7Padding(s):
            pad_length = cls.block_size - len(s) % cls.block_size
            return s + bytes([pad_length] * pad_length)
        
        if padding_type == cls.PKCS5Padding or padding_type == cls.PKCS7Padding:
            return PKCS7Padding(data)
        if padding_type == cls.zero_padding:
            return data

    @classmethod
    def unpadding(cls, data, padding_type):
        def PKCS7Unpadding(s):
            pad_length = s[-1]
            return s[:-pad_length]
        
        if padding_type == cls.PKCS5Padding or padding_type == cls.PKCS7Padding:
            return PKCS7Unpadding(data)
        if padding_type == cls.zero_padding:
            return data

    def hex2bytes(self, raw):
        if isinstance(raw, str):
            raw = bytes.fromhex(raw)
        return raw

    def encrypt(self, raw):
        raw = AESCipher.padding(raw.encode('utf-8'), self.PKCS5Padding)
        return self.aes.encrypt(raw).hex().upper()

    def decrypt(self, enc):
        enc = self.hex2bytes(enc)
        raw = self.unpadding(self.aes.decrypt(enc), self.PKCS5Padding).decode('utf-8').strip()
        return raw

# Initialize AESCipher with the key
key = '1234567890123456'
aes = AESCipher(key)

# Process input from stdin
for line in sys.stdin:
    line = line.strip()
    if line:
        enc = aes.encrypt(line)
        print(enc)
        raw = aes.decrypt(enc)
        print(raw)
