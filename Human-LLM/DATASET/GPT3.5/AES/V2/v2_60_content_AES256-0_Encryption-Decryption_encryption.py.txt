import tkinter as tk
import encryption_decryption

def encrypt_file():
    # Retrieve input values from the GUI
    filename = str(entry_filename.get())
    key_length = int(entry_keylength.get())
    key = str(entry_keyfile.get())
    algorithm = str(entry_algorithm.get())

    # Select encryption method based on the chosen algorithm
    if algorithm == "AESCCM":
        encryption_decryption.aesccm_mode_encryption(key_length, filename, key)
    elif algorithm == "AESGCM":
        encryption_decryption.aesgcm_mode_encryption(key_length, filename, key)
    else:
        print("Please select a valid algorithm")

# Initialize the Tkinter GUI window
root = tk.Tk()
root.geometry("500x300")
root.title("ENCRYPTOR")

# Create labels and entry fields for filename, key length, key file, and algorithm
label_filename = tk.Label(root, text="Filename:")
label_filename.place(x=0, y=0)
entry_filename = tk.Entry(root, width=30)
entry_filename.place(x=200, y=0)

label_keylength = tk.Label(root, text="Key Length:")
label_keylength.place(x=0, y=20)
entry_keylength = tk.Entry(root, width=30)
entry_keylength.place(x=200, y=20)

label_keyfile = tk.Label(root, text="Key File:")
label_keyfile.place(x=0, y=40)
entry_keyfile = tk.Entry(root, width=30)
entry_keyfile.place(x=200, y=40)

label_algorithm = tk.Label(root, text="Algorithm (AESGCM or AESCCM):")
label_algorithm.place(x=0, y=60)
entry_algorithm = tk.Entry(root, width=30)
entry_algorithm.place(x=200, y=60)

# Create an "Encrypt" button to trigger the encryption process
button_encrypt = tk.Button(root, text="Encrypt", command=encrypt_file, bg="grey", fg="black")
button_encrypt.place(x=200, y=100)

# Start the GUI event loop
root.mainloop()
