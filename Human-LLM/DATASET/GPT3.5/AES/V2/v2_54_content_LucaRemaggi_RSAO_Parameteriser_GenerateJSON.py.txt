import numpy as np
import json

class GenerateJSON_RSAO:
    def __init__(self, paramEarly, paramLate, name, maxEarly, filename, objtype):
        self.paramEarly = paramEarly
        self.paramLate = paramLate
        self.name = name
        self.maxEarly = maxEarly
        self.filename = filename
        self.objtype = objtype
        self.libentry = []

    def get_object_vector_room_library(self):
        self.libentry = {
            'name': self.name,
            'type': self.objtype,
            'id': 0,
            'channels': 0,
            'priority': 0,
            'level': 1.0,
            'position': {
                'az': '{:.2f}'.format(self.paramEarly['Direct_sound']['doa'][0]),
                'el': '{:.2f}'.format(self.paramEarly['Direct_sound']['doa'][1]),
                'radius': '1.00'
            },
            'room': {'ereflect': []}
        }

        for idx_refl in range(1, self.maxEarly + 1):
            reflect_data = self.paramEarly['Reflection' + str(idx_refl)]
            tmpDictionary = {
                'level': '{:.3e}'.format(reflect_data['level']),
                'delay': '{:.3e}'.format(reflect_data['toa']),
                'position': {
                    'az': '{:.1f}'.format(reflect_data['doa'][0]),
                    'el': '{:.1f}'.format(reflect_data['doa'][1]),
                    'refdist': '1.0'
                },
                'biquadsos': []
            }

            for filter_params in reflect_data['filtersos']:
                biquad = {
                    'b0': '{:.3e}'.format(filter_params[0]),
                    'b1': '{:.3e}'.format(filter_params[1]),
                    'b2': '{:.3e}'.format(filter_params[2]),
                    'a0': '{:.3e}'.format(filter_params[3]),
                    'a1': '{:.3e}'.format(filter_params[4]),
                    'a2': '{:.3e}'.format(filter_params[5])
                }
                tmpDictionary['biquadsos'].append(biquad)

            self.libentry['room']['ereflect'].append(tmpDictionary)

        self.libentry['room'].update({
            'lreverb': {
                'delay': '{:.3e}'.format(self.paramLate['Late']['toa']),
                'level': format_list(self.paramLate['Late']['level']),
                'attacktime': format_list(self.paramLate['Late']['attacktimes']),
                'decayconst': format_list(self.paramLate['Late']['expdecays'])
            }
        })

        return self

    def save_json(self):
        with open(self.filename, 'w') as outfile:
            json.dump(self.libentry, outfile)

def format_list(list_val):
    val = [f'{item:.2e}' for item in list_val.values()]
    return ', '.join(val)

# Example usage:
if __name__ == "__main__":
    # Define your parameters
    paramEarly = {...}
    paramLate = {...}
    name = "Object1"
    maxEarly = 3
    filename = "output.json"
    objtype = "Type1"

    # Create instance and generate JSON
    generator = GenerateJSON_RSAO(paramEarly, paramLate, name, maxEarly, filename, objtype)
    generator.get_object_vector_room_library().save_json()
