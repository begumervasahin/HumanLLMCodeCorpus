from Crypto.Cipher import AES

# Function to pad data to ensure its length is a multiple of 16 bytes
def Padding(data, padder=b"\0"):
    if not isinstance(data, bytes):
        try:
            data = data.encode("utf-8")
        except:
            pass
    while len(data) % 16 != 0:
        data += padder
    return data

# Function to remove padding from data
def Depadding(data, depadder=b"\0"):
    return data.rstrip(depadder)

# AES encryption key and IV (Initialization Vector)
AES_PASSWORD = b"F<{Z[C?AY)L7!Yc@"
AES_IV = b"nrB27PNZTnn-N^dR"

# Function to encrypt data using AES CBC mode
def Encrypt_AES(data):
    aes = AES.new(AES_PASSWORD, AES.MODE_CBC, AES_IV)
    return aes.encrypt(Padding(data))

# Function to decrypt data using AES CBC mode
def Decrypt_AES(data):
    aes = AES.new(AES_PASSWORD, AES.MODE_CBC, AES_IV)
    return Depadding(aes.decrypt(data))
