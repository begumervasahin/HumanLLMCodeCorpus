# Function to split a 16-bit string into four 4-bit strings
def string_split(VAR):
    HEX0 = str(VAR[0:4])
    HEX1 = str(VAR[4:8])
    HEX2 = str(VAR[8:12])
    HEX3 = str(VAR[12:16])
    return HEX0, HEX1, HEX2, HEX3

# S-box encryption function
def sbox_enc(BITS):
    RTRBITS = []
    for v in BITS:
        HEX0, HEX1, HEX2, HEX3 = string_split(v)
        F8B = '{0:08b}'.format(int(Sbox[int(HEX0, 2)][int(HEX1, 2)]))
        L8B = '{0:08b}'.format(int(Sbox[int(HEX2, 2)][int(HEX3, 2)]))
        joinBITS = F8B + L8B
        RTRBITS.append(joinBITS)
    return RTRBITS

# S-box decryption function
def sbox_dec(filename):
    logging.info("Reading file contents...")
    CONTENT = []
    with open(filename, 'r', encoding='utf-8') as FILE:
        for x in FILE:
            for y in x:
                CONTENT.append(y)
    BITS = []
    for x in CONTENT:
        BITS.append('{0:016b}'.format(ord(x)))
    RTRBITS = []
    for x in BITS:
        HEX0, HEX1, HEX2, HEX3 = string_split(x)
        F8B = '{0:08b}'.format(int(InvSbox[int(HEX0, 2)][int(HEX1, 2)]))
        L8B = '{0:08b}'.format(int(InvSbox[int(HEX2, 2)][int(HEX3, 2)]))
        joinBITS = F8B + L8B
        RTRBITS.append(joinBITS)
    return RTRBITS
