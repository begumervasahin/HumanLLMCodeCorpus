def add(byte1, byte2):
    """
    Adds two 8-bit strings in GF(256).
    """
    if len(byte1) != 8 or len(byte2) != 8 or type(byte1) != str or type(byte2) != str:
        print("Error in gf256.add(): Please enter 8-bit strings")
        return

    int1 = converter.byte_to_int(byte1)
    int2 = converter.byte_to_int(byte2)
    xored = int1 ^ int2
    return converter.int_to_byte(xored)


def mul_word(word1, word2):
    """
    Multiplies two words (4-byte array) in GF(256).
    """
    if len(word1) != 8 or len(word2) != 8 or type(word1) != str or type(word2) != str:
        print("Error in gf256.mul_word(): Please enter words (4-byte array)")
        return

    mod_red = 0b10001
    out = ""
    return out % mod_red


def mul_bytes(q, w):
    """
    Multiplies two bytes in GF(256).
    """
    e = 0
    for x in range(8):
        if w & 1 != 0:
            e ^= q
        bit_set = q & 0x80 != 0
        q <<= 1
        if bit_set:
            q ^= 0x1B
        w >>= 1
    return e % 256


def get_inverse(byt):
    """
    Returns the multiplicative inverse of a byte in GF(256).
    """
    if byt == 0:
        return 0

    for q in range(256):
        if mul_bytes(byt, q) == 1:
            return q

    return None


def pol_mod(pol, mod):
    """
    Performs polynomial modular division.
    """
    to_divide = pol
    while to_divide >= mod:
        sub_to_divide = ""
        pol_str = bin(to_divide)[2:]

        for bit in range(len(pol_str)):
            sub_to_divide += pol_str[bit]
            divide = int(sub_to_divide, 2)

            if divide > mod:
                to_divide = bin(divide ^ mod)[2:]

                if bit < len(pol_str):
                    to_divide += pol_str[bit + 1:]

                to_divide = int(to_divide, 2)
                break

    return to_divide


def mul_byte(byte1, byte2):
    """
    Multiplies two bytes in GF(256).
    """
    if len(byte1) != 8 or len(byte2) != 8 or type(byte1) != str or type(byte2) != str:
        print("Error in gf256.mul_byte(): Please enter 8-bit strings")
        return

    mod_red = 0b100011011
    out = 0
    factor = converter.byte_to_int(byte1)

    for q in range(8):
        if byte2[q] == "1":
            shifted = factor << 7 - q
            out ^= shifted

    return pol_mod(out, mod_red)
