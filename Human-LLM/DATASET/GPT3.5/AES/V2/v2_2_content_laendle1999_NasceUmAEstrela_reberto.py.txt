import logging
import sys
import minesweeper as mns
from AEstrela import AEstrela
from BestFirst import BestFirst

class Game:
    def __init__(self, algorithm_choice):
        logging.basicConfig(filename='minesweeperIa.log', level=logging.DEBUG,
                            format='%(levelname)s %(funcName)s => %(message)s')
        if algorithm_choice == 1:
            self.algorithm = AEstrela()
        elif algorithm_choice == 2:
            self.algorithm = BestFirst()

    def play_game(self, grid_size, number_of_mines):
        current_grid = [[' ' for _ in range(grid_size)] for _ in range(grid_size)]
        grid = []
        flags = []
        win = True
        game = True
        previous_cell = None

        # Get the initial cell to open from the AI
        initial_cell = self.algorithm.jogadaInicial()
        cell_to_open = (initial_cell.getX(), initial_cell.getY())

        # Setup the grid with mines if not already done
        if not grid:
            grid, mines = mns.setupgrid(grid_size, cell_to_open, number_of_mines)

        while game:
            # Determine the next move from the AI
            next_cell = self.algorithm.definirJogada(current_grid)

            # Check if the cell is None (no more valid moves)
            if next_cell is None:
                print('Perdeu, a lista de abertos ficou vazia')
                win = False
                game = False
            else:
                cell_to_open = (next_cell.getX(), next_cell.getY())
                print(cell_to_open, next_cell.getData())

                # Play the game based on the AI's decision
                if next_cell.getData() == 'F':
                    game = mns.jogar(cell_to_open, current_grid, grid, flags, mines, flag=True)
                else:
                    game = mns.jogar(cell_to_open, current_grid, grid, flags, mines)
                    previous_cell = next_cell

        return win

if len(sys.argv) > 1:
    algorithm_choice = int(sys.argv[1])
    grid_size = int(sys.argv[2])
    number_of_mines = int(sys.argv[3])
else:
    print('Algoritmos:\nA*:1\nBestFirst:2')
    algorithm_choice = int(input('Selecione o Algoritmo(1 ou 2):'))
    grid_size = int(input('Selecione o tamanho do tabuleiro:'))  
    number_of_mines = int(input('Selecione o numero de bombas:')) 

# Initialize the game with the chosen algorithm
mine_sweeper_game = Game(algorithm_choice)
mine_sweeper_game.play_game(grid_size, number_of_mines)
