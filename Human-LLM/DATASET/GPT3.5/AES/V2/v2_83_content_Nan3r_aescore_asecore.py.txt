import base64
import zlib
import time
import hashlib
from Crypto.Cipher import AES

class AESCipher:
    def __init__(self, IV=None, Key=None):
        self.block_size = 16
        self.IV = IV if IV else self.generate_IV()
        self.Key = Key if Key else hashlib.md5(str(self.IV).encode()).hexdigest()

    def generate_IV(self):
        """
        Generate a random initialization vector (IV).
        """
        return hashlib.md5(str(time.time()).encode()).hexdigest()[:self.block_size]

    def encrypt(self, plaintext):
        """
        Encrypts plaintext using AES in CBC mode with PKCS7 padding.

        Args:
            plaintext (str): The plaintext to be encrypted.

        Returns:
            str: The encrypted ciphertext.
        """
        pad = lambda s: s + (self.block_size - len(s) % self.block_size) * chr(self.block_size - len(s) % self.block_size)
        padded_plaintext = pad(plaintext)
        cipher = AES.new(self.Key.encode(), AES.MODE_CBC, self.IV.encode())
        ciphertext = cipher.encrypt(padded_plaintext)
        compressed_ciphertext = zlib.compress(self.IV.encode() + ciphertext)
        return base64.b64encode(compressed_ciphertext)

    def decrypt(self, ciphertext):
        """
        Decrypts ciphertext using AES in CBC mode with PKCS7 padding.

        Args:
            ciphertext (str): The ciphertext to be decrypted.

        Returns:
            str: The decrypted plaintext.
        """
        unpad = lambda s: s[0:-ord(s[-1])]
        compressed_ciphertext = base64.b64decode(ciphertext)
        IV = zlib.decompress(compressed_ciphertext)[:self.block_size].decode()
        ciphertext = zlib.decompress(compressed_ciphertext)[self.block_size:].decode()
        Key = hashlib.md5(IV.encode()).hexdigest()
        cipher = AES.new(Key.encode(), AES.MODE_CBC, IV.encode())
        plaintext = cipher.decrypt(ciphertext)
        return unpad(plaintext).decode()
