import socket
import threading

# Server configuration
IP = "127.0.0.1"
Port = 5555
MSG_HEADER = 10

# Thread lock for handling concurrent access
lock = threading.Lock()

# Lists to keep track of sockets and clients
socket_list = []
clients_list = {}

# Function to handle client connections
def handle_client(client_socket):
    while True:
        data = client_socket.recv(1024)
        if not data:
            print('Connection closed by client')
            lock.release()
            break
        # Reverse the received data and send it back
        data_reversed = data[::-1]
        client_socket.send(data_reversed)
    # Close the client socket when done
    client_socket.close()

# Main server function
def main():
    # Create a server socket
    server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server_socket.bind((IP, Port))
    print(f"Server socket bound at {IP}:{Port}")
    server_socket.listen()

    # Add the server socket to the list of sockets
    socket_list.append(server_socket)
    print("Socket is now listening for connections...")

    # Main server loop to accept incoming connections
    while True:
        # Accept incoming connection
        client_socket, client_address = server_socket.accept()
        lock.acquire()  # Acquire lock before processing the client
        print(f"Connected to {client_address[0]}:{client_address[1]}")

        # Start a new thread to handle the client
        threading.Thread(target=handle_client, args=(client_socket,)).start()

    # Close the server socket when done
    server_socket.close()

if __name__ == "__main__":
    main()
