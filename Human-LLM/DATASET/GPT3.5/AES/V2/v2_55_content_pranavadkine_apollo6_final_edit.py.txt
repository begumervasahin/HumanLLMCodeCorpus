import numpy as np
import cv2
import math
import os
import random
import struct
from Crypto.Cipher import AES


def slice_image(input_image, output_directory, slices):
    img = cv2.imread(input_image)
    height, width, channels = img.shape
    print(f"Image dimensions: Height={height}, Width={width}")
    width_of_slice = math.floor(width / slices)
    for x in range(1, slices):
        start = (x - 1) * width_of_slice + 1
        end = x * width_of_slice
        cropped = img[0:height, start:end]
        image_name = f"{output_directory}/d{x}.jpg"
        cv2.imwrite(image_name, cropped)
    cropped = img[0:height, (slices - 1) * width_of_slice:width]
    image_name = f"{output_directory}/d{slices}.jpg"
    cv2.imwrite(image_name, cropped)


def encrypt_file(key, in_filename, out_filename=None, chunksize=64 * 1024):
    if not out_filename:
        out_filename = in_filename + ".enc"
    iv = "".join(chr(random.randint(0, 0xFF)) for i in range(16))
    encryptor = AES.new(key, AES.MODE_CBC, iv)
    filesize = os.path.getsize(in_filename)
    with open(in_filename, "rb") as infile:
        with open(out_filename, "wb") as outfile:
            outfile.write(struct.pack("<Q", filesize))
            outfile.write(iv)
            while True:
                chunk = infile.read(chunksize)
                if len(chunk) == 0:
                    break
                elif len(chunk) % 16 != 0:
                    chunk += b" " * (16 - len(chunk) % 16)
                outfile.write(encryptor.encrypt(chunk))


def distribute_encrypted_files(slices):
    for x in range(1, slices + 1):
        os.rename(f"d{x}.jpg.enc", f"server{x}/d{x}.jpg.enc")


def decrypt_file(key, in_filename, out_filename=None, chunksize=24 * 1024):
    if not out_filename:
        out_filename = os.path.splitext(in_filename)[0]
    with open(in_filename, "rb") as infile:
        origsize = struct.unpack("<Q", infile.read(struct.calcsize("Q")))[0]
        iv = infile.read(16)
        decryptor = AES.new(key, AES.MODE_CBC, iv)
        with open(out_filename, "wb") as outfile:
            while True:
                chunk = infile.read(chunksize)
                if len(chunk) == 0:
                    break
                outfile.write(decryptor.decrypt(chunk))
            outfile.truncate(origsize)


def concatenate_images(slices):
    output_image = cv2.imread("d1.jpg")
    for x in range(2, slices + 1):
        image_name = f"d{x}_decrypted.jpg"
        img = cv2.imread(image_name)
        output_image = np.concatenate((output_image, img), axis=1)
    cv2.imwrite("concatenated_output_final.jpg", output_image)
