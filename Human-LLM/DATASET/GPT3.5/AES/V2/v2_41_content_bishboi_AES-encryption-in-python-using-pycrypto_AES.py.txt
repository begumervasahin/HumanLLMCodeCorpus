import os
import base64
from Crypto.Cipher import AES
from Crypto import Random
from Crypto.Random import get_random_bytes
from Crypto.Util import Counter

BLOCK_SIZE = 16

def pad(s):
    padding = BLOCK_SIZE - len(s) % BLOCK_SIZE
    return s + bytes([padding]) * padding

def unpad(s):
    padding = s[-1]
    return s[:-padding]

class CipherAES:
    def __init__(self, key):
        self.key = key

    def encrypt(self, raw, iv=None):
        raw = pad(raw)
        cipher = AES.new(self.key, AES.MODE_CBC, iv)
        encrypted = cipher.encrypt(raw)
        if iv:
            return iv + encrypted
        return encrypted

    def decrypt(self, enc, iv=None):
        if iv:
            enc = enc[len(iv):]
        cipher = AES.new(self.key, AES.MODE_CBC, iv)
        decrypted = cipher.decrypt(enc)
        return unpad(decrypted)

def generate_key():
    return os.urandom(BLOCK_SIZE)

def generate_iv():
    return Random.new().read(BLOCK_SIZE)

def generate_plaintext():
    return get_random_bytes(BLOCK_SIZE)

def generate_counter():
    return Counter.new(128)

plaintext = generate_plaintext().hex()
key = generate_key()
iv = generate_iv()

print('plaintext:', plaintext)
print('iv:', iv.hex())
print('key:', key.hex())

cbc_cipher = CipherAES(key)
C = cbc_cipher.encrypt(plaintext, iv)
print('encrypted in CBC:', base64.b64encode(C).decode())
C = cbc_cipher.decrypt(C, iv)
print('decrypted in CBC:', C)

ecb_cipher = CipherAES(key)
C = ecb_cipher.encrypt(plaintext)
print('encrypted in ECB:', base64.b64encode(C).decode())
C = ecb_cipher.decrypt(C)
print('decrypted in ECB:', C)

ofb_cipher = CipherAES(key)
C = ofb_cipher.encrypt(plaintext, iv)
print('encrypted in OFB:', base64.b64encode(C).decode())
C = ofb_cipher.decrypt(C, iv)
print('decrypted in OFB:', C)

ctr_cipher = CipherAES(key)
C = ctr_cipher.encrypt(plaintext, iv)
print('encrypted in CTR:', base64.b64encode(C).decode())
C = ctr_cipher.decrypt(C, iv)
print('decrypted in CTR:', C)

print('CTR:', generate_counter())
