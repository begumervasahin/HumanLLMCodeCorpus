# Operations
def ffAdd(a, b):
    """Performs finite field addition (XOR) of two bytes."""
    return a ^ b

def xtime(byte):
    """Performs multiplication by x (0x02) in GF(2^8)."""
    return ((byte << 1) & 0xFF) ^ (0x1B if byte & 0x80 else 0x00)

def ffMultiply(a, b):
    """Performs finite field multiplication of two bytes."""
    product = 0
    for _ in range(8):
        if b & 1:
            product ^= a
        high_bit_set = a & 0x80
        a = (a << 1) & 0xFF
        if high_bit_set:
            a ^= 0x1B
        b >>= 1
    return product
