class GridNode:
    def __init__(self, x, y, gridsize=9):
        self.__x = x
        self.__y = y
        self.__data = None
        self.__gridsize = gridsize

    def __eq__(self, other):
        return self.__x == other.getX() and self.__y == other.getY()

    def setF(self, f):
        self.__f = f

    def getF(self):
        return self.__f

    def getX(self):
        return self.__x

    def getY(self):
        return self.__y

    def getXY(self):
        return (self.__x, self.__y)

    def setData(self, data):
        self.__data = data

    def getData(self):
        return self.__data

    def getNeighbors(self, grid):
        neighbors = []
        for a in range(self.__x - 1, self.__x + 2):
            for b in range(self.__y - 1, self.__y + 2):
                if (a == self.__x and b == self.__y):
                    continue
                if (self.__isValid(a, b)):
                    node = GridNode(a, b)
                    node.setData(grid[a][b])
                    neighbors.append(node)
        return neighbors

    def __isValid(self, x, y):
        if (x < 0 or x > self.__gridsize - 1 or
                y < 0 or y > self.__gridsize - 1):
            return False
        return True

    def closedNeighborsToOpen(self, grid):
        neighbors = []
        closed = []
        neighbors = self.getNeighbors(grid)
        for cell in neighbors:
            if cell.getData() == ' ':
                closed.append(cell)
        return closed

    def checkCell(self, grid):
        if self.__data == 'F' or self.__data == ' ':
            return 0
        bombs = 0
        neighbors = []
        closed = []
        neighbors = self.getNeighbors(grid)
        closed = self.closedNeighborsToOpen(grid)
        closedCount = len(closed)
        for cell in neighbors:
            if cell not in closed:
                if cell.getData() == 'F':
                    bombs += 1
        if closedCount != 0 and grid[self.__x][self.__y] != ' ':
            return (int(grid[self.__x][self.__y]) - bombs) / closedCount
        else:
            return 0

# Example usage:
if __name__ == "__main__":
    # Example grid
    grid = [
        ['1', 'F', ' ', '2', ' '],
        [' ', '2', 'F', ' ', ' '],
        ['1', ' ', ' ', '2', 'F'],
        [' ', '1', '1', ' ', '1'],
        [' ', ' ', ' ', '2', ' ']
    ]

    # Create a grid node
    node = GridNode(1, 1)

    # Set data for the node
    node.setData(grid[node.getX()][node.getY()])

    # Check the cell
    result = node.checkCell(grid)
    print("Result:", result)
