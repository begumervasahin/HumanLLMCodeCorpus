import random

if __name__ == "__main__":
    number = random.randrange(2**127, (2**128)-1)
    key = random.randrange(2**127, (2**128)-1)
    print("Result of XOR: " + bin(number ^ key))
    print("Plain text: " + bin(number))
    
    State = ["0" * 8 for _ in range(16)]
    for i in range(16):
        State[i] = bin(number)[8*i+2:8*i+10]

    print("State Array:")
    print(State)
    
    Sbox = [random.randrange(2**7, (2**8)-1) for _ in range(256)]
    
    for i in range(16):
        z = State[i]
        row = int(z[:4], 2)
        col = int(z[4:], 2)
        State[i] = Sbox[(col + row) % 16]

    print("State Array after the Substitution process:")
    print(State)
