import paho.mqtt.client as mqtt
import configparser
from ecdsa import SigningKey, VerifyingKey, BadSignatureError

# Read configuration from publisher.ini file
config = configparser.RawConfigParser()
config.read('config/publisher.ini')

# Load private and public keys
sk = SigningKey.from_pem(open("config/private192.pem").read())
vk = VerifyingKey.from_pem(open("config/public192.pem").read())

# Define message to be signed and published
messageS = 'sds'
message = messageS.encode('utf-8')

# Sign the message with the private key
sig = sk.sign(message)

# Verify the signature with the public key
try:
    vk.verify(sig, message)
    print("Good signature")
except BadSignatureError:
    print("BAD SIGNATURE")

# Publish the message over MQTT
client = mqtt.Client()
client.connect(config.get('mqtt', 'broker'), int(config.get('mqtt', 'port')))
client.publish(config.get('mqtt', 'topic'), message)
client.disconnect()
