def AddRoundKey(state, w):
    return [state[i] ^ w[i] for i in range(len(state))]

def SubWord(word, box):
    return [box[b] for b in word]

def RotWord(word, n):
    return word[n:] + word[:n]

def KeyExpansion(key, box, rcon):
    w = []
    for i in range(4):
        w.append([key[4 * i], key[4 * i + 1], key[4 * i + 2], key[4 * i + 3]])
    
    for i in range(4, 4 * 11):
        temp = w[i - 1]
        if i % 4 == 0:
            temp = SubWord(RotWord(temp, 1), box)
            temp[0] ^= rcon[i // 4 - 1]
        w.append([w[i - 4][j] ^ temp[j] for j in range(4)])
    
    return w

def ShiftRows(state):
    output_state = [0] * len(state)
    for i in range(4):
        row = [state[j] for j in range(i, len(state), 4)]
        output_state[i:] = RotWord(row, i)
    return output_state

def MixColumns(state, box, into_func):
    new_state = []
    for i in range(0, len(state), 4):
        col = state[i:i+4]
        new_col = [0] * 4
        for j in range(4):
            new_col[j] = into_func(
                col[0] * box[0 + j*4] ^ 
                col[1] * box[1 + j*4] ^ 
                col[2] * box[2 + j*4] ^ 
                col[3] * box[3 + j*4]
            )
        new_state.extend(new_col)
    return new_state

def Cipher(block, key, box, rcon, into_func):
    KeyEx = KeyExpansion(key, box, rcon)
    state = block
    state = AddRoundKey(state, KeyEx[0:4])      
    for round in range(1, 10):
        state = SubWord(state, box)
        state = ShiftRows(state)
        state = MixColumns(state, box, into_func)
        state = AddRoundKey(state, KeyEx[4 * round:4 * (round + 1)])
    state = SubWord(state, box)
    state = ShiftRows(state)
    state = AddRoundKey(state, KeyEx[4 * 10:])  
    return state

def to_int_list(s):
    return [ord(c) for c in s]

def to_hex_list(s):
    return [int(c, 16) for c in s.split()]

def EncryptFile(plaintext_file, ciphertext_file, key, box, rcon, into_func):
    read_file = open(plaintext_file, "r")
    s = read_file.read()
    read_file.close()
    key_list = to_int_list(key)
    plaintext_int = to_int_list(s)
    l = len(s)
    ciphertext = []
    for i in range(l // 16):
        ciphertext.append(Cipher(plaintext_int[i*16:(i+1)*16], key_list, box, rcon, into_func))
    if l % 16 > 0:
        ciphertext.append(Cipher(plaintext_int[l//16*16:] + [0] * (16 - l % 16), key_list, box, rcon, into_func))
    write_file = open(ciphertext_file, "w")
    for encrypted_block in ciphertext:
        block_list = ['{:02x}'.format(i) for i in encrypted_block]
        block_string = " ".join(block_list)
        write_file.write(block_string + "\n")
    write_file.close()
