class SuperNesne:
    def __init__(self,title,size):
        window = Tk()
        window.title(title)
        theme = ThemedStyle(window)
        theme.set_theme("radiance")
        window.geometry(size)
        window.resizable(False,False)
        self.center(window)
        return window

    def listbox_config(self,listbox):
        listbox['selectbackground'] = ""

    def combobox_config(self,combobox):
        combobox.option_add("*TCombobox*Listbox*selectBackground", "")

    def window_freeze(self,window,window2=False):
        if window2==False:
            window.wm_attributes("-disabled", True)
        else:
            window.wm_attributes("-disabled", False)
            window2.destroy()

    def center(self,win):
        win.update_idletasks()
        width = win.winfo_width()
        frm_width = win.winfo_rootx() - win.winfo_x()
        win_width = width + 2 * frm_width
        height = win.winfo_height()
        titlebar_height = win.winfo_rooty() - win.winfo_y()     
        win_height = height + titlebar_height + frm_width       
        x = win.winfo_screenwidth() // 2 - win_width // 2       
        y = win.winfo_screenheight() // 2 - win_height // 2     
        win.geometry('{}x{}+{}+{}'.format(width, height, x, y)) 
        win.deiconify()

    def maxlenght(self,text,limit,password_and_login=False):    
        if len(text) > int(limit) or (password_and_login == False and not text.isalpha()):
            return False
        return True

    def state_active(self,btn,btn1=False):
        btn["state"] = "normal"
        if btn1:
            btn["bg"] = "white"

    def state_disabled(self,btn=False,btn1=False):
        if btn == False and btn1:
            btn1["state"] = "readonly"
        elif btn and btn1:
            btn["bg"] = ""
            btn["state"] = "disabled"
        else:
            btn["state"] = "disabled"
