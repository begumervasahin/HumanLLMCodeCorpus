import os
import pandas as pd
import torch as t
import numpy as np
from PIL import Image
from torch.utils.data import Dataset

class AADBDataset(Dataset):
    def __init__(self, root, labelroot, transforms=None):
        self.root = root
        self.labelroot = labelroot
        self.transforms = transforms
        self.label = pd.read_csv(labelroot)
        self.imgs = [os.path.join(root, img) for img in self.label['filename']]
        
    def __getitem__(self, index):
        img_path = self.imgs[index]
        label = self.label.loc[index, '1':].values.astype(np.float32)
        
        pil_img = Image.open(img_path).convert('RGB')
        
        if self.transforms:
            pil_img = self.transforms(pil_img)
        
        data = t.from_numpy(np.array(pil_img))
        
        return data, label
    
    def __len__(self):
        return len(self.imgs)
