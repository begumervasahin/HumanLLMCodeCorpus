def p2p_echo():
    try:
        sconn = find_bot()
        sconn.verbose = True
        sconn.send(bytes("ECHO", "ascii"))
        while True:
            msg = input("Echo> ")
            byte_msg = bytes(msg, "ascii")
            sconn.send(byte_msg)
            echo = sconn.recv(1024)
            assert echo == byte_msg
            if msg.lower() in ['x', 'exit', 'quit']:
                sconn.close()
                break
    except socket.error:
        print("Connection closed unexpectedly")
