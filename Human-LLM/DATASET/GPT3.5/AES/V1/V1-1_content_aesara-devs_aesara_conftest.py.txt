import os
import pytest

# Function to set environment variables before the test session starts
def pytest_sessionstart(session):
    os.environ["AESARA_FLAGS"] = ",".join(
        [
            os.environ.setdefault("AESARA_FLAGS", ""),
            "warn__ignore_bug_before=all,on_opt_error=raise,on_shape_error=raise,cmodule__warn_no_version=True",
        ]
    )
    os.environ["NUMBA_BOUNDSCHECK"] = "1"

# Function to add a command-line option to pytest
def pytest_addoption(parser):
    parser.addoption(
        "--runslow", action="store_true", default=False, help="run slow tests"
    )

# Function to configure pytest markers
def pytest_configure(config):
    config.addinivalue_line("markers", "slow: mark test as slow to run")

# Function to modify test collection items
def pytest_collection_modifyitems(config, items):
    # Check if the --runslow option is provided
    if config.getoption("--runslow"):
        return
    # If --runslow option is not provided, skip slow tests
    skip_slow = pytest.mark.skip(reason="need --runslow option to run")
    for item in items:
        if "slow" in item.keywords:
            item.add_marker(skip_slow)

# Example test function with the slow marker
@pytest.mark.slow
def test_example_slow():
    assert True

# Example test function without the slow marker
def test_example_fast():
    assert True

if __name__ == "__main__":
    pytest.main()
