import sys
import os
import argparse
import json
from base64 import b64decode
from Crypto.Random import get_random_bytes
from Crypto.Cipher import AES

MAX_BUFFER_SIZE = 4096
KEY_LENGTH = 32

class AesEncipher:
    def __init__(self, mode, key, nonce):
        self.cipher = AES.new(key, mode, nonce=nonce)

    def encrypt(self, plaintext):
        return self.cipher.encrypt(plaintext)

    def decrypt(self, ciphertext):
        return self.cipher.decrypt(ciphertext)

    def get_nonce(self):
        return b64decode(self.cipher.nonce)

    def get_key(self):
        return b64decode(self.cipher.key)

def parse_args():
    parser = argparse.ArgumentParser(description='AES(CTR Mode) Encipher Program')
    parser.add_argument('tool_mode', help='choose e (encrypt) or d (decrypt)')
    parser.add_argument('input_file', help='file to en/de-crypt')
    parser.add_argument('--nonce', help='nonce to decrypt', type=str, default="")
    parser.add_argument('--key', help='key to decrypt', type=str, default="")
    return parser.parse_args()

def check_args(args):
    if args.tool_mode not in ('e', 'd'):
        print("error: args1: undefined parameter")
        sys.exit()
    if not os.path.isfile(args.input_file):
        print("error: args2: file not exists")
        sys.exit()
    if args.tool_mode == 'd' and not args.nonce:
        print("error: --nonce is null: need nonce to use decrypt mode")
        sys.exit()
    if args.tool_mode == 'd' and not args.key:
        print("error: --key is null: need key to use decrypt mode")
        sys.exit()

if __name__ == '__main__':
    args = parse_args()
    check_args(args)
    encrypt_mode = args.tool_mode == 'e'
    key = get_random_bytes(KEY_LENGTH) if not args.key else b64decode(args.key)
    nonce = b64decode(args.nonce) if args.nonce else None
    aes_encipher = AesEncipher(AES.MODE_CTR, key, nonce)

    translated_bytes = bytearray()

    with open(args.input_file, 'rb') as f:
        for chunk in iter(lambda: f.read(MAX_BUFFER_SIZE), b''):
            if encrypt_mode:
                translated_bytes.extend(aes_encipher.encrypt(chunk))
            else:
                translated_bytes.extend(aes_encipher.decrypt(chunk))

    out_file_name = args.input_file + "_encrypted" if encrypt_mode else args.input_file + "_decrypted"
    
    with open(out_file_name, 'wb') as translated_file:
        translated_file.write(translated_bytes)

    if encrypt_mode:
        encrypt_info = json.dumps({'nonce': aes_encipher.get_nonce().decode(), 'key': aes_encipher.get_key().decode()})
        print(encrypt_info)
