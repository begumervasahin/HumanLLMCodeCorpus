import tkinter as tk
import encryption_decryption

def result():
    name = str(filename.get())
    length = int(keylength.get())
    key = str(keyfile.get())
    algo_name = gcm_ccm.get()
    if algo_name == "AESCCM":
        encryption_decryption.aesccm_mode_encryption(length, name, key)
    elif algo_name == "AESGCM":
        encryption_decryption.aesgcm_mode_encryption(length, name, key)
    else:
        print("Provide the correct algorithm")

root = tk.Tk()
root.geometry("500x300")
root.title("ENCRYPTOR")

label1 = tk.Label(root, text="filename:")
label1.place(x=0, y=0)
filename = tk.Entry(root, width=30)
filename.place(x=200, y=0)

label2 = tk.Label(root, text="keylength:")
label2.place(x=0, y=20)
keylength = tk.Entry(root, width=30)
keylength.place(x=200, y=20)

label3 = tk.Label(root, text="keyfile:")
label3.place(x=0, y=40)
keyfile = tk.Entry(root, width=30)
keyfile.place(x=200, y=40)

label4 = tk.Label(root, text="AESGCM OR AESCCM:")
label4.place(x=0, y=60)
gcm_ccm = tk.Entry(root, width=30)
gcm_ccm.place(x=200, y=60)

button = tk.Button(root, text="Encrypt", command=result, bg="grey", fg="black")
button.place(x=200, y=100)

root.mainloop()
