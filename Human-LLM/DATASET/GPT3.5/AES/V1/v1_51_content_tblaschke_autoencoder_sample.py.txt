import os
import torch
from torch.utils.data import DataLoader
from random import randint
from rdkit import Chem, RDLogger
from src.datareader import SMILESReader, OnehotDecoder
from train import get_args, load_autoencoder

def setup_data_loaders(args, use_cuda):
    """
    Prepare data loaders for training and validation datasets.
    """
    dataset_path = os.path.join(os.path.dirname(__file__), "data/prior_trainingset_DRD2_actives_removed.smi")
    alphabet_path = os.path.join(os.path.dirname(__file__), "data/alphabet.json")

    train_dataset = SMILESReader(dataset_path, alphabet_path, subset=(0, 1200001))
    val_dataset = SMILESReader(dataset_path, alphabet_path, subset=(1200001, None))

    train_loader = DataLoader(train_dataset, batch_size=args.batch_size, shuffle=True,
                              num_workers=4, drop_last=True, pin_memory=use_cuda)
    val_loader = DataLoader(val_dataset, batch_size=args.batch_size, shuffle=True,
                            num_workers=4, drop_last=True, pin_memory=use_cuda)
    return train_loader, val_loader

def validate_point(autoencoder, onehot_decoder, encoded, startsmi, groundTruth=None, print_all=False, repeat=100, batches=1):
    """
    Validate a given point by decoding and comparing with the original or generating new molecules.
    """
    encoded_stack = torch.stack([encoded.squeeze() for _ in range(batches)])
    groundTruth_stack = torch.stack([groundTruth.squeeze() for _ in range(batches)]) if groundTruth is not None else None

    invalid = 0
    same = 0
    mol_set = []

    logger = RDLogger.logger()
    logger.setLevel(RDLogger.CRITICAL)

    for _ in range(repeat):
        sampled = autoencoder.decode(encoded_stack, groundTruth_stack) if groundTruth is not None else autoencoder.decode(encoded_stack)
        smi = onehot_decoder.decode(sampled.data)
        mols = [Chem.MolFromSmiles(s) for s in smi]

        compare_with = Chem.MolToSmiles(Chem.MolFromSmiles(startsmi)) if startsmi is not None else None

        for mol in mols:
            if mol is None:
                invalid += 1
                continue
            
            s = Chem.MolToSmiles(mol)
            if compare_with and s == compare_with:
                same += 1
            if not compare_with:
                mol_set.append(s)

    if compare_with is None:
        return invalid, set(mol_set)
    else:
        return invalid, same

def main():
    args, use_cuda = get_args()
    train_loader, val_loader = setup_data_loaders(args, use_cuda)

    autoencoder = load_autoencoder(args)
    onehot_decoder = OnehotDecoder(os.path.join(os.path.dirname(__file__), "data/alphabet.json"))

    # Example validation
    x, y = val_loader.dataset[randint(0, 1000)]  # Random selection from validation dataset
    startsmi = onehot_decoder.decode(x.unsqueeze(0))[0]  # Decode the randomly selected SMILES

    encoded_smi = autoencoder.encode(x.unsqueeze(0))
    encoded_noise = torch.randn_like(encoded_smi)

    print("Validating molecule with GroundTruth:", startsmi)      
    invalid, same = validate_point(autoencoder, onehot_decoder, encoded_smi, startsmi, groundTruth=train_loader.dataset.onehotencoder(startsmi), print_all=False)
    print(f"{invalid} (out of {args.batch_size * repeat}) invalid")
    print(f"{same} (out of {(args.batch_size * repeat) - invalid}) reconstructed")

    print("Validating molecule in free mode:", startsmi)
    invalid, same = validate_point(autoencoder, onehot_decoder, encoded_smi, startsmi)       
    print(f"{invalid} (out of {args.batch_size * repeat}) invalid")
    print(f"{same} (out of {(args.batch_size * repeat) - invalid}) reconstructed")

    print("Decoding noise")
    invalid, molset = validate_point(autoencoder, onehot_decoder, encoded_noise, None)
    print(f"{invalid} (out of {args.batch_size * repeat}) invalid")
    print(molset)

if __name__ == "__main__":
    main()
