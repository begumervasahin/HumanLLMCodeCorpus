import os
import cv2  # Ensure cv2 is imported for OpenCV functionality
import torch
import numpy as np
import torchvision.transforms as transforms
import torchvision.datasets as datasets
from PIL import Image
from torch.utils.data import DataLoader, Dataset

# Assuming you're running this on a system with a CUDA-capable GPU and PyTorch installed.
print(torch.cuda.get_device_name(0))

class ImageDataset(Dataset):
    def __init__(self, data_path, transform=None, label_transform=None):
        self.data_path = data_path['data']
        self.label_path = data_path['labels']
        self.data_files = os.listdir(self.data_path)
        self.label_files = os.listdir(self.label_path)
        self.transform = transform
        self.label_transform = label_transform
        
        # Check if data and labels counts match
        if len(self.data_files) != len(self.label_files):
            raise ValueError("The number of data files and label files do not match.")

    def __len__(self):
        return len(self.data_files)

    def __getitem__(self, index):
        if index >= len(self):
            raise IndexError("Index should be less than {}".format(len(self)))
        
        # Load image and label
        image_path = os.path.join(self.data_path, self.data_files[index])
        label_path = os.path.join(self.label_path, self.label_files[index])
        image = Image.open(image_path).convert('RGB')
        label_image = cv2.imread(label_path)
        label_image = cv2.cvtColor(label_image, cv2.COLOR_BGR2RGB)
        label_image = cv2.resize(label_image, (480, 480))
        label_image = Image.fromarray(label_image)

        # Apply transformations
        if self.transform is not None:
            image = self.transform(image)
        if self.label_transform is not None:
            label_image = self.label_transform(label_image)
        else:
            # Default transform for label if not specified
            label_image = transforms.ToTensor()(label_image)

        return image, label_image

# Example usage:
if __name__ == "__main__":
    data_path = {
        'data': 'path_to_data_directory',
        'labels': 'path_to_labels_directory'
    }
    
    # Define transformations for the input images and labels
    transform = transforms.Compose([
        transforms.ToTensor(),
        transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
    ])

    # Optionally define different transformations for the labels if needed
    label_transform = transforms.ToTensor()

    dataset = ImageDataset(data_path, transform=transform, label_transform=label_transform)

    # Create a DataLoader
    dataloader = DataLoader(dataset, batch_size=4, shuffle=True)

    # Fetch a batch of data
    images, labels = next(iter(dataloader))
    print(images.shape, labels.shape)
