import base64
import os
from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC
from cryptography.fernet import Fernet

def encryption():
    print("\nWelcome to gostpy Encryption Tool...!!\n")

    def take_password():
        password_provided = input('Choose Password: ')
        if len(password_provided) < 1:
            print("Please provide a password.")
            return take_password()
        else:
            reconfirm_password = input("Confirm Password: ")
            if len(reconfirm_password) < 1:
                print("Please provide a password.")
                return take_password()
            elif password_provided != reconfirm_password:
                print("Passwords do not match. Please try again.")
                return take_password()
        return password_provided

    def encrypt_file(file_path, key):
        with open(file_path, 'rb') as f:
            data = f.read()
        fernet = Fernet(key)
        encrypted_data = fernet.encrypt(data)
        with open(file_path, 'wb') as f:
            f.write(encrypted_data)
        print("File '" + file_path + "' is Encrypted.")

    def encrypt_folder(folder_path, key):
        for root, dirs, files in os.walk(folder_path):
            for file_name in files:
                input_file = os.path.join(root, file_name)
                with open(input_file, 'rb') as f:
                    data = f.read()
                fernet = Fernet(key)
                encrypted_data = fernet.encrypt(data)
                with open(input_file, 'wb') as f:
                    f.write(encrypted_data)
                print("File '" + input_file + "' is Encrypted.")

    password_provided = take_password()
    password = password_provided.encode()
    salt = b'ghostpy'
    kdf = PBKDF2HMAC(
        algorithm=hashes.SHA256(),
        length=32,
        salt=salt,
        iterations=100000,
        backend=default_backend()
    )
    key = base64.urlsafe_b64encode(kdf.derive(password))
    files_path = input("\nEnter the path of your file or folder: ")

    if os.path.isfile(files_path):
        encrypt_file(files_path, key)
    elif os.path.isdir(files_path):
        encrypt_folder(files_path, key)
    else:
        print("Invalid file or folder path.")

if __name__ == "__main__":
    encryption()
