import sys

def dec_decrypt(int_list, int_key):
    """Decrypts a list of integers using the specified key."""
    crypt_bytes = bytearray(int_list)
    crypt_nibbles = convert_bytes_to_nibbles(crypt_bytes)
    key_nibbles = byte_to_nibbles(int_key)
    decrypted_nibbles = [decrypt_byte_nibbles(n, key_nibbles) for n in crypt_nibbles]
    decrypted_bytes = convert_nibbles_to_bytes(decrypted_nibbles)    
    return bytes_to_string(decrypted_bytes)

def convert_bytes_to_nibbles(byte_array):
    """Converts each byte in the array to four nibbles."""
    return [byte_to_nibbles(b) for b in byte_array]

def byte_to_nibbles(byte):
    """Splits a byte into four 2-bit nibbles."""
    return bytearray([(byte >> 6) & 0x03, (byte >> 4) & 0x03, (byte >> 2) & 0x03, byte & 0x03])

def convert_nibbles_to_bytes(nibbles_list):
    """Converts lists of nibbles back into bytes."""
    return bytearray([nibbles_to_byte(n) for n in nibbles_list])

def nibbles_to_byte(nibbles):
    """Combines four 2-bit nibbles into a byte."""
    return (nibbles[0] << 6) | (nibbles[1] << 4) | (nibbles[2] << 2) | nibbles[3]

def decrypt_byte_nibbles(nibble_quad, key_quad):
    """Applies decryption steps to a set of nibbles."""
    result = xor(nibble_quad, key_quad)
    result = swap_last_two(result)
    result = s_box_transform(result)
    result = xor(result, key_quad)
    result = subtract_last_two_first_two(result)
    result = swap_last_two(result)
    result = s_box_transform(result)
    result = xor(result, key_quad)
    return result

def xor(nibbles, key):
    """Performs XOR operation on two sets of nibbles."""
    return bytearray(n ^ k for n, k in zip(nibbles, key))

def swap_last_two(nibbles):
    """Swaps the last two nibbles."""
    return bytearray([nibbles[0], nibbles[1], nibbles[3], nibbles[2]])

def s_box_transform(nibbles):
    """Applies a simple substitution box (S-box) transformation."""
    s_box = {0: 2, 1: 0, 2: 3, 3: 1}
    return bytearray(s_box[n] for n in nibbles)

def subtract_last_two_first_two(nibbles):
    """Subtracts the last two nibbles from the first two, modulo 4."""
    return bytearray([(nibbles[0] - nibbles[2]) % 4, (nibbles[1] - nibbles[3]) % 4, nibbles[2], nibbles[3]])

def bytes_to_string(byte_list):
    """Converts a list of bytes to a string."""
    return ''.join(chr(b) for b in byte_list)
