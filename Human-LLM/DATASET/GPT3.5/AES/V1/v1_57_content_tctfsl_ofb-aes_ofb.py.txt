import random
import aes

# Function to divide a string into fixed-size chunks
def chunks(string, blockSize):
    for i in range(0, len(string), blockSize):
        yield string[i:i + blockSize]

# Function to generate a random initialization vector (IV) of a given length
def randomIV(length):
    IV = ""
    for l in range(0, length):
        IV += str(random.randint(0, 1))
    return IV

# Function to generate a dummy initialization vector (IV)
def dummyIV():
    ivArray = list("01010000")
    for i in range(0, len(ivArray)):
        ivArray[i] = int(ivArray[i])
    return ivArray

# Function to convert a value to its binary representation of a specified size
def binvalue(val, bitsize):
    binval = bin(val)[2:] if isinstance(val, int) else bin(ord(val))[2:]
    if len(binval) > bitsize:
        raise ValueError("binary value larger than the expected size")
    while len(binval) < bitsize:
        binval = "0" + binval
    return binval

# Function to split a string into smaller strings of length n
def nsplit(s, n):
    return [s[k:k + n] for k in range(0, len(s), n)]

# Function to convert a string to a bit array
def string_to_bit_array(text):
    array = []
    for char in text:
        binval = binvalue(char, 8)
        array.extend([int(x) for x in list(binval)])
    return array

# Function to convert a bit array to a string
def bit_array_to_string(array):
    res = ''.join([chr(int(y, 2)) for y in [''.join([str(x) for x in bytes]) for bytes in nsplit(array, 8)]])
    return res

# Function to convert a string of '0's and '1's to a list of integers
def string_to_ind_bit(string):
    array = list(string)
    for q in range(0, len(array)):
        array[q] = int(array[q])
    return array

# Function to convert a list of integers to a string of '0's and '1's
def ind_bit_to_string(array):
    string = ""
    for r in range(0, len(array)):
        string += str(array[r])
    return string

# Function to encrypt plaintext blocks
def encrypt(blockSize, plaintextBlockList, shiftRegister):
    key = "11001001110010011100100111001001110010011100100111001001110010011100100111001001110010011100100111001001110010011100100111001001"
    ciphertextBlockList = []
    for j in range(0, len(plaintextBlockList)):
        plaintextBlock = plaintextBlockList[j]
        tempRegister = string_to_ind_bit(aes.encryption(shiftRegister, key))
        ciphertextBlock = []
        for m in range(0, blockSize):
            ciphertextBlock.append(tempRegister[m] ^ plaintextBlock[m])
        shiftRegister = string_to_ind_bit(shiftRegister)
        for n in range(0, blockSize):
            shiftRegister.pop(0)
            shiftRegister.append(tempRegister[n])
        shiftRegister = ind_bit_to_string(shiftRegister)
        ciphertextBlockList.append(ciphertextBlock)
    return ciphertextBlockList

# Function to encrypt plaintext blocks with a one-bit error introduced in the second block
def wrongencrypt(blockSize, plaintextBlockList, shiftRegister):
    key = "11001001110010011100100111001001110010011100100111001001110010011100100111001001110010011100100111001001110010011100100111001001"
    ciphertextBlockList = []
    for j in range(0, len(plaintextBlockList)):
        plaintextBlock = plaintextBlockList[j]
        tempRegister = string_to_ind_bit(aes.encryption(shiftRegister, key))
        ciphertextBlock = []
        for m in range(0, blockSize):
            if j == 1 and m == 1:  # Introducing a one-bit error in the second block
                bit = tempRegister[m] ^ plaintextBlock[m]
                if bit == 1:
                    bit = 0
                else:
                    bit = 1
                ciphertextBlock.append(bit)
            else:
                ciphertextBlock.append(tempRegister[m] ^ plaintextBlock[m])
        shiftRegister = string_to_ind_bit(shiftRegister)
        for n in range(0, blockSize):
            shiftRegister.pop(0)
            shiftRegister.append(tempRegister[n])
        shiftRegister = ind_bit_to_string(shiftRegister)
        ciphertextBlockList.append(ciphertextBlock)
    return ciphertextBlockList

# Function to decrypt ciphertext blocks
def decrypt_blocks(decrypted_blocks):
    decrypted_text = []
    for k in range(0, len(decrypted_blocks)):
        for p in range(0, blockSize):
            decrypted_text.append(int(decrypted_blocks[k][p]))
    decrypted_text_string = bit_array_to_string(decrypted_text)
    return decrypted_text_string
