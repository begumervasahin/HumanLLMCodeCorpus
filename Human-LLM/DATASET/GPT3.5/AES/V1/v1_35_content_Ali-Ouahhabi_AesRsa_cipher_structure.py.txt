import pickle
import sys

class CipherStructure:
    def __init__(self, ciphered_text, key, initialization_vector, tag_sentence, signature):
        self.ciphered_text = ciphered_text
        self.key = key
        self.initialization_vector = initialization_vector
        self.tag_sentence = tag_sentence
        self.signature = signature
    
    def get_ciphered_text(self):
        return self.ciphered_text
    
    def get_key(self):
        return self.key
    
    def get_initialization_vector(self):
        return self.initialization_vector
    
    def get_tag_sentence(self):
        return self.tag_sentence
    
    def get_signature(self):
        return self.signature
    
    def dump_to_file(self, output_file):
        # Dump the instance to the output file using pickle
        pickle.dump(self, output_file)
    
    @staticmethod
    def load_from_file(input_file):
        try:
            # Load the object from the input file using pickle
            tmp = pickle.load(input_file)
            # Check if the loaded object is an instance of CipherStructure
            if isinstance(tmp, CipherStructure):
                return tmp
        except:
            # Print an error message and exit if the file is corrupted
            print(">> ERROR: Corrupted ciphered file")
            sys.exit()

# Example usage:
if __name__ == "__main__":
    # Create an instance of CipherStructure
    cipher = CipherStructure("ciphered_text", "key", "initialization_vector", "tag_sentence", "signature")
    
    # Dump the instance to a file
    with open("cipher_file.pkl", "wb") as f:
        cipher.dump_to_file(f)
    
    # Load the instance from the file
    with open("cipher_file.pkl", "rb") as f:
        loaded_cipher = CipherStructure.load_from_file(f)
    
    # Example: Accessing attributes of the loaded object
    print("Ciphered Text:", loaded_cipher.get_ciphered_text())
    print("Key:", loaded_cipher.get_key())
    print("Initialization Vector:", loaded_cipher.get_initialization_vector())
    print("Tag Sentence:", loaded_cipher.get_tag_sentence())
    print("Signature:", loaded_cipher.get_signature())
