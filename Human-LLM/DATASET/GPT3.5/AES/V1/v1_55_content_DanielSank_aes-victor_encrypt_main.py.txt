from constants import Nr
from encrypt_functions import get_key, get_pt_data, out_ct_data, print_hex
from encrypt_functions import key_expansion, add_round_key, sub_bytes, shift_rows, mix_columns

def main():
    # Read the key from key.txt and perform key expansion
    key = get_key()
    w = key_expansion(key)

    # Read plain text data from plaintex.txt
    state = get_pt_data()
    print('  Main: Finished GetPTData   state = ')
    print_hex(state)

    # Initial round
    rround = 0
    print('Main: round = ', rround)
    add_round_key(state, w, rround)

    # Perform AES encryption rounds
    for rround in range(1, Nr):
        print('  Main:                round = ', rround)
        sub_bytes(state)
        shift_rows(state)
        mix_columns(state)
        add_round_key(state, w, rround)

    # Final round
    rround = 14
    print('  Main:                round = ', rround)
    sub_bytes(state)
    shift_rows(state)
    add_round_key(state, w, rround)

    # Write encrypted data to CTData.txt
    out_ct_data(state)
    print('Encrypted Data written to CTData.txt')

if __name__ == "__main__":
    main()
