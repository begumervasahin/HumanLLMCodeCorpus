    p2 = str(hex(SBOX[int(state_array[4:6], 16)]))

        p2 = p2[:2] + '0' + p2[2:]
    p3 = str(hex(SBOX[int(state_array[6:8], 16)]))

def shift_rows(state_array):

        column[i] = column[i][i:] + column[i][:i]
    return column

def xor(var1, var2):
    result = str(int(var1, 16) ^ int(var2, 16))
    result = hex(int(result))[2:].zfill(32)
    return result
