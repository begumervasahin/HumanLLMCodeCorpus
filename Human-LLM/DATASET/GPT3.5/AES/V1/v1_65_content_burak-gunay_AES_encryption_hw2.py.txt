import os
import sys
from cryptography.hazmat.primitives.ciphers import Cipher, algorithms, modes
from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives import padding

def generate_key():
    """
    Generate a random AES key and save it to a file.
    """
    key_file = open('AESkey.key', 'wb')
    key_file.write(os.urandom(32))
    key_file.close()

def encrypt_file(msg_filename):
    """
    Encrypt a file using AES encryption with a generated key.
    """
    msg_file = open(msg_filename, 'rb')
    plaintext = msg_file.read()
    key_file = open('AESkey.key', 'rb')
    key = key_file.read()
    iv = os.urandom(16)
    iv_file = open(msg_filename + '.enc.iv', 'wb')
    iv_file.write(iv)

    padder = padding.PKCS7(256).padder()
    padded_plaintext = padder.update(plaintext)
    padded_plaintext += padder.finalize()

    backend = default_backend()
    cipher = Cipher(algorithms.AES(key), modes.CBC(iv), backend=backend)
    encryptor = cipher.encryptor()
    ct = encryptor.update(padded_plaintext) + encryptor.finalize()

    encrypted_file = open(msg_filename + '.enc', 'wb')
    encrypted_file.write(ct)

    msg_file.close()
    key_file.close()
    iv_file.close()
    encrypted_file.close()

def decrypt_file(msg_filename):
    """
    Decrypt an encrypted file using AES decryption with the stored key.
    """
    encrypted_file = open(msg_filename, 'rb')
    padded_ciphertext = encrypted_file.read()
    key_file = open('AESkey.key', 'rb')
    key = key_file.read()
    iv_file = open(msg_filename + '.iv', 'rb')
    iv = iv_file.read()

    backend = default_backend()
    cipher = Cipher(algorithms.AES(key), modes.CBC(iv), backend=backend)
    decryptor = cipher.decryptor()
    decrypted_msg = decryptor.update(padded_ciphertext) + decryptor.finalize()

    unpadder = padding.PKCS7(256).unpadder()
    ciphertext = unpadder.update(decrypted_msg) + unpadder.finalize()

    decrypted_file_name = msg_filename + 'DECRYPTED'
    print("Your decrypted message is saved in: {}".format(decrypted_file_name))
    decrypted_file = open(decrypted_file_name, 'wb')
    decrypted_file.write(ciphertext)

    key_file.close()
    iv_file.close()
    decrypted_file.close()

if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("Usage: python3 script_name.py generate|encrypt|decrypt FILENAME.EXTENSION")
        sys.exit(1)

    if sys.argv[1] == 'generate':
        generate_key()
        print("Key generated")
    elif sys.argv[1] == 'encrypt':
        encrypt_file(sys.argv[2])
        print("File encrypted")
    elif sys.argv[1] == 'decrypt':
        decrypt_file(sys.argv[2])
    else:
        print('Wrong command')
