import sys
import os
import struct
from Crypto.Cipher import AES
from Crypto.Hash import MD5, SHA256
import Crypto.Util.Counter

blockSize = 2048
padding = "@"

def encrypt(mode, IV, myKey, input_file, output_file):
    # Validate encryption mode
    if mode not in ["ECB", "CBC", "CFB", "OFB", "CTR"]:
        print("Available modes: ECB, CBC, CFB, OFB, CTR. Choose again!")
        sys.exit(1)

    # Map mode strings to AES encryption modes
    mode_map = {"ECB": AES.MODE_ECB, "CBC": AES.MODE_CBC,
                "CFB": AES.MODE_CFB, "OFB": AES.MODE_OFB,
                "CTR": AES.MODE_CTR}
    mode = mode_map[mode]

    # Hash the key and IV
    key_hash = SHA256.new(myKey.encode()).digest()
    IV_hash = MD5.new(IV.encode()).digest()

    # Create AES encryptor with the given mode, key, and IV
    if mode != AES.MODE_CTR:
        encryptor = AES.new(key_hash, mode, IV_hash)
    else:
        ctr = Crypto.Util.Counter.new(128, initial_value=int(IV, 16))
        encryptor = AES.new(key_hash, AES.MODE_CTR, counter=ctr)

    # Get file size and encrypt the input file
    filesize = os.path.getsize(input_file)
    with open(input_file, "rb") as fin:
        with open(output_file, "wb") as fout:
            fout.write(struct.pack('<Q', filesize))
            fout.write(IV_hash)
            while True:
                block = fin.read(blockSize)
                if len(block) == 0:
                    break
                elif len(block) % 16 != 0:
                    block = block + padding * (16 - len(block) % 16)
                fout.write(encryptor.encrypt(block))

def main(argv):
    try:
        opts, args = getopt.getopt(argv, "m:i:k:")
    except getopt.GetoptError:
        print("encrypt.py –m <mode> -i <IV> –k <key> <input_file> <output_file>")
        sys.exit(2)

    # Check for correct number of arguments
    if len(args) != 2:
        print("encrypt.py –m <mode> -i <IV> –k <key> <input_file> <output_file>")
        sys.exit(2)

    input_file, output_file = args
    mode, IV, myKey = "", "", ""
    for opt, arg in opts:
        if opt == "-h":
            print("encrypt.py -m <mode> -i <IV> -k <key> <input_file> <output_file>")
            sys.exit()
        elif opt == "-m":
            mode = arg.upper()
        elif opt == "-i":
            IV = arg
        elif opt == "-k":
            myKey = arg

    encrypt(mode, IV, myKey, input_file, output_file)

if __name__ == "__main__":
    main(sys.argv[1:])
