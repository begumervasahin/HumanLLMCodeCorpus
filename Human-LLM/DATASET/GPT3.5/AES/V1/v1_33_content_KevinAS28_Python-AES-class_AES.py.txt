from Crypto.Cipher import AES

def Padding(data, padder=b"\0"):
    """Pad data to ensure its length is a multiple of 16 bytes."""
    if not isinstance(data, bytes):
        try:
            data = data.encode("utf-8")
        except:
            pass
    while len(data) % 16 != 0:
        data += padder
    return data

def Depadding(data, depadder=b"\0"):
    """Remove padding from data."""
    return data.rstrip(depadder)

AES_PASSWORD = b"F<{Z[C?AY)L7!Yc@"  # Ensure the password is in bytes
AES_IV = b"nrB27PNZTnn-N^dR"         # Ensure the IV is in bytes

def Encrypt_AES(data):
    """Encrypt data using AES CBC mode."""
    aes = AES.new(AES_PASSWORD, AES.MODE_CBC, AES_IV)
    return aes.encrypt(Padding(data))

def Decrypt_AES(data):
    """Decrypt data using AES CBC mode."""
    aes = AES.new(AES_PASSWORD, AES.MODE_CBC, AES_IV)
    return Depadding(aes.decrypt(data))
