from Crypto_Examples.DES import run_des
from Crypto_Examples.AES import AES
from Crypto_Examples.RSA import RSA
from Crypto_Examples.MD5 import MD5

def is_hex(string):
    try:
        int(string, 16)
        return True
    except ValueError:
        return False

def get_hex(text):
    text_byte_array = list(map(bin, bytearray(text, encoding='utf-8')))
    text_byte_array_string = ''.join([foo[2:] for foo in text_byte_array])
    return str(hex(int(text_byte_array_string, 2))[2:].upper())

def get_hex_array(string):
    return [int(string[i:i + 2], 16) for i in range(0, len(string), 2)]

def fixed_len_hex(string):
    return '{0:0{1}X}'.format(int(string, 16), len(string))

def main():
    print("Welcome to Crypto Flask Application!")
    while True:
        print("\nAvailable Options:")
        print("1. DES")
        print("2. AES")
        print("3. RSA")
        print("4. MD5")
        print("5. Quit")

        choice = input("Enter your choice (1-5): ")

        if choice == '1':
            message = input("Enter the message: ")
            key = input("Enter the key: ")

            if not is_hex(key):
                print("Key must be in hexadecimal format.")
                continue

            action = input("Enter 'encrypt', 'decrypt', or 'both': ")
            if action not in ['encrypt', 'decrypt', 'both']:
                print("Invalid action.")
                continue

            result = run_des(message, key, action)
            print(result)

        elif choice == '2':
            message = input("Enter the message: ")
            key = input("Enter the key: ")

            if not is_hex(key) or not is_hex(message):
                print("Message and key must be in hexadecimal format.")
                continue

            message = fixed_len_hex(message)
            key = fixed_len_hex(key)

            aes = AES()
            aes_data = aes.do_rounds(get_hex_array(message), get_hex_array(key))
            print(aes_data)

        elif choice == '3':
            message = int(input("Enter the message (integer): "))
            e = int(input("Enter the public exponent (e): "))
            p = int(input("Enter the prime number (p): "))
            q = int(input("Enter the prime number (q): "))

            rsa = RSA()
            result = rsa.do_rsa(message, p, q, e)
            print(result)

        elif choice == '4':
            message = input("Enter the message: ")
            action = input("Enter 'all_steps' to show all steps: ")

            md5 = MD5()
            hash = md5.get_md5_hash(message, show_all=(action == 'all_steps'))
            print(hash)

        elif choice == '5':
            print("Exiting...")
            break

        else:
            print("Invalid choice. Please choose again.")

if __name__ == "__main__":
    main()
