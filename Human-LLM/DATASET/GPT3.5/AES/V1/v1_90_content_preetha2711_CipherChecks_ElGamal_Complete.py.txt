# Miller-Rabin primality test
def miller_rabin(number):
    B = False
    num_1 = number - 1
    k = 0
    while(num_1 % 2 == 1 ):
        num_1 = num_1/2
        k += 1
    m = num_1//(2**k)
    a = random.randint(0, num_1)
    b = pow(a, m, number)
    if b == 1:
        B = True
    else:
        for i in range(k):
            if b % number == -1:
                B = True
            else:
                b = pow(b, 2, number)
    return B

# Generate a prime number
def generate_prime(length=300):
    flag = False
    while flag != True:
        random_num = random.getrandbits(length)
        random_num |= (1 << length - 1) | 1
        flag = miller_rabin(random_num)
        prime_number = random_num
        return prime_number

# Get two prime numbers p and q
def get_two_primes():
    flag = False
    while flag != True:
        p = generate_prime()
        q = 2 * p + 1
        flag = miller_rabin(q)
    return p, q

# Lucas test for primitive roots
def lucas_test(integer):
    p = integer
    Q = []
    for i in range(2, p):
        if ((p-1) % i == 0):
            Q.append(i)
    prim_roots = []
    for g in range(2, p):
        flag = True
        for q in Q:
            if (g ** int((p-1)/q)) % p == 1:
                flag = False
                break
        if flag == True:
            prim_roots.append(g)
    return prim_roots
