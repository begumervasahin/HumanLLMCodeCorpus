from odoo import models, api, fields

class ExtraschoolEncodageManuelValidationWizard(models.TransientModel):
    _name = 'extraschool.encodage_manuel_validation_wizard'
    
    place_id = fields.Many2one('extraschool.place')

    @api.multi
    def validate(self):
        prestations = self.env['extraschool.prestation_times_encodage_manuel'].browse(self._context.get('active_ids'))
        prestations.validate()
        return True

    @api.multi
    def set_to_draft(self):
        prestations = self.env['extraschool.prestation_times_encodage_manuel'].browse(self._context.get('active_ids'))
        prestations.set_to_draft()
        return True

    @api.multi
    def change_place_id(self):
        prestations = self.env['extraschool.prestation_times_encodage_manuel'].browse(self._context.get('active_ids'))
        for prestation in prestations:
            prestation.set_to_draft()
            prestation.write({'place_id': self.place_id.id})
            prestation.validate()
        return True
