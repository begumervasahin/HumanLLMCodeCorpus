import math

def encrypt_message(message, key):
    cipher = ""
    key_length = len(key)
    message_length = len(message)
    num_rows = math.ceil(message_length / key_length)
    padded_message = message.ljust(num_rows * key_length, '_')
    matrix = [list(padded_message[i:i+key_length]) for i in range(0, len(padded_message), key_length)]
    
    sorted_key = sorted(key)
    for k in range(key_length):
        index = key.index(sorted_key[k])
        cipher += ''.join(row[index] for row in matrix)
    
    return cipher

def decrypt_message(cipher, key):
    message = ""
    key_length = len(key)
    num_rows = math.ceil(len(cipher) / key_length)
    
    sorted_key = sorted(key)
    matrix = []
    for i in range(num_rows):
        matrix.append(list(cipher[i*key_length:(i+1)*key_length]))
    
    for k in range(key_length):
        index = key.index(sorted_key[k])
        message += ''.join(matrix[row][index] for row in range(num_rows))
    
    return message.rstrip('_')
