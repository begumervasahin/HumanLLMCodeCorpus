import socket
from AES256 import AES256

def main():
    # Set up the server IP address and port
    IP = '127.0.0.1'
    Port = 5555
    
    # Create a socket object
    server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    
    # Connect to the server
    server_socket.connect((IP, Port))
    
    # Create an instance of AES256 for encryption and decryption
    aes = AES256("YourSecretKeyHere")
    
    # Send and receive messages to/from the server
    while True:
        # Send a message to the server
        msg = input("Enter message: ")
        encrypted_msg = aes.encrypt(msg)
        server_socket.send(encrypted_msg)
        
        # Receive a response from the server
        encrypted_data = server_socket.recv(1024)
        decrypted_data = aes.decrypt(encrypted_data)
        print("Server response:", decrypted_data)
        
    # Close the connection
    server_socket.close()

if __name__ == "__main__":
    main()
