        cost_to_a = nodes['b'][2] + distances['A']
        cost_to_d = nodes['b'][4] + distances['D']
        if cost_to_a > cost_to_d:
            path.append('D')
            current_node = 'd'
        else:
            path.append('A')
            current_node = 'a'
    # Continue similar checks for other nodes...
    else:

# Print the path
