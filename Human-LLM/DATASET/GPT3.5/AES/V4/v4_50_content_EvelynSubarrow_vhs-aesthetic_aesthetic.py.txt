def parse_arguments():
    """Parse command-line arguments."""
    parser = argparse.ArgumentParser(description='Apply effects to an image.')
    parser.add_argument('input_file', help='Input image file path.')
    parser.add_argument('output_file', help='Output image file path.')
    parser.add_argument('-t', '--text', help='Text to add to the image, formatted as "[x,y]Text".')
    parser.add_argument('-s', '--text_size', type=int, default=50, help='Font size for the added text.')
    parser.add_argument('--ghost-il', action='store_true', help='Use inline Python for ghosting (slower).')
    parser.add_argument('--ghost-np', action='store_true', default=True, help='Use NumPy for ghosting (faster).')
    parser.add_argument('-g', '--ghost_width', type=float, default=1.0, help='Width of the ghost effect as a percentage of the image width.')
    parser.add_argument('-y', '--chrominance_static', type=int, default=7, help='Maximum +- variation for chrominance in YUV color space.')
    parser.add_argument('-v', '--luminance_static', type=int, default=11, help='Maximum +- variation for luminance in YUV color space.')
    parser.add_argument('-r', '--rescale', type=decimal.Decimal, help='Rescale the image (as a decimal fraction of the original size).')
    return parser.parse_args()

class PrintTimer:
    """Context manager for timing operations."""
    def __init__(self, description):
        self.description = description

    def __enter__(self):
        self.start_time = timeit.default_timer()
        return self

    def __exit__(self, *args):
        elapsed_time = timeit.default_timer() - self.start_time
        print(f"{self.description} took {elapsed_time:.2f} seconds.")
