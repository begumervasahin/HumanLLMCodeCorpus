import matplotlib.pyplot as plt
import numpy as np
from pysph.solver.utils import get_files, load

def process_data(folder):
    """Process simulation data to compute normalized vmax and average KE."""
    vmax_list, KE_list, time_list = [], [], []

    # Load initial data to normalize future values
    initial_data = load(f'{folder}/vortex_spin_down_0.hdf5')['arrays']['fluid']
    u, v = initial_data.u, initial_data.v
    norm = np.max(np.sqrt(u**2 + v**2))
    KE_norm = np.average(initial_data.m * 0.5 * (u**2 + v**2))

    # Iterate through files in the directory to process data
    for file_name in get_files(folder):
        data = load(file_name)
        u, v = data['arrays']['fluid'].u, data['arrays']['fluid'].v
        vmag = np.sqrt(u**2 + v**2)
        vmax = np.max(vmag) / norm
        KE = np.average(data['arrays']['fluid'].m * 0.5 * vmag**2) / KE_norm
        time = data['solver_data']['t']

        vmax_list.append(vmax)
        KE_list.append(KE)
        time_list.append(time)

    return time_list, vmax_list, KE_list

def plot_results(time_lists, vmax_lists, KE_lists):
    """Plot and save the results for normalized velocity and average kinetic energy."""
    scenarios = ['C = 1.25 (TVF)', 'C = 2.5 (TVF)', 'C = 5 (TVF)']

    # Plot normalized velocity
    plt.figure(figsize=(10, 6))
    for time, vmax, scenario in zip(time_lists, vmax_lists, scenarios):
        plt.plot(time, vmax, label=scenario)
    plt.title('(TVF) Variation of Normalized Velocity with Time')
    plt.xlabel('Time (s)')
    plt.ylabel('Normalized Velocity')
    plt.legend()
    plt.savefig('results_TVF_VMAX.png')
    plt.clf()

    # Plot normalized average kinetic energy
    plt.figure(figsize=(10, 6))
    for time, KE, scenario in zip(time_lists, KE_lists, scenarios):
        plt.plot(time, KE, label=scenario)
    plt.title('(TVF) Variation of Average Kinetic Energy with Time')
    plt.xlabel('Time (s)')
    plt.ylabel('Normalized Average Kinetic Energy')
    plt.legend()
    plt.savefig('results_TVF_KE.png')
    plt.clf()

# Directories for each simulation scenario
folders = [
    'TVFoutput/a/vortex_spin_down_output',
    'TVFoutput/b/vortex_spin_down_output',
    'TVFoutput/c/vortex_spin_down_output'
]

# Process data for each scenario
results = [process_data(folder) for folder in folders]

# Unpack results
time_lists, vmax_lists, KE_lists = zip(*results)

# Plot results
plot_results(time_lists, vmax_lists, KE_lists)
