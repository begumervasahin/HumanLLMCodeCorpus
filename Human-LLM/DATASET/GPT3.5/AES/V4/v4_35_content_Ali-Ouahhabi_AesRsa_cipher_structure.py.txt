import pickle
import sys

class CipherStructure:
    """
    Represents the structure of ciphered data, including ciphered text, key,
    initialization vector, tag sentence, and signature.
    """
    def __init__(self, ciphered_text, key, initialization_vector, tag_sentence, signature):
        """
        Initializes the CipherStructure object with the provided parameters.
        """
        self.ciphered_text = ciphered_text
        self.key = key
        self.initialization_vector = initialization_vector
        self.tag_sentence = tag_sentence
        self.signature = signature

    def get_ciphered_text(self):
        """Returns the ciphered text."""
        return self.ciphered_text

    def get_key(self):
        """Returns the key."""
        return self.key

    def get_initialization_vector(self):
        """Returns the initialization vector."""
        return self.initialization_vector

    def get_tag_sentence(self):
        """Returns the tag sentence."""
        return self.tag_sentence

    def get_signature(self):
        """Returns the signature."""
        return self.signature

    def dump_to_file(self, output_file):
        """
        Dumps the CipherStructure object to the output file using pickle.
        """
        pickle.dump(self, output_file)

    @staticmethod
    def load_from_file(input_file):
        """
        Loads a CipherStructure object from the input file using pickle.
        """
        try:
            loaded_object = pickle.load(input_file)
            if isinstance(loaded_object, CipherStructure):
                return loaded_object
        except:
            print(">> ERROR: Corrupted ciphered file")
            sys.exit()
