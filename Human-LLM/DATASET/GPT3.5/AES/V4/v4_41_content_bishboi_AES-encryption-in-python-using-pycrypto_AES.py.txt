import os
import base64
from Crypto.Cipher import AES
from Crypto import Random
from Crypto.Random import get_random_bytes
from Crypto.Util import Counter

BLOCK_SIZE = 16

def pad(s):
    padding = BLOCK_SIZE - len(s) % BLOCK_SIZE
    return s + bytes([padding]) * padding

def unpad(s):
    padding = s[-1]
    return s[:-padding]

class CBC:
    def __init__(self, key):
        self.key = key

    def encrypt(self, plaintext, iv):
        plaintext = pad(plaintext)
        cipher = AES.new(self.key, AES.MODE_CBC, iv)
        encrypted = cipher.encrypt(plaintext)
        return base64.b64encode(iv + encrypted)

    def decrypt(self, ciphertext):
        ciphertext = base64.b64decode(ciphertext)
        iv = ciphertext[:BLOCK_SIZE]
        cipher = AES.new(self.key, AES.MODE_CBC, iv)
        decrypted = cipher.decrypt(ciphertext[BLOCK_SIZE:])
        return unpad(decrypted)

class ECB:
    def __init__(self, key):
        self.key = key

    def encrypt(self, plaintext):
        plaintext = pad(plaintext)
        cipher = AES.new(self.key, AES.MODE_ECB)
        encrypted = cipher.encrypt(plaintext)
        return base64.b64encode(encrypted)

    def decrypt(self, ciphertext):
        ciphertext = base64.b64decode(ciphertext)
        cipher = AES.new(self.key, AES.MODE_ECB)
        decrypted = cipher.decrypt(ciphertext)
        return unpad(decrypted)

class OFB:
    def __init__(self, key):
        self.key = key

    def encrypt(self, plaintext, iv):
        plaintext = pad(plaintext)
        cipher = AES.new(self.key, AES.MODE_OFB, iv)
        encrypted = cipher.encrypt(plaintext)
        return base64.b64encode(iv + encrypted)

    def decrypt(self, ciphertext):
        ciphertext = base64.b64decode(ciphertext)
        iv = ciphertext[:BLOCK_SIZE]
        cipher = AES.new(self.key, AES.MODE_OFB, iv)
        decrypted = cipher.decrypt(ciphertext[BLOCK_SIZE:])
        return unpad(decrypted)

class CTR:
    def __init__(self, key):
        self.key = key

    def encrypt(self, plaintext, iv):
        plaintext = pad(plaintext)
        ctr = Counter.new(128)
        cipher = AES.new(self.key, AES.MODE_CTR, counter=ctr)
        encrypted = cipher.encrypt(plaintext)
        return base64.b64encode(iv + encrypted)

    def decrypt(self, ciphertext):
        ciphertext = base64.b64decode(ciphertext)
        iv = ciphertext[:BLOCK_SIZE]
        ctr = Counter.new(128)
        cipher = AES.new(self.key, AES.MODE_CTR, counter=ctr)
        decrypted = cipher.decrypt(ciphertext[BLOCK_SIZE:])
        return unpad(decrypted)

def generate_key():
    return os.urandom(BLOCK_SIZE)

def generate_iv():
    return Random.new().read(BLOCK_SIZE)

def generate_plaintext():
    return get_random_bytes(BLOCK_SIZE)

def generate_counter():
    return Counter.new(128)

def main():
    plaintext = generate_plaintext().hex()
    key = generate_key()
    iv = generate_iv()

    print('plaintext:', plaintext)
    print('iv:', iv.hex())
    print('key:', key.hex())

    cbc_cipher = CBC(key)
    ciphertext = cbc_cipher.encrypt(plaintext, iv)
    print('encrypted in CBC:', ciphertext)
    plaintext_decrypted = cbc_cipher.decrypt(ciphertext)
    print('decrypted in CBC:', plaintext_decrypted)

    ecb_cipher = ECB(key)
    ciphertext = ecb_cipher.encrypt(plaintext)
    print('encrypted in ECB:', ciphertext)
    plaintext_decrypted = ecb_cipher.decrypt(ciphertext)
    print('decrypted in ECB:', plaintext_decrypted)

    ofb_cipher = OFB(key)
    ciphertext = ofb_cipher.encrypt(plaintext, iv)
    print('encrypted in OFB:', ciphertext)
    plaintext_decrypted = ofb_cipher.decrypt(ciphertext)
    print('decrypted in OFB:', plaintext_decrypted)

    ctr_cipher = CTR(key)
    ciphertext = ctr_cipher.encrypt(plaintext, iv)
    print('encrypted in CTR:', ciphertext)
    plaintext_decrypted = ctr_cipher.decrypt(ciphertext)
    print('decrypted in CTR:', plaintext_decrypted)

    print('CTR:', generate_counter())

if __name__ == "__main__":
    main()
