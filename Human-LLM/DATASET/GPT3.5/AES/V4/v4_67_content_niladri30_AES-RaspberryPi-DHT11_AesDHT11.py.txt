import base64
import hashlib
from Crypto.Cipher import AES
from Crypto import Random
import Adafruit_DHT

# Define DHT sensor and GPIO pin
sensor = Adafruit_DHT.DHT11
gpio = 18

# Define block size for padding
BLOCK_SIZE = 16

# Padding function
def pad(s):
    pad_length = BLOCK_SIZE - len(s) % BLOCK_SIZE
    return s + pad_length * chr(pad_length)

# Unpadding function
def unpad(s):
    return s[:-ord(s[-1])]

# Read humidity and temperature from DHT sensor
humidity, temperature = Adafruit_DHT.read_retry(sensor, gpio)

# Define password and message
password = "Neel"
message = "Temperature={0:0.1f}*C  Humidity={1:0.1f}%".format(temperature, humidity)

# Encryption function
def encrypt(raw, password):
    private_key = hashlib.sha256(password.encode("utf-8")).digest()
    raw = pad(raw)
    iv = Random.new().read(AES.block_size)
    cipher = AES.new(private_key, AES.MODE_CBC, iv)
    return base64.b64encode(iv + cipher.encrypt(raw))

# Decryption function
def decrypt(enc, password):
    private_key = hashlib.sha256(password.encode("utf-8")).digest()
    enc = base64.b64decode(enc)
    iv = enc[:AES.block_size]
    cipher = AES.new(private_key, AES.MODE_CBC, iv)
    return unpad(cipher.decrypt(enc[AES.block_size:]))
