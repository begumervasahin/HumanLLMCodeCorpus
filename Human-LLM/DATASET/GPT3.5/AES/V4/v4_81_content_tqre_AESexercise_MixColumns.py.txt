import gf

def do(state):
    """
    Apply the MixColumns operation to the AES state matrix.

    Args:
        state (bytes): The state matrix.

    Returns:
        bytes: The new state matrix after the MixColumns operation.
    """
    fixed_matrix = bytearray([0x02, 0x03, 0x01, 0x01])
    columns = [state[i * 4: (i + 1) * 4] for i in range(4)]
    mixed_columns = [mix_column(column) for column in columns]
    return bytearray().join(mixed_columns)

def inverse(m):
    """
    Compute the inverse of the MixColumns operation.

    Args:
        m (bytes): The input matrix.

    Returns:
        bytes: The inverse matrix.
    """
    fixed_matrix = bytearray([0x0e, 0x0b, 0x0d, 0x09])
    columns = [m[i * 4: (i + 1) * 4] for i in range(4)]
    inverse_columns = [inv_column(column) for column in columns]
    return bytearray().join(inverse_columns)

def mix_column(column):
    """
    Mix a single column of the AES state matrix.

    Args:
        column (bytes): The input column.

    Returns:
        bytes: The mixed column.
    """
    fixed_matrix = bytearray([0x02, 0x03, 0x01, 0x01])
    mix_column_matrix = bytearray()

    for i in range(4):
        rijndael = gf.makeblist(0x11b)
        resulting_list = []

        for j in range(4):
            fixed_byte_list = gf.makeblist(fixed_matrix[j])
            column_vector = gf.makeblist(column[j])
            resulting_byte = gf.mul(fixed_byte_list, column_vector)

            if gf.value(resulting_byte) > 255:
                resulting_byte = gf.div(resulting_byte, rijndael)[1]

            resulting_list.append(resulting_byte)

        final_value = gf.add(resulting_list[0], resulting_list[1])
        final_value = gf.add(final_value, resulting_list[2])
        final_value = gf.add(final_value, resulting_list[3])
        mix_column_matrix.append(gf.value(final_value))
        fixed_matrix = gf.circrotateright(fixed_matrix)

    return mix_column_matrix

def inv_column(column):
    """
    Compute the inverse of a single column.

    Args:
        column (bytes): The input column.

    Returns:
        bytes: The inverse column.
    """
    fixed_matrix = bytearray([0x0e, 0x0b, 0x0d, 0x09])
    inv_column_matrix = bytearray()

    for i in range(4):
        rijndael = gf.makeblist(0x11b)
        resulting_list = []

        for j in range(4):
            fixed_bit_list = gf.makeblist(fixed_matrix[j])
            column_vector = gf.makeblist(column[j])
            resulting_byte = gf.mul(fixed_bit_list, column_vector)

            while gf.value(resulting_byte) > 255:
                resulting_byte = gf.div(rijndael, resulting_byte)[1]

            resulting_list.append(resulting_byte)

        final_value = gf.add(resulting_list[0], resulting_list[1])
        final_value = gf.add(final_value, resulting_list[2])
        final_value = gf.add(final_value, resulting_list[3])
        inv_column_matrix.append(gf.value(final_value))
        fixed_matrix = gf.circrotateright(fixed_matrix)

    return inv_column_matrix
