import os
import smtplib
from getpass import getpass
from Crypto.Cipher import AES
from email.message import EmailMessage

# Get email and password from environment variables
email = os.environ.get('EMAIL')
passwd = os.environ.get('PASSWORD')

# Generate random key and initialization vector (IV)
key = os.urandom(16)
iv = os.urandom(16)

# Get original data from user input, ensuring it's 16 bytes
original_data = getpass("Enter the '16-byte' message to be encrypted:\n")        
if len(original_data) > 16:
    original_data = original_data[:16]
elif len(original_data) < 16:
    original_data += ' ' * (16 - len(original_data))
original_data = str.encode(original_data)

# Read dummy data from an image file
image_path = input("Enter the path to an image file:\n")
with open(image_path, 'rb') as f:
    dummy_data = f.read()

# Encrypt the original data using AES-CBC mode
aes_enc = AES.new(key, AES.MODE_CBC, iv)
enc = aes_enc.encrypt(original_data)

# Combine dummy data, encrypted data, IV, and key
send_data = dummy_data + enc + iv + key

# Compose email message
msg = EmailMessage()
msg['To'] = input("Enter the email-ID of the receiver:\n")
msg['Subject'] = 'Cryptography Demo'
msg['From'] = email
msg.set_content(str(input("Type any extra message or instructions to be sent:\n")))
image_name = input("Enter the name of the image to be sent (without the extension):\n")
msg.add_attachment(send_data, maintype='image', subtype='jpeg', filename=image_name + '.jpg')

# Send email using SMTP
with smtplib.SMTP_SSL('smtp.gmail.com', 465) as smtp:
    smtp.login(email, passwd)
    smtp.send_message(msg)

print('The message has been successfully encrypted, embedded, and sent!')
