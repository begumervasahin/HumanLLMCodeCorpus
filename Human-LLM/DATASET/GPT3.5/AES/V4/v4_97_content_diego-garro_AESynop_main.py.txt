from datetime import datetime
from utils import tools as tl
from synoptic.section1 import Group_iRixhVV

def fetch_and_format_url():
    """
    Fetches today's date and 6 hours prior, then formats the URL for fetching synoptic reports.
    Returns:
        str: The formatted URL.
    """
    today, today_minus_6h = tl.fecha_para_synops()
    url_es = ('http://ogimet.com/display_synops2.php?lugar=78764+78762+78774+78767&tipo=ALL&ord=DIR&nil=SI'
              '&fmt=txt&ano={}&mes={}&day={}&hora={}&anof={}&mesf={}&dayf={}&horaf={}&enviar=Ver').format(
                  today_minus_6h[0], str(today_minus_6h[1]).zfill(2), str(today_minus_6h[2]).zfill(2), 
                  str(today_minus_6h[3]).zfill(2), today[0], str(today[1]).zfill(2), 
                  str(today[2]).zfill(2), str(today[3]).zfill(2))
    return url_es

def evaluate_synoptic_reports(filename, station_code):
    """
    Extracts and evaluates synoptic reports from a given file for a specific weather station.
    Args:
        filename (str): The file containing synoptic reports.
        station_code (int): The code of the weather station to evaluate reports for.
    """
    synoptic_reports = tl.extraer_synops(filename, station_code)
    for report in synoptic_reports:
        print(report)
        report_parts = report.split(' ')
        evaluator = Group_iRixhVV(report_parts[4], log_file)
        evaluator.evaluate_group()

if __name__ == "__main__":
