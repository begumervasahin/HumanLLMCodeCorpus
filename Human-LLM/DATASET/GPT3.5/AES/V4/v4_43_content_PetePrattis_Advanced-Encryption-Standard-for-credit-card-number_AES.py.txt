from Crypto.Cipher import AES
from Crypto import Random

# Define the XOR function for binary strings
def xor(a, b):
    result = ''
    for i in range(len(a)):
        result += str(int(a[i]) ^ int(b[i]))
    return result

# Encrypt the credit card number using AES encryption
def encrypt(card_number, rounds, produce_keys):
    # Convert the credit card number to a binary string
    encoded_number = '{0:054b}'.format(int(card_number))        
    left_half = encoded_number[:27]
    right_half = encoded_number[27:]
    round_keys = []

    # Generate round keys if needed
    if produce_keys:
        for i in range(rounds):
            key = Random.new().read(16).encode('hex')
            round_keys.append(key)

    # Encrypt the number for the specified number of rounds
    for i in range(rounds):
        key = round_keys[i]
        obj = AES.new(key.decode('hex'))
        hex_right_half = '{:08x}'.format(int(right_half + '0'*5, 2))
        block = hex_right_half + '0'*23 + str(i+1)
        encrypted_result = obj.encrypt(block.decode('hex')).encode('hex')[:7] + '0'
        right_half = xor(encrypted_result, left_half + '0'*5)
        left_half = right_half

    return (left_half , right_half)

# Decrypt the cipher using AES decryption
def decrypt(cipher, rounds):
    encoded_number = '{0:054b}'.format(int(cipher))
    left_half = encoded_number[:27]
    right_half = encoded_number[27:]
    for i in range(rounds, 0, -1):
        key = round_keys[i-1]
        obj = AES.new(key.decode('hex'))
        hex_left_half = '{:08x}'.format(int(left_half + '0'*5, 2))
        block = hex_left_half + '0'*23 + str(i)
        encrypted_result = obj.encrypt(block.decode('hex')).encode('hex')[:7] + '0'
        left_half = xor(encrypted_result, right_half + '0'*5)
        right_half = left_half
    return (left_half , right_half)
