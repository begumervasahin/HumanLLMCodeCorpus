import sys
import getopt
import os
import struct
from Crypto.Cipher import AES
from Crypto.Hash import MD5, SHA256
import Crypto.Util.Counter

BLOCK_SIZE = 2048
PADDING = "@"

def encrypt_file(mode, IV, key, input_file, output_file):
    # Validate encryption mode
    valid_modes = ["ECB", "CBC", "CFB", "OFB", "CTR"]
    if mode not in valid_modes:
        print("Invalid encryption mode. Available modes:", ", ".join(valid_modes))
        sys.exit(1)

    # Map mode strings to AES encryption modes
    aes_modes = {
        "ECB": AES.MODE_ECB,
        "CBC": AES.MODE_CBC,
        "CFB": AES.MODE_CFB,
        "OFB": AES.MODE_OFB,
        "CTR": AES.MODE_CTR
    }
    aes_mode = aes_modes[mode]

    # Hash the key and IV
    key_hash = SHA256.new(key.encode()).digest()
    IV_hash = MD5.new(IV.encode()).digest()

    # Create AES encryptor with the given mode, key, and IV
    if aes_mode != AES.MODE_CTR:
        encryptor = AES.new(key_hash, aes_mode, IV_hash)
    else:
        ctr = Crypto.Util.Counter.new(128, initial_value=int(IV, 16))
        encryptor = AES.new(key_hash, AES.MODE_CTR, counter=ctr)

    # Get file size and encrypt the input file
    filesize = os.path.getsize(input_file)
    with open(input_file, "rb") as fin:
        with open(output_file, "wb") as fout:
            fout.write(struct.pack('<Q', filesize))
            fout.write(IV_hash)
            while True:
                block = fin.read(BLOCK_SIZE)
                if not block:
                    break
                if len(block) % 16 != 0:
                    block += PADDING * (16 - len(block) % 16)
                fout.write(encryptor.encrypt(block))

def main(argv):
    try:
        opts, args = getopt.getopt(argv, "m:i:k:")
    except getopt.GetoptError:
        print("Usage: encrypt.py -m <mode> -i <IV> -k <key> <input_file> <output_file>")
        sys.exit(2)

    # Check for correct number of arguments
    if len(args) != 2:
        print("Usage: encrypt.py -m <mode> -i <IV> -k <key> <input_file> <output_file>")
        sys.exit(2)

    input_file, output_file = args
    mode, IV, key = "", "", ""
    for opt, arg in opts:
        if opt == "-h":
            print("Usage: encrypt.py -m <mode> -i <IV> -k <key> <input_file> <output_file>")
            sys.exit()
        elif opt == "-m":
            mode = arg.upper()
        elif opt == "-i":
            IV = arg
        elif opt == "-k":
            key = arg

    encrypt_file(mode, IV, key, input_file, output_file)

if __name__ == "__main__":
    main(sys.argv[1:])
