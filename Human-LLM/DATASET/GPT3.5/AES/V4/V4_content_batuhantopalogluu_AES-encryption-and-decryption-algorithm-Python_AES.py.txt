# AES Mix Columns Matrix
mix_matrix = [
    [0x02, 0x03, 0x01, 0x01],
    [0x01, 0x02, 0x03, 0x01],
    [0x01, 0x01, 0x02, 0x03],
    [0x03, 0x01, 0x01, 0x02]

# AES Round Constants
RCON = [
    [0x01, 0x00, 0x00, 0x00],
    [0x02, 0x00, 0x00, 0x00],
    [0x04, 0x00, 0x00, 0x00],
    [0x08, 0x00, 0x00, 0x00],
    [0x10, 0x00, 0x00, 0x00],
    [0x20, 0x00, 0x00, 0x00],
    [0x40, 0x00, 0x00, 0x00],

    [0x1b, 0x00, 0x00, 0x00],
    [0x36, 0x00, 0x00, 0x00]
]

# SubBytes transformation
def subBytes(state):
    new_state = [[0 for _ in range(4)] for _ in range(4)]
    for x in range(4):
        for y in range(4):
            new_state[x][y] = Sbox[state[x][y] >> 4][state[x][y] & 0x0f]
    return new_state

# ShiftRows transformation
def shiftRows(state):
    new_state = [[0 for _ in range(4)] for _ in range(4)]
    for i in range(4):
        for j in range(4):
            new_state[i][(j + i) % 4] = state[i][j]
    return new_state

# MixColumns transformation
def mixColumns(state):
    new_state = [[0 for _ in range(4)] for _ in range(4)]
    for i in range(4):
        for j in range(4):
            new_state[i][j] = mix_column(state, i, j)
    return new_state

# MixColumn operation for a single column
def mix_column(state, col, row):
    val = 0
    for i in range(4):
        val ^= gf_mul(state[i][col], mix_matrix[row][i])
    return val

# Galois Field multiplication
def gf_mul(a, b):
    p = 0
    for _ in range(8):
        if b & 1:
            p ^= a
        hi_bit_set = a & 0x80
        a <<= 1
        if hi_bit_set:
            a ^= 0x1b
        b >>= 1
    return p

# Key expansion
def keySchedule(cipher_key, roundNumber):
    table = [[0 for _ in range(4)] for _ in range(4)]
    for i in range(4):
        for j in range(4):
            table[i][j] = cipher_key[i][j]
    rot_word = [Sbox[table[1][(roundNumber + 1) * 4 - 1] >> 4][table[1][(roundNumber + 1) * 4 - 1] & 0x0f],
                Sbox[table[2][(roundNumber + 1) * 4 - 1] >> 4][table[2][(roundNumber + 1) * 4 - 1] & 0x0f],
                Sbox[table[3][(roundNumber + 1) * 4 - 1] >> 4][table[3][(roundNumber + 1) * 4 - 1] & 0x0f],
                Sbox[table[0][(roundNumber + 1) * 4 - 1] >> 4][table[0][(roundNumber + 1) * 4 - 1] & 0x0f]]
    for i in range(4):
        for j in range(4):
            if i == 0:
                table[j][(roundNumber + 1) * 4 + i] = (table[j][roundNumber * 4 + i] ^ rot_word[j] ^ RCON[roundNumber][j])
            else:
                table[j][(roundNumber + 1) * 4 + i] = table[j][roundNumber * 4 + i] ^ table[j][(roundNumber + 1) * 4 + i - 1]
    return table

# Encryption function
def encryption(state, cipher_key):
    key_schedule = [[0 for _ in range(44)] for _ in range(4)]
    for i in range(4):
        for j in range(4):
            key_schedule[i][j] = cipher_key[i][j]
    add_round_key_state = addRoundKey(state, cipher_key)
    for co in range(9):
        sub_bytes_state = subBytes(add_round_key_state)
        shift_rows_state = shiftRows(sub_bytes_state)
        mix_columns_state = mixColumns(shift_rows_state)
        add_round_key_state = addRoundKey(mix_columns_state, keySchedule(key_schedule, co))
    sub_bytes_state = subBytes(add_round_key_state)
    shift_rows_state = shiftRows(sub_bytes_state)
    add_round_key_state = addRoundKey(shift_rows_state, keySchedule(key_schedule, 9))
    return add_round_key_state

# Decryption function
def decryption(state, cipher_key):
    key_schedule = [[0 for _ in range(44)] for _ in range(4)]
    for i in range(4):
        for j in range(4):
            key_schedule[i][j] = cipher_key[i][j]
    for f in range(10):
        keySchedule(key_schedule, f)
    add_round_key_state = addRoundKey(state, parser(key_schedule, 10))
    shift_rows_state = invShiftRows(add_round_key_state)
    sub_bytes_state = invSubBytes(shift_rows_state)
    for co in range(9, 0, -1):
        add_round_key_state = addRoundKey(sub_bytes_state, parser(key_schedule, co))
        mix_columns_state = invMixColumns(add_round_key_state)
        shift_rows_state = invShiftRows(mix_columns_state)
        sub_bytes_state = invSubBytes(shift_rows_state)
    add_round_key_state = addRoundKey(sub_bytes_state, parser(key_schedule, 0))
    return add_round_key_state

# Addition of round key
def addRoundKey(state, key_param):
    new_state = [[0 for _ in range(4)] for _ in range(4)]
    for i in range(4):
        for j in range(4):
            new_state[i][j] = state[i][j] ^ key_param[i][j]
    return new_state

# Parsing the key schedule
def parser(table, block_no):
    new_state = [[0 for _ in range(4)] for _ in range(4)]
    for i in range(4):
        for j in range(4):
            new_state[i][j] = table[i][block_no * 4 + j]
    return new_state

# Inverse MixColumns transformation
def invMixColumns(state):
    return mixColumns(mixColumns(mixColumns(state)))

# Inverse ShiftRows transformation
def invShiftRows(state):
    new_state = [[0 for _ in range(4)] for _ in range(4)]
    for i in range(4):
        for j in range(4):
            new_state[i][(j + 4 - i) % 4] = state[i][j]
    return new_state

# Inverse SubBytes transformation
def invSubBytes(state):
    new_state = [[0 for _ in range(4)] for _ in range(4)]
    for x in range(4):
        for y in range(4):
            new_state[x][y] = inverseSbox[state[x][y] >> 4][state[x][y] & 0x0f]
    return new_state
