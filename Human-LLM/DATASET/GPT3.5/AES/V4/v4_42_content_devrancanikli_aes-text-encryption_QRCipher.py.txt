from AESCipher import AESCipher
from PIL import Image
from pyzbar.pyzbar import decode
import pyqrcode

class QRCipher:
    def __init__(self, key, file_name):
        """
        Initialize QRCipher object with the provided key and file name.
        """
        self.key = key
        self.file_name = file_name
        self.cipher = AESCipher(bytes(self.key, encoding='utf8'))

    def set_key(self, key):
        """
        Set a new encryption key.
        """
        self.key = key
        self.cipher = AESCipher(bytes(self.key, encoding='utf8'))

    @staticmethod
    def create_qr(text, qr_name):
        """
        Create a QR code with the given text and save it as an image with the specified name.
        """
        qr_code = pyqrcode.create(text)
        qr_code.png(qr_name, scale=6)
        return qr_code

    @staticmethod
    def decode_qr_image(image_name):
        """
        Decode the text from a QR code image file.
        """
        data = decode(Image.open(image_name))
        text = data[0].data
        return text

    def encrypt(self, plain_text):
        """
        Encrypt plain text using AES encryption and save the result as a QR code image.
        """
        encrypted_text = self.cipher.encrypt(plain_text)
        qr_name = 'encrypted_' + self.file_name + '.png'
        return self.create_qr(encrypted_text, qr_name)

    def decrypt(self, encrypted_text):
        """
        Decrypt encrypted text using AES decryption and save the result as a QR code image.
        """
        decrypted_text = self.cipher.decrypt(encrypted_text)
        qr_name = 'decrypted_' + self.file_name + '.png'
        return self.create_qr(decrypted_text, qr_name)
