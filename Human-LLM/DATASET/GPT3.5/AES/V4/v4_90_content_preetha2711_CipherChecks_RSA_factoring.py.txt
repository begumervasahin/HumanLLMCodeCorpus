import fractions
import random

def factor(n, e, d):
    # Calculate s and t
    s = 0
    t = e * d - 1
    while t % 2 == 0:
        s += 1
        t = t / 2
    
    # Generate Z_n*
    z_star_n = []
    for i in range(2, n + 1):
        if fractions.gcd(i, n) == 1:
            z_star_n.append(i)
    
    # Find a suitable value for b
    b = 0
    flag = True
    while flag:
        a = random.choice(z_star_n)
        b = pow(a, t, n)
        while pow(b, 2, n) != 1:
            b = (b * b) % n
        if (b % n != 1) and ((b + 1) % n != 0):
            flag = False
    
    # Find the factors p and q
    p = fractions.gcd(b - 1, n)
    q = n // p
    
    return p, q

# Test the factorization function
