# Utility function to remove the '0x' prefix from hex strings and return the raw hex value
def format_hex(hex_str):
    return hex_str[2:]

# Converts a message into its hexadecimal representation using the AES S-box
def sbox_substitute(message):
    substituted = []
    for char in message:
        # Convert each character to its corresponding S-box value
        hex_value = format_hex(hex(ord(char)))
        sbox_value = Sbox[int(hex_value, 16)]
        substituted.append(format_hex(hex(sbox_value)))
    return substituted

def main():
    key = 'Thats my Kung Fu'
    print(f"Key length: {len(key.encode('utf-8'))} bytes")

    msg = 'Two One Nine Two'
    print(f"Message: {msg}")
    print(f"Message length: {len(msg.encode('utf-8'))} bytes")

    # Convert message using S-box and print the result
    substituted_msg = sbox_substitute(msg)
    print("S-box substituted values:", substituted_msg)

    # Example of dividing the substituted message into blocks (assuming block size of 4 for demonstration)
    blocks = [substituted_msg[i:i+4] for i in range(0, len(substituted_msg), 4)]
    for i, block in enumerate(blocks):
        print(f"Block {i+1}: {block}")

if __name__ == "__main__":
    main()
