# Importing required library
import numpy as np

# Round constants used in key expansion
RCON = [0x01, 0x02, 0x04, 0x08, 0x10, 0x20, 0x40, 0x80, 0x1B, 0x36]

# Substitution Box (S-box)
S_BOX = []

# Forward Mix Columns matrix
FORWARD_MIX_MATRIX = [
    [0x02, 0x03, 0x01, 0x01],
    [0x01, 0x02, 0x03, 0x01],
    [0x01, 0x01, 0x02, 0x03],
    [0x03, 0x01, 0x01, 0x02]
]

# Initial key
KEY = [0x00, 0x01, 0x02, 0x03, 0x04, 0x05, 0x06, 0x07,
       0x08, 0x09, 0x0A, 0x0B, 0x0C, 0x0D, 0x0E, 0x0F]

# Plain text to be encrypted
PLAIN_TEXT = [
    0x00, 0x11, 0x22, 0x33, 0x44, 0x55, 0x66, 0x77,
    0x88, 0x99, 0xAA, 0xBB, 0xCC, 0xDD, 0xEE, 0xFF
]

# Initialization Vector (IV) for Cipher Feedback (CFB) mode
IV = [
    0x08, 0x07, 0x06, 0x05, 0x04, 0x03, 0x02, 0x01,
    0x00, 0x99, 0xAA, 0xBB, 0xCC, 0xDD, 0xEE, 0xFF
]

# Input string to be encrypted
input_string = (
    "AES became effective as a federal government standard after approval by "
    "the Secretary of Commerce It is the first publicly accessible cipher approved by "
    "the National Security Agency for top secret information"
)

# Splitting input string into blocks of 16 bytes
group_num = len(input_string) // 16
plain_text_array = [[0] * 16 for _ in range(group_num)]
for i in range(group_num):
    for j in range(16):
        plain_text_array[i][j] = ord(input_string[i * 16 + j])

# Function to expand the initial key into a round key
def key_expansion(key):
    round_key = [[0] * 4 for _ in range(44)]
    for i in range(4):
        for j in range(4):
            round_key[i][j] = key[4 * i + j]
    for i in range(4, 44):
        temp = round_key[i - 1].copy()
        if i % 4 == 0:
            temp = key_sub(key_rotate(temp))
            temp[0] ^= RCON[i // 4 - 1]
        for j in range(4):
            round_key[i][j] = round_key[i - 4][j] ^ temp[j]
    return round_key

# Function to perform the key schedule rotation
def key_rotate(temp):
    temp.append(temp.pop(0))
    return temp

# Function to substitute bytes using the S-box
def key_sub(temp):
    return [S_BOX[val] for val in temp]

# Function to perform byte substitution using the S-box during encryption
def encrypt_sub(state):
    return [[S_BOX[val] for val in row] for row in state]

# Function to perform the shift rows operation
def shift_row(state):
    for i in range(1, 4):
        state[i] = state[i][i:] + state[i][:i]
    return state

# Function to perform the mix columns operation
def mix_column(state):
    res = np.zeros((4, 4), dtype=int)
    for i in range(4):
        for j in range(4):
            for k in range(4):
                res[i][j] ^= mul(FORWARD_MIX_MATRIX[i][k], state[k][j])
    return res

# Function to perform multiplication in the Galois Field
def mul(a, b):
    if a == 1:
        return b
    if a == 2:
        return (b << 1) ^ (0x1B if b >= 128 else 0)
    if a == 3:
        return mul(2, b) ^ b

# Function to add the round key to the state
def add_round_key(state, key_state):
    return [[state[i][j] ^ key_state[i][j] for j in range(4)] for i in range(4)]

# Function to perform AES encryption
def aes_encrypt(plain_text, round_key):
    state = [[0] * 4 for _ in range(4)]
    key_state = [[0] * 4 for _ in range(4)]
    for j in range(4):
        for i in range(4):
            state[i][j] = plain_text[j * 4 + i]
    for i in range(11):
        key_state = round_key[i * 4:(i + 1) * 4]
        if i == 0:
            state = add_round_key(state, key_state)
        elif i == 10:
            state = shift_row(encrypt_sub(state))
            state = add_round_key(state, key_state)
        else:
            state = shift_row(encrypt_sub(state))
            state = mix_column(state)
            state = add_round_key(state, key_state)
    return [state[i][j] for j in range(4) for i in range(4)]

# Function to perform AES encryption in CFB mode
def cfb_mode(IV, key, plain_text_array, group_num):
    round_key = key_expansion(key)
    res = []
    temp = IV.copy()
    for group in plain_text_array:
        temp = aes_encrypt(temp, round_key)
        temp = [temp[j] ^ group[j] for j in range(16)]
        res.append(temp.copy())
    return res

# Encrypting the plain text using CFB mode
encrypted_text = cfb_mode(IV, KEY, plain_text_array, group_num)

# Printing the encrypted result
for array in encrypted_text:
    print(" ".join([hex(element)[2:].zfill(2) for element in array]))
