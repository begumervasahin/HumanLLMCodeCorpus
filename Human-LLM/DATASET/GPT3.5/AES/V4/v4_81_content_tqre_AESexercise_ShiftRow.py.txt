import gf

def do(state):
    """Perform the 'do' operation."""
    r0, r1, r2, r3 = split_state(state)
    r1 = gf.circrotateleft(r1)
    r2 = gf.circrotateleft(gf.circrotateleft(r2))
    r3 = gf.circrotateright(r3)
    new_state_matrix = concatenate_state(r0, r1, r2, r3)
    return new_state_matrix

def inverse(state):
    """Perform the 'inverse' operation."""
    r0, r1, r2, r3 = split_state(state)
    r1 = gf.circrotateright(r1)
    r2 = gf.circrotateleft(gf.circrotateleft(r2))
    r3 = gf.circrotateleft(r3)
    new_state_matrix = change_matrix_mode(concatenate_state(r0, r1, r2, r3))
    return new_state_matrix

def split_state(state):
    """Split the state into 4 blocks."""
    r0, r1, r2, r3 = bytearray(), bytearray(), bytearray(), bytearray()
    for i in range(4):
        r0.append(state[i * 4])
        r1.append(state[1 + i * 4])
        r2.append(state[2 + i * 4])
        r3.append(state[3 + i * 4])
    return r0, r1, r2, r3

def concatenate_state(r0, r1, r2, r3):
    """Concatenate 4 blocks into a new state."""
    return r0 + r1 + r2 + r3

def change_matrix_mode(state):
    """Change the matrix mode."""
    r0, r1, r2, r3 = bytearray(), bytearray(), bytearray(), bytearray()
    for i in range(4):
        r0.append(state[i * 4])
        r1.append(state[1 + i * 4])
        r2.append(state[2 + i * 4])
        r3.append(state[3 + i * 4])
    new_state_matrix = r0 + r1 + r2 + r3
    return new_state_matrix
