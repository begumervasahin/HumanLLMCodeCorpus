import base64
import random
import string
from Crypto import Random
from Crypto.Cipher import AES

BLOCK_SIZE = 32

def pad(s):
    padding_length = BLOCK_SIZE - len(s.encode('utf-8')) % BLOCK_SIZE
    return s + (padding_length * chr(padding_length)).encode('utf-8')

def unpad(s):
    return s[:-ord(s[len(s) - 1:])]

class AES256:
    def __init__(self, key):
        self.key = self.generate_key(key)

    def encrypt(self, raw):
        raw = pad(raw)
        iv = Random.new().read(AES.block_size)
        cipher = AES.new(self.key, AES.MODE_CBC, iv)
        return base64.b64encode(iv + cipher.encrypt(raw))

    def decrypt(self, enc):
        enc = base64.b64decode(enc)
        iv = enc[:AES.block_size]
        cipher = AES.new(self.key, AES.MODE_CBC, iv)
        return unpad(cipher.decrypt(enc[AES.block_size:])).decode('utf-8')

    def generate_key(self, key):
        string_length = 16 - len(key)
        random.seed(key)
        letters = string.ascii_letters
        return (key + ''.join(random.choice(letters) for _ in range(string_length))).encode('utf-8')
