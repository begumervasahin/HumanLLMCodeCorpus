import os
import json
import sys
import getpass
from hashlib import sha256
from termcolor import colored
from halo import Halo
from modules.encryption import DataManip
from modules.exceptions import UserExits, PasswordFileDoesNotExist
from modules.menu import Manager

def exit_program():
    print(colored("Exiting...", "red"))
    sys.exit()

def start(data_manipulator: DataManip):
    # Check if the master password file exists
    master_password_file = "db/masterpassword.json"
    if os.path.isfile(master_password_file):
        with open(master_password_file, 'r') as json_data_file:
            master_password_data = json.load(json_data_file)
        stored_master_password = master_password_data["Master"]

        # Prompt user to enter master password
        user_input_password = getpass.getpass("Enter Your Master Password: ")
        spinner = Halo(text=colored("Unlocking", "green"), color="green", spinner=data_manipulator.dots_)

        # Validate the entered master password
        if sha256(user_input_password.encode("utf-8")).hexdigest() == stored_master_password:
            print(colored(f"{data_manipulator.checkmark_} Thank you! Choose an option below:", "green"))
            menu = Manager(data_manipulator, "db/passwords.json", master_password_file, user_input_password)
            try:
                menu.begin()
            except UserExits:
                exit_program()
            except PasswordFileDoesNotExist:
                print(colored(f"{data_manipulator.x_mark_} DB not found. Try adding a password {data_manipulator.x_mark_}", "red"))
        else:
            print(colored(f"{data_manipulator.x_mark_} Master password is incorrect {data_manipulator.x_mark_}", "red"))        
            return start(data_manipulator)
    else:
        try:
            os.mkdir("db/")
        except FileExistsError:
            pass

        # Prompt user to create a new master password
        print(colored("To start, we'll have you create a master password. Be careful not to lose it as it is unrecoverable.", "green"))
        master_password = getpass.getpass("Create a master password for the program: ")
        second_input = getpass.getpass("Verify your master password: ")

        # Validate the entered master password and save it
        if master_password == second_input:
            spinner = Halo(text=colored("initializing base...", "green"), color="green", spinner=data_manipulator.dots_)        
            hashed_master_password = sha256(master_password.encode("utf-8")).hexdigest()
            master_password_data = {"Master": {}}
            master_password_data["Master"] = hashed_master_password

            with open(master_password_file, 'w') as json_data_file:
                json.dump(master_password_data, json_data_file, sort_keys=True, indent=4)

            spinner.stop()
            print(colored(f"{data_manipulator.checkmark_} Thank you! Restart the program and enter your master password to begin.", "green"))
        else:
            print(colored(f"{data_manipulator.x_mark_} Passwords do not match. Please try again {data_manipulator.x_mark_}", "red"))
            return start(data_manipulator)

if __name__ == "__main__":
    data_manipulator = DataManip()
    start(data_manipulator)
