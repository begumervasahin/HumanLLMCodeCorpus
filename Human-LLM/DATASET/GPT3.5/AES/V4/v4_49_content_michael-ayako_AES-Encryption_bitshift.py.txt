import numpy as np

def shift_left(bits):
    shifted_bits = []
    for block in bits:
        shifted_block = []
        for row in block:
            shifted_row = [row[0], row[1], row[2], row[3]]
            shifted_block.append(shifted_row)
        shifted_bits.append(np.array(shifted_block).reshape(4, 4))
    return [''.join(str(bit) for bit in block.flatten()) for block in shifted_bits]

def shift_right(bits):
    shifted_bits = []
    for block in bits:
        shifted_block = []
        for row in block:
            shifted_row = [row[0], row[1], row[2], row[3]]
            shifted_row = [shifted_row[3], shifted_row[0], shifted_row[1], shifted_row[2]]
            shifted_block.append(shifted_row)
        shifted_bits.append(np.array(shifted_block).reshape(4, 4))
    return [''.join(str(bit) for bit in block.flatten()) for block in shifted_bits]
