import base64
import zlib
import time
import hashlib
from Crypto.Cipher import AES

class AESCipher:
    def __init__(self, IV=None, Key=None):
        self.block_size = 16
        self.IV = IV if IV else self.generate_IV()
        self.Key = Key if Key else hashlib.md5(str(self.IV).encode()).hexdigest()

    def generate_IV(self):
        """
        Generate a random initialization vector (IV).
        """
        return hashlib.md5(str(time.time()).encode()).hexdigest()[:self.block_size]

    def encrypt(self, raw):
        """
        Encrypts raw data using AES in CBC mode with zlib compression.

        Args:
            raw (str): The data to be encrypted.

        Returns:
            str: The encrypted data.
        """
        pad = lambda s: s + (self.block_size - len(s) % self.block_size) * chr(self.block_size - len(s) % self.block_size)
        raw = pad(raw)
        cipher = AES.new(self.Key.encode(), AES.MODE_CBC, self.IV.encode())
        encrypted_data = cipher.encrypt(raw)
        compressed_data = zlib.compress(self.IV.encode() + encrypted_data)
        return base64.b64encode(compressed_data)

    def decrypt(self, enc):
        """
        Decrypts encrypted data using AES in CBC mode with zlib compression.

        Args:
            enc (str): The encrypted data to be decrypted.

        Returns:
            str: The decrypted data.
        """
        unpad = lambda s: s[0:-ord(s[-1])]
        decoded_data = base64.b64decode(enc)
        IV = zlib.decompress(decoded_data)[:self.block_size].decode()
        encrypted_data = zlib.decompress(decoded_data)[self.block_size:].decode()
        Key = hashlib.md5(IV.encode()).hexdigest()
        cipher = AES.new(Key.encode(), AES.MODE_CBC, IV.encode())
        decrypted_data = cipher.decrypt(encrypted_data)
        return unpad(decrypted_data)
