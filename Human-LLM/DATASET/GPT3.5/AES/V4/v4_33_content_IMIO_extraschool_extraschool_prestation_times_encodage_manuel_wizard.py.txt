from openerp import models, api, fields, _
from openerp.exceptions import except_orm, Warning, RedirectWarning 

class ExtraschoolEncodageManuelValidationWizard(models.TransientModel):
    _name = 'extraschool.encodage_manuel_validation_wizard'

    place_id = fields.Many2one('extraschool.place')

    @api.multi
    def validate(self):
        # Iterate over each record in active_ids
        for record in self.env['extraschool.prestation_times_encodage_manuel'].browse(self._context.get('active_ids')):
            record.validate()
        return True

    @api.multi
    def set_to_draft(self):
        # Iterate over each record in active_ids
        for record in self.env['extraschool.prestation_times_encodage_manuel'].browse(self._context.get('active_ids')):
            record.set_to_draft()
        return True

    @api.multi
    def change_place_id(self):
        # Iterate over each record in active_ids
        for record in self.env['extraschool.prestation_times_encodage_manuel'].browse(self._context.get('active_ids')):
            # Set record to draft state
            record.set_to_draft()
            # Update place_id for the record
            record.write({'place_id': self.place_id.id})
            # Validate the record
            record.validate()
        return True
