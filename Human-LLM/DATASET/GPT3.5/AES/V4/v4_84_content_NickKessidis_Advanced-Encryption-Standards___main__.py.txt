import random

if __name__ == "__main__":
    # Generate a random number and key
    number = random.randrange(2**127, (2**128)-1)
    key = random.randrange(2**127, (2**128)-1)

    # Display the result of XOR and the plain text
    xor_result = number ^ key
    print("Result of XOR:", bin(xor_result))
    print("Plain text:", bin(number))

    # Initialize the State array with zeros
    State = ["0"] * 16

    # Populate the State array with 8-bit slices of the random number
    for i in range(16):
        binary_slice = bin(number)[8*i+2:8*i+10]
        State[i] = binary_slice

    # Display the State array
    print("\nState Array:")
    for row in State:
        print(row)

    # Generate a random S-box
    Sbox = [random.randrange(2**7, (2**8)-1) for _ in range(256)]

    # Perform the Substitution process on the State array
    for i in range(16):
        # Extract row and column indices from each 8-bit slice
        row = int(State[i][:4], 2)
        col = int(State[i][4:], 2)

        # Perform substitution using the S-box
        State[i] = Sbox[(col + row) % 16]

    # Display the State array after the Substitution process
    print("\nState Array after the Substitution process:")
    print(State)
