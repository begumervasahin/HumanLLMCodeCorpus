from Crypto.Cipher import AES, PKCS1_v1_5
from Crypto.PublicKey import RSA
from Crypto.Signature import pkcs1_15
from Crypto.Hash import SHA256
import json
import base64

class AESCrypt:
    def __init__(self, key):
        self.key = key.encode('utf-8')
        self.mode = AES.MODE_ECB
        self.block_size = 16

    def encrypt(self, data):
        padded_data = self._pad(data)
        cipher = AES.new(self.key, self.mode)
        encrypted_data = cipher.encrypt(padded_data)
        return base64.b64encode(encrypted_data).decode()

    def decrypt(self, encrypted_data):
        cipher = AES.new(self.key, self.mode)
        decrypted_data = cipher.decrypt(base64.b64decode(encrypted_data))
        return self._unpad(decrypted_data).decode()

    def _pad(self, data):
        padding_length = self.block_size - len(data) % self.block_size
        padding = padding_length * chr(padding_length).encode()
        return data + padding

    def _unpad(self, data):
        return data[:-data[-1]]

class RSACrypt:
    @staticmethod
    def _serialize_data(data_dict):
        serialized_data = ''
        for key in sorted(data_dict.keys()):
            value = data_dict[key]
            if isinstance(value, dict) or isinstance(value, list):
                serialized_data += json.dumps(value, sort_keys=True)
            else:
                serialized_data += str(value)
        return serialized_data

class RSAPubCrypt(RSACrypt):
    def __init__(self, key):
        self.key = RSA.import_key(key)

    def encrypt(self, data, length=200):
        try:
            data = data.encode('utf-8')
            cipher = PKCS1_v1_5.new(self.key)
            encrypted_parts = []
            for i in range(0, len(data), length):
                encrypted_parts.append(cipher.encrypt(data[i:i + length]))
            return str(base64.b64encode(b"".join(encrypted_parts)), encoding='utf-8')
        except Exception as e:
            return False

    def verify_signature(self, data, signature):
        try:
            if isinstance(data, dict):
                data = self._serialize_data(data)
            data = data.encode('utf-8')
            hashed_data = SHA256.new(data)
            pkcs1_15.new(self.key).verify(hashed_data, base64.b64decode(signature))
            return True
        except (ValueError, TypeError):
            return False

class RSAPrvCrypt(RSACrypt):
    def __init__(self, key):
        self.key = RSA.import_key(key)

    def decrypt(self, encrypted_data, length=256):
        try:
            cipher = PKCS1_v1_5.new(self.key)
            encrypted_data = base64.b64decode(encrypted_data)
            decrypted_parts = []
            for i in range(0, len(encrypted_data), length):
                decrypted_parts.append(cipher.decrypt(encrypted_data[i:i + length], 'xyz'))
            return str(b"".join(decrypted_parts), encoding='utf-8')
        except Exception as e:
            return False

    def sign(self, data):
        try:
            if isinstance(data, dict):
                data = self._serialize_data(data)
            data = data.encode('utf-8')
            hashed_data = SHA256.new(data)
            signature = pkcs1_15.new(self.key).sign(hashed_data)
            return str(base64.b64encode(signature), encoding='utf-8')
        except Exception as e:
            return False
