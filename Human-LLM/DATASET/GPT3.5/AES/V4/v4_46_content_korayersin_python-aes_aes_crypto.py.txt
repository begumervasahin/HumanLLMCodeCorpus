from Crypto import Random
from Crypto.Cipher import AES
import base64

class AESCrypto:
    def __init__(self, key, value_of):
        # Store the last 6 characters of value_of
        self.value_of = value_of[-6:]
        # Generate the key using the key and value_of
        self.key = self._generate_key(key)
        # Block size for AES encryption
        self.BLOCK_SIZE = 16

    def _generate_key(self, key):
        # Helper function to modify the key based on value_of
        def change_name(name, index, rate):
            while len(name) - 1 < index:
                index = index - len(name)
            _str = name[:index]
            total = ""
            for iw in range(rate):
                total += name[index]
            _str = _str + total + name[index:]
            return _str

        # Modify the key based on value_of
        for i in range(3):
            defined = self.value_of[i * 2:i * 2 + 2]
            defined_index = int(defined[:1])
            defined_rate = int(defined[1:])
            key = change_name(key, defined_index, defined_rate) 

        # Ensure the key is 16 bytes long
        length_name = len(key)
        if length_name > 16:
            key = key[:16]
        elif length_name < 16:
            for i in range(16 - length_name):
                key = key + "_"
        return key

    def pad(self, data):
        # Pad the data to be a multiple of the block size
        return data + (self.BLOCK_SIZE - len(data) % self.BLOCK_SIZE) * chr(self.BLOCK_SIZE - len(data) % self.BLOCK_SIZE)

    def unpad(self, data):
        # Remove padding from the data
        return data[:-ord(data[len(data) - 1:])]

    def encrypt(self, message):
        # Pad the message and generate an initialization vector
        message = self.pad(message)
        iv = Random.new().read(AES.block_size)
        # Encrypt the message using AES in CBC mode
        cipher = AES.new(self.key.encode("utf-8"), AES.MODE_CBC, iv)
        encrypted = cipher.encrypt(message.encode("utf-8"))
        # Return the encrypted message along with the IV
        return base64.b64encode(iv + encrypted)

    def decrypt(self, encrypted):
        # Decode the encrypted message
        encrypted = base64.b64decode(encrypted)
        # Extract the IV from the encrypted message
        IV = encrypted[:self.BLOCK_SIZE]
        # Create an AES cipher object with the key and IV
        aes = AES.new(self.key.encode("utf-8"), AES.MODE_CBC, IV)
        # Decrypt the message and remove padding
        decrypted = aes.decrypt(encrypted[self.BLOCK_SIZE:])
        return self.unpad(decrypted).decode("utf-8")
