import matplotlib.pyplot as plt
import seaborn as sns
import pandas as pd
import numpy as np
from copy import copy
import tensorflow as tf
from keras import optimizers, Sequential
from keras.models import Model
from keras.layers import Dense, LSTM, RepeatVector, TimeDistributed
from keras.callbacks import ModelCheckpoint, TensorBoard

# Setting seeds for reproducibility
seed(7)
set_random_seed(11)

# Global constants
LABELS = ["Normal", "Break"]
DATA_SPLIT_PCT = 0.2
SEED = 123

# Load dataset
df = pd.read_csv("outfinaltest62.csv", header=None)

# Function to shift binary labels in a dataframe
def curve_shift(df, shift_by):
    vector = df[2].copy()
    for s in range(abs(shift_by)):
        tmp = vector.shift(np.sign(shift_by))
        tmp = tmp.fillna(0)
        vector += tmp
    labelcol = 2
    tmp = 29
    df.insert(loc=0, column=labelcol + tmp, value=vector)
    df = df.drop(df[df[labelcol] == 1].index)
    df = df.drop(labelcol, axis=1)
    df = df.rename(columns={labelcol + tmp: labelcol})
    df.loc[df[labelcol] > 0, labelcol] = 1
    return df

# Print before shifting
one_indexes = df.index[df[2] == 1]
print('Before shifting')
print(df.iloc[(one_indexes[0] - 3):(one_indexes[0] + 2), 0:5].head(n=5))

# Apply curve shifting
df_shift = curve_shift(df, shift_by=-2)

# Print after shifting
print('After shifting')
print(df_shift.iloc[(one_indexes[0] - 4):(one_indexes[0] + 1), 0:5].head(n=5))

# Prepare data for LSTM autoencoder
input_X = df.loc[:, 1].values
input_X = input_X.reshape(-1, 1)
input_y = df[2].values
input_y = input_y.reshape(-1, 1)
n_features = input_X.shape[1]

# Function to temporalize data
def temporalize(X, y, lookback):
    X_temp = []
    y_temp = []
    for i in range(len(input_X) - lookback - 1):
        t = []
        for j in range(1, lookback + 1):
            t.append(input_X[[(i + j + 1)], :])
        X_temp.append(t)
        y_temp.append(input_y[i + lookback + 1])
    return X_temp, y_temp

# Temporalize data
lookback = 1
X, y = temporalize(X=input_X, y=input_y, lookback=lookback)

# Train-test split
X_train, X_test, y_train, y_test = train_test_split(np.array(X), np.array(y), test_size=DATA_SPLIT_PCT, random_state=SEED)

# Model architecture
model = Sequential([
    LSTM(128, activation='relu', input_shape=(lookback, n_features), return_sequences=True),
    LSTM(64, activation='relu', return_sequences=False),
    RepeatVector(lookback),
    LSTM(64, activation='relu', return_sequences=True),
    LSTM(128, activation='relu', return_sequences=True),
    TimeDistributed(Dense(n_features))
])

# Compile model
model.compile(loss='mse', optimizer=optimizers.Adam(0.0001))

# Model training
history = model.fit(X_train, X_train,
                    epochs=30,
                    batch_size=64,
                    validation_split=0.2,
                    verbose=2)

# Plot training history
plt.plot(history.history['loss'], linewidth=2, label='Train')
plt.plot(history.history['val_loss'], linewidth=2, label='Valid')
plt.legend(loc='upper right')
plt.title('Model loss')
plt.ylabel('Loss')
plt.xlabel('Epoch')
plt.show()

# Make predictions
train_x_predictions = model.predict(X_train)
mse = np.mean(np.power(X_train - train_x_predictions, 2), axis=1)

# Reconstruction error plot
error_df = pd.DataFrame({'Reconstruction_error': mse, 'True_class': y_train.reshape(-1)})
groups = error_df.groupby('True_class')
fig, ax = plt.subplots()
for name, group in groups:
    ax.plot(group.index, group.Reconstruction_error, marker='o', ms=3.5, linestyle='',
            label="Break" if name == 1 else "Normal")
ax.legend()
plt.title("Reconstruction error for different classes")
plt.ylabel("Reconstruction error")
plt.xlabel("Data point index")
plt.show()
