import os
import glob
import cv2
import random
import numpy as np
from scipy.spatial import distance

# Custom module for saving results
import ypoften as of

def perform_edge_detection(img_path, tf_folder, blur_first=True, blur_size=3,
                            threshold1_default=100, threshold2_default=250,
                            adaptive_threshold=True, ratio1_default=0.4, ratio2_default=0.8,
                            save_tf=True, select_random=True, n_random=1000):
    """Perform edge detection on an image."""

    # Read the image in grayscale
    img = cv2.imread(img_path, cv2.IMREAD_GRAYSCALE)

    # Apply Gaussian blur if specified
    if blur_first:
        img = cv2.GaussianBlur(img, (blur_size, blur_size), 0)

    # Calculate adaptive thresholds if enabled
    if adaptive_threshold:
        threshold1 = min(100, np.quantile(img, q=ratio1_default))
        threshold2 = max(200, np.quantile(img, q=ratio2_default))
        print("Edge detection thresholds:", threshold1, threshold2)

    # Perform Canny edge detection
    edge = cv2.Canny(img, threshold1=threshold1_default, threshold2=threshold2_default)

    # Save the transformed image if required
    if save_tf:
        img_name = os.path.basename(img_path)
        img_save_path = os.path.join(tf_folder, "edge_canny", os.path.splitext(img_name)[0] + '.png')
        of.create_path(img_save_path)
        cv2.imwrite(img_save_path, edge)

    # Extract edge points and calculate edge density and distribution
    epoints_twodims = np.nonzero(edge)
    epoints = np.transpose(epoints_twodims)
    h, w = edge.shape
    dia = np.sqrt(h**2 + w**2)
    etotal = len(epoints)
    edensity = etotal / (h * w)

    # Randomly select edge points if required
    if etotal > 0 and select_random:
        random.seed(42)
        epoints = random.sample(list(epoints), min(n_random, etotal))

    # Calculate average distance between edge points
    if etotal > 0:
        dists = distance.pdist(epoints, 'euclidean')
        edist = np.mean(dists) / dia
    else:
        edist = -99999

    edge_data = [edensity, edist]
    print("Edge density & distribution:", edge_data)

    return edge_data

def process_images(img_folder, tf_folder, result_path):
    """Process images in the specified folder."""
    img_paths = glob.glob(os.path.join(img_folder, '*'))

    for j, img_path in enumerate(img_paths):
        print("-" * 100)
        img_name = os.path.basename(img_path)
        print(j, img_name)
        edge_data = [img_name] + perform_edge_detection(img_path, tf_folder)
        of.save_list_to_txt(edge_data, result_path)

    print("DONE" * 50)

if __name__ == "__main__":
    img_folder = os.path.join('img_all')
    tf_folder = os.path.join("img_transform")
    result_path = os.path.join('img_result', 'edge.txt')

    process_images(img_folder, tf_folder, result_path)
