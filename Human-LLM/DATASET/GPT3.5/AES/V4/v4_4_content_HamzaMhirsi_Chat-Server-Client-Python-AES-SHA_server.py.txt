import socket
from Crypto.Hash import SHA256
from Crypto.Cipher import AES

class Colors:
    MOVE = '\033[95m'
    BLUE = '\033[94m'
    GREEN = '\033[92m'
    YELLOW = '\033[93m'
    RED = '\033[91m'
    ENDC = '\033[0m'
    BBLUE = '\033[1m'
    UNDERLINE = '\033[4m'

def sha256_hash(message):
    sha256 = SHA256.new()
    sha256.update(message)
    return sha256.hexdigest()

def encrypt(message, key):
    cipher_aes = AES.new(key.encode(), AES.MODE_CFB, key)
    encrypted_aes = cipher_aes.encrypt(message.encode())
    return encrypted_aes

def decrypt(message, key):
    dec_cipher_aes = AES.new(key.encode(), AES.MODE_CFB, key)      
    decrypted_aes = dec_cipher_aes.decrypt(message)
    return decrypted_aes

def get_key():
    return input(Colors.MOVE + "AES key must be either 16, 24, or 32 bytes long: " + Colors.ENDC)

def accept_connection(server_socket):
    client, address = server_socket.accept()
    print(Colors.YELLOW + 'Connection from ' + str(address) + Colors.ENDC)  
    return client

def send_message(client, message):
    client.send(message)

def receive_message(client):
    return client.recv(1024)

def print_message(message):
    print(message)

def ask_question():
    return input(Colors.BLUE + "Ask a private question to your allies, Sir: " + Colors.ENDC)

def input_number():
    valid_input = False
    while not valid_input:
        number = int(input(Colors.BLUE + "Write a number between 3000 and 4000, Sir: " + Colors.ENDC))
        if 3000 < number < 4000:
            valid_input = True
            number = str(number)
        elif number < 3000:
            print(Colors.BBLUE + "You are under 3000, Sir" + Colors.ENDC)
        else:
            print(Colors.BBLUE + "You are writing a big number, Sir!" + Colors.ENDC)
    return number

def get_decision():
    return input("Is this the right answer? (y/n): ").upper()

server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
host = socket.gethostname()
port = 8091
server.bind((host, port))
server.listen(5)

client = accept_connection(server)
send_message(client, Colors.YELLOW + "We will use AES and SHA-256 for our key here!!" + Colors.ENDC)

print_message("We will use AES and SHA-256 for our key here!!" + Colors.ENDC)  
reply = receive_message(client)
print_message(Colors.GREEN + reply + Colors.ENDC)

question = ask_question()
send_message(client, encrypt(question, key))
send_message(client, sha256_hash(question))

number = input_number()
send_message(client, number)

print_message(Colors.YELLOW + "Waiting for answer..." + Colors.ENDC)

response = receive_message(client)
answer = receive_message(client)
decrypted_answer = decrypt(answer, key)
print(decrypted_answer)

decision = get_decision()

if decision != 'Y':
    client.close()
    print(Colors.RED + "Access has been denied :)" + Colors.ENDC)   
else:
    client.send("Everything is okay, we are here for orders, Sir !!")
    print(Colors.YELLOW + "Waiting for orders..." + Colors.ENDC)     

    while True:
        data = receive_message(client)
        if not data:
            break
        print("Encrypted message: " + str(data))

        print("Received: " + order)
        client.send(data)

client.close()
print(Colors.MOVE + "END OF COMMUNICATION, GOOD LUCK FOR WAR. MAY GOD PROTECT YOU!!" + Colors.ENDC)
