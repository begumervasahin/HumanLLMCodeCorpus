def gf_multiply(a, b):
    # Initialize the array to store intermediate values during Galois Field multiplication
    intermediate_values = [0, 0, 0, 0]
    res = 0
    
    # Initialize the first element of the intermediate array with the value of 'b'
    intermediate_values[0] = b
    
    # Perform the Galois Field multiplication
    for i in range(1, 4):
        intermediate_values[i] = intermediate_values[i - 1] << 1
        if intermediate_values[i - 1] & 0x80:
            intermediate_values[i] ^= 0x1b
        intermediate_values[i] &= 0xff
    
    # Calculate the result by performing XOR operations
    for i in range(4):
        if (a >> i) & 0x01:
            res ^= intermediate_values[i]
    
    return res

def generate_s_box(n):
    # Initialize an empty list to store the substitution table
    substitution_table = []
    
    # Generate the substitution table by iterating over all possible input values
    for i in range(256):
        # Calculate the value for the current input using the Galois Field multiplication function
        value = gf_multiply(n, i)
        
        # Append the calculated value to the table
        substitution_table.append(value)
    
    # Convert the list to a tuple for efficiency
    substitution_table = tuple(substitution_table)
    
    # Print the substitution table
    print(substitution_table)

# Test the S-box generation for different values of 'n'
for n in [2, 3, 9, 11, 13, 14]:
    print("Value of n:", n)
    generate_s_box(n)
