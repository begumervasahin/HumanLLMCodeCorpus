import os
from cryptography.hazmat.primitives.ciphers import Cipher, algorithms, modes
from cryptography.hazmat.backends import default_backend

class AES:
    def __init__(self):
        self.salt = ""
        self.iv = ""
        self.key = os.urandom(32)
        self.password = ""
        self.message = ""

    def encrypt(self, message):
        self.salt = os.urandom(16)
        backend = default_backend()
        self.iv = os.urandom(16)

        # Create AES cipher in CBC mode
        cipher = Cipher(algorithms.AES(self.key), modes.CBC(self.iv), backend=backend)

        # Create AES cipher in ECB mode for IV encryption
        iv_cipher = Cipher(algorithms.AES(self.key), modes.ECB(), backend=backend)
        iv_encryptor = iv_cipher.encryptor()

        # Encrypt IV using ECB mode
        ct_iv = iv_encryptor.update(self.iv) + iv_encryptor.finalize()

        # Encrypt the message using CBC mode
        encryptor = cipher.encryptor()
        ct = encryptor.update(bytes(message, 'utf8')) + encryptor.finalize()

        # Decrypt the message for demonstration purposes
        decryptor = cipher.decryptor()
        msg = decryptor.update(ct) + decryptor.finalize()

        self.message = ct
        self.iv = ct_iv

    def decrypt(self, ct):
        backend = default_backend()

        # Create AES cipher in ECB mode for IV decryption
        iv_cipher = Cipher(algorithms.AES(self.key), modes.ECB(), backend=backend)
        decryptor_iv = iv_cipher.decryptor()

        # Decrypt IV using ECB mode
        self.iv = decryptor_iv.update(self.iv) + decryptor_iv.finalize()

        # Create AES cipher in CBC mode
        cipher = Cipher(algorithms.AES(self.key), modes.CBC(self.iv), backend=backend)
        decryptor = cipher.decryptor()

        # Decrypt the message using CBC mode
        msg = decryptor.update(ct) + decryptor.finalize()
        self.message = msg.decode('utf-8')

# Usage example
aes = AES()
aes.message = "Aqui temos uma palavras com um num de caracteres multiplos de 16, para funcionar o algoritmo AES"
aes.password = "12345678"
aes.encrypt(aes.message)
print("Encrypted message:", str(aes.message)[2:-1])

aes.decrypt(aes.message)
print("Decrypted message:", aes.message)
