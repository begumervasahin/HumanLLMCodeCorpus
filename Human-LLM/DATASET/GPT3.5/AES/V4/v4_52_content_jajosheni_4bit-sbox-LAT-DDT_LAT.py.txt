import sys

state = {
    'size': 16,
    'sBox': [],
    'hex': {
        '0': 0x0, '1': 0x1, '2': 0x2, '3': 0x3,
        '4': 0x4, '5': 0x5, '6': 0x6, '7': 0x7,
        '8': 0x8, '9': 0x9, 'a': 0xa, 'b': 0xb,
        'c': 0xc, 'd': 0xd, 'e': 0xe, 'f': 0xf,
    },
    'LAT': [],
}

def get_values(state, mode):
    if mode:
        try:
            with open("sBox.txt", "r") as f:
                contents = f.read().strip()
                state['sBox'] = [state['hex'][char] for char in contents[:state['size']]]
        except FileNotFoundError:
            print("sBox.txt not found!")
            sys.exit(1)
    else:
        for i in range(state['size']):
            val = check_value(state, i)
            state['sBox'].append(state['hex'][val])

def check_value(state, i):
    val = input(f'Enter the sBox value for {hex(i)}: ')
    while True:
        if len(val) == 1 and (val.isdigit() and int(val) in range(10)) or (val in 'abcdef'):
            return val
        else:
            val = input(f'Enter the sBox value for {hex(i)}: ')

def print_table(state):
    for i in range(state['size']):
        print(hex(i), end=' | ')
    print('')
    print('----|-'*state['size'])
    for i in range(state['size']):
        print(hex(state['sBox'][i]), end=' | ')
    print('\n\nCalculating ...', end='\r')

def traverse(state):
    for row in state['hex']:
        latRow=[]
        for col in state['hex']:
            latRow.append(calc_value(state, row, col))
        state['LAT'].append(latRow)

def calc_value(state, inputmask, outputmask):
    total = 0
    inputmask = state['hex'][inputmask]
    outputmask = state['hex'][outputmask]
    for eachHex in state['hex']:
        eachHex = state['hex'][eachHex]
        sboxi = eachHex
        sboxo = state['sBox'][int(eachHex)]
        inputbit = [(0b0001 << i) & inputmask >> i for i in range(4)]
        outputbit = [(0b0001 << i) & outputmask >> i for i in range(4)]
        if (sum(inputbit) % 2) == (sum(outputbit) % 2):
            total += 1
    return total - 8

def print_LAT(state):
    columns = '0123456789ABCDEF'
    index = 0
    print('    ', end='')
    for hexno in state['hex']:
        print(' {:4} '.format(hexno[-1].upper()), end='')
    print('\n  |' + '-----|' * 16)
    for row in state['LAT']:
        print('{}'.format(columns[index]), end=' |')
        index += 1
        for element in row:
            print('{:3}  |'.format(element), end='')
        print('\n  |' + '-----|'*16)

def lat_max(state):
    max_val = -10000000
    for row in state['LAT'][1:]:
        for element in row:
            max_val = max(max_val, abs(element))
    return max_val
