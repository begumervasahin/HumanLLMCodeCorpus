# Import the AES module from the 'aes' package.
from aes import aes

# Define the encryption key and the message to be encrypted.
key = "Thats my Kung Fu"
message = "Two One Nine Two"

# Encrypt the message using the AES module's encrypt function.
encrypted_message = aes.encrypt(key, message)

# Print the original message and its encrypted form to the console.
print("Message:\t", message)
print("Encrypted:\t", encrypted_message)
