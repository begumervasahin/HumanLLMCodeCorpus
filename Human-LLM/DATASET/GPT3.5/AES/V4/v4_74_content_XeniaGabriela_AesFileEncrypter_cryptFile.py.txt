import os
import platform
import sys
import binascii
import hashlib
from base64 import b64encode, b64decode
from datetime import datetime
from Crypto.Cipher import AES

# Detecting the operating system platform
os_platform = platform.system().lower()

# Dictionary to map actions to their descriptions
actions = {
    'e': 'Encrypting',
    'd': 'Decrypting',
    't': 'Testing'
}

# Function to encode string to bytes
def v_bytes(string):
    return bytes(string, "utf-8") if sys.version_info[0] == 3 else string

# Function to handle user input compatibility between Python 2 and 3
def v_input(string):
    return input(string) if sys.version_info[0] == 3 else raw_input(string)

# Function to clear the console screen
def clear_console():
    if os_platform.startswith("win"):
        os.system("cls")
    elif os_platform.startswith("linux"):
        os.system("history -c")
    else:
        print("Make sure to clear your command line history after running the script!!!")

# Function to determine terminal size
def terminal_size():
    if os_platform.startswith("linux"):
        os.system("resize -s 40 120")
        return 120
    elif os_platform.startswith("win"):
        os.system("mode con: cols=120 lines=40")
        return 120
    else:
        return 80

# Class for handling encryption and decryption
class Cipher(object):
    start_crypto = 0
    end_crypto = 0
    AES_modes = {"OFB": AES.MODE_OFB}
    required_args = ["-a", "-f", "-u", "-k"]

    usage = """
    This is the file encrypter! Run it like this:
    python cryptFile.py -a action -f file -u user -k key
    1. where action is e for encrypt, d for decrypt or t for testing
    2. file is the path to the file
    3. user is a username (min length = 6 characters)
    4. key is the key (min length = 8 characters)
    Keep user and key in your own mind!
    They will not be stored or hidden by the script.
    If you forget them, encrypted files cannot not be decrypted anymore in reasonable time.
    """

    def start(self, argv):
        # Checking if arguments are provided
        if len(argv) == 1:
            print(self.usage)
            sys.exit()

        # Parsing command line arguments
        self.parse_args(argv)
        self.prepare_args()

        # Performing the specified action
        if AES_args["-a"] == "e":
            self.encrypt_file()
        if AES_args["-a"] == "d":
            self.decrypt_file()
        if AES_args["-a"] == "t":
            self.test_decryption()

    def parse_args(self, strings):
        # Checking for missing arguments
        missing_args = set(self.required_args) - set(strings)
        if len(strings) < 9:
            print("\nMissing arguments!")
        if len(missing_args) > 0:
            for arg in missing_args:
                print("   You forgot to define %s" % (arg))
            sys.exit()

        # Parsing arguments
        for arg in self.required_args:
            try:
                AES_args[arg] = strings[strings.index(arg) + 1]
            except:
                print("   You have to define the value of %s" % (arg))

        # Validating arguments for encryption action
        if AES_args["-a"] == "e" and (len(AES_args["-u"]) < 5 or len(AES_args["-k"]) < 12):
            print("\n   The key must consist of 12 characters and the username must consists of 5 characters at least")
            sys.exit()
        if AES_args["-a"] == "e" and (len(AES_args["-k"]) < 12):
            print("\n   It is recommended to use a key and user name with 24 characters in total at least")
            go_on = v_input("\n   Are you sure that you want to continue? Type y/n for yes/no: ")
            if go_on == "n":
                sys.exit()

        # Clearing console and printing action
        clear_console()
        print("\n%s %s ..." % (actions[AES_args["-a"]], AES_args["-f"]))
        return AES_args

    def prepare_args(self):
        # Validating the action
        if AES_args["-a"] not in ["e", "d", "t"]:
            print("\n-a must be e for encryption or d for decryption")
            sys.exit()

        # Getting current time for performance measurement
        self.start_crypto = datetime.now()

        # Adjusting file path and building the key
        AES_args["-f"] = AES_args["-f"].replace("\\", "/")
        AES_args["-k"] = self.build_key(AES_args["-k"], 251)
        AES_args["-u"] = self.build_key(AES_args["-u"], 239)[11:27]

    def build_key(self, key, modulus):
        # Building key using PBKDF2 HMAC
        salt, index = v_bytes(""), 0
        for i in 4*key:
            index += 1
            hex_position = hex(int(binascii.hexlify(v_bytes(i)), 16)**index % modulus)[2:4].replace("L", "")
            hex_position = "0" + hex_position if len(hex_position) == 1 else hex_position
            salt += binascii.unhexlify(hex_position)
        new_key = hashlib.pbkdf2_hmac("sha512", v_bytes(key), salt, 2746398)
        return new_key[17:49]

    def encrypt_file(self):
        # Encrypting the file
        with open(AES_args["-f"], "rb") as read_file:
            plaintext = b64encode(read_file.read())
            plaintext += (16-len(plaintext)%16)*b'\x00'
        obj = AES.new(AES_args["-k"], self.AES_modes["OFB"], AES_args["-u"])
        ciphertext = obj.encrypt(plaintext)
        self.write_file(ciphertext, "_encrypted.")

            decrypted_text = obj.decrypt(ciphertext)

            self.write_file(decrypted_text, "_decrypted.")
        except:
            print("\n   Are you sure you chose the right action for the file? \n   %s seems not to be encrypted with this script.\n" % (AES_args["-f"]))

    def test_decryption(self):
        # Testing encryption and decryption process
        print("\ntesting the encryption-decryption-process without saving encryption to a file")
        with open(AES_args["-f"], "rb") as read_file:
            plaintext = b64encode(read_file.read())
            plaintext += (16-len(plaintext)%16)*b'\x00'
        obj = AES.new(AES_args["-k"], self.AES_modes["OFB"], AES_args["-u"])
        ciphertext = obj.encrypt(plaintext)
        obj2 = AES.new(AES_args["-k"], self.AES_modes["OFB"], AES_args["-u"])
        decrypted_text = obj2.decrypt(ciphertext)
        decoded_text = b64decode(decrypted_text)
        self.write_file(decoded_text, "_tested.")

    def write_file(self, text, direction):
        # Writing text to a file
        self.end_crypto = datetime.now()
        path_parts = AES_args["-f"].split(".")
        file_path = path_parts[0] + direction + path_parts[1]
        file_name = file_path.split("/").pop()
        changed_name = v_input("\nThe new file name will be %s. If you want another name, type it, else just press enter: " % (file_name))
        print(changed_name)
        new_file_name = file_name if changed_name == "" else changed_name.split(".")[0] + "." + file_name.split(".").pop()
        new_file_path = file_path.replace(file_name, new_file_name)
        with open(new_file_path, "wb") as new_file:
            new_file.write(text)
        self.clean_up(new_file_name, new_file_path)

    def clean_up(self, new_file_name, new_file_path):
        # Cleaning up and displaying result
        width = terminal_size()
        print("\n%s" % (width * "-"))
        print("%s saved in %s" % (new_file_name, new_file_path.replace(new_file_name, "")))
        print("\nComputation time: %s" % (str(self.end_crypto - self.start_crypto)[2:]))
        print("%s" % (width * "-"))
        delete = v_input("\nDo you want to delete the original file? Type y/n for yes/no: ")
        if delete == "y":
            os.remove(AES_args["-f"])
            print("Deleted %s\n" % (AES_args["-f"]))
        if os_platform.startswith("linux"):
            print("type 'history -c' to clear the current command line history")

# Starting the script
if __name__ == '__main__':
    Cipher().start(sys.argv)
