def parse_arguments():
    """Parse command-line arguments."""
    parser = argparse.ArgumentParser(description='Apply artistic effects to an image.')
    parser.add_argument('input_file', help='Path to the input image file.')
    parser.add_argument('output_file', help='Path to save the output image file.')
    parser.add_argument('-t', '--text', help='Text to overlay on the image, with format "[x,y]text".')
    parser.add_argument('-s', '--text_size', type=int, default=50, help='Font size for the overlay text.')
    parser.add_argument('--ghost-il', action='store_true', help='Apply ghost effect using inline Python (slower).')
    parser.add_argument('--ghost-np', action='store_true', default=True, help='Apply ghost effect using NumPy (faster).')
    parser.add_argument('-g', '--ghost_width', type=float, default=1.0, help='Ghost effect width as a percentage of image width.')
    parser.add_argument('-y', '--chrominance_static', type=int, default=7, help='Max +/- variation for chrominance in YUV space.')
    parser.add_argument('-v', '--luminance_static', type=int, default=11, help='Max +/- variation for luminance in YUV space.')
    parser.add_argument('-r', '--rescale', type=decimal.Decimal, help='Rescale the image (as a decimal fraction of original size).')
    return parser.parse_args()
