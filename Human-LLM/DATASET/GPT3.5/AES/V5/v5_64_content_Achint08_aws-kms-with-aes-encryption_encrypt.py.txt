import os
from Crypto.Cipher import AES
from Crypto import Random
import hashlib
import base64
import boto3

# AWS KMS configuration
KEY_ID = 'arn:aws:kms:xxxxxxx:key/xxxxxxxxxxxxxxxxxxxx'
MASTER_KEY = 'This is a secret key'
REGION_NAME = 'xx-xxxx-1'

def pad_string(s):
    """
    Pad a string to the block size of 16 for cryptographic functions.
    """
    BLOCK_SIZE = 16
    padding_size = BLOCK_SIZE - len(s) % BLOCK_SIZE
    padded_string = s + (padding_size * chr(padding_size))
    return padded_string

def generate_encrypted_key_from_kms(master_key):
    """
    Generate an encrypted master key from AWS KMS.
    """
    kms = boto3.client('kms', region_name=REGION_NAME)
    try:
        response = kms.encrypt(
            KeyId=KEY_ID,
            Plaintext=master_key.encode("utf-8"),
            region_name=REGION_NAME
        )
        encrypted_master_key = response['CiphertextBlob']
        print("Encrypted master key:", encrypted_master_key)
        print("Store the encrypted master key securely for later use.")
        return encrypted_master_key
    except Exception as e:
        print("[ERROR]: Error during KMS call:", e)

def encrypt_data(master_key, data):
    """
    Encrypt data using the master key.
    """
    try:
        hashed_key = hashlib.sha256(master_key.encode("utf-8")).digest()
        iv = Random.new().read(AES.block_size)
        cipher = AES.new(hashed_key, AES.MODE_CBC, iv)
        padded_data = pad_string(data)
        encrypted_data = iv + cipher.encrypt(padded_data)
        return base64.b64encode(encrypted_data)
    except Exception as e:
        print("[ERROR]: Error while encrypting data:", e)

# Encrypt the master key using AWS KMS
encrypted_master_key = generate_encrypted_key_from_kms(MASTER_KEY)

# Data to be encrypted
data_to_encrypt = "Your data here"

# Encrypt the data using the master key
encrypted_data = encrypt_data(MASTER_KEY, data_to_encrypt)
print("Encrypted data:", encrypted_data)
