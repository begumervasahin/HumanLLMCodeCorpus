import paho.mqtt.client as mqtt
from ecdsa import SigningKey, VerifyingKey, BadSignatureError
import configparser

def read_config(file_path):
    """
    Read MQTT configuration from the specified file.

    Args:
        file_path (str): The path to the configuration file.

    Returns:
        config: The parsed configuration object.
    """
    config = configparser.RawConfigParser()
    config.read(file_path)
    return config

def load_keys(private_key_path, public_key_path):
    """
    Load the private and public keys from the specified paths.

    Args:
        private_key_path (str): The path to the private key file.
        public_key_path (str): The path to the public key file.

    Returns:
        private_key: The loaded private key.
        public_key: The loaded public key.
    """
    with open(private_key_path, "rb") as private_key_file:
        private_key = SigningKey.from_pem(private_key_file.read())
    with open(public_key_path, "rb") as public_key_file:
        public_key = VerifyingKey.from_pem(public_key_file.read())
    return private_key, public_key

def sign_message(private_key, message):
    """
    Sign the message using the private key.

    Args:
        private_key: The private key used for signing.
        message (bytes): The message to sign.

    Returns:
        signature: The signature of the message.
    """
    return private_key.sign(message)

def verify_signature(public_key, signature, message):
    """
    Verify the signature of the message using the public key.

    Args:
        public_key: The public key used for verification.
        signature: The signature to verify.
        message (bytes): The original message.

    Returns:
        bool: True if the signature is valid, False otherwise.
    """
    try:
        public_key.verify(signature, message)
        return True
    except BadSignatureError:
        return False

def publish_message(mqtt_config, topic, message):
    """
    Publish the message over MQTT.

    Args:
        mqtt_config: The MQTT configuration.
        topic (str): The MQTT topic to publish to.
        message (bytes): The message to publish.
    """
    mqtt_client = mqtt.Client()
    mqtt_client.connect(mqtt_config.get('mqtt', 'broker'), int(mqtt_config.get('mqtt', 'port'))) 
    mqtt_client.publish(topic, message)
    mqtt_client.disconnect()

# Main function
if __name__ == "__main__":
    # Read MQTT configuration
    mqtt_config = read_config('config/publisher.ini')

    # Load keys
    private_key, public_key = load_keys("config/private192.pem", "config/public192.pem")

    # Define message to be signed and published
    message = 'sds'
    message_bytes = message.encode('utf-8')

    # Sign the message
    signature = sign_message(private_key, message_bytes)

    # Verify the signature
    if verify_signature(public_key, signature, message_bytes):
        print("Signature verified successfully.")
    else:
        print("BAD SIGNATURE")

    # Publish the message over MQTT
    publish_message(mqtt_config, mqtt_config.get('mqtt', 'topic'), message_bytes)
