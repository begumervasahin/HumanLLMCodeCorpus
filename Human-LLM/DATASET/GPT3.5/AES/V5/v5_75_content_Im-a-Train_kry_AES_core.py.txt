# Importing the keyExpand function from the key_schedule module
from src.key_schedule import keyExpand

# Define the main function to generate and print the expanded key
def main(key):
    # Generate the expanded key using keyExpand function
    expanded_key = keyExpand(key, key_size=4, block_size=4, rounds=10)

    # Print each word of the expanded key along with its index
    for index, word in enumerate(expanded_key):
        print(f"Word {index+1}: {hex(word)}")

# Check if the script is being executed directly
if __name__ == '__main__':
    # Call the main function with a sample key value
    main(0x2b7e151628aed2a6abf7158809cf4f3c)
