from AEstrela import AEstrela
from BestFirst import BestFirst
import logging
import minesweeper as mns
import sys

class Game:
    def __init__(self):
        logging.basicConfig(filename='minesweeperIa.log', level=logging.DEBUG,
                            format='%(levelname)s %(funcName)s => %(message)s')
        self.algorithm = self._initialize_algorithm()

    def _initialize_algorithm(self):
        algorithm_choice = self._get_algorithm_choice()
        if algorithm_choice == 1:
            return AEstrela()
        elif algorithm_choice == 2:
            return BestFirst()

    def _get_algorithm_choice(self):
        if len(sys.argv) > 1:
            return int(sys.argv[1])
        else:
            print('Algoritmos:\nA*:1\nBestFirst:2')
            return int(input('Selecione o Algoritmo(1 ou 2):')) 

    def play_game(self):
        grid_size, num_mines = self._get_game_settings()     
        curr_grid = [[' ' for _ in range(grid_size)] for _ in range(grid_size)]
        grid = []
        flags = []
        cell_ant = None
        win = True
        game = True
        initial_cell = self.algorithm.jogadaInicial()
        cell_to_open = (initial_cell.getX(), initial_cell.getY())
        if not grid:
            grid, mines = mns.setupgrid(grid_size, cell_to_open, num_mines)
        while game:
            cell = self.algorithm.definirJogada(curr_grid)
            if cell is None:
                print('Perdeu, a lista de abertos ficou vazia') 
                win = False
                game = False
            else:
                cell_to_open = (cell.getX(), cell.getY())
                print(cell_to_open, cell.getData())
                if cell.getData() == 'F':
                    game = mns.jogar(cell_to_open, curr_grid, grid, flags, mines, flag=True)
                else:
                    game = mns.jogar(cell_to_open, curr_grid, grid, flags, mines)
                    cell_ant = cell
        return win

    def _get_game_settings(self):
        if len(sys.argv) > 3:
            return int(sys.argv[2]), int(sys.argv[3])
        else:
            grid_size = int(input('Selecione o tamanho do tabuleiro:'))
            num_mines = int(input('Selecione o numero de bombas:'))
            return grid_size, num_mines

mine = Game()
mine.play_game()
