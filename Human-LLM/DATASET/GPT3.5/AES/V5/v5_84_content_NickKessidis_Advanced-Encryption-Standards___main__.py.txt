import random

def generate_state_array(number):
    """
    Generate the State array from the given random number.

    Args:
        number (int): The random number.

    Returns:
        list: The State array.
    """
    state = ["0"] * 16
    for i in range(16):
        binary_slice = bin(number)[8*i+2:8*i+10]
        state[i] = binary_slice
    return state

def perform_substitution(state, sbox):
    """
    Perform the Substitution process on the State array.

    Args:
        state (list): The State array.
        sbox (list): The S-box.

    Returns:
        list: The State array after substitution.
    """
    for i in range(16):
        row = int(state[i][:4], 2)
        col = int(state[i][4:], 2)
        state[i] = sbox[(col + row) % 16]
    return state

if __name__ == "__main__":
    # Generate a random number and key
    number = random.randrange(2**127, (2**128)-1)
    key = random.randrange(2**127, (2**128)-1)

    # Display the result of XOR and the plain text
    xor_result = number ^ key
    print("Result of XOR:", bin(xor_result))
    print("Plain text:", bin(number))

    # Generate the State array
    state = generate_state_array(number)

    # Display the State array
    print("\nState Array:")
    for row in state:
        print(row)

    # Generate a random S-box
    sbox = [random.randrange(2**7, (2**8)-1) for _ in range(256)]

    # Perform the Substitution process
    state_after_substitution = perform_substitution(state, sbox)

    # Display the State array after the Substitution process
    print("\nState Array after the Substitution process:")
    for row in state_after_substitution:
        print(row)
