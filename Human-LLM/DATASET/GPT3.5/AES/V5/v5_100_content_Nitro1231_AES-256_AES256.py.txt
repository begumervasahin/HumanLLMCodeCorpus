import base64
import random
import string
from Crypto import Random
from Crypto.Cipher import AES

BLOCK_SIZE = 32

def pad(s):
    padding_length = BLOCK_SIZE - len(s.encode('utf-8')) % BLOCK_SIZE
    padding = padding_length * chr(padding_length).encode('utf-8')
    return s.encode('utf-8') + padding

def unpad(s):
    return s[:-s[-1]]

class AES256:
    def __init__(self, key):
        self.key = self.generate_key(key)

    def encrypt(self, raw):
        padded_raw = pad(raw)
        iv = Random.new().read(AES.block_size)
        cipher = AES.new(self.key, AES.MODE_CBC, iv)
        encrypted = cipher.encrypt(padded_raw)
        return base64.b64encode(iv + encrypted)

    def decrypt(self, enc):
        enc = base64.b64decode(enc)
        iv = enc[:AES.block_size]
        cipher = AES.new(self.key, AES.MODE_CBC, iv)
        decrypted = cipher.decrypt(enc[AES.block_size:])
        return unpad(decrypted).decode('utf-8')

    def generate_key(self, key):
        missing_chars = 16 - len(key)
        random.seed(key)
        letters = string.ascii_letters
        random_chars = ''.join(random.choice(letters) for _ in range(missing_chars))
        return (key + random_chars).encode('utf-8')
