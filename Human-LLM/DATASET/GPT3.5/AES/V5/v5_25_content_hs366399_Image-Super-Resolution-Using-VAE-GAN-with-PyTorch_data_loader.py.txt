import os
import cv2
import torch
from PIL import Image
from torchvision import transforms
from torch.utils.data import Dataset, DataLoader

# Check and print the name of the available CUDA device
if torch.cuda.is_available():
    print(f"Available CUDA Device: {torch.cuda.get_device_name(0)}")
else:
    print("CUDA device not available.")

class ImageLabelDataset(Dataset):
    """
    A custom dataset class for loading image-label pairs.
    """
    def __init__(self, data_path, image_transform=None, label_transform=None):
        """
        Initializes the dataset with paths and transforms.

        Args:
        - data_path (dict): Contains paths to images and labels with keys 'data' and 'labels'.
        - image_transform (callable, optional): Transform for image preprocessing.
        - label_transform (callable, optional): Transform for label preprocessing.
        """
        self.data_path = data_path['data']
        self.label_path = data_path['labels']
        self.image_transform = image_transform
        self.label_transform = label_transform

        self.data_files = sorted(os.listdir(self.data_path))
        self.label_files = sorted(os.listdir(self.label_path))
        
        if len(self.data_files) != len(self.label_files):
            raise ValueError("Mismatch in number of images and labels.")

    def __len__(self):
        """Returns the dataset size."""
        return len(self.data_files)

    def __getitem__(self, index):
        """Retrieves an image-label pair by index."""
        if index >= len(self):
            raise IndexError("Index out of range")

        image_path = os.path.join(self.data_path, self.data_files[index])
        label_path = os.path.join(self.label_path, self.label_files[index])

        # Load and convert image and label
        image = Image.open(image_path).convert('RGB')
        label = Image.fromarray(cv2.cvtColor(cv2.imread(label_path), cv2.COLOR_BGR2RGB))

        # Apply transformations if specified
        if self.image_transform:
            image = self.image_transform(image)
        if self.label_transform:
            label = self.label_transform(label)
        else:
            # Default transform for label (if not specified)
            label = transforms.ToTensor()(label)

        return image, label
