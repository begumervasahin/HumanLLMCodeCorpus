from utils import tools as tl
from synoptic.section1 import Group_iRixhVV

def get_formatted_url_for_reports():
    """
    Constructs and returns a URL for fetching synoptic weather reports based on the current date and time.
    
    Returns:
        str: A URL formatted with the current date and time parameters.
    """
    today, six_hours_ago = tl.fecha_para_synops()
    base_url = ('http://ogimet.com/display_synops2.php?lugar=78764+78762+78774+78767&tipo=ALL&ord=DIR&nil=SI'
                '&fmt=txt&ano={}&mes={}&day={}&hora={}&anof={}&mesf={}&dayf={}&horaf={}&enviar=Ver')
    formatted_url = base_url.format(six_hours_ago[0], six_hours_ago[1].zfill(2), six_hours_ago[2].zfill(2),
                                     six_hours_ago[3].zfill(2), today[0], today[1].zfill(2),
                                     today[2].zfill(2), today[3].zfill(2))
    return formatted_url
