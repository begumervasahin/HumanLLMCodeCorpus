import sys

# Global state variable
state = {
    'size': 16,
    'sBox': [],
    'hex': {
        '0': 0x0, '1': 0x1, '2': 0x2, '3': 0x3,
        '4': 0x4, '5': 0x5, '6': 0x6, '7': 0x7,
        '8': 0x8, '9': 0x9, 'a': 0xa, 'b': 0xb,
        'c': 0xc, 'd': 0xd, 'e': 0xe, 'f': 0xf,
    },
    'LAT': [],
}

# Function to read sBox values
def get_values(state, mode):
    if mode:
        try:
            with open("sBox.txt", "r") as f:
                contents = f.read().strip()
                state['sBox'] = [state['hex'][char] for char in contents[:state['size']]]
        except FileNotFoundError:
            print("sBox.txt not found!")
            sys.exit(1)
    else:
        for i in range(state['size']):
            val = check_value(state, i)
            state['sBox'].append(state['hex'][val])

# Function to check sBox value validity
def check_value(state, i):
    while True:
        val = input(f'Enter the sBox value for {hex(i)}: ')
        if len(val) == 1 and (val.isdigit() and int(val) in range(10)) or (val in 'abcdef'):
            return val
        else:
            print(f'Invalid input for {hex(i)}!')
            
# Function to print sBox table
def print_table(state):
    print('S-Box Table:')
    for i in range(state['size']):
        print(hex(i), end=' | ')
    print('\n----|-'*state['size'])
    for i in range(state['size']):
        print(hex(state['sBox'][i]), end=' | ')
    print('\n\nCalculating ...', end='\r')

# Function to calculate Linear Approximation Table (LAT)
def traverse(state):
    for row in state['hex']:
        lat_row = []
        for col in state['hex']:
            lat_row.append(calc_value(state, row, col))
        state['LAT'].append(lat_row)

# Function to calculate values for LAT
def calc_value(state, input_mask, output_mask):
    total = 0
    input_mask = state['hex'][input_mask]
    output_mask = state['hex'][output_mask]
    for each_hex in state['hex']:
        each_hex = state['hex'][each_hex]
        sbox_i = each_hex
        sbox_o = state['sBox'][int(each_hex)]
        input_bit = [(0b0001 << i) & input_mask >> i for i in range(4)]
        output_bit = [(0b0001 << i) & output_mask >> i for i in range(4)]
        if sum(input_bit) % 2 == sum(output_bit) % 2:
            total += 1
    return total - 8

# Function to print LAT
def print_LAT(state):
    columns = '0123456789ABCDEF'
    print('\nLinear Approximation Table (LAT):')
    print('    ', end='')
    for hex_no in state['hex']:
        print(' {:4} '.format(hex_no[-1].upper()), end='')
    print('\n  |' + '-----|' * 16)
    for index, row in enumerate(state['LAT']):
        print('{}'.format(columns[index]), end=' |')
        for element in row:
            print('{:3}  |'.format(element), end='')
        print('\n  |' + '-----|'*16)

# Function to calculate maximum absolute value in LAT
def lat_max(state):
    max_val = -10000000
    for row in state['LAT'][1:]:
        for element in row:
            max_val = max(max_val, abs(element))
    return max_val
