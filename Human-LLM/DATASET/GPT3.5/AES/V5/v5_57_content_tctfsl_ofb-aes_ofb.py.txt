import random
import aes

def chunks(string, block_size):
    """Divides a string into fixed-size chunks."""
    for i in range(0, len(string), block_size):
        yield string[i:i + block_size]

def generate_random_IV(length):
    """Generates a random initialization vector (IV) of a given length."""
    return "".join(str(random.randint(0, 1)) for _ in range(length))

def binvalue(val, bitsize):
    """Converts a value to its binary representation of a specified size."""
    binval = bin(val)[2:] if isinstance(val, int) else bin(ord(val))[2:]
    if len(binval) > bitsize:
        raise ValueError("Binary value larger than the expected size")
    return binval.zfill(bitsize)

def string_to_bit_array(text):
    """Converts a string to a bit array."""
    return [int(bit) for char in text for bit in binvalue(ord(char), 8)]

def bit_array_to_string(array):
    """Converts a bit array to a string."""
    return "".join([chr(int("".join(map(str, array[i:i+8])), 2)) for i in range(0, len(array), 8)])

def encrypt(block_size, plaintext_blocks, shift_register):
    """Encrypts plaintext blocks."""
    key = "11001001110010011100100111001001110010011100100111001001110010011100100111001001110010011100100111001001110010011100100111001001"
    ciphertext_blocks = []
    for plaintext_block in plaintext_blocks:
        temp_register = aes.encryption(shift_register, key)
        ciphertext_block = [temp_register[m] ^ plaintext_block[m] for m in range(block_size)]
        shift_register = temp_register[:]
        ciphertext_blocks.append(ciphertext_block)
    return ciphertext_blocks

def encrypt_with_error(block_size, plaintext_blocks, shift_register):
    """Encrypts plaintext blocks with a one-bit error introduced in the second block."""
    key = "11001001110010011100100111001001110010011100100111001001110010011100100111001001110010011100100111001001110010011100100111001001"
    ciphertext_blocks = []
    for i, plaintext_block in enumerate(plaintext_blocks):
        temp_register = aes.encryption(shift_register, key)
        ciphertext_block = [temp_register[m] ^ (plaintext_block[m] if not (i == 1 and m == 1) else 1 - plaintext_block[m]) for m in range(block_size)]
        shift_register = temp_register[:]
        ciphertext_blocks.append(ciphertext_block)
    return ciphertext_blocks

def decrypt_blocks(decrypted_blocks):
    """Decrypts ciphertext blocks."""
    decrypted_text = [bit for block in decrypted_blocks for bit in block]
    return bit_array_to_string(decrypted_text)

# User input
plaintext = input('Enter plaintext:')
block_size = int(input('Block size:'))

# Convert plaintext to bit array
plaintext_bit_array = string_to_bit_array(plaintext)
print("PlaintextArrayLength:", len(plaintext_bit_array))

# Adjust plaintext to fit block size
padding = block_size - len(plaintext_bit_array) % block_size
plaintext_bit_array += [0] * padding

# Divide plaintext into blocks
plaintext_blocks = list(chunks(plaintext_bit_array, block_size))

# Generate initialization vector (IV)
IV_size = 128
shift_register = generate_random_IV(IV_size)

# Encrypt plaintext blocks
ciphertext_blocks = encrypt(block_size, plaintext_blocks, shift_register)
wrong_ciphertext_blocks = encrypt_with_error(block_size, plaintext_blocks, shift_register)

# Decrypt ciphertext blocks
decrypted_blocks = encrypt(block_size, ciphertext_blocks, shift_register)
decrypted_text_string = decrypt_blocks(decrypted_blocks)

# Encrypt ciphertext blocks with a one-bit error introduced in the second block
wrong_blocks = encrypt_with_error(block_size, wrong_ciphertext_blocks, shift_register)
wrong_decrypted_text_string = decrypt_blocks(wrong_blocks)

# Display results
print("\nCipherTextBlockList               : ", ciphertext_blocks)
print("CipherTextBlockList (1 bit error) : ", wrong_ciphertext_blocks)
print("\nDecrypted BlockList (no error)    : ", decrypted_blocks)
print("Decrypted BlockList (1 bit error) : ", wrong_blocks)
print("\nDecrypted string          : ", decrypted_text_string)
print("Incorrect decrypted string: ", wrong_decrypted_text_string)
