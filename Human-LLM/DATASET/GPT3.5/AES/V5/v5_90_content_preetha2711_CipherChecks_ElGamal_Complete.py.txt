import random
import math
import sys

# Set recursion limit to prevent stack overflow
sys.setrecursionlimit(100000)

# Miller-Rabin primality test
def miller_rabin_test(number):
    is_prime = False
    num_minus_1 = number - 1
    k = 0

    # Find the number of times num_minus_1 can be divided by 2
    while num_minus_1 % 2 == 1:
        num_minus_1 = num_minus_1 / 2
        k += 1

    # Compute m and choose a random integer a
    m = num_minus_1 // (2 ** k)
    a = random.randint(0, num_minus_1)

    # Compute b = a^m mod number
    b = pow(a, m, number)

    # Check if b is congruent to 1 or -1 (mod number)
    if b == 1:
        is_prime = True
    else:
        for i in range(k):
            if b % number == -1:
                is_prime = True
            else:
                b = pow(b, 2, number)

    return is_prime

# Generate a prime number of specified length
def generate_prime_number(length=300):
    is_prime = False

    # Keep generating random numbers until a prime number is found
    while not is_prime:
        random_num = random.getrandbits(length)
        # Ensure that the number is odd and has both the most significant and least significant bits set to 1
        random_num |= (1 << length - 1) | 1
        is_prime = miller_rabin_test(random_num)

    return random_num

# Get two prime numbers p and q such that q = 2p + 1
def get_two_primes():
    is_prime_q = False

    # Keep generating random numbers for p until a corresponding prime q is found
    while not is_prime_q:
        p = generate_prime_number()
        q = 2 * p + 1
        is_prime_q = miller_rabin_test(q)

    return p, q

# Lucas test for finding primitive roots modulo p
def lucas_test_for_primitive_roots(p):
    Q = []

    # Find the divisors of (p-1)
    for i in range(2, p):
        if (p - 1) % i == 0:
            Q.append(i)

    primitive_roots = []

    # Check if each integer from 2 to p-1 is a primitive root modulo p
    for g in range(2, p):
        is_primitive_root = True
        for q in Q:
            if (g ** int((p - 1) / q)) % p == 1:
                is_primitive_root = False
                break
        if is_primitive_root:
            primitive_roots.append(g)

    return primitive_roots
