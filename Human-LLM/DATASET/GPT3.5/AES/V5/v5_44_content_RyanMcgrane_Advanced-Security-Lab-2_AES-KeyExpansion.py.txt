# KeyExpansion: expands the cipher key into a key schedule      
def KeyExpansion(CipherKeyArray, CipherKeySize, expandedKeySize):
    expandedKey = CipherKeyArray[:]
    i = CipherKeySize
    rconIteration = 1
    while i < expandedKeySize:
        temp = expandedKey[i-4:i]
        if i % CipherKeySize == 0:
            temp = RotWord(temp)
            temp = [SubWord(byte) for byte in temp]
            temp[0] ^= getRconValue(rconIteration)
            rconIteration += 1
        if CipherKeySize == 256 / 8 and (i % CipherKeySize) == 16:
            temp = [SubWord(byte) for byte in temp]
        expandedKey.extend([expandedKey[i-CipherKeySize] ^ temp[j] for j in range(4)])
        i += 4
    return expandedKey
