import Aes

# Main loop for choosing between encryption and decryption
while True:
    operation = input('Please choose cipher (c) or decipher (d): ')

    # Encryption operation
    if operation == 'c':
        while True:
            plain_text = input('Enter the plaintext string: ')

            # Validate plaintext length
            if len(plain_text) != 16:
                print("The length of the string is not 16 ({})".format(len(plain_text)))
                continue

            key_length = input('Enter the key length (16, 24, 32): ')

            # Validate key length
            if key_length not in ["16", "24", "32"]:
                print('The key length is not available.')
                continue

            cipher_key = input('Enter the cipher key: ')

            # Validate cipher key length
            if len(cipher_key) != int(key_length):
                print('The length of the cipher key is not {} ({})'.format(key_length, len(cipher_key)))
                continue

            # Perform encryption
            aes_cipher = Aes.AesCore(key_length, list(plain_text), list(cipher_key))
            res_cipher_text = aes_cipher.Cipher()
            aes_cipher.FormatPrint('Cipher text', res_cipher_text)
            break

    # Decryption operation
    elif operation == 'd':
        while True:
            cipher_text_string = input('Enter the cipher text string: ')
            cipher_text = cipher_text_string.split()

            # Validate cipher text length
            if len(cipher_text) != 16:
                print("The length of the string is not 16 ({})".format(len(cipher_text)))
                continue

            key_length = input('Enter the key length (16, 24, 32): ')

            # Validate key length
            if key_length not in ["16", "24", "32"]:
                print('The key length is not available.')
                continue

            cipher_key = input('Enter the cipher key: ')

            # Validate cipher key length
            if len(cipher_key) != int(key_length):
                print('The length of the cipher key is not {} ({})'.format(key_length, len(cipher_key)))
                continue

            # Perform decryption
            aes_decipher = Aes.AesCore(key_length, cipher_text, list(cipher_key))  
            res_plain_text = aes_decipher.InvCipher()
            print("Plain text: {}".format(res_plain_text))
            break

    else:
        print("Please input 'c' or 'd'!")
        continue
