import aes

class EasyAES:
    def __init__(self):
        self.cipher = aes.aes(256)

    def encrypt(self, text, key):
        blocks = self._sub_divide(text, 16)
        if not blocks:
            return
        padding = 16 - len(blocks[-1])
        blocks[-1] += padding * "="
        blocks.append(format(padding, "x") * 16)
        key_hash = self._expand_key(key)
        encrypted_blocks = ""
        for block in blocks:
            encrypted_block = self.cipher.encrypt(str(key_hash), block)
            hex_encrypted_block = self._convert_to_hex(encrypted_block)
            encrypted_blocks += hex_encrypted_block
        return encrypted_blocks

    def decrypt(self, cipher_text, key):
        blocks = self._sub_divide(cipher_text, 32)
        if not blocks:
            return
        key_hash = self._expand_key(key)
        plain_text = ""
        for block in blocks:
            cipher_block = self._hex_to_int_list(block)
            decrypted_block = self.cipher.decrypt(str(key_hash), cipher_block)
            plain_text += self.cipher.to_plain(decrypted_block)
        return self._remove_padding(plain_text)

    def _sub_divide(self, data, index):
        return [data[i:i + index] for i in range(0, len(data), index)]

    def _expand_key(self, key):
        key_hash = 1
        for i, c in enumerate(key):
            key_hash += ord(c) << (8 * i)
        while key_hash < 0xfffffffffffff:
            key_hash *= key_hash
        key_hash &= 0xfffffffffffff
        return key_hash

    def _convert_to_hex(self, block):
        hex_block = ""
        for i in block:
            hex_val = hex(i)[2:]
            if len(hex_val) == 1:
                hex_val = "0" + hex_val
            hex_block += hex_val
        return hex_block

    def _hex_to_int_list(self, block):
        return [int(block[i:i+2], 16) for i in range(0, len(block), 2)]

    def _remove_padding(self, text):
        padding = int(text[-1], 16)
        return text[:-padding]

if __name__ == "__main__":
    cipher = EasyAES()
    KEY = "Tis but a scratch!"
    DATA = "A scratch?! your arm's off!"
    cipher_text = cipher.encrypt(DATA, KEY)
    print("Cipher text:", cipher_text)
    plain_text = cipher.decrypt(cipher_text, KEY)
    print("Plain text:", plain_text)
