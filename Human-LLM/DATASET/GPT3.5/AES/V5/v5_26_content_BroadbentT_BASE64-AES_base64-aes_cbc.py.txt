def pad(s):
    """Pad the string to match the block size."""
    block_size = 32
    padding_length = block_size - len(s) % block_size
    return s + padding_length * chr(padding_length)

def unpad(s):
    """Unpad the string."""
    padding_length = ord(s[-1])
    return s[:-padding_length]

def generate_key(company_key):
    """Generate a key using PBKDF2."""
    salt = 'Pots de sel et de poivre'
    key = PBKDF2(company_key, salt, 64, 1000)
    return key[:32]

def encrypt(plain_text, company_key):
    """Encrypt the plain text using AES."""
    private_key = generate_key(company_key)
    padded_text = pad(plain_text)
    iv = Random.new().read(AES.block_size)
    cipher = AES.new(private_key, AES.MODE_CBC, iv)
    encrypted_text = cipher.encrypt(padded_text)
    return base64.b64encode(iv + encrypted_text)

def decrypt(encryption, company_key):
    """Decrypt the encrypted text using AES."""
    private_key = generate_key(company_key)
    decrypted_data = base64.b64decode(encryption)
    iv = decrypted_data[:AES.block_size]
    cipher = AES.new(private_key, AES.MODE_CBC, iv)
    decrypted_text = cipher.decrypt(decrypted_data[AES.block_size:])
    return unpad(decrypted_text)
