from Crypto.Cipher import AES
from Crypto.Random import get_random_bytes
import base64
import hashlib
import sys
import os

class AESCipher:
    def __init__(self, key):
        self.block_size = AES.block_size
        self.key = hashlib.sha256(key.encode()).digest()

    def encrypt(self, plaintext):
        plaintext = self._pad(plaintext)
        iv = get_random_bytes(self.block_size)
        cipher = AES.new(self.key, AES.MODE_CBC, iv)
        encrypted_bytes = cipher.encrypt(plaintext.encode())
        return base64.b64encode(iv + encrypted_bytes).decode('utf-8')

    def decrypt(self, encrypted_text):
        encrypted_data = base64.b64decode(encrypted_text)
        iv = encrypted_data[:self.block_size]
        cipher = AES.new(self.key, AES.MODE_CBC, iv)
        decrypted_bytes = cipher.decrypt(encrypted_data[self.block_size:])
        return self._unpad(decrypted_bytes).decode('utf-8')

    def _pad(self, s):
        padding_length = self.block_size - len(s) % self.block_size
        return s + (padding_length * chr(padding_length))

    @staticmethod
    def _unpad(s):
        return s[:-ord(s[len(s)-1:])]

def handle_file(operation, input_path, output_path, key):
    if not os.path.exists(input_path):
        print(f"Error: The file '{input_path}' does not exist.")
        sys.exit(1)
    
    aes = AESCipher(key)
    try:
        with open(input_path, 'r') as file:
            content = file.read()
        
        if operation == 'encrypt':
            processed_content = aes.encrypt(content)
        else:  # decrypt
            processed_content = aes.decrypt(content)

        with open(output_path, 'w') as file:
            file.write(processed_content)
        
        print(f"File {operation}ed successfully.")
    
    except Exception as e:
        print(f"An error occurred: {e}")
        sys.exit(1)

if __name__ == "__main__":
    if len(sys.argv) != 5:
        print("Usage: script.py -e|-d input_file output_file key")
        sys.exit(1)

    operation_flag, input_file, output_file, key = sys.argv[1:5]
    operation = 'encrypt' if operation_flag == '-e' else 'decrypt'

    handle_file(operation, input_file, output_file, key)
