import os
import numpy as np
import torch
from torch.utils.data import DataLoader, Dataset
from models.CASS_UNET import CASS
from utils import reverse_stft, compute_lx_error, compute_sdr

# Load data files
print("Loading data files...")
data_dir = '/data/yongzheng/cass/samples/'
stft_vocal_train = np.load(os.path.join(data_dir, 's1_train_mag.npy')).astype(np.float32)
stft_accom_train = np.load(os.path.join(data_dir, 's2_train_mag.npy')).astype(np.float32)
stft_mix_train = stft_vocal_train + stft_accom_train

stft_vocal_test = np.load(os.path.join(data_dir, 's1_test_mag.npy')).astype(np.float32)
stft_accom_test = np.load(os.path.join(data_dir, 's2_test_mag.npy')).astype(np.float32)
stft_mix_test = stft_vocal_test + stft_accom_test

stft_vocal_test_phase = np.load(os.path.join(data_dir, 's1_test_pha.npy')).astype(np.float32)
stft_accom_test_phase = np.load(os.path.join(data_dir, 's1_test_pha.npy')).astype(np.float32)
stft_mix_test_phase = np.load(os.path.join(data_dir, 'mix_test_pha.npy')).astype(np.float32)

print("Data files loaded!")

# Constants
BATCH_SIZE = 4
NUM_NETWORKS = 2
NUM_EPOCHS = 1000
LEARNING_RATE = 0.00002
WEIGHT_DECAY = 0
THRESHOLD = 20
DEVICE = torch.device("cuda:0" if torch.cuda.is_available() else "cpu")
SAVE_DIR = '/data/yongzheng/cass_musdb/results/cass_unet_2/'

# Custom dataset class
class IndexDataset(Dataset):
    def __init__(self, length):
        self.samples = np.arange(length)
    def __len__(self):
        return len(self.samples)
    def __getitem__(self, idx):
        return self.samples[idx]

# Data loaders
train_dataset = IndexDataset(len(stft_mix_train))
train_loader = DataLoader(train_dataset, batch_size=BATCH_SIZE, shuffle=True, num_workers=4)

test_dataset = IndexDataset(len(stft_mix_test))
test_loader = DataLoader(test_dataset, batch_size=BATCH_SIZE, shuffle=False, num_workers=4)

# Initialize CASS model
cass = CASS(NUM_NETWORKS, LEARNING_RATE, DEVICE, 100, WEIGHT_DECAY)
cass.build_model()

# Training loop
best_error = float('inf')
best_sdr = 0
counter = 0

for epoch in range(1, NUM_EPOCHS + 1):
    print("=" * 10)
    print("Epoch:", epoch)

    # Training phase
    for i, batch in enumerate(train_loader):
        indices = batch.numpy()
        x = stft_mix_train[indices]
        a = stft_vocal_train[indices]
        b = stft_accom_train[indices]
        enc_losses, dec_losses, dis_losses = cass.train(x, [a, b])

    # Testing phase
    vocal_errors = []
    accom_errors = []
    vocal_sdrs = []
    accom_sdrs = []

    for i, batch in enumerate(test_loader):
        indices = batch.numpy()
        x = stft_mix_test[indices]
        vocal_results, accom_results = cass.test(x)

        vocal_error = compute_lx_error(stft_vocal_test, vocal_results, stft_mix_test_phase)
        accom_error = compute_lx_error(stft_accom_test, accom_results, stft_mix_test_phase)
        vocal_sdr, _, _, _, _ = compute_sdr(stft_vocal_test, vocal_results, stft_mix_test_phase, stft_mix_test)
        accom_sdr, _, _, _, _ = compute_sdr(stft_accom_test, accom_results, stft_mix_test_phase, stft_mix_test)

        vocal_errors.append(vocal_error)
        accom_errors.append(accom_error)
        vocal_sdrs.append(vocal_sdr)
        accom_sdrs.append(accom_sdr)

    vocal_error = np.mean(vocal_errors)
    accom_error = np.mean(accom_errors)
    vocal_sdr = np.mean(vocal_sdrs)
    accom_sdr = np.mean(accom_sdrs)

    total_error = 0.5 * (vocal_error + accom_error)
    total_sdr = 0.5 * (vocal_sdr + accom_sdr)

    print("Total error:", total_error)
    print("Total SDR:", total_sdr)

    # Update best results and save models
    if total_error < best_error:
        best_error = total_error
        counter = 0
        np.save(os.path.join(SAVE_DIR, 's1_result'), vocal_results)
        np.save(os.path.join(SAVE_DIR, 's2_result'), accom_results)

    if total_sdr > best_sdr:
        best_sdr = total_sdr
        counter = 0
        cass.save_model(SAVE_DIR)
        np.save(os.path.join(SAVE_DIR, 's1_result'), vocal_results)
        np.save(os.path.join(SAVE_DIR, 's2_result'), accom_results)

    counter += 1

    # Check for early stopping
    if counter >= THRESHOLD:
        break

print("Training ended at epoch {}, with best testing error {:.4f}".format(epoch, best_error))
