import sys
from Crypto.Cipher import AES

class AESCipher:
    block_size = AES.block_size
    zero_padding = 1
    PKCS5Padding = 2
    PKCS7Padding = 3

    def __init__(self, key, mode=AES.MODE_ECB):
        if not isinstance(key, bytes):
            key = bytes(key, 'utf-8')
        self.key = key
        self.aes = AES.new(self.key, mode)

    @staticmethod
    def PKCS7_padding(s):
        pad_length = AESCipher.block_size - len(s) % AESCipher.block_size
        return s + bytes([pad_length] * pad_length)

    @staticmethod
    def PKCS7_unpadding(s):
        pad_length = s[-1]
        return s[:-pad_length]

    def hex_to_bytes(self, raw):
        if isinstance(raw, str):
            raw = bytes.fromhex(raw)
        return raw

    def encrypt(self, raw):
        padded_raw = AESCipher.PKCS7_padding(raw.encode("utf-8"))
        encrypted = self.aes.encrypt(padded_raw)
        return encrypted.hex().upper()

    def decrypt(self, enc):
        enc = self.hex_to_bytes(enc)
        decrypted = self.aes.decrypt(enc)
        unpadded_raw = AESCipher.PKCS7_unpadding(decrypted).decode('utf-8').strip()
        return unpadded_raw

# Initialize AESCipher with the key
key = '1234567890123456'.encode('utf-8')
aes = AESCipher(key)

# Process input from stdin
for line in sys.stdin:
    line = line.strip()
    if line:
        enc = aes.encrypt(line)
        print(enc)
        raw = aes.decrypt(enc)
        print(raw)
