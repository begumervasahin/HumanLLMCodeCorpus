import os
import sys
import getopt
from Crypto import Random
from Crypto.Cipher import AES
from Crypto.Hash import SHA256

class FileEncryptor:
    def __init__(self):
        """Initializes the FileEncryptor with an empty encryption key."""
        self.key = None

    def set_encryption_key(self, password):
        """Sets the encryption key based on the given password."""
        hash_of_password = SHA256.new(password.encode()).digest()
        self.key = hash_of_password

    def encrypt_data(self, data):
        """Encrypts the given data using AES-CBC mode."""
        initialization_vector = Random.new().read(AES.block_size)
        cipher = AES.new(self.key, AES.MODE_CBC, initialization_vector)
        padded_data = self._pad(data)
        encrypted_data = cipher.encrypt(padded_data)
        return initialization_vector + encrypted_data

    def decrypt_data(self, encrypted_data):
        """Decrypts the given data using AES-CBC mode."""
        initialization_vector = encrypted_data[:AES.block_size]
        cipher = AES.new(self.key, AES.MODE_CBC, initialization_vector)
        decrypted_padded_data = cipher.decrypt(encrypted_data[AES.block_size:])
        return self._unpad(decrypted_padded_data)

    def _pad(self, data):
        """Pads the data to make its length a multiple of AES block size."""
        padding_length = AES.block_size - len(data) % AES.block_size
        padding = bytes([padding_length]) * padding_length
        return data + padding

    def _unpad(self, data):
        """Removes padding from the data."""
        padding_length = data[-1]
        return data[:-padding_length]

    def encrypt_file(self, filepath):
        """Encrypts the file at the specified path."""
        with open(filepath, 'rb') as file:
            plaintext = file.read()
        encrypted_data = self.encrypt_data(plaintext)
        with open(filepath + ".enc", 'wb') as file:
            file.write(encrypted_data)
        os.remove(filepath)
        print(f"Encrypted {filepath}")

    def decrypt_file(self, filepath):
        """Decrypts the encrypted file at the specified path."""
        with open(filepath, 'rb') as file:
            encrypted_data = file.read()
        plaintext = self.decrypt_data(encrypted_data)
        original_filepath = filepath[:-4]  # Remove the '.enc' extension
        with open(original_filepath, 'wb') as file:
            file.write(plaintext)
        os.remove(filepath)
        print(f"Decrypted {filepath} to {original_filepath}")

    def process_directory(self, action):
        """Processes all files in the current directory based on the action ('encrypt' or 'decrypt')."""
        for root, _, filenames in os.walk("."):
            for filename in filenames:
                if filename != sys.argv[0]:  # Avoid processing the script itself
                    filepath = os.path.join(root, filename)
                    if action == "encrypt":
                        self.encrypt_file(filepath)
                    elif action == "decrypt" and filepath.endswith('.enc'):
                        self.decrypt_file(filepath)

    def display_help_message(self):
        """Displays the usage instructions for the script."""
        usage_text = """
        Usage:
            python FileEncryptor.py -p <password> [-e <filename> | -d <filename> | -E | -D]
        
        Options:
            -p, --password <password> : Set the password for encryption or decryption.
            -e, --encrypt <filename>  : Encrypt the specified file.
            -d, --decrypt <filename>  : Decrypt the specified file.
            -E                         : Encrypt all files in the current directory.
            -D                         : Decrypt all files in the current directory.
        """
        print(usage_text.strip())

    def run(self):
        """Parses command-line arguments and invokes the corresponding functionality."""
        try:
            options, _ = getopt.getopt(sys.argv[1:], "hp:e:d:ED", ["help", "password=", "encrypt=", "decrypt="])
        except getopt.GetoptError as error:
            print(error)
            self.display_help_message()
            sys.exit(2)

        for option, argument in options:
            if option in ("-h", "--help"):
                self.display_help_message()
                sys.exit()
            elif option in ("-p", "--password"):
                self.set_encryption_key(argument)
            elif option in ("-e", "--encrypt"):
                self.encrypt_file(argument)
            elif option in ("-d", "--decrypt"):
                self.decrypt_file(argument)
            elif option == "-E":
                self.process_directory("encrypt")
            elif option == "-D":
                self.process_directory("decrypt")

if __name__ == "__main__":
    encryptor = FileEncryptor()
    encryptor.run()
