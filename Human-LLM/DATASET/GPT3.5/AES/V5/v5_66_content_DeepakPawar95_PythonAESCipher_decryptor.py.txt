import os
from Crypto.Cipher import AES
from Crypto.Hash import SHA256
from Crypto import Random

class Colors:
    BLUE = '\033[94m'
    GREEN = '\033[92m'
    RED = '\033[31m'
    ENDC = '\033[0m'

def decrypt_message(key, filename):
    chunk_size = 64 * 256
    output_file = "de_" + filename

    with open(filename, 'rb') as infile:
        file_size = int(infile.read(16))
        IV = infile.read(16)
        decryptor = AES.new(key, AES.MODE_CBC, IV)

        with open(output_file, 'wb') as outfile:
            while True:
                chunk = infile.read(chunk_size)
                if not chunk:
                    break

                decrypted_chunk = decryptor.decrypt(chunk)
                decrypted_length = len(decrypted_chunk)

                if file_size > decrypted_length:
                    outfile.write(decrypted_chunk)
                else:
                    outfile.write(decrypted_chunk[:file_size])

                file_size -= decrypted_length

    print(Colors.GREEN + f"\n\n[Decrypted message stored in file]: de_{filename}\n\n")

def create_key(password):
    hasher = SHA256.new(password.encode('utf-8'))
    return hasher.digest()

def main():
    print(Colors.RED + "\n           Python AES Decryptor            ")
    filename = input(Colors.BLUE + "[Message Filename]: " + Colors.ENDC)
    password = input(Colors.BLUE + "[Password]: " + Colors.ENDC)
    key = create_key(password)
    decrypt_message(key, filename)

if __name__ == "__main__":
    main()
