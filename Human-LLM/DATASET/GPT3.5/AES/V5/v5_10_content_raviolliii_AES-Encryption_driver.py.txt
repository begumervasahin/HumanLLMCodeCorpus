from Crypto.Cipher import AES
from Crypto.Util.Padding import pad
from Crypto.Random import get_random_bytes

class AESCipher:
    def __init__(self, key):
        self.key = pad(key.encode(), AES.block_size)  # Ensure the key fits the block size with padding

    def encrypt(self, plaintext):
        cipher = AES.new(self.key, AES.MODE_CBC)  # Initialize cipher in CBC mode
        iv = cipher.iv  # Initialization vector for decryption
        encrypted = cipher.encrypt(pad(plaintext.encode(), AES.block_size))  # Encrypt and pad the plaintext
        return iv + encrypted  # Prepend IV to the encrypted data for use in decryption

def main():
    # Define key and message
    key = "Thats my Kung Fu"
    message = "Two One Nine Two"

    # Initialize AES cipher with the given key
    aes_cipher = AESCipher(key)

    # Encrypt the message
    encrypted_message = aes_cipher.encrypt(message)

    # Display the original and encrypted messages
    print("Message:\t", message)
    print("Encrypted:\t", encrypted_message.hex())  # Convert bytes to hex for readable output

if __name__ == "__main__":
    main()
