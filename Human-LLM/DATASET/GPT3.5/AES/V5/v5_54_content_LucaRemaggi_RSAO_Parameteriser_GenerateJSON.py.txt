import numpy as np
import json

class JSONGenerator:
    def __init__(self, param_early, param_late, name, max_early, filename, obj_type):
        self.param_early = param_early
        self.param_late = param_late
        self.name = name
        self.max_early = max_early
        self.filename = filename
        self.obj_type = obj_type
        self.lib_entry = []

    def generate_object_vector_room_library(self):
        self.lib_entry = {
            'name': self.name,
            'type': self.obj_type,
            'id': 0,
            'channels': 0,
            'priority': 0,
            'level': 1.0,
            'position': self._format_position(self.param_early['Direct_sound']['doa']),
            'room': {'ereflect': []}
        }

        for idx_refl in range(1, self.max_early + 1):
            reflect_data = self.param_early['Reflection' + str(idx_refl)]
            reflect_entry = {
                'level': self._format_value(reflect_data['level']),
                'delay': self._format_value(reflect_data['toa']),
                'position': self._format_position(reflect_data['doa']),
                'biquadsos': [self._format_biquad(filter_params) for filter_params in reflect_data['filtersos']]
            }
            self.lib_entry['room']['ereflect'].append(reflect_entry)

        self.lib_entry['room'].update({
            'lreverb': {
                'delay': self._format_value(self.param_late['Late']['toa']),
                'level': self._format_list(self.param_late['Late']['level']),
                'attacktime': self._format_list(self.param_late['Late']['attacktimes']),
                'decayconst': self._format_list(self.param_late['Late']['expdecays'])
            }
        })

        return self

    def save_to_json(self):
        with open(self.filename, 'w') as outfile:
            json.dump(self.lib_entry, outfile)

    @staticmethod
    def _format_position(position):
        return {'az': '{:.2f}'.format(position[0]), 'el': '{:.2f}'.format(position[1]), 'radius': '1.00'}

    @staticmethod
    def _format_biquad(params):
        return {f'{coef}': JSONGenerator._format_value(value) for coef, value in zip(['b0', 'b1', 'b2', 'a0', 'a1', 'a2'], params)}

    @staticmethod
    def _format_value(value):
        return '{:.3e}'.format(value)

    @staticmethod
    def _format_list(list_val):
        return ', '.join(JSONGenerator._format_value(val) for val in list_val)

# Example usage:
if __name__ == "__main__":
    # Define parameters
    param_early = {...}
    param_late = {...}
    name = "Object1"
    max_early = 3
    filename = "output.json"
    obj_type = "Type1"

    # Create instance and generate JSON
    generator = JSONGenerator(param_early, param_late, name, max_early, filename, obj_type)
    generator.generate_object_vector_room_library().save_to_json()
