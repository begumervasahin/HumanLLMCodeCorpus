from flask import Flask, session, request
from Crypto import Random
from Crypto.Cipher import AES
from secret import flag, aes_key, secret_key

app = Flask(__name__)
app.secret_key = secret_key
app.config['DEBUG'] = False

class AESCipher:
    def __init__(self):
        self.block_size = AES.block_size

    def pad(self, s):
        pad_length = self.block_size - len(s) % self.block_size
        padding = chr(pad_length) * pad_length
        return s + padding

    def encrypt(self, raw):
        iv = session.get("IV") or Random.new().read(AES.block_size)
        cipher = AES.new(aes_key, AES.MODE_CBC, iv)
        padded_data = self.pad(raw + flag)
        encrypted = cipher.encrypt(padded_data)
        session["IV"] = encrypted[-AES.block_size:]
        return encrypted.hex()

@app.route("/")
def main():
    with open(__file__, 'r') as file:
        return file.read()

@app.route("/flag_format")
def flag_format():
    return """The flag format for this challenge is non-standard.
The flag to obtain for this challenge (stored in the flag variable) is 32-HEX only.
Once you obtain this flag, submit your solution in the regular 247CTF{32-HEX} format."""

@app.route("/encrypt")
def encrypt():
    try:
        plaintext = request.args.get('plaintext')
        return AESCipher().encrypt(plaintext.decode('hex'))
    except Exception as e:
        return f"Error: {e}"

if __name__ == "__main__":
    app.run()
