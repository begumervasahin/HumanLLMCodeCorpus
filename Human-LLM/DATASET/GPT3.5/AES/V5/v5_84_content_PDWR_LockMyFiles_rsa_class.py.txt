import json
from cryptography.hazmat.primitives import serialization
from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives.asymmetric import rsa, padding
from cryptography.hazmat.primitives import hashes

class RSA_Class:
    @staticmethod
    def generate_private_key():
        """
        Generate a private key.

        Returns:
            private_key: The generated private key.
        """
        private_key = rsa.generate_private_key(
            public_exponent=65537,
            key_size=2048,
            backend=default_backend()
        )
        return private_key

    @staticmethod
    def generate_public_key(private_key):
        """
        Generate a public key from the given private key.

        Args:
            private_key: The private key.

        Returns:
            public_key: The generated public key.
        """
        public_key = private_key.public_key()
        return public_key

    @staticmethod
    def save_private_key(file_path, private_key):
        """
        Save the private key to a file.

        Args:
            file_path: The file path to save the private key.
            private_key: The private key to save.
        """
        with open(file_path, "wb") as f:
            f.write(private_key.private_bytes(
                encoding=serialization.Encoding.PEM,
                format=serialization.PrivateFormat.TraditionalOpenSSL,
                encryption_algorithm=serialization.NoEncryption(),
            ))

    @staticmethod
    def save_public_key(file_path, public_key):
        """
        Save the public key to a file.

        Args:
            file_path: The file path to save the public key.
            public_key: The public key to save.
        """
        with open(file_path, "wb") as f:
            f.write(public_key.public_bytes(
                encoding=serialization.Encoding.PEM,
                format=serialization.PublicFormat.SubjectPublicKeyInfo,
            ))

    @staticmethod
    def load_private_key(file_path):
        """
        Load the private key from a file.

        Args:
            file_path: The file path to load the private key from.

        Returns:
            private_key: The loaded private key.
        """
        private_key = serialization.load_pem_private_key(
            open(file_path, 'rb').read(), password=None, backend=default_backend() 
        )
        return private_key

    @staticmethod
    def load_public_key(file_path):
        """
        Load the public key from a file.

        Args:
            file_path: The file path to load the public key from.

        Returns:
            public_key: The loaded public key.
        """
        public_key = serialization.load_pem_public_key(
            open(file_path, 'rb').read(), backend=default_backend()
        )
        return public_key

    @staticmethod
    def encrypt(public_key, plaintext):
        """
        Encrypt plaintext using the given public key.

        Args:
            public_key: The public key for encryption.
            plaintext: The plaintext to encrypt.

        Returns:
            ciphertext: The encrypted ciphertext.
        """
        ciphertext = public_key.encrypt(
            plaintext,
            padding.OAEP(
                mgf=padding.MGF1(algorithm=hashes.SHA256()),
                algorithm=hashes.SHA256(),
                label=None
            )
        )
        return ciphertext

    @staticmethod
    def decrypt(private_key, ciphertext):
        """
        Decrypt ciphertext using the given private key.

        Args:
            private_key: The private key for decryption.
            ciphertext: The ciphertext to decrypt.

        Returns:
            plaintext: The decrypted plaintext.
        """
        plaintext = private_key.decrypt(
            ciphertext,
            padding.OAEP(
                mgf=padding.MGF1(algorithm=hashes.SHA256()),
                algorithm=hashes.SHA256(),
                label=None
            )
        )
        return plaintext

    @staticmethod
    def save(path, text):
        """
        Save text to a file.

        Args:
            path: The file path to save the text to.
            text: The text to save.
        """
        data = {"aes_ciphered_password": text}
        with open(path, 'w') as f:
            json.dump(data, f)

    @staticmethod
    def load(path):
        """
        Load text from a file.

        Args:
            path: The file path to load text from.

        Returns:
            data: The loaded data.
        """
        with open(path, "r") as json_file:
            data = json.load(json_file)
        return data

# Example usage:
if __name__ == "__main__":
    # Generate a private key
    private_key = RSA_Class.generate_private_key()

    # Generate a public key
    public_key = RSA_Class.generate_public_key(private_key)

    # Save the private key to a file
    RSA_Class.save_private_key("private_key.pem", private_key)

    # Save the public key to a file
    RSA_Class.save_public_key("public_key.pem", public_key)

    # Load the private key from a file
    loaded_private_key = RSA_Class.load_private_key("private_key.pem")

    # Load the public key from a file
    loaded_public_key = RSA_Class.load_public_key("public_key.pem")

    # Encrypt plaintext using the public key
    plaintext = b"Hello, World!"
    ciphertext = RSA_Class.encrypt(loaded_public_key, plaintext)

    # Decrypt ciphertext using the private key
    decrypted_plaintext = RSA_Class.decrypt(loaded_private_key, ciphertext)        

    print("Original plaintext:", plaintext)
    print("Decrypted plaintext:", decrypted_plaintext.decode())
