def add(byte1, byte2):
    """
    Adds two 8-bit strings in GF(256).

    Args:
        byte1 (str): The first 8-bit string.
        byte2 (str): The second 8-bit string.

    Returns:
        str: The result of the addition in 8-bit string format.

    Raises:
        ValueError: If the input strings are not of length 8 or not of type str.
    """
    if len(byte1) != 8 or len(byte2) != 8 or not isinstance(byte1, str) or not isinstance(byte2, str):
        raise ValueError("Error in gf256.add(): Please enter 8-bit strings")

    int1 = converter.byte_to_int(byte1)
    int2 = converter.byte_to_int(byte2)
    xored = int1 ^ int2
    return converter.int_to_byte(xored)

    mod_red = 0b10001

def mul_bytes(q, w):
    """
    Multiplies two bytes in GF(256).

    Args:
        q (int): The first byte.
        w (int): The second byte.

    Returns:
        int: The result of the multiplication.
    """
    e = 0
    for _ in range(8):
        if w & 1 != 0:
            e ^= q
        bit_set = q & 0x80 != 0
        q <<= 1
        if bit_set:
            q ^= 0x1B
        w >>= 1
    return e % 256


def get_inverse(byt):
    """
    Returns the multiplicative inverse of a byte in GF(256).

    Args:
        byt (int): The byte for which the inverse is to be found.

    Returns:
        int: The multiplicative inverse of the byte.
    """
    if byt == 0:
        return 0

    for q in range(256):
        if mul_bytes(byt, q) == 1:
            return q

    return None


def pol_mod(pol, mod):
    """
    Performs polynomial modular division.

    Args:
        pol (int): The polynomial to be divided.
        mod (int): The modulus for division.

    Returns:
        int: The result of the polynomial modular division.
    """
    to_divide = pol
    while to_divide >= mod:
        sub_to_divide = ""
        pol_str = bin(to_divide)[2:]

        for bit in range(len(pol_str)):
            sub_to_divide += pol_str[bit]
            divide = int(sub_to_divide, 2)

            if divide > mod:
                to_divide = bin(divide ^ mod)[2:]

                if bit < len(pol_str):
                    to_divide += pol_str[bit + 1:]

                to_divide = int(to_divide, 2)
                break

    return to_divide


def mul_byte(byte1, byte2):
    """
    Multiplies two bytes in GF(256).

    Args:
        byte1 (str): The first 8-bit string.
        byte2 (str): The second 8-bit string.

    Returns:
        int: The result of the multiplication.

    Raises:
        ValueError: If the input strings are not of length 8 or not of type str.
    """
    if len(byte1) != 8 or len(byte2) != 8 or not isinstance(byte1, str) or not isinstance(byte2, str):
        raise ValueError("Error in gf256.mul_byte(): Please enter 8-bit strings")

    mod_red = 0b100011011
    out = 0
    factor = converter.byte_to_int(byte1)

    for q in range(8):
        if byte2[q] == "1":
            shifted = factor << 7 - q
            out ^= shifted

    return pol_mod(out, mod_red)
