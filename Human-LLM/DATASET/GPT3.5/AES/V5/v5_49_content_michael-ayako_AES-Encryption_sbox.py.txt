def string_split(bit_string):
    """Split a 16-bit string into four 4-bit strings."""
    hex0 = bit_string[0:4]
    hex1 = bit_string[4:8]
    hex2 = bit_string[8:12]
    hex3 = bit_string[12:16]
    return hex0, hex1, hex2, hex3

def sbox_enc(bits):
    """Perform S-box encryption."""
    encrypted_bits = []
    for bit in bits:
        hex0, hex1, hex2, hex3 = string_split(bit)
        first_byte = '{0:08b}'.format(int(Sbox[int(hex0, 2)][int(hex1, 2)]))
        second_byte = '{0:08b}'.format(int(Sbox[int(hex2, 2)][int(hex3, 2)]))
        encrypted_bits.append(first_byte + second_byte)
    return encrypted_bits

def sbox_dec(filename):
    """Perform S-box decryption."""
    logging.info("Reading file contents...")
    content = []
    with open(filename, 'r', encoding='utf-8') as file:
        for line in file:
            for char in line:
                content.append(char)
    bits = ['{0:016b}'.format(ord(char)) for char in content]
    decrypted_bits = []
    for bit in bits:
        hex0, hex1, hex2, hex3 = string_split(bit)
        first_byte = '{0:08b}'.format(int(InvSbox[int(hex0, 2)][int(hex1, 2)]))
        second_byte = '{0:08b}'.format(int(InvSbox[int(hex2, 2)][int(hex3, 2)]))
        decrypted_bits.append(first_byte + second_byte)
    return decrypted_bits
