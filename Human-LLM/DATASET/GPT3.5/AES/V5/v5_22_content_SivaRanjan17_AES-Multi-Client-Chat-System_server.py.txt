from socket import AF_INET, socket, SOCK_STREAM
from threading import Thread
import AES

# Define the master encryption key
key = b'master key'

def accept_incoming_connections():
    """Accepts incoming client connections."""
    while True:
        client, client_address = SERVER.accept()
        print(f"Client {client_address[0]}:{client_address[1]} has connected.")

        # Send a greeting message to the client
        greeting = "Greetings from the server! Please type your name and press enter."
        encrypted_greeting = AES.encrypt(key, greeting.encode('UTF-8'))
        client.send(encrypted_greeting)

        # Add client to the clients dictionary and start handling its messages
        addresses[client] = client_address
        Thread(target=handle_client, args=(client,)).start()    

def handle_client(client):
    """Handles a single client connection."""
    # Receive and decrypt the client's name
    name = AES.decrypt(key, client.recv(BUFSIZ)).decode("utf-8")
    welcome_msg = f"Welcome {name}! If you ever want to quit, type {{quit}} to exit."
    encrypted_welcome_msg = AES.encrypt(key, welcome_msg.encode('UTF-8'))
    client.send(encrypted_welcome_msg)
    clients[client] = name

    while True:
        # Receive and decrypt the client's message
        message = AES.decrypt(key, client.recv(BUFSIZ)).decode("utf-8")

        # Check if the client wants to quit
        if message != "{quit}":
            # Broadcast the message to all clients
            broadcast(message, name + ": ")
        else:
            client.close()
            del clients[client]

            # If there are no more clients, close the server
            if not clients:
                SERVER.close()
            break

def broadcast(msg, prefix=""):
    """Broadcasts a message to all the clients."""
    message = prefix + msg
    encrypted_message = AES.encrypt(key, message.encode('UTF-8'))
    for sock in clients:
        sock.send(encrypted_message)

# Initialize dictionaries to keep track of clients and their addresses
clients = {}
addresses = {}

# Define server constants
HOST = ''
PORT = 33000
BUFSIZ = 1024
ADDR = (HOST, PORT)

# Create the server socket
SERVER = socket(AF_INET, SOCK_STREAM)
SERVER.bind(ADDR)

if __name__ == "__main__":
    SERVER.listen(5)
    print("Waiting for connections...")

    # Start accepting incoming connections in a separate thread 
    ACCEPT_THREAD = Thread(target=accept_incoming_connections)  
    ACCEPT_THREAD.start()
    ACCEPT_THREAD.join()

    # Close the server socket
    SERVER.close()
