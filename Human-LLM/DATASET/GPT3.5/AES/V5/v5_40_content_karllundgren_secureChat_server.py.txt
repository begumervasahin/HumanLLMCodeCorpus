import socket
from Crypto.Cipher import AES
from Crypto.PublicKey import RSA
import threading

class Client:
    def __init__(self, host='127.0.0.1', port=5022):
        self.host = host
        self.port = port
        self.server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.client_socket = None
        self.cipher = None

    def send_message(self):
        while True:
            message = input("\nEnter a message: ")
            encrypted_message = self.cipher.encrypt(message.encode())
            self.client_socket.send(encrypted_message)

    def establish_connection(self):
        self.server_socket.bind((self.host, self.port))
        self.server_socket.listen(1)
        self.client_socket, _ = self.server_socket.accept()

    def import_rsa_key(self):
        with open('mykey.pem', 'r') as key_file:
            key_data = key_file.read()
        return RSA.importKey(key_data)

    def send_public_rsa_key(self, rsa_key):
        public_rsa_key = rsa_key.publickey().exportKey('DER')
        self.client_socket.send(public_rsa_key)

    def receive_and_decrypt_aes_key(self, rsa_key):
        aes_encrypted_key = self.client_socket.recv(1024)
        aes_key = rsa_key.decrypt(aes_encrypted_key)
        return aes_key

    def initialize_aes_cipher(self, aes_key):
        self.cipher = AES.new(aes_key, AES.MODE_CFB, aes_key)

    def receive_and_decrypt_messages(self):
        while True:
            data = self.client_socket.recv(1024)
            if not data:
                break
            decrypted_data = self.cipher.decrypt(data).decode() 
            print(decrypted_data)

    def run(self):
        self.establish_connection()
        rsa_key = self.import_rsa_key()
        print("RSA key imported.")
        self.send_public_rsa_key(rsa_key)
        print("Public RSA key sent.")
        aes_key = self.receive_and_decrypt_aes_key(rsa_key)
        print("AES encrypted key received and decrypted.")
        self.initialize_aes_cipher(aes_key)
        input_thread = threading.Thread(target=self.send_message)
        input_thread.daemon = True
        input_thread.start()
        self.receive_and_decrypt_messages()

def main():
    client = Client()
    client.run()

if __name__ == "__main__":
    main()
