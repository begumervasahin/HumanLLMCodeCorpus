import os
import pytest

# Configure environment variables before starting the test session
def configure_environment():
    configure_aesara_flags()
    configure_numba_bounds_check()

# Configure Aesara flags for optimization and shape error handling
def configure_aesara_flags():
    aesara_flags = os.environ.setdefault("AESARA_FLAGS", "")
    aesara_flags += ",warn__ignore_bug_before=all,on_opt_error=raise,on_shape_error=raise,cmodule__warn_no_version=True"
    os.environ["AESARA_FLAGS"] = aesara_flags

# Enable bounds checking for Numba
def configure_numba_bounds_check():
    os.environ["NUMBA_BOUNDSCHECK"] = "1"

# Add a command-line option to pytest for running slow tests
def add_slow_test_option(parser):
    parser.addoption(
        "--runslow", 
        action="store_true", 
        default=False, 
        help="Run tests marked as slow"
    )

# Configure pytest markers
def configure_pytest():
    configure_slow_marker()

# Configure the pytest slow marker
def configure_slow_marker():
    pytest.mark.slow = pytest.mark.marker("slow", mark="Mark a test as slow to run")

# Skip slow tests if the --runslow option is not provided
def skip_slow_tests_if_not_requested(config, items):
    if not config.getoption("--runslow"):
        mark_slow_tests_to_skip(items)

# Mark slow tests to be skipped
def mark_slow_tests_to_skip(items):
    for item in items:
        if "slow" in item.keywords:
            skip_slow_test(item)

# Helper function to skip a slow test
def skip_slow_test(item):
    skip_reason = "Use --runslow option to run this test"
    item.add_marker(pytest.mark.skip(reason=skip_reason))

if __name__ == "__main__":
    configure_environment()
    add_slow_test_option(pytest)
    configure_pytest()
    skip_slow_tests_if_not_requested(pytest.config, pytest.config.getoption("--runslow"))
    pytest.main()
