def gcd(a, b):
    """Calculate the greatest common divisor."""
    while a != 0:
        a, b = b % a, a
    return b

def find_modular_inverse(a, m):
    """Find the modular inverse."""
    if gcd(a, m) != 1:
        return None
    u1, u2, u3 = 1, 0, a
    v1, v2, v3 = 0, 1, m
    while v3 != 0:
        q = u3 // v3
        u1, u2, u3, v1, v2, v3 = v1, v2, v3, u1 - q * v1, u2 - q * v2, u3 - q * v3
    return u1 % m

def rabin_miller_find_prime(num):
    """Check if a number is prime using Rabin-Miller algorithm."""
    s = num - 1
    t = 0
    while s % 2 == 0:
        s //= 2
        t += 1
    for _ in range(5):
        a = random.randrange(2, num - 1)
        v = pow(a, s, num)
        if v != 1:
            i = 0
            while v != num - 1:
                if i == t - 1:
                    return False
                i += 1
                v = pow(v, 2, num)
    return True

def test_if_prime(num):
    """Test if a number is prime."""
    if num < 2:
        return False
    if num in LOW_PRIMES:
        return True
    for prime in LOW_PRIMES:
        if num % prime == 0:
            return False
    return rabin_miller_find_prime(num)

def generate_large_prime_of_key_size(key_size=DEFAULT_KEY_SIZE):
    """Generate a large prime number of the given key size."""
    while True:
        num = random.randrange(2**(key_size-1), 2**(key_size))
        if test_if_prime(num):
            return num

def generate_private_public_key(default_key_size):
    """Generate private and public keys."""
    print('Generating p prime...')
    p = generate_large_prime_of_key_size(default_key_size)
    print('Generating q prime...')
    q = generate_large_prime_of_key_size(default_key_size)
    n = p * q
    print('Generating e for (e, N) which is relatively prime to phi(n) = (p-1)*(q-1)...')
    while True:
        e = random.randrange(2 ** (default_key_size - 1), 2 ** default_key_size)
        if gcd(e, (p - 1) * (q - 1)) == 1:
            break
    print('Calculating d for private key that is mod inverse of e...')
    d = find_modular_inverse(e, (p - 1) * (q - 1))
    rsa_public_key = (n, e)
    rsa_private_key = (n, d)
    return rsa_public_key, rsa_private_key
