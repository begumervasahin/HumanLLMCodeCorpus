import tkinter as tk
from tkinter.ttk import ThemedStyle


class SuperNesne:
    def __init__(self, title, size):
        self.window = tk.Tk()
        self.window.title(title)
        self.set_theme()
        self.set_window_size(size)
        self.center_window()

    def set_theme(self):
        theme = ThemedStyle(self.window)
        theme.set_theme("radiance")

    def set_window_size(self, size):
        self.window.geometry(size)
        self.window.resizable(False, False)

    def center_window(self):
        self.window.update_idletasks()
        width = self.window.winfo_width()
        frm_width = self.window.winfo_rootx() - self.window.winfo_x()
        win_width = width + 2 * frm_width
        height = self.window.winfo_height()
        titlebar_height = self.window.winfo_rooty() - self.window.winfo_y()
        win_height = height + titlebar_height + frm_width
        x = self.window.winfo_screenwidth() // 2 - win_width // 2
        y = self.window.winfo_screenheight() // 2 - win_height // 2
        self.window.geometry(f'{width}x{height}+{x}+{y}')
        self.window.deiconify()

    def listbox_config(self, listbox):
        listbox['selectbackground'] = ""

    def combobox_config(self, combobox):
        combobox.option_add("*TCombobox*Listbox*selectBackground", "")

    def window_freeze(self, window, window2=None):
        if not window2:
            window.wm_attributes("-disabled", True)
        else:
            window.wm_attributes("-disabled", False)
            window2.destroy()

    def max_length(self, text, limit, password_and_login=False):
        if len(text) > int(limit) or (not password_and_login and not text.isalpha()):
            return False
        return True

    def state_active(self, btn, btn1=None):
        btn["state"] = "normal"
        if btn1:
            btn["bg"] = "white"

    def state_disabled(self, btn=None, btn1=None):
        if not btn and btn1:
            btn1["state"] = "readonly"
        elif btn and btn1:
            btn["bg"] = ""
            btn["state"] = "disabled"
        else:
            btn["state"] = "disabled"
