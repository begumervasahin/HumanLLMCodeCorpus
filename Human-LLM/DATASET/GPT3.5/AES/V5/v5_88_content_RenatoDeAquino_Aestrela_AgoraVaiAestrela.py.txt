# Define the adjacency lists for each node
nodes = {
    'a': [2, 'B', 36, 'C', 61],
    'b': [2, 'A', 36, 'D', 31],
    'c': [4, 'A', 61, 'D', 32, 'F', 31, 'L', 80],
    'd': [3, 'B', 31, 'C', 32, 'E', 52],
    'e': [2, 'D', 52, 'G', 43],
    'f': [3, 'C', 31, 'K', 112, 'J', 122],
    'g': [2, 'E', 43, 'H', 40],
    'h': [2, 'G', 40, 'I', 40],
    'i': [2, 'H', 40, 'J', 45],
    'j': [3, 'I', 45, 'F', 122, 'K', 36],
    'k': [3, 'J', 36, 'F', 112, 'M', 32],
    'l': [2, 'C', 80, 'M', 102]
}

# Define the distances between nodes
distances = {
    'A': 223, 'B': 222, 'C': 166, 'D': 192,
    'E': 165, 'F': 136, 'G': 122, 'H': 111,
    'I': 100, 'J': 60, 'K': 32, 'L': 102
}

# Initialize variables
current_node = None
path = []

# Input starting node
current_node = input("Enter the starting node: ").lower()

# Add starting node to the path
path.append(current_node)

# Loop until the final node 'M' is reached
while current_node != 'm':
    adjacent_nodes = nodes[current_node]
    min_cost_node = None
    min_cost = float('inf')
    
    # Find adjacent node with minimum cost
    for i in range(1, len(adjacent_nodes), 2):
        node = adjacent_nodes[i]
        cost = distances[node.upper()] + adjacent_nodes[i - 1]
        if cost < min_cost:
            min_cost = cost
            min_cost_node = node
    
    # Update current node and add it to the path
    current_node = min_cost_node.lower()
    path.append(current_node)

# Print the path
print("Shortest path to node 'M':", path)
