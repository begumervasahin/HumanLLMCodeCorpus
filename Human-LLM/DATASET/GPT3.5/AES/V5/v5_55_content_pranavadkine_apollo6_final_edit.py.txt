import cv2
import math
import os
import random
import struct
from Crypto.Cipher import AES

def read_image(image_path):
    """Reads an image and returns its dimensions."""
    img = cv2.imread(image_path)
    height, width, channels = img.shape
    print(f"Image dimensions: Height={height}, Width={width}")
    return img, height, width

def slice_image(img, slices):
    """Slices the image into multiple parts."""
    height, width, _ = img.shape
    width_of_slice = math.floor(width / slices)
    sliced_images = []
    for x in range(1, slices):
        start = (x - 1) * width_of_slice + 1
        end = x * width_of_slice
        cropped = img[0:height, start:end]
        sliced_images.append(cropped)
        cv2.imwrite(f'd{x}.jpg', cropped)
    cropped = img[0:height, (slices - 1) * width_of_slice:width]
    sliced_images.append(cropped)
    cv2.imwrite(f'd{slices}.jpg', cropped)
    return sliced_images

def encrypt_image(key, image_path):
    """Encrypts an image using AES encryption."""
    iv = ''.join(chr(random.randint(0, 0xFF)) for i in range(16))
    encryptor = AES.new(key, AES.MODE_CBC, iv)
    chunksize = 64 * 1024
    filesize = os.path.getsize(image_path)
    with open(image_path, 'rb') as infile:
        with open(image_path + '.enc', 'wb') as outfile:
            outfile.write(struct.pack('<Q', filesize))
            outfile.write(iv.encode())
            while True:
                chunk = infile.read(chunksize)
                if len(chunk) == 0:
                    break
                elif len(chunk) % 16 != 0:
                    chunk += b' ' * (16 - len(chunk) % 16)
                outfile.write(encryptor.encrypt(chunk))

def move_encrypted_images(slices):
    """Moves encrypted image files to server directories."""
    for x in range(1, slices + 1):
        os.rename(f'd{x}.jpg.enc', f'server{x}/d{x}.jpg.enc')

        iv = infile.read(16)
