import sys

# Encrypted text and key
text = [
    132, 201, 141, 74, 140, 94, 141, 140, 141, 15, 31, 164, 90, 229, 201, 141,
    78, 114, 241, 217, 141, 217, 140, 180, 141, 164, 51, 141, 188, 221, 31, 164,
    241, 177, 141, 140, 51, 217, 141, 201, 229, 152, 141, 78, 241, 114, 78, 102,
    94, 141, 74, 152, 31, 152, 141, 94, 201, 31, 164, 102, 164, 51, 90, 141, 201,
    229, 164, 31, 201, 152, 152, 51, 115
]
key = 84

# Function to decrypt the text
def decrypt(text, key):
    # Convert text to bytes and split into nibbles
    crypt_bytes = bytearray(text)
    crypt_bytes_as_nibbles = [byte_to_nibbles(byte) for byte in crypt_bytes]
    
    # Convert key to nibbles
    key_as_nibbles = byte_to_nibbles(key)
    
    # Decrypt each byte
    decrypt_bytes_as_nibbles = [decrypt_byte_as_nibbles(nibbles, key_as_nibbles) for nibbles in crypt_bytes_as_nibbles]
    
    # Convert nibbles back to bytes
    decrypt_bytes = [convert_nibbles_to_byte(nibbles) for nibbles in decrypt_bytes_as_nibbles]
    
    # Convert bytes to string
    decrypt_string = convert_bytes_to_string(decrypt_bytes)

    return decrypt_string

# Convert byte to nibbles
def byte_to_nibbles(a_byte):
    nibble4 = a_byte & 0b00000011
    nibble3 = (a_byte >> 2) & 0b00000011
    nibble2 = (a_byte >> 4) & 0b00000011
    nibble1 = (a_byte >> 6) & 0b00000011
    return bytearray([nibble1, nibble2, nibble3, nibble4])

# Convert nibbles to byte
def convert_nibbles_to_byte(nibbles):
    a_byte = (
        (nibbles[0] << 6) |
        (nibbles[1] << 4) |
        (nibbles[2] << 2) |
        nibbles[3]
    )
    return a_byte

# Decrypt byte as nibbles
def decrypt_byte_as_nibbles(nibble_quadruple, key_quadruple):
    ans = xor(nibble_quadruple, key_quadruple)
    ans = swap_last_two(ans)
    ans = s_box(ans)
    ans = xor(ans, key_quadruple)
    ans = subtract_last_two_first_two(ans)
    ans = swap_last_two(ans)
    ans = s_box(ans)
    ans = xor(ans, key_quadruple)
    return ans

# XOR operation
def xor(nibbles, key):
    return [nib ^ k for nib, k in zip(nibbles, key)]

# Swap last two bits
def swap_last_two(nibbles):
    nibbles[-2], nibbles[-1] = nibbles[-1], nibbles[-2]
    return nibbles

# S-box substitution
def s_box(nibbles):
    sbox = {0: 2, 1: 0, 2: 3, 3: 1}
    return [sbox[nib] for nib in nibbles]

# Subtract last two bits from first two bits
def subtract_last_two_first_two(nibbles):
    nibbles[0] = (nibbles[0] - nibbles[2]) % 4
    nibbles[1] = (nibbles[1] - nibbles[3]) % 4
    return nibbles

# Convert bytes to string
def convert_bytes_to_string(byte_list):
    return ''.join(chr(byte) for byte in byte_list)
