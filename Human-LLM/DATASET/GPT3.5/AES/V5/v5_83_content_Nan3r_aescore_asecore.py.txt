import base64
import zlib
import time
import hashlib
from Crypto.Cipher import AES

class AESCipher:
    def __init__(self, IV=None, Key=None):
        self.block_size = 16
        self.IV = IV if IV else self.generate_IV()
        self.Key = Key if Key else hashlib.md5(str(self.IV).encode()).hexdigest()  

    def generate_IV(self):
        """
        Generate a random initialization vector (IV).
        """
        return hashlib.md5(str(time.time()).encode()).hexdigest()[:self.block_size]

    def encrypt(self, plaintext):
        """
        Encrypts plaintext using AES in CBC mode with zlib compression.

        Args:
            plaintext (str): The plaintext to be encrypted.

        Returns:
            str: The encrypted ciphertext.
        """
        padded_plaintext = self.pad(plaintext)
        cipher = AES.new(self.Key.encode(), AES.MODE_CBC, self.IV.encode())        
        encrypted_data = cipher.encrypt(padded_plaintext)
        compressed_data = zlib.compress(self.IV.encode() + encrypted_data)
        return base64.b64encode(compressed_data)

    def decrypt(self, ciphertext):
        """
        Decrypts ciphertext using AES in CBC mode with zlib compression.

        Args:
            ciphertext (str): The ciphertext to be decrypted.

        Returns:
            str: The decrypted plaintext.
        """
        decoded_data = base64.b64decode(ciphertext)
        IV = zlib.decompress(decoded_data)[:self.block_size].decode()
        encrypted_data = zlib.decompress(decoded_data)[self.block_size:].decode()  
        Key = hashlib.md5(IV.encode()).hexdigest()
        cipher = AES.new(Key.encode(), AES.MODE_CBC, IV.encode())
        decrypted_data = cipher.decrypt(encrypted_data)
        return self.unpad(decrypted_data).decode()

    def pad(self, s):
        """
        Apply PKCS7 padding to the input string.

        Args:
            s (str): The input string.

        Returns:
            str: The padded string.
        """
        padding_length = self.block_size - len(s) % self.block_size
        return s + padding_length * chr(padding_length)

    def unpad(self, s):
        """
        Remove PKCS7 padding from the input string.

        Args:
            s (str): The padded string.

        Returns:
            str: The unpadded string.
        """
        padding_length = ord(s[-1])
        return s[:-padding_length]
