from constants import Nr
from encrypt_functions import (
    get_key, 
    get_pt_data, 
    out_ct_data, 
    print_hex,
    key_expansion, 
    add_round_key, 
    sub_bytes, 
    shift_rows, 
    mix_columns
)

def main():
    # Read the encryption key and perform key expansion
    key = get_key()
    expanded_key = key_expansion(key)

    # Read the plain text data
    plain_text = get_pt_data()
    print('Main: Finished retrieving plain text data. State:')
    print_hex(plain_text)

    # Initial round
    current_round = 0
    print('Main: Round =', current_round)
    add_round_key(plain_text, expanded_key, current_round)

    # Perform AES encryption rounds
    for current_round in range(1, Nr):
        print('Main: Round =', current_round)
        sub_bytes(plain_text)
        shift_rows(plain_text)
        mix_columns(plain_text)
        add_round_key(plain_text, expanded_key, current_round)

    # Final round
    current_round = Nr - 1
    print('Main: Round =', current_round)
    sub_bytes(plain_text)
    shift_rows(plain_text)
    add_round_key(plain_text, expanded_key, current_round)

    # Write encrypted data to CTData.txt
    out_ct_data(plain_text)
    print('Encrypted data has been written to CTData.txt')

if __name__ == "__main__":
    main()
