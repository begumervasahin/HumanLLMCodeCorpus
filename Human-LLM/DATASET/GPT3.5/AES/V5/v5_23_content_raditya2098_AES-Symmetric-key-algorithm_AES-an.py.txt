# AES S-box and Rcon (round constant) definitions
Sbox = [
    0x63, 0x7C, 0x77, 0x7B, 0xF2, 0x6B, 0x6F, 0xC5, 0x30, 0x01, 0x67, 0x2B, 0xFE, 0xD7, 0xAB, 0x76,
    # Add the rest of the S-box values here...
]

Rcon = [0x8d, 0x01, 0x02, 0x04, 0x08, 0x10, 0x20, 0x40, 0x80, 0x1b, 0x36, 0x6c, 0xd8, 0xab, 0x4d, 0x9a]

# Utility function to format a hex string (remove the '0x' prefix)
def format_hex(hex_str):
    return hex_str[2:]

# Substitutes characters of a message using the AES S-box
def sbox_substitute(message):
    substituted = []
    for char in message:
        # Convert character to S-box value and format as hex without '0x' prefix
        sbox_value = Sbox[ord(char)]
        substituted.append(format_hex(hex(sbox_value)))
    return substituted

def main():
    key = 'Thats my Kung Fu'
    print(f"Key length: {len(key.encode('utf-8'))} bytes")

    msg = 'Two One Nine Two'
    print(f"Message: {msg}")
    print(f"Message length: {len(msg.encode('utf-8'))} bytes")

    # Convert the message using the S-box and display the result
    substituted_msg = sbox_substitute(msg)
    print("S-box substituted values:", substituted_msg)

    # Dividing the substituted message into 4-character blocks for demonstration
    blocks = [substituted_msg[i:i+4] for i in range(0, len(substituted_msg), 4)]
    for i, block in enumerate(blocks):
        print(f"Block {i+1}: {block}")

if __name__ == "__main__":
    main()
