import numpy as np

class EstimatePerceptualMixingTime:
    def __init__(self, room_dimensions):
        self.room_dimensions = room_dimensions
        self.mixing_time_estimate = {}

    def model_based(self):
        print('----------------------------------')
        print('MODEL-BASED MIXING TIME PREDICTION')
        print('----------------------------------')
        print('ROOM PROPERTIES:')

        # Unpack room dimensions
        height, length, width = self.room_dimensions

        # Check if dimensions are valid
        if any(dim <= 0 for dim in self.room_dimensions):
            raise ValueError('Room dimensions must be positive numeric values only.')

        print(f'Room dimensions: height = {height}m, length = {length}m, width = {width}m')

        # Calculate volume and surface area
        volume = height * length * width
        surface_area = 2 * (height * length + height * width + length * width)

        print(f'Volume: {volume}m3')
        print(f'Surface area: {surface_area}m2')

        # Calculate mixing time estimates
        root_volume = np.sqrt(volume)
        mean_free_path = 47 * volume / surface_area
        tmp50 = 20.08 * volume / surface_area + 12
        tmp95 = 0.0117 * volume + 50.1

        print(f'tmp50: {tmp50}ms')
        print(f'tmp95: {tmp95}ms')

        # Store estimates in dictionary
        self.mixing_time_estimate['model'] = {
            'tmp50': tmp50,
            'tmp95': tmp95,
            'root_volume': root_volume,
            'mean_free_path': mean_free_path
        }

        return self
