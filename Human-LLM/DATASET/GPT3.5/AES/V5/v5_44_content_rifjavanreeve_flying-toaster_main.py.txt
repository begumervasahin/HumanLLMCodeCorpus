from flying_toaster.toaster import Toaster
from flying_toaster.super_toaster import SuperToaster

def print_toaster_info(toaster):
    """
    Print information about the provided toaster.
    """
    print('Model:', type(toaster).__name__)
    if isinstance(toaster, SuperToaster):
        print('Alias:', toaster.alias)
    print('Colour:', toaster.colour)
    print('Number of slots:', len(toaster.slots))

def get_toaster_details(toaster_type):
    """
    Prompt the user to input details of the toaster.
    """
    try:
        slots = int(input('\nHow many slots does your toaster have?: '))
        colour = input('\nWhat colour is your toaster?: ')

        if toaster_type == '1':
            toaster = Toaster(slots, colour)
        elif toaster_type == '2':
            alias = input('\nWhat is the alias of your toaster?: ')
            toaster = SuperToaster(slots, alias, colour)

        print('\nHere is your wonderful toaster!\n')
        print_toaster_info(toaster)

        toasts = int(input('\nHow many toasts would you like?: '))
        bread = input('\nWhich bread would you like to toast?: ')

        toaster.add_toasts(toasts, bread)
        time = int(input('\nHow many seconds would you like to toast for?: '))
        toaster.set_timer(time)
        toaster.toast()

    except Exception as e:
        print(e)

def main():
    """
    Main function to interactively choose and use a toaster.
    """
    print('What type of toaster would you like?\n')
    print('1. Toaster')
    print('2. Super Toaster')

    toaster_type = input('\nChoose the type of toaster: ')

    if toaster_type in ['1', '2']:
        get_toaster_details(toaster_type)
    else:
        print('\nInvalid input! Please choose either 1 or 2.')

    end_loop = input('\nDo you want to exit the program? [y/n]: ')

    if end_loop.lower() == 'n':
        main()

if __name__ == '__main__':
    main()
