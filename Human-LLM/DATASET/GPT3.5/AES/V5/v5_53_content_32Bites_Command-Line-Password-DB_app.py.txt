import json
import base64
import os
from cryptography.fernet import Fernet
from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC
from terminaltables import AsciiTable

# Define the format for the vault
vault_format = {"logins": []}

# Function to create a new vault
def create_vault():
    name_of_vault = input("Enter the name of the vault: ")
    vault_password = input("Enter the password for the vault: ").encode()

    with open(f"vaults/{name_of_vault}.vault", "wb") as vault_file:
        vault_contents = json.dumps(vault_format).encode()
        salt = b'vault_OOFYDOOFY'
        kdf = PBKDF2HMAC(algorithm=hashes.SHA256(), length=32, salt=salt, iterations=100000, backend=default_backend())
        key = base64.urlsafe_b64encode(kdf.derive(vault_password))
        encrypter = Fernet(key)
        encrypted_data = encrypter.encrypt(vault_contents)
        vault_file.write(encrypted_data)

# Function to print the contents of a vault
def print_vault_contents(password=None, name=None):
    if not password and not name:
        name_of_vault = input("Enter the name of the vault: ")
        vault_password = input("Enter the password for the vault: ").encode()
    else:
        name_of_vault = name
        vault_password = password

    with open(f"vaults/{name_of_vault}.vault", "rb") as vault_file:
        vault_contents = vault_file.read()
        salt = b'vault_OOFYDOOFY'
        kdf = PBKDF2HMAC(algorithm=hashes.SHA256(), length=32, salt=salt, iterations=100000, backend=default_backend())
        key = base64.urlsafe_b64encode(kdf.derive(vault_password))
        decrypter = Fernet(key)
        decrypted_data = decrypter.decrypt(vault_contents).decode()
        json_data = json.loads(decrypted_data)

        table_data = [["ID", "Website URL", "Username", "Password"]]

        for i, data in enumerate(json_data['logins']):
            table_data.append([str(i + 1), data['website_url'], data['username'], data['password']])

        table = AsciiTable(table_data)
        print(table.table)

# Function to add login details to a vault
def add_login_to_vault():
    name_of_vault = input("Enter the name of the vault: ")
    vault_password = input("Enter the password for the vault: ").encode()

    with open(f"vaults/{name_of_vault}.vault", "rb+") as vault_file:
        vault_contents = vault_file.read()
        salt = b'vault_OOFYDOOFY'
        kdf = PBKDF2HMAC(algorithm=hashes.SHA256(), length=32, salt=salt, iterations=100000, backend=default_backend())
        key = base64.urlsafe_b64encode(kdf.derive(vault_password))
        decrypter = Fernet(key)
        decrypted_data = decrypter.decrypt(vault_contents).decode()
        json_data = json.loads(decrypted_data)

        website_url = input("Enter the website URL: ")
        username = input("Enter the username: ")
        password = input("Enter the password: ")

        json_data['logins'].append({"website_url": website_url, "username": username, "password": password})

        encrypted_data = decrypter.encrypt(json.dumps(json_data).encode())

        vault_file.seek(0)
        vault_file.truncate()
        vault_file.write(encrypted_data)

    print_vault_contents(vault_password, name_of_vault)

# Main menu function
def main_menu():
    print("Welcome to Password Vault!")
    print("What would you like to do?")
    print("[1] Create a vault")
    print("[2] Decrypt a vault and view logins")
    print("[3] Add a login")
    print("[4] Exit")

    user_input = input("Enter your choice: ")

    if user_input == '1':
        create_vault()
    elif user_input == '2':
        print_vault_contents()
    elif user_input == '3':
        add_login_to_vault()
    elif user_input == '4':
        exit()
    else:
        print("Unknown option")

# Entry point of the program
if __name__ == "__main__":
    while True:
        main_menu()
