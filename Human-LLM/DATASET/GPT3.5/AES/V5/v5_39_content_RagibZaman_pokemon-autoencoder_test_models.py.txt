import unittest
import torch
from torch.autograd import Variable
import models

class TestEncoder(unittest.TestCase):
    def test_output_size(self):
        """Test the output size of the Encoder."""
        batch_size = 32
        image_size = 64
        latent_size = 128
        encoder = models.Encoder(latent_size, image_size)       
        input_tensor = Variable(2.0 * (torch.rand((batch_size, 3, image_size, image_size)) - 0.5))
        output_tensor = encoder(input_tensor)
        self.assertEqual(output_tensor.size(), torch.Size([batch_size, latent_size]))

    def test_layer_output_size(self):
        """Test the output size of each layer in the Encoder."""
        batch_size = 32
        image_size = 64
        latent_size = 128
        encoder = models.Encoder(latent_size, image_size)       
        input_tensor = Variable(2.0 * (torch.rand((batch_size, 3, image_size, image_size)) - 0.5))
        input_tensor = input_tensor.view(input_tensor.size(0), -1)
        hidden_output = encoder.hidden_layer(input_tensor)      
        self.assertEqual(hidden_output.size(), torch.Size([batch_size, latent_size]))
        output_tensor = encoder.output_layer(hidden_output)     
        self.assertEqual(output_tensor.size(), torch.Size([batch_size, latent_size]))

class TestDecoder(unittest.TestCase):
    def test_output_size(self):
        """Test the output size of the Decoder."""
        batch_size = 32
        image_size = 64
        latent_size = 128
        decoder = models.Decoder(latent_size, image_size)       
        input_tensor = Variable(2.0 * (torch.rand((batch_size, latent_size, 1, 1)) - 0.5))
        output_tensor = decoder(input_tensor)
        self.assertEqual(output_tensor.size(), torch.Size([batch_size, 3, image_size, image_size]))

    def test_layer_output_size(self):
        """Test the output size of each layer in the Decoder."""
        batch_size = 32
        image_size = 64
        latent_size = 128
        decoder = models.Decoder(latent_size, image_size)       
        input_tensor = Variable(2.0 * (torch.rand((batch_size, latent_size, 1, 1)) - 0.5))
        tensor = decoder.layer1(input_tensor)
        self.assertEqual(tensor.size(), torch.Size([batch_size, 128, 7, 7]))
        tensor = decoder.layer2(tensor)
        self.assertEqual(tensor.size(), torch.Size([batch_size, 64, 15, 15]))
        tensor = decoder.layer3(tensor)
        self.assertEqual(tensor.size(), torch.Size([batch_size, 64, 31, 31]))
        tensor = decoder.layer4(tensor)
        self.assertEqual(tensor.size(), torch.Size([batch_size, 32, 63, 63]))
        tensor = decoder.layer5(tensor)
        self.assertEqual(tensor.size(), torch.Size([batch_size, 3, image_size, image_size]))

if __name__ == '__main__':
    unittest.main()
