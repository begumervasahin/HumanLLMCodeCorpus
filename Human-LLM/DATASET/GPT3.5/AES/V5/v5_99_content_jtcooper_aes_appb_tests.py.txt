def multiply_by_x(byte):
    """Multiplies by x (0x02) in GF(2^8)."""
    return ((byte << 1) & 0xFF) ^ (0x1B if byte & 0x80 else 0)

def finite_field_multiplication(a, b):
    """Finite field multiplication of two bytes."""
    product = 0
    for _ in range(8):
        if b & 1:
            product ^= a
        high_bit_set = a & 0x80
        a = ((a << 1) & 0xFF) ^ (0x1B if high_bit_set else 0)
        b >>= 1
    return product
