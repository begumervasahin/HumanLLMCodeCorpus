from HopfClass import *
import numpy as np
from itertools import product
import scipy.sparse as sps

def taft_element_names(n, g, x):
    """Generate element names for the Taft algebra."""
    out = []
    for i in range(n):
        for j in range(n):
            out.append(f"{g}^{j}{x}^{i}")
    return out

def taft_mult(n):
    """Generate the multiplication matrix for the Taft algebra."""
    poly = [0] * (n + 1)
    poly[0] = 1
    poly[n] = -1
    omega = np.roots(poly)
    out = np.zeros((n**2, n**4), dtype=complex)
    N = range(n)
    for i, j, k in product(N, N, N):
        for l in range(n - j):
            out[((i + k) % n) + n * (j + l), i + n * j + (k + n * l) * n**2] = omega[(n - 1) - (-j * k) % n] 
    return sps.csr_matrix(out, dtype=complex)

def taft_comult(n, mult):
    """Generate the comultiplication matrix for the Taft algebra."""
    dim = n ** 2
    out = np.zeros((dim**2, dim), dtype=complex)
    x_plus_g = np.zeros(dim)
    x_plus_g[1] = 1
    x_plus_g[n] = 1
    temp = np.zeros(dim)
    temp[0] = 1
    prods = [temp]
    N = range(1, n)
    for i in N:
        temp2 = np.kron(temp, x_plus_g)
        temp = mult.dot(temp2)
        prods.append(temp)
    for i, j in product(N, N):
        for l in range(j + 1):
            out[((i + l) % n) + (j - l) * n + (i + n * l) * dim, i + j * n] = prods[j][l + (j - l) * n]    
    return out

def taft_counit(n):
    """Generate the counit vector for the Taft algebra."""
    return np.ones(n, dtype=complex)

def taft_antipode(n):
    """Generate the antipode matrix for the Taft algebra."""
    dim = n ** 2
    poly = [0] * (n + 1)
    poly[0] = 1
    poly[n] = -1
    omega = np.roots(poly)
    out = np.zeros((dim, dim), dtype=complex)
    N = range(n)
    for i, j in product(N, N):
        out[((-i) % n) + j * n, i + j * n] = (-1) ** j * omega[(i * j) % n]
    return sps.csr_matrix(out, dtype=complex)

# Example usage
n = 3
g = 'g'
x = 'x'

element_names = taft_element_names(n, g, x)
mult_matrix = taft_mult(n)
comult_matrix = taft_comult(n, mult_matrix)
counit_vector = taft_counit(n)
antipode_matrix = taft_antipode(n)

# Create Hopf algebra
taft_algebra = HopfAlgebra("TaftAlgebra", element_names, mult_matrix, comult_matrix, counit_vector, antipode_matrix)
