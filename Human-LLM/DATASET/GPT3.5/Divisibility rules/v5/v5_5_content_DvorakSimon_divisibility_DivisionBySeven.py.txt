# Initialize the number to check
number_to_check = 1946

# Convert the number to a string for manipulation
number_str = str(number_to_check)

# Continue looping while the length of the number is not 2 and it satisfies other conditions
while len(number_str) != 2 and number_to_check >= 0 and number_to_check != 0:
    # Print the current number
    print(number_str)

    # Get the prefix of the number
    prefix = number_str[:-1]

    # Get the last digit of the number
    last_digit = int(number_str[-1])

    # Update the number by subtracting twice the last digit from the prefix
    number_str = str(int(prefix) - (2 * last_digit))

    # Update the integer representation of the number
    number_to_check = int(number_str)

# Check if the resulting number is divisible by 7
if number_to_check % 7 == 0:
    # If divisible by 7, print a message
    print("Number " + number_str + " is divisible by 7!")
