def get_input():
    # Get input for the first list from the user
    list1 = [n.strip() for n in input("Enter the first list (separate elements with commas): ").split(",")]
    print("List1:", list1)
    print("List1 is", len(list1), "characters long")

    # Get input for the second list from the user
    list2 = [n.strip() for n in input("Enter the second list (separate elements with commas): ").split(",")]
    print("List2:", list2)
    print("List2 is", len(list2), "characters long")

    # Calculate and print Fizzbuzz result
    print(Fizzbuzz(list1, list2))

def Fizzbuzz(a, b):
    # Calculate the total length of both lists
    totallength = len(a) + len(b)

    # Check if the total length is divisible by 3 and 5
    if totallength % 3 == 0 and totallength % 5 == 0:
        return 'FizzBuzz'
    # Check if the total length is divisible by 3
    elif totallength % 3 == 0:
        return 'Fizz'
    # Check if the total length is divisible by 5
    elif totallength % 5 == 0:
        return 'Buzz'
    # If none of the conditions are met, return the total length
    else:
        return totallength

if __name__ == "__main__":
    # Call the get_input function when the script is run
    get_input()
