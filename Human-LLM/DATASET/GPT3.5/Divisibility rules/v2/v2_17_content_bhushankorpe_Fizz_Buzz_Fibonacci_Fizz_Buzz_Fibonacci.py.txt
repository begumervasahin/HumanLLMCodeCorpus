def check_prime(number):
    """
    Check if a number is prime.

    Args:
        number (int): The number to check.

    Returns:
        bool: True if the number is prime, False otherwise.
    """
    if number <= 1:
        return False
    for divisor in range(2, number):
        if number % divisor == 0:
            return False
    return True

def perform_operations(num):
    """
    Perform operations based on the input number.

    Args:
        num (int): The input number.
    """
    result = []

    if num % 3 == 0 or num % 5 == 0 or num % 15 == 0 or check_prime(num):
        if num % 3 == 0:
            result.append("Buzz")
        if num % 5 == 0:
            result.append("Fizz")
        if num % 15 == 0:
            result.append("FizzBuzz")
        if check_prime(num):
            result.append("BuzzFizz")

        for item in result:
            print(item)
    else:
        fibonacci_sequence = []
        a, b = 0, 1

        for i in range(num):
            if i == 0:
                fibonacci_sequence.append(a)
            elif i == 1:
                fibonacci_sequence.append(b)
            else:
                c = a + b
                fibonacci_sequence.append(c)
                a, b = b, c

        print(fibonacci_sequence)

if __name__ == '__main__':
    print("How many Fibonacci numbers do you want to see?")
    n = int(input())
    perform_operations(n)
