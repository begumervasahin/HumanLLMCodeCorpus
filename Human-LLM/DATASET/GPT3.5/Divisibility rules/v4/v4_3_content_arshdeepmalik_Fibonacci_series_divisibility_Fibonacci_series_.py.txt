import random

def fibonacci(n):
    """Calculate the nth Fibonacci number."""
    if n <= 1:
        return n
    else:
        return fibonacci(n - 1) + fibonacci(n - 2)

def fizzbuzz(n):
    """Apply the FizzBuzz rules to a number."""
    result = ""
    if is_prime(n):
        return "BuzzFizz"
    if n % 3 == 0:
        result += "Fizz"
    if n % 5 == 0:
        result += "Buzz"
    return result if result else n

def is_prime(n):
    """Check if a number is prime."""
    if n < 2:
        return False
    if n > 2 and n % 2 == 0:
        return False
    for i in range(3, int(n ** 0.5) + 1, 2):
        if n % i == 0:
            return False
    return True

    n = 15
