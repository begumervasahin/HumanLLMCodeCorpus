# Get user input for the maximum multiple
max_multiple = int(input("What is the max multiple? "))

# Initialize variables
found = False
smallest = 0
current = max_multiple

# Loop until the smallest evenly divisible number is found
while not found:
    div_count = 0
    # Check divisibility for each number from 1 to max_multiple
    for i in range(1, max_multiple + 1):
        if current % i == 0:
            div_count += 1

    # If the current number is divisible by all numbers from 1 to max_multiple
    if div_count == max_multiple:
        found = True
        smallest = current
    else:
        # Increment current by max_multiple to check the next potential smallest number
        current += max_multiple

# Print the smallest number that is evenly divisible by all numbers from 1 to max_multiple
print(f"The smallest number that is evenly divisible by all of the numbers from 1 to {max_multiple} is {smallest}")
