def load(filename="input.txt"):
    # Open the input file and read its contents
    with open(filename, 'r') as file:
        # Read lines from the file and extract the values for fizz, buzz, and end
        fizz, buzz, end = map(int, file.readline().split())

    # Iterate through numbers from 1 to end
    for i in range(1, end + 1):
        # Check for divisibility by fizz and buzz
        if i % fizz == 0 and i % buzz == 0:
            print("FB")
        # Check for divisibility by fizz only
        elif i % fizz == 0:
            print("F")
        # Check for divisibility by buzz only
        elif i % buzz == 0:
            print("B")
        # If not divisible by either, print the number itself
        else:
            print(i)

# Load and process the input file, assuming 'input.txt' exists in the same directory    
load()
