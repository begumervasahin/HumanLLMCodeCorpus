def count_odd_even_bits(n):
    odd_counter = even_counter = 0
    
    # Counting odd and even bits in the binary representation of n
    while n:
        # Check if the least significant bit (LSB) is 1 (odd)
        if n & 1 == 1:
            odd_counter += 1
        n = n >> 1  # Right shift n by 1 to move to the next bit

        # Check if the LSB is 1 (even)
        if n & 1 == 1:
            even_counter += 1
        n = n >> 1  # Right shift n by 1 to move to the next bit

    return odd_counter, even_counter

def is_divisible_by_three(n):
    # Base cases: 0 and 1
    if n == 0:
        return True
    elif n == 1:
        return False
    
    odd_count, even_count = count_odd_even_bits(abs(n))
    difference = abs(odd_count - even_count)
    
    # Recursively check if the absolute difference between odd and even bit counts is divisible by 3
    return is_divisible_by_three(difference)

if __name__ == "__main__":
    # Prompt the user to input an integer
    print("Enter an integer:")
    n = int(input())

    # Check if the absolute value of the input integer is divisible by 3
    if is_divisible_by_three(abs(n)):
        print("%d is divisible by 3." % n)
    else:
        print("%d is not divisible by 3." % n)
