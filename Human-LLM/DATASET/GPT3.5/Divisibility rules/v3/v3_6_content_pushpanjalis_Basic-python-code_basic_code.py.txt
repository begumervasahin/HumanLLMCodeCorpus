def check_divisibility(number):
    results = ""

    # Check divisibility by 2, 3, 5, and 7
    if number % 2 == 0:
        results += "The number is divisible by 2.\n"

    if number % 3 == 0:
        results += "The number is divisible by 3.\n"

    if number % 5 == 0:
        results += "The number is divisible by 5.\n"

    if number % 7 == 0:
        results += "The number is divisible by 7.\n"

    # If the number is not divisible by any of the specified divisors, add a message
    if not results:
        results += "The number is not divisible by 2, 3, 5, or 7.\n"

    return results

def get_user_input():
    try:
        number = int(input("Please enter a number of your choice:\n"))
        return number
    except ValueError:
        print("Invalid input. Please enter a valid integer.")
        return get_user_input()

def main():
    number = get_user_input()
    results = check_divisibility(number)
    print(results)

if __name__ == "__main__":
    main()
