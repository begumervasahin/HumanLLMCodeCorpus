import random
import time

# Function to calculate the percentage difference between two numbers
def calculate_percentage_difference(num1, num2):
    """
    Calculates the percentage difference between two numbers.

    Args:
    - num1: First number.
    - num2: Second number.

    Returns:
    - Percentage difference between num1 and num2.
    """
    return (abs(num1 - num2) / max(num1, num2)) * 100

# Function to print an array
def print_array(arr, name):
    """
    Prints an array with a label.

    Args:
    - arr: The array to print.
    - name: The label for the array.
    """
    print(f"Sorted array from {name} Sort:")
    print(*arr)

# Function to measure the time taken by a sorting algorithm
def measure_sorting_time(sorting_function, array):
    """
    Measures the time taken by a sorting algorithm.

    Args:
    - sorting_function: The sorting function to test.
    - array: The array to be sorted.

    Returns:
    - The elapsed time for sorting.
    """
    start_time = time.time()
    sorting_function(array)
    elapsed_time = time.time() - start_time
    print(f"Function [{sorting_function.__name__}] finished in {elapsed_time:.3f} seconds")
    return elapsed_time

# Sorting algorithms
def bubble_sort(arr):
    """
    Implements the Bubble Sort algorithm.
    """
    n = len(arr)
    for i in range(n):
        for j in range(0, n - i - 1):
            if arr[j] > arr[j + 1]:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]

def insertion_sort(arr):
    """
    Implements the Insertion Sort algorithm.
    """
    for i in range(1, len(arr)):
        key = arr[i]
        j = i - 1
        while j >= 0 and key < arr[j]:
            arr[j + 1] = arr[j]
            j -= 1
        arr[j + 1] = key

def selection_sort(arr):
    """
    Implements the Selection Sort algorithm.
    """
    for i in range(len(arr)):
        min_index = i
        for j in range(i + 1, len(arr)):
            if arr[min_index] > arr[j]:
                min_index = j
        arr[i], arr[min_index] = arr[min_index], arr[i]

def quick_sort(arr):
    """
    Implements the Quick Sort algorithm.
    """
    if len(arr) <= 1:
        return arr
    pivot = arr[len(arr) // 2]
    left = [x for x in arr if x < pivot]
    middle = [x for x in arr if x == pivot]
    right = [x for x in arr if x > pivot]
    return quick_sort(left) + middle + quick_sort(right)

def merge_sort(arr):
    """
    Implements the Merge Sort algorithm.
    """
    if len(arr) <= 1:
        return arr
    mid = len(arr) // 2
    left, right = merge_sort(arr[:mid]), merge_sort(arr[mid:])
    return merge(left, right)

def merge(left, right):
    """
    Merges two arrays in sorted order.

    Args:
    - left: The left array.
    - right: The right array.

    Returns:
    - The merged and sorted array.
    """
    result = []
    i = j = 0
    while i < len(left) and j < len(right):
        if left[i] < right[j]:
            result.append(left[i])
            i += 1
        else:
            result.append(right[j])
            j += 1
    result.extend(left[i:])
    result.extend(right[j:])
    return result

def heap_sort(arr):
    """
    Implements the Heap Sort algorithm.
    """
    n = len(arr)
    for i in range(n, -1, -1):
        heapify(arr, n, i)
    for i in range(n - 1, 0, -1):
        arr[i], arr[0] = arr[0], arr[i]
        heapify(arr, i, 0)

def heapify(arr, n, i):
    """
    Helper function for heap sort.

    Args:
    - arr: The array to heapify.
    - n: Size of the heap.
    - i: Index of the root.
    """
    largest = i
    left = 2 * i + 1
    right = 2 * i + 2
    if left < n and arr[i] < arr[left]:
        largest = left
    if right < n and arr[largest] < arr[right]:
        largest = right
    if largest != i:
        arr[i], arr[largest] = arr[largest], arr[i]
        heapify(arr, n, largest)
