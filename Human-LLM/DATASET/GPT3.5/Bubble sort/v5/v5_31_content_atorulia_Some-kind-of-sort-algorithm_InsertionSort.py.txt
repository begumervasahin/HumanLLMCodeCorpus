import random

def generate_random_list(size: int) -> list[int]:
    """
    Generates a random list of integers.

    Args:
        size (int): The size of the list to generate.

    Returns:
        list[int]: The randomly generated list.
    """
    return [random.randrange(100) for _ in range(size)]

def check_sorted(array: list[int]) -> bool:
    """
    Checks if an array is sorted in ascending order.

    Args:
        array (list): The array to be checked.

    Returns:
        bool: True if the array is sorted, False otherwise.
    """
    return all(array[i] <= array[i + 1] for i in range(len(array) - 1))

def insertion_sort(array: list[int]) -> list[int]:
    """
    Sorts an array in ascending order using the insertion sort algorithm.

    Args:
        array (list): The array to be sorted.

    Returns:
        list: The sorted array.
    """
    for i in range(1, len(array)):
        key = array[i]
        j = i - 1
        while j >= 0 and key < array[j]:
            array[j + 1] = array[j]
            j -= 1
        array[j + 1] = key
    return array

# Generate a random list of integers
arr = generate_random_list(10)

# Print original array
print("Original array:", arr)

# Sort the array using insertion sort
sorted_arr = insertion_sort(arr)

# Print sorted array
print("Sorted array:", sorted_arr)
