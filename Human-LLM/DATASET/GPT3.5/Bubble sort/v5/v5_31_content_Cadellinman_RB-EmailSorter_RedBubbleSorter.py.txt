import imaplib

# Connect to the IMAP server
M = imaplib.IMAP4_SSL('imap.gmail.com')

# Attempt login
try:
    M.login('youremailinhere', 'yourpasswordinhere')
    print("Logged in successfully")
except imaplib.IMAP4.error:
    print("Login failed")

# Select the INBOX folder
M.select("INBOX")

# Lists to store email dates and corresponding amounts
list_dates = []
list_amounts = []

# Search for emails from 'congratulations@redbubble.com'
typ, data_raw = M.search(None, '(FROM "congratulations@redbubble.com")')

# Process email data
for num in data_raw[0].split():
    # Move email to destination folder and mark it as deleted
    M.store(num, '+X-GM-LABELS', 'yourdestinationfolder')
    M.store(num, '+FLAGS', '\\Deleted')

    # Fetch email subject
    typ, data = M.fetch(num, '(BODY.PEEK[HEADER.FIELDS (SUBJECT)])')
    subject_data = data[-1][0].decode('utf-8')

    # Extract amount from subject line
    if "=28=C2=A3" in subject_data:
        amount = subject_data.split('=28=C2=A3')[-1].split('=')[0]
    elif "=28=E2=82=AC" in subject_data:
        amount = float(subject_data.split('=28=E2=82=AC')[-1].split('=')[0]) * 0.88
    else:
        amount = float(subject_data.split('(US$')[-1].split(')')[0]) * 0.77

    # Round the amount to 2 decimal places and add it to the list
    list_amounts.append(round(float(amount), 2))

    # Fetch email received date
    typ, dates = M.fetch(num, '(BODY[HEADER.FIELDS (RECEIVED DATE)])')
    received_date = dates[-1][0].decode('utf-8').split(' ')[8:11]
    received_date = ' '.join(received_date).replace('\\r\\n', '')

    # Add the received date to the list
    list_dates.append(received_date)

# Combine amounts for emails received on the same date
for i in range(len(list_dates) - 1):
    if list_dates[i] == list_dates[i + 1]:
        list_amounts[i] = str(round(float(list_amounts[i]) + float(list_amounts[i + 1]), 2))
        list_amounts[i + 1] = str(0)

# Remove entries with zero amounts
list_dates = [d for d, a in zip(list_dates, list_amounts) if a != "0" and a != "0.0"]
list_amounts = [a for a in list_amounts if a != "0" and a != "0.0"]

# Print dates and corresponding amounts
for date, amount in zip(list_dates, list_amounts):
    print(f"{date}: £{amount}")

# Expunge deleted emails, close the connection, and log out
M.expunge()
M.close()
M.logout()
