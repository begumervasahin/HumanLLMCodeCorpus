import sys
import argparse
from sorting_algorithms import bubbleSort, selectionSort, mergeSort, quickSort

# Mapping of command-line algorithm names to their corresponding class names
class_names = {'bubble': 'bubbleSort', 'selection': 'selectionSort', 'merge': 'mergeSort', 'quick': 'quickSort'}

def display_list(lst):
    """
    Display a list in a readable format.
    """
    lst = [str(i) for i in lst]
    if not lst:
        print('| |')
    else:
        output = '|' + ' | '.join(lst) + ' |'
        print(output)
    input("Press Enter to continue...")

def main(argv):
    """
    Main function to sort a list using a specified algorithm.
    """
    trace_mode = False
    parser = argparse.ArgumentParser()
    parser.add_argument("input_list", nargs='+', type=int, help="List of integers separated by space")
    parser.add_argument("alg", type=str, help="bubble, selection, merge, or quick")
    parser.add_argument("-t", "--trace", action="store_true", help="Print the sorting algorithm step-by-step")
    args = parser.parse_args()

    if args.trace:
        trace_mode = True

    if args.alg not in class_names:
        print('\nInvalid algorithm name')
        parser.print_help()
        return

    print('Input: ' + str(args.input_list))
    class_name = class_names[args.alg]
    sorting_algorithm = getattr(globals()[class_name], class_name)
    obj = sorting_algorithm(args.input_list, trace_mode)
    print('Result: ' + str(obj.sort()))

if __name__ == '__main__':
    main(sys.argv[1:])
