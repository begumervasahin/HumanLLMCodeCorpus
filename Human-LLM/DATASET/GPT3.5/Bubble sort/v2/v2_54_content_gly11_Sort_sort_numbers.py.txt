def insertion_sort(lst):
    # Get the length of the list
    length = len(lst)

    # Iterate over the list starting from the second element
    for j in range(1, length):
        # Get the current element to be inserted
        key = lst[j]
        i = j - 1

        # Move elements of the sorted part of the list that are greater than the key
        # to one position ahead of their current position
        while i >= 0 and lst[i] > key:
            lst[i + 1] = lst[i]
            i -= 1

        # Insert the key into the correct position
        lst[i + 1] = key

    # Return the sorted list
    return lst

def bubble_sort(lst):
    # Get the length of the list
    count = len(lst)

    # Iterate over the list
    for i in range(0, count):
        # Compare each pair of adjacent elements
        for j in range(i + 1, count):
            # Swap if the elements are in the wrong order
            if lst[i] > lst[j]:
                lst[i], lst[j] = lst[j], lst[i]

    # Return the sorted list
    return lst

# Example usage
my_list = [64, 34, 25, 12, 22, 11, 90]
print("Original list:", my_list)

# Sort the list using insertion sort
sorted_list_insertion = insertion_sort(my_list.copy())
print("Sorted list (Insertion Sort):", sorted_list_insertion)

# Sort the list using bubble sort
sorted_list_bubble = bubble_sort(my_list.copy())
print("Sorted list (Bubble Sort):", sorted_list_bubble)
