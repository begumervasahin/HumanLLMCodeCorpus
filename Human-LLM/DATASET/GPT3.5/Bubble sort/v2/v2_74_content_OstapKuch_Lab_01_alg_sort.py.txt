class Student:
    def __init__(self, name, rating, growth):
        self.name = name
        self.rating = rating
        self.growth = growth

    def __repr__(self):
        return f"Student({self.name}, {self.rating}, {self.growth})"

def bubble_sort(arr):
    comparisons = 0
    swaps = 0
    is_sorted = False
    while not is_sorted:
        is_sorted = True
        for i in range(len(arr) - 1):
            comparisons += 1
            if int(arr[i].rating) < int(arr[i + 1].rating):
                swaps += 1
                arr[i], arr[i + 1] = arr[i + 1], arr[i]
                is_sorted = False
    print("Bubble Sort\nComparisons:", comparisons, "\nSwaps:", swaps)

def quick_sort(start, end, arr):
    comparisons = 0
    swaps = 0
    if start < end:
        pivot_index = start
        smaller_element = start
        for i in range(start, end):
            comparisons += 1
            if int(arr[i].growth) < int(arr[pivot_index].growth):
                smaller_element += 1
                arr[i], arr[smaller_element] = arr[smaller_element], arr[i]
                swaps += 1
        arr[start], arr[smaller_element] = arr[smaller_element], arr[start]
        swaps += 1
        quick_sort(start, smaller_element, arr)
        quick_sort(smaller_element + 1, end, arr)
    print("Quick Sort\nComparisons:", comparisons, "\nSwaps:", swaps)

# Example usage
students = [
    Student("Alice", 80, 5),
    Student("Bob", 70, 7),
    Student("Charlie", 90, 3),
    Student("David", 85, 6),
    Student("Eve", 75, 4)
]

print("Unsorted students:")
for student in students:
    print(student)

bubble_sort(students.copy())
print("\nSorted students using Bubble Sort:")
for student in students:
    print(student)

students = [
    Student("Alice", 80, 5),
    Student("Bob", 70, 7),
    Student("Charlie", 90, 3),
    Student("David", 85, 6),
    Student("Eve", 75, 4)
]

quick_sort(0, len(students), students)
print("\nSorted students using Quick Sort:")
for student in students:
    print(student)
