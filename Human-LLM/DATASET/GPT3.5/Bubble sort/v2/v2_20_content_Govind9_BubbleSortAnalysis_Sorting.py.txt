def bubble_sort(arr):
    moves = 0
    for j in range(len(arr)):
        moves += 3  # Increment for loop initialization and condition checks
        flag = False
        for i in range(len(arr) - 1 - j):
            moves += 2  # Increment for loop initialization and condition checks
            if arr[i] > arr[i + 1]:
                moves += 2  # Increment for swap operation
                flag = True
                arr[i], arr[i + 1] = arr[i + 1], arr[i]
        if not flag:
            break
    return moves

def quick_sort(arr):
    moves = 0
    for j in range(len(arr)):
        moves += 3  # Increment for loop initialization and condition checks
        swap = 0
        for i in range(len(arr) - 1 - j):
            moves += 2  # Increment for loop initialization and condition checks
            if arr[i] > arr[i + 1]:
                moves += 2  # Increment for swap operation
                swap += 1
                arr[i], arr[i + 1] = arr[i + 1], arr[i]
        if swap < 2:
            break
    return moves

def better_bubble_sort(arr):
    moves = 0
    for j in range(len(arr)):
        moves += 3  # Increment for loop initialization and condition checks
        flag = False
        for i in range(len(arr) - 1 - j):
            moves += 2  # Increment for loop initialization and condition checks
            if arr[i] > arr[i + 1]:
                moves += 2  # Increment for swap operation
                flag = True
                arr[i], arr[i + 1] = arr[i + 1], arr[i]
        if not flag:
            break
    return moves

def analyze_moves(Qmoves, Bmoves, BBmoves):
    print("Qsort\tBubble\tBetterBubble")
    for Q, B, BB in zip(Qmoves, Bmoves, BBmoves):
        print(f"{Q}\t{B}\t{BB}")
    total_Q = sum(Qmoves)
    total_B = sum(Bmoves)
    total_BB = sum(BBmoves)
    print(f"{total_Q}\t{total_B}\t{total_BB}")

def permute(s, arr, Qmoves, Bmoves, BBmoves):
    if len(arr) == len(s):
        sort(arr, Qmoves, Bmoves, BBmoves)
    for i in range(len(s)):
        if str(i) not in arr:
            permute(s, arr + str(i), Qmoves, Bmoves, BBmoves)

def sort(arr, Qmoves, Bmoves, BBmoves):
    a = arr[:]
    Bmoves.append(bubble_sort(a))
    a = arr[:]
    Qmoves.append(quick_sort(a))
    a = arr[:]
    BBmoves.append(better_bubble_sort(a))
    analyze_moves(Qmoves, Bmoves, BBmoves)
