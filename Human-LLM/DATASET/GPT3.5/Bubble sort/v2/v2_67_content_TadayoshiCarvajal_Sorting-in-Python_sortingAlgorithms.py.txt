from matplotlib import pyplot as plt
from tqdm import tqdm
import random
import time

# Utility Functions
def generate_random_list(n, low=0, high=100):
    return [random.randint(1, n) for _ in range(n)]

def measure_sort_time(sort_function, array):
    shuffled_array = array.copy()
    random.shuffle(shuffled_array)
    start_time = time.time()
    sort_function(shuffled_array)
    stop_time = time.time()
    return stop_time - start_time

def average_sort_time(sort_function, array, trials=500):
    total_time = 0
    for _ in range(trials):
        total_time += measure_sort_time(sort_function, array)
    return total_time / trials

def is_sorted(array):
    for i in range(1, len(array)):
        if array[i] < array[i - 1]:
            return False
    return True

# Sorting Algorithms
def insertion_sort(array):
    for j in range(len(array)):
        key, i = array[j], j - 1
        while i > -1 and array[i] > key:
            array[i + 1], i = array[i], i - 1
        array[i + 1] = key
    return array

def bubble_sort(array):
    is_sorted = False
    while not is_sorted:
        is_sorted = True
        for i in range(1, len(array)):
            if array[i] < array[i - 1]:
                array[i], array[i - 1], is_sorted = array[i - 1], array[i], False
    return array

def selection_sort(array):
    for i in range(len(array)):
        min_index = i
        for j in range(i + 1, len(array)):
            if array[j] < array[min_index]:
                min_index = j
        array[i], array[min_index] = array[min_index], array[i]
    return array

def heap_sort(array):
    def sift_down(array, start, end):
        root = start
        while True:
            child = 2 * root + 1
            if child > end:
                break
            if child + 1 <= end and array[child] < array[child + 1]:
                child += 1
            if array[root] < array[child]:
                array[root], array[child] = array[child], array[root]
                root = child
            else:
                break

    for start in range(len(array) // 2 - 1, -1, -1):
        sift_down(array, start, len(array) - 1)

    for end in range(len(array) - 1, 0, -1):
        array[end], array[0] = array[0], array[end]
        sift_down(array, 0, end - 1)

    return array

def merge_sort(array):
    if len(array) <= 1:
        return array

    mid = len(array) // 2
    left_half = array[:mid]
    right_half = array[mid:]

    merge_sort(left_half)
    merge_sort(right_half)

    i = j = k = 0

    while i < len(left_half) and j < len(right_half):
        if left_half[i] < right_half[j]:
            array[k] = left_half[i]
            i += 1
        else:
            array[k] = right_half[j]
            j += 1
        k += 1

    while i < len(left_half):
        array[k] = left_half[i]
        i += 1
        k += 1

    while j < len(right_half):
        array[k] = right_half[j]
        j += 1
        k += 1

    return array

def quick_sort(array):
    def partition(array, low, high):
        pivot = array[(low + high) // 2]
        i = low - 1
        j = high + 1
        while True:
            i += 1
            while array[i] < pivot:
                i += 1
            j -= 1
            while array[j] > pivot:
                j -= 1
            if i >= j:
                return j
            array[i], array[j] = array[j], array[i]

    def quick_sort_helper(array, low, high):
        if low < high:
            split_index = partition(array, low, high)
            quick_sort_helper(array, low, split_index)
            quick_sort_helper(array, split_index + 1, high)

    quick_sort_helper(array, 0, len(array) - 1)
    return array

def radix_sort(array):
    RADIX = 10
    placement = 1
    max_digit = max(array)

    while placement < max_digit:
        buckets = [list() for _ in range(RADIX)]
        for num in array:
            tmp = int((num / placement) % RADIX)
            buckets[tmp].append(num)

        a = 0
        for b in range(RADIX):
            buck = buckets[b]
            for i in buck:
                array[a] = i
                a += 1
        placement *= RADIX

    return array

def tim_sort(array):
    return sorted(array)

if __name__ == '__main__':
    input_sizes = [i for i in range(100, 25000, 500)]
    sort_times = {
        'Bubble Sort': [average_sort_time(bubble_sort, generate_random_list(i)) for i in tqdm(input_sizes[:10], 'Bubble Sort')],
        'Selection Sort': [average_sort_time(selection_sort, generate_random_list(i)) for i in tqdm(input_sizes[:10], 'Selection Sort')],
        'Insertion Sort': [average_sort_time(insertion_sort, generate_random_list(i)) for i in tqdm(input_sizes[:10], 'Insertion Sort')],
        'Merge Sort': [average_sort_time(merge_sort, generate_random_list(i)) for i in tqdm(input_sizes, 'Merge Sort')],
        'Heap Sort': [average_sort_time(heap_sort, generate_random_list(i)) for i in tqdm(input_sizes, 'Heap Sort')],
        'Quick Sort': [average_sort_time(quick_sort, generate_random_list(i)) for i in tqdm(input_sizes, 'Quick Sort')],
        'Radix Sort': [average_sort_time(radix_sort, generate_random_list(i)) for i in tqdm(input_sizes, 'Radix Sort')]
    }

    n_squared = {'Bubble Sort', 'Selection Sort', 'Insertion Sort'}
    for sort_name, sort_times_list in sort_times.items():
        if sort_name in n_squared:
            plt.plot(input_sizes[:10], sort_times_list, label=sort_name)
        else:
            plt.plot(input_sizes, sort_times_list, label=sort_name)

    plt.legend(loc='lower right')
    plt.ylabel('Time (seconds)')
    plt.xlabel('List Size')
    plt.show()
