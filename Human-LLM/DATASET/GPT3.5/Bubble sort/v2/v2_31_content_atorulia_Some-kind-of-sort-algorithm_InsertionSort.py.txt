import random

def generate_random_list(length: int) -> list[int]:
    """
    Generates a list of random numbers.

    Args:
        length (int): The length of the list.

    Returns:
        list[int]: The generated list of random numbers.
    """
    return [random.randrange(100) for _ in range(length)]

def is_sorted(array: list) -> bool:
    """
    Checks if an array is sorted in non-decreasing order.

    Args:
        array (list): The array to be checked.

    Returns:
        bool: True if the array is sorted, False otherwise.
    """
    for i in range(len(array) - 1):
        if array[i] > array[i + 1]:
            return False
    return True

def insertion_sort(array: list) -> list:
    """
    Sorts an array using the Insertion Sort algorithm.

    Args:
        array (list): The array to be sorted.

    Returns:
        list: The sorted array.
    """
    for i in range(1, len(array)):
        key = array[i]
        j = i - 1
        while j >= 0 and key < array[j]:
            array[j + 1] = array[j]
            j -= 1
        array[j + 1] = key
    return array

def main():
    # Generate a list of 10 random numbers
    original_array = generate_random_list(10)

    # Print the original array
    print("Original array:")
    print(original_array)

    # Sort the array using insertion sort and print the sorted array
    sorted_array = insertion_sort(original_array)
    print("Sorted array:")
    print(sorted_array)

    # Check if the array is sorted
    if is_sorted(sorted_array):
        print("Array is sorted.")
    else:
        print("Array is not sorted.")

if __name__ == "__main__":
    main()
