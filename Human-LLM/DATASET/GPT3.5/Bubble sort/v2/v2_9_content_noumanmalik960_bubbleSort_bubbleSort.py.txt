def bubble_sort(arr):
    # Flag to track if any elements were swapped
    swapped = True
    
    # Continue looping until no more swaps are needed
    while swapped:
        swapped = False
        
        # Loop through the array
        for i in range(len(arr)-1):
            # Check if adjacent elements are in the wrong order
            if arr[i] > arr[i+1]:
                # Swap the elements
                arr[i], arr[i+1] = arr[i+1], arr[i]
                # Set swapped flag to True
                swapped = True

# Example usage
arr = [9, 8, 6, 6, 5, 4, 3, 2, 1, 0]
bubble_sort(arr)
print(arr)
