import random
from time import time

# Generate a list of unique random numbers
list_length = 10000
max_value = 10000
unsorted_list = random.sample(range(1, max_value + 1), list_length)

def bubble_sort(arr):
    """
    Sorts a list of elements using the bubble sort algorithm.

    Args:
        arr (list): The list to be sorted.

    Returns:
        list: The sorted list.
    """
    is_sorted = False
    while not is_sorted:
        is_sorted = True
        for i in range(len(arr) - 1):
            if arr[i] > arr[i + 1]:
                arr[i], arr[i + 1] = arr[i + 1], arr[i]
                is_sorted = False
    return arr

# Copy the unsorted list for each sorting algorithm
unsorted_list_copy = unsorted_list[:]

# Measure the time taken for bubble sort
start_time = time()
bubble_sort(unsorted_list_copy)
end_time = time()
print(f"Bubble sort took {end_time - start_time:.6f} seconds to sort a list of {list_length} items.")

# Copy the unsorted list for the system sort algorithm
unsorted_list_copy = unsorted_list[:]

# Measure the time taken for the system sort (Timsort)
start_time = time()
unsorted_list_copy.sort()
end_time = time()
print(f"System Python sort (Timsort) took {end_time - start_time:.6f} seconds to sort a list of {list_length} items.")
