import random

def bubble_sort(arr):
    """
    Sort the array using the bubble sort algorithm.

    Args:
    arr (list): The list to be sorted.
    """
    length = len(arr)
    for pass_num in range(length):
        for i in range(1, length - pass_num):
            if arr[i - 1] > arr[i]:
                # Swap elements if they are in the wrong order
                arr[i - 1], arr[i] = arr[i], arr[i - 1]
        print(f"Array after {pass_num} sort is:")
        print_array(arr)

def print_array(arr):
    """
    Print the elements of the array.

    Args:
    arr (list): The list to be printed.
    """
    for i in range(len(arr)):
        print(arr[i], end="")
    print()

def fill_array(arr):
    """
    Fill the array with random integers.

    Args:
    arr (list): The list to be filled with random integers.
    """
    for i in range(8):
        arr.append(random.randint(1, 9))
    print("The array that will be worked on is:")
    print_array(arr)

array = []
fill_array(array)
bubble_sort(array)
print(" ")
print("THE FINAL SORTED ARRAY IS")
print_array(array)
