from time import time
import configuraciones

def quick_sort(alist):
    """Sorts the list using the quick sort algorithm."""
    quick_sort_helper(alist, 0, len(alist) - 1)

def quick_sort_helper(alist, first, last):
    """Helper function for quick sort."""
    if first < last:
        split_point = partition(alist, first, last)
        quick_sort_helper(alist, first, split_point - 1)
        quick_sort_helper(alist, split_point + 1, last)

def partition(alist, first, last):
    """Partitions the list for quick sort."""
    pivot_value = alist[first]
    left_mark = first + 1
    right_mark = last
    done = False
    while not done:
        while left_mark <= right_mark and alist[left_mark] <= pivot_value:
            left_mark += 1
        while alist[right_mark] >= pivot_value and right_mark >= left_mark:
            right_mark -= 1
        if right_mark < left_mark:
            done = True
        else:
            alist[left_mark], alist[right_mark] = alist[right_mark], alist[left_mark]
    alist[first], alist[right_mark] = alist[right_mark], alist[first]
    return right_mark

def measure_sorting_time(configurations, file_name):
    """Measures sorting time for each configuration."""
    total_time = 0
    with open(file_name, 'a') as output_file:
        for config_name, config_data in configurations.items():
            total_config_time = 0
            for data in config_data:
                start_time = time()
                quick_sort(data)
                elapsed_time = time() - start_time
                total_config_time += elapsed_time
                output_file.write(f"{config_name}: {data}\n")
            total_time += total_config_time
    return total_time

def main():
    configurations.load_all_configurations()
    file_name = 'tiempoQuick.txt'
    total_time = measure_sorting_time(configurations.get_all_configurations(), file_name)
    with open(file_name, 'a') as time_file:
        time_file.write(f'Total time: {total_time}\n')

if __name__ == "__main__":
    main()
