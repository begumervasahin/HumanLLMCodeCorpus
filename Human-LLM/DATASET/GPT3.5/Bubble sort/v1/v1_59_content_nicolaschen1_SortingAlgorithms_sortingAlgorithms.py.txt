def insertion_sort(lst):
    for i in range(1, len(lst)):
        element = lst[i]
        j = i
        while j > 0 and lst[j - 1] > element:
            lst[j] = lst[j - 1]
            j -= 1
        lst[j] = element
    return lst

def bubble_sort(lst):
    permutation = True
    passage = 0
    while permutation:
        permutation = False
        passage += 1
        for element in range(0, len(lst) - passage):
            if lst[element] > lst[element + 1]:
                permutation = True
                lst[element], lst[element + 1] = lst[element + 1], lst[element]
    return lst

def shaker_sort(lst):
    permutation = True
    way, element = 1, 0
    start, end = 0, len(lst) - 2
    while permutation:
        permutation = False
        while (element < end and way == 1) or (element > start and way == -1):
            if lst[element] > lst[element + 1]:
                permutation = True
                lst[element], lst[element + 1] = lst[element + 1], lst[element]
            element += way
        if way == 1:
            end -= 1
        else:
            start += 1
        way = -way
    return lst

def gnome_sort(lst):
    i_b, i_i, size = 1, 2, len(lst)
    while i_b < size:
        if lst[i_b - 1] <= lst[i_b]:
            i_b, i_i = i_i, i_i + 1
        else:
            lst[i_b - 1], lst[i_b] = lst[i_b], lst[i_b - 1]
            i_b -= 1
            if i_b == 0:
                i_b, i_i = i_i, i_i + 1
    return lst

def merge(left, right):
    result = []
    left_index, right_index = 0, 0
    while left_index < len(left) and right_index < len(right):
        if left[left_index] <= right[right_index]:
            result.append(left[left_index])
            left_index += 1
        else:
            result.append(right[right_index])
            right_index += 1
    if left:
        result.extend(left[left_index:])
    if right:
        result.extend(right[right_index:])
    return result

def merge_sort(lst):
    if len(lst) <= 1:
        return lst
    middle = len(lst) // 2
    left = lst[:middle]
    right = lst[middle:]
    left = merge_sort(left)
    right = merge_sort(right)
    return merge(left, right)

def selection_sort(lst):
    nb = len(lst)
    for element in range(0, nb):
        min_index = element
        for j in range(element + 1, nb):
            if lst[j] < lst[min_index]:
                min_index = j
        if min_index != element:
            lst[element], lst[min_index] = lst[min_index], lst[element]
    return lst

list1 = [6, 3, 2, 89, 10, 2.3, 5, 80.2]
print("Insertion Sort:", insertion_sort(list1[:]))
print("Bubble Sort:", bubble_sort(list1[:]))
print("Shaker Sort:", shaker_sort(list1[:]))
print("Gnome Sort:", gnome_sort(list1[:]))
print("Merge Sort:", merge_sort(list1[:]))
print("Selection Sort:", selection_sort(list1[:]))
