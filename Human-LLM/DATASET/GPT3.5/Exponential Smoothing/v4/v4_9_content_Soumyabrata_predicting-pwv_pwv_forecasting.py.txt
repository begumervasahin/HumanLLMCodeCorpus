import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from read_matfile import read_matfile
from statsmodels.tsa.api import ExponentialSmoothing
from matplotlib.dates import DateFormatter

# Read data from MATLAB file
matlab_file_path = './data/PWV_2010from_WS_2_withGradient.mat'
timestamps, pwv = read_matfile(matlab_file_path)

# Create DataFrame from the data
df = pd.DataFrame(data=np.column_stack((timestamps, pwv)), columns=['timestamps', 'pwv']).set_index('timestamps')

# Define start index and lengths for train and test data
start_index = 73513
train_length = 10000
test_length = 50

# Split data into train and test sets
train_data = df[start_index:start_index + train_length]
test_data = df[start_index + train_length:start_index + train_length + test_length]
y_hat_avg = test_data.copy()

# Fit exponential smoothing model
print('Computation started')
fit1 = ExponentialSmoothing(np.asarray(train_data['pwv']), seasonal_periods=288, trend='add', seasonal='add').fit()
y_hat_avg['Holt_Winter'] = fit1.forecast(len(test_data))
print('Computation completed')

# Plotting
plt.figure(figsize=(10, 5))
plt.plot(train_data['pwv'][-150:], 'b:', label='Train')
plt.plot(test_data['pwv'], 'r--', label='Test')
plt.plot(y_hat_avg['Holt_Winter'], 'k-', label='Predicted')     
plt.legend(loc='best', fontsize=18)
plt.gcf().autofmt_xdate()
date_formatter = DateFormatter('%d-%m-%y %H:%M')
plt.gca().xaxis.set_major_formatter(date_formatter)
plt.grid(True)
plt.xlabel('Timestamps', fontsize=14)
plt.ylabel('Precipitable Water Vapor (in mm)', fontsize=14)
plt.tight_layout()

# Save and display the plot
save_path = './results/73513-example.pdf'
plt.savefig(save_path)
plt.show()
