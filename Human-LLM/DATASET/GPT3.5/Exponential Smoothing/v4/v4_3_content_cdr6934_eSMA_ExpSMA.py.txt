import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

def generate_exponential_sma(data, window):
    """
    Generate exponential Simple Moving Average (SMA) from data.

    Parameters:
        data (DataFrame): Time series data.
        window (int): Span parameter for the exponential SMA.

    Returns:
        DataFrame: Exponential SMA result.
    """
    return data.ewm(span=window).mean()

def import_clean_ts(filename):
    """
    Read and clean time series data from a CSV file.

    Parameters:
        filename (str): Name of the CSV file.

    Returns:
        DataFrame: Cleaned time series data.
    """
    ts_data = pd.read_csv(filename)
    ts_data['Date'] = pd.to_datetime(ts_data['Date'])
    return ts_data.sort_values(by='Date')

def collect_user_input(window_default):
    """
    Collect user input for the window parameter or use the default value.

    Parameters:
        window_default (int): Default value for the window parameter.

    Returns:
        int: User input or default value for the window parameter.
    """
    user_input = input("Enter the Long Term Window or press Enter for default ({0} Weeks): ".format(window_default))
    if user_input == '':
        user_input = window_default
    return int(user_input * 5)

def main():
    """
    Main function to calculate and plot the second derivative of the difference
    between long-term and short-term exponential moving averages.
    """
    # Import and clean time series data
    stock = import_clean_ts("data/goog.csv")
    data = pd.Series(np.float32(stock['Close']), index=stock['Date'])

    # Calculate long-term and short-term exponential moving averages
    lt_av = generate_exponential_sma(data, collect_user_input(4))
    st_av = generate_exponential_sma(data, collect_user_input(2))

    # Calculate the difference between long-term and short-term exponential moving averages
    diff_ltst = lt_av - st_av

    # Calculate the second derivative of the difference
    diff_ltst_av = generate_exponential_sma(diff_ltst, 7.5)
    diff_av_diff = diff_ltst_av - diff_ltst

    # Plot the second derivative of the difference
    diff_av_diff.plot(style='k', title="2nd Derivative of GOOG Close Price")
    plt.show()

if __name__ == '__main__':
    main()
