class Graph:
    """Graph class representing a collection of vertices and edges."""

    def __init__(self):
        """Initialize the Graph with an empty dictionary of vertices."""
        self.vertices = {}

    def add_vertex(self, key):
        """Add a vertex with the given key to the graph."""
        vertex = Vertex(key)
        self.vertices[key] = vertex

    def get_vertex(self, key):
        """Return the vertex object with the corresponding key."""
        return self.vertices[key]

    def __contains__(self, key):
        """Check if the graph contains a vertex with the given key."""
        return key in self.vertices

    def add_edge(self, src_key, dest_key, weight=1):
        """Add an edge from src_key to dest_key with the given weight."""
        self.vertices[src_key].add_neighbour(self.vertices[dest_key], weight)

    def does_edge_exist(self, src_key, dest_key):
        """Return True if there is an edge from src_key to dest_key."""
        return self.vertices[src_key].does_it_point_to(self.vertices[dest_key])

    def __len__(self):
        """Return the number of vertices in the graph."""
        return len(self.vertices)

    def __iter__(self):
        """Return an iterator over the vertices of the graph."""
        return iter(self.vertices.values())


class Vertex:
    """Vertex class representing a single vertex in the graph."""

    def __init__(self, key):
        """Initialize the vertex with the given key and an empty dictionary of neighbors."""
        self.key = key
        self.points_to = {}

    def get_key(self):
        """Return the key corresponding to this vertex object."""
        return self.key

    def add_neighbour(self, dest, weight):
        """Make this vertex point to dest with the given edge weight."""
        self.points_to[dest] = weight

    def get_neighbours(self):
        """Return all vertices pointed to by this vertex."""
        return self.points_to.keys()

    def get_weight(self, dest):
        """Get the weight of the edge from this vertex to dest."""
        return self.points_to[dest]

    def does_it_point_to(self, dest):
        """Return True if this vertex points to dest."""
        return dest in self.points_to


def floyd_warshall(g):
    """Apply the Floyd-Warshall algorithm to find shortest paths in the graph."""
    distance = {v: dict.fromkeys(g, float('inf')) for v in g}
    next_v = {v: dict.fromkeys(g, None) for v in g}

    for v in g:
        for n in v.get_neighbours():
            distance[v][n] = v.get_weight(n)
            next_v[v][n] = n

    for v in g:
        distance[v][v] = 0
        next_v[v][v] = None

    for p in g:
        for v in g:
            for w in g:
                if distance[v][w] > distance[v][p] + distance[p][w]:
                    distance[v][w] = distance[v][p] + distance[p][w]
                    next_v[v][w] = next_v[v][p]

    return distance, next_v


def print_path(next_v, u, v):
    """Print the shortest path from vertex u to v using the next_v dictionary."""
    p = u
    path_list = []

    while next_v[p][v]:
        path_list.append(p.get_key())
        p = next_v[p][v]

    path_list.append(v.get_key())
    return path_list

    return list_for_sketch
