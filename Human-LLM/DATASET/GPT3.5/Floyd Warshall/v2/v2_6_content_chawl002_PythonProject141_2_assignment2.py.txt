import os
import re
import sys
import time

# Regular expressions for parsing input
graph_re = re.compile(r"(\d+)\s(\d+)")
edge_re = re.compile(r"(\d+)\s(\d+)\s(\d+)")

def bellman_ford(graph):
    """
    Implementation of the Bellman-Ford algorithm.
    """
    path_pairs = []
    d = []

    for a in range(len(graph[0])):
        for b in range(len(graph[0])):
            d.append(float("inf"))
        d[a] = 0

        for c in range(1, len(graph[0])-1):
            for e in range(len(graph[0])):
                for f in range(len(graph[0])):
                    if float(graph[1][e][f]) < float("inf"):
                        if d[f] > d[e] + float(graph[1][e][f]):
                            d[f] = int(d[e]) + int(float(graph[1][e][f]))

        path_pairs.append(d)
        d = []

    return path_pairs

def floyd_warshall(graph):
    """
    Implementation of the Floyd-Warshall algorithm.
    """
    path_pairs = []
    d = []

    for a in range(len(graph[0])):
        drow = []
        for b in range(len(graph[0])):
            drow.append(float("inf"))
        d.append(drow)

    for c in range(len(graph[0])):
        for e in range(len(graph[0])):
            if c == e:
                d[c][c] = 0
            elif float(graph[1][c][e]) < float("inf"):
                d[c][e] = int(float(graph[1][c][e]))
            else:
                d[c][e] = float("inf")

    for c in range(len(graph[0])):
        for a in range(len(graph[0])):
            for b in range(len(graph[0])):
                if d[a][b] < d[a][c] + d[c][b]:
                    d[a][b] = d[a][b]
                else:
                    d[a][b] = d[a][c] + d[c][b]

    path_pairs.append(d)
    print(d)
    return path_pairs

def read_file(filename):
    """
    Read the input file and parse the graph information.
    """
    inFile = open(filename, 'r')
    line1 = inFile.readline()
    graph_match = graph_re.match(line1)

    if not graph_match:
        print(line1 + " not properly formatted")
        quit(1)

    vertices = list(range(int(graph_match.group(1))))
    edges = []

    for i in range(len(vertices)):
        row = []
        for j in range(len(vertices)):
            row.append(float("inf"))
        edges.append(row)

    for line in inFile.readlines():
        line = line.strip()
        edge_match = edge_re.match(line)
        if edge_match:
            source = edge_match.group(1)
            sink = edge_match.group(2)
            if int(source) >= len(vertices) or int(sink) >= len(vertices):
                print("Attempting to insert an edge between " + str(source) + " and " + str(sink) + " in a graph with " + str(len(vertices)) + " vertices")
                quit(1)
            weight = edge_match.group(3)
            edges[int(source)][int(sink)] = weight

    return (vertices, edges)

def write_file(length_matrix, filename):
    """
    Write the output to a file.
    """
    filename = os.path.splitext(os.path.split(filename)[1])[0]
    outFile = open('output/' + filename + '_output.txt', 'w+')

    for vertex in length_matrix:
        for length in vertex:
            outFile.write(str(length) + ',')
        outFile.write('\n')

def main(filename, algorithm):
    """
    Main function to run the Bellman-Ford or Floyd-Warshall algorithms based on user input.
    """
    algorithm = algorithm[1:]
    graph = read_file(filename)
    path_lengths = []

    if algorithm == 'b' or algorithm == 'B':
        path_lengths = bellman_ford(graph)

    if algorithm == 'f' or algorithm == 'F':
        path_lengths = floyd_warshall(graph)

    if algorithm == "both":
        start = time.clock()
        bellman_ford(graph)
        end = time.clock()
        BFTime = end - start

        floyd_warshall(graph)
        start = time.clock()
        end = time.clock()
        FWTime = end - start

        print("Bellman-Ford timing: " + str(BFTime))
        print("Floyd-Warshall timing: " + str(FWTime))

    write_file(path_lengths, filename)

if __name__ == '__main__':
    if len(sys.argv) < 3:
        print("python bellman_ford.py -<f|b> <input_file>")
        quit(1)
    main(sys.argv[2], sys.argv[1])
