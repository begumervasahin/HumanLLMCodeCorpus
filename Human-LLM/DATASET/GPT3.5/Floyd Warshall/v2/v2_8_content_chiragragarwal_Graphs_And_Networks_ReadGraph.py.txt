import numpy as np
from path import Path

class GraphReader:
    """
    Reads a graph matrix from a text file and stores it in a numpy matrix.
    """

    def __init__(self, file_path):
        """
        Initializes the GraphReader with the file path.

        Args:
        file_path (str): The path to the input file.
        """
        self.file_path = file_path

    def read_graph_matrix(self):
        """
        Reads the contents of the file and converts it into a matrix.

        Returns:
        numpy.matrix: The graph matrix.
        """
        with open(self.file_path, 'r') as file:
            graph_string = file.read()
            graph_matrix = np.matrix(graph_string)
        return graph_matrix

# Example usage
if __name__ == "__main__":
    # Replace "your_graph_file.txt" with the actual path to your graph file
    input_file_path = "your_graph_file.txt"
    graph_reader = GraphReader(input_file_path)
    graph_matrix = graph_reader.read_graph_matrix()
    print("Graph Matrix:")
    print(graph_matrix)
