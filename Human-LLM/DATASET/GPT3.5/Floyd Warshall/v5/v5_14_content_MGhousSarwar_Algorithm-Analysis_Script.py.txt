import random
import sys
import time
import matplotlib.pyplot as plt
import networkx as nx
from scipy.interpolate import interp1d
from graph import Graph, Algorithms

class Algorithms:
    def __init__(self):
        self.vertices_count = 0

    def floyd_warshall(self, graph):
        self.vertices_count = len(graph)
        dist = [[0] * self.vertices_count for _ in range(self.vertices_count)]

        for i in range(self.vertices_count):
            for j in range(self.vertices_count):
                dist[i][j] = graph[i][j]

        for k in range(self.vertices_count):
            for i in range(self.vertices_count):
                for j in range(self.vertices_count):
                    if dist[i][k] + dist[k][j] < dist[i][j]:
                        dist[i][j] = dist[i][k] + dist[k][j]

        self.print_solution(dist)

    def print_solution(self, distances):
        for i in range(self.vertices_count):
            for j in range(self.vertices_count):
                if distances[i][j] == float('inf'):
                    print("INF\t", end='')
                else:
                    print(str(distances[i][j]) + "\t", end='')
            print()
        print()

    def minimum_distance(self, distances, shortest_path_tree_set):
        min_distance = sys.maxsize
        min_index = 0

        for v in range(self.vertices_count):
            if not shortest_path_tree_set[v] and distances[v] <= min_distance:
                min_distance = distances[v]
                min_index = v

        return min_index

    def print_dijkstra(self, from_vertex, distances):
        print('Distance from:', from_vertex)
        print('Vertex\tDistance from source')

        for i in range(self.vertices_count):
            print(str(i) + '\t\t' + str(distances[i]))

    def dijkstra(self, graph):
        self.vertices_count = len(graph)

        for source in range(self.vertices_count):
            distances = [sys.maxsize] * self.vertices_count
            shortest_path_tree_set = [False] * self.vertices_count
            distances[source] = 0

            for _ in range(self.vertices_count - 1):
                u = self.minimum_distance(distances, shortest_path_tree_set)
                shortest_path_tree_set[u] = True

                for v in range(self.vertices_count):
                    if not shortest_path_tree_set[v] and graph[u][v] and distances[u] != sys.maxsize \
                            and distances[u] + graph[u][v] < distances[v]:
                        distances[v] = distances[u] + graph[u][v]

            self.print_dijkstra(source, distances)

def generate_random_graph(vertices):
    graph = nx.DiGraph()
    weights = [[random.randint(1, 1000) for _ in range(vertices)] for _ in range(vertices)]

    for i in range(vertices):
        weights[i][i] = 0  # Avoid self-loops

    for i in range(vertices):
        for j in range(vertices):
            if weights[i][j] != 0:
                graph.add_edges_from([(i, j)], weight=weights[i][j])

    return graph, weights

def remove_random_edges(graph, reduction_factor):
    edges_to_remove = int(len(graph.edges) / reduction_factor)

    for _ in range(edges_to_remove):
        edge = random.choice(list(graph.edges()))
        graph.remove_edge(*edge)

def main():
    vertices_list = []
    floyd_times = []
    dijkstra_times = []

    for graph_index in range(5):
        while True:
            vertices = random.randint(4, 8)

            if vertices not in vertices_list:
                vertices_list.append(vertices)
                break

        graph, weights = generate_random_graph(vertices)
        remove_random_edges(graph, random.uniform(1.0, 1.1))

        print('Total Remaining Edges for Graph {}: {}'.format(graph_index, len(graph.edges)))

        algorithms = Algorithms()

        start_time = time.time()
        algorithms.floyd_warshall(weights)
        floyd_elapsed_time = time.time() - start_time
        floyd_times.append(floyd_elapsed_time)

        start_time1 = time.time()
        algorithms.dijkstra(weights)
        dijkstra_elapsed_time = time.time() - start_time1
        dijkstra_times.append(dijkstra_elapsed_time)

        print("Floyd Execution Time:", floyd_elapsed_time)
        print("Dijkstra Execution Time:", dijkstra_elapsed_time)

    floyd_times.sort()
    dijkstra_times.sort()
    vertices_list.sort()

    f = interp1d(vertices_list, floyd_times)
    ff = interp1d(vertices_list, dijkstra_times)
    f2 = interp1d(vertices_list, floyd_times, kind='cubic')
    ff2 = interp1d(vertices_list, dijkstra_times, kind='cubic')

    plt.plot(vertices_list, floyd_times, 'o', vertices_list, f(vertices_list), '-', vertices_list, f2(vertices_list), '-')
    plt.plot(vertices_list, dijkstra_times, 'o', vertices_list, ff(vertices_list), '-', vertices_list, ff2(vertices_list), '-')
    plt.legend(['Floyd', '', 'Floyd Line', 'Dijkstra', '', 'Dijkstra Line'], loc='best')
    plt.show()

if __name__ == "__main__":
    main()
