import sys
from pathlib import Path
from PlotGraph import PlotGraph
from ReadGraph import ReadGraph

def main():
    # Check if the input file is provided as a command-line argument
    if len(sys.argv) <= 1:
        raise ValueError("Please provide the file path as a command-line argument")

    # Extract the input file path from the command-line argument
    input_file_path = Path(sys.argv[1])

    # Ensure that the input file exists
    if not input_file_path.exists():
        raise FileNotFoundError("File not found:", input_file_path)

    # Read the graph from the input file
    graph_reader = ReadGraph(input_file_path)
    graph_matrix = graph_reader.read()

    # Display the graph matrix
    print("Analyzing the graph matrix...\n")
    print(graph_matrix)

    # Plot the graph
    print("\nPlotting the graph...")
    graph_plotter = PlotGraph(graph_matrix)
    graph_plotter.plot()

if __name__ == '__main__':
    main()
