import random

# List of number of vertices for the graphs
num_vertices_list = [5, 10, 20, 50, 100, 200, 500, 750, 1000, 1250, 1500, 2000, 3000, 5000]

# Generate random graphs and save them to files
for num_vertices in num_vertices_list:
    # Generate a random adjacency matrix for the graph
    matrix = [[random.randint(1, 10) for _ in range(num_vertices)] for _ in range(num_vertices)]

    # Write the adjacency matrix to a file
    with open("randograph-{}.txt".format(num_vertices), "w") as file:
        # Write the number of vertices on the first line
        file.write("{}\n".format(num_vertices))

        # Write the edges with their weights
        for i in range(num_vertices):
            # Randomly select 10 edges for each vertex
            edges = [random.randint(0, num_vertices - 1) for _ in range(10)]
            for j in edges:
                # Write the edge and its weight to the file
                file.write("{} {} {}\n".format(i + 1, j + 1, matrix[i][j]))
