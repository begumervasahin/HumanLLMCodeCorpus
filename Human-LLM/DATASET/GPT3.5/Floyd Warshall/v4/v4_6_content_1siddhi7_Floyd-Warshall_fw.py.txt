import numpy as np
import math

# Set options for numpy printing
np.set_printoptions(precision=2)
np.set_printoptions(suppress=True)

# Define infinity
infinity = math.inf

def get_adjacency_matrix():
    """
    Prompt the user to enter the adjacency matrix.

    Returns:
        tuple: A tuple containing the number of vertices and the adjacency matrix.
    """
    num_vertices = int(input("Enter the number of vertices: "))
    print("Enter the values of the adjacency matrix row-wise:")
    graph = np.zeros((num_vertices, num_vertices))
    for i in range(num_vertices):
        for j in range(num_vertices):
            graph[i][j] = int(input())
    return num_vertices, graph

def floyd_warshall(graph, num_vertices):
    """
    Apply the Floyd-Warshall algorithm to find the shortest paths between all pairs of vertices.

    Args:
        graph (ndarray): The adjacency matrix representing the graph.
        num_vertices (int): The number of vertices in the graph.

    Returns:
        None. Prints the intermediate distance matrices.
    """
    print("\n\n")
    print("Initial Matrix (D[0]):\n ", graph)
    print("\n")

    for k in range(num_vertices):
        for i in range(num_vertices):
            for j in range(num_vertices):
                graph[i][j] = min(graph[i][j], graph[i][k] + graph[k][j])

        print("Matrix after iteration %d (D[%d]):\n " % (k + 1, k + 1), graph)
        print("\n")

# Get the adjacency matrix from the user
num_vertices, graph = get_adjacency_matrix()

# Apply Floyd-Warshall algorithm
floyd_warshall(graph, num_vertices)
