from shortest import Database, ConnectingFlights, console

if __name__ == '__main__':
    # Initialize the database connection
    db = Database('localhost', 27017, "connecting_flight")
    
    # Initialize the ConnectingFlights object
    cf = ConnectingFlights(db)
    
    # Run the console menu
    console.menu(db, cf)
    
    # Clear the console after exiting the menu
    console.clear()
