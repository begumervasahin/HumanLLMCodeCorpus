from copy import deepcopy
import networkx as nx
import matplotlib.pyplot as plt

def FloydWarshall(graph):
    """
    Applies the Floyd-Warshall algorithm to find the shortest paths between all pairs of vertices in a graph.

    Args:
        graph (list of lists): The adjacency matrix representing the graph.

    Returns:
        list of lists: The shortest path distances between all pairs of vertices.
    """
    # Create a deep copy of the input graph to avoid modifying the original data
    matrix_floyd = deepcopy(graph)
    V = len(graph)

    # Apply the Floyd-Warshall algorithm to find shortest paths
    for k in range(V):
        for i in range(V):
            for j in range(V):
                aux = matrix_floyd[i][k] + matrix_floyd[k][j]
                if matrix_floyd[i][j] > aux:
                    matrix_floyd[i][j] = aux
    return matrix_floyd

def print_matrix(matrix):
    """Prints the contents of a matrix."""
    for row in matrix:
        for item in row:
            print(item, ",", end=" ")
        print("")

def get_input():
    """Prompts the user to enter data for the adjacency matrix."""
    INF = 99999
    print("Create Adjacency Matrix")
    V = int(input("Number of vertices: "))
    graph = []

    for v1 in range(V):
        graph.append([])
        for v2 in range(V):
            if v1 == v2:
                graph[v1].append(0)
            else:
                print("Distance from vertex", v1 + 1, "to", v2 + 1, ": ", end="")
                aux = input()
                if aux == '':
                    distance = INF
                else:
                    distance = int(aux)
                graph[v1].append(distance)
    return graph

# Main function
if __name__ == "__main__":
    # Get input data for the adjacency matrix
    adjacency_matrix = get_input()

    # Apply the Floyd-Warshall algorithm to find shortest paths
    shortest_paths = FloydWarshall(adjacency_matrix)

    # Print the shortest path distances between all pairs of vertices
    print("Shortest path distances:")
    print_matrix(shortest_paths)

    # Create a directed graph using NetworkX
    g = nx.DiGraph()
    for v1 in range(len(adjacency_matrix)):
        for v2 in range(len(adjacency_matrix[v1])):
            weight = adjacency_matrix[v1][v2]
            if weight != 0 and weight != float("inf"):
                g.add_edge(v1, v2, weight=weight)

    # Draw the graph
    nx.draw(g, with_labels=True)
    plt.show()
