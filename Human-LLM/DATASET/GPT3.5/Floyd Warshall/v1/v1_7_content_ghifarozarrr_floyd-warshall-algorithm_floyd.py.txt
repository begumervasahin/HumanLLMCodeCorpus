import numpy as np

def floyd(graph, node, nextt):
    """
    Apply Floyd-Warshall algorithm to find the shortest paths between all pairs of vertices.

    Args:
        graph (ndarray): The adjacency matrix representing the graph.
        node (int): The number of nodes in the graph.
        nextt (ndarray): Matrix to store next vertices for each pair of vertices.

    Returns:
        ndarray: The updated adjacency matrix with shortest distances between all pairs of vertices.
    """
    for k in range(node):
        for i in range(node):
            for j in range(node):
                if graph[i][j] > (graph[i][k] + graph[k][j]):
                    graph[i][j] = graph[i][k] + graph[k][j]
                    nextt[i][j] = nextt[i][k]
    return graph

def path(nextt, source, destination):
    """
    Find the shortest path between two vertices using the next vertex matrix.

    Args:
        nextt (ndarray): Matrix storing next vertices for each pair of vertices.
        source (int): The source vertex.
        destination (int): The destination vertex.

    Returns:
        list: The shortest path between the source and destination vertices.
    """
    path = [source]
    while source != destination:
        source = nextt[source][destination]
        path.append(source)
    return path

def main():
    node = int(input("Number of nodes: "))
    dist_list = np.zeros(shape=(node, node), dtype=np.int)
    nextt = np.zeros(shape=(node, node), dtype=np.int)
    
    # Input distance values
    for i in range(node):
        for j in range(node):
            c = input("Distance from node %d to node %d: " % (i, j))
            if i == j:
                c = 0
            if c == "-":
                c = 9999  # Assuming infinity as 9999
            else:
                c = int(c)
            dist_list[i][j] = c
            nextt[i][j] = j
    
    print("\nIteration 0\n", dist_list)
    
    # Apply Floyd-Warshall algorithm
    new_dist = floyd(dist_list, node, nextt)
    print("\nShortest paths between nodes:")
    
    # Print shortest paths and distances
    for i in range(node):
        for j in range(node):
            if i < j:
                print("Node", i, "and node", j, "are connected through nodes =", 
                      path(nextt, i, j), ", total distance =", new_dist[i][j])

if __name__ == "__main__":
    main()
