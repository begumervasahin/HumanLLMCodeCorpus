INF = float("inf")

# Define the graph as an adjacency matrix
graph = [
    [0, 5, INF, 10],
    [INF, 0, 3, INF],
    [INF, INF, 0, 1],
    [INF, INF, INF, 0]
]

def floyd_warshall(graph):
    """
    Applies the Floyd-Warshall algorithm to find the shortest paths between all pairs of vertices in a graph.

    Args:
    - graph: The adjacency matrix representing the graph.

    Returns:
    - None. Prints the shortest paths between all pairs of vertices.
    """
    num_vertices = len(graph)

    # Apply Floyd-Warshall algorithm to find shortest paths
    for k in range(num_vertices):
        for i in range(num_vertices):
            for j in range(num_vertices):
                # Update the distance if a shorter path is found through vertex k
                graph[i][j] = min(graph[i][j], graph[i][k] + graph[k][j])

    # Print the shortest paths for each pair of vertices
    print("Shortest paths in the graph:")
    for row in graph:
        print("\t".join(str(item) for item in row))

if __name__ == "__main__":
    # Example usage with the provided graph
    floyd_warshall(graph)
