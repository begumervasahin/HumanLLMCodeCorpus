import argparse
import os
import re
import sys
import time
import timeit

class Graph:
    """Graph class to represent the graph."""
    def __init__(self, vertices, edges):
        self.vertices = vertices
        self.edges = edges

# Command-line argument parser
parser = argparse.ArgumentParser(description='Calculate the shortest path between all pairs of vertices in a graph')
parser.add_argument('--algorithm', default='a', help='Algorithm: Select the algorithm to run, default is all. (a)ll, (b)ellman-ford only or (f)loyd-warshall only')
parser.add_argument('-v', '--verbose', action='store_true')
parser.add_argument('--profile', action='store_true')
parser.add_argument('filename', metavar='<filename>', help='Input file containing graph')

# Regular expressions for parsing the graph file
graph_regex = re.compile(r"(\d+)\s(\d+)")
edge_regex = re.compile(r"(\d+)\s(\d+)\s(-?\d+)")

def bellman_ford(graph):
    """Bellman-Ford algorithm for finding shortest paths."""
    path_pairs = []
    for source in range(len(graph.vertices)):
        distances = [float("inf")] * len(graph.vertices)
        distances[source] = 0.0

        for _ in range(len(graph.vertices) - 1):
            for u in range(len(graph.vertices)):
                for v in range(len(graph.vertices)):
                    if distances[v] > distances[u] + graph.edges[u][v]:
                        distances[v] = distances[u] + graph.edges[u][v]

        for u in range(len(graph.vertices)):
            for v in range(len(graph.vertices)):
                if distances[v] > distances[u] + graph.edges[u][v]:
                    return (path_pairs, False)

        path_pairs.append(distances)

    return (path_pairs, True)

def floyd_warshall(graph):
    """Floyd-Warshall algorithm for finding shortest paths."""
    path_pairs = []
    distances = [[float("inf")] * len(graph.vertices) for _ in range(len(graph.vertices))]

    for i in range(len(graph.vertices)):
        for j in range(len(graph.vertices)):
            distances[i][j] = graph.edges[i][j]

    for k in range(len(graph.vertices)):
        for i in range(len(graph.vertices)):
            for j in range(len(graph.vertices)):
                distances[i][j] = min(distances[i][j], distances[i][k] + distances[k][j])

    return distances

def read_graph(filename):
    """Reads the graph file and returns a Graph object."""
    with open(filename, 'r') as file:
        line1 = file.readline()
        graph_match = graph_regex.match(line1)
        if not graph_match:
            print(f"First line of {filename} is not properly formatted")
            sys.exit(1)

        vertices = list(range(int(graph_match.group(1))))
        edges = [[float("inf")] * len(vertices) for _ in range(len(vertices))]

        for line in file.readlines():
            edge_match = edge_regex.match(line.strip())
            if edge_match:
                source, sink, weight = map(int, edge_match.groups())
                if source > len(vertices) or sink > len(vertices):
                    print(f"Attempting to insert an edge between {source} and {sink} in a graph with {len(vertices)} vertices")
                    sys.exit(1)
                edges[source - 1][sink - 1] = weight

    return Graph(vertices, edges)

def matrix_equal(a, b):
    """Check if two matrices are equal."""
    if len(a) != len(b) or len(a[0]) != len(b[0]):
        return False

    for i in range(len(a)):
        for j in range(len(a[0])):
            if a[i][j] != b[i][j]:
                return False

    return True

def main(filename, algorithm):
    """Main function to run the selected algorithm."""
    graph = read_graph(filename)
    path_pairs = []
    no_cycle = True

    if algorithm.lower() == 'b':
        start_time = timeit.default_timer()
        path_pairs, no_cycle = bellman_ford(graph)
        end_time = timeit.default_timer()
        print(end_time - start_time)
    elif algorithm.lower() == 'f':
        start_time = timeit.default_timer()
        path_pairs = floyd_warshall(graph)
        end_time = timeit.default_timer()
        print(end_time - start_time)
    elif algorithm.lower() == 'a':
        print('Running both')
        path_pairs_bellman, no_cycle_bellman = bellman_ford(graph)
        path_pairs_floyd = floyd_warshall(graph)
        path_pairs = path_pairs_bellman
        no_cycle = no_cycle_bellman

        if not matrix_equal(path_pairs_bellman, path_pairs_floyd) or not no_cycle:      
            print('Floyd-Warshall and Bellman-Ford did not produce the same result or negative cycle detected')
    else:
        print("Invalid algorithm option. Please choose 'b', 'f', or 'a'.")

    with open(os.path.splitext(filename)[0]+'_shortestPaths.txt', 'w') as f:
        if not no_cycle:
            f.write("Negative Cycle Detected \n")
        for row in path_pairs:
            for weight in row:
                f.write(str(weight)+' ')
            f.write('\n')

if __name__ == '__main__':
    args = parser.parse_args()
    main(args.filename, args.algorithm)
