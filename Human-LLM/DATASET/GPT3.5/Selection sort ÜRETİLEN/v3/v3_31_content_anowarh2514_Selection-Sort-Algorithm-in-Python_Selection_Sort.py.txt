def selection_sort(nums):
    """
    Sorts a list of numbers using the selection sort algorithm.

    Args:
        nums (list): The list of numbers to be sorted.

    Returns:
        None: The list is sorted in-place.
    """
    # Iterate over the list
    for i in range(len(nums)):
        # Assume the current index has the minimum value
        min_pos = i

        # Find the index of the minimum value in the unsorted portion of the list
        for j in range(i, len(nums)):
            if nums[j] < nums[min_pos]:
                min_pos = j

        # Swap elements if necessary
        nums[i], nums[min_pos] = nums[min_pos], nums[i]
        print(nums)  # Print the current state of the list after each iteration

# Example usage
nums = [5, 3, 7, 2, 4, 1, 11, 8, 10, 9, 6]
print("Original List:", nums)
selection_sort(nums)
print("Sorted List:", nums)
