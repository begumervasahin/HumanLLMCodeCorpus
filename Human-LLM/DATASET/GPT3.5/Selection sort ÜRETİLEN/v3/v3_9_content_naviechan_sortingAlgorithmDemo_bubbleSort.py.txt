class BubbleSort:
    def __init__(self, lst, trace_mode=False):
        """
        Initialize BubbleSort object with the given list and trace mode.

        Args:
            lst (list): The list to be sorted.
            trace_mode (bool, optional): Indicates whether to enable trace mode or not. Defaults to False.
        """
        self.lst = lst
        self.trace_mode = trace_mode

    def sort(self):
        """
        Sorts the list using the bubble sort algorithm.

        Returns:
            list: The sorted list.
        """
        sorted = False
        lst = self.lst
        while not sorted:
            sorted = True
            for i in range(len(lst) - 1):
                j = i + 1
                if lst[i] > lst[j]:
                    lst[i], lst[j] = lst[j], lst[i]
                    sorted = False
                if self.trace_mode:
                    self.display_list(lst, i, j)
        return lst

    @staticmethod
    def display_list(lst, i, j):
        """
        Displays the list with markers (*) at indices i and j.

        Args:
            lst (list): The list to be displayed.
            i (int): Index of the first element to be marked.
            j (int): Index of the second element to be marked.
        """
        lst_with_markers = [f"{elm}*" if k in (i, j) else elm for k, elm in enumerate(lst)]
        output = ' | '.join(map(str, lst_with_markers))
        print(output)
        input("Press Enter to continue...")

def main():
    """
    Main function to demonstrate bubble sort.
    """
    input_list = [64, 25, 12, 22, 11]
    trace_mode = True
    sorter = BubbleSort(input_list, trace_mode)
    sorted_list = sorter.sort()
    print("Sorted array:", sorted_list)

if __name__ == "__main__":
    main()
