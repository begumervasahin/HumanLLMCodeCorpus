def selection_sort(num_list):
    """
    Sorts a list using the selection sort algorithm.

    Args:
        num_list (list): The list of numbers to be sorted.

    Returns:
        None: The list is sorted in place.
    """
    sorted_list = []
    while num_list:
        # Find the minimum value in the remaining list
        min_value = min(num_list)
        
        # Append the minimum value to the sorted list and remove it from the original list
        sorted_list.append(min_value)
        num_list.remove(min_value)
        
        # Print progress
        print_progress(sorted_list, num_list)

    # Print the sorted list
    print("\nSorted list:", sorted_list)

def print_progress(sorted_list, remaining_list):
    """Prints the current progress of the sorting algorithm."""
    print("\nStep", len(sorted_list))
    print("Remaining:", remaining_list)
    print("Sorted:", sorted_list)

# Test the function
numbers = [2, 1, 5, 3, 4, 7, 6, 9, 8, 10]
selection_sort(numbers)
