import random
import timeit
import matplotlib.pyplot as plt

# Global variables
lista = []
num_elementos = [100, 1000, 3000, 6000, 9000, 12000, 15000, 18000, 21000, 24000]
tempo_geracao = []
tempo_ordena = []

# Function to generate a list of random numbers
def gerar_aleatorios(tamanho):
    random.seed()
    return random.sample(range(1, 10 * tamanho + 1), tamanho)

# Function to perform selection sort
def selection_sort(tamanho):
    arr = lista[:tamanho]
    for i in range(len(arr)):
        menor = i
        for j in range(i + 1, len(arr)):
            if arr[j] < arr[menor]:
                menor = j
        arr[i], arr[menor] = arr[menor], arr[i]
    return arr

# Perform timing for list generation and sorting
for i in num_elementos:
    tempo_geracao.append(timeit.timeit("gerar_aleatorios({})".format(i), setup="from __main__ import gerar_aleatorios", number=1))
    tempo_ordena.append(timeit.timeit("selection_sort({})".format(i), setup="from __main__ import selection_sort", number=1))
    print(i)

# Plot the results
plt.plot(num_elementos, tempo_geracao, '*-', label='Tempo de Geração')
plt.plot(num_elementos, tempo_ordena, 'o-', label='Tempo de Ordenação')
plt.ylabel('Tempo(s)')
plt.xlabel('Quantidade de elementos')
plt.legend()
plt.show()
