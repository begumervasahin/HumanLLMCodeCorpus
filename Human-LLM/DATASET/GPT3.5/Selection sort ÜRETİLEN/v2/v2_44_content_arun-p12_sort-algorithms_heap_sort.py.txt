import common as c  # Import the common module for the swap function

def heap_sort(arr, verbose=0, desc=0):
    """
    Sorts the input array using the heap sort algorithm.
    
    Args:
        arr (list): The input list to be sorted.
        verbose (int): Verbosity level for printing intermediate steps (0 for no print, 1 for minimal, 2 for detailed).
        desc (int): Flag indicating whether to sort in descending order (0 for ascending, 1 for descending).
    
    Returns:
        list: The sorted array.
    """
    def heapify(i, size):
        """
        Heapify the subtree rooted at index i.
        
        Args:
            i (int): Index of the root of the subtree.
            size (int): Size of the array.
        """
        parent = i
        left_child = i * 2 + 1
        right_child = left_child + 1
        
        # Find the index of the largest element among parent and its children
        if left_child < size and arr[left_child] > arr[parent]:
            parent = left_child
        if right_child < size and arr[right_child] > arr[parent]:
            parent = right_child
        
        if verbose == 2:
            print("  Subtree:", i, size, " :: ", arr)
        
        # Swap parent with the largest child if necessary and recursively heapify the affected subtree
        if i != parent:
            arr[i], arr[parent] = c.swap(arr[i], arr[parent])
            heapify(parent, size)

    def sort():
        """
        Perform heap sort on the array.
        """
        size = len(arr)
        
        # Build max heap by heapifying each subtree
        for i in range((size // 2) - 1, -1, -1):
            heapify(i, size)
            if verbose:
                print("Iteration")
        
        # Extract elements from the heap one by one
        for i in range(size - 1, 0, -1):
            arr[0], arr[i] = c.swap(arr[0], arr[i])
            heapify(0, i)
            if verbose:
                print("Iteration")

    # Call the sorting function
    sort()
    
    # Reverse the array if sorting in descending order
    if desc:
        arr = arr[::-1]
    
    return arr
