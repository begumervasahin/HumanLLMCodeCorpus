def buildfields(s):
    fields = ()
    match = re.match('^(\S+)\s+(\S+)\s+(\S+)\s+(\S+)', s)
    if match:
        fields = (match.group(1), match.group(2), match.group(3), match.group(4))
    else:
        match = re.match('^(\S+)\s+(\S+)\s+(\S+)', s)
        if not match:
            print('broken 1')
            os._exit(0)
        else:
            fields = (match.group(1), match.group(2), match.group(3), '')
    if len(fields) != 4:
        print('broken 2')
        os._exit(0)
    return fields

def buildkey(fields):
    key = fields[0] + ' ' + fields[1] + ' ' + fields[2]
    return key
