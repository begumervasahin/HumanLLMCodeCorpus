from random import randint
from time import time
from tkinter import *

# Function to perform selection sort
def selection_sort(values):
    for i in range(len(values)):
        min_pos = minimum_position(values, i)
        temp = values[min_pos]
        values[min_pos] = values[i]
        values[i] = temp

# Function to find the position of the minimum value in a list
def minimum_position(values, start):
    min_pos = start
    for i in range(start + 1, len(values)):
        if values[i] < values[min_pos]:
            min_pos = i
    return min_pos

# Main function to perform sorting and display results
def main():
    print()
    nmin = int(input("Enter the minimum list size: "))
    print()
    nmax = int(input("Enter the maximum list size: "))
    print()
    runs = int(input("Enter the number of different measurements to run: "))
    
    n = nmin
    table_array = []

    # Run the sorting algorithm for different list sizes
    for i in range(runs):
        values = [randint(1, 1000) for _ in range(n)]
        start_time = time()
        selection_sort(values)
        end_time = time()
        table_array.append([n, round((end_time - start_time), 3)])
        n = int(n + ((nmax - nmin) / (runs - 1))) - int(n + ((nmax - nmin) / (runs - 1))) % 5
    
    # Create a GUI window to display results
    window = Tk()
    window.title("Results of Sample Runs - Selection Sort")
    
    # Labels for table headers
    t1 = Label(text='Sort Size')
    t2 = Label(text='Seconds to Sort')
    t1.grid(row=0, column=1, sticky=NSEW)
    t2.grid(row=0, column=2, sticky=NSEW)
    
    # Display results in the table
    for i in range(runs):
        l1 = Label(text='%.0f' % table_array[i][0], relief=RIDGE, width=20, height=2)
        l2 = Label(text='%.3f' % table_array[i][1], relief=RIDGE, width=20, height=2)
        l1.grid(row=i + 1, column=1, sticky=NSEW)
        l2.grid(row=i + 1, column=2, sticky=NSEW)
    
    window.mainloop()

# Call the main function to execute the program
main()
