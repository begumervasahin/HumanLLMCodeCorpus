import time

# Line separator
line_separator = '_' * 80

# Title of the program
title = "Sort techniques on dataset"

# Print title
print(line_separator)
print(title.center(80))
print(line_separator + "\n\n")

# Main loop for sorting techniques
while True:
    # Menu for sorting techniques
    print("Enter \n 1 for bubble sort \n 2 for insertion sort \n 3 for merge sort \n 4 for selection sort")
    choice = int(input("\nEnter your choice: "))

    # Bubble sort
    if choice == 1:
        start_time = time.time()

        # Bubble sort algorithm
        def bubble_sort(lst):
            for j in range(len(lst) - 1, 0, -1):
                for i in range(j):
                    if lst[i] > lst[i + 1]:
                        temp = lst[i]
                        lst[i] = lst[i + 1]
                        lst[i + 1] = temp

        # Input list from user
        lst = []
        num_count = int(input("\nHow many numbers do you want to enter? "))
        for i in range(num_count):
            num = int(input("Enter the number: "))
            lst.append(num)

        # Sort the list
        bubble_sort(lst)
        print("\nThe list after bubble sorting is:", lst)

        # Print sorting time
        end_time = time.time()
        print("The time taken by sorting process:", end_time - start_time)

    # Insertion sort
    elif choice == 2:
        start_time = time.time()

        # Insertion sort algorithm
        def insertion_sort(lst):
            for i in range(1, len(lst)):
                current_value = lst[i]
                position = i
                while position > 0 and lst[position - 1] > current_value:
                    lst[position] = lst[position - 1]
                    position -= 1
                lst[position] = current_value

        # Input list from user
        lst = []
        num_count = int(input("\nHow many numbers do you want to enter? "))
        for i in range(num_count):
            num = int(input("Enter the number: "))
            lst.append(num)

        # Sort the list
        insertion_sort(lst)
        print("\nThe list after insertion sorting is:", lst)

        # Print sorting time
        end_time = time.time()
        print("The time taken by sorting process:", end_time - start_time)

    # Merge sort
    elif choice == 3:
        start_time = time.time()

        # Merge sort algorithm
        def merge_sort(alist):
            if len(alist) > 1:
                mid = len(alist) // 2
                lefthalf = alist[:mid]
                righthalf = alist[mid:]
                merge_sort(lefthalf)
                merge_sort(righthalf)
                i = j = k = 0
                while i < len(lefthalf) and j < len(righthalf):
                    if lefthalf[i] < righthalf[j]:
                        alist[k] = lefthalf[i]
                        i += 1
                    else:
                        alist[k] = righthalf[j]
                        j += 1
                    k += 1
                while i < len(lefthalf):
                    alist[k] = lefthalf[i]
                    i += 1
                    k += 1
                while j < len(righthalf):
                    alist[k] = righthalf[j]
                    j += 1
                    k += 1

        # Input list from user
        alist = []
        num_count = int(input("\nHow many numbers do you want to enter? "))
        for i in range(num_count):
            num = int(input("Enter the number: "))
            alist.append(num)

        # Sort the list
        merge_sort(alist)
        print("\nThe list after merge sorting is:", alist)

        # Print sorting time
        end_time = time.time()
        print("The time taken by sorting process:", end_time - start_time)

    # Selection sort
    elif choice == 4:
        start_time = time.time()

        # Selection sort algorithm
        def selection_sort(lst):
            for fillslot in range(len(lst) - 1, 0, -1):
                position_of_max = 0
                for location in range(1, fillslot + 1):
                    if lst[location] > lst[position_of_max]:
                        position_of_max = location
                temp = lst[fillslot]
                lst[fillslot] = lst[position_of_max]
                lst[position_of_max] = temp

        # Input list from user
        lst = []
        num_count = int(input("\nHow many numbers do you want to enter? "))
        for i in range(num_count):
            num = int(input("Enter the number: "))
            lst.append(num)

        # Sort the list
        selection_sort(lst)
        print("\nThe list after selection sorting is:", lst)

        # Print sorting time
        end_time = time.time()
        print("The time taken by sorting process:", end_time - start_time)

    # Ask user if they want to continue or exit
    continue_option = int(input("\nEnter 1 to continue and 0 to exit: "))
    print("\n\n")

    # Check if user wants to exit
    if continue_option == 0:
        break
