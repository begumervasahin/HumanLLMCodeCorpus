class BubbleSort:
    def __init__(self, lst, trace_mode):
        """
        Initialize the BubbleSort object with the given list and trace mode.

        Parameters:
            lst (list): The list to be sorted.
            trace_mode (bool): Flag indicating whether to display the sorting process.
        """
        self.lst = lst
        self.trace_mode = trace_mode

    def sort(self):
        """
        Sorts the list using the bubble sort algorithm and optionally displays the sorting process.
        
        Returns:
            list: The sorted list.
        """
        sorted = False
        lst = self.lst
        while not sorted:
            sorted = True
            for i in range(len(lst) - 1):
                j = i + 1
                if lst[i] > lst[j]:
                    lst[i], lst[j] = lst[j], lst[i]
                    sorted = False
                if self.trace_mode:
                    self.display_list(lst, i, j)
        return lst

    def display_list(self, lst, i, j):
        """
        Display the list with markers (*) indicating the elements being compared.

        Parameters:
            lst (list): The list to be displayed.
            i (int): Index of the first element being compared.
            j (int): Index of the second element being compared.
        """
        lst_str = [str(k) for k in lst]
        output = '|'
        for k, elm in enumerate(lst_str):
            if k == i or k == j:
                elm = elm + '*'
            output = output + ' ' + str(elm) + ' |'
        print(output)
        input("Press Enter to continue...")
