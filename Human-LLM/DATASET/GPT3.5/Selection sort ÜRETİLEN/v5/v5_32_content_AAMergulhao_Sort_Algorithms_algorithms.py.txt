def quick_sort(lst):
    """
    Sorts the given list using the quick sort algorithm.

    Args:
        lst (list): The list of elements to be sorted.

    Returns:
        list: The sorted list.
    """
    if len(lst) <= 1:
        return lst
    pivot = lst[0]
    equals = [x for x in lst if x == pivot]
    less = [x for x in lst if x < pivot]
    greater = [x for x in lst if x > pivot]
    return quick_sort(less) + equals + quick_sort(greater)


def selection_sort(lst):
    """
    Sorts the given list using the selection sort algorithm.

    Args:
        lst (list): The list of elements to be sorted.

    Returns:
        list: The sorted list.
    """
    sorted_list = []
    temp = lst[:]
    while temp:
        min_val = min(temp)
        sorted_list.append(min_val)
        temp.remove(min_val)
    return sorted_list


def merge(left, right):
    """
    Merges two sorted lists into a single sorted list.

    Args:
        left (list): The left sorted list.
        right (list): The right sorted list.

    Returns:
        list: The merged sorted list.
    """
    merged = []
    i, j = 0, 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            merged.append(left[i])
            i += 1
        else:
            merged.append(right[j])
            j += 1
    merged += left[i:]
    merged += right[j:]
    return merged


def merge_sort(lst):
    """
    Sorts the given list using the merge sort algorithm.

    Args:
        lst (list): The list of elements to be sorted.

    Returns:
        list: The sorted list.
    """
    if len(lst) <= 1:
        return lst
    else:
        middle = len(lst) // 2
        left = merge_sort(lst[:middle])
        right = merge_sort(lst[middle:])
        return merge(left, right)
