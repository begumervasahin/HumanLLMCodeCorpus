import sys
import argparse
from quickSort import QuickSort
from bubbleSort import BubbleSort
from selectionSort import SelectionSort
from mergeSort import MergeSort

class_names = {'bubble': BubbleSort, 'selection': SelectionSort, 'merge': MergeSort, 'quick': QuickSort}

def display_list(lst):
    """Display the list in a formatted table."""
    lst_str = ' | '.join(str(i) for i in lst)
    output = f"| {lst_str} |" if lst else "| |"
    print(output)
    input("Press Enter to continue...")

def main(argv):
    """Main function to execute the sorting algorithm based on the input arguments."""
    trace_mode = False
    parser = argparse.ArgumentParser()
    parser.add_argument("input_list", nargs='+', type=int, help="List of integers separated by space")
    parser.add_argument("alg", type=str, choices=class_names.keys(), help="Choose from bubble, selection, merge, or quick")
    parser.add_argument("-t", "--trace", action="store_true", help="Print the sorting algorithm step-by-step")
    args = parser.parse_args()

    if args.trace:
        trace_mode = True

    print('Input:', args.input_list)
    sort_algorithm = class_names[args.alg](args.input_list, trace_mode)
    sorted_list = sort_algorithm.sort()
    print('Result:', sorted_list)

if __name__ == '__main__':
    main(sys.argv[1:])
