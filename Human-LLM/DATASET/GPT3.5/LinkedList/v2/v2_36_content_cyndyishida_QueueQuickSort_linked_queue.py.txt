class Node:
    """A class for storing a singly linked node."""
    
    def __init__(self, value, next_node):
        """Initialize a node with a value and a reference to the next node."""
        self.value = value
        self.next = next_node

    def __lt__(self, other):
        """Compare node values."""
        return self.value <= other.value

    def __le__(self, other):
        """Compare node values."""
        return self.value <= other.value

class LinkedQueue:
    """FIFO queue implementation using a singly linked list for storage."""

    def __init__(self):
        """Create an empty queue."""
        self.head = None
        self.tail = None
        self.size = 0

    def __str__(self):
        """Return a string representation of the queue."""
        current = self.head
        values = []
        while current:
            values.append(str(current.value))
            current = current.next
        return ", ".join(values)

    __repr__ = __str__

    def __len__(self):
        """Return the number of elements in the queue."""
        return self.size

    def is_empty(self):
        """Check if the queue is empty."""
        return self.size == 0

    def dequeue(self):
        """Remove and return the first element of the queue (i.e., FIFO)."""
        if self.is_empty():
            raise IndexError("Queue is empty")
        value = self.head.value
        self.head = self.head.next
        self.size -= 1
        if self.is_empty():
            self.tail = None
        return value

    def enqueue(self, element):
        """Add an element to the back of the queue."""
        new_node = Node(element, None)
        if self.is_empty():
            self.head = new_node
        else:
            self.tail.next = new_node
        self.tail = new_node
        self.size += 1

    def __getitem__(self, index):
        """Get the value at the specified index."""
        if not (0 <= index < self.size):
            raise IndexError("Index out of range")
        current = self.head
        for _ in range(index):
            current = current.next
        return current.value

    def __setitem__(self, index, value):
        """Set the value of the item at the specified index."""
        if not (0 <= index < self.size):
            raise IndexError("Index out of range")
        current = self.head
        for _ in range(index):
            current = current.next
        current.value = value

# Example usage
if __name__ == "__main__":
    queue = LinkedQueue()
    queue.enqueue(1)
    queue.enqueue(2)
    queue.enqueue(3)
    print("Queue:", queue)
    print("Length:", len(queue))
    print("Dequeue:", queue.dequeue())
    print("Queue after dequeue:", queue)
    queue[0] = 5
    print("Queue after setting value at index 0:", queue)
