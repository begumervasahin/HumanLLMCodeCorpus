import urllib.request
from bs4 import BeautifulSoup
import ssl

def main():
    # Ignore SSL certificate errors
    ssl_context = ssl.create_default_context()
    ssl_context.check_hostname = False
    ssl_context.verify_mode = ssl.CERT_NONE

    # Prompt user for URL, count, and position
    url = input('Enter URL: ')
    count = int(input('Enter count: '))
    position = int(input('Enter position: '))

    # Loop for the specified count
    for _ in range(count):
        print('Retrieving:', url)
        
        # Open and read the HTML content of the URL
        html = urllib.request.urlopen(url, context=ssl_context).read()
        
        # Parse the HTML using BeautifulSoup
        soup = BeautifulSoup(html, 'html.parser')
        
        # Find all anchor tags
        anchor_tags = soup('a')
        
        # Get the URL at the specified position
        url = anchor_tags[position - 1].get('href', None)

if __name__ == '__main__':
    main()
