class Heap:
    def __init__(self):
        self.heap_list = []
        self.current_size = 0

    def left_child(self, i):
        left_index = 2 * i + 1
        if left_index < self.current_size:
            return left_index
        return None

    def right_child(self, i):
        right_index = 2 * i + 2
        if right_index < self.current_size:
            return right_index
        return None

    def max_heapify(self, i):
        largest = i
        left = self.left_child(i)
        right = self.right_child(i)

        if left is not None and self.heap_list[left] > self.heap_list[largest]:
            largest = left
        if right is not None and self.heap_list[right] > self.heap_list[largest]:
            largest = right

        if largest != i:
            self.heap_list[i], self.heap_list[largest] = self.heap_list[largest], self.heap_list[i]
            self.max_heapify(largest)

    def build_heap(self, array):
        self.current_size = len(array)
        self.heap_list = list(array)
        for i in range(self.current_size // 2, -1, -1):
            self.max_heapify(i)

    def get_max(self):
        if self.current_size >= 1:
            max_element = self.heap_list[0]
            self.heap_list[0], self.heap_list[self.current_size - 1] = self.heap_list[self.current_size - 1], self.heap_list[0]
            self.current_size -= 1
            self.max_heapify(0)
            return max_element
        return None

    def heap_sort(self):
        size = self.current_size
        while self.current_size - 1 >= 0:
            self.heap_list[0], self.heap_list[self.current_size - 1] = self.heap_list[self.current_size - 1], self.heap_list[0]
            self.current_size -= 1
            self.max_heapify(0)
        self.current_size = size

    def insert(self, data):
        self.heap_list.append(data)
        curr = self.current_size
        self.current_size += 1
        while curr > 0 and self.heap_list[curr] > self.heap_list[(curr - 1) // 2]:
            self.heap_list[curr], self.heap_list[(curr - 1) // 2] = self.heap_list[(curr - 1) // 2], self.heap_list[curr]
            curr = (curr - 1) // 2

    def display(self):
        print(self.heap_list)


def main():
    print("=== Heap (max) - A specialized tree-based data structure that satisfies the heap property (max-min) which is in max heap, the keys of parent nodes are always greater than or equal to those of the children and the highest key is in the root node")
    sentence = input("[STR] What is your sentence? ").split()
    heap = Heap()
    heap.build_heap(sentence)
    heap.heap_sort()
    heap.display()


if __name__ == '__main__':
    main()
