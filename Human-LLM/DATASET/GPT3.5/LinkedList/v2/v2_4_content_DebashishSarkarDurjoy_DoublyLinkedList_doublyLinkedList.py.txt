class Node:
    def __init__(self, value):
        self.value = value  # Value stored in the node
        self.next = None  # Reference to the next node
        self.prev = None  # Reference to the previous node

class DoublyLinkedList:
    def __init__(self, value):
        # Initialize the doubly linked list with a single node as both head and tail
        self.head = Node(value)
        self.tail = self.head

    # Method to add a new node at the end of the doubly linked list
    def add_node_last(self, value):
        current = self.head
        # Traverse to the last node
        while current.next is not None:
            current = current.next
        # Create a new node with the given value
        new_node = Node(value)
        # Set the new node as the next node of the current last node
        current.next = new_node
        # Set the previous node of the new node to the current last node
        new_node.prev = current
        # Update the tail to point to the new last node
        self.tail = new_node

    # Method to insert a new node after a node with a specific value
    def insert_node(self, target_value, new_value):
        # If the target value is the value of the tail node, add the new node after the tail
        if self.tail.value == target_value:
            self.add_node_last(new_value)
        # If the target value is the value of the head node, add the new node after the head
        elif self.head.value == target_value:
            new_node = Node(new_value)
            new_node.next = self.head.next
            new_node.prev = self.head
            new_node.next.prev = new_node
            self.head.next = new_node
        else:
            current = self.head.next
            # Traverse to find the node with the target value
            while current.value != target_value:
                current = current.next
            # Create a new node with the given value
            new_node = Node(new_value)
            # Insert the new node between the current node and its next node
            new_node.next = current.next
            new_node.next.prev = new_node
            new_node.prev = current
            current.next = new_node

    # Method to remove a node with a specific value from the doubly linked list
    def remove_node(self, target_value):
        # If the target value is the value of the head node
        if self.head.value == target_value:
            self.head = self.head.next
            self.head.prev = None
        # If the target value is the value of the tail node
        elif self.tail.value == target_value:
            self.tail = self.tail.prev
            self.tail.next = None
        else:
            current = self.head.next
            # Traverse to find the node with the target value
            while current.value != target_value:
                current = current.next
            # Remove the current node from the doubly linked list
            current.prev.next = current.next
            current.next.prev = current.prev

    # Method to display the elements of the doubly linked list in reverse order
    def show_reverse(self):
        current = self.tail
        while current is not None:
            print(current.value)
            current = current.prev

    # Method to display the elements of the doubly linked list
    def show(self):
        current = self.head
        while current is not None:
            print(current.value)
            current = current.next
