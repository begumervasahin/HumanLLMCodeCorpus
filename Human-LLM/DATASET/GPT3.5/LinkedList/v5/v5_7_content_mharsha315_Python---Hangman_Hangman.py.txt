import random
import time

class Node:
    def __init__(self, data="-", data_check="-"):
        self.data = data
        self.data_check = data_check
        self.next = None

class HangmanGame:
    def __init__(self):
        self.head = Node()

    def insert_word(self, word):
        current_node = self.head
        for letter in word:
            new_node = Node(letter)
            current_node.next = new_node
            current_node = current_node.next

    def display_word(self):
        current_node = self.head.next
        print()
        while current_node is not None:
            print(current_node.data_check, end="")
            current_node = current_node.next
        print()

    def play(self, letter, hangman):
        found_letter = False
        current_node = self.head.next
        while current_node is not None:
            if current_node.data == letter:
                current_node.data_check = letter
                found_letter = True
            current_node = current_node.next
        if not found_letter:
            hangman.pop(0)
        self.display_word()
        print(hangman)

    def view_answer(self):
        current_node = self.head.next
        print("ANSWER IS...")
        time.sleep(2.0)
        while current_node is not None:
            print(current_node.data, end="")
            current_node = current_node.next

def play_hangman():
    game = HangmanGame()
    print("Type 'view' to view answer")
    print("Type 'exit' to EXIT")
    words = ["python", "jumble", "easy", "difficult", "computer", "hangman", "failure", "brilliant", "worthy",
             "xylophone", "awkward", "gypsy", "jinx", "burglar", "bankrupt", "crisis", "hyphen", "memento",
             "mystery", "pajama", "pixel", "rogue", "rhythmic", "twelfth", "jealous", "zombie", "yacht", "yak",
             "zippy", "unknown", "battleground", "player", "psycho", "beast", "buzzard", "boycott", "coffin",
             "witchcraft", "rickshaw", "mnemonic", "pneumonia", "peekaboo", "diarrhoea", "jaundice", "gossip",
             "despacito"]
    selected_word = random.choice(words)
    game.insert_word(selected_word)
    game.display_word()
    option = None
    hangman = ["H", "A", "N", "G", "M", "A", "N"]
    while option not in ["exit", "view"] and len(hangman) != 0:
        option = input()
        if option != "exit" and option != "view":
            game.play(option, hangman)
        elif option == "view":
            game.view_answer()
            break
        elif option == "exit":
            print("Exiting...")
            break

play_hangman()

def ask_play_again():
    print()
    print("1. PLAY AGAIN")
    return input("Do you want to play again? (yes/no): ").strip().lower()

while True:
    play_again = ask_play_again()
    if play_again in ["yes", "y"]:
        play_hangman()
    else:
        break
