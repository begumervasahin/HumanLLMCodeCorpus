class LinkedListNode:
    """Class to represent a node in a doubly linked list."""

    def __init__(self, data, post=None, pred=None):
        """Initialize a node with data and optional pointers to the next and previous nodes."""
        self.data = data
        self.post = post
        self.pred = pred

    def remove(self, data):
        """Remove a node with the given data from the linked list."""
        if self.data == data:
            if self.pred is None:
                if self.post:
                    # Update the data and post pointer of the first node
                    self.data = self.post.data
                    self.post = self.post.post
                else:
                    # If there's only one node in the list
                    self.data = None
                return 0
            elif self.post is None:
                # If the node to be removed is the last node
                self.pred.post = None
                return 0
            else:
                # If the node to be removed is in the middle
                self.pred.post = self.post
                return 0
        elif self.post:
            # Continue searching in the next node
            return self.post.remove(data)
        else:
            # Data not found
            return -1

    def insert(self, data):
        """Insert a new node with the given data in sorted order."""
        if data < self.data:
            # Insert before the current node
            hold = LinkedListNode(data, self, None)
            self = hold
        elif self.post is None:
            # Insert after the current node if it's the last one
            hold = LinkedListNode(data, None, self)
            self.post = hold
        else:
            # Recursively insert
            self.post.insert(data)

    def contains(self):
        """Print the contents of the linked list."""
        current = self
        if current.data is None:
            print("Empty.")
        else:
            while current:
                print(current.data, end=", ")
                current = current.post
            print()
