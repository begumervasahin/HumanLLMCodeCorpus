import tensorflow as tf
import numpy as np

def generate_random_data(size=100):
    """Generate random data for demonstration."""
    xd = np.random.rand(size).astype(np.float32)
    yd = xd * 0.1 + 0.3
    return xd, yd

def build_neural_network(x):
    """Build the neural network."""
    with tf.name_scope('layer'):
        with tf.name_scope('weights'):
            weights = tf.Variable(tf.random_uniform([1], -1.0, 1.0), name="W")
            tf.summary.histogram('weights', weights)
        with tf.name_scope('biases'):
            biases = tf.Variable(tf.zeros([1]), name="b")
            tf.summary.histogram('biases', biases)
        with tf.name_scope('Wx_add_b'):
            y = weights * x + biases
    return y

def train_neural_network(xs, ys, y):
    """Train the neural network."""
    with tf.name_scope('loss'):
        loss = tf.reduce_mean(tf.square(y - ys))
        tf.summary.scalar('loss', loss)

    with tf.name_scope('train'):
        optimizer = tf.train.GradientDescentOptimizer(0.5)
        train = optimizer.minimize(loss)
    return train, loss

def save_model(sess, saver, save_path):
    """Save the trained model."""
    save_path = saver.save(sess, save_path)
    print("Model saved in path: %s" % save_path)

def main():
    # Generate random data
    xd, yd = generate_random_data()

    # Define input placeholders
    with tf.name_scope('inputs'):
        xs = tf.placeholder(tf.float32, name='x_input')
        ys = tf.placeholder(tf.float32, name='y_input')

    # Build the neural network
    y = build_neural_network(xs)

    # Train the neural network
    train, loss = train_neural_network(xs, ys, y)

    # Initialize variables
    init = tf.global_variables_initializer()
    saver = tf.train.Saver()

    # Start a TensorFlow session
    with tf.Session() as sess:
        # Merge all summaries
        merged = tf.summary.merge_all()
        writer = tf.summary.FileWriter("logs/", sess.graph)

        # Initialize variables
        sess.run(init)

        # Training loop
        for step in range(2017):
            sess.run(train, feed_dict={xs: xd, ys: yd})
            if step % 20 == 0:
                result = sess.run(merged, feed_dict={xs: xd, ys: yd})
                writer.add_summary(result, step)
                print(step, sess.run(weights), sess.run(biases))

        # Make predictions
        feed_dict = {xs: [10, 20, 40]}
        prediction = sess.run(y, feed_dict)
        print(prediction)

        # Save the model
        save_model(sess, saver, "output/save.ckpt")

if __name__ == "__main__":
    main()
