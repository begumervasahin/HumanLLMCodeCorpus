class Node:
    def __init__(self, data, next_node=None):
        self.data = data
        self.next_node = next_node

    def __str__(self):
        return str(self.data)


class MyLinkedList:
    def __init__(self):
        self.head = None
        self.tail = None
        self.size = 0

    def __len__(self):
        return self.size

    def add(self, data):
        new_node = Node(data)
        if not self.head:
            self.head = new_node
        else:
            self.tail.next_node = new_node
        self.tail = new_node
        self.size += 1

    def get(self, index):
        if index < 0 or index >= self.size:
            raise IndexError('Index out of bounds')
        current_node = self.head
        for _ in range(index):
            current_node = current_node.next_node
        return current_node.data


# Example usage:
list1 = MyLinkedList()
list1.add(1)
list1.add(5)
list1.add(-7)
for i in range(len(list1)):
    print(list1.get(i))
