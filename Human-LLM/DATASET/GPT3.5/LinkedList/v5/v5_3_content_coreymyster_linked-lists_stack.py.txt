class Node:
    """Class to represent a node in a linked list."""
    def __init__(self, data):
        """Initialize a node with the given data."""
        self.data = data
        self.next = None

    def get_data(self):
        """Return the data stored in the node."""
        return self.data

    def set_data(self, new_data):
        """Set the data of the node to the given value."""
        self.data = new_data

    def get_next(self):
        """Return the reference to the next node."""
        return self.next

    def set_next(self, new_next):
        """Set the reference to the next node."""
        self.next = new_next


class UnorderedList:
    """Class representing an unordered linked list."""
    def __init__(self):
        """Initialize an empty linked list."""
        self.head = None

    def is_empty(self):
        """Check if the linked list is empty."""
        return self.head is None

    def add_first(self, item):
        """Add a new node with the given item at the beginning of the linked list."""
        new_node = Node(item)
        new_node.set_next(self.head)
        self.head = new_node

    def size(self):
        """Return the number of nodes in the linked list."""
        current = self.head
        count = 0
        while current:
            count += 1
            current = current.get_next()
        return count

    def search(self, item):
        """
        Search for the given item in the linked list.

        Returns:
            bool: True if the item is found, False otherwise.
        """
        current = self.head
        while current:
            if current.get_data() == item:
                return True
            current = current.get_next()
        return False

    def remove_first(self):
        """
        Remove and return the first item from the linked list.

        Returns:
            Any: The value of the removed item.
        """
        if self.is_empty():
            return None
        removed_item = self.head.get_data()
        self.head = self.head.get_next()
        return removed_item


# Example usage
mylist = UnorderedList()
mylist.add_first(80)
print(mylist.size())  # Output: 1
mylist.add_first(3)
mylist.add_first(67)
mylist.add_first(15)
print(mylist.size())  # Output: 4
print(mylist.search(15))  # Output: True
mylist.remove_first()
print(mylist.size())  # Output: 3
print(mylist.search(15))  # Output: False
mylist.add_first(15)
print(mylist.size())  # Output: 4
