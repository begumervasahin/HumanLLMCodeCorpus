class Node:
    def __init__(self, val=None):
        self.val = val
        self.nextval = None

class LinkedList:
    def __init__(self):
        self.headval = None

    def addEnd(self, endData):
        newNode = Node(endData)
        if self.headval is None:
            self.headval = newNode
            return
        last = self.headval
        while last.nextval:
            last = last.nextval
        last.nextval = newNode

    def removeNode(self, key):
        headVal = self.headval
        if headVal is not None:
            if headVal.val == key:
                self.head = headVal.nextval
                headVal = None
                return
        while headVal is not None:
            if headVal.val == key:
                break
            prev = headVal
            headVal = headVal.nextval
        if headVal is None:
            return
        prev.nextval = headVal.nextval
        headVal = None

    def printlist(self):
        printval = self.headval
        while printval is not None:
            print(printval.val)
            printval = printval.nextval

def nodeGen(lst):
    """Generate nodes from the list items."""
    try:
        for i in range(2, len(lst) + 1):
            print("d" + str(i) + " = Node('" + lst[i - 1] + "')")
    except (TypeError, ValueError):
        print("Usage: nodeGen(list)\nArgs:\nlist = [] any list")

def nextGen(lst):
    """Generate the code for adding Node.nextval in the range of the list."""
    dlist = []
    try:
        for i in range(2, len(lst) + 1):
            dlist.append("d" + str(i))
        for d in range(len(dlist) - 1):
            print(dlist[d] + ".nextval = " + dlist[d + 1])
    except (TypeError, ValueError):
        print("Usage: nextGen(list)\nArgs:\nlist = [] any list")
