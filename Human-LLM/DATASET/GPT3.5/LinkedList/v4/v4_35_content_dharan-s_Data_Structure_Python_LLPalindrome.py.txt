class Node:
    """
    Class to represent a node in a linked list.
    """

    def __init__(self, value):
        """
        Initialize a node with the given value.

        Parameters:
        value: The value to be stored in the node.
        """
        self.info = value
        self.next = None


class Pointer:
    """
    Class to manage a linked list and perform various operations on it.
    """

    def __init__(self):
        """
        Initialize the linked list with head and tail pointers.
        """
        self.head = None
        self.tail = None

    def insert_at_end(self, data):
        """
        Insert a new node with the given data at the end of the linked list.

        Parameters:
        data: The data to be stored in the new node.
        """
        temp = Node(data)
        if self.head is None:
            self.head = temp
            self.tail = temp
        else:
            self.tail.next = temp
            self.tail = temp

    def insert_at_end_notail(self, data):
        """
        Insert a new node with the given data at the end of the linked list without using the tail pointer.

        Parameters:
        data: The data to be stored in the new node.
        """
        temp = Node(data)
        if self.head is None:
            self.head = temp
        else:
            p = self.head
            while p.next is not None:
                p = p.next
            p.next = temp

    def display(self):
        """
        Display the elements of the linked list.
        """
        if self.head is None:
            print("List is empty")
        else:
            p = self.head
            while p is not None:
                print(p.info, " ", end='')
                p = p.next
            print()

    def create(self):
        """
        Create a linked list by taking input from the user.
        """
        num_nodes = int(input("Enter the number of nodes: "))
        for _ in range(num_nodes):
            data = int(input("Enter data to be inserted: "))
            self.insert_at_end_notail(data)

    def tail_element(self):
        """
        Print the value of the tail element of the linked list.
        """
        if self.head is not None:
            print(self.tail.info)

    def middle_last_element(self):
        """
        Print the middle and last elements of the linked list.
        """
        if self.head is not None:
            p1 = self.head
            p2 = self.head
            while p2.next:
                p1 = p1.next
                p2 = p2.next
                if p2.next is not None:
                    p2 = p2.next
            print("Middle element:", p1.info)
            print("Last element:", p2.info)

    def reverse(self):
        """
        Reverse the linked list.
        """
        if self.head and self.head.next:
            prev = None
            p = self.head
            while p is not None:
                next_node = p.next
                p.next = prev
                prev = p
                p = next_node
            self.head = prev

    def palindrome(self):
        """
        Check if the linked list is a palindrome.
        """
        if self.head and self.head.next:
            p = self.head
            self.reverse()
            q = self.head
            while p and q:
                if p.info != q.info:
                    print("Not a palindrome")
                    return
                p = p.next
                q = q.next
            print("Palindrome")
        elif self.head is None:
            print("List is Empty")
        else:
            print("Palindrome")
