class Node:
    def __init__(self, data="-", data_check="-"):
        self.data = data
        self.data_check = data_check
        self.next = None

class HangmanGame:
    def __init__(self):
        self.head = Node()

    def insert_word(self, word):
        n = len(word)
        pos = self.head
        i = 0
        while i < n:
            new_node = Node(word[i])
            pos.next = new_node
            pos = pos.next
            i += 1

    def display_word(self):
        pos = self.head.next
        print()
        while pos is not None:
            print(pos.data_check, end="")
            pos = pos.next
        print()

    def play(self, letter, hangman):
        flag = 0
        pos = self.head.next
        while pos is not None:
            if pos.data == letter:
                pos.data_check = letter
                flag = 1
            pos = pos.next
        if flag == 0:
            hangman.pop(0)
        pos = self.head.next
        while pos is not None:
            print(pos.data_check, end="")
            pos = pos.next
        print(hangman)

    def view_answer(self):
        import time
        pos = self.head.next
        print("ANSWER IS...")
        time.sleep(2.0)
        while pos is not None:
            print(pos.data, end="")
            pos = pos.next

def play_hangman():
    game = HangmanGame()
    print("Type 'view' to view answer")
    print("Type 'exit' to EXIT")
    words = ["python", "jumble", "easy", "difficult", "computer", "hangman", "failure", "brilliant", "worthy",

             "witchcraft", "rickshaw", "mnemonic", "pneumonia", "peekaboo", "diarrhoea", "jaundice", "gossip", 
             "despacito"]
    import random
    selected_word = random.choice(words)
    game.insert_word(selected_word)
    game.display_word()
    option = None
    hangman = ["H", "A", "N", "G", "M", "A", "N"]
    while (option != "exit" and option != "view") and len(hangman) != 0:
        option = input()
        if option != "exit" and option != "view":
            game.play(option, hangman)
        elif option == "view":
            game.view_answer()
            break
        elif option == "exit":
            print("Exiting...")
            break
