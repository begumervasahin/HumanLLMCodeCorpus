class Node:
    def __init__(self, data=None):
        """Initialize a node with optional data."""
        self.data = data
        self.next = None

class LinkedList:
    def __init__(self):
        """Initialize an empty linked list."""
        self.head = None

    def printing(self):
        """Print the values of nodes in the linked list."""
        print_value = self.head
        while print_value is not None:
            print(print_value.data)
            print_value = print_value.next

    def begin(self, data):
        """Insert a new node at the beginning of the linked list."""
        new_node = Node(data)
        new_node.next = self.head
        self.head = new_node

    def between(self, node, data):
        """Insert a new node after a specific node in the linked list."""
        new_node = Node(data)
        new_node.next = node.next
        node.next = new_node

    def end(self, data):
        """Insert a new node at the end of the linked list."""
        newest = Node(data)
        if self.head is None:
            self.head = newest
            return
        last_node = self.head
        while last_node.next is not None:
            last_node = last_node.next
        last_node.next = newest

    def delete(self, data):
        """Delete a node with a specific data value."""
        current = self.head
        if current is not None and current.data == data:
            self.head = current.next
            current = None
            return
        prev = None
        while current is not None and current.data != data:
            prev = current
            current = current.next
        if current is None:
            return
        prev.next = current.next
        current = None

# Create a linked list instance and populate it with nodes
x = LinkedList()
x.head = Node("Aimie Ojuba")
data2 = Node("latifs")
data3 = Node("Toni")
data4 = Node("slap")
x.head.next = data2
data2.next = data3
data3.next = data4

# Perform various operations on the linked list
x.begin("imaa")
x.end("beast")
x.between(x.head.next, "shile")
x.delete("slap")

# Print the linked list
x.printing()
