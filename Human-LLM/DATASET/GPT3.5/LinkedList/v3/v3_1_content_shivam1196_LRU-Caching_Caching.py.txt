class DoublyLinkedList:
    """
    Doubly linked list implementation.
    """

    class Node:
        """
        Node class for the doubly linked list.
        """

        def __init__(self, key=None, value=None):
            """
            Initialize a node with key, value, and references to previous and next nodes.

            Parameters:
            key: Key of the node.
            value: Value of the node.
            """
            self.key = key
            self.value = value
            self.prev = None
            self.next = None

    def __init__(self):
        """
        Initialize the doubly linked list with head and tail sentinel nodes.
        """
        self.head = self.Node()
        self.tail = self.Node()
        self.head.next = self.tail
        self.tail.prev = self.head

    def append(self, key, value):
        """
        Append a new node to the end of the doubly linked list.

        Parameters:
        key: Key of the new node.
        value: Value of the new node.

        Returns:
        Node: The newly appended node.
        """
        new_node = self.Node(key, value)
        self._add_to_tail(new_node)
        return new_node

    def move_to_end(self, node):
        """
        Move a node to the end of the doubly linked list.

        Parameters:
        node: Node to be moved to the end.
        """
        self._remove_node(node)
        self._add_to_tail(node)

    def _replace_tail(self, key, value):
        """
        Replace the tail node with a new node.

        Parameters:
        key: Key of the new node.
        value: Value of the new node.

        Returns:
        Node: The newly added node.
        """
        new_node = self.Node(key, value)
        self._remove_from_head()
        self._add_to_tail(new_node)
        return new_node

    def _remove_node(self, node):
        """
        Remove a node from the doubly linked list.

        Parameters:
        node: Node to be removed.
        """
        node.prev.next = node.next
        node.next.prev = node.prev

    def _add_to_tail(self, node):
        """
        Add a node to the end of the doubly linked list.

        Parameters:
        node: Node to be added to the end.
        """
        node.prev = self.tail.prev
        node.next = self.tail
        self.tail.prev.next = node
        self.tail.prev = node

    def _remove_from_head(self):
        """
        Remove a node from the head of the doubly linked list.

        Returns:
        Node: The removed node.
        """
        if self.head.next != self.tail:
            removed_node = self.head.next
            self._remove_node(removed_node)
            return removed_node

    def get_head_node(self):
        """
        Get the head node of the doubly linked list.

        Returns:
        Node: The head node.
        """
        if self.head.next != self.tail:
            return self.head.next

    def print_list(self):
        """
        Print the elements of the doubly linked list.
        """
        current = self.head.next
        while current != self.tail:
            print(f"({current.key}, {current.value})")
            current = current.next


class Caching:
    """
    Implementation of a caching mechanism using a doubly linked list.
    """

    def __init__(self, capacity):
        """
        Initialize the caching mechanism with a specified capacity.

        Parameters:
        capacity: Maximum capacity of the cache.
        """
        self.cache = {}
        self.capacity = capacity
        self.size = 0
        self.dll = DoublyLinkedList()

    def get(self, key):
        """
        Get the value corresponding to the given key from the cache.

        Parameters:
        key: Key to retrieve the value.

        Returns:
        int: Value corresponding to the key, or -1 if key not found.
        """
        if key in self.cache:
            self.dll.move_to_end(self.cache[key])
            print(self.cache[key].value)
        else:
            print(-1)

    def set(self, key, value):
        """
        Set a key-value pair in the cache.

        Parameters:
        key: Key to set.
        value: Value corresponding to the key.
        """
        if key in self.cache:
            self.dll.move_to_end(self.cache[key])
            self.cache[key].value = value
        else:
            if self.size >= self.capacity:
                del self.cache[self.dll._remove_from_head().key]
                self.size -= 1
            self.cache[key] = self.dll.append(key, value)
            self.size += 1

    def print_cache(self):
        """
        Print the elements of the cache.
        """
        self.dll.print_list()
