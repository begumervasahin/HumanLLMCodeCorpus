class Node:
    def __init__(self, data=None, next_node=None):
        """
        Initialize a node with data and a reference to the next node.

        Parameters:
        data: Data to be stored in the node.
        next_node: Reference to the next node.
        """
        self.data = data
        self.next = next_node

class LinkedList:
    def __init__(self):
        """
        Initialize a linked list with a head node.
        """
        self.head = None

    def print_list(self):
        """
        Print the elements of the linked list.
        """
        temp = self.head
        while temp:
            print(temp.data, end=" ")
            temp = temp.next
        print()

    def insert_tail_node(self, data):
        """
        Insert a node with given data at the end of the linked list.

        Parameters:
        data: Data to be stored in the new node.
        """
        if self.head is None:
            self.head = Node(data)
        else:
            current = self.head
            while current.next:
                current = current.next
            current.next = Node(data)

    def insert_head_node(self, data):
        """
        Insert a node with given data at the beginning of the linked list.

        Parameters:
        data: Data to be stored in the new node.
        """
        new_node = Node(data)
        new_node.next = self.head
        self.head = new_node

    def insert_specific_position(self, data, position):
        """
        Insert a node with given data at a specific position in the linked list.

        Parameters:
        data: Data to be stored in the new node.
        position: Position at which the new node should be inserted.
        """
        if position == 0:
            self.insert_head_node(data)
            return
        current = self.head
        for _ in range(position - 1):
            if current is None:
                return
            current = current.next
        if current is None:
            return
        new_node = Node(data)
        new_node.next = current.next
        current.next = new_node

    def delete(self, position):
        """
        Delete the node at the specified position in the linked list.

        Parameters:
        position: Position of the node to be deleted.
        """
        if position == 0:
            if self.head is None:
                return
            temp = self.head
            self.head = self.head.next
            temp.next = None
            return temp
        current = self.head
        for _ in range(position - 1):
            if current is None:
                return
            current = current.next
        if current is None or current.next is None:
            return
        temp = current.next
        current.next = current.next.next
        temp.next = None
        return temp

    def print_reverse_iterative(self):
        """
        Print the elements of the linked list in reverse order using iteration.
        """
        stack = []
        current = self.head
        while current:
            stack.append(current.data)
            current = current.next
        while stack:
            print(stack.pop(), end=" ")
        print()

    def reverse(self):
        """
        Reverse the linked list.
        """
        previous = None
        current = self.head
        while current:
            next_node = current.next
            current.next = previous
            previous = current
            current = next_node
        self.head = previous

    llist = LinkedList()
    llist.insert_head_node(1)
    llist.insert_specific_position(2, 1)
    llist.insert_specific_position(3, 2)
    llist.insert_tail_node(4)
    print("Linked list:")
    llist.print_list()
    print("\nReverse linked list:")
    llist.reverse()
    llist.print_list()
