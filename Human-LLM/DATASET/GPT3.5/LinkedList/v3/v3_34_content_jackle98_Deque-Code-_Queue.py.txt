from collections import deque

class Queue:
    def __init__(self):
        # Initialize a deque data structure to store elements in the queue
        self._queue = deque()

    def __str__(self):
        # Convert the deque to a string for display purposes
        return str(self._queue)

    def __len__(self):
        # Return the length of the queue
        return len(self._queue)

    def enqueue(self, value):
        # Add an element to the end of the queue
        self._queue.append(value)

    def dequeue(self):
        # Remove and return the first element from the queue
        if not self._queue:
            # If the queue is empty, return None
            return None
        return self._queue.popleft()

# Example usage:
if __name__ == "__main__":
    # Create a queue instance
    my_queue = Queue()

    # Enqueue elements into the queue
    my_queue.enqueue(1)
    my_queue.enqueue(2)
    my_queue.enqueue(3)

    # Display the queue
    print("Queue:", my_queue)

    # Display the length of the queue
    print("Length:", len(my_queue))

    # Dequeue an element from the queue
    dequeued_value = my_queue.dequeue()
    print("Dequeued value:", dequeued_value)

    # Display the queue after dequeue operation
    print("Queue after dequeue:", my_queue)
