from collections import deque

class Queue:
    def __init__(self):
        self._dq = deque()

    def __str__(self):
        return str(self._dq)

    def __len__(self):
        return len(self._dq)

    def enqueue(self, val):
        self._dq.append(val)

    def dequeue(self):
        if len(self._dq) == 0:
            return None
        return self._dq.popleft()

# Example usage:
if __name__ == "__main__":
    q = Queue()
    q.enqueue(1)
    q.enqueue(2)
    q.enqueue(3)
    print("Queue:", q)
    print("Length:", len(q))
    print("Dequeue:", q.dequeue())
    print("Queue after dequeue:", q)
