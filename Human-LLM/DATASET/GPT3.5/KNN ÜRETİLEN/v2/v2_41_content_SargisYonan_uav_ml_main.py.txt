import knn
import map_driver
import matplotlib.pyplot as plt
import matplotlib.animation as animation

def main():
    # Initialize UAVs and place interest points on the map
    map_driver.initialize_uavs(3)
    map_driver.place_interest_point(25, 25, 20)
    map_driver.place_interest_point(19, 25, 20)

    # Get the predicted map from the KNN module
    predicted_map = knn.predicted_map

    # Perform UAV flight simulation and save the animation
    map_driver.uav_flight(predicted_map, 'predicted.mp4')

    # Plot and display the predicted map
    map_driver.plot_map(predicted_map)
    plt.show()

if __name__ == "__main__":
    main()
