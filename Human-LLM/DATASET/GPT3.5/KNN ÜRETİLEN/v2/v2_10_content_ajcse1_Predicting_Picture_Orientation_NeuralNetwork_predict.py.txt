import time
import math
import random

# Function to create a matrix with specified dimensions and initial value
def create_matrix(I, J, fill=0.0):
    matrix = []
    for i in range(I):
        matrix.append([fill] * J)
    return matrix

class Predict:
    def __init__(self, ni, nh, no, lr):
        # Initialize the neural network with the specified parameters
        self.ninput = int(ni)
        self.nhidden = int(nh)
        self.noutput = int(no)
        self.lrate = float(lr)
        self.iactive = [1.0] * self.ninput
        self.hactive = [1.0] * self.nhidden
        self.oactive = [1.0] * self.noutput
        self.iweight = create_matrix(self.ninput, self.nhidden)
        self.oweight = create_matrix(self.nhidden, self.noutput)
        self.bias = 1.0

        # Initialize weights randomly
        for i in range(self.ninput):
            for j in range(self.nhidden):
                self.iweight[i][j] = random.uniform(0, 1)
        for j in range(self.nhidden):
            for k in range(self.noutput):
                self.oweight[j][k] = random.uniform(0, 1)

    def update(self, inputs):
        # Update the neural network with the given inputs
        for i in range(self.ninput):
            self.iactive[i] = float(inputs[i]) / float(255)

        # Update hidden layer
        for j in range(self.nhidden):
            sum = 0.0
            for i in range(self.ninput):
                sum += (self.iactive[i] * self.iweight[i][j])
            sum += self.bias
            self.hactive[j] = sum

        # Normalize and apply activation function to hidden layer
        t = max(self.hactive)
        r = min(self.hactive)
        for i in range(self.nhidden):
            self.hactive[i] = (self.hactive[i] - r) / (t - r)
            self.hactive[i] = 1 / (1 + math.exp(-self.hactive[i]))

        # Update output layer
        for k in range(self.noutput):
            sum = 0.0
            for j in range(self.nhidden):
                sum += (self.hactive[j] * self.oweight[j][k])
            self.oactive[k] = sum + self.bias

        # Normalize and apply activation function to output layer
        t = max(self.oactive)
        for i in range(self.noutput):
            self.oactive[i] = self.oactive[i] / t
            self.oactive[i] = 1 / (1 + math.exp(-self.oactive[i]))

        # Return the predicted orientation
        return self.oactive.index(t) * 90

    def backpropagate(self, targets, lrate):
        # Backpropagation algorithm to adjust weights based on target values
        odeltas = [0.0] * self.noutput

        # Calculate output layer deltas
        for k in range(self.noutput):
            error = targets[k] - self.oactive[k]
            odeltas[k] = error * self.oactive[k] * (1 - self.oactive[k])

        # Update output layer weights
        for j in range(self.nhidden):
            for k in range(self.noutput):
                change = odeltas[k] * self.hactive[j]
                self.oweight[j][k] += lrate * change

        # Calculate hidden layer deltas
        hdeltas = [0.0] * self.nhidden
        for j in range(self.nhidden):
            error = 0.0
            for k in range(self.noutput):
                error += odeltas[k] * self.oweight[j][k]
            hdeltas[j] = error * self.hactive[j] * (1 - self.hactive[j])

        # Update hidden layer weights
        for i in range(self.ninput):
            for j in range(self.nhidden):
                change = hdeltas[j] * self.iactive[i]
                self.iweight[i][j] += lrate * change

    def test(self, patterns, s):
        # Test the neural network with given patterns and expected orientation
        t = self.update(patterns)
        if t == s:
            return s
        else:
            return -1

    def train(self, train_data):
        # Train the neural network with the given training data
        for i in range(len(train_data) // 2):
            l = list(map(int, train_data[i][2:]))
            s = int(train_data[i][1])
            p = [l, [s]]
            targets = [0] * 4
            inputs = p[0]
            targets[s // 90] = 1
            self.update(inputs)
            self.backpropagate(targets, self.lrate)

    def tests(self, test_data):
        # Test the neural network with the given test data
        count = 0
        confusion_matrix = [[0 for _ in range(4)] for _ in range(4)]
        with open("nnet_output.txt", 'w') as fl:
            for i in range(len(test_data)):
                l = list(map(int, test_data[i][2:]))
                s = int(test_data[i][1])
                p = [l, [s]]
                inputs = p[0]
                t = self.test(inputs, s)
                confusion_matrix[s // 90][t // 90] += 1
                orientation = test_data[i][0] + " " + test_data[i][1] + "\n"
                fl.write(orientation)
                if t == s:
                    count += 1
        accuracy = float(count) / float(len(test_data)) * 100
        print("Percentage of Accuracy:", accuracy)
        print("Confusion Matrix:")
        for row in confusion_matrix:
            print(row)

    def knnd(self, train_data, test_data, k):
        # k-Nearest Neighbors algorithm to predict orientation
        klist = [float('inf') for _ in range(int(k))]
        kind = [None] * int(k)
        idata = list(test_data[2:])
        for j in range(len(train_data) // 20):
            distance = 0
            jdata = list(train_data[j][2:])
            for k in range(192):
                distance += abs(float(idata[k]) - float(jdata[k]))
            t = max(klist)
            if distance < t:
                ind = klist.index(t)
                klist[ind] = float(distance)
                kind[ind] = j
        ori = []
        for kindl in range(len(kind)):
            ori.append(train_data[kind[kindl]][1])
        word_counter = {}
        for word in ori:
            if word in word_counter:
                word_counter[word] += 1
            else:
                word_counter[word] = 1
        return max(word_counter, key=word_counter.get)

    def knn(self, train_data, test_data, k):
        # k-Nearest Neighbors algorithm to predict orientation and calculate accuracy
        count = 0
        confusion_matrix = [[0 for _ in range(4)] for _ in range(4)]
        with open("knn_output.txt", 'w') as fl:
            for i in range(len(test_data)):
                t = list(test_data[i])
                orientation = self.knnd(train_data, t, k)
                if orientation == t[1]:
                    count += 1
                confusion_matrix[int(t[1]) // 90][int(orientation) // 90] += 1
                orientation = t[0] + " " + orientation + "\n"
                fl.write(orientation)
        accuracy = float(count) / float(len(test_data)) * 100
        print("Percentage of Accuracy:", accuracy)
        print("Confusion Matrix:")
        for row in confusion_matrix:
            print(row)

# Example usage:
def example_usage():
    # Initialize the neural network
    p = Predict(192, 100, 4, 0.1)

    # Generate some random training and testing data (for demonstration purposes)
    train_data = [[random.randint(0, 255) for _ in range(194)] for _ in range(100)]
    test_data = [[random.randint(0, 255) for _ in range(194)] for _ in range(20)]

    # Train the neural network
    p.train(train_data)

    # Test the neural network
    p.tests(test_data)

    # Use k-Nearest Neighbors algorithm
    k = 5
    p.knn(train_data, test_data, k)

if __name__ == "__main__":
    example_usage()
