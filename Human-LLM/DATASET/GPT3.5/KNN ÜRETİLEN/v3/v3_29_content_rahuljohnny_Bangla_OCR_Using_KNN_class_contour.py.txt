import copy

MIN_CONTOUR_AREA = 100  # Define MIN_CONTOUR_AREA as needed

class ContourWithData():
    def __init__(self):
        self.npaContour = None
        self.boundingRect = None
        self.intRectX = 0
        self.intRectY = 0
        self.intRectWidth = 0
        self.intRectHeight = 0
        self.fltArea = 0.0

    def calculate_rect_top_left_point_and_width_and_height(self):
        intX, intY, intWidth, intHeight = self.boundingRect
        self.intRectX = intX
        self.intRectY = intY
        self.intRectWidth = intWidth
        self.intRectHeight = intHeight

    def check_if_contour_is_valid(self):
        return self.fltArea >= MIN_CONTOUR_AREA

    def adjust_int_rect_y(self):
        self.intRectY -= 555

def nested_list(init, *args):
    dp = init
    for x in reversed(args):
        dp = [copy.deepcopy(dp) for _ in range(x)]
    return dp

if __name__ == "__main__":
    # Create an instance of ContourWithData
    contour = ContourWithData()

    # Perform operations on contour
    contour.boundingRect = [10, 20, 30, 40]
    contour.fltArea = 200.0
    contour.calculate_rect_top_left_point_and_width_and_height()
    print("Initial intRectY:", contour.intRectY)
    contour.adjust_int_rect_y()
    print("Adjusted intRectY:", contour.intRectY)

    # Create a nested list using ndlist
    nested_list = nested_list(0, 3, 4, 2)
    print("Nested list:", nested_list)
