import numpy as np
import scipy.sparse.csgraph

def generate_random_weighted_graph(size):
    """
    Generate a random weighted graph with the specified size.
    """
    W = np.random.random((size, size))
    W += W.transpose()
    W[W < 1.0] = np.inf
    return scipy.sparse.csr_matrix(W)
