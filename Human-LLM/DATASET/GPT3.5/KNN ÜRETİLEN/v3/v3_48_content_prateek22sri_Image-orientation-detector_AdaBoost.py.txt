class Model:
    def __init__(self):
        # Initialize the model type as generic
        self.model_type = "generic"

class AdaBoost(Model):
    def __init__(self):
        # Call the parent class constructor to set the model type
        super().__init__()
        # Update the model type to AdaBoost
        self.model_type = "AdaBoost"

# Example usage
# Create an instance of the AdaBoost model
adaboost_model = AdaBoost()
# Print the model type
print(adaboost_model.model_type)  # Output: AdaBoost
