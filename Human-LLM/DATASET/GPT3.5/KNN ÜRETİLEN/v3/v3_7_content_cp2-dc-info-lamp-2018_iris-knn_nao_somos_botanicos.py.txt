import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split

# Load data from the "plantinhas.txt" file into a pandas DataFrame
data = pd.read_table("plantinhas.txt", sep=',', header=0, names=["sepal-length", "sepal-width", "petal-length", "petal-width", "class"])

# Separate features (X) and target variable (y)
features = data.iloc[:, :-1]  # Features
target = data.iloc[:, -1]     # Target variable

# Split the data into training and testing sets
X_train, X_test, y_train, y_test = train_test_split(features, target, test_size=0.2, random_state=42)

# Print the shapes of the training and testing sets
print("Training set shapes:")
print(f"X_train shape: {X_train.shape}")
print(f"y_train shape: {y_train.shape}\n")
print("Testing set shapes:")
print(f"X_test shape: {X_test.shape}")
print(f"y_test shape: {y_test.shape}")
