import numpy as np
import operator

# Example of basic arithmetic operations
a = 1
b = 2
c = a + b
print(c)

# Example of getting the number of elements in an array
b = np.array([6, 7, 8])
c = b.shape[0]
print(c)

# Example of summing all elements in a matrix
a = np.sum([[0, 1, 2], [2, 1, 3]])
print(a)

# Example of summing elements along a specific axis
a = np.sum([[0, 1, 2], [2, 1, 3]], axis=0)
print(a)

# Example of summing elements along a specific axis
a = np.sum([[8, 1, 2], [2, 1, 3], [0, 0, 0]], axis=1)
print(a)

# Example of sorting array indices based on element values
d = a.argsort()
print(d)

# Example of sorting a dictionary by values
classCount = {"A": 6, "B": 1, "C": 0, "D": 2}
sortedClassCount = sorted(classCount.items(), key=operator.itemgetter(1), reverse=True)
print(sortedClassCount)
