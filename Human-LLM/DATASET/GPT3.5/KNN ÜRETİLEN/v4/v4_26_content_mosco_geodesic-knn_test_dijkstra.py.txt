import numpy as np
import scipy.sparse.csgraph
from dijkstra import dijkstra

def test_dijkstra_against_scipy():
    # Generate a random weighted graph
    W = np.random.random((100, 100))
    W += W.transpose()
    W[W < 1.0] = np.inf
    W = scipy.sparse.csr_matrix(W)
    
    # Test Dijkstra's algorithm implementation against scipy's implementation
    for seed in [0, 1, 2]:
        # Use custom Dijkstra's algorithm implementation
        result0 = dijkstra(W, seed)
        # Use scipy's Dijkstra's algorithm implementation
        result1 = scipy.sparse.csgraph.dijkstra(W, indices=[seed])[0]
        # Assert that the results are equal
        assert all(result0 == result1)

# Run the test
test_dijkstra_against_scipy()
