from sklearn.feature_extraction.text import TfidfVectorizer
import os
from pymongo import MongoClient

def load_documents(directory):
    documents = []
    for book in os.listdir(directory):
        if not book.startswith('.'):
            with open(os.path.join(directory, book), 'rb') as f:
                content = f.read()
                content = content.decode('utf-8', errors='replace')
                documents.append(content)
    return documents

def extract_title_and_author(file_path):
    title = None
    author = None
    with open(file_path, 'rb') as f:
        content = f.read().splitlines()
    for line in content[:80]:  # Assuming title and author information within first 80 lines
        if line.startswith("Title: "):
            title = line[7:]
        elif line.startswith("Author: "):
            author = line[8:]
    return title, author

def build_database(tfidf_matrix, feature_names, title_author_store):
    database = {}
    for i in range(tfidf_matrix.shape[0]):
        doc = tfidf_matrix[i].toarray()[0]
        phrase_scores = [(index, score) for index, score in enumerate(doc) if score > 0]
        sorted_phrase_scores = sorted(phrase_scores, key=lambda t: t[1], reverse=True)
        local_word_dict = {feature_names[index]: score for index, score in sorted_phrase_scores}
        database[title_author_store[i]] = local_word_dict
    return database

def insert_to_database(posts_collection, database):
    for title_author_pair, words_dict in database.items():
        title, author = title_author_pair
        post = {"title_id_0011": title, "author_id_0011": author}
        post.update(words_dict)
        try:
            post_id = posts_collection.insert_one(post).inserted_id
        except Exception as e:
            print(f"Insertion failed for {title}, {author}: {str(e)}")

if __name__ == '__main__':
    client = MongoClient()
    db = client.bookdb
    posts = db.posts

    docs_directory = "/Users/jamesledoux/Documents/Drew2"

    # Load documents
    documents = load_documents(docs_directory)

    # Compute TF-IDF matrix
    tfidf = TfidfVectorizer(max_df=0.9, ngram_range=(1, 1), stop_words='english', strip_accents='unicode', analyzer='word')
    tfidf_matrix = tfidf.fit_transform(documents)
    feature_names = tfidf.get_feature_names()

    # Extract titles and authors
    title_author_store = [extract_title_and_author(os.path.join(docs_directory, book)) for book in os.listdir(docs_directory) if not book.startswith('.')]

    # Build database
    database = build_database(tfidf_matrix, feature_names, title_author_store)

    # Insert into MongoDB
    insert_to_database(posts, database)
