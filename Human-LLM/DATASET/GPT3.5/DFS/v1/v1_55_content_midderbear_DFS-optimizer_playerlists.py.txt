import csv
import string

def get_player_list():
    filename = input('Enter filename (.csv): ') + '.csv'
    with open(filename, 'r', newline='') as file:
        player_list = []
        reader = csv.reader(file)
        next(reader)  # Skip header row
        for row in reader:
            player = {
                'FIRST': row[2],
                'LAST': row[4],
                'POS': row[1],
                'PRICE': row[6],
                'TEAM': row[9],
                'OPP': row[10],
                'INJ': row[11]
            }
            player_list.append(player)
    return player_list

def trim_names(p_list):
    for p in p_list:
        p['FIRST'] = p['FIRST'].translate(str.maketrans('', '', string.punctuation))
        p['LAST'] = p['LAST'].translate(str.maketrans('', '', string.punctuation))
    return p_list

# Example usage:
if __name__ == "__main__":
    players = get_player_list()
    trimmed_players = trim_names(players)
    for player in trimmed_players:
        print(player)
