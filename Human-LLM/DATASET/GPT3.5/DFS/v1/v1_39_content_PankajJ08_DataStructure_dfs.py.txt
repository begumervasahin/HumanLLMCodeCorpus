from collections import defaultdict

class Graph:
    def __init__(self):
        """Initialize an empty graph."""
        self.graph = defaultdict(list)

    def add_edge(self, u, v):
        """Add an edge between vertices u and v."""
        self.graph[u].append(v)

    def dfs_visit(self, v, visited):
        """Utility function to recursively visit vertices."""
        visited[v] = True
        print(v, end=" ")  # Print the visited vertex
        for i in self.graph[v]:
            if not visited[i]:
                self.dfs_visit(i, visited)

    def dfs(self):
        """Perform Depth First Traversal."""
        visited = defaultdict(bool)
        for v in self.graph:
            if not visited[v]:
                self.dfs_visit(v, visited)

if __name__ == "__main__":
    # Create a new graph
    g = Graph()

    # Add edges to the graph
    edges = [(0, 1), (1, 2), (2, 4), (2, 5), (4, 6), (5, 10), (11, 7)]
    for edge in edges:
        g.add_edge(*edge)

    # Perform DFS starting from vertex 0
    print(" " * 10, "Depth First Traversal starting from vertex 0: ")
    g.dfs()
