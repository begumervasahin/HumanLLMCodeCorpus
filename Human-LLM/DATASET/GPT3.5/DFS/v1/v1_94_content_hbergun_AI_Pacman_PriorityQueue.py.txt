import heapq

class PriorityQueue:
    def __init__(self):
        self.elements = []

    def empty(self):
        """
        Check if the priority queue is empty.
        
        Returns:
            bool: True if the priority queue is empty, False otherwise.
        """
        return len(self.elements) == 0

    def put(self, item, priority):
        """
        Put an item into the priority queue with a specified priority.
        
        Args:
            item: The item to be inserted into the priority queue.
            priority: The priority of the item.
        """
        heapq.heappush(self.elements, (priority, item))

    def get(self):
        """
        Remove and return the item with the highest priority from the priority queue.
        
        Returns:
            item: The item with the highest priority.
        """
        return heapq.heappop(self.elements)[1]
