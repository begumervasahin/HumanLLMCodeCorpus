data = [50, 17, 72, 12, 23, 54, 76, 9, 14, 19, 0, 0, 67]

def dfs(data, x, i):
    if i <= len(data):
        if data[i-1] != 0:
            print("Checking", data[i-1])
        if data[i-1] == x:
            return i
        result = dfs(data, x, i*2)
        if result != -1:
            return result
        result = dfs(data, x, (i*2)+1)
        if result != -1:
            return result
        return -1
    else:
        return -1

def bfs(data, x, i):
    if i <= len(data):
        q = []
        temp = i
        for y in range(i):
            q.append(data[temp-1])
            if len(data) > temp:
                temp += 1
        for y in range(i):
            print("Checking", q[y])
            if q[y] == x:
                print("Yay,", x, "is found")
                return i
        print("Change row")
        for y in range(i):
            q.pop()
        result = bfs(data, x, i*2)
        if result != -1:
            return result
        return -1
    else:
        return -1

# DFS example
print("DFS:")
result_dfs = dfs(data, 72, 1)
if result_dfs != -1:
    print("Value found at index:", result_dfs)
else:
    print("Value not found")

# BFS example
print("\nBFS:")
result_bfs = bfs(data, 72, 1)
if result_bfs != -1:
    print("Value found at index:", result_bfs)
else:
    print("Value not found")
