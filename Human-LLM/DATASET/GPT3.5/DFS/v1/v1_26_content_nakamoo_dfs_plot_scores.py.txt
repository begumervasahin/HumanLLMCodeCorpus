import argparse
import matplotlib.pyplot as plt
import pandas as pd

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('scores', type=str, help='specify path of scores.txt')
    parser.add_argument('--title', type=str, default=None)
    args = parser.parse_args()

    # Read scores data from the specified file
    scores = pd.read_csv(args.scores, delimiter='\t')

    # Plot mean and median scores against steps
    for col in ['mean', 'median']:
        plt.plot(scores['steps'], scores[col], label=col)

    # Set title if provided as a command-line argument
    if args.title is not None:
        plt.title(args.title)

    # Set labels for x and y axes, and add legend
    plt.xlabel('steps')
    plt.ylabel('score')
    plt.legend(loc='best')

    # Save the figure as a PNG file with the same name as the input scores file
    fig_fname = args.scores + '.png'
    plt.savefig(fig_fname)
    print('Saved a figure as {}'.format(fig_fname))

if __name__ == '__main__':
    main()
