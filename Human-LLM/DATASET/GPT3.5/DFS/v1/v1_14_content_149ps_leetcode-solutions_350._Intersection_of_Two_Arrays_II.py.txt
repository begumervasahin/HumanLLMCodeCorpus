import collections

class Solution:
    def intersect(self, nums1, nums2):
        res = []
        hmap2 = dict(collections.Counter(nums2))
        for num in nums1:
            if hmap2.get(num):
                res.append(num)
                hmap2[num] -= 1
        return res

# Example usage
nums1 = [1, 2, 2, 1]
nums2 = [2, 2]
solution = Solution()
intersection = solution.intersect(nums1, nums2)
print("Intersection:", intersection)
