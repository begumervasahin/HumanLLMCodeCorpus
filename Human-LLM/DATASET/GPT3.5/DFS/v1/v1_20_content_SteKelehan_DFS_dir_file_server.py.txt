import requests
from collections import defaultdict

class DirFileServer:
    def __init__(self):
        self.dir_id_counter = 1
        self.dirs = defaultdict(list)
        self.file_server = 'http://localhost:'

    def get_files_on_all_servers(self, **kwargs):
        message = kwargs.get('message')
        if message is None:
            return None
        else:
            add = []
            for k in self.dirs.keys():
                x = requests.get(self.file_server + str(k) + '/files', **kwargs)
                add.append(x)
            return add

    def add_file_server(self, **kwargs):
        message = kwargs.get('message')
        if message is None:
            return None
        else:
            return requests.post()

    def add_one_file(self, **kwargs):
        message = kwargs.get('message')
        if message is None:
            return None
        else:
            min_files = 300
            for key in self.file_server:
                num_files = len(self.file_server[key])
                if num_files < min_files:
                    min_files = num_files
                    min_key = key
            return requests.post(self.file_server + str(min_key) + '/file', **kwargs)

    def get_file(self, **kwargs):
        message = kwargs.get('message')
        if message is None:
            return None
        else:
            if '_id' in message and message['_id'] is not None:
                key = [k for k, v in self.dirs.items() if message['_id'] in v]
                key = key[0]
                return requests.get(self.file_server + str(key) + '/file', json={'_id': message['_id']})
            else:
                if 'name' in message and message['name'] is not None:
                    all_files = self.get_files_on_all_servers(message=message['name'])
                    return all_files
                else:
                    return None

# Example usage
dir_file_server = DirFileServer()
files = dir_file_server.get_files_on_all_servers(message={'name': 'example.txt'})
print(files)
