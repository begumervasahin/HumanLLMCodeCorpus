peta = {
    'A': {'B', 'C', 'D'},
    'B': {'A', 'C', 'E', 'F', 'G'},
    'C': {'A', 'B', 'D'},
    'D': {'A', 'C', 'H', 'I'},
    'E': {'B', 'L'},
    'F': {'B', 'G', 'L'},
    'G': {'B', 'F', 'J', 'H'},
    'H': {'D', 'G', 'J'},
    'I': {'D', 'J'},
    'J': {'I', 'G', 'H', 'K', 'M'},
    'K': {'L', 'J', 'M'},
    'L': {'E', 'F', 'M', 'K'},
    'M': {'L', 'K', 'J'}
}

def route_dfs(peta, mulai, tujuan):
    stack = [(mulai, [mulai])]
    while stack:
        node, edge = stack.pop()
        for next_node in peta[node].difference(edge):
            if next_node == tujuan:
                yield edge + [next_node]
            else:
                stack.append((next_node, edge + [next_node]))

print("---------Program Result-------")
# Find all possible routes from 'E' to 'G'
all_routes = list(route_dfs(peta, 'E', 'G'))
route_count = len(all_routes)
# Print the result
if route_count == 0:
    print("No route found.")
else:
    print("Possible routes:", all_routes)
    print("DFS route:", all_routes[0])
