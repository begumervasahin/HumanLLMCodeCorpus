class Solution:
    def letterCombinations(self, digits):
        d = {
            '2': ['a', 'b', 'c'],
            '3': ['d', 'e', 'f'],
            '4': ['g', 'h', 'i'],
            '5': ['j', 'k', 'l'],
            '6': ['m', 'n', 'o'],
            '7': ['p', 'q', 'r', 's'],
            '8': ['t', 'u', 'v'],
            '9': ['w', 'x', 'y', 'z']
        }
        
        results = []
        if digits:
            self.dfs(digits, 0, '', results, d)
        return results

    def dfs(self, digits, index, string, results, d):
        if index == len(digits):
            results.append(string)
            return
        for letter in d[digits[index]]:
            self.dfs(digits, index + 1, string + letter, results, d)

# Example usage:
solution = Solution()
result = solution.letterCombinations("23")
print(result)  # Output: ["ad", "ae", "af", "bd", "be", "bf", "cd", "ce", "cf"]
