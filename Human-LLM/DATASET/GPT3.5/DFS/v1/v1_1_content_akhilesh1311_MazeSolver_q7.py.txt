import pickle
from os import listdir
from os.path import isfile, join
from maze import Maze
from maze_solver import DFSSolver, BFSSolver
import matplotlib.pyplot as plt

# Define probability values and maze dimension
pvals = [x / 100 for x in range(10, 35, 5)]
dim = 1000

# Initialize dictionary to store nodes expanded for each solver and probability value
nodes_expanded = {
    'dfs': {p: [] for p in pvals},
    'bfs': {p: [] for p in pvals}
}

# Loop through each probability value
for p in pvals:
    count = 0
    # Generate and solve mazes multiple times for each probability value
    while count < 30:
        m = Maze(dim, p)
        d = DFSSolver(m)
        d.display_solution(display=False)
        # Ensure the maze is solvable
        if d.is_solvable:
            count += 1
            s = BFSSolver(m)
            print(f"Solving maze {count} for p={p}...")
            s.display_solution(display=False)
            assert s.is_solvable == True
            # Store the number of nodes expanded for both solvers
            nodes_expanded['dfs'][p].append(d.nodes_expanded)
            nodes_expanded['bfs'][p].append(s.nodes_expanded)

# Calculate average number of nodes expanded for each solver and probability value
bfs = [sum(nodes_expanded['bfs'][p]) / len(nodes_expanded['bfs'][p]) for p in pvals]
dfs = [sum(nodes_expanded['dfs'][p]) / len(nodes_expanded['dfs'][p]) for p in pvals]

# Plot the results
plt.plot(pvals, bfs, pvals, dfs)
ax = plt.gca()
ax.legend(('BFS', 'DFS'))
plt.xlabel("p-value")
plt.ylabel("Avg. No. of Nodes Expanded")
plt.savefig("q7.png")
plt.show()
