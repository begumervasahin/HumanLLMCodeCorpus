from sklearn.neighbors import NearestNeighbors
import numpy as np

# Define the input data array
X = np.array([[-1, -1], [-2, -1], [-3, -2], [1, 1], [2, 1], [3, 2]])

# Initialize NearestNeighbors with parameters
nbrs = NearestNeighbors(n_neighbors=2, algorithm='ball_tree').fit(X)

# Print the NearestNeighbors object
print("NearestNeighbors Object:")
print(nbrs)

# Find the nearest neighbors and distances for each data point
distances, indices = nbrs.kneighbors(X)

# Print the distances and indices of nearest neighbors
print("\nDistances to Nearest Neighbors:")
print(distances)

print("\nIndices of Nearest Neighbors:")
print(indices)
