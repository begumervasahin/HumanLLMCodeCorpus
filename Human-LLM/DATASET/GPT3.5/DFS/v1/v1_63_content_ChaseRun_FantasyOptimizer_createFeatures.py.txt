import pandas as pd
import math
import csv

def playerFeat1(player, game):
    returnData = {}
    # Initialize all features to 0
    features = [
        'Home', 'MP', 'FG', 'FGA', 'FG%', '3P', '3PA', '3P%', 'FT', 'FTA',
        'FT%', 'ORB', 'DRB', 'TRB', 'AST', 'STL', 'BLK', 'TOV', 'PF', 'PTS',
        '+/-', 'TS%', 'eFG%', '3PAr', 'FTr', 'ORB%', 'DRB%', 'TRB%', 'AST%',
        'STL%', 'BLK%', 'TOV%', 'USG%', 'ORtg', 'DRtg'
    ]
    for feat in features:
        returnData[feat] = 0
    
    data = pd.read_csv('data/LakersPlayerData.csv')
    prevGame = game - 1
    
    for index, row in data.iterrows():
        if ((row['Players'] == player) and (row['Game'] == prevGame)):
            # Update feature values from the dataset
            returnData['Home'] = float(row['Home'])
            realTime = 0
            time = row['MP']
            # Calculate real time from 'MP' column
            if time:
                min1 = time[0]
                if min1.isdigit():
                    if (time[1] != ':'):
                        min2 = time[1]
                        sec1 = time[3]
                        sec2 = time[4]
                        realTime = float(min1) * 10 + float(min2) + float(sec1) * 10 / 60 + float(sec2) / 60
                    else:
                        sec1 = time[2]
                        sec2 = time[3]
                        realTime = float(min1) * 10 + float(sec1) * 10 / 60 + float(sec2) / 60
            returnData['MP'] = realTime if realTime else 0
            if realTime == 0:
                continue
            # Update other features
            for feat in features[2:]:
                returnData[feat] = checkNaN(float(row[feat]))
            break

    return returnData

def checkNaN(val):
    return val if not math.isnan(val) else 0

def createTest(player, game, var, features):
    trainData = pd.read_csv(f'testData/{var}/{player}.csv')
    newDataSheet = []
    for index, row in trainData.iterrows():
        if row['Game'] == game:
            newDataSheet.append(row)
            break

    with open(f'test/{var}/{player}_{game}.csv', mode='w', newline='') as playerFile:
        playerWriter = csv.writer(playerFile)
        playerWriter.writerow(features)
        playerWriter.writerow(newDataSheet)

# Example usage
player_data = playerFeat1('LeBron James', 5)
print(player_data)

# Example usage of createTest function
features = ['Game', 'MP', 'FG', '3P', 'FT', 'PTS']
createTest('LeBron James', 5, 'players', features)
