from breadth_first_search import bfs
from state import State

def test_bfs_algorithm():
    print("=======================")
    print("BFS Algorithm Test")
    print("=======================")
    
    # Initial State
    initial_state = State(red=13, blue=16, green=17)
    
    # Execute BFS algorithm
    result = bfs(initial_state)
    
    # Extract final state from result
    final_state = result['final_state']
    
    # Assert final state
    assert final_state.red_count == 0
    assert final_state.green_count == 46
    assert final_state.blue_count == 0
    
    # Extract full path and path from result
    full_path = result['full_path']
    path = result['path']
    
    # Print lengths of full path and path
    print(f"Full path length: {len(full_path)}")
    print(f"Path length: {len(path)}")
    
    # Assert lengths of full path and path
    assert len(full_path) == 180169
    assert len(path) == 17
    
    # Print nodes in the path
    print("Nodes in the Path:")
    for i, node in enumerate(path, start=1):
        print(f"{i}: {node}")

if __name__ == "__main__":
    test_bfs_algorithm()
