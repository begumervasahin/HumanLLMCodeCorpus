from foundation import Node, State, Block
from searchMethods import SearchMethods

# Define the start state
start_movables = [Block('A', 0, 0), Block('B', 1, 0), Block('C', 2, 0)]
start_agent = Block('P', 3, 0)
start_state = State(start_movables, start_agent, None)
start_node = Node(start_state, None, None)

# Define the goal state
fin_movables = [Block('A', 1, 2), Block('B', 1, 1), Block('C', 1, 0)]
fin_agent = Block('P', 3, 0)
fin_state = State(fin_movables, fin_agent, None)

# Initialize SearchMethods with start node and goal state
sm = SearchMethods(start_node, fin_state)

# Prompt user to select a search method
print('Select Search Method:\n1. DFS\n2. BFS\n3. Iterative Deepening\n4. A*')
inp = int(input())

# Perform the selected search method
if inp == 1:
    print('DEPTH FIRST SEARCH')
    sm.DFS()
elif inp == 2:
    print('BREADTH FIRST SEARCH')
    sm.BFS()
elif inp == 3:
    print('ITERATIVE DEEPENING SEARCH')
    sm.iterativeDeepening()
elif inp == 4:
    print('A* SEARCH')
    sm.aStar()
else:
    print('Invalid input. Please select a valid search method.')
