class Node:
    def __init__(self, index):
        self.children = []
        self.index = index
        self.discovered = False

    def add_child(self, index):
        self.children.append(index)

class Graph:
    def __init__(self, head, nodes):
        self.head = head
        self.nodes = nodes

def read_file(path):
    head = 0
    nodes = []
    with open(path, "r") as file:
        head = int(file.readline())
        index = -1
        node_i = None
        for edge in file.readlines():
            split1, split2 = map(int, edge.split(','))
            if index == split1:
                node_i.add_child(split2)
            else:
                if index != -1:
                    nodes.append(node_i)
                index = split1
                node_i = Node(split1)
                node_i.add_child(split2)
        nodes.append(node_i)
    return Graph(head, nodes)
