def input_board(inputfile, board, n):
    for i in range(n):
        lst = [char for char in inputfile.readline().rstrip('\n')]
        board.append(lst)

def failCase(outputfile):
    outputfile.write('FAIL')

def passCase(outputfile, board):
    outputfile.write('OK\n')
    lst1 = []
    for lst in board:
        string1 = ''.join(lst)
        lst1.append(string1)
    string = '\n'.join(lst1)
    outputfile.write(string)

def check_col_diagonal(board, row, col, n):
    flag = '0'
    for i in range(row - 1, -1, -1):
        if board[i][col] == '2':
            break
        if board[i][col] == '1':
            flag = '-1'
            break
    if flag == '-1':
        return False
    for i in range(row + 1, n, 1):
        if board[i][col] == '2':
            break
        if board[i][col] == '1':
            flag = '-1'
            break
    if flag == '-1':
        return False
    for i, j in zip(range(row - 1, -1, -1), range(col - 1, -1, -1)):
        if board[i][j] == '2':
            break
        if board[i][j] == '1':
            flag = '-1'
            break
    if flag == '-1':
        return False
    for i, j in zip(range(row - 1, -1, -1), range(col + 1, n, 1)):
        if board[i][j] == '2':
            break
        if board[i][j] == '1':
            flag = '-1'
            break
    if flag == '-1':
        return False
    for i, j in zip(range(row + 1, n, 1), range(col + 1, n, 1)):
        if board[i][j] == '2':
            break
        if board[i][j] == '1':
            flag = '-1'
            break
    if flag == '-1':
        return False
    for i, j in zip(range(row + 1, n, 1), range(col - 1, -1, -1)):
        if board[i][j] == '2':
            break
        if board[i][j] == '1':
            flag = '-1'
            break
    if flag == '-1':
        return False
    return True

def safe(board, n):
    flag = '0'
    for row in range(n):
        if sum(x.count('0') for x in board[row]) <= 0:
            continue
        if '1' in board[row]:
            if '2' not in board[row]:
                continue
            for col in range(n):
                if board[row][col] == '2' or board[row][col] == '1':
                    continue
                for i in range(col - 1, -1, -1):
                    if board[row][i] == '2':
                        break
                    if board[row][i] == '1':
                        flag = '-1'
                        break
                if flag == '-1':
                    continue
                for j in range(col + 1, n, 1):
                    if board[row][i] == '2':
                        break
                    if board[row][i] == '1':
                        flag = '-1'
                        break
                if flag == '-1':
                    continue
                if check_col_diagonal(board, row, col, n):
                    return row, col
        for col in range(n):
            if board[row][col] == '2':
                continue
            if check_col_diagonal(board, row, col, n):
                return row, col
    return -1, -1

def dfs(board, placed_liz, total_liz, t, n):
    if placed_liz >= total_liz:
        return True
    for i in range(total_liz):
        row, col = -1, -1
        row, col = safe(board, n)
        if row != -1:
            board[row][col] = '1'
            if dfs(board, placed_liz + 1, total_liz, t, n):
                return True
            board[row][col] = '0'
    return False

inputfile = open("input.txt", "r")
outputfile = open("output.txt", "w")
algorithm_name = inputfile.readline()
n = int(inputfile.readline())
total_liz = int(inputfile.readline())
board = []
input_board(inputfile, board, n)
t = sum(x.count('2') for x in board)

if algorithm_name == 'BFS\n':
    pass
elif algorithm_name == 'DFS\n':
    if total_liz > (n + t):
        failCase(outputfile)
    elif dfs(board, 0, total_liz, t, n):
        passCase(outputfile, board)
    else:
        failCase(outputfile)
else:
    pass

inputfile.close()
outputfile.close()
