def bfs(graph, start, end):
    queue = [[start]]
    visited = set()
    
    while queue:
        path = queue.pop(0)
        state = path[-1]
        
        if state == end:
            return path
        elif state not in visited:
            for branch in graph.get(state, []):
                new_path = list(path)
                new_path.append(branch)
                queue.append(new_path)
            visited.add(state)
        
        if len(queue) == 0:
            print("Tidak ditemukan")
            return None
