from mds_db import MDS_DB

# Create an instance of the MDS_DB class with the database file path
db = MDS_DB("dfs.db")

# Print status messages
print("Connecting to database")
# Connect to the database
db.Connect()

# Print status message
print("Testing node addition")
# Add data nodes
id1 = db.AddDataNode("localhost", 4000)
id2 = db.AddDataNode("localhost2", 5080)

# Print empty line
print()

# Print status message
print("Testing if node was inserted")
print("A tuple with node name and connection info must appear")
# Check if the node was inserted
print(db.CheckNode(id1))

# Print empty line
print()

# Print status message
print("Testing all Available data nodes")
# Get and print available data nodes
for address, port in db.GetDataNodes():
    print(address, port)

# Print empty line
print()

# Print status message
print("Inserting two files to DB")
# Insert files into the database
db.InsertFile("/hola/cheo.txt", 20)
db.InsertFile("/opt/blah.txt", 30)

# Print empty line
print()

# Print status message
print("Files in the database")
# Get and print files in the database
for file, size in db.GetFiles():
    print(file, size)

# Print empty line
print()

# Print status message
print("Adding blocks to the file, duplicate message if not the first time running this script")
try:
    # Add blocks to the file
    db.AddBlockToInode("/hola/cheo.txt", [(id1, "1"), (id2, "1")])
except:
    print("Won't duplicate")

# Print empty line
print()

# Print status message
print("Testing retrieving Inode info")
# Get file inode information
fsize, chunks_info = db.GetFileInode("/hola/cheo.txt")
print("File Size is:", fsize)
print("and can be constructed from:")
for address, port, chunk in chunks_info:
    print(address, port, chunk)

# Print empty line
print()

# Print status message
print("Closing connection")
# Close the database connection
db.Close()
