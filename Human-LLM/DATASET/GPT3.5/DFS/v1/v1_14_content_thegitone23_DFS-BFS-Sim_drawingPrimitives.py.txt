import pygame
import math

# Constants
RAD = 30
WIDTH = 800
HEIGHT = 600

# Initialize Pygame
pygame.init()
screen = pygame.display.set_mode((WIDTH, HEIGHT))
clock = pygame.time.Clock()

# Function to calculate distance between two points
def distance(x1, y1, x2, y2):
    return math.sqrt((x2 - x1)**2 + (y2 - y1)**2)

# Function to draw a pixel
def drawPixel(x, y):
    pygame.draw.circle(screen, (255, 255, 255), (x, y), 1)

# Function to draw a line
def drawLine(a, b, c, d):
    pygame.draw.line(screen, (255, 255, 255), (a, b), (c, d))

# Function to draw a hollow circle
def drawHollowCircle(x, y, r=RAD):
    lineAmount = 100
    for i in range(lineAmount):
        angle = i * 2 * math.pi / lineAmount
        x1 = int(x + r * math.cos(angle))
        y1 = int(y + r * math.sin(angle))
        x2 = int(x + r * math.cos(angle + 2 * math.pi / lineAmount))
        y2 = int(y + r * math.sin(angle + 2 * math.pi / lineAmount))
        drawLine(x1, y1, x2, y2)

# Function to draw a filled circle
def drawFilledCircle(x, y, r=RAD):
    triangleAmount = 30
    for i in range(triangleAmount):
        angle1 = i * 2 * math.pi / triangleAmount
        angle2 = (i + 1) * 2 * math.pi / triangleAmount
        x1 = int(x)
        y1 = int(y)
        x2 = int(x + r * math.cos(angle1))
        y2 = int(y + r * math.sin(angle1))
        x3 = int(x + r * math.cos(angle2))
        y3 = int(y + r * math.sin(angle2))
        pygame.draw.polygon(screen, (255, 255, 255), [(x1, y1), (x2, y2), (x3, y3)])

# Main loop
def main():
    running = True
    while running:
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                running = False

        screen.fill((0, 0, 0))  # Clear the screen
        drawHollowCircle(WIDTH // 2, HEIGHT // 2)  # Draw a hollow circle
        drawFilledCircle(WIDTH // 4, HEIGHT // 4)  # Draw a filled circle
        pygame.display.flip()  # Update the display
        clock.tick(60)  # Cap the frame rate

    pygame.quit()

if __name__ == "__main__":
    main()
