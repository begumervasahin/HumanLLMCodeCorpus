import json
import rep
import busqueda
import sys

# Check if the correct number of command-line arguments are provided
if len(sys.argv) != 3:
    print('Debe ingresar dos argumentos en el comando de línea:')
    print('Tipo de búsqueda: bfs, dfs')
    print('Mostrar Q: si, no')
    quit()

# Check if the first argument is valid (bfs or dfs)
if sys.argv[1] not in ['bfs', 'dfs']:
    print('Por favor ingrese un argumento válido para el primer parámetro: bfs, dfs')
    quit()

# Check if the second argument is valid (si or no)
if sys.argv[2] not in ['si', 'no']:
    print('Por favor ingrese un argumento válido para el segundo parámetro: si, no')
    quit()

# Determine whether to show Q or not based on the second argument
mostrarQ = True if sys.argv[2] == 'si' else False

# Load data from JSON file
with open('data.json') as json_file:
    data = json.load(json_file)
    print(data['estadoInicial'])
    print(data['estadoObjetivo'])

# Set up the world based on the loaded data
world = rep.setWorld(data)
inicial = world['inicial']
final = world['final']

# Search for the solution using the specified search algorithm
solucion = busqueda.buscarSolucion(inicial, final, sys.argv[1], mostrarQ)

# Print the steps for the solution
print('Pasos para la solución:')
for paso in solucion['estadosAnteriores']:
    print(paso)

# Print the applied operators
print('\nOperadores aplicados:')
for operacion in solucion['operadores']:
    print(operacion)
