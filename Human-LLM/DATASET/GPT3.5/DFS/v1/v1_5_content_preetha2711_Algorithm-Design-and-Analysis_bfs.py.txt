from collections import deque

def bfs_traversal(vertices, adj_list, source):
    visited = [None] * len(vertices)
    visited[source] = 0
    queue = deque()
    queue.append(source)

    while queue:
        vertex_check = queue.popleft()
        print(vertex_check)
        for i in adj_list[vertex_check]:
            if visited[i] is None:
                visited[i] = 0
                queue.append(i)

    if None in visited:
        print("Graph is disconnected")
    else:
        print("All nodes of the graph have been traversed")

def cycle_bfs(array, source):
    parent = [None] * len(array)
    visited = [None] * len(array)
    parent[source] = None
    visited[source] = 0
    queue = deque()
    queue.append(source)

    while queue:
        vertex_check = queue.popleft()
        for i in array[vertex_check]:
            if parent[i] is not None:
                print("Cycle exists")
                return
            else:
                visited[i] = 0
                parent[i] = vertex_check
                queue.append(i)

# Example data
vertices = [0, 1, 2, 3, 4]
adj_list = [[1, 2], [2, 3], [3, 4], [4, 0], [0, 1]]

# BFS traversal
source_vertex = int(input("Enter source vertex: "))
bfs_traversal(vertices, adj_list, source_vertex)

# Check for cycle using BFS
cycle_source = int(input("Enter source vertex to check for cycle: "))
cycle_bfs(adj_list, cycle_source)
