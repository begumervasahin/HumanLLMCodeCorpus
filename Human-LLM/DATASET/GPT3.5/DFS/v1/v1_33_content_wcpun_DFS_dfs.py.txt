from fs.memoryfs import MemoryFS
from fs.osfs import OSFS

# Create a MemoryFS instance
dfs = MemoryFS()

# Create directories 'test' and 'foo' in MemoryFS
dfs.makedir('test')
dfs.makedir('foo')

# Walk through the directories and print info
for path, info in dfs.walk.info(max_depth=1):
    if info.is_dir:
        print("[dir] {}".format(path))
    else:
        print("[file] {}".format(path))
