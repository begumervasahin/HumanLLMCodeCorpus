import nflgame

class TeamDefense:
    """
    A class for computing defensive stats for a full team
    """
    def __init__(self):
        self.defense_sk = 0
        self.defense_int = 0
        self.defense_frec = 0
        self.kickret_tds = 0
        self.puntret_tds = 0
        self.defense_int_tds = 0
        self.defense_frec_tds = 0
        self.defense_misc_tds = 0
        self.defense_safe = 0
        self.blocked_kicks = 0

    def get_team_stats(self, players, t):
        self.__init__()
        for p in players.defense().filter(team=t):
            self.defense_sk += p.defense_sk
            self.defense_int += p.defense_int
            self.defense_frec += p.defense_frec
            self.kickret_tds += p.kickret_tds
            self.puntret_tds += p.puntret_tds
            self.defense_int_tds += p.defense_int_tds
            self.defense_frec_tds += p.defense_frec_tds
            self.defense_misc_tds += p.defense_misc_tds
            self.defense_safe += p.defense_safe
            self.blocked_kicks += p.defense_puntblk
            self.blocked_kicks += p.defense_xpblk
            self.blocked_kicks += p.defense_fgblk

if __name__ == "__main__":
    # Create an instance of the TeamDefense class
    team_defense = TeamDefense()

    # Get NFL game players data
    players = nflgame.combine_game_stats(nflgame.games(2019))

    # Specify team abbreviation
    team_abbreviation = 'NE'  # Example: New England Patriots

    # Get defensive stats for the specified team
    team_defense.get_team_stats(players, team_abbreviation)

    # Print the defensive stats
    print("Defensive Stats for Team {}: ".format(team_abbreviation))
    print("Sacks:", team_defense.defense_sk)
    print("Interceptions:", team_defense.defense_int)
    print("Fumble Recoveries:", team_defense.defense_frec)
    print("Kick Return Touchdowns:", team_defense.kickret_tds)
    print("Punt Return Touchdowns:", team_defense.puntret_tds)
    print("Interception Return Touchdowns:", team_defense.defense_int_tds)
    print("Fumble Recovery Touchdowns:", team_defense.defense_frec_tds)
    print("Miscellaneous Defensive Touchdowns:", team_defense.defense_misc_tds)
    print("Safeties:", team_defense.defense_safe)
    print("Blocked Kicks:", team_defense.blocked_kicks)
