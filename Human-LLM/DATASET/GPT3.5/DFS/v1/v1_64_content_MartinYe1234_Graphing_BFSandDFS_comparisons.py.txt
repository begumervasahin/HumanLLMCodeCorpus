import matplotlib.pyplot as plt
from random import randint
from time import time

def gen_graph(num_of_nodes, num_of_connections):
    adj = {}
    adjacent = {}
    for i in range(num_of_nodes):
        adj[i] = []
        adjacent[i] = []
    for i in range(num_of_connections):
        x = randint(0, num_of_nodes - 1)
        y = randint(0, num_of_nodes - 1)
        adj[x].append(y)
        adj[y].append(x)
        adjacent[x].append(y)
        adjacent[y].append(x)
    for i in range(num_of_nodes):
        adj[i] = list(set(adj[i]) - set([i]))
        adjacent[i] = list(set(adjacent[i]) - set([i]))
    return adj, adjacent

def dfs(node):
    dfs_visited[node] = 1
    for neighbor in adj[node]:
        if not dfs_visited[neighbor]:
            dfs(neighbor)

def bfs(node):
    queue = [node]
    bfs_visited[node] = 1
    while queue:
        for neighbour in adjacent[queue[0]]:
            if bfs_visited[neighbour] == 0:
                queue.append(neighbour)
                bfs_visited[neighbour] = 1
        queue.pop(0)

def run_tests(num_of_nodes, num_of_connections, runs):
    dfs_runtime = 0.0
    bfs_runtime = 0.0
    dfs_data = []
    bfs_data = []
    for _ in range(num_of_connections):
        for _ in range(runs):
            adj, adjacent = gen_graph(num_of_nodes, num_of_connections)
            dfs_visited = [0 for _ in range(num_of_nodes)]
            bfs_visited = [0 for _ in range(num_of_nodes)]

            dfs_time = time()
            dfs(0)
            dfs_runtime += (time() - dfs_time)

            bfs_time = time()
            bfs(0)
            bfs_runtime += (time() - bfs_time)

        dfs_data.append(round((dfs_runtime / runs), 13))
        bfs_data.append(round((bfs_runtime / runs), 13))

        dfs_runtime = 0.0
        bfs_runtime = 0.0

    return dfs_data, bfs_data

def plot_results(x_axis, dfs_data, bfs_data):
    plt.figure(figsize=(16, 5))
    plt.plot(x_axis, dfs_data, label='DFS', color='blue')
    plt.plot(x_axis, bfs_data, label='BFS', color='orange')
    plt.xlabel('Number of Connections')
    plt.ylabel('Runtime (seconds)')
    plt.title('BFS (Orange) / DFS (Blue) Comparison')
    plt.legend()
    plt.show()

if __name__ == "__main__":
    num_of_nodes = 200
    num_of_connections = 2000
    runs = 200
    x_axis = [i + 1 for i in range(num_of_connections)]
    dfs_data, bfs_data = run_tests(num_of_nodes, num_of_connections, runs)
    plot_results(x_axis, dfs_data, bfs_data)
