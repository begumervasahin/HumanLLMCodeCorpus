import pandas as pd
import requests
from bs4 import BeautifulSoup as bsoup

def grab_first():
    for year in [2016, 2015, 2014, 2013]:
        for week in range(1, 17):
            url = f'http://rotoguru1.com/cgi-bin/fyday.pl?week={week}&year={year}&game=fd&scsv=1'
            html = requests.get(url)
            soup = bsoup(html.content, 'html.parser')
            pre = soup.find_all('pre')
            with open('test.csv', 'a+') as f:
                for i in pre[0]:
                    f.write(str(i))
    return pre

if __name__ == '__main__':
    soup = grab_first()
