import math
import numpy as np

def stft(source, winlen, slide):
    # Pad the input signal with zeros to ensure its length is a power of 2
    top_index = 2 ** power2_near(len(source))
    if top_index - len(source) > 0:
        data = source + [0] * (top_index - len(source))
    else:
        data = source

    # Calculate the number of windows
    win_num = int((len(data) - winlen) / slide) + 1

    # Initialize a matrix to store the windowed segments of the signal
    Map = np.arange(win_num * winlen).reshape(winlen, win_num)

    # Fill the matrix with the windowed segments of the signal
    for i in range(win_num):
        Map[:, i] = data[slide * i:slide * i + winlen]

    # Convert the matrix to a complex data type
    cMap = Map.astype(complex)

    # Apply the Fast Fourier Transform (FFT) to each column of the matrix
    for i in range(win_num):
        cMap[:, i] = np.fft.fft(Map[:, i])

    return cMap

def power2_near(k):
    assert k > 0, "Input must be bigger than 0"
    power = 0
    while True:
        if k > 1:
            k = math.ceil(k / 2)
            power += 1
        elif k <= 1:
            break
    return power
