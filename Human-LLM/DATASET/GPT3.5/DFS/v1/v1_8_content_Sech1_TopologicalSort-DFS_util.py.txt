import itertools
import sys
import time
from graph import Graph

class Util:
    go = True

    def __init__(self):
        pass

    @classmethod
    def spinning_cursor(cls):
        spinner = itertools.cycle(['-Doing Algo Stuff. ', '/Doing Algo Stuff.. ',
                                   '|Doing Algo Stuff... ', '\\Doing Algo Stuff.... '])
        while Util.go:
            sys.stdout.write(next(spinner))
            sys.stdout.flush()
            time.sleep(.25)
            sys.stdout.write('\r')

    @classmethod
    def read_in_graph(cls, file):
        with open(file) as f:
            lines = f.readlines()

        vert_count = len(lines)
        graph = Graph(vert_count)

        for i in range(len(lines)):
            lines[i] = lines[i].strip('\n')

        for line in lines:
            vert, edges = line.split(':')
            edges = edges.split(' ')
            for edge in edges:
                graph.graph[vert].append(edge)

        for i in range(1, vert_count + 1):
            graph.graph[str(i)] = list(filter(None, graph.graph[str(i)]))

        return graph
