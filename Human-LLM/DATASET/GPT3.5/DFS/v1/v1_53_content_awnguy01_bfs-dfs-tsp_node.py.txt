from typing import List

class Node:
    def __init__(self, name: str, x: float = 0, y: float = 0, dest_nodes: List['Node'] = None):
        self.name = name
        self.x = x
        self.y = y
        self.dest_nodes = dest_nodes if dest_nodes is not None else []

    def __eq__(self, other):
        return self.name == other.name

    def dest_nodes_to_string(self) -> str:
        next_nodes_str: str = '['
        if self.dest_nodes:
            for i, node in enumerate(self.dest_nodes):
                if i == 0:
                    next_nodes_str += node.name
                else:
                    next_nodes_str += f',{node.name}'
        next_nodes_str += ']'
        return next_nodes_str

# Example usage:
if __name__ == "__main__":
    node1 = Node("A")
    node2 = Node("B")
    node3 = Node("C")
    
    node1.dest_nodes = [node2, node3]
    node2.dest_nodes = [node1]
    node3.dest_nodes = [node2]
    
    print(node1.dest_nodes_to_string())  # Output: [B,C]
    print(node2.dest_nodes_to_string())  # Output: [A]
    print(node3.dest_nodes_to_string())  # Output: [B]
