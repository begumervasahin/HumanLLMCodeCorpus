def combination_sum(candidates, target):
    candidates.sort()
    result = []
    current_index = 0
    result_seq = []
    combination_sum_helper(candidates, target, current_index, result_seq, result)
    return result

def combination_sum_helper(candidates, target, current_index, result_seq, result):
    for j in range(current_index, len(candidates)):
        num = candidates[j]
        if num > target:
            return
        result_seq.append(num)
        if num < target:
            combination_sum_helper(candidates, target - num, j, result_seq, result)
        else:
            result.append(list(result_seq))
        result_seq.pop()

# Example usage:
candidates = [2, 3, 6, 7]
target = 7
print(combination_sum(candidates, target))  # Output: [[2, 2, 3], [7]]
