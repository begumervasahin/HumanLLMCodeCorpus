from collections import deque
from mymap import myMap  # Assuming myMap is a dictionary containing the map data
from modules import showResultWithAttr, showStep  # Assuming these functions are defined in a separate module

def LCBFS(start, goal):
    q = deque()
    q.append(start)
    previous = {}
    for city in myMap.keys():
        previous[city] = {'from': None, 'total_cost': 0}
    counter = 0
    while True:
        counter += 1
        showStep(counter, q, previous)  # Assuming showStep function displays the current step
        curCitys = []
        while len(q) != 0:
            curCitys.append(q.popleft())
        for curCity in curCitys:
            curCityTotalCost = previous[curCity]['total_cost']
            if curCity != goal:
                for city in myMap[curCity].keys():
                    cityTotalCost = previous[city]['total_cost']
                    totalCost = myMap[curCity][city]['cost'] + curCityTotalCost
                    if previous[city]['from'] is None or totalCost < cityTotalCost:
                        if city in q:
                            q.remove(city)
                        q.append(city)
                        previous[city]['from'] = curCity
                        previous[city]['total_cost'] = totalCost
        if len(q) == 0:
            break
    if previous[goal]['from'] is not None:
        showResultWithAttr(previous, start, goal)  # Assuming showResultWithAttr displays the result
        return True
    else:
        return False

# Example usage:
start_city = 'A'  # Replace with actual start city
goal_city = 'G'  # Replace with actual goal city
result = LCBFS(start_city, goal_city)
if result:
    print("Route found!")
else:
    print("No route found.")
