from PIL import Image

# Define a list of RGB tuples
rgb_tuples = [(0, 0, 0), (0, 0, 0), (128, 128, 128), (256, 256, 256), (256, 256, 256)] * 2000

# Create a new image with white background
image_size = (100, 100)
background_color = "white"
image = Image.new("RGB", image_size, background_color)

# Put the RGB tuples data into the image
image.putdata(rgb_tuples)

# Display the image
image.show()
