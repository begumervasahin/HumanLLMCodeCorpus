def dfs(graph, start, goal):
    stack = [[start]]
    visited = set()

    while stack:
        current_path = stack.pop()  # Get the current path
        current_node = current_path[-1]  # Get the current node from the path

        if current_node == goal:  # Check if we reached the goal
            return current_path

        if current_node not in visited:
            for neighbor in graph.get(current_node, []):  # Get neighbors of the current node
                new_path = list(current_path)  # Create a new path
                new_path.append(neighbor)  # Add the neighbor to the path
                stack.append(new_path)  # Add the new path to the stack

            visited.add(current_node)  # Mark the current node as visited

    print("Path not found")
    return None

# Example usage
graph = {
    'A': {'B'},
    'B': {'A', 'C', 'F', 'G'},
    'C': {'B', 'D'},
    'D': {'C', 'E'},
    'E': {'D', 'F'},
    'F': {'G', 'E', 'B'},
    'G': {'F', 'H', 'B'},
    'H': {'I', 'G', 'L'},
    'I': {'H', 'J'},
    'J': {'I', 'K'},
    'K': {'L', 'J'},
    'L': {'K', 'H'}
}

start_node = 'A'
goal_node = 'I'
print(dfs(graph, start_node, goal_node))
