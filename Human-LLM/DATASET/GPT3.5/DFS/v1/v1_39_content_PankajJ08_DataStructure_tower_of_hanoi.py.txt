def tower_of_hanoi_sol(h, initial, end, aux):
    x = 1  # Initialize the move counter
    def move_disk(disk, from_peg, to_peg):
        print(f"{x}: Move disk {disk} from peg {from_peg} to peg {to_peg}")
        nonlocal x
        x += 1
    
    def hanoi(n, source, target, auxiliary):
        if n > 0:
            hanoi(n - 1, source, auxiliary, target)
            move_disk(n, source, target)
            hanoi(n - 1, auxiliary, target, source)

    hanoi(h, initial, end, aux)

# Example usage: Solve Tower of Hanoi for 3 disks
tower_of_hanoi_sol(3, 1, 3, 2)
