from graphviz import Digraph
import copy

class TreeNode:
    def __init__(self, name, last_time_of_visit):
        self.name = name
        self.last_time_of_visit = last_time_of_visit
        self.list_of_nodes = []

# Function to get a new name for duplicate nodes
def get_new_name(nodes, name):
    name = name.split('-')[0]
    for i in range(1, len(nodes) + 1):
        new_name = f"{name}-{i}"
        if new_name not in nodes:
            return new_name

# Function to change duplicate names in the tree
def change_duplicate_names(copy_tree_node):
    nodes = [copy_tree_node.name]
    queue = [copy_tree_node]
    while queue:
        node = queue.pop(0)
        for child in node.list_of_nodes:
            queue.append(child)
            if child.name not in nodes:
                nodes.append(child.name)
            else:
                child.name = get_new_name(nodes, child.name)
                nodes.append(child.name)

# Function to draw the tree using Graphviz
def draw_tree(tree_node_root, name_pdf):
    copy_tree_node = copy.deepcopy(tree_node_root)
    change_duplicate_names(copy_tree_node)
    g = Digraph()
    g.node(copy_tree_node.name)
    queue = [copy_tree_node]
    while queue:
        node = queue.pop(0)
        for child in node.list_of_nodes:
            queue.append(child)
            g.node(child.name)
            g.edge(node.name, child.name, str(child.last_time_of_visit), minlen='2', dir="forward")
    g.view(name_pdf)

# Sample usage
if __name__ == "__main__":
    root = TreeNode("A", 0)
    b = TreeNode("B", 1)
    c = TreeNode("C", 2)
    d = TreeNode("A", 3)
    root.list_of_nodes = [b, c, d]
    draw_tree(root, "tree_output")
