def bfs(graph, start, end):
    queue = [[start]]
    visited = set()

    while queue:
        route = queue.pop(0)
        node = route[-1]

        if node == end:
            return route

        if node not in visited:
            visited.add(node)
            for neighbor in graph.get(node, []):
                new_route = route + [neighbor]
                queue.append(new_route)
    
    print("Not found, try again")

def dfs(graph, start, end):
    stack = [[start]]
    visited = set()

    while stack:
        route = stack.pop()
        node = route[-1]

        if node == end:
            return route

        if node not in visited:
            visited.add(node)
            for neighbor in graph.get(node, []):
                new_route = route + [neighbor]
                stack.append(new_route)

    print("Not found, try again")

spiders = {
    '1': ['2', '3', '4'],
    '2': ['3', '8'],
    '3': ['4', '5', '6', '7', '8'],
    '4': ['5'],
    '5': ['6'],
    '6': ['7', '11'],
    '7': ['8', '9', '10', '11'],
    '8': ['9'],
    '9': ['10'],
    '10': ['11']
}

start_node = '1'
end_node = '11'

print("BFS route:", bfs(spiders, start_node, end_node))
print("DFS route:", dfs(spiders, start_node, end_node))
