import time
import numpy as np
from sklearn.svm import SVC
from sklearn.model_selection import StratifiedKFold
from input_data import read_data_sets

num_labeled = 5

dataSet = read_data_sets('I_feature.h5', num_labeled)
train_data = dataSet.train.images
train_label = dataSet.train.labels
test_data = dataSet.test.images
test_label = dataSet.test.labels

print('Train.......')

skf = StratifiedKFold(n_splits=4)

max_ac = []
cc = []
gg = []

for skf1, skf2 in skf.split(train_data, train_label):
    train_x = train_data[skf1]
    train_y = train_label[skf1]
    test_x = train_data[skf2]
    test_y = train_label[skf2]
    C = np.logspace(-2, 8, 11, base=2)
    gamma = np.logspace(-2, 8, 11, base=2)
    max_accuracy = 0
    best_c = 0
    best_g = 0
    for i in C:
        for j in gamma:
            svr = SVC(decision_function_shape='ovo', kernel='rbf', C=i, gamma=j)
            svr.fit(train_x, train_y)
            accuracy = svr.score(test_x, test_y)
            print('Max Accuracy:', max_accuracy, 'Test Accuracy:', accuracy)
            if accuracy > max_accuracy:
                max_accuracy = accuracy
                best_c = i
                best_g = j
    max_ac.append(max_accuracy)
    cc.append(best_c)
    gg.append(best_g)

max_temp = 0
index = 0
for i in range(4):
    if max_ac[i] > max_temp:
        max_temp = max_ac[i]
        index = i

svr = SVC(decision_function_shape='ovo', kernel='rbf', C=cc[index], gamma=gg[index])
svr.fit(train_data, train_label)
accuracy = svr.score(test_data, test_label)
print('Accuracy:', accuracy)

end = time.time()
