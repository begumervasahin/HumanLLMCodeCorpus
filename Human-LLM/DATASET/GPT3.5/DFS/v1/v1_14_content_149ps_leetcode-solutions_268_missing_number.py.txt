def missing_number(nums):
    # Initialize lower bound, upper bound, and length
    l = 0
    u = max(nums)
    length = u - l + 1
    print(l, u, length)  # For debugging purposes

    # Initialize index and iterate until length becomes 0
    i = 0
    while length != 0:
        if l + i in nums:
            i += 1
            length -= 1
        else:
            return l + i  # Return the missing number
    return l + i  # Return the missing number (if nums is empty, it returns l)

# Example usage
nums = [0]  # List containing numbers
print(missing_number(nums))  # Call the function and print the result
