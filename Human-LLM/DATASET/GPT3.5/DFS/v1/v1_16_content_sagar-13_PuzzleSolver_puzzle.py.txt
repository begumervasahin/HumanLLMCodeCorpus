from abc import ABC, abstractmethod

class Puzzle(ABC):
    """
    Snapshot of a full-information puzzle, which may be solved, unsolved,
    or even unsolvable.
    """

    def fail_fast(self):
        """
        Return True iff Puzzle self can never be extended to a solution.
        Override this in a subclass where you can determine early that
        this Puzzle can't be solved.
        @type self: Puzzle
        @rtype: bool
        """
        return False

    @abstractmethod
    def is_solved(self):
        """
        Return True iff Puzzle self is solved.
        This is an abstract method that must be implemented
        in a subclass.
        @type self: Puzzle
        @rtype: bool
        """
        pass

    @abstractmethod
    def extensions(self):
        """
        Return list of legal extensions of Puzzle self.
        This is an abstract method that must be implemented
        in a subclass.
        @type self: Puzzle
        @rtype: list[Puzzle]
        """
        pass


# Example subclass of Puzzle
class MyPuzzle(Puzzle):
    def __init__(self, state):
        self.state = state

    def is_solved(self):
        return self.state == "Solved"

    def extensions(self):
        return [MyPuzzle("Extension 1"), MyPuzzle("Extension 2")]

# Example usage
if __name__ == "__main__":
    puzzle = MyPuzzle("Unsolved")
    print("Is solved:", puzzle.is_solved())
    print("Extensions:", puzzle.extensions())
    print("Fail fast:", puzzle.fail_fast())
