import queue
from mymap import myMap, heuristic
from modules import showResultWithAttr, showStep, aweSomeSort

def AStar(start, goal):
    q = queue.deque()
    q.append(start)
    previous = []
    
    # Initialize the previous dictionary with initial values
    for city in myMap.keys():
        previous.append((city, {'from': None, 'total_cost': heuristic[city]}))
    previous = dict(previous)
    
    counter = 0
    while True:
        counter += 1
        showStep(counter, q, previous)
        curCity = q.popleft()
        
        if curCity == goal:
            print('Success!')
            showResultWithAttr(previous, start, goal)
            return True
        
        curCityTotalCost = previous[curCity]['total_cost'] - heuristic[curCity]
        
        if curCity != goal:
            for city in myMap[curCity].keys():
                cityTotalCost = previous[city]['total_cost']
                totalCost = myMap[curCity][city]['cost'] + curCityTotalCost + heuristic[city]
                
                if previous[city]['from'] is None or totalCost < cityTotalCost:
                    if city in q:
                        q.remove(city)
                    q.append(city)
                    previous[city]['from'] = curCity
                    previous[city]['total_cost'] = totalCost
        
        if len(q) == 0:
            print('Fail!')
            return False
        
        q = queue.deque(aweSomeSort(q, previous))

# Example usage
start_city = 'A'
goal_city = 'G'
AStar(start_city, goal_city)
