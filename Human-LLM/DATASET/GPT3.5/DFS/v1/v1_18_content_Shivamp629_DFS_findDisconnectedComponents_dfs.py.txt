from collections import defaultdict

class Graph:
    def __init__(self):
        self.graph = defaultdict(list)

    def add_edge(self, v1, v2):
        self.graph[v1].append(v2)

    def dfs(self, start):
        visited_set = set()
        count_graphs = 0
        components = []

        for neighbor in self.graph.keys():
            vertices_in_component = []
            if neighbor not in visited_set:
                self.dfs_helper(neighbor, visited_set, vertices_in_component)
                components.append(vertices_in_component)
                count_graphs += 1

        output_text = "Graph has {} components and the components each contain the following values {}"
        print(output_text.format(count_graphs, components))

    def dfs_helper(self, node, visited_set, vertices_in_component):
        visited_set.add(node)
        vertices_in_component.append(node)

        for neighbor in self.graph.get(node):
            if neighbor not in visited_set:
                self.dfs_helper(neighbor, visited_set, vertices_in_component)
