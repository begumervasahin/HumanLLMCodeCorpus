g = {'A':['B','D','G'],
        'B':['A','E','F'],
         'C':['F','H'],
         'D':['A','F'],
         'E':['B','G'],
         'F':['B','C','D'],
         'G':['A','E'],
         'H':['C']
    }
def dfs(g,start):
    if start not in g:
        print("Start state not in graph")
        return False
    visited = []
    stack = [start]
    while stack:
        curr = stack.pop()
        if curr not in visited:
            visited.append(curr)
            for value in g[curr]:
                if value not in visited:
                    stack.append(value)
    return visited
def bfs(g,start):
    if start not in g:
        return None
    visited = []
    queue = [start]
    while queue:
        curr = queue.pop(0)
        if curr not in visited:
            visited.append(curr)
            for value in g[curr]:
                if value not in visited:
                    queue.append(value)
    return visited
