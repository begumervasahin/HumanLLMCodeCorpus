peta =  {'A':set(['B']),
         'B':set(['A','C','F','G']),
         'C':set(['B','D']),
         'D':set(['C','E']),
         'E':set(['D','F']),
         'F':set(['G','E','B']),
         'G':set(['F','H','B']),
         'H':set(['I','G','L']),
         'I':set(['H','J']),
         'J':set(['I','K']),
         'K':set(['L','J']),
         'L':set(['K','H'])}
def dfs(graf, mulai, tujuan):
    stack = [[mulai]]
    visited = set()
    while stack:
        panjang_tumpukan = len(stack)-1
        jalur = stack.pop(panjang_tumpukan)
        state = jalur[-1]
        if state == tujuan:
            return jalur
        elif state not in visited:
            for cabang in graf.get(state, []):
                jalur_baru = list(jalur)
                jalur_baru.append(cabang)
                stack.append(jalur_baru)
            visited.add(state)
        isi = len(stack)
        if isi == 0:
            print("Tidak ditemukan")
print(dfs(peta,'A','I'))
