import json
import rep
import busqueda
import sys
if(len(sys.argv) != 3):
    print('Debe ingresar dos argumentos en comando de línea:')
    print('Tipo de búsqueda: bfs, dfs')
    print('Mostrar Q: si, no')
    quit()
if(sys.argv[1] != 'bfs' and sys.argv[1] != 'dfs'):
    print('Por favor ingrese argumento N°1 válido: bfs, dfs')
    quit()
if(sys.argv[2] != 'si' and sys.argv[2] != 'no'):
    print('Por favor ingrese argumento N°2 válido: si, no')
    quit()
if(sys.argv[2] == 'si'):
    mostrarQ = True
else:
    mostrarQ =False
with open('data.json') as json_file:
    data = json.load(json_file)
    print(data['estadoInicial'])
    print(data['estadoObjetivo'])
world = rep.setWorld(data)
inicial = world['inicial']
final = world['final']
solucion = busqueda.buscarSolucion(inicial, final, sys.argv[1], mostrarQ)
print('Pasos para la solucion:')
for paso in solucion['estadosAnteriores']:
    print(paso)
print('\nOperadores aplicados:')
for operacion in solucion['operadores']:
    print(operacion)
