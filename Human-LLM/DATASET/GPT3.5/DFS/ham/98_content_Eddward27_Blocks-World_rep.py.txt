import clase
def setWorld(data):
    inicial = setEstado(data['estadoInicial'])
    final = setEstado(data['estadoObjetivo'])
    world = { 'inicial': inicial, 'final': final }
    return world
def setEstado(arrTorres):
    estado = { 'operadores': [], 'estados': [], 'estadosAnteriores': []}
    for strTorre in arrTorres:
        torre = list(strTorre)
        for i in range(0,len(torre)):
            nombre = torre[i]
            on = None
            clear = False
            onTable = True
            if i < len(torre)-1:
                on = torre[i+1]
                onTable = False
            if i == 0:
                clear = True
            estado['estados'].append(clase.cubo(nombre, on, clear, onTable))
    estado['estadosAnteriores'].append(estadoString(estado['estados']))
    return estado
def mismoEstado(estado1, estado2):
    torres1 = 0
    torres2 = 0
    for cantidad in estado1:
        torres1+=1
    for cantidad in estado2:
        torres2+=1
    if torres1 != torres2:
        return False
    for i in range(0, len(estado1)):
        estadoEncontrado = False
        for j in range(0, len(estado2)):
            if(estado1[i].nombre == estado2[j].nombre):
                if(estado1[i].on == estado2[j].on and estado1[i].onTable == estado2[j].onTable and estado1[i].clear == estado2[j].clear):
                    estadoEncontrado = True
                    break
        if not estadoEncontrado:
            return False
    return True
def estadoString(estado):
    estadoSTR = []
    for cubosRep in estado:
        if cubosRep.clear:
            estadoSTR.append(cubosRep.nombre)
    for i in range(0,len(estadoSTR)):
        notLast = True
        cuboActual = estadoSTR[i]
        while notLast:
            for j in range(0,len(estado)):
                if cuboActual == estado[j].nombre:
                    if estado[j].on == None:
                        notLast = False
                        break
                    estadoSTR[i] = estadoSTR[i] + estado[j].on
                    cuboActual = estado[j].on
                    break
    return estadoSTR
