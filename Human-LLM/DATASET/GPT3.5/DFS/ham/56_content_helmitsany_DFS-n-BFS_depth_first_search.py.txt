peta =  {'A':set(['B','C']),
         'B':set(['A','E']),
         'C':set(['A','D','F','K']),
         'D':set(['C','E','J']),
         'E':set(['B','D','H']),
         'F':set(['C','G']),
         'G':set(['F','J']),
         'H':set(['E','I']),
         'I':set(['H','J','L']),
         'J':set(['D','G','I']),
         'K':set(['C']),
         'L':set(['I'])}
def dfs(graf, mulai, tujuan):
    stack = [[mulai]]
    visited = set()
    while stack:
        panjang_tumpukan = len(stack)-1
        jalur = stack.pop(panjang_tumpukan)
        state = jalur[-1]
        if state == tujuan:
            return jalur
        elif state not in visited:
            for cabang in graf.get(state, []):
                jalur_baru = list(jalur)
                jalur_baru.append(cabang)
                stack.append(jalur_baru)
            visited.add(state)
        isi = len(stack)
        if isi == 0:
            print("Tidak ditemukan")
print(dfs(peta,'A','H'))
