# Define the dimensions of the Pascal's triangle
num_rows = 1000
num_cols = 100

# Initialize a 2D list to store Pascal's triangle values
pascals_triangle = [[None for _ in range(num_cols + 1)] for _ in range(num_rows + 1)]

# Calculate Pascal's triangle values
for row in range(1, num_rows + 1):
    for col in range(1, num_cols + 1):
        if col == 1:
            # Set the first column values to the row number
            pascals_triangle[row][col] = row
        elif col > row:
            # If column index exceeds row index, set value to 0
            pascals_triangle[row][col] = 0
        else:
            # Calculate values using the recursive Pascal's triangle formula
            pascals_triangle[row][col] = pascals_triangle[row - 1][col] + pascals_triangle[row - 1][col - 1]

# Print the desired value from Pascal's triangle
print(pascals_triangle[num_rows][num_cols])
