import sys

def check_permutation(string):
    # Flags to track if each direction is present
    has_R = False
    has_L = False
    has_U = False
    has_D = False
    
    # Check each character in the string
    for char in string:
        if char == "R":
            has_R = True
        elif char == "L":
            has_L = True
        elif char == "U":
            has_U = True
        elif char == "D":
            has_D = True
    
    # Check if all directions are present
    if not (has_R and has_L and has_U and has_D):
        print("Invalid method parameter!")
        sys.exit()

def check_numbers(array):
    # Flatten the 2D array
    flat_list = [num for sublist in array for num in sublist]
    
    # Check for duplicate numbers
    for i in range(len(flat_list)):
        for j in range(i + 1, len(flat_list)):
            if flat_list[i] == flat_list[j]:
                print("Incorrect numbering of puzzle elements")
                sys.exit()

def generate_pattern(rows, columns):
    # Initialize counter for numbering elements
    counter = 1
    array = []
    
    # Generate the pattern
    for i in range(rows):
        row = []
        for j in range(columns):
            if counter < rows * columns:
                row.append(counter)
            else:
                row.append(0)
            counter += 1
        array.append(row)
    
    # Print and return the generated pattern
    print(array)
    return array
