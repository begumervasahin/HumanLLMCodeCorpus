# Import necessary classes from modules
from Webcrawler import Crawler
from bfs import BFS

# Define the URL to start crawling
url = "https://en.wikipedia.org/wiki/Sustainable_energy"

# Create an instance of the BFS search algorithm
searcher = BFS()

# Create a web crawler instance with the specified URL and search algorithm
myCrawler = Crawler(url, searcher)

# Add '/wiki/' to the mandatory list of URLs to visit during crawling
myCrawler.addToMandatoryList('/wiki/')

# Add ':' to the disallowed list of characters in URLs
myCrawler.addToDisallowedList(':')

# Set the focused text to 'solar', indicating that the crawler should search for this text on web pages
myCrawler.setFocusedText('solar')

# Start the crawling process
myCrawler.start()
