import cmd

class MyCLI(cmd.Cmd):
    def __init__(self):
        super().__init__()
        self.current_path = ''
        self.prompt = "FYP >>> "

    def do_hi(self, arg):
        """Command to print a greeting message."""
        print("Hello World.")

def start_cli():
    """Starts the command-line interface."""
    cli = MyCLI()
    cli.cmdloop()

if __name__ == '__main__':
    start_cli()
