class Solution:
    def zigzagLevelOrder(self, root):
        # Check if the root is None
        if not root:
            return []

        # Initialize a list to store the result
        result = []
        # Initialize a list to store nodes at the current level
        current_level = [root]
        # Initialize a flag to track the direction of traversal
        left_to_right = True

        # Continue traversal while there are nodes at the current level
        while current_level:
            # Initialize a list to store values at the current level
            level_res = []
            # Initialize a list to store nodes at the next level
            next_level = []

            # Process nodes at the current level
            while current_level:
                # Pop a node from the current level
                node = current_level.pop()
                # Append the value of the node to the level result
                level_res.append(node.val)

                # Add child nodes to the next level based on the traversal direction
                if left_to_right:
                    if node.left:
                        next_level.append(node.left)
                    if node.right:
                        next_level.append(node.right)
                else:
                    if node.right:
                        next_level.append(node.right)
                    if node.left:
                        next_level.append(node.left)

            # Update the current level with nodes from the next level
            current_level = next_level
            # Toggle the traversal direction
            left_to_right = not left_to_right

            # Append the level result to the final result
            result.append(level_res)

        return result
