import logging
from collections import defaultdict
from errno import ENOENT, ENOTEMPTY
from stat import S_IFDIR, S_IFLNK, S_IFREG
from sys import argv, exit
from time import time
from fuse import FUSE, FuseOSError, Operations, LoggingMixIn

if not hasattr(__builtins__, 'bytes'):
    bytes = str

MAX_BLOCK_SIZE = 512

class Memory(LoggingMixIn, Operations):
    """Implements a hierarchical file system using FUSE virtual filesystem.
    The file structure and data are stored in local memory.
    Data is lost when the filesystem is unmounted."""

    def __init__(self):
        self.files = {}
        self.data = defaultdict(bytes)
        self.fd = 0
        now = time()
        self.files['/'] = {
            'st_mode': (S_IFDIR | 0o755),
            'st_ctime': now,
            'st_mtime': now,
            'st_atime': now,
            'st_nlink': 2,
            'files': {}
        }

    def traverse(self, path, traverse_data=False):
        """Traverse the file structure to locate the current file."""
        target = self.data if traverse_data else self.files['/']
        for name in path.split('/'):
            if name:
                target = target[name]
        return target

    def traverse_parent(self, path, traverse_data=False):
        """Traverse to the parent directory of the current file."""
        target = self.data if traverse_data else self.files['/']
        components = path.split('/')
        parent = target
        for name in components[:-1]:
            if name:
                parent = parent[name]
        return parent, components[-1]

    def chmod(self, path, mode):
        """Change the mode (permissions) of a file."""
        node = self.traverse(path)
        node['st_mode'] &= 0o770000
        node['st_mode'] |= mode
        return 0

    def chown(self, path, uid, gid):
        """Change the owner and group of a file."""
        node = self.traverse(path)
        node['st_uid'] = uid
        node['st_gid'] = gid

    def create(self, path, mode):
        """Create a new file."""
        parent, name = self.traverse_parent(path)
        parent['files'][name] = {
            'st_mode': (S_IFREG | mode),
            'st_nlink': 1,
            'st_size': 0,
            'st_ctime': time(),
            'st_mtime': time(),
            'st_atime': time()
        }
        self.fd += 1
        data_node, _ = self.traverse_parent(path, traverse_data=True)
        data_node[name] = []
        print("file_created", data_node[name])
        return self.fd
