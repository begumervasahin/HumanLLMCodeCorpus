import requests
from bs4 import BeautifulSoup
import re
import time

def my_crawler(seed_url):
    parent_urls = [seed_url]
    child_urls = []
    MAX_PARENTS = 1000
    MAX_DEPTH = 7
    current_depth = 1

    while len(parent_urls) < MAX_PARENTS and current_depth <= MAX_DEPTH:
        for parent_url in parent_urls:
            child_urls.extend(crawl_child_links(parent_url))
            if len(child_urls) >= MAX_PARENTS:
                break
        
        parent_urls.extend([child_url for child_url in child_urls if child_url not in parent_urls])
        current_depth += 1

    save_urls_to_file(parent_urls)

def crawl_child_links(url):
    time.sleep(1)
    BASE_URL = "https://en.wikipedia.org"
    child_links = []

    page = requests.get(url)
    soup = BeautifulSoup(page.text, 'html.parser')
    content_div = soup.find('div', {'id': 'mw-content-text'})

    remove_unwanted_elements(soup)

    for link in content_div.find_all('a', {'href': re.compile("^/wiki")}):
        if ':' not in link.get('href'):
            link_text = BASE_URL + link.get('href')
            refined_link = link_text.split('#')[0]
            if len(refined_link) > 1 and refined_link not in child_links:
                child_links.append(refined_link)

    return child_links

def remove_unwanted_elements(soup):
    for element in soup(['div', 'table', 'ol', 'span', 'sup', 'i']):
        element.decompose()

def save_urls_to_file(urls):
    with open('Task-1-E-URLs.txt', 'w') as file:
        for index, url in enumerate(urls, start=1):
            file.write(f"{index} {url}\n")

my_crawler('https://en.wikipedia.org/wiki/Tropical_cyclone')
