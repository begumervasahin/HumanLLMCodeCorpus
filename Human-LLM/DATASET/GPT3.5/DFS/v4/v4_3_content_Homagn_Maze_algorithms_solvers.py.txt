import numpy as np
import copy

class Solver:
    def __init__(self, state_space, num_cols, start, goal, technique):
        self.start = start
        self.num_cols = num_cols
        self.goal = goal
        self.state_space = state_space
        self.path = []
        self.exact_path = []
        self.explored = []
        print("Initialized solver, using technique", technique)
        
        if technique == "bfs":
            self.bfs()
        elif technique == "dfs":
            self.dfs()
        elif technique == "astar":
            self.astar()

    def path_from_visited(self, path):
        p = []
        pointer = self.goal
        while pointer != self.start:
            p.append(pointer)
            pointer = copy.copy(self.state_space[pointer]["parent "])
        p.reverse()
        return p

    def expand_stack(self, stack):
        head = stack.pop(0)
        connects = head["connections "]
        if not connects:
            return stack
        for i in connects:
            if i not in self.explored:
                stack.append(self.state_space[i])
        return stack

    def not_worth_visiting(self, node, closed_nodes, open_nodes):
        for i in open_nodes:
            if i["parent "] == self.state_space[node["name "]]["parent "] and i["g"] + i["h"] < node["g"] + node["h"]:
                return True
        for i in closed_nodes:
            if i["parent "] == self.state_space[node["name "]]["parent "] and i["g"] + i["h"] < node["g"] + node["h"]:
                return True
        return False

    def heuristic_cost(self, current, goal):
        y_dist = np.abs((goal - current) / self.num_cols)
        x_dist = np.abs((goal - current)) % self.num_cols
        return y_dist + x_dist

    def expand_value_stack(self, stack):
        f_vals = [i["g"] + i["h"] for i in stack]
        head = stack.pop(np.argmin(f_vals))
        connects = head["connections "]
        for i in connects:
            self.state_space[i]["parent "] = head["name "]
            try:
                self.state_space[i]["connections "].remove(head["name "])
            except:
                pass
            self.state_space[i]["g"] = head["g"] + 1
            self.state_space[i]["h"] = self.heuristic_cost(self.state_space[i]["name "], self.goal)
        if not connects:
            return stack
        open_list = copy.copy(stack)
        for i in connects:
            if not self.not_worth_visiting(self.state_space[i], self.explored, open_list):
                stack.append(self.state_space[i])
        self.explored.append(head)
        return stack

    def explore_stack(self, stack):
        tail = stack[-1]
        connects = tail["connections "]
        if not connects:
            stack.pop(-1)
            return stack
        for i in connects:
            if i not in self.explored:
                stack.append(self.state_space[i])
                return stack
        stack.pop(-1)
        return stack

    def bfs(self):
        stack = [self.state_space[self.start]]
        while stack:
            current = stack.pop(0)
            self.explored.append(current["name "])
            self.path.append(current["name "])
            if current["name "] == self.goal:
                print("CONGRATULATIONS! A path exists, printing all expanded nodes")
                print(self.path)
                return
            stack.extend([self.state_space[neighbor] for neighbor in current["connections "] if neighbor not in self.explored])
        print("WARNING! A valid path does not exist, printing all expanded nodes")
        print(self.path)

    def dfs(self):
        stack = [self.state_space[self.start]]
        while stack:
            current = stack.pop()
            self.explored.append(current["name "])
            self.path.append(current["name "])
            if current["name "] == self.goal:
                print("CONGRATULATIONS! A path exists, printing all expanded nodes")
                print(self.path)
                return
            stack.extend([self.state_space[neighbor] for neighbor in current["connections "] if neighbor not in self.explored])
        print("WARNING! A valid path does not exist, printing all expanded nodes")
        print(self.path)

    def astar(self):
        stack = [self.state_space[self.start]]
        for i in self.state_space:
            i["g"] = 0
            i["h"] = 0
        self.state_space[self.start]["parent "] = -1
        while stack:
            f_vals = [i["g"] + i["h"] for i in stack]
            head = stack.pop(np.argmin(f_vals))
            connects = head["connections "]
            self.explored.append(head["name "])
            self.path.append(head["name "])
            if head["name "] == self.goal:
                print("CONGRATULATIONS! A path exists, printing all expanded nodes")
                print(self.path)
                return
            for i in connects:
                neighbor = self.state_space[i]
                neighbor_g = head["g"] + 1
                if neighbor not in stack and i not in self.explored:
                    stack.append(neighbor)
                    neighbor["parent "] = head["name "]
                    neighbor["g"] = neighbor_g
                    neighbor["h"] = self.heuristic_cost(neighbor["name "], self.goal)
                elif neighbor_g < neighbor["g"]:
                    neighbor["parent "] = head["name "]
                    neighbor["g"] = neighbor_g
        print("WARNING! A valid path does not exist, printing all expanded nodes")
        print(self.path)
