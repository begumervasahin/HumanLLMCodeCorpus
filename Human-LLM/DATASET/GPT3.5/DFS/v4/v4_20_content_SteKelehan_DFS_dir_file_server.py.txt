import requests
from collections import defaultdict

class DirFileServer:
    def __init__(self):
        self.dir_id_counter = 1
        self.dirs = defaultdict(list)
        self.file_server_base_url = 'http://localhost:'

    def _extract_message(self, **kwargs):
        if 'message' in kwargs and kwargs['message'] is not None:
            return kwargs['message']
        else:
            return None

    def get_files_on_all_servers(self, *args, **kwargs):
        message = self._extract_message(**kwargs)
        if message is None:
            return None
        else:
            files = []
            for server_id in self.dirs.keys():
                url = f"{self.file_server_base_url}{server_id}/files"
                response = requests.get(url, **kwargs)
                files.append(response)
            return files

    def add_file_server(self, *args, **kwargs):
        message = self._extract_message(**kwargs)
        if message is None:
            return None
        else:
            return requests.post()

    def add_one_file(self, args, **kwargs):
        message = self._extract_message(**kwargs)
        if message is None:
            return None
        else:
            min_files = float('inf')
            min_server = None
            for server_id, files in self.file_server.items():
                num_files = len(files)
                if num_files < min_files:
                    min_files = num_files
                    min_server = server_id
            url = f"{self.file_server_base_url}{min_server}/file"
            return requests.post(url, **kwargs)

    def get_file(self, *args, **kwargs):
        message = self._extract_message(**kwargs)
        if message is None:
            return None
        else:
            if '_id' in message and message['_id'] is not None:
                for server_id, files in self.dirs.items():
                    if message['_id'] in files:
                        url = f"{self.file_server_base_url}{server_id}/file"
                        return requests.get(url, json={'_id': message['_id']})
            else:
                if 'name' in message and message['name'] is not None:
                    return self.get_files_on_all_servers(message=message['name'])
                else:
                    return None
