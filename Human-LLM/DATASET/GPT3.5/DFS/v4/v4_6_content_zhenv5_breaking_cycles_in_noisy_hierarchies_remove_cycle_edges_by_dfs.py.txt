import networkx as nx
import argparse
import numpy as np
from measures import F1
from file_io import write_pairs_to_file
import sys

# Set recursion limit to avoid potential stack overflow
sys.setrecursionlimit(5500000)

def dfs_visit_recursively(graph, node, node_colors, edges_to_be_removed):
    '''
    Depth-first search (DFS) to recursively visit nodes and remove back edges.
    
    Node colors:
    0: White, not visited
    1: Grey, being visited
    2: Black, already visited
    '''
    node_colors[node] = 1  # Mark node as being visited
    successors = list(graph.successors(node))
    np.random.shuffle(successors)  # Randomly shuffle successors to break ties
    for child in successors:
        if node_colors[child] == 0:
            dfs_visit_recursively(graph, child, node_colors, edges_to_be_removed)
        elif node_colors[child] == 1:
            edges_to_be_removed.append((node, child))  # Record back edges
    node_colors[node] = 2  # Mark node as already visited

def dfs_remove_back_edges(graph_file, nodetype=int):
    '''
    Perform DFS to remove back edges in a directed graph.
    '''
    graph = nx.read_edgelist(graph_file, create_using=nx.DiGraph(), nodetype=nodetype)
    node_colors = {node: 0 for node in graph.nodes()}  # Initialize nodes' colors
    edges_to_be_removed = []

    # Perform DFS for each unvisited node
    for node in np.random.permutation(list(graph.nodes())):
        if node_colors[node] == 0:
            dfs_visit_recursively(graph, node, node_colors, edges_to_be_removed)

    # Write removed edges to a file
    edges_to_be_removed_file = graph_file[:-6] + "_removed_by_dfs.edges"
    print("Edges to be removed, saved in file:", edges_to_be_removed_file)
    write_pairs_to_file(edges_to_be_removed, edges_to_be_removed_file)

    return edges_to_be_removed

def dfs_performance(graph_file, gt_edges_file):
    '''
    Evaluate DFS performance in removing back edges.
    '''
    edges_to_be_removed = dfs_remove_back_edges(graph_file)
    from measures import report_performance
    report_performance(gt_edges_file, edges_to_be_removed, "DFS")

if __name__ == "__main__":
    '''
    DFS Remove Back Edges
    '''
    parser = argparse.ArgumentParser()
    parser.add_argument("-g", "--graph_file", default=" ", help="Input graph file name (edges list)")
    parser.add_argument("-t", "--gt_edges_file", default=None, help="Ground truth edges file")
    args = parser.parse_args()
    print("Graph file:", args.graph_file)
    dfs_performance(args.graph_file, args.gt_edges_file)
