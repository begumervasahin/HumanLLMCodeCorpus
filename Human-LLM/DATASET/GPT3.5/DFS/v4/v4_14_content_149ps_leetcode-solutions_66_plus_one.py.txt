def plus_one(nums):
    # Convert the list of digits to an integer
    num_as_int = int(''.join(map(str, nums)))

    # Increment the integer by 1
    incremented_num = num_as_int + 1

    # Convert the incremented integer back to a list of digits
    result = [int(digit) for digit in str(incremented_num)]

    return result

# Example usage
nums = [1, 2, 4, 5, 3, 2, 2, 9]
result = plus_one(nums)
print("Result:", result)
