def shortest_path(graph, start, goal):
    """
    Find the shortest path from the start node to the goal node using breadth-first search.
    """
    explored = []  # List to track explored nodes
    queue = [[start]]  # Queue to store paths

    # Check if the start node is the goal node
    if start == goal:
        return "Start and goal nodes are the same."

    # Explore paths until the queue is empty
    while queue:
        path = queue.pop(0)  # Get the first path from the queue
        node = path[-1]  # Get the current node from the path

        # Check if the current node has not been explored
        if node not in explored:
            neighbours = graph[node]  # Get the neighbours of the current node
            for neighbour in neighbours:
                new_path = list(path)  # Create a new path by appending the neighbour
                new_path.append(neighbour)
                queue.append(new_path)  # Add the new path to the queue
                # Check if the neighbour is the goal node
                if neighbour == goal:
                    return new_path
            explored.append(node)  # Mark the current node as explored

    return "Sorry, the selected node does not exist in the graph."


# Define the graph
graph = {
    'A': {'B'},
    'B': {'A', 'C'},
    'C': {'B', 'D', 'H', 'I'},
    'D': {'C', 'E', 'F', 'H'},
    'E': {'D'},
    'F': {'D', 'G'},
    'G': {'F', 'H'},
    'H': {'C', 'D', 'G', 'L'},
    'I': {'C', 'J', 'K'},
    'J': {'I'},
    'K': {'I', 'L'},
    'L': {'H', 'K'}
}

# Get input from the user
start_node = input("Enter the start node: ")
goal_node = input("Enter the goal node: ")

# Call the function and print the result
print(shortest_path(graph, start_node, goal_node))
