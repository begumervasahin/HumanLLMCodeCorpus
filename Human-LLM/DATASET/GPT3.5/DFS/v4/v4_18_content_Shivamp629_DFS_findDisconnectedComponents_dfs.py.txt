from collections import defaultdict

class Graph:
    def __init__(self):
        # Initialize an empty graph using defaultdict
        self.graph = defaultdict(list)

    def add_edge(self, vertex1, vertex2):
        # Add an edge between vertices vertex1 and vertex2
        self.graph[vertex1].append(vertex2)

    def dfs(self, start_vertex):
        # Initialize an empty set to keep track of visited vertices
        visited_set = set()

        # Initialize variables to count graphs and store components
        num_components = 0
        components = []

        # Iterate through each vertex in the graph
        for neighbor in self.graph.keys():
            vertices_in_component = []

            # If the vertex has not been visited, perform DFS traversal
            if neighbor not in visited_set:
                self.dfs_helper(neighbor, visited_set, vertices_in_component)
                components.append(vertices_in_component)
                num_components += 1

        # Print the number of components and their respective vertices
        output_text = "Graph has {} components and the components each contain the following values {}"
        print(output_text.format(num_components, components))

    def dfs_helper(self, node, visited_set, vertices_in_component):
        # Mark the current node as visited and add it to the component
        visited_set.add(node)
        vertices_in_component.append(node)

        # Recursively traverse all neighbors of the current node
        for neighbor in self.graph.get(node):
            if neighbor not in visited_set:
                self.dfs_helper(neighbor, visited_set, vertices_in_component)
