class Solution:
    """
    A class to generate all possible letter combinations of a phone number.

    Attributes:
        digit_to_letters (dict): A dictionary mapping digits to their corresponding letters.
    """

    digit_to_letters = {
        '2': ['a', 'b', 'c'],
        '3': ['d', 'e', 'f'],
        '4': ['g', 'h', 'i'],
        '5': ['j', 'k', 'l'],
        '6': ['m', 'n', 'o'],
        '7': ['p', 'q', 'r', 's'],
        '8': ['t', 'u', 'v'],
        '9': ['w', 'x', 'y', 'z']
    }

    def letterCombinations(self, digits):
        """
        Generate all possible letter combinations for a given digit string.

        Args:
            digits (str): A string containing digits.

        Returns:
            list: A list of all possible letter combinations.
        """
        results = []
        if digits:
            self.generate_combinations(digits, 0, '', results)
        return results

    def generate_combinations(self, digits, index, current_combination, results):
        """
        Recursively generate all possible letter combinations for the given digit string.

        Args:
            digits (str): A string containing digits.
            index (int): The current index in the digits string.
            current_combination (str): The current combination of letters.
            results (list): A list to store the generated combinations.
        """
        if index == len(digits):
            results.append(current_combination)
            return
        for letter in self.digit_to_letters[digits[index]]:
            self.generate_combinations(digits, index + 1, current_combination + letter, results)
