def find_missing_number(nums):
    # Determine the lower bound (l), upper bound (u), and range length
    lower_bound = 0
    upper_bound = max(nums)
    range_length = upper_bound - lower_bound + 1
    print("Lower Bound:", lower_bound)
    print("Upper Bound:", upper_bound)
    print("Range Length:", range_length)

    # Initialize index (i) to track missing numbers
    i = 0
    while range_length != 0:
        # Check if the number lower_bound + i is present in the list nums
        if lower_bound + i in nums:
            i += 1
            range_length -= 1
        else:
            # If the number lower_bound + i is missing, return it as the missing number
            return lower_bound + i

    # If all numbers from lower_bound to upper_bound are present, return upper_bound + 1
    return lower_bound + i

# Example usage
numbers = [0]  # List containing numbers
missing_number = find_missing_number(numbers)
print("Missing Number:", missing_number)
