from queue import Queue, PriorityQueue
from math import sqrt

class Node:
    def __init__(self, value):
        self.value = value
        self.id = None
        self.up = None
        self.down = None
        self.left = None
        self.right = None
        self.previousNode = None
        self.edgeCost = None
        self.gOfN = None
        self.hOfN = None
        self.heuristicFn = None
        self.visited = False

class SearchAlgorithms:
    def __init__(self, mazeStr, edgeCost=None):
        self.path = []
        self.fullPath = []
        self.totalCost = -1
        self.grid = []
        self.list_of_rows = []
        self.id_counter = 0
        self.start_id = 0
        self.row_number = 0
        self.col_number = 0
        self.cost_counter = 0
        self.end_x = 0
        self.end_y = 0
        self.stack = []
        self.myqueue = Queue()
        self.pqueue = PriorityQueue()

        self.initializeGrid(mazeStr, edgeCost)

    def initializeGrid(self, mazeStr, edgeCost):
        self.grid.clear()
        self.list_of_rows.clear()
        self.fullPath.clear()
        self.stack.clear()

        for char in mazeStr:
            if char == ',':
                continue
            if char == " ":
                self.row_number += 1
        self.row_number += 1

        for char in mazeStr:
            if char == ',':
                self.col_number += 1
            if char == " ":
                self.col_number += 1
                break

        self.list_of_rows = mazeStr.split()
        for row in self.list_of_rows:
            self.grid.append(row.split(","))

        for i in range(len(self.grid)):
            for j in range(len(self.grid[0])):
                if self.grid[i][j] == "S":
                    self.start_id = self.id_counter
                elif self.grid[i][j] == "E":
                    self.end_x = i
                    self.end_y = j
                node = Node(self.grid[i][j])
                node.id = self.id_counter
                self.id_counter += 1
                self.grid[i][j] = node

        for i in range(len(self.grid)):
            for j in range(len(self.grid[0])):
                if i != 0:
                    if self.grid[i-1][j].value != '':
                        self.grid[i][j].up = self.grid[i-1][j].id
                if i != len(self.grid) - 1:
                    if self.grid[i+1][j].value != '':
                        self.grid[i][j].down = self.grid[i+1][j].id
                if j != 0:
                    if self.grid[i][j-1].value != '':
                        self.grid[i][j].left = self.grid[i][j-1].id
                if j != len(self.grid[0]) - 1:
                    if self.grid[i][j+1].value != '':
                        self.grid[i][j].right = self.grid[i][j+1].id

        if edgeCost is not None:
            for i in range(len(self.grid)):
                for j in range(len(self.grid[0])):
                    self.grid[i][j].edgeCost = edgeCost[self.cost_counter]
                    self.cost_counter += 1

    def get_flatten(self, id):
        row = int(id // self.col_number)
        col = int(id % self.col_number)
        return row, col

    def heuristic_euclidean_distance(self, i, j):
        distance = sqrt((self.end_x - i) ** 2 + (self.end_y - j) ** 2)
        return distance

    def heuristic_absolute(self, i, j):
        distance = abs(i - self.end_x) + abs(j - self.end_y)
        return distance

    def get_children(self, row_idx, col_idx):
        children = []
        if self.grid[row_idx][col_idx].right is not None:
            children.append(self.grid[row_idx][col_idx].right)
        if self.grid[row_idx][col_idx].left is not None:
            children.append(self.grid[row_idx][col_idx].left)
        if self.grid[row_idx][col_idx].down is not None:
            children.append(self.grid[row_idx][col_idx].down)
        if self.grid[row_idx][col_idx].up is not None:
            children.append(self.grid[row_idx][col_idx].up)
        self.grid[row_idx][col_idx].visited = True
        return children

    def assign_costs(self):
        for i in range(len(self.grid)):
            for j in range(len(self.grid[0])):
                if self.grid[i][j].value == 'S':
                    self.grid[i][j].edgeCost = 0
                else:
                    self.grid[i][j].edgeCost = 1

    def DFS(self):
        self.stack.append(self.start_id)
        while len(self.stack) != 0:
            ID = self.stack.pop()
            row_idx, col_idx = self.get_flatten(ID)
            if self.grid[row_idx][col_idx].value == "E":
                self.fullPath.append(ID)
                break
            elif not self.grid[row_idx][col_idx].visited and ID not in self.stack:
                self.fullPath.append(ID)
                self.grid[row_idx][col_idx].visited = True
                children = self.get_children(row_idx, col_idx)
                self.stack.extend(children)
        return self.path, self.fullPath

    def BFS(self):
        self.myqueue.put(self.start_id)
        while not self.myqueue.empty():
            ID = self.myqueue.get()
            row_idx, col_idx = self.get_flatten(ID)
            if self.grid[row_idx][col_idx].value == "E":
                self.fullPath.append(ID)
                break
            elif not self.grid[row_idx][col_idx].visited:
                self.fullPath.append(ID)
                self.grid[row_idx][col_idx].visited = True
                children = self.get_children(row_idx, col_idx)
                children.reverse()
                self.myqueue.queue.extend(children)
        return self.path, self.fullPath

    def UCS(self):
        self.pqueue.put((0, self.start_id))
        while not self.pqueue.empty():
            cost, ID = self.pqueue.get()
            row_idx, col_idx = self.get_flatten(ID)
            if not self.grid[row_idx][col_idx].visited:
                self.fullPath.append(ID)
                if self.grid[row_idx][col_idx].value == 'E':
                    break
                children = self.get_children(row_idx, col_idx)
                children.reverse()
                self.grid[row_idx][col_idx].visited = True
                for childId in children:
                    child_row, child_col = self.get_flatten(childId)
                    child_node = self.grid[child_row][child_col]
                    if childId not in self.fullPath:
                        child_gOfN = cost + child_node.edgeCost
                        self.totalCost = child_gOfN
                        self.pqueue.put((self.totalCost, childId))
                        child_node.gOfN = child_gOfN
        return self.path, self.fullPath, self.totalCost

    def AStarEuclideanHeuristic(self):
        for i in range(len(self.grid)):
            for j in range(len(self.grid[0])):
                self.grid[i][j].hOfN = self.heuristic_euclidean_distance(i, j)
        self.pqueue.put((0, self.start_id))
        row_idx, col_idx = self.get_flatten(self.start_id)
        self.grid[row_idx][col_idx].gOfN = 0
        while not self.pqueue.empty():
            _, ID = self.pqueue.get()
            row_idx, col_idx = self.get_flatten(ID)
            cost = self.grid[row_idx][col_idx].gOfN
            if not self.grid[row_idx][col_idx].visited:
                self.fullPath.append(ID)
                if self.grid[row_idx][col_idx].value == 'E':
                    break
                children = self.get_children(row_idx, col_idx)
                children.reverse()
                self.grid[row_idx][col_idx].visited = True
                for childId in children:
                    child_row, child_col = self.get_flatten(childId)
                    child_node = self.grid[child_row][child_col]
                    if childId not in self.fullPath:
                        child_gOfN = cost + child_node.edgeCost
                        self.totalCost = child_gOfN
                        child_node.gOfN = child_gOfN
                        child_node.heuristicFn = child_node.gOfN + child_node.hOfN
                        self.pqueue.put((child_node.heuristicFn, childId))
        return self.path, self.fullPath, self.totalCost

    def AStarManhattanHeuristic(self):
        for i in range(len(self.grid)):
            for j in range(len(self.grid[0])):
                self.grid[i][j].hOfN = self.heuristic_absolute(i, j)
        self.assign_costs()
        self.pqueue.put((0, self.start_id))
        row_idx, col_idx = self.get_flatten(self.start_id)
        self.grid[row_idx][col_idx].gOfN = 0
        while not self.pqueue.empty():
            _, ID = self.pqueue.get()
            row_idx, col_idx = self.get_flatten(ID)
            cost = self.grid[row_idx][col_idx].gOfN
            if not self.grid[row_idx][col_idx].visited:
                self.fullPath.append(ID)
                if self.grid[row_idx][col_idx].value == 'E':
                    break
                children = self.get_children(row_idx, col_idx)
                children.reverse()
                self.grid[row_idx][col_idx].visited = True
                for childId in children:
                    child_row, child_col = self.get_flatten(childId)
                    child_node = self.grid[child_row][child_col]
                    if childId not in self.fullPath:
                        child_gOfN = cost + child_node.edgeCost
                        self.totalCost = child_gOfN
                        child_node.gOfN = child_gOfN
                        child_node.heuristicFn = child_node.gOfN + child_node.hOfN
                        self.pqueue.put((child_node.heuristicFn, childId))
        return self.path, self.fullPath, self.totalCost

def main():
    searchAlgo = SearchAlgorithms('S,.,.',
                                  [100, 2, 15, 60, 100, 30, 2, 100, 2, 2, 2, 40, 30, 2, 2, 100, 100, 3, 15, 30, 100, 2, 100, 0, 2, 100, 30])
    path, fullPath = searchAlgo.DFS()
    print('**DFS**\nPath is:', path, '\nFull Path is:', fullPath, '\n\n')

    searchAlgo = SearchAlgorithms('S,.,.')
    path, fullPath = searchAlgo.BFS()
    print('**BFS**\nPath is:', path, '\nFull Path is:', fullPath, '\n\n')

    searchAlgo = SearchAlgorithms('S,.,.',
                                  [100, 2, 15, 60, 100, 30, 2, 100, 2, 2, 2, 40, 30, 2, 2, 100, 100, 3, 15, 30, 100, 2, 100, 0, 2, 100, 30])
    path, fullPath, TotalCost = searchAlgo.UCS()
    print('**UCS**\nPath is:', path, '\nFull Path is:', fullPath, '\nTotal Cost:', TotalCost, '\n\n')

    searchAlgo = SearchAlgorithms('S,.,.',
                                  [100, 2, 15, 60, 100, 30, 2, 100, 2, 2, 2, 40, 30, 2, 2, 100, 100, 3, 15, 30, 100, 2, 100, 0, 2, 100, 30])
    path, fullPath, TotalCost = searchAlgo.AStarEuclideanHeuristic()
    print('**ASTAR with Euclidean Heuristic**\nPath is:', path, '\nFull Path is:', fullPath, '\nTotal Cost:', TotalCost, '\n\n')

    searchAlgo = SearchAlgorithms('S,.,.')
    path, fullPath, TotalCost = searchAlgo.AStarManhattanHeuristic()
    print('**ASTAR with Manhattan Heuristic**\nPath is:', path, '\nFull Path is:', fullPath, '\nTotal Cost:', TotalCost, '\n\n')

if __name__ == "__main__":
    main()
