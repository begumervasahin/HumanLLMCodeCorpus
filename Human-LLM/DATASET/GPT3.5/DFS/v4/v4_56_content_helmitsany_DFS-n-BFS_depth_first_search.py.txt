def depth_first_search(graph, start, end):
    """
    Perform Depth-First Search (DFS) on the given graph to find a path from the start node to the end node.

    Args:
        graph (dict): The graph represented as a dictionary.
        start (str): The starting node.
        end (str): The destination node.

    Returns:
        list or None: The path from the start node to the end node, or None if no path is found.
    """
    stack = [[start]]  # Initialize the stack with a list containing the starting node
    visited = set()    # Initialize a set to keep track of visited nodes

    while stack:
        current_path = stack.pop()               # Pop the last path from the stack
        current_state = current_path[-1]         # Get the current state (last node) from the path

        if current_state == end:                 # If the current state is the destination, return the path
            return current_path

        if current_state not in visited:         # If the current state has not been visited yet
            for neighbor in graph.get(current_state, []):   # Explore all adjacent nodes of the current state
                new_path = list(current_path) + [neighbor]  # Create a new path by extending the current path
                stack.append(new_path)                      # Push the new path onto the stack

            visited.add(current_state)            # Mark the current state as visited

    print("No path found")                         # If no path is found, print a message
    return None

# Define the graph
graph = {
    'A': {'B', 'C'},
    'B': {'A', 'E'},
    'C': {'A', 'D', 'F', 'K'},
    'D': {'C', 'E', 'J'},
    'E': {'B', 'D', 'H'},
    'F': {'C', 'G'},
    'G': {'F', 'J'},
    'H': {'E', 'I'},
    'I': {'H', 'J', 'L'},
    'J': {'D', 'G', 'I'},
    'K': {'C'},
    'L': {'I'}
}

# Find a path from node 'A' to node 'H' using DFS
path = depth_first_search(graph, 'A', 'H')
print(path)
