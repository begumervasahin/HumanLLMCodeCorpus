from DFS_Unlimited import DFSUnlimited
from uniform_cost import UniformCost
from bidirectional import Bidirectional
from Astar import AStar

class Problem1:
    def __init__(self, path=None):
        self.path = path or 'test1.txt'
        self.m = 0
        self.n = 0
        self.num_wall = 0
        self.walls = []
        self.start = [1, 1]
        self.goal = [self.m, self.n]
        self.move = [[0, -1, 'u'], [0, 1, 'd'], [1, 0, 'r'], [-1, 0, 'l']]
        self.initial_state()

    def actions(self, state):
        x, y = state
        res = []
        for dx, dy, action in self.move:
            new_x, new_y = x + dx, y + dy
            if 0 < new_x <= self.m and 0 < new_y <= self.n:
                t1 = [x, y, new_x, new_y]
                t2 = [new_x, new_y, x, y]
                if t1 not in self.walls and t2 not in self.walls:
                    res.append(action)
        return res

    def open_file(self):
        with open(self.path) as f:
            lines = f.read().splitlines()
        self.m, self.n = map(int, lines[0].split())
        self.num_wall = int(lines[1])
        for line in lines[2:]:
            a1, a2, a3, a4 = map(int, line.split())
            self.walls.append([a1, a2, a3, a4])
        self.goal = [self.m, self.n]

    def initial_state(self):
        self.open_file()
        return self.start

    def goal_test(self, state):
        x, y = state
        return x == self.m and y == self.n

    def result(self, state, action):
        x, y = state
        if action == 'u':
            return [x, y - 1]
        elif action == 'd':
            return [x, y + 1]
        elif action == 'r':
            return [x + 1, y]
        elif action == 'l':
            return [x - 1, y]

    def path_cost(self, c, state1, action, state2):
        pass

    def get_cost(self, state1, state2):
        return 1

    def get_reverse_action(self, action):
        if action == 'u':
            return 'd'
        elif action == 'd':
            return 'u'
        elif action == 'r':
            return 'l'
        elif action == 'l':
            return 'r'

    def heuristic(self, state):
        x, y = state
        d = ((self.goal[1] - y) ** 2 + (self.goal[0] - x) ** 2) ** 0.5
        return d

    def call_dfs_unlimited(self):
        dfs = DFSUnlimited(self.initial_state, self.actions, self.result, self.goal_test)
        dfs.search_dfs()

    def call_uniform_cost(self):
        uc = UniformCost(self.initial_state, self.actions, self.result, self.goal_test, self.get_cost)
        uc.search_uniform()

    def call_bidirectional(self):
        bd = Bidirectional(self.initial_state, self.actions, self.result, self.goal_test, self.get_cost,
                           self.get_reverse_action)
        bd.search_bidirectional(self.goal)

    def call_astar(self):
        astar = AStar(self.initial_state, self.actions, self.result, self.goal_test, self.get_cost, self.heuristic)
        astar.search_astar()

p = Problem1()
p.call_astar()
