import sys

class Node:
    def __init__(self, index):
        self.children = []
        self.index = index
        self.discovered = False
    
    def add_child(self, index):
        self.children.append(index)

class Graph:
    def __init__(self, head, nodes):
        self.head = head
        self.nodes = nodes

def read_graph_from_file(path):
    head = 0
    nodes = []
    with open(path, "r") as file:
        head = int(file.readline())
        current_index = -1
        current_node = None
        for line in file.readlines():
            split1, split2 = map(int, line.strip().split(','))
            if current_index == split1:
                current_node.add_child(split2)
            else:
                if current_index != -1:
                    nodes.append(current_node)
                current_index = split1
                current_node = Node(split1)
                current_node.add_child(split2)
        nodes.append(current_node)
    return Graph(head, nodes)

if __name__ == "__main__":
    if len(sys.argv) != 2:
        print("Usage: python script.py input_file.txt")
        sys.exit(1)

    input_file = sys.argv[1]
    graph = read_graph_from_file(input_file)
    print("Graph Head:", graph.head)
    print("Graph Nodes:")
    for node in graph.nodes:
        print(f"Node {node.index}: Children {node.children}")
