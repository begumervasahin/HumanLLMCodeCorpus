import collections

class Solution:
    def intersect(self, nums1, nums2):
        # Initialize an empty list to store the intersection
        intersection = []

        # Create a dictionary to store the count of elements in nums2
        nums2_count = collections.Counter(nums2)

        # Iterate through each element in nums1
        for num in nums1:
            # Check if the element exists in nums2_count and its count is greater than 0
            if nums2_count[num] > 0:
                # Add the element to the intersection list
                intersection.append(num)
                # Decrement the count of the element in nums2_count
                nums2_count[num] -= 1

        return intersection
