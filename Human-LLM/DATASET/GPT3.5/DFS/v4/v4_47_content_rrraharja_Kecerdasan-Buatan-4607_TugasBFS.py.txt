def bfs(graph, start, finish):
    """
    Perform Breadth-First Search (BFS) on a graph to find a path from start to finish.

    Args:
    - graph: The graph represented as a dictionary of nodes and their neighbors.
    - start: The starting node for the search.
    - finish: The target node to reach.

    Returns:
    - The path from start to finish if found, or None if no path exists.
    """
    queue = [[start]]  # Initialize the queue with the start node
    visited = set()  # Keep track of visited nodes

    while queue:
        path = queue.pop(0)  # Dequeue the first path from the queue
        state = path[-1]  # Get the last node in the current path

        if state == finish:
            return path  # If the finish node is reached, return the path

        if state not in visited:
            for neighbor in graph.get(state, []):
                new_path = list(path)  # Create a new path by extending the current path
                new_path.append(neighbor)  # Append the neighbor to the new path
                queue.append(new_path)  # Enqueue the new path

            visited.add(state)  # Mark the current node as visited

    print("NOT FOUND 404")  # If the finish node is not reached, print a message
    return None
