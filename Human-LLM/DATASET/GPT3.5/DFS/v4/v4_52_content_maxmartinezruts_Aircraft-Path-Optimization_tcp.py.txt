# Importing the socket module
import socket

# Defining the server's IP address, port, buffer size, and message
TCP_IP = '52.43.121.77'
TCP_PORT = 2329
BUFFER_SIZE = 1024
MESSAGE = "Hello, World!"

# Creating a socket object
socket_obj = socket.socket(socket.AF_INET, socket.SOCK_STREAM)

# Connecting to the server
socket_obj.connect((TCP_IP, TCP_PORT))
print('Connected to the server')

# Sending the message
socket_obj.send(MESSAGE.encode())

# Receiving data from the server
received_data = socket_obj.recv(BUFFER_SIZE)

# Closing the connection
socket_obj.close()

# Printing the received data
print("Received data:", received_data.decode())
