from collections import defaultdict

class Graph:
    def __init__(self):
        """Initializes an empty graph."""
        self.graph = defaultdict(list)

    def add_edge(self, start_vertex, end_vertex):
        """Adds an edge between vertices start_vertex and end_vertex."""
        self.graph[start_vertex].append(end_vertex)

    def dfs_visit(self, vertex, visited):
        """Recursively visits vertices in depth-first order."""
        visited[vertex] = True
        print(vertex, end=" ")  # Print the visited vertex
        for neighbor in self.graph[vertex]:
            if not visited[neighbor]:
                self.dfs_visit(neighbor, visited)

    def dfs(self):
        """Performs Depth First Traversal."""
        visited = defaultdict(bool)
        for vertex in self.graph:
            if not visited[vertex]:
                self.dfs_visit(vertex, visited)

if __name__ == "__main__":
    # Create a new graph
    graph = Graph()

    # Add edges to the graph
    edges = [(0, 1), (1, 2), (2, 4), (2, 5), (4, 6), (5, 10), (11, 7)]
    for edge in edges:
        graph.add_edge(*edge)

    # Perform DFS starting from vertex 0
    print(" " * 10, "Depth First Traversal starting from vertex 0: ")
    graph.dfs()
