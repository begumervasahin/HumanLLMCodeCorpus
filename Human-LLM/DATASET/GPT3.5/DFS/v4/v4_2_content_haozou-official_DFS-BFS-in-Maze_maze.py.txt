from random import seed, randint
from myarray2d import Array2D

class Maze:
    EMPTY = 0
    OBSTACLE = -1
    OCCUPIED = 1

    def __init__(self, height, width):
        seed()
        self.maze = Array2D(height, width)
        self.start = (0, 0)
        self.goal = (height - 1, width - 1)
        self.initialize_maze()

    def initialize_maze(self):
        self.maze.clear(Maze.EMPTY)
        num_obstacles = int(self.maze.numRows() * self.maze.numCols() * 0.1)
        for _ in range(num_obstacles):
            self.place_random_obstacle()

    def place_random_obstacle(self):
        while True:
            row = randint(0, self.maze.numRows() - 1)
            col = randint(0, self.maze.numCols() - 1)
            if (row, col) not in [self.start, self.goal] and self.maze[row, col] != Maze.OBSTACLE:
                self.maze[row, col] = Maze.OBSTACLE
                break

    def getAllMoves(self, row, col):
        moves = []
        offsets = [(0, -1), (-1, 0), (1, 0), (0, 1)]
        for dx, dy in offsets:
            new_x, new_y = col + dx, row + dy
            if 0 <= new_x < self.maze.numCols() and 0 <= new_y < self.maze.numRows() \
                    and self.maze[new_y, new_x] != Maze.OBSTACLE:
                moves.append((new_y, new_x))
        return moves

    def numRows(self):
        return self.maze.numRows()

    def numCols(self):
        return self.maze.numCols()

    def printMaze(self):
        rows = self.numRows()
        cols = self.numCols()
        for row in range(rows):
            for col in range(cols):
                if self.maze[row, col] == Maze.EMPTY:
                    print('_', end=' ')
                elif self.maze[row, col] == Maze.OBSTACLE:
                    print('|', end=' ')
                else:
                    print('O', end=' ')
            print()

def main():
    maze = Maze(20, 20)
    maze.printMaze()
    print()
    print(maze.getAllMoves(3, 3))

if __name__ == '__main__':
    main()
