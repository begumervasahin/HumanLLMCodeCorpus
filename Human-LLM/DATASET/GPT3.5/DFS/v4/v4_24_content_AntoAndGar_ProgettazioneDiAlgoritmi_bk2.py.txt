def generate_binary_strings(n, k, index=0, ones_count=0, current=[]):
    """
    Generate binary strings of length n with exactly k '1's.

    Args:
    - n: Length of the binary strings.
    - k: Number of '1's in the binary strings.
    - index: Current index during recursion (default: 0).
    - ones_count: Count of '1's added so far (default: 0).
    - current: Current partial binary string being constructed (default: []).

    This function uses backtracking to generate binary strings with exactly k '1's.
    """
    if index == n:
        # Base case: if the string length reaches n, print the current string
        print(''.join(current))
    else:
        # Try adding '0' to the current string
        current.append('0')
        generate_binary_strings(n, k, index + 1, ones_count, current)
        current.pop()  # Backtrack by removing the last character

        # If we can add '1' (total '1's added < k), try adding '1' to the current string
        if ones_count < k:
            current.append('1')
            generate_binary_strings(n, k, index + 1, ones_count + 1, current)
            current.pop()  # Backtrack by removing the last character

# Generate binary strings of length 4 with exactly 1 '1'
generate_binary_strings(4, 1)
