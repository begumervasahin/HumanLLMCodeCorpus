from collections import deque

# Graph data
vertices = [0, 1, 2, 3, 4]
adjacency_list = [[1, 2], [2, 3], [3, 4], [4, 0], [0, 1]]

# Initialize visited nodes list
visited = [None] * len(vertices)

# Input source vertex from user
source_vertex = int(input("Enter the source vertex: "))
visited[source_vertex] = 0

# Initialize queue for BFS traversal
queue = deque()
queue.append(source_vertex)

# Perform BFS traversal
while queue:
    current_vertex = queue.popleft()
    print("Visited vertex:", current_vertex)
    for neighbor in adjacency_list[current_vertex]:
        if visited[neighbor] is None:
            visited[neighbor] = 0
            queue.append(neighbor)

# Check for disconnected graph
if None in visited:
    print("Graph is disconnected")
else:
    print("All nodes of the graph have been traversed")

# Reset visited and parent arrays for cycle detection

# Function to detect cycle using BFS
def detect_cycle_BFS(adj_list, source, parent):

    visited[source] = 0
    queue = deque()
    queue.append(source)
    
    while queue:
        vertex = queue.popleft()
        for neighbor in adj_list[vertex]:
            if parent[neighbor] is not None:
                print("Cycle exists")
                return
            else:
                visited[neighbor] = 0
                parent[neighbor] = vertex
