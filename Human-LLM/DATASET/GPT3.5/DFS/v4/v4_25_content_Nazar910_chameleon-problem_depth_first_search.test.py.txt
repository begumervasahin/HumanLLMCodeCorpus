import unittest
from depth_first_search import dfs
from state import State

class TestDfs(unittest.TestCase):
    def test_dfs_algorithm(self):
        # Display test header
        print("=======================")
        print("DFS Algorithm Test")
        print("=======================")
        
        # Execute DFS algorithm with an initial state
        result = dfs(State(red=13, blue=16, green=17))
        
        # Extract final state from result
        final_state = result['final_state']
        
        # Verify the final state's color counts
        self.assertEqual(final_state.red_count, 0)
        self.assertEqual(final_state.green_count, 46)
        self.assertEqual(final_state.blue_count, 0)
        
        # Extract the solution path and verify its length
        path = result['path']
        self.assertEqual(len(path), 89)
        print(f"Path length: {len(path)}")
        
        # Print the solution path
        print("Solution Path:")
        for step, state in enumerate(path, start=1):
            print(f"Step {step}: {state}")
        
        # Extract the full path and verify its length
        full_path = result['full_path']
        self.assertEqual(len(full_path), 98)
        print(f"Full path length: {len(full_path)}")

if __name__ == "__main__":
    unittest.main()
