from typing import List
from cell import Cell

class UniqueCellArea:
    allowed_numbers = list(range(1, 10))

    def __init__(self, number: int, cells: List[Cell]):
        self.number = number
        self.cells = cells

    def get_missing(self) -> set:
        """Returns a set of missing numbers in the cells."""
        return set(num for num in self.allowed_numbers if num not in [cell.value for cell in self.cells])

    def get_present(self) -> set:
        """Returns a set of present numbers in the cells."""
        return set(self.allowed_numbers) - self.get_missing()

    def check(self) -> bool:
        """Checks if the cells contain all numbers from 1 to 9 without duplicates."""
        return set([cell.value for cell in self.cells]) == set(self.allowed_numbers) and not self.has_duplicates()

    def has_duplicates(self) -> bool:
        """Checks if there are duplicate numbers in the cells."""
        return any([cell.value for cell in self.cells if not cell.is_empty].count(x) > 1 for x in self.allowed_numbers)

    def __repr__(self) -> str:
        """Returns a string representation of the unique cell area."""
        return f"{self.__class__.__name__} {self.number}: {self.cells}"

class Row(UniqueCellArea):
    """Represents a row of cells in a Sudoku puzzle."""
    pass

class Col(UniqueCellArea):
    """Represents a column of cells in a Sudoku puzzle."""
    pass

class Grid(UniqueCellArea):
    """Represents a 3x3 grid of cells in a Sudoku puzzle."""
    pass
