from foundation import Node, State, Block
from searchMethods import SearchMethods

# Create the initial state
initial_blocks = [Block('A', 0, 0), Block('B', 1, 0), Block('C', 2, 0)]
initial_agent = Block('P', 3, 0)
initial_state = State(initial_blocks, initial_agent, None)
initial_node = Node(initial_state, None, None)

# Create the goal state
goal_blocks = [Block('A', 1, 2), Block('B', 1, 1), Block('C', 1, 0)]
goal_agent = Block('P', 3, 0)
goal_state = State(goal_blocks, goal_agent, None)

# Initialize SearchMethods with the initial node and goal state
search_methods = SearchMethods(initial_node, goal_state)

# Prompt the user to select a search method
print('Select Search Method:')
print('1. Depth First Search (DFS)')
print('2. Breadth First Search (BFS)')
print('3. Iterative Deepening Search')
print('4. A* Search (AStar)')

# Get user input for the search method selection
selection = int(input())

# Perform the selected search method
if selection == 1:
    print('Performing Depth First Search (DFS) ...')
    search_methods.DFS()
elif selection == 2:
    print('Performing Breadth First Search (BFS) ...')
    search_methods.BFS()
elif selection == 3:
    print('Performing Iterative Deepening Search ...')
    search_methods.iterativeDeepening()
elif selection == 4:
    print('Performing A* Search (AStar) ...')
    search_methods.aStar()
else:
    print('Invalid input. Please select a valid search method.')
