import unittest
from harmony import Harmony
from solve import get_harmony_text, get_path

class TestHarmonyCases(unittest.TestCase):
    """Unit tests for the Harmony class."""

    def test_small_puzzle_1(self):
        """Tests the correct path for a 2 x 2 puzzle."""
        expected_length = 1
        path = get_path("cases/1.in")
        self.assertEqual(expected_length, len(path))

    def test_medium_puzzle_1(self):
        """Tests the correct path for a 3 x 3 puzzle."""
        expected_length = 2
        path = get_path("cases/2.in")
        self.assertEqual(expected_length, len(path))

    def test_medium_puzzle_2_harmony_3_puzzle_1_pack_1(self):
        """Tests the correct path for a 3 x 3 puzzle (Harmony 3, puzzle 1 of pack 1)."""
        expected_length = 5
        path = get_path("cases/3.in")
        self.assertEqual(expected_length, len(path))

    def test_medium_puzzle_3_holiday_pack_2015_puzzle_1(self):
        """Tests the correct path for a 3 x 3 puzzle (Harmony 3, puzzle 1 of 2015 holiday pack)."""
        expected_length = 5
        path = get_path("cases/4.in")
        self.assertEqual(expected_length, len(path))

    def test_medium_puzzle_4_invalid_path(self):
        """Tests for an invalid path for a 3 x 3 puzzle."""
        path = get_path("cases/5.in")
        self.assertIsNone(path)

    def test_large_puzzle_1_trivial_path(self):
        """Tests for a trivial path for a 4 x 4 puzzle."""
        expected_length = 4
        path = get_path("cases/6.in")
        self.assertEqual(expected_length, len(path))

    def test_large_puzzle_2_trivial_path(self):
        """Tests for a trivial path for a 4 x 4 puzzle."""
        expected_length = 12
        path = get_path("cases/7.in")
        self.assertEqual(expected_length, len(path))

    def test_large_puzzle_3_harmony_3_puzzle_8_pack_1(self):
        """Tests for a trivial path for Harmony 3 puzzle 8 of pack 1."""
        expected_length = 11
        path = get_path("cases/8.in")
        self.assertEqual(expected_length, len(path))

if __name__ == '__main__':
    unittest.main()
