def tower_of_hanoi_solver(num_disks, start_peg, end_peg, auxiliary_peg):
    move_count = 1  # Initialize the move counter

    def move_disk(disk, from_peg, to_peg):
        nonlocal move_count
        print(f"Move {move_count}: Move disk {disk} from peg {from_peg} to peg {to_peg}")
        move_count += 1

    def hanoi_recursive(num, source, target, auxiliary):
        if num > 0:
            hanoi_recursive(num - 1, source, auxiliary, target)
            move_disk(num, source, target)
            hanoi_recursive(num - 1, auxiliary, target, source)

    # Start solving the Tower of Hanoi problem
    hanoi_recursive(num_disks, start_peg, end_peg, auxiliary_peg)

# Example usage: Solve Tower of Hanoi for 3 disks
tower_of_hanoi_solver(3, 1, 2, 3)
