def create_directory(dir_name, remove_files=False):
    """
    Create a directory if it doesn't exist and optionally remove files inside it.
    """
    if not os.path.isdir(dir_name):
        try:
            os.mkdir(dir_name)
        except OSError:
            print(f"Cannot create folder {dir_name}!")
    else:
        if remove_files:
            files = glob(os.path.join(dir_name, '*.*'))
            if files:
                try:
                    for file in files:
                        os.remove(file)
                except OSError:
                    print("Cannot remove files")
            else:
                print(f"Folder {dir_name} is empty!")

def get_data_from_website(url, filename):
    """
    Download data from a URL and save it to a file.
    """
    try:
        r = requests.get(url, allow_redirects=True)
        open(filename, 'wb').write(r.content)
    except ValueError:
        print(f"Oops! Cannot get data from {url}")

def convert_raw_data_to_txt_format():
    """
    Convert raw data from the website to text format.
    """
    temp = np.loadtxt(fname='./DataFromWebsite/luongmua.txt', delimiter=';', skiprows=1, dtype='str')
    temp_tram = 0
    filename = ""
    file_handle = 0
    list_of_data = []
    create_directory('./TempData', remove_files=True)

    for i in temp:
        if temp_tram != int(i[0]):
            temp_tram = int(i[0])
            filename = tempDataFolder.get_tram_folder(temp_tram) + i[0]
            if file_handle != 0:
                if list_of_data:
                    for line_index in list_of_data:
                        file_handle.write(line_index)
                    file_handle.close()
            list_of_data = []
            file_handle = open(filename, "w")
            line = f"{i[2]},{i[1]}\n"
            list_of_data.insert(0, line)
        else:
            line = f"{i[2]},{i[1]}\n"
            list_of_data.insert(0, line)

    if file_handle != 0:
        if list_of_data:
            for line_index in list_of_data:
                file_handle.write(line_index)
        file_handle.close()

def convert_txt_data_to_dfs0_format():
    """
    Convert text data to DFS0 format using the multDFS0Generating function.
    """
