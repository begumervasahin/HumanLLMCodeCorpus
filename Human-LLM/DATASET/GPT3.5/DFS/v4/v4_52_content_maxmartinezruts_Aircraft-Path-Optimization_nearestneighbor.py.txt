# Importing necessary libraries
from sklearn.neighbors import NearestNeighbors
import numpy as np

# Creating a dataset
data_points = np.array([[-1, -1], [-2, -1], [-3, -2], [1, 1], [2, 1], [3, 2]])

# Initializing the NearestNeighbors model
nearest_neighbors_model = NearestNeighbors(n_neighbors=2, algorithm='ball_tree').fit(data_points)

# Printing the model
print("Nearest Neighbors Model:")
print(nearest_neighbors_model)

# Finding nearest neighbors for each data point
distances, indices = nearest_neighbors_model.kneighbors(data_points)

# Printing distances and indices of nearest neighbors
print("Distances to Nearest Neighbors:")
print(distances)
print("Indices of Nearest Neighbors:")
print(indices)
