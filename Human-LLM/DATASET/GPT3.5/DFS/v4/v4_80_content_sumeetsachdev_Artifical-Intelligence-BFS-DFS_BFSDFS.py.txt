def dfs(graph, start):
    """
    Perform Depth-First Search (DFS) traversal on the given graph starting from the specified node.

    Args:
        graph (dict): The graph represented as a dictionary where keys are nodes and values are lists of adjacent nodes.
        start (str): The starting node for DFS traversal.

    Returns:
        list: The list of nodes visited during DFS traversal.
    """
    if start not in graph:
        print("Start state not in graph")
        return False
    
    visited = []
    stack = [start]
    
    while stack:
        current = stack.pop()
        if current not in visited:
            visited.append(current)
            for neighbor in graph[current]:
                if neighbor not in visited:
                    stack.append(neighbor)
    
    return visited

def bfs(graph, start):
    """
    Perform Breadth-First Search (BFS) traversal on the given graph starting from the specified node.

    Args:
        graph (dict): The graph represented as a dictionary where keys are nodes and values are lists of adjacent nodes.
        start (str): The starting node for BFS traversal.

    Returns:
        list: The list of nodes visited during BFS traversal.
    """
    if start not in graph:
        return None
    
    visited = []
    queue = [start]
    
    while queue:
        current = queue.pop(0)
        if current not in visited:
            visited.append(current)
            for neighbor in graph[current]:
                if neighbor not in visited:
                    queue.append(neighbor)
    
    return visited

# Define the graph
graph = {
    'A': ['B', 'D', 'G'],
    'B': ['A', 'E', 'F'],
    'C': ['F', 'H'],
    'D': ['A', 'F'],
    'E': ['B', 'G'],
    'F': ['B', 'C', 'D'],
    'G': ['A', 'E'],
    'H': ['C']
}

# Perform DFS and BFS
dfs_result = dfs(graph, 'A')
bfs_result = bfs(graph, 'A')

print("DFS result:", dfs_result)
print("BFS result:", bfs_result)
