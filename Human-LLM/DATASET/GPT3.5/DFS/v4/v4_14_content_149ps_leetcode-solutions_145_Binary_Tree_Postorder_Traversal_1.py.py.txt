class Solution:
    def postorderTraversal(self, root):
        """
        Perform postorder traversal of a binary tree.

        :param root: TreeNode, the root node of the binary tree.
        :return: List[int], containing the postorder traversal of the tree.
        """
        result = []  # List to store the traversal result
        stack = []   # Stack to simulate iterative traversal
        node = root  # Start traversal from the root node
        visited = set()  # Set to track visited nodes

        while node or stack:
            if node:
                # Traverse left subtree until reaching the leftmost leaf
                stack.append(node)
                node = node.left
            else:
                # Process the right subtree and backtrack if needed
                node = stack.pop()
                if node.right and node.right not in visited:
                    # If right subtree exists and hasn't been visited yet
                    stack.append(node)  # Push the current node back to the stack
                    node = node.right  # Move to the right subtree
                else:
                    # If no right subtree or already visited, add to result
                    visited.add(node)  # Mark the current node as visited
                    result.append(node.val)  # Add node value to result
                    node = None  # Reset node to None for backtracking
        
        return result
