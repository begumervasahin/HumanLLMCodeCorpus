import requests
from bs4 import BeautifulSoup
import csv
import datetime
import sys

def main():
    position = select_position()
    url = build_url(position)
    filename = generate_filename(position)

    with open(filename, 'w', newline='') as csvfile:
        writer = csv.writer(csvfile)
        writer.writerow(['Name', 'Team', 'Pos', 'Fantasy Team'])

        response = requests.get(url)
        soup = BeautifulSoup(response.text, 'html.parser')
        stats = extract_stats(soup)

        page_num = 0
        max_pages = 10  # Limiting the number of pages for demonstration purposes
        while True:
            page_count = str(page_num * 25)
            print(f"Loading page {page_num+1}")
            content = requests.get(url + page_count)
            soup = BeautifulSoup(content.text, 'html.parser')
            players = soup.find_all('div', class_='ysf-player-name Nowrap Grid-u Relative Lh-xs Ta-start')
            data_list = soup.find_all('td', class_='Ta-end')
            fantasy_teams = soup.find_all('div', style='text-overflow: ellipsis; overflow: hidden;')

            for player, data, fantasy_team in zip(players, data_list, fantasy_teams):
                player_stats = extract_player_stats(player, data, fantasy_team, stats)
                writer.writerow(player_stats)

            page_num += 1
            if page_num >= max_pages:
                break

def select_position():
    print("\nPosition:")
    print("-------------")
    print("1. Quarterbacks\n2. Runningbacks\n3. Receivers\n4. Tightends\n5. Defenses")
    print("-------------")
    try:
        position = int(input("Enter 1 - 5: "))
        if position not in range(1, 6):
            raise ValueError("Invalid position number.")
    except ValueError as e:
        print(f"Error: {e}. Exiting...")
        sys.exit()
    print('Gathering Data...')
    return position

def build_url(position):
    league_id = 12345  # Your league ID
    base_url = f'http://baseball.fantasysports.yahoo.com/b1/{league_id}/players?status='
    return f"{base_url}&myteam=0&sort=OR&sdir=1&count="

def generate_filename(position):
    positions = {
        1: 'Quarterbacks', 2: 'Runningbacks', 3: 'Receivers', 4: 'Tightends', 5: 'Defenses'
    }
    position_name = positions.get(position, 'Unknown')
    return f'DFS_data_{position_name}_{datetime.date.today()}.csv'

def extract_stats(soup):
    stats_list = soup.find_all('th', class_='Ta-end')
    stats = ['Name', 'Team', 'Pos', 'Fantasy Team']
    for s in stats_list:
        text = str(s.get_text())
        text = text[3:len(text) - 2]
        stats.append(text.split(",")[0])
        try:
            stats.remove("Rankings")
        except:
            continue
    return stats

def extract_player_stats(player, data, fantasy_team, stats):
    player_stats = []
    name = get_name(player.get_text())
    team, pos = get_team_and_position(data.get_text())
    fan_team = fix_text(fantasy_team.get_text())
    player_stats.extend([name, team, pos, fan_team])

    for i in range(len(stats) - 4):
        tmp = str(data[i].get_text())
        tmp = tmp[3:len(tmp) - 2] if tmp.find("/") != -1 else tmp
        player_stats.append(f"'{tmp}'" if tmp.find("/") != -1 else tmp)

    return player_stats

def get_name(data):
    player_data = data.split("'") if data[2] == '"' else data.split('"')
    return fix_text(player_data[1])

def get_team_and_position(data):
    player_data = data.split("'")
    teampos = player_data[4] if data[2] == '"' else player_data[5]
    team = teampos[:teampos.find("-") - 1]
    pos = teampos[teampos.find("-") + 2:len(teampos)]
    return team, pos

def fix_text(string):
    replacements = {
        '\\xe1': 'a', '\\xe0': 'a', '\\xc1': 'A', '\\xe9': 'e', '\\xc9': 'E',
        '\\xed': 'i', '\\xcd': 'I', '\\xf3': 'o', '\\xd3': 'O', '\\xfa': 'u',
        '\\xda': 'U', '\\xf1': 'n'
    }
    for key, value in replacements.items():
        string = string.replace(key, value)
    return string

if __name__ == "__main__":
    main()
