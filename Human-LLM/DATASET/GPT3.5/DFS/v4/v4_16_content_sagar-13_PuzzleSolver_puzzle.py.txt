class Puzzle:
    """
    Represents a snapshot of a full-information puzzle, which can be in various states.
    """

    def fail_fast(self):
        """
        Checks whether this puzzle can never be extended to a solution.
        
        Returns:
            bool: True if the puzzle cannot be extended to a solution, False otherwise.
        """
        return False

    def is_solved(self):
        """
        Checks whether this puzzle is solved.
        
        Raises:
            NotImplementedError: This method must be implemented in a subclass.
        """
        raise NotImplementedError("Subclasses must implement is_solved method.")

    def extensions(self):
        """
        Returns a list of legal extensions of this puzzle.
        
        Raises:
            NotImplementedError: This method must be implemented in a subclass.
        """
        raise NotImplementedError("Subclasses must implement extensions method.")
