import pandas as pd

# Define the column headers for the DataFrame
column_headers = ['Player', 'FP', 'Cost', 'Value', 'C/A', 'PYDS', 'PTDS', 'INTS', 'RUATT', 'RUYDS', 'RUTDS', 'REREC', 'REYDS', 'RETDS', 'RETGTS']

# Read the Excel file into a DataFrame
file_path = 'C:\\Users\\Alan\\Documents\\Fantasy\\Football\\DFS\\10292017\\FD_ALL_POS_PROJ.xlsx'
dk_df = pd.read_excel(file_path, sheet_name='Sheet1', names=column_headers)

# Initialize lists to store parsed player information
players = []
positions = []
matchups = []
times = []

# Parse player information and populate lists
for i, row in dk_df.iterrows():
    if 'GTD' in row['Player']:
        row['Player'] = row['Player'].replace('GTD', '')
    player_info = row['Player'].split()
    players.append(' '.join(player_info[:2]))
    positions.append(player_info[2])
    matchups.append(f"{player_info[3]} @ {player_info[5][:-3]}")
    times.append(f"{player_info[5][-3:]} {player_info[6]}")

# Remove the 'Player' column from the DataFrame
dk_df.drop(columns=['Player'], inplace=True)

# Add parsed columns to the DataFrame
dk_df['Player'] = players
dk_df['Position'] = positions
dk_df['Matchup'] = matchups
dk_df['Game Time'] = times

# Filter DataFrames by position
dk_df_qb = dk_df[dk_df['Position'] == 'QB']
dk_df_wr = dk_df[dk_df['Position'] == 'WR']
dk_df_rb = dk_df[dk_df['Position'] == 'RB']
dk_df_te = dk_df[dk_df['Position'] == 'TE']
dk_df_def = dk_df[dk_df['Position'] == 'D/ST']

# Write DataFrames to separate sheets in an Excel file
output_file_path = 'C:\\Users\\Alan\\Documents\\Fantasy\\Football\\DFS\\10292017\\FD_ALL_POS_PROJ_PROCESSED.xlsx'
with pd.ExcelWriter(output_file_path) as writer:
    dk_df.to_excel(writer, 'DK Projections', index=False)
    dk_df_qb.to_excel(writer, 'DK QB', index=False)
    dk_df_wr.to_excel(writer, 'DK WR', index=False)
    dk_df_rb.to_excel(writer, 'DK RB', index=False)
    dk_df_te.to_excel(writer, 'DK TE', index=False)
    dk_df_def.to_excel(writer, 'DK DEF', index=False)
