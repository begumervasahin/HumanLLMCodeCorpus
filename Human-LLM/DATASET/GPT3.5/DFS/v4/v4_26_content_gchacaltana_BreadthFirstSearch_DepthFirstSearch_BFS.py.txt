from Node import Node
from UninformedSearchAlgorithm import UninformedSearchAlgorithm

class BFS(UninformedSearchAlgorithm):
    def __init__(self, list_nodes, searched):
        super().__init__(list_nodes, searched)

    def add_queue(self, node):
        if node.name not in [n.name for n in self.queue]:
            self.queue.append(node)

    def search(self):
        try:
            count = 0
            while True:
                self.validate_len_queue()
                count += 1
                print("\nIteration:", count)
                node = self.dequeue()
                print("Reading node:", node.name)
                self.check_searched(node.name)
                self.add_children_to_queue(node)
                print("Queue:", [n.name for n in self.queue])
        except Exception as error:
            print(error)
