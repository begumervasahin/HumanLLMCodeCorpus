from collections import deque

# Define the graph
graph = {
    'A': {'B', 'C', 'D'},
    'B': {'A', 'C', 'E', 'F', 'G'},
    'C': {'A', 'B', 'D'},
    'D': {'A', 'C', 'H', 'I'},
    'E': {'B', 'L'},
    'F': {'B', 'G', 'L'},
    'G': {'B', 'F', 'J', 'H'},
    'H': {'D', 'G', 'J'},
    'I': {'D', 'J'},
    'J': {'I', 'G', 'H', 'K', 'M'},
    'K': {'L', 'J', 'M'},
    'L': {'E', 'F', 'M', 'K'},
    'M': {'L', 'K', 'J'}
}

def bfs_shortest_path(graph, start, end):
    # Use a queue to perform BFS
    queue = deque([(start, [start])])
    while queue:
        node, path = queue.popleft()
        # Iterate over neighbors of the current node
        for neighbor in graph[node].difference(path):
            # If the neighbor is the destination, yield the path
            if neighbor == end:
                yield path + [neighbor]
            # Otherwise, continue exploring
            else:
                queue.append((neighbor, path + [neighbor]))

print("---------Program Result-------")
# Find the shortest path from 'E' to 'G'
all_paths = list(bfs_shortest_path(graph, 'E', 'G'))
path_count = len(all_paths)
# Print the result
if path_count == 0:
    print("No path found.")
else:
    print("Possible paths:", all_paths)
    print("Shortest path:", all_paths[0])
