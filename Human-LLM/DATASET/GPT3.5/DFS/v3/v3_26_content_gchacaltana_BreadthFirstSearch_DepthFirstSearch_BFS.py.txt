class Node:
    def __init__(self, name):
        self.name = name
        self.children = []

    def add_child(self, child_node):
        self.children.append(child_node)

class UninformedSearchAlgorithm:
    def __init__(self, nodes, searched_nodes):
        self.queue = []
        self.visited = []
        self.nodes = nodes
        self.searched_nodes = searched_nodes

    def enqueue(self, node):
        if node.name not in [n.name for n in self.queue]:
            self.queue.append(node)

    def validate_queue(self):
        if len(self.queue) == 0:
            raise Exception("Queue is empty!")

    def dequeue(self):
        return self.queue.pop(0)

    def check_searched(self, node_name):
        if node_name in self.searched_nodes:
            print(f"Found {node_name}!")

    def add_children_to_queue(self, node):
        for child in node.children:
            if child.name not in self.visited:
                self.enqueue(child)
                self.visited.append(child.name)

    def search(self):
        try:
            iteration = 0
            while True:
                self.validate_queue()
                iteration += 1
                print("\nIteration:", iteration)
                current_node = self.dequeue()
                print("Current node:", current_node.name)
                self.check_searched(current_node.name)
                self.add_children_to_queue(current_node)
                print("Queue:", [n.name for n in self.queue])
        except Exception as error:
            print(error)

class BFS(UninformedSearchAlgorithm):
    def __init__(self, nodes, searched_nodes):
        super().__init__(nodes, searched_nodes)

# Create nodes
node_a = Node("A")
node_b = Node("B")
node_c = Node("C")
node_d = Node("D")
node_e = Node("E")
node_f = Node("F")
node_g = Node("G")

# Connect nodes
node_a.add_child(node_b)
node_a.add_child(node_c)
node_b.add_child(node_d)
node_b.add_child(node_e)
node_c.add_child(node_f)
node_c.add_child(node_g)

# Create BFS algorithm and perform search
bfs = BFS([node_a, node_b, node_c, node_d, node_e, node_f, node_g], ["F"])
bfs.enqueue(node_a)  # Start from node A
bfs.search()
