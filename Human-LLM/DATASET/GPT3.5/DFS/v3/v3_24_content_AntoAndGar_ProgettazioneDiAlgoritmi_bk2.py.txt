def generate_binary_strings(length, ones_count, index=0, ones_so_far=0, current=[]):
    """
    Generate binary strings of a specified length with a given number of '1's.
    
    Args:
    - length: The length of the binary strings to generate.
    - ones_count: The number of '1's to include in the binary strings.
    - index: Internal index for recursion (default: 0).
    - ones_so_far: Count of '1's added so far (default: 0).
    - current: Current partial binary string being constructed (default: []).
    """
    if index == length:
        print(''.join(current))
    else:
        # Try adding '0'
        current.append('0')
        generate_binary_strings(length, ones_count, index + 1, ones_so_far, current)
        current.pop()  # Backtrack

        # Try adding '1' if '1's added so far are less than the specified count
        if ones_so_far < ones_count:
            current.append('1')
            generate_binary_strings(length, ones_count, index + 1, ones_so_far + 1, current)
            current.pop()  # Backtrack

# Example usage: generate binary strings of length 4 with exactly 1 '1'
generate_binary_strings(4, 1)
