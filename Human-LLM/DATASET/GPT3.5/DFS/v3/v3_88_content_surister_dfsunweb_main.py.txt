from selenium import webdriver
from selenium.webdriver.support.ui import Select
from time import sleep

# Set the path to the geckodriver executable
geckodriver_path = 'D:/geckodriver/geckodriver.exe'

# Create a new instance of Firefox WebDriver
driver = webdriver.Firefox(executable_path=geckodriver_path)

# Open the website
driver.get("http://checkyourflight.info")

# Click on the departure date input field to activate it
departure_date_input = driver.find_element_by_id('fecha_salida')
departure_date_input.click()

# Enter the departure date
departure_date_input.send_keys('2019-08-14')

# Close the date picker by clicking again
departure_date_input.click()

# Wait for 1 second to ensure the date picker is closed
sleep(1)

# Select flights
flight_select = Select(driver.find_element_by_id('flight'))
flights = flight_select.options

# Iterate over each flight option
for flight in flights:
    # Click on the flight option
    flight.click()
    # Print the value of the flight option
    print(flight.get_attribute('value'))
    # Wait for 1 second
    sleep(1)
    # Select hotels
    hotel_select = Select(driver.find_element_by_id('hoteles'))
    hotels = hotel_select.options
    # Iterate over each hotel option
    for hotel in hotels:
        # Click on the hotel option
        hotel.click()
        # Print the value of the hotel option
        print(hotel.get_attribute('value'))

# Wait for 100 seconds before closing the browser
sleep(100)

# Close the browser
driver.close()
