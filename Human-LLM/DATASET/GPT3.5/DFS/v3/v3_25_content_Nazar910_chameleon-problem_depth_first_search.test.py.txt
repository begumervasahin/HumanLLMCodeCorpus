import unittest
from depth_first_search import dfs
from state import State

class TestDfs(unittest.TestCase):
    def test_dfs_algorithm(self):
        # Display test header
        self.print_test_header("DFS Algorithm Test")
        
        # Execute DFS algorithm with an initial state
        result = dfs(State(red=13, blue=16, green=17))
        
        # Extract and verify the final state
        final_state = result['final_state']
        self.verify_final_state(final_state)
        
        # Extract and verify the solution path
        path = result['path']
        self.verify_solution_path(path)
        
        # Print the solution path
        self.print_solution_path(path)
        
        # Extract and verify the full path
        full_path = result['full_path']
        self.verify_full_path(full_path)

    def print_test_header(self, header_text):
        print("=======================")
        print(header_text)
        print("=======================")

    def verify_final_state(self, state):
        self.assertEqual(state.red_count, 0)
        self.assertEqual(state.green_count, 46)
        self.assertEqual(state.blue_count, 0)

    def verify_solution_path(self, path):
        self.assertEqual(len(path), 89)

    def print_solution_path(self, path):
        print(f"Length of Solution Path: {len(path)}")
        print("Solution Path:")
        for step, state in enumerate(path, start=1):
            print(f"Step {step}: {state}")

    def verify_full_path(self, full_path):
        self.assertEqual(len(full_path), 98)
        print(f"Length of Full Path: {len(full_path)}")

if __name__ == "__main__":
    unittest.main()
