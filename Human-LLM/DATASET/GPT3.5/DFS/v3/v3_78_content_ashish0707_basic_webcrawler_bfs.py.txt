class Searcher:
    """A base class for search algorithms."""

class BFS(Searcher):
    """A class representing Breadth-First Search algorithm."""
    name = '2.A_bfs.txt'  # Name of the BFS algorithm

    def merge_lists(self, list1, list2):
        """
        Merge two lists.
        
        Args:
            list1 (list): The first list.
            list2 (list): The second list.
        
        Returns:
            list: The merged list.
        """
        return list1 + list2
