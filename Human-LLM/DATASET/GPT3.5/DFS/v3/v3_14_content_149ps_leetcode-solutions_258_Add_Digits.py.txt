class Solution:
    def addDigits(self, num):
        # Continuously sum the digits until the result becomes a single digit
        while num >= 10:
            # Extract the last digit and add it to the remaining digits
            num = self.sum_digits(num)
        return num

    def sum_digits(self, num):
        # Calculate the sum of the digits of the given number
        total = 0
        while num:
            total += num % 10
            num //= 10
        return total

# Example usage:
if __name__ == "__main__":
    # Create an instance of the Solution class
    solution = Solution()

    # Define the input number
    num = 12345

    # Call the addDigits method to obtain the result
    result = solution.addDigits(num)
    print("Result:", result)
