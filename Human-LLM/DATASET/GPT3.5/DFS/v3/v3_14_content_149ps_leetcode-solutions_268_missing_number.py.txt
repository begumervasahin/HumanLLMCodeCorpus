def find_missing_number(nums):
    # Find the lower and upper bounds of the range
    lower_bound = 0
    upper_bound = max(nums)
    range_length = upper_bound - lower_bound + 1

    print("Lower Bound:", lower_bound)
    print("Upper Bound:", upper_bound)
    print("Range Length:", range_length)

    # Iterate through the range and check for missing numbers
    current_number = lower_bound
    while range_length > 0:
        if current_number not in nums:
            return current_number
        current_number += 1
        range_length -= 1

    # If the list is complete or empty, return the next number after the upper bound
    return upper_bound + 1

# Example usage
numbers = [0]  # List containing numbers
missing_number = find_missing_number(numbers)
print("Missing Number:", missing_number)
