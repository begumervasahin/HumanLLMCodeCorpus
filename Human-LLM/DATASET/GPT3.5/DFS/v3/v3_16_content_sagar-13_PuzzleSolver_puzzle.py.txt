from abc import ABC, abstractmethod

class Puzzle(ABC):
    """Represents a snapshot of a full-information puzzle."""

    @abstractmethod
    def fail_fast(self) -> bool:
        """Checks if the puzzle can never be extended to a solution."""
        return False

    @abstractmethod
    def is_solved(self) -> bool:
        """Checks if the puzzle is solved."""
        pass

    @abstractmethod
    def extensions(self) -> list:
        """Returns a list of legal extensions of the puzzle."""
        pass

# Example subclass of Puzzle
class MyPuzzle(Puzzle):
    """An example subclass of Puzzle."""
    
    def __init__(self, state: str):
        self.state = state

    def is_solved(self) -> bool:
        """Checks if the puzzle is solved."""
        return self.state == "Solved"

    def extensions(self) -> list:
        """Returns legal extensions of the puzzle."""
        return [MyPuzzle("Extension 1"), MyPuzzle("Extension 2")]

# Example usage
if __name__ == "__main__":
    puzzle = MyPuzzle("Unsolved")
    print("Is solved:", puzzle.is_solved())
    print("Extensions:", puzzle.extensions())
    print("Fail fast:", puzzle.fail_fast())
