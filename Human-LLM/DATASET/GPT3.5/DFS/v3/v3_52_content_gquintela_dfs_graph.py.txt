class Graph:
    """
    A class representing a graph.

    Attributes:
    - adjacency_list (dict): A dictionary storing vertices and their corresponding neighbours.
    - size (int): The number of vertices in the graph.
    """

    def __init__(self):
        """Initialize an empty graph."""
        self.adjacency_list = {}
        self.size = 0

    def get_vertices(self):
        """Return a set of vertices in the graph."""
        return set(self.adjacency_list.keys())

    def get_neighbours(self, vertex):
        """Return a list of neighbours for the given vertex."""
        return self.adjacency_list.get(vertex, [])

    def print_vertices(self):
        """Print all vertices in the graph."""
        print("Vertices:")
        for vertex in self.adjacency_list:
            print(vertex)
        print()

    def print_neighbours(self, vertex):
        """Print neighbours of the given vertex."""
        print(f"Neighbours of {vertex}:")
        neighbours = self.adjacency_list.get(vertex, [])
        for neighbour, weight in neighbours:
            print(f"{neighbour}, weight of edge: {weight}")

    def add_vertex(self, vertex):
        """Add a vertex to the graph."""
        if vertex not in self.adjacency_list:
            self.adjacency_list[vertex] = []

    def add_undirected_edge(self, u, v, weight=1):
        """
        Add an undirected edge between vertices u and v with the specified weight.

        Args:
        - u: The first vertex.
        - v: The second vertex.
        - weight (optional): The weight of the edge (default is 1).
        """
        self.add_vertex(u)
        self.add_vertex(v)
        self.adjacency_list[u].append((v, weight))
        self.adjacency_list[v].append((u, weight))

    def add_directed_edge(self, u, v, weight=1):
        """
        Add a directed edge from vertex u to vertex v with the specified weight.

        Args:
        - u: The source vertex.
        - v: The destination vertex.
        - weight (optional): The weight of the edge (default is 1).
        """
        self.add_vertex(u)
        self.add_vertex(v)
        self.adjacency_list[u].append((v, weight))
