import time
import numpy as np
from sklearn.svm import SVC
from sklearn.model_selection import StratifiedKFold
from input_data import read_data_sets

def train_and_test_model(train_data, train_label, test_data, test_label, num_labeled=5, n_splits=4):      
    print('Training and testing the model...')
    skf = StratifiedKFold(n_splits=n_splits)

    max_accuracies = []
    best_C_values = []
    best_gamma_values = []

    for train_indices, test_indices in skf.split(train_data, train_label):
        train_x, train_y = train_data[train_indices], train_label[train_indices]
        test_x, test_y = train_data[test_indices], train_label[test_indices]

        C_values = np.logspace(-2, 8, 11, base=2)
        gamma_values = np.logspace(-2, 8, 11, base=2)

        max_accuracy = 0
        best_C = 0
        best_gamma = 0

        for C in C_values:
            for gamma in gamma_values:
                svr = SVC(decision_function_shape='ovo', kernel='rbf', C=C, gamma=gamma)
                svr.fit(train_x, train_y)
                accuracy = svr.score(test_x, test_y)
                print('Max Accuracy:', max_accuracy, 'Test Accuracy:', accuracy)

                if accuracy > max_accuracy:
                    max_accuracy = accuracy
                    best_C = C
                    best_gamma = gamma

        max_accuracies.append(max_accuracy)
        best_C_values.append(best_C)
        best_gamma_values.append(best_gamma)

    best_index = np.argmax(max_accuracies)
    best_C = best_C_values[best_index]
    best_gamma = best_gamma_values[best_index]

    svr = SVC(decision_function_shape='ovo', kernel='rbf', C=best_C, gamma=best_gamma)
    svr.fit(train_data, train_label)
    test_accuracy = svr.score(test_data, test_label)

    print('Best Test Accuracy:', test_accuracy)

def main():
    start_time = time.time()

    num_labeled = 5
    dataSet = read_data_sets('I_feature.h5', num_labeled)
    train_data = dataSet.train.images
    train_label = dataSet.train.labels
    test_data = dataSet.test.images
    test_label = dataSet.test.labels

    train_and_test_model(train_data, train_label, test_data, test_label)

    end_time = time.time()
    print('Time taken:', end_time - start_time)

if __name__ == "__main__":
    main()
