import collections

class Solution:
    def intersect(self, nums1, nums2):
        # Create a Counter object to count the occurrences of elements in nums2
        nums2_counter = collections.Counter(nums2)

        # Initialize an empty list to store the intersection of nums1 and nums2
        intersection = []

        # Iterate through each element in nums1
        for num in nums1:
            # Check if the element exists in nums2_counter and its count is greater than 0
            if nums2_counter[num] > 0:
                # Add the element to the intersection list
                intersection.append(num)
                # Decrement the count of the element in nums2_counter
                nums2_counter[num] -= 1

        return intersection

# Example usage
nums1 = [1, 2, 2, 1]
nums2 = [2, 2]
solution = Solution()
intersection = solution.intersect(nums1, nums2)
print("Intersection:", intersection)
