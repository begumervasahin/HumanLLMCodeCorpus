import csv
import string

def get_player_list():
    """
    Prompt the user to enter the filename of the CSV file and read player information from it.

    Returns:
        list: A list containing dictionaries of player information.
    """
    filename = input('Enter filename (.csv): ') + '.csv'

    player_list = []
    with open(filename, 'r', newline='') as file:
        reader = csv.reader(file)
        next(reader)  # Skip header row
        for row in reader:
            player = {
                'FIRST': row[2],
                'LAST': row[4],
                'POS': row[1],
                'PRICE': row[6],
                'TEAM': row[9],
                'OPP': row[10],
                'INJ': row[11]
            }
            player_list.append(player)
    return player_list

def trim_names(player_list):
    """
    Remove punctuation from the first and last names of each player in the player list.

    Args:
        player_list (list): A list containing dictionaries of player information.

    Returns:
        list: A list containing dictionaries of player information with trimmed names.
    """
    for player in player_list:
        player['FIRST'] = player['FIRST'].translate(str.maketrans('', '', string.punctuation))
        player['LAST'] = player['LAST'].translate(str.maketrans('', '', string.punctuation))
    return player_list

# Example usage:
if __name__ == "__main__":
    # Get the list of players from the CSV file
    players = get_player_list()

    # Trim names of players in the list
    trimmed_players = trim_names(players)

    # Print the modified player information
    for player in trimmed_players:
        print(player)
