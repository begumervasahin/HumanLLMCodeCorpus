class Queue:
    """
    A class representing a queue data structure.

    Attributes:
        items (list): A list to store the elements of the queue.
    """

    def __init__(self):
        """Initialize an empty queue."""
        self.items = []

    def is_empty(self):
        """Check if the queue is empty."""
        return not self.items

    def enqueue(self, item):
        """Add an element to the end of the queue."""
        self.items.insert(0, item)

    def dequeue(self):
        """Remove and return the first element from the queue."""
        return self.items.pop()

    def first(self):
        """Return the first element of the queue."""
        return self.items[0]

    def size(self):
        """Return the number of elements in the queue."""
        return len(self.items)

    def enqueue_items(self, items):
        """
        Enqueue multiple items into the queue.

        Args:
            items (list): A list of items to enqueue.
        """
        for item in items:
            self.enqueue(item)


class BQueue:
    """
    A class representing a bounded queue data structure.

    Attributes:
        items (list): A list to store the elements of the queue.
    """

    def __init__(self):
        """Initialize an empty bounded queue."""
        self.items = []

    def is_empty(self):
        """Check if the bounded queue is empty."""
        return not self.items

    def enqueue(self, item):
        """Add an element to the end of the bounded queue."""
        self.items.append(item)

    def dequeue(self):
        """Remove and return the first element from the bounded queue."""
        return self.items.pop()

    def size(self):
        """Return the number of elements in the bounded queue."""
        return len(self.items)


# Example usage:
# Create a Queue and enqueue items
queue = Queue()
queue.enqueue_items([1, 2, 3, 4, 5])

# Print the size of the queue
print("Queue size:", queue.size())

# Dequeue an item from the queue
print("Dequeued item:", queue.dequeue())

# Print the first item in the queue
print("First item:", queue.first())

# Create a BQueue and enqueue items
bqueue = BQueue()
bqueue.enqueue(10)
bqueue.enqueue(20)
bqueue.enqueue(30)

# Print the size of the BQueue
print("BQueue size:", bqueue.size())

# Dequeue an item from the BQueue
print("Dequeued item:", bqueue.dequeue())
