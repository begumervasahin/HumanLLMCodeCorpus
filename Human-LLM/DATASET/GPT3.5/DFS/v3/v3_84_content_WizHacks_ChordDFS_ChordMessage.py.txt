class ChordMessageType:
    # Define constants for different types of messages
    FIND_SUCCESSOR = "FIND_SUCCESSOR"
    RETURN_SUCCESSOR = "RETURN_SUCCESSOR"
    GET_PREDECESSOR = "GET_PREDECESSOR"
    RETURN_PREDECESSOR = "RETURN_PREDECESSOR"
    NOTIFY_PREDECESSOR = "NOTIFY_PREDECESSOR"
    CHECK_ALIVE = "CHECK_ALIVE"
    AM_ALIVE = "AM_ALIVE"
    SEND_FILE = "SEND_FILE"
    REQUEST_FILE = "REQUEST_FILE"
    SOMEONE_DIED = "SOMEONE_DIED"
    LEAVING = "LEAVING"
    INSERT_FILE = "INSERT_FILE"
    GET_FILE = "GET_FILE"
    GET_FILE_LIST = "GET_FILE_LIST"
    ERR = "ERR"
    SUCCESS = "SUCCESS"
    ENTRIES = "ENTRIES"
    OP_SEND_FILE = "OP_SEND_FILE"
    OP_REQ_FILE = "OP_REQUEST_FILE"
    OP_INSERT_FILE = "OP_INSERT_FILE"
    EXIT = "EXIT"
    HELP = "HELP"

def new_message_dict():
    """
    Initialize a new message dictionary with default values.
    """
    return {
        'msg_type': None,       # Type of message
        'filename': None,       # Name of the file (if applicable)
        'finger': None,         # Finger table entry (if applicable)
        'client_ip': None,      # IP address of the client (if applicable)
        'successor_ip': None,   # Successor's IP address (if applicable)
        'key': None,            # Key associated with the message (if applicable)
        'target': None,         # Target of the message (if applicable)
        'predecessor_ip': None, # Predecessor's IP address (if applicable)
        'content': None,        # Content of the message (if applicable)
        'hash_value': None,     # Hash value (if applicable)
        'dead_node': None,      # Dead node (if applicable)
        'file_list': None,      # List of files (if applicable)
        'hops': 0               # Number of hops (if applicable)
    }

# Example usage:
msg = new_message_dict()
msg['msg_type'] = ChordMessageType.FIND_SUCCESSOR
msg['filename'] = "example.txt"
msg['client_ip'] = "192.168.1.100"
print(msg)
