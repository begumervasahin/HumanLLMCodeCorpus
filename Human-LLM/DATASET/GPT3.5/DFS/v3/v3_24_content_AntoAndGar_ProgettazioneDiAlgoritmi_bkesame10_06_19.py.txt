def generate_sequences(n, index=0, current=[]):
    """
    Generate sequences of length n where adjacent elements differ by more than 1.

    Args:
    - n: Length of the sequences.
    - index: Current index during recursion (default: 0).
    - current: Current partial sequence being constructed (default: []).

    This function uses backtracking to generate sequences where adjacent elements differ by more than 1.
    """
    # Base case: if the sequence length reaches n, print the current sequence
    if index == n:
        print(''.join(current))
    else:
        # Iterate through possible values ('0', '1', '2', '3') for the current position
        for value in {'0', '1', '2', '3'}:
            # Check if the current value is valid based on the previous element (if exists)
            if index == 0 or (value != str(int(current[index - 1]) + 1) and value != str(int(current[index - 1]) - 1)):
                # Add the current value to the sequence
                current.append(value)
                # Recur to the next position
                generate_sequences(n, index + 1, current)
                # Backtrack by removing the last element to explore other possibilities
                current.pop()

# Generate sequences of length 6 where adjacent elements differ by more than 1
generate_sequences(6)
