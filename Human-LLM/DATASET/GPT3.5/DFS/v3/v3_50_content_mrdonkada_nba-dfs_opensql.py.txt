import MySQLdb

# Define the path to the SQL file
sql_file_path = 'sql/consistency.sql'

# Read the SQL file and extract the query
with open(sql_file_path) as file:
    sql_query = file.read().replace('\n', ' ')

# Establish a connection to the MySQL database
with MySQLdb.connect('localhost', 'root', '', 'dfs-nba') as connection:
    # Create a cursor object to execute SQL commands
    with connection.cursor() as cursor:
        # Execute the SQL query
        cursor.execute(sql_query)

        # Fetch and print the resulting rows
        rows = cursor.fetchall()
        for row in rows:
            print(row)
