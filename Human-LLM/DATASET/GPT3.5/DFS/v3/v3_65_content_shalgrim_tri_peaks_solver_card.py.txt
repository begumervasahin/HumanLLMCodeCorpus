from enum import IntEnum

class Suit(IntEnum):
    CLUBS = 1
    DIAMONDS = 2
    HEARTS = 3
    SPADES = 4

class Rank(IntEnum):
    ACE = 1
    TWO = 2
    THREE = 3
    FOUR = 4
    FIVE = 5
    SIX = 6
    SEVEN = 7
    EIGHT = 8
    NINE = 9
    TEN = 10
    JACK = 11
    QUEEN = 12
    KING = 13

class Card:
    """
    Represents a playing card with a rank and a suit.
    """
    def __init__(self, rank, suit):
        """
        Initializes a Card instance with the given rank and suit.

        :param rank: The rank of the card (from Rank enum)
        :param suit: The suit of the card (from Suit enum)
        """
        self.rank = rank
        self.suit = suit
        self.covering = set()
        self.covered_by = set()

    def covers(self, other_card):
        """
        Adds another card to the set of cards this card is covering, and vice versa.

        :param other_card: Another Card instance
        :return: None
        """
        self.covering.add(other_card)
        other_card.covered_by.add(self)

    def can_play(self, other_card):
        """
        Determines if this card can be played with another card based on their ranks.

        :param other_card: Another Card instance
        :return: True if the cards can be played together, False otherwise
        """
        if {self.rank, other_card.rank} == {Rank.KING, Rank.ACE}:
            return True
        if self.rank + 1 == other_card.rank or self.rank - 1 == other_card.rank:
            return True
        return False

    def __str__(self):
        """
        Returns a human-readable string representation of the card.

        :return: String representation of the card (e.g., "ACE of CLUBS")
        """
        return f"{self.rank.name} of {self.suit.name}"

    def __repr__(self):
        """
        Returns a string representation of the card for debugging purposes.

        :return: String representation of the card
        """
        return f"Card(rank={self.rank.name}, suit={self.suit.name})"

# Example usage:
if __name__ == "__main__":
    # Create a card
    card1 = Card(Rank.ACE, Suit.CLUBS)

    # Print the card
    print(card1)  # Output: ACE of CLUBS
