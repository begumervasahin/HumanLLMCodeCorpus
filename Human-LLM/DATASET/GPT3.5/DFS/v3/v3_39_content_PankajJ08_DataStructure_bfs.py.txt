from collections import defaultdict, deque

class Graph:
    def __init__(self):
        """Initialize an empty graph."""
        self.adjacency_list = defaultdict(list)

    def add_edge(self, start_vertex, end_vertex):
        """Add an edge between the start_vertex and end_vertex."""
        self.adjacency_list[start_vertex].append(end_vertex)

    def bfs(self, start_vertex):
        """Perform Breadth First Search traversal starting from the specified start_vertex."""
        visited = defaultdict(bool)  # Keeps track of visited vertices
        distance = defaultdict(int)  # Stores the distance of each vertex from the start_vertex
        queue = deque()  # Queue for BFS traversal
        queue.append(start_vertex)  # Enqueue the start_vertex
        visited[start_vertex] = True  # Mark start_vertex as visited
        distance[start_vertex] = 0  # Set distance of start_vertex to 0

        while queue:
            current_vertex = queue.popleft()  # Dequeue a vertex
            print(current_vertex, end=" ")  # Print the visited vertex

            # Visit all neighbors of the current_vertex
            for neighbor in self.adjacency_list[current_vertex]:
                if not visited[neighbor]:
                    queue.append(neighbor)  # Enqueue unvisited neighbors
                    visited[neighbor] = True  # Mark neighbor as visited
                    distance[neighbor] = distance[current_vertex] + 1  # Update distance

    graph = Graph()
