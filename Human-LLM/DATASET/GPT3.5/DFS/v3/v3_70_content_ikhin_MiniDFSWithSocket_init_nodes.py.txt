import os
from commute.multiclient import MultiClient
from commute.multiserver import MultiServer
from core.common import *
from core.datanodehandler import DataNodeHandler
from core.namenodehandler import NameNodeHandler

def initialize_data_nodes(name_prefix, num_servers):
    """
    Initialize data nodes.

    Args:
        name_prefix (str): Prefix for the data node names.
        num_servers (int): Number of data nodes to initialize.
    """
    for server_id in range(num_servers):
        data_node_name = f"{name_prefix}{server_id}"
        data_node = MultiServer(DataNodeHandler)
        DataNodeHandler.server_id = server_id
        data_node.name = data_node_name
        data_node.bind((DATA_NODE_FIRST[0], DATA_NODE_FIRST[1] + server_id))
        data_node.listen()
        data_node.run()

def initialize_name_node(name):
    """
    Initialize the name node.

    Args:
        name (str): Name of the name node.
    """
    def create_dfs_directories():
        """
        Create necessary directories for the DFS if they don't exist.
        """
        dfs_dir = "dfs"
        if not os.path.isdir(dfs_dir):
            os.makedirs(dfs_dir)
            for i in range(NUM_DATA_SERVER):
                os.makedirs(os.path.join(dfs_dir, f"datanode{i}"))
            os.makedirs(os.path.join(dfs_dir, "namenode"))

    # Create DFS directories
    create_dfs_directories()

    # Define addresses for connecting to data nodes
    data_node_addresses = [(DATA_NODE_FIRST[0], DATA_NODE_FIRST[1] + i) for i in range(NUM_DATA_SERVER)]

    # Initialize name node
    name_node = MultiClient(NameNodeHandler, NAME_NODE)
    name_node.name = name
    for address in data_node_addresses:
        name_node.connect(address)
    name_node.run()

if __name__ == "__main__":
    # Initialize data nodes
    initialize_data_nodes("datanode", NUM_DATA_SERVER)

    # Initialize name node
    initialize_name_node("namenode")
