class Cube:
    def __init__(self, name, on=None, clear=False, on_table=True):
        self.name = name
        self.on = on
        self.clear = clear
        self.on_table = on_table

def set_world(data):
    initial_state = create_state(data['initialState'])
    final_state = create_state(data['finalState'])
    return {'initial': initial_state, 'final': final_state}

def create_state(tower_arrangement):
    state = {'operators': [], 'cubes': [], 'previous_states': []}
    for tower in tower_arrangement:
        cubes_in_tower = list(tower)
        for i in range(len(cubes_in_tower)):
            cube_name = cubes_in_tower[i]
            cube_on = None
            cube_clear = False
            cube_on_table = True
            if i < len(cubes_in_tower) - 1:
                cube_on = cubes_in_tower[i + 1]
                cube_on_table = False
            if i == 0:
                cube_clear = True
            state['cubes'].append(Cube(cube_name, cube_on, cube_clear, cube_on_table))
    state['previous_states'].append(state_to_string(state['cubes']))
    return state

def are_states_equal(state1, state2):
    if len(state1['cubes']) != len(state2['cubes']):
        return False
    for cube1 in state1['cubes']:
        found_matching_cube = False
        for cube2 in state2['cubes']:
            if cube1.name == cube2.name:
                if (cube1.on == cube2.on and cube1.on_table == cube2.on_table and cube1.clear == cube2.clear):
                    found_matching_cube = True
                    break
        if not found_matching_cube:
            return False
    return True

def state_to_string(state):
    state_str = []
    for cube in state:
        if cube.clear:
            state_str.append(cube.name)
    for i in range(len(state_str)):
        not_last = True
        current_cube = state_str[i]
        while not_last:
            for cube in state:
                if current_cube == cube.name:
                    if cube.on is None:
                        not_last = False
                        break
                    state_str[i] += cube.on
                    current_cube = cube.on
                    break
    return state_str

# Example usage:
data = {
    'initialState': ['ABCD', 'EF'],
    'finalState': ['AB', 'CDEF']
}

world = set_world(data)
print(world)
