def stringify_command(command, param_list):
    """
    Convert command and parameters to a JSON string.
    Args:
        command (int): Command to be executed.
        param_list (list): List of parameters for the command.
    Returns:
        str: JSON string representation of the command.
    """
    command_obj = {'command': command, 'param_list': param_list}
    return json.dumps(command_obj)

def stringify_result(result):
    """
    Convert result to a JSON string.
    Args:
        result: Result to be converted.
    Returns:
        str: JSON string representation of the result.
    """
    return json.dumps(result)
