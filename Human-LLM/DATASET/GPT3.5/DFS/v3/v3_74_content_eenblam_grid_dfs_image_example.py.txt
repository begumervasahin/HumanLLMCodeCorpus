from PIL import Image

def create_image_with_rgb_tuples(rgb_tuples, image_size, background_color):
    """
    Create a new image with specified RGB tuples, size, and background color.

    Args:
        rgb_tuples (list): List of RGB tuples.
        image_size (tuple): Size of the image (width, height).
        background_color (str or tuple): Background color of the image.

    Returns:
        Image: PIL Image object.
    """
    # Create a new image with the specified background color
    image = Image.new("RGB", image_size, background_color)
    
    # Put the RGB tuples data into the image
    image.putdata(rgb_tuples)
    
    return image

def display_image(image):
    """
    Display the given PIL Image object.

    Args:
        image (Image): PIL Image object to display.
    """
    # Display the image
    image.show()

if __name__ == "__main__":
    # Define a list of RGB tuples
    rgb_tuples = [
        (0, 0, 0),
        (0, 0, 0),
        (128, 128, 128),
        (256, 256, 256),
        (256, 256, 256)
    ] * 2000

    # Define the image size and background color
    image_size = (100, 100)
    background_color = "white"

    # Create the image
    image = create_image_with_rgb_tuples(rgb_tuples, image_size, background_color)

    # Display the image
    display_image(image)
