import numpy as np
from collections import defaultdict

class Solver:
    def __init__(self, state_space, num_cols, start, goal, technique):
        self.start = start
        self.num_cols = num_cols
        self.goal = goal
        self.state_space = state_space
        self.path = []
        self.explored = []
        print(f"Initialized solver using technique: {technique}")
        
        if technique == "bfs":
            self.bfs()
        elif technique == "dfs":
            self.dfs()
        elif technique == "astar":
            self.astar()

    def get_path_from_visited(self):
        path = []
        pointer = self.goal
        while pointer != self.start:
            path.append(pointer)
            pointer = self.state_space[pointer]["parent "]
        path.reverse()
        return path

    def bfs(self):
        queue = []
        queue.append(self.state_space[self.start])
        while queue:
            current = queue.pop(0)
            self.explored.append(current["name "])
            self.path.append(current["name "])
            if current["name "] == self.goal:
                print("CONGRATULATIONS! A path exists, printing all expanded nodes")
                print(self.path)
                return
            queue.extend([self.state_space[neighbor] for neighbor in current["connections "] if neighbor not in self.explored])
        print("WARNING! A valid path does not exist, printing all expanded nodes")
        print(self.path)

    def dfs(self):
        stack = []
        stack.append(self.state_space[self.start])
        while stack:
            current = stack.pop()
            self.explored.append(current["name "])
            self.path.append(current["name "])
            if current["name "] == self.goal:
                print("CONGRATULATIONS! A path exists, printing all expanded nodes")
                print(self.path)
                return
            stack.extend([self.state_space[neighbor] for neighbor in current["connections "] if neighbor not in self.explored])
        print("WARNING! A valid path does not exist, printing all expanded nodes")
        print(self.path)

    def astar(self):
        for node in self.state_space:
            node["g"] = 0
            node["h"] = 0
        self.state_space[self.start]["parent "] = -1
        open_list = [self.state_space[self.start]]
        while open_list:
            open_list.sort(key=lambda x: x["g"] + x["h"])
            current = open_list.pop(0)
            self.explored.append(current["name "])
            self.path.append(current["name "])
            if current["name "] == self.goal:
                print("CONGRATULATIONS! A path exists, printing all expanded nodes")
                print(self.path)
                return
            for neighbor_name in current["connections "]:
                neighbor = self.state_space[neighbor_name]
                neighbor_g = current["g"] + 1
                if neighbor not in open_list and neighbor_name not in self.explored:
                    open_list.append(neighbor)
                    neighbor["parent "] = current["name "]
                    neighbor["g"] = neighbor_g
                    neighbor["h"] = self.heuristic_cost(neighbor["name "])
                elif neighbor_g < neighbor["g"]:
                    neighbor["parent "] = current["name "]
                    neighbor["g"] = neighbor_g
        print("WARNING! A valid path does not exist, printing all expanded nodes")
        print(self.path)

    def heuristic_cost(self, current):
        y_dist = np.abs((self.goal - current) / self.num_cols)
        x_dist = np.abs((self.goal - current)) % self.num_cols
        return y_dist + x_dist
