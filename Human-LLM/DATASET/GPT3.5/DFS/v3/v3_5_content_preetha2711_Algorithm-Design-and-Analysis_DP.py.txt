# Define the parameters for Pascal's triangle
num_rows = 1000
num_cols = 100

# Initialize Pascal's triangle as a 2D list
pascals_triangle = [[None] * (num_cols + 1) for _ in range(num_rows + 1)]

# Calculate Pascal's triangle values
for row in range(1, num_rows + 1):
    for col in range(1, min(row, num_cols) + 1):
        if col == 1:
            # The first element in each row is the row number
            pascals_triangle[row][col] = row
        else:
            # Calculate the value using the recursive Pascal's triangle formula
            pascals_triangle[row][col] = pascals_triangle[row - 1][col] + pascals_triangle[row - 1][col - 1]

# Print the desired value from Pascal's triangle
target_row = num_rows
target_col = num_cols
target_value = pascals_triangle[target_row][target_col]
print(f"The value at row {target_row} and column {target_col} in Pascal's triangle is: {target_value}")
