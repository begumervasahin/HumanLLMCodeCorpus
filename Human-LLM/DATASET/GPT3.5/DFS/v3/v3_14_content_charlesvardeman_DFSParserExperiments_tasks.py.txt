import os
from invoke import task, run

# Define directories
DOCS_DIR = 'docs'
BUILD_DIR = os.path.join(DOCS_DIR, '_build')


@task
def test():
    """Run tests."""
    run('python setup.py test', pty=True)


@task
def clean():
    """Clean up build artifacts."""
    run("rm -rf build")
    run("rm -rf dist")
    run("rm -rf dfs_parser.egg-info")
    clean_docs()
    print("Cleaned up.")


@task
def clean_docs():
    """Clean up documentation artifacts."""
    run(f"rm -rf {BUILD_DIR}")


@task
def browse_docs():
    """Open documentation in browser."""
    run(f"open {os.path.join(BUILD_DIR, 'index.html')}")


@task
def build_docs(clean=False, browse=False):
    """Build documentation."""
    if clean:
        clean_docs()
    run(f"sphinx-build {DOCS_DIR} {BUILD_DIR}", pty=True)
    if browse:
        browse_docs()


@task
def generate_readme():
    """Generate HTML from README."""
    run('rst2html.py README.rst > README.html')


@task
def publish(test=False):
    """Publish to the cheeseshop."""
    if test:
        run('python setup.py register -r test sdist upload -r test')
    else:
        run("python setup.py register sdist upload")


if __name__ == "__main__":
    # Example usage
    test()
    clean()
    build_docs(clean=True, browse=True)
    generate_readme()
    publish(test=True)
