def depth_first_search(graph, start, end):
    """
    Perform Depth-First Search (DFS) on the given graph to find a path from the start node to the end node.

    Args:
        graph (dict): The graph represented as a dictionary.
        start (str): The starting node.
        end (str): The destination node.

    Returns:
        list or None: The path from the start node to the end node, or None if no path is found.
    """
    # Initialize stack with the starting node
    stack = [[start]]
    # Initialize set to keep track of visited nodes
    visited = set()

    while stack:
        # Pop the last path from the stack
        current_path = stack.pop()
        # Get the current state (last node) from the path
        current_state = current_path[-1]

        # If the current state is the destination, return the path
        if current_state == end:
            return current_path

        # If the current state has not been visited yet
        elif current_state not in visited:
            # Explore all adjacent nodes of the current state
            for neighbor in graph.get(current_state, []):
                # Create a new path by extending the current path with the neighbor node
                new_path = current_path + [neighbor]
                # Push the new path onto the stack
                stack.append(new_path)

            # Mark the current state as visited
            visited.add(current_state)

    # If no path is found, print a message
    print("No path found")
    return None

if __name__ == "__main__":
    # Define the graph
    graph = {
        'A': {'B', 'C'},
        'B': {'A', 'E'},
        'C': {'A', 'D', 'F', 'K'},
        'D': {'C', 'E', 'J'},
        'E': {'B', 'D', 'H'},
        'F': {'C', 'G'},
        'G': {'F', 'J'},
        'H': {'E', 'I'},
        'I': {'H', 'J', 'L'},
        'J': {'D', 'G', 'I'},
        'K': {'C'},
        'L': {'I'}
    }

    # Find a path from node 'A' to node 'H' using DFS
    path = depth_first_search(graph, 'A', 'H')
    print(path)
