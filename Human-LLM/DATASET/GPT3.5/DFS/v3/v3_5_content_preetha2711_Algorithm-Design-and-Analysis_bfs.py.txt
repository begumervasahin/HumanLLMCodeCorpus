from collections import deque

def bfs_traversal(vertices, adjacency_list, source):
    visited = [None] * len(vertices)
    visited[source] = 0
    queue = deque([source])

    while queue:
        current_vertex = queue.popleft()
        print(f"Visited vertex: {current_vertex}")
        for neighbor in adjacency_list[current_vertex]:
            if visited[neighbor] is None:
                visited[neighbor] = 0
                queue.append(neighbor)

    if None in visited:
        print("The graph is disconnected")
    else:
        print("All nodes of the graph have been traversed")

def detect_cycle_bfs(adjacency_list, source):
    parent = [None] * len(adjacency_list)
    visited = [None] * len(adjacency_list)
    parent[source] = None
    visited[source] = 0
    queue = deque([source])

    while queue:
        current_vertex = queue.popleft()
        for neighbor in adjacency_list[current_vertex]:
            if parent[neighbor] is not None:
                print("Cycle exists")
                return
            else:
                visited[neighbor] = 0
                parent[neighbor] = current_vertex
                queue.append(neighbor)

# Example data
vertices = [0, 1, 2, 3, 4]
adjacency_list = [[1, 2], [2, 3], [3, 4], [4, 0], [0, 1]]

# BFS traversal
source_vertex = int(input("Enter source vertex for BFS traversal: "))
bfs_traversal(vertices, adjacency_list, source_vertex)

# Check for cycle using BFS
cycle_source = int(input("Enter source vertex to check for cycle: "))
detect_cycle_bfs(adjacency_list, cycle_source)
