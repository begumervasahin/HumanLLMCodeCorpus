from graphviz import Digraph
import copy

class TreeNode:
    def __init__(self, name, last_time_of_visit):
        self.name = name
        self.last_time_of_visit = last_time_of_visit
        self.children = []

def get_unique_name(existing_names, name):
    base_name = name.split('-')[0]
    for i in range(1, len(existing_names) + 1):
        new_name = f"{base_name}-{i}"
        if new_name not in existing_names:
            return new_name

def resolve_duplicate_names(root_node):
    existing_names = {root_node.name}
    queue = [root_node]
    while queue:
        current_node = queue.pop(0)
        for child in current_node.children:
            queue.append(child)
            if child.name not in existing_names:
                existing_names.add(child.name)
            else:
                child.name = get_unique_name(existing_names, child.name)
                existing_names.add(child.name)

def visualize_tree(root_node, output_file):
    resolved_root = copy.deepcopy(root_node)
    resolve_duplicate_names(resolved_root)
    graph = Digraph()
    graph.node(resolved_root.name)
    queue = [resolved_root]
    while queue:
        current_node = queue.pop(0)
        for child in current_node.children:
            queue.append(child)
            graph.node(child.name)
            graph.edge(current_node.name, child.name, str(child.last_time_of_visit), minlen='2', dir="forward")
    graph.view(output_file)

if __name__ == "__main__":
    root = TreeNode("A", 0)
    b = TreeNode("B", 1)
    c = TreeNode("C", 2)
    d = TreeNode("A", 3)
    root.children = [b, c, d]
    visualize_tree(root, "tree_output")
