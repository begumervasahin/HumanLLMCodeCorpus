from fs.memoryfs import MemoryFS

# Create a MemoryFS instance
memory_fs = MemoryFS()

# Create directories 'test' and 'foo' in MemoryFS
memory_fs.makedir('test')
memory_fs.makedir('foo')

# Walk through the directories and print information
for path, info in memory_fs.walk.info(max_depth=1):
    if info.is_dir:
        print(f"[Directory] {path}")
    else:
        print(f"[File] {path}")
