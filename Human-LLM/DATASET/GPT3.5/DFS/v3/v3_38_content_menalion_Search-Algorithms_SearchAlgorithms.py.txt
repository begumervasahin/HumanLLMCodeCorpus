from queue import Queue, PriorityQueue
from math import sqrt, pow, abs

class Node:
    def __init__(self, value):
        self.value = value
        self.id = None
        self.up = None
        self.down = None
        self.left = None
        self.right = None
        self.previousNode = None
        self.edgeCost = None
        self.gOfN = None
        self.hOfN = None
        self.heuristicFn = None
        self.visited = False

class SearchAlgorithms:
    def __init__(self, mazeStr, edgeCost=None):
        self.grid = []
        self.rows = []
        self.fullPath = []
        self.stack = []
        self.queue = Queue()
        self.priorityQueue = PriorityQueue()
        self.path = []
        self.totalCost = -1
        self.idCounter = 0
        self.startId = 0
        self.rowsCount = 0
        self.colsCount = 0
        self.costCounter = 0
        self.endX = 0
        self.endY = 0

        self.initializeGrid(mazeStr)
        self.initializeNodes(edgeCost)

    def initializeGrid(self, mazeStr):
        self.rows = mazeStr.replace(',', '').split()
        self.rowsCount = len(self.rows)
        self.colsCount = len(self.rows[0])

        for row in self.rows:
            self.grid.append([Node(char) for char in row])

        for i in range(self.rowsCount):
            for j in range(self.colsCount):
                node = self.grid[i][j]
                node.id = self.idCounter
                self.idCounter += 1

                if node.value == 'S':
                    self.startId = node.id
                elif node.value == 'E':
                    self.endX = i
                    self.endY = j

                if i > 0 and self.grid[i-1][j].value != '':
                    node.up = self.grid[i-1][j].id
                if i < self.rowsCount - 1 and self.grid[i+1][j].value != '':
                    node.down = self.grid[i+1][j].id
                if j > 0 and self.grid[i][j-1].value != '':
                    node.left = self.grid[i][j-1].id
                if j < self.colsCount - 1 and self.grid[i][j+1].value != '':
                    node.right = self.grid[i][j+1].id

    def initializeNodes(self, edgeCost):
        if edgeCost is not None:
            for i in range(self.rowsCount):
                for j in range(self.colsCount):
                    if self.grid[i][j].value != '':
                        self.grid[i][j].edgeCost = edgeCost[self.costCounter]
                        self.costCounter += 1

    def flattenId(self, id):
        row = id // self.colsCount
        col = id % self.colsCount
        return row, col

    def heuristicEuclidean(self, i, j):
        return sqrt(pow((self.endX - i), 2) + pow((self.endY - j), 2))

    def heuristicManhattan(self, i, j):
        return abs(i - self.endX) + abs(j - self.endY)

    def getChildren(self, rowIdx, colIdx):
        children = []
        node = self.grid[rowIdx][colIdx]
        node.visited = True

        if node.right is not None:
            children.append(node.right)
        if node.left is not None:
            children.append(node.left)
        if node.down is not None:
            children.append(node.down)
        if node.up is not None:
            children.append(node.up)

        return children

    def depthFirstSearch(self):
        self.stack = [self.startId]

        while self.stack:
            currentId = self.stack.pop()
            rowIdx, colIdx = self.flattenId(currentId)
            currentNode = self.grid[rowIdx][colIdx]

            if currentNode.value == "E":
                self.fullPath.append(currentId)
                break

            if not currentNode.visited and currentId not in self.stack:
                self.fullPath.append(currentId)
                currentNode.visited = True
                children = self.getChildren(rowIdx, colIdx)
                self.stack.extend(children)

        return self.path, self.fullPath

    def breadthFirstSearch(self):
        self.queue.put(self.startId)

        while not self.queue.empty():
            currentId = self.queue.get()
            rowIdx, colIdx = self.flattenId(currentId)
            currentNode = self.grid[rowIdx][colIdx]

            if currentNode.value == "E":
                self.fullPath.append(currentId)
                break

            if not currentNode.visited:
                self.fullPath.append(currentId)
                currentNode.visited = True
                children = self.getChildren(rowIdx, colIdx)
                self.queue.queue.extend(children)

        return self.path, self.fullPath

    def uniformCostSearch(self):
        self.priorityQueue.put((0, self.startId))

        while not self.priorityQueue.empty():
            cost, currentId = self.priorityQueue.get()
            rowIdx, colIdx = self.flattenId(currentId)
            currentNode = self.grid[rowIdx][colIdx]

            if not currentNode.visited:
                self.fullPath.append(currentId)

                if currentNode.value == 'E':
                    break

                children = self.getChildren(rowIdx, colIdx)
                children.reverse()
                currentNode.visited = True

                for childId in children:
                    childRow, childCol = self.flattenId(childId)
                    childNode = self.grid[childRow][childCol]

                    if childId not in self.fullPath:
                        childGOfN = cost + childNode.edgeCost
                        self.totalCost = childGOfN
                        self.priorityQueue.put((self.totalCost, childId))
                        childNode.gOfN = childGOfN

        return self.path, self.fullPath, self.totalCost

    def aStarEuclideanHeuristic(self):
        for i in range(self.rowsCount):
            for j in range(self.colsCount):
                self.grid[i][j].hOfN = self.heuristicEuclidean(i, j)

        self.priorityQueue.put((0, self.startId))
        rowIdx, colIdx = self.flattenId(self.startId)
        self.grid[rowIdx][colIdx].gOfN = 0

        while not self.priorityQueue.empty():
            _, currentId = self.priorityQueue.get()
            rowIdx, colIdx = self.flattenId(currentId)
            cost = self.grid[rowIdx][colIdx].gOfN

            if not self.grid[rowIdx][colIdx].visited:
                self.fullPath.append(currentId)

                if self.grid[rowIdx][colIdx].value == 'E':
                    break

                children = self.getChildren(rowIdx, colIdx)
                children.reverse()
                self.grid[rowIdx][colIdx].visited = True

                for childId in children:
                    if childId not in self.fullPath:
                        childRow, childCol = self.flattenId(childId)
                        childNode = self.grid[childRow][childCol]
                        childNode.gOfN = cost + childNode.edgeCost
                        self.totalCost = childNode.gOfN
                        childNode.heuristicFn = childNode.gOfN + childNode.hOfN
                        self.priorityQueue.put((childNode.heuristicFn, childId))

        return self.path, self.fullPath, self.totalCost

    def aStarManhattanHeuristic(self):
        for i in range(self.rowsCount):
            for j in range(self.colsCount):
                self.grid[i][j].hOfN = self.heuristicManhattan(i, j)

        self.priorityQueue.put((0, self.startId))
        rowIdx, colIdx = self.flattenId(self.startId)
        self.grid[rowIdx][colIdx].gOfN = 0

        while not self.priorityQueue.empty():
            _, currentId = self.priorityQueue.get()
            rowIdx, colIdx = self.flattenId(currentId)
            cost = self.grid[rowIdx][colIdx].gOfN

            if not self.grid[rowIdx][colIdx].visited:
                self.fullPath.append(currentId)

                if self.grid[rowIdx][colIdx].value == 'E':
                    break

                children = self.getChildren(rowIdx, colIdx)
                children.reverse()
                self.grid[rowIdx][colIdx].visited = True

                for childId in children:
                    if childId not in self.fullPath:
                        childRow, childCol = self.flattenId(childId)
                        childNode = self.grid[childRow][childCol]
                        childNode.gOfN = cost + childNode.edgeCost
                        self.totalCost = childNode.gOfN
                        childNode.heuristicFn = childNode.gOfN + childNode.hOfN
                        self.priorityQueue.put((childNode.heuristicFn, childId))

        return self.path, self.fullPath, self.totalCost

def main():
    searchAlgo = SearchAlgorithms('S,.,.,\n.,E,.\n.,.,.')
    path, fullPath = searchAlgo.depthFirstSearch()
    print('**DFS**\nPath is:', path, '\nFull Path is:', fullPath, '\n\n')

    searchAlgo = SearchAlgorithms('S,.,.,\n.,E,.\n.,.,.')
    path, fullPath = searchAlgo.breadthFirstSearch()
    print('**BFS**\nPath is:', path, '\nFull Path is:', fullPath, '\n\n')

    searchAlgo = SearchAlgorithms('S,.,.,\n.,E,.\n.,.,.', [100, 2, 15, 60, 100, 30, 2, 100, 2, 2, 2, 40, 30, 2, 2, 100, 100, 3, 15, 30, 100, 2, 100, 0, 2, 100, 30])
    path, fullPath, TotalCost = searchAlgo.uniformCostSearch()
    print('**UCS**\nPath is:', path, '\nFull Path is:', fullPath, '\nTotal Cost:', TotalCost, '\n\n')

    searchAlgo = SearchAlgorithms('S,.,.,\n.,E,.\n.,.,.', [100, 2, 15, 60, 100, 30, 2, 100, 2, 2, 2, 40, 30, 2, 2, 100, 100, 3, 15, 30, 100, 2, 100, 0, 2, 100, 30])
    path, fullPath, TotalCost = searchAlgo.aStarEuclideanHeuristic()
    print('**ASTAR with Euclidean Heuristic**\nPath is:', path, '\nFull Path is:', fullPath, '\nTotal Cost:', TotalCost, '\n\n')

    searchAlgo = SearchAlgorithms('S,.,.,\n.,E,.\n.,.,.')
    path, fullPath, TotalCost = searchAlgo.aStarManhattanHeuristic()
    print('**ASTAR with Manhattan Heuristic**\nPath is:', path, '\nFull Path is:', fullPath, '\nTotal Cost:', TotalCost, '\n\n')

if __name__ == "__main__":
    main()
