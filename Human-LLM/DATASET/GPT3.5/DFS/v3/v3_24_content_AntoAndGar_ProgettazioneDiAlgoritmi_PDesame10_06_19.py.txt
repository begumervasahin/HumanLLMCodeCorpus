# Define the value of n
n = 5

# Initialize the list T with zeros
sequence_values = [0] * (n + 1)

# Function to calculate the sequence value at index n
def calculate_sequence_value(n):
    # Base cases
    sequence_values[0] = 1
    sequence_values[1] = 4
    sequence_values[2] = 10

    # Calculate values for indices 3 to n
    for i in range(3, n + 1):
        sequence_values[i] = 3 * sequence_values[i - 1] - sequence_values[i - 2]

    return sequence_values[n]

# Call the function and print the result
print(calculate_sequence_value(n))
