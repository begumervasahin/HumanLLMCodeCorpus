def breadth_first_search(graph, start, end):
    queue = [[start]]
    visited = set()

    while queue:
        path = queue.pop(0)
        node = path[-1]

        if node == end:
            return path

        if node not in visited:
            visited.add(node)
            neighbors = graph.get(node, [])
            for neighbor in neighbors:
                new_path = path + [neighbor]
                queue.append(new_path)
    
    print("Route not found. Please try again.")

def depth_first_search(graph, start, end):
    stack = [[start]]
    visited = set()

    while stack:
        path = stack.pop()
        node = path[-1]

        if node == end:
            return path

        if node not in visited:
            visited.add(node)
            neighbors = graph.get(node, [])
            for neighbor in neighbors:
                new_path = path + [neighbor]
                stack.append(new_path)

    print("Route not found. Please try again.")

# Define the spiders graph
spiders = {
    '1': ['2', '3', '4'],
    '2': ['3', '8'],
    '3': ['4', '5', '6', '7', '8'],
    '4': ['5'],

    '7': ['8', '9', '10', '11'],
    '8': ['9'],
    '9': ['10'],
    '10': ['11']
}

# Define start and end nodes
start_node = '1'
end_node = '11'

# Print routes found by BFS and DFS algorithms
print("BFS route:", breadth_first_search(spiders, start_node, end_node))
