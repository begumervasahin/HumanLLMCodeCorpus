class Solution:
    def postorderTraversal(self, root):
        """
        Traverse a binary tree using postorder traversal.

        :param root: TreeNode, the root node of the binary tree.
        :return: List[int], containing the postorder traversal of the tree.
        """
        result = []        # List to store the traversal result
        stack = []         # Stack for iterative traversal
        current_node = root  # Start from the root node
        visited = set()    # Set to keep track of visited nodes

        while current_node or stack:
            if current_node:
                # Traverse left subtree until reaching the leftmost leaf
                stack.append(current_node)
                current_node = current_node.left
            
            else:
                # Process the right subtree and backtrack if necessary
                current_node = stack.pop()
                
                if current_node.right and current_node.right not in visited:
                    # If right subtree exists and hasn't been visited yet
                    stack.append(current_node)  # Push the current node back to stack
                    current_node = current_node.right  # Traverse the right subtree
                
                else:
                    # If no right subtree or already visited, add to result
                    visited.add(current_node)   # Mark the current node as visited
                    result.append(current_node.val)  # Add node value to result
                    current_node = None  # Reset current node to None to backtrack to parent
        
        return result
