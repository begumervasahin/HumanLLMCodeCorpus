import argparse
import matplotlib.pyplot as plt
import pandas as pd

def plot_scores(data, title=None):
    """Plot mean and median scores against steps."""
    for col in ['mean', 'median']:
        plt.plot(data['steps'], data[col], label=col)

    if title:
        plt.title(title)

    plt.xlabel('steps')
    plt.ylabel('score')
    plt.legend(loc='best')

def save_plot_as_png(filename):
    """Save the current plot as a PNG file."""
    plt.savefig(filename)
    print(f'Saved the figure as {filename}')

def main():
    # Set up command-line argument parser
    parser = argparse.ArgumentParser(description='Plot mean and median scores from a data file.')
    parser.add_argument('scores', type=str, help='path to scores data file (e.g., scores.txt)')
    parser.add_argument('--title', type=str, default=None, help='title for the plot (optional)')
    args = parser.parse_args()

    # Read scores data from the specified file
    scores_data = pd.read_csv(args.scores, delimiter='\t')

    # Plot the scores data
    plot_scores(scores_data, args.title)

    # Save the plot as a PNG file
    save_plot_as_png(args.scores + '.png')

if __name__ == '__main__':
    main()
