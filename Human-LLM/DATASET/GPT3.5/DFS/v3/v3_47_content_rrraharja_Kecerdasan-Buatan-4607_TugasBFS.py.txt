def bfs(adjacency_list, start_node, end_node):
    """
    Perform breadth-first search on a graph to find a path from start_node to end_node.

    Args:
    - adjacency_list: The graph represented as a dictionary of nodes and their neighbors.
    - start_node: The starting node for the search.
    - end_node: The target node to reach.

    Returns:
    - The path from start_node to end_node if found, or None if no path exists.
    """
    queue = [[start_node]]  # Initialize the queue with the start node
    visited = set()  # Keep track of visited nodes

    while queue:
        current_path = queue.pop(0)  # Dequeue the first path from the queue
        current_node = current_path[-1]  # Get the last node in the current path

        if current_node == end_node:
            return current_path  # If the end node is reached, return the path

        if current_node not in visited:
            for neighbor in adjacency_list.get(current_node, []):
                new_path = list(current_path)  # Create a new path by extending the current path
                new_path.append(neighbor)  # Append the neighbor to the new path
                queue.append(new_path)  # Enqueue the new path

            visited.add(current_node)  # Mark the current node as visited

    print(f"No path found from {start_node} to {end_node}")
    return None

# Define the graph as an adjacency list
adjacency_list = {
    'ALASKA': {'CALIFORNIA'},
    'CALIFORNIA': {'CAROLINA', 'ALASKA'},
    'CAROLINA': {'CALIFORNIA', 'INDIANA', 'HAWAII', 'DAREWALE'},
    'DAREWALE': {'CAROLINA', 'EDINBURH', 'HAWAII', 'FLORIDA'},
    'EDINBURH': {'DAREWALE'},
    'FLORIDA': {'DAREWALE', 'GEORGIA'},
    'GEORGIA': {'FLORIDA', 'HAWAII'},
    'HAWAII': {'DARAWALE', 'CAROLINA', 'GEORGIA', 'LOUSIANA'},
    'INDIANA': {'CAROLINA', 'IDAKO', 'KENTUCKY'},
    'IDAKO': {'INDIANA'},
    'KENTUCKY': {'LOUSIANA', 'INDIANA'},
    'LOUSIANA': {'KENTUCKY', 'HAWAII'}
}

# Example usage: Find a path from 'ALASKA' to 'GEORGIA'
result = bfs(adjacency_list, 'ALASKA', 'GEORGIA')
