import re
import sys

class TestParser:
    def __init__(self, path):
        self.path = path

    def remove_comments(self, rawlines):
        """Remove comments from raw lines."""
        fixed_lines = []
        for line in rawlines:
            idx = line.find('#')
            if idx == -1:
                fixed_lines.append(line)
            else:
                fixed_lines.append(line[:idx])
        return '\n'.join(fixed_lines)

    def parse(self):
        """Parse the test file."""
        test = {}
        with open(self.path) as handle:
            raw_lines = handle.read().split('\n')
        
        # Remove comments from raw lines
        test_text = self.remove_comments(raw_lines)
        
        # Initialize test dictionary
        test['__raw_lines__'] = raw_lines
        test['path'] = self.path
        test['__emit__'] = []
        
        # Split test text into lines
        lines = test_text.split('\n')
        i = 0
        while i < len(lines):
            if re.match(r'\A\s*\Z', lines[i]):
                # Append raw line to emit list
                test['__emit__'].append(("raw", raw_lines[i]))
                i += 1
                continue
            m = re.match(r'\A([^"]*?):\s*"([^"]*)"\s*\Z', lines[i])
            if m:
                # Store key-value pair in test dictionary
                test[m.group(1)] = m.group(2)
                # Append key to emit list
                test['__emit__'].append(("oneline", m.group(1)))
                i += 1
                continue
            m = re.match(r'\A([^"]*?):\s*"""\s*\Z', lines[i])
            if m:
                # Store multi-line string in test dictionary
                msg = []
                i += 1
                while not re.match(r'\A\s*"""\s*\Z', lines[i]):
                    msg.append(raw_lines[i])
                    i += 1
                test[m.group(1)] = '\n'.join(msg)
                # Append key to emit list
                test['__emit__'].append(("multiline", m.group(1)))
                i += 1
                continue
            # Error handling for invalid test file format
            print('error parsing test file: %s' % self.path)
            sys.exit(1)
        return test

def emit_test_dict(test_dict, handle):
    """Write the parsed test dictionary to a file."""
    for kind, data in test_dict['__emit__']:
        if kind == "raw":
            handle.write(data + "\n")
        elif kind == "oneline":
            handle.write('%s: "%s"\n' % (data, test_dict[data]))
        elif kind == "multiline":
            handle.write('%s: """\n%s\n"""\n' % (data, test_dict[data]))
        else:
            raise Exception("Bad __emit__")

# Example usage:
if __name__ == "__main__":
    parser = TestParser("test.txt")
    test_dict = parser.parse()
    with open("output.txt", "w") as output_file:
        emit_test_dict(test_dict, output_file)
