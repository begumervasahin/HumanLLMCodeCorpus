def combination_sum(candidates, target):
    candidates.sort()  # Sort the candidates list in ascending order
    result = []  # Initialize an empty list to store the result combinations
    combination_sum_helper(candidates, target, 0, [], result)  # Call the helper function to find combinations
    return result  # Return the final result list containing all valid combinations

def combination_sum_helper(candidates, target, start_index, path, result):
    # Iterate through the candidates starting from the specified index
    for i in range(start_index, len(candidates)):
        num = candidates[i]  # Get the current number from the candidates list
        # If the current number is greater than the target, stop the iteration
        if num > target:
            return
        # If the current number is equal to the target, add the current path to the final result list
        if num == target:
            result.append(path + [num])
            return
        # Recursively call the helper function with updated parameters
        combination_sum_helper(candidates, target - num, i, path + [num], result)

# Example usage:
candidates = [2, 3, 6, 7]
target = 7
# Find all unique combinations that sum up to the target
print(combination_sum(candidates, target))  # Output: [[2, 2, 3], [7]]
