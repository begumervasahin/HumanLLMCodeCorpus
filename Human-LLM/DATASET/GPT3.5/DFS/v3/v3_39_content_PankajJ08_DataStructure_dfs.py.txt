from collections import defaultdict

class Graph:
    def __init__(self):
        """Initialize an empty graph."""
        self.graph = defaultdict(list)

    def add_edge(self, start_vertex, end_vertex):
        """Add an edge between start_vertex and end_vertex."""
        self.graph[start_vertex].append(end_vertex)

    def dfs_visit(self, current_vertex, visited):
        """Recursively visit vertices in depth-first order."""
        visited[current_vertex] = True
        print(current_vertex, end=" ")  # Print the visited vertex
        for neighbor in self.graph[current_vertex]:
            if not visited[neighbor]:
                self.dfs_visit(neighbor, visited)

    def dfs(self, start_vertex=0):
        """Perform Depth First Traversal starting from start_vertex."""
        visited = defaultdict(bool)
        self.dfs_visit(start_vertex, visited)

if __name__ == "__main__":
    # Create a new graph
    graph = Graph()

    # Add edges to the graph
    edges = [(0, 1), (1, 2), (2, 4), (2, 5), (4, 6), (5, 10), (11, 7)]
    for edge in edges:
        graph.add_edge(*edge)

    # Perform DFS starting from vertex 0
    print(" " * 10, "Depth First Traversal starting from vertex 0: ")
    graph.dfs()
