import numpy as np

def stft(source, winlen, slide):
    # Ensure the input signal length is a power of 2 by padding with zeros if necessary
    padded_length = next_power_of_2(len(source))
    if padded_length > len(source):
        source = np.pad(source, (0, padded_length - len(source)), mode='constant')

    # Calculate the number of windows
    num_windows = (len(source) - winlen) // slide + 1

    # Create a matrix to store the windowed segments of the signal
    window_matrix = np.empty((winlen, num_windows))

    # Fill the matrix with the windowed segments of the signal
    for i in range(num_windows):
        start_index = slide * i
        end_index = start_index + winlen
        window_matrix[:, i] = source[start_index:end_index]

    # Apply the Fast Fourier Transform (FFT) to each column of the matrix
    complex_matrix = np.fft.fft(window_matrix, axis=0)

    return complex_matrix

def next_power_of_2(k):
    """Return the next power of 2 greater than or equal to k."""
    return 2 ** (len(bin(k)) - 2)
