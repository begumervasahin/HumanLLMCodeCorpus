def pattern_count(n):
    # Define the set size and initialize the table
    num_elements = 4  # Four elements: '1', '2', '3', '4'
    table = [[0] * num_elements for _ in range(n + 1)]

    # Base cases
    table[1] = [1] * num_elements
    table[2] = [3, 2, 2, 3]

    # Populate the table using dynamic programming
    for i in range(3, n + 1):
        table[i][0] = sum(table[i - 1][j] for j in [0, 2, 3])
        table[i][1] = table[i - 1][1] + table[i - 1][3]
        table[i][2] = table[i - 1][2] + table[i - 1][0]
        table[i][3] = sum(table[i - 1][j] for j in [0, 2, 3])

    # Return the result for n
    return sum(table[n])

# Example usage
n = 6
print(pattern_count(n))
