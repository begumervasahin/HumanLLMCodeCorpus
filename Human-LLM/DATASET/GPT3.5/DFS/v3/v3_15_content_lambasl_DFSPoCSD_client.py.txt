import getopt
import sys
import pickle
from xmlrpc.client import Binary
import xmlrpc.client

class RPCClient:
    def __init__(self, server_url):
        self.server = xmlrpc.client.Server(server_url)

    def put(self, key, value, time_to_live):
        return self.server.put(Binary(key), Binary(value), time_to_live)

    def get(self, key):
        return self.server.get(Binary(key))

    def write_file(self, filename):
        return self.server.write_file(Binary(filename))

    def read_file(self, filename):
        return self.server.read_file(Binary(filename))

    def list_methods(self):
        return self.server.system.listMethods()

    try:
        options, arguments = getopt.getopt(sys.argv[1:], "", ["port="])
    except getopt.GetoptError as error:
        print(str(error))
        sys.exit(2)

    for option, value in options:
        parsed_options[option] = value
