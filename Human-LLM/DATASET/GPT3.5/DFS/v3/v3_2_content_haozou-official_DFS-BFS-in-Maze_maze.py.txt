from random import randint
from myarray2d import Array2D  # Assuming myarray2d is a custom class for 2D arrays

class Maze:
    EMPTY = 0
    OBSTACLE = -1

    def __init__(self, height, width):
        self.maze = Array2D(height, width)
        self.start = (0, 0)
        self.goal = (height - 1, width - 1)
        self.initialize_maze()

    def initialize_maze(self):
        self.maze.clear(Maze.EMPTY)
        num_obstacles = int(self.maze.num_rows() * self.maze.num_cols() * 0.1)
        for _ in range(num_obstacles):
            self.place_random_obstacle()

    def place_random_obstacle(self):
        while True:
            row = randint(0, self.maze.num_rows() - 1)
            col = randint(0, self.maze.num_cols() - 1)
            if (row, col) not in [self.start, self.goal] and self.maze[row, col] != Maze.OBSTACLE:
                self.maze[row, col] = Maze.OBSTACLE
                break

    def get_all_moves(self, row, col):
        moves = []
        for dx, dy in [(0, -1), (-1, 0), (1, 0), (0, 1)]:
            new_x, new_y = col + dx, row + dy
            if 0 <= new_x < self.maze.num_cols() and 0 <= new_y < self.maze.num_rows() \
                    and self.maze[new_y, new_x] != Maze.OBSTACLE:
                moves.append((new_y, new_x))
        return moves

    def print_maze(self):
        for row in range(self.maze.num_rows()):
            for col in range(self.maze.num_cols()):
                if self.maze[row, col] == Maze.EMPTY:
                    print('_', end=' ')
                elif self.maze[row, col] == Maze.OBSTACLE:
                    print('|', end=' ')
                else:
                    print('O', end=' ')
            print()

def main():
    maze = Maze(20, 20)
    maze.print_maze()
    print()
    print(maze.get_all_moves(3, 3))

if __name__ == '__main__':
    main()
