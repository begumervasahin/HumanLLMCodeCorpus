def depth_first_search(graph, start, goal):
    # Initialize a stack with the start node as the initial path
    stack = [[start]]
    # Initialize an empty set to keep track of visited nodes
    visited = set()

    # Continue searching until the stack is empty
    while stack:
        # Pop the last path from the stack (depth-first traversal)
        current_path = stack.pop()
        # Get the current node from the last node of the current path
        current_node = current_path[-1]

        # Check if the current node is the goal node
        if current_node == goal:
            return current_path

        # Check if the current node has not been visited yet
        if current_node not in visited:
            # Get the neighbors of the current node from the graph
            neighbors = graph.get(current_node, [])
            # Loop through the neighbors
            for neighbor in neighbors:
                # Create a new path by appending the neighbor to the current path
                new_path = current_path + [neighbor]
                # Add the new path to the stack
                stack.append(new_path)

            # Mark the current node as visited
            visited.add(current_node)

    # If the loop completes without finding the goal node, print a message and return None
    print("Path not found")
    return None

# Example usage
graph = {
    'A': {'B'},
    'B': {'A', 'C', 'F', 'G'},
    'C': {'B', 'D'},
    'D': {'C', 'E'},
    'E': {'D', 'F'},
    'F': {'G', 'E', 'B'},
    'G': {'F', 'H', 'B'},
    'H': {'I', 'G', 'L'},
    'I': {'H', 'J'},
    'J': {'I', 'K'},
    'K': {'L', 'J'},
    'L': {'K', 'H'}
}

start_node = 'A'
goal_node = 'I'
print(depth_first_search(graph, start_node, goal_node))
