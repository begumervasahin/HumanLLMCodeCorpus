import re

def open_problem(file_name):
    # Read the problem text from the file
    with open(file_name, 'r') as file:
        problem_text = file.read()

    # Clean up the problem text
    problem_text = re.sub(r'[ ]+', ' ', re.sub(r'[ ]*\)\(', ')\n(', re.sub(r'\n', '', problem_text)))

    # Extract segmented data from the problem text
    segmented_data = re.compile(r'\(:objects.*\)\s*|\(:INIT.*\)\s*|\(:goal.*\)\s*', re.IGNORECASE).findall(problem_text)

    # Extract objects, initial state, and goal state from segmented data
    objects = re.findall(r' (\w*)', segmented_data[0])
    init_state = re.findall(r'\((\w*) (\w*)\s?(\w*)?\)', segmented_data[1])
    goal_state = re.findall(r'\(\w* (\w*) (\w*)\)', segmented_data[2])

    # Initialize the initial state dictionary with objects
    initial_state_dict = {obj: ['table', True] for obj in objects}

    # Update the initial state based on INIT statements in the problem text
    for action, item1, item2 in init_state:
        if action == 'ON':
            initial_state_dict[item1][0] = item2
            initial_state_dict[item2][1] = False

    # Initialize the goal state dictionary with objects
    goal_state_dict = {obj: ['table', True] for obj in objects}

    # Update the goal state based on goal statements in the problem text
    for item1, item2 in goal_state:
        goal_state_dict[item1][0] = item2
        goal_state_dict[item2][1] = False

    return initial_state_dict, goal_state_dict, objects

def write_solution(solution, output=''):
    # If output file name is not provided, use a default name
    if output == '':
        output = 'solution.txt'

    # Write the solution to the output file
    with open(output, 'w') as file:
        move_number = 0
        for move in solution:
            move_number += 1
            file.write(f'{move_number}. Move({move[0]}, {move[1]}, {move[2]})\n')
