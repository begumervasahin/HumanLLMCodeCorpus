class ChordMessageType:
    FIND_SUCCESSOR = "FIND_SUCCESSOR"
    RETURN_SUCCESSOR = "RETURN_SUCCESSOR"
    GET_PREDECESSOR = "GET_PREDECESSOR"
    RETURN_PREDECESSOR = "RETURN_PREDECESSOR"
    NOTIFY_PREDECESSOR = "NOTIFY_PREDECESSOR"
    CHECK_ALIVE = "CHECK_ALIVE"
    AM_ALIVE = "AM_ALIVE"
    SEND_FILE = "SEND_FILE"
    REQUEST_FILE = "REQUEST_FILE"
    SOMEONE_DIED = "SOMEONE_DIED"
    LEAVING = "LEAVING"
    INSERT_FILE = "INSERT_FILE"
    GET_FILE = "GET_FILE"
    GET_FILE_LIST = "GET_FILE_LIST"
    ERR = "ERR"
    SUCCESS = "SUCCESS"
    ENTRIES = "ENTRIES"
    OP_SEND_FILE = "OP_SEND_FILE"
    OP_REQ_FILE = "OP_REQUEST_FILE"
    OP_INSERT_FILE = "OP_INSERT_FILE"
    EXIT = "EXIT"
    HELP = "HELP"

def create_new_message():
    """
    Create a new message dictionary with default values.
    """
    return {
        'msg_type': None,           # Type of message
        'filename': None,           # Name of the file
        'finger': None,             # Finger table entry
        'client_ip': None,          # Client's IP address
        'successor_ip': None,       # Successor's IP address
        'key': None,                # Key associated with the message
        'target': None,             # Target of the message
        'pred_ip': None,            # Predecessor's IP address
        'content': None,            # Content of the message
        'hash_value': None,         # Hash value
        'dead_node': None,          # Dead node information
        'file_list': None,          # List of files
        'hops': 0                   # Number of hops
    }

# Example usage:
message = create_new_message()
message['msg_type'] = ChordMessageType.FIND_SUCCESSOR
message['filename'] = "example.txt"
message['client_ip'] = "192.168.1.100"
print(message)
