from PIL import Image

# Define RGB tuples for the image
rgb_tuples = [
    (0, 0, 0),         # Black
    (0, 0, 0),         # Black
    (128, 128, 128),   # Gray
    (256, 256, 256),   # White
    (256, 256, 256)    # White
] * 2000  # Repeat the tuples to create a pattern

# Set image size and background color
image_size = (100, 100)
background_color = "white"

# Create a new image
image = Image.new("RGB", image_size, background_color)

# Put the RGB tuples data into the image
image.putdata(rgb_tuples)

# Display the image
image.show()
