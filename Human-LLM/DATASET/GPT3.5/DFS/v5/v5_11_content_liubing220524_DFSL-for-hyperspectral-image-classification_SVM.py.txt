import time
import numpy as np
from sklearn.svm import SVC
from sklearn.model_selection import StratifiedKFold
import input_data

def train_and_evaluate_model(train_data, train_label, test_data, test_label, num_labeled=5, n_splits=4):
    print('Training...')

    # Stratified K-Fold cross-validation
    skf = StratifiedKFold(n_splits=n_splits)

    # Lists to store max accuracies and corresponding parameters
    max_accuracies = []
    best_C_values = []
    best_gamma_values = []

    # Performing cross-validation
    for train_indices, test_indices in skf.split(train_data, train_label):
        train_x, train_y = train_data[train_indices], train_label[train_indices]
        test_x, test_y = train_data[test_indices], train_label[test_indices]

        # Setting the range of parameters for grid search
        C_values = np.logspace(-2, 8, 11, base=2)
        gamma_values = np.logspace(-2, 8, 11, base=2)

        max_accuracy = 0
        best_C = 0
        best_gamma = 0

        # Grid search for best parameters
        for C in C_values:
            for gamma in gamma_values:
                svr = SVC(decision_function_shape='ovo', kernel='rbf', C=C, gamma=gamma)
                svr.fit(train_x, train_y)
                accuracy = svr.score(test_x, test_y)
                print('Max Accuracy:', max_accuracy, 'Test Accuracy:', accuracy)

                if accuracy > max_accuracy:
                    max_accuracy = accuracy
                    best_C = C
                    best_gamma = gamma

        max_accuracies.append(max_accuracy)
        best_C_values.append(best_C)
        best_gamma_values.append(best_gamma)

    # Selecting the best model based on cross-validation results
    best_index = np.argmax(max_accuracies)
    best_C = best_C_values[best_index]
    best_gamma = best_gamma_values[best_index]

    # Training the final model with the best parameters
    svr = SVC(decision_function_shape='ovo', kernel='rbf', C=best_C, gamma=best_gamma)
    svr.fit(train_data, train_label)

    # Evaluating the model on the test set
    test_accuracy = svr.score(test_data, test_label)
    print('Test Accuracy:', test_accuracy)

    # Printing the time taken for parameter optimization
    end = time.time()
    print('Parameters Optimization Time:', end - start)

def main():
    start = time.time()

    # Setting the number of labeled samples
    num_labeled = 5

    # Loading dataset
    dataSet = input_data.read_data_sets('I_feature.h5', num_labeled)
    train_data = dataSet.train.images
    train_label = dataSet.train.labels
    test_data = dataSet.test.images
    test_label = dataSet.test.labels

    # Training and evaluating the model
    train_and_evaluate_model(train_data, train_label, test_data, test_label)

if __name__ == "__main__":
    main()
