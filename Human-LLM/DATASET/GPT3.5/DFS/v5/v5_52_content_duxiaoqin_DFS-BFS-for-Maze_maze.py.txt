from random import randint
from myarray2d import Array2D  # Assuming Array2D is a custom class provided elsewhere

class Maze:
    EMPTY = 0
    OBSTACLE = -1
    OCCUPIED = 1

    def __init__(self, height, width):
        self.maze = Array2D(height, width)
        self.start = (0, 0)
        self.goal = (height - 1, width - 1)
        self.clear_maze()
        self.generate_obstacles()

    def clear_maze(self):
        """Clear the maze by setting all cells to the EMPTY state."""
        self.maze.clear(Maze.EMPTY)

    def generate_obstacles(self):
        """Generate random obstacles in the maze."""
        obstacle_count = int(self.maze.num_rows() * self.maze.num_cols() * 0.1)
        for _ in range(obstacle_count):
            row, col = self._generate_random_position()
            if (row, col) == self.start or (row, col) == self.goal:
                continue
            self.maze[row, col] = Maze.OBSTACLE

    def get_all_moves(self, row, col):
        """Get all valid moves from the specified cell (row, col)."""
        moves = []
        offsets = [(0, -1), (-1, 0), (1, 0), (0, 1)]
        for dx, dy in offsets:
            new_row, new_col = row + dy, col + dx
            if self._is_valid_position(new_row, new_col) and self.maze[new_row, new_col] != Maze.OBSTACLE:
                moves.append((new_row, new_col))
        return moves

    def print_maze(self):
        """Print the maze in a readable format."""
        for row in range(self.maze.num_rows()):
            for col in range(self.maze.num_cols()):
                print(self._get_cell_char(row, col), end=' ')
            print()

    def _generate_random_position(self):
        """Generate a random position within the maze."""
        return randint(0, self.maze.num_rows() - 1), randint(0, self.maze.num_cols() - 1)

    def _is_valid_position(self, row, col):
        """Check if the position (row, col) is within the maze boundaries."""
        return 0 <= row < self.maze.num_rows() and 0 <= col < self.maze.num_cols()

    def _get_cell_char(self, row, col):
        """Get the character representation of the cell at position (row, col)."""
        if (row, col) == self.start:
            return 'S'
        elif (row, col) == self.goal:
            return 'G'
        elif self.maze[row, col] == Maze.EMPTY:
            return '_'
        elif self.maze[row, col] == Maze.OBSTACLE:
            return '|'
        else:
            return 'O'

def main():
    """Create a Maze object, print the maze, and get all moves from a specific cell."""
    maze = Maze(20, 20)
    maze.print_maze()
    print()
    print(maze.get_all_moves(3, 3))

if __name__ == '__main__':
    main()
