import pandas as pd
import requests
from bs4 import BeautifulSoup as bsoup

def fetch_and_save_data():
    # Define the base URL and iterate over years and weeks
    base_url = 'http://rotoguru1.com/cgi-bin/fyday.pl'
    for year in [2016, 2015, 2014, 2013]:
        for week in range(1, 17):
            # Construct the complete URL with parameters
            url = f'{base_url}?week={week}&year={year}&game=fd&scsv=1'
            # Send a GET request to the URL
            response = requests.get(url)
            # Create a BeautifulSoup object to parse the HTML content
            soup = bsoup(response.content, 'html.parser')
            # Find all 'pre' tags in the HTML content
            pre_tags = soup.find_all('pre')
            # Append the data to the CSV file
            with open('test.csv', 'a+') as csv_file:
                for tag in pre_tags[0]:
                    csv_file.write(str(tag))

if __name__ == '__main__':
    # Call the function to fetch and save data
    fetch_and_save_data()
