from collections import deque

def bfs_traversal(vertices, adjacency_list, source):
    visited = [None] * len(vertices)
    visited[source] = 0
    queue = deque([source])

    while queue:
        current_vertex = queue.popleft()
        print(f"Visited vertex: {current_vertex}")
        for neighbor in adjacency_list[current_vertex]:
            if visited[neighbor] is None:
                visited[neighbor] = 0
                queue.append(neighbor)

    if None in visited:
        print("Graph is disconnected")
    else:
        print("All nodes of the graph have been traversed")

def detect_cycle_bfs(adj_list, source, parent):
    parent[source] = None
    visited[source] = 0
    queue = deque([source])

    while queue:
        vertex = queue.popleft()
        for neighbor in adj_list[vertex]:
            if parent[neighbor] is not None:
                print("Cycle exists")
                return
            else:
                visited[neighbor] = 0
                parent[neighbor] = vertex
                queue.append(neighbor)

# Graph data
vertices = [0, 1, 2, 3, 4]
adjacency_list = [[1, 2], [2, 3], [3, 4], [4, 0], [0, 1]]

# Input source vertex from user
source_vertex = int(input("Enter the source vertex: "))

# Perform BFS traversal
print("BFS Traversal:")
bfs_traversal(vertices, adjacency_list, source_vertex)

# Reset visited and parent arrays for cycle detection
parent = [None] * len(vertices)
visited = [None] * len(vertices)

# Example usage: Detect cycle starting from vertex 2
print("\nCycle Detection:")
detect_cycle_bfs(adjacency_list, 2, parent)
