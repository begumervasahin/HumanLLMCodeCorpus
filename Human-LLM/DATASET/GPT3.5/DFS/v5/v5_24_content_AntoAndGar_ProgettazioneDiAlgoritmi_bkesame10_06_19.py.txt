def generate_sequences(sequence_length, current_index=0, current_sequence=[]):
    """
    Generate sequences of a given length where adjacent elements differ by more than 1.

    Args:
    - sequence_length: Length of the sequences to generate.
    - current_index: Current index during recursion (default: 0).
    - current_sequence: Current partial sequence being constructed (default: []).

    This function uses backtracking to generate sequences where adjacent elements differ by more than 1.
    """
    # Base case: if the sequence length reaches the desired length, print the current sequence
    if current_index == sequence_length:
        print(''.join(current_sequence))
    else:
        # Iterate through possible values ('0', '1', '2', '3') for the current position
        for value in {'0', '1', '2', '3'}:
            # Check if the current value is valid based on the previous element (if it exists)
            if current_index == 0 or (
                value != str(int(current_sequence[current_index - 1]) + 1)
                and value != str(int(current_sequence[current_index - 1]) - 1)
            ):
                # Add the current value to the sequence
                current_sequence.append(value)
                # Recur to the next position
                generate_sequences(sequence_length, current_index + 1, current_sequence)
                # Backtrack by removing the last element to explore other possibilities
                current_sequence.pop()

# Generate sequences of length 6 where adjacent elements differ by more than 1
generate_sequences(6)
