# Define the spider graph
spiders = {
    '1': ['2', '3', '4'],
    '2': ['3', '8'],
    '3': ['4', '5', '6', '7', '8'],
    '4': ['5'],
    '5': ['6'],
    '6': ['7', '11'],
    '7': ['8', '9', '10', '11'],
    '8': ['9'],
    '9': ['10'],
    '10': ['11']
}

# Breadth-First Search (BFS) algorithm
def bfs(graph, start, end):
    queue = [[start]]
    
    while queue:
        route = queue.pop(0)
        node = route[-1]
        
        if node == end:
            return route
        
        for neighbor in graph.get(node, []):
            new_route = route + [neighbor]
            queue.append(new_route)
    
    print("Route not found. Please try again.")

# Depth-First Search (DFS) algorithm
def dfs(graph, start, end):
    stack = [[start]]
    
    while stack:
        route = stack.pop()
        node = route[-1]
        
        if node == end:
            return route
        
        for neighbor in graph.get(node, []):
            new_route = route + [neighbor]
            stack.append(new_route)
    
    print("Route not found. Please try again.")

# Example usage:
start_node = '1'
end_node = '11'

# Print routes found by BFS and DFS algorithms
print("BFS route:", bfs(spiders, start_node, end_node))
print("DFS route:", dfs(spiders, start_node, end_node))
