def pattern_count(n):
    # Define the number of elements in the set
    num_elements = 4

    # Initialize the table with zeros
    table = [[0 for _ in range(num_elements)] for _ in range(n + 1)]

    # Populate the table using dynamic programming
    for x in range(n + 1):
        for y in range(num_elements):
            if x == 0:
                table[x][y] = 0
            elif x == 1:
                table[x][y] = 1
            elif x == 2:
                if y == 0:
                    table[x][y] = 3
                elif y in (1, 2):
                    table[x][y] = 2
                elif y == 3:
                    table[x][y] = 3
            else:
                if y == 0:
                    table[x][y] = table[x - 1][y] + table[x - 1][y + 2] + table[x - 1][y + 3]
                elif y == 1:
                    table[x][y] = table[x - 1][y] + table[x - 1][y + 2]
                elif y == 2:
                    table[x][y] = table[x - 1][y] + table[x - 1][y - 2]
                elif y == 3:
                    table[x][y] = table[x - 1][y] + table[x - 1][y - 2] + table[x - 1][y - 3]

    # Return the count for n
    return table[n]

# Example usage
n = 6
print(pattern_count(n))
