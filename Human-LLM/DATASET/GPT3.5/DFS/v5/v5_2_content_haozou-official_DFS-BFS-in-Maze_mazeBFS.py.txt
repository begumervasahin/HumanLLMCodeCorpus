from myqueue import Queue
from graphics import GraphWin, Line, Point, Text
from myarray2d import Array2D
from maze import Maze
from mazedraw import MazeDraw

def bfs(maze, start, goal, came_from):
    frontier = Queue()
    frontier.enqueue(start)
    came_from[start] = None

    while not frontier.is_empty():
        current_cell = frontier.dequeue()
        if maze[current_cell[0], current_cell[1]] == Maze.EMPTY:
            if current_cell == goal:
                return current_cell
            else:
                maze[current_cell[0], current_cell[1]] = Maze.OCCUPIED
                for neighbor in maze.get_all_moves(current_cell[0], current_cell[1]):
                    if maze[neighbor[0], neighbor[1]] == Maze.EMPTY:
                        frontier.enqueue(neighbor)
                        came_from[neighbor] = current_cell

    return None

def draw_path_bfs(window, maze, came_from):
    offsets = [(0, 1), (0, -1), (1, 0), (-1, 0)]
    add_coordinates = lambda x, y: (x[0] + y[0], x[1] + y[1])
    current_cell = maze.goal

    while current_cell != maze.start:
        next_cell = came_from[current_cell]
        for offset in offsets:
            next_one = tuple(map(add_coordinates, current_cell, offset))
            if next_one == next_cell:
                line = Line(Point(next_cell[1] + 1 + 0.5, maze.num_rows() - next_cell[0] + 1 - 0.5),
                            Point(current_cell[1] + 1 + 0.5, maze.num_rows() - current_cell[0] + 1 - 0.5))
                line.setOutline('white')
                line.setArrow('last')
                line.draw(window)
        current_cell = next_cell

def main_bfs():
    window = GraphWin('BFS for Maze', 600, 600, autoflush=False)
    maze = Maze(20, 20)
    maze_draw = MazeDraw(window, maze)
    maze_draw.draw()
    came_from = {}
    found_goal = bfs(maze, maze.start, maze.goal, came_from)

    print("Found goal:", found_goal)
    text = Text(Point(11, 0.5), 'Goal found!' if found_goal == maze.goal else 'Goal not found!')
    text.draw(window)

    if found_goal == maze.goal:
        draw_path_bfs(window, maze, came_from)

    while window.checkKey() != 'Escape':
        pass
    window.close()

if __name__ == '__main__':
    main_bfs()
