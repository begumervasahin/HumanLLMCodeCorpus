from collections import defaultdict, deque

class Graph:
    def __init__(self):
        """Initialize an empty graph."""
        self.graph = defaultdict(list)

    def add_edge(self, u, v):
        """Add an edge between vertices u and v."""
        self.graph[u].append(v)

    def bfs(self, start_vertex):
        """Perform Breadth First Traversal starting from the specified source vertex 'start_vertex'."""
        visited = defaultdict(bool)  # Tracks visited vertices
        dist = defaultdict(int)  # Tracks distances from start_vertex
        queue = deque([start_vertex])  # Queue for BFS, initialized with start_vertex
        visited[start_vertex] = True  # Mark start_vertex as visited
        dist[start_vertex] = 0  # Set distance of start_vertex to 0

        while queue:
            current_vertex = queue.popleft()  # Dequeue a vertex
            print(current_vertex, end=" ")  # Print the visited vertex

            # Visit all neighbors of the current_vertex
            for neighbor in self.graph[current_vertex]:
                if not visited[neighbor]:
                    queue.append(neighbor)  # Enqueue unvisited neighbors
                    visited[neighbor] = True  # Mark neighbor as visited
                    dist[neighbor] = dist[current_vertex] + 1  # Update distance

if __name__ == "__main__":
    # Create a new graph
    g = Graph()

    # Add edges to the graph
    edges = [(0, 1), (1, 2), (2, 4), (2, 5), (4, 6), (5, 10), (5, 7)]
    for edge in edges:
        g.add_edge(*edge)

    # Perform BFS starting from vertex 0
    print(" " * 10, "Breadth First Traversal starting from vertex 0: ")
    g.bfs(0)
