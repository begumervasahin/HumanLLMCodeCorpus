# Define the dimensions of the Pascal's triangle
num_rows = 1000
num_cols = 100

# Initialize a 2D list to store Pascal's triangle values
pascals_triangle = [[None] * (num_cols + 1) for _ in range(num_rows + 1)]

# Calculate Pascal's triangle values
for row in range(1, num_rows + 1):
    for col in range(1, min(row, num_cols) + 1):
        if col == 1:
            # Set the value in the first column to the row number
            pascals_triangle[row][col] = row
        else:
            # Calculate values using the Pascal's triangle formula
            pascals_triangle[row][col] = pascals_triangle[row - 1][col] + pascals_triangle[row - 1][col - 1]

# Print the desired value from Pascal's triangle
target_value = pascals_triangle[num_rows][num_cols]
print(f"The value at row {num_rows} and column {num_cols} in Pascal's triangle is: {target_value}")
