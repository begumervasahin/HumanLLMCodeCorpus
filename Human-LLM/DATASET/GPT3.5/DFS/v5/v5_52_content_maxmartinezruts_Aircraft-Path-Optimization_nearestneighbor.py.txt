# Importing necessary libraries
from sklearn.neighbors import NearestNeighbors
import numpy as np

# Define the dataset
data_points = np.array([[-1, -1], [-2, -1], [-3, -2], [1, 1], [2, 1], [3, 2]])

# Initialize the NearestNeighbors model
nbrs = NearestNeighbors(n_neighbors=2, algorithm='ball_tree').fit(data_points)

# Print the model
print("Nearest Neighbors Model:\n", nbrs)

# Find nearest neighbors for each data point
distances, indices = nbrs.kneighbors(data_points)

# Print distances and indices of nearest neighbors
print("\nDistances to Nearest Neighbors:\n", distances)
print("\nIndices of Nearest Neighbors:\n", indices)
