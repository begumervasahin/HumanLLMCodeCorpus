from Node import Node
from UninformedSearchAlgorithm import UninformedSearchAlgorithm

class DFS(UninformedSearchAlgorithm):
    def __init__(self, node_list, target_nodes):
        """
        Initialize the DFS algorithm.

        Args:
        - node_list (list): List of nodes in the graph.
        - target_nodes (list): List of nodes to be searched for.
        """
        super().__init__(node_list, target_nodes)

    def enqueue(self, node):
        """
        Add a node to the front of the queue if it's not already in the queue.

        Args:
        - node (Node): Node to be added to the queue.
        """
        if node.name not in [n.name for n in self.queue]:
            self.queue.insert(0, node)

    def search(self):
        """
        Perform depth-first search.

        Prints the current iteration, the node being read, and the updated queue at each step.
        """
        try:
            iteration_count = 0
            while True:
                self.validate_queue_length()
                iteration_count += 1
                print("\nIteration:", iteration_count)
                current_node = self.dequeue()
                print("Reading node:", current_node.name)
                self.check_searched(current_node.name)
                self.add_children_to_queue(current_node)
                print("Queue:", [n.name for n in self.queue])
        except Exception as error:
            print(error)
