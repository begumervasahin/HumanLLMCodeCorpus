# Import the base class Searcher from the module Searcher
from Searcher import Searcher

# Define a subclass BFS inheriting from the Searcher class
class BFS(Searcher):
    # Class variable to store the name of the BFS algorithm
    name = '2.A_bfs.txt'

    def merge_list(self, list1, list2):
        """
        Merge two lists.
        
        Args:
            list1 (list): The first list.
            list2 (list): The second list.
        
        Returns:
            list: The merged list.
        """
        # Concatenate the two input lists
        merged_list = list1 + list2
        return merged_list
