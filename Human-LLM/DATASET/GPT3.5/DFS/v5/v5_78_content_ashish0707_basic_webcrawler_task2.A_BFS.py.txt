from Webcrawler import Crawler
from bfs import BFS

def initialize_crawler(url):
    """
    Initialize the web crawler with the specified URL and search algorithm.

    Args:
        url (str): The URL to start crawling.
    Returns:
        Crawler: The initialized web crawler instance.
    """
    # Create an instance of the BFS search algorithm
    searcher = BFS()

    # Create a web crawler instance with the specified URL and search algorithm
    crawler = Crawler(url, searcher)
    return crawler

def configure_crawler(crawler):
    """
    Configure the web crawler with mandatory and disallowed lists, and set focused text.

    Args:
        crawler (Crawler): The web crawler instance to configure.
    """
    # Add '/wiki/' to the mandatory list of URLs to visit during crawling
    crawler.addToMandatoryList('/wiki/')

    # Add ':' to the disallowed list of characters in URLs
    crawler.addToDisallowedList(':')

    # Set the focused text to 'solar', indicating that the crawler should search for this text on web pages
    crawler.setFocusedText('solar')

def start_crawling(crawler):
    """
    Start the crawling process.

    Args:
        crawler (Crawler): The web crawler instance to start.
    """
    crawler.start()

if __name__ == "__main__":
    # Define the URL to start crawling
    url = "https://en.wikipedia.org/wiki/Sustainable_energy"

    # Initialize the web crawler
    my_crawler = initialize_crawler(url)

    # Configure the web crawler
    configure_crawler(my_crawler)

    # Start the crawling process
    start_crawling(my_crawler)
