import os
import shutil
import tensorflow as tf
import numpy as np

def check_and_create_directory(dir_path, overwrite=False):
    """
    Check if a directory exists and create it if it doesn't.

    Args:
    - dir_path: Path of the directory to be checked/created.
    - overwrite: Boolean indicating whether to overwrite the directory if it exists (default: False).
    """
    if not os.path.exists(dir_path):
        os.mkdir(dir_path)
    elif overwrite:
        shutil.rmtree(dir_path)
        os.mkdir(dir_path)

def create_directory_tree(dir_path_list):
    """
    Create directories recursively based on a list of directory names.

    Args:
    - dir_path_list: List of directory names representing the directory tree structure.
    """
    for i in range(len(dir_path_list)):
        check_and_create_directory(os.path.join(*dir_path_list[:i + 1]))

def remove_directory(dir_path):
    """
    Remove a directory and its contents.

    Args:
    - dir_path: Path of the directory to be removed.
    """
    shutil.rmtree(dir_path, ignore_errors=True)

def relu_init(shape, dtype=tf.float32):
    """
    Initialize weights using the ReLU activation function.

    Args:
    - shape: Shape of the weight tensor.
    - dtype: Data type of the weight tensor (default: tf.float32).

    Returns:
    - initial: Initialized weight tensor.
    """
    init_range = np.sqrt(2.0 / shape[1])
    initial = tf.random_normal(shape, dtype=dtype) * init_range
    return initial

def zeros(shape, dtype=tf.float32):
    """
    Initialize weights with zeros.

    Args:
    - shape: Shape of the weight tensor.
    - dtype: Data type of the weight tensor (default: tf.float32).

    Returns:
    - initial: Initialized weight tensor.
    """
    initial = tf.zeros(shape, dtype=dtype)
    return initial

def const(shape, value=0.1, dtype=tf.float32):
    """
    Initialize weights with a constant value.

    Args:
    - shape: Shape of the weight tensor.
    - value: Constant value for initialization (default: 0.1).
    - dtype: Data type of the weight tensor (default: tf.float32).

    Returns:
    - initial: Initialized weight tensor.
    """
    initial = tf.constant(value, shape=shape, dtype=dtype)
    return initial

def tanh_init(shape, dtype=tf.float32):
    """
    Initialize weights using the tanh activation function.

    Args:
    - shape: Shape of the weight tensor.
    - dtype: Data type of the weight tensor (default: tf.float32).

    Returns:
    - initial: Initialized weight tensor.
    """
    init_range = np.sqrt(6.0 / (shape[0] + shape[1]))
    initial = tf.random_uniform(shape, minval=-init_range, maxval=init_range, dtype=tf.float32)
    return initial
