class TreeNode:
    def __init__(self, value):
        self.value = value
        self.left = None
        self.right = None

class TreeDLS:
    def __init__(self, root):
        self.root = root

    def search(self, target, level_threshold):
        found = False
        stack = [[self.root, 0]]  # Initialize stack with root and level 0
        
        # Perform depth-limited search
        while stack:
            node, current_level = stack.pop()
            
            # Check if the current level is within the threshold
            if current_level < level_threshold:
                if node.value == target:
                    print("Target found.")
                    found = True
                    break
                
                # Add left child to the stack if exists and level is within the threshold
                if node.left and current_level < level_threshold - 1:
                    stack.append([node.left, current_level + 1])
                
                # Add right child to the stack if exists and level is within the threshold
                if node.right and current_level < level_threshold - 1:
                    stack.append([node.right, current_level + 1])
            else:
                break
        
        # Print message if target is not found
        if not found:
            print("Target not found within the specified level threshold.")

# Define the nodes of the binary tree
A = TreeNode(1)
B = TreeNode(2)
C = TreeNode(3)
D = TreeNode(4)
E = TreeNode(5)
F = TreeNode(6)
G = TreeNode(7)

# Construct the binary tree structure
B.left = D
B.right = E
C.left = F
C.right = G
A.left = B
A.right = C

# Create an instance of TreeDLS
tree = TreeDLS(A)

# Search for the target within the specified level threshold
tree.search(4, 2)
