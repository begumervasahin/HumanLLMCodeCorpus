def bfs(graph, start, finish):
    """
    Perform Breadth-First Search (BFS) on a graph to find a path from start to finish.

    Args:
    - graph: The graph represented as a dictionary of nodes and their neighbors.
    - start: The starting node for the search.
    - finish: The target node to reach.

    Returns:
    - The path from start to finish if found, or None if no path exists.
    """
    # Initialize the queue with the start node
    queue = [[start]]
    # Keep track of visited nodes
    visited = set()

    while queue:
        # Dequeue the first path from the queue
        path = queue.pop(0)
        # Get the last node in the current path
        state = path[-1]

        if state == finish:
            # If the finish node is reached, return the path
            return path

        if state not in visited:
            for neighbor in graph.get(state, []):
                # Create a new path by extending the current path
                new_path = list(path)
                new_path.append(neighbor)
                # Enqueue the new path
                queue.append(new_path)

            # Mark the current node as visited
            visited.add(state)

    # If the finish node is not reached, print a message and return None
    print("NOT FOUND 404")
    return None
