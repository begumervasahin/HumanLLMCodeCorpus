from __future__ import print_function
import h5py
import numpy as np

class DataSet:
    def __init__(self, images, labels, fake_data=False):
        self._num_examples = 10000 if fake_data else len(images)
        assert len(images) == len(labels), "Number of images and labels must be the same"
        self._images = images
        self._labels = labels
        self._epochs_completed = 0
        self._index_in_epoch = 0

    @property
    def images(self):
        return self._images

    @property
    def labels(self):
        return self._labels

    @property
    def num_examples(self):
        return self._num_examples

    @property
    def epochs_completed(self):
        return self._epochs_completed

    def next_batch(self, batch_size):
        start = self._index_in_epoch
        self._index_in_epoch += batch_size
        if self._index_in_epoch > self._num_examples:
            self._epochs_completed += 1
            perm = np.random.permutation(self._num_examples)
            self._images = self._images[perm]
            self._labels = self._labels[perm]
            start = 0
            self._index_in_epoch = batch_size
            assert batch_size <= self._num_examples
        end = self._index_in_epoch
        return self._images[start:end], self._labels[start:end]

class SemiDataSet:
    def __init__(self, images, labels, n_labeled):
        self.n_labeled = n_labeled
        self.unlabeled_ds = DataSet(images, labels)
        self.num_examples = self.unlabeled_ds.num_examples
        indices = np.arange(self.num_examples)
        np.random.shuffle(indices)
        images = images[indices]
        labels = labels[indices]
        unique_labels = np.unique(labels)
        labeled_indices = []
        for label in unique_labels:
            label_indices = np.where(labels == label)[0][:n_labeled // len(unique_labels)]
            labeled_indices.extend(label_indices)
        labeled_indices = np.array(labeled_indices)
        labeled_images = images[labeled_indices]
        labeled_labels = labels[labeled_indices]
        self.labeled_ds = DataSet(labeled_images, labeled_labels)

    def next_batch(self, batch_size):
        unlabeled_images, _ = self.unlabeled_ds.next_batch(batch_size)
        labeled_images, labeled_labels = self.labeled_ds.next_batch(min(batch_size, self.n_labeled))
        images = np.vstack([labeled_images, unlabeled_images])
        labels = np.hstack([labeled_labels, np.zeros(len(unlabeled_images), dtype=labeled_labels.dtype)])
        return images, labels

def read_data_sets(file_name='I_feature.h5', n_labeled=100):
    class DataSets:
        pass
    data_sets = DataSets()
    with h5py.File(file_name, 'r') as f:
        train_images = f['data'][:]
        train_labels = np.argmax(f['label'][:], axis=1)
    train_images = (train_images - train_images.min()) / (train_images.max() - train_images.min())
    indices = np.random.permutation(len(train_images))
    train_images, train_labels = train_images[indices], train_labels[indices]
    data_sets.train = SemiDataSet(train_images, train_labels, n_labeled)
    data_sets.test = DataSet(train_images, train_labels)
    return data_sets
