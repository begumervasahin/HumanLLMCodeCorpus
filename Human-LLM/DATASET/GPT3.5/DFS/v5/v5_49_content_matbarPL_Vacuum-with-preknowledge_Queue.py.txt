class Queue:
    """A class representing a queue."""

    def __init__(self):
        """Initialize an empty queue."""
        self.items = []

    def is_empty(self):
        """Check if the queue is empty."""
        return len(self.items) == 0

    def enqueue(self, item):
        """Add an item to the end of the queue."""
        self.items.insert(0, item)

    def dequeue(self):
        """Remove and return the first item from the queue."""
        return self.items.pop()

    def first(self):
        """Return the first item in the queue."""
        return self.items[-1]

    def size(self):
        """Return the number of items in the queue."""
        return len(self.items)

    def enqueue_items(self, items):
        """Add multiple items to the queue."""
        for item in items:
            self.enqueue(item)


class BQueue:
    """A class representing a bounded queue."""

    def __init__(self):
        """Initialize an empty bounded queue."""
        self.items = []

    def is_empty(self):
        """Check if the bounded queue is empty."""
        return len(self.items) == 0

    def enqueue(self, item):
        """Add an item to the end of the bounded queue."""
        self.items.append(item)

    def dequeue(self):
        """Remove and return the first item from the bounded queue."""
        return self.items.pop(0)

    def size(self):
        """Return the number of items in the bounded queue."""
        return len(self.items)
