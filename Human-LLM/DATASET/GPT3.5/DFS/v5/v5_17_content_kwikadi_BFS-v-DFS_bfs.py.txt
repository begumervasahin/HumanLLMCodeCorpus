from collections import defaultdict

class Graph:
    def __init__(self, num_vertices):
        # Initialize the graph with adjacency lists and the total number of vertices
        self.adjacency_lists = defaultdict(list)
        self.num_vertices = num_vertices

    def add_edge(self, u, v):
        # Add an edge from vertex u to vertex v in the graph
        self.adjacency_lists[u].append(v)

    def bfs(self, start_vertex, goal_vertex):
        # Perform Breadth-First Search (BFS) traversal in the graph from start to goal
        visited = [False] * self.num_vertices  # Track visited vertices
        queue = [start_vertex]  # Initialize a queue for BFS with the start vertex
        visited[start_vertex] = True  # Mark the start vertex as visited
        while queue:
            current_vertex = queue.pop(0)  # Get the next vertex from the queue
            print(current_vertex, end=" ")  # Print the current vertex
            if current_vertex == goal_vertex:
                print("Goal found.")  # Print if the goal vertex is found
                return
            for neighbor in self.adjacency_lists[current_vertex]:
                # Traverse all neighbors of the current vertex
                if not visited[neighbor]:
                    queue.append(neighbor)  # Add unvisited neighbors to the queue
                    visited[neighbor] = True  # Mark neighbors as visited
        print("Search failed.")  # Print if the goal vertex is not found

def main():
    # Read input from the file and perform BFS
    with open('input.txt', 'r') as input_file:
        num_vertices = int(input_file.readline())  # Total number of vertices in the graph
        start_vertex = int(input_file.readline())  # Start vertex for BFS
        goal_vertex = int(input_file.readline())  # Goal vertex for BFS
        graph = Graph(num_vertices)  # Create an instance of the Graph class
        for line in input_file:
            u, v = map(int, line.split())  # Read edges from input file
            graph.add_edge(u, v)  # Add edges to the graph
    graph.bfs(start_vertex, goal_vertex)  # Perform BFS from start to goal

if __name__ == "__main__":
    main()
