import networkx as nx
from helper import (
    mark_trail_between_cities_and_print_report,
    mark_starting_point,
    show_return_to_city,
    execute_instructions_related_to_visited_city,
    show_verify_neighbours,
    begin_note,
    end_note,
)

def dfs(G, cities, start_vertex):
    """
    Perform Depth-First Search (DFS) traversal on a directed graph starting from a specified vertex.

    Args:
    - G (networkx.DiGraph): Directed graph.
    - cities (dict): Mapping of vertex IDs to city names.
    - start_vertex (int): Starting vertex for DFS traversal.
    """
    def dfs_node(current_vertex, parent_vertex, visited):
        """
        Recursive function to perform DFS traversal starting from a given vertex.

        Args:
        - current_vertex (int): Current vertex being processed.
        - parent_vertex (int): Parent vertex of the current vertex.
        - visited (list): List to track visited vertices.
        """
        visited[current_vertex] = True
        
        if parent_vertex != current_vertex:
            mark_trail_between_cities_and_print_report(G, current_vertex, parent_vertex, cities[current_vertex], G.pos)
        else:
            mark_starting_point(G, current_vertex)
        
        for neighbor_vertex in G.neighbors(current_vertex):
            if not visited[neighbor_vertex]:
                dfs_node(neighbor_vertex, current_vertex, visited)
                show_return_to_city(G, current_vertex, cities[current_vertex])
            else:
                execute_instructions_related_to_visited_city(G, current_vertex, parent_vertex, neighbor_vertex, cities[neighbor_vertex], G.pos)
        
        show_verify_neighbours(G, current_vertex, cities[current_vertex])

    visited = [False] * len(cities)
    begin_note(G, cities[start_vertex])
    dfs_node(start_vertex, start_vertex, visited)
    end_note(G)
