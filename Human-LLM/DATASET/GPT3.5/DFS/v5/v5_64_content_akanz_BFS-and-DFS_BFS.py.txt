from collections import deque

# Define the graph
graph = {
    'A': ['B', 'E', 'C'],
    'B': ['A', 'D', 'E'],
    'C': ['A', 'F', 'G'],
    'D': ['B', 'E'],
    'E': ['A', 'B', 'D'],
    'F': ['C'],
    'G': ['C']
}

def bfs_search(graph, start_node):
    """Perform breadth-first search traversal of the graph."""
    visited = []
    queue = deque([start_node])
    
    while queue:
        current_node = queue.popleft()
        if current_node not in visited:
            visited.append(current_node)
            neighbors = graph[current_node]
            queue.extend(neighbors)
    
    return visited

def bfs_shortest_path(graph, start_node, end_node):
    """Find the shortest path from start_node to end_node using breadth-first search."""
    queue = deque([[start_node]])
    
    if start_node == end_node:
        return [start_node]
    
    visited = []
    
    while queue:
        path = queue.popleft()
        node = path[-1]
        if node not in visited:
            neighbors = graph[node]
            for neighbor in neighbors:
                new_path = list(path)
                new_path.append(neighbor)
                queue.append(new_path)
                if neighbor == end_node:
                    return new_path
            visited.append(node)
    
    return "Sorry, but a connecting path doesn't exist."

# Example usage:
start_node = 'A'
end_node = 'G'
print("BFS traversal from node", start_node, ":", bfs_search(graph, start_node))
print("Shortest path from", start_node, "to", end_node, ":", bfs_shortest_path(graph, start_node, end_node))
