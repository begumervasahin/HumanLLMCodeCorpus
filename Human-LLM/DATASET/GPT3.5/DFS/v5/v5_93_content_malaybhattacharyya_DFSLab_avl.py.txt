import sys

class AVLNode:
    """A single node in an AVL tree."""
    def __init__(self, data, left=None, right=None, parent=None, height=0):
        self.data = data
        self.left = left
        self.right = right
        self.parent = parent
        self.height = height

class AVLTree:
    """AVL Tree implementation."""
    def __init__(self):
        self.root = None

    def height(self, node):
        if node is None:
            return -1
        else:
            return node.height

    def rotate_LL(self, parent, start_node):
        k2 = start_node
        k1 = k2.left
        Z = k2.right
        X = k1.left
        Y = k1.right

        k2.left = Y
        k1.right = k2
        k1.parent = parent
        k2.parent = k1

        if Y is not None:
            Y.parent = k2

        k2.height = 1 + max(self.height(Y), self.height(Z))
        k1.height = 1 + max(self.height(X), k2.height)

        return k1
