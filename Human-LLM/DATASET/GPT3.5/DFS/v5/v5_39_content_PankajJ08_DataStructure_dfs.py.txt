from collections import defaultdict

class Graph:
    def __init__(self):
        """Initialize an empty graph."""
        self.adjacency_list = defaultdict(list)

    def add_edge(self, start_vertex, end_vertex):
        """Add an edge between start_vertex and end_vertex."""
        self.adjacency_list[start_vertex].append(end_vertex)

    def dfs_visit(self, vertex, visited):
        """Recursively visit vertices in depth-first order."""
        visited[vertex] = True
        print(vertex, end=" ")  # Print the visited vertex
        for neighbor in self.adjacency_list[vertex]:
            if not visited[neighbor]:
                self.dfs_visit(neighbor, visited)

    def dfs(self):
        """Perform Depth-First Search (DFS)."""
        visited = defaultdict(bool)
        for vertex in self.adjacency_list:
            if not visited[vertex]:
                self.dfs_visit(vertex, visited)

if __name__ == "__main__":
    # Create a new graph
    graph = Graph()

    # Add edges to the graph
    edges = [(0, 1), (1, 2), (2, 4), (2, 5), (4, 6), (5, 10), (11, 7)]
    for edge in edges:
        graph.add_edge(*edge)

    # Perform DFS starting from vertex 0
    print(" " * 10, "Depth-First Traversal starting from vertex 0:")
    graph.dfs()
