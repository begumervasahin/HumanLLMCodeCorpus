from bs4 import BeautifulSoup, Comment
import re
import urllib.request

def scrape_baseball_reference(resource, table_ids=None, server_url="http://www.baseball-reference.com/"):
    """
    Scrape data from Baseball Reference.

    Args:
        resource (str): The path to the resource to scrape.
        table_ids (list): A list of table IDs to scrape data from.
        server_url (str): The base URL of the website.

    Returns:
        list: A list containing dictionaries of scraped data and their corresponding table IDs.
    """
    data = {}
    all_data = []

    print("Scraping URL:", server_url + resource)

    with urllib.request.urlopen(server_url + resource) as response:
        html_content = response.read()

    soup = BeautifulSoup(html_content, 'html.parser')
    comments = soup.find_all(text=lambda text: isinstance(text, Comment))

    for comment in comments:
        for table_id in table_ids:
            if f'table class="row_summable sortable stats_table" id="{table_id}"' in comment:
                data[table_id] = []
                rows = comment.string.split("</tr>")
                table_found = False

                for row in rows:
                    stats = []
                    if f'table class="row_summable sortable stats_table" id="{table_id}"' in row:
                        headers = re.findall(r'<th.*>([^<]*)</th>', row)
                        table_found = True
                        continue

                    if table_found:
                        entries = row.split(">")
                        for entry in entries:
                            if "</th" in entry and "</thead" not in entry:
                                stat = re.findall(r'(.*)<', entry)
                                stats.extend(stat)

                            if "</td" in entry:
                                stat = re.findall(r'(.*)<', entry)
                                stats.extend(stat)

                        if stats:
                            data[table_id].append(dict(zip(headers, stats)))

                all_data.append([data, table_id])

    return all_data

if __name__ == '__main__':
    # Example usage:
    resource = "path_to_resource"  # Replace with the actual resource path
    table_ids = ["table_id1", "table_id2"]  # Replace with the actual table IDs
    server_url = "http://www.baseball-reference.com/"  # Replace with the actual server URL
    scraped_data = scrape_baseball_reference(resource, table_ids, server_url)
    print(scraped_data)
