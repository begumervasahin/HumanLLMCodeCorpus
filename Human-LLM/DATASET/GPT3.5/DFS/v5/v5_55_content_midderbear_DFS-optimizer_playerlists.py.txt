import csv
import string

def get_player_list_from_csv():
    """
    Prompt the user to enter the filename of a CSV file and read player information from it.

    Returns:
        list: A list containing dictionaries of player information.
    """
    filename = input('Enter filename (.csv): ') + '.csv'
    
    with open(filename, 'r', newline='') as file:
        player_list = []
        reader = csv.reader(file)
        
        # Skip the header row
        next(reader)
        
        for row in reader:
            player = {
                'FIRST_NAME': row[2],
                'LAST_NAME': row[4],
                'POSITION': row[1],
                'PRICE': row[6],
                'TEAM': row[9],
                'OPPONENT': row[10],
                'INJURY_STATUS': row[11]
            }
            player_list.append(player)
    
    # Remove the header row from the list
    player_list.pop(0)
    
    return player_list

def trim_player_names(player_list):
    """
    Remove punctuation characters from the first and last names of each player in the player list.

    Args:
        player_list (list): A list containing dictionaries of player information.

    Returns:
        list: A list containing dictionaries of player information with trimmed names.
    """
    for player in player_list:
        player['FIRST_NAME'] = player['FIRST_NAME'].translate(str.maketrans('', '', string.punctuation))
        player['LAST_NAME'] = player['LAST_NAME'].translate(str.maketrans('', '', string.punctuation))
    
    return player_list

# Example usage:
if __name__ == "__main__":
    players = get_player_list_from_csv()
    trimmed_players = trim_player_names(players)
    for player in trimmed_players:
        print(player)
