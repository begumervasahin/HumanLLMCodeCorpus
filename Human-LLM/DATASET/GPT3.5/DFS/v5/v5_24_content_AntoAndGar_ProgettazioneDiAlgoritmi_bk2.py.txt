def generate_binary_strings(length, ones_count, index=0, current=[], current_ones=0):
    """
    Generate binary strings of a specified length with a specific count of '1's.

    Args:
    - length: Length of the binary strings.
    - ones_count: Number of '1's in the binary strings.
    - index: Current index during recursion (default: 0).
    - current: Current partial binary string being constructed (default: []).
    - current_ones: Count of '1's added so far (default: 0).

    This function uses backtracking to generate binary strings with a specified number of '1's.
    """
    if index == length:
        # Base case: if the string length reaches the specified length, print the current string
        print(''.join(current))
    else:
        # Try adding '0' to the current string
        current.append('0')
        generate_binary_strings(length, ones_count, index + 1, current, current_ones)
        current.pop()  # Backtrack by removing the last character

        # If we can add '1' (total '1's added < specified count), try adding '1' to the current string
        if current_ones < ones_count:
            current.append('1')
            generate_binary_strings(length, ones_count, index + 1, current, current_ones + 1)
            current.pop()  # Backtrack by removing the last character

# Example: Generate binary strings of length 4 with exactly 1 '1'
generate_binary_strings(4, 1)
