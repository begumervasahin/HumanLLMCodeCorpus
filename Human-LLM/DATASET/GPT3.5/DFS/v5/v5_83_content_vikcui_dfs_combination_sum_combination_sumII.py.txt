"""
Given a collection of candidate numbers (candidates) and a target number (target),
find all unique combinations in candidates where the candidate numbers sum to target.
Each number in candidates may only be used once in the combination.
Note:
All numbers (including target) will be positive integers.
The solution set must not contain duplicate combinations.
This implementation follows a Depth-First Search (DFS) approach.
"""

def find_combinations_unique(candidates, target):
    # Sort the input list of candidates in ascending order
    candidates.sort()
    # Initialize an empty list to store the result combinations
    result_combinations = []
    # Start from the first index of the input list
    start_index = 0
    # Initialize an empty list to store the current combination sequence
    current_combination = []
    # Call the recursive function to find combinations
    find_combinations(candidates, target, start_index, current_combination, result_combinations)
    return result_combinations

def find_combinations(candidates, target, start_index, current_combination, result_combinations):
    """
    Recursively find combinations that sum up to the target.
    
    Args:
        candidates (list): The list of candidate numbers.
        target (int): The target sum.
        start_index (int): The index to start searching for combinations.
        current_combination (list): The current combination sequence being built.
        result_combinations (list): The list to store the valid combinations.
    """
    # Iterate through the candidate numbers starting from the specified index
    for i in range(start_index, len(candidates)):
        # Check if it's the first element at this index or if the current element is different from the previous one
        if i == start_index or candidates[i] != candidates[i - 1]:
            num = candidates[i]
            # If the current number exceeds the target, terminate the loop
            if num > target:
                return
            # Add the current number to the current sequence
            current_combination.append(num)
            # If the current number is less than the target, recursively search for combinations
            if num < target:
                find_combinations(candidates, target - num, i + 1, current_combination, result_combinations)
            # If the current number equals the target, add the current sequence to the result list
            else:
                result_combinations.append(list(current_combination))
            # Backtrack by removing the last added number from the current sequence
            current_combination.pop()

# Example usage:
candidates = [10, 1, 2, 7, 6, 1, 5]
target = 8
print(find_combinations_unique(candidates, target))  # Output: [[1, 1, 6], [1, 2, 5], [1, 7], [2, 6]]
