def bfs_paths(graph, start, goal):
    """
    Find all possible paths from the start node to the goal node using Breadth-First Search.
    
    Args:
        graph (dict): The graph represented as an adjacency list.
        start (str): The starting node.
        goal (str): The target node.
        
    Yields:
        list: A list representing a path from the start node to the goal node.
    """
    queue = [(start, [start])]
    while queue:
        node, path = queue.pop(0)
        for next_node in graph[node] - set(path):
            if next_node == goal:
                yield path + [next_node]
            else:
                queue.append((next_node, path + [next_node]))

def shortest_bfs(graph, start, goal):
    """
    Find the shortest path from the start node to the goal node using Breadth-First Search.
    
    Args:
        graph (dict): The graph represented as an adjacency list.
        start (str): The starting node.
        goal (str): The target node.
        
    Returns:
        list or None: The shortest path from the start node to the goal node, or None if no path exists.
    """
    try:
        return next(bfs_paths(graph, start, goal))
    except StopIteration:
        return None

def dfs_paths(graph, start, goal, path=None):
    """
    Find all possible paths from the start node to the goal node using Depth-First Search.
    
    Args:
        graph (dict): The graph represented as an adjacency list.
        start (str): The starting node.
        goal (str): The target node.
        path (list): The current path being explored.
        
    Yields:
        list: A list representing a path from the start node to the goal node.
    """
    if path is None:
        path = [start]
    if start == goal:
        yield path
    for next_node in graph[start] - set(path):
        yield from dfs_paths(graph, next_node, goal, path + [next_node])

def shortest_dfs(graph, start, goal, path=None):
    """
    Find the shortest path from the start node to the goal node using Depth-First Search.
    
    Args:
        graph (dict): The graph represented as an adjacency list.
        start (str): The starting node.
        goal (str): The target node.
        path (list): The current path being explored.
        
    Returns:
        list or None: The shortest path from the start node to the goal node, or None if no path exists.
    """
    try:
        return next(dfs_paths(graph, start, goal, path))
    except StopIteration:
        return None

# Example graph
graph = {
    'A': {'B', 'C'},
    'B': {'A', 'D', 'E'},
    'C': {'A', 'F'},
    'D': {'B'},
    'E': {'B', 'F'},
    'F': {'C', 'E'}
}

# Example usage
start_node = 'A'
goal_node = 'F'

print("Shortest path using BFS:", shortest_bfs(graph, start_node, goal_node))
print("Shortest path using DFS:", shortest_dfs(graph, start_node, goal_node))
