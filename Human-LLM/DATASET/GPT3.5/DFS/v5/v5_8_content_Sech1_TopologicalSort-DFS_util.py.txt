import itertools
import sys
import time
from graph import Graph

class Util:
    # Static variable to control the spinning animation
    go = True

    @classmethod
    def spinning_cursor(cls):
        """
        Display a spinning cursor animation in the console.
        """
        spinner = itertools.cycle(['- Doing Algo Stuff. ', '/ Doing Algo Stuff.. ',
                                   '| Doing Algo Stuff... ', '\\ Doing Algo Stuff.... '])
        while Util.go:
            sys.stdout.write(next(spinner))
            sys.stdout.flush()
            time.sleep(0.25)
            sys.stdout.write('\r')

    @classmethod
    def read_in_graph(cls, file_path):
        """
        Read graph data from a file and create a Graph object.

        Args:
            file_path (str): The path to the file containing graph data.

        Returns:
            Graph: A Graph object representing the graph read from the file.
        """
        # Read the lines from the file
        with open(file_path) as file:
            lines = file.readlines()

        # Initialize variables
        vert_count = len(lines)
        graph = Graph(vert_count)

        # Process each line in the file
        for i, line in enumerate(lines):
            lines[i] = line.strip('\n')

        # Parse each line to extract vertex and edges
        for line in lines:
            vert, edges = line.split(':')
            edges = edges.split(' ')
            for edge in edges:
                graph.graph[vert].append(edge)

        # Remove any empty entries in the graph
        for i in range(1, vert_count + 1):
            graph.graph[str(i)] = list(filter(None, graph.graph[str(i)]))

        return graph
