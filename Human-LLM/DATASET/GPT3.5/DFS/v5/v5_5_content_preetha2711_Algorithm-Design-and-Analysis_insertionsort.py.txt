def insertion_sort(array):
    # Iterate through the array, starting from the second element
    for i in range(1, len(array)):
        current_element = array[i]
        hole = i
        
        # Find the correct position for the current element
        while hole > 0 and array[hole - 1] > current_element:
            array[hole] = array[hole - 1]
            hole -= 1
        
        # Insert the current element into its correct position
        array[hole] = current_element

# Define an array to be sorted
unsorted_array = [12, 11, 4, 3, 1, 1, 6, 7]

# Call the insertion sort function
insertion_sort(unsorted_array)

# Print the sorted array
print("Sorted Array:", unsorted_array)
