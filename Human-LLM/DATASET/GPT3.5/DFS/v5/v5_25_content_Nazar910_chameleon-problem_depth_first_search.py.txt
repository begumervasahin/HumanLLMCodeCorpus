from collections import deque

def depth_first_search(state):
    # Initialize data structures
    previous_states = deque()
    visited_states = set()
    full_path = []
    step_count = 0
    
    while state is not None and not state.has_only_one_color():
        # Add current state to full path
        full_path.append(str(state))
        
        # Check possible transitions and update state accordingly
        if state.has_red_ones() and state.has_green_ones() and str(state.red_met_green()) not in visited_states:
            new_state = state.red_met_green()
        elif state.has_blue_ones() and state.has_red_ones() and str(state.blue_met_red()) not in visited_states:
            new_state = state.blue_met_red()
        elif state.has_blue_ones() and state.has_green_ones() and str(state.green_met_blue()) not in visited_states:
            new_state = state.green_met_blue()
        else:
            # Backtrack if no valid transitions are available
            state = previous_states.pop()
            print('Backtracking one step')
            continue
        
        # Update state and tracking variables
        previous_states.append(state)
        print('{}: {}'.format(step_count, state))
        step_count += 1
        state = new_state
        visited_states.add(str(new_state))
    
    print('Final state after {} steps: {}'.format(step_count, state))
    
    return {
        'final_state': state,
        'visited_states': visited_states,
        'path': state.get_path(),  # Assuming State has a get_path() method
        'full_path': full_path
    }
