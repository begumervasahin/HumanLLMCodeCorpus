def stringify_command(command, param_list):
    """
    Convert command and parameters to a JSON string.
    
    Args:
        command (int): The command identifier.
        param_list (list): List of parameters associated with the command.
    
    Returns:
        str: JSON representation of the command and parameters.
    """
    command_obj = {'command': command, 'param_list': param_list}
    return json.dumps(command_obj)

def stringify_result(result):
    """
    Convert a result to a JSON string.
    
    Args:
        result: The result to be converted to JSON.
    
    Returns:
        str: JSON representation of the result.
    """
    return json.dumps(result)
