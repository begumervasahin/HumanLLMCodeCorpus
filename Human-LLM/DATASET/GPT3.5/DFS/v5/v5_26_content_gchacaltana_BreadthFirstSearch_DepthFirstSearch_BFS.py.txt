from Node import Node
from UninformedSearchAlgorithm import UninformedSearchAlgorithm

class BFS(UninformedSearchAlgorithm):
    def __init__(self, nodes, searched_nodes):
        super().__init__(nodes, searched_nodes)

    def add_to_queue(self, node):
        if node.name not in [n.name for n in self.queue]:
            self.queue.append(node)

    def search(self):
        try:
            iteration_count = 0
            while True:
                self.validate_queue_length()
                iteration_count += 1
                print("\nIteration:", iteration_count)
                current_node = self.dequeue()
                print("Processing node:", current_node.name)
                self.check_searched(current_node.name)
                self.add_children_to_queue(current_node)
                print("Queue:", [n.name for n in self.queue])
        except Exception as error:
            print("Error:", error)
