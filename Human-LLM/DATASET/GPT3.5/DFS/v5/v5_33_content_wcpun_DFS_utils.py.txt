class Meta:
    """Represents metadata information."""
    def __init__(self, user_id, file_name, file_size=None, iv=None):
        self.user_id = user_id
        self.file_name = file_name
        self.file_size = file_size
        self.iv = iv

    def __str__(self):
        return f"User ID: {self.user_id}\n" \
               f"File Name: {self.file_name}\n" \
               f"File Size: {self.file_size}\n" \
               f"IV: {self.iv}"

class Code:
    """Represents a code with a key."""
    def __init__(self, key):
        self.key = key

    def __str__(self):
        return f"Key: {self.key}"

class Cipher:
    """Represents a cipher with encrypted text."""
    def __init__(self, cipher):
        self.cipher = cipher

    def __str__(self):
        return f"Cipher: {self.cipher}"
