import pygame
import pygame.locals as pl

class Button:
    def __init__(self, window, color, x, y, width, height, text=''):
        self.window = window  # Pygame window surface
        self.color = color  # Button color
        self.x = x  # X-coordinate of the button's top-left corner
        self.y = y  # Y-coordinate of the button's top-left corner
        self.width = width  # Width of the button
        self.height = height  # Height of the button
        self.text = text  # Text displayed on the button

    def draw(self, outline=False):
        """
        Draw the button on the window.

        Args:
            outline (bool): Whether to draw an outline around the button.

        Returns:
            None
        """
        if outline:
            pygame.draw.rect(self.window, (0, 0, 0), (self.x - 2, self.y - 2, self.width + 4, self.height + 4))
        pygame.draw.rect(self.window, self.color, (self.x, self.y, self.width, self.height))
        if self.text:
            font = pygame.font.SysFont('Times New Roman', 20)
            text_surface = font.render(self.text, True, (255, 255, 255))
            text_rect = text_surface.get_rect(center=(self.x + self.width / 2, self.y + self.height / 2))
            self.window.blit(text_surface, text_rect)

    def is_over(self, pos):
        """
        Check if a position (x, y) is within the button's boundaries.

        Args:
            pos (tuple): Position (x, y) to check.

        Returns:
            bool: True if the position is over the button, False otherwise.
        """
        return self.x < pos[0] < self.x + self.width and self.y < pos[1] < self.y + self.height
