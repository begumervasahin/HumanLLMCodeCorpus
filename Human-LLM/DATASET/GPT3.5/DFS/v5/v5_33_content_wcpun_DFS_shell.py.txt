import cmd

class MyCLI(cmd.Cmd):
    """
    Custom command-line interface class.
    """

    def __init__(self):
        """
        Initialize the command-line interface.
        """
        super().__init__()
        self.current_path = ''
        self.prompt = "FYP >>> "

    def do_hi(self, arg):
        """
        Command to print a greeting message.
        """
        print("Hello World.")

def start_command_line_interface():
    """
    Start the command-line interface.
    """
    cli = MyCLI()
    cli.cmdloop()

if __name__ == '__main__':
    start_command_line_interface()
