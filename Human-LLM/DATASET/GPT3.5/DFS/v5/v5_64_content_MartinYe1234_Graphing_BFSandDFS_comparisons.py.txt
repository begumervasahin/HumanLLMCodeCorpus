import matplotlib.pyplot as plt
from random import randint
from time import time

# Initialize lists for storing data
dfs_runtimes = []
bfs_runtimes = []

# Generate x-axis values
num_connections_range = range(1, 2002)
x_axis = list(num_connections_range)

# Set figure size
plt.figure(figsize=(16, 5))

def generate_graph(num_of_nodes, num_of_connections):
    """Generate a random graph."""
    adj_list = {i: [] for i in range(num_of_nodes)}
    adjacent = {i: [] for i in range(num_of_nodes)}
    
    # Generate connections
    for _ in range(num_of_connections):
        x, y = randint(0, num_of_nodes - 1), randint(0, num_of_nodes - 1)
        adj_list[x].append(y)
        adj_list[y].append(x)
        adjacent[x].append(y)
        adjacent[y].append(x)
    
    # Remove self-loops
    for i in range(num_of_nodes):
        adj_list[i] = [node for node in adj_list[i] if node != i]
        adjacent[i] = [node for node in adjacent[i] if node != i]
    
    return adj_list, adjacent

def breadth_first_search(node):
    """Breadth-first search algorithm."""
    queue = [node]
    bfs_visited[node] = True
    while queue:
        for neighbour in adjacent[queue[0]]:
            if not bfs_visited[neighbour]:
                queue.append(neighbour)
                bfs_visited[neighbour] = True
        queue.pop(0)

def run_tests(num_of_nodes, num_of_connections, runs):
    """Run tests for DFS and BFS."""
    dfs_avg_runtime = 0.0
    bfs_avg_runtime = 0.0
    
    for _ in range(num_of_connections):
        dfs_total_runtime = 0.0
        bfs_total_runtime = 0.0
        for _ in range(runs):
            adj, adjacent = generate_graph(num_of_nodes, num_of_connections)
            
            global dfs_visited, bfs_visited
            dfs_visited = [False] * num_of_nodes
            bfs_visited = [False] * num_of_nodes
            
            # Depth-first search
            dfs_start_time = time()
            depth_first_search(0)
            dfs_total_runtime += time() - dfs_start_time
            
            # Breadth-first search
            bfs_start_time = time()
            breadth_first_search(0)
            bfs_total_runtime += time() - bfs_start_time
        
        # Calculate average runtimes
        dfs_avg_runtime = dfs_total_runtime / runs
        bfs_avg_runtime = bfs_total_runtime / runs
        
        dfs_runtimes.append(round(dfs_avg_runtime, 13))
        bfs_runtimes.append(round(bfs_avg_runtime, 13))

run_tests(num_of_nodes=200, num_of_connections=2000, runs=200)

# Plot the results
plt.plot(x_axis, dfs_runtimes, label='DFS', color='blue')
plt.plot(x_axis, bfs_runtimes, label='BFS', color='orange')

plt.xlabel("Number of Connections")
plt.ylabel("Runtime (seconds)")
plt.title("BFS (Orange) / DFS (Blue) Comparison")
plt.legend()
plt.show()
