def remove_element(nums, val):
    # Remove all occurrences of the specified value from the list
    while val in nums:
        nums.remove(val)
    
    # Optional: Print the modified list after all removals for debugging
    print("Modified list:", nums)
    
    return nums

# Example usage
nums = [3, 2, 2, 3]
result = remove_element(nums, 2)
print("Final list:", result)
