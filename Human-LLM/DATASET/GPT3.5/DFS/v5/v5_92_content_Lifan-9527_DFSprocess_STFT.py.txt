import math
import numpy as np

def stft(source, winlen, slide):
    """
    Compute the Short-Time Fourier Transform (STFT) of a signal.

    Args:
        source (array-like): Input signal.
        winlen (int): Length of the window.
        slide (int): Number of samples the window is advanced for each step.

    Returns:
        ndarray: STFT of the input signal.
    """
    # Pad the input signal with zeros to ensure its length is a power of 2
    padded_length = next_power_of_2(len(source))
    if padded_length > len(source):
        data = source + [0] * (padded_length - len(source))
    else:
        data = source

    # Calculate the number of windows
    num_windows = (len(data) - winlen) // slide + 1

    # Initialize a matrix to store the windowed segments of the signal
    window_matrix = np.empty((winlen, num_windows))

    # Fill the matrix with the windowed segments of the signal
    for i in range(num_windows):
        start_index = slide * i
        end_index = start_index + winlen
        window_matrix[:, i] = data[start_index:end_index]

    # Apply the Fast Fourier Transform (FFT) to each column of the matrix
    complex_matrix = np.fft.fft(window_matrix, axis=0)

    return complex_matrix

def next_power_of_2(k):
    """
    Return the next power of 2 greater than or equal to k.

    Args:
        k (int): Input value.

    Returns:
        int: Next power of 2.
    """
    return 2 ** math.ceil(math.log2(k))
