from collections import defaultdict

class Graph:
    def __init__(self):
        # Initialize an empty graph using defaultdict
        self.adjacency_list = defaultdict(list)

    def add_edge(self, source, destination):
        # Add an edge between source and destination vertices
        self.adjacency_list[source].append(destination)

    def dfs(self, start_vertex):
        # Initialize an empty set to keep track of visited vertices
        visited = set()

        # Initialize variables to count connected components and store them
        num_components = 0
        connected_components = []

        # Iterate through each vertex in the graph
        for vertex in self.adjacency_list.keys():
            if vertex not in visited:
                # If the vertex has not been visited, perform DFS traversal
                component_vertices = []
                self.dfs_util(vertex, visited, component_vertices)
                connected_components.append(component_vertices)
                num_components += 1

        # Print the number of connected components and their respective vertices
        output_message = "The graph has {} connected components. Components: {}"
        print(output_message.format(num_components, connected_components))

    def dfs_util(self, vertex, visited, component_vertices):
        # Mark the current vertex as visited and add it to the current component
        visited.add(vertex)
        component_vertices.append(vertex)

        # Recursively traverse all neighbors of the current vertex
        for neighbor in self.adjacency_list[vertex]:
            if neighbor not in visited:
                self.dfs_util(neighbor, visited, component_vertices)
