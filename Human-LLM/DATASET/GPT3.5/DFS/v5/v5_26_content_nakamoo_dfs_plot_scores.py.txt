import argparse
import matplotlib.pyplot as plt
import pandas as pd

def main():
    # Set up command-line argument parser
    parser = argparse.ArgumentParser(description='Plot mean and median scores from a data file.')
    parser.add_argument('scores', type=str, help='path to the scores data file (e.g., scores.txt)')
    parser.add_argument('--title', type=str, default=None, help='title for the plot (optional)')
    args = parser.parse_args()

    # Read scores data from the specified file
    scores_data = pd.read_csv(args.scores, delimiter='\t')

    # Plot mean and median scores against steps
    for column in ['mean', 'median']:
        plt.plot(scores_data['steps'], scores_data[column], label=column)

    # Set the plot title if provided as a command-line argument
    if args.title is not None:
        plt.title(args.title)

    # Set labels for x and y axes, and add legend
    plt.xlabel('steps')
    plt.ylabel('score')
    plt.legend(loc='best')

    # Save the figure as a PNG file with the same name as the input scores file
    figure_filename = args.scores + '.png'
    plt.savefig(figure_filename)
    print(f'Saved the figure as {figure_filename}')

if __name__ == '__main__':
    main()
