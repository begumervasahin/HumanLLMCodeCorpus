def combination_sum(candidates, target):
    # Sort the candidates in ascending order to optimize the search
    candidates.sort()
    # Initialize an empty list to store the result combinations
    result = []
    # Start finding combinations from the beginning of the candidates list
    start_index = 0
    # Initialize an empty list to store the current combination sequence
    current_sequence = []
    # Call the recursive helper function to find combinations
    find_combinations(candidates, target, start_index, current_sequence, result)
    # Return the list of all valid combinations
    return result

def find_combinations(candidates, target, start_index, current_sequence, result):
    # Iterate through the candidates starting from the specified index
    for i in range(start_index, len(candidates)):
        num = candidates[i]
        # If the current number exceeds the target, terminate the loop
        if num > target:
            return
        # Add the current number to the current sequence
        current_sequence.append(num)
        # If the current number is less than the target, recursively search for combinations
        if num < target:
            find_combinations(candidates, target - num, i, current_sequence, result)
        # If the current number equals the target, add the current sequence to the result list
        else:
            result.append(list(current_sequence))
        # Backtrack by removing the last added number from the current sequence
        current_sequence.pop()
