import sys
import os
import collections
from helper import write_to_file, next_movement

# Set recursion limit to avoid stack overflow
sys.setrecursionlimit(4500)

def depth_limited_search(maze_map, maze_dict, pose_queue, depth, end, obstacle):
    """Perform depth-limited search to find a path from start to end within given depth."""
    if not pose_queue:
        return False

    current = pose_queue.popleft()
    if current == end:
        return True

    if depth <= 0:
        return False

    for next_pos in next_movement(current):
        if next_pos not in maze_dict and next_pos not in obstacle:
            maze_dict[next_pos] = current
            pose_queue.append(next_pos)

    return depth_limited_search(maze_map, maze_dict, pose_queue, depth - 1, end, obstacle)

def iterative_deepening_depth_first_search(maze_map):
    """Perform Iterative Deepening Depth-First Search (IDDFS) to find paths from start to goal."""
    obstacle = []
    goal = []
    start = None

    # Parse maze_map to extract obstacles, goals, and start position
    for i in range(len(maze_map)):
        for j in range(len(maze_map[0]) - 1):
            if maze_map[i][j] in ("|", "="):
                obstacle.append((i, j))
            elif maze_map[i][j] == "*":
                goal.append((i, j))
            elif maze_map[i][j] == "s":
                start = (i, j)

    # Set maximum depth for IDDFS
    max_depth = len(maze_map) * len(maze_map[0])

    all_paths = []

    # Iterate over each goal and perform IDDFS
    for end in goal:
        for depth in range(max_depth + 1):
            maze_dict = {start: None}
            pose_queue = collections.deque([start])
            if depth_limited_search(maze_map, maze_dict, pose_queue, depth, end, obstacle):
                # Trace back the path from goal to start
                path = []
                current = end
                while current != start:
                    path.append(current)
                    current = maze_dict[current]
                all_paths.append(path)
                break

    return all_paths

if __name__ == '__main__':
    # Set the default map directory
    map_directory = 'maps'

    # Check if a map directory is provided as a command-line argument
    if len(sys.argv) > 1:
        map_directory = sys.argv[1]

    # Get the current working directory
    working_directory = os.getcwd()

    # Read maps from files
    maps = []
    for i in range(1, 4):
        file_path_map = os.path.join(working_directory, map_directory, f'map{i}.txt')
        with open(file_path_map) as f:
            maps.append(f.readlines())

    # Perform IDDFS on each map and write results to output files
    for idx, maze_map in enumerate(maps):
        paths = iterative_deepening_depth_first_search(maze_map)
        write_to_file(f"./results/iddfs_map{idx + 1}.txt", paths, maze_map)
