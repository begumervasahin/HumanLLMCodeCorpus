class Solution:
    def addDigits(self, num):
        # Continuously sum the digits until the result becomes a single digit
        while num >= 10:
            num = self.sum_digits(num)
        return num

    def sum_digits(self, num):
        # Initialize a variable to store the sum of digits
        digit_sum = 0

        # Sum the digits of the number
        while num > 0:
            digit_sum += num % 10
            num //= 10

        return digit_sum
