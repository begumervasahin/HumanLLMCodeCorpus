def dfs_route(graph, start, end):
    """
    Find routes from start to end using Depth-First Search algorithm.
    
    Args:
    - graph (dict): The graph representation.
    - start (str): The starting node.
    - end (str): The destination node.
    
    Yields:
    - list: A list containing the path from start to end.
    """
    stack = [(start, [start])]  # Initialize the stack with the starting node and path
    while stack:
        node, path = stack.pop()  # Pop the current node and its path from the stack
        for neighbor in graph[node].difference(path):  # Iterate over neighbors not visited yet
            if neighbor == end:  # If the neighbor is the destination, yield the path
                yield path + [neighbor]
            else:
                stack.append((neighbor, path + [neighbor]))  # Add the neighbor and its path to the stack

print("---------Program Result-------")
# Find all possible routes from 'E' to 'G'
all_routes = list(dfs_route(graph, 'E', 'G'))
route_count = len(all_routes)
# Print the result
if route_count == 0:
    print("No route found.")
else:
    print("Possible routes:", all_routes)
    print("DFS route:", all_routes[0])
