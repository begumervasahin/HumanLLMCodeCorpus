import nflgame

class TeamDefense:
    """
    A class for computing defensive stats for an NFL team.
    """

    def __init__(self):
        # Initialize defensive statistics attributes
        self.sacks = 0
        self.interceptions = 0
        self.fumble_recoveries = 0
        self.kick_return_touchdowns = 0
        self.punt_return_touchdowns = 0
        self.interception_return_touchdowns = 0
        self.fumble_recovery_touchdowns = 0
        self.misc_defensive_touchdowns = 0
        self.safeties = 0
        self.blocked_kicks = 0

    def calculate_team_stats(self, players, team_abbr):
        """
        Calculate defensive statistics for the specified team.

        :param players: NFL game players data.
        :param team_abbr: Team abbreviation (e.g., 'NE' for New England Patriots).
        """
        # Reset defensive statistics
        self.reset_defensive_stats()

        # Iterate through defensive players of the specified team
        for player in players.defense().filter(team=team_abbr):
            # Accumulate defensive statistics
            self.sacks += player.defense_sk
            self.interceptions += player.defense_int
            self.fumble_recoveries += player.defense_frec
            self.kick_return_touchdowns += player.kickret_tds
            self.punt_return_touchdowns += player.puntret_tds
            self.interception_return_touchdowns += player.defense_int_tds
            self.fumble_recovery_touchdowns += player.defense_frec_tds
            self.misc_defensive_touchdowns += player.defense_misc_tds
            self.safeties += player.defense_safe

            # Accumulate blocked kicks
            self.blocked_kicks += player.defense_puntblk
            self.blocked_kicks += player.defense_xpblk
            self.blocked_kicks += player.defense_fgblk

    def reset_defensive_stats(self):
        """
        Reset all defensive statistics to zero.
        """
        self.sacks = 0
        self.interceptions = 0
        self.fumble_recoveries = 0
        self.kick_return_touchdowns = 0
        self.punt_return_touchdowns = 0
        self.interception_return_touchdowns = 0
        self.fumble_recovery_touchdowns = 0
        self.misc_defensive_touchdowns = 0
        self.safeties = 0
        self.blocked_kicks = 0

if __name__ == "__main__":
    # Create an instance of the TeamDefense class
    team_defense = TeamDefense()

    # Get NFL game players data
    players_data = nflgame.combine_game_stats(nflgame.games(2019))

    # Specify team abbreviation for which defensive statistics are to be calculated
    team_abbreviation = 'NE'  # Example: New England Patriots

    # Calculate defensive statistics for the specified team
    team_defense.calculate_team_stats(players_data, team_abbreviation)

    # Print the defensive statistics for the specified team
    print("Defensive Statistics for Team {}: ".format(team_abbreviation))
    print("Sacks:", team_defense.sacks)
    print("Interceptions:", team_defense.interceptions)
    print("Fumble Recoveries:", team_defense.fumble_recoveries)
    print("Kick Return Touchdowns:", team_defense.kick_return_touchdowns)
    print("Punt Return Touchdowns:", team_defense.punt_return_touchdowns)
    print("Interception Return Touchdowns:", team_defense.interception_return_touchdowns)
    print("Fumble Recovery Touchdowns:", team_defense.fumble_recovery_touchdowns)
    print("Miscellaneous Defensive Touchdowns:", team_defense.misc_defensive_touchdowns)
    print("Safeties:", team_defense.safeties)
    print("Blocked Kicks:", team_defense.blocked_kicks)
