import os
import sys
import requests
import pymongo
from flask import Flask, request, jsonify
from encryption import AESCipher

# Initialize Flask application
app = Flask(__name__)

# Configuration
AES_KEY = "94CA61A3CFC9BB7B8FF07C723917851A"
SERVER_KEY = "7596DE01913A20EC9069DBE508C5FEA3"
MONGO_SERVER_ADDR = "localhost"
MONGO_SERVER_PORT = "27017"
FULL_SERVER_ADDR = "http://127.0.0.1:5000"
HEADERS = {"Content-type": "application/json"}

# Initialize AES cipher
cipher = AESCipher(AES_KEY)

# Connect to MongoDB
mongo_db_addr = f"mongodb://{MONGO_SERVER_ADDR}:{MONGO_SERVER_PORT}"
mongo_client = pymongo.MongoClient(mongo_db_addr)
mongo_db = mongo_client.dfs

# Function to register server
def register_server():
    server_name = input("Enter server name: ")
    if mongo_db.servers.find_one({"server_name": server_name}):
        print("ERROR: Server with this name already exists!")
        sys.exit(1)
    data = {
        "server_name": server_name,
        "server_key": cipher.encode_string(SERVER_KEY).decode()
    }
    response = requests.post(f"{FULL_SERVER_ADDR}/register_server", data=json.dumps(data), headers=HEADERS)

# Main function
if __name__ == '__main__':
    register_server()
    app.run()
