class Element:
    def __init__(self, value: int):
        self.value = value
        self.next = None

class LinkedList:
    def __init__(self, head: Element = None):
        self.head = head

    def append_last(self, new_element: Element):
        """Append a new element at the end of the linked list."""
        if not self.head:
            self.head = new_element
        else:
            current = self.head
            while current.next:
                current = current.next
            current.next = new_element

    def insert_first(self, new_element: Element):
        """Insert a new element at the beginning of the linked list."""
        new_element.next = self.head
        self.head = new_element

    def delete_first(self) -> Element:
        """Delete and return the first element of the linked list."""
        if self.head:
            temp = self.head
            self.head = self.head.next
            temp.next = None
            return temp
        else:
            return None

class Stack:
    def __init__(self, top: Element = None):
        self.ll = LinkedList(top)

    def push(self, new_element: Element):
        """Push a new element onto the stack."""
        self.ll.insert_first(new_element)

    def pop(self) -> Element:
        """Pop the top element from the stack."""
        return self.ll.delete_first()

# Create elements
e1 = Element(1)
e2 = Element(2)
e3 = Element(3)
e4 = Element(4)

# Create a stack and perform operations
stack = Stack(e1)
stack.push(e2)
stack.push(e3)

# Output the popped values
print(stack.pop().value)  # Output: 3
print(stack.pop().value)  # Output: 2
print(stack.pop().value)  # Output: 1
print(stack.pop())  # Output: None (as the stack is empty after popping all elements)

# Push a new element onto the stack and output its value after popping
stack.push(e4)
print(stack.pop().value)  # Output: 4
