import json
import rep
import busqueda
import sys

def validate_arguments(arguments):
    """
    Validate the command-line arguments.
    """
    if len(arguments) != 3:
        print('Debe ingresar dos argumentos en el comando de línea:')
        print('Tipo de búsqueda: bfs, dfs')
        print('Mostrar Q: si, no')
        quit()

    search_type = arguments[1]
    if search_type not in ['bfs', 'dfs']:
        print('Por favor ingrese un argumento válido para el primer parámetro: bfs, dfs')
        quit()

    show_q = arguments[2]
    if show_q not in ['si', 'no']:
        print('Por favor ingrese un argumento válido para el segundo parámetro: si, no')
        quit()

def load_data():
    """
    Load data from the JSON file.
    """
    with open('data.json') as json_file:
        data = json.load(json_file)
        print(data['estadoInicial'])
        print(data['estadoObjetivo'])
        return data

def set_show_q(argument):
    """
    Set the showQ variable based on the second argument.
    """
    return argument == 'si'

def print_solution(solution):
    """
    Print the steps and applied operators for the solution.
    """
    print('Pasos para la solución:')
    for step in solution['estadosAnteriores']:
        print(step)

    print('\nOperadores aplicados:')
    for operation in solution['operadores']:
        print(operation)

if __name__ == "__main__":
    # Validate command-line arguments
    validate_arguments(sys.argv)

    # Set the showQ variable based on the second argument
    mostrarQ = set_show_q(sys.argv[2])

    # Load data from JSON file
    data = load_data()

    # Set up the world based on the loaded data
    world = rep.setWorld(data)
    inicial = world['inicial']
    final = world['final']

    # Search for the solution using the specified search algorithm
    solucion = busqueda.buscarSolucion(inicial, final, sys.argv[1], mostrarQ)

    # Print the solution
    print_solution(solucion)
