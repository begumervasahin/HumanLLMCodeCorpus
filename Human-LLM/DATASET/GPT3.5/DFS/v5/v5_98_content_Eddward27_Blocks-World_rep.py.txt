import clase

def set_world(data):
    initial_state = set_state(data['initialState'])
    final_state = set_state(data['finalState'])
    world = {'initial': initial_state, 'final': final_state}
    return world

def set_state(tower_arrangement):
    state = {'operators': [], 'states': [], 'previous_states': []}
    for tower_str in tower_arrangement:
        tower = list(tower_str)
        for i in range(len(tower)):
            name = tower[i]
            on = None
            clear = False
            on_table = True
            if i < len(tower) - 1:
                on = tower[i + 1]
                on_table = False
            if i == 0:
                clear = True
            state['states'].append(clase.Cubo(name, on, clear, on_table))
    state['previous_states'].append(state_to_string(state['states']))
    return state

def same_state(state1, state2):
    towers1 = len(state1)
    towers2 = len(state2)
    if towers1 != towers2:
        return False
    for i in range(len(state1)):
        state_found = False
        for j in range(len(state2)):
            if state1[i].nombre == state2[j].nombre:
                if (state1[i].on == state2[j].on and state1[i].on_table == state2[j].on_table and state1[i].clear == state2[j].clear):
                    state_found = True
                    break
        if not state_found:
            return False
    return True

def state_to_string(state):
    state_str = []
    for cube_rep in state:
        if cube_rep.clear:
            state_str.append(cube_rep.nombre)
    for i in range(len(state_str)):
        not_last = True
        current_cube = state_str[i]
        while not_last:
            for j in range(len(state)):
                if current_cube == state[j].nombre:
                    if state[j].on is None:
                        not_last = False
                        break
                    state_str[i] += state[j].on
                    current_cube = state[j].on
                    break
    return state_str
