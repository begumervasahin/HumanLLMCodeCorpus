import unittest
from breadth_first_search import bfs
from state import State

class TestBfs(unittest.TestCase):
    def test_result(self):
        # Display test header
        print("=======================")
        print("Breadth-First Search Algorithm Test")
        print("=======================")
        
        # Execute BFS algorithm with an initial state
        initial_state = State(red=13, blue=16, green=17)
        result = bfs(initial_state)
        
        # Extract information from the result
        final_state = result['final_state']
        full_path = result['full_path']
        path = result['path']
        
        # Verify final state properties and path lengths
        self.assertEqual(final_state.red_count, 0)
        self.assertEqual(final_state.green_count, 46)
        self.assertEqual(final_state.blue_count, 0)
        self.assertEqual(len(full_path), 180169)
        self.assertEqual(len(path), 17)
        
        # Print path lengths for clarity
        print(f"Full path length: {len(full_path)}")
        print(f"Solution path length: {len(path)}")
        
        # Display nodes in the solution path
        print("Nodes in the Solution Path:")
        for i, node in enumerate(path, start=1):
            print(f"{i}: {node}")

if __name__ == "__main__":
    unittest.main()
