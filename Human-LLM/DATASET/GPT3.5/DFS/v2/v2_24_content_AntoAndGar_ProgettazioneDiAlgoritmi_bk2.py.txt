def generate_binary_strings(n, k, i=0, tot1=0, current_solution=[]):
    # Base case: if the length of the current solution equals n, print it
    if i == n:
        print(''.join(current_solution))
    else:
        # Add '0' to the current solution and continue recursion
        current_solution.append('0')
        generate_binary_strings(n, k, i + 1, tot1, current_solution)
        current_solution.pop()  # Backtrack by removing the last element

        # If we can add '1' (total 1s < k), add it and continue recursion
        if tot1 < k:
            current_solution.append('1')
            generate_binary_strings(n, k, i + 1, tot1 + 1, current_solution)
            current_solution.pop()  # Backtrack by removing the last element

# Example usage: generate binary strings of length 4 with exactly 1 '1'
generate_binary_strings(4, 1)
