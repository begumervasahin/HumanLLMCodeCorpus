class Problem1:
    def __init__(self, path=None):
        # Initialize problem parameters
        self.m = 0
        self.n = 0
        self.num_wall = 0
        self.walls = []
        self.start = [1, 1]
        self.goal = [self.m, self.n]
        self.path = path
        self.u = [0, -1, 'u']
        self.d = [0, 1, 'd']
        self.r = [1, 0, 'r']
        self.l = [-1, 0, 'l']
        self.move = [self.u, self.d, self.r, self.l]
        if not path:
            self.path = 'test1.txt'
        self.initial_state()

    def actions(self, state):
        # Determine possible actions from a given state
        x, y = state
        res = []
        for m in self.move:
            r = [x + m[0], y + m[1]]
            if 0 < r[0] <= self.m and 0 < r[1] <= self.n:
                t1 = [x, y, x + m[0], y + m[1]]
                t2 = [x + m[0], y + m[1], x, y]
                if t1 not in self.walls and t2 not in self.walls:
                    res.append(m[2])
        return res

    def open_file(self):
        # Open and read the input file to set up the problem
        with open(self.path) as f:
            lines = f.readlines()
        self.m, self.n = map(int, lines[0].split())
        self.num_wall = int(lines[1])
        for line in lines[2:]:
            a1, a2, a3, a4 = map(int, line.split())
            self.walls.append([a1, a2, a3, a4])
        self.goal = [self.m, self.n]

    def initial_state(self):
        # Set up the initial state of the problem
        self.open_file()
        return self.start

    def goal_test(self, state):
        # Check if a given state is the goal state
        x, y = state
        return x == self.m and y == self.n

    def result(self, state, action):
        # Calculate the result of taking an action from a given state
        x, y = state
        if action == 'u':
            return [x, y - 1]
        elif action == 'd':
            return [x, y + 1]
        elif action == 'r':
            return [x + 1, y]
        elif action == 'l':
            return [x - 1, y]

    def path_cost(self, c, state1, action, state2):
        # Calculate the cost of a path
        pass

    def get_cost(self, state1, state2):
        # Get the cost between two states
        return 1

    def get_reverse_Action(self, action):
        # Get the reverse action of a given action
        if action == 'u':
            return 'd'
        elif action == 'd':
            return 'u'
        elif action == 'r':
            return 'l'
        elif action == 'l':
            return 'r'

    def heuristic(self, state):
        # Calculate the heuristic value for a state
        x, y = state
        d = (self.goal[1] - y) ** 2 + (self.goal[0] - x) ** 2
        return d ** 0.5

    def call_dfs_unlimited(self):
        # Call the DFS Unlimited search algorithm
        dfs = DFSUnlimited(self.initial_state, self.actions, self.result, self.goal_test)
        dfs.search_dfs()

    def call_uniform_cost(self):
        # Call the Uniform Cost search algorithm
        uc = UniformCost(self.initial_state, self.actions, self.result, self.goal_test, self.get_cost)
        uc.search_uniform()

    def call_bidirectional(self):
        # Call the Bidirectional search algorithm
        bd = Bidirectional(self.initial_state, self.actions, self.result, self.goal_test, self.get_cost,
                           self.get_reverse_Action)
        bd.search_bidirection(self.goal)

    def call_astar(self):
        # Call the A* search algorithm
        astar = AStar(self.initial_state, self.actions, self.result, self.goal_test, self.get_cost, self.heuristic)
        astar.search_astar()
