class TreeNode:
    def __init__(self, value):
        self.value = value
        self.left = None
        self.right = None

class TreeDLS:
    def __init__(self, root):
        self.root = root

    def search(self, target, level_threshold):
        level = 0
        found = False
        stack = [[self.root, level]]

        while stack:
            node, current_level = stack.pop()
            
            if node.value == target and current_level < level_threshold:
                print("Target found within the specified level threshold.")
                found = True
                break
            
            if node.left and current_level < level_threshold - 1:
                stack.append([node.left, current_level + 1])
            if node.right and current_level < level_threshold - 1:
                stack.append([node.right, current_level + 1])

        if not found:
            print("Target not found within the specified level threshold.")

# Create the binary tree
node1 = TreeNode(1)
node2 = TreeNode(2)
node3 = TreeNode(3)
node4 = TreeNode(4)
node5 = TreeNode(5)
node6 = TreeNode(6)
node7 = TreeNode(7)
node2.left = node4
node2.right = node5
node3.left = node6
node3.right = node7
node1.left = node2
node1.right = node3

# Initialize the TreeDLS instance
tree = TreeDLS(node1)

# Search for a target value within a given level threshold
tree.search(4, 2)
