import requests
from collections import defaultdict

class DirFileServer:
    def __init__(self):
        self.dir_id_counter = 1
        self.dirs = defaultdict(list)
        self.file_server = 'http://localhost:'

    def get_files_on_all_servers(self, **kwargs):
        message = kwargs.get('message')
        if message is None:
            return None
        else:
            files = []
            for server_id in self.dirs.keys():
                response = requests.get(f"{self.file_server}{server_id}/files", **kwargs)
                files.append(response)
            return files

    def add_file_server(self, **kwargs):
        message = kwargs.get('message')
        if message is None:
            return None
        else:
            return requests.post()

    def add_one_file(self, **kwargs):
        message = kwargs.get('message')
        if message is None:
            return None
        else:
            min_files = float('inf')
            min_server = None
            for server_id, files in self.file_server.items():
                num_files = len(files)
                if num_files < min_files:
                    min_files = num_files
                    min_server = server_id
            return requests.post(f"{self.file_server}{min_server}/file", **kwargs)

    def get_file(self, **kwargs):
        message = kwargs.get('message')
        if message is None:
            return None
        else:
            if '_id' in message and message['_id'] is not None:
                for server_id, files in self.dirs.items():
                    if message['_id'] in files:
                        return requests.get(f"{self.file_server}{server_id}/file", json={'_id': message['_id']})
            else:
                if 'name' in message and message['name'] is not None:
                    return self.get_files_on_all_servers(message=message['name'])
                else:
                    return None

# Example usage
dir_file_server = DirFileServer()
files = dir_file_server.get_files_on_all_servers(message={'name': 'example.txt'})
print(files)
