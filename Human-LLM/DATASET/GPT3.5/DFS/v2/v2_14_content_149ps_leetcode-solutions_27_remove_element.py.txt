def remove_element(nums, val):
    # Iterate over the list and remove all occurrences of the specified value
    while val in nums:
        nums.remove(val)
        # Print the list after each removal (optional for debugging)
        print("Modified list:", nums)
    return nums

# Example usage
nums = [3, 2, 2, 3]
result = remove_element(nums, 2)
print("Final list:", result)
