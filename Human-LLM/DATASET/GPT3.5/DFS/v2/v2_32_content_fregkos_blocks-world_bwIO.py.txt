import re

def open_problem(file_name):
    # Read the problem file
    with open(file_name, 'r') as file:
        problem = file.read()

    # Clean up the problem text
    problem = re.sub('[ ]+', ' ', re.sub('[ ]*\)\(', ')\n(', re.sub('\n', '', problem)))

    # Extract segmented data from the problem text
    segmented_list = re.compile('\(:objects.*\)\s*|\(:INIT.*\)\s*|\(:goal.*\)\s*', re.IGNORECASE).findall(problem)

    # Extract objects, initial state, and goal state from segmented data
    objects = re.findall(' (\w*)', segmented_list[0])
    init_temp = re.findall('\((\w*) (\w*)\s?(\w*)?\)', segmented_list[1])
    goal_temp = re.findall('\(\w* (\w*) (\w*)\)', segmented_list[2])

    # Initialize the initial state dictionary with objects
    init = {i: ['table', True] for i in objects}

    # Update the initial state based on INIT statements in the problem text
    for item in init_temp:
        if item[0] == 'ON':
            init[item[1]][0] = item[2]
            init[item[2]][1] = False

    # Initialize the goal state dictionary with objects
    goal = {i: ['table', True] for i in objects}

    # Update the goal state based on goal statements in the problem text
    for item in goal_temp:
        goal[item[0]][0] = item[1]
        goal[item[1]][1] = False

    return init, goal, objects

def write_solution(solution, output=''):
    # If no output file name is provided, use 'solution.txt'
    if output == '':
        output = 'solution.txt'

    # Write the solution to the output file
    with open(output, 'w') as file:
        i = 0
        for move in solution:
            i += 1
            file.write('{}. Move({}, {}, {})\n'.format(i, move[0], move[1], move[2]))
