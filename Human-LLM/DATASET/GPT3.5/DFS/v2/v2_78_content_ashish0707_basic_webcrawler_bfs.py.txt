class Searcher:
    """
    A base class for search algorithms.
    """

class BFS(Searcher):
    """
    A class representing Breadth-First Search algorithm.
    """
    # Name of the BFS algorithm
    name = '2.A_bfs.txt'

    def merge_list(self, list1, list2):
        """
        Merge two lists.
        
        Args:
            list1 (list): The first list.
            list2 (list): The second list.
        
        Returns:
            list: The merged list.
        """
        merged_list = list1 + list2
        return merged_list
