import sys
import os
import collections
from helper import write_to_file, next_movement

# Set recursion limit to avoid stack overflow
sys.setrecursionlimit(4500)

# Depth-limited search
def dls(maze_map, maze_dict, pose_queue, depth, end, obstacle):
    if len(pose_queue) == 0:
        return 0

    current = pose_queue.popleft()
    if current == end:
        return 1

    if depth <= 0:
        return 0
    else:
        for i in next_movement(current):
            if i not in maze_dict and i not in obstacle:
                maze_dict[i] = current
                pose_queue.append(i)
        result = dls(maze_map, maze_dict, pose_queue, depth-1, end, obstacle)
        if result == 1:
            return 1
        return 0

# Iterative Deepening Depth-First Search
def iterative_deepening_depth_first_search(maze_map):
    obstacle = []
    goal = []
    start = ()

    # Parse maze_map to extract obstacles, goals, and start position
    for i in range(len(maze_map)):
        for j in range(len(maze_map[0])-1):
            if maze_map[i][j] in ("|", "="):
                obstacle.append((i, j))
            elif maze_map[i][j] == "*":
                goal.append((i, j))
            elif maze_map[i][j] == "s":
                start = (i, j)

    # Set maximum depth for IDDFS
    maximum_depth = len(maze_map) * len(maze_map[0])

    # Iterate over each goal and perform IDDFS
    all_paths = []
    for i in goal:
        for depth in range(0, maximum_depth + 1):
            maze_dict = {}
            maze_dict[start] = None
            pose_queue = collections.deque()
            pose_queue.append(start)
            result = dls(maze_map, maze_dict, pose_queue, depth, i, obstacle)
            if result == 1:
                break

        # Extract paths for each goal
        if maze_dict.get(i) is not None:
            path = []
            k = maze_dict[i]
            while k != start:
                path.append(k)
                k = maze_dict[k]
            all_paths.append(path)

    return all_paths

if __name__ == '__main__':
    working_directory = os.getcwd()
    map_directory = 'maps'  # Default map directory

    # Check if map directory is provided as command-line argument
    if len(sys.argv) > 1:
        map_directory = sys.argv[1]

    # Read maps from files
    maps = []
    for i in range(1, 4):
        file_path_map = os.path.join(working_directory, map_directory, f'map{i}.txt')
        with open(file_path_map) as f:
            maps.append(f.readlines())

    # Perform IDDFS on each map and write results to output files
    for idx, maze_map in enumerate(maps):
        paths = iterative_deepening_depth_first_search(maze_map)
        write_to_file(f"./results/iddfs_map{idx+1}.txt", paths, maze_map)
