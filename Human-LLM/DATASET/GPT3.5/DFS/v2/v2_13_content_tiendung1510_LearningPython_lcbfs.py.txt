from collections import deque
from mymap import myMap  # Assuming myMap is a dictionary containing the map data
from modules import showResultWithAttr, showStep  # Assuming these functions are defined in a separate module

def LCBFS(start, goal):
    # Initialize the queue with the starting city
    queue = deque([start])
    
    # Initialize the dictionary to track previous cities and their total cost
    previous = {city: {'from': None, 'total_cost': 0} for city in myMap.keys()}
    
    # Initialize the counter for tracking steps
    step_counter = 0
    
    # Main loop for the BFS search
    while queue:
        step_counter += 1
        showStep(step_counter, queue, previous)  # Display the current step
        
        # Process current cities in the queue
        current_cities = list(queue)
        queue.clear()  # Clear the queue for the next level
        
        # Explore neighbors of current cities
        for curCity in current_cities:
            curCityTotalCost = previous[curCity]['total_cost']
            if curCity != goal:
                for neighbor_city, info in myMap[curCity].items():
                    cityTotalCost = previous[neighbor_city]['total_cost']
                    totalCost = info['cost'] + curCityTotalCost
                    if previous[neighbor_city]['from'] is None or totalCost < cityTotalCost:
                        if neighbor_city in queue:
                            queue.remove(neighbor_city)
                        queue.append(neighbor_city)
                        previous[neighbor_city]['from'] = curCity
                        previous[neighbor_city]['total_cost'] = totalCost
        
        # Exit the loop if there are no more cities to explore
        if not queue:
            break
    
    # Display the result if a route is found
    if previous[goal]['from'] is not None:
        showResultWithAttr(previous, start, goal)
        return True
    else:
        return False

# Example usage:
start_city = 'A'  # Replace with actual start city
goal_city = 'G'  # Replace with actual goal city
result = LCBFS(start_city, goal_city)
if result:
    print("Route found!")
else:
    print("No route found.")
