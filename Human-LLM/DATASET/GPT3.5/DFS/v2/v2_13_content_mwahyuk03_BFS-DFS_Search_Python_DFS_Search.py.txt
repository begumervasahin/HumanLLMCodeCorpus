# Define the graph
graph = {
    'A': {'B', 'C', 'D'},
    'B': {'A', 'C', 'E', 'F', 'G'},
    'C': {'A', 'B', 'D'},
    'D': {'A', 'C', 'H', 'I'},
    'E': {'B', 'L'},
    'F': {'B', 'G', 'L'},
    'G': {'B', 'F', 'J', 'H'},
    'H': {'D', 'G', 'J'},
    'I': {'D', 'J'},
    'J': {'I', 'G', 'H', 'K', 'M'},
    'K': {'L', 'J', 'M'},
    'L': {'E', 'F', 'M', 'K'},
    'M': {'L', 'K', 'J'}
}

def dfs_route(graph, start, end):
    """
    Find routes from start to end using Depth-First Search algorithm.
    """
    stack = [(start, [start])]
    while stack:
        node, path = stack.pop()
        for neighbor in graph[node].difference(path):
            if neighbor == end:
                yield path + [neighbor]
            else:
                stack.append((neighbor, path + [neighbor]))

print("---------Program Result-------")
# Find all possible routes from 'E' to 'G'
all_routes = list(dfs_route(graph, 'E', 'G'))
route_count = len(all_routes)
# Print the result
if route_count == 0:
    print("No route found.")
else:
    print("Possible routes:", all_routes)
    print("DFS route:", all_routes[0])
