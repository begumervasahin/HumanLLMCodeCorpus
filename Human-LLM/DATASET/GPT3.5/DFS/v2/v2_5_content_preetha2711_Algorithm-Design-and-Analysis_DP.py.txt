# Define the number of rows (i) and columns (k) for Pascal's triangle
num_rows = 1000
num_cols = 100

# Initialize a 2D list to store Pascal's triangle values
pascals_triangle = [[None for _ in range(num_cols + 1)] for _ in range(num_rows + 1)]

# Calculate Pascal's triangle values
for row in range(1, num_rows + 1):
    for col in range(1, num_cols + 1):
        if col == 1:
            # For the first column, set the value equal to the row number
            pascals_triangle[row][col] = row
        elif col > row:
            # For columns beyond the row index, set the value to 0
            pascals_triangle[row][col] = 0
        else:
            # Calculate the value using the recursive Pascal's triangle formula
            pascals_triangle[row][col] = pascals_triangle[row - 1][col] + pascals_triangle[row - 1][col - 1]

# Print the value of p[i][k]
print("Value at row", num_rows, "and column", num_cols, "in Pascal's triangle is:", pascals_triangle[num_rows][num_cols])
