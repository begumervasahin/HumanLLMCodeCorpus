# Import necessary libraries
from __future__ import print_function
import h5py
import numpy as np

# Define a class for managing dataset
class DataSet(object):
    def __init__(self, images, labels, fake_data=False):
        # Initialize dataset properties
        if fake_data:
            self._num_examples = 10000
        else:
            assert images.shape[0] == labels.shape[0], "Number of images and labels must be the same"
            self._num_examples = images.shape[0]
        self._images = images
        self._labels = labels
        self._epochs_completed = 0
        self._index_in_epoch = 0

    @property
    def images(self):
        return self._images

    @property
    def labels(self):
        return self._labels

    @property
    def num_examples(self):
        return self._num_examples

    @property
    def epochs_completed(self):
        return self._epochs_completed

    # Retrieve the next batch of data
    def next_batch(self, batch_size):
        start = self._index_in_epoch
        self._index_in_epoch += batch_size
        if self._index_in_epoch > self._num_examples:
            # Finished epoch
            self._epochs_completed += 1
            # Shuffle the data
            perm = np.arange(self._num_examples)
            np.random.shuffle(perm)
            self._images = self._images[perm]
            self._labels = self._labels[perm]
            start = 0
            self._index_in_epoch = batch_size
            assert batch_size <= self._num_examples
        end = self._index_in_epoch
        return self._images[start:end], self._labels[start:end]

# Define a class for managing semi-supervised dataset
class SemiDataSet(object):
    def __init__(self, images, labels, n_labeled):
        # Initialize semi-supervised dataset properties
        self.n_labeled = n_labeled
        self.unlabeled_ds = DataSet(images, labels)
        self.num_examples = self.unlabeled_ds.num_examples
        indices = np.arange(self.num_examples)
        shuffled_indices = np.random.permutation(indices)
        images = images[shuffled_indices]
        labels = labels[shuffled_indices]
        y = np.array([np.arange(9)[l==1][0] for l in labels])
        n_classes = y.max() + 1
        n_from_each_class = n_labeled // n_classes
        i_labeled = []
        for c in range(n_classes):
            i = indices[y==c][:n_from_each_class]
            i_labeled += list(i)
        l_images = images[i_labeled]
        l_labels = labels[i_labeled]
        self.labeled_ds = DataSet(l_images, l_labels)

    # Retrieve the next batch of data
    def next_batch(self, batch_size):
        unlabeled_images, _ = self.unlabeled_ds.next_batch(batch_size)
        if batch_size > self.n_labeled:
            labeled_images, labels = self.labeled_ds.next_batch(self.n_labeled)
        else:
            labeled_images, labels = self.labeled_ds.next_batch(batch_size)
        images = np.vstack([labeled_images, unlabeled_images])
        return images, labels

# Function to read dataset from a file
def read_data_sets(file_name='I_feature.h5', n_labeled=100):
    class DataSets(object):
        pass
    data_sets = DataSets()
    # Read data from the HDF5 file
    f = h5py.File(file_name, 'r')
    train_images = f['data'][:]
    train_labels = f['label'][:]
    train_labels = np.argmax(train_labels, 1)
    f.close()
    # Normalize the image data
    max_value = train_images.max()
    min_value = train_images.min()
    train_images = (train_images - min_value) * 1.0 / (max_value - min_value)
    indices = np.arange(train_images.shape[0])
    shuffled_indices = np.random.permutation(indices)
    images = train_images[shuffled_indices]
    labels = train_labels[shuffled_indices]
    y = labels
    n_classes = y.max() + 1
    i_labeled = []
    # Select labeled data from each class
    for c in range(n_classes):
        i = indices[y == c][:n_labeled]
        i_labeled += list(i)
    l_images = images[i_labeled]
    l_labels = labels[i_labeled]
    # Create DataSet objects for training and testing
    data_sets.train = DataSet(l_images, l_labels)
    data_sets.test = DataSet(train_images, train_labels)
    return data_sets

# Usage example:
data = read_data_sets()
batch_x, batch_y = data.train.next_batch(128)
