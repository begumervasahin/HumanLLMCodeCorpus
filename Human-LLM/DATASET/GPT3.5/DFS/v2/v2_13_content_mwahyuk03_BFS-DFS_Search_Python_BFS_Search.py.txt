from collections import deque

# Define the graph
peta = {
    'A': {'B', 'C', 'D'},
    'B': {'A', 'C', 'E', 'F', 'G'},
    'C': {'A', 'B', 'D'},
    'D': {'A', 'C', 'H', 'I'},
    'E': {'B', 'L'},
    'F': {'B', 'G', 'L'},
    'G': {'B', 'F', 'J', 'H'},
    'H': {'D', 'G', 'J'},
    'I': {'D', 'J'},
    'J': {'I', 'G', 'H', 'K', 'M'},
    'K': {'L', 'J', 'M'},
    'L': {'E', 'F', 'M', 'K'},
    'M': {'L', 'K', 'J'}
}

def rute_bfs(peta, mulai, tujuan):
    # Use a queue to perform BFS
    queue = deque([(mulai, [mulai])])
    while queue:
        node, edge = queue.popleft()
        # Iterate over neighbors of the current node
        for next_node in peta[node].difference(edge):
            # If the neighbor is the destination, yield the path
            if next_node == tujuan:
                yield edge + [next_node]
            # Otherwise, continue exploring
            else:
                queue.append((next_node, edge + [next_node]))

print("---------Hasil Program-------")
# Find all possible paths from 'E' to 'G'
list_jalur = list(rute_bfs(peta, 'E', 'G'))
cek = len(list_jalur)
# Print the result
if cek == 0:
    print("Tidak Ketemu")
else:
    print("Jalur yang mungkin -> ", list_jalur)
    print("jalur BFS = ", list_jalur[0])
