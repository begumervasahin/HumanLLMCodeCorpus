from collections import deque

class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

class Solution:
    def zigzagLevelOrder(self, root):
        # Check if the root is None
        if not root:
            return []

        result = []  # List to store the result
        queue = deque([root])  # Initialize a deque with the root node
        left_to_right = True  # Flag to track the direction of traversal

        # Continue traversal while the queue is not empty
        while queue:
            level = []  # List to store the current level nodes
            size = len(queue)  # Get the size of the current level

            # Iterate over the nodes in the current level
            for _ in range(size):
                # Process nodes based on the traversal direction
                if left_to_right:
                    node = queue.popleft()
                    level.append(node.val)
                    if node.left:
                        queue.append(node.left)
                    if node.right:
                        queue.append(node.right)
                else:
                    node = queue.pop()
                    level.append(node.val)
                    if node.right:
                        queue.appendleft(node.right)
                    if node.left:
                        queue.appendleft(node.left)

            # Toggle the traversal direction
            left_to_right = not left_to_right

            # Append the current level to the result
            result.append(level)

        return result

# Example usage:
# Define the binary tree
root = TreeNode(3)
root.left = TreeNode(9)
root.right = TreeNode(20)
root.right.left = TreeNode(15)
root.right.right = TreeNode(7)

# Create an instance of the Solution class
solution = Solution()

# Call the zigzagLevelOrder method to obtain the zigzag level order traversal
zigzag_order = solution.zigzagLevelOrder(root)
print("Zigzag Level Order Traversal:", zigzag_order)
