# Import necessary libraries
from sklearn.neighbors import NearestNeighbors
import numpy as np

# Define the input data array
data_points = np.array([[-1, -1], [-2, -1], [-3, -2], [1, 1], [2, 1], [3, 2]])

# Initialize the NearestNeighbors algorithm with parameters
k_neighbors = 2  # Number of neighbors to find
algorithm_type = 'ball_tree'  # Algorithm for nearest neighbor search
neighbor_finder = NearestNeighbors(n_neighbors=k_neighbors, algorithm=algorithm_type).fit(data_points)

# Print the NearestNeighbors object
print("NearestNeighbors Object:")
print(neighbor_finder)

# Find the nearest neighbors and distances for each data point
distances, indices = neighbor_finder.kneighbors(data_points)

# Print the distances and indices of nearest neighbors
print("\nDistances to Nearest Neighbors:")
print(distances)

print("\nIndices of Nearest Neighbors:")
print(indices)
