def combination_sum(candidates, target):
    # Sort the candidates list in ascending order
    candidates.sort()
    # Initialize an empty list to store the result combinations
    result = []
    # Start the combination sum calculation from index 0
    current_index = 0
    # Initialize an empty list to store the current combination sequence
    result_seq = []
    # Call the helper function to find combinations
    combination_sum_helper(candidates, target, current_index, result_seq, result)
    # Return the final result list containing all valid combinations
    return result

def combination_sum_helper(candidates, target, current_index, result_seq, result):
    # Iterate through the candidates starting from the current index
    for j in range(current_index, len(candidates)):
        # Get the current number from the candidates list
        num = candidates[j]
        # If the current number is greater than the target, stop the iteration
        if num > target:
            return
        # Add the current number to the result sequence
        result_seq.append(num)
        # If the current number is less than the target, recursively call the helper function
        if num < target:
            combination_sum_helper(candidates, target - num, j, result_seq, result)
        # If the current number is equal to the target, add the current result sequence to the final result list
        else:
            result.append(list(result_seq))
        # Backtrack by removing the last number from the result sequence
        result_seq.pop()

# Example usage:
candidates = [2, 3, 6, 7]
target = 7
# Find all unique combinations that sum up to the target
print(combination_sum(candidates, target))  # Output: [[2, 2, 3], [7]]
