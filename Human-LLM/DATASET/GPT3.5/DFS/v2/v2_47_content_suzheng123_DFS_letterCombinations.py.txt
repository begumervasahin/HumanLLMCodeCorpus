class Solution:
    def letterCombinations(self, digits):
        # Define a dictionary mapping digits to corresponding letters
        digit_to_letters = {
            '2': ['a', 'b', 'c'],
            '3': ['d', 'e', 'f'],
            '4': ['g', 'h', 'i'],
            '5': ['j', 'k', 'l'],
            '6': ['m', 'n', 'o'],
            '7': ['p', 'q', 'r', 's'],
            '8': ['t', 'u', 'v'],
            '9': ['w', 'x', 'y', 'z']
        }
        
        # Initialize an empty list to store results
        results = []

        # Check if the input digits string is not empty
        if digits:
            # Start depth-first search to generate combinations
            self.dfs(digits, 0, '', results, digit_to_letters)

        # Return the generated combinations
        return results

    def dfs(self, digits, index, current_combination, results, digit_to_letters):
        # If the current combination is complete, append it to the results list
        if index == len(digits):
            results.append(current_combination)
            return

        # Get the possible letters corresponding to the current digit
        possible_letters = digit_to_letters[digits[index]]

        # Iterate through each possible letter
        for letter in possible_letters:
            # Recursively call dfs with the next index and updated combination
            self.dfs(digits, index + 1, current_combination + letter, results, digit_to_letters)

# Example usage:
solution = Solution()
result = solution.letterCombinations("23")
print(result)  # Output: ["ad", "ae", "af", "bd", "be", "bf", "cd", "ce", "cf"]
