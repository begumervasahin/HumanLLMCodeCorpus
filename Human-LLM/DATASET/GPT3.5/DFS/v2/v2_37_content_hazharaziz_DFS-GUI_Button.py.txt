import pygame
import pygame.locals as pl

class Button:
    def __init__(self, window, color, x, y, width, height, text=''):
        self.window = window
        self.color = color
        self.x = x
        self.y = y
        self.width = width
        self.height = height
        self.text = text

    def draw(self, outline=False):
        if outline:
            pygame.draw.rect(self.window, outline, (self.x - 2, self.y - 2, self.width + 4, self.height + 4))
        pygame.draw.rect(self.window, self.color, (self.x, self.y, self.width, self.height))
        if self.text != '':
            font = pygame.font.SysFont('Times New Roman', 20)
            text = font.render(self.text, 1, (255, 255, 255))
            self.window.blit(text, (
                self.x + (self.width / 2 - text.get_width() / 2), self.y + (self.height / 2 - text.get_height() / 2)))

    def is_over(self, pos):
        if pos[0] > self.x and pos[0] < self.x + self.width:
            if pos[1] > self.y and pos[1] < self.y + self.height:
                return True
        return False

# Pygame initialization
pygame.init()

running = True
while running:
    for event in pygame.event.get():
        if event.type == pl.QUIT:
            running = False
        if event.type == pl.MOUSEBUTTONDOWN:
            pos = pygame.mouse.get_pos()
            if button1.is_over(pos):
                print("Button 1 clicked")
            elif button2.is_over(pos):
                print("Button 2 clicked")
