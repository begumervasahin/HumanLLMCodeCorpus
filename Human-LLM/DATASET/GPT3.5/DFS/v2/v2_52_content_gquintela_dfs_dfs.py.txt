from graph import Graph

def depth_first_search(graph):
    """
    Perform depth-first search traversal on the given graph.

    Args:
    - graph (Graph): The graph to traverse.

    Returns:
    - tuple: A tuple containing the DFS traversal order and vertices marked.
    """
    dfs_order = []
    vertices = graph.get_vertices()
    vertices_marked = {}

    # Initialize vertices_marked dictionary
    for vertex in vertices:
        vertices_marked[vertex] = None

    # Perform DFS traversal for each vertex
    for vertex in vertices:
        mark = vertex
        _dfs_aux(graph, vertex, vertices_marked, mark, dfs_order)

    # Return the DFS traversal order and vertices marked
    return (dfs_order, vertices_marked)

def _dfs_aux(graph, vertex, vertices_marked, mark, dfs_order):
    """
    Recursive function to perform depth-first search traversal.

    Args:
    - graph (Graph): The graph being traversed.
    - vertex: The current vertex being visited.
    - vertices_marked (dict): Dictionary to mark visited vertices.
    - mark: Current marking value.
    - dfs_order (list): List to store the DFS traversal order.
    """
    if vertices_marked[vertex] is not None:
        return
    vertices_marked[vertex] = mark
    dfs_order.append(vertex)
    for neighbour in graph.get_neighbours(vertex):
        _dfs_aux(graph, neighbour, vertices_marked, mark, dfs_order)
