def process_command(conn):
    """Process commands received from a client connection."""
    while True:
        data = conn.recv(1024)
        if not data:
            break
        execute_command(conn, data)
    conn.close()

def execute_command(conn, command_string):
    """Execute a command received from a client."""
    command_obj = json.loads(command_string)
    param_list = command_obj['param_list']
    command = command_obj['command']
    if command == 'heartbeat':
        heartBeat(param_list)
    else:
        send_command(conn, command, param_list)

def receiving():
    """Handle incoming client connections and spawn threads to process commands."""
    log.debug('Receiving commands')
    global client_sock
    client_sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    client_sock.bind((HOST, M_PORT))
    client_sock.listen(1)
    while True:
        conn, addr = client_sock.accept()
        log.debug('Connected by: %s', addr)
        threading.Thread(target=process_command, args=(conn,)).start()
    client_sock.close()

if __name__ == "__main__":
    receiving()
