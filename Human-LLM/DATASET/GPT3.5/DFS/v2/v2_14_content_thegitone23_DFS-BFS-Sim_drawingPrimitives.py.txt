import pygame
import math

# Constants
RADIUS = 30
WIDTH = 800
HEIGHT = 600

# Initialize Pygame
pygame.init()
screen = pygame.display.set_mode((WIDTH, HEIGHT))
clock = pygame.time.Clock()

# Function to calculate distance between two points
def calculate_distance(x1, y1, x2, y2):
    return math.sqrt((x2 - x1) ** 2 + (y2 - y1) ** 2)

# Function to draw a pixel
def draw_pixel(x, y):
    pygame.draw.circle(screen, (255, 255, 255), (x, y), 1)

# Function to draw a line
def draw_line(start_x, start_y, end_x, end_y):
    pygame.draw.line(screen, (255, 255, 255), (start_x, start_y), (end_x, end_y))

# Function to draw a hollow circle
def draw_hollow_circle(center_x, center_y, radius=RADIUS):
    num_segments = 100
    for i in range(num_segments):
        angle = i * 2 * math.pi / num_segments
        start_x = int(center_x + radius * math.cos(angle))
        start_y = int(center_y + radius * math.sin(angle))
        end_x = int(center_x + radius * math.cos(angle + 2 * math.pi / num_segments))
        end_y = int(center_y + radius * math.sin(angle + 2 * math.pi / num_segments))
        draw_line(start_x, start_y, end_x, end_y)

# Function to draw a filled circle
def draw_filled_circle(center_x, center_y, radius=RADIUS):
    num_triangles = 30
    for i in range(num_triangles):
        angle1 = i * 2 * math.pi / num_triangles
        angle2 = (i + 1) * 2 * math.pi / num_triangles
        start_x = int(center_x)
        start_y = int(center_y)
        x1 = int(center_x + radius * math.cos(angle1))
        y1 = int(center_y + radius * math.sin(angle1))
        x2 = int(center_x + radius * math.cos(angle2))
        y2 = int(center_y + radius * math.sin(angle2))
        pygame.draw.polygon(screen, (255, 255, 255), [(start_x, start_y), (x1, y1), (x2, y2)])

# Main loop
def main():
    running = True
    while running:
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                running = False

        screen.fill((0, 0, 0))  # Clear the screen
        draw_hollow_circle(WIDTH // 2, HEIGHT // 2)  # Draw a hollow circle
        draw_filled_circle(WIDTH // 4, HEIGHT // 4)  # Draw a filled circle
        pygame.display.flip()  # Update the display
        clock.tick(60)  # Cap the frame rate

    pygame.quit()

if __name__ == "__main__":
    main()
