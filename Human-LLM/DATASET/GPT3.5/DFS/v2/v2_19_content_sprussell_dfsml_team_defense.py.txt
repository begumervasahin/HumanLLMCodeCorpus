import nflgame

class TeamDefense:
    """
    A class representing defensive stats for an NFL team.
    """

    def __init__(self):
        # Initialize defensive stats attributes
        self.defense_sk = 0                   # Sacks
        self.defense_int = 0                  # Interceptions
        self.defense_frec = 0                 # Fumble recoveries
        self.kickret_tds = 0                  # Kick return touchdowns
        self.puntret_tds = 0                  # Punt return touchdowns
        self.defense_int_tds = 0              # Interception return touchdowns
        self.defense_frec_tds = 0             # Fumble recovery touchdowns
        self.defense_misc_tds = 0             # Miscellaneous defensive touchdowns
        self.defense_safe = 0                 # Safeties
        self.blocked_kicks = 0                # Blocked kicks

    def get_team_stats(self, players, team_abbr):
        """
        Calculate and store defensive stats for the specified team.

        :param players: NFL game players data.
        :param team_abbr: Team abbreviation (e.g., 'NE' for New England Patriots).
        """
        # Reset defensive stats
        self.__init__()

        # Iterate through defensive players of the specified team
        for player in players.defense().filter(team=team_abbr):
            # Accumulate defensive stats
            self.defense_sk += player.defense_sk
            self.defense_int += player.defense_int
            self.defense_frec += player.defense_frec
            self.kickret_tds += player.kickret_tds
            self.puntret_tds += player.puntret_tds
            self.defense_int_tds += player.defense_int_tds
            self.defense_frec_tds += player.defense_frec_tds
            self.defense_misc_tds += player.defense_misc_tds
            self.defense_safe += player.defense_safe

            # Accumulate blocked kicks
            self.blocked_kicks += player.defense_puntblk
            self.blocked_kicks += player.defense_xpblk
            self.blocked_kicks += player.defense_fgblk

if __name__ == "__main__":
    # Create an instance of the TeamDefense class
    team_defense = TeamDefense()

    # Get NFL game players data
    players_data = nflgame.combine_game_stats(nflgame.games(2019))

    # Specify team abbreviation
    team_abbreviation = 'NE'  # Example: New England Patriots

    # Get defensive stats for the specified team
    team_defense.get_team_stats(players_data, team_abbreviation)

    # Print the defensive stats
    print("Defensive Stats for Team {}: ".format(team_abbreviation))
    print("Sacks:", team_defense.defense_sk)
    print("Interceptions:", team_defense.defense_int)
    print("Fumble Recoveries:", team_defense.defense_frec)
    print("Kick Return Touchdowns:", team_defense.kickret_tds)
    print("Punt Return Touchdowns:", team_defense.puntret_tds)
    print("Interception Return Touchdowns:", team_defense.defense_int_tds)
    print("Fumble Recovery Touchdowns:", team_defense.defense_frec_tds)
    print("Miscellaneous Defensive Touchdowns:", team_defense.defense_misc_tds)
    print("Safeties:", team_defense.defense_safe)
    print("Blocked Kicks:", team_defense.blocked_kicks)
