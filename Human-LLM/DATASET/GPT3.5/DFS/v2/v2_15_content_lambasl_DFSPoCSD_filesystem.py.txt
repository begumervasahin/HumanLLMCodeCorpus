import logging
from collections import defaultdict
from errno import ENOENT, ENOTEMPTY
from stat import S_IFDIR, S_IFLNK, S_IFREG
from sys import argv, exit
from time import time
from fuse import FUSE, FuseOSError, Operations, LoggingMixIn

if not hasattr(__builtins__, 'bytes'):
    bytes = str

MaxBLOCKSIZE = 512

class Memory(LoggingMixIn, Operations):
    """Implements a hierarchical file system using FUSE virtual filesystem.
    File structure and data are stored in local memory.
    Data is lost when the filesystem is unmounted."""

    def __init__(self):
        self.files = {}
        self.data = defaultdict(bytes)
        self.fd = 0
        now = time()
        self.files['/'] = {
            'st_mode': (S_IFDIR | 0o755),
            'st_ctime': now,
            'st_mtime': now,
            'st_atime': now,
            'st_nlink': 2,
            'files': {}
        }

    def traverse(self, path, tdata=False):
        """Traverse the file structure to locate the current file."""
        p = self.data if tdata else self.files['/']
        if tdata:
            for i in path.split('/'):
                p = p[i] if i else p
        else:
            for i in path.split('/'):
                p = p['files'][i] if i else p
        return p

    def traverseparent(self, path, tdata=False):
        """Traverse to the parent directory of the current file."""
        p = self.data if tdata else self.files['/']
        target = path[path.rfind('/') + 1:]
        path = path[:path.rfind('/')]
        if tdata:
            for i in path.split('/'):
                p = p[i] if i else p
        else:
            for i in path.split('/'):
                p = p['files'][i] if i else p
        return p, target

    # (chown, chmod, create, getattr, getxattr, listxattr, mkdir, open, read,
    # readdir, readlink, removexattr, rename, rmdir, setxattr, statfs, symlink,
    # truncate, unlink, utimens, write)

if __name__ == '__main__':
    if len(argv) != 2:
        print('usage: %s <mountpoint>' % argv[0])
        exit(1)
    logging.basicConfig(level=logging.DEBUG)
    fuse = FUSE(Memory(), argv[1], foreground=True)
