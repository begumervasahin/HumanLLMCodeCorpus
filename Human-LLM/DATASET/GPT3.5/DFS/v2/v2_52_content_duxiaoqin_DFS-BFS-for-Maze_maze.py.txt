from random import randint
from myarray2d import Array2D  # Assuming Array2D is a custom class provided elsewhere

class Maze:
    EMPTY = 0
    OBSTACLE = -1
    OCCUPIED = 1

    def __init__(self, height, width):
        self.height = height
        self.width = width
        self.maze = Array2D(height, width)
        self.start = (0, 0)
        self.goal = (height - 1, width - 1)
        self.clear_maze()
        self.generate_obstacles()

    def clear_maze(self):
        self.maze.clear(Maze.EMPTY)

    def generate_obstacles(self):
        obstacle_count = int(self.height * self.width * 0.1)
        for _ in range(obstacle_count):
            row = randint(0, self.height - 1)
            col = randint(0, self.width - 1)
            if (row, col) == self.start or (row, col) == self.goal:
                continue
            self.maze[row, col] = Maze.OBSTACLE

    def get_all_moves(self, row, col):
        moves = []
        offsets = [(0, -1), (-1, 0), (1, 0), (0, 1)]
        for dx, dy in offsets:
            new_row = row + dy
            new_col = col + dx
            if 0 <= new_row < self.height and 0 <= new_col < self.width \
                    and self.maze[new_row, new_col] != Maze.OBSTACLE:
                moves.append((new_row, new_col))
        return moves

    def num_rows(self):
        return self.height

    def num_cols(self):
        return self.width

    def print_maze(self):
        for row in range(self.height):
            for col in range(self.width):
                if self.maze[row, col] == Maze.EMPTY:
                    print('_', end=' ')
                elif self.maze[row, col] == Maze.OBSTACLE:
                    print('|', end=' ')
                else:
                    print('O', end=' ')
            print()

def main():
    maze = Maze(20, 20)
    maze.print_maze()
    print()
    print(maze.get_all_moves(3, 3))

if __name__ == '__main__':
    main()
