import pandas as pd
import requests
from bs4 import BeautifulSoup as bsoup

def grab_data_and_save_csv():
    # Iterate through years and weeks
    for year in [2016, 2015, 2014, 2013]:
        for week in range(1, 17):
            # Construct the URL for each week and year
            url = f'http://rotoguru1.com/cgi-bin/fyday.pl?week={week}&year={year}&game=fd&scsv=1'
            # Send a GET request to the URL
            response = requests.get(url)
            # Create a BeautifulSoup object to parse the HTML content
            soup = bsoup(response.content, 'html.parser')
            # Find all 'pre' tags in the HTML content
            pre_tags = soup.find_all('pre')
            # Open the CSV file in append mode
            with open('test.csv', 'a+') as csv_file:
                # Iterate through the content of the first 'pre' tag
                for content in pre_tags[0]:
                    # Write the content to the CSV file
                    csv_file.write(str(content))

if __name__ == '__main__':
    # Call the function to grab data and save to CSV
    grab_data_and_save_csv()
