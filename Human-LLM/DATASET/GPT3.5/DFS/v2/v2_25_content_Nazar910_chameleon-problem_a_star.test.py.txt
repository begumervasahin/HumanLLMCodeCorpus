import unittest
from a_star import a_star
from state import State

class TestAStar(unittest.TestCase):
    def test_result(self):
        print('=======================')
        print('A* algorithm')
        print('=======================')
        
        # Define initial state
        initial_state = State(red=13, blue=16, green=17)
        
        # Execute A* algorithm
        result = a_star(initial_state)
        
        # Extract final state from result
        final_state = result['final_state']
        
        # Assertions for final state
        self.assertEqual(final_state.red_count, 0)
        self.assertEqual(final_state.green_count, 46)
        self.assertEqual(final_state.blue_count, 0)
        
        # Extract full path and path from result
        full_path = result['full_path']
        path = result['path']
        
        # Print lengths of full path and path
        print(f'Full path length: {len(full_path)}')
        print(f'Path length is {len(path)}')
        
        # Assertions for lengths of full path and path
        self.assertEqual(len(full_path), 3295)
        self.assertEqual(len(path), 17)
        
        # Print nodes in the path
        for i, node in enumerate(path, start=1):
            print(f'{i}:{node}')

if __name__ == "__main__":
    unittest.main()
