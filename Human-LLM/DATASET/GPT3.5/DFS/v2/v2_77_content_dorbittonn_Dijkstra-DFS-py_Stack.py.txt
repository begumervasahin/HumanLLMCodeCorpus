class Node:
    def __init__(self, value):
        self.value = value
        self.next = None

class Stack:
    def __init__(self):
        # Initialize the stack with a head node
        self.head = Node("head")
        self.size = 0

    def __str__(self):
        # Convert the stack to a string for printing
        cur = self.head.next
        stack_str = ""
        while cur:
            stack_str += str(cur.value) + "->"
            cur = cur.next
        return stack_str[:-2]  # Remove the extra "->" at the end

    def getSize(self):
        # Get the size of the stack
        return self.size

    def isEmpty(self):
        # Check if the stack is empty
        return self.size == 0

    def Top(self):
        # Get the value of the top element of the stack
        if self.isEmpty():
            raise Exception("Peeking from an empty stack")
        return self.head.next.value

    def push(self, value):
        # Push a new element onto the stack
        new_node = Node(value)
        new_node.next = self.head.next
        self.head.next = new_node
        self.size += 1

    def pop(self):
        # Remove and return the top element from the stack
        if self.isEmpty():
            raise Exception("Popping from empty stack")
        remove = self.head.next
        self.head.next = remove.next
        self.size -= 1
