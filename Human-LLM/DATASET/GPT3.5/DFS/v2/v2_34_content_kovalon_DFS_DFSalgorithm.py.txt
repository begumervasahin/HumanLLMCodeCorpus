# Function to print the graph
def printGraph(graph):
    for node, data in graph.items():
        print(f"Node {node}: {data[0]}")
    print()

# Depth-First Search (DFS) resolver function
def dfsResolver(key):
    time.sleep(1)  # Simulate processing time
    mathGraph[key][1] = True
    printGraph(mathGraph)
    
    for count, edge in enumerate(mathGraph[key][0], start=1):
        if not mathGraph[count][1] and edge:
            dfsResolver(count)

# Start DFS from node 1
dfsResolver(1)
print("DFS Complete!")
