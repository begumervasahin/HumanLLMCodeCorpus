from collections import defaultdict, deque

class Graph:
    def __init__(self):
        """Initialize an empty graph."""
        self.graph = defaultdict(list)

    def add_edge(self, u, v):
        """Add an edge between vertices u and v."""
        self.graph[u].append(v)

    def bfs(self, start_vertex):
        """Perform Breadth First Traversal starting from the specified vertex."""
        visited = defaultdict(bool)  # Keep track of visited vertices
        distance = defaultdict(int)  # Keep track of distances from the start vertex
        queue = deque()  # Initialize a queue for BFS
        queue.append(start_vertex)  # Enqueue the start vertex
        visited[start_vertex] = True  # Mark start vertex as visited
        distance[start_vertex] = 0  # Set distance of start vertex to 0

        while queue:
            current_vertex = queue.popleft()  # Dequeue a vertex
            print(current_vertex, end=" ")  # Print the vertex being visited

            # Visit all neighbors of the current vertex
            for neighbor in self.graph[current_vertex]:
                if not visited[neighbor]:
                    queue.append(neighbor)  # Enqueue unvisited neighbors
                    visited[neighbor] = True  # Mark neighbor as visited
                    distance[neighbor] = distance[current_vertex] + 1  # Update distance

    graph = Graph()
