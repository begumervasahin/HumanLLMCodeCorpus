# Importing the socket module
import socket

# Define the TCP server's IP address, port, buffer size, and message
TCP_IP = '52.43.121.77'
TCP_PORT = 2329
BUFFER_SIZE = 1024
MESSAGE = "Hello, World!"

# Create a socket object
socket_object = socket.socket(socket.AF_INET, socket.SOCK_STREAM)

# Connect to the server
socket_object.connect((TCP_IP, TCP_PORT))
print('Connected to the server')

# Send the message
socket_object.send(MESSAGE.encode())

# Receive data from the server
received_data = socket_object.recv(BUFFER_SIZE)

# Close the connection
socket_object.close()

# Print the received data
print("Received data:", received_data.decode())
