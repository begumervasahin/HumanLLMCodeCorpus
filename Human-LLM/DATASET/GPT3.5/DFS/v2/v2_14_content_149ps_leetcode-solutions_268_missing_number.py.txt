def missing_number(nums):
    # Determine the lower bound (l), upper bound (u), and range length
    l = 0
    u = max(nums)
    length = u - l + 1  # Range of numbers from l to u inclusive
    print("Lower Bound:", l)
    print("Upper Bound:", u)
    print("Range Length:", length)

    # Iterate through the range from l to u
    i = 0
    while length != 0:
        # Check if the number l+i is present in the list nums
        if l + i in nums:
            i += 1
            length -= 1
        else:
            # If the number l+i is missing, return it as the missing number
            return l + i
    # If nums is empty or contains a complete range, return the upper bound + 1
    return l + i

# Example usage
nums = [0]  # List containing numbers
print("Missing Number:", missing_number(nums))
