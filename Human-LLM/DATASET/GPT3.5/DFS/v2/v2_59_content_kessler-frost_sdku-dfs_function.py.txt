# Import necessary functions from the 'utils' module
from utils import grid_values, search, display

# Define the Sudoku puzzle as a string
puzzle_string = '4.....8.5.3..........7......2.....6.....8.4......1.......6.3.7.5..2.....1.4......'

# Convert the puzzle string into a Sudoku grid using the grid_values function
initial_grid = grid_values(puzzle_string)

# Solve the Sudoku puzzle using the search function
solved_grid = search(initial_grid)

# Display the solved Sudoku puzzle using the display function
display(solved_grid)
