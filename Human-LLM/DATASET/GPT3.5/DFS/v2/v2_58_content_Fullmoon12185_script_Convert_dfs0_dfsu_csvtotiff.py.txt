import os
from glob import glob
from osgeo import gdal

# Set the directory containing CSV files
csv_directory = r"C:/anhHoangKTTV/PythonScripts_DFS0/Project_NTB/"
os.chdir(csv_directory)

def find_csv_files(directory, suffix=".csv"):
    """
    Find all CSV files in the specified directory.
    """
    files_pattern = os.path.join(directory, "*" + suffix)
    filenames = [os.path.relpath(file) for file in glob(files_pattern)]
    return filenames

# Get a list of CSV files in the directory
csv_files = find_csv_files(csv_directory)

# Process each CSV file
for csv_file in csv_files:
    vrt_file = csv_file.replace(".csv", ".vrt")
    layer_name = os.path.splitext(os.path.basename(csv_file))[0]
    tiff_file = csv_file.replace('.csv', '.tiff')

    # Create a VRT file for the CSV file
    with open(vrt_file, 'w') as vrt_handle:
        vrt_handle.write('<OGRVRTDataSource>\n')
        vrt_handle.write('\t<OGRVRTLayer name="%s">\n' % layer_name)
        vrt_handle.write('\t\t<SrcDataSource>%s</SrcDataSource>\n' % csv_file)
        vrt_handle.write('\t\t<GeometryType>wkbPoint</GeometryType>\n')
        vrt_handle.write('\t\t<GeometryField encoding="PointFromColumns" x="Lon" y="Lat" z="Ref"/>\n')
        vrt_handle.write('\t</OGRVRTLayer>\n')
        vrt_handle.write('</OGRVRTDataSource>\n')

    # Convert the VRT file to a TIFF file using gdal.Grid
    gdal.Grid(tiff_file, vrt_file)
    print("Generated TIFF file:", tiff_file)

# Generate outcome2.tif using gdal.Grid with specific parameters
gdal.Grid('outcome2.tif', 'name.vrt', algorithm='invdist:power=2.0:smoothing=1.0')
print("Generated outcome2.tif")
