class Solution:
    def postorderTraversal(self, root):
        """
        Traverse a binary tree using postorder traversal.

        :param root: TreeNode, the root node of the binary tree.
        :return: List[int], containing the postorder traversal of the tree.
        """
        result = []  # List to store the traversal result
        stack = []   # Stack for iterative traversal
        node = root  # Start from the root node
        visited = set()  # Set to keep track of visited nodes

        while node or stack:
            # Traverse left subtree until reaching the leftmost leaf
            if node:
                stack.append(node)
                node = node.left
            
            # Process the right subtree and backtrack if necessary
            else:
                node = stack.pop()
                
                # Check if right subtree exists and hasn't been visited yet
                if node.right and node.right not in visited:
                    stack.append(node)  # Push the current node back to stack
                    node = node.right  # Traverse the right subtree
                
                # If no right subtree or already visited, add to result
                else:
                    visited.add(node)   # Mark the current node as visited
                    result.append(node.val)  # Add node value to result
                    node = None  # Reset node to None to backtrack to parent
        
        return result
