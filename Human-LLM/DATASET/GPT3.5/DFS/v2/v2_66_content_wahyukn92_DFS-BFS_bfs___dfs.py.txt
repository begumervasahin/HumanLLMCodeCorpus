# Breadth-First Search (BFS) algorithm
def bfs(graph, start, end):
    # Initialize queue with the start node
    queue = [[start]]
    # Initialize set to keep track of visited nodes
    visited = set()

    while queue:
        # Get the first route from the queue
        route = queue.pop(0)
        # Get the last node of the route
        node = route[-1]

        # If the last node is the end node, return the route
        if node == end:
            return route

        # If the node has not been visited yet
        if node not in visited:
            # Mark the node as visited
            visited.add(node)
            # Get neighbors of the current node
            neighbors = graph.get(node, [])
            # Add new routes to the queue
            for neighbor in neighbors:
                new_route = route + [neighbor]
                queue.append(new_route)
    
    # If no route is found, print a message
    print("Not found, try again")

# Depth-First Search (DFS) algorithm
def dfs(graph, start, end):
    # Initialize stack with the start node
    stack = [[start]]
    # Initialize set to keep track of visited nodes
    visited = set()

    while stack:
        # Get the last route from the stack
        route = stack.pop()
        # Get the last node of the route
        node = route[-1]

        # If the last node is the end node, return the route
        if node == end:
            return route

        # If the node has not been visited yet
        if node not in visited:
            # Mark the node as visited
            visited.add(node)
            # Get neighbors of the current node
            neighbors = graph.get(node, [])
            # Add new routes to the stack
            for neighbor in neighbors:
                new_route = route + [neighbor]
                stack.append(new_route)

    # If no route is found, print a message
    print("Not found, try again")

# Define the spiders graph
spiders = {
    '1': ['2', '3', '4'],
    '2': ['3', '8'],
    '3': ['4', '5', '6', '7', '8'],
    '4': ['5'],
    '5': ['6'],
    '6': ['7', '11'],
    '7': ['8', '9', '10', '11'],
    '8': ['9'],
    '9': ['10'],
    '10': ['11']
}

# Define start and end nodes
start_node = '1'
end_node = '11'

# Print routes found by BFS and DFS algorithms
print("BFS route:", bfs(spiders, start_node, end_node))
print("DFS route:", dfs(spiders, start_node, end_node))
