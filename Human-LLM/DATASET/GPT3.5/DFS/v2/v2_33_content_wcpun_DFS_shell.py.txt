import cmd

class MyCLI(cmd.Cmd):
    def __init__(self):
        # Initialize the base class (cmd.Cmd)
        super().__init__()
        # Initialize the current_path attribute to an empty string
        self.current_path = ''
        # Set the command prompt string
        self.prompt = "FYP >>> "

    def do_hi(self, arg):
        """Command to print a greeting message."""
        print("Hello World.")

def main():
    # Create an instance of MyCLI
    cli = MyCLI()
    # Start the command loop
    cli.cmdloop()

if __name__ == '__main__':
    # Entry point of the script
    main()
