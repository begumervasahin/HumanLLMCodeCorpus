from queue import Queue, PriorityQueue
from math import sqrt, pow, abs

class Node:
    def __init__(self, value):
        self.value = value
        self.id = None
        self.up = None
        self.down = None
        self.left = None
        self.right = None
        self.previousNode = None
        self.edgeCost = None
        self.gOfN = None
        self.hOfN = None
        self.heuristicFn = None
        self.visited = False

class SearchAlgorithms:
    def __init__(self, mazeStr, edgeCost=None):
        self.grid = []
        self.list_of_rows = []
        self.fullPath = []
        self.stack = []
        self.myqueue = Queue()
        self.pqueue = PriorityQueue()
        self.path = []
        self.totalCost = -1
        self.id_counter = 0
        self.start_id = 0
        self.row_number = 0
        self.col_number = 0
        self.cost_counter = 0
        self.end_x = 0
        self.end_y = 0

        self.initialize_grid(mazeStr)
        self.initialize_nodes(edgeCost)

    def initialize_grid(self, mazeStr):
        self.list_of_rows = mazeStr.replace(',', '').split()
        self.row_number = len(self.list_of_rows)
        self.col_number = len(self.list_of_rows[0])

        for row in self.list_of_rows:
            self.grid.append([Node(char) for char in row])

        for i in range(self.row_number):
            for j in range(self.col_number):
                node = self.grid[i][j]
                node.id = self.id_counter
                self.id_counter += 1

                if node.value == 'S':
                    self.start_id = node.id
                elif node.value == 'E':
                    self.end_x = i
                    self.end_y = j

                if i > 0 and self.grid[i-1][j].value != '':
                    node.up = self.grid[i-1][j].id
                if i < self.row_number - 1 and self.grid[i+1][j].value != '':
                    node.down = self.grid[i+1][j].id
                if j > 0 and self.grid[i][j-1].value != '':
                    node.left = self.grid[i][j-1].id
                if j < self.col_number - 1 and self.grid[i][j+1].value != '':
                    node.right = self.grid[i][j+1].id

    def initialize_nodes(self, edgeCost):
        if edgeCost is not None:
            for i in range(self.row_number):
                for j in range(self.col_number):
                    if self.grid[i][j].value != '':
                        self.grid[i][j].edgeCost = edgeCost[self.cost_counter]
                        self.cost_counter += 1

    def get_flatten(self, id):
        row = id // self.col_number
        col = id % self.col_number
        return row, col

    def heuristic_euclidean_distance(self, i, j):
        return sqrt(pow((self.end_x - i), 2) + pow((self.end_y - j), 2))

    def heuristic_absolute(self, i, j):
        return abs(i - self.end_x) + abs(j - self.end_y)

    def get_children(self, row_idx, col_idx):
        children = []
        node = self.grid[row_idx][col_idx]
        node.visited = True

        if node.right is not None:
            children.append(node.right)
        if node.left is not None:
            children.append(node.left)
        if node.down is not None:
            children.append(node.down)
        if node.up is not None:
            children.append(node.up)

        return children

    def DFS(self):
        self.stack = [self.start_id]

        while self.stack:
            current_id = self.stack.pop()
            row_idx, col_idx = self.get_flatten(current_id)
            current_node = self.grid[row_idx][col_idx]

            if current_node.value == "E":
                self.fullPath.append(current_id)
                break

            if not current_node.visited and current_id not in self.stack:
                self.fullPath.append(current_id)
                current_node.visited = True
                children = self.get_children(row_idx, col_idx)
                self.stack.extend(children)

        return self.path, self.fullPath

    def BFS(self):
        self.myqueue.put(self.start_id)

        while not self.myqueue.empty():
            current_id = self.myqueue.get()
            row_idx, col_idx = self.get_flatten(current_id)
            current_node = self.grid[row_idx][col_idx]

            if current_node.value == "E":
                self.fullPath.append(current_id)
                break

            if not current_node.visited:
                self.fullPath.append(current_id)
                current_node.visited = True
                children = self.get_children(row_idx, col_idx)
                self.myqueue.queue.extend(children)

        return self.path, self.fullPath

    def UCS(self):
        self.pqueue.put((0, self.start_id))

        while not self.pqueue.empty():
            cost, current_id = self.pqueue.get()
            row_idx, col_idx = self.get_flatten(current_id)
            current_node = self.grid[row_idx][col_idx]

            if not current_node.visited:
                self.fullPath.append(current_id)

                if current_node.value == 'E':
                    break

                children = self.get_children(row_idx, col_idx)
                children.reverse()
                current_node.visited = True

                for child_id in children:
                    child_row, child_col = self.get_flatten(child_id)
                    child_node = self.grid[child_row][child_col]

                    if child_id not in self.fullPath:
                        child_gOfN = cost + child_node.edgeCost
                        self.totalCost = child_gOfN
                        self.pqueue.put((self.totalCost, child_id))
                        child_node.gOfN = child_gOfN

        return self.path, self.fullPath, self.totalCost

    def AStarEuclideanHeuristic(self):
        for i in range(self.row_number):
            for j in range(self.col_number):
                self.grid[i][j].hOfN = self.heuristic_euclidean_distance(i, j)

        self.pqueue.put((0, self.start_id))
        row_idx, col_idx = self.get_flatten(self.start_id)
        self.grid[row_idx][col_idx].gOfN = 0

        while not self.pqueue.empty():
            _, current_id = self.pqueue.get()
            row_idx, col_idx = self.get_flatten(current_id)
            cost = self.grid[row_idx][col_idx].gOfN

            if not self.grid[row_idx][col_idx].visited:
                self.fullPath.append(current_id)

                if self.grid[row_idx][col_idx].value == 'E':
                    break

                children = self.get_children(row_idx, col_idx)
                children.reverse()
                self.grid[row_idx][col_idx].visited = True

                for child_id in children:
                    if child_id not in self.fullPath:
                        child_row, child_col = self.get_flatten(child_id)
                        child_node = self.grid[child_row][child_col]
                        child_node.gOfN = cost + child_node.edgeCost
                        self.totalCost = child_node.gOfN
                        child_node.heuristicFn = child_node.gOfN + child_node.hOfN
                        self.pqueue.put((child_node.heuristicFn, child_id))

        return self.path, self.fullPath, self.totalCost

    def AStarManhattanHeuristic(self):
        for i in range(self.row_number):
            for j in range(self.col_number):
                self.grid[i][j].hOfN = self.heuristic_absolute(i, j)

        self.pqueue.put((0, self.start_id))
        row_idx, col_idx = self.get_flatten(self.start_id)
        self.grid[row_idx][col_idx].gOfN = 0

        while not self.pqueue.empty():
            _, current_id = self.pqueue.get()
            row_idx, col_idx = self.get_flatten(current_id)
            cost = self.grid[row_idx][col_idx].gOfN

            if not self.grid[row_idx][col_idx].visited:
                self.fullPath.append(current_id)

                if self.grid[row_idx][col_idx].value == 'E':
                    break

                children = self.get_children(row_idx, col_idx)
                children.reverse()
                self.grid[row_idx][col_idx].visited = True

                for child_id in children:
                    if child_id not in self.fullPath:
                        child_row, child_col = self.get_flatten(child_id)
                        child_node = self.grid[child_row][child_col]
                        child_node.gOfN = cost + child_node.edgeCost
                        self.totalCost = child_node.gOfN
                        child_node.heuristicFn = child_node.gOfN + child_node.hOfN
                        self.pqueue.put((child_node.heuristicFn, child_id))

        return self.path, self.fullPath, self.totalCost

def main():
    searchAlgo = SearchAlgorithms('S,.,.,\n.,E,.\n.,.,.')
    path, fullPath = searchAlgo.DFS()
    print('**DFS**\nPath is:', path, '\nFull Path is:', fullPath, '\n\n')

    searchAlgo = SearchAlgorithms('S,.,.,\n.,E,.\n.,.,.')
    path, fullPath = searchAlgo.BFS()
    print('**BFS**\nPath is:', path, '\nFull Path is:', fullPath, '\n\n')

    searchAlgo = SearchAlgorithms('S,.,.,\n.,E,.\n.,.,.', [100, 2, 15, 60, 100, 30, 2, 100, 2, 2, 2, 40, 30, 2, 2, 100, 100, 3, 15, 30, 100, 2, 100, 0, 2, 100, 30])
    path, fullPath, TotalCost = searchAlgo.UCS()
    print('**UCS**\nPath is:', path, '\nFull Path is:', fullPath, '\nTotal Cost:', TotalCost, '\n\n')

    searchAlgo = SearchAlgorithms('S,.,.,\n.,E,.\n.,.,.', [100, 2, 15, 60, 100, 30, 2, 100, 2, 2, 2, 40, 30, 2, 2, 100, 100, 3, 15, 30, 100, 2, 100, 0, 2, 100, 30])
    path, fullPath, TotalCost = searchAlgo.AStarEuclideanHeuristic()
    print('**ASTAR with Euclidean Heuristic**\nPath is:', path, '\nFull Path is:', fullPath, '\nTotal Cost:', TotalCost, '\n\n')

    searchAlgo = SearchAlgorithms('S,.,.,\n.,E,.\n.,.,.')
    path, fullPath, TotalCost = searchAlgo.AStarManhattanHeuristic()
    print('**ASTAR with Manhattan Heuristic**\nPath is:', path, '\nFull Path is:', fullPath, '\nTotal Cost:', TotalCost, '\n\n')

if __name__ == "__main__":
    main()
