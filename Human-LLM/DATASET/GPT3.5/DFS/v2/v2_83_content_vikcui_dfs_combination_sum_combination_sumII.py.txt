def combination_sum(candidates, target):
    # Sort the candidate numbers in ascending order
    candidates.sort()
    # Initialize an empty list to store the result combinations
    result = []
    # Start finding combinations from index 0
    start_index = 0
    # Initialize an empty list to store the current combination sequence
    current_sequence = []
    # Call the helper function to find combinations
    combination_sum_helper(candidates, target, start_index, current_sequence, result)
    # Return the list of all valid combinations
    return result

def combination_sum_helper(candidates, target, start_index, current_sequence, result):
    # Iterate through the candidate numbers starting from the specified index
    for i in range(start_index, len(candidates)):
        # Skip duplicate candidate numbers to avoid duplicate combinations
        if i > start_index and candidates[i] == candidates[i - 1]:
            continue
        num = candidates[i]
        # If the current number exceeds the target, terminate the loop
        if num > target:
            return
        # Add the current number to the current sequence
        current_sequence.append(num)
        # If the current number equals the target, add the current sequence to the result list
        if num == target:
            result.append(list(current_sequence))
        # If the current number is less than the target, recursively search for combinations
        else:
            combination_sum_helper(candidates, target - num, i + 1, current_sequence, result)
        # Backtrack by removing the last added number from the current sequence
        current_sequence.pop()

# Example usage:
candidates = [10, 1, 2, 7, 6, 1, 5]
target = 8
print(combination_sum(candidates, target))  # Output: [[1, 1, 6], [1, 2, 5], [1, 7], [2, 6]]
