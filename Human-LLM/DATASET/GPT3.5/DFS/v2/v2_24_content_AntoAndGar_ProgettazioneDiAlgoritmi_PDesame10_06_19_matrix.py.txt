def pattern_count(n):
    # Define the set size and initialize the table
    num_elements = len({'1', '2', '3', '4'})
    table = [[0 for _ in range(num_elements)] for _ in range(n + 1)]

    # Populate the table using dynamic programming
    for i in range(n + 1):
        for j in range(num_elements):
            if i == 0:
                table[i][j] = 0
            elif i == 1:
                table[i][j] = 1
            elif i == 2:
                if j == 0:
                    table[i][j] = 3
                if j == 1 or j == 2:
                    table[i][j] = 2
                if j == 3:
                    table[i][j] = 3
            else:
                if j == 0:
                    table[i][j] = table[i - 1][j] + table[i - 1][j + 2] + table[i - 1][j + 3]
                if j == 1:
                    table[i][j] = table[i - 1][j] + table[i - 1][j + 2]
                if j == 2:
                    table[i][j] = table[i - 1][j] + table[i - 1][j - 2]
                if j == 3:
                    table[i][j] = table[i - 1][j] + table[i - 1][j - 2] + table[i - 1][j - 3]

    # Return the result for n
    return table[n]

# Example usage
n = 6
print(pattern_count(n))
