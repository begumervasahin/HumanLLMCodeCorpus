import pandas as pd
from db import actor

# SQL queries
select_stat_lines_by_date = """
    SELECT
        player_id AS pid,
        fd_name AS name,
        fd_positions AS pos,
        fd_salary AS sal,
        fd_points AS pts
    FROM
        stat_lines sl,
        games g,
        players p
    WHERE
        sl.game_id = g.id
        AND sl.player_id = p.id
        AND g.date='%s'
        AND sl.fd_positions IS NOT NULL
        AND sl.fd_points IS NOT NULL
        AND sl.player_id NOT IN %s
        AND sl.active;
"""

select_main_str = """
    SELECT
        SUM(fd_points) AS points,
        SUM(fd_salary) AS salary
    FROM
        stat_line_points
    WHERE
        player_id IN %s
        AND date='%s';
"""

query_string_self_projection = '''
    SELECT
        slp.player_id AS pid,
        slp.name AS name,
        slp.fd_positions AS pos,
        slp.fd_salary AS sal,
        proj.fd_points AS pts
    FROM
        stat_line_points slp,
        projections proj
    WHERE
        proj.stat_line_id = slp.stat_line_id
        AND proj.version = %s
        AND slp.date = %s
        AND slp.fd_positions IS NOT NULL
        AND proj.fd_points IS NOT NULL
        AND proj.minutes > 0.0
        AND slp.player_id NOT IN %s
        AND slp.active
        AND proj.active;
'''

# Functions
def get_actual_points_sal_for_ids(player_ids, date):
    if not player_ids:
        raise ValueError('No player ids.')

    query = select_main_str % (tuple(player_ids), date)
    cursor = actor.cursor
    cursor.execute(query)
    result = cursor.fetchone()
    print('actual info')
    print('done actual info')
    points = result['points']
    salary = result['salary']
    if points is not None:
        points = float(points)
    return points, salary

def get_stat_lines_for_date(date, version, exclude=[0]):
    params = (version, date, tuple(exclude),)
    df = pd.read_sql_query(query_string_self_projection, actor.conn, params=params)
    print(df)
    return df
