import getopt
import sys
import pickle
from xmlrpc.client import Binary
import xmlrpc.client

class Helper:
    def __init__(self, rpc_server):
        self.rpc_server = rpc_server

    def put(self, key, value, time_to_live):
        return self.rpc_server.put(Binary(key), Binary(value), time_to_live)

    def get(self, key):
        return self.rpc_server.get(Binary(key))

    def write_file(self, filename):
        return self.rpc_server.write_file(Binary(filename))

    def read_file(self, filename):
        return self.rpc_server.read_file(Binary(filename))

    def list_methods(self):
        return self.rpc_server.system.listMethods()

def main():
    try:
        options, arguments = getopt.getopt(sys.argv[1:], "", ["port="])
    except getopt.GetoptError as error:
        print(str(error))
        sys.exit(2)

    parsed_options = {}
    for option, value in options:
        parsed_options[option] = value

    port = 51234
    if "--port" in parsed_options:
        port = int(parsed_options["--port"])

    server_url = "http://127.0.0.1:" + str(port)
    rpc_client = xmlrpc.client.Server(server_url)
    helper = Helper(rpc_client)

    data = {'a': 'a', 'files_name': 'd'}
    helper.put("/hello.c", pickle.dumps(data), 10000)

    retrieved_data = pickle.loads(helper.get("/hello.c").data)
    if retrieved_data == data:
        print("Data insertion was successful")

if __name__ == "__main__":
    main()
