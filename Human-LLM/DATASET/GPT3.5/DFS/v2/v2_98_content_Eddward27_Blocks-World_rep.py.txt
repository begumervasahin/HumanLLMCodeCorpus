class Cubo:
    def __init__(self, nombre, on=None, clear=False, onTable=True):
        self.nombre = nombre
        self.on = on
        self.clear = clear
        self.onTable = onTable

def set_world(data):
    initial_state = set_estado(data['initialState'])
    final_state = set_estado(data['finalState'])
    world = {'initial': initial_state, 'final': final_state}
    return world

def set_estado(tower_arrangement):
    state = {'operators': [], 'states': [], 'previousStates': []}
    for tower in tower_arrangement:
        tower_list = list(tower)
        for i in range(len(tower_list)):
            cube_name = tower_list[i]
            cube_on = None
            cube_clear = False
            cube_on_table = True
            if i < len(tower_list) - 1:
                cube_on = tower_list[i + 1]
                cube_on_table = False
            if i == 0:
                cube_clear = True
            state['states'].append(Cubo(cube_name, cube_on, cube_clear, cube_on_table))
    state['previousStates'].append(state_to_string(state['states']))
    return state

def same_state(state1, state2):
    towers1 = len(state1)
    towers2 = len(state2)
    if towers1 != towers2:
        return False
    for cube1 in state1:
        state_found = False
        for cube2 in state2:
            if cube1.nombre == cube2.nombre:
                if (cube1.on == cube2.on and cube1.onTable == cube2.onTable and cube1.clear == cube2.clear):
                    state_found = True
                    break
        if not state_found:
            return False
    return True

def state_to_string(state):
    state_str = []
    for cube_rep in state:
        if cube_rep.clear:
            state_str.append(cube_rep.nombre)
    for i in range(len(state_str)):
        not_last = True
        current_cube = state_str[i]
        while not_last:
            for cube in state:
                if current_cube == cube.nombre:
                    if cube.on is None:
                        not_last = False
                        break
                    state_str[i] += cube.on
                    current_cube = cube.on
                    break
    return state_str

# Example usage:
data = {
    'initialState': ['ABCD', 'EF'],
    'finalState': ['AB', 'CDEF']
}

world = set_world(data)
print(world)
