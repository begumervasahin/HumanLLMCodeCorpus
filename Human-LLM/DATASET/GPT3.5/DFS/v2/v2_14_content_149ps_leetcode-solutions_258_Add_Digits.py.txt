class Solution:
    def addDigits(self, num):
        # Continuously sum the digits until the result becomes a single digit
        while num >= 10:
            num = (num // 10) + (num % 10)
        return num

# Example usage:
# Create an instance of the Solution class
solution = Solution()

# Call the addDigits method to obtain the result
num = 12345
result = solution.addDigits(num)
print("Result:", result)
