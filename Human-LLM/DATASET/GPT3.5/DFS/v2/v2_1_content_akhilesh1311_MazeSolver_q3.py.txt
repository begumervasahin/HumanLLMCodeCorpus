from maze import Maze
from maze_solver import DFSSolver
import matplotlib.pyplot as plt
import time
import pickle

# Define probability values and maze dimensions
pvals = [x / 10 for x in range(1, 10)]
dim = 1000
solvable_counts = {p: 0 for p in pvals}

# Iterate over different probabilities and solve mazes
for p in pvals:
    count = 0
    for i in range(100):
        # Create a maze and DFS solver
        maze = Maze(dim, p)
        solver = DFSSolver(maze)
        
        # Solve the maze and measure solving time
        start_time = time.time()
        solver.display_solution(display=False)
        solving_time = time.time() - start_time
        
        # Check if the maze is solvable
        if solver.is_solvable:
            solvable_counts[p] += 1
            count += 1
            
            # Save the solved maze as a pickle file
            with open(f"../mazes/{dim}x{dim}_{p}_{count:03}.p", 'wb') as f:
                pickle.dump(maze, f, pickle.HIGHEST_PROTOCOL)
            
            print(f"Maze ({maze.dim}, {maze.p}) solved in {solving_time:.2f} seconds.")
        else:
            print(f"Maze ({maze.dim}, {maze.p}) is not solvable.")
    
    print(f"For p={p}, {solvable_counts[p]} out of 100 mazes were solvable.")

# Plot the solvability rate vs. probability values
counts = [solvable_counts[p] for p in pvals]
plt.plot(pvals, counts)
plt.xlabel("Probability (p-value)")
plt.ylabel("Number of Solvable Mazes")
plt.title("Solvability Rate vs. Probability")
plt.savefig("solvability_vs_probability.png")
plt.show()
