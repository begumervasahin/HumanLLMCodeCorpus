from breadth_first_search import bfs
from state import State

def test_bfs_algorithm():
    print("=======================")
    print("BFS Algorithm Test")
    print("=======================")
    
    # Define the initial state of the problem
    initial_state = State(red=13, blue=16, green=17)
    
    # Apply the Breadth-First Search algorithm to solve the problem
    result = bfs(initial_state)
    
    # Extract the final state from the result
    final_state = result['final_state']
    
    # Verify that the final state meets the expected conditions
    assert final_state.red_count == 0
    assert final_state.green_count == 46
    assert final_state.blue_count == 0
    
    # Extract the full path and the solution path from the result
    full_path = result['full_path']
    path = result['path']
    
    # Print the lengths of the full path and the solution path
    print(f"Length of the full path: {len(full_path)}")
    print(f"Length of the solution path: {len(path)}")
    
    # Verify the lengths of the full path and the solution path
    assert len(full_path) == 180169
    assert len(path) == 17
    
    # Print the nodes in the solution path
    print("Nodes in the Solution Path:")
    for i, node in enumerate(path, start=1):
        print(f"Step {i}: {node}")

if __name__ == "__main__":
    # Run the test function if this script is executed directly
    test_bfs_algorithm()
