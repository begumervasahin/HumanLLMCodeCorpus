import collections

class Solution:
    def intersect(self, nums1, nums2):
        # Initialize an empty list to store the intersection
        intersection = []

        # Create a dictionary to store the count of elements in nums2
        num_count = dict(collections.Counter(nums2))

        # Iterate through each element in nums1
        for num in nums1:
            # Check if the element exists in nums2 and has a count greater than 0
            if num_count.get(num, 0) > 0:
                # Add the element to the intersection list
                intersection.append(num)
                # Decrement the count of the element in num_count
                num_count[num] -= 1

        return intersection

# Example usage
nums1 = [1, 2, 2, 1]
nums2 = [2, 2]
solution = Solution()
intersection = solution.intersect(nums1, nums2)
print("Intersection:", intersection)
