from typing import List
from cell import Cell

class UniqueCellArea:
    allowed_numbers = list(range(1, 10))

    def __init__(self, number: int, cells: List[Cell]):
        self.number = number
        self.cells = cells

    def get_missing_numbers(self):
        return set(num for num in self.allowed_numbers if num not in [cell.value for cell in self.cells])

    def get_present_numbers(self):
        return set(self.allowed_numbers) - self.get_missing_numbers()

    def is_valid(self):
        return set(cell.value for cell in self.cells) == set(self.allowed_numbers) and not self.has_duplicates()

    def has_duplicates(self):
        return any([cell.value for cell in self.cells if not cell.is_empty].count(x) > 1 for x in self.allowed_numbers)

    def __repr__(self):
        return f"{self.__class__.__name__} {self.number}: {self.cells}"

class Row(UniqueCellArea):
    pass

class Col(UniqueCellArea):
    pass

class Grid(UniqueCellArea):
    pass
