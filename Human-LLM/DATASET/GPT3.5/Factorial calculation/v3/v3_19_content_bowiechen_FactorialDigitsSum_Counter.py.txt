class Counter:
    def __init__(self, count, initial_value=0, fill_with_zeros=True):
        """
        Initialize the Counter object with the given number of digits.

        Args:
        - count (int): The number of digits in the counter.
        - initial_value (int): The initial value of the counter (default is 0).
        - fill_with_zeros (bool): Whether to fill missing digits with zeros (default is True).
        """
        self.count = count
        self.digits = []
        self.overflow = False

        if initial_value != 0:
            self._initialize_from_value(initial_value)
        else:
            self._initialize_with_zeros(fill_with_zeros)

    def _initialize_from_value(self, value):
        """
        Initialize the counter from a given initial value.

        Args:
        - value (int): The initial value of the counter.
        """
        self.digits = [int(digit) for digit in str(value)]
        while len(self.digits) < self.count:
            self.digits.insert(0, 0)

    def _initialize_with_zeros(self, fill_with_zeros):
        """
        Initialize the counter with zeros or ones.

        Args:
        - fill_with_zeros (bool): Whether to fill with zeros (True) or ones (False).
        """
        initial_digit = 0 if fill_with_zeros else 1
        self.digits = [initial_digit] * self.count

    def increment(self):
        """
        Increment the counter by one.
        """
        self.digits[-1] += 1
        for i in range(self.count - 1, -1, -1):
            if self.digits[i] > 9:
                if i == 0:
                    self.overflow = True
                self.digits[i] = 0
                if i - 1 >= 0:
                    self.digits[i - 1] += 1

    def get_value(self):
        """
        Get the current value of the counter as an integer.
        """
        return int(''.join(map(str, self.digits)))

    def get_list(self):
        """
        Get the current value of the counter as a list of digits.
        """
        return self.digits


# Usage example:
if __name__ == "__main__":
    # Create a counter with 3 digits
    counter = Counter(3)

    # Increment the counter multiple times
    for _ in range(10):
        counter.increment()
        print("Current value:", counter.get_value())
