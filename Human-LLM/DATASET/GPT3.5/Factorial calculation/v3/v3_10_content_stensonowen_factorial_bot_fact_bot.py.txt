import praw
import sqlite3
import re
import mpmath
import atoi
import info  # Assuming info.py contains your Reddit app credentials

# Regular expression pattern to match numerical values possibly formatted with commas
fact_pattern = re.compile("\d{1,3}(,?\d{3})*!")


def login():
    """
    Log in to Reddit using the provided app credentials.
    Returns:
        praw.Reddit: A Reddit instance authenticated with the app credentials.
    """
    reddit = praw.Reddit(
        user_agent=info.app_ua,
        client_id=info.app_id,
        client_secret=info.app_secret,
        redirect_uri=info.app_uri
    )
    reddit.refresh_access_information(info.app_refresh)
    return reddit


def shorten_number(x):
    """
    Shorten a large number by calculating its order of magnitude.
    Args:
        x (float): The number to be shortened.
    Returns:
        tuple: A tuple containing the shortened number and its order of magnitude.
    """
    order_of_magnitude = mpmath.floor(mpmath.log(x) / mpmath.log(10)) - 1
    return x / mpmath.power(10, order_of_magnitude), order_of_magnitude


def stirling_approximation(x):
    """
    Calculate the Stirling approximation of a number.
    Args:
        x (float): The number to approximate.
    Returns:
        float: The Stirling approximation of the number.
    """
    n = mpmath.mpf(x)
    coeff = mpmath.sqrt(2 * mpmath.pi * n)
    expon = mpmath.power(n / mpmath.e, n)
    return coeff * expon


def extract_numerical_values(comment):
    """
    Extract numerical values from a comment using regular expressions.
    Args:
        comment (str): The comment to search for numerical values.
    Returns:
        list: A list of numerical values found in the comment.
    """
    numbers = []
    matches = fact_pattern.findall(comment)
    for match in matches:
        result = atoi.extract(match)
        if result is not None:
            numbers.append(result)
    return numbers


def process_comments(reddit, cursor):
    """
    Retrieve comments from Reddit, extract numerical values, and store them in the database.
    Args:
        reddit (praw.Reddit): A Reddit instance.
        cursor (sqlite3.Cursor): A SQLite database cursor.
    """
    all_comments = reddit.subreddit('all').comments()
    for comment in all_comments:
        numbers = extract_numerical_values(comment.body)
        if numbers:
            for number in numbers:
                cursor.execute('INSERT INTO history (id) VALUES (?)', (number,))
    cursor.connection.commit()


def main():
    """
    Main function to connect to Reddit, process comments, and store data in the database.
    """
    # Connect to SQLite database
    db_connection = sqlite3.connect('history.db')
    cursor = db_connection.cursor()

    # Create history table if it doesn't exist
    cursor.execute('CREATE TABLE IF NOT EXISTS history(id TEXT)')

    # Log in to Reddit
    reddit = login()

    # Retrieve comments and store relevant data in the database
    process_comments(reddit, cursor)

    # Close database connection
    cursor.close()
    db_connection.close()


if __name__ == "__main__":
    main()
