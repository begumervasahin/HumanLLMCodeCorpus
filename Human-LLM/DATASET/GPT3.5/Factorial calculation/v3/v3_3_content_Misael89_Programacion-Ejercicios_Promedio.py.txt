import time
import os

def get_integer_input(prompt):
    """
    Prompt the user for an integer input and validate it.
    """
    while True:
        user_input = input(prompt)
        try:
            value = int(user_input)
            return value
        except ValueError:
            print('Por favor, introduzca solo números.')
            print('Vuelva a intentarlo.')
            time.sleep(1)
            clear_screen()

def clear_screen():
    """
    Clear the console screen.
    """
    os.system('cls' if os.name == 'nt' else 'clear')

def main():
    """
    Main function to execute the program.
    """
    while True:
        numero_elementos = get_integer_input("Introduzca el número de elementos: ")      
        lista_elementos = [None] * numero_elementos
        break

    for i in range(numero_elementos):
        while True:
            try:
                elementos = get_integer_input('Elemento {}: '.format(i + 1))
                lista_elementos[i] = elementos
                break
            except ValueError:
                print('Por favor, introduzca solo números.')
                print('Vuelva a intentarlo.')
                time.sleep(1)
                clear_screen()

    total = sum(lista_elementos)
    print('El promedio es:', total / numero_elementos)

if __name__ == "__main__":
    main()
