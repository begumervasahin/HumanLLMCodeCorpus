import math
from typing import List

def divide_swing_and_conquer(number: int) -> int:
    """
    Calculate the factorial of a number using the swinging and conquering algorithm.    
    """
    if number == 1 or number == 0:
        return 1

    primes = generate_primes_up_to(number)
    prime_exponents = calculate_prime_exponents(number, primes)
    swinging_factorial = calculate_swinging_factorial(primes, prime_exponents)

    return divide_swing_and_conquer(number // 2) ** 2 * swinging_factorial

def calculate_prime_exponents(swing_number: int, primes: List[int]) -> List[int]:
    """
    Calculate the prime exponents of a swinging number.
    """
    max_exponent = math.floor(math.log2(swing_number))
    prime_exponents = []

    for prime in primes:
        exponent = 0
        for i in range(1, max_exponent + 1):
            exponent += swing_number // (prime ** i) % 2
        prime_exponents.append(exponent)

    return prime_exponents

def calculate_swinging_factorial(primes: List[int], prime_exponents: List[int]) -> int:
    """
    Calculate the swinging factorial using prime numbers and their exponents.
    """
    swinging_factorial = 1

    for prime, exponent in zip(primes, prime_exponents):
        if exponent == 0:
            continue
        swinging_factorial *= prime ** exponent

    return swinging_factorial

def is_prime(n: int) -> bool:
    """
    Check if a number is prime.
    """
    if n == 1:
        return False
    if n == 2:
        return True
    if n > 2 and n % 2 == 0:
        return False

    max_divisor = math.floor(math.sqrt(n))
    for divisor in range(3, max_divisor + 1, 2):
        if n % divisor == 0:
            return False

    return True

def generate_primes_up_to(n: int) -> List[int]:
    """
    Generate all prime numbers up to a given number.
    """
    primes = []

    for num in range(2, n + 1):
        if is_prime(num):
            primes.append(num)

    return primes
