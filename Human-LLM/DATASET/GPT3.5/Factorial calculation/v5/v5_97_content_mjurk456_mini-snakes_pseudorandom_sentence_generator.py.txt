import random

def read_words_from_file(filename):
    """
    Read words from a file and return a list of stripped words.

    Parameters:
    filename (str): The name of the file to read words from.

    Returns:
    list: A list of stripped words read from the file.
    """
    with open(filename, "r") as file:
        return [word.strip() for word in file.readlines()]

def get_user_input(message):
    """
    Get user input and handle non-integer inputs.

    Parameters:
    message (str): The message to display when asking for input.

    Returns:
    int: The integer value entered by the user.
    """
    while True:
        try:
            return int(input(message))
        except ValueError:
            print("Wrong input. Please use integer numbers!")

def generate_sentences(adjective, noun, verb, adverb, count):
    """
    Generate and print sentences based on given word lists.

    Parameters:
    adjective (list): List of adjectives.
    noun (list): List of nouns.
    verb (list): List of verbs.
    adverb (list): List of adverbs.
    count (int): Number of sentences to generate.
    """
    for _ in range(count):
        sentence = (
            "A " + random.choice(adjective) + " " +
            random.choice(noun) + " " + random.choice(verb) +
            " a " + random.choice(adjective) + " " +
            random.choice(noun) + " " + random.choice(adverb) + "."
        )
        print(sentence)

if __name__ == "__main__":
    # Read words from files
    adjective = read_words_from_file("adjective.txt")
    noun = read_words_from_file("noun.txt")
    verb = read_words_from_file("verb.txt")
    adverb = read_words_from_file("adverb.txt")

    # Get the number of sentences to generate from user input
    user_answer = get_user_input("How many sentences do you want to generate? ")

    # Generate and print sentences
    generate_sentences(adjective, noun, verb, adverb, user_answer)
