def factorial(positive_int):
    """
    Computes the factorial of a positive integer.
    Accepts integers, floats with integer values, and complex numbers with zero imaginary parts.
    """
    # Check if the input is a complex number
    if isinstance(positive_int, complex):
        if positive_int.imag != 0:
            raise ValueError("Factorial is only defined for positive integers, not complex numbers with a nonzero imaginary part")
        positive_int = positive_int.real

    # Check if the input is a string
    if isinstance(positive_int, str):
        raise ValueError("Factorial is only defined for positive integers, not strings")

    # Check if the input is an integer
    if positive_int == int(positive_int):
        if positive_int > 0:
            # Calculate the factorial
            fac_prod = 1
            for n in range(1, int(positive_int) + 1):
                fac_prod *= n
            return fac_prod
        else:
            raise ValueError("Factorial is only defined for positive integers")
    else:
        raise ValueError("Factorial is only defined for positive integers")

# Example usage
try:
    n = input('Enter a positive integer: ')
    if n.lower() == 'q':
        exit()
    n = int(n)
    result = factorial(n)
    print(f"The factorial of {n} is {result}")
except ValueError as e:
    print(e)
