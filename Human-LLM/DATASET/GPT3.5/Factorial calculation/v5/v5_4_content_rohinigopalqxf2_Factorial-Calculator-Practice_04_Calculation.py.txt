import time
from selenium import webdriver

def test_factorial_calculation():
    """
    Test the factorial calculation functionality on the specified webpage.
    """
    # Launch Chrome Driver
    driver = webdriver.Chrome()
    driver.maximize_window()

    try:
        # Navigate to the page
        driver.get("https://qainterview.pythonanywhere.com/")

        # Enter an integer value in the textbox
        input_box = driver.find_element_by_id("number")
        input_box.send_keys('12')

        # Click on the calculate! button
        calculate_button = driver.find_element_by_xpath("//*[@id='getFactorial']")
        calculate_button.click()

        # Check the validation message
        validation_message = driver.find_element_by_xpath("//*[@id='resultDiv']").text

        # Verify factorial value is printed
        if validation_message == "Please enter an integer":
            print("Failure: Factorial calculation failed for integer value")
        else:
            print("Success: Factorial calculated successfully")

    finally:
        # Close the browser
        driver.quit()

# Run the test
if __name__ == "__main__":
    test_factorial_calculation()
