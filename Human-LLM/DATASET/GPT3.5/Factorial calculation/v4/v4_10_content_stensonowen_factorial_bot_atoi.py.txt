import re
import text2num

# Define translations for specific words or phrases
Translations = {
    'a': 'one',
    'dozen': 'twelve',
    'and': '',
}

# Create a set of valid tokens including numbers up to 'hundred'
valid_tokens = set(["hundred"])

# Include keys from text2num.Small, text2num.Magnitude, and Translations in valid_tokens
for d in [text2num.Small, text2num.Magnitude, Translations]:
    valid_tokens = valid_tokens.union(set(d.keys()))

# Function to prepare text by removing non-alphanumeric characters
def prep(text):
    text = re.sub("[\W_]", ' ', text)  # Replace non-alphanumeric characters with spaces
    tokens = []  # Initialize an empty list to store processed tokens
    for token in text.split():  # Split the text into tokens
        tr = Translations.get(token)  # Check if the token is in the Translations dictionary
        if tr is not None and tr != '':  # If translation exists and is not empty
            tokens.append(Translations[token])  # Append the translated token
        elif tr is None:  # If no translation exists
            tokens.append(token)  # Append the token as it is
    return tokens  # Return the list of processed tokens

# Function to extract numerical value from text
def extract(text):
    tokens = prep(text)  # Prepare the text and get the list of tokens
    first = len(tokens)  # Initialize 'first' with the total number of tokens
    for i in range(len(tokens))[::-1]:  # Iterate through tokens in reverse order
        if tokens[i] in valid_tokens:  # If the token is in the valid_tokens set
            first = i  # Update 'first' to the index of the valid token
        else:
            break  # Stop iteration if a non-valid token is encountered
    return text2num.text2num(tokens[first:])  # Convert tokens after 'first' index to a numerical value

# Function to convert text to integer
def atoi(text):
    tokens = prep(text)  # Prepare the text and get the list of tokens
    result = text2num.text2num(tokens)  # Convert tokens to numerical value using text2num library
    return result  # Return the numerical result
