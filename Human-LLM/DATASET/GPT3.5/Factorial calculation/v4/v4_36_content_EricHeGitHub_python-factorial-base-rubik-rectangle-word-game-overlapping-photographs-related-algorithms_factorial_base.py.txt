from sys import exit
import time
from itertools import product

# Function to compute the factorial of a nonnegative integer
def factorial(n):
    result = 1
    for i in range(n):
        result *= i + 1
    return result

# Function to convert a list of digits to an integer
def list_to_integer(lst):
    result = 0
    power = len(lst) - 1
    for digit in lst:
        if not str(digit).isdigit():  # Check if the digit is a valid number
            return None
        else:
            result += digit * 10 ** power
            power -= 1
    return result

# Prompt the user to input a nonnegative integer
number = input('Input a nonnegative integer: ')

# Validate the input
if not number.isdigit() or number[0] == '0':
    print('Incorrect input, giving up...')
    exit()

# Convert the input to an integer
number = int(number)

# Compute the maximum factorial required
Max = 1
while factorial(Max) <= number:
    Max += 1
Max -= 1

# Initialize variables
middle_result = 0
factorial_results = []
number_display = number

# Convert the decimal number into factorial base representation
while number:
    for i in range(Max, 0, -1):
        while middle_result * factorial(i) < number:
            middle_result += 1
        if middle_result * factorial(i) > number:
            middle_result -= 1
        else:
            pass
        factorial_results.append(middle_result)
        number -= middle_result * factorial(i)
        middle_result = 0

# Output the decimal number and its factorial base representation
print('Decimal {} reads as {} in factorial base.'.format(number_display, list_to_integer(factorial_results)))
