import time
import os

# Function to get the integer input from the user
def get_integer_input(prompt):
    """
    Prompt the user for an integer input and validate it.
    """
    while True:
        try:
            value = int(input(prompt))
            return value
        except ValueError:
            print('Por favor, introduzca solo números.')
            print('Inténtelo nuevamente.')
            time.sleep(1)
            clear_screen()

# Function to clear the screen
def clear_screen():
    """
    Clear the console screen.
    """
    os.system('cls' if os.name == 'nt' else 'clear')

def main():
    """
    Main function to execute the program.
    """
    # Prompt for the number of elements
    while True:
        try:
            num_elements = get_integer_input("Introduzca el número de elementos: ")
            break
        except KeyboardInterrupt:
            print('Operación cancelada.')
            return

    # Create a list to store elements
    elements_list = [None] * num_elements

    # Loop to get elements from the user
    for i in range(num_elements):
        while True:
            try:
                element = get_integer_input('Elemento {}: '.format(i + 1))
                elements_list[i] = element
                break
            except KeyboardInterrupt:
                print('Operación cancelada.')
                return
            except ValueError:
                print('Por favor, introduzca solo números.')
                print('Inténtelo nuevamente.')
                time.sleep(1)
                clear_screen()

    # Calculate the average of elements
    total_sum = sum(elements_list)
    average = total_sum / num_elements

    # Display the average
    print('El promedio es:', average)

if __name__ == "__main__":
    main()
