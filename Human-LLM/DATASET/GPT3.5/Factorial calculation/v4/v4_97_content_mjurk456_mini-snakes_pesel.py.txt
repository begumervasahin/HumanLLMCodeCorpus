import unittest

def check_pesel(pesel_nr):
    """
    Validates a PESEL number.

    Parameters:
    pesel_nr (str): The PESEL number to be validated.

    Returns:
    bool: True if the PESEL number is valid, False otherwise.

    The PESEL validation algorithm was taken from:
    http://www.infor.pl/prawo/gmina/dowod-osobisty/262184,Jak-sprawdzic-czy-masz-poprawny-PESEL.html
    """
    if len(pesel_nr) < 11:
        return False
    if not pesel_nr.isdigit():
        return False

    check_array = [1, 3, 7, 9, 1, 3, 7, 9, 1, 3]
    final = sum(int(pesel_nr[i]) * check_array[i] for i in range(10))
    
    return (10 - int(str(final)[-1])) == int(pesel_nr[-1])

class PeselTest(unittest.TestCase):
    def test_correct_pesel(self):
        self.assertTrue(check_pesel("92121510595"))
        self.assertTrue(check_pesel("99810235403"))

    def test_wrong_pesel(self):
        self.assertFalse(check_pesel("44051401358"))
        self.assertFalse(check_pesel("51401358"))

if __name__ == "__main__":
    unittest.main()
