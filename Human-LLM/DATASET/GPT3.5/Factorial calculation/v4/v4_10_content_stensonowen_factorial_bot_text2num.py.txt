# Define dictionaries for small numbers and magnitudes
Small = {
    'zero': 0, 'one': 1, 'two': 2, 'three': 3, 'four': 4, 'five': 5, 'six': 6, 'seven': 7, 'eight': 8, 'nine': 9,
    'ten': 10, 'eleven': 11, 'twelve': 12, 'thirteen': 13, 'fourteen': 14, 'fifteen': 15, 'sixteen': 16,
    'seventeen': 17, 'eighteen': 18, 'nineteen': 19, 'twenty': 20, 'thirty': 30, 'forty': 40, 'fifty': 50,
    'sixty': 60, 'seventy': 70, 'eighty': 80, 'ninety': 90
}

Magnitude = {
    'thousand': 1000, 'million': 1000000, 'billion': 1000000000, 'trillion': 1000000000000,
    'quadrillion': 1000000000000000, 'quintillion': 1000000000000000000, 'sextillion': 1000000000000000000000,
    'septillion': 1000000000000000000000000, 'octillion': 1000000000000000000000000000,
    'nonillion': 1000000000000000000000000000000, 'decillion': 1000000000000000000000000000000000
}

class NumberException(Exception):
    """Custom exception for handling number-related errors."""
    def __init__(self, msg):
        super().__init__(msg)

def text_to_number(tokens):
    """Converts textual numbers to numerical values."""
    n = 0  # Total numerical value
    g = 0  # Current numerical value being accumulated

    for word in tokens:
        x = Small.get(word, None)  # Check if the word is a small number
        if x is not None:
            g += x
        elif word == "hundred" and g != 0:  # If the word is "hundred" and there's a current value
            g *= 100
        else:
            x = Magnitude.get(word, None)  # Check if the word is a magnitude
            if x is not None:
                n += g * x
                g = 0
            else:
                raise NumberException("Unknown number: " + word)

    return n + g  # Return the total numerical value

if __name__ == "__main__":
    # Test cases to validate the functionality
    assert 1 == text_to_number("one".split())
    assert 12 == text_to_number("twelve".split())
    assert 72 == text_to_number("seventy two".split())
    assert 300 == text_to_number("three hundred".split())
    assert 1200 == text_to_number("twelve hundred".split())
    assert 12304 == text_to_number("twelve thousand three hundred four".split())
    assert 6000000 == text_to_number("six million".split())
    assert 6400005 == text_to_number("six million four hundred thousand five".split())
    assert 123456789012 == text_to_number("one hundred twenty three billion four hundred fifty six million seven hundred eighty nine thousand twelve".split())
    assert 4000000000000000000000000000000000 == text_to_number("four decillion".split())

    print("All assertions passed.")
