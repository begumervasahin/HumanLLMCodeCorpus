import time
from selenium import webdriver

# Set desired capabilities for the mobile device
desired_cap = {
    'device': 'iPhone 7',
    'realMobile': 'true',
    'platform': 'iOS',
    'browserName': 'safari',
    'browserstack.debug': 'true'
}

# Replace 'USERNAME' and 'ACCESS_KEY' with your actual BrowserStack credentials
username = 'USERNAME'
access_key = 'ACCESS_KEY'

# Initialize the WebDriver with BrowserStack remote execution
command_executor = f'http://{username}:{access_key}@hub.browserstack.com:80/wd/hub'
driver = webdriver.Remote(command_executor=command_executor, desired_capabilities=desired_cap)

# Navigate to the QA Interview Application page
driver.get("https://qainterview.pythonanywhere.com/")

# Check if the page title is correct
if driver.title == "Factoriall":
    print("Success: QA Interview Application page launched successfully")
else:
    print("Failure: QA Interview Application page title is incorrect")

# Input the value '5' and click on the Calculate button
number_input = driver.find_element_by_xpath("//input[@id='number']")
number_input.send_keys('5')
calculate_button = driver.find_element_by_id('getFactorial')
calculate_button.click()

# Wait for the result to appear
time.sleep(5)

# Get the calculated factorial result and check if it's correct
result_element = driver.find_element_by_xpath("//p[@id='resultDiv']")
result_text = result_element.text
result_value = float(result_text.split(':')[1].strip())

if result_value == 120:
    print("Result for 5! is correct:", result_value)
else:
    print("Result for 5! is incorrect:", result_value)

# Close the browser
driver.close()
