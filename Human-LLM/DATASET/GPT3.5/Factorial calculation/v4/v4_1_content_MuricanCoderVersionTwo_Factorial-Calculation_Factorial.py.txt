import sys
import datetime
import operator
import functools
import decimal

# Input maximum count and number added per round
max_number = int(input('What will the maximum count be?: '))
number_increase = int(input('What will be the number added per round?: '))
number1 = 0

# Iterate until reaching the maximum count
while number1 < max_number:
    # Increment number1 and calculate number2
    number2 = number1 + 1
    number1 += number_increase
    
    # Initialize list for multipliers
    multipliers = []
    
    # Record start time
    now = datetime.datetime.now()
    
    # Write start time to file
    with open("Factorial.txt", "a") as file:
        file.write("Start time " + now.strftime('%H%m%S.%f') + "\n")

    # Calculate factorial multipliers
    for i in range(number1, 0, -1):
        multipliers.append(i)

    # Calculate factorial result
    result = functools.reduce(operator.mul, multipliers, 1)
    
    # Record end time
    now1 = datetime.datetime.now()
    
    # Write end time to file
    with open("Factorial.txt", "a") as file:
        file.write("End time " + now1.strftime('%H%m%S.%f') + "\n")
    
    # Format factorial result
    x = decimal.Decimal(result)
    
    # Write factorial and time difference to file
    with open("Factorial.txt", "a") as file:
        file.write(f"Factorial of {number1} is {format(x, '.6e')}\n")
        file.write(f"The time difference is {(now1 - now).total_seconds()} seconds\n")
