import time
from selenium import webdriver

# Set desired capabilities for the mobile device
desired_cap = {
    'device': 'iPhone 7',
    'realMobile': 'true',
    'platform': 'iOS',
    'browserName': 'safari',
    'browserstack.debug': 'true'
}

# Initialize the WebDriver with BrowserStack remote execution
driver = webdriver.Remote(
    command_executor='http://USERNAME:ACCESS_KEY@hub.browserstack.com:80/wd/hub',
    desired_capabilities=desired_cap  # Replace USERNAME and ACCESS_KEY with your BrowserStack credentials
)

# Navigate to the QA Interview Application page
driver.get("https://qainterview.pythonanywhere.com/")

# Check if the page title is correct
if driver.title == "Factoriall":
    print("Success: QA Interview Application page launched successfully")
else:
    print("Failure: QA Interview Application page Title is incorrect")

# Input the value '5' and click on the Calculate button
driver.find_element_by_xpath("//input[@id='number']").send_keys('5')
driver.find_element_by_id('getFactorial').click()

# Wait for the result to appear
time.sleep(5)

# Get the calculated factorial result and check if it's correct
result_text = driver.find_element_by_xpath("//p[@id='resultDiv']").text
result = float(result_text.split(':')[1].strip())
if result == 120:
    print("Result for 5! is correct:", result)
else:
    print("Result for 5! is incorrect:", result)

# Close the browser
driver.close()
