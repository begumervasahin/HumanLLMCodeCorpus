import requests
import os
import bs4

def download_images_from_url(url, download_path):
    """
    Download images from a specified URL and save them to a download directory.

    Parameters:
    url (str): The URL of the webpage containing images.
    download_path (str): The directory path to save the downloaded images.
    """
    # Create the download directory if it doesn't exist
    os.makedirs(download_path, exist_ok=True)

    while True:
        print("Downloading page:", url)
        res = requests.get(url)
        res.raise_for_status()
        soup = bs4.BeautifulSoup(res.text, "lxml")

        # Find image elements on the webpage
        img_elements = soup.select(".image-size-full")
        for element in img_elements:
            img_url = element.get('src')
            print("Downloading image:", img_url)
            image_res = requests.get(img_url)
            image_res.raise_for_status()

            # Save the downloaded image to the specified directory
            with open(os.path.join(download_path, os.path.basename(img_url)), 'wb') as image_file:
                for chunk in image_res.iter_content(100000):
                    image_file.write(chunk)

        # Find the link to the next page
        next_link = soup.find("a", class_="pagination")
        if next_link:
            url = next_link.get('href')
        else:
            break

    print("Download completed.")
