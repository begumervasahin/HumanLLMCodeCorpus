def main():
    min_limit = 0
    max_limit = 0
    step = 1
    fizzes = 0
    buzzes = 0
    fizzbuzzes = 0
    normal = 0
    
    # Input start and stop values
    while True:
        try:
            min_limit, max_limit = map(int, input("Input start and stop values for counting dividing them with a space: ").split())
            break
        except ValueError:
            print("One or all values are wrong. Input should be like 12 10")
    
    # Determine step direction
    if min_limit > max_limit:
        step = -1
    
    # Perform counting
    for i in range(min_limit, max_limit + step, step):
        if i % 3 == 0 and i % 5 != 0 and i != 0:
            print("fizz", end=" ")
            fizzes += 1
        elif i % 5 == 0 and i % 3 != 0 and i != 0:
            print("buzz", end=" ")
            buzzes += 1
        elif i % 3 == 0 and i % 5 == 0 and i != 0:
            print("fizzbuzz", end=" ")
            fizzbuzzes += 1
        else:
            print(i, end=" ")
            normal += 1
    
    # Display counts
    print("\nFizzes in range:", fizzes)
    print("Buzzes in range:", buzzes)
    print("Fizzbuzzes in range:", fizzbuzzes)
    print("Normal numbers in range:", normal)

if __name__ == "__main__":
    main()
