def text_file_aligning(file_path, align, line_length):
    """Splits text file into lines with given length and align these lines
       as given.
       Parameters:
       file_path - path to .txt file. String.
       align - type of aligning. One-symbol string.
          r - right
          l - left
          c - center
        line_length - length of a line into which the text file will be split.
          Integer.
        Returns:
        output on the screen
     """
    if align == "r":
        pattern = "{:>"
    elif align == "l":
        pattern = "{:"
    elif align == "c":
        pattern = "{:^"
    pattern = pattern + str(line_length) + "}"
    
    with open(file_path, "r") as f:
        for line in f:
            while len(line) > line_length:
                last_space = line[:line_length].rfind(" ")
                print(pattern.format(line[:last_space]))
                line = line[last_space + 1:]
            else:
                print(pattern.format(line.rstrip()))

# Example usage:
text_file_aligning("example.txt", "c", 20)
