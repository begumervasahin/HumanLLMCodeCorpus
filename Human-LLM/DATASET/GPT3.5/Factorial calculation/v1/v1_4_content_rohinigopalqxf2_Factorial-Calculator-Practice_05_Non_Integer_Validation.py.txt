import time
from selenium import webdriver

def test_non_integer_factorial_calculation():
    """
    Test the factorial calculation functionality with a non-integer input.
    """
    # Launch Chrome Driver
    driver = webdriver.Chrome()
    driver.maximize_window()

    try:
        # Navigate to the page
        driver.get("https://qainterview.pythonanywhere.com/")

        # Enter a non-integer value in the textbox
        input_box = driver.find_element_by_id("number")
        input_box.send_keys('aa')

        # Click on the calculate! button
        calculate_button = driver.find_element_by_xpath("//*[@id='getFactorial']")
        calculate_button.click()

        # Check the validation message
        validation_message = driver.find_element_by_xpath("//*[@id='resultDiv']").text

        # Verify that the factorial calculation failed for non-integer input
        if validation_message == "Please enter an integer":
            print("Success: Factorial calculation failed for non-integer input")
        else:
            print("Failure: Factorial calculation succeeded for non-integer input")

    finally:
        # Close the browser
        driver.quit()

# Run the test
if __name__ == "__main__":
    test_non_integer_factorial_calculation()
