class Counter:
    def __init__(self, count, initial_value=0, zeros=True):
        self.count = count
        self.digits = []
        self.overflow = False

        if initial_value != 0:
            # Convert the initial value to a list of digits
            self.digits = [int(digit) for digit in str(initial_value)]
            # If the initial value has fewer digits than the count, pad with zeros
            while len(self.digits) < count:
                self.digits.insert(0, 0)
        else:
            # If initial value is not provided, initialize with zeros or ones based on the flag
            self.digits = [0 if zeros else 1] * count

    def increment(self):
        # Increment the least significant digit
        self.digits[-1] += 1
        # Handle carry-over
        for i in range(self.count - 1, -1, -1):
            if self.digits[i] > 9:
                if i == 0:
                    self.overflow = True
                self.digits[i] = 0
                if i - 1 >= 0:
                    self.digits[i - 1] += 1

    def get_value(self):
        # Convert the list of digits back to an integer
        return int(''.join(map(str, self.digits)))

    def get_list(self):
        return self.digits


# Usage example:
if __name__ == "__main__":
    # Create a counter with 3 digits
    counter = Counter(3)

    # Increment the counter multiple times
    for _ in range(10):
        counter.increment()
        print("Current value:", counter.get_value())
