# Function to validate a PESEL number
def check_pesel(pesel_nr):
    """
    Validator for PESEL numbers.
    
    Parameters:
    pesel_nr (str): A string containing the PESEL number to be checked.
    
    Returns:
    bool: True if the PESEL number is correct, False otherwise.
    
    The PESEL validation algorithm was taken from:
    http://www.infor.pl/prawo/gmina/dowod-osobisty/262184,Jak-sprawdzic-czy-masz-poprawny-PESEL.html
    """
    # Check if the length of the PESEL number is less than 11
    if len(pesel_nr) < 11:
        return False
    
    # Check if the PESEL number contains only digits
    if not pesel_nr.isdigit():
        return False
    
    # Define the weights for each digit in the PESEL number
    check_array = [1, 3, 7, 9, 1, 3, 7, 9, 1, 3]
    
    # Calculate the weighted sum of the digits
    final = sum(int(pesel_nr[i]) * check_array[i] for i in range(10))
    
    # Validate the PESEL number
    return (10 - int(str(final)[-1])) == int(pesel_nr[-1])

import unittest

# Unit tests for the check_pesel function
class PeselTest(unittest.TestCase):
    def test_correct_pesel(self):
        self.assertTrue(check_pesel("92121510595"))
        self.assertTrue(check_pesel("99810235403"))

    def test_wrong_pesel(self):
        self.assertFalse(check_pesel("44051401358"))
        self.assertFalse(check_pesel("51401358"))

# Run the unit tests if the script is executed directly
if __name__ == "__main__":
    unittest.main()
