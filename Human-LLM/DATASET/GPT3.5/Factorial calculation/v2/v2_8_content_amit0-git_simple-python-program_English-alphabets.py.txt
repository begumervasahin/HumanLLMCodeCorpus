def print_alphabet_A():
    for row in range(6):
        for col in range(11):
            if (row + col == 5) or (col - row == 5) or (row == 3 and row + col in [7, 9]):
                print('*', end='')
            else:
                print(end=' ')
        print()

def print_alphabet_M():
    for row in range(6):
        for col in range(11):
            if (col == 0) or (col - row == 0) or (row + col == 10) or (col == 10):      
                print('*', end='')
            else:
                print(end=' ')
        print()

def print_alphabet_I():
    for row in range(7):
        for col in range(11):
            if (col == 5):
                print('*', end='')
            else:
                print(end=' ')
        print()

def print_alphabet_T():
    for row in range(6):
        for col in range(11):
            if (col != 5 and row == 0) or (col == 5):
                print('*', end='')
            else:
                print(end=' ')
        print()

def print_alphabet_V():
    for row in range(6):
        for col in range(11):
            if (row + col == 10) or (row - col == 0):
                print('*', end='')
            else:
                print(end=' ')
        print()

# Print English alphabets
print("Alphabet A:")
print_alphabet_A()
print("\nAlphabet M:")
print_alphabet_M()
print("\nAlphabet I:")
print_alphabet_I()
print("\nAlphabet T:")
print_alphabet_T()
print("\nAlphabet V:")
print_alphabet_V()
