def text_file_aligning(file_path, align, line_length):
    """
    Splits a text file into lines with the given length and aligns these lines
    as specified.

    Parameters:
    file_path (str): Path to the .txt file.
    align (str): Type of aligning. One-symbol string.
                 'r' - right
                 'l' - left
                 'c' - center
    line_length (int): Length of a line into which the text file will be split.

    Returns:
    None. Output is printed on the screen.
    """
    # Determine the pattern for formatting based on the alignment type
    if align == "r":
        pattern = "{:>"
    elif align == "l":
        pattern = "{:"
    elif align == "c":
        pattern = "{:^"
    pattern = pattern + str(line_length) + "}"

    # Open the file and process each line
    with open(file_path, "r") as file:
        for line in file:
            # Split the line into segments of the specified length
            while len(line) > line_length:
                # Find the last space within the segment length
                last_space = line[:line_length].rfind(" ")
                # Print the segment with the determined alignment
                print(pattern.format(line[:last_space]))
                # Update the line to process the remaining part
                line = line[last_space + 1:]
            else:
                # Print the remaining part of the line with the determined alignment
                print(pattern.format(line.rstrip()))

# Example usage:
text_file_aligning("example.txt", "c", 20)
