from __future__ import division
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.cbook as cbook

def union_changepoints(allchangepoints):
    """
    Combine multiple sets of changepoints into a single set.

    Args:
    - allchangepoints (list): List of changepoints lists.

    Returns:
    - list: Combined list of changepoints.
    """
    startpoints = sorted(set(cbook.flatten(allchangepoints)))
    return [(startpoint, nextstartpoint) for startpoint, nextstartpoint in zip(startpoints[:-1], startpoints[1:])]

def plot_with_changepoints(data, changepoints):
    """
    Plot the data with vertical lines indicating the changepoints.

    Args:
    - data (array-like): Data to plot.
    - changepoints (list): List of changepoints tuples.
    """
    plt.figure()
    plt.plot(data)
    rect = plt.axis()
    plt.vlines([c[1] for c in changepoints[:-1]], rect[2], rect[3], color='r', linestyles='dashed')
    plt.axis(rect)
    plt.show()

def indicators_to_changepoints(indicators):
    """
    Convert indicators to changepoints.

    Indicators are 0 everywhere except the index at which a new segment starts.

    Args:
    - indicators (list): List of boolean indicators.

    Returns:
    - list: List of changepoints tuples.
    """
    pos, = np.where(np.diff(np.array(indicators, dtype=int)) == 1)
    pos = np.concatenate(((0,), pos + 1, (len(indicators),)))
    return [(a, b) for a, b in zip(pos[:-1], pos[1:])]

# Example usage
if __name__ == "__main__":
    data = np.random.rand(100)
    indicators = np.random.choice([False, True], size=100)
    changepoints = indicators_to_changepoints(indicators)
    plot_with_changepoints(data, changepoints)
