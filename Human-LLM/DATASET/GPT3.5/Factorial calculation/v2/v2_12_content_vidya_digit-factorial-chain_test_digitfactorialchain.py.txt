from digitfactorialchain.digitfactorialchain import DigitFactorialChain

# Test for calculating the factorial sum chain length for a given number
def test_factorial_sum_chain_length_169():
    # Initialize the shared area for the DigitFactorialChain
    DigitFactorialChain.init_shared_area()
    
    # Create an instance of DigitFactorialChain for the number 169
    dfc = DigitFactorialChain(169)
    
    # Define the expected length of the factorial sum chain for 169
    expected_length = 3
    
    # Calculate the factorial sum chain length
    result_length = dfc.fact_sum_chain_len()
    
    # Assert that the calculated length matches the expected length
    assert result_length == expected_length
