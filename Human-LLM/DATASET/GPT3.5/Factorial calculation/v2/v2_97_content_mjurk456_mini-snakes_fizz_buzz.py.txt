def main():
    # Initialize variables
    min_limit = 0
    max_limit = 0
    step = 1
    fizzes = 0
    buzzes = 0
    fizzbuzzes = 0
    normal = 0

    # Input start and stop values
    while True:
        try:
            min_limit, max_limit = map(int, input("Enter the start and stop values separated by a space: ").split())
            break
        except ValueError:
            print("Invalid input. Please enter two integers separated by a space.")

    # Determine the step direction
    if min_limit > max_limit:
        step = -1

    # Perform counting
    for i in range(min_limit, max_limit + step, step):
        if i % 3 == 0 and i % 5 != 0 and i != 0:
            print("fizz", end=" ")
            fizzes += 1
        elif i % 5 == 0 and i % 3 != 0 and i != 0:
            print("buzz", end=" ")
            buzzes += 1
        elif i % 3 == 0 and i % 5 == 0 and i != 0:
            print("fizzbuzz", end=" ")
            fizzbuzzes += 1
        else:
            print(i, end=" ")
            normal += 1

    # Display counts
    print("\nNumber of fizzes in the range:", fizzes)
    print("Number of buzzes in the range:", buzzes)
    print("Number of fizzbuzzes in the range:", fizzbuzzes)
    print("Number of normal numbers in the range:", normal)

if __name__ == "__main__":
    main()
