import pandas as pd
from sklearn.decomposition import PCA
from sklearn.manifold import TSNE
from sklearn.preprocessing import scale
import matplotlib.pyplot as plt

# Load data from CSV file
data = pd.read_csv('endometrial.csv')

# Remove rows with missing values
data.dropna(inplace=True)

# Exclude the first column (assuming it's an index column)
data = data.iloc[:, 1:]

# Scale the data
scaled_data = scale(data.T)  # Transpose for column-wise scaling

# Apply t-SNE
tsne = TSNE(n_components=2, n_iter=2500, random_state=0)
tsne_data = tsne.fit_transform(scaled_data)

# Visualize t-SNE result
plt.scatter(tsne_data[:, 0], tsne_data[:, 1])
plt.title('t-SNE Visualization of EC Dataset')
plt.xlabel('t-SNE Component 1')
plt.ylabel('t-SNE Component 2')
plt.show()
