import pandas as pd
from sklearn.decomposition import PCA
from sklearn.manifold import TSNE
from sklearn.preprocessing import scale
import matplotlib.pyplot as plt

# Read data from CSV file
data_with_nan = pd.read_csv('endometrial.csv')

# Remove rows with missing values
data = data_with_nan.dropna()

# Remove the first column (assuming it's an index column)
data = data.iloc[:, 1:]

# Scale the data
scaled_data = scale(data.T)  # Transpose data for scaling

# Apply t-SNE
tsne = TSNE(n_components=2, n_iter=2500, random_state=0)
tsne_data = tsne.fit_transform(scaled_data)

# Plot t-SNE result
plt.scatter(tsne_data[:, 0], tsne_data[:, 1])
plt.title('t-SNE Visualization of EC Dataset')
plt.xlabel('t-SNE Component 1')
plt.ylabel('t-SNE Component 2')
plt.show()
