import math
import matplotlib.pyplot as plt
from pandas import read_csv, to_datetime

def parse_datetime(x):
    """
    Parse a datetime string into a datetime object.
    """
    return to_datetime(x, format='%Y-%m-%d %H:%M:%S')

def mean_absolute_percentage_error(y_true, y_pred):
    """
    Calculate the Mean Absolute Percentage Error (MAPE).

    Parameters:
    y_true (list): List of actual values.
    y_pred (list): List of predicted values.

    Returns:
    float: The MAPE value as a percentage.
    """
    total_error = sum(abs((y_p - y_t) / y_t) for y_t, y_p in zip(y_true, y_pred))    
    return (total_error / len(y_true)) * 100

# Define the path to the CSV file
csv_file_path = '/Users/daihanru/Desktop/arima-lstm/DataSet/FEB15.csv'

# Read the CSV file and parse dates using the defined parse_datetime function
series = read_csv(
    csv_file_path,
    usecols=[2, 3],
    header=0,
    parse_dates=[0],
    index_col=0,
    squeeze=True,
    date_parser=parse_datetime
)

# Extract the first 1000 values for plotting
X = series.values[:1000]

# Plot the time series
plt.plot(X, '-', label="Real Flow")
plt.legend(loc='upper right')
plt.xlabel("Period (15-minute intervals)")
plt.ylabel("Volume (vehicle/period)")
plt.title("Time Series of Traffic Volume")
plt.show()

# Example usage of the mean_absolute_percentage_error function
# Replace y_true and y_pred with your actual data
y_true = [1, 2, 3, 4, 5]
y_pred = [0.9, 2.1, 3.2, 4.3, 5.1]

mape = mean_absolute_percentage_error(y_true, y_pred)
print(f"Mean Absolute Percentage Error (MAPE): {mape:.2f}%")
