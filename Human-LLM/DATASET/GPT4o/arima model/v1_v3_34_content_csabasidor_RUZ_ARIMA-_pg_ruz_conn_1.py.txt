def connect_to_database(connection_params):
    """
    Establish a connection to the PostgreSQL database using the provided parameters.

    Parameters:
    connection_params (dict): Dictionary containing database connection parameters.

    Returns:
    connection: psycopg2 connection object if successful, None otherwise.
    """
    try:
        connection = pg.connect(**connection_params)
        print("Connected to the database!")
        return connection
    except Exception as e:
        print(f"Error connecting to the database: {e}")
        return None

def execute_query(connection, query):
    """
    Execute a SQL query on the database and retrieve the results in a Pandas DataFrame.

    Parameters:
    connection: psycopg2 connection object.
    query (str): SQL query to execute.

    Returns:
    pd.DataFrame: DataFrame containing the query results if successful, None otherwise.
    """
    try:
        # Execute the query and retrieve results into a Pandas DataFrame
        data_frame = pd.read_sql(query, connection)
        return data_frame
    except Exception as e:
        print(f"Error executing query: {e}")
        return None

def close_connection(connection):
    """
    Close the connection to the PostgreSQL database.

    Parameters:
    connection: psycopg2 connection object.
    """
    try:
        if connection is not None:
            connection.close()
            print("Connection closed.")
    except Exception as e:
        print(f"Error closing the connection: {e}")
