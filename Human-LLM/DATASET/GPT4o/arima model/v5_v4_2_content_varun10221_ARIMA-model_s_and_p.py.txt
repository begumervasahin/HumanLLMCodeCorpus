import os
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import statsmodels.api as sm
import seaborn as sns

# Set seaborn style for visualizations
sns.set_style('darkgrid')

def main():
    # Load and preprocess stock data
    stock_data = load_stock_data('q_table.csv')

    # Calculate additional columns for analysis
    stock_data = calculate_additional_columns(stock_data)

    # Calculate autocorrelation and partial autocorrelation
    lag_corr, lag_partial_corr = calculate_autocorrelations(stock_data['Logged First Difference'])

    # Perform seasonal decomposition
    decomposition = decompose_seasonal(stock_data['Natural Log'])

    # Fit ARIMA model and generate forecast
    stock_data = fit_arima_and_forecast(stock_data)

    # Plot actual vs forecasted values
    plot_forecast(stock_data)

def load_stock_data(filename):
    """Load stock data from a CSV file and preprocess it."""
    path = os.path.join(os.getcwd(), filename)
    stock_data = pd.read_csv(path)

    # Convert 'Date' column to datetime format and sort data by date
    stock_data['Date'] = pd.to_datetime(stock_data['Date'])
    stock_data = stock_data.sort_values(by='Date', ascending=True).set_index('Date')
    
    return stock_data

def calculate_additional_columns(stock_data):
    """Calculate additional columns needed for analysis."""
    stock_data['First Difference'] = stock_data['Close'].diff()
    stock_data['Natural Log'] = np.log(stock_data['Close'])
    stock_data['Original Variance'] = stock_data['Close'].rolling(window=30, center=True).var()
    stock_data['Log Variance'] = stock_data['Natural Log'].rolling(window=30, center=True).var()
    stock_data['Logged First Difference'] = stock_data['Natural Log'].diff()
    stock_data['Lag 20'] = stock_data['Logged First Difference'].shift(20)
    
    return stock_data

def calculate_autocorrelations(logged_diff):
    """Calculate autocorrelation and partial autocorrelation."""
    lag_corr = sm.tsa.acf(logged_diff.dropna())
    lag_partial_corr = sm.tsa.pacf(logged_diff.dropna())
    
    return lag_corr, lag_partial_corr

def decompose_seasonal(natural_log):
    """Perform seasonal decomposition of time series data."""
    decomposition = sm.tsa.seasonal_decompose(natural_log, model='additive', period=30)
    return decomposition

def fit_arima_and_forecast(stock_data):
    """Fit ARIMA model and generate forecast."""
    model = sm.tsa.ARIMA(stock_data['Logged First Difference'].dropna(), order=(0, 0, 1))
    results = model.fit(disp=-1)
    stock_data['Forecast'] = results.fittedvalues
    
    return stock_data

def plot_forecast(stock_data):
    """Plot actual vs forecasted values."""
    plt.figure(figsize=(16, 12))
    plt.plot(stock_data['Logged First Difference'], label='Actual')
    plt.plot(stock_data['Forecast'], label='Forecast')
    plt.legend()
    plt.title('Actual vs Forecasted Logged First Difference')
    plt.xlabel('Date')
    plt.ylabel('Logged First Difference')
    plt.show()

if __name__ == "__main__":
    main()
