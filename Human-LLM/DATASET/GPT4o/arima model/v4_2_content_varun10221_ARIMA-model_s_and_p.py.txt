import os
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import statsmodels.api as sm
import seaborn as sb
from statsmodels.tsa.stattools import acf, pacf
from statsmodels.tsa.seasonal import seasonal_decompose

# Set the seaborn style
sb.set_style('darkgrid')

# Define the path to the CSV file and load the data
path = os.path.join(os.getcwd(), 'q_table.csv')
stock_data = pd.read_csv(path)

# Convert the 'Date' column to datetime and sort by date
stock_data['Date'] = pd.to_datetime(stock_data['Date'])
stock_data = stock_data.sort_values(by='Date', ascending=True)
stock_data = stock_data.set_index('Date')

# Calculate the first difference of the 'Close' prices
stock_data['First Difference'] = stock_data['Close'] - stock_data['Close'].shift()

# Calculate the natural logarithm of the 'Close' prices
stock_data['Natural Log'] = stock_data['Close'].apply(np.log)

# Calculate the rolling variance of the 'Close' prices and their natural logs
stock_data['Original Variance'] = stock_data['Close'].rolling(window=30, center=True).var()
stock_data['Log Variance'] = stock_data['Natural Log'].rolling(window=30, center=True).var()

# Calculate the first difference of the natural logs
stock_data['Logged First Difference'] = stock_data['Natural Log'] - stock_data['Natural Log'].shift(1)

# Create a lagged series of the logged first difference
stock_data['Lag 20'] = stock_data['Logged First Difference'].shift(20)

# Calculate the autocorrelation and partial autocorrelation functions
lag_corr = acf(stock_data['Logged First Difference'].dropna())
lag_partial_corr = pacf(stock_data['Logged First Difference'].dropna())

# Perform seasonal decomposition on the natural log of 'Close' prices
decomposition = seasonal_decompose(stock_data['Natural Log'], model='additive', period=30)

# Fit an ARIMA model to the logged first difference series
model = sm.tsa.ARIMA(stock_data['Logged First Difference'].dropna(), order=(0, 0, 1))
results = model.fit(disp=-1)

# Store the fitted values as 'Forecast'
stock_data['Forecast'] = results.fittedvalues

# Plot the logged first difference and the forecast
plt.figure(figsize=(16, 12))
stock_data[['Logged First Difference', 'Forecast']].plot()
plt.show()
