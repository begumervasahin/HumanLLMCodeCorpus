import torch
import torch.nn as nn
from torch.nn import init

class YourModel(nn.Module):
    def __init__(self, input_features=10, output_features=5):
        super(YourModel, self).__init__()
        # Define a linear layer with specified input and output features
        self.linear_layer = nn.Linear(in_features=input_features, out_features=output_features)

    def forward(self, x):
        # Implement the forward pass of your model
        return self.linear_layer(x)

def initialize_weights(model):
    # Apply Xavier uniform initialization to linear layers in the model
    if isinstance(model, nn.Linear):
        init.xavier_uniform_(model.weight.data)

# Create an instance of the YourModel class
your_model = YourModel()

# Apply weight initialization to the model
your_model.apply(initialize_weights)

# Print initialized weights for linear layers
for name, param in your_model.named_parameters():
    if 'weight' in name:
        print(f'Initialized weights for {name} with shape {param.shape}')

# Example: Now you can use your_model for further training or evaluation
if __name__ == "__main__":
    # Example input tensor
    example_input = torch.randn(1, 10)
    # Forward pass
    output = your_model(example_input)
    print(f'Output: {output}')
