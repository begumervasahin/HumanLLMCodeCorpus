import math
import matplotlib.pyplot as plt
import pandas as pd

def parse_date(date_str):
    """
    Parse date string into a datetime object.
    
    Args:
        date_str (str): Date string in the format '%Y-%m-%d %H:%M:%S'.
    
    Returns:
        datetime: Parsed datetime object.
    """
    return pd.to_datetime(date_str, format='%Y-%m-%d %H:%M:%S')

def mean_absolute_percentage_error(y_true, y_pred):
    """
    Calculate Mean Absolute Percentage Error (MAPE).
    
    Args:
        y_true (list): List of true values.
        y_pred (list): List of predicted values.
    
    Returns:
        float: Mean Absolute Percentage Error.
    """
    total_error = sum(abs(pred - true) / true for true, pred in zip(y_true, y_pred))
    return total_error / len(y_true)

def read_and_process_data(file_path, date_column, selected_columns):
    """
    Read CSV file, parse dates, and extract relevant columns.
    
    Args:
        file_path (str): Path to the CSV file.
        date_column (str): Name of the date column in the CSV file.
        selected_columns (list): List of column indices to select from the CSV file.
    
    Returns:
        DataFrame: Processed DataFrame with date parsing and selected columns.
    """
    return pd.read_csv(
        file_path,
        usecols=selected_columns,
        header=0,
        parse_dates=[date_column],
        index_col=date_column,
        date_parser=parse_date
    )

def plot_data(data, title="Real Flow", xlabel="Period (15-minute intervals)", ylabel="Volume (vehicles/period)"):
    """
    Plot the data with appropriate labels and title.
    
    Args:
        data (DataFrame): Data to be plotted.
        title (str): Title of the plot.
        xlabel (str): Label for the x-axis.
        ylabel (str): Label for the y-axis.
    """
    plt.plot(data, '-', label=title)
    plt.legend(loc='upper right')
    plt.xlabel(xlabel)
    plt.ylabel(ylabel)
    plt.show()
