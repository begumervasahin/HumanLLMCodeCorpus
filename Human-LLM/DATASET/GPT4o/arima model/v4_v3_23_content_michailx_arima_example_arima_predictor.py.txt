import numpy as np
from statsmodels.tsa.arima.model import ARIMA

def generate_arima_predictions(historical_data, order=(4, 1, 1), start_index=16, end_index=19):
    """
    Generates future predictions using an ARIMA model.
    
    Parameters:
    historical_data (array-like): The historical time series data.
    order (tuple): The order (p,d,q) for the ARIMA model.
    start_index (int): The start index for prediction.
    end_index (int): The end index for prediction.
    
    Returns:
    array: Predicted values for the specified future time points.
    """
    # Initialize the ARIMA model
    arima_model = ARIMA(historical_data, order=order)
    
    # Fit the ARIMA model to the historical data
    fitted_model = arima_model.fit(disp=0)
    
    # Generate predictions for the specified range
    future_predictions = fitted_model.predict(start=start_index, end=end_index, typ='levels')
    
    return future_predictions

if __name__ == "__main__":
    # Historical time series data
    historical_data = np.array([16, 20, 32, 40, 20, 18, 11, 21, 4, 6, 31, 48, 43, 49, 37]) 
    
    # ARIMA model order (p, d, q)
    arima_order = (4, 1, 1)
    
    # Generate predictions for the next 4 time points
    predictions = generate_arima_predictions(historical_data, order=arima_order)
    
    # Display the predictions
    print("Predicted values for the next 4 time points:", predictions)
