import math
import matplotlib.pyplot as plt
from pandas import read_csv, to_datetime
from statsmodels.tsa.arima.model import ARIMA
from sklearn.metrics import mean_squared_error, mean_absolute_error

def parse_datetime(x):
    """Parse datetime from string format."""
    return to_datetime(x, format='%Y-%m-%d %H:%M:%S')

def mean_absolute_percentage_error(y_true, y_pred):
    """Calculate Mean Absolute Percentage Error (MAPE)."""
    sum_error = sum(math.fabs(pred - true) / true for true, pred in zip(y_true, y_pred))
    return sum_error / len(y_true)

def load_dataset(dataset_path):
    """Load and prepare the dataset."""
    series = read_csv(dataset_path, usecols=[2, 3], header=0, parse_dates=[0], index_col=0, squeeze=True, date_parser=parse_datetime)
    return series.values.astype('float32')

def split_dataset(X, start_index, train_size, test_size):
    """Split the dataset into training and testing sets."""
    train = X[start_index:start_index + train_size]
    test = X[start_index + train_size:start_index + train_size + test_size]
    return train, test

def arima_forecast(history, order=(4, 1, 0)):
    """Perform ARIMA modeling and forecasting."""
    model = ARIMA(history, order=order)
    model_fit = model.fit(disp=0)
    yhat = model_fit.forecast()[0]
    return yhat

def evaluate_model(test, predictions):
    """Evaluate the model performance."""
    error = mean_squared_error(test, predictions)
    mae = mean_absolute_error(test, predictions)
    mape = mean_absolute_percentage_error(test, predictions)
    return mae, error, mape

def plot_results(test, predictions, test_start, windows_size):
    """Plot the results."""
    plt.plot(test[test_start:], '-', label="Real Flow")
    plt.plot(predictions[test_start:], '--', color='red', label="ARIMA")
    plt.legend(loc='upper right')
    plt.xlabel("Period (15-minute intervals)")
    plt.ylabel("Volume (Vehicle/Period)")
    plt.ylim(0, 800)
    plt.show()

def main():
    # File path to the dataset
    dataset_path = '/Users/daihanru/Desktop/arima-lstm/DataSet/FEB15-2.csv'
    
    # Load the dataset
    X = load_dataset(dataset_path)
    
    # Define data splitting indices
    start_index = 2000
    train_size = 1100
    test_size = 500
    line_test_size = 300
    windows_size = 10
    test_start = line_test_size + windows_size - 1
    
    # Split the dataset into training and testing sets
    arima_train, arima_test = split_dataset(X, start_index, train_size, test_size)
    
    # ARIMA modeling and prediction
    history = list(arima_train)
    predictions = []
    
    for t in range(len(arima_test)):
        yhat = arima_forecast(history)
        predictions.append(yhat)
        obs = arima_test[t]
        history.append(obs)
        print(f'Predicted: {yhat}, Expected: {obs}')
    
    # Evaluate the model performance
    mae, error, mape = evaluate_model(arima_test, predictions)
    
    # Print the evaluation metrics
    print(f'ARIMA Test - MAE: {mae:.3f}, MSE: {error:.3f}, RMSE: {math.sqrt(error):.3f}, MAPE: {mape:.3f}')
    
    # Plot the results
    plot_results(arima_test, predictions, test_start, windows_size)

if __name__ == "__main__":
    main()
