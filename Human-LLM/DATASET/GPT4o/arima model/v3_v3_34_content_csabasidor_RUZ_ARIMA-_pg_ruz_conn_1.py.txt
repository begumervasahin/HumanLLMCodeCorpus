def connect_to_database(params):
    """
    Establish a connection to the PostgreSQL database using the provided parameters.

    Parameters:
    params (dict): Dictionary containing database connection parameters.

    Returns:
    connection: psycopg2 connection object if successful, None otherwise.
    """
    try:
        connection = pg.connect(**params)
        print("Connected to the database!")
        return connection
    except Exception as error:
        print(f"Error connecting to the database: {error}")
        return None

def execute_query(connection, query):
    """
    Execute a SQL query on the database and retrieve the results in a Pandas DataFrame.

    Parameters:
    connection: psycopg2 connection object.
    query (str): SQL query to execute.

    Returns:
    pd.DataFrame: DataFrame containing the query results if successful, None otherwise.
    """
    try:
        data_frame = pd.read_sql(query, connection)
        return data_frame
    except Exception as error:
        print(f"Error executing query: {error}")
        return None

def close_connection(connection):
    """
    Close the connection to the PostgreSQL database.

    Parameters:
    connection: psycopg2 connection object.
    """
    try:
        if connection:
            connection.close()
            print("Connection closed.")
    except Exception as error:
        print(f"Error closing the connection: {error}")

    """
    Main function to handle the database connection, query execution, and closing the connection.
    """
    connection = connect_to_database(DB_PARAMS)

            result_df = execute_query(connection, query)

        finally:
            close_connection(connection)
