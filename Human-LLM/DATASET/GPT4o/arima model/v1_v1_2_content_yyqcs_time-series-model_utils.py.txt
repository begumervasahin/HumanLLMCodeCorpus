import torch
import torch.nn as nn
from torch.nn import init

class YourModel(nn.Module):
    def __init__(self):
        super(YourModel, self).__init__()
        self.linear_layer = nn.Linear(in_features=10, out_features=5)  # Adjust the input and output features accordingly

    def forward(self, x):
        # Your model's forward pass implementation
        return self.linear_layer(x)

def initialize_weights(m):
    if isinstance(m, nn.Linear):
        init.xavier_uniform_(m.weight.data)

def main():
    # Create an instance of YourModel
    your_model = YourModel()

    # Apply the weight initialization to the model
    your_model.apply(initialize_weights)

    # Check the initialized weights
    for name, param in your_model.named_parameters():
        if 'weight' in name:
            print(f'Initialized {name} with shape {param.shape}')

    # Example: Use your_model for further training or evaluation
    # Here is a dummy input tensor with a batch size of 2 for demonstration
    dummy_input = torch.randn(2, 10)
    output = your_model(dummy_input)
    print('Output:', output)

if __name__ == "__main__":
    main()
