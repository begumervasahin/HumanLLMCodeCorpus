import torch
import torch.nn as nn
from torch.nn import init

class YourModel(nn.Module):
    def __init__(self):
        super(YourModel, self).__init__()
        self.linear_layer = nn.Linear(in_features=10, out_features=5)  # Adjust the input and output features accordingly

    def forward(self, x):
        # Define the forward pass
        return self.linear_layer(x)

def initialize_weights(m):
    # Apply Xavier uniform initialization to Linear layers
    if isinstance(m, nn.Linear):
        init.xavier_uniform_(m.weight.data)

def main():
    # Create an instance of the model
    model = YourModel()

    # Initialize the model weights
    model.apply(initialize_weights)

    # Print the initialized weights
    for name, param in model.named_parameters():
        if 'weight' in name:
            print(f'Initialized {name} with shape {param.shape}')

    # Example usage: forward pass with dummy input
    dummy_input = torch.randn(2, 10)  # Batch size of 2
    output = model(dummy_input)
    print('Output:', output)

if __name__ == "__main__":
    main()
