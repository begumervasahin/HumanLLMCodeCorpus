import math
import matplotlib.pyplot as plt
from pandas import read_csv
from pandas import datetime

# Function to parse dates from the CSV file
def parser(x):
    return datetime.strptime(x, '%Y-%m-%d %H:%M:%S')

# Function to calculate Mean Absolute Percentage Error (MAPE)
def mean_absolute_percentage_error(y_true, y_pred):
    sum_error = 0
    for i in range(len(y_true)):
        sum_error += math.fabs((y_pred[i] - y_true[i]) / y_true[i])
    return sum_error / len(y_true) * 100  # Convert to percentage

# Replace the file path with your actual CSV file path
csv_file_path = '/Users/daihanru/Desktop/arima-lstm/DataSet/FEB15.csv'

# Read CSV file and parse dates using the defined parser function
series = read_csv(csv_file_path, usecols=[2, 3], header=0, parse_dates=[0],
                  index_col=0, squeeze=True, date_parser=parser)

# Extract the first 1000 values for plotting
X = series.values[0:1000]

# Plot the time series data
plt.plot(X, '-', label="real flow")
plt.legend(loc='upper right')
plt.xlabel("Period (15-minute intervals)")
plt.ylabel("Volume (vehicle/period)")
plt.title("Traffic Volume Time Series")
plt.show()

# Example data for MAPE calculation
y_true = [1, 2, 3, 4, 5]
y_pred = [0.9, 2.1, 3.2, 4.3, 5.1]

# Calculate and print the Mean Absolute Percentage Error (MAPE)
mape = mean_absolute_percentage_error(y_true, y_pred)
print(f"Mean Absolute Percentage Error (MAPE): {mape:.2f}%")
