import math
import matplotlib.pyplot as plt
import pandas as pd
from datetime import datetime

# Define the parser function for date parsing
def parser(x):
    return datetime.strptime(x, '%Y-%m-%d %H:%M:%S')

# Define the Mean Absolute Percentage Error (MAPE) function
def mean_a_p_e(y_true, y_pred):
    """
    Calculate the Mean Absolute Percentage Error (MAPE).
    
    Parameters:
    y_true: list or array of true values
    y_pred: list or array of predicted values
    
    Returns:
    float: MAPE value
    """
    total_error = 0
    for true, pred in zip(y_true, y_pred):
        total_error += abs(pred - true) / true
    return total_error / len(y_true)
