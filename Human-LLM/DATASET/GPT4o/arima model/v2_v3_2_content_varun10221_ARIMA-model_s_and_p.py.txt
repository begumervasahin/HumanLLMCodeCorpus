import os
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import statsmodels.api as sm
from statsmodels.tsa.seasonal import seasonal_decompose
import seaborn as sns

def read_stock_data(file_path):
    """
    Read stock data from a CSV file and preprocess it by converting the Date column
    to datetime, sorting by date, and setting the Date column as the index.
    """
    stock_data = pd.read_csv(file_path)
    stock_data['Date'] = pd.to_datetime(stock_data['Date'])
    stock_data.sort_values(by='Date', inplace=True)
    stock_data.set_index('Date', inplace=True)
    return stock_data

def calculate_statistics(stock_data):
    """
    Calculate various statistical measures for the stock data, including first differences,
    natural log, and variances. Also, prepare the data for ARIMA modeling.
    """
    stock_data['First Difference'] = stock_data['Close'].diff()
    stock_data['Natural Log'] = np.log(stock_data['Close'])
    stock_data['Original Variance'] = stock_data['Close'].rolling(window=30).var()
    stock_data['Log Variance'] = stock_data['Natural Log'].rolling(window=30).var()
    stock_data['Logged First Difference'] = stock_data['Natural Log'].diff()
    stock_data['Lag 20'] = stock_data['Logged First Difference'].shift(20)
    return stock_data

def visualize_data(stock_data):
    """
    Visualize the logged first differences and their ARIMA forecasted values.
    Decompose the log-transformed data seasonally and plot the actual and forecasted values.
    """
    decomposition = seasonal_decompose(stock_data['Natural Log'], model='additive', period=30)
    
    # Fit ARIMA model to the logged first differences
    model = sm.tsa.ARIMA(stock_data['Logged First Difference'].dropna(), order=(0, 0, 1))
    results = model.fit()
    stock_data['Forecast'] = results.fittedvalues

    # Plot the logged first differences and the forecasted values
    plt.figure(figsize=(16, 12))
    plt.plot(stock_data['Logged First Difference'], label='Logged First Difference')
    plt.plot(stock_data['Forecast'], label='Forecast', linestyle='--')
    plt.legend()
    plt.title('Logged First Difference and Forecast')
    plt.xlabel('Date')
    plt.ylabel('Value')
    plt.show()

def main():
    """
    Main function to set seaborn style, read stock data, calculate statistics, and visualize the data.
    """
    sns.set_style('darkgrid')

    # Define the file path for the stock data CSV
    file_path = os.path.join(os.getcwd(), 'q_table.csv')
    
    # Read and preprocess the stock data
    stock_data = read_stock_data(file_path)

    # Calculate required statistical measures
    stock_data = calculate_statistics(stock_data)

    # Visualize the stock data and its forecasts
    visualize_data(stock_data)

if __name__ == "__main__":
    main()
