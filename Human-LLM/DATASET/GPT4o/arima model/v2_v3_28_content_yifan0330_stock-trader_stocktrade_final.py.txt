import stocktrader

def main():
    # Load portfolio and historical stock data
    portfolio_file = 'portfolio.csv'
    stocktrader.loadPortfolio(portfolio_file)
    stocktrader.loadAllStocks()

    # Evaluate portfolio on a specific date
    valuation_date = '2024-03-28'
    evaluate_portfolio(valuation_date)

    # Execute trading strategies
    execute_trading_strategies()

    # Save updated portfolio
    save_updated_portfolio('updated_portfolio.csv')

def evaluate_portfolio(date):
    """
    Evaluate the total value of the portfolio on a specific date.
    
    Args:
        date (str): The date on which to evaluate the portfolio value.
    """
    total_value = stocktrader.valuatePortfolio(date=date, verbose=True)
    print(f"Total portfolio value on {date}: {total_value:.2f}")

def execute_trading_strategies():
    """
    Execute predefined trading strategies.
    """
    print("\nExecuting Strategy 1: Buy as much as possible and sell all")
    stocktrader.tradeStrategy1(verbose=True)

    print("\nExecuting Strategy 2: Buy stocks predicted to increase in price and sell when predicted to decrease")
    stocktrader.tradeStrategy2(verbose=True)

def save_updated_portfolio(filename):
    """
    Save the updated portfolio to a CSV file.
    
    Args:
        filename (str): The name of the file to save the updated portfolio.
    """
    print("\nSaving updated portfolio...")
    stocktrader.savePortfolio(filename)

if __name__ == "__main__":
    main()
