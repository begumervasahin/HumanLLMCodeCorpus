import math
import matplotlib.pyplot as plt
import pandas as pd
from datetime import datetime

# Define a function to parse date strings
def parser(x):
    return datetime.strptime(x, '%Y-%m-%d %H:%M:%S')

# Define a function to calculate the mean absolute percentage error
def mean_absolute_percentage_error(y_true, y_pred):
    total_error = 0
    for i in range(len(y_true)):
        total_error += abs(y_pred[i] - y_true[i]) / y_true[i]
    return total_error / len(y_true)

# Read the CSV file and extract relevant columns
def read_data(file_path):
    series = pd.read_csv(file_path,
                         usecols=[2, 3],
                         header=0,
                         parse_dates=[0],
                         index_col=0,
                         squeeze=True,
                         date_parser=parser)
    return series

# Plot the data
def plot_data(data, title="Traffic Volume over Time"):
    plt.plot(data, '-', label="Real Flow")
    plt.legend(loc='upper right')
    plt.xlabel("Period (15-minute intervals)")
    plt.ylabel("Volume (vehicles/period)")
    plt.title(title)
    plt.show()
