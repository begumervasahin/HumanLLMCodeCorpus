import math
import matplotlib.pyplot as plt
from pandas import read_csv, to_datetime
import statsmodels.tsa.stattools as st

# Function to parse the date column
def parser(x):
    return to_datetime(x, format='%Y-%m-%d %H:%M:%S')

# Function to calculate Mean Absolute Percentage Error (MAPE)
def mean_absolute_percentage_error(y_true, y_pred):
    total_error = 0
    for true_value, pred_value in zip(y_true, y_pred):
        total_error += abs(pred_value - true_value) / true_value
    return total_error / len(y_true)

# Read the CSV file and parse dates
csv_file_path = '/Users/daihanru/Desktop/arima-lstm/DataSet/FEB15-2.csv'
series = read_csv(
    csv_file_path,
    usecols=[2, 3],
    header=0,
    parse_dates=[0],
    index_col=0,
    date_parser=parser
)

# Extract values from the series
values = series.values

# Define the start index and the size of the training data
start_index = 2
training_size = 200
training_data = values[start_index:start_index + training_size]

# Perform Augmented Dickey-Fuller (ADF) test for stationarity
adf_test_result = st.adfuller(training_data)

# Print the ADF test results
print(f'ADF Statistic: {adf_test_result[0]:.6f}')
print(f'p-value: {adf_test_result[1]:.6f}')
print('Critical Values:')
for key, value in adf_test_result[4].items():
    print(f'\t{key}: {value:.3f}')

# If you want to visualize the data (optional)
plt.plot(series)
plt.title('Time Series Data')
plt.xlabel('Date')
plt.ylabel('Values')
plt.show()
