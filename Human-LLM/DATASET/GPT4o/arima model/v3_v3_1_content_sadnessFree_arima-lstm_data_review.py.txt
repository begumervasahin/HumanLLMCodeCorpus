import math
import matplotlib.pyplot as plt
from pandas import read_csv, to_datetime

def parse_datetime(x):
    """Convert a string date-time into a datetime object."""
    return to_datetime(x, format='%Y-%m-%d %H:%M:%S')

def calculate_mape(y_true, y_pred):
    """
    Calculate the Mean Absolute Percentage Error (MAPE).
    
    :param y_true: List of actual values.
    :param y_pred: List of predicted values.
    :return: MAPE value as a percentage.
    """
    total_error = sum(math.fabs((predicted - actual) / actual) for actual, predicted in zip(y_true, y_pred))
    return (total_error / len(y_true)) * 100

def load_time_series_data(csv_file_path, n_values=1000):
    """
    Load and return the time series data from a CSV file.
    
    :param csv_file_path: Path to the CSV file.
    :param n_values: Number of values to extract from the time series data.
    :return: Extracted time series data.
    """
    # Read CSV file and parse dates using the defined parse_datetime function
    series = read_csv(
        csv_file_path, 
        usecols=[2, 3], 
        header=0, 
        parse_dates=[0], 
        index_col=0, 
        squeeze=True, 
        date_parser=parse_datetime
    )
    
    # Extract the first n_values for plotting
    return series.values[:n_values]

def plot_time_series(data):
    """
    Plot the time series data.
    
    :param data: Time series data to plot.
    """
    plt.plot(data, '-', label="Real Flow")
    plt.legend(loc='upper right')
    plt.xlabel("Period (15-minute intervals)")
    plt.ylabel("Volume (vehicle/period)")
    plt.title("Traffic Flow Time Series")
    plt.show()

def main():
    # Replace the file path with the actual CSV file path
    csv_file_path = '/Users/daihanru/Desktop/arima-lstm/DataSet/FEB15.csv'
    
    # Load and plot the time series data
    time_series_data = load_time_series_data(csv_file_path)
    plot_time_series(time_series_data)
    
    # Example of using the calculate_mape function
    # Replace y_true and y_pred with your actual data
    actual_values = [1, 2, 3, 4, 5]
    predicted_values = [0.9, 2.1, 3.2, 4.3, 5.1]
    
    mape = calculate_mape(actual_values, predicted_values)
    print(f"Mean Absolute Percentage Error (MAPE): {mape:.2f}%")

if __name__ == "__main__":
    main()
