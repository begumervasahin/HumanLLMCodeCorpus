import numpy as np
from statsmodels.tsa.arima.model import ARIMA

def generate_arima_predictions(historical_data, order=(4, 1, 1), start_index=16, end_index=19):
    """
    Generate future predictions using an ARIMA model.
    
    Parameters:
    historical_data (array-like): The historical time series data.
    order (tuple): The ARIMA model order (p, d, q).
    start_index (int): The start index for generating predictions.
    end_index (int): The end index for generating predictions.
    
    Returns:
    np.ndarray: Predicted values for the specified future time points.
    """
    # Create and fit the ARIMA model
    model = ARIMA(historical_data, order=order)
    fitted_model = model.fit(disp=0)
    
    # Generate predictions
    predictions = fitted_model.predict(start=start_index, end=end_index, typ='levels')
    
    return predictions

def main():
    # Historical time series data
    historical_data = np.array([16, 20, 32, 40, 20, 18, 11, 21, 4, 6, 31, 48, 43, 49, 37])
    
    # Define ARIMA model parameters
    arima_order = (4, 1, 1)
    
    # Generate predictions
    predictions = generate_arima_predictions(historical_data, order=arima_order)
    
    # Print predictions
    print("Predicted values for the next 4 time points:", predictions)

if __name__ == "__main__":
    main()
