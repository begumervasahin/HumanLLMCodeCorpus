import numpy as np
import pandas as pd
from statsmodels.tsa.arima.model import ARIMA

def main():
    # Read the CSV file into a pandas DataFrame
    SPX = pd.read_csv("SPX Weekly 2000-2017.csv")

    # Display the first 5 rows of the DataFrame
    print("First 5 rows of the SPX DataFrame:")
    print(SPX.head(n=5))

    # Create an ARIMA model with order (0, 1, 0)
    model = ARIMA(SPX, order=(0, 1, 0))

    # Fit the ARIMA model to the data
    model_fit = model.fit()

    # Forecast the next data point using the fitted model
    forecasted_value = model_fit.forecast()[0]

    # Display the forecasted value
    print("Forecasted value:", forecasted_value)

    # Display the first 5 rows of the model summary
    print("First 5 rows of the ARIMA model summary:")
    print(model_fit.summary().tables[1].as_html())  # Convert to HTML for better formatting in some IDEs

if __name__ == "__main__":
    main()
