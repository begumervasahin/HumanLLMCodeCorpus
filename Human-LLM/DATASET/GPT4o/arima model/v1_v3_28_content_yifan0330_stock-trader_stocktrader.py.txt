# Define custom exceptions
class TransactionError(Exception):
    pass

class DateError(Exception):
    pass

class LinAlgError(Exception):
    pass

# Global variables
stocks = {}
portfolio = {}
transactions = []

# Function to normalize date formats
def normalize_date(date_string):
    """
    Convert date_string to 'YYYY-MM-DD' format.
    Accepts three date formats: YYYY-MM-DD, YYYY/MM/DD, and DD.MM.YYYY.
    """
    date_formats = ['%Y-%m-%d', '%Y/%m/%d', '%d.%m.%Y']
    for fmt in date_formats:
        try:
            date_object = datetime.strptime(date_string, fmt)
            return date_object.strftime("%Y-%m-%d")
        except ValueError:
            continue
    raise DateError(f"Invalid date format: {date_string}")

# Function to load the portfolio from a CSV file
def load_portfolio(file_path):
    """
    Load portfolio from a CSV file.
    """
    global portfolio
    try:
        portfolio = pd.read_csv(file_path, index_col=0).to_dict('index')
    except FileNotFoundError:
        raise TransactionError(f"File not found: {file_path}")

# Function to load all stock data
def load_all_stocks(directory='stocks'):
    """
    Load all stock data from CSV files in the given directory.
    """
    global stocks
    for filename in os.listdir(directory):
        if filename.endswith('.csv'):
            stock_name = filename[:-4]
            file_path = os.path.join(directory, filename)
            try:
                stocks[stock_name] = pd.read_csv(file_path, index_col=0, parse_dates=True)
            except Exception as e:
                raise TransactionError(f"Error loading stock data from {file_path}: {e}")

# Function to execute a trading strategy
def execute_trading_strategy(verbose=False):
    """
    Execute a trading strategy and log transactions.
    """
    global transactions
    # Example strategy: Buy one share of each stock if not already in portfolio
    for stock_name, stock_data in stocks.items():
        if stock_name not in portfolio:
            price = stock_data['Close'][-1]
            portfolio[stock_name] = {'shares': 1, 'purchase_price': price}
            transactions.append((stock_name, 'buy', 1, price))
            if verbose:
                print(f"Bought 1 share of {stock_name} at {price}")

# Function to evaluate the portfolio on a given date
def evaluate_portfolio(date, verbose=False):
    """
    Evaluate the portfolio value on the given date.
    """
    date = normalize_date(date)
    total_value = 0
    for stock_name, stock_data in portfolio.items():
        if stock_name in stocks:
            try:
                stock_price = stocks[stock_name].loc[date]['Close']
                stock_value = stock_data['shares'] * stock_price
                total_value += stock_value
                if verbose:
                    print(f"{stock_name}: {stock_data['shares']} shares at {stock_price} each, total value: {stock_value}")
            except KeyError:
                raise TransactionError(f"Data for {stock_name} not available on {date}")
    if verbose:
        print(f"Total portfolio value on {date}: {total_value}")

# Main logic of the program
def main():
    """
    Execute the main logic of the program.
    """
    load_portfolio('portfolio0.csv')
    load_all_stocks()
    execute_trading_strategy(verbose=True)
    evaluate_portfolio(date="2018-03-13", verbose=True)

# Check if the script is being executed directly
if __name__ == '__main__':
    main()
