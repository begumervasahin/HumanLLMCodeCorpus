import sys
import pandas as pd

def test_legitimacy(grouped):
    """
    Check if the max temperature exceeds 100˚F for any given day.
    There are three situations where it does:
    
    Time                min   max    mean
    2011-05-04 00:00:00 53.6  101.8  68.463
    2017-09-01 00:00:00 66.6  107.5  84.266
    2017-09-02 00:00:00 73.4  104.4  86.590
    
    After looking up these dates, we can confirm data legitimacy due to news stories
    that cited high temperatures during these periods.
    """
    for index, row in grouped.iterrows():
        if row["max"] > 100:
            print(f"{index} {row['min']} {row['max']} {row['mean']}")

def fill_nans(grouped):
    """
    Fill NaN values using forward fill method.
    """
    grouped.fillna(method="ffill", inplace=True)
    return grouped

def make_negatives_nans(grouped):
    """
    Convert negative temperatures to NaN, and mean temperatures below 12.46 to NaN.
    """
    grouped["min"] = grouped["min"].apply(lambda x: float("nan") if x < 0 else x)
    grouped["max"] = grouped["max"].apply(lambda x: float("nan") if x < 0 else x)
    grouped["mean"] = grouped["mean"].apply(lambda x: float("nan") if x < 12.46 else x)    
    return grouped

def group_data(df):
    """
    Group the data by day and aggregate min, max, and mean temperatures.
    """
    grouped = df.groupby(pd.Grouper(freq="D"))["TemperatureF"].agg(["min", "max", "mean"])
    return grouped

def process_data(df):
    """
    Process the raw data by grouping, handling NaNs, and adjusting invalid values.
    """
    grouped = group_data(df)
    grouped = make_negatives_nans(grouped)
    grouped = fill_nans(grouped)
    return grouped

def main():
    """
    Main function to read raw data, process it, check for legitimacy, and save the cleaned data.
    """
    try:
        df = pd.read_csv("temp_data_raw.csv", index_col="Time", parse_dates=True)
    except FileNotFoundError:
        print("Please run `scrape_data.py` to generate the raw data first.")
        sys.exit(1)

    df.drop(columns=[col for col in df.columns if "Unnamed" in col], inplace=True)

    grouped = process_data(df)
    test_legitimacy(grouped)
    grouped.to_csv("temp_data_cleaned.csv")

if __name__ == "__main__":
    main()
