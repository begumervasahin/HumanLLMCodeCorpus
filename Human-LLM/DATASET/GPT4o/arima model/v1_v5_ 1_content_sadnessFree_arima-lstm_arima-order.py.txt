import math
import matplotlib.pyplot as plt
import seaborn as sns
import statsmodels.tsa.stattools as st
from pandas import read_csv, to_datetime

def parse_datetime(x):
    """
    Parses a datetime string into a datetime object.
    
    Parameters:
        x (str): The datetime string to parse.
    
    Returns:
        datetime: The parsed datetime object.
    """
    return to_datetime(x, format='%Y-%m-%d %H:%M:%S')

def calculate_mape(y_true, y_pred):
    """
    Calculates the Mean Absolute Percentage Error (MAPE).
    
    Parameters:
        y_true (list): The list of actual values.
        y_pred (list): The list of predicted values.
    
    Returns:
        float: The MAPE value.
    """
    total_error = sum(math.fabs(pred - true) / true for true, pred in zip(y_true, y_pred))
    return total_error / len(y_true)

def main():
    """
    Main function to read the CSV file, extract data, and perform ADF test.
    """
    # File path to the CSV file
    file_path = '/Users/daihanru/Desktop/arima-lstm/DataSet/FEB15-2.csv'
    
    # Reading the CSV file
    series = read_csv(file_path, usecols=[2, 3], header=0, parse_dates=[0],
                      index_col=0, squeeze=True, date_parser=parse_datetime)
    
    # Extracting values from the series
    data_values = series.values
    
    # Setting start index and training size for ARIMA
    start_index = 2
    training_size = 200
    arima_training_data = data_values[start_index:start_index + training_size]
    
    # Performing the Augmented Dickey-Fuller test for stationarity
    adf_result = st.adfuller(arima_training_data)
    
    # Printing the ADF test results
    print('ADF Statistic: %f' % adf_result[0])
    print('p-value: %f' % adf_result[1])
    for key, value in adf_result[4].items():
        print('Critical Values:')
        print('%s: %.3f' % (key, value))

if __name__ == "__main__":
    main()
