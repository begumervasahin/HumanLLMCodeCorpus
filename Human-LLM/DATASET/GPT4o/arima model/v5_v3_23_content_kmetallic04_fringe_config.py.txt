import numpy as np
import tensorflow as tf
from tensorflow.keras.layers import Input, LSTM, Dense
from tensorflow.keras.models import Model
from tensorflow.keras.optimizers import Adam

# Parameters for time series forecasting
SEQUENCE_LENGTH = 168
OUTPUT_SEQUENCE_LENGTH = 168
TOTAL_SEQUENCE_LENGTH = 33600 + OUTPUT_SEQUENCE_LENGTH
INPUT_DIMENSION = 1
OUTPUT_DIMENSION = 1
BATCH_SIZE = 16
TRAINING_STEPS = 100
LEARNING_RATE = 1e-3

# Generate synthetic data for demonstration (replace with actual data loading)
data = np.random.randn(TOTAL_SEQUENCE_LENGTH, INPUT_DIMENSION)

# Prepare training data
X_train = data[:33600]
y_train = data[OUTPUT_SEQUENCE_LENGTH:]

# Reshape data to fit LSTM input shape (samples, time steps, features)
X_train = X_train.reshape(-1, SEQUENCE_LENGTH, INPUT_DIMENSION)
y_train = y_train.reshape(-1, OUTPUT_SEQUENCE_LENGTH, OUTPUT_DIMENSION)

# Build the LSTM model
def build_model(input_shape, output_units):
    input_layer = Input(shape=input_shape)
    lstm_layer = LSTM(units=50)(input_layer)
    output_layer = Dense(units=output_units)(lstm_layer)
    return Model(inputs=input_layer, outputs=output_layer)

model = build_model(input_shape=(SEQUENCE_LENGTH, INPUT_DIMENSION), output_units=OUTPUT_SEQUENCE_LENGTH)

# Compile the model
optimizer = Adam(learning_rate=LEARNING_RATE)
model.compile(optimizer=optimizer, loss='mse')

# Train the model
model.fit(X_train, y_train, batch_size=BATCH_SIZE, epochs=TRAINING_STEPS)
