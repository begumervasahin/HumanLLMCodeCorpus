import math
import matplotlib.pyplot as plt
from pandas import read_csv, to_datetime
import statsmodels.tsa.stattools as st

def parse_datetime(x):
    """
    Parse a datetime string to a datetime object.
    
    Parameters:
    x (str): The datetime string in the format '%Y-%m-%d %H:%M:%S'.
    
    Returns:
    datetime: The parsed datetime object.
    """
    return to_datetime(x, format='%Y-%m-%d %H:%M:%S')

def calculate_mape(y_true, y_pred):
    """
    Calculate the Mean Absolute Percentage Error (MAPE) between actual and predicted values.
    
    Parameters:
    y_true (list): The actual values.
    y_pred (list): The predicted values.
    
    Returns:
    float: The MAPE value.
    """
    total_error = sum(math.fabs(pred - true) / true for true, pred in zip(y_true, y_pred))
    return total_error / len(y_true)

def load_data(file_path):
    """
    Load time series data from a CSV file.
    
    Parameters:
    file_path (str): The path to the CSV file.
    
    Returns:
    Series: The loaded time series data.
    """
    return read_csv(
        file_path,
        usecols=[2, 3],
        header=0,
        parse_dates=[0],
        index_col=0,
        squeeze=True,
        date_parser=parse_datetime
    )

def perform_adf_test(data):
    """
    Perform the Augmented Dickey-Fuller test for stationarity.
    
    Parameters:
    data (array-like): The time series data.
    
    Returns:
    tuple: The ADF test statistic, p-value, and critical values.
    """
    adf_result = st.adfuller(data)
    return adf_result[0], adf_result[1], adf_result[4]

def main():
    # Path to the CSV file
    file_path = '/Users/daihanru/Desktop/arima-lstm/DataSet/FEB15-2.csv'
    
    # Load the data
    series = load_data(file_path)
    
    # Extract values from the series
    data_values = series.values
    
    # Define the start index and size for ARIMA training
    start_index = 2
    training_size = 200
    arima_training_data = data_values[start_index:start_index + training_size]
    
    # Perform Augmented Dickey-Fuller test for stationarity
    adf_stat, p_value, critical_values = perform_adf_test(arima_training_data)
    
    # Display the ADF test results
    print("ADF Statistic:", adf_stat)
    print("p-value:", p_value)
    for key, value in critical_values.items():
        print(f'Critical Value ({key}): {value}')

if __name__ == "__main__":
    main()
