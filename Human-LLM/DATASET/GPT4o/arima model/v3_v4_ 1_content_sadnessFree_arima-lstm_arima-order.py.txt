import math
from datetime import datetime
import matplotlib.pyplot as plt
import pandas as pd
import statsmodels.tsa.stattools as st
import seaborn as sns

def parser(x):
    """
    Parse datetime string to datetime object.
    
    Parameters:
    x (str): datetime string in the format '%Y-%m-%d %H:%M:%S'
    
    Returns:
    datetime: parsed datetime object
    """
    return datetime.strptime(x, '%Y-%m-%d %H:%M:%S')

def mean_absolute_percentage_error(y_true, y_pred):
    """
    Calculate the Mean Absolute Percentage Error (MAPE).
    
    Parameters:
    y_true (list or array): Array of true values
    y_pred (list or array): Array of predicted values
    
    Returns:
    float: MAPE value
    """
    total_error = 0
    for i in range(len(y_true)):
        total_error += math.fabs(y_pred[i] - y_true[i]) / y_true[i]
    return total_error / len(y_true)

def read_series(file_path):
    """
    Read the CSV file and parse the datetime column.
    
    Parameters:
    file_path (str): Path to the CSV file
    
    Returns:
    pandas.Series: Time series data
    """
    return pd.read_csv(
        file_path, 
        usecols=[2, 3], 
        header=0, 
        parse_dates=[0], 
        index_col=0, 
        squeeze=True, 
        date_parser=parser
    )

def perform_adf_test(data):
    """
    Perform the Augmented Dickey-Fuller test for stationarity.
    
    Parameters:
    data (array-like): Time series data for the ADF test
    
    Returns:
    tuple: ADF test result
    """
    adf_result = st.adfuller(data)
    print_adf_result(adf_result)
    return adf_result

def print_adf_result(adf_result):
    """
    Print the results of the Augmented Dickey-Fuller test.
    
    Parameters:
    adf_result (tuple): Result from the ADF test
    """
    print('ADF Statistic: %f' % adf_result[0])
    print('p-value: %f' % adf_result[1])
    print('Critical Values:')
    for key, value in adf_result[4].items():
        print('\t%s: %.3f' % (key, value))

def plot_time_series(series):
    """
    Plot the time series data.
    
    Parameters:
    series (pandas.Series): Time series data
    """
    plt.figure(figsize=(10, 6))
    plt.plot(series.index, series.values)
    plt.title('Time Series Data')
    plt.xlabel('Date')
    plt.ylabel('Values')
    plt.show()

def main():
    # Define the path to the CSV file
    file_path = '/Users/daihanru/Desktop/arima-lstm/DataSet/FEB15-2.csv'
    
    # Read the time series data
    series = read_series(file_path)
    
    # Extract values from the series
    X = series.values
    
    # Set the starting point and size for ARIMA training data
    start = 2
    size = 200
    arima_train_data = X[start:start + size]
    
    # Perform the Augmented Dickey-Fuller test for stationarity
    perform_adf_test(arima_train_data)
    
    # Optionally, visualize the time series data
    plot_time_series(series)

if __name__ == "__main__":
    main()
