# Import the pandas library and assign it an alias 'pd'
import pandas as pd

# Define a function named 'load_data' that takes a parameter 'data_name'
def load_data(data_name):
    # Use pandas to read a CSV file specified by 'data_name' and store it in a variable called 'file'
    file = pd.read_csv(data_name)
    
    # Return the loaded data
    return file

# Example usage:
# Specify the name or path of your CSV file in 'data_file_name'
data_file_name = "your_data_file.csv"

# Call the 'load_data' function with 'data_file_name' as an argument and store the result in 'loaded_data'
loaded_data = load_data(data_file_name)

# Print the first few rows of the loaded data
print("Preview of the loaded data:")
print(loaded_data.head())
