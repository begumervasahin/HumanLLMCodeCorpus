# Import necessary libraries
import numpy as np
import pandas as pd
from statsmodels.tsa.arima.model import ARIMA

# Read the CSV file into a pandas DataFrame
SPX = pd.read_csv("SPX Weekly 2000-2017.csv")

# Display the first 5 rows of the DataFrame
print("First 5 rows of the SPX DataFrame:")
print(SPX.head(n=5))

# Create an ARIMA model with order (0, 1, 0)
arima_order = (0, 1, 0)
arima_model = ARIMA(SPX, order=arima_order)

# Fit the ARIMA model to the data
fitted_model = arima_model.fit()

# Forecast the next data point using the fitted model
forecasted_value = fitted_model.forecast()[0]

# Display the forecasted value
print("Forecasted next data point:", forecasted_value)

# Display the first 5 rows of the ARIMA model summary
# Note: The summary() method provides a comprehensive summary of the model fit, which includes several tables.
# Here, we are displaying the second table (index 1), which typically contains the coefficients and statistics.
print("First 5 rows of the ARIMA model summary:")
print(fitted_model.summary().tables[1].as_html())
