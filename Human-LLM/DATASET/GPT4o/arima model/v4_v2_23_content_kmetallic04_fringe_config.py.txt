import numpy as np
import tensorflow as tf
from tensorflow.keras.layers import Input, LSTM, Dense
from tensorflow.keras.models import Model
from tensorflow.keras.optimizers import Adam

# Set parameters for time series forecasting
sequence_length = 168
output_sequence_length = 168
total_sequence_length = 33600 + output_sequence_length
input_dimension = 1
output_dimension = 1
batch_size = 16
training_steps = 100
learning_rate = 1e-3
p, d, q = 24, 1, 2
order = (p, d, q)

# Generate synthetic data for demonstration (replace with your actual data loading)
data = np.random.randn(total_sequence_length, input_dimension)

# Prepare data for training
X_train = data[:33600]
y_train = data[output_sequence_length:]

# Reshape data to fit the LSTM input shape (samples, time steps, features)
X_train = X_train.reshape(-1, sequence_length, input_dimension)
y_train = y_train.reshape(-1, output_sequence_length, output_dimension)

# Build the LSTM model
input_layer = Input(shape=(sequence_length, input_dimension))
lstm_layer = LSTM(50)(input_layer)
output_layer = Dense(output_sequence_length)(lstm_layer)

model = Model(inputs=input_layer, outputs=output_layer)
optimizer = Adam(learning_rate=learning_rate)
model.compile(optimizer=optimizer, loss='mse')

# Train the model
model.fit(X_train, y_train, batch_size=batch_size, epochs=training_steps)

# model.save("sarima_model.h5")

print("Model training complete.")
