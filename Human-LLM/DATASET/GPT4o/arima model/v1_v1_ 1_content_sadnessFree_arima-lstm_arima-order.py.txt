import math
import matplotlib.pyplot as plt
from pandas import read_csv
from pandas import to_datetime
import statsmodels.tsa.stattools as st

# Function to parse the date column
def parser(x):
    return to_datetime(x, format='%Y-%m-%d %H:%M:%S')

# Function to calculate Mean Absolute Percentage Error (MAPE)
def mean_a_p_e(y_true, y_pred):
    total_error = 0
    for i in range(len(y_true)):
        total_error += math.fabs(y_pred[i] - y_true[i]) / y_true[i]
    return total_error / len(y_true)

# Read the CSV file
series = read_csv('/Users/daihanru/Desktop/arima-lstm/DataSet/FEB15-2.csv', usecols=[2, 3], header=0, parse_dates=[0],
                  index_col=0, date_parser=parser)

# Extract values from the series
X = series.values

# Define start and size for training data
start = 2
size = 200
arima_train = X[start:start + size]

# Print ADF (Augmented Dickey-Fuller) test result for stationarity
adf_test_result = st.adfuller(arima_train)
print('ADF Statistic: %f' % adf_test_result[0])
print('p-value: %f' % adf_test_result[1])
print('Critical Values:')
for key, value in adf_test_result[4].items():
    print('\t%s: %.3f' % (key, value))
