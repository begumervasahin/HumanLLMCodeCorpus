import re
from tqdm import tqdm

def preprocess_tweet(tweet):
    """Preprocess a tweet by replacing specific patterns with placeholders."""
    new_tweet = []
    for word in tweet.split():
        if re.match(r'^.*@.*', word):
            word = '<NAME/>'
        elif re.match(r'^.*http://.*', word):
            word = '<LINK/>'
        word = word.replace('&quot;', ' " ')
        word = word.replace('&amp;', ' & ')
        word = word.replace('&gt;', ' > ')
        word = word.replace('&lt;', ' < ')
        new_tweet.append(word)
    return ' '.join(new_tweet).strip()

def process_dataset(input_file, pos_output_file, neg_output_file):
    """Process the input dataset and write positive and negative tweets to separate files."""
    try:
        with open(input_file, "r") as full_dataset, \
             open(pos_output_file, "w") as pos_dataset, \
             open(neg_output_file, "w") as neg_dataset:

            csv_lines = full_dataset.readlines()

            for line in tqdm(csv_lines, desc="Processing Tweets"):
                line_parts = line.split(",", 3)
                tweet = line_parts[3].strip()
                processed_tweet = preprocess_tweet(tweet) + '\n'

                if line_parts[1].strip() == '1':
                    pos_dataset.write(processed_tweet)
                else:
                    neg_dataset.write(processed_tweet)

    except IOError:
        print("Failed to open file")

if __name__ == "__main__":
    dataset_url = "http://thinknook.com/wp-content/uploads/2012/09/Sentiment-Analysis-Dataset.zip"
    input_file = "twitter-sentiment-dataset/sentiment-dataset.csv"
    pos_output_file = "twitter-sentiment-dataset/tw-data.pos"
    neg_output_file = "twitter-sentiment-dataset/tw-data.neg"
    
    process_dataset(input_file, pos_output_file, neg_output_file)
