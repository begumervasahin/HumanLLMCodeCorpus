import json
import jieba.posseg as pseg
from pyspark import SparkContext

# Initialize Spark context
sc = SparkContext(appName='CutDocument')

def cut(text):
    """
    Tokenizes the input text and filters the tokens based on allowed POS tags.

    Args:
    text (str): The text to be tokenized.

    Returns:
    list: A list of filtered tokens.
    """
    seg_list = list(pseg.cut(text))
    allowed_flags = {
        "n", "nr", "nr1", "nr2", "nrj", "nrf", "ns", "nsf", "nt", "nz", "nl", "ng",
        "s", "f", "v", "vd", "vn", "vf", "vx", "vi", "vl", "vga", "ad", "an", "ag", "al"
    }
    filtered_list = [word for word, flag in seg_list if flag in allowed_flags]
    return filtered_list

def output(items):
    """
    Writes the list of tokens to a file.

    Args:
    items (list): The list of tokens to be written to the file.
    """
    filename = "/vagrant/vocabulary/mominal.txt"
    join = " ".join(items)
    with open(filename, "a+", encoding="utf-8") as fo:
        fo.write(join)
        fo.write("\n")

def filter_stopword(tokens):
    """
    Filters out stop words from the list of tokens.

    Args:
    tokens (list): The list of tokens to be filtered.

    Returns:
    list: The filtered list of tokens.
    """
    return [token for token in tokens if token not in stopWordRdd]

# Load the stop words from file
stop_words_file = "stop_words.txt"
stopWordRdd = set(sc.textFile(stop_words_file).map(lambda x: x.strip()).collect())

# Load and process the source file
src_file = '/vagrant/data/data.txt'
rdd = sc.textFile(src_file).map(lambda x: json.loads(x)['content'])

# Apply tokenization, stop word filtering, and write the results to a file
rdd.map(cut).map(filter_stopword).foreach(output)

# Stop the Spark context
sc.stop()
