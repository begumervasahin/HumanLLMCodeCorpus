import wikipedia
from nltk.stem import WordNetLemmatizer
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from sklearn.feature_extraction.text import TfidfVectorizer
import os

# Initialize the lemmatizer and stop words
lemmatizer = WordNetLemmatizer()
stop_words = set(stopwords.words("english"))

# Define the Wikipedia pages to process
pages_array = ['Alfa_Romeo', 'Aston_Martin', 'Audi', 'Automobile_Dacia', 'Bentley']

# Create directory for storing generated text files if it doesn't exist
output_dir = "Generated_txt"
os.makedirs(output_dir, exist_ok=True)

# Get the number of top keywords from user input
keyword_number = int(input('Broj željenih najfrekventnijih ključnih riječi po dokumentu: '))

# Initialize a dictionary to store top keywords for each page
dictionary = {}

# Process each Wikipedia page
for page_name in pages_array:
    # Fetch the page content
    page = wikipedia.page(page_name)
    document = page.content
    
    # Save the raw content to a text file
    raw_file_path = os.path.join(output_dir, f"{page_name}.txt")
    with open(raw_file_path, 'w', encoding='utf-8') as file:
        file.write(document)
    
    # Read the content and tokenize it
    with open(raw_file_path, "r", encoding='utf-8') as file:
        contents = file.read()
        words = word_tokenize(contents)
    
    # Convert words to lowercase and filter out non-alphabetic and non-numeric tokens
    words_lowercase = [word.lower() for word in words if word.isalpha() or word.isnumeric()]
    
    # Lemmatize the words and remove stop words
    lemmatized_words = [lemmatizer.lemmatize(word, pos="v") for word in words_lowercase if word not in stop_words]
    
    # Save the lemmatized content to a text file
    lemmatized_file_path = os.path.join(output_dir, f"{page_name}_Lemmatized.txt")
    with open(lemmatized_file_path, 'w', encoding='utf-8') as file:
        file.write(", ".join(lemmatized_words))
    
    # Print the number of lemmatized words
    print(f"{page_name:30} Broj riječi: {len(lemmatized_words):5}")
    
    # Calculate TF-IDF weights for the top keywords
    vectorizer = TfidfVectorizer(max_features=keyword_number)
    response = vectorizer.fit_transform([" ".join(lemmatized_words)])
    
    # Save the TF-IDF weights to a text file
    weights_file_path = os.path.join(output_dir, f"{page_name}_Weights.txt")
    with open(weights_file_path, 'w', encoding='utf-8') as file:
        for col in response.nonzero()[1]:
            file.write(f"{response[0, col]} - {vectorizer.get_feature_names_out()[col]}\n")
    
    # Save the top keywords to a text file
    top_keywords_file_path = os.path.join(output_dir, f"{page_name}_Top_Keywords.txt")
    top_keywords = [vectorizer.get_feature_names_out()[col] for col in response.nonzero()[1]]
    with open(top_keywords_file_path, 'w', encoding='utf-8') as file:
        file.write("\n".join(top_keywords))
    
    # Store the top keywords in the dictionary
    dictionary[page_name] = top_keywords
    
    # Print the top keywords
    print("\n".join(top_keywords))
    print("\n")

# Print the final report
print("="*88)
print("REPORT")
print("Preuzet tekst pojedine stranice sa Wikipedije, odrađena lematizacija i izbacivanje stop riječi, izračunate težine za", keyword_number, "najfrekventnijih ključnih riječi. Generirano 4 .txt dokumenta za pojedinu Wikipedia stranicu.")
print("="*88)
print("\n\n")
