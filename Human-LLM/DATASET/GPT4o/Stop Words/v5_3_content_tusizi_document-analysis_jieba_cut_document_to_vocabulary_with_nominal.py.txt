import json
import jieba.posseg as pseg
from pyspark import SparkContext

def cut(text):
    """Segment text and filter by specific part-of-speech tags."""
    seg_list = list(pseg.cut(text))
    allowed_pos = {"n", "nr", "nr1", "nr2", "nrj", "nrf", "ns", "nsf", "nt", "nz", "nl", "ng", "s", "f", "v", "vd", "vn", 
                   "vf", "vx", "vi", "vl", "vga", "ad", "an", "ag", "al"}
    filtered_words = [word for word, flag in seg_list if flag in allowed_pos]
    encoded_words = [word.encode("utf-8") for word in filtered_words]
    return encoded_words

def output(items):
    """Write segmented items to a file."""
    filename = "/vagrant/vocabulary/nominal.txt"
    with open(filename, "a+", encoding='utf-8') as file:
        file.write(" ".join(items) + "\n")

def filter_stopwords(words, stopwords):
    """Filter out stopwords from the list of words."""
    return [word for word in words if word not in stopwords]

if __name__ == "__main__":
    sc = SparkContext(appName='CutDocument')

    # Load data and stopwords
    src_file = '/vagrant/data/data.txt'
    rdd = sc.textFile(src_file).map(lambda x: json.loads(x)['content'])
    stopwords = set(sc.textFile("stop_words.txt").map(lambda x: x.encode("utf-8")).collect())

    # Process data
    rdd.map(cut).map(lambda words: filter_stopwords(words, stopwords)).foreach(output)

    sc.stop()
