import urllib.request as urllib2
from bs4 import BeautifulSoup
from nltk.tokenize import RegexpTokenizer
from collections import Counter

# Initialize lists to store Wikipedia URLs and their titles
wiki_urls = []
wiki_titles = []

# URL of the Wikipedia page with selected articles
wiki_page_url = "https://az.wikipedia.org/wiki/Vikipediya:Seçilmiş_məqalələr"

# Open the Wikipedia page and parse its content
page = urllib2.urlopen(wiki_page_url)
soup = BeautifulSoup(page, 'html.parser')

# Find the table containing the selected articles
table_row = soup.find('div', class_='mw-parser-output').find('tr')

# Extract all the table cells from the second table row
table_cells = table_row.find_next('tr').find_all('td')

# Extract URLs and titles of the articles
for cell in table_cells:
    links = cell.find_all('a')
    for link in links:
        if link.get('title'):
            wiki_urls.append('https://az.wikipedia.org' + link['href'])
            wiki_titles.append(link.get('title'))

# List of test URLs for demonstration purposes
wiki_urls_test = [
    'https://az.wikipedia.org/wiki/GRB_970508',
    'https://az.wikipedia.org/wiki/Günəş_sistemi'
]

# Initialize list to store tokenized words
tokenized_words = []

# Tokenize the content of each Wikipedia page
for i, url in enumerate(wiki_urls, start=1):
    print(f"Processing article {i}")

    # Open and parse the content of the Wikipedia page
    page_single = urllib2.urlopen(url)
    soup_single = BeautifulSoup(page_single, 'html.parser')
    content = soup_single.find('div', class_='mw-parser-output').get_text()

    # Tokenize the text content
    tokenizer = RegexpTokenizer(r'\w+')
    tokenized_text = tokenizer.tokenize(content)
    tokenized_words += tokenized_text

# Count the frequency of each tokenized word
word_count = Counter(tokenized_words)

# Write the word counts to a file
with open('stopwordsAZE.txt', 'a') as file:
    file.write(str(word_count))
