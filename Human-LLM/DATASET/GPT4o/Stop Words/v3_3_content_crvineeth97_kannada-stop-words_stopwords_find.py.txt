import nltk
nltk.download('punkt')
from textblob import TextBlob as tb
import json

class TfIdf:
    def __init__(self, corpus_paths):
        """
        Initialize the TfIdf object with a list of corpus file paths.
        """
        self.corpus_paths = corpus_paths
        self.corpus = []
        self.word_df_dict = {}
        self.blob_list = []

    def setup(self):
        """
        Setup the TfIdf object by loading the corpus files, building the corpus,
        and calculating word frequencies.
        """
        self.load_corpus_files()
        self.build_corpus()
        self.calculate_word_frequency()

    def load_corpus_files(self):
        """
        Load the JSON files from the provided paths and extend the corpus list.
        """
        for path in self.corpus_paths:
            with open(path, 'r', encoding='utf-8') as file:
                self.corpus.extend(json.load(file))

    def build_corpus(self):
        """
        Build the corpus by concatenating the content of each document and creating
        TextBlob objects.
        """
        for document in self.corpus:
            content = '. '.join(document['content'])
            content = content.replace('..', '.')
            self.blob_list.append(tb(content))

    def calculate_word_frequency(self):
        """
        Calculate the word frequency across all documents in the corpus.
        """
        for blob in self.blob_list:
            for word in set(blob.words):
                if word not in self.word_df_dict:
                    self.word_df_dict[word] = 0
                self.word_df_dict[word] += 1

    def write_to_file(self, filename="stop-words.txt", top_n=102):
        """
        Write the top N most frequent words to a file.
        
        Parameters:
        - filename: The name of the output file (default is "stop-words.txt").
        - top_n: The number of top frequent words to write (default is 102).
        """
        with open(filename, 'w', encoding='utf-8') as out_file:
            for word, freq in sorted(self.word_df_dict.items(), key=lambda x: x[1], reverse=True)[:top_n]:
                out_file.write(f"{word}\n")

# Specify the paths to the corpus files
corpus_paths = ["udayavani.json"]

# Create a TfIdf object
tfidf = TfIdf(corpus_paths)

# Setup the TfIdf object by loading the corpus files, building the corpus, and calculating word frequencies
tfidf.setup()

# Write the top 102 most frequent words to a file
tfidf.write_to_file()
