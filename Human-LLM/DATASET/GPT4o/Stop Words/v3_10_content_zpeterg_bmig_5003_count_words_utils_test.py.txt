import argparse

class Options:
    def __init__(self, file, start, stop, finish, format, output, stats, csv):
        self.file = file
        self.start = start
        self.stop = stop
        self.finish = finish
        self.format = format
        self.output = output
        self.stats = stats
        self.csv = csv

    def to_object(self):
        return {
            'file': self.file,
            'start': self.start,
            'stop': self.stop,
            'finish': self.finish,
            'format': self.format,
            'output': self.output,
            'stats': self.stats,
            'csv': self.csv,
        }

def dealArgs(args):
    parser = argparse.ArgumentParser(description="Process some arguments.")
    parser.add_argument('--input', required=True, help="Input file name")
    parser.add_argument('--start', required=True, help="Start word")
    parser.add_argument('--stop', required=True, help="Stop word")
    parser.add_argument('--finish', required=True, help="Finish word")
    parser.add_argument('--output', required=True, help="Output file name (without extension)")
    parser.add_argument('-s', action='store_true', dest='stats', help="Include statistics")
    parser.add_argument('-f', action='store_false', dest='format', help="Format flag")

    parsed_args = parser.parse_args(args)
    
    return Options(
        file=parsed_args.input,
        start=parsed_args.start,
        stop=parsed_args.stop,
        finish=parsed_args.finish,
        format=parsed_args.format,
        output=f'{parsed_args.output}.json',
        stats=parsed_args.stats,
        csv=False  # Default value for csv
    )

def cleanWord(word):
    return ''.join(filter(str.isalnum, word))

import unittest
from utils import dealArgs, cleanWord

class UtilsTest(unittest.TestCase):
    def setUp(self):
        self.args = [
            '--input=small_test.txt',
            '--start=foo',
            '--stop=bar',
            '--finish=enough',
            '--output=output',
            '-s',
            '-f',
        ]

    def test_deal_args(self):
        expected_result = {
            'file': 'small_test.txt',
            'start': 'foo',
            'stop': 'bar',
            'finish': 'enough',
            'format': False,
            'output': 'output.json',
            'stats': True,
            'csv': False,
        }
        options = dealArgs(self.args)
        self.assertEqual(expected_result, options.to_object())

    def test_change_format(self):
        options = dealArgs(self.args)
        options.format = True
        self.assertEqual(True, options.format)

    def test_change_format_override(self):
        options = dealArgs(self.args)
        options.format = 'foo'
        self.assertEqual(False, options.format)

    def test_change_file(self):
        options = dealArgs(self.args)
        options.file = 'foo.txt'
        self.assertEqual('foo.txt', options.file)

    def test_change_output_override(self):
        options = dealArgs(self.args)
        options.output = 'foo'
        self.assertEqual('foo.json', options.output)

    def test_change_csv_override(self):
        options = dealArgs(self.args)
        options.csv = True
        self.assertEqual(True, options.csv)
        self.assertEqual(False, options.format)

    def test_change_stats(self):
        options = dealArgs(self.args)
        options.stats = True
        self.assertEqual(True, options.stats)
        self.assertEqual(False, options.format)

    def test_change_stats_override(self):
        options = dealArgs(self.args)
        options.stats = 'foo'
        self.assertEqual(False, options.stats)

    def test_deal_no_args(self):
        with self.assertRaises(SystemExit):
            dealArgs([])

    def test_clean_word(self):
        words = ['it', 'is', 't1me', 'for', 'a11', 'g00d', '4', '@', '#']
        expected_result = ['it', 'is', 't1me', 'for', 'a11', 'g00d', '4', '', '']
        cleaned_words = [cleanWord(word) for word in words]
        self.assertEqual(expected_result, cleaned_words)

if __name__ == '__main__':
    unittest.main()
