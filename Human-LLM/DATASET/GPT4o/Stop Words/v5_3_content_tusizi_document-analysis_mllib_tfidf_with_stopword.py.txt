import json
import jieba
from pyspark import SparkContext
from sklearn.feature_extraction.text import CountVectorizer, TfidfTransformer

def compute_tfidf(corpus, output_file):
    """
    Compute TF-IDF for the given corpus and write the results to a file.
    
    :param corpus: List of documents in the corpus
    :param output_file: Path to the output file
    """
    # Initialize the vectorizer and transformer
    vectorizer = CountVectorizer()
    transformer = TfidfTransformer()

    # Fit and transform the corpus
    tfidf_matrix = transformer.fit_transform(vectorizer.fit_transform(corpus))

    # Get the feature names and weights
    words = vectorizer.get_feature_names_out()
    weights = tfidf_matrix.toarray()

    # Write the results to the output file
    with open(output_file, "a+", encoding="utf-8") as fo:
        for doc_weights in weights:
            for word, weight in zip(words, doc_weights):
                record = json.dumps([word, weight], ensure_ascii=False)
                fo.write(record + "\n")

def cut_words(text):
    """
    Cut the given text into words using jieba.
    
    :param text: The input text
    :return: The cut text with words separated by spaces
    """
    return " ".join(jieba.cut(text, cut_all=True))

def main(input_file, output_file):
    """
    Main function to read data, process it, and compute TF-IDF.
    
    :param input_file: Path to the input file containing the data
    :param output_file: Path to the output file to write the results
    """
    # Initialize SparkContext
    sc = SparkContext(appName='DocumentsRewrite')

    # Read and process the input data
    rdd = sc.textFile(input_file).map(lambda x: json.loads(x)['content'])
    corpus = rdd.map(cut_words).collect()

    # Compute TF-IDF and write the results
    compute_tfidf(corpus, output_file)

if __name__ == "__main__":
    input_file = '/vagrant/data/data.txt'
    output_file = '/vagrant/vocabulary/mllib.txt'
    main(input_file, output_file)
