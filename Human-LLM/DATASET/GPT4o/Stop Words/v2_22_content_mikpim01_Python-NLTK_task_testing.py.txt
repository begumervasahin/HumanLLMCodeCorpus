import os
import sys
from textract import process
from nltk import wordpunct_tokenize
from nltk.corpus import stopwords
from stopword_removal import remove_stopwords, perform_imports

def get_filenames_from_directory(directory):
    """
    Reads the names of all files in the specified directory and returns them as a list.
    
    :param directory: Path to the directory from which the files must be read.
    :return: List of filenames in the directory.
    """
    files = []
    for dirpath, dirnames, filenames in os.walk(directory):
        files.extend(filenames)
        break
    return files

def verify_no_stopwords(infile, language):
    """
    Checks if there are any stop words from the specified language in the file.
    
    :param infile: File whose text needs to be checked.
    :param language: Language whose stop words need to be checked.
    :return: True if no stopwords are found, False otherwise.
    """
    # Extract text from the input file
    text = process(infile).decode('UTF-8')
    
    # Tokenize the text
    tokens = wordpunct_tokenize(text)
    
    # Get the list of stop words for the specified language
    stop_words = stopwords.words(language)
    
    # Check if any stop words are in the tokens
    for word in stop_words:
        if word in tokens:
            return False
    return True

def run_test(input_folder):
    """
    Tests the stopword_removal function using sample files from the input folder.
    
    :param input_folder: Folder containing the sample PDF files.
    """
    if not input_folder:
        print("Please specify the folder for input sample files.")
        sys.exit(1)

    # Ensure all necessary imports are performed
    perform_imports()
    
    # Get the list of sample files
    samples = get_filenames_from_directory(input_folder)
    count = 0

    # Process each sample file
    for sample in samples:
        input_file = os.path.join(input_folder, sample)
        output_file = os.path.join(input_folder, sample[:-3] + 'txt')
        
        # Remove stop words and save to the output file
        remove_stopwords(in_file=input_file, out_file=output_file)
        
        # Determine the language from the filename (assumes format "language-filename.pdf")
        language = sample.split('.')[0].split('-')[0]
        
        # Verify no stopwords in the output file
        if verify_no_stopwords(infile=output_file, language=language):
            count += 1

    # Print the test results
    if count < len(samples):
        print(f'Test failed for {len(samples) - count} files')
    else:
        print('All tests passed')

    # Clean up the generated text files
    for sample in samples:
        try:
            os.remove(os.path.join(input_folder, sample[:-3] + 'txt'))
        except FileNotFoundError:
            pass

if __name__ == '__main__':
    run_test(input_folder='input_files')
