import logging
import re
from pymystem3 import Mystem

# Set up logging
logging.basicConfig(level=logging.DEBUG, format='%(asctime)s %(message)s', handlers=[logging.StreamHandler()])
logging.info("Loading Mystem")
mystem = Mystem()
logging.info("Mystem loaded")

def parse_gr(grammar):
    """Extract and yield grammar options from the given string."""
    options_match = re.search(r'\(([^)]*)\)', grammar, re.IGNORECASE)
    if options_match:
        options = options_match.group(1)
        for option in options.split('|'):
            yield grammar.replace(f"({options})", option)
    else:
        yield grammar

# Read input file and analyze text
input_filepath = "data/test.txt"
output_filepath = "data/pairs_with_grammar.tsv"
relations_filepath = "data/relations.pairs.tsv"

lines = set()
with open(input_filepath, "r") as input_file:
    logging.info("File opened for reading")
    for line in input_file:
        analyzed_words = mystem.analyze(line)
        for word in analyzed_words:
            if 'analysis' in word:
                for analysis_item in word['analysis']:
                    for grammar_info in parse_gr(analysis_item['gr']):
                        line_entry = "\t".join([grammar_info, analysis_item['lex'], word['text'].lower()])
                        lines.add(line_entry.encode("utf-8") + b"\n")

# Write analyzed data to output file
with open(output_filepath, "w+") as output_file:
    for line in lines:
        output_file.write(line.decode("utf-8"))

# Read pairs from output file and build relations
grammar_dict = {}
with open(output_filepath, "r") as input_file:
    for line in input_file:
        if line.strip():
            desc, normal, form = line.strip().split("\t")
            if desc not in grammar_dict:
                grammar_dict[desc] = []
            grammar_dict[desc].append((normal, form))
logging.info("Pairs acquired")

# Write relations to output file
with open(relations_filepath, "w") as output_file:
    for desc, pairs in grammar_dict.items():
        for pair1 in pairs:
            for pair2 in pairs:
                if pair1 != pair2:
                    output_file.write("\t".join([pair1[0], pair1[1], pair2[0], pair2[1]]) + "\n")
logging.info("Relations pairs acquired")
