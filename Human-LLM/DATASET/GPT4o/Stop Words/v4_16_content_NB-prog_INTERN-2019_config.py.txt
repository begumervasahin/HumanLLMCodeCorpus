# Configuration for SkipGram model training

# Path to the text file used for training the SkipGram model
corpus_file = 'corpus.txt'

# Maximum vocabulary size: SkipGram is trained on the most frequent `max_vocab_size` words from the corpus.
# To include all words, set this to a very high value like 1e12.
max_vocab_size = 50000

# Maximum corpus size: SkipGram is trained using only the first `max_corpus_size` words from the corpus_file.
# To use the entire corpus, set this to a very high value like 1e12.
max_corpus_size = 1e12

# Frequency-based subsampling threshold: if set, words are randomly removed from the corpus with a probability:
# discard_probability = 1 - (fw_subsampling_th / word_freq)^0.5
fw_subsampling_th = None

# Embedding dimensions: the size of the word vectors
embedding_dims = 50

# Number of negative samples: for each context word, `neg_samples` negative words are sampled from the distribution:
# neg_sampling_dis = (word_freq^0.75) / (sum(word_freq^0.75))
neg_samples = 20

# Number of training epochs
epochs = 3

# Mini-batch size for training
mini_batch_size = 256

# Window size: the context window size for SkipGram
window_size = 5

# Continue training: if True, SkipGram continues training from the previous state
cont_traning = False

# Number of words to plot in the visualization script (plot.py)
words_to_plot = 1000
