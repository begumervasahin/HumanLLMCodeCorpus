import codecs
import win_unicode_console
from nltk.tokenize import sent_tokenize, word_tokenize

# Enable Unicode console support on Windows
win_unicode_console.enable()

def read_file(file_path):
    """
    Reads a text file and returns its content as a single string.
    
    Parameters:
    file_path (str): The path to the text file.
    
    Returns:
    str: The content of the file as a single string.
    """
    with codecs.open(file_path, 'r', encoding='utf-8') as file:
        text = ""
        for line in file:
            text += " " + line.strip()
    return text

def tokenize_text(text):
    """
    Tokenizes the text into sentences and words.
    
    Parameters:
    text (str): The text to be tokenized.
    
    Returns:
    tuple: A tuple containing a list of sentences and a list of word tokens for each sentence.
    """
    sentences = sent_tokenize(text)
    word_tokens = [word_tokenize(sentence) for sentence in sentences]
    return sentences, word_tokens

def display_sentence_tokens(sentences, word_tokens, sentence_number):
    """
    Displays the tokens of the specified sentence.
    
    Parameters:
    sentences (list): A list of sentences.
    word_tokens (list): A list of word tokens for each sentence.
    sentence_number (int): The index of the sentence to display.
    """
    print("\n[Sentence {}] -> {}\n".format(sentence_number, sentences[sentence_number]))
    tokens = word_tokens[sentence_number]
    
    for i, token in enumerate(tokens):
        print("Token {} -> {}".format(i, token))
    
    print("\nTotal word tokens in sentence {}: {} tokens".format(sentence_number, len(tokens)))
    print("\nTotal sentences: {}".format(len(sentences) - 1))

def main():
    file_path = 'texto.txt'
    text = read_file(file_path)
    sentences, word_tokens = tokenize_text(text)
    
    num_sentences = len(sentences)
    print("\nThe text contains {} sentences.\nStarting from 0 to {}\n".format(num_sentences, num_sentences - 1))
    
    try:
        sentence_number = int(input("Enter the sentence number: "))
    except ValueError:
        print("\n\nInvalid sentence number")
        return
    
    if 0 <= sentence_number < num_sentences:
        display_sentence_tokens(sentences, word_tokens, sentence_number)
    else:
        print("\n\nInvalid sentence number")

if __name__ == "__main__":
    main()
