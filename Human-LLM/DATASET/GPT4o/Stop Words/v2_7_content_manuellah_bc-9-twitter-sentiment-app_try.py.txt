def get_all_tweets(screen_name):
    """
    Fetch all tweets from a specific Twitter user.

    Args:
        screen_name (str): Twitter handle of the user.

    Returns:
        dict: A dictionary where keys are tweet IDs and values are tuples of tweet text and creation date.
    """
    tweet_dict = {}
    all_tweets = []

    # Request the most recent 200 tweets
    new_tweets = api.user_timeline(screen_name=screen_name, count=200)
    all_tweets.extend(new_tweets)

    if not all_tweets:
        tweet_dict[0] = ("No tweets so far", "00/00/00")
        return tweet_dict

    # Save the ID of the oldest tweet minus one
    oldest = all_tweets[-1].id - 1

    # Keep fetching tweets until there are no more
    while new_tweets:
        new_tweets = api.user_timeline(screen_name=screen_name, count=200, max_id=oldest)
        all_tweets.extend(new_tweets)
        oldest = all_tweets[-1].id - 1

    # Populate the dictionary with tweet ID as the key and (text, created_at) as the value
    for tweet in all_tweets:
        tweet_dict[tweet.id] = (tweet.text, tweet.created_at)

    return tweet_dict


def display_tweets(screen_name):
    """
    Display all tweets from a specific Twitter user.

    Args:
        screen_name (str): Twitter handle of the user.
    """
    tweet_dict = get_all_tweets(screen_name)
    header = '{:<30} {:<30} {}'.format("Tweet ID", "Date Posted", "Tweet Text")
    print(header)
    print('-' * len(header))
    for tweet_id, (text, created_at) in tweet_dict.items():
        print('{:<30} {:<30} {}'.format(tweet_id, created_at, text))


def frequency_analyzer(screen_name):
    """
    Analyze the frequency of words in tweets from a specific Twitter user.

    Args:
        screen_name (str): Twitter handle of the user.
    """
    tweet_dict = get_all_tweets(screen_name)
    freq_dict = {}
    stop_words = set()  # Add any stop words you want to exclude here

    for tweet_id, (text, created_at) in tweet_dict.items():
        words = text.split()
        for word in words:
            if word in stop_words:
                continue
            if word not in freq_dict:
                freq_dict[word] = 1
            else:
                freq_dict[word] += 1

    for word, count in freq_dict.items():
        print(f"{word}: {count}")
