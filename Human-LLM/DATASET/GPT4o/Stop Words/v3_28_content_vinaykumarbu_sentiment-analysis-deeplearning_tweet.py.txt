# Get command line arguments
ARGS = sys.argv[1:]

class TweetListener(StreamListener):
    count = 0

    def on_data(self, data):
        try:
            all_data = json.loads(data)

            # Extracting required fields
            user_name = all_data["user"]["name"]
            verified = all_data["user"]["verified"]
            text = all_data.get("extended_tweet", {}).get("full_text", all_data["text"])

            # Clean the text
            cleaned_text = self.clean_text(text)

            # Print and save the tweet data
            output_line = f'Username: {user_name} , verified: {verified} , text: {cleaned_text}'
            self.save_tweet(output_line)

            # Increment the tweet count and check the limit
            TweetListener.count += 1
            if TweetListener.count == 30:
                sys.exit("30 tweets fetched")

            return True
        except Exception as e:
            print(f"Error on_data: {str(e)}")
            return True

    def on_error(self, status):
        print(f"Error: {status}")
        return True

    @staticmethod
    def clean_text(text):
        """Clean the tweet text by removing unwanted characters."""
        return re.sub(r"[^A-Za-z0-9_@./#&: ]+", "", text)

    @staticmethod
    def save_tweet(output_line):
        """Save the cleaned tweet to a file."""
        with open("new_tweets.txt", "a") as output:
            output.write(output_line + '\n')
        print(output_line)

def main():
    # OAuth authentication
    auth = OAuthHandler(CKEY, CSECRET)
    auth.set_access_token(ATOKEN, ASECRET)

    # Join the command line arguments to form the query string
    query_string = " ".join(ARGS)

    # Create and start the stream
    twitter_stream = Stream(auth, TweetListener())
    twitter_stream.filter(languages=["en"], track=[query_string.strip()])

if __name__ == "__main__":
    main()
