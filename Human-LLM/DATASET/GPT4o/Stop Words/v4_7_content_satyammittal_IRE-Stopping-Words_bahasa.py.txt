import os
import json
import operator

def list_files(directory):
    """List all files in the given directory and its subdirectories."""
    files_list = []
    for subdir, _, files in os.walk(directory):
        for file in files:
            files_list.append(os.path.join(subdir, file))
    return files_list

def process_files(file_list):
    """Process the given list of files to compute word frequencies."""
    counter = {}
    check_doc = {}
    total_files = 0

    for file in file_list:
        with open(file) as f:
            for line in f:
                jfile = json.loads(line)
                text = jfile.get('desc')
                if text:
                    total_files += 1
                    check_word = {}
                    for word in text.split():
                        if "," not in word and "-" not in word:
                            # Update counter
                            counter[word] = counter.get(word, 0) + 1
                            # Update check_word
                            check_word[word] = check_word.get(word, 0) + 1
                    
                    # Update check_doc
                    for word in check_word:
                        check_doc[word] = check_doc.get(word, 0) + 1

    return counter, check_doc

def compute_result(counter, check_doc):
    """Compute the result based on word frequencies and document occurrences."""
    result = {}
    for word in check_doc:
        result[word] = counter[word] * check_doc[word] ** 2
    return result

def main():
    # Directory containing files
    directory = 'extracted'
    
    # List of files to process
    file_list = list_files(directory)
    
    # For testing, we override file_list with a single file
    file_list = ['out.txt']

    # Process the files and get word frequencies
    counter, check_doc = process_files(file_list)

    # Compute the result based on frequencies
    result = compute_result(counter, check_doc)

    # Sort the result in descending order
    sorted_result = sorted(result.items(), key=operator.itemgetter(1), reverse=True)

    # Print the top 50 words
    for i, (word, score) in enumerate(sorted_result[:50]):
        print(word)

if __name__ == "__main__":
    main()
