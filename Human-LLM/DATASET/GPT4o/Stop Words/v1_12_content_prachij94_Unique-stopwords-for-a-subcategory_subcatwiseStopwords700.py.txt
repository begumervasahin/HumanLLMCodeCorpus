import pandas as pd

# Load the data from the Excel and CSV files
df2 = pd.read_excel("C:/Users/IMART/Downloads/Subcat_super_PMCAT.xlsx", sheet_name="Export Worksheet")
df3 = pd.read_csv("C:/Users/IMART/Desktop/My_Data/Himanshu Stopwords/wordfrequencyfinal.csv")

# Initialize an empty DataFrame with the specified columns
mergedf1 = pd.DataFrame(columns=['Subcat_ID', 'Stopwords'])

# Get the unique SUBCAT_ID values from df2
uniquesubcats = list(df2['SUBCAT_ID'].unique())

# Convert strings to lowercase for consistency
df3["HIGH FREQUENCY WORDS"] = df3["HIGH FREQUENCY WORDS"].str.lower()
df2["GLCAT_MCAT_NAME"] = df2["GLCAT_MCAT_NAME"].str.lower()

# Create a dictionary from df3 with SUBCAT ID as keys
highfreqdict = df3.set_index('SUBCAT ID').T.to_dict('list')

a = 0
i = 0
row = 0

# Iterate over the rows of df3
while a < len(df3):
    if df3.iloc[a]["SUBCAT ID"] in uniquesubcats and a < len(df3):
        subcatid = df3.iloc[a]["SUBCAT ID"]
        j = i
        concatstring = ""

        # Concatenate GLCAT_MCAT_NAME values for the current SUBCAT_ID
        while i <= len(df2) - 1 and df2.iloc[i]["SUBCAT_ID"] == subcatid:
            concatstring += df2.iloc[i]["GLCAT_MCAT_NAME"] + " "
            i += 1
            if i == len(df2):
                break

        # Create a set of unique words
        uniq = set(concatstring.split(' '))
        string1 = ""

        # Create a string of unique words
        for x in uniq:
            if x not in string1:
                string1 += x + " "

        # Set values in the merged DataFrame
        mergedf1.loc[row, 'Subcat_ID'] = df2.iloc[j]["SUBCAT_ID"]
        s = string1 + " " + str(highfreqdict[df2.iloc[j]["SUBCAT_ID"]][0])
        mergedf1.loc[row, 'Stopwords'] = s
        row += 1
        a += 1
    elif df3.iloc[a]["SUBCAT ID"] not in uniquesubcats and a < len(df3):
        # Set values in the merged DataFrame for non-unique SUBCAT_ID
        mergedf1.loc[row, 'Subcat_ID'] = df3.iloc[a]["SUBCAT ID"]
        mergedf1.loc[row, 'Stopwords'] = df3.iloc[a]["HIGH FREQUENCY WORDS"]
        row += 1
        a += 1

# Save the merged DataFrame to a CSV file
mergedf1.to_csv("700subcatwisefinalstopwords.csv", index=False)
