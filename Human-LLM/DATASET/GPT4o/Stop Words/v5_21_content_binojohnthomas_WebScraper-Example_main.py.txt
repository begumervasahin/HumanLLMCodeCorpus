import requests
from bs4 import BeautifulSoup
import re
import operator
import json
from tabulate import tabulate
from stop_words import get_stop_words

def get_word_list(url):
    """Fetches the page content and returns a list of cleaned words."""
    word_list = []
    response = requests.get(url)
    soup = BeautifulSoup(response.text, 'lxml')

    for paragraph in soup.find_all('p'):
        if paragraph.text:
            words = paragraph.text.lower().split()
            for word in words:
                cleaned_word = clean_word(word)
                if cleaned_word:
                    word_list.append(cleaned_word)
    
    return word_list

def clean_word(word):
    """Removes non-alphabetical characters from a word."""
    return re.sub('[^a-z]+', '', word)

def create_frequency_table(word_list):
    """Creates a frequency table from a list of words."""
    word_count = {}
    for word in word_list:
        word_count[word] = word_count.get(word, 0) + 1
    return word_count

def remove_stop_words(frequency_list):
    """Removes stop words from the frequency list."""
    stop_words = set(get_stop_words('en'))
    return [[word, freq] for word, freq in frequency_list if word not in stop_words]

def fetch_wikipedia_page(query):
    """Fetches the Wikipedia page URL for a given query."""
    wikipedia_api_link = "https://en.wikipedia.org/w/api.php?format=json&action=query&list=search&srsearch="
    wikipedia_link = "https://en.wikipedia.org/wiki/"

    response = requests.get(wikipedia_api_link + query)
    data = response.json()
    page_title = data['query']['search'][0]['title']
    return wikipedia_link + page_title

def display_word_frequencies(word_frequencies):
    """Displays the word frequencies in a tabular format."""
    total_words = sum(freq for _, freq in word_frequencies)
    top_words = word_frequencies[:20]

    final_list = [
        [word, freq, round(freq * 100 / total_words, 4)]
        for word, freq in top_words
    ]

    print_headers = ['Word', 'Frequency', 'Frequency Percentage']
    print(tabulate(final_list, headers=print_headers, tablefmt='orgtbl'))

def main(query, search_mode=False):
    """Main function to fetch word frequencies from a Wikipedia page."""
    try:
        page_url = fetch_wikipedia_page(query)
        word_list = get_word_list(page_url)
        word_count = create_frequency_table(word_list)
        sorted_word_frequencies = sorted(word_count.items(), key=operator.itemgetter(1), reverse=True)

        if search_mode:
            sorted_word_frequencies = remove_stop_words(sorted_word_frequencies)
        
        display_word_frequencies(sorted_word_frequencies)
    except requests.exceptions.Timeout:
        print("The server didn't respond. Please, try again later.")

if __name__ == "__main__":
    import sys
    if len(sys.argv) < 2:
        print("Enter a valid string")
        sys.exit(1)

    query = sys.argv[1]
    search_mode = len(sys.argv) > 2
    main(query, search_mode)
