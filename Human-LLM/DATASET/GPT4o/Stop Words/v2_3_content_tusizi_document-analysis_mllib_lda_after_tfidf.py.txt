import json
import numpy as np
import matplotlib.pyplot as plt
from sklearn.feature_extraction.text import CountVectorizer
import lda

def main():
    # Load the corpus
    corpus = load_corpus('/vagrant/data/160928/6335045150218551554')
    
    # Vectorize the corpus
    weight = vectorize_corpus(corpus)
    
    # Perform LDA and obtain topic-word and document-topic distributions
    topic_word, doc_topic = perform_lda(weight)
    
    # Print some information about the weight matrix and LDA results
    display_information(weight, doc_topic)
    
    # Plot the topic distribution for selected documents
    plot_doc_topics(doc_topic)
    
    # Plot the word distribution for each topic
    plot_topic_words(topic_word)

def load_corpus(file_path):
    """
    Load the corpus from the given file path.

    Parameters:
    file_path (str): The path to the file containing the corpus.

    Returns:
    list: A list of document contents.
    """
    corpus = []
    with open(file_path, 'r') as file:
        for line in file:
            content_ = json.loads(line)['content']
            corpus.append(content_.strip())
    return corpus

def vectorize_corpus(corpus):
    """
    Vectorize the corpus using CountVectorizer.

    Parameters:
    corpus (list): A list of document contents.

    Returns:
    np.array: The weight matrix obtained from CountVectorizer.
    """
    vectorizer = CountVectorizer()
    X = vectorizer.fit_transform(corpus)
    return X.toarray()

def perform_lda(weight):
    """
    Perform Latent Dirichlet Allocation (LDA) on the weight matrix.

    Parameters:
    weight (np.array): The weight matrix obtained from CountVectorizer.

    Returns:
    np.array, np.array: The topic-word and document-topic distributions.
    """
    model = lda.LDA(n_topics=2, n_iter=500, random_state=1)
    model.fit(weight)
    return model.topic_word_, model.doc_topic_

def display_information(weight, doc_topic):
    """
    Display information about the weight matrix and document-topic matrix.

    Parameters:
    weight (np.array): The weight matrix obtained from CountVectorizer.
    doc_topic (np.array): The document-topic matrix obtained from LDA.
    """
    print(f"Weight matrix shape: {weight.shape}")
    print("First 5 rows and columns of the weight matrix:")
    print(weight[:5, :5])
    print(f"type(doc_topic): {type(doc_topic)}")
    print(f"shape: {doc_topic.shape}")

    # Determine the most probable topic for the first 10 documents
    for n in range(10):
        topic_most_pr = doc_topic[n].argmax()
        print(f"doc: {n} topic: {topic_most_pr}")

def plot_doc_topics(doc_topic):
    """
    Plot the topic distribution for selected documents.

    Parameters:
    doc_topic (np.array): The document-topic matrix obtained from LDA.
    """
    fig, axes = plt.subplots(6, 1, figsize=(8, 8), sharex=True)
    for i, k in enumerate([0, 1, 2, 3, 8, 9]):
        axes[i].stem(doc_topic[k, :], linefmt='r-', markerfmt='ro', basefmt='w-')
        axes[i].set_xlim(-1, 2)
        axes[i].set_ylim(0, 1.2)
        axes[i].set_ylabel("Prob")
        axes[i].set_title(f"Document {k}")
    axes[5].set_xlabel("Topic")
    plt.tight_layout()
    plt.show()

def plot_topic_words(topic_word):
    """
    Plot the word distribution for each topic.

    Parameters:
    topic_word (np.array): The topic-word matrix obtained from LDA.
    """
    fig, axes = plt.subplots(2, 1, figsize=(6, 6), sharex=True)
    for i, k in enumerate([0, 1]):
        axes[i].stem(topic_word[k, :], linefmt='b-', markerfmt='bo', basefmt='w-')
        axes[i].set_xlim(-2, 20)
        axes[i].set_ylim(0, 1)
        axes[i].set_ylabel("Prob")
        axes[i].set_title(f"Topic {k}")
    axes[1].set_xlabel("Word")
    plt.tight_layout()
    plt.show()

if __name__ == "__main__":
    main()
