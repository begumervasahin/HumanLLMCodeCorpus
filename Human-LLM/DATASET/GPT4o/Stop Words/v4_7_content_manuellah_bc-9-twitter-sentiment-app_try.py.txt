def get_all_tweets(screen_name):
    """
    Retrieve all tweets from a specific Twitter user.
    
    Args:
        screen_name (str): The Twitter handle of the user.
    
    Returns:
        dict: A dictionary with tweet ID as keys and a tuple of tweet text and creation date as values.
    """
    tweets_dict = {}
    all_tweets = []

    # Initial request for the most recent tweets (up to 200)
    new_tweets = api.user_timeline(screen_name=screen_name, count=200)
    all_tweets.extend(new_tweets)

    # If no tweets are found, return a dictionary with a default message
    if not all_tweets:
        tweets_dict[0] = ("No tweet so far", "00/00/00")
        return tweets_dict

    # Save the ID of the oldest tweet less one
    oldest = all_tweets[-1].id - 1

    # Keep retrieving tweets until no new tweets are found
    while new_tweets:
        new_tweets = api.user_timeline(screen_name=screen_name, count=200, max_id=oldest)
        all_tweets.extend(new_tweets)
        oldest = all_tweets[-1].id - 1

    # Populate the dictionary with tweet information
    for tweet in all_tweets:
        tweets_dict[tweet.id] = (tweet.text, tweet.created_at)

    return tweets_dict

def display_tweets():
    """
    Display all tweets for a specified user in a formatted manner.
    """
    tweets_dict = get_all_tweets("emmanuelmuthui")
    header = '{:<30} {:<30} {}'.format("Tweet ID", "Date Posted", "Tweet")
    print(header)
    print('-' * len(header))

    for key in tweets_dict:
        print('{:<30} {:<30} {}'.format(str(key), str(tweets_dict[key][1]), tweets_dict[key][0]))

def frequency_analyzer():
    """
    Analyze the frequency of words in the tweets of a specified user, excluding stop words.
    """
    tweets_dict = get_all_tweets("emmanuelmuthui")
    freq_dict = {}

    for key in tweets_dict:
        tweet_words = tweets_dict[key][0].split()
        for word in tweet_words:
            if word.lower() in stop_words:
                continue
            freq_dict[word] = freq_dict.get(word, 0) + 1

    # Print or return the frequency dictionary
    for word, freq in freq_dict.items():
        print(f"{word}: {freq}")
