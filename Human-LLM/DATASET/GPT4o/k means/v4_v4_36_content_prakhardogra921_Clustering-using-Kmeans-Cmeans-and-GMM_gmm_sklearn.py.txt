import numpy as np
from sklearn import mixture
import matplotlib.pyplot as plt
import random

# Generate synthetic data sets
size = 500
set1 = np.random.normal(loc=1, scale=0.1, size=size)
set2 = np.random.normal(loc=1.5, scale=0.1, size=size)
set3 = np.random.normal(loc=2, scale=0.2, size=size)

# Define mixture proportions
p1 = 0.25
p2 = 0.5
p3 = 0.25

# Combine data sets based on proportions
dset = np.concatenate([random.sample(list(set1), int(p1*size)),
                       random.sample(list(set2), int(p2*size)),
                       random.sample(list(set3), int(p3*size))])

# Another set of synthetic data
set4 = np.random.normal(loc=1, scale=0.3, size=size)
set5 = np.random.normal(loc=1.5, scale=0.4, size=size)
set6 = np.random.normal(loc=2, scale=0.3, size=size)
dset2 = np.concatenate([random.sample(list(set4), int(p1*size)),
                        random.sample(list(set5), int(p2*size)),
                        random.sample(list(set6), int(p3*size))])

# Function to generate Gaussian Mixture Model (GMM) plot
def generate_gmm(data):
    # Fit GMM to the data
    gmm = mixture.GaussianMixture(n_components=3)
    gmm.fit(data.reshape(-1, 1))

    # Predict clusters
    y = gmm.predict(data.reshape(-1, 1))

    # Plot the data points colored by cluster
    plt.scatter(data, y)
    plt.title('Gaussian Mixture Model')
    plt.xlabel('Data Points')
    plt.ylabel('Cluster')
    plt.show()

# Generate GMM plot for the first data set
generate_gmm(dset)
