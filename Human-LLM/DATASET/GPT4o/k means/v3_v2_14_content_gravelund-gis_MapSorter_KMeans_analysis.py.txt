import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
import numpy as np
import datetime

def cluster_analysis(data_array, k):
    """
    Groups the input data points into 'k' clusters using K-Means clustering.
    Outputs a plot of the resulting clusters, including the cluster centers.
    No data scaling is performed.

    Parameters:
        data_array (list of lists): The input data points.
        k (int): The number of clusters to create.

    Returns:
        list: Cluster labels for each data point.
    """
    # Convert the input list of lists to a NumPy array
    data_array = np.array(data_array)

    # Initialize the KMeans model with the specified number of clusters
    kmeans = KMeans(n_clusters=k)

    # Fit the KMeans model to the data
    kmeans.fit(data_array)

    # Predict cluster labels for each data point
    labels = kmeans.predict(data_array)

    # Extract x and y coordinates from the data array
    x_coords = data_array[:, 0]
    y_coords = data_array[:, 1]

    # Extract cluster centers
    centroids = kmeans.cluster_centers_
    centroids_x = centroids[:, 0]
    centroids_y = centroids[:, 1]

    # Plot data points with colors corresponding to their cluster labels
    plt.scatter(x_coords, y_coords, c=labels, alpha=0.5)

    # Plot cluster centers with distinct markers
    plt.scatter(centroids_x, centroids_y, marker='D', s=100, c='red')

    # Set plot title with the current date and time
    plt.title(datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S"))

    # Inform the user to inspect the plot
    print("\nInspect the cluster plot shown.\nWhen ready, close the plot.")

    # Display the plot
    plt.show()

    # Return the cluster labels
    return labels

# Example usage
if __name__ == "__main__":
    # Example data points
    example_data = [[1, 2], [2, 3], [3, 4], [10, 9], [8, 7], [7, 6]]
    num_clusters = 2  # Number of clusters

    # Perform cluster analysis
    cluster_labels = cluster_analysis(example_data, num_clusters)

    # Print the cluster labels
    print("Cluster labels:", cluster_labels)
