# Import necessary libraries
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
import seaborn as sns

# Set seaborn style
sns.set()

# Read data from CSV file
file_path = "./yelp.csv"
data = pd.read_csv(file_path, sep=',', quotechar='"', header=0)

# Select relevant columns for clustering
data = data[['latitude', 'longitude', 'reviewCount', 'checkins']]

# Extract values for clustering
latitude = data['latitude'].values
longitude = data['longitude'].values

# Create feature matrix X
X = np.array(list(zip(latitude, longitude)))

# Set the number of clusters
K = 4

# Initialize KMeans clustering model
kmeans = KMeans(n_clusters=K)

# Fit the model to the data
kmeans.fit(X)

# Predict cluster labels for the data points
labels = kmeans.predict(X)

# Get the centroids of the clusters
centroids = kmeans.cluster_centers_

# Print within-cluster sum of squares
print(f"Within-cluster sum of squares: {kmeans.inertia_}")

# Print centroids of each cluster
for i, centroid in enumerate(centroids, start=1):
    print(f"Centroid {i}: {centroid}")

# Visualize the clustering results
plt.scatter(X[:, 0], X[:, 1], c=labels, s=50, cmap='viridis')
plt.scatter(centroids[:, 0], centroids[:, 1], c='red', marker='*', s=200, alpha=0.5)
plt.title("Latitude vs. Longitude")
plt.xlabel("Latitude")
plt.ylabel("Longitude")
plt.savefig('Latitude-Longitude.jpg')  # Save the plot as an image file
plt.show()  # Display the plot
