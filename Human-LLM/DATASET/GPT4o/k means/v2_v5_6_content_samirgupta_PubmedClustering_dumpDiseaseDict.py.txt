import sys
from time import time
import pickle
import os

def process_pubtator_file(input_file_path):
    """
    Process the PubTator file to create a dictionary mapping PMIDs to diseases.

    Parameters:
    input_file_path (str): Path to the input PubTator file.

    Returns:
    dict: A dictionary where keys are PMIDs and values are lists of diseases.
    """
    # Initialize the dictionary to store PMID-disease mappings
    pmid_disease_dict = {}

    # Read the input file line by line
    with open(input_file_path, "r") as file:
        lines = file.readlines()

    # Start the timer to measure processing time
    start_time = time()

    # Iterate through each line in the file
    for line in lines:
        # Remove leading and trailing whitespaces and split the line into tokens
        tokens = line.strip().split("\t")

        # Ensure the line has the required number of fields
        if len(tokens) == 4:
            pmid, _, diseases_mentioned, _ = tokens

            # Split the diseases_mentioned field into individual diseases
            diseases = diseases_mentioned.split("|")

            # Update the PMID-disease dictionary
            if pmid in pmid_disease_dict:
                pmid_disease_dict[pmid].extend(diseases)
            else:
                pmid_disease_dict[pmid] = diseases

    # Print the time taken to process the file
    print(f"Processing completed in {time() - start_time:.4f} seconds")
    print()

    # Return the PMID-disease dictionary
    return pmid_disease_dict

def save_dictionary(dictionary, file_path):
    """
    Save the dictionary to a file using pickle.

    Parameters:
    dictionary (dict): The dictionary to save.
    file_path (str): The path to the output file.
    """
    # Ensure the directory exists
    os.makedirs(os.path.dirname(file_path), exist_ok=True)

    # Serialize and save the dictionary using pickle
    with open(file_path, 'wb') as file:
        pickle.dump(dictionary, file, protocol=pickle.HIGHEST_PROTOCOL)

def run():
    """
    Main function to process the PubTator file and save the PMID-disease dictionary.
    """
    # Ensure the input file path is provided as a command-line argument
    if len(sys.argv) != 2:
        print("Usage: python script_name.py <input_file_path>")
        sys.exit(1)

    # Get the input file path from the command-line arguments
    input_file_path = sys.argv[1]

    # Process the PubTator file
    pmid_disease_dict = process_pubtator_file(input_file_path)

    # Define the path for storing the dictionary
    output_file_path = './pubtator_dumps/pmid_disease_dict.pickle'

    # Save the dictionary to a file
    save_dictionary(pmid_disease_dict, output_file_path)

if __name__ == '__main__':
    # Execute the main function
    run()
