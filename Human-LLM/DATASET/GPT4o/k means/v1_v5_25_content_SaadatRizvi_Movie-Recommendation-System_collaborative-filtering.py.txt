import pandas as pd

def load_data(ratings_path, movies_path):
    # Read movie ratings data
    ratings_cols = ['user_id', 'movie_id', 'rating']
    ratings = pd.read_csv(ratings_path, sep='\t', names=ratings_cols, usecols=range(3))

    # Read movie information data
    movies_cols = ['movie_id', 'title']
    movies = pd.read_csv(movies_path, sep='|', names=movies_cols, usecols=range(2))

    # Merge movie ratings and movie information
    ratings = pd.merge(movies, ratings)
    return ratings

def create_user_ratings_pivot(ratings):
    # Create a pivot table for user ratings
    user_ratings = ratings.pivot_table(index=['user_id'], columns=['title'], values='rating')
    return user_ratings

def calculate_correlation_matrix(user_ratings):
    # Calculate correlation matrix between movies based on user ratings
    corr_matrix = user_ratings.corr(method='pearson', min_periods=100)
    return corr_matrix

def get_user_ratings(user_ratings, user_id):
    # Extract user ratings
    my_ratings = user_ratings.loc[user_id].dropna()
    return my_ratings

def find_similar_movies(corr_matrix, my_ratings):
    similar_candidates = pd.Series(dtype=float)
    
    for i in range(len(my_ratings.index)):
        # Find correlations with each rated movie
        similarities = corr_matrix[my_ratings.index[i]].dropna()

        # Scale similarities by user ratings
        similarities = similarities.map(lambda x: x * my_ratings[i])

        # Add scaled similarities to candidates
        similar_candidates = similar_candidates.append(similarities)
    
    # Sort and aggregate similar candidates
    similar_candidates.sort_values(inplace=True, ascending=False)
    similar_candidates = similar_candidates.groupby(similar_candidates.index).sum()
    similar_candidates.sort_values(inplace=True, ascending=False)
    
    # Filter out already rated movies
    filtered_sims = similar_candidates.drop(my_ratings.index, errors='ignore')
    return filtered_sims

def main():
    ratings_path = 'u.data'
    movies_path = 'u.item'
    user_id = 2

    ratings = load_data(ratings_path, movies_path)
    user_ratings = create_user_ratings_pivot(ratings)
    corr_matrix = calculate_correlation_matrix(user_ratings)
    my_ratings = get_user_ratings(user_ratings, user_id)
    filtered_sims = find_similar_movies(corr_matrix, my_ratings)

    # Print and export filtered similar movies
    print(filtered_sims.head())
    filtered_sims.to_csv("filteredSims.csv")

if __name__ == "__main__":
    main()
