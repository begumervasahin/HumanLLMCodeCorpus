import os
import pandas as pd
import re
from settings import ENCODING

def load_data(directory_path):
    """
    Loads data from text files in a specified directory.

    :param directory_path: Path to the directory containing text files.
    :return: A list of lists, each containing up to three floating point numbers extracted from the text files.
    """
    data = []

    # Get all .txt files in the directory
    txt_files = [file for file in os.listdir(directory_path)
                 if file.endswith(".txt") and os.path.isfile(os.path.join(directory_path, file))]

    # Read content from each file and process the lines
    for file in txt_files:
        file_path = os.path.join(directory_path, file)
        with open(file_path, "r", encoding=ENCODING) as f:
            lines = f.read().splitlines()
            for line in lines:
                filtered_numbers = extract_numbers(line)
                if filtered_numbers:
                    data.append(filtered_numbers)

    return data

def extract_numbers(text):
    """
    Extracts up to three floating point numbers from a string using a regex pattern.

    :param text: Input string to extract numbers from.
    :return: A list of up to three floating point numbers found in the string.
    """
    pattern = r"\d+(\.\d+)?"
    matches = re.findall(pattern, text)
    numbers = [float(match) for match in matches]
    return numbers[:3]

def save_to_csv(data, output_file_path):
    """
    Saves a list of data to a CSV file.

    :param data: List of data to be saved.
    :param output_file_path: Path to the output CSV file.
    """
    df = pd.DataFrame(data, columns=['x', 'y', 'z'])
    df.to_csv(output_file_path, index=False, encoding=ENCODING)
