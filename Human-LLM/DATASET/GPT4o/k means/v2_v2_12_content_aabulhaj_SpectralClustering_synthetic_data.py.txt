# Import necessary libraries
import numpy as np
from sklearn.datasets import make_blobs
import matplotlib.pyplot as plt

def create_circles():
    """Generates noisy circular data for four circles with increasing radii."""
    # Define angles for generating circles
    angles = np.linspace(0, 2 * np.pi, 210)
    
    # Generate noisy circular data for four circles
    circle1 = np.array([np.cos(angles) + 0.1 * np.random.randn(len(angles)),
                        np.sin(angles) + 0.1 * np.random.randn(len(angles))]).T
    circle2 = np.array([2 * np.cos(angles) + 0.1 * np.random.randn(len(angles)),
                        2 * np.sin(angles) + 0.1 * np.random.randn(len(angles))]).T
    circle3 = np.array([3 * np.cos(angles) + 0.1 * np.random.randn(len(angles)),
                        3 * np.sin(angles) + 0.1 * np.random.randn(len(angles))]).T
    circle4 = np.array([4 * np.cos(angles) + 0.1 * np.random.randn(len(angles)),
                        4 * np.sin(angles) + 0.1 * np.random.randn(len(angles))]).T
    
    # Combine all circles into a single dataset
    circles_data = np.vstack((circle1, circle2, circle3, circle4))
    
    return circles_data

def create_blobs():
    """Generates blob-shaped data centered around specified points."""
    # Define centers for generating blobs
    centers = [(-8, -8), (0, 0), (8, 8), (20, 20), (-20, -20)]
    # Generate blob-shaped data with specified centers
    X, _ = make_blobs(n_samples=3000, n_features=2, cluster_std=1.0, centers=centers, random_state=99)
    
    return X

# Generate circular and blob-shaped data
circles_data = create_circles()
blobs_data = create_blobs()

# Create a figure to plot the data
plt.figure(figsize=(16, 8))

# Plot the circular data
plt.subplot(1, 2, 1)
plt.title('Circles')
plt.scatter(circles_data[:, 0], circles_data[:, 1], s=5, color='blue')
plt.xlabel('X')
plt.ylabel('Y')

# Plot the blob-shaped data
plt.subplot(1, 2, 2)
plt.title('Blobs')
plt.scatter(blobs_data[:, 0], blobs_data[:, 1], s=5, color='red')
plt.xlabel('X')
plt.ylabel('Y')

# Display the plots
plt.tight_layout()
plt.show()
