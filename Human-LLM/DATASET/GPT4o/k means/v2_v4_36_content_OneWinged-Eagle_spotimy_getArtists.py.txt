import spotipy
import spotipy.util as util
from json import dump
from typing import Any, List
from time import perf_counter
from sys import stderr

# Configuration details (fill in your own Spotify API credentials here)
username = 'YOUR_SPOTIFY_USERNAME'
scope = 'user-follow-read'
client_id = 'YOUR_SPOTIFY_CLIENT_ID'
client_secret = 'YOUR_SPOTIFY_CLIENT_SECRET'
redirect_uri = 'YOUR_SPOTIFY_REDIRECT_URI'

# Attempting to retrieve token
try:
    token = util.prompt_for_user_token(username, scope, client_id, client_secret, redirect_uri)
    # Creating a Spotify object with the obtained token
    sp = spotipy.Spotify(auth=token)
except Exception as e:
    # Printing an error message if token retrieval fails
    print(f"Token is not accessible for {username}. Error: {e}", file=stderr)
    exit(1)  # Exit the script if token retrieval fails

# Function to retrieve followed artists
def retrieve_artists() -> List[Any]:
    artists = []  # List to store retrieved artists
    last_artist_id = None  # Variable to track the last artist ID

    # Loop until all followed artists are retrieved
    while True:
        # Retrieving followed artists in batches of 50
        response = sp.current_user_followed_artists(limit=50, after=last_artist_id)
        items = response['artists']['items']
        
        # If no more artists are left, exit the loop
        if not items:
            break
        
        # Update last artist ID for the next batch
        last_artist_id = items[-1]['id']
        
        # Extend the list of artists with the retrieved batch
        artists.extend(items)
    
    return artists

# Print a message to indicate the start of artist retrieval
print("Start retrieving artists...")
start_time = perf_counter()  # Record start time

# Retrieve artists
artists = retrieve_artists()

# Calculate elapsed time
elapsed_time = perf_counter() - start_time

# Print the number of retrieved artists and elapsed time
print(f"Retrieved {len(artists)} artists in {elapsed_time:.2f} seconds.")

# Write retrieved artists to a JSON file
with open("artists.json", "w") as outfile:
    dump(artists, outfile, indent=4)
