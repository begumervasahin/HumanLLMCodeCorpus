import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
import datetime
import numpy as np

def cluster_analysis(data_array, k):
    """
    Performs K-Means clustering on the input data and plots the resulting clusters along with cluster centers.

    Parameters:
        data_array (list of lists or np.ndarray): The input data points.
        k (int): The number of clusters to create.

    Returns:
        np.ndarray: Cluster labels for each data point.
    """
    # Ensure data_array is a NumPy array for consistency
    data_array = np.array(data_array)

    # Initialize KMeans model with specified number of clusters
    kmeans_model = KMeans(n_clusters=k, random_state=42)

    # Fit the model to the data and predict cluster labels
    labels = kmeans_model.fit_predict(data_array)

    # Extract coordinates from data_array
    x_coords = data_array[:, 0]
    y_coords = data_array[:, 1]

    # Extract cluster centers
    cluster_centers = kmeans_model.cluster_centers_
    center_x = cluster_centers[:, 0]
    center_y = cluster_centers[:, 1]

    # Plot data points with cluster colors
    plt.figure(figsize=(10, 6))
    plt.scatter(x_coords, y_coords, c=labels, cmap='viridis', alpha=0.6, edgecolor='w', s=100)
    plt.scatter(center_x, center_y, c='red', marker='D', s=150, label='Centroids')

    # Add plot title and labels
    current_time = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    plt.title(f"Cluster Analysis - {current_time}")
    plt.xlabel("Feature 1")
    plt.ylabel("Feature 2")
    plt.legend()

    # Display the plot
    plt.show()

    return labels

# Example usage:
if __name__ == "__main__":
    # Example data
    example_data = np.array([[1, 2], [2, 3], [3, 4], [10, 9], [8, 7], [7, 6]])
    num_clusters = 2  # Number of clusters

    # Call the function
    cluster_labels = cluster_analysis(example_data, num_clusters)
    print("Cluster Labels:", cluster_labels)
