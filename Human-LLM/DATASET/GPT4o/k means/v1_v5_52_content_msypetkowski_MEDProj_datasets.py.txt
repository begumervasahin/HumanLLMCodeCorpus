import pandas as pd

def load_iris():
    """
    Loads the Iris dataset into a pandas DataFrame.

    Returns:
        DataFrame: Iris dataset.
    """
    return pd.read_csv('https://raw.githubusercontent.com/mwaskom/seaborn-data/master/iris.csv')


def load_adult():
    """
    Loads the Adult dataset into a pandas DataFrame.

    Returns:
        DataFrame: Adult dataset.
    """
    column_names = ["Age", "Workclass", "fnlwgt", "Education", "Education-Num", "Marital Status",
                    "Occupation", "Relationship", "Race", "Sex", "Capital Gain", "Capital Loss",
                    "Hours per week", "Country", "Target"]
    return pd.read_csv('https://archive.ics.uci.edu/ml/machine-learning-databases/adult/adult.data',
                       names=column_names,
                       sep=r'\s*,\s*',
                       engine='python',
                       na_values="?")


def load_ctg():
    """
    Loads the Cardiotocography dataset into a pandas DataFrame.

    Returns:
        DataFrame: Cardiotocography dataset.
    """
    ctg_data = pd.read_csv('http://staff.ii.pw.edu.pl/~gprotazi/dydaktyka/dane/cardioto_all_corr.csv')
    return ctg_data.drop('Unnamed: 0', axis=1)


def load_cars():
    """
    Loads the Car Evaluation dataset into a pandas DataFrame.

    Returns:
        DataFrame: Car Evaluation dataset.
    """
    column_names = ["class", "buying", "maint", "doors", "persons", "lug_boot", "safety"]
    return pd.read_csv('https://archive.ics.uci.edu/ml/machine-learning-databases/car/car.data',
                       names=column_names)

# Example usage
if __name__ == "__main__":
    iris_df = load_iris()
    print("Iris Dataset:")
    print(iris_df.head())

    adult_df = load_adult()
    print("\nAdult Dataset:")
    print(adult_df.head())

    ctg_df = load_ctg()
    print("\nCardiotocography Dataset:")
    print(ctg_df.head())

    cars_df = load_cars()
    print("\nCar Evaluation Dataset:")
    print(cars_df.head())
