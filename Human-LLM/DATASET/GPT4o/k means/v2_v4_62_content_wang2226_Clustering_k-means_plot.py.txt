# Import necessary libraries
from sklearn.cluster import KMeans
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
import numpy as np

# Set seaborn style for better aesthetics
sns.set()

# File path to the CSV data
file_path = "./yelp.csv"

# Read data from CSV file
data = pd.read_csv(file_path, sep=',', quotechar='"', header=0)

# Select relevant columns for clustering
data = data[['latitude', 'longitude', 'reviewCount', 'checkins']]

# Extract latitude and longitude values for clustering
latitude = data['latitude'].values
longitude = data['longitude'].values

# Create feature matrix X using latitude and longitude
X = np.array(list(zip(latitude, longitude)))

# Set the number of clusters
num_clusters = 4

# Initialize KMeans clustering model with specified number of clusters
kmeans = KMeans(n_clusters=num_clusters)

# Fit the model to the data
kmeans.fit(X)

# Predict cluster labels for each data point
labels = kmeans.predict(X)

# Get the coordinates of the centroids of the clusters
centroids = kmeans.cluster_centers_

# Print within-cluster sum of squares (inertia)
print(f"Within-cluster sum of squares: {kmeans.inertia_:.2f}")

# Print the coordinates of each cluster centroid
for i, centroid in enumerate(centroids, start=1):
    print(f"Centroid {i}: {centroid}")

# Visualize the clustering results
plt.figure(figsize=(10, 6))
plt.scatter(X[:, 0], X[:, 1], c=labels, s=50, cmap='viridis', label='Data Points')
plt.scatter(centroids[:, 0], centroids[:, 1], c='red', marker='*', s=200, alpha=0.5, label='Centroids')
plt.title("Clustering of Yelp Data: Latitude vs. Longitude")
plt.xlabel("Latitude")
plt.ylabel("Longitude")
plt.legend()
plt.savefig('Latitude-Longitude.jpg')  # Save the plot as an image file
plt.show()  # Display the plot
