import pandas as pd

def load_dataset(url, column_names=None, sep=',', na_values=None, drop_column=None):
    """
    Loads a dataset from a given URL into a pandas DataFrame.

    Args:
        url (str): URL of the dataset.
        column_names (list of str, optional): List of column names to use. Defaults to None.
        sep (str, optional): Separator to use. Defaults to ','.
        na_values (str, optional): String to recognize as NA/NaN. Defaults to None.
        drop_column (str, optional): Column to drop from the DataFrame. Defaults to None.

    Returns:
        DataFrame: Loaded dataset.
    """
    df = pd.read_csv(url, names=column_names, sep=sep, na_values=na_values, engine='python')
    if drop_column:
        df = df.drop(drop_column, axis=1)
    return df

def load_iris():
    """
    Loads the Iris dataset into a pandas DataFrame.

    Returns:
        DataFrame: Iris dataset.
    """
    url = 'https://raw.githubusercontent.com/mwaskom/seaborn-data/master/iris.csv'
    return load_dataset(url)

def load_adult():
    """
    Loads the Adult dataset into a pandas DataFrame.

    Returns:
        DataFrame: Adult dataset.
    """
    url = 'https://archive.ics.uci.edu/ml/machine-learning-databases/adult/adult.data'
    column_names = ["Age", "Workclass", "fnlwgt", "Education", "Education-Num", "Marital Status",
                    "Occupation", "Relationship", "Race", "Sex", "Capital Gain", "Capital Loss",
                    "Hours per week", "Country", "Target"]
    return load_dataset(url, column_names, sep=r'\s*,\s*', na_values="?")

def load_ctg():
    """
    Loads the Cardiotocography dataset into a pandas DataFrame.

    Returns:
        DataFrame: Cardiotocography dataset.
    """
    url = 'http://staff.ii.pw.edu.pl/~gprotazi/dydaktyka/dane/cardioto_all_corr.csv'
    return load_dataset(url, drop_column='Unnamed: 0')

def load_cars():
    """
    Loads the Car Evaluation dataset into a pandas DataFrame.

    Returns:
        DataFrame: Car Evaluation dataset.
    """
    url = 'https://archive.ics.uci.edu/ml/machine-learning-databases/car/car.data'
    column_names = ["class", "buying", "maint", "doors", "persons", "lug_boot", "safety"]
    return load_dataset(url, column_names)

# Example usage
if __name__ == "__main__":
    datasets = {
        "Iris Dataset": load_iris(),
        "Adult Dataset": load_adult(),
        "Cardiotocography Dataset": load_ctg(),
        "Car Evaluation Dataset": load_cars()
    }

    for name, df in datasets.items():
        print(f"\n{name}:")
        print(df.head())
