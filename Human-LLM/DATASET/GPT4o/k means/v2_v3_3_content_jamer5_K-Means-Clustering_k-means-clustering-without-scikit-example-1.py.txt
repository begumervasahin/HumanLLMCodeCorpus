import numpy as np
import os

def compute_euclidean_distance(point, centroid):
    """Compute the Euclidean distance between a data point and a centroid."""
    return np.linalg.norm(point - centroid)

def assign_label_cluster(distances, data_point, centroids):
    """Assign a data point to the nearest centroid."""
    nearest_centroid_index = min(distances, key=distances.get)
    return [nearest_centroid_index, data_point, centroids[nearest_centroid_index]]

def compute_new_centroids(cluster_labels, centroids):
    """Compute new centroids based on the mean of all data points assigned to each centroid."""
    new_centroids = []
    for i in range(len(centroids)):
        points_assigned_to_centroid = [data[1] for data in cluster_labels if data[0] == i]
        if points_assigned_to_centroid:
            new_centroid = np.mean(points_assigned_to_centroid, axis=0)
        else:
            new_centroid = centroids[i]
        new_centroids.append(new_centroid)
    return np.array(new_centroids)

def iterate_k_means(data_points, centroids, total_iterations):
    """Perform the k-means clustering algorithm for a specified number of iterations."""
    for iteration in range(total_iterations):
        cluster_labels = []
        for point in data_points:
            distances = {i: compute_euclidean_distance(point, centroids[i]) for i in range(len(centroids))}
            label = assign_label_cluster(distances, point, centroids)
            cluster_labels.append(label)
        centroids = compute_new_centroids(cluster_labels, centroids)
    return cluster_labels, centroids

def print_label_data(result):
    """Print the clustering results and final centroid positions."""
    print("Result of k-Means Clustering:\n")
    for data in result[0]:
        print("Data point:", data[1])
        print("Cluster number:", data[0], "\n")
    print("Final centroids position:\n", result[1])

def create_initial_centroids():
    """Initialize centroids with predefined values."""
    return np.array([[5.0, 0.0], [45.0, 70.0], [50.0, 90.0]])

if __name__ == "__main__":
    # Load data points from a CSV file
    filename = os.path.join(os.path.dirname(__file__), "data-example-1.csv")
    data_points = np.genfromtxt(filename, delimiter=",")
    
    # Initialize centroids and set the number of iterations
    centroids = create_initial_centroids()
    total_iterations = 100
    
    # Perform k-means clustering
    cluster_labels, new_centroids = iterate_k_means(data_points, centroids, total_iterations)
    
    # Print the clustering results
    print_label_data([cluster_labels, new_centroids])
