import os
import re
import pandas as pd

# Define the text encoding to use
ENCODING = 'utf-8'

def load_data_from_directory(directory_path):
    """
    Load and filter data from text files in a specified directory.

    :param directory_path: Path to the directory containing text files.
    :return: List of filtered data containing numerical values.
    """
    data = []

    # Get a list of all '.txt' files in the directory
    txt_files = [file for file in os.listdir(directory_path) if file.endswith(".txt")]

    # Read content from each file and collect the data
    for txt_file in txt_files:
        file_path = os.path.join(directory_path, txt_file)
        with open(file_path, "r", encoding=ENCODING) as file:
            lines = file.readlines()
            data.extend(lines)

    # Filter the data to extract numerical values
    filtered_data = [extract_numerical_values(line) for line in data]
    return filtered_data

def extract_numerical_values(text):
    """
    Extract the first three numerical values from a given string.

    :param text: Input string to be filtered.
    :return: List of the first three numerical values found in the string.
    """
    # Regular expression pattern to find numerical values
    pattern = r"[0-9]+"
    matches = re.findall(pattern, text)
    numbers = [float(num) for num in matches]

    # Return the first three numerical values
    return numbers[:3]

def save_data_to_csv(data, output_path):
    """
    Save the filtered data to a CSV file.

    :param data: List of filtered data to be saved.
    :param output_path: Path where the CSV file will be saved.
    """
    df = pd.DataFrame(data, columns=['x', 'y', 'z'])
    df.to_csv(output_path, index=False, encoding=ENCODING)

def main():
    # Example usage:
    
    # Directory containing the text files
    data_directory = "/path/to/directory"
    # Output file path
    output_file = "output.csv"

    # Load and filter data from the specified directory
    filtered_data = load_data_from_directory(data_directory)

    # Save the filtered data to a CSV file
    save_data_to_csv(filtered_data, output_file)

if __name__ == "__main__":
    main()
