import sys
from time import time
import pickle

def process_pubtator_file(input_file_path):
    """
    Process the PubTator file to create a dictionary mapping PMIDs to diseases.

    Parameters:
    input_file_path (str): Path to the input PubTator file.

    Returns:
    dict: A dictionary where keys are PMIDs and values are lists of diseases.
    """
    # Open the file and read lines
    with open(input_file_path, "r") as file:
        lines = file.readlines()

    # Dictionary to store PMID-disease mappings
    pmid_disease_dict = {}

    # Start the timer
    start_time = time()

    # Iterate through each line in the file
    for line in lines:
        # Remove leading and trailing whitespaces, split the line into tokens
        tokens = line.strip().split("\t")

        # Check if the line has all required fields
        if len(tokens) == 4:
            # Extract PMID, Mesh ID, diseases_mentioned, and disease_source
            pmid, _, diseases_mentioned, _ = tokens

            # Split diseases_mentioned into individual diseases
            diseases = diseases_mentioned.split("|")

            # Update the PMID-disease dictionary
            pmid_disease_dict.setdefault(pmid, []).extend(diseases)

    # Print the time taken
    print(f"Processing done in {time() - start_time:.4f} seconds")
    print()

    # Return the PMID-disease dictionary
    return pmid_disease_dict

def run():
    """
    Main function to process the PubTator file and save the PMID-disease dictionary.
    """
    # Ensure the input file name is provided as a command-line argument
    if len(sys.argv) != 2:
        print("Usage: python script_name.py <input_file_path>")
        sys.exit(1)

    # Get the input file name from command-line arguments
    input_file_path = sys.argv[1]

    # Process the PubTator file
    pmid_disease_dict = process_pubtator_file(input_file_path)

    # Define the file name for storing the dictionary
    output_file_path = './pubtator_dumps/pmid_disease_dict.pickle'

    # Serialize and save the dictionary using pickle
    with open(output_file_path, 'wb') as handle:
        pickle.dump(pmid_disease_dict, handle, protocol=pickle.HIGHEST_PROTOCOL)

if __name__ == '__main__':
    # Call the main run function
    run()
