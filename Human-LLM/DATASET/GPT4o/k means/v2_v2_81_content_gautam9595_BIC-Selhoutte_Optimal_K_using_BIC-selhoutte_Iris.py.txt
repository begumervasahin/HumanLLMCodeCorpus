import numpy as np
import matplotlib.pyplot as plt
import math
from scipy.spatial import distance
from sklearn.datasets import load_iris
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score

# Load the Iris dataset
iris = load_iris()

# Function to calculate the Bayesian Information Criterion (BIC)
def calculate_bic(kmeans, X):
    # Extract the centers and labels from the KMeans model
    centers = kmeans.cluster_centers_
    labels = kmeans.labels_
    num_clusters = kmeans.n_clusters
    cluster_sizes = np.bincount(labels)
    num_samples, num_features = X.shape

    # Calculate the cluster variance
    cluster_variance = (1.0 / (num_samples - num_clusters) / num_features) * sum(
        [sum(distance.cdist(X[labels == i], [centers[i]], 'euclidean') ** 2) for i in range(num_clusters)]
    )

    # Calculate the constant term for BIC
    constant_term = 0.5 * num_clusters * np.log(num_samples) * (num_features + 1)

    # Calculate BIC
    bic = np.sum(
        [
            cluster_sizes[i] * np.log(cluster_sizes[i]) - cluster_sizes[i] * np.log(num_samples)
            - ((cluster_sizes[i] * num_features) / 2) * np.log(2 * np.pi * cluster_variance)
            - ((cluster_sizes[i] - 1) * num_features / 2)
            for i in range(num_clusters)
        ]
    ) - constant_term

    return bic

# Calculate BIC/silhouette ratio for different cluster sizes
bic_silhouette_ratio = []
for num_clusters in range(2, int(math.sqrt(len(iris.data))) + 1):
    # Fit KMeans with the current number of clusters
    kmeans = KMeans(n_clusters=num_clusters, random_state=42)
    labels = kmeans.fit_predict(iris.data)

    # Calculate BIC for the current number of clusters
    bic = calculate_bic(kmeans, iris.data)

    # Calculate the silhouette score for the current number of clusters
    silhouette_avg = silhouette_score(iris.data, labels)

    # Append the BIC/silhouette ratio to the list
    bic_silhouette_ratio.append(bic / silhouette_avg)

# Plot the BIC/silhouette ratio against the number of clusters
plt.figure(figsize=(10, 6))
plt.plot(range(2, int(math.sqrt(len(iris.data))) + 1), bic_silhouette_ratio, marker='o')
plt.xlabel('Number of Clusters')
plt.ylabel('BIC/Silhouette Ratio')
plt.title('BIC/Silhouette Ratio vs Number of Clusters')
plt.grid(True)
plt.show()
