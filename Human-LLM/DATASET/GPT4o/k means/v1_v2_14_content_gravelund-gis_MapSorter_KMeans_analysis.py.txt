import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
import numpy as np
import datetime

def cluster_analysis(data_array, k):
    """
    Groups (clusters) the input images into 'k' groups (user input for k is expected),      
    based on K-Means clustering.
    Outputs a plot of the resulting clusters, including the cluster centers.
    No data scaling is performed.

    Parameters:
        data_array (list of lists): The input data points.
        k (int): The number of clusters to create.

    Returns:
        list: Cluster labels for each data point.
    """
    # Convert data_array to numpy array
    data_array = np.array(data_array)

    # Initialize KMeans model with specified number of clusters
    model = KMeans(n_clusters=int(k))

    # Fit the model to the data
    model.fit(data_array)

    # Predict cluster labels for each data point
    labels = model.predict(data_array)

    # Extract x and y coordinates from data_array
    xs = data_array[:, 0]
    ys = data_array[:, 1]

    # Extract cluster centers
    centroids = model.cluster_centers_
    centroids_x = centroids[:, 0]
    centroids_y = centroids[:, 1]

    # Plot data points with cluster colors
    plt.scatter(xs, ys, c=labels, alpha=0.5)

    # Plot cluster centers
    plt.scatter(centroids_x, centroids_y, marker='D', s=100, c='red')

    # Set plot title with current date and time
    plt.title(datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S"))

    # Inform the user to inspect the plot
    print("\nInspect the cluster plot shown.\nWhen ready, close the plot and the images will be sorted\n"
          "and copied into subdirectories.")

    # Show the plot
    plt.show()

    # Return the cluster labels
    return labels

# Example usage:
if __name__ == "__main__":
    # Example data
    data_array = [[1, 2], [2, 3], [3, 4], [10, 9], [8, 7], [7, 6]]
    k = 2  # Number of clusters

    # Call the function
    labels = cluster_analysis(data_array, k)
    print("Cluster labels:", labels)
