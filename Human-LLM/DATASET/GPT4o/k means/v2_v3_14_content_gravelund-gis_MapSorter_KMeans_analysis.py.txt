import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
import datetime
import numpy as np

def cluster_analysis(data_array, num_clusters):
    """
    Perform K-Means clustering on the input data points and visualize the clusters.

    Parameters:
        data_array (list of lists or np.ndarray): Input data points.
        num_clusters (int): Number of clusters to create.

    Returns:
        np.ndarray: Cluster labels for each data point.
    """
    # Ensure data_array is a numpy array for compatibility with scikit-learn
    data_array = np.array(data_array)

    # Initialize and fit the KMeans model
    kmeans = KMeans(n_clusters=num_clusters)
    kmeans.fit(data_array)

    # Predict cluster labels for each data point
    labels = kmeans.predict(data_array)

    # Extract x and y coordinates from data_array
    xs, ys = data_array[:, 0], data_array[:, 1]

    # Extract cluster centers
    centroids = kmeans.cluster_centers_
    centroids_x, centroids_y = centroids[:, 0], centroids[:, 1]

    # Plot data points with cluster colors
    plt.scatter(xs, ys, c=labels, alpha=0.5, label='Data Points')

    # Plot cluster centers
    plt.scatter(centroids_x, centroids_y, marker='D', s=100, c='red', label='Cluster Centers')

    # Set plot title with current date and time
    current_time = datetime.datetime.now().strftime('%Y-%m-%d %H:%M:%S')
    plt.title(f"Cluster Analysis ({current_time})")
    plt.xlabel('X')
    plt.ylabel('Y')
    plt.legend()

    # Inform the user to inspect the plot
    print("\nInspect the cluster plot shown.\nWhen ready, close the plot to proceed.")      

    # Show the plot
    plt.show()

    # Return the cluster labels
    return labels

# Example usage
if __name__ == "__main__":
    # Sample data
    data = [[1, 2], [2, 3], [3, 4], [10, 9], [8, 7], [7, 6]]
    num_clusters = 2  # Number of clusters

    # Perform cluster analysis
    cluster_labels = cluster_analysis(data, num_clusters)
    print("Cluster Labels:", cluster_labels)
