import numpy as np
from sklearn.datasets import make_classification
from sklearn.mixture import GaussianMixture
from itertools import groupby

def generate_data():
    """
    Generate synthetic classification data with 178 samples and 13 features.
    """
    X, _ = make_classification(n_samples=178, n_features=13, n_classes=3, n_clusters_per_class=1, random_state=42)
    return X

def apply_em_algorithm(X, n_clusters):
    """
    Apply the Expectation-Maximization (EM) algorithm using a Gaussian Mixture Model.
    
    Parameters:
    X (ndarray): Input data.
    n_clusters (int): Number of clusters to form.
    """
    model = GaussianMixture(n_components=n_clusters, covariance_type='diag')
    fitted_model = model.fit(X)

    print("Means of the Gaussian components:")
    print(fitted_model.means_)
    print("\nWeights of the Gaussian components:")
    print(fitted_model.weights_)

    em_predictions = model.predict(X)

    print("\nCluster sizes for each subset:")
    print("Subset 1 (samples 0-58):")
    print_cluster_sizes(em_predictions[:59])
    print("Subset 2 (samples 59-129):")
    print_cluster_sizes(em_predictions[59:130])
    print("Subset 3 (samples 130-177):")
    print_cluster_sizes(em_predictions[130:178])

def print_cluster_sizes(labels):
    """
    Print sizes of clusters based on predicted labels.
    
    Parameters:
    labels (list): List of predicted labels.
    """
    labels.sort()
    cluster_sizes = [len(list(group)) for key, group in groupby(labels)]
    print(cluster_sizes)

def load_data(file_path):
    """
    Load data from a specified file path or generate synthetic data if the file cannot be opened.
    
    Parameters:
    file_path (str): Path to the input data file.
    
    Returns:
    ndarray: Input data.
    """
    try:
        data = np.genfromtxt(file_path, delimiter=',')
        print("Data loaded from file successfully.")
    except IOError:
        print("Could not open file. Generating synthetic data instead.")
        data = generate_data()
    
    return data

def main():
    file_path = "C:\\Users\\SUMANTH C\\Desktop\\Deep Learning\\Datasets\\wine_sort.csv"
    input_data = load_data(file_path)

    input_data = input_data[:, :13]  # Select the first 13 features
    print("Input data shape:", input_data.shape)

    apply_em_algorithm(input_data, 3)
    print("EM algorithm completed.")

if __name__ == "__main__":
    main()
