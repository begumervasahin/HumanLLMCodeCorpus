import tkinter as tk
from tkinter import Canvas, Label
from pandas import DataFrame
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
from matplotlib.backends.backend_tkagg import FigureCanvasTkAgg

def create_dataframe():
    """Creates a DataFrame with predefined data points."""
    data = {
        'x': [25, 34, 22, 27, 33, 33, 31, 22, 35, 34, 67, 54, 57, 43, 50, 57, 59, 52, 65, 47, 49, 48, 35, 33, 44, 45, 38, 43, 51, 46],
        'y': [79, 51, 53, 78, 59, 74, 73, 57, 69, 75, 51, 32, 40, 47, 53, 36, 35, 58, 59, 50, 25, 20, 14, 12, 20, 5, 29, 27, 8, 7]
    }
    return DataFrame(data, columns=['x', 'y'])

def perform_kmeans(df, n_clusters=3):
    """Performs KMeans clustering and returns the model and centroids."""
    kmeans = KMeans(n_clusters=n_clusters).fit(df)
    return kmeans, kmeans.cluster_centers_

def create_plot(df, kmeans, centroids):
    """Creates a matplotlib plot with clustered data and centroids."""
    fig = plt.Figure(figsize=(6, 4))
    ax = fig.add_subplot(111)
    ax.scatter(df['x'], df['y'], c=kmeans.labels_.astype(float), s=50, alpha=0.5, label='Data Points')
    ax.scatter(centroids[:, 0], centroids[:, 1], c='red', s=100, marker='s', label='Centroids')
    ax.set_xlabel('X')
    ax.set_ylabel('Y')
    ax.set_title('KMeans Clustering')
    ax.legend()
    return fig

def main():
    # Create data and perform clustering
    df = create_dataframe()
    kmeans, centroids = perform_kmeans(df)

    # Initialize Tkinter window
    root = tk.Tk()
    root.title("KMeans Clustering Visualization")

    # Create frame for layout
    frame = tk.Frame(root)
    frame.pack()

    # Display centroids label
    centroids_label = Label(frame, text=f"Centroids:\n{centroids}")
    centroids_label.grid(row=0, column=0)

    # Create canvas for matplotlib figure
    canvas = Canvas(frame, width=600, height=400)
    canvas.grid(row=0, column=1)

    # Create and embed the matplotlib figure into Tkinter
    fig = create_plot(df, kmeans, centroids)
    figure_canvas = FigureCanvasTkAgg(fig, master=canvas)
    figure_canvas.draw()
    figure_canvas.get_tk_widget().pack()

    # Run Tkinter event loop
    root.mainloop()

if __name__ == "__main__":
    main()
