import pandas as pd

# Load the movie ratings data
ratings_cols = ['user_id', 'movie_id', 'rating']
ratings = pd.read_csv('u.data', sep='\t', names=ratings_cols, usecols=range(3))

# Load the movie information data
movies_cols = ['movie_id', 'title']
movies = pd.read_csv('u.item', sep='|', names=movies_cols, usecols=range(2))

# Merge the two datasets on 'movie_id'
ratings = pd.merge(movies, ratings)

# Create a pivot table where rows are users and columns are movie titles, values are ratings
user_ratings = ratings.pivot_table(index='user_id', columns='title', values='rating')

# Calculate the correlation matrix between movies based on user ratings
corr_matrix = user_ratings.corr(method='pearson', min_periods=100)

# Define the user ID for whom we want to generate recommendations
user_id = 2

# Extract the ratings given by this user
my_ratings = user_ratings.loc[user_id].dropna()

# Initialize a series to store similarity scores
similar_candidates = pd.Series()

# Calculate similarity scores for all movies rated by the user
for movie in my_ratings.index:
    # Find movies similar to the one rated by the user
    similarities = corr_matrix[movie].dropna()
    
    # Scale the similarity scores by the user's rating of the movie
    similarities = similarities.map(lambda x: x * my_ratings[movie])
    
    # Append the similarity scores to the candidates series
    similar_candidates = similar_candidates.append(similarities)

# Aggregate the similarity scores and sort them
similar_candidates = similar_candidates.groupby(similar_candidates.index).sum()
similar_candidates.sort_values(inplace=True, ascending=False)

# Remove movies that the user has already rated
recommended_movies = similar_candidates.drop(my_ratings.index, errors='ignore')

# Print the top recommendations and export to CSV
print(recommended_movies.head())
recommended_movies.to_csv("filteredSims.csv")
