import numpy as np
import os

def compute_euclidean_distance(point, centroid):
    return np.linalg.norm(point - centroid)

def assign_label_cluster(distance, data_point, centroids):
    index_of_minimum = min(distance, key=distance.get)
    return [index_of_minimum, data_point, centroids[index_of_minimum]]

def compute_new_centroids(cluster_data, centroids):
    new_centroids = []
    for i in range(len(centroids)):
        points = [data[1] for data in cluster_data if data[0] == i]
        if points:
            new_centroid = np.mean(points, axis=0)
        else:
            new_centroid = centroids[i]
        new_centroids.append(new_centroid)
    return np.array(new_centroids)

def iterate_k_means(data_points, centroids, total_iteration):
    for iteration in range(total_iteration):
        cluster_label = []
        for point in data_points:
            distance = {i: compute_euclidean_distance(point, centroids[i]) for i in range(len(centroids))}
            label = assign_label_cluster(distance, point, centroids)
            cluster_label.append(label)
        centroids = compute_new_centroids(cluster_label, centroids)
    return [cluster_label, centroids]

def print_label_data(result):
    print("Result of k-Means Clustering:\n")
    for data in result[0]:
        print("Data point:", data[1])
        print("Cluster number:", data[0], "\n")
    print("Last centroids position:\n", result[1])

def create_centroids():
    return np.array([[5.0, 0.0], [45.0, 70.0], [50.0, 90.0]])

if __name__ == "__main__":
    filename = os.path.join(os.path.dirname(__file__), "data-example-1.csv")
    data_points = np.genfromtxt(filename, delimiter=",")
    centroids = create_centroids()
    total_iteration = 100
    [cluster_label, new_centroids] = iterate_k_means(data_points, centroids, total_iteration)
    print_label_data([cluster_label, new_centroids])
    print()
