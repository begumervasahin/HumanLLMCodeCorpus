import numpy as np

def load_data(filename: str) -> np.array:
    """
    Opens the file 'filename', loads the dataset, and outputs it as an array.
    Each line in the file should contain tab-separated values representing a data point.
    """
    data = []
    with open(filename, "r") as file:
        lines = file.readlines()
        for line in lines:
            # Convert each line into a list of floats and append a cluster ID (initially 0)
            data.append([float(value) for value in line.split('\t')] + [0])
    return np.asarray(data)

def compute_error(X: np.array, M: np.array) -> float:
    """
    Evaluates the quality of clustering by calculating the average distance
    of each data point to its assigned cluster centroid.
    """
    total_error = 0.0
    for x in X:
        cluster_id = int(x[-1])
        total_error += np.linalg.norm(x[:-1] - M[cluster_id])
    return total_error / X.shape[0]

def calculate_mean(X: np.array, M: np.array) -> np.array:
    """
    Computes the new centroids based on the current cluster assignments.
    """
    new_centroids = np.zeros(M.shape)
    cluster_counts = np.zeros(M.shape[0])

    for x in X:
        cluster_id = int(x[-1])
        new_centroids[cluster_id] += x[:-1]
        cluster_counts[cluster_id] += 1

    for i in range(M.shape[0]):
        if cluster_counts[i] != 0:
            new_centroids[i] /= cluster_counts[i]
        else:
            new_centroids[i] = M[i]  # Retain the old centroid if no points were assigned to this cluster

    return new_centroids

def group_data(X: np.array, M: np.array) -> np.array:
    """
    Assigns each data point in the dataset to the nearest cluster based on Euclidean distance.
    """
    for i in range(X.shape[0]):
        point = X[i][:-1]
        shortest_distance = float("inf")
        nearest_cluster_id = 0

        for idx, centroid in enumerate(M):
            distance = np.linalg.norm(point - centroid)
            if distance < shortest_distance:
                shortest_distance = distance
                nearest_cluster_id = idx

        X[i][-1] = nearest_cluster_id

    return X

def k_means_clustering(filename: str, k: int, iterations: int) -> tuple:
    """
    Executes the K-means clustering algorithm on the dataset loaded from 'filename'.
    """
    X = load_data(filename)
    num_features = X.shape[1] - 1

    # Initialize centroids randomly from the dataset
    initial_indices = np.random.choice(X.shape[0], k, replace=False)
    M = X[initial_indices, :-1]

    for _ in range(iterations):
        X = group_data(X, M)
        M = calculate_mean(X, M)

    final_error = compute_error(X, M)
    print(f"Final clustering error: {final_error}")
    return X, M

# Example usage
if __name__ == "__main__":
    filename = "data.txt"
    k = 3  # Number of clusters
    iterations = 100  # Number of iterations

    clustered_data, centroids = k_means_clustering(filename, k, iterations)
    print("Clustered Data:\n", clustered_data)
    print("Centroids:\n", centroids)
