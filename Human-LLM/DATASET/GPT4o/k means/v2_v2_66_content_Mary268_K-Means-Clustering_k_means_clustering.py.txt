# Import necessary libraries and modules
from __future__ import print_function
import numpy as np
from sklearn.cluster import KMeans
from sklearn.datasets import load_digits
from sklearn.metrics import accuracy_score
from sklearn.model_selection import train_test_split
from matplotlib import pyplot as plt
from scipy.stats import mode
import sklearn

# Check sklearn version for compatibility
if int(sklearn.__version__.split(".")[1]) < 18:
    from sklearn.cross_validation import train_test_split
else:
    from sklearn.model_selection import train_test_split

# Load the digits dataset
digits = load_digits()
data = digits.data
targets = digits.target

# Print the shape of the data
print("Shape of the digits data:", data.shape)

# Perform KMeans clustering with 10 clusters
kmeans = KMeans(n_clusters=10, random_state=0)
clusters = kmeans.fit_predict(data)

# Print the shape of the cluster centers
print("Shape of the cluster centers:", kmeans.cluster_centers_.shape)

# Visualize the cluster centers
fig, axes = plt.subplots(2, 5, figsize=(8, 3))
cluster_centers = kmeans.cluster_centers_.reshape(10, 8, 8)

for ax, center in zip(axes.flat, cluster_centers):
    ax.set(xticks=[], yticks=[])
    ax.imshow(center, interpolation='nearest', cmap=plt.cm.binary)

plt.show()

# Assign labels to clusters based on the mode of true labels in each cluster
assigned_labels = np.zeros_like(clusters)
for i in range(10):
    mask = (clusters == i)
    assigned_labels[mask] = mode(targets[mask])[0]

# Calculate and print the accuracy of clustering
accuracy = accuracy_score(targets, assigned_labels)
print("Accuracy of clustering:", accuracy)
