import kmeans
from sklearn import datasets
from sklearn.cluster import KMeans
from sklearn.metrics import confusion_matrix
import matplotlib.pyplot as plt

def load_iris_dataset():
    """Load the Iris dataset and return the data and target labels."""
    iris = datasets.load_iris()
    return iris.data, iris.target

def run_custom_kmeans(data, k, max_iterations, init_method):
    """Run the custom KMeans implementation and return the predicted labels."""
    return kmeans.KMeans(data, k, max_iterations, init_method)

def run_sklearn_kmeans(data, k):
    """Run the SKLearn KMeans implementation and return the predicted labels."""
    kmeans_model = KMeans(n_clusters=k, init='k-means++').fit(data)
    return kmeans_model.labels_

def plot_clusters(data, labels, title):
    """Plot the data points with colors indicating cluster assignments."""
    plt.scatter(data[:, 0], data[:, 1], c=labels, cmap='viridis')
    plt.title(title)
    plt.show()

def print_confusion_matrix(y_true, y_pred):
    """Print the confusion matrix comparing true and predicted labels."""
    cm = confusion_matrix(y_true, y_pred)
    print(cm)

# Constants
NUM_CLUSTERS = 3
MAX_ITERATIONS = 100
INIT_METHOD = 'kmeans++'

# Load the Iris dataset
data, true_labels = load_iris_dataset()

# Run the custom KMeans algorithm
predicted_labels_custom = run_custom_kmeans(data, NUM_CLUSTERS, MAX_ITERATIONS, INIT_METHOD)

# Run the SKLearn KMeans algorithm
predicted_labels_sklearn = run_sklearn_kmeans(data, NUM_CLUSTERS)

# Plot the results
plot_clusters(data, true_labels, 'Ground Truth')
plot_clusters(data, predicted_labels_custom, 'Custom KMeans Output')
plot_clusters(data, predicted_labels_sklearn, 'SKLearn KMeans Output')

# Print confusion matrices to compare clustering results
print('Confusion Matrix: True Labels vs Custom KMeans Predicted Labels:')
print_confusion_matrix(true_labels, predicted_labels_custom)

print('Confusion Matrix: True Labels vs SKLearn KMeans Predicted Labels:')
print_confusion_matrix(true_labels, predicted_labels_sklearn)

print('Confusion Matrix: SKLearn KMeans vs Custom KMeans Predicted Labels:')
print_confusion_matrix(predicted_labels_sklearn, predicted_labels_custom)
