import numpy as np

def load_data(filename: str) -> np.array:
    """
    Loads the dataset from the file 'filename' and returns it as a numpy array.
    """
    data = []
    with open(filename, "r") as file:
        for line in file.readlines():
            data.append([float(word) for word in line.split('\t')] + [0])
    return np.asarray(data)

def compute_error(X: np.array, M: np.array) -> float:
    """
    Computes the quality of clustering.
    """
    total_error = 0.0
    for x in X:
        cluster_id = int(x[-1])
        total_error += np.linalg.norm(x[:-1] - M[cluster_id])
    return (1.0 / X.shape[0]) * total_error

def calculate_mean(X: np.array, M: np.array) -> np.array:
    """
    Calculates the new centroids based on the current clusters.
    """
    new_centroids = np.zeros(M.shape)
    cluster_counts = [0 for _ in range(M.shape[0])]

    for x in X:
        cluster_id = int(x[-1])
        new_centroids[cluster_id] += x[:-1]
        cluster_counts[cluster_id] += 1

    for i in range(M.shape[0]):
        if cluster_counts[i] != 0:
            new_centroids[i] /= cluster_counts[i]

    return new_centroids

def group_data(X: np.array, M: np.array) -> np.array:
    """
    Assigns each data point in the dataset to a cluster based on Euclidean distance.        
    """
    for i in range(X.shape[0]):
        point = X[i][:-1]
        shortest_distance = float("inf")
        closest_cluster = 0

        for idx, centroid in enumerate(M):
            distance = np.linalg.norm(point - centroid)
            if distance < shortest_distance:
                shortest_distance = distance
                closest_cluster = idx

        X[i][-1] = closest_cluster

    return X

def kmeans_clustering(filename: str, num_clusters: int, max_iterations: int = 100) -> (np.array, float):
    """
    Perform K-means clustering on the dataset provided in the filename.
    
    Parameters:
    - filename: The path to the data file.
    - num_clusters: The number of clusters to form.
    - max_iterations: The maximum number of iterations to run the algorithm.

    Returns:
    - The final centroids and the error of the clustering.
    """
    X = load_data(filename)
    # Initialize centroids randomly from the dataset
    initial_centroids_idx = np.random.choice(range(X.shape[0]), num_clusters, replace=False)
    M = X[initial_centroids_idx, :-1]

    for iteration in range(max_iterations):
        # Group the data points based on the current centroids
        X = group_data(X, M)
        # Compute new centroids
        new_centroids = calculate_mean(X, M)
        # If centroids do not change, break out of loop
        if np.allclose(M, new_centroids):
            print(f"Converged after {iteration} iterations.")
            break
        M = new_centroids

    # Calculate the final error
    error = compute_error(X, M)
    return M, error
