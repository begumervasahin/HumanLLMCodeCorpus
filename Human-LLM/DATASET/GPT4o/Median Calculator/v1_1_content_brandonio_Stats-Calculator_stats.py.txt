import statistics as s

def mean(lst):
    return sum(lst) / len(lst)

def standev(lst):
    avg = mean(lst)
    total = sum((num - avg) ** 2 for num in lst)
    total /= len(lst)
    return total ** 0.5

def variance(lst):
    return standev(lst) ** 2

def quartile(lst, n):
    return lst[int((len(lst) - 1) * n)]

def median(lst):
    assert s.median_low(lst) == quartile(lst, 0.5)
    return quartile(lst, 0.5)

def rng(lst):
    return max(lst) - min(lst)

def rms(lst):
    total = sum(num * num for num in lst)
    total /= len(lst)
    return total ** 0.5

def lq(lst):
    return quartile(lst, 0.25)

def uq(lst):
    return quartile(lst, 0.75)

def main():
    lst = input("What's the list?\n")
    lst = lst.replace(",", "")
    lst = lst.replace(" ", ", ")
    lst = list(eval(lst))
    og = list(lst)
    lst = sorted(lst)
    
    print("\nSorted Data:", lst)
    print("standev:", round(standev(lst), 3))
    print("variance:", round(variance(lst), 3))
    print("rms:", round(rms(lst), 3))
    print("range:", round(rng(lst), 3))
    print("mean:", round(mean(lst), 3))
    
    m = mean(lst)
    sd = standev(lst)
    z_scores = [round((x - m) / sd, 3) for x in og]
    print("Z-Score:", z_scores)
    
    print("\n5 number summary for box plot")
    print("-----------------")
    print("min:", min(lst))
    print("lq:", lq(lst))
    print("median:", median(lst))
    print("uq:", uq(lst))
    print("max:", max(lst))

if __name__ == "__main__":
    main()
