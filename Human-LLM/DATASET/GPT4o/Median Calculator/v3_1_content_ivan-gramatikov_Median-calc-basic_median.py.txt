import statistics
import sys

class MedianFinder:
    def __init__(self):
        self.numbers = []

    def add_number(self, number):
        """
        Adds a number to the list after converting it to float.
        Prints an error message if the input is invalid.
        """
        try:
            self.numbers.append(float(number))
        except ValueError:
            print(f'Invalid input: {number} is not a valid number.')

    def calculate_median(self):
        """
        Calculates and returns the median of the numbers.
        Prints a message and returns None if no numbers are available.
        """
        if not self.numbers:
            print('No numbers available to calculate the median.')
            return None
        return statistics.median(self.numbers)

def run_tests():
    """
    Function to test the MedianFinder class.
    Asserts that the median calculation is correct.
    """
    mf = MedianFinder()
    
    # Test adding valid numbers
    mf.add_number(1)
    mf.add_number(2)
    mf.add_number(3)
    assert mf.calculate_median() == 2, "Test case 1 failed"
    
    # Test adding an invalid number
    mf.add_number('a')
    assert mf.calculate_median() == 2, "Test case 2 failed"
    
    # Test with more numbers
    mf.add_number(4)
    mf.add_number(5)
    assert mf.calculate_median() == 3, "Test case 3 failed"
    
    print("All test cases passed!")

def main():
    """
    Main function to handle command-line arguments and run the MedianFinder.
    If no arguments are provided, it runs the tests.
    """
    mf = MedianFinder()
    
    # If arguments are passed from the command line, use them
    if len(sys.argv) > 1:
        for arg in sys.argv[1:]:
            mf.add_number(arg)
        median = mf.calculate_median()
        if median is not None:
            print('The median of the numbers given is', median)
    else:
        # Run the test cases if no arguments are provided
        run_tests()

if __name__ == '__main__':
    main()
