import pandas as pd

class MedianCalculator:
    '''
    A class to calculate the median day of each month in a dataframe and create an index
    relative to this median.
    '''
    def __init__(self, dataframe, date_column):
        '''
        Initialize the MedianCalculator with a dataframe and a date column.

        Parameters:
        dataframe (pd.DataFrame): The dataframe containing the date column.
        date_column (str): The name of the date column in the dataframe.
        '''
        self.dataframe = dataframe
        self.date_column = date_column

    def add_date_components(self):
        '''
        Add year, month, and day columns to the dataframe based on the date column.
        
        Returns:
        pd.DataFrame: The dataframe with additional year, month, and day columns.
        '''
        self.dataframe['year'] = pd.DatetimeIndex(self.dataframe[self.date_column]).year   
        self.dataframe['month'] = pd.DatetimeIndex(self.dataframe[self.date_column]).month 
        self.dataframe['day'] = pd.DatetimeIndex(self.dataframe[self.date_column]).day     
        return self.dataframe

    def calculate_median(self):
        '''
        Calculate the median day for each month of each year and create an index column
        showing the difference between each day and the median day.
        
        Returns:
        pd.DataFrame: The dataframe with the calculated index column.
        '''
        self.dataframe[self.date_column] = pd.to_datetime(self.dataframe[self.date_column])  
        df_with_components = self.add_date_components()
        result = pd.DataFrame()

        # Iterate over each unique year in the dataframe
        for year in df_with_components['year'].unique():
            year_data = df_with_components[df_with_components['year'] == year]

            # Iterate over each month of the year
            for month in range(1, 13):
                month_data = year_data[year_data['month'] == month].copy()

                if not month_data.empty:
                    median_day = month_data['day'].median()
                    month_data['index'] = month_data['day'].apply(lambda day: day - median_day)
                    result = pd.concat([result, month_data])
        
        return result
