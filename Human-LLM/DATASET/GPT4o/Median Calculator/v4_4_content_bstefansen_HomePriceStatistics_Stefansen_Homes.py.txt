def prompt(cost):
    while True:
        try:
            one_cost = float(input("Enter cost of one home or -99 to quit: "))
            if one_cost == -99:
                break
            cost.append(one_cost)
        except ValueError:
            print("Please enter a valid number.")

def median(cost):
    n = len(cost)
    sorted_cost = sorted(cost)
    mid = n // 2
    if n % 2 == 0:
        return round((sorted_cost[mid - 1] + sorted_cost[mid]) / 2, 1)
    else:
        return round(sorted_cost[mid], 1)

def mean(cost):
    return round(sum(cost) / len(cost), 1)

def output(cost):
    median_price = median(cost)
    mean_price = mean(cost)
    min_price = min(cost)
    max_price = max(cost)

    print("*" * 35)
    print("Prices of homes in your area: ")
    print(cost)
    print("*" * 35)
    print(f"The median value is $ {median_price}")
    print(f"The average sale price is $ {mean_price}")
    print(f"The minimum sale price is $ {min_price}")
    print(f"The maximum sale price is $ {max_price}")
    print("*" * 35)
    print("Thank you for using this program")

def main():
    cost = []
    prompt(cost)
    output(cost)

if __name__ == "__main__":
    main()
