class Calculator:
    def __init__(self, _a_lst):
        self._a_lst = _a_lst

    def sum_my_lst(self):
        return sum(self._a_lst)

    def calc_mean(self):
        return self.sum_my_lst() / len(self._a_lst)

    def calc_median(self):
        sorted_lst = sorted(self._a_lst)
        n = len(sorted_lst)
        mid = n // 2
        if n % 2 == 0:
            return (sorted_lst[mid - 1] + sorted_lst[mid]) / 2.0
        else:
            return sorted_lst[mid]

    def calc_mode(self):
        from collections import Counter
        count = Counter(self._a_lst)
        max_count = max(count.values())
        mode = {k: v for k, v in count.items() if v == max_count and max_count > 1}
        if mode:
            return mode
        else:
            return 'No value appeared more than once.'

    def standard_deviation(self):
        mean = self.calc_mean()
        variance = sum((x - mean) ** 2 for x in self._a_lst) / len(self._a_lst)
        return variance ** 0.5

    def calc_range(self):
        return max(self._a_lst) - min(self._a_lst)


if __name__ == '__main__':
    lst_of_num = []
    print("This calculator will calculate mean, median, mode, range, and standard deviation. Enter data points one by one, then type 'done'.")
    while True:
        prompt = input("Enter your data: ").strip()
        if prompt.lower() == "done":
            break
        try:
            lst_of_num.append(float(prompt))
        except ValueError:
            print("Please enter a valid number.")

    if lst_of_num:
        usr_input = Calculator(lst_of_num)
        print("\nYou entered {} numbers".format(len(lst_of_num)))
        print("The mean is: {:.5f}".format(usr_input.calc_mean()))
        print("The median is: {:.5f}".format(usr_input.calc_median()))
        mode_vals = usr_input.calc_mode()
        if isinstance(mode_vals, dict):
            print("Mode: These numbers were entered this many times: ")
            for key, value in mode_vals.items():
                print("{}: {}".format(key, value))
        else:
            print("Mode: " + mode_vals)
        print("The range is: {:.5f}".format(usr_input.calc_range()))
        print("The standard deviation (sigma) is: {:.5f}".format(usr_input.standard_deviation()))
        print("Have a wonderful day!")
    else:
        print('No valid numbers entered. Please try again with integers or decimals.')
