from collections import Counter

class Calculator:
    def __init__(self, numbers_list):
        """
        Initialize the calculator with a list of numbers.
        """
        self.numbers_list = numbers_list

    def sum_list(self):
        """
        Calculate the sum of the numbers in the list.
        """
        return sum(self.numbers_list)

    def calc_mean(self):
        """
        Calculate the mean (average) of the numbers in the list.
        """
        return self.sum_list() / len(self.numbers_list)

    def calc_median(self):
        """
        Calculate the median of the numbers in the list.
        """
        sorted_list = sorted(self.numbers_list)
        n = len(sorted_list)
        mid = n // 2
        if n % 2 == 0:
            return (sorted_list[mid - 1] + sorted_list[mid]) / 2.0
        else:
            return sorted_list[mid]

    def calc_mode(self):
        """
        Calculate the mode of the numbers in the list.
        """
        count = Counter(self.numbers_list)
        max_count = max(count.values())
        mode = {k: v for k, v in count.items() if v == max_count and max_count > 1}
        return mode if mode else 'No value appeared more than once.'

    def standard_deviation(self):
        """
        Calculate the standard deviation of the numbers in the list.
        """
        mean = self.calc_mean()
        variance = sum((x - mean) ** 2 for x in self.numbers_list) / len(self.numbers_list)
        return variance ** 0.5

    def calc_range(self):
        """
        Calculate the range of the numbers in the list.
        """
        return max(self.numbers_list) - min(self.numbers_list)


def get_user_input():
    """
    Collect numbers from the user until 'done' is entered.
    """
    numbers_list = []
    print("This calculator will calculate the mean, median, mode, range, and standard deviation.")
    print("Enter data points one by one, then type 'done'.")

    while True:
        user_input = input("Enter your data: ").strip()
        if user_input.lower() == "done":
            break
        try:
            numbers_list.append(float(user_input))
        except ValueError:
            print("Please enter a valid number.")

    return numbers_list


def display_results(calculator):
    """
    Display the calculated results.
    """
    print("\nYou entered {} numbers".format(len(calculator.numbers_list)))
    print("The mean is: {:.5f}".format(calculator.calc_mean()))
    print("The median is: {:.5f}".format(calculator.calc_median()))
    
    mode_values = calculator.calc_mode()
    if isinstance(mode_values, dict):
        print("Mode: These numbers were entered this many times: ")
        for key, value in mode_values.items():
            print("{}: {}".format(key, value))
    else:
        print("Mode: " + mode_values)
    
    print("The range is: {:.5f}".format(calculator.calc_range()))
    print("The standard deviation (sigma) is: {:.5f}".format(calculator.standard_deviation()))
    print("Have a wonderful day!")


if __name__ == '__main__':
    numbers_list = get_user_input()

    if numbers_list:
        calculator = Calculator(numbers_list)
        display_results(calculator)
    else:
        print('No valid numbers entered. Please try again with integers or decimals.')
