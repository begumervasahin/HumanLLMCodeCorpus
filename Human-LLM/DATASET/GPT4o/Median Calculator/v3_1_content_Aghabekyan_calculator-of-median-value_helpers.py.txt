import math

def read_file_to_list(file_name, number_type='int'):
    """
    Reads a file and returns a list of numbers.

    Args:
        file_name (str): The name of the file to read.
        number_type (str): The type of numbers to convert the file contents to ('int' or 'float').
                           Default is 'int'.

    Returns:
        list: A list of numbers from the file.
    """
    with open(file_name, "r") as text_file:
        lines = text_file.readlines()
        
    if number_type == 'int':
        return [int(line.strip()) for line in lines]
    elif number_type == 'float':
        return [float(line.strip()) for line in lines]
    else:
        raise ValueError("number_type must be 'int' or 'float'")

def list_median_calculator(data):
    """
    Calculates the median of a list of numbers.

    Args:
        data (list): The list of numbers.

    Returns:
        float: The median of the list.
    """
    data.sort()  # Ensure the data is sorted before calculating the median
    length = len(data)
    
    if length == 0:
        raise ValueError("The data list is empty.")
    
    middle = length // 2
    
    if length % 2 != 0:
        return float(data[middle])
    else:
        return (data[middle - 1] + data[middle]) / 2
