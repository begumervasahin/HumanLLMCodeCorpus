import statistics as s

def mean(lst):
    return sum(lst) / len(lst)

def standard_deviation(lst):
    avg = mean(lst)
    variance = sum((num - avg) ** 2 for num in lst) / len(lst)
    return variance ** 0.5

def variance(lst):
    return standard_deviation(lst) ** 2

def quartile(lst, n):
    return lst[int((len(lst) - 1) * n)]

def median(lst):
    median_val = quartile(lst, 0.5)
    assert s.median_low(lst) == median_val, "Median calculation mismatch"
    return median_val

def data_range(lst):
    return max(lst) - min(lst)

def root_mean_square(lst):
    rms = (sum(num ** 2 for num in lst) / len(lst)) ** 0.5
    return rms

def lower_quartile(lst):
    return quartile(lst, 0.25)

def upper_quartile(lst):
    return quartile(lst, 0.75)

def main():
    lst = input("What's the list?\n")
    lst = lst.replace(",", "")
    lst = lst.replace(" ", ", ")
    lst = list(eval(lst))
    original_lst = list(lst)
    lst.sort()

    print("\nSorted Data:", lst)
    print("Standard Deviation:", round(standard_deviation(lst), 3))
    print("Variance:", round(variance(lst), 3))
    print("Root Mean Square:", round(root_mean_square(lst), 3))
    print("Range:", round(data_range(lst), 3))
    print("Mean:", round(mean(lst), 3))

    mean_val = mean(lst)
    stddev = standard_deviation(lst)
    z_scores = [round((x - mean_val) / stddev, 3) for x in original_lst]
    print("Z-Scores:", z_scores)

    print("\n5-Number Summary for Box Plot")
    print("-----------------")
    print("Min:", min(lst))
    print("Lower Quartile:", lower_quartile(lst))
    print("Median:", median(lst))
    print("Upper Quartile:", upper_quartile(lst))
    print("Max:", max(lst))

if __name__ == "__main__":
    main()
