# helpers.py

def read_file_to_list(file_path):
    """
    Reads integers from a file and returns them as a list.
    The file should have one integer per line.
    """
    with open(file_path, 'r') as file:
        lines = file.readlines()
    return [int(line.strip()) for line in lines]

def list_median_calculator(numbers):
    """
    Calculates the median of a list of numbers.
    Assumes the input list is already sorted.
    """
    n = len(numbers)
    if n % 2 == 1:
        return numbers[n // 2]
    else:
        return (numbers[n // 2 - 1] + numbers[n // 2]) / 2
# main.py

from helpers import read_file_to_list, list_median_calculator

def main():
    # Read the data from the file
    data = read_file_to_list("data/input01.txt")
    
    # Extract the count of numbers from the first element
    count = data.pop(0)
    
    tmp_list = []
    
    # Calculate and print the median after each insertion
    for i in range(count):
        tmp_list.append(data[i])
        sorted_list = sorted(tmp_list)
        median = list_median_calculator(sorted_list)
        print(median)

if __name__ == "__main__":
    main()
