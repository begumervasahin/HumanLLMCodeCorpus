def read_file_to_list(file_name, number_type='int'):
    with open(file_name, "r") as text_file:
        lines = text_file.readlines()
        
    if number_type == 'int':
        return [int(line.strip()) for line in lines]
    elif number_type == 'float':
        return [float(line.strip()) for line in lines]
    else:
        raise ValueError("number_type must be either 'int' or 'float'")

def list_median_calculator(data):
    length = len(data)
    sorted_data = sorted(data)
    
    if length % 2 == 1:
        median = sorted_data[length // 2]
    else:
        mid1 = length // 2 - 1
        mid2 = length // 2
        median = (sorted_data[mid1] + sorted_data[mid2]) / 2.0
    
    return median
