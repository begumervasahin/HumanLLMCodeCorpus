from helpers import read_file_to_list, list_median_calculator

# Dictionary mapping input files to their corresponding output files
tested_files_dict = {
    "data/input00.txt": "data/output00.txt",
    "data/input01.txt": "data/output01.txt",
    "data/input02.txt": "data/output02.txt"
}

# Iterate over each input-output file pair in the dictionary
for input_file, output_file in tested_files_dict.items():
    # Read the input data from the input file
    input_data = read_file_to_list(input_file)
    
    # Read the expected output data from the output file
    output_data = read_file_to_list(output_file, 'float')
    
    # Extract the count of numbers to process from the input data
    count = int(input_data.pop(0))
    
    # Initialize an empty list to store the numbers for median calculation
    tmp_list = []
    
    # Process each number in the input data
    for i in range(count):
        # Add the current number to the temporary list
        tmp_list.append(input_data[i])
        
        # Sort the list and calculate the median
        sorted_list = sorted(tmp_list)
        result = list_median_calculator(sorted_list)
        
        # Print the calculated median and the expected median from the output file
        print(result, output_data[i])
        
        # Assert that the calculated median matches the expected median
        assert result == output_data[i], f"Inconsistency in line {i+1} of output file {output_file}."
