import math

def read_file_to_list(file_name, number_type='int'):
    """
    Reads a file and converts its contents into a list of numbers.

    Parameters:
    - file_name (str): The name of the file to read.
    - number_type (str): The type of numbers to convert to ('int' or 'float'). Default is 'int'.

    Returns:
    - list: A list of numbers (int or float) read from the file.
    """
    with open(file_name, "r") as text_file:
        lines = text_file.readlines()
        
    if number_type == 'int':
        return [int(line.strip()) for line in lines]
    elif number_type == 'float':
        return [float(line.strip()) for line in lines]
    else:
        raise ValueError("number_type must be either 'int' or 'float'")

def list_median_calculator(data):
    """
    Calculates the median of a list of numbers.

    Parameters:
    - data (list): A list of numbers.

    Returns:
    - float: The median of the list.
    """
    sorted_data = sorted(data)
    length = len(sorted_data)
    
    if length % 2 == 1:
        # Odd number of elements, return the middle one
        median = sorted_data[length // 2]
    else:
        # Even number of elements, return the average of the two middle ones
        mid1 = (length // 2) - 1
        mid2 = length // 2
        median = (sorted_data[mid1] + sorted_data[mid2]) / 2.0
    
    return median
