from helpers import read_file_to_list, list_median_calculator

# Dictionary mapping input files to their corresponding output files
tested_files_dict = {
    "data/input00.txt": "data/output00.txt",
    "data/input01.txt": "data/output01.txt",
    "data/input02.txt": "data/output02.txt"
}

# Function to process each file pair
def process_files(input_file, output_file):
    # Read input and output data
    input_data = read_file_to_list(input_file)
    output_data = read_file_to_list(output_file, 'float')

    # Extract the number of elements to process
    count = int(input_data.pop(0))

    # List to store the current numbers
    current_numbers = []

    # Iterate through the numbers, calculate the median, and compare with expected output
    for i in range(count):
        current_numbers.append(input_data[i])
        median = list_median_calculator(sorted(current_numbers))
        expected_output = output_data[i]
        print(median, expected_output)
        assert median == expected_output, f"Inconsistency in line {i + 1} of output file {output_file}."

# Iterate through each file pair in the dictionary and process them
for input_file, output_file in tested_files_dict.items():
    process_files(input_file, output_file)
