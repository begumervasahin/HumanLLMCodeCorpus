def prompt_for_costs():
    costs = []
    while True:
        try:
            cost = float(input("Enter the cost of one home or -99 to quit: "))
            if cost == -99:
                break
            costs.append(cost)
        except ValueError:
            print("Please enter a valid number.")
    return costs

def calculate_median(costs):
    sorted_costs = sorted(costs)
    n = len(sorted_costs)
    if n % 2 == 0:
        median_value = (sorted_costs[n // 2 - 1] + sorted_costs[n // 2]) / 2
    else:
        median_value = sorted_costs[n // 2]
    return round(median_value, 1)

def calculate_mean(costs):
    return round(sum(costs) / len(costs), 1)

def output_statistics(costs):
    print("*" * 35)
    print("Prices of homes in your area: ")
    print(costs)
    print("*" * 35)
    print(f"The median value is $ {calculate_median(costs)}")
    print(f"The average sale price is $ {calculate_mean(costs)}")
    print(f"The minimum sale price is $ {min(costs)}")
    print(f"The maximum sale price is $ {max(costs)}")
    print("*" * 35)
    print("Thank you for using this program")

def main():
    costs = prompt_for_costs()
    if costs:
        output_statistics(costs)
    else:
        print("No data to display. Exiting the program.")

if __name__ == "__main__":
    main()
