import pandas as pd

class MedianCalculator:
    """
    Class to calculate the median of days within each month of a given dataframe's date column.
    """

    def __init__(self, dataframe, date_column):
        """
        Initialize the MedianCalculator with a dataframe and a date column.

        Parameters:
        dataframe (pd.DataFrame): The dataframe containing the data.
        date_column (str): The name of the date column in the dataframe.
        """
        self.dataframe = dataframe.copy()  # Avoid modifying the original dataframe
        self.date_column = date_column
        self._prepare_dataframe()

    def _prepare_dataframe(self):
        """
        Prepare the dataframe by converting the date column to datetime and adding year, month, and day columns.
        """
        self.dataframe[self.date_column] = pd.to_datetime(self.dataframe[self.date_column])
        self.dataframe['year'] = self.dataframe[self.date_column].dt.year
        self.dataframe['month'] = self.dataframe[self.date_column].dt.month
        self.dataframe['day'] = self.dataframe[self.date_column].dt.day

    def calculate_median(self):
        """
        Calculate the median day for each month and add an 'index' column indicating the difference from the median day.

        Returns:
        pd.DataFrame: The dataframe with an added 'index' column.
        """
        result_df = self.dataframe.copy()
        result_df['index'] = None  # Initialize the 'index' column

        unique_years = result_df['year'].unique()

        for year in unique_years:
            yearly_data = result_df[result_df['year'] == year]
            for month in range(1, 13):
                monthly_data = yearly_data[yearly_data['month'] == month]
                if not monthly_data.empty:
                    median_day = monthly_data['day'].median()
                    result_df.loc[monthly_data.index, 'index'] = monthly_data['day'] - median_day

        return result_df
