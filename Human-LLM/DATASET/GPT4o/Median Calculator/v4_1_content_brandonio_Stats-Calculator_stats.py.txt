import statistics as s

# Function definitions
def mean(lst):
    return sum(lst) / len(lst)

def standev(lst):
    avg = mean(lst)
    variance_sum = sum((num - avg) ** 2 for num in lst) / len(lst)
    return variance_sum ** 0.5

def variance(lst):
    return standev(lst) ** 2

def quartile(lst, n):
    return lst[int((len(lst) - 1) * n)]

def median(lst):
    assert(s.median_low(lst) == quartile(lst, 0.5))
    return quartile(lst, 0.5)

def data_range(lst):
    return max(lst) - min(lst)

def rms(lst):
    rms_sum = sum(num ** 2 for num in lst) / len(lst)
    return rms_sum ** 0.5

def lq(lst):
    return quartile(lst, 0.25)

def uq(lst):
    return quartile(lst, 0.75)

# Main script
lst = input("What's the list?\n")
lst = lst.replace(",", "")
lst = lst.replace(" ", ", ")
lst = list(eval(lst))
og = list(lst)
lst = sorted(lst)

# Calculations
m = mean(lst)
sd = standev(lst)

# Print results
print("\nSorted Data:", lst)
print("Standard Deviation:", round(sd, 3))
print("Variance:", round(variance(lst), 3))
print("RMS:", round(rms(lst), 3))
print("Range:", round(data_range(lst), 3))
print("Mean:", round(m, 3))
print("Z-Score:", [round((x - m) / sd, 3) for x in og])
print("\n5 number summary for box plot")
print("-----------------")
print("Min:", min(lst))
print("LQ:", lq(lst))
print("Median:", median(lst))
print("UQ:", uq(lst))
print("Max:", max(lst))
