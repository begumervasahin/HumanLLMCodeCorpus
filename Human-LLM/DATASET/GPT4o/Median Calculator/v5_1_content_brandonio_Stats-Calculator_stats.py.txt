import statistics as s

# Function definitions
def mean(lst):
    return sum(lst) / len(lst)

def standard_deviation(lst):
    avg = mean(lst)
    variance_sum = sum((num - avg) ** 2 for num in lst) / len(lst)
    return variance_sum ** 0.5

def variance(lst):
    return standard_deviation(lst) ** 2

def quartile(lst, fraction):
    return lst[int((len(lst) - 1) * fraction)]

def median(lst):
    return quartile(lst, 0.5)

def data_range(lst):
    return max(lst) - min(lst)

def root_mean_square(lst):
    rms_sum = sum(num ** 2 for num in lst) / len(lst)
    return rms_sum ** 0.5

def lower_quartile(lst):
    return quartile(lst, 0.25)

def upper_quartile(lst):
    return quartile(lst, 0.75)

# Main script
lst = input("What's the list?\n")
lst = lst.replace(",", "").replace(" ", ", ")
lst = list(eval(lst))

original_list = lst[:]
sorted_list = sorted(lst)

# Calculations
mean_value = mean(sorted_list)
std_dev = standard_deviation(sorted_list)
z_scores = [(x - mean_value) / std_dev for x in original_list]

# Print results
print("\nSorted Data:", sorted_list)
print("Standard Deviation:", round(std_dev, 3))
print("Variance:", round(variance(sorted_list), 3))
print("RMS:", round(root_mean_square(sorted_list), 3))
print("Range:", round(data_range(sorted_list), 3))
print("Mean:", round(mean_value, 3))
print("Z-Score:", [round(z, 3) for z in z_scores])
print("\n5 Number Summary for Box Plot")
print("------------------------------")
print("Min:", min(sorted_list))
print("Lower Quartile:", lower_quartile(sorted_list))
print("Median:", median(sorted_list))
print("Upper Quartile:", upper_quartile(sorted_list))
print("Max:", max(sorted_list))
