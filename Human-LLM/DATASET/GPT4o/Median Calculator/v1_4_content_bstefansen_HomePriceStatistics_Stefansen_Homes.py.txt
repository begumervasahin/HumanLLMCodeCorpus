def prompt(cost):
    one_cost = 0
    while one_cost != -99:
        one_cost = float(input("Enter cost of one home or -99 to quit: "))
        if one_cost != -99:
            cost.append(one_cost)

def median(cost):
    cost.sort()
    n = len(cost)
    middle = n // 2
    if n % 2 == 0:
        median_value = (cost[middle - 1] + cost[middle]) / 2
    else:
        median_value = cost[middle]
    return round(median_value, 1)

def mean(cost):
    total = sum(cost)
    list_total = len(cost)
    avg = total / list_total
    return round(avg, 1)

def output(cost):
    median_price = str(median(cost))
    mean_price = str(mean(cost))
    min_price = str(min(cost))
    max_price = str(max(cost))
    print("*" * 35)
    print("Prices of homes in your area: ")
    print(str(cost))
    print("*" * 35)
    print("The median value is $ " + median_price)
    print("The average sale price is $ " + mean_price)
    print("The minimum sale price is $ " + min_price)
    print("The maximum sale price is $ " + max_price)
    print("*" * 35)
    print("Thank you for using this program")

def main():
    cost = []
    prompt(cost)
    output(cost)

if __name__ == "__main__":
    main()
