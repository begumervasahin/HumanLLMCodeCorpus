def prompt_for_costs():
    costs = []
    while True:
        try:
            one_cost = float(input("Enter cost of one home or -99 to quit: "))
            if one_cost == -99:
                break
            costs.append(one_cost)
        except ValueError:
            print("Please enter a valid number.")
    return costs

def calculate_median(costs):
    n = len(costs)
    sorted_costs = sorted(costs)
    mid = n // 2
    if n % 2 == 0:
        return round((sorted_costs[mid - 1] + sorted_costs[mid]) / 2, 1)
    else:
        return round(sorted_costs[mid], 1)

def calculate_mean(costs):
    return round(sum(costs) / len(costs), 1)

def display_statistics(costs):
    median_price = calculate_median(costs)
    mean_price = calculate_mean(costs)
    min_price = min(costs)
    max_price = max(costs)

    print("*" * 35)
    print("Prices of homes in your area: ")
    print(costs)
    print("*" * 35)
    print(f"The median value is $ {median_price}")
    print(f"The average sale price is $ {mean_price}")
    print(f"The minimum sale price is $ {min_price}")
    print(f"The maximum sale price is $ {max_price}")
    print("*" * 35)
    print("Thank you for using this program")

def main():
    costs = prompt_for_costs()
    display_statistics(costs)

if __name__ == "__main__":
    main()
