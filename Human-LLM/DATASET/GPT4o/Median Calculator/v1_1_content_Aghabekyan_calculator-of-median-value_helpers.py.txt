import math

def read_file_to_list(file_name, number_type='int'):
    """
    Reads a file and returns a list of numbers.
    
    Args:
    file_name (str): The name of the file to read.
    number_type (str): The type of numbers to convert the file contents to ('int' or 'float'). Default is 'int'.
    
    Returns:
    list: A list of numbers from the file.
    """
    with open(file_name, "r") as text_file:
        lines = text_file.readlines()
        if number_type == 'int':
            return [int(item) for item in lines]
        elif number_type == 'float':
            return [float(item) for item in lines]

def list_median_calculator(data):
    """
    Calculates the median of a list of numbers.
    
    Args:
    data (list): The list of numbers.
    
    Returns:
    float: The median of the list.
    """
    length = len(data)
    if length % 2 != 0:
        pos = int(math.floor(length / 2))
        return float(data[pos])
    else:
        max_pos = length // 2
        min_pos = max_pos - 1
        return float((data[min_pos] + data[max_pos]) / 2)

# Example usage
if __name__ == "__main__":
    # Replace 'your_file.txt' with your actual file name
    data = read_file_to_list('your_file.txt', number_type='int')
    data.sort()  # Ensure the data is sorted before calculating the median
    median = list_median_calculator(data)
    print(f"The median of the list is: {median}")
