"""
    This is a simple calculator that calculates the mean, median, range, mode, and standard deviation to five
    decimal points. It prompts the user to enter their data one by one, then it calculates and prints all five 
    statistics to the console.
"""

class Calculator:
    def __init__(self, data_list):
        self.data_list = data_list
    
    def sum_list(self):
        return sum(self.data_list)
    
    def calc_mean(self):
        return self.sum_list() / len(self.data_list)
    
    def calc_median(self):
        sorted_list = sorted(self.data_list)
        n = len(sorted_list)
        mid = n // 2
        if n % 2 == 0:
            return (sorted_list[mid - 1] + sorted_list[mid]) / 2
        else:
            return sorted_list[mid]
    
    def calc_mode(self):
        frequency = {}
        for num in self.data_list:
            frequency[num] = frequency.get(num, 0) + 1
        max_freq = max(frequency.values())
        modes = [key for key, value in frequency.items() if value == max_freq]
        if max_freq == 1:
            return 'No value appeared more than once.'
        return modes
    
    def calc_range(self):
        return max(self.data_list) - min(self.data_list)
    
    def calc_standard_deviation(self):
        mean = self.calc_mean()
        variance = sum((x - mean) ** 2 for x in self.data_list) / len(self.data_list)
        return variance ** 0.5

def main():
    data_list = []
    print("This calculator will calculate mean, median, mode, range, and standard deviation. Enter data points one by one, then type 'done'.")
    
    while True:
        user_input = input("Enter your data: ").strip()
        if user_input.lower() == 'done':
            break
        try:
            data_list.append(float(user_input))
        except ValueError:
            print("Please enter a valid number.")

    if not data_list:
        print('No data entered. Exiting the program.')
        return

    calc = Calculator(data_list)
    
    print("\nYou entered {} numbers.".format(len(data_list)))
    print("The mean is: {:.5f}".format(calc.calc_mean()))
    print("The median is: {:.5f}".format(calc.calc_median()))
    mode_result = calc.calc_mode()
    if isinstance(mode_result, str):
        print("Mode: " + mode_result)
    else:
        print("Mode: These numbers were entered the most times: " + ', '.join(map(str, mode_result)))
    print("The range is: {:.5f}".format(calc.calc_range()))
    print("The standard deviation is: {:.5f}".format(calc.calc_standard_deviation()))
    print("Have a wonderful day!")

if __name__ == '__main__':
    main()
