from helpers import read_file_to_list, list_median_calculator

# Dictionary mapping input files to their corresponding output files
tested_files_dict = {
    "data/input00.txt": "data/output00.txt",
    "data/input01.txt": "data/output01.txt",
    "data/input02.txt": "data/output02.txt"
}

def process_file(input_file, output_file):
    """
    Process the input file to calculate medians and compare with expected output.

    Args:
        input_file (str): Path to the input file.
        output_file (str): Path to the output file with expected results.
    """
    # Read the input data and expected output data from files
    input_data = read_file_to_list(input_file)
    expected_output = read_file_to_list(output_file, 'float')
    
    # Extract the number of data points to process
    count = int(input_data.pop(0))
    
    # Initialize an empty list to store numbers for median calculation
    numbers = []
    
    # Process each number in the input data
    for i in range(count):
        # Add the current number to the list
        numbers.append(input_data[i])
        
        # Calculate the median of the sorted list
        median = list_median_calculator(sorted(numbers))
        
        # Print the calculated median and the expected median
        print(median, expected_output[i])
        
        # Verify that the calculated median matches the expected output
        assert median == expected_output[i], f"Inconsistency in line {i+1} of output file {output_file}."

# Main loop to process all file pairs in the dictionary
for input_file, output_file in tested_files_dict.items():
    process_file(input_file, output_file)
