import statistics
import sys

class MedianCalculator:
    """Class to calculate the median of a list of numbers."""
    
    def __init__(self):
        self.numbers = []

    def add_number(self, number):
        """
        Adds a number to the list.
        
        Args:
            number (float): The number to add.
        
        Raises:
            ValueError: If the input is not a valid float.
        """
        try:
            self.numbers.append(float(number))
        except ValueError:
            raise ValueError(f'Invalid input: {number} is not a number.')

    def calculate_median(self):
        """
        Calculates the median of the list of numbers.
        
        Returns:
            float: The median of the numbers.
        
        Raises:
            ValueError: If there are no numbers in the list.
        """
        if not self.numbers:
            raise ValueError('No numbers in the list to calculate the median.')
        return statistics.median(self.numbers)

def parse_arguments(arguments):
    """
    Parses the command line arguments.
    
    Args:
        arguments (list): The command line arguments.
    
    Returns:
        list: A list of numbers provided as arguments.
    
    Raises:
        ValueError: If insufficient number of arguments are provided.
    """
    if len(arguments) < 2:
        raise ValueError('Insufficient number of arguments provided.')
    return arguments[1:]

def main(arguments):
    """
    Main function to add numbers and calculate the median.
    
    Args:
        arguments (list): The command line arguments.
    """
    calculator = MedianCalculator()
    
    try:
        numbers = parse_arguments(arguments)
        for number in numbers:
            calculator.add_number(number)

        median = calculator.calculate_median()
        print('The median of the numbers given is', median)
    except ValueError as ve:
        print(f'Error: {ve}')

if __name__ == '__main__':
    main(sys.argv)
