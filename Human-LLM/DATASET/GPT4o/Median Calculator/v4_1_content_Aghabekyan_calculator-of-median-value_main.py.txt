from helpers import read_file_to_list, list_median_calculator

def main():
    # Read data from file
    data = read_file_to_list("data/input01.txt")
    
    # Extract the count of numbers to process
    count = int(data.pop(0))
    
    # Initialize an empty list to store the numbers
    numbers = []
    
    # Process each number and calculate the median
    for i in range(count):
        numbers.append(data[i])
        sorted_numbers = sorted(numbers)
        median = list_median_calculator(sorted_numbers)
        print(median)

if __name__ == "__main__":
    main()
