class Node:
    def __init__(self, key):
        self.key = key
        self.left = None
        self.right = None

def build_tree(T, root):
    """Recursively build the binary tree from the adjacency matrix."""
    if root is None:
        return None
    
    node = Node(root)
    children = [i for i, val in enumerate(T[root]) if val == 1]
    
    if children:
        if children[0] < root:
            node.left = build_tree(T, children[0])
        if len(children) > 1 and children[1] > root:
            node.right = build_tree(T, children[1])
    
    return node

def find_lca(root, n1, n2):
    """Find the Lowest Common Ancestor (LCA) of two given nodes in the binary tree."""
    if root is None:
        return None

    if root.key == n1 or root.key == n2:
        return root

    left_lca = find_lca(root.left, n1, n2)
    right_lca = find_lca(root.right, n1, n2)

    if left_lca and right_lca:
        return root

    return left_lca if left_lca else right_lca

def question4(T, r, n1, n2):
    """Find the LCA of two nodes in a binary tree represented by an adjacency matrix."""
    root = build_tree(T, r)
    lca = find_lca(root, n1, n2)
    return lca.key if lca else None
