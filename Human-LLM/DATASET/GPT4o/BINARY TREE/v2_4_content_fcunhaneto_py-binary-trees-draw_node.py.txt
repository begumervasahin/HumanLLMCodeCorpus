class Node:
    def __init__(self, key):
        self.key = key
        self.left = None
        self.right = None
        self.parent = None
        self.height = 0
        self.real_height = 0
        self.color = 1  # Assuming 1 represents a specific color, typically used in Red-Black Trees

# Example usage
if __name__ == "__main__":
    # Create a new node with key value 10
    root = Node(10)
    
    # Set left and right children
    root.left = Node(5)
    root.right = Node(15)
    
    # Set parent for children
    root.left.parent = root
    root.right.parent = root
    
    # Print the key values to verify
    print(f"Node key: {root.key}")
    print(f"Left child key: {root.left.key}")
    print(f"Right child key: {root.right.key}")
    
    # Print the parent key values to verify
    print(f"Parent of left child: {root.left.parent.key}")
    print(f"Parent of right child: {root.right.parent.key}")
    
    # Print heights and color
    print(f"Node height: {root.height}")
    print(f"Node real height: {root.real_height}")
    print(f"Node color: {root.color}")
