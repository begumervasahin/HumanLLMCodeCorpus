import shelve

class QuestionNode:
    def __init__(self, data):
        self.data = data
        self.left = None
        self.right = None

class QuestionTree:
    def __init__(self):
        self.root = QuestionNode("hippo")

    def play_game(self):
        """Starts the game of 20 Questions."""
        print("Welcome to 20 Questions.")
        self.root = self._play_game_helper(self.root)

    def _play_game_helper(self, node):
        """Helper function for playing the game recursively."""
        if self._is_leaf(node):
            return self._handle_leaf_node(node)
        else:
            return self._handle_non_leaf_node(node)

    def _is_leaf(self, node):
        """Checks if a node is a leaf node."""
        return node.left is None and node.right is None

    def _handle_leaf_node(self, node):
        """Handles the situation when a leaf node is reached."""
        print(f"You're thinking of: {node.data}")
        response = self._get_input("Is it correct (y/n)? ")
        if response == "y":
            print("I win!!")
            return node
        else:
            return self._add_answer(node)

    def _handle_non_leaf_node(self, node):
        """Handles the situation when a non-leaf node is reached."""
        print(node.data)
        response = self._get_input("y/n? ")
        if response == "y":
            node.right = self._play_game_helper(node.right)
        elif response == "n":
            node.left = self._play_game_helper(node.left)
        return node

    def _add_answer(self, old_node):
        """Adds a new answer to the tree when the AI loses."""
        your_thing = input("Enter the thing you were thinking of: ").strip()
        question = input("Enter your disambiguating question: ").strip()
        yes_or_no = self._get_input("Is your thing the answer to the question (y/n)? ")

        your_thing_node = QuestionNode(your_thing)
        new_question_node = QuestionNode(question)

        if yes_or_no == "y":
            new_question_node.right = your_thing_node
            new_question_node.left = old_node
        elif yes_or_no == "n":
            new_question_node.right = old_node
            new_question_node.left = your_thing_node

        return new_question_node

    def _get_input(self, prompt):
        """Gets user input and ensures it's either 'y' or 'n'."""
        while True:
            response = input(prompt).strip().lower()
            if response in {"y", "n"}:
                return response
            print("Invalid input. Please enter 'y' or 'n'.")

def main():
    """Main function to run the 20 Questions game."""
    with shelve.open("twenty_questions.save") as savefile:
        twenty_questions = savefile.get("tree", QuestionTree())

        twenty_questions.play_game()

        while True:
            savefile["tree"] = twenty_questions
            response = input("Do you want to play again (y/n)? ").strip().lower()
            if response == "y":
                twenty_questions.play_game()
            elif response == "n":
                print("Thanks for playing! Goodbye.")
                break
            else:
                print("Invalid input. Please enter 'y' or 'n'.")

if __name__ == "__main__":
    main()
