import sys
from tree import BST, Node

# List of traversal methods to be executed
TRAVERSALS = ['zigzag', 'preorder', 'postorder', 'inorder', 'levelorder', 'max_width']
# Global variable to store the traversal results
vis = []

def max_width(root):
    """Calculate the maximum width of the tree."""
    if not root:
        return
    queue = [{'node': root, 'level': 0}]
    current_level = []
    widest_level = []
    while queue:
        current = queue.pop(0)
        if current_level and current_level[0]['level'] != current['level']:
            if len(current_level) > len(widest_level):
                widest_level = current_level[:]
            current_level = []
        current_level.append(current)
        if current['node'].right:
            queue.append({'node': current['node'].right, 'level': current['level'] + 1})
        if current['node'].left:
            queue.append({'node': current['node'].left, 'level': current['level'] + 1})
    if len(current_level) > len(widest_level):
        widest_level = current_level[:]
    vis.append(len(widest_level))
    vis.append("at level")
    if widest_level:
        vis.append(widest_level[0]['level'])

def zigzag(root):
    """Perform zigzag level order traversal of the tree."""
    if not root:
        return
    current_stack = [root]
    next_stack = []
    go_right = True
    while current_stack:
        current = current_stack.pop()
        vis.append(current.val)
        if go_right:
            if current.right:
                next_stack.append(current.right)
            if current.left:
                next_stack.append(current.left)
        else:
            if current.left:
                next_stack.append(current.left)
            if current.right:
                next_stack.append(current.right)
        if not current_stack:
            go_right = not go_right
            current_stack, next_stack = next_stack, []

def levelorder(root):
    """Perform level order traversal of the tree."""
    if not root:
        return
    queue = [root]
    while queue:
        current = queue.pop(0)
        vis.append(current.val)
        if current.left:
            queue.append(current.left)
        if current.right:
            queue.append(current.right)

def preorder(root):
    """Perform preorder traversal of the tree."""
    if not root:
        return
    vis.append(root.val)
    preorder(root.left)
    preorder(root.right)

def postorder(root):
    """Perform postorder traversal of the tree."""
    if not root:
        return
    postorder(root.left)
    postorder(root.right)
    vis.append(root.val)

def inorder(root):
    """Perform inorder traversal of the tree."""
    if not root:
        return
    inorder(root.left)
    vis.append(root.val)
    inorder(root.right)

def main():
    global vis
    default_tree = [5, 6, 1, 2, 5, 8, 3, 10, 12, 13, 15]
    tree = BST()
    for node in default_tree:
        tree.insert(node)

    print("{:11s}: {}".format("Traversal", "Sequence"))
    print("-" * 45)
    
    for traversal in TRAVERSALS:
        globals()[traversal](tree.root)
        print("{:11s}: {}".format(traversal, ', '.join(map(str, vis))))
        vis = []

if __name__ == "__main__":
    main()
