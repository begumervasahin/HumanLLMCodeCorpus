from collections import deque

class treeNode:
    def __init__(self, val=-1, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

class binaryTree:
    """
    A binary tree representation that supports deserialization from strings of serialized binary trees, preOrder, inOrder, and postOrder iteration.
    """
    def __init__(self, data=None):
        if data is None:
            self.root = None
        else:
            try:
                self.root = binaryTree.serialize(data)
            except Exception as e:
                raise e

    def __iter__(self):
        """
        Default iterator set to preorder iterator
        """
        return self.preOrderIter()

    @staticmethod
    def serialize(data):
        """
        Decodes your encoded data to construct a binary tree.
        :type data: str
        :rtype: treeNode
        """
        if data is None:
            raise TypeError('input can not be None')
        if not isinstance(data, str):
            raise TypeError('input must be str')
        if len(data) < 3 or not data.startswith('[') or not data.endswith(']'):
            raise TypeError('input must be surrounded by brackets and can not be empty')
        
        data = data[1:-1]
        datas = data.split(', ')
        if datas[0] == 'null':
            return None
        
        i = 1
        l = len(datas)
        root = treeNode(int(datas[0]))
        frontier = [root]
        
        while i < l:
            t = []
            for n in frontier:
                if i < l and datas[i] != 'null':
                    n.left = treeNode(int(datas[i]))
                    t.append(n.left)
                i += 1
                if i < l and datas[i] != 'null':
                    n.right = treeNode(int(datas[i]))
                    t.append(n.right)
                i += 1
            frontier = t
        
        return root

    def inOrderIter(self):
        """
        inOrder iterator
        """
        if self.root:
            stack = [self.root]
            visited = set()
            while stack:
                cur = stack[-1]
                if not cur.left or cur.left in visited:
                    stack.pop()
                    visited.add(cur)
                    yield cur.val
                    if cur.right:
                        stack.append(cur.right)
                else:
                    stack.append(cur.left)
        raise StopIteration

    def preOrderIter(self):
        """
        preOrder iterator
        """
        if self.root:
            stack = [self.root]
            while stack:
                cur = stack.pop()
                yield cur.val
                if cur.right:
                    stack.append(cur.right)
                if cur.left:
                    stack.append(cur.left)
        raise StopIteration

    def postOrderIter(self):
        """
        postOrder iterator
        """
        if self.root:
            stack = [self.root]
            visited = set()
            while stack:
                cur = stack[-1]
                if (not cur.left or cur.left in visited) and (not cur.right or cur.right in visited):
                    stack.pop()
                    visited.add(cur)
                    yield cur.val
                elif cur.left and cur.left not in visited:
                    stack.append(cur.left)
                else:
                    stack.append(cur.right)
        raise StopIteration

    def bfsIter(self):
        """
        bfs iterator
        """
        if self.root:
            queue = deque([self.root])
            while queue:
                front = queue.popleft()
                yield front.val
                if front.left:
                    queue.append(front.left)
                if front.right:
                    queue.append(front.right)
        raise StopIteration

# Example usage:
if __name__ == '__main__':
    # Example serialized tree: "[1, 2, 3, null, null, 4, 5]"
    data = "[1, 2, 3, null, null, 4, 5]"
    bt = binaryTree(data)
    
    print("Pre-order traversal:")
    for value in bt.preOrderIter():
        print(value, end=' ')
    print("\nIn-order traversal:")
    for value in bt.inOrderIter():
        print(value, end=' ')
    print("\nPost-order traversal:")
    for value in bt.postOrderIter():
        print(value, end=' ')
    print("\nBFS traversal:")
    for value in bt.bfsIter():
        print(value, end=' ')
