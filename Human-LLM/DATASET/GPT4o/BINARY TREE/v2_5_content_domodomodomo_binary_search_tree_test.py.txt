import unittest
import random
from binary_search_tree import iterator, generator, BinarySearchTree, BinarySearchNode

class TestBinarySearchTree(unittest.TestCase):
    def test_insert_list_delete_left(self):
        """Test insertion and deletion of nodes using delete_left method."""
        bst = BinarySearchTree()
        # Insert 1000 random values into the BST
        for value in (random.randint(0, 99) for _ in range(1000)):
            bst.insert(value)
        # Get the list representation of the BST and shuffle it
        lst = bst.list()
        random.shuffle(lst)
        # Delete each value from the BST using delete_left
        for value in lst:
            bst.delete_left(value)
        # Check if the BST is empty
        self.assertEqual(bst.list(), [])

    def test_insert_list_sequentially_delete_right(self):
        """Test insertion and sequential deletion of nodes using delete_right method."""
        bst = BinarySearchTree()
        # Insert 1000 random values into the BST
        for value in (random.randint(0, 99) for _ in range(1000)):
            bst.insert(value)
        # Get the sequential list representation of the BST and shuffle it
        lst = bst.list_sequentially()
        random.shuffle(lst)
        # Delete each value from the BST using delete_right
        for value in lst:
            bst.delete_right(value)
        # Check if the BST is empty
        self.assertEqual(bst.list(), [])

    def test_iterator(self):
        """Test the iterator method of the BST."""
        bst = BinarySearchTree()
        # Insert 1000 random values into the BST
        for value in (random.randint(0, 99) for _ in range(1000)):
            bst.insert(value)
        # Assign the iterator method to the BinarySearchNode
        BinarySearchNode.__iter__ = iterator
        # Check if the list representation matches the iterator output
        self.assertEqual(bst.list(), list(bst))

    def test_generator(self):
        """Test the generator method of the BST."""
        bst = BinarySearchTree()
        # Insert 1000 random values into the BST
        for value in (random.randint(0, 99) for _ in range(1000)):
            bst.insert(value)
        # Assign the generator method to the BinarySearchNode
        BinarySearchNode.__iter__ = generator
        # Check if the list representation matches the generator output
        self.assertEqual(bst.list(), list(bst))

if __name__ == '__main__':
    # Load the test cases and run them
    suite = unittest.defaultTestLoader.loadTestsFromTestCase(TestBinarySearchTree)
    unittest.TextTestRunner().run(suite)
