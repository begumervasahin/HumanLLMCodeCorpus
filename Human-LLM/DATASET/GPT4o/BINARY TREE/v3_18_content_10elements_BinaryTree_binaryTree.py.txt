from collections import deque

class TreeNode:
    def __init__(self, val=-1, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

class BinaryTree:
    """
    A binary tree representation that supports deserialization from strings of serialized binary trees,
    as well as pre-order, in-order, post-order, and BFS traversal.
    """
    def __init__(self, data=None):
        self.root = self.deserialize(data) if data else None

    def __iter__(self):
        """Default iterator set to pre-order iterator."""
        return self.pre_order_iter()

    @staticmethod
    def deserialize(data):
        """
        Decodes a serialized string to construct a binary tree.
        :param data: str
        :return: TreeNode
        """
        if not isinstance(data, str):
            raise TypeError('Input must be a string')
        if len(data) < 3 or not (data.startswith('[') and data.endswith(']')):
            raise ValueError('Input must be surrounded by brackets and cannot be empty')
        
        elements = data[1:-1].split(', ')
        if elements[0] == 'null':
            return None
        
        root = TreeNode(int(elements[0]))
        queue = deque([root])
        i = 1
        
        while i < len(elements):
            node = queue.popleft()
            if elements[i] != 'null':
                node.left = TreeNode(int(elements[i]))
                queue.append(node.left)
            i += 1
            if i < len(elements) and elements[i] != 'null':
                node.right = TreeNode(int(elements[i]))
                queue.append(node.right)
            i += 1
        
        return root

    def in_order_iter(self):
        """In-order iterator."""
        if not self.root:
            return
        stack = []
        current = self.root
        while stack or current:
            while current:
                stack.append(current)
                current = current.left
            current = stack.pop()
            yield current.val
            current = current.right

    def pre_order_iter(self):
        """Pre-order iterator."""
        if not self.root:
            return
        stack = [self.root]
        while stack:
            node = stack.pop()
            yield node.val
            if node.right:
                stack.append(node.right)
            if node.left:
                stack.append(node.left)

    def post_order_iter(self):
        """Post-order iterator."""
        if not self.root:
            return
        stack, visited = [self.root], set()
        while stack:
            node = stack[-1]
            if (not node.left or node.left in visited) and (not node.right or node.right in visited):
                stack.pop()
                visited.add(node)
                yield node.val
            else:
                if node.right and node.right not in visited:
                    stack.append(node.right)
                if node.left and node.left not in visited:
                    stack.append(node.left)

    def bfs_iter(self):
        """BFS iterator."""
        if not self.root:
            return
        queue = deque([self.root])
        while queue:
            node = queue.popleft()
            yield node.val
            if node.left:
                queue.append(node.left)
            if node.right:
                queue.append(node.right)

# Example usage:
if __name__ == '__main__':
    data = "[1, 2, 3, null, null, 4, 5]"
    bt = BinaryTree(data)
    
    print("Pre-order traversal:")
    for value in bt.pre_order_iter():
        print(value, end=' ')
    
    print("\nIn-order traversal:")
    for value in bt.in_order_iter():
        print(value, end=' ')
    
    print("\nPost-order traversal:")
    for value in bt.post_order_iter():
        print(value, end=' ')
    
    print("\nBFS traversal:")
    for value in bt.bfs_iter():
        print(value, end=' ')
