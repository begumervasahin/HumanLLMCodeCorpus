import csv
import random
import math
import operator

def load_ds(filename, split, train_set=None, test_set=None, signal=4):
    if train_set is None:
        train_set = []
    if test_set is None:
        test_set = []

    with open(filename, 'r') as csvfile:
        lines = csv.reader(csvfile)
        dataset = list(lines)
        for x in range(len(dataset) - 1):
            for y in range(signal):
                dataset[x][y] = float(dataset[x][y])
            if random.random() < split:
                train_set.append(dataset[x])
            else:
                test_set.append(dataset[x])
    return train_set, test_set

def distance(instance1, instance2, length):
    return math.sqrt(sum(pow(instance1[x] - instance2[x], 2) for x in range(length)))

def get_nn(train_set, test_instance, k):
    distances = [(train_instance, distance(test_instance, train_instance, len(test_instance) - 1)) for train_instance in train_set]
    distances.sort(key=operator.itemgetter(1))
    return [distances[x][0] for x in range(k)]

def resp(neighbors):
    class_votes = {}
    for neighbor in neighbors:
        response = neighbor[-1]
        class_votes[response] = class_votes.get(response, 0) + 1
    sorted_votes = sorted(class_votes.items(), key=operator.itemgetter(1), reverse=True)
    return sorted_votes[0][0]

def calculate_accuracy(test_set, predictions):
    correct = sum(1 for x in range(len(test_set)) if test_set[x][-1] == predictions[x])
    return (correct / float(len(test_set))) * 100.0

def main():
    train_set, test_set = [], []
    split = 0.67
    train_set, test_set = load_ds('traindata.csv', split, train_set, test_set, 4)
    print('Train set:', len(train_set))
    print('Test set:', len(test_set))
    
    predictions = []
    k = 3
    for test_instance in test_set:
        neighbors = get_nn(train_set, test_instance, k)
        result = resp(neighbors)
        predictions.append(result)
        print(f'> predicted Decision={result}, Actual Decision={test_instance[-1]}')
    
    accuracy = calculate_accuracy(test_set, predictions)
    print(f'Accuracy: {accuracy}%')

if __name__ == '__main__':
    main()
