import scipy.io as sio
import numpy as np
import heapq

def euclidean_dist(x, y):
    """
    Compute the Euclidean distance between two points.
    
    Parameters:
    x (ndarray): First point.
    y (ndarray): Second point.
    
    Returns:
    float: Euclidean distance between x and y.
    """
    return np.linalg.norm(x - y)

def knn_classification(k, num_training_pts, training_pixels, training_labels, test_pixels, test_labels):
    """
    Perform k-nearest neighbors classification on the dataset.

    Parameters:
    k (int): Number of nearest neighbors to consider.
    num_training_pts (int): Number of training data points to use.
    training_pixels (ndarray): Training data pixels.
    training_labels (ndarray): Training data labels.
    test_pixels (ndarray): Test data pixels.
    test_labels (ndarray): Test data labels.

    Returns:
    tuple: Number of correct and incorrect classifications.
    """
    correct_wrong_list = [0, 0]
    RIGHT, WRONG = 0, 1

    # Iterate over each test sample
    for test_idx in range(len(test_pixels)):
        knn = []  # List to store the k nearest neighbors

        # Find the k nearest neighbors
        for i in range(len(training_pixels)):
            curr_dist = euclidean_dist(training_pixels[i], test_pixels[test_idx])
            if len(knn) < k:
                heapq.heappush(knn, (curr_dist, i))
                heapq._heapify_max(knn)
            elif curr_dist < knn[0][0]:
                heapq.heappop(knn)
                heapq.heappush(knn, (curr_dist, i))
                heapq._heapify_max(knn)

        # Determine the most common label among the neighbors
        counts = [0] * 10
        for _, idx in knn:
            label = training_labels[idx][0]
            counts[label] += 1

        label_guess = np.argmax(counts)
        actual_label = test_labels[test_idx][0]

        # Check if the prediction is correct
        if actual_label == label_guess:
            correct_wrong_list[RIGHT] += 1
        else:
            correct_wrong_list[WRONG] += 1

        print(f"Test image {num_training_pts + test_idx} was identified as {label_guess}")
        print(f"Actual label was {actual_label}")

    return correct_wrong_list

def main():
    """
    Main function to perform k-nearest neighbors classification on a dataset.
    """
    # Input parameters
    k = int(input("Please type a k for how many nearest neighbors you want to compute: "))
    num_training_pts = int(input("There are 10,000 pictures of numbers.\nHow many would you like to use as training data? "))

    # Load the dataset
    data = sio.loadmat('ML_hw1data.mat')
    training_pixels = data['X'][:num_training_pts]
    training_labels = data['Y'][:num_training_pts]
    test_pixels = data['X'][num_training_pts:]
    test_labels = data['Y'][num_training_pts:]

    # Perform k-nearest neighbors classification
    correct_wrong_list = knn_classification(k, num_training_pts, training_pixels, training_labels, test_pixels, test_labels)

    # Output the results
    print(f"\nk was {k}")
    print(f"We used {num_training_pts} training data points")
    print("\nCorrect vs. wrong answers:")
    print(correct_wrong_list)
    print("\nPercentage correct:")
    accuracy = (correct_wrong_list[0] / sum(correct_wrong_list)) * 100
    print(f"{accuracy:.2f}%")

if __name__ == '__main__':
    main()
