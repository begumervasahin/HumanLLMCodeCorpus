import cv2
import os
import numpy as np
import array

# Global list to store segmented words
segmented_words = []

def word_segmenter(count):
    """
    Segments words from an image and saves the segmented images.
    
    Parameters:
    count (int): The current count for saving the segmented images.
    
    Returns:
    np.ndarray: The thresholded image after segmentation.
    """
    img_path = "words.png"
    img = cv2.imread(img_path)
    
    if img is None:
        print("Error: Image not read from file\n\n")
        os.system("pause")
        return
    
    height, width, _ = img.shape
    print(f"Height: {height}")
    print(f"Width: {width}")
    
    word_ratio = float(width) / float(height)
    print(f"Ratio: {word_ratio}")
    
    if word_ratio >= 0.9:
        max_value = 9999
        min_index = -999
        pixel_count = array.array('i', (0,) * 100)
        
        for i in range(height // 2):
            for j in range(width):
                if np.any(img[i, j] == 0):
                    pixel_count[i] += 1
            
            if pixel_count[i] < max_value:
                max_value = pixel_count[i]
                min_index = i
            elif pixel_count[i] - max_value <= int(width / 3):
                new_min_index = i
        
        irange = max(new_min_index, min_index)
        
        for i in range(min_index - 2, irange + 2):
            for j in range(width):
                img[i, j] = 1
    
    img_gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    img_thresh = cv2.adaptiveThreshold(img_gray, 255, cv2.ADAPTIVE_THRESH_MEAN_C, cv2.THRESH_BINARY_INV, 111, 2)
    
    segmented_words.append(img_thresh)
    
    save_path = f"CreS2/c{count - 1}.png" if count != 0 else "CreS2/c-1.png"
    cv2.imwrite(save_path, segmented_words[count - 1])
    cv2.imwrite("wordseg.png", segmented_words[count])
    
    cv2.imshow("Segmented Word", img_thresh)
    cv2.waitKey(0)
    
    return img_thresh

if __name__ == "__main__":
    count = 1  # Example count value, adjust as needed
    word_segmenter(count)
