import matplotlib.pyplot as plt
import numpy as np

def plot_2d_classification(data, labels, correct_data, correct_labels, wrong_data, wrong_labels, block=True):
    """
    Plot 2D data points with different markers and colors based on their classification and correctness.
    
    Parameters:
    - data (array): The main dataset.
    - labels (array): The labels for the main dataset.
    - correct_data (array): The correctly classified data points.
    - correct_labels (array): The labels for the correctly classified data points.
    - wrong_data (array): The incorrectly classified data points.
    - wrong_labels (array): The labels for the incorrectly classified data points.
    - block (bool): Whether to block the plot display.
    """
    fig, ax = plt.subplots()
    
    datasets = [(data, labels, 'b'), (correct_data, correct_labels, 'g'), (wrong_data, wrong_labels, 'r')]
    markers = ['o', 'x']
    
    for dataset, label_set, color in datasets:
        for label, marker in zip([1, 0], markers):
            indices = np.where(label_set == label)[0]
            if indices.size > 0:
                ax.scatter(dataset[indices, 0], dataset[indices, 1], marker=marker, c=color, label=f'{color} {"true" if label == 1 else "false"}')
    
    ax.legend()
    plt.show(block=block)

# Example usage
if __name__ == "__main__":
    data = np.array([[1, 2], [2, 3], [3, 4], [4, 5]])
    labels = np.array([1, 0, 1, 0])
    
    correct_data = np.array([[1, 2], [3, 4]])
    correct_labels = np.array([1, 1])
    
    wrong_data = np.array([[2, 3], [4, 5]])
    wrong_labels = np.array([0, 0])
    
    plot_2d_classification(data, labels, correct_data, correct_labels, wrong_data, wrong_labels)
