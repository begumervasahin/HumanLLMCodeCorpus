import numpy as np
import operator

# Basic addition
a = 1
b = 2
c = a + b
print(c)  # Output: 3

# Create a numpy array and get its shape
array_b = np.array([6, 7, 8])
array_b_shape = array_b.shape[0]
print(array_b_shape)  # Output: 3

# Sum of all elements in a 2D array
sum_all_elements = np.sum([[0, 1, 2], [2, 1, 3]])
print(sum_all_elements)  # Output: 9

# Sum of elements along axis 0 (column-wise)
sum_axis_0 = np.sum([[0, 1, 2], [2, 1, 3]], axis=0)
print(sum_axis_0)  # Output: [2 2 5]

# Sum of elements along axis 1 (row-wise)
sum_axis_1 = np.sum([[8, 1, 2], [2, 1, 3], [0, 0, 0]], axis=1)
print(sum_axis_1)  # Output: [11  6  0]

# Argsort the result to get the indices that would sort the array
argsorted_indices = sum_axis_1.argsort()
print(argsorted_indices)  # Output: [2 1 0]

# Sort a dictionary by value in descending order
class_count = {"A": 6, "B": 1, "C": 0, "D": 2}
sorted_class_count = sorted(class_count.items(), key=operator.itemgetter(1), reverse=True)
print(sorted_class_count)  # Output: [('A', 6), ('D', 2), ('B', 1), ('C', 0)]

# Define a 2x2 numpy array
array_f = np.array([[5, 8], [1, 2]])

# Function to convert an image to a vector (example)
def img_to_vector(matrix):
    rows, cols = matrix.shape
    vector = np.zeros((1, rows * cols))
    for i in range(rows):
        for j in range(cols):
            vector[0, cols * i + j] = int(matrix[i, j])
    return vector

# Example usage of the img_to_vector function
vector_f = img_to_vector(array_f)
print(vector_f)

# Initialize a zero vector and print an element
zero_vector = np.zeros((1, 4))
print(zero_vector[0, 1])  # Output: 0.0
