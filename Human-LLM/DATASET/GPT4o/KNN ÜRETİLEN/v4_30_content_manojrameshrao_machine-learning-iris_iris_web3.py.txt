import os
import psutil
import numpy as np
import matplotlib.pyplot as plt
import mpld3
from sklearn import svm, datasets
from flask import Flask
import timeit

app = Flask(__name__)

def memory_usage_psutil():
    """
    Get the memory usage of the current process in MB.
    """
    process = psutil.Process(os.getpid())
    mem = process.memory_info()[0] / float(2 ** 20)
    return mem

@app.route("/")
def hello():
    """
    Flask route to render SVM classification plots for the Iris dataset.
    """
    start = timeit.default_timer()

    # Load Iris dataset
    iris = datasets.load_iris()
    X = iris.data[:, :2]
    y = iris.target

    # SVM parameters
    h = 0.02
    C = 1.0

    # Train SVM models with different kernels
    svc = svm.SVC(kernel='linear', C=C).fit(X, y)
    rbf_svc = svm.SVC(kernel='rbf', gamma=0.7, C=C).fit(X, y)
    poly_svc = svm.SVC(kernel='poly', degree=3, C=C).fit(X, y)
    lin_svc = svm.LinearSVC(C=C).fit(X, y)

    # Create a mesh to plot in
    x_min, x_max = X[:, 0].min() - 1, X[:, 0].max() + 1
    y_min, y_max = X[:, 1].min() - 1, X[:, 1].max() + 1
    xx, yy = np.meshgrid(np.arange(x_min, x_max, h), np.arange(y_min, y_max, h))

    # Titles for the plots
    titles = [
        'SVC with linear kernel',
        'LinearSVC (linear kernel)',
        'SVC with RBF kernel',
        'SVC with polynomial (degree 3) kernel'
    ]

    # Plotting
    for i, clf in enumerate((svc, lin_svc, rbf_svc, poly_svc)):
        plt.subplot(2, 2, i + 1)
        plt.subplots_adjust(wspace=0.4, hspace=0.4)
        Z = clf.predict(np.c_[xx.ravel(), yy.ravel()])
        Z = Z.reshape(xx.shape)
        plt.contourf(xx, yy, Z, cmap=plt.cm.coolwarm, alpha=0.8)
        plt.scatter(X[:, 0], X[:, 1], c=y, cmap=plt.cm.coolwarm)
        plt.xlabel('Sepal length')
        plt.ylabel('Sepal width')
        plt.xlim(xx.min(), xx.max())
        plt.ylim(yy.min(), yy.max())
        plt.xticks(())
        plt.yticks(())
        plt.title(titles[i])

    fig = plt.gcf()
    d3plot = mpld3.fig_to_html(fig, template_type="simple")

    stop = timeit.default_timer()
    print("Time(secs):", stop - start)
    print("Memory usage (MB):", memory_usage_psutil())

    return d3plot

if __name__ == "__main__":
    app.run(debug=True)
