import matplotlib.pyplot as plt
import numpy as np

def p2d2cls(data, label, correctData, correctLabel, wrongData, wrongLabel, blk=True):
    """
    Plot 2D classification results.

    Parameters:
    data (array): Data points to plot.
    label (array): Labels corresponding to the data points.
    correctData (array): Correctly classified data points.
    correctLabel (array): Labels corresponding to the correctly classified data points.
    wrongData (array): Incorrectly classified data points.
    wrongLabel (array): Labels corresponding to the incorrectly classified data points.
    blk (bool): If True, the plot blocks execution until closed. Default is True.
    """
    
    fig, ax = plt.subplots()
    
    # Data sets and their corresponding labels and colors
    datasets = [data, correctData, wrongData]
    labels = [label, correctLabel, wrongLabel]
    colors = ['b', 'g', 'r']
    
    # Plot each dataset with its labels and colors
    for dat, lab, clr in zip(datasets, labels, colors):
        true_indices = np.where(lab == 1)[0]
        false_indices = np.where(lab == 0)[0]
        
        if true_indices.size > 0:
            ax.scatter(dat[true_indices, 0], dat[true_indices, 1], marker='o', c=clr, label=f'True ({clr})')
        if false_indices.size > 0:
            ax.scatter(dat[false_indices, 0], dat[false_indices, 1], marker='x', c=clr, label=f'False ({clr})')
    
    ax.legend()
    plt.show(block=blk)
