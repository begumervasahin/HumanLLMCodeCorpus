import sys
from operator import itemgetter

def main():
    distances = []
    estimateVal = 0.0
    k = 5

    # Read input from standard input (e.g., piped or redirected from a file)
    for line in sys.stdin:
        line = line.strip()
        elements = line.split()
        distances.append(elements)

    # Sort distances based on the first element
    distances = sorted(distances, key=itemgetter(0))

    # Sum the k smallest values
    for i in range(k):
        try:
            estimateVal += float(distances[i][1])
        except ValueError:
            continue

    # Print the average
    print(estimateVal / k)

if __name__ == "__main__":
    main()
