import os
from abhishek_sharma_knn_agent import Agent
from abhishek_sharma_knn_graph import Graph

class Environment:
    """
    Environment class:
    This class receives two inputs from the main program - 1) directory name, 2) value of k.
    It then iterates over the files present in the given directory, and passes the training set, percept, and k to the agent.
    On receiving a prediction from the agent, the environment class compares it with the expected output and calculates the accuracy.
    Based on the accuracy for different values of k, it prints the output and plots a graph using the plotGraph() method of the Graph class.
    """
    def send_to_agent(self, directory, k):
        agent = Agent()
        files = os.listdir(directory)
        testing_files = [file_name for file_name in files if file_name.endswith("tst.dat")]
        number_of_files = len(testing_files)
        number_of_iterations = int(round(int(k) / 2.0))

        self._print_header(k)

        computed_stats = [[0 for _ in range(number_of_iterations + 1)] for _ in range(number_of_files)]

        for j, testing_file in enumerate(testing_files):
            testing_file_path = os.path.join(directory, testing_file)
            training_file_path = testing_file_path.replace("tst.dat", "tra.dat")
            print("|", testing_file, " " * (36 - len(testing_file)), "|", end='')

            with open(testing_file_path) as test_file:
                read_data = False
                for line in test_file:
                    if "@data" in line:
                        read_data = True
                        continue
                    if read_data:
                        percept = line.strip().split(",")
                        expected_prediction = percept[-1]

                        result = agent.sensor(k, training_file_path, percept)
                        alphabet_flag = self._is_alphabet(result)
                        self._update_computed_stats(result, expected_prediction, alphabet_flag, computed_stats[j])

            self._print_results(computed_stats[j])

        self._print_average_accuracies(computed_stats)

        # Plot the graph of accuracies
        all_stats = [sum(stats) for stats in zip(*computed_stats)]
        graph = Graph()
        graph.plotGraph(all_stats[1:], directory)

    def _print_header(self, k):
        header = "| File Name" + " " * (36 - len("File Name"))
        for i in range(1, int(k) + 1, 2):
            header += "| Accuracy (k=" + str(i) + ")" + " " * (17 - len("Accuracy (k=" + str(i) + ")"))
        print(header + "|")

    def _is_alphabet(self, result):
        try:
            list(map(float, result))
            return False
        except ValueError:
            return True

    def _update_computed_stats(self, result, expected_prediction, alphabet_flag, stats):
        count = 1
        for item in result:
            if alphabet_flag:
                if item.strip() == expected_prediction.strip():
                    stats[count] += 1
            elif str(float(item)).strip() == str(float(expected_prediction)).strip():
                stats[count] += 1
            count += 1
        stats[0] += 1

    def _print_results(self, stats):
        for n in range(1, len(stats)):
            accuracy = (float(stats[n]) / stats[0]) * 100
            accuracy_str = f"{accuracy:.2f}% ({stats[n]:03}/{stats[0]:03})"
            print(accuracy_str + " " * (27 - len(accuracy_str)) + "|", end='')
        print("")

    def _print_average_accuracies(self, computed_stats):
        all_stats = [sum(stats) for stats in zip(*computed_stats)]
        print("-" * 42)
        print("| Average Accuracies" + " " * (36 - len("Average Accuracies")) + "|")
        print("-" * 42)

        for i in range(1, len(all_stats)):
            average_accuracy = (float(all_stats[i]) / all_stats[0]) * 100
            print(f"K={i*2-1}: {average_accuracy:.2f}%")
