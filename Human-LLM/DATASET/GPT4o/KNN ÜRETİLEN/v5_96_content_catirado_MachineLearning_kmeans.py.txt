import numpy as np
import math

# Maximum number of iterations for convergence
MAX_ITERATIONS = 100

def run(data_set, k):
    """
    Run the K-means clustering algorithm on the given dataset.

    Parameters:
    data_set (array-like): The dataset to be clustered.
    k (int): The number of clusters.

    Returns:
    tuple: A tuple containing the final centroids, clusters, and number of iterations.
    """
    iterations = 0
    centroids = get_random_centroids(data_set, k)
    old_centroids = np.zeros_like(centroids)

    while not has_converged(old_centroids, centroids, iterations):
        old_centroids = centroids.copy()
        iterations += 1
        clusters = assign_clusters(data_set, centroids)
        centroids = calculate_new_centroids(clusters, data_set, k)

    return centroids, clusters, iterations

def has_converged(old_centroids, centroids, iterations):
    """
    Check if the centroids have converged or if the maximum number of iterations has been reached.

    Parameters:
    old_centroids (array-like): The centroids from the previous iteration.
    centroids (array-like): The current centroids.
    iterations (int): The current iteration number.

    Returns:
    bool: True if the centroids have converged or if the maximum number of iterations has been reached, False otherwise.
    """
    return iterations > MAX_ITERATIONS or np.array_equal(old_centroids, centroids)

def get_random_centroids(data_set, k):
    """
    Initialize k random centroids from the dataset.

    Parameters:
    data_set (array-like): The dataset to be clustered.
    k (int): The number of clusters.

    Returns:
    array: An array of k random centroids.
    """
    indices = np.random.choice(len(data_set), k, replace=False)
    return data_set[indices]

def assign_clusters(data_set, centroids):
    """
    Assign each data point to the nearest centroid to form clusters.

    Parameters:
    data_set (array-like): The dataset to be clustered.
    centroids (array-like): The current centroids.

    Returns:
    list: A list of clusters, each containing the data points assigned to that cluster.
    """
    clusters = [[] for _ in centroids]
    for point in data_set:
        closest_centroid_idx = np.argmin([euclidean_distance(point, centroid) for centroid in centroids])
        clusters[closest_centroid_idx].append(point)
    return clusters

def calculate_new_centroids(clusters, data_set, k):
    """
    Calculate the new centroids as the mean of the points in each cluster.

    Parameters:
    clusters (list): The current clusters.
    data_set (array-like): The dataset to be clustered.
    k (int): The number of clusters.

    Returns:
    array: An array of the new centroids.
    """
    new_centroids = []
    for cluster in clusters:
        if cluster:
            new_centroids.append(np.mean(cluster, axis=0))
        else:
            new_centroids.append(data_set[np.random.randint(0, len(data_set))])
    return np.array(new_centroids)

def euclidean_distance(x, y):
    """
    Calculate the Euclidean distance between two points.

    Parameters:
    x (array-like): The first point.
    y (array-like): The second point.

    Returns:
    float: The Euclidean distance between the two points.
    """
    return np.sqrt(np.sum((x - y) ** 2))

if __name__ == "__main__":
    # Example dataset
    data = np.array([
        [1.0, 2.0],
        [1.5, 1.8],
        [5.0, 8.0],
        [8.0, 8.0],
        [1.0, 0.6],
        [9.0, 11.0],
        [8.0, 2.0],
        [10.0, 2.0],
        [9.0, 3.0]
    ])
    k = 3
    centroids, clusters, iterations = run(data, k)
    print(f"Centroids:\n{centroids}")
    print(f"Iterations: {iterations}")
