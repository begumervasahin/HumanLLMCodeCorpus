import knn
import time
import gc
import numpy as np

def main():
    experiment_start_time = time.time()
    
    train_dir = './digits/trainingDigits'
    test_dir = './digits/testDigits'
    result_file_path = 'result.txt'
    result_csv_path = 'result_table.csv'
    
    # Load training and testing samples
    train_samples, train_labels = knn.load_sample_set(train_dir)
    test_samples, ground_labels = knn.load_sample_set(test_dir)
    
    ks = [3, 5, 7, 9]
    get_knn_funcs = [knn.get_knn_e_dist, knn.get_knn_e_dist_with_kdtree, knn.get_knn_m_dist]
    get_knn_funcs_str = ['get_knn_e_dist', 'get_knn_e_dist_with_kdtree', 'get_knn_m_dist']
    get_label_funcs = [knn.get_label_by_knn, knn.get_label_by_wknn]
    get_label_funcs_str = ['get_label_by_knn', 'get_label_by_wknn']
    pca_parameters = [0, 4, 8, 16, 32, 64, 128, 256]
    
    results = [['k', 'get_knn_function', 'get_label_function', 'pca_parameters', 'accuracy', 'macro_precision', 'macro_recall',
                'macro_F1', 'execution_time', 'dimension']]
    
    with open(result_file_path, 'w') as result_file:
        for k in ks:
            for get_knn_func, get_knn_func_str in zip(get_knn_funcs, get_knn_funcs_str):
                for get_label_func, get_label_func_str in zip(get_label_funcs, get_label_funcs_str):
                    for pca_parameter in pca_parameters:
                        single_result = [k, get_knn_func_str, get_label_func_str, pca_parameter]
                        
                        # Log the experiment conditions
                        print('\n******** Conditions ******', file=result_file)
                        print(f'k = {k}', file=result_file)
                        print(f'get_knn_function = {get_knn_func_str}', file=result_file)
                        print(f'get_label_function = {get_label_func_str}', file=result_file)
                        print(f'pca_parameters = {pca_parameter}', file=result_file)
                        print('**************************', file=result_file)
                        
                        # Prepare datasets
                        train_set = np.copy(train_samples)
                        test_set = np.copy(test_samples)
                        train_labels_copy = np.copy(train_labels)
                        ground_labels_copy = np.copy(ground_labels)
                        
                        # Measure the execution time
                        start_time = time.time()
                        result_labels, dimension = knn.get_test_samples_labels(
                            k, train_set, train_labels_copy, test_set, get_knn_func, get_label_func, pca_parameter
                        )
                        elapsed_time = time.time() - start_time
                        
                        # Evaluate results
                        accuracy, precision, recall, F1, macro_precision, macro_recall, macro_F1 = knn.result_evaluate(
                            ground_labels_copy, result_labels
                        )
                        
                        # Log the results
                        print(f'accuracy = {accuracy}', file=result_file)
                        single_result.append(accuracy)
                        print(f'precision = {precision}', file=result_file)
                        print(f'recall = {recall}', file=result_file)
                        print(f'F1 = {F1}', file=result_file)
                        print(f'macro_precision = {macro_precision}', file=result_file)
                        single_result.append(macro_precision)
                        print(f'macro_recall = {macro_recall}', file=result_file)
                        single_result.append(macro_recall)
                        print(f'macro_F1 = {macro_F1}', file=result_file)
                        single_result.append(macro_F1)
                        print(f'execution time = {elapsed_time}', file=result_file)
                        single_result.append(elapsed_time)
                        print(f'dimension = {dimension}', file=result_file)
                        single_result.append(dimension)
                        print(file=result_file)
                        
                        results.append(single_result)
                        gc.collect()
    
    experiment_end_time = time.time()
    total_time = experiment_end_time - experiment_start_time
    
    # Log the total experiment time
    with open(result_file_path, 'a') as result_file:
        print('***********************', file=result_file)
        print(f'Total Time = {total_time}', file=result_file)
        print('***********************', file=result_file)
    
    # Write the results to a CSV file
    with open(result_csv_path, 'w') as csv_file:
        for result in results:
            csv_file.write(', '.join(map(str, result)) + '\n')

if __name__ == '__main__':
    main()
