import time
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.neighbors import KNeighborsClassifier
from sklearn.feature_selection import SelectKBest, chi2

class bcolors:
    HEADER = '\033[95m'
    OKBLUE = '\033[94m'
    OKGREEN = '\033[92m'
    WARNING = '\033[93m'
    FAIL = '\033[91m'
    ENDC = '\033[0m'
    BOLD = '\033[1m'
    UNDERLINE = '\033[4m'

def get_features(data):
    """Extract feature names from the dataset, excluding the label."""
    features = list(data.columns)
    features.remove("label")
    return features

def feature_selection(data):
    """Select the top 5 features using the chi-squared test."""
    features = get_features(data)
    X = data[features]
    y = data["label"]
    selector = SelectKBest(score_func=chi2, k=5)
    selector.fit(X, y)
    selected_indexes = selector.get_support(indices=True)
    selected_features = [features[i] for i in selected_indexes]
    print(f"[+] Selected features -> {bcolors.OKBLUE}{selected_features}{bcolors.ENDC}")
    return data[selected_features]

def train_and_evaluate_model(X, y, description):
    """Train and evaluate the KNN model."""
    print(f"{bcolors.HEADER}{bcolors.UNDERLINE}Testing with {description}{bcolors.ENDC}")
    X_train, X_test, y_train, y_test = train_test_split(X, y, random_state=0)
    knn = KNeighborsClassifier(n_neighbors=5)

    start_time = time.time()
    knn.fit(X_train, y_train)
    training_time = time.time() - start_time
    print(f"[+] Classifier trained in {bcolors.OKGREEN}{training_time:.2f} seconds{bcolors.ENDC}")

    start_time = time.time()
    score = knn.score(X_test, y_test)
    evaluation_time = time.time() - start_time
    print(f"[+] Model evaluated in {bcolors.OKGREEN}{evaluation_time:.2f} seconds{bcolors.ENDC}")
    print(f"[!] Test score is {bcolors.OKBLUE}{score:.4f}{bcolors.ENDC}")
    print(f"{bcolors.BOLD}{bcolors.WARNING}{'-' * 50}{bcolors.ENDC}")

def with_full_features(data):
    """Train and evaluate the model with all features."""
    features = get_features(data)
    X = data[features]
    y = data["label"]
    train_and_evaluate_model(X, y, "full data")

def with_selected_features(data):
    """Train and evaluate the model with selected features."""
    X = feature_selection(data)
    y = data["label"]
    train_and_evaluate_model(X, y, "selected features")

def test_with_real_data(data, test_filename):
    """Test the model with real data from a file."""
    X = data[get_features(data)]
    y = data["label"]
    knn = KNeighborsClassifier(n_neighbors=5)
    knn.fit(X, y)

    test_results = []
    with open(test_filename, "r") as test_file:
        for line in test_file:
            tokens = list(map(float, line.strip().split(",")))
            test_results.append(knn.predict([tokens])[0])

    count_dict = {}
    for result in test_results:
        count_dict[result] = count_dict.get(result, 0) + 1
    print(count_dict)

def main():
    """Main function to execute the script."""
    filename = "preprocessed_kddcup_data.csv"
    data = pd.read_csv(filename)
    with_full_features(data)
    with_selected_features(data)

if __name__ == "__main__":
    main()
