import csv
from collections import defaultdict

def loadCsv(filename):
    # Open the output file in write mode
    with open('output.csv', 'w', newline='') as clusterfile:
        writer = csv.writer(clusterfile)
        
        # Initialize a defaultdict to store columns
        columns = defaultdict(list)
        
        # Open the input file in read mode
        with open(filename, 'r') as f:
            reader = csv.reader(f)
            
            # Read the header row
            headers = next(reader)
            
            # Write specific headers to the output file
            writer.writerow([headers[12], headers[13], headers[9]])
            
            # Use DictReader to read the CSV file into a dictionary
            f.seek(0)  # Reset file pointer to the beginning
            reader = csv.DictReader(f)
            
            # Extract specified columns and store them in the dictionary
            for row in reader:
                for (k, v) in row.items():
                    columns[k].append(v)
            
            # Write the extracted data to the output file
            for i in range(len(columns["Lon"])):
                writer.writerow([columns["Lon"][i], columns["Lat"][i], columns["Text_General_Code"][i]])

def main():
    filename = 'input.csv'
    loadCsv(filename)

if __name__ == "__main__":
    main()
