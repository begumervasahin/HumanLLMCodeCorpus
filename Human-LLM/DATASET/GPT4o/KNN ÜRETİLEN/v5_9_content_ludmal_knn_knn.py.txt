import csv
import random
import math
import operator

def load_dataset(filename, split_ratio, signal_length=4):
    """
    Load dataset from a CSV file and split into training and test sets.

    Args:
        filename (str): Path to the CSV file.
        split_ratio (float): Ratio to split data into training and test sets.
        signal_length (int, optional): Number of features in the dataset. Defaults to 4.

    Returns:
        tuple: Two lists, training set and test set.
    """
    train_set = []
    test_set = []

    with open(filename, 'r') as csvfile:
        lines = csv.reader(csvfile)
        dataset = list(lines)
        for row in dataset[:-1]:
            for i in range(signal_length):
                row[i] = float(row[i])
            if random.random() < split_ratio:
                train_set.append(row)
            else:
                test_set.append(row)
    
    return train_set, test_set

def calculate_distance(instance1, instance2, length):
    """
    Calculate the Euclidean distance between two data instances.

    Args:
        instance1 (list): First data instance.
        instance2 (list): Second data instance.
        length (int): Number of features.

    Returns:
        float: Euclidean distance between the two instances.
    """
    return math.sqrt(sum((instance1[i] - instance2[i]) ** 2 for i in range(length)))

def get_neighbors(train_set, test_instance, k):
    """
    Get k nearest neighbors of the test instance from the training set.

    Args:
        train_set (list): Training set.
        test_instance (list): Test instance.
        k (int): Number of neighbors to retrieve.

    Returns:
        list: k nearest neighbors.
    """
    length = len(test_instance) - 1
    distances = [(train_instance, calculate_distance(test_instance, train_instance, length)) for train_instance in train_set]
    distances.sort(key=operator.itemgetter(1))
    return [distances[i][0] for i in range(k)]

def get_response(neighbors):
    """
    Get the most common class label among the neighbors.

    Args:
        neighbors (list): List of neighbors.

    Returns:
        str: Most common class label.
    """
    class_votes = {}
    for neighbor in neighbors:
        response = neighbor[-1]
        class_votes[response] = class_votes.get(response, 0) + 1
    sorted_votes = sorted(class_votes.items(), key=operator.itemgetter(1), reverse=True)
    return sorted_votes[0][0]

def calculate_accuracy(test_set, predictions):
    """
    Calculate the accuracy of predictions.

    Args:
        test_set (list): Test set.
        predictions (list): List of predictions.

    Returns:
        float: Accuracy percentage.
    """
    correct = sum(1 for i in range(len(test_set)) if test_set[i][-1] == predictions[i])
    return (correct / float(len(test_set))) * 100.0

def main():
    """
    Main function to run the k-NN algorithm.
    """
    split_ratio = 0.67
    filename = 'traindata.csv'
    signal_length = 4
    k = 3

    train_set, test_set = load_dataset(filename, split_ratio, signal_length)

    print(f'Train set: {len(train_set)}')
    print(f'Test set: {len(test_set)}')

    predictions = []
    for test_instance in test_set:
        neighbors = get_neighbors(train_set, test_instance, k)
        result = get_response(neighbors)
        predictions.append(result)
        print(f'> Predicted Decision={result}, Actual Decision={test_instance[-1]}')

    accuracy = calculate_accuracy(test_set, predictions)
    print(f'Accuracy: {accuracy:.2f}%')

if __name__ == "__main__":
    main()
