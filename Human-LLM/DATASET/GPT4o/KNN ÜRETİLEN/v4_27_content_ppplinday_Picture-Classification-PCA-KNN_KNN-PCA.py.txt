import numpy as np
import sys
from sklearn.decomposition import PCA
import os
import pickle

def unpickle(file):
    """Load data from a pickle file."""
    with open(file, 'rb') as fo:
        data_dict = pickle.load(fo, encoding='latin1')
    return data_dict

def divide_train_and_test(data_dict, n):
    """Divide the data into training and testing sets."""
    data = data_dict['data']
    labels = data_dict['labels']

    data_test = data[:n].reshape(n, 3, 32, 32).transpose(0, 2, 3, 1).astype("float")
    data_train = data[n:1000].reshape(1000 - n, 3, 32, 32).transpose(0, 2, 3, 1).astype("float")

    labels_test = np.array(labels[:n])
    labels_train = np.array(labels[n:1000])

    return data_test, data_train, labels_test, labels_train

def rgb_to_gray(data):
    """Convert RGB images to grayscale."""
    num, height, width, _ = data.shape
    gray_data = np.zeros((num, height, width), float)

    for n in range(num):
        for row in range(height):
            for col in range(width):
                r, g, b = data[n, row, col]
                gray_data[n, row, col] = 0.299 * r + 0.587 * g + 0.114 * b

    return gray_data

def knn(test_data, train_data, train_labels, k):
    """k-Nearest Neighbors algorithm implementation."""
    if k <= 0:
        return None

    num_test = test_data.shape[0]
    predictions = np.zeros(num_test, int)

    for i in range(num_test):
        diff = np.tile(test_data[i], (train_data.shape[0], 1)) - train_data
        sq_diff = diff ** 2
        sq_distance = sq_diff.sum(axis=1)
        sorted_indices = sq_distance.argsort()

        class_counts = {}
        for j in range(k):
            label = train_labels[sorted_indices[j]]
            class_counts[label] = class_counts.get(label, 0) + 1

        predictions[i] = max(class_counts, key=class_counts.get)

    return predictions

def write_predictions(predictions, labels):
    """Write predictions and actual labels to a file."""
    filename = '4251258720.txt'
    if os.path.exists(filename):
        os.remove(filename)

    with open(filename, 'a') as f:
        for pred, label in zip(predictions, labels):
            f.write(f"{pred} {label}\n")

def main(k, d, n, file):
    data_dict = unpickle(file)
    data_test, data_train, labels_test, labels_train = divide_train_and_test(data_dict, n)

    data_gray_test = rgb_to_gray(data_test).reshape(n, -1)
    data_gray_train = rgb_to_gray(data_train).reshape(1000 - n, -1)

    pca = PCA(n_components=d, svd_solver='full')
    data_pca_train = pca.fit_transform(data_gray_train)
    data_pca_test = pca.transform(data_gray_test)

    predictions = knn(data_pca_test, data_pca_train, labels_train, k)
    write_predictions(predictions, labels_test)

    for pred, label in zip(predictions, labels_test):
        print(pred, label)

if __name__ == "__main__":
    k = int(sys.argv[1])
    d = int(sys.argv[2])
    n = int(sys.argv[3])
    file = sys.argv[4]
    main(k, d, n, file)
