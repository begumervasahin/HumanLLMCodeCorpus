import numpy as np
from sklearn.neighbors import KNeighborsClassifier

# Helper function to convert a value to float, returning 0 if conversion fails
def convert_num(num):
    try:
        return float(num)
    except ValueError:
        return 0

# Function to return label based on the draft pick number
def return_label(num):
    label_dist = ["Top 10 picks", "Mid 1st round", "Late 1st round",
                  "Early 2nd round", "Mid 2nd round", "Late 2nd round/undrafted"]
    return label_dist[(num - 1) // 10]

# Function to extract player statistics from a list
def extract_stats(stat_list):
    stats = [
        convert_num(stat_list[18]),           # Points
        convert_num(stat_list[20]) * 20,      # Assists (scaled)
        convert_num(stat_list[21])            # Rebounds
    ]
    return np.array(stats)

# Load data from 1999 to 2016
nba_stats = []
nba_labels = []

for year in range(1999, 2017):
    filename = f"{year}.txt"
    with open(filename, "r") as file:
        for line in file.readlines()[2:]:
            stat_list = line.split(",")
            if stat_list[11] == "":  # Skip if the player did not play
                continue

            nba_stats.append(extract_stats(stat_list))
            nba_labels.append(return_label(int(stat_list[1])))

# Load test data for the year 2003
test_year = 2003
filename = f"{test_year}.txt"
with open(filename, "r") as file:
    for line in file.readlines()[2:60]:
        stat_list = line.split(",")
        player_name = stat_list[3]  # Assuming player name is at index 3
        player_stats = extract_stats(stat_list)

        # Initialize and train KNN classifier
        knn = KNeighborsClassifier(n_neighbors=50)
        knn.fit(nba_stats, np.array(nba_labels))

        # Predict and print the label for the current player
        prediction = knn.predict([player_stats])
        print(f"Player: {player_name}, Prediction: {prediction[0]}")
