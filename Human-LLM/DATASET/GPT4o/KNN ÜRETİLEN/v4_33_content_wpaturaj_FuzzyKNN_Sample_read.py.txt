import math
import random
import pandas as pd

def get_stratified_sample(filepath, error_rate=0.01):
    """
    Generate a stratified sample from the dataset.

    Parameters:
    filepath (str): The path to the CSV file.
    error_rate (float): The acceptable error rate for the sample size calculation.

    Returns:
    pd.DataFrame: A DataFrame containing the stratified sample.
    """
    target_data = pd.read_csv(filepath, usecols=[0])
    percent_default = target_data.iloc[:, 0].mean()
    num_elements = len(target_data)
    
    num_sample = get_sample_size(percent_default, num_elements, error_rate)
    num_default = int(math.ceil(percent_default * num_sample))
    num_non_default = num_sample - num_default

    default_indices = target_data[target_data.iloc[:, 0] == 1].index.tolist()
    non_default_indices = target_data[target_data.iloc[:, 0] == 0].index.tolist()

    sampled_default_indices = random.sample(default_indices, num_default)
    sampled_non_default_indices = random.sample(non_default_indices, num_non_default)

    iter_csv = pd.read_csv(filepath, iterator=True, chunksize=10000)
    df_default = pd.concat([chunk[chunk.index.isin(sampled_default_indices)] for chunk in iter_csv])
    
    iter_csv = pd.read_csv(filepath, iterator=True, chunksize=10000)
    df_non_default = pd.concat([chunk[chunk.index.isin(sampled_non_default_indices)] for chunk in iter_csv])

    sample = pd.concat([df_non_default, df_default])

    return sample

def get_sample_size(percent, num_elements, error_rate):
    """
    Calculate the required sample size based on the specified error rate.

    Parameters:
    percent (float): The percentage of default cases in the dataset.
    num_elements (int): The total number of elements in the dataset.
    error_rate (float): The acceptable error rate for the sample size calculation.

    Returns:
    int: The calculated sample size.
    """
    Z = 2.59  # Z-score for a 99% confidence level
    sample_size = percent * (1 - percent) / ((error_rate / Z) ** 2 + (percent * (1 - percent) / num_elements))
    return int(sample_size)
