import csv
from collections import defaultdict

def load_csv(input_filename, output_filename='output.csv'):
    """
    Load data from an input CSV file, extract specific columns, and write them to an output CSV file.

    Parameters:
    - input_filename (str): The path to the input CSV file.
    - output_filename (str): The path to the output CSV file. Defaults to 'output.csv'.
    """
    # Initialize a defaultdict to store columns
    columns = defaultdict(list)

    # Open the input file and read the data
    with open(input_filename, 'r') as infile:
        reader = csv.reader(infile)
        headers = next(reader)
        
        # Create a DictReader to read the CSV file into a dictionary
        infile.seek(0)  # Reset file pointer to the beginning
        dict_reader = csv.DictReader(infile)
        
        # Extract specified columns and store them in the dictionary
        for row in dict_reader:
            for key, value in row.items():
                columns[key].append(value)

    # Define the headers to be written to the output file
    output_headers = ['Lon', 'Lat', 'Text_General_Code']

    # Open the output file and write the extracted data
    with open(output_filename, 'w', newline='') as outfile:
        writer = csv.writer(outfile)
        writer.writerow(output_headers)
        
        for i in range(len(columns['Lon'])):
            writer.writerow([columns['Lon'][i], columns['Lat'][i], columns['Text_General_Code'][i]])

def main():
    input_filename = 'input.csv'
    load_csv(input_filename)

if __name__ == "__main__":
    main()
