from sklearn.feature_extraction.text import TfidfVectorizer
import sys
import re
import numpy as np
import pymongo
from pymongo import MongoClient
import os

# Connect to MongoDB
client = MongoClient()
db = client.bookdb
posts = db.posts

# Directory containing the documents
docsDir = "/Users/jamesledoux/Documents/Drew2"
documents = []

# Read documents
for book in os.listdir(docsDir):
    if not book.startswith('.'):
        book = str(book)
        with open(os.path.join(docsDir, book), 'rb') as f:
            content = f.read()
            content = content.decode('utf-8', errors='replace')
            documents.append(content)

# TF-IDF Vectorization
tfidf = TfidfVectorizer(max_df=0.9,
                        ngram_range=(1, 1),
                        stop_words='english',
                        strip_accents='unicode', analyzer='word')
tfidf_matrix = tfidf.fit_transform(documents)
feature_names = tfidf.get_feature_names_out()

# Extract title and author information
title_author_store = []
for book in os.listdir(docsDir):
    title = None
    author = None
    if not book.startswith('.'):
        book = str(book)
        with open(os.path.join(docsDir, book), 'rb') as f:
            content = f.read().decode('utf-8').splitlines()
        for i in range(min(80, len(content))):
            if "Title: " in content[i]:
                title = content[i][7:]
            if "Author: " in content[i]:
                author = content[i][8:]
        title_author_tuple = (title, author)
        title_author_store.append(title_author_tuple)

# Build database
database = {}
for i in range(tfidf_matrix.shape[0]):
    doc = tfidf_matrix[i].toarray()[0]
    phrase_scores = [(pair[0], pair[1]) for pair in zip(range(len(doc)), doc) if pair[1] > 0]
    sorted_phrase_scores = sorted(phrase_scores, key=lambda t: t[1], reverse=True)
    local_word_dict = {feature_names[pair[0]]: pair[1] for pair in sorted_phrase_scores}
    database[title_author_store[i]] = local_word_dict

# Insert into MongoDB
for i in database.keys():
    title = i[0]
    author = i[1]
    try:
        post = {"title_id_0011": title, "author_id_0011": author}
        words = database[i]
        post.update(words)
        post_id = posts.insert_one(post).inserted_id
    except Exception as e:
        print(f"{title}, {author} failed: {e}")

print("Data successfully inserted into MongoDB")
