from text import Text

class Conversation:
    def __init__(self, dialogue_strings, start_position):
        """
        Initialize a Conversation object.

        Parameters:
        dialogue_strings (list of str): List of dialogue strings to be displayed.
        start_position (tuple): The (x, y) position where dialogue text will start to be drawn.
        """
        self.start_position = start_position
        self.dialogues = []
        self._split_into_dialogues(dialogue_strings)
        self.is_read = False
        self.current_dialogue_index = 0

    def _split_into_dialogues(self, strings):
        """
        Split the list of strings into chunks, each chunk containing up to 3 Text objects.

        Parameters:
        strings (list of str): List of dialogue strings to be split into chunks.
        """
        chunk_size = 3
        current_chunk = []

        for i, string in enumerate(strings):
            current_chunk.append(Text(string, scrollable=True))

            if len(current_chunk) >= chunk_size or i == len(strings) - 1:
                # Add empty Text objects if the chunk size is less than chunk_size
                while len(current_chunk) < chunk_size:
                    current_chunk.append(Text('', scrollable=True))
                self.dialogues.append(current_chunk)
                current_chunk = []

    def draw(self, surface, position=None):
        """
        Draw the current dialogue on the surface.

        Parameters:
        surface: The surface to draw on (e.g., a list or graphical context).
        position (tuple): The (x, y) position to start drawing. Defaults to self.start_position.
        """
        if position is None:
            position = self.start_position

        if self.dialogues:
            y_offset = 0
            current_dialogue = self.dialogues[self.current_dialogue_index]
            
            for text in current_dialogue:
                text.draw(surface, (position[0], position[1] + y_offset))
                if text.current_position < len(text.content):
                    break
                y_offset += 16

# Example usage
if __name__ == "__main__":
    dialogue_strings = [
        "Hello!",
        "How are you?",
        "This is a long message that might span multiple lines.",
        "This is another message.",
        "Final message here."
    ]
    start_position = (10, 20)
    
    conversation = Conversation(dialogue_strings, start_position)
    surface = []
    conversation.draw(surface)
    
    # Print the surface to see the output
    for item in surface:
        print(item)
