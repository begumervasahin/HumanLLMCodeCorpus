from typing import List

class Map:
    """
    Map object. Fetches the map from a .txt file and renders it to a list of lists.
    """
    def __init__(self, filename: str):
        """
        Initializes the Map object by loading the map data from a .txt file.
        Stores the height and width of the map in terms of cells, as well as the cell size in pixels.

        :param filename: Path to the file (string).
        """
        self.filename = filename
        self.board = self._load_board_from_file(filename)
        self.height = len(self.board)
        self.width = len(self.board[0]) if self.board else 0
        self.cell_size = 30  # Cell size in pixels

    @staticmethod
    def _load_file_content(filename: str) -> str:
        """
        Opens the map .txt file and reads its content.

        :param filename: Path to the file
        :return: Content of the file as a string
        """
        with open(filename, 'r') as file:
            return file.read()

    @staticmethod
    def _convert_file_to_board(file_content: str) -> List[List[str]]:
        """
        Converts the file content into a list of lists representing the map.

        :param file_content: Content of the map file
        :return: List of lists representing the map
        """
        lines = file_content.splitlines()
        return [list(line) for line in lines]

    def _load_board_from_file(self, filename: str) -> List[List[str]]:
        """
        Loads the board from a .txt file by reading its content and converting it to a list of lists.

        :param filename: Path to the file
        :return: List of lists representing the map
        """
        file_content = self._load_file_content(filename)
        return self._convert_file_to_board(file_content)

# Example usage
if __name__ == "__main__":
    # Assuming 'map.txt' is a file with the map representation
    filename = 'map.txt'
    map_object = Map(filename)

    # Print map dimensions
    print("Map Height (cells):", map_object.height)
    print("Map Width (cells):", map_object.width)

    # Print the board
    for row in map_object.board:
        print(''.join(row))
