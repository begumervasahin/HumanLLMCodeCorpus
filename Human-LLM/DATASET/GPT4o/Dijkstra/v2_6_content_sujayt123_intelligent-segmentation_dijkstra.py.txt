from DijkstraPriorityQueue import DijkstraPriorityQueue as DPQ

def is_within_bounding_box(start, end, point):
    """
    Check if a point is within the bounding box defined by start and end.

    Parameters:
    start (tuple): The starting point (x, y).
    end (tuple): The ending point (x, y). If None, there is no bounding box constraint.
    point (tuple): The point (x, y) to check.

    Returns:
    bool: True if the point is within the bounding box, False otherwise.
    """
    if end is None:
        return True
    
    min_x, max_x = min(start[0], end[0]), max(start[0], end[0])
    min_y, max_y = min(start[1], end[1]), max(start[1], end[1])
    
    return min_x <= point[0] <= max_x and min_y <= point[1] <= max_y

def shortest_path(adj_list, source, end=None):
    """
    Compute the shortest path from the source node using Dijkstra's algorithm.

    Parameters:
    adj_list (dict): Dictionary mapping each node to a list of (neighbor, edge_weight) tuples.
    source: The starting node for the pathfinding.
    end: The target node for the pathfinding. If None, find shortest paths to all nodes.

    Returns:
    dist (dict): Dictionary mapping each node to its shortest distance from the source.
    p (dict): Dictionary mapping each node to its predecessor in the shortest path.
    """
    # Initialize distances and predecessors
    dist = {node: float('inf') for node in adj_list if is_within_bounding_box(source, end, node)}
    p = {node: None for node in adj_list if is_within_bounding_box(source, end, node)}
    
    dist[source] = 0
    priority_queue = DPQ()
    priority_queue.build_heap([[dist[node], node] for node in dist])

    while priority_queue:
        # Get the node with the smallest distance
        current_distance, current_node = priority_queue.deleteMin()

        # Stop if the end node is reached
        if current_node == end:
            return dist, p
        
        # Update distances to neighbors
        for neighbor, weight in adj_list.get(current_node, []):
            if is_within_bounding_box(source, end, neighbor):
                new_distance = current_distance + weight
                if new_distance < dist[neighbor]:
                    dist[neighbor] = new_distance
                    p[neighbor] = current_node
                    priority_queue.update_priority(neighbor, new_distance)

    return dist, p

def main():
    """
    Main function to execute the pathfinding application.
    """
    # Define the adjacency list for the graph
    adj_list = {
        1: [(2, 15), (3, 71)],
        2: [(3, 7), (4, 1)],
        3: [(4, 19)],
        4: [(3, 1)]
    }
    
    # Source node
    source = 1
    
    # Compute shortest paths from the source
    distances, predecessors = shortest_path(adj_list, source)
    
    # Print the results
    print("Distances from source:", distances)
    print("Predecessors in path:", predecessors)

if __name__ == "__main__":
    main()
