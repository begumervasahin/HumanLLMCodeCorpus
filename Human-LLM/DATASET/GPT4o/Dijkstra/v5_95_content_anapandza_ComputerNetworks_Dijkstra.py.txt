def pop_min(priority_queue):
    """
    Remove and return the key with the minimum value from the priority queue.

    Parameters:
    priority_queue (dict): A dictionary where keys are nodes and values are distances.

    Returns:
    key_with_min_value (any): The key associated with the minimum value in the priority queue.
    """
    min_value = float('inf')
    key_with_min_value = None

    # Find the key with the minimum value
    for key, value in priority_queue.items():
        if value < min_value:
            min_value = value
            key_with_min_value = key

    # Remove the key with the minimum value
    if key_with_min_value is not None:
        del priority_queue[key_with_min_value]

    return key_with_min_value

def dijkstra(graph, start):
    """
    Compute the shortest paths from the start node to all other nodes using Dijkstra's algorithm.

    Parameters:
    graph (dict): An adjacency list where keys are nodes and values are dictionaries of neighbors with edge weights.
    start (any): The starting node for the algorithm.

    Returns:
    tuple: A tuple containing two dictionaries:
        - distances (dict): Shortest distances from the start node to each node.
        - predecessors (dict): Predecessors of each node in the shortest path.
    """
    # Initialize data structures
    priority_queue = {}
    distances = {}
    predecessors = {}

    # Set initial distances to infinity and predecessors to None
    for node in graph:
        distances[node] = float('inf')
        predecessors[node] = None
        priority_queue[node] = distances[node]

    # Distance to the start node is zero
    distances[start] = 0
    priority_queue[start] = distances[start]

    while priority_queue:
        # Extract the node with the minimum distance
        current_node = pop_min(priority_queue)

        # Update distances for neighboring nodes
        for neighbor, weight in graph.get(current_node, {}).items():
            new_distance = distances[current_node] + weight
            if new_distance < distances[neighbor]:
                distances[neighbor] = new_distance
                predecessors[neighbor] = current_node
                priority_queue[neighbor] = new_distance

    return distances, predecessors
