from graph import Graph
from shortest_paths import shortest_path_lengths, shortest_path_tree

def load_rail_system(file_path):
    """
    Loads rail system data from a CSV file and creates a graph representation.

    Parameters:
    file_path (str): The path to the CSV file containing rail system data.

    Returns:
    tuple: A tuple containing:
        - Graph: A graph representing the rail system.
        - dict: A dictionary mapping city names to graph vertex objects.
    """
    rail_system = Graph()
    cities = {}
    routes = set()

    with open(file_path, 'r') as file:
        for line in file:
            start_city, end_city, time_str = line.strip().split(',')
            hours, minutes = map(int, time_str.split(':'))
            duration = hours * 60 + minutes
            routes.add((start_city, end_city, duration))
            cities.setdefault(start_city)
            cities.setdefault(end_city)

    # Create vertices for each city
    for city in cities:
        cities[city] = rail_system.insert_vertex(city)

    # Add edges for each route
    for start_city, end_city, duration in routes:
        rail_system.insert_edge(cities[start_city], cities[end_city], duration)

    return rail_system, cities

def display_shortest_paths(rail_system, cities, start_city):
    """
    Displays the shortest paths from the start city to all other cities.

    Parameters:
    rail_system (Graph): The graph representing the rail system.
    cities (dict): A dictionary mapping city names to graph vertex objects.
    start_city (str): The name of the starting city.
    """
    start_vertex = cities[start_city]
    durations = shortest_path_lengths(rail_system, start_vertex)
    paths = shortest_path_tree(rail_system, start_vertex, durations)
    print(paths)

def main():
    # Path to the input file containing rail system data
    file_path = 'eurail.txt'
    
    # Load the rail system and city mappings
    rail_system, cities = load_rail_system(file_path)
    
    # Display the shortest paths from London
    display_shortest_paths(rail_system, cities, 'London')

if __name__ == "__main__":
    main()
