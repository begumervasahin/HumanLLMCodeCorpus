import numpy as np

def dijkstra(graph, source):
    """
    Dijkstra's algorithm to find the shortest path from a source node to all other nodes in a graph.

    Parameters:
        graph (2D list or np.array): The adjacency matrix of the graph.
        source (int): The starting node.

    Returns:
        list: A list of tuples containing nodes and their respective shortest distance from the source.
        dict: A dictionary mapping each node to its preceding node on the shortest path.
    """
    n = len(graph)
    infinity = float('inf')
    distance = [infinity] * n
    distance[source] = 0
    visited_nodes = []
    visited_nodes_and_cost = []
    priority_queue = {source: distance[source]}
    preceding_nodes = {}

    while priority_queue:
        current_node, current_cost = min(priority_queue.items(), key=lambda x: x[1])
        del priority_queue[current_node]
        visited_nodes.append(current_node)
        visited_nodes_and_cost.append((current_node, current_cost))

        for neighbor in range(n):
            if graph[current_node][neighbor] != 0 and neighbor not in visited_nodes:
                new_cost = distance[current_node] + graph[current_node][neighbor]
                if new_cost < distance[neighbor]:
                    distance[neighbor] = new_cost
                    priority_queue[neighbor] = new_cost
                    preceding_nodes[neighbor] = current_node

    return visited_nodes_and_cost, preceding_nodes

def get_node_input(prompt, graph_size):
    """
    Get a valid node input from the user.

    Parameters:
        prompt (str): The input prompt for the user.
        graph_size (int): The size of the graph.

    Returns:
        int: A valid node index.
    """
    node = int(input(prompt))
    while node >= graph_size or node < 0:
        node = int(input(f"The value entered is out of bounds. Try again: {prompt}"))
    return node

def main():
    filename = input('Enter the file name (without extension): ') + '.txt'
    graph = np.loadtxt(filename)
    
    start_node = get_node_input('Enter starting node: ', len(graph))
    end_node = get_node_input('Enter end node: ', len(graph))

    target_node = end_node
    costs, previous_nodes = dijkstra(graph, start_node)

    print(f"The shortest path from the start node {start_node} to every other node in the graph is:\n{costs}")

    shortest_path = []
    while end_node != start_node:
        shortest_path.append(end_node)
        end_node = previous_nodes[end_node]
    shortest_path.append(start_node)
    shortest_path.reverse()

    print(f"The shortest path from the start node {start_node} to the target node {target_node} is {shortest_path} with a distance of {dict(costs).get(target_node)}")

if __name__ == "__main__":
    main()
