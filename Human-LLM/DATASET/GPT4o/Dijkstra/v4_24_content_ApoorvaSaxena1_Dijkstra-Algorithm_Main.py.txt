def get_path(sender, start, current_sum, path_sum, ht_times):
    """
    Recursively calculates the shortest path from the start node to all other nodes.

    Parameters:
    sender (int): The sender node.
    start (int): The starting node.
    current_sum (int): The current sum of the path.
    path_sum (list): List to store the shortest path sum to each node.
    ht_times (dict): Dictionary representing the graph with adjacency list.

    Returns:
    list: Updated path_sum with the shortest paths.
    """
    if start in ht_times:
        for neighbor, weight in ht_times[start]:
            new_sum = current_sum + weight
            if path_sum[neighbor - 1] is None or new_sum < path_sum[neighbor - 1]:
                path_sum[neighbor - 1] = new_sum
                path_sum = get_path(start, neighbor, new_sum, path_sum, ht_times)
    return path_sum

def main():
    times = [
        [3, 5, 78], [2, 1, 1], [1, 3, 0], [4, 3, 59], [5, 3, 85], 
        [5, 2, 22], [2, 4, 23], [1, 4, 43], [4, 5, 75], [5, 1, 15], 
        [1, 5, 91], [4, 1, 16], [3, 2, 98], [3, 4, 22], [5, 4, 31], 
        [1, 2, 0], [2, 5, 4], [4, 2, 51], [3, 1, 36], [2, 3, 59]
    ]
    N = 5
    K = 5

    # Create a dictionary to store the graph
    ht_times = {}
    for u, v, w in times:
        if u in ht_times:
            ht_times[u].append((v, w))
        else:
            ht_times[u] = [(v, w)]

    # Initialize path_sum with None and set the starting point
    path_sum = [None] * N
    path_sum[K - 1] = 0

    # Get the shortest path sum
    path_sum = get_path(K, K, 0, path_sum, ht_times)

    # Check if all nodes are reachable
    if None not in path_sum:
        result = max(path_sum)
    else:
        result = -1

    print(result)

if __name__ == "__main__":
    main()
