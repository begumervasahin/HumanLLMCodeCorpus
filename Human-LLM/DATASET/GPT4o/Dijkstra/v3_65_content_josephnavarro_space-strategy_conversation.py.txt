class Text:
    def __init__(self, content, scrollable=False):
        """
        Initialize a Text object.

        Parameters:
        content (str): The text to be displayed.
        scrollable (bool): If True, the text can be scrolled.
        """
        self.content = content
        self.scrollable = scrollable
        self.current_position = 0

    def draw(self, surface, position):
        """
        Draw the text on the surface at the given position.

        Parameters:
        surface: The drawing surface (e.g., a list or a graphical context).
        position (tuple): The (x, y) position where the text should be drawn.
        """
        # Assuming `surface` is a list where text draw actions are appended for demonstration.
        surface.append(f"Drawing '{self.content[self.current_position:]}' at {position}")

class Conversation:
    def __init__(self, texts, start_position):
        """
        Initialize a Conversation object.

        Parameters:
        texts (list of str): List of strings to display in the conversation.
        start_position (tuple): The (x, y) position to start drawing text.
        """
        self.start_position = start_position
        self.dialogues = []
        self._format_dialogues(texts)
        self.is_read = False
        self.current_dialogue_index = 0

    def _format_dialogues(self, texts):
        """
        Format the list of texts into dialogue chunks.

        Parameters:
        texts (list of str): List of strings to be split into chunks of dialogues.
        """
        chunk_size = 3
        dialogue_chunk = []

        for i, text in enumerate(texts):
            dialogue_chunk.append(Text(text, scrollable=True))

            if len(dialogue_chunk) >= chunk_size or i == len(texts) - 1:
                # Ensure each chunk has exactly `chunk_size` Text objects
                while len(dialogue_chunk) < chunk_size:
                    dialogue_chunk.append(Text('', scrollable=True))
                self.dialogues.append(dialogue_chunk)
                dialogue_chunk = []

    def draw(self, surface, position=None):
        """
        Draw the current dialogue on the surface.

        Parameters:
        surface: The drawing surface (e.g., a list or a graphical context).
        position (tuple): The (x, y) position to start drawing. Defaults to self.start_position.
        """
        if position is None:
            position = self.start_position
        
        if self.dialogues:
            y_offset = 0
            current_dialogue = self.dialogues[self.current_dialogue_index]

            for text in current_dialogue:
                text.draw(surface, (position[0], position[1] + y_offset))
                if text.current_position < len(text.content):
                    break
                y_offset += 16
