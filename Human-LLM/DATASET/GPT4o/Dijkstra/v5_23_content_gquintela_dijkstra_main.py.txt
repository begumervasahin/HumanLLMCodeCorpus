
from dijkstra import Graph, get_dijkstra_path

def create_directed_graph():
    """
    Creates a graph with directed edges.
    """
    graph = Graph()
    edges = [
        ('a', 'b', 4), ('a', 'h', 8), ('b', 'c', 8), ('c', 'd', 7),
        ('b', 'h', 11), ('h', 'i', 7), ('i', 'c', 2), ('i', 'g', 6),
        ('h', 'g', 1), ('g', 'f', 2), ('c', 'f', 4), ('d', 'f', 14),
        ('d', 'e', 9), ('f', 'e', 10)
    ]
    for from_node, to_node, weight in edges:
        graph.add_directed_edge(from_node, to_node, weight)
    return graph

def create_undirected_graph():
    """
    Creates a graph with undirected edges.
    """
    graph = Graph()
    edges = [
        ('s', 't', 10), ('s', 'y', 5), ('t', 'x', 1), ('y', 'z', 2),
        ('t', 'y', 2), ('y', 't', 3), ('x', 'z', 4), ('z', 'x', 6),
        ('y', 'x', 9), ('z', 's', 7)
    ]
    for node1, node2, weight in edges:
        graph.add_undirected_edge(node1, node2, weight)
    return graph

def main():
    # Create graphs
    directed_graph = create_directed_graph()
    undirected_graph = create_undirected_graph()

    # Get the shortest path from 't' to 's' in the undirected graph
    path = get_dijkstra_path(undirected_graph, 't', 's')
    print(f"The shortest path from 't' to 's' is: {path}")

if __name__ == "__main__":
    main()
