"""
Best First Search (Greedy Algorithm).
Always moves towards the endpoint; not efficient with obstacles and may not find the shortest path.
"""

from queue import PriorityQueue
from pprint import pprint
import grid_helper as gh

def find_path_greedy(grid, start, end):
    """
    Find a path from start to end using the greedy algorithm.

    Parameters:
        grid (list of lists): The grid with obstacles.
        start (tuple): The starting position in the grid.
        end (tuple): The ending position in the grid.

    Returns:
        dict: A dictionary representing the path from start to end.
    """
    pq = PriorityQueue()
    pq.put((0, start))
    came_from = {start: None}
    
    while not pq.empty():
        current_pos = pq.get()[1]
        
        if current_pos == end:
            break
        
        neighbors = gh.get_neighbors(grid, current_pos[0], current_pos[1])
        
        for neighbor in neighbors:
            if neighbor not in came_from:
                priority = gh.heuristic_distance(neighbor, end, type="m")
                pq.put((priority, neighbor))
                came_from[neighbor] = current_pos
    
    return came_from

def display_path(grid, start, end, came_from):
    """
    Find and display the path on the grid from start to end.
    
    Parameters:
        grid (list of lists): The grid with obstacles.
        start (tuple): The starting position in the grid.
        end (tuple): The ending position in the grid.
        came_from (dict): The directions to reach each cell from the start position.
    """

def initialize():
    """
    Initialize the grid, find the path using greedy algorithm, and display the result.
    """
    initial_grid = gh.generate_grid_empty()
    start, end = (0, 0), (8, 8)
    
    # Find path using the greedy algorithm
    came_from = find_path_greedy(initial_grid, start, end)
    
    # Display the path on the grid
    display_path(initial_grid, start, end, came_from)
