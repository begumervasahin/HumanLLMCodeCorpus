import os
import csv

def parse_and_create_map(coordinates_directory):
    """
    Parse coordinates from CSV files in the given directory and create a map.

    Parameters:
    coordinates_directory (str): Path to the directory containing CSV files with coordinates.

    Returns:
    list of lists: The generated map, represented as a list of lists.
    """
    points = []

    # Read and collect coordinates from all CSV files in the directory
    for filename in os.listdir(coordinates_directory):
        filepath = os.path.join(coordinates_directory, filename)
        with open(filepath, 'r') as file:
            reader = csv.reader(file, delimiter=',')
            for row in reader:
                # Strip whitespace and extend points list
                cleaned_row = [entry.strip() for entry in row if entry.strip()]
                points.extend(cleaned_row)

    # Filter out any empty strings from the points list
    filtered_points = [point for point in points if point]

    # Parse the coordinates from the filtered points
    x_coords = []
    y_coords = []
    for point in filtered_points:
        x_str, y_str = point[1:].split('y')
        x_coords.append(int(x_str))
        y_coords.append(int(y_str))

    # Create the map with '.' for empty spaces
    max_x = max(x_coords, default=0)
    max_y = max(y_coords, default=0)
    map_ = [['.' for _ in range(max_x + 1)] for _ in range(max_y + 1)]

    # Place 'x' on the map at the specified coordinates
    for x, y in zip(x_coords, y_coords):
        map_[y][x] = 'x'

    # Print the map to the console
    for row in map_:
        print(''.join(row))

    # Write the map to a text file
    with open('map.txt', 'w') as file:
        for row in map_:
            file.write(''.join(row) + '\n')

    return map_

# Example usage
if __name__ == "__main__":
    parse_and_create_map("CoordinateSystem")
