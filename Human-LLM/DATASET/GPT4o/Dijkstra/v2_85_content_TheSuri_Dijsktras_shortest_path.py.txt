import heapq

class Graph:
    def __init__(self):
        """
        Initializes an empty graph using an adjacency list.
        """
        self.adjacency_list = {}

    def add_edge(self, u, v, weight):
        """
        Adds an edge between nodes u and v with the given weight.
        
        Parameters:
        u (int): The starting node.
        v (int): The ending node.
        weight (float): The weight of the edge.
        """
        if u not in self.adjacency_list:
            self.adjacency_list[u] = []
        if v not in self.adjacency_list:
            self.adjacency_list[v] = []
        self.adjacency_list[u].append((v, weight))
        self.adjacency_list[v].append((u, weight))  # For undirected graph

    def get_neighbors(self, node):
        """
        Retrieves the neighbors and weights for a given node.

        Parameters:
        node (int): The node whose neighbors are to be retrieved.

        Returns:
        list: A list of (neighbor, edge_weight) tuples.
        """
        return self.adjacency_list.get(node, [])

def shortest_path(graph, source, target):
    """
    Finds the shortest path from source to target using Dijkstra's algorithm.

    Parameters:
    graph (Graph): The graph to search within.
    source (int): The starting node.
    target (int): The destination node.

    Returns:
    tuple: A tuple containing:
        - path (list): The shortest path from source to target as a list of nodes.
        - distance (float): The total weight of the shortest path.
    """
    # Priority queue for maintaining nodes to explore, initialized with the source node
    priority_queue = []
    heapq.heappush(priority_queue, (0, source))
    
    # Track visited nodes and the path leading to each node
    visited = set()
    path = {source: None}
    remaining_distances = {}

    while priority_queue:
        curr_distance, curr_node = heapq.heappop(priority_queue)

        if curr_node in visited:
            continue
        
        if curr_node == target:
            # Return the shortest path and distance once the target is reached
            return get_path(source, curr_node, path), curr_distance
        
        for neighbor, edge_weight in graph.get_neighbors(curr_node):
            if neighbor not in visited:
                new_distance = curr_distance + edge_weight
                if neighbor not in remaining_distances or new_distance < remaining_distances[neighbor]:
                    remaining_distances[neighbor] = new_distance
                    path[neighbor] = curr_node
                    heapq.heappush(priority_queue, (new_distance, neighbor))
        
        visited.add(curr_node)
    
    return None, float('inf')  # Return None if no path is found

def get_path(source, target, path_dict):
    """
    Reconstructs the path from source to target using the path dictionary.

    Parameters:
    source (int): The starting node.
    target (int): The destination node.
    path_dict (dict): A dictionary mapping nodes to their predecessors.

    Returns:
    list: The path from source to target as a list of nodes.
    """
    path = []
    node = target
    while node is not None:
        path.append(node)
        node = path_dict[node]
    path.reverse()  # Reverse the path to start from the source
    return path

def main():
    """
    Main function to demonstrate the shortest path calculation.
    """
    # Create a graph instance and add edges
    graph = Graph()
    graph.add_edge(1, 2, 15)
    graph.add_edge(1, 3, 71)
    graph.add_edge(2, 3, 7)
    graph.add_edge(2, 4, 1)
    graph.add_edge(3, 4, 19)
    graph.add_edge(4, 3, 1)  # Adding bidirectional edge

    # Define source and target nodes
    source = 1
    target = 4
    
    # Find and print the shortest path and its distance
    path, distance = shortest_path(graph, source, target)
    
    if path:
        print(f"Shortest path from {source} to {target}: {path}")
        print(f"Total distance: {distance}")
    else:
        print(f"No path found from {source} to {target}")

if __name__ == "__main__":
    main()
