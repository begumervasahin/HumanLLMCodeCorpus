import heapq

def shortest_path(graph, source, target):
    """
    Computes the shortest path from source to target using Dijkstra's algorithm.

    Parameters:
    graph (Graph): The graph to search within.
    source (int): The starting node.
    target (int): The destination node.

    Returns:
    tuple: A tuple containing:
        - path (list): The shortest path from source to target as a list of nodes.
        - distance (float): The total weight of the shortest path.
    """
    # Priority queue to explore nodes, initialized with the source node
    priority_queue = []
    heapq.heappush(priority_queue, (0, source))

    # Dictionaries to track visited nodes, shortest path to each node, and remaining distances
    visited = set()
    path = {source: None}
    distances = {source: 0}

    while priority_queue:
        curr_distance, curr_node = heapq.heappop(priority_queue)

        if curr_node in visited:
            continue
        
        if curr_node == target:
            # Return the path and its total distance when the target is reached
            return get_path(source, target, path), curr_distance
        
        # Mark the node as visited
        visited.add(curr_node)
        
        # Explore neighbors
        for neighbor, edge_weight in graph.get_neighbors(curr_node):
            if neighbor not in visited:
                new_distance = curr_distance + edge_weight
                if neighbor not in distances or new_distance < distances[neighbor]:
                    distances[neighbor] = new_distance
                    path[neighbor] = curr_node
                    heapq.heappush(priority_queue, (new_distance, neighbor))
    
    # Return None if no path is found
    return None, float('inf')

def get_path(source, target, path_dict):
    """
    Reconstructs the path from source to target using the path dictionary.

    Parameters:
    source (int): The starting node.
    target (int): The destination node.
    path_dict (dict): A dictionary mapping nodes to their predecessors.

    Returns:
    list: The path from source to target as a list of nodes.
    """
    path = []
    node = target
    while node is not None:
        path.append(node)
        node = path_dict.get(node)
    path.reverse()  # Reverse to get the path from source to target
    return path
