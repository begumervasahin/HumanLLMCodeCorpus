from pprint import pprint
from collections import deque
import copy
import grid_helper as gh

def scan_grid(grid, start=(0, 0)):
    """
    Scan the grid to find a path from 'start' to any point.
    
    Parameters:
        grid (list of lists): The grid to be scanned.
        start (tuple): The starting position in the grid.
    
    Returns:
        dict: A dictionary representing the path from the start to each reachable cell.
    """
    queue = deque([start])
    came_from = {start: None}
    
    while queue:
        current = queue.popleft()
        neighbors = gh.get_neighbors(grid, current[0], current[1])
        
        for neighbor in neighbors:
            if neighbor not in came_from:
                queue.append(neighbor)
                came_from[neighbor] = current
    
    return came_from

def print_path(grid, start_position, end_position, directions):
    """
    Find and print the path from start to end position on the grid.
    
    Parameters:
        grid (list of lists): The grid with obstacles.
        start_position (tuple): The starting position in the grid.
        end_position (tuple): The ending position in the grid.
        directions (dict): The directions to reach each cell from the start position.
    """
    path = gh.find_path(start_position, end_position, directions)
    grid_with_path = gh.draw_path(path, copy.deepcopy(grid))
    
    print(f"Path to {end_position}:")
    pprint(grid_with_path)
    print(f"Steps: {len(path)}")

def initialize():
    """
    Initialize the grid, find paths, and print the results.
    """
    grid = gh.generate_grid_obstacle()
    start_position = (3, 0)
    
    # Scan the grid to get directions
    directions = scan_grid(grid, start_position)
    
    # Find and print paths to specified end positions
    print_path(grid, start_position, (8, 8), directions)
    print_path(grid, start_position, (4, 7), directions)

if __name__ == "__main__":
    initialize()
