import networkx as nx
import matplotlib.pyplot as plt

def draw_graph_with_path(edges, highlighted_path):
    """
    Draw a directed graph with weighted edges, highlighting a specific path.

    Parameters:
    edges (list of tuples): List of edges where each edge is (source, target, weight).
    highlighted_path (list of tuples): List of edges to highlight in red.
    """
    # Create a directed graph and add weighted edges
    graph = nx.DiGraph()
    graph.add_weighted_edges_from(edges)

    # Define a value map for node colors
    node_color_map = {1: 1.0, 4: 0.5714285714285714, 8: 0.0}
    node_colors = [node_color_map.get(node, 0.25) for node in graph.nodes()]

    # Create a dictionary for edge labels
    edge_labels = {(u, v): data['weight'] for u, v, data in graph.edges(data=True)}

    # Determine which edges to color red and which to keep black
    red_edges = highlighted_path
    edge_colors = ['red' if edge in red_edges else 'black' for edge in graph.edges()]

    # Compute positions for nodes using spring layout
    positions = nx.spring_layout(graph)

    # Draw the graph
    nx.draw_networkx_nodes(graph, positions, node_color=node_colors, node_size=500, cmap=plt.get_cmap('jet'))
    nx.draw_networkx_edges(graph, positions, edgelist=red_edges, edge_color='red', arrows=True)
    nx.draw_networkx_edges(graph, positions, edgelist=[e for e in graph.edges() if e not in red_edges], edge_color='black', arrows=False)
    nx.draw_networkx_labels(graph, positions)
    nx.draw_networkx_edge_labels(graph, positions, edge_labels=edge_labels)

    # Display the graph
    plt.show()

# Example usage
if __name__ == "__main__":
    # Define edges and path to highlight
    edges = [(1, 2, 4), (1, 3, 1), (2, 3, 2), (3, 4, 5)]
    highlighted_path = [(1, 2), (2, 3)]

    # Draw the graph with the specified path highlighted
    draw_graph_with_path(edges, highlighted_path)
