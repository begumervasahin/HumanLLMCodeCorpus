import collections
import heapq

class Queue:
    """
    First in first out queue. Implemented using collections.deque.
    Used for BFS.
    """
    def __init__(self):
        self.elements = collections.deque()

    def is_empty(self) -> bool:
        """Check if the queue is empty."""
        return not self.elements

    def add(self, item):
        """Add an item to the queue."""
        self.elements.append(item)

    def pop(self):
        """Remove and return an item from the queue."""
        return self.elements.popleft()

    def __len__(self) -> int:
        """Return the number of items in the queue."""
        return len(self.elements)


class PriorityQueue:
    """
    Priority queue. Implemented using heapq from the collections module.
    Used in Dijkstra and A* algorithms.
    """
    def __init__(self):
        self.elements = []

    def is_empty(self) -> bool:
        """Check if the priority queue is empty."""
        return not self.elements

    def put(self, item, priority):
        """Add an item to the priority queue with a given priority."""
        heapq.heappush(self.elements, (priority, item))

    def pop(self):
        """Remove and return the item with the highest priority."""
        return heapq.heappop(self.elements)[1]

    def __len__(self) -> int:
        """Return the number of items in the priority queue."""
        return len(self.elements)


# Example usage of Queue
if __name__ == "__main__":
    queue = Queue()
    queue.add(1)
    queue.add(2)
    queue.add(3)
    
    print("Queue contents:")
    while not queue.is_empty():
        print(queue.pop())

    # Example usage of PriorityQueue
    priority_queue = PriorityQueue()
    priority_queue.put("task1", 1)
    priority_queue.put("task2", 2)
    priority_queue.put("task3", 0)
    
    print("\nPriorityQueue contents:")
    while not priority_queue.is_empty():
        print(priority_queue.pop())
