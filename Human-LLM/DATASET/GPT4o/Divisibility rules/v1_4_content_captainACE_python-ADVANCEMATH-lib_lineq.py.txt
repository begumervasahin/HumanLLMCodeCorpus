def solve_linear_equation_cramers_rule():
    """
    Solve a system of linear equations using Cramer's Rule.
    The equations are of the form:
    a*x + b*y = c
    k*x + l*y = m
    """
    print("SOLVE A LINEAR EQUATION BY CRAMER'S RULE OF DISCRIMINANTS")
    
    print("Equation no. 1")
    a = int(input("Number for 'a' term: "))
    b = int(input("Number for 'b' term: "))
    c = int(input("Number for 'c' term: "))
    
    print_equation(a, b, c)
    
    print("----------------------------")
    
    print("Equation no. 2")
    k = int(input("Number for 'k' term: "))
    l = int(input("Number for 'l' term: "))
    m = int(input("Number for 'm' term: "))
    
    print_equation(k, l, m)
    
    print("--------------------------------------------")
    print("The equations will be solved by Cramer's rule")
    
    d = a * l - b * k
    dx = c * l - b * m
    dy = a * m - c * k
    
    if d == 0:
        if dx == 0 and dy == 0:
            print("The system has infinitely many solutions.")
        else:
            print("The system has no solution.")
    else:
        x = dx / d
        y = dy / d
        
        print(f'd = {d}')
        print(f'dx = {dx}')
        print(f'dy = {dy}')
        print(f'x = {x}')
        print(f'y = {y}')

def print_equation(a, b, c):
    """
    Print the linear equation in the form ax + by = c.
    
    Args:
        a (int): Coefficient of x.
        b (int): Coefficient of y.
        c (int): Constant term.
    """
    if a == 0:
        print(f'{b}y = {c}')
    elif b == 0:
        print(f'{a}x = {c}')
    else:
        print(f'{a}x + {b}y = {c}')

if __name__ == "__main__":
    solve_linear_equation_cramers_rule()
