import math
from turtle import *

# Constants
G = 6.67428e-11  # Gravitational constant
AU = 149.6e6 * 1000  # Astronomical unit in meters
SCALE = 250 / 384400000  # Scale factor for visualization

class Body(Turtle):
    """
    Subclass of Turtle representing a gravitationally-acting body.
    Extra attributes:
    mass : mass in kg
    ax, ay: x, y acceleration in m/s^2
    vx, vy: x, y velocities in m/s
    px, py: x, y positions in m
    """
    def __init__(self, name, mass, px, py, vx, vy, color):
        """
        Initialize a Body object.

        Args:
            name (str): The name of the body.
            mass (float): The mass of the body in kg.
            px (float): The initial x position of the body in meters.
            py (float): The initial y position of the body in meters.
            vx (float): The initial x velocity of the body in m/s.
            vy (float): The initial y velocity of the body in m/s.
            color (str): The color of the body for visualization.
        """
        super().__init__()
        self.name = name
        self.mass = mass
        self.px = px
        self.py = py
        self.vx = vx
        self.vy = vy
        self.penup()
        self.hideturtle()
        self.pencolor(color)
        self.ax = 0.0
        self.ay = 0.0

    def attraction(self, other):
        """
        Calculate the gravitational force exerted on this body by another body.

        Args:
            other (Body): Another Body object.

        Returns:
            tuple: The x and y components of the gravitational force.
        """
        if self is other:
            raise ValueError(f"Attraction of object {self.name} to itself requested")

        dx = other.px - self.px
        dy = other.py - self.py
        distance = math.sqrt(dx**2 + dy**2)

        if distance == 0:
            raise ValueError(f"Collision between objects {self.name} and {other.name}")

        force = G * self.mass * other.mass / distance**2
        theta = math.atan2(dy, dx)
        fx = math.cos(theta) * force
        fy = math.sin(theta) * force
        return fx, fy

def update_info(step, bodies):
    """
    Display information about the status of the simulation.

    Args:
        step (int): The current step of the simulation.
        bodies (list): The list of Body objects.
    """
    print(f'Step {step}')
    for body in bodies:
        print(f'{body.name:<8}  Pos.={body.px / AU:>6.2f} {body.py / AU:>6.2f} Vel.={body.vx:>10.3f} {body.vy:>10.3f}')
    print()

def loop(bodies):
    """
    Run the simulation loop, updating the positions of all the provided bodies.

    Args:
        bodies (list): The list of Body objects.
    """
    timestep = 600
    step = 1

    while True:
        update_info(step, bodies)
        step += 1

        forces = {}
        for body in bodies:
            total_fx, total_fy = 0.0, 0.0
            for other in bodies:
                if body is other:
                    continue
                fx, fy = body.attraction(other)
                total_fx += fx
                total_fy += fy
            forces[body] = (total_fx, total_fy)

        for body in bodies:
            fx, fy = forces[body]
            body.vx += fx / body.mass * timestep
            body.vy += fy / body.mass * timestep
            body.px += body.vx * timestep
            body.py += body.vy * timestep
            body.goto(body.px * SCALE, body.py * SCALE)
            body.dot(3)

def create_bodies():
    """
    Create a list of Body objects to simulate.

    Returns:
        list: A list of Body objects.
    """
    bodies = [
        Body('rock', 10**25, 384400000, 0, 0, -1200, 'green'),
        Body('rock2', 10**25, -384400000, 0, 0, 1200, 'green'),
        Body('rock3', 10**25, 0, 384400000, 1200, 0, 'green'),
        Body('rock3a', 10**25, 0, 484400000, 3700, 0, 'green'),
        Body('rock4', 10**25, 0, -384400000, -1200, 0, 'green'),
        Body('rock4a', 10**22, 0, -484400000, -3700, 0, 'green'),
        Body('rock40', 10**27, 0, 0, 0, 0, 'green'),
        Body('rock41', 10**22, 0, -100000000, -2500, 0, 'green')
    ]
    return bodies

def main():
    """
    Initialize the bodies and start the simulation loop.
    """
    bodies = create_bodies()
    loop(bodies)

if __name__ == '__main__':
    main()
