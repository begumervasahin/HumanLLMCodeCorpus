from divisibility_rule import div_rule

def is_divisible(n, m):
    """
    Check if a number n is divisible by m using a custom divisibility rule.

    Parameters:
    n (str): The number to check, provided as a string.
    m (int): The divisor.

    Returns:
    bool: True if n is divisible by m, False otherwise.
    """
    divisibility_rule = div_rule(m)
    total = 0
    n_reversed = n[::-1]  # Reverse the number string

    for digit in n_reversed:
        param = next(divisibility_rule)
        total += int(digit) * param
        print(total)
        
        # Adjust total to prevent it from becoming too large
        if abs(total) > 3 * m:
            if total > 0:
                total -= 3 * m
            else:
                total += 3 * m

    return total % m == 0

# Example usage
if __name__ == "__main__":
    number = "1946"
    divisor = 7
    result = is_divisible(number, divisor)
    if result:
        print(f"The number {number} is divisible by {divisor}.")
    else:
        print(f"The number {number} is not divisible by {divisor}.")
