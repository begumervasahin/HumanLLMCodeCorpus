from itertools import product
import scipy.sparse as sps
import sympy as sp
import numpy as np
from math import floor
from HopfConstructions_Functions import Left_Action_Matrix, Right_Action_Matrix, CreateBasisVectors, Tensor_Mult

def char_poly(M):
    """
    Compute the characteristic polynomial of a matrix.

    Parameters:
    M (scipy.sparse.csr_matrix): The matrix.

    Returns:
    sympy.Poly: The characteristic polynomial.
    """
    matrix = sp.SparseMatrix(M.toarray())
    return matrix.berkowitz_charpoly()

def multiplicity(poly, root, derivatives):
    """
    Compute the multiplicity of a root in a polynomial.

    Parameters:
    poly (sympy.Poly): The polynomial.
    root (float): The root to check.
    derivatives (list): List of derivatives.

    Returns:
    list: Multiplicity and updated list of derivatives.
    """
    D = derivatives
    i = 1
    while True:
        if i < len(D):
            if D[i].eval(root) != 0:
                return [i, D]
        else:
            temp = D[-1].diff()
            D.append(temp)
            if temp.eval(root) != 0:
                return [i, D]
        i += 1

def divisors(n):
    """
    Compute the divisors of an integer.

    Parameters:
    n (int): The integer.

    Returns:
    list: List of divisors.
    """
    return [i for i in range(1, floor(n**0.5) + 1) if n % i == 0]

def remove_zeros_at_zero(poly):
    """
    Remove zeros at the origin of a polynomial.

    Parameters:
    poly (sympy.Poly): The polynomial.

    Returns:
    sympy.Poly: The polynomial without zeros at the origin.
    """
    monomial = poly.EM().as_expr()
    return sp.exquo(poly, monomial)

def higman_trace(A):
    """
    Compute the Higman trace of an algebra.

    Parameters:
    A (Algebra): The algebra object.

    Returns:
    scipy.sparse.csr_matrix: The Higman trace matrix.
    """
    dim = A.dim
    if A.casimir_flag == 'no':
        A.GetCasimir()
    
    mult = A.mult
    V = CreateBasisVectors(dim)
    casimir = A.casimir
    higman_trace = sps.csr_matrix((dim, dim))

    for i, j in product(range(dim), repeat=2):
        if casimir[dim * i + j] != 0:
            higman_trace += Left_Action_Matrix(V[i], mult).dot(Right_Action_Matrix(V[j], mult))
    
    return higman_trace

def compute_M(A):
    """
    Compute the matrix M of an algebra.

    Parameters:
    A (Algebra): The algebra object.

    Returns:
    scipy.sparse.csr_matrix: The computed matrix M.
    """
    dim = A.dim
    if A.casimir_flag == 'no':
        A.GetCasimir()
    
    mult = A.mult
    casimir = A.casimir
    tensor_mult = Tensor_Mult(A, A)
    C = mult.dot(tensor_mult.dot(np.kron(casimir, casimir)))
    
    return Left_Action_Matrix(C, A.mult)
