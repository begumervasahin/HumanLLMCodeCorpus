def check_divisibility_by_7(number):
    """
    Check if a given number, after repeatedly applying a specific transformation, is divisible by 7.

    Parameters:
    number (int): The initial number to check.

    Returns:
    str: Message indicating whether the final number is divisible by 7 or not.
    """
    current_number_str = str(number)
    
    # Continue transforming the number until it has 2 digits or becomes zero
    while len(current_number_str) > 2 and number > 0:
        print(current_number_str)
        prefix = current_number_str[:-1]  # All digits except the last one
        last_digit = int(current_number_str[-1])  # The last digit
        transformed_number = int(prefix) - (2 * last_digit)
        current_number_str = str(transformed_number)
        number = transformed_number
    
    # Check if the resulting number is divisible by 7
    if number % 7 == 0:
        return f"Number {current_number_str} is divisible by 7!"
    else:
        return f"Number {current_number_str} is not divisible by 7."

# Example usage
initial_number = 1946
result = check_divisibility_by_7(initial_number)
print(result)
