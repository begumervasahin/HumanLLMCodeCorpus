def division_func(listA, listB, c):
    """
    Filter elements in listA based on the number of divisors they have in listB.
    If an element in listA is divisible by at least `c` elements in listB, it is added to the result.

    Args:
        listA (list): The list of integers to be filtered.
        listB (list): The list of divisors.
        c (int): The minimum number of divisors required.
    """
    result = {i for i in listA if sum(1 for j in listB if i % j == 0) >= c}
    print(result)

def main():
    """
    Main function to handle user input and call the division function.
    """
    try:
        inputA = input("Enter a list of numbers separated by spaces: ")
        listA = list(map(int, inputA.split()))

        inputB = input("Enter another list of numbers separated by spaces: ")
        listB = list(map(int, inputB.split()))

        c = len(listB) // 2  # Use integer division for threshold

        division_func(listA, listB, c)
    except ValueError:
        print("Please enter valid lists of integers.")

if __name__ == "__main__":
    main()
