def division_func(listA, listB, c):
    """
    This function filters elements in listA based on the number of divisors they have in listB.
    If an element in listA is divisible by at least `c` elements in listB, it is added to the result.

    Args:
        listA (list): The list of integers to be filtered.
        listB (list): The list of divisors.
        c (int): The minimum number of divisors required.

    Returns:
        list: A list of integers from listA that are divisible by at least `c` elements in listB.
    """
    list_answer = []
    for num in listA:
        count = sum(1 for divisor in listB if num % divisor == 0)
        if count >= c:
            list_answer.append(num)
    list_answer = list(set(list_answer))  # Remove duplicates and convert back to list
    print(list_answer)

def main():
    # Input lists and convert them to lists of integers
    inputA = input("Enter a list of numbers separated by a space: ")
    listA = list(map(int, inputA.split()))

    inputB = input("Enter a list of numbers separated by a space: ")
    listB = list(map(int, inputB.split()))

    # Calculate the threshold `c` as half the length of listB
    c = len(listB) // 2

    # Call the function
    division_func(listA, listB, c)

    # Wait for user input to exit
    input("Press any key to exit ")

if __name__ == "__main__":
    main()
