from flask import Flask, render_template, request
import divisibility_dfa

app = Flask(__name__)

@app.route('/')
def homepage():
    """
    Render the homepage template.
    """
    return render_template('index.html')

@app.route('/faq')
def faq():
    """
    Render the FAQ page template.
    """
    return render_template('faq.html')

@app.route('/regex', methods=['POST'])
def generate_regex():
    """
    Handle the form submission to generate a regex for divisibility.

    Returns:
        str: Rendered template with the generated regex or an error message.
    """
    form = request.form
    
    # Validate form fields
    if 'base' not in form or 'divisor' not in form:
        return 'Missing form fields', 400
    
    base = form['base']
    div = form['divisor']
    
    # Validate base
    if base not in ('binary', 'decimal', 'hexadecimal'):
        return 'Invalid base', 400
    
    # Validate divisor
    if div not in [str(i) for i in range(1, 10)]:
        return 'Invalid divisor', 400
    
    div = int(div)
    
    # Build DFA and generate regex
    dfa = divisibility_dfa.build_dfa(base, div)
    dfa.set_start(div)
    regex = divisibility_dfa.dfa_to_regex(dfa, 0)
    
    return render_template('regex.html', regex=regex)
