def fibonacci(n, memo={}):
    """
    Calculate the nth Fibonacci number using memoization.
    
    Args:
        n (int): The position in the Fibonacci sequence.
    
    Returns:
        int: The nth Fibonacci number.
    """
    if n in memo:
        return memo[n]
    if n <= 1:
        memo[n] = n
    else:
        memo[n] = fibonacci(n - 1, memo) + fibonacci(n - 2, memo)
    return memo[n]

def is_prime(n):
    """
    Check if a number is prime.
    
    Args:
        n (int): The number to check.
    
    Returns:
        bool: True if the number is prime, False otherwise.
    """
    if n < 2:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False
    for i in range(3, int(n ** 0.5) + 1, 2):
        if n % i == 0:
            return False
    return True

def fizzbuzz(n):
    """
    Generate FizzBuzz output for a given number.
    
    Args:
        n (int): The number to check.
    
    Returns:
        str: The FizzBuzz output for the number.
    """
    if is_prime(n):
        return "BuzzFizz"
    
    result = ""
    if n % 3 == 0:
        result += "Fizz"
    if n % 5 == 0:
        result += "Buzz"
    return result if result else str(n)

def main():
    """
    Main function to generate Fibonacci numbers and their FizzBuzz equivalents.
    """
    n = 15
    fibonacci_numbers = [fibonacci(i) for i in range(n)]
    fizzbuzz_results = [fizzbuzz(num) for num in fibonacci_numbers]
    
    print("Fibonacci - FizzBuzz:")
    for fib_num, fb_result in zip(fibonacci_numbers, fizzbuzz_results):
        print(f"{fib_num} -> {fb_result}")

if __name__ == "__main__":
    main()
