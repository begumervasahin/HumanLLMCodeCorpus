from collections import namedtuple

# Define a namedtuple to store elements with their index and value
Element = namedtuple('Element', ['index', 'value'])

class DivisiblePairSums:
    def __init__(self, n, k, ar):
        self.n = n
        self.k = k
        self.ar = ar
        # Create a list of Element namedtuples with index and value
        self.ints = [Element(index=index, value=value) for index, value in enumerate(ar)]
    
    def answer_question(self):
        # Validate the input data
        if self.validate_data():
            # Count pairs whose sums are divisible by k
            return len(
                [
                    (x.value, y.value) for x in self.ints
                    for y in self.ints if
                    (x.value + y.value) % self.k == 0 and
                    x.index < y.index
                ]
            )
        return "Error: invalid input"
    
    def validate_data(self):
        # Validate constraints for n, k, and length of array ar
        if self.n > 100 or self.k > 100 or len(self.ar) > 100 or self.n < 2 or self.k < 1 or len(self.ar) < 1:
            return False
        return True

# Example usage:
if __name__ == "__main__":
    n = 6
    k = 3
    ar = [1, 3, 2, 6, 1, 2]
    dps = DivisiblePairSums(n, k, ar)
    result = dps.answer_question()
    print(result)  # Output should be 5
