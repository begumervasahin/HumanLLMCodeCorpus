from collections import namedtuple

# Define a namedtuple to store elements with their index and value
Element = namedtuple('Element', ['index', 'value'])

class DivisiblePairSums:
    def __init__(self, n, k, ar):
        self.n = n
        self.k = k
        self.ar = ar
        # Create a list of Element namedtuples with index and value
        self.elements = [Element(index, value) for index, value in enumerate(ar)]
    
    def answer_question(self):
        """Counts the number of pairs whose sum is divisible by k."""
        if not self._validate_data():
            return "Error: invalid input"
        
        # Find and count pairs whose sums are divisible by k
        pairs = [
            (x.value, y.value) for x in self.elements
            for y in self.elements
            if (x.value + y.value) % self.k == 0 and x.index < y.index
        ]
        return len(pairs)
    
    def _validate_data(self):
        """Validates input constraints."""
        return 2 <= self.n <= 100 and 1 <= self.k <= 100 and len(self.ar) <= 100

# Example usage
if __name__ == "__main__":
    n = 6
    k = 3
    ar = [1, 3, 2, 6, 1, 2]
    dps = DivisiblePairSums(n, k, ar)
    result = dps.answer_question()
    print(result)  # Output should be 5
