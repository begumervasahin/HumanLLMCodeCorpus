def is_prime(x):
    """
    Check if a number x is a prime number.
    
    Parameters:
    x (int): The number to check.

    Returns:
    int or None: The number itself if it is prime, otherwise None.
    """
    for i in range(2, x):
        if x % i == 0:
            return None
    return x

def generate_prime_related_numbers(limit):
    """
    Generate a list of numbers based on prime numbers between 9 and a specified limit.
    The list includes squares of primes and products of distinct primes.

    Parameters:
    limit (int): The upper limit for the range of numbers to check for primes.

    Returns:
    list: A sorted list of generated numbers.
    """
    primes = list(filter(None, map(is_prime, range(9, limit))))
    prime_products = [1]

    for i, prime_i in enumerate(primes):
        square = prime_i * prime_i
        if square < limit:
            prime_products.append(square)
            for j in range(i + 1, len(primes)):
                product = prime_i * primes[j]
                if product < limit:
                    prime_products.append(product)
                else:
                    break
        else:
            break

    all_numbers = sorted(set(primes + prime_products))
    return all_numbers

def main():
    """
    Main function to generate and print the list of prime-related numbers up to 200.
    """
    result = generate_prime_related_numbers(201)
    print(result)

if __name__ == "__main__":
    main()
