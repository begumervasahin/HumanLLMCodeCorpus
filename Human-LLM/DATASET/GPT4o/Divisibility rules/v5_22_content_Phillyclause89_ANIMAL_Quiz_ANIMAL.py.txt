def number_checker():
    while True:
        user_input = input("Enter a number or enter 'q' to quit: ")
        
        if user_input.lower() in ['q', 'quit']:
            print("Goodbye!")
            break
        
        try:
            number = int(user_input)
            
            if number % 32 == 0:
                print("ANIMAL")
            elif number % 16 == 0:
                print("ANIM")
            elif number % 8 == 0:
                print("AN")
            else:
                print("Try a different number next time.")
        
        except ValueError:
            print("Entry is unacceptable. Please enter a valid number.")

number_checker()
