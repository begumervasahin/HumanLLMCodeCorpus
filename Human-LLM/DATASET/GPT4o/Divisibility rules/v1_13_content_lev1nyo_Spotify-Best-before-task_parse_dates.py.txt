import sys
import datetime
import itertools

def parse_date(year, month, day):
    """
    Parse the date from year, month, and day strings. Adjusts year if less than 1000.
    """
    try:
        year = int(year) + 2000 if int(year) < 1000 else int(year)
        return datetime.date(year, int(month), int(day))
    except ValueError:
        return False

def main():
    """
    Main function to read the file, parse dates, and print the earliest valid date.
    """
    try:
        filename = sys.argv[1]
    except IndexError:
        print("Missing file name!")
        sys.exit(1)
    
    try:
        with open(filename) as f:
            lines = [line.strip() for line in f]
    except FileNotFoundError:
        print(f"File not found: {filename}")
        sys.exit(1)

    for line in lines:
        parts = line.split('/')
        dates = []

        for year, month, day in itertools.permutations(parts, 3):
            date = parse_date(year, month, day)
            if date and 2000 <= date.year < 3000:
                dates.append(date)
        
        if dates:
            print(min(dates))
        else:
            print(f"{line} is illegal")

if __name__ == '__main__':
    main()
