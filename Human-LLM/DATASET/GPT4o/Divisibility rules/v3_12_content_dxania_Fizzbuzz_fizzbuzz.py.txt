def get_input():
    """
    Prompt the user to enter two lists of elements and print the results of the FizzBuzz calculation
    based on the total length of both lists.
    """
    # Prompt user for the first list input
    list1 = get_list_from_user("Enter the first list (separate elements with commas): ")
    print(f"List1: {list1}")
    print(f"List1 is {len(list1)} elements long")

    # Prompt user for the second list input
    list2 = get_list_from_user("Enter the second list (separate elements with commas): ")
    print(f"List2: {list2}")
    print(f"List2 is {len(list2)} elements long")

    # Call fizzbuzz function and print the result
    result = fizzbuzz(len(list1), len(list2))
    print(f"Result: {result}")

def get_list_from_user(prompt):
    """
    Helper function to get a list of elements from the user input.
    
    Parameters:
    prompt (str): The input prompt message for the user.
    
    Returns:
    list: A list of elements entered by the user.
    """
    return [item.strip() for item in input(prompt).split(",")]

def fizzbuzz(len1, len2):
    """
    Calculate the FizzBuzz result based on the combined length of two lists.
    
    Parameters:
    len1 (int): The length of the first list.
    len2 (int): The length of the second list.
    
    Returns:
    str: 'Fizz' if the combined length is divisible by 3.
         'Buzz' if the combined length is divisible by 5.
         'FizzBuzz' if the combined length is divisible by both 3 and 5.
         Otherwise, return the combined length as a string.
    """
    # Calculate the total length of both lists
    total_length = len1 + len2

    # Determine the result based on the total length
    if total_length % 3 == 0 and total_length % 5 == 0:
        return 'FizzBuzz'
    elif total_length % 3 == 0:
        return 'Fizz'
    elif total_length % 5 == 0:
        return 'Buzz'
    else:
        return str(total_length)

# Execute the script by calling the get_input function
if __name__ == "__main__":
    get_input()
