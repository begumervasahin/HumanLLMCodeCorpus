"""
Given any two numbers, this program can help the user
decide if the first number is divisible by the second
number. This program can validate for positive
integers.
SUBSTITUTION FOR MODULUS OPERATOR (%)
A - B * (A // B), where A and B are positive integers
Created by:    Victor Nguyen
Updated on:    12/12/2016
"""

def validate(char):
    """
    Validate user input to ensure it is 'Y', 'y', 'N', or 'n'.
    
    Args:
        char (str): The input character to validate.
    
    Returns:
        str: A valid character ('Y', 'y', 'N', 'n').
    """
    while char.lower() not in ['y', 'n']:
        char = input("Invalid input. Please try again. Are those numbers correct? (Y/N) ")
    return char

def get_positive_integer(prompt):
    """
    Prompt the user to enter a positive integer.
    
    Args:
        prompt (str): The input prompt for the user.
    
    Returns:
        int: A positive integer entered by the user.
    """
    while True:
        try:
            num = int(input(prompt))
            if num < 0:
                raise ValueError
            return num
        except ValueError:
            print("Value error. Remember that this program will only take positive INTEGERS. Starting over.\n")

def calculate_x_factor(second_num):
    """
    Calculate the X factor for the given second number.
    
    Args:
        second_num (int): The second number (divisor).
    
    Returns:
        int: The calculated X factor.
    """
    x_factor = 1
    multiplier = 1
    while x_factor % 10 != 9:
        x_factor = second_num * multiplier
        multiplier += 1
    return (x_factor // 10) + 1

def check_divisibility(first_num, second_num, x_factor):
    """
    Check if the first number is divisible by the second number using the calculated X factor.
    
    Args:
        first_num (int): The first number (dividend).
        second_num (int): The second number (divisor).
        x_factor (int): The calculated X factor.
    
    Returns:
        str: A message indicating whether the first number is divisible by the second number.
    """
    ones_digit = first_num % 10
    ones_digit *= x_factor
    penultimate = (first_num // 10) + ones_digit

    print(f"Is {penultimate} divisible by {second_num}?")
    if penultimate % second_num == 0:
        return "Yes, it is divisible."
    else:
        return "No, it is not divisible."

def main():
    """
    Main function to check if the first number is divisible by the second number
    using a custom method without the modulus operator.
    """
    first_num = get_positive_integer("Please enter the first positive integer: ")
    second_num = get_positive_integer("Please enter the second positive integer: ")

    print(f"Your first number is {first_num} and your second number is {second_num}")
    choice = input("Are these correct? (Y/N) ")
    choice = validate(choice)

    if choice.lower() == 'n':
        print("Please restart the program and enter the correct numbers.")
        return

    x_factor = calculate_x_factor(second_num)
    print(f"Your X Factor is {x_factor}")

    result = check_divisibility(first_num, second_num, x_factor)
    print(result)

if __name__ == "__main__":
    main()
