from tkinter import *
from tkinter import ttk, filedialog
from pageone import PageOne  # Ensure PageOne class is properly defined in the pageone module

# Define a large font style for consistency
LARGE_FONT = ("Verdana", 16)

class MainApp(Tk):
    """Main application class for managing frames and switching between them."""
    
    def __init__(self, *args, **kwargs):
        """Initialize the main application window."""
        super().__init__(*args, **kwargs)
        self.title("KetokMagicHalal")

        # Create a container frame to hold all other frames
        container = Frame(self)
        container.pack(side="top", fill="both", expand=True)

        # Dictionary to keep track of frames
        self.frames = {}
        self.currentFrame = None

        # Initialize and store frames
        self.initialize_frames(container)
        
        # Display the initial frame
        self.show_frame(PageOne)

    def initialize_frames(self, container):
        """Initialize and store the frames in a dictionary."""
        for F in (PageOne,):  # Add more page classes to this tuple if needed
            frame = F(container, self)
            self.frames[F] = frame
            frame.pack_forget()

    def show_frame(self, cont):
        """Show a frame for the given class name."""
        if self.currentFrame is not None:
            self.currentFrame.pack_forget()
        
        self.currentFrame = self.frames[cont]
        self.currentFrame.pack(side=TOP, fill=BOTH, expand=True, padx=20, pady=20)
        self.currentFrame.load_db(db_path)  # Assuming load_db is a method in PageOne

if __name__ == '__main__':
    # Set your database path here
    db_path = 'path/to/your/database'
    
    # Create and configure the main application
    app = MainApp()
    app.geometry("800x600")
    
    # Start the main application loop
    app.mainloop()
