import numpy as np
import pandas as pd
from sklearn.feature_extraction.text import CountVectorizer, TfidfTransformer
from sklearn.metrics.pairwise import cosine_similarity

def load_data(file_path):
    """
    Load data from a CSV file.

    Args:
        file_path (str): Path to the CSV file.

    Returns:
        pd.DataFrame: Loaded data as a pandas DataFrame.
    """
    return pd.read_csv(file_path)

def preprocess_texts(df, text_column):
    """
    Extract text data from a DataFrame.

    Args:
        df (pd.DataFrame): DataFrame containing text data.
        text_column (str): Column name containing the text data.

    Returns:
        list: List of texts.
    """
    return df[text_column].tolist()

def compute_tfidf_matrix(texts, stop_words='english'):
    """
    Compute the TF-IDF matrix for a list of texts.

    Args:
        texts (list): List of texts.
        stop_words (str or list): Stop words to use for vectorization.

    Returns:
        sparse matrix: TF-IDF matrix.
        CountVectorizer: Fitted CountVectorizer instance.
    """
    vectorizer = CountVectorizer(stop_words=stop_words)
    freq_term_matrix = vectorizer.fit_transform(texts)
    tfidf_transformer = TfidfTransformer(norm="l2")
    tf_idf_matrix = tfidf_transformer.fit_transform(freq_term_matrix)
    return tf_idf_matrix, vectorizer

def compute_cosine_similarity(tf_idf_matrix, target_index=0):
    """
    Compute cosine similarity of texts in a TF-IDF matrix.

    Args:
        tf_idf_matrix (sparse matrix): TF-IDF matrix.
        target_index (int): Index of the target text.

    Returns:
        np.ndarray: Cosine similarity scores.
    """
    return cosine_similarity(tf_idf_matrix[target_index:target_index + 1], tf_idf_matrix)[0]

def add_cosine_similarity_scores(df, cos_sim_scores):
    """
    Add cosine similarity scores to a DataFrame.

    Args:
        df (pd.DataFrame): DataFrame containing the original data.
        cos_sim_scores (np.ndarray): Cosine similarity scores.

    Returns:
        pd.DataFrame: DataFrame with cosine similarity scores added.
    """
    cos_sim_df = pd.DataFrame(cos_sim_scores, columns=['Cos_Sim'])
    return pd.concat([df, cos_sim_df], axis=1)

def sort_by_cosine_similarity(df):
    """
    Sort DataFrame by cosine similarity scores.

    Args:
        df (pd.DataFrame): DataFrame containing cosine similarity scores.

    Returns:
        pd.DataFrame: Sorted DataFrame.
    """
    return df.sort_values(by='Cos_Sim', ascending=False)

if __name__ == '__main__':
    # Load the article data from a CSV file
    df_articles = load_data('article_data.csv')

    # Extract all article texts
    texts = preprocess_texts(df_articles, 'content')

    # Compute the TF-IDF matrix
    tf_idf_matrix, vectorizer = compute_tfidf_matrix(texts, stop_words='english')

    # Compute cosine similarity scores
    cosine_similarities = compute_cosine_similarity(tf_idf_matrix, target_index=0)

    # Add cosine similarity scores to the DataFrame
    df_with_scores = add_cosine_similarity_scores(df_articles, cosine_similarities)

    # Sort the DataFrame by cosine similarity scores
    df_final_sorted = sort_by_cosine_similarity(df_with_scores)

    # Print the top 6 results
    print(df_final_sorted.head(6))
