import re
from nltk.stem import PorterStemmer
import os

def extract_data(filepath, stop_word_list):
    ps = PorterStemmer()
    total_doc = ""
    final_word_list = []
    doc_num_list = []

    with open(filepath, 'r') as file:
        line = file.readline()
        while line:
            striped_string = line.strip() + " "
            total_doc += striped_string
            if "<DOCNO>" in striped_string:
                doc_num = re.search(r'<DOCNO>(.*?)</DOCNO>', striped_string).group(1)
                doc_num_list.append(doc_num)
            line = file.readline()

    total_texts = re.findall(r'<TEXT>(.*?)</TEXT>', total_doc)
    for text in total_texts:
        cleaned_text = clean_text(text)
        word_list = remove_stopwords(cleaned_text, stop_word_list)
        stemmed_list = stem_words(word_list, ps)
        final_word_list.extend(stemmed_list)

    return final_word_list, doc_num_list

def clean_text(text):
    """Cleans the text by removing unwanted characters and numbers."""
    text = text.strip()
    text = re.sub(r"\w*-*\d+-*\w*", " ", text)
    text = re.sub(r"\d+", " ", text)
    text = re.sub(r"\W+", " ", text)
    return text.lower()

def remove_stopwords(text, stop_word_list):
    """Removes stop words from the text."""
    word_list = re.split(r'\s+', text)
    return [word for word in word_list if word not in stop_word_list]

def stem_words(word_list, stemmer):
    """Stems the words in the list."""
    return [stemmer.stem(word) for word in word_list if word]

def main():
    filepath = 'path_to_your_file'
    stop_word_list = ['your', 'stop', 'words', 'here']
    final_word_list, doc_num_list = extract_data(filepath, stop_word_list)
    print(f"Final Word List: {final_word_list}")
    print(f"Document Numbers: {doc_num_list}")

if __name__ == "__main__":
    main()
