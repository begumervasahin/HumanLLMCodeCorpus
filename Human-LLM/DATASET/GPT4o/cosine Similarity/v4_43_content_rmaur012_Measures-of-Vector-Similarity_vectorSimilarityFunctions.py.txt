from math import sqrt

def cosine_similarity(v, w):
    """
    Calculate the cosine similarity between two vectors v and w.
    
    Args:
    - v (list): First vector.
    - w (list): Second vector.
    
    Returns:
    - float: Cosine similarity between v and w.
    """
    numerator_sum = 0
    v_vector_sum = 0
    w_vector_sum = 0
    
    for index, value in enumerate(v):
        numerator_sum += value * w[index]
        v_vector_sum += value ** 2
        w_vector_sum += w[index] ** 2
    
    denominator_total = sqrt(v_vector_sum) * sqrt(w_vector_sum)
    
    if denominator_total == 0:
        return 0.0
    
    similarity = numerator_sum / denominator_total
    return similarity

def jaccard_similarity(v, w):
    """
    Calculate the Jaccard similarity between two sets v and w.
    
    Args:
    - v (list): First set (as a list).
    - w (list): Second set (as a list).
    
    Returns:
    - float: Jaccard similarity between v and w.
    """
    numerator_min_sum = 0
    denominator_max_sum = 0
    
    for index, value in enumerate(v):
        numerator_min_sum += min(value, w[index])
        denominator_max_sum += max(value, w[index])
    
    if denominator_max_sum == 0:
        return 0.0
    
    similarity = numerator_min_sum / denominator_max_sum
    return similarity

def dice_similarity(v, w):
    """
    Calculate the Dice similarity between two sets v and w.
    
    Args:
    - v (list): First set (as a list).
    - w (list): Second set (as a list).
    
    Returns:
    - float: Dice similarity between v and w.
    """
    numerator_min_sum = 0
    denominator_sum = 0
    
    for index, value in enumerate(v):
        numerator_min_sum += min(value, w[index])
        denominator_sum += value + w[index]
    
    if denominator_sum == 0:
        return 0.0
    
    similarity = (2 * numerator_min_sum) / denominator_sum
    return similarity

# Example usage:
v1 = [1, 2, 3, 4]
v2 = [2, 3, 4, 5]

# Calculate and print similarities
print(f"Cosine Similarity: {cosine_similarity(v1, v2)}")
print(f"Jaccard Similarity: {jaccard_similarity(v1, v2)}")
print(f"Dice Similarity: {dice_similarity(v1, v2)}")
