import fitz  # PyMuPDF
from storage import enumrateFilenames

def read_all_pdfs():
    """
    Read all PDF files from storage and return the text from each in a list along with the file names.

    Returns:
        list: A list of tuples, each containing a PDF name and the extracted text data from that PDF.
    """
    pdf_pages = []
    for pdf_filename in enumrateFilenames():
        pdf_pages.extend(read_pdf(pdf_filename))
    return pdf_pages

def read_pdf(pdf_filename):
    """
    Read a PDF file from storage and return the text from it in a list along with the file name.

    Args:
        pdf_filename (str): Path to the PDF file.

    Returns:
        list: A list of tuples, each containing the PDF name and the extracted text data from the PDF.
    """
    pdf_pages = []
    with fitz.open(pdf_filename) as pdf_file:
        for page in pdf_file:
            pdf_pages.append((pdf_filename, page.get_text()))
    return pdf_pages

def display_extracted_text(pdfs):
    """
    Display the extracted text from PDFs.

    Args:
        pdfs (list): List of tuples, each containing a PDF name and the extracted text data from the PDF.
    """
    for pdf, text in pdfs:
        print(f"PDF: {pdf}\nText: {text[:100]}...\n")  # Print first 100 characters of each page's text for brevity

if __name__ == "__main__":
    all_pdfs = read_all_pdfs()
    display_extracted_text(all_pdfs)
