import json
from config import SETTINGS
from TwitterAPI import TwitterAPI

# Define the file to store tweets
FILE_NAME = "tweets.json"

# Initialize TwitterAPI with credentials from SETTINGS
api = TwitterAPI(
    SETTINGS["consumer_key"], 
    SETTINGS["consumer_secret"],
    SETTINGS["access_token_key"], 
    SETTINGS["access_token_secret"]
)

def get_search_query(query):
    """
    Get the results of a search query from Twitter API.

    Args:
    - query (str): The search query string.

    Returns:
    - TwitterAPI.Response: The response from the Twitter API containing search results.
    """
    response = api.request('search/tweets', {'q': query})
    return response

def get_user_timeline(username):
    """
    Get tweets from a user's timeline.

    Args:
    - username (str): The Twitter username.

    Returns:
    - TwitterAPI.Response: The response from the Twitter API containing the user's timeline tweets.
    """
    response = api.request('statuses/user_timeline', {'screen_name': username, 'count': 200})
    return response

def save_tweets_to_file():
    """
    Fetch tweets from a specific user's timeline and save them to a JSON file.
    """
    with open(FILE_NAME, 'a+') as tweet_file:
        for tweet in get_user_timeline("hasgeek").get_iterator():
            json.dump(tweet, tweet_file, separators=(',', ':'))
            tweet_file.write('\n')
            tweet_file.flush()

if __name__ == "__main__":
    save_tweets_to_file()
