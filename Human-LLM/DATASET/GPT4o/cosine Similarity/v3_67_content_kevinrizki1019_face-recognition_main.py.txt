from tkinter import *

from pageone import PageOne  # Ensure PageOne class is properly defined in the pageone module

LARGE_FONT = ("Verdana", 16)  # Define a large font style for consistency

class MainApp(Tk):
    """Main application class for managing frames and switching between them."""
    
    def __init__(self, *args, **kwargs):
        """Initialize the main application window."""
        super().__init__(*args, **kwargs)
        self.title("KetokMagicHalal")

        # Create a container frame to hold all other frames
        container = Frame(self)
        container.pack(side="top", fill="both", expand=True)

        # Dictionary to keep track of frames
        self.frames = {}
        self.current_frame = None

        # Initialize and store frames
        self._initialize_frames(container)
        
        # Display the initial frame
        self.show_frame(PageOne)

    def _initialize_frames(self, container):
        """Initialize and store the frames in a dictionary."""
        for FrameClass in (PageOne,):  # Add more page classes to this tuple if needed
            frame = FrameClass(container, self)
            self.frames[FrameClass] = frame
            frame.pack_forget()

    def show_frame(self, frame_class):
        """Show a frame for the given class name."""
        if self.current_frame is not None:
            self.current_frame.pack_forget()
        
        self.current_frame = self.frames[frame_class]
        self.current_frame.pack(side=TOP, fill=BOTH, expand=True, padx=20, pady=20)
        self.current_frame.load_db(db_path)  # Assuming load_db is a method in PageOne
