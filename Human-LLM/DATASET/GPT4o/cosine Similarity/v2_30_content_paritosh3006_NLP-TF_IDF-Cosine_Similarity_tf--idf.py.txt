# Define two sample documents
docA = "I am going to Bangalore"
docB = "I am coming from Delhi"

# Split documents into words
bowA = docA.split(" ")
bowB = docB.split(" ")

# Create a set of unique words from both documents
wordSet = set(bowA).union(set(bowB))

# Initialize dictionaries to store word frequencies for each document
wordDictA = dict.fromkeys(wordSet, 0)
wordDictB = dict.fromkeys(wordSet, 0)

# Count word occurrences in each document
for word in bowA:
    wordDictA[word] += 1
for word in bowB:
    wordDictB[word] += 1

# Print word frequency dictionaries for each document
print("Word frequencies in docA:", wordDictA)
print("Word frequencies in docB:", wordDictB)

import pandas as pd

# Function to compute Term Frequency (TF) for a document
def computeTF(wordDict, bow):
    tfDict = {}
    bowCount = len(bow)
    for word, count in wordDict.items():
        tfDict[word] = count / float(bowCount)
    return tfDict

# Compute TF for each document
tfBowA = computeTF(wordDictA, bowA)
tfBowB = computeTF(wordDictB, bowB)

# Print TF for each document
print("TF for docA:", tfBowA)
print("TF for docB:", tfBowB)

# Function to compute Inverse Document Frequency (IDF) for a list of documents
def computeIDF(docList):
    import math
    idfDict = {}
    N = len(docList)
    
    # Initialize IDF dictionary with zeros based on the first document
    idfDict = dict.fromkeys(docList[0].keys(), 0)
    
    # Count documents containing each word
    for doc in docList:
        for word, val in doc.items():
            if val > 0:
                idfDict[word] += 1
    
    # Compute IDF values
    for word, val in idfDict.items():
        idfDict[word] = math.log(N / float(val))
    
    return idfDict

# Compute IDF for the documents
idfs = computeIDF([wordDictA, wordDictB])

# Print IDF values
print("IDF values:", idfs)

# Function to compute TF-IDF for a document
def computeTFIDF(tfBow, idfs):
    tfidf = {}
    for word, val in tfBow.items():
        tfidf[word] = val * idfs[word]
    return tfidf

# Compute TF-IDF for each document
tfidfBowA = computeTFIDF(tfBowA, idfs)
tfidfBowB = computeTFIDF(tfBowB, idfs)

# Create a Pandas DataFrame to display TF-IDF values
tfidf_df = pd.DataFrame([tfidfBowA, tfidfBowB])
print("TF-IDF values:")
print(tfidf_df)
