import re
from collections import Counter

def load_words(file_path):
    """
    Load words from a file and count their occurrences.

    Args:
        file_path (str): Path to the file.

    Returns:
        Counter: A dictionary-like object containing word counts.
    """
    with open(file_path, 'r') as file:
        text = file.read().lower()
    word_list = re.findall(r'\w+', text)
    return Counter(word_list)

def probability(word, N):
    """
    Calculate the probability of a word.

    Args:
        word (str): The word to calculate the probability for.
        N (int): Total number of words.

    Returns:
        float: Probability of the word.
    """
    return WORDS_DICT[word] / N

def correct(word):
    """
    Find the most probable spelling correction for a word.

    Args:
        word (str): The word to correct.

    Returns:
        str: The corrected word.
    """
    return max(candidates(word), key=probability)

def candidates(word):
    """
    Generate possible spelling corrections for a word.

    Args:
        word (str): The word to generate corrections for.

    Returns:
        set: A set of candidate words.
    """
    known_words = known([word])
    if known_words:
        return known_words

    known_words = known(edits1(word))
    if known_words:
        return known_words

    known_words = known(edits2(word))
    if known_words:
        return known_words

    return {word}

def known(words):
    """
    Return the subset of words that are actually in the dictionary.

    Args:
        words (iterable): An iterable of words.

    Returns:
        set: A set of known words.
    """
    return {w for w in words if w in WORDS_DICT}

def edits1(word):
    """
    Generate all edits that are one edit away from the input word.

    Args:
        word (str): The input word.

    Returns:
        set: A set of words that are one edit away.
    """
    letters = 'abcdefghijklmnopqrstuvwxyz'
    splits = [(word[:i], word[i:]) for i in range(len(word) + 1)]
    deletes = [L + R[1:] for L, R in splits if R]
    transposes = [L + R[1] + R[0] + R[2:] for L, R in splits if len(R) > 1]
    replaces = [L + c + R[1:] for L, R in splits if R for c in letters]
    inserts = [L + c + R for L, R in splits for c in letters]
    return set(deletes + transposes + replaces + inserts)

def edits2(word):
    """
    Generate all edits that are two edits away from the input word.

    Args:
        word (str): The input word.

    Returns:
        set: A set of words that are two edits away.
    """
    return {e2 for e1 in edits1(word) for e2 in edits1(e1)}
