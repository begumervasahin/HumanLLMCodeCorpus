import cv2
import numpy as np
import pickle
import os
import matplotlib.pyplot as plt
from scipy.misc import imread

def extract_features(image_path, vector_size=32):
    """
    Extract features from an image using the KAZE algorithm.

    Args:
        image_path (str): The path to the image file.
        vector_size (int): The number of key points to consider.

    Returns:
        np.ndarray: A flattened array of the image's features.
    """
    # Read the image in RGB mode
    image = imread(image_path, mode="RGB")
    
    try:
        # Initialize KAZE feature detector
        kaze = cv2.KAZE_create()
        
        # Detect key points in the image
        keypoints = kaze.detect(image)
        
        # Sort key points by response and keep the best ones
        keypoints = sorted(keypoints, key=lambda x: -x.response)[:vector_size]
        
        # Compute descriptors for the key points
        keypoints, descriptors = kaze.compute(image, keypoints)
        
        # Flatten the descriptors array
        descriptors = descriptors.flatten()
        
        # Ensure the descriptors array has the required size
        needed_size = vector_size * 64
        if descriptors.size < needed_size:
            descriptors = np.concatenate([descriptors, np.zeros(needed_size - descriptors.size)])
    
    except cv2.error as e:
        print(f'Error: {e}')
        return None
    
    return descriptors

def batch_extractor(images_path, pickled_db_path="features.pck"):
    """
    Extract features from a batch of images and save them in a pickle file.

    Args:
        images_path (str): The path to the directory containing images.
        pickled_db_path (str): The path to save the pickle file.

    Returns:
        None
    """
    # Get the list of image files in the directory
    image_files = [os.path.join(images_path, p) for p in sorted(os.listdir(images_path))]
    features = {}
    
    for image_file in image_files:
        print(f'Extracting features from image {image_file}')
        image_name = os.path.basename(image_file).lower()
        features[image_name] = extract_features(image_file)
    
    # Save the features dictionary to a pickle file
    with open(pickled_db_path, 'wb') as fp:
        pickle.dump(features, fp, protocol=pickle.HIGHEST_PROTOCOL)

if __name__ == '__main__':
    # Path to the directory containing images
    images_path = 'path/to/images'
    
    # Path to save the pickle file
    pickled_db_path = 'features.pck'
    
    # Extract features from the images and save them in a pickle file
    batch_extractor(images_path, pickled_db_path)
