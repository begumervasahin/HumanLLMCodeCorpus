import io
from pdfminer.pdfinterp import PDFResourceManager, PDFPageInterpreter
from pdfminer.converter import TextConverter
from pdfminer.layout import LAParams
from pdfminer.pdfpage import PDFPage

def pdf_to_text(file_path):
    """
    Extract text from a PDF file.

    Args:
        file_path (str): The path to the PDF file.

    Returns:
        str: The extracted text from the PDF.
    """
    # Initialize the PDF resource manager
    resource_manager = PDFResourceManager()
    
    # Create a StringIO object to capture the text output
    output_string = io.StringIO()
    
    # Set parameters for layout analysis
    laparams = LAParams()
    
    # Create a TextConverter object
    with TextConverter(resource_manager, output_string, laparams=laparams) as device:
        # Create a PDF interpreter object
        interpreter = PDFPageInterpreter(resource_manager, device)

        with open(file_path, 'rb') as file:
            # Process each page in the PDF file
            for page in PDFPage.get_pages(file, check_extractable=True):
                interpreter.process_page(page)

        # Get the extracted text from the StringIO object
        text = output_string.getvalue()

    # Return the extracted text
    return text

    pdf_path = 'Computer Graphics.pdf'  # Path to your PDF file
    extracted_text = pdf_to_text(pdf_path)
