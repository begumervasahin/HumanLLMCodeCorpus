from nltk.tokenize import word_tokenize
import gensim

def load_word2vec_model(model_path):
    """
    Load a pre-trained Word2Vec model from a file.

    Args:
    - model_path (str): Path to the Word2Vec model file.

    Returns:
    - gensim.models.Word2Vec: Loaded Word2Vec model.
    """
    model = gensim.models.Word2Vec.load(model_path, mmap='r')
    return model

def score_similarity(seq1, seq2, model, tail=100, head=10):
    """
    Calculate the average maximum similarity between two sequences of words using a Word2Vec model.

    Args:
    - seq1 (str): First sequence of words.
    - seq2 (str): Second sequence of words.
    - model (gensim.models.Word2Vec): Pre-trained Word2Vec model.
    - tail (int): Number of words to consider from the end of seq1 (default: 100).
    - head (int): Number of words to consider from the beginning of seq2 (default: 10).

    Returns:
    - float: Average maximum similarity score between seq1 and seq2.
    """
    seq1_word_list = word_tokenize(seq1.strip().lower())[-tail:]
    seq2_word_list = word_tokenize(seq2.strip().lower())[:head]
    return calculate_similarity_score(seq1_word_list, seq2_word_list, model)

def calculate_similarity_score(wordlist1, wordlist2, model):
    """
    Calculate the average maximum similarity score between two lists of words using a Word2Vec model.

    Args:
    - wordlist1 (list of str): List of words from the first sequence.
    - wordlist2 (list of str): List of words from the second sequence.
    - model (gensim.models.Word2Vec): Pre-trained Word2Vec model.

    Returns:
    - float: Average maximum similarity score between wordlist1 and wordlist2.
    """
    maxes = []
    for word in wordlist1:
        cur_max = 0
        for word2 in wordlist2:
            if word == word2:
                sim = 1
                cur_max = sim
            elif word in model.vocab and word2 in model.vocab:
                sim = model.similarity(word, word2)
                if sim > cur_max:
                    cur_max = sim
        if cur_max != 0:
            maxes.append(cur_max)
    if sum(maxes) == 0:
        return 0
    return float(sum(maxes)) / len(maxes)

# Example usage:
model_path = 'vectors'  # Update with your Word2Vec model path
word2vec_model = load_word2vec_model(model_path)

sequence1 = "This is a sample sequence for scoring similarity."
sequence2 = "Sample sequence scoring similarity using Word2Vec model."

similarity_score = score_similarity(sequence1, sequence2, word2vec_model)
print(f"Similarity Score: {similarity_score}")
