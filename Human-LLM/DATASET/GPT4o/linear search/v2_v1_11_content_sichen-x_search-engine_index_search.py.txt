from collections import defaultdict
from words import get_text, words  # Assuming these functions are implemented correctly

def create_index(files):
    """
    Given a list of fully-qualified filenames, build an index from word to set of document IDs.
    A document ID is just the index into the files parameter (indexed from 0) to get the file name.
    For each word w in file i, add i to the set of document IDs containing w.
    Return a dict object mapping a word to a set of doc IDs.
    """
    if not files:
        return None

    index = defaultdict(set)
    for doc_id, file_path in enumerate(files):
        file_content = get_text(file_path)
        key_words = words(file_content)
        for word in key_words:
            index[word].add(doc_id)
    return index

def index_search(files, index, terms):
    """
    Given an index and a list of fully-qualified filenames, return a list of doc IDs whose file contents
    contain all words in the terms parameter as normalized by the words() function.
    Parameter terms is a list of strings. You can only use the index to find matching files; you cannot open the files
    and look inside.
    """
    if not (files and index and terms):
        return None

    doc_id_sets = []
    for term in terms:
        doc_ids = index.get(term)
        if doc_ids:
            doc_id_sets.append(doc_ids)
        else:
            return None

    if doc_id_sets:
        matching_doc_ids = set.intersection(*doc_id_sets)
        return [files[doc_id] for doc_id in matching_doc_ids]
    else:
        return None

# Example usage:
if __name__ == "__main__":
    # Assuming you have some filenames in a list named `files`
    files = ["file1.txt", "file2.txt", "file3.txt"]

    # Create index
    index = create_index(files)

    # Search for terms
    search_terms = ["term1", "term2"]
    result = index_search(files, index, search_terms)
    print("Matching files:", result)
