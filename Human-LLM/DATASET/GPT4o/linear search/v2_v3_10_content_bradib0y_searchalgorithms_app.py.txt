import time
import math

def recursive_binary_search(lst, target):
    """
    Perform recursive binary search to find the target in the list.
    """
    if not lst:
        return False
    
    midpoint = len(lst) // 2
    if lst[midpoint] == target:
        return True
    elif lst[midpoint] < target:
        return recursive_binary_search(lst[midpoint + 1:], target)
    else:
        return recursive_binary_search(lst[:midpoint], target)

def binary_search(lst, target):
    """
    Perform binary search to find the target in the list.
    """
    first = 0
    last = len(lst) - 1
    
    while first <= last:
        midpoint = (first + last) // 2
        if lst[midpoint] == target:
            return midpoint
        elif lst[midpoint] < target:
            first = midpoint + 1
        else:
            last = midpoint - 1
            
    return None

def linear_search(lst, target):
    """
    Perform linear search to find the target in the list.
    """
    for i, num in enumerate(lst):
        if num == target:
            return i
    return None

def log_performance():
    """
    Test and log the performance of different search algorithms.
    """
    for i in range(10, 19):
        size = int(math.pow(10, i))
        numbers = list(range(size))
        target = size - 3
        
        print(f"\n-> Quantity: {size} (10 to the power of {i})")

        # Linear search
        print("Linear search: null")

        # Binary search
        start_time = time.time()
        binary_search(numbers, target)
        binary_search_time = time.time() - start_time
        print(f"Binary search: {binary_search_time:.6f} seconds")

        # Recursive binary search
        start_time = time.time()
        recursive_binary_search(numbers, target)
        recursive_search_time = time.time() - start_time
        print(f"Recursive binary search: {recursive_search_time:.6f} seconds")

        # Write results to file
        with open("data.csv", "a") as file:
            file.write(f"{size};null;{binary_search_time:.6f};{recursive_search_time:.6f}\n")

if __name__ == "__main__":
    log_performance()
