import sys
import webbrowser
from index_search import index_search, create_index
from linear_search import linear_search
from myhtable_search import myhtable_index_search, myhtable_create_index
from words import filelist, words, results

def main():
    # Validate command-line arguments
    if len(sys.argv) != 3:
        print_usage()
        sys.exit(1)

    # Extract search implementation and root directory from command-line arguments
    impl, rootdir = sys.argv[1], sys.argv[2]

    # Get the list of files in the specified directory
    files = filelist(rootdir)
    print(f"{len(files)} files")

    index = None

    # Main search loop
    while True:
        search_terms = input("Search terms: ")
        terms = words(search_terms)

        docs, index = perform_search(impl, files, terms, index)

        if docs is None:
            break

        display_results(docs, terms)

def print_usage():
    """Prints the usage information for the script."""
    print("Usage:")
    print("$ python search.py linear ~/data/slate")
    print("$ python search.py index ~/data/slate")
    print("$ python search.py myhtable ~/data/slate")

def perform_search(impl, files, terms, index):
    """Performs the search based on the specified implementation."""
    if impl == 'linear':
        return linear_search(files, terms), index
    elif impl == 'index':
        if index is None:
            index = create_index(files)
            print("Index complete")
        return index_search(files, index, terms), index
    elif impl == 'myhtable':
        if index is None:
            index = myhtable_create_index(files)
            print("Index complete")
        return myhtable_index_search(files, index, terms), index
    else:
        print("Invalid search type:", impl)
        return None, index

def display_results(docs, terms):
    """Generates and displays the search results in a browser."""
    search_results_page = results(docs, terms)
    results_file_path = "/tmp/results.html"
    
    with open(results_file_path, "w") as results_file:
        results_file.write(search_results_page)
    
    webbrowser.open_new_tab(f"file://{results_file_path}")

if __name__ == "__main__":
    main()
