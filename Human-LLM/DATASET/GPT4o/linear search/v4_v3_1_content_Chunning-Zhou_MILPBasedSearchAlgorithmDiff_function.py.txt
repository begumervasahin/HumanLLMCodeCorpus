def gen_filefolder(cipher_name, goal):
    """
    Generate required file folders based on cipher and goal.

    :param cipher_name: Name of the cipher
    :param goal: Goal for which the folder is being generated
    """
    folder_path = f"{cipher_name}_{goal}"
    if not os.path.exists(folder_path):
        os.makedirs(folder_path)
