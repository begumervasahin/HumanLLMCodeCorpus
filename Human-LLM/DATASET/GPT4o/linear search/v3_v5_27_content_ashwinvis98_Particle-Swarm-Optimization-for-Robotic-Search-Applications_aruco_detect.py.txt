import cv2
import cv2.aruco as aruco

def configure_camera_settings(cap):
    """
    Configure the camera settings.

    Parameters:
    cap (cv2.VideoCapture): The video capture object.
    """
    # Uncomment the lines below to adjust camera settings
    '''
    cap.set(cv2.CAP_PROP_AUTOFOCUS, 0)
    cap.set(cv2.CAP_PROP_FRAME_WIDTH, 720)  # Set width
    cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 720)  # Set height
    '''

def detect_and_draw_markers(frame, aruco_dict, parameters):
    """
    Detect and draw ArUco markers on the frame.

    Parameters:
    frame (ndarray): The grayscale image frame.
    aruco_dict (cv2.aruco.Dictionary): The ArUco dictionary.
    parameters (cv2.aruco.DetectorParameters): The detector parameters.

    Returns:
    ndarray: The frame with detected markers drawn.
    """
    # Detect ArUco markers in the grayscale frame
    corners, ids, _ = aruco.detectMarkers(frame, aruco_dict, parameters=parameters)

    # Draw detected markers on the grayscale frame
    frame_with_markers = aruco.drawDetectedMarkers(frame.copy(), corners, ids)

    return frame_with_markers

def main():
    """
    Main function to capture video and detect ArUco markers.
    """
    # Initialize the video capture device
    cap = cv2.VideoCapture(0)

    # Configure camera settings
    configure_camera_settings(cap)

    # Define the ArUco dictionary and parameters
    aruco_dict = aruco.Dictionary_get(aruco.DICT_6X6_250)
    parameters = aruco.DetectorParameters_create()

    while True:
        # Read a frame from the video capture device
        ret, frame = cap.read()

        if not ret:
            print("Failed to grab frame")
            break

        # Convert the frame to grayscale
        gray_frame = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

        # Detect and draw markers on the frame
        frame_with_markers = detect_and_draw_markers(gray_frame, aruco_dict, parameters)

        # Display the frame with detected markers
        cv2.imshow('Frame with ArUco Markers', frame_with_markers)

        # Check for user input to quit the application
        if cv2.waitKey(1) & 0xFF == ord('q'):
            break

    # Release the video capture device and close all windows
    cap.release()
    cv2.destroyAllWindows()

if __name__ == "__main__":
    main()
