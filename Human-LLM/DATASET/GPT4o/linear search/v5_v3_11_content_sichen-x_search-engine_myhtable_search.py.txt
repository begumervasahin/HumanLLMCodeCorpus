from words import get_text, words

def create_index(files):
    """
    Create an index mapping words to sets of document indexes using a hash table.
    """
    if not files:
        return None

    index_table = htable(4011)  # Creating a hash table with 4011 buckets

    for file_index, file_name in enumerate(files):
        # Extract words from the text content of each file
        words_in_file = words(get_text(file_name))

        # Add words and their corresponding document indexes to the hash table
        for word in words_in_file:
            existing_set = htable_get(index_table, word)
            if existing_set is None:
                existing_set = set()
            existing_set.add(file_index)
            htable_put(index_table, word, existing_set)

    return index_table

def search_index(files, index, terms):
    """
    Search for terms in the index and retrieve corresponding files.
    """
    if not (files and index and terms):
        return None

    term_sets = []

    # Retrieve sets of document indexes for each search term
    for term in terms:
        term_set = htable_get(index, term)
        if term_set is None:
            return None
        term_sets.append(term_set)

    # Find the intersection of all sets of document indexes
    intersection_ids = set.intersection(*term_sets)

    # Retrieve corresponding files
    matching_files = [files[file_index] for file_index in intersection_ids]

    return matching_files

# Example usage
if __name__ == "__main__":
    # Assuming you have some files in a list
    files = ["file1.txt", "file2.txt", "file3.txt"]

    # Creating the index
    index = create_index(files)

    # Searching for terms
    search_terms = ["term1", "term2", "term3"]
    result = search_index(files, index, search_terms)

    # Printing the result
    if result:
        print("Search Result:")
        for file in result:
            print(file)
    else:
        print("No results found.")
