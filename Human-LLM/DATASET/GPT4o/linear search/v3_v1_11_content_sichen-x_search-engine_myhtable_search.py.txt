# Importing the necessary functions from the htable module
from htable import htable, htable_put, htable_get

# Importing the necessary functions from the words module
from words import get_text, words

def myhtable_create_index(files):
    """
    Build an index from words to sets of document indices.
    This function creates an index using a hashtable with 4011 buckets.
    
    Parameters:
    files (list): List of file paths to be indexed.

    Returns:
    table (list): Hashtable representation of the index.
    """
    if not files:
        return None

    # Create a hashtable with 4011 buckets
    table = htable(4011)

    # Iterate over the files and populate the hashtable
    for i, file in enumerate(files):
        file_content = get_text(file)  # Get the text content of the file
        key_words = words(file_content)  # Split the content into words

        for word in key_words:
            existing_set = htable_get(table, word)  # Get the current set for the word
            if existing_set is None:
                htable_put(table, word, {i})  # Create a new set with the current file index
            else:
                existing_set.add(i)  # Add the current file index to the existing set
                htable_put(table, word, existing_set)

    return table

def myhtable_index_search(files, index, terms):
    """
    Search for files containing all the given terms using the hashtable index.
    
    Parameters:
    files (list): List of file paths.
    index (list): Hashtable index created by myhtable_create_index.
    terms (list): List of search terms.

    Returns:
    ret_file_list (list): List of files containing all the search terms.
    """
    if not files or not index or not terms:
        return None

    list_of_sets = []

    # Retrieve sets of document indices for each search term
    for term in terms:
        value = htable_get(index, term)
        if value:
            list_of_sets.append(value)
        else:
            return None

    if not list_of_sets:
        return None

    # Find the intersection of all sets to get the common document indices
    intersection_ids = set.intersection(*list_of_sets)
    ret_file_list = [files[id] for id in intersection_ids]

    return ret_file_list

# Example usage:
if __name__ == "__main__":
    # List of files to be indexed
    files = ["file1.txt", "file2.txt", "file3.txt"]

    # Creating the index
    index = myhtable_create_index(files)

    # Searching for terms
    search_terms = ["term1", "term2", "term3"]
    result = myhtable_index_search(files, index, search_terms)

    # Printing the result
    if result:
        print("Search Result:")
        for file in result:
            print(file)
    else:
        print("No results found.")
