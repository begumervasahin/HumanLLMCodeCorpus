def linear_search(files, terms):
    """
    Perform a linear search to find files containing all search terms.
    
    Parameters:
    files (list of str): A list of file paths.
    terms (list of str): A list of search terms.
    
    Returns:
    list of str: A list of filenames containing all search terms. 
    If either files or terms is None or empty, returns None.
    """
    if not files or not terms:
        return None

    result_files = []

    for file in files:
        file_content = get_text(file)
        if file_content:
            content_words = extract_words(file_content)
            if all(term in content_words for term in terms):
                result_files.append(file)

    return result_files
