class HashTable:
    def __init__(self, size):
        """Initialize the hash table with the given size."""
        self.size = size
        self.buckets = [[] for _ in range(size)]

    def __str__(self):
        """Return a string representation of the hash table."""
        pairs = [f"{key}:{value}" for bucket in self.buckets for key, value in bucket]
        return "{" + ", ".join(pairs) + "}"

            result += "\n"
        return result

    def hash_function(self, key):
        """Compute the hash value for the given key."""
        return hash(key) % self.size

    def put(self, key, value):
        """Insert or update the (key, value) pair in the hash table."""
        index = self.hash_function(key)
        for i, (existing_key, _) in enumerate(self.buckets[index]):
            if existing_key == key:
                self.buckets[index][i] = (key, value)  # Update existing key
                return
        self.buckets[index].append((key, value))  # Insert new key

def test_empty():
    """Test the hash table when it is empty."""
    table = HashTable(5)
    assert str(table) == "{}"
    assert table.buckets_str() == "0000 ->\n0001 ->\n0002 ->\n0003 ->\n0004 ->\n"

def test_single():
    """Test the hash table with a single (key, value) pair."""
    table = HashTable(5)
    table.put("parrt", 99)
    assert str(table) == "{parrt:99}"
    assert table.buckets_str() == "0000 ->\n0001 ->\n0002 ->\n0003 -> parrt:99\n0004 ->\n"

def test_a_few():
    """Test the hash table with a few (key, value) pairs."""
    table = HashTable(5)
    for i in range(1, 11):
        table.put(i, i)
    assert str(table) == "{5:5, 10:10, 1:1, 6:6, 2:2, 7:7, 3:3, 8:8, 4:4, 9:9}"
    assert table.buckets_str() == (
        "0000 -> 5:5, 10:10\n"
        "0001 -> 1:1, 6:6\n"
        "0002 -> 2:2, 7:7\n"
        "0003 -> 3:3, 8:8\n"
        "0004 -> 4:4, 9:9\n"
    )

def test_str_to_set():
    """Test the hash table with (key, value) pairs where values are sets."""
    table = HashTable(5)
    table.put("parrt", {2, 99, 3942})
    table.put("tombu", {6, 3, 1024, 99, 102342})
    assert str(table) == "{tombu:{1024, 99, 3, 102342, 6}, parrt:{2, 99, 3942}}"
    assert table.buckets_str() == (
        "0000 ->\n"
        "0001 -> tombu:{1024, 99, 3, 102342, 6}\n"
        "0002 ->\n"
        "0003 -> parrt:{2, 99, 3942}\n"
        "0004 ->\n"
    )

if __name__ == "__main__":
    test_empty()
    test_single()
    test_a_few()
    test_str_to_set()
    print("All tests passed!")
