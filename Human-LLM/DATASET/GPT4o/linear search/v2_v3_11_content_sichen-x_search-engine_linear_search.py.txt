import os

# Function to retrieve text content from a file
def get_text(file_path):
    """Reads and returns the text content from a given file."""
    try:
        with open(file_path, 'r') as file:
            return file.read()
    except Exception as e:
        print(f"Error reading file {file_path}: {e}")
        return None

# Function to extract words from text
def extract_words(text):
    """Splits the given text into words based on whitespace."""
    return text.split()

# Function to perform linear search for files containing all search terms
def linear_search(files, search_terms):
    """Searches for files containing all specified search terms.

    Args:
        files (list): List of file paths to search.
        search_terms (list): List of search terms to look for in the files.

    Returns:
        list: List of filenames containing all search terms.
    """
    # Validate inputs
    if not files or not search_terms:
        return None

    # List to store filenames that match the search criteria
    matching_files = []

    # Iterate through each file
    for file_path in files:
        # Retrieve text content from the file
        file_content = get_text(file_path)

        # Proceed if file content is successfully retrieved
        if file_content:
            # Extract words from the file content
            file_words = extract_words(file_content)

            # Check if all search terms are present in the file words
            if all(term in file_words for term in search_terms):
                matching_files.append(file_path)

    return matching_files

# Example usage
if __name__ == "__main__":
    # Example list of files
    files = ["file1.txt", "file2.txt", "file3.txt"]

    # Example list of search terms
    search_terms = ["term1", "term2", "term3"]

    # Perform linear search to find files containing all search terms
    result = linear_search(files, search_terms)

    # Print the result
    if result:
        print("Files containing all search terms:")
        for file_path in result:
            print(file_path)
    else:
        print("No files found containing all search terms.")
