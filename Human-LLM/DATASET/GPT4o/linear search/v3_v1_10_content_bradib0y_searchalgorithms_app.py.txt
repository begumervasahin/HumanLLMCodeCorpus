import time
import math

def recursive_binary_search(lst, target):
    """Recursively searches for the target in a sorted list."""
    if not lst:
        return False
    
    midpoint = len(lst) // 2
    if lst[midpoint] == target:
        return True
    elif lst[midpoint] < target:
        return recursive_binary_search(lst[midpoint + 1:], target)
    else:
        return recursive_binary_search(lst[:midpoint], target)

def binary_search(lst, target):
    """Performs an iterative binary search on a sorted list and returns the index of the target if found, else returns None."""
    first, last = 0, len(lst) - 1
    
    while first <= last:
        midpoint = (first + last) // 2
        if lst[midpoint] == target:
            return midpoint
        elif lst[midpoint] < target:
            first = midpoint + 1
        else:
            last = midpoint - 1
    return None

def linear_search(lst, target):
    """Performs a linear search on a list and returns the index of the target if found, else returns None."""
    for index, value in enumerate(lst):
        if value == target:
            return index
    return None

def log_search_results(numbers, target, i):
    """Logs the results of binary and recursive binary search."""
    # Linear search is not performed, so it's logged as null
    print("Linear search: null")
    data_string = f"{len(numbers)};null;"

    # Test binary search
    start_time = time.time()
    binary_search(numbers, target)
    binary_search_time = time.time() - start_time
    print(f"Binary search: {binary_search_time:.6f} seconds")
    data_string += f"{binary_search_time:.6f};"

    # Test recursive binary search
    start_time = time.time()
    recursive_binary_search(numbers, target)
    recursive_binary_search_time = time.time() - start_time
    print(f"Recursive binary search: {recursive_binary_search_time:.6f} seconds")
    data_string += f"{recursive_binary_search_time:.6f}"

    return data_string

def test_log():
    """Tests the search functions with increasingly large lists and logs the results."""
    for i in range(10, 19):
        numbers = list(range(int(math.pow(10, i))))
        target = int(math.pow(10, i)) - 3
        
        print(f"\n-> Quantity: {target + 3} (10 to the power of {i})")

        data_string = log_search_results(numbers, target, i)
        
        # Append the results to the CSV file
        with open("data.csv", "a") as file:
            file.write(data_string + "\n")

if __name__ == "__main__":
    test_log()
