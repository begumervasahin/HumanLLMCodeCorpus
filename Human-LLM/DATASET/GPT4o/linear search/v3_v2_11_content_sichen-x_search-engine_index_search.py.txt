from collections import defaultdict

def get_text(file_path):
    """
    Reads the content of a file and returns it as a string.
    
    Parameters:
        file_path (str): The path to the file.
        
    Returns:
        str: The content of the file.
    """
    with open(file_path, 'r', encoding='utf-8') as file:
        return file.read()

def words(text):
    """
    Splits the text into words and returns a list of words.
    
    Parameters:
        text (str): The text to split into words.
        
    Returns:
        list: A list of words.
    """
    return text.lower().split()

def create_index(files):
    """
    Builds an index from words to sets of document IDs.

    Parameters:
        files (list): List of fully-qualified filenames.

    Returns:
        defaultdict: Mapping from word to set of document IDs.
    """
    index = defaultdict(set)
    
    for doc_id, file_path in enumerate(files):
        file_content = get_text(file_path)
        key_words = words(file_content)
        
        for word in key_words:
            index[word].add(doc_id)
    
    return index

def index_search(files, index, terms):
    """
    Searches for documents containing all terms.

    Parameters:
        files (list): List of fully-qualified filenames.
        index (defaultdict): Mapping from word to set of document IDs.
        terms (list): List of strings to search for.

    Returns:
        list: Matching filenames.
    """
    if not (files and index and terms):
        return None

    # Gather sets of document IDs for each term
    document_sets = [index.get(term) for term in terms if term in index]
    
    # If any term is not found, return None
    if len(document_sets) != len(terms):
        return None

    # Find intersection of all sets
    matching_ids = set.intersection(*document_sets)
    
    # Return the filenames corresponding to the matching IDs
    return [files[doc_id] for doc_id in matching_ids] if matching_ids else None

# Example usage:
if __name__ == "__main__":
    # Example filenames
    files = ["file1.txt", "file2.txt", "file3.txt"]

    # Create the index
    index = create_index(files)

    # Search for terms
    search_terms = ["term1", "term2"]
    result = index_search(files, index, search_terms)
    print("Matching files:", result)
