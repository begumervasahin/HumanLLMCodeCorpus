import sys
import webbrowser
from index_search import index_search, create_index
from linear_search import linear_search
from myhtable_search import myhtable_index_search, myhtable_create_index
from words import filelist, words, results

# Check the command-line arguments
if len(sys.argv) != 3:
    print("Usage:")
    print("$ python search.py linear ~/data/slate")
    print("$ python search.py index ~/data/slate")
    print("$ python search.py myhtable ~/data/slate")
    sys.exit(1)

# Extract search implementation and root directory from command-line arguments
impl = sys.argv[1]
rootdir = sys.argv[2]

# Get the list of files in the specified directory
files = filelist(rootdir)
N = len(files)
print(N, "files")

index = None

# Start an infinite loop to accept search terms
while True:
    terms = input("Search terms: ")  # Input search terms from the user
    terms = words(terms)  # Tokenize the search terms

    # Perform search based on the specified implementation
    if impl == 'linear':
        docs = linear_search(files, terms)
    elif impl == 'index':
        if index is None:
            index = create_index(files)
            print("Index complete")
        docs = index_search(files, index, terms)
    elif impl == 'myhtable':
        if index is None:
            index = myhtable_create_index(files)
            print("Index complete")
        docs = myhtable_index_search(files, index, terms)
    else:
        print("Invalid search type:", impl)
        break

    # Generate search results page
    page = results(docs, terms)

    # Write search results to a temporary HTML file
    with open("/tmp/results.html", "w") as f:
        f.write(page)

    # Open the search results in a new browser tab
    webbrowser.open_new_tab("file:///tmp/results.html")
