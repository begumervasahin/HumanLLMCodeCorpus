class HashTable:
    def __init__(self, size):
        """
        Initialize the hash table with a specified size.
        Each bucket is initialized as an empty list.
        """
        self.size = size
        self.buckets = [[] for _ in range(size)]

    def __str__(self):
        """
        Return a string representation of the hash table.
        Displays key-value pairs in the format {key1:value1, key2:value2, ...}.
        """
        pairs = [
            f"{key}:{value}"
            for bucket in self.buckets
            for key, value in bucket
        ]
        return "{" + ", ".join(pairs) + "}"

    def buckets_str(self):
        """
        Return a string representation of the hash table's buckets.
        Displays the bucket index and its contents.
        """
        result = ""
        for i, bucket in enumerate(self.buckets):
            result += f"{str(i).zfill(4)}->"
            if bucket:
                pairs = ", ".join([f"{key}:{value}" for key, value in bucket])
                result += pairs
            result += "\n"
        return result

    def hash_function(self, key):
        """
        Compute the hash value for a given key.
        """
        return hash(key) % self.size

    def put(self, key, value):
        """
        Insert a key-value pair into the hash table.
        If the key already exists, update its value.
        """
        index = self.hash_function(key)
        for i, (existing_key, _) in enumerate(self.buckets[index]):
            if existing_key == key:
                self.buckets[index][i] = (key, value)
                return
        self.buckets[index].append((key, value))


def test_empty():
    """
    Test the hash table when it is empty.
    """
    table = HashTable(5)
    assert str(table) == "{}"
    assert table.buckets_str() == """0000->
0001->
0002->
0003->
0004->
"""


def test_single():
    """
    Test the hash table with a single key-value pair.
    """
    table = HashTable(5)
    table.put("parrt", 99)
    assert str(table) == "{parrt:99}"
    assert table.buckets_str() == """0000->
0001->
0002->
0003->parrt:99
0004->
"""


def test_a_few():
    """
    Test the hash table with multiple key-value pairs.
    """
    table = HashTable(5)
    for i in range(1, 11):
        table.put(i, i)
    s = str(table)
    assert s == "{5:5, 10:10, 1:1, 6:6, 2:2, 7:7, 3:3, 8:8, 4:4, 9:9}"
    s = table.buckets_str()
    assert s == """0000->5:5, 10:10
0001->1:1, 6:6
0002->2:2, 7:7
0003->3:3, 8:8
0004->4:4, 9:9
"""


def test_str_to_set():
    """
    Test the hash table with key-value pairs where values are sets.
    """
    table = HashTable(5)
    table.put("parrt", {2, 99, 3942})
    table.put("tombu", {6, 3, 1024, 99, 102342})
    assert str(table) == "{tombu:{1024, 99, 3, 102342, 6}, parrt:{2, 99, 3942}}"
    assert table.buckets_str() == """0000->
0001->tombu:{1024, 99, 3, 102342, 6}
0002->
0003->parrt:{2, 99, 3942}
0004->
"""


if __name__ == "__main__":
    # Run the test functions to ensure the hash table works as expected
    test_empty()
    test_single()
    test_a_few()
    test_str_to_set()
    print("All tests passed!")
