import time
import math

# Recursive binary search function
def recursive_binary_search(arr, target):
    if not arr:
        return False
    
    midpoint = len(arr) // 2
    
    if arr[midpoint] == target:
        return True
    elif arr[midpoint] < target:
        return recursive_binary_search(arr[midpoint + 1:], target)
    else:
        return recursive_binary_search(arr[:midpoint], target)

# Iterative binary search function
def binary_search(arr, target):
    """
    Return the index position of the target if found, else returns None
    """
    first, last = 0, len(arr) - 1
    
    while first <= last:
        midpoint = (first + last) // 2
        
        if arr[midpoint] == target:
            return midpoint
        elif arr[midpoint] < target:
            first = midpoint + 1
        else:
            last = midpoint - 1
    
    return None

# Linear search function
def linear_search(arr, target):
    """
    Return the index position of the target if found, else returns None
    """
    for index, value in enumerate(arr):
        if value == target:
            return index
    
    return None

# Function to test and log search algorithms
def test_log():
    # Open the file in append mode
    with open("data.csv", "a") as file:
        # Iterate over a range of exponents for practical execution
        for exponent in range(1, 8):
            numbers = list(range(0, int(math.pow(10, exponent))))
            target = int(math.pow(10, exponent)) - 3
            data_string = f"{len(numbers)};"
            
            # Display the current test details
            print(f"\n-> Quantity: {target + 3} (10 to the power of {exponent})")
            print("Linear search: null")
            data_string += "null;"

            # Measure the time taken for iterative binary search
            start_time = time.time()
            binary_search(numbers, target)
            elapsed_time = time.time() - start_time
            print(f"Binary search: {elapsed_time}")
            data_string += f"{elapsed_time};"

            # Measure the time taken for recursive binary search
            start_time = time.time()
            recursive_binary_search(numbers, target)
            elapsed_time = time.time() - start_time
            print(f"Recursive binary search: {elapsed_time}")
            data_string += f"{elapsed_time}"

            # Log the data to the CSV file
            file.write(data_string + "\n")

# Run the test and logging function
test_log()
