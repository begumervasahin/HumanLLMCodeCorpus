def generate_file_folder(cipher_name, goal):
    """
    Generate the required file folders based on the cipher name and goal.

    Args:
        cipher_name (str): Name of the cipher.
        goal (str): Goal for which the folders are generated.
    """
    folder_name = f"{cipher_name}_{goal}"
    if not os.path.exists(folder_name):
        os.makedirs(folder_name)
        print(f"Folder '{folder_name}' created.")
    else:
        print(f"Folder '{folder_name}' already exists.")


def remove_file(filename):
    """
    Remove a file if it exists.

    Args:
        filename (str): Name of the file to be removed.
    """
    if os.path.exists(filename):
        os.remove(filename)
        print(f"File '{filename}' removed.")
    else:
        print(f"File '{filename}' does not exist.")
