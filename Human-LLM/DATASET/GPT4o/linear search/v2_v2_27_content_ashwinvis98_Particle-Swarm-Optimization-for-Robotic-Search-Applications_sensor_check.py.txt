import cv2
import cv2.aruco as aruco
from nxt.motor import *
from nxt.sensor import *
import tty
import sys
import termios
import nxt.locator

def getch():
    """
    Get a single character from the keyboard input.
    
    Returns:
        str: A single character from the keyboard input.
    """
    fd = sys.stdin.fileno()
    old_settings = termios.tcgetattr(fd)
    try:
        tty.setraw(fd)
        ch = sys.stdin.read(1)
    finally:
        termios.tcsetattr(fd, termios.TCSADRAIN, old_settings)
    return ch

def initialize_brick():
    """
    Connect to the LEGO Mindstorms brick.
    
    Returns:
        Brick: Connected LEGO Mindstorms brick.
    """
    return nxt.locator.find_one_brick(host='00:16:53:0F:0F:D8')

def initialize_motors_and_sensors(brick):
    """
    Initialize motors and sensors.
    
    Args:
        brick (Brick): Connected LEGO Mindstorms brick.
    
    Returns:
        tuple: Initialized left motor, right motor, light sensor, synchronized motors for moving forward/backward and turning.
    """
    left_motor = Motor(brick, PORT_C)
    right_motor = Motor(brick, PORT_A)
    light_sensor = Light(brick, PORT_1, False)
    both_motors = SynchronizedMotors(left_motor, right_motor, 0)
    right_both_motors = SynchronizedMotors(left_motor, right_motor, 50)
    left_both_motors = SynchronizedMotors(right_motor, left_motor, 50)
    
    return left_motor, right_motor, light_sensor, both_motors, right_both_motors, left_both_motors

def control_robot():
    """
    Main function to control the LEGO Mindstorms robot.
    """
    # Connect to the LEGO Mindstorms brick
    brick = initialize_brick()

    # Initialize motors and sensors
    left_motor, right_motor, light_sensor, both_motors, right_both_motors, left_both_motors = initialize_motors_and_sensors(brick)

    print("Ready")
    print(brick)

    # Main control loop
    user_input = ' '
    while user_input != 'q':
        user_input = getch()  # Get user input

        # Read light sensor value
        light_value = light_sensor.get_input_values()[0]
        print("Light value:", light_value)

        # Perform actions based on user input
        if user_input == 'w':
            both_motors.turn(75, 360, True)  # Move forward
        elif user_input == 's':
            both_motors.turn(-75, 90, True)  # Move backward
        elif user_input == 'a':
            left_both_motors.turn(75, 90, True)  # Turn left
        elif user_input == 'd':
            right_both_motors.turn(75, 90, True)  # Turn right

    print("Aborted!")

if __name__ == "__main__":
    control_robot()
