# Assuming the implementation of get_text and words functions are available

def get_text(file_path):
    """
    Reads and returns the content of the specified file.

    Args:
        file_path (str): The path to the file.

    Returns:
        str: The content of the file.
    """
    with open(file_path, 'r') as file:
        return file.read()

def words(text):
    """
    Extracts and returns a list of words from the given text.

    Args:
        text (str): The input text.

    Returns:
        list: A list of words extracted from the text.
    """
    return text.split()

def linear_search(files, terms):
    """
    Searches for files containing all the specified search terms.

    Args:
        files (list): A list of file paths to search within.
        terms (list): A list of search terms to look for.

    Returns:
        list: A list of file paths that contain all the search terms.
    """
    # Return None if either files or terms are None or empty
    if not files or not terms:
        return None

    matching_files = []

    for file_path in files:
        # Retrieve the text content of the file
        file_content = get_text(file_path)

        if file_content:
            # Extract words from the file content
            file_words = words(file_content)
            # Check if all search terms are present in the file words
            if all(term in file_words for term in terms):
                matching_files.append(file_path)

    return matching_files

if __name__ == "__main__":
    # Example list of files
    files = ["file1.txt", "file2.txt", "file3.txt"]

    # Example list of search terms
    search_terms = ["term1", "term2", "term3"]

    # Perform linear search to find files containing all search terms
    result = linear_search(files, search_terms)

    # Print the result
    print("Files containing all search terms:")
    if result:
        for file in result:
            print(file)
    else:
        print("No files contain all search terms.")
