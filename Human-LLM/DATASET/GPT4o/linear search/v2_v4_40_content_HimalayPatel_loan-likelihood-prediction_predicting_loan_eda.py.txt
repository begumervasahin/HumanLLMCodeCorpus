# Importing necessary libraries
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

# Load the dataset
dataset = pd.read_csv("C:\\Users\\himal\\Desktop\\Machine Learning Practicals\\P5- Predicting the likelihood of e-signing a loan based on Financial History\\P39-Financial-Data.csv") 

# Remove unnecessary columns
columns_to_drop = ['entry_id', 'pay_schedule', 'e_signed']
dataset_cleaned = dataset.drop(columns=columns_to_drop)

# Plot histograms for numerical columns
fig, axes = plt.subplots(nrows=6, ncols=3, figsize=(15, 12))
fig.suptitle('Histograms of Numerical Columns', fontsize=20)

for i, column in enumerate(dataset_cleaned.columns):
    ax = axes[i // 3, i % 3]
    unique_values_count = min(np.size(dataset_cleaned[column].unique()), 100)
    ax.hist(dataset_cleaned[column], bins=unique_values_count, color='blue')
    ax.set_title(column)

plt.tight_layout(rect=[0, 0.03, 1, 0.95])
plt.show()

# Plot correlation with the target variable (e_signed)
correlation_with_target = dataset_cleaned.corrwith(dataset['e_signed'])
correlation_with_target.plot.bar(figsize=(20, 10), title="Correlation with e_signed", fontsize=15, rot=45, grid=True)
plt.tight_layout()
plt.show()

# Create a correlation matrix heatmap
sns.set(style="white")
correlation_matrix = dataset_cleaned.corr()

# Create a mask for the upper triangle
mask = np.triu(np.ones_like(correlation_matrix, dtype=bool))

# Set up the matplotlib figure
fig, ax = plt.subplots(figsize=(18, 15))

# Generate a custom diverging colormap
color_map = sns.diverging_palette(220, 10, as_cmap=True)

# Draw the heatmap with the mask and correct aspect ratio
sns.heatmap(correlation_matrix, mask=mask, cmap=color_map, vmax=0.4, center=0,
            square=True, linewidths=.5, cbar_kws={"shrink": .5})

plt.title("Correlation Matrix", fontsize=20)
plt.show()
