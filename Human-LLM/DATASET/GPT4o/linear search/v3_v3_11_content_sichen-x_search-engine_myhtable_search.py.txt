# Import necessary functions from the words and htable modules
from words import get_text, words
from htable import htable, htable_put, htable_get

def create_index(files):
    """
    Create an index mapping words to sets of document indexes using a hash table.
    
    Args:
        files (list): List of file names to index.
        
    Returns:
        dict: Hash table with words as keys and sets of document indexes as values.
    """
    if not files:
        return None

    # Initialize the hash table with a prime number of buckets for better distribution
    index_table = htable(4011)

    for file_index, file_name in enumerate(files):
        # Extract words from the text content of each file
        words_in_file = words(get_text(file_name))

        # Add words and their corresponding document indexes to the hash table
        for word in words_in_file:
            current_set = htable_get(index_table, word)
            if current_set is None:
                htable_put(index_table, word, {file_index})
            else:
                current_set.add(file_index)
                htable_put(index_table, word, current_set)

    return index_table

def search_index(files, index, terms):
    """
    Search for terms in the index and retrieve corresponding files.
    
    Args:
        files (list): List of file names.
        index (htable): Hash table created by create_index.
        terms (list): List of terms to search for.
        
    Returns:
        list: List of file names that contain all search terms.
    """
    if not (files and index and terms):
        return None

    term_sets = []

    # Collect sets of document indexes for each search term
    for term in terms:
        term_set = htable_get(index, term)
        if term_set is None:
            return []

        term_sets.append(term_set)

    # Find the intersection of all sets of document indexes
    intersection_ids = set.intersection(*term_sets) if term_sets else set()

    # Retrieve corresponding file names
    matching_files = [files[file_index] for file_index in intersection_ids]

    return matching_files

# Example usage
if __name__ == "__main__":
    # List of files to index
    files = ["file1.txt", "file2.txt", "file3.txt"]

    # Create the index
    index = create_index(files)

    # Terms to search for
    search_terms = ["term1", "term2", "term3"]

    # Search the index for the terms
    result = search_index(files, index, search_terms)

    # Print the search results
    if result:
        print("Search Result:")
        for file in result:
            print(file)
    else:
        print("No results found.")
