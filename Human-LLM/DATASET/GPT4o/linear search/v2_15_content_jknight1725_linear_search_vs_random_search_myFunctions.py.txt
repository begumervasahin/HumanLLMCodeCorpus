import random
from time import process_time

def number_prompt(lower_bound, upper_bound):
    """
    Prompt the user to enter a number within the specified bounds.
    """
    num = -1
    while num < lower_bound or num > upper_bound:
        num = int(input(f"Enter a number between {lower_bound} and {upper_bound}: "))
    return num

def compare_prompt():
    """
    Provide the comparison prompt for user input.
    """
    return "Type 'l' if larger, 's' if smaller, 'c' if correct: "

def computer_says(cpu_guess):
    """
    Print the computer's guess.
    """
    print(f'Computer says: {cpu_guess}')

def mid(lower, upper):
    """
    Calculate the midpoint of the current range.
    """
    return (upper + lower) // 2

def get_adjustment():
    """
    Get the user's adjustment input ('l', 's', or 'c').
    """
    adjustment = '0'
    while adjustment not in 'lsc':
        adjustment = input(compare_prompt())[0]
    return adjustment

def make_adjustment(adjustment, cpu_guess, lower_bound, upper_bound):
    """
    Adjust the bounds based on the user's feedback.
    """
    if adjustment == 'l':
        return [cpu_guess, upper_bound]
    elif adjustment == 's':
        return [lower_bound, cpu_guess]
    elif adjustment == 'c':
        return [lower_bound, upper_bound]
    else:
        return "Error\n"

def found(cpu_guess, guesses):
    """
    Announce that the computer has found the correct number.
    """
    print(f'Done! Your number is {cpu_guess}. Guessed in {guesses} times!')

def time_efficiency(func, *args):
    """
    Measure the time efficiency of a given function.
    """
    start_time = process_time()
    func(*args)
    end_time = process_time()
    return {
        'start': start_time,
        'end': end_time,
        'total': end_time - start_time
    }

def print_time_report(time):
    """
    Print the time report for the function execution.
    """
    print(f"Starts at: {time['start']}\nEnds at: {time['end']}\nTime taken to execute the function: {time['total']}\n")

def sum_up(large_num):
    """
    Calculate the sum of all numbers from 1 to large_num.
    """
    return int((large_num * (large_num + 1)) / 2)

def get_three_random_numbers():
    """
    Generate three random numbers between 0 and 9.
    """
    x = random.randrange(10)
    y = random.randrange(10)
    z = random.randrange(10)
    return [x, y, z]

def prompt_for_num():
    """
    Prompt the user to enter a number.
    """
    return int(input("Enter a number: "))

def get_user_guesses():
    """
    Get three guesses from the user.
    """
    x = prompt_for_num()
    y = prompt_for_num()
    z = prompt_for_num()
    return [x, y, z]

def check_for_win(user_numbers, rand_numbers):
    """
    Check if the user's guesses match the random numbers.
    """
    return user_numbers == rand_numbers

def all_1000_options():
    """
    Generate all possible three-digit combinations.
    """
    options = []
    for x in range(0, 1000):
        x_str = str(x).zfill(3)
        options.append(x_str)
    return options

def worst_case_algorithm(options, three_random_numbers):
    """
    Simulate the worst-case scenario algorithm for guessing the numbers.
    """
    idx = 0
    guesses = 1
    while [int(i) for i in str(options[idx])] != three_random_numbers:
        idx += 1
        guesses += 1
    return guesses

def random_case_algorithm(three_random_numbers):
    """
    Simulate a random guessing algorithm with an upper constraint on the number of guesses.
    """
    upper_constraint = 5000
    guesses = 1
    while get_three_random_numbers() != three_random_numbers and guesses < upper_constraint:
        guesses += 1
    return guesses
