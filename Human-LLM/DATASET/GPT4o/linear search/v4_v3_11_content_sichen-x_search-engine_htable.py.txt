class Hashtable:
    def __init__(self, nbuckets):
        """Initialize a hashtable with a specified number of buckets."""
        self.buckets = [[] for _ in range(nbuckets)]

    def _hash(self, key):
        """Compute the hash code for a given key.

        Args:
            key (int or str): The key to hash.

        Returns:
            int: The hash code of the key.
        """
        if isinstance(key, int):
            return key
        elif isinstance(key, str):
            hash_val = 0
            for char in key:
                hash_val = hash_val * 31 + ord(char)
            return hash_val
        else:
            raise TypeError("Keys must be either int or str")

    def put(self, key, value):
        """Add or update a key-value pair in the hashtable.

        Args:
            key (int or str): The key for the value.
            value: The value to be stored.
        """
        index = self._hash(key) % len(self.buckets)
        bucket = self.buckets[index]

        for i, (k, v) in enumerate(bucket):
            if k == key:
                bucket[i] = (key, value)  # Update existing key
                return

        bucket.append((key, value))  # Add new key-value pair

    def get(self, key):
        """Retrieve the value associated with the given key.

        Args:
            key (int or str): The key to look for.

        Returns:
            The value associated with the key, or None if the key is not found.
        """
        index = self._hash(key) % len(self.buckets)
        bucket = self.buckets[index]

        for k, v in bucket:
            if k == key:
                return v
        return None

    def __str__(self):
        """Return a string representation of the hashtable."""
        pairs = [f"{k}:{v}" for bucket in self.buckets for k, v in bucket]
        return "{" + ", ".join(pairs) + "}"

    def __repr__(self):
        """Return a string representation of the hashtable (for debugging)."""
        return str(self)

    def buckets_str(self):
        """Return a string representing the internal structure of the hashtable."""
        output = ""
        for i, bucket in enumerate(self.buckets):
            pairs = [f"{k}:{v}" for k, v in bucket]
            output += f"{str(i).zfill(4)}->" + ", ".join(pairs) + "\n"
        return output


# Example usage:
ht = Hashtable(5)
ht.put("apple", 10)
ht.put("banana", 20)
ht.put("apple", 5)

print("Hashtable:", ht)
print("Buckets:")
print(ht.buckets_str())
print("Get 'apple':", ht.get("apple"))
print("Get 'banana':", ht.get("banana"))
print("Get 'orange':", ht.get("orange"))
