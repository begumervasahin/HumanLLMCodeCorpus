# Importing necessary functions from the words module
from words import get_text, words
from htable import htable, htable_put, htable_get

# Function to create an index using a hash table
def myhtable_create_index(files):
    """
    Create an index mapping words to sets of document indexes using a hash table.
    """
    if not files:
        return None

    index_table = htable(4011)  # Creating a hash table with 4011 buckets

    for file_index, file_name in enumerate(files):
        # Extracting words from the text content of each file
        words_in_file = words(get_text(file_name))

        # Adding words and their corresponding document indexes to the hash table
        for word in words_in_file:
            current_set = htable_get(index_table, word)
            if current_set is None:
                htable_put(index_table, word, set([file_index]))
            else:
                current_set.add(file_index)
                htable_put(index_table, word, current_set)

    return index_table

# Function to search for terms in the index
def myhtable_index_search(files, index, terms):
    """
    Search for terms in the index and retrieve corresponding files.
    """
    if not (files and index and terms):
        return None

    term_sets = []

    # Iterating through the search terms
    for term in terms:
        term_set = htable_get(index, term)

        if term_set is None:
            return []

        term_sets.append(term_set)

    # Find the intersection of all sets of document indexes
    intersection_ids = set.intersection(*term_sets) if term_sets else set()

    # Retrieve corresponding files
    matching_files = [files[file_index] for file_index in intersection_ids]

    return matching_files

# Example usage:
if __name__ == "__main__":
    # Assuming you have some files in a list
    files = ["file1.txt", "file2.txt", "file3.txt"]

    # Creating the index
    index = myhtable_create_index(files)

    # Searching for terms
    search_terms = ["term1", "term2", "term3"]
    result = myhtable_index_search(files, index, search_terms)

    # Printing the result
    if result:
        print("Search Result:")
        for file in result:
            print(file)
    else:
        print("No results found.")
