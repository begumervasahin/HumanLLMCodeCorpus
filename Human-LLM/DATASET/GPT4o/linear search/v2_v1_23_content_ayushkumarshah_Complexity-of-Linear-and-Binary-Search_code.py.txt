import random
from time import time

# Define the search and utility functions
def linearSearch(theValues, target):
    """Perform a linear search in an unsorted list."""
    for value in theValues:
        if value == target:
            return True
    return False

def sortedLinearSearch(theValues, target):
    """Perform a linear search in a sorted list."""
    for value in theValues:
        if value == target:
            return True
        elif value > target:
            return False
    return False

def findSmallest(theValues):
    """Find the smallest element in a list."""
    smallest = theValues[0]
    for value in theValues[1:]:
        if value < smallest:
            smallest = value
    return smallest

def binarySearch(theValues, target):
    """Perform a binary search in a sorted list."""
    low, high = 0, len(theValues) - 1
    while low <= high:
        mid = (high + low) // 2
        if theValues[mid] == target:
            return True
        elif target < theValues[mid]:
            high = mid - 1
        else:
            low = mid + 1
    return False

# Generate random numbers and prepare test data
random.seed(42)
random_numbers = random.sample(range(1000000), 100000)
sorted_random_numbers = sorted(random_numbers)
target_number = random.choice(sorted_random_numbers)

# Lists to store elapsed times for each search method
elapsed_unsorted = []
elapsed_sorted = []
elapsed_smallest = []
elapsed_binary = []

# Measure the time taken for linear search in unsorted lists
for size in range(10000, 100001, 10000):
    start_time = time()
    linearSearch(random_numbers[:size], target_number)
    end_time = time()
    elapsed_unsorted.append(end_time - start_time)

# Print unsorted linear search times
print("\nUnsorted Linear Search times:")
for i in range(10):
    print(elapsed_unsorted[i])

# Measure the time taken for linear search in sorted lists
for size in range(10000, 100001, 10000):
    start_time = time()
    sortedLinearSearch(sorted_random_numbers[:size], target_number)
    end_time = time()
    elapsed_sorted.append(end_time - start_time)

# Print sorted linear search times
print("\nSorted Linear Search times:")
for i in range(10):
    print(elapsed_sorted[i])

# Measure the time taken to find the smallest element in lists
for size in range(10000, 100001, 10000):
    start_time = time()
    findSmallest(random_numbers[:size])
    end_time = time()
    elapsed_smallest.append(end_time - start_time)

# Print times for finding the smallest element
print("\nFinding Smallest element times:")
for i in range(10):
    print(elapsed_smallest[i])

# Measure the time taken for binary search in sorted lists
for size in range(10000, 100001, 10000):
    start_time = time()
    binarySearch(sorted_random_numbers[:size], target_number)
    end_time = time()
    elapsed_binary.append(end_time - start_time)

# Print binary search times
print("\nBinary Search times:")
for i in range(10):
    print(elapsed_binary[i])
