import sys
import webbrowser
from index_search import index_search, create_index
from linear_search import linear_search
from myhtable_search import myhtable_index_search, myhtable_create_index
from words import filelist, words, results

def main():
    # Check for correct number of command-line arguments
    if len(sys.argv) != 3:
        print("Usage:")
        print("$ python search.py linear ~/data/slate")
        print("$ python search.py index ~/data/slate")
        print("$ python search.py myhtable ~/data/slate")
        sys.exit(1)

    # Extract the search implementation type and root directory from arguments
    impl = sys.argv[1]
    rootdir = sys.argv[2]

    # Get the list of files in the specified directory
    files = filelist(rootdir)
    num_files = len(files)
    print(f"{num_files} files")

    index = None

    # Start an infinite loop to accept and process search terms
    while True:
        # Prompt user for search terms and tokenize them
        search_terms = input("Search terms: ")
        terms = words(search_terms)

        # Select the search implementation based on user input
        if impl == 'linear':
            docs = linear_search(files, terms)
        elif impl == 'index':
            if index is None:
                index = create_index(files)
                print("Index complete")
            docs = index_search(files, index, terms)
        elif impl == 'myhtable':
            if index is None:
                index = myhtable_create_index(files)
                print("Index complete")
            docs = myhtable_index_search(files, index, terms)
        else:
            print(f"Invalid search type: {impl}")
            break

        # Generate the search results page
        search_results_page = results(docs, terms)

        # Write search results to a temporary HTML file
        temp_results_path = "/tmp/results.html"
        with open(temp_results_path, "w") as results_file:
            results_file.write(search_results_page)

        # Open the search results in a new browser tab
        webbrowser.open_new_tab(f"file://{temp_results_path}")

if __name__ == "__main__":
    main()
