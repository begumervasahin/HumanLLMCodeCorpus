import os
import time
import re
from hash import hash

def read_directory(directory):
    """Read all files from the given directory and return a list of file paths."""
    return [os.path.join(directory, name) for name in os.listdir(directory) if os.path.isfile(os.path.join(directory, name))]

def normalize_text(text):
    """Normalize text by converting to lowercase and removing punctuation and extra whitespace."""
    return re.sub(r'[^\w\s]', '', text.lower()).replace('\r', '').replace('\t', '').replace('\n', ' ')

def generate_index(file_path, file_num, hashing):
    """Generate an index for words in the given file and update the hashing table."""
    with open(file_path, 'r') as file:
        words = normalize_text(file.read()).split()
    
    for word in words:
        count = words.count(word)
        entries = hashing.getValue(word) or []
        
        # Check if the word is already indexed for this file with the same count
        if not any(entry[0] == count and entry[1] == file_num + 1 for entry in entries):
            hashing.insertValue(word, [count, file_num + 1])

def print_hashing_results(hashing, files):
    """Print the results of the hashing."""
    for key in sorted(hashing.getKeys()):
        count, file_index = hashing.getValue(key)[0]
        print(key, count, files[file_index - 1])

def process_files_with_hashing_methods(directory):
    """Process files with different hashing methods and print results."""
    files = read_directory(directory)
    hashing_methods = {
        'Multiplication-Linear': hash(1000000, 3, 'multiplication', 'linear'),
        'Multiplication-Quadratic': hash(1000000, 3, 'multiplication', 'quadratic'),
        'Division-Linear': hash(1000000, 3, 'division', 'linear'),
        'Division-Quadratic': hash(1000000, 3, 'division', 'quadratic')
    }

    for method_name, hashing in hashing_methods.items():
        start_time = time.time()
        
        for i, file in enumerate(files):
            generate_index(file, i, hashing)
        
        print(f'Results for {method_name}:')
        print_hashing_results(hashing, files)
        
        end_time = time.time()
        print(f'Time for {method_name}: {end_time - start_time}\n')

def main():
    directory = 'base'
    process_files_with_hashing_methods(directory)
    
    while True:
        word = input("Enter a word to search (or press Enter to exit):\n").lower()
        if word:
            print(hashing_methods['Multiplication-Linear'].getValue(word))
        else:
            break

if __name__ == "__main__":
    main()
