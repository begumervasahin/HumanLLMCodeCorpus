# Function to retrieve text content from a file
def get_text(file):
    # Implementation not provided, replace this with your own implementation
    with open(file, 'r') as f:
        return f.read()

# Function to extract words from text
def words(text):
    # Implementation not provided, replace this with your own implementation
    # This example splits the text into words based on whitespace
    return text.split()

# Function to perform linear search for files containing all search terms
def linear_search(files, terms):
    # Check if either files or terms is None or empty
    if files is None or terms is None or len(files) == 0 or len(terms) == 0:
        return None

    # List to store filenames containing all search terms
    ret_docs = []

    # Iterate through each file in the list of files
    for file in files:
        # Retrieve text content from the file
        file_content = get_text(file)

        # Check if file_content is not None
        if file_content is not None:
            # Check if all search terms are found in the words extracted from file_content 
            all_terms_found = all(term in words(file_content) for term in terms)

            # If all search terms are found, add the file to the list of result documents  
            if all_terms_found:
                ret_docs.append(file)

    # Return the list of filenames containing all search terms
    return ret_docs

# Example usage
if __name__ == "__main__":
    # Example list of files
    files = ["file1.txt", "file2.txt", "file3.txt"]

    # Example list of search terms
    search_terms = ["term1", "term2", "term3"]

    # Perform linear search to find files containing all search terms
    result = linear_search(files, search_terms)

    # Print the result
    print("Files containing all search terms:")
    if result:
        for file in result:
            print(file)
    else:
        print("No files contain all search terms.")
