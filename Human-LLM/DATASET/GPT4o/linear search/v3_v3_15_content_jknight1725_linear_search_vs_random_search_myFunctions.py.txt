from random import randrange
from time import process_time

# Input functions
def number_prompt(lower_bound, upper_bound):
    """Prompt the user to enter a number within a specified range."""
    while True:
        try:
            num = int(input(f"Enter a number between {lower_bound} and {upper_bound}: "))
            if lower_bound <= num <= upper_bound:
                return num
            else:
                print("Please enter a valid number.")
        except ValueError:
            print("Please enter a valid number.")

def prompt_for_num():
    """Prompt the user to enter a number."""
    while True:
        try:
            return int(input("Enter a number: "))
        except ValueError:
            print("Please enter a valid number.")

def get_user_guesses():
    """Get three numbers guessed by the user."""
    return [prompt_for_num() for _ in range(3)]

# Output functions
def computer_says(cpu_guess):
    """Output the computer's guess."""
    print(f'Computer says: {cpu_guess}')

def found(cpu_guess, guesses):
    """Output the success message with the number of guesses taken."""
    print(f'Done! Your number is {cpu_guess}. Guessed in {guesses} times!')

def print_time_report(time):
    """Print the time report for function execution."""
    print(f"Starts at: {time['start']}\nEnds at: {time['end']}\nTime taken to execute the function: {time['total']}\n")

# Game logic functions
def get_adjustment():
    """Get adjustment input from the user."""
    while True:
        adjustment = input("Type 'l' if larger, 's' if smaller, 'c' if correct: ").strip().lower()
        if adjustment in {'l', 's', 'c'}:
            return adjustment

def make_adjustment(adjustment, cpu_guess, lower_bound, upper_bound):
    """Adjust the guessing range based on user input."""
    if adjustment == 'l':
        return [cpu_guess, upper_bound]
    elif adjustment == 's':
        return [lower_bound, cpu_guess]
    elif adjustment == 'c':
        return [lower_bound, upper_bound]

def check_for_win(user_numbers, rand_numbers):
    """Check if the user's numbers match the random numbers."""
    return user_numbers == rand_numbers

# Time measurement function
def measure_time(func, *args):
    """Measure the time taken to execute a function."""
    start_time = process_time()
    result = func(*args)
    end_time = process_time()
    time_taken = end_time - start_time
    time_report = {
        'start': start_time,
        'end': end_time,
        'total': time_taken
    }
    return result, time_report

# Game algorithms
def worst_case_algorithm(options, three_random_numbers):
    """Find the position of the random numbers in the list of options (worst case)."""
    for idx, option in enumerate(options):
        if [int(i) for i in str(option)] == three_random_numbers:
            return idx + 1

def random_case_algorithm(three_random_numbers):
    """Randomly guess numbers until the correct combination is found (random case)."""
    upper_constraint = 5000
    guesses = 1
    while guesses < upper_constraint:
        if get_three_random_numbers() == three_random_numbers:
            return guesses
        guesses += 1
    return guesses

# Utility functions
def sum_up(large_num):
    """Calculate the sum of integers from 1 to large_num."""
    return int((large_num * (large_num + 1)) / 2)

def get_three_random_numbers():
    """Generate a list of three random numbers between 0 and 9."""
    return [randrange(10) for _ in range(3)]

def all_1000_options():
    """Generate a list of all 1000 possible three-digit combinations."""
    return [str(x).zfill(3) for x in range(1000)]

# Main function
if __name__ == "__main__":
    lower_bound = 0
    upper_bound = 9

    user_numbers = get_user_guesses()
    rand_numbers = get_three_random_numbers()

    print("Random Numbers: ", rand_numbers)
    print("User Numbers: ", user_numbers)

    if check_for_win(user_numbers, rand_numbers):
        print("Congratulations! You guessed the correct numbers!")
    else:
        print("Try again!")

    options = all_1000_options()
    result, time_report = measure_time(worst_case_algorithm, options, rand_numbers)
    print(f"Worst Case Algorithm Result: {result}")
    print_time_report(time_report)

    result, time_report = measure_time(random_case_algorithm, rand_numbers)
    print(f"Random Case Algorithm Result: {result}")
    print_time_report(time_report)
