import cv2
import cv2.aruco as aruco

def configure_camera_settings(cap):
    """
    Configure the settings for the video capture device.
    Uncomment the lines to adjust specific settings as needed.
    """
    # cap.set(cv2.CAP_PROP_AUTOFOCUS, 0)  # Disable autofocus
    # cap.set(cv2.CAP_PROP_FRAME_WIDTH, 720)  # Set width
    # cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 720)  # Set height

def detect_and_draw_markers(frame, aruco_dict, parameters):
    """
    Detect ArUco markers in the given frame and draw them.

    Args:
        frame (numpy.ndarray): The input grayscale frame.
        aruco_dict (cv2.aruco.Dictionary): The ArUco dictionary to use for detection.
        parameters (cv2.aruco.DetectorParameters): The parameters for marker detection.

    Returns:
        numpy.ndarray: The frame with detected markers drawn.
    """
    # Detect ArUco markers in the frame
    corners, ids, _ = aruco.detectMarkers(frame, aruco_dict, parameters=parameters)

    # Draw the detected markers on the frame
    frame_with_markers = aruco.drawDetectedMarkers(frame.copy(), corners, ids)

    return frame_with_markers

def main():
    # Initialize the video capture device (0 for the default camera, 1 for the external camera)
    cap = cv2.VideoCapture(1)

    # Configure the camera settings
    configure_camera_settings(cap)

    # Define the ArUco dictionary and detection parameters
    aruco_dict = aruco.Dictionary_get(aruco.DICT_6X6_250)
    parameters = aruco.DetectorParameters_create()

    while True:
        # Capture a frame from the video capture device
        ret, frame = cap.read()

        if not ret:
            print("Failed to capture frame. Exiting...")
            break

        # Convert the captured frame to grayscale
        gray_frame = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

        # Detect and draw ArUco markers on the grayscale frame
        frame_with_markers = detect_and_draw_markers(gray_frame, aruco_dict, parameters)

        # Display the frame with detected markers
        cv2.imshow('Frame with ArUco Markers', frame_with_markers)

        # Exit the loop if the 'q' key is pressed
        if cv2.waitKey(1) & 0xFF == ord('q'):
            break

    # Release the video capture device and close all OpenCV windows
    cap.release()
    cv2.destroyAllWindows()

if __name__ == "__main__":
    main()
