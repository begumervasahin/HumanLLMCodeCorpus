Digit Factorials
Problem 34

145 is a curious number, as 1! + 4! + 5! = 1 + 24 + 120 = 145.
Find the sum of all numbers which are equal to the sum of the factorial of their digits.
Note: as 1! = 1 and 2! = 2 are not sums they are not included.
"""

def factorial(x):
    """
    Computes the factorial of a given number x.
    
    :param x: Integer
    :return: Factorial of x
    """
    fac = 1
    for i in range(2, x + 1):
        fac *= i
    return fac

def separate_into_chars(num):
    """
    Separates a number into its individual digits.
    
    :param num: Integer
    :return: List of digits
    """
    return list(str(num))

def is_special_number(num):
    """
    Checks if a number is equal to the sum of the factorial of its digits.
    
    :param num: Integer
    :return: True if the number is special, False otherwise
    """
    digits = separate_into_chars(num)
    total = sum(factorial(int(digit)) for digit in digits)
    return total == num

def find_all_special_numbers(max_num):
    """
    Finds all special numbers less than max_num.
    
    :param max_num: Maximum number to check
    :return: List of special numbers
    """
    special_numbers = []
    for i in range(3, max_num):
        if i % 100000 == 0:
            print(f"Trying i as {i}, so there's another hundred thousand tested")
        if is_special_number(i):
            special_numbers.append(i)
    
    print(f"Special numbers are {special_numbers}, so there are {len(special_numbers)} that are less than {max_num}.")
    return special_numbers

if __name__ == "__main__":
    find_all_special_numbers(100000000)
