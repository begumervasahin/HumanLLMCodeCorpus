from linkedlist import LinkedList
from vartree import BinarySearchTree
from infixtotree import to_expr_tree

class VariableTree(BinarySearchTree):
    def assign(self, variable, value):
        """
        Assigns a value to a variable in the binary search tree.
        
        :param variable: Variable to assign value to
        :param value: Value to assign
        """
        self.insert(variable, value)

    def lookup(self, variable):
        """
        Looks up the value of a variable in the binary search tree.
        
        :param variable: Variable to look up
        :return: Value of the variable
        """
        node = self.search(variable)
        if node:
            return node.value
        raise ValueError(f"Variable '{variable}' not found in the tree")

# Initialize variable tree
varTree = VariableTree()

def eval_postfix(postfixExpr):
    """
    Evaluates a postfix expression using a stack and a binary search tree for variable storage.
    
    :param postfixExpr: Postfix expression as a list of tokens
    :return: Evaluation result
    """
    stack = LinkedList()
    
    for token in postfixExpr:
        if token[0].isdigit() or token[0].isalpha():
            stack.push(token)
        elif token == '=':
            top = stack.pop()
            bottom = stack.pop()
            if top[0].isalpha():
                top = varTree.lookup(top)
            varTree.assign(bottom, top)
            stack.push(bottom)
        else:
            top = stack.pop()
            bottom = stack.pop()
            if top[0].isalpha():
                top = varTree.lookup(top)
            if bottom[0].isalpha():
                bottom = varTree.lookup(bottom)
            temp = eval(f"{bottom} {token} {top}")
            stack.push(str(temp))
    
    result = stack.top()
    if result[0].isdigit():
        return float(result)
    return varTree.lookup(result)

    # Example postfix expression: 12 3 * 5 + a =
    postfix_expr = ['12', '3', '*', '5', '+', 'a', '=']
    result = eval_postfix(postfix_expr)
    print("Result:", result)
