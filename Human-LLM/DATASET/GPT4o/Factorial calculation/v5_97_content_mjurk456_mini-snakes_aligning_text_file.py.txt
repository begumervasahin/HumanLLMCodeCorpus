def text_file_aligning(file_path, align, line_length):
    """
    Splits a text file into lines of a specified length and aligns these lines
    according to the specified alignment type.

    Parameters:
    - file_path (str): Path to the .txt file.
    - align (str): Type of alignment. Use:
        'r' for right alignment,
        'l' for left alignment,
        'c' for center alignment.
    - line_length (int): Maximum length of each line after splitting.

    Returns:
    - None: Prints the aligned lines to the screen.
    """
    # Determine the alignment pattern based on the 'align' parameter
    alignment_patterns = {
        'r': '{:>',   # Right alignment
        'l': '{:',    # Left alignment
        'c': '{:^'    # Center alignment
    }

    if align not in alignment_patterns:
        raise ValueError("Invalid alignment option. Use 'r', 'l', or 'c'.")

    pattern = alignment_patterns[align] + f"{line_length}}"

    try:
        with open(file_path, 'r') as file:
            for line in file:
                line = line.rstrip('\n')  # Remove any trailing newline characters

                while len(line) > line_length:
                    # Find the last space within the line length limit
                    split_pos = line[:line_length].rfind(' ')
                    if split_pos == -1:
                        # If no space is found, split at line_length
                        split_pos = line_length
                    
                    # Print the aligned substring
                    print(pattern.format(line[:split_pos].rstrip()))
                    
                    # Update the line to the remaining part
                    line = line[split_pos:].lstrip()
                
                # Print the remaining part of the line
                print(pattern.format(line))
    except FileNotFoundError:
        print(f"Error: The file '{file_path}' does not exist.")
    except Exception as e:
        print(f"An error occurred: {e}")

if __name__ == "__main__":
    import sys

    # Check for the correct number of command-line arguments
    if len(sys.argv) != 4:
        print("Usage: python script.py <file_path> <alignment> <line_length>")
        sys.exit(1)

    file_path = sys.argv[1]
    align = sys.argv[2]

    try:
        line_length = int(sys.argv[3])
    except ValueError:
        print("Error: Line length must be an integer.")
        sys.exit(1)

    text_file_aligning(file_path, align, line_length)
