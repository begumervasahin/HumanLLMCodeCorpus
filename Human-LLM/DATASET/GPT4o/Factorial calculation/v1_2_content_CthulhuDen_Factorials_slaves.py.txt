import multiprocessing

def worker_function(function, operating, queue_in, queue_out, terminator, lock):
    """
    Worker function for processing tasks in a separate process.
    
    :param function: Function to apply to each task.
    :param operating: Event to control whether the worker should process tasks.
    :param queue_in: Queue to get tasks from.
    :param queue_out: Queue to put results in.
    :param terminator: Event to signal when to stop the worker.
    :param lock: Lock to synchronize access to the queues.
    """
    while not terminator.is_set():
        operating.wait()  # Wait until operating is set
        with lock:  # Automatically acquire and release the lock
            if queue_in.empty():
                continue  # If queue is empty, continue waiting
            task = queue_in.get()
        result = function(task)
        queue_out.put(result)
    queue_out.close()

class Slaves:
    def __init__(self, function, number=multiprocessing.cpu_count()):
        """
        Initializes the Slaves class with a given function and number of worker processes.
        
        :param function: Function to apply to each task.
        :param number: Number of worker processes to create.
        """
        self.queue_in = multiprocessing.Queue()
        self.queue_out = multiprocessing.Queue()
        self.terminator = multiprocessing.Event()
        self.operating = multiprocessing.Event()
        self.lock = multiprocessing.Lock()
        self.processes = []
        for _ in range(number):
            p = multiprocessing.Process(target=worker_function,
                                        args=(function, self.operating, self.queue_in, self.queue_out, self.terminator, self.lock))
            self.processes.append(p)

    def start(self):
        """
        Starts all worker processes and sets the operating event.
        """
        self.operating.set()
        for p in self.processes:
            p.start()

    def put(self, info):
        """
        Puts a task into the queue for processing.
        
        :param info: Task to be processed.
        """
        self.queue_in.put(info)

    def get(self):
        """
        Gets a result from the output queue.
        
        :return: Processed result.
        """
        return self.queue_out.get()

    def pause(self):
        """
        Pauses all worker processes by clearing the operating event.
        """
        self.operating.clear()

    def resume(self):
        """
        Resumes all worker processes by setting the operating event.
        """
        self.operating.set()

    def terminate(self):
        """
        Terminates all worker processes and closes the queues.
        """
        self.operating.set()  # Ensure processes stop waiting
        self.terminator.set()  # Signal termination
        self.queue_in.close()
        for p in self.processes:
            p.join()  # Wait for processes to finish

if __name__ == "__main__":
    import time

    def example_function(x):
        time.sleep(1)  # Simulate a task that takes time
        return x * x

    # Create a Slaves instance with the example_function
    slaves = Slaves(function=example_function, number=4)

    # Start the worker processes
    slaves.start()

    # Put tasks into the queue
    for i in range(10):
        slaves.put(i)

    # Get results
    for _ in range(10):
        print(slaves.get())

    # Terminate the worker processes
    slaves.terminate()
