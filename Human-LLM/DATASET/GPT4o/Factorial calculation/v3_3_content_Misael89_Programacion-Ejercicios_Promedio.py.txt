import time
import os

def clear_screen():
    """Clears the console screen."""
    os.system('cls' if os.name == 'nt' else 'clear')

def prompt_for_number(prompt):
    """
    Prompts the user to input a number until a valid number is entered.
    
    :param prompt: The prompt message to display
    :return: The entered number as an integer
    """
    while True:
        try:
            return int(input(prompt))
        except ValueError:
            print('Solo introduzca números.')
            print('Inténtelo nuevamente.')
            time.sleep(1)
            clear_screen()

def get_number_of_elements():
    """
    Prompts the user to input the number of elements until a valid number is entered.
    
    :return: The number of elements as an integer
    """
    return prompt_for_number("Introduzca el número de elementos: ")

def get_elements(number_of_elements):
    """
    Prompts the user to input the specified number of elements until all are valid numbers.
    
    :param number_of_elements: The number of elements to input
    :return: A list of the entered elements
    """
    elements_list = []
    for i in range(1, number_of_elements + 1):
        element = prompt_for_number(f'Elemento {i}: ')
        elements_list.append(element)
    return elements_list

def calculate_average(elements_list):
    """
    Calculates and prints the average of the elements in the list.
    
    :param elements_list: List of elements
    """
    total = sum(elements_list)
    average = total / len(elements_list)
    print('Promedio: ', average)

def main():
    """Main function to run the script."""
    number_of_elements = get_number_of_elements()
    elements_list = get_elements(number_of_elements)
    calculate_average(elements_list)

if __name__ == "__main__":
    main()
