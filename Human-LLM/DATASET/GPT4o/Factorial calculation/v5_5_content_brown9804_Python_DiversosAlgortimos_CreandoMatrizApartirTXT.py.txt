import numpy as np
import sys

def read_file(file_name, num_columns, num_rows):
    """
    Reads a CSV file and returns its contents as a 2D list.

    :param file_name: The name of the file to read, including its extension.
    :param num_columns: The number of elements expected per row.
    :param num_rows: The number of rows expected in the file.
    :return: A 2D list representing the matrix read from the file.
    """
    matrix = []
    
    try:
        with open(file_name, "r") as file:
            for line in file:
                line_values = line.strip().split(",")
                
                if len(line_values) != num_columns:
                    print(f"Warning: Expected {num_columns} elements per row, but found {len(line_values)}.")
                
                matrix.append(line_values)
                
                if len(matrix) >= num_rows:
                    break

    except FileNotFoundError:
        print(f"Error: The file '{file_name}' was not found.")
        sys.exit()
    except ValueError:
        print("Error: There was a problem with the data format in the file.")
        sys.exit()
    
    return matrix

def main():
    """
    Main function to handle user input, process the file, and display the matrix.
    """
    file_name = input("Enter the file name (including extension, e.g., .txt): ").strip()
    
    try:
        num_columns = int(input("Enter the number of elements per row: ").strip())
        num_rows = int(input("Enter the number of rows: ").strip())
    except ValueError:
        print("Error: Please enter valid integer values for columns and rows.")
        sys.exit()
    
    matrix = read_file(file_name, num_columns, num_rows)
    
    print("Requested matrix (NxM):")
    matrix_np = np.array(matrix)
    print(matrix_np)

if __name__ == "__main__":
    main()
