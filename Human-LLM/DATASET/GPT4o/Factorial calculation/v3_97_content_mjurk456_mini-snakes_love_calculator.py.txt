def count_characters(source, characters):
    """
    Counts how many times characters from 'characters' appear in 'source'.
    
    Parameters:
    - source (str): The string to search.
    - characters (str): The characters to count in 'source'.
    
    Returns:
    - int: The count of characters from 'characters' in 'source'.
    """
    return sum(1 for char in source if char in characters)

def calculate_love_points(name1, name2):
    """
    Calculates love points based on various criteria between two names.
    
    Parameters:
    - name1 (str): The first name.
    - name2 (str): The second name.
    
    Returns:
    - int: The total love points.
    """
    vowels = "aeiou"
    consonants = "qrtpsdfghjklzxcvbnmy"
    
    love_points = 0

    # Add points based on the first letter of each name
    if name1[0] == name2[0]:
        love_points += 20
    if name1[0] in vowels and name2[0] in vowels:
        love_points += 10
    if name1[0] in consonants and name2[0] in consonants:
        love_points += 5

    # Compare the count of vowels and consonants in both names
    if count_characters(name1, vowels) == count_characters(name2, vowels):
        love_points += 12
    if count_characters(name1, consonants) == count_characters(name2, consonants):
        love_points += 12

    # Check if both names contain any letter from 'love'
    if any(letter in name1 and letter in name2 for letter in 'love'):
        love_points += 7

    return love_points

def main():
    while True:
        try:
            # Prompt user for input
            input_string = input("Enter two names separated by a comma (e.g., 'Jan, Joanna'): ").strip()
            name1, name2 = [name.strip().lower() for name in input_string.split(",")]
            
            # Calculate love points and display results
            love_points = calculate_love_points(name1, name2)
            print(f"Your love points are {love_points}.")
            print("REMEMBER: your relationship depends on you and your partner, not on this silly love calculator!")
            break
        except ValueError:
            print("Invalid input. Please enter two names separated by a comma.")

if __name__ == "__main__":
    main()
