def print_A():
    """
    Prints the letter 'A' in a pattern.
    """
    for row in range(6):
        for col in range(11):
            if (row + col == 5) or (col - row == 5) or (row == 3 and col in [7, 9]):
                print('*', end='')
            else:
                print(' ', end='')
        print()

def print_M():
    """
    Prints the letter 'M' in a pattern.
    """
    for row in range(6):
        for col in range(11):
            if col in {0, 10} or row == col or row + col == 10:
                print('*', end='')
            else:
                print(' ', end='')
        print()

def print_I():
    """
    Prints the letter 'I' in a pattern.
    """
    for row in range(7):
        for col in range(11):
            if col == 5:
                print('*', end='')
            else:
                print(' ', end='')
        print()

def print_T():
    """
    Prints the letter 'T' in a pattern.
    """
    for row in range(6):
        for col in range(11):
            if row == 0 or col == 5:
                print('*', end='')
            else:
                print(' ', end='')
        print()

def print_V():
    """
    Prints the letter 'V' in a pattern.
    """
    for row in range(6):
        for col in range(11):
            if row + col == 10 or row == col:
                print('*', end='')
            else:
                print(' ', end='')
        print()

def main():
    """
    Prints the letters A, M, I, T, and V with vertical spacing in between.
    """
    print_A()
    print()  # Print vertical space between letters
    print_M()
    print()  # Print vertical space between letters
    print_I()
    print()  # Print vertical space between letters
    print_T()
    print()  # Print vertical space between letters
    print_V()

if __name__ == "__main__":
    main()
