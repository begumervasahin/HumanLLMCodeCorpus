from flask import Flask, render_template, request
import util

app = Flask(__name__)

@app.route("/")
def root():
    """
    Render the index.html page.
    """
    return render_template("index.html")

@app.route("/results", methods=["GET", "POST"])
def results():
    """
    Handle form submission from index.html and display results based on user query.
    """
    if request.method == "GET":
        query = request.args.get('question', '')
        mode = request.args.get('type', '')

        # Retrieve a list of URLs based on the query
        urls = util.get_list_of_urls(query, 10)
        current_results = {}

        for url in urls:
            # Extract text from each URL
            text = util.get_text_from_url(url)
            
            # Process the text based on the mode
            if mode == 'Who':
                current_results = util.find_name(text, current_results)
            else:
                current_results = util.find_date(text, current_results)

        # Sort the results and render the results.html page
        sorted_results = util.sort_dict_by_value(current_results, 10)
        return render_template('results.html', question=query, results=sorted_results)
    else:
        # If method is POST or any other method, return to index.html
        return render_template('index.html')

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=8000, debug=True)
