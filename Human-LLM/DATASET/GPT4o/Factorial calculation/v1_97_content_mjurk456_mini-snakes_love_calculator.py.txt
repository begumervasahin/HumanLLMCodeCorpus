def counting(source, example):
    """
    Counts the number of characters in 'source' that are present in 'example'.
    
    Parameters:
    - source (str): The string to search.
    - example (str): The characters to count in 'source'.
    
    Returns:
    - int: The count of characters from 'example' in 'source'.
    """
    count = 0
    for char in source:
        if char in example:
            count += 1
    return count

def main():
    # Initialize love points
    love_points = 0
    
    # Define vowels and consonants
    vowels = "aeiou"
    consonants = "qrtpsdfghjklzxcvbnmy"
    
    while True:
        try:
            # Input two names separated by a comma
            names = input("Input two names separated by a comma (e.g., 'Jan, Joanna'): ").split(",")
            name1, name2 = names[0].strip().lower(), names[1].strip().lower()
            break
        except (IndexError, ValueError):
            print("Invalid input. Please enter two names separated by a comma.")

    # Calculate love points based on the first letter of each name
    if name1[0] == name2[0]:
        love_points += 20
    if name1[0] in vowels and name2[0] in vowels:
        love_points += 10
    if name1[0] in consonants and name2[0] in consonants:
        love_points += 5

    # Compare the number of vowels and consonants in both names
    if counting(name1, vowels) == counting(name2, vowels):
        love_points += 12
    if counting(name1, consonants) == counting(name2, consonants):
        love_points += 12

    # Check if both names contain certain letters
    if any(letter in name1 and letter in name2 for letter in 'love'):
        love_points += 7

    # Output the result
    print(f"Your love points are {love_points}")
    print("REMEMBER: your relationship depends only on you and your partner, not on this silly love calculator!")

if __name__ == "__main__":
    main()
