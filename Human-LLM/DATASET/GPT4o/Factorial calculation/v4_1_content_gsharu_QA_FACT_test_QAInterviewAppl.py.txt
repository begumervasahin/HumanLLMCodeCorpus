import time
from selenium import webdriver

# Set up desired capabilities for BrowserStack
desired_cap = {
    'device': 'iPhone 7',
    'realMobile': 'true',
    'platform': 'iOS',
    'browserName': 'safari',
    'browserstack.debug': 'true'
}

# Initialize the WebDriver
driver = webdriver.Remote(
    command_executor='http://sharadag1:ve5y9kXqdCoseBQYsxQ9@hub.browserstack.com:80/wd/hub',
    desired_capabilities=desired_cap
)

# Open the QA Interview application page
driver.get("https://qainterview.pythonanywhere.com/")

# Verify the page title
if driver.title == "Factoriall":
    print("Success: QA Interview Application page launched successfully")
else:
    print("Failure: QA Interview Application page title is incorrect")

# Input the value 5 in the input field
input_field = driver.find_element_by_xpath("//input[@id='number']")
input_field.send_keys('5')

# Click the 'Calculate' button
calculate_button = driver.find_element_by_id('getFactorial')
calculate_button.click()

# Wait for the result to be calculated
time.sleep(5)

# Get the result text and parse the factorial value
result_text = driver.find_element_by_xpath("//p[@id='resultDiv']").text
result_value = int(result_text.split(':')[1].strip())

# Check if the calculated factorial value is 120
if result_value == 120:
    print("Result for 5! is correct:", result_value)
else:
    print("Result for 5! is incorrect:", result_value)

# Close the browser
driver.close()
