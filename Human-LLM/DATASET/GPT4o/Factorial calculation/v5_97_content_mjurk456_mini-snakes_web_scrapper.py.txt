import requests
import os
from bs4 import BeautifulSoup

def download_images_from_page(url, save_directory):
    """
    Downloads images from a webpage and saves them to a specified directory.

    Parameters:
    - url (str): URL of the webpage to scrape images from.
    - save_directory (str): Directory path where images will be saved.
    """
    while True:
        print(f"Fetching page: {url}")
        response = requests.get(url)
        response.raise_for_status()  # Raise an error for bad responses

        soup = BeautifulSoup(response.text, "lxml")
        img_elements = soup.select(".image-size-full")

        # Download each image found on the page
        for img_element in img_elements:
            img_url = img_element.get('src')
            if img_url:
                download_image(img_url, save_directory)

        # Find the link to the next page
        next_page_link = soup.find("a", class_="pagination")
        if next_page_link and 'href' in next_page_link.attrs:
            url = next_page_link.get('href')
        else:
            print("No more pages to download.")
            break

    print("All images have been downloaded.")

def download_image(img_url, save_directory):
    """
    Downloads an image from a URL and saves it to a specified directory.

    Parameters:
    - img_url (str): URL of the image to download.
    - save_directory (str): Directory path where the image will be saved.
    """
    print(f"Downloading image: {img_url}")
    response = requests.get(img_url)
    response.raise_for_status()  # Raise an error for bad responses

    # Construct the full path to save the image
    img_filename = os.path.join(save_directory, os.path.basename(img_url))
    
    # Save the image
    with open(img_filename, 'wb') as img_file:
        for chunk in response.iter_content(1024):  # Download in chunks
            img_file.write(chunk)

    # Define the starting URL and directory to save images
    start_url = 'https://www.boredpanda.com/funny-pet-jerks/'
