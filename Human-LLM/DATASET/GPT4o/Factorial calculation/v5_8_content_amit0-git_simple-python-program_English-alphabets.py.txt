def print_A():
    """
    Prints the letter 'A' in a pattern.
    """
    for row in range(6):
        for col in range(11):
            if (row + col == 5) or (col - row == 5) or (row == 3 and col in {7, 9}):
                print('*', end='')
            else:
                print(' ', end='')
        print()

def print_M():
    """
    Prints the letter 'M' in a pattern.
    """
    for row in range(6):
        for col in range(11):
            if (col == 0) or (col - row == 0) or (row + col == 10) or (col == 10):
                print('*', end='')
            else:
                print(' ', end='')
        print()

def print_I():
    """
    Prints the letter 'I' in a pattern.
    """
    for row in range(7):
        for col in range(11):
            if col == 5:
                print('*', end='')
            else:
                print(' ', end='')
        print()

def print_T():
    """
    Prints the letter 'T' in a pattern.
    """
    for row in range(6):
        for col in range(11):
            if row == 0 or col == 5:
                print('*', end='')
            else:
                print(' ', end='')
        print()

def print_V():
    """
    Prints the letter 'V' in a pattern.
    """
    for row in range(6):
        for col in range(11):
            if (row + col == 10) or (row - col == 0):
                print('*', end='')
            else:
                print(' ', end='')
        print()

def print_letter(letter_func):
    """
    Calls a letter function to print the letter, followed by vertical spacing.
    """
    letter_func()
    print()  # Vertical space between letters

def main():
    """
    Prints the letters A, M, I, T, and V with vertical spacing between them.
    """
    letters = [print_A, print_M, print_I, print_T, print_V]
    for letter_func in letters:
        print_letter(letter_func)

if __name__ == "__main__":
    main()
