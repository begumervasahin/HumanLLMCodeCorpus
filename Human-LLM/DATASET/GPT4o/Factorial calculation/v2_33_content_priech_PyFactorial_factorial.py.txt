def factorial(value):
    """
    Computes the factorial of a positive integer.
    Accepts integers, integer floats, and integer complex numbers with zero imaginary parts as input.
    
    :param value: The positive integer for which to compute the factorial.
    :return: The factorial of the given positive integer.
    :raises ValueError: If the input is not a positive integer or is a complex number with a non-zero imaginary part.
    """
    # Handle complex numbers
    if isinstance(value, complex):
        if value.imag == 0:
            value = value.real
        else:
            raise ValueError("Factorial is only defined for positive integers, not complex values with non-zero imaginary parts.")
    
    # Handle strings
    if isinstance(value, str):
        raise ValueError("Factorial is only defined for positive integers, not strings.")
    
    # Check if the number is a positive integer
    if isinstance(value, (int, float)) and value == int(value):
        value = int(value)
        if value > 0:
            result = 1
            for n in range(1, value + 1):
                result *= n
            return result
        else:
            raise ValueError("Factorial is only defined for positive integers.")
    else:
        raise ValueError("Factorial is only defined for positive integers.")

        print(f"Factorial of {num} is {factorial(num)}")
    except ValueError as e:
