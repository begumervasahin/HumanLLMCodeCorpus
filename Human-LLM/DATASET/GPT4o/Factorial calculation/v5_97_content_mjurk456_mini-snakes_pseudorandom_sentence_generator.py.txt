import random

def load_words(file_path):
    """
    Loads words from a file and returns them as a list.

    Parameters:
    - file_path (str): Path to the file containing words.

    Returns:
    - list: A list of words, each stripped of leading and trailing whitespace.
    """
    with open(file_path, 'r') as file:
        return [line.strip() for line in file]

def generate_sentence(adjectives, nouns, verbs, adverbs):
    """
    Creates a random sentence using the provided lists of words.

    Parameters:
    - adjectives (list): List of adjectives.
    - nouns (list): List of nouns.
    - verbs (list): List of verbs.
    - adverbs (list): List of adverbs.

    Returns:
    - str: A randomly generated sentence.
    """
    return (
        f"A {random.choice(adjectives)} {random.choice(nouns)} "
        f"{random.choice(verbs)} a {random.choice(adjectives)} "
        f"{random.choice(nouns)} {random.choice(adverbs)}."
    )

def get_positive_integer(prompt):
    """
    Prompts the user to enter a positive integer.

    Parameters:
    - prompt (str): The message to display to the user.

    Returns:
    - int: A positive integer entered by the user.
    """
    while True:
        try:
            value = int(input(prompt))
            if value > 0:
                return value
            else:
                print("Please enter a positive integer.")
        except ValueError:
            print("Invalid input. Please enter an integer.")

def main():
    # Load words from files
    adjectives = load_words("adjective.txt")
    nouns = load_words("noun.txt")
    verbs = load_words("verb.txt")
    adverbs = load_words("adverb.txt")

    # Get the number of sentences to generate
    num_sentences = get_positive_integer("How many sentences do you want to generate? ")

    # Generate and print the requested number of sentences
    for _ in range(num_sentences):
        print(generate_sentence(adjectives, nouns, verbs, adverbs))

if __name__ == "__main__":
    main()
