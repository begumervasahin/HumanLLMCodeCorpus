import time
from selenium import webdriver
from selenium.webdriver.common.by import By

def setup_driver():
    """
    Sets up the Firefox WebDriver.
    
    :return: Initialized WebDriver
    """
    return webdriver.Firefox()

def navigate_to_page(driver, url):
    """
    Navigates to the specified URL.
    
    :param driver: WebDriver instance
    :param url: URL to navigate to
    """
    driver.get(url)

def enter_text(driver, element_id, text):
    """
    Enters text into a textbox identified by element ID.
    
    :param driver: WebDriver instance
    :param element_id: ID of the textbox element
    :param text: Text to enter into the textbox
    """
    driver.find_element(By.ID, element_id).send_keys(text)

def click_button(driver, xpath):
    """
    Clicks a button identified by its XPath.
    
    :param driver: WebDriver instance
    :param xpath: XPath of the button element
    """
    driver.find_element(By.XPATH, xpath).click()

def get_validation_result(driver, xpath):
    """
    Retrieves the validation result text from an element identified by XPath.
    
    :param driver: WebDriver instance
    :param xpath: XPath of the result element
    :return: Text content of the result element
    """
    return driver.find_element(By.XPATH, xpath).text

def main():
    """
    Main function to execute the test script.
    """
    driver = setup_driver()

    try:
        # Step 1 & 2: Launch Firefox Driver and navigate to the page
        navigate_to_page(driver, 'https://qainterview.pythonanywhere.com/')
        
        # Step 3: Enter integer value in the textbox
        enter_text(driver, 'number', '12')

        # Step 4: Click on the calculate button
        click_button(driver, "//*[@id='getFactorial']")

        # Step 5: Check the validation message
        time.sleep(3)  # Wait for the result to be displayed
        validation_result = get_validation_result(driver, "//p[@id='resultDiv']")
        
        # Step 6: Verify the validation message
        expected_result = "The factorial of 12 is: 479001600"
        if validation_result == expected_result:
            print("Success: Message displayed successfully")
        else:
            print("Failed: Validation message is incorrect")
    
    finally:
        # Step 7: Close the browser
        driver.quit()

if __name__ == "__main__":
    main()
