def factorial_loop(n):
    """
    Computes the factorial of a positive integer n using an iterative approach.

    :param n: The number to compute the factorial for.
    :return: The factorial of n.
    """
    factorial = 1
    for i in range(n, 0, -1):
        factorial *= i
    return factorial

def factorial_recursive(n):
    """
    Computes the factorial of a positive integer n using a recursive approach.

    :param n: The number to compute the factorial for.
    :return: The factorial of n.
    """
    if n == 0:
        return 1
    elif n == 1:
        return 1
    else:
        return n * factorial_recursive(n - 1)

def get_positive_integer(prompt):
    """
    Prompts the user to enter a positive integer and validates the input.

    :param prompt: Message to display to the user.
    :return: A valid positive integer entered by the user.
    """
    while True:
        user_input = input(prompt)
        if user_input.isdigit():
            return int(user_input)
        print("Invalid input. Please enter a whole number.")

def main():
    """
    Demonstrates calculating the factorial of a number using both iterative and recursive methods.
    """
    # Get input and calculate factorial using a loop
    number = get_positive_integer("Loop Method:\nEnter a whole number to find its factorial: ")
    loop_result = factorial_loop(number)
    print(f"Factorial (loop method) of {number} is: {loop_result}")

    # Get input and calculate factorial using recursion
    number = get_positive_integer("Recursion Method:\nEnter a whole number to find its factorial: ")
    recursion_result = factorial_recursive(number)
    print(f"Factorial (recursion method) of {number} is: {recursion_result}")

if __name__ == "__main__":
    main()
