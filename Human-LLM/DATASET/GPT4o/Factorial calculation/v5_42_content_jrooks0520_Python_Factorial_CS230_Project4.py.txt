def iterative_factorial(n):
    """
    Computes the factorial of a positive integer using an iterative approach.
    
    :param n: Positive integer
    :return: Factorial of n
    """
    result = 1
    for i in range(1, n + 1):
        result *= i
    return result

def recursive_factorial(n):
    """
    Computes the factorial of a positive integer using a recursive approach.
    
    :param n: Positive integer
    :return: Factorial of n
    """
    if n == 1:
        return 1
    return n * recursive_factorial(n - 1)

def get_positive_integer(prompt):
    """
    Prompts the user to input a positive integer.
    
    :param prompt: The prompt message to display
    :return: The entered positive integer
    """
    while True:
        value = input(prompt)
        if value.isdigit():
            value = int(value)
            if value > 0:
                return value
            elif value == 0:
                print("The factorial of 0 is 1!")
                exit()
        print("Invalid input. Please enter a positive integer.")

def choose_factorial_method():
    """
    Prompts the user to choose a method for computing the factorial.
    
    :return: The chosen method ('1' for iterative, '2' for recursive)
    """
    while True:
        option = input("How would you like to solve the factorial?\n"
                       "Option '1': Iterative approach.\n"
                       "Option '2': Recursive approach.\n"
                       "Please enter either '1' or '2': ")
        if option in ('1', '2'):
            return option
        print("That is not an option! Please enter '1' or '2'.")

def main():
    """
    Main function to run the script.
    """
    pos_int = get_positive_integer("Please enter a positive integer: ")
    option = choose_factorial_method()

    if option == '1':
        result = iterative_factorial(pos_int)
    else:
        result = recursive_factorial(pos_int)
    
    print(f"The factorial of {pos_int} is {result}.")

if __name__ == "__main__":
    main()
