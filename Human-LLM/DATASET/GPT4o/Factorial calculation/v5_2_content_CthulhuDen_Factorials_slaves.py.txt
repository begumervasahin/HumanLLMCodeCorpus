import multiprocessing
import time

def worker_function(task_function, operating_event, input_queue, output_queue, terminate_event, lock):
    """
    Worker function that processes tasks in separate processes.
    
    :param task_function: Function to apply to each task.
    :param operating_event: Event to signal whether the worker should process tasks.
    :param input_queue: Queue from which to get tasks.
    :param output_queue: Queue to which results are put.
    :param terminate_event: Event to signal when to stop the worker.
    :param lock: Lock to synchronize access to the queues.
    """
    while not terminate_event.is_set():
        operating_event.wait()  # Wait until operating_event is set
        with lock:
            if not input_queue.empty():
                task = input_queue.get()
                result = task_function(task)
                output_queue.put(result)
    output_queue.close()

class Slaves:
    def __init__(self, task_function, num_processes=None):
        """
        Initializes the Slaves class with a given task function and number of processes.
        
        :param task_function: Function to apply to each task.
        :param num_processes: Number of worker processes to create. Defaults to the number of CPUs.
        """
        if num_processes is None:
            num_processes = multiprocessing.cpu_count()
        
        self.task_function = task_function
        self.input_queue = multiprocessing.Queue()
        self.output_queue = multiprocessing.Queue()
        self.terminate_event = multiprocessing.Event()
        self.operating_event = multiprocessing.Event()
        self.lock = multiprocessing.Lock()
        
        self.processes = [
            multiprocessing.Process(
                target=worker_function,
                args=(
                    self.task_function,
                    self.operating_event,
                    self.input_queue,
                    self.output_queue,
                    self.terminate_event,
                    self.lock
                )
            )
            for _ in range(num_processes)
        ]

    def start(self):
        """
        Starts all worker processes and sets the operating_event.
        """
        self.operating_event.set()
        for process in self.processes:
            process.start()

    def put(self, task):
        """
        Adds a task to the input queue.
        
        :param task: Task to be processed.
        """
        self.input_queue.put(task)

    def get(self):
        """
        Retrieves a result from the output queue.
        
        :return: Processed result.
        """
        return self.output_queue.get()

    def pause(self):
        """
        Pauses task processing by clearing the operating_event.
        """
        self.operating_event.clear()

    def resume(self):
        """
        Resumes task processing by setting the operating_event.
        """
        self.operating_event.set()

    def terminate(self):
        """
        Stops all worker processes and closes the queues.
        """
        self.operating_event.set()  # Ensure workers stop waiting
        self.terminate_event.set()  # Signal workers to terminate
        self.input_queue.close()
        for process in self.processes:
            process.join()  # Wait for processes to finish

# Example usage
if __name__ == "__main__":
    def example_function(x):
        """Simulates a task that takes some time to complete."""
        time.sleep(1)
        return x * x

    # Create an instance of Slaves with the example_function
    slaves = Slaves(task_function=example_function, num_processes=4)

    # Start the worker processes
    slaves.start()

    # Add tasks to the queue
    for i in range(10):
        slaves.put(i)

    # Retrieve and print results
    for _ in range(10):
        print(slaves.get())

    # Terminate the worker processes
    slaves.terminate()
