import asyncio
import math
import sys

async def produce_factorial(queue: asyncio.Queue, start: int = 0, finish: int = None):
    """
    Produces factorial values and places them into the provided queue.

    :param queue: asyncio.Queue where factorial values are put.
    :param start: The starting number for factorial calculations.
    :param finish: The number to stop factorial calculations. If None, produces indefinitely.
    """
    loop = asyncio.get_event_loop()

    async def calculate_factorial(n):
        """
        Calculate the factorial of `n` using a blocking call and put the result into the queue.

        :param n: The number to calculate the factorial for.
        """
        factorial_result = await loop.run_in_executor(None, math.factorial, n)
        await queue.put(factorial_result)

    # Validate and adjust start value
    try:
        current = abs(int(start))
    except ValueError as e:
        raise ValueError(f"Invalid start value: {e}")

    # Produce factorials indefinitely or up to the specified finish value
    if finish is None:
        while True:
            await calculate_factorial(current)
            current += 1
    else:
        try:
            end = abs(int(finish))
        except ValueError as e:
            raise ValueError(f"Invalid finish value: {e}")

        while current < end:
            await calculate_factorial(current)
            current += 1

    # Signal completion by placing None in the queue
    await queue.put(None)

async def consume_factorial(queue: asyncio.Queue):
    """
    Consumes factorial values from the queue and prints them.

    :param queue: asyncio.Queue to retrieve factorial values from.
    """
    while True:
        item = await queue.get()
        if item is None:
            break
        sys.stdout.write(f"Current value: {item}\n")
        await asyncio.sleep(0.2)

async def main():
    queue = asyncio.Queue(maxsize=1)
    producer = produce_factorial(queue, start=0, finish=10)  # Adjust start and finish as needed
    consumer = consume_factorial(queue)
    
    await asyncio.gather(producer, consumer)

if __name__ == "__main__":
    asyncio.run(main())
