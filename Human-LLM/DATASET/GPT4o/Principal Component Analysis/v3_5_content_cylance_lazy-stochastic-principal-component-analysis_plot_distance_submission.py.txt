import numpy as np
import matplotlib.pyplot as plt
from matplotlib.offsetbox import AnchoredText
from matplotlib.patheffects import withStroke

def add_inner_title(title, loc, ax=None, size=None, **kwargs):
    """
    Add an inner title to a matplotlib plot.

    Parameters:
    - title (str): The title text to add to the plot.
    - loc (int): The location code for the anchored text.
    - ax (matplotlib.axes._subplots.AxesSubplot, optional): The axis to add the title to. Defaults to the current axis.
    - size (dict, optional): The size properties for the title text. Defaults to legend font size.
    """
    if ax is None:
        ax = plt.gca()
    if size is None:
        size = {'size': plt.rcParams['legend.fontsize']}
    
    at = AnchoredText(title, loc=loc, prop=size, pad=0.7, borderpad=0.01, frameon=False, **kwargs)
    ax.add_artist(at)
    at.txt._text.set_path_effects([withStroke(foreground="w", linewidth=3)])
    return at

# Mapping of row indices to distance type and column indices to k values
ROW_TO_NAME = {0: "subspace_distance", 1: "Steifel_distance"}
COL_TO_K = {0: 100, 1: 2000}
LABELS = {(0, 0): "(a)", (0, 1): "(b)", (1, 0): "(c)", (1, 1): "(d)"}
PROJECTION_ALGORITHMS = ['alg1', 'alg2', 'alg3', 'alg4', 'alg5']  # Update with actual algorithm names

def plot_matrices():
    """
    Plots a 2x2 grid of matrices with log10 scaling.
    """
    fig, axs = plt.subplots(nrows=2, ncols=2)
    ims = [[None, None], [None, None]]

    for i in range(2):
        for j in range(2):
            name = ROW_TO_NAME[i]
            k = COL_TO_K[j]
            mat = np.loadtxt(f"{name}/{name}__k_{k}.txt")

            # Plot the matrix with log10 scaling
            im = axs[i][j].matshow(np.log10(mat))
            ims[i][j] = im

            # Customize x and y ticks
            if i == 0:
                axs[i][j].set_xticks(np.arange(mat.shape[0]))
                axs[i][j].set_xticklabels(PROJECTION_ALGORITHMS, rotation="vertical")
            else:
                axs[i][j].set_xticks([])

            if j == 0:
                axs[i][j].set_yticks(np.arange(mat.shape[0]))
                axs[i][j].set_yticklabels(PROJECTION_ALGORITHMS)
            else:
                axs[i][j].set_yticks([])

            # Add inner title and colorbar
            add_inner_title(LABELS[(i, j)], loc=2, ax=axs[i][j])
            fig.colorbar(ims[i][j], ax=axs[i][j])

    # Adjust layout and save figure
    plt.subplots_adjust(wspace=0, hspace=0.1)
    plt.savefig("subspace_distance.pdf", bbox_inches="tight")
    plt.show()

# Run the plotting function
if __name__ == "__main__":
    plot_matrices()
