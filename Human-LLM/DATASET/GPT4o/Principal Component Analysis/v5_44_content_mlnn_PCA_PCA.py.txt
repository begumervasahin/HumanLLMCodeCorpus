import numpy as np
import matplotlib.pyplot as plt
from sklearn.decomposition import PCA, KernelPCA
from sklearn.datasets import make_circles

# Set a random seed for reproducibility
np.random.seed(0)

# Generate a synthetic dataset using make_circles
X, y = make_circles(n_samples=400, factor=0.3, noise=0.05)

# Apply Kernel PCA with RBF kernel
kpca = KernelPCA(kernel="rbf", fit_inverse_transform=True, gamma=10)
X_kpca = kpca.fit_transform(X)
X_back = kpca.inverse_transform(X_kpca)

# Apply standard PCA
pca = PCA()
X_pca = pca.fit_transform(X)

# Define colors for plotting
reds = y == 0
blues = y == 1

# Create a figure for the subplots
plt.figure(figsize=(12, 10))

# Original space plot
plt.subplot(2, 2, 1, aspect='equal')
plt.title("Original space")
plt.plot(X[reds, 0], X[reds, 1], "ro", label='Class 0')
plt.plot(X[blues, 0], X[blues, 1], "bo", label='Class 1')
plt.xlabel("$x_1$")
plt.ylabel("$x_2$")
plt.legend()

# Create a grid for contour plot in the original space
X1, X2 = np.meshgrid(np.linspace(-1.5, 1.5, 50), np.linspace(-1.5, 1.5, 50))
X_grid = np.array([X1.ravel(), X2.ravel()]).T
Z_grid = kpca.transform(X_grid)[:, 0].reshape(X1.shape)
plt.contour(X1, X2, Z_grid, colors='grey', linewidths=1, origin='lower')

# Projection by PCA plot
plt.subplot(2, 2, 2, aspect='equal')
plt.title("Projection by PCA")
plt.plot(X_pca[reds, 0], X_pca[reds, 1], "ro")
plt.plot(X_pca[blues, 0], X_pca[blues, 1], "bo")
plt.xlabel("1st principal component")
plt.ylabel("2nd component")

# Projection by KPCA plot
plt.subplot(2, 2, 3, aspect='equal')
plt.title("Projection by KPCA")
plt.plot(X_kpca[reds, 0], X_kpca[reds, 1], "ro")
plt.plot(X_kpca[blues, 0], X_kpca[blues, 1], "bo")
plt.xlabel("1st principal component in space induced by $\phi$")
plt.ylabel("2nd component")

# Original space after inverse transform plot
plt.subplot(2, 2, 4, aspect='equal')
plt.title("Original space after inverse transform")
plt.plot(X_back[reds, 0], X_back[reds, 1], "ro")
plt.plot(X_back[blues, 0], X_back[blues, 1], "bo")
plt.xlabel("$x_1$")
plt.ylabel("$x_2$")

# Adjust layout and show the plots
plt.tight_layout(pad=2.0)
plt.show()
