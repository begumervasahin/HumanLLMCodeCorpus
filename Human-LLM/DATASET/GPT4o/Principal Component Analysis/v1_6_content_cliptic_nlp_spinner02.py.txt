import nltk
import random
import pandas as pd
import numpy as np
from bs4 import BeautifulSoup

# Download the necessary NLTK data
nltk.download('punkt')

# Load and parse the positive reviews
with open('electronics/positive.review', 'r', encoding='utf-8') as f:
    positive_reviews = BeautifulSoup(f.read(), 'html.parser')
positive_reviews = positive_reviews.findAll("review_text")

# Create trigrams
trigrams = {}
for review in positive_reviews:
    s = review.text.lower()
    tokens = nltk.tokenize.word_tokenize(s)
    for i in range(len(tokens) - 2):
        k = (tokens[i], tokens[i + 2])
        if k not in trigrams:
            trigrams[k] = []
        trigrams[k].append(tokens[i + 1])

# Calculate probabilities
probabilities = {}
for k, words in trigrams.items():
    if len(set(words)) > 1:
        n = 0
        d = {}
        for w in words:
            if w not in d:
                d[w] = 0
            d[w] += 1
            n += 1
        for w, c in d.items():
            d[w] = c / n
        probabilities[k] = d

# Function to sample from the distribution
def random_sample(d):
    r = random.random()
    cumulative = 0
    for w, p in d.items():
        cumulative += p
        if r < cumulative:
            return w

# Function to test the spinner
def test_spinner():
    review = random.choice(positive_reviews)
    review = review.text.lower()
    print('Original text: \n', review)
    tokens = nltk.tokenize.word_tokenize(review)
    for i in range(len(tokens) - 2):
        if random.random() < 0.2:
            k = (tokens[i], tokens[i + 2])
            if k in probabilities:
                w = random_sample(probabilities[k])
                tokens[i + 1] = w
    print("Spun: \n")
    print(' '.join(tokens).replace(" :", ":").replace(" .", ".").replace(" ,", ",").replace(" !", "!").replace(" ?", "?").replace("  ' ", "'"))

# Run the spinner test
test_spinner()
