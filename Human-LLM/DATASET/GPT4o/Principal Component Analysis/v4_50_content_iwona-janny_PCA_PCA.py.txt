import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
import matplotlib.pyplot as plt

# Load the dataset and rename the columns
data = pd.read_csv("Wine.csv", header=1)
data.columns = ["Alkohol", "Kwas jabłkowy", "Popiół", "Zasadowość popiołu", "Magnez", 
                "Całkowita zawartość fenoli", "Flawonoidy", "Fenole nieflawonoidowe", 
                "Proantocyjaniny", "Intensywność koloru", "Odcień", "Transmitacja", 
                "Prolina", "Etykieta"]

# Display the first few rows of the dataset
print(data.head())

# Split the data into features (X) and labels (y)
X = data.iloc[:, :-1].values
y = data.iloc[:, -1].values

# Split the dataset into training and testing sets
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=0)

# Standardize the features
sc = StandardScaler()
X_train_stand = sc.fit_transform(X_train)
X_test_stand = sc.transform(X_test)

# Compute the covariance matrix of the standardized training data
cov_mat = np.cov(X_train_stand.T)

# Calculate eigenvalues and eigenvectors of the covariance matrix
eigen_val, eigen_vec = np.linalg.eig(cov_mat)

# Compute the explained variance from the eigenvalues
total = sum(eigen_val)
variance_explained = [(i / total) for i in sorted(eigen_val, reverse=True)]
cumulated_variance_explained = np.cumsum(variance_explained)

# Create pairs of eigenvalues and eigenvectors, and sort them in descending order
eigen_pairs = [(np.abs(eigen_val[i]), eigen_vec[:, i]) for i in range(len(eigen_val))]
eigen_pairs.sort(key=lambda k: k[0], reverse=True)

# Construct the projection matrix using the top 2 eigenvectors
projection_matrix = np.hstack((eigen_pairs[0][1][:, np.newaxis], eigen_pairs[1][1][:, np.newaxis]))

# Project the standardized training data onto the new 2D subspace
Pca = X_train_stand.dot(projection_matrix)

# Plot the PCA-transformed data
targets = np.unique(y_train)
colors = ["r", "g", "b"]
markers = ["s", "x", "o"]

for t, c, m in zip(targets, colors, markers):
    plt.scatter(Pca[y_train == t, 0], Pca[y_train == t, 1], c=c, label=t, marker=m)

plt.xlabel("PC1")
plt.ylabel("PC2")
plt.legend(loc="lower right")
plt.show()
