import numpy as np
import os
import h5py
from sklearn.decomposition import IncrementalPCA

def generate_random_batch(rows, columns):
    """
    Generate a random array with the specified number of rows and columns.
    
    Parameters:
        rows (int): Number of rows in the array.
        columns (int): Number of columns in the array.
        
    Returns:
        numpy.ndarray: Randomly generated array.
    """
    return np.random.rand(rows, columns)

def create_dense_dataset(batch_rows=1000, total_rows=5000, columns=1000, save_fp="dense.h5"):
    """
    Create a large dense dataset in HDF5 format in batches.
    
    Parameters:
        batch_rows (int): Number of rows per batch.
        total_rows (int): Total number of rows in the dataset.
        columns (int): Number of columns in the dataset.
        save_fp (str): File path to save the dataset.
    """
    # Remove existing file if it exists
    if os.path.exists(save_fp):
        os.remove(save_fp)
    
    # Create HDF5 file and dataset
    with h5py.File(save_fp, mode="a") as f:
        dset = f.create_dataset('float', (total_rows, columns), dtype=np.float)
        
        # Generate and save data in batches
        for i in range(int(total_rows / batch_rows)):
            batch_data = generate_random_batch(batch_rows, columns)
            dset[i * batch_rows:(i + 1) * batch_rows] = batch_data
            del batch_data
    
    print("Dataset created and saved to", save_fp)

def apply_incremental_pca(batch_rows=500, compressed_columns=200, uncompressed_fp="dense.h5", compressed_fp="compressed_dense.h5"):
    """
    Read the dense dataset in batches, apply Incremental PCA, and save the compressed dataset.
    
    Parameters:
        batch_rows (int): Number of rows per batch for reading and transforming.
        compressed_columns (int): Number of columns for the compressed dataset.
        uncompressed_fp (str): File path of the uncompressed dataset.
        compressed_fp (str): File path to save the compressed dataset.
    """
    # Open the uncompressed dataset
    with h5py.File(uncompressed_fp, 'r') as data:
        total_rows = data["float"].shape[0]
        ipca = IncrementalPCA(n_components=compressed_columns)
        
        # Fit the PCA model in batches
        for i in range(int(total_rows / batch_rows) + 1):
            batch_data = data["float"][i * batch_rows:(i + 1) * batch_rows]
            if len(batch_data) == 0:
                break
            ipca.partial_fit(batch_data)
        
        print("Incremental PCA model fitted.")
        
        # Remove existing file if it exists
        if os.path.exists(compressed_fp):
            os.remove(compressed_fp)
        
        # Create HDF5 file and compressed dataset
        with h5py.File(compressed_fp, mode="a") as output:
            dset = output.create_dataset('float', (total_rows, compressed_columns), dtype=np.float)
            
            # Transform and save data in batches
            for i in range(int(total_rows / batch_rows) + 1):
                if i * batch_rows >= total_rows:
                    break
                transformed_batch = ipca.transform(data["float"][i * batch_rows:(i + 1) * batch_rows])
                dset[i * batch_rows:(i + 1) * batch_rows] = transformed_batch
                del transformed_batch
    
    print("Compressed dataset created and saved to", compressed_fp)

if __name__ == "__main__":
    create_dense_dataset()
    apply_incremental_pca()
