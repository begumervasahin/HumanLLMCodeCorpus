import numpy as np

# Function to zero mean the data
def zeroMean(datamat):
    mean = np.mean(datamat, axis=0)  # Calculate the mean of each column
    newdata = datamat - mean  # Subtract the mean from each column
    return newdata, mean

# Function to perform PCA on the data
def pca(datamat):
    newdata, mean = zeroMean(datamat)  # Zero mean the data
    covmat = np.cov(newdata, rowvar=0)  # Calculate the covariance matrix
    eigvals, eigvects = np.linalg.eig(np.mat(covmat))  # Compute eigenvalues and eigenvectors
    eigvals_sort = np.argsort(eigvals)[::-1]  # Sort eigenvalues in descending order
    eigval_final = eigvals[eigvals_sort]  # Reorder eigenvalues
    eigvect_final = eigvects[:, eigvals_sort]  # Reorder eigenvectors
    return eigval_final, eigvect_final

# Function to calculate the percentage of variance explained by the top components
def percent_percentage(eigvals, percentage):
    total_variance = sum(eigvals)  # Sum of all eigenvalues
    cumulative_variance = 0
    num_components = 0
    
    for eigval in eigvals:
        cumulative_variance += eigval
        num_components += 1
        if cumulative_variance >= total_variance * percentage:
            break
    
    result_percent = np.zeros(num_components)
    for i in range(num_components):
        result_percent[i] = eigvals[i] / total_variance  # Calculate variance ratio for each component
    
    return num_components, result_percent

# Function to return the percentage of variance for all components
def percent_all(eigvals):
    _, percent = percent_percentage(eigvals, 1)
    return percent

# Function to return the number of components with variance greater than 10%
def percent_over_10(eigvals):
    _, percent = percent_percentage(eigvals, 1)
    num_components = sum(1 for p in percent if p > 0.1)  # Count components with >10% variance
    percent = percent[:num_components]
    return num_components, percent

# Function to project the data onto the top principal components
def product(datamat, eigvect, num):
    eigvect_final = eigvect[:, :num]  # Select the top 'num' eigenvectors
    low_data = np.dot(datamat, eigvect_final)  # Project the data
    return low_data

# Example usage
if __name__ == "__main__":
    datamat = np.array([
        [2.5, 2.4],
        [0.5, 0.7],
        [2.2, 2.9],
        [1.9, 2.2],
        [3.1, 3.0],
        [2.3, 2.7],
        [2.0, 1.6],
        [1.0, 1.1],
        [1.5, 1.6],
        [1.1, 0.9]
    ])

    eigvals, eigvects = pca(datamat)
    print("Eigenvalues:", eigvals)
    print("Eigenvectors:\n", eigvects)

    num, percent = percent_over_10(eigvals)
    print("Number of components with >10% variance:", num)
    print("Percentage of variance for these components:", percent)

    low_data = product(datamat, eigvects, num)
    print("Reduced data:\n", low_data)
