from time import time
from numpy import log2, array, load, save
from sklearn.decomposition import PCA
from joblib import Parallel, delayed
from open3d import read_point_cloud

def log_var(pts):
    pca = PCA()
    pca.fit(pts)
    pca.explained_variance_[:-1] -= pca.explained_variance_[1:]
    return log2(pca.explained_variance_)

def export_pca(ply_vox, npy_knn, temp_out, n_jobs):
    vox = array(read_point_cloud(ply_vox).points)
    knn = load(npy_knn)
    save(temp_out, Parallel(n_jobs=n_jobs, verbose=1)(
        delayed(log_var)(vox[n]) for n in knn))
    return None

def main():
    begin = int(input('begin: '))
    end = int(input('end: '))
    n_jobs = input('n_jobs: ')
    if not n_jobs:
        n_jobs = 8
    else:
        n_jobs = int(n_jobs)
    
    t = time()
    
    if begin <= 0 <= end:

    if begin <= 1 <= end:
        print('1: mn')
        import mn

    if begin <= 2 <= end:
        print('2: pcamn')
        export_pca('vox1.ply', '128_1.npy', 'pca_1.npy', n_jobs)
        export_pca('vox2.ply', '128_2.npy', 'pca_2.npy', n_jobs)
        export_pca('vox4.ply', '128_4.npy', 'pca_4.npy', n_jobs)

    if begin <= 3 <= end:
        print('3: concat')
        import concat
