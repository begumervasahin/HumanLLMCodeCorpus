import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA

# Load the Iris dataset
df = pd.read_csv("D:\Python_programs\ML\iris.csv")

# Define the features and target
FEATURES = ['sepal length', 'sepal width', 'petal length', 'petal width']
TARGET = 'target'

# Extract features and target from the dataframe
X = df[FEATURES].values
Y = df[[TARGET]].values

# Standardize the features
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# Perform PCA with 4 components
pca_4 = PCA(n_components=4)
principal_components_4 = pca_4.fit_transform(X_scaled)
explained_variance_ratio = pca_4.explained_variance_ratio_

# Plot the explained variance ratio
plt.figure(figsize=(8, 6))
plt.bar(range(1, 5), explained_variance_ratio * 100, color='b', label='Principal Components')
plt.xlabel('Principal Components')
plt.ylabel('Variance Ratio (%)')
plt.title('Variance Ratio of IRIS Dataset')
plt.xticks(range(1, 5), ['PC-1', 'PC-2', 'PC-3', 'PC-4'], fontsize=10, rotation=30)
plt.legend()
plt.grid(True)
plt.show()

# Perform PCA with 2 components for visualization
pca_2 = PCA(n_components=2)
principal_components_2 = pca_2.fit_transform(X_scaled)
principal_df = pd.DataFrame(data=principal_components_2, columns=['PC-1', 'PC-2'])

# Concatenate the target variable for plotting
final_df = pd.concat([principal_df, df[[TARGET]]], axis=1)

# Plot the PCA result
fig, ax = plt.subplots(figsize=(8, 8))
ax.set_xlabel('PC-1', fontsize=15)
ax.set_ylabel('PC-2', fontsize=15)
ax.set_title('PCA on IRIS Dataset', fontsize=20)

TARGETS = ['Iris-setosa', 'Iris-versicolor', 'Iris-virginica']
COLORS = ['r', 'g', 'b']

for target, color in zip(TARGETS, COLORS):
    indices_to_keep = final_df[TARGET] == target
    ax.scatter(final_df.loc[indices_to_keep, 'PC-1'], final_df.loc[indices_to_keep, 'PC-2'], c=color, s=50, label=target)

ax.legend()
ax.grid(True)
plt.show()

# Save the PCA result to a CSV file
final_df.to_csv('iris_after_pca.csv', index=False)
