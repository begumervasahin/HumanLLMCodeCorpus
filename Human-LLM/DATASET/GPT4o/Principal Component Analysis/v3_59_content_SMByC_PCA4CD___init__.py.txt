import os
import site
import pkg_resources

def add_extra_libs_to_path():
    """
    Adds the extra libraries needed by the plugin to the Python path.
    Ensures that any dependencies located in the 'extlibs' directory are available.
    """
    # Determine the absolute path to the 'extlibs' directory
    extlibs_path = os.path.abspath(os.path.join(os.path.dirname(__file__), "extlibs"))
    
    # If the 'extlibs' directory exists, add it to the Python site-packages and pkg_resources
    if os.path.isdir(extlibs_path):
        site.addsitedir(extlibs_path)
        pkg_resources.working_set.add_entry(extlibs_path)

def classFactory(iface):
    """
    Initializes the PCA4CD plugin within QGIS.
    
    :param iface: A QGIS interface instance.
    :type iface: QgsInterface
    :return: An instance of the PCA4CD class
    """
    # Initialize any additional libraries required by the plugin
    add_extra_libs_to_path()
    
    # Import the PCA4CD class from the 'pca4cd' module
    from .pca4cd import PCA4CD
    
    # Return an instance of the PCA4CD class
    return PCA4CD(iface)
