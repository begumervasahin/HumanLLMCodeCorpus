import nltk
import numpy as np
import pandas as pd
from bs4 import BeautifulSoup
from nltk.tokenize import RegexpTokenizer

# Define a tokenizer that matches word characters
tokenizer = RegexpTokenizer(r'\w+')

def html_to_probs_dict(html_txt_link):
    # Parse the HTML content
    with open(html_txt_link, 'r') as file:
        soup = BeautifulSoup(file.read(), features="html.parser")
    
    # Extract and tokenize text from 'review_text' tags
    text_data = []
    for review in soup.find_all('review_text'):
        txt = review.get_text().lower()
        lst_txt = tokenizer.tokenize(txt)
        text_data.append(lst_txt)
    
    # Initialize dictionaries for triples and counts
    triple_dictionary = {}
    count_dictionary = {}
    
    # Populate dictionaries with word triples and their counts
    for tokens in text_data:
        for index, token in enumerate(tokens):
            if 0 < index < len(tokens) - 1:
                prev_token = tokens[index - 1]
                next_token = tokens[index + 1]
                if token not in triple_dictionary:
                    triple_dictionary[token] = [[prev_token, next_token]]
                    count_dictionary[token] = 1
                else:
                    triple_dictionary[token].append([prev_token, next_token])
                    count_dictionary[token] += 1
    
    # Initialize dictionaries for probabilities and tuple counts
    probability_dict = {}
    tuple_count_dict = {}
    
    # Populate dictionaries with tuple probabilities
    for key, value_list in triple_dictionary.items():
        for value in value_list:
            value.sort()
            value_tuple = tuple(value)
            if value_tuple not in probability_dict:
                probability_dict[value_tuple] = {key: 1}
                tuple_count_dict[value_tuple] = 1
            else:
                if key not in probability_dict[value_tuple]:
                    probability_dict[value_tuple][key] = 1
                else:
                    probability_dict[value_tuple][key] += 1
                tuple_count_dict[value_tuple] += 1
    
    # Calculate probabilities for each tuple
    for value_tuple, word_dict in probability_dict.items():
        for word in word_dict:
            word_dict[word] = word_dict[word] / tuple_count_dict[value_tuple]
    
    return probability_dict

# Example usage
probs_dict = html_to_probs_dict('electronics/positive.review')
print(probs_dict[('i', 'this')]['bought'])
