"""model.py - Module to store the training and testing models."""

import numpy as np
from preprocess import read_images, create_data_matrix
from subspace import pca, knn, predict, accuracy

class Model:
    """Base class to define a data model for image processing."""

    def __init__(self, path, K):
        """
        Initialize the model by preprocessing images.

        Args:
            path (str): Path to the images.
            K (list): List of ranks for PCA.
        """
        self.path = path
        self.images, self.labels = read_images(self.path)
        self.matrix = create_data_matrix(self.images)
        self.K = K


class TrainingModel(Model):
    """Child class to define a training data model."""

    def __init__(self, path, K):
        """
        Initialize the training model by inheriting from the base class.

        Args:
            path (str): Path to the training images.
            K (list): List of ranks for PCA.
        """
        super().__init__(path, K)


class TestingModel(Model):
    """Child class to define a testing data model."""

    def __init__(self, path, K):
        """
        Initialize the testing model by inheriting from the base class.

        Args:
            path (str): Path to the testing images.
            K (list): List of ranks for PCA.
        """
        super().__init__(path, K)

    def reduce(self, training_matrix):
        """
        Perform PCA and dimensional reduction for all ranks in K.

        Args:
            training_matrix (np.ndarray): Data matrix of the training set.

        Returns:
            list: List of tuples containing the reduced matrices for each rank.
        """
        return [pca(training_matrix, self.matrix, rank) for rank in self.K]

    def knn(self, reduced, labels):
        """
        Return the KNN classifier for all ranks in K.

        Args:
            reduced (list): List of reduced matrices from PCA.
            labels (np.ndarray): Labels of the training set.

        Returns:
            list: List of KNN classifiers for each rank.
        """
        return [knn(reduced[i][0], labels) for i in range(len(self.K))]

    def predict(self, knn, reduced):
        """
        Return the predicted subject labels for all ranks in K.

        Args:
            knn (list): List of KNN classifiers for each rank.
            reduced (list): List of reduced matrices from PCA.

        Returns:
            list: List of predicted labels for each rank.
        """
        return [predict(knn[i], self.K[i], reduced[i][1], self.labels) for i in range(len(self.K))]

    def accuracy(self, knn, reduced):
        """
        Return the accuracy of each prediction for all ranks in K.

        Args:
            knn (list): List of KNN classifiers for each rank.
            reduced (list): List of reduced matrices from PCA.

        Returns:
            list: List of accuracy scores for each rank.
        """
        return [accuracy(knn[i], self.K[i], reduced[i][1], self.labels) for i in range(len(self.K))]
