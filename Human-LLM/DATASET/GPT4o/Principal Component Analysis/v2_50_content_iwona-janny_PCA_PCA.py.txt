import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
import matplotlib.pyplot as plt

# Load the dataset and rename columns
data = pd.read_csv("Wine.csv", header=1)
data.columns = ["Alkohol", "Kwas jabłkowy", "Popiół", "Zasadowość popiołu", "Magnez", "Całkowita zawartość fenoli", "Flawonoidy", "Fenole nieflawonoidowe", "Proantocyjaniny", "Intensywność koloru", "Odcień", "Transmitacja", "Prolina", "Etykieta"]

# Print the first few rows of the dataset
print(data.head())

# Split data into features (X) and labels (y)
X = data.iloc[:, :-1].values
y = data.iloc[:, -1].values

# Split the dataset into training and test sets (70% train, 30% test)
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=42)

# Standardize the features
scaler = StandardScaler()
X_train_stand = scaler.fit_transform(X_train)
X_test_stand = scaler.transform(X_test)

# Compute the covariance matrix of the standardized training data
cov_matrix = np.cov(X_train_stand.T)

# Compute eigenvalues and eigenvectors of the covariance matrix
eigenvalues, eigenvectors = np.linalg.eig(cov_matrix)

# Calculate the explained variance
total_variance = sum(eigenvalues)
variance_explained = [(i / total_variance) for i in sorted(eigenvalues, reverse=True)]
cumulative_variance_explained = np.cumsum(variance_explained)

# Sort eigenvalues and eigenvectors
eigen_pairs = [(np.abs(eigenvalues[i]), eigenvectors[:, i]) for i in range(len(eigenvalues))]
eigen_pairs.sort(key=lambda x: x[0], reverse=True)

# Create a projection matrix from the top 2 eigenvectors
projection_matrix = np.hstack((eigen_pairs[0][1][:, np.newaxis], eigen_pairs[1][1][:, np.newaxis]))

# Project the standardized training data onto the new 2D feature space
PCA_transformed = X_train_stand.dot(projection_matrix)

# Plot the PCA results
targets = np.unique(y_train)
colors = ["r", "g", "b"]
markers = ["s", "x", "o"]

plt.figure(figsize=(10, 6))
for target, color, marker in zip(targets, colors, markers):
    plt.scatter(PCA_transformed[y_train == target, 0], PCA_transformed[y_train == target, 1], color=color, label=target, marker=marker)

plt.xlabel("Principal Component 1 (PC1)")
plt.ylabel("Principal Component 2 (PC2)")
plt.legend(loc="lower right")
plt.title("PCA of Wine Dataset")
plt.show()
