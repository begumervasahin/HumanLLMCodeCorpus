import numpy as np
from numpy import linalg as LA

def compute_Z(X, centering=True, scaling=True):
    """
    Computes the standardized data matrix Z from input data matrix X.

    Parameters:
    X (numpy.ndarray): The input data matrix.
    centering (bool): If True, center the data by subtracting the mean.
    scaling (bool): If True, scale the data by dividing by the standard deviation.

    Returns:
    numpy.ndarray: The standardized data matrix Z.
    """
    Z = X.astype(float)  # Convert to float for numerical operations
    rows, cols = Z.shape

    if centering:
        means = X.mean(axis=0)  # Compute the mean of each column
        Z -= means  # Center the data

    if scaling:
        std = np.std(Z, axis=0)  # Compute the standard deviation of each column
        Z /= std  # Scale the data

    return Z

def compute_covariance_matrix(Z):
    """
    Computes the covariance matrix of the standardized data matrix Z.

    Parameters:
    Z (numpy.ndarray): The standardized data matrix.

    Returns:
    numpy.ndarray: The covariance matrix.
    """
    return Z.T @ Z  # Compute the covariance matrix

def find_pcs(COV):
    """
    Finds the principal components (eigenvalues and eigenvectors) of the covariance matrix.

    Parameters:
    COV (numpy.ndarray): The covariance matrix.

    Returns:
    tuple: Eigenvalues and eigenvectors sorted in descending order of eigenvalues.
    """
    eigenValues, eigenVectors = LA.eig(COV)
    idx = eigenValues.argsort()[::-1]  # Sort eigenvalues in descending order
    eigenValues = eigenValues[idx]
    eigenVectors = eigenVectors[:, idx]
    return eigenValues, eigenVectors

def project_data(Z, PCS, L, k=0, var=0):
    """
    Projects the standardized data onto the principal components.

    Parameters:
    Z (numpy.ndarray): The standardized data matrix.
    PCS (numpy.ndarray): The matrix of principal component vectors.
    L (numpy.ndarray): The eigenvalues.
    k (int): The number of principal components to keep (if non-zero).
    var (float): The fraction of total variance to retain (if k is zero).

    Returns:
    numpy.ndarray: The projected data matrix Z*.
    """
    if k != 0:
        # Keep the first k principal components
        component_matrix = PCS[:, :k]
    else:
        # Determine the number of components needed to retain the desired variance
        total_variance = np.sum(L)
        cumulative_variance = 0
        num_components = 0

        for eigenvalue in L:
            cumulative_variance += eigenvalue
            num_components += 1
            if cumulative_variance / total_variance >= var:
                break

        component_matrix = PCS[:, :num_components]

    Z_star = Z @ component_matrix  # Project the data
    return Z_star

# Example usage
if __name__ == "__main__":
    # Example data
    X = np.array([
        [2.5, 2.4],
        [0.5, 0.7],
        [2.2, 2.9],
        [1.9, 2.2],
        [3.1, 3.0],
        [2.3, 2.7],
        [2.0, 1.6],
        [1.0, 1.1],
        [1.5, 1.6],
        [1.1, 0.9]
    ])

    # Compute the Z matrix
    Z = compute_Z(X, centering=True, scaling=True)

    # Compute the covariance matrix
    COV = compute_covariance_matrix(Z)

    # Find the principal components
    eigenValues, eigenVectors = find_pcs(COV)

    # Project the data onto the principal components
    Z_star = project_data(Z, eigenVectors, eigenValues, k=1)

    print("Projected data:\n", Z_star)
