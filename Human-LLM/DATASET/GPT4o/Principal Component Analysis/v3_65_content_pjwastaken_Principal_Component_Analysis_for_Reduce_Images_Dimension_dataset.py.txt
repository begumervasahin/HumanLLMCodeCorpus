import os

class DatasetClass:
    def __init__(self, required_no):
        # Initialize the dataset directory
        dataset_name = "ORL"
        dir_path = os.path.join("images", dataset_name)
        
        # Initialize lists to store image paths and labels
        self.image_path_for_training = []
        self.labels_for_training = []
        self.no_of_images_for_training = []
        self.image_path_for_testing = []
        self.labels_for_testing = []
        self.no_of_images_for_testing = []
        self.images_target = []
        per_no = 0

        # Iterate through each person directory in the dataset
        for person_name in os.listdir(dir_path):
            person_dir = os.path.join(dir_path, person_name)
            
            # Check if it is a directory and has the required number of images
            if os.path.isdir(person_dir) and len(os.listdir(person_dir)) >= required_no:
                self._process_person_directory(person_dir, person_name, per_no, required_no)
                per_no += 1

    def _process_person_directory(self, person_dir, person_name, person_id, required_no):
        """Process images in a person's directory."""
        training_count = 0
        testing_count = 0

        for i, img_name in enumerate(os.listdir(person_dir)):
            img_path = os.path.join(person_dir, img_name)
            
            if i < required_no:
                self.image_path_for_training.append(img_path)
                self.labels_for_training.append(person_id)
                training_count += 1

                if i == 0:
                    self.images_target.append(person_name)
            else:
                self.image_path_for_testing.append(img_path)
                self.labels_for_testing.append(person_id)
                testing_count += 1

        self._update_image_counts(training_count, testing_count, person_id)

    def _update_image_counts(self, training_count, testing_count, person_id):
        """Update the count of images for training and testing sets."""
        self._update_count_list(self.no_of_images_for_training, training_count, person_id)
        self._update_count_list(self.no_of_images_for_testing, testing_count, person_id)

    def _update_count_list(self, count_list, count, person_id):
        """Update the count list for the specified person."""
        if len(count_list) > person_id:
            count_list[person_id] += count
        else:
            count_list.append(count)
