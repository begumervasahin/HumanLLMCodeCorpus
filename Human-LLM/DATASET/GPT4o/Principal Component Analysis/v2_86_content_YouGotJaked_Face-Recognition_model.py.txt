import numpy as np
from preprocess import read_images, create_data_matrix
from subspace import pca, knn, predict, accuracy

class Model:
    """Base class to define a data model."""
    def __init__(self, path, K):
        """
        Initialize the Model with image data and labels.
        
        Args:
            path (str): Path to the dataset.
            K (list): List of ranks for dimensionality reduction.
        """
        self.path = path
        self.images, self.labels = read_images(self.path)
        self.matrix = create_data_matrix(self.images)
        self.K = K

class TrainingModel(Model):
    """Model for training data."""
    def __init__(self, path, K):
        """
        Initialize the TrainingModel by inheriting from Model.
        
        Args:
            path (str): Path to the training dataset.
            K (list): List of ranks for dimensionality reduction.
        """
        super().__init__(path, K)

class TestingModel(Model):
    """Model for testing data."""
    def __init__(self, path, K):
        """
        Initialize the TestingModel by inheriting from Model.
        
        Args:
            path (str): Path to the testing dataset.
            K (list): List of ranks for dimensionality reduction.
        """
        super().__init__(path, K)

    def reduce(self, training_matrix):
        """
        Perform PCA and dimensional reduction for all ranks in K.
        
        Args:
            training_matrix (ndarray): Training data matrix.
        
        Returns:
            list: List of reduced data for each rank in K.
        """
        return [pca(training_matrix, self.matrix, rank) for rank in self.K]

    def knn(self, reduced, labels):
        """
        Train KNN classifiers for all ranks in K.
        
        Args:
            reduced (list): List of reduced data for each rank.
            labels (ndarray): Labels of the training data.
        
        Returns:
            list: List of KNN classifiers for each rank in K.
        """
        return [knn(reduced[i][0], labels) for i in range(len(self.K))]

    def predict(self, knn_models, reduced):
        """
        Predict labels using KNN classifiers for all ranks in K.
        
        Args:
            knn_models (list): List of KNN classifiers.
            reduced (list): List of reduced data for each rank.
        
        Returns:
            list: List of predicted labels for each rank in K.
        """
        return [predict(knn_models[i], self.K[i], reduced[i][1], self.labels) for i in range(len(self.K))]

    def accuracy(self, knn_models, reduced):
        """
        Calculate the accuracy of predictions for all ranks in K.
        
        Args:
            knn_models (list): List of KNN classifiers.
            reduced (list): List of reduced data for each rank.
        
        Returns:
            list: List of accuracy scores for each rank in K.
        """
        return [accuracy(knn_models[i], self.K[i], reduced[i][1], self.labels) for i in range(len(self.K))]

# Example usage
if __name__ == "__main__":
    # Paths to the training and testing datasets
    training_path = 'path/to/training/data'
    testing_path = 'path/to/testing/data'
    
    # Example ranks for PCA
    K_values = [10, 20, 30]

    # Initialize training and testing models
    training_model = TrainingModel(training_path, K_values)
    testing_model = TestingModel(testing_path, K_values)

    # Perform dimensional reduction on testing data
    reduced_data = testing_model.reduce(training_model.matrix)

    # Train KNN classifiers on the reduced data
    knn_models = testing_model.knn(reduced_data, training_model.labels)

    # Predict labels and evaluate accuracy
    predictions = testing_model.predict(knn_models, reduced_data)
    accuracies = testing_model.accuracy(knn_models, reduced_data)

    # Output the results
    print("Predictions:", predictions)
    print("Accuracies:", accuracies)
