import numpy as np
from numpy.linalg import eig
from scipy.stats import chi2

def gsis(snp_mat, qr_smy_mat, proj_mat):
    """
    Global sure independence screening (GSIS) procedure.
    
    Parameters:
        snp_mat (np.ndarray): SNP data matrix of shape (n x g), where n is the number of samples and g is the number of SNPs.
        qr_smy_mat (np.ndarray): Common part in the global test statistic of shape (n x n).
        proj_mat (np.ndarray): Projection matrix of shape (n x n).
    
    Returns:
        g_pv_log10 (np.ndarray): -log10 p-values across all SNPs.
        g_stat (np.ndarray): The global Wald test statistics across all SNPs.
    """
    # Calculate ZX matrix
    zx_mat = np.dot(proj_mat, snp_mat).T

    # Calculate Q(ZX)
    q_zx = np.sum(zx_mat * zx_mat, axis=1)
    
    # Avoid division by zero by adding a small value to zero elements
    q_zx[q_zx == 0] += 1e-6
    
    # Compute the inverse of Q(ZX)
    inv_q_zx = 1.0 / q_zx
    
    # Eigen decomposition of QR summary matrix
    eigenvalues, eigenvectors = eig(qr_smy_mat)
    
    # Ensure eigenvalues are non-negative and take the square root
    eigenvalues = np.real(eigenvalues)
    eigenvalues[eigenvalues < 0] = 0
    sqrt_eigenvalues = np.sqrt(eigenvalues)
    
    # Compute the square root of the QR summary matrix
    sqrt_qr_smy_mat = np.dot(np.dot(eigenvectors, np.diag(sqrt_eigenvalues)), eigenvectors.T)
    sqrt_qr_smy_mat = np.real(sqrt_qr_smy_mat)
    
    # Calculate global Wald test statistics
    g_stat = np.sum(np.dot(zx_mat, sqrt_qr_smy_mat) ** 2, axis=1) * inv_q_zx
    
    # Compute moments for the chi-squared distribution
    mean_g_stat = np.mean(g_stat)
    var_g_stat = np.var(g_stat)
    skew_g_stat = np.mean((g_stat - mean_g_stat) ** 3)
    
    # Calculate parameters for chi-squared distribution
    a = skew_g_stat / (4 * var_g_stat)
    b = mean_g_stat - (2 * var_g_stat ** 2 / skew_g_stat)
    d = (8 * var_g_stat ** 3) / (skew_g_stat ** 2)
    
    # Calculate p-values and their -log10 values
    g_pv = 1 - chi2.cdf((g_stat - b) / a, d)
    g_pv_log10 = -np.log10(g_pv)
    
    return g_pv_log10, g_stat

def main():
    # Generate example data for demonstration purposes
    n = 100  # Number of samples
    g = 10   # Number of SNPs
    snp_mat = np.random.rand(n, g)
    qr_smy_mat = np.random.rand(n, n)
    proj_mat = np.random.rand(n, n)
    
    # Perform GSIS procedure
    g_pv_log10, g_stat = gsis(snp_mat, qr_smy_mat, proj_mat)
    
    # Print results
    print("g_pv_log10:", g_pv_log10)
    print("g_stat:", g_stat)

if __name__ == "__main__":
    main()
