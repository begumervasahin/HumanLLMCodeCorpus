import numpy as np
import matplotlib.pyplot as plt
from matplotlib.backends.backend_pdf import PdfPages
import sys

# Check for correct number of arguments
if len(sys.argv) != 5:
    print(sys.argv[0], "takes 4 arguments. Not ", len(sys.argv) - 1)
    sys.exit()

# Parse input arguments
first = sys.argv[1]
second = sys.argv[2]
third = sys.argv[3]
four = sys.argv[4]

# Load data
Xt = np.genfromtxt(first, delimiter=',')
y = np.genfromtxt(second, autostrip=True)
m, n = Xt.shape
unique_labels = set(y)

# Initialize global mean and scatter matrices
mu_global = np.mean(Xt, axis=0)
W_class = np.zeros((n, n))
B_class = np.zeros((n, n))

# Calculate within-class scatter and between-class scatter
for c in unique_labels:
    B1 = np.zeros((n, n))
    BT1 = np.zeros((n, n))
    s1 = np.zeros((1, n))
    C1 = np.zeros((n, n))
    m1 = 0
    for i, xt in enumerate(Xt):
        if y[i] == c:
            B1 += np.outer(xt, xt)
            s1 += xt
            m1 += 1
    mu1 = s1 / m1
    for i, xt in enumerate(Xt):
        if y[i] == c:
            ct = xt - mu1
            bt = mu1 - mu_global
            C1 += np.outer(ct, ct)
            BT1 += np.outer(bt, bt)
    W_class += C1
    B_class += BT1

# Solve the eigenvalue problem
x = np.dot(np.linalg.inv(W_class), B_class)
evals, evecs = np.linalg.eig(x)
r = 2
V_r = evecs[:, :r]
result = np.dot(Xt, V_r)
x1 = result[:, 0]
x2 = result[:, 1]

# Plot the result
fig, ax = plt.subplots()
for label in unique_labels:
    ax.scatter(x1[y == label], x2[y == label], label=f"Class {label}")
plt.title("Maximize the ratio of between-class scatter to within-class scatter")
plt.xlabel("PC1")
plt.ylabel("PC2")
plt.legend()
plotname = 'scatter_3_plot'
pdf = PdfPages(plotname + '.pdf')
pdf.savefig(fig)
pdf.close()

# Save the results
np.savetxt(third, V_r.T, delimiter=',')
np.savetxt(four, result, delimiter=',')
