import sys
import pickle
import time
import numpy as np
from scipy.io import loadmat
from S3_BSTP import wild_bstp

def main(LorR, i):
    # Define directories
    input_dir = 'data/'
    interm_dir = f'res/{LorR}vars/'
    bstp_dir = f'res/{LorR}bstp/'

    # Load image data
    y_design = loadmat(input_dir + f"img_data_{LorR}.mat")['img_data']
    
    # Ensure y_design has 3 dimensions
    if y_design.ndim == 2:
        y_design = y_design[np.newaxis, :, :]

    # Initialize projected image data array
    proj_y_design = np.zeros_like(y_design)
    m = y_design.shape[0]

    # Load projection matrix
    with open(interm_dir + 'proj_mat.dat', 'rb') as f:
        proj_mat = pickle.load(f)

    # Project image data using the projection matrix
    for mii in range(m):
        proj_y_design[mii, :, :] = np.dot(proj_mat, np.squeeze(y_design[mii, :, :]))

    print(f'The matrix dimension of image data is {y_design.shape}')

    # Load additional data
    img_size = np.loadtxt(input_dir + 'img_size.txt').astype(int)
    img_idx = (np.loadtxt(input_dir + 'img_idx.txt').astype(int) - 1).astype(int)
    snp = np.loadtxt(input_dir + 'snp_data.txt')

    with open(interm_dir + 'efit_eta.dat', 'rb') as f:
        efit_eta = pickle.load(f)

    with open(interm_dir + 'coord_data.dat', 'rb') as f:
        coord_data = pickle.load(f)

    with open(interm_dir + 'h_opt.dat', 'rb') as f:
        h_opt = pickle.load(f)

    # Set parameters for the bootstrap procedure
    alpha = 0.005
    alpha_log10 = -np.log10(alpha)
    b_num = 25
    g_num = 2000

    # Perform wild bootstrap
    start_time = time.time()
    max_gstat_bstp, max_lstat_bstp, max_area_bstp = wild_bstp(snp, proj_y_design, efit_eta, proj_mat, 
                                                              coord_data, h_opt, img_size, img_idx, 
                                                              alpha_log10, g_num, b_num)
    end_time = time.time()

    print(f'Elapsed time in wild_bstp is {end_time - start_time} seconds')

    # Save the bootstrap results
    np.savetxt(bstp_dir + f'max_gstat_bstp_{i}', max_gstat_bstp)
    np.savetxt(bstp_dir + f'max_lstat_bstp_{i}', max_lstat_bstp)
    np.savetxt(bstp_dir + f'max_area_bstp_{i}', max_area_bstp)

if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("Usage: python script.py <LorR> <i>")
    else:
        main(sys.argv[1], sys.argv[2])
