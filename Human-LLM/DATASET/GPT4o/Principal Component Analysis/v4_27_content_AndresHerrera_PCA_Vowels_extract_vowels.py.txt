import os
import cv2
import numpy as np

# Define constants
w = 20
h = 20
min_contour_area = 100

def main():
    # Load the image
    imgVowels = cv2.imread("raw_vowels.jpg")
    
    # Convert the image to grayscale
    imgGray = cv2.cvtColor(imgVowels, cv2.COLOR_BGR2GRAY)
    
    # Apply Gaussian blur to the grayscale image
    imgBlurred = cv2.GaussianBlur(imgGray, (5, 5), 0)
    
    # Apply adaptive thresholding to the blurred image
    imgThresh = cv2.adaptiveThreshold(imgBlurred, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C, cv2.THRESH_BINARY_INV, 11, 2)
    
    # Display the threshold image
    cv2.imshow("Threshold Image", imgThresh)
    
    # Copy the threshold image for contour detection
    imgThreshCopy = imgThresh.copy()
    
    # Find contours in the threshold image
    contours, hierarchy = cv2.findContours(imgThreshCopy, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    
    # Initialize a counter for found vowels
    found = 0
    
    # Loop through each contour
    for contour in contours:
        # Check if the contour area is larger than the minimum contour area
        if cv2.contourArea(contour) > min_contour_area:
            # Get the bounding rectangle for the contour
            intX, intY, intW, intH = cv2.boundingRect(contour)
            
            # Draw a rectangle around the detected contour on the original image
            cv2.rectangle(imgVowels, (intX, intY), (intX + intW, intY + intH), (0, 0, 255), 2)
            
            # Extract the region of interest (ROI) from the threshold image
            imgROI = imgThresh[intY:intY + intH, intX:intX + intW]
            
            # Resize the ROI to the defined width and height
            imgROIResized = cv2.resize(imgROI, (w, h))
            
            # Display the original image with rectangles
            cv2.imshow("Original File", imgVowels)
            
            # Increment the found counter
            found += 1
            
            # Save the resized ROI as an image file
            cv2.imwrite(f'vowels/vowel_{found}.png', cv2.bitwise_not(imgROIResized))
    
    # Print the number of files written and instructions for organizing them
    print(f"{found} files written into (vowels) folder!")
    print("Notice: You have to arrange (vowels) files into (training) folder tree!")
    print("in folder :  training/A  <-  store only A images")
    print("in folder :  training/E  <-  store only E images")
    print("in folder :  training/I  <-  store only I images")
    print("in folder :  training/O  <-  store only O images")
    print("in folder :  training/U  <-  store only U images")
    
    # Wait for a key press and close all OpenCV windows
    cv2.waitKey(0)
    cv2.destroyAllWindows()

if __name__ == "__main__":
    main()
