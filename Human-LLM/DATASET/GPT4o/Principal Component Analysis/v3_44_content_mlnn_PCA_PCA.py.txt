import numpy as np
import matplotlib.pyplot as plt
from sklearn.decomposition import PCA, KernelPCA
from sklearn.datasets import make_circles

# Set random seed for reproducibility
np.random.seed(0)

# Generate synthetic data
X, y = make_circles(n_samples=400, factor=0.3, noise=0.05)

# Apply Kernel PCA with RBF kernel
kpca = KernelPCA(kernel="rbf", fit_inverse_transform=True, gamma=10)
X_kpca = kpca.fit_transform(X)
X_back = kpca.inverse_transform(X_kpca)

# Apply PCA
pca = PCA()
X_pca = pca.fit_transform(X)

# Plotting
def plot_original_space(ax, X, y):
    ax.set_title("Original space")
    reds = y == 0
    blues = y == 1
    ax.plot(X[reds, 0], X[reds, 1], "ro")
    ax.plot(X[blues, 0], X[blues, 1], "bo")
    ax.set_xlabel("$x_1$")
    ax.set_ylabel("$x_2$")

    # Generate a grid for contour plot
    X1, X2 = np.meshgrid(np.linspace(-1.5, 1.5, 50), np.linspace(-1.5, 1.5, 50))
    X_grid = np.array([X1.ravel(), X2.ravel()]).T
    Z_grid = kpca.transform(X_grid)[:, 0].reshape(X1.shape)
    ax.contour(X1, X2, Z_grid, colors='grey', linewidths=1, origin='lower')

def plot_pca_projection(ax, X_pca, y):
    ax.set_title("Projection by PCA")
    reds = y == 0
    blues = y == 1
    ax.plot(X_pca[reds, 0], X_pca[reds, 1], "ro")
    ax.plot(X_pca[blues, 0], X_pca[blues, 1], "bo")
    ax.set_xlabel("1st principal component")
    ax.set_ylabel("2nd component")

def plot_kpca_projection(ax, X_kpca, y):
    ax.set_title("Projection by KPCA")
    reds = y == 0
    blues = y == 1
    ax.plot(X_kpca[reds, 0], X_kpca[reds, 1], "ro")
    ax.plot(X_kpca[blues, 0], X_kpca[blues, 1], "bo")
    ax.set_xlabel("1st principal component in space induced by $\phi$")
    ax.set_ylabel("2nd component")

def plot_inverse_transform(ax, X_back, y):
    ax.set_title("Original space after inverse transform")
    reds = y == 0
    blues = y == 1
    ax.plot(X_back[reds, 0], X_back[reds, 1], "ro")
    ax.plot(X_back[blues, 0], X_back[blues, 1], "bo")
    ax.set_xlabel("$x_1$")
    ax.set_ylabel("$x_2$")

# Create subplots
fig, axes = plt.subplots(2, 2, figsize=(12, 12), subplot_kw={'aspect': 'equal'})

# Plot each subplot
plot_original_space(axes[0, 0], X, y)
plot_pca_projection(axes[0, 1], X_pca, y)
plot_kpca_projection(axes[1, 0], X_kpca, y)
plot_inverse_transform(axes[1, 1], X_back, y)

# Adjust layout and show plot
plt.subplots_adjust(0.02, 0.10, 0.98, 0.94, 0.04, 0.35)
plt.show()
