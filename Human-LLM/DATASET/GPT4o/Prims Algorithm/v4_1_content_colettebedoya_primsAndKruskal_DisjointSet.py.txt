class DisjointSet:
    def __init__(self, size):
        """
        Constructor to initialize a disjoint set of the specified size.
        :param size: Number of elements in the disjoint set.
        """
        if size < 0:
            raise ValueError("Size must be >= 0")
        self.size = size
        self.parent = [-1] * (size + 1)

    def union(self, root1, root2):
        """
        Perform a union of two sets identified by root1 and root2.
        :param root1: The root of the first set.
        :param root2: The root of the second set.
        """
        if not (0 <= root1 <= self.size) or not (0 <= root2 <= self.size):
            raise ValueError("Illegal value")
        self.parent[root2] = root1

    def find(self, root):
        """
        Find the root of the set containing the element.
        :param root: The element to find the root of.
        :return: The root of the set containing the element.
        """
        if not (0 <= root <= self.size):
            raise ValueError("Illegal value")
        if self.parent[root] < 0:
            return root
        else:
            self.parent[root] = self.find(self.parent[root])  # Path compression
            return self.parent[root]
