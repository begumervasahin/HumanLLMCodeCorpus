class PriorityQueue:
    def __init__(self):
        self.queue = [(0, 0)]  # Initialize the queue with a dummy element at index 0
        self.size = 0

    def get_size(self):
        """Return the size of the priority queue."""
        return self.size

    def find_min(self):
        """Find the minimum element in the priority queue."""
        if self.size > 0:
            return self.queue[1]
        return None

    def remove_min(self):
        """Remove and return the minimum element from the priority queue."""
        min_element = self.find_min()
        if min_element:
            self.queue[1] = self.queue[self.size]
            self.size -= 1
            self.queue.pop()
            self._fix_down(1)
        return min_element

    def insert(self, element):
        """Insert a new element into the priority queue."""
        self.queue.append(element)
        self.size += 1
        self._fix_up(self.size)

    def increase_key(self, node, value):
        """Increase the key value of a specific node."""
        for i in range(1, self.size + 1):
            if self.queue[i][1] == node:
                self.queue[i] = (value, node)
                self._fix_up(i)
                break

    def __contains__(self, element):
        """Check if an element is in the priority queue."""
        return element in self.queue

    def _fix_down(self, index):
        """Maintain the heap property by moving an element down the tree."""
        while 2 * index <= self.size:
            child = 2 * index
            if child < self.size and self.queue[child][0] > self.queue[child + 1][0]:
                child += 1
            if self.queue[index][0] <= self.queue[child][0]:
                break
            self.queue[index], self.queue[child] = self.queue[child], self.queue[index]
            index = child

    def build_min_heap(self, elements):
        """Build a min-heap from a list of elements."""
        self.queue = [(0, 0)] + elements
        self.size = len(elements)
        for i in range(self.size // 2, 0, -1):
            self._fix_down(i)

    def _fix_up(self, index):
        """Maintain the heap property by moving an element up the tree."""
        while index > 1 and self.queue[index][0] < self.queue[index // 2][0]:
            self.queue[index], self.queue[index // 2] = self.queue[index // 2], self.queue[index]
            index //= 2
