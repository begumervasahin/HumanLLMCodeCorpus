import random

def construct_random_graph(n):
    # Initialize vertex sets
    v_set = [[] for _ in range(n)]
    
    # Create random connections and weights between nodes
    for i in range(2, n):
        x = random.randint(1, i - 1)
        for j in range(x):
            node_connect = random.randint(0, n - 1)
            weight = random.randint(10, 100)
            v_set[i].append((node_connect, weight))
            v_set[node_connect].append((i, weight))
    
    # Initialize adjacency matrix with zeros
    matrix = [[0 for _ in range(n)] for _ in range(n)]
    
    # Fill the adjacency matrix with weights
    for i in range(n):
        for node_connect, matrix_weight in v_set[i]:
            matrix[i][node_connect] = matrix_weight
    
    return matrix

def bfs(G):
    # Choose a random start vertex
    v = random.randint(0, len(G) - 1)
    total = 0
    Q = []
    visited = [0] * len(G)
    
    visited[v] = 1
    Q.append(v)
    
    while Q:
        x = Q.pop(0)
        for y in range(len(G)):
            if G[x][y] > 0 and visited[y] == 0:
                visited[y] = 1
                Q.append(y)
                total += G[x][y]
    
    return total

def prim(G):
    total = 0
    start_vertex = random.randint(0, len(G) - 1)
    A = [["empty"] * len(G) for _ in range(3)]
    
    A[0][start_vertex] = "N"
    for i in range(len(G)):
        if i != start_vertex:
            A[0][i] = "Y"
            A[2][i] = 1000
    
    for i in range(len(G)):
        if G[start_vertex][i] > 0:
            A[1][i] = start_vertex
            A[2][i] = G[start_vertex][i]
    
    T = [start_vertex]
    while len(T) < len(G):
        min_weight = 1000
        for i in range(len(G)):
            if A[0][i] == "Y" and A[2][i] < min_weight:
                min_weight = A[2][i]
                x = i
        
        T.append(x)
        total += A[2][x]
        A[0][x] = "N"
        
        for y in range(len(G)):
            if G[x][y] > 0 and A[0][y] == "Y" and G[x][y] < A[2][y]:
                A[1][y] = x
                A[2][y] = G[x][y]
    
    return total

def experiment():
    n_values = [20, 30, 40, 50, 60]
    k = 1000
    
    for n in n_values:
        sum_diff = 0
        for _ in range(k):
            graph = construct_random_graph(n)
            bfs_total = bfs(graph)
            prim_total = prim(graph)
            diff = ((bfs_total / prim_total) - 1) * 100
            sum_diff += diff
        
        avg_diff = sum_diff / k
        print(f"Average of Diff for n = {n}: {avg_diff}")

experiment()
