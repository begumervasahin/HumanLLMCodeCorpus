import GenGraph
import Sollins
import Prims
import Kruskals
import time
import copy
from Node import Node

def driver():
    # Introduction to the program
    display_intro()
    
    # Choosing the edge generation method
    edge_method = get_edge_method()
    
    # Getting user inputs for graph generation
    total_nodes = get_positive_integer("How many nodes are in the graph (any number greater than 0)?")
    graph_size = get_positive_integer("How big is the graph (any number greater than 0)?")
    k_value = get_positive_integer("What is the k-value that should be used (any number greater than 0)?")

    max_weight = 0
    if edge_method == 2:
        max_weight = get_positive_integer("What is the maximum weight of an edge (any number greater than 0)?")

    # Generating the graph
    trees = GenGraph.GenerateGraph(total_nodes, graph_size, max_weight, k_value, edge_method)

    # Displaying the generated trees
    print("Here are the adjacency lists of the trees randomly generated using your inputs (format = ((x,y,) weight)):")
    print_trees(trees, "Tree")

    # Finding MSTs using different algorithms and measuring their runtime
    print("Here are the MSTs for each of the generated trees.")
    display_msts(trees)

def display_intro():
    print("Hello! This is a program which randomly generates nodes and edges to connect those nodes.")
    print("After the graph is generated, you can then choose from finding an MST of each tree created")
    print("by using either Prim's, Kruskal's, or Sollin's algorithm. Additionally, there are two methods")
    print("of generating edges. The first method is to randomly assign a weight between 1 and the user-defined")
    print("maximum edge weight, the second is to assign the weight based on the actual distance between")
    print("the two nodes. Finally, the amount of nodes in the graph, the size of the graph, and the k-value,")
    print("which determines the likelihood any two nodes will have an edge between them.")

def get_edge_method():
    print("Which method of edge generation would you like to use?")
    print("1: Weight based on actual distance between nodes")
    print("2: Weight based on user-defined maximum edge weight")
    return int(input('> '))

def get_positive_integer(prompt):
    while True:
        value = int(input(f"{prompt}\n> "))
        if value > 0:
            return value
        print("Please enter a number greater than 0.")

def display_msts(trees):
    # Prim's Algorithm
    print("Result of Prim's")
    prims_msts, prims_time = run_algorithm(Prims.runPrims, trees)
    print(f"Prim's Algorithm Run Time = {prims_time}")

    # Kruskal's Algorithm
    print("Result of Kruskal's")
    kruskals_msts, kruskals_time = run_algorithm(Kruskals.runKruskals, trees)
    print(f"Kruskal's Algorithm Run Time = {kruskals_time}")

    # Sollin's Algorithm
    print("Result of Sollin's")
    sollins_msts, sollins_time = run_algorithm(Sollins.runSollins, trees)
    print(f"Sollin's Algorithm Run Time = {sollins_time}")

def run_algorithm(algorithm, trees):
    trees_copy = copy.deepcopy(trees)
    start_time = time.time()
    msts = algorithm(trees_copy)
    elapsed_time = time.time() - start_time
    print_trees(msts, "MST")
    return msts, elapsed_time

def print_trees(trees, term):
    # Function to print the adjacency list of trees
    print(f"{term}s:")
    for t in trees:
        print(f"{term}:{{")
        for x in t:
            print(f"{print_node(x)}: {{", end=" ")
            for y in x.adjList:
                print(f"({print_node(y)}, {round(x.adjList[y], 2)})", end=" ")
            print("}")
        print("}")

def print_node(n):
    # Function to print the coordinates of a node
    return f"({n.xloc},{n.yloc})"

# Running the driver function
driver()
