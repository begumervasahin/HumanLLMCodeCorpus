class Set:
    """
    Union-Find Data Structure with path compression and union by rank.
    """

    def __init__(self, value):
        """
        Initialize a set containing a single element.
        """
        self.value = value
        self.parent = self
        self.next = self
        self.rank = 0

    def findSet(self):
        """
        Find the representative element (root) of the set containing this element.
        Apply path compression to speed up future queries.
        """
        elements_to_compress = []
        current = self

        while current is not current.parent:
            elements_to_compress.append(current)
            current = current.parent

        root = current
        for element in elements_to_compress:
            element.parent = root

        return root

    def union(self, other):
        """
        Merge this set with another set, using union by rank to keep the tree shallow.
        """
        root1 = self.findSet()
        root2 = other.findSet()

        if root1 is not root2:
            if root1.rank < root2.rank:
                root1.parent = root2
            else:
                root2.parent = root1
                if root1.rank == root2.rank:
                    root1.rank += 1

            # Maintain the circular linked list structure
            root1.next, root2.next = root2.next, root1.next

    def add(self, value):
        """
        Add a new element to the set.
        """
        new_set = Set(value)
        self.union(new_set)

    def getElements(self):
        """
        Generator to iterate over all elements in the set.
        """
        yield self.value
        current = self.next
        while current != self:
            yield current.value
            current = current.next

    def __str__(self):
        """
        Return a string representation of the set.
        """
        return "{{{}}}".format(", ".join(map(str, self.getElements())))

    def __repr__(self):
        """
        Return a string representation of the set.
        """
        return str(self)

# Example usage
if __name__ == "__main__":
    # Create sets with initial values
    set1 = Set(1)
    set2 = Set(2)
    set3 = Set(3)

    # Union sets
    set1.union(set2)
    set2.union(set3)

    # Add an element
    set1.add(4)

    # Print sets
    print(set1)  # Output: {1, 2, 3, 4}
    print(set2)  # Output: {1, 2, 3, 4}
    print(set3)  # Output: {1, 2, 3, 4}
