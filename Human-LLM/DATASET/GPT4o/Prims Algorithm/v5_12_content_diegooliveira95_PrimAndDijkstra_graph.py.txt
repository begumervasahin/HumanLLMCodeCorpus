import sys

class Vertex:
    def __init__(self, node):
        self.id = node
        self.adjacent = {}
        self.distance = sys.maxsize
        self.visited = False
        self.previous = None

    def add_neighbor(self, neighbor, weight=0):
        """Add a neighbor along with the edge weight."""
        self.adjacent[neighbor] = weight

    def get_connections(self):
        """Return all neighbors of this vertex."""
        return self.adjacent.keys()

    def get_id(self):
        """Return the ID of this vertex."""
        return self.id

    def get_weight(self, neighbor):
        """Return the weight of the edge connecting this vertex to the neighbor."""
        return self.adjacent[neighbor]

    def set_distance(self, distance):
        """Set the distance from the source."""
        self.distance = distance

    def get_distance(self):
        """Return the distance from the source."""
        return self.distance

    def set_previous(self, previous):
        """Set the previous vertex in the shortest path."""
        self.previous = previous

    def set_visited(self):
        """Mark this vertex as visited."""
        self.visited = True

    def __str__(self):
        """Return the string representation of this vertex and its neighbors."""
        return f'{self.id} adjacent(s): {[x.id for x in self.adjacent]}'

class Graph:
    def __init__(self):
        self.vertex_dict = {}
        self.num_vertices = 0

    def __iter__(self):
        """Iterate over the vertices in the graph."""
        return iter(self.vertex_dict.values())

    def add_vertex(self, node):
        """Add a new vertex to the graph."""
        self.num_vertices += 1
        new_vertex = Vertex(node)
        self.vertex_dict[node] = new_vertex
        return new_vertex

    def get_vertex(self, node):
        """Return the vertex if it exists."""
        return self.vertex_dict.get(node)

    def add_edge(self, from_node, to_node, cost=0):
        """Add an edge between two vertices with a given cost."""
        if from_node not in self.vertex_dict or to_node not in self.vertex_dict:
            return 'One or both nodes are not in the graph'
        
        self.vertex_dict[from_node].add_neighbor(self.vertex_dict[to_node], cost)
        self.vertex_dict[to_node].add_neighbor(self.vertex_dict[from_node], cost)

    def get_vertices(self):
        """Return all the vertices in the graph."""
        return self.vertex_dict.keys()

    def set_previous(self, current):
        """Set the previous vertex (for pathfinding)."""
        self.previous = current

    def get_previous(self, current):
        """Get the previous vertex (for pathfinding)."""
        return self.previous
