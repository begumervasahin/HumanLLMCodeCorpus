import prims
import kruskals
import glb
import sys

def test_read(file, delimiter, algo):
    """
    Tests the read_graph function of the specified algorithm.
    """
    if algo == 'k':
        glb.WGRAPH = kruskals.read_graph(file, delimiter)
        keys = glb.VERTICES
    elif algo == 'p':
        glb.WGRAPH = prims.read_graph(file, delimiter)
        keys = glb.WGRAPH.keys()
    else:
        usage()
        return False

    expected_keys = 3
    responded_keys = len(keys)
    
    if responded_keys != expected_keys:
        print("ERROR:\t read_graph() returned the wrong number of keys")
        print(f"Expected: {expected_keys}")
        print(f"Responded: {responded_keys}")
        return False

    if algo == 'p':
        for key in keys:
            if key in glb.WGRAPH[key]:
                print("ERROR:\t Key neighbors have the same values as Key")
                print(f"Expected: Key = {key}, graph[Key] != Key")
                print(f"Responded: {glb.WGRAPH[key]}")
                return False

    return True

def test_getmin():
    """
    Tests the get_min function of the Prim's algorithm.
    """
    glb.Vr = ['a']
    expected = ('a', 'b', '1')
    responded = prims.get_min()

    if expected == responded:
        return True
    else:
        print("ERROR:\t get_min returns the wrong minimum edge")
        print(f"Expected: {expected}")
        print(f"Responded: {responded}")
        return False

def test_algo(algo):
    """
    Tests the main algorithm function (Prim's or Kruskal's).
    """
    if algo == 'k':
        algo_name = 'kruskals'
        responded, cumulative_dist = kruskals.kruskals()
    elif algo == 'p':
        algo_name = 'prims'
        glb.Vr.clear()
        responded, cumulative_dist = prims.prims()
    else:
        usage()
        return False

    expected_mst = [('a', 'b', '1', '1'), ('a', 'c', '2', '3')]
    expected_cdist = 4

    pass_mst = (expected_mst == responded)
    if not pass_mst:
        print(f"ERROR:\t {algo_name}.{algo_name} returned an unexpected MST")
        print(f"Expected: {expected_mst}")
        print(f"Responded: {responded}")

    pass_cdist = (cumulative_dist == expected_cdist)
    if not pass_cdist:
        print(f"ERROR:\t {algo_name}.{algo_name} returned an incorrect cumulative distance")
        print(f"Expected: {expected_cdist}")
        print(f"Responded: {cumulative_dist}")

    return pass_mst and pass_cdist

def usage():
    """
    Displays the usage information for the script.
    """
    print("USAGE: test.py [algo] where p = prims, k = kruskals")
    print("$ test.py k")
    sys.exit()

def main():
    """
    Main function to test the specified algorithm.
    """
    file = "sample_data.txt"
    delimiter = ' '

    if len(sys.argv) < 2:
        usage()

    algo = sys.argv[1]

    if algo == 'p':
        print("....TESTING PRIMS....")
        if test_read(file, delimiter, algo):
            print("prims.read_graph() : PASS")
        else:
            print("prims.read_graph() : FAIL")

        if test_getmin():
            print("prims.get_min() : PASS")
        else:
            print("prims.get_min() : FAIL")

        if test_algo(algo):
            print("prims.prims() : PASS")
        else:
            print("prims.prims() : FAIL")

    elif algo == 'k':
        print("....TESTING KRUSKALS....")
        if test_read(file, delimiter, algo):
            print("kruskals.read_graph() : PASS")
        else:
            print("kruskals.read_graph() : FAIL")

        if test_algo(algo):
            print("kruskals.kruskals() : PASS")
        else:
            print("kruskals.kruskals() : FAIL")

    else:
        usage()

if __name__ == "__main__":
    main()
