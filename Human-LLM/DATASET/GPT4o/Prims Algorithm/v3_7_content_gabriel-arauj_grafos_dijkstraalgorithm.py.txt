import time
import networkx as nx
import heapq
import sys

def dijkstra(digraph, source, target):
    # Initialize distances and labels for each node
    for node in digraph:
        digraph.nodes[node]['distance'] = sys.maxsize
        digraph.nodes[node]['labelled'] = False
    digraph.nodes[source]['distance'] = 0

    # Priority queue with (distance, node) tuples
    priority_queue = [(0, source)]
    heapq.heapify(priority_queue)
    
    while priority_queue:
        current_distance, current_node = heapq.heappop(priority_queue)
        
        if digraph.nodes[current_node]['labelled']:
            continue
        
        digraph.nodes[current_node]['labelled'] = True
        
        for neighbor in digraph.successors(current_node):
            if not digraph.nodes[neighbor]['labelled']:
                new_distance = current_distance + digraph.edges[current_node, neighbor]['weight']
                if new_distance < digraph.nodes[neighbor]['distance']:
                    digraph.nodes[neighbor]['distance'] = new_distance
                    digraph.nodes[neighbor]['precedent'] = current_node
                    heapq.heappush(priority_queue, (new_distance, neighbor))
    
    # Retrieve the shortest path
    path_distance = digraph.nodes[target]['distance']
    path = [{'weight': path_distance}]
    node = target
    
    while node != source:
        path.append(node)
        node = digraph.nodes[node].get('precedent', source)
    path.append(source)
    
    return path[::-1]

def build_digraph():
    digraph = nx.DiGraph()
    digraph.add_nodes_from(range(1, 10))
    digraph.add_edges_from([
        (1, 2, {'weight': 11}), (1, 3, {'weight': 9}),
        (2, 4, {'weight': 4}), (2, 5, {'weight': 8}),
        (3, 4, {'weight': 8}), (3, 5, {'weight': 6}),
        (4, 6, {'weight': 6}), (4, 7, {'weight': 5}),
        (5, 7, {'weight': 6}), (5, 8, {'weight': 4}),
        (6, 9, {'weight': 6}), (7, 9, {'weight': 4}),
        (8, 9, {'weight': 6})
    ])
    return digraph

def get_user_input():
    try:
        source = int(input("Enter the source vertex: "))
        target = int(input("Enter the target vertex: "))
        return source, target
    except ValueError:
        print("The vertices must be integers.")
        return None, None

def main():
    digraph = build_digraph()
    print("Nodes\n", digraph.nodes())
    print("Edges\n", digraph.edges(data=True))
    
    source, target = get_user_input()
    if source is None or target is None:
        return

    if source in digraph and target in digraph:
        start_time = time.time()
        shortest_path = dijkstra(digraph, source, target)
        elapsed_time = time.time() - start_time
        print(f"Dijkstra's algorithm executed in {elapsed_time:.6f} seconds.")
        print("Shortest Path: ", shortest_path)
    else:
        print("The source or target vertex is not in the graph")

if __name__ == "__main__":
    main()
