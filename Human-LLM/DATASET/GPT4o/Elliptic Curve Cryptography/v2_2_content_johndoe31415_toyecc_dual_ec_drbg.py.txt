from toyecc import getcurvebyname, ECPrivateKey

def generate_points(curve, has_backdoor=True, d=987654321):
    """
    Generate the points P and Q based on whether the backdoor is present.
    
    Args:
        curve: The elliptic curve object.
        has_backdoor (bool): Flag to indicate if the backdoor is present.
        d (int): Scalar multiplier for backdoor, default is 987654321.
    
    Returns:
        Tuple: Points P and Q, and the inverse of d if backdoor is present.
    """
    P = ECPrivateKey.generate(curve).pubkey.point

    if has_backdoor:
        Q = d * P
        dinv = pow(d, -1, curve.n)
        print(f"Dual_EC_DBRG implementation is backdoored (d = {d}, d^{{-1}} = {dinv})")
        print("Backdoor prerequisite: Q = d P")
        return P, Q, dinv
    else:
        Q = ECPrivateKey.generate(curve).pubkey.point
        print("Dual_EC_DBRG implementation is not backdoored")
        return P, Q, None

def generate_initial_state(P):
    """
    Generate the initial state and subsequent state transitions.
    
    Args:
        P: The generator point.
    
    Returns:
        Tuple: Initial state, s, new state, sQ, and output r.
    """
    t = 0x123456789
    print("Initial state:", hex(t))

    s = int((t * P).x)
    print("s            :", hex(s))

    t = int((s * P).x)
    sQ = s * Q
    r = int(sQ.x)

    return t, s, sQ, r

def recover_point(r, dinv, curve):
    """
    Recover the point from the given x-coordinate and inverse.
    
    Args:
        r (int): x-coordinate.
        dinv (int): Inverse of the scalar multiplier.
        curve: The elliptic curve object.
    
    Returns:
        Tuple: Points A and B after recovery.
    """
    A, B = curve.getpointwithx(r)
    return A * dinv, B * dinv

def main():
    # Select the curve
    curve = getcurvebyname("secp112r1")
    print("Selected curve:", curve)

    # Determine if the implementation has a backdoor
    has_backdoor = True

    # Generate points
    P, Q, dinv = generate_points(curve, has_backdoor)

    print("P:", P)
    print("Q:", Q)
    print()

    # Generate initial state and compute transitions
    t, s, sQ, r = generate_initial_state(P)

    print("sQ           :", sQ)
    print("New state    :", hex(t))
    print("Output       :", hex(r))

    # Recover the point if backdoor is present
    if has_backdoor:
        print()
        print("Recovered point from r:")
        recovered_A, recovered_B = recover_point(r, dinv, curve)
        print(recovered_A)
        print(recovered_B)

if __name__ == "__main__":
    main()
