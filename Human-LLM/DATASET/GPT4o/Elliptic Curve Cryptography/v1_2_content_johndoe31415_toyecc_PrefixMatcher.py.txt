class PrefixMatcher:
    def __init__(self, options):
        """
        Initialize the PrefixMatcher with a list of options.

        Args:
            options (list): List of options to match against.
        """
        self._opts = options

    def matchunique(self, value):
        """
        Match the given value uniquely against the options.

        Args:
            value (str): The value to match.

        Returns:
            str: The unique matched option.

        Raises:
            Exception: If no matches or multiple matches are found.
        """
        result = self.match(value)
        if len(result) != 1:
            if len(result) == 0:
                raise Exception(f"'{value}' did not match any options.")
            else:
                raise Exception(f"'{value}' is ambiguous. Please clarify further. Available: {', '.join(sorted(list(result)))}")
        return result[0]

    def match(self, value):
        """
        Match the given value against the options.

        Args:
            value (str): The value to match.

        Returns:
            list: A list of matched options.
        """
        return [option for option in self._opts if option.startswith(value)]

if __name__ == "__main__":
    # Initialize the PrefixMatcher with a list of options
    pm = PrefixMatcher(["import", "install", "foo"])

    # Test the match method
    print(pm.match("i"))  # Output: ['import', 'install']

    # Test the matchunique method
    try:
        print(pm.matchunique("i"))  # Should raise an exception because 'i' is ambiguous
    except Exception as e:
        print(e)  # Output: "'i' is ambiguous. Please clarify further. Available: import, install"

    try:
