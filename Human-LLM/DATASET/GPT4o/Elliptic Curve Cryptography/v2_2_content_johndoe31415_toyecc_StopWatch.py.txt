import time

class StopWatch:
    def __init__(self, component=None, noisy=False):
        """
        Initializes a StopWatch that can time operations, optionally outputting the duration.

        Args:
            component (str): Optional. The name of the component or operation being timed.
            noisy (bool): Optional. If True, outputs the duration when the timing stops.
        """
        self._component = component
        self._noisy = noisy
        self.reset()

    @property
    def finishtime(self):
        """Returns the last recorded finish time in seconds."""
        return self._finishtime

    def stop(self):
        """Stops the timer and calculates the elapsed time since it was last started."""
        self._finishtime = time.time() - self._start_time
        return self.finishtime

    def finish(self):
        """Stops the timer and prints the elapsed time if this stopwatch is set to noisy."""
        self.stop()
        if self._noisy:
            print(f"{self._component} took {self}")

    def reset(self):
        """Resets the stopwatch to start timing from the current moment."""
        self._finishtime = None
        self._start_time = time.time()

    def __str__(self):
        """Returns a string representing the elapsed time in a human-friendly format."""
        elapsed_time = self.stop()
        if elapsed_time < 1:
            return f"{int(elapsed_time * 1000)} ms"
        if elapsed_time < 60:
            return f"{elapsed_time:.2f} sec"
        minutes, seconds = divmod(int(elapsed_time), 60)
        if minutes < 60:
            return f"{minutes} min {seconds} sec"
        hours, minutes = divmod(minutes, 60)
        days, hours = divmod(hours, 24)
        if days > 0:
            return f"{days} days {hours} hours {minutes} min {seconds} sec"
        return f"{hours} hours {minutes} min {seconds} sec"

        """Starts or resets the stopwatch context."""
