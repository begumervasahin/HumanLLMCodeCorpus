from Mods.EC.Elliptic_Curve import EllipticCurve
from Mods.EC.Rational_Point_In_EC import RationalPointInEC

def display_points(o, r1, r2):
    """
    Displays the given elliptic curve points.
    """
    print('Show values')
    print(f'o = {o}')
    print(f'r1 = {r1}')
    print(f'r2 = {r2}')

def display_calculations(o, r1, r2):
    """
    Performs and displays calculations with the given elliptic curve points.
    """
    print('Show calculations')
    print(f'r1 + o = {r1 + o}')
    print(f'r2 + r1 = {r2 + r1}')
    print(f'2 * r1 = {2 * r1}')
    print(f'4 * r1 == 3 * r1 + r1 is {4 * r1 == 3 * r1 + r1}')

def main():
    """
    Main function to create an elliptic curve and points, and display their values and calculations.
    """
    # Create an elliptic curve instance
    curve = EllipticCurve.get_instance(0, -2)
    print(curve)

    # Define points on the elliptic curve
    o = RationalPointInEC()  # Point at infinity
    r1 = RationalPointInEC(-1, 1, curve)
    r2 = RationalPointInEC(2, 2, curve)

    # Display points and their calculations
    display_points(o, r1, r2)
    display_calculations(o, r1, r2)

if __name__ == "__main__":
    main()
