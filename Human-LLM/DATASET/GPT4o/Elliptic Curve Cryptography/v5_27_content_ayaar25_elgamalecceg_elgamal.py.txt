import random
import pickle
import array
import re
from base.general import is_prime  # Importing is_prime for prime checking

class ElGamal:
    def __init__(self):
        pass

    @staticmethod
    def read_file(filename):
        """
        Read binary data from a file.

        :param filename: Path to the file.
        :return: Binary data read from the file.
        """
        with open(filename, 'rb') as file:
            return file.read()

    @staticmethod
    def read_file_list(filename):
        """
        Read a list of data from a file using pickle.

        :param filename: Path to the file.
        :return: List of data read from the file.
        """
        with open(filename, 'rb') as file:
            return pickle.load(file)

    @staticmethod
    def write_file(filename, data):
        """
        Write binary data to a file.

        :param filename: Path to the file.
        :param data: Binary data to write.
        """
        with open(filename, 'wb') as file:
            file.write(data)

    @staticmethod
    def write_file_list(filename, data):
        """
        Write a list of data to a file using pickle.

        :param filename: Path to the file.
        :param data: List of data to write.
        """
        with open(filename, 'wb') as file:
            pickle.dump(data, file)

    def key_gen(self, p):
        """
        Generate public and private keys for ElGamal encryption.

        :param p: A prime number.
        :return: None
        """
        if not is_prime(p):
            raise ValueError("P is not prime")

        g = random.randrange(2, p)
        x = random.randrange(1, p - 1)
        y = pow(g, x, p)

        key_public = [y, g, p]
        key_private = [x, p]

        self.write_file_list("key.pub", key_public)
        self.write_file_list("key.pri", key_private)

    def encrypt(self, filename, key_public):
        """
        Encrypt a file using ElGamal encryption.

        :param filename: Path to the plaintext file.
        :param key_public: Path to the public key file.
        :return: None
        """
        plaintext = self.read_file(filename)
        list_plain_int = list(plaintext)
        key_pub = self.read_file_list(key_public)
        y, g, p = key_pub

        k = random.randrange(1, p - 1)
        list_cipher_int = []

        for plain_int in list_plain_int:
            a = pow(g, k, p)
            b = (pow(y, k, p) * plain_int) % p
            list_cipher_int.extend([a, b])

        regex = re.compile(r'\.\w+$').search(filename)
        if regex:
            cipher_filename = f"cipher{regex.group(0)}"
        else:
            cipher_filename = "cipher"

        self.write_file_list(cipher_filename, list_cipher_int)

    def decrypt(self, filename, key_private):
        """
        Decrypt a file using ElGamal encryption.

        :param filename: Path to the ciphertext file.
        :param key_private: Path to the private key file.
        :return: None
        """
        list_cipher_int = self.read_file_list(filename)
        key_pri = self.read_file_list(key_private)
        x, p = key_pri
        list_plain_int = []

        for i in range(0, len(list_cipher_int), 2):
            a, b = list_cipher_int[i], list_cipher_int[i + 1]
            ax_inv = pow(a, p - 1 - x, p)
            message = (ax_inv * b) % p
            list_plain_int.append(message)

        regex = re.compile(r'\.\w+$').search(filename)
        if regex:
            output_filename = f"out{regex.group(0)}"
        else:
            output_filename = "out"

        self.write_file(output_filename, array.array('B', list_plain_int).tobytes())
