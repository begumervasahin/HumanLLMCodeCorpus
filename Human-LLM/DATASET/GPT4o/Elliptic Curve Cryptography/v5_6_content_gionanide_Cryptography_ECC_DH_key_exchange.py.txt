from random import randint

def compute_b(a, x, y, N):
    """
    Computes the elliptic curve equation value for the given point (x, y).

    Args:
        a (int): Coefficient 'a' of the elliptic curve equation.
        x (int): x-coordinate of the point.
        y (int): y-coordinate of the point.
        N (int): Modulus for the finite field.

    Returns:
        int: Result of the elliptic curve equation modulo N.
    """
    return (y**2 - x**3 - a * x) % N

def modinv(a, m):
    """
    Computes the modular inverse of a number modulo m.

    Args:
        a (int): The number to invert.
        m (int): The modulus.

    Returns:
        int: Modular inverse of a modulo m.
    
    Raises:
        ValueError: If modular inverse does not exist.
    """
    gcd, x, _ = extended_gcd(a, m)
    if gcd != 1:
        raise ValueError("Modular inverse does not exist.")
    return x % m

def extended_gcd(a, b):
    """
    Computes the extended greatest common divisor of two numbers.

    Args:
        a (int): First number.
        b (int): Second number.

    Returns:
        tuple: gcd, and Bézout's coefficients (x, y).
    """
    last_remainder, remainder = abs(a), abs(b)
    x, last_x = 0, 1
    y, last_y = 1, 0

    while remainder:
        last_remainder, (quotient, remainder) = remainder, divmod(last_remainder, remainder)
        x, last_x = last_x - quotient * x, x
        y, last_y = last_y - quotient * y, y

    return last_remainder, last_x * (-1 if a < 0 else 1), last_y * (-1 if b < 0 else 1)

def double_point(point, a, b, N):
    """
    Doubles a point on the elliptic curve.

    Args:
        point (tuple): The point (x, y) to be doubled.
        a (int): Coefficient 'a' of the elliptic curve equation.
        b (int): Coefficient 'b' of the elliptic curve equation.
        N (int): Modulus for the finite field.

    Returns:
        tuple: The doubled point on the elliptic curve.
    """
    x, y = point
    if y == 0:
        return x, y

    m = (3 * x**2 + a) * modinv(2 * y, N)
    x3 = (m**2 - 2 * x) % N
    y3 = (m * (x - x3) - y) % N
    return x3, y3

def add_points(p1, p2, a, b, N):
    """
    Adds two points on the elliptic curve.

    Args:
        p1 (tuple): The first point (x1, y1).
        p2 (tuple): The second point (x2, y2).
        a (int): Coefficient 'a' of the elliptic curve equation.
        b (int): Coefficient 'b' of the elliptic curve equation.
        N (int): Modulus for the finite field.

    Returns:
        tuple: The result of adding the two points.
    """
    x1, y1 = p1
    x2, y2 = p2

    if x1 == x2 and y1 == y2:
        return double_point(p1, a, b, N)

    m = (y2 - y1) * modinv(x2 - x1, N)
    x3 = (m**2 - x1 - x2) % N
    y3 = (m * (x1 - x3) - y1) % N
    return x3, y3

def multiply_point(point, scalar, a, b, N):
    """
    Multiplies a point on the elliptic curve by a scalar.

    Args:
        point (tuple): The point (x, y) to be multiplied.
        scalar (int): The scalar value.
        a (int): Coefficient 'a' of the elliptic curve equation.
        b (int): Coefficient 'b' of the elliptic curve equation.
        N (int): Modulus for the finite field.

    Returns:
        tuple: The result of scalar multiplication.
    """
    x, y = point
    result_x, result_y = x, y
    for _ in range(scalar - 1):
        result_x, result_y = add_points((x, y), (result_x, result_y), a, b, N)
    return result_x, result_y

def validate_coefficients(a, b):
    """
    Validates the elliptic curve coefficients.

    Args:
        a (int): Coefficient 'a' of the elliptic curve equation.
        b (int): Coefficient 'b' of the elliptic curve equation.

    Returns:
        bool: True if the coefficients are valid, False otherwise.
    """
    return 4 * a**3 + 27 * b**2 != 0
