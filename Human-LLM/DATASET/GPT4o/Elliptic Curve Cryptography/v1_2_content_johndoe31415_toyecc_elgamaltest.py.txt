import os
from toyecc import getcurvebyname, ECPrivateKey

def main():
    # Select the elliptic curve
    curve = getcurvebyname("secp521r1")
    
    def msg_to_point(curve, msg, msg_width_bits):
        """
        Convert a message to a point on the curve.
        
        Args:
            curve: The elliptic curve object.
            msg: The message to convert.
            msg_width_bits: The bit width of the message.
        
        Returns:
            Tuple: Number of trials and the resulting point.
        """
        int_message = int.from_bytes(msg, byteorder="little")
        for i in range(100):
            try_message = int_message | (i << msg_width_bits)
            point = curve.getpointwithx(try_message)
            if point:
                return i + 1, point[0]
        return None

    def elgamal_encrypt(recipient_pubkey, msg, msg_width_bits):
        """
        Encrypt a message using ElGamal encryption.
        
        Args:
            recipient_pubkey: The recipient's public key.
            msg: The message to encrypt.
            msg_width_bits: The bit width of the message.
        
        Returns:
            Tuple: The ciphertext consisting of C1 and C2.
        """
        k = ECPrivateKey.generate(curve)
        C1 = k.pubkey.point
        C2 = k.scalar * recipient_pubkey.point
        trials, P_m = msg_to_point(curve, msg, msg_width_bits)
        ciphertext = (C1, C2 + P_m)
        return ciphertext

    def elgamal_decrypt(recipient_privkey, ciphertext, msg_width_bits):
        """
        Decrypt a message using ElGamal decryption.
        
        Args:
            recipient_privkey: The recipient's private key.
            ciphertext: The ciphertext to decrypt.
            msg_width_bits: The bit width of the message.
        
        Returns:
            bytes: The decrypted message.
        """
        C1, C2 = ciphertext
        Cp = C1 * recipient_privkey.scalar
        P_m = C2 + (-Cp)
        int_message = int(P_m.x) & ((1 << msg_width_bits) - 1)
        msg = int_message.to_bytes((msg_width_bits + 7) // 8, byteorder="little")
        return msg

    # Generate a private key and corresponding public key
    privkey = ECPrivateKey.generate(curve)
    pubkey = privkey.pubkey

    # Define the message to be encrypted
    message = b"foobar"
    print("Message:", message)

    # Encrypt the message
    ciphertext = elgamal_encrypt(pubkey, message, msg_width_bits=256)
    print("Ciphertext:")
    print("    C1 =", ciphertext[0])
    print("    C2 =", ciphertext[1])

    # Decrypt the message
    plaintext = elgamal_decrypt(privkey, ciphertext, msg_width_bits=256)
    print("Plaintext:", plaintext)

if __name__ == "__main__":
    main()
