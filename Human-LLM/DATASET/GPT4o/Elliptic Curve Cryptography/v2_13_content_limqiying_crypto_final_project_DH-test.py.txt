from Crypto.PublicKey import ECC
import ecc_simple

def format_ecc_point(ecc_point):
    """
    Formats EccPoint object to allow comparison with custom ecc_simple function outputs.
    
    :param ecc_point: EccPoint Object
    :return: A tuple of integers (x, y)
    """
    return (int(ecc_point.x), int(ecc_point.y))

def generate_keys_and_shared_secret():
    """
    Generates private and public keys for Alice and Bob using pycryptodome's ECC module
    and computes shared secrets using both pycryptodome and custom ecc_simple methods.
    """
    # Generate private keys for Alice and Bob using pycryptodome's ECC module
    keyA = ECC.generate(curve='P-256')
    keyB = ECC.generate(curve='P-256')
    privateA, publicA = keyA.d, keyA.pointQ
    privateB, publicB = keyB.d, keyB.pointQ

    # Compute shared secrets using pycryptodome's methods
    shared_keyA = format_ecc_point(publicB * privateA)
    shared_keyB = format_ecc_point(publicA * privateB)

    # Verify custom ecc_simple methods
    ecc_simple.set_p256_param()
    g = ecc_simple.g
    publicA_1 = ecc_simple.times(int(privateA), g)
    publicB_1 = ecc_simple.times(int(privateB), g)
    shared_keyA_1 = ecc_simple.times(int(privateA), publicB_1)
    shared_keyB_1 = ecc_simple.times(int(privateB), publicA_1)

    return {
        "pycryptodome": {"publicA": format_ecc_point(publicA), "publicB": format_ecc_point(publicB), "shared_keyA": shared_keyA, "shared_keyB": shared_keyB},
        "ecc_simple": {"publicA_1": publicA_1, "publicB_1": publicB_1, "shared_keyA_1": shared_keyA_1, "shared_keyB_1": shared_keyB_1}
    }

def main():
    """
    Main function to perform Diffie-Hellman key exchange and compare outputs with custom ecc_simple methods.
    """
    keys_and_secrets = generate_keys_and_shared_secret()

    # Display keys and shared secrets computed using pycryptodome's methods
    print("PyCryptodome:")
    print(f"Public key (Alice): {keys_and_secrets['pycryptodome']['publicA']}")
    print(f"Public key (Bob): {keys_and_secrets['pycryptodome']['publicB']}")
    print(f"Shared key (Alice): {keys_and_secrets['pycryptodome']['shared_keyA']}")
    print(f"Shared key (Bob): {keys_and_secrets['pycryptodome']['shared_keyB']}")

    # Display keys and shared secrets computed using custom ecc_simple methods
    print("\nECC Simple:")
    print(f"Public key (Alice): {keys_and_secrets['ecc_simple']['publicA_1']}")
    print(f"Public key (Bob): {keys_and_secrets['ecc_simple']['publicB_1']}")
    print(f"Shared key (Alice): {keys_and_secrets['ecc_simple']['shared_keyA_1']}")
    print(f"Shared key (Bob): {keys_and_secrets['ecc_simple']['shared_keyB_1']}")

if __name__ == "__main__":
    main()
