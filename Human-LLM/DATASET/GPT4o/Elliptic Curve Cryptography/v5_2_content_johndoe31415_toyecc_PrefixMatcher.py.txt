class PrefixMatcher:
    def __init__(self, options):
        """
        Initialize the PrefixMatcher with a list of options.

        Args:
            options (list): List of options to match against.
        """
        self.options = options

    def match(self, value):
        """
        Match the given value against the options.

        Args:
            value (str): The value to match.

        Returns:
            list: A list of matched options.
        """
        return [option for option in self.options if option.startswith(value)]

    def match_unique(self, value):
        """
        Match the given value uniquely against the options.

        Args:
            value (str): The value to match.

        Returns:
            str: The unique matched option.

        Raises:
            ValueError: If no matches or multiple matches are found.
        """
        matches = self.match(value)
        if len(matches) == 0:
            raise ValueError(f"'{value}' did not match any options.")
        elif len(matches) > 1:
            raise ValueError(f"'{value}' is ambiguous. Please clarify further. Available: {', '.join(sorted(matches))}")
        return matches[0]

if __name__ == "__main__":
    # Initialize the PrefixMatcher with a list of options
    options = ["import", "install", "foo"]
    matcher = PrefixMatcher(options)

    # Test the match method
    try:
        matches = matcher.match("i")
        print("Matches for 'i':", matches)  # Output: ['import', 'install']
    except ValueError as e:
        print("Error:", e)

    # Test the match_unique method with ambiguous input
    try:
        unique_match = matcher.match_unique("i")
        print("Unique match for 'i':", unique_match)
    except ValueError as e:
        print("Error:", e)  # Output: "'i' is ambiguous. Please clarify further. Available: import, install"

    # Test the match_unique method with a unique match
    try:
        unique_match = matcher.match_unique("im")
        print("Unique match for 'im':", unique_match)  # Output: 'import'
    except ValueError as e:
        print("Error:", e)
