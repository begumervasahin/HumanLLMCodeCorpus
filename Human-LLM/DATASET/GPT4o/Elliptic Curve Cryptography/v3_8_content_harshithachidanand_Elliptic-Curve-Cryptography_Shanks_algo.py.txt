def print_shanks_algorithm():
    """
    Prints Shank's Algorithm tables and finds matching values between S and T tables.
    """
    # Header for Shank's Algorithm
    print("Shank's Algorithm!\n")
    
    # Compute and print the S table
    print("S table:\n")
    s_values = {}  # Dictionary to store S table values
    for x in range(46):
        # Compute value for S table
        value = (5**(45 * x)) % 2017
        s_values[value] = x  # Map value to its index
        print(f"{x} {value}")
    
    # Compute and print the T table
    print("\nT table:\n")
    for y in range(46):
        # Compute value for T table
        value = (1736 * (5**y)) % 2017
        print(f"{y}  {value}")
    
    # Find and print matching values between S and T tables
    print("\nMatching Values:\n")
    for y in range(46):
        # Compute value for T table
        t_value = (1736 * (5**y)) % 2017
        if t_value in s_values:
            # Print matching indices
            print(f"{s_values[t_value]} {y}")

# Execute the function
print_shanks_algorithm()
