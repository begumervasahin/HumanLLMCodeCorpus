from ecc.ecc import EllipticCurve
from ecc.galois_field.galois_field import GaloisField
from ecc.galois_field.generic import is_prime

def find_top_k_prime_orders(k=10, field_size=103):
    """
    Finds the top k elliptic curves with prime orders over a given finite field.

    Args:
        k (int): Number of top orders to find.
        field_size (int): Size of the finite field.

    Returns:
        list: Top k points with their prime orders.
    """
    points = []
    orders = []
    top_k_points = []
    top_k_orders = []

    gf = GaloisField(field_size)

    # Iterate over all possible values of a and b in the finite field
    for a in range(gf.p):
        for b in range(gf.p):
            elliptic_curve = EllipticCurve(a, b, field_size)
            order = elliptic_curve.order_of_curve()
            if is_prime(order):
                orders.append(order)
                points.append((a, b))

    # Find the top k points with the highest prime orders
    for _ in range(k):
        if not orders:
            break
        maximum = max(orders)
        max_index = orders.index(maximum)
        top_k_orders.append(maximum)
        top_k_points.append(points[max_index])
        del orders[max_index]
        del points[max_index]

    return list(zip(top_k_points, top_k_orders))

# Example usage
if __name__ == "__main__":
    top_k = 10  # Number of top points to find
    field_size = 103  # Size of the finite field

    top_k_prime_orders = find_top_k_prime_orders(top_k, field_size)

    for point, order in top_k_prime_orders:
        print(f"{point} : {order}")
