class Fp:
    def __init__(self, value, p):
        self.value = value % p
        self.p = p

    @staticmethod
    def get_instance(p):
        return Fp(0, p)

    def __repr__(self):
        return f'Fp({self.value}, {self.p})'

    def __add__(self, other):
        if self.p != other.p:
            raise ValueError("Cannot add two numbers in different fields")
        return Fp(self.value + other.value, self.p)

    def __sub__(self, other):
        if self.p != other.p:
            raise ValueError("Cannot subtract two numbers in different fields")
        return Fp(self.value - other.value, self.p)

    def __mul__(self, other):
        if self.p != other.p:
            raise ValueError("Cannot multiply two numbers in different fields")
        return Fp(self.value * other.value, self.p)

    def __truediv__(self, other):
        if self.p != other.p:
            raise ValueError("Cannot divide two numbers in different fields")
        return self * other.inverse()

    def inverse(self):
        t, newt = 0, 1
        r, newr = self.p, self.value
        while newr != 0:
            quotient = r // newr
            t, newt = newt, t - quotient * newt
            r, newr = newr, r - quotient * newr
        if r > 1:
            raise ValueError("No inverse exists")
        if t < 0:
            t = t + self.p
        return Fp(t, self.p)

class EllipticCurve:
    def __init__(self, a, b, p):
        self.a = Fp(a, p)
        self.b = Fp(b, p)
        self.p = p

    @staticmethod
    def get_instance(a, b, p):
        return EllipticCurve(a, b, p)

    def __repr__(self):
        return f'EllipticCurve(y^2 = x^3 + {self.a}x + {self.b} over F_{self.p})'

class RationalPointInEC:
    def __init__(self, x=None, y=None, curve=None):
        self.curve = curve if curve else EllipticCurve.get_instance(0, 1, 5)
        if x is None and y is None:
            self.is_infinity = True
            self.x = None
            self.y = None
        else:
            self.is_infinity = False
            self.x = Fp(x, self.curve.p)
            self.y = Fp(y, self.curve.p)

    def __repr__(self):
        if self.is_infinity:
            return 'RationalPointInEC(Infinity)'
        return f'RationalPointInEC({self.x}, {self.y})'

    def __add__(self, other):
        if self.is_infinity:
            return other
        if other.is_infinity:
            return self

        if self.x == other.x and self.y != other.y:
            return RationalPointInEC(curve=self.curve)

        if self == other:
            s = (Fp(3, self.curve.p) * self.x**2 + self.curve.a) / (Fp(2, self.curve.p) * self.y)
        else:
            s = (other.y - self.y) / (other.x - self.x)

        x3 = s**2 - self.x - other.x
        y3 = s * (self.x - x3) - self.y

        return RationalPointInEC(x3.value, y3.value, self.curve)

    def __mul__(self, n):
        result = RationalPointInEC(curve=self.curve)
        addend = self

        while n:
            if n & 1:
                result = result + addend
            addend = addend + addend
            n >>= 1

        return result
