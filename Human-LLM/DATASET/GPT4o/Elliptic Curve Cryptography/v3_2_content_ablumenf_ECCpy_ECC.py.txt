from TabbedGUI import *  # Importing necessary components from TabbedGUI
from tkinter import Tk, RIDGE  # Importing Tkinter for GUI elements

__author__ = 'Aaron Blumenfeld'

def main():
    """
    Main function to set up and run the ECC Toolkit GUI.
    """
    # Initialize the main window
    root = Tk()
    root.title("ECC Toolkit")

    # Create the tab bar
    tab_bar = TabBar(root, "GF(p)")

    # Create the tabs
    prime_tab = PrimeTab(root, "GF(p)")
    binary_tab = BinaryTab(root, "GF(2^r)")
    misc_tab = MiscTab(root, "Misc.")

    # Add the tabs to the tab bar
    tab_bar.add(prime_tab)
    tab_bar.add(binary_tab)
    tab_bar.add(misc_tab)

    # Configure the tab bar's appearance
    tab_bar.config(bd=2, relief=RIDGE)
    tab_bar.show()

    # Run the main loop
    root.mainloop()

if __name__ == '__main__':
    main()
