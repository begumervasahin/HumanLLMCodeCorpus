def print_shanks_algorithm():
    """
    Prints Shank's Algorithm tables and finds matching values between S and T tables.
    """
    MODULO = 2017
    BASE = 5
    MULTIPLIER = 1736
    RANGE_LIMIT = 46

    # Print header for Shank's Algorithm
    print("Shank's Algorithm!\n")
    
    # Print the S table
    print("S table:\n")
    s_values = {}  # Dictionary to store S table values and their indices
    for x in range(RANGE_LIMIT):
        s_value = (BASE**(45 * x)) % MODULO
        s_values[s_value] = x
        print(f"{x} {s_value}")
    
    # Print the T table
    print("\nT table:\n")
    t_values = []  # List to store T table values
    for y in range(RANGE_LIMIT):
        t_value = (MULTIPLIER * (BASE**y)) % MODULO
        t_values.append((y, t_value))
        print(f"{y} {t_value}")
    
    # Find and print matching values between S and T tables
    print("\nMatching Values:\n")
    for y, t_value in t_values:
        if t_value in s_values:
            x = s_values[t_value]
            print(f"{x} {y}")

# Execute the function
print_shanks_algorithm()
