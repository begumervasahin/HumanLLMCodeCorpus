def compute_b(a, x, y, N):
    """
    Computes the right-hand side of the elliptic curve equation.

    Args:
        a (int): The 'a' coefficient of the elliptic curve equation.
        x (int): The x-coordinate of the point.
        y (int): The y-coordinate of the point.
        N (int): The modulus for the finite field.

    Returns:
        int: The result of the elliptic curve equation modulo N.
    """
    return (y**2 - x**3 - a*x) % N

def modinv(a, m):
    """
    Computes the modular inverse of a number.

    Args:
        a (int): The number to invert.
        m (int): The modulus.

    Returns:
        int: The modular inverse of a modulo m.
    """
    g, x, _ = extended_gcd(a, m)
    if g != 1:
        raise ValueError("Modular inverse does not exist.")
    return x % m

def extended_gcd(aa, bb):
    """
    Computes the extended greatest common divisor of a and b.

    Args:
        aa (int): The first number.
        bb (int): The second number.

    Returns:
        tuple: A tuple containing the gcd and the coefficients of Bézout's identity.
    """
    lastremainder, remainder = abs(aa), abs(bb)
    x, lastx = 0, 1
    y, lasty = 1, 0

    while remainder:
        lastremainder, (quotient, remainder) = remainder, divmod(lastremainder, remainder)
        x, lastx = lastx - quotient * x, x
        y, lasty = lasty - quotient * y, y

    return lastremainder, lastx * (-1 if aa < 0 else 1), lasty * (-1 if bb < 0 else 1)

def single_sum(point, a, b, N):
    """
    Computes the point doubling operation on the elliptic curve.

    Args:
        point (tuple): A tuple (x, y) representing a point on the elliptic curve.
        a (int): The 'a' coefficient of the elliptic curve equation.
        b (int): The 'b' coefficient of the elliptic curve equation.
        N (int): The modulus for the finite field.

    Returns:
        tuple: The result of the point doubling operation.
    """
    x, y = point
    if y == 0:
        return x, y

    m = (3 * (x**2) + a) * modinv(2 * y, N)
    x3 = (m**2 - 2 * x) % N
    y3 = (m * (x - x3) - y) % N
    return x3, y3

def add(point1, point2, a, b, N):
    """
    Computes the point addition operation on the elliptic curve.

    Args:
        point1 (tuple): The first point (x1, y1) on the elliptic curve.
        point2 (tuple): The second point (x2, y2) on the elliptic curve.
        a (int): The 'a' coefficient of the elliptic curve equation.
        b (int): The 'b' coefficient of the elliptic curve equation.
        N (int): The modulus for the finite field.

    Returns:
        tuple: The result of the point addition operation.
    """
    x1, y1 = point1
    x2, y2 = point2

    if x1 == x2 and y1 == y2:
        return single_sum(point1, a, b, N)

    m = (y2 - y1) * modinv(x2 - x1, N)
    x3 = (m**2 - x1 - x2) % N
    y3 = (m * (x1 - x3) - y1) % N
    return x3, y3

def multiply(point, n, a, b, N):
    """
    Computes the scalar multiplication of a point on the elliptic curve.

    Args:
        point (tuple): The point (x, y) on the elliptic curve.
        n (int): The scalar value.
        a (int): The 'a' coefficient of the elliptic curve equation.
        b (int): The 'b' coefficient of the elliptic curve equation.
        N (int): The modulus for the finite field.

    Returns:
        tuple: The result of the scalar multiplication operation.
    """
    x, y = point
    xr, yr = x, y
    for _ in range(n - 1):
        xr, yr = add((x, y), (xr, yr), a, b, N)
    return xr, yr

    Args:

        b (int): The 'b' coefficient of the elliptic curve equation.

    Returns:
        bool: True if the coefficients are valid, False otherwise.
    """
    return 4 * a**3 + 27 * b**2 != 0
