import zipfile

def extract_zip_with_password(zip_path, password_file):
    """
    Attempt to extract a zip file using passwords from a given file.

    Args:
        zip_path (str): The path to the zip file.
        password_file (str): The path to the file containing potential passwords.
    """
    try:
        # Attempt to open the zip file
        with zipfile.ZipFile(zip_path) as myzip:
            # Read the potential passwords from the file
            with open(password_file, 'r') as f:
                passwords = f.readlines()
                # Iterate over each password
                for password in passwords:
                    password = password.strip()
                    try:
                        # Attempt to extract the zip file with the current password
                        print(f"Trying password: {password}")
                        myzip.extractall(pwd=password.encode())
                        print(f"Password cracked with: {password}")
                        return
                    except (RuntimeError, zipfile.BadZipFile, zipfile.LargeZipFile):
                        continue
                print("Password not found")
    except zipfile.BadZipFile:
        print("ERROR: Not a valid zip file")
        return

def main():
    zip_path = 'test_zip.zip'
    password_file = 'english.txt'
    extract_zip_with_password(zip_path, password_file)

if __name__ == "__main__":
    main()
