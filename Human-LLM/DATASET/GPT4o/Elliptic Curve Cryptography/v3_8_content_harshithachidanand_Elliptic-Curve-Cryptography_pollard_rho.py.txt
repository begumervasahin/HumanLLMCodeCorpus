# Compute the initial value
initial_value = (5**30) % 2017
print(f"S1 {initial_value}\n")

# Initialize a list to store values
values = []

# Define a function to classify values
def classify_value(val):
    """
    Classify the value into one of the categories.
    
    Args:
        val (int): The value to classify.

    Returns:
        str: The classification of the value.
    """
    if 1 <= val <= 672:
        return f"S0 {val}\n"
    elif 673 <= val <= 1345:
        return f"S1 {val}\n"
    elif 1346 <= val <= 2016:
        return f"S2 {val}\n"
    else:
        return "oops\n"

# Iterate to generate and classify values
val = initial_value
for _ in range(1, 10):
    # Compute the next value in the sequence
    val2 = (val**2) % 2017
    
    # Classify and print val2
    print(classify_value(val2))
    
    # Append current values to the list
    values.append(val)
    values.append(val2)
    
    # Update val for the next iteration
    val = (1736 * val2) % 2017
    
    # Classify and print the updated val
    print(classify_value(val))

# Print the final list of values
print(values)

# Create a dictionary to map each value to its indices in the list
value_indices = {}
for index, value in enumerate(values):
    if value not in value_indices:
        value_indices[value] = []
    value_indices[value].append(index)

# Print the dictionary of values and their indices
print(value_indices)
