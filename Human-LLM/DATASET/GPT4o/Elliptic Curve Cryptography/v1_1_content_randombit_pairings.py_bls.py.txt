import bn256
import time

"""
Demonstrate the bn256 module using the BLS short signature scheme
"""

def bls_keygen():
    """
    Generate a BLS key pair.
    :return: Tuple containing the private key and the public key
    """
    privkey, pubkey = bn256.g2_random()
    return privkey, pubkey

def bls_sign(privkey, msg):
    """
    Sign a message using the BLS scheme.
    :param privkey: Private key
    :param msg: Message to be signed
    :return: Compressed signature
    """
    pt = bn256.g1_hash_to_point(msg)
    assert pt.is_on_curve()
    return bn256.g1_compress(pt.scalar_mul(privkey))

def bls_verify(pubkey, msg, csig):
    """
    Verify a BLS signature.
    :param pubkey: Public key
    :param msg: Message that was signed
    :param csig: Compressed signature
    :return: Boolean indicating whether the signature is valid
    """
    sig = bn256.g1_uncompress(csig)
    assert isinstance(pubkey, bn256.curve_twist)
    assert isinstance(sig, bn256.curve_point)
    msg_pt = bn256.g1_hash_to_point(msg)
    assert msg_pt.is_on_curve()
    v1 = bn256.optimal_ate(pubkey, msg_pt)
    v2 = bn256.optimal_ate(bn256.twist_G, sig)
    return v1 == v2

def test():
    """
    Test the BLS signature scheme by signing and verifying messages.
    """
    priv, pub = bls_keygen()
    
    for i in range(10):  # Reduced to 10 for quick testing; change to 1000 for full test
        msg = ("message @ %f" % time.time()).encode("utf-8")
        print("Message:", msg)
        sig = bls_sign(priv, msg)
        print("Signature:", sig)
        ok = bls_verify(pub, msg, sig)
        assert ok, "Signature verification failed"
        print("Verification:", ok)

if __name__ == "__main__":
    test()
