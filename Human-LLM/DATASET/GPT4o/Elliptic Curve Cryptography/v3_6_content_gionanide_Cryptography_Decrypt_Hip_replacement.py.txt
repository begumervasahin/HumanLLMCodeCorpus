import string

def decrypt_caesar_cipher(encrypted_text, shift_amount):
    """
    Decrypts a Caesar cipher encrypted text by shifting the characters back.

    Args:
        encrypted_text (str): The text encrypted with a Caesar cipher.
        shift_amount (int): The number of positions to shift characters back.

    Returns:
        str: The decrypted text.
    """
    decrypted_chars = []

    for char in encrypted_text:
        if char in string.ascii_uppercase:
            # Calculate the decrypted character by shifting back
            shifted_char_code = (ord(char) - shift_amount - ord('A')) % 26 + ord('A')
            decrypted_chars.append(chr(shifted_char_code))
        else:
            # Non-alphabetic characters remain unchanged
            decrypted_chars.append(char)
    
    return ''.join(decrypted_chars)

def main():
    encrypted_message = 'YMJJDJXHFSRNXQJFIYMJXRNQJHFSQNJGZYYMJXMTJXYJQQYMJYWZYM'
    print(f'Encrypted message: {encrypted_message}')

    # Decrypt the message using all possible shifts from 1 to 25
    for shift in range(1, 26):
        decrypted_message = decrypt_caesar_cipher(encrypted_message, shift)
        print(f'Shift {shift}: {decrypted_message}')

if __name__ == "__main__":
    main()
