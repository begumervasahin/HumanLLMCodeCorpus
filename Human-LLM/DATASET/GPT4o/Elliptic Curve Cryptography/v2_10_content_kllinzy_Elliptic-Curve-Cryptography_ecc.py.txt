def encode_string(s, curve, length):
    """Encodes a string s into a list of points on the curve."""
    return [ECencode(chunk, curve) for chunk in chunkstring(s, length)]

def encrypt_long(arr, key):
    """Encrypts a long message represented by arr using the public key key."""
    curve = key.curve
    p = curve.p
    P = key.P
    Q = key.Q
    l = randint(1, p - 1)
    lP = curve.multiply(l, P)
    lQ = curve.multiply(l, Q)
    return [lP] + [curve.add_points(lQ, point) for point in arr]

def decrypt_long(arr, key, k):
    """Decrypts a long encrypted message arr using the private key k and public key key."""
    curve = key.curve
    neg_kC1 = curve.multiply(k, arr[0]).negate()
    return [curve.add_points(point, neg_kC1) for point in arr[1:]]

def decode_string(arr):
    """Decodes a list of points arr into the original string."""
    return ''.join(ECdecode(point) for point in arr)

def chunkstring(string, length):
    """Chunks the string into pieces of specified length."""
    return [string[i:i + length] for i in range(0, len(string), length)]

def ECdecode(P):
    """Decodes a point P on the curve into a string."""
    n = (P.x - P.x % 1000) // 1000
    chars = []
    while n != 0:
        chars.append(chr(n % 256))
        n //= 256
    return ''.join(chars)

def encrypt(s, key):
    """Encrypts a string s using the public key key."""
    curve = key.curve
    p = curve.p
    encoded_message = encode_string(s, curve, int(math.log(p // (1000 - 1), 256)))
    return encrypt_long(encoded_message, key)

def decrypt(arr, key, k):
    """Decrypts an encrypted array arr using the public key key and private key k."""
    decoded_message = decrypt_long(arr, key, k)
    return decode_string(decoded_message)

def miller_rabin(n, k):
    """Performs the Miller-Rabin primality test on n with k iterations."""
    if n in (2, 3):
        return True
    if n % 2 == 0:
        return False
    r, s = 0, n - 1
    while s % 2 == 0:
        r += 1
        s //= 2
    for _ in range(k):
        a = randint(2, n - 1)
        x = pow(a, s, n)
        if x in (1, n - 1):
            continue
        for _ in range(r - 1):
            x = pow(x, 2, n)
            if x == n - 1:
                break
        else:
            return False
    return True

def random_prime(low, high):
    """Generates a random prime number in the range [low, high]."""
    while True:
        candidate = randint(low, high)
        if miller_rabin(candidate, 45):
            return candidate
