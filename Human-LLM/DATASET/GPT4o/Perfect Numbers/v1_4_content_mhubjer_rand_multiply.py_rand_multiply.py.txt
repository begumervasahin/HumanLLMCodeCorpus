import random
import time
import os
import sys

def main():
    # Generate two random numbers between 1 and 10
    rand1 = random.randint(1, 10)
    rand2 = random.randint(1, 10)

    # Print the first random number
    print("THE FIRST NUMBER IS %s." % rand1)
    time.sleep(3)

    # Print the second random number
    print("THE SECOND NUMBER IS %s." % rand2)
    time.sleep(3)

    # Calculate and print the product of the two numbers
    sol1 = rand1 * rand2
    print("%d X %d = %d" % (rand1, rand2, sol1))
    time.sleep(3)

    # Clear the screen
    os.system("clear" if os.name == "posix" else "cls")

    # Exit the script
    sys.exit()

if __name__ == "__main__":
    main()
