def is_anagram(s1, s2):
    """
    Check if two strings are anagrams of each other.
    
    Parameters:
    s1 (str): The first string.
    s2 (str): The second string.
    
    Returns:
    bool: True if the strings are anagrams, False otherwise.
    """
    # Convert both strings to lists for character manipulation
    alist = list(s2)
    pos1 = 0
    still_ok = True
    
    while pos1 < len(s1) and still_ok:
        pos2 = 0
        found = False
        
        while pos2 < len(alist) and not found:
            if s1[pos1] == alist[pos2]:
                found = True
            else:
                pos2 += 1
        
        if found:
            # Mark the found character as used
            alist[pos2] = None
        else:
            still_ok = False
        
        pos1 += 1
    
    return still_ok

# Example usage
print(is_anagram('ey edip', 'pide ye'))  # Output: True
