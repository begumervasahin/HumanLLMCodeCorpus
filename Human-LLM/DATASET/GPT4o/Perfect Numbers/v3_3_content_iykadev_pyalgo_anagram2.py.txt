def are_anagrams(s1: str, s2: str) -> bool:
    """
    Checks if two strings are anagrams.

    An anagram is a word or phrase formed by rearranging the letters of a different word or phrase,
    typically using all the original letters exactly once.

    Parameters:
    s1 (str): The first string.
    s2 (str): The second string.

    Returns:
    bool: True if the strings are anagrams, False otherwise.
    """
    # Remove spaces and convert to lowercase for accurate comparison
    cleaned_s1 = s1.replace(" ", "").lower()
    cleaned_s2 = s2.replace(" ", "").lower()

    # Compare sorted versions of the cleaned strings
    return sorted(cleaned_s1) == sorted(cleaned_s2)

def main() -> None:
    """
    Main function to test the are_anagrams function.
    """
    # Test the are_anagrams function with provided strings
    s1 = 'ey edip'
    s2 = 'pide ye'
    result = are_anagrams(s1, s2)
    print(f"Are '{s1}' and '{s2}' anagrams? {result}")

if __name__ == "__main__":
    main()
