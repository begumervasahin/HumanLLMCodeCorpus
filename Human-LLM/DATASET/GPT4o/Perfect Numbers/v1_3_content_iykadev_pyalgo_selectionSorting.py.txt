def selectionSort(alist):
    # Traverse through the list from the end to the beginning
    for fillslot in range(len(alist) - 1, 0, -1):
        positionOfMax = 0
        # Find the maximum element in the unsorted part of the list
        for location in range(1, fillslot + 1):
            if alist[location] > alist[positionOfMax]:
                positionOfMax = location
        # Swap the found maximum element with the last element in the unsorted part
        temp = alist[fillslot]
        alist[fillslot] = alist[positionOfMax]
        alist[positionOfMax] = temp

# Test the function with the provided list
alist = [54, 26, 93, 17, 77, 31, 44, 55, 20]
selectionSort(alist)
print(alist)  # Output should be a sorted list
