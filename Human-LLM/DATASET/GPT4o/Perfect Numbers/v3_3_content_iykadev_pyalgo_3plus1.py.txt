def collatz_sequence_steps(n: int) -> None:
    """
    Generates and prints the Collatz sequence starting from a given integer n.

    The Collatz sequence is defined as:
    - If the number is even, divide it by 2.
    - If the number is odd, multiply it by 3 and add 1.
    The sequence ends when the number reaches 1.

    Parameters:
    n (int): The starting integer for the Collatz sequence.
    """
    step_count = 0  # Initialize step counter

    while n > 1:
        step_count += 1  # Increment step counter

        if n % 2 == 0:
            n //= 2  # If n is even, divide by 2
        else:
            n = 3 * n + 1  # If n is odd, multiply by 3 and add 1

        print(f"Step {step_count}: {n}")  # Print the current step and value of n

def main(start_value: int) -> None:
    """
    Main function to execute the Collatz sequence steps.

    Parameters:
    start_value (int): The starting integer for the Collatz sequence.
    """
    print(f"Starting Collatz sequence with initial value: {start_value}")
    collatz_sequence_steps(start_value)

if __name__ == "__main__":
    start_value = 15000  # Starting value for the Collatz sequence
    main(start_value)
