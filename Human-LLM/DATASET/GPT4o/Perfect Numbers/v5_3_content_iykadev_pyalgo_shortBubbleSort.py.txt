def short_bubble_sort(alist):
    """Sorts a list using an optimized bubble sort algorithm that stops early if no exchanges are made."""
    passnum = len(alist) - 1
    exchanges = True

    while passnum > 0 and exchanges:
        exchanges = False

        for i in range(passnum):
            if alist[i] > alist[i + 1]:
                # Swap the elements
                alist[i], alist[i + 1] = alist[i + 1], alist[i]
                exchanges = True

        passnum -= 1

# Test the short_bubble_sort function
alist = [20, 30, 40, 90, 50, 60, 70, 80, 100, 110]
short_bubble_sort(alist)
