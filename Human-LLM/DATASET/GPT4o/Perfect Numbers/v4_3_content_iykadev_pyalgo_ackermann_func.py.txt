import sys

# Increase the recursion limit to handle deep recursion required by the Ackermann function
sys.setrecursionlimit(3000)

def ackermann(m, n):
    """
    Compute the Ackermann function for given non-negative integers m and n.
    
    Parameters:
    m (int): The first non-negative integer.
    n (int): The second non-negative integer.
    
    Returns:
    int: The result of the Ackermann function.
    """
    if m == 0:
        return n + 1
    elif n == 0:
        return ackermann(m - 1, 1)
    else:
        return ackermann(m - 1, ackermann(m, n - 1))

# Compute and print the result of the Ackermann function for m=3 and n=4
print(ackermann(3, 4))
