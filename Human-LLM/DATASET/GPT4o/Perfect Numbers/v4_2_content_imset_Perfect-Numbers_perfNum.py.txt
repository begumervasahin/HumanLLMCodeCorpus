import sys

def is_perfect_number(n):
    """Check if a number is perfect."""
    divisors = [i for i in range(1, n) if n % i == 0]
    return sum(divisors) == n

def perf_check(mode, limit):
    """Check for perfect numbers based on the provided mode and limit."""
    try:
        limit = int(limit)
    except ValueError:
        print("The limit must be an integer.")
        return
    
    if mode.lower() == 'check':
        numbers_to_check = [limit]
    elif mode.lower() == 'iterate':
        numbers_to_check = range(2, limit + 1)
    else:
        print(f"Argument '{mode}' is not valid. Use 'check' or 'iterate'.")
        return
    
    for num in numbers_to_check:
        if is_perfect_number(num):
            print(f"{num} is perfect!")
        else:
            print(f"{num} is not perfect!")

if __name__ == '__main__':
    if len(sys.argv) != 3:
        print("Usage: python script.py <mode> <limit>")
    else:
        perf_check(sys.argv[1], sys.argv[2])
