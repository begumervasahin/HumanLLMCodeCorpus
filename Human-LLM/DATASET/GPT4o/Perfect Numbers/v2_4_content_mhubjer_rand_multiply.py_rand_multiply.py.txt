import random
import time
import os
import sys

def generate_random_number() -> int:
    """
    Generates a random integer between 1 and 10.

    Returns:
    int: A random integer between 1 and 10.
    """
    return random.randint(1, 10)

def clear_screen() -> None:
    """
    Clears the console screen.
    """
    os.system("clear" if os.name == "posix" else "cls")

def main() -> None:
    """
    Main function to generate two random numbers, print them, calculate their product, and clear the screen.
    """
    # Generate two random numbers between 1 and 10
    rand1 = generate_random_number()
    rand2 = generate_random_number()

    # Print the first random number
    print(f"THE FIRST NUMBER IS {rand1}.")
    time.sleep(3)

    # Print the second random number
    print(f"THE SECOND NUMBER IS {rand2}.")
    time.sleep(3)

    # Calculate and print the product of the two numbers
    product = rand1 * rand2
    print(f"{rand1} X {rand2} = {product}")
    time.sleep(3)

    # Clear the screen
    clear_screen()

    # Exit the script
    sys.exit()

if __name__ == "__main__":
    main()
