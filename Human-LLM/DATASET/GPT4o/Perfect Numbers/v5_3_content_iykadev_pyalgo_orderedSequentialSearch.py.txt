def ordered_sequential_search(alist, item):
    """Perform an ordered sequential search on a sorted list."""
    pos = 0

    while pos < len(alist):
        if alist[pos] == item:
            return True
        elif alist[pos] > item:
            return False
        pos += 1

    return False

# Test the ordered_sequential_search function
test_list = [0, 1, 2, 8, 13, 17, 19, 32, 42]
print(ordered_sequential_search(test_list, 3))   # Output: False
print(ordered_sequential_search(test_list, 13))  # Output: True
