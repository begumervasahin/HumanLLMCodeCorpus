def anagram(s1, s2):
    # Create two lists to count the frequency of each letter in both strings
    count1 = [0] * 26
    count2 = [0] * 26
    
    # Count the frequency of each letter in the first string
    for char in s1:
        index = ord(char) - ord('a')
        count1[index] += 1
    
    # Count the frequency of each letter in the second string
    for char in s2:
        index = ord(char) - ord('a')
        count2[index] += 1
    
    # Compare the frequency counts of both strings
    for i in range(26):
        if count1[i] != count2[i]:
            return False
    
    return True

# Test the function
print(anagram('pide', 'epid'))  # This should return True
print(anagram('hello', 'bello'))  # This should return False
