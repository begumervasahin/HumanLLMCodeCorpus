def sieve_of_eratosthenes(limit: int):
    """
    Generates prime numbers up to a specified limit using the Sieve of Eratosthenes.

    Parameters:
    limit (int): The upper limit for generating prime numbers.

    Yields:
    int: The next prime number.
    """
    multiples = set()
    for number in range(2, limit + 1):
        if number not in multiples:
            yield number
            multiples.update(range(number * number, limit + 1, number))

def main() -> None:
    """
    Main function to generate and print the list of prime numbers up to a specified limit.
    """
    limit = 100
    primes = list(sieve_of_eratosthenes(limit))
    print(f"Prime numbers up to {limit}: {primes}")

if __name__ == "__main__":
    main()
