def insertionSort(alist):
    """
    Sorts a list using the insertion sort algorithm.

    Parameters:
    alist (list): The list to be sorted.

    Returns:
    None: The list is sorted in place.
    """
    for index in range(1, len(alist)):
        currentvalue = alist[index]
        position = index

        # Shift elements of the sorted segment that are greater than the current value
        while position > 0 and alist[position - 1] > currentvalue:
            alist[position] = alist[position - 1]
            position -= 1

        # Place the current value in its correct position
        alist[position] = currentvalue

def insertion_sort_main():
    # Example usage of insertionSort function
    alist = [54, 26, 93, 17, 77, 31, 44, 55, 20]
    insertionSort(alist)
    print("Sorted list:", alist)

if __name__ == "__main__":
    insertion_sort_main()
