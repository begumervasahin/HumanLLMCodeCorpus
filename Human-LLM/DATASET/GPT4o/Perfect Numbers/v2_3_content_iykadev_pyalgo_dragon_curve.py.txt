from turtle import *

def dragon_curve(step, length):
    draw_curve_right(step, length)

def draw_curve_right(step, length):
    step -= 1
    length /= 1.41421  # Scaling factor for the dragon curve
    if step > 0:
        right(45)
        draw_curve_right(step, length)
        left(90)
        draw_curve_left(step, length)
        right(45)
    else:
        right(45)
        forward(length)
        left(90)
        forward(length)
        right(45)

def draw_curve_left(step, length):
    step -= 1
    length /= 1.41421  # Scaling factor for the dragon curve
    if step > 0:
        left(45)
        draw_curve_right(step, length)
        right(90)
        draw_curve_left(step, length)
        left(45)
    else:
        left(45)
        forward(length)
        right(90)
        forward(length)
        left(45)

# Setup the turtle environment
speed(0)         # Set the turtle speed to the maximum
bgcolor('white') # Set the background color to white
color('blue')    # Set the drawing color to blue
pensize(2)       # Set the pen size

# Start drawing the dragon curve
initial_length = 200  # Length of the initial step
iterations = 10       # Number of iterations

# Position the turtle for drawing
penup()
goto(-initial_length // 2, 0)
pendown()

# Draw the dragon curve
dragon_curve(iterations, initial_length)

# Complete the drawing
done()
