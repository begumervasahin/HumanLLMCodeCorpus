def collatz_sequence(n):
    """
    Compute the Collatz sequence (3n + 1 problem) for a given integer n.
    
    Parameters:
    n (int): The starting integer for the Collatz sequence.
    
    Returns:
    sequence (list): A list containing the sequence of values from n to 1.
    """
    sequence = []
    while n > 1:
        sequence.append(n)
        if n % 2:  # If n is odd
            n = 3 * n + 1
        else:  # If n is even
            n = n // 2  # Use integer division
    sequence.append(1)  # The sequence always ends with 1
    return sequence

def main(start_value):
    """
    Start the Collatz sequence for a given integer start_value and print the sequence.
    
    Parameters:
    start_value (int): The starting integer for the Collatz sequence.
    
    Returns:
    None
    """
    sequence = collatz_sequence(start_value)
    for value in sequence:
        print(value)

# Run the main function with the initial value 15000
if __name__ == "__main__":
    main(15000)
