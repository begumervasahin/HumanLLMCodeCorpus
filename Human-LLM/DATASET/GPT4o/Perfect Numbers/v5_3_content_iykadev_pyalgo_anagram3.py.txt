def is_anagram(s1, s2):
    # Check if lengths are different; no need to proceed further if they are
    if len(s1) != len(s2):
        return False

    # Initialize counters for each character
    count1 = [0] * 26
    count2 = [0] * 26

    # Count characters in the first string
    for char in s1:
        count1[ord(char) - ord('a')] += 1

    # Count characters in the second string
    for char in s2:
        count2[ord(char) - ord('a')] += 1

    # Compare character counts
    return count1 == count2

print(is_anagram('pide', 'epid'))  # Output: True
