from turtle import *

def dragon_curve(steps, length):
    """Draws a Dragon Curve with the given number of steps and length."""
    draw_curve_right(steps, length)

def draw_curve_right(steps, length):
    """Recursively draws the right part of the Dragon Curve."""
    if steps == 0:
        forward(length)
        return
    
    steps -= 1
    length /= 1.41421  # Scale down the length

    right(45)
    draw_curve_right(steps, length)
    left(90)
    draw_curve_left(steps, length)
    right(45)

def draw_curve_left(steps, length):
    """Recursively draws the left part of the Dragon Curve."""
    if steps == 0:
        forward(length)
        return

    steps -= 1
    length /= 1.41421  # Scale down the length

    left(45)
    draw_curve_right(steps, length)
    right(90)
    draw_curve_left(steps, length)
    left(45)

# Example usage
if __name__ == "__main__":
    speed(0)  # Set the speed of the turtle
    length = 200  # Initial length of the line segments
    steps = 10  # Number of iterations

    penup()
    goto(-length / 2, 0)  # Start position
    pendown()

    dragon_curve(steps, length)
    done()  # Finish the drawing
