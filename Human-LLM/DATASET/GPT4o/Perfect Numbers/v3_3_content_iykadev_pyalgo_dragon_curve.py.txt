from turtle import *

def dragon_curve(step, length):
    """
    Draws the dragon curve using turtle graphics.
    
    Args:
    step (int): The number of iterations for the dragon curve.
    length (float): The initial length of the step.
    """
    draw_curve_right(step, length)

def draw_curve_right(step, length):
    """
    Draws the right-hand segment of the dragon curve.
    
    Args:
    step (int): The number of iterations remaining.
    length (float): The current length of the step.
    """
    step -= 1
    length /= 1.41421  # Scaling factor for the dragon curve
    if step > 0:
        right(45)
        draw_curve_right(step, length)
        left(90)
        draw_curve_left(step, length)
        right(45)
    else:
        draw_line_segment(length, direction='right')

def draw_curve_left(step, length):
    """
    Draws the left-hand segment of the dragon curve.
    
    Args:
    step (int): The number of iterations remaining.
    length (float): The current length of the step.
    """
    step -= 1
    length /= 1.41421  # Scaling factor for the dragon curve
    if step > 0:
        left(45)
        draw_curve_right(step, length)
        right(90)
        draw_curve_left(step, length)
        left(45)
    else:
        draw_line_segment(length, direction='left')

def draw_line_segment(length, direction):
    """
    Draws a single line segment in the specified direction.
    
    Args:
    length (float): The length of the line segment.
    direction (str): The direction to draw ('right' or 'left').
    """
    if direction == 'right':
        right(45)
        forward(length)
        left(90)
        forward(length)
        right(45)
    elif direction == 'left':
        left(45)
        forward(length)
        right(90)
        forward(length)
        left(45)

# Setup the turtle environment
def setup_turtle():
    speed(0)         # Set the turtle speed to the maximum
    bgcolor('white') # Set the background color to white
    color('blue')    # Set the drawing color to blue
    pensize(2)       # Set the pen size

def main():
    initial_length = 200  # Length of the initial step
    iterations = 10       # Number of iterations

    # Setup turtle and position it for drawing
    setup_turtle()
    penup()
    goto(-initial_length // 2, 0)
    pendown()

    # Draw the dragon curve
    dragon_curve(iterations, initial_length)

    # Complete the drawing
    done()

if __name__ == "__main__":
    main()
