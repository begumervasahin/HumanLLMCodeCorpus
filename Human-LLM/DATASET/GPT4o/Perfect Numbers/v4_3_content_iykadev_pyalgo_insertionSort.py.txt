def insertion_sort(alist):
    """
    Sorts a list in ascending order using the insertion sort algorithm.
    
    Parameters:
    alist (list): The list to be sorted.
    
    Returns:
    None: The list is sorted in place.
    """
    for index in range(1, len(alist)):
        current_value = alist[index]
        position = index
        
        # Shift elements of the sorted segment to the right to make space for the current value
        while position > 0 and alist[position - 1] > current_value:
            alist[position] = alist[position - 1]
            position -= 1
        
        # Place the current value at its correct position
        alist[position] = current_value

# Example usage
alist = [54, 26, 93, 17, 77, 31, 44, 55, 20]
insertion_sort(alist)
print(alist)  # Output: [17, 20, 26, 31, 44, 54, 55, 77, 93]
