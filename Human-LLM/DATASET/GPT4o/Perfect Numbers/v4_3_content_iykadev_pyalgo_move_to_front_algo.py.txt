from __future__ import print_function
from string import ascii_lowercase

SYMBOL_TABLE = list(ascii_lowercase)

def move_to_front_encode(string, symbol_table):
    """Encodes a string using the Move-to-Front algorithm."""
    sequence = []
    pad = symbol_table[:]
    for char in string:
        index = pad.index(char)
        sequence.append(index)
        pad = [pad.pop(index)] + pad
    return sequence

def move_to_front_decode(sequence, symbol_table):
    """Decodes a sequence using the Move-to-Front algorithm."""
    chars = []
    pad = symbol_table[:]
    for index in sequence:
        char = pad[index]
        chars.append(char)
        pad = [pad.pop(index)] + pad
    return ''.join(chars)

if __name__ == '__main__':
    test_strings = ['broood', 'bananaaa', 'hiphophiphop']
    for s in test_strings:
        encoded = move_to_front_encode(s, SYMBOL_TABLE)
        print(f'{s:14} encodes to {encoded}', end=', ')
        decoded = move_to_front_decode(encoded, SYMBOL_TABLE)
        print(f'which decodes back to {decoded}')
