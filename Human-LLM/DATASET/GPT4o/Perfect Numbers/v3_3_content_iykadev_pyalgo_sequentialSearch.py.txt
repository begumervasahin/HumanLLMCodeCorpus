def sequential_search(arr: list, target: any) -> bool:
    """
    Performs a sequential search on a list to find a specified item.

    Parameters:
    arr (list): The list to search.
    target (any): The item to find.

    Returns:
    bool: True if the item is found, False otherwise.
    """
    for item in arr:
        if item == target:
            return True
    return False

def main() -> None:
    """
    Main function to test the sequential search function.
    """
    test_list = [1, 2, 32, 8, 17, 19, 42, 13, 0]
    print(f"Is 3 in the list? {sequential_search(test_list, 3)}")   # Output: False
    print(f"Is 13 in the list? {sequential_search(test_list, 13)}")  # Output: True

if __name__ == "__main__":
    main()
