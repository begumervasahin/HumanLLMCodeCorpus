def sequentialSearch(alist, item):
    """
    Performs a sequential search on a list to find a specified item.

    Parameters:
    alist (list): The list to search.
    item (any): The item to find.

    Returns:
    bool: True if the item is found, False otherwise.
    """
    pos = 0
    found = False
    
    while pos < len(alist) and not found:
        if alist[pos] == item:
            found = True
        else:
            pos += 1
    
    return found

def main():
    testlist = [1, 2, 32, 8, 17, 19, 42, 13, 0]
    print(sequentialSearch(testlist, 3))  # Output: False
    print(sequentialSearch(testlist, 13))  # Output: True

if __name__ == "__main__":
    main()
