import pickle

def remove_duplicates(lst):
    if lst:
        # Sort the list to bring duplicates together
        lst.sort()
        # Initialize the unique_list with the last element of the sorted list
        unique_list = [lst[-1]]
        # Iterate over the sorted list in reverse order (excluding the last element already added)
        for i in range(len(lst) - 2, -1, -1):
            # If the current element is different from the first element of unique_list, insert it at the beginning
            if lst[i] != unique_list[0]:
                unique_list.insert(0, lst[i])
        return unique_list
    return lst

# Load the list from the pickle file
with open("Fail.txt", 'rb') as file:
    list_data = pickle.load(file)

print("Original length:", len(list_data))

# Remove duplicates
list_data = remove_duplicates(list_data)

print("Length after removing duplicates:", len(list_data))
