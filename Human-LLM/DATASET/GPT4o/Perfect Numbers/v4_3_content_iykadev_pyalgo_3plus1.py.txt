def collatz_sequence(x):
    """
    Function to compute the Collatz sequence (3n + 1 problem) for a given integer x.
    
    Parameters:
    x (int): The starting integer for the Collatz sequence.
    
    Returns:
    None: The function prints the sequence steps until x becomes 1.
    """
    i = 0  # Initialize a counter to keep track of the number of iterations
    while x > 1:
        i += 1  # Increment the counter
        if x % 2:  # If x is odd
            x = 3 * x + 1
        else:  # If x is even
            x = x / 2
        print(x)  # Print the current value of x

def main(x):
    """
    Main function to start the Collatz sequence for a given integer x.
    
    Parameters:
    x (int): The starting integer for the Collatz sequence.
    
    Returns:
    None
    """
    collatz_sequence(x)
