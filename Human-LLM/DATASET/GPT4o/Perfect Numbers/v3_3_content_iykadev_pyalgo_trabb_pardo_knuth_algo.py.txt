import math

def f(x):
    """
    Compute the function f(x) = sqrt(|x|) + 5 * x^3.
    
    Args:
    x (float): The input number.
    
    Returns:
    float: The result of the function.
    """
    return math.sqrt(abs(x)) + 5 * x ** 3

def ask():
    """
    Prompt the user to enter 11 numbers and return them as a list of floats.
    
    Returns:
    list of float: The list of 11 numbers entered by the user.
    """
    while True:
        try:
            numbers = input('\nEnter 11 numbers: ').strip().split()
            if len(numbers) != 11:
                print("Please enter exactly 11 numbers.")
                continue
            return [float(y) for y in numbers]
        except ValueError:
            print("Please enter valid numbers.")

def main():
    """
    Main function to get user input, compute the function for each number, 
    and print the results.
    """
    numbers = ask()
    numbers.reverse()
    
    for x in numbers:
        result = f(x)
        if result > 400:
            print(f' {x}: TOO LARGE!', end='')
        else:
            print(f' {x}: {result}', end='')
    print('')

if __name__ == '__main__':
    main()
