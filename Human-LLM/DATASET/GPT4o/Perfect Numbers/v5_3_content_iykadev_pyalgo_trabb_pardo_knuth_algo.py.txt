import math

def f(x):
    """Calculate the function f(x) = sqrt(abs(x)) + 5 * x^3."""
    return math.sqrt(abs(x)) + 5 * x ** 3

def ask():
    """Prompt the user to input 11 numbers and return them as a list of floats."""
    input_numbers = input('\nEnter 11 numbers separated by spaces: ')
    return [float(num) for num in input_numbers.strip().split()[:11]]

def main():
    """Main function to process the input numbers and print the results."""
    numbers = ask()
    numbers.reverse()

    for x in numbers:
        result = f(x)
        if result > 400:
            print(f'{x}: TOO LARGE!', end=' ')
        else:
            print(f'{x}: {result}', end=' ')

    print('')

if __name__ == '__main__':
    main()
