def selection_sort(alist):
    """Sorts a list using the selection sort algorithm."""
    for fillslot in range(len(alist) - 1, 0, -1):
        position_of_max = 0

        # Find the position of the maximum element in the unsorted portion
        for location in range(1, fillslot + 1):
            if alist[location] > alist[position_of_max]:
                position_of_max = location

        # Swap the largest item found with the item in the fillslot
        alist[fillslot], alist[position_of_max] = alist[position_of_max], alist[fillslot]

# Test the selection_sort function
alist = [54, 26, 93, 17, 77, 31, 44, 55, 20]
selection_sort(alist)
