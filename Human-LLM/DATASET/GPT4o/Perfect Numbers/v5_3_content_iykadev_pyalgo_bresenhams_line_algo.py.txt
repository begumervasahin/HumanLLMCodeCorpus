class Bitmap:
    def __init__(self, width, height):
        """Initialize a bitmap with the given width and height."""
        self.width = width
        self.height = height
        self.pixels = [[' ' for _ in range(width)] for _ in range(height)]

    def set_pixel(self, x, y):
        """Set a pixel at (x, y) to '#'."""
        if 0 <= x < self.width and 0 <= y < self.height:
            self.pixels[y][x] = '#'

    def draw_line(self, x0, y0, x1, y1):
        """Draw a line from (x0, y0) to (x1, y1) using Bresenham's line algorithm."""
        dx, dy = abs(x1 - x0), abs(y1 - y0)
        x, y = x0, y0
        sx, sy = (-1 if x0 > x1 else 1), (-1 if y0 > y1 else 1)
        
        if dx > dy:
            err = dx / 2.0
            while x != x1:
                self.set_pixel(x, y)
                err -= dy
                if err < 0:
                    y += sy
                    err += dx
                x += sx
        else:
            err = dy / 2.0
            while y != y1:
                self.set_pixel(x, y)
                err -= dx
                if err < 0:
                    x += sx
                    err += dy
                y += sy
        self.set_pixel(x, y)

    def display(self):
        """Display the bitmap in the console."""
        for row in self.pixels:
            print(''.join(row))

def main():
    """Create a bitmap and draw specified lines."""
    bitmap = Bitmap(17, 17)
    lines_to_draw = [(1, 8, 8, 16), (8, 16, 16, 8), (16, 8, 8, 1), (8, 1, 1, 8)]

    for points in lines_to_draw:
        bitmap.draw_line(*points)

    bitmap.display()

if __name__ == '__main__':
    main()
