def binary_search(alist, item):
    """
    Recursively performs a binary search on a sorted list to find an item.
    
    Args:
    alist (list): The sorted list to search.
    item (int): The item to search for.
    
    Returns:
    bool: True if the item is found, False otherwise.
    """
    if len(alist) == 0:
        return False
    
    midpoint = len(alist) // 2
    if alist[midpoint] == item:
        return True
    elif item < alist[midpoint]:
        return binary_search(alist[:midpoint], item)
    else:
        return binary_search(alist[midpoint+1:], item)

# Test the binary search function
test_list = [0, 1, 2, 8, 13, 17, 19, 32, 42]
print(binary_search(test_list, 3))   # Output: False
print(binary_search(test_list, 13))  # Output: True
