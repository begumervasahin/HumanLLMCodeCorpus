def ordered_sequential_search(alist, item):
    """
    Performs an ordered sequential search on a sorted list to find an item.
    
    Args:
    alist (list): The sorted list to search.
    item (int): The item to search for.
    
    Returns:
    bool: True if the item is found, False otherwise.
    """
    pos = 0
    found = False
    stop = False
    
    while pos < len(alist) and not found and not stop:
        if alist[pos] == item:
            found = True
        elif alist[pos] > item:
            stop = True
        else:
            pos += 1
    
    return found

# Test the function with the provided test list
test_list = [0, 1, 2, 8, 13, 17, 19, 32, 42]
print(ordered_sequential_search(test_list, 3))   # Output: False
print(ordered_sequential_search(test_list, 13))  # Output: True
