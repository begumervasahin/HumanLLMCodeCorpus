import functools

def sierpinski(n):
    """
    Generate the Sierpinski triangle pattern of order n.
    
    Parameters:
    n (int): The order of the Sierpinski triangle.
    
    Returns:
    list: A list of strings representing the Sierpinski triangle.
    """
    def aggregate(triangle, i):
        """
        Helper function to aggregate the Sierpinski triangle at each iteration.
        
        Parameters:
        triangle (list): The current state of the triangle.
        i (int): The current iteration index.
        
        Returns:
        list: The updated state of the triangle.
        """
        space = " " * (2 ** i)
        top = [space + row + space for row in triangle]
        bottom = [row + " " + row for row in triangle]
        return top + bottom

    initial_triangle = ["*"]
    return functools.reduce(aggregate, range(n), initial_triangle)

def main():
    """
    Main function to generate and print the Sierpinski triangle of a specified order.
    
    Returns:
    None
    """
    order = 4
    triangle = sierpinski(order)
    print("\n".join(triangle))

# Run the main function
if __name__ == "__main__":
    main()
