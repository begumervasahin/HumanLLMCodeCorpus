def transform_list(s: list, j: int, k: int, iterations: int) -> None:
    """
    Transforms a list based on specific rules for a given number of iterations.

    Parameters:
    s (list): The list to be transformed.
    j (int): The index used for the transformation calculation (1-based index).
    k (int): The index used for the transformation calculation (1-based index).
    iterations (int): The number of iterations to perform the transformation.

    Returns:
    None: The list is modified in place.
    """
    for _ in range(iterations):
        for i in range(len(s)):
            if i == 0:
                # Calculate the new value to be inserted at the end of the list
                new_value = (s[j - 1] + s[k - 1]) % 10
            elif 0 < i < len(s) - 1:
                # Shift the elements to the left
                s[i] = s[i + 1]
            else:
                # Insert the calculated value at the end of the list
                s[i] = new_value
                print(s[i])

def main() -> None:
    """
    Main function to demonstrate the transformation of a list based on specific rules.
    """
    j = 3
    k = 7
    s = [8, 6, 7, 5, 3, 0, 9]
    iterations = 10

    print("Original list:", s)
    transform_list(s, j, k, iterations)
    print("Transformed list:", s)

if __name__ == "__main__":
    main()
