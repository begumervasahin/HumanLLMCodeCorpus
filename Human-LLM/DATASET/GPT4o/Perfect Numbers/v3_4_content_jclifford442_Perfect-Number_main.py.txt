def find_divisors(num):
    """
    Find all divisors of a given number (excluding the number itself).
    """
    return [i for i in range(1, num) if num % i == 0]

def is_perfect_number(num):
    """
    Check if the given number is a perfect number.
    
    A perfect number is one whose divisors (excluding itself) sum up to that number.
    """
    divisors = find_divisors(num)
    if sum(divisors) == num:
        print(f"{num} is a perfect number. Divisors: {divisors}")
        return True
    return False

def main():
    """
    Main function to check for perfect numbers in a given range.
    """
    print("A perfect number is a number whose divisors, excluding itself, add up to that number.")
    print("It is unknown if there are any odd perfect numbers, and it is also unknown if there are an infinite number of perfect numbers.")
    
    # Check for perfect numbers in the range 1 to 9999
    for number in range(1, 10000):
        is_perfect_number(number)

if __name__ == "__main__":
    main()
