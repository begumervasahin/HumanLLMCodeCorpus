def zigzag(n):
    """
    Generate a dictionary representing a zigzag matrix of size n x n.
    
    Args:
        n (int): The size of the matrix.
        
    Returns:
        dict: A dictionary where keys are (x, y) tuples and values are numbers in zigzag order.
    """
    index_order = sorted(((x, y) for x in range(n) for y in range(n)),
                         key=lambda xy: (xy[0] + xy[1], -xy[1] if (xy[0] + xy[1]) % 2 else xy[1]))
    return {index: num for num, index in enumerate(index_order)}

def print_zigzag(matrix):
    """
    Print the zigzag matrix in a formatted way.
    
    Args:
        matrix (dict): A dictionary where keys are (x, y) tuples and values are numbers in zigzag order.
    """
    n = int(len(matrix) ** 0.5 + 0.5)
    for x in range(n):
        for y in range(n):
            print(f"{matrix[(x, y)]:2d}", end=" ")
        print()

# Generate and print a zigzag matrix of size 6
zigzag_matrix = zigzag(6)
print_zigzag(zigzag_matrix)
