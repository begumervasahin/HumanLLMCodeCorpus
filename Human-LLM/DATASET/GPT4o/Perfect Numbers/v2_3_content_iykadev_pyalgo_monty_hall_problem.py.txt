import random

# Initialize counters for stay and switch strategies
stay_wins = 0
switch_wins = 0

# Simulate the game 1000 times
for _ in range(1000):
    # Set up the game with one prize (1) and two goats (0)
    doors = [1, 0, 0]
    random.shuffle(doors)
    
    # Contestant makes an initial choice
    initial_choice = random.randrange(3)
    chosen_door = doors[initial_choice]
    
    # Remove the chosen door from the list
    remaining_doors = doors[:initial_choice] + doors[initial_choice + 1:]
    
    # Host reveals a goat by removing one 0 from the remaining doors
    for i in range(len(remaining_doors)):
        if remaining_doors[i] == 0:
            del remaining_doors[i]
            break
    
    # Check the outcomes for staying or switching
    if chosen_door == 1:
        stay_wins += 1
    if remaining_doors[0] == 1:
        switch_wins += 1

# Print the results
print("Stay wins =", stay_wins)
print("Switch wins =", switch_wins)
