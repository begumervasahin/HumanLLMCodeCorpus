aunsorted = [6, 2, 7, 8, 3, 1, 10, 5, 4, 9]
asorted = []

while aunsorted:
    amin = aunsorted[0]
    aminindex = 0
    
    # Find the minimum element and its index in the unsorted list
    for i in range(len(aunsorted)):
        if aunsorted[i] < amin:
            amin = aunsorted[i]
            aminindex = i
    
    # Remove the minimum element from the unsorted list
    del aunsorted[aminindex]
    
    # Append the minimum element to the sorted list
    asorted.append(amin)

print(asorted)
