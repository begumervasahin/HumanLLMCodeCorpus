import functools

def sierpinski(n):
    """
    Generates the Sierpinski triangle of order n.

    Parameters:
    n (int): The order of the Sierpinski triangle.

    Returns:
    list: A list of strings representing the Sierpinski triangle.
    """
    def aggregate(TRIANGLE, I):
        SPACE = " " * (2 ** I)
        return [SPACE + X + SPACE for X in TRIANGLE] + [X + " " + X for X in TRIANGLE]
    
    return functools.reduce(aggregate, range(n), ["*"])

def main():
    # Generate the Sierpinski triangle of order 4 and print it
    triangle = sierpinski(4)
    print("\n".join(triangle))

if __name__ == "__main__":
    main()
