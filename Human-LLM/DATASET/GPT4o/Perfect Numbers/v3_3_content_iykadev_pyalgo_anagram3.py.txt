def anagram(s1, s2):
    """
    Determines if two strings are anagrams of each other.
    
    Args:
    s1 (str): The first string.
    s2 (str): The second string.
    
    Returns:
    bool: True if the strings are anagrams, False otherwise.
    """
    
    # Early exit if lengths of the strings are not the same
    if len(s1) != len(s2):
        return False
    
    # Initialize frequency counters for each string
    count1 = [0] * 26
    count2 = [0] * 26
    
    # Function to increment the frequency counter for a given string
    def count_frequency(string, count):
        for char in string:
            index = ord(char) - ord('a')
            count[index] += 1
    
    # Count the frequency of each letter in both strings
    count_frequency(s1, count1)
    count_frequency(s2, count2)
    
    # Compare the frequency counts of both strings
    return count1 == count2

# Test the function
print(anagram('pide', 'epid'))  # Expected output: True
print(anagram('hello', 'bello'))  # Expected output: False
