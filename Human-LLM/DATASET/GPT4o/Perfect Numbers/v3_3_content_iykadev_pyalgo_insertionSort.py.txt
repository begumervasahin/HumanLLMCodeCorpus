def insertion_sort(arr: list) -> None:
    """
    Sorts a list using the insertion sort algorithm.

    Parameters:
    arr (list): The list to be sorted.

    Returns:
    None: The list is sorted in place.
    """
    for index in range(1, len(arr)):
        current_value = arr[index]
        position = index

        # Shift elements of the sorted segment that are greater than the current value
        while position > 0 and arr[position - 1] > current_value:
            arr[position] = arr[position - 1]
            position -= 1

        # Place the current value in its correct position
        arr[position] = current_value

def main() -> None:
    """
    Main function to test the insertion sort function.
    """
    # Example usage of insertion_sort function
    unsorted_list = [54, 26, 93, 17, 77, 31, 44, 55, 20]
    print("Unsorted list:", unsorted_list)
    insertion_sort(unsorted_list)
    print("Sorted list:", unsorted_list)

if __name__ == "__main__":
    main()
