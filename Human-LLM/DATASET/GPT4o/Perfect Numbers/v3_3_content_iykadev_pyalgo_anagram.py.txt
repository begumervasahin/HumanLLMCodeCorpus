def are_anagrams(s1: str, s2: str) -> bool:
    """
    Checks if two strings are anagrams.

    Parameters:
    s1 (str): The first string.
    s2 (str): The second string.

    Returns:
    bool: True if the strings are anagrams, False otherwise.
    """
    # Early return if lengths do not match
    if len(s1) != len(s2):
        return False

    # Sort both strings and compare
    return sorted(s1) == sorted(s2)

def main() -> None:
    """
    Main function to test the are_anagrams function.
    """
    # Test the are_anagrams function
    result = are_anagrams('ey edip', 'pide ye')
    print(f"Are 'ey edip' and 'pide ye' anagrams? {result}")  # Output: True

if __name__ == "__main__":
    main()
