import random

stay = 0
switch = 0

for _ in range(1000):
    # Initialize the list with one prize (1) and two goats (0)
    lst = [1, 0, 0]
    random.shuffle(lst)
    
    # Contestant makes an initial choice
    ran = random.randrange(3)
    user_choice = lst[ran]
    
    # Remove the chosen door from the list
    del lst[ran]
    
    # Host reveals a goat
    for i in range(len(lst)):
        if lst[i] == 0:
            del lst[i]
            break
    
    # Check the outcomes
    if user_choice == 1:
        stay += 1
    if lst[0] == 1:
        switch += 1

print("Stay =", stay)
print("Switch =", switch)
