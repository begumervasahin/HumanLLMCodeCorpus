import time

def is_prime(num):
    """
    Check if a number is a prime number.
    
    Args:
        num (int): The number to check.
        
    Returns:
        bool: True if the number is prime, False otherwise.
    """
    if num < 2:
        return False
    if num == 2:
        return True
    for i in range(2, int(num ** 0.5) + 1):
        if num % i == 0:
            return False
    return True

def find_primes(limit):
    """
    Find all prime numbers up to a given limit.
    
    Args:
        limit (int): The upper limit for finding prime numbers.
        
    Returns:
        list: A list of prime numbers up to the given limit.
    """
    primes = [i for i in range(limit) if is_prime(i)]
    return primes

def main():
    start_time = time.time()
    
    # Find all prime numbers up to 10000
    primes = find_primes(10000)
    
    # Print the list of prime numbers
    print(primes)
    
    # Find and print Mersenne primes
    for prime in primes:
        mersenne_number = 2 ** prime - 1
        if is_prime(mersenne_number):
            elapsed_time = time.time() - start_time
            print(f"Time elapsed: {elapsed_time} seconds")
            print(f"Mersenne prime: {2 ** (prime - 1) * mersenne_number}")

if __name__ == '__main__':
    main()
