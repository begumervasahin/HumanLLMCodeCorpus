class Bitmap:
    def __init__(self, width, height):
        self.width = width
        self.height = height
        self.pixels = [[' ' for _ in range(width)] for _ in range(height)]

    def set(self, x, y):
        """Set a pixel at (x, y) to '#'."""
        if 0 <= x < self.width and 0 <= y < self.height:
            self.pixels[y][x] = '#'

    def line(self, x0, y0, x1, y1):
        """Draw a line from (x0, y0) to (x1, y1) using Bresenham's line algorithm."""
        dx = abs(x1 - x0)
        dy = abs(y1 - y0)
        x, y = x0, y0
        sx = -1 if x0 > x1 else 1
        sy = -1 if y0 > y1 else 1
        if dx > dy:
            err = dx / 2.0
            while x != x1:
                self.set(x, y)
                err -= dy
                if err < 0:
                    y += sy
                    err += dx
                x += sx
        else:
            err = dy / 2.0
            while y != y1:
                self.set(x, y)
                err -= dx
                if err < 0:
                    x += sx
                    err += dy
                y += sy
        self.set(x, y)

    def chardisplay(self):
        """Display the bitmap in the console."""
        for row in self.pixels:
            print(''.join(row))

# Create a bitmap and draw lines
bitmap = Bitmap(17, 17)
lines_to_draw = [(1, 8, 8, 16), (8, 16, 16, 8), (16, 8, 8, 1), (8, 1, 1, 8)]

for points in lines_to_draw:
    bitmap.line(*points)

bitmap.chardisplay()
