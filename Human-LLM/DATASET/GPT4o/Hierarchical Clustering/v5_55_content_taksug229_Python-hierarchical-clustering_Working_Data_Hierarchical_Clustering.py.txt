import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
import scipy.cluster.hierarchy as sch
from sklearn.cluster import AgglomerativeClustering

def load_data(filepath):
    """
    Load the dataset from a CSV file.

    Parameters:
    - filepath (str): The path to the CSV file.

    Returns:
    - ndarray: The feature matrix containing Annual Income and Spending Score.
    """
    dataset = pd.read_csv(filepath)
    return dataset.iloc[:, [3, 4]].values  # Select the features (Annual Income and Spending Score)

def plot_dendrogram(X):
    """
    Plot a dendrogram to help determine the optimal number of clusters.

    Parameters:
    - X (ndarray): The feature matrix.
    """
    plt.figure(figsize=(10, 7))
    sch.dendrogram(sch.linkage(X, method='ward'))
    plt.title('Dendrogram')
    plt.xlabel('Customers')
    plt.ylabel('Euclidean distances')
    plt.show()

def apply_agglomerative_clustering(X, n_clusters=5):
    """
    Apply Agglomerative Clustering to the data.

    Parameters:
    - X (ndarray): The feature matrix.
    - n_clusters (int): The number of clusters to form.

    Returns:
    - ndarray: The cluster labels for each data point.
    """
    model = AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage='ward')
    return model.fit_predict(X)

def plot_clusters(X, y_hc, cluster_labels, colors):
    """
    Plot the clusters formed by Agglomerative Clustering.

    Parameters:
    - X (ndarray): The feature matrix.
    - y_hc (ndarray): The cluster labels for each data point.
    - cluster_labels (list): The labels for each cluster.
    - colors (list): The colors for each cluster.
    """
    plt.figure(figsize=(10, 7))
    for i, color in enumerate(colors):
        plt.scatter(X[y_hc == i, 0], X[y_hc == i, 1], s=100, c=color, label=cluster_labels[i])
    
    plt.title('Clusters of Customers')
    plt.xlabel('Annual Income (k$)')
    plt.ylabel('Spending Score (1-100)')
    plt.legend()
    plt.show()

def main():
    # File path to the dataset
    filepath = 'Mall_Customers.csv'
    
    # Load the data
    X = load_data(filepath)
    
    # Plot the dendrogram
    plot_dendrogram(X)
    
    # Apply Agglomerative Clustering
    y_hc = apply_agglomerative_clustering(X)
    
    # Define cluster labels and colors
    cluster_labels = ['Careful', 'Standard', 'Target', 'Careless', 'Sensible']
    colors = ['red', 'blue', 'green', 'cyan', 'magenta']
    
    # Plot the clusters
    plot_clusters(X, y_hc, cluster_labels, colors)

if __name__ == "__main__":
    main()
