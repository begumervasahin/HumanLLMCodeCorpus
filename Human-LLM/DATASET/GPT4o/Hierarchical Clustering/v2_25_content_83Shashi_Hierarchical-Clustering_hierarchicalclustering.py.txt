import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import scipy.cluster.hierarchy as sch
from sklearn.cluster import AgglomerativeClustering

def load_data(file_path):
    """Loads the dataset from a CSV file."""
    return pd.read_csv(file_path)

def plot_dendrogram(X):
    """Plots a dendrogram to help determine the optimal number of clusters."""
    plt.figure(figsize=(10, 7))
    dendrogram = sch.dendrogram(sch.linkage(X, method='ward'))
    plt.title('Dendrogram')
    plt.xlabel('Customers')
    plt.ylabel('Euclidean Distance')
    plt.show()

def apply_agglomerative_clustering(X, n_clusters):
    """Applies Agglomerative Clustering to the data."""
    hc = AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage='ward')
    return hc.fit_predict(X)

def plot_clusters(X, y_hc):
    """Visualizes the clusters identified by Agglomerative Clustering."""
    plt.figure(figsize=(10, 7))
    plt.scatter(X[y_hc == 0, 0], X[y_hc == 0, 1], s=100, c='red', label='Careful')
    plt.scatter(X[y_hc == 1, 0], X[y_hc == 1, 1], s=100, c='blue', label='Standard')
    plt.scatter(X[y_hc == 2, 0], X[y_hc == 2, 1], s=100, c='green', label='Target')
    plt.scatter(X[y_hc == 3, 0], X[y_hc == 3, 1], s=100, c='cyan', label='Careless')
    plt.scatter(X[y_hc == 4, 0], X[y_hc == 4, 1], s=100, c='magenta', label='Sensible')
    plt.title('Clusters of Clients')
    plt.xlabel('Annual Income (K$)')
    plt.ylabel('Spending Score (1-100)')
    plt.legend()
    plt.show()

def main():
    # Load the dataset
    dataset = load_data("Mall_Customers.csv")
    X = dataset.iloc[:, [3, 4]].values  # Extract the relevant columns for clustering

    # Plot the Dendrogram
    plot_dendrogram(X)

    # Apply Agglomerative Clustering
    y_hc = apply_agglomerative_clustering(X, n_clusters=5)

    # Plot the clusters
    plot_clusters(X, y_hc)

if __name__ == "__main__":
    main()
