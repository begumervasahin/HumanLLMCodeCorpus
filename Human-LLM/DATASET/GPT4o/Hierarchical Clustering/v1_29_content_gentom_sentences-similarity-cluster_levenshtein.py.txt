def edit_distance(s1, s2):
    """
    Calculates the edit distance (Levenshtein distance) between two strings s1 and s2.

    Parameters:
    s1 (str): The first string.
    s2 (str): The second string.

    Returns:
    int: The edit distance between the two strings.
    """
    if len(s1) > len(s2):
        s1, s2 = s2, s1  # Ensure that s1 is the shorter string

    distances = range(len(s1) + 1)  # Initialize distance range

    for i2, c2 in enumerate(s2):
        distances_ = [i2 + 1]  # Start new row with the incremented index

        for i1, c1 in enumerate(s1):
            if c1 == c2:
                distances_.append(distances[i1])  # Characters match, carry forward the previous distance
            else:
                distances_.append(1 + min(
                    distances[i1],    # Substitution
                    distances[i1 + 1],  # Insertion
                    distances_[-1]     # Deletion
                ))

        distances = distances_  # Move to the next row

    return distances[-1]  # The last element contains the edit distance

def distance(s1, s2):
    """
    An alternative implementation of calculating the edit distance (Levenshtein distance)
    between two strings s1 and s2.

    Parameters:
    s1 (str): The first string.
    s2 (str): The second string.

    Returns:
    int: The edit distance between the two strings.
    """
    if len(s1) < len(s2):
        return distance(s2, s1)  # Ensure s1 is the longer string

    if len(s2) == 0:
        return len(s1)  # If s2 is empty, the distance is the length of s1

    previous_row = range(len(s2) + 1)  # Initialize the first row

    for i, c1 in enumerate(s1):
        current_row = [i + 1]  # Start the row with the incremented index

        for j, c2 in enumerate(s2):
            insertions = previous_row[j + 1] + 1
            deletions = current_row[j] + 1
            substitutions = previous_row[j] + (c1 != c2)

            current_row.append(min(insertions, deletions, substitutions))  # Choose the minimum edit operation

        previous_row = current_row  # Move to the next row

    return previous_row[-1]  # The last element contains the edit distance

# Example usage:
s1 = "kitten"
s2 = "sitting"

print(f"Edit distance between '{s1}' and '{s2}' using edit_distance function: {edit_distance(s1, s2)}")
print(f"Edit distance between '{s1}' and '{s2}' using distance function: {distance(s1, s2)}")
