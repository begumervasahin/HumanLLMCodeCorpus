import pandas as pd
from sklearn.preprocessing import StandardScaler
from matplotlib import pyplot as plt
from scipy.cluster.hierarchy import dendrogram, linkage, cophenet, maxdists
from scipy.spatial.distance import pdist

def load_and_standardize_data(file_path):
    """
    Load the dataset from a CSV file and standardize its features.

    Args:
        file_path (str): The path to the CSV file containing the dataset.

    Returns:
        np.ndarray: Standardized data ready for clustering.
    """
    data = pd.read_csv(file_path)
    scaler = StandardScaler()
    standardized_data = scaler.fit_transform(data)
    return standardized_data

def hierarchical_clustering(data, method='average'):
    """
    Perform hierarchical clustering on the standardized data.

    Args:
        data (np.ndarray): The standardized data.
        method (str): The linkage method to use for clustering. Defaults to 'average'.

    Returns:
        tuple: Contains the linkage matrix, cophenet correlation coefficient, cophenet distances, and max distances.
    """
    linkage_matrix = linkage(data, method=method)
    cophenet_corr, cophenet_distances = cophenet(linkage_matrix, pdist(data))
    max_cluster_distances = maxdists(linkage_matrix)
    return linkage_matrix, cophenet_corr, cophenet_distances, max_cluster_distances

def plot_dendrogram(linkage_matrix, title, truncate_mode=None, p=None):
    """
    Plot a dendrogram for the hierarchical clustering results.

    Args:
        linkage_matrix (np.ndarray): The linkage matrix resulting from hierarchical clustering.
        title (str): The title for the dendrogram plot.
        truncate_mode (str, optional): Truncation mode for the dendrogram ('lastp' to show only the last `p` merged clusters).
        p (int, optional): The number of last clusters to show when truncation is enabled.
    """
    plt.figure(figsize=(30, 15))
    plt.title(title)
    plt.xlabel('Sample Index')
    plt.ylabel('Distance')
    
    dendrogram(
        linkage_matrix,
        leaf_rotation=90.,
        leaf_font_size=10.,
        truncate_mode=truncate_mode,
        p=p,
        show_contracted=True if truncate_mode else False
    )
    plt.show()

def main():
    """
    Main function to execute the hierarchical clustering process on the Flag dataset.
    """
    # Define the path to the dataset
    file_path = 'Flagdata.csv'
    
    # Load and standardize the data
    standardized_data = load_and_standardize_data(file_path)
    
    # Perform hierarchical clustering
    linkage_matrix, cophenet_corr, cophenet_distances, max_distances = hierarchical_clustering(standardized_data)

    # Output the results
    print(f'Cophenet Correlation Coefficient: {cophenet_corr}')
    print(f'Cophenet Pairwise Distances: {cophenet_distances[:10]}')  # Print the first 10 for brevity
    print(f'First Cluster: {linkage_matrix[0]}')
    print(f'Max Cluster Distances: {max_distances[:10]}')  # Print the first 10 for brevity

    # Plot the full dendrogram
    plot_dendrogram(linkage_matrix, 'Dendrogram for Flag Data')
    
    # Plot a truncated dendrogram showing the last 12 merged clusters
    plot_dendrogram(linkage_matrix, 'Truncated Dendrogram (Last 12 Merged Clusters)', truncate_mode='lastp', p=12)

if __name__ == "__main__":
    main()
