import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
import scipy.cluster.hierarchy as sch
from sklearn.cluster import AgglomerativeClustering

def load_data(file_path):
    """
    Load dataset from a CSV file.

    Args:
        file_path (str): The path to the CSV file.

    Returns:
        pd.DataFrame: Loaded data as a pandas DataFrame.
    """
    return pd.read_csv(file_path)

def extract_features(data, columns):
    """
    Extract specific columns for clustering.

    Args:
        data (pd.DataFrame): The dataset from which to extract features.
        columns (list): List of column indices to extract.

    Returns:
        np.ndarray: The extracted features as a NumPy array.
    """
    return data.iloc[:, columns].values

def plot_dendrogram(X):
    """
    Plot a dendrogram for hierarchical clustering.

    Args:
        X (np.ndarray): The data to be used for plotting the dendrogram.
    """
    plt.figure(figsize=(10, 7))
    sch.dendrogram(sch.linkage(X, method='ward'))
    plt.title('Dendrogram')
    plt.xlabel('Customers')
    plt.ylabel('Euclidean Distances')
    plt.show()

def perform_clustering(X, n_clusters=5):
    """
    Apply Agglomerative Clustering to the data.

    Args:
        X (np.ndarray): The data to cluster.
        n_clusters (int, optional): Number of clusters to find. Defaults to 5.

    Returns:
        np.ndarray: Cluster labels for each data point.
    """
    clustering_model = AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage='ward')
    return clustering_model.fit_predict(X)

def plot_clusters(X, cluster_labels):
    """
    Visualize the clusters of customers.

    Args:
        X (np.ndarray): The data used for clustering.
        cluster_labels (np.ndarray): The cluster labels for each data point.
    """
    plt.figure(figsize=(10, 7))

    colors = ['red', 'blue', 'green', 'cyan', 'magenta']
    labels = [f'Cluster {i+1}' for i in range(len(colors))]

    for i, color in enumerate(colors):
        plt.scatter(X[cluster_labels == i, 0], X[cluster_labels == i, 1],
                    s=100, c=color, label=labels[i])

    plt.title('Clusters of Customers')
    plt.xlabel('Annual Income (k$)')
    plt.ylabel('Spending Score (1-100)')
    plt.legend()
    plt.show()

def main():
    """
    Main function to perform hierarchical clustering on customer data.
    """
    # Load the dataset
    data = load_data('Mall_Customers.csv')

    # Extract the features for clustering
    X = extract_features(data, [3, 4])

    # Plot the Dendrogram to determine the number of clusters
    plot_dendrogram(X)

    # Perform Agglomerative Clustering
    cluster_labels = perform_clustering(X)

    # Plot the resulting clusters
    plot_clusters(X, cluster_labels)

if __name__ == "__main__":
    main()
