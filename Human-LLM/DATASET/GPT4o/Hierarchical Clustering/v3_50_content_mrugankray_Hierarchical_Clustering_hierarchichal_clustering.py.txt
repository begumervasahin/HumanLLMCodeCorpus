import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
import scipy.cluster.hierarchy as sch
from sklearn.cluster import AgglomerativeClustering

def load_dataset(filename):
    """Load dataset from a CSV file and extract relevant features."""
    dataset = pd.read_csv(filename)
    return dataset.iloc[:, [3, 4]].values

def plot_dendrogram(X):
    """Plot a dendrogram to help determine the optimal number of clusters."""
    plt.figure(figsize=(10, 7))
    plt.title('Dendrogram')
    plt.xlabel('Customers')
    plt.ylabel('Euclidean Distances')
    sch.dendrogram(sch.linkage(X, method='ward'))
    plt.show()

def apply_agglomerative_clustering(X, n_clusters):
    """Apply Agglomerative Clustering to the data and return the cluster labels."""
    hc = AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage='ward')
    return hc.fit_predict(X)

def plot_clusters(X, y_hc, n_clusters):
    """Visualize the clusters with a scatter plot."""
    plt.figure(figsize=(10, 7))
    
    colors = ['red', 'blue', 'magenta', 'cyan', 'green']
    labels = ['Careful', 'Standard', 'Target', 'Careless', 'Sensible']
    
    for i in range(n_clusters):
        plt.scatter(X[y_hc == i, 0], X[y_hc == i, 1], s=100, c=colors[i], label=labels[i])
    
    plt.title('Hierarchical Clustering of Customers')
    plt.xlabel('Annual Income (k$)')
    plt.ylabel('Spending Score (1-100)')
    plt.legend()
    plt.show()

def main():
    """Main function to execute the clustering process."""
    # Load the dataset
    X = load_dataset('Mall_Customers.csv')
    
    # Plot the dendrogram to find the optimal number of clusters
    plot_dendrogram(X)
    
    # Apply Agglomerative Clustering with the chosen number of clusters
    n_clusters = 5
    y_hc = apply_agglomerative_clustering(X, n_clusters)
    
    # Visualize the resulting clusters
    plot_clusters(X, y_hc, n_clusters)

if __name__ == "__main__":
    main()
