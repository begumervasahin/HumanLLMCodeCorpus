import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from sklearn.cluster import AgglomerativeClustering
import scipy.cluster.hierarchy as sch

def load_dataset(file_path):
    """Loads the dataset from a CSV file and returns the features needed for clustering."""
    dataset = pd.read_csv(file_path)
    return dataset.iloc[:, [3, 4]].values  # Selects the columns for Annual Income and Spending Score

def plot_dendrogram(data):
    """Plots a dendrogram to help determine the optimal number of clusters."""
    plt.figure(figsize=(10, 7))
    dendrogram = sch.dendrogram(sch.linkage(data, method='ward'))
    plt.title('Dendrogram')
    plt.xlabel('Customers')
    plt.ylabel('Euclidean Distances')
    plt.show()

def perform_agglomerative_clustering(data, n_clusters):
    """Applies agglomerative clustering to the dataset."""
    model = AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage='ward')
    return model.fit_predict(data)

def plot_clusters(data, cluster_labels):
    """Plots the clusters identified by hierarchical clustering."""
    plt.figure(figsize=(10, 7))
    colors = ['red', 'blue', 'green', 'cyan', 'magenta']

    for i, color in enumerate(colors):
        plt.scatter(data[cluster_labels == i, 0], data[cluster_labels == i, 1],
                    s=100, c=color, label=f'Cluster {i + 1}')
    
    plt.title('Clusters of Customers')
    plt.xlabel('Annual Income (k$)')
    plt.ylabel('Spending Score (1-100)')
    plt.legend()
    plt.show()

def main():
    # Step 1: Load the dataset and extract relevant features
    data = load_dataset('Mall_Customers.csv')

    # Step 2: Plot the dendrogram to visualize potential clusters
    plot_dendrogram(data)

    # Step 3: Perform agglomerative clustering based on the dendrogram
    cluster_labels = perform_agglomerative_clustering(data, n_clusters=5)

    # Step 4: Visualize the resulting clusters
    plot_clusters(data, cluster_labels)

if __name__ == "__main__":
    main()
