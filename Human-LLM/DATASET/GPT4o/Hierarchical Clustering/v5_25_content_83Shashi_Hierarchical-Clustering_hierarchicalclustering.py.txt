import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import scipy.cluster.hierarchy as sch
from sklearn.cluster import AgglomerativeClustering

def load_data(file_path):
    """Loads the dataset from a CSV file and extracts the relevant features."""
    dataset = pd.read_csv(file_path)
    return dataset.iloc[:, [3, 4]].values  # Extract Annual Income and Spending Score

def plot_dendrogram(X):
    """Plots a dendrogram to help determine the optimal number of clusters."""
    plt.figure(figsize=(10, 7))
    dendrogram = sch.dendrogram(sch.linkage(X, method='ward'))
    plt.title('Dendrogram')
    plt.xlabel('Customers')
    plt.ylabel('Euclidean Distance')
    plt.show()

def apply_agglomerative_clustering(X, n_clusters):
    """Applies Agglomerative Clustering to the dataset and returns the cluster labels."""
    hc = AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage='ward')
    return hc.fit_predict(X)

def plot_clusters(X, y_hc):
    """Visualizes the clusters identified by Agglomerative Clustering."""
    plt.figure(figsize=(10, 7))
    cluster_colors = ['red', 'blue', 'green', 'cyan', 'magenta']
    cluster_labels = ['Careful', 'Standard', 'Target', 'Careless', 'Sensible']
    
    for i, color in enumerate(cluster_colors):
        plt.scatter(X[y_hc == i, 0], X[y_hc == i, 1], s=100, c=color, label=cluster_labels[i])
    
    plt.title('Clusters of Clients')
    plt.xlabel('Annual Income (K$)')
    plt.ylabel('Spending Score (1-100)')
    plt.legend()
    plt.show()

def main():
    # Load the dataset and extract features
    X = load_data("Mall_Customers.csv")

    # Plot the dendrogram to determine the optimal number of clusters
    plot_dendrogram(X)

    # Apply Agglomerative Clustering and get cluster labels
    y_hc = apply_agglomerative_clustering(X, n_clusters=5)

    # Plot the resulting clusters
    plot_clusters(X, y_hc)

if __name__ == "__main__":
    main()
