import nltk
import re
from nltk.corpus import stopwords as sw
from nltk.corpus import wordnet as wn

# Ensure necessary NLTK resources are available
nltk.download('stopwords')
nltk.download('wordnet')

def preprocess_text(text):
    """
    Processes the input text by removing stopwords, converting to lowercase, and applying stemming.

    Parameters:
    text (str): The input text to preprocess.

    Returns:
    list: A list of preprocessed words.
    """
    words = text.split()
    porter_stemmer = nltk.PorterStemmer()

    # Filter out stopwords and convert to lowercase
    filtered_words = [word.lower() for word in words if word.lower() not in sw.words('english')]

    # Apply stemming
    stemmed_words = [porter_stemmer.stem(word) for word in filtered_words]

    return stemmed_words

def calculate_similarity(words1, words2):
    """
    Calculates the semantic similarity between two lists of words using WordNet path similarity.

    Parameters:
    words1 (list): A list of words from the first text.
    words2 (list): A list of words from the second text.

    Returns:
    float: The average semantic similarity score between the two word lists.
    """
    total_similarity = 0

    try:
        for word1 in words1:
            synsets1 = wn.synsets(word1)
            if synsets1:
                synset1 = synsets1[0]
                for word2 in words2:
                    synsets2 = wn.synsets(word2)
                    if synsets2:
                        synset2 = synsets2[0]
                        similarity = synset1.path_similarity(synset2)
                        if similarity:
                            total_similarity += similarity
    except Exception as e:
        print(f"Error occurred: {str(e)}")

    # Calculate the average similarity
    num_comparisons = len(words1) * len(words2)
    average_similarity = total_similarity / num_comparisons if num_comparisons > 0 else 0
    
    return average_similarity

def print_similarity(text1, text2):
    """
    Prints the semantic similarity between two texts.

    Parameters:
    text1 (str): The first text.
    text2 (str): The second text.
    """
    words1 = preprocess_text(text1)
    words2 = preprocess_text(text2)

    similarity = calculate_similarity(words1, words2)
    print(f"Semantic similarity between '{text1}' and '{text2}': {similarity:.4f}")

# Example usage
print_similarity("I am loved by everyone", "Everyone loves me")
