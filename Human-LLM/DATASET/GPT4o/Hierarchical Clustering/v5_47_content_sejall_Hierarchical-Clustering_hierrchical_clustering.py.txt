import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
import scipy.cluster.hierarchy as sch
from sklearn.cluster import AgglomerativeClustering

def load_dataset(file_path):
    """
    Load the dataset and extract relevant features (Annual Income and Spending Score).

    Args:
        file_path (str): The path to the CSV file containing the dataset.

    Returns:
        np.ndarray: A NumPy array containing the extracted features for clustering.
    """
    dataset = pd.read_csv(file_path)
    return dataset.iloc[:, [3, 4]].values

def plot_dendrogram(X):
    """
    Plot the dendrogram to determine the optimal number of clusters.

    Args:
        X (np.ndarray): The data to be used for plotting the dendrogram.
    """
    plt.figure(figsize=(10, 7))
    sch.dendrogram(sch.linkage(X, method='ward'))
    plt.title('Dendrogram')
    plt.xlabel('Customers')
    plt.ylabel('Euclidean Distance')
    plt.show()

def perform_clustering(X, n_clusters=5):
    """
    Perform Agglomerative Clustering on the dataset.

    Args:
        X (np.ndarray): The data to be clustered.
        n_clusters (int): The number of clusters to form.

    Returns:
        np.ndarray: Cluster labels for each data point.
    """
    model = AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage='ward')
    return model.fit_predict(X)

def plot_clusters(X, y_hc):
    """
    Visualize the clusters formed by Agglomerative Clustering.

    Args:
        X (np.ndarray): The data used for clustering.
        y_hc (np.ndarray): Cluster labels for the data points.
    """
    plt.figure(figsize=(10, 7))

    # Define the cluster labels and colors
    cluster_definitions = {
        0: ('red', 'Careful'),
        1: ('blue', 'Standard'),
        2: ('green', 'Target'),
        3: ('cyan', 'Careless'),
        4: ('magenta', 'Sensible')
    }

    # Plot each cluster
    for cluster_id, (color, label) in cluster_definitions.items():
        plt.scatter(X[y_hc == cluster_id, 0], X[y_hc == cluster_id, 1], 
                    s=100, c=color, label=label)

    plt.title('Clusters of Customers')
    plt.xlabel('Annual Income (k$)')
    plt.ylabel('Spending Score (1-100)')
    plt.legend()
    plt.show()

def main():
    """
    Main function to execute the clustering and visualization process.
    """
    # Define the file path to the dataset
    file_path = 'Mall_Customers.csv'

    # Load the dataset and extract features
    X = load_dataset(file_path)

    # Plot the dendrogram to determine the optimal number of clusters
    plot_dendrogram(X)

    # Perform Agglomerative Clustering
    y_hc = perform_clustering(X, n_clusters=5)

    # Visualize the resulting clusters
    plot_clusters(X, y_hc)

if __name__ == "__main__":
    main()
