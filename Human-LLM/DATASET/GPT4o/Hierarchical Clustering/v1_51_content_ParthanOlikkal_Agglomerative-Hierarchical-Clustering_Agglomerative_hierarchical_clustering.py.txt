import numpy as np
import pandas as pd
from scipy import ndimage
from scipy.cluster import hierarchy
from scipy.spatial import distance_matrix
from matplotlib import pyplot as plt
from sklearn.cluster import AgglomerativeClustering
from sklearn.datasets import make_blobs

# Generate synthetic data
X1, y1 = make_blobs(n_samples=50, centers=[[4, 4], [-2, -1], [1, 1], [10, 4]], cluster_std=0.9)

# Plot the generated data points
plt.scatter(X1[:, 0], X1[:, 1], marker='o')
plt.title('Generated Data Points')
plt.xlabel('Feature 1')
plt.ylabel('Feature 2')
plt.show()

# Apply Agglomerative Clustering
agglom = AgglomerativeClustering(n_clusters=4, linkage='average')
agglom.fit(X1)

# Plot the clustering results
plt.figure(figsize=(6, 4))
x_min, x_max = np.min(X1, axis=0), np.max(X1, axis=0)
X1_normalized = (X1 - x_min) / (x_max - x_min)

for i in range(X1_normalized.shape[0]):
    plt.text(X1_normalized[i, 0], X1_normalized[i, 1], str(y1[i]),
             color=plt.cm.nipy_spectral(agglom.labels_[i] / 10.),
             fontdict={'weight': 'bold', 'size': 9})

plt.xticks([])
plt.yticks([])
plt.scatter(X1_normalized[:, 0], X1_normalized[:, 1], marker='.')
plt.title('Agglomerative Clustering')
plt.show()

# Compute the distance matrix
dist_matrix = distance_matrix(X1_normalized, X1_normalized)
print("Distance Matrix:\n", dist_matrix)

# Perform hierarchical clustering and plot the dendrogram
Z = hierarchy.linkage(dist_matrix, 'complete')
plt.figure(figsize=(10, 7))
dendro = hierarchy.dendrogram(Z)
plt.title('Dendrogram')
plt.xlabel('Sample Index')
plt.ylabel('Distance')
plt.show()
