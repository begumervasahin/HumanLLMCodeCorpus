import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from scipy.spatial.distance import pdist
from scipy.cluster.hierarchy import linkage, fcluster

def load_country_data(filepath):
    """
    Load country data from a CSV file.

    Parameters:
    - filepath (str): The path to the CSV file.

    Returns:
    - pd.DataFrame: A DataFrame containing the loaded data.
    """
    return pd.read_csv(filepath)

def plot_country_locations(coords, country_names):
    """
    Plot country locations on a scatter plot with their names.

    Parameters:
    - coords (ndarray): An array of (x, y) coordinates for the countries.
    - country_names (ndarray): An array of country names corresponding to the coordinates.
    """
    plt.figure(figsize=(8, 12))
    plt.subplot(211)

    for (x, y), name in zip(coords, country_names):
        plt.scatter(x, y)
        plt.text(x, y, name, fontsize=6)

    plt.title('Countries')
    plt.grid(True)
    plt.xticks([])
    plt.yticks([])

def plot_country_clusters(coords, clusters):
    """
    Plot the clustered countries on a scatter plot.

    Parameters:
    - coords (ndarray): An array of (x, y) coordinates for the countries.
    - clusters (ndarray): An array of cluster labels for each country.
    """
    cluster_colors = ['r', 'g', 'b', 'm']
    plt.subplot(212)

    for color, cluster_id in zip(cluster_colors, np.unique(clusters)):
        cluster_points = clusters == cluster_id
        plt.scatter(coords[cluster_points, 0], coords[cluster_points, 1], c=color, label=f'Cluster {cluster_id}')

    plt.title('Clusters')
    plt.grid(True)
    plt.xticks([])
    plt.yticks([])
    plt.legend()

def perform_hierarchical_clustering(coords, num_clusters=4, method='single'):
    """
    Perform hierarchical clustering on country coordinates.

    Parameters:
    - coords (ndarray): An array of (x, y) coordinates for the countries.
    - num_clusters (int): The number of clusters to form.
    - method (str): The linkage method to use for clustering (default is 'single').

    Returns:
    - ndarray: An array of cluster labels for each country.
    """
    distance_matrix = pdist(coords, 'euclidean')
    Z = linkage(distance_matrix, method=method)
    clusters = fcluster(Z, t=num_clusters, criterion='maxclust')
    return clusters

def main(filepath):
    """
    Main function to execute the data loading, clustering, and plotting.

    Parameters:
    - filepath (str): The path to the CSV file containing country data.
    """
    data = load_country_data(filepath)
    coords = data[['x-coord', 'y-coord']].values
    country_names = data['CountryName'].values

    plot_country_locations(coords, country_names)

    clusters = perform_hierarchical_clustering(coords)
    plot_country_clusters(coords, clusters)

    plt.tight_layout()
    plt.show()

if __name__ == "__main__":
    filepath = 'final_countries_data.csv'
    main(filepath)
