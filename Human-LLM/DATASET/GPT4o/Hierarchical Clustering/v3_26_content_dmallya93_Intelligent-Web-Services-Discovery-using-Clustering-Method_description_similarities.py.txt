import nltk
from nltk.corpus import stopwords
from nltk.corpus import wordnet as wn

# Ensure the necessary NLTK resources are downloaded
nltk.download('stopwords')
nltk.download('wordnet')

def preprocess_text(text):
    """
    Processes the input text by removing stopwords, converting to lowercase, and applying stemming.

    Parameters:
    text (str): The input text to preprocess.

    Returns:
    list: A list of preprocessed words.
    """
    # Initialize the Porter stemmer
    porter_stemmer = nltk.PorterStemmer()

    # Split text into words, remove stopwords, convert to lowercase, and apply stemming
    filtered_words = [
        porter_stemmer.stem(word.lower()) 
        for word in text.split() 
        if word.lower() not in stopwords.words('english')
    ]
    
    return filtered_words

def calculate_similarity(text1, text2):
    """
    Calculates the semantic similarity between two sentences using WordNet path similarity.

    Parameters:
    text1 (str): The first sentence.
    text2 (str): The second sentence.

    Returns:
    float: The average semantic similarity score between the two sentences.
    """
    words1 = preprocess_text(text1)
    words2 = preprocess_text(text2)
    total_similarity = 0

    try:
        for word1 in words1:
            synsets1 = wn.synsets(word1)
            if synsets1:
                synset1 = synsets1[0]
                for word2 in words2:
                    synsets2 = wn.synsets(word2)
                    if synsets2:
                        synset2 = synsets2[0]
                        similarity = synset1.path_similarity(synset2)
                        if similarity:
                            total_similarity += similarity
    except Exception as e:
        print(f"Error occurred: {str(e)}")
    
    # Calculate the average similarity
    num_comparisons = len(words1) * len(words2)
    average_similarity = total_similarity / num_comparisons if num_comparisons > 0 else 0
    
    print(f"Semantic similarity: {average_similarity:.4f}")
    return average_similarity

# Example usage
calculate_similarity("I am loved by everyone", "Everyone loves me")
