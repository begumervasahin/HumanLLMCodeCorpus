import pandas as pd

def load_prediction_data(file_path):
    """
    Load prediction data from a CSV file and return it as a flattened list.

    Args:
        file_path (str): Path to the CSV file.

    Returns:
        list: A flattened list containing the prediction data.
    """
    return pd.read_csv(file_path).values.flatten().tolist()

def identify_valid_sequences(pred_data, min_length=4):
    """
    Identify valid sequences in the prediction data where values are 0.5 or 1.

    A valid sequence is one that:
    - Has a length greater than `min_length`.
    - Contains at least one value of 1.

    Args:
        pred_data (list): The prediction data as a list.
        min_length (int): Minimum length for a sequence to be considered valid.

    Returns:
        tuple: Two lists containing the start and end indices of each valid sequence.
    """
    start_indices = []
    end_indices = []
    idx = 0
    data_length = len(pred_data)

    while idx < data_length:
        count = 0
        ones_count = 0

        if pred_data[idx] in [0.5, 1]:
            start_idx = idx

            while idx < data_length and pred_data[idx] in [0.5, 1]:
                if pred_data[idx] == 1:
                    ones_count += 1
                count += 1
                idx += 1

            end_idx = idx

            if count >= min_length and ones_count > 0:
                start_indices.append(start_idx)
                end_indices.append(end_idx)
        else:
            idx += 1

    return start_indices, end_indices

def save_sequences_to_csv(start_indices, end_indices, output_file):
    """
    Save the start and end indices of valid sequences to a CSV file.

    Args:
        start_indices (list): List of start indices of valid sequences.
        end_indices (list): List of end indices of valid sequences.
        output_file (str): Path to the output CSV file.
    """
    sequences = pd.DataFrame({
        "Start Time": start_indices,
        "End Time": end_indices
    })
    sequences.to_csv(output_file, index=False)

def main():
    """
    Main function to load prediction data, identify valid sequences, and save the results.
    """
    # Load the prediction data
    prediction_data = load_prediction_data("y_prediction_itzy.csv")
    
    # Identify valid sequences in the prediction data
    start_indices, end_indices = identify_valid_sequences(prediction_data)
    
    # Save the identified sequences to a CSV file
    save_sequences_to_csv(start_indices, end_indices, "timedata_itzy.csv")

if __name__ == "__main__":
    main()
