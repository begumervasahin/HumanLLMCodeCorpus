import sys
import pandas as pd

def intersect(l1, l2):
    """Find the intersection of two lists."""
    return list(set(l1) & set(l2))

def issubset(l, m):
    """Check if list l is a subset of list m."""
    return all(elem in m for elem in l)

def diff(l1, l2):
    """Find the difference between two lists."""
    return list(set(l1) - set(l2))

def eclat(P, minsupport, F, FrequentSets):
    """Eclat algorithm for frequent itemset mining."""
    if len(P) == 1:
        FrequentSets.append(P[0])

    for i in range(len(P)):
        F.append(P[i])
        Pa = []
        for j in range(i + 1, len(P)):
            Xab = list(set(P[i][0] + P[j][0]))
            Tab = intersect(P[i][1], P[j][1])
            if len(Tab) >= minsupport:
                Pa.append((Xab, Tab, len(Xab)))
        if Pa:
            eclat(Pa, minsupport, F, FrequentSets)
        else:
            if len(P) != 1:
                FrequentSets.append(P[i])

def declat(Q, minsupport, IF, InFrequentSets):
    """Declat algorithm for infrequent itemset mining."""
    if len(Q) == 1:
        InFrequentSets.append(Q[0])

    for i in range(len(Q)):
        IF.append(Q[i])
        Qa = []
        for j in range(i + 1, len(Q)):
            Xab = list(set(Q[i][0] + Q[j][0]))
            Dab = diff(Q[j][2], Q[i][2])
            Sab = Q[i][1] - len(Dab)
            if Sab >= minsupport:
                Qa.append((Xab, Sab, Dab, len(Xab)))
        if Qa:
            declat(Qa, minsupport, IF, InFrequentSets)
        else:
            if len(Q) != 1:
                InFrequentSets.append(Q[i])

def main():
    """Main function to execute the Eclat and Declat algorithms."""
    F = []
    IF = []
    P = []
    Q = []
    
    minsupport = int(input('Enter minsupport: '))
    
    database = pd.read_csv('./db.csv', header=None)
    itemsets = ["TID"] + [f"i{col}" for col in range(1, len(database.columns))]
    database.columns = itemsets

    rowCount = database["TID"].count()
    database.drop(columns=["TID"], inplace=True)
    itemsets.remove("TID")

    T = list(range(1, rowCount + 1))
    for colName in itemsets:
        tmp = [index + 1 for index in range(rowCount) if database[colName][index] == 1]
        diffIndex = diff(T, tmp)
        if len(tmp) >= minsupport:
            P.append((colName, tmp, 1))
            Q.append((colName, len(tmp), diffIndex, 1))
    
    print("\nECLAT Algorithm")
    FrequentSets = []
    eclat(P, minsupport, F, FrequentSets)
    
    FrequentSets = sorted(FrequentSets, key=lambda x: x[2], reverse=True)
    print("Frequent Itemsets || Transaction IDs where present")
    for a, b, c in FrequentSets:
        print(''.join(a), "=>", b)
    
    MFSeclat = []
    for f in FrequentSets:
        if all(not issubset(f[0], m[0]) for m in MFSeclat):
            MFSeclat.append(f)

    print("\nMaximal Frequent Sets || Transaction IDs where present")
    for a, b, c in MFSeclat:
        print(''.join(a), "=>", b)
    
    print("\nDECLAT Algorithm")
    InFrequentSets = []
    declat(Q, minsupport, IF, InFrequentSets)

    InFrequentSets = sorted(InFrequentSets, key=lambda x: x[3], reverse=True)
    print("Infrequent Itemsets || Support of Itemsets")
    for a, b, c, d in InFrequentSets:
        print(''.join(a), "=>", b)

    MFSeclat = []
    for f in InFrequentSets:
        if all(not issubset(f[0], m[0]) for m in MFSeclat):
            MFSeclat.append(f)

    print("\nMaximal Infrequent Sets || Transaction IDs where present")
    for a, b, c, d in MFSeclat:
        print(''.join(a), "=>", b)

if __name__ == '__main__':
    main()
