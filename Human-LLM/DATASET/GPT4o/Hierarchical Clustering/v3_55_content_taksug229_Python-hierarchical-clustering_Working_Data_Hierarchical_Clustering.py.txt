import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
import scipy.cluster.hierarchy as sch
from sklearn.cluster import AgglomerativeClustering

def load_data(filepath):
    """
    Load dataset from a CSV file and extract the relevant features.

    Parameters:
    - filepath (str): The path to the CSV file.

    Returns:
    - ndarray: The feature matrix (Annual Income and Spending Score).
    """
    dataset = pd.read_csv(filepath)
    return dataset.iloc[:, [3, 4]].values  # Extract Annual Income and Spending Score

def plot_dendrogram(X):
    """
    Plot a dendrogram to help determine the optimal number of clusters.

    Parameters:
    - X (ndarray): The feature matrix.
    """
    plt.figure(figsize=(10, 7))
    sch.dendrogram(sch.linkage(X, method='ward'))
    plt.title('Dendrogram')
    plt.xlabel('Customers')
    plt.ylabel('Euclidean distances')
    plt.show()

def perform_clustering(X, n_clusters=5):
    """
    Perform Agglomerative Clustering on the data.

    Parameters:
    - X (ndarray): The feature matrix.
    - n_clusters (int): The number of clusters.

    Returns:
    - ndarray: The cluster labels for each data point.
    """
    clustering_model = AgglomerativeClustering(n_clusters=n_clusters, affinity='euclidean', linkage='ward')
    return clustering_model.fit_predict(X)

def plot_clusters(X, labels, cluster_labels):
    """
    Plot the clusters formed by Agglomerative Clustering.

    Parameters:
    - X (ndarray): The feature matrix.
    - labels (list): The list of labels for the clusters.
    - cluster_labels (ndarray): The cluster labels for each data point.
    """
    plt.figure(figsize=(10, 7))
    colors = ['red', 'blue', 'green', 'cyan', 'magenta']
    
    for i, color in enumerate(colors):
        plt.scatter(X[cluster_labels == i, 0], X[cluster_labels == i, 1], s=100, c=color, label=labels[i])
    
    plt.title('Clusters of customers')
    plt.xlabel('Annual Income (k$)')
    plt.ylabel('Spending Score (1-100)')
    plt.legend()
    plt.show()

def main():
    # Define cluster labels
    cluster_labels = ['Careful', 'Standard', 'Target', 'Careless', 'Sensible']
    
    # Load the data
    X = load_data('Mall_Customers.csv')
    
    # Plot the dendrogram
    plot_dendrogram(X)
    
    # Perform clustering
    y_hc = perform_clustering(X)
    
    # Plot the clusters
    plot_clusters(X, cluster_labels, y_hc)

if __name__ == "__main__":
    main()
