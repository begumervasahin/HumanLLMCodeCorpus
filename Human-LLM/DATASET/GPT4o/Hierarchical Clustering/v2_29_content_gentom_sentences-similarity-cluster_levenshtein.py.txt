def edit_distance(s1, s2):
    """
    Calculate the edit distance (Levenshtein distance) between two strings s1 and s2.

    Parameters:
    s1 (str): The first string.
    s2 (str): The second string.

    Returns:
    int: The edit distance between s1 and s2.
    """
    # Ensure s1 is the shorter string
    if len(s1) > len(s2)):
        s1, s2 = s2, s1  

    # Initialize the distances list
    distances = range(len(s1) + 1)  

    # Compute the edit distance
    for i2, c2 in enumerate(s2):
        new_distances = [i2 + 1]  # Start new row with incremented index

        for i1, c1 in enumerate(s1):
            if c1 == c2:
                new_distances.append(distances[i1])  # No operation needed if characters match
            else:
                new_distances.append(1 + min(
                    distances[i1],      # Substitution
                    distances[i1 + 1],  # Insertion
                    new_distances[-1]   # Deletion
                ))

        distances = new_distances  # Update distances for the next iteration

    return distances[-1]  # Return the final computed edit distance


def alternative_distance(s1, s2):
    """
    Another implementation of the Levenshtein distance calculation between two strings.

    Parameters:
    s1 (str): The first string.
    s2 (str): The second string.

    Returns:
    int: The edit distance between s1 and s2.
    """
    # Ensure s1 is the longer string
    if len(s1) < len(s2):
        return alternative_distance(s2, s1)  

    # If the second string is empty, the distance is the length of the first string
    if len(s2) == 0:
        return len(s1)  

    previous_row = range(len(s2) + 1)  # Initialize the first row

    # Compute the edit distance row by row
    for i, c1 in enumerate(s1):
        current_row = [i + 1]  # Start the row with the incremented index

        for j, c2 in enumerate(s2):
            insertions = previous_row[j + 1] + 1
            deletions = current_row[j] + 1
            substitutions = previous_row[j] + (c1 != c2)

            current_row.append(min(insertions, deletions, substitutions))  # Choose the minimum operation cost

        previous_row = current_row  # Update previous_row for the next iteration

    return previous_row[-1]  # Return the final computed edit distance

    print(f"Edit distance between '{s1}' and '{s2}' using edit_distance function: {edit_distance(s1, s2)}")
    print(f"Edit distance between '{s1}' and '{s2}' using alternative_distance function: {alternative_distance(s1, s2)}")
