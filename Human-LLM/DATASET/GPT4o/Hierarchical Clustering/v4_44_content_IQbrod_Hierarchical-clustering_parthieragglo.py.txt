class Group:
    def __init__(self, elements, similarity, index, node):
        """
        Initialize a Group object.

        Args:
            elements (list): List of elements in the group.
            similarity (float): The similarity score of the group.
            index (int): The index of the element with the highest similarity.
            node (Node): The node representing the group in the binary tree.
        """
        self.elements = elements
        self.similarity = similarity
        self.index = index
        self.node = node

    def __str__(self):
        return f"{self.elements} => {self.similarity} (Index: {self.index})"


class Node:
    def __init__(self, value=None, left=None, right=None):
        """
        Initialize a Node object.

        Args:
            value (any): Value of the node, typically a list of elements.
            left (Node): Left child of the node.
            right (Node): Right child of the node.
        """
        self.value = value
        self.left = left
        self.right = right

    def basic_display(self):
        """Display the tree structure in a simple format."""
        result = "\n"
        current_level = [self]
        while current_level:
            next_level = []
            for elem in current_level:
                result += f"{str(elem.value)} "
                if elem.left:
                    next_level.append(elem.left)
                if elem.right:
                    next_level.append(elem.right)
            result += "\n"
            current_level = next_level
        return result

    def __bool__(self):
        return self.value is not None

    def __str__(self):
        current_level = [self]
        result = []
        while current_level:
            result.append(" ".join(str(node.value) for node in current_level))
            next_level = []
            for node in current_level:
                if node.value is not None:
                    next_level.extend([node.left, node.right])
            current_level = [n for n in next_level if n]
        return "\n" + "\n".join(result)


def initialize_groups(similarity_matrix):
    """
    Initialize groups based on the similarity matrix.

    Args:
        similarity_matrix (list of list of int): The similarity matrix.

    Returns:
        list of Group: List of initialized groups.
    """
    groups = []
    for i, row in enumerate(similarity_matrix):
        max_sim = max(row)
        index = row.index(max_sim)
        group = Group([i], max_sim, index, Node([i]))
        groups.append(group)
    return groups


def merge_groups(groups, work_matrix):
    """
    Merge groups iteratively based on maximum similarity.

    Args:
        groups (list of Group): The list of groups to be merged.
        work_matrix (list of list of int): The working similarity matrix.

    Returns:
        Group: The final merged group.
    """
    while len(groups) > 1:
        max_sim = max(groups, key=lambda g: g.similarity).similarity
        dest_group_idx, source_group_idx = None, None

        for j, group in enumerate(groups):
            if group.similarity == max_sim:
                dest_group_idx = j
                source_group_idx = next(i for i, g in enumerate(groups) if group.index in g.elements)
                break

        dest_group = groups[dest_group_idx]
        source_group = groups[source_group_idx]

        # Update the working matrix to remove similarities between merged groups
        for element in source_group.elements:
            for dest_element in dest_group.elements:
                work_matrix[element][dest_element] = -1
                work_matrix[dest_element][element] = -1

        print(f"Merging groups: {dest_group.elements} with {source_group.elements}")
        dest_group.elements.extend(source_group.elements)
        dest_group.node = Node(dest_group.elements, dest_group.node, source_group.node)
        groups.pop(source_group_idx)

        # Update indices and similarities for the merged group
        max_sim, max_idx = -1, -1
        for element in dest_group.elements:
            for i, sim in enumerate(work_matrix[element]):
                if sim > max_sim:
                    max_sim, max_idx = sim, i

        dest_group.index = max_idx
        dest_group.similarity = max_sim

    return groups[0]


def display_matrix(matrix):
    """
    Display the matrix.

    Args:
        matrix (list of list of int): The matrix to display.
    """
    print("Matrix:")
    for row in matrix:
        print(row)
    print()


def remove_self_similarity(matrix):
    """
    Remove self-similarity from the similarity matrix by setting the diagonal to -1.

    Args:
        matrix (list of list of int): The matrix to be modified.
    """
    for i in range(len(matrix)):
        matrix[i][i] = -1

    work_matrix = [row[:] for row in similarity_matrix]  # Copy of the matrix

    groups = initialize_groups(work_matrix)
