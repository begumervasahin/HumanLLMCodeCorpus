import pandas as pd

def load_prediction_data(file_path):
    """
    Load prediction data from a CSV file and flatten it into a 1D list.

    Args:
        file_path (str): Path to the CSV file.

    Returns:
        list: Flattened list of prediction data.
    """
    return pd.read_csv(file_path).values.flatten().tolist()

def find_valid_sequences(pred_data, min_length=4):
    """
    Identify sequences of 0.5 and 1 in the prediction data where:
    - The sequence length is greater than min_length.
    - The sequence contains at least one value of 1.

    Args:
        pred_data (list): List of prediction data values.
        min_length (int): Minimum length of the sequence to be considered valid.

    Returns:
        tuple: Two lists containing the start and end indices of each valid sequence.
    """
    start_indices = []
    end_indices = []
    idx = 0
    data_length = len(pred_data)

    while idx < data_length:
        count = 0
        ones_count = 0
        
        if pred_data[idx] in [0.5, 1]:
            start_idx = idx
            
            while idx < data_length and pred_data[idx] in [0.5, 1]:
                if pred_data[idx] == 1:
                    ones_count += 1
                count += 1
                idx += 1
            
            end_idx = idx
            
            if count >= min_length and ones_count > 0:
                start_indices.append(start_idx)
                end_indices.append(end_idx)
        else:
            idx += 1

    return start_indices, end_indices

def save_time_data(start_indices, end_indices, output_file):
    """
    Save the start and end indices as a CSV file.

    Args:
        start_indices (list): List of start indices.
        end_indices (list): List of end indices.
        output_file (str): Path to the output CSV file.
    """
    time_data = pd.DataFrame({
        "Start Time": start_indices,
        "End Time": end_indices
    })
    time_data.to_csv(output_file, index=False)

def main():
    """
    Main function to load prediction data, find valid sequences, and save the results.
    """
    # Load the prediction data
    pred_data = load_prediction_data("y_prediction_bts.csv")
    
    # Find valid sequences in the prediction data
    start_indices, end_indices = find_valid_sequences(pred_data)
    
    # Save the identified sequences to a CSV file
    save_time_data(start_indices, end_indices, "timedata_bts.csv")

if __name__ == "__main__":
    main()
