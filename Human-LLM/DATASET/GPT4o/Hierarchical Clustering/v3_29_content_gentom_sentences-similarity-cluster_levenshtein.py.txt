def edit_distance(s1, s2):
    """
    Calculates the edit distance (Levenshtein distance) between two strings s1 and s2.

    Parameters:
    s1 (str): The first string.
    s2 (str): The second string.

    Returns:
    int: The edit distance between s1 and s2.
    """
    # Ensure s1 is the shorter string to optimize the space complexity
    if len(s1) > len(s2):
        s1, s2 = s2, s1  

    # Initialize the distance matrix for the first row
    previous_row = list(range(len(s1) + 1))

    # Compute the edit distance row by row
    for i2, c2 in enumerate(s2):
        # Start the new row with the incremented index
        current_row = [i2 + 1]

        for i1, c1 in enumerate(s1):
            # Calculate costs for different operations
            if c1 == c2:
                current_row.append(previous_row[i1])  # Characters match, no operation needed
            else:
                insertion_cost = previous_row[i1 + 1] + 1
                deletion_cost = current_row[i1] + 1
                substitution_cost = previous_row[i1] + 1
                current_row.append(min(insertion_cost, deletion_cost, substitution_cost))

        # Move to the next row
        previous_row = current_row

    # The last element in the last row is the edit distance
    return previous_row[-1]


def alternative_distance(s1, s2):
    """
    An alternative implementation for calculating the edit distance (Levenshtein distance)
    between two strings s1 and s2.

    Parameters:
    s1 (str): The first string.
    s2 (str): The second string.

    Returns:
    int: The edit distance between s1 and s2.
    """
    # Ensure s1 is the longer string for recursive case handling
    if len(s1) < len(s2):
        return alternative_distance(s2, s1)  

    # Base case: If the second string is empty, return the length of the first string
    if len(s2) == 0:
        return len(s1)  

    # Initialize the first row of the matrix
    previous_row = list(range(len(s2) + 1))

    # Compute the edit distance row by row
    for i, c1 in enumerate(s1):
        current_row = [i + 1]

        for j, c2 in enumerate(s2):
            insertion_cost = previous_row[j + 1] + 1
            deletion_cost = current_row[j] + 1
            substitution_cost = previous_row[j] + (c1 != c2)
            current_row.append(min(insertion_cost, deletion_cost, substitution_cost))

        previous_row = current_row  # Move to the next row

    return previous_row[-1]  # The last element contains the edit distance
