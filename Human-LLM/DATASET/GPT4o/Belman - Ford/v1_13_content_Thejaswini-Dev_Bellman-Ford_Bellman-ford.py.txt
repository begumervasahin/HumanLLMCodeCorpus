class Graph:
    def __init__(self, vertices):
        """
        Initialize the graph with a given number of vertices.
        :param vertices: Number of vertices in the graph.
        """
        self.V = vertices
        self.graph = []

    def add_edge(self, u, v, w):
        """
        Add an edge to the graph.
        :param u: The starting vertex of the edge.
        :param v: The ending vertex of the edge.
        :param w: The weight of the edge.
        """
        self.graph.append([u, v, w])

    def print_distances(self, dist):
        """
        Print the distances from the source vertex to all other vertices.
        :param dist: List of distances from the source vertex.
        """
        print("Vertex   Distance from Source")
        for i in range(self.V):
            print(f"{i}\t\t{dist[i]}")

    def bellman_ford(self, src):
        """
        Bellman-Ford algorithm to find the shortest paths from the source vertex to all other vertices.
        :param src: The source vertex.
        """
        # Step 1: Initialize distances from src to all other vertices as INFINITE
        dist = [float("Inf")] * self.V
        dist[src] = 0

        # Step 2: Relax all edges |V| - 1 times
        for _ in range(self.V - 1):
            for u, v, w in self.graph:
                if dist[u] != float("Inf") and dist[u] + w < dist[v]:
                    dist[v] = dist[u] + w

        # Step 3: Check for negative-weight cycles
        for u, v, w in self.graph:
            if dist[u] != float("Inf") and dist[u] + w < dist[v]:
                print("Graph contains negative weight cycle")
                return

        # Print the calculated shortest distances
        self.print_distances(dist)

# Example usage
if __name__ == "__main__":
    g = Graph(5)
    g.add_edge(0, 1, -1)
    g.add_edge(0, 2, 4)
    g.add_edge(1, 2, 3)
    g.add_edge(1, 3, 2)
    g.add_edge(1, 4, 2)
    g.add_edge(3, 2, 5)
    g.add_edge(3, 1, 1)
    g.add_edge(4, 3, -3)

    g.bellman_ford(0)
