edges2 = {
    "v_start": {"1_start": 0, "2_start": 0},
    "1_start": {"1_finish": 40},
    "2_start": {"2_finish": 36},
    "1_finish": {"v_finish": 0},
    "2_finish": {"v_finish": 0, "1_start": 0}
}

def connections(v_start):
    """
    Retrieve connections for a given start node.

    :param v_start: The start node.
    :return: Dictionary of connected nodes and their weights.
    """
    return edges2.get(v_start, {})

def weight(start, finish):
    """
    Retrieve the weight of the edge between start and finish nodes.

    :param start: The start node.
    :param finish: The finish node.
    :return: Weight of the edge.
    """
    return edges2.get(start, {}).get(finish, None)

def LongestPath(edges2, Ppaths):
    """
    Generate paths for longest path calculation.

    :param edges2: The graph represented as a dictionary.
    :param Ppaths: List of paths to be extended.
    """
    for i in Ppaths:
        for j in connections(i):
            Ppaths.append([i])
    print(Ppaths)

def ShortestPath(edges2):
    """
    Calculate the shortest paths in the graph.

    :param edges2: The graph represented as a dictionary.
    :return: Tuple of paths and their corresponding weights.
    """
    Ppaths = []
    Pweights = []

    # Initialize paths and weights from the starting node
    for key, value in edges2['v_start'].items():
        Ppaths.append([key])
        Pweights.append(value)

    # Traverse through paths to build all possible paths
    for path in Ppaths:
        path_end = path[-1]
        if len(edges2[path_end].keys()) == 1:
            path.extend(edges2[path_end].keys())
            Pweights.append(edges2[path_end][list(edges2[path_end].keys())[0]])
        elif len(edges2[path_end].keys()) > 1:
            for key in edges2[path_end].keys():
                new_path = path.copy()
                new_path.append(key)
                Ppaths.append(new_path)

    return Ppaths, Pweights

    # Shortest path calculation example
    paths, weights = ShortestPath(edges2)
