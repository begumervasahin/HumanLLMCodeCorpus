from grafo import Grafo
from bellman_ford import bellman_ford
from utils import caminho_minimo

class NegativeWeightException(Exception):
    """Custom exception to handle negative weight cycles in the graph."""
    pass

def create_graph():
    """
    Create a graph based on the example from Professor Fernando Lobo's lectures at the University of Algarve in Portugal.
    :return: A graph object with vertices and edges inserted.
    """
    # Create a directed graph
    graph = Grafo(direcionado=True)

    # Define vertices
    vertices = ['a', 'b', 'c', 'd', 'e']

    # Insert vertices into the graph
    for vertex in vertices:
        graph.inserir_vertice(vertex)

    # Define edges with weights
    edges = [
        ('a', 'b', 10),
        ('a', 'c', 3),
        ('b', 'c', 1),
        ('b', 'd', 2),
        ('c', 'b', 4),
        ('c', 'd', 8),
        ('c', 'e', 2),
        ('d', 'e', 7),
        ('e', 'd', 9)
    ]

    # Insert edges into the graph
    for origem, destino, peso in edges:
        graph.inserir_aresta(origem, destino, peso)

    return graph

def print_shortest_paths(graph, start_vertex_id):
    """
    Run the Bellman-Ford algorithm on the graph and print the shortest paths and costs for each vertex.
    :param graph: The graph object.
    :param start_vertex_id: The starting vertex ID for the Bellman-Ford algorithm.
    """
    if bellman_ford(graph, start_vertex_id):
        # Print the shortest paths and costs for each vertex
        for vertex in graph.get_vertices():
            path = [vertex.get_id()]
            caminho_minimo(vertex, path)
            print(f'The shortest path to vertex {vertex.get_id()} is: {path[::-1]} with a cost of {vertex.get_distancia()}.')
    else:
        print('Negative weight cycle found')

def test_graph():
    """
    Test the graph example from Professor Fernando Lobo's lectures at the University of Algarve in Portugal.
    """
    print('Testing graph example from Professor Fernando Lobo\'s lectures at the University of Algarve in Portugal.')

    # Create the graph
    graph = create_graph()

    # Print the shortest paths from vertex 'a'
    print_shortest_paths(graph, 'a')
