from Edges import Edge

class Vertex:
    def __init__(self, id):
        """
        Initialize a Vertex instance.

        :param id: The identifier of the vertex.
        """
        self.id = id
        self.min_distance = float('inf')
        self.adjacencies = []
        self.prev_vertex = None
        self.visited = False

    def get_min_distance(self):
        """
        Get the minimum distance to this vertex.

        :return: The minimum distance.
        """
        return self.min_distance

    def set_min_distance(self, distance):
        """
        Set the minimum distance to this vertex.

        :param distance: The minimum distance to set.
        """
        self.min_distance = distance

    def get_previous_vertex(self):
        """
        Get the previous vertex in the path.

        :return: The previous vertex.
        """
        return self.prev_vertex

    def set_previous_vertex(self, prev_vertex):
        """
        Set the previous vertex in the path.

        :param prev_vertex: The previous vertex to set.
        """
        self.prev_vertex = prev_vertex

    def add_edge(self, edge):
        """
        Add an edge to the vertex's adjacency list.

        :param edge: The edge to add.
        """
        self.adjacencies.append(edge)

    def get_adjacencies(self):
        """
        Get the list of adjacencies (edges) for this vertex.

        :return: The list of adjacencies.
        """
        return self.adjacencies

    def is_visited(self):
        """
        Check if the vertex has been visited.

        :return: True if the vertex has been visited, False otherwise.
        """
        return self.visited

    def set_visited(self, visited):
        """
        Set the visited status of the vertex.

        :param visited: The visited status to set.
        """
        self.visited = visited

    def __str__(self):
        """
        String representation of the vertex.

        :return: The identifier of the vertex.
        """
        return self.id

# Example usage
def main():
    # Create vertices
    vertex_A = Vertex('A')
    vertex_B = Vertex('B')

    # Create an edge between vertex A and vertex B
    edge_AB = Edge(vertex_A, vertex_B, 10)
    
    # Add the edge to the adjacency list of vertex A
    vertex_A.add_edge(edge_AB)

    # Set some properties
    vertex_A.set_min_distance(0)
    vertex_B.set_min_distance(10)
    vertex_A.set_visited(True)

    # Print vertex details
    print(f"Vertex A: {vertex_A}, Min Distance: {vertex_A.get_min_distance()}, Visited: {vertex_A.is_visited()}")
    print(f"Vertex B: {vertex_B}, Min Distance: {vertex_B.get_min_distance()}, Visited: {vertex_B.is_visited()}")
    print(f"Edges from Vertex A: {[str(edge.target) for edge in vertex_A.get_adjacencies()]}")

if __name__ == "__main__":
    main()
