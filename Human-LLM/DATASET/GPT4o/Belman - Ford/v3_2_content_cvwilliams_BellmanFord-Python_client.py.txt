import socket
import pickle
import sys

MAXINT = sys.maxsize

# Initial routing table
initial_table = [
    [0, -1, 0],
    [1, 0, 1],
    [2, 1, 3],
    [3, 2, 7]
]

def print_table(table):
    """
    Print the routing table in a formatted manner.

    :param table: The routing table to print.
    """
    print("DESTINATION-------INTERFACE------LINKCOST")
    for entry in table:
        destination, interface, link_cost = entry
        if interface == -1:
            print(f"    {destination}            {interface}           {link_cost}")
        elif interface == MAXINT:
            print(f"    {destination}            inf            inf")
        else:
            print(f"    {destination}            {interface}            {link_cost}")

def main():
    """
    Main function to establish connection, send and receive routing table updates.
    """
    print("Initial Routing Table:")
    print_table(initial_table)

    # Create a socket object
    with socket.socket() as s:
        host = "afsconnect2.njit.edu"
        port = 57171

        print("Client: Connecting to the server...")
        s.connect((host, port))

        print("Client: Sending initial table...")
        s.send(pickle.dumps(initial_table))

        print("Client: Waiting for update...")
        updated_table = s.recv(1024)
        updated_table = pickle.loads(updated_table)

        print("Client: Received updated table.")
        print_table(updated_table)

        print('Client: Connection closed')

if __name__ == '__main__':
    main()
