import os

def read_file(filepath):
    """
    Reads a file containing neighbor information and constructs a dictionary representation.

    Args:
        filepath (str): The full path to the file containing the neighbor information.

    Returns:
        dict: A dictionary containing neighbors and their corresponding cost and port.
    """
    neighbors = {}
    
    with open(filepath, 'r') as infile:
        for line_number, line in enumerate(infile):
            line = line.strip()
            if line_number == 0:
                try:
                    no_of_neighbors = int(line)
                except ValueError:
                    print(f"Invalid number of neighbors: {line}")
                    return {}
            else:
                words = line.split()
                if len(words) != 3:
                    print(f"Invalid line format: {line}")
                    continue
                
                neighbor, weight, port = words[0], words[1], words[2]
                
                try:
                    weight = float(weight)
                    port = int(port)
                except ValueError:
                    print(f"Invalid weight or port in line: {line}")
                    continue
                
                neighbors[neighbor] = {'cost': weight, 'port': port}

    return neighbors

def print_neighbors(neighbors):
    """
    Prints the neighbor information in a readable format.

    Args:
        neighbors (dict): A dictionary containing neighbors and their corresponding cost and port.
    """
    print("Neighbors:")
    for neighbor, details in neighbors.items():
        print(f"{neighbor}: cost={details['cost']}, port={details['port']}")

def main():
    """
    Main function to read the neighbor information from a file and print it.
    """
    filename = "input.txt"  # Example filename
    filepath = os.path.join("C:\\", filename)
    
    if not os.path.isfile(filepath):
        print(f"File {filepath} does not exist.")
        return

    neighbors = read_file(filepath)
    print_neighbors(neighbors)

if __name__ == "__main__":
    main()
