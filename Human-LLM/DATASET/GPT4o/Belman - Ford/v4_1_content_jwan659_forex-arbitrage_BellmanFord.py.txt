class Vertex:
    def __init__(self, name):
        """
        Initialize a vertex with a given name.
        The minimum distance is set to infinity and the previous vertex is set to None.
        """
        self.name = name
        self.min_distance = float('inf')
        self.previous_vertex = None

    def set_min_distance(self, dist):
        """
        Set the minimum distance from the source vertex.
        """
        self.min_distance = dist

    def get_min_distance(self):
        """
        Get the minimum distance from the source vertex.
        """
        return self.min_distance

    def set_previous_vertex(self, vertex):
        """
        Set the previous vertex in the shortest path.
        """
        self.previous_vertex = vertex

    def get_previous_vertex(self):
        """
        Get the previous vertex in the shortest path.
        """
        return self.previous_vertex

    def __str__(self):
        """
        Return the string representation of the vertex (its name).
        """
        return self.name
