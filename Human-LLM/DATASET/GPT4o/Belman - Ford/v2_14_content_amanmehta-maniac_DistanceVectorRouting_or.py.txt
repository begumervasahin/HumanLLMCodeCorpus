from collections import defaultdict
import socket
import time

def initialize(graph, source):
    """
    Initialize the distance dictionary for the Bellman-Ford algorithm.
    :param graph: The graph represented as a dictionary of dictionaries.
    :param source: The source node.
    :return: A dictionary with initial distances set to infinity, except for the source node.
    """
    distances = {node: float('Inf') for node in graph}
    distances[source] = 0
    return distances

def relax():
    """
    Simulate the relax function by establishing a socket connection and receiving data.
    """
    print("Relax called")
    with socket.socket() as s:
        host = "127.0.0.1"  # Use localhost for demonstration purposes
        port = 60000
        s.connect((host, port))
        data = s.recv(1024)
        print(data.decode('utf-8'))

def bellman_ford(graph, source):
    """
    Bellman-Ford algorithm to find the shortest paths from the source to all other nodes.
    :param graph: The graph represented as a dictionary of dictionaries.
    :param source: The source node.
    :return: A dictionary containing the shortest distances from the source to each node.
    """
    distances = initialize(graph, source)
    
    for _ in range(len(graph) - 1):
        for u in graph:
            for v in graph[u]:
                weight = graph[u][v]
                if distances[u] != float('Inf') and distances[u] + weight < distances[v]:
                    distances[v] = distances[u] + weight
            relax()

    # Check for negative-weight cycles
    for u in graph:
        for v in graph[u]:
            weight = graph[u][v]
            if distances[u] != float('Inf') and distances[u] + weight < distances[v]:
                print("Graph contains a negative weight cycle")
                return

    return distances

def test(graph):
    """
    Test the Bellman-Ford algorithm with the provided graph.
    :param graph: The graph represented as a dictionary of dictionaries.
    """
    source = 0  # Assuming the source node is 0
    distances = bellman_ford(graph, source)
    print("Distances from source:")
    for node in distances:
        print(f"Node {node}: {distances[node]}")

def load_graph(filename):
    """
    Load a graph from a file.
    :param filename: The name of the file containing the graph data.
    :return: A graph represented as a dictionary of dictionaries.
    """
    graph = defaultdict(dict)
    with open(filename) as f:
        lines = f.readlines()
        for i in range(1, int(lines[0]) + 1):
            gf = lines[i].split()
            node = int(gf[0])
            for k in range(1, len(gf), 2):
                graph[node][int(gf[k])] = int(gf[k + 1])
    return graph

if __name__ == '__main__':
    graph = load_graph('test')
    test(graph)
