class Graph:
    """Represents a graph with vertices and edges."""
    
    def __init__(self, vertices=None, edges=None, directed=False):
        self.V = vertices if vertices is not None else []
        self.E = edges if edges is not None else []
        self.directed = directed
        self.adj = {vertex.value: [] for vertex in self.V}
        
    def get_adj(self, v):
        """Returns the adjacency list of vertex v."""
        return self.adj[v.value]
    
    def add_vertex(self, value):
        """Adds a vertex to the graph."""
        u = Vertex(value=value)
        self.V.append(u)
        self.adj[u.value] = []
    
    def add_edge(self, u, v, weight):
        """Adds an edge to the graph."""
        e = Edge(u, v, weight)
        self.adj[u.value].append(e)
        self.E.append(e)
        if not self.directed:
            f = Edge(v, u, weight)
            self.adj[v.value].append(f)

class Vertex:
    """Represents a vertex in the graph."""
    
    def __init__(self, value=None, d=None, pre=None):
        self.value = value
        self.d = d
        self.pre = pre

class Edge:
    """Represents an edge in the graph."""
    
    def __init__(self, u=None, v=None, weight=None):
        self.u = u
        self.v = v
        self.weight = weight
    
    def equals(self, other):
        """Checks if this edge is equal to another edge."""
        return self.u == other.u and self.v == other.v

def main():
    # Example usage of the Graph, Vertex, and Edge classes
    g = Graph(directed=False)
    g.add_vertex(1)
    g.add_vertex(2)
    g.add_vertex(3)
    
    v1 = g.V[0]
    v2 = g.V[1]
    v3 = g.V[2]
    
    g.add_edge(v1, v2, 10)
    g.add_edge(v2, v3, 20)
    g.add_edge(v3, v1, 30)
    
    print("Vertices in the graph:")
    for v in g.V:
        print(f"Vertex {v.value}")
    
    print("\nEdges in the graph:")
    for e in g.E:
        print(f"Edge from {e.u.value} to {e.v.value} with weight {e.weight}")

if __name__ == "__main__":
    main()
