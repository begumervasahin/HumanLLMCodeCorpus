class Graph:
    def __init__(self, vertices):
        """Initializes the graph with a given number of vertices."""
        self.vertices = vertices
        self.edges = []

    def add_edge(self, source, destination, weight):
        """Adds an edge to the graph."""
        self.edges.append([source, destination, weight])

    def print_distances(self, distances):
        """Prints the distances from the source to all vertices."""
        print("Vertex Distance from Source")
        for vertex, distance in enumerate(distances):
            print(f"{vertex}\t\t{distance}")

    def bellman_ford(self, source):
        """Runs the Bellman-Ford algorithm from the source vertex."""
        # Initialize distances from source to all vertices as infinite
        distances = [float("inf")] * self.vertices
        distances[source] = 0

        # Relax all edges |V| - 1 times
        for _ in range(self.vertices - 1):
            for u, v, w in self.edges:
                if distances[u] != float("inf") and distances[u] + w < distances[v]:
                    distances[v] = distances[u] + w

        # Check for negative-weight cycles
        for u, v, w in self.edges:
            if distances[u] != float("inf") and distances[u] + w < distances[v]:
                print("Graph contains a negative weight cycle")
                return

        self.print_distances(distances)

def main():
    """Main function to create the graph and run the Bellman-Ford algorithm."""
    # Create a graph with 5 vertices
    graph = Graph(5)

    # Define edges in the format (source, destination, weight)
    edges = [
        (0, 1, -1),
        (0, 2, 4),
        (1, 2, 3),
        (1, 3, 2),
        (1, 4, 2),
        (3, 2, 5),
        (3, 1, 1),
        (4, 3, -3)
    ]

    # Add edges to the graph
    for source, destination, weight in edges:
        graph.add_edge(source, destination, weight)

    # Run the Bellman-Ford algorithm from vertex 0
    source_vertex = 0
    graph.bellman_ford(source_vertex)

if __name__ == "__main__":
    main()
