from utils import initialize_single_source, relax

def bellman_ford(graph, source):
    """
    Bellman-Ford algorithm to find the shortest paths from the source vertex to all other vertices.
    :param graph: The graph on which to run the algorithm.
    :param source: The starting vertex.
    :return: True if no negative weight cycles are detected, False otherwise.
    """
    # Step 1: Initialize the graph
    initialize_single_source(graph, source)
    
    # Step 2: Relax all edges |V| - 1 times
    for _ in range(len(graph.get_vertices()) - 1):
        for u, v in graph.get_edges():
            if not v.is_visited():
                relax(u, v)

    # Step 3: Check for negative-weight cycles
    for u, v in graph.get_edges():
        if v.get_distance() > u.get_distance() + u.get_weight(v):
            return False  # A negative weight cycle is found

    return True  # No negative weight cycles found

    # Define the source vertex
    source_vertex = graph.get_vertex(0)
    
    # Run the Bellman-Ford algorithm
    if bellman_ford(graph, source_vertex):
        print("No negative weight cycle detected")
    else:
        print("Graph contains a negative weight cycle")
