import socket
import sys
import pickle

def main():
    if len(sys.argv) != 2:
        print("Usage: python script.py <port>")
        sys.exit(1)

    port = int(sys.argv[1])

    # Create a UDP socket
    with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as recvsock:
        recvsock.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)

        # Bind the socket to the given port
        recvsock.bind(("", port))
        print(f"Listening on port {port}")

        # Receive data from the socket
        data, addr = recvsock.recvfrom(4096)

    # Deserialize the data using pickle
    recovered = pickle.loads(data)

    # Print the sender's address and the recovered data
    print(f"Sender: {addr}")
    print(f"Recovered entry: {recovered}")

if __name__ == "__main__":
    main()
