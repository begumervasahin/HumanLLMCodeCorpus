import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import statsmodels.formula.api as smf
from sklearn.metrics import mean_squared_error
from math import sqrt

# Load the dataset
data_path = 'C:/Users/HP/Desktop/assignments submission/simple linear regression/Salary_Data.csv'
sal_hike = pd.read_csv(data_path)

# Scatter plot of YearsExperience vs Salary
def plot_scatter(x, y, xlabel, ylabel, title):
    plt.scatter(x, y)
    plt.xlabel(xlabel)
    plt.ylabel(ylabel)
    plt.title(title)
    plt.show()

plot_scatter(sal_hike.YearsExperience, sal_hike.Salary, 'Years of Experience', 'Salary', 'Scatter plot of Years of Experience vs Salary')

# Display the correlation matrix
corr_matrix = sal_hike.corr()
print("Correlation matrix:\n", corr_matrix)

# Function to fit a model and display results
def fit_model(formula, data, model_name):
    model = smf.ols(formula, data=data).fit()
    print(f"{model_name} Parameters:\n", model.params)
    print(f"{model_name} Summary:\n", model.summary())
    
    # Predictions and Errors
    predictions = model.predict(data)
    errors = data.Salary - predictions

    # Plot model predictions
    plt.scatter(data.YearsExperience, data.Salary, c='r')
    plt.plot(data.YearsExperience, predictions, c='b')
    plt.xlabel('Years of Experience')
    plt.ylabel('Salary')
    plt.title(f'{model_name}: Years of Experience vs Salary')
    plt.show()

    # Calculate and print RMSE
    rmse = sqrt(mean_squared_error(data.Salary, predictions))
    print(f"{model_name} RMSE:", rmse)
    
    # Return the predictions
    return predictions

# Fit and evaluate Linear Model
lin_predict = fit_model('Salary ~ YearsExperience', sal_hike, 'Linear Model')

# Fit and evaluate Logarithmic Model
log_predict = fit_model('Salary ~ np.log(YearsExperience)', sal_hike, 'Logarithmic Model')

# Fit and evaluate Exponential Model
exp_model = smf.ols('np.log(Salary) ~ YearsExperience', data=sal_hike).fit()
print("Exponential Model Parameters:\n", exp_model.params)
print("Exponential Model Summary:\n", exp_model.summary())

# Predictions and Errors for Exponential Model
exp_predict = np.exp(exp_model.predict(sal_hike))
exp_error = sal_hike.Salary - exp_predict

# Plot Exponential Model predictions
plt.scatter(sal_hike.YearsExperience, sal_hike.Salary, c='r')
plt.plot(sal_hike.YearsExperience, exp_predict, c='b')
plt.xlabel('Years of Experience')
plt.ylabel('Salary')
plt.title('Exponential Model: Years of Experience vs Salary')
plt.show()

# RMSE for Exponential Model
exp_rmse = sqrt(mean_squared_error(sal_hike.Salary, exp_predict))
print("Exponential Model RMSE:", exp_rmse)

# Fit and evaluate Quadratic Model
quad_predict = fit_model('Salary ~ YearsExperience + I(YearsExperience**2)', sal_hike, 'Quadratic Model')

# Display correlation coefficients
def display_corr_coeff(model_name, actual, predicted):
    corr_coeff = np.corrcoef(actual, predicted)[0, 1]
    print(f"Correlation coefficient ({model_name}):", corr_coeff)

display_corr_coeff("Linear Model", sal_hike.Salary, lin_predict)
display_corr_coeff("Logarithmic Model", sal_hike.Salary, log_predict)
display_corr_coeff("Exponential Model", sal_hike.Salary, exp_predict)
display_corr_coeff("Quadratic Model", sal_hike.Salary, quad_predict)
