import pandas as pd
import numpy as np
import os
import matplotlib.pyplot as plt
from datetime import date
from sklearn.linear_model import LinearRegression
import seaborn as sns
from sklearn.model_selection import train_test_split
from tkinter import messagebox as msg
from tkinter import *
from sklearn.preprocessing import LabelEncoder
from sklearn.metrics import r2_score

class SalaryPredictor:
    def __init__(self, root):
        # Initialize the GUI
        self.frame = Frame(root, height=200, width=300)
        self.frame.pack()

        # Create and place widgets
        self.messageLabel = Label(self.frame, text='Predict Salary Using Experience In Years', font='Arial')
        self.inputLabelExp = Label(self.frame, text='Enter experience', font=('Arial', 14))
        self.outputLabel = Label(self.frame, text='', font=('Arial', 14))
        self.experienceEntry = Entry(self.frame, text='', width=10, font=('Arial', 14))
        self.predictButton = Button(self.frame, text='Predict', font=('Arial', 12), command=self.predict_salary)
        self.exitButton = Button(self.frame, text='Exit', font=('Arial', 12), command=root.destroy)

        # Layout management
        self.messageLabel.grid(row=0, column=0, columnspan=2)
        self.inputLabelExp.grid(row=1, column=0)
        self.experienceEntry.grid(row=1, column=1)
        self.predictButton.grid(row=2, column=0)
        self.exitButton.grid(row=2, column=1)
        self.outputLabel.grid(row=3, column=0, columnspan=2)

    def predict_salary(self):
        if not os.path.exists('empsal.csv'):
            msg.showerror('Error', 'File path is wrong or file not found')
            return
        
        try:
            # Load the data
            data = pd.read_csv('empsal.csv', parse_dates=True, infer_datetime_format=True, index_col='empno')

            # Helper function to calculate age from date of birth
            def calculate_age(dob):
                today = date.today()
                return today.year - dob.year - ((today.month, today.day) < (dob.month, dob.day))

            # Data preprocessing
            data['dob'] = pd.to_datetime(data['dob'])
            data['age'] = data['dob'].apply(calculate_age)
            data.drop(columns=['empname', 'dob', 'hra'], inplace=True)

            # Encode categorical variables
            label_encoder = LabelEncoder()
            data['sex'] = label_encoder.fit_transform(data['sex'])
            data['state'] = label_encoder.fit_transform(data['state'])
            data['city'] = label_encoder.fit_transform(data['city'])

            # Display correlation heatmap and pairplot
            self._plot_data(data)

            # Prepare features and target variable
            X = data.drop(columns=['state', 'salary', 'city', 'sex', 'age'])
            y = data['salary']

            # Train and evaluate the model
            best_model = self._train_and_evaluate_model(X, y)

            # Predict salary based on input experience
            experience_years = float(self.experienceEntry.get())
            predicted_salary = best_model.predict(np.array([[experience_years]]))
            self.outputLabel.config(text='Expected Salary is: {:.2f}'.format(predicted_salary[0]))

        except Exception as e:
            msg.showerror('Error', str(e))

    def _plot_data(self, data):
        """Plot correlation heatmap and pairplot."""
        correlation = data.corr()
        sns.heatmap(correlation)
        plt.show()
        sns.pairplot(data)
        plt.show()

    def _train_and_evaluate_model(self, X, y):
        """Train and evaluate the linear regression model, returning the best model."""
        linear_reg = LinearRegression()
        best_score = -1
        best_model = None

        for i in range(1000):
            X_train, X_test, y_train, y_test = train_test_split(X, y, random_state=i, test_size=0.1)
            model = linear_reg.fit(X_train, y_train)
            score = model.score(X_test, y_test)
            if score > best_score:
                best_score = score
                best_model = model

        print('Best R2 score:', best_score)
        return best_model

if __name__ == "__main__":
    root = Tk()
    root.title("Salary Predictor")
    root.geometry('1000x300')
    app = SalaryPredictor(root)
    root.mainloop()
