import numpy as np
import pandas as pd
from sklearn.model_selection import KFold
from sklearn.discriminant_analysis import LinearDiscriminantAnalysis
from sklearn.linear_model import LogisticRegression
from sklearn.utils import shuffle

def load_data1(filename):
    """Load and preprocess the wine quality dataset."""
    data_df = pd.read_csv(filename, header=0, sep=';')
    data = data_df.values
    data = shuffle_data(data)
    
    X = np.array(data[:, 0:-1])
    Y = data[:, -1]
    Y = np.asarray([1 if i > 5 else 0 for i in Y])
    
    return X, Y

def load_data2(filename):
    """Load and preprocess the breast cancer dataset."""
    data_clean = []
    
    with open(filename) as input_file:
        lines = input_file.readlines()
        for line in lines:
            newline = line.strip().split(',')
            temp = newline[1:]
            temp = [-1 if value == '?' else int(value) for value in temp]
            data_clean.append(temp)
    
    data_clean = np.array(data_clean)
    data_clean = shuffle_data(data_clean)
    
    X = data_clean[:, 0:-1]
    Y = data_clean[:, -1]
    Y = np.asarray([1 if i == 2 else 0 for i in Y])
    
    return X, Y

def dummy_feature(X):
    """Add a dummy feature (column of ones) to the dataset."""
    n1, m1 = np.shape(X)
    X0 = np.ones((n1, 1))
    X_new = np.column_stack((X0, X))
    
    return X_new

def split_train_test(X, Y):
    """Split the dataset into training and testing sets."""
    num = len(Y)
    train_x, test_x = X[:int(0.9*num), :], X[int(0.9*num):, :]
    train_y, test_y = Y[:int(0.9*num)], Y[int(0.9*num):]
    
    return train_x, train_y, test_x, test_y

def shuffle_data(data):
    """Shuffle the data."""
    np.random.shuffle(data)
    return data

def sk_learn_test(X, Y):
    """Evaluate classifiers using K-Fold cross-validation."""
    fold = 5
    kf = KFold(n_splits=fold, shuffle=True)
    
    accuracy_LR = np.zeros(fold)
    accuracy_LDA = np.zeros(fold)
    
    for k, (train_index, test_index) in enumerate(kf.split(X)):
        train_x, val_x = X[train_index], X[test_index]
        train_y, val_y = Y[train_index], Y[test_index]
        
        # Logistic Regression
        lg = LogisticRegression(random_state=0, max_iter=1000, penalty='l2', solver='liblinear', C=1)
        lg.fit(train_x, train_y)
        accuracy_LR[k] = lg.score(val_x, val_y) * 100
        
        # Linear Discriminant Analysis
        clf = LinearDiscriminantAnalysis()
        clf.fit(train_x, train_y)
        accuracy_LDA[k] = clf.score(val_x, val_y) * 100
    
    print('Logistic Regression')
    print(accuracy_LR)
    print('Average accuracy =', np.mean(accuracy_LR))
    
    print('Linear Discriminant Analysis')
    print(accuracy_LDA)
    print('Average accuracy =', np.mean(accuracy_LDA))

# Load and split the datasets
X1, Y1 = load_data1('winequality-red.csv')
X1_train, Y1_train, X1_test, Y1_test = split_train_test(X1, Y1)

X2, Y2 = load_data2('breast-cancer-wisconsin.data')
X2_train, Y2_train, X2_test, Y2_test = split_train_test(X2, Y2)

# Display data information
print('Red wine data:')
sk_learn_test(X1_train, Y1_train)

print('\nBreast cancer data:')
sk_learn_test(X2_train, Y2_train)
