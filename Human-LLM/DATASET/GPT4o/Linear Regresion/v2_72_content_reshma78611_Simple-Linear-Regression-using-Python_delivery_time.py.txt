import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.metrics import mean_squared_error
from math import sqrt
import statsmodels.formula.api as smf

# Load the dataset
data_path = 'C:/Users/HP/Desktop/assignments submission/simple linear regression/delivery_time.csv'
delivery_time_data = pd.read_csv(data_path)

# Scatter plot of Sorting Time vs Delivery Time
plt.scatter(delivery_time_data.Sorting_time, delivery_time_data.Delivery_time)
plt.xlabel('Sorting Time')
plt.ylabel('Delivery Time')
plt.title('Scatter Plot of Sorting Time vs Delivery Time')
plt.show()

# Calculate and print the correlation coefficient
correlation_matrix = np.corrcoef(delivery_time_data.Sorting_time, delivery_time_data.Delivery_time)
print(f'Correlation Coefficient: {correlation_matrix[0, 1]}')

# Linear Regression Model
linear_model = smf.ols('Delivery_time ~ Sorting_time', data=delivery_time_data).fit()
print("Linear Model Parameters:\n", linear_model.params)
print("Linear Model Summary:\n", linear_model.summary())

# Predictions and Errors for Linear Model
linear_predictions = linear_model.predict(delivery_time_data)
linear_errors = delivery_time_data.Delivery_time - linear_predictions

# Plot Linear Regression Results
plt.scatter(delivery_time_data.Sorting_time, delivery_time_data.Delivery_time, c='r')
plt.plot(delivery_time_data.Sorting_time, linear_predictions, c='b')
plt.xlabel('Sorting Time')
plt.ylabel('Delivery Time')
plt.title('Linear Regression: Sorting Time vs Delivery Time')
plt.show()

# Print Confidence Interval and RMSE for Linear Model
print("Linear Model Confidence Interval:\n", linear_model.conf_int(0.05))
linear_rmse = sqrt(mean_squared_error(delivery_time_data.Delivery_time, linear_predictions))
print(f'Linear Model RMSE: {linear_rmse}')

# Logarithmic Model
log_model = smf.ols('Delivery_time ~ np.log(Sorting_time)', data=delivery_time_data).fit()
print("Logarithmic Model Parameters:\n", log_model.params)
print("Logarithmic Model Summary:\n", log_model.summary())

# Predictions and Errors for Logarithmic Model
log_predictions = log_model.predict(delivery_time_data)
log_errors = delivery_time_data.Delivery_time - log_predictions

# Plot Logarithmic Regression Results
plt.scatter(delivery_time_data.Sorting_time, delivery_time_data.Delivery_time, c='r')
plt.plot(delivery_time_data.Sorting_time, log_predictions, c='b')
plt.xlabel('Sorting Time')
plt.ylabel('Delivery Time')
plt.title('Logarithmic Regression: Sorting Time vs Delivery Time')
plt.show()

# Print Confidence Interval and RMSE for Logarithmic Model
print("Logarithmic Model Confidence Interval:\n", log_model.conf_int(0.05))
log_rmse = sqrt(mean_squared_error(delivery_time_data.Delivery_time, log_predictions))
print(f'Logarithmic Model RMSE: {log_rmse}')

# Exponential Model
exp_model = smf.ols('np.log(Delivery_time) ~ Sorting_time', data=delivery_time_data).fit()
print("Exponential Model Parameters:\n", exp_model.params)
print("Exponential Model Summary:\n", exp_model.summary())

# Predictions and Errors for Exponential Model
exp_predictions_log = exp_model.predict(delivery_time_data)
exp_predictions = np.exp(exp_predictions_log)
exp_errors = delivery_time_data.Delivery_time - exp_predictions

# Plot Exponential Regression Results
plt.scatter(delivery_time_data.Sorting_time, delivery_time_data.Delivery_time, c='r')
plt.plot(delivery_time_data.Sorting_time, exp_predictions, c='b')
plt.xlabel('Sorting Time')
plt.ylabel('Delivery Time')
plt.title('Exponential Regression: Sorting Time vs Delivery Time')
plt.show()

# Print Confidence Interval and RMSE for Exponential Model
print("Exponential Model Confidence Interval:\n", exp_model.conf_int(0.05))
exp_rmse = sqrt(mean_squared_error(delivery_time_data.Delivery_time, exp_predictions))
print(f'Exponential Model RMSE: {exp_rmse}')

# Quadratic Model
delivery_time_data['sq_sorting_time'] = delivery_time_data.Sorting_time ** 2
quad_model = smf.ols('Delivery_time ~ Sorting_time + sq_sorting_time', data=delivery_time_data).fit()
print("Quadratic Model Parameters:\n", quad_model.params)
print("Quadratic Model Summary:\n", quad_model.summary())

# Predictions and Errors for Quadratic Model
quad_predictions = quad_model.predict(delivery_time_data)
quad_errors = delivery_time_data.Delivery_time - quad_predictions

# Plot Quadratic Regression Results
plt.scatter(delivery_time_data.Sorting_time, delivery_time_data.Delivery_time, c='r')
plt.plot(delivery_time_data.Sorting_time, quad_predictions, c='b')
plt.xlabel('Sorting Time')
plt.ylabel('Delivery Time')
plt.title('Quadratic Regression: Sorting Time vs Delivery Time')
plt.show()

# Print Confidence Interval and RMSE for Quadratic Model
print("Quadratic Model Confidence Interval:\n", quad_model.conf_int(0.05))
quad_rmse = sqrt(mean_squared_error(delivery_time_data.Delivery_time, quad_predictions))
print(f'Quadratic Model RMSE: {quad_rmse}')
