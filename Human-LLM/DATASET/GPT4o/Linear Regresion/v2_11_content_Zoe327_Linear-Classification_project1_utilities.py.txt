import numpy as np

class LDAClassification:
    def __init__(self):
        self.w0 = None
        self.weights = None

    def fit(self, train_x, train_y):
        N, M = train_x.shape
        N1 = np.sum(train_y)
        P1 = N1 / N
        P0 = 1 - P1
        
        u0, u1 = np.zeros(M), np.zeros(M)
        for i in range(N):
            if train_y[i] == 0:
                u0 += train_x[i, :]
            else:
                u1 += train_x[i, :]
        u0 /= (N - N1)
        u1 /= N1
        
        sigma = np.zeros((M, M))
        for i in range(N):
            if train_y[i] == 0:
                diff = train_x[i, :] - u0
            else:
                diff = train_x[i, :] - u1
            sigma += np.outer(diff, diff)
        sigma /= (N - 2)
        
        sigma_inv = np.linalg.inv(sigma)
        self.w0 = (np.log(P1 / P0) - 0.5 * np.dot(u1, np.dot(sigma_inv, u1)) +
                   0.5 * np.dot(u0, np.dot(sigma_inv, u0)))
        self.weights = np.dot(sigma_inv, (u1 - u0))

    def predict(self, test_x):
        log_odds = self.w0 + np.dot(test_x, self.weights)
        return np.array([1 if i > 0 else 0 for i in log_odds])

    def evaluate(self, test_x, test_y):
        predictions = self.predict(test_x)
        accuracy = np.mean(predictions == test_y) * 100
        return accuracy

class LogisticRegression:
    def __init__(self):
        self.weights = None

    @staticmethod
    def sigmoid(z):
        return 1 / (1 + np.exp(-z))

    def fit(self, train_x, train_y, tolerance=1e-4, max_iter=10000):
        N, M = train_x.shape
        self.weights = np.zeros((M, 1))
        train_y = train_y.reshape(-1, 1)
        for i in range(max_iter):
            alpha = 1 / (1 + i) + 0.00001
            predictions = self.sigmoid(np.dot(train_x, self.weights))
            errors = train_y - predictions
            gradient = np.dot(train_x.T, errors)
            new_weights = self.weights + alpha * gradient
            if np.mean(np.abs(new_weights - self.weights)) < tolerance:
                break
            self.weights = new_weights

    def predict(self, test_x):
        probabilities = self.sigmoid(np.dot(test_x, self.weights))
        return (probabilities > 0.5).astype(int).flatten()

    def evaluate(self, test_x, test_y):
        predictions = self.predict(test_x)
        accuracy = np.mean(predictions == test_y) * 100
        return accuracy

if __name__ == "__main__":
    # Sample data
    train_x = np.array([[1, 2], [2, 3], [3, 4], [4, 5]])
    train_y = np.array([0, 0, 1, 1])
    test_x = np.array([[1, 2], [4, 5]])
    test_y = np.array([0, 1])

    # LDA
    lda = LDAClassification()
    lda.fit(train_x, train_y)
    lda_accuracy = lda.evaluate(test_x, test_y)
    print(f"LDA Accuracy: {lda_accuracy:.2f}%")

    # Logistic Regression
    log_model = LogisticRegression()
    log_model.fit(train_x, train_y)
    log_accuracy = log_model.evaluate(test_x, test_y)
    print(f"Logistic Regression Accuracy: {log_accuracy:.2f}%")
