DECIMAL_PRECISION = 4  # Define the decimal precision for rounding the results

def transposed(matrix):
    """Return the transpose of the given matrix."""
    return [[matrix[row][col] for row in range(len(matrix))] for col in range(len(matrix[0]))]

def minor(matrix, omitted_row, omitted_col):
    """Return the minor matrix after removing the specified row and column."""
    return [
        row[:omitted_col] + row[omitted_col + 1:] 
        for row in (matrix[:omitted_row] + matrix[omitted_row + 1:])
    ]

def determinant(matrix):
    """Recursively calculate the determinant of the given matrix."""
    if len(matrix) == 1:
        return matrix[0][0]

    det = 0
    sign = 1
    for col in range(len(matrix[0])):
        det += sign * matrix[0][col] * determinant(minor(matrix, 0, col))
        sign *= -1
    return det

def inverse(matrix):
    """Calculate and return the inverse of the given matrix."""
    det = determinant(matrix)
    if det == 0:
        raise ValueError("Matrix is not invertible")

    cofactor_matrix = [
        [
            ((-1) ** (row + col)) * determinant(minor(matrix, row, col))
            for col in range(len(matrix[0]))
        ] 
        for row in range(len(matrix))
    ]
    
    transposed_cofactor_matrix = transposed(cofactor_matrix)
    return [
        [transposed_cofactor_matrix[row][col] / det for col in range(len(matrix))]
        for row in range(len(matrix))
    ]

def multiply(matrix_a, matrix_b):
    """Multiply two matrices and return the product matrix."""
    rows_a, cols_a = len(matrix_a), len(matrix_a[0])
    cols_b = len(matrix_b[0])

    product_matrix = [[0 for _ in range(cols_b)] for _ in range(rows_a)]
    
    for i in range(rows_a):
        for j in range(cols_b):
            for k in range(cols_a):
                product_matrix[i][j] += matrix_a[i][k] * matrix_b[k][j]
                
    return product_matrix

def main():
    """Main function to perform multiple linear regression."""
    print('\nMultiple Linear Regression Software\n')
    print('Describing the input file:\n')
    print('First line: number of independent variables - k')
    print('Second line: number of observations - n')
    print('Next n lines: tuples of data corresponding to each observation: x1 x2 ... xk y')    
    print('Last line: exponents of independent variables x1, x2, ... xk')

    with open('input_data.txt') as input_file:
        k = int(input_file.readline().strip())
        n = int(input_file.readline().strip())
        observations = [tuple(map(float, input_file.readline().split())) for _ in range(n)]
        exponents = list(map(int, input_file.readline().split()))

    print(f'\nInput processed: k = {k}, n = {n}, observations are: {observations}')
    print(f'Coefficients for independent variables are: {exponents}')
    print('\nCalculating coefficients...')

    # Construct matrix X
    X = [[1] + [obs[j] ** exponents[j] for j in range(k)] for obs in observations]
    # Construct matrix Y
    Y = [[obs[-1]] for obs in observations]

    print(f'Auxiliary matrices: \nX = {X}, \nY = {Y}')

    # Compute the transpose of X
    X_transpose = transposed(X)
    # Compute X'X
    X_transpose_X = multiply(X_transpose, X)
    # Compute X'Y
    X_transpose_Y = multiply(X_transpose, Y)
    # Compute the coefficients matrix B
    B = multiply(inverse(X_transpose_X), X_transpose_Y)

    print(f'\nEquation of hyperplane is: y = {round(B[0][0], DECIMAL_PRECISION)}', end=' ')
    for i in range(1, k):
        print(f'+ {round(B[i][0], DECIMAL_PRECISION)} * x{i}', end=' ')
    print(f'+ {round(B[k][0], DECIMAL_PRECISION)} * x{k}')

if __name__ == "__main__":
    main()
