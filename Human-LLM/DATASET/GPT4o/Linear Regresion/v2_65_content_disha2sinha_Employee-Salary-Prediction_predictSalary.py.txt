import pandas as pd
import numpy as np
import os
import matplotlib.pyplot as plt
from datetime import date
from sklearn.linear_model import LinearRegression
import seaborn as sns
from sklearn.model_selection import train_test_split
from tkinter import messagebox as msg
from tkinter import *
from sklearn.preprocessing import LabelEncoder
from sklearn.metrics import r2_score

class PredictSalary:
    def __init__(self, root):
        """
        Initialize the GUI components and set up the layout.
        """
        self.frame = Frame(root, height=200, width=300)
        self.frame.pack()
        
        self.messageLabel = Label(self.frame, text='Predict Salary Using Experience In Years', font='Arial')
        self.inputLabelexp = Label(self.frame, text='Enter experience', font=('Arial', 14))
        self.outputLabel = Label(self.frame, text='', font=('Arial', 14))
        self.experience = Entry(self.frame, width=10, font=('Arial', 14))
        self.predictedLabel = Label(self.frame, text='', font=('Arial', 15))
        self.predictButton = Button(self.frame, text='Predict', font=('Arial', 12), command=self.predict_salary)
        self.exitButton = Button(self.frame, text='Exit', font=('Arial', 12), command=root.destroy)
        
        # Grid layout
        self.messageLabel.grid(row=0, column=0, columnspan=2)
        self.inputLabelexp.grid(row=1, column=0)
        self.experience.grid(row=1, column=1)
        self.predictedLabel.grid(row=5, column=0)
        self.predictButton.grid(row=5, column=1)
        self.exitButton.grid(row=7, column=0)
        self.outputLabel.grid(row=6, column=0, columnspan=2)
    
    def calculate_age(self, dob):
        """
        Calculate age based on date of birth.
        """
        today = date.today()
        return today.year - dob.year - ((today.month, today.day) < (dob.month, dob.day))
    
    def predict_salary(self):
        """
        Predict the salary based on the experience entered by the user.
        """
        try:
            # Check if the CSV file exists
            if os.path.exists('empsal.csv'):
                data = pd.read_csv("empsal.csv", parse_dates=True, infer_datetime_format=True, index_col='empno')
                
                # Calculate age from date of birth
                data['dob'] = pd.to_datetime(data['dob'])
                data['age'] = data['dob'].apply(self.calculate_age)
                
                # Drop unnecessary columns
                data.drop(columns=['empname', 'dob', 'hra'], inplace=True)
                
                # Encode categorical variables
                le = LabelEncoder()
                data['sex'] = le.fit_transform(data['sex'])
                data['state'] = le.fit_transform(data['state'])
                data['city'] = le.fit_transform(data['city'])
                
                # Display correlation matrix and pair plot
                correlation = data.corr()
                sns.heatmap(correlation, annot=True)
                sns.pairplot(data)
                
                # Prepare data for training
                X = data.drop(columns=['state', 'salary', 'city', 'sex', 'age'])
                y = data['salary']
                
                # Initialize Linear Regression model
                lreg = LinearRegression()
                scores = []
                
                # Find the best random state for splitting the data
                for i in range(1000):
                    X_train, X_test, y_train, y_test = train_test_split(X, y, random_state=i, test_size=0.1)
                    lreg.fit(X_train, y_train)
                    scores.append(lreg.score(X_test, y_test))
                
                best_state = scores.index(max(scores))
                X_train, X_test, y_train, y_test = train_test_split(X, y, random_state=best_state, test_size=0.1)
                lreg.fit(X_train, y_train)
                y_pred = lreg.predict(X_test)
                
                # Print R2 score
                print('R2 score:', r2_score(y_test, y_pred))
                
                # Predict the salary based on user input
                new_experience = np.array([[float(self.experience.get())]])
                predicted_salary = lreg.predict(new_experience)
                self.outputLabel.config(text='Expected Salary is: ' + str(predicted_salary[0]), font=('Arial', 15))
                
        except FileNotFoundError as e:
            msg.showerror('File path is wrong or file not found', str(e))

if __name__ == "__main__":
    root = Tk()
    root.title("Salary Prediction Application")
    root.geometry('500x300')
    app = PredictSalary(root)
    root.mainloop()
