import pandas as pd

import csv
import random

import operator
def loaddataset(filename,split,training=[],test = []):
    with open(filename,'r') as csvfile:
        lines = csv.reader(csvfile)
        dataset = list(lines)
        for x in range(1,len(dataset)-1):
            for y in range(1,3):
                dataset[x][y] = float(dataset[x][y])
                if random.random() <split:
                    training.append(dataset[x])
                else:
                    test.append(dataset[x])
def ecsdist(i1,i2,leng):
    dist = 0
    for x in range(1,leng):
        dist+=pow((i1[x]-i2[x]),2)
    return math.sqrt(dist)
def getneigh(trainingset,i1,k):
    dist = []
    l1 = len(i1)-1
    for x in range(len(trainingset)-1):
        dist1 = ecsdist(i1,trainingset[x],l1)
        dist.append((trainingset[x],dist1))
    dist.sort(key = operator.itemgetter(1))
    neighbours = []
    for x in range(k):
        neighbours.append(dist[x][0])
    return neighbours
def getresp(neighbours):
    vote = {}
    for x in range(len(neighbours)):
        response = neighbours[x][0]
        if response in vote:
            vote[response]+=1
        else:
            vote[response]= 1
    sortedvotes = sorted(vote.items(),key = operator.itemgetter(1),reverse = True)
    return sortedvotes[0][0]
"""function for calculating accuracy"""
def getaccu(test,predict):
    correct = 0
    for x in range(len(test)):
        if test[x][0] == predict[x]:
            correct+=1
    p = (correct/float(len(test)))*100
    return p
"""function for confusion matrix"""
def cf(predict,test):
    cm = {'mm':0,'mf':0,'fm':0,'ff':0}
    for x in range(len(test)):
        if predict[x] == 'Male':
            if test[x][0] == predict[x]:
                cm['mm']+=1
            else:
                cm['mf']+=1
        else:
            if test[x][0] == predict[x]:
                cm['ff']+=1
            else:
                cm['fm']+=1
    return cm
