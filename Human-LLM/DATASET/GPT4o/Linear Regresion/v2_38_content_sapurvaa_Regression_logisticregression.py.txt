import math
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.metrics import confusion_matrix, accuracy_score, classification_report

def cost_func(theta, X_train, y_train):
    """Calculate the cost function for logistic regression."""
    z = np.dot(X_train, theta)
    hyp = 1.0 / (1.0 + np.exp(-z))
    total = np.sum(-(y_train * np.log(hyp)) - ((1 - y_train) * np.log(1 - hyp)))
    cost = total / len(X_train)
    return cost

def gradient_descent(theta, X_train, y_train, alpha=0.001, num_iterations=400):
    """Perform gradient descent to learn theta."""
    for _ in range(num_iterations):
        z = np.dot(X_train, theta)
        hyp = 1.0 / (1.0 + np.exp(-z))
        total = np.dot(X_train.T, (hyp - y_train))
        theta = theta - ((alpha * total) / len(X_train))
    return theta

def feature_scaling(unscaled_data):
    """Scale the features of the dataset."""
    data = unscaled_data.copy()
    data[:, :-1] = (unscaled_data[:, :-1] - np.mean(unscaled_data[:, :-1], axis=0)) / (np.max(unscaled_data[:, :-1], axis=0) - np.min(unscaled_data[:, :-1], axis=0))
    return data

if __name__ == '__main__':
    # Load the dataset
    complete_data = pd.read_csv("BSOM_DataSet_for_HW2.csv", usecols=["all_mcqs_avg_n20", "all_NBME_avg_n4", "LEVEL"])
    features = complete_data.shape[1]
    unscaled_data = complete_data.to_numpy()
    
    # Feature scaling
    scaled_data = feature_scaling(unscaled_data)
    data = np.append(np.ones([len(scaled_data), 1]), scaled_data, axis=1)
    
    X = data[:, :-1]
    y = data[:, features]
    
    # Split the data into training and test sets
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.33, random_state=27)
    
    y_train = y_train[:, None]
    y_test = y_test[:, None]
    
    labels = ["A", "B", "C", "D"]
    y_master_train = y_train.copy()
    
    final_theta = []
    
    # Train logistic regression classifiers for each label
    for label in labels:
        y_train_binary = np.where(y_master_train == label, 1, 0)
        theta = np.random.rand(X_train.shape[1], 1)
        theta = gradient_descent(theta, X_train, y_train_binary)
        final_theta.append(theta.T.flatten().tolist())
    
    theta_labels = np.array(final_theta)
    z = np.dot(X_test, theta_labels.T)
    hyp = 1.0 / (1.0 + np.exp(-z))
    hypothesis_table = np.argmax(hyp, axis=1)
    
    # Prepare the actual labels for comparison
    y_master_test = y_test.copy()
    actual = np.array([labels.index(label) for label in y_master_test.flatten()])
    
    # Print results
    print("Predicted values:", hypothesis_table)
    print("Actual values:", actual)
    
    results = confusion_matrix(actual, hypothesis_table)
    print("Confusion Matrix:\n", results)
    
    accuracy = accuracy_score(actual, hypothesis_table)
    print("Accuracy:", accuracy)
    
    print(classification_report(actual, hypothesis_table, target_names=labels))
