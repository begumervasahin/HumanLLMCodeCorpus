import argparse
from typing import Tuple, List

# Messaging functions for logging
def normal_message(msg: str):
    print(msg)

def success_message(msg: str):
    print(msg)

# Function to normalize data
def normalize_data(values: List[float]) -> List[float]:
    """
    Normalizes the data to a range between 0 and 1.
    """
    maximum = max(values)
    minimum = min(values)
    if maximum != minimum:
        return [(num - minimum) / (maximum - minimum) for num in values]
    return [num - minimum for num in values]

# Function to predict values based on theta
def predict(X: List[float], theta: List[float]) -> List[float]:
    """
    Predicts the output using the linear model parameters theta.
    """
    return [theta[0] + x * theta[1] for x in X]

# Function to compute the cost function
def compute_cost(X: List[float], y: List[float], theta: List[float], use_squared_error: bool) -> float:
    """
    Computes the cost for given theta values.
    """
    predictions = predict(X, theta)
    if use_squared_error:
        return 1 / (2 * len(X)) * sum([(target - pred)**2 for target, pred in zip(y, predictions)])
    else:
        return sum([(target - pred) for target, pred in zip(y, predictions)])

# Function to train the model using gradient descent
def gradient_descent(X: List[float], y: List[float], initial_thetas: List[float], options: argparse.Namespace) -> Tuple[List[float], List[List[float]], List[float]]:
    """
    Trains the model using gradient descent.
    """
    theta0, theta1 = initial_thetas
    m = len(X)
    theta_history: List[List[float]] = []
    cost_history: List[float] = []

    if options.verbose:
        normal_message('              Thetas values              | Cost function value')

    for iteration in range(options.iter):
        sum_errors_theta0, sum_errors_theta1 = 0.0, 0.0

        for i in range(m):
            error = (theta0 + theta1 * X[i]) - y[i]
            sum_errors_theta0 += error
            sum_errors_theta1 += error * X[i]

        theta0 -= options.alpha * (sum_errors_theta0 / m)
        theta1 -= options.alpha * (sum_errors_theta1 / m)
        
        current_cost = compute_cost(X, y, [theta0, theta1], options.square)
        if options.verbose:
            normal_message(f'{theta0: .16f}, {theta1: .16f} | {current_cost: .16f}')
        
        if options.auto_break and cost_history and cost_history[-1] < current_cost:
            normal_message('Model is fitted')
            if options.verbose:
                normal_message(f'Number of iterations: {iteration}')
            normal_message(f'Final theta values: {initial_thetas}')
            return [theta0, theta1], theta_history, cost_history

        initial_thetas = [theta0, theta1]
        theta_history.append(initial_thetas)
        cost_history.append(current_cost)

    success_message('Model is fitted')
    if options.verbose:
        normal_message(f'Number of iterations: {options.iter}')
    success_message(f'Final theta values: {initial_thetas}')
    return initial_thetas, theta_history, cost_history

# Function to estimate price based on raw theta values
def raw_estimated_price(x: float, theta: List[float]) -> float:
    return theta[0] + theta[1] * x

# Function to scale estimated prices to the original scale
def scale_estimated_prices(X: List[float], theta: List[float], mile_lim: List[float], mile_delta: float, price_lim: List[float]) -> List[float]:
    scaled_prices = []
    for x in X:
        normalized_miles = (x - mile_lim[0]) / mile_delta
        price = raw_estimated_price(normalized_miles, theta)
        scaled_price = price * (price_lim[1] - price_lim[0]) + price_lim[0]
        scaled_prices.append(scaled_price)
    return scaled_prices

# Main function to run the script
def main():
    parser = argparse.ArgumentParser(description='Train a simple linear regression model.')
    parser.add_argument('--alpha', type=float, default=0.01, help='Learning rate')
    parser.add_argument('--iter', type=int, default=1000, help='Number of iterations')
    parser.add_argument('--verbose', action='store_true', help='Print progress')
    parser.add_argument('--square', action='store_true', help='Use squared cost function')
    parser.add_argument('--auto_break', action='store_true', help='Automatically stop if cost increases')
    options = parser.parse_args()

    # Example data
    X = [1, 2, 3, 4, 5]
    y = [2, 4, 6, 8, 10]
    initial_thetas = [0.0, 0.0]

    # Normalize data
    X_norm = normalize_data(X)
    y_norm = normalize_data(y)

    # Train the model
    final_thetas, theta_history, cost_history = gradient_descent(X_norm, y_norm, initial_thetas, options)

    # Predict prices
    predicted_prices = scale_estimated_prices(X, final_thetas, [min(X), max(X)], max(X) - min(X), [min(y), max(y)])
    print("Predicted prices:", predicted_prices)

if __name__ == "__main__":
    main()
