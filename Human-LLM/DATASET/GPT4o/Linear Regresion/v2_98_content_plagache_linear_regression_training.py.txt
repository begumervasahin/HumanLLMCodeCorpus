import pandas as pd
from linear_function import linear_function

# Load data from CSV file
data = pd.read_csv('data.csv')
data_frame = pd.DataFrame(data)
x_values = data_frame["km"]
y_values = data_frame["price"]

# Calculate basic statistics for normalization
data_set_size = x_values.count()
x_min, x_max = x_values.min(), x_values.max()
y_min, y_max = y_values.min(), y_values.max()

def normalize_values(values, min_value, max_value):
    """
    Normalize a list of values to the range [0, 1].
    
    Parameters:
    values (pd.Series): The values to normalize.
    min_value (float): The minimum value for normalization.
    max_value (float): The maximum value for normalization.
    
    Returns:
    list: Normalized values.
    """
    return [(value - min_value) / (max_value - min_value) for value in values]

def normalize_data(x_values, y_values):
    """
    Normalize the x and y values and return a DataFrame with the normalized data.
    
    Parameters:
    x_values (pd.Series): The x values (independent variable).
    y_values (pd.Series): The y values (dependent variable).
    
    Returns:
    pd.DataFrame: DataFrame containing normalized x and y values.
    """
    x_normalized = normalize_values(x_values, x_min, x_max)
    y_normalized = normalize_values(y_values, y_min, y_max)
    return pd.DataFrame({"xs": x_normalized, "ys": y_normalized})

normalized_matrix = normalize_data(x_values, y_values)
xs, ys = normalized_matrix["xs"], normalized_matrix["ys"]

def denormalize_value(value, min_value, max_value):
    """
    Denormalize a single value from the range [0, 1] to the original range.
    
    Parameters:
    value (float): The normalized value to denormalize.
    min_value (float): The minimum value for denormalization.
    max_value (float): The maximum value for denormalization.
    
    Returns:
    float: Denormalized value.
    """
    return value * (max_value - min_value) + min_value

def denormalize_values(normalized_values, min_value, max_value):
    """
    Denormalize a list of values from the range [0, 1] to the original range.
    
    Parameters:
    normalized_values (list): The normalized values to denormalize.
    min_value (float): The minimum value for denormalization.
    max_value (float): The maximum value for denormalization.
    
    Returns:
    list: Denormalized values.
    """
    return [denormalize_value(value, min_value, max_value) for value in normalized_values]

def denormalize_matrix(normalized_matrix):
    """
    Denormalize the x and y values from the normalized DataFrame.
    
    Parameters:
    normalized_matrix (pd.DataFrame): DataFrame containing normalized x and y values.
    
    Returns:
    pd.DataFrame: DataFrame containing denormalized x and y values.
    """
    denormalized_x = denormalize_values(normalized_matrix["xs"], x_min, x_max)
    denormalized_y = denormalize_values(normalized_matrix["ys"], y_min, y_max)
    return pd.DataFrame({"km": denormalized_x, "price": denormalized_y})

def denormalize_thetas(normalized_theta_0, normalized_theta_1):
    """
    Denormalize the theta values to the original scale.
    
    Parameters:
    normalized_theta_0 (float): The normalized theta_0 value.
    normalized_theta_1 (float): The normalized theta_1 value.
    
    Returns:
    tuple: Denormalized theta_0 and theta_1 values.
    """
    y_mean = y_values.mean()
    x_mean = x_values.mean()
    y_range = y_max - y_min
    x_range = x_max - x_min
    range_ratio = y_range / x_range
    
    d_theta_1 = normalized_theta_1 * range_ratio
    d_theta_0 = (normalized_theta_0 * range_ratio + y_mean) - (d_theta_1 * x_mean)
    return d_theta_0, d_theta_1

def save_thetas(theta_0, theta_1):
    """
    Save theta values to a CSV file.
    
    Parameters:
    theta_0 (float): The theta_0 value.
    theta_1 (float): The theta_1 value.
    """
    with open("thetas.csv", "a+") as f:
        f.write(f"{theta_0}, {theta_1}\n")

def get_thetas():
    """
    Retrieve the last saved theta values from the CSV file.
    
    Returns:
    tuple: The last saved theta_0 and theta_1 values.
    """
    thetas = pd.read_csv('thetas.csv')
    return thetas["theta0"].values[-1], thetas["theta1"].values[-1]

def gradient_descent(theta_0, theta_1, total_iterations=1000, learning_rate=0.3):
    """
    Perform gradient descent to find the optimal theta values.
    
    Parameters:
    theta_0 (float): The initial theta_0 value.
    theta_1 (float): The initial theta_1 value.
    total_iterations (int): The number of iterations for gradient descent.
    learning_rate (float): The learning rate for gradient descent.
    
    Returns:
    tuple: The optimized theta_0 and theta_1 values.
    """
    for _ in range(total_iterations):
        sum_loss_y = sum_loss_x = 0
        for x, y in zip(xs, ys):
            estimate = linear_function(theta_1, theta_0, float(x))
            loss_y = estimate - y
            loss_x = loss_y * x
            sum_loss_y += loss_y
            sum_loss_x += loss_x
        gradient_theta_0 = sum_loss_y / data_set_size
        gradient_theta_1 = sum_loss_x / data_set_size
        theta_0 -= learning_rate * gradient_theta_0
        theta_1 -= learning_rate * gradient_theta_1
        dt0, dt1 = denormalize_thetas(theta_0, theta_1)
        save_thetas(dt0, dt1)
    return denormalize_thetas(theta_0, theta_1)

# Load initial theta values
base_theta_0, base_theta_1 = get_thetas()

# Perform gradient descent to optimize theta values
dt0, dt1 = gradient_descent(base_theta_0, base_theta_1)

# Print the final denormalized theta values
print("\nLast denormalized theta 0 =", dt0)
print("\nLast denormalized theta 1 =", dt1)
