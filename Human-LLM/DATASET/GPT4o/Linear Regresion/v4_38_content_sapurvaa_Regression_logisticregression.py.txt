import math
import pandas as pd
from sklearn.model_selection import train_test_split
import numpy as np
from sklearn.metrics import confusion_matrix, accuracy_score, classification_report

# Define the cost function
def cost_func(theta, X_train, y_train):
    z = np.dot(X_train, theta)
    hyp = 1.0 / (1.0 + np.exp(-z))
    total = np.sum(-(y_train * np.log(hyp)) - ((1 - y_train) * np.log(1 - hyp)))
    cost = total / len(X_train)
    return cost

# Define the gradient descent function
def gradient_descent(theta, X_train, y_train, alpha=0.001, iterations=400):
    for _ in range(iterations):
        z = np.dot(X_train, theta)
        hyp = 1.0 / (1.0 + np.exp(-z))
        gradient = np.dot(X_train.T, (hyp - y_train))
        theta -= (alpha * gradient) / len(X_train)
    return theta

# Define the feature scaling function
def feature_scaling(unscaled_data):
    scaled_data = unscaled_data.copy()
    scaled_data[:, :-1] = (unscaled_data[:, :-1] - np.mean(unscaled_data[:, :-1], axis=0)) / (np.max(unscaled_data[:, :-1], axis=0) - np.min(unscaled_data[:, :-1], axis=0))
    return scaled_data

# Main execution
if __name__ == '__main__':
    # Load and prepare the dataset
    complete_data = pd.read_csv("BSOM_DataSet_for_HW2.csv", usecols=["all_mcqs_avg_n20", "all_NBME_avg_n4", "LEVEL"])
    unscaled_data = complete_data.to_numpy()
    temp_data = feature_scaling(unscaled_data)
    data = np.append(np.ones([len(temp_data), 1]), temp_data, axis=1)
    X = data[:, :-1]
    y = data[:, -1]
    
    # Split the data into training and test sets
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.33, random_state=27)
    
    # Reshape y for compatibility
    y_train = y_train[:, None]
    y_test = y_test[:, None]
    
    labels = ["A", "B", "C", "D"]
    y_master_train = y_train.copy()
    final_theta = []

    # Train the model for each label
    for label in labels:
        y_train = np.where(y_master_train == label, 1, 0)
        theta = np.random.rand(X_train.shape[1], 1)
        
        initial_cost = cost_func(theta, X_train, y_train)
        print(f"Initial cost for label {label}: {initial_cost}")
        
        theta = gradient_descent(theta, X_train, y_train)
        final_cost = cost_func(theta, X_train, y_train)
        print(f"Final cost for label {label}: {final_cost}")
        
        final_theta.append(theta.T.flatten().tolist())
    
    # Evaluate the model
    theta_labels = np.array(final_theta)
    z = np.dot(X_test, theta_labels.T)
    hyp = 1.0 / (1.0 + np.exp(-z))
    hypothesis_table = np.argmax(hyp, axis=1)
    print("Predicted values:", hypothesis_table)
    
    y_test_labels = [labels.index(label) for label in y_test.flatten()]
    print("Actual values:", y_test_labels)
    
    # Confusion matrix and evaluation metrics
    results = confusion_matrix(y_test_labels, hypothesis_table)
    print("Confusion matrix:\n", results)
    
    accuracy = accuracy_score(y_test_labels, hypothesis_table)
    print("Accuracy:", accuracy)
    
    print("Classification report:\n", classification_report(y_test_labels, hypothesis_table, target_names=labels))
