import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import statsmodels.formula.api as smf
from sklearn.metrics import mean_squared_error
from math import sqrt

# Load the dataset
data_path = 'C:/Users/HP/Desktop/assignments submission/simple linear regression/Salary_Data.csv'
sal_hike = pd.read_csv(data_path)

# Scatter plot of YearsExperience vs Salary
plt.scatter(sal_hike.YearsExperience, sal_hike.Salary, color='r')
plt.xlabel('Years of Experience')
plt.ylabel('Salary')
plt.title('Years of Experience vs Salary')
plt.show()

# Correlation between YearsExperience and Salary
correlation = np.corrcoef(sal_hike.YearsExperience, sal_hike.Salary)[0, 1]
print(f"Correlation: {correlation}")

# Linear Regression Model
lin_model = smf.ols('Salary ~ YearsExperience', data=sal_hike).fit()
print(lin_model.summary())

# Predictions using Linear Regression Model
lin_predict = lin_model.predict(sal_hike)
lin_rmse = sqrt(mean_squared_error(sal_hike.Salary, lin_predict))
print(f"Linear Regression RMSE: {lin_rmse}")

# Plot Linear Regression results
plt.scatter(sal_hike.YearsExperience, sal_hike.Salary, color='r')
plt.plot(sal_hike.YearsExperience, lin_predict, color='b')
plt.xlabel('Years of Experience')
plt.ylabel('Salary')
plt.title('Linear Regression: Years of Experience vs Salary')
plt.show()

# Logarithmic Regression Model
log_model = smf.ols('Salary ~ np.log(YearsExperience)', data=sal_hike).fit()
print(log_model.summary())

# Predictions using Logarithmic Regression Model
log_predict = log_model.predict(sal_hike)
log_rmse = sqrt(mean_squared_error(sal_hike.Salary, log_predict))
print(f"Logarithmic Regression RMSE: {log_rmse}")

# Plot Logarithmic Regression results
plt.scatter(sal_hike.YearsExperience, sal_hike.Salary, color='r')
plt.plot(sal_hike.YearsExperience, log_predict, color='b')
plt.xlabel('Years of Experience')
plt.ylabel('Salary')
plt.title('Logarithmic Regression: Years of Experience vs Salary')
plt.show()

# Exponential Regression Model
exp_model = smf.ols('np.log(Salary) ~ YearsExperience', data=sal_hike).fit()
print(exp_model.summary())

# Predictions using Exponential Regression Model
exp_predict = np.exp(exp_model.predict(sal_hike))
exp_rmse = sqrt(mean_squared_error(sal_hike.Salary, exp_predict))
print(f"Exponential Regression RMSE: {exp_rmse}")

# Plot Exponential Regression results
plt.scatter(sal_hike.YearsExperience, sal_hike.Salary, color='r')
plt.plot(sal_hike.YearsExperience, exp_predict, color='b')
plt.xlabel('Years of Experience')
plt.ylabel('Salary')
plt.title('Exponential Regression: Years of Experience vs Salary')
plt.show()

# Quadratic Regression Model
quad_model = smf.ols('Salary ~ YearsExperience + I(YearsExperience ** 2)', data=sal_hike).fit()
print(quad_model.summary())

# Predictions using Quadratic Regression Model
quad_predict = quad_model.predict(sal_hike)
quad_rmse = sqrt(mean_squared_error(sal_hike.Salary, quad_predict))
print(f"Quadratic Regression RMSE: {quad_rmse}")

# Plot Quadratic Regression results
plt.scatter(sal_hike.YearsExperience, sal_hike.Salary, color='r')
plt.plot(sal_hike.YearsExperience, quad_predict, color='b')
plt.xlabel('Years of Experience')
plt.ylabel('Salary')
plt.title('Quadratic Regression: Years of Experience vs Salary')
plt.show()
