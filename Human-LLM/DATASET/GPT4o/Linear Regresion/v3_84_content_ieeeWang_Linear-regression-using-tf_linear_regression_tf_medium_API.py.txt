import tensorflow as tf
import numpy as np
from matplotlib import pyplot as plt
from tensorflow.keras import layers, losses, optimizers

# Function to print a timestamp bar for logging purposes
@tf.function
def print_bar():
    today_ts = tf.timestamp() % (24 * 60 * 60)
    hour = tf.cast(today_ts // 3600 + 2, tf.int32) % tf.constant(24)
    minute = tf.cast((today_ts % 3600) // 60, tf.int32)
    second = tf.cast(tf.floor(today_ts % 60), tf.int32)

    def format_time(value):
        return tf.strings.format("{:02d}", value)

    time_string = tf.strings.join([format_time(hour), format_time(minute), format_time(second)], separator=":")
    tf.print("=" * 64 + time_string)

# Generate synthetic data
def generate_data(num_samples=400):
    X = tf.random.uniform([num_samples, 2], minval=-10, maxval=10)
    w_true = tf.constant([[2.0], [-3.0]])
    b_true = tf.constant([[3.0]])
    Y = X @ w_true + b_true + tf.random.normal([num_samples, 1], mean=0.0, stddev=2.0)
    return X, Y

# Plot the synthetic data
def plot_data(X, Y):
    plt.figure(figsize=(12, 5))

    ax1 = plt.subplot(121)
    ax1.scatter(X[:, 0], Y[:, 0], c="b")
    ax1.set_xlabel("x1")
    ax1.set_ylabel("y", rotation=0)

    ax2 = plt.subplot(122)
    ax2.scatter(X[:, 1], Y[:, 0], c="g")
    ax2.set_xlabel("x2")
    ax2.set_ylabel("y", rotation=0)

    plt.show()

# Create TensorFlow dataset
def create_dataset(X, Y, buffer_size=100, batch_size=10):
    return tf.data.Dataset.from_tensor_slices((X, Y)).shuffle(buffer_size).batch(batch_size).prefetch(tf.data.experimental.AUTOTUNE)

# Training step function
@tf.function
def train_step(model, features, labels):
    with tf.GradientTape() as tape:
        predictions = model(features)
        loss = model.loss_func(tf.reshape(labels, [-1]), tf.reshape(predictions, [-1]))
    gradients = tape.gradient(loss, model.variables)
    model.optimizer.apply_gradients(zip(gradients, model.variables))
    return loss

# Function to train the model
def train_model(model, dataset, epochs):
    for epoch in tf.range(1, epochs + 1):
        epoch_loss = tf.constant(0.0)
        for features, labels in dataset:
            epoch_loss = train_step(model, features, labels)
        if epoch % 50 == 0:
            print_bar()
            tf.print(f"Epoch = {epoch}, Loss = {epoch_loss}")
            tf.print(f"Weights = {model.variables[0]}")
            tf.print(f"Bias = {model.variables[1]}")

# Plot the model predictions against the data
def plot_model_predictions(X, Y, weights, bias):
    plt.figure(figsize=(12, 5))

    ax1 = plt.subplot(121)
    ax1.scatter(X[:, 0], Y[:, 0], c="b", label="Samples")
    ax1.plot(X[:, 0], weights[0] * X[:, 0] + bias[0], "-r", linewidth=5.0, label="Model")
    ax1.legend()
    ax1.set_xlabel("x1")
    ax1.set_ylabel("y", rotation=0)

    ax2 = plt.subplot(122)
    ax2.scatter(X[:, 1], Y[:, 0], c="g", label="Samples")
    ax2.plot(X[:, 1], weights[1] * X[:, 1] + bias[0], "-r", linewidth=5.0, label="Model")
    ax2.legend()
    ax2.set_xlabel("x2")
    ax2.set_ylabel("y", rotation=0)

    plt.show()

# Main function to execute the workflow
def main():
    # Generate and plot synthetic data
    X, Y = generate_data()
    plot_data(X, Y)

    # Create dataset
    dataset = create_dataset(X, Y)

    # Define and compile model
    model = layers.Dense(units=1)
    model.build(input_shape=(2,))
    model.loss_func = losses.mean_squared_error
    model.optimizer = optimizers.SGD(learning_rate=0.001)

    # Train the model
    train_model(model, dataset, epochs=200)

    # Get the trained weights and bias
    weights, bias = model.variables
    tf.print(f"Trained Weights = {weights}")
    tf.print(f"Trained Bias = {bias}")

    # Plot the model predictions
    plot_model_predictions(X, Y, weights, bias)

# Run the main function
if __name__ == "__main__":
    main()
