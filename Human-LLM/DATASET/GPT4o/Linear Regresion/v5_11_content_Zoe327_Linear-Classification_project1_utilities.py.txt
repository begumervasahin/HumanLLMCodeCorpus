import numpy as np

class LDAClassification:
    def __init__(self):
        self.w0 = None
        self.weights = None
    
    def fit(self, train_x, train_y):
        """
        Fit the LDA model to the training data.
        """
        N, M = train_x.shape
        N1 = np.sum(train_y)
        P1 = N1 / N
        P0 = 1 - P1
        
        # Calculate means of the two classes
        u0, u1 = np.zeros(M), np.zeros(M)
        for i in range(N):
            if train_y[i] == 0:
                u0 += train_x[i]
            else:
                u1 += train_x[i]
        
        u0 /= (N - N1)
        u1 /= N1
        
        # Calculate the pooled covariance matrix
        sigma = np.zeros((M, M))
        for i in range(N):
            diff = train_x[i] - (u0 if train_y[i] == 0 else u1)
            sigma += np.outer(diff, diff)
        
        sigma /= (N - 2)
        sigma_inv = np.linalg.inv(sigma)
        
        # Calculate the intercept and weights
        self.w0 = np.log(P1 / P0) - 0.5 * (u1 @ sigma_inv @ u1) + 0.5 * (u0 @ sigma_inv @ u0)
        self.weights = sigma_inv @ (u1 - u0)
    
    def predict(self, test_x):
        """
        Predict the class labels for the test data.
        """
        log_odds = self.w0 + test_x @ self.weights
        return (log_odds > 0).astype(int)
    
    def evaluate(self, test_x, test_y):
        """
        Evaluate the accuracy of the model on the test data.
        """
        predictions = self.predict(test_x)
        accuracy = np.mean(predictions == test_y) * 100
        return accuracy
    
    def test(self, test_x, test_y):
        """
        Test the model on the test data and return the accuracy.
        """
        return self.evaluate(test_x, test_y)

class LogisticRegressionModel:
    def __init__(self):
        self.weights = None
        self.alpha = None
    
    def sigmoid(self, z):
        """
        Compute the sigmoid function.
        """
        return 1.0 / (1.0 + np.exp(-z))
    
    def fit(self, train_x, train_y, tol=1e-4):
        """
        Fit the logistic regression model to the training data.
        """
        N, M = train_x.shape
        train_y = train_y.reshape(-1, 1)
        self.weights = np.zeros((M, 1))
        
        mse = float('inf')
        i = 0
        
        # Gradient descent optimization
        while mse > tol:
            self.alpha = 1 / (1 + i) + 1e-5
            h = self.sigmoid(train_x @ self.weights)
            loss = train_y - h
            gradient = train_x.T @ loss
            new_weights = self.weights + self.alpha * gradient
            
            mse = np.mean((new_weights - self.weights)**2)
            self.weights = new_weights
            i += 1
    
    def predict(self, test_x):
        """
        Predict the class labels for the test data.
        """
        probabilities = self.sigmoid(test_x @ self.weights)
        return (probabilities > 0.5).astype(int).flatten()
    
    def evaluate(self, test_x, test_y):
        """
        Evaluate the accuracy of the model on the test data.
        """
        predictions = self.predict(test_x)
        accuracy = np.mean(predictions == test_y) * 100
        return accuracy
    
    def test(self, test_x, test_y):
        """
        Test the model on the test data and return the accuracy.
        """
        return self.evaluate(test_x, test_y)
