import numpy as np
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error
from numpy.linalg import inv

# Define training and testing data
x_train = np.array([[0], [0.5], [2]])
y_train = np.array([-1.25, -0.6, -4.85])
x_test = np.array([[-1], [1], [3]])
y_test = np.array([-5.2, -0.9, -13])

# Linear Regression Model
lr_model = LinearRegression()
lr_model.fit(x_train, y_train)

w0 = lr_model.intercept_
w1 = lr_model.coef_

print(f'Intercept (w0) = {w0}')
print(f'Coefficient (w1) = {w1}')

y_train_pred = lr_model.predict(x_train)
y_test_pred = lr_model.predict(x_test)

train_mse = mean_squared_error(y_train, y_train_pred)
test_mse = mean_squared_error(y_test, y_test_pred)

print(f'Train MSE: {train_mse}')
print(f'Test MSE: {test_mse}')

# Plot Linear Regression
x_range = np.linspace(-3, 5, 100)
y_fit = w1 * x_range + w0

plt.figure(figsize=(10, 6))
plt.plot(x_range, y_fit, '--', color='black', label='Linear Regression Line')
plt.scatter(x_train, y_train, marker='*', label='Training Data')
plt.scatter(x_test, y_test, marker='o', label='Testing Data')
plt.xlabel('X')
plt.ylabel('Y')
plt.legend()
plt.title('Linear Regression')
plt.show()

# Least Squares Estimator Function
def least_squares_estimator(X, y):
    X_with_intercept = np.hstack((np.ones((X.shape[0], 1)), X))
    X_transpose = np.transpose(X_with_intercept)
    X_transpose_X = X_transpose @ X_with_intercept
    X_transpose_X_inv = inv(X_transpose_X)
    X_transpose_y = X_transpose @ y
    return X_transpose_X_inv @ X_transpose_y

# Prepare data for Polynomial Regression
X_train_poly = np.hstack((x_train, x_train**2))
X_test_poly = np.hstack((x_test, x_test**2))

# Estimate polynomial coefficients using least squares
poly_coeffs = least_squares_estimator(X_train_poly, y_train)
gamma0, gamma1, gamma2 = poly_coeffs

print(f'Polynomial Coefficients (gamma0, gamma1, gamma2): {poly_coeffs}')

# Polynomial Regression Model
pr_model = LinearRegression()
pr_model.fit(X_train_poly, y_train)

y_train_poly_pred = pr_model.predict(X_train_poly)
y_test_poly_pred = pr_model.predict(X_test_poly)

poly_train_mse = mean_squared_error(y_train, y_train_poly_pred)
poly_test_mse = mean_squared_error(y_test, y_test_poly_pred)

print(f'Polynomial Train MSE: {poly_train_mse}')
print(f'Polynomial Test MSE: {poly_test_mse}')

# Plot Polynomial Regression
x_range_poly = np.linspace(-3, 5, 100)
y_poly_fit = gamma0 + gamma1 * x_range_poly + gamma2 * (x_range_poly**2)

plt.figure(figsize=(10, 6))
plt.plot(x_range_poly, y_poly_fit, '--', color='red', label='Polynomial Regression Line')
plt.scatter(x_train, y_train, marker='*', label='Training Data')
plt.scatter(x_test, y_test, marker='o', label='Testing Data')
plt.xlabel('X')
plt.ylabel('Y')
plt.legend()
plt.title('Polynomial Regression vs. Linear Regression')
plt.show()

# Conclusion about model performance
"""
The linear regression model fits the training data well but performs poorly on the test data.
In contrast, the polynomial regression model, which accounts for non-linearity,
better fits both the training and test data, demonstrating improved performance.
"""
