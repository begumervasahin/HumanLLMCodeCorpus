def prepare_fraud_data(df):
    """
    Prepares the fraud data for modeling by handling missing values,
    encoding categorical variables, and normalizing numerical variables.

    Args:
        df (pd.DataFrame): Input dataframe.

    Returns:
        pd.DataFrame: Processed dataframe.
    """
    selected_columns = ['TransactionAmt', 'card4', 'card6']
    X = df[selected_columns]

    # One-hot encode categorical variables
    X = pd.get_dummies(X, columns=['card4', 'card6'], drop_first=True)

    # Handle missing values
    X['TransactionAmt'] = X['TransactionAmt'].fillna(X['TransactionAmt'].mean())

    # Normalize numerical variables
    X = (X - X.mean()) / X.std()

    return X

def sigmoid(u):
    """
    Computes the sigmoid function.

    Args:
        u (float): Input value.

    Returns:
        float: Sigmoid of the input value.
    """
    return 1 / (1 + np.exp(-u))

def log_likelihood(beta, X, Y):
    """
    Computes the log likelihood for logistic regression.

    Args:
        beta (np.ndarray): Coefficient vector.
        X (np.ndarray): Feature matrix.
        Y (np.ndarray): Target vector.

    Returns:
        float: Log likelihood.
    """
    dot_products = X.dot(beta)
    return np.sum(Y * dot_products - np.log(1 + np.exp(dot_products)))

def logistic_regression(X, Y, alpha=0.0001, max_iterations=30):
    """
    Performs logistic regression using gradient descent.

    Args:
        X (np.ndarray): Feature matrix.
        Y (np.ndarray): Target vector.
        alpha (float): Learning rate.
        max_iterations (int): Maximum number of iterations.

    Returns:
        tuple: Coefficient vector, list of gradient norms, list of log likelihoods.
    """
    N, d = X.shape
    X = np.hstack((np.ones((N, 1)), X))  # Add intercept term
    beta = np.random.randn(d + 1)
    gradient_norms = []
    log_likelihoods = []

    for iteration in range(max_iterations):
        predictions = sigmoid(X.dot(beta))
        gradient = X.T.dot(predictions - Y)
        gradient_norms.append(np.linalg.norm(gradient))
        beta -= alpha * gradient
        log_likelihoods.append(log_likelihood(beta, X, Y))
        print(f"Iteration {iteration}: Log Likelihood = {log_likelihoods[-1]}")

    return beta, gradient_norms, log_likelihoods
