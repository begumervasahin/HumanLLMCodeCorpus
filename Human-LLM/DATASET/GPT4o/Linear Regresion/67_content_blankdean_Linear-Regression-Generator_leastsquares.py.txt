import sys
import csv
print('\nWelcome to the Linear Regression Generator')
print('Enter File Name Containing Data: ', end='')
f = input()
x = 0
y = 0
count = 0
m_num = 0
m_den = 0
xarray = []
yarray = []
i = 0
est = []
err = []
tempt = 0
errorsum = 0
try:
    with open(f, 'r') as csvfile:
        myfile = csv.reader(csvfile)
        next(myfile)
        for row in myfile:
            count += 1
            try:
                row[0] = float(row[0])
                row[1] = float(row[1])
            except ValueError as e:
                print('  Error: A value in the file cannot be read')
                sys.exit(0)
            x = x + float(row[0])
            y = y + float(row[1])
            xarray.append(float(row[0]))
            yarray.append(float(row[1]))
            i += 1
    csvfile.close()
    xavg = float(x/count)
    yavg = float(y/count)
    for i in range(len(xarray)):
        m_num += (xarray[i]-xavg)*(yarray[i]-yavg)
        m_den += (xarray[i]-xavg)**2.0
    m = m_num/m_den
    b = yavg - (m*xavg)
    for i in range(len(xarray)):
        est.append((m*xarray[i]) + b)
    for i in range(len(xarray)):
        err.append(abs(yarray[i]-est[i]))
    for i in range(len(err)):
        errorsum = errorsum + float(err[i])
    for i in range(len(xarray)):
        tempt += (yarray[i]-est[i])**2.0
    erroravg = round(errorsum/len(err), 5)
    mse = tempt/(len(xarray)-2)
    s = round(mse**0.5, 5)
except FileNotFoundError as e:
    print('  Error: File could not be opened.')
    sys.exit(0)
mrounded = round(m, 5)
brounded = round(b, 5)
if brounded >= 0:
    plus_minus = '+'
else:
    plus_minus = ''
print('\n  The Linear Regression Line is y = ' + str(mrounded) + '*x' + str(plus_minus) + str(brounded) + '.')
print('  Average Error for Known Values was +/-' + str(erroravg) + '.')
print('  Regression Standard Error for Known Values was ' + str(s) + '.')
print('  System Ready to make predictions.')
print("  To quit, type 'exit' as the year.")
try:
    with open(f, 'r') as csvfile:
        myfile = csv.reader(csvfile)
        data = []
        for row in csvfile:
            values = row.split(',')
            for value in values:
                data.append(value)
        xheader = data[0]
        yheader = data[1]
        yheader = (yheader)
except FileNotFoundError as e:
    print('  Error: File could not be opened.')
    sys.exit(0)
    csvfile.close()
yheader = yheader.strip()
year = input('  Enter ' + str(xheader) + ': ')
while True:
    try:
        year = int(year)
        prediction = round(m*year + b, 5)
        print('  Prediction when ' + str(xheader) + ' = ' + str(year) + ' is '
        + str(yheader) + ' = ' + str(prediction) + '.')
        print('  ________________________________')
    except:
            if year == 'exit':
                sys.exit(0)
            else:
                print('  Input could not be understood. Please try again.')
    year = input('  Enter ' + str(xheader) + ': ')
