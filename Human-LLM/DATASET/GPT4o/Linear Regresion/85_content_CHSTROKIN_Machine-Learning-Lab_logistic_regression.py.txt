class neuro:
    size_of_one_traning_example=0
    learning_rate=0
    the_train_set_x=[]
    the_train_set_y=[]
    the_weight=[]
    the_constant=0
    def __init__(self,learn,size_examples,train_value_x,trian_value_y):
        self.learning_rate=learn
        self.size_of_one_traning_example=size_examples
        self.the_train_set_x=train_value_x
        self.the_train_set_y=trian_value_y
        self.the_weight=np.random.rand(self.size_of_one_traning_example)
    def random_weight(self):
        self.the_weight=np.random.rand(self.size_of_one_traning_example)
        return
    def print_weight(self):
        print(self.the_weight,' ',self.the_constant)
    def segmoid(self,n):
        return (float)(1/(float)(1+math.exp(-n)))
    def get_input(self,the_input_list=[]):
        self.the_train_set=the_input_list
        return
    def error_func(self,y_pred,y):
        return (y-y_pred)*(y-y_pred)
    def gradient_numeric(self,variable_number):
        pass
    def train(self,train_cycle):
        for i in tqdm(range(0,train_cycle)):
            for j in range(0,len(self.the_train_set_x)):
                one_train_example=self.the_train_set_x[j]
                one_correct_answer=self.the_train_set_y[j]
                the_ret_value=self.get_value(one_train_example)
                mis=self.error_func(the_ret_value,one_correct_answer)
                for k in range(0,len(self.the_weight)):
                    self.the_weight[k]=self.the_weight[k]-self.learning_rate*2*(the_ret_value-one_correct_answer)*the_ret_value*(1-the_ret_value)*one_train_example[k]
                    self.the_constant=self.the_constant-self.learning_rate*2*(the_ret_value-one_correct_answer)*the_ret_value*(1-the_ret_value)
        return
    def get_value(self,one_input_value):
        z=np.dot(one_input_value,self.the_weight)
        a=np.sum(z)
        a+=self.the_constant
        return (float)(self.segmoid(a))
