import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
import statsmodels.formula.api as smf
import statsmodels.api as sm
from sklearn.metrics import mean_squared_error
from math import sqrt

def load_data(file_path, columns):
    """
    Load the data from a CSV file and select relevant columns.
    
    Parameters:
    file_path (str): The path to the CSV file.
    columns (list): List of columns to select from the CSV file.
    
    Returns:
    pd.DataFrame: The selected columns of the data.
    """
    data = pd.read_csv(file_path, encoding="ISO-8859-1")
    return data[columns]

def visualize_data(data):
    """
    Visualize the relationships between variables using pairplot.
    
    Parameters:
    data (pd.DataFrame): The data to visualize.
    """
    sns.pairplot(data)
    plt.show()

def compute_correlation_matrix(data):
    """
    Compute and display the correlation matrix.
    
    Parameters:
    data (pd.DataFrame): The data to compute the correlation matrix for.
    """
    print("Correlation matrix:\n", data.corr())

def fit_regression_model(formula, data):
    """
    Fit a regression model using the specified formula and data.
    
    Parameters:
    formula (str): The formula for the regression model.
    data (pd.DataFrame): The data to fit the model on.
    
    Returns:
    RegressionResultsWrapper: The fitted regression model.
    """
    model = smf.ols(formula, data=data).fit()
    print(f"Model parameters ({formula}):\n", model.params)
    print(f"Model summary ({formula}):\n", model.summary())
    return model

def calculate_vif(data, predictors):
    """
    Calculate the Variance Inflation Factor (VIF) for each predictor.
    
    Parameters:
    data (pd.DataFrame): The data containing the predictors.
    predictors (list): List of predictor variables.
    
    Returns:
    pd.DataFrame: DataFrame containing VIF values for each predictor.
    """
    vif_data = pd.DataFrame()
    vif_data["variables"] = predictors
    vif_data["VIF"] = [
        1 / (1 - smf.ols(f'{predictor} ~ ' + '+'.join([p for p in predictors if p != predictor]), data=data).fit().rsquared)
        for predictor in predictors
    ]
    return vif_data

def plot_partial_regression(model):
    """
    Plot partial regression plots for the specified model.
    
    Parameters:
    model (RegressionResultsWrapper): The fitted regression model.
    """
    sm.graphics.plot_partregress_grid(model)
    plt.show()

def plot_influence(model):
    """
    Plot influence plot for the specified model.
    
    Parameters:
    model (RegressionResultsWrapper): The fitted regression model.
    """
    sm.graphics.influence_plot(model)
    plt.show()

def remove_influential_points(data, points):
    """
    Remove specified influential points from the data.
    
    Parameters:
    data (pd.DataFrame): The data to remove points from.
    points (list): List of indices of influential points to remove.
    
    Returns:
    pd.DataFrame: The data with the influential points removed.
    """
    return data.drop(data.index[points], axis=0)

def calculate_rmse(predictions, actual):
    """
    Calculate the Root Mean Squared Error (RMSE) between predictions and actual values.
    
    Parameters:
    predictions (pd.Series): The predicted values.
    actual (pd.Series): The actual values.
    
    Returns:
    float: The calculated RMSE.
    """
    return sqrt(mean_squared_error(actual, predictions))
