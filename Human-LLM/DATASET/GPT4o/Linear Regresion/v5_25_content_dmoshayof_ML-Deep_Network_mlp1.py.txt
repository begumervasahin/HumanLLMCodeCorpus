def classifier_output(x, params):
    W, b, U, b_tag = params
    h = np.tanh(np.dot(x, W) + b)
    probs = softmax(np.dot(h, U) + b_tag)
    return probs

def predict(x, params):
    """
    Predict the class of input x using the given parameters.
    params: a list of the form [W, b, U, b_tag]
    """
    return np.argmax(classifier_output(x, params))

def loss_and_gradients(x, y, params):
    """
    Compute the loss and gradients for a single training example.
    params: a list of the form [W, b, U, b_tag]
    returns:
        loss: scalar
        gradients: list of [gW, gb, gU, gb_tag]
    """
    W, b, U, b_tag = params
    pred_vec = classifier_output(x, params)
    y_hat = pred_vec[y]
    loss = -np.log(y_hat)

    y_hot_vector = np.zeros_like(pred_vec)
    y_hot_vector[y] = 1

    dl_dz = pred_vec - y_hot_vector
    h = np.dot(x, W) + b

    # Gradients
    gb_tag = dl_dz
    gU = np.outer(dl_dz, np.tanh(h)).T
    dz_dh = U.T * (1 - np.tanh(h) ** 2)
    gb = np.dot(dl_dz, dz_dh)
    gW = np.outer(x, gb)

    return loss, [gW, gb, gU, gb_tag]

def create_classifier(in_dim, hid_dim, out_dim):
    """
    Create and initialize the parameters for a multi-layer perceptron.
    in_dim: input dimension
    hid_dim: hidden layer dimension
    out_dim: output dimension
    return: a list of parameters [W, b, U, b_tag]
    """
    W = np.random.randn(in_dim, hid_dim)
    b = np.random.randn(hid_dim)
    U = np.random.randn(hid_dim, out_dim)
    b_tag = np.random.randn(out_dim)
    return [W, b, U, b_tag]
