import pandas as pd
import numpy as np

class BinaryChoice:
    def __init__(self, data=None, x_cols=None, y_col=None, lr=0.000001, tol=1e-7):
        self.lr = lr
        self.tol = tol
        self.data = data
        self.n = len(data)
        
        self.x = np.c_[np.ones(self.n), self.data[x_cols]]
        self.y = self.data[[y_col]].values
        
        self.i, self.log_like, self.w = self.estimate_parameters(self.x, self.y)
        self.std_errors = self.calculate_standard_errors(self.x, self.w)
        self.mcfadden_r_squared, self.nagelkerke_r_squared = self.calculate_r_squared()
        
        self.result = self.build_result_dataframe(x_cols)
    
    def sigmoid(self, x, w):
        pred = np.dot(x, w)
        return np.where(pred >= 0, 1 / (1 + np.exp(-pred)), np.exp(pred) / (1 + np.exp(pred)))
    
    def update_weights(self, prob, x, y, w):
        gradient = np.dot(x.T, y - prob)
        return w + self.lr * gradient
    
    def calculate_log_likelihood(self, prob, y):
        y_1_prob = np.dot(y.T, np.log(prob + 1e-9))
        y_0_prob = np.dot((1 - y).T, np.log(1 - prob + 1e-9))
        return y_1_prob + y_0_prob
    
    def estimate_parameters(self, x, y):
        w = np.zeros((x.shape[1], 1))
        prob = self.sigmoid(x, w)
        log_likelihood = self.calculate_log_likelihood(prob, y)
        iteration = 1
        
        while True:
            new_w = self.update_weights(prob, x, y, w)
            new_prob = self.sigmoid(x, new_w)
            new_log_likelihood = self.calculate_log_likelihood(new_prob, y)
            
            if log_likelihood >= new_log_likelihood:
                break
            
            log_likelihood = new_log_likelihood
            w = new_w
            prob = new_prob
            iteration += 1
        
        return iteration, log_likelihood, w
    
    def calculate_standard_errors(self, x, w):
        W = np.zeros((x.shape[0], x.shape[0]))
        
        for i in range(x.shape[0]):
            pred = np.dot(x[i], w)
            W[i, i] = pred * (1 - pred)
        
        XtWX_inv = np.linalg.inv(x.T @ W @ x)
        return np.sqrt(np.diag(XtWX_inv))
    
    def calculate_r_squared(self):
        log_like_full = self.log_like
        log_like_intercept = self.calculate_log_likelihood_intercept()
        
        mcfadden_r_squared = 1 - log_like_full / log_like_intercept
        nagelkerke_r_squared = (1 - (np.exp(log_like_intercept) / np.exp(log_like_full)) ** (2 / self.n)) / \
                               (1 - np.exp(log_like_intercept) ** (2 / self.n))
        
        return mcfadden_r_squared, nagelkerke_r_squared
    
    def calculate_log_likelihood_intercept(self):
        x_intercept_only = self.x[:, :1]
        _, log_like_intercept, _ = self.estimate_parameters(x_intercept_only, self.y)
        return log_like_intercept
    
    def build_result_dataframe(self, x_cols):
        index = ['const'] + x_cols
        result = pd.DataFrame(0, index=index, columns=['Coefficient', 'Std. Error', 'Z-value'])
        
        for i, feat in enumerate(index):
            z_value = self.w[i, 0] / self.std_errors[i]
            result.loc[feat, 'Coefficient'] = round(self.w[i, 0], 3)
            result.loc[feat, 'Std. Error'] = round(self.std_errors[i], 3)
            result.loc[feat, 'Z-value'] = round(z_value, 3)
        
        return result
    
    def __repr__(self):
        return (
            f"Regression results:\n\n{self.result}\n\n"
            f"Log-likelihood: {self.log_like[0, 0]:.3f}\n\n"
            f"McFadden R squared: {self.mcfadden_r_squared[0, 0]:.3f}\n\n"
            f"Nagelkerke R squared: {self.nagelkerke_r_squared[0, 0]:.3f}"
        )

# Example usage
data = pd.DataFrame({
    'feature1': [1, 2, 3, 4, 5],
    'feature2': [2, 3, 4, 5, 6],
    'outcome': [0, 1, 0, 1, 0]
})

model = BinaryChoice(data, x_cols=['feature1', 'feature2'], y_col='outcome')
print(model)
