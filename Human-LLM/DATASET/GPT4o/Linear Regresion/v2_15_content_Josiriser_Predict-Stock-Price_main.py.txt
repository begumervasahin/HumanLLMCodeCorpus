import os
import csv
import numpy as np
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression

def load_data(filepath):
    """Load stock data from a CSV file."""
    high_prices = []
    low_prices = []
    close_prices = []
    
    with open(filepath, 'r', newline="", encoding='utf-8') as csvfile:
        reader = csv.reader(csvfile)
        for i, row in enumerate(reader):
            if i == 0:  # Skip the header
                continue
            high_prices.append(float(row[2]))
            low_prices.append(float(row[3]))
            close_prices.append(float(row[4]))
    
    return [high_prices, low_prices, close_prices]

def get_stock_prices(directory):
    """Load stock data from all CSV files in a directory."""
    all_prices = []
    filenames = sorted(os.listdir(directory))
    
    for filename in filenames:
        all_prices.append(load_data(os.path.join(directory, filename)))
    
    return all_prices

def split_data(prices):
    """Split the data into features and targets for training and prediction."""
    high_prices, low_prices, close_prices = prices
    X = []
    Y = []
    real_x = []
    num_samples = len(high_prices) // 13
    
    for i in range(num_samples):
        features = []
        targets = []
        for j in range(i * 13, i * 13 + 3):
            features.extend([high_prices[j], low_prices[j], close_prices[j]])
        for j in range(i * 13 + 3, i * 13 + 13):
            targets.append(close_prices[j])
        X.append(np.array(features))
        Y.append(np.array(targets))
    
    for k in range(len(close_prices) - 13, len(close_prices) - 10):
        real_x.extend([high_prices[k], low_prices[k], close_prices[k]])
    
    real_y = close_prices[-10:]
    return X, Y, real_x, real_y

def plot_predictions(days, real_y, predicted_y, label_real, label_predicted, color_real, color_predicted):
    """Plot the real and predicted stock prices."""
    plt.plot(days, real_y, color_real, label=label_real)
    plt.plot(days, predicted_y, color_predicted, label=label_predicted)

def main():
    """Main function to execute the stock price prediction and plotting."""
    days = list(range(1, 11))
    directory = 'stock_data'
    all_prices = get_stock_prices(directory)

    # Process and plot data for YFY
    yfy_prices = all_prices[0]
    yfy_X, yfy_Y, yfy_real_x, yfy_real_y = split_data(yfy_prices)
    yfy_model = LinearRegression()
    yfy_model.fit(yfy_X, yfy_Y)
    yfy_predicted_y = yfy_model.predict([yfy_real_x])[0]
    plot_predictions(days, yfy_real_y, yfy_predicted_y, "YFY-Real", "YFY-Predict", 'b', 'g')

    # Process and plot data for TMSC
    tmsc_prices = all_prices[1]
    tmsc_X, tmsc_Y, tmsc_real_x, tmsc_real_y = split_data(tmsc_prices)
    tmsc_model = LinearRegression()
    tmsc_model.fit(tmsc_X, tmsc_Y)
    tmsc_predicted_y = tmsc_model.predict([tmsc_real_x])[0]
    plot_predictions(days, tmsc_real_y, tmsc_predicted_y, "TSMC-Real", "TSMC-Predict", 'r', 'c')

    # Process and plot data for MTK
    mtk_prices = all_prices[2]
    mtk_X, mtk_Y, mtk_real_x, mtk_real_y = split_data(mtk_prices)
    mtk_model = LinearRegression()
    mtk_model.fit(mtk_X, mtk_Y)
    mtk_predicted_y = mtk_model.predict([mtk_real_x])[0]
    plot_predictions(days, mtk_real_y, mtk_predicted_y, "MTK-Real", "MTK-Predict", 'm', 'y')

    plt.legend()
    plt.show()

if __name__ == "__main__":
    main()
