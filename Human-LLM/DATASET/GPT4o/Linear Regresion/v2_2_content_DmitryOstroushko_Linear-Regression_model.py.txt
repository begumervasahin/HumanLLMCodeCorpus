import argparse
from typing import Tuple, List

# Messaging functions for logging
def normal_message(msg: str):
    print(msg)

def success_message(msg: str):
    print(msg)

# Function to normalize data
def normalize_data(val: List[float]) -> List[float]:
    """
    Normalizes the data to a range between 0 and 1.
    """
    maximum = max(val)
    minimum = min(val)
    if maximum != minimum:
        return [(num - minimum) / (maximum - minimum) for num in val]
    return [num - minimum for num in val]

# Function to predict values based on theta
def predict(X: List[float], theta: List[float]) -> List[float]:
    """
    Predicts the output using the linear model parameters theta.
    """
    return [theta[0] + x * theta[1] for x in X]

# Function to compute the cost function
def cost(X: List[float], y: List[float], theta: List[float], options: argparse.Namespace) -> float:
    """
    Computes the cost for given theta values.
    """
    y_pred = predict(X, theta)
    if options.square:
        return 1 / (2 * len(X)) * sum([(target - target_pred)**2 for target, target_pred in zip(y, y_pred)])
    else:
        return sum([(target - target_pred) for target, target_pred in zip(y, y_pred)])

# Function to train the model using gradient descent
def fit(X: List[float], y: List[float], thetas: List[float], options: argparse.Namespace) -> Tuple[List[float], List[List[float]], List[float]]:
    """
    Trains the model using gradient descent.
    """
    theta0, theta1 = thetas
    length = len(X)
    theta_history: List[List[float]] = []
    J_history: List[float] = []

    if options.verbose:
        normal_message('              Thetas values              | Cost function value')

    for count in range(options.iter):
        accum_theta0, accum_theta1 = 0.0, 0.0
        for idx in range(length):
            error = theta0 + theta1 * X[idx] - y[idx]
            accum_theta0 += error
            accum_theta1 += error * X[idx]
        
        theta0 -= options.alpha * (accum_theta0 / length)
        theta1 -= options.alpha * (accum_theta1 / length)
        
        J_cost = cost(X, y, [theta0, theta1], options)
        if options.verbose:
            normal_message('{: .16f}, {: .16f} | {: .16f}'.format(theta0, theta1, J_cost))
        
        if options.auto_break and J_history and J_history[-1] < J_cost:
            normal_message('Model is fitted')
            if options.verbose:
                normal_message(f'Number of iterations: {count}')
            normal_message(f'Final theta values: {thetas}')
            return [theta0, theta1], theta_history, J_history

        thetas = [theta0, theta1]
        theta_history.append(thetas)
        J_history.append(J_cost)

    success_message('Model is fitted')
    if options.verbose:
        normal_message(f'Number of iterations: {options.iter}')
    success_message(f'Final theta values: {thetas}')
    return thetas, theta_history, J_history

# Function to estimate price based on raw theta values
def raw_estimated_price(x: float, theta: List[float]) -> float:
    return theta[0] + theta[1] * x

# Function to scale estimated prices to the original scale
def estimated_price(X: List[float], theta: List[float], mile_lim: List[float], mile_delta: float, price_lim: List[float]) -> List[float]:
    price_scaled = []
    for x in X:
        price = raw_estimated_price((x - mile_lim[0]) / mile_delta, theta)
        if not price:
            price_scaled.append(0.0)
        else:
            price_scaled.append(price * (price_lim[1] - price_lim[0]) + price_lim[0])
    return price_scaled

# Main function to run the script
def main():
    parser = argparse.ArgumentParser(description='Train a simple linear regression model.')
    parser.add_argument('--alpha', type=float, default=0.01, help='Learning rate')
    parser.add_argument('--iter', type=int, default=1000, help='Number of iterations')
    parser.add_argument('--verbose', action='store_true', help='Print progress')
    parser.add_argument('--square', action='store_true', help='Use squared cost function')
    parser.add_argument('--auto_break', action='store_true', help='Automatically stop if cost increases')
    options = parser.parse_args()

    # Example data
    X = [1, 2, 3, 4, 5]
    y = [2, 4, 6, 8, 10]
    thetas = [0.0, 0.0]

    # Normalize data
    X_norm = normalize_data(X)
    y_norm = normalize_data(y)

    # Train the model
    thetas, theta_history, J_history = fit(X_norm, y_norm, thetas, options)

    # Predict prices
    price_predictions = estimated_price(X, thetas, [min(X), max(X)], max(X) - min(X), [min(y), max(y)])
    print("Predicted prices:", price_predictions)

if __name__ == "__main__":
    main()
