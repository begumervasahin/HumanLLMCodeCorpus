import pandas as pd
import matplotlib.pyplot as plt
import numpy as np
import csv
import random
import math
import operator

def load_dataset(filename, split_ratio):
    """
    Load dataset from a CSV file and split it into training and test sets.
    
    Args:
    filename (str): Path to the CSV file.
    split_ratio (float): Ratio for splitting the data into training and test sets.

    Returns:
    tuple: Training and test sets.
    """
    training_set = []
    test_set = []
    with open(filename, 'r') as csvfile:
        lines = csv.reader(csvfile)
        dataset = list(lines)
        for row in dataset[1:-1]:  # Skip header and last line
            row[1] = float(row[1])
            row[2] = float(row[2])
            if random.random() < split_ratio:
                training_set.append(row)
            else:
                test_set.append(row)
    return training_set, test_set

def euclidean_distance(instance1, instance2, length):
    """
    Calculate the Euclidean distance between two data points.
    
    Args:
    instance1 (list): First data point.
    instance2 (list): Second data point.
    length (int): Number of features.

    Returns:
    float: Euclidean distance.
    """
    distance = sum(pow(instance1[i] - instance2[i], 2) for i in range(1, length))
    return math.sqrt(distance)

def get_neighbors(training_set, test_instance, k):
    """
    Find k nearest neighbors for a test instance.
    
    Args:
    training_set (list): Training dataset.
    test_instance (list): A single test data point.
    k (int): Number of neighbors to find.

    Returns:
    list: k nearest neighbors.
    """
    distances = [(train_instance, euclidean_distance(test_instance, train_instance, len(test_instance) - 1))
                 for train_instance in training_set]
    distances.sort(key=operator.itemgetter(1))
    neighbors = [distances[i][0] for i in range(k)]
    return neighbors

def get_response(neighbors):
    """
    Get the most common class among neighbors.
    
    Args:
    neighbors (list): k nearest neighbors.

    Returns:
    str: Predicted class.
    """
    class_votes = {}
    for neighbor in neighbors:
        response = neighbor[0]
        class_votes[response] = class_votes.get(response, 0) + 1
    sorted_votes = sorted(class_votes.items(), key=operator.itemgetter(1), reverse=True)
    return sorted_votes[0][0]

def get_accuracy(test_set, predictions):
    """
    Calculate the accuracy of predictions.
    
    Args:
    test_set (list): Test dataset.
    predictions (list): Predicted classes.

    Returns:
    float: Accuracy as a percentage.
    """
    correct = sum(1 for i in range(len(test_set)) if test_set[i][0] == predictions[i])
    return (correct / float(len(test_set))) * 100.0

def confusion_matrix(predictions, test_set):
    """
    Create a confusion matrix for the predictions.
    
    Args:
    predictions (list): Predicted classes.
    test_set (list): Test dataset.

    Returns:
    dict: Confusion matrix.
    """
    cm = {'mm': 0, 'mf': 0, 'fm': 0, 'ff': 0}
    for i in range(len(test_set)):
        if predictions[i] == 'Male':
            if test_set[i][0] == predictions[i]:
                cm['mm'] += 1
            else:
                cm['mf'] += 1
        else:
            if test_set[i][0] == predictions[i]:
                cm['ff'] += 1
            else:
                cm['fm'] += 1
    return cm

def plot_data(training_set):
    """
    Plot height vs weight data with different colors for each gender.
    
    Args:
    training_set (list): Training dataset.
    """
    heights_male = [data[2] for data in training_set if data[0] == 'Male']
    weights_male = [data[1] for data in training_set if data[0] == 'Male']
    heights_female = [data[2] for data in training_set if data[0] == 'Female']
    weights_female = [data[1] for data in training_set if data[0] == 'Female']
    
    plt.scatter(heights_male, weights_male, c='blue', label='Male')
    plt.scatter(heights_female, weights_female, c='red', label='Female')
    plt.xlabel("Height")
    plt.ylabel("Weight")
    plt.legend()
    plt.show()

def main():
    # Load and prepare data
    training_set, test_set = load_dataset("weight-height.csv", 0.7)
    print(f'Train set: {len(training_set)}')
    print(f'Test set: {len(test_set)}')

    # Plot data
    plot_data(training_set)

    # Set the value of k
    k = int(math.sqrt(len(training_set)))
    print(f'Value of k: {k}')

    # Make predictions
    predictions = [get_response(get_neighbors(training_set, test_instance, k)) for test_instance in test_set]

    # Calculate accuracy
    accuracy = get_accuracy(test_set, predictions)
    print(f'Accuracy: {accuracy:.2f}%')

    # Confusion matrix
    cm = confusion_matrix(predictions, test_set)
    print(f'Confusion Matrix: {cm}')

    # Recall scores
    recall_male = cm['mm'] / (cm['mm'] + cm['mf']) if cm['mm'] + cm['mf'] > 0 else 0
    recall_female = cm['ff'] / (cm['ff'] + cm['fm']) if cm['ff'] + cm['fm'] > 0 else 0
    print(f'Recall for Male: {recall_male:.2f}')
    print(f'Recall for Female: {recall_female:.2f}')

    # F1 scores
    f1_score_male = (2 * accuracy * recall_male) / (accuracy + recall_male) if (accuracy + recall_male) > 0 else 0
    f1_score_female = (2 * accuracy * recall_female) / (accuracy + recall_female) if (accuracy + recall_female) > 0 else 0
    print(f'F1 Score for Male: {f1_score_male:.2f}')
    print(f'F1 Score for Female: {f1_score_female:.2f}')

if __name__ == "__main__":
    main()
