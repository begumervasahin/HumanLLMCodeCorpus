def softmax(x):
    """
    Compute the softmax vector.
    
    Parameters:
    x: numpy array, input vector
    
    Returns:
    numpy array, softmax values
    """
    c = np.max(x)
    ex = np.exp(x - c)
    softmax_vals = ex / np.sum(ex)
    return softmax_vals

def classifier_output(x, params):
    """
    Return the output layer (class probabilities) of a log-linear classifier.
    
    Parameters:
    x: numpy array, input vector
    params: tuple, parameters (W, b)
    
    Returns:
    numpy array, class probabilities
    """
    W, b = params
    z = np.dot(x, W) + b
    probs = softmax(z)
    return probs

def predict(x, params):
    """
    Return the prediction (highest scoring class id) of a log-linear classifier.
    
    Parameters:
    x: numpy array, input vector
    params: tuple, parameters (W, b)
    
    Returns:
    int, predicted class id
    """
    return np.argmax(classifier_output(x, params))

def loss_and_gradients(x, y, params):
    """
    Compute the loss and gradients at point x with given parameters.
    
    Parameters:
    x: numpy array, input vector
    y: int, correct label
    params: tuple, parameters (W, b)
    
    Returns:
    loss: float, loss value
    gradients: list, gradients [gW, gb]
    """
    W, b = params
    pred_vec = classifier_output(x, params)
    y_hat = pred_vec[y]
    loss = -np.log(y_hat)
    
    y_hot_vector = np.zeros_like(pred_vec)
    y_hot_vector[y] = 1
    gb = pred_vec - y_hot_vector
    gW = np.outer(x, gb)
    
    return loss, [gW, gb]

def create_classifier(in_dim, out_dim):
    """
    Create parameters for a log-linear classifier.
    
    Parameters:
    in_dim: int, input dimension
    out_dim: int, output dimension
    
    Returns:
    list, parameters [W, b]
    """
    W = np.zeros((in_dim, out_dim))
    b = np.zeros(out_dim)
    return [W, b]
