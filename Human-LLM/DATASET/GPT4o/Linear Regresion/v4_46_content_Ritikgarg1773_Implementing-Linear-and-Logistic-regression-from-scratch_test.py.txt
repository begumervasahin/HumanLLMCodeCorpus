from scratch import MyLinearRegression, MyLogisticRegression, MyPreProcessor
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd

# Initialize preprocessor
preprocessor = MyPreProcessor()

# Linear Regression
print('Linear Regression')
X, y = preprocessor.pre_process(0)
num_rows = int(X.shape[0] * 0.7)
X_train, X_test = X[:num_rows], X[num_rows:]
y_train, y_test = y[:num_rows], y[num_rows:]

linear = MyLinearRegression()
k = 5

# Split data into k folds for cross-validation
X_folds = np.array_split(X_train, k)
y_folds = np.array_split(y_train, k)

for i in range(k):
    # Create training and validation sets
    X_train_folds = X_folds[:i] + X_folds[i+1:]
    y_train_folds = y_folds[:i] + y_folds[i+1:]
    X_val_fold = X_folds[i]
    y_val_fold = y_folds[i]
    
    X_data_train = np.concatenate(X_train_folds)
    y_data_train = np.concatenate(y_train_folds)
    
    # Fit the model
    rmse_errors, mae_errors, rmse_coef, rmse_intercept, mae_coef, mae_intercept = linear.fit(X_data_train, y_data_train)
    
    # Predict and calculate validation loss
    y_pred = linear.predict(X_val_fold)
    rmse_val_loss, mae_val_loss = linear.validationLoss(X_val_fold, y_val_fold, rmse_coef, rmse_intercept, mae_coef, mae_intercept)
    
    # Plot RMSE Training and Validation Loss
    plt.plot(rmse_errors, color='red', linewidth=2, label='RMSE Training Loss')
    plt.plot(rmse_val_loss, color='orange', linewidth=2, label='RMSE Validation Loss')
    plt.xlabel("Number of Iterations")
    plt.ylabel("Loss")
    plt.legend()
    plt.show()
    
    # Plot MAE Training and Validation Loss
    plt.plot(mae_errors, color='red', linewidth=2, label='MAE Training Loss')
    plt.plot(mae_val_loss, color='blue', linewidth=2, label='MAE Validation Loss')
    plt.xlabel("Number of Iterations")
    plt.ylabel("Loss")
    plt.legend()
    plt.show()
    
    print(f'RMSE Error on Test Data: {rmse_val_loss[-1]}')
    print(f'MAE Error on Test Data: {mae_val_loss[-1]}')
