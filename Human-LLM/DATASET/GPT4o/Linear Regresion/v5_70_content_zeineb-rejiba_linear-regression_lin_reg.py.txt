from sklearn.preprocessing import StandardScaler
import numpy as np
import tensorflow as tf
from data_utils import load_training_data, prepare_data
from plot_utils import plot_scatter, plot_data_with_line, plot_surface_and_contour, plot_cost_per_lr

class LinearRegressionModel:
    def __init__(self, input_dim):
        self.theta = tf.Variable(tf.zeros([input_dim, 1]), dtype=tf.float32)

    def train(self, iterations, X, y, learning_rate):
        optimizer = tf.optimizers.Adam(learning_rate)
        for _ in range(iterations):
            with tf.GradientTape() as tape:
                cost = self._compute_cost(X, y)
            grads = tape.gradient(cost, [self.theta])
            optimizer.apply_gradients(zip(grads, [self.theta]))

    def _compute_cost(self, X, y):
        predictions = self.predict(X)
        cost = tf.reduce_mean(tf.square(predictions - y))
        return cost

    def predict(self, X):
        return tf.linalg.matmul(X, self.theta)

def main():
    # Step 2: Linear Regression with One Variable
    print('Step 2: Linear Regression with One Variable')

    # Step 2.1: Plotting the Data
    print('Step 2.1: Plotting the Data')
    data_path_single = 'data/ex1data1.txt'
    column_names_single = ['population', 'profit']
    df_single, X_single, y_single = load_training_data(data_path_single, column_names_single)
    plot_scatter(df_single)
    print()

    # Step 2.2: Gradient Descent
    print('Step 2.2: Gradient Descent')
    X_single, y_single = prepare_data(X_single.to_numpy(), y_single)
    model_single = LinearRegressionModel(X_single.shape[1])
    iterations_single = 1500
    learning_rate_single = 0.01
    print(f'Training model using {iterations_single} iterations and learning rate= {learning_rate_single:.2f}')
    model_single.train(iterations_single, X_single, y_single, learning_rate_single)

    # Predicting Profits for Population Sizes
    predict_population_profit(model_single, 3.5, '35,000')
    predict_population_profit(model_single, 7.0, '70,000')
    print()

    # Step 2.3: Debugging
    print('Step 2.3: Debugging')
    plot_data_with_line(df_single, model_single.theta)
    print()

    # Step 2.4: Visualizing Cost Function
    print('Step 2.4: Visualizing Cost Function')
    theta_single = model_single.theta.numpy()
    plot_surface_and_contour(theta_single, X_single, y_single)
    print()

    # Step 3: Linear Regression with Multiple Variables
    print('Step 3: Linear Regression with Multiple Variables')

    # Step 3.1: Feature Normalization
    print('Step 3.1: Feature Normalization')
    data_path_multi = 'data/ex1data2.txt'
    column_names_multi = ['size', 'bedrooms', 'price']
    df_multi, X_multi, y_multi = load_training_data(data_path_multi, column_names_multi)
    scaler = StandardScaler().fit(X_multi)
    X_multi_normalized = scaler.transform(X_multi)
    print()

    # Step 3.2: Gradient Descent
    print('Step 3.2: Gradient Descent')
    X_multi_normalized, y_multi = prepare_data(X_multi_normalized, y_multi)
    model_multi = LinearRegressionModel(X_multi_normalized.shape[1])
    iterations_multi = 500
    learning_rate_multi = 0.01
    print(f'Training model using {iterations_multi} iterations and learning rate= {learning_rate_multi:.2f}')
    model_multi.train(iterations_multi, X_multi_normalized, y_multi, learning_rate_multi)
    print()

    # Step 3.2.1: Selecting Learning Rates
    print('Step 3.2.1: Selecting Learning Rates')
    plot_cost_per_lr(X_multi_normalized, y_multi)
    model_multi = LinearRegressionModel(X_multi_normalized.shape[1])
    learning_rate_multi_high = 0.3
    print(f'Training model using {iterations_multi} iterations and learning rate= {learning_rate_multi_high:.2f}')
    model_multi.train(iterations_multi, X_multi_normalized, y_multi, learning_rate_multi_high)

    # Predicting Prices for a House with Given Features
    predict_house_price(model_multi, scaler, 1650, 3)
    print()

    # Step 3.3: Normal Equations
    print('Step 3.3: Normal Equations')
    theta_normal_eq = calculate_normal_equation(X_multi, y_multi)
    predict_price_normal_eq(theta_normal_eq, 1650, 3)

def predict_population_profit(model, population, population_size):
    example_population = tf.constant([[1.0, population]], dtype=tf.float32)
    predicted_profit = model.predict(example_population).numpy()[0][0]
    print(f'When the size of the population is {population_size} => The predicted profit is {predicted_profit * 10000:.2f}$')

def predict_house_price(model, scaler, size, bedrooms):
    example_house_features = np.array([size, bedrooms]).reshape(1, -1)
    example_house_normalized = scaler.transform(example_house_features)
    example_house_input = tf.constant(np.concatenate([np.ones((1, 1)), example_house_normalized], axis=1), dtype=tf.float32)
    predicted_price = model.predict(example_house_input).numpy()[0][0]
    print(f'When the house has an area of {size} square feet and {bedrooms} bedrooms => Its predicted price is {predicted_price:.2f}$')

def calculate_normal_equation(X, y):
    X = np.concatenate([np.ones((X.shape[0], 1)), X], axis=1)
    theta_normal_eq = np.linalg.inv(X.T @ X) @ X.T @ y
    return theta_normal_eq

def predict_price_normal_eq(theta, size, bedrooms):
    predicted_price_normal_eq = np.array([1, size, bedrooms]).reshape(1, -1) @ theta
    print(f'When the house has an area of {size} square feet and {bedrooms} bedrooms => Its predicted price is {predicted_price_normal_eq[0]:.2f}$')

if __name__ == "__main__":
    main()
