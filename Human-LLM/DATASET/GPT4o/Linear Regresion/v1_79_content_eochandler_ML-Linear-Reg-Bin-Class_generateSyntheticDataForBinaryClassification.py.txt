import numpy as np
import matplotlib.pyplot as plt
from sklearn.linear_model import LogisticRegression

def plot_line(beta, x_min, x_max, y_min, y_max):
    """Plot a decision boundary line based on logistic regression coefficients."""
    x_vals = np.linspace(x_min, x_max, 100)
    y_vals = (-beta[1] * x_vals - beta[0]) / beta[2]
    idxs = np.where((y_vals >= y_min) & (y_vals <= y_max))
    plt.plot(x_vals[idxs], y_vals[idxs])

def generate_data(num_samples, mean, cov):
    """Generate multivariate normal data."""
    return np.random.multivariate_normal(mean, cov, num_samples)

def cross_entropy(p, q):
    """Calculate the cross-entropy loss."""
    return -p * np.log(q) - (1 - p) * np.log(1 - q)

def sigmoid(u):
    """Compute the sigmoid function."""
    return np.exp(u) / (1 + np.exp(u))

def h(u, y):
    """Calculate the logistic regression hypothesis."""
    exp = np.exp(u)
    return -y * u + np.log(1 + exp)

def loss(beta, X, Y):
    """Compute the logistic regression loss function."""
    N = X.shape[0]
    total_loss = 0
    for i in range(N):
        dot_product = np.vdot(X[i], beta)
        total_loss += h(dot_product, Y[i])
    return total_loss

def logistic_regression(X, Y, alpha=0.001, max_iterations=1000):
    """Perform logistic regression using gradient descent."""
    N, d = X.shape
    X = np.hstack((np.ones((N, 1)), X))  # Add intercept term
    beta = np.random.randn(d + 1)
    norm_gradients = []
    losses = []

    for _ in range(max_iterations):
        gradient = np.zeros(d + 1)
        for i in range(N):
            dot_product = np.vdot(X[i], beta)
            qi = sigmoid(dot_product)
            gradient += (qi - Y[i]) * X[i]
        norm_gradient = np.linalg.norm(gradient)
        norm_gradients.append(norm_gradient)
        beta -= alpha * gradient
        losses.append(loss(beta, X, Y))

    return beta, norm_gradients, losses

# Parameters
num_samples = 100
np.random.seed(7)
mu_pos = [1.0, 1.0]
cov_pos = np.array([[1.0, 0.0], [0.0, 1.0]])
mu_neg = [-1.0, -1.0]
cov_neg = np.array([[1.0, 0.0], [0.0, 1.0]])

# Generate data
X_pos = generate_data(num_samples, mu_pos, cov_pos)
X_neg = generate_data(num_samples, mu_neg, cov_neg)
X = np.concatenate((X_pos, X_neg))
Y = np.concatenate((np.ones(num_samples), np.zeros(num_samples)))

# Logistic Regression using Gradient Descent
beta, norm_gradients, losses = logistic_regression(X, Y)

# Plotting
plt.figure()
plt.semilogy(norm_gradients, label='Norm of Gradients')
plt.semilogy(losses, label='Loss')
plt.legend()
plt.title('Gradient Norm and Loss during Training')
plt.xlabel('Iteration')
plt.ylabel('Value')
plt.show()

plt.figure()
plt.scatter(X_pos[:, 0], X_pos[:, 1], label='Positive Class')
plt.scatter(X_neg[:, 0], X_neg[:, 1], label='Negative Class')
plot_line(beta, X[:, 0].min(), X[:, 0].max(), X[:, 1].min(), X[:, 1].max())
plt.title('Logistic Regression Decision Boundary')
plt.xlabel('X1')
plt.ylabel('X2')
plt.legend()
plt.axis('equal')
plt.show()

# Logistic Regression using Scikit-Learn
model = LogisticRegression(solver='lbfgs', penalty='none')
model.fit(X, Y)
bias = model.intercept_
weights = model.coef_[0]
beta_sk = [bias[0], weights[0], weights[1]]

plt.figure()
plt.scatter(X_pos[:, 0], X_pos[:, 1], label='Positive Class')
plt.scatter(X_neg[:, 0], X_neg[:, 1], label='Negative Class')
plot_line(beta_sk, X[:, 0].min(), X[:, 0].max(), X[:, 1].min(), X[:, 1].max())
plt.title('Logistic Regression Decision Boundary (Scikit-Learn)')
plt.xlabel('X1')
plt.ylabel('X2')
plt.legend()
plt.axis('equal')
plt.show()

print("Beta (Gradient Descent):", beta)
print("Beta (Scikit-Learn):", beta_sk)
