import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.metrics import mean_squared_error
from math import sqrt
import statsmodels.formula.api as smf

# Load the dataset
data_path = 'C:/Users/HP/Desktop/assignments submission/simple linear regression/delivery_time.csv'
delivery_time_data = pd.read_csv(data_path)

# Function to plot scatter plot of Sorting Time vs Delivery Time
def plot_scatter(x, y, title, xlabel, ylabel):
    plt.scatter(x, y)
    plt.xlabel(xlabel)
    plt.ylabel(ylabel)
    plt.title(title)
    plt.show()

# Function to print model summary and parameters
def print_model_summary(model):
    print(f"Model Parameters:\n{model.params}")
    print(f"Model Summary:\n{model.summary()}")

# Function to calculate and print RMSE
def calculate_rmse(actual, predicted, model_name):
    rmse = sqrt(mean_squared_error(actual, predicted))
    print(f'{model_name} RMSE: {rmse}')
    return rmse

# Function to plot regression results
def plot_regression_results(x, actual, predicted, title, xlabel, ylabel):
    plt.scatter(x, actual, c='r')
    plt.plot(x, predicted, c='b')
    plt.xlabel(xlabel)
    plt.ylabel(ylabel)
    plt.title(title)
    plt.show()

# Plot initial scatter plot
plot_scatter(delivery_time_data.Sorting_time, delivery_time_data.Delivery_time,
             'Scatter Plot of Sorting Time vs Delivery Time', 'Sorting Time', 'Delivery Time')

# Calculate and print the correlation coefficient
correlation_matrix = np.corrcoef(delivery_time_data.Sorting_time, delivery_time_data.Delivery_time)
print(f'Correlation Coefficient: {correlation_matrix[0, 1]}')

# Linear Regression Model
linear_model = smf.ols('Delivery_time ~ Sorting_time', data=delivery_time_data).fit()
print_model_summary(linear_model)
linear_predictions = linear_model.predict(delivery_time_data)
plot_regression_results(delivery_time_data.Sorting_time, delivery_time_data.Delivery_time, linear_predictions,
                        'Linear Regression: Sorting Time vs Delivery Time', 'Sorting Time', 'Delivery Time')
print(f"Linear Model Confidence Interval:\n{linear_model.conf_int(0.05)}")
calculate_rmse(delivery_time_data.Delivery_time, linear_predictions, 'Linear Model')

# Logarithmic Model
log_model = smf.ols('Delivery_time ~ np.log(Sorting_time)', data=delivery_time_data).fit()
print_model_summary(log_model)
log_predictions = log_model.predict(delivery_time_data)
plot_regression_results(delivery_time_data.Sorting_time, delivery_time_data.Delivery_time, log_predictions,
                        'Logarithmic Regression: Sorting Time vs Delivery Time', 'Sorting Time', 'Delivery Time')
print(f"Logarithmic Model Confidence Interval:\n{log_model.conf_int(0.05)}")
calculate_rmse(delivery_time_data.Delivery_time, log_predictions, 'Logarithmic Model')

# Exponential Model
exp_model = smf.ols('np.log(Delivery_time) ~ Sorting_time', data=delivery_time_data).fit()
print_model_summary(exp_model)
exp_predictions_log = exp_model.predict(delivery_time_data)
exp_predictions = np.exp(exp_predictions_log)
plot_regression_results(delivery_time_data.Sorting_time, delivery_time_data.Delivery_time, exp_predictions,
                        'Exponential Regression: Sorting Time vs Delivery Time', 'Sorting Time', 'Delivery Time')
print(f"Exponential Model Confidence Interval:\n{exp_model.conf_int(0.05)}")
calculate_rmse(delivery_time_data.Delivery_time, exp_predictions, 'Exponential Model')

# Quadratic Model
delivery_time_data['sq_sorting_time'] = delivery_time_data.Sorting_time ** 2
quad_model = smf.ols('Delivery_time ~ Sorting_time + sq_sorting_time', data=delivery_time_data).fit()
print_model_summary(quad_model)
quad_predictions = quad_model.predict(delivery_time_data)
plot_regression_results(delivery_time_data.Sorting_time, delivery_time_data.Delivery_time, quad_predictions,
                        'Quadratic Regression: Sorting Time vs Delivery Time', 'Sorting Time', 'Delivery Time')
print(f"Quadratic Model Confidence Interval:\n{quad_model.conf_int(0.05)}")
calculate_rmse(delivery_time_data.Delivery_time, quad_predictions, 'Quadratic Model')
