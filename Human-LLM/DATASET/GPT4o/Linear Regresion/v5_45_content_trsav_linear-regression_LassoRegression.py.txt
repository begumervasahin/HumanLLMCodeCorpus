import numpy as np
from scipy.optimize import minimize, NonlinearConstraint
from tqdm import tqdm
import matplotlib.pyplot as plt
import pandas as pd

# Define the loss function for training
def loss(B, data_train, response_train):
    B0 = B[0]
    B = B[1:]
    error = np.sum((response_train - B0 - np.dot(data_train, B)) ** 2)
    return error

# Define the loss function for testing
def loss_test(B, test_data, test_response):
    B0 = B[0]
    Bj = B[1:]
    error = np.sum((test_response - B0 - np.dot(test_data, Bj)) ** 2)
    return error

# Define the constraint function
def constraint_fun(B):
    Bj = B[1:]
    return np.dot(Bj, Bj)

# Function to split the data into k folds and return the ith fold
def fold(i, k, data):
    percen = 1 / k
    fold_amount = int(percen * len(data))
    start_in = fold_amount * i
    end_in = start_in + fold_amount
    data_test = data[start_in:end_in]
    data_train = np.concatenate((data[:start_in], data[end_in:]))
    return data_train, data_test

# Load and preprocess the data
data = pd.read_excel('/Users/tomsavage/Dropbox/Documents/Projects/ElementsOfStatisticalLearning/prostate_data.xlsx').to_numpy()

# Normalize the data
data = (data - np.mean(data, axis=0)) / np.std(data, axis=0)

# Data augmentation
data_aug1 = data * np.random.normal(1, 0.05, data.shape)
data_aug2 = data * np.random.normal(1, 0.05, data.shape)
data = np.concatenate((data, data_aug1, data_aug2), axis=0)

# Shuffle the data
np.random.shuffle(data)

# Load the column labels
labels = pd.read_excel('/Users/tomsavage/Dropbox/Documents/Projects/ElementsOfStatisticalLearning/prostate_data.xlsx', nrows=0).columns

# Set up parameters
t_limit = 0.7
folds = 4
t_fidel = 20
t_vals = np.linspace(0, t_limit, t_fidel)
B_init = np.zeros(len(data[0]))

# Perform cross-validation
bias = np.zeros((folds, t_fidel))
var = np.zeros((folds, t_fidel))

for j in range(folds):
    data_train_full, data_test_full = fold(j, folds, data)
    test_data = data_test_full[:, :-1]
    test_response = data_test_full[:, -1]
    data_train = data_train_full[:, :-1]
    response_train = data_train_full[:, -1]
    
    for i in tqdm(range(t_fidel)):
        t = t_vals[i]
        constraint = NonlinearConstraint(constraint_fun, 0, t)
        sol = minimize(loss, B_init, args=(data_train, response_train), constraints=constraint, method='SLSQP')
        B_opt = sol.x
        bias[j, i] = sol.fun
        var[j, i] = loss_test(B_opt, test_data, test_response)

# Calculate mean bias and variance
bias_mean = np.mean(bias, axis=0) / len(data_train)
var_mean = np.mean(var, axis=0) / len(test_data)

# Plot bias and variance
plt.figure()
plt.plot(t_vals, bias_mean, label='Train')
plt.plot(t_vals, var_mean, label='Test')
plt.xlabel(r'$t$ where, $\sum _j| \beta _j| \leq t$')
plt.ylabel('MSE')
plt.legend()
plt.grid()
plt.show()

# Store beta coefficients
data = data[:, :-1]
response = data[:, -1]
B_store = np.zeros((t_fidel, len(data[0]) + 1))

for i in tqdm(range(t_fidel)):
    t = t_vals[i]
    constraint = NonlinearConstraint(constraint_fun, 0, t)
    sol = minimize(loss, B_init, args=(data_train, response_train), constraints=constraint, method='SLSQP')
    B_store[i] = sol.x

# Plot beta coefficients
plt.figure()
for i in range(1, len(B_init)):
    plt.plot(t_vals, B_store[:, i])
    plt.text(t_vals[-1], B_store[-1, i], labels[i-1])
plt.grid()
plt.ylabel(r'$\beta$')
plt.xlabel(r'$t$ where, $\sum _j| \beta _j| \leq t$')
plt.show()
