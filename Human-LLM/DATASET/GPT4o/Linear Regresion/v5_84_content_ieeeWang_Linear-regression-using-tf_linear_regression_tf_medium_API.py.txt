import tensorflow as tf
import numpy as np
from matplotlib import pyplot as plt
from tensorflow.keras import layers, losses, optimizers

# Function to print a timestamp bar
@tf.function
def print_timestamp_bar():
    """Prints a timestamp bar to indicate progress."""
    current_ts = tf.timestamp() % (24 * 60 * 60)
    hour = tf.cast(current_ts // 3600 + 2, tf.int32) % tf.constant(24)
    minute = tf.cast((current_ts % 3600) // 60, tf.int32)
    second = tf.cast(tf.floor(current_ts % 60), tf.int32)

    def format_time(unit):
        return tf.strings.format("0{}", unit) if tf.strings.length(tf.strings.format("{}", unit)) == 1 else tf.strings.format("{}", unit)

    time_string = tf.strings.join([format_time(hour), format_time(minute), format_time(second)], separator=":")
    tf.print("=" * 80 + time_string)

# Generate synthetic dataset
n_samples = 400
X = tf.random.uniform([n_samples, 2], minval=-10, maxval=10)
X_np = X.numpy()

# True weights and bias
true_weights = tf.constant([[2.0], [-3.0]])
true_bias = tf.constant([[3.0]])

# Print true weights and bias
print(true_weights)
tf.print(tf.rank(true_weights))
print(true_bias)
tf.print(tf.rank(true_bias))

# Generate labels with some noise
Y = X @ true_weights + true_bias + tf.random.normal([n_samples, 1], mean=0.0, stddev=2.0)
Y_np = Y.numpy()

# Plot the generated dataset
plt.figure(figsize=(12, 5))
plt.subplot(121)
plt.scatter(X[:, 0], Y[:, 0], c="b")
plt.xlabel("x1")
plt.ylabel("y", rotation=0)
plt.subplot(122)
plt.scatter(X[:, 1], Y[:, 0], c="g")
plt.xlabel("x2")
plt.ylabel("y", rotation=0)
plt.show()

# Create a TensorFlow dataset
dataset = tf.data.Dataset.from_tensor_slices((X, Y)).shuffle(buffer_size=100).batch(10).prefetch(tf.data.experimental.AUTOTUNE)

# Define the model
model = layers.Dense(units=1)
model.build(input_shape=(2,))
model.loss_function = losses.mean_squared_error
model.optimizer = optimizers.SGD(learning_rate=0.001)

# Training step function
@tf.function
def train_step(model, features, labels):
    """Performs a single training step."""
    with tf.GradientTape() as tape:
        predictions = model(features)
        loss = model.loss_function(tf.reshape(labels, [-1]), tf.reshape(predictions, [-1]))
    gradients = tape.gradient(loss, model.variables)
    model.optimizer.apply_gradients(zip(gradients, model.variables))
    return loss

# Example of a single training step
features, labels = next(iter(dataset))
loss = train_step(model, features, labels)
print(loss)

# Function to train the model
def train_model(model, epochs):
    """Trains the model for a specified number of epochs."""
    for epoch in tf.range(1, epochs + 1):
        epoch_loss = tf.constant(0.0)
        for features, labels in dataset:
            epoch_loss = train_step(model, features, labels)
        if epoch % 50 == 0:
            print_timestamp_bar()
            tf.print("Epoch =", epoch, "Loss =", epoch_loss)
            tf.print("Weights =", model.variables[0])
            tf.print("Bias =", model.variables[1])

# Train the model
train_model(model, epochs=200)

# Get the trained weights and bias
trained_weights, trained_bias = model.variables
tf.print("Trained Weights =", trained_weights)
tf.print("Trained Bias =", trained_bias)

# Plot the fitted model against the data
plt.figure(figsize=(12, 5))
plt.subplot(121)
plt.scatter(X[:, 0], Y[:, 0], c="b", label="Samples")
plt.plot(X[:, 0], trained_weights[0] * X[:, 0] + trained_bias[0], "-r", linewidth=5.0, label="Model")
plt.legend()
plt.xlabel("x1")
plt.ylabel("y", rotation=0)
plt.subplot(122)
plt.scatter(X[:, 1], Y[:, 0], c="g", label="Samples")
plt.plot(X[:, 1], trained_weights[1] * X[:, 1] + trained_bias[0], "-r", linewidth=5.0, label="Model")
plt.legend()
plt.xlabel("x2")
plt.ylabel("y", rotation=0)
plt.show()
