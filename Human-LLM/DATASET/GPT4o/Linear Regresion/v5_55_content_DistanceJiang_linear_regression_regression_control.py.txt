import numpy as np
import pypcd
from linear_regressor import SegmentedLinearRegressor
from points_divider import PointsDivider, ContinuousPart
from utils import filter_points, get_intersection

class RegressionController:
    def __init__(self, path=None, verbose=False):
        """
        Initialize the RegressionController.
        
        :param path: str, optional
            Path to the PCD file.
        :param verbose: bool, optional
            If True, print more information to the console.
        """
        self.points = self._load_points(path) if path else []
        self.verbose = verbose
        self.parameters = []
        self.parts = []
        self.intersections = []

    def _load_points(self, path):
        """Load points from a PCD file."""
        pc_data = pypcd.PointCloud.from_path(path).pc_data
        return [[point['x'], point['y']] for point in pc_data]

    def clear_state(self):
        """Clear the internal state of the controller."""
        self.parameters = []
        self.parts = []
        self.intersections = []

    def get_points(self):
        """Get the list of points."""
        return self.points

    def set_points(self, points):
        """Set the list of points."""
        self.points = points

    def fit(self, regressor):
        """
        Use a linear regressor to fit the points.

        :param regressor: SegmentedLinearRegressor
            An instance of the regressor.
        :raises Exception: If parts are not set before fitting.
        """
        if not self.parts:
            raise Exception("Nothing to fit, check if set_parts() method is called.")

        for count, part in enumerate(self.parts, start=1):
            if part.isolated:
                self.intersections.append(None)
                self.parameters.append(None)
                continue

            if self.verbose:
                print(f"\n\n************************ Fitting part {count} ************************")

            self.intersections.append(regressor.process(part.points))
            self.parameters.append(regressor.get_parameters())

    def set_parts(self, divider, resolution=1):
        """
        Divide all points into parts for fitting.

        :param divider: PointsDivider
            An instance of PointsDivider to divide points into parts.
        :param resolution: int, optional
            Resolution for dividing the points.
        """
        divider.set_points(self.points)
        divider.set_blocks(resolution)
        self.parts = divider.divide()

    def get_intersections(self):
        """Get the list of intersections."""
        return self.intersections

class SimpleRegressionController(RegressionController):
    def set_parts(self):
        """Divide points into predefined parts for simple regression."""
        limits = [
            [-2, -1, float('-inf'), float('inf')],
            [4, 5, float('-inf'), float('inf')],
            [float('-inf'), float('inf'), -1, -0.2],
            [float('-inf'), float('inf'), 3, 4]
        ]

        for limit in limits:
            points = filter_points(self.points, *limit)
            self.parts.append(ContinuousPart(points))

    def get_intersections(self):
        """Calculate intersections based on fitted parameters."""
        parameters = [param[0] for param in self.parameters]

        points = [
            get_intersection(parameters[0], parameters[2]),
            get_intersection(parameters[0], parameters[3]),
            get_intersection(parameters[1], parameters[2]),
            get_intersection(parameters[1], parameters[3])
        ]

        intersections = [
            [[points[0][0], points[0][1] + 2], points[1]],
            [points[0], points[2]],
            [points[2], points[3]],
            [points[3], points[1]]
        ]

        return intersections
