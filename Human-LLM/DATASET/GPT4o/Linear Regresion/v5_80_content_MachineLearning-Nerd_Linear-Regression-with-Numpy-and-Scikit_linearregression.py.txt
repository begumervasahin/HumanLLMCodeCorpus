import matplotlib.pyplot as plt
import numpy as np
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import PolynomialFeatures
from mpl_toolkits.mplot3d import Axes3D

def extract_data(input_file='./input03.txt', output_file='./output03.txt'):
    """
    Extracts training and testing data from input files.
    
    Args:
        input_file (str): Path to the input file containing training data.
        output_file (str): Path to the output file containing test data.
    
    Returns:
        tuple: Arrays of training features, training labels, test features, and test labels.
    """
    with open(input_file, 'r') as f:
        lines = list(map(int, f.readline().split()))
        num_train = lines[1]
        
        X_train, Y_train = [], []
        for _ in range(num_train):
            train_data = list(map(float, f.readline().strip().split()))
            X_train.append(train_data[:2])
            Y_train.append(train_data[2])
        
        num_test = int(f.readline())
        X_test = [list(map(float, f.readline().strip().split())) for _ in range(num_test)]
    
    with open(output_file, 'r') as f:
        Y_test = [float(f.readline().strip().split()[0]) for _ in range(num_test)]
    
    return np.array(X_train), np.array(Y_train), np.array(X_test), np.array(Y_test)

def linear_regression_numpy(X_train, X_test, Y_train):
    """
    Performs linear regression using NumPy.
    
    Args:
        X_train (np.ndarray): Training features.
        X_test (np.ndarray): Testing features.
        Y_train (np.ndarray): Training labels.
    
    Returns:
        np.ndarray: Predicted values for the test set.
    """
    x_train = np.column_stack((np.ones(X_train.shape[0]), X_train))
    x_test = np.column_stack((np.ones(X_test.shape[0]), X_test))
    
    coef = np.linalg.pinv(x_train.T @ x_train) @ x_train.T @ Y_train
    Y_pred = coef @ x_test.T
    
    return Y_pred

def linear_regression_sklearn(X_train, Y_train, X_test):
    """
    Performs linear regression using scikit-learn.
    
    Args:
        X_train (np.ndarray): Training features.
        Y_train (np.ndarray): Training labels.
        X_test (np.ndarray): Testing features.
    
    Returns:
        np.ndarray: Predicted values for the test set.
    """
    model = LinearRegression()
    model.fit(X_train, Y_train)
    return model.predict(X_test)

def plot_3d_scatter(X_train, Y_train, Y_pred, title):
    """
    Plots a 3D scatter plot of the training data and predictions.
    
    Args:
        X_train (np.ndarray): Training features.
        Y_train (np.ndarray): Training labels.
        Y_pred (np.ndarray): Predicted values.
        title (str): Title of the plot.
    """
    fig = plt.figure()
    ax = fig.add_subplot(111, projection='3d')
    ax.scatter(X_train[:, 0], X_train[:, 1], Y_train, c='b', marker='x', label='X_train vs Y_train', s=10)
    ax.scatter(X_train[:, 0], X_train[:, 1], Y_pred, c='r', marker='s', label='X_train vs Y_pred', s=10)
    plt.legend(loc='lower right')
    plt.title(title)
    plt.show()

def main():
    X_train, Y_train, X_test, Y_test = extract_data()

    print("NumPy Linear Regression Predictions:", linear_regression_numpy(X_train, X_test, Y_train))
    print("scikit-learn Linear Regression Predictions:", linear_regression_sklearn(X_train, Y_train, X_test))

    Y_train_pred = linear_regression_sklearn(X_train, Y_train, X_train)
    plot_3d_scatter(X_train, Y_train, Y_train_pred, "Without Polynomial Features")

    poly = PolynomialFeatures(degree=3)
    X_poly_train = poly.fit_transform(X_train)
    X_poly_test = poly.fit_transform(X_test)

    print("NumPy Polynomial Regression Predictions:", linear_regression_numpy(X_poly_train, X_poly_test, Y_train))
    print("scikit-learn Polynomial Regression Predictions:", linear_regression_sklearn(X_poly_train, Y_train, X_poly_test))

    Y_train_poly_pred = linear_regression_sklearn(X_poly_train, Y_train, X_poly_train)
    plot_3d_scatter(X_train, Y_train, Y_train_poly_pred, "With Polynomial Features")

if __name__ == "__main__":
    main()
