import numpy as np
import pandas as pd

from sklearn.metrics import mean_absolute_error as MAE

# Setting a seed for reproducibility
np.random.seed(999)

def load_wine_data():
    """Loads and preprocesses the wine dataset."""
    wine = load_wine()
    df_wine = pd.DataFrame(data=wine.data, columns=wine.feature_names)
    df_wine['target'] = wine.target
    df_wine = df_wine[df_wine['target'] < 2]  # Filter classes to binary
    X = df_wine.drop('target', axis=1).values
    y = df_wine['target'].values.reshape(-1, 1)
    return X, y
