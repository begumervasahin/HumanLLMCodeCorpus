import numpy as np
import csv
import matplotlib.pyplot as plt

# Function to compute the loss (cost function)
def compute_loss(inputs, outputs, theta):
    m = len(inputs[0])
    total_loss = 0
    for i in range(m):
        predicted = theta[0] * inputs[0][i] + theta[1] * inputs[1][i] + theta[2] * inputs[2][i] + theta[3] * inputs[3][i]
        actual = outputs[i]
        total_loss += (predicted - actual) ** 2
    return total_loss / (2 * m)

# Function to compute the gradient of the loss
def compute_gradient(inputs, outputs, theta, alpha):
    m = len(inputs[0])
    gradient = [0, 0, 0, 0]
    for i in range(m):
        predicted = theta[0] + theta[1] * inputs[1][i] + theta[2] * inputs[2][i] + theta[3] * inputs[3][i]
        actual = outputs[i]
        for j in range(4):
            gradient[j] += alpha * (actual - predicted) * inputs[j][i] / m
    return gradient

# Function to standardize features
def standardize(data):
    mean = np.mean(data)
    std = np.std(data)
    return (data - mean) / std

def standardize_single(value, data):
    mean = np.mean(data)
    std = np.std(data)
    return (value - mean) / std

# Read data from CSV
def read_data(file_path):
    x1, x2, x3, y = [], [], [], []
    with open(file_path, newline='') as file:
        reader = csv.reader(file)
        for row in reader:
            x1.append(float(row[0]))
            x2.append(float(row[1]))
            x3.append(float(row[2]))
            y.append(float(row[3]))
    return x1, x2, x3, y

# Display first few rows of data
def display_initial_data(x1, x2, x3, y, rows=5):
    print("--------------------------------------------------------------------------------------------")
    print("(1) First 5 rows of raw input data are:")
    for i in range(rows):
        print(f"{x1[i]}\t{x2[i]}\t{x3[i]}\t{y[i]}")

# Standardize data
def standardize_data(x1, x2, x3):
    x1s = standardize(x1)
    x2s = standardize(x2)
    x3s = standardize(x3)
    standardized_data = [[1.0] * len(x1s), x1s, x2s, x3s]
    return standardized_data

# Display standardized data
def display_standardized_data(x1s, x2s, x3s, y, rows=5):
    print("--------------------------------------------------------------------------------------------")
    print("(2) First 5 rows of standardized data are:")
    for i in range(rows):
        print(f"1\t{x1s[i]}\t{x2s[i]}\t{x3s[i]}\t{y[i]}")

# Gradient descent for parameter optimization
def gradient_descent(xs, y, iterations, alpha):
    theta = [0, 0, 0, 0]
    loss_history = []
    print("--------------------------------------------------------------------------------------------")
    print("Theta \t\t\t\t\t\t\t\t Loss Function (J)")
    print("--------------------------------------------------------------------------------------------")
    for iteration in range(iterations):
        gradients = compute_gradient(xs, y, theta, alpha)
        for j in range(4):
            theta[j] += gradients[j]
        loss = compute_loss(xs, y, theta)
        loss_history.append(loss)
        theta_rounded = [round(t, 5) for t in theta]
        loss_rounded = round(loss, 5)
        print(f"{theta_rounded} \t\t {loss_rounded}")
    return theta, loss_history

# Predict house price using the optimized parameters
def predict_price(theta, n_bed, liv_area, lot_area, x1, x2, x3):
    xtests = [1, standardize_single(n_bed, x1), standardize_single(liv_area, x2), standardize_single(lot_area, x3)]
    price = round((theta[0] + theta[1] * xtests[1] + theta[2] * xtests[2] + theta[3] * xtests[3]), 2)
    return price

# Plot the loss function over iterations
def plot_loss(iterations, loss_history):
    plt.title("J vs n")
    plt.plot(range(1, iterations + 1), loss_history)
    plt.xlabel('Iterations')
    plt.ylabel('Loss value (J)')
    plt.grid(axis='both')
    plt.show()

# Main function to execute the program
def main():
    file_path = 'data/KCSmall_NS2.csv'
    x1, x2, x3, y = read_data(file_path)
    
    display_initial_data(x1, x2, x3, y)
    
    xs = standardize_data(x1, x2, x3)
    
    display_standardized_data(xs[1], xs[2], xs[3], y)
    
    initial_theta = [0, 0, 0, 0]
    print(f"The cost (J) value for theta = {initial_theta} is: J = {compute_loss(xs, y, initial_theta)}")
    
    iterations = 50
    alpha = 1.5
    theta, loss_history = gradient_descent(xs, y, iterations, alpha)
    
    n_bed = 3
    liv_area = 2000
    lot_area = 8550
    price = predict_price(theta, n_bed, liv_area, lot_area, x1, x2, x3)
    print(f"Approximate house price for [{n_bed}, {liv_area}, {lot_area}] is: ${price}")
    
    plot_loss(iterations, loss_history)

if __name__ == "__main__":
    main()
