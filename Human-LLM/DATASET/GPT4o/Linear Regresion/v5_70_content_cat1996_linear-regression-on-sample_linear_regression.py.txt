import pandas as pd
import matplotlib.pyplot as plt
import numpy as np
import seaborn as sns
import random
import csv

def load_dataset(filename, split_ratio, training=None, test=None):
    """
    Load dataset from a CSV file and split it into training and test sets.

    Parameters:
    filename (str): The path to the CSV file.
    split_ratio (float): The ratio to split the data into training and test sets.
    training (list): List to store training data.
    test (list): List to store test data.
    """
    if training is None:
        training = []
    if test is None:
        test = []

    with open(filename, 'r') as csvfile:
        lines = csv.reader(csvfile)
        dataset = list(lines)
        for x in range(1, len(dataset) - 1):
            for y in range(1, 3):
                dataset[x][y] = float(dataset[x][y])
                if random.random() < split_ratio:
                    training.append(dataset[x])
                else:
                    test.append(dataset[x])
    return training, test

def linear_model(X, w):
    """
    Compute the linear model.

    Parameters:
    X (array-like): Input features.
    w (list): Model parameters.

    Returns:
    array-like: Computed values.
    """
    return w[1] * np.array(X[:, 0]) + w[0]

def cost_function(w, X, y):
    """
    Compute the cost function.

    Parameters:
    w (list): Model parameters.
    X (array-like): Input features.
    y (array-like): Target values.

    Returns:
    float: Computed cost.
    """
    m = len(y)
    return (0.5 / m) * np.sum(np.square(linear_model(X, w) - np.array(y)))

def gradient(w, X, y):
    """
    Compute the gradient.

    Parameters:
    w (list): Model parameters.
    X (array-like): Input features.
    y (array-like): Target values.

    Returns:
    list: Computed gradient.
    """
    m = len(y)
    grad_0 = (1 / m) * np.sum(linear_model(X, w) - np.array(y))
    grad_1 = (1 / m) * np.sum((linear_model(X, w) - np.array(y)) * np.array(X[:, 0]))
    return [grad_0, grad_1]

def gradient_descent(w_init, learning_rate, X, y, tolerance=1e-6, max_iterations=500):
    """
    Perform gradient descent.

    Parameters:
    w_init (list): Initial model parameters.
    learning_rate (float): Learning rate for gradient descent.
    X (array-like): Input features.
    y (array-like): Target values.
    tolerance (float): Tolerance for stopping criterion.
    max_iterations (int): Maximum number of iterations.

    Returns:
    list: Optimized model parameters.
    """
    w_prev = w_init
    for iteration in range(max_iterations):
        grad_values = gradient(w_prev, X, y)
        w_new = [w_prev[0] - learning_rate * grad_values[0],
                 w_prev[1] - learning_rate * grad_values[1]]

        cost_diff = (w_new[0] - w_prev[0])**2 + (w_new[1] - w_prev[1])**2
        if cost_diff < tolerance:
            break

        w_prev = w_new

    return w_new

def plot_regression_line(formula, x_range):
    """
    Plot the regression line.

    Parameters:
    formula (function): The regression formula.
    x_range (array-like): Range of x values.
    """
    x = np.array(x_range)
    y = formula(x)
    plt.plot(x, y)

def regression_formula(x, w):
    """
    Compute the regression formula.

    Parameters:
    x (float): Input feature.
    w (list): Model parameters.

    Returns:
    float: Computed value.
    """
    return w[0] + w[1] * x

def calculate_r2_score(test_weights, test_heights, w):
    """
    Calculate the R^2 score.

    Parameters:
    test_weights (array-like): Test set weights.
    test_heights (array-like): Test set heights.
    w (list): Model parameters.

    Returns:
    float: R^2 score.
    """
    mean_height = np.mean(test_heights)
    ss_total = np.sum((test_heights - mean_height) ** 2)
    ss_residual = np.sum((test_heights - regression_formula(np.array(test_weights), w)) ** 2)
    return 1 - (ss_residual / ss_total)

training, test = load_dataset("weight-height.csv", 0.7)

train_weights = [row[1] for row in training]

# Prepare training data for linear regression
X = np.array(train_weights).reshape(-1, 1)
y = np.array(train_heights)

plt.scatter(train_weights, train_heights)
plt.xlabel('Weight')
plt.ylabel('Height')
plt.show()

# Calculate initial coefficients using ordinary least squares
mean_height = np.mean(train_heights)
mean_weight = np.mean(train_weights)
numerator = np.sum((train_weights - mean_weight) * (train_heights - mean_height))
denominator = np.sum((train_weights - mean_weight) ** 2)
b1 = numerator / denominator
b0 = mean_height - b1 * mean_weight
initial_weights = [b0, b1]

print(f"Initial coefficients: b1={b1}, b0={b0}")

# Perform gradient descent to optimize coefficients
optimized_weights = gradient_descent(initial_weights, 0.001, X, y)

print(f"Optimized coefficients: {optimized_weights}")

# Plotting the linear regression line
x_values = np.linspace(min(train_weights), max(train_weights), 20000)
plt.scatter(X, y, c="red", alpha=0.5, marker='o')
plot_regression_line(lambda x: regression_formula(x, optimized_weights), x_values)
plt.xlabel('Weight')
plt.ylabel('Height')
plt.show()

# Calculate and print R^2 score
r2_score = calculate_r2_score(test_weights, test_heights, optimized_weights)
print(f'The R^2 score is {r2_score}')
