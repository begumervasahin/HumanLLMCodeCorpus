import math
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.metrics import confusion_matrix, accuracy_score, classification_report

def cost_func(theta, X, y):
    """Calculate the cost function for logistic regression."""
    z = np.dot(X, theta)
    hyp = 1.0 / (1.0 + np.exp(-z))
    total_cost = np.sum(-(y * np.log(hyp)) - ((1 - y) * np.log(1 - hyp)))
    cost = total_cost / len(X)
    return cost

def gradient_descent(theta, X, y, alpha=0.001, num_iterations=400):
    """Perform gradient descent to optimize theta."""
    for _ in range(num_iterations):
        z = np.dot(X, theta)
        hyp = 1.0 / (1.0 + np.exp(-z))
        gradient = np.dot(X.T, (hyp - y))
        theta -= (alpha * gradient) / len(X)
    return theta

def feature_scaling(data):
    """Scale the features of the dataset."""
    scaled_data = data.copy()
    scaled_data[:, :-1] = (data[:, :-1] - np.mean(data[:, :-1], axis=0)) / (np.max(data[:, :-1], axis=0) - np.min(data[:, :-1], axis=0))
    return scaled_data

def prepare_data(file_path):
    """Load and prepare the data from a CSV file."""
    df = pd.read_csv(file_path, usecols=["all_mcqs_avg_n20", "all_NBME_avg_n4", "LEVEL"])
    unscaled_data = df.to_numpy()
    scaled_data = feature_scaling(unscaled_data)
    data = np.append(np.ones([len(scaled_data), 1]), scaled_data, axis=1)
    X = data[:, :-1]
    y = data[:, -1]
    return X, y

def train_one_vs_all(X_train, y_train, labels, alpha=0.001, num_iterations=400):
    """Train one-vs-all logistic regression classifiers."""
    final_theta = []
    for label in labels:
        y_train_binary = np.where(y_train == label, 1, 0)
        theta = np.random.rand(X_train.shape[1], 1)
        theta = gradient_descent(theta, X_train, y_train_binary, alpha, num_iterations)
        final_theta.append(theta.T.flatten().tolist())
    return np.array(final_theta)

def predict(X, theta_labels):
    """Predict the class labels for a set of inputs."""
    z = np.dot(X, theta_labels.T)
    hyp = 1.0 / (1.0 + np.exp(-z))
    predictions = np.argmax(hyp, axis=1)
    return predictions

def main():
    # Load and prepare the data
    X, y = prepare_data("BSOM_DataSet_for_HW2.csv")
    
    # Split the data into training and test sets
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.33, random_state=27)
    
    # Define the class labels
    labels = ["A", "B", "C", "D"]
    
    # Train the logistic regression classifiers
    theta_labels = train_one_vs_all(X_train, y_train, labels)
    
    # Predict on the test set
    y_pred = predict(X_test, theta_labels)
    
    # Convert actual labels to numerical values
    y_actual = np.array([labels.index(label) for label in y_test])
    
    # Print results
    print("Predicted values:", y_pred)
    print("Actual values:", y_actual)
    
    results = confusion_matrix(y_actual, y_pred)
    print("Confusion Matrix:\n", results)
    
    accuracy = accuracy_score(y_actual, y_pred)
    print("Accuracy:", accuracy)
    
    print(classification_report(y_actual, y_pred, target_names=labels))

if __name__ == '__main__':
    main()
