import pandas as pd
import numpy as np

def data_preprocessing(file_path):
    df = pd.read_csv(file_path)
    species_to_target = {species: idx for idx, species in enumerate(df['species'].unique())}
    df['target'] = df['species'].map(species_to_target)
    return df

def data_shuffling(df):
    shuffled_df = df.sample(frac=1).reset_index(drop=True)
    X = shuffled_df.iloc[:, :-2].values
    Y = shuffled_df.iloc[:, -1].values
    return shuffled_df, X, Y

def data_split(X, Y, split_ratio=0.8):
    total_records = len(X)
    train_len = int(total_records * split_ratio)
    X_train, X_test = X[:train_len], X[train_len:]
    Y_train, Y_test = Y[:train_len], Y[train_len:]
    return X_train, Y_train, X_test, Y_test

def init_parameters(feature_count):
    w = np.random.rand(feature_count)
    b = 0
    return w, b

def forward_prop(X, w, b):
    return np.dot(X, w) + b

def cost_function(y_hat, y):
    m = len(y)
    cost = (1 / (2 * m)) * np.sum(np.square(y_hat - y))
    return cost

def back_prop(X, y, y_hat):
    m = len(y)
    dy_hat = (y_hat - y) / m
    dw = np.dot(X.T, dy_hat)
    db = np.sum(dy_hat)
    return dw, db

def gradient_descent(w, b, dw, db, lr):
    w -= lr * dw
    b -= lr * db
    return w, b

def train(X_train, Y_train, epochs=10, lr=0.001):
    w, b = init_parameters(X_train.shape[1])
    for epoch in range(epochs):
        for i in range(len(X_train)):
            X_sample, y_sample = X_train[i], Y_train[i]
            y_hat = forward_prop(X_sample, w, b)
            cost = cost_function(y_hat, y_sample)
            dw, db = back_prop(X_sample, y_sample, y_hat)
            w, b = gradient_descent(w, b, dw, db, lr)
        if epoch % 2 == 0:
            print(f"Epoch {epoch}: Cost {cost}")
    print(f"Trained Parameters -> w: {w}, b: {b}")
    return w, b

def predict(X_test, Y_test, w, b):
    predictions = np.round(forward_prop(X_test, w, b))
    accuracy = np.mean(predictions == Y_test) * 100
    print(f"Accuracy: {accuracy}%")
    return accuracy

def cross_validation(X, Y, folds=5):
    fold_size = len(X) // folds
    accuracies = {}
    
    for fold in range(folds):
        start, end = fold * fold_size, (fold + 1) * fold_size
        X_train = np.concatenate((X[:start], X[end:]), axis=0)
        Y_train = np.concatenate((Y[:start], Y[end:]), axis=0)
        X_test, Y_test = X[start:end], Y[start:end]

        w, b = train(X_train, Y_train)
        accuracy = predict(X_test, Y_test, w, b)
        accuracies[fold] = accuracy

    print(f"Cross-validation Accuracies: {accuracies}")

def main():
    df = data_preprocessing("iris.csv")
    df, X, Y = data_shuffling(df)
    X_train, Y_train, X_test, Y_test = data_split(X, Y)

    print("Training and predicting results for the test data using the trained model:")
    w, b = train(X_train, Y_train)
    predict(X_test, Y_test, w, b)

    print("\nCross-validation results for the trained model:")
    cross_validation(X, Y)

if __name__ == "__main__":
    main()
