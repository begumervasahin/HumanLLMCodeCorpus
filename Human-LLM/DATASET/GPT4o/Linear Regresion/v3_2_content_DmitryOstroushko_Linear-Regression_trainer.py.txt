import sys
import argparse
import matplotlib.pyplot as plt
from typing import List, Tuple

# Import custom functions
from stream_funcs import normal_message, error_message
from csv_funcs import load_data_from_csv, load_theta_from_csv, save_theta_to_csv
from model import normalize_data, fit, estimated_price

def parse_options() -> argparse.Namespace:
    """
    Parse command line arguments.
    """
    parser = argparse.ArgumentParser(description="Linear Regression Model")
    parser.add_argument('--verbose', '-v', action='store_true', default=False,
                        help='Enable verbose output')
    parser.add_argument('--auto_break', '-b', action='store_true', default=False,
                        help='Stop program when cost function has minimum value')
    parser.add_argument('--iteration', '-i', type=int, default=500,
                        help='Number of iterations (default: 500)')
    parser.add_argument('--coeff', '-c', action='store_true', default=False,
                        help='Start from known coefficients from CSV file')
    parser.add_argument('--learningRate', '-l', type=float, default=0.2,
                        help='Learning rate (default: 0.2)')
    parser.add_argument('--draw', '-d', action='store_true', default=False,
                        help='Enable drawing mode')
    parser.add_argument('--mean_square', '-q', action='store_true', default=False,
                        help='Calculate mean-square deviation')
    return parser.parse_args()

def main() -> None:
    """
    Main function for linear regression project.
    """
    options = parse_options()

    if options.verbose:
        normal_message(options)

    if options.alpha <= 0:
        error_message(f"Learning rate should be greater than 0. Given: {options.alpha}")
        sys.exit(1)

    # Load and normalize data
    X_original, y_original, mileage_limits, mileage_range, price_limits = load_data_from_csv()
    X_normalized = normalize_data(X_original)
    y_normalized = normalize_data(y_original)
    thetas = [0.0, 0.0] if not options.coeff else load_theta_from_csv()

    if options.verbose:
        normal_message(f"Initial thetas: {thetas}")

    try:
        thetas, thetas_history, J_history = fit(X_normalized, y_normalized, thetas, options)
        save_theta_to_csv(thetas)
    except Exception as e:
        error_message("Error during gradient descent computation")
        error_message(str(e))
        sys.exit(1)

    if options.draw:
        plot_cost_function(J_history)
        plot_predictions(X_original, y_original, thetas, mileage_limits, mileage_range, price_limits)

def plot_cost_function(J_history: List[float]) -> None:
    """
    Plot the cost function over iterations.
    """
    plt.figure()
    plt.title('Cost Function Over Iterations')
    plt.xlabel('Iteration')
    plt.ylabel('Cost')
    plt.grid(True)
    plt.plot(range(len(J_history)), J_history, 'r--', label='Cost')
    plt.legend()
    plt.show()

def plot_predictions(X_original: List[float], y_original: List[float], thetas: List[float],
                     mileage_limits: Tuple[float, float], mileage_range: Tuple[float, float], 
                     price_limits: Tuple[float, float]) -> None:
    """
    Plot the real vs predicted prices.
    """
    y_pred = estimated_price(X_original, thetas, mileage_limits, mileage_range, price_limits)
    plt.figure()
    plt.title('Real vs Predicted Prices')
    plt.xlabel('Index')
    plt.ylabel('Price')
    plt.grid(True)
    plt.plot(range(len(y_original)), y_original, 'b--', label='Real Price')
    plt.plot(range(len(y_pred)), y_pred, 'r--', label='Predicted Price')
    plt.legend(loc="upper left")
    plt.show()

if __name__ == '__main__':
    main()
