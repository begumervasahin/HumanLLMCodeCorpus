DECIMAL_PRECISION = 4

def transpose(matrix):
    """Returns the transpose of the given matrix."""
    return [[matrix[row][col] for row in range(len(matrix))] for col in range(len(matrix[0]))]

def get_minor(matrix, omitted_row, omitted_col):
    """Returns the minor of the matrix after removing the specified row and column."""
    return [row[:omitted_col] + row[omitted_col + 1:] for row in (matrix[:omitted_row] + matrix[omitted_row + 1:])]

def calculate_determinant(matrix):
    """Calculates the determinant of a given square matrix."""
    if len(matrix) == 1:
        return matrix[0][0]
    
    determinant = 0
    sign = 1
    for col in range(len(matrix[0])):
        determinant += sign * matrix[0][col] * calculate_determinant(get_minor(matrix, 0, col))
        sign *= -1
    return determinant

def inverse_matrix(matrix):
    """Calculates the inverse of a given square matrix."""
    determinant = calculate_determinant(matrix)
    if determinant == 0:
        raise ValueError("Matrix is not invertible")
    
    cofactor_matrix = [
        [((-1) ** (row + col)) * calculate_determinant(get_minor(matrix, row, col)) for col in range(len(matrix[0]))]
        for row in range(len(matrix))
    ]

    transposed_cofactor = transpose(cofactor_matrix)
    for row in range(len(transposed_cofactor)):
        for col in range(len(transposed_cofactor[0])):
            transposed_cofactor[row][col] /= determinant
    
    return transposed_cofactor

def multiply_matrices(a, b):
    """Multiplies two matrices a and b."""
    rows_a = len(a)
    cols_a = len(a[0])
    cols_b = len(b[0])
    
    product = [[0 for _ in range(cols_b)] for _ in range(rows_a)]
    for i in range(rows_a):
        for j in range(cols_b):
            for k in range(cols_a):
                product[i][j] += a[i][k] * b[k][j]
    
    return product

def read_input_file(file_path):
    """Reads and parses the input file."""
    with open(file_path) as input_file:
        k = int(input_file.readline().strip())
        n = int(input_file.readline().strip())
        observations = [tuple(map(float, input_file.readline().split())) for _ in range(n)]
        exponents = list(map(int, input_file.readline().split()))
    return k, n, observations, exponents

def construct_matrices(observations, exponents, k):
    """Constructs the matrices X and Y for regression calculation."""
    n = len(observations)
    X = [[1] + [obs[i] ** exponents[i] for i in range(k)] for obs in observations]
    Y = [[obs[k]] for obs in observations]
    return X, Y

def calculate_coefficients(X, Y):
    """Calculates the regression coefficients."""
    Xt = transpose(X)
    XtX = multiply_matrices(Xt, X)
    XtX_inv = inverse_matrix(XtX)
    XtY = multiply_matrices(Xt, Y)
    B = multiply_matrices(XtX_inv, XtY)
    return B

def print_equation(B, k):
    """Prints the regression equation."""
    equation = f"y = {round(B[0][0], DECIMAL_PRECISION)}"
    for i in range(1, k + 1):
        equation += f" + {round(B[i][0], DECIMAL_PRECISION)} * x{i}"
    print(f'\nEquation of hyperplane is: {equation}')

def main():
    print('\nMultiple Linear Regression Software\n')
    print('Describing the input file:\n')
    print('First line: number of independent variables - k')
    print('Second line: number of observations - n')
    print('Next n lines: tuples of data corresponding to each observation: x1 x2 ... xk y')
    print('Last line: exponents of independent variables x1, x2, ... xk')
    
    file_path = 'input_data.txt'
    k, n, observations, exponents = read_input_file(file_path)
    
    print(f'\nInput processed: k = {k}, n = {n}, observations are: {observations}')
    print(f'Coefficients for independent variables are: {exponents}')
    
    print('\nCalculating coefficients...')
    X, Y = construct_matrices(observations, exponents, k)
    B = calculate_coefficients(X, Y)
    
    print(f'Auxiliary matrices: \nX = {X}, \nY = {Y}')
    
    print_equation(B, k)

if __name__ == "__main__":
    main()
