import numpy as np
import numpy.random as npr
import matplotlib.pyplot as plt
from mnist import MNIST

# Load MNIST data
mndata = MNIST('.')
x_train, y_train = mndata.load_training()
x_test, y_test = mndata.load_testing()

# Helper function to convert data to numpy arrays and reshape
def to_np_array(data, shape):
    return np.array(data).reshape(shape)

# Convert and reshape data
x_train = to_np_array(x_train, [-1, 28 * 28])
y_train = to_np_array(y_train, [-1, 1])
x_test = to_np_array(x_test, [-1, 28 * 28])
y_test = to_np_array(y_test, [-1, 1])

# Convert multi-classification to binary classification (1 vs all others)
y_train = np.where(y_train != 1.0, -1.0, 1.0)
y_test = np.where(y_test != 1.0, -1.0, 1.0)

# Define Linear Regression model
class LinearReg:
    def __init__(self, in_dim: int, out_dim: int, bias: bool = False):
        self.bias = bias
        self.A = npr.uniform(-0.2, 0.2, [out_dim, in_dim])
        if bias:
            self.B = np.ones([out_dim, 1], dtype=np.float64)
    
    def __call__(self, x, apply_sign=False):
        yh = self.A @ x
        if self.bias:
            yh += self.B
        if apply_sign:
            return np.sign(yh)
        return yh
    
    @staticmethod
    def loss_fn(yh, y):
        return (yh - y) ** 2.0

# Train the Linear Regression model
def train(model, train_data, epochs=2, verbose=False, train_with_bias=False):
    x_train, y_train = train_data
    lrate = 9e-9
    N = x_train.shape[0]

    for epoch in range(epochs):
        loss = 0.0
        dJdA = np.zeros_like(model.A, dtype=np.float64)
        
        for i in range(N):
            x = x_train[i][..., None]
            y = y_train[i][..., None]
            yh = model(x)
            loss += model.loss_fn(yh, y)
            dJdyh = (yh - y)
            dJdA += dJdyh @ x.T

        loss /= (N * 2.0)
        dJdA /= N
        model.A -= lrate * dJdA

        if train_with_bias:
            pass  # Placeholder for bias update logic

        if verbose or epoch == epochs - 1:
            print(f"Training iteration: {epoch + 1}, Loss: {float(loss)}")
    
    return model

# Evaluate the Linear Regression model
def evaluate(model, test_data):
    x_test, y_test = test_data
    loss, accuracy = 0.0, 0.0
    N = x_test.shape[0]

    for i in range(N):
        x = x_test[i][..., None]
        y = y_test[i][..., None]
        yh = model(x)
        loss += model.loss_fn(yh, y)
        accuracy += (np.sign(yh) == y).sum() / y.size

    loss /= (N * 2.0)
    accuracy /= N
    print(f"Loss: {float(loss)}, Accuracy: {accuracy}")

# Initialize, train, and evaluate the model
model = LinearReg(in_dim=784, out_dim=1, bias=False)
model = train(model, (x_train, y_train), epochs=15)
evaluate(model, (x_test, y_test))

# Print a prediction for the first test sample
print(f"Pred: {model(x_test[0][..., None], apply_sign=True)}, True: {y_test[0]}")
