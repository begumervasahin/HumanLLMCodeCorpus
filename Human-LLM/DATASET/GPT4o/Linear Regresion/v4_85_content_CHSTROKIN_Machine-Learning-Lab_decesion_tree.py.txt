from math import log2
import operator

def calculate_shannon_entropy(dataset):
    """
    Calculate the Shannon Entropy of a dataset.
    
    Args:
        dataset (list of lists): The dataset where each inner list is a data instance.
    
    Returns:
        float: The Shannon Entropy of the dataset.
    """
    num_entries = len(dataset)
    label_counts = {}

    # Count occurrences of each label
    for feature_vector in dataset:
        current_label = feature_vector[-1]
        if current_label not in label_counts:
            label_counts[current_label] = 0
        label_counts[current_label] += 1

    # Calculate Shannon Entropy
    shannon_entropy = 0.0
    for key in label_counts:
        prob = label_counts[key] / num_entries
        shannon_entropy -= prob * log2(prob)
    
    return shannon_entropy

def create_dataset():
    """
    Create a simple dataset for demonstration.
    
    Returns:
        tuple: A tuple containing the dataset and labels.
    """
    dataset = [
        [1, 1, 'yes'],
        [1, 1, 'yes'],
        [1, 0, 'no'],
        [0, 1, 'no'],
        [0, 1, 'no']
    ]
    labels = ['input 1', 'input 2']
    return dataset, labels

def split_dataset(dataset, axis, value):
    """
    Split the dataset based on a feature value.
    
    Args:
        dataset (list of lists): The dataset to split.
        axis (int): The index of the feature to split on.
        value: The value of the feature to split on.
    
    Returns:
        list of lists: The subset of the dataset where the feature matches the given value.
    """
    ret_dataset = []
    for feature_vector in dataset:
        if feature_vector[axis] == value:
            reduced_feature_vector = feature_vector[:axis] + feature_vector[axis+1:]
            ret_dataset.append(reduced_feature_vector)
    return ret_dataset

def choose_best_feature_to_split(dataset):
    """
    Choose the best feature to split the dataset on to maximize information gain.
    
    Args:
        dataset (list of lists): The dataset to split.
    
    Returns:
        int: The index of the best feature to split on.
    """
    num_features = len(dataset[0]) - 1
    base_entropy = calculate_shannon_entropy(dataset)
    best_info_gain = 0.0
    best_feature = -1

    for i in range(num_features):
        # Create a set of all unique values for this feature
        feature_values = [example[i] for example in dataset]
        unique_values = set(feature_values)
        new_entropy = 0.0

        # Calculate the new entropy for each unique value
        for value in unique_values:
            subset = split_dataset(dataset, i, value)
            prob = len(subset) / len(dataset)
            new_entropy += prob * calculate_shannon_entropy(subset)

        # Calculate information gain
        info_gain = base_entropy - new_entropy

        if info_gain > best_info_gain:
            best_info_gain = info_gain
            best_feature = i

    return best_feature

def majority_count(class_list):
    """
    Determine the majority class in a list of classes.
    
    Args:
        class_list (list): The list of classes.
    
    Returns:
        The majority class.
    """
    class_count = {}
    for vote in class_list:
        if vote not in class_count:
            class_count[vote] = 0
        class_count[vote] += 1

    sorted_class_count = sorted(class_count.items(), key=operator.itemgetter(1), reverse=True)
    return sorted_class_count[0][0]

def create_tree(dataset, labels):
    """
    Create a decision tree from the dataset.
    
    Args:
        dataset (list of lists): The dataset to create the tree from.
        labels (list): The list of feature labels.
    
    Returns:
        dict: The decision tree represented as a nested dictionary.
    """
    class_list = [example[-1] for example in dataset]

    # If all classes are the same, return that class
    if class_list.count(class_list[0]) == len(class_list):
        return class_list[0]

    # If no more features, return the majority class
    if len(dataset[0]) == 1:
        return majority_count(class_list)

    best_feature = choose_best_feature_to_split(dataset)
    best_feature_label = labels[best_feature]
    tree = {best_feature_label: {}}

    # Remove the best feature from labels
    sub_labels = labels[:]
    del(sub_labels[best_feature])

    # Split dataset and create subtree for each value of the best feature
    feature_values = [example[best_feature] for example in dataset]
    unique_values = set(feature_values)
    for value in unique_values:
        tree[best_feature_label][value] = create_tree(split_dataset(dataset, best_feature, value), sub_labels)

    return tree

def classify(input_tree, feature_labels, test_vector):
    """
    Classify a test vector using the decision tree.
    
    Args:
        input_tree (dict): The decision tree.
        feature_labels (list): The list of feature labels.
        test_vector (list): The test vector to classify.
    
    Returns:
        The class label predicted by the decision tree.
    """
    first_str = next(iter(input_tree))
    second_dict = input_tree[first_str]
    feature_index = feature_labels.index(first_str)
    key = test_vector[feature_index]
    value_of_feat = second_dict[key]

    if isinstance(value_of_feat, dict):
        class_label = classify(value_of_feat, feature_labels, test_vector)
    else:
        class_label = value_of_feat

    return class_label
