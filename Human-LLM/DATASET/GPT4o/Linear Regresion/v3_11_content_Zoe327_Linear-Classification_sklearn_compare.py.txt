import numpy as np
import pandas as pd
from sklearn.model_selection import KFold
from sklearn.discriminant_analysis import LinearDiscriminantAnalysis
from sklearn.linear_model import LogisticRegression
from sklearn.utils import shuffle

# Load and preprocess the wine quality dataset
def load_wine_data(filename):
    data_df = pd.read_csv(filename, header=0, sep=';')
    data = shuffle_data(data_df.values)
    X = data[:, :-1]
    Y = np.asarray([1 if label > 5 else 0 for label in data[:, -1]])
    return X, Y

# Load and preprocess the breast cancer dataset
def load_cancer_data(filename):
    with open(filename) as input_file:
        lines = input_file.readlines()
        data_clean = []
        for line in lines:
            newline = line.strip().split(',')
            temp = newline[1:]
            temp = [-1 if val == '?' else int(val) for val in temp]
            data_clean.append(temp)
        data_clean = shuffle_data(np.array(data_clean))
        X = data_clean[:, :-1]
        Y = np.asarray([1 if label == 2 else 0 for label in data_clean[:, -1]])
    return X, Y

# Add a dummy feature to the dataset (not used in this script)
def add_dummy_feature(X):
    n, m = X.shape
    X0 = np.ones((n, 1))
    return np.column_stack((X0, X))

# Split the dataset into training and testing sets
def split_train_test(X, Y, train_ratio=0.9):
    split_index = int(len(Y) * train_ratio)
    return X[:split_index], Y[:split_index], X[split_index:], Y[split_index:]

# Shuffle the dataset
def shuffle_data(data):
    np.random.shuffle(data)
    return data

# Perform K-fold cross-validation using Logistic Regression and Linear Discriminant Analysis
def evaluate_models(X, Y):
    kf = KFold(n_splits=5, shuffle=True)
    accuracies_lr = []
    accuracies_lda = []
    
    for train_index, test_index in kf.split(X):
        train_x, val_x = X[train_index], X[test_index]
        train_y, val_y = Y[train_index], Y[test_index]
        
        # Logistic Regression
        lr_model = LogisticRegression(random_state=0, max_iter=1000, penalty='l2', solver='liblinear', C=1)
        lr_model.fit(train_x, train_y)
        accuracies_lr.append(lr_model.score(val_x, val_y) * 100)
        
        # Linear Discriminant Analysis
        lda_model = LinearDiscriminantAnalysis()
        lda_model.fit(train_x, train_y)
        accuracies_lda.append(lda_model.score(val_x, val_y) * 100)
    
    print_model_performance('Logistic Regression', accuracies_lr)
    print_model_performance('Linear Discriminant Analysis', accuracies_lda)

def print_model_performance(model_name, accuracies):
    print(f'{model_name}')
    print('Accuracy per fold:', accuracies)
    print('Average accuracy:', np.mean(accuracies))

# Load and split the datasets
X_wine, Y_wine = load_wine_data('winequality-red.csv')
train_x_wine, train_y_wine, test_x_wine, test_y_wine = split_train_test(X_wine, Y_wine)

X_cancer, Y_cancer = load_cancer_data('breast-cancer-wisconsin.data')
train_x_cancer, train_y_cancer, test_x_cancer, test_y_cancer = split_train_test(X_cancer, Y_cancer)

# Perform tests and print results
print('Red-wine data')
evaluate_models(X_wine, Y_wine)

print('Breast-cancer data')
evaluate_models(X_cancer, Y_cancer)
