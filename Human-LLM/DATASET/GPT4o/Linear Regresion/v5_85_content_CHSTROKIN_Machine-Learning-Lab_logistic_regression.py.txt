import numpy as np
import math
import time
from tqdm import tqdm

class NeuralNetwork:
    def __init__(self, learning_rate, input_size, train_x, train_y):
        """
        Initialize the NeuralNetwork class with learning rate, input size, training data, and labels.
        
        :param learning_rate: Learning rate for gradient descent
        :param input_size: Size of one training example
        :param train_x: Training data (features)
        :param train_y: Training labels
        """
        self.learning_rate = learning_rate
        self.input_size = input_size
        self.train_x = train_x
        self.train_y = train_y
        self.weights = np.random.rand(input_size)
        self.bias = 0.0
    
    def randomize_weights(self):
        """Randomize the weights."""
        self.weights = np.random.rand(self.input_size)
    
    def print_weights(self):
        """Print the current weights and bias."""
        print(f"Weights: {self.weights}, Bias: {self.bias}")
    
    @staticmethod
    def sigmoid(z):
        """
        Compute the sigmoid function.
        
        :param z: Input value
        :return: Sigmoid of the input
        """
        return 1 / (1 + math.exp(-z))
    
    @staticmethod
    def squared_error(y_pred, y_true):
        """
        Compute the squared error.
        
        :param y_pred: Predicted value
        :param y_true: True value
        :return: Squared error
        """
        return (y_true - y_pred) ** 2
    
    def train(self, epochs):
        """
        Train the model using gradient descent.
        
        :param epochs: Number of training iterations
        """
        for epoch in tqdm(range(epochs)):
            for i in range(len(self.train_x)):
                x = self.train_x[i]
                y_true = self.train_y[i]
                y_pred = self.predict(x)
                
                # Update weights and bias using gradient descent
                for j in range(self.input_size):
                    self.weights[j] -= self.learning_rate * 2 * (y_pred - y_true) * y_pred * (1 - y_pred) * x[j]
                self.bias -= self.learning_rate * 2 * (y_pred - y_true) * y_pred * (1 - y_pred)
    
    def predict(self, x):
        """
        Predict the output for a given input.
        
        :param x: Input value
        :return: Predicted output
        """
        z = np.dot(x, self.weights) + self.bias
        return self.sigmoid(z)

def generate_primes(n):
    """
    Generate the first n prime numbers.
    
    :param n: Number of prime numbers to generate
    :return: List of prime numbers
    """
    primes = []
    num = 2
    while len(primes) < n:
        if is_prime(num):
            primes.append(num)
        num += 1
    return primes

def is_prime(n):
    """
    Check if a number is prime.
    
    :param n: Number to check
    :return: True if prime, False otherwise
    """
    if n <= 1:
        return False
    for i in range(2, int(math.sqrt(n)) + 1):
        if n % i == 0:
            return False
    return True

def arithmetic_function(n):
