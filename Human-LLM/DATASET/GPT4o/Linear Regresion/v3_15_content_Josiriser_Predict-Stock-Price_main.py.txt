import os
import csv
import numpy as np
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression

def load_data(filepath):
    """Load stock data from a CSV file."""
    high_prices = []
    low_prices = []
    close_prices = []
    
    with open(filepath, 'r', newline="", encoding='utf-8') as csvfile:
        reader = csv.reader(csvfile)
        next(reader)  # Skip the header row
        for row in reader:
            high_prices.append(float(row[2]))
            low_prices.append(float(row[3]))
            close_prices.append(float(row[4]))
    
    return high_prices, low_prices, close_prices

def get_stock_prices(directory):
    """Load stock data from all CSV files in a directory."""
    all_prices = []
    filenames = sorted(os.listdir(directory))
    
    for filename in filenames:
        filepath = os.path.join(directory, filename)
        all_prices.append(load_data(filepath))
    
    return all_prices

def split_data(prices):
    """Split the data into features and targets for training and prediction."""
    high_prices, low_prices, close_prices = prices
    X = []
    Y = []
    real_x = []
    num_samples = len(high_prices) // 13
    
    for i in range(num_samples):
        features = []
        targets = []
        for j in range(i * 13, i * 13 + 3):
            features.extend([high_prices[j], low_prices[j], close_prices[j]])
        for j in range(i * 13 + 3, i * 13 + 13):
            targets.append(close_prices[j])
        X.append(np.array(features))
        Y.append(np.array(targets))
    
    for k in range(len(close_prices) - 13, len(close_prices) - 10):
        real_x.extend([high_prices[k], low_prices[k], close_prices[k]])
    
    real_y = close_prices[-10:]
    return X, Y, real_x, real_y

def plot_predictions(days, real_y, predicted_y, labels, colors):
    """Plot the real and predicted stock prices."""
    real_label, predicted_label = labels
    real_color, predicted_color = colors
    
    plt.plot(days, real_y, real_color, label=real_label)
    plt.plot(days, predicted_y, predicted_color, label=predicted_label)

def process_and_plot_stock_data(stock_prices, stock_name, colors):
    """Process stock data, train model, and plot results."""
    X, Y, real_x, real_y = split_data(stock_prices)
    model = LinearRegression()
    model.fit(X, Y)
    predicted_y = model.predict([real_x])[0]
    plot_predictions(
        days=list(range(1, 11)),
        real_y=real_y,
        predicted_y=predicted_y,
        labels=(f"{stock_name}-Real", f"{stock_name}-Predict"),
        colors=colors
    )

def main():
    """Main function to execute the stock price prediction and plotting."""
    directory = 'stock_data'
    all_prices = get_stock_prices(directory)

    # Process and plot data for each stock
    process_and_plot_stock_data(all_prices[0], "YFY", ('b', 'g'))
    process_and_plot_stock_data(all_prices[1], "TSMC", ('r', 'c'))
    process_and_plot_stock_data(all_prices[2], "MTK", ('m', 'y'))

    plt.legend()
    plt.show()

if __name__ == "__main__":
    main()
