class RegressionController:
    def __init__(self, path=None, verbose=False):
        """
        Initialize the RegressionController.
        @param path: Path to the PCD file.
        @param verbose: If True, print more information to the console.
        """
        self.points = self.load_points(path) if path else []
        self.verbose = verbose
        self.parameters = []
        self.parts = []
        self.intersections = []

    def load_points(self, path):
        """
        Load points from a PCD file.
        @param path: Path to the PCD file.
        @return: List of points.
        """
        point_cloud = pypcd.PointCloud.from_path(path)
        return [[point['x'], point['y']] for point in point_cloud.pc_data]

    def clear_state(self):
        """Clear the internal state of the controller."""
        self.regressors = []
        self.parts = []
        self.intersections = []

    def get_points(self):
        """Get the loaded points."""
        return self.points

    def set_points(self, points):
        """Set the points to be processed."""
        self.points = points

    def fit(self, regressor):
        """
        Fit the points using the provided regressor.
        @param regressor: Instance of a regressor.
        """
        if not self.parts:
            raise Exception("Nothing to fit. Ensure set_parts() method is called first.")
        
        for count, part in enumerate(self.parts, start=1):
            if part.isolated:
                self.intersections.append(None)
                self.parameters.append(None)
                continue
            if self.verbose:
                print(f"\n\n************************ Fitting part {count} ************************")
            self.intersections.append(regressor.process(part.points))
            self.parameters.append(regressor.get_parameters())

    def set_parts(self, divider, resolution=1):
        """
        Divide all points into parts for fitting.
        @param divider: PointsDivider object to divide points into parts.
        @param resolution: Resolution for dividing the points.
        """
        divider.set_points(self.points)
        divider.set_blocks(resolution)
        self.parts = divider.divide()

    def get_intersections(self):
        """Get the intersections calculated during fitting."""
        return self.intersections


class SimpleRegressionController(RegressionController):
    def set_parts(self):
        """
        Set parts based on predefined limits for filtering points.
        """
        limits = [
            [-2, -1, float('-inf'), float('inf')],
            [4, 5, float('-inf'), float('inf')],
            [float('-inf'), float('inf'), -1, -0.2],
            [float('-inf'), float('inf'), 3, 4]
        ]

        for limit in limits:
            points = filter_points(self.points, limit[0], limit[1], limit[2], limit[3])
            self.parts.append(ContinuousPart(points))

    def get_intersections(self):
        """
        Calculate intersections based on fitted parameters.
        @return: List of intersections.
        """
        parameters = [param[0] for param in self.parameters]
        print(parameters)

        points = [
            get_intersection(parameters[0], parameters[2]),
            get_intersection(parameters[0], parameters[3]),
            get_intersection(parameters[1], parameters[2]),
            get_intersection(parameters[1], parameters[3])
        ]

        intersections = [
            [[points[0][0], points[0][1] + 2], points[1]],
            [points[0], points[2]],

        ]
        
        return intersections
