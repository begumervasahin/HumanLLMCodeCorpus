import numpy as np
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

def load_data(file_path):
    """
    Load data from a CSV file using pandas.

    :param file_path: Path to the CSV file
    :return: DataFrame containing the loaded data
    """
    return pd.read_csv(file_path)

def calculate_weights(query_point, data, tau):
    """
    Calculate the weights for each data point based on the query point using a Gaussian kernel.

    :param query_point: The query point x
    :param data: The dataset X
    :param tau: Bandwidth parameter controlling the weight fall-off
    :return: Diagonal matrix of weights
    """
    num_samples, _ = data.shape
    weights = np.mat(np.eye(num_samples))
    for i in range(num_samples):
        diff = data[i] - query_point
        weights[i, i] = np.exp(diff * diff.T / (-2.0 * tau**2))
    return weights

def compute_local_weight(query_point, data, labels, tau):
    """
    Compute the local weight for a given query point.

    :param query_point: The query point x
    :param data: The dataset X
    :param labels: The label vector y
    :param tau: Bandwidth parameter controlling the weight fall-off
    :return: Theta (θ) that minimizes the cost function
    """
    weights = calculate_weights(query_point, data, tau)
    theta = (data.T * (weights * data)).I * (data.T * weights * labels.T)
    return theta

def local_weight_regression(data, labels, tau, show_process=False):
    """
    Perform Local Weighted Regression.

    :param data: The dataset X
    :param labels: The label vector y
    :param tau: Bandwidth parameter controlling the weight fall-off
    :param show_process: Flag to print the intermediate calculations
    :return: Predictions vector
    """
    num_samples, _ = data.shape
    predictions = np.zeros(num_samples)
    for i in range(num_samples):
        theta = compute_local_weight(data[i], data, labels, tau)
        predictions[i] = data[i] * theta
        if show_process:
            print(f"h(θ) = {theta[0]} + {theta[1]}")
    return predictions

def visualize_data(data, plot_title, x_label, y_label):
    """
    Visualize the given data using a scatter plot.

    :param data: DataFrame containing the data
    :param plot_title: Title of the plot
    :param x_label: Label for the x-axis
    :param y_label: Label for the y-axis
    :return: None
    """
    ax = sns.scatterplot(x=x_label, y=y_label, data=data)
    ax.set_title(plot_title)
    plt.show()
