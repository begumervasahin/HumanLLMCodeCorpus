import numpy as np
import pandas as pd
from sklearn.model_selection import KFold
from sklearn.discriminant_analysis import LinearDiscriminantAnalysis
from sklearn.linear_model import LogisticRegression
from sklearn.utils import shuffle

def load_wine_data(filename):
    """Load and preprocess the wine quality dataset."""
    data_df = pd.read_csv(filename, header=0, sep=';')
    data = shuffle_data(data_df.values)
    X = data[:, :-1]
    Y = np.where(data[:, -1] > 5, 1, 0)
    return X, Y

def load_breast_cancer_data(filename):
    """Load and preprocess the breast cancer dataset."""
    data_clean = []

    with open(filename) as input_file:
        lines = input_file.readlines()
        for line in lines:
            temp = [int(value) if value != '?' else -1 for value in line.strip().split(',')[1:]]
            data_clean.append(temp)
    
    data_clean = shuffle_data(np.array(data_clean))
    X = data_clean[:, :-1]
    Y = np.where(data_clean[:, -1] == 2, 1, 0)
    return X, Y

def add_dummy_feature(X):
    """Add a dummy feature (column of ones) to the dataset."""
    return np.hstack([np.ones((X.shape[0], 1)), X])

def split_train_test(X, Y, test_ratio=0.1):
    """Split the dataset into training and testing sets."""
    split_index = int((1 - test_ratio) * len(Y))
    return X[:split_index], Y[:split_index], X[split_index:], Y[split_index:]

def shuffle_data(data):
    """Shuffle the data."""
    return shuffle(data)

def evaluate_models(X, Y):
    """Evaluate classifiers using K-Fold cross-validation."""
    kf = KFold(n_splits=5, shuffle=True)

    accuracy_lr = []
    accuracy_lda = []

    for train_index, test_index in kf.split(X):
        train_x, val_x = X[train_index], X[test_index]
        train_y, val_y = Y[train_index], Y[test_index]

        # Logistic Regression
        lr_model = LogisticRegression(random_state=0, max_iter=1000, penalty='l2', solver='liblinear', C=1)
        lr_model.fit(train_x, train_y)
        accuracy_lr.append(lr_model.score(val_x, val_y) * 100)

        # Linear Discriminant Analysis
        lda_model = LinearDiscriminantAnalysis()
        lda_model.fit(train_x, train_y)
        accuracy_lda.append(lda_model.score(val_x, val_y) * 100)

    print('Logistic Regression')
    print(np.array(accuracy_lr))
    print('Average accuracy =', np.mean(accuracy_lr))

    print('Linear Discriminant Analysis')
    print(np.array(accuracy_lda))
    print('Average accuracy =', np.mean(accuracy_lda))

# Load and split the datasets
X_wine, Y_wine = load_wine_data('winequality-red.csv')
X_wine_train, Y_wine_train, X_wine_test, Y_wine_test = split_train_test(X_wine, Y_wine)

X_cancer, Y_cancer = load_breast_cancer_data('breast-cancer-wisconsin.data')
X_cancer_train, Y_cancer_train, X_cancer_test, Y_cancer_test = split_train_test(X_cancer, Y_cancer)

# Evaluate models on the datasets
print('Red wine data:')
evaluate_models(X_wine_train, Y_wine_train)

print('\nBreast cancer data:')
evaluate_models(X_cancer_train, Y_cancer_train)
