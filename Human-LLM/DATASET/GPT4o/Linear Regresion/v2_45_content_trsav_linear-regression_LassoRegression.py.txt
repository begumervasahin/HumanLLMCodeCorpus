import numpy as np
import pandas as pd
from scipy.optimize import minimize
from scipy.optimize import NonlinearConstraint
from tqdm import tqdm
import matplotlib.pyplot as plt

# Define the loss function for training data
def loss(B):
    B0 = B[0]
    B = B[1:]
    error = 0
    for i in range(len(data_train)):
        error += (response_train[i, :] - B0 - np.dot(data_train[i], B)) ** 2
    return error

# Define the loss function for test data
def loss_test(B):
    B0 = B[0]
    Bj = B[1:]
    error = 0
    for i in range(len(test_data)):
        error += (test_response[i, :] - B0 - np.dot(test_data[i], Bj)) ** 2
    return error

# Define the constraint function
def constraint_fun(B):
    Bj = B[1:]
    return np.dot(Bj, Bj)

# Define the function to get the ith fold of k total folds
def get_fold(i, k, data):
    fold_size = len(data) // k
    start_idx = i * fold_size
    end_idx = start_idx + fold_size
    
    data_test = data[start_idx:end_idx]
    data_train = np.concatenate((data[:start_idx], data[end_idx:]))
    
    return data_train, data_test

# Load and preprocess the data
data = pd.read_excel('prostate_data.xlsx').to_numpy()

# Standardize the data
data = (data - data.mean(axis=0)) / data.std(axis=0)

# Data augmentation
data_aug1 = data * np.random.normal(1, 0.05, data.shape)
data_aug2 = data * np.random.normal(1, 0.05, data.shape)
data = np.concatenate((data, data_aug1, data_aug2))

# Shuffle the data
np.random.shuffle(data)

# Load column labels
labels = pd.read_excel('prostate_data.xlsx', nrows=0).columns

# Define constants
t_limit = 0.7
folds = 4
t_fidel = 20
t_vals = np.linspace(0, t_limit, t_fidel)
B_init = np.zeros(data.shape[1])

# Initialize storage for bias and variance
bias = np.zeros((folds, t_fidel))
var = np.zeros((folds, t_fidel))

# Perform k-fold cross-validation
for j in range(folds):
    data_train_full, data_test_full = get_fold(j, folds, data)
    test_data = data_test_full[:, :-1]
    test_response = data_test_full[:, -1:]
    global data_train, response_train
    data_train = data_train_full[:, :-1]
    response_train = data_train_full[:, -1:]
    
    for i in tqdm(range(t_fidel)):
        t = t_vals[i]
        constraint = NonlinearConstraint(constraint_fun, 0, t)
        sol = minimize(loss, B_init, constraints=[constraint], method='SLSQP')
        B_opt = sol.x
        bias[j, i] = sol.fun
        var[j, i] = loss_test(B_opt)

# Calculate mean bias and variance
bias_mean = np.mean(bias, axis=0) / len(data_train)
var_mean = np.mean(var, axis=0) / len(test_data)

# Plot training and test errors
plt.figure()
plt.plot(t_vals, bias_mean, label='Train')
plt.plot(t_vals, var_mean, label='Test')
plt.xlabel(r'$t$ where $\sum_j |\beta_j| \leq t$')
plt.ylabel('MSE')
plt.legend()
plt.grid()
plt.show()

# Store the optimized coefficients
B_store = np.zeros((t_fidel, 1 + data.shape[1]))

for i in tqdm(range(t_fidel)):
    t = t_vals[i]
    constraint = NonlinearConstraint(constraint_fun, 0, t)
    sol = minimize(loss, B_init, constraints=[constraint], method='SLSQP')
    B_store[i, :] = sol.x

# Plot the coefficients
plt.figure()
for i in range(1, len(B_init)):
    plt.plot(t_vals, B_store[:, i])
    plt.text(t_vals[-1], B_store[-1, i], labels[i - 1])
plt.grid()
plt.ylabel(r'$\beta$')
plt.xlabel(r'$t$ where $\sum_j |\beta_j| \leq t$')
plt.show()
