import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import statsmodels.formula.api as smf
from sklearn.metrics import mean_squared_error
from math import sqrt

# Load the dataset
sal_hike = pd.read_csv('C:/Users/HP/Desktop/assignments submission/simple linear regression/Salary_Data.csv')

# Plotting the scatter plot of YearsExperience vs Salary
plt.scatter(sal_hike.YearsExperience, sal_hike.Salary)
plt.xlabel('Years of Experience')
plt.ylabel('Salary')
plt.title('Scatter plot of Years of Experience vs Salary')
plt.show()

# Correlation matrix
corr_matrix = sal_hike.corr()
print("Correlation matrix:\n", corr_matrix)

# Linear Model
lin_model = smf.ols('Salary ~ YearsExperience', data=sal_hike).fit()
print("Linear Model Parameters:\n", lin_model.params)
print("Linear Model Summary:\n", lin_model.summary())

# Predictions and Errors for Linear Model
lin_predict = lin_model.predict(sal_hike)
lin_error = sal_hike.Salary - lin_predict

# Plot Linear Model predictions
plt.scatter(sal_hike.YearsExperience, sal_hike.Salary, c='r')
plt.plot(sal_hike.YearsExperience, lin_predict, c='b')
plt.xlabel('Years of Experience')
plt.ylabel('Salary')
plt.title('Linear Model: Years of Experience vs Salary')
plt.show()

# RMSE for Linear Model
lin_rmse = sqrt(mean_squared_error(sal_hike.Salary, lin_predict))
print("Linear Model RMSE:", lin_rmse)

# Logarithmic Model
log_model = smf.ols('Salary ~ np.log(YearsExperience)', data=sal_hike).fit()
print("Logarithmic Model Parameters:\n", log_model.params)
print("Logarithmic Model Summary:\n", log_model.summary())

# Predictions and Errors for Logarithmic Model
log_predict = log_model.predict(sal_hike)
log_error = sal_hike.Salary - log_predict

# Plot Logarithmic Model predictions
plt.scatter(sal_hike.YearsExperience, sal_hike.Salary, c='r')
plt.plot(sal_hike.YearsExperience, log_predict, c='b')
plt.xlabel('Years of Experience')
plt.ylabel('Salary')
plt.title('Logarithmic Model: Years of Experience vs Salary')
plt.show()

# RMSE for Logarithmic Model
log_rmse = sqrt(mean_squared_error(sal_hike.Salary, log_predict))
print("Logarithmic Model RMSE:", log_rmse)

# Exponential Model
exp_model = smf.ols('np.log(Salary) ~ YearsExperience', data=sal_hike).fit()
print("Exponential Model Parameters:\n", exp_model.params)
print("Exponential Model Summary:\n", exp_model.summary())

# Predictions and Errors for Exponential Model
exp_predict = np.exp(exp_model.predict(sal_hike))
exp_error = sal_hike.Salary - exp_predict

# Plot Exponential Model predictions
plt.scatter(sal_hike.YearsExperience, sal_hike.Salary, c='r')
plt.plot(sal_hike.YearsExperience, exp_predict, c='b')
plt.xlabel('Years of Experience')
plt.ylabel('Salary')
plt.title('Exponential Model: Years of Experience vs Salary')
plt.show()

# RMSE for Exponential Model
exp_rmse = sqrt(mean_squared_error(sal_hike.Salary, exp_predict))
print("Exponential Model RMSE:", exp_rmse)

# Quadratic Model
quad_model = smf.ols('Salary ~ YearsExperience + I(YearsExperience**2)', data=sal_hike).fit()
print("Quadratic Model Parameters:\n", quad_model.params)
print("Quadratic Model Summary:\n", quad_model.summary())

# Predictions and Errors for Quadratic Model
quad_predict = quad_model.predict(sal_hike)
quad_error = sal_hike.Salary - quad_predict

# Plot Quadratic Model predictions
plt.scatter(sal_hike.YearsExperience, sal_hike.Salary, c='r')
plt.plot(sal_hike.YearsExperience, quad_predict, c='b')
plt.xlabel('Years of Experience')
plt.ylabel('Salary')
plt.title('Quadratic Model: Years of Experience vs Salary')
plt.show()

# RMSE for Quadratic Model
quad_rmse = sqrt(mean_squared_error(sal_hike.Salary, quad_predict))
print("Quadratic Model RMSE:", quad_rmse)

# Correlation coefficients
print("Correlation coefficient (Linear Model):", np.corrcoef(sal_hike.Salary, lin_predict)[0, 1])
print("Correlation coefficient (Logarithmic Model):", np.corrcoef(sal_hike.Salary, log_predict)[0, 1])
print("Correlation coefficient (Exponential Model):", np.corrcoef(sal_hike.Salary, exp_predict)[0, 1])
print("Correlation coefficient (Quadratic Model):", np.corrcoef(sal_hike.Salary, quad_predict)[0, 1])
