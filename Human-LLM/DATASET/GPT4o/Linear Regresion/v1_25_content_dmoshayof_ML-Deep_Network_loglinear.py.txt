def softmax(x):
    """
    Compute the softmax vector.
    x: a n-dim vector (numpy array)
    returns: an n-dim vector (numpy array) of softmax values
    """
    c = np.max(x)
    ex = np.exp(x - c)
    x = ex / np.sum(ex)
    return x

def classifier_output(x, params):
    """
    Return the output layer (class probabilities)
    of a log-linear classifier with given params on input x.
    """
    W, b = params
    z = np.dot(x, W) + b
    probs = softmax(z)
    return probs

def predict(x, params):
    """
    Returns the prediction (highest scoring class id) of a
    log-linear classifier with given parameters on input x.
    params: a list of the form [(W, b)]
    W: matrix
    b: vector
    """
    return np.argmax(classifier_output(x, params))

def loss_and_gradients(x, y, params):
    """
    Compute the loss and the gradients at point x with given parameters.
    y is a scalar indicating the correct label.
    returns:
        loss,[gW,gb]
    loss: scalar
    gW: matrix, gradients of W
    gb: vector, gradients of b
    """
    W, b = params
    pred_vec = classifier_output(x, params)
    y_hat = pred_vec[y]
    loss = -np.log(y_hat)
    
    y_hot_vector = np.zeros(pred_vec.shape)
    y_hot_vector[y] = 1
    gb = pred_vec - y_hot_vector
    gW = np.outer(x, gb)
    return loss, [gW, gb]

def create_classifier(in_dim, out_dim):
    """
    returns the parameters (W,b) for a log-linear classifier
    with input dimension in_dim and output dimension out_dim.
    """
    W = np.zeros((in_dim, out_dim))
    b = np.zeros(out_dim)
    return [W, b]

def gradient_check(f, x, epsilon=1e-5, threshold=1e-4):
    """
    Perform gradient check for a function f
    f: function that takes a single argument and outputs the loss and its gradient
    x: the point (numpy array) to check the gradient at
    epsilon: the epsilon value for numerical gradient calculation
    threshold: the threshold for considering the gradients as close
    """
    fx, grad = f(x) # Evaluate function value at original point
    grad_approx = np.zeros_like(x)
    it = np.nditer(x, flags=['multi_index'], op_flags=['readwrite'])
    
    while not it.finished:
        ix = it.multi_index
        x[ix] += epsilon
        fxh, _ = f(x)
        x[ix] -= 2 * epsilon
        fxnh, _ = f(x)
        x[ix] += epsilon
        grad_approx[ix] = (fxh - fxnh) / (2 * epsilon)
        it.iternext()
    
    # Compare gradients
    if not np.allclose(grad, grad_approx, atol=threshold):
        print(f"Gradient check failed at index {ix}. Analytical: {grad[ix]}, Numerical: {grad_approx[ix]}")
    else:
        print("Gradient check passed!")
