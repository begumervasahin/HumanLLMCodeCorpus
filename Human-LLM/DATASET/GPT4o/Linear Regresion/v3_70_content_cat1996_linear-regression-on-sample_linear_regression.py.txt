import pandas as pd
import matplotlib.pyplot as plt
import numpy as np
import seaborn as sns
import random
import csv

def load_dataset(filename, split_ratio):
    """
    Load dataset from CSV file and split it into training and test sets.

    Parameters:
    filename (str): The path to the CSV file.
    split_ratio (float): The ratio to split the dataset into training and test sets.

    Returns:
    tuple: Training and test datasets as lists.
    """
    training, test = [], []
    with open(filename, 'r') as csvfile:
        lines = csv.reader(csvfile)
        dataset = list(lines)
        for row in dataset[1:]:
            for col in range(1, 3):
                row[col] = float(row[col])
            if random.random() < split_ratio:
                training.append(row)
            else:
                test.append(row)
    return training, test

def linear_model(X, w):
    """
    Linear model: y = w1 * X + w0

    Parameters:
    X (numpy.ndarray): The input feature matrix.
    w (list): The weights [w0, w1].

    Returns:
    numpy.ndarray: The predicted values.
    """
    return w[1] * np.array(X[:, 0]) + w[0]

def cost_function(w, X, y):
    """
    Cost function for linear regression.

    Parameters:
    w (list): The weights [w0, w1].
    X (numpy.ndarray): The input feature matrix.
    y (numpy.ndarray): The actual target values.

    Returns:
    float: The cost value.
    """
    return (0.5 / len(y)) * np.sum(np.square(linear_model(X, w) - y))

def gradient(w, X, y):
    """
    Calculate gradients for linear regression.

    Parameters:
    w (list): The weights [w0, w1].
    X (numpy.ndarray): The input feature matrix.
    y (numpy.ndarray): The actual target values.

    Returns:
    list: The gradients [grad_0, grad_1].
    """
    grad_0 = (1 / len(y)) * np.sum(linear_model(X, w) - y)
    grad_1 = (1 / len(y)) * np.sum((linear_model(X, w) - y) * X[:, 0])
    return [grad_0, grad_1]

def gradient_descent(w, learning_rate, X, y):
    """
    Perform gradient descent to minimize the cost function.

    Parameters:
    w (list): The initial weights [w0, w1].
    learning_rate (float): The learning rate.
    X (numpy.ndarray): The input feature matrix.
    y (numpy.ndarray): The actual target values.

    Returns:
    list: The optimized weights [w0, w1].
    """
    iteration = 0
    while iteration <= 500:
        grad_vals = gradient(w, X, y)
        w_new = [w[0] - learning_rate * grad_vals[0], w[1] - learning_rate * grad_vals[1]]
        if np.sum((np.array(w_new) - np.array(w)) ** 2) < 1e-6:
            return w_new
        w = w_new
        iteration += 1
    return w

def plot_regression_line(formula, x_range):
    """
    Plot the regression line.

    Parameters:
    formula (function): The regression formula function.
    x_range (numpy.ndarray): The range of x values to plot.
    """
    x = np.array(x_range)
    y = formula(x)
    plt.plot(x, y)

def main():
    # Read and preprocess data
    data = pd.read_csv("weight-height.csv")
    print(data.shape)
    print(data.head())

    # Load and split dataset
    training, test = load_dataset("weight-height.csv", 0.7)

    # Extract training data
    w_t = np.array([row[1] for row in training])
    h_t = np.array([row[2] for row in training])

    # Reshape data for calculations
    X = w_t.reshape(-1, 1)
    y = h_t

    # Plot the training data
    plt.scatter(X, y)
    plt.xlabel('Weight')
    plt.ylabel('Height')
    plt.show()

    # Initialize weights using the closed-form solution
    mean_weight = np.mean(X)
    mean_height = np.mean(y)
    numerator = np.sum((X - mean_weight) * (y - mean_height))
    denominator = np.sum((X - mean_weight) ** 2)
    b1 = numerator / denominator
    b0 = mean_height - b1 * mean_weight
    w = [b0, b1]

    # Perform gradient descent
    w = gradient_descent(w, 0.001, X, y)

    # Plot linear regression line
    def regression_formula(x):
        return w[0] + w[1] * x

    plt.scatter(X, y, c="red", alpha=0.5, marker='o')
    plot_regression_line(regression_formula, np.linspace(min(X), max(X), 200))
    plt.xlabel('Weight')
    plt.ylabel('Height')
    plt.show()

    # Calculate R^2 score
    ss_total = np.sum((y - np.mean(y)) ** 2)
    ss_residual = np.sum((y - regression_formula(X[:, 0])) ** 2)
    r2_score = 1 - (ss_residual / ss_total)
    print(f'The R^2 score is {r2_score}')

    # Plot with Seaborn
    sns.scatterplot(x=w_t, y=h_t)
    plt.xlabel('Weight')
    plt.ylabel('Height')
    plt.show()

if __name__ == "__main__":
    main()
