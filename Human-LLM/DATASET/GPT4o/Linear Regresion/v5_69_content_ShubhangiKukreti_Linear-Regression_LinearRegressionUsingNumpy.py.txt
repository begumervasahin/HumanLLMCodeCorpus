import numpy as np
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d import Axes3D

def predict(X, theta):
    """Compute the prediction for given inputs X and parameters theta."""
    return X @ theta

def compute_cost(X, y, theta):
    """Compute the cost function for linear regression."""
    m = len(y)
    predictions = X @ theta
    cost = (1 / (2 * m)) * np.sum((predictions - y) ** 2)
    return cost

def compute_gradient(X, y, theta):
    """Compute the gradient of the cost function."""
    m = len(y)
    predictions = X @ theta
    gradient = (1 / m) * (X.T @ (predictions - y))
    return gradient

def gradient_descent(X, y, num_params, alpha=0.02, iterations=5000):
    """Perform gradient descent to learn theta parameters."""
    theta = np.zeros(num_params)
    cost_history = np.zeros(iterations)
    
    for i in range(iterations):
        theta -= alpha * compute_gradient(X, y, theta)
        cost_history[i] = compute_cost(X, y, theta)
        
    return theta, cost_history

def normalize_data(x):
    """Normalize the dataset X."""
    scale = x.max(axis=0)
    return x / scale, scale

def split_data(X, y):
    """Split the data into training and testing sets."""
    return X, y, X, y

def main():
    data = np.loadtxt('stockprices.csv', usecols=(1, 2))
    X = data[:, 0]
    y = data[:, 1]
    
    # Plotting the initial data
    plt.figure(figsize=(12, 8))
    plt.scatter(X, y, label='Data')
    plt.xlabel('Amazon')
    plt.ylabel('Google')
    plt.title('Google stock price vs Amazon')
    plt.legend()
    plt.savefig('graph.png')
    
    # Prepare the data
    X_train, y_train, X_test, y_test = split_data(X, y)
    m = len(y)
    X_train = np.column_stack((np.ones((m, 1)), X_train))
    
    # Normalize the data
    X_train_normalized, X_scale = normalize_data(X_train)
    y_train_normalized, y_scale = normalize_data(y_train)
    
    # Test prediction function
    theta = np.array([1, 2])
    print('Testing the prediction function ...')
    print(f'When X = [1, 1] and theta = [1, 2]: {predict(np.ones(2), theta)}')
    print('Expected prediction: 3')
    print(f'When X = [[1, 1], [5, 5]] and theta = [1, 2]: {predict(np.array([[1, 1], [5, 5]]), theta)}')
    print('Expected prediction: [3, 15]')
    
    input('Press Enter to continue...')
    
    # Test cost function
    theta = np.zeros(X_train.shape[1])
    print('Testing the cost function ...')
    print(f'When theta is zero, cost = {compute_cost(X_train_normalized, y_train_normalized, theta)}')
    print('Expected cost: approximately 0.318')
    
    input('Press Enter to continue...')
    
    # Test gradient function
    print('Testing the gradient function ...')
    print(f'When theta is zero, gradient = {compute_gradient(X_train_normalized, y_train_normalized, theta)}')
    print('Expected gradient: approximately [-0.79, -0.59]')
    
    input('Press Enter to continue...')
    
    # Run gradient descent
    print('Running gradient descent ...')
    theta, cost_history = gradient_descent(X_train_normalized, y_train_normalized, X_train.shape[1])
    print(f'After running gradient descent, theta = {theta}')
    print('Expected theta: approximately [0.34, 0.61]')
    
    # Plot predictions
    X_pred = np.linspace(X.min(), X.max(), 100)
    X_pred = np.column_stack((np.ones((100, 1)), X_pred))
    y_pred = predict(X_pred / X_scale, theta) * y_scale
    
    plt.figure(figsize=(12, 8))
    plt.scatter(X_test, y_test, color='b', label='Test Data')
    plt.plot(X_pred[:, 1], y_pred, 'r', label='Prediction')
    plt.xlabel('Amazon')
    plt.ylabel('Google')
    plt.legend(loc=2)
    plt.savefig('pred.png')
    
    # Plot cost history
    plt.figure(figsize=(12, 8))
    plt.semilogy(cost_history, 'r')
    plt.xlabel('Iteration')
    plt.ylabel('Cost')
    plt.savefig('cost.png')
    
    # Plot 3D cost surface
    fig = plt.figure()
    ax = fig.add_subplot(1, 1, 1, projection='3d')
    n = 100
    theta0, theta1 = np.meshgrid(np.linspace(-3, 3, n), np.linspace(-3, 2, n))
    cost_surface = np.empty((n, n))
    
    for i in range(n):
        for j in range(n):
            cost_surface[i, j] = compute_cost(X_train_normalized, y_train_normalized, np.array([theta0[i, j], theta1[i, j]]))
    
    ax.plot_surface(theta0, theta1, cost_surface)
    ax.set_xlabel('theta0')
    ax.set_ylabel('theta1')
    ax.set_zlabel('Cost')
    plt.savefig('J.png')

if __name__ == '__main__':
    main()
