import statsmodels.api as sm
import pandas as pd
import numpy as np

class BoostedSLR:
    def __init__(self,
                 input_endog,
                 n_season=12,
                 forecast_length=12,
                 smooth_factor=1,
                 trend_dampening=1,
                 model_type='add',
                 boost=True,
                 boost_iter=5):
        self.input_endog = input_endog
        self.n_season = n_season
        self.forecast_length = forecast_length
        self.smooth_factor = smooth_factor
        self.trend_dampening = trend_dampening
        self.model_type = model_type
        self.boost = boost
        self.boost_iter = boost_iter

    def get_avg_seasonal_idx(self, ts_list, nk, smooth_factor):
        ak_list = [sum(ts)/len(ts) for ts in ts_list]
        s_list = [ts/ak for ts, ak in zip(ts_list, ak_list)]
        
        if nk < 1.0:
            s_avg = np.mean(s_list)
        else:
            s_avg = [np.mean([s[i] for s in s_list if len(s) > i]) for i in range(len(max(s_list, key=len)))]
        
        s_avg_s = np.array(s_avg) * smooth_factor
        return s_avg_s

    def fit_slr(self):
        n_total = len(self.input_endog)
        nk = n_total / self.n_season
        ts = self.input_endog.values
        ts_list = [ts[i:i + self.n_season] for i in range(0, len(ts), self.n_season)]
        
        s_avg_s = self.get_avg_seasonal_idx(ts_list, nk, self.smooth_factor)
        
        if self.model_type == 'add':
            X = sm.add_constant(range(len(ts)))
            y = ts - np.resize(s_avg_s, len(ts))
        elif self.model_type == 'mult':
            X = sm.add_constant(range(len(ts)))
            y = ts / np.resize(s_avg_s, len(ts))
        else:
            raise ValueError("model_type should be either 'add' or 'mult'")

        if not self.boost:
            model = sm.OLS(y, X)
            res = model.fit()
            coefs = res.params
        else:
            boosted_data = y
            boosted_output = np.zeros(len(y))
            coefs = np.zeros(2)
            for _ in range(self.boost_iter):
                model = sm.OLS(boosted_data, X)
                res = model.fit()
                fitted = res.predict(X)
                boosted_output += 0.2 * fitted
                boosted_data = y - boosted_output
                coefs += 0.2 * res.params

        X_forecast = sm.add_constant(range(len(ts) + self.forecast_length))
        yhat = model.predict(coefs, X_forecast)
        y_pred = yhat * np.resize(s_avg_s, len(yhat))
        
        frc_in = y_pred[:len(ts)]
        frc_out = y_pred[len(ts):]
        
        return {
            'model': model,
            'in_sample_forecast': pd.Series(frc_in),
            'out_of_sample_forecast': pd.Series(frc_out),
            'trend': yhat
        }
