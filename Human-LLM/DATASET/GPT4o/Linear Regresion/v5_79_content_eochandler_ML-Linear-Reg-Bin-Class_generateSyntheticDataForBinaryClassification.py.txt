import numpy as np
import matplotlib.pyplot as plt
from sklearn.linear_model import LogisticRegression

def plot_decision_boundary(beta, x_min, x_max, y_min, y_max):
    """
    Plots the decision boundary given the model coefficients.
    
    Parameters:
    beta (array): Model coefficients.
    x_min (float): Minimum x value.
    x_max (float): Maximum x value.
    y_min (float): Minimum y value.
    y_max (float): Maximum y value.
    """
    x_vals = np.linspace(x_min, x_max, 100)
    y_vals = (-beta[1] * x_vals - beta[0]) / beta[2]
    idxs = np.where((y_vals >= y_min) & (y_vals <= y_max))
    plt.plot(x_vals[idxs], y_vals[idxs])

def generate_data(num_samples, mean, cov):
    """
    Generates multivariate normal data.
    
    Parameters:
    num_samples (int): Number of samples to generate.
    mean (list): Mean of the distribution.
    cov (array): Covariance matrix of the distribution.
    
    Returns:
    array: Generated samples.
    """
    return np.random.multivariate_normal(mean, cov, num_samples)

def sigmoid(u):
    """
    Computes the sigmoid function.
    
    Parameters:
    u (float): Input value.
    
    Returns:
    float: Sigmoid of the input.
    """
    return np.exp(u) / (1 + np.exp(u))

def log_likelihood(beta, X, Y):
    """
    Computes the log-likelihood function for logistic regression.
    
    Parameters:
    beta (array): Model coefficients.
    X (array): Feature matrix.
    Y (array): Labels.
    
    Returns:
    float: Log-likelihood value.
    """
    N = X.shape[0]
    total_likelihood = 0
    for i in range(N):
        x_i = X[i]
        y_i = Y[i]
        dot_product = np.dot(x_i, beta)
        total_likelihood += -y_i * dot_product + np.log(1 + np.exp(dot_product))
    return total_likelihood

def logistic_regression(X, Y, alpha=0.001, max_iterations=1000):
    """
    Performs logistic regression using gradient descent.
    
    Parameters:
    X (array): Feature matrix.
    Y (array): Labels.
    alpha (float): Learning rate.
    max_iterations (int): Maximum number of iterations.
    
    Returns:
    tuple: Final model coefficients, list of gradient norms, and list of log-likelihood values.
    """
    N, d = X.shape
    X = np.hstack((np.ones((N, 1)), X))
    beta = np.random.randn(d + 1)
    gradient_norms = []
    log_likelihoods = []

    for _ in range(max_iterations):
        gradient = np.zeros(d + 1)
        for i in range(N):
            x_i = X[i]
            y_i = Y[i]
            q_i = sigmoid(np.dot(x_i, beta))
            gradient += (q_i - y_i) * x_i
        norm_gradient = np.linalg.norm(gradient)
        gradient_norms.append(norm_gradient)
        beta -= alpha * gradient
        log_likelihoods.append(log_likelihood(beta, X, Y))
    
    return beta, gradient_norms, log_likelihoods

def main():
    # Parameters
    num_samples = 100
    np.random.seed(7)
    mu_pos = [1.0, 1.0]
    cov_pos = np.array([[1.0, 0.0], [0.0, 1.0]])
    mu_neg = [-1.0, -1.0]
    cov_neg = np.array([[1.0, 0.0], [0.0, 1.0]])

    # Generate data
    X_pos = generate_data(num_samples, mu_pos, cov_pos)
    X_neg = generate_data(num_samples, mu_neg, cov_neg)
    X = np.vstack((X_pos, X_neg))
    Y = np.hstack((np.ones(num_samples), np.zeros(num_samples)))

    # Train custom logistic regression model
    beta, gradient_norms, log_likelihoods = logistic_regression(X, Y)

    # Plot gradient norms and log-likelihoods
    plt.figure()
    plt.semilogy(gradient_norms, label='Gradient Norm')
    plt.semilogy(log_likelihoods, label='Log-Likelihood')
    plt.legend()
    plt.show()

    # Plot data and decision boundary
    plt.figure()
    plt.scatter(X_pos[:, 0], X_pos[:, 1], label='Positive')
    plt.scatter(X_neg[:, 0], X_neg[:, 1], label='Negative')
    plot_decision_boundary(beta, min(X[:, 0]), max(X[:, 0]), min(X[:, 1]), max(X[:, 1]))
    plt.axis("equal")
    plt.legend()
    plt.show()

    # Train sklearn logistic regression model
    model = LogisticRegression(solver='lbfgs', penalty='none')
    model.fit(X, Y)
    beta_sklearn = np.hstack((model.intercept_, model.coef_.flatten()))

    # Plot sklearn model decision boundary
    plt.figure()
    plt.scatter(X_pos[:, 0], X_pos[:, 1], label='Positive')
    plt.scatter(X_neg[:, 0], X_neg[:, 1], label='Negative')
    plot_decision_boundary(beta_sklearn, min(X[:, 0]), max(X[:, 0]), min(X[:, 1]), max(X[:, 1]))
    plt.axis("equal")
    plt.legend()
    plt.show()

if __name__ == "__main__":
    main()
