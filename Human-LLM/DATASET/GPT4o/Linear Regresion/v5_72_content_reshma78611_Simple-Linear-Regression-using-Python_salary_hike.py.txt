import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import statsmodels.formula.api as smf
from sklearn.metrics import mean_squared_error
from math import sqrt

# Load the dataset
data_path = 'C:/Users/HP/Desktop/assignments submission/simple linear regression/Salary_Data.csv'
sal_hike = pd.read_csv(data_path)

def plot_experience_vs_salary(experience, salary, predictions=None, title='Years of Experience vs Salary'):
    """Plot Years of Experience vs Salary with optional prediction line."""
    plt.scatter(experience, salary, color='r', label='Actual')
    if predictions is not None:
        plt.plot(experience, predictions, color='b', label='Predicted')
    plt.xlabel('Years of Experience')
    plt.ylabel('Salary')
    plt.title(title)
    plt.legend()
    plt.show()

def calculate_rmse(actual, predicted):
    """Calculate Root Mean Squared Error (RMSE)."""
    return sqrt(mean_squared_error(actual, predicted))

def print_summary(model, model_name):
    """Print the summary of the regression model."""
    print(f"{model_name} Summary:")
    print(model.summary())
    print("\n")

# Scatter plot of YearsExperience vs Salary
plot_experience_vs_salary(sal_hike.YearsExperience, sal_hike.Salary)

# Correlation between YearsExperience and Salary
correlation = np.corrco
ef(sal_hike.YearsExperience, sal_hike.Salary)[0, 1]
print(f"Correlation: {correlation}")

# Linear Regression Model
lin_model = smf.ols('Salary ~ YearsExperience', data=sal_hike).fit()
print_summary(lin_model, "Linear Regression")

# Predictions and RMSE using Linear Regression Model
lin_predict = lin_model.predict(sal_hike)
lin_rmse = calculate_rmse(sal_hike.Salary, lin_predict)
print(f"Linear Regression RMSE: {lin_rmse}")

# Plot Linear Regression results
plot_experience_vs_salary(sal_hike.YearsExperience, sal_hike.Salary, lin_predict, 'Linear Regression: Years of Experience vs Salary')

# Logarithmic Regression Model
log_model = smf.ols('Salary ~ np.log(YearsExperience)', data=sal_hike).fit()
print_summary(log_model, "Logarithmic Regression")

# Predictions and RMSE using Logarithmic Regression Model
log_predict = log_model.predict(sal_hike)
log_rmse = calculate_rmse(sal_hike.Salary, log_predict)
print(f"Logarithmic Regression RMSE: {log_rmse}")

# Plot Logarithmic Regression results
plot_experience_vs_salary(sal_hike.YearsExperience, sal_hike.Salary, log_predict, 'Logarithmic Regression: Years of Experience vs Salary')

# Exponential Regression Model
exp_model = smf.ols('np.log(Salary) ~ YearsExperience', data=sal_hike).fit()
print_summary(exp_model, "Exponential Regression")

# Predictions and RMSE using Exponential Regression Model
exp_predict = np.exp(exp_model.predict(sal_hike))
exp_rmse = calculate_rmse(sal_hike.Salary, exp_predict)
print(f"Exponential Regression RMSE: {exp_rmse}")

# Plot Exponential Regression results
plot_experience_vs_salary(sal_hike.YearsExperience, sal_hike.Salary, exp_predict, 'Exponential Regression: Years of Experience vs Salary')

# Quadratic Regression Model
quad_model = smf.ols('Salary ~ YearsExperience + I(YearsExperience ** 2)', data=sal_hike).fit()
print_summary(quad_model, "Quadratic Regression")

# Predictions and RMSE using Quadratic Regression Model
quad_predict = quad_model.predict(sal_hike)
quad_rmse = calculate_rmse(sal_hike.Salary, quad_predict)
print(f"Quadratic Regression RMSE: {quad_rmse}")

# Plot Quadratic Regression results
plot_experience_vs_salary(sal_hike.YearsExperience, sal_hike.Salary, quad_predict, 'Quadratic Regression: Years of Experience vs Salary')
