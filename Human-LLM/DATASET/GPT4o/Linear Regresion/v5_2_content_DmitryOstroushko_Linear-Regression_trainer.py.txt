import sys
import math
import argparse
from typing import Tuple, List
import matplotlib.pyplot as plt
from stream_funcs import *
from csv_funcs import *
from model import *

def parse_options() -> argparse.Namespace:
    """
    Define and parse command line arguments.
    """
    parser = argparse.ArgumentParser(description='Linear Regression Model')
    parser.add_argument('--verbose', '-v', action='store_true', help='Enable verbosity mode')
    parser.add_argument('--auto_break', '-b', action='store_true', help='Stop when cost function reaches minimum value')
    parser.add_argument('--iteration', '-i', type=int, default=500, help='Number of iterations (default: 500)')
    parser.add_argument('--coeff', '-c', action='store_true', help='Start from known coefficients from CSV file')
    parser.add_argument('--learningRate', '-l', type=float, default=0.2, help='Learning coefficient (default: 0.2)')
    parser.add_argument('--draw', '-d', action='store_true', help='Enable drawing mode')
    parser.add_argument('--mean_square', '-q', action='store_true', help='Calculate mean-square deviation')
    return parser.parse_args()

def main() -> None:
    """
    Main function for the linear regression project.
    """
    options = parse_options()

    if options.verbose:
        normal_message(options)

    if options.alpha <= 0:
        error_message(f'{options.alpha}: Invalid learning coefficient. It must be greater than 0.')
        sys.exit(1)

    X_original, y_original, mileage_limits, mileage_range, price_limits = load_data_from_csv()
    X = normalize_data(X_original)
    y = normalize_data(y_original)
    thetas = load_theta_from_csv() if options.coeff else [0.0, 0.0]

    if options.verbose:
        normal_message(f'Starting thetas: {thetas}')

    try:
        thetas, thetas_history, J_history = fit(X, y, thetas, options)
        save_theta_to_csv(thetas)
    except Exception as e:
        error_message('Gradient descent computation failed')
        error_message(str(e))
        sys.exit(1)

    if options.draw:
        plot_cost_function(J_history)
        plot_real_vs_predicted_prices(X_original, y_original, thetas, mileage_limits, mileage_range, price_limits)

def plot_cost_function(J_history: List[float]) -> None:
    """
    Plot the cost function over iterations.
    """
    plt.figure()
    plt.title('Cost Function')
    plt.xlabel('Iteration')
    plt.ylabel('Cost')
    plt.grid(True)
    plt.plot(range(len(J_history)), J_history, 'r--', label='Cost')
    plt.legend()
    plt.show()

def plot_real_vs_predicted_prices(X_original: List[float], y_original: List[float], thetas: List[float],
                                  mileage_limits: Tuple[float, float], mileage_range: float, price_limits: Tuple[float, float]) -> None:
    """
    Plot real vs predicted prices.
    """
    y_pred = estimated_price(X_original, thetas, mileage_limits, mileage_range, price_limits)
    plt.figure()
    plt.title('Real vs Predicted Prices')
    plt.xlabel('Index')
    plt.ylabel('Price')
    plt.grid(True)
    plt.plot(range(len(y_original)), y_original, 'b--', label='Real price')
    plt.plot(range(len(y_pred)), y_pred, 'r--', label='Predicted price')
    plt.legend()
    plt.show()

if __name__ == '__main__':
    main()
