import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import math
import statistics
import time
from tqdm import tqdm

def load_data(file_path):
    data = pd.read_csv(file_path)
    if len(data['quadratmeter']) != len(data['preis']):
        raise ValueError('Ups es ist was schief gelaufen daten stimmen nicht überein!!')
    return data

def plot_data(data):
    plt.scatter(data["preis"], data["quadratmeter"], color='red')
    plt.xlabel('Preis')
    plt.ylabel('Quadratmeter')
    plt.title('Preis vs Quadratmeter')
    plt.show()

def calculate_nearest_and_farthest_points(data):
    nearest = None
    farthest = None
    d_max = None
    d_min = None

    for i in range(len(data['quadratmeter'])):
        x = data['preis'][i]
        y = data['quadratmeter'][i]
        d = math.sqrt(x**2 + y**2)
        if nearest is None:
            nearest = (x, y)
            farthest = (x, y)
            d_min = d
            d_max = d
        elif d_min > d:
            nearest = (x, y)
            d_min = d
        elif d_max < d:
            farthest = (x, y)
            d_max = d

    return nearest, farthest

def plot_points_with_marks(data, nearest, farthest):
    plt.scatter(data["preis"], data["quadratmeter"], color='red')
    plt.scatter(*nearest, color='green')
    plt.scatter(*farthest, color='green')
    plt.xlabel('Preis')
    plt.ylabel('Quadratmeter')
    plt.title('Markierte Punkte')
    plt.show()

def calculate_initial_line(nearest, farthest):
    x1, y1 = nearest
    x2, y2 = farthest
    m = (y2 - y1) / (x2 - x1)
    b = y1 - m * x1
    return m, b

def plot_initial_line(data, nearest, farthest, m, b):
    plt.scatter(data["preis"], data["quadratmeter"], color='red')
    plt.scatter(*nearest, color='green')
    plt.scatter(*farthest, color='green')
    plt.plot(data['preis'], data['preis'] * m + b, color='blue', label='InitLine')
    plt.xlabel('Preis')
    plt.ylabel('Quadratmeter')
    plt.title('Initiale Gerade')
    plt.legend()
    plt.show()

def fit_line(data):
    xmean = statistics.mean(data['preis'])
    ymean = statistics.mean(data['quadratmeter'])
    m1, m2 = 0, 0

    for i, x in enumerate(tqdm(data['preis'])):
        m1 += (x - xmean) * (data['quadratmeter'][i] - ymean)
        m2 += (x - xmean)**2
        m = m1 / m2
        b = ymean - m * xmean
        plt.scatter(data["preis"], data["quadratmeter"], color='red')
        plt.plot(data['preis'], data['preis'] * m + b, color='blue')
        plt.xlabel('Preis')
        plt.ylabel('Quadratmeter')
        plt.title('Anpassung der Linie')
        plt.pause(0.05)
        plt.cla()
    
    return m, b

def plot_fitted_line(data, m, b):
    plt.scatter(data["preis"], data["quadratmeter"], color='red')
    plt.plot(data['preis'], data['preis'] * m + b, color='blue', label='FittedLine')
    plt.xlabel('Preis')
    plt.ylabel('Quadratmeter')
    plt.title('Finale Gerade')
    plt.legend()
    plt.show()

def predict_price(m, b):
    while True:
        try:
            size = float(input('Geben sie eine Grösse eines Hotels ein um denn vorrausgesagten Preis zu bekommen: '))
            price = m * size + b
            print(f'Die KI hat denn preis = {price:.2f} Euro vorrausgesagt.')
        except ValueError as e:
            print('Bitte nur Zahlen eingeben, ohne "m" oder ähnliches!', e)

def main():
    data = load_data("hotels.csv")

    print('Hier sind die Daten veranschaulicht')
    plot_data(data)

    print('Jetzt werden die Anfangspunkte ausgerechnet...')
    time.sleep(1)
    nearest, farthest = calculate_nearest_and_farthest_points(data)

    print('Nähester Punkt:', nearest)
    print('Entferntester Punkt:', farthest)
    time.sleep(0.5)

    print('Hier sind die nähesten/ entferntesten Punkte markiert.')
    plot_points_with_marks(data, nearest, farthest)

    print('Jetzt werden m und b für die Anfangslinie ausgerechnet...')
    time.sleep(1)
    m, b = calculate_initial_line(nearest, farthest)

    print(f'Die Anfangsgerade lautet: y = {m:.2f}x + {b:.2f}')
    print('Hier ist die Gerade zu denn Daten gezeichnet.')
    plot_initial_line(data, nearest, farthest, m, b)

    print('Jetzt wird die gerade angepasst...')
    m, b = fit_line(data)

    print('Die perfekte Linie wurde ausgerechenet!')
    print(f'Die perfekte Linie lautet: f(x) = {m:.2f}x + {b:.2f}')
    print(f'Oder auch y = {m:.2f}x + {b:.2f}')

    print('Die Finale Gerade wurde gefunden!')
    plot_fitted_line(data, m, b)

    print('Jetzt kann man das Program nutzen um vorraussagen zu treffen...')
    predict_price(m, b)

if __name__ == "__main__":
    main()
