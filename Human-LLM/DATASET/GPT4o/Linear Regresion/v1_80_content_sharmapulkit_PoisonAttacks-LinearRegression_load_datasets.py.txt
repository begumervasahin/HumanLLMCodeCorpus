import pandas as pd
import numpy as np

class DatasetStruct:
    """
    A class to store features (X) and target (Y) datasets.
    """
    def __init__(self, X, Y):
        self._X = X
        self._Y = Y

    @property
    def X(self):
        return self._X

    @property
    def Y(self):
        return self._Y

    def get_size(self):
        return len(self._X)


class Datasets:
    """
    A base class for loading and managing datasets.
    """
    def __init__(self, train_frac=0.7):
        self.train_frac = train_frac
        self.train = None
        self.val = None
        self.whole = None

    def load(self):
        self.df = pd.read_csv(self.data_path)
        feature_columns = self.df.columns.drop(self.target_columns)
        X = self.df[feature_columns].values
        Y = self.df[self.target_columns].values
        self.whole = DatasetStruct(X, Y)

        N = len(X)
        all_idxs = np.arange(N)
        np.random.shuffle(all_idxs)
        train_idxs = all_idxs[:int(self.train_frac * N)]
        val_idxs = all_idxs[int(self.train_frac * N):]

        self.train = DatasetStruct(X[train_idxs], Y[train_idxs])
        self.val = DatasetStruct(X[val_idxs], Y[val_idxs])

    def get_df(self):
        return self.df

    def get_size(self):
        return len(self.df)

    def get_train(self):
        if self.train is None:
            raise ValueError("Train data not loaded")
        return self.train

    def get_val(self):
        if self.val is None:
            raise ValueError("Validation data not loaded")
        return self.val

class InitialDataSet(Datasets):
    """
    A class for generating and managing initial datasets with specific transformations.
    """
    def __init__(self):
        super().__init__()

    def load_inv_flip(self, data_tr, N):
        """
        Load an Inverse Flip dataset randomly from training data.

        Parameters:
        - data_tr: training data (DatasetStruct)
        - N: number of samples to load
        """
        train_data_size = data_tr.get_size()
        assert N < train_data_size, "N should be less than the training data size"
        selected_idxs = np.random.choice(train_data_size, size=N, replace=False)
        self.X = data_tr.X[selected_idxs]
        self.Y = 1 - data_tr.Y[selected_idxs]
        self.whole = DatasetStruct(self.X, self.Y)

    def load_b_flip(self, data_tr, N):
        """
        Load a Boundary Flip dataset randomly from training data.

        Parameters:
        - data_tr: training data (DatasetStruct)
        - N: number of samples to load
        """
        train_data_size = data_tr.get_size()
        assert N < train_data_size, "N should be less than the training data size"
        selected_idxs = np.random.choice(train_data_size, size=N, replace=False)
        self.X = data_tr.X[selected_idxs]
        self.Y = np.round(1 - data_tr.Y[selected_idxs])
        self.whole = DatasetStruct(self.X, self.Y)

    def load_random(self, N, mean, variance):
        """
        Load random data points sampled from a Multivariate Normal distribution.

        Parameters:
        - N: number of samples to generate
        - mean: mean of the distribution
        - variance: covariance matrix of the distribution
        """
        self.X, self.Y = np.random.multivariate_normal(mean, variance, N).T
        self.whole = DatasetStruct(self.X, self.Y)

    def get_size(self):
        return len(self.X)

    def get_df(self):
        """
        Convert the dataset to a pandas DataFrame.
        """
        return pd.DataFrame(self.X)
