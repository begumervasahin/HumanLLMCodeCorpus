def linear_regression_one_variable():
    print('Step 2: Linear regression with one variable')
    
    # Step 2.1: Plotting the data
    print('Step 2.1: Plotting the data')

    print('Step 2.2: Gradient descent')

    plot_data_with_line(df, model.theta)

    # Step 2.4: Visualizing J(theta)
    print('Step 2.4: Visualizing J(theta)')

def predict_profit(model, population):
    example_x = tf.constant([1.0, population], shape=(1, 2))
    predicted_y = model.predict(example_x).numpy()[0][0]
    print(f'When the size of the population is {population * 10000:.0f} => The predicted profit is {predicted_y * 10000:.2f}$')

def predict_price(model, scaler, size, bedrooms):
    example_features = scaler.transform(np.array([size, bedrooms]).reshape(1, -1))
    example_x = tf.constant(np.concatenate([np.ones((1, 1)), example_features], axis=1), shape=(1, 3), dtype=tf.float32)
    predicted_price = model.predict(example_x).numpy()[0][0]
    print(f'When the house has an area of {size} square feet and {bedrooms} bedrooms => Its predicted price is {predicted_price:.2f}$')

def predict_price_normal_equation(X, y, size, bedrooms):
    X = X.to_numpy()
    X = np.concatenate([np.ones((X.shape[0], 1)), X], axis=1)
    theta_neq = np.linalg.inv(X.T @ X) @ X.T @ y
    prediction = (np.array([1, size, bedrooms]).reshape(1, -1) @ theta_neq)[0][0]
    print(f'When the house has an area of {size} square feet and {bedrooms} bedrooms => Its predicted price is {prediction:.2f}$')
