import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
import statsmodels.api as sm
import statsmodels.formula.api as smf
from sklearn.metrics import mean_squared_error
from math import sqrt

def load_data(file_path):
    """Load dataset from a CSV file."""
    return pd.read_csv(file_path, encoding="ISO-8859-1")

def select_columns(data, columns):
    """Select specific columns from the dataset."""
    return data[columns]

def check_missing_values(data):
    """Check for missing values in the dataset."""
    return data.isna().sum()

def plot_pairplot(data):
    """Plot pairwise relationships in the dataset."""
    sns.pairplot(data)
    plt.show()

def calculate_vif(data, predictors):
    """Calculate Variance Inflation Factor (VIF) for each predictor."""
    vif_data = []
    for predictor in predictors:
        formula = f'{predictor} ~ ' + ' + '.join([p for p in predictors if p != predictor])
        rsq = smf.ols(formula, data=data).fit().rsquared
        vif = 1 / (1 - rsq)
        vif_data.append({'Variable': predictor, 'VIF': vif})
    return pd.DataFrame(vif_data)

def plot_regression_diagnostics(model):
    """Plot partial regression and influence diagnostics."""
    sm.graphics.plot_partregress_grid(model)
    plt.show()
    sm.graphics.influence_plot(model)
    plt.show()

def fit_model(formula, data):
    """Fit a linear regression model."""
    return smf.ols(formula, data=data).fit()

def main():
    # Load the dataset
    data_path = 'C:/Users/HP/Desktop/assignments submission/multi linear regression/ToyotaCorolla.csv'
    corolla_data = load_data(data_path)

    # Select relevant columns
    selected_columns = ["Price", "Age_08_04", "KM", "HP", "cc", "Doors", "Gears", "Quarterly_Tax", "Weight"]
    Toyota = select_columns(corolla_data, selected_columns)

    # Check for missing values and drop them
    print(check_missing_values(Toyota))
    Toyota = Toyota.dropna()

    # Plot pairwise relationships
    plot_pairplot(Toyota)

    # Display correlation matrix
    print(Toyota.corr())

    # Fit the initial regression model
    formula = 'Price ~ Age_08_04 + KM + HP + cc + Doors + Gears + Quarterly_Tax + Weight'
    toyota_model = fit_model(formula, Toyota)
    print(toyota_model.summary())

    # Calculate and display VIF values
    predictors = ['Age_08_04', 'KM', 'HP', 'cc', 'Doors', 'Gears', 'Quarterly_Tax', 'Weight']
    vif_frame = calculate_vif(Toyota, predictors)
    print(vif_frame)

    # Plot regression diagnostics
    plot_regression_diagnostics(toyota_model)

    # Remove outlier and refit the model
    outliers = [80]
    Toyota1 = Toyota.drop(Toyota.index[outliers])
    toyota_model1 = fit_model(formula, Toyota1)
    print(toyota_model1.summary())

    # Plot diagnostics after removing outlier
    plot_regression_diagnostics(toyota_model1)

    # Remove additional outliers and refit the model
    additional_outliers = [221, 960, 601]
    Toyota2 = Toyota.drop(Toyota.index[outliers + additional_outliers])
    toyota_model2 = fit_model(formula, Toyota2)
    print(toyota_model2.summary())

    # Plot diagnostics after removing additional outliers
    plot_regression_diagnostics(toyota_model2)

    # Log transform of certain predictors and refit the model
    formula_log = 'Price ~ Age_08_04 + KM + HP + np.log(cc) + np.log(Doors) + Gears + Quarterly_Tax + Weight'
    toyota_model3 = fit_model(formula_log, Toyota2)
    print(toyota_model3.summary())

    # Square transform of certain predictors and refit the model
    Toyota2['cc_sq'] = Toyota2['cc'] ** 2
    Toyota2['Door_sq'] = Toyota2['Doors'] ** 2
    formula_sq = 'Price ~ Age_08_04 + KM + HP + cc + Doors + Gears + Quarterly_Tax + Weight + Door_sq'
    toyota_model4 = fit_model(formula_sq, Toyota2)
    print(toyota_model4.summary())

    # Predictions and RMSE calculation
    Toyota_predict = toyota_model4.predict(Toyota2)
    Toyota_rmse = sqrt(mean_squared_error(Toyota2['Price'], Toyota_predict))
    print(f'RMSE: {Toyota_rmse}')

if __name__ == "__main__":
    main()
