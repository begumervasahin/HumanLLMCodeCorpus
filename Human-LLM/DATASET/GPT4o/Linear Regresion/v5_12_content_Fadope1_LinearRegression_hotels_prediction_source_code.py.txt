import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import math
import statistics
import time
from tqdm import tqdm

# Load data
data = pd.read_csv("hotels.csv")

# Check if the data columns have the same length
if len(data['quadratmeter']) != len(data['preis']):
    print('Ups, es ist etwas schief gelaufen. Daten stimmen nicht überein!')
    exit()

# Visualize the data
print('Hier sind die Daten veranschaulicht')
plt.scatter(data["preis"], data["quadratmeter"], color='red')
plt.xlabel('Preis')
plt.ylabel('Quadratmeter')
plt.title('Preis vs. Quadratmeter')
plt.show()

# Calculate the initial points
print('Jetzt werden die Anfangspunkte ausgerechnet...')
time.sleep(1)
nearest = None
farthest = None
d_min = float('inf')
d_max = float('-inf')

for i in range(len(data['quadratmeter'])):
    x = data['preis'][i]
    y = data['quadratmeter'][i]
    d = math.sqrt(x**2 + y**2)
    if d < d_min:
        nearest = (x, y)
        d_min = d
    if d > d_max:
        farthest = (x, y)
        d_max = d

x1, y1 = nearest
x2, y2 = farthest

print('Nähester Punkt:', nearest)
print('Entferntester Punkt:', farthest)

# Mark the nearest and farthest points
print('Hier sind die nähesten/ entferntesten Punkte markiert.')
plt.scatter(data["preis"], data["quadratmeter"], color='red')
plt.scatter(x1, y1, color='green')
plt.scatter(x2, y2, color='green')
plt.xlabel('Preis')
plt.ylabel('Quadratmeter')
plt.title('Preis vs. Quadratmeter mit markierten Punkten')
plt.show()

# Calculate the initial line
print('Jetzt werden m und b für die Anfangslinie ausgerechnet...')
time.sleep(1)
m = (y2 - y1) / (x2 - x1)
b = y1 - m * x1

print('Die Anfangsgerade lautet: y = {:.2f}x + {:.2f}'.format(m, b))

# Plot the initial line
print('Hier ist die Gerade zu den Daten gezeichnet.')
plt.scatter(data["preis"], data["quadratmeter"], color='red')
plt.scatter(x1, y1, color='green')
plt.scatter(x2, y2, color='green')
plt.plot(data['preis'], data['preis'] * m + b, color='blue', label='InitLine')
plt.xlabel('Preis')
plt.ylabel('Quadratmeter')
plt.title('Preis vs. Quadratmeter mit Anfangsgerade')
plt.legend()
plt.show()

# Calculate the mean values
xmean = statistics.mean(data['preis'])
ymean = statistics.mean(data['quadratmeter'])

# Adjust the line using least squares method
print('Jetzt wird die Gerade angepasst...')
m1 = 0
m2 = 0

for x in tqdm(data['preis']):
    y = data['quadratmeter'][data['preis'] == x].values[0]
    m1 += (x - xmean) * (y - ymean)
    m2 += (x - xmean) ** 2
    m = m1 / m2
    b = ymean - m * xmean
    plt.scatter(data["preis"], data["quadratmeter"], color='red')
    plt.plot(data['preis'], data['preis'] * m + b, color='blue')
    plt.xlabel('Preis')
    plt.ylabel('Quadratmeter')
    plt.title('Anpassung der Gerade')
    plt.show(block=False)
    plt.pause(0.05)
    plt.cla()

print('Die perfekte Linie wurde ausgerechnet!')
print('Die perfekte Linie lautet: y = {:.2f}x + {:.2f}'.format(m, b))

# Plot the final fitted line
print('Die Finale Gerade wurde gefunden!')
plt.scatter(data["preis"], data["quadratmeter"], color='red')
plt.plot(data['preis'], data['preis'] * m + b, color='blue', label='FittedLine')
plt.xlabel('Preis')
plt.ylabel('Quadratmeter')
plt.title('Preis vs. Quadratmeter mit finaler Gerade')
plt.legend()
plt.show()

# Function to predict hotel price based on size
print('Jetzt kann man das Programm nutzen, um Vorhersagen zu treffen...')
while True:
    try:
        size = int(input('Geben Sie eine Größe eines Hotels ein, um den vorhergesagten Preis zu bekommen: '))
        predicted_price = m * size + b
        print('Die KI hat den Preis = {:.2f} Euro vorhergesagt.'.format(predicted_price))
    except ValueError:
        print('Bitte nur Zahlen eingeben, ohne "m" oder Ähnliches!')
