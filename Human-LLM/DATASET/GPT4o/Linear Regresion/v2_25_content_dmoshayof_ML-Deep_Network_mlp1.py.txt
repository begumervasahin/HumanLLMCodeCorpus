def classifier_output(x, params):
    """
    Compute the classifier output for the input x given the parameters.
    """
    W, b, U, b_tag = params
    h = np.tanh(np.dot(x, W) + b)
    probs = softmax(np.dot(h, U) + b_tag)
    return probs

def predict(x, params):
    """
    Predict the class label for the input x given the parameters.
    """
    return np.argmax(classifier_output(x, params))

def loss_and_gradients(x, y, params):
    """
    Compute the loss and gradients for a single data point (x, y) given the parameters.
    """
    W, b, U, b_tag = params
    pred_vec = classifier_output(x, params)
    y_hat = pred_vec[y]
    
    # Compute the loss
    loss = -np.log(y_hat)
    
    # Compute the gradients
    y_hot_vector = np.zeros(pred_vec.shape)
    y_hot_vector[y] = 1
    dl_dz = pred_vec - y_hot_vector
    
    h = np.dot(x, W) + b
    dh_dw = x
    
    # Gradient with respect to b_tag
    gb_tag = dl_dz
    
    # Gradient with respect to U
    dz_dU = np.tanh(h)
    gU = np.outer(gb_tag, dz_dU).T
    
    # Gradient with respect to b
    dz_dh = 1 - np.tanh(h) ** 2
    dz_dh = U.T * dz_dh
    gb = np.dot(gb_tag, dz_dh)
    
    # Gradient with respect to W
    gW = np.outer(x, gb)
    
    return loss, [gW, gb, gU, gb_tag]

def create_classifier(in_dim, hid_dim, out_dim):
    """
    Initialize the parameters for a multi-layer perceptron with the given dimensions.
    """
    W = np.random.randn(in_dim, hid_dim)
    b = np.random.randn(hid_dim)
    U = np.random.randn(hid_dim, out_dim)
    b_tag = np.random.randn(out_dim)
    
    return [W, b, U, b_tag]
