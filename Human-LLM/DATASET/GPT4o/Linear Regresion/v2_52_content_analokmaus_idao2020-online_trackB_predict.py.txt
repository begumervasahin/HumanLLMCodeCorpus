import datetime
import pandas as pd
import numpy as np
from scipy.interpolate import interp1d
import pickle

# Define probability list and target keys
probabilities = [0.04, 0.08, 0.12, 0.2, 0.24, 0.32, 0.40, 0.48, 0.56, 0.64, 0.72, 0.80, 0.88, 0.92, 0.96]
target_keys = ["max", "min"] + [f"p{p:.2f}" for p in probabilities] + [f"pp{p:.2f}" for p in probabilities]

# Define start and end dates
START_DATE = pd.to_datetime("2014-01-01 00:00:00")
END_DATE = pd.to_datetime("2014-03-01 00:00:00")
max_elapsed_seconds = int((END_DATE - START_DATE).total_seconds() * 1.2)

def predict(param_dict):
    fit_max = param_dict["max_1"]
    roots = (np.poly1d(fit_max) - max_elapsed_seconds).roots

    if len(roots) == 2:
        m1, m2 = roots[0], roots[1]
        if m1 < param_dict["n_wave_min"] < m2:
            m = m2
        elif m2 < param_dict["n_wave_min"] < m1:
            m = m1
        elif param_dict["n_wave_min"] < m1 < m2:
            m = m1
        elif param_dict["n_wave_min"] < m2 < m1:
            m = m2
    elif len(roots) == 1:
        m = roots[0]
    else:
        raise ValueError("No valid roots found.")
        
    m = int(m * 1.7)

    predictions = []
    for key in target_keys:
        fit_1 = param_dict[f"{key}_1"]
        fit_2 = param_dict[f"{key}_2"]

        indices = np.arange(param_dict["n_wave_min"], m)
        elapsed_seconds = np.poly1d(fit_1)(indices)
        targets = np.poly1d(fit_2)(elapsed_seconds)

        prediction_df = pd.DataFrame({
            "elapsed_sec": elapsed_seconds,
            "target": targets
        })
        predictions.append(prediction_df)

    concatenated_predictions = pd.concat(predictions).sort_values("elapsed_sec")
    return concatenated_predictions

def make_submission(predictions, test_elapsed_seconds):
    fitted_curve = interp1d(predictions["elapsed_sec"], predictions["target"], kind='linear')
    return fitted_curve(test_elapsed_seconds)

def main():
    TARGET_COLUMNS = ["x", "y", "z", "Vx", "Vy", "Vz"]
    SIM_COLUMNS = [col + "_sim" for col in TARGET_COLUMNS]

    # Load fitted parameters
    with open('fit_param_dictB2.dump', 'rb') as file:
        fit_param_dict = pickle.load(file)

    # Load test data
    test_data = pd.read_csv('test.csv', index_col=0)
    test_data['epoch'] = pd.to_datetime(test_data['epoch'])
    test_data["elapsed_sec"] = (test_data["epoch"] - START_DATE).dt.total_seconds()

    unique_sat_ids = test_data["sat_id"].unique()
    submission = test_data[SIM_COLUMNS].copy()
    submission.columns = TARGET_COLUMNS

    for sat_id in unique_sat_ids:
        satellite_data = test_data[test_data["sat_id"] == sat_id].sort_values("epoch")
        test_indices = satellite_data.index

        for target_column in TARGET_COLUMNS:
            predictions = predict(fit_param_dict[(sat_id, target_column)])
            interpolated_values = make_submission(predictions, satellite_data["elapsed_sec"].values)
            submission.loc[test_indices, target_column] = interpolated_values

    submission.to_csv('submission.csv', index=True)

if __name__ == "__main__":
    main()
