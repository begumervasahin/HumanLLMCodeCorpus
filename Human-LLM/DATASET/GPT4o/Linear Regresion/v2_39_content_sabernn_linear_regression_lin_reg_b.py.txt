import numpy as np
import csv
import matplotlib.pyplot as plt

# Function to compute the loss
def loss_func(inp, out, theta):
    m = len(inp[0])
    J = 0
    for i in range(m):
        y_hat = theta[0] * inp[0][i] + theta[1] * inp[1][i] + theta[2] * inp[2][i] + theta[3] * inp[3][i]
        y = out[i]
        J += (y_hat - y) ** 2
    return J / (2 * m)

# Function to compute the gradient of the loss
def loss_func_der(inp, out, theta, alpha):
    m = len(inp[0])
    dJ = [0, 0, 0, 0]
    for i in range(m):
        y_hat = theta[0] + theta[1] * inp[1][i] + theta[2] * inp[2][i] + theta[3] * inp[3][i]
        y = out[i]
        dJ[0] += alpha * (y - y_hat) * inp[0][i] / m
        dJ[1] += alpha * (y - y_hat) * inp[1][i] / m
        dJ[2] += alpha * (y - y_hat) * inp[2][i] / m
        dJ[3] += alpha * (y - y_hat) * inp[3][i] / m
    return dJ

# Standardization functions
def standardize(x):
    mean = np.average(x)
    std = np.std(x)
    return (x - mean) / std

def standardize_s(x0, x):
    mean = np.average(x)
    std = np.std(x)
    return (x0 - mean) / std

# Read data from CSV
x1, x2, x3, y = [], [], [], []
with open('data/KCSmall_NS2.csv', newline='') as f:
    reader = csv.reader(f)
    for row in reader:
        x1.append(float(row[0]))
        x2.append(float(row[1]))
        x3.append(float(row[2]))
        y.append(float(row[3]))

# Print first 5 rows of raw data
print("--------------------------------------------------------------------------------------------")
print("(1) First 5 rows of raw input data are:")
for i in range(5):
    print(f"{x1[i]}\t{x2[i]}\t{x3[i]}\t{y[i]}")

# Standardize data
x1s = standardize(x1)
x2s = standardize(x2)
x3s = standardize(x3)
xs = [[1.0] * len(x1s), x1s, x2s, x3s]

# Print first 5 rows of standardized data
print("--------------------------------------------------------------------------------------------")
print("(2) First 5 rows of standardized data are:")
for i in range(5):
    print(f"1\t{x1s[i]}\t{x2s[i]}\t{x3s[i]}\t{y[i]}")

# Initialize theta and calculate initial cost
theta1 = [0, 0, 0, 0]
print(f"The cost (J) value for theta = {theta1} is: J = {loss_func(xs, y, theta1)}")

# Gradient Descent
theta = [0, 0, 0, 0]
J = []
n = 50
alpha = 1.5

print("--------------------------------------------------------------------------------------------")
print("Theta \t\t\t\t\t\t\t\t Loss Function (J)")
print("--------------------------------------------------------------------------------------------")
for iter in range(n):
    dJ = loss_func_der(xs, y, theta, alpha)
    for j in range(4):
        theta[j] += dJ[j]
    Jtemp = loss_func(xs, y, theta)
    J.append(Jtemp)
    theta_r = [round(t, 5) for t in theta]
    Jtemp = round(Jtemp, 5)
    print(f"{theta_r} \t\t {Jtemp}")

# Test with new data
xtests = [1, standardize_s(3, x1), standardize_s(2000, x2), standardize_s(8550, x3)]
price = round((theta[0] + theta[1] * xtests[1] + theta[2] * xtests[2] + theta[3] * xtests[3]), 2)
xtest = [3, 2000, 8550]
print(f"Approximate house price for {xtest} is: ${price}")

# Plot the loss function
print("--------------------------------------------------------------------------------------------")
plt.title("J vs n")
plt.plot(range(1, n + 1), J)
plt.xlabel('Iterations')
plt.ylabel('Loss value (J)')
plt.grid(axis='both')
plt.show()
