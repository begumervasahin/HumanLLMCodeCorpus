import pandas as pd
import numpy as np

def data_preprocessing():
    df = pd.read_csv("iris.csv")
    target_dict = {species: i for i, species in enumerate(df['species'].unique())}
    df['target'] = df['species'].map(target_dict)
    return df

def data_shuffling(df):
    q = df.values
    np.random.shuffle(q)
    df = pd.DataFrame(q, columns=df.columns)
    X = df.iloc[:, :-2].values
    Y = df.iloc[:, 5].values
    return df, X, Y

def data_split(df, X, Y):
    total_records = len(X)
    split_ratio = 0.8
    train_len = int(total_records * split_ratio)
    X_train = X[:train_len]
    Y_train = Y[:train_len]
    X_test = X[train_len:]
    Y_test = Y[train_len:]
    return X_train, Y_train, X_test, Y_test

def init_parameters(lenw):
    w = np.random.rand(lenw)
    b = 0
    return w, b

def forward_prop(X, w, b):
    y_hat = np.dot(w, X) + b
    return y_hat

def cost_function(y_hat, y):
    m = len(y)
    J = (1 / (2 * m)) * np.sum(np.square(y_hat - y))
    return J

def back_prop(X, y, y_hat):
    m = len(y)
    dy_hat = (1 / m) * (y_hat - y)
    dw = np.dot(dy_hat, X)
    db = np.sum(dy_hat)
    return dw, db

def gradient_descent(w, b, dw, db, lr):
    w -= lr * dw
    b -= lr * db
    return w, b

def train(X_train, Y_train, epochs=10, lr=1e-3):
    w, b = init_parameters(X_train.shape[1])
    cost_history = []

    for epoch in range(epochs):
        for x_feed, y_feed in zip(X_train, Y_train):
            y_hat = forward_prop(x_feed, w, b)
            cost = cost_function(y_hat, y_feed)
            dw, db = back_prop(x_feed, y_feed, y_hat)
            w, b = gradient_descent(w, b, dw, db, lr)

        if epoch % 2 == 0:
            print(f"Epoch {epoch} Cost: {cost:.4f}")

    print(f"Training complete. w: {w}, b: {b}")
    return w, b

def predict(X_test, Y_test, w, b):
    predictions = [np.round(forward_prop(x, w, b)) for x in X_test]
    accuracy = np.mean([pred == y for pred, y in zip(predictions, Y_test)]) * 100
    for pred, actual in zip(predictions, Y_test):
        print(f"Predicted: {pred}, Actual: {actual}")
    return accuracy

def data_sampler(X, Y, i, fold):
    length_data = len(X)
    partition_size = length_data // fold

    test_start = i * partition_size
    test_stop = (i + 1) * partition_size

    X_test = X[test_start:test_stop]
    Y_test = Y[test_start:test_stop]
    X_train = np.concatenate((X[:test_start], X[test_stop:]))
    Y_train = np.concatenate((Y[:test_start], Y[test_stop:]))

    return X_train, Y_train, X_test, Y_test

def cross_validation(X, Y, fold=5, epochs=10, lr=1e-3):
    ACC_dict = {}
    for i in range(fold):
        X_train, Y_train, X_test, Y_test = data_sampler(X, Y, i, fold)
        w, b = train(X_train, Y_train, epochs, lr)
        acc = predict(X_test, Y_test, w, b)
        ACC_dict[i] = acc
    print("Cross-validation results:", ACC_dict)

def main():
    df = data_preprocessing()
    df, X, Y = data_shuffling(df)
    X_train, Y_train, X_test, Y_test = data_split(df, X, Y)

    print("Training and predicting results for the test data using the trained model:")
    w, b = train(X_train, Y_train)
    acc = predict(X_test, Y_test, w, b)
    print(f"Accuracy: {acc:.2f}%")

    print("\nCross-validation for the trained model")
    cross_validation(X, Y)

if __name__ == "__main__":
    main()
