import numpy as np
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error
from numpy.linalg import inv

# Training and test data
x_train = np.array([[0], [0.5], [2]])
y_train = [-1.25, -0.6, -4.85]
x_test = np.array([[-1], [1], [3]])
y_test = [-5.2, -0.9, -13]

# Function to perform least squares estimation
def least_squares_estimator(X, Y):
    X_with_intercept = np.hstack((np.ones((len(X), 1)), X))
    X_transpose = X_with_intercept.T
    XTX = np.matmul(X_transpose, X_with_intercept)
    XTX_inv = inv(XTX)
    XTY = np.matmul(X_transpose, Y)
    return np.matmul(XTX_inv, XTY)

# Function to plot data and models
def plot_models(x_train, y_train, x_test, y_test, lr_model, pr_model, gammas):
    # Plot Linear Regression
    x_range = np.linspace(-3, 5, 100)
    linear_y = lr_model.coef_[0] * x_range + lr_model.intercept_
    plt.plot(x_range, linear_y, '--', color='black', label='Linear Regression Line')
    
    # Plot Polynomial Regression
    poly_y = gammas[0] + gammas[1] * x_range + gammas[2] * (x_range ** 2)
    plt.plot(x_range, poly_y, '--', color='red', label='Polynomial Regression Curve')
    
    # Plot Training and Test Data
    plt.scatter(x_train, y_train, marker='*', label='Train Data')
    plt.scatter(x_test, y_test, marker='o', label='Test Data')
    plt.xlabel('X axis')
    plt.ylabel('Y axis')
    plt.title('Polynomial Regression vs. Linear Regression')
    plt.legend()
    plt.show()

# Linear Regression model
linear_model = LinearRegression()
linear_model.fit(x_train, y_train)

# Extract coefficients for Linear Regression
linear_coef = linear_model.coef_[0]
linear_intercept = linear_model.intercept_
print(f'Linear Regression Intercept (w0) = {linear_intercept}')
print(f'Linear Regression Coefficient (w1) = {linear_coef}')

# Predictions for Linear Regression
y_train_pred_linear = linear_model.predict(x_train)
y_test_pred_linear = linear_model.predict(x_test)
print('Predicted values for x_test (Linear):', y_test_pred_linear)

# Mean Squared Error for Linear Regression
train_mse_linear = mean_squared_error(y_train, y_train_pred_linear)
test_mse_linear = mean_squared_error(y_test, y_test_pred_linear)
print(f'Train MSE (Linear): {train_mse_linear}')
print(f'Test MSE (Linear): {test_mse_linear}')

# Polynomial features
z_train = np.hstack((x_train, np.power(x_train, 2)))
z_test = np.hstack((x_test, np.power(x_test, 2)))

# Least Squares Estimators for Polynomial Regression
gammas = least_squares_estimator(z_train, y_train)
gamma0, gamma1, gamma2 = gammas
print(f'\nThe LS estimators (gamma) are: {gammas}')

# Polynomial Regression model
poly_model = LinearRegression()
poly_model.fit(z_train, y_train)

# Predictions for Polynomial Regression
y_train_pred_poly = poly_model.predict(z_train)
y_test_pred_poly = poly_model.predict(z_test)

# Mean Squared Error for Polynomial Regression
train_mse_poly = mean_squared_error(y_train, y_train_pred_poly)
test_mse_poly = mean_squared_error(y_test, y_test_pred_poly)
print(f'Train MSE (Polynomial): {train_mse_poly}')
print(f'Test MSE (Polynomial): {test_mse_poly}')

# Plot models
plot_models(x_train, y_train, x_test, y_test, linear_model, poly_model, gammas)
