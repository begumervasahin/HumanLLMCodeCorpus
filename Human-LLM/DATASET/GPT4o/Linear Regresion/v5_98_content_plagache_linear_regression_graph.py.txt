import matplotlib.pyplot as plt
import pandas as pd
import numpy as np
import matplotx
import os
import imageio
import argparse

# Argument parser setup
def setup_argument_parser():
    """Setup the argument parser."""
    parser = argparse.ArgumentParser(
        prog='Graph',
        description='Plot different graphs',
        exit_on_error=False
    )
    parser.add_argument('-p', '--polyfit',
                        help='Polyfit example',
                        action='store_true',
                        required=False)
    parser.add_argument('-l', '--linear',
                        help='My linear function',
                        action='store_true',
                        required=False)
    parser.add_argument('-g', '--gif',
                        help='Create Animated Gif',
                        action='store_true',
                        required=False)
    return parser.parse_args()

# Load data from CSV files
def load_data():
    """Load data from CSV files."""
    data = pd.read_csv('data.csv')
    theta_values = pd.read_csv('thetas.csv')
    return data, theta_values

# Extract data for plotting
def extract_plot_data(data, theta_values):
    """Extract data for plotting."""
    x_values = data["km"]
    y_values = data["price"]
    theta_0 = theta_values["theta0"].values[-1]
    theta_1 = theta_values["theta1"].values[-1]
    return x_values, y_values, theta_0, theta_1, len(theta_values)

# Setup base graph
def base_graph(xs, ys):
    """Setup the base graph with scatter plot of data points."""
    plt.style.use(matplotx.styles.gruvbox["dark"])
    plt.xlabel('km')
    plt.ylabel('price')
    plt.scatter(xs, ys)

# Plot linear regression
def plot_linear_regression(theta_0, theta_1, x_min, x_max, label, fmt):
    """Plot a linear regression line based on given theta values."""
    x = np.linspace(x_min, x_max, NUM_POINTS_TO_PLOT)
    y = theta_1 * x + theta_0
    plt.plot(x, y, fmt, label=label)

# Plot polyfit example
def polyfit_example(x_values, y_values):
    """Plot an example using numpy's polyfit."""
    model = np.polyfit(x_values, y_values, 1)
    plot_linear_regression(model[1], model[0], x_values.min(), x_values.max(), "Polyfit Example", "r:.")  

# Create directory if it doesn't exist
def create_directory(path):
    """Create a directory if it doesn't exist."""
    if not os.path.exists(path):
        os.makedirs(path)

# Create PNG files for each step of the linear regression animation
def create_png_files(x_values, y_values, theta_values, theta_count):
    """Create PNG files for each step of the linear regression animation."""
    images = []
    total_images = 120
    create_directory("png")
    for index in range(1, total_images):
        step = int(theta_count / total_images)
        theta_0 = theta_values["theta0"].values[index * step]
        theta_1 = theta_values["theta1"].values[index * step]
        base_graph(x_values, y_values)
        plot_linear_regression(theta_0, theta_1, x_values.min(), x_values.max(), "My Linear Regression", "g")
        filename = f'png/{index}.png'
        images.append(filename)
        plt.savefig(filename)
        plt.close()
    base_graph(x_values, y_values)
    plot_linear_regression(theta_0, theta_1, x_values.min(), x_values.max(), "My Linear Regression", "g") 
    polyfit_example(x_values, y_values)
    filename = f'png/{total_images + 1}.png'
    images.append(filename)
    plt.legend()
    plt.savefig(filename)
    plt.close()
    return images

# Create an animated GIF from PNG files
def create_animated_gif(x_values, y_values, theta_values, theta_count):
    """Create an animated GIF from PNG files."""
    images = create_png_files(x_values, y_values, theta_values, theta_count)
    frames = [imageio.v3.imread(image) for image in images]
    imageio.mimsave("training.gif", frames, duration=50)

# Main execution
def main():
    args = setup_argument_parser()
    data, theta_values = load_data()
    x_values, y_values, theta_0, theta_1, theta_count = extract_plot_data(data, theta_values)
    
    base_graph(x_values, y_values)
    
    if args.linear:
        plot_linear_regression(theta_0, theta_1, x_values.min(), x_values.max(), "My Linear Regression", "g") 
    
    if args.polyfit:
        polyfit_example(x_values, y_values)
    
    plt.legend()
    
    if args.gif:
        create_animated_gif(x_values, y_values, theta_values, theta_count)
    
    plt.show()

if __name__ == "__main__":
    main()
