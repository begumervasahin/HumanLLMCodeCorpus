import math
import pandas as pd
from sklearn.model_selection import train_test_split
import numpy as np
from sklearn.metrics import confusion_matrix, accuracy_score, classification_report

# Define the cost function
def cost_func(theta, X, y):
    z = np.dot(X, theta)
    hyp = 1.0 / (1.0 + np.exp(-z))
    cost = np.mean(-(y * np.log(hyp)) - ((1 - y) * np.log(1 - hyp)))
    return cost

# Define the gradient descent function
def gradient_descent(theta, X, y, alpha=0.001, iterations=400):
    for _ in range(iterations):
        z = np.dot(X, theta)
        hyp = 1.0 / (1.0 + np.exp(-z))
        gradient = np.dot(X.T, (hyp - y))
        theta -= (alpha * gradient) / len(X)
    return theta

# Define the feature scaling function
def feature_scaling(data):
    scaled_data = data.copy()
    scaled_data[:, :-1] = (data[:, :-1] - np.mean(data[:, :-1], axis=0)) / (np.max(data[:, :-1], axis=0) - np.min(data[:, :-1], axis=0))
    return scaled_data

# Main execution
if __name__ == '__main__':
    # Load and prepare the dataset
    complete_data = pd.read_csv("BSOM_DataSet_for_HW2.csv", usecols=["all_mcqs_avg_n20", "all_NBME_avg_n4", "LEVEL"])
    data_array = complete_data.to_numpy()
    
    # Scale features
    scaled_data = feature_scaling(data_array)
    
    # Add bias term
    data_with_bias = np.hstack((np.ones((scaled_data.shape[0], 1)), scaled_data))
    
    # Split features and labels
    X = data_with_bias[:, :-1]
    y = data_with_bias[:, -1]
    
    # Split the data into training and test sets
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.33, random_state=27)
    
    # Reshape y for compatibility
    y_train = y_train[:, np.newaxis]
    y_test = y_test[:, np.newaxis]
    
    labels = ["A", "B", "C", "D"]
    final_theta = []

    # Train the model for each label
    for label in labels:
        binary_y_train = np.where(y_train == label, 1, 0)
        theta = np.random.rand(X_train.shape[1], 1)
        
        initial_cost = cost_func(theta, X_train, binary_y_train)
        print(f"Initial cost for label {label}: {initial_cost:.4f}")
        
        theta = gradient_descent(theta, X_train, binary_y_train)
        final_cost = cost_func(theta, X_train, binary_y_train)
        print(f"Final cost for label {label}: {final_cost:.4f}")
        
        final_theta.append(theta.flatten())

    # Evaluate the model
    theta_matrix = np.array(final_theta).T
    z = np.dot(X_test, theta_matrix)
    hyp = 1.0 / (1.0 + np.exp(-z))
    predictions = np.argmax(hyp, axis=1)
    print("Predicted values:", predictions)
    
    actual_labels = [labels.index(label) for label in y_test.flatten()]
    print("Actual values:", actual_labels)
    
    # Confusion matrix and evaluation metrics
    conf_matrix = confusion_matrix(actual_labels, predictions)
    print("Confusion matrix:\n", conf_matrix)
    
    accuracy = accuracy_score(actual_labels, predictions)
    print("Accuracy:", accuracy)
    
    print("Classification report:\n", classification_report(actual_labels, predictions, target_names=labels))
