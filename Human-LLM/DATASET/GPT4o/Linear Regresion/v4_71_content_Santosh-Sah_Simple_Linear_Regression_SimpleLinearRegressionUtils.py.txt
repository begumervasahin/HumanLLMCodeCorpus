import pandas as pd
import pickle
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler

def import_linear_regression_dataset(file_name):
    """
    Imports the dataset, reads specific columns, and splits the dataset into training and testing sets.

    Args:
    file_name (str): The name of the file containing the dataset.

    Returns:
    tuple: Split training and testing sets (X_train, X_test, y_train, y_test).
    """
    dataset = pd.read_csv(file_name)
    X = dataset.iloc[:, :-1].values
    y = dataset.iloc[:, -1].values
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=0)
    return X_train, X_test, y_train, y_test

def save_standard_scaler(scaler, file_name="StandardScaler.pkl"):
    """
    Saves the StandardScaler object as a pickle file.

    Args:
    scaler (StandardScaler): The StandardScaler object to save.
    file_name (str): The name of the pickle file.
    """
    with open(file_name, 'wb') as file:
        pickle.dump(scaler, file, protocol=pickle.HIGHEST_PROTOCOL)

def load_standard_scaler(file_name="StandardScaler.pkl"):
    """
    Loads the StandardScaler object from a pickle file.

    Args:
    file_name (str): The name of the pickle file.

    Returns:
    StandardScaler: The loaded StandardScaler object.
    """
    with open(file_name, 'rb') as file:
        return pickle.load(file)

def save_dataset_splits(X_train, X_test, y_train, y_test):
    """
    Saves the training and testing datasets as pickle files.

    Args:
    X_train (array-like): Training data.
    X_test (array-like): Testing data.
    y_train (array-like): Training labels.
    y_test (array-like): Testing labels.
    """
    with open("X_train.pkl", 'wb') as file:
        pickle.dump(X_train, file, protocol=pickle.HIGHEST_PROTOCOL)
    with open("X_test.pkl", 'wb') as file:
        pickle.dump(X_test, file, protocol=pickle.HIGHEST_PROTOCOL)
    with open("y_train.pkl", 'wb') as file:
        pickle.dump(y_train, file, protocol=pickle.HIGHEST_PROTOCOL)
    with open("y_test.pkl", 'wb') as file:
        pickle.dump(y_test, file, protocol=pickle.HIGHEST_PROTOCOL)

def save_model(model, file_name="LinearRegressionModel.pkl"):
    """
    Saves the Linear Regression model as a pickle file.

    Args:
    model (object): The Linear Regression model to save.
    file_name (str): The name of the pickle file.
    """
    with open(file_name, 'wb') as file:
        pickle.dump(model, file, protocol=pickle.HIGHEST_PROTOCOL)
