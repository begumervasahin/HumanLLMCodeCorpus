# Importing the necessary libraries
import pandas as pd
import pickle
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler

def import_linear_regression_dataset(file_name):
    """
    Import the dataset and split it into training and testing sets.
    
    Parameters:
    file_name (str): The path to the CSV file containing the dataset.

    Returns:
    tuple: Containing arrays of X_train, X_test, y_train, y_test
    """
    dataset = pd.read_csv(file_name)
    X = dataset.iloc[:, :-1].values
    y = dataset.iloc[:, 1].values
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=0)
    return X_train, X_test, y_train, y_test

def save_standard_scaler(scaler_file_name="StandardScaler.pkl"):
    """
    Save the StandardScaler object as a pickle file.

    Parameters:
    scaler_file_name (str): The file name for the saved StandardScaler object.
    """
    scaler = StandardScaler()
    with open(scaler_file_name, 'wb') as file:
        pickle.dump(scaler, file)

def save_dataset(X_train, X_test, y_train, y_test, prefix=""):
    """
    Save the training and testing datasets as pickle files.

    Parameters:
    X_train, X_test, y_train, y_test: Arrays containing the training and testing data.
    prefix (str): Prefix for the pickle file names.
    """
    datasets = {
        f"{prefix}X_train.pkl": X_train,
        f"{prefix}X_test.pkl": X_test,
        f"{prefix}y_train.pkl": y_train,
        f"{prefix}y_test.pkl": y_test
    }
    for file_name, data in datasets.items():
        with open(file_name, 'wb') as file:
            pickle.dump(data, file)

def save_model(model, model_file_name="LinearRegressionModel.pkl"):
    """
    Save the linear regression model as a pickle file.

    Parameters:
    model: The linear regression model to be saved.
    model_file_name (str): The file name for the saved model.
    """
    with open(model_file_name, 'wb') as file:
        pickle.dump(model, file)

def load_standard_scaler(scaler_file_name="StandardScaler.pkl"):
    """
    Load the StandardScaler object from a pickle file.

    Parameters:
    scaler_file_name (str): The file name of the saved StandardScaler object.

    Returns:
    StandardScaler object
    """
    with open(scaler_file_name, 'rb') as file:
        scaler = pickle.load(file)
    return scaler

def load_model(model_file_name="LinearRegressionModel.pkl"):
    """
    Load the linear regression model from a pickle file.

    Parameters:
    model_file_name (str): The file name of the saved model.

    Returns:
    The loaded linear regression model.
    """
    with open(model_file_name, 'rb') as file:
        model = pickle.load(file)
    return model

def load_dataset(prefix=""):
    """
    Load the training and testing datasets from pickle files.

    Parameters:
    prefix (str): Prefix for the pickle file names.

    Returns:
    tuple: Containing arrays of X_train, X_test, y_train, y_test
    """
    file_names = [
        f"{prefix}X_train.pkl",
        f"{prefix}X_test.pkl",
        f"{prefix}y_train.pkl",
        f"{prefix}y_test.pkl"
    ]
    datasets = []
    for file_name in file_names:
        with open(file_name, 'rb') as file:
            datasets.append(pickle.load(file))
    return tuple(datasets)
