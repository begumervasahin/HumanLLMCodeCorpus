import os
import csv
import numpy as np
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression

def load_data(file_path):
    """
    Load stock data from a CSV file.

    Parameters:
    - file_path: str, path to the CSV file.

    Returns:
    - tuple: (list of high prices, list of low prices, list of close prices)
    """
    high_price, low_price, close_price = [], [], []

    with open(file_path, 'r', newline="", encoding='utf-8') as csvfile:
        rows = csv.reader(csvfile)
        next(rows)  # Skip the header row

        for row in rows:
            high_price.append(float(row[2]))
            low_price.append(float(row[3]))
            close_price.append(float(row[4]))

    return high_price, low_price, close_price

def get_stock_data(directory_path):
    """
    Load stock data from all CSV files in a directory.

    Parameters:
    - directory_path: str, path to the directory containing CSV files.

    Returns:
    - list of tuples: each containing high prices, low prices, and close prices for each stock
    """
    all_prices = []
    filenames = sorted(os.listdir(directory_path))

    for filename in filenames:
        file_path = os.path.join(directory_path, filename)
        all_prices.append(load_data(file_path))

    return all_prices

def split_data(stock_prices):
    """
    Split data into features and targets for model training.

    Parameters:
    - stock_prices: tuple of lists, containing high, low, and close prices.

    Returns:
    - tuple: (feature set, target set, real features for prediction, real target values)
    """
    high_price, low_price, close_price = stock_prices
    X, Y = [], []
    real_x = []
    num_rounds = len(high_price) // 13

    for run in range(num_rounds):
        features = []
        target = []

        for i in range(run * 13, run * 13 + 3):
            features.extend([high_price[i], low_price[i], close_price[i]])

        for i in range(run * 13 + 3, run * 13 + 13):
            target.append(close_price[i])

        X.append(np.array(features))
        Y.append(np.array(target))

    for i in range(len(close_price) - 13, len(close_price) - 10):
        real_x.extend([high_price[i], low_price[i], close_price[i]])

    real_y = close_price[-10:]
    return X, Y, real_x, real_y

def plot_stock_predictions(stock_names, colors, predict_colors, all_prices):
    """
    Train models and plot stock price predictions.

    Parameters:
    - stock_names: list of str, names of the stocks.
    - colors: list of str, colors for actual stock prices.
    - predict_colors: list of str, colors for predicted stock prices.
    - all_prices: list of tuples, each containing high, low, and close prices for each stock.
    """
    days = list(range(1, 11))
    plt.figure()

    for i, stock_name in enumerate(stock_names):
        stock_prices = all_prices[i]
        X, Y, real_x, real_y = split_data(stock_prices)
        
        model = LinearRegression()
        model.fit(X, Y)
        predicted_y = model.predict([real_x])[0]

        plt.plot(days, real_y, colors[i], label=f"{stock_name}-Real")
        plt.plot(days, predicted_y, predict_colors[i], label=f"{stock_name}-Predict")

    plt.legend()
    plt.show()

def main():
    """
    Main function to load data, train models, and plot results.
    """
    directory_path = 'stock_data'
    all_prices = get_stock_data(directory_path)

    stock_names = ["YFY", "TSMC", "MTK"]
    colors = ['b', 'r', 'm']
    predict_colors = ['g', 'c', 'y']

    plot_stock_predictions(stock_names, colors, predict_colors, all_prices)

if __name__ == "__main__":
    main()
