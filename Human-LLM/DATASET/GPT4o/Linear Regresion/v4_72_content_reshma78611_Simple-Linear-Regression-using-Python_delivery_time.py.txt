import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import statsmodels.formula.api as smf
from sklearn.metrics import mean_squared_error
from math import sqrt

# Load data
data_path = 'C:/Users/HP/Desktop/assignments submission/simple linear regression/delivery_time.csv'
delv_time = pd.read_csv(data_path)

# Scatter plot of delivery time vs sorting time
plt.figure(figsize=(10, 6))
plt.scatter(delv_time.Sorting_time, delv_time.Delivery_time, color='r')
plt.xlabel('Sorting Time')
plt.ylabel('Delivery Time')
plt.title('Scatter plot of Delivery Time vs Sorting Time')
plt.show()

# Calculate correlation coefficient
correlation = np.corrcoef(delv_time.Sorting_time, delv_time.Delivery_time)[0, 1]
print(f"Correlation coefficient: {correlation}")

# Linear Regression Model
lin_model = smf.ols('Delivery_time ~ Sorting_time', data=delv_time).fit()
print(lin_model.summary())

# Predictions and errors
lin_predict = lin_model.predict(delv_time)
lin_error = delv_time.Delivery_time - lin_predict

# Plot linear regression
plt.figure(figsize=(10, 6))
plt.scatter(delv_time.Sorting_time, delv_time.Delivery_time, color='r')
plt.plot(delv_time.Sorting_time, lin_predict, color='b')
plt.xlabel('Sorting Time')
plt.ylabel('Delivery Time')
plt.title('Linear Regression: Delivery Time vs Sorting Time')
plt.show()

# Confidence intervals and RMSE for linear model
lin_conf_int = lin_model.conf_int(0.05)
lin_rmse = sqrt(mean_squared_error(delv_time.Delivery_time, lin_predict))
lin_corr = np.corrcoef(delv_time.Delivery_time, lin_predict)[0, 1]
print(f"Linear model RMSE: {lin_rmse}, Correlation: {lin_corr}")

# Log-Linear Regression Model
log_model = smf.ols('Delivery_time ~ np.log(Sorting_time)', data=delv_time).fit()
print(log_model.summary())

# Predictions and errors for log model
log_predict = log_model.predict(delv_time)
log_error = delv_time.Delivery_time - log_predict

# Plot log-linear regression
plt.figure(figsize=(10, 6))
plt.scatter(delv_time.Sorting_time, delv_time.Delivery_time, color='r')
plt.plot(delv_time.Sorting_time, log_predict, color='b')
plt.xlabel('Sorting Time')
plt.ylabel('Delivery Time')
plt.title('Log-Linear Regression: Delivery Time vs Sorting Time')
plt.show()

# Confidence intervals and RMSE for log model
log_conf_int = log_model.conf_int(0.05)
log_rmse = sqrt(mean_squared_error(delv_time.Delivery_time, log_predict))
log_corr = np.corrcoef(delv_time.Delivery_time, log_predict)[0, 1]
print(f"Log model RMSE: {log_rmse}, Correlation: {log_corr}")

# Exponential Regression Model
exp_model = smf.ols('np.log(Delivery_time) ~ Sorting_time', data=delv_time).fit()
print(exp_model.summary())

# Predictions and errors for exponential model
exp_predict_log = exp_model.predict(delv_time)
exp_predict = np.exp(exp_predict_log)
exp_error = delv_time.Delivery_time - exp_predict

# Plot exponential regression
plt.figure(figsize=(10, 6))

plt.ylabel('Delivery Time')

# Confidence intervals and RMSE for exponential model
exp_conf_int = exp_model.conf_int(0.05)
exp_rmse = sqrt(mean_squared_error(delv_time.Delivery_time, exp_predict))

# Quadratic Regression Model
delv_time['sq_sort_time'] = delv_time.Sorting_time ** 2
quad_model = smf.ols('Delivery_time ~ Sorting_time + sq_sort_time', data=delv_time).fit()
print(quad_model.summary())

# Predictions and errors for quadratic model
quad_predict = quad_model.predict(delv_time)

# Plot quadratic regression

plt.ylabel('Delivery Time')
plt.title('Quadratic Regression: Delivery Time vs Sorting Time')
plt.show()

# Confidence intervals and RMSE for quadratic model
quad_conf_int = quad_model.conf_int(0.05)
