import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
import statsmodels.formula.api as smf
import statsmodels.api as sm
from sklearn.metrics import mean_squared_error
from math import sqrt

# Load the data
corolla_data = pd.read_csv('C:/Users/HP/Desktop/assignments submission/multi linear regression/ToyotaCorolla.csv', encoding="ISO-8859-1")

# Select relevant columns
Toyota = corolla_data[["Price", "Age_08_04", "KM", "HP", "cc", "Doors", "Gears", "Quarterly_Tax", "Weight"]]

# Check for missing values
print("Missing values in each column:\n", Toyota.isna().sum())

# Drop rows with missing values (if any)
Toyota = Toyota.dropna()

# Pairplot to visualize relationships between variables
sns.pairplot(Toyota)
plt.show()

# Compute correlation matrix
print("Correlation matrix:\n", Toyota.corr())

# Fit initial regression model
toyota_model = smf.ols('Price~Age_08_04+KM+HP+cc+Doors+Gears+Quarterly_Tax+Weight', data=Toyota).fit()
print("Initial model parameters:\n", toyota_model.params)
print("Initial model summary:\n", toyota_model.summary())

# Calculate VIF for each predictor
def calculate_vif(data, predictors):
    vif_data = pd.DataFrame()
    vif_data["variables"] = predictors
    vif_data["VIF"] = [1 / (1 - smf.ols(f'{predictor}~' + '+'.join([p for p in predictors if p != predictor]), data=data).fit().rsquared) for predictor in predictors]
    return vif_data

predictors = ["Age_08_04", "KM", "HP", "cc", "Doors", "Gears", "Quarterly_Tax", "Weight"]
vif_frame = calculate_vif(Toyota, predictors)
print("VIF DataFrame:\n", vif_frame)

# Plot partial regression plots
sm.graphics.plot_partregress_grid(toyota_model)
plt.show()

# Plot influence plot
sm.graphics.influence_plot(toyota_model)
plt.show()

# Remove influential points and refit model
influential_points = [80, 221, 960, 601]
Toyota_filtered = Toyota.drop(Toyota.index[influential_points], axis=0)

# Refit model after removing influential points
toyota_model2 = smf.ols('Price~Age_08_04+KM+HP+cc+Doors+Gears+Quarterly_Tax+Weight', data=Toyota_filtered).fit()
print("Model parameters after removing influential points:\n", toyota_model2.params)
print("Model summary after removing influential points:\n", toyota_model2.summary())

# Plot partial regression plots for the updated model
sm.graphics.plot_partregress_grid(toyota_model2)
plt.show()

# Plot influence plot for the updated model
sm.graphics.influence_plot(toyota_model2)
plt.show()

# Test different transformations on 'cc' and 'Doors'
toyota_model3 = smf.ols('Price~Age_08_04+KM+HP+np.log(cc)+np.log(Doors)+Gears+Quarterly_Tax+Weight', data=Toyota_filtered).fit()
print("Model parameters with log transformations:\n", toyota_model3.params)
print("Model summary with log transformations:\n", toyota_model3.summary())

# Test adding polynomial terms
Toyota_filtered['cc_sq'] = Toyota_filtered.cc ** 2
Toyota_filtered['Door_sq'] = Toyota_filtered.Doors ** 2
toyota_model4 = smf.ols('Price~Age_08_04+KM+HP+cc+Doors+Gears+Quarterly_Tax+Weight+Door_sq', data=Toyota_filtered).fit()
print("Model parameters with polynomial terms:\n", toyota_model4.params)
print("Model summary with polynomial terms:\n", toyota_model4.summary())

# Predict and calculate RMSE
Toyota_predict = toyota_model4.predict(Toyota_filtered)
Toyota_rmse = sqrt(mean_squared_error(Toyota_filtered.Price, Toyota_predict))
print("RMSE of the final model:", Toyota_rmse)
