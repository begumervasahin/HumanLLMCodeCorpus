import pandas as pd
import numpy as np

class Dataset:
    """
    A class to store feature (X) and target (Y) data for a dataset.
    """
    def __init__(self, X, Y):
        self._X = X
        self._Y = Y

    @property
    def X(self):
        return self._X

    @property
    def Y(self):
        return self._Y

    def get_size(self):
        """
        Return the size of the dataset.
        """
        return len(self._X)


class Datasets:
    """
    A base class to handle dataset loading and splitting into training and validation sets.
    """
    def __init__(self, train_frac):
        self.train_frac = train_frac
        self.train = None
        self.val = None
        self.whole = None

    def load(self):
        """
        Load the dataset, shuffle it, and split into training and validation sets.
        """
        self.df = pd.read_csv(self.data_path)
        feature_columns = self.df.columns.drop(self.target_columns)
        X = self.df[feature_columns].values
        Y = self.df[self.target_columns].values
        self.whole = Dataset(X, Y)

        # Shuffle and split the dataset
        num_samples = len(X)
        all_indices = np.arange(num_samples)
        np.random.shuffle(all_indices)
        train_indices = all_indices[:int(self.train_frac * num_samples)]
        val_indices = all_indices[int(self.train_frac * num_samples):]

        self.train = Dataset(X[train_indices], Y[train_indices])
        self.val = Dataset(X[val_indices], Y[val_indices])

    def get_df(self):
        """
        Return the loaded DataFrame.
        """
        return self.df

    def get_size(self):
        """
        Return the size of the DataFrame.
        """
        return len(self.df)

    def get_train(self):
        """
        Return the training dataset.
        """
        if self.train is None:
            raise ValueError("Data not loaded")
        return self.train

    def get_val(self):
        """
        Return the validation dataset.
        """
        if self.val is None:
            raise ValueError("Validation data not loaded")
        return self.val

class InitialDataSet(Datasets):
    """
    A class to handle creating new datasets with specific transformations.
    """
    def __init__(self):
        pass

    def load_inv_flip(self, data_tr, num_samples):
        """
        Load an Inverse Flip dataset randomly from training data.
        """
        train_data_size = data_tr.get_size()
        assert num_samples < train_data_size, "num_samples must be smaller than the size of the training data"

        selected_indices = np.random.choice(train_data_size, size=num_samples, replace=False)
        self.X = data_tr.whole.X[selected_indices]
        self.Y = 1 - data_tr.whole.Y[selected_indices]

        self.whole = Dataset(self.X, self.Y)

    def load_b_flip(self, data_tr, num_samples):
        """
        Load a Boundary Flip dataset randomly from training data.
        """
        train_data_size = data_tr.get_size()
        assert num_samples < train_data_size, "num_samples must be smaller than the size of the training data"

        selected_indices = np.random.choice(train_data_size, size=num_samples, replace=False)
        self.X = data_tr.whole.X[selected_indices]
        self.Y = np.round(1 - data_tr.whole.Y[selected_indices])

        self.whole = Dataset(self.X, self.Y)

    def load_random(self, num_samples, mean, variance):
        """
        Load random data points sampled from a multivariate normal distribution.
        """
        self.X, self.Y = np.random.multivariate_normal(mean, variance, num_samples).T
        self.whole = Dataset(self.X, self.Y)

    def get_size(self):
        """
        Return the size of the dataset.
        """
        return len(self.X)

    def get_df(self):
        """
        Returns a DataFrame constructed from the dataset.
        """
        data = np.hstack((self.X, self.Y.reshape(-1, 1)))
        columns = [f"Feature_{i}" for i in range(self.X.shape[1])] + ["Target"]
        return pd.DataFrame(data, columns=columns)
