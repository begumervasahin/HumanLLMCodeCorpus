"""
Linear Regression Implementation from Scratch on Housing Data
"""

import numpy as np
import matplotlib.pyplot as plt

# Load data from the file
datafile = 'Housing_data.txt'
cols = np.loadtxt(datafile, delimiter=',', usecols=(0,1,2,3), unpack=True)

# Prepare the data matrices
X = np.transpose(np.array(cols[:-1]))
y = np.transpose(np.array(cols[-1:]))
m = y.size  # Number of training examples

# Add intercept term to X
X = np.insert(X, 0, 1, axis=1)

# Print shapes and statistics for debugging
print("Shape of X:", X.shape)
print("Shape of y:", y.shape)
print("Mean of first feature (before normalization):", np.mean(X[:,1]))
print("Standard deviation of first feature (before normalization):", np.std(X[:,1]))

# Normalize features
stored_feature_means, stored_feature_stds = [], []
Xnorm = X.copy()

for col in range(1, Xnorm.shape[1]):
    mean = np.mean(Xnorm[:, col])
    std = np.std(Xnorm[:, col])
    stored_feature_means.append(mean)
    stored_feature_stds.append(std)
    Xnorm[:, col] = (Xnorm[:, col] - mean) / std

# Print normalized statistics for debugging
print("Mean of first feature (after normalization):", np.mean(Xnorm[:, 1]))
print("Standard deviation of first feature (after normalization):", np.std(Xnorm[:, 1]))
print("Stored feature means:", stored_feature_means)
print("Stored feature standard deviations:", stored_feature_stds)

# Hypothesis function
def hypothesis(X, theta):
    return np.dot(X, theta)

# Cost function
def compute_cost(theta, X, y):
    pred = hypothesis(X, theta)
    sqr_error = (pred - y) ** 2
    J = 1 / (2 * m) * sum(sqr_error)
    return J[0]

# Initial theta (weights)
initial_theta = np.zeros((X.shape[1], 1))

# Print initial cost
print("Initial cost with theta initialized to zero:")
print(compute_cost(initial_theta, X, y))

# Gradient Descent function
def gradient_descent(X, y, theta, num_iters, alpha):
    J_history = np.zeros((num_iters, 1))
    for i in range(num_iters):
        errors = hypothesis(X, theta) - y
        theta -= (alpha / m) * np.dot(X.T, errors)
        J_history[i] = compute_cost(theta, X, y)
    return theta, J_history

# Parameters for gradient descent
iterations = 1500
alpha = 0.01
initial_theta = np.zeros((Xnorm.shape[1], 1))

# Run gradient descent
theta, J_history = gradient_descent(Xnorm, y, initial_theta, iterations, alpha)

# Print updated theta values
print("Updated values of theta:")
print(theta)

# Plot convergence graph
def plot_convergence(J_history):
    plt.figure(figsize=(10, 6))
    plt.plot(range(len(J_history)), J_history, 'bo')
    plt.grid(True)
    plt.title("Convergence of Cost Function")
    plt.xlabel("Iteration number")
    plt.ylabel("Cost function")
    plt.xlim([-0.05 * iterations, 1.05 * iterations])
    plt.show()

plot_convergence(J_history)

# Prediction function
def predict(features):
    print(f"Predicting price for house with {features[0]} square feet, {features[1]} bedrooms, and {features[2]} years old:")
    normalized_features = [(features[i] - stored_feature_means[i]) / stored_feature_stds[i] for i in range(len(features))]
    normalized_features.insert(0, 1)
    predicted_price = hypothesis(normalized_features, theta)
    print(f"Predicted price: ${predicted_price[0]:.2f}")

# Test prediction
test_features = np.array([1203, 3, 11])
predict(test_features)
