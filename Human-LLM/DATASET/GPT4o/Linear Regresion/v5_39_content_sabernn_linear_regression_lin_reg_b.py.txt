import numpy as np
import csv
import matplotlib.pyplot as plt

# Define the loss function
def loss_func(inp, out, theta):
    m = len(inp[0])
    J = 0
    for i in range(m):
        y_hat = sum(theta[j] * inp[j][i] for j in range(4))
        J += (y_hat - out[i])**2
    return J / (2 * m)

# Define the derivative of the loss function
def loss_func_der(inp, out, theta, alpha):
    m = len(inp[0])
    dJ = [0] * 4
    for i in range(m):
        y_hat = sum(theta[j] * inp[j][i] for j in range(4))
        for j in range(4):
            dJ[j] += alpha * (out[i] - y_hat) * inp[j][i] / m
    return dJ

# Standardize function
def standardize(x):
    mean = np.mean(x)
    std = np.std(x)
    return (x - mean) / std

# Standardize a single value
def standardize_s(x0, x):
    mean = np.mean(x)
    std = np.std(x)
    return (x0 - mean) / std

# Load data from CSV
x1, x2, x3, y = [], [], [], []
with open('data/KCSmall_NS2.csv', newline='') as f:
    reader = csv.reader(f)
    for row in reader:
        x1.append(float(row[0]))
        x2.append(float(row[1]))
        x3.append(float(row[2]))
        y.append(float(row[3]))

# Display the first 5 rows of raw data
print("--------------------------------------------------------------------------------------------")
print("(1) First 5 rows of raw input data are:")
for i in range(5):
    print("{0}\t{1}\t{2}\t{3}".format(x1[i], x2[i], x3[i], y[i]))

# Standardize the data
x1s = standardize(x1)
x2s = standardize(x2)
x3s = standardize(x3)
xs = [[1.0]*len(x1s), x1s, x2s, x3s]

# Display the first 5 rows of standardized data
print("--------------------------------------------------------------------------------------------")
print("(2) First 5 rows of standardized data are:")
for i in range(5):
    print("1\t{0}\t{1}\t{2}\t{3}".format(x1s[i], x2s[i], x3s[i], y[i]))

# Initialize theta and calculate initial cost
theta = [0, 0, 0, 0]
initial_cost = loss_func(xs, y, theta)
print("The cost (J) value for theta = {0} is: J = {1}".format(theta, initial_cost))

# Gradient descent parameters
n_iterations = 50
alpha = 1.5
J_values = []

# Perform gradient descent
print("--------------------------------------------------------------------------------------------")
print("Theta \t\t\t\t\t\t\t\t Loss Function (J)")
print("--------------------------------------------------------------------------------------------")
for _ in range(n_iterations):
    gradients = loss_func_der(xs, y, theta, alpha)
    for j in range(4):
        theta[j] += gradients[j]
    current_cost = loss_func(xs, y, theta)
    J_values.append(current_cost)
    theta_rounded = [round(t, 5) for t in theta]
    current_cost_rounded = round(current_cost, 5)
    print("{0} \t\t {1}".format(theta_rounded, current_cost_rounded))

# Test the model with new input
new_features = [3, 2000, 8550]
xtests = [1, standardize_s(new_features[0], x1), standardize_s(new_features[1], x2), standardize_s(new_features[2], x3)]
predicted_price = round(sum(theta[j] * xtests[j] for j in range(4)), 2)
print("Approximate house price for {0} is: ${1}".format(new_features, predicted_price))

# Plot the cost function
print("--------------------------------------------------------------------------------------------")
plt.title("J vs n")
plt.plot(range(1, n_iterations + 1), J_values)
plt.xlabel('Iterations')
plt.ylabel('Loss value (J)')
plt.grid(axis='both')
plt.show()
