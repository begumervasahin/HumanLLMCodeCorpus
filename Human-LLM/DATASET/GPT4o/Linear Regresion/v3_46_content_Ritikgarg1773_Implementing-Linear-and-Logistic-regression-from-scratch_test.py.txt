from scratch import MyLinearRegression, MyLogisticRegression, MyPreProcessor
import numpy as np
import matplotlib.pyplot as plt

# Initialize preprocessor
preprocessor = MyPreProcessor()

def plot_loss(training_errors, validation_errors, training_label, validation_label, ylabel):
    plt.plot(training_errors, color='red', linewidth=2, label=training_label)
    plt.plot(validation_errors, color='orange', linewidth=2, label=validation_label)
    plt.xlabel("Number of Iterations")
    plt.ylabel(ylabel)
    plt.legend()
    plt.show()
    plt.clf()

def perform_linear_regression(preprocessor):
    print('Linear Regression')
    X, y = preprocessor.pre_process(0)
    numberOfRows = int(X.shape[0] * 0.7)
    X_train, X_test = X[:numberOfRows], X[numberOfRows:]
    y_train, y_test = y[:numberOfRows], y[numberOfRows:]

    linear = MyLinearRegression()
    k = 5
    X_folds = np.array_split(X_train, k)
    y_folds = np.array_split(y_train, k)

    for i in range(k):
        X_data_train = np.concatenate(X_folds[:i] + X_folds[i+1:])
        y_data_train = np.concatenate(y_folds[:i] + y_folds[i+1:])
        X_data_test = X_folds[i]
        y_data_test = y_folds[i]

        RMSE_errors, MAE_errors, Rmse_coef, Rmse_intercept, Mae_coef, Mae_intercept = linear.fit(X_data_train, y_data_train)
        RMSE_validation, MAE_validation = linear.validationLoss(X_data_test, y_data_test, Rmse_coef, Rmse_intercept, Mae_coef, Mae_intercept)

        plot_loss(RMSE_errors, RMSE_validation, 'RMSE Training Loss', 'RMSE Validation Loss', "Loss")
        print('RMSE Error on Test Data:', RMSE_validation[-1])

        plot_loss(MAE_errors, MAE_validation, 'MAE Training Loss', 'MAE Validation Loss', "Loss")
        print('MAE Error on Test Data:', MAE_validation[-1])

perform_linear_regression(preprocessor)
