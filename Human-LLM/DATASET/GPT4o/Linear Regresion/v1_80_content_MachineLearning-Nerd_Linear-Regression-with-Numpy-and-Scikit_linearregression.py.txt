import numpy as np
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d import Axes3D
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import PolynomialFeatures


def extract_data(input_file='./input03.txt', output_file='./output03.txt'):
    """
    Extracts training and testing data from given input files.
    
    Args:
    input_file (str): Path to the input file containing training and test data.
    output_file (str): Path to the output file containing test data results.
    
    Returns:
    tuple: Tuple containing training features, training targets, test features, and test targets.
    """
    with open(input_file, 'r') as f:
        lines = list(map(int, f.readline().split()))
        num_train = lines[1]

        X_train = []
        Y_train = []
        for _ in range(num_train):
            train_data = list(map(float, f.readline().strip().split()))
            X_train.append(train_data[:2])
            Y_train.append(train_data[2])

        num_test = int(f.readline().strip())
        X_test = []
        for _ in range(num_test):
            test_data = list(map(float, f.readline().strip().split()))
            X_test.append(test_data)

    X_train = np.array(X_train)
    Y_train = np.array(Y_train)
    X_test = np.array(X_test)
    
    Y_test = []
    with open(output_file, 'r') as f:
        for _ in range(num_test):
            test_val = float(f.readline().strip())
            Y_test.append(test_val)

    return X_train, Y_train, X_test, Y_test


def linear_regression_numpy(X_train, Y_train, X_test):
    """
    Performs linear regression using numpy.
    
    Args:
    X_train (np.array): Training features.
    Y_train (np.array): Training targets.
    X_test (np.array): Test features.
    
    Returns:
    np.array: Predicted values for the test features.
    """
    x_train = np.column_stack((np.ones([X_train.shape[0], 1]), X_train))
    x_test = np.column_stack((np.ones([X_test.shape[0], 1]), X_test))
    coef = np.linalg.pinv(x_train.T @ x_train) @ x_train.T @ Y_train
    Y_pred = x_test @ coef
    return Y_pred


def linear_regression_sklearn(X_train, Y_train, X_test):
    """
    Performs linear regression using scikit-learn.
    
    Args:
    X_train (np.array): Training features.
    Y_train (np.array): Training targets.
    X_test (np.array): Test features.
    
    Returns:
    np.array: Predicted values for the test features.
    """
    lin_reg = LinearRegression()
    lin_reg.fit(X_train, Y_train)
    Y_pred = lin_reg.predict(X_test)
    return Y_pred


def plot_results(X_train, Y_train, Y_pred, title):
    """
    Plots 3D scatter plot of training data and predictions.
    
    Args:
    X_train (np.array): Training features.
    Y_train (np.array): Training targets.
    Y_pred (np.array): Predicted values for the training features.
    title (str): Title of the plot.
    """
    fig = plt.figure()
    ax = fig.add_subplot(111, projection='3d')
    ax.scatter(X_train[:, 0], X_train[:, 1], Y_train, c='b', marker='x', label='X_train vs Y_train', s=10)
    ax.scatter(X_train[:, 0], X_train[:, 1], Y_pred, c='r', marker='s', label='X_train vs Y_pred', s=10)
    plt.legend(loc='lower right')
    plt.title(title)
    plt.show()


if __name__ == "__main__":
    X_train, Y_train, X_test, Y_test = extract_data()

    # Linear Regression without polynomial features
    print("Linear Regression without polynomial features (Numpy):", linear_regression_numpy(X_train, Y_train, X_test))
    print("Linear Regression without polynomial features (Sklearn):", linear_regression_sklearn(X_train, Y_train, X_test))

    Yhat_train = linear_regression_sklearn(X_train, Y_train, X_train)
    plot_results(X_train, Y_train, Yhat_train, "Without polynomial features")

    # Linear Regression with polynomial features
    poly = PolynomialFeatures(degree=3)
    X_poly_train = poly.fit_transform(X_train)
    X_poly_test = poly.fit_transform(X_test)

    print("Linear Regression with polynomial features (Numpy):", linear_regression_numpy(X_poly_train, Y_train, X_poly_test))
    print("Linear Regression with polynomial features (Sklearn):", linear_regression_sklearn(X_poly_train, Y_train, X_poly_test))

    Yhat_poly_train = linear_regression_sklearn(X_poly_train, Y_train, X_poly_train)
    plot_results(X_train, Y_train, Yhat_poly_train, "With polynomial features")
