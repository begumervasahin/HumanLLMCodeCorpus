import numpy as np
import numpy.random as npr
from mnist import MNIST
import matplotlib.pyplot as plt

# Load MNIST data
mndata = MNIST('.')
x_train, y_train = mndata.load_training()
x_test, y_test = mndata.load_testing()

# Convert lists to numpy arrays and reshape
to_np_array = lambda x, shape: np.array(x).reshape(shape)
x_train = to_np_array(x_train, [-1, 28 * 28])
y_train = to_np_array(y_train, [-1, 1])
x_test = to_np_array(x_test, [-1, 28 * 28])
y_test = to_np_array(y_test, [-1, 1])

# Convert multi-class labels to binary labels (-1 for not '1', 1 for '1')
y_train = np.where(y_train != 1.0, -1.0, 1.0)
y_test = np.where(y_test != 1.0, -1.0, 1.0)

# Define Linear Regression model
class LinearReg:
    def __init__(self, in_dim: int, out_dim: int, bias: bool = False):
        self.bias = bias
        self.A = npr.uniform(-0.2, 0.2, [out_dim, in_dim])
        if bias:
            self.B = np.ones([out_dim, 1], dtype=np.float64)
    
    def __call__(self, x, apply_sign=False):
        yh = self.A @ x
        if self.bias:
            yh += self.B
        if apply_sign:
            return np.sign(yh)
        return yh
    
    def loss_fn(self, yh, y):
        return (yh - y) ** 2.0

# Train the model
def train(model: LinearReg, train_data: tuple, epochs: int = 2, verbose: bool = False, train_with_bias: bool = False):
    assert len(train_data) == 2, "Train data should be a tuple (x, y)"
    lrate = 9e-9
    if train_with_bias:
        assert model.bias, 'The bias parameter is not enabled.'
    
    x_train, y_train = train_data
    N = x_train.shape[0]
    
    for epoch in range(epochs):
        loss = 0.0
        dJdA = np.zeros_like(model.A, dtype=np.float64)
        
        for i in range(N):
            x = x_train[i][..., None]
            y = y_train[i][..., None]
            yh = model(x)
            loss += model.loss_fn(yh, y)
            
            dJdyh = (yh - y)
            dyhdA = x
            dJdA += dJdyh @ dyhdA.T
            
            if train_with_bias:
                # Calculate dJ w.r.t Bias or B, i.e., dJ/dB
                continue
        
        loss /= (N * 2.0)
        dJdA /= N
        model.A -= lrate * dJdA
        
        if train_with_bias:
            # Update Bias --- declare dJdB somewhere and make sure the shape is correct
            pass
        
        if verbose:
            print(f"Training iteration: {epoch + 1}, Loss: {float(loss)}")
        if epoch == epochs - 1:
            print(f"Training iteration: {epoch + 1}, Loss: {float(loss)}")
    
    return model

# Evaluate the model
def evaluate(model: LinearReg, test_data: tuple):
    assert len(test_data) == 2, "Test data should be a tuple (x, y)"
    
    x_test, y_test = test_data
    N = x_test.shape[0]
    loss, accuracy = 0.0, 0.0
    
    for i in range(N):
        x = x_test[i][..., None]
        y = y_test[i][..., None]
        yh = model(x)
        loss += model.loss_fn(yh, y)
        yh = np.sign(yh)
        accuracy += np.where(yh == y, 1.0, 0.0).sum() / y.size
    
    loss /= (N * 2.0)
    accuracy /= N
    print(f"Loss: {float(loss)}, Accuracy: {accuracy}")

# Instantiate and train the model
model = LinearReg(in_dim=784, out_dim=1, bias=False)
model = train(model, (x_train, y_train), epochs=15, verbose=True)

# Evaluate the model
evaluate(model, (x_test, y_test))

# Print prediction for the first test example
print(f"Pred: {model(x_test[0][..., None], apply_sign=True)}, True: {y_test[0]}")
