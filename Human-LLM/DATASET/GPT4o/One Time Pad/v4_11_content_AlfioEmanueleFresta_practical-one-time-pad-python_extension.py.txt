from cp_otp import strxor
from extension.utils import keystream_generator

def stream_cipher(key, message, key_length=8):
    """
    Encrypts or decrypts a message using a stream cipher.

    Args:
        key (bytes): The key used for the encryption/decryption.
        message (bytes): The message to be encrypted/decrypted.
        key_length (int): The length of the key stream.

    Returns:
        bytes: The encrypted or decrypted message.
    """
    # Generate a keystream based on the provided key and the required key length
    keystream = keystream_generator(key, key_length)
    
    # XOR the keystream with the message to get the cipher text
    return strxor(keystream, message)

# Example usage
if __name__ == "__main__":
    plaintext = b'WOULD YOU KINDLY ENCRYPT ME, PLEASE'
    key = b'I will be very useful as a key for your encryption task, curious stranger'
    secret = b'\xcf\xab|\xb40\x92y\x03r&\x014\xee\xae\xc92r(U\x98\xdee\xe8\x8fG,\x00B\xdb\xcf\xa8L6F\xa8c\x15\x89\x94>*J\xc8q\xf2"\xd9\xeb\xc5\xb4\x15i\xad\xbc!n\x92I\xee\x8a\x18\x93\x94\xfc\x11'

    # Encrypt the plaintext
    encrypted_message = stream_cipher(key, plaintext, key_length=len(plaintext))
    print("Encrypted message:", encrypted_message)

    # Decrypt the secret
    decrypted_message = stream_cipher(key, secret, key_length=len(secret))
    print("Decrypted message:", decrypted_message)
