from numbers import Number

class One:
    def __add__(self, other):
        """
        Custom addition for the One class.
        Returns One if other is 0, else returns 0.
        """
        return self if other == 0 else 0
    
    # Addition and subtraction are the same for One
    __sub__ = __add__

    def __mul__(self, other):
        """
        Custom multiplication for the One class.
        Returns 0 if other is 0, else returns One.
        """
        if isinstance(other, Number):
            return 0 if other == 0 else self
        return other

    def __truediv__(self, other):
        """
        Custom true division for the One class.
        Raises ZeroDivisionError if other is 0, else returns One.
        """
        if other == 0:
            raise ZeroDivisionError("division by zero")
        return self

    # True division and classic division are the same for One
    __div__ = __truediv__

    def __rdiv__(self, other):
        """
        Custom reverse division for the One class.
        Returns other.
        """
        return other

    # Reverse true division is the same as reverse classic division
    __rtruediv__ = __rdiv__
    
    # Reverse addition and subtraction are the same as addition
    __radd__ = __add__
    __rsub__ = __add__

    # Reverse multiplication is the same as multiplication
    __rmul__ = __mul__

    def __lt__(self, other):
        """
        Custom less-than comparison for the One class.
        Always returns False.
        """
        return False

    def __eq__(self, other):
        """
        Custom equality comparison for the One class.
        Returns True if other is One or 0, else raises TypeError.
        """
        if isinstance(other, self.__class__) or other == 0:
            return other != 0
        raise TypeError(f"Cannot compare One with {type(other)}")

    def __hash__(self):
        """
        Custom hash for the One class.
        Returns 1.
        """
        return 1

    def __str__(self):
        """
        Custom string representation for the One class.
        Returns 'one'.
        """
        return 'one'

    __repr__ = __str__

    def __neg__(self):
        """
        Custom negation for the One class.
        Returns One.
        """
        return self

    def __bool__(self):
        """
        Custom boolean conversion for the One class.
        Always returns True.
        """
        return True

    def __format__(self, format_spec):
        """
        Custom format for the One class.
        Returns formatted string representation.
        """
        return format(str(self), format_spec)

# Create instances of One and zero
one = One()
