from modules import algorithms

def decrypt(ciphertext: str, pad: str) -> str:
    """
    Decrypts the given ciphertext using the provided one-time pad.
    
    Args:
        ciphertext (str): The ciphertext to decrypt.
        pad (str): The one-time pad used for decryption.
        
    Returns:
        str: The decrypted plaintext.
    """
    # Convert characters to their corresponding positions in the alphabet
    c_text_num = algorithms.alph_pos(list(ciphertext))
    pad_num = algorithms.alph_pos(list(pad))
    
    # Calculate the plaintext numeric values using the modular inverse
    p_text_num = [
        algorithms.modinverse((c_text_num[i] - pad_num[i]), 26)
        for i in range(len(c_text_num))
    ]
    
    # Convert numeric values back to characters
    plaintext = algorithms.num_alph(p_text_num)
    return "".join(plaintext)

def main():
    """
    Main function to handle user input and perform decryption.
    """
    c_text = input("Enter ciphertext: ")
    pad = input("Enter pad: ")
    
    # Check if the lengths of the ciphertext and the pad are the same
    if len(c_text) != len(pad):
        print("Error: The length of the ciphertext and the pad must be the same.")
    else:
        # Decrypt the ciphertext and print the result
        decrypted_text = decrypt(c_text, pad)
        print("Decrypted text:", decrypted_text)

if __name__ == "__main__":
    main()
