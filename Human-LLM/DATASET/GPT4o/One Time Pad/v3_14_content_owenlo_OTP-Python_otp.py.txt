import random

# Define the character set
CHARSET = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"

def main():
    """
    Demonstrates the usage of the encryption and decryption functions.
    """
    vector = "Once more into the fray."
    otp, encrypted_message = encrypt(vector)
    decrypted_message = decrypt(otp, encrypted_message)
    
    print(f"Test Vector: {vector}")
    print(f"One-Time Pad (OTP): {otp}")
    print(f"Encrypted: {encrypted_message}")
    print(f"Decrypted: {decrypted_message}")

def generate_otp():
    """
    Generates a randomized one-time pad using the defined character set.

    Returns:
    str: The generated one-time pad.
    """
    return "".join(random.sample(CHARSET, len(CHARSET)))

def encrypt(plaintext):
    """
    Encrypts the plaintext using a randomized one-time pad.

    Args:
    plaintext (str): The plaintext value to encrypt.

    Returns:
    tuple: A tuple containing the one-time pad and the encrypted message.
    """
    otp = generate_otp()
    encrypted_message = "".join(
        otp[CHARSET.index(c)] if c in CHARSET else c for c in plaintext.upper()
    )
    return otp, encrypted_message

def decrypt(otp, encrypted_message):
    """
    Decrypts the encrypted message using the one-time pad.

    Args:
    otp (str): The one-time pad used for encryption.
    encrypted_message (str): The encrypted message to be decrypted.

    Returns:
    str: The decrypted message.
    """
    decrypted_message = "".join(
        CHARSET[otp.index(c)] if c in otp else c for c in encrypted_message.upper()
    )
    return decrypted_message

if __name__ == "__main__":
    main()
