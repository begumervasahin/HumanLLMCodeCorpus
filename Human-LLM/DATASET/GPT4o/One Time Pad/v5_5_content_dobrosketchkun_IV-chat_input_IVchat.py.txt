import ipfsapi

DEBUG = 0
point = 0

# Connect to the IPFS API
api = ipfsapi.connect('127.0.0.1', 5001)

# Read the random key file
with open('TEST_RANDOM_DO_NOT_USE_IN_REAL_CONVERSATION.txt', 'r') as f:
    all_random = f.read()

def save_point(point):
    """
    Save the current point to a file.
    
    Parameters
    ----------
    point : int
        The current point to save.
    """
    with open('pointt', 'w') as fpoint:
        fpoint.write(str(point))

def debug_print(message, *args):
    """
    Print debug messages if DEBUG mode is enabled.
    
    Parameters
    ----------
    message : str
        The message to print.
    args : tuple
        Additional arguments to print.
    """
    if DEBUG:
        print(message, *args)

def encrypt(string, key):
    """
    Encrypt a string using a key with XOR operation.
    
    Parameters
    ----------
    string : str
        The string to encrypt.
    key : str
        The key to use for encryption.
    
    Returns
    -------
    str
        The encrypted string.
    """
    return "".join(chr(ord(s) ^ ord(k)) for s, k in zip(string, key))

def main():
    """
    Main function to handle the encryption and IPFS publishing process.
    """
    global point
    
    print("You:")
    while True:
        data = input("")
        debug_print('point:', point)
        
        end = point + len(data)
        debug_print('end:', end)
        
        key = all_random[point:end]
        debug_print('key:', key)
        
        data_enc = encrypt(data, key)
        debug_print('data_enc:', data_enc)
        
        data_ipfs = api.add_pyobj([data_enc, point, end])
        debug_print('data_ipfs:', [data_enc, point, end])
        
        data_ipns = api.name_publish('/ipfs/' + data_ipfs)
        debug_print('IPFS published:', data_ipns)
        
        point = end
        save_point(point)

if __name__ == "__main__":
    main()
