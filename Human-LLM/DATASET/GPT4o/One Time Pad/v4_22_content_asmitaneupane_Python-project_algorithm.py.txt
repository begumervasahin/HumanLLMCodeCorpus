def vernam_cipher(key, text):
    """
    Returns the Vernam Cipher for the given string and key.
    
    Parameters:
    key (str): The key to be used for the Vernam Cipher.
    text (str): The text to be encrypted or decrypted.
    
    Returns:
    str: The result of the Vernam Cipher encryption/decryption.
    """
    answer = []
    key_length = len(key)
    
    for i, char in enumerate(text):
        answer.append(chr(ord(char) ^ ord(key[i % key_length])))
    
    return ''.join(answer)

# Example usage
key = "mysecretkey"
text = "Hello, World!"
cipher_text = vernam_cipher(key, text)
print("Cipher Text:", cipher_text)

# Decrypting the cipher text (using the same function)
decrypted_text = vernam_cipher(key, cipher_text)
print("Decrypted Text:", decrypted_text)
