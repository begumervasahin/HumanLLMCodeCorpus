def hex_to_nums(hex_string):
    """
    Converts a message of hexadecimal pairs to a list of decimal numbers.

    Args:
        hex_string (str): String containing a stream of hexadecimal bytes.

    Returns:
        list: List of integers representing the decimal equivalents of the hex pairs.
    """
    num_bytes = len(hex_string) // 2
    decimal_values = [0] * num_bytes

    for i in range(num_bytes):
        digit1 = hex_digit_to_decimal(hex_string[2 * i])
        digit2 = hex_digit_to_decimal(hex_string[2 * i + 1])
        decimal_values[i] = 16 * digit1 + digit2

    return decimal_values

def nums_to_hex(decimal_values):
    """
    Converts a list of integers to their 2-digit hexadecimal string representations.

    Args:
        decimal_values (list): List of integers.

    Returns:
        list: List of strings representing the hexadecimal values of the integers.
    """
    return [hex(n)[2:].upper().zfill(2) for n in decimal_values]

def hex_digit_to_decimal(hex_digit):
    """
    Determines the decimal equivalent of a hex digit.

    Args:
        hex_digit (str): A single hexadecimal digit.

    Returns:
        int: Decimal equivalent of the hex digit.
    """
    if hex_digit.isdigit():
        return int(hex_digit)
    return ord(hex_digit.upper()) - 55

def bytes_to_chars(byte_values):
    """
    Converts a list of bytes to their corresponding character representations.

    Args:
        byte_values (list): List of byte values.

    Returns:
        list: List of characters corresponding to the byte values.
    """
    return [chr(b) for b in byte_values]

def xor_chars(chars):
    """
    Returns the XOR of all characters in the input list.

    Args:
        chars (list): List of characters.

    Returns:
        str: A single character resulting from the XOR operation.
    """
    xor_result = 0
    for char in chars:
        xor_result ^= ord(char)
    return chr(xor_result)

# Example usage
hex_message = "48656C6C6F"  # Represents "Hello" in hex
decimal_values = hex_to_nums(hex_message)
hex_values = nums_to_hex(decimal_values)
char_values = bytes_to_chars(decimal_values)
xor_result = xor_chars(char_values)

print(f"Hex Message: {hex_message}")
print(f"Decimal Values: {decimal_values}")
print(f"Hex Values: {hex_values}")
print(f"Character Values: {char_values}")
print(f"XOR Result: {xor_result}")
