import os
import sys

def main():
    """
    Main function to set up Django environment variables and execute command line commands.
    """
    # Set default environment variables for Django settings and configuration
    os.environ.setdefault("DJANGO_SETTINGS_MODULE", "FacePad.config")
    os.environ.setdefault("DJANGO_CONFIGURATION", "Local")

    try:
        # Attempt to import the execute_from_command_line function from configurations.management
        from configurations.management import execute_from_command_line
    except ImportError:
        try:
            # If the first import fails, check if Django is installed
            import django
        except ImportError:
            # Raise an error if Django is not installed or not found in the PYTHONPATH
            raise ImportError(
                "Couldn't import Django. Are you sure it's installed and "
                "available on your PYTHONPATH environment variable? Did you "
                "forget to activate a virtual environment?"
            )
        # If configurations is not available, re-raise the initial ImportError
        raise

    # Execute the command line command passed in sys.argv
    execute_from_command_line(sys.argv)

if __name__ == "__main__":
    main()
