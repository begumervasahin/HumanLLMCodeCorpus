import sys
import encrypt

def main() -> None:
    """
    Main function to handle command-line arguments and call the encryption/decryption function.
    """
    if len(sys.argv) != 6:
        print("Usage: python script.py <encrypt/decrypt> <passphrase> <path> <outputType> <saveLocation>")
        return

    # Parse command-line arguments
    operation = sys.argv[1]
    passphrase = sys.argv[2][1:]  # Remove the first character
    path = sys.argv[3][1:]  # Remove the first character
    output_type = sys.argv[4]
    save_location = sys.argv[5][1:]  # Remove the first character

    # Call the encrypt/decrypt function
    result = encrypt.enc(operation, passphrase, path, output_type, save_location)
    print(result)

if __name__ == "__main__":
    main()
