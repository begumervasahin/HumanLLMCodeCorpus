import sys
from PyQt5 import QtWidgets
import config.config
from gui.mainwindow import MainWindow

# Load the active configuration
activeConfig = config.config.getConfig()

def main(args):
    """
    Main function to run the PyQt5 application.
    
    Parameters:
    args (list): Command line arguments passed to the script.
    """
    # Create the application
    app = QtWidgets.QApplication(sys.argv)
    
    # Create the main window with the active configuration
    window = MainWindow(activeConfig)
    window.show()
    
    # Execute the application
    app.exec_()

if __name__ == "__main__":
    # Run the main function with command line arguments
    main(sys.argv[1:])
