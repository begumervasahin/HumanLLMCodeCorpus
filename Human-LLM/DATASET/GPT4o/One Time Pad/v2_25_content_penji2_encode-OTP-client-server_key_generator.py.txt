import random
import string

def generate_random_keys(num_keys: int, key_file: str, binary_key_file: str) -> None:
    """
    Generates random ASCII letters and their binary representations, writing them to specified files.
    
    Parameters:
    num_keys (int): The number of random ASCII letters to generate.
    key_file (str): The file path to write the ASCII letters.
    binary_key_file (str): The file path to write the binary representations of the ASCII letters.
    """
    with open(key_file, 'w') as ascii_file, open(binary_key_file, 'w') as binary_file:
        for _ in range(num_keys):
            # Choose a random ASCII letter
            random_letter = random.choice(string.ascii_letters)
            # Write the letter to key_file
            ascii_file.write(random_letter)
            # Convert the letter to its ASCII value, then to an 8-bit binary string
            binary_representation = f'{ord(random_letter):08b}'
            # Write the binary string to binary_key_file
            binary_file.write(binary_representation)

def main() -> None:
    """
    Main function to generate and write 1,000,000 random ASCII letters and their binary representations.
    """
    num_keys = 1000000
    generate_random_keys(num_keys, "key.dat", "key1.dat")

if __name__ == "__main__":
    main()
