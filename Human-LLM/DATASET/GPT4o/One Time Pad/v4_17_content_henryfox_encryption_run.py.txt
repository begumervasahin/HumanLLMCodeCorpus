import sys
import encrypt

def main(decrypt_or_encrypt, passphrase, path, output_type, save_location):
    """
    Main function to perform encryption or decryption based on command-line arguments.
    
    Parameters:
    decrypt_or_encrypt (str): The operation to perform ('encrypt' or 'decrypt').
    passphrase (str): The passphrase for encryption or decryption.
    path (str): The path to the file to be encrypted or decrypted.
    output_type (str): The type of output desired.
    save_location (str): The location to save the output file.
    
    Returns:
    None
    """
    result = encrypt.enc(decrypt_or_encrypt, passphrase, path, output_type, save_location)
    print(result)

if __name__ == "__main__":
    if len(sys.argv) != 6:
        print("Usage: python script.py <encrypt|decrypt> <passphrase> <path> <outputType> <saveLocation>")
        sys.exit(1)
    
    decrypt_or_encrypt = sys.argv[1]
    passphrase = sys.argv[2][1:]  # Remove leading character if necessary
    path = sys.argv[3][1:]        # Remove leading character if necessary
    output_type = sys.argv[4]
    save_location = sys.argv[5][1:]  # Remove leading character if necessary
    
    main(decrypt_or_encrypt, passphrase, path, output_type, save_location)
