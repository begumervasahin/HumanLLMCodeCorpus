def demonstrate_ord_chr_xor():
    """
    Demonstrate the use of ord, chr, and bitwise XOR operations.
    
    This function prints the hexadecimal values of the characters 'A' and ' ',
    performs a bitwise XOR operation on their ordinal values, and prints the
    resulting character.
    """
    char_a = 'A'
    char_space = ' '
    
    ord_a = ord(char_a)
    ord_space = ord(char_space)
    
    print(f"ord('A') = {hex(ord_a)}")
    print(f"ord(' ') = {hex(ord_space)}")
    
    axorb = chr(ord_a ^ ord_space)
    print(f"chr(ord('A') ^ ord(' ')) = {axorb}")

def main():
    """
    Main function to run the demonstration of ord, chr, and bitwise XOR operations.
    """
    demonstrate_ord_chr_xor()

if __name__ == "__main__":
    main()
