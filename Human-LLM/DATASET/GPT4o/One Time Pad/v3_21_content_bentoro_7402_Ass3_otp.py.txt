import sys
import random
import argparse

def main(argv):
    """
    Main function to handle command-line arguments and initiate the encryption process.

    Args:
    argv (list): List of command-line arguments.
    """
    args = parse_arguments(argv)
    content = read_content(args.filename, args.message)

    if not content:
        print("No content to encrypt")
        return

    xor_encrypt_and_save(content, args.output)

def parse_arguments(argv):
    """
    Parses command-line arguments.

    Args:
    argv (list): List of command-line arguments.

    Returns:
    argparse.Namespace: Parsed arguments.
    """
    parser = argparse.ArgumentParser(description="Encrypt a message using XOR encryption.")
    parser.add_argument('-f', '--filename', help="Input file containing the message")
    parser.add_argument('-m', '--message', help="Message to be encrypted")
    parser.add_argument('-o', '--output', required=True, help="Output file for the encrypted message")
    return parser.parse_args(argv)

def read_content(filename, message):
    """
    Reads the content either from a file or directly from the provided message.

    Args:
    filename (str): The name of the file to read from.
    message (str): The message to be used if no file is provided.

    Returns:
    str: The content to be encrypted.
    """
    if filename:
        with open(filename, 'r', encoding='utf-8') as file:
            return file.read()
    return message

def generate_random_key():
    """
    Generates a random key for XOR encryption.

    Returns:
    int: A random integer between 0 and 255.
    """
    return random.randint(0, 255)

def xor_encrypt_and_save(content, outfile):
    """
    Encrypts the content using XOR encryption and writes the result to an output file.

    Args:
    content (str): The content to be encrypted.
    outfile (str): The file to write the encrypted content to.
    """
    key = generate_random_key()
    encrypted_content = xor_encrypt(content, key)
    
    with open(outfile, 'w', encoding='utf-8') as fileout:
        fileout.write(encrypted_content)
        # Save the key at the end for decryption (for demonstration purposes)
        fileout.write(f"\nKEY={key}")

def xor_encrypt(content, key):
    """
    Encrypts the content using XOR encryption.

    Args:
    content (str): The content to be encrypted.
    key (int): The XOR encryption key.

    Returns:
    str: The encrypted content.
    """
    return ''.join(chr(ord(char) ^ key) for char in content)

if __name__ == "__main__":
    main(sys.argv[1:])
