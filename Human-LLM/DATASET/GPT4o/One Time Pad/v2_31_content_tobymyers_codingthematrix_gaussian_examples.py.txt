from mat import Mat
from GF2 import one

def create_matrix(domain, elements) -> Mat:
    """
    Creates a matrix with the given domain and elements.

    Parameters:
    domain (tuple): A tuple containing the set of row labels and column labels.
    elements (dict): A dictionary representing the non-zero elements of the matrix.
    
    Returns:
    Mat: The created matrix.
    """
    return Mat(domain, elements)

def main() -> None:
    """
    Main function to define and print matrices A and B.
    """
    # Define the domain and elements of matrix A
    A_domain = (set([1, 2, 3, 4]), set(['A', 'B', 'C', 'D']))
    A_elements = {
        (2, 'A'): one, (2, 'C'): one, (2, 'D'): one,
        (1, 'C'): one, (1, 'D'): one,
        (3, 'A'): one, (3, 'D'): one,
        (4, 'A'): one, (4, 'B'): one, (4, 'C'): one, (4, 'D'): one
    }
    A = create_matrix(A_domain, A_elements)

    # Define the domain and elements of matrix B
    B_domain = (set([1, 2, 3, 4]), set(['A', 'B', 'C', 'D']))
    B_elements = {
        (1, 'A'): one, (1, 'B'): one,
        (2, 'A'): one, (2, 'C'): one,
        (3, 'B'): one, (3, 'C'): one, (3, 'D'): one,
        (4, 'A'): one
    }
    B = create_matrix(B_domain, B_elements)

    # Print matrices A and B
    print("Matrix A:")
    print(A)
    print("\nMatrix B:")
    print(B)

if __name__ == "__main__":
    main()
