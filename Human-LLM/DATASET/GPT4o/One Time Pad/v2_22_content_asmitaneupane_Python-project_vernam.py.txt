import sys
import click

def vernam(text, key, return_str=False, alphanumerical=False):
    """
    Encrypts or decrypts text using the Vernam cipher.

    Parameters
    ----------
    text : str
        The input text to be encrypted or decrypted.
    key : str
        The key used for encryption or decryption.
    return_str : bool, optional
        If True, returns the result as a string; otherwise, returns a list. Default is False.
    alphanumerical : bool, optional
        If True, encodes alphanumerically; otherwise, uses Unicode points. Default is False.

    Returns
    -------
    str or list
        The encrypted or decrypted result.
    """
    # Define conversion functions based on the encoding type
    if alphanumerical:
        alphanumerics = "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ"
        to_num = lambda x: alphanumerics.index(x)
        to_char = lambda x: alphanumerics[x]
    else:
        to_num = ord
        to_char = chr

    # Convert text and key to their numeric representations
    bintext = [to_num(x) for x in text]
    binkey = [to_num(x) for x in key]

    # Extend the key to match the length of the text
    while len(binkey) < len(bintext):
        binkey.append(binkey[len(binkey) % len(key)])

    # Apply the Vernam cipher
    vernamed = [bintext[i] ^ binkey[i] for i in range(len(bintext))]
    result = [to_char(i) for i in vernamed]

    # Return the result as a string or list based on the parameter
    if return_str:
        return "".join(result)
    return result

@click.command()
@click.argument("text")
@click.argument("key")
@click.option('--string/--list', '-s/-l', "return_str", default=False, help="Return as string [default: list]")
@click.option('--alphanumerical/--unicode', '-a/-u', "alphanumerical", default=False, help="Encode alphanumerically [default: use Unicode points]")
def cli(text, key, return_str, alphanumerical):
    """
    CLI for the Vernam cipher.

    Parameters
    ----------
    text : str
        The input text to be encrypted or decrypted.
    key : str
        The key used for encryption or decryption.
    return_str : bool, optional
        If True, returns the result as a string; otherwise, returns a list. Default is False.
    alphanumerical : bool, optional
        If True, encodes alphanumerically; otherwise, uses Unicode points. Default is False.
    """
    result = vernam(text, key, return_str, alphanumerical)
    click.echo(result)

if __name__ == "__main__":
    cli()
