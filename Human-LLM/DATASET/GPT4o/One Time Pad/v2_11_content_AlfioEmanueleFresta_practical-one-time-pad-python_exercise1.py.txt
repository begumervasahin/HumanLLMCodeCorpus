from itertools import combinations

def strxor(a: bytes, b: bytes) -> bytes:
    """
    XORs two byte strings and returns the result.

    Parameters:
    a (bytes): The first byte string.
    b (bytes): The second byte string.

    Returns:
    bytes: The result of XORing the two byte strings.
    """
    return bytes(x ^ y for x, y in zip(a, b))

def list_of_words(of_length: int) -> list:
    """
    Generates a list of words of a given length.

    Parameters:
    of_length (int): The length of the words to generate.

    Returns:
    list: A list of byte strings representing words of the specified length.
    """
    dummy_words = ['hello', 'world', 'apple', 'lemon', 'peach']
    return [word.encode('utf-8') for word in dummy_words if len(word) == of_length]

def main() -> None:
    """
    Main function to demonstrate the XORing of word pairs.
    """
    c1 = b'\x0b\x0e\x1e\x0b\x17'
    c2 = b'\x15\x0a\x1b\x1d\x01'
    
    words = list_of_words(of_length=len(c1))
    pairs = combinations(words, 2)
    
    for word1, word2 in pairs:
        xor_result = strxor(word1, word2)
        print(f'Word 1: {word1.decode("utf-8")}, Word 2: {word2.decode("utf-8")}, XOR Result: {xor_result}')

if __name__ == "__main__":
    main()
