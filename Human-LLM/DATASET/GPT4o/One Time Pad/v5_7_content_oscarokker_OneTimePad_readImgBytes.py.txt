import base64

def encode_image_to_base64(file_path):
    """
    Encode an image file to a base64 string.
    
    Parameters:
    file_path (str): The path to the image file.
    
    Returns:
    str: The base64 encoded string of the image.
    """
    with open(file_path, "rb") as image_file:
        encoded_string = base64.b64encode(image_file.read())
    return encoded_string.decode('utf-8')

def decode_base64_to_image(encoded_string, output_file_path):
    """
    Decode a base64 string to an image file.
    
    Parameters:
    encoded_string (str): The base64 encoded string of the image.
    output_file_path (str): The path to save the decoded image file.
    
    Returns:
    None
    """
    decoded_image = base64.b64decode(encoded_string.encode('utf-8'))
    with open(output_file_path, 'wb') as image_file:
        image_file.write(decoded_image)

def main():
    input_file_path = "rode_in_club.png"
    output_file_path = "new_rode_in_club.png"
    
    # Encode the image to a base64 string
    encoded_string = encode_image_to_base64(input_file_path)
    
    # Print the base64 encoded string
    print("Base64 Encoded String:")
    print(encoded_string)
    
    # Convert the base64 string to an integer (for demonstration purposes)
    encoded_integer = int.from_bytes(base64.b64decode(encoded_string), "big")
    print("\nEncoded Integer:")
    print(encoded_integer)
    
    # Decode the base64 string back to an image
    decode_base64_to_image(encoded_string, output_file_path)

if __name__ == "__main__":
    main()
