from orthogonalization import orthogonalize, aug_orthogonalize
from matutil import *
from vecutil import *
from math import sqrt

def find_norm(vec):
    """
    Calculates the Euclidean norm of a vector.
    
    Args:
    vec (Vec): The vector for which the norm is to be calculated.
    
    Returns:
    float: The Euclidean norm of the vector.
    """
    return sqrt(vec * vec)

def orthonormalize(L):
    """
    Orthonormalizes a list of vectors.
    
    Args:
    L (list of Vec): The list of vectors to be orthonormalized.
    
    Returns:
    list of Vec: The orthonormalized vectors.
    """
    Lstar = orthogonalize(L)
    norms = [find_norm(l) for l in Lstar]
    return [(1 / norm) * l for l, norm in zip(Lstar, norms)]

def adjust(v, multipliers):
    """
    Adjusts a vector by multiplying its elements by the given multipliers.
    
    Args:
    v (Vec): The vector to be adjusted.
    multipliers (dict): The multipliers for each element of the vector.
    
    Returns:
    Vec: The adjusted vector.
    """
    return Vec(v.D, {i: v[i] * multipliers[i] for i in v.D})

def aug_orthonormalize(L):
    """
    Performs augmented orthonormalization on a list of vectors.
    
    Args:
    L (list of Vec): The list of vectors to be orthonormalized.
    
    Returns:
    tuple: The orthonormalized vectors and the adjustment matrix.
    """
    Lstar, sigmadict = aug_orthogonalize(L)
    norms = [find_norm(l) for l in Lstar]
    R = [adjust(sigma, norms) for sigma in sigmadict]
    Q = [(1 / norm) * l for l, norm in zip(Lstar, norms)]
    return Q, R
