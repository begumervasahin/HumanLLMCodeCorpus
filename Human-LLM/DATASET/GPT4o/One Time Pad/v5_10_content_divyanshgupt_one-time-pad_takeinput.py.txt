def take_input(argv):
    """
    Processes command line arguments for encryption or decryption.

    Args:
        argv (list): List of command line arguments.

    Returns:
        tuple: A tuple containing the action (encrypt/decrypt), file directory, and key directory.
               If there is an error, returns (0, 0, 0).
    """
    # Ensure there are enough arguments
    if len(argv) < 2:
        print("Error: found no input")
        return 0, 0, 0

    # Extract the action (encrypt or decrypt)
    action = argv[1]
    file_directory = ""
    key_directory = ""

    # Validate the action and process further if valid
    if action in ["encrypt", "decrypt"]:
        # Check if file directory is provided as a command line argument
        if len(argv) > 2 and argv[2].startswith("file"):
            file_directory = argv[2][5:]
        else:
            file_directory = input("Enter File-Directory: ").strip()

        # Check if key directory is provided as a command line argument
        if len(argv) > 3 and argv[3].startswith("key"):
            key_directory = argv[3][4:]
        else:
            key_directory = input("Enter Key-Directory: ").strip()

        return action, file_directory, key_directory

    # Print error message if the action is invalid
    print("Error: invalid action specified")
    return 0, 0, 0
