# Define a roster with IP addresses and names as tuples within a tuple
roster = (
    ('ipns address one', 'Name one'),
    ('ipns address two', 'Name two'),
    ('etc', 'etc')
)

def print_roster(roster):
    """
    Print the IPNS addresses and names from the given roster.
    
    Parameters:
    roster (tuple): A tuple containing tuples of IPNS addresses and names.
    
    Returns:
    None
    """
    for address, name in roster:
        print(f"IPNS Address: {address}, Name: {name}")

def main():
    """
    Main function to demonstrate printing the roster.
    
    Returns:
    None
    """
    print_roster(roster)

if __name__ == "__main__":
    main()
