def printt(x, *kargs):

def encrypt(string, key):
    """
    Encrypt a string using a key with XOR encryption.
    
    Args:
    string (str): The input string to encrypt.
    key (str): The key to use for encryption.
    
    Returns:
    str: The encrypted string.
    """
    return "".join(chr(ord(i) ^ ord(j)) for (i, j) in zip(string, key))
