def tuple_sum(A, B):
    '''
    Sum corresponding elements of two lists of tuples.

    Input:
      - A: a list of tuples
      - B: a list of tuples
    Output:
      - list of tuples where each tuple is the element-wise sum of the corresponding tuples in A and B

    Examples:
    >>> tuple_sum([(1,2), (10,20)],[(3,4), (30,40)])
    [(4, 6), (40, 60)]
    >>> tuple_sum([(0,1),(-1,0),(2,2)], [(3,4),(5,6),(7,8)])
    [(3, 5), (4, 6), (9, 10)]
    '''
    return [(a[0] + b[0], a[1] + b[1]) for a, b in zip(A, B)]

def inv_dict(d):
    '''
    Invert a dictionary.

    Input:
      - d: dictionary representing an invertible function f
    Output:
      - dictionary representing the inverse of f

    Examples:
    >>> inv_dict({'goodbye': 'au revoir', 'thank you': 'merci'})
    {'au revoir': 'goodbye', 'merci': 'thank you'}
    '''
    return {v: k for k, v in d.items()}

def row(p, n):
    '''
    Create a list of n elements starting from p.

    Input:
      - p: the starting number
      - n: the number of elements
    Output:
      - n-element list where the i-th element is p + i

    Examples:
    >>> row(10, 4)
    [10, 11, 12, 13]
    '''
    return [p + i for i in range(n)]

# List comprehension using the row function
comprehension_with_row = [row(i, 20) for i in range(15)]

# List comprehension without using the row function
comprehension_without_row = [[i + j for j in range(20)] for i in range(15)]

# Assuming f and g are defined as follows for the sake of example:
from random import randint

def f(x):
    return x % 2  # Example function that returns 0 for even numbers and 1 for odd numbers

def g(x):
    return randint(0, 2)  # Example function that returns a random integer 0, 1, or 2

# Calculate probabilities
def calculate_probabilities(sample_size=10000):
    f_results = [f(randint(0, 10)) for _ in range(sample_size)]
    g_results = [g(randint(0, 10)) for _ in range(sample_size)]

    Pr_f_is_even = f_results.count(0) / sample_size
    Pr_f_is_odd = f_results.count(1) / sample_size
    Pr_g_is_1 = g_results.count(1) / sample_size
    Pr_g_is_0or2 = (g_results.count(0) + g_results.count(2)) / sample_size

    return Pr_f_is_even, Pr_f_is_odd, Pr_g_is_1, Pr_g_is_0or2
