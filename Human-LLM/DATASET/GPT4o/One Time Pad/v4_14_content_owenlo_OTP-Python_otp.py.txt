"""Randomized One-Time Pad encryption and decryption functions in Python."""

import random

CHARSET = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"

def main():
    """
    Demonstrates the usage of the encrypt and decrypt functions.
    """
    vector = "Once more into the fray."
    otp, encrypted_message = encrypt(vector)
    decrypted_message = decrypt(otp, encrypted_message)
    
    print(f"Test Vector: {vector}")
    print(f"OTP: {otp}")
    print(f"Encrypted: {encrypted_message}")
    print(f"Decrypted: {decrypted_message}")

def encrypt(plaintext):
    """
    Encrypts the given plaintext using a randomly generated one-time pad.
    
    Args:
        plaintext (str): The plaintext value to encrypt.
        
    Returns:
        tuple: A tuple containing the one-time pad and the encrypted message.
    """
    otp = "".join(random.sample(CHARSET, len(CHARSET)))
    result = "".join(otp[CHARSET.find(c)] for c in plaintext.upper() if c in CHARSET)
    return otp, result

def decrypt(otp, secret):
    """
    Decrypts the given secret using the provided one-time pad.
    
    Args:
        otp (str): The one-time pad used when the secret value was encrypted.
        secret (str): The value to be decrypted.
        
    Returns:
        str: The decrypted message.
    """
    result = "".join(CHARSET[otp.find(c)] for c in secret.upper() if c in otp)
    return result

if __name__ == "__main__":
    main()
