def make_vernam_cipher(text, key):
    """
    Returns the Vernam Cipher for the given string and key.

    Args:
        text (str): The text to be encrypted or decrypted.
        key (str): The key used for the Vernam cipher.

    Returns:
        str: The resulting cipher text.
    """
    cipher_text = []
    key_index = 0

    for char in text:
        cipher_char = chr(ord(char) ^ ord(key[key_index]))
        cipher_text.append(cipher_char)
        key_index = (key_index + 1) % len(key)
    
    return ''.join(cipher_text)

MY_KEY = "cvwopslweinedvq9fnasdlkfn2"

def main():
    """
    Main function to handle user input for encryption and decryption
    using the Vernam cipher.
    """
    while True:
        print("\n\n--- Vernam Cipher ---")
        plaintext = input("Enter text to encrypt: ")
        cipher = make_vernam_cipher(plaintext, MY_KEY)
        print("Cipher text:", cipher)
        decrypted_text = make_vernam_cipher(cipher, MY_KEY)
        print("Decrypted text:", decrypted_text)

if __name__ == "__main__":
    main()
