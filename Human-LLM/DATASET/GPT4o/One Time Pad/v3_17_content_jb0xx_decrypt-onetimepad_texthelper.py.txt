def hex_to_nums(msg):
    """
    Converts a message of hexadecimals to a list of numbers.

    Args:
    msg (str): String containing a stream of hexadecimal bytes.

    Returns:
    list of int: Stream of integers corresponding to the hexadecimal input.
    """
    len_msg = len(msg) // 2
    return [16 * hex_num_helper(msg[2 * i]) + hex_num_helper(msg[2 * i + 1]) for i in range(len_msg)]

def nums_to_hex(nums):
    """
    Converts a list of integers to their 2-digit hexadecimal string equivalents.

    Args:
    nums (list of int): List of integers to be converted to hexadecimal strings.

    Returns:
    list of str: List of hexadecimal strings.
    """
    return [format(n, '02X') for n in nums]

def hex_num_helper(hex_val):
    """
    Converts a hexadecimal digit to its decimal equivalent.

    Args:
    hex_val (str): Single hexadecimal character.

    Returns:
    int: Decimal equivalent of the hexadecimal character.
    """
    return int(hex_val, 16)

def bytes_to_chars(bytes_array):
    """
    Converts a list of bytes to a list of characters.

    Args:
    bytes_array (list of int): List of bytes to be converted to characters.

    Returns:
    list of str: List of characters.
    """
    return [chr(b) for b in bytes_array]

def xor_chrs(chrs):
    """
    Computes the XOR of all characters in the input list.

    Args:
    chrs (list of str): List of characters to be XORed.

    Returns:
    str: Resulting character after XORing all input characters.
    """
    val = 0
    for c in chrs:
        val ^= ord(c)
    return chr(val)

# Example usage
if __name__ == "__main__":
    hex_message = "4A6F686E20446F65"
    nums = hex_to_nums(hex_message)
    print(f"Hex to Nums: {nums}")
    
    hex_vals = nums_to_hex(nums)
    print(f"Nums to Hex: {hex_vals}")
    
    chars = bytes_to_chars(nums)
    print(f"Bytes to Chars: {chars}")
    
    xor_result = xor_chrs(chars)
    print(f"XOR of Chars: {xor_result}")
