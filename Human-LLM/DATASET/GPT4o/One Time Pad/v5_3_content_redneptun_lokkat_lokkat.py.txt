import sys
from PyQt5 import QtWidgets
from config.config import getConfig
from gui.mainwindow import MainWindow

def initialize_application():
    """
    Initialize the Qt application with the given configuration.
    
    Returns:
    QtWidgets.QApplication: The initialized application instance.
    MainWindow: The main window of the application.
    """
    active_config = getConfig()
    app = QtWidgets.QApplication(sys.argv)
    window = MainWindow(active_config)
    return app, window

def main():
    """
    Main function to initialize and start the Qt application.
    """
    app, window = initialize_application()
    window.show()
    sys.exit(app.exec_())

if __name__ == "__main__":
    main()
