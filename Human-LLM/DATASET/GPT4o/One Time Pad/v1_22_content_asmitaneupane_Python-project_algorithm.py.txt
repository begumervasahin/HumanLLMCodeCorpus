def vernam(key, text):
    """
    Returns the Vernam Cipher for a given string and key.
    
    Parameters:
    key (str): The key used for the cipher.
    text (str): The text to be encrypted or decrypted.
    
    Returns:
    str: The resulting Vernam cipher text.
    """
    answer = ""
    p = 0
    for char in text:
        answer += chr(ord(char) ^ ord(key[p]))
        p += 1
        if p == len(key):
            p = 0
    return answer

def main():
    key = "SECRET"
    text = "HELLO WORLD"
    encrypted = vernam(key, text)
    print("Encrypted:", encrypted)
    
    decrypted = vernam(key, encrypted)
    print("Decrypted:", decrypted)

if __name__ == "__main__":
    main()
