import pandas as pd
import re

# Dictionary to store binning information for test data
test_bins_dict = {}

def preprocess(structure_file, df, num_of_intervals):
    """
    Preprocess the training data by setting attribute types, handling missing values, and performing discretization.
    
    :param structure_file: File containing attribute structure information
    :param df: DataFrame containing the training data
    :param num_of_intervals: Number of intervals for discretization
    :return: Preprocessed DataFrame
    """
    attribute_dict = set_attribute_dict(structure_file)
    df_no_missing_vals = handle_missing_values(df, attribute_dict)
    df_final = discretize_attributes(attribute_dict, df_no_missing_vals, num_of_intervals)
    return df_final

def preprocess_test(structure_file, df):
    """
    Preprocess the test data by setting attribute types, handling missing values, and performing discretization using training data bins.
    
    :param structure_file: File containing attribute structure information
    :param df: DataFrame containing the test data
    :return: Preprocessed DataFrame
    """
    attribute_dict = set_attribute_dict(structure_file)
    df_no_missing_vals = handle_missing_values(df, attribute_dict)
    df_final = discretize_test_attributes(test_bins_dict, df_no_missing_vals)
    return df_final

def set_attribute_dict(structure_file):
    """
    Set the attribute dictionary with attribute types based on the structure file.
    
    :param structure_file: File containing attribute structure information
    :return: Dictionary with attribute names as keys and types ('N' for numeric, 'C' for categorical) as values
    """
    attribute_dict = {}
    for line in structure_file:
        split_line = re.split(r'\s+', line)
        attribute_type = "N" if split_line[2] == "NUMERIC" else "C"
        attribute_dict[split_line[1]] = attribute_type
    return attribute_dict

def set_attribute_values_dict(structure_file):

    :param structure_file: File containing attribute structure information
    :return: Dictionary with attribute names as keys and possible values as values

def handle_missing_values(df, attribute_dict):
    """
    Handle missing values in the DataFrame based on attribute types.
    
    :param df: DataFrame containing the data
    :param attribute_dict: Dictionary with attribute names as keys and types as values
    :return: DataFrame with missing values handled
    """
    for key, attr_type in attribute_dict.items():
        if attr_type == "N":
            df[key].fillna(df.groupby("class")[key].transform("mean"), inplace=True)
        elif attr_type == "C":
            df[key].fillna(df[key].mode()[0], inplace=True)
    return df

def binning(col, k, key):
    """
    Perform binning on a column for discretization.
    
    :param col: Column to be binned
    :param k: Number of intervals for discretization
    :param key: Attribute name
    :return: Binned column
    """
    global test_bins_dict

    min_val = col.min()
    max_val = col.max()
    cut_points = [min_val + (i + 1) * (max_val - min_val) / k for i in range(k - 1) if min_val < (min_val + (i + 1) * (max_val - min_val) / k) < max_val]
    break_points = [min_val] + cut_points + [max_val]
    
    global labels
    labels = range(len(cut_points) + 1)
    
    test_bins_dict[key] = [break_points, labels]
    col_bin = pd.cut(col, bins=break_points, labels=labels, include_lowest=True)
    return col_bin

def binning_test(col, break_points, labels):
    """
    Perform binning on a column for test data using training data break points and labels.
    
    :param col: Column to be binned
    :param break_points: Break points for binning
    :param labels: Labels for the bins
    :return: Binned column
    """
    return pd.cut(col, bins=break_points, labels=labels, include_lowest=True)

def discretize_attributes(attribute_dict, df, num_of_intervals):
    """
    Discretize numeric attributes in the DataFrame.
    
    :param attribute_dict: Dictionary with attribute names as keys and types as values
    :param df: DataFrame containing the data
    :param num_of_intervals: Number of intervals for discretization
    :return: DataFrame with discretized attributes
    """
    for key, attr_type in attribute_dict.items():
        if attr_type == "N":
            df[key] = binning(df[key], num_of_intervals, key)
    return df

def discretize_test_attributes(test_bins_dict, df):
    """
    Discretize numeric attributes in the test DataFrame using training data bins.
    
    :param test_bins_dict: Dictionary with binning information from training data
    :param df: DataFrame containing the test data
    :return: DataFrame with discretized attributes
    """
    for key, (break_points, labels) in test_bins_dict.items():
        df[key] = binning_test(df[key], break_points, labels)
    return df
