import math
import random
import re
import glob
from collections import Counter, defaultdict
from machine_learning import split_data

def tokenize(message):
    """
    Convert a message to a set of unique lowercase words.
    
    Args:
    message (str): The message to tokenize.

    Returns:
    set: A set of unique lowercase words.
    """
    message = message.lower()
    return set(re.findall(r"[a-z0-9']+", message))

def count_words(training_set):
    """
    Count the occurrences of each word in spam and non-spam messages.
    
    Args:
    training_set (list): A list of tuples where each tuple contains a message and a label indicating spam or not.

    Returns:
    defaultdict: A dictionary where keys are words and values are lists of counts [spam_count, non_spam_count].
    """
    counts = defaultdict(lambda: [0, 0])
    for message, is_spam in training_set:
        for word in tokenize(message):
            counts[word][0 if is_spam else 1] += 1
    return counts

def word_probabilities(counts, total_spams, total_non_spams, k=0.5):
    """
    Calculate word probabilities given word counts, total spams, and total non-spams.
    
    Args:
    counts (dict): A dictionary of word counts.
    total_spams (int): Total number of spam messages.
    total_non_spams (int): Total number of non-spam messages.
    k (float): Smoothing parameter.

    Returns:
    list: A list of tuples containing word, probability of word given spam, and probability of word given non-spam.
    """
    return [
        (word, (spam_count + k) / (total_spams + 2 * k), (non_spam_count + k) / (total_non_spams + 2 * k))
        for word, (spam_count, non_spam_count) in counts.items()
    ]

def spam_probability(word_probs, message):
    """
    Calculate the probability of a message being spam.
    
    Args:
    word_probs (list): A list of word probabilities.
    message (str): The message to classify.

    Returns:
    float: The probability of the message being spam.
    """
    message_words = tokenize(message)
    log_prob_if_spam = log_prob_if_not_spam = 0.0

    for word, prob_if_spam, prob_if_not_spam in word_probs:
        if word in message_words:
            log_prob_if_spam += math.log(prob_if_spam)
            log_prob_if_not_spam += math.log(prob_if_not_spam)
        else:
            log_prob_if_spam += math.log(1.0 - prob_if_spam)
            log_prob_if_not_spam += math.log(1.0 - prob_if_not_spam)

    prob_if_spam = math.exp(log_prob_if_spam)
    prob_if_not_spam = math.exp(log_prob_if_not_spam)
    return prob_if_spam / (prob_if_spam + prob_if_not_spam)

class NaiveBayesClassifier:
    def __init__(self, k=0.5):
        self.k = k
        self.word_probs = []

    def train(self, training_set):
        """
        Train the Naive Bayes classifier.

        Args:
        training_set (list): A list of tuples where each tuple contains a message and a label indicating spam or not.
        """
        num_spams = sum(1 for _, is_spam in training_set if is_spam)
        num_non_spams = len(training_set) - num_spams

        word_counts = count_words(training_set)
        self.word_probs = word_probabilities(word_counts, num_spams, num_non_spams, self.k)

    def classify(self, message):
        """
        Classify a message as spam or not spam.

        Args:
        message (str): The message to classify.

        Returns:
        float: The probability of the message being spam.
        """
        return spam_probability(self.word_probs, message)

def get_subject_data(path):
    """
    Extract subject lines and labels from emails in the specified path.
    
    Args:
    path (str): Path to the directory containing email files.

    Returns:
    list: A list of tuples where each tuple contains a subject line and a label indicating spam or not.
    """
    data = []
    subject_regex = re.compile(r"^Subject:\s+")
    for filepath in glob.glob(path):
        is_spam = "ham" not in filepath
        with open(filepath, 'r', encoding='ISO-8859-1') as file:
            for line in file:
                if line.startswith("Subject:"):
                    subject = subject_regex.sub("", line).strip()
                    data.append((subject, is_spam))
    return data

def p_spam_given_word(word_prob):
    """
    Calculate the probability of spam given a word's probabilities.
    
    Args:
    word_prob (tuple): A tuple containing a word, its probability given spam, and its probability given non-spam.

    Returns:
    float: The probability of spam given the word.
    """
    word, prob_if_spam, prob_if_not_spam = word_prob
    return prob_if_spam / (prob_if_spam + prob_if_not_spam)

def train_and_test_model(path):
    """
    Train the Naive Bayes classifier and test it on the dataset.
    
    Args:
    path (str): Path to the directory containing email files.
    """
    data = get_subject_data(path)
    random.seed(0)
    train_data, test_data = split_data(data, 0.75)

    classifier = NaiveBayesClassifier()
    classifier.train(train_data)

    classified = [(subject, is_spam, classifier.classify(subject)) for subject, is_spam in test_data]

    counts = Counter((is_spam, spam_prob > 0.5) for _, is_spam, spam_prob in classified)
    print(counts)

    classified.sort(key=lambda row: row[2])
    spammiest_hams = [row for row in classified if not row[1]][-5:]
    hammiest_spams = [row for row in classified if row[1]][:5]

    print("spammiest_hams:", spammiest_hams)
    print("hammiest_spams:", hammiest_spams)

    words = sorted(classifier.word_probs, key=p_spam_given_word)
    spammiest_words = words[-5:]
    hammiest_words = words[:5]

    print("spammiest_words:", spammiest_words)
    print("hammiest_words:", hammiest_words)

if __name__ == "__main__":
    train_and_test_model(r"./emails/*/*")
