import time
import librosa
import glob
import numpy as np
from sklearn.metrics import confusion_matrix, accuracy_score
from sklearn.naive_bayes import GaussianNB

start = time.time()

# Function to extract MFCC features from an audio file
def extract_feature_audio(file_path):
    y, _ = librosa.load(file_path)
    mfcc = librosa.feature.mfcc(y)
    mfcc /= np.amax(np.absolute(mfcc))
    return np.ndarray.flatten(mfcc)[:300]

# Function to generate features and labels from audio files
def generate_features_and_labels():
    all_features = []
    all_labels = []
    ganjils = ['satu', 'tiga', 'lima', 'tujuh', 'sembilan']
    for ganjil in ganjils:
        sound_files = glob.glob(f'dataset/{ganjil}/*.wav')
        print(f'Processing {len(sound_files)} sound files in {ganjil} ...')
        for file_path in sound_files:
            features = extract_feature_audio(file_path)
            all_features.append(features)
            all_labels.append(ganjil)
    return np.stack(all_features), all_labels

# Function to apply softmax normalization
def softmax(x):
    e_x = np.exp(x - np.max(x))
    return e_x / e_x.sum(axis=0)

# Generate features and labels
features, labels = generate_features_and_labels()
features = softmax(features)
all_data = np.column_stack((features, labels))

# Shuffle and split data into training and test sets
np.random.seed(200)
np.random.shuffle(all_data)
training_split = 0.9
split_index = int(len(all_data) * training_split)
train_data, test_data = all_data[:split_index, :], all_data[split_index:, :]

train_input = train_data[:, :-1].astype(np.float)
train_labels = train_data[:, -1]
test_input = test_data[:, :-1].astype(np.float)
test_labels = test_data[:, -1]

# Function to train and evaluate the Naive Bayes classifier
def train_and_evaluate_classifier():
    nb = GaussianNB()
    nb.fit(train_input, train_labels)
    predictions = nb.predict(test_input)
    print("Result Prediction: ", predictions)
    print("Real label data: ", test_labels)
    print('\nEvaluate Naive Bayes')
    print(confusion_matrix(test_labels, predictions))
    print("\nAccuracy: ", accuracy_score(test_labels, predictions))

# Train and evaluate the classifier
train_and_evaluate_classifier()

end = time.time()
print("Execution time: ", end - start)
