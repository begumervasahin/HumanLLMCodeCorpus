import csv
import fungsi as fg

def load_data(file_path):
    """
    Load data from a CSV file.

    Parameters:
    file_path (str): Path to the CSV file.

    Returns:
    list: List of rows from the CSV file.
    """
    data = []
    with open(file_path) as file:
        reader = csv.reader(file, delimiter=',')
        next(reader)  # Skip the header
        for row in reader:
            data.append(row)
    return data

def calculate_class_probabilities(train_data):
    """
    Calculate the class probabilities for the training data.

    Parameters:
    train_data (list): List of rows from the training data.

    Returns:
    tuple: Class probabilities and counts for '>50K' and '<=50K'.
    """
    PC1, nC1 = fg.PC(train_data, '>50K')
    PC2, nC2 = fg.PC(train_data, '<=50K')
    return PC1, nC1, PC2, nC2

def predict(test_data, train_data, PC1, nC1, PC2, nC2, num_features):
    """
    Predict the class labels for the test data.

    Parameters:
    test_data (list): List of rows from the test data.
    train_data (list): List of rows from the training data.
    PC1 (float): Class probability for '>50K'.
    nC1 (int): Count of '>50K' instances in the training data.
    PC2 (float): Class probability for '<=50K'.
    nC2 (int): Count of '<=50K' instances in the training data.
    num_features (int): Number of features (excluding the label).

    Returns:
    list: List of predicted class labels.
    """
    predictions = []
    for test_instance in test_data:
        PC1X, PC2X = 1, 1
        PXC1, PXC2 = 1, 1
        for feature_index in range(1, num_features):
            prob_c1, prob_c2 = fg.PXC(train_data, feature_index, nC1, nC2, test_instance)
            PXC1 *= prob_c1
            PXC2 *= prob_c2
        PC1X = PXC1 * PC1
        PC2X = PXC2 * PC2
        if PC1X > PC2X:
            predictions.append('>50K')
        else:
            predictions.append('<=50K')
    return predictions

def write_predictions(predictions, output_file):
    """
    Write predictions to a CSV file.

    Parameters:
    predictions (list): List of predicted class labels.
    output_file (str): Path to the output CSV file.
    """
    with open(output_file, 'w', newline='') as file:
        writer = csv.writer(file, dialect='excel')
        for result in predictions:
            writer.writerow([result])

def main():
    """
    Main function to load data, calculate probabilities, predict classes, and write predictions.
    """
    train_data = load_data('TrainsetTugas1ML.csv')
    test_data = load_data('TestsetTugas1ML.csv')

    PC1, nC1, PC2, nC2 = calculate_class_probabilities(train_data)
    
    num_features = 8
    predictions = predict(test_data, train_data, PC1, nC1, PC2, nC2, num_features)

    write_predictions(predictions, 'TebakanTugas1ML.csv')

if __name__ == "__main__":
    main()
