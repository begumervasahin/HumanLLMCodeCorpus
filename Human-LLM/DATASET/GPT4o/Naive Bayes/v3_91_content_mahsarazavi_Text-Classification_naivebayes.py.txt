import re
import random
from collections import Counter, OrderedDict
from nltk.corpus import stopwords
import math

# Define file paths
class1file = "raw_data_sport.txt"
class2file = "raw_data_politics.txt"

# Dictionaries to store important words for each class
impWOrd85 = {}
impWOrd77 = {}

def random_partition(text, percent):
    """
    Randomly partition the text into training and testing sets based on the given percentage.

    Parameters:
    text (str): The input text.
    percent (float): The percentage of words to be included in the test set.

    Returns:
    tuple: A tuple containing the remaining (training) text and the partitioned (testing) text.
    """
    words = text.split()
    partition_size = int(len(words) * percent)
    partition = random.sample(words, partition_size)
    remaining = [word for word in words if word not in partition]
    return ' '.join(remaining), ' '.join(partition)

def clean_text(text):
    """
    Clean the input text by converting to lowercase, removing digits and stopwords.

    Parameters:
    text (str): The input text.

    Returns:
    str: The cleaned text.
    """
    text = text.lower()
    text = re.sub(r'\d+', '', text)
    text = text.replace("\n", " ").replace("-", "")
    stop_words = set(stopwords.words('english'))
    text = ' '.join(word for word in text.split() if word not in stop_words)
    return text

def normal_pdf(x, mean, std):
    """
    Calculate the normal probability density function for the given value.

    Parameters:
    x (float): The value.
    mean (float): The mean of the distribution.
    std (float): The standard deviation of the distribution.

    Returns:
    float: The calculated probability density.
    """
    return (1 / (math.sqrt(2 * math.pi) * std)) * math.exp(-((x - mean) ** 2) / (2 * std ** 2))

def calculate_probability(word, class_words, class_word_count, total_words, distinct_words):
    """
    Calculate the log probability of a word given the class.

    Parameters:
    word (str): The word to calculate the probability for.
    class_words (Counter): The word counts for the class.
    class_word_count (int): The total word count for the class.
    total_words (int): The total word count for the dataset.
    distinct_words (int): The number of distinct words in the class.

    Returns:
    float: The calculated log probability.
    """
    word_count = class_words.get(word, 0)
    return math.log10((word_count + 1) / (class_word_count + distinct_words))

def calculate_p(text, pclass1, pclass2, countallword1, countallword2, count_dist_words1, count_dist_words2, class1words, class2words):
    """
    Calculate the class probabilities for the given text and determine the class.

    Parameters:
    text (str): The input text.
    pclass1 (float): The prior probability for class 1.
    pclass2 (float): The prior probability for class 2.
    countallword1 (int): The total word count for class 1.
    countallword2 (int): The total word count for class 2.
    count_dist_words1 (int): The number of distinct words in class 1.
    count_dist_words2 (int): The number of distinct words in class 2.
    class1words (Counter): The word counts for class 1.
    class2words (Counter): The word counts for class 2.

    Returns:
    int: The predicted class (1 or 2).
    """
    text = clean_text(text).split()
    p_word_in_class1 = pclass1
    p_word_in_class2 = pclass2
    p_max1, p_max2 = -float('inf'), -float('inf')
    imp_word1, imp_word2 = '', ''

    for word in text:
        if len(word) < 3:
            continue
        p1 = calculate_probability(word, class1words, countallword1, countallword1, count_dist_words1)
        p2 = calculate_probability(word, class2words, countallword2, countallword2, count_dist_words2)

        if p1 > p_max1:
            p_max1 = p1
            imp_word1 = word
        if p2 > p_max2:
            p_max2 = p2
            imp_word2 = word

        p_word_in_class1 += p1
        p_word_in_class2 += p2

    if p_word_in_class1 > p_word_in_class2:
        impWOrd77[imp_word1] = p_max1
        return 1
    else:
        impWOrd85[imp_word2] = p_max2
        return 2

def count_words(train1, train2):
    """
    Count the words in the training datasets for each class.

    Parameters:
    train1 (str): The training text for class 1.
    train2 (str): The training text for class 2.

    Returns:
    tuple: A tuple containing word counts for class 1 and class 2.
    """
    class1words = Counter(re.findall(r'\w+', train1))
    class2words = Counter(re.findall(r'\w+', train2))
    return class1words, class2words

def stringify_every_n_words(arr, n=50):
    """
    Create a list of strings where each string contains up to n words.

    Parameters:
    arr (list): The input list of words.
    n (int): Maximum number of words per string.

    Returns:
    list: A list of strings.
    """
    LEN = len(arr) // n
    return [' '.join(arr[i * n:(i + 1) * n]) for i in range(LEN)] + [' '.join(arr[LEN * n:])]

def classifier():
    """
    Main function to perform text classification using Naive Bayes.
    """
    # Read and clean the datasets
    with open(class1file, 'r', encoding='utf8', errors='ignore') as f:
        class1 = f.read()
    with open(class2file, 'r', encoding='utf8', errors='ignore') as f:
        class2 = f.read()

    class1, class2 = clean_text(class1), clean_text(class2)

    # Partition the data into training and testing sets
    train1, test1 = random_partition(class1, 0.10)
    train2, test2 = random_partition(class2, 0.10)

    # Count words in training sets
    class1words, class2words = count_words(train1, train2)
    countallword1, countallword2 = sum(class1words.values()), sum(class2words.values())
    countDistWords1, countDistWords2 = len(class1words), len(class2words)

    # Calculate prior probabilities
    countSentence1 = len(re.findall(r'\w+', train1))
    countSentence2 = len(re.findall(r'\w+', train2))
    pclass1 = countSentence1 / (countSentence1 + countSentence2)
    pclass2 = countSentence2 / (countSentence1 + countSentence2)

    # Prepare test sentences
    sentence1 = stringify_every_n_words(re.findall(r'\w+', test1))
    sentence2 = stringify_every_n_words(re.findall(r'\w+', test2))

    # Initialize counts for evaluation metrics
    tp, fn, fp, tn = 0, 0, 0, 0

    # Classify test sentences from class 1
    for sentence in sentence1:
        c = calculate_p(sentence, pclass1, pclass2, countallword1, countallword2, countDistWords1, countDistWords2, class1words, class2words)
        if c == 1:
            tp += 1
        else:
            fn += 1

    # Classify test sentences from class 2
    for sentence in sentence2:
        c = calculate_p(sentence, pclass1, pclass2, countallword1, countallword2, countDistWords1, countDistWords2, class1words, class2words)
        if c == 2:
            tn += 1
        else:
            fp += 1

    # Print evaluation results
    print(f'False Positives: {fp}, True Positives: {tp}, False Negatives: {fn}, True Negatives: {tn}')
    precision = tp / (tp + fp) if (tp + fp) > 0 else 0
    recall = tp / (tp + fn) if (tp + fn) > 0 else 0
    print(f'Precision: {precision}, Recall: {recall}')
    print('Important words in class 1:', OrderedDict(sorted(impWOrd77.items(), key=lambda x: x[1], reverse=True)))
    print('Important words in class 2:', OrderedDict(sorted(impWOrd85.items(), key=lambda x: x[1], reverse=True)))

if __name__ == "__main__":
    classifier()
