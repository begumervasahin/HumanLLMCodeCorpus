import numpy as np
from sklearn.metrics import accuracy_score
from sklearn.model_selection import train_test_split
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.naive_bayes import MultinomialNB
from app.src.spam_classify import HAM_DATA, SPAM_DATA, load_email, TEST_DATA
from app.src.import_data import crawl_data

# Crawl data
crawl_data()
crawl_data(test=True)

# Constants
LIMIT = 2000

# Load emails
ham_emails = [email for email in load_email(HAM_DATA, limit=LIMIT)]
spam_emails = [email for email in load_email(SPAM_DATA, limit=LIMIT)]
test_emails = [email for email in load_email(TEST_DATA)]

# Prepare dataset
X = np.array(ham_emails + spam_emails)
Y = np.array([0] * len(ham_emails) + [1] * len(spam_emails))

# Vectorize text data using TF-IDF
vectorizer = TfidfVectorizer(stop_words='english')
X_prepared = vectorizer.fit_transform(X)

# Split data into training and test sets
X_train, X_test, Y_train, Y_test = train_test_split(X_prepared, Y, test_size=0.2, random_state=42)

# Train Multinomial Naive Bayes classifier
mnb = MultinomialNB(alpha=0.2)
mnb.fit(X_train, Y_train)

# Evaluate on training and test set
Y_train_pred = mnb.predict(X_train)
Y_test_pred = mnb.predict(X_test)
train_accuracy = accuracy_score(Y_train, Y_train_pred)
test_accuracy = accuracy_score(Y_test, Y_test_pred)

print(f"Training Accuracy: {train_accuracy:.4f}")
print(f"Test Accuracy: {test_accuracy:.4f}")

# Test on new data
X_test_emails = np.array(test_emails)
X_test_emails_prepared = vectorizer.transform(X_test_emails)
test_emails_pred = mnb.predict(X_test_emails_prepared)

# If the test emails are labeled as spam (1)
Y_test_emails = np.array([1] * len(test_emails))
test_emails_accuracy = accuracy_score(Y_test_emails, test_emails_pred)
print(f"Test Emails Accuracy: {test_emails_accuracy:.4f}")
