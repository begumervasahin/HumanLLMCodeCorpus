import glob
import pickle
import numpy as np
from multiprocessing import Process, Lock
from nltk.stem import WordNetLemmatizer

def compute_probability(vocabulary, ignore=0):
    """
    Compute conditional and priori probabilities for spam classification.

    Args:
    vocabulary (dict): Vocabulary dictionary.
    ignore (int): Index of the partition to ignore for training.

    Returns:
    tuple: Conditional probabilities and priori probabilities.
    """
    conditional_probability = [{key: 1 for key in vocabulary.keys()}, {key: 1 for key in vocabulary.keys()}]
    priori_probability = [0, 0]
    lemmatizer = WordNetLemmatizer()
    
    for i in range(1, 11):
        if i != ignore:
            filenames = glob.glob(f"lingspam/part{i}/*.txt")
            for filename in filenames:
                flag = {key: True for key in vocabulary.keys()}
                label = int("spmsg" in filename)
                priori_probability[label] += 1
                
                with open(filename, "rb") as f:
                    text = f.readlines()
                    for line in text:
                        for word in line.decode().split():
                            word = lemmatizer.lemmatize(word)
                            if word in vocabulary and flag[word]:
                                conditional_probability[label][word] += 1
                                flag[word] = False
    
    for label in range(2):
        conditional_probability[label] = {key: value / priori_probability[label] for key, value in conditional_probability[label].items()}
    
    total = sum(priori_probability)
    priori_probability = [count / total for count in priori_probability]
    
    return conditional_probability, priori_probability

def classify(conditional_probability, priori_probability, filename):
    """
    Classify a text file as spam or ham.

    Args:
    conditional_probability (list): List of dictionaries for conditional probabilities.
    priori_probability (list): List of priori probabilities.
    filename (str): Path to the file to be classified.

    Returns:
    int: 1 for spam, 0 for ham.
    """
    spam, ham = 1, 1
    words = set(conditional_probability[0].keys())
    lemmatizer = WordNetLemmatizer()
    
    with open(filename, "rb") as f:
        text = f.readlines()
        for line in text:
            for word in line.decode().split():
                word = lemmatizer.lemmatize(word)
                if word in words:
                    ham *= conditional_probability[0][word]
                    spam *= conditional_probability[1][word]
                    words.remove(word)
    
    for word in words:
        ham *= 1 - conditional_probability[0][word]
        spam *= 1 - conditional_probability[1][word]
    
    ham *= priori_probability[0]
    spam *= priori_probability[1]
    
    return 1 if spam > ham else 0

def test(conditional_probability, priori_probability, folder):
    """
    Test the classification performance on a specific folder.

    Args:
    conditional_probability (list): List of dictionaries for conditional probabilities.
    priori_probability (list): List of priori probabilities.
    folder (int): Index of the partition to test.

    Returns:
    tuple: True positives, false positives, true negatives, false negatives.
    """
    filenames = glob.glob(f"lingspam/part{folder}/*.txt")
    TP, FP, TN, FN = 0, 0, 0, 0
    
    for filename in filenames:
        label = int("spmsg" in filename)
        predicted_label = classify(conditional_probability, priori_probability, filename)
        
        if label == 1 and predicted_label == 1:
            TP += 1
        elif label == 1 and predicted_label == 0:
            FN += 1
        elif label == 0 and predicted_label == 0:
            TN += 1
        else:
            FP += 1
    
    return TP, FP, TN, FN

def validate(vocabulary, partition_index, lock):
    """
    Validate the classifier on a specific partition.

    Args:
    vocabulary (dict): Vocabulary dictionary.
    partition_index (int): Index of the partition to validate.
    lock (Lock): Lock for synchronizing output.
    """
    conditional_probability, priori_probability = compute_probability(vocabulary, partition_index)
    TP, FP, TN, FN = test(conditional_probability, priori_probability, partition_index)
    
    with lock:
        print(f"{partition_index} , {TP} , {FP} , {TN} , {FN} ,")

if __name__ == '__main__':
    for v in range(1, 5):
        vocabulary_file = f"vocabulary{v}.pickle"
        print(vocabulary_file)
        
        with open(vocabulary_file, "rb") as f:
            vocabulary = pickle.load(f)
        
        lock = Lock()
        processes = [Process(target=validate, args=(vocabulary, i, lock)) for i in range(1, 11)]
        
        for process in processes:
            process.start()
        
        for process in processes:
            process.join()
