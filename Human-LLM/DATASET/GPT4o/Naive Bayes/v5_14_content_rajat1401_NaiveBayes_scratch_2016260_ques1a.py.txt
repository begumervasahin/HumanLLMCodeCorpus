import numpy as np
import pandas as pd
import math
import matplotlib.pyplot as plt
import pickle
from mlxtend.data import loadlocal_mnist

# Function to load MNIST data
def load_mnist_data(images_path, labels_path):
    return loadlocal_mnist(images_path=images_path, labels_path=labels_path)

# Function to filter binary labels (1 or 2) and adjust them to 0 or 1
def filter_binary_labels(labels):
    return [label - 1 for label in labels if label in [1, 2]]

# Function to calculate priors
def calculate_priors(y_train_binary):
    a = y_train_binary.count(0)
    b = y_train_binary.count(1)
    prior_trouser = a / (a + b)
    prior_pullover = b / (a + b)
    return prior_trouser, prior_pullover, a, b

# Function to normalize counts
def normalize_counts(counts, a, b):
    for i in range(len(counts)):
        counts[i] /= [a, b]
    return counts

# Function to predict using Naive Bayes with different thresholds
def naive_bayes_predict(x_test, thresholds, counttrain0, counttrain1, prior_trouser, prior_pullover):
    y_pred = np.zeros((len(thresholds), len(x_test)))
    for k, threshold in enumerate(thresholds):
        for i, x in enumerate(x_test):
            pdt1 = sum(math.log(counttrain1[j][0 if x[j] == 1 else 1]) for j in range(len(x)))
            pdt2 = sum(math.log(counttrain0[j][0 if x[j] == 1 else 1]) for j in range(len(x)))
            prob0 = (pdt1 * prior_trouser) / (pdt1 * prior_trouser + pdt2 * prior_pullover)
            y_pred[k][i] = 0 if prob0 <= threshold else 1
    return y_pred

# Function to evaluate model and plot ROC curve
def evaluate_and_plot_roc(y_pred, y_test_binary, thresholds):
    tparray = []
    fparray = []

    for i in range(len(thresholds)):
        tp = fp = fn = tn = 0
        for j, y_true in enumerate(y_test_binary):
            y_hat = y_pred[i][j]
            if y_hat == 0 and y_true == 0:
                tp += 1
            elif y_hat == 1 and y_true == 1:
                tn += 1
            elif y_hat == 0 and y_true == 1:
                fp += 1
            elif y_hat == 1 and y_true == 0:
                fn += 1

        tpr = tp / (tp + fn)
        fpr = fp / (fp + tn)
        tparray.append(tpr)
        fparray.append(fpr)

        print(f"CASE {i+1} :-")
        print(f"Confusion Matrix:\n[[{tp} {fp}]\n [{fn} {tn}]]")
        print(f"The precision is: {tp / (tp + fp):.4f}")
        print(f"The recall is: {tpr:.4f}")

    plt.plot(fparray, tparray)
    plt.ylabel("True Positive Rate")
    plt.xlabel("False Positive Rate")
    plt.title("ROC Curve")
    plt.show()

# Load MNIST data
X_train, y_train = load_mnist_data('./train-images.idx3-ubyte', './train-labels.idx1-ubyte')
X_test, y_test = load_mnist_data('./t10k-images.idx3-ubyte', './t10k-labels.idx1-ubyte')

# Filter labels for binary classification (1 or 2)
y_train_binary = filter_binary_labels(y_train)
y_test_binary = filter_binary_labels(y_test)

# Load preprocessed train and test data
with open('./bintrain.pkl', 'rb') as f:
    x_train = pickle.load(f)
with open('./bintest.pkl', 'rb') as f:
    x_test = pickle.load(f)

# Load count data
counttrain0 = np.load('./counttrain0.npy')
counttrain1 = np.load('./counttrain1.npy')

# Calculate priors
prior_trouser, prior_pullover, a, b = calculate_priors(y_train_binary)

# Normalize counts
counttrain0 = normalize_counts(counttrain0, a, b)
counttrain1 = normalize_counts(counttrain1, a, b)

print("The prior trouser is:", prior_trouser)
print("The prior pullover is:", prior_pullover)

# Predict using Naive Bayes with different thresholds
thresholds = [0.3, 0.4, 0.5, 0.6, 0.7]
y_pred = naive_bayes_predict(x_test, thresholds, counttrain0, counttrain1, prior_trouser, prior_pullover)

print("Lengths:", len(y_pred), len(y_test_binary))
print(y_pred[:, 0], y_pred[:, 1])

# Evaluate model and plot ROC curve
evaluate_and_plot_roc(y_pred, y_test_binary, thresholds)
