import os
import csv
import string
from nltk.corpus import stopwords

def save(filename, dictionary):
    """
    Save the dictionary to a CSV file.

    Parameters:
    filename (str): The name of the file to save the dictionary.
    dictionary (dict): The dictionary to save.
    """
    with open(filename, "w", newline='') as file:
        writer = csv.writer(file)
        for key, val in dictionary.items():
            writer.writerow([key, val])

def load(filename):
    """
    Load a dictionary from a CSV file.

    Parameters:
    filename (str): The name of the file to load the dictionary from.

    Returns:
    dict: The loaded dictionary.
    """
    dictionary = {}
    with open(filename, mode='r') as file:
        reader = csv.reader(file)
        for rows in reader:
            if len(rows) == 2:
                dictionary[rows[0]] = float(rows[1])
    return dictionary

def clean_text(text):
    """
    Clean the text by converting to lowercase, removing punctuation and stopwords.

    Parameters:
    text (str): The text to clean.

    Returns:
    list: A list of cleaned words.
    """
    text = text.lower()
    translation_table = str.maketrans('', '', string.punctuation + '\0')
    stop_words = set(stopwords.words('english'))
    stop_words.add("subject")

    text = text.replace('\n', ' ')
    text = text.translate(translation_table)
    words = text.split()
    cleaned_words = [word for word in words if word not in stop_words]

    return cleaned_words

def train(directory):
    """
    Train the classifier by reading all files in the given directory,
    cleaning the text, and counting word frequencies.

    Parameters:
    directory (str): The directory containing the training files.

    Returns:
    dict: A dictionary with word frequencies.
    """
    files = [f for f in os.listdir(directory) if os.path.isfile(os.path.join(directory, f))]
    word_counts = {}
    total_words = 0

    for file in files:
        with open(os.path.join(directory, file), errors="ignore") as f:
            text = f.read()
        
        words = clean_text(text)
        for word in words:
            total_words += 1
            word_counts[word] = word_counts.get(word, 0) + 1

    num_files = len(files)
    for word in word_counts:
        word_counts[word] = float(word_counts[word]) / num_files

    word_counts["fileCount"] = num_files
    return word_counts

def classify(text, word_freqs, prior_prob):
    """
    Classify the given text using the trained dictionary and prior probability.

    Parameters:
    text (str): The text to classify.
    word_freqs (dict): The word frequency dictionary.
    prior_prob (float): The prior probability.

    Returns:
    float: The calculated probability.
    """
    words = clean_text(text)
    probability = 1

    for word in words:
        if word in word_freqs:
            probability *= word_freqs[word]
        else:
            probability *= 1 / (word_freqs["fileCount"] + len(word_freqs) + 1)

    probability *= prior_prob
    return probability

def train_and_save():
    """
    Train the classifier on the 'spam' and 'ham' directories and save the results.
    """
    save("spam_train_results.csv", train("spam"))
    save("ham_train_results.csv", train("ham"))

def test():
    """
    Test the classifier by classifying files in the 'input' directory.
    """
    ham_dict = load("ham_train_results.csv")
    spam_dict = load("spam_train_results.csv")
    files = [f for f in os.listdir("input") if os.path.isfile(os.path.join("input", f))]

    spam_count = 0
    ham_count = 0

    for file in files:
        with open(os.path.join("input", file), errors="ignore") as f:
            text = f.read()
        
        ham_prob = classify(text, ham_dict, 0.67)
        spam_prob = classify(text, spam_dict, 0.33)

        if ham_prob <= spam_prob:
            spam_count += 1
        else:
            ham_count += 1

    print(f"Spam: {spam_count}")
    print(f"Ham: {ham_count}")

if __name__ == "__main__":
    train_and_save()
    test()
