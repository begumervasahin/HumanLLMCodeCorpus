import cv2
import numpy as np
import glob
import os
import matplotlib.pyplot as plt

def calculate_crv(base_dir, step_size):
    """
    Calculate and plot the Channel Response Variability (CRV) over a series of 
    singularity index images using a sliding window approach.

    Parameters:
    base_dir (str): The base directory containing the .TIF images.
    step_size (int): The size of the sliding window.
    """
    
    # Get a sorted list of .TIF images in the directory
    psi_images = sorted(glob.glob(os.path.join(base_dir, '*/*.TIF')))
    
    # Calculate the number of windows
    number_sections = (len(psi_images) - step_size) + 1
    
    # Print information about the number of images and windows
    print(f'Number of singularity index images: {len(psi_images)}')
    print(f'Number of windows: {number_sections}')
    print(f'Step size: {step_size}')
    
    # Read the first image to get its shape
    I1 = cv2.imread(psi_images[0], cv2.IMREAD_UNCHANGED)
    
    # Initialize an array to store the singularity index images
    psi_array = np.zeros((step_size, I1.shape[0], I1.shape[1]), dtype='float32')
    
    # List to store the average CRV for each window
    averages = []
    xval = list(range(number_sections))
    
    # Iterate through each window
    for i in range(number_sections):
        # Populate psi_array with images in the current window
        for j in range(i, i + step_size):
            psi = cv2.imread(psi_images[j], cv2.IMREAD_UNCHANGED)
            psi_array[j - i, :, :] = psi
        
        # Calculate the channel variance and average CRV for the window
        channel_var = np.var(psi_array, axis=0)
        avg = np.mean(channel_var)
        print(f'Window {i}')
        averages.append(avg)
        
        # Reset the psi_array for the next window
        psi_array.fill(0)
    
    # Plot the results
    plt.scatter(xval, averages)
    plt.xlabel('Window number')
    plt.ylabel('Average CRV for window images')
    
    # Save the plot in both TIF and EPS formats
    output_filename_base = os.path.join(base_dir, f"demo_slidingwindow_{step_size}")
    plt.savefig(f"{output_filename_base}.TIF")
    plt.savefig(f"{output_filename_base}.eps", format='eps')
    plt.close()
    
    print('Completed')

def main():
    base_dir = '/home/engrla/rivamap/SIESD_related/SlidingWindow/demo_PSIs'
    
    # Run the CRV calculation for different step sizes
    for step_size in range(15, 60, 5):
        calculate_crv(base_dir, step_size)

if __name__ == "__main__":
    main()
