import numpy as np

multifile_name = 'savefiles/640test_'
filenumbers = list(range(1, 60))  # Assuming filenumbers are sequential from 1 to 59
output_filename = '640_final.dat'

def process_files(multifile_name, filenumbers, output_filename):
    with open(output_filename, 'w') as mainfile:
        prev_time = 0.0
        prev_step = 0

        for f in filenumbers:
            with open(f"{multifile_name}{f}.dat", 'r') as singlefile:
                i = 0
                newtime = 0.0
                newstep = 0
                savetime = 0.0

                for line in singlefile:
                    split_line = line.split('[')
                    temp = split_line[0]
                    time = float(temp.split(' ')[0])
                    step = int(temp.split(' ')[1])

                    if f == 1:
                        newtime = prev_time + time
                        newstep = prev_step + step
                        newline = f"{newtime} {newstep} [{split_line[1]}"
                        mainfile.write(newline)
                    else:
                        if i == 0:
                            savetime = time
                        else:
                            newtime = prev_time + (time - savetime)
                            newstep = prev_step + step
                            newline = f"{newtime} {newstep} [{split_line[1]}"
                            mainfile.write(newline)
                    
                    i += 1

            print(f"Finished with file {f}")
            prev_time = newtime
            prev_step = newstep

if __name__ == "__main__":
    process_files(multifile_name, filenumbers, output_filename)
