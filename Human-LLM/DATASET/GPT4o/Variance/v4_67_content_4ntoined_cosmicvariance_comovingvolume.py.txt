import matplotlib.pyplot as plt
import numpy as np
import math
from scipy.integrate import quad

# Define parameters for different cases
cases = [
    {"omegaM": 1, "omegaK": 0, "omegaL": 0},
    {"omegaM": 0.05, "omegaK": 0.95, "omegaL": 0},
    {"omegaM": 0.2, "omegaK": 0, "omegaL": 0.8}
]

# Function to calculate xE
def xE(z, omegaM, omegaK, omegaL):
    return (omegaM * (1 + z)**3 + omegaK * (1 + z)**2 + omegaL)**0.5

# Function to calculate the inverse of xE
def inverseE(z, omegaM, omegaK, omegaL):
    return 1 / xE(z, omegaM, omegaK, omegaL)

# Function to calculate dVc
def ploot(z, omegaM, omegaK, omegaL):
    entries = np.rint(z * 100 + 1).astype(int)
    k = entries - 1
    E = xE(z, omegaM, omegaK, omegaL)
    answer_and_error = [quad(inverseE, 0, c, args=(omegaM, omegaK, omegaL)) for c in np.arange(0, z, 0.01)]
    dC = np.zeros(entries)
    
    for i in range(1, entries):
        dC[i] = answer_and_error[i - 1][0]
    
    if omegaK > 0:
        dM = math.sinh((omegaK**0.5) * dC[k]) / (omegaK**0.5)
    elif omegaK < 0:
        dM = math.sin((abs(omegaK)**0.5) * dC[k]) / (abs(omegaK)**0.5)
    else:
        dM = dC[k]
    
    dA = dM / (1 + z)
    dVc = (((1 + z)**2) * (dA**2)) / E
    return dVc

# Prepare arrays for plotting
x = np.arange(0, 5.02, 0.01)
ploot_arrays = []

# Compute ploot arrays for each case
for case in cases:
    omegaM = case["omegaM"]
    omegaK = case["omegaK"]
    omegaL = case["omegaL"]
    ploot_array = [ploot(z, omegaM, omegaK, omegaL) for z in x]
    ploot_arrays.append(ploot_array)

# Plot the results
plt.plot(x, ploot_arrays[0], label="Case 1")
plt.plot(x, ploot_arrays[1], label="Case 2")
plt.plot(x, ploot_arrays[2], label="Case 3")
plt.axis([0, 5, 0, 1.2])
plt.legend()
plt.show()
