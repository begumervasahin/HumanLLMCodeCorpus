from numpy import zeros, ones, diag, apply_along_axis, amax
from scipy import linalg as la
from scipy.sparse import linalg as spla
from numpy import ndarray

def L_Seed(
    A: spla.LinearOperator,
    B: ndarray,
    tol: float = 5e-3,
    p_freq: int = 5,
    maxit: int = 50,
    verbose: bool = True
) -> dict:
    """
    Implementation of the L-Seed algorithm.

    Parameters:
    A (spla.LinearOperator): Linear operator representing matrix A.
    B (ndarray): Right-hand side matrix B.
    tol (float): Tolerance for convergence. Default is 5e-3.
    p_freq (int): Frequency of printing progress. Default is 5.
    maxit (int): Maximum number of iterations. Default is 50.
    verbose (bool): If True, prints progress and convergence messages. Default is True.

    Returns:
    dict: Dictionary containing matrices U, β, δ, and ρ.
    """
    n, t = A.shape[0], B.shape[1]

    # Initialize matrices and vectors
    X = zeros((n, t, maxit))
    R = zeros((n, t, maxit))
    U = zeros((n, t, maxit))
    ρ = zeros((t, maxit))
    β = zeros((t, maxit))
    ω = zeros((t, maxit))
    γ = ones((t, maxit))
    δ = zeros((t, maxit))

    # Initial values
    ρ[:, 0] = apply_along_axis(la.norm, 0, B)
    β[:, 0] = ρ[:, 0]
    R[:, :, 0] = B
    U[:, :, 0] = B / β[:, 0]

    j = 0
    cnvg = False

    while j < maxit - 1:
        # Compute tmp and δ
        tmp = A @ U[:, :, j]
        δ[:, j] = diag(U[:, :, j].T @ tmp)

        # Update tmp
        if j > 0:
            tmp = tmp - U[:, :, j] * δ[:, j] - β[:, j] * U[:, :, j - 1]
        else:
            tmp = tmp - U[:, :, j] * δ[:, j]

        # Compute β and U
        β[:, j + 1] = apply_along_axis(la.norm, 0, tmp)
        U[:, :, j + 1] = tmp / β[:, j + 1]

        # Update γ, ω, and ρ
        if j > 0:
            γ[:, j] = 1 / (δ[:, j] - ω[:, j - 1] / γ[:, j - 1])
        else:
            γ[:, j] = 1 / δ[:, j]

        ω[:, j] = (β[:, j + 1] * γ[:, j]) ** 2
        ρ[:, j + 1] = -β[:, j + 1] * γ[:, j] * ρ[:, j]

        # Update R
        R[:, :, j + 1] = ρ[:, j + 1] * U[:, :, j + 1]

        # Increment iteration counter
        j += 1

        # Compute the residual norm
        res_norm = amax(apply_along_axis(la.norm, 0, R[:, :, j]))

        if verbose and j % p_freq == 0:
            print(f"Error at step {j} is {res_norm}")

        if res_norm <= tol:
            cnvg = True
            break

    if verbose:
        if cnvg:
            print(f"Converged after {j} iterations.")
        else:
            print(f"Failed to converge after {j} iterations.")

    return {
        "U": U[:, :, :j],
        "β": β[:, :j],
        "δ": δ[:, :j],
        "ρ": ρ[:, :j]
    }
