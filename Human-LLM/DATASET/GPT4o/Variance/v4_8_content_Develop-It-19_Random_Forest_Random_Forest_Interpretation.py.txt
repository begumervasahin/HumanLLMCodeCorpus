# Function to get out-of-bag score
def get_oob(df):
    model = RandomForestRegressor(n_estimators=40, min_samples_leaf=3, max_features=0.5, n_jobs=-1, oob_score=True)
    x, _ = split_vals(df, n_trn)
    model.fit(x, y_train)
    return model.oob_score
