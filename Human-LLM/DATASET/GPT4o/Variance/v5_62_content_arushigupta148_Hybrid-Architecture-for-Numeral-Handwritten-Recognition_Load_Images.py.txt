import os
import struct
from array import array as pyarray
import numpy as np
import cv2

def load_mnist(dataset="training", digits=np.arange(10), path="."):
    """
    Loads MNIST files into 3D numpy arrays.

    Parameters:
    dataset (str): "training" or "testing"
    digits (array-like): Array of digits to load
    path (str): Path to the MNIST files

    Returns:
    tuple: (images, labels)
    """
    if dataset == "training":
        fname_img = os.path.join(path, 'train-images.idx3-ubyte')
        fname_lbl = os.path.join(path, 'train-labels.idx1-ubyte')
    elif dataset == "testing":
        fname_img = os.path.join(path, 't10k-images.idx3-ubyte')
        fname_lbl = os.path.join(path, 't10k-labels.idx1-ubyte')
    else:
        raise ValueError("dataset must be 'testing' or 'training'")

    with open(fname_lbl, 'rb') as flbl:
        magic_nr, size = struct.unpack(">II", flbl.read(8))
        lbl = pyarray("b", flbl.read())

    with open(fname_img, 'rb') as fimg:
        magic_nr, size, rows, cols = struct.unpack(">IIII", fimg.read(16))
        img = pyarray("B", fimg.read())

    indices = [k for k in range(size) if lbl[k] in digits]
    N = len(indices)

    images = np.zeros((N, rows, cols), dtype=np.uint8)
    labels = np.zeros((N, 1), dtype=np.int8)

    for i, idx in enumerate(indices):
        images[i] = np.array(img[idx*rows*cols : (idx+1)*rows*cols]).reshape((rows, cols))
        labels[i] = lbl[idx]

    return images, labels

def apply_otsu_thresholding(images):
    """
    Apply Otsu's thresholding to a list of images.

    Parameters:
    images (array-like): Array of images to process

    Returns:
    array-like: Processed images
    """
    processed_images = []
    for image in images:
        _, thresholded_image = cv2.threshold(image, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)
        processed_images.append(thresholded_image)
    return np.array(processed_images)

def save_data(images, labels, image_filename="X_test.npy", label_filename="y_test.npy"):
    """
    Save images and labels to .npy files.

    Parameters:
    images (array-like): Array of images to save
    labels (array-like): Array of labels to save
    image_filename (str): Filename for images
    label_filename (str): Filename for labels
    """
    np.save(image_filename, images)
    np.save(label_filename, labels)

# Load and process the MNIST testing dataset
images, labels = load_mnist(dataset="testing")
processed_images = apply_otsu_thresholding(images)

# Save processed images and labels
save_data(processed_images, labels)
