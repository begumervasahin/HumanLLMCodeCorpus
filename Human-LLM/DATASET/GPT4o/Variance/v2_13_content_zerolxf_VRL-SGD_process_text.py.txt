import os
import numpy as np
import pandas as pd
from torchtext.vocab import GloVe

# Function to process the text data
def process_text(text, glove_embeddings, vocab, max_length=50):
    """
    Processes the input text by tokenizing, filtering, and converting words to vectors.
    
    Parameters:
    - text: str, the input text to process
    - glove_embeddings: GloVe object, the pre-trained GloVe embeddings
    - vocab: set, the set of words in the vocabulary
    - max_length: int, the fixed length of the output vector
    
    Returns:
    - np.array, the processed text represented as a fixed-length vector
    """
    words = text.split()
    words = [word for word in words if word in vocab]
    
    # Adjust the length of words to match max_length
    if len(words) > max_length:
        words = words[:max_length]
    else:
        words.extend([" "] * (max_length - len(words)))
    
    # Convert words to vectors and flatten the result
    word_vectors = np.array([glove_embeddings.get_vecs_by_tokens(word).numpy() for word in words]).ravel()
    
    return word_vectors

# Function to save the processed data to files
def save_processed_data(train_data, test_data, train_labels, test_labels):
    """
    Saves the processed data and labels to .npy files.
    
    Parameters:
    - train_data: pd.Series, the processed training data
    - test_data: pd.Series, the processed testing data
    - train_labels: pd.Series, the training labels
    - test_labels: pd.Series, the testing labels
    """
    np.save("./db_pedia_train_data.npy", train_data.values)
    np.save("./db_pedia_test_data.npy", test_data.values)
    np.save("./db_pedia_train_label.npy", train_labels.values)
    np.save("./db_pedia_test_label.npy", test_labels.values)

# Function to generate and process the DBpedia dataset
def generate_db_pedia_dataset(path="./"):
    """
    Generates and processes the DBpedia dataset, converting text to vectors using GloVe embeddings.
    
    Parameters:
    - path: str, the path to the dataset files
    """
    print("Start reading files")
    train_data = pd.read_csv(os.path.join(path, "train1.csv"), header=None)
    test_data = pd.read_csv(os.path.join(path, "test1.csv"), header=None)
    print("Files read successfully")

    # Combine the text columns for training and testing data
    train_data['text'] = train_data[1] + train_data[2]
    test_data['text'] = test_data[1] + test_data[2]

    # Load GloVe embeddings
    glove = GloVe(name='6B', dim=50)
    vocab = set(glove.itos)
    print("Start processing files")

    # Apply the processing function to the text data
    train_data['processed_text'] = train_data['text'].apply(lambda text: process_text(text, glove, vocab))
    test_data['processed_text'] = test_data['text'].apply(lambda text: process_text(text, glove, vocab))
    print("Files processed successfully")

    # Extract the labels for training and testing data
    train_labels = train_data[0].copy()
    test_labels = test_data[0].copy()

    print("Start saving files")
    save_processed_data(train_data['processed_text'], test_data['processed_text'], train_labels, test_labels)
    print("Files saved successfully")

# Run the generate_db_pedia_dataset function with the specified path
if __name__ == "__main__":
    generate_db_pedia_dataset("./db_pedia/")
