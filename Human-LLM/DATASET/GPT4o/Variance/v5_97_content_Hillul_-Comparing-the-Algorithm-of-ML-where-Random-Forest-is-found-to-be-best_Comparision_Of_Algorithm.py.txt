import pandas as pd
import matplotlib.pyplot as plt
from sklearn.model_selection import KFold, cross_val_score
from sklearn.linear_model import LogisticRegression
from sklearn.tree import DecisionTreeClassifier
from sklearn.neighbors import KNeighborsClassifier
from sklearn.discriminant_analysis import LinearDiscriminantAnalysis
from sklearn.naive_bayes import GaussianNB
from sklearn.svm import SVC

def load_data(url, column_names):
    """Load the dataset from the given URL and return a DataFrame."""
    return pd.read_csv(url, names=column_names)

def evaluate_models(models, X, Y, seed, scoring='accuracy'):
    """Evaluate each model using cross-validation and return the results."""
    results = []
    for name, model in models:
        kfold = KFold(n_splits=10, random_state=seed, shuffle=True)
        cv_results = cross_val_score(model, X, Y, cv=kfold, scoring=scoring)
        results.append((name, cv_results))
        print(f"{name}: {cv_results.mean():.4f} ({cv_results.std():.4f})")
    return results

def plot_results(results):
    """Plot the cross-validation results as a boxplot."""
    fig, ax = plt.subplots()
    fig.suptitle('Algorithm Comparison')
    
    model_names = [name for name, _ in results]
    scores = [cv_results for _, cv_results in results]
    
    ax.boxplot(scores)
    ax.set_xticklabels(model_names, rotation=45)
    plt.show()

# Main script
if __name__ == "__main__":
    # Define constants
    DATA_URL = "https://raw.githubusercontent.com/jbrownlee/Datasets/master/pima-indians-diabetes.data.csv"
    COLUMN_NAMES = ['preg', 'plas', 'pres', 'skin', 'test', 'mass', 'pedi', 'age', 'class']
    SEED = 7

    # Load the dataset
    dataframe = load_data(DATA_URL, COLUMN_NAMES)

    # Separate the dataset into input features (X) and output target (Y)
    X = dataframe.iloc[:, :-1].values
    Y = dataframe.iloc[:, -1].values

    # Define the models to be evaluated
    models = [
        ('Logistic Regression', LogisticRegression(max_iter=200)),
        ('Linear Discriminant Analysis', LinearDiscriminantAnalysis()),
        ('K-Nearest Neighbors', KNeighborsClassifier()),
        ('Decision Tree', DecisionTreeClassifier()),
        ('Naive Bayes', GaussianNB()),
        ('Support Vector Machine', SVC())
    ]

    # Evaluate the models
    results = evaluate_models(models, X, Y, SEED)

    # Plot the results
    plot_results(results)
