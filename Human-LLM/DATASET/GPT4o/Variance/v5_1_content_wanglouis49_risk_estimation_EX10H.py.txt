from __future__ import division
import time
import numpy as np
import numpy.random as npr
import blspricer as bp
import customML as cm
import scipy.stats as scs
from scipy.stats.distributions import norm
from doe_lhs import lhs

class EX10:
    def __init__(self):
        # Initialize parameters
        self.D = 10
        self.S0 = 100.0
        self.mu = 0.08
        self.sigma = 0.3
        self.rfr = 0.05
        self.T = 0.1
        self.tau = 0.04
        self.K = 100.0
        self.H = 95.0
        self.c = 306.8763
        self.perc = 0.99
        self.I_pred = 2 ** 15
        self.dS = 1e-6
        self.Value0 = 0.0
        self.delta = np.zeros(self.D)

        self._initialize_opt_val()

        print(self.delta)

    def _initialize_opt_val(self):
        # Define the option valuation function
        opt_val = lambda S: (
            5.0 * bp.blsprice(S, self.K, self.rfr, self.T, self.sigma, 'put') +
            10.0 * bp.rear_end_dnOutCall(S, self.K, self.rfr, self.T, self.sigma, self.H, self.tau, 0.0)
        )

        # Calculate the initial delta and option value
        for d in range(self.D):
            self.delta[d] = (opt_val(self.S0 + self.dS) - opt_val(self.S0 - self.dS)) / self.dS
            self.Value0 += -opt_val(self.S0) + self.S0 * self.delta[d]

    def analy(self, N_o):
        """Analytical solution"""
        # Define the option valuation function
        opt_val = lambda S: (
            5.0 * bp.blsprice(S, self.K, self.rfr, self.T - self.tau, self.sigma, 'put') +
            10.0 * bp.dnOutCall(S, self.K, self.rfr, self.T - self.tau, self.sigma, self.H)
        )

        # Generate random samples using Latin Hypercube Sampling
        ran1 = norm(loc=0, scale=1).ppf(lhs(self.D, samples=N_o))

        # Initialize the asset prices matrix
        S1 = np.full((N_o, self.D), self.S0)
        S1 *= np.exp((self.mu - 0.5 * self.sigma ** 2) * self.tau + self.sigma * np.sqrt(self.tau) * ran1)

        # Calculate the option values at tau
        ValueTau = np.zeros(N_o)
        for n in range(N_o):
            for d in range(self.D):
                ValueTau[n] += -opt_val(S1[n, d]) + S1[n, d] * self.delta[d]

        # Sort the losses and calculate VaR and Expected Excess Loss
        L_analy = np.sort(self.Value0 - ValueTau)
        print(L_analy)
        var = scs.scoreatpercentile(L_analy, self.perc * 100.0)
        eel = np.mean(np.maximum(L_analy - var, 0))
        
        return var, eel
