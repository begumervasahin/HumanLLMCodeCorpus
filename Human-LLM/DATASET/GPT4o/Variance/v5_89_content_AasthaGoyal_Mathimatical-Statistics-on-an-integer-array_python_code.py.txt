import math

def input_numbers():
    """Prompts the user to input a list of numbers, validates the input, and returns a list of numbers and errors."""
    numbers = []
    errors = []
    while True:
        phrase = input("Enter a list of numbers separated by commas (or type 'Terminate' to exit): ")
        if phrase.lower() == 'terminate':
            print("You terminated the program")
            break
        
        temp_list = phrase.split(",")
        for item in temp_list:
            item = item.strip()
            if item.isdigit():
                numbers.append(int(item))
            else:
                print(f"{item} is not an integer value")
                errors.append(item)
    return numbers, errors

def selection_sort(lst):
    """Sorts a list of numbers using the selection sort algorithm and returns the sorted list."""
    sorted_list = []
    while lst:
        min_value = min(lst)
        sorted_list.append(min_value)
        lst.remove(min_value)
    return sorted_list

def calculate_statistics(sorted_list):
    """Calculates and returns the maximum, minimum, range, mean, variance, and standard deviation of a sorted list of numbers."""
    maximum = sorted_list[-1]
    minimum = sorted_list[0]
    range_value = maximum - minimum
    total_count = len(sorted_list)
    mean = sum(sorted_list) / total_count
    variance = sum((x - mean) ** 2 for x in sorted_list) / total_count
    std_deviation = math.sqrt(variance)
    
    return maximum, minimum, range_value, mean, variance, std_deviation

def count_occurrences(lst):
    """Counts the occurrences of each number in the list and returns a dictionary with the counts."""
    occurrences = {}
    for item in lst:
        occurrences[item] = occurrences.get(item, 0) + 1
    return occurrences

def display_statistics(sorted_list, occurrences, statistics, errors):
    """Displays the calculated statistics and occurrences of numbers."""
    maximum, minimum, range_value, mean, variance, std_deviation = statistics
    
    print("1) The number of occurrences of each number:")
    for number, count in occurrences.items():
        print(f"{number}: {count}")
    
    print(f"2) The Maximum value is: {maximum}")
    print(f"3) The Minimum value is: {minimum}")
    print(f"4) The Range of the numbers is: {range_value}")
    print(f"5) The Arithmetic Mean of the numbers is: {mean}")
    print(f"6) The Variance of the numbers is: {variance}")
    print(f"7) The Standard Deviation of the numbers is: {std_deviation}")
    print(f"8) Errors encountered: {errors}")

def main():
    numbers, errors = input_numbers()
    
    if not numbers:
        print("No valid numbers entered.")
        return
    
    sorted_list = selection_sort(numbers)
    statistics = calculate_statistics(sorted_list)
    occurrences = count_occurrences(sorted_list)
    
    display_statistics(sorted_list, occurrences, statistics, errors)

if __name__ == "__main__":
    main()
