import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# Set the plot size
plt.rcParams['figure.figsize'] = (20.0, 10.0)

# Load the dataset
data = pd.read_csv('/Users/swaruptripathy/Desktop/Data Science/headbrain.csv')
print(f"Dataset shape: {data.shape}")
print(data.head())

# Extract features and labels
X = data['Head Size(cm^3)'].values
Y = data['Brain Weight(grams)'].values

# Calculate means
mean_x = np.mean(X)
mean_y = np.mean(Y)

# Calculate coefficients b0 and b1
m = len(X)
numer = sum((X[i] - mean_x) * (Y[i] - mean_y) for i in range(m))
denom = sum((X[i] - mean_x) ** 2 for i in range(m))
b1 = numer / denom
b0 = mean_y - (b1 * mean_x)
print(f"Coefficients: b1 = {b1}, b0 = {b0}")

# Plot the regression line
max_x = np.max(X) + 100
min_x = np.min(X) - 100
x = np.linspace(min_x, max_x, 1000)
y = b0 + b1 * x

plt.plot(x, y, color='blue', label='Regression Line')
plt.scatter(X, Y, c='red', label='Scatter Plot')
plt.xlabel('Head Size in cm³')
plt.ylabel('Brain Weight in grams')
plt.legend()
plt.show()

# Calculate Root Mean Squared Error (RMSE)
rmse = np.sqrt(sum((Y[i] - (b0 + b1 * X[i])) ** 2 for i in range(m)) / m)
print(f"RMSE: {rmse}")

# Calculate R-squared (R²)
ss_t = sum((Y[i] - mean_y) ** 2 for i in range(m))
ss_r = sum((Y[i] - (b0 + b1 * X[i])) ** 2 for i in range(m))
r2 = 1 - (ss_r / ss_t)
print(f"R²: {r2}")

# Using sklearn for Linear Regression
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error

X = X.reshape((m, 1))  # Reshape X to fit the model
reg = LinearRegression().fit(X, Y)
Y_pred = reg.predict(X)

# Calculate RMSE and R² using sklearn
mse = mean_squared_error(Y, Y_pred)
rmse = np.sqrt(mse)
r2_score = reg.score(X, Y)

print(f"Sklearn RMSE: {rmse}")
print(f"Sklearn R²: {r2_score}")
