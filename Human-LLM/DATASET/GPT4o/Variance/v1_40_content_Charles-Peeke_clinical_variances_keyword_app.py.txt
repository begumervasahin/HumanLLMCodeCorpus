import os
import time
import pandas as pd
import matplotlib.pyplot as plt
from dotenv import load_dotenv
import psycopg2

def load_environment_variables():
    """Load environment variables from .env file."""
    load_dotenv()
    env_vars = {
        'hostname': os.getenv('hostname'),
        'port': os.getenv('port'),
        'database': os.getenv('database'),
        'username': os.getenv('username'),
        'password': os.getenv('password')
    }
    return env_vars

def connect_to_database(env_vars):
    """Establish a connection to the PostgreSQL database."""
    conn = psycopg2.connect(
        host=env_vars['hostname'],
        port=env_vars['port'],
        database=env_vars['database'],
        user=env_vars['username'],
        password=env_vars['password']
    )
    return conn

def fetch_keywords(conn):
    """Fetch keywords from the database."""
    print('Fetching keywords...')
    start_time = time.time()
    query = '''
        SELECT studies.nct_id, downcase_name 
        FROM studies 
        RIGHT JOIN keywords ON studies.nct_id = keywords.nct_id
    '''
    df_keywords = pd.read_sql(query, con=conn)
    print(f'Keywords fetched in {time.time() - start_time:.2f} seconds')
    return df_keywords

def calculate_keyword_variances(df_keywords):
    """Calculate keyword variances."""
    print('Calculating keyword variances...')
    start_time = time.time()
    key_dict = {}
    for nct_id, keyw in zip(df_keywords['nct_id'], df_keywords['downcase_name']):
        if nct_id in key_dict:
            key_dict[nct_id] += 1
        else:
            key_dict[nct_id] = 1
    print(f'Keyword variances calculated in {time.time() - start_time:.2f} seconds')
    return key_dict

def plot_keyword_variances(key_dict):
    """Plot the keyword variances."""
    ids = list(key_dict.keys())
    counts = list(key_dict.values())
    plt.figure(figsize=(10, 6))
    plt.bar(ids, counts)
    plt.xlabel('NCT ID')
    plt.ylabel('Keyword Count')
    plt.title('Keyword Variances by NCT ID')
    plt.xticks(rotation=90)
    plt.tight_layout()
    plt.show()

def main():
    env_vars = load_environment_variables()
    conn = connect_to_database(env_vars)
    
    df_keywords = fetch_keywords(conn)
    key_dict = calculate_keyword_variances(df_keywords)
    plot_keyword_variances(key_dict)

    conn.close()

if __name__ == '__main__':
    main()
