import numpy as np
from my_augmentations import zero_pad, random_crop, horizontal_flip

def run_augmentation_v1(image):
    """
    Apply a series of augmentations to the input image.
    
    Parameters:
    image (numpy.ndarray): Input image to be augmented.
    
    Returns:
    numpy.ndarray: Augmented image.
    """
    image = zero_pad(image, (40, 40, 3), 4)
    image = random_crop(image, 32)
    image = horizontal_flip(image, 0.5)
    return image

def load_data(path, test=False):
    """
    Load CIFAR-10 data from the given path.
    
    Parameters:
    path (str): Path to the CIFAR-10 data.
    test (bool): Whether to load test data. Defaults to False.
    
    Returns:
    tuple: Tuple of data and labels.
    """
    data = []
    labels = []
    file_prefix = 'test_batch' if test else 'data_batch_'
    
    for slice in range(1, 6):
        file_suffix = '' if test else str(slice)
        data_dict = np.load(f"{path}{file_prefix}{file_suffix}", allow_pickle=True)
        images = data_dict['data'].reshape((len(data_dict['data']), 3, 32, 32)).transpose(0, 2, 3, 1)
        data.extend(images)
        labels.extend(data_dict['labels'])
        if test:
            break
            
    return np.array(data), np.array(labels)

def change_lr(optimizer, lr):
    """
    Change the learning rate of the optimizer.
    
    Parameters:
    optimizer (Optimizer): The optimizer whose learning rate needs to be changed.
    lr (float): The new learning rate.
    """
    for param_group in optimizer.param_groups:
        param_group['lr'] = lr
        print(f"LR CHANGED: {param_group['lr']}")

class AverageMeter:
    """
    Computes and stores the average and current value.
    """
    def __init__(self):
        self.reset()

    def reset(self):
        """
        Reset all statistics.
        """
        self.val = 0.0
        self.avg = 0.0
        self.sum = 0.0
        self.count = 0.0

    def update(self, val, n=1):
        """
        Update the statistics.
        
        Parameters:
        val (float): The value to update.
        n (int): The weight of the value. Defaults to 1.
        """
        self.val = val
        self.sum += val * n
        self.count += n
        self.avg = self.sum / self.count
