import os
import time
import psycopg2
import pandas as pd
from dotenv import load_dotenv

# Global variable for the database connection
conn = None

def load_environment_variables():
    """Load environment variables from .env file"""
    load_dotenv()
    env_vars = {
        "hostname": os.getenv('hostname'),
        "port": os.getenv('port'),
        "database": os.getenv('database'),
        "username": os.getenv('username'),
        "password": os.getenv('password')
    }
    return env_vars

def connect_to_database(env_vars):
    """Establish a connection to the PostgreSQL database"""
    connection = psycopg2.connect(
        host=env_vars["hostname"],
        port=env_vars["port"],
        database=env_vars["database"],
        user=env_vars["username"],
        password=env_vars["password"]
    )
    return connection

def get_data_from_table(table_name):
    """Retrieve all data from a specified table"""
    query = f'SELECT * FROM {table_name}'
    data = pd.read_sql(query, con=conn)
    return data

def join_tables(table1, table2):
    """Perform a RIGHT JOIN on two tables using the 'nct_id' column"""
    query = f"""
    SELECT * FROM {table1}
    RIGHT JOIN {table2} ON {table1}.nct_id = {table2}.nct_id
    """
    joined_data = pd.read_sql(query, con=conn)
    return joined_data

def measure_query_time(query_function, *args):
    """Measure the execution time of a query function"""
    start_time = time.time()
    query_function(*args)
    end_time = time.time()
    return end_time - start_time

def main():
    # Load environment variables
    env_vars = load_environment_variables()

    # Establish database connection
    global conn
    conn = connect_to_database(env_vars)

    # Measure time for join operation using join_tables function
    join_time = measure_query_time(join_tables, 'ctgov.studies', 'ctgov.keywords')
    print(f"Join operation time: {join_time:.4f} seconds")

    # Measure time for a raw SQL join query
    raw_sql_time = measure_query_time(pd.read_sql, """
    SELECT * FROM ctgov.studies 
    RIGHT JOIN ctgov.keywords ON ctgov.studies.nct_id = ctgov.keywords.nct_id
    """, con=conn)
    print(f"Raw SQL join query time: {raw_sql_time:.4f} seconds")

    # Close the database connection
    conn.close()

if __name__ == '__main__':
    main()
