import matplotlib.pyplot as plt
from matplotlib.colors import LinearSegmentedColormap
from matplotlib.lines import Line2D
from cycler import cycler
import seaborn as sns

# Define color variables
n_red = "#FF0000"
n_blue = "#0000FF"
n_green = "#00FF00"
n_gray = "#808080"
n_black = "#000000"
n_palette = [n_red, n_blue, n_green, n_gray]

# Create colorblind-friendly palette and colormap
cblind_palette = sns.color_palette("colorblind", 9)
cblind_cmap = LinearSegmentedColormap.from_list("cmap", cblind_palette)
plt.rcParams['axes.prop_cycle'] = cycler(color=cblind_palette)

# Define HSLuv values for blue and red
n_blue_hsluv = 248.2
n_red_hsluv = 26.8

# Create diverging colormap for heatmaps
heatmap_cmap = sns.diverging_palette(n_blue_hsluv, n_red_hsluv, 85, 54, n=15, as_cmap=True)

# Define figure sizes
figsize_1 = (6.5, 3)
figsize_2 = (6.5, 6)

def set_visuals():
    """
    Set visual parameters for matplotlib plots.
    """
    font_settings = {
        "family": "serif",
        "size": 10
    }
    fig_settings = {
        "figsize": figsize_1
    }
    xtick_settings = {
        "labelsize": 10
    }
    ytick_settings = {
        "alignment": "center"
    }
    axes_settings = {
        "grid": True
    }
    grid_settings = {
        "linestyle": '-',
        "alpha": 0.75
    }
    legend_settings = {
        "fontsize": 10
    }
    
    plt.rc("xtick", **xtick_settings)
    plt.rc("ytick", **ytick_settings)
    plt.rc("figure", **fig_settings)
    plt.rc("font", **font_settings)
    plt.rc("axes", **axes_settings)
    plt.rc("grid", **grid_settings)
    plt.rc("legend", **legend_settings)

def add_line(**kwargs):
    """
    Add a line to a plot.
    
    Parameters
    ----------
    kwargs : dict
        Valid parameters: color, lw, linestyle, label
    
    Returns
    -------
    Line2D
        A Line2D object with the specified parameters.
    """
    return Line2D([0], [0], **kwargs)
