import numpy as np
import scipy.stats as stats

# Define the range of x values
range_of_x = [-1.73, 1.73]

def lessthan(values, alpha):
    """Calculate the percentage of values less than alpha."""
    count = sum(value < alpha for value in values)
    percentage = count / len(values)
    return percentage

def pdf(x, uniform_value):
    """Probability density function that combines normal distribution and uniform distribution."""
    return (1 - uniform_value) * stats.norm.pdf(x) + 0.289 * uniform_value

def find_min_and_max(range_values, uniform_value):
    """Find the minimum and maximum values of the pdf within the given range."""
    list_of_range = np.arange(range_values[0], range_values[1], 0.01)
    pdf_values = [pdf(x, uniform_value) for x in list_of_range]
    min_value = min(pdf_values)
    max_value = max(pdf_values)
    return min_value, max_value

# Iterate over p values from 0 to 0.01 with a step of 0.1
p = 0
while p < 0.01:
    min_of_y, max_of_y = find_min_and_max(range_of_x, p)
    p_value_list = []

    # Perform 1000 iterations for generating samples and calculating p-values
    for _ in range(1000):
        samples = []

        # Generate 100 samples that satisfy the pdf condition
        while len(samples) < 100:
            x = np.random.uniform(range_of_x[0], range_of_x[1])
            y = np.random.uniform(min_of_y, max_of_y)

            if y < pdf(x, p):
                samples.append(x)

        # Calculate the p-value using the normal test
        p_value = stats.normaltest(samples)[1]
        p_value_list.append(p_value)

    # Print the results
    print(f"p: {p:.2f}")
    print(f"Percentage less than 0.05: {lessthan(p_value_list, 0.05):.2%}")
    p += 0.1
