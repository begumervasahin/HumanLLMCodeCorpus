import os
import sys
import nsc
import hdw

def check_python_version():
    """Ensure the correct Python version."""
    if sys.version_info[0:2] < (2, 4):
        from sets import Set as set, ImmutableSet as frozenset

def handle_arguments():
    """Handle command-line arguments."""
    return hdw.handle_commands(sys.argv, 's:', ['separator=', 'classid='])

def initialize_universe(ifile_name, separator, klasse_index):
    """Initialize the universe."""
    universe = hdw.abstract_file(ifile_name, separator, klasse_index)
    hdw.fill_world(universe)
    return universe

def compute_prototypes(sigma):
    """Compute prototypes."""
    prototypes = {}
    for kl in nsc.welt.keys():
        nsc.computeRLs(kl)
        for pr in nsc.mvc(kl, sigma):
            if kl not in prototypes:
                prototypes[kl] = set()
            if not pr.isVoid():
                prototypes[kl].add(pr.mean)
    return prototypes

def write_prototypes_to_file(prototypes, ifile_name, sigma):
    """Write prototypes to file."""
    ofile_name = f'{os.path.splitext(ifile_name)[0]}-mvc-{sigma:.4f}.txt'
    with open(ofile_name, 'w') as ofile:
        for kl in prototypes.keys():
            for point in prototypes[kl]:
                ofile.write(f'{point}\n')

def plot_using_gnuplot(prototypes, ifile_name, sigma):
    """Plot using Gnuplot."""
    try:
        import Gnuplot
    except ImportError:
        print('\nCANNOT FIND GNUPLOT-PYTHON MODULE\n')
        sys.exit()

    f, s = hdw.rnd_dim(nsc.dim)
    g = Gnuplot.Gnuplot(debug=0)
    g.title(f'MVC applied to {os.path.basename(ifile_name)} with sigma^2={sigma:.4f}')
    if (f, s) != (0, 1):
        g.xlabel(f'feature space reduced to {f + 1}x{s + 1}')

    plot_data(g, prototypes, f, s, '** {} **')
    plot_data(g, nsc.welt, f, s, '{}')

    gfile_name = f'{os.path.splitext(ifile_name)[0]}-mvc-{sigma:.4f}.ps'
    g.hardcopy(gfile_name, color=1)

def plot_data(g, data, f, s, title_template):
    """Helper function to plot data."""
    for kl in sorted(data.keys()):
        if len(data[kl]) == 0:
            continue
        points = [item.features for item in data[kl]]
        data[kl] = points
        g.replot(Gnuplot.Data(data[kl], title=title_template.format(kl), cols=(f, s)))

def main():
    check_python_version()
    
    ifile_name, sigma, _, _, separator, klasse_index = handle_arguments()
    
    universe = initialize_universe(ifile_name, separator, klasse_index)
    
    prototypes = compute_prototypes(sigma)
    
    write_prototypes_to_file(prototypes, ifile_name, sigma)
    
    plot_using_gnuplot(prototypes, ifile_name, sigma)

if __name__ == "__main__":
    main()
