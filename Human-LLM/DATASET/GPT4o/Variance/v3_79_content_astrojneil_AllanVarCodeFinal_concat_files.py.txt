import numpy as np

multifile_name = 'savefiles/640test_'
filenumbers = list(range(1, 60))  # Assuming file numbers are sequential from 1 to 59
output_filename = '640_final.dat'

def process_files(multifile_name, filenumbers, output_filename):
    prev_time = 0.0
    prev_step = 0

    with open(output_filename, 'w') as mainfile:
        for f in filenumbers:
            with open(f"{multifile_name}{f}.dat", 'r') as singlefile:
                savetime = 0.0

                for i, line in enumerate(singlefile):
                    time, step, split_line = parse_line(line)

                    if f == 1 or i > 0:
                        if i == 0 and f != 1:
                            savetime = time
                        newtime = calculate_new_time(f, prev_time, time, savetime)
                        newstep = prev_step + step
                        write_line(mainfile, newtime, newstep, split_line)

            print(f"Finished processing file {f}")
            prev_time, prev_step = newtime, newstep

def parse_line(line):
    split_line = line.split('[')
    temp = split_line[0]
    time, step = map(float, temp.split()[:2])
    step = int(step)
    return time, step, split_line[1]

def calculate_new_time(file_number, prev_time, time, savetime):
    if file_number == 1:
        return prev_time + time
    return prev_time + (time - savetime)

def write_line(file, newtime, newstep, split_line):
    newline = f"{newtime} {newstep} [{split_line}"
    file.write(newline)

if __name__ == "__main__":
    process_files(multifile_name, filenumbers, output_filename)
