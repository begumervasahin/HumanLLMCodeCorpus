import folium
import pandas as pd
from geopy.geocoders import Nominatim
from geopy.distance import great_circle

def create_map(names, latitudes, longitudes, output_file='map.html'):
    """Creates a Folium map with markers for given locations."""
    # Convert latitude and longitude lists to floats
    latitudes = [float(lat) for lat in latitudes]
    longitudes = [float(lon) for lon in longitudes]
    
    # Create a DataFrame to hold the coordinates and names
    data = pd.DataFrame({
        'latitude': latitudes,
        'longitude': longitudes,
        'name': names
    })
    
    # Create a Folium map centered at a given location
    map_center = [20, 0]
    folium_map = folium.Map(location=map_center, tiles="Mapbox Bright", zoom_start=2)
    
    # Add markers to the map
    for i in range(len(data)):
        folium.Marker(
            [data.iloc[i]['latitude'], data.iloc[i]['longitude']],
            popup=data.iloc[i]['name']
        ).add_to(folium_map)
    
    # Save the map to an HTML file
    folium_map.save(output_file)
    print(f"Map saved to {output_file}")

def calculate_edges(data):
    """Calculates the edges and distances between airports."""
    edges = []
    
    for j in range(len(data)):
        loc_chosen = data[j][0]
        coord_chosen = (float(data[j][6]), float(data[j][7]))
        
        for i in range(len(data)):
            loc_other = data[i][0]
            coord_other = (float(data[i][6]), float(data[i][7]))
            
            if loc_chosen == loc_other:
                edges.append([int(loc_chosen), int(loc_other), 99999])
            else:
                distance = great_circle(coord_chosen, coord_other).km
                edges.append([int(loc_chosen), int(loc_other), distance])
    
    return edges

def main(data):
    """Processes the input data, creates a map, and calculates edge distances."""
    latitudes = [entry[6] for entry in data]
    longitudes = [entry[7] for entry in data]
    airport_names = [entry[1] for entry in data]
    
    # Create the map
    create_map(airport_names, latitudes, longitudes)
    
    # Calculate edges
    edges = calculate_edges(data)
    
    return edges, len(data)
