import numpy as np
import math

# Set numpy print options for better readability
np.set_printoptions(precision=2, suppress=True)

# Define infinity for use in the adjacency matrix
infinity = math.inf

def get_graph():
    """
    Prompt the user to input the number of vertices and the adjacency matrix values.
    
    Returns:
    tuple: A tuple containing the number of vertices and the adjacency matrix.
    """
    V = int(input("Enter number of vertices: "))
    print("Enter the values of adjacency matrix row-wise:")
    graph = np.zeros((V, V))
    for i in range(V):
        for j in range(V):
            graph[i][j] = int(input())
    return V, graph

def floyd_warshall(graph, n):
    """
    Implement the Floyd-Warshall algorithm to find the shortest paths between all pairs of vertices.
    
    Parameters:
    graph (numpy.ndarray): The adjacency matrix of the graph.
    n (int): The number of vertices in the graph.
    """
    print("\n\n")
    print("D[0] :\n", graph)
    print("\n")
    for k in range(n):
        for i in range(n):
            for j in range(n):
                graph[i][j] = min(graph[i][j], graph[i][k] + graph[k][j])
        print(f"D[{k+1}] :\n", graph)
        print("\n")

def main():
    """
    Main function to execute the Floyd-Warshall algorithm on a user-provided graph.
    """
    V, graph = get_graph()
    floyd_warshall(graph, V)

if __name__ == '__main__':
    main()
