import sys
from pathlib import Path
from plot_graph import PlotGraph  # Ensure PlotGraph is implemented in plot_graph.py
from read_graph import ReadGraph  # Ensure ReadGraph is implemented in read_graph.py

def main(input_file_path):
    """
    Main function to process a graph from a file: read the graph data, print the matrix,
    and plot the graph.

    Parameters:
    input_file_path (Path): The path to the input file containing the graph data.
    """
    # Create a ReadGraph instance and read the graph matrix from the file
    graph_reader = ReadGraph(input_file_path)
    graph_matrix = graph_reader.read()

    # Output the graph matrix for analysis
    print("Analysing graph matrix...\n")
    print(graph_matrix)

    # Create a PlotGraph instance and plot the graph
    print("\nPlotting the graph...")
    graph_plotter = PlotGraph(graph_matrix)
    graph_plotter.plot()

if __name__ == '__main__':
    # Check if the script is provided with the necessary command-line argument
    if len(sys.argv) != 2:
        raise ValueError("Please provide exactly one argument: the path to the input file.")

    # Extract the input file path from command-line arguments
    input_file_path = Path(sys.argv[1])

    # Validate the existence of the input file
    if not input_file_path.is_file():
        raise FileNotFoundError(f"The file '{input_file_path}' does not exist.")

    # Execute the main function with the provided input file path
    main(input_file_path)
