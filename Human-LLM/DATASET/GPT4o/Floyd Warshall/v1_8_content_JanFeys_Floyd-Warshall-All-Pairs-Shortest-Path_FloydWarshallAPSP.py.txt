import time
from collections import defaultdict
import sys

class WeightedDirectedGraph:
    """
    A class to represent a weighted directed graph and perform shortest path calculations using the Floyd-Warshall algorithm.

    Attributes:
    nr_vs (int): Number of vertices in the graph.
    es (defaultdict): Adjacency list to store edges.
    ws (dict): Dictionary to store weights of edges.
    A (list of lists): Matrix to store shortest path distances.
    B (list of lists): Temporary matrix used in Floyd-Warshall algorithm.
    """

    def __init__(self, nr_vs: int):
        """
        Initializes the graph with the number of vertices.

        Parameters:
        nr_vs (int): Number of vertices in the graph.
        """
        self.nr_vs = nr_vs
        self.es = defaultdict(set)
        self.ws = dict()

    def show_nr_vs(self) -> int:
        """
        Returns the number of vertices in the graph.

        Returns:
        int: Number of vertices.
        """
        return self.nr_vs

    def add_edge(self, t: int, h: int, w: int):
        """
        Adds an edge to the graph with a specified weight.

        Parameters:
        t (int): Tail vertex.
        h (int): Head vertex.
        w (int): Weight of the edge.
        """
        self.es[t].add(h)
        self.ws[(t, h)] = w

    def get_edges(self, t: int) -> set:
        """
        Returns the set of edges originating from a given vertex.

        Parameters:
        t (int): Tail vertex.

        Returns:
        set: Set of head vertices connected by edges from the tail vertex.
        """
        return self.es[t]

    def get_weight(self, t: int, h: int) -> int:
        """
        Returns the weight of an edge between two vertices.

        Parameters:
        t (int): Tail vertex.
        h (int): Head vertex.

        Returns:
        int: Weight of the edge.
        """
        return self.ws[(t, h)]

    def remove_edge(self, t: int, h: int):
        """
        Removes an edge from the graph.

        Parameters:
        t (int): Tail vertex.
        h (int): Head vertex.
        """
        self.es[t].remove(h)
        del self.ws[(t, h)]

    def display(self):
        """
        Displays all edges and their weights in the graph.
        """
        nr_es_present = 0
        for t in self.es:
            for h in self.es[t]:
                print(f"Edge from {t} to {h} with weight {self.ws[(t, h)]}")
                nr_es_present += 1
        print(f"There are {nr_es_present} edges in total.")

    def run_floyd_warshall(self) -> list:
        """
        Executes the Floyd-Warshall algorithm to find the shortest paths between all pairs of vertices.

        Returns:
        list of lists: Matrix representing shortest path distances.
        """
        # Initialize distance matrix
        self.A = [[0 if t == h else sys.maxsize for h in range(self.nr_vs)] for t in range(self.nr_vs)]

        for t in range(self.nr_vs):
            for h in self.es[t]:
                self.A[t][h] = self.ws[(t, h)]

        # Floyd-Warshall algorithm
        for k in range(self.nr_vs):
            self.B = [row[:] for row in self.A]
            for t in range(self.nr_vs):
                for h in range(self.nr_vs):
                    self.A[t][h] = min(self.B[t][h], self.B[t][k] + self.B[k][h])

        # Check for negative weight cycles
        for t in range(self.nr_vs):
            if self.A[t][t] < 0:
                print("There is a negative cycle!")
                sys.exit()

        return self.A

if __name__ == "__main__":
    # Define the path to the file containing the graph matrix
    file_name = 'APSPtest3.txt'

    start_time = time.time()

    # Read the graph from the file
    with open(file_name, 'r') as file:
        nr_vs, _ = map(int, file.readline().strip().split())
        graph = WeightedDirectedGraph(nr_vs)
        for line in file:
            t, h, w = map(int, line.strip().split())
            graph.add_edge(t - 1, h - 1, w)

    # Compute shortest paths
    shortest_paths = graph.run_floyd_warshall()

    # Print shortest paths
    for t in range(nr_vs):
        for h in range(nr_vs):
            print(f"From {t} to {h} the shortest path is {shortest_paths[t][h]}")

    # Print minimum distance
    min_distance = min(min(row) for row in shortest_paths)
    print(f"Minimum distance is {min_distance}")

    end_time = time.time()
    print(f"Time taken: {end_time - start_time:.4f} seconds")
