import math

def floyd_warshall(graph, num_vertices):
    """
    Implements the Floyd-Warshall algorithm to find the shortest paths between all pairs of vertices.
    
    Parameters:
    graph (dict): A dictionary representing the graph where keys are (i, j) tuples representing edges and values are weights.
    num_vertices (int): The number of vertices in the graph.
    
    Returns:
    float: The shortest path found in the graph. Returns 'NULL' if a negative cycle is detected.
    """
    # Initialize the bookkeeper matrix
    book_keeper = [[[0 if k == 0 else 0 for k in range(2)] for j in range(num_vertices + 1)] for i in range(num_vertices + 1)]
    
    # Populate the initial distances in bookkeeper matrix
    for i in range(1, num_vertices + 1):
        for j in range(1, num_vertices + 1):
            if i == j:
                book_keeper[i][j][0] = 0
            elif graph.get((i, j)) is not None:
                book_keeper[i][j][0] = graph[(i, j)]
            else:
                book_keeper[i][j][0] = math.inf

    # Floyd-Warshall algorithm
    shortest_path = math.inf
    for k in range(1, num_vertices + 1):
        for i in range(1, num_vertices + 1):
            for j in range(1, num_vertices + 1):
                book_keeper[i][j][1] = min(book_keeper[i][j][0], book_keeper[i][k][0] + book_keeper[k][j][0])
                if k == num_vertices and shortest_path > book_keeper[i][j][1]:
                    shortest_path = book_keeper[i][j][1]
                if i == j and book_keeper[i][j][1] < 0:
                    return 'NULL'
                book_keeper[i][j][0] = book_keeper[i][j][1]

    return shortest_path

def read_graph_from_file(file_path):
    """
    Reads the graph from a given file and constructs a dictionary representation of the graph.
    
    Parameters:
    file_path (str): The path to the file containing the graph data.
    
    Returns:
    tuple: A tuple containing the graph dictionary and the number of vertices.
    """
    graph = {}
    with open(file_path) as file:
        first_line = file.readline()
        num_vertices, num_edges = map(int, first_line.split())
        data = file.readlines()
        for line in data:
            u, v, w = map(int, line.split())
            if (u, v) not in graph or graph[(u, v)] > w:
                graph[(u, v)] = w

    return graph, num_vertices

def main():
    """
    Main function to execute the Floyd-Warshall algorithm on a graph read from a file.
    """
    file_path = 'g3.txt'
    graph, num_vertices = read_graph_from_file(file_path)
    result = floyd_warshall(graph, num_vertices)
    print(result)

if __name__ == '__main__':
    main()
