from shortest import Database, ConnectingFlights, console

def main():
    """
    Main function to establish a database connection, initialize the ConnectingFlights class,
    and display the console menu.
    """
    # Database connection parameters
    DATABASE_HOST = 'localhost'
    DATABASE_PORT = 27017
    DATABASE_NAME = "connecting_flight"
    
    # Establish a connection to the database
    db = Database(DATABASE_HOST, DATABASE_PORT, DATABASE_NAME)
    
    # Initialize ConnectingFlights with the database connection
    connecting_flights = ConnectingFlights(db)
    
    # Display the console menu
    console.menu(db, connecting_flights)
    
    # Clear the console
    console.clear()

if __name__ == '__main__':
    main()
