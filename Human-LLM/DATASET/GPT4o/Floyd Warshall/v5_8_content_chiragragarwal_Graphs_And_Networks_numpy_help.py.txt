import numpy as np

def create_matrix_from_string(matrix_str):
    """
    Converts a formatted string into a NumPy matrix.

    Parameters:
    matrix_str (str): The matrix data as a string with rows separated by semicolons.

    Returns:
    np.matrix: The NumPy matrix created from the string.
    """
    # Replace semicolons with newlines to format the string for matrix creation
    formatted_str = matrix_str.replace(';', '\n')
    
    # Create and return a NumPy matrix from the formatted string
    return np.matrix(formatted_str)

def display_matrix_operations(matrix):
    """
    Performs and displays various operations on the given NumPy matrix.

    Parameters:
    matrix (np.matrix): The NumPy matrix on which operations will be performed.
    """
    # Print the shape of the matrix
    print("Matrix shape:", matrix.shape)
    
    # Access and print specific elements from the matrix
    print("Element at position (0, 1):", matrix[0, 1])
    print("Element at position (2, 3):", matrix[2, 3])
    
    # Modify an element in the matrix
    matrix[0, 1] = -8
    
    # Print the modified row
    print("Modified row 1:", matrix[1])
    
    # Print the entire column
    print("Column 1:", matrix[:, 1])

def main():
    """
    Main function to demonstrate operations on a NumPy matrix created from a string.
    """
    # Define the matrix as a string with rows separated by semicolons
    matrix_str = '1 2 3 4 ; 5 6 7 8 ; 0 1 2 3'
    
    # Create a NumPy matrix from the formatted string
    matrix = create_matrix_from_string(matrix_str)
    
    # Perform and display operations on the matrix
    display_matrix_operations(matrix)

if __name__ == '__main__':
    main()
