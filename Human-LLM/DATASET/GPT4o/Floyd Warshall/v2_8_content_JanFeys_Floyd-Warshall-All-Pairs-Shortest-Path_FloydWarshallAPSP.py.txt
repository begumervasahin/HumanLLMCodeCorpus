import time
from collections import defaultdict
import sys

class WeightedDirectedGraph:
    """
    Represents a weighted directed graph and supports shortest path calculations using the Floyd-Warshall algorithm.

    Attributes:
    nr_vs (int): Number of vertices in the graph.
    edges (defaultdict): Adjacency list of edges.
    weights (dict): Dictionary storing the weight of each edge.
    distance_matrix (list of lists): Matrix to store shortest path distances.
    temp_matrix (list of lists): Temporary matrix used in the Floyd-Warshall algorithm.
    """

    def __init__(self, nr_vs: int):
        """
        Initializes the graph with the number of vertices.

        Parameters:
        nr_vs (int): Number of vertices in the graph.
        """
        self.nr_vs = nr_vs
        self.edges = defaultdict(set)
        self.weights = dict()

    def get_number_of_vertices(self) -> int:
        """
        Returns the number of vertices in the graph.

        Returns:
        int: Number of vertices.
        """
        return self.nr_vs

    def add_edge(self, tail: int, head: int, weight: int):
        """
        Adds an edge to the graph with a specified weight.

        Parameters:
        tail (int): Tail vertex.
        head (int): Head vertex.
        weight (int): Weight of the edge.
        """
        self.edges[tail].add(head)
        self.weights[(tail, head)] = weight

    def get_edges(self, tail: int) -> set:
        """
        Returns the set of edges originating from a given vertex.

        Parameters:
        tail (int): Tail vertex.

        Returns:
        set: Set of head vertices connected by edges from the tail vertex.
        """
        return self.edges[tail]

    def get_weight(self, tail: int, head: int) -> int:
        """
        Returns the weight of an edge between two vertices.

        Parameters:
        tail (int): Tail vertex.
        head (int): Head vertex.

        Returns:
        int: Weight of the edge.
        """
        return self.weights.get((tail, head), sys.maxsize)

    def remove_edge(self, tail: int, head: int):
        """
        Removes an edge from the graph.

        Parameters:
        tail (int): Tail vertex.
        head (int): Head vertex.
        """
        if head in self.edges[tail]:
            self.edges[tail].remove(head)
            del self.weights[(tail, head)]

    def display_edges(self):
        """
        Displays all edges and their weights in the graph.
        """
        edge_count = 0
        for tail in self.edges:
            for head in self.edges[tail]:
                print(f"Edge from {tail} to {head} with weight {self.weights[(tail, head)]}")
                edge_count += 1
        print(f"There are {edge_count} edges in total.")

    def run_floyd_warshall(self) -> list:
        """
        Executes the Floyd-Warshall algorithm to find the shortest paths between all pairs of vertices.

        Returns:
        list of lists: Matrix representing shortest path distances.
        """
        # Initialize the distance matrix
        self.distance_matrix = [[0 if i == j else sys.maxsize for j in range(self.nr_vs)] for i in range(self.nr_vs)]

        for tail in range(self.nr_vs):
            for head in self.edges[tail]:
                self.distance_matrix[tail][head] = self.weights[(tail, head)]

        # Floyd-Warshall algorithm
        for k in range(self.nr_vs):
            self.temp_matrix = [row[:] for row in self.distance_matrix]
            for i in range(self.nr_vs):
                for j in range(self.nr_vs):
                    self.distance_matrix[i][j] = min(self.temp_matrix[i][j], self.temp_matrix[i][k] + self.temp_matrix[k][j])

        # Check for negative weight cycles
        for i in range(self.nr_vs):
            if self.distance_matrix[i][i] < 0:
                print("Negative cycle detected!")
                sys.exit()

        return self.distance_matrix

if __name__ == "__main__":
    # Path to the file containing the graph data
    file_name = 'APSPtest3.txt'

    start_time = time.time()

    # Initialize and build the graph from the file
    with open(file_name, 'r') as file:
        num_vertices, _ = map(int, file.readline().strip().split())
        graph = WeightedDirectedGraph(num_vertices)
        for line in file:
            tail, head, weight = map(int, line.strip().split())
            graph.add_edge(tail - 1, head - 1, weight)

    # Compute shortest paths
    shortest_paths = graph.run_floyd_warshall()

    # Print the shortest paths
    for i in range(num_vertices):
        for j in range(num_vertices):
            print(f"From {i} to {j} the shortest path is {shortest_paths[i][j]}")

    # Find and print the minimum distance
    min_distance = min(min(row) for row in shortest_paths)
    print(f"Minimum distance is {min_distance}")

    end_time = time.time()
    print(f"Time taken: {end_time - start_time:.4f} seconds")
