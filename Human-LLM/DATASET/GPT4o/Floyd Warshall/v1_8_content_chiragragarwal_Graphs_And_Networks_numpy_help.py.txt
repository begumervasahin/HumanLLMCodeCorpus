import numpy as np

def main():
    # Define the matrix as a string with semicolons as row delimiters
    mat_str = '1 2 3 4 ; 5 6 7 8 ; 0 1 2 3'
    
    # Convert the string into a format suitable for numpy matrix creation
    mat_str = mat_str.replace(';', '\n')
    
    # Create a numpy matrix from the formatted string
    graph_mat = np.matrix(mat_str)
    
    # Print the shape of the matrix
    print(f"Matrix shape: {graph_mat.shape}")
    
    # Access and print specific elements from the matrix
    print(f"Element at position (0, 1): {graph_mat[0, 1]}")
    print(f"Element at position (2, 3): {graph_mat[2, 3]}")
    
    # Modify an element in the matrix
    graph_mat[0, 1] = -8
    
    # Print the modified row
    print(f"Modified row 1: {graph_mat[1]}")
    
    # Print the entire column
    print(f"Column 1: {graph_mat[:, 1]}")

if __name__ == '__main__':
    main()
