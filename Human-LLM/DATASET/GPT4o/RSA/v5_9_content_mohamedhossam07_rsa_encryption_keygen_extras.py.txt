from random import randint

def power(x, y, p):
    """Computes (x ** y) % p."""
    return (x ** y) % p

def decimal_to_binary(decimalin):
    """Converts a decimal number to binary."""
    binary_str = bin(int(decimalin))[2:]
    return binary_str

def adjust_size(key, size):
    """Adjusts the size of a binary string to the specified size."""
    if len(key) < size:
        return key.rjust(size, '0')
    elif len(key) > size:
        return key[:size]
    return key

def binary_to_decimal(binaryin):
    """Converts a binary string to a decimal number."""
    return int(binaryin, 2)

def random_binary(length):
    """Generates a random binary string of a given length and converts it to decimal."""
    binary_str = ''.join(adjust_size(bin(randint(0, 255))[2:], 8) for _ in range(length // 8))
    binary_str = adjust_size(binary_str, length)
    return binary_to_decimal(binary_str)

def multiplicative_inverse(a, b):
    """Computes the multiplicative inverse of a modulo b."""
    x, lx = 0, 1
    y, ly = 1, 0
    original_b = b

    while b != 0:
        q = a // b
        a, b = b, a % b
        lx, x = x, lx - q * x
        ly, y = y, ly - q * y

    if lx < 0:
        lx += original_b

    return lx
