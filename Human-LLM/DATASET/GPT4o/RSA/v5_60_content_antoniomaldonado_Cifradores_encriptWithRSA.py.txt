import math
import sys
import utils.utils as utils

def get_int_blocks(msg, block_size):
    """
    Converts a message into a list of integer blocks based on allowed characters.

    Args:
        msg (str): The message to be converted.
        block_size (int): The size of each block.

    Returns:
        list: A list of integer blocks.
    """
    allowed_chars = utils.allowed_chars()

    # Validate that all characters in the message are allowed
    for character in msg:
        if character not in allowed_chars:
            print(f'Character in the message cannot be encrypted: {character}')
            sys.exit()

    block_ints = []
    num_allowed_chars = len(allowed_chars)

    for block_start in range(0, len(msg), block_size):
        block_int = 0
        for i in range(block_start, min(block_start + block_size, len(msg))):
            char_index = allowed_chars.index(msg[i])
            block_int += char_index * (num_allowed_chars ** (i % block_size))
        block_ints.append(block_int)

    return block_ints

def get_encrypted_blocks(msg, key, block_size):
    """
    Encrypts the integer blocks of a message using the provided key.

    Args:
        msg (str): The message to be encrypted.
        key (tuple): The public key (num, enc).
        block_size (int): The size of each block.

    Returns:
        list: A list of encrypted blocks.
    """
    num, enc = key
    int_blocks = get_int_blocks(msg, block_size)
    encrypted_blocks = [pow(block, enc, num) for block in int_blocks]
    return encrypted_blocks

def encrypt(input_file_name, encrypted_file_name, pub_key_file_name):
    """
    Encrypts a message from an input file using a public key from a file and writes the encrypted message to an output file.

    Args:
        input_file_name (str): The name of the file containing the message to encrypt.
        encrypted_file_name (str): The name of the file to save the encrypted message.
        pub_key_file_name (str): The name of the file containing the public key.
    """
    msg = utils.read_file(input_file_name)
    key_size, n, e = utils.read_key_file(pub_key_file_name)
    allowed_chars = utils.allowed_chars()

    block_size = int(math.log(2 ** key_size, len(allowed_chars)))

    if not utils.is_valid(key_size, block_size):
        sys.exit('Wrong block size. Encryption aborted.')

    encrypted_blocks = get_encrypted_blocks(msg, (n, e), block_size)
    encrypted_blocks_str = [str(block) for block in encrypted_blocks]

    with open(encrypted_file_name, 'w') as fo:
        fo.write(f'{len(msg)}|{block_size}|{",".join(encrypted_blocks_str)}')

def main():
    """
    Main function to run the encryption process.
    """
    input_file_name = input('1. Enter the file name with the message to encrypt: ')
    pub_key_file_name = input('2. Enter the file name containing the public key: ')
    output_file_name = input('3. Enter the file name to save the encrypted message: ')

    print('Encrypting...')
    encrypt(input_file_name, output_file_name, pub_key_file_name)
    print('Encryption complete.')

if __name__ == '__main__':
    main()
