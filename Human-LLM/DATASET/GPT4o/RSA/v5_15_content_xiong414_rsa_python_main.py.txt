import math
import random

class RSA:
    def __init__(self, p, q):
        self.p = p
        self.q = q
        self._validate_primes()
        self._generate_keys()

    def _validate_primes(self):
        if not self._is_prime(self.p):
            raise ValueError('p is not a prime number')
        if not self._is_prime(self.q):
            raise ValueError('q is not a prime number')

    @staticmethod
    def _is_prime(n):
        if n <= 1:
            return False
        if n <= 3:
            return True
        if n % 2 == 0 or n % 3 == 0:
            return False
        for i in range(5, int(math.sqrt(n)) + 1, 6):
            if n % i == 0 or n % (i + 2) == 0:
                return False
        return True

    @staticmethod
    def _generate_e(phi):
        while True:
            e = random.randint(2, phi - 1)
            if math.gcd(e, phi) == 1:
                return e

    @staticmethod
    def _extended_euclid(a, b):
        x1, x2, x3 = 1, 0, a
        y1, y2, y3 = 0, 1, b
        while y3 != 0 and y3 != 1:
            q = x3 // y3
            x1, x2, x3, y1, y2, y3 = y1, y2, y3, x1 - q * y1, x2 - q * y2, x3 - q * y3
        return None if y3 == 0 else (y2 if y2 > 0 else y2 + a)

    def _generate_keys(self):
        self.n = self.p * self.q
        phi = (self.p - 1) * (self.q - 1)
        self.e = self._generate_e(phi)
        self.d = self._extended_euclid(phi, self.e)
        self.public_key = (self.n, self.e)
        self.private_key = self.d

    def decrypt(self, ciphertext):
        return pow(ciphertext, self.d, self.n)
