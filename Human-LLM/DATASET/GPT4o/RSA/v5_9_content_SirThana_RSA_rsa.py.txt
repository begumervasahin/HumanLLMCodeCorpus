import random
import math

def rsa_keygen(p, q):
    """Generate RSA keys."""
    # Calculate n and Euler's Totient function (phi)
    n = p * q
    phi = (p - 1) * (q - 1)
    
    # Find all possible relative primes e such that 1 < e < phi and gcd(e, phi) = 1
    relative_primes = [e for e in range(2, phi) if math.gcd(e, phi) == 1]
    
    # Randomly select e from the relative primes list
    e = random.choice(relative_primes)
    
    # Find d, the modular multiplicative inverse of e modulo phi
    d = pow(e, -1, phi)
    
    return relative_primes, phi, n, d, e

def encrypt(public_exponent, modulus, plaintext):
    """Encrypt the plaintext using the public key (public_exponent, modulus)."""
    return pow(plaintext, public_exponent, modulus)

def decrypt(private_exponent, modulus, ciphertext):
    """Decrypt the ciphertext using the private key (private_exponent, modulus)."""
    return pow(ciphertext, private_exponent, modulus)

def main():
    # Define prime numbers p and q
    p = 1733
    q = 1301
    
    # Generate RSA keys
    relative_primes, phi, n, d, e = rsa_keygen(p, q)
    
    # Display the generated values
    print(f"Relative Primes: {relative_primes}")
    print(f"phi (Euler's Totient): {phi}")
    print(f"Modulus (n): {n}")
    print(f"Private Exponent (d): {d}")
    print(f"Public Exponent (e): {e}\n")
    
    # Sample payload
    plaintext = 2999
    print(f"Starting value: {plaintext}")
    
    # Encrypt the payload
    ciphertext = encrypt(e, n, plaintext)
    print(f"Encrypted: {ciphertext}")
    
    # Decrypt the payload
    decrypted_plaintext = decrypt(d, n, ciphertext)
    print(f"Decrypted: {decrypted_plaintext}")

if __name__ == "__main__":
    main()
