def decimal(decimalin):
    decimalin = int(decimalin)
    rest = ""
    while True:
        rest += str(decimalin % 2)
        decimalin = decimalin // 2
        if decimalin == 0:
            break
    return rest[::-1]

def size_be_R(key, size):
    if len(key) < size:
        x_key = key.rjust(size, '0')
    elif len(key) > size:
        x_key = key[:size]
    else:
        x_key = key
    return x_key

def binary2dec(binaryin):
    binaryin = str(binaryin)
    binaryin = binaryin[::-1]
    binarydecimal = sum(int(bit) * (2 ** idx) for idx, bit in enumerate(binaryin))
    return binarydecimal

def rand_bin(leng):
    out = ''
    for _ in range(int(leng / 8)):
        new = str(bin(randint(0, 10)))[2:]
        new = size_be_R(new, 8)
        out += new
    out = size_be_R(out, leng)
    return binary2dec(out)

def multiplicative_inverse(a, b):
    x, y, lx, ly = 0, 1, 1, 0
    oa, ob = a, b
    while b != 0:
        q = a // b
        a, b = b, a % b
        x, lx = lx - q * x, x
        y, ly = ly - q * y, y
    if lx < 0:
        lx += ob
    if ly < 0:
        ly += oa
    return lx
