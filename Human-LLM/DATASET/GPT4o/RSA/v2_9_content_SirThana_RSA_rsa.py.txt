import random
import math

def generate_keys(p, q):
    n = p * q
    phi_n = (p - 1) * (q - 1)
    
    # Find all relative primes for e
    relative_primes = []
    for e in range(2, phi_n):
        print("Finding possible relative primes...", int((e / phi_n) * 100), "%")
        if math.gcd(e, n) == 1 and math.gcd(e, phi_n) == 1:
            relative_primes.append(e)
    
    e = random.choice(relative_primes)
    d = find_d(e, phi_n)
    
    return n, e, d

def find_d(e, phi_n):
    for d in range(1, phi_n):
        if (d * e) % phi_n == 1:
            return d
    return None
