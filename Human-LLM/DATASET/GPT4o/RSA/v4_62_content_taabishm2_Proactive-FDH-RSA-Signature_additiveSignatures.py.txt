def init_prime_group(n):
    """Generate a prime group with order greater than n."""
    return nextprime.next_prime(n)

def pick_sum(p):
    """Pick a sum such that di + di_dash = sum and p < sum < 2p."""
    return random.randrange(p, 2 * p)

def generate_challenge(shares, p, c):
    """Generate challenge and verifier. Return as [challenge(list), verifier(int)]."""
    challenge = []
    a = random.randrange(2, p)
    verifier = pow(a, c, p)
    for di in shares:
        challenge.append(pow(a, di, p))
    return challenge, verifier, a

def generate_response(challenge, shares, p, c, verifier, gen):
    """Generate response by a party holding a 'share' to the challenge for all additive shares."""
    responses = []
    for i in range(len(challenge)):
        response = (pow(gen, c - shares[i], p) * challenge[i]) % p
        responses.append(response == verifier)
    return responses
