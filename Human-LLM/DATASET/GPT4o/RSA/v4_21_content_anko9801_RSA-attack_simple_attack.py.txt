import random

def modular_exp(a, b, n):
    res = 1
    while b > 0:
        if b & 1 == 1:
            res = (res * a) % n
        a = (a * a) % n
        b >>= 1
    return res

def gen_rand(bit_length):
    bits = [random.randint(0, 1) for _ in range(bit_length - 2)]
    ret = 1
    for b in bits:
        ret = ret * 2 + int(b)
    return ret * 2 + 1

def mr_primality_test(n, k=100):
    if n == 1:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False
    
    d = n - 1
    s = 0
    while d % 2 == 0:
        d //= 2
        s += 1
    
    def check_composite(a, d, n, s):
        if modular_exp(a, d, n) == 1:
            return False
        for i in range(s):
            if modular_exp(a, d * (2 ** i), n) == n - 1:
                return False
        return True
    
    for _ in range(k):
        a = random.randint(2, n - 2)
        if check_composite(a, d, n, s):
            return False
    
    return True

def gen_prime(bit_length):
    while True:
        candidate = gen_rand(bit_length)
        if mr_primality_test(candidate):
            return candidate

def gcd(x, y):
    while y != 0:
        x, y = y, x % y
    return x

def extended_gcd(a, b):
    x0, x1 = 1, 0
    y0, y1 = 0, 1
    while b != 0:
        q, a, b = a // b, b, a % b
        x0, x1 = x1, x0 - q * x1
        y0, y1 = y1, y0 - q * y1
    return a, x0, y0

def gen_d(e, phi):
    _, x, _ = extended_gcd(e, phi)
    return x % phi

def pollards_rho(n):
    def f(x):
        return (x * x + 1) % n
    
    x, y, d = 2, 2, 1
    while d == 1:
        x = f(x)
        y = f(f(y))
        d = gcd(abs(x - y), n)
    return d

def simple_attack(n, e, c):
    p = pollards_rho(n)
    q = n // p
    phi = (p - 1) * (q - 1)
    d = gen_d(e, phi)
    return p, q, d
