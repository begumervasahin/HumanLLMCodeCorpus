import random
import time
import Prime

def encryption(plaintext, public_key):
    """Encrypt the plaintext using the public key."""
    n, e = public_key
    return Prime.quick_pow_mod(plaintext, e, n)

def decryption(ciphertext, private_key):
    """Decrypt the ciphertext using the private key."""
    n, d = private_key
    return Prime.quick_pow_mod(ciphertext, d, n)

def generate_rsa_key():
    """Generate an RSA key pair."""
    start_time = time.perf_counter()
    
    p, q = Prime.get_rand_prime_arr(2)
    while p == q:
        q = random.choice(Prime.get_rand_prime_arr(2))
    
    end_time = time.perf_counter()
    
    n = p * q
    phi = (p - 1) * (q - 1)
    e = 65537
    d = Prime.mod_inverse(e, phi)
    
    print(f"随机生成的素数 p = {p}")
    print(f"随机生成的素数 q = {q}")
    print(f'素数生成的时间为： {end_time - start_time:.6f} 秒')
    print(f"n = p * q = {n}")
    print(f"利用Euclidean算法生成的公钥 e = {e}")
    print(f"利用扩展Euclidean算法生成的私钥 d = {d}")
    
    return (n, e), (n, d)

def measure_time(func, *args):
    """Measure the execution time of a function."""
    start_time = time.perf_counter()
    result = func(*args)
    end_time = time.perf_counter()
    return result, end_time - start_time

def main():
    """Main function to handle RSA encryption and decryption."""
    public_key, private_key = generate_rsa_key()
    
    print('请输入明文：')
    plaintext = int(input())
    
    ciphertext, encryption_time = measure_time(encryption, plaintext, public_key)
    print(f"RSA加密后的密文为: {ciphertext}")
    print(f"加密时间为: {encryption_time:.6f} 秒")
    
    decrypted_text, decryption_time = measure_time(decryption, ciphertext, private_key)
    print(f"RSA解密后的明文为: {decrypted_text}")
    print(f"解密时间为: {decryption_time:.6f} 秒")
    
    if plaintext == decrypted_text:
        print('解密成功')
    
    total_encryption_time = 0
    total_decryption_time = 0
    
    for _ in range(10):
        print('请输入明文：')
        plaintext = int(input())
        
        ciphertext, encryption_time = measure_time(encryption, plaintext, public_key)
        total_encryption_time += encryption_time
        
        print(f"RSA加密后的密文为: {ciphertext}")
        
        decrypted_text, decryption_time = measure_time(decryption, ciphertext, private_key)
        total_decryption_time += decryption_time
        
        print(f"RSA解密后的明文为: {decrypted_text}")
    
    print(f'10组明文的RSA加密时间为： {total_encryption_time:.6f} 秒')
    print(f'10组明文的RSA解密时间为： {total_decryption_time:.6f} 秒')

if __name__ == '__main__':
    main()
