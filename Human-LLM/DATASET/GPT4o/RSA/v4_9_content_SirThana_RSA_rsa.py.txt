import random

def RSA(p, q):
    # Calculate n and Euler's Totient function (phi)
    n = p * q
    phi = (p - 1) * (q - 1)
    
    # Find all possible relative primes e such that 1 < e < phi and gcd(e, phi) = 1
    relative_primes = [e for e in range(2, phi) if math.gcd(e, phi) == 1]
    
    # Randomly select e from the relative primes list
    e = random.choice(relative_primes)
    
    # Find d, the modular multiplicative inverse of e modulo phi
    d = pow(e, -1, phi)
    
    return relative_primes, phi, n, d, e

def encrypt(e, n, plaintext):
    # Encrypt the plaintext using the public key (e, n)
    return pow(plaintext, e, n)

def decrypt(d, n, ciphertext):
    # Decrypt the ciphertext using the private key (d, n)
    return pow(ciphertext, d, n)

    p = 1733
    q = 1301
    
    # Generate RSA keys
    relative_primes, phi, n, d, e = RSA(p, q)
    
    # Display the generated values
    print(f"Relative Primes: {relative_primes}")
    print(f"phi: {phi}, n: {n}, d: {d}, e: {e}\n")
    
    # Sample payload
    payload = 2999
    print(f"Starting value: {payload}")
    
    # Encrypt the payload
    encrypted_payload = encrypt(e, n, payload)
    print(f"Encrypted: {encrypted_payload}")
    
    # Decrypt the payload
    decrypted_payload = decrypt(d, n, encrypted_payload)
    print(f"Decrypted: {decrypted_payload}")
