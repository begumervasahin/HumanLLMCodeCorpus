import random
from random import randrange

# Convert a binary string to a decimal integer
def binary2dec(binary):
    return int(binary, 2)

# Convert a decimal integer to a binary string of a given length
def dec2binary(dec, length):
    return bin(dec)[2:].zfill(length)

# Generate a random binary string of a given length
def rand_bin(length):
    return ''.join([str(randrange(0, 2)) for _ in range(length)])

# Compute the two's complement of a binary string of a given length
def two_com(binary, length):
    if binary[0] == '1':  # if the number is negative
        complement = ''.join('1' if bit == '0' else '0' for bit in binary)
        return bin(binary2dec(complement) + 1)[2:].zfill(length)
    return binary.zfill(length)

# Compute the greatest common divisor using the Euclidean algorithm
def gcd(a, b):
    while b != 0:
        a, b = b, a % b
    return a

# Extended Euclidean Algorithm to find the modular inverse
def egcd(a, b):
    if a == 0:
        return b, 0, 1
    else:
        g, x, y = egcd(b % a, a)
        return g, y - (b // a) * x, x

def multiplicative_inverse(e, phi):
    g, x, y = egcd(e, phi)
    if g != 1:
        raise Exception('No modular inverse')
    else:
        return x % phi

# Miller-Rabin primality test
def isPrime(n, k=5):  # Number of tests = k
    if n <= 1:
        return False
    if n <= 3:
        return True
    if n % 2 == 0 or n % 3 == 0:
        return False
    r, s = 0, n - 1
    while s % 2 == 0:
        r += 1
        s //= 2
    for _ in range(k):
        a = randrange(2, n - 1)
        x = pow(a, s, n)
        if x == 1 or x == n - 1:
            continue
        for _ in range(r - 1):
            x = pow(x, 2, n)
            if x == n - 1:
                break
        else:
            return False
    return True

def main():
    mlen = 512
    rlen = 512
    eleng = 1024

    # Generate a random prime number m
    m = rand_bin(mlen)
    m = two_com(m, mlen)
    while not isPrime(binary2dec(m)):
        m = rand_bin(mlen)
        m = two_com(m, mlen)

    # Generate a random prime number r
    r = rand_bin(rlen)
    r = two_com(r, rlen)
    while not isPrime(binary2dec(r)):
        r = rand_bin(rlen)
        r = two_com(r, rlen)

    # Generate a random number e
    e = rand_bin(eleng)
    e = two_com(e, eleng)
    
    # Convert m and r to decimal
    m = binary2dec(m)
    r = binary2dec(r)
    n = m * r
    phi = (m - 1) * (r - 1)
    
    e = binary2dec(e)
    # Ensure e is coprime with phi and is a prime number
    while gcd(e, phi) != 1 or not isPrime(e):
        e = rand_bin(eleng)
        e = two_com(e, eleng)
        e = binary2dec(e)
    
    # Compute the private key d
    d = multiplicative_inverse(e, phi)
    
    # Print the results
    print('n =', n)
    print('Public Key (e) =', e)
    print('Private Key (d) =', d)

if __name__ == "__main__":
    main()
