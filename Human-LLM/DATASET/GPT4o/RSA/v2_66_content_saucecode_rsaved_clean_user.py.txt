import os
import sys
import json
import time
import pickle

# Placeholder for the rsaved module
import rsaved
__version__ = rsaved.__version__

def print_job(job, item):
    """Print job details with a delay for better readability."""
    def sprint(*args):
        time.sleep(0.2)
        print(*args)
        
    sprint()
    sprint('Name:', job['name'], 'Completed:', job['completed'])
    sprint('Title:', item['data'].get('title'))
    sprint('URL:', item['data'].get('url'))
    sprint('Commands:')
    for code, cmd in zip(job['returncodes'], job['commands'][1:]):
        sprint('    $', ' '.join(cmd))
        sprint('Exit code:', code)
        sprint()
    sprint()

def load_file(fname, json_load=False):
    """Load the contents of a file, optionally parsing it as JSON."""
    with open(fname, 'r') as f:
        return json.load(f) if json_load else f.read()

def main():
    print(f'rsaved/{__version__} clean_user.py')

    # Ensure the correct number of arguments are provided
    if len(sys.argv) != 2:
        print('Usage:', sys.argv[0], '[username]')
        sys.exit(1)

    username = sys.argv[1]
    user_folder = f'user/{username}'
    library_folder = f'{user_folder}/library'

    # Check if the user folder exists
    if not os.path.exists(user_folder):
        print('User does not exist.')
        sys.exit(2)

    changes_made = False
    index = rsaved.load_index(username)
    
    # Load all JSON files in the library folder
    json_files = [
        load_file(f'{library_folder}/{file}', True)
        for file in os.listdir(library_folder) if file.endswith('json')
    ]
    
    # Find jobs with non-zero exit codes
    errored_jobs = [
        file for file in json_files if any(code != 0 for code in file.get('returncodes', [0]))
    ]

    print(f'Found {len(errored_jobs)} jobs which exited with error codes.')
    print('I will now enumerate through them.')

    # Process each errored job
    for job in errored_jobs:
        item = next((i for i in index if i['data']['name'] == job['name']), None)
        if item is None:
            continue
        
        print_job(job, item)
        
        # Ask user if they want to mark the job as ignored
        if input(f'Would you like to mark item {job["name"]} as ignored? [Y/n] ').lower() != 'n':
            item['rsaved']['ignore'] = True
            print('Ignored.')
            changes_made = True
        else:
            print('Not ignored (no change).')

    # Write changes to index if any modifications were made
    if changes_made:
        print('Writing changes to index...')
        rsaved.dump_index(username, index)
    else:
        print('No changes to the index were needed.')

    # Clean/archive completed job files
    print('Cleaning/archiving job files...')
    rsaved.library_clean_completed(username)
    print('Done.')

if __name__ == "__main__":
    main()
