def calc(expression):
    if not isinstance(expression, list):
        expression = expression.split(' ')
    operators = ['+', '-', '*', '/']
    
    for i, item in enumerate(expression):
        if item in operators:
            result = eval(f"{expression[i-2]} {item} {expression[i-1]}")
            expression.insert(i-2, str(result))
            for _ in range(3):
                del expression[i-1]
            return calc(expression)
    
    return sum(map(eval, expression))

def translate(expression):
    elements, stack, output = [], [], []
    element = ''
    
    # Tokenize the input expression
    for char in expression:
        if char.isdigit() or char == '.':
            element += char
        else:
            if element:
                elements.append(element)
                element = ''
            if char in ['+', '-', '*', '/', '(', ')']:
                elements.append(char)
    
    if element:
        elements.append(element)
    
    elements.insert(0, '(')
    elements.append(')')
    
    # Convert infix expression to postfix (RPN) using the shunting yard algorithm
    for token in elements:
        if token.isdigit() or '.' in token:
            output.append(token)
        elif token == '(':
            stack.append(token)
        elif token == ')':
            while stack and stack[-1] != '(':
                output.append(stack.pop())
            stack.pop()
        elif token in ['+', '-']:
            while stack and stack[-1] != '(':
                output.append(stack.pop())
            stack.append(token)
        elif token in ['*', '/']:
            while stack and stack[-1] in ['*', '/']:
                output.append(stack.pop())
            stack.append(token)
    
    return output
