import numpy as np

class EstimatePerceptualMixingTime:
    def __init__(self, RoomDims):
        self.RoomDims = RoomDims
        self.mixing_time_estimate = {}

    def model_based(self):
        print('----------------------------------')
        print('MODEL-BASED MIXING TIME PREDICTION')
        print('----------------------------------')
        print('ROOM PROPERTIES:')
        x_d, y_d, z_d = self.RoomDims

        if not all(isinstance(dim, (int, float)) and dim > 0 for dim in [x_d, y_d, z_d]):
            raise ValueError('RoomDims must contain positive, numeric values only.')

        print(f'Room dimensions: height = {x_d}m, length = {y_d}m, width = {z_d}m')
        print('Perceptual mixing times tmp50 and tmp95 (in ms) from model-based predictors:')

        volume = x_d * y_d * z_d
        surface = 2 * (x_d * y_d + x_d * z_d + y_d * z_d)

        print(f'Volume: {volume}m³')
        print(f'Surface area: {surface}m²')

        rootvol = np.sqrt(volume)
        MFP = 47 * volume / surface
        tmp50 = 20.08 * volume / surface + 12
        tmp95 = 0.0117 * volume + 50.1

        print(f'tmp50: {tmp50}ms')
        print(f'tmp95: {tmp95}ms')

        self.mixing_time_estimate.update({
            'model': {
                'tmp50': tmp50,
                'tmp95': tmp95,
                'rootV': rootvol,
                'mfp_based': MFP
            }
        })

        return self

# Example usage
dims = [5.0, 6.0, 4.0]
estimator = EstimatePerceptualMixingTime(dims)
estimator.model_based()
