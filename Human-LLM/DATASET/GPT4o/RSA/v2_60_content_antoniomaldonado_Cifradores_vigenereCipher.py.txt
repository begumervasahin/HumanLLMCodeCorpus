
import string

# Define the alphabet to be used for encryption and decryption
ALPHABET_UPPER = string.ascii_uppercase

def encrypt_msg(key, message):
    """
    Encrypt the message using the provided key.
    """
    return translate_msg(key, message, 'encrypt')

def decrypt_msg(key, message):
    """
    Decrypt the message using the provided key.
    """
    return translate_msg(key, message, 'decrypt')

def translate_msg(key, message, mode):
    """
    Translate the message either for encryption or decryption based on the mode.
    """
    translated = []
    key = key.upper()
    key_index = 0
    is_encrypt = mode == 'encrypt'
    
    for symbol in message:
        offset = ALPHABET_UPPER.find(symbol.upper())
        if exist_symbol(offset):
            offset = get_alphabet_offset(is_encrypt, key, key_index, offset)
            if symbol.isupper():
                translated.append(ALPHABET_UPPER[offset])
            elif symbol.islower():
                translated.append(ALPHABET_UPPER[offset].lower())
            key_index = key_iter(key, key_index)
        else:
            translated.append(symbol)
    
    return ''.join(translated)

def exist_symbol(offset):
    """
    Check if the symbol exists in the alphabet.
    """
    return offset != -1

def get_alphabet_offset(is_encrypt, key, key_index, offset):
    """
    Calculate the new offset for the symbol based on whether we are encrypting or decrypting.
    """
    key_offset = ALPHABET_UPPER.find(key[key_index])
    if is_encrypt:
        offset += key_offset
    else:
        offset -= key_offset
    return offset % len(ALPHABET_UPPER)

def key_iter(key, key_index):
    """
    Move to the next character in the key, wrapping around if necessary.
    """
    key_index += 1
    if key_index == len(key):
        key_index = 0
    return key_index

def main():
    """
    Main function to run the script. Handles user input and output.
    """
    print('1. Introduce a message: ')
    message = input()
    print('2. Introduce the key: ')
    key = input()
    print('3. Introduce the word "encrypt" to encrypt the message or press Enter to decrypt: ')
    mode = input().lower()
    if mode == 'encrypt':
        translated = encrypt_msg(key, message)
    else:
        translated = decrypt_msg(key, message)
    print(f'{mode.title() if mode else "Decrypt"}ed message:')
    print(translated)

if __name__ == '__main__':
    main()
