def decimal(decimalin):
    decimalin =int(decimalin)
    rest =""
    while True:
        rest += str(decimalin % 2)
        decimalin = decimalin // 2
        if decimalin == 0:
            break
    return rest[::-1]
def size_be_R(key,size):
    if len(key) < size :
        x_key = key.rjust(size,'0')
    elif len(key) > size :
        x_key = key[:size]
    else:
        x_key = key
    return x_key

def multiplicative_inverse(a, b):
    """Returns a tuple (r, i, j) such that r = gcd(a, b) = ia + jb
    """
    x = 0
    y = 1
    lx = 1
    ly = 0
    oa = a
    ob = b
    while b != 0:
        q = a // b
        (a, b) = (b, a % b)
        (x, lx) = ((lx - (q * x)), x)
        (y, ly) = ((ly - (q * y)), y)
    if lx < 0:
        lx += ob
    if ly < 0:
        ly += oa
    return lx
