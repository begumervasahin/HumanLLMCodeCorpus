"""Django's command-line utility for administrative tasks."""

import os
import sys

def main():
    """
    Set the default Django settings module and execute the command-line utility.

    This function sets the default Django settings module for the 'mypubsub' project and
    executes the command-line utility with the provided arguments. If Django is not installed
    or cannot be imported, it raises an ImportError with a helpful message.
    """
    os.environ.setdefault('DJANGO_SETTINGS_MODULE', 'mypubsub.settings')

    try:
        from django.core.management import execute_from_command_line
    except ImportError as exc:
        raise ImportError(
            "Couldn't import Django. Make sure it's installed and available "
            "on your PYTHONPATH environment variable. Did you forget to activate a virtual environment?"
        ) from exc

    execute_from_command_line(sys.argv)

if __name__ == '__main__':
    main()
