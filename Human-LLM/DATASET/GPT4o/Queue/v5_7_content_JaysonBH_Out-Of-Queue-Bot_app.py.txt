import os
import slack
import flask
from flask import Flask, request, redirect
from pymongo import MongoClient
from slackbot import SlackBot
from roster import Roster
import requests
from tasks import choose_command, processEvent

# Load environment variables
client_id = os.environ["SLACK_CLIENT_ID"]
client_secret = os.environ["SLACK_CLIENT_SECRET"]
oauth_scope = os.environ["SLACK_SCOPE"]
CONNECT_STRING = os.environ["CONNECT_STRING"]
ZOOM_MONGO = os.environ["ZOOM_MONGO"]

# Initialize Flask app
app = Flask(__name__)

# MongoDB connection
client = MongoClient(CONNECT_STRING)
db = client.queue
employees = db.employees

zoomDBClient = MongoClient(ZOOM_MONGO)
zdb = zoomDBClient.zoom
zoomUsers = zdb.users

# Initialize SlackBot
slack_bot = SlackBot()

# Define commands and privileged commands/users
COMMANDS = {"list", "listall", "run", "runall", "refresh", "test", "zoom"}
PRIVILEGED_COMMANDS = {"listall", "runall"}
PRIVILEGED_USERS = {"UF57DA49F", "U054G9J88", "U26R8ML07"}

@app.errorhandler(404)
def page_not_found(e):
    return "Command not found"

@app.route("/", methods=["GET"])
def pre_install():
    """Handles the pre-installation authorization."""
    name = request.args['name'].split(' ')
    first_name = name[0]
    last_name = name[-1]
    if len(name) == 3:
        first_name += '+' + name[1]
    return f'<a href="https://slack.com/oauth/authorize?scope={oauth_scope}&client_id={client_id}&state={first_name}+{last_name}">Authorize App (you will be redirected to Slack)</a>'

@app.route("/finish_auth", methods=["GET", "POST"])
def post_install():
    """Handles post-installation of the Slack app."""
    auth_code = request.args['code']
    name = request.args['state'].split(' ')
    first_name = name[0]
    if len(name) == 3:
        last_name = name[2]
        first_name += ' ' + name[1]
    else:
        last_name = name[1]

    client = slack.WebClient(token="")
    response = client.oauth_access(client_id=client_id, client_secret=client_secret, code=auth_code)
    
    person = {
        'access_token': response['access_token'],
        'user_id': response['user_id']
    }
    employees.update(
        {'first_name': first_name, 'last_name': last_name},
        {'$set': person},
        upsert=False
    )

    roster = Roster("password.json", "EAST")
    roster.setOutOfQueue()
    choose_command.apply_async(args=("run", response['user_id']), queue="commands")
    
    return "Auth complete! You will receive a notification on your Out of Queue day, and your status will be updated!"

@app.route("/command", methods=["POST"])
def exec_command():
    """Executes a command received from Slack."""
    user_id = request.form.get("user_id")
    command = request.form.get("text")
    
    if command not in COMMANDS:
        return redirect(404)
    
    if command in PRIVILEGED_COMMANDS and user_id not in PRIVILEGED_USERS:
        return "You are not authorized to use this command. Please reach out in Slack for assistance."
    
    choose_command.apply_async(args=(command, user_id), queue="commands")
    return "Executing command."

@app.route("/events", methods=["POST"])
def events():
    """Handles events received from the Slack API."""
    event_data = request.get_json()
    print(event_data)
    processEvent.apply_async(args=(event_data,), queue="events")
    return "Event received"

@app.route("/zoom", methods=["POST"])
def zoom():
    """Handles Zoom events such as participant joining or leaving a meeting."""
    event_data = request.get_json()
    event_type = event_data.get('event')

    if event_type == 'meeting.participant_joined':
        handle_zoom_participant_joined(event_data)
    elif event_type == 'meeting.participant_left':
        handle_zoom_participant_left(event_data)
    
    return "Event received"

def handle_zoom_participant_joined(event):
    """Handles participant joined Zoom event."""
    try:
        participant_id = event['payload']['object']['participant']['id']
    except KeyError:
        return "There is no ID"

    match = zoomUsers.update_one({'zoomID': participant_id}, {'$inc': {'num_meetings': 1}})
    if match.matched_count != 0:
        user_name = event['payload']['object']['participant']['user_name']
        print(f"\033[96m {user_name} joined a zoom meeting! \033[00m")
        slack_bot.slackBotUser.chat_postMessage(
            channel='#general', 
            text=f"{user_name} just joined a zoom meeting.",
            username='Availability Bot', 
            link_names=1, 
            as_user=True
        )

def handle_zoom_participant_left(event):
    """Handles participant left Zoom event."""
    try:
        participant_id = event['payload']['object']['participant']['id']
    except KeyError:
        return "There is no ID"

    match = zoomUsers.update_one({'zoomID': participant_id, 'num_meetings': {'$gt': 0}}, {'$inc': {'num_meetings': -1}})
    if match.matched_count != 0:
        user_name = event['payload']['object']['participant']['user_name']
        print(f"\033[96m {user_name} left a zoom meeting! \033[00m")
        slack_bot.slackBotUser.chat_postMessage(
            channel='#general', 
            text=f"{user_name} just left a zoom meeting.",
            username='Availability Bot', 
            link_names=1, 
            as_user=True
        )

if __name__ == "__main__":
    app.run()
