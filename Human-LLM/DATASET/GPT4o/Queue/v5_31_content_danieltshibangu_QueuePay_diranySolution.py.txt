class Queue:
    def __init__(self):
        """Initialize an empty queue."""
        self.items = []

    def is_empty(self):
        """Check if the queue is empty."""
        return not self.items

    def enqueue(self, item):
        """Add an item to the end of the queue."""
        self.items.append(item)

    def dequeue(self):
        """Remove and return the item from the front of the queue."""
        return self.items.pop(0) if not self.is_empty() else None

    def size(self):
        """Return the number of items in the queue."""
        return len(self.items)

class Employee:
    def __init__(self, first, last, pay):
        """
        Initialize an Employee object.

        :param first: First name of the employee.
        :param last: Last name of the employee.
        :param pay: Pay of the employee.
        """
        self.first = first
        self.last = last
        self.pay = float(pay)
        self.email = f'{first}.{last}@company.com'
        self.bonus = 0.0

    def get_pay(self):
        """Return the pay of the employee."""
        return self.pay

    def set_pay(self, pay):
        """Set the pay of the employee."""
        self.pay = float(pay)

    def set_bonus(self, amount):
        """Set the bonus of the employee."""
        self.bonus = float(amount)

    def get_bonus(self):
        """Return the bonus of the employee."""
        return self.bonus

    def full_name(self):
        """Return the full name of the employee."""
        return f'{self.first} {self.last}'

    def __str__(self):
        """Return the string representation of the employee."""
        return (
            f'\nEmployee name: {self.full_name()}'
            f'\nEmployee pay: {self.pay}'
            f'\nEmployee bonus: {self.bonus:.2f}'
        )

def read_employees(file_path):
    """
    Read employee data from a file and return a queue of Employee objects.

    :param file_path: Path to the file containing employee data.
    :return: A queue of Employee objects.
    """
    employee_queue = Queue()
    pay_rate = 0.2
    total_bonus = 0.0

    with open(file_path, 'r') as file:
        for line in file:
            data_attr = line.split()
            if len(data_attr) < 3:
                continue
            employee = Employee(data_attr[0], data_attr[1], data_attr[2])
            employee.set_bonus(employee.get_pay() * pay_rate)
            total_bonus += employee.get_bonus()
            employee_queue.enqueue(employee)
            pay_rate = max(0, pay_rate - 0.01)

    return employee_queue, total_bonus

def display_results(employee_queue, total_bonus):
    """
    Display the results of the employee data processing.

    :param employee_queue: A queue of Employee objects.
    :param total_bonus: The total bonus amount.
    """
    print(f"The total number of employees: {employee_queue.size()}")
    print(f"The total bonus amount: {total_bonus:.2f}")
    print("\nDisplays all the objects in the queue:")

    while not employee_queue.is_empty():
        print(employee_queue.dequeue())

def main():
    """
    Main function to read employee data from a file, calculate bonuses, 
    and display the results.
    """
    file_path = '/Users/danieltshibangu/Desktop/dirany.txt'
    employee_queue, total_bonus = read_employees(file_path)
    display_results(employee_queue, total_bonus)

if __name__ == "__main__":
    main()
