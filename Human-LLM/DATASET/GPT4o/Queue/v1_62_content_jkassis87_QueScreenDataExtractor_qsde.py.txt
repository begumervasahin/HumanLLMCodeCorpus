import requests
import json
from flatten_dict import flatten
import gspread
from oauth2client.service_account import ServiceAccountCredentials
from datetime import datetime

def fetch_ticket_data(url, user, password):
    """
    Fetches and flattens the ticket data from the given URL using basic authentication.

    Args:
        url (str): The URL to fetch data from.
        user (str): The username for authentication.
        password (str): The password for authentication.

    Returns:
        list: A list of ticket values.
    """
    response = requests.get(url, auth=(user, password))
    ticket_data = json.loads(response.text)
    flattened_data = flatten(ticket_data)
    return list(flattened_data.values())

def authorize_google_sheets(credentials_file, scope):
    """
    Authorizes and returns a Google Sheets client.

    Args:
        credentials_file (str): Path to the JSON credentials file.
        scope (list): List of scopes for Google Sheets API.

    Returns:
        gspread.client.Client: Authorized Google Sheets client.
    """
    creds = ServiceAccountCredentials.from_json_keyfile_name(credentials_file, scope)
    client = gspread.authorize(creds)
    return client

def update_google_sheet(client, ticket_values, template_file):
    """
    Updates the Google Sheets with the given ticket values and template.

    Args:
        client (gspread.client.Client): Authorized Google Sheets client.
        ticket_values (list): List of ticket values.
        template_file (str): Path to the CSV template file.
    """
    now = datetime.now()
    daynow = now.day
    yearmonth = f"{now.year}-{now.month}"

    # Open or create the Google Sheet for the current month
    try:
        sh = client.open(yearmonth)
    except gspread.SpreadsheetNotFound:
        sh = client.create(yearmonth)
        sh.share('gaetano.egisto@hostopia.com.au', perm_type='user', role='writer')

    # Add a new worksheet for the current day
    sh.add_worksheet(title=str(daynow), rows="17", cols="25")

    # Import CSV template to the worksheet
    with open(template_file, 'r') as file:
        csv_template = file.read()
    client.import_csv(sh.id, csv_template)

    # Get the newly created worksheet
    worksheet = sh.worksheet(str(daynow))

    # Update the worksheet with ticket values
    row = 2
    col = 2
    ticket_count = 0
    for value in ticket_values:
        if ticket_count < 11:
            worksheet.update_cell(row, col, value)
            row += 1
            ticket_count += 1
        else:
            row = 2
            col += 1
            worksheet.update_cell(row, col, value)
            ticket_count = 1  # Reset ticket count to 1 because we have just placed the first value in new column
