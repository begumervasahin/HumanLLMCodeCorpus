import os
import signal
from multiprocessing import Process, Lock, Semaphore, Value, Queue, cpu_count
from inspect import signature

class DataQueue:
    def __init__(self, generator, size=16, child_count=6):
        """
        Initialize the DataQueue.
        
        :param generator: The generator object with batchGenerator and batchProcessor methods.
        :param size: Size of the queue (default: 16).
        :param child_count: Number of child processes (default: 6).
        """
        self.size = size
        self.child_count = child_count
        self.status = Value('b', False)
        self.generator = generator

        if not (hasattr(self.generator, 'batchGenerator') and hasattr(self.generator, 'batchProcessor')):
            raise Exception('batchGenerator and batchProcessor in Generator object are mandatory')
    
    def load_batch(self, status, input_sm, batch_sm):
        """
        Load batches of data processed by the generator.
        
        :param status: Shared status value to indicate running state.
        :param input_sm: Shared memory structure for input data.
        :param batch_sm: Shared memory structure for processed batches.
        """
        while status.value:
            input_sm['empty_semaphore'].acquire()
            input_sm['mutex'].acquire()
            data = input_sm['q'].get()
            input_sm['mutex'].release()
            input_sm['full_semaphore'].release()

            sig = signature(self.generator.batchProcessor)
            if len(sig.parameters) == 1:
                data = self.generator.batchProcessor(list(data))
            else:
                data = self.generator.batchProcessor(*data)

            batch_sm['full_semaphore'].acquire()
            batch_sm['mutex'].acquire()
            batch_sm['q'].put(data)
            batch_sm['mutex'].release()
            batch_sm['empty_semaphore'].release()
    
    def get_batch(self):
        """
        Retrieve a batch from the batch queue.
        
        :return: A batch of data if available.
        """
        if self.status.value and self.batch_sm['q'].qsize() != 0 and self.input_sm['q'].qsize() != 0:
            self.batch_sm['empty_semaphore'].acquire()
            self.batch_sm['mutex'].acquire()
            batch = self.batch_sm['q'].get()
            self.batch_sm['mutex'].release()
            self.batch_sm['full_semaphore'].release()
            return batch
    
    def monitor(self, status, input_sm):
        """
        Monitor the input queue and generate batches using the generator.
        
        :param status: Shared status value to indicate running state.
        :param input_sm: Shared memory structure for input data.
        """
        while status.value:
            data = tuple(self.generator.batchGenerator())
            input_sm['full_semaphore'].acquire()
            input_sm['mutex'].acquire()
            input_sm['q'].put(data)
            input_sm['mutex'].release()
            input_sm['empty_semaphore'].release()
    
    def start(self):
        """
        Start the DataQueue processes for monitoring and loading batches.
        """
        if hasattr(self.generator, 'init'):
            self.generator.init()
        
        self.status.value = True

        self.input_sm = {
            "q": Queue(),
            "empty_semaphore": Semaphore(value=0),
            "full_semaphore": Semaphore(value=self.size),
            "mutex": Lock()
        }

        self.batch_sm = {
            "q": Queue(),
            "empty_semaphore": Semaphore(value=0),
            "full_semaphore": Semaphore(value=self.size),
            "mutex": Lock()
        }

        child_count = self.child_count if self.child_count != -1 else cpu_count()
        self.child_processes = []

        self.monitor_process = Process(target=self.monitor, args=(self.status, self.input_sm))
        self.monitor_process.start()

        for _ in range(child_count):
            p = Process(target=self.load_batch, args=(self.status, self.input_sm, self.batch_sm))
            p.start()
            self.child_processes.append(p)
    
    def stop(self):
        """
        Stop the DataQueue processes.
        """
        self.status.value = False

        for c_process in self.child_processes:
            if c_process.pid != 0:
                try:
                    os.kill(c_process.pid, signal.SIGTERM)
                    print(f"Successfully killed child process {c_process.pid}")
                except Exception as e:
                    print(f"Access denied: can't kill child process {c_process.pid}. Error: {e}")

        try:
            os.kill(self.monitor_process.pid, signal.SIGTERM)
            print(f"Successfully killed monitor process {self.monitor_process.pid}")
        except Exception as e:
            print(f"Access denied: can't kill monitor process {self.monitor_process.pid}. Error: {e}")

        print(f"Job Queue size at stop: {self.input_sm['q'].qsize()}")
        print(f"Batch Queue size at stop: {self.batch_sm['q'].qsize()}")

        self.input_sm['q'].close()
        self.batch_sm['q'].close()
    
    def get_size(self):
        """
        Get the size of the batch queue.
        
        :return: Size of the batch queue.
        """
        return self.batch_sm['q'].qsize()
    
    def is_empty(self):
        """
        Check if the batch queue is empty.
        
        :return: True if the batch queue is empty, False otherwise.
        """
        return self.batch_sm['q'].qsize() == 0
