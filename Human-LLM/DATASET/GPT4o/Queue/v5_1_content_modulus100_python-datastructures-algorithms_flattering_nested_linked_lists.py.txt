class Node:
    """A class representing a single node in a linked list."""
    def __init__(self, value):
        self.value = value
        self.next = None
    
    def __repr__(self):
        return str(self.value)

class LinkedList:
    """A class representing a singly linked list."""
    def __init__(self, head=None):
        self.head = head
    
    def append(self, value):
        """Append a new node with the given value to the end of the linked list."""
        new_node = Node(value)
        if not self.head:
            self.head = new_node
            return
        current = self.head
        while current.next:
            current = current.next
        current.next = new_node

    def __repr__(self):
        """Return a string representation of the linked list."""
        values = []
        current = self.head
        while current:
            values.append(str(current.value))
            current = current.next
        return " -> ".join(values)

def merge(list1, list2):
    """
    Merge two sorted linked lists into a single sorted linked list.
    
    Args:
        list1 (Node): Head node of the first sorted linked list.
        list2 (Node): Head node of the second sorted linked list.
    
    Returns:
        LinkedList: A new linked list containing all nodes from list1 and list2 in sorted order.
    """
    dummy = Node(0)
    current = dummy

    while list1 and list2:
        if list1.value <= list2.value:
            current.next = list1
            list1 = list1.next
        else:
            current.next = list2
            list2 = list2.next
        current = current.next

    current.next = list1 if list1 else list2

    return LinkedList(dummy.next)

class NestedLinkedList(LinkedList):
    """A class representing a nested linked list, where each node contains a linked list."""
    def flatten(self):
        """Flatten the nested linked list into a single sorted linked list."""
        return self._flatten(self.head)
    
    def _flatten(self, node):
        if not node.next:
            return node.value
        return merge(node.value, self._flatten(node.next))

# Example usage:
if __name__ == '__main__':
    linked_list1 = LinkedList(Node(1))
    linked_list1.append(3)
    linked_list1.append(5)

    linked_list2 = LinkedList(Node(2))
    linked_list2.append(4)

    merged_list = merge(linked_list1.head, linked_list2.head)
    print("Merged Linked List:", merged_list)

    nested_linked_list = NestedLinkedList(Node(linked_list1))
    nested_linked_list.append(linked_list2)

    flattened_list = nested_linked_list.flatten()
    print("Flattened Nested Linked List:", flattened_list)
