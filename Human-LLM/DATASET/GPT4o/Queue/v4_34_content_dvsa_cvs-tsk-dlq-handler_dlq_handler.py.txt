import json
import logging
import base64
import boto3
import datetime
import os
from aws_xray_sdk.core import patch_all, xray_recorder
from boto3_type_annotations.lambda_ import Client as LambdaClient
from pathlib import Path
from typing import List, Dict

# Initialize AWS X-Ray and set logging level
patch_all()
boto3.set_stream_logger('', logging.INFO)

@xray_recorder.capture('invoke')
def handler(event, context):
    """
    Lambda function handler to process SQS messages and send notifications.

    :param event: The event data containing SQS messages.
    :param context: The context in which the function is called.
    """
    lamb: LambdaClient = boto3.client('lambda')
    records: List[Dict] = event.get('Records')
    
    # Extract the source queue and environment from the SQS ARN
    sqs_arn: str = records[0].get('eventSourceARN')
    source_queue = sqs_arn.split(':')[5]
    source_environment = source_queue.rsplit('-', 1)[1]
    
    # Generate the current time in ISO format
    current_time = f"{datetime.datetime.utcnow().isoformat(timespec='minutes')}Z"
    
    # Prepare the payload for the notification Lambda function
    payload = {
        'message_type': 'email',
        'to': os.getenv('TO_EMAIL'),
        'template_id': os.getenv('TEMPLATE_ID'),
        'template_vars': {
            'source_queue': source_queue,
            'source_environment': source_environment
        },
        'attachment': base64.b64encode(json.dumps(records, indent=2).encode('utf-8')).decode("utf-8"),
        'attachment_name': f'{source_queue}_dead_messages_{current_time}.json'
    }
    
    # Invoke the notification Lambda function
    lambda_name = os.getenv('NOTIFY_LAMBDA_NAME')
    resp = lamb.invoke(FunctionName=lambda_name, Payload=json.dumps(payload))
    
    # Check for errors in the Lambda invocation response
    if resp.get('FunctionError') is not None:
        raise RuntimeError(f'{lambda_name} failed to notify with {payload}')
    else:
        return resp['Payload'].read()

if __name__ == '__main__':
    import argparse

    # Set up command-line argument parsing
    parser = argparse.ArgumentParser(description='Invoke the Lambda function with a test event.')
    parser.add_argument('--event', help='Path to event JSON file', type=Path, required=True)
    args = parser.parse_args()
    
    # Read the event data from the specified file
    with open(args.event) as event_file:
        evt = json.load(event_file)
    
    # Invoke the handler with the event data
    print(handler(evt, {}))
