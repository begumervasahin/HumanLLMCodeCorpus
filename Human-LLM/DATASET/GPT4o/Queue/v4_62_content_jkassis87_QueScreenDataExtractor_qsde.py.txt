import requests
import json
from flatten_dict import flatten
import gspread
from oauth2client.service_account import ServiceAccountCredentials
from datetime import datetime
import sys

def fetch_ticket_data(url, username, password):
    """
    Fetch ticket data from the given URL with provided credentials.

    Parameters:
    url (str): The API URL to fetch ticket data.
    username (str): The username for authentication.
    password (str): The password for authentication.

    Returns:
    dict: The flattened ticket data.
    """
    response = requests.get(url, auth=(username, password))
    ticket_data = json.loads(response.text)
    return flatten(ticket_data)

def initialize_google_sheet(credentials_file, sheet_name, template_file):
    """
    Initialize the Google Sheet for the current day and month.

    Parameters:
    credentials_file (str): The path to the JSON credentials file.
    sheet_name (str): The base name of the Google Sheet.
    template_file (str): The path to the CSV template file.

    Returns:
    gspread.Spreadsheet: The initialized Google Sheet.
    """
    # Authorize and initialize the client
    scope = ['https://spreadsheets.google.com/feeds', 'https://www.googleapis.com/auth/drive']
    creds = ServiceAccountCredentials.from_json_keyfile_name(credentials_file, scope)
    client = gspread.authorize(creds)

    now = datetime.now()
    day_now = now.day
    year_month = f"{now.year}-{now.month}"

    # Open or create the monthly sheet
    try:
        sheet = client.open(year_month)
    except gspread.exceptions.SpreadsheetNotFound:
        client.create(year_month)
        sheet = client.open(year_month)
        sheet.share('gaetano.egisto@hostopia.com.au', perm_type='user', role='writer')

    # Add a worksheet for the current day
    sheet.add_worksheet(title=str(day_now), rows="17", cols="25")

    # Import the CSV template into the new worksheet
    with open(template_file, 'r') as file:
        csv_template = file.read()
    client.import_csv(sheet.id, csv_template)

    return sheet

def update_google_sheet(sheet, data):
    """
    Update the Google Sheet with the provided data.

    Parameters:
    sheet (gspread.Spreadsheet): The Google Sheet to update.
    data (list): The data to be inserted into the sheet.
    """
    now = datetime.now()
    day_now = str(now.day)
    worksheet = sheet.worksheet(day_now)

    row = 2
    col = 2
    count = 0

    for item in data:
        if count < 11:
            worksheet.update_cell(row, col, item)
            row += 1
            count += 1
        else:
            row = 2
            col += 1
            worksheet.update_cell(row, col, item)
            count = 1
            row += 1
