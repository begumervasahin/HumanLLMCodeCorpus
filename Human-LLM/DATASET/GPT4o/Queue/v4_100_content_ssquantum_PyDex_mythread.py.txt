from PyQt5.QtCore import QThread, pyqtSlot
from PyQt5.QtWidgets import QApplication

enco = 'mbcs'

def reset_slot(signal, slot, reconnect=True):
    """
    Make sure all instances of slot are disconnected from signal.
    Prevents multiple connections to the same slot.
    If reconnect=True, then reconnect slot to signal.
    
    :param signal: The signal to disconnect from.
    :param slot: The slot to disconnect.
    :param reconnect: Whether to reconnect the slot to the signal.
    """
    while True:
        try:
            signal.disconnect(slot)
        except TypeError:
            break
    if reconnect:
        signal.connect(slot)

class PyDexThread(QThread):
    """
    A template thread that continuously iterates an action on a FIFO queue of items.
    """
    stop = False
    queue = []

    def __init__(self):
        """Initialize the thread and the QApplication instance."""
        super().__init__()
        self.app = QApplication.instance()

    @pyqtSlot(object)
    def add_item(self, new_item, *args, **kwargs):
        """
        Append a new item to the queue for processing.
        
        :param new_item: The item to add to the queue.
        """
        self.queue.append(new_item)

    def process(self, item, *args, **kwargs):
        """
        Process an item in the queue.
        This method should be overridden in subclasses.
        
        :param item: The item to process.
        """
        raise NotImplementedError

    def run(self, *args, **kwargs):
        """
        Run the thread continuously processing items from the queue until the stop bool is toggled.
        """
        while True:
            self.app.processEvents()
            if self.check_stop():
                break
            elif self.queue:
                self.process(self.queue.pop(0), *args, **kwargs)

    def check_stop(self):
        """
        Check the value of stop - must be a function in order to work in a while loop.
        
        :return: The value of the stop flag.
        """
        return self.stop

    def reset_stop(self):
        """
        Reset the stop toggle so that the event loop can run.
        """
        self.stop = False

    def close(self):
        """
        Stop the thread safely. Once the thread has stopped,
        reset the stop toggle so that it doesn't block the thread
        from starting again the next time.
        """
        reset_slot(self.finished, self.reset_stop)
        self.stop = True
