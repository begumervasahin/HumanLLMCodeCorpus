import logging
from collections import deque
from queue import Queue
import slack
from client_helpers import fetchImageUrlForUserId

logging.getLogger().setLevel(logging.INFO)

class QueueBot:
    LINE_BLOCK = {
        "type": "section",
        "text": {
            "type": "mrkdwn",
            "text": (
                "Welcome to queueBot! :wave: \n\n"
                "*Here are the people in the line:*"
            ),
        },
        "accessory": {
            "type": "button",
            "text": {
                "type": "plain_text",
                "text": "Line Up"
            },
            "action_id": "action_id",
            "value": "view_alternate_2"
        }
    }

    def __init__(self, channel, name):
        """
        Initialize the QueueBot.
        
        :param channel: Slack channel ID where the bot operates.
        :param name: Name of the bot.
        """
        self.channel = channel
        self.name = name
        self.username = "queueBot"
        self.icon_emoji = ":robot_face:"
        self.timestamp = ""
        self.queue = deque()

    def get_message_payload(self):
        """
        Get the message payload to be sent to Slack.
        
        :return: A dictionary representing the message payload.
        """
        return {
            "ts": self.timestamp,
            "channel": self.channel,
            "username": self.username,
            "icon_emoji": self.icon_emoji,
            "blocks": [
                self.LINE_BLOCK,
                self._get_queue_block()
            ],
        }

    def _get_queue_block(self):
        """
        Get the block representing the current queue.
        
        :return: A dictionary representing the queue block.
        """
        elements = list(map(user_to_block_kit_element, self.queue))

        if not elements:
            return {
                "type": "section",
                "text": {
                    "type": "mrkdwn",
                    "text": 'There is no one in this queue yet! Click the "Line Up" button to be the first in the line!'
                }
            }

        elements.insert(0, {
            "type": "mrkdwn",
            "text": ":one: : "
        })

        return {
            "type": "context",
            "elements": elements
        }

    def insert_queue(self, user):
        """
        Insert a user into the queue.
        
        :param user: A dictionary representing the user.
        """
        if user in self.queue:
            logging.info('Error: User is already in the queue.')
            return
        self.queue.append(user)

def user_to_block_kit_element(user):
    """
    Convert a user to a Block Kit element for Slack.
    
    :param user: A dictionary representing the user.
    :return: A dictionary representing the Block Kit element.
    """
    image_url = fetchImageUrlForUserId(user["id"])
    return {
        "type": "image",
        "image_url": image_url,
        "alt_text": user["name"]
    }
