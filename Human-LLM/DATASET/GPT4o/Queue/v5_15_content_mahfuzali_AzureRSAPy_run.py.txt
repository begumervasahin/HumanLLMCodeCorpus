def create_database(client, database_id):
    """
    Create a database in Azure Cosmos DB.

    Parameters:
    client (CosmosClient): The Cosmos DB client.
    database_id (str): The ID of the database to create.
    """
    try:
        client.CreateDatabase({"id": database_id})
        print(f'Database with id \'{database_id}\' created')
    except errors.HTTPFailure as e:
        if e.status_code == 409:
            print(f'Database with id \'{database_id}\' already exists')
        else:
            raise

def create_collection(client, database_link, collection_id):
    """
    Create a collection in Azure Cosmos DB.

    Parameters:
    client (CosmosClient): The Cosmos DB client.
    database_link (str): The link to the database.
    collection_id (str): The ID of the collection to create.
    """
    try:
        client.CreateContainer(database_link, {"id": collection_id})
        print(f'Collection with id \'{collection_id}\' created')
    except errors.HTTPFailure as e:
        if e.status_code == 409:
            print(f'Collection with id \'{collection_id}\' already exists')
        else:
            raise
