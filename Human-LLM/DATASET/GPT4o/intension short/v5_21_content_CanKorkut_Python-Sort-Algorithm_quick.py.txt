from random import randint

def generate_random_array(size, upper_limit=10000):
    """Generates a list of random integers of a given size and within a given range."""
    return [randint(0, upper_limit) for _ in range(size)]

class SortAlgorithm:
    """Implements the QuickSort algorithm with step counting for performance tracking."""
    
    def __init__(self):
        self.quick_sort_steps = 0

    def quick_sort(self, array, low, high):
        """Recursively applies the QuickSort algorithm to sort the array."""
        if low < high:
            pivot_index = self.partition(array, low, high)
            self.quick_sort(array, low, pivot_index - 1)
            self.quick_sort(array, pivot_index + 1, high)
        self.quick_sort_steps += 1

    def partition(self, array, low, high):
        """Partitions the array around a pivot, sorting elements relative to the pivot."""
        pivot = array[high]
        i = low - 1

        for j in range(low, high):
            if array[j] <= pivot:
                i += 1
                array[i], array[j] = array[j], array[i]
            self.quick_sort_steps += 1

        array[i + 1], array[high] = array[high], array[i + 1]
        return i + 1

def run_quick_sort(array):
    """Runs the QuickSort algorithm on the provided array and prints the results."""
    print("Original array:", array)
    sorter = SortAlgorithm()
    sorter.quick_sort(array, 0, len(array) - 1)
    print("Sorted array:", array)
    print("QuickSort steps:", sorter.quick_sort_steps)
