import time

# Initialize global count variable to track the number of comparisons
count = 0

def mergeSort(arr):
    """Recursively divides and sorts the array while counting comparisons."""
    global count
    if len(arr) > 1:
        mid = len(arr) // 2  # Finding the mid of the array
        L = arr[:mid]  # Dividing the array elements into 2 halves
        R = arr[mid:]

        mergeSort(L)  # Sorting the first half
        mergeSort(R)  # Sorting the second half

        i = j = k = 0

        # Merge the sorted halves back together
        while i < len(L) and j < len(R):
            count += 1  # Increment comparison count
            if L[i] < R[j]:
                arr[k] = L[i]
                i += 1
            else:
                arr[k] = R[j]
                j += 1
            k += 1

        # Check if any element was left in the left half
        while i < len(L):
            count += 1  # Increment comparison count
            arr[k] = L[i]
            i += 1
            k += 1

        # Check if any element was left in the right half
        while j < len(R):
            count += 1  # Increment comparison count
            arr[k] = R[j]
            j += 1
            k += 1

def printList(arr):
    """Prints the elements of the array."""
    for i in range(len(arr)):
        print(arr[i], end=" ")
    print()

if __name__ == '__main__':
    start = time.time()  # Start the timer

    # Open and read the input file "case"
    with open("case") as fp:
        arr = fp.read().split()

    # Convert the read strings to integers
    list2 = [int(num) for num in arr]

    # Perform merge sort on the list
    mergeSort(list2)

    # Print the sorted array
    print("Sorted array is: ", end="\n")
    printList(list2)

    # Print the total number of comparisons
    print("Number of comparisons:", count)

    end = time.time()  # Stop the timer

    # Print the time taken to execute the sorting
    print("Time taken:", end - start)
