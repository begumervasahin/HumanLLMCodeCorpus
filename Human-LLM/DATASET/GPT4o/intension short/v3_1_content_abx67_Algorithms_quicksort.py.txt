def partition(array, left, right):
    """
    Partitions the array around a pivot element, which is chosen as the last element.
    Elements smaller than or equal to the pivot are moved to the left of the pivot,
    while elements greater than the pivot are moved to the right.
    
    Args:
        array (list): The list of elements to be partitioned.
        left (int): The starting index of the subarray.
        right (int): The ending index of the subarray (pivot element).
    
    Returns:
        int: The final index of the pivot element after partitioning.
    """
    pivot = array[right]
    print(f"Pivot selected: {pivot}")

    split_index = left

    for current_index in range(left, right + 1):
        print(f"Inspecting element at index {current_index}: {array[current_index]}")
        if array[current_index] <= pivot:
            if current_index != split_index:
                # Swap the current element with the element at the split_index
                array[current_index], array[split_index] = array[split_index], array[current_index]
            split_index += 1
    
    return split_index - 1

def quicksort(array, left, right):
    """
    Applies the quicksort algorithm to sort the array.
    
    Args:
        array (list): The list of elements to be sorted.
        left (int): The starting index of the subarray to be sorted.
        right (int): The ending index of the subarray to be sorted.
    """
    if left < right:
        split_index = partition(array, left, right)
        print(f"Array after partitioning at index {split_index}: {array}")

        # Recursively apply quicksort to the subarrays
        quicksort(array, left, split_index - 1)
        quicksort(array, split_index + 1, right)

if __name__ == "__main__":

    array = [3, 6, 8, 10, 1, 2, 1]
    print("Original array:", array)
    quicksort(array, 0, len(array) - 1)
