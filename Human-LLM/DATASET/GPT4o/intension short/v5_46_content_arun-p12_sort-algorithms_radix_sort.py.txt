def radix_sort(arr, verbose=0, desc=0):
    def get_digit_count(n, base=10):
        """Return the number of digits in a number for the given base."""
        count = 0
        while n > 0:
            count += 1
            n //= base
        return count

    def get_digit_at_position(n, position, base=10):
        """Return the digit at the specified position (1-indexed) in the given base."""
        return (n // (base ** (position - 1))) % base

    def bucket_sort(arr, position, base=10):
        """Sort the array based on the digit at the given position using bucket sort."""
        buckets = [[] for _ in range(base)]
        for num in arr:
            digit = get_digit_at_position(num, position, base)
            buckets[digit].append(num)
            if verbose == 2:
                print(f"Digit {digit} at position {position}: {buckets}")
        return [num for bucket in buckets for num in bucket]

    base = 10
    offset = -min(arr) if min(arr) < 0 else 0
    
    # Normalize the array if it contains negative numbers
    arr = [x + offset for x in arr]

    max_value = max(arr)
    num_passes = get_digit_count(max_value, base)
    
    # Perform sorting based on each digit position
    for position in range(1, num_passes + 1):
        arr = bucket_sort(arr, position, base)
        if verbose:
            print(f"After pass {position}: {arr}")
    
    # Adjust back if normalization was applied
    arr = [x - offset for x in arr]
    
    # Reverse if descending order is requested
    if desc:
        arr.reverse()

    return arr
