import sys
from tkinter import *
import tkSimpleDialog

def busqueda():
    # Get the number of elements to input
    num_elements = int(entry_num_elements.get())
    numbers = []

    # Ask for the number to search for
    search_number = tkSimpleDialog.askinteger("", "Ingresa el número a buscar")

    # Gather the numbers from the user
    for i in range(num_elements):
        number = tkSimpleDialog.askinteger("", "Ingresa los números en orden")
        numbers.append(number)

    # Display the list of numbers
    Label2.config(text=str(numbers))

    # Perform the sequential search
    search_result = secuencial(numbers, search_number)
    label3.config(text=search_result)

def secuencial(numbers, search_number):
    for i, number in enumerate(numbers):
        if number == search_number:
            return f"Encontrado en la posición: {i}"
    return "Número no encontrado"

# Set up the main application window
app = Tk()
app.title("Radix Sort")
app.geometry('250x200')
app.configure(bg='SkyBlue2')

# Label and entry to ask for the number of elements
Label1 = Label(app, text="¿Cuántos números vas a ingresar?", font='Helvetica 12', bg='SkyBlue2')
Label1.pack()

entry_num_elements = Entry(app, width=8)
entry_num_elements.pack()

# Button to trigger the search
button1 = Button(app, text="Ok!", command=busqueda)
button1.pack(pady=(10, 0))

# Labels to display results
Label2 = Label(app, text="", bg='SkyBlue2')
Label2.pack(pady=(10, 0))

label3 = Label(app, text="", bg='SkyBlue2')
label3.pack(pady=(10, 0))

# Start the main loop
app.mainloop()
