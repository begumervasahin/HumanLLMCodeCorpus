import timeit
import random

# Function to copy a section of one array to another
def copy_array(src, src_start, dest, dest_start, length):
    for i in range(length):
        dest[dest_start + i] = src[src_start + i]

# Recursive function that sorts an array by splitting and merging
def merge_sort(array):
    # Base case: If the array has one or zero elements, it's already sorted
    if len(array) <= 1:
        return array
    
    # Find the middle index to split the array
    mid = len(array) // 2
    left_half = array[:mid]
    right_half = array[mid:]
    
    # Recursively sort both halves
    merge_sort(left_half)
    merge_sort(right_half)
    
    # Merge the sorted halves back into the original array
    merge(left_half, right_half, array)
    
    return array

# Function to merge two sorted arrays into a single sorted array
def merge(left, right, merged):
    i = j = k = 0
    
    # Compare elements and merge them into the merged array
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            merged[k] = left[i]
            i += 1
        else:
            merged[k] = right[j]
            j += 1
        k += 1
    
    # Copy any remaining elements from the left and right arrays
    copy_array(left, i, merged, k, len(left) - i)
    copy_array(right, j, merged, k, len(right) - j)

# Generate a list of random numbers, including an initial predefined list
def generate_random_numbers(size=5000, upper_limit=99999):
    initial_numbers = [7, 4, 3, 8, 1, 12, 9, 14]
    random_numbers = [random.randint(1, upper_limit) for _ in range(size)]
    return initial_numbers + random_numbers

# Main function to execute the sort and measure the time taken
def main():
    numbers = generate_random_numbers()
    
    start_time = timeit.default_timer()
    sorted_numbers = merge_sort(numbers)
    elapsed_time = timeit.default_timer() - start_time
    
    print(f"Merge Sort Time: {elapsed_time:.6f} seconds")
    print(sorted_numbers)

# Run the main function
if __name__ == "__main__":
    main()
