import matplotlib.pyplot as plt
import matplotlib.animation as animation
import random

# Function to animate the sorting process
def plot_sort_animation(sort_steps):
    fig, ax = plt.subplots()

    # Retrieve the initial state of the list and the index being highlighted
    values, highlight_index = next(sort_steps)
    num_elements = len(values)
    max_frames = num_elements ** 2  # Estimate for the number of animation frames

    # Function to update the plot for each frame
    def update(frame):
        try:
            values, highlight_index = next(sort_steps)
            ax.clear()
            bars = ax.bar(range(num_elements), values, color='blue')
            bars[highlight_index].set_color('red')
            print(f'Iteration: {frame}')
        except StopIteration:
            print("Sorting complete.")
            animation.event_source.stop()

    # Create and start the animation
    anim = animation.FuncAnimation(fig, update, frames=max_frames, repeat=False)
    plt.show()

# Generator function to perform insertion sort, yielding the list at each step
def insertion_sort_generator(lst):
    n = len(lst)
    for i in range(1, n):
        j = i
        while j > 0 and lst[j] < lst[j - 1]:
            lst[j], lst[j - 1] = lst[j - 1], lst[j]
            yield lst, j
            j -= 1
        yield lst, i

    # Final yield to indicate sorting is complete
    yield lst, -1

# Main execution
if __name__ == "__main__":
    num_elements = 25
    random_list = [random.randint(1, 200) for _ in range(num_elements)]

    # Start the sorting animation
    plot_sort_animation(insertion_sort_generator(random_list))
