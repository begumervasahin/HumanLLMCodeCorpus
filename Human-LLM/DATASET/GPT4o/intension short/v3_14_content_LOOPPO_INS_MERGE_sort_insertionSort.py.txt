def insertion_sort(array):
    """
    Sorts the input array in ascending order using the insertion sort algorithm.

    Parameters:
    array (list): A list of elements to be sorted.

    Returns:
    None: The input array is sorted in place.
    """

    # Traverse the array starting from the second element
    for i in range(1, len(array)):
        current_value = array[i]
        position = i - 1

        # Shift elements of the sorted portion to the right
        # to make space for the current_value
        while position >= 0 and array[position] > current_value:
            array[position + 1] = array[position]
            position -= 1

        # Insert the current_value at its correct position
        array[position + 1] = current_value

# Example usage
if __name__ == "__main__":
    data = [12, 11, 13, 5, 6]
    print("Original array:", data)
    insertion_sort(data)
    print("Sorted array:", data)
