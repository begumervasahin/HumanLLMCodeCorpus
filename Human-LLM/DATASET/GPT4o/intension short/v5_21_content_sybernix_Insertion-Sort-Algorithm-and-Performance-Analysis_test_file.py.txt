from is_140422H import InsertionSort

def test_insertion_sort(number_list):
    """
    Tests the InsertionSort function by checking if the provided list is sorted correctly.
    
    Parameters:
    number_list (list): The list of numbers to be sorted and tested.
    """
    print(f'Unsorted list: {number_list}')
    
    sorted_list = InsertionSort(number_list)
    print(f'Sorted list: {sorted_list}')
    
    # Check if the list is sorted
    is_sorted = all(sorted_list[i] <= sorted_list[i + 1] for i in range(len(sorted_list) - 1))
    
    # Verify that the sorted list contains the same elements as the original list
    if set(number_list) != set(sorted_list):
        is_sorted = False
    
    if is_sorted:
        print("The list is correctly sorted.")
    else:
        print("The list is not sorted correctly.")
    
    print()  # Add a blank line for readability

def main():
    """
    Main function to run the insertion sort tests on multiple lists.
    """
    test_cases = [
        [20, 12, 8, 5, 7, 10, 14],  # A list with positive numbers
        [20, -7, 10, 14],            # A list with a mix of positive and negative numbers
        [],                          # An empty list
        [0, 0, 0]                    # A list with identical elements
    ]
    
    for case in test_cases:
        test_insertion_sort(case)

if __name__ == "__main__":
    main()
