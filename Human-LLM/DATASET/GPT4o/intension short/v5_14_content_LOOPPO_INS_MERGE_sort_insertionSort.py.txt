def insertion_sort(arr):
    """Sorts a list in ascending order using the insertion sort algorithm."""
    
    # Traverse the list starting from the second element
    for i in range(1, len(arr)):
        current_value = arr[i]  # Element to be positioned correctly
        position = i - 1  # Initialize the position for comparison

        # Shift elements of the sorted segment to the right to create space for current_value
        while position >= 0 and arr[position] > current_value:
            arr[position + 1] = arr[position]
            position -= 1
        
        # Place the current value in its correct position within the sorted segment
        arr[position + 1] = current_value
