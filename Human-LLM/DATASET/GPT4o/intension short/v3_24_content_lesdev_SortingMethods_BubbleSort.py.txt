import tkinter as tk
from tkinter import simpledialog

def bubble_sort():
    """Performs bubble sort on a list of numbers provided by the user and displays the sorted result."""
    try:
        # Get the number of elements to sort
        num_elements = int(entry_numeros.get())
        numbers = collect_numbers(num_elements)

        if not numbers:
            label_unsorted.config(text="No numbers entered.")
            return

        display_list(label_unsorted, "Unsorted", numbers)
        
        sorted_numbers = perform_bubble_sort(numbers)
        display_list(label_sorted, "Sorted", sorted_numbers)
    
    except ValueError:
        label_unsorted.config(text="Invalid input. Please enter a valid number.")
        label_sorted.config(text="")

def collect_numbers(num_elements):
    """Collects a specified number of integers from the user."""
    numbers = []
    for i in range(num_elements):
        number = simpledialog.askinteger("Input", f"Enter number {i+1}:")
        if number is not None:
            numbers.append(number)
        else:
            break
    return numbers

def perform_bubble_sort(numbers):
    """Sorts a list of numbers using the bubble sort algorithm."""
    n = len(numbers)
    for i in range(n - 1):
        for j in range(n - i - 1):
            if numbers[j] > numbers[j + 1]:
                numbers[j], numbers[j + 1] = numbers[j + 1], numbers[j]
    return numbers

def display_list(label, title, lst):
    """Displays a list in the specified label widget."""
    text = f"{title}: {lst}"
    label.config(text=text)

# Set up the main application window
app = tk.Tk()
app.title("Bubble Sort")
app.geometry('300x200')
app.configure(bg='SkyBlue2')

# Label and entry for the number of elements
label_prompt = tk.Label(app, text="How many numbers do you want to enter?", font="Helvetica 12", bg='SkyBlue2')
label_prompt.pack(pady=(10, 5))

entry_numeros = tk.Entry(app, width=8)
entry_numeros.pack()

# Button to trigger the sorting process
button_ok = tk.Button(app, text="Ok!", command=bubble_sort)
button_ok.pack(pady=(10, 0))

# Labels to display the unsorted and sorted lists
label_unsorted = tk.Label(app, text="", bg='SkyBlue2')
label_unsorted.pack(pady=(10, 0))

label_sorted = tk.Label(app, text="", bg='SkyBlue2')
label_sorted.pack(pady=(10, 0))

# Start the application's event loop
app.mainloop()
