def quick_sort(array, verbose=0, desc=0):
    def partition(low, high):
        pivot = array[low]
        left, right = low + 1, high

        while True:
            while left <= right and array[left] <= pivot:
                left += 1
            while left <= right and array[right] > pivot:
                right -= 1
            if left <= right:
                array[left], array[right] = array[right], array[left]
                if verbose == 2:
                    print(f"  sub: pivot={pivot}, left={left}, right={right} :: {array}")
            else:
                break

        array[low], array[right] = array[right], array[low]
        return right

    def quicksort_recursive(low, high):
        if low < high:
            pivot_index = partition(low, high)
            if verbose:
                print(f"iter: pivot_index={pivot_index}, low={low}, high={high} :: {array}")
            quicksort_recursive(low, pivot_index - 1)
            quicksort_recursive(pivot_index + 1, high)

    quicksort_recursive(0, len(array) - 1)
    
    if desc:
        array.reverse()
    
    return array
