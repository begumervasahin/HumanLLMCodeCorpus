import statistics

def find_pivot_position(arr, first, last):
    # Identify the first, middle, and last elements
    low = arr[first]
    mid = arr[(first + last) // 2]
    high = arr[last]
    
    # Determine the median of these elements to be the pivot
    pivot_value = statistics.median([low, mid, high])
    
    # Find and move the pivot to the last position
    if pivot_value == low:
        pivot_index = first
    elif pivot_value == high:
        pivot_index = last
    else:
        pivot_index = (first + last) // 2
    
    arr[pivot_index], arr[last] = arr[last], arr[pivot_index]
    
    # Partitioning step
    left = first
    right = last - 1
    
    while True:
        # Move the left pointer until an element larger than the pivot is found
        while left <= right and arr[left] <= pivot_value:
            left += 1
        # Move the right pointer until an element smaller than the pivot is found
        while left <= right and arr[right] >= pivot_value:
            right -= 1
        # If the pointers cross, break the loop
        if left > right:
            break
        # Swap the elements at the left and right pointers
        arr[left], arr[right] = arr[right], arr[left]
    
    # Swap the pivot to its correct position
    arr[last], arr[right] = arr[right], arr[last]
    return right

def quicksort(arr, first, last):
    if first < last:
        pivot_index = find_pivot_position(arr, first, last)
        quicksort(arr, first, pivot_index - 1)
        quicksort(arr, pivot_index + 1, last)
