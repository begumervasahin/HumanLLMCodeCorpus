import time
from tqdm import tqdm

def simulate_work(total_iterations, delay):
    """
    Simulate a task with a progress bar, introducing a delay in each iteration.

    Parameters:
        total_iterations (int): The total number of iterations to simulate.
        delay (float): The delay in seconds for each iteration.
    """
    for _ in tqdm(range(total_iterations), desc="Processing", unit="iteration"):
        time.sleep(delay)

if __name__ == "__main__":
    # Configuration: number of iterations and delay per iteration
    total_iterations = 1000
    delay_per_iteration = 0.02

    # Run the simulation
    simulate_work(total_iterations, delay_per_iteration)
