def shell_sort(A, verbose=0, desc=0):
    def insertion_sort_with_gap(array, gap):
        """Perform insertion sort on elements that are gap distance apart."""
        for i in range(gap, len(array)):
            current_value = array[i]
            position = i

            # Shift elements in the sorted subarray until the correct position is found
            while position >= gap and array[position - gap] > current_value:
                array[position] = array[position - gap]
                position -= gap

            array[position] = current_value

    # Start with a large gap, then reduce the gap size
    gap = len(A) // 2

    while gap > 0:
        if verbose:
            print(f"Current gap: {gap}")

        # Perform insertion sort with the current gap
        insertion_sort_with_gap(A, gap)

        # Reduce the gap size for the next iteration
        gap //= 2

    # If descending order is requested, reverse the sorted list
    if desc:
        A.reverse()

    return A

# Example usage:
if __name__ == "__main__":
    sample_array = [19, 2, 31, 45, 6, 11, 121, 27]
    sorted_array = shell_sort(sample_array, verbose=1)
    print("Sorted array:", sorted_array)
