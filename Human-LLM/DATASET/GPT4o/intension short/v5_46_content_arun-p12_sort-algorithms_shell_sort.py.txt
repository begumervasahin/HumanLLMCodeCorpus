def shell_sort(arr, verbose=0, desc=False):
    """
    Sorts an array using the Shell Sort algorithm.
    
    Parameters:
    arr (list): The list of elements to be sorted.
    verbose (int): The verbosity level (0: no output, 1: gap info, 2: detailed subarray info).
    desc (bool): If True, sorts the array in descending order.

    Returns:
    list: The sorted array.
    """
    def insertion_sort_with_gap(arr, gap):
        """
        Performs an insertion sort on elements with a specified gap.

        Parameters:
        arr (list): The list of elements to be sorted.
        gap (int): The gap between elements to compare.
        """
        for i in range(gap, len(arr)):
            current_value = arr[i]
            j = i
            while j >= gap and arr[j - gap] > current_value:
                arr[j] = arr[j - gap]
                j -= gap
            arr[j] = current_value
            if verbose == 2:
                print(f"  Current state after gap {gap}: {arr}")

    gap = len(arr) // 2
    while gap > 0:
        if verbose >= 1:
            print(f"Gap: {gap}")
        insertion_sort_with_gap(arr, gap)
        gap //= 2

    if desc:
        arr.reverse()

    return arr
