import time
import random

def merge_sort(arr):
    """Recursively sort an array using the merge sort algorithm."""
    if len(arr) <= 1:
        return

    # Split the array into two halves
    mid = len(arr) // 2
    left_half = arr[:mid]
    right_half = arr[mid:]

    # Recursively sort both halves
    merge_sort(left_half)
    merge_sort(right_half)

    # Merge the sorted halves back together
    merge(arr, left_half, right_half)

def merge(arr, left_half, right_half):
    """Merge two sorted halves into a single sorted array."""
    i = j = k = 0

    # Merge the halves by comparing elements
    while i < len(left_half) and j < len(right_half):
        if left_half[i] < right_half[j]:
            arr[k] = left_half[i]
            i += 1
        else:
            arr[k] = right_half[j]
            j += 1
        k += 1

    # Copy any remaining elements from the left half
    while i < len(left_half):
        arr[k] = left_half[i]
        i += 1
        k += 1

    # Copy any remaining elements from the right half
    while j < len(right_half):
        arr[k] = right_half[j]
        j += 1
        k += 1

def insertion_sort(arr):
    """Sort an array in place using the insertion sort algorithm."""
    for i in range(1, len(arr)):
        current_value = arr[i]
        pos = i

        # Shift elements to the right to create the correct position for current_value
        while pos > 0 and arr[pos - 1] > current_value:
            arr[pos] = arr[pos - 1]
            pos -= 1

        # Insert the current value into its correct position
        arr[pos] = current_value

def run_sorting_tests(array_sizes):
    """Run timing tests on merge sort and insertion sort for various array sizes."""
    for size in array_sizes:
        arr = [random.randint(1, 1000) for _ in range(size)]

        # Measure the time taken by merge sort
        merge_time = measure_sort_time(merge_sort, arr)

        # Measure the time taken by insertion sort
        insertion_time = measure_sort_time(insertion_sort, arr)

        # Display the results
        display_results(size, merge_time, insertion_time)

def measure_sort_time(sort_function, arr):
    """Measure the time taken to sort an array using the specified sort function."""
    start_time = time.time()
    sort_function(arr.copy())
    return time.time() - start_time

def display_results(size, merge_time, insertion_time):
    """Display the sorting time results for a given array size."""
    print(f"Array Size: {size}")
    print(f"Merge Sort Time: {merge_time:.6f} seconds")
    print(f"Insertion Sort Time: {insertion_time:.6f} seconds")
    print("-" * 40)

# Define the array sizes to test
array_sizes = [2000, 8000, 32000, 128000, 512000, 1024000, 4096000]

# Run the tests
run_sorting_tests(array_sizes)
