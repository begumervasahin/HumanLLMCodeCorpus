def cipher_sub(text, shift_letter):
    """
    Encrypts the input text using a simple substitution cipher based on the shift_letter.
    """
    alphabet = list("abcdefghijklmnopqrstuvwxyz")
    shift_index = alphabet.index(shift_letter)

    # Create the substitution alphabet
    shifted_alphabet = alphabet[shift_index:] + alphabet[:shift_index]

    # Encode the message
    encoded_message = [
        shifted_alphabet[alphabet.index(char)] if char.isalpha() else char
        for char in text
    ]

    encoded_message_str = ''.join(encoded_message)

    print("\nOriginal Text:", text)
    print("\nShift Index:", shift_index)
    print("Substitution Alphabet: <", ' '.join(shifted_alphabet), ">")

    return encoded_message_str


def decode_sub(text, shift_letter):
    """
    Decrypts the input text using a simple substitution cipher based on the shift_letter.
    """
    alphabet = list("abcdefghijklmnopqrstuvwxyz")
    shift_index = alphabet.index(shift_letter)

    # Create the substitution alphabet
    shifted_alphabet = alphabet[shift_index:] + alphabet[:shift_index]

    # Decode the message
    decoded_message = [
        alphabet[shifted_alphabet.index(char)] if char.isalpha() else char
        for char in text
    ]

    decoded_message_str = ''.join(decoded_message)

    print("\nEncoded Text:", text)
    print("\nShift Index:", shift_index)
    print("Substitution Alphabet: <", ' '.join(shifted_alphabet), ">")

    return decoded_message_str


def output_encoded(text):
    """
    Outputs each character of the text separated by a space.
    """
    print(' '.join(text))


def empty(size):
    """
    Returns a list of zeros with the specified size.
    """
    return [0] * size


def alpha_index(letter):
    """
    Returns the index of the given letter in the alphabet.
    If the input is not a single letter, it prints an error message.
    """
    if len(letter) != 1 or not letter.isalpha():
        print("That's not a letter!")
    else:
        alphabet = list("abcdefghijklmnopqrstuvwxyz")
        return alphabet.index(letter)
