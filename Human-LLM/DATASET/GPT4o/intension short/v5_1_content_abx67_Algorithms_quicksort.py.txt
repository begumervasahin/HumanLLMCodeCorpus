def partition(arr, left, right):
    """Partitions the array around a pivot and returns the split index."""
    pivot = arr[right]  # Choose the last element as the pivot
    split = left  # This will track where to split the array

    print(f"Pivot: {pivot}")

    for i in range(left, right + 1):
        print(f"Index: {i}")

        # If the current element is less than or equal to the pivot
        if arr[i] <= pivot:
            # Swap the element at i with the element at the split index
            arr[i], arr[split] = arr[split], arr[i]
            split += 1
    
    # Return the index of the pivot after partitioning
    return split - 1

def quicksort(arr, left, right):
    """Recursively sorts the array using the quicksort algorithm."""
    if left < right:
        # Partition the array and get the pivot index
        split = partition(arr, left, right)
        
        print(f"Split index: {split}, Left: {left}, Right: {right}")
        print(f"Array after partition: {arr}")
        
        # Recursively sort the left and right subarrays
        quicksort(arr, left, split - 1)
        quicksort(arr, split + 1, right)

    arr = [3, 6, 8, 10, 1, 2, 1]
    quicksort(arr, 0, len(arr) - 1)
