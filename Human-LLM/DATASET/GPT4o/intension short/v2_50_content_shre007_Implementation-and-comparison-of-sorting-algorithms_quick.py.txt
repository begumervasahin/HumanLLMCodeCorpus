import statistics

def find_pivot_position(arr, first, last):
    # Select the first, middle, and last elements for the pivot calculation
    low, mid, high = arr[first], arr[(first + last) // 2], arr[last]
    
    # Determine the median value among the selected elements
    pivot_value = statistics.median([low, mid, high])
    
    # Find the index of the pivot value
    if pivot_value == low:
        pivot_index = first
    elif pivot_value == high:
        pivot_index = last
    else:
        pivot_index = (first + last) // 2
    
    # Move the pivot to the end of the array for partitioning
    arr[pivot_index], arr[last] = arr[last], arr[pivot_index]
    pivot = arr[last]
    
    # Initialize pointers for partitioning
    left, right = first, last - 1
    
    while True:
        # Move left pointer until an element larger than the pivot is found
        while left <= right and arr[left] <= pivot:
            left += 1
        # Move right pointer until an element smaller than the pivot is found
        while left <= right and arr[right] >= pivot:
            right -= 1
        # If pointers have crossed, exit the loop
        if left > right:
            break
        # Swap elements at the left and right pointers
        arr[left], arr[right] = arr[right], arr[left]
    
    # Place the pivot in its final position
    arr[first], arr[right] = arr[right], arr[first]
    return right

def quicksort(arr, first, last):
    # Recursively apply quicksort to sub-arrays
    if first < last:
        pivot_index = find_pivot_position(arr, first, last)
        quicksort(arr, first, pivot_index - 1)
        quicksort(arr, pivot_index + 1, last)
