import random

def quicksort(lst):
    # Base case: if the list is empty or contains a single element, return it as is
    if len(lst) <= 1:
        return lst
    
    # Choose a random pivot from the list
    pivot_index = random.randint(0, len(lst) - 1)
    pivot = lst[pivot_index]
    
    # Print the current state for debugging
    print(f"Pivoting {pivot} in {lst}")
    
    # Create lists to hold elements less than, greater than, and equal to the pivot
    left = []
    right = []
    pivot_clones = [pivot]
    
    # Partition the list into left, right, and pivot_clones
    for i in range(len(lst)):
        if i == pivot_index:
            continue
        if lst[i] < pivot:
            left.append(lst[i])
        elif lst[i] > pivot:
            right.append(lst[i])
        else:
            pivot_clones.append(lst[i])
    
    # Print the partitions for debugging
    print(f"Left becomes {left}; right becomes {right}")
    
    # Recursively apply quicksort to the partitions and concatenate results
    return quicksort(left) + pivot_clones + quicksort(right)

# Example usage
lst = [3, 6, 8, 10, 1, 2, 1]
print("Result:", quicksort(lst))
