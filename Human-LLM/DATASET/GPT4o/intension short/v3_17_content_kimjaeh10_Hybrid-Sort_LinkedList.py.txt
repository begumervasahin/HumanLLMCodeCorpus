class Node:
    __slots__ = 'value', 'next_node'

    def __init__(self, value, next_node=None):
        """
        Initialize a Node with a value and an optional reference to the next node.
        :param value: The value of the node.
        :param next_node: The next node in the linked list (default is None).
        """
        self.value = value
        self.next_node = next_node

    def __eq__(self, other):
        """
        Compare two nodes for equality based on their values.
        :param other: The node to compare to.
        :return: True if the nodes have the same value, False otherwise.
        """
        return other is not None and self.value == other.value

    def __repr__(self):
        """
        Return a string representation of the node.
        :return: String of the node's value.
        """
        return str(self.value)


def insertion_sort_wrapper(sort_func):
    """
    Wrapper to count the number of times the insertion sort is called.
    :param sort_func: The original insertion sort function.
    :return: The wrapped function.
    """
    def wrapped_sort(self, *args, **kwargs):
        if self.size > 1:
            LinkedList.insertion_count += 1
        return sort_func(self)
    return wrapped_sort


class LinkedList:
    insertion_count = 0  # Static variable to count the number of insertion sort calls.

    def __init__(self, data=None):
        """
        Initialize an empty linked list, optionally with initial data.
        :param data: A list of values to populate the linked list (default is None).
        """
        self.head = None
        self.tail = None
        self.size = 0

        if data:
            for value in data:
                self.push_back(value)

    def __len__(self):
        return self.size

    def __eq__(self, other):
        """
        Compare two linked lists for equality.
        :param other: The linked list to compare to.
        :return: True if the linked lists are identical, False otherwise.
        """
        if self.size != other.size:
            return False

        current_self, current_other = self.head, other.head

        while current_self and current_other:
            if current_self != current_other:
                return False
            current_self = current_self.next_node
            current_other = current_other.next_node

        return current_self is None and current_other is None

    def __repr__(self):
        """
        Return a string representation of the linked list.
        :return: String of the list values.
        """
        values = []
        current_node = self.head

        while current_node:
            values.append(current_node.value)
            current_node = current_node.next_node

        return str(values)

    def is_empty(self):
        """
        Check if the linked list is empty.
        :return: True if the list is empty, False otherwise.
        """
        return self.size == 0

    def front_value(self):
        """
        Get the value of the first node in the list.
        :return: The value of the head node, or None if the list is empty.
        """
        return self.head.value if self.head else None

    def push_front(self, value):
        """
        Add a node with the specified value to the front of the list.
        :param value: The value to add.
        """
        new_node = Node(value, self.head)
        if self.is_empty():
            self.tail = new_node
        self.head = new_node
        self.size += 1

    def push_back(self, value):
        """
        Add a node with the specified value to the back of the list.
        :param value: The value to add.
        """
        new_node = Node(value)
        if self.is_empty():
            self.head = new_node
        else:
            self.tail.next_node = new_node
        self.tail = new_node
        self.size += 1

    def pop_front(self):
        """
        Remove and return the value of the node at the front of the list.
        :return: The value of the removed node, or None if the list is empty.
        """
        if self.is_empty():
            return None

        value = self.head.value
        self.head = self.head.next_node
        self.size -= 1

        if self.is_empty():
            self.tail = None

        return value

    @insertion_sort_wrapper
    def insertion_sort(self):
        """
        Sort the linked list using the insertion sort algorithm.
        """
        if self.is_empty() or self.head.next_node is None:
            return

        sorted_list = LinkedList()
        sorted_list.push_back(self.pop_front())

        while not self.is_empty():
            current_value = self.pop_front()

            if current_value >= sorted_list.tail.value:
                sorted_list.push_back(current_value)
            elif current_value <= sorted_list.front_value():
                sorted_list.push_front(current_value)
            else:
                current = sorted_list.head
                while current.next_node:
                    if current_value <= current.next_node.value:
                        new_node = Node(current_value, current.next_node)
                        current.next_node = new_node
                        sorted_list.size += 1
                        break
                    current = current.next_node

        self.head, self.tail, self.size = sorted_list.head, sorted_list.tail, sorted_list.size


# Example usage
if __name__ == "__main__":
    linked_list = LinkedList([6, 2, 3, 1, 4, 5])
    print("Before sorting:", linked_list)
    linked_list.insertion_sort()
    print("After sorting:", linked_list)
