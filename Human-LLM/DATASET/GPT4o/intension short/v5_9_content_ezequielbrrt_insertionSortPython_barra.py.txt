from time import sleep
from tqdm import tqdm

def process_items(item_count, delay):
    """
    Process a given number of items, simulating work with a delay for each item.

    Args:
        item_count (int): The total number of items to process.
        delay (float): The delay (in seconds) to simulate processing time for each item.
    """
    for _ in tqdm(range(item_count), desc="Processing items", unit="item"):
        sleep(delay)

if __name__ == "__main__":
    ITEM_COUNT = 1000
    DELAY_PER_ITEM = 0.02

    process_items(ITEM_COUNT, DELAY_PER_ITEM)
