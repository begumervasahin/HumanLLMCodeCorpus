import timeit
import random

def array_copy(src, src_pos, dest, dest_pos, length):
    """
    Copies a segment from the source array to the destination array.
    
    Parameters:
    src (list): Source array.
    src_pos (int): Starting position in the source array.
    dest (list): Destination array.
    dest_pos (int): Starting position in the destination array.
    length (int): Number of elements to copy.
    """
    for i in range(length):
        dest[i + dest_pos] = src[i + src_pos]

def split_sort(array):
    """
    Recursively splits the array into subarrays, sorts, and merges them.
    
    Parameters:
    array (list): The array to be sorted.
    
    Returns:
    list: A sorted array.
    """
    if len(array) <= 1:
        return array

    mid = len(array) // 2
    left_half = split_sort(array[:mid])
    right_half = split_sort(array[mid:])

    return merge_sort(left_half, right_half)

def merge_sort(left, right):
    """
    Merges two sorted arrays into a single sorted array.
    
    Parameters:
    left (list): The left sorted subarray.
    right (list): The right sorted subarray.
    
    Returns:
    list: A merged and sorted array.
    """
    merged = []
    l_index, r_index = 0, 0

    while l_index < len(left) and r_index < len(right):
        if left[l_index] <= right[r_index]:
            merged.append(left[l_index])
            l_index += 1
        else:
            merged.append(right[r_index])
            r_index += 1

    # Append any remaining elements from both subarrays
    merged.extend(left[l_index:])
    merged.extend(right[r_index:])

    return merged

def generate_random_numbers(size, lower_bound=1, upper_bound=99999):
    """
    Generates a list of random numbers.
    
    Parameters:
    size (int): The number of random numbers to generate.
    lower_bound (int): The minimum value of the random number.
    upper_bound (int): The maximum value of the random number.
    
    Returns:
    list: A list containing random numbers.
    """
    return [random.randint(lower_bound, upper_bound) for _ in range(size)]

# Initialize the numbers list with predefined numbers and append random numbers
numbers = [7, 4, 3, 8, 1, 12, 9, 14] + generate_random_numbers(5000)

# Measure the time taken to sort the list
start_time = timeit.default_timer()
sorted_numbers = split_sort(numbers)
elapsed_time = timeit.default_timer() - start_time

# Output the results
print(f"Merge Sort: {elapsed_time:.6f} seconds")
print(sorted_numbers)
