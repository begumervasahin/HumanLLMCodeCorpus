import random

# Global counter for tracking comparisons
comparison_count = 0

def merge_sort(lst):
    """
    Performs the merge sort algorithm on the given list.

    Args:
        lst (list): The list to be sorted.

    Returns:
        list: A sorted list.
    """
    global comparison_count
    
    # Base case: a list of one or fewer elements is already sorted
    if len(lst) <= 1:
        return lst
    
    # Recursive case: split the list into halves and sort each half
    mid = len(lst) // 2
    left_half = merge_sort(lst[:mid])
    right_half = merge_sort(lst[mid:])
    
    # Merge the sorted halves
    return merge(left_half, right_half)

def merge(left, right):
    """
    Merges two sorted lists into one sorted list.

    Args:
        left (list): The first sorted list.
        right (list): The second sorted list.

    Returns:
        list: A merged and sorted list.
    """
    global comparison_count
    merged = []
    i = j = 0
    
    # Merge the two lists by comparing their elements
    while i < len(left) and j < len(right):
        comparison_count += 1
        if left[i] < right[j]:
            merged.append(left[i])
            i += 1
        else:
            merged.append(right[j])
            j += 1

    # Append remaining elements from left and right
    merged.extend(left[i:])
    merged.extend(right[j:])
    
    return merged

def load_words(filename):
    """
    Loads words from a given file.

    Args:
        filename (str): The name of the file containing the words.

    Returns:
        list: A list of words from the file.
    """
    with open(filename, 'r') as file:
        return [line.strip() for line in file]

def test_sorting(words, description):
    """
    Tests sorting performance on a given list of words and prints the results.

    Args:
        words (list): The list of words to be sorted.
        description (str): A description of the current test case.
    """
    global comparison_count
    
    print(f'Testing on {description}')
    
    # Test on sorted words
    comparison_count = 0
    print(f'{description} sorted')
    merge_sort(words.copy())
    print('Number of comparisons:', comparison_count)
    
    # Test on reversed words
    comparison_count = 0
    print(f'{description} reversed')
    merge_sort(list(reversed(words)))
    print('Number of comparisons:', comparison_count)
    
    # Test on randomly shuffled words
    comparison_count = 0
    print(f'{description} random')
    random.shuffle(words)
    merge_sort(words.copy())
    print('Number of comparisons:', comparison_count)
    
    print('-------------\n')

def main():
    """
    Main routine to execute the merge sort tests on various word sets.
    """
    words = load_words('words.txt')
    total_words = len(words)
    print(f'Total words: {total_words}\n')
    
    test_cases = [
        (words[0:10], "10 words"),
        (words[0:30], "30 words"),
        (words[0:100], "100 words"),
        (words[0:300], "300 words"),
        (words[0:1000], "1000 words"),
        (words[0:3000], "3000 words"),
        (words[0:10000], "10000 words"),
        (words, "all words"),
    ]
    
    for case, description in test_cases:
        test_sorting(case, description)

if __name__ == "__main__":
    main()
