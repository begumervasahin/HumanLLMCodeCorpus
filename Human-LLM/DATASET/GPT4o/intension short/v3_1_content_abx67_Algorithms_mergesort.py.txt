def merge(arr, left, mid, right):
    # Split the array into two subarrays
    left_subarray = arr[left:mid + 1]
    right_subarray = arr[mid + 1:right + 1]

    # Initialize pointers for the subarrays and the merged array
    left_index, right_index = 0, 0
    merge_index = left

    # Merge the subarrays back into the main array
    while left_index < len(left_subarray) and right_index < len(right_subarray):
        if left_subarray[left_index] <= right_subarray[right_index]:
            arr[merge_index] = left_subarray[left_index]
            left_index += 1
        else:
            arr[merge_index] = right_subarray[right_index]
            right_index += 1
        merge_index += 1

    # Copy any remaining elements from the left subarray
    while left_index < len(left_subarray):
        arr[merge_index] = left_subarray[left_index]
        left_index += 1
        merge_index += 1

    # Copy any remaining elements from the right subarray
    while right_index < len(right_subarray):
        arr[merge_index] = right_subarray[right_index]
        right_index += 1
        merge_index += 1

def merge_sort(arr, left, right):
    if left < right:
        # Calculate the middle index
        mid = left + (right - left) // 2

        # Recursively sort the left and right halves
        merge_sort(arr, left, mid)
        merge_sort(arr, mid + 1, right)

        # Merge the sorted halves
        merge(arr, left, mid, right)

# Example usage
if __name__ == "__main__":
    arr = [12, 11, 13, 5, 6, 7]
    print("Original array:", arr)
    
    merge_sort(arr, 0, len(arr) - 1)
    
    print("Sorted array:", arr)
