import math

def stooge_sort(arr, start, end):
    if end - start + 1 == 2 and arr[start] > arr[end]:
        arr[start], arr[end] = arr[end], arr[start]
    elif end - start + 1 > 2:
        m = (end - start + 1) // 3
        stooge_sort(arr, start, end - m)
        stooge_sort(arr, start + m, end)
        stooge_sort(arr, start, end - m)

def read_input_file(file_path):
    nums = []
    with open(file_path, "r") as file:
        for line in file:
            nums.append([int(num) for num in line.split()[1:]])
    return nums

def write_output_file(file_path, sorted_arrays):
    with open(file_path, "w") as file:
        for array in sorted_arrays:
            file.write(" ".join(map(str, array)) + "\n")

def main():
    input_file = "data.txt"
    output_file = "stooge.out"

    nums = read_input_file(input_file)

    print("Merge Sort")
    print("The unsorted arrays are:")
    for array in nums:
        print(array)

    for array in nums:
        stooge_sort(array, 0, len(array) - 1)

    print("The sorted arrays are:")
    for array in nums:
        print(array)

    write_output_file(output_file, nums)

if __name__ == "__main__":
    main()
