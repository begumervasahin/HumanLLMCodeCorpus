import math

def read_data_from_file(file_path):
    """
    Reads the data from the specified file and returns it as a list of integer lists.
    
    Args:
        file_path (str): The path to the input file.
    
    Returns:
        list: A list of lists containing integers read from the file.
    """
    data = []
    with open(file_path, "r") as file:
        for line in file:
            # Convert the space-separated numbers (ignoring the first value) to integers
            numbers = [int(num) for num in line.split()[1:]]
            data.append(numbers)
    return data

def stooge_sort(array, start, end):
    """
    Recursively sorts the array using the Stooge Sort algorithm.
    
    Args:
        array (list): The list of integers to sort.
        start (int): The starting index of the sub-array.
        end (int): The ending index of the sub-array.
    """
    if start >= end:
        return
    
    # If there are exactly two elements, swap if out of order
    if array[start] > array[end]:
        array[start], array[end] = array[end], array[start]
    
    # If there are more than two elements, recursively apply Stooge Sort
    if end - start + 1 > 2:
        third = (end - start + 1) // 3
        stooge_sort(array, start, end - third)
        stooge_sort(array, start + third, end)
        stooge_sort(array, start, end - third)

def write_data_to_file(file_path, data):
    """
    Writes the sorted data to the specified output file.
    
    Args:
        file_path (str): The path to the output file.
        data (list): The list of sorted integer lists to write to the file.
    """
    with open(file_path, "w") as file:
        for line in data:
            file.write(" ".join(map(str, line)) + "\n")

def main():
    """
    Main function to handle reading, sorting, and writing data.
    """
    # Read the data from the input file
    data = read_data_from_file("data.txt")
    
    # Display the unsorted arrays
    print("Merge Sort")
    print("The unsorted arrays are:")
    for array in data:
        print(array)
    
    # Sort each array using Stooge Sort
    for array in data:
        stooge_sort(array, 0, len(array) - 1)
    
    # Display the sorted arrays
    print("The sorted arrays are:")
    for array in data:
        print(array)
    
    # Write the sorted arrays to the output file
    write_data_to_file("stooge.out", data)

if __name__ == "__main__":
    main()
