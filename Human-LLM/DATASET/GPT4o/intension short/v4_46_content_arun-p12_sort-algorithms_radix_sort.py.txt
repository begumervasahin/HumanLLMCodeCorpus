def radix_sort(arr, verbose=0, desc=0):
    def get_digit_count(n, base):
        """Returns the number of digits in a number based on the given base."""
        count = 0
        while n > 0:
            count += 1
            n //= base
        return count

    def get_digit(n, position, base=10):
        """Extracts the digit at the given position (1-indexed) from the number based on the given base."""
        for _ in range(position - 1):
            n //= base
        return n % base

    def distribute_to_buckets(arr, position, base):
        """Distributes numbers into buckets based on the digit at the specified position."""
        buckets = [[] for _ in range(base)]
        for num in arr:
            digit = get_digit(num, position, base)
            buckets[digit].append(num)
            if verbose == 2:
                print(f"Bucket distribution for digit position {position}: {buckets}")
        return [num for bucket in buckets for num in bucket]

    base = 10
    min_value = min(arr)
    
    # Normalize the array if there are negative numbers
    if min_value < 0:
        arr = [x - min_value for x in arr]

    max_value = max(arr)
    num_passes = get_digit_count(max_value, base)
    
    # Perform sorting passes for each digit position
    for i in range(1, num_passes + 1):
        arr = distribute_to_buckets(arr, i, base)
        if verbose:
            print(f"After pass {i}: {arr}")
    
    # Restore original values if normalization was applied
    if min_value < 0:
        arr = [x + min_value for x in arr]
    
    # Reverse the array if descending order is requested
    if desc:
        arr.reverse()

    return arr
