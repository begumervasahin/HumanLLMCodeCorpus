def insertion_sort(array):
    # Start from the second element as the first element is considered sorted
    for i in range(1, len(array)):
        current_value = array[i]
        position = i - 1

        # Move elements of the sorted segment forward to find the correct position for the current_value
        while position >= 0 and array[position] > current_value:
            array[position + 1] = array[position]
            position -= 1
        
        # Place the current_value in its correct position
        array[position + 1] = current_value

# Example usage
if __name__ == "__main__":
    data = [12, 11, 13, 5, 6]
    print("Original array:", data)
    insertion_sort(data)
    print("Sorted array:", data)
