def counting_sort(arr, verbose=0, desc=0):
    # Find the smallest and largest values in the array
    smallest = min(arr)
    largest = max(arr)

    # Normalize the array by shifting the smallest value to 0
    if smallest < 0:
        normalized_arr = [x - smallest for x in arr]
    else:
        normalized_arr = arr.copy()

    if verbose:
        print("Normalized array:", normalized_arr)

    # Initialize the count array based on the range of the normalized array
    count = [0] * (largest - smallest + 1)

    # Count the occurrences of each element
    for num in normalized_arr:
        count[num] += 1

    # Update the count array to be cumulative
    for i in range(1, len(count)):
        count[i] += count[i - 1]
        if verbose == 2:
            print(f"Cumulative count at index {i}: {count[i]}")

    # Create the output array
    output = [0] * len(arr)

    # Build the output array by scanning the input array from right to left
    for num in reversed(normalized_arr):
        count[num] -= 1
        output[count[num]] = num

    if verbose:
        print("Sorted output (before denormalization):", output)

    # Denormalize the output array if necessary
    if smallest < 0:
        output = [x + smallest for x in output]

    # Reverse the output array if descending order is required
    if desc:
        output.reverse()

    return output
