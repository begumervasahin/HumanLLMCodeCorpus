import statistics

def find_pivot_index(arr, first, last):
    """Determine the pivot index using the median-of-three method."""
    mid = (first + last) // 2
    candidates = [arr[first], arr[mid], arr[last]]
    pivot_value = statistics.median(candidates)
    
    if pivot_value == arr[first]:
        return first
    elif pivot_value == arr[last]:
        return last
    else:
        return mid

def partition(arr, first, last):
    """Partition the array around the chosen pivot."""
    pivot_index = find_pivot_index(arr, first, last)
    arr[pivot_index], arr[last] = arr[last], arr[pivot_index]  # Move pivot to the end
    pivot = arr[last]
    
    left, right = first, last - 1
    
    while True:
        # Move the left pointer to the right as long as elements are smaller than the pivot
        while left <= right and arr[left] <= pivot:
            left += 1
        # Move the right pointer to the left as long as elements are larger than the pivot
        while left <= right and arr[right] >= pivot:
            right -= 1
        
        # If pointers cross, break out of the loop
        if left > right:
            break
        
        # Swap elements at left and right pointers
        arr[left], arr[right] = arr[right], arr[left]
    
    # Place the pivot in its correct position
    arr[left], arr[last] = arr[last], arr[left]
    
    return left

def quicksort(arr, first, last):
    """Recursive quicksort algorithm to sort the array."""
    if first < last:
        pivot_index = partition(arr, first, last)
        quicksort(arr, first, pivot_index - 1)
        quicksort(arr, pivot_index + 1, last)

# Example usage
if __name__ == "__main__":
    arr = [56, 26, 93, 17, 31, 44]
    quicksort(arr, 0, len(arr) - 1)
    print("Sorted array:", arr)
