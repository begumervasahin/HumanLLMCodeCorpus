class FileReadError(Exception):
    """Exception raised when there is an error reading the file."""
    pass

class NotIntError(Exception):
    """Exception raised when conversion to integers fails."""
    pass

def read_lines(file_path):
    """Reads lines from a file and returns them as a list of strings."""
    try:
        with open(file_path, 'r') as file:
            lines = file.read().splitlines()
        return lines
    except Exception as e:
        raise FileReadError(f"Error reading the file: {e}")

def write_file(write_mode, append, file_path, data):
    """Writes a list of data to a file. The data can be either integers or strings."""
    mode = 'a' if append else 'w'
    try:
        with open(file_path, mode) as file:
            for item in data:
                file.write(f"{item}\n")
    except Exception as e:
        raise FileReadError(f"Error writing to the file: {e}")

def convert_lines_to_ints(strict, lines):
    """Converts a list of strings to integers. Raises an error if conversion fails."""
    result = []
    for line in lines:
        try:
            result.append(int(line))
        except ValueError:
            if strict:
                raise NotIntError(f"Cannot convert '{line}' to an integer.")
            else:
                continue
    return result
