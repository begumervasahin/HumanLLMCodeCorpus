import math

def read_data_from_file(file_path):
    """
    Reads data from the specified file and returns it as a list of lists of integers.
    
    Args:
        file_path (str): Path to the input file.
    
    Returns:
        list: A list of lists, where each inner list contains integers from one line of the file.
    """
    with open(file_path, "r") as file:
        return [[int(num) for num in line.split()[1:]] for line in file]

def stooge_sort(array, start, end):
    """
    Recursively sorts an array using the Stooge Sort algorithm.
    
    Args:
        array (list): The list of integers to sort.
        start (int): The starting index of the sub-array.
        end (int): The ending index of the sub-array.
    """
    if start >= end:
        return

    if array[start] > array[end]:
        array[start], array[end] = array[end], array[start]

    if end - start + 1 > 2:
        third = (end - start + 1) // 3
        stooge_sort(array, start, end - third)
        stooge_sort(array, start + third, end)
        stooge_sort(array, start, end - third)

def write_data_to_file(file_path, data):
    """
    Writes the sorted data to the specified output file.
    
    Args:
        file_path (str): Path to the output file.
        data (list): List of lists of sorted integers to write to the file.
    """
    with open(file_path, "w") as file:
        for line in data:
            file.write(" ".join(map(str, line)) + "\n")

def print_arrays(title, arrays):
    """
    Prints arrays with a given title.
    
    Args:
        title (str): The title to print before the arrays.
        arrays (list): The list of arrays to print.
    """
    print(title)
    for array in arrays:
        print(array)

def main():
    """
    Main function to read, sort, and write data.
    """
    input_file = "data.txt"
    output_file = "stooge.out"

    # Read data from the input file
    data = read_data_from_file(input_file)
    
    # Print unsorted arrays
    print_arrays("Unsorted Arrays:", data)
    
    # Sort each array using Stooge Sort
    for array in data:
        stooge_sort(array, 0, len(array) - 1)
    
    # Print sorted arrays
    print_arrays("Sorted Arrays:", data)
    
    # Write sorted arrays to the output file
    write_data_to_file(output_file, data)

if __name__ == "__main__":
    main()
