import tkinter as tk
from tkinter import simpledialog

def select_numbers():
    try:
        # Get the number of elements to be entered by the user
        num_elements = int(entry_numeros.get())
        numbers = []

        # Prompt the user to input each number
        for _ in range(num_elements):
            num = simpledialog.askinteger("Input", "Ingresa el número", parent=app)
            if num is not None:
                numbers.append(num)

        # Display the entered numbers
        label_input_numbers.config(text=f"Números ingresados: {numbers}")

        # Sort the numbers using the Selection Sort algorithm
        sorted_numbers = selection_sort(numbers)

        # Display the sorted numbers
        label_sorted_numbers.config(text=f"Números ordenados: {sorted_numbers}")

    except ValueError:
        label_input_numbers.config(text="Por favor ingresa un número válido.")

def selection_sort(numbers):
    # Implementation of the Selection Sort algorithm
    for i in range(len(numbers) - 1):
        min_index = i
        for j in range(i + 1, len(numbers)):
            if numbers[j] < numbers[min_index]:
                min_index = j
        # Swap the found minimum element with the current element
        numbers[i], numbers[min_index] = numbers[min_index], numbers[i]
    return numbers

# Initialize the main application window
app = tk.Tk()
app.title("Selection Sort")
app.configure(bg='SkyBlue2')

# Label and entry for the number of elements to be input
label_prompt = tk.Label(app, text="¿Cuántos números vas a ingresar?", font='Helvetica 12', bg='SkyBlue2')
label_prompt.pack()

entry_numeros = tk.Entry(app, width=8)
entry_numeros.pack()

# Button to trigger the selection process
button_sort = tk.Button(app, text="Ok!", command=select_numbers)
button_sort.pack(pady=(10, 0))

# Labels to display the input and sorted numbers
label_input_numbers = tk.Label(app, text="", bg='SkyBlue2')
label_input_numbers.pack(pady=(10, 0))

label_sorted_numbers = tk.Label(app, text="", bg='SkyBlue2')
label_sorted_numbers.pack(pady=(10, 0))

# Run the application loop
app.mainloop()
