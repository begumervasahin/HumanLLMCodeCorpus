import tkinter as tk
from tkinter import simpledialog

def radix_sort(numbers):
    """
    Perform radix sort on the provided list of numbers.
    """
    def counting_sort(numbers, exp):
        """
        Perform counting sort based on the digit represented by exp.
        """
        n = len(numbers)
        output = [0] * n
        count = [0] * 10

        # Count occurrences of each digit in the given place value
        for number in numbers:
            index = number // exp
            count[index % 10] += 1

        # Update count[i] to contain the actual position of this digit in output
        for i in range(1, 10):
            count[i] += count[i - 1]

        # Build the output array
        i = n - 1
        while i >= 0:
            index = numbers[i] // exp
            output[count[index % 10] - 1] = numbers[i]
            count[index % 10] -= 1
            i -= 1

        # Copy the output array back to numbers[]
        for i in range(n):
            numbers[i] = output[i]

    # Find the maximum number to determine the number of digits
    max_number = max(numbers)
    exp = 1
    while max_number // exp > 0:
        counting_sort(numbers, exp)
        exp *= 10

def get_numbers_and_sort():
    """
    Get the number of elements and the elements themselves from the user, then sort them.
    """
    num_elements = int(entry_num.get())
    numbers = []
    
    for _ in range(num_elements):
        number = simpledialog.askinteger("Input", "Enter a number:")
        numbers.append(number)
    
    # Display the original list
    label_original.config(text="Original List: " + str(numbers))
    
    # Sort the numbers using radix sort
    radix_sort(numbers)
    
    # Display the sorted list
    label_sorted.config(text="Sorted List: " + str(numbers))

# Set up the main application window
app = tk.Tk()
app.title("Radix Sort Application")
app.geometry('300x200')
app.configure(bg='SkyBlue2')

# Create and place the UI elements
label_prompt = tk.Label(app, text="How many numbers will you enter?", font='Helvetica 12', bg='SkyBlue2')
label_prompt.pack(pady=10)

entry_num = tk.Entry(app, width=8)
entry_num.pack()

button_sort = tk.Button(app, text="Sort", command=get_numbers_and_sort)
button_sort.pack(pady=10)

label_original = tk.Label(app, text="", bg='SkyBlue2')
label_original.pack(pady=5)

label_sorted = tk.Label(app, text="", bg='SkyBlue2')
label_sorted.pack(pady=5)

# Start the Tkinter event loop
app.mainloop()
