def radix_sort(arr, base=10, verbose=False, desc=False):
    def passes(n, base):
        """Returns the number of digits in the largest number based on the given base."""
        cnt = 0
        while n > 0:
            cnt += 1
            n //= base
        return cnt

    def positional_digit(n, pos, base=10):
        """Extracts the digit at the specified position from the number."""
        return (n // base**pos) % base

    def bucketize(arr, pos, base):
        """Distributes numbers into buckets based on the digit at the current position."""
        buckets = [[] for _ in range(base)]
        for number in arr:
            digit = positional_digit(number, pos, base)
            buckets[digit].append(number)
            if verbose == 2:
                print(f"Number {number} goes to bucket {digit}: {buckets}")
        # Flatten the list of buckets back into a single list
        return [num for bucket in buckets for num in bucket]

    # Adjust the array for negative values (if any)
    min_value = min(arr)
    if min_value < 0:
        arr = [num - min_value for num in arr]

    # Determine the number of passes needed based on the largest number
    max_value = max(arr)
    num_passes = passes(max_value, base)

    # Perform sorting by positional digits
    for pos in range(num_passes):
        arr = bucketize(arr, pos, base)
        if verbose:
            print(f"After pass {pos + 1}: {arr}")

    # Restore the original values if adjustment was made for negatives
    if min_value < 0:
        arr = [num + min_value for num in arr]

    # Reverse the list if sorting in descending order
    if desc:
        arr = arr[::-1]

    return arr


# Example usage:
if __name__ == "__main__":
    sample_array = [170, 45, 75, 90, 802, 24, 2, 66]
    sorted_array = radix_sort(sample_array, verbose=True, desc=False)
    print("Sorted Array:", sorted_array)
