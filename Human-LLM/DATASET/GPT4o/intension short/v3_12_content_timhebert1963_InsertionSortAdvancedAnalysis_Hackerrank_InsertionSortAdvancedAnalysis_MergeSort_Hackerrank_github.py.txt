import time

class InversionCounter:
    def __init__(self):
        self.count = 0

def split_array(array):
    """Splits the array into two halves and returns them."""
    mid_index = len(array) // 2
    return array[:mid_index], array[mid_index:]

def count_and_merge_inversions(left_half, right_half, counter):
    """Counts inversions between two sorted halves and merges them."""
    left_half.sort()
    right_half.sort()
    i, j = 0, 0
    
    while i < len(left_half):
        while j < len(right_half) and right_half[j] < left_half[i]:
            counter.count += len(left_half) - i
            j += 1
        i += 1

def merge_sort_and_count(array, counter):
    """Recursively sorts the array and counts inversions."""
    if len(array) > 1:
        left_half, right_half = split_array(array)
        merge_sort_and_count(left_half, counter)
        merge_sort_and_count(right_half, counter)
        count_and_merge_inversions(left_half, right_half, counter)

def calculate_inversions(array):
    """Calculates the number of inversions in the array."""
    counter = InversionCounter()
    merge_sort_and_count(array, counter)
    return counter.count

def get_current_time():
    """Returns the current time in seconds."""
    return int(time.time())

def format_duration(seconds):
    """Formats the duration into minutes and seconds."""
    minutes = seconds // 60
    return minutes, seconds % 60

def display_test_results(test_name, calculated_inversions, expected_inversions, start_time):
    """Displays the results of a test case."""
    elapsed_time = get_current_time() - start_time
    minutes, seconds = format_duration(elapsed_time)
    status = 'PASS' if calculated_inversions == expected_inversions else 'FAIL'
    
    print(f"\n{test_name} - Inversions: {calculated_inversions} (Expected: {expected_inversions}) - {status}")
    print(f"Execution time: {minutes} minutes, {seconds} seconds\n")

def main():
    test_cases = [
        ('TC2_1', 'TC2_1_length_441_answer_46768', 46768),
        ('TC2_2', 'TC2_2_length_18_answer_77', 77)
    ]
    
    for test_name, module_name, expected_inversions in test_cases:
        test_module = __import__(module_name)
        start_time = get_current_time()
        calculated_inversions = calculate_inversions(test_module.arr)
        display_test_results(test_name, calculated_inversions, expected_inversions, start_time)

if __name__ == '__main__':
    main()
