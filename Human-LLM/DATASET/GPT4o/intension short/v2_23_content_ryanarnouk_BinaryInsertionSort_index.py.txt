def binary_search(arr, val, start, end):
    """
    Perform a binary search to determine the index where `val` should be inserted
    to maintain the sorted order in the array `arr`.
    """
    while start <= end:
        mid = (start + end) // 2

        if arr[mid] < val:
            start = mid + 1
        elif arr[mid] > val:
            end = mid - 1
        else:
            return mid

    return start

def insertion_sort(arr):
    """
    Sort the array `arr` using the insertion sort algorithm enhanced with binary search.
    """
    for i in range(1, len(arr)):
        val = arr[i]
        # Find the correct insertion point for the current element using binary search
        pos = binary_search(arr, val, 0, i - 1)
        # Shift elements to the right and insert the current element at the correct position
        arr = arr[:pos] + [val] + arr[pos:i] + arr[i+1:]

    return arr

    sorted_arr = insertion_sort(arr)
