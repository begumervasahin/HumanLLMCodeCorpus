import tkinter as tk
from tkinter import simpledialog

def obtener_numeros(cantidad):
    """
    Solicita al usuario que ingrese una cantidad específica de números y los devuelve en una lista.
    """
    numeros = []
    for i in range(cantidad):
        numero = simpledialog.askinteger("Ingresar Número", f"Ingrese el número {i + 1}:")
        if numero is not None:
            numeros.append(numero)
    return numeros

def buscar_en_lista(numeros, numero_buscado):
    """
    Realiza una búsqueda secuencial del número_buscado en la lista de números.
    Devuelve la posición del número si se encuentra, o None si no se encuentra.
    """
    for index, num in enumerate(numeros):
        if num == numero_buscado:
            return index
    return None

def buscar_numero():
    try:
        # Obtener la cantidad de números a ingresar
        cantidad_numeros = int(entry_cantidad.get())

        # Obtener los números ingresados por el usuario
        numeros = obtener_numeros(cantidad_numeros)

        # Solicitar el número que se desea buscar
        numero_buscado = simpledialog.askinteger("Buscar Número", "Ingresa el número que deseas buscar:")

        # Mostrar los números ingresados en la interfaz
        lista_numeros = ', '.join(map(str, numeros))
        label_lista.config(text=f"Números ingresados: {lista_numeros}")

        # Realizar la búsqueda secuencial del número
        posicion = buscar_en_lista(numeros, numero_buscado)
        if posicion is not None:
            label_resultado.config(text=f"Número encontrado en la posición: {posicion}")
        else:
            label_resultado.config(text="Número no encontrado")

    except ValueError:
        label_resultado.config(text="Por favor, ingresa un número válido")

# Configuración de la ventana principal de la aplicación
def crear_interfaz():
    app = tk.Tk()
    app.title("Búsqueda Secuencial")
    app.geometry('300x200')
    app.configure(bg='SkyBlue2')

    # Etiqueta y entrada para ingresar la cantidad de números
    tk.Label(app, text="¿Cuántos números vas a ingresar?", font='Helvetica 12', bg='SkyBlue2').pack()
    
    global entry_cantidad
    entry_cantidad = tk.Entry(app, width=8)
    entry_cantidad.pack()

    # Botón para iniciar la búsqueda
    tk.Button(app, text="Buscar", command=buscar_numero).pack(pady=(10, 0))

    # Etiqueta para mostrar la lista de números ingresados
    global label_lista
    label_lista = tk.Label(app, text="", bg='SkyBlue2')
    label_lista.pack(pady=(10, 0))

    # Etiqueta para mostrar el resultado de la búsqueda
    global label_resultado
    label_resultado = tk.Label(app, text="", bg='SkyBlue2')
    label_resultado.pack(pady=(10, 0))

    app.mainloop()

if __name__ == "__main__":
    crear_interfaz()
