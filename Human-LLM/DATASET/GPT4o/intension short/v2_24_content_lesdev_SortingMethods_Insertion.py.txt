import tkinter as tk
from tkinter import simpledialog

def insertion_sort():
    # Retrieve the number of elements to be sorted from the input field
    num_elements = int(entry_numbers.get())
    
    # Initialize an empty list to store the numbers entered by the user
    numbers = []
    
    # Prompt the user to input each number, and append it to the list
    for _ in range(num_elements):
        number = simpledialog.askinteger("Input", "Enter a number:")
        if number is not None:
            numbers.append(number)
    
    # Display the original list of numbers on the GUI
    label_original.config(text=f"Original List: {numbers}")
    
    # Perform the insertion sort algorithm on the list
    for i in range(1, len(numbers)):
        key = numbers[i]
        j = i - 1
        while j >= 0 and numbers[j] > key:
            numbers[j + 1] = numbers[j]
            j -= 1
        numbers[j + 1] = key
    
    # Display the sorted list on the GUI
    label_sorted.config(text=f"Sorted List: {numbers}")

# Set up the main application window
app = tk.Tk()
app.title("Insertion Sort")
app.geometry('300x200')
app.configure(bg='SkyBlue2')

# Create and position the label prompting the user for input
label_prompt = tk.Label(app, text="How many numbers will you enter?", font="Helvetica 12", bg='SkyBlue2')
label_prompt.pack(pady=(10, 0))

# Create and position the input field for the number of elements
entry_numbers = tk.Entry(app, width=8)
entry_numbers.pack(pady=(5, 0))

# Create and position the button to trigger the sorting process
button_sort = tk.Button(app, text="Sort", command=insertion_sort)
button_sort.pack(pady=(10, 0))

# Create and position labels to display the original and sorted lists
label_original = tk.Label(app, text="", bg='SkyBlue2')
label_original.pack(pady=(10, 0))

label_sorted = tk.Label(app, text="", bg='SkyBlue2')
label_sorted.pack(pady=(10, 0))

# Start the application's main event loop
app.mainloop()
