def quicksort_iterative(arr, start, end):
    """Performs an in-place iterative quicksort on the given array."""
    # Initialize stack to simulate recursion
    stack = [(start, end)]

    # Process the stack until it is empty
    while stack:
        start, end = stack.pop()

        # Partition the array and get the pivot index
        pivot_index = partition(arr, start, end)

        # Push the left and right subarrays to the stack if they have more than one element
        if pivot_index - 1 > start:
            stack.append((start, pivot_index - 1))
        if pivot_index + 1 < end:
            stack.append((pivot_index + 1, end))


def partition(arr, start, end):
    """Helper function to partition the array around a pivot."""
    pivot = arr[end]
    i = start - 1

    for j in range(start, end):
        if arr[j] <= pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]

    arr[i + 1], arr[end] = arr[end], arr[i + 1]
    return i + 1


def find_special_value(arr):
    """Finds and prints the special value based on the sorted array."""
    quicksort_iterative(arr, 0, len(arr) - 1)

    for i in range(len(arr)):
        if arr[i] == len(arr) - i:
            print(arr[i])
            return
        elif arr[i] > len(arr) - i:
            print(len(arr) - i - 1)
            return

    # If no special value found, print the length of the array
    print(len(arr))


def main():
    """Main function to take input and run the algorithm."""
    n = int(input("Enter the value of n: ").strip())
    arr_input = input("Enter array: [Example: {1,2,3}]: ").strip()[1:-1]
    arr = list(map(int, arr_input.split(',')))

    if len(arr) == n:
        find_special_value(arr)
    else:
        print("N and array size don't match. Validate your input. Exiting program.")


if __name__ == "__main__":
    main()
