def merge(array, left, mid, right):
    # Create temporary arrays for left and right subarrays
    left_subarray = array[left:mid+1]
    right_subarray = array[mid+1:right+1]

    i = j = 0  # Pointers for left_subarray and right_subarray
    k = left   # Pointer for the main array

    # Merge the subarrays back into the original array
    while i < len(left_subarray) and j < len(right_subarray):
        if left_subarray[i] <= right_subarray[j]:
            array[k] = left_subarray[i]
            i += 1
        else:
            array[k] = right_subarray[j]
            j += 1
        k += 1

    # Copy any remaining elements of left_subarray, if any
    while i < len(left_subarray):
        array[k] = left_subarray[i]
        i += 1
        k += 1

    # Copy any remaining elements of right_subarray, if any
    while j < len(right_subarray):
        array[k] = right_subarray[j]
        j += 1
        k += 1

def mergesort(array, left, right):
    if left < right:
        mid = left + (right - left) // 2

        # Recursively sort the first and second halves
        mergesort(array, left, mid)
        mergesort(array, mid + 1, right)

        # Merge the sorted halves
        merge(array, left, mid, right)
