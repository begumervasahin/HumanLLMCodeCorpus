import random

# Generate a list of 10 random numbers between 0 and 99
arr = [random.randrange(100) for _ in range(10)]

# Function to check if the array is sorted
def is_sorted(array) -> bool:
    for i in range(len(array) - 1):
        if array[i] > array[i + 1]:
            return False
    return True

# Function to perform Insertion Sort on an array
def insertion_sort(array):
    for i in range(1, len(array)):
        key = array[i]
        j = i - 1
        # Move elements of array[0..i-1] that are greater than key to one position ahead of their current position
        while j >= 0 and array[j] > key:
            array[j + 1] = array[j]
            j -= 1
        array[j + 1] = key
    return array

# Print the original array
print("Original array:", arr)

# Sort the array using Insertion Sort and print the sorted array
sorted_arr = insertion_sort(arr)
print("Sorted array:", sorted_arr)

# Check if the array is sorted
print("Is the array sorted?", is_sorted(sorted_arr))
