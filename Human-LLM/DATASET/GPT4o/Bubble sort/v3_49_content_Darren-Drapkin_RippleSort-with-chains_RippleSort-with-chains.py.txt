names = ["", "", "ROSE", "ALGERNON", "CLARENCE", "VIOLET", "CECIL", "CUTHBERT"]
genders = ["", "", "FEMALE", "MALE", "MALE", "FEMALE", "MALE", "MALE"]
links = [0] * len(names)

def initialise():
    """
    Creates a linked list of indices based on gender.
    Females are linked starting at links[0] and males at links[1].
    """
    links[0] = 0  # Start of the female list
    links[1] = 0  # Start of the male list

    for index in range(2, len(names)):
        if genders[index] == "FEMALE":
            queue = 0  # Queue for females
        else:
            queue = 1  # Queue for males
        
        links[index] = links[queue]
        links[queue] = index

def ripple_sort(queue_start):
    """
    Sorts the linked list of names starting from the given queue_start index.
    
    Parameters:
    queue_start (int): The index in the links array where sorting starts.
    """
    for _ in range(2, len(names)):
        swapped = False
        current = links[queue_start]
        
        while current != 0 and links[current] != 0:
            next_index = links[current]
            if names[current] > names[next_index]:
                # Swap the links
                links[current], links[next_index] = links[next_index], links[current]
                swapped = True
            current = links[current]
        
        if not swapped:
            break

def print_sorted_list():
    """
    Prints the sorted list of names based on the linked list.
    """
    print("Sorted Females:")
    current = links[0]
    while current != 0:
        print(names[current])
        current = links[current]
    
    print("\nSorted Males:")
    current = links[1]
    while current != 0:
        print(names[current])
        current = links[current]

def main():
    initialise()
    ripple_sort(0)  # Sort females
    ripple_sort(1)  # Sort males
    print_sorted_list()

if __name__ == "__main__":
    main()
