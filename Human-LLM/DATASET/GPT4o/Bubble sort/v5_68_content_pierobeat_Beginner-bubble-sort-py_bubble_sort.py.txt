def bubble_sort(arr):
    """
    Perform Bubble Sort on the given array.

    Args:
        arr (list): The list of elements to be sorted.

    Returns:
        list: The sorted list.
    """
    n = len(arr)
    while True:
        swapped = False
        for i in range(n - 1):
            if arr[i] > arr[i + 1]:
                arr[i], arr[i + 1] = arr[i + 1], arr[i]
                swapped = True
        if not swapped:
            break
    return arr

# Testing the bubble_sort function with different inputs
print("Already sorted:", bubble_sort([1, 2, 3, 4, 5, 6]))
print("Partially unsorted:", bubble_sort([2, 1, 4, 3, 6, 5]))
print("Completely unsorted:", bubble_sort([6, 5, 4, 3, 2, 1]))
