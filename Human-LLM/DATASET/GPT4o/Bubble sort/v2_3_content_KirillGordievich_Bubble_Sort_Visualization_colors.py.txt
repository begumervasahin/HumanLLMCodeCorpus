# Define color variables using RGB values
colors = {
    "white": (255, 255, 255),
    "red": (255, 0, 0),
    "blue": (0, 0, 255),
    "green": (0, 255, 0)
}

def print_colors():
    """
    Prints the defined colors and their RGB values.
    """
    for color_name, rgb_values in colors.items():
        print(f"{color_name.capitalize()}: {rgb_values}")

def main():
    """
    Main function to run the script.
    """
    print_colors()

if __name__ == "__main__":
    main()
