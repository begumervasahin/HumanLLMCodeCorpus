class Node:
    def __init__(self, data):
        """Initialize a new node with the given data."""
        self.left = None
        self.right = None
        self.data = data

    def insert(self, data):
        """Insert a new node with the given data into the tree."""
        if self.data:
            if data < self.data:
                if self.left is None:
                    self.left = Node(data)
                else:
                    self.left.insert(data)
            elif data > self.data:
                if self.right is None:
                    self.right = Node(data)
                else:
                    self.right.insert(data)
        else:
            self.data = data

    def findval(self, lkpval):
        """Find a value in the tree and return a message if it's not found."""
        if lkpval < self.data:
            if self.left is None:
                return str(lkpval) + " Not Found"
            return self.left.findval(lkpval)
        elif lkpval > self.data:
            if self.right is None:
                return str(lkpval) + " Not Found"
            return self.right.findval(lkpval)
        else:
            return str(self.data) + ' is found'

    def PrintTree(self):
        """Print the tree in order."""
        if self.left:
            self.left.PrintTree()
        print(self.data, end=' ')
        if self.right:
            self.right.PrintTree()

if __name__ == "__main__":
    root = Node(14)
    root.insert(6)
    root.insert(18)
    root.insert(3)

    print(root.findval(7))  # Output: "7 Not Found"
    print(root.findval(14)) # Output: "14 is found"
    root.PrintTree()        # Output: 3 6 14 18

def clouds(x, y, diameter, widths=5, lengths=2, transparency=100):
    """
    Draw the clouds.
    @param int x: The x-coordinate of the cloud.
    @param int y: The y-coordinate of the cloud.
    @param int diameter: The diameter of the cloud circles.
    @param int widths: The number of circles in width.
    @param int lengths: The number of circles in length.
    @param int transparency: The transparency of the cloud.
    @return None
    """
    fill(250, 250, 250, transparency if diameter < 60 else 150)
    noStroke()
    for m in range(widths):
        for n in range(lengths):
            ellipse(x + 25 * m, y + 25 * n, diameter, diameter)

def setGradient(x, y, w, h, c1, c2, axis):
    """
    Draw a color gradient.
    @param int x: The x-coordinate of the gradient.
    @param int y: The y-coordinate of the gradient.
    @param int w: The width of the gradient.
    @param int h: The height of the gradient.
    @param color c1: The start color.
    @param color c2: The end color.
    @param str axis: The axis of the gradient ("X_AXIS" or "Y_AXIS").
    @return None
    """
    noFill()
    if axis == "Y_AXIS":
        for i in range(y, y + h + 1):
            inter = map(i, y, y + h, 0, 1)
            c = lerpColor(c1, c2, inter)
            stroke(c)
            line(x, i, x + w, i)
    elif axis == "X_AXIS":
        for i in range(x, x + w + 1):
            inter = map(i, x, x + w, 0, 1)
            c = lerpColor(c1, c2, inter)
            stroke(c)
            line(i, y, i, y + h)

class TextBox:
    """
    A text box.
    @param tuple rect_digits: parameters of the rect (x, y, width, height)
    @param tuple rect_fill: Color of the rectangle (R, G, B, [optional A])
    @param str text_fill: the fill of the text
    @param int text_size: size of the text
    @param bool selected: True if the text_box is selected
    @param str default_stroke: stroke for when the box is not selected
    @param str selected_stroke: stroke for when the box is selected
    """

    def __init__(self, rect_digits, rect_fill, text_fill, text_size, texts, selected_stroke, default_stroke):
        """
        Initialize a text box.
        @param tuple rect_digits: parameters of the rect (x, y, width, height)
        @param tuple rect_fill: Color of the rectangle (R, G, B, [optional A])
        @param str text_fill: the fill of the text
        @param int text_size: size of the text
        @param str texts: text inside the box
        @param str selected_stroke: stroke for when the box is selected
        @param str default_stroke: stroke for when the box is not selected
        @return None
        """
        self.x = rect_digits[0]
        self.y = rect_digits[1]
        self.width = rect_digits[2]
        self.height = rect_digits[3]
        self.rect_fill = rect_fill
        self.text_fill = text_fill
        self.text_size = text_size
        self.text_box = texts
        self.selected_stroke = selected_stroke
        self.default_stroke = default_stroke
        self.selected = False

    def draw_box(self):
        """
        Draw the text box.
        @return None
        """
        transparency = self.rect_fill[3] if len(self.rect_fill) == 4 else 200
        fill(self.rect_fill[0], self.rect_fill[1], self.rect_fill[2], transparency)
        stroke(self.selected_stroke if self.selected else self.default_stroke)
        rect(self.x, self.y, self.width, self.height)
        fill(self.text_fill)
        textSize(self.text_size)
        text(self.text_box, self.x + 5, self.y + 25)
