names = ["", "", "ROSE", "ALGERNON", "CLARENCE", "VIOLET", "CECIL", "CUTHBERT"]
genders = ["", "", "FEMALE", "MALE", "MALE", "FEMALE", "MALE", "MALE"]
links = [0] * len(names)

def initialise():
    """
    Creates a linked list of indices based on gender.
    Females are linked starting at links[0] and males at links[1].
    """
    links[0] = 0  # Start of the female list
    links[1] = 0  # Start of the male list
    for index in range(2, len(names)):
        if genders[index] == "FEMALE":
            queue = 0  # Queue for females
            links[index] = links[queue]
            links[queue] = index
        elif genders[index] == "MALE":
            queue = 1  # Queue for males
            links[index] = links[queue]
            links[queue] = index

def ripple_sort():
    """
    Sorts the linked list of names initialized by the initialise function.
    """
    for index in range(2, len(names)):
        swapped = False
        current = links[0]
        for _ in range(2, len(names) - index + 1):
            previous = current
            current = links[current]
            following = links[current]
            if names[current] > names[following]:
                swapped = True
                # Perform the swap in the linked list
                links[previous], links[current], links[following] = (
                    links[current],
                    links[following],
                    links[previous],
                )
                current = following
        if not swapped:
            break

def print_sorted_list():
    """
    Prints the sorted list of names based on the linked list.
    """
    current = links[0]
    print("Sorted Females:")
    while current != 0:
        print(names[current])
        current = links[current]
    
    current = links[1]
    print("Sorted Males:")
    while current != 0:
        print(names[current])
        current = links[current]

def main():
    initialise()
    ripple_sort()
    print_sorted_list()

if __name__ == "__main__":
    main()
