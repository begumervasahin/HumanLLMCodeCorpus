import random

# Generate a list of 10 random integers between 0 and 99
arr = [random.randrange(100) for i in range(10)]

def check_sort(array) -> bool:
    for i in range(len(array) - 1):
        if array[i] > array[i + 1]:
            return False
    return True

def insertion_sort(array):
    for i in range(1, len(array)):
        key = array[i]
        j = i - 1
        while j >= 0 and key < array[j]:
            array[j + 1] = array[j]
            j -= 1
        array[j + 1] = key
    return array

# Print the original array
print("Original array:", arr)

# Print the sorted array
sorted_arr = insertion_sort(arr.copy())
print("Sorted array:", sorted_arr)

# Check if the sorted array is actually sorted
is_sorted = check_sort(sorted_arr)
print("Is the array sorted correctly?", is_sorted)
