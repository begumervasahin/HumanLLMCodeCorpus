# Define color variables using RGB values
white = (255, 255, 255)
red = (255, 0, 0)
blue = (0, 0, 255)
green = (0, 255, 0)

# Function to print the defined colors
def print_colors():
    print(f"White: {white}")
    print(f"Red: {red}")
    print(f"Blue: {blue}")
    print(f"Green: {green}")

# Main function to run the script
def main():
    print_colors()

if __name__ == "__main__":
    main()
