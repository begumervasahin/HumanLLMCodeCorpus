import random
from time import time

MAX_RANGE = 10001

# Generate a list of random numbers
unsorted_list = random.sample(range(1, MAX_RANGE), MAX_RANGE - 1)

def bubble_sort(input_list):
    """
    Sorts a list using the bubble sort algorithm.

    Args:
        input_list (list): The list of elements to be sorted.

    Returns:
        list: The sorted list.
    """
    is_sorted = False
    while not is_sorted:
        is_sorted = True
        for i in range(len(input_list) - 1):
            if input_list[i] > input_list[i + 1]:
                # Swap the elements
                input_list[i], input_list[i + 1] = input_list[i + 1], input_list[i]
                is_sorted = False
    return input_list

# Create a copy of the unsorted list for bubble sort
unsorted_list_cpy = unsorted_list[:]

# Measure the time taken by bubble sort
start = time()
bubble_sort(unsorted_list_cpy)
bubble_sort_time = time() - start
print(f"Bubble sort algorithm ends in {bubble_sort_time:.4f} seconds to sort a list of {MAX_RANGE - 1} items.")

# Create a copy of the unsorted list for Python's built-in sort
unsorted_list_cpy = unsorted_list[:]

# Measure the time taken by Python's built-in sort (Timsort)
start = time()
unsorted_list_cpy.sort()
timsort_time = time() - start
print(f"System Python sort() Timsort algorithm ends in {timsort_time:.4f} seconds to sort a list of {MAX_RANGE - 1} items.")
