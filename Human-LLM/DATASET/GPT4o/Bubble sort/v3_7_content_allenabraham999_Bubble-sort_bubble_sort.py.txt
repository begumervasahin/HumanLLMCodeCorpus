import random

def bubble_sort(arr):
    """
    Sorts the array using the Bubble Sort algorithm and prints the array after each pass.
    
    Args:
        arr (list): The list of elements to be sorted.
    """
    length = len(arr)
    for j in range(length):
        for i in range(1, length):
            if arr[i - 1] > arr[i]:
                arr[i - 1], arr[i] = arr[i], arr[i - 1]
        print(f"\nArray after {j + 1} pass:")
        print_array(arr)

def print_array(arr):
    """
    Prints the elements of the array on a single line.
    
    Args:
        arr (list): The list of elements to be printed.
    """
    print(" ".join(map(str, arr)))

def fill_array(size, lower_bound, upper_bound):
    """
    Fills an array with random integers within a specified range.
    
    Args:
        size (int): The number of elements in the array.
        lower_bound (int): The lower bound of the random integers.
        upper_bound (int): The upper bound of the random integers.
    
    Returns:
        list: The filled array.
    """
    arr = [random.randint(lower_bound, upper_bound) for _ in range(size)]
    print("The array that will be worked on is:")
    print_array(arr)
    return arr

def main():
    """
    Main function to demonstrate the Bubble Sort algorithm.
    """
    array = fill_array(8, 1, 9)
    print()
    bubble_sort(array)
    print("\nThe final sorted array is:")
    print_array(array)

if __name__ == "__main__":
    main()
