import numpy as np
import cv2
import os
import sys
import glob
from grabCutClass import GrabCutter
from videoProcessor import VideoProcessor
from annotation_suggester import annotation_suggester
from color_hist_frame_select import color_hist_frame_select
from ResNet_preprocess import resnet_process_data_dir
from BubbleNets_frame_select import BubbleNets_sort
from BubbleNets import bn_utils

# Adding methods to sys.path
cwd = os.getcwd()
sys.path.insert(0, os.path.join(cwd, 'methods', 'preprocess'))
sys.path.insert(0, os.path.join(cwd, 'methods', 'annotate_suggest'))

# Flags for user interaction
USER_SCALE = True
USER_SELECT = True

def get_user_annotation(video_dir):
    """Generate user-guided annotations for a video."""
    video_name = os.path.basename(video_dir)
    print(f'\n\nGenerating user-guided annotation for {video_name}')
    
    annotation_dir = os.path.join(video_dir, 'usrAnnotate')
    os.makedirs(annotation_dir, exist_ok=True)
    
    image_dir = os.path.join(video_dir, 'src')
    image_files = sorted(glob.glob(os.path.join(image_dir, '*')))
    
    ant_idx, ant_file = read_annotation_list(os.path.join(video_dir, 'frame_selection', 'all.txt'))
    suggester = annotation_suggester(video_dir)
    user_annotating = True
    
    while user_annotating:
        ant_image_files = glob.glob(os.path.join(annotation_dir, '*'))
        n_ant_imgs = len(ant_image_files)
        
        print(f'Currently {n_ant_imgs} annotation image(s):')
        for ant_file_name in ant_image_files:
            ant_name = os.path.basename(ant_file_name)
            print(ant_name)
            if ant_name in ant_file:
                ant_file.remove(ant_name)
        
        print('Suggested annotation frames remaining:')
        print(ant_file)
        
        response = input('Annotate another image? (y or n)\n')
        if response.lower() not in {'y', 'yes'}:
            user_annotating = False
        if user_annotating:
            while True:
                try:
                    annotation_image_idx = get_annotation_image_index(image_files, suggester)
                    image_path = image_files[annotation_image_idx]
                    annotation_image = cv2.imread(image_path)
                    
                    scale = get_image_scale(annotation_image) if USER_SCALE else 1
                    
                    annotation_image_scaled = cv2.resize(annotation_image, (0, 0), fx=scale, fy=scale)
                    cv2.imshow('Scaled Annotation Image', annotation_image_scaled)
                    cv2.moveWindow('Scaled Annotation Image', 100, 100)
                    cv2.waitKey(20)
                    
                    response = input('Is annotation frame acceptable? (y or n)\n')
                    if response.lower() in {'y', 'yes'}:
                        cv2.destroyAllWindows()
                        break
                except Exception as e:
                    print(f'Error: {e}. Image {annotation_image_idx} does not exist!')
            
            output_mask_dir = os.path.join(annotation_dir, f"{os.path.basename(image_path).split('.')[0]}.png")
            GrabCutter(image_path, output_mask_dir, 100, 100, scale)
            save_extra_image_copy(image_path, video_dir, n_ant_imgs)

def get_annotation_image_index(image_files, suggester):
    """Get the index of the annotation image."""
    if USER_SELECT:
        return int(input(f'What is preferred annotation image index? ({os.path.basename(image_files[0])}-{os.path.basename(image_files[-1])} possible)\n')) - suggester.manip_start_idx
    return suggester.suggest_frame()

def get_image_scale(annotation_image):
    """Get the scale for the annotation image."""
    cv2.imshow('Annotation Image', annotation_image)
    cv2.moveWindow('Annotation Image', 100, 100)
    cv2.waitKey(20)
    return float(input('What is preferred scale? (e.g., 1, 2, or 0.5)\n'))

def save_extra_image_copy(image_path, video_dir, annotation_frame_num):
    """Save an extra copy of the annotation image for development."""
    print('Saving extra copy of annotation image for development.')
    extra_image_dir = os.path.join(video_dir, 'annotation_imgs')
    os.makedirs(extra_image_dir, exist_ok=True)
    
    extra_image_path = os.path.join(extra_image_dir, f"{annotation_frame_num:02d}_annotation_{os.path.basename(image_path).split('.')[0]}.jpg")
    cv2.imwrite(extra_image_path, cv2.imread(image_path))

def read_annotation_list(text_file):
    """Read the annotation list from a text file."""
    read_list = bn_utils.read_list_file(text_file)
    n_ant = int(read_list[0].split(' ')[0])
    ant_idx = [int(read_list[i * 2 + 1]) for i in range(n_ant)]
    ant_file = [read_list[i * 2 + 2] for i in range(n_ant)]
    return ant_idx, ant_file

def main():
    """Main function to process video annotations."""
    main_dir = os.getcwd()
    data_dir = os.path.join(main_dir, 'data')
    raw_data_dir = os.path.join(data_dir, 'rawData')
    video_list = sorted(next(os.walk(raw_data_dir))[1])
    
    resnet_process_data_dir(raw_data_dir)
    BubbleNets_sort(raw_data_dir, model='BNLF')
    BubbleNets_sort(raw_data_dir, model='BN0')
    color_hist_frame_select(raw_data_dir, annotate_rate=int(10e6))
    
    for video_name in video_list:
        video_dir = os.path.join(raw_data_dir, video_name)
        VideoProcessor(video_dir)
        get_user_annotation(video_dir)
        print(f'Finished with {video_name} annotation.\n\n')
    
    print('\n\nFinished with all annotations!\n\n')

if __name__ == "__main__":
    main()
