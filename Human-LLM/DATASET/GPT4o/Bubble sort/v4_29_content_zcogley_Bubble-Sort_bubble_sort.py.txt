def bubble_sort(lst):
    new_list = lst[:]  # Create a copy of the original list
    is_sorted = False
    
    while not is_sorted:
        is_sorted = True  # Assume the list is sorted
        for idx in range(len(new_list) - 1):
            if new_list[idx] > new_list[idx + 1]:
                # Swap the elements
                new_list[idx], new_list[idx + 1] = new_list[idx + 1], new_list[idx]
                is_sorted = False  # If a swap occurred, the list wasn't sorted
                
    return new_list
