def bubble_sort(results):
    """
    Sorts the results in place using the bubble sort algorithm.
    Sorts primarily by points (descending), then by goal difference (descending), 
    and finally by goals scored (descending) if previous criteria are equal.
    """
    for pass_num in range(len(results) - 1, 0, -1):
        for i in range(pass_num):
            if int(results[i][1]) < int(results[i + 1][1]):
                results[i], results[i + 1] = results[i + 1], results[i]
            elif results[i][1] == results[i + 1][1]:
                goal_diff_i = int(results[i][2]) - int(results[i][3])
                goal_diff_next = int(results[i + 1][2]) - int(results[i + 1][3])
                if goal_diff_i < goal_diff_next:
                    results[i], results[i + 1] = results[i + 1], results[i]
                elif goal_diff_i == goal_diff_next:
                    if int(results[i][2]) < int(results[i + 1][2]):
                        results[i], results[i + 1] = results[i + 1], results[i]

def result_table():
    """
    Reads a table of football match results from a file, sorts them, and prints the sorted table.
    """
    try:
        with open('table1.txt', 'r') as infile:
            table = infile.read()
    except FileNotFoundError:
        print("Error: The file 'table1.txt' was not found.")
        return

    table_list = table.strip().split('\n')
    print('    {0:30} {1:5}   {2:7}  {3:2}'.format('Team', 'Points', 'Diff', 'Goals'))

    results = []
    for line in table_list:
        results.append(line.split(','))

    bubble_sort(results)

    for number, item in enumerate(results, start=1):
        name, points, goals_scored, goals_against = item[0], int(item[1]), int(item[2]), int(item[3])
        diff_goals = goals_scored - goals_against
        print(f'{number:2}. {name:26}  {points:7}  {diff_goals:5}  {goals_scored:5} : {goals_against:2}')

def main():
    print("Results table by gsoa420")
    print()
    result_table()

if __name__ == "__main__":
    main()
