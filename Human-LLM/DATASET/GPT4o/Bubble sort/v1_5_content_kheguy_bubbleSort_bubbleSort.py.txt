import random

class BubbleSort:
    def __init__(self, sizeOfArray, intervalFrom, intervalTo):
        self.arraySize = sizeOfArray
        self.array = [random.randint(intervalFrom, intervalTo) for _ in range(sizeOfArray)]
    
    def __del__(self):
        self.array.clear()
    
    def printArray(self):
        print(self.array)
    
    def sort(self):
        isSorted = False
        while not isSorted:
            isSorted = True
            for i in range(self.arraySize - 1):
                if self.array[i + 1] < self.array[i]:
                    self.array[i], self.array[i + 1] = self.array[i + 1], self.array[i]
                    isSorted = False

def main():
    while True:
        n = int(input('Enter size of array:\n'))
        a = int(input('Enter primary position for random:\n'))
        b = int(input('Enter final position for random:\n'))
        sort = BubbleSort(n, a, b)
        sort.printArray()
        sort.sort()
        sort.printArray()
        del sort

if __name__ == "__main__":
    main()
