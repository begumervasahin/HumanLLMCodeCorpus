from random import randrange

def swap(arr, index_1, index_2):
    """Swaps elements at index_1 and index_2 in the array."""
    arr[index_1], arr[index_2] = arr[index_2], arr[index_1]

def merge(left, right):
    """Merges two sorted lists into one sorted list."""
    result = []
    while left and right:
        if left[0] < right[0]:
            result.append(left.pop(0))
        else:
            result.append(right.pop(0))
    result += left
    result += right
    return result

def bubble_sort(arr):
    """Sorts the array using Bubble Sort algorithm."""
    for _ in range(len(arr)):
        for i in range(len(arr) - 1):
            if arr[i] > arr[i + 1]:
                swap(arr, i, i + 1)

def merge_sort(arr):
    """Sorts the array using Merge Sort algorithm."""
    if len(arr) <= 1:
        return arr
    middle_index = len(arr) // 2
    left_split = arr[:middle_index]
    right_split = arr[middle_index:]
    left_sorted = merge_sort(left_split)
    right_sorted = merge_sort(right_split)
    return merge(left_sorted, right_sorted)

def quick_sort(arr, start, end):
    """Sorts the array using Quick Sort algorithm."""
    if start >= end:
        return arr
    pivot_index = randrange(start, end + 1)
    pivot_element = arr[pivot_index]
    swap(arr, pivot_index, end)
    less_than_pointer = start
    for i in range(start, end):
        if arr[i] < pivot_element:
            swap(arr, i, less_than_pointer)
            less_than_pointer += 1
    swap(arr, less_than_pointer, end)
    quick_sort(arr, start, less_than_pointer - 1)
    quick_sort(arr, less_than_pointer + 1, end)

# Example usage
if __name__ == "__main__":
    array = [randrange(1, 100) for _ in range(10)]
    print("Original array:", array)
    
    sorted_array = array.copy()
    bubble_sort(sorted_array)
    print("Bubble Sorted array:", sorted_array)
    
    sorted_array = merge_sort(array.copy())
    print("Merge Sorted array:", sorted_array)
    
    sorted_array = array.copy()
    quick_sort(sorted_array, 0, len(sorted_array) - 1)
    print("Quick Sorted array:", sorted_array)
