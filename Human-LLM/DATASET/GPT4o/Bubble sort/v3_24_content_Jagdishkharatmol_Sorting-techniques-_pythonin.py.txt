import time

def bubble_sort(lst):
    """Sorts a list using the bubble sort algorithm."""
    n = len(lst)
    for j in range(n - 1, 0, -1):
        for i in range(j):
            if lst[i] > lst[i + 1]:
                lst[i], lst[i + 1] = lst[i + 1], lst[i]

def insertion_sort(lst):
    """Sorts a list using the insertion sort algorithm."""
    for i in range(1, len(lst)):
        current_value = lst[i]
        position = i
        while position > 0 and lst[position - 1] > current_value:
            lst[position] = lst[position - 1]
            position -= 1
        lst[position] = current_value

def merge_sort(lst):
    """Sorts a list using the merge sort algorithm."""
    if len(lst) > 1:
        mid = len(lst) // 2
        left_half = lst[:mid]
        right_half = lst[mid:]

        merge_sort(left_half)
        merge_sort(right_half)

        i = j = k = 0
        while i < len(left_half) and j < len(right_half):
            if left_half[i] < right_half[j]:
                lst[k] = left_half[i]
                i += 1
            else:
                lst[k] = right_half[j]
                j += 1
            k += 1

        while i < len(left_half):
            lst[k] = left_half[i]
            i += 1
            k += 1

        while j < len(right_half):
            lst[k] = right_half[j]
            j += 1
            k += 1

def selection_sort(lst):
    """Sorts a list using the selection sort algorithm."""
    n = len(lst)
    for fill_slot in range(n - 1, 0, -1):
        position_of_max = 0
        for location in range(1, fill_slot + 1):
            if lst[location] > lst[position_of_max]:
                position_of_max = location
        lst[fill_slot], lst[position_of_max] = lst[position_of_max], lst[fill_slot]

def get_sort_choice():
    """Get the user's choice of sorting algorithm."""
    print("Enter \n 1 for Bubble Sort \n 2 for Insertion Sort \n 3 for Merge Sort \n 4 for Selection Sort")
    while True:
        try:
            choice = int(input("\nEnter your choice: "))
            if choice in {1, 2, 3, 4}:
                return choice
            else:
                print("Invalid choice! Please enter a valid option.")
        except ValueError:
            print("Invalid input! Please enter a number.")

def get_numbers():
    """Get the list of numbers to sort from the user."""
    lst = []
    while True:
        try:
            n = int(input("\nHow many numbers do you want to enter: "))
            for _ in range(n):
                num = int(input("Enter the number: "))
                lst.append(num)
            return lst
        except ValueError:
            print("Invalid input! Please enter a number.")

def sort_list(choice, lst):
    """Sort the list based on the user's choice of algorithm."""
    if choice == 1:
        bubble_sort(lst)
    elif choice == 2:
        insertion_sort(lst)
    elif choice == 3:
        merge_sort(lst)
    elif choice == 4:
        selection_sort(lst)
    return lst

def main():
    separator = '_' * 80
    title = "Sort Techniques on Dataset"
    print(f"{separator}\n{title.center(80)}\n{separator}\n")

    while True:
        choice = get_sort_choice()
        lst = get_numbers()

        start = time.time()
        sorted_list = sort_list(choice, lst)
        end = time.time()

        sort_name = ["Bubble Sort", "Insertion Sort", "Merge Sort", "Selection Sort"][choice - 1]
        print(f"\nThe list after {sort_name} is: {sorted_list}")
        print(f"The time taken by the sorting process: {end - start:.6f} seconds")

        try:
            continue_choice = int(input("\nEnter 1 to continue and 0 to exit: "))
            if continue_choice == 0:
                break
        except ValueError:
            print("Invalid input! Exiting the program.")
            break

if __name__ == "__main__":
    main()
