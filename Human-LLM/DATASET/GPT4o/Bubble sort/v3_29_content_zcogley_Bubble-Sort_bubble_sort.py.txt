def bubble_sort(lst):
    """
    Sorts a list using the bubble sort algorithm.

    Args:
    lst (list): The list to be sorted.

    Returns:
    list: A new sorted list.
    """
    # Create a copy of the list to avoid modifying the original list
    sorted_list = lst[:]
    n = len(sorted_list)
    is_sorted = False  # Flag to keep track of whether the list is sorted

    while not is_sorted:
        is_sorted = True  # Assume the list is sorted
        # Loop through the list up to the second-to-last element
        for idx in range(n - 1):
            if sorted_list[idx] > sorted_list[idx + 1]:
                # Swap the elements if they are in the wrong order
                sorted_list[idx], sorted_list[idx + 1] = sorted_list[idx + 1], sorted_list[idx]
                is_sorted = False  # If we had to swap, then it's not sorted yet

    return sorted_list

# Example usage
if __name__ == "__main__":
    lst = [64, 25, 12, 22, 11]
    sorted_lst = bubble_sort(lst)
    print("Sorted list:", sorted_lst)
