class SelectionSort:
    def __init__(self, lst, trace_mode):
        """
        Initializes the SelectionSort object.
        
        Args:
            lst (list): The list to be sorted.
            trace_mode (bool): Flag to enable or disable step-by-step tracing.
        """
        self.lst = lst
        self.trace_mode = trace_mode

    def sort(self):
        """
        Sorts the list using the Selection Sort algorithm.
        
        Returns:
            list: The sorted list.
        """
        start_idx = 0
        lst = self.lst
        while start_idx < len(lst):
            min_val = lst[start_idx]
            min_idx = start_idx
            for i, elm in enumerate(lst[start_idx:]):
                if elm < min_val:
                    min_val = elm
                    min_idx = i + start_idx
            lst[start_idx], lst[min_idx] = lst[min_idx], lst[start_idx]
            if self.trace_mode:
                self.display_list(lst, start_idx, min_idx)
            start_idx += 1
        return lst

    def display_list(self, lst, i, j):
        """
        Displays the list with the current state of the sorting process.
        
        Args:
            lst (list): The current state of the list.
            i (int): The index of the current element being compared.
            j (int): The index of the minimum element found.
        """
        lst = [str(k) for k in lst]
        output = '|'
        for k, elm in enumerate(lst):
            if k == i or k == j:
                elm += '*'
            output += f' {elm} |'
        print(output)
        input("Press Enter to continue...")

def main():
    """
    Main function to demonstrate the Selection Sort algorithm with optional tracing.
    """
    trace_mode = True  # Set to False to disable step-by-step tracing
    lst = [64, 25, 12, 22, 11]
    sorter = SelectionSort(lst, trace_mode)
    sorted_list = sorter.sort()
    print('Sorted List:', sorted_list)

if __name__ == "__main__":
    main()
