import random
import pickle
import time
import matplotlib.pyplot as plt

def generate_random_array(size, range_start, range_end):
    return random.sample(range(range_start, range_end), size)

def save_array_to_file(array, filename):
    with open(filename, 'wb') as file:
        pickle.dump(array, file)

def load_array_from_file(filename):
    with open(filename, 'rb') as file:
        return pickle.load(file)

def bubble_sort(arr):
    n = len(arr)
    for i in range(n):
        for j in range(n - 1):
            if arr[j] > arr[j + 1]:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]

def insertion_sort(arr):
    n = len(arr)
    for j in range(1, n):
        key = arr[j]
        i = j - 1
        while i >= 0 and arr[i] > key:
            arr[i + 1] = arr[i]
            i -= 1
        arr[i + 1] = key

def measure_sorting_time(sort_function, array):
    start_time = time.time()
    sort_function(array)
    end_time = time.time()
    return end_time - start_time

def main():
    input_sizes = [2000 * i for i in range(1, 26)]
    avg_time_bubble_sort = []
    avg_time_insertion_sort = []

    original_array = generate_random_array(200000, 1, 300000)
    save_array_to_file(original_array, "data.txt")

    array = load_array_from_file("data.txt")

    for size in input_sizes:
        bubble_sort_times = []
        insertion_sort_times = []

        for _ in range(10):
            array_sample_bubble = random.sample(array, size)
            array_sample_insertion = random.sample(array, size)

            bubble_sort_time = measure_sorting_time(bubble_sort, array_sample_bubble)
            bubble_sort_times.append(bubble_sort_time)

            insertion_sort_time = measure_sorting_time(insertion_sort, array_sample_insertion)
            insertion_sort_times.append(insertion_sort_time)

        avg_time_bubble_sort.append(sum(bubble_sort_times) / 10)
        avg_time_insertion_sort.append(sum(insertion_sort_times) / 10)

        print(f"Average bubble sort time for size {size}: {avg_time_bubble_sort[-1]}")
        print(f"Average insertion sort time for size {size}: {avg_time_insertion_sort[-1]}")

    plt.plot(input_sizes, avg_time_bubble_sort, 'r--', label='Bubble Sort')
    plt.plot(input_sizes, avg_time_insertion_sort, 'b--', label='Insertion Sort')
    plt.xlabel('Input size')
    plt.ylabel('Average time (s)')
    plt.title('Sorting Algorithm Performance')
    plt.legend()
    plt.show()

if __name__ == "__main__":
    main()
