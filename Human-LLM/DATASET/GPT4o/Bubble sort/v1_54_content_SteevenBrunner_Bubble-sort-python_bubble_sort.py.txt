import random
from time import time

# Set the maximum range for generating random numbers
max_range = 10001

# Generate an unsorted list of random numbers
unsorted_list = random.sample(range(1, max_range), max_range - 1)

def bubble_sort(input_list):
    """
    Sorts a list using the bubble sort algorithm.

    Parameters:
    input_list (list): The list to be sorted.

    Returns:
    list: The sorted list.
    """
    is_sorted = False
    while not is_sorted:
        is_sorted = True
        for i in range(len(input_list) - 1):
            if input_list[i] > input_list[i + 1]:
                input_list[i], input_list[i + 1] = input_list[i + 1], input_list[i]
                is_sorted = False
    return input_list

# Copy the unsorted list for bubble sort
unsorted_list_cpy = unsorted_list[:]

# Measure time taken by bubble sort
start = time()
bubble_sort(unsorted_list_cpy)
print("Bubble sort algorithm ends in {:.3f} seconds to sort a list of {} items.".format(time() - start, max_range - 1))

# Copy the unsorted list for built-in sort
unsorted_list_cpy = unsorted_list[:]

# Measure time taken by Python's built-in sort() method (Timsort)
start = time()
unsorted_list_cpy.sort()
print("System Python sort() Timsort algorithm ends in {:.3f} seconds to sort a list of {} items.".format(time() - start, max_range - 1))
