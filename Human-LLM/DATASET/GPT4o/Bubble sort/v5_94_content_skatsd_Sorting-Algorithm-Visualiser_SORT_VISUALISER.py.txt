import random
import time
import matplotlib.pyplot as plt
import matplotlib.animation as animation

def swap(A, i, j):
    """Swap elements at indices i and j in array A."""
    if i != j:
        A[i], A[j] = A[j], A[i]

def bubble_sort(A):
    """Generate states for Bubble Sort."""
    if len(A) == 1:
        return
    for i in range(len(A) - 1):
        swapped = False
        for j in range(len(A) - 1 - i):
            if A[j] > A[j + 1]:
                swap(A, j, j + 1)
                swapped = True
            yield A
        if not swapped:
            break

def insertion_sort(A):
    """Generate states for Insertion Sort."""
    for i in range(1, len(A)):
        j = i
        while j > 0 and A[j] < A[j - 1]:
            swap(A, j, j - 1)
            j -= 1
            yield A

def merge_sort(A, start, end):
    """Generate states for Merge Sort."""
    if end <= start:
        return
    mid = start + (end - start) // 2
    yield from merge_sort(A, start, mid)
    yield from merge_sort(A, mid + 1, end)
    yield from merge(A, start, mid, end)
    yield A

def merge(A, start, mid, end):
    """Merge helper function for Merge Sort."""
    merged = []
    left_idx, right_idx = start, mid + 1
    while left_idx <= mid and right_idx <= end:
        if A[left_idx] < A[right_idx]:
            merged.append(A[left_idx])
            left_idx += 1
        else:
            merged.append(A[right_idx])
            right_idx += 1
    while left_idx <= mid:
        merged.append(A[left_idx])
        left_idx += 1
    while right_idx <= end:
        merged.append(A[right_idx])
        right_idx += 1
    for i, sorted_val in enumerate(merged):
        A[start + i] = sorted_val
        yield A

def quick_sort(A, start, end):
    """Generate states for Quick Sort."""
    if start >= end:
        return
    pivot = A[end]
    pivot_idx = start
    for i in range(start, end):
        if A[i] < pivot:
            swap(A, i, pivot_idx)
            pivot_idx += 1
        yield A
    swap(A, end, pivot_idx)
    yield A
    yield from quick_sort(A, start, pivot_idx - 1)
    yield from quick_sort(A, pivot_idx + 1, end)
