def bubble_sort(lst):
    # Create a copy of the original list to avoid modifying it
    sorted_list = lst[:]
    
    # Flag to indicate if the list is sorted
    is_sorted = False
    
    while not is_sorted:
        is_sorted = True  # Assume the list is sorted initially
        
        for idx in range(len(sorted_list) - 1):
            # Compare adjacent elements
            if sorted_list[idx] > sorted_list[idx + 1]:
                # Swap the elements if they are in the wrong order
                sorted_list[idx], sorted_list[idx + 1] = sorted_list[idx + 1], sorted_list[idx]
                is_sorted = False  # A swap means the list wasn't fully sorted yet
                
    return sorted_list
