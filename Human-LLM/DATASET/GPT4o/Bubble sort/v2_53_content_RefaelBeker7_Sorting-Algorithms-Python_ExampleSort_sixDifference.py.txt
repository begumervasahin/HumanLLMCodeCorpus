import random
import time

def check_different_time(num1, num2):
    """
    Returns the percentage difference between two numbers.
    """
    if num1 > num2:
        return (num2 / num1) * 100
    else:
        return (num1 / num2) * 100

def print_array(arr, name):
    """
    Prints the sorted array with a label.
    
    Parameters:
    arr (list): The sorted array.
    name (str): The name of the sorting algorithm used.
    """
    print(f'Sorted array from {name} Sort is:')
    for num in arr:
        print(num, end=' ')
    print()

def check_time(func, arr):
    """
    Measures the time taken by a sorting function to sort an array.
    
    Parameters:
    func (function): The sorting function.
    arr (list): The array to be sorted.
    
    Returns:
    float: The time taken to sort the array.
    """
    start_time = time.time()
    func(arr)
    elapsed_time = time.time() - start_time
    print(f'Function [{func.__name__}] finished in {elapsed_time:.3f} sec')
    return elapsed_time

def heapify(arr, n, i):
    """
    Heapifies a subtree rooted with node i which is an index in arr[].
    
    Parameters:
    arr (list): The array to be heapified.
    n (int): The size of the heap.
    i (int): The index of the root element of the current subtree.
    """
    largest = i
    left = 2 * i + 1
    right = 2 * i + 2

    if left < n and arr[i] < arr[left]:
        largest = left

    if right < n and arr[largest] < arr[right]:
        largest = right

    if largest != i:
        arr[i], arr[largest] = arr[largest], arr[i]
        heapify(arr, n, largest)

def heap_sort(arr):
    """
    Sorts an array using the heap sort algorithm.
    
    Parameters:
    arr (list): The array to be sorted.
    """
    n = len(arr)
    for i in range(n // 2 - 1, -1, -1):
        heapify(arr, n, i)
    for i in range(n - 1, 0, -1):
        arr[i], arr[0] = arr[0], arr[i]
        heapify(arr, i, 0)

def merge_sort(arr):
    """
    Sorts an array using the merge sort algorithm.
    
    Parameters:
    arr (list): The array to be sorted.
    """
    if len(arr) > 1:
        mid = len(arr) // 2
        left_half = arr[:mid]
        right_half = arr[mid:]

        merge_sort(left_half)
        merge_sort(right_half)

        i = j = k = 0
        while i < len(left_half) and j < len(right_half):
            if left_half[i] < right_half[j]:
                arr[k] = left_half[i]
                i += 1
            else:
                arr[k] = right_half[j]
                j += 1
            k += 1

        while i < len(left_half):
            arr[k] = left_half[i]
            i += 1
            k += 1

        while j < len(right_half):
            arr[k] = right_half[j]
            j += 1
            k += 1

def partition(arr, low, high):
    """
    Partitions the array for quick sort.
    
    Parameters:
    arr (list): The array to be partitioned.
    low (int): The starting index.
    high (int): The ending index.
    
    Returns:
    int: The partition index.
    """
    pivot = arr[high]
    i = low - 1
    for j in range(low, high):
        if arr[j] <= pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    return i + 1

def quick_sort_helper(arr, low, high):
    """
    Helper function for the quick sort algorithm.
    
    Parameters:
    arr (list): The array to be sorted.
    low (int): The starting index.
    high (int): The ending index.
    """
    if low < high:
        pi = partition(arr, low, high)
        quick_sort_helper(arr, low, pi - 1)
        quick_sort_helper(arr, pi + 1, high)

def quick_sort(arr):
    """
    Sorts an array using the quick sort algorithm.
    
    Parameters:
    arr (list): The array to be sorted.
    """
    quick_sort_helper(arr, 0, len(arr) - 1)

def selection_sort(arr):
    """
    Sorts an array using the selection sort algorithm.
    
    Parameters:
    arr (list): The array to be sorted.
    """
    for i in range(len(arr)):
        min_index = i
        for j in range(i + 1, len(arr)):
            if arr[min_index] > arr[j]:
                min_index = j
        arr[i], arr[min_index] = arr[min_index], arr[i]

def insertion_sort(arr):
    """
    Sorts an array using the insertion sort algorithm.
    
    Parameters:
    arr (list): The array to be sorted.
    """
    for i in range(1, len(arr)):
        key = arr[i]
        j = i - 1
        while j >= 0 and key < arr[j]:
            arr[j + 1] = arr[j]
            j -= 1
        arr[j + 1] = key

def bubble_sort(arr):
    """
    Sorts an array using the bubble sort algorithm.
    
    Parameters:
    arr (list): The array to be sorted.
    """
    n = len(arr)
    for i in range(n):
        for j in range(0, n - i - 1):
            if arr[j] > arr[j + 1]:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]

def main():
    # Generate random arrays
    array_size = 10000
    arrays = {
        'Bubble': [random.randint(1, 500) for _ in range(array_size)],
        'Insertion': [random.randint(1, 500) for _ in range(array_size)],
        'Selection': [random.randint(1, 500) for _ in range(array_size)],
        'Quick': [random.randint(1, 5000) for _ in range(array_size)],
        'Merge': [random.randint(1, 5000) for _ in range(array_size)],
        'Heap': [random.randint(1, 5000) for _ in range(array_size)]
    }

    # Print initial arrays
    for name, arr in arrays.items():
        print(f"----- Array Random --- {name} sort: ---- ")
        print(*arr)
    print("")

    # Sort arrays and measure time
    times = {}
    for name, arr in arrays.items():
        arr_copy = arr.copy()
        start_time = time.time()
        if name == 'Bubble':
            bubble_sort(arr_copy)
        elif name == 'Insertion':
            insertion_sort(arr_copy)
        elif name == 'Selection':
            selection_sort(arr_copy)
        elif name == 'Quick':
            quick_sort(arr_copy)
        elif name == 'Merge':
            merge_sort(arr_copy)
        elif name == 'Heap':
            heap_sort(arr_copy)
        elapsed_time = time.time() - start_time
        times[name] = elapsed_time
        print(f'Function [{name} Sort] finished in {elapsed_time:.3f} sec')

    # Print sorted arrays
    for name, arr in arrays.items():
        print_array(arr, name)

if __name__ == "__main__":
    main()
