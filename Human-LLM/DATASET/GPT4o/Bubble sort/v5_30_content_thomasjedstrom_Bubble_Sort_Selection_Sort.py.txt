import random
import datetime

def populate_list(size=100, upper_limit=10000):
    """
    Function to populate a list with random numbers.
    
    Args:
        size (int): Number of random numbers to generate.
        upper_limit (int): The upper limit for the random numbers.
    
    Returns:
        list: A list of random numbers.
    """
    lst = [random.randint(0, upper_limit) for _ in range(size)]
    print('\nThis is your Sample List')
    print(lst)
    return lst

def bubble_sort(lst):
    """
    Function to perform Bubble Sort on a list.
    
    Args:
        lst (list): The list to be sorted.
    
    Returns:
        list: The sorted list.
    """
    start_time = datetime.datetime.now()
    n = len(lst)
    
    for i in range(n):
        for j in range(0, n - i - 1):
            if lst[j] > lst[j + 1]:
                lst[j], lst[j + 1] = lst[j + 1], lst[j]
                
    end_time = datetime.datetime.now()
    print('\nThis is your Bubble Sort')
    print(lst)
    print(f'This operation took {end_time - start_time} seconds to complete')
    return lst

def selection_sort(lst):
    """
    Function to perform Selection Sort on a list.
    
    Args:
        lst (list): The list to be sorted.
    
    Returns:
        list: The sorted list.
    """
    start_time = datetime.datetime.now()
    n = len(lst)
    
    for i in range(n):
        min_idx = i
        for j in range(i + 1, n):
            if lst[j] < lst[min_idx]:
                min_idx = j
        lst[i], lst[min_idx] = lst[min_idx], lst[i]
                
    end_time = datetime.datetime.now()
    print('\nThis is your Selection Sort')
    print(lst)
    print(f'This operation took {end_time - start_time} seconds to complete')
    return lst

# Main logic
sample_list = populate_list()

# Make copies of the sample list for sorting
bubble_sort_list = sample_list[:]
selection_sort_list = sample_list[:]

# Perform sorting
bubble_sort(bubble_sort_list)
selection_sort(selection_sort_list)
