def bubble_sort(nums):
    """
    Sorts a list of numbers using the bubble sort algorithm.

    Args:
        nums (list): The list of numbers to be sorted.
    """
    n = len(nums)
    for i in range(n - 1):
        # Track if any swaps are made during the pass
        swapped = False
        for j in range(n - 1 - i):
            if nums[j] > nums[j + 1]:
                # Swap the elements
                nums[j], nums[j + 1] = nums[j + 1], nums[j]
                swapped = True
        # Print the list after each pass
        print(f"After pass {i + 1}: {nums}")
        # If no swaps are made, the list is already sorted
        if not swapped:
            break

if __name__ == "__main__":
    nums = [5, 3, 8, 6, 7, 2]
    print("Unsorted List:", nums)
    bubble_sort(nums)
    print("Sorted List:", nums)
