list1 = [6, 3, 2, 89, 10, 2.3, 5, 80.2]

def insertion_sort(lst):
    """
    Sorts a list using the insertion sort algorithm.
    """
    for i in range(1, len(lst)):
        element = lst[i]
        j = i
        while j > 0 and lst[j - 1] > element:
            lst[j] = lst[j - 1]
            j -= 1
        lst[j] = element
    return lst

def bubble_sort(lst):
    """
    Sorts a list using the bubble sort algorithm.
    """
    permutation = True
    passage = 0
    while permutation:
        permutation = False
        passage += 1
        for element in range(len(lst) - passage):
            if lst[element] > lst[element + 1]:
                permutation = True
                lst[element], lst[element + 1] = lst[element + 1], lst[element]
    return lst

def shaker_sort(lst):
    """
    Sorts a list using the shaker sort algorithm.
    """
    permutation = True
    way = 1
    element = 0
    start, end = 0, len(lst) - 2
    while permutation:
        permutation = False
        while (element < end and way == 1) or (element > start and way == -1):
            if lst[element] > lst[element + 1]:
                permutation = True
                lst[element], lst[element + 1] = lst[element + 1], lst[element]
            element += way
        if way == 1:
            end -= 1
        else:
            start += 1
        way = -way
    return lst

def gnome_sort(lst):
    """
    Sorts a list using the gnome sort algorithm.
    """
    i_b, i_i = 1, 2
    size = len(lst)
    while i_b < size:
        if lst[i_b - 1] <= lst[i_b]:
            i_b, i_i = i_i, i_i + 1
        else:
            lst[i_b - 1], lst[i_b] = lst[i_b], lst[i_b - 1]
            i_b -= 1
            if i_b == 0:
                i_b, i_i = i_i, i_i + 1
    return lst

def merge(left, right):
    """
    Merges two sorted lists.
    """
    result = []
    left_index, right_index = 0, 0
    while left_index < len(left) and right_index < len(right):
        if left[left_index] <= right[right_index]:
            result.append(left[left_index])
            left_index += 1
        else:
            result.append(right[right_index])
            right_index += 1
    result.extend(left[left_index:])
    result.extend(right[right_index:])
    return result

def merge_sort(lst):
    """
    Sorts a list using the merge sort algorithm.
    """
    if len(lst) <= 1:
        return lst
    middle = len(lst) // 2
    left = lst[:middle]
    right = lst[middle:]
    return merge(merge_sort(left), merge_sort(right))

def selection_sort(lst):
    """
    Sorts a list using the selection sort algorithm.
    """
    nb = len(lst)
    for element in range(nb):
        min_index = element
        for j in range(element + 1, nb):
            if lst[j] < lst[min_index]:
                min_index = j
        if min_index != element:
            lst[element], lst[min_index] = lst[min_index], lst[element]
    return lst

print("Gnome Sort:", gnome_sort(list1.copy()))
print("Merge Sort:", merge_sort(list1.copy()))
print("Selection Sort:", selection_sort(list1.copy()))
