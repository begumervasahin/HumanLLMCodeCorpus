class Node:
    def __init__(self, data):
        """Initialize a new node with the given data."""
        self.left = None
        self.right = None
        self.data = data

    def insert(self, data):
        """Insert a new node with the given data into the tree."""
        if data < self.data:
            if self.left is None:
                self.left = Node(data)
            else:
                self.left.insert(data)
        elif data > self.data:
            if self.right is None:
                self.right = Node(data)
            else:
                self.right.insert(data)

    def find_value(self, value):
        """Find a value in the tree and return a message if it's not found."""
        if value < self.data:
            if self.left is None:
                return f"{value} Not Found"
            return self.left.find_value(value)
        elif value > self.data:
            if self.right is None:
                return f"{value} Not Found"
            return self.right.find_value(value)
        else:
            return f"{self.data} is found"

    def print_tree(self):
        """Print the tree in order."""
        if self.left:
            self.left.print_tree()
        print(self.data, end=' ')
        if self.right:
            self.right.print_tree()

if __name__ == "__main__":
    root = Node(14)
    root.insert(6)
    root.insert(18)
    root.insert(3)

    print(root.find_value(7))  # Output: "7 Not Found"
    print(root.find_value(14)) # Output: "14 is found"
    root.print_tree()          # Output: 3 6 14 18

def draw_clouds(x, y, diameter, widths=5, lengths=2, transparency=100):
    """Draw the clouds."""
    fill(250, 250, 250, transparency if diameter < 60 else 150)
    noStroke()
    for m in range(widths):
        for n in range(lengths):
            ellipse(x + 25 * m, y + 25 * n, diameter, diameter)

def set_gradient(x, y, w, h, c1, c2, axis):
    """Draw a color gradient."""
    noFill()
    if axis == "Y_AXIS":
        for i in range(y, y + h + 1):
            inter = map(i, y, y + h, 0, 1)
            c = lerpColor(c1, c2, inter)
            stroke(c)
            line(x, i, x + w, i)
    elif axis == "X_AXIS":
        for i in range(x, x + w + 1):
            inter = map(i, x, x + w, 0, 1)
            c = lerpColor(c1, c2, inter)
            stroke(c)
            line(i, y, i, y + h)

class TextBox:
    """A text box class."""

    def __init__(self, rect_params, rect_fill, text_fill, text_size, text, selected_stroke, default_stroke):
        """Initialize a text box."""
        self.x, self.y, self.width, self.height = rect_params
        self.rect_fill = rect_fill
        self.text_fill = text_fill
        self.text_size = text_size
        self.text = text
        self.selected_stroke = selected_stroke
        self.default_stroke = default_stroke
        self.selected = False

    def draw_box(self):
        """Draw the text box."""
        transparency = self.rect_fill[3] if len(self.rect_fill) == 4 else 200
        fill(self.rect_fill[0], self.rect_fill[1], self.rect_fill[2], transparency)
        stroke(self.selected_stroke if self.selected else self.default_stroke)
        rect(self.x, self.y, self.width, self.height)
        fill(self.text_fill)
        textSize(self.text_size)
        text(self.text, self.x + 5, self.y + 25)
