import time
import configuraciones

def shell_sort(arr):
    """
    Sorts the array using the Shell Sort algorithm.
    """
    sublist_count = len(arr) // 2
    while sublist_count > 0:
        for start_position in range(sublist_count):
            gap_insertion_sort(arr, start_position, sublist_count)
        print(f"After increments of size {sublist_count}, the list is {arr}")
        sublist_count = sublist_count // 2

def gap_insertion_sort(arr, start, gap):
    """
    Performs a gapped insertion sort for the Shell Sort algorithm.
    """
    for i in range(start + gap, len(arr), gap):
        current_value = arr[i]
        position = i

        while position >= gap and arr[position - gap] > current_value:
            arr[position] = arr[position - gap]
            position -= gap

        arr[position] = current_value

def sort_and_log(configuration_func, config_name, output_file, time_file):
    """
    Sorts arrays using Shell Sort, logs the sorted arrays and the time taken.
    """
    arrays = configuration_func('ShellSort')
    total_time = 0

    with open(output_file, 'a') as file_output, open(time_file, 'a') as file_time:
        for sublist in arrays:
            start_time = time.time()
            shell_sort(sublist)
            elapsed_time = time.time() - start_time
            file_output.write(f"{sublist}\n")
            total_time += elapsed_time
        
        file_time.write(f'{config_name}: {total_time}\n')

if __name__ == "__main__":
    configurations = [
        (configuraciones.confA, 'ConfA', 'ordenadoAShellSort.txt'),
        (configuraciones.confB, 'ConfB', 'ordenadoBShellSort.txt'),
        (configuraciones.confC, 'ConfC', 'ordenadoCShellSort.txt'),
        (configuraciones.confD, 'ConfD', 'ordenadoDShellSort.txt'),
        (configuraciones.confE, 'ConfE', 'ordenadoEShellSort.txt'),
        (configuraciones.confF, 'ConfF', 'ordenadoFShellSort.txt')
    ]
    
    for config_func, config_name, output_file in configurations:
        sort_and_log(config_func, config_name, output_file, 'tiempoShell.txt')
