import datetime
import statistics

def quick_sort(arr):
    """
    Sort an array using the Quick Sort algorithm with a median-of-three pivot selection.

    Args:
    arr (list): The list of elements to be sorted.

    Returns:
    list: The sorted list.
    """
    # Base case: if the array has 0 or 1 elements, it is already sorted
    if len(arr) <= 1:
        return arr

    # Median-of-three pivot selection
    # Select the first, middle, and last elements and find their median
    items = [arr[0], arr[len(arr) // 2], arr[-1]]
    pivot = statistics.median(items)

    # Partition the array into three parts:
    # Elements less than the pivot, elements equal to the pivot, and elements greater than the pivot
    left = [x for x in arr if x < pivot]
    middle = [x for x in arr if x == pivot]
    right = [x for x in arr if x > pivot]

    # Recursively sort the left and right partitions and concatenate the results
    return quick_sort(left) + middle + quick_sort(right)

if __name__ == "__main__":
    # Test array to be sorted
    test_array = [21, 4, 1, 3, 9, 20, 25, 6, 21, 14]

    # Measure the start time
    start_time = datetime.datetime.now()
    
    # Sort the array using quick_sort
    sorted_array = quick_sort(test_array)
    
    # Measure the finish time
    finish_time = datetime.datetime.now()

    # Print the sorted array and the time taken to sort it
    print("Sorted array:", sorted_array)
    print("Time taken:", finish_time - start_time)
