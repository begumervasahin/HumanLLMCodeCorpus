def quick_sort(arr, left, right):
    """
    Sorts an array in place using the quicksort algorithm.

    Args:
        arr (list): The list of elements to be sorted.
        left (int): The starting index of the segment to be sorted.
        right (int): The ending index of the segment to be sorted.
    """
    if left < right:
        pivot_index = partition(arr, left, right)
        quick_sort(arr, left, pivot_index - 1)
        quick_sort(arr, pivot_index + 1, right)

def partition(arr, left, right):
    """
    Partitions the array segment into elements less than the pivot and elements greater than or equal to the pivot.

    Args:
        arr (list): The list of elements to be partitioned.
        left (int): The starting index of the segment to be partitioned.
        right (int): The ending index of the segment to be partitioned.

    Returns:
        int: The index of the pivot element after partitioning.
    """
    pivot = arr[left]
    store_index = left + 1
    
    for i in range(left + 1, right + 1):
        if arr[i] < pivot:
            arr[store_index], arr[i] = arr[i], arr[store_index]
            store_index += 1
    
    store_index -= 1
    arr[left], arr[store_index] = arr[store_index], arr[left]
    
    return store_index

# Example usage
if __name__ == "__main__":
    array = [24, 3, 45, 29, 37, 12, 4]
    print("Original array:", array)
    quick_sort(array, 0, len(array) - 1)
    print("Sorted array:", array)
