import argparse
import random
import sys
import pivots
from comparisons import Comparisons

def setup_argument_parser():
    """
    Set up the argument parser for the script.
    
    Returns:
        argparse.ArgumentParser: Configured argument parser.
    """
    parser = argparse.ArgumentParser(
        description='Comparisons client.',
        formatter_class=argparse.RawTextHelpFormatter
    )

    group = parser.add_mutually_exclusive_group()
    group.add_argument(
        '-b', '--bound',
        type=int,
        help='Upper bound number'
    )
    group.add_argument(
        'infile',
        nargs='?',
        type=argparse.FileType('r'),
        default=sys.stdin,
        help='Input file with numbers (default stdin)'
    )

    parser.add_argument(
        '-p', '--pivot',
        choices=pivots_choice.keys(),
        metavar='PIVOT',
        default='first',
        help='first: use first element as a pivot (default)\n'
             'last: use last element as a pivot\n'
             'median: use median of three elements as a pivot'
    )

    return parser

def get_numbers_from_args(args):
    """
    Generate or read the list of numbers based on the provided arguments.
    
    Args:
        args: Parsed command line arguments.

    Returns:
        list: List of numbers to be sorted.
    """
    if args.bound is not None:
        nums = list(range(1, args.bound + 1))
        random.shuffle(nums)
    else:
        nums = [int(num) for num in args.infile]

    return nums

def main():
    """
    Main function to run the comparisons client.
    """
    parser = setup_argument_parser()
    args = parser.parse_args()

    nums = get_numbers_from_args(args)

    comp = Comparisons(nums, pivots_choice[args.pivot])

    print(comp.count())

if __name__ == "__main__":
    pivots_choice = {
        'first': pivots.first_index,
        'last': pivots.last_index,
        'median': pivots.median_of_three_index
    }
    main()
