import matplotlib.pyplot as plt
import numpy as np
import time

# Global variables for tracking counts and times
values = []
times = []
counts = []
timesAve = np.zeros(5)
count = 0

def merge_sort(arr):
    """
    Perform a merge sort on the given list.

    Args:
    arr (list): The list to be sorted.
    """
    global count
    if len(arr) > 1:
        mid = len(arr) // 2
        left_half = arr[:mid]
        right_half = arr[mid:]

        merge_sort(left_half)
        merge_sort(right_half)

        i = j = k = 0

        while i < len(left_half) and j < len(right_half):
            if left_half[i] < right_half[j]:
                arr[k] = left_half[i]
                i += 1
            else:
                arr[k] = right_half[j]
                j += 1
            k += 1
            count += 1

        while i < len(left_half):
            arr[k] = left_half[i]
            i += 1
            k += 1
            count += 1

        while j < len(right_half):
            arr[k] = right_half[j]
            j += 1
            k += 1
            count += 1

def merge():
    """
    Read data from files, sort the data using merge sort, and track the performance.
    """
    import createDataSets as cds
    global count
    for file_name in cds.fileNames:
        values.clear()
        start = time.time()
        with open(file_name, "r") as reader:
            values.extend(int(value) for value in reader.readlines())
        
        count = 0
        merge_sort(values)
        counts.append(count)
        end = time.time()
        times.append(end - start)
    
    cds.generateAverageValues(times, timesAve)

def generate_cases():
    """
    Generate and plot the best, average, and worst case scenarios for the merge sort algorithm.
    """
    global counts
    best_cases = []
    average_cases = []
    worst_cases = []

    for i in range(len(counts)):
        if i % 9 in {0, 4}:
            best_cases.append(counts[i])

    best_cases_avg = [(best_cases[2 * i] + best_cases[2 * i + 1]) / 2 for i in range(len(best_cases) // 2)]

    for i in range(len(counts)):
        if i % 9 == 5:
            worst_cases.append(counts[i])

    for i in range(len(counts)):
        if i % 9 in {1, 2, 3}:
            average_cases.append(counts[i])

    average_cases_avg = [(average_cases[3 * i] + average_cases[3 * i + 1] + average_cases[3 * i + 2]) / 3 for i in range(len(average_cases) // 3)]

    sizes = [200, 400, 600, 800, 1000]
    plt.plot(sizes, average_cases_avg, '-bo', label="Average Case")
    plt.plot(sizes, best_cases_avg, '-ro', label="Best Case")
    plt.plot(sizes, worst_cases, '-go', label="Worst Case")
    plt.legend()
    plt.grid()
    plt.xlabel("Sizes")
    plt.ylabel("Counts")
    plt.title("Merge Sort Performance")
    plt.show()

# Example usage:
if __name__ == "__main__":
    merge()
    generate_cases()
