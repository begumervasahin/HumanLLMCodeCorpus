import random

def partition(arr, low, high):
    """
    Partition the array around a pivot element chosen randomly.

    This function rearranges the elements in the array such that elements less than or equal to the pivot
    are on its left, and elements greater than the pivot are on its right.

    Parameters:
    arr (list): The list of elements to be sorted.
    low (int): The starting index of the array slice.
    high (int): The ending index of the array slice.

    Returns:
    int: The index position of the pivot element after partition.
    """
    # Choose a random pivot index and move the pivot element to the end
    pivot_index = random.randint(low, high)
    pivot = arr[pivot_index]
    arr[pivot_index], arr[high] = arr[high], arr[pivot_index]
    
    # Partitioning logic
    pindex = low
    for j in range(low, high):
        if arr[j] <= pivot:
            arr[pindex], arr[j] = arr[j], arr[pindex]
            pindex += 1

    # Place the pivot element in its correct position
    arr[pindex], arr[high] = arr[high], arr[pindex]
    return pindex

def quick_sort(arr, low, high):
    """
    Perform QuickSort on the array.

    This function uses the QuickSort algorithm to sort the elements of the array in place.

    Parameters:
    arr (list): The list of elements to be sorted.
    low (int): The starting index of the array slice.
    high (int): The ending index of the array slice.
    """
    if low < high:
        pi = partition(arr, low, high)
        quick_sort(arr, low, pi - 1)
        quick_sort(arr, pi + 1, high)

if __name__ == '__main__':
    # Sample array to be sorted
    sample_array = [10, 7, 8, 9, 1, 5]
    n = len(sample_array)
    
    # Perform QuickSort
    quick_sort(sample_array, 0, n - 1)
