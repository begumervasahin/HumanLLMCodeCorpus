# Initialize an empty list to store TV show information
tv_list = []

# Open the unsorted TV show list file for reading
with open("tvshow_list_unsorted.txt", "r") as myfile:
    for line in myfile:
        # Split each line into components based on the underscore character
        temp_split = line.split('_')
        # Append a tuple (show_name, show_year) to the tv_list
        tv_list.append((temp_split[0], temp_split[1].strip('\n')))

# Sort the tv_list by year first and then by name
tv_list.sort(key=lambda x: (x[1], x[0]))

# Open the output file for writing the sorted list
with open('tvshow_list_sorted.txt', 'w') as sorted_file:
    for item in tv_list:
        # Write each sorted item to the file
        sorted_file.write(f"{item[0]}_{item[1]}\n")
