from rasa_nlu.training_data import load_data
from rasa_nlu.config import RasaNLUModelConfig
from rasa_nlu.model import Trainer, Interpreter
from rasa_nlu import config
from rasa_nlu.components import ComponentBuilder

# Enable component caching
builder = ComponentBuilder(use_cache=True)

def train_nlu(data_path, config_path, model_dir):
    """
    Train the NLU model with the given data and configuration file, and save it to the specified directory.
    
    :param data_path: Path to the training data.
    :param config_path: Path to the configuration file.
    :param model_dir: Directory where the model should be saved.
    """
    # Load training data
    training_data = load_data(data_path)
    
    # Load configuration
    nlu_config = config.load(config_path)
    
    # Initialize trainer with configuration and component builder
    trainer = Trainer(nlu_config, builder)
    
    # Train the model
    trainer.train(training_data)
    
    # Save the trained model to the specified directory
    trainer.persist(model_dir, fixed_model_name='restaurantnlu')

def run_nlu():
    """
    Load the trained NLU model and parse a sample text input.
    """
    # Load the trained model
    interpreter = Interpreter.load('./models/nlu/default/restaurantnlu', builder)
    
    # Parse a sample text input
    result = interpreter.parse("can you please suggest food")
    print(result)

if __name__ == '__main__':
    # Train the NLU model with specified data and configuration
    train_nlu('./data/data.json', 'config_spacy.json', './models/nlu')
    
    # Run the NLU model to parse a sample input
    run_nlu()
