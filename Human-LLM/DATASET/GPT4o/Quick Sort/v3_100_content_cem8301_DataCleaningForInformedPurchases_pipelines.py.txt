class CraigslistPipeline:
    def process_item(self, item, spider):
        """Process the given item and return it."""
        return item

def main():
    # Create an instance of the CraigslistPipeline
    pipeline = CraigslistPipeline()

    # Example item and spider (for demonstration purposes)
    example_item = {
        'title': 'Example Item',
        'price': '$100',
        'description': 'This is an example item description.'
    }
    example_spider = 'craigslist_spider'

    # Process the item using the pipeline
    processed_item = pipeline.process_item(example_item, example_spider)

    # Print the processed item
    print("Processed Item:", processed_item)

if __name__ == "__main__":
    main()
