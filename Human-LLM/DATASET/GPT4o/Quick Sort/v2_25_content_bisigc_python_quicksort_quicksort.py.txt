def quickSort(arr, left, right):
    """
    Sorts the list 'arr' using the Quick Sort algorithm.
    
    Args:
    arr (list): The list to be sorted.
    left (int): The starting index of the segment to be sorted.
    right (int): The ending index of the segment to be sorted.
    """
    if left < right:
        pivot = arr[left]  # Choose the pivot element
        s = left  # Initialize the partitioning index
        
        # Partition the array around the pivot
        for i in range(left + 1, right):
            if arr[i] < pivot:
                s += 1
                arr[s], arr[i] = arr[i], arr[s]  # Swap elements
        
        arr[left], arr[s] = arr[s], arr[left]  # Place the pivot in the correct position
        
        quickSort(arr, left, s)  # Recursively sort the left partition
        quickSort(arr, s + 1, right)  # Recursively sort the right partition

def main():
    """
    Main function to demonstrate the Quick Sort algorithm.
    """
    # Example usage
    array = [21, 4, 1, 3, 9, 20, 25, 6, 21, 14]
    print("Original array:", array)
    
    quickSort(array, 0, len(array))  # Perform quick sort on the array
    
    print("Sorted array:", array)

if __name__ == "__main__":
    main()
