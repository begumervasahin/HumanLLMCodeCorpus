def quickSortLomutoHelper(arr, start, end):
    """
    Helper function to perform Quick Sort using the Lomuto partition scheme.

    Args:
    arr (list): The list to be sorted.
    start (int): The starting index of the sublist.
    end (int): The ending index of the sublist.

    Returns:
    list: The sorted list.
    """
    if len(arr) == 1:
        return arr
    elif start < end:
        piv = lomutoPartition(arr, start, end)
        quickSortLomutoHelper(arr, start, piv - 1)
        quickSortLomutoHelper(arr, piv + 1, end)
    return arr

def quickSortLomuto(arr):
    """
    Function to sort a list using Quick Sort with Lomuto partition scheme.

    Args:
    arr (list): The list to be sorted.

    Returns:
    list: The sorted list.
    """
    return quickSortLomutoHelper(arr, 0, len(arr) - 1)

def lomutoPartition(arr, start, end):
    """
    Partition the list using the Lomuto partition scheme.

    Args:
    arr (list): The list to be partitioned.
    start (int): The starting index of the sublist.
    end (int): The ending index of the sublist.

    Returns:
    int: The pivot index after partitioning.
    """
    pivot = arr[end]
    i = start - 1
    for j in range(start, end):
        if arr[j] <= pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    arr[i + 1], arr[end] = arr[end], arr[i + 1]
    return i + 1

def quickSortHoareHelper(arr, start, end):
    """
    Helper function to perform Quick Sort using the Hoare partition scheme.

    Args:
    arr (list): The list to be sorted.
    start (int): The starting index of the sublist.
    end (int): The ending index of the sublist.

    Returns:
    list: The sorted list.
    """
    if len(arr) == 1:
        return arr
    if start < end:
        piv = hoarePartition(arr, start, end)
        quickSortHoareHelper(arr, start, piv)
        quickSortHoareHelper(arr, piv + 1, end)
    return arr

def hoarePartition(arr, start, end):
    """
    Partition the list using the Hoare partition scheme.

    Args:
    arr (list): The list to be partitioned.
    start (int): The starting index of the sublist.
    end (int): The ending index of the sublist.

    Returns:
    int: The pivot index after partitioning.
    """
    pivot = arr[start]
    i = start - 1
    j = end + 1
    while True:
        i += 1
        while arr[i] < pivot:
            i += 1
        j -= 1
        while arr[j] > pivot:
            j -= 1
        if i >= j:
            return j
        arr[i], arr[j] = arr[j], arr[i]

def quickSortHoare(arr):
    """
    Function to sort a list using Quick Sort with Hoare partition scheme.

    Args:
    arr (list): The list to be sorted.

    Returns:
    list: The sorted list.
    """
    return quickSortHoareHelper(arr, 0, len(arr) - 1)
