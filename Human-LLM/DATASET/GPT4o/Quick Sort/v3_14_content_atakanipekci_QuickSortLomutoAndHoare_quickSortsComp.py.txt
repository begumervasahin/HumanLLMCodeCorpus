def quick_sort_lomuto(arr):
    """
    Function to sort a list using Quick Sort with Lomuto partition scheme.

    Args:
    arr (list): The list to be sorted.

    Returns:
    list: The sorted list.
    """
    def lomuto_partition(arr, start, end):
        """
        Partition the list using the Lomuto partition scheme.

        Args:
        arr (list): The list to be partitioned.
        start (int): The starting index of the sublist.
        end (int): The ending index of the sublist.

        Returns:
        int: The pivot index after partitioning.
        """
        pivot = arr[end]
        i = start - 1
        for j in range(start, end):
            if arr[j] <= pivot:
                i += 1
                arr[i], arr[j] = arr[j], arr[i]
        arr[i + 1], arr[end] = arr[end], arr[i + 1]
        return i + 1

    def quick_sort_lomuto_helper(arr, start, end):
        """
        Helper function to perform Quick Sort using the Lomuto partition scheme.

        Args:
        arr (list): The list to be sorted.
        start (int): The starting index of the sublist.
        end (int): The ending index of the sublist.
        """
        if start < end:
            pivot_index = lomuto_partition(arr, start, end)
            quick_sort_lomuto_helper(arr, start, pivot_index - 1)
            quick_sort_lomuto_helper(arr, pivot_index + 1, end)

    quick_sort_lomuto_helper(arr, 0, len(arr) - 1)
    return arr

def quick_sort_hoare(arr):
    """
    Function to sort a list using Quick Sort with Hoare partition scheme.

    Args:
    arr (list): The list to be sorted.

    Returns:
    list: The sorted list.
    """
    def hoare_partition(arr, start, end):
        """
        Partition the list using the Hoare partition scheme.

        Args:
        arr (list): The list to be partitioned.
        start (int): The starting index of the sublist.
        end (int): The ending index of the sublist.

        Returns:
        int: The pivot index after partitioning.
        """
        pivot = arr[start]
        i = start - 1
        j = end + 1
        while True:
            i += 1
            while arr[i] < pivot:
                i += 1
            j -= 1
            while arr[j] > pivot:
                j -= 1
            if i >= j:
                return j
            arr[i], arr[j] = arr[j], arr[i]

    def quick_sort_hoare_helper(arr, start, end):
        """
        Helper function to perform Quick Sort using the Hoare partition scheme.

        Args:
        arr (list): The list to be sorted.
        start (int): The starting index of the sublist.
        end (int): The ending index of the sublist.
        """
        if start < end:
            pivot_index = hoare_partition(arr, start, end)
            quick_sort_hoare_helper(arr, start, pivot_index)
            quick_sort_hoare_helper(arr, pivot_index + 1, end)

    quick_sort_hoare_helper(arr, 0, len(arr) - 1)
    return arr
