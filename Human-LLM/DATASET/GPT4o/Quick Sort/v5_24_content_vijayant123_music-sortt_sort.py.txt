import os
import sys
from os import listdir
from os.path import isfile, join
from pygame import mixer

# Define paths
SOURCE_PATH = "/home/error-404/Desktop/songs"
SAVE_PATH = "/home/error-404/Desktop/dance"
ALT_PATH = "/home/error-404/Desktop/others"

def get_song_files(path):
    """
    Get a list of all files in the given path.

    Args:
    path (str): The directory path to list files from.

    Returns:
    list: A list of filenames in the given path.
    """
    return [f for f in listdir(path) if isfile(join(path, f))]

def play_song(song_path, start_time=40):
    """
    Load and play a song from the specified start time.

    Args:
    song_path (str): The full path of the song to be played.
    start_time (int): The time in seconds from which to start playing the song.
    """
    mixer.music.load(song_path)
    mixer.music.play(0, start_time)

def handle_user_input(song_path):
    """
    Handle user input to control playback and move files.

    Args:
    song_path (str): The full path of the current song being played.
    """
    global is_paused

    while True:
        user_input = input().strip()

        if user_input == 'q':
            sys.exit(0)  # Quit the program
        elif user_input.isdigit():
            # Rewind and play from the specified time
            mixer.music.rewind()
            mixer.music.play(0, int(user_input) * 10)
        elif user_input == '':
            # Stop the music and move the file to alt_path
            mixer.music.stop()
            os.rename(song_path, join(ALT_PATH, os.path.basename(song_path)))
            break
        elif user_input == '/':
            # Stop the music and move the file to save_path
            mixer.music.stop()
            os.rename(song_path, join(SAVE_PATH, os.path.basename(song_path)))
            break
        elif user_input == 'p':
            if not is_paused:
                mixer.music.pause()  # Pause the music
                is_paused = True
            else:
                mixer.music.unpause()  # Unpause the music
                is_paused = False

def main():
    # Initialize the mixer
    mixer.init()

    # Get a list of all song files in the source path
    song_files = get_song_files(SOURCE_PATH)

    # Loop through each file in the source path
    for song in song_files:
        song_path = join(SOURCE_PATH, song)
        print(song)
        play_song(song_path)
        handle_user_input(song_path)

if __name__ == "__main__":
    main()
