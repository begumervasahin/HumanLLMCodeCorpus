def quicksort(arr):
    """
    Sort an array using the quicksort algorithm.

    Args:
    arr (list): The list of elements to be sorted.

    Returns:
    list: The sorted list.
    """
    # Base case: if the array is empty or contains a single element, it's already sorted
    if len(arr) <= 1:
        return arr

    # Choose the first element as the pivot
    pivot = arr[0]

    # Partition the array into three lists: lowVal, pivoter, and highVal
    lowVal = [x for x in arr if x < pivot]
    pivoter = [x for x in arr if x == pivot]
    highVal = [x for x in arr if x > pivot]

    # Recursively sort the lowVal and highVal lists
    sorted_low = quicksort(lowVal)
    sorted_high = quicksort(highVal)

    # Concatenate the sorted lowVal list, pivoter list, and sorted highVal list
    return sorted_low + pivoter + sorted_high
