def quicksort(arr):
    """
    Sort an array using the quicksort algorithm.

    Args:
    arr (list): The list of elements to be sorted.

    Returns:
    list: The sorted list.
    """
    # Base case: if the array is empty or contains a single element, it's already sorted
    if len(arr) <= 1:
        return arr

    # Choose the first element as the pivot
    pivot = arr[0]

    # Partition the array into three lists: less, equal, and greater
    less = [x for x in arr if x < pivot]
    equal = [x for x in arr if x == pivot]
    greater = [x for x in arr if x > pivot]

    # Recursively sort the less and greater lists
    return quicksort(less) + equal + quicksort(greater)

# Test the quicksort function
if __name__ == "__main__":
    test_array = [1, 3, 5, 2]
    sorted_array = quicksort(test_array)
    print(sorted_array)  # Output: [1, 2, 3, 5]
