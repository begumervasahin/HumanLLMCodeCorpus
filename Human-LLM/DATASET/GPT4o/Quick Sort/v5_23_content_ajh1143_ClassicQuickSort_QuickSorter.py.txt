def quicksort(arr):
    """
    Sorts an array using the quicksort algorithm.

    Parameters:
    arr (list): The list of elements to be sorted.

    Returns:
    list: A new list with the elements from `arr` sorted in ascending order.
    """
    # Base case: if the array has 1 or no elements, it is already sorted
    if len(arr) <= 1:
        return arr

    # Choose the first element as the pivot
    pivot = arr[0]

    # Partition the array into three lists based on the pivot
    low_vals = [x for x in arr if x < pivot]
    pivots = [x for x in arr if x == pivot]
    high_vals = [x for x in arr if x > pivot]

    # Recursively sort the low and high value lists
    low_sorted = quicksort(low_vals)
    high_sorted = quicksort(high_vals)

    # Concatenate the sorted low values, pivots, and sorted high values
    sorted_arr = low_sorted + pivots + high_sorted

    return sorted_arr

# Example usage
if __name__ == "__main__":
    sorted_list = quicksort([1, 3, 5, 2])
    print("Sorted list:", sorted_list)
