def partition(arr, low, high):
    """
    Partition the array on the basis of pivot element.

    Args:
        arr (list): List of elements to be sorted.
        low (int): Starting index of the array slice.
        high (int): Ending index of the array slice.

    Returns:
        int: Index position of the pivot element after partition.
    """
    i = low - 1  # Index of the smaller element
    pivot = arr[high]  # Pivot element

    for j in range(low, high):
        if arr[j] <= pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]

    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    return i + 1

def quick_sort(arr, low, high):
    """
    Perform non-recursive QuickSort on the array.

    Args:
        arr (list): List of elements to be sorted.
        low (int): Starting index of the array slice.
        high (int): Ending index of the array slice.
    """
    # Initialize stack for storing subarray indices
    size = high - low + 1
    stack = [0] * size

    top = -1

    # Push initial low and high indices to the stack
    top += 1
    stack[top] = low
    top += 1
    stack[top] = high

    # Process the stack until it's empty
    while top >= 0:
        # Pop high and low indices
        high = stack[top]
        top -= 1
        low = stack[top]
        top -= 1

        # Partition the array and get the pivot index
        p = partition(arr, low, high)

        # If there are elements on the left side of the pivot, push them to the stack
        if p - 1 > low:
            top += 1
            stack[top] = low
            top += 1
            stack[top] = p - 1

        # If there are elements on the right side of the pivot, push them to the stack
        if p + 1 < high:
            top += 1
            stack[top] = p + 1
            top += 1
            stack[top] = high

if __name__ == "__main__":
    array = [4, 3, 5, 2, 1, 3, 2, 3]
    n = len(array)
    quick_sort(array, 0, n - 1)

    print("Sorted array is:")
