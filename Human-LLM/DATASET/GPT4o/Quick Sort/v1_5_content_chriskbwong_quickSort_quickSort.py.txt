def partition(arr, l, h):
    """
    Partition the array on the basis of pivot element.

    :param arr: List of elements to be sorted.
    :param l: Starting index of the array slice.
    :param h: Ending index of the array slice.
    :return: Index position of the pivot element after partition.
    """
    i = l - 1  # Index of the smaller element
    pivot = arr[h]  # Pivot element

    for j in range(l, h):
        if arr[j] <= pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]

    arr[i + 1], arr[h] = arr[h], arr[i + 1]
    return i + 1

def quickSort(arr, l, h):
    """
    Perform non-recursive QuickSort on the array.

    :param arr: List of elements to be sorted.
    :param l: Starting index of the array slice.
    :param h: Ending index of the array slice.
    """
    size = h - l + 1
    stack = [0] * size  # Initialize stack

    top = -1
    top += 1
    stack[top] = l
    top += 1
    stack[top] = h

    while top >= 0:
        h = stack[top]
        top -= 1
        l = stack[top]
        top -= 1

        p = partition(arr, l, h)

        if p - 1 > l:
            top += 1
            stack[top] = l
            top += 1
            stack[top] = p - 1

        if p + 1 < h:
            top += 1
            stack[top] = p + 1
            top += 1
            stack[top] = h

    arr = [4, 3, 5, 2, 1, 3, 2, 3]
    n = len(arr)
    quickSort(arr, 0, n - 1)
