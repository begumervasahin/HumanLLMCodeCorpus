import time
from random import randint

def partition(arr, low, high):
    """
    Partition function used in quicksort.
    This function selects the last element as the pivot, places the pivot element at its correct
    position in the sorted array, and places all smaller elements to the left of the pivot and
    all greater elements to the right of the pivot.
    
    Args:
        arr (list): The list to be sorted.
        low (int): The starting index of the sublist.
        high (int): The ending index of the sublist.

    Returns:
        int: The partitioning index.
    """
    pivot = arr[high]
    i = low - 1

    for j in range(low, high):
        if arr[j] <= pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]

    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    return i + 1

def quick_sort(arr, low, high):
    """
    Quicksort function.
    Recursively sorts the sublists.

    Args:
        arr (list): The list to be sorted.
        low (int): The starting index of the sublist.
        high (int): The ending index of the sublist.
    """
    if low < high:
        pi = partition(arr, low, high)
        quick_sort(arr, low, pi - 1)
        quick_sort(arr, pi + 1, high)

def main():
    """
    Main function to perform quicksort on lists of increasing size and measure the time taken.
    """
    print("Quick Sort Performance Test")

    # Initialize parameters for testing
    input_size = 100000
    max_size = 10000000
    step = 1100000

    while input_size <= max_size:
        # Generate a list of random numbers
        randlist = [randint(1, input_size) for _ in range(input_size)]
        
        print(f"Sorting list of size {input_size}...")

        # Record start time
        start_time = time.time()
        
        # Perform quicksort
        quick_sort(randlist, 0, len(randlist) - 1)
        
        # Record end time
        end_time = time.time()
        
        print(f"Time taken for input size {input_size}: {end_time - start_time} seconds")
        
        # Increase the input size for the next iteration
        input_size += step

if __name__ == "__main__":
    main()
