def partition(array, low, high):
    """
    Partition the array around the pivot element.

    This function rearranges the elements in the array such that elements less than or equal to the pivot
    are on its left, and elements greater than the pivot are on its right.

    :param array: The list to be partitioned.
    :param low: The starting index of the partition.
    :param high: The ending index of the partition.
    :return: The index of the pivot element after partitioning.
    """
    pivot = array[high]  # Pivot element
    i = low - 1  # Index of the smaller element

    for j in range(low, high):
        if array[j] <= pivot:
            i += 1
            array[i], array[j] = array[j], array[i]

    array[i + 1], array[high] = array[high], array[i + 1]
    return i + 1

def quick_sort(array, low, high):
    """
    Perform the QuickSort algorithm on the list.

    :param array: The list to be sorted.
    :param low: The starting index of the sort.
    :param high: The ending index of the sort.
    """
    if low < high:
        pi = partition(array, low, high)
        quick_sort(array, low, pi - 1)
        quick_sort(array, pi + 1, high)

def main():
    """
    Main function to demonstrate the QuickSort algorithm.
    """
    nums = [3, 1, 6, 8, 0]
    print("Original array:", nums)
    quick_sort(nums, 0, len(nums) - 1)
    print("Sorted array:", nums)

if __name__ == '__main__':
    main()
