import random

def partition(arr, low, high):
    """
    Function to partition the array on the basis of pivot element.
    The pivot element is chosen randomly.
    
    :param arr: List of elements to be sorted.
    :param low: Starting index of the array slice.
    :param high: Ending index of the array slice.
    :return: Index position of the pivot element after partition.
    """
    pindex = low
    pivot_index = random.randint(low, high)
    pivot = arr[pivot_index]
    arr[pivot_index], arr[high] = arr[high], arr[pivot_index]

    for j in range(low, high):
        if arr[j] <= pivot:
            arr[pindex], arr[j] = arr[j], arr[pindex]
            pindex += 1

    arr[pindex], arr[high] = arr[high], arr[pindex]
    return pindex

def quickSort(arr, low, high):
    """
    Function to perform QuickSort on the array.
    
    :param arr: List of elements to be sorted.
    :param low: Starting index of the array slice.
    :param high: Ending index of the array slice.
    """
    if low < high:
        pi = partition(arr, low, high)
        quickSort(arr, low, pi - 1)
        quickSort(arr, pi + 1, high)

if __name__ == '__main__':
    # Sample array to be sorted
    arr = [10, 7, 8, 9, 1, 5]
    n = len(arr)
    
    # Perform QuickSort
    quickSort(arr, 0, n - 1)
    
    # Print the sorted array
    print("Sorted array is:")
