import datetime
import statistics

def quick_sort(arr):
    """
    Sort an array using the Quick Sort algorithm with a median-of-three pivot selection.

    Args:
    arr (list): The list of elements to be sorted.

    Returns:
    list: The sorted list.
    """
    if len(arr) <= 1:
        return arr

    # Median-of-three pivot selection
    items = [arr[0], arr[len(arr)//2], arr[len(arr)-1]]
    pivot = statistics.median(items)

    # Partition the array
    left = [x for x in arr if x < pivot]
    middle = [x for x in arr if x == pivot]
    right = [x for x in arr if x > pivot]

    # Recursively sort left and right partitions
    return quick_sort(left) + middle + quick_sort(right)

if __name__ == "__main__":
    # Test array
    test = [21, 4, 1, 3, 9, 20, 25, 6, 21, 14]

    # Measure the time taken to sort the array
    start = datetime.datetime.now()
    sorted_array = quick_sort(test)
    finish = datetime.datetime.now()

    # Print the sorted array and the time taken
    print("Sorted array:", sorted_array)
    print("Time taken:", finish - start)
