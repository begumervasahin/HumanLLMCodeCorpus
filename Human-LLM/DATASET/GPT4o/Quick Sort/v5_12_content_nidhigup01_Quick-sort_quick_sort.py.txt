import datetime
import statistics

def quick_sort(arr):
    """
    Perform a quick sort on the given array.

    Parameters:
    arr (list): The list of elements to be sorted.

    Returns:
    list: A new list with the elements sorted.
    """
    if len(arr) <= 1:
        return arr

    # Choosing pivot as the median of the first, middle, and last elements
    first, middle, last = arr[0], arr[len(arr) // 2], arr[-1]
    pivot = statistics.median([first, middle, last])

    # Partitioning the array into left, middle, and right
    left = [x for x in arr if x < pivot]
    middle = [x for x in arr if x == pivot]
    right = [x for x in arr if x > pivot]

    # Recursively applying quick sort and concatenating results
    return quick_sort(left) + middle + quick_sort(right)

# Test the quick_sort function
if __name__ == "__main__":
    test_array = [21, 4, 1, 3, 9, 20, 25, 6, 21, 14]

    start_time = datetime.datetime.now()
    sorted_array = quick_sort(test_array)
    end_time = datetime.datetime.now()

    print("Sorted array:", sorted_array)
    print("Time taken:", end_time - start_time)
