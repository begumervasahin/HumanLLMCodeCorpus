def read_tv_show_list(filename):
    """
    Read TV show list from a file and return a list of tuples (show_name, show_year).

    Args:
        filename (str): The name of the file to read from.

    Returns:
        list: A list of tuples containing show names and their corresponding years.
    """
    tv_list = []
    with open(filename, "r") as file:
        for line in file:
            show_name, show_year = line.strip().split('_')
            tv_list.append((show_name, show_year))
    return tv_list

def write_sorted_tv_show_list(filename, tv_list):
    """
    Write the sorted TV show list to a file.

    Args:
        filename (str): The name of the file to write to.
        tv_list (list): The sorted list of tuples containing show names and their corresponding years.
    """
    with open(filename, "w") as file:
        for show_name, show_year in tv_list:
            file.write(f"{show_name}_{show_year}\n")

def main():
    """
    Main function to read, sort, and write the TV show list.
    """
    input_filename = "tvshow_list_unsorted.txt"
    output_filename = "tvshow_list_sorted.txt"

    # Read TV show list from the input file
    tv_list = read_tv_show_list(input_filename)

    # Sort the list by year first and then by show name
    tv_list.sort(key=lambda show: (show[1], show[0]))

    # Write the sorted list to the output file
    write_sorted_tv_show_list(output_filename, tv_list)

if __name__ == "__main__":
    main()
