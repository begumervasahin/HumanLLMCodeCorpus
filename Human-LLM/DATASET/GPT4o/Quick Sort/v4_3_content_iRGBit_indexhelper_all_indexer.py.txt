import sys
import string
from collections import Counter

DEFAULT_FILE = 'files/sample.txt'
DEFAULT_STOPWORDS = 'stop_words.txt'

def main():
    if len(sys.argv) > 3:
        print_usage()
        sys.exit()

    your_file, your_stopwords = get_file_paths()

    print(f'Using {your_file} as file and {your_stopwords} as stop word reference.\n')
    index_words(your_file, your_stopwords)

def print_usage():
    print("\nUsage: python indexer.py <yourFile> <stopWords>")
    print(f"If no arguments are given, {DEFAULT_FILE} and {DEFAULT_STOPWORDS} will be used as default files\n")

def get_file_paths():
    if len(sys.argv) == 3:
        return sys.argv[1], sys.argv[2]
    elif len(sys.argv) == 2:
        return sys.argv[1], DEFAULT_STOPWORDS
    else:
        return DEFAULT_FILE, DEFAULT_STOPWORDS

def index_words(file_path, stopwords_path):
    with open(file_path, 'r', encoding='utf-8') as file:
        book_words = file.read().lower().split()

    book_words = [word.strip(string.punctuation) for word in book_words]

    with open(stopwords_path, 'r', encoding='utf-8-sig') as file:
        stop_words = file.read().splitlines()

    filtered_words = [word for word in book_words if word and word not in stop_words]

    word_counts = Counter(filtered_words)

    sorted_words = sorted(word_counts.items())

    for word, count in sorted_words:
        print(f'{word}: {count}')

if __name__ == '__main__':
    main()
