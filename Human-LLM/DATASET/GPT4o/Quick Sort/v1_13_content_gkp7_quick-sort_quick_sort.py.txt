def quicksort(alist):
    """
    Sorts a list in place using the Quick Sort algorithm.

    Args:
    alist (list): The list to be sorted.
    """
    quicksorthelper(alist, 0, len(alist) - 1)

def quicksorthelper(alist, first, last):
    """
    Helper function to recursively sort sublists.

    Args:
    alist (list): The list to be sorted.
    first (int): The starting index of the sublist.
    last (int): The ending index of the sublist.
    """
    if first < last:
        splitpoint = partition(alist, first, last)
        quicksorthelper(alist, first, splitpoint - 1)
        quicksorthelper(alist, splitpoint + 1, last)

def partition(alist, first, last):
    """
    Partition the sublist and return the split point.

    Args:
    alist (list): The list to be partitioned.
    first (int): The starting index of the sublist.
    last (int): The ending index of the sublist.

    Returns:
    int: The split point index.
    """
    pivotvalue = alist[first]
    leftmark = first + 1
    rightmark = last
    done = False

    while not done:
        while leftmark <= rightmark and alist[leftmark] <= pivotvalue:
            leftmark = leftmark + 1
        while alist[rightmark] >= pivotvalue and rightmark >= leftmark:
            rightmark = rightmark - 1
        if rightmark < leftmark:
            done = True
        else:
            alist[leftmark], alist[rightmark] = alist[rightmark], alist[leftmark]

    alist[first], alist[rightmark] = alist[rightmark], alist[first]
    return rightmark
