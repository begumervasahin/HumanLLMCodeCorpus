import os
import datetime
import logging
import pathlib as pl
import re
from typing import Union, Optional, TypeVar
from exifread import process_file
import click

# Type definitions
PathLike = Union[str, bytes, pl.Path]
T = TypeVar('T')
Maybe = Optional[T]

# Setup logger
logger = logging.getLogger(__name__)
if not logger.hasHandlers():
    logger.addHandler(logging.StreamHandler())

# Extend pathlib.Path to include a copy method
def _copy(self, target: PathLike):
    import shutil
    assert self.is_file()
    shutil.copy(self, target)
pl.Path.copy = _copy

def sort_directory(source_path: PathLike, target_path: PathLike = None, remove_empty: bool = True, recursive: bool = True, copy: bool = False):
    """
    Sorts photos in the source directory into the target directory based on their date.

    Args:
        source_path (PathLike): The source directory path.
        target_path (PathLike): The target directory path.
        remove_empty (bool): Remove empty directories if True.
        recursive (bool): Recursively process directories if True.
        copy (bool): Copy files if True, otherwise move.
    """
    source_path = pl.Path(source_path)
    target_path = pl.Path(target_path) if target_path else source_path

    assert source_path.is_dir()
    assert target_path.is_dir()

    logger.info(f'Sorting photos in "{source_path}"')
    if source_path != target_path:
        logger.info(f'Outputting to "{target_path}"')

    logger.debug(f'Options - remove_empty: {remove_empty}, recursive: {recursive}, copy: {copy}')
    logger.info('=' * 20)

    glob_pattern = '**/*.*' if recursive else '*.*'
    sorted_files = set()
    directories_to_delete = set()

    for file in source_path.glob(glob_pattern):
        if file not in sorted_files:
            new_file = sort_file(file, target_path, copy=copy)
            if new_file:
                sorted_files.add(new_file)
            if remove_empty and is_empty_dir(file.parent):
                directories_to_delete.add(file.parent)

    for directory in directories_to_delete:
        logger.info('Removing empty directories')
        if is_empty_dir(directory):
            logger.debug(f'Removing empty directory: {directory}')
            directory.rmdir()

    logger.info(f'Sorting photos in "{source_path}" completed successfully!')

def sort_file(path: pl.Path, target_path: PathLike, copy: bool = False) -> Maybe[pl.Path]:
    """
    Sorts a single file into the target directory based on its date.

    Args:
        path (pl.Path): The file to sort.
        target_path (PathLike): The target directory path.
        copy (bool): Copy the file if True, otherwise move.

    Returns:
        Maybe[pl.Path]: The new path of the file if sorted, otherwise None.
    """
    path = pl.Path(path)
    logger.debug(f'Sorting file: {path}')

    date = get_file_date(path)
    logger.debug(f'Inferred date: {date}')

    if date:
        new_path = target_path / date.strftime('%Y') / date.strftime('%Y-%m') / date.strftime('%Y-%m-%d') / path.name
    else:
        new_path = target_path / 'Unknown_Date' / path.name

    if new_path != path:
        if not new_path.parent.exists():
            logger.debug(f'Creating new directory: {new_path.parent}')
            new_path.parent.mkdir(parents=True)

        if new_path.exists():
            backup_path = get_backup_path(new_path)
            logger.warning(f'File "{new_path.name}" already exists in "{new_path.parent}". Saving as "{backup_path.name}" instead.')
            new_path = backup_path

        if copy:
            logger.info(f'Copying "{new_path.name}" to "{new_path.parent}"')
            path.copy(new_path)
        else:
            logger.info(f'Moving "{new_path.name}" to "{new_path.parent}"')
            path.rename(new_path)
    else:
        logger.debug('File already sorted.')

    return new_path

def get_backup_path(path: pl.Path) -> pl.Path:
    """
    Generates a backup path if the target file already exists.

    Args:
        path (pl.Path): The original file path.

    Returns:
        pl.Path: The backup file path.
    """
    def make_path(source_path: pl.Path, number: int) -> pl.Path:
        return source_path.with_name(f'{source_path.stem}_{number}{source_path.suffix}')

    backup_count = 0
    new_path = make_path(path, backup_count)
    while new_path.exists():
        backup_count += 1
        new_path = make_path(path, backup_count)
    return new_path

def is_empty_dir(path: pl.Path) -> bool:
    """
    Checks if the directory is empty.

    Args:
        path (pl.Path): The directory path.

    Returns:
        bool: True if the directory is empty, otherwise False.
    """
    return path.is_dir() and sum(1 for _ in path.iterdir()) == 0

def get_file_date(path: pl.Path) -> Maybe[datetime.date]:
    """
    Gets the file date from metadata or filename.

    Args:
        path (pl.Path): The file path.

    Returns:
        Maybe[datetime.date]: The date of the file if found, otherwise None.
    """
    return get_file_date_from_metadata(path) or get_file_date_from_filename(path)

def get_file_date_from_metadata(path: pl.Path) -> Maybe[datetime.date]:
    """
    Gets the file date from metadata.

    Args:
        path (pl.Path): The file path.

    Returns:
        Maybe[datetime.date]: The date of the file if found in metadata, otherwise None.
    """
    if path.suffix.lower() in ('.jpg', '.jpeg'):
        date_tag_names = ('EXIF DateTimeOriginal', 'EXIF DateTimeDigitized', 'Image DateTime')
        tags = load_tags(path)
        for tag in date_tag_names:
            if tag in tags:
                date_str = tags[tag].values
                dt = datetime.datetime.strptime(date_str, '%Y:%m:%d %H:%M:%S')
                return dt.date()
    return None

def get_file_date_from_filename(path: pl.Path) -> Maybe[datetime.date]:
    """
    Gets the file date from filename.

    Args:
        path (pl.Path): The file path.

    Returns:
        Maybe[datetime.date]: The date of the file if found in filename, otherwise None.
    """
    pattern = r'(?P<year>\d{4})(.|\s+)?(?P<month>\d{2})(.|\s+)?(?P<day>\d{2})'
    match = re.search(pattern, path.name)
    if match:
        date_kwargs = {k: int(v) for k, v in match.groupdict().items()}
        try:
            return datetime.date(**date_kwargs)
        except ValueError:
            return None
    return None

def load_tags(path: pl.Path) -> dict:
    """
    Loads EXIF tags from an image file.

    Args:
        path (pl.Path): The image file path.

    Returns:
        dict: The EXIF tags.
    """
    with open(path, 'rb') as image_file:
        return process_file(image_file)

@click.command()
@click.option('--copy/--move', '-c/-m', 'copy', default=False, help="Copy or move files")
@click.option('--output', '-o', 'target_path', default=None, type=click.Path(file_okay=False, dir_okay=True, writable=True, readable=True, resolve_path=True), help="Output directory")
@click.option('--remove-empty/--no-remove-empty', '-e/-E', default=True, help="Remove empty directories")
@click.option('--logger-level', '-l', default='info', type=click.Choice(['debug', 'info', 'warning', 'error', 'critical']), help="Set logger level")
@click.option('--recursive/--no-recursive', '-r/-R', default=True, help="Recursively process directories")
@click.argument('source_path', type=click.Path(exists=True, file_okay=False, dir_okay=True, writable=True, readable=True, resolve_path=True))
def cli(source_path, target_path, remove_empty, copy, recursive, logger_level):
    logger.setLevel(getattr(logging, logger_level.upper()))
    sort_directory(source_path=source_path, target_path=target_path, remove_empty=remove_empty, recursive=recursive, copy=copy)

if __name__ == '__main__':
    cli()
