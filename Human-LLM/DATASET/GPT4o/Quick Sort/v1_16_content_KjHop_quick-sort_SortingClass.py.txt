class Sorting:
    def __init__(self):
        self.listOfNumbers = []

    def addNumber(self, number):
        """
        Add a number to the list.
        
        Args:
        number (int): The number to add to the list.
        """
        self.listOfNumbers.append(number)

    def returnList(self):
        """
        Return the list of numbers.
        
        Returns:
        list: The list of numbers.
        """
        return self.listOfNumbers

    def printEachElementOfList(self):
        """
        Print each element of the list.
        """
        for number in self.listOfNumbers:
            print(number)

    def quickSort(self, arr):
        """
        Perform quick sort on the list.
        
        Args:
        arr (list): The list to sort.
        
        Returns:
        list: The sorted list.
        """
        lessThanPivot = []
        equalPivot = []
        greaterThanPivot = []
        
        if len(arr) > 1:
            pivot = arr[len(arr) - 1]
            for number in arr:
                if number < pivot:
                    lessThanPivot.append(number)
                elif number == pivot:
                    equalPivot.append(number)
                else:
                    greaterThanPivot.append(number)
            return self.quickSort(lessThanPivot) + equalPivot + self.quickSort(greaterThanPivot)
        else:
            return arr

if __name__ == "__main__":
    sorting = Sorting()
    numbers_to_add = [34, 7, 23, 32, 5, 62, 32, 7]
    
    # Add numbers to the list
    for number in numbers_to_add:
        sorting.addNumber(number)
    
    # Print the list
    print("Original list:")
    sorting.printEachElementOfList()
    
    # Perform quick sort and print the sorted list
    sorted_list = sorting.quickSort(sorting.returnList())
    print("\nSorted list:")
    for number in sorted_list:
        print(number)
