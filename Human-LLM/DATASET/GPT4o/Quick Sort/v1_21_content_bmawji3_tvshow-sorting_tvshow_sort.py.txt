# Define the file paths
input_file = "tvshow_list_unsorted.txt"
output_file = "tvshow_list_sorted.txt"

# Initialize an empty list to store TV show tuples
tv_list = []

# Read the unsorted list of TV shows from the input file
with open(input_file, "r") as myfile:
    for line in myfile:
        # Split each line by the underscore and strip newline characters
        temp_split = line.strip().split('_')
        # Append the tuple (first part, second part) to the list
        tv_list.append((temp_split[0], temp_split[1]))

# Sort the list first by the second part and then by the first part of the tuple
tv_list.sort(key=lambda x: (x[1], x[0]))

# Write the sorted list to the output file
with open(output_file, 'w') as f:
    for item in tv_list:
        f.write(f"{item[0]}_{item[1]}\n")
