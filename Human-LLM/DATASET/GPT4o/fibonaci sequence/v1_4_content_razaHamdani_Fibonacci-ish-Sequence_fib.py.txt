def list_fib_seq(s, x):
    """
    Generate a Fibonacci sequence starting with 0 and s, stopping when the next number exceeds x.

    Args:
        s (int): The second number in the Fibonacci sequence.
        x (int): The maximum value for the Fibonacci sequence.

    Returns:
        list: A list containing the Fibonacci sequence.
    """
    res = [0, s]
    while (res[-1] + res[-2] <= x):
        res.append(res[-1] + res[-2])
    print(res)
    return res

def find_fibish_seq_number(orgFs, x):
    """
    Find the minimum starting integer in the original Fibonacci sequence that divides x with no remainder.

    Args:
        orgFs (list): The original Fibonacci sequence.
        x (int): The target integer to be divided.

    Returns:
        int: The minimum starting integer that divides x, or None if no such integer is found.
    """
    min_start_int = None
    for member in orgFs:
        if member > x:
            break
        if member == 0 or member == 1:
            continue
        if x % member == 0:
            quotient = x // member
            if min_start_int is None or quotient < min_start_int:
                min_start_int = quotient
    return min_start_int
