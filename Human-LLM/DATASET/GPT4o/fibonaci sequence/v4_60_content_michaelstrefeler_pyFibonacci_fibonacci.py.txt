from turtle import *

def get_fibonacci_sequence(n):
    """Generate a Fibonacci sequence up to n terms."""
    sequence = [0, 1]
    for i in range(2, n):
        sequence.append(sequence[-1] + sequence[-2])
    return sequence

def draw_square(size, fib):
    """Draw a square for the given Fibonacci number."""
    forward(25 * size)
    write(fib)
    forward(25 * size)
    left(90)
    for _ in range(3):
        forward(25 * size)
        left(90)

def draw_fibonacci_spiral(sequence):
    """Draw the Fibonacci spiral using turtle graphics."""
    penup()
    goto(135, -20)
    pendown()
    
    for fib in sequence:
        if fib == 0:
            continue
        if fib == 1:
            forward(25 * fib / 2)
            write(fib)
            forward(25 * fib / 2)
        else:
            draw_square(fib, fib)
        left(90)

    penup()
    goto(135, 5)
    pendown()
    color('red')
    shape('turtle')
    pensize(2)
    setheading(180)
    
    for fib in sequence:
        if fib < 9:
            circle(25 * fib, 90)
        else:
            break

    done()

def main():
    """Main function to execute the Fibonacci sequence generation and drawing."""
    while True:
        amount = input('Choose up to which number of the Fibonacci sequence you want to go up to (min 5): ')
        if amount.isdecimal() and int(amount) >= 5:
            amount = int(amount)
            break
        print('Please enter a number greater than or equal to 5.')
    
    fibonacci_sequence = get_fibonacci_sequence(amount)
    
    for index, number in enumerate(fibonacci_sequence):
        print(f'F°{index}: {number}')
    
    draw_fibonacci_spiral(fibonacci_sequence)

if __name__ == "__main__":
    main()
