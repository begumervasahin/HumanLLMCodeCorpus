class Fibonacci:
    def __init__(self, fibtype=2, stop=10):
        self.steps = 0
        self.ratio = 0
        self.fibtype = fibtype
        self.stop = stop
        self.previous = [1] * fibtype

    def fibonacci(self):
        if self.steps == self.stop:
            print(f"Ratio {self.fibtype}: {self.ratio}")
            return
        else:
            mysum = sum(self.previous)
            self.ratio = mysum / self.previous[-1]
            self.steps += 1
            if len(self.previous) >= self.fibtype:
                newarray = self.previous[-self.fibtype + 1:]
                newarray.append(mysum)
                self.previous = newarray
            else:
                self.previous.append(mysum)
            self.fibonacci()

if __name__ == "__main__":
    for i in range(2, 56):  # Adjusted range to start from 2 and include 55
        myfib = Fibonacci(i, 54)
        myfib.fibonacci()
