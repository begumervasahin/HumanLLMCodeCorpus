import csv
from timeit import timeit

def benchmark_fibonacci():
    """
    Benchmark the execution time of different Fibonacci calculation methods and save the results to a CSV file.
    """
    with open('./fibonacci_benchmark.csv', 'w', newline='') as reportfile:
        writer = csv.writer(reportfile)
        writer.writerow(['n_arg', 'recursive_seconds', 'iterative_seconds', 'explicit_seconds'])

        for n in range(1, 46):
            print(f'n={n}...', end='')

            if n < 15:
                rec_time = timeit(f'fib.f_recursive({n})', setup='import fib', number=1)
            else:
                rec_time = 'NA'
                print(' ~recur~', end='')

            iter_time = timeit(f'fib.f_iterative({n})', setup='import fib', number=1)
            exp_time = timeit(f'fib.f_explicit({n})', setup='import fib', number=1)

            print(' OK')

            writer.writerow([n, rec_time, iter_time, exp_time])

if __name__ == "__main__":
    benchmark_fibonacci()
