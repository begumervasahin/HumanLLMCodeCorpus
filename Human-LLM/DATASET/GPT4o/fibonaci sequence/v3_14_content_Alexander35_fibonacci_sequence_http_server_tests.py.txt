from flask import Flask, request, jsonify

app = Flask(__name__)

def fibonacci(n):
    """Calculate the nth Fibonacci number."""
    a, b = 0, 1
    for _ in range(n):
        a, b = b, a + b
    return a

def validate_parameters(request):
    """Validate query parameters for 'from' and 'to'."""
    try:
        start = int(request.args.get('from'))
        end = int(request.args.get('to'))
    except (TypeError, ValueError):
        return None, None, {"error": "Invalid query parameters"}, 400

    if start < 0 or end < 0 or start > end:
        return None, None, {"error": "Invalid range parameters"}, 400

    return start, end, None, 200

def generate_fibonacci_sequence(start, end):
    """Generate a Fibonacci sequence from start to end."""
    return {i: fibonacci(i) for i in range(start, end)}

@app.route('/fibonacci', methods=['GET'])
def fibonacci_route():
    """Endpoint to get a sequence of Fibonacci numbers from 'start' to 'end'."""
    start, end, error_response, status_code = validate_parameters(request)
    if error_response:
        return jsonify(error_response), status_code

    result = generate_fibonacci_sequence(start, end)
    return jsonify(result)

@app.route('/get-sequence', methods=['GET'])
def get_sequence_route():
    """Endpoint to get a sequence of Fibonacci numbers from 'start' to 'end'."""
    start, end, error_response, status_code = validate_parameters(request)
    if error_response:
        return jsonify(error_response), status_code

    result = generate_fibonacci_sequence(start, end)
    return jsonify(result)

if __name__ == '__main__':
    app.run(port=5000)
