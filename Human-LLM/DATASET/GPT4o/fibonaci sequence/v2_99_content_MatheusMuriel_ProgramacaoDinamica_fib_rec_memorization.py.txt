import sys
from Memoria import Cerebro

class Memorization:
    def __init__(self):
        # Define the memory limit and cleaning rate for the Cerebro instance
        LIMIT = 10000000
        CLEANING_RATE = 0.7
        self.memory = Cerebro(LIMIT, CLEANING_RATE)

    def calculate(self, n):
        # Base cases for Fibonacci sequence
        if n == 0:
            return 0
        if n == 1:
            return 1

        # Convert indices to strings for use with the memory storage
        str_n1 = str(n - 1)
        str_n2 = str(n - 2)

        # Retrieve the values from memory if they exist
        n_1 = self.memory.get_lembranca(str_n1)
        n_2 = self.memory.get_lembranca(str_n2)

        # If the values are not in memory, calculate them recursively
        if n_1 == -1:
            n_1 = self.calculate(n - 1)
            self.memory.nova_memoria(str_n1, n_1)

        if n_2 == -1:
            n_2 = self.calculate(n - 2)
            self.memory.nova_memoria(str_n2, n_2)

        return n_1 + n_2

    def get_calculation_function(self):
        return lambda n: self.calculate(n)

    def __str__(self):
        return "Recursive + Memoization"

def main():
    # Ensure that exactly one argument is passed (the number to calculate the Fibonacci sequence for)
    if len(sys.argv) != 2:
        print("Usage: python script_name.py <number>")
        return

    try:
        # Convert the argument to an integer
        n = int(sys.argv[1])
    except ValueError:
        print("Please enter a valid integer.")
        return

    # Create an instance of the Memorization class and calculate the Fibonacci number
    fib_calculator = Memorization()
    result = fib_calculator.calculate(n)
    print(f"Fibonacci number for {n} is: {result}")

if __name__ == "__main__":
    main()
