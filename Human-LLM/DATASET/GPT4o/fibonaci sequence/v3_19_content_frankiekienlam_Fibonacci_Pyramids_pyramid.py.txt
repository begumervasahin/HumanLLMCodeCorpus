"""
Print a pyramid to the terminal.
A pyramid of height 3 would look like:
--=--
-===-
=====
"""

from argparse import ArgumentParser, RawDescriptionHelpFormatter

def print_pyramid(rows):
    """
    Print a pyramid of a given height.

    Args:
        rows (int): Number of rows to print.

    Example:
        input rows = 2
        output:
            -=-
            ===
    """
    if not isinstance(rows, int) or rows <= 0:
        print('Please enter a number greater than 0')
        return
    
    for i in range(1, rows + 1):
        dashes = rows - i
        equals = 2 * i - 1
        print('-' * dashes + '=' * equals + '-' * dashes)

def main():
    parser = ArgumentParser(
        description=__doc__, formatter_class=RawDescriptionHelpFormatter
    )
    parser.add_argument(
        "-r", "--rows", 
        default=10, 
        type=int, 
        help="Number of rows for the pyramid"
    )
    args = parser.parse_args()
    print_pyramid(args.rows)

if __name__ == "__main__":
    main()
