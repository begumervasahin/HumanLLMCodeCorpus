def calculate_fibonacci(n):
    """
    Calculate the first n Fibonacci numbers.
    
    Args:
    n (int): The number of Fibonacci numbers to calculate.
    
    Returns:
    list: A list containing the first n Fibonacci numbers.
    """
    first_number = 1
    second_number = 0
    fibonacci_list = []
    for _ in range(n):
        answer = first_number + second_number
        fibonacci_list.append(answer)
        first_number, second_number = second_number, answer
    return fibonacci_list

def print_fibonacci(fibonacci_list):
    """
    Print the Fibonacci numbers with their calculation steps.
    
    Args:
    fibonacci_list (list): A list of Fibonacci numbers to print.
    """
    for i, number in enumerate(fibonacci_list):
        if i == 0:
            print(f"{number} + 0 = {number}")
        elif i == 1:
            print(f"0 + {number} = {number}")
        else:
            print(f"{fibonacci_list[i - 2]} + {fibonacci_list[i - 1]} = {number}")

def main():
    """Main function to calculate and print the first 100 Fibonacci numbers."""
    fibonacci_numbers = calculate_fibonacci(100)
    print_fibonacci(fibonacci_numbers)

if __name__ == "__main__":
    main()
