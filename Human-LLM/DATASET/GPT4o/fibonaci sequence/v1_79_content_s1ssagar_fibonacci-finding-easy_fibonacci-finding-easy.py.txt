def multiply(mat1, mat2):
    """
    Multiply two 2x2 matrices under modulo 10^9+7.
    
    Args:
        mat1 (list): The first matrix as a list of 4 elements.
        mat2 (list): The second matrix as a list of 4 elements.

    Returns:
        list: The resulting matrix as a list of 4 elements.
    """
    MOD = 10**9 + 7
    return [
        (mat1[0] * mat2[0] + mat1[1] * mat2[2]) % MOD,
        (mat1[0] * mat2[1] + mat1[1] * mat2[3]) % MOD,
        (mat1[2] * mat2[0] + mat1[3] * mat2[2]) % MOD,
        (mat1[2] * mat2[1] + mat1[3] * mat2[3]) % MOD
    ]

def power(mat, n):
    """
    Raise a matrix to the power n using binary exponentiation.
    
    Args:
        mat (list): The matrix as a list of 4 elements.
        n (int): The exponent.

    Returns:
        list: The resulting matrix as a list of 4 elements.
    """
    result = [1, 0, 0, 1]  # Identity matrix
    base = mat[:]
    while n > 0:
        if n % 2 == 1:
            result = multiply(result, base)
        base = multiply(base, base)
        n //= 2
    return result

def fibonacci_mod(f0, f1, n):
    """
    Compute the n-th Fibonacci number modulo 10^9+7 using matrix exponentiation.
    
    Args:
        f0 (int): The 0th Fibonacci number.
        f1 (int): The 1st Fibonacci number.
        n (int): The position of the Fibonacci number to compute.

    Returns:
        int: The n-th Fibonacci number modulo 10^9+7.
    """
    if n == 0:
        return f0
    if n == 1:
        return f1
    
    mat = [1, 1, 1, 0]
    mat = power(mat, n - 1)
    MOD = 10**9 + 7
    return (mat[0] * f1 + mat[1] * f0) % MOD

def main():
    tc = int(input("Enter the number of test cases: "))
    for _ in range(tc):
        f0, f1, n = map(int, input("Enter f0, f1, n: ").strip().split())
        result = fibonacci_mod(f0, f1, n)
        print(result)

if __name__ == "__main__":
    main()
