import unittest
import json
import requests

class TestHttpServer(unittest.TestCase):

    def setUp(self):
        """Load configuration before running tests."""
        with open('config.json') as file:
            self.conf = json.load(file)

    def test_fibonacci_from_a_to_b(self):
        """Test the /fibonacci endpoint for a specific range."""
        url = f"http://localhost:{self.conf['http_server_port']}/fibonacci?from=9&to=13"
        response = requests.get(url)
        control_sequence = response.json()
        etalon_sequence = {9: 34, 10: 55, 11: 89, 12: 144}
        self.assertEqual(control_sequence, etalon_sequence)

    def test_get_sequence_from_a_to_b(self):
        """Test the /get-sequence endpoint for a specific range."""
        url = f"http://localhost:{self.conf['http_server_port']}/get-sequence?from=9&to=13"
        response = requests.get(url)
        control_sequence = response.json()
        etalon_sequence = {9: 34, 10: 55, 11: 89, 12: 144}
        self.assertEqual(control_sequence, etalon_sequence)

if __name__ == '__main__':
    unittest.main()
