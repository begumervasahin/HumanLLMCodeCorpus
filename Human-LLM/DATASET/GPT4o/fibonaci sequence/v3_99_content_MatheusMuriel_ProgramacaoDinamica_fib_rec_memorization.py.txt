import sys
from Memoria import Cerebro

class Memorization:
    def __init__(self, limit=10000000, cleaning_rate=0.7):
        """
        Initialize the Memorization class with a Cerebro instance.

        Args:
            limit (int): The memory limit for the Cerebro instance.
            cleaning_rate (float): The cleaning rate for the Cerebro instance.
        """
        self.memory = Cerebro(limit, cleaning_rate)

    def calculate(self, n):
        """
        Calculate the nth Fibonacci number using memoization.

        Args:
            n (int): The position in the Fibonacci sequence to calculate.

        Returns:
            int: The nth Fibonacci number.
        """
        if n == 0:
            return 0
        if n == 1:
            return 1

        # Convert indices to strings for use with the memory storage
        str_n1, str_n2 = str(n - 1), str(n - 2)

        # Retrieve the values from memory if they exist
        n_1 = self.memory.get_lembranca(str_n1)
        n_2 = self.memory.get_lembranca(str_n2)

        # If the values are not in memory, calculate them recursively
        if n_1 == -1:
            n_1 = self.calculate(n - 1)
            self.memory.nova_memoria(str_n1, n_1)

        if n_2 == -1:
            n_2 = self.calculate(n - 2)
            self.memory.nova_memoria(str_n2, n_2)

        return n_1 + n_2

    def get_calculation_function(self):
        """
        Get a lambda function to calculate Fibonacci numbers.

        Returns:
            function: A lambda function that calculates Fibonacci numbers.
        """
        return lambda n: self.calculate(n)

    def __str__(self):
        return "Recursive + Memoization"

def main():
    """
    Main function to handle user input and calculate the Fibonacci number.
    """
    if len(sys.argv) != 2:
        print("Usage: python script_name.py <number>")
        return

    try:
        n = int(sys.argv[1])
    except ValueError:
        print("Please enter a valid integer.")
        return

    fib_calculator = Memorization()
    result = fib_calculator.calculate(n)
    print(f"Fibonacci number for {n} is: {result}")

if __name__ == "__main__":
    main()
