from flask import Flask

app = Flask(__name__)
n = 0

def fibo(n: int) -> int:
    """Calculate the n-th Fibonacci number recursively."""
    if n < 1:
        return 0
    elif n == 1:
        return 1
    else:
        return fibo(n - 1) + fibo(n - 2)

@app.route("/")
def home():
    """Display the available routes."""
    return (
        f"Available Routes:<br/>"
        f"/previous<br/>"
        f"/current<br/>"
        f"/next<br/>"
    )

@app.route("/current")
def current_fib():
    """Return the current Fibonacci number."""
    global n
    return str(fibo(n))

@app.route("/next")
def next_fib():
    """Increment n and return the next Fibonacci number."""
    global n
    n += 1
    return str(fibo(n))

@app.route("/previous")
def prev_fib():
    """Decrement n and return the previous Fibonacci number."""
    global n
    if n > 0:
        n -= 1
    return str(fibo(n))

if __name__ == '__main__':
    app.run(debug=True)
