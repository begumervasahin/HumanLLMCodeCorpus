import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.tsa.stattools import adfuller, acf, pacf
from statsmodels.tsa.seasonal import seasonal_decompose
from statsmodels.tsa.arima_model import ARIMA

# Load the data
data = pd.read_csv('C:/Users/Dell/Desktop/Pairs Trading/FuturePrices.csv', parse_dates=['Date'], index_col='Date')

# Display basic information about the dataset
print(data.head())
print(data.dtypes)

# Select YESBANK data
ts = data['YESBANK']
print(ts.head(10))

# Plot the YESBANK time series
plt.figure()
plt.plot(ts)
plt.title('YESBANK Time Series')
plt.show()

# Function to test stationarity
def test_stationarity(timeseries):
    rolmean = timeseries.rolling(window=8).mean()
    rolstd = timeseries.rolling(window=8).std()

    plt.figure()
    plt.plot(timeseries, color='blue', label='Original')
    plt.plot(rolmean, color='red', label='Rolling Mean')
    plt.plot(rolstd, color='black', label='Rolling Std')
    plt.legend(loc='best')
    plt.title('Rolling Mean & Standard Deviation')
    plt.show()

    print('Dickey-Fuller Test:')
    dftest = adfuller(timeseries, autolag='AIC')
    dfoutput = pd.Series(dftest[0:4], index=['Test Statistic', 'p-value', '#Lags Used', 'Number of Observations Used'])
    for key, value in dftest[4].items():
        dfoutput[f'Critical Value ({key})'] = value
    print(dfoutput)

# Test stationarity of the original time series
test_stationarity(ts)

# Log transform the time series
ts_log = np.log(ts)
plt.figure()
plt.plot(ts_log)
plt.title('Log Transformed Time Series')
plt.show()

# Moving average
moving_avg = ts_log.rolling(window=8).mean()
plt.figure()
plt.plot(ts_log, label='Log Transformed')
plt.plot(moving_avg, color='red', label='Moving Average')
plt.legend(loc='best')
plt.title('Log Transformed with Moving Average')
plt.show()

# Remove moving average to get stationary data
ts_log_moving_avg_diff = ts_log - moving_avg
ts_log_moving_avg_diff.dropna(inplace=True)
test_stationarity(ts_log_moving_avg_diff)

# Exponential Weighted Moving Average (EWMA)
expwighted_avg = ts_log.ewm(halflife=8).mean()
plt.figure()
plt.plot(ts_log, label='Log Transformed')
plt.plot(expwighted_avg, color='red', label='EWMA')
plt.legend(loc='best')
plt.title('Log Transformed with EWMA')
plt.show()

# Remove EWMA to get stationary data
ts_log_ewma_diff = ts_log - expwighted_avg
test_stationarity(ts_log_ewma_diff)

# Differencing to make the series stationary
ts_log_diff = ts_log - ts_log.shift()
ts_log_diff.dropna(inplace=True)
plt.figure()
plt.plot(ts_log_diff)
plt.title('Differenced Log Transformed Time Series')
plt.show()
test_stationarity(ts_log_diff)

# Decompose the time series
decomposition = seasonal_decompose(ts_log, period=100)
trend = decomposition.trend
seasonal = decomposition.seasonal
residual = decomposition.resid

plt.figure()
plt.subplot(411)
plt.plot(ts_log, label='Original')
plt.legend(loc='best')
plt.subplot(412)
plt.plot(trend, label='Trend')
plt.legend(loc='best')
plt.subplot(413)
plt.plot(seasonal, label='Seasonality')
plt.legend(loc='best')
plt.subplot(414)
plt.plot(residual, label='Residuals')
plt.legend(loc='best')
plt.tight_layout()
plt.show()

# Test stationarity of the residuals
ts_log_decompose = residual
ts_log_decompose.dropna(inplace=True)
test_stationarity(ts_log_decompose)

# ACF and PACF plots
lag_acf = acf(ts_log_diff, nlags=20)
lag_pacf = pacf(ts_log_diff, nlags=20, method='ols')

plt.figure()
plt.subplot(121)
plt.plot(lag_acf)
plt.axhline(y=0, linestyle='--', color='gray')
plt.axhline(y=-1.96/np.sqrt(len(ts_log_diff)), linestyle='--', color='gray')
plt.axhline(y=1.96/np.sqrt(len(ts_log_diff)), linestyle='--', color='gray')
plt.title('Autocorrelation Function')
plt.subplot(122)
plt.plot(lag_pacf)
plt.axhline(y=0, linestyle='--', color='gray')
plt.axhline(y=-1.96/np.sqrt(len(ts_log_diff)), linestyle='--', color='gray')
plt.axhline(y=1.96/np.sqrt(len(ts_log_diff)), linestyle='--', color='gray')
plt.title('Partial Autocorrelation Function')
plt.tight_layout()
plt.show()

# Function to fit ARIMA model and plot results
def fit_arima_model(ts_log, ts_log_diff, order):
    model = ARIMA(ts_log, order=order)
    results = model.fit(disp=-1)
    plt.figure()
    plt.plot(ts_log_diff)
    plt.plot(results.fittedvalues, color='red')
    plt.title(f'RSS: {sum((results.fittedvalues-ts_log_diff)**2):.4f}')
    plt.show()
    return results

# Fit different ARIMA models
results_AR = fit_arima_model(ts_log, ts_log_diff, (2, 1, 0))
results_MA = fit_arima_model(ts_log, ts_log_diff, (0, 1, 2))
results_ARIMA = fit_arima_model(ts_log, ts_log_diff, (2, 1, 2))

# Predictions from the ARIMA model
predictions_ARIMA_diff = pd.Series(results_ARIMA.fittedvalues, copy=True)
predictions_ARIMA_diff_cumsum = predictions_ARIMA_diff.cumsum()
predictions_ARIMA_log = pd.Series(ts_log.iloc[0], index=ts_log.index)
predictions_ARIMA_log = predictions_ARIMA_log.add(predictions_ARIMA_diff_cumsum, fill_value=0)
predictions_ARIMA = np.exp(predictions_ARIMA_log)

# Plot predictions
plt.figure()
plt.plot(ts, label='Original')
plt.plot(predictions_ARIMA, label='Predicted')
plt.title(f'RMSE: {np.sqrt(sum((predictions_ARIMA-ts)**2)/len(ts)):.4f}')
plt.legend(loc='best')
plt.show()
