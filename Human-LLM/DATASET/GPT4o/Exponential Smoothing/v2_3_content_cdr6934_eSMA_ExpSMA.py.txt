import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

def generate_exponential_sma(data, window):
    """
    Generates exponential Simple Moving Average (eSMA) from data.
    
    Parameters:
    data (DataFrame or Series): Time series data.
    window (int): Sliding window size.
    
    Returns:
    DataFrame or Series: Exponential SMA result.
    """
    return data.ewm(span=window).mean()

def import_clean_ts(filename):
    """
    Reads and selects time series data from a CSV file.
    
    Parameters:
    filename (str): Path to the CSV file.
    
    Returns:
    DataFrame: Cleaned time series data sorted by date.
    """
    ts_data = pd.read_csv(filename)
    ts_data['Date'] = pd.to_datetime(ts_data['Date'])
    return ts_data.sort_values(by='Date')

def collect_user_input(window_default):
    """
    Collects window size from user or uses the default value.
    
    Parameters:
    window_default (int): Default window size in weeks.
    
    Returns:
    int: User-specified or default window size in days.
    """
    user_input = input(f'Enter Long Term Window or Enter to Default ({window_default} Weeks): ')
    if not user_input:
        user_input = window_default
    return int(user_input) * 5

def main():
    """
    Main function to read data, apply exponential smoothing, and plot results.
    """
    # Import and clean the stock data
    stock = import_clean_ts("data/goog.csv")
    data = pd.Series(np.float32(stock['Close']), index=stock['Date'])
    
    # Collect user input and generate exponential SMAs
    long_term_window = collect_user_input(4)
    short_term_window = collect_user_input(2)
    
    lt_av = generate_exponential_sma(data, long_term_window)
    st_av = generate_exponential_sma(data, short_term_window)
    
    # Calculate differences between long-term and short-term averages
    diff_ltst = lt_av - st_av
    diff_ltst_av = generate_exponential_sma(diff_ltst, 7.5)
    diff_av_diff = diff_ltst_av - diff_ltst
    
    # Plot the results
    diff_av_diff.plot(style='k', title="2nd Derivative of GOOG Close Price")
    plt.xlabel('Date')
    plt.ylabel('Value')
    plt.show()

if __name__ == '__main__':
    main()
