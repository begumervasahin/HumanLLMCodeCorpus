import pandas as pd
import numpy as np
import datetime
from datetime import timedelta
import time
import json

def prepare_data(input_file, output_file='processed_data.csv'):
    """
    Reads the pricing data, filters by the most recorded product ID, and prepares the data with simulated timestamps.

    Parameters:
    - input_file (str): Path to the input CSV file containing the pricing data.
    - output_file (str): Path to the output CSV file to save the processed data. Default is 'processed_data.csv'.
    """
    # Read and sort data by 'id'
    price_df = pd.read_csv(input_file)
    most_recorded_id = price_df['id'].value_counts().idxmax()
    
    # Filter data for the most recorded id and prepare the final list
    filtered_data = price_df[price_df['id'] == most_recorded_id][['id', 'dateUpdated', 'prices.amountMax']].copy()
    
    # Simulate new timestamps for the data
    current_time = datetime.datetime.now()
    for i in range(len(filtered_data)):
        filtered_data.iat[i, 1] = (current_time + timedelta(days=i)).strftime('%Y-%m-%dT%H:%M:%S.%fZ')
    
    # Rename columns and save to CSV
    filtered_data.columns = ['id', 'timestamp', 'demand']
    filtered_data.to_csv(output_file, index=False)

def exponential_smoothing_algorithm(file_path, alpha=0.2, log_cli=True):
    """
    Applies exponential smoothing to the data and calculates the Mean Absolute Deviation (MAD).

    Parameters:
    - file_path (str): Path to the CSV file containing the processed data.
    - alpha (float): Smoothing factor for exponential smoothing. Default is 0.2.
    - log_cli (bool): Flag to print log messages to the console. Default is True.

    Returns:
    - str: JSON string containing the MSE value and calculation time.
    """
    start_time = time.time()
    
    # Read data from CSV
    df = pd.read_csv(file_path)
    records = df.values.tolist()
    
    absolute_deviation = 0
    for index, item in enumerate(records):
        if index == 0:
            item.append(item[2])  # Initial prediction is the first actual value
        else:
            previous_predict_value = records[index - 1][3]
            previous_actual_value = records[index - 1][2]
            predicted_value = previous_predict_value + alpha * (previous_actual_value - previous_predict_value)
            item.append(predicted_value)
            absolute_deviation += abs(item[2] - item[3])
    
    mean_absolute_deviation = absolute_deviation / len(records)
    end_time = time.time()
    
    if log_cli:
        print('-------------------------')
        print(f'MAD: {mean_absolute_deviation:.4f}')
        print(f'Calculation time: {end_time - start_time:.4f} seconds')
        print('-------------------------')
    
    result = {
        'mse_value': mean_absolute_deviation,
        'calculate_time': round(end_time - start_time, 4)
    }
    return json.dumps(result)

def find_best_alpha(file_path, frequency=0.2, criterion='mse_value'):
    """
    Finds the best alpha value for exponential smoothing based on the specified criterion.

    Parameters:
    - file_path (str): Path to the CSV file containing the processed data.
    - frequency (float): Step size for alpha values. Default is 0.2.
    - criterion (str): Criterion for selecting the best alpha. Options are 'mse_value' and 'calculate_time'. Default is 'mse_value'.

    Returns:
    - float: The best alpha value.
    """
    alpha_values = np.arange(0.01, 1, frequency)
    best_alpha = 0.2
    min_mse_value = float('inf')
    min_calculate_time = float('inf')
    
    print('*********************')
    for alpha in alpha_values:
        print(f'Processing with alpha = {alpha}')
        result = json.loads(exponential_smoothing_algorithm(file_path=file_path, alpha=alpha, log_cli=False))
        mse_value = result['mse_value']
        calculate_time = result['calculate_time']
        
        if criterion == 'mse_value' and mse_value < min_mse_value:
            min_mse_value = mse_value
            min_calculate_time = calculate_time
            best_alpha = alpha
        
        if criterion == 'calculate_time' and calculate_time < min_calculate_time:
            min_mse_value = mse_value
            min_calculate_time = calculate_time
            best_alpha = alpha
    
    print(f'Best alpha: {best_alpha} with MSE value: {min_mse_value:.4f} and calculation time: {min_calculate_time:.4f} seconds')
    print('*********************')
    
    return best_alpha
