import psycopg2
import pandas as pd
import datetime

def process_anomalies(from_id, to_id):
    """
    Process anomalies from tick_anomalies table between specified IDs and insert the processed data into train_tick_data table.
    
    Parameters:
    from_id (int): Starting ID for processing anomalies.
    to_id (int): Ending ID for processing anomalies.
    """
    # Database connection parameters
    db_params = {
        "database": "postgres",
        "user": "postgres",
        "password": "swapnil",
        "host": "35.190.146.57",
        "port": "5432"
    }
    
    # Establish database connections
    conn = psycopg2.connect(**db_params)
    connw = psycopg2.connect(**db_params)
    
    # Load domain security data into a DataFrame
    with conn.cursor() as cur_sec:
        cur_sec.execute('SELECT * FROM domain_security')
        security_df = pd.DataFrame(cur_sec.fetchall(), columns=[desc[0] for desc in cur_sec.description])
    
    # Process tick anomalies
    with conn.cursor() as cur, connw.cursor() as curw:
        sql_stmt = f"""
            SELECT tickdate, security, id, anomaly_score 
            FROM tick_anomalies 
            WHERE id BETWEEN {from_id} AND {to_id} 
            ORDER BY id;
        """
        print(sql_stmt)
        cur.execute(sql_stmt)
        
        count = 0
        for row in cur.fetchall():
            anodate, security, anomid, anoscore = row
            security_id = int(security_df.loc[security_df['security'] == security, 'security_id'].iloc[0])
            
            sql_stmt = f"""
                SELECT tickdate, field, value 
                FROM import_raw_ticks 
                WHERE security='{security}' 
                AND field IN ('PX_LAST', 'VOLUME') 
                AND tickdate BETWEEN to_timestamp('{anodate.date()}', 'YYYY-MM-DD') - interval '30 day' 
                AND to_timestamp('{anodate.date()}', 'YYYY-MM-DD') + interval '10 day' 
                ORDER BY field, tickdate;
            """
            cur_read = conn.cursor()
            cur_read.execute(sql_stmt)
            result_df = pd.DataFrame(cur_read.fetchall(), columns=[desc[0] for desc in cur_read.description])
            
            price_df = result_df[result_df['field'] == 'PX_LAST']
            ffidx = price_df.index[price_df['tickdate'] == anodate]
            fidx = ffidx[0] if not ffidx.empty else 0
            
            if 10 < fidx < price_df.shape[0] - 2:
                label = 1 if price_df['value'].iloc[fidx + 1] > price_df['value'].iloc[fidx] else -1
                vol_df = result_df[result_df['field'] == 'VOLUME']
                vvidx = vol_df.index[vol_df['tickdate'] == anodate]
                vidx = vvidx[0] if not vvidx.empty else 0
                
                if vidx > 10:
                    sql_insert = f"""
                        INSERT INTO train_tick_data 
                        VALUES ({anomid}, '{anodate}', {anoscore}, '{security}', {security_id}, {label}, 
                        {', '.join(map(str, price_df['value'].iloc[fidx-9:fidx].tolist()))}, 
                        {', '.join(map(str, vol_df['value'].iloc[vidx-9:vidx].tolist()))});
                    """
                    curw.execute(sql_insert)
            
            count += 1
            
            if count % 1000 == 0:
                connw.commit()
                print(f"{datetime.datetime.now().time()} Count: {count}")
        
        connw.commit()
    
    conn.close()
    connw.close()
    print("DONE..........")

if __name__ == "__main__":
    process_anomalies(400001, 450000)
