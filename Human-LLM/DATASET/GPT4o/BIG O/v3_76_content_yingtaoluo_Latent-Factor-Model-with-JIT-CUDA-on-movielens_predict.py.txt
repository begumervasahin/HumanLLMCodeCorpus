import numpy as np
from numba import cuda, float64
import time
import heapq
import csv

# Constants
TPB = 32  # Threads per block

@cuda.jit
def fast_matmul(a, b, c, hid):
    """
    Perform matrix multiplication on the GPU using shared memory for optimization.

    Args:
        a: Input matrix A.
        b: Input matrix B.
        c: Output matrix C (result of A * B).
        hid: Number of tiles needed for the multiplication.
    """
    # Define shared memory arrays for the tiles of A and B
    sa = cuda.shared.array(shape=(TPB, TPB), dtype=float64)
    sb = cuda.shared.array(shape=(TPB, TPB), dtype=float64)

    # Calculate the global thread indices
    x, y = cuda.grid(2)
    tx = cuda.threadIdx.x
    ty = cuda.threadIdx.y

    # Check if the thread is within the bounds of the output matrix
    if x >= c.shape[0] or y >= c.shape[1]:
        return

    tmp = 0
    for i in range(hid):
        # Load tiles into shared memory
        sa[tx, ty] = a[x, ty + i * TPB]
        sb[tx, ty] = b[tx + i * TPB, y]
        cuda.syncthreads()

        # Perform the partial matrix multiplication
        for j in range(TPB):
            tmp += sa[tx, j] * sb[j, ty]
        cuda.syncthreads()

    # Write the result to the output matrix
    c[x, y] = tmp

def load_matrix(file_path):
    """
    Load a matrix from a .npy file.

    Args:
        file_path (str): The path to the .npy file.

    Returns:
        np.ndarray: The loaded matrix.
    """
    return np.load(file_path)

def save_predictions(predictions, file_path):
    """
    Save the predictions to a CSV file.

    Args:
        predictions (list): List of predictions to save.
        file_path (str): The path to the CSV file.
    """
    with open(file_path, 'w', newline='', encoding='utf-8') as csv_file:
        writer = csv.writer(csv_file)
        for item in predictions:
            writer.writerow(item)

def get_top_predictions(predictions, data, top_n=100):
    """
    Get the top N predictions.

    Args:
        predictions (np.ndarray): The array of predictions.
        data (list): The list of movie data.
        top_n (int): The number of top predictions to get.

    Returns:
        list: The top N predictions.
    """
    top_predictions_indices = heapq.nlargest(top_n, range(len(predictions)), predictions.take)
    return [data[idx] for idx in top_predictions_indices]

def main():
    # Matrix dimensions
    m_, n_, k = 640, 9760, 128
    hid = (k - 1) // TPB + 1

    # Load input matrices
    Q = load_matrix("Q.npy")
    P = load_matrix("P.npy")
    multiple = np.zeros((m_, n_))

    # Define CUDA grid and block dimensions
    blockdim = (TPB, TPB)
    griddim1 = ((m_ - 1) // TPB + 1, (n_ - 1) // TPB + 1)

    # NumPy dot product
    start_time = time.time()
    Movie_predict = np.dot(Q, P)
    print(f"np.dot elapsed time: {time.time() - start_time:.2f}s")

    # CUDA matrix multiplication
    start_time = time.time()
    fast_matmul[griddim1, blockdim](Q, P, multiple, hid)
    print(f"CUDA elapsed time: {time.time() - start_time:.2f}s")

    # Load movie data
    with open("movies.csv", encoding='utf-8') as csv_file:
        reader = csv.reader(csv_file)
        movie_data = [row for idx, row in enumerate(reader) if idx != 0]

    # Get top 100 movie predictions for the second user (index 1)
    top_predictions = get_top_predictions(Movie_predict[1, :], movie_data, top_n=100)

    # Save predictions to CSV
    save_predictions(top_predictions, 'top_100_movie_predictions.csv')

    print("Top 100 movie predictions saved to 'top_100_movie_predictions.csv'")
    print(top_predictions)

if __name__ == "__main__":
    main()
