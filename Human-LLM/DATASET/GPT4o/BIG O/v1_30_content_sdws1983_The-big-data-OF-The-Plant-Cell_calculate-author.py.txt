import sys
import getopt
import time
import pandas as pd
import os

def get_option():
    opts, args = getopt.getopt(sys.argv[1:], "hi:o:")
    input_file = ""
    output_file = ""
    h = ""
    for op, value in opts:
        if op == "-i":
            input_file = value
        elif op == "-o":
            output_file = value
        elif op == "-h":
            h = 'Usage:\nRemove the sequence which contains "N"\n-i : inputfile\n-o : outputfile\n'
    return input_file, output_file, h

def main(input_file, output_file):
    all_sequences = []
    with open(input_file) as f:
        for line in f:
            if line[0] == ">":
                continue
            else:
                all_sequences.append(line.strip())
    
    tag_all = {}
    for sequence in all_sequences:
        if sequence not in tag_all:
            tag_all[sequence] = 1
        else:
            tag_all[sequence] += 1
    
    with open('tmp.txt', 'w') as fou:
        for sequence, count in tag_all.items():
            fou.write(f"{sequence}\t{count}\n")
    
    data = pd.read_table('tmp.txt', names=['sequence', 'count'])
    data = data.sort_values(by='count', ascending=False)
    data.to_csv(output_file, sep='\t', index=False)
    
    os.remove('tmp.txt')

if __name__ == "__main__":
    time_start = time.time()
    input_file, output_file, h = get_option()
    if h == "":
        main(input_file, output_file)
        print("Execution time: {:.2f} seconds".format(time.time() - time_start))
    else:
        print(h)
