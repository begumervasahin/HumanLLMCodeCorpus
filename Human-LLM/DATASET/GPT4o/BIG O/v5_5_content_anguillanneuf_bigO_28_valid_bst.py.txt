This module implements a Binary Search Tree (BST) and includes methods 
to check whether a given binary tree is a valid BST.
"""

class BinaryTreeNode:
    """
    Binary Search Trees (BSTs) are great for quick lookups with O(logN) time complexity.
    """

    def __init__(self, value):
        """
        Initialize a Binary Tree node with a given value.
        
        Args:
            value (int): The value of the node.
        """
        self.value = value
        self.left = None
        self.right = None

    def insert_left(self, value):
        """
        Insert a new node with the given value to the left of the current node.
        
        Args:
            value (int): The value of the new node to be inserted.
        
        Returns:
            BinaryTreeNode: The newly inserted left node.
        """
        self.left = BinaryTreeNode(value)
        return self.left

    def insert_right(self, value):
        """
        Insert a new node with the given value to the right of the current node.
        
        Args:
            value (int): The value of the new node to be inserted.
        
        Returns:
            BinaryTreeNode: The newly inserted right node.
        """
        self.right = BinaryTreeNode(value)
        return self.right

    def check_binary_search_tree_iterative(self):
        """
        Iteratively check if the binary tree is a valid BST.
        
        Returns:
            bool: True if the tree is a valid BST, False otherwise.
        """
        stack = [(self, -float('inf'), float('inf'))]
        while stack:
            node, lower_bound, upper_bound = stack.pop()
            if not lower_bound < node.value < upper_bound:
                return False
            if node.left:
                stack.append((node.left, lower_bound, node.value))
            if node.right:
                stack.append((node.right, node.value, upper_bound))
        return True

    @staticmethod
    def check_bst_recursive(node, lower_bound, upper_bound):
        """
        Recursively check if the binary tree is a valid BST.
        
        Args:
            node (BinaryTreeNode): The current node being checked.
            lower_bound (float): The lower bound for the node value.
            upper_bound (float): The upper bound for the node value.
        
        Returns:
            bool: True if the tree is a valid BST, False otherwise.
        """
        if not node:
            return True
        if not lower_bound < node.value < upper_bound:
            return False
        return (BinaryTreeNode.check_bst_recursive(node.left, lower_bound, node.value) and
                BinaryTreeNode.check_bst_recursive(node.right, node.value, upper_bound))


# Example usage:
if __name__ == "__main__":
    # Create a binary tree
    bt = BinaryTreeNode(5)
    bt.insert_left(3)
    bt.insert_right(7)
    bt.left.insert_left(1)
    bt.left.insert_right(9)  # This will make the tree invalid as a BST

    # Check if the binary tree is a valid BST
    print("Iterative check:", bt.check_binary_search_tree_iterative())  # Output: False
    print("Recursive check:", BinaryTreeNode.check_bst_recursive(bt, -float('inf'), float('inf')))  # Output: False
