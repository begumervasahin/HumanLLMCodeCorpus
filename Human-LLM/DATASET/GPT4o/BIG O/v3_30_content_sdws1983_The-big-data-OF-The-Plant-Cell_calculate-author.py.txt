import sys
import getopt
import time
import pandas as pd
import os

def print_usage():
    """
    Prints the usage information for the script.
    """
    usage_info = (
        'Usage:\n'
        'Remove the sequences which contain "N".\n'
        '-i : inputfile\n'
        '-o : outputfile\n'
    )
    print(usage_info)

def get_options():
    """
    Parses and returns command line options.
    
    Returns:
        tuple: The input and output file paths.
    """
    try:
        opts, _ = getopt.getopt(sys.argv[1:], "hi:o:")
        input_file, output_file = "", ""
        for op, value in opts:
            if op == "-i":
                input_file = value
            elif op == "-o":
                output_file = value
            elif op == "-h":
                print_usage()
                sys.exit()
        if not input_file or not output_file:
            print_usage()
            sys.exit()
        return input_file, output_file
    except getopt.GetoptError:
        print_usage()
        sys.exit()

def process_sequences(input_file, output_file):
    """
    Reads sequences from the input file, counts occurrences, and writes sorted results to the output file.
    
    Args:
        input_file (str): The path to the input file.
        output_file (str): The path to the output file.
    """
    sequences = []

    # Read sequences from the input file
    with open(input_file, 'r') as f:
        for line in f:
            if not line.startswith(">"):
                sequences.append(line.strip())

    # Count occurrences of each sequence
    sequence_counts = {}
    for sequence in sequences:
        if sequence not in sequence_counts:
            sequence_counts[sequence] = 1
        else:
            sequence_counts[sequence] += 1

    # Write counts to a temporary file
    tmp_file = 'tmp.txt'
    with open(tmp_file, 'w') as temp_out:
        for sequence, count in sequence_counts.items():
            temp_out.write(f"{sequence}\t{count}\n")

    # Load data into pandas DataFrame, sort, and write to the output file
    data = pd.read_csv(tmp_file, sep='\t', names=['sequence', 'count'])
    data.sort_values(by='count', ascending=False, inplace=True)
    data.to_csv(output_file, sep='\t', index=False)

    # Remove the temporary file
    os.remove(tmp_file)

def main():
    """
    Main function to execute the script.
    """
    start_time = time.time()
    input_file, output_file = get_options()
    process_sequences(input_file, output_file)
    print(f"Execution time: {time.time() - start_time:.2f} seconds")

if __name__ == "__main__":
    main()
