import firebase_admin
from firebase_admin import firestore
import flask
from flask import request, jsonify, abort

# Initialize Flask app and enable CORS
app = flask.Flask(__name__)
CORS(app)

# Initialize Firebase app and Firestore client
firebase_admin.initialize_app()
db = firestore.client()

@app.route('/')

    return "Hello World"

@app.route('/game')
def game():
    season_id = request.args.get('seasonId')
    team_id = request.args.get('teamId')
    game_id = request.args.get('gameId')
    
    if not (season_id and team_id and game_id):
        return abort(400)
    
    game = get_game(db, season_id, team_id, game_id)
    
    if not game:
        return abort(404)
    
    return jsonify(game)

@app.route('/score')
def score():
    season_id = request.args.get('seasonId')
    team_id = request.args.get('teamId')
    game_id = request.args.get('gameId')
    data = request.args.get('data')
    portion = request.args.get('portion')
    
    if not (season_id and team_id and game_id):
        return abort(400)
    
    game = get_game(db, season_id, team_id, game_id)
    opp_game = get_opp_game(db, season_id, team_id, game_id)
    
    if not game:
        return abort(404)
    
    team = get_team_by_id(db, game["TEAM_ID"])
    opp_team = get_team_by_id(db, opp_game["TEAM_ID"])
    
    if not (team and opp_team):
        return abort(404)
    
    if not data:
        data_season_id = int(game["SEASON_ID"])
        data, portion = _get_default_data(data_season_id, game)
    
    if portion == "before" and game["SEASON_ID"] != data:
        return abort(422)
    
    list_data_games = get_games_stats(db, data, portion, team_id, game)
    
    if not list_data_games:
        return abort(404)
    
    response = _get_scores(list_data_games, game)
    response["team"] = team
    response["opp_team"] = opp_team
    
    return jsonify(response)

@app.route('/games')
def games():
    season_id = request.args.get('seasonId')
    team_id = request.args.get('teamId')
    with_obt = request.args.get('withOBT')
    with_summary = request.args.get('withSummary')
    
    if not (season_id and team_id):
        return abort(400)
    
    games = get_games_info_by_team_and_season(db, season_id, team_id, with_obt, with_summary)
    
    return jsonify(games)

if __name__ == '__main__':
