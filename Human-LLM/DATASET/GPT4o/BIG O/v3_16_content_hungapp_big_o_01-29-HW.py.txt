def check_array_elements():
    """
    Check if the k-th smallest element in array 'a' is less than the m-th largest element in array 'b'.
    """
    # Read sizes of arrays 'a' and 'b'
    na, nb = map(int, input().split())
    
    # Read the positions k and m
    k, m = map(int, input().split())
    
    # Read arrays 'a' and 'b'
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))

    # Check if the k-th smallest element in 'a' is less than the m-th largest element in 'b'
    if a[k - 1] < b[-m]:
        print("YES")
    else:
        print("NO")

def find_min_max_index():
    """
    Find the index of the element where the value in list 'a' is the minimum
    and the corresponding value in list 'b' is the maximum.
    """
    n = int(input())  # Read the number of elements
    
    a = []
    b = []

    # Read the elements into lists 'a' and 'b'
    for _ in range(n):
        x, y = map(int, input().split())
        a.append(x)
        b.append(y)

    min_value = min(a)
    max_value = max(b)
    answer = -1

    # Find the index where the element in 'a' is min_value and the corresponding element in 'b' is max_value
    for i in range(n):
        if a[i] == min_value and b[i] == max_value:
            answer = i + 1  # +1 to convert zero-based index to one-based index
            break

    print(answer)

def compare_arrays():
    """
    Compare two arrays 'a' and 'b' and output two binary sequences
    indicating the selection from each array.
    """
    n = int(input())  # Read the number of elements
    
    a = []
    b = []

    # Read the elements into lists 'a' and 'b'
    for _ in range(n):
        x, y = map(int, input().split())
        a.append(x)
        b.append(y)

    final_1 = [False] * n
    final_2 = [False] * n

    i = j = chosen = 0

    # Compare elements from both arrays
    while chosen < n:
        if a[i] < b[j]:
            final_1[i] = True
            i += 1
        else:
            final_2[j] = True
            j += 1
        chosen += 1

    # Output the binary sequence for array 'a'
    for i in range(n):
        if final_1[i] or i < n // 2:
            print(1, end='')
        else:
            print(0, end='')
    print()

    # Output the binary sequence for array 'b'
    for i in range(n):
        if final_2[i] or i < n // 2:
            print(1, end='')
        else:
            print(0, end='')
    print()

def calculate_ways():
    """
    Calculate the number of ways to pick pairs of genres from an array 'a' of genres.
    """
    n, m = map(int, input().split())  # Read the number of elements and the number of genres
    a = list(map(int, input().split()))  # Read the genre array

    count = [0] * (m + 1)

    # Count occurrences of each genre
    for genre in a:
        count[genre] += 1

    ways = 0

    # Calculate the number of ways to pick pairs of different genres
    for i in range(1, m):
        for j in range(i + 1, m + 1):
            ways += count[i] * count[j]

    print(ways)
