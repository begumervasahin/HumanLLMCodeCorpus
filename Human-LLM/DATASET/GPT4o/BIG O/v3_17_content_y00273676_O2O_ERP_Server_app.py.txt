class Application(web.Application):
    def __init__(self):
        # Define application settings
        settings = {
            'xsrf_cookies': False,
            'compress_response': True,
            'debug': options.options.debug,
            'ui_modules': uimodules,
            'ui_methods': uimethods,
            'static_path': STATIC_PATH,
            'cookie_secret': base64.b64encode(uuid.uuid3(uuid.NAMESPACE_DNS, 'android').bytes).decode(),
            'sentry_url': SENTRY_URL if not options.options.debug else ''
        }

        # Initialize the Tornado web application with the specified settings
        super().__init__(**settings)

        # Add URL handlers to the application
        for host_pattern, handlers in URLS:
            self.add_handlers(host_pattern, handlers)

def run_server():
    """
    Run the Tornado server.
    """
    app = Application()
    app.sentry_client = AsyncSentryClient(app.settings['sentry_url'])
    http_server = httpserver.HTTPServer(app, xheaders=True)
    http_server.listen(options.options.port)
    print(f'Server is running on port {options.options.port}')

if __name__ == "__main__":
    # Define command-line options
    options.define('port', default=8000, help='Port to listen on')
    options.define('debug', default=False, help='Run in debug mode')

    # Parse command-line options
    options.parse_command_line()

    # Run the server
    run_server()
