from itertools import product, combinations

class TextFormatter:
    def __init__(self, text="", width=80):
        self.text = text
        self.width = width
        self.word_borders = []
        self.linebreaks = []

    def find_word_borders(self):
        """Find the starting indices of words in the text."""
        for i in range(len(self.text)):
            if not self.text[i].isspace():
                if i == 0 or self.text[i - 1].isspace():
                    self.word_borders.append(i)
        self.word_borders.append(len(self.text))

    def break_text_naive(self):
        """Naive implementation of breaking text into lines."""
        self.find_word_borders()
        lines = ['']
        offset = 0
        for b in self.word_borders:
            if b - offset < self.width:
                lines[-1] += self.text[offset:b]
            else:
                lines.append(self.text[offset:b])
                offset = b
        return lines

    def find_next_linebreak(self, start_index, reverse=False):
        """Find the next optimal line break index."""
        if not reverse:
            if start_index == len(self.word_borders) - 1:
                return None
            i, j = start_index, len(self.word_borders) - 1
            sign = 1
        else:
            if start_index == 0:
                return None
            i, j = 0, start_index
            sign = -1

        while i < j:
            mid = (i + j) // 2
            if self.word_borders[mid - 1] <= self.word_borders[start_index] + sign * self.width <= self.word_borders[mid]:
                start_index = mid - 1 * (not reverse)
                break
            elif self.word_borders[mid] > self.word_borders[start_index] + sign * self.width:
                j = mid
            else:
                i = mid
        return start_index

    def find_linebreaks(self):
        """Find optimal line breaks to minimize jaggedness."""
        front_push, back_push = [], []
        p = 0
        while p is not None and p <= len(self.word_borders):
            front_push.append((p, self.word_borders[p]))
            p = self.find_next_linebreak(p)

        p = len(self.word_borders) - 1
        while p is not None and p >= 0:
            back_push.append((p, self.word_borders[p]))
            p = self.find_next_linebreak(p, reverse=True)

        min_cost = float('inf')
        possible_ranges = [range(back[0], front[0] + 1) for back, front in zip(back_push[::-1], front_push)]

        for scenario in product(*possible_ranges):
            cost = 0
            for k in range(1, len(scenario)):
                diff = self.word_borders[scenario[k]] - self.word_borders[scenario[k - 1]]
                if diff > self.width:
                    cost = float('inf')
                    break
                else:
                    cost += (self.width - diff) ** 2
            if cost < min_cost:
                min_cost = cost
                self.linebreaks = [self.word_borders[k] for k in scenario]

        for b in range(1, len(self.linebreaks)):
            print(self.text[self.linebreaks[b - 1]:self.linebreaks[b]])

    def find_linebreaks_brute_force(self):
        """Find line breaks using brute force."""
        self.find_word_borders()
        borders = self.word_borders[1:-1]
        n = len(self.text) // self.width
        min_cost = float('inf')

        for scenario in combinations(borders, n):
            cost = 0
            scenario = [0] + list(scenario) + [len(self.text)]
            for k in range(1, len(scenario)):
                diff = scenario[k] - scenario[k - 1]
                if diff > self.width:
                    cost = float('inf')
                    break
                else:
                    cost += (self.width - diff) ** 2
            if cost < min_cost:
                min_cost = cost
                self.linebreaks = scenario

        for b in range(1, len(self.linebreaks)):
            print(self.text[self.linebreaks[b - 1]:self.linebreaks[b]])
