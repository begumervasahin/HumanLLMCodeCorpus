from pyspark import SparkConf, SparkContext
import sys
import re

# Define constants for file paths
ALL_MOVIES = 'hdfs:///user/kirvenjt/oscar_data/omdb_responses'
TEST_OUTPUT = 'hdfs:///user/kirvenjt/oscar_data/actor_experience'
IMDB_FILE = 'hdfs:///user/kirvenjt/oscar_data/top10ActorsPerMovie.txt'

def create_y(spark):
    """
    Process the IMDB and movie data to generate a dataset of actor experiences
    and save the results to HDFS.
    """
    # Load IMDB and movie data from HDFS
    imdb_data = spark.textFile(IMDB_FILE)
    movie_data = spark.textFile(ALL_MOVIES)
    
    # Process IMDB data: extract fields, filter valid entries, and key by (title, year)
    imdb_data = (imdb_data.map(get_imdb_fields)
                           .filter(lambda x: len(x) == 5)
                           .keyBy(lambda x: (x[0], get_int_clean(x[1]))))
    
    # Process movie data: extract fields, filter valid entries, map to (title, year), and key by (title, year)
    movie_data = (movie_data.map(get_fields)
                             .filter(lambda x: has_name_and_year(x[1], x[2]))
                             .map(lambda x: (x[1], x[2]))
                             .keyBy(lambda x: (x[0], get_int_clean(x[1]))))
    
    # Join movie data with IMDB data, extract unique actor experiences, and save to HDFS
    (movie_data.join(imdb_data)
               .map(lambda key_info: tuple(key_info[1][1]))
               .distinct()
               .saveAsTextFile(TEST_OUTPUT))

def has_name_and_year(title, year):
    """
    Check if the title is non-empty and the year is a valid 4-digit number.
    """
    return title != "" and len(year) == 4

def get_int_clean(item):
    """
    Extract a 4-digit integer from the input string. Return 0 if extraction fails.
    """
    try:
        return int(re.findall(r'\d{4}', item).pop())
    except Exception:
        return 0

def get_fields(text):
    """
    Convert a string representation of a list into an actual list.
    """
    return eval(text)

def get_imdb_fields(text):
    """
    Split the input string by '|' and strip whitespace from each field.
    """
    return [x.strip() for x in text.split('|')]

if __name__ == '__main__':
    # Initialize Spark configuration
    conf = SparkConf()

    # Set Spark master based on command line argument
    if len(sys.argv) > 1:
        if sys.argv[1] == 'local':
            conf.setMaster("local[3]")
            print('Running locally')
        elif sys.argv[1] == 'cluster':
            conf.setMaster("spark://10.0.22.241:7077")
            print('Running on cluster')

    # Set Spark memory configurations
    conf.set("spark.executor.memory", "10g")
    conf.set("spark.driver.memory", "10g")
    
    # Create Spark context and execute the main function
    spark = SparkContext(conf=conf)
    create_y(spark)
