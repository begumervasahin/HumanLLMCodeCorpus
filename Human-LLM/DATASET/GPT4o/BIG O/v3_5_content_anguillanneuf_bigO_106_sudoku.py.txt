def update_cell(row, col, board):
    """
    Update the possible choices for a cell in the Sudoku board.

    Args:
        row (int): The row index of the cell.
        col (int): The column index of the cell.
        board (list of list of str): The Sudoku board.

    Returns:
        list of str: A list of possible values that can be placed in the cell.
    """
    choices = []
    for k in map(str, range(1, 10)):
        collision = False
        for y in range(9):
            if (board[row][y] == k or
                board[y][col] == k or
                board[row - row % 3 + y // 3][col - col % 3 + y % 3] == k):
                collision = True
                break
        if not collision:
            choices.append(k)
    return choices

def print_board(board):
    """
    Print the Sudoku board in a readable format.

    Args:
        board (list of list of str): The Sudoku board.
    """
    for row in board:
        print(" ".join(row))

def sudoku_solve(board):
    """
    Solve the Sudoku board using a backtracking algorithm.

    Args:
        board (list of list of str): The Sudoku board.

    Returns:
        bool: True if the board is solved, False otherwise.
    """
    choices = None
    row, col = -1, -1

    # Find the cell with the fewest possible choices
    for i in range(9):
        for j in range(9):
            if board[i][j] == '.':
                temp_choices = update_cell(i, j, board)
                if choices is None or (len(temp_choices) > 0 and len(temp_choices) < len(choices)):
                    choices = temp_choices
                    row, col = i, j
    
    # If there are no empty cells left, the board is solved
    if choices is None:
        print_board(board)
        return True
    
    # Try each possible choice for the current cell
    for value in choices:
        board[row][col] = value
        if sudoku_solve(board):
            return True
        board[row][col] = '.'  # Reset the cell if the choice did not lead to a solution
    
    return False

def solve_and_print(board, board_name):
    """
    Solve a given Sudoku board and print the result.

    Args:
        board (list of list of str): The Sudoku board to be solved.
        board_name (str): The name of the board for display purposes.
    """
    print(f"Solving {board_name}:")
    if sudoku_solve(board):
        print(f"{board_name} solved successfully!\n")
    else:
        print(f"{board_name} could not be solved.\n")
