import sys
import operator

def main():
    # Dictionary to store the key-value pairs
    storage = {}

    # Reading input from standard input (stdin)
    for line in sys.stdin:
        value, key = line.split()
        storage[key] = int(value)

    # Sorting the dictionary by value in descending order
    sorted_storage = sorted(storage.items(), key=operator.itemgetter(1), reverse=True)

    # Printing the sorted key-value pairs
    for key, value in sorted_storage:
        print(f"{key},{value}")

if __name__ == "__main__":
    main()
