import sys
import wotdapp

def make_mod(username):
    """
    Promote a user to moderator.
    
    Args:
        username (str): The username of the user to be promoted.
    
    Returns:
        str: Status message.
    """
    with wotdapp.app.app_context():
        user = wotdapp.models.User.query.filter_by(username=username).first()
        if user is None:
            return f'{username} is not a recognized user'
        if user.isMod:
            return f'{username} is already a moderator'
        user.isMod = True
        wotdapp.db.session.add(user)
        wotdapp.db.session.commit()
        return f'{username} is now a moderator'

def remove_mod(username):
    """
    Demote a moderator to a regular user.
    
    Args:
        username (str): The username of the moderator to be demoted.
    
    Returns:
        str: Status message.
    """
    with wotdapp.app.app_context():
        user = wotdapp.models.User.query.filter_by(username=username).first()
        if user is None:
            return f'{username} is not a recognized user'
        if not user.isMod:
            return f'{username} is not a current moderator'
        user.isMod = False
        wotdapp.db.session.add(user)
        wotdapp.db.session.commit()
        return f'{username} is no longer a moderator'

def replace_db():
    """
    Drop all tables and recreate the database schema.
    
    Returns:
        str: Status message.
    """
    with wotdapp.app.app_context():
        wotdapp.db.drop_all()
        wotdapp.db.create_all()
        return 'Dropped all tables. Created all tables.'

if __name__ == '__main__':
    if len(sys.argv) < 2:
        print("Usage: python script.py <method> [username]")
        sys.exit(1)

    method = sys.argv[1]

    if method == 'mod':
        if len(sys.argv) != 3:
            print("Usage: python script.py mod <username>")
            sys.exit(1)
        print(make_mod(sys.argv[2]))

    elif method == 'rmmod':
        if len(sys.argv) != 3:
            print("Usage: python script.py rmmod <username>")
            sys.exit(1)
        print(remove_mod(sys.argv[2]))

    elif method == 'newdb':
        wipe = input('This will delete all data. Are you sure? (Y/n): ')
        if wipe.lower() == 'y':
            print(replace_db())
        else:
            print("Operation cancelled.")
    
    else:
        print(f"Unknown method: {method}")
        print("Usage: python script.py <method> [username]")
        sys.exit(1)
