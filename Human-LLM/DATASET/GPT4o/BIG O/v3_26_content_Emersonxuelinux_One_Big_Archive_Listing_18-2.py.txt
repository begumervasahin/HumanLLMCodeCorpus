import pygame
import sys

# Initialize Pygame
pygame.init()

# Set up display
SCREEN_WIDTH, SCREEN_HEIGHT = 640, 480
screen = pygame.display.set_mode([SCREEN_WIDTH, SCREEN_HEIGHT])
pygame.display.set_caption("Ball Movement")

# Set up background
background = pygame.Surface(screen.get_size())
background.fill([255, 255, 255])
clock = pygame.time.Clock()

class Ball(pygame.sprite.Sprite):
    def __init__(self, image_file, speed, location):
        super().__init__()
        self.image = pygame.image.load(image_file)
        self.rect = self.image.get_rect()
        self.rect.topleft = location
        self.speed = speed

    def move(self):
        """Move the ball and bounce off the screen edges."""
        if self.rect.left <= 0 or self.rect.right >= SCREEN_WIDTH:
            self.speed[0] = -self.speed[0]
        self.rect = self.rect.move(self.speed)

def handle_events(ball):
    """Handle user input events."""
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            pygame.quit()
            sys.exit()
        elif event.type == pygame.KEYDOWN:
            if event.key == pygame.K_UP:
                ball.rect.y -= 10
            elif event.key == pygame.K_DOWN:
                ball.rect.y += 10

def update_screen(ball):
    """Update the screen with the ball's current position."""
    screen.blit(background, (0, 0))
    ball.move()
    screen.blit(ball.image, ball.rect)
    pygame.display.flip()

def main():
    """Main function to run the game loop."""
    my_ball = Ball('beach_ball.png', [10, 0], [20, 20])

    while True:
        handle_events(my_ball)
        update_screen(my_ball)
        clock.tick(30)  # Control the frame rate

if __name__ == "__main__":
    main()
