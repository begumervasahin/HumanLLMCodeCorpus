def update_cell(row, col, board):
    """
    Update the possible choices for a cell in the Sudoku board.

    Args:
        row (int): The row index of the cell.
        col (int): The column index of the cell.
        board (list of list of str): The Sudoku board.

    Returns:
        list of str: A list of possible values that can be placed in the cell.
    """
    choices = []
    for k in [str(v) for v in range(1, 10)]:
        collision = False
        for y in range(9):
            if board[row][y] == k or \
               board[y][col] == k or \
               board[row - row % 3 + y // 3][col - col % 3 + y % 3] == k:
                collision = True
                break
        if not collision:
            choices.append(k)
    return choices

def print_board(board):
    """
    Print the Sudoku board in a readable format.

    Args:
        board (list of list of str): The Sudoku board.
    """
    for row in board:
        print(" ".join(row))

def sudoku_solve(board):
    """
    Solve the Sudoku board using a backtracking algorithm.

    Args:
        board (list of list of str): The Sudoku board.

    Returns:
        bool: True if the board is solved, False otherwise.
    """
    choices = None
    row, col = -1, -1
    
    # Find the cell with the fewest possible choices
    for i in range(9):
        for j in range(9):
            if board[i][j] == '.':
                temp = update_cell(i, j, board)
                if choices is None or (len(temp) > 0 and len(temp) < len(choices)):
                    choices = temp
                    row, col = i, j
    
    # If there are no empty cells left, the board is solved
    if choices is None:
        print_board(board)
        return True
    
    # Try each possible choice for the current cell
    for v in choices:
        board[row][col] = v
        if sudoku_solve(board):
            return True
        board[row][col] = '.'  # Reset the cell if the choice did not lead to a solution
    
    return False
