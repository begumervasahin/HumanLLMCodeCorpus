import os
import sys
import json
import shutil

class Organize:
    def __init__(self, folder='Downloads'):
        self.folder = self.get_folder_path(folder)
        self.extensions_map = self.load_extensions()

    def get_folder_path(self, folder):
        """Generate absolute folder path from a relative folder name."""
        path = os.path.join('~', folder)
        return os.path.expanduser(path)

    def load_extensions(self):
        """Load file extensions mapping from a JSON file."""
        with open('extensions.json', 'r') as extension_file:
            return json.load(extension_file)

    def execute(self):
        """Execute the main flow of organizing files."""
        print(f'\nWorking in directory: {self.folder}\n')
        files = os.listdir(self.folder)

        for file in files:
            file_name, file_extension = os.path.splitext(file)
            target_folder = self.map_folder(file_extension[1:])

            if target_folder:
                target_folder = target_folder.upper()
                print(f'{target_folder} <--- {file}')
                file_path = os.path.join(self.folder, file)
                destination_folder = os.path.join(self.folder, target_folder)

                if not os.path.isdir(destination_folder):
                    os.mkdir(destination_folder)
                
                self.move_file(file_path, destination_folder)

        print(f'\nProcess complete.\n')

    def map_folder(self, extension):
        """Map file extension to target folder based on JSON configuration."""
        for folder, extensions in self.extensions_map.items():
            if extension.lower() in extensions:
                return folder
        return None

    def move_file(self, file_path, destination_folder):
        """Move file to the target folder, handling potential conflicts."""
        try:
            shutil.move(file_path, destination_folder)
        except shutil.Error as err:
            print(f"Replacing old file >>> {err}")
            shutil.copy(file_path, destination_folder)
            os.remove(file_path)

if __name__ == '__main__':
    arguments = sys.argv

    try:
        if len(arguments) > 1:
            for argument in arguments[1:]:
                Organize(argument).execute()
        else:
            Organize().execute()
    except FileNotFoundError as err:
        sys.exit(f'No such directory found. {err}')
