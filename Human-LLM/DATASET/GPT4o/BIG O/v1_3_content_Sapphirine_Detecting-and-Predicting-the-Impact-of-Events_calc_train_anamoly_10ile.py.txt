import psycopg2
import pandas as pd
import datetime

def process_anomalies(from_id, to_id):
    # Database connection parameters
    db_params = {
        "database": "postgres",
        "user": "postgres",
        "password": "swapnil",
        "host": "35.190.146.57",
        "port": "5432"
    }

    # Establish connections to the database
    conn = psycopg2.connect(**db_params)
    connw = psycopg2.connect(**db_params)
    
    # Load security data into DataFrame
    with conn.cursor() as cursec:
        cursec.execute('SELECT * FROM domain_security')
        security_df = pd.DataFrame(cursec.fetchall(), columns=[desc[0] for desc in cursec.description])
    
    # Query tick anomalies
    sql_stmt = f"""
    SELECT tickdate, security, id, anamoly_score 
    FROM tick_anomalies 
    WHERE id BETWEEN {from_id} AND {to_id} 
    ORDER BY id;
    """
    print(sql_stmt)

    with conn.cursor() as cur:
        cur.execute(sql_stmt)
        anomalies = cur.fetchall()
    
    count = 0
    
    for row in anomalies:
        anodate, security, anomid, anoscore = row
        security_id = int(security_df[security_df['security'] == security]['security_id'])
        
        sql_stmt = f"""
        SELECT tickdate, field, value 
        FROM import_raw_ticks 
        WHERE security = '{security}' AND field IN ('PX_LAST', 'VOLUME') AND 
        tickdate BETWEEN to_timestamp('{anodate.date()}', 'YYYY-MM-DD') - INTERVAL '30 day' AND 
        to_timestamp('{anodate.date()}', 'YYYY-MM-DD') + INTERVAL '10 day' 
        ORDER BY field, tickdate;
        """
        
        with conn.cursor() as curread:
            curread.execute(sql_stmt)
            result_df = pd.DataFrame(curread.fetchall(), columns=[desc[0] for desc in curread.description])
        
        price_df = result_df[result_df['field'] == 'PX_LAST']
        ffidx = price_df.index[price_df['tickdate'] == anodate]
        fidx = ffidx[0] if ffidx.shape[0] > 0 else 0
        
        if fidx > 10 and fidx < price_df.shape[0] - 2:
            label = 1 if price_df['value'].iloc[fidx + 1] > price_df['value'].iloc[fidx] else -1
            vol_df = result_df[result_df['field'] == 'VOLUME']
            vvidx = vol_df.index[vol_df['tickdate'] == anodate]
            vidx = vvidx[0] if vvidx.shape[0] > 0 else 0
            
            if vidx > 10:
                values = [anomid, anodate, anoscore, security, security_id, label] + \
                         price_df['value'].iloc[fidx-9:fidx+1].tolist() + \
                         vol_df['value'].iloc[vidx-9:vidx+1].tolist()
                sql_insert = f"""
                INSERT INTO train_tick_data VALUES ({', '.join(map(str, values))});
                """
                
                with connw.cursor() as curw:
                    curw.execute(sql_insert)
        
        count += 1
        if count % 1000 == 0:
            connw.commit()
            print(f"{datetime.datetime.now().time()} - Count: {count}")

    connw.commit()
    conn.close()
    connw.close()
    print("DONE..........")

if __name__ == '__main__':
    process_anomalies(0, 200000)
