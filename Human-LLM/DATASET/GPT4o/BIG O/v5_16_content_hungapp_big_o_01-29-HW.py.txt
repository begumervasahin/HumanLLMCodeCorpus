# Input number of books and genres
n, m = map(int, input().split())

# Input list of genres for the books
a = list(map(int, input().split()))

# Initialize a list to count occurrences of each genre
count = [0] * (m + 1)

# Count occurrences of each genre
for genre in a:
    count[genre] += 1

# Calculate the number of ways to pick two different genres
ways = 0
for i in range(1, m):
    for j in range(i + 1, m + 1):
        ways += count[i] * count[j]

print(ways)
