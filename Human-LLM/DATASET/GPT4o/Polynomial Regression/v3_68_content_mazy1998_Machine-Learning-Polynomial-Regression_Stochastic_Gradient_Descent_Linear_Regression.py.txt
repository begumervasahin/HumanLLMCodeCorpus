import random
from tkinter import *
import time

class LinearRegressionAnimation(Frame):
    def __init__(self):
        super().__init__()
        self.pack()
        
        self.num_points = 600
        self.canvas = Canvas(self, width=self.num_points, height=self.num_points, bg="black")
        self.canvas.grid()
        
        self.initial_slope = random.uniform(-1, 1)
        self.initial_intercept = random.uniform(-1, 1)

        self.data_points = self.generate_data()
        self.draw_data_points()

        self.current_slope = self.initial_slope
        self.current_intercept = self.initial_intercept

        self.animate_regression()

    def generate_data(self):
        def linear_function(x):
            return 0.5 * x + 250

        return [(i, linear_function(i) + random.randint(-10, 10)) for i in range(self.num_points)]

    def draw_data_points(self):
        for x, y in self.data_points:
            self.canvas.create_line(x, self.num_points - y, x + 1, self.num_points - y + 1, fill="green")

    def draw_regression_line(self, slope, intercept, color="red"):
        return self.canvas.create_line(
            0, self.num_points - intercept,
            self.num_points, self.num_points - (slope * self.num_points + intercept),
            fill=color
        )

    def animate_regression(self):
        num_iterations = 120
        learning_rate = 0.001

        for _ in range(num_iterations):
            self.current_intercept += (self.data_points[0][1] - self.current_intercept) * 0.05
            for x, y in self.data_points:
                prediction = self.current_slope * x + self.current_intercept
                error = y - prediction
                self.current_slope += (1 / len(self.data_points)) * x * error * learning_rate

            regression_line = self.draw_regression_line(self.current_slope, self.current_intercept)
            self.update_canvas(regression_line)

        final_line = self.draw_regression_line(self.current_slope, self.current_intercept)
        print(self.current_slope, self.current_intercept)

    def update_canvas(self, line_id):
        self.canvas.update()
        time.sleep(0.1)
        self.canvas.after(10, self.canvas.delete, line_id)

if __name__ == "__main__":
    app = LinearRegressionAnimation()
    app.mainloop()
