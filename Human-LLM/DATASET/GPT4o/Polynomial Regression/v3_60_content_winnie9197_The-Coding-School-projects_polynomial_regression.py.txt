import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn import linear_model, preprocessing
from sklearn.metrics import mean_squared_error, r2_score
import operator
import scipy.stats as stats

# Set plot style
plt.style.use('seaborn-whitegrid')

# Function to generate a synthetic dataset
def generate_data_set(seed=0, num_samples=20):
    np.random.seed(seed)
    x = 2 - 3 * np.random.normal(0, 1, num_samples)
    y = x - 2 * (x ** 2) + 0.5 * (x ** 3) + np.random.normal(-3, 3, num_samples)
    x = x[:, np.newaxis]
    y = y[:, np.newaxis]
    return x, y

# Generate training data
x_train, y_train = generate_data_set()

# Transform features to polynomial features
degree = 2
poly_features = preprocessing.PolynomialFeatures(degree=degree)
x_train_poly = poly_features.fit_transform(x_train)

# Train the polynomial regression model
poly_model = linear_model.LinearRegression()
poly_model.fit(x_train_poly, y_train)

# Predict using the polynomial regression model
y_poly_pred = poly_model.predict(x_train_poly)

# Plot the polynomial regression results
def plot_polynomial_regression(x, y, y_pred):
    sorted_zip = sorted(zip(x, y_pred), key=operator.itemgetter(0))
    x_sorted, y_pred_sorted = zip(*sorted_zip)
    
    plt.scatter(x, y, color='black')
    plt.plot(x_sorted, y_pred_sorted, linewidth=3, color='blue')
    plt.title("Polynomial Regression")
    plt.xlabel("Observed Values")
    plt.ylabel("Predicted Values")
    plt.show()

plot_polynomial_regression(x_train, y_train, y_poly_pred)

# Plot a Q-Q plot for normality check
def plot_qq(x):
    stats.probplot(x[:, 0], dist="norm", plot=plt)
    plt.title("Q-Q Plot for Normality")
    plt.xlabel("Quantiles")
    plt.ylabel("Observed Values")
    plt.show()

plot_qq(x_train)

# Calculate residuals
residuals_train = y_poly_pred - y_train

# Plot residuals versus predicted values
def plot_residuals_vs_predicted(residuals, y_pred):
    plt.scatter(y_pred, residuals, color='black')
    plt.axhline(0, color='red', linestyle='--')
    plt.title("Residuals Versus Predicted Values")
    plt.xlabel("Predicted Values")
    plt.ylabel("Residuals")
    plt.show()

plot_residuals_vs_predicted(residuals_train, y_poly_pred)

# Function to calculate Variance Inflation Factor (VIF)
def calculate_vif(exogenous_vars, data):
    vif_dict, tolerance_dict = {}, {}
    for exo in exogenous_vars:
        endog_vars = [i for i in exogenous_vars if i != exo]
        X, y = data[endog_vars], data[exo]
        r_squared = linear_model.LinearRegression().fit(X, y).score(X, y)
        vif = 1 / (1 - r_squared)
        tolerance = 1 - r_squared
        vif_dict[exo] = vif
        tolerance_dict[exo] = tolerance
    
    df_vif = pd.DataFrame({'VIF': vif_dict, 'Tolerance': tolerance_dict})
    return df_vif

# Evaluate model performance
rmse_train = np.sqrt(mean_squared_error(y_train, y_poly_pred))
r2_train = r2_score(y_train, y_poly_pred)

print("The model performance for the training set")
print("-------------------------------------------")
print(f"RMSE for training: {rmse_train:.5f}")
print(f"R2 for training: {r2_train:.5f}")
