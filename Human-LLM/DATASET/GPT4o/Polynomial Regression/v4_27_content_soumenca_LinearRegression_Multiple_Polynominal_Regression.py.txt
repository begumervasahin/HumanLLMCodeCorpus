import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.preprocessing import PolynomialFeatures
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error
from sklearn.model_selection import train_test_split

# Load the dataset
data = pd.read_csv('/home/soumen/Desktop/TML_HW3_AT/winequalityRed.csv', header=None)
print(f"Shape of the training data is {data.shape}")

# Prepare features and target variable
X = data.iloc[:, :-1].values
y = data.iloc[:, -1].values

# Split the data into training and testing sets
x_train, x_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=1)

# Lists to store MSE values
train_errors = []
test_errors = []

# Iterate over polynomial degrees
for degree in range(1, 3):
    # Transform features to polynomial features
    poly = PolynomialFeatures(degree=degree)
    x_train_poly = poly.fit_transform(x_train)
    x_test_poly = poly.transform(x_test)
    
    # Fit the model
    model = LinearRegression()
    model.fit(x_train_poly, y_train)
    
    # Predict and calculate errors
    y_train_pred = model.predict(x_train_poly)
    y_test_pred = model.predict(x_test_poly)
    
    train_mse = mean_squared_error(y_train, y_train_pred)
    test_mse = mean_squared_error(y_test, y_test_pred)
    
    train_errors.append(train_mse)
    test_errors.append(test_mse)
    
    print(f"Degree of the polynomial: {degree}")
    print(f"Training MSE: {train_mse}")
    print(f"Testing MSE: {test_mse}")
    print('----------------------------------')

# Plot the results
plt.figure(figsize=(10, 6))

plt.subplot(2, 1, 1)
plt.plot(range(1, 3), train_errors, marker='o')
plt.ylabel('Training Error (MSE)')
plt.xlabel('Degree of Polynomial')
plt.grid(True)

plt.subplot(2, 1, 2)
plt.plot(range(1, 3), test_errors, marker='o')
plt.ylabel('Testing Error (MSE)')
plt.xlabel('Degree of Polynomial')
plt.grid(True)

plt.tight_layout()
plt.show()
