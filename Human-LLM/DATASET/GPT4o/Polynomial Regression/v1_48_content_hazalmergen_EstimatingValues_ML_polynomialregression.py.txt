import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split, KFold
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import PolynomialFeatures
from sklearn.metrics import explained_variance_score, mean_absolute_error, mean_squared_error, r2_score

# Load dataset
dataset = pd.read_csv('datas.csv')

# Select features and target variable
X = dataset.iloc[0:100, 1:6].values
y = dataset.iloc[0:100, 6].values

# Initialize KFold cross-validation
kf = KFold(n_splits=10)

# Initialize lists to store true values and predictions
y_tests = []
y_preds = []

# Perform KFold cross-validation
for train_index, test_index in kf.split(X):
    X_train, X_test = X[train_index], X[test_index]
    y_train, y_test = y[train_index], y[test_index]
    
    # Polynomial feature transformation
    poly_reg = PolynomialFeatures(degree=3)
    X_poly = poly_reg.fit_transform(X_train)
    
    # Linear regression model
    lin_reg = LinearRegression()
    lin_reg.fit(X_poly, y_train)
    
    # Make predictions
    y_pred = lin_reg.predict(poly_reg.transform(X_test))
    
    # Store true values and predictions
    y_preds += list(y_pred)
    y_tests += list(y_test)

# Plot true values vs predictions
plt.scatter(y_tests, y_preds)
plt.plot(y_tests, y_tests, color="black")
plt.xlabel("True Values")
plt.ylabel("Predictions")
plt.show()

# Calculate evaluation metrics
ex_var_score = explained_variance_score(y_tests, y_preds)
m_absolute_error = mean_absolute_error(y_tests, y_preds)
m_squared_error = mean_squared_error(y_tests, y_preds)
r_2_score = r2_score(y_tests, y_preds)

# Print evaluation metrics
print("Explained Variance Score: ", ex_var_score)
print("Mean Absolute Error: ", m_absolute_error)
print("Mean Squared Error: ", m_squared_error)
print("R Squared Error: ", r_2_score)
