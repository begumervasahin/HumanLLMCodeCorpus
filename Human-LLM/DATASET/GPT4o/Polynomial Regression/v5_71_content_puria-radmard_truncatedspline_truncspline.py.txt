import pandas as pd
import seaborn as sns
from sklearn.linear_model import LinearRegression
from sklearn.pipeline import make_pipeline

# Load the dataset
wagedf = pd.read_csv(
    filepath_or_buffer="https://raw.githubusercontent.com/puria-radmard/ISLR-python/master/Notebooks/Data/Wage.csv", 
    index_col=0
)

class TruncatedSpline:
    def __init__(self, degree=3):
        """
        Initialize the Truncated Spline class with a given degree.
        Degree represents the power to which the splines will be computed.
        """
        self.degree = degree

    def fit(self, X, y, knots):
        """
        Fit the model by setting up the knots for truncation.
        
        Parameters:
        - X: DataFrame containing predictors
        - y: Series containing the target variable
        - knots: Dictionary where keys are predictor names and values are lists of knots
        """
        if type(X) != pd.DataFrame:
            X = pd.DataFrame(data=X)
        
        self.cols = list(knots.keys())
        self.truncs = {}

        for param in knots:
            for k in knots[param]:
                self.truncs[f"{param}_{k}"] = f"(df['{param}'] - {k}) * (df['{param}'].apply(lambda x: int(x > {k})))"
        
        return self

    def transform(self, X):
        """
        Transform the input DataFrame by adding truncated spline features.
        
        Parameters:
        - X: DataFrame containing the predictors
        
        Returns:
        - Transformed DataFrame with additional spline features
        """
        df = X.copy()[self.cols]

        for trunc in self.truncs:
            for n in range(1, self.degree + 1):
                df[f"{trunc}_^{n}"] = eval(self.truncs[trunc])**n

        for col in self.cols:
            for n in range(2, self.degree + 1):
                df[f"{col}_^{n}"] = df[col]**n

        return pd.concat((X, df), axis=1)

# Prepare the dataset with one-hot encoding
testdf = pd.get_dummies(wagedf[["age", "race", "wage"]])

# Initialize and fit the TruncatedSpline model
truncated_spline = TruncatedSpline(degree=3).fit(
    X=pd.get_dummies(wagedf[["age", "race"]]),
    y=testdf["wage"],
    knots={"age": [40, 60]}
)

# Transform the dataset using the fitted TruncatedSpline
transformed_df = truncated_spline.transform(testdf)

# Create a pipeline with TruncatedSpline and LinearRegression
pipe = make_pipeline(TruncatedSpline(), LinearRegression())

# Fit the pipeline to the data
pipe.fit(
    pd.get_dummies(wagedf[["age", "race"]]), 
    wagedf["wage"], 
    truncatedspline__knots={"age": [30, 40, 50, 60, 70]}
)

# Extract the TruncatedSpline step from the pipeline
print(pipe.named_steps['truncatedspline'])

# Predict wages using a simple linear regression model
linear_model = LinearRegression()
linear_model.fit(
    pd.get_dummies(wagedf[["age", "race"]]),
    wagedf["wage"]
)
predicted_wages_simple = linear_model.predict(pd.get_dummies(wagedf[["age", "race"]]))

# Plot the actual wages and the simple linear regression predictions
sns.scatterplot(x=wagedf["age"], y=wagedf["wage"], s=10)
sns.lineplot(x=wagedf["age"], y=predicted_wages_simple)

# Predict wages using the pipeline with truncated splines
predicted_wages_pipe = pipe.predict(pd.get_dummies(wagedf[["age", "race"]]))

# Plot the actual wages and the pipeline predictions
sns.scatterplot(x=wagedf["age"], y=wagedf["wage"], s=10)
sns.lineplot(x=wagedf["age"], y=predicted_wages_pipe)

# Show the plots
sns.plt.show()
