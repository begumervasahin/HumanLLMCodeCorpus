import numpy as np
import matplotlib.pyplot as plt

# Load datasets
datasets = [np.loadtxt(f'data/svar-set{i}.dat.txt') for i in range(1, 5)]

def linear_regression_train(z, y):
    """Train linear regression model."""
    return np.linalg.pinv(z).dot(y)

def linear_regression_fit(z, y):
    """Fit linear regression model and return fitted values and parameters."""
    theta = linear_regression_train(z, y)
    y_fit = z.dot(theta)
    return y_fit, theta

def linear_regression_predict(theta, x_in):
    """Predict using linear regression model."""
    z = np.ones((1, x_in.size + 1))
    z[0, 1:] = x_in
    return z.dot(theta)

def create_design_matrix(x, order=1):
    """Create a design matrix for polynomial regression."""
    m, n = x.shape
    z = np.ones((m, n + order))
    for i in range(1, order + 1):
        z[:, n + i - 1] = np.power(x[:, 0], i)
    return z

def calculate_average_sse(y_fit, y):
    """Calculate the average sum of squared errors."""
    return np.mean(np.square(y_fit - y))

def split_data(x, n_splits):
    """Split data into n_splits parts."""
    row, col = x.shape
    split_size = row // n_splits
    return np.array([x[i * split_size:(i + 1) * split_size] for i in range(n_splits)])

def combine_except(data_splits, exclude_index):
    """Combine data splits except one specified part."""
    test_data = data_splits[exclude_index]
    train_data = np.vstack([data_splits[i] for i in range(len(data_splits)) if i != exclude_index])
    print(f"Training shape: {train_data.shape}")
    print(f"Testing shape: {test_data.shape}")
    return train_data, test_data

def run_experiment(data):
    """Run single linear regression experiment."""
    X = data[:, :-1]
    y = data[:, -1:].reshape(-1, 1)

    # Linear regression
    Z_linear = create_design_matrix(X, order=1)
    y_fit_linear, _ = linear_regression_fit(Z_linear, y)

    # Polynomial regression of degree 2
    Z_quad = create_design_matrix(X, order=2)
    y_fit_quad, _ = linear_regression_fit(Z_quad, y)

    # Polynomial regression of degree 3
    Z_cubic = create_design_matrix(X, order=3)
    y_fit_cubic, _ = linear_regression_fit(Z_cubic, y)

    # Calculate average SSE
    sse_linear = calculate_average_sse(y_fit_linear, y)
    sse_quad = calculate_average_sse(y_fit_quad, y)
    sse_cubic = calculate_average_sse(y_fit_cubic, y)

    print(f"Average SSE for linear: {sse_linear:.4f}, Polynomial of degree 2: {sse_quad:.4f}, Polynomial of degree 3: {sse_cubic:.4f}")

    # Plot results
    plt.plot(X, y, 'r+', label='Data')
    plt.plot(X, y_fit_linear, '.', label='Linear Fit')
    plt.plot(X, y_fit_quad, 'r.', label='Quadratic Fit')
    plt.plot(X, y_fit_cubic, 'g.', label='Cubic Fit')
    plt.legend()
    plt.show()

# Execute experiments
for dataset in datasets:
    run_experiment(dataset)

# Split data and combine except one part
data_splits = split_data(datasets[0], 10)
combine_except(data_splits, 1)
