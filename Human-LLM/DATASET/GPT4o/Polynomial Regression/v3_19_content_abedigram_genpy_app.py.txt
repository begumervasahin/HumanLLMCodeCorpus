import random
import copy
import numpy as np
import matplotlib.pyplot as plt
import Ind

def crossover(parents):
    """
    Perform crossover between two parent individuals to produce a child.

    Args:
    - parents (list of Ind.Ind): List containing two parent individuals.

    Returns:
    - child (Ind.Ind): The child individual after crossover.
    """
    child = Ind.Ind()
    r1, r2 = random.sample(range(4), 2)
    
    # Inherit two coefficients from the first parent
    child.cof[r1] = parents[0].cof[r1]
    child.cof[r2] = parents[0].cof[r2]
    
    # Inherit the remaining two coefficients from the second parent
    remaining_indices = [i for i in range(4) if i not in (r1, r2)]
    child.cof[remaining_indices[0]] = parents[1].cof[remaining_indices[0]]
    child.cof[remaining_indices[1]] = parents[1].cof[remaining_indices[1]]
    
    return child

def mutation(individual, mutation_rate, sigma):
    """
    Perform mutation on an individual.

    Args:
    - individual (Ind.Ind): The individual to be mutated.
    - mutation_rate (float): The probability of mutation occurring.
    - sigma (float): The standard deviation of the Gaussian noise added during mutation.

    Returns:
    - individual (Ind.Ind): The mutated individual.
    """
    if random.random() <= mutation_rate:
        noise = np.random.normal(0, sigma, 4)
        individual.cof += noise
    return individual

def generate_next_population(population, mutation_rate, sigma, ys, xs):
    """
    Generate the next population through selection, crossover, and mutation.

    Args:
    - population (list of Ind.Ind): The current population of individuals.
    - mutation_rate (float): The probability of mutation occurring.
    - sigma (float): The standard deviation of the Gaussian noise added during mutation.
    - ys (list of float): The target y-values for fitness calculation.
    - xs (list of float): The x-values for fitness calculation.

    Returns:
    - next_population (list of Ind.Ind): The next generation of the population.
    """
    # Select pairs of parents
    parents_pairs = [
        (copy.deepcopy(random.choice(population)), copy.deepcopy(random.choice(population))) 
        for _ in range(len(population))
    ]
    
    # Perform crossover and mutation to produce new individuals
    offspring = [crossover(pair) for pair in parents_pairs]
    offspring = [mutation(ind, mutation_rate, sigma) for ind in offspring]
    
    # Evaluate fitness of the new individuals
    for ind in offspring:
        ind.fit_ness(ys, xs)
    
    # Combine current population and offspring, then select the top individuals
    combined_population = population + offspring
    combined_population.sort(key=lambda ind: ind.fitness, reverse=True)
    
    return combined_population[:50]

def read_input(file_path):
    """
    Read input data from a CSV file.

    Args:
    - file_path (str): Path to the input CSV file.

    Returns:
    - ys (list of float): List of y-values read from the file.
    """
    with open(file_path, 'r') as f:
        ys = [float(f.readline().strip()) for _ in range(100)]
    return ys

def initialize_population(population_size):
    """
    Initialize a population of individuals.

    Args:
    - population_size (int): The size of the population to initialize.

    Returns:
    - population (list of Ind.Ind): The initialized population.
    """
    return [Ind.Ind() for _ in range(population_size)]

def evaluate_population(population, ys, xs):
    """
    Evaluate the fitness of a population.

    Args:
    - population (list of Ind.Ind): The population to evaluate.
    - ys (list of float): The target y-values for fitness calculation.
    - xs (list of float): The x-values for fitness calculation.
    """
    for ind in population:
        ind.fit_ness(ys, xs)

def plot_results(xs, ys, best_fit_curve, generations_count, best_fitness, mean_fitness, worst_fitness):
    """
    Plot the results of the genetic algorithm.

    Args:
    - xs (list of float): The x-values for the plot.
    - ys (list of float): The target y-values for the plot.
    - best_fit_curve (list of float): The y-values of the best fit curve.
    - generations_count (list of int): List of generation numbers.
    - best_fitness (list of float): List of best fitness values over generations.
    - mean_fitness (list of float): List of mean fitness values over generations.
    - worst_fitness (list of float): List of worst fitness values over generations.
    """
    plt.plot(xs, ys, 'co', label='Data')
    plt.plot(xs, best_fit_curve, '-g', label='Best Fit')
    plt.legend()
    plt.show()
    
    plt.plot(generations_count, best_fitness, 'g^', label='Best')
    plt.plot(generations_count, mean_fitness, 'co', label='Mean')
    plt.plot(generations_count, worst_fitness, 'rs', label='Worst')
    plt.legend()
    plt.show()

def main(generations, population_size, tournament_size, mutation_rate, sigma):
    """
    Main function to run the genetic algorithm.

    Args:
    - generations (int): The number of generations to evolve.
    - population_size (int): The size of the population.
    - tournament_size (int): The size of the tournament for selection.
    - mutation_rate (float): The probability of mutation occurring.
    - sigma (float): The standard deviation of the Gaussian noise added during mutation.
    """
    # Read input data
    ys = read_input('input.csv')
    xs = [round(x * 0.1, 1) for x in range(100)]
    
    # Initialize population
    population = initialize_population(population_size)
    evaluate_population(population, ys, xs)
    
    # Evolution loop
    best_fitness, mean_fitness, worst_fitness, generations_count = [], [], [], []
    for gen_num in range(generations):
        population = generate_next_population(population, mutation_rate, sigma, ys, xs)
        generations_count.append(gen_num)
        best_fitness.append(population[0].fitness)
        mean_fitness.append((population[0].fitness + population[-1].fitness) / 2)
        worst_fitness.append(population[-1].fitness)
        print(f'Generation {gen_num + 1} - Best Fitness: {population[0].fitness}')
    
    # Prepare data for plotting
    x = np.linspace(0, 10, 100)
    best_cof = population[0].cof
    best_fit_curve = best_cof[3] * x**3 + best_cof[2] * x**2 + best_cof[1] * x + best_cof[0]
    
    # Plot results
    plot_results(xs, ys, best_fit_curve, generations_count, best_fitness, mean_fitness, worst_fitness)

if __name__ == "__main__":
    main(generations=5000, population_size=50, tournament_size=2, mutation_rate=0.1, sigma=0.1)
