import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import time

def main():
    # Load and clean the data
    data = pd.read_csv('CleanData.csv')
    data.drop(['Unnamed: 0'], axis=1, inplace=True)
    data.drop(data.index[[0, 1, 5, 9, 73, 74, 75, 710, 711, 712, 713, 714, 715, 716, 717, 718, 719]], inplace=True)

    # Display data info and statistics
    print(data.info())
    print(data.describe())

    # Extract data for plotting and model training/testing
    x = data['Decimal Date'].tolist()
    y = data['Carbon Dioxide (ppm)'].tolist()
    z = data['Carbon Dioxide Fit (ppm)'].tolist()
    a = data['Seasonally Adjusted CO2 (ppm)'].tolist()
    b = data['Seasonally Adjusted CO2 Fit (ppm)'].tolist()

    x_train = x[0:600]
    a_train = a[0:600]
    x_test = x[601:703]
    a_actual = a[601:703]

    degree = 2

    # Polynomial fitting
    start_time = time.time()
    coef = np.polyfit(x_train, a_train, degree)
    print("Polynomial Coefficients:", coef.tolist())

    # Generate training curve
    training_curve = [np.polyval(coef, x) for x in x_train]

    # Plot the training data and regression line
    plt.scatter(x_train, a_train, color='red', label='Carbon Dioxide (ppm)')
    plt.plot(x_train, training_curve, color='blue', label='Regression Line')
    plt.legend(loc='upper left')
    
    # Calculate training error
    training_error = np.mean((np.array(a_train) - np.array(training_curve)) ** 2)
    print("==================Error During Training=======================")
    print("MSE =", training_error)

    # Predict on the test data
    predicted_list = [np.polyval(coef, x) for x in x_test]

    # Display predictions
    print("==================Prediction===========================")
    for year, prediction in zip(x_test, predicted_list):
        print(f"Predicted value of CO2 (in ppm) for Year {year}: {prediction}")

    # Calculate prediction error
    prediction_error = np.mean((np.array(a_actual) - np.array(predicted_list)) ** 2)
    print("==================Prediction Error=======================")
    print("MSE for Prediction =", prediction_error)

    # Display execution time
    end_time = time.time()
    print("Approx execution time:", end_time - start_time, "seconds")

    # Show plot
    plt.show()

if __name__ == "__main__":
    main()
