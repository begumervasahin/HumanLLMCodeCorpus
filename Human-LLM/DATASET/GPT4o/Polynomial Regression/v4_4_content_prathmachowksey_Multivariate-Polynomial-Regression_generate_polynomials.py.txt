import numpy as np
import pandas as pd
from itertools import combinations_with_replacement

# Function to generate combinations of powers up to a given degree
def get_powers(degree):
    l = [0, 1]
    powers = []
    for i in range(1, degree + 1):
        powers.extend(combinations_with_replacement(l, i))
    return powers

# Function to transform the data based on the given powers
def transform_data(X, powers):
    X_new = np.ones((X.shape[0], len(powers)))
    for n in range(X.shape[0]):
        for i, power in enumerate(powers):
            for p in power:
                X_new[n, i] *= X[n, p]
    return X_new

# Load and preprocess the data
data = pd.read_csv('3D_spatial_network.txt', names=["id", "latitude", "longitude", "altitude"])
print(data.head())

# Drop the 'id' column and standardize the data
data = data.drop("id", axis=1)
data = (data - data.mean()) / data.std()
print(data.head())

# Extract features and target
X = data.iloc[:, 0:2].values
y = data.iloc[:, 2:3].values

# Generate powers and transform the data for different degrees
degrees = [1, 2, 3, 4, 5, 6]
transformed_data = {}

for degree in degrees:
    powers = get_powers(degree)
    X_transformed = transform_data(X, powers)
    X_transformed = (X_transformed - X_transformed.mean(axis=0)) / X_transformed.std(axis=0)
    transformed_data[degree] = X_transformed

# Standardize the target variable
y = (y - y.mean()) / y.std()

# Save the transformed data and target variable to .npy files
for degree, X_transformed in transformed_data.items():
    np.save(f'X_{degree}.npy', X_transformed)

np.save('y.npy', y)

# Print sample data for verification
print("Sample powers for degree 3:", get_powers(3))
print("Original X samples:", X[:3])
print("Transformed X samples for degree 3:", transformed_data[3][:3])
