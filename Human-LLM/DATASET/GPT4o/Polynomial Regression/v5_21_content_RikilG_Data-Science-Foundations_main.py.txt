import PolynomialModel
import DataUtils
import pandas as pd
import numpy as np
from time import time
import matplotlib.pyplot as plt

# Load and preprocess the dataset
def load_and_preprocess_data(file_path, drop_column, degree):
    dataset = pd.read_csv(file_path)
    dataset = dataset.drop(columns=drop_column)
    dataset = PolynomialModel.transform_dataset(dataset, degree)
    dataset = DataUtils.normalize(dataset, type="min-max")
    return dataset

# Split the dataset into training and testing sets
def split_data(dataset, split_ratio):
    train, test = DataUtils.data_split(dataset, split_at=split_ratio)
    x_train, y_train = DataUtils.xy_split(train)
    x_test, y_test = DataUtils.xy_split(test)
    return x_train, y_train, x_test, y_test

# Add a constant term to the features
def add_constant_term(x_train, x_test):
    x_train.insert(0, "Const", np.ones(x_train.shape[0]))
    x_test.insert(0, "Const", np.ones(x_test.shape[0]))
    return x_train, x_test

# Fit the model and plot validation errors
def fit_and_plot_model(train, x_test, y_test, method, degree, alpha, epsilon):
    w_list, lambdas, val_errs, train_errs = PolynomialModel.reg_fit(train, alpha=alpha, epsilon=epsilon, method=method, degree=degree)
    
    test_errs = [PolynomialModel.test(w, x_test, y_test) for w in w_list]
    
    plt.title(f'Error w.r.t lambda - degree: {degree}, method: {method}')
    plt.ylabel('Error')
    plt.xlabel('Lambda')
    plt.plot(lambdas, val_errs, label="Validation Error")
    plt.legend()
    plt.grid(True)
    plt.show()
    
    return w_list, lambdas, test_errs

# Select the best lambda and corresponding weights
def select_best_lambda(w_list, lambdas):
    min_index = np.argmin(lambdas)
    best_lambda = lambdas[min_index]
    best_weights = w_list[min_index]
    return best_lambda, best_weights

# Calculate and print errors
def calculate_and_print_errors(w, x_train, y_train, x_test, y_test):
    train_error_mse = PolynomialModel.error(w, x_train.values, y_train.values)
    test_error_mse = PolynomialModel.test(w, x_test, y_test)
    print('Train Error (MSE):', train_error_mse)
    print('Test Error (MSE):', test_error_mse)
    print('Train Error (RMSE):', np.sqrt(train_error_mse))
    print('Test Error (RMSE):', np.sqrt(test_error_mse))
    print('Train Error (R2):', PolynomialModel.r2_error(w, x_train, y_train))
    print('Test Error (R2):', PolynomialModel.r2_error(w, x_test, y_test))

# Main script
def main():
    file_path = "dataset.csv"
    drop_column = "OSM_ID"
    degree = 1
    method = "L2GD"
    alpha = 7e-7
    epsilon = 1e-3
    split_ratio = 0.80

    dataset = load_and_preprocess_data(file_path, drop_column, degree)
    x_train, y_train, x_test, y_test = split_data(dataset, split_ratio)
    x_train, x_test = add_constant_term(x_train, x_test)

    print(f"Using polynomial of degree: {degree}")
    st_time = time()

    w_list, lambdas, test_errs = fit_and_plot_model((x_train, y_train), x_test, y_test, method, degree, alpha, epsilon)
    
    best_lambda, best_weights = select_best_lambda(w_list, lambdas)
    print("\nSelected Regularization parameter: ", best_lambda)
    print(f"\nExecution Time: {time() - st_time}")
    print(best_weights)

    calculate_and_print_errors(best_weights, x_train, y_train, x_test, y_test)

if __name__ == "__main__":
    main()
