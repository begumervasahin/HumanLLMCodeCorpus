import numpy as np
import lmfit as lm
import itertools

class MA_Coefficients:
    def __init__(self, y, acf, q, index):
        self.y = y
        self.acf = acf
        self.q = q
        self.index = index

    @staticmethod
    def fitter_fn(params, x1, x2, x3, data):
        a = params['a']
        b = params['b']
        k = params['k']
        model = [k * k for k in x1] + [a * i for i in x2] + [b * i for i in x3]
        return [m - d for m, d in zip(model, data)]

    def get_thetas(self, y, et):
        x1 = et
        x2 = et[1:]
        x3 = et[2:]
        
        params = lm.Parameters()
        params.add('a', value=0, min=-1, max=1)
        params.add('b', value=0, min=-1, max=1)
        params.add('k', value=0, min=-1, max=1)
        
        result = lm.minimize(self.fitter_fn, params, args=(x1, x2, x3, y))
        
        final_thetas = [
            result.params['a'].value,
            result.params['b'].value,
            result.params['k'].value
        ]
        
        return final_thetas

    def prelim_thetas(self, acf, q):
        p = [[1 - acf[1], 1, -acf[1]]]
        eliminations = []
        prelim_theta = []

        x1_arr = np.roots(p[0]).tolist()
        for val in x1_arr:
            if not isinstance(val, complex) and (val > 1 or val < -1):
                eliminations.append(val)
        
        theta = [val for val in x1_arr if val not in eliminations]

        if q == 2:
            eliminations = []
            for k in theta:
                p.append([acf[1] + acf[2], 1 - 2 * k, acf[1] + acf[2] + (acf[1] + acf[2]) * k ** 2 + k])
                x2_arr = np.roots(p.pop()).tolist()
                for val in x2_arr:
                    if not isinstance(val, complex) and (val > 1 or val < -1):
                        eliminations.append(val)
            
            theta_2 = [val for val in x2_arr if val not in eliminations]
            prelim_theta = list(itertools.product(theta, theta_2))
        else:
            prelim_theta = theta

        return prelim_theta

    def get_ma_coeff(self):
        prelim_theta = self.prelim_thetas(self.acf, self.q)
        et_list = []

        if len(self.y) < 10:
            print("Input time series not suitable for forecasting")
            return

        for theta in prelim_theta:
            if isinstance(theta, float):
                et = [0, self.y[0], self.y[1] + theta * self.y[0]]
            else:
                et = [
                    0, 
                    self.y[0], 
                    self.y[1] + theta[0] * self.y[0], 
                    self.y[2] + theta[1] * self.y[1] + theta[1] * theta[0] * self.y[0]
                ]
            et_list.append(et)

        final_thetas = [self.get_thetas(self.y, et) for et in et_list]

        if self.index >= len(final_thetas):
            self.index = 0

        return final_thetas[self.index][:self.q]
