import numpy as np
import matplotlib.pyplot as plt

def generate_poly_points(start, stop, num_points, coeff, noise_level=1, plot=1):
    """
    Generate evenly spaced x and y coordinates in Euclidean space from a polynomial with Gaussian noise.
    
    Parameters
    ----------
    start : float
        Minimum value for the generated x-coordinates.
    stop : float
        Maximum value for the generated x-coordinates.
    num_points : int
        Number of coordinates to generate.
    coeff : list or array-like
        Coefficients of the polynomial. The index of each coefficient corresponds to the power of x.
    noise_level : float, optional
        Indicates the amount of Gaussian noise to add to the y-coordinates generated from the polynomial.
        Zero indicates no noise, while each integer increment results in a ten-fold noise increase.
    plot : int, optional
        If 1, a matplotlib plot of the generated coordinates is displayed.
    
    Returns
    -------
    x_pts : ndarray
        Generated x-coordinates.
    y_pts : ndarray
        Generated y-coordinates.
    """
    
    # Generate evenly spaced x-coordinates between start and stop
    x_pts = np.linspace(start, stop, num_points)
    
    # Compute y-coordinates using the polynomial coefficients
    y_pts = np.polyval(coeff[::-1], x_pts)
    
    # Add Gaussian noise to y-coordinates if noise_level is greater than 0
    if noise_level > 0:
        noise = np.random.normal(0, 10 ** noise_level, num_points)
        y_pts += noise
    
    # Plot the generated points if plot is set to 1
    if plot == 1:
        plt.figure()
        plt.scatter(x_pts, y_pts, label='Data points')
        plt.xlabel('x')
        plt.ylabel('y')
        plt.title('Generated Polynomial Points with Noise')
        plt.legend()
        plt.show()
    
    return x_pts, y_pts

# Example usage
if __name__ == "__main__":
    start = 0
    stop = 10
    num_points = 100
    coeff = [1, 2, 3]  # Example polynomial coefficients for 1 + 2*x + 3*x^2
    noise_level = 1

    x_pts, y_pts = generate_poly_points(start, stop, num_points, coeff, noise_level, plot=1)
