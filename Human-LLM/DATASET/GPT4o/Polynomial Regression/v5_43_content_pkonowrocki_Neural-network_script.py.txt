import csvReader as reader
import neuralNetwork as nn
import activationFunctions as F
import lossFunctions as L
import networkPrinter as printer

def classification3Classes():
    """Train and validate a neural network for 3-class classification."""
    # Load training and testing data
    X_train, Y_train = reader.readClassification3ClassesFile('classification/data.three_gauss.train.100.csv')
    X_test, Y_test = reader.readClassification3ClassesFile('classification/data.three_gauss.test.100.csv')
    
    # Initialize the neural network
    net = nn.NeuralNetwork(momentumSize=0, seed=0)
    net.addLayer(2, 20, F.LReLU, False)
    net.addLayer(20, 3, F.softmax, False)
    net.setCostFunction(L.crossEntropy)
    
    # Train and validate the network using 5-fold cross-validation
    net.kFoldsTrainAndValidate(
        X_train, Y_train,
        k=5, epochs=50, learningRate=0.1,
        showError=True,
        showNodes=True,
        print=lambda e: printer.print_classification(net, X_train, Y_train, size=1.5, dS=0.1),
        showEvery=5
    )
    
    # Validate the network on the test set
    mean, std, error = net.validate(X_test, Y_test)
    print(mean, std)
    printer.print_accuracy(net, X_test, Y_test)

def classification2Classes():
    """Train and validate a neural network for 2-class classification."""
    # Load training and testing data
    X_train, Y_train = reader.readClassificationFile('classification/data.XOR.train.100.csv')
    X_test, Y_test = reader.readClassificationFile('classification/data.XOR.test.100.csv')
    
    # Initialize the neural network
    net = nn.NeuralNetwork(momentumSize=1, seed=0)
    net.addLayer(2, 10, F.tanh, True)
    net.addLayer(10, 1, F.sigmoid, True)
    net.setCostFunction(L.l1)
    
    # Train and validate the network using 5-fold cross-validation
    net.kFoldsTrainAndValidate(
        X_train, Y_train,
        k=5, epochs=500, learningRate=0.5,
        showError=True,
        showNodes=False,
        print=lambda: printer.print_classification(net, X_test, Y_test, size=1.1, dS=0.1),
        showEvery=10
    )
    
    # Validate the network on the test set
    mean, std, error = net.validate(X_test, Y_test)
    print(mean, std)
    printer.print_accuracy(net, X_test, Y_test)

def regression():
    """Train and validate a neural network for regression."""
    # Load training and testing data
    X_train, Y_train = reader.readRegressionFile('regression/data.square.train.100.csv')
    X_test, Y_test = reader.readRegressionFile('regression/data.square.test.100.csv')
    
    # Initialize the neural network
    net = nn.NeuralNetwork(momentumSize=1, seed=100)
    net.addLayer(1, 2, F.polly, bias=True)
    net.addLayer(2, 1, F.linear, bias=True)
    net.setCostFunction(L.l1)
    
    # Train and validate the network using 20-fold cross-validation
    net.kFoldsTrainAndValidate(
        X_train, Y_train,
        k=20, batchSize=1, epochs=2000, learningRate=6e-4, momentumRate=2e-5,
        showError=True,
        showNodes=False,
        print=lambda: printer.print_regression(net, X_train, Y_train, size=5, dx=0.25),
        showEvery=10
    )
    
    # Validate the network on the test set
    mean, std, error = net.validate(X_test, Y_test)
    print(mean, std)
