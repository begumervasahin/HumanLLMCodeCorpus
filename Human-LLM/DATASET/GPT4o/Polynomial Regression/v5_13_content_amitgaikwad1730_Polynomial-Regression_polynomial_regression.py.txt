import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import PolynomialFeatures

# Load dataset
dataset = pd.read_csv('Position_Salaries.csv')
X = dataset.iloc[:, 1:2].values
y = dataset.iloc[:, 2].values

# Linear regression model
linear_regressor = LinearRegression()
linear_regressor.fit(X, y)

# Polynomial regression model
polynomial_features = PolynomialFeatures(degree=4)
X_poly = polynomial_features.fit_transform(X)
polynomial_regressor = LinearRegression()
polynomial_regressor.fit(X_poly, y)

def visualize_results(X, y, model, title, xlabel, ylabel, poly_model=None):
    plt.scatter(X, y, color='red')
    if poly_model:
        plt.plot(X, model.predict(poly_model.fit_transform(X)), color='blue')
    else:
        plt.plot(X, model.predict(X), color='blue')
    plt.title(title)
    plt.xlabel(xlabel)
    plt.ylabel(ylabel)
    plt.show()

# Visualize linear regression results
visualize_results(X, y, linear_regressor, 'Truth or Bluff (Linear Regression)', 'Position level', 'Salary')

# Visualize polynomial regression results
visualize_results(X, y, polynomial_regressor, 'Truth or Bluff (Polynomial Regression)', 'Position level', 'Salary', polynomial_features)

# Higher resolution visualization for polynomial regression
X_grid = np.arange(min(X), max(X), 0.1).reshape(-1, 1)
plt.scatter(X, y, color='red')
plt.plot(X_grid, polynomial_regressor.predict(polynomial_features.fit_transform(X_grid)), color='blue')
plt.title('Truth or Bluff (Polynomial Regression)')
plt.xlabel('Position level')
plt.ylabel('Salary')
plt.show()

# Predicting a new result with Linear Regression
linear_prediction = linear_regressor.predict([[6.5]])
print(f"Linear Regression prediction for level 6.5: {linear_prediction}")

# Predicting a new result with Polynomial Regression
poly_prediction = polynomial_regressor.predict(polynomial_features.fit_transform([[6.5]]))
print(f"Polynomial Regression prediction for level 6.5: {poly_prediction}")
