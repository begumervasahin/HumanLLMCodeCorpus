import os
import numpy as np
import requests
from flask import Flask, render_template, request

app = Flask(__name__, template_folder='web', static_folder='static')

# Define x-coordinates for the temperature data points
X_COORDS = [1, 4, 7, 10, 13, 16, 19, 22, 25, 28]

API_KEY = '8da365f65510cd3b548ab105287b6f28'
BASE_URL = 'http://api.openweathermap.org/data/2.5/forecast'

def predict_temperature(temps):
    """Predict the temperature using a quadratic polynomial fit."""
    coefficients = np.polyfit(X_COORDS, temps, 2)
    polynomial = np.poly1d(coefficients)
    predicted_temp = round(polynomial(1), 2)
    return predicted_temp

def fetch_weather_data(city):
    """Fetch weather data from OpenWeatherMap API for a given city."""
    url = f"{BASE_URL}?q={city}&appid={API_KEY}"
    response = requests.get(url)
    return response.json()

def extract_temperatures(json_data):
    """Extract and convert temperatures from the API response data."""
    temps = [
        round(json_data['list'][i]['main']['temp'] - 273.15, 1) 
        for i in range(0, 30, 3)
    ]
    return temps

@app.route('/', methods=["GET", "POST"])
def index():
    city = request.values.get('test', 'London')  # Default to London if no city is provided
    weather_data = fetch_weather_data(city)
    temperatures = extract_temperatures(weather_data)

    if len(temperatures) < 10:
        return "Not enough data to make predictions"

    predicted_temp = predict_temperature(temperatures[:10])
    current_weather = weather_data['list'][0]['weather'][0]['description'].capitalize()

    return render_template('index.html', t=temperatures[0], w=current_weather, n=predicted_temp)

if __name__ == '__main__':
    port = int(os.environ.get("PORT", 5000))
    app.run(debug=True, host='0.0.0.0', port=port)
