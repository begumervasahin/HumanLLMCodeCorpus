import sympy as sp
import numpy as np

def montar_equacao(incognitas, x):
    """
    Constructs a polynomial equation with the given coefficients.
    
    Args:
    incognitas (list): List of coefficients.
    x (symbol): Symbolic variable.
    
    Returns:
    y (expression): Polynomial equation.
    """
    degree = len(incognitas)
    y = sum(incognitas[i] * x ** (degree - i - 1) for i in range(degree))
    return y

def is_float(s):
    """
    Checks if a string can be converted to a float.
    
    Args:
    s (str): Input string.
    
    Returns:
    bool: True if s can be converted to float, False otherwise.
    """
    try:
        float(s)
        return True
    except ValueError:
        return False

def read_data(file_name='data.csv'):
    """
    Reads data from a CSV file and returns two lists of floats.
    
    Args:
    file_name (str): Path to the CSV file.
    
    Returns:
    x_list (list): List of x values.
    y_list (list): List of y values.
    """
    x_list = []
    y_list = []
    with open(file_name, 'r') as file:
        for line in file:
            x_str, y_str = line.split(',')
            if is_float(x_str) and is_float(y_str):
                x_list.append(float(x_str))
                y_list.append(float(y_str.strip()))
    return x_list, y_list

def main():
    """
    Main function to perform polynomial regression.
    """
    # Input for the degree of the polynomial regression
    n = int(input("Digite a ordem da regressão (ordem máxima = 24): "))
    if n > 24:
        print("\n\n------------------\n  Ordem inválida  \n------------------\n\n")
        return

    # Define the symbolic variable 'x' and the list of coefficients
    x = sp.symbols('x')
    incognitas = [sp.symbols(chr(i)) for i in range(97, 97 + n + 1)]  # a to the nth letter
    
    # Read data from file
    x_list, y_list = read_data()
    
    # Construct the polynomial equation
    h = montar_equacao(incognitas, x)
    
    # Calculate the sum of squared differences
    sum_of_squared_diffs = sum((y_list[i] - h.subs(x, x_list[i]))**2 for i in range(len(x_list)))
    
    # Calculate the derivatives of the sum of squared differences with respect to each coefficient
    diffs = [sp.diff(sum_of_squared_diffs, incognitas[i]) for i in range(len(incognitas))]
    
    # Solve the system of equations to find the optimal coefficients
    solutions = sp.solve(diffs, incognitas)
    
    # Substitute the solutions back into the polynomial equation
    k = h.subs(solutions)
    
    # Print the resulting model
    print("Modelo: ", k)

if __name__ == "__main__":
    main()
