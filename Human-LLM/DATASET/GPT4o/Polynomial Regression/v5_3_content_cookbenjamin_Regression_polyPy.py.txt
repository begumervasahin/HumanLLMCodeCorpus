import numpy as np
from numpy import linalg
import itertools
import polynomialModel as pm

class PolyPy:
    def __init__(self, split=0.15, max_validation=5):
        self.lowest_test_error = None
        self.max_validation = max_validation
        self.validation_timer = None
        self.split_ratio = split
        self.best_degree = 0
        self.errors = []
        
    def fit(self, dataset):
        train_X, train_Y, test_X, test_Y = self.split_dataset(dataset)
        self.validate_models(train_X, train_Y, test_X, test_Y)
        model = self.fit_model_with_degree(dataset[:, :-1], dataset[:, -1], self.best_degree)
        return model
    
    def split_dataset(self, dataset):
        np.random.shuffle(dataset)
        split_index = int(len(dataset) * self.split_ratio)
        
        train_X = dataset[:split_index, :-1]
        train_Y = dataset[:split_index, -1]
        test_X = dataset[split_index:, :-1]
        test_Y = dataset[split_index:, -1]
        
        return train_X, train_Y, test_X, test_Y
    
    def validate_models(self, train_X, train_Y, test_X, test_Y):
        self.validation_timer = self.max_validation
        degree = 1
        
        while self.validation_timer > 0:
            model = self.fit_model_with_degree(train_X, train_Y, degree)
            test_error = model.error(test_X, test_Y)
            
            print(f"Degree: {degree}")
            print(model)
            print(f"Train Error: {model.error(train_X, train_Y)}")
            print(f"Test Error: {test_error}")
            print()
            
            self.errors.append(test_error)
            self.update_best_model(test_error, degree)
            degree += 1
    
    def update_best_model(self, error, degree):
        if self.lowest_test_error is None or error < self.lowest_test_error:
            self.lowest_test_error = error
            self.best_degree = degree
            self.validation_timer = self.max_validation
        else:
            self.validation_timer -= 1
    
    def fit_model_with_degree(self, X, Y, degree):
        Y = np.asarray(Y).squeeze()
        X = np.asarray(X)
        
        num_covariates = X.shape[1]
        X = np.hstack((np.ones((X.shape[0], 1), dtype=X.dtype), X))  # Add intercept term
        
        generators = [self.basis_vector(num_covariates + 1, i) for i in range(num_covariates + 1)]
        powers = list(map(sum, itertools.combinations_with_replacement(generators, degree)))
        
        A = np.hstack([self.as_tall((X ** p).prod(1)) for p in powers])
        beta = linalg.lstsq(A, Y, rcond=None)[0]
        
        model = pm.PolyModel(powers, beta, degree, num_covariates)
        return model
    
    @staticmethod
    def basis_vector(n, i):
        x = np.zeros(n, dtype=int)
        x[i] = 1
        return x
    
    @staticmethod
    def as_tall(x):
        return x.reshape(x.shape + (1,))
