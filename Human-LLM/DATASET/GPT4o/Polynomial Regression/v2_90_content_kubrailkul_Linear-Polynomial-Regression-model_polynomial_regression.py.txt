import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# Load the data
data = pd.read_csv("C:/Users/pc/Desktop/kurs/HW1_DATA.csv")
x = data['x']
y = data['y']

# Plot the original data
plt.plot(x, y, '.')
plt.title("Original Data")
plt.xlabel("x")
plt.ylabel("y")
plt.show()

# Calculate the sums of x and y
xsum = np.sum(x)
ysum = np.sum(y)
yx = y.dot(x)

# Define a function to calculate the mean
def mean(num):
    return num / 1000

# Calculate the coefficients for the linear regression
den = mean(x.dot(x)) - mean(xsum)**2
a = (mean(yx) - mean(xsum) * mean(ysum)) / den
b = (mean(ysum) * mean(x.dot(x)) - mean(yx) * mean(xsum)) / den

# Calculate the predicted y values using the linear regression model
yhat = a * x + b

# Plot the original data and the linear fit
plt.plot(x, y, '.', label="Original Data")
plt.plot(x, yhat, '.', label="Linear Fit")
plt.title("Linear Regression Fit")
plt.xlabel("x")
plt.ylabel("y")
plt.legend()
plt.show()

# Calculate R-squared for the linear fit
sren = np.sum((y - yhat)**2)
stot = np.sum((y - mean(ysum))**2)
rsquare = 1 - sren / stot
print(f"R-squared for linear fit: {rsquare:.4f}")

# Define a function for polynomial regression
def ypoly(p, x):
    yhat = 0
    l = len(p) - 1
    for i in range(len(p)):
        yhat += p[i] * x**(l - i)
    return yhat

# Define a function to calculate R-squared
def rsquare(yhat, y, stot):
    sren = np.sum((y - yhat)**2)
    return 1 - sren / stot

# Polynomial regression and R-squared calculations for different degrees
ones = np.ones(1000)
xhat = np.c_[x, ones]
w = np.linalg.solve(np.transpose(xhat).dot(xhat), np.transpose(xhat).dot(y))
print(f"Coefficients for degree 1: {w}")

for degree in range(2, 8):
    xhat = np.c_[x**degree, xhat]
    w = np.linalg.solve(np.transpose(xhat).dot(xhat), np.transpose(xhat).dot(y))
    print(f"Coefficients for degree {degree}: {w}")

# Calculate and plot the polynomial regression fit for the highest degree
y2 = ypoly(w, x)
a2 = rsquare(y2, y, stot)
plt.plot(x, y, '.', color='b', label="Original Data")
plt.plot(x, y2, '.', color='g', label="Polynomial Fit (degree 7)")
plt.title("Polynomial Regression Fit")
plt.xlabel("x")
plt.ylabel("y")
plt.legend()
plt.show()

print(f"R-squared for polynomial fit (degree 7): {a2:.4f}")
