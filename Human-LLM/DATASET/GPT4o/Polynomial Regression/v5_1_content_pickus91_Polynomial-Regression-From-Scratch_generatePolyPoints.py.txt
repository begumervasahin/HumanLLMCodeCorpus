import numpy as np
import matplotlib.pyplot as plt
from matplotlib import style

style.use('ggplot')

def generate_poly_points(start, stop, num_points, coeff, noise_level=1, plot=True):
    """
    Generate evenly spaced x and y coordinates in Euclidean space from a polynomial with Gaussian noise.

    Parameters
    ----------
    start : float
        Minimum value for the generated x-coordinates.
    stop : float
        Maximum value for the generated x-coordinates.
    num_points : int
        Number of coordinates to generate.
    coeff : 1-d array
        Coefficients of the polynomial.
    noise_level : float, optional
        Indicates the amount of Gaussian noise to add to the y-coordinates generated from the polynomial characterized by coeff array.
        Zero indicates no noise, while each integer increment results in a ten-fold noise increase. Default is 1.
    plot : bool, optional
        If True, returns a matplotlib plot of the generated coordinates. Default is True.

    Returns
    -------
    x_pts : 1-d array, shape = [num_points,]
        Generated x-coordinates.
    y_pts : 1-d array, shape = [num_points,]
        Generated y-coordinates.
    """
    # Generate evenly spaced x-coordinates
    x_pts = np.linspace(start, stop, num_points)
    
    # Generate y-coordinates from the polynomial coefficients
    y_pts = np.polyval(coeff[::-1], x_pts)

    # Add Gaussian noise to the y-coordinates
    if noise_level > 0:
        noise = np.random.normal(0, 10 ** noise_level, num_points)
        y_pts += noise

    # Plot the points if requested
    if plot:
        plt.figure()
        plt.scatter(x_pts, y_pts)
        plt.xlabel('x')
        plt.ylabel('y')
        plt.title('Generated Polynomial Points with Noise')
        plt.show()

    return x_pts, y_pts

# Example usage
if __name__ == "__main__":
    # Parameters for generating points
    start = 0
    stop = 10
    num_points = 100
    coeff = [1, -2, 3]  # Example polynomial coefficients for 3x^2 - 2x + 1
    noise_level = 1
    plot = True

    # Generate and plot polynomial points
    x_pts, y_pts = generate_poly_points(start, stop, num_points, coeff, noise_level, plot)
