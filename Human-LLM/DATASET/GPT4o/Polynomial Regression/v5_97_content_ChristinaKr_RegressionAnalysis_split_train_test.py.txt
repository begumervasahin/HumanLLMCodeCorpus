import csv
import numpy as np

def split_data(ifname, delimiter=None, has_header=False, columns=None, seed=42, fraction=0.15):
    """
    Imports a tab/comma/semi-colon/... separated data file as an array of
    floating point numbers.

    Parameters
    ----------
    ifname : str
        Filename/path of the data file.
    delimiter : str, optional
        Delimiter of data values. Defaults to tab.
    has_header : bool, optional
        Indicates if the data file has a header line. Defaults to False.
    columns : list of int, optional
        List of integers specifying which columns of the file to import (counting from 0).
    seed : int, optional
        Seed for the pseudo-random number generator. Defaults to 42.
    fraction : float, optional
        Fraction of data to be used as the test set. Defaults to 0.15.

    Returns
    -------
    tuple
        training_data_as_array : numpy.ndarray
            Training data as a numpy array.
        test_data_as_array : numpy.ndarray
            Test data as a numpy array.
        field_names : list of str or None
            List of field names if the header is present, otherwise None.
    """
    # Set the random seed for reproducibility
    np.random.seed(seed)

    # Placeholder for the number of rows; should be determined dynamically
    num_rows = 1599  # Update this if the number of rows is known or can be determined

    # Determine test row indices based on the fraction
    test_row_indices = np.unique(np.random.choice(num_rows, size=int(fraction * num_rows), replace=False))

    # Default delimiter to tab if not specified
    if delimiter is None:
        delimiter = '\t'

    with open(ifname, 'r') as ifile:
        datareader = csv.reader(ifile, delimiter=delimiter)

        # Read the header if present
        field_names = next(datareader) if has_header else None

        training_data = []
        test_data = []

        for count, row in enumerate(datareader):
            # Select only the specified columns if provided
            if columns is not None:
                row = [row[c] for c in columns]

            # Assign the row to test or training data
            if count in test_row_indices:
                test_data.append(row)
            else:
                training_data.append(row)

    # Print summary information
    print(f"There are {len(training_data)} training entries")
    print(f"There are {len(test_data)} test entries")
    print(f"Each row has {len(training_data[0])} elements")

    # Convert lists to numpy arrays
    training_data_as_array = np.array(training_data).astype(float)
    test_data_as_array = np.array(test_data).astype(float)

    return training_data_as_array, test_data_as_array, field_names
