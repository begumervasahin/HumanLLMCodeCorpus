import quandl
import pandas as pd
import numpy as np
from sklearn.linear_model import LinearRegression
import matplotlib.pyplot as plt
from sklearn.preprocessing import PolynomialFeatures

# Fetch Google stock data
google_stock_data = quandl.get("WIKI/GOOGL")
google_stock_data = google_stock_data.reset_index()
df = pd.DataFrame(google_stock_data)

# Prepare training data
index_ = df['index']
days_ = np.array(index_)[:, np.newaxis]
price_open = df['Adj. Open']
price_train = np.array(price_open[:])

# Simple Linear Regression model
simple_regression = LinearRegression()
simple_regression.fit(days_, price_train)

# Polynomial Regression model (degree 4)
polynomial_regression = LinearRegression()
poly_features = PolynomialFeatures(degree=4)
days_transformed = poly_features.fit_transform(days_)
polynomial_regression.fit(days_transformed, price_train)

# Load test data for prediction
test_data = pd.read_csv('data_to_be_predicted.csv')
test_days = np.array(test_data['NO'])[:, np.newaxis]

# Predict using both models
pred_simple_regression = simple_regression.predict(test_days)
test_days_transformed = poly_features.fit_transform(test_days)
pred_poly_regression = polynomial_regression.predict(test_days_transformed)

# Create DataFrame for predictions
predictions = np.array([pred_poly_regression, pred_simple_regression]).T
pred_dataframe = pd.DataFrame(predictions, index=test_data['DATE_PREDICTED'], columns=['Polynomial Model', 'Linear Model'])

# Display predictions
print("A = Google stock prices (Adj. Open) predicted by Polynomial model")
print("B = Google stock prices (Adj. Open) predicted by Linear model")
print(pred_dataframe)
print("\nCONCLUSION: The Polynomial model predicts better and fits the graph better, as can be clearly seen in the graph.")

# Plot results
plt.scatter(days_, price_train, color='black', s=1, label='Data')
plt.plot(days_, simple_regression.predict(days_), color='blue', linewidth=2.0, label='Linear Model')
plt.plot(days_, polynomial_regression.predict(days_transformed), color='green', linewidth=2.0, label='Polynomial Model')
plt.xlabel("Days (Approx)")
plt.ylabel('Stock Price in USD')
plt.title('Implementation on Training Data')
plt.legend()
plt.autoscale(tight=True)
plt.show()
