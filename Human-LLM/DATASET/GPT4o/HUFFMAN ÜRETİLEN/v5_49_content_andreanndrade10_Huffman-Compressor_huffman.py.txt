import os
from buildTree import firstRound

def main():
    display_intro()
    input_data = get_user_input()

    if input_data.endswith(".txt"):
        file_content = read_file_content(input_data)
        frequency_dict = compute_frequency_dict(file_content)
        proportions = compute_proportions(frequency_dict, len(file_content))
        firstRound(proportions, list(frequency_dict.keys()))
    else:
        handle_invalid_input()

def display_intro():
    print("\n\nAndre Luiz Lourenço de Andrade - 14/0016295")
    print("Teoria da Informação - Huffman Compressor\n")

def get_user_input():
    return input("Please enter a string or a file path (e.g., 'filename.txt') >>> ").strip()

def handle_invalid_input():
    print("\nThis function is not ready yet... Please choose a .txt file from your directory: " + os.getcwd())
    main()

def read_file_content(file_path):
    try:
        with open(file_path, "rb") as file:
            content = file.read()
        print(f"You selected '{file_path}' for compression.")
        return content
    except FileNotFoundError:
        print(f"Error: The file '{file_path}' does not exist.")
        return main()

def compute_frequency_dict(content):
    frequency_dict = {}
    for byte in content:
        frequency_dict[byte] = frequency_dict.get(byte, 0) + 1
    return frequency_dict

def compute_proportions(frequency_dict, total_size):
    return sorted((count / total_size for count in frequency_dict.values()), reverse=True)

if __name__ == "__main__":
    main()
