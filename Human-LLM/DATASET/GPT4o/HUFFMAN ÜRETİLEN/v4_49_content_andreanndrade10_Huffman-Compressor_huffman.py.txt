import os
from buildTree import firstRound

def main():
    print_intro()
    input_data = get_input()
    
    if input_data.endswith(".txt"):
        file_content = read_file(input_data)
        frequency_dict = build_frequency_dictionary(file_content)
        proportion_list = calculate_proportions(frequency_dict, file_content)
        firstRound(proportion_list, list(frequency_dict.keys()))
    else:
        handle_unsupported_input()

def print_intro():
    print("\n\nAndre Luiz Lourenço de Andrade - 14/0016295")
    print("Teoria da Informação - Huffman Compressor\n")

def get_input():
    return input("Please enter a string or a file path (e.g., '-c filename.txt') >>> ").strip()

def handle_unsupported_input():
    print("\nThis function is not ready yet... Please choose a .txt file from your directory: " + os.getcwd())
    main()

def read_file(file_path):
    try:
        with open(file_path, "rb") as file:
            content = file.read()
        print(f"You chose {file_path} as the file to be compressed...")
        return content
    except FileNotFoundError:
        print(f"Error: The file '{file_path}' does not exist.")
        main()

def build_frequency_dictionary(content):
    frequency_dict = {}
    for byte in content:
        frequency_dict[byte] = frequency_dict.get(byte, 0) + 1
    return frequency_dict

def calculate_proportions(frequency_dict, content):
    total_size = len(content)
    for key in frequency_dict:
        frequency_dict[key] /= total_size
    proportions = sorted(frequency_dict.values(), reverse=True)
    return proportions

if __name__ == "__main__":
    main()
