def calculate_frequencies(text):
    """
    Calculate the frequency of each character in the string.

    Args:
        text (str): The input string.

    Returns:
        dict: A dictionary of character frequencies.
    """
    frequency = {}
    for char in text:
        frequency[char] = frequency.get(char, 0) + 1
    return frequency

def huffman_encoding(frequency):
    """
    Generate Huffman encoding based on character frequencies.

    Args:
        frequency (dict): A dictionary of character frequencies.

    Returns:
        dict: A dictionary mapping characters to their Huffman codes.
    """
    # Initialize priority queue with characters sorted by frequency
    chars = list(frequency.keys())
    counts = list(frequency.values())
    huffman_codes = {char: "" for char in chars}
    
    while len(counts) > 1:
        # Sort the characters by their frequencies
        combined = sorted(zip(counts, chars))
        counts, chars = zip(*combined)
        
        # Get the two smallest frequencies and their corresponding characters
        left, right = counts[:2], chars[:2]
        
        # Merge these frequencies and characters
        counts = counts[2:]
        chars = chars[2:]
        
        merged_freq = sum(left)
        merged_chars = "".join(sorted(right))
        
        # Assign Huffman codes
        for i, char in enumerate(right):
            huffman_codes[char] = ("0" if i == 0 else "1") + huffman_codes[char]
        
        # Add the merged node back to the list
        counts = [merged_freq] + list(counts)
        chars = [merged_chars] + list(chars)
    
    return huffman_codes

def encode_text(text, huffman_codes):
    """
    Encode the text using Huffman codes.

    Args:
        text (str): The input text.
        huffman_codes (dict): The dictionary of Huffman codes.

    Returns:
        str: The encoded text.
    """
    return ''.join(huffman_codes[char] for char in text)

def write_to_file(filename, content):
    """
    Write content to a file.

    Args:
        filename (str): The name of the file.
        content (str): The content to write to the file.
    """
    with open(filename, 'w') as file:
        file.write(content)

def main():
    # Read input text
    with open("input.txt", "r") as file:
        text = file.readline().strip().lower()
    
    print("Input Text:", text)
    
    # Calculate character frequencies
    frequency = calculate_frequencies(text)
    print("Frequency Table:", frequency)
    
    # Generate Huffman codes
    huffman_codes = huffman_encoding(frequency)
    print("Huffman Codes:", huffman_codes)
    
    # Encode the text using Huffman codes
    encoded_text = encode_text(text, huffman_codes)
    print("Encoded Text:", encoded_text)
    
    # Write the encoded text to output.txt
    write_to_file("output.txt", encoded_text)
    
    # Write the Huffman dictionary to dictionary.txt
    huffman_dict_content = "\n".join(f"{char}={code}" for char, code in huffman_codes.items())
    write_to_file("dictionary.txt", huffman_dict_content)
    
    print("Huffman Encoding Complete.")

if __name__ == "__main__":
    main()
