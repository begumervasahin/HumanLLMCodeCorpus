class Comparable:
    def compare(self, other):
        raise NotImplementedError("Subclasses should implement this method!")


class HuffElement(Comparable):
    """
    Represents an element in a Huffman encoding map, storing a character,
    its frequency, and its corresponding Huffman code.
    """

    def __init__(self, char: str):
        """
        Initializes a HuffElement with the given character.
        The frequency starts at zero, and the Huffman code is an empty string.
        """
        self._char = char
        self._frequency = 0
        self._code = ""

    def increment_frequency(self) -> None:
        """Increments the frequency count of the character."""
        self._frequency += 1

    def get_frequency(self) -> int:
        """Returns the frequency count of the character."""
        return self._frequency

    def set_frequency(self, frequency: int) -> None:
        """Sets the frequency count of the character."""
        self._frequency = frequency

    def get_code(self) -> str:
        """Returns the Huffman code for the character."""
        return self._code

    def set_code(self, code: str) -> None:
        """Sets the Huffman code for the character."""
        self._code = code

    def get_char(self) -> str:
        """Returns the character."""
        return self._char

    def set_char(self, char: str) -> None:
        """Sets the character."""
        self._char = char

    def compare(self, other: 'HuffElement') -> int:
        """
        Compares this HuffElement with another based on frequency count.
        Returns 1 if this element has a higher frequency, -1 if lower, and 0 if equal.
        """
        return (self._frequency > other.get_frequency()) - (self._frequency < other.get_frequency())

    def __str__(self) -> str:
        """Returns a string representation of the HuffElement."""
        return f"Char: {self._char} | Code: {self._code} | Frequency: {self._frequency}"


# Example usage
if __name__ == "__main__":
    huff_element = HuffElement('a')
    huff_element.increment_frequency()
    huff_element.set_code('101')
    
    print(huff_element)
