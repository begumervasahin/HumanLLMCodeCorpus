class HuffmanHeap:
    """
    A heap structure to manage Huffman trees during the encoding process.
    
    Attributes:
        old (list): A list of Huffman trees sorted by frequency (used for initial trees).
        new (list): A list of Huffman trees sorted by frequency (used for newly created trees).
    """
    
    def __init__(self, old, new=None):
        """
        Initialize the HuffmanHeap with old and new Huffman trees.
        
        Args:
            old (list): A list of Huffman trees sorted by frequency.
            new (list, optional): A list of newly created Huffman trees sorted by frequency. Defaults to an empty list.
        """
        self.old = old
        self.new = new if new is not None else []

    def enqueue(self, item):
        """
        Add a Huffman tree to the new list.
        
        Args:
            item: A Huffman tree to be added to the heap.
        """
        self.new.append(item)

    def dequeue(self):
        """
        Remove and return the Huffman tree with the smallest frequency from the heap.
        
        Returns:
            The Huffman tree with the smallest frequency. Returns None if both heaps are empty.
        """
        if not self.old and not self.new:
            print("Both old and new heaps are empty!")
            return None

        if not self.old:
            return self.new.pop(0)
        if not self.new:
            return self.old.pop(0)

        # Both heaps are non-empty, compare the smallest elements
        if self.old[0].get_freq() <= self.new[0].get_freq():
            return self.old.pop(0)
        else:
            return self.new.pop(0)
