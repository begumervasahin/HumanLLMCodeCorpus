from collections import Counter
from queue import PriorityQueue

# Step 1: Read the input text from the file
with open('rand_file') as file:
    text = file.read()

# Step 2: Calculate the frequency of each character in the text
char_count = Counter(text)

# Step 3: Specify the relevant characters
selected_chars = ['A', 'B', 'C', 'D', 'E']

# Step 4: Calculate the sum of frequencies of the selected characters
total_count = sum(char_count[char] for char in selected_chars)

# Step 5: Calculate the probability of each selected character
probabilities = {char: char_count[char] / total_count for char in selected_chars}

# Step 6: Define the HuffmanNode class to represent nodes in the Huffman tree
class HuffmanNode:
    def __init__(self, left=None, right=None):
        self.left = left
        self.right = right

    def children(self):
        return self.left, self.right

# Step 7: Create the Huffman tree based on the character probabilities
def create_huffman_tree(probabilities):
    pq = PriorityQueue()
    for char, prob in probabilities.items():
        pq.put((prob, char))

    while pq.qsize() > 1:
        left_node, right_node = pq.get(), pq.get()
        node = HuffmanNode(left_node, right_node)
        pq.put((left_node[0] + right_node[0], node))

    return pq.get()

# Step 8: Traverse the Huffman tree to generate the binary codes
def generate_huffman_codes(node, prefix="", code=None):
    if code is None:
        code = {}
    if isinstance(node[1], HuffmanNode):
        generate_huffman_codes(node[1].left, prefix + "0", code)
        generate_huffman_codes(node[1].right, prefix + "1", code)
    else:
        code[node[1]] = prefix
    return code

# Step 9: Create the Huffman tree and generate the Huffman codes
huffman_tree = create_huffman_tree(probabilities)
huffman_codes = generate_huffman_codes(huffman_tree)

# Step 10: Print the probabilities and corresponding Huffman codes
print("Character  Probability       Huffman Code")
for char in sorted(probabilities, key=probabilities.get, reverse=True):
    print(f"{char}         {probabilities[char]:.10f}     {huffman_codes[char]}")

# Step 11: Encode the text using the Huffman codes
encoded_text = ''.join(huffman_codes[char] for char in text if char in huffman_codes)

# Step 12: Output the encoded text
print(f"Encoded text: {encoded_text}")
