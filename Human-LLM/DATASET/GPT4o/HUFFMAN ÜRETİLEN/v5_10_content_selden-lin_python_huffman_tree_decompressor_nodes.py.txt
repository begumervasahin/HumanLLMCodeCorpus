class HuffmanNode:
    """
    A node in a Huffman tree.
    
    Symbols occur only at leaves. Each node has a `number` attribute that can be used for node-numbering.
    
    Attributes:
    ===========
    symbol (int, optional): Symbol located in this node, if any.
    left (HuffmanNode, optional): Left subtree.
    right (HuffmanNode, optional): Right subtree.
    number (int, optional): Node number.
    """

    def __init__(self, symbol=None, left=None, right=None):
        """
        Initialize a new HuffmanNode.
        
        Args:
            symbol (int, optional): Symbol to be stored in this node, if any.
            left (HuffmanNode, optional): A tree rooted at 'left', if any.
            right (HuffmanNode, optional): A tree rooted at 'right', if any.
        """
        self.symbol = symbol
        self.left = left
        self.right = right
        self.number = None

    def __eq__(self, other):
        """
        Check if self is equivalent to other.
        
        Args:
            other (HuffmanNode): Another HuffmanNode to compare.
        
        Returns:
            bool: True if self is equivalent to other, False otherwise.
        """
        if not isinstance(other, HuffmanNode):
            return False
        return (
            self.symbol == other.symbol and
            self.left == other.left and
            self.right == other.right
        )

    def __lt__(self, other):
        """
        Compare if self is less than other. 
        
        This method is a placeholder and always returns False.
        
        Args:
            other (HuffmanNode): Another HuffmanNode to compare.
        
        Returns:
            bool: False always.
        """
        return False

    def __repr__(self):
        """
        Return a constructor-style string representation of the node.
        
        Returns:
            str: String representation of the HuffmanNode.
        """
        return (
            f"HuffmanNode(symbol={self.symbol}, "
            f"left={self.left}, right={self.right})"
        )

    def is_leaf(self):
        """
        Check if the node is a leaf.
        
        Returns:
            bool: True if the node is a leaf, False otherwise.
        """
        return self.left is None and self.right is None


class ReadNode:
    """
    A node as read from a compressed file.
    
    Each node consists of type and data information as described in the handout.
    
    Attributes:
    ===========
    l_type (int): 0/1 (if the corresponding HuffmanNode's left is a leaf).
    l_data (int): A symbol or the node number of a HuffmanNode's left.
    r_type (int): 0/1 (if the corresponding HuffmanNode's right is a leaf).
    r_data (int): A symbol or the node number of a HuffmanNode's right.
    """

    def __init__(self, l_type, l_data, r_type, r_data):
        """
        Initialize a new ReadNode.
        
        Args:
            l_type (int): Used to initialize self.l_type.
            l_data (int): Used to initialize self.l_data.
            r_type (int): Used to initialize self.r_type.
            r_data (int): Used to initialize self.r_data.
        """
        self.l_type = l_type
        self.l_data = l_data
        self.r_type = r_type
        self.r_data = r_data

    def __repr__(self):
        """
        Return a constructor-style string representation of the node.
        
        Returns:
            str: String representation of the ReadNode.
        """
        return (
            f"ReadNode(l_type={self.l_type}, l_data={self.l_data}, "
            f"r_type={self.r_type}, r_data={self.r_data})"
        )
