import heapq
from collections import namedtuple, Counter

class Node(namedtuple('Node', ['left', 'right'])):
    def walk(self, code, acc):
        """
        Traverse the Huffman tree, building the encoding map.
        
        Parameters
        ----------
        code : dict
            The dictionary storing the Huffman codes for each character.
        acc : str
            The accumulated binary string for the current traversal path.
        """
        self.left.walk(code, acc + '0')
        self.right.walk(code, acc + '1')

class Leaf(namedtuple('Leaf', ['char'])):
    def walk(self, code, acc):
        """
        Assign the accumulated binary string to the character in the leaf node.
        
        Parameters
        ----------
        code : dict
            The dictionary storing the Huffman codes for each character.
        acc : str
            The accumulated binary string for the current traversal path.
        """
        code[self.char] = acc or '0'

def huffman_encode(text):
    """
    Build the Huffman encoding for a given string.
    
    Parameters
    ----------
    text : str
        The string to encode.
    
    Returns
    -------
    dict
        A dictionary mapping each character in the string to its Huffman code.
    """
    heap = [(freq, len(heap), Leaf(char)) for char, freq in Counter(text).items()]
    heapq.heapify(heap)
    
    count = len(heap)
    while len(heap) > 1:
        freq1, _count1, left = heapq.heappop(heap)
        freq2, _count2, right = heapq.heappop(heap)
        heapq.heappush(heap, (freq1 + freq2, count, Node(left, right)))
        count += 1
    
    code = {}
    if heap:
        [(_freq, _count, root)] = heap
        root.walk(code, '')
    return code

def huffman_decode(encoded_text, code):
    """
    Decode a Huffman-encoded string using the provided code map.
    
    Parameters
    ----------
    encoded_text : str
        The encoded binary string.
    code : dict
        The dictionary mapping characters to their Huffman codes.
    
    Returns
    -------
    str
        The decoded string.
    """
    decoded_text = []
    current_code = ''
    
    reverse_code = {v: k for k, v in code.items()}
    
    for bit in encoded_text:
        current_code += bit
        if current_code in reverse_code:
            decoded_text.append(reverse_code[current_code])
            current_code = ''
    
    return ''.join(decoded_text)

def save_encoded_binary(encoded_text, filename):
    """
    Save the encoded text as a binary file.
    
    Parameters
    ----------
    encoded_text : str
        The binary string representing the encoded text.
    filename : str
        The name of the file to save the binary data to.
    """
    with open(filename, 'wb') as output_file:
        for i in range(0, len(encoded_text), 8):
            byte = encoded_text[i:i+8]
            output_file.write(int(byte, 2).to_bytes(1, byteorder='big'))

def load_text(filename):
    """
    Load the text from a file.
    
    Parameters
    ----------
    filename : str
        The path to the file to load the text from.
    
    Returns
    -------
    str
        The content of the file as a string.
    """
    with open(filename, 'r', encoding='utf-8') as file:
        return file.read()

def save_text(text, filename):
    """
    Save the given text to a file.
    
    Parameters
    ----------
    text : str
        The text to save.
    filename : str
        The name of the file to save the text to.
    """
    with open(filename, 'w', encoding='utf-8') as file:
        file.write(text)

    code_map = huffman_encode(input_text)

    print(f"Number of symbols: {len(code_map)}")

    for char in sorted(code_map, key=code_map.get):
        print(f'{char}: {code_map[char]}')
    
    # Save the encoded text as a binary file
    save_encoded_binary(encoded_text, 'encoded.bin')
    
    # Decode the encoded text to verify the correctness
    decoded_text = huffman_decode(encoded_text, code_map)
    
    # Print the decoded text
    print("\nDecoded String:")
