import numpy as np
from scipy.fftpack import dct
from PIL import Image
import os

# JPEG quantization matrix
QUANTIZATION_MATRIX = np.array([
    [16, 11, 10, 16, 24, 40, 51, 61],
    [12, 12, 14, 19, 26, 58, 60, 55],
    [14, 13, 16, 24, 40, 57, 69, 56],
    [14, 17, 22, 29, 51, 87, 80, 62],
    [18, 22, 37, 56, 68, 109, 103, 77],
    [24, 35, 55, 64, 81, 104, 113, 92],
    [49, 64, 78, 87, 103, 121, 120, 101],
    [72, 92, 95, 98, 112, 100, 103, 99]
])

# Global dictionaries to track temporal values and their frequencies
temporal_values = {}
frequency_map = {}
temporal_index = 0

def apply_dct(block):
    """
    Apply 2D DCT to an 8x8 block and quantize it using the JPEG quantization matrix.

    Args:
        block (numpy.ndarray): 8x8 block of image data.

    Returns:
        numpy.ndarray: Quantized DCT coefficients.
    """
    block -= 128  # Shift pixel values to range [-128, 127]
    dct_block = dct(dct(block, axis=0, norm='ortho'), axis=1, norm='ortho')
    quantized_block = np.round(dct_block / QUANTIZATION_MATRIX)
    return quantized_block

def zigzag_order(matrix):
    """
    Perform zigzag ordering on an 8x8 matrix and track the frequency of non-zero coefficients.

    Args:
        matrix (numpy.ndarray): 8x8 matrix of DCT coefficients.
    """
    global temporal_index
    rows, cols = matrix.shape
    result = [[] for _ in range(rows + cols - 1)]

    for i in range(rows):
        for j in range(cols):
            sum_idx = i + j
            if sum_idx % 2 == 0:
                result[sum_idx].insert(0, matrix[i, j])
            else:
                result[sum_idx].append(matrix[i, j])
    
    ordered_values = [val for sublist in result for val in sublist]
    non_zero_idx = next((i for i, v in enumerate(ordered_values) if v != 0), None)
    
    if non_zero_idx is not None:
        for val in ordered_values[non_zero_idx:]:
            temporal_values[temporal_index] = val
            frequency_map[val] = frequency_map.get(val, 0) + 1
            temporal_index += 1

def save_probabilities_to_file(probabilities, filename="result.txt"):
    """
    Save calculated probabilities to a file.

    Args:
        probabilities (dict): Probability map of DCT coefficients.
        filename (str): File to save the results.
    """
    with open(filename, "w") as file:
        for key, prob in probabilities.items():
            file.write(f"{key}\t{prob:.6f}\n")

def process_image(image_path):
    """
    Process the image: convert to grayscale, apply DCT, and save the results.

    Args:
        image_path (str): Path to the image file.
    """
    global temporal_index

    # Load and convert the image to grayscale
    img = Image.open(image_path).convert('L')
    img.save("gray.jpg")
    img.show()

    # Get image dimensions and convert to numpy array
    width, height = img.size
    img_array = np.asarray(img, dtype=np.float32)

    # Initialize a matrix to store the DCT results
    dct_result = np.zeros_like(img_array)

    # Apply DCT to each 8x8 block
    for i in range(0, height, 8):
        for j in range(0, width, 8):
            block = img_array[i:i+8, j:j+8]
            dct_result[i:i+8, j:j+8] = apply_dct(block)
    
    # Save the DCT image
    Image.fromarray(dct_result.clip(0, 255).astype(np.uint8)).save("dct.jpg")
    Image.open("dct.jpg").show()

    # Perform zigzag ordering on the DCT blocks
    for i in range(0, height, 8):
        for j in range(0, width, 8):
            zigzag_order(dct_result[i:i+8, j:j+8])

    # Save the DCT coefficients to a file
    np.savetxt("dct.txt", dct_result, fmt='%d')

    # Calculate probabilities of DCT coefficients
    total_elements = list(temporal_values.values())
    total_count = len(total_elements)
    probabilities = {k: v / total_count for k, v in frequency_map.items()}
    
    # Save probabilities to a file
    save_probabilities_to_file(probabilities)

if __name__ == "__main__":
    process_image("lena.jpg")
    print("Image processing complete. Results saved to files.")
