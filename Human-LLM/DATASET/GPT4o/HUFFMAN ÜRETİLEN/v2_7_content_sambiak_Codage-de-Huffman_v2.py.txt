class Pair:
    """An orderable pair object used in the priority queue."""
    def __init__(self, element, priority):
        self.element = element
        self.priority = priority

    def __lt__(self, other):
        return self.priority < other.priority

    def __gt__(self, other):
        return self.priority > other.priority

    def __eq__(self, other):
        return self.priority == other.priority and self.element == other.element

    def __le__(self, other):
        return self.priority <= other.priority

    def __ge__(self, other):
        return self.priority >= other.priority

    def __str__(self):
        return f"({self.element}, {self.priority})"

    def __repr__(self):
        return self.__str__()

class PriorityQueue:
    """A simple priority queue implemented using a list."""
    def __init__(self):
        self.data = []

    def qsize(self):
        return len(self.data)

    def get(self):
        """Removes and returns the element with the highest priority."""
        return self.data.pop()

    def put(self, elem):
        """Inserts an element into the queue based on its priority."""
        for i, el in enumerate(self.data):
            if elem > el:
                self.data = self.data[:i] + [elem] + self.data[i:]
                return
        self.data.append(elem)

    def __str__(self):
        return str(self.data)

class TreeNode:
    """A node in the Huffman tree."""
    def __init__(self, value):
        """Initializes a tree node with a value."""
        self.value = value
        self.left = None
        self.right = None

    def __str__(self):
        return f"({self.left}, {self.value}, {self.right})"

def create_huffman_tree(letter_frequencies):
    """
    Creates a Huffman tree from a dictionary of letter frequencies.
    
    Parameters
    ----------
    letter_frequencies : dict
        A dictionary mapping letters to their frequencies.
    
    Returns
    -------
    TreeNode
        The root of the Huffman tree.
    """
    priority_queue = PriorityQueue()
    
    # Populate the priority queue with initial leaf nodes
    for letter, frequency in letter_frequencies.items():
        priority_queue.put(Pair(TreeNode(letter), frequency))

    # Build the Huffman tree
    while priority_queue.qsize() > 1:
        left = priority_queue.get()
        right = priority_queue.get()
        merged_node = TreeNode(None)
        merged_node.left = left.element
        merged_node.right = right.element
        priority_queue.put(Pair(merged_node, left.priority + right.priority))

    return priority_queue.get().element

def calculate_frequencies(text):
    """
    Calculates the frequency of each letter in the given text.
    
    Parameters
    ----------
    text : str
        The input text.
    
    Returns
    -------
    dict
        A dictionary mapping letters to their frequencies.
    """
    frequencies = {}
    for letter in text:
        frequencies[letter] = frequencies.get(letter, 0) + 1
    return frequencies

def generate_huffman_codes(tree, path="", code_dict=None):
    """
    Recursively generates Huffman codes from the Huffman tree.
    
    Parameters
    ----------
    tree : TreeNode
        The root of the Huffman tree.
    path : str, optional
        The current traversal path (default is an empty string).
    code_dict : dict, optional
        The dictionary to store the generated codes (default is None).
    
    Returns
    -------
    dict
        A dictionary mapping letters to their corresponding Huffman codes.
    """
    if code_dict is None:
        code_dict = {}

    if tree.value is not None:
        code_dict[tree.value] = path
    else:
        if tree.left is not None:
            generate_huffman_codes(tree.left, path + "0", code_dict)
        if tree.right is not None:
            generate_huffman_codes(tree.right, path + "1", code_dict)

    return code_dict

    # Calculate frequencies
    frequencies = calculate_frequencies(text)
    print("Frequencies:", frequencies)

    huffman_tree = create_huffman_tree(frequencies)
