import numpy as np
import cv2

class Node:
    """A node in the Huffman tree."""
    def __init__(self):
        self.grayvalue = None
        self.isleaf = False
        self.left = None
        self.right = None

def build_tree(root, bit_sequence, grayscale_value):
    """
    Recursively builds the Huffman tree from the bit sequence.

    Parameters
    ----------
    root : Node
        The current node in the tree.
    bit_sequence : list of int
        The bit sequence representing the path in the Huffman tree.
    grayscale_value : int
        The grayscale value associated with the bit sequence.
    """
    for bit in bit_sequence:
        if bit == 0:
            if root.left is None:
                root.left = Node()
            root = root.left
        else:
            if root.right is None:
                root.right = Node()
            root = root.right
    
    root.isleaf = True
    root.grayvalue = grayscale_value

def decode_bitstring(bit_string, node):
    """
    Decodes the bit sequence to find the corresponding grayscale value.

    Parameters
    ----------
    bit_string : str
        The bit sequence to decode.
    node : Node
        The current node in the Huffman tree.

    Returns
    -------
    tuple
        A tuple containing the decoded grayscale value and the length of the matched bit sequence.
    """
    if node.isleaf:
        return node.grayvalue, 0
    
    next_node = node.left if bit_string[0] == '0' else node.right
    return decode_bitstring(bit_string[1:], next_node)

def load_image_dimensions(tree_array):
    """
    Extracts the image dimensions from the tree array.

    Parameters
    ----------
    tree_array : list of list of int
        The array containing the tree bit sequences and dimension data.

    Returns
    -------
    tuple
        A tuple containing the image height and width.
    """
    height = int(''.join(map(str, tree_array.pop())))
    width = int(''.join(map(str, tree_array.pop())))
    return height, width

def main():
    # Step 1: Read the dictionary and bitstream files
    with open('dictionary_meo.txt', 'r') as dict_file, open('imageinBit_meo.txt', 'r') as bit_file:
        tree_array = [list(map(int, line.split()[1].strip())) for line in dict_file]

        # Extract image dimensions
        height, width = load_image_dimensions(tree_array)
        print(f"Image dimensions: {height}x{width}")

        # Step 2: Build the Huffman tree
        root = Node()
        for grayscale_value, bit_sequence in enumerate(tree_array):
            build_tree(root, bit_sequence, grayscale_value)

        # Step 3: Decode the bitstream to reconstruct the image
        image = []
        for line in bit_file:
            bit_string = line.strip()
            decoded_row = []
            i = 0
            while i < len(bit_string):
                gray_value, length = decode_bitstring(bit_string[i:], root)
                decoded_row.append(gray_value)
                i += length + 1  # Move to the next bit after the last decoded symbol
            image.append(decoded_row)

        image = np.array(image, dtype=np.uint8).reshape((height, width))

        # Step 4: Display and save the decompressed image
        cv2.imshow('Decompressed Image', image)
        cv2.imwrite('meo-decompress.png', image)
        cv2.waitKey(0)
        cv2.destroyAllWindows()

if __name__ == "__main__":
    main()
