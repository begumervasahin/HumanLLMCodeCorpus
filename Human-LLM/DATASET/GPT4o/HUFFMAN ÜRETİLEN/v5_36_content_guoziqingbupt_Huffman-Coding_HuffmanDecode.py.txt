def huffman_decode(encoded_text, root):
    """
    Decode a Huffman encoded string using the given Huffman tree root.

    Args:
        encoded_text (str): The Huffman encoded string to decode.
        root (Node): The root node of the Huffman tree.

    Returns:
        str: The decoded string.
    """
    decoded_text = []
    current_node = root

    for bit in encoded_text:
        # Traverse the tree according to the current bit
        current_node = current_node.left if bit == "0" else current_node.right

        # If a leaf node is reached, append the decoded character and reset to the root
        if is_leaf(current_node):
            decoded_text.append(current_node.val)
            current_node = root

    return ''.join(decoded_text)

def is_leaf(node):
    """
    Determine whether a tree node is a leaf.

    Args:
        node (Node): The node to check.

    Returns:
        bool: True if the node is a leaf, False otherwise.
    """
    return node.left is None and node.right is None
