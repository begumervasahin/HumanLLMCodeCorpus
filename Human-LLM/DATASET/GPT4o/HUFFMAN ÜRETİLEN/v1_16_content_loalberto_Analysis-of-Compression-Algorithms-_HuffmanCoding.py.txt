import random
import string
import time
from heapq import heappush, heappop, heapify
from collections import defaultdict

def encode(values):
    """
    Huffman encode the given dictionary mapping symbols to weights.
    
    Parameters:
    ===========
    values : dict
        Dictionary with symbols as keys and their respective weights as values.
    
    Returns:
    ===========
    tuple
        A sorted list of symbols with their encoded values, and the total operation count.
    """
    count = 0
    heap = []
    
    # Initialize the heap with symbols and their weights
    for sym, wt in values.items():
        count += 1
        heap.append([wt, [sym, ""]])
    
    heapify(heap)
    
    # Build the Huffman tree
    while len(heap) > 1:
        count += 1
        lo = heappop(heap)
        hi = heappop(heap)
        
        for pair in lo[1:]:
            pair[1] = '0' + pair[1]
        for pair in hi[1:]:
            pair[1] = '1' + pair[1]
        
        heappush(heap, [lo[0] + hi[0]] + lo[1:] + hi[1:])
    
    return sorted(heappop(heap)[1:], key=lambda val: (len(val[-1]), val)), count

def random_string(string_length):
    """
    Generate a random string of fixed length.
    
    Parameters:
    ===========
    string_length : int
        The length of the generated random string.
    
    Returns:
    ===========
    str
        A random string of the specified length.
    """
    letters = string.ascii_lowercase
    return ''.join(random.choice(letters) for _ in range(string_length))

def main():
    # Number of iterations for testing
    iterations = 10
    
    # Dictionaries to store operation and time measurements
    op_matrix = {f'Iteration{i + 1}': [] for i in range(iterations)}
    time_matrix = {f'Iteration{i + 1}': [] for i in range(iterations)}
    
    # Run multiple iterations for performance measurement
    for itr in range(iterations):
        iteration_key = f'Iteration{itr + 1}'
        
        # Test string lengths from 10 to 100000
        n = 10
        while n <= 100000:
            # Generate a random string
            txt = random_string(n)
            
            # Build frequency dictionary
            huffman = defaultdict(int)
            first_count = 0
            for ch in txt:
                first_count += 1
                huffman[ch] += 1
            
            # Measure the time for Huffman encoding
            start_time = time.time()
            encoded_values, encode_operations = encode(huffman)
            end_time = time.time()
            
            # Calculate time taken in nanoseconds
            total_time = (end_time - start_time) * 1e9
            
            # Calculate total operations (frequency count + encode operations)
            total_operations = encode_operations + first_count
            
            # Store the results
            op_matrix[iteration_key].append(total_operations)
            time_matrix[iteration_key].append(total_time)
            
            # Increase the string length by an order of magnitude
            n *= 10
    
    # Calculate averages across iterations
    calculate_averages(op_matrix)
    calculate_averages(time_matrix)
    
    # Print the operation and time measurement results
    print_results(op_matrix, 'operation')
    print_results(time_matrix, 'time')

def calculate_averages(matrix):
    """
    Calculate the average of values across all iterations in the matrix.
    
    Parameters:
    ===========
    matrix : dict
        Dictionary containing lists of measurements for each iteration.
    """
    average = []
    num_iterations = len(matrix)
    
    for i in range(len(matrix['Iteration1'])):
        avg_value = sum(matrix[key][i] for key in matrix) / num_iterations
        average.append(avg_value)
    
    matrix['Average'] = average

def print_results(matrix, label):
    """
    Print the results stored in the matrix.
    
    Parameters:
    ===========
    matrix : dict
        Dictionary containing the results to print.
    label : str
        The label for the type of results (operation or time).
    """
    print(f'\n\n------------This is for the {label} measurement------------\n\n')
    for key in matrix:
        print(f'{key}: {matrix[key]}')

if __name__ == "__main__":
    main()
