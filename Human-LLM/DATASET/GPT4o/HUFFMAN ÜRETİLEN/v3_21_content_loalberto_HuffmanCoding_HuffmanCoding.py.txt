import sys
from collections import defaultdict

class Node:
    def __init__(self, frequency, key=''):
        """
        Initialize a node for the Huffman tree.

        Args:
            frequency (int): Frequency of the character or combined frequencies for internal nodes.
            key (str): The character associated with the node. Empty for internal nodes.
        """
        self.key = key
        self.frequency = frequency
        self.code = None
        self.left = None
        self.right = None

def assign_codes(node, code=''):
    """
    Recursively assign binary codes to the leaves of the Huffman tree.

    Args:
        node (Node): Current node in the Huffman tree.
        code (str): Current binary code prefix.
    """
    if node is None:
        return

    if node.left is None and node.right is None:
        node.code = code
        return

    assign_codes(node.left, code + '0')
    assign_codes(node.right, code + '1')

def build_huffman_tree(frequencies):
    """
    Build the Huffman tree based on character frequencies.

    Args:
        frequencies (dict): Dictionary with characters as keys and their frequencies as values.

    Returns:
        Node: Root of the Huffman tree.
    """
    nodes = [Node(freq, char) for char, freq in frequencies.items()]

    while len(nodes) > 1:
        nodes.sort(key=lambda node: node.frequency)
        left = nodes.pop(0)
        right = nodes.pop(0)
        merged_node = Node(left.frequency + right.frequency)
        merged_node.left = left
        merged_node.right = right
        nodes.append(merged_node)

    return nodes[0] if nodes else None

def generate_huffman_codes(root):
    """
    Generate the Huffman codes for all characters based on the Huffman tree.

    Args:
        root (Node): Root of the Huffman tree.

    Returns:
        dict: Dictionary with characters as keys and their Huffman codes as values.
    """
    code_dict = {}
    assign_codes(root)
    fill_code_dict(root, code_dict)
    return code_dict

def fill_code_dict(node, code_dict):
    """
    Populate the code dictionary with the codes assigned to each leaf node.

    Args:
        node (Node): Current node in the Huffman tree.
        code_dict (dict): Dictionary to store the codes.
    """
    if node.left is None and node.right is None:
        code_dict[node.key] = node.code
        return

    fill_code_dict(node.left, code_dict)
    fill_code_dict(node.right, code_dict)

def huffman_encoding(data):
    """
    Encode the input data using Huffman encoding.

    Args:
        data (str): Input string to encode.

    Returns:
        str, Node: Encoded data and the root of the Huffman tree.
    """
    if not data:
        return None, None

    frequencies = defaultdict(int)
    for char in data:
        frequencies[char] += 1

    if len(frequencies) == 1:
        single_char = next(iter(frequencies))
        encoded_data = '0' * len(data)
        root = Node(frequencies[single_char])
        root.left = Node(frequencies[single_char], single_char)
        return encoded_data, root

    root = build_huffman_tree(frequencies)
    code_dict = generate_huffman_codes(root)
    encoded_data = ''.join(code_dict[char] for char in data)

    return encoded_data, root

def huffman_decoding(encoded_data, root):
    """
    Decode the encoded data using the Huffman tree.

    Args:
        encoded_data (str): Encoded string of data.
        root (Node): Root of the Huffman tree.

    Returns:
        str: Decoded string.
    """
    if not encoded_data or not root:
        return None

    decoded_data = []
    current_node = root

    for bit in encoded_data:
        current_node = current_node.left if bit == '0' else current_node.right
        if current_node.left is None and current_node.right is None:
            decoded_data.append(current_node.key)
            current_node = root

    return ''.join(decoded_data)

def test_huffman(data):
    """
    Test the Huffman encoding and decoding on the provided data.

    Args:
        data (str): Input string to test.
    """
    print(f"Original data size: {sys.getsizeof(data)} bytes")
    print(f"Original data: {data}")

    encoded_data, tree = huffman_encoding(data)
    if encoded_data:
        print(f"Encoded data size: {sys.getsizeof(int(encoded_data, base=2))} bytes")
        print(f"Encoded data: {encoded_data}")

        decoded_data = huffman_decoding(encoded_data, tree)
        print(f"Decoded data size: {sys.getsizeof(decoded_data)} bytes")
        print(f"Decoded data: {decoded_data}")
    else:
        print("Encoding failed. Data is empty or None.")
