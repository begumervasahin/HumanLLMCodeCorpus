import sys
import unittest
import os
from WriteBitFile import WriteBitFile
from ReadBitFile import ReadBitFile

class ReadWriteTest(unittest.TestCase):
    def setUp(self):
        self.test_file = "data.bin"
        self.test_file_2 = "data2.bin"

    def tearDown(self):
        # Cleanup the created test files
        if os.path.exists(self.test_file):
            os.remove(self.test_file)
        if os.path.exists(self.test_file_2):
            os.remove(self.test_file_2)

    def test_mixed_operations(self):
        # Test writing and reading a mix of bits and data types
        w = WriteBitFile(self.test_file)
        w.writeBit(1)
        w.writeBit(0)
        w.writeBit(1)
        w.writeBit(1)
        w.writeUInt(10000042)
        w.writeBit(1)
        w.writeBit(0)
        w.writeUShort(50000)
        w.writeBit(1)
        w.writeUByte(42)
        w.close()

        r = ReadBitFile(self.test_file)
        self.assertEqual(r.readBit(), 1)
        self.assertEqual(r.readBit(), 0)
        self.assertEqual(r.readBit(), 1)
        self.assertEqual(r.readBit(), 1)
        self.assertEqual(r.readUInt(), 10000042)
        self.assertEqual(r.readBit(), 1)
        self.assertEqual(r.readBit(), 0)
        self.assertEqual(r.readUShort(), 50000)
        self.assertEqual(r.readBit(), 1)
        self.assertEqual(r.readUByte(), 42)
        r.close()

    def test_repeated_bit_patterns(self):
        # Test writing and reading repeated bit patterns
        w = WriteBitFile(self.test_file_2)
        for _ in range(4):
            w.writeBit(1)
            w.writeBit(0)
        w.close()

        r = ReadBitFile(self.test_file_2)
        for _ in range(4):
            self.assertEqual(r.readBit(), 1)
            self.assertEqual(r.readBit(), 0)
        r.close()

if __name__ == '__main__':
    unittest.main(argv=sys.argv, exit=False)
