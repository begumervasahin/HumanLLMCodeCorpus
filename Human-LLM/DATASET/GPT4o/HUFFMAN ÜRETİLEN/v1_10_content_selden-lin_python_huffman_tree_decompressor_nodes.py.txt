class HuffmanNode:
    """
    A node in a Huffman tree. Symbols occur only at leaves.
    
    Attributes:
    ===========
    symbol : int or None
        Symbol located in this node, if any.
    left : HuffmanNode or None
        Left subtree.
    right : HuffmanNode or None
        Right subtree.
    number : int or None
        Node number.
    """
    
    def __init__(self, symbol=None, left=None, right=None):
        """
        Create a new HuffmanNode with the given parameters.
        
        Parameters:
        ===========
        symbol : int or None
            Symbol to be stored in this node, if any.
        left : HuffmanNode or None
            A tree rooted at 'left', if any.
        right : HuffmanNode or None
            A tree rooted at 'right', if any.
        """
        self.symbol = symbol
        self.left = left
        self.right = right
        self.number = None
    
    def __eq__(self, other):
        """
        Return True if self is equivalent to other.
        
        Parameters:
        ===========
        other : HuffmanNode or Any
            A tree rooted at the HuffmanNode 'other'.
        
        Returns:
        ===========
        bool
            True if the nodes are equivalent.
        """
        return (isinstance(other, HuffmanNode) and 
                self.symbol == other.symbol and
                self.left == other.left and 
                self.right == other.right)
    
    def __lt__(self, other):
        """
        Return True if self is less than other.
        
        This method can be expanded to compare nodes based on some criteria.
        
        Parameters:
        ===========
        other : HuffmanNode or Any
            A tree rooted at the HuffmanNode 'other'.
        
        Returns:
        ===========
        bool
            Currently always returns False.
        """
        return False
    
    def __repr__(self):
        """
        Return constructor-style string representation.
        
        Returns:
        ===========
        str
            String representation of the HuffmanNode.
        """
        return f'HuffmanNode({self.symbol}, {self.left}, {self.right})'
    
    def is_leaf(self):
        """
        Return True if self is a leaf (no children).
        
        Returns:
        ===========
        bool
            True if the node is a leaf.
        """
        return self.left is None and self.right is None


class ReadNode:
    """
    A node as read from a compressed file. Each node consists of type and data
    information as described in the handout.
    
    Attributes:
    ===========
    l_type : int
        0/1 (if the corresponding HuffmanNode's left is a leaf).
    l_data : int
        A symbol or the node number of a HuffmanNode's left.
    r_type : int
        0/1 (if the corresponding HuffmanNode's right is a leaf).
    r_data : int
        A symbol or the node number of a HuffmanNode's right.
    """
    
    def __init__(self, l_type, l_data, r_type, r_data):
        """
        Create a new ReadNode with the given parameters.
        
        Parameters:
        ===========
        l_type : int
            Used to initialize self.l_type.
        l_data : int
            Used to initialize self.l_data.
        r_type : int
            Used to initialize self.r_type.
        r_data : int
            Used to initialize self.r_data.
        """
        self.l_type = l_type
        self.l_data = l_data
        self.r_type = r_type
        self.r_data = r_data
    
    def __repr__(self):
        """
        Return constructor-style string representation.
        
        Returns:
        ===========
        str
            String representation of the ReadNode.
        """
        return f'ReadNode({self.l_type}, {self.l_data}, {self.r_type}, {self.r_data})'
