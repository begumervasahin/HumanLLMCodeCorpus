class HuffElement:
    """
    Represents an element in a Huffman coding tree, containing a character,
    its frequency, and its corresponding Huffman code.
    """

    def __init__(self, char: str):
        """
        Initializes a HuffElement with the given character.
        Frequency is initialized to zero, and the Huffman code is an empty string.
        """
        self._char = char
        self._frequency = 0
        self._code = ""

    def increment_frequency(self):
        """Increments the frequency count by one."""
        self._frequency += 1

    @property
    def frequency(self) -> int:
        """Gets the frequency count of the character."""
        return self._frequency

    @frequency.setter
    def frequency(self, value: int):
        """Sets the frequency count of the character."""
        self._frequency = value

    @property
    def code(self) -> str:
        """Gets the Huffman code for the character."""
        return self._code

    @code.setter
    def code(self, value: str):
        """Sets the Huffman code for the character."""
        self._code = value

    @property
    def char(self) -> str:
        """Gets the character associated with this element."""
        return self._char

    @char.setter
    def char(self, value: str):
        """Sets the character associated with this element."""
        self._char = value

    def compare(self, other: 'HuffElement') -> int:
        """
        Compares this HuffElement with another based on frequency.
        Returns:
            1  : if this element's frequency is greater
           -1  : if this element's frequency is lesser
            0  : if frequencies are equal
        """
        if self._frequency > other.frequency:
            return 1
        elif self._frequency < other.frequency:
            return -1
        else:
            return 0

    def __str__(self) -> str:
        """Returns a string representation of the HuffElement."""
        return f"Char: '{self._char}', Code: '{self._code}', Frequency: {self._frequency}"
