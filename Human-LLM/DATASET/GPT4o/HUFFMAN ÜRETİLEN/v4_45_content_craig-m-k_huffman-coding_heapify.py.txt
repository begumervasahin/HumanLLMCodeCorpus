class Heap:
    def __init__(self, heap_type: str):
        if heap_type not in {'max', 'min'}:
            raise ValueError("heap_type must be either 'max' or 'min'")
        self.queue = []
        self.heap_size = 0
        self.heap_type = heap_type

    def __len__(self) -> int:
        return self.heap_size

    def perc_up(self, index: int):
        parent = (index - 1) // 2
        while index > 0 and self._compare(self.queue[parent], self.queue[index]):
            self.queue[index], self.queue[parent] = self.queue[parent], self.queue[index]
            index = parent
            parent = (index - 1) // 2

    def _compare(self, parent_val, child_val) -> bool:
        """Returns True if a swap is needed based on heap type."""
        if self.heap_type == 'max':
            return parent_val > child_val
        else:
            return parent_val < child_val

    def insert(self, key):
        self.queue.append(key)
        self.heap_size += 1
        self.perc_up(self.heap_size - 1)

    def pop(self):
        if self.heap_size == 0:
            raise IndexError("pop from empty heap")
        
        self.queue[0], self.queue[-1] = self.queue[-1], self.queue[0]
        popped_key = self.queue.pop()
        self.heap_size -= 1
        
        if self.heap_size > 0:
            self._heapify(0)
        
        return popped_key

    def _heapify(self, index: int):
        left = 2 * index + 1
        right = 2 * index + 2
        extreme = index

        if left < self.heap_size and self._compare(self.queue[extreme], self.queue[left]):
            extreme = left
        if right < self.heap_size and self._compare(self.queue[extreme], self.queue[right]):
            extreme = right

        if extreme != index:
            self.queue[index], self.queue[extreme] = self.queue[extreme], self.queue[index]
            self._heapify(extreme)

    def build_heap(self):
        for i in range((self.heap_size // 2) - 1, -1, -1):
            self._heapify(i)
