import random

class HuffmanCoding:
    """
    A simple mock HuffmanCoding class for demonstration purposes.
    """

    def encode(self, data):
        """
        Encodes the input data into a binary string using mock Huffman coding.

        Parameters
        ----------
        data : str
            The input string to encode.

        Returns
        -------
        str
            The encoded binary string.
        """
        return ''.join(format(ord(char), '08b') for char in data)

    def decode(self, encoded_data):
        """
        Decodes a binary string back into the original string.

        Parameters
        ----------
        encoded_data : str
            The encoded binary string to decode.

        Returns
        -------
        str
            The decoded original string.
        """
        return ''.join(chr(int(encoded_data[i:i + 8], 2)) for i in range(0, len(encoded_data), 8))

class ConvolutionalCoding:
    """
    A simple mock ConvolutionalCoding class for demonstration purposes.
    """

    def encode(self, data):
        """
        Encodes the input data using mock convolutional coding.

        Parameters
        ----------
        data : str
            The input binary string to encode.

        Returns
        -------
        str
            The encoded data (identity function in this mock).
        """
        return data

    def decode(self, encoded_data):
        """
        Decodes the encoded data using mock convolutional decoding.

        Parameters
        ----------
        encoded_data : str
            The encoded binary string to decode.

        Returns
        -------
        str
            The decoded data (identity function in this mock).
        """
        return encoded_data

def introduce_noise(data, noise_probability=0.1):
    """
    Introduces noise into a binary string with a given probability.

    Parameters
    ----------
    data : str
        The input binary string.
    noise_probability : float, optional
        The probability of flipping each bit (default is 0.1).

    Returns
    -------
    str
        The binary string with noise introduced.
    """
    return ''.join(
        str(1 - int(bit)) if random.uniform(0, 1) < noise_probability else bit
        for bit in data
    )

def main():
    # Instantiate coding classes
    huffman = HuffmanCoding()
    convolutional = ConvolutionalCoding()

    # Input string
    input_data = "negarmirgati"
    print('Input:', input_data)

    # Huffman encoding
    huffman_encoded = huffman.encode(input_data)
    print('Huffman encoded:', huffman_encoded)

    # Convolutional encoding
    convolutional_encoded = convolutional.encode(huffman_encoded)
    print('Convolutional encoded:', convolutional_encoded)

    # Adding noise to the encoded data
    noisy_data = introduce_noise(convolutional_encoded)
    print('Noisy data:', noisy_data)

    # Convolutional decoding
    convolutional_decoded = convolutional.decode(noisy_data)
    print('Convolutional decoded:', convolutional_decoded)

    # Huffman decoding
    huffman_decoded = huffman.decode(convolutional_decoded)
    print('Huffman decoded:', huffman_decoded)

if __name__ == "__main__":
    main()
