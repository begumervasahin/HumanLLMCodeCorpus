class Node:
    def __init__(self, value, probability):
        self.value = value
        self.probability = probability
        self.code = ""

    def __str__(self):
        return f"({self.value})"

    def get_prob(self):
        return self.probability

    def get_val(self):
        return self.value

    def set_code(self, code):
        self.code = code

    def get_code(self):
        return self.code

class NodeJoin:
    def __init__(self, node1, node2):
        self.node1 = node1
        self.node2 = node2
        self.probability = node1.get_prob() + node2.get_prob()
        self.code = ""

    def __str__(self):
        return f"({self.node1},{self.node2})"

    def get_prob(self):
        return self.probability

    def get_val(self):
        return self.node1.get_val() + self.node2.get_val()

    def set_code(self, code):
        self.code = code

    def get_code(self):
        return self.code

def split_nodes(node, final_codes):
    code = node.get_code()
    node1, node2 = node.node1, node.node2
    node1.set_code(code + "1")
    node2.set_code(code + "0")

    if isinstance(node1, NodeJoin):
        split_nodes(node1, final_codes)
    else:
        final_codes.append(node1)

    if isinstance(node2, NodeJoin):
        split_nodes(node2, final_codes)
    else:
        final_codes.append(node2)

def main():
    nodes = []
    final_codes = []

    # Input node data
    while True:
        n = input("Enter node (format: value,probability) or leave empty to finish: ")
        if not n:
            break
        try:
            value, probability = n.split(",")
            probability = float(probability)
            nodes.append(Node(value.strip(), probability))
        except ValueError:
            print("Invalid input format. Please enter in format: value,probability")

    # Sort nodes by probability in descending order
    if nodes and sorted(nodes, key=lambda x: x.get_prob(), reverse=True) != nodes:
        print("Nodes not sorted! Sorting...")
        nodes.sort(key=lambda x: x.get_prob(), reverse=True)
        print("New nodes order:")
        for node in nodes:
            print(f"  {node}")

    # Build Huffman tree
    while len(nodes) > 1:
        new_nodes = nodes[:-2]
        new_node = NodeJoin(nodes[-1], nodes[-2])

        for i, obj in enumerate(new_nodes):
            if new_node.get_prob() >= obj.get_prob():
                new_nodes.insert(i, new_node)
                break
        else:
            new_nodes.append(new_node)

        nodes = new_nodes

    # Generate Huffman codes
    if len(nodes) == 1:
        root_node = nodes[0]
        root_node.set_code("")

        if isinstance(root_node, Node):
            final_codes.append(root_node)
        else:
            split_nodes(root_node, final_codes)

    # Output the final Huffman codes
    final_codes.sort(key=lambda x: x.get_val())
    print("\nHuffman Codes:")
    for node in final_codes:
        print(f"Node {node.get_val()} has code {node.get_code()}")

if __name__ == "__main__":
    main()
