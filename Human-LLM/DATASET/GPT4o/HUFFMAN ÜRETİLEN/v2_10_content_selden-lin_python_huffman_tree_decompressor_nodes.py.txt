class HuffmanNode:
    """
    A node in a Huffman tree. Symbols are stored only at leaf nodes.
    
    Attributes:
    ----------
    symbol : int or None
        The symbol stored at this node, if any.
    left : HuffmanNode or None
        The left subtree.
    right : HuffmanNode or None
        The right subtree.
    number : int or None
        An optional node number.
    """
    
    def __init__(self, symbol=None, left=None, right=None):
        """
        Initialize a HuffmanNode instance.

        Parameters:
        ----------
        symbol : int or None, optional
            The symbol to store in this node, if any. Default is None.
        left : HuffmanNode or None, optional
            The left subtree. Default is None.
        right : HuffmanNode or None, optional
            The right subtree. Default is None.
        """
        self.symbol = symbol
        self.left = left
        self.right = right
        self.number = None
    
    def __eq__(self, other):
        """
        Check if this node is equivalent to another node.

        Parameters:
        ----------
        other : HuffmanNode
            The other node to compare with.

        Returns:
        -------
        bool
            True if the nodes are equivalent, False otherwise.
        """
        if not isinstance(other, HuffmanNode):
            return False
        return (self.symbol == other.symbol and
                self.left == other.left and 
                self.right == other.right)
    
    def __lt__(self, other):
        """
        Compare this node with another node.

        This method can be expanded to implement custom comparison logic.

        Parameters:
        ----------
        other : HuffmanNode
            The other node to compare with.

        Returns:
        -------
        bool
            Currently always returns False.
        """
        return False
    
    def __repr__(self):
        """
        Return a string representation of the node.

        Returns:
        -------
        str
            A string that represents the HuffmanNode.
        """
        return f"HuffmanNode(symbol={self.symbol}, left={self.left}, right={self.right})"
    
    def is_leaf(self):
        """
        Check if the node is a leaf (i.e., has no children).

        Returns:
        -------
        bool
            True if the node is a leaf, False otherwise.
        """
        return self.left is None and self.right is None


class ReadNode:
    """
    A node read from a compressed file, containing type and data information.
    
    Attributes:
    ----------
    l_type : int
        Indicates if the left child is a leaf (1) or a subtree (0).
    l_data : int
        The symbol or node number for the left child.
    r_type : int
        Indicates if the right child is a leaf (1) or a subtree (0).
    r_data : int
        The symbol or node number for the right child.
    """
    
    def __init__(self, l_type, l_data, r_type, r_data):
        """
        Initialize a ReadNode instance.

        Parameters:
        ----------
        l_type : int
            Type indicator for the left child (leaf or subtree).
        l_data : int
            Data for the left child (symbol or node number).
        r_type : int
            Type indicator for the right child (leaf or subtree).
        r_data : int
            Data for the right child (symbol or node number).
        """
        self.l_type = l_type
        self.l_data = l_data
        self.r_type = r_type
        self.r_data = r_data
    
    def __repr__(self):
        """
        Return a string representation of the node.

        Returns:
        -------
        str
            A string that represents the ReadNode.
        """
        return (f"ReadNode(l_type={self.l_type}, l_data={self.l_data}, "
                f"r_type={self.r_type}, r_data={self.r_data})")
