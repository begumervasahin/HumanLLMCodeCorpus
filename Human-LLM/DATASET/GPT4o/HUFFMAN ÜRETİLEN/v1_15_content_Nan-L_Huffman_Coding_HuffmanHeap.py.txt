class HuffmanTree:
    """
    A simple Huffman tree node class for demonstration purposes.
    This class should have a get_freq() method that returns the frequency
    associated with the tree node.
    """
    def __init__(self, freq, symbol=None, left=None, right=None):
        self.freq = freq
        self.symbol = symbol
        self.left = left
        self.right = right

    def get_freq(self):
        return self.freq

    def __repr__(self):
        return f"HuffmanTree(freq={self.freq}, symbol={self.symbol})"

class HuffmanHeap:
    """
    A heap-like structure for managing two lists of Huffman trees.
    
    Attributes:
    old: The old list of Huffman trees, sorted.
    new: The new list of Huffman trees, sorted.
    """
    def __init__(self, old, new=None):
        self.old = old if old is not None else []
        self.new = new if new is not None else []

    def enqueue(self, item):
        """
        Insert a Huffman tree into the new list.
        
        :param item: HuffmanTree object to be added to the new list.
        :return: None
        """
        self.new.append(item)
        self.new.sort(key=lambda x: x.get_freq())  # Ensure new list is sorted

    def dequeue(self):
        """
        Remove and return the smallest item from the heap.
        
        :return: The HuffmanTree object with the smallest frequency; None if both lists are empty.
        """
        if len(self.old) == 0 and len(self.new) > 0:
            return self.new.pop(0)
        elif len(self.new) == 0 and len(self.old) > 0:
            return self.old.pop(0)
        elif len(self.new) > 0 and len(self.old) > 0:
            if self.old[0].get_freq() >= self.new[0].get_freq():
                return self.new.pop(0)
            else:
                return self.old.pop(0)
        else:
            print("Both old and new lists are empty!")
            return None

if __name__ == "__main__":
    # Example usage:
    old_list = [HuffmanTree(freq=5), HuffmanTree(freq=10)]
    new_list = [HuffmanTree(freq=2), HuffmanTree(freq=8)]

    huffman_heap = HuffmanHeap(old_list, new_list)

    # Enqueue a new Huffman tree
    huffman_heap.enqueue(HuffmanTree(freq=3))

    # Dequeue the smallest Huffman tree
    print("Dequeued:", huffman_heap.dequeue())

    # Dequeue another Huffman tree
    print("Dequeued:", huffman_heap.dequeue())

    # Dequeue remaining Huffman trees
    while True:
        item = huffman_heap.dequeue()
        if item is None:
            break
        print("Dequeued:", item)
