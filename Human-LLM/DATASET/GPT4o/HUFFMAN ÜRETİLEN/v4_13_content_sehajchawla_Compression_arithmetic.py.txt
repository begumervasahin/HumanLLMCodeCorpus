from math import floor, ceil
from sys import stdout as so
from bisect import bisect

def encode(x, p):
    """
    Encode the input sequence `x` using arithmetic encoding with probabilities `p`.

    Args:
        x (list): The input sequence to encode.
        p (dict): A dictionary of symbol probabilities.

    Returns:
        list: A list of encoded bits.
    """
    precision = 32
    one = (1 << precision) - 1
    quarter = (one + 1) // 4
    half = 2 * quarter
    threequarters = 3 * quarter

    # Filter and normalize probabilities
    p = {a: p[a] for a in p if p[a] > 0}
    
    # Calculate cumulative frequency dictionary
    cumulative_freq = [0]
    for a in p:
        cumulative_freq.append(cumulative_freq[-1] + p[a])
    cumulative_freq.pop()
    cumulative_freq = {a: freq for a, freq in zip(p, cumulative_freq)}

    # Encoding process
    y = []
    lo, hi = 0, one
    straddle = 0

    for k, symbol in enumerate(x):
        if k % 100 == 0:
            so.write(f'Arithmetic encoded {int(floor(k / len(x) * 100))}%    \r')
            so.flush()

        lohi_range = hi - lo + 1
        lo = int(ceil(lo + cumulative_freq[symbol] * lohi_range))
        hi = int(floor(lo + p[symbol] * lohi_range))

        if lo == hi:
            raise ValueError('Zero interval!')

        while True:
            if hi < half:
                y.append(0)
                y.extend([1] * straddle)
                straddle = 0
            elif lo >= half:
                y.append(1)
                y.extend([0] * straddle)
                straddle = 0
                lo -= half
                hi -= half
            elif quarter <= lo < threequarters and quarter <= hi < threequarters:
                straddle += 1
                lo -= quarter
                hi -= quarter
            else:
                break

            lo *= 2
            hi = 2 * hi + 1

    # Final bits to output
    straddle += 1
    if lo < quarter:
        y.append(0)
        y.extend([1] * straddle)
    else:
        y.append(1)
        y.extend([0] * straddle)

    return y

def decode(y, p, n):
    """
    Decode the encoded bit sequence `y` using arithmetic decoding with probabilities `p`.

    Args:
        y (list): The encoded bit sequence.
        p (dict): A dictionary of symbol probabilities.
        n (int): The number of symbols to decode.

    Returns:
        list: The decoded sequence of symbols.
    """
    precision = 32
    one = (1 << precision) - 1
    quarter = (one + 1) // 4
    half = 2 * quarter
    threequarters = 3 * quarter

    # Filter and normalize probabilities
    p = {a: p[a] for a in p if p[a] > 0}
    alphabet = list(p)

    # Calculate cumulative frequency list
    cumulative_freq = [0]
    for a in p:
        cumulative_freq.append(cumulative_freq[-1] + p[a])
    cumulative_freq.pop()
    probabilities = list(p.values())

    y.extend([0] * precision)
    x = [0] * n

    value = int(''.join(map(str, y[:precision])), 2)
    y_position = precision
    lo, hi = 0, one
    x_position = 0

    while x_position < n:
        if x_position % 100 == 0:
            so.write(f'Arithmetic decoded {int(floor(x_position / n * 100))}%    \r')
            so.flush()

        lohi_range = hi - lo + 1
        a_index = bisect(cumulative_freq, (value - lo) / lohi_range) - 1
        x[x_position] = alphabet[a_index]

        lo = int(ceil(lo + cumulative_freq[a_index] * lohi_range))
        hi = int(floor(lo + probabilities[a_index] * lohi_range))

        if lo == hi:
            raise ValueError('Zero interval!')

        while True:
            if hi < half:
                pass
            elif lo >= half:
                lo -= half
                hi -= half
                value -= half
            elif quarter <= lo < threequarters and quarter <= hi < threequarters:
                lo -= quarter
                hi -= quarter
                value -= quarter
            else:
                break

            lo *= 2
            hi = 2 * hi + 1
            value = 2 * value + y[y_position]
            y_position += 1

        x_position += 1

    return x
