def sub_sequence(st1, st2):
    """
    Finding the longest common subsequence in string1 and string2 in order.
    Examples:
        * 'ABCABA' , 'ABBA' --> 'ABBA'
        * 'aaaaa' , 'aa' --> 'aa'
        * 'AGGTAB' , 'GXTXAYB' --> 'GTAB'
        * 'ABAZDC' , 'BACBAD' --> 'ABAD'
    """
    st1 = list(st1)
    st2 = list(st2)
    
    # Determine the shorter and longer strings
    if len(st1) <= len(st2):
        lower = st1
        bigger = st2
    else:
        lower = st2
        bigger = st1

    longest_subseq = []

    for start in range(len(lower)):
        current_subseq = []
        index_bigger = 0

        for char in lower[start:]:
            if char in bigger[index_bigger:]:
                current_subseq.append(char)
                index_bigger += bigger[index_bigger:].index(char) + 1

        # Check for duplicates and invalid subsequences
        if all(current_subseq.count(char) <= lower.count(char) and current_subseq.count(char) <= bigger.count(char) for char in current_subseq):
            if len(current_subseq) > len(longest_subseq):
                longest_subseq = current_subseq

    return ''.join(longest_subseq)

# Examples
print(sub_sequence('ABCABA', 'ABBA'))  # Output: 'ABBA'
print(sub_sequence('aaaaa', 'aa'))     # Output: 'aa'
print(sub_sequence('AGGTAB', 'GXTXAYB'))  # Output: 'GTAB'
print(sub_sequence('ABAZDC', 'BACBAD'))  # Output: 'ABAD'
