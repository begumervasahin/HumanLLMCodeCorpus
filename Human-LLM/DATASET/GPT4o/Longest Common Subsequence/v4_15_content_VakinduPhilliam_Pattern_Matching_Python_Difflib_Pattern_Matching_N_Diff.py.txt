from difflib import ndiff

# Define the original and modified strings
original_text = 'one\ntwo\nthree\n'
modified_text = 'ore\ntree\nemu\n'

# Split the strings into lines while keeping the line breaks
original_lines = original_text.splitlines(keepends=True)
modified_lines = modified_text.splitlines(keepends=True)

# Generate the diff using ndiff
diff = ndiff(original_lines, modified_lines)

# Print the diff result as a single string
print(''.join(diff), end="")
