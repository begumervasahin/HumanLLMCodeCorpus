import sys
from time import time

def matrix_chain_order(p, n):
    # Initialize matrices for storing minimum costs and split points
    m = [[0 for _ in range(n)] for _ in range(n)]
    s = [[0 for _ in range(n)] for _ in range(n)]
    
    # Set diagonal values to zero (base case)
    for i in range(1, n):
        m[i][i] = 0
    
    # Compute minimum multiplication costs and store split points
    for L in range(2, n):
        for i in range(1, n - L + 1):
            j = i + L - 1
            m[i][j] = sys.maxsize
            for k in range(i, j):
                q = m[i][k] + m[k + 1][j] + p[i - 1] * p[k] * p[j]
                if q < m[i][j]:
                    m[i][j] = q
                    s[i][j] = k
    
    return m, s

def print_optimal_parens(s, i, j):
    if i == j:
        print(f"A{i}", end="")
    else:
        print("(", end="")
        print_optimal_parens(s, i, s[i][j])
        print_optimal_parens(s, s[i][j] + 1, j)
        print(")", end="")

# Test matrices dimensions
arr = [
    [2, 3], [3, 4, 6], [1, 3, 6, 7], [5, 4, 6, 2, 7], 
    [2, 4, 5, 6, 7, 8], [2, 4, 5, 3, 6, 7, 8], [4, 3, 2, 5, 6, 4, 3, 2],
    [3, 5, 6, 4, 3, 5, 6, 7, 8], [3, 4, 5, 3, 6, 7, 8, 9, 5, 10], 
    [2, 4, 5, 7, 8, 2, 4, 10, 12, 5, 6]
]

n = []
elapsed_times = []

# Perform matrix chain multiplication for each test case
for dims in arr:
    n_val = len(dims) - 1
    n.append(n_val)
    start_time = time()
    
    m, s = matrix_chain_order(dims, len(dims))
    
    print(f"\nP={dims}")
    print("\nm table\n")
    for row in m[:n_val]:
        print("\t".join(map(str, row[:n_val])))
    
    print("\ns table\n")
    for row in s[:n_val]:
        print("\t".join(map(str, row[:n_val])))
    
    print("\nAns:", end="")
    print_optimal_parens(s, 1, n_val)
    print("")
    
    end_time = time()
    elapsed_times.append(end_time - start_time)

print("\n\nn\tElapsed Time")
for i in range(10):
    print(f"{n[i]}\t{elapsed_times[i]}")
