import sys

# Global variable to keep track of the number of recursive calls
num_recursive_calls = 0

def recursive_lcs(x, y, reconstruct=False):
    """
    Compute the Longest Common Subsequence (LCS) of two strings x and y using recursion.

    :param x: First string sequence
    :param y: Second string sequence
    :param reconstruct: Boolean flag to indicate whether to reconstruct the LCS string. Default is False.
    :return: Length of the LCS or the LCS string itself if reconstruct is True
    """
    global num_recursive_calls
    num_recursive_calls += 1

    # Base case: If either string is empty, return 0 or an empty string
    if len(x) == 0 or len(y) == 0:
        return "" if reconstruct else 0

    # If the last characters of both strings match
    if x[-1] == y[-1]:
        if reconstruct:
            return recursive_lcs(x[:-1], y[:-1], reconstruct) + x[-1]
        else:
            return 1 + recursive_lcs(x[:-1], y[:-1], reconstruct)
    else:
        # If the last characters do not match
        if reconstruct:
            return max(recursive_lcs(x, y[:-1], reconstruct), recursive_lcs(x[:-1], y, reconstruct), key=len)
        else:
            return max(recursive_lcs(x, y[:-1], reconstruct), recursive_lcs(x[:-1], y, reconstruct))

def test(x, y, reconstruct):
    """
    A simple test function to run the recursive LCS algorithm.

    :param x: First string sequence
    :param y: Second string sequence
    :param reconstruct: Boolean flag to indicate whether to reconstruct the LCS string
    :return: Length of the LCS or the LCS string itself if reconstruct is True
    """
    return recursive_lcs(x, y, reconstruct)

if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Usage: python3 " + __file__ + " <filename> [0/1]")
        print("filename - file containing two sequences")
        print("0 - without reconstruction, 1 - with reconstruction")
    else:
        file = sys.argv[1]
        with open(file, 'r') as f:
            x = f.readline().strip()
            y = f.readline().strip()
            reconstruct = False
            if len(sys.argv) == 3:
                reconstruct = int(sys.argv[2]) == 1
            if reconstruct:
                lcs = test(x, y, reconstruct)
                print("LCS: " + lcs)
                print("LCS Length: " + str(len(lcs)))
            else:
                print("LCS Length: " + str(test(x, y, reconstruct)))
            print("Number of Recursive Calls: " + str(num_recursive_calls))
