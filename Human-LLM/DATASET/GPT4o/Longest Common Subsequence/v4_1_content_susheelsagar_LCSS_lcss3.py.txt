def similarity_lcss(traj1, traj2):
    def classify_by_day(traj):
        days = {"Fri": [], "Sat": [], "Sun": [], "Mon": [], "Tue": [], "Wed": [], "Thu": []}
        for row in traj:
            day = row[1].strip('"')
            if day in days:
                days[day].append(row)
        return days

    def cal_sim(traj1, traj2):
        score = 0
        while traj2:
            len2 = len(traj1) - 1
            while len2 > -1:
                if traj2[-1][2] == traj1[len2][2] and traj2[-1][0] == traj1[len2][0]:
                    score += 1
                len2 -= 1
            traj2.pop()
        return score

    score = 0
    days1 = classify_by_day(traj1)
    days2 = classify_by_day(traj2)

    for day in days1:
        score += cal_sim(days1[day], days2[day]) if len(days1[day]) < len(days2[day]) else cal_sim(days2[day], days1[day])

    distance = float(score) / min(len(traj1), len(traj2))
    return 1.0 - distance
