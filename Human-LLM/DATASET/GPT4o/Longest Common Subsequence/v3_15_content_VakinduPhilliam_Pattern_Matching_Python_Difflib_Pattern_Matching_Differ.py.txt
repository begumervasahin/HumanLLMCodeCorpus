import sys
import difflib

def generate_context_diff(seq1, seq2, from_file, to_file):
    """
    Generate a context diff between two sequences of lines.

    Args:
        seq1 (list of str): The first sequence of lines.
        seq2 (list of str): The second sequence of lines.
        from_file (str): The name of the first file (used in the diff output).
        to_file (str): The name of the second file (used in the diff output).

    Returns:
        generator: A generator that yields lines of the context diff.
    """
    return difflib.context_diff(seq1, seq2, fromfile=from_file, tofile=to_file)

def main():
    # Define the lists to compare
    list1 = ['bacon\n', 'eggs\n', 'ham\n', 'guido\n']
    list2 = ['python\n', 'eggy\n', 'hamster\n', 'guido\n']

    # Generate the context diff
    diff_result = generate_context_diff(list1, list2, fromfile='before.py', tofile='after.py')

    # Output the diff to the standard output
    sys.stdout.writelines(diff_result)

if __name__ == "__main__":
    main()
