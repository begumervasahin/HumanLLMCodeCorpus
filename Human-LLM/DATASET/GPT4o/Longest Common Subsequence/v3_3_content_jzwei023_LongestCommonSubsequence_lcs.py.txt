def calculate_LCS(A, B):
    """
    Calculate the Longest Common Subsequence (LCS) of two strings.

    Args:
    A (str): First string.
    B (str): Second string.

    Returns:
    tuple: Length of the LCS and the LCS string itself.
    """
    n = len(A)
    m = len(B)

    # Initialize the DP and direction tables
    opt = [[0] * (m + 1) for _ in range(n + 1)]
    pi = [[0] * (m + 1) for _ in range(n + 1)]

    # Fill the DP and direction tables
    for i in range(1, n + 1):
        for j in range(1, m + 1):
            if A[i - 1] == B[j - 1]:
                opt[i][j] = opt[i - 1][j - 1] + 1
                pi[i][j] = 0  # Diagonal
            elif opt[i][j - 1] >= opt[i - 1][j]:
                opt[i][j] = opt[i][j - 1]
                pi[i][j] = 1  # Left
            else:
                opt[i][j] = opt[i - 1][j]
                pi[i][j] = 2  # Up

    # Trace back to find the LCS
    i, j = n, m
    LCS = []
    while i > 0 and j > 0:
        if pi[i][j] == 0:  # Diagonal
            LCS.append(A[i - 1])
            i -= 1
            j -= 1
        elif pi[i][j] == 2:  # Up
            i -= 1
        else:  # Left
            j -= 1

    return str(opt[n][m]), ''.join(reversed(LCS))

def read_input(file_path):
    """
    Read input strings from a file.

    Args:
    file_path (str): Path to the input file.

    Returns:
    tuple: Two strings read from the file.
    """
    with open(file_path, "r") as file:
        lines = file.readlines()
    return lines[0].strip(), lines[1].strip()

def write_output(file_path, length, sequence):
    """
    Write the LCS length and sequence to a file.

    Args:
    file_path (str): Path to the output file.
    length (str): Length of the LCS.
    sequence (str): LCS string.
    """
    with open(file_path, "w") as file:
        file.write(length + "\n")
        file.write(sequence)

# File paths
input_file = "input.txt"
output_file = "output.txt"

# Read input strings from file
A, B = read_input(input_file)

# Calculate LCS
length, sequence = calculate_LCS(A, B)

# Write results to output file
write_output(output_file, length, sequence)
