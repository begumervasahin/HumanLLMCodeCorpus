from typing import Iterable, Optional, Callable, Any, TypeVar
import ctypes
from sklearn.metrics.pairwise import cosine_similarity

_T = TypeVar('_T')

def c_mul(a: int, b: int) -> int:
    return ctypes.c_int64((a * b) & 0xffffffff).value

class TokenSequence(list[_T]):
    def extend(self, iterable: Iterable[_T]) -> None:
        super().extend(iterable)

    def insert(self, index: int, obj: _T) -> None:
        super().insert(index, obj)

    def append(self, obj: _T) -> None:
        super().append(obj)

    def remove(self, obj: _T) -> None:
        super().remove(obj)

    def pop(self, index: int = ...) -> _T:
        return super().pop(index)

    def sort(self, *, key: Optional[Callable[[_T], Any]] = None, reverse: bool = False) -> None:
        super().sort(key=key, reverse=reverse)

    def clear(self) -> None:
        super().clear()

    def reverse(self) -> None:
        super().reverse()

    def __hash__(self) -> int:
        value = 0x345678
        for item in self:
            value = c_mul(1000003, value) ^ hash(item)
        value ^= len(self)
        if value == -1:
            value = -2
        return value
