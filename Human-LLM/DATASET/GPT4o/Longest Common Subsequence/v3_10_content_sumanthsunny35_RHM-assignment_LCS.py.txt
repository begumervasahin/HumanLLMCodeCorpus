import re

def verify(sequence):
    """
    Verify that the sequence contains only valid DNA nucleotides ('A', 'T', 'C', 'G') and spaces.
    
    Args:
    sequence (str): The DNA sequence to verify.
    
    Returns:
    bool: True if the sequence is valid, False otherwise.
    """
    return bool(re.match("^[ATCG ]*$", sequence))

def lcs(X, Y):
    """
    Compute the longest common subsequence (LCS) of sequences X and Y using dynamic programming.
    
    Args:
    X (str): The first DNA sequence.
    Y (str): The second DNA sequence.
    
    Returns:
    str: The longest common subsequence of X and Y.
    """
    m, n = len(X), len(Y)
    L = [[0] * (n + 1) for _ in range(m + 1)]

    # Build the LCS table in bottom-up fashion.
    for i in range(m + 1):
        for j in range(n + 1):
            if i == 0 or j == 0:
                L[i][j] = 0
            elif X[i - 1] == Y[j - 1]:
                L[i][j] = L[i - 1][j - 1] + 1
            else:
                L[i][j] = max(L[i - 1][j], L[i][j - 1])

    # The length of LCS is L[m][n].
    index = L[m][n]

    # Create a list to store the LCS characters.
    lcs_str = [""] * index

    # Start from the bottom-right corner of the LCS table.
    i, j = m, n
    while i > 0 and j > 0:
        # If characters match, they are part of LCS.
        if X[i - 1] == Y[j - 1]:
            lcs_str[index - 1] = X[i - 1]
            i -= 1
            j -= 1
            index -= 1
        # Move in the direction of the larger value.
        elif L[i - 1][j] > L[i][j - 1]:
            i -= 1
        else:
            j -= 1

    return "".join(lcs_str)

# Input sequences
s1 = input("Enter the First Sequence: ").upper()
s2 = input("Enter the Querying Sequence: ").upper()

# Verify the sequences and call LCS function if valid.
if verify(s1) and verify(s2):
    if len(s1) >= len(s2):
        print(f"First Sequence: {s1}\nQuerying Sequence: {s2}")
        result = lcs(s1, s2)
        print(f"\nLCS of the given two DNA sequences is {result}\nand length of it is: {len(result)}")
    else:
        print("Querying Sequence should be smaller than or equal to the First Sequence")
else:
    print("The Sequences should contain 'A', 'T', 'C', 'G', or spaces only")
