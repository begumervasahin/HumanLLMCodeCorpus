import pandas as pd
import numpy as np
from sklearn.feature_extraction.text import CountVectorizer, TfidfTransformer
from sklearn.decomposition import TruncatedSVD
from sklearn import preprocessing
from sklearn.model_selection import KFold
from sklearn.ensemble import RandomForestClassifier, VotingClassifier
from sklearn.neighbors import KNeighborsClassifier
from sklearn.pipeline import Pipeline
from sklearn.metrics import accuracy_score

def compute_and_print(X_train, Y_train, pipeline, kf):
    """
    Perform K-Fold cross-validation and compute accuracy scores for each fold.
    
    Parameters:
    X_train (numpy array): Feature data.
    Y_train (numpy array): Target labels.
    pipeline (Pipeline): Machine learning pipeline.
    kf (KFold): K-Fold cross-validator.
    
    Returns:
    list: Accuracy scores for each fold.
    """
    stats = []
    for train_index, test_index in kf.split(X_train):
        X_train_fold, X_test_fold = X_train[train_index], X_train[test_index]
        Y_train_fold, Y_test_fold = Y_train[train_index], Y_train[test_index]
        
        # Fit the pipeline and predict
        predictions = pipeline.fit(X_train_fold, Y_train_fold).predict(X_test_fold)
        
        # Calculate accuracy and append to stats
        accuracy = accuracy_score(Y_test_fold, predictions)
        stats.append(accuracy)
    
    return stats

def main():
    # Load the data
    df = pd.read_csv("grids.csv")

    # Encode the labels
    label_encoder = preprocessing.LabelEncoder()
    label_encoder.fit(df["TripId"])
    Y_train = label_encoder.transform(df["TripId"])

    # Extract the features
    X_train = df['Grids'].values

    # Define the vectorizer and transformers
    vectorizer = CountVectorizer()
    transformer = TfidfTransformer()
    svd = TruncatedSVD(n_components=300, random_state=42)

    # Define the K-Fold cross-validator
    kf = KFold(n_splits=10, shuffle=True, random_state=42)

    # Define the classifiers
    clf1 = RandomForestClassifier(n_estimators=40, n_jobs=-1)
    clf2 = RandomForestClassifier(n_estimators=50, n_jobs=-1)
    clf3 = KNeighborsClassifier(n_neighbors=7, n_jobs=-1)

    # Combine the classifiers in a voting classifier
    voting_clf = VotingClassifier(estimators=[('rf1', clf1), ('rf2', clf2), ('knn', clf3)], voting='hard')

    # Create the pipeline
    pipeline = Pipeline([
        ('vect', vectorizer),
        ('tfidf', transformer),
        ('svd', svd),
        ('clf', voting_clf)
    ])

    # Compute and print the accuracy scores
    stats = compute_and_print(X_train, Y_train, pipeline, kf)
    print(stats)

if __name__ == "__main__":
    main()
