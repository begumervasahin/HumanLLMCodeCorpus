def sub_sequence(st1, st2):
    """
    Finds the longest common subsequence in order from two given strings.
    
    Examples:
        * 'ABCABA', 'ABBA' --> 'ABBA'
        * 'aaaaa', 'aa' --> 'aa'
        * 'AGGTAB', 'GXTXAYB' --> 'GTAB'
        * 'ABAZDC', 'BACBAD' --> 'ABAD'
    """
    # Convert strings to lists for easy manipulation
    st1, st2 = list(st1), list(st2)

    # Determine which string is shorter
    if len(st1) <= len(st2):
        shorter, longer = st1, st2
    else:
        shorter, longer = st2, st1

    # Function to find the longest subsequence starting from a given index
    def find_subsequence(start):
        temp_subseq = []
        index = 0
        for char in shorter[start:]:
            if char in longer[index:]:
                temp_subseq.append(char)
                index = longer.index(char, index) + 1
        return temp_subseq

    # Initialize the longest subsequence
    longest_subseq = []

    # Iterate through all possible starting points in the shorter string
    for start in range(len(shorter)):
        current_subseq = find_subsequence(start)

        # Check if the current subsequence is valid
        is_valid = all(
            current_subseq.count(char) <= min(shorter.count(char), longer.count(char))
            for char in current_subseq
        )

        # Update the longest subsequence if the current one is valid and longer
        if is_valid and len(current_subseq) > len(longest_subseq):
            longest_subseq = current_subseq

    # Join the list into a string for the result
    return ''.join(longest_subseq)

# Test cases
print(sub_sequence('ABCABA', 'ABBA'))  # Output: 'ABBA'
print(sub_sequence('aaaaa', 'aa'))     # Output: 'aa'
print(sub_sequence('AGGTAB', 'GXTXAYB'))  # Output: 'GTAB'
print(sub_sequence('ABAZDC', 'BACBAD'))   # Output: 'ABAD'
