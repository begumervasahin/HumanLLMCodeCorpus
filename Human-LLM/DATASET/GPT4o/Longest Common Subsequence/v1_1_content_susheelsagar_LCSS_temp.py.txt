# Open the file "distance_matrix_1698.csv" in read and write mode
with open("distance_matrix_1698.csv", "r+") as f:
    a = []  # Initialize an empty list
    k = 0  # Initialize a counter variable

    # Iterate over the lines in the file
    for line in iter(f):
        # Break the loop if the counter reaches 5
        if k == 5:
            break
        k += 1  # Increment the counter

    # Split the line by comma and assign to list 'a'
    a = line.split(",")

# Print the length of list 'a' plus 1698
print(len(a) + 1698)

# Print the third element in list 'a'
print(a[2])
