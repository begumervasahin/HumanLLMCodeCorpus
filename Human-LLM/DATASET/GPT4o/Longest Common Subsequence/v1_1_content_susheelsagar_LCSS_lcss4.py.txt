import numpy as np
import copy

def similarity_lcss(traj1, traj2):
    score = 0
    min_traj = min(len(traj1), len(traj2))

    days = ["Fri", "Sat", "Sun", "Mon", "Tue", "Wed", "Thu"]
    traj1_days = {day: [] for day in days}
    traj2_days = {day: [] for day in days}

    for row in traj1:
        traj1_days[row[1]].append(row)

    for row in traj2:
        traj2_days[row[1]].append(row)

    for day in days:
        if len(traj1_days[day]) < len(traj2_days[day]):
            score += cal_sim(traj1_days[day], traj2_days[day])
        else:
            score += cal_sim(traj2_days[day], traj1_days[day])

    distance = float(score) / min_traj
    distance = 1 - distance
    return distance

def cal_sim(traj1, traj2):
    score = 0
    while traj2:
        for point in traj1:
            if traj2[-1][2] == point[2] and traj2[-1][0] == point[0]:
                score += 1
        traj2.pop()
    return score

# Initialize the distance matrix
num_users = 8357
distance = np.zeros((num_users, num_users))

# Read the trajectories from the CSV file
file_path = 'v01_anonymized_mobility.csv'
output_file_with_id = 'distance_matrix_1698.csv'
output_file_without_id = 'distance_matrix_1698_withoutid.csv'

traj1 = []
traj2 = []
lst_users = []

with open(file_path, 'r') as f:
    next(f)  # Skip header
    for line in f:
        m = line.strip().split(",")
        user_id = int(m[0])
        tuple = [m[1], m[2], m[3]]
        
        if lst_users and user_id == int(lst_users[-1]):
            traj2.append(tuple)
        else:
            if traj2:
                traj1.append(traj2)
            traj2 = [tuple]
            lst_users.append(m[0])
            
    if traj2:
        traj1.append(traj2)

print(lst_users[1697])

# Calculate the distance matrix
with open(output_file_with_id, 'w') as f1, open(output_file_without_id, 'w') as f2:
    j = 1698
    while j < num_users - 1:
        print(f"j={j}")
        user1 = copy.deepcopy(traj1[j])
        i = j + 1
        while i < num_users:
            user2 = copy.deepcopy(traj1[i])
            distance[j][i] = similarity_lcss(user1, user2)
            distance[i][j] = distance[j][i]
            i += 1
        f1.write(f"{lst_users[j]},{','.join(map(str, distance[j][j:]))}\n")
        f2.write(f"{','.join(map(str, distance[j][j:]))}\n")
        j += 1
