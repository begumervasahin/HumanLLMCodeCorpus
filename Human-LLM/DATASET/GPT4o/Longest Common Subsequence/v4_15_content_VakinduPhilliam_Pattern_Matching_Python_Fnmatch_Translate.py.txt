import fnmatch
import re

# Convert the fnmatch pattern to a regular expression
pattern = '*.txt'
regex = fnmatch.translate(pattern)

# Compile the regular expression
regex_object = re.compile(regex)

# Test the regular expression against a string
filename = 'foobar.txt'
match = regex_object.match(filename)

# Check if there was a match and print the result
if match:
    print(f'The filename "{filename}" matches the pattern "{pattern}".')
else:
    print(f'The filename "{filename}" does not match the pattern "{pattern}".')
