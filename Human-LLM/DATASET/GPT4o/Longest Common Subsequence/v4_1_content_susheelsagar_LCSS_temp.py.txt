# Open the file 'distance_matrix_1698.csv' in read and write mode
with open("distance_matrix_1698.csv", "r+") as f:
    # Initialize an empty list to store lines
    lines = []
    
    # Iterate through the file lines
    for i, line in enumerate(f):
        # Break the loop if the 5th line is reached
        if i == 5:
            break
        lines.append(line.strip())
    
    # Split the 5th line by commas into a list
    if lines:
        last_line = lines[-1]
        elements = last_line.split(",")
        
        # Print the length of the list plus 1698
        print(len(elements) + 1698)
        
        # Print the third element in the list
        if len(elements) > 2:
            print(elements[2])
        else:
            print("The third element does not exist.")
    else:
        print("The file contains less than 5 lines.")
