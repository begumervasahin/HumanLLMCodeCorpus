import difflib

# Define the two strings to be compared
str1 = 'one\ntwo\nthree\n'
str2 = 'ore\ntree\nemu\n'

# Split the strings into lines, keeping the end-of-line characters
lines1 = str1.splitlines(keepends=True)
lines2 = str2.splitlines(keepends=True)

# Create a Differ object to compute the differences
differ = difflib.Differ()
diff = differ.ndiff(lines1, lines2)

# Print the computed differences
print(''.join(diff), end="")
