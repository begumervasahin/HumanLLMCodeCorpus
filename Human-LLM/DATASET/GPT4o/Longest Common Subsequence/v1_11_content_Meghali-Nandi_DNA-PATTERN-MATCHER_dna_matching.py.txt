import random

# Function to generate random DNA sequences and write to files
def randomword(length):
    letters = ["a", "c", "t", "g"]
    word = ''.join(random.choice(letters) for _ in range(length))
    with open("dna-sequence.txt", "w") as file_txt:
        file_txt.write(word)
    with open("dna-sequence2.txt", "w") as file_txt:
        file_txt.write(''.join(random.choice(letters) for _ in range(length)))

# Function to find the Longest Common Subsequence (LCS)
def lcs(X, Y, m, n):
    L = [[0 for _ in range(n+1)] for _ in range(m+1)]

    for i in range(m+1):
        for j in range(n+1):
            if i == 0 or j == 0:
                L[i][j] = 0
            elif X[i-1] == Y[j-1]:
                L[i][j] = L[i-1][j-1] + 1
            else:
                L[i][j] = max(L[i-1][j], L[i][j-1])

    index = L[m][n]
    lcs = [""] * (index+1)
    lcs[index] = ""

    i = m
    j = n
    while i > 0 and j > 0:
        if X[i-1] == Y[j-1]:
            lcs[index-1] = X[i-1]
            i -= 1
            j -= 1
            index -= 1
        elif L[i-1][j] > L[i][j-1]:
            i -= 1
        else:
            j -= 1

    lcs_str = "".join(lcs)
    print("LCS of " + X + " and " + Y + " is " + lcs_str)
    if len(lcs_str) > 500:
        print("Matching")
    else:
        print("Not matching")

# Generate random DNA sequences and write to files
randomword(1000)

# Read the DNA sequences from the files
X = open("dna-sequence.txt", "r").read()
Y = open("dna-sequence2.txt", "r").read()

# Get the lengths of the sequences
m = len(X)
n = len(Y)

# Find and print the LCS
lcs(X, Y, m, n)
