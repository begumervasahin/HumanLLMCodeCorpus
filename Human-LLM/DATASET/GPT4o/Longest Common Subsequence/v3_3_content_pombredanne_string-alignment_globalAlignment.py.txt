def globalAlignmentScore(s1, s2):
    MATCH = 5
    MISMATCH = -4
    GAP = -6
    NUM_ROWS = len(s2) + 1
    NUM_COLS = len(s1) + 1

    # Initialize cost and direction tables
    costs = createTable(NUM_ROWS, NUM_COLS, 0)
    directions = createTable(NUM_ROWS, NUM_COLS, "A")

    # Initialize first row and first column
    for i in range(NUM_COLS):
        costs[0][i] = -i * GAP
        directions[0][i] = "L"

    for j in range(1, NUM_ROWS):
        costs[j][0] = -j * GAP
        directions[j][0] = "T"

    directions[0][0] = 'F'

    # Fill in the rest of the cost and direction tables
    for y in range(1, NUM_ROWS):
        for x in range(1, NUM_COLS):
            valTop = costs[y-1][x] + GAP
            valLeft = costs[y][x-1] + GAP
            if s1[x-1] == s2[y-1]:
                valDiag = costs[y-1][x-1] + MATCH
            else:
                valDiag = costs[y-1][x-1] + MISMATCH

            val = max(valTop, valLeft, valDiag)
            costs[y][x] = val

            if val == valLeft:
                directions[y][x] = "L"
            elif val == valDiag:
                directions[y][x] = "D"
            else:
                directions[y][x] = "T"

    # Align the sequences and write the result to a file
    alignSequences(directions, s1, s2, "alignment.txt")

    return costs[NUM_ROWS-1][NUM_COLS-1]

def createTable(numRows, numCols, value):
    return [[value for _ in range(numCols)] for _ in range(numRows)]

def alignSequences(direction, s1, s2, filename):
    with open(filename, 'w') as file:
        x, y = len(direction[0]) - 1, len(direction) - 1
        topSeq, botSeq = "", ""

        while direction[y][x] != "F":
            if direction[y][x] == "T":
                topSeq = "-" + topSeq
                botSeq = s2[y-1] + botSeq
                y -= 1
            elif direction[y][x] == "L":
                topSeq = s1[x-1] + topSeq
                botSeq = "-" + botSeq
                x -= 1
            elif direction[y][x] == "D":
                topSeq = s1[x-1] + topSeq
                botSeq = s2[y-1] + botSeq
                x -= 1
                y -= 1

        for i in range(0, len(topSeq), 50):
            file.write(topSeq[i:i+50] + '\n')
            file.write(botSeq[i:i+50] + '\n')
            file.write('\n')

if __name__ == "__main__":
    s = "AGCGTCTA"
    t = "TGCATCTCG"
    optimalScore = globalAlignmentScore(s, t)
    print(s)
    print(t)
    print("Global alignment score: " + str(optimalScore))
