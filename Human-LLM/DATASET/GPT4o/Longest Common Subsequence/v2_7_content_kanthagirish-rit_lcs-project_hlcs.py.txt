import numpy as np
import sys

def alg_b(m, n, A, B):
    """
    Hirschberg's algorithm B for computing the LCS length.
    
    :param m: Length of string A
    :param n: Length of string B
    :param A: String sequence A
    :param B: String sequence B
    :return: Final row of the DP table
    """
    K = np.zeros((2, n + 1), dtype='int64')
    for i in range(m):
        K[0, :] = K[1, :]
        for j in range(n):
            if A[i] == B[j]:
                K[1, j + 1] = K[0, j] + 1
            else:
                K[1, j + 1] = max(K[1, j], K[0, j + 1])
    return K[1, :]

def alg_c(m, n, A, B, reconstruct=False):
    """
    Hirschberg's algorithm C for computing and optionally reconstructing the LCS.
    
    :param m: Length of string A
    :param n: Length of string B
    :param A: String sequence A
    :param B: String sequence B
    :param reconstruct: Boolean flag to indicate if the LCS should be reconstructed
    :return: Reconstructed LCS or its length
    """
    if n == 0:
        return ""
    elif m == 1:
        return A if A in B else ""
    else:
        i = m // 2
        L1 = alg_b(i, n, A[:i], B)
        L2 = alg_b(m - i, n, A[i:][::-1], B[::-1])
        if reconstruct:
            k = np.argmax(L1 + L2[::-1])
            return alg_c(i, k, A[:i], B[:k], reconstruct) + alg_c(m - i, n - k, A[i:], B[k:], reconstruct)
        else:
            return int(np.amax(L1 + L2[::-1]))

def test(A, B, reconstruct):
    """
    Test function to compute the LCS.
    
    :param A: String sequence A
    :param B: String sequence B
    :param reconstruct: Boolean flag to indicate if the LCS should be reconstructed
    :return: LCS length or the reconstructed LCS
    """
    return alg_c(len(A), len(B), A, B, reconstruct)

if __name__ == '__main__':
    if len(sys.argv) < 2:
        print("Usage: python3 " + __file__ + " <input_type> <reconstruct_flag>")
        print("<input_type> - 'acgt' for sequences of acgt, 'bits' for binary sequences")
        print("<reconstruct_flag> - '0' to not reconstruct LCS, '1' to reconstruct LCS")
    else:
        file = sys.argv[1] + ".txt"
        with open(file, 'r') as f:
            x = f.readline().strip()
            y = f.readline().strip()
            reconstruct = False
            if len(sys.argv) == 3:
                reconstruct = int(sys.argv[2]) == 1
            if reconstruct:
                lcs = test(x, y, reconstruct)
                print("LCS length: " + str(len(lcs)))
                print("LCS: " + lcs)
            else:
                print("LCS length: " + str(test(x, y, reconstruct)))
