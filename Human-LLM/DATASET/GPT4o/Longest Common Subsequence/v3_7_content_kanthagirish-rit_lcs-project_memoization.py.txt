import sys
import numpy as np

# Initialize the number of recursive calls
num_recursive_calls = 0

def memoization(x, y):
    """
    Computes the length of the longest common subsequence (LCS) between two strings using memoization.

    :param x: First string sequence
    :param y: Second string sequence
    :return: Length of the LCS of x and y
    """
    m, n = len(x), len(y)
    # Create a 2D array initialized to -1 for memoization
    memo_table = np.full((m, n), -1, dtype='int64')
    return _memoized_LCS(x, y, memo_table)

def _memoized_LCS(x, y, memo_table):
    """
    Helper function that uses recursion and memoization to find the LCS length.

    :param x: First string sequence
    :param y: Second string sequence
    :param memo_table: 2D numpy array initialized to -1 for memoization
    :return: Length of the LCS of x and y
    """
    global num_recursive_calls
    num_recursive_calls += 1

    if not x or not y:
        return 0

    i, j = len(x) - 1, len(y) - 1

    if memo_table[i, j] != -1:
        return memo_table[i, j]

    if x[-1] == y[-1]:
        memo_table[i, j] = 1 + _memoized_LCS(x[:-1], y[:-1], memo_table)
    else:
        memo_table[i, j] = max(_memoized_LCS(x, y[:-1], memo_table), _memoized_LCS(x[:-1], y, memo_table))

    return memo_table[i, j]

def test_lcs(x, y):
    """
    Test function to compute the LCS length of two strings.

    :param x: First string sequence
    :param y: Second string sequence
    :return: Length of the LCS of x and y
    """
    return memoization(x, y)

def main():
    if len(sys.argv) != 2:
        print(f"Usage: python3 {sys.argv[0]} <input_file>")
        return
    
    input_file = sys.argv[1]
    with open(input_file, 'r') as file:
        x = file.readline().strip()
        y = file.readline().strip()
        
    lcs_length = test_lcs(x, y)
    print(f"LCS length: {lcs_length}")
    print(f"Number of recursive calls: {num_recursive_calls}")

if __name__ == "__main__":
    main()
