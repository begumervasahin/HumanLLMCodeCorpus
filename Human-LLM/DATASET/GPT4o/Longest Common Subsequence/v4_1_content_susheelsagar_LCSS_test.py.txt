import numpy as np

# Open the mobility data file
with open('/home/s/Dropbox/Thesis/Telenor/fwdtelenordata/v01_anonymized_mobility.csv') as f:
    traj1 = []
    traj2 = []
    lst_users = []
    userid = None
    
    # Skip header
    next(f)
    
    # Read each line in the file
    for line in f:
        m = line.split(",")
        
        # If current user ID matches the last one, append the data to traj2
        if userid == int(m[0]):
            traj2.append([m[1], str(m[2]), m[3]])
        else:
            # When user ID changes, append traj2 to traj1 and reset traj2
            userid = int(m[0])
            if traj2:
                traj1.append(traj2)
            traj2 = [[m[1], str(m[2]), m[3]]]
            lst_users.append(m[0])
    
    # Append the last traj2 to traj1
    if traj2:
        traj1.append(traj2)

print("Loading done")

# Open the distance matrix file
distance_matrix_path = 'distance_matrix_final.csv'
distance = np.zeros((8357, 8357))

# Load the distance matrix
with open(distance_matrix_path) as f1:
    for i, line in enumerate(f1):
        a = line.split(",")
        
        # Check if the user ID matches
        if lst_users[i] != a[0]:
            print('Error at user ID index', i)
        
        for j, value in enumerate(a[1:], start=0):
            distance[i][j] = float(value)
            
            # Check for zero distance for non-diagonal elements
            if distance[i][j] == 0 and i != j:
                print(f"Error at distance {i}, {j}")

# Check if distance matrix is symmetric
for i in range(8357):
    for j in range(8357):
        if distance[i][j] != distance[j][i]:
            print(f"Symmetry error at {i}, {j}")
