import fnmatch
import re

# Function to check if a given string matches a given fnmatch pattern
def matches_pattern(test_string, pattern):
    # Translate the pattern into a regular expression
    regex = fnmatch.translate(pattern)
    print(f"Translated regex: {regex}")
    
    # Compile the translated regular expression
    compiled_regex = re.compile(regex)
    
    # Use the compiled regular expression to check for a match
    return bool(compiled_regex.match(test_string))

# Define the pattern to match .txt files
pattern = '*.txt'

# Define the test string to match against the pattern
test_string = 'foobar.txt'

# Check if the test string matches the pattern
if matches_pattern(test_string, pattern):
    print(f"The string '{test_string}' matches the pattern '{pattern}'")
else:
    print(f"The string '{test_string}' does not match the pattern '{pattern}'")
