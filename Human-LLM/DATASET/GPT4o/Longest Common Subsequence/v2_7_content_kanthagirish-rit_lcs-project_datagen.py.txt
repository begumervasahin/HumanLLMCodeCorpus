import numpy.random as rand
import sys

# Define the character sets
acgt = "acgt"
bits = "01"

def get_random_acgt(length, delta=None):
    """
    Generates random sequences of 'acgt' characters.
    
    Parameters:
    length (int): Length of the sequences to generate.
    delta (int, optional): Number of characters that should differ between x and y. 
                           Should be in the range (0 < delta < length).
    
    Returns:
    tuple: Two generated strings x and y where y has `delta` characters different from x.
    """
    x = "".join([acgt[rand.randint(0, len(acgt))] for _ in range(length)])
    if delta is None:
        return x, x
    else:
        delta_idx = rand.permutation(len(x))[:delta]
        y = "".join([acgt[rand.randint(0, len(acgt))]
                     if i in delta_idx
                     else x[i]
                     for i in range(len(x))])
        return x, y

def get_random_bits(length, delta=None):
    """
    Generates random sequences of '01' bits.
    
    Parameters:
    length (int): Length of the sequences to generate.
    delta (int, optional): Number of characters that should differ between x and y. 
                           Should be in the range (0 < delta < length).
    
    Returns:
    tuple: Two generated strings x and y where y has `delta` characters different from x.
    """
    x = "".join([bits[rand.randint(0, len(bits))] for _ in range(length)])
    if delta is None:
        return x, x
    else:
        delta_idx = rand.permutation(len(x))[:delta]
        y = "".join([bits[rand.randint(0, len(bits))]
                     if i in delta_idx
                     else x[i]
                     for i in range(len(x))])
        return x, y

# Map the function names to their respective functions
functions = {
    "bits": get_random_bits,
    "acgt": get_random_acgt
}

if __name__ == "__main__":
    # Ensure the script is run with appropriate arguments
    if len(sys.argv) < 3:
        print("Usage: python " + __file__ + " acgt/bits N")
        print("acgt - generate sequences of acgt")
        print("bits - generate sequences of binary digits")
        print("N - length of the random strings to generate")
    else:
        # Parse arguments
        type = sys.argv[1]
        size = int(sys.argv[2])
        delta = round(size * 0.3)
        
        # Generate sequences based on the specified type
        if type in functions:
            x, y = functions[type](size, delta)
            with open(type + ".txt", 'w') as f:
                f.write(x + "\n" + y)
        else:
            print("Unknown type, please specify a valid type (acgt/bits)")
