import re

def verify(sequence):
    """
    Verify if the sequence contains only the characters A, T, C, G, and spaces.
    """
    return bool(re.match("^[ATCG ]*$", sequence))

def lcs(X, Y):
    """
    Find the Longest Common Subsequence (LCS) between sequences X and Y.
    """
    m, n = len(X), len(Y)
    L = [[0 for _ in range(n+1)] for _ in range(m+1)]
    
    for i in range(m+1):
        for j in range(n+1):
            if i == 0 or j == 0:
                L[i][j] = 0
            elif X[i-1] == Y[j-1]:
                L[i][j] = L[i-1][j-1] + 1
            else:
                L[i][j] = max(L[i-1][j], L[i][j-1])
    
    index = L[m][n]
    lcs = [""] * (index + 1)
    lcs[index] = "\0"
    i, j = m, n
    
    while i > 0 and j > 0:
        if X[i-1] == Y[j-1]:

            i -= 1
            j -= 1
            index -= 1
        elif L[i-1][j] > L[i][j-1]:
            i -= 1
        else:
            j -= 1
    
    lcs_string = "".join(lcs).strip("\0")
    print(f"\n\nLCS of {X} and {Y} is {lcs_string} \nand length of it is: {len(lcs_string)}")
