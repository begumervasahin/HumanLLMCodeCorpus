def lcs(x, y):
    """
    Calculate the length of the longest common subsequence (LCS) between two strings.

    Args:
    x (str): The first string.
    y (str): The second string.

    Returns:
    int: The length of the LCS.
    """
    m = len(x)
    n = len(y)

    # Initialize a 2D list with zeros to store the LCS lengths
    lcs_table = [[0] * (n + 1) for _ in range(m + 1)]

    # Build the LCS table using dynamic programming
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if x[i - 1] == y[j - 1]:
                lcs_table[i][j] = lcs_table[i - 1][j - 1] + 1
            else:
                lcs_table[i][j] = max(lcs_table[i - 1][j], lcs_table[i][j - 1])

    # The length of the LCS is in the bottom-right cell
    return lcs_table[m][n]

def main():
    """
    Main function to prompt the user for input and display the LCS length.
    """
    x = input("Enter the first string: ")
    y = input("Enter the second string: ")

    # Compute the LCS length
    lcs_length = lcs(x, y)

    # Print the result
    print(f"The length of the LCS is: {lcs_length}")

if __name__ == "__main__":
    main()
