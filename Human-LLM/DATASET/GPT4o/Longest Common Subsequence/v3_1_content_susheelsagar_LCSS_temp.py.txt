# Open the file "distance_matrix_1698.csv" in read and write mode
with open("distance_matrix_1698.csv", "r+") as file:
    # Read the first 5 lines from the file
    lines = [next(file).strip() for _ in range(5)]

# Calculate the length of the list 'values' plus 1698 and print it
length_plus_1698 = len(values) + 1698
print(length_plus_1698)

# Print the third element in the list 'values'
third_value = values[2]
