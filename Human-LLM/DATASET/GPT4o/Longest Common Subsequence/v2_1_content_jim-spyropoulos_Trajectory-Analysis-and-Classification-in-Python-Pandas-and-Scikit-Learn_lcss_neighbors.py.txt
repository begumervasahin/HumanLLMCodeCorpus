import csv
import os
import re
import numpy as np
import pandas as pd
import time
from auxiliaryfunctions import lcss_trigger, plot_traj_red     

# Load the preprocessed dataframe
df = pd.read_pickle('./final_cleaned.df')

# Read the test set data from the CSV file
with open('test_set_a2.csv', 'r') as f:
    reader = csv.reader(f)
    trajectories_read = []

    # Skip the header and process each row
    for i, row in enumerate(reader):
        if i == 0:
            continue  # Skip the header row

        trajectory = []
        # Process each trajectory point
        for j in range(0, len(row), 3):
            point = [
                float(re.sub(r'\[', '', row[j])),  # Clean and convert the first coordinate
                float(row[j + 1]),  # Convert the second coordinate
                float(re.sub(r'\]', '', row[j + 2]))  # Clean and convert the third coordinate
            ]
            trajectory.append(point)
        trajectories_read.append(trajectory)

print("Dataset read successfully.")

# Create a directory to store LCSS results
os.makedirs("LCSSresults", exist_ok=True)

# Process each trajectory
for k, traj in enumerate(trajectories_read, start=1):
    start_time = time.time()

    # Calculate matching points using LCSS algorithm
    matching_points = lcss_trigger(traj, df, 1, 0)
    distancesnp = np.array(matching_points)
    sorted_ind = np.argsort(-distancesnp)  # Sort indices by matching points in descending order

    # Find and plot the top 5 nearest trajectories
    for m, elem in enumerate(sorted_ind[:5], start=1):
        print(f"Nearest trajectory {df['JourneyPatternId'].iloc[elem]} Matching points: {distancesnp[elem]}.")
        
        # Get common points for the matched trajectory
        common_points = lcss_trigger(traj, df, 2, elem)
        
        # Plot and save the matched trajectory
        plot_traj_red(common_points, df['timestamp_longitude_latitude'].iloc[elem],
                      f"./LCSSresults/trajectory{k}_matchingpointsneighb_{m}.png")

    end_time = time.time()
    print(f"Took {(end_time - start_time) / 60:.3f} mins for traj {k} to finish")
