import pandas as pd
import numpy as np
from sklearn.preprocessing import LabelEncoder, StandardScaler
from sklearn.model_selection import train_test_split
from sklearn.manifold import TSNE
import matplotlib.pyplot as plt

def load_and_clean_data(url):
    """
    Loads and cleans the dataset from the provided URL by removing columns with null values
    and discarding the first seven columns.

    Args:
        url (str): The URL to the CSV file.

    Returns:
        pd.DataFrame: A cleaned DataFrame with selected columns.
    """
    # Load the dataset from the given URL
    dataframe_all = pd.read_csv(url)

    # Identify columns that do not have any null values
    columns_without_null = dataframe_all.columns[dataframe_all.isnull().sum() == 0]
    
    # Select only those columns without null values
    dataframe_cleaned = dataframe_all[columns_without_null]

    # Remove the first seven columns as per the requirement
    dataframe_cleaned = dataframe_cleaned.iloc[:, 7:]

    return dataframe_cleaned

def preprocess_data(dataframe):
    """
    Preprocesses the data by standardizing features and encoding labels.
    It then splits the data into training and testing sets.

    Args:
        dataframe (pd.DataFrame): The cleaned DataFrame.

    Returns:
        tuple: Arrays of training and testing features and labels.
    """
    # Extract feature columns and standardize them
    x = dataframe.iloc[:, :-1].values
    scaler = StandardScaler()
    x_scaled = scaler.fit_transform(x)

    # Extract and encode the target labels
    y = dataframe.iloc[:, -1].values
    encoder = LabelEncoder()
    y_encoded = encoder.fit_transform(y)

    # Split the data into training and test sets (90% train, 10% test)
    x_train, x_test, y_train, y_test = train_test_split(x_scaled, y_encoded, test_size=0.1, random_state=0)

    return x_train, x_test, y_train, y_test

def visualize_tsne(x_test, y_test):
    """
    Visualizes the test data using t-SNE by reducing dimensions to two and plotting the data points.

    Args:
        x_test (np.ndarray): Test features.
        y_test (np.ndarray): Test labels.
    """
    # Apply t-SNE to reduce data to 2 dimensions
    tsne = TSNE(n_components=2, random_state=0)
    x_test_2d = tsne.fit_transform(x_test)

    # Define markers and colors for different classes
    markers = ['s', 'd', 'o', '^', 'v']
    colors = ['red', 'blue', 'lightgreen', 'purple', 'cyan']
    unique_classes = np.unique(y_test)

    # Plot each class with a unique marker and color
    plt.figure()
    for idx, class_label in enumerate(unique_classes):
        plt.scatter(x=x_test_2d[y_test == class_label, 0], 
                    y=x_test_2d[y_test == class_label, 1],
                    c=colors[idx], marker=markers[idx], label=class_label)

    plt.xlabel('t-SNE component 1')
    plt.ylabel('t-SNE component 2')
    plt.legend(loc='upper left')
    plt.title('t-SNE visualization of test data')
    plt.show()

def main():
    """
    Main function to load the dataset, preprocess it, and visualize the test data using t-SNE.
    """
    # URL to the dataset
    url = "https://d396qusza40orc.cloudfront.net/predmachlearn/pml-training.csv"

    # Load and clean the data
    dataframe_cleaned = load_and_clean_data(url)

    # Preprocess the data
    x_train, x_test, y_train, y_test = preprocess_data(dataframe_cleaned)

    # Visualize the test data using t-SNE
    visualize_tsne(x_test, y_test)

if __name__ == "__main__":
    main()
