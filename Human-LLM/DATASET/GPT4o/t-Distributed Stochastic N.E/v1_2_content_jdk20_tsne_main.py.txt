import torch
import torch.nn as nn
import matplotlib.pyplot as plt
from torchvision import datasets

def pca(X, k=30):
    """
    Apply Principal Component Analysis (PCA) to reduce the dimensionality of X.
    
    Args:
        X (torch.Tensor): High-dimensional data set in the form [n=samples, N=number of high dimensions].
        k (int): Number of PCA components to retain, with k << N.
    
    Returns:
        torch.Tensor: Reduced dimensionality data set.
    """
    X = X / torch.max(X)  # Normalize X to range [0, 1]
    X = X - torch.mean(X, dim=0)  # Center the data for PCA
    U, S, Vt = torch.svd(X)
    X_reduced = torch.mm(U[:, :k], torch.diag(S[:k]))  # Select top k principal components
    return X_reduced

def gaussian_kernel_variance_search(variance, X, perp, variance_range, tol=0.01):
    """
    Estimate Gaussian kernel variances using binary search to match given perplexity.
    
    Args:
        variance (torch.Tensor): Vector of Gaussian kernel variances for each sample.
        X (torch.Tensor): High-dimensional data set in the form [n=samples, N=number of high dimensions].
        perp (float): Perplexity parameter.
        variance_range (torch.Tensor): Tensor with upper and lower limits for variance search.
        tol (float): Tolerance for matching Shannon entropy with Perplexity.
    """
    n = X.size(0)
    for i in range(n):
        binary_search = True
        while binary_search:
            # Calculate Shannon entropy
            H = torch.exp(-torch.norm(X[i] - X, dim=1) ** 2 / (2 * variance[i]))
            H[i] = 0
            H /= torch.sum(H)
            log_H = torch.log2(H)
            log_H[i] = 0
            H_shannon = torch.pow(2, -torch.sum(H * log_H))
            
            if H_shannon > perp + (tol * perp):
                variance_range[1] = variance[i]
            elif H_shannon < perp - (tol * perp):
                variance_range[0] = variance[i]
            else:
                binary_search = False
            variance[i] = (variance_range[1] + variance_range[0]) / 2

def estimate_p(X, variance):
    """
    Compute pairwise affinities using the Gaussian kernel function and estimated variances.
    
    Args:
        X (torch.Tensor): High-dimensional data set in the form [n=samples, N=number of high dimensions].
        variance (torch.Tensor): Vector of Gaussian kernel variances for each sample.
    
    Returns:
        torch.Tensor: Symmetrical conditional probabilities matrix P.
    """
    P = torch.exp(-torch.norm(X[:, None] - X[None, :], dim=2) ** 2 / (2 * variance))
    torch.diagonal(P).zero_()
    P /= torch.sum(P, dim=1, keepdim=True)
    p = (P + P.T) / (2 * X.size(0))
    torch.diagonal(p).fill_(float('nan'))
    return p

def estimate_q(Y, v):
    """
    Compute low-dimensional affinities using the t-distribution function.
    
    Args:
        Y (torch.Tensor): Low-dimensional data set in the form [n=samples, M=number of low dimensions].
        v (int): Degrees of freedom for the t-distribution.
    
    Returns:
        torch.Tensor: Joint probability matrix Q.
    """
    q = torch.pow(1 + torch.norm(Y[:, None] - Y[None, :], dim=2) ** 2, -(v + 1) / 2)
    torch.diagonal(q).zero_()
    q /= torch.sum(q)
    torch.diagonal(q).fill_(float('nan'))
    return q

def tsne(X, perp=40, T=1000, lr=100, a=(0.5, 0.8), k=30, M=2, v=1, verbose=True):
    """
    Perform t-Distributed Stochastic Neighbor Embedding (t-SNE) on high-dimensional data.
    
    Args:
        X (torch.Tensor): High-dimensional data set in the form [n=samples, N=number of high dimensions].
        perp (int): Perplexity parameter.
        T (int): Number of gradient descent iterations.
        lr (int): Learning rate.
        a (tuple): Momentum terms for early and late iterations.
        k (int): Number of PCA components to reduce X.
        M (int): Number of dimensions for the low-dimensional output.
        v (int): Degrees of freedom for the t-distribution.
        verbose (bool): Whether to print progress information.
    
    Returns:
        torch.Tensor: Low-dimensional embedding of X.
    """
    n = X.size(0)
    X_reduced = pca(X, k)
    Y = 1e-4 * torch.randn(n, M)

    variance_range = torch.tensor([0.0, 50.0])
    variance = (variance_range[1] - variance_range[0]) / 2 * torch.ones(n, 1)

    if torch.cuda.is_available():
        X_reduced, Y, variance = X_reduced.cuda(), Y.cuda(), variance.cuda()
    Y.requires_grad = True

    loss_function = nn.KLDivLoss(reduction='sum')
    optimizer = torch.optim.SGD([Y], lr=lr, momentum=a[0])

    gaussian_kernel_variance_search(variance, X_reduced, perp, variance_range)
    
    if verbose:
        print(f'Gaussian kernel variance (mean): {torch.mean(variance).item()}')
        print(f'Gaussian kernel variance (min): {torch.min(variance).item()}')
        print(f'Gaussian kernel variance (max): {torch.max(variance).item()}')

    p = estimate_p(X_reduced, variance)

    for t in range(T + 1):
        if t == 250:
            optimizer = torch.optim.SGD([Y], lr=lr, momentum=a[1])

        optimizer.zero_grad()
        q = estimate_q(Y, v)

        if t <= 50:
            loss = loss_function(torch.log(q), 4 * p)
        else:
            loss = loss_function(torch.log(q), p)

        loss.backward()
        optimizer.step()

        if verbose and t % 20 == 0:
            print(f'Iteration: {t} | KL Loss: {loss.item()} | Gradient Norm: {torch.norm(Y.grad).item()}')

    return Y

def main():
    torch.manual_seed(1)

    perp = 40
    T = 1000
    lr = 100
    a = [0.5, 0.8]
    k = 30
    n = 6000
    M = 2
    v = 1
    verbose = True

    mnist = datasets.MNIST('../data', train=False, download=True)
    X = mnist.data.view(mnist.data.shape[0], -1).float()
    index = torch.randint(mnist.data.shape[0], (n,))
    X = X[index]
    targets = mnist.targets[index]

    Y = tsne(X, perp, T, lr, a, k, M, v, verbose)

    plt.figure()
    for i in range(10):
        plt.scatter(Y[targets == i, 0].cpu().detach().numpy(), Y[targets == i, 1].cpu().detach().numpy(), label=str(i))
    plt.legend()
    plt.show()

if __name__ == "__main__":
    main()
