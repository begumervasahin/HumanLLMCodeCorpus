import time
import data
import tsne
import bh_tsne
import matplotlib.pyplot as plt

def load_mnist_data(image_file, label_file, n_samples=1000):
    """
    Load and preprocess the MNIST dataset.

    Args:
        image_file (str): Path to the MNIST images file in IDX format.
        label_file (str): Path to the MNIST labels file in IDX format.
        n_samples (int): Number of samples to load and process.

    Returns:
        tuple: A tuple containing:
            - np.ndarray: Preprocessed image data.
            - np.ndarray: Corresponding labels.
    """
    images, labels = data.read_MNIST(image_file, label_file)
    images = images[:n_samples] / 255.0  # Normalize the images
    labels = labels[:n_samples]
    return images, labels

def run_tsne(images, labels, use_bh_tsne=False, max_iter=1000, animate=False):
    """
    Run t-SNE or Barnes-Hut t-SNE on the dataset.

    Args:
        images (np.ndarray): The preprocessed image data.
        labels (np.ndarray): The corresponding labels.
        use_bh_tsne (bool): If True, use Barnes-Hut t-SNE; otherwise, use standard t-SNE.
        max_iter (int): Maximum number of iterations for t-SNE.
        animate (bool): If True, generate an animation of the t-SNE process.

    Returns:
        object: The fitted t-SNE model.
    """
    if use_bh_tsne:
        vis = bh_tsne.BH_TSNE(max_iter=max_iter, bh_threshold=0.5)
    else:
        vis = tsne.TSNE(max_iter=max_iter)

    vis.fit(images, animate=animate, labels=labels, anim_file="tsne_movie.mp4" if animate else None)
    return vis

def plot_embedding(vis, labels):
    """
    Plot the 2D embedding produced by t-SNE.

    Args:
        vis (object): The fitted t-SNE model.
        labels (np.ndarray): The corresponding labels.
    """
    fig, ax = plt.subplots()
    vis.plot_embedding2D(labels, ax)
    plt.show()

def main():
    start_time = time.time()

    # Load and preprocess MNIST data
    print("Loading data...")
    filename_train_images = "MNIST/train-images.idx3-ubyte"
    filename_train_labels = "MNIST/train-labels.idx1-ubyte"
    images_train, labels_train = load_mnist_data(filename_train_images, filename_train_labels)

    # Apply PCA to reduce dimensionality before t-SNE
    images_train = tsne.get_pca_proj(images_train, n_components=30)

    print(f"Data loaded and preprocessed. Time elapsed: {time.time() - start_time:.2f} seconds")

    # Set up for t-SNE
    animate = False
    use_bh_tsne = False  # Set to True to use Barnes-Hut t-SNE

    if animate:
        import matplotlib
        matplotlib.use("Agg")

    # Run t-SNE or Barnes-Hut t-SNE
    vis = run_tsne(images_train, labels_train, use_bh_tsne=use_bh_tsne, max_iter=1000, animate=animate)

    print(f"t-SNE completed. Total time: {time.time() - start_time:.2f} seconds")

    # Plot the results if not animating
    if not animate:
        plot_embedding(vis, labels_train)

if __name__ == "__main__":
    main()
