import pandas as pd
import numpy as np
from sklearn.decomposition import PCA
from sklearn.manifold import TSNE
from sklearn.preprocessing import scale
import matplotlib.pyplot as plt

def load_and_preprocess_data(file_path):
    """
    Load the dataset, remove missing values, and scale the data.

    Args:
        file_path (str): Path to the CSV file containing the dataset.

    Returns:
        np.ndarray: Scaled data ready for t-SNE.
    """
    # Load dataset
    data = pd.read_csv(file_path)
    
    # Remove rows with missing values
    data_cleaned = data.dropna()
    
    # Drop the first column (assuming it's non-numeric or an identifier)
    data_features = data_cleaned.iloc[:, 1:]
    
    # Transpose and scale the data (rows are samples for scaling)
    scaled_data = scale(data_features.T)
    
    return scaled_data

def perform_tsne(scaled_data, n_components=2, n_iter=2500, random_state=0):
    """
    Perform t-SNE on the scaled data to reduce dimensionality.

    Args:
        scaled_data (np.ndarray): The scaled data matrix.
        n_components (int): Number of dimensions for the t-SNE output.
        n_iter (int): Number of iterations for t-SNE.
        random_state (int): Random seed for reproducibility.

    Returns:
        np.ndarray: 2D t-SNE transformed data.
    """
    tsne = TSNE(n_components=n_components, n_iter=n_iter, random_state=random_state)
    tsne_result = tsne.fit_transform(scaled_data)
    return tsne_result

def plot_tsne(tsne_data):
    """
    Plot the 2D t-SNE result.

    Args:
        tsne_data (np.ndarray): The t-SNE transformed data.
    """
    plt.figure(figsize=(8, 6))
    plt.scatter(tsne_data[:, 0], tsne_data[:, 1], alpha=0.7, edgecolors='k')
    plt.xlabel('t-SNE Component 1')
    plt.ylabel('t-SNE Component 2')
    plt.title('2D t-SNE Visualization')
    plt.grid(True)
    plt.show()

def main():
    file_path = 'endometrial.csv'  # Update with the correct path to your CSV file
    
    # Load and preprocess data
    scaled_data = load_and_preprocess_data(file_path)
    
    # Perform t-SNE
    tsne_result = perform_tsne(scaled_data)
    
    # Plot the t-SNE result
    plot_tsne(tsne_result)

if __name__ == "__main__":
    main()
