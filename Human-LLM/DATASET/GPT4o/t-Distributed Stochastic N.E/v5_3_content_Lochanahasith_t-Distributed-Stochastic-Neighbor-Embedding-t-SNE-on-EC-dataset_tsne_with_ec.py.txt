import pandas as pd
import numpy as np
from sklearn.manifold import TSNE
from sklearn.preprocessing import scale
import matplotlib.pyplot as plt

def load_and_preprocess_data(file_path):
    """
    Load and preprocess the dataset: remove missing values and scale features.

    Args:
        file_path (str): Path to the CSV file containing the dataset.

    Returns:
        np.ndarray: Scaled features ready for t-SNE.
    """
    # Load dataset from CSV
    data = pd.read_csv(file_path)
    
    # Drop rows with missing values
    cleaned_data = data.dropna()
    
    # Extract features (excluding the first column)
    features = cleaned_data.iloc[:, 1:]
    
    # Transpose features to ensure samples are rows and scale the data
    scaled_features = scale(features.T)
    
    return scaled_features

def perform_tsne(data, n_components=2, n_iter=2500, random_state=0):
    """
    Apply t-SNE to the scaled data to reduce its dimensionality.

    Args:
        data (np.ndarray): Scaled data matrix.
        n_components (int): Number of dimensions for the t-SNE output.
        n_iter (int): Number of iterations for t-SNE.
        random_state (int): Random seed for reproducibility.

    Returns:
        np.ndarray: 2D t-SNE transformed data.
    """
    # Initialize and fit t-SNE
    tsne = TSNE(n_components=n_components, n_iter=n_iter, random_state=random_state)
    tsne_result = tsne.fit_transform(data)
    
    return tsne_result

def plot_tsne(tsne_data):
    """
    Plot the 2D t-SNE result.

    Args:
        tsne_data (np.ndarray): The t-SNE transformed data.
    """
    plt.figure(figsize=(10, 8))
    plt.scatter(tsne_data[:, 0], tsne_data[:, 1], alpha=0.6, edgecolor='k')
    plt.xlabel('t-SNE Component 1')
    plt.ylabel('t-SNE Component 2')
    plt.title('2D t-SNE Visualization')
    plt.grid(True)
    plt.show()

def main():
    """
    Main function to load data, apply t-SNE, and plot the results.
    """
    # Path to the dataset
    file_path = 'endometrial.csv'
    
    # Load and preprocess the data
    scaled_data = load_and_preprocess_data(file_path)
    
    # Perform t-SNE
    tsne_result = perform_tsne(scaled_data)
    
    # Plot the t-SNE results
    plot_tsne(tsne_result)

if __name__ == "__main__":
    main()
