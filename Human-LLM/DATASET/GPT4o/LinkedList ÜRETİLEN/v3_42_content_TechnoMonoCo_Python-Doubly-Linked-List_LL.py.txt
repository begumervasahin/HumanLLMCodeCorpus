class Node:
    def __init__(self, data=None, next_node=None, prev_node=None):
        """
        Initialize a new node with the given data and optional next and previous nodes.

        Args:
            data: The data to store in the node. Default is None.
            next_node: The next node in the linked list. Default is None.
            prev_node: The previous node in the linked list. Default is None.
        """
        self.data = data
        self.next = next_node
        self.prev = prev_node

class LinkedList:
    def __init__(self, data=None):
        """
        Initialize an empty linked list or a list with a single node.

        Args:
            data: The data for the initial node. Default is None.
        """
        self.head = Node(data) if data is not None else None

    def remove(self, data):
        """
        Remove the first node with the specified data.

        Args:
            data: The data to remove from the list.

        Returns:
            int: 0 if the node was removed, -1 if the node was not found.
        """
        current = self.head
        while current is not None:
            if current.data == data:
                if current.prev is not None:
                    current.prev.next = current.next
                if current.next is not None:
                    current.next.prev = current.prev
                if current == self.head:
                    self.head = current.next
                del current
                return 0
            current = current.next
        return -1

    def insert(self, data):
        """
        Insert a new node with the specified data in a sorted manner.

        Args:
            data: The data to insert into the list.
        """
        new_node = Node(data)
        if self.head is None or data < self.head.data:
            new_node.next = self.head
            if self.head is not None:
                self.head.prev = new_node
            self.head = new_node
        else:
            current = self.head
            while current.next is not None and current.next.data < data:
                current = current.next
            new_node.next = current.next
            if current.next is not None:
                current.next.prev = new_node
            current.next = new_node
            new_node.prev = current

    def display(self):
        """
        Print all elements in the linked list.
        """
        if self.head is None:
            print("Empty.")
        else:
            current = self.head
            elements = []
            while current is not None:
                elements.append(current.data)
                current = current.next
            print(", ".join(map(str, elements)))

def test_linked_list():
    """
    Test the LinkedList class with various operations.
    """
    ll = LinkedList(0)
    print("Initial list:")
    ll.display()
    
    print("\nInserting 1...")

    print("\nInserting 5...")
