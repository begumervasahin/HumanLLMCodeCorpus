import urllib.request
from bs4 import BeautifulSoup
import ssl

# Ignore SSL certificate errors
ssl_context = ssl.create_default_context()
ssl_context.check_hostname = False
ssl_context.verify_mode = ssl.CERT_NONE

def get_html(url, context):
    """
    Retrieve HTML content from the given URL.

    Args:
        url (str): The URL to retrieve HTML content from.
        context (ssl.SSLContext): The SSL context to use for the request.

    Returns:
        str: The HTML content of the page.
    """
    with urllib.request.urlopen(url, context=context) as response:
        return response.read()

def find_next_url(soup, position):
    """
    Find the URL at the specified position in the list of anchor tags.

    Args:
        soup (BeautifulSoup): The BeautifulSoup object containing the parsed HTML.
        position (int): The position of the anchor tag to retrieve the URL from.

    Returns:
        str or None: The URL at the specified position, or None if not found.
    """
    tags = soup.find_all('a')
    if len(tags) >= position:
        return tags[position - 1].get('href')
    else:
        print(f"No link found at position {position}")
        return None

def main():
    """
    Main function to retrieve URLs from a web page by following links.
    """
    url = input('Enter URL: ')
    count = int(input('Enter count: '))
    position = int(input('Enter position: '))

    for i in range(count + 1):
        print('Retrieving:', url)
        html = get_html(url, ssl_context)
        soup = BeautifulSoup(html, 'html.parser')
        next_url = find_next_url(soup, position)
        if next_url:
            url = next_url
        else:
            break

if __name__ == "__main__":
    main()
