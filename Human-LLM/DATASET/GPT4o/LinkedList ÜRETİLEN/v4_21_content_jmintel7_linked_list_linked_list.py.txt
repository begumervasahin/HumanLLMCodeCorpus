class Node:
    def __init__(self, data=None):
        self.data = data
        self.next = None

class LinkedList:
    def __init__(self):
        self.head = Node()

    def append(self, data):
        """Append a new node with the given data to the end of the list."""
        new_node = Node(data)
        cur_node = self.head
        while cur_node.next is not None:
            cur_node = cur_node.next
        cur_node.next = new_node

    def length(self):
        """Return the length of the linked list."""
        cur_node = self.head
        total = 0
        while cur_node.next is not None:
            cur_node = cur_node.next
            total += 1
        return total

    def display_images(self):
        """Return a list of images from the linked list."""
        cur_node = self.head
        images = []
        while cur_node.next is not None:
            cur_node = cur_node.next
            if cur_node.data:
                images.append(cur_node.data[0])
        return images

    def get_rollno(self, rollno):
        """Return the data of the node with the given roll number."""
        cur_node = self.head
        while cur_node.next is not None:
            cur_node = cur_node.next
            data = cur_node.data
            if data and data.rollno == rollno:
                return data
        print('Student with the roll number does not exist')
        return None

    def erase(self, rollno):
        """Erase the node with the given roll number."""
        cur_node = self.head
        while cur_node.next is not None:
            last_node = cur_node
            cur_node = cur_node.next
            data = cur_node.data
            if data and data.rollno == rollno:
                last_node.next = cur_node.next
                print('Record erased')
                return
        print('Student with the roll number does not exist')

    def get_image(self, index):
        """Return the data of the node at the given index."""
        if index >= self.length():
            print('ERROR: Index out of range')
            return None
        cur_node = self.head
        for _ in range(index + 1):
            cur_node = cur_node.next
        return cur_node.data

    def erase_image(self, index):
        """Erase the node at the given index."""
        if index >= self.length():
            print('ERROR: Index out of range')
            return None
        cur_node = self.head
        for _ in range(index + 1):
            last_node = cur_node
            cur_node = cur_node.next
        last_node.next = cur_node.next
