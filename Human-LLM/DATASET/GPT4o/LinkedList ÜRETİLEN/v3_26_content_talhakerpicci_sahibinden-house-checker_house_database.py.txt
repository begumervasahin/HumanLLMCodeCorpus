import sqlite3

class House:
    """A class representing a house."""

    def __init__(self, house_link, title, price, m2, date, neighborhood, room, img):
        self.house_link = house_link
        self.title = title
        self.price = price
        self.m2 = m2
        self.date = date
        self.neighborhood = neighborhood
        self.room = room
        self.img = img

class DatabasePost:
    """A class to handle database operations for house posts."""

    def __init__(self, db_name="Sahibinden.db"):
        self.db_name = db_name
        self.connection = None
        self.cursor = None
        self.connect_database()

    def connect_database(self):
        """Connect to the SQLite database and create tables if they do not exist."""
        self.connection = sqlite3.connect(self.db_name)
        self.cursor = self.connection.cursor()
        self.create_tables()

    def create_tables(self):
        """Create the necessary tables if they do not exist."""
        query = """
        CREATE TABLE IF NOT EXISTS Tbl_Posts (
            Link TEXT PRIMARY KEY,
            Title TEXT,
            Price TEXT,
            M2 TEXT,
            Date TEXT,
            Neighborhood TEXT,
            Room TEXT,
            Img TEXT
        );
        CREATE TABLE IF NOT EXISTS Tbl_Posts_Backup (
            Link TEXT PRIMARY KEY,
            Title TEXT,
            Price TEXT,
            M2 TEXT,
            Date TEXT,
            Neighborhood TEXT,
            Room TEXT,
            Img TEXT
        );
        """
        self.cursor.executescript(query)
        self.connection.commit()

    def check_if_house_exists(self, link):
        """Check if a house with the given link exists in the Tbl_Posts table."""
        query = "SELECT 1 FROM Tbl_Posts WHERE Link = ?"
        self.cursor.execute(query, (link,))
        return self.cursor.fetchone() is not None

    def add_house(self, house, backup=False):
        """Add a house to the database. Use backup table if specified."""
        table = "Tbl_Posts_Backup" if backup else "Tbl_Posts"
        query = f"""
        INSERT OR IGNORE INTO {table} 
        (Link, Title, Price, M2, Date, Neighborhood, Room, Img) 
        VALUES (?, ?, ?, ?, ?, ?, ?, ?)
        """
        self.cursor.execute(query, (house.house_link, house.title, house.price, house.m2, house.date, house.neighborhood, house.room, house.img))
        self.connection.commit()

    def clear_posts(self, day):
        """Delete all posts from the Tbl_Posts table with the given date."""
        query = "DELETE FROM Tbl_Posts WHERE Date LIKE ?"
        self.cursor.execute(query, (f"%{day}%",))
        self.connection.commit()

    def close(self):
        """Close the database connection."""
        if self.connection:
            self.connection.close()

if __name__ == "__main__":
    db = DatabasePost()
    
    # Creating a new house instance
    house = House(
        house_link="http://example.com/house1",
        title="Beautiful House",
        price="500000",
        m2="120",
        date="2024-01-01",
        neighborhood="Downtown",
        room="3+1",
        img="http://example.com/image1.jpg"
    )
    
    # Checking if house exists
    exists = db.check_if_house_exists(house.house_link)
    print(f"House exists: {exists}")
    
    # Adding house to database
    if not exists:
        db.add_house(house)
        print("House added to database")
    
    # Clearing posts from a specific day
    db.clear_posts("2024-01-01")
    print("Posts cleared for the date 2024-01-01")

    db.close()
