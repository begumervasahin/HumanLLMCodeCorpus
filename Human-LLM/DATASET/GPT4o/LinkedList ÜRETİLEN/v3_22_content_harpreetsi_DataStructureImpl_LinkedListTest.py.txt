import unittest
from LinkedList import LinkedList

class LinkedListTest(unittest.TestCase):

    def setUp(self):
        """
        Set up the linked list for testing.
        """
        self.mylist = LinkedList()

    def test_add_at_the_beginning_empty_list(self):
        """
        Test adding a node at the beginning of an empty list.
        """
        self.mylist.add_at_the_beginning(10)
        self.assertEqual(self.mylist.add_at_the_beginning(20).data, 20)

    def test_add_at_the_beginning_non_empty_list(self):
        """
        Test adding a node at the beginning of a non-empty list.
        """
        self.mylist.add_at_the_beginning(10)
        self.mylist.add_at_the_beginning(20)
        self.assertEqual(self.mylist.add_at_the_beginning(30).data, 30)

    def test_add_at_the_end_empty_list(self):
        """
        Test adding a node at the end of an empty list.
        """
        self.assertEqual(self.mylist.add_at_the_end(500).data, 500)

    def test_add_at_the_end_non_empty_list(self):
        """
        Test adding a node at the end of a non-empty list.
        """
        self.mylist.add_at_the_end(10)
        self.mylist.add_at_the_end(20)
        self.assertEqual(self.mylist.head.next.data, 20)

    def test_add_at_nth_position_1st_position_non_empty_list(self):
        """
        Test adding a node at the 1st position in a non-empty list.
        """
        self.mylist.add_at_the_end(10)
        self.mylist.add_at_the_end(20)
        self.mylist.add_at_the_end(30)
        self.assertEqual(self.mylist.add_at_nth_position(5, 1).data, 5)

    def test_add_at_nth_position_0th_position_invalid(self):
        """
        Test adding a node at the 0th position, which is invalid.
        """
        self.assertEqual(self.mylist.add_at_nth_position(5, 0), "Invalid position, it cannot be less than or equal to 0")

    def test_add_at_nth_position_valid(self):
        """
        Test adding a node at a valid nth position in the list.
        """
        self.mylist.add_at_the_end(10)
        self.mylist.add_at_the_end(20)
        self.mylist.add_at_the_end(30)
        self.mylist.add_at_the_end(40)
        self.assertEqual(self.mylist.add_at_nth_position(25, 3).next.next.data, 25)

    def test_add_at_nth_position_invalid_position(self):
        """
        Test adding a node at an invalid position that exceeds the list size.
        """
        self.mylist.add_at_the_end(10)
        self.mylist.add_at_the_end(20)
        self.mylist.add_at_the_end(30)
        self.mylist.add_at_the_end(40)
        self.assertEqual(self.mylist.add_at_nth_position(50, 6), "Invalid position, it exceeds the size of the linked list")

    def test_remove_at_the_beginning_empty_list(self):
        """
        Test removing a node at the beginning of an empty list.
        """
        self.assertEqual(self.mylist.remove_at_the_beginning(), "List is empty, nothing to remove")

    def test_remove_at_the_beginning_non_empty_list(self):
        """
        Test removing a node at the beginning of a non-empty list.
        """
        self.mylist.add_at_the_end(10)
        self.mylist.add_at_the_end(20)
        self.mylist.add_at_the_end(30)
        self.assertEqual(self.mylist.remove_at_the_beginning().data, 20)

    def test_remove_at_the_end_empty_list(self):
        """
        Test removing a node at the end of an empty list.
        """
        self.assertEqual(self.mylist.remove_at_the_end(), "List is empty, nothing to remove")

    def test_remove_at_the_end_list_has_one_element(self):
        """
        Test removing a node at the end of a list that has one element.
        """
        self.mylist.add_at_the_end(10)
        self.assertEqual(self.mylist.remove_at_the_end(), None)

    def test_remove_at_the_end_list_has_more_than_one_element(self):
        """
        Test removing a node at the end of a list that has more than one element.
        """
        self.mylist.add_at_the_end(10)
        self.mylist.add_at_the_end(20)
        self.mylist.add_at_the_end(30)
        self.assertEqual(self.mylist.remove_at_the_end().next.data, 20)

    def test_remove_at_nth_position_0th_position_invalid(self):
        """
        Test removing a node at the 0th position, which is invalid.
        """
        self.assertEqual(self.mylist.remove_at_nth_position(0), "Invalid position, it cannot be less than or equal to 0")

    def test_remove_at_nth_position_1st_position(self):
        """
        Test removing a node at the 1st position in the list.
        """
        self.mylist.add_at_the_end(10)
        self.mylist.add_at_the_end(20)
        self.assertEqual(self.mylist.remove_at_nth_position(1).data, 20)

    def test_remove_at_nth_position_valid(self):
        """
        Test removing a node at a valid nth position in the list.
        """
        self.mylist.add_at_the_end(10)
        self.mylist.add_at_the_end(20)
        self.mylist.add_at_the_end(30)
        self.mylist.add_at_the_end(40)
        self.assertEqual(self.mylist.remove_at_nth_position(3).next.next.data, 40)

    def test_remove_at_nth_position_invalid_position(self):
        """
        Test removing a node at an invalid position that exceeds the list size.
        """
        self.mylist.add_at_the_end(10)
        self.mylist.add_at_the_end(20)
        self.mylist.add_at_the_end(30)
        self.assertEqual(self.mylist.remove_at_nth_position(4), "Invalid position, it exceeds the size of the linked list")

    def test_reverse_empty_list(self):
        """
        Test reversing an empty list.
        """
        self.assertEqual(self.mylist.reverse(), "List is empty, nothing to reverse")

    def test_reverse_non_empty_list(self):
        """
        Test reversing a non-empty list.
        """
        self.mylist.add_at_the_end(10)
        self.mylist.add_at_the_end(20)
        self.mylist.add_at_the_end(30)
        self.mylist.add_at_the_end(40)
        self.assertEqual(self.mylist.reverse().data, 40)

    def test_reverse_one_element_list(self):
        """
        Test reversing a list that has one element.
        """
        self.mylist.add_at_the_end(10)
        self.assertEqual(self.mylist.reverse().data, 10)

    def test_reverse_using_recursion_empty_list(self):
        """
        Test reversing an empty list using recursion.
        """
        self.assertEqual(self.mylist.reverse_using_recursion(self.mylist.head), "List is empty, nothing to reverse")

    def test_reverse_using_recursion_one_element_list(self):
        """
        Test reversing a list that has one element using recursion.
        """
        self.mylist.add_at_the_end(10)
        self.assertEqual(self.mylist.reverse_using_recursion(self.mylist.head).data, 10)

    def test_reverse_using_recursion_non_empty_list(self):
        """
        Test reversing a non-empty list using recursion.
        """
        self.mylist.add_at_the_end(10)
        self.mylist.add_at_the_end(20)
        self.mylist.add_at_the_end(30)
        self.mylist.add_at_the_end(40)
        self.mylist.add_at_the_end(50)
        self.mylist.add_at_the_end(60)
        self.assertEqual(self.mylist.reverse_using_recursion(self.mylist.head).data, 60)

if __name__ == '__main__':
    unittest.main()
