class Node:
    def __init__(self, data):
        self.data = data
        self.next_node = None

class LinkedList:
    def __init__(self):
        self.head = None

    def append(self, data):
        new_node = Node(data)
        if not self.head:
            self.head = new_node
            return
        last = self.head
        while last.next_node:
            last = last.next_node
        last.next_node = new_node

    def display(self):
        current = self.head
        while current:
            print(current.data, end=" -> " if current.next_node else "\n")
            current = current.next_node

def locate_middle_node(ll):
    """
    Locate the middle node of the linked list and split the list into two halves.
    
    Args:
        ll (LinkedList): The linked list to process.
        
    Returns:
        Node: The middle node of the linked list.
    """
    slow_ptr = ll.head
    fast_ptr = ll.head
    prev_ptr = None
    
    while fast_ptr and fast_ptr.next_node:
        prev_ptr = slow_ptr
        slow_ptr = slow_ptr.next_node
        fast_ptr = fast_ptr.next_node.next_node
    
    # Disconnect the first half from the second half
    if prev_ptr:
        prev_ptr.next_node = None
        
    return slow_ptr

def reverse_list(head):
    """
    Reverse the linked list starting from the given node.
    
    Args:
        head (Node): The starting node of the list to reverse.
        
    Returns:
        Node: The new head of the reversed list.
    """
    prev = None
    current = head
    while current:
        next_node = current.next_node
        current.next_node = prev
        prev = current
        current = next_node
    return prev

def reorder_list(ll, second_half_head):
    """
    Reorder the linked list by merging the first half and the reversed second half.
    
    Args:
        ll (LinkedList): The original linked list.
        second_half_head (Node): The head of the reversed second half of the list.
    """
    first_half_ptr = ll.head
    second_half_ptr = second_half_head
    
    while first_half_ptr and second_half_ptr:
        # Save the next nodes
        first_half_next = first_half_ptr.next_node
        second_half_next = second_half_ptr.next_node
        
        # Reorder nodes
        first_half_ptr.next_node = second_half_ptr
        if first_half_next:
            second_half_ptr.next_node = first_half_next
        
        # Move pointers forward
        first_half_ptr = first_half_next
        second_half_ptr = second_half_next

# Example usage
if __name__ == "__main__":
    # Create and populate the linked list
    ll = LinkedList()
    for i in range(1, 10):
        ll.append(i)

    print("Original List:")
    ll.display()

    # Locate the middle node
    middle_node = locate_middle_node(ll)
    print("Middle Node:", middle_node.data)

    # Reverse the second half of the list
    second_half_head = reverse_list(middle_node)

    print("Second Half Reversed:")
    temp_list = LinkedList()
    temp_list.head = second_half_head
    temp_list.display()

    # Reorder the original list
    reorder_list(ll, second_half_head)

    print("Reordered List:")
    ll.display()
