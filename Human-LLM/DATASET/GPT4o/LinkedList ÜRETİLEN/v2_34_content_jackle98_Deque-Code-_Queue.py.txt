from collections import deque

def get_deque(size):
    """
    Return a deque (double-ended queue) with a maximum size.

    Args:
        size (int): The maximum size of the deque.

    Returns:
        deque: A deque object with the specified maximum size.
    """
    return deque(maxlen=size)

class Queue:
    def __init__(self):
        """Initialize the queue with a deque of size 1."""
        self._dq = get_deque(1)

    def __str__(self):
        """Return a string representation of the queue."""
        return str(self._dq)

    def __len__(self):
        """Return the number of elements in the queue."""
        return len(self._dq)

    def enqueue(self, val):
        """
        Add an element to the end of the queue.

        Args:
            val: The value to be added to the queue.
        """
        self._dq.append(val)

    def dequeue(self):
        """
        Remove and return the element from the front of the queue.

        Returns:
            The value removed from the front of the queue, or None if the queue is empty.
        """
        if len(self._dq) == 0:
            return None
        return self._dq.popleft()

# Example usage
if __name__ == "__main__":
    q = Queue()
    
    # Enqueue elements
    q.enqueue(10)
    q.enqueue(20)
    q.enqueue(30)
    print("Queue after enqueuing 10, 20, 30:")
    print(q)

    # Dequeue elements and display the queue after each dequeue
    print("Dequeue element:", q.dequeue())
    print("Queue after dequeue:")
    print(q)

    print("Dequeue element:", q.dequeue())
    print("Queue after dequeue:")
    print(q)

    print("Dequeue element:", q.dequeue())
    print("Queue after dequeue:")
    print(q)

    # Attempt to dequeue from an empty queue
    print("Dequeue element (should return None):", q.dequeue())
