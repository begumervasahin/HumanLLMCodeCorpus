class Node:
    def __init__(self, value):
        """
        Initialize a node with a given value.
        
        Args:
            value: The value to be stored in the node.
        """
        self.value = value
        self.next = None

if __name__ == "__main__":
    # Create the linked list with three nodes
    head = Node(0)          # First node with value 0
    head.next = Node(1)     # Second node with value 1
    head.next.next = Node(2) # Third node with value 2

    # Print the values of the linked list nodes
    current_node = head
    while current_node is not None:
        print(f'value = {current_node.value}')
        current_node = current_node.next
