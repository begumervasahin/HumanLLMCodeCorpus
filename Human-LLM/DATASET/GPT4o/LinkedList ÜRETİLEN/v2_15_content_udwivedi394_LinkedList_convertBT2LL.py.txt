def in_order_traverse(root):
    """
    Perform in-order traversal of the binary tree and print node data.
    
    Args:
        root (Node): The root of the binary tree.
    """
    current = root
    stack = []
    while True:
        while current:
            stack.append(current)
            current = current.left
        if not stack:
            break
        current = stack.pop()
        print(current.data, end=' ')
        current = current.right


def convert_bt_to_dll(root):
    """
    Convert a binary tree to a circular doubly linked list.
    
    Args:
        root (Node): The root of the binary tree.
    
    Returns:
        Node: The head of the doubly linked list.
    """
    current = root
    stack = []
    prev_node = None
    head = None
    tail = None
    while True:
        while current:
            stack.append(current)
            current = current.left
        if not stack:
            break
        current = stack.pop()
        current.left = prev_node
        if prev_node:
            prev_node.right = current
        else:
            head = current
        prev_node = current
        if not current.right:
            tail = current
        current = current.right

    if head and tail:
        head.left = tail
        tail.right = head

    return head


def traverse_dll(head):
    """
    Traverse and print the doubly linked list.
    
    Args:
        head (Node): The head of the doubly linked list.
    """
    if not head:
        return
    current = head
    while True:
        print(f"{current.data} ->", end=' ')
        current = current.right
        if current == head:
            break
    print("None")
