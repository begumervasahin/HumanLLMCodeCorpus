class Node:
    def __init__(self, key, value):
        """
        Initialize a new Node.

        Args:
            key: The key of the node.
            value: The value of the node.
        """
        self.key = key
        self.value = value
        self.prev = None
        self.next = None


class DoublyLinkedList:
    def __init__(self):
        """
        Initialize an empty Doubly Linked List.
        """
        self.head = None
        self.tail = None

    def enter_value(self, key, value):
        """
        Add a new node with the given key and value to the end of the list.

        Args:
            key: The key of the new node.
            value: The value of the new node.

        Returns:
            Node: The newly created node.
        """
        new_node = Node(key, value)
        if self.head is None:
            self.head = self.tail = new_node
        else:
            self.tail.next = new_node
            new_node.prev = self.tail
            self.tail = new_node
        return new_node

    def access_a_node(self, node):
        """
        Move the given node to the end of the list.

        Args:
            node (Node): The node to be moved.
        """
        if node == self.tail:
            return
        if node == self.head:
            self.head = node.next
            if self.head:
                self.head.prev = None
        else:
            node.prev.next = node.next
            node.next.prev = node.prev
        node.next = None
        node.prev = self.tail
        if self.tail:
            self.tail.next = node
        self.tail = node

    def get_head_node(self):
        """
        Get the head node of the list.

        Returns:
            Node: The head node.
        """
        return self.head

    def replace_a_node(self, key, value):
        """
        Replace the head node with a new node with the given key and value.

        Args:
            key: The key of the new node.
            value: The value of the new node.

        Returns:
            Node: The newly created node.
        """
        old_head = self.head
        self.head = self.head.next
        if self.head:
            self.head.prev = None
        else:
            self.tail = None
        new_node = Node(key, value)
        if self.tail is None:
            self.head = self.tail = new_node
        else:
            self.tail.next = new_node
            new_node.prev = self.tail
            self.tail = new_node
        return new_node

    def print_doubly_linked_list(self):
        """
        Print all elements in the list.
        """
        current = self.head
        while current:
            print(f"Key: {current.key}, Value: {current.value}")
            current = current.next


class Caching:
    def __init__(self, capacity):
        """
        Initialize a caching system with a specified capacity.

        Args:
            capacity (int): The maximum number of entries in the cache.
        """
        self.cache_array = {}
        self.array_size = capacity
        self.total_entries = 0
        self.doubly_linked_list = DoublyLinkedList()

    def get(self, key):
        """
        Get the value associated with the given key and move the accessed node to the end.

        Args:
            key: The key to access.

        Returns:
            The value associated with the key, or -1 if the key is not found.
        """
        if key in self.cache_array:
            self.doubly_linked_list.access_a_node(self.cache_array[key])
            return self.cache_array[key].value
        else:
            return -1

    def set(self, key, value):
        """
        Set the value for the given key in the cache. If the cache is full, remove the least recently used item.

        Args:
            key: The key to set.
            value: The value to set.
        """
        if self.total_entries < self.array_size:
            self.cache_array[key] = self.doubly_linked_list.enter_value(key, value)
            self.total_entries += 1
        else:
            node = self.doubly_linked_list.get_head_node()
            del self.cache_array[node.key]
            self.cache_array[key] = self.doubly_linked_list.replace_a_node(key, value)

    def print_all_entries(self):
        """
        Print all entries in the cache.
        """
        self.doubly_linked_list.print_doubly_linked_list()
