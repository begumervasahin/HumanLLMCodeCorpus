class HashTable:
    def __init__(self):
        # Initialize the hash table with empty slots
        self.table = [None] * 10000

    def store(self, string):
        """
        Store the given string in the hash table.
        If the hash slot is empty, initialize a list with the string.
        If the hash slot is not empty, append the string to the list.
        """
        hash_value = self.calculate_hash_value(string)
        if self.table[hash_value] is None:
            self.table[hash_value] = [string]
        else:
            self.table[hash_value].append(string)

    def lookup(self, string):
        """
        Look up the given string in the hash table.
        Return the hash value if the string is found, otherwise return -1.
        """
        hash_value = self.calculate_hash_value(string)
        if self.table[hash_value] and string in self.table[hash_value]:
            return hash_value
        return -1

    def calculate_hash_value(self, string):
        """
        Calculate the hash value of the given string.
        The hash value is calculated using the ASCII values of the first two characters.
        """
        if string:
            return ord(string[0]) * 100 + ord(string[1])
        return -1

# Example usage:
hash_table = HashTable()

# Calculate hash value for 'UDACITY'
print("Hash value for 'UDACITY':", hash_table.calculate_hash_value('UDACITY'))

# Lookup 'UDACITY' before storing
print("Lookup 'UDACITY' before storing:", hash_table.lookup('UDACITY'))

# Store 'UDACITY' in the hash table
hash_table.store('UDACITY')

# Lookup 'UDACITY' after storing
print("Lookup 'UDACITY' after storing:", hash_table.lookup('UDACITY'))

# Store 'UDACIOUS' in the hash table
hash_table.store('UDACIOUS')

# Lookup 'UDACIOUS' after storing
print("Lookup 'UDACIOUS' after storing:", hash_table.lookup('UDACIOUS'))
