from caching.doublylinkedlist.DoublyLinkedList import DoublyLinkedList
from caching.node.HashNode import HashNode

class Cache:
    def __init__(self, capacity):
        """
        Initialize the cache with a given capacity.
        
        Args:
            capacity (int): The maximum number of entries the cache can hold.
        """
        self.cache = {}
        self.capacity = capacity
        self.total_entries = 0
        self.doubly_linked_list = DoublyLinkedList()

    def get(self, key):
        """
        Retrieve the value associated with the given key from the cache.
        
        Args:
            key: The key to look up in the cache.
        
        Returns:
            The value associated with the key if found, otherwise -1.
        """
        if key in self.cache:
            self.doubly_linked_list.access_node(self.cache[key])
            return self.cache[key].value
        else:
            print(-1)
            return -1

    def set(self, key, value):
        """
        Set the value for a given key in the cache. If the cache is full,
        remove the least recently used (LRU) entry before adding the new entry.
        
        Args:
            key: The key to store in the cache.
            value: The value to store in the cache.
        """
        if self.total_entries < self.capacity:
            self.cache[key] = self.doubly_linked_list.add_node(key, value)
            self.total_entries += 1
        else:
            lru_node = self.doubly_linked_list.get_head_node()
            del self.cache[lru_node.key]
            self.cache[key] = self.doubly_linked_list.replace_node(lru_node, key, value)

    def hash_of_key(self, key):
        """
        Calculate the hash of the given key based on the cache capacity.
        
        Args:
            key: The key to hash.
        
        Returns:
            int: The hash value of the key.
        """
        return key % self.capacity

    def get_total_entries(self):
        """
        Get the total number of entries currently in the cache.
        
        Returns:
            int: The total number of entries in the cache.
        """
        return self.total_entries

    def print_all_entries(self):
        """
        Print all entries in the cache.
        """
        self.doubly_linked_list.print_list()
