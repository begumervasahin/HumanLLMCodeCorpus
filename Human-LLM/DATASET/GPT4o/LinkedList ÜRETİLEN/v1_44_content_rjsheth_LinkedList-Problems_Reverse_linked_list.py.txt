class Node:
    def __init__(self, data):
        self.data = data
        self.next = None

class Singly_Linked_List:
    def __init__(self):
        self.head = None

    def add(self, data):
        new_node = Node(data)
        new_node.next = self.head
        self.head = new_node

    def delete(self, data):
        current = self.head
        prev = None
        while current:
            if current.data == data:
                if prev:
                    prev.next = current.next
                else:
                    self.head = current.next
                return
            prev = current
            current = current.next
        print(f"Value {data} not found in the list.")

    def __str__(self):
        result = []
        current = self.head
        while current:
            result.append(str(current.data))
            current = current.next
        return " -> ".join(result) + " -> None"

def reverse_list(input_list):
    l = Singly_Linked_List()
    for i in input_list:
        l.add(i)
    print("input list: ", l)
    
    current = l.head
    prev = None
    next_node = None
    
    while current:
        next_node = current.next
        current.next = prev
        prev = current
        current = next_node
    
    l.head = prev
    return l
