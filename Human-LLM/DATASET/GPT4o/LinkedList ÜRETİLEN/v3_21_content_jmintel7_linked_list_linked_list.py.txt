class Node:
    def __init__(self, data=None):
        """
        Initialize a new Node.

        Args:
            data: The data to be stored in the node.
        """
        self.data = data
        self.next = None


class LinkedList:
    def __init__(self):
        """
        Initialize an empty Linked List.
        """
        self.head = Node()

    def append(self, data):
        """
        Append a new node with the given data to the end of the list.

        Args:
            data: The data to be stored in the new node.
        """
        new_node = Node(data)
        cur_node = self.head
        while cur_node.next:
            cur_node = cur_node.next
        cur_node.next = new_node

    def length(self):
        """
        Calculate the total number of nodes in the list.

        Returns:
            int: The total number of nodes.
        """
        cur_node = self.head
        total = 0
        while cur_node.next:
            cur_node = cur_node.next
            total += 1
        return total

    def display_images(self):
        """
        Collect and return all images from the nodes in the list.

        Returns:
            list: A list of images.
        """
        images = []
        cur_node = self.head
        while cur_node.next:
            cur_node = cur_node.next
            if cur_node.data:
                images.append(cur_node.data[0])
        return images

    def get_student_by_rollno(self, rollno):
        """
        Retrieve the data of a student by their roll number.

        Args:
            rollno: The roll number of the student.

        Returns:
            The data of the student if found, otherwise None.
        """
        cur_node = self.head
        while cur_node.next:
            cur_node = cur_node.next
            if cur_node.data[1] == rollno:
                return cur_node.data
        print('Student with the roll number does not exist')
        return None

    def erase_by_rollno(self, rollno):
        """
        Erase the node containing the student with the given roll number.

        Args:
            rollno: The roll number of the student.
        """
        cur_node = self.head
        while cur_node.next:
            last_node = cur_node
            cur_node = cur_node.next
            if cur_node.data[1] == rollno:
                last_node.next = cur_node.next
                print('Record erased')
                return
        print('Student with the roll number does not exist')

    def get_image_by_index(self, index):
        """
        Retrieve the image data at a specific index in the list.

        Args:
            index (int): The index of the node.

        Returns:
            The image data at the specified index, or None if the index is out of range.
        """
        if index >= self.length():
            print('ERROR: Index out of range')
            return None
        cur_node = self.head
        for _ in range(index + 1):
            cur_node = cur_node.next
        return cur_node.data

    def erase_image_by_index(self, index):
        """
        Erase the node containing the image at the specified index.

        Args:
            index (int): The index of the node to be erased.
        """
        if index >= self.length():
            print('ERROR: Index out of range')
            return None
        cur_node = self.head
        for _ in range(index + 1):
            last_node = cur_node
            cur_node = cur_node.next
        last_node.next = cur_node.next


# Example usage
if __name__ == "__main__":
    linked_list = LinkedList()
    linked_list.append(("image1.jpg", 101))
    linked_list.append(("image2.jpg", 102))
    linked_list.append(("image3.jpg", 103))
    
    print("List of images:")
    print(linked_list.display_images())
    
    print("\nGet image by index 1:")
    print(linked_list.get_image_by_index(1))
    
    print("\nErase image by index 1:")
    linked_list.erase_image_by_index(1)
    print("List of images after erasing:")
    print(linked_list.display_images())
    
    print("\nErase student by roll number 103:")
    linked_list.erase_by_rollno(103)
    print("List of images after erasing student with roll number 103:")
    print(linked_list.display_images())
