class Node:
    def __init__(self, value):
        self.value = value
        self.next_node = None

    def __eq__(self, other):
        return self.value == other.value if other else False

class LinkedList:
    def __init__(self):
        self.head = None

    def __contains__(self, node):
        found = False
        if self.head:
            cursor = self.head
            while cursor:
                if cursor == node:
                    found = True
                    break
                cursor = cursor.next_node
        return found

    def __len__(self):
        cnt = 0
        cursor = self.head
        while cursor:
            cnt += 1
            cursor = cursor.next_node
        return cnt

    def append(self, node):
        if not self.head:
            self.head = node
            return
        cursor = self.head
        while cursor.next_node:
            cursor = cursor.next_node
        cursor.next_node = node

    def insert(self, flag_node, node):
        if self.head == flag_node:
            node.next_node = self.head
            self.head = node
        else:
            _node = self.head
            while _node.next_node:
                if _node.next_node == flag_node:
                    break
                _node = _node.next_node
            node.next_node = _node.next_node
            _node.next_node = node

    def delete(self, node):
        if self.head:
            if self.head == node:
                self.head = self.head.next_node
                return
            prev_node = self.head
            while prev_node.next_node:
                if prev_node.next_node == node:
                    break
                prev_node = prev_node.next_node
            if prev_node.next_node:
                prev_node.next_node = prev_node.next_node.next_node

    def remove_duplicates(self):
        oc = self.head
        while oc:
            pc = oc
            ic = oc.next_node
            while ic:
                if ic == oc:
                    pc.next_node = ic.next_node
                else:
                    pc = ic
                ic = ic.next_node
            oc = oc.next_node

    def reverse(self):
        if not self.head or not self.head.next_node:
            return
        cursor = self.head.next_node
        self.head.next_node = None
        while cursor:
            next_cursor = cursor.next_node
            cursor.next_node = self.head
            self.head = cursor
            cursor = next_cursor

    def print_list(self):
        cursor = self.head
        while cursor:
            print(cursor.value)
            cursor = cursor.next_node

if __name__ == "__main__":
    linked_list = LinkedList()
    
    # Append nodes
    linked_list.append(Node(1))
    linked_list.append(Node(2))
    linked_list.append(Node(3))
    
    # Insert nodes
    linked_list.insert(linked_list.head, Node(0))  # Insert at beginning
    linked_list.insert(linked_list.head.next_node, Node(1.5))  # Insert after first node
    
    # Print list
    print("List after insertions:")
    linked_list.print_list()
    
    # Remove duplicates
    linked_list.append(Node(2))  # Add a duplicate for testing
    linked_list.remove_duplicates()
    print("List after removing duplicates:")
    linked_list.print_list()
    
    # Reverse list
    linked_list.reverse()
    print("List after reversing:")
    linked_list.print_list()
    
    # Delete node
    linked_list.delete(Node(1.5))
    print("List after deleting node with value 1.5:")
    linked_list.print_list()
    
    # Print the length of the list
    print(f"Length of list: {len(linked_list)}")
    
    # Check if a node is in the list
    print(f"Node with value 2 in list: {Node(2) in linked_list}")
