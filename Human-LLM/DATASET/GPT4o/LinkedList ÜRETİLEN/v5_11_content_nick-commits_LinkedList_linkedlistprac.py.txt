class Node:
    def __init__(self, data=None):
        """
        Initialize a new node with the given data.
        
        Args:
            data: The data to store in the node. Default is None.
        """
        self.data = data
        self.next = None

class LinkedList:
    def __init__(self):
        """
        Initialize a new empty linked list.
        """
        self.head = Node()

    def append(self, data):
        """
        Append a new node with the given data to the end of the list.
        
        Args:
            data: The data to store in the new node.
        """
        new_node = Node(data)
        current = self.head
        while current.next:
            current = current.next
        current.next = new_node

    def length(self):
        """
        Return the length of the linked list.
        
        Returns:
            int: The number of nodes in the linked list.
        """
        current = self.head
        total = 0
        while current.next:
            total += 1
            current = current.next
        return total

    def display(self):
        """
        Print all elements in the linked list.
        """
        elements = []
        current = self.head
        while current.next:
            current = current.next
            elements.append(current.data)
        print(elements)

    def get(self, index):
        """
        Get the data of the node at the given index.
        
        Args:
            index (int): The index of the node to retrieve.
        
        Returns:
            The data of the node at the given index, or None if the index is out of range.
        """
        if index >= self.length():
            print('Index out of range')
            return None
        
        current_idx = 0
        current = self.head
        while current.next:
            current = current.next
            if current_idx == index:
                return current.data
            current_idx += 1

    def erase(self, index):
        """
        Erase the node at the given index.
        
        Args:
            index (int): The index of the node to erase.
        """
        if index >= self.length():
            print('Index out of range')
            return
        
        current_idx = 0
        current = self.head
        while current.next:
            previous_node = current
            current = current.next
            if current_idx == index:
                previous_node.next = current.next
                return
            current_idx += 1

# Example usage
if __name__ == "__main__":
    my_list = LinkedList()
    my_list.append(2)
    my_list.append(1)
    my_list.erase(1)
    my_list.display()
