class ListNode:
    def __init__(self, value):
        """
        Initialize a node with a given value and a pointer to the next node.
        
        Args:
            value (int): The value to be stored in the node.
        """
        self.val = value
        self.next = None

class Solution:
    def addTwoNumbers(self, l1, l2):
        """
        Add two numbers represented by linked lists.
        
        Args:
            l1 (ListNode): The head of the first linked list.
            l2 (ListNode): The head of the second linked list.
        
        Returns:
            ListNode: The head of the linked list representing the sum of the two numbers.
        """
        carry = 0
        dummy = ListNode(0)
        current = dummy
        
        while l1 or l2 or carry:
            total = carry
            if l1:
                total += l1.val
                l1 = l1.next
            if l2:
                total += l2.val
                l2 = l2.next
            
            carry, val = divmod(total, 10)
            current.next = ListNode(val)
            current = current.next
        
        return dummy.next

def print_linked_list(node):
    """
    Print the linked list in a readable format.
    
    Args:
        node (ListNode): The head of the linked list.
    """
    values = []
    while node:
        values.append(str(node.val))
        node = node.next
    print(" -> ".join(values))
