class HashTable(object):
    def __init__(self):
        self.table = [None] * 10000

    def store(self, string):
        """Input a string that's stored in the table."""
        hash_value = self.calculate_hash_value(string)
        if self.table[hash_value] is None:
            self.table[hash_value] = [string]
        else:
            self.table[hash_value].append(string)

    def lookup(self, string):
        """Return the hash value if the string is already in the table.
        Return -1 otherwise."""
        hash_value = self.calculate_hash_value(string)
        if self.table[hash_value] is not None and string in self.table[hash_value]:
            return hash_value
        return -1

    def calculate_hash_value(self, string):
        """Helper function to calculate a hash value from a string."""
        if string:
            return ord(string[0]) * 100 + ord(string[1])
        return -1

# Example usage
if __name__ == "__main__":
    hash_table = HashTable()

    # Calculate the hash value for 'UDACITY'
    print(hash_table.calculate_hash_value('UDACITY'))  # Should print 8568

    # Try to look up 'UDACITY' in the hash table
    print(hash_table.lookup('UDACITY'))  # Should print -1 because 'UDACITY' is not yet stored

    # Store 'UDACITY' in the hash table
    hash_table.store('UDACITY')

    # Look up 'UDACITY' again
    print(hash_table.lookup('UDACITY'))  # Should print 8568

    # Store 'UDACIOUS' in the hash table
    hash_table.store('UDACIOUS')

    # Look up 'UDACIOUS'
    print(hash_table.lookup('UDACIOUS'))  # Should print 8568
