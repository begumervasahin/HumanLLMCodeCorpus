import os
import unittest
import tempfile
from app import app

class AppTestCase(unittest.TestCase):
    def setUp(self):
        """Set up a temporary database and configure the app for testing."""
        self.db_fd, app.config['DATABASE'] = tempfile.mkstemp()
        app.config['TESTING'] = True
        self.app = app.test_client()
        with app.app_context():
            app.init_db()

    def tearDown(self):
        """Close the temporary database and remove the file."""
        os.close(self.db_fd)
        os.unlink(app.config['DATABASE'])

    def test_empty_db(self):
        """Test the homepage for an empty database."""
        response = self.app.get('/')
        self.assertIn(b'No entries here so far', response.data)

    def login(self, username, password):
        """Helper function to log in a user."""
        return self.app.post('/login', data=dict(
            username=username,
            password=password
        ), follow_redirects=True)

    def logout(self):
        """Helper function to log out a user."""
        return self.app.get('/logout', follow_redirects=True)

    def test_login_logout(self):
        """Test user login and logout functionality."""
        response = self.login('admin', 'default')
        self.assertIn(b'You were logged in', response.data)
        
        response = self.logout()
        self.assertIn(b'You were logged out', response.data)
        
        response = self.login('adminx', 'default')
        self.assertIn(b'Invalid username', response.data)
        
        response = self.login('admin', 'defaultx')
        self.assertIn(b'Invalid password', response.data)

    def test_add_list(self):
        """Test adding an item to the list."""
        self.login('admin', 'default')
        response = self.app.post('/add_list', data=dict(
            id='1',
            title='Test Title',
            qnty='10',
            date='2024-07-27'
        ), follow_redirects=True)
        self.assertIn(b'1', response.data)
        self.assertIn(b'Test Title', response.data)
        self.assertIn(b'10', response.data)
        self.assertIn(b'2024-07-27', response.data)

if __name__ == '__main__':
    unittest.main()
