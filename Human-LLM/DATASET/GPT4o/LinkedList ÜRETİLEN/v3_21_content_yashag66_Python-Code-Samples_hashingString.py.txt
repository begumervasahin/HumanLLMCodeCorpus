class HashTable:
    def __init__(self):
        """Initialize the hash table with a fixed size of 10000 slots."""
        self.size = 10000
        self.table = [None] * self.size

    def store(self, string):
        """
        Store a string in the hash table.

        Args:
            string (str): The string to store in the table.
        """
        hash_value = self.calculate_hash_value(string)
        if self.table[hash_value] is None:
            self.table[hash_value] = [string]
        else:
            self.table[hash_value].append(string)

    def lookup(self, string):
        """
        Look up a string in the hash table and return its hash value.

        Args:
            string (str): The string to look up in the table.

        Returns:
            int: The hash value of the string if found, -1 otherwise.
        """
        hash_value = self.calculate_hash_value(string)
        if self.table[hash_value] is not None and string in self.table[hash_value]:
            return hash_value
        return -1

    def calculate_hash_value(self, string):
        """
        Calculate the hash value for a given string.

        Args:
            string (str): The string to calculate the hash value for.

        Returns:
            int: The calculated hash value.
        """
        if string:
            return ord(string[0]) * 100 + ord(string[1])
        return -1

# Example usage
if __name__ == "__main__":
    hash_table = HashTable()

    # Calculate the hash value for 'UDACITY'
    print("Hash value for 'UDACITY':", hash_table.calculate_hash_value('UDACITY'))  # Should print 8568

    # Try to look up 'UDACITY' in the hash table
    print("Lookup 'UDACITY':", hash_table.lookup('UDACITY'))  # Should print -1 because 'UDACITY' is not yet stored

    # Store 'UDACITY' in the hash table
    hash_table.store('UDACITY')

    # Look up 'UDACITY' again
    print("Lookup 'UDACITY' after storing:", hash_table.lookup('UDACITY'))  # Should print 8568

    # Store 'UDACIOUS' in the hash table
    hash_table.store('UDACIOUS')

    # Look up 'UDACIOUS'
    print("Lookup 'UDACIOUS':", hash_table.lookup('UDACIOUS'))  # Should print 8568
