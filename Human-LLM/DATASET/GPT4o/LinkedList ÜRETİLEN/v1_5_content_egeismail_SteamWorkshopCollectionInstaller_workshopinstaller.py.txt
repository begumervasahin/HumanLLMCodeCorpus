import requests
import sys
import json
import os
import zipfile

# Check and parse command-line arguments
if len(sys.argv) < 4:
    sys.exit("Usage: script.py <workshopdir> <links_file> <install|download>")

workshopdir = sys.argv[1]
links_file = sys.argv[2]
mode = True if sys.argv[3].lower() == "install" else False

def create_folder(directory):
    if not os.path.exists(directory):
        os.makedirs(directory)
        return directory
    return None

def extract_zip(filename, extract_dir):
    with zipfile.ZipFile(filename, 'r') as zip_ref:
        zip_ref.extractall(extract_dir)
    os.remove(filename)

class PluginContent:
    def __init__(self, data):
        self.data = data
        self.fid = data["publishedfileid"]
        self.filename = data["filename"]
        self.file_size = data["file_size"]
        self.file_url = data["file_url"]
        self.file_title = data["title"]
        self.nfilename = f"{self.fid}-{self.file_title}.zip"
        self.fmid = f"M{self.fid}-{self.file_title}"
        self.mfolder = os.path.join(workshopdir, self.fmid)

    def create_folder(self):
        return create_folder(self.mfolder)

    def install(self):
        if self.create_folder():
            local_filename = os.path.join(self.mfolder, self.nfilename)
            response = requests.get(self.file_url, stream=True)
            with open(local_filename, 'wb') as f:
                for chunk in response.iter_content(chunk_size=1024):
                    if chunk:
                        f.write(chunk)
            extract_zip(local_filename, self.mfolder)
            return local_filename
        return None

    def download(self):
        if self.create_folder():
            local_filename = os.path.join(self.mfolder, self.filename)
            response = requests.get(self.file_url, stream=True)
            with open(local_filename, 'wb') as f:
                for chunk in response.iter_content(chunk_size=1024):
                    if chunk:
                        f.write(chunk)
            return local_filename
        return None

def get_plugin_content(plugin_id):
    payload = {'itemcount': 1, 'publishedfileids[0]': plugin_id}
    response = requests.post("https://api.steampowered.com/ISteamRemoteStorage/GetPublishedFileDetails/v1/", data=payload)
    data = json.loads(response.text)["response"]["publishedfiledetails"][0]
    return PluginContent(data)

def get_pid(url):
    args = dict(tuple(arg.split("=")) for arg in url.split("?")[-1].split("&"))
    return int(args.get("id", None))

def main():
    with open(links_file, "r") as f:
        plugins = [line.strip() for line in f.readlines() if line.strip()]
    
    for plugin_url in plugins:
        plugin_id = get_pid(plugin_url)
        if plugin_id:
            plugin_content = get_plugin_content(plugin_id)
            if mode:
                if plugin_content.install():
                    print(f"{plugin_content.fmid} install complete.")
                else:
                    print(f"{plugin_content.fmid} already installed.")
            else:
                if plugin_content.download():
                    print(f"{plugin_content.fmid} download complete.")
                else:
                    print(f"{plugin_content.fmid} already downloaded.")

if __name__ == '__main__':
    main()
