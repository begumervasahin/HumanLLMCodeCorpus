class Heap:
    def __init__(self):
        """
        Initialize an empty heap.
        """
        self.h = []
        self.currsize = 0

    def _left_child(self, i):
        """
        Get the index of the left child.

        Args:
            i: The index of the parent node.

        Returns:
            The index of the left child if it exists, None otherwise.
        """
        left = 2 * i + 1
        return left if left < self.currsize else None

    def _right_child(self, i):
        """
        Get the index of the right child.

        Args:
            i: The index of the parent node.

        Returns:
            The index of the right child if it exists, None otherwise.
        """
        right = 2 * i + 2
        return right if right < self.currsize else None

    def _max_heapify(self, node):
        """
        Ensure the max heap property is maintained starting from the given node.

        Args:
            node: The index of the node to start heapifying from.
        """
        if node < self.currsize:
            largest = node
            left = self._left_child(node)
            right = self._right_child(node)
            if left is not None and self.h[left] > self.h[largest]:
                largest = left
            if right is not None and self.h[right] > self.h[largest]:
                largest = right
            if largest != node:
                self.h[node], self.h[largest] = self.h[largest], self.h[node]
                self._max_heapify(largest)

    def build_heap(self, a):
        """
        Build a heap from an array.

        Args:
            a: The array to build the heap from.
        """
        self.currsize = len(a)
        self.h = list(a)
        for i in range(self.currsize // 2 - 1, -1, -1):
            self._max_heapify(i)

    def get_max(self):
        """
        Extract the maximum element from the heap.

        Returns:
            The maximum element if the heap is not empty, None otherwise.
        """
        if self.currsize >= 1:
            max_elem = self.h[0]
            self.h[0], self.h[self.currsize - 1] = self.h[self.currsize - 1], self.h[0]
            self.currsize -= 1
            self._max_heapify(0)
            return max_elem
        return None

    def heap_sort(self):
        """
        Perform heap sort on the heap.
        """
        size = self.currsize
        while self.currsize > 1:
            self.h[0], self.h[self.currsize - 1] = self.h[self.currsize - 1], self.h[0]
            self.currsize -= 1
            self._max_heapify(0)
        self.currsize = size

    def insert(self, data):
        """
        Insert a new element into the heap.

        Args:
            data: The data to be inserted.
        """
        self.h.append(data)
        self.currsize += 1
        curr = self.currsize - 1
        parent = (curr - 1) // 2
        while curr > 0 and self.h[curr] > self.h[parent]:
            self.h[curr], self.h[parent] = self.h[parent], self.h[curr]
            curr = parent
            parent = (curr - 1) // 2

    def display(self):
        """
        Display the elements of the heap.
        """
        print(self.h)


def main():
    print("=== Heap (max) - A specialized tree-based data structure that satisfies the max-heap property.")
    sentence = input("[STR] What is your sentence? ").split()
    h = Heap()
    h.build_heap(sentence)
    h.heap_sort()
    h.display()


if __name__ == '__main__':
    main()
