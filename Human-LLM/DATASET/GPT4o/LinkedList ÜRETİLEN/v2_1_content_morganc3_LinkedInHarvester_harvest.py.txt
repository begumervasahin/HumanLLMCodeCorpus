import json
import sys
import argparse
import re
from urllib.request import build_opener
from unidecode import unidecode
import ssl

# Disable SSL verification
ssl._create_default_https_context = ssl._create_unverified_context

# Set up argument parser
parser = argparse.ArgumentParser(description='Creates email addresses with optional formatters from names of employees of a company on LinkedIn.')
parser.add_argument('COMPANY', help='Company ID')
parser.add_argument('DOMAIN', help='Domain to be used in email address')
parser.add_argument('COOKIE', help='Cookie file', metavar="FILE")
parser.add_argument("-o", "--output", default='emails.txt', help="Path to output text file for emails")
parser.add_argument("-f", action="store_true", dest="abbrevF", default=False, help="Abbreviate first name")
parser.add_argument("-l", action="store_true", dest="abbrevL", default=False, help="Abbreviate last name")
parser.add_argument("-s", action="store_true", dest="swap", default=False, help="Switch order of first and last names")

args = parser.parse_args()
company_id = args.COMPANY
domain = args.DOMAIN

# Read cookies from file
with open(args.COOKIE, "r") as f:
    csrf_token = f.readline().strip()
    session_id = f.readline().strip()

# Set cookies
cookie1 = f'li_at={session_id}'
cookie2 = f'JSESSIONID={csrf_token}'

emails = []
last_page = False

# List of credentials and titles to be removed from names
creds = '''AAMS
ACA
ADPA
AIF
AWMA
CAIA
CAP
CDFA
CEP
CFA
CFP
ChFC
CIMA
CLU
CPA
CMA
CMM
CMP
CDFA
CPWA
CRPC
CRPS
CTP
CWS
Jr.
MBA
M.A.
RICP
Sr.
WMS'''.split('\n')

def format_name(original_name):
    """Format the name to create an email address."""
    original_name = unidecode(original_name)
    pattern = r'\s*(?:(?:(?:[\x80-\xff]|\)(?:[Rr]|[Tt][Mm])\()?(?:%s),?)+|[^,]+,)*\s*([A-Za-z \-\.\'\(\)]+)\s*' % ('|'.join([x[::-1] for x in creds]))
    match = re.match(pattern, original_name[::-1])
    
    if not match:
        print(f'Unexpected format for {original_name}')
        return None
    
    name = match.group(1)[::-1]
    final_name = re.sub(r'(\(.*\)%s|[-\'\.,\\])' % (r'|[A-Z]\.' if not args.abbrevF and not args.abbrevL else ''), '', name).lower().split()
    email = '.'.join(final_name) + '@' + domain
    return email

def harvest(current_index):
    """Harvest employee data from LinkedIn and format emails."""
    global last_page
    url = (
        f"https://www.linkedin.com/voyager/api/search/blended?count=49&origin=OTHER&queryContext=List(spellCorrectionEnabled-%3Etrue,"
        f"relatedSearchesEnabled-%3Etrue,kcardTypes-%3EPROFILE%7CCOMPANY)&q=all&filters=List(currentCompany-%3E{company_id},resultType-%3EPEOPLE)&start={current_index}"
    )
    opener = build_opener()
    opener.addheaders.append(('csrf-token', csrf_token))
    opener.addheaders.append(('Cookie', cookie1 + '; ' + cookie2))
    opener.addheaders.append(('x-restli-protocol-version', '2.0.0'))
    
    response = opener.open(url)
    data = json.load(response)["elements"][0]["elements"]
    
    if len(data) != 49:
        last_page = True
    
    for employee in data:
        profile = employee["image"]["attributes"][0]["miniProfile"]
        first_name = profile["firstName"].replace(' ', '')
        last_name = profile["lastName"].replace(' ', '')
        
        if not first_name:
            continue
        
        if args.abbrevF:
            first_name = first_name[0]
        
        if args.abbrevL:
            last_name = last_name[0]
        
        full_name = f"{last_name} {first_name}" if args.swap else f"{first_name} {last_name}"
        email = format_name(full_name)
        
        if email:
            emails.append(email)

def main():
    """Main function to execute the script."""
    current_index = 0
    
    while not last_page:
        harvest(current_index)
        current_index += 49
    
    with open(args.output, "w") as output_file:
        for email in emails:
            try:
                output_file.write(f"{email}\n")
            except Exception as e:
                print(f'Error writing email "{email}" due to: {e}')
    
    print(f"Done! {len(emails)} emails have been written to {args.output}")

if __name__ == "__main__":
    main()
