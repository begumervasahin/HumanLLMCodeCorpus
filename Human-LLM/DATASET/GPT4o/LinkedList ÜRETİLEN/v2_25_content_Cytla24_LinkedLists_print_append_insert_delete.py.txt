class Node:
    def __init__(self, data_value=None):
        """
        Initialize a new Node.
        
        Args:
            data_value: The data to be stored in the node.
        """
        self.data_value = data_value
        self.next_value = None

class LinkedList:
    def __init__(self):
        """
        Initialize an empty Linked List.
        """
        self.head_value = None

    def print_list(self):
        """
        Print all elements in the list.
        """
        current_node = self.head_value
        while current_node is not None:
            print(current_node.data_value)
            current_node = current_node.next_value

    def insert_at_beginning(self, data_value):
        """
        Insert a new node with the given data at the beginning of the list.
        
        Args:
            data_value: The data to be stored in the new node.
        """
        new_node = Node(data_value)
        new_node.next_value = self.head_value
        self.head_value = new_node

    def insert_after(self, prev_node, data_value):
        """
        Insert a new node with the given data after the specified node.
        
        Args:
            prev_node (Node): The node after which the new node should be inserted.
            data_value: The data to be stored in the new node.
        """
        if prev_node is None:
            print("The given previous node must be in the LinkedList.")
            return
        new_node = Node(data_value)
        new_node.next_value = prev_node.next_value
        prev_node.next_value = new_node

    def insert_at_end(self, data_value):
        """
        Insert a new node with the given data at the end of the list.
        
        Args:
            data_value: The data to be stored in the new node.
        """
        new_node = Node(data_value)
        if self.head_value is None:
            self.head_value = new_node
            return
        last_node = self.head_value
        while last_node.next_value is not None:
            last_node = last_node.next_value
        last_node.next_value = new_node

    def delete_node(self, key):
        """
        Delete the first node with the specified data.
        
        Args:
            key: The data of the node to be deleted.
        """
        current_node = self.head_value
        if current_node is not None:
            if current_node.data_value == key:
                self.head_value = current_node.next_value
                current_node = None
                return
        prev = None
        while current_node is not None and current_node.data_value != key:
            prev = current_node
            current_node = current_node.next_value
        if current_node is None:
            return
        prev.next_value = current_node.next_value
        current_node = None

# Example usage
if __name__ == "__main__":
    linked_list = LinkedList()

    # Manually creating nodes and linking them
    linked_list.head_value = Node("Aimie Ojuba")
    node2 = Node("Latifs")
    node3 = Node("Toni")
    node4 = Node("Slap")

    linked_list.head_value.next_value = node2
    node2.next_value = node3
    node3.next_value = node4

    # Performing various operations on the linked list
    linked_list.insert_at_beginning("Imaa")
    linked_list.insert_at_end("Beast")
    linked_list.insert_after(linked_list.head_value.next_value, "Shile")
    linked_list.delete_node("Slap")

    # Printing the list
    linked_list.print_list()
