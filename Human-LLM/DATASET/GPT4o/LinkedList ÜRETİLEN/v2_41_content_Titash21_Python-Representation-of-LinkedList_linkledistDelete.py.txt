class Node:
    """A class representing a node in a linked list."""
    
    def __init__(self, data):
        self.data = data
        self.next = None

class LinkedList:
    """A class representing a singly linked list."""
    
    def __init__(self):
        self.head = None

    def insert_at_front(self, data):
        """Insert a new node with the given data at the front of the list."""
        new_node = Node(data)
        new_node.next = self.head
        self.head = new_node

    def delete_value(self, data):
        """Delete the first node with the given data."""
        if self.head is None:
            print("You are trying to delete from an empty list")
            return

        if self.head.data == data:
            self.head = self.head.next
            return

        current = self.head
        previous = None
        while current and current.data != data:
            previous = current
            current = current.next

        if current is None:
            print("The data you want to delete is not in the list")
            return

        previous.next = current.next

    def print_list(self):
        """Print the contents of the linked list."""
        if self.head is None:
            print("Empty linked list")
            return

        current = self.head
        while current:
            print(current.data, end=" -> ")
            current = current.next
        print("None")

def main():
    linked_list = LinkedList()
    while True:
        print("\nOPTIONS FOR THIS PROGRAM")
        print("1. Insert at front")
        print("2. Delete a particular node")
        print("3. Print contents of the linked list")
        print("4. Exit")
        try:
            choice = int(input("Enter your choice: "))
        except ValueError:
            print("Invalid input! Please enter a number.")
            continue
        
        if choice == 1:
            try:
                value = int(input("Enter data to add in front: "))
                linked_list.insert_at_front(value)
                linked_list.print_list()
            except ValueError:
                print("Invalid input! Please enter a valid number.")
        elif choice == 2:
            try:
                value = int(input("Enter data to delete: "))
                linked_list.delete_value(value)
                linked_list.print_list()
            except ValueError:
                print("Invalid input! Please enter a valid number.")
        elif choice == 3:
            linked_list.print_list()
        elif choice == 4:
            break
        else:
            print("Wrong input! Please try again.")

if __name__ == "__main__":
    main()
