class DisjointSet:
    """
    Class representing a disjoint set data structure.
    
    Usage:
        ds = DisjointSet(size)
        ds.find(value)
        ds.union(value1, value2)
    
    This implementation supports the union-find operations with path compression.
    
    @author Greg Gagne
    @date November 2014
    """

    def __init__(self, size):
        """
        Constructor to initialize the disjoint set.

        Parameters:
            size: The number of elements in the disjoint set. Each element is initialized
                  as its own set with a negative value representing its rank.
        """
        if size < 0:
            raise ValueError("Size must be >= 0")
        
        self.size = size
        self.parent = [-1] * (size + 1)  # Initialize all elements as their own root

    def union(self, root1, root2):
        """
        Performs a union of two sets by making root2 join the set of root1.

        Parameters:
            root1: The root of the first set.
            root2: The root of the second set.
        """
        if root1 < 0 or root1 > self.size or root2 < 0 or root2 > self.size:
            raise ValueError("Illegal value")
        
        self.parent[root2] = root1

    def find(self, root):
        """
        Finds the representative (root) of the set containing the specified element with path compression.

        Parameters:
            root: The element to find the set representative for.

        Returns:
            The root of the set containing the specified element.
        """
        if root < 0 or root > self.size:
            raise ValueError("Illegal value")
        
        if self.parent[root] < 0:
            return root
        else:
            self.parent[root] = self.find(self.parent[root])  # Path compression
            return self.parent[root]
