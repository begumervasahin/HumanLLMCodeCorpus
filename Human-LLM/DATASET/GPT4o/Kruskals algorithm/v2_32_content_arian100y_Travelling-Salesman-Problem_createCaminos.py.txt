import statistics

def db_to_list(filename):
    """
    Reads a CSV file and converts it into a list of tuples.
    Each tuple contains two float values representing x and y coordinates.
    """
    with open(filename, "r") as file:
        return [(float(x), float(y)) for line in file for x, y in [line.strip().split(",")]]

def create_paths_for_quadrant(quadrant, filename):
    """
    Creates paths for a quadrant by finding the closest points and writes them to a file.

    The function writes each point in the quadrant along with the 10 closest points to it
    based on Euclidean distance. The output is written to the specified file.
    """
    with open(filename, "w") as file:
        num_points = len(quadrant)
        
        for i, point in enumerate(quadrant):
            connections = []
            
            # Generate connections by considering half of the points in both directions
            for j in range(1, num_points // 2 + 1):
                for direction in [-1, 1]:  # Check both backward and forward directions
                    neighbor_index = (i + j * direction) % num_points
                    neighbor = quadrant[neighbor_index]
                    distance = ((neighbor[0] - point[0]) ** 2 + (neighbor[1] - point[1]) ** 2) ** 0.5
                    connections.append((neighbor, distance))
            
            # Sort connections by distance and select the 10 closest ones
            connections.sort(key=lambda x: x[1])
            closest_points = " ".join(f"{x[0][0]},{x[0][1]}" for x in connections[:10])
            
            # Write the point and its closest connections to the file
            file.write(f"{point[0]},{point[1]} {closest_points}\n")

def separate_lists(points):
    """
    Separates a list of (x, y) tuples into two sorted lists: one for x values and one for y values.

    Returns:
        tuple: A tuple containing two lists: one sorted by x-values and the other by y-values.
    """
    sorted_by_x = sorted(points, key=lambda p: p[0])
    x_values = [p[0] for p in sorted_by_x]
    y_values = [p[1] for p in points]
    return x_values, y_values

def create_quadrants(points):
    """
    Splits a list of points into four quadrants based on the median x and y values.

    The function calculates the median of x and y values and uses them to divide
    the points into four quadrants.

    Returns:
        tuple: A tuple containing four lists representing the points in each quadrant,
               followed by the median x and y values.
    """
    x_values, y_values = separate_lists(points)
    x_median = statistics.median(x_values)
    y_median = statistics.median(y_values)
    
    q1, q2, q3, q4 = [], [], [], []
    
    for x, y in points:
        if x <= x_median and y >= y_median:
            q1.append((x, y))
        elif x > x_median and y >= y_median:
            q2.append((x, y))
        elif x <= x_median and y < y_median:
            q3.append((x, y))
        else:
            q4.append((x, y))
    
    return q1, q2, q3, q4, x_median, y_median

# Example usage:
filename = "data.csv"  # Replace with your actual file path
points = db_to_list(filename)
quadrants, x_median, y_median = create_quadrants(points)

for idx, quadrant in enumerate(quadrants, 1):
    output_file = f"quadrant_{idx}_paths.txt"
    create_paths_for_quadrant(quadrant, output_file)
