import heapq

# Global variables to store graph-related data
EDGES = []        # Priority queue (min-heap) to store edges
VERTICES = set()  # Set to store unique vertices
PARENT = {}       # Dictionary to store the parent of each vertex for union-find
RANK = {}         # Dictionary to store the rank of each vertex for union-find
MST = []          # List to store the edges of the Minimum Spanning Tree

def read_graph(file_path, delimiter=' '):
    """
    Reads a graph from a file and stores the edges and vertices in global variables.

    :param file_path: The path to the file containing the graph data.
    :param delimiter: The delimiter used in the file to separate values (default is space).
    :return: The list of edges in the priority queue (min-heap).
    """
    with open(file_path, 'r') as file:
        for line in file:
            v1, v2, weight = [x.strip() for x in line.split(delimiter)]
            weight = int(weight)
            # Avoid adding duplicate edges
            if (weight, v2, v1) not in EDGES:
                heapq.heappush(EDGES, (weight, v1, v2))
                VERTICES.add(v1)
                VERTICES.add(v2)
    return EDGES

def make_set(vertex):
    """
    Initializes the parent and rank of a vertex.

    :param vertex: The vertex to initialize.
    """
    PARENT[vertex] = vertex
    RANK[vertex] = 0

def find(vertex):
    """
    Finds the root parent of a vertex using path compression.

    :param vertex: The vertex to find the root parent for.
    :return: The root parent of the vertex.
    """
    if PARENT[vertex] != vertex:
        PARENT[vertex] = find(PARENT[vertex])
    return PARENT[vertex]

def union(vertex1, vertex2):
    """
    Unites two sets containing the vertices vertex1 and vertex2 using union by rank.

    :param vertex1: The first vertex.
    :param vertex2: The second vertex.
    """
    root1 = find(vertex1)
    root2 = find(vertex2)

    if root1 != root2:
        if RANK[root1] > RANK[root2]:
            PARENT[root2] = root1
        else:
            PARENT[root1] = root2
            if RANK[root1] == RANK[root2]:
                RANK[root2] += 1

def kruskals():
    """
    Implements Kruskal's algorithm to find the Minimum Spanning Tree (MST).

    :return: A tuple containing the MST list and the total cost.
    """
    total_cost = 0
    for vertex in VERTICES:
        make_set(vertex)
    
    while EDGES:
        weight, v1, v2 = heapq.heappop(EDGES)
        if find(v1) != find(v2):
            union(v1, v2)
            total_cost += weight
            MST.append((v1, v2, weight))
    
    return MST, total_cost

# Example usage
if __name__ == "__main__":
    # Example: Reading the graph from 'graph_data.txt' with space as a delimiter
    read_graph('graph_data.txt')
    
    # Finding the Minimum Spanning Tree (MST) using Kruskal's algorithm
    mst, total_cost = kruskals()
    
    # Output the result
    print("Minimum Spanning Tree:")
    for edge in mst:
        print(f"Edge {edge[0]} - {edge[1]} with weight {edge[2]}")
    
    print("\nTotal Cost of MST:", total_cost)
