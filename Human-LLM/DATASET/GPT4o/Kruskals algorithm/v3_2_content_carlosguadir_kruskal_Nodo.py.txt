class Enlace:
    """Represents a connection to another node with an associated weight."""

    def __init__(self, destino, peso):
        """
        Initialize an Enlace instance.

        Args:
            destino (Nodo): The destination node of the link.
            peso (float): The weight of the link.
        """
        self.destino = destino
        self.peso = peso

    def obtener_destino(self):
        """Return the destination node of the link."""
        return self.destino

    def obtener_peso(self):
        """Return the weight of the link."""
        return self.peso


class Nodo:
    """Represents a graph node with a name and a list of links (enlaces) to other nodes."""

    def __init__(self, nombre):
        """
        Initialize a Nodo instance.

        Args:
            nombre (str): The name of the node.
        """
        self.nombre = nombre
        self.enlaces = []

    def agregar_enlace(self, destino, peso):
        """
        Add a link to the node if it doesn't already exist.

        Args:
            destino (Nodo): The destination node for the new link.
            peso (float): The weight of the new link.
        """
        if not self._enlace_existe(destino):
            self.enlaces.append(Enlace(destino, peso))

    def _enlace_existe(self, destino):
        """
        Check if a link to a given destination already exists.

        Args:
            destino (Nodo): The destination node to check.

        Returns:
            bool: True if a link exists, False otherwise.
        """
        return any(enlace.obtener_destino() == destino for enlace in self.enlaces)

    def obtener_enlaces(self):
        """
        Get all links associated with the node.

        Returns:
            list[Enlace]: A list of all links.
        """
        return self.enlaces

    def contar_enlaces(self):
        """
        Get the number of links associated with the node.

        Returns:
            int: The number of links.
        """
        return len(self.enlaces)

    def obtener_nombre(self):
        """
        Get the name of the node.

        Returns:
            str: The name of the node.
        """
        return self.nombre

    def obtener_peso_enlace_por_posicion(self, posicion):
        """
        Get the weight of a link by its position in the list.

        Args:
            posicion (int): The position of the link in the list.

        Returns:
            float or None: The weight of the link, or None if the position is invalid.
        """
        return self.enlaces[posicion].obtener_peso() if 0 <= posicion < len(self.enlaces) else None

    def obtener_destino_enlace_por_posicion(self, posicion):
        """
        Get the destination of a link by its position in the list.

        Args:
            posicion (int): The position of the link in the list.

        Returns:
            Nodo or None: The destination node, or None if the position is invalid.
        """
        return self.enlaces[posicion].obtener_destino() if 0 <= posicion < len(self.enlaces) else None

    def eliminar_enlace(self, posicion):
        """
        Remove a link by its position in the list.

        Args:
            posicion (int): The position of the link to remove.

        Returns:
            bool: True if the link was successfully removed, False if the position was invalid.
        """
        if 0 <= posicion < len(self.enlaces):
            del self.enlaces[posicion]
            return True
        return False
