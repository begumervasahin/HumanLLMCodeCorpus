def generate_kruskal_input(matrix):
    """
    Generates a list of edges for Kruskal's algorithm from an adjacency matrix.
    
    Parameters:
        matrix: A 2D list representing the adjacency matrix of a graph.
    
    The function writes the edges to a file named 'kruskal.txt', where each line
    contains two vertices and the weight of the edge connecting them.
    """
    with open('kruskal.txt', 'w') as output_file:
        for row in range(len(matrix)):
            for col in range(row + 1, len(matrix[row])):  # Only consider upper triangle
                weight = matrix[row][col]
                if weight != 0:
                    output_file.write(f'{row} {col} {weight}\n')

def read_matrix_from_file(filename):
    """
    Reads an adjacency matrix from a file and returns it as a 2D list.
    
    Parameters:
        filename: The name of the file containing the adjacency matrix.
    
    Returns:
        matrix: A 2D list representing the adjacency matrix of a graph.
    """
    with open(filename, 'r') as file:
        lines = file.readlines()
        size = len(lines)
        matrix = [[0 for _ in range(size)] for _ in range(size)]
        
        for i, line in enumerate(lines):
            values = list(map(int, line.split()))
            matrix[i] = values
    
    return matrix

if __name__ == '__main__':
    # Read the adjacency matrix from 'matrix.txt'
    adjacency_matrix = read_matrix_from_file('matrix.txt')
    
    # Generate the Kruskal input file 'kruskal.txt' from the adjacency matrix
    generate_kruskal_input(adjacency_matrix)
