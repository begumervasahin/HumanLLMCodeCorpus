# glb.py
EDGES = []       # List to store the edges in a priority queue (min-heap)
VERTICES = set() # Set to store unique vertices
PARENT = {}      # Dictionary to store the parent of each vertex for union-find
RANK = {}        # Dictionary to store the rank of each vertex for union-find
MST = []         # List to store the edges of the Minimum Spanning Tree
import heapq
import glb

def read_graph(file, delimiter):
    """
    Reads a graph from a file and stores the edges and vertices in global variables.

    Parameters:
    - file (str): The path to the file containing the graph data.
    - delimiter (str): The delimiter used in the file to separate values.

    Returns:
    - list: The list of edges in the priority queue (min-heap).
    """
    with open(file, 'r') as file_path:
        for line in file_path:
            v1, v2, weight = [x.strip() for x in line.split(delimiter)]
            weight = int(weight)
            if (weight, v2, v1) in glb.EDGES:
                continue
            heapq.heappush(glb.EDGES, (weight, v1, v2))
            glb.VERTICES.add(v1)
            glb.VERTICES.add(v2)
    return glb.EDGES

def make_set(v):
    """
    Initializes the parent and rank of a vertex.

    Parameters:
    - v (str or int): The vertex to initialize.
    """
    glb.PARENT[v] = v
    glb.RANK[v] = 0

def find(v):
    """
    Finds the root parent of a vertex using path compression.

    Parameters:
    - v (str or int): The vertex to find the root parent for.

    Returns:
    - str or int: The root parent of the vertex.
    """
    if glb.PARENT[v] != v:
        glb.PARENT[v] = find(glb.PARENT[v])
    return glb.PARENT[v]

def union(v1, v2):
    """
    Unites two sets containing the vertices v1 and v2 using union by rank.

    Parameters:
    - v1, v2 (str or int): The vertices to unite.
    """
    root1 = find(v1)
    root2 = find(v2)

    if root1 != root2:
        if glb.RANK[root1] > glb.RANK[root2]:
            glb.PARENT[root2] = root1
        else:
            glb.PARENT[root1] = root2
            if glb.RANK[root1] == glb.RANK[root2]:
                glb.RANK[root2] += 1

def kruskals():
    """
    Implements Kruskal's algorithm to find the Minimum Spanning Tree (MST).

    Returns:
    - tuple: A tuple containing the MST list and the total cost.
    """
    cdist = 0
    for v in glb.VERTICES:
        make_set(v)
    
    while glb.EDGES:
        weight, v1, v2 = heapq.heappop(glb.EDGES)
        if find(v1) != find(v2):
            union(v1, v2)
            cdist += weight
            glb.MST.append((v1, v2, str(weight), str(cdist)))
    
    return glb.MST, cdist

# Example usage
if __name__ == "__main__":
    # Example: Reading the graph from 'graph_data.txt' with space as a delimiter
    read_graph('graph_data.txt', ' ')
    
    # Finding the Minimum Spanning Tree (MST) using Kruskal's algorithm
    mst, total_cost = kruskals()
    
    # Output the result
    print("Minimum Spanning Tree:")
    for edge in mst:
        print(edge)
    
    print("\nTotal Cost of MST:", total_cost)
