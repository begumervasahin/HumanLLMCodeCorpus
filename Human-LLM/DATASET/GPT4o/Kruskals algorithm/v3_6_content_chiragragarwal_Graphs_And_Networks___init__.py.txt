# Import necessary modules for graph reading and plotting
from PlotGraph import PlotGraph
from ReadGraph import ReadGraph

def main():
    """
    Main function to read graph data and visualize it.
    
    This function initializes the necessary components to read graph data from a 
    file and then visualize it using the PlotGraph class.
    """
    # Initialize the graph reader to handle reading of graph data
    graph_reader = ReadGraph()

    # Read the graph data from a specified file or data source
    graph_file = 'graph_data.txt'  # Replace with your actual data source
    graph = graph_reader.read(graph_file)

    # Initialize the graph plotter to handle visualization
    graph_plotter = PlotGraph()

    # Plot the graph using the data read by ReadGraph
    graph_plotter.plot(graph)

if __name__ == "__main__":
    main()
