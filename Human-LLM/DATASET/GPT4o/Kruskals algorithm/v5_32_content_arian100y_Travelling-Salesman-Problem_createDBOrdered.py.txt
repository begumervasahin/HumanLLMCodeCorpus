def partition(arr, low, high):
    """
    Partition the array for quicksort based on the latitude values.

    Parameters:
        arr (list of tuples): The array of tuples to be partitioned.
        low (int): The starting index of the segment to be partitioned.
        high (int): The ending index of the segment to be partitioned.

    Returns:
        int: The partition index where the pivot element is placed.
    """
    i = low - 1
    pivot = arr[high][0]  # Pivot is the last element's latitude

    for j in range(low, high):
        if arr[j][0] <= pivot:  # Compare latitude values
            i += 1
            arr[i], arr[j] = arr[j], arr[i]  # Swap elements

    arr[i + 1], arr[high] = arr[high], arr[i + 1]  # Place pivot in correct position
    return i + 1

def quicksort(arr, low, high):
    """
    QuickSort algorithm to sort an array of tuples based on latitude values.

    Parameters:
        arr (list of tuples): The array of tuples to be sorted.
        low (int): The starting index of the segment to be sorted.
        high (int): The ending index of the segment to be sorted.
    """
    if low < high:
        pi = partition(arr, low, high)  # Partition the array
        quicksort(arr, low, pi - 1)     # Recursively sort the left segment
        quicksort(arr, pi + 1, high)    # Recursively sort the right segment

def filter_and_sort_coordinates(filename, location):
    """
    Filter the coordinates from the file for a specific location and sort them by latitude.

    Parameters:
        filename (str): The name of the input file containing the data.
        location (str): The location to filter by.

    Returns:
        list of tuples: A sorted list of tuples containing the coordinates.
    """
    coordinates = []

    with open(filename, encoding="utf-8") as file:
        for line in file:
            data = line.strip().split(",")
            if data[3] == location:
                latitude = float(data[5])
                longitude = float(data[6])
                coordinates.append((latitude, longitude))

    # Sort the coordinates by latitude using QuickSort
    quicksort(coordinates, 0, len(coordinates) - 1)

    return coordinates

def write_coordinates_to_file(coordinates, output_filename):
    """
    Write the sorted coordinates to an output file.

    Parameters:
        coordinates (list of tuples): The sorted coordinates to write.
        output_filename (str): The name of the output file.
    """
    with open(output_filename, "w") as output_file:
        for latitude, longitude in coordinates:
            output_file.write(f"{latitude},{longitude}\n")

def main():
    input_filename = "db2.txt"
    output_filename = "testSET4.txt"
    location = "LOS OLIVOS"

    # Filter and sort the coordinates
    sorted_coordinates = filter_and_sort_coordinates(input_filename, location)

    # Print the sorted coordinates
    for coord in sorted_coordinates:
        print(coord)

    # Write the sorted coordinates to the output file
    write_coordinates_to_file(sorted_coordinates, output_filename)

    # Print the number of filtered coordinates
    print(f"Total coordinates: {len(sorted_coordinates)}")

if __name__ == "__main__":
    print("Sorting coordinates for LOS OLIVOS...")
    main()
