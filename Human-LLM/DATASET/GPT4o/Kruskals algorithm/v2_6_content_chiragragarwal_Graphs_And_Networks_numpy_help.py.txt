import numpy as np

# Define a matrix using a string where semicolons separate the rows
matrix_str = '1 2 3 4 ; 5 6 7 8 ; 0 1 2 3'

# Convert the string representation to a NumPy matrix
graph_matrix = np.matrix(matrix_str)

# Display the shape of the matrix (rows, columns)
print("Matrix shape:", graph_matrix.shape)

# Access specific elements in the matrix
element_at_0_1 = graph_matrix[0, 1]  # Element at the first row, second column
element_at_2_3 = graph_matrix[2, 3]  # Element at the third row, fourth column

# Print the accessed elements
print("Element at (0,1):", element_at_0_1)
print("Element at (2,3):", element_at_2_3)

# Modify a specific element in the matrix
graph_matrix[0, 1] = -8

# Access and print the entire second row (index 1)
second_row = graph_matrix[1]
print("Second row:", second_row)

# Access and print the entire second column (index 1)
second_column = graph_matrix[:, 1]
print("Second column:", second_column)
