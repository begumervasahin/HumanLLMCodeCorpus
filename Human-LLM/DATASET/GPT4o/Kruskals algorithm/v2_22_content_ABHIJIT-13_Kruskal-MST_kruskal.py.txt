def find_root(vertex):
    """
    Find the root of the given vertex using path compression.
    This function flattens the structure by making each node point directly to the root.
    """
    while root_ids[vertex] != vertex:
        # Path compression: make the vertex point directly to its grandparent
        root_ids[vertex] = root_ids[root_ids[vertex]]
        vertex = root_ids[vertex]
    return vertex

def union_sets(u, v):
    """
    Union the sets containing vertices 'u' and 'v' by linking their roots.
    """
    root_u = find_root(u)
    root_v = find_root(v)
    root_ids[root_v] = root_u  # Link the root of v's set to the root of u's set

def kruskal(graph_edges):
    """
    Kruskal's algorithm to compute the Minimum Spanning Tree (MST).
    
    Parameters:
        graph_edges: List of edges in the graph, where each edge is a tuple (u, v, weight)
    
    Returns:
        The total weight of the Minimum Spanning Tree.
    """
    global root_ids
    total_weight = 0
    
    # Iterate through the sorted edge list
    for u, v, weight in graph_edges:
        # If u and v are in different sets, include this edge in the MST
        if find_root(u) != find_root(v):
            total_weight += weight
            union_sets(u, v)
    
    return total_weight

if __name__ == "__main__":
    # Input the number of vertices and edges
    V, E = map(int, input("Enter number of vertices and edges: ").split())
    graph_edges = []

    # Input the edges with their weights
    for _ in range(E):
        u, v, weight = map(int, input("Enter edge (u v weight): ").split())
        graph_edges.append([u, v, weight])

    # Sort the edges by weight
    graph_edges.sort(key=lambda x: x[2])

    # Initialize the root ids for each vertex
    root_ids = list(range(V + 1))

    # Compute and print the total weight of the Minimum Spanning Tree
    mst_weight = kruskal(graph_edges)
    print("Total weight of Minimum Spanning Tree:", mst_weight)
