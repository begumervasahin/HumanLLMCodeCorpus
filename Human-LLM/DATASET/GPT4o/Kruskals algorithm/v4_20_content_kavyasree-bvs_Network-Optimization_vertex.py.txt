class Vertex:
    def __init__(self, name):
        self.name = name         # Name of the vertex
        self.neighbors = []      # List to store neighboring vertices

    def add_neighbor(self, neighbor):
        if isinstance(neighbor, Vertex):
            # Add the neighbor if it's not already present
            if neighbor.name not in self.neighbors:
                self.neighbors.append(neighbor.name)
                neighbor.neighbors.append(self.name)

                # Keep neighbors lists sorted for both vertices
                self.neighbors.sort()
                neighbor.neighbors.sort()
        else:
            return False  # Return False if the neighbor is not a Vertex instance

    def __repr__(self):
        return f"Vertex({self.name}): Neighbors -> {self.neighbors}"
