class DisjointSet:
    def __init__(self):
        self.parent = {}

    def add(self, item):
        """Add a new item to the disjoint set, initializing its parent to itself."""
        self.parent[item] = item

    def find(self, item):
        """Find the root of the item with path compression."""
        if self.parent[item] != item:
            # Path compression: update the parent to the root
            self.parent[item] = self.find(self.parent[item])
        return self.parent[item]

    def union(self, item1, item2):
        """Union the sets containing item1 and item2 by linking their roots."""
        root1 = self.find(item1)
        root2 = self.find(item2)
        if root1 != root2:
            self.parent[root2] = root1


def kruskal(nodes, sorted_edges):
    """Kruskal's algorithm to find the Minimum Spanning Tree (MST)."""
    forest = DisjointSet()
    mst = []

    # Initialize the disjoint set with each node as its own tree
    for node in nodes:
        forest.add(node)

    # The MST will contain exactly len(nodes) - 1 edges
    edges_needed = len(nodes) - 1

    for src, dest, weight in sorted_edges:
        # Find the roots of the trees containing src and dest
        root_src = forest.find(src)
        root_dest = forest.find(dest)

        # If src and dest are in different trees, unite them and add the edge to the MST
        if root_src != root_dest:
            mst.append((src, dest, weight))
            forest.union(src, dest)
            edges_needed -= 1

            # Stop if we've added enough edges for the MST
            if edges_needed == 0:
                break

    return mst
