import networkx as nx
import matplotlib.pyplot as plt
from fibonacci_heap_mod import FibonacciHeap  # Assumed to be available
from prim_algorithm import prim  # Custom implementation assumed to be available

def build_heap(V, key_dict, heap):
    """
    Build a Fibonacci heap with initial keys for all vertices.

    Parameters:
    - V (iterable): The set of vertices.
    - key_dict (dict): A dictionary containing keys (weights) for each vertex.
    - heap (FibonacciHeap): An instance of a FibonacciHeap.

    Returns:
    - dict: A dictionary mapping each vertex to its corresponding Fibonacci heap node.
    """
    nodes_heap = {}
    for v in V:
        nodes_heap[v] = heap.insert(key_dict[v], v)
    return nodes_heap

def main():
    # Create a graph with NetworkX
    G = nx.Graph()

    # Add nodes with positions and initial key values
    node_set = ('a', 'b', 'c', 'd')
    G.add_node('a', pos=(0, 1), key=0)
    G.add_node('b', pos=(2, 1), key=0)
    G.add_node('c', pos=(1, 0), key=0)
    G.add_node('d', pos=(1, 2), key=0)

    # Add edges with weights
    G.add_edge('a', 'b', weight=2)
    G.add_edge('a', 'c', weight=30)
    G.add_edge('b', 'c', weight=90)
    G.add_edge('d', 'b', weight=1)
    G.add_edge('d', 'a', weight=3)

    # Extract weights and positions for drawing
    weight = nx.get_edge_attributes(G, 'weight')
    pos = nx.get_node_attributes(G, 'pos')

    # Apply Prim's algorithm to find the Minimum Spanning Tree (MST)
    start_node = 'a'
    MST = prim(G, weight, start_node, node_set)

    # Get edges and weights from the MST
    edgesMST = MST.edges()
    weightMST = nx.get_edge_attributes(MST, 'weight')

    # Plot the original graph
    plt.subplot(121)
    nx.draw(G, pos, with_labels=True)
    nx.draw_networkx_edge_labels(G, pos, edge_labels=weight)
    nx.draw(G, pos, edges=G.edges())

    # Plot the MST
    plt.subplot(122)
    nx.draw(MST, pos, with_labels=True)
    nx.draw_networkx_edge_labels(MST, pos, edge_labels=weightMST)
    nx.draw(MST, pos, edges=edgesMST)

    # Show the plots
    plt.show()

if __name__ == "__main__":
    main()
