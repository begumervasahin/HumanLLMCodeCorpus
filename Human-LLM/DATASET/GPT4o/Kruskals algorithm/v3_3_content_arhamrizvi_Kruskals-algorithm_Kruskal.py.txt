def swap(array, i, j):
    """
    Swap two elements in an array.
    
    Args:
        array (List): The list in which to swap elements.
        i (int): Index of the first element.
        j (int): Index of the second element.
    """
    array[i], array[j] = array[j], array[i]

def quicksort(array):
    """
    Sort an array of edges using the quicksort algorithm.
    
    Args:
        array (List[List[int]]): The array of edges to sort.
    """
    _quicksort(array, 0, len(array) - 1)

def _quicksort(array, start, end):
    """
    Auxiliary recursive function for quicksort.
    
    Args:
        array (List[List[int]]): The array to sort.
        start (int): The starting index of the array segment.
        end (int): The ending index of the array segment.
    """
    if start < end:
        partition_index = _partition(array, start, end)
        _quicksort(array, start, partition_index - 1)
        _quicksort(array, partition_index + 1, end)

def _partition(array, start, end):
    """
    Partition the array around a pivot element and return its final position.
    
    Args:
        array (List[List[int]]): The array to partition.
        start (int): The starting index of the segment.
        end (int): The ending index of the segment.
    
    Returns:
        int: The final position of the pivot element.
    """
    pivot_index = (start + end) // 2
    pivot_value = array[pivot_index][2]
    swap(array, pivot_index, end)  # Move pivot to end
    store_index = start
    
    for i in range(start, end):
        if array[i][2] < pivot_value:
            swap(array, i, store_index)
            store_index += 1
    
    swap(array, store_index, end)  # Move pivot to its final place
    return store_index

def find(x, parent):
    """
    Find the root of the set containing x with path compression.
    
    Args:
        x (int): The element to find.
        parent (List[int]): The parent array for union-find.
    
    Returns:
        int: The root of the set containing x.
    """
    if parent[x] != x:
        parent[x] = find(parent[x], parent)
    return parent[x]

def union(x, y, parent, rank):
    """
    Union the sets containing x and y using union by rank.
    
    Args:
        x (int): The first element.
        y (int): The second element.
        parent (List[int]): The parent array for union-find.
        rank (List[int]): The rank array for union-find.
    """
    root_x = find(x, parent)
    root_y = find(y, parent)
    
    if root_x != root_y:
        if rank[root_x] > rank[root_y]:
            parent[root_y] = root_x
        elif rank[root_x] < rank[root_y]:
            parent[root_x] = root_y
        else:
            parent[root_y] = root_x
            rank[root_x] += 1

def kruskal_mst(input_filename, output_filename):
    """
    Implements Kruskal's algorithm to find the Minimum Spanning Tree (MST)
    from a given graph and writes the result to a file.
    
    Args:
        input_filename (str): The path to the input file containing the graph edges.
        output_filename (str): The path to the output file to write the MST edges.
    """
    edges = _read_edges_from_file(input_filename)
    num_vertices = _find_num_vertices(edges)
    
    parent, rank = _initialize_union_find(num_vertices)
    
    quicksort(edges)
    
    mst = _build_mst(edges, parent, rank)
    
    _write_mst_to_file(mst, output_filename)

def _read_edges_from_file(filename):
    """
    Reads the edges of a graph from a file.
    
    Args:
        filename (str): The path to the file containing the edges.
    
    Returns:
        List[List[int]]: A list of edges, each represented as [u, v, w].
    """
    edges = []
    with open(filename, 'r') as file:
        for line in file:
            u, v, w = map(int, line.split())
            edges.append([u, v, w])
    return edges

def _find_num_vertices(edges):
    """
    Determines the number of vertices in the graph based on the edges.
    
    Args:
        edges (List[List[int]]): The list of edges.
    
    Returns:
        int: The number of vertices in the graph.
    """
    return max(max(u, v) for u, v, _ in edges) + 1

def _initialize_union_find(num_vertices):
    """
    Initializes the union-find data structures.
    
    Args:
        num_vertices (int): The number of vertices in the graph.
    
    Returns:
        Tuple[List[int], List[int]]: The parent and rank arrays for union-find.
    """
    parent = list(range(num_vertices))
    rank = [0] * num_vertices
    return parent, rank

def _build_mst(edges, parent, rank):
    """
    Builds the Minimum Spanning Tree (MST) using Kruskal's algorithm.
    
    Args:
        edges (List[List[int]]): The sorted list of edges.
        parent (List[int]): The parent array for union-find.
        rank (List[int]): The rank array for union-find.
    
    Returns:
        List[List[int]]: The edges in the MST.
    """
    mst = []
    for u, v, w in edges:
        if find(u, parent) != find(v, parent):
            mst.append([u, v, w])
            union(u, v, parent, rank)
    return mst

def _write_mst_to_file(mst, filename):
    """
    Writes the edges of the MST to a file.
    
    Args:
        mst (List[List[int]]): The edges in the MST.
        filename (str): The path to the output file.
    """
    with open(filename, 'w') as file:
        for u, v, w in mst:
            file.write(f"{u} {v} {w}\n")
