import matplotlib.pyplot as plt
from DynammicProgramming import *

def plot_route_with_info(reco, db, cost, time, xMedian, yMedian):
    """
    Plots the route on a graph and adds distance and time information.
    
    Parameters:
        reco: List of recommended points (indices) for the route.
        db: Dictionary or list containing coordinates (latitude, longitude) for each point.
        cost: Total distance of the route.
        time: Total time taken for the route.
        xMedian: Median value of x-coordinates (used for plotting vertical line).
        yMedian: Median value of y-coordinates (used for plotting horizontal line).
    """
    x_coords = []
    y_coords = []

    # Extract coordinates from the recommended points
    for point in reco:
        x_coords.append(db[point[0]][0])
        y_coords.append(db[point[0]][1])

    # Close the loop by appending the first point at the end
    x_coords.append(x_coords[0])
    y_coords.append(y_coords[0])

    # Plot the route
    plt.plot(x_coords, y_coords)

    # Add distance and time information as text on the plot
    plt.text(-84, -17.5, f"Distancia: {cost}")
    plt.text(-84, -19.5, f"Tiempo: {time}")

    # Plot points and mark the starting point with a red 'x'
    plt.plot(x_coords, y_coords, '.')
    plt.plot(x_coords[0], y_coords[0], 'rx')

    # Add median lines for reference
    plt.axhline(yMedian, color='black')
    plt.axvline(xMedian, color='black')

    # Display the plot
    plt.show()

# Assuming reco, db, cost, time, xMedian, yMedian are defined elsewhere
plot_route_with_info(reco, db, cost, time, xMedian, yMedian)
