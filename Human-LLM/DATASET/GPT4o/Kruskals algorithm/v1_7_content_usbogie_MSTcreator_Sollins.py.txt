import copy
from Node import Node

def run_sollins(trees):
    # Runs the algorithm on each tree and returns a list of MSTs
    MSTs = []
    for tree in trees:
        MSTs.append(run_sollins_on_tree(tree))
    return MSTs

def run_sollins_on_tree(tree):
    subtrees = []
    tree_to_mst = {}
    mst_to_tree = {}

    # Initialize subtrees: each node starts as its own subtree
    for n in tree:
        t = []
        m = copy.copy(n)
        m.adjList = {}
        tree_to_mst[n] = m
        mst_to_tree[m] = n
        t.append(m)
        subtrees.append(t)

    # Continue until there is only one subtree left
    while len(subtrees) > 1:
        subtrees = run_phase(subtrees, tree, tree_to_mst, mst_to_tree)
    
    return subtrees.pop(0)  # Return the final MST

def run_phase(subtrees, tree, tree_to_mst, mst_to_tree):
    # Perform a single phase of Sollin's algorithm
    for subtree in subtrees:
        child = None
        lowest_weight = float('inf')
        parent = None

        for n in subtree:
            tree_node = mst_to_tree[n]
            for neighbor, weight in tree_node.adjList.items():
                temp = tree_to_mst[neighbor]
                if temp not in subtree and weight < lowest_weight:
                    lowest_weight = weight
                    child = temp
                    parent = n
        
        if parent and child:
            combine_components(subtrees, subtree, parent, child, lowest_weight)
    
    return subtrees

def combine_components(subtrees, subtree, parent, child, lowest_weight):
    # Combine two components (subtrees) by connecting parent and child nodes
    child_subtree = next((sub for sub in subtrees if child in sub), None)
    
    if child_subtree:
        parent.adjList[child] = lowest_weight
        child.adjList[parent] = lowest_weight

        # Merge child_subtree into the current subtree
        for node in subtree:
            child_subtree.append(node)
        
        # Remove the current subtree as it is now merged
        subtrees.remove(subtree)
    
    return subtrees
