class Nodo:
    """Clase nodo"""
    def __init__(self, dato=None):
        """Nodo con dato, puntero al siguiente elemento y puntero a su lista"""
        self.dato = dato
        self.sig = None

    def get_dato(self):
        """Obtener dato"""
        return self.dato

    def get_sig(self):
        """Obtener nodo siguiente"""
        return self.sig

    def set_dato(self, newdato):
        """Asignar dato"""
        self.dato = newdato

    def set_sig(self, newsig):
        """Asignar elemento siguiente"""
        self.sig = newsig


class ListaLigada:
    """Clase para representar una lista ligada simple"""
    def __init__(self):
        """Crea nueva lista con puntero al inicio y al final"""
        self.head = None
        self.tail = None

    def is_empty(self):
        """Retorna True si la lista está vacía y False si no"""
        return self.head is None

    def agregar(self, item):
        """Agrega un nuevo elemento al final de la lista"""
        temp = Nodo(item)
        if self.is_empty():
            self.head = temp
            self.tail = temp
        else:
            self.tail.set_sig(temp)
            self.tail = temp

    def size(self):
        """Devuelve la cantidad de elementos en la lista"""
        actual = self.head
        count = 0
        while actual is not None:
            count += 1
            actual = actual.get_sig()
        return count

    def buscar(self, item):
        """Busca un elemento en la lista"""
        actual = self.head
        found = False
        while actual is not None and not found:
            if actual.get_dato() == item:
                found = True
            else:
                actual = actual.get_sig()
        return found

    def eliminar(self, item):
        """Elimina un elemento de la lista"""
        actual = self.head
        prev = None
        found = False
        while actual is not None and not found:
            if actual.get_dato() == item:
                found = True
            else:
                prev = actual
                actual = actual.get_sig()

        if found:
            if prev is None:
                self.head = actual.get_sig()
            else:
                prev.set_sig(actual.get_sig())

            if actual == self.tail:
                self.tail = prev

    def mostrar(self):
        """Muestra la lista"""
        actual = self.head
        while actual is not None:
            print(actual.get_dato(), end='')
            actual = actual.get_sig()
            if actual is not None:
                print(" -> ", end="")
        print()


# Ejemplo de uso
if __name__ == "__main__":
    lista = ListaLigada()
    lista.agregar(1)
    lista.agregar(2)
    lista.agregar(3)
    lista.mostrar()  # Muestra: 1 -> 2 -> 3

    print("Size:", lista.size())  # Muestra: Size: 3

    print("Buscar 2:", lista.buscar(2))  # Muestra: Buscar 2: True
    print("Buscar 5:", lista.buscar(5))  # Muestra: Buscar 5: False

    lista.eliminar(2)
    lista.mostrar()  # Muestra: 1 -> 3
