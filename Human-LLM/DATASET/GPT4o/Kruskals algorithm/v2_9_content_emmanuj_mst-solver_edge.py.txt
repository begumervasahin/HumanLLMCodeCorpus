class Edge:
    def __init__(self, start_vertex, end_vertex, weight):
        """
        Initializes an Edge object representing a connection between two vertices.

        Parameters:
        - start_vertex (int or str): The starting vertex of the edge.
        - end_vertex (int or str): The ending vertex of the edge.
        - weight (float or int): The weight or cost associated with traversing this edge.
        """
        self.start_vertex = start_vertex  # The starting vertex (u)
        self.end_vertex = end_vertex  # The ending vertex (v)
        self.weight = weight  # The weight of the edge

    def __repr__(self):
        """
        Returns a string representation of the Edge object.
        Format: "e start_vertex end_vertex weight"
        """
        return f"e {self.start_vertex} {self.end_vertex} {self.weight}"

# Example usage
if __name__ == "__main__":
    # Create an edge from vertex 1 to vertex 2 with a weight of 3.5
    edge = Edge(1, 2, 3.5)
    
    # Print the edge
    print(edge)  # Expected output: e 1 2 3.5
