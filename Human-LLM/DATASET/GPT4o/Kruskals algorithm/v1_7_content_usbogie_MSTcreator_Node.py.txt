class Node:
    def __init__(self, xloc, yloc, kval, adjList):
        """
        Initializes a Node object.

        Parameters:
        xloc (float or int): The x-coordinate of the node.
        yloc (float or int): The y-coordinate of the node.
        kval (int): A value associated with the node.
        adjList (list): A list of adjacent nodes (or other relevant data).
        """
        self.xloc = xloc  # x-coordinate
        self.yloc = yloc  # y-coordinate
        self.kval = kval  # Some value associated with the node
        self.adjList = adjList  # List of adjacent nodes

    def __eq__(self, other):
        """
        Checks if two Node objects are equal based on their coordinates and value.

        Parameters:
        other (Node): The other Node object to compare with.

        Returns:
        bool: True if both nodes have the same xloc, yloc, and kval, False otherwise.
        """
        return self.xloc == other.xloc and self.yloc == other.yloc and self.kval == other.kval

# Example usage
if __name__ == "__main__":
    # Create two nodes with the same attributes
    node1 = Node(1, 2, 3, [])
    node2 = Node(1, 2, 3, [])

    # Check if the nodes are equal
    print(node1 == node2)  # Output: True

    # Create a node with different attributes
    node3 = Node(1, 2, 4, [])

    # Check if node1 and node3 are equal
    print(node1 == node3)  # Output: False
