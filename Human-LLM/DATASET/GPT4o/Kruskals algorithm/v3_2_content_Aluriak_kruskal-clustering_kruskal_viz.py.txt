"""
This module provides functions for visualizing graphs using NetworkX and Matplotlib.
"""

import matplotlib.pyplot as plt
import networkx as nx

def visualize_graph(edges, node_labels=None, title="Graph Visualization"):
    """
    Visualizes a graph using NetworkX and Matplotlib.

    Parameters:
    - edges (list of tuples): List of edges in the graph. Each edge is represented as a tuple (node1, node2).
    - node_labels (dict, optional): Dictionary of node labels, where keys are nodes and values are labels.
    - title (str, optional): Title for the graph visualization. Defaults to "Graph Visualization".
    """
    # Create a NetworkX graph object
    G = nx.Graph()

    # Add edges to the graph
    G.add_edges_from(edges)

    # Generate layout for the graph using the Fruchterman-Reingold force-directed algorithm
    pos = nx.spring_layout(G)

    # Draw the graph with or without node labels
    draw_options = {
        'pos': pos,
        'with_labels': True,
        'node_color': 'lightblue',
        'node_size': 3000,
        'font_size': 12,
        'font_weight': 'bold'
    }
    
    if node_labels:
        draw_options['labels'] = node_labels
    
    nx.draw(G, **draw_options)

    # Draw edges
    nx.draw_networkx_edges(G, pos)

    # Add a title to the graph
    plt.title(title)

    # Display the graph visualization
    plt.show()

def main():
    """
    Example usage of visualize_graph function.
    """
    # Define a sample list of edges (node connections)
    edges = [(1, 2), (1, 3), (2, 4), (3, 4), (4, 5)]

    # Optionally, define node labels for better clarity
    node_labels = {1: 'A', 2: 'B', 3: 'C', 4: 'D', 5: 'E'}

    # Visualize the sample graph with labels and a title
    visualize_graph(edges, node_labels, title="Sample Graph Visualization")

if __name__ == "__main__":
    main()
