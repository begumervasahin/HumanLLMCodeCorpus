class Edge:
    def __init__(self, vertex1, vertex2, weight, extra=0, selected=False):
        """
        Initializes an edge between two vertices in a graph.

        Parameters:
        - vertex1: The first vertex connected by the edge.
        - vertex2: The second vertex connected by the edge.
        - weight: The weight or cost associated with traversing this edge.
        - extra (optional): An additional attribute for any extra information (default is 0).
        - selected (optional): Boolean indicating if this edge is selected in a specific context (default is False).
        """
        self.vertex1 = vertex1
        self.vertex2 = vertex2
        self.weight = weight
        self.extra = extra
        self.selected = selected

    def __repr__(self):
        """
        Returns a string representation of the edge, including vertices, weight, and selection status.
        """
        return (f"V1: {self.vertex1}  V2: {self.vertex2}  W: {self.weight}  SEL: {self.selected}")

# Example usage
if __name__ == "__main__":
    # Create an edge with vertices 1 and 2, weight 10, and mark it as selected
    edge = Edge(1, 2, 10, selected=True)
    
    # Print the edge's details
    print(edge)
