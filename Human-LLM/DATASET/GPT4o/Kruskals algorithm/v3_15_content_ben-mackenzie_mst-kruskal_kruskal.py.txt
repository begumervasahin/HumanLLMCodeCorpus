import heapq

class DisjointSet:
    def __init__(self):
        self.parent = {}
        self.rank = {}

    def make_group(self, item):
        """
        Create a new group with 'item' as its own parent.
        Initialize the rank of the item to 0.
        """
        self.parent[item] = item
        self.rank[item] = 0

    def find(self, item):
        """
        Find the root ancestor of the set containing 'item'.
        Apply path compression for efficiency.
        """
        if self.parent[item] != item:
            self.parent[item] = self.find(self.parent[item])  # Path compression
        return self.parent[item]

    def union_sets(self, item1, item2):
        """
        Union the sets containing 'item1' and 'item2'.
        Attach the smaller rank tree under the root of the higher rank tree.
        """
        root1 = self.find(item1)
        root2 = self.find(item2)
        
        if root1 != root2:
            # Union by rank
            if self.rank[root1] > self.rank[root2]:
                self.parent[root2] = root1
            else:
                self.parent[root1] = root2
                if self.rank[root1] == self.rank[root2]:
                    self.rank[root2] += 1

class HeapPriorityQueue:
    def __init__(self):
        self.heap = []

    def insert(self, item):
        """Insert an item into the priority queue."""
        heapq.heappush(self.heap, item)

    def delete_min(self):
        """Remove and return the item with the smallest value from the priority queue."""
        return heapq.heappop(self.heap)

    def is_empty(self):
        """Check if the priority queue is empty."""
        return len(self.heap) == 0

def kruskal_mst(graph):
    """
    Compute the Minimum Spanning Tree (MST) of a graph using Kruskal's algorithm.
    
    Parameters:
        graph: A dictionary with:
            'V': List of vertices.
            'E': List of edges, where each edge is a tuple (weight, vertex_u, vertex_v).
    
    Returns:
        A tuple containing:
            - A list of edges that comprise the MST.
            - The total weight of the MST.
    """
    mst = []  # List to store the edges of the MST
    pq = HeapPriorityQueue()
    disjoint_set = DisjointSet()

    # Initialize each vertex as a separate set
    for vertex in graph['V']:
        disjoint_set.make_group(vertex)

    # Insert all edges into the priority queue
    for edge in graph['E']:
        pq.insert(edge)

    total_weight = 0

    # Kruskal's algorithm to construct the MST
    while len(mst) < len(graph['V']) - 1 and not pq.is_empty():
        weight, u, v = pq.delete_min()
        root_u = disjoint_set.find(u)
        root_v = disjoint_set.find(v)

        # If u and v are in different sets, add the edge to the MST
        if root_u != root_v:
            mst.append((u, v, weight))
            total_weight += weight
            disjoint_set.union_sets(root_u, root_v)

    return mst, total_weight

# Example usage:
graph = {
    'V': ['A', 'B', 'C', 'D', 'E'],
    'E': [
        (1, 'A', 'B'),
        (2, 'B', 'C'),
        (3, 'C', 'D'),
        (4, 'D', 'E'),
        (5, 'A', 'E'),
        (6, 'B', 'D')
    ]
}

mst, total_weight = kruskal_mst(graph)
print("Minimum Spanning Tree:", mst)
print("Total Weight of MST:", total_weight)
