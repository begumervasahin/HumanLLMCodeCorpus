class Node:
    def __init__(self, xloc, yloc, kval, adj_list=None):
        """
        Initializes a Node object.

        :param xloc: X-coordinate of the node.
        :param yloc: Y-coordinate of the node.
        :param kval: Value associated with the node.
        :param adj_list: List of adjacent nodes (default is None).
        """
        self.xloc = xloc          # X-coordinate of the node
        self.yloc = yloc          # Y-coordinate of the node
        self.kval = kval          # Some value associated with the node
        self.adj_list = adj_list if adj_list is not None else []  # List of adjacent nodes

    def __eq__(self, other):
        """
        Compares this Node to another Node.

        :param other: The other node to compare to.
        :return: True if nodes are equal, False otherwise.
        """
        if not isinstance(other, Node):
            return False
        
        return (
            self.xloc == other.xloc and 
            self.yloc == other.yloc and 
            self.kval == other.kval
        )
