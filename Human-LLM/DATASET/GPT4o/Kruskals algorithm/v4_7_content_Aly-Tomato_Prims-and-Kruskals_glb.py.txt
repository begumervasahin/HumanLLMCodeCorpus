# Global data structures
WGRAPH = {}  # Weighted graph representation
MST = []  # List to store the edges of the Minimum Spanning Tree
Vr = []  # List to store vertices
PARENT = {}  # Dictionary to store the parent of each vertex in the disjoint set
RANK = {}  # Dictionary to store the rank of each vertex in the disjoint set
EDGES = []  # List to store all the edges in the graph
VERTICES = set()  # Set to store all the vertices in the graph

def print_pretty():
    """
    Prints the Minimum Spanning Tree (MST) in a formatted table.
    
    The table includes:
        - Vertex 1: The first vertex of the edge.
        - Vertex 2: The second vertex of the edge.
        - Distance: The weight of the edge.
        - Cumulative Distance: The cumulative weight up to that edge.
    """
    # Headers for the table
    header_v1 = "Vertex 1"
    header_v2 = "Vertex 2"
    header_distance = "Distance"
    header_cumulative = "Cumulative Distance"
    
    # Print the table headers
    print("******************************************************************************")
    print(f"{header_v1.ljust(15)}\t{header_v2.ljust(15)}\t{header_distance.ljust(10)}\t{header_cumulative.ljust(15)}")
    print("******************************************************************************")
    
    # Print each edge in the MST with its corresponding details
    for edge in MST:
        vertex1, vertex2, distance, cumulative_distance = edge
        print(f"{vertex1.ljust(15)}\t{vertex2.ljust(15)}\t{str(distance).ljust(10)}\t{str(cumulative_distance).ljust(15)}")
