def floyd(graph):
    """
    Implements the Floyd-Warshall algorithm to find the shortest paths between all pairs of nodes.

    Parameters:
    - graph (list of list of int): The adjacency matrix of the graph.

    Returns:
    - tuple: A tuple containing the updated graph with shortest path distances and a path dictionary.
    """
    num_nodes = len(graph)
    path = {i: {j: [i, j] for j in range(num_nodes) if i != j} for i in range(num_nodes)}

    for k in range(num_nodes):
        for i in range(num_nodes):
            for j in range(num_nodes):
                if graph[i][j] > graph[i][k] + graph[k][j]:
                    graph[i][j] = graph[i][k] + graph[k][j]
                    path[i][j].insert(-1, k)

    return graph, path

def floyd_dict(graph):
    """
    Implements the Floyd-Warshall algorithm to find the shortest paths between all pairs of nodes,
    using a dictionary representation of the graph.

    Parameters:
    - graph (dict): The adjacency list of the graph, where each key is a source node and each value 
                    is a dictionary of destination nodes and edge weights.

    Returns:
    - tuple: A tuple containing the updated graph with shortest path distances and a path dictionary.
    """
    path = {src: {dst: [src, dst] for dst in graph[src] if src != dst} for src in graph}

    for mid in graph:
        for src in graph:
            for dst in graph[src]:
                if graph[src][dst] > graph[src][mid] + graph[mid][dst]:
                    graph[src][dst] = graph[src][mid] + graph[mid][dst]
                    path[src][dst].insert(-1, mid)

    return graph, path
