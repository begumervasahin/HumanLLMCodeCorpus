class Edge:
    def __init__(self, vertex1, vertex2, weight, extra=0, selected=False):
        self.vertex1 = vertex1
        self.vertex2 = vertex2
        self.weight = weight
        self.extra = extra
        self.selected = selected

    def __repr__(self):
        return (f"V1: {self.vertex1}  V2: {self.vertex2}  W: {self.weight}  SEL: {self.selected}")

# Example usage
if __name__ == "__main__":
    edge = Edge(1, 2, 10, selected=True)
    print(edge)
