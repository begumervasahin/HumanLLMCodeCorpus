import sys

def kruskal(graph):
    """
    Kruskal's algorithm to find the Minimum Spanning Tree (MST) of a graph.
    
    Parameters:
        graph: 2D list representing the adjacency matrix of the graph
    
    Returns:
        tree: 2D list representing the adjacency matrix of the MST
    """
    num_edges_in_tree = 1
    size = len(graph)
    tree = [[0 for _ in range(size)] for _ in range(size)]
    parent = list(range(size))  # Initialize each node as its own parent

    # Find the Minimum Spanning Tree
    while num_edges_in_tree < size:
        min_weight = sys.maxsize
        node1, node2 = -1, -1

        # Find the minimum weight edge that doesn't form a cycle
        for i in range(size):
            for j in range(size):
                if graph[i][j] != 0 and graph[i][j] < min_weight and parent[i] != parent[j]:
                    min_weight = graph[i][j]
                    node1 = i
                    node2 = j

        if node1 != -1 and node2 != -1:
            # Add the edge to the MST
            tree[node1][node2] = min_weight

            # Update the parent array to reflect the union of sets
            old_parent = parent[node2]
            new_parent = parent[node1]
            for i in range(size):
                if parent[i] == old_parent:
                    parent[i] = new_parent

        num_edges_in_tree += 1

    return tree

if __name__ == '__main__':
    print("Kruskal's Algorithm for Minimum Spanning Tree")

    # Get the size of the graph (number of vertices)
    size = int(input("Enter the number of vertices in the graph: "))

    # Initialize the adjacency matrix
    graph = []
    print("Enter the adjacency matrix (row by row):")
    for i in range(size):
        graph.append(list(map(int, input().split())))

    # Find the Minimum Spanning Tree
    mst = kruskal(graph)

    # Print the resulting MST as an adjacency matrix
    print("Minimum Spanning Tree (as an adjacency matrix):")
    for row in mst:
        print(" ".join(map(str, row)))
