import numpy as np

# Define a matrix using a string with semicolons separating rows
mat = '1 2 3 4 ; 5 6 7 8 ; 0 1 2 3'

# Convert the string to a NumPy matrix
graphMat = np.matrix(mat)

# Get the shape of the matrix (rows, columns)
print("Matrix shape:", graphMat.shape)

# Access specific elements in the matrix
element_0_1 = graphMat[0, 1]  # Element at first row, second column
element_2_3 = graphMat[2, 3]  # Element at third row, fourth column

# Print the accessed elements
print("Element at (0,1):", element_0_1)
print("Element at (2,3):", element_2_3)

# Modify a specific element in the matrix
graphMat[0, 1] = -8

# Access and print the entire second row (index 1)
second_row = graphMat[1]
print("Second row:", second_row)

# Access and print the entire second column (index 1)
second_column = graphMat[:, 1]
print("Second column:", second_column)
