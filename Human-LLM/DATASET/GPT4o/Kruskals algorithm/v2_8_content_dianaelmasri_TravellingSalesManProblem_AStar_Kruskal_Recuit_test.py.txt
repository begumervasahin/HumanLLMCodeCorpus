import queue as Q

class ComparableString(str):
    """
    A string subclass that compares based on string length.
    This allows strings to be used in a priority queue with custom comparison logic.
    """
    def __lt__(self, other):
        # Compare strings by their length
        return len(self) < len(other)

def main():
    # Create a priority queue
    priority_queue = Q.PriorityQueue()

    # Insert strings into the priority queue
    priority_queue.put(ComparableString('jeej'))
    priority_queue.put(ComparableString('kek'))
    priority_queue.put(ComparableString('topkek'))
    priority_queue.put(ComparableString('non'))

    # Extract and print the smallest item based on string length
    # Since priority_queue.queue is a list, we can sort it to get the smallest item
    smallest_string = sorted(priority_queue.queue)[0]
    print(smallest_string)

if __name__ == "__main__":
    main()
