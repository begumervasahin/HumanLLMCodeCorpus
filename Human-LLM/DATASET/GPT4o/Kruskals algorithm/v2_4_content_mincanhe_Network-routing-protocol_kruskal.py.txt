class DisjointSet:
    def __init__(self):
        self.parent = {}
        self.rank = {}

    def make_set(self, vertice):
        """
        Initialize the set for each vertex.
        Each vertex is its own parent initially, and the rank is set to 0.
        """
        self.parent[vertice] = vertice
        self.rank[vertice] = 0

    def find(self, vertice):
        """
        Find the root of the set containing the vertex with path compression.
        
        Parameters:
            vertice: The vertex whose root is to be found.
        
        Returns:
            The root vertex of the set containing the given vertex.
        """
        if self.parent[vertice] != vertice:
            self.parent[vertice] = self.find(self.parent[vertice])  # Path compression
        return self.parent[vertice]

    def union(self, vertice1, vertice2):
        """
        Unite the sets that contain vertice1 and vertice2 using union by rank.
        
        Parameters:
            vertice1: The first vertex.
            vertice2: The second vertex.
        """
        root1 = self.find(vertice1)
        root2 = self.find(vertice2)

        if root1 != root2:
            # Union by rank
            if self.rank[root1] < self.rank[root2]:
                self.parent[root1] = root2
            else:
                self.parent[root2] = root1
                if self.rank[root1] == self.rank[root2]:
                    self.rank[root1] += 1

def kruskal(graph):
    """
    Implement Kruskal's algorithm to find the maximum bandwidth path.
    
    Parameters:
        graph: A dictionary with 'vertices' as the list of vertices and 'edges' as the list of edges,
               where each edge is represented as a tuple (weight, vertice1, vertice2).
    
    Returns:
        A set of edges that form the maximum bandwidth path.
    """
    disjoint_set = DisjointSet()
    
    for vertice in graph['vertices']:
        disjoint_set.make_set(vertice)
    
    max_bandwidth_path = set()
    edges = sorted(graph['edges'], reverse=True)  # Sort edges by weight in descending order

    for edge in edges:
        weight, vertice1, vertice2 = edge
        if disjoint_set.find(vertice1) != disjoint_set.find(vertice2):
            disjoint_set.union(vertice1, vertice2)
            max_bandwidth_path.add(edge)
    
    return max_bandwidth_path
