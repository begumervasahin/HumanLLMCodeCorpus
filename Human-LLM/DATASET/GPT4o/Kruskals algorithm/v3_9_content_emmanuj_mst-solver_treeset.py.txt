class Node:
    def __init__(self, value):
        self.value = value
        self.parent = self  # Each node is initially its own parent
        self.rank = 0  # Rank is used to keep the tree balanced

class DisjointSet:
    def __init__(self):
        self.nodes = {}  # Dictionary to store the nodes

    def make_set(self, value):
        """
        Create a new set containing the single element 'value'.
        
        Parameters:
            value: The value of the new set element.
        """
        self.nodes[value] = Node(value)

    def find(self, value):
        """
        Find the representative (root) of the set containing 'value'.
        Implements path compression to optimize future queries.
        
        Parameters:
            value: The element whose set representative is to be found.
        
        Returns:
            The value of the representative element (root) of the set.
        """
        node = self.nodes[value]
        if node.parent != node:
            node.parent = self.nodes[self.find(node.parent.value)]
        return node.parent.value

    def union(self, value1, value2):
        """
        Unite the sets containing 'value1' and 'value2'.
        Uses union by rank to maintain a balanced tree structure.
        
        Parameters:
            value1: The first element.
            value2: The second element.
        
        Returns:
            The root of the united set.
        """
        root1 = self.nodes[self.find(value1)]
        root2 = self.nodes[self.find(value2)]

        if root1.value != root2.value:
            if root1.rank > root2.rank:
                root2.parent = root1
                return root1
            elif root1.rank < root2.rank:
                root1.parent = root2
                return root2
            else:
                root2.parent = root1
                root1.rank += 1
                return root1

    def print_structure(self):
        """
        Print the internal structure of the DisjointSet, showing each node,
        its parent, and its rank.
        """
        print("Disjoint Set Structure:")
        for value, node in self.nodes.items():
            print(f"Node: {value}, Parent: {node.parent.value}, Rank: {node.rank}")

if __name__ == "__main__":
    # Example usage:
    ds = DisjointSet()
    elements = [1, 2, 3, 4]

    # Create sets for each element
    for element in elements:
        ds.make_set(element)

    print("Initial sets:")
    ds.print_structure()

    # Perform union operations
    ds.union(1, 2)
    ds.union(3, 4)
    ds.union(2, 3)

    print("\nAfter some unions:")
    ds.print_structure()

    # Find the representative of element 4
    print("\nFind representative of element 4:", ds.find(4))
