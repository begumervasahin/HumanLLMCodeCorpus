# Color definitions
COLORS = [
    ("or", (255, 210, 0)),
    ("argent", (192, 192, 192)),
    ("azure", (0, 0, 139)),
    ("gules", (139, 0, 0)),

# Function to get the message based on dimension and delta
def get_msg(dim, delta):
    """Retrieve a message based on the dimension and delta value."""
    return MSGS[dim].get(delta)

# Function to get the color by index
def get_color(idx):
    """Retrieve the RGB color tuple based on the index."""
    return COLORS[idx][1]
