def floyd(graph):
    """
    Implements the Floyd-Warshall algorithm for finding the shortest paths between all pairs of nodes.

    Parameters:
    graph (list of list of int): The adjacency matrix of the graph.

    Returns:
    tuple: A tuple containing the updated graph with shortest path distances and a path dictionary.
    """
    length = len(graph)
    path = {}

    for i in range(length):
        path.setdefault(i, {})
        for j in range(length):
            if i == j:
                continue
            path[i].setdefault(j, [i, j])
            new_node = None

            for k in range(length):
                if k == j:
                    continue
                new_len = graph[i][k] + graph[k][j]
                if graph[i][j] > new_len:
                    graph[i][j] = new_len
                    new_node = k

            if new_node:
                path[i][j].insert(-1, new_node)

    return graph, path

def floyd_dict(graph):
    """
    Implements the Floyd-Warshall algorithm for finding the shortest paths between all pairs of nodes,
    using a dictionary representation of the graph.

    Parameters:
    graph (dict): The adjacency list of the graph, where each key is a source node and each value is 
                  a dictionary of destination nodes and edge weights.

    Returns:
    tuple: A tuple containing the updated graph with shortest path distances and a path dictionary.
    """
    path = {}

    for src in graph:
        path.setdefault(src, {})
        for dst in graph[src]:
            if src == dst:
                continue
            path[src].setdefault(dst, [src, dst])
            new_node = None

            for mid in graph:
                if mid == dst:
                    continue
                new_len = graph[src][mid] + graph[mid][dst]
                if graph[src][dst] > new_len:
                    graph[src][dst] = new_len
                    new_node = mid

            if new_node:
                path[src][dst].insert(-1, new_node)

    return graph, path
