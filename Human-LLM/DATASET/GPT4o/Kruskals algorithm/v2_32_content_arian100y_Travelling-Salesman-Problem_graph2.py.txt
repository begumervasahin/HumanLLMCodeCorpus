import matplotlib.pyplot as plt

# Example data: replace these with your actual data
db = {
    0: (-84.1, -17.6),
    1: (-83.9, -17.4),
    2: (-83.8, -17.3),
    # Add more points as necessary
}
reco = [(0, 1), (1, 2), (2, 0)]  # Example path; replace with your actual path data
cost = 100  # Example distance; replace with your actual distance value
time = 10   # Example time; replace with your actual time value
y_median = -17.5  # Example y median; replace with your actual y median value
x_median = -84.0  # Example x median; replace with your actual x median value

# Initialize lists to store the x and y coordinates of the path
x_coords = []
y_coords = []

# Collect coordinates based on the path indices (reco)
for start, end in reco:
    x_coords.append(db[start][0])
    y_coords.append(db[start][1])

# Closing the path by adding the starting point at the end
x_coords.append(x_coords[0])
y_coords.append(y_coords[0])

# Plot the path
plt.plot(x_coords, y_coords, label='Path')

# Annotate the plot with distance and time information
plt.text(x_median - 0.2, y_median - 0.2, f"Distancia: {cost}")
plt.text(x_median - 0.2, y_median - 0.4, f"Tiempo: {time}")

# Plot the points
plt.plot(x_coords, y_coords, 'o', label='Points')
plt.plot(x_coords[0], y_coords[0], 'rx', label='Start Point')  # Highlight the starting point

# Plot median lines for x and y
plt.axhline(y_median, color='black', linestyle='--', label='Y Median')
plt.axvline(x_median, color='black', linestyle='--', label='X Median')

# Add labels and a legend
plt.xlabel('Longitude')
plt.ylabel('Latitude')
plt.legend()

# Display the plot
plt.show()
