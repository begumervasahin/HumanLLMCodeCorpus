def root(vertex):
    """
    Finds the root of the given vertex with path compression.
    """
    while root_ids[vertex] != vertex:
        root_ids[vertex] = root_ids[root_ids[vertex]]  # Path compression
        vertex = root_ids[vertex]
    return vertex

def union(u, v):
    """
    Unites the sets containing u and v by linking their roots.
    """
    u_root = root(u)
    v_root = root(v)
    root_ids[v_root] = u_root  # Link root of v to root of u

def kruskal(graph_edges):
    """
    Implements Kruskal's algorithm to find the minimum spanning tree.
    Returns the total weight of the minimum spanning tree.
    """
    global root_ids
    min_weight = 0
    
    # Process each edge in the sorted edge list
    for edge in graph_edges:
        u, v, weight = edge
        if root(u) != root(v):
            min_weight += weight
            union(u, v)
    
    return min_weight

# Example input
V, E = map(int, input("Enter number of vertices and edges: ").split())
graph_edges = []

# Read edges
for i in range(E):
    u, v, weight = map(int, input("Enter edge (u v weight): ").split())
    graph_edges.append([u, v, weight])

# Sort edges by weight
graph_edges = sorted(graph_edges, key=lambda x: x[2])

# Initialize root ids for each vertex
root_ids = list(range(V + 1))

# Compute and print the total weight of the MST
print("Total weight of Minimum Spanning Tree:", kruskal(graph_edges))
