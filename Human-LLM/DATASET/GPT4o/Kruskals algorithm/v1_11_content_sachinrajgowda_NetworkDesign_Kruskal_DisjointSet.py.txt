class DisjointSet:
    def __init__(self):
        self.parent = {}

    def add(self, item):
        # Initially, the parent of the item is the item itself
        self.parent[item] = item

    def find(self, item):
        # Find the root of the item and perform path compression
        if self.parent[item] != item:
            self.parent[item] = self.find(self.parent[item])
        return self.parent[item]

    def union(self, item1, item2):
        # Union the sets of item1 and item2
        root1 = self.find(item1)
        root2 = self.find(item2)
        if root1 != root2:
            self.parent[root2] = root1

def kruskal(nodes, sorted_edges):
    forest = DisjointSet()
    mst = []
    
    # Initialize the disjoint set with each node
    for node in nodes:
        forest.add(node)
    
    no_edges = len(nodes) - 1
    
    # Iterate over the sorted edges
    for edge in sorted_edges:
        src, dest, weight = edge
        root1 = forest.find(src)
        root2 = forest.find(dest)
        
        # If the roots are different, no cycle is formed
        if root1 != root2:
            mst.append(edge)
            no_edges -= 1
            forest.union(root1, root2)
            
            # If we've found enough edges for MST, we can stop
            if no_edges == 0:
                break
    
    return mst

# Example usage:
nodes = ['A', 'B', 'C', 'D', 'E']
sorted_edges = [
    ('A', 'B', 1),
    ('B', 'C', 2),
    ('C', 'D', 3),
    ('D', 'E', 4),
    ('A', 'E', 5),
    ('B', 'D', 6)
]

mst = kruskal(nodes, sorted_edges)
print("Minimum Spanning Tree:", mst)
