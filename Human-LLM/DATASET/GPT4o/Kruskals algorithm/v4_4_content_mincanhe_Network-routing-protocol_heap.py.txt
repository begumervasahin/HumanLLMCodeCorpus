def max_heapify(A, i):
    """
    Maintains the max-heap property for the heap subtree rooted at index i.

    Parameters:
        A: List representing the heap.
        i: Index of the root of the subtree to be heapified.
    """
    n = len(A)
    left = 2 * i + 1  # Left child index (adjusted for zero-based indexing)
    right = 2 * i + 2  # Right child index (adjusted for zero-based indexing)
    largest = i

    # Compare left child with root
    if left < n and A[left] > A[i]:
        largest = left

    # Compare right child with the current largest
    if right < n and A[right] > A[largest]:
        largest = right

    # If the largest is not the root, swap and continue heapifying
    if largest != i:
        A[i], A[largest] = A[largest], A[i]  # Swap
        max_heapify(A, largest)  # Recursively heapify the affected subtree

def build_max_heap(A):
    """
    Converts an unsorted array into a max-heap.

    Parameters:
        A: List to be converted into a max-heap.
    """
    n = len(A)
    # Start from the last non-leaf node and heapify each node
    for i in range(n // 2 - 1, -1, -1):
        max_heapify(A, i)

def heapsort(A):
    """
    Sorts an array in ascending order using the heapsort algorithm.

    Parameters:
        A: List to be sorted.
    """
    build_max_heap(A)
    n = len(A)
    # Extract elements one by one from the heap
    for i in range(n - 1, 0, -1):
        A[0], A[i] = A[i], A[0]  # Swap the root (max element) with the last element
        max_heapify(A[:i], 0)  # Heapify the reduced heap

def heap_maximum(A):
    """
    Returns the maximum element in the heap.

    Parameters:
        A: List representing the heap.

    Returns:
        The maximum element of the heap.
    """
    return A[0]

def heap_insert(A, key):
    """
    Inserts a new key into the heap.

    Parameters:
        A: List representing the heap.
        key: The new element to be inserted.
    """
    A.append(key)  # Add the new element to the end of the heap
    i = len(A) - 1
    # Fix the max-heap property if it is violated
    while i > 0 and A[(i - 1) // 2] < A[i]:
        A[i], A[(i - 1) // 2] = A[(i - 1) // 2], A[i]
        i = (i - 1) // 2

def heap_delete(A, i):
    """
    Deletes an element from the heap at a given index.

    Parameters:
        A: List representing the heap.
        i: Index of the element to be deleted.
    """
    A[i] = A[-1]  # Replace the element to be deleted with the last element
    A.pop()  # Remove the last element
    max_heapify(A, i)  # Heapify the affected subtree
