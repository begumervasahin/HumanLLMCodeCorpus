def heapify(arr, n, i):
    """
    Maintains the heap property for a subtree rooted at index i in a max heap.

    Parameters:
    - arr (list): The heap (or array) to be heapified.
    - n (int): The size of the heap.
    - i (int): The index of the root of the subtree.
    """
    largest = i  # Assume the root is the largest element
    left = 2 * i + 1  # Left child index
    right = 2 * i + 2  # Right child index

    # If the left child exists and is larger than the root
    if left < n and arr[left] > arr[largest]:
        largest = left

    # If the right child exists and is larger than the largest so far
    if right < n and arr[right] > arr[largest]:
        largest = right

    # If the largest element is not the root
    if largest != i:
        arr[i], arr[largest] = arr[largest], arr[i]  # Swap the root with the largest
        heapify(arr, n, largest)  # Recursively heapify the affected subtree

def heap_sort(arr):
    """
    Sorts a list using the heap sort algorithm.

    Parameters:
    - arr (list): The list to be sorted.
    """
    n = len(arr)

    # Build a max heap from the list
    for i in range(n // 2 - 1, -1, -1):
        heapify(arr, n, i)

    # Extract elements one by one from the heap and rebuild the heap
    for i in range(n - 1, 0, -1):
        arr[i], arr[0] = arr[0], arr[i]  # Move current root to end
        heapify(arr, i, 0)  # Call heapify on the reduced heap

# Example usage
if __name__ == "__main__":
    arr = [12, 11, 13, 5, 6, 7]
    print("Original array:", arr)
    heap_sort(arr)
    print("Sorted array:", arr)
