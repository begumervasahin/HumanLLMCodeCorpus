def max_heapify(A, i):
    """
    Ensure the subtree rooted at index 'i' is a max-heap.
    
    Parameters:
        A: List representing the heap (1-based index).
        i: Index of the node to heapify.
    """
    n = len(A) - 1  # Adjust for 1-based index assumption
    left = 2 * i
    right = 2 * i + 1
    largest = i

    if left <= n and A[left] > A[i]:
        largest = left
    if right <= n and A[right] > A[largest]:
        largest = right

    if largest != i:
        A[i], A[largest] = A[largest], A[i]  # Swap with the largest child
        max_heapify(A, largest)

def build_max_heap(A):
    """
    Convert the list 'A' into a max-heap.
    
    Parameters:
        A: List to be transformed into a max-heap (1-based index).
    """
    n = len(A) - 1  # Adjust for 1-based index assumption
    for i in range(n // 2, 1 - 1, -1):
        max_heapify(A, i)

def heapsort(A):
    """
    Perform heap sort on the list 'A'.
    
    Parameters:
        A: List to be sorted (1-based index).
    """
    build_max_heap(A)
    n = len(A) - 1  # Adjust for 1-based index assumption
    for i in range(n, 1, -1):
        A[1], A[i] = A[i], A[1]  # Swap the first and last elements
        max_heapify(A[:i], 1)  # Heapify the reduced heap

def heap_maximum(A):
    """
    Return the maximum element in the heap.
    
    Parameters:
        A: List representing the heap (1-based index).
    
    Returns:
        The maximum element, which is at index 1.
    """
    return A[1]

def heap_insert(A, value):
    """
    Insert a new element into the max-heap.
    
    Parameters:
        A: List representing the heap (1-based index).
        value: The value to insert into the heap.
    """
    A.append(value)
    i = len(A) - 1
    while i > 1 and A[i // 2] < A[i]:
        A[i], A[i // 2] = A[i // 2], A[i]  # Swap with the parent
        i = i // 2

def heap_delete(A, i):
    """
    Delete the element at index 'i' from the heap.
    
    Parameters:
        A: List representing the heap (1-based index).
        i: The index of the element to delete.
    """
    n = len(A) - 1
    A[i] = A[n]
    A.pop()  # Remove the last element
    max_heapify(A, i)

def print_heap(A, message=""):
    """
    Print the heap list in a readable format (1-based index).
    
    Parameters:
        A: List representing the heap (1-based index).
        message: Optional message to print before the heap.
    """
    if message:
        print(message)
    print(A[1:])

if __name__ == "__main__":
    # Example usage of heap operations
    A = [0, 4, 10, 3, 5, 1]  # 0 is a placeholder to facilitate 1-based indexing
    print_heap(A, "Original array:")

    build_max_heap(A)
    print_heap(A, "Max-heap:")

    heapsort(A)
    print_heap(A, "Sorted array:")

    # Reset the array and perform insert and delete operations
    A = [0, 4, 10, 3, 5, 1]
    heap_insert(A, 6)
    print_heap(A, "After inserting 6:")

    heap_delete(A, 2)
    print_heap(A, "After deleting element at index 2:")

    print("Maximum element:", heap_maximum(A))
