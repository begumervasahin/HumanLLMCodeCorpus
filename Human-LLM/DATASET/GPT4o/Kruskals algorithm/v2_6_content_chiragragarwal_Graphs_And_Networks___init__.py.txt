# Import necessary modules for graph reading and plotting
from PlotGraph import PlotGraph
from ReadGraph import ReadGraph

def main():
    # Instantiate ReadGraph to handle the reading of graph data
    graph_reader = ReadGraph()

    # Read the graph data from a file or other data source
    graph = graph_reader.read('graph_data.txt')  # Replace 'graph_data.txt' with your actual data source

    # Instantiate PlotGraph to handle graph visualization
    graph_plotter = PlotGraph()

    # Plot the graph using the data read by ReadGraph
    graph_plotter.plot(graph)

if __name__ == "__main__":
    main()
