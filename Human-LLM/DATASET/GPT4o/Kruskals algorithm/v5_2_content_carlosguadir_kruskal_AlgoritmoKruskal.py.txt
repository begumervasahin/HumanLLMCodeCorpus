from Grafo import Grafo

class AlgoritmoKruskal:
    def aplicar_kruskal(self, grafo):
        """
        Applies Kruskal's algorithm to find the Minimum Spanning Tree (MST) of the given graph.

        Parameters:
        - grafo (Grafo): The graph on which Kruskal's algorithm is applied.

        Returns:
        - Grafo: The Minimum Spanning Tree (MST) of the graph.
        """
        arbol = Grafo()  # Create an empty graph to store the MST
        nodos = grafo.getNombres()  # Get the list of node names from the graph

        # Add all nodes to the MST (without edges initially)
        for nodo in nodos:
            arbol.ingresarNodo(nodo)

        aristas = grafo.getAristas()  # Get the list of edges (aristas) from the graph
        aristas.sort(key=lambda arco: arco.getPeso())  # Sort edges by weight for Kruskal's algorithm

        # Process each edge to build the MST
        for arco in aristas:
            nodo_inicial = arbol.getNodo(arco.getInicial())
            nodo_terminal = arbol.getNodo(arco.getTerminal())

            # Check if adding this edge creates a cycle
            if not self.hay_ciclo(arbol, arco, nodo_terminal, nodo_inicial.getNombre()):
                # If no cycle is formed, add the edge to the MST
                arbol.adicionarEnlace(nodo_inicial.getNombre(), nodo_terminal.getNombre(), arco.getPeso())

        return arbol

    def hay_ciclo(self, grafo, arco_verificar, terminal, origen):
        """
        Checks if adding an edge to the graph will create a cycle.

        Parameters:
        - grafo (Grafo): The graph to check for cycles.
        - arco_verificar: The edge to verify.
        - terminal: The current node being checked.
        - origen: The original node from which the check started.

        Returns:
        - bool: True if adding the edge creates a cycle, False otherwise.
        """
        enlaces = terminal.getEnlaces()  # Get the list of edges connected to the terminal node

        # Base case: If the terminal node has no edges, no cycle can be formed
        if not enlaces:
            return False

        # Check if the terminal node directly connects back to the origin node
        if terminal.existeEnlace(origen) != -1:
            return True

        # Recursively check each connected node to detect a cycle
        for enlace in enlaces:
            nodo_destino = enlace.getDestino()

            # Avoid revisiting the node we came from
            if nodo_destino != origen:
                if self.hay_ciclo(grafo, arco_verificar, grafo.getNodo(nodo_destino), terminal.getNombre()):
                    return True

        return False

    # Example usage of AlgoritmoKruskal
    grafo = Grafo()
    # Populate the graph with nodes and edges
    # ...

    algoritmo_kruskal = AlgoritmoKruskal()
    mst = algoritmo_kruskal.aplicar_kruskal(grafo)
    print("Minimum Spanning Tree:", mst)
