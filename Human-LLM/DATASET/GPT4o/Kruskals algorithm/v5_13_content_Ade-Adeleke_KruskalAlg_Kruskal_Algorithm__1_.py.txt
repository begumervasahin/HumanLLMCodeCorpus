class DisjointSet:
    def __init__(self):
        self.parent = {}

    def make_set(self, vertex):
        """Initialize the set for a given vertex."""
        self.parent[vertex] = vertex

    def find_set(self, vertex):
        """Find the representative of the set containing the given vertex with path compression."""
        if self.parent[vertex] != vertex:
            self.parent[vertex] = self.find_set(self.parent[vertex])
        return self.parent[vertex]

    def union(self, u, v):
        """Union the sets containing vertices u and v."""
        root_u = self.find_set(u)
        root_v = self.find_set(v)
        if root_u != root_v:
            self.parent[root_u] = root_v


def kruskal(graph):
    """Kruskal's algorithm to find the Minimum Spanning Tree (MST)."""
    disjoint_set = DisjointSet()
    mst = set()  # Set to store the edges of the MST
    
    # Initialize each vertex as its own set
    for vertex in graph['V']:
        disjoint_set.make_set(vertex)
    
    # Sort edges by their weights
    sorted_edges = sorted(graph['E'])
    
    # Process each edge in the sorted order
    for weight, u, v in sorted_edges:
        # If u and v are in different sets, add the edge to the MST
        if disjoint_set.find_set(u) != disjoint_set.find_set(v):
            mst.add((weight, u, v))
            disjoint_set.union(u, v)
    
    return mst
