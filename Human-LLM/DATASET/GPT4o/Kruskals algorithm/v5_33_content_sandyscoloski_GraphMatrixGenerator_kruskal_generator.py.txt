def generate_kruskal_input(matrix, output_filename='kruskal.txt'):
    """
    Generates a list of edges for Kruskal's algorithm from an adjacency matrix and writes them to a file.
    
    Parameters:
        matrix (list of list of int): A 2D list representing the adjacency matrix of a graph.
        output_filename (str): The name of the output file to write the edges to (default is 'kruskal.txt').
    """
    with open(output_filename, 'w') as output_file:
        for row in range(len(matrix)):
            for col in range(row + 1, len(matrix[row])):  # Only consider the upper triangle
                weight = matrix[row][col]
                if weight != 0:
                    output_file.write(f'{row} {col} {weight}\n')

def read_matrix_from_file(filename):
    """
    Reads an adjacency matrix from a file and returns it as a 2D list.
    
    Parameters:
        filename (str): The name of the file containing the adjacency matrix.
    
    Returns:
        list of list of int: A 2D list representing the adjacency matrix of a graph.
    """
    with open(filename, 'r') as file:
        lines = file.readlines()
        matrix = [list(map(int, line.split())) for line in lines]
    
    return matrix

def main():
    # Read the adjacency matrix from 'matrix.txt'
    input_filename = 'matrix.txt'
    adjacency_matrix = read_matrix_from_file(input_filename)
    
    # Generate the Kruskal input file 'kruskal.txt' from the adjacency matrix
    output_filename = 'kruskal.txt'
    generate_kruskal_input(adjacency_matrix, output_filename)
    print(f"Kruskal input file '{output_filename}' has been generated successfully.")

if __name__ == '__main__':
    main()
