import heapq
import glb

def read_graph(file, delimiter):
    """
    Reads a graph from a file and populates the global EDGES and VERTICES sets.

    Parameters:
    file (str): The path to the file containing the graph.
    delimiter (str): The delimiter separating the vertices and weight in the file.

    Returns:
    list: A list of edges in the format (weight, vertex1, vertex2).
    """
    with open(file, 'r') as file_path:
        for line in file_path:
            # Split the line by the delimiter and strip any whitespace
            v1, v2, weight = [x.strip() for x in line.split(delimiter)]
            weight = int(weight)

            # Avoid adding duplicate edges in the reverse direction
            if (weight, v2, v1) in glb.EDGES:
                continue

            # Add the edge to the heap and the vertices to the set
            heapq.heappush(glb.EDGES, (weight, v1, v2))
            glb.VERTICES.add(v1)
            glb.VERTICES.add(v2)

    return glb.EDGES

def make_set(vertex):
    """
    Initializes the set for the vertex in the disjoint-set data structure.

    Parameters:
    vertex: The vertex to initialize in the disjoint-set.
    """
    glb.PARENT[vertex] = vertex
    glb.RANK[vertex] = 0

def find(vertex):
    """
    Finds the representative of the set containing the vertex.

    Parameters:
    vertex: The vertex to find the representative for.

    Returns:
    The representative of the set containing the vertex.
    """
    if glb.PARENT[vertex] != vertex:
        glb.PARENT[vertex] = find(glb.PARENT[vertex])
    return glb.PARENT[vertex]

def union(vertex1, vertex2):
    """
    Unites the sets containing vertex1 and vertex2.

    Parameters:
    vertex1: The first vertex.
    vertex2: The second vertex.
    """
    root1 = find(vertex1)
    root2 = find(vertex2)

    if root1 != root2:
        # Attach the shorter tree under the root of the taller tree
        if glb.RANK[root1] > glb.RANK[root2]:
            glb.PARENT[root2] = root1
        else:
            glb.PARENT[root1] = root2
            if glb.RANK[root1] == glb.RANK[root2]:
                glb.RANK[root2] += 1

def kruskals():
    """
    Implements Kruskal's algorithm to find the Minimum Spanning Tree (MST) of a graph.

    Returns:
    tuple: A tuple containing the MST and the total cost of the MST.
    """
    cdist = 0

    # Initialize the disjoint sets for each vertex
    for vertex in glb.VERTICES:
        make_set(vertex)

    # Process the edges in the order of increasing weight
    while glb.EDGES:
        weight, v1, v2 = heapq.heappop(glb.EDGES)

        # If v1 and v2 are in different sets, include the edge in the MST
        if find(v1) != find(v2):
            union(v1, v2)
            cdist += weight
            glb.MST.append((v1, v2, str(weight), str(cdist)))

    return glb.MST, cdist
