from osgeo import gdal
import numpy as np
from scipy.interpolate import interp1d
import argparse

def perform_color_balance(reference_image, adjust_image, output_image):
    """
    Perform color balance using CDF matching.
    
    Parameters:
    reference_image (str): Path to the reference image.
    adjust_image (str): Path to the image to be adjusted.
    output_image (str): Path to the output image.
    """
    
    ref = gdal.Open(reference_image)
    adj = gdal.Open(adjust_image)
    
    if ref.RasterCount != adj.RasterCount:
        print("Images have different number of bands.")
        return
    
    num_bands = ref.RasterCount
    
    ref_histograms = [ref.GetRasterBand(b+1).GetHistogram() for b in range(num_bands)]
    ref_cdfs = [np.cumsum(hist) / float(sum(hist)) for hist in ref_histograms]
    
    adj_histograms = [adj.GetRasterBand(b+1).GetHistogram() for b in range(num_bands)]
    adj_cdfs = [np.cumsum(hist) / float(sum(hist)) for hist in adj_histograms]
    
    inv_cdf_functions = [interp1d(cdf, np.arange(len(cdf)), bounds_error=False, fill_value="extrapolate") for cdf in ref_cdfs]
    transfer_bands = [inv_cdf_functions[b](adj_cdfs[b]) for b in range(num_bands)]
    
    transfer_functions = [lambda x, t=transfer_bands[b]: t[x] for b in range(num_bands)]
    
    driver = gdal.GetDriverByName('GTiff')
    output_dataset = driver.CreateCopy(output_image, adj, 0, ['COMPRESS=LZW'])
    
    for b in range(num_bands):
        adjusted_band = transfer_functions[b](adj.GetRasterBand(b+1).ReadAsArray().astype(int))
        output_dataset.GetRasterBand(b+1).WriteArray(adjusted_band)
    
    ref, adj, output_dataset = None, None, None

def main():
    parser = argparse.ArgumentParser(description='Perform histogram matching between reference image and image to be adjusted.')
    parser.add_argument("reference_image", help="Path to the reference image.")
    parser.add_argument("adjust_image", help="Path to the image to be adjusted.")
    parser.add_argument("output_image", help="Path to the output image.")
    args = parser.parse_args()
    
    perform_color_balance(args.reference_image, args.adjust_image, args.output_image)

if __name__ == '__main__':
    main()
