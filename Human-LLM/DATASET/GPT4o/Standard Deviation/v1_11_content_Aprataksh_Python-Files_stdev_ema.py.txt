import csv
import pandas as pd
import statistics
import os

def get_ticker_list(filepath):
    ticker_list = []
    with open(filepath, 'r') as f:
        lines = csv.reader(f)
        for line in lines:
            if "Symbol" not in line:
                ticker_list.append(line[2])
    return ticker_list

def process_stock_data(ticker_list, input_dir, output_dir, ndays=9):
    stdev_list = []

    for ticker in ticker_list:
        if '&' in ticker:
            ticker = ticker.replace('&', '%26')

        input_file = os.path.join(input_dir, ticker + ".csv")
        output_file = os.path.join(output_dir, ticker + ".csv")

        if not os.path.exists(input_file):
            stdev_list.append(0)
            continue

        df = pd.read_csv(input_file)

        if len(df.columns) >= 6 and len(df.index) >= ndays:
            SMA = pd.Series(round(df['Close'].rolling(window=ndays).mean(), 2), name='SMA')
            df = df.join(SMA)
            SMA_list = df['SMA'].tolist()
            SD = round(statistics.stdev(SMA_list[ndays-1:]), 2)
            stdev_list.append(SD)

            EMA = pd.Series(round(df['Close'].ewm(span=15, min_periods=ndays).mean(), 2), name="EMA")
            df = df.join(EMA)
            df.to_csv(output_file, index=False)
        else:
            stdev_list.append(0)

    return stdev_list

def save_stdev_list(ticker_list, stdev_list, output_file):
    sdf = pd.DataFrame(stdev_list, index=pd.Series(ticker_list, name="Ticker"), columns=["Standard Deviation"])
    sdf.to_csv(output_file)

def main():
    ticker_file = r"C:\Users\Hp\Desktop\New folder\nifty500_list.csv"
    input_dir = r"C:\Users\Hp\Desktop\New folder\Top500_stock\Original data"
    output_dir = r"C:\Users\Hp\Desktop\New folder\Top500_stock\With SMA and EMA"
    stdev_output_file = r"C:\Users\Hp\Desktop\New folder\st_dev.csv"

    ticker_list = get_ticker_list(ticker_file)
    stdev_list = process_stock_data(ticker_list, input_dir, output_dir)
    save_stdev_list(ticker_list, stdev_list, stdev_output_file)

if __name__ == "__main__":
    main()
