import argparse
import os
import ntpath
import glob

# Set up argument parser
parser = argparse.ArgumentParser(description='Process gzipped samtools mpileup output.')
parser.add_argument("bams", help="Text file containing BAM files, one per line.")
parser.add_argument("dir", help="Directory containing *.mpileup.gz files.")
parser.add_argument("-v", "--verbose", help="Increase output verbosity", action="store_true")
args = parser.parse_args()

def read_bam_file(bam_file):
    """Read the list of BAM files from a text file."""
    with open(bam_file, 'r') as f:
        bam_files = f.readlines()
    return [bam.strip() for bam in bam_files]

def clean_file_names(file_names):
    """Remove path and extension from filenames."""
    cleaned_names = [os.path.splitext(ntpath.basename(file_name))[0] for file_name in file_names]
    return cleaned_names

def get_mpileup_files(directory):
    """Get a list of *.mpileup.gz files from the specified directory."""
    return glob.glob(os.path.join(directory, "*.mpileup.gz"))

# Verbose logging if enabled
if args.verbose:
    print("Verbosity turned on")
    print(f"Input BAM files: {args.bams}")
    print(f"Input directory: {args.dir}")

# Read BAM files
bam_files = read_bam_file(args.bams)
if args.verbose:
    print(f"Number of BAM files: {len(bam_files)}")
    if bam_files:
        print(f"First BAM file: {bam_files[0]}")

# Clean BAM filenames
sample_names = clean_file_names(bam_files)
if args.verbose:
    print("Sample names:")
    for name in sample_names:
        print(name)

# Get mpileup files from the directory
mpileup_files = get_mpileup_files(args.dir)

# Clean mpileup filenames
locus_names = clean_file_names(mpileup_files)
if args.verbose:
    print(f"Number of mpileup files: {len(mpileup_files)}")

# Print mpileup files with their corresponding cleaned names
for mp_file, locus_name in zip(mpileup_files, locus_names):
    print(f"{mp_file}  {locus_name}")

# Initialize list to hold non-zero values
nonzeros = [[] for _ in range(len(sample_names))]
print(nonzeros)
