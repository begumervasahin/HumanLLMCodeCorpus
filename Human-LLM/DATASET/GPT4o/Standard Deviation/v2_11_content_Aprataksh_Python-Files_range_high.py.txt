import csv
import numpy as np
import math
import pandas as pd
import xlsxwriter

def resistance(ticker):
    file_path = f"C:/Users/Hp/Desktop/New folder/Top500_stock/Original data/{ticker}.csv"
    df = pd.read_csv(file_path)
    
    if len(df.columns) < 6:
        return

    high_data = df['High'].tolist()
    max_value = math.ceil(max(high_data))
    min_value = math.floor(min(high_data))
    interval = np.around(np.linspace(min_value, max_value, no_of_intervals + 1), 2).tolist()
    
    interval_data = []
    highest_frequency = 0
    highest_interval_start = min_value

    for start in interval[:-1]:
        end = interval[interval.index(start) + 1]
        count = sum(start <= price < end for price in high_data)
        
        if count > highest_frequency:
            highest_frequency = count
            highest_interval_start = start
            
        interval_data.append((start, end, count))
    
    interval_data.append((highest_interval_start, interval[interval.index(highest_interval_start) + 1], highest_frequency))
    intervals_data.append(interval_data)
    new_ticker_list.append(ticker)

def get_ticker_list():
    file_path = r"C:/Users/Hp/Desktop/New folder/nifty500_list.csv"
    with open(file_path, 'r') as f:
        reader = csv.reader(f)
        for row in reader:
            if "Symbol" not in row:
                ticker_list.append(row[2])

def main():
    get_ticker_list()
    
    for ticker in ticker_list:
        ticker = ticker.replace('&', '%26') if '&' in ticker else ticker
        print(ticker)
        resistance(ticker)
    
    data_frame = pd.DataFrame(intervals_data, index=pd.Series(new_ticker_list, name="SYMBOL"), columns=columns)
    
    # Saving to CSV
    data_frame.to_csv(r"C:/Users/Hp/Desktop/New folder/Top500_stock/resistance.csv")
    
    # Saving to XLSX
    with pd.ExcelWriter(r"C:/Users/Hp/Desktop/New folder/Top500_stock/resistance.xlsx", engine="xlsxwriter") as writer:
        data_frame.to_excel(writer, sheet_name="Sheet1")

if __name__ == "__main__":
    ticker_list = []
    intervals_data = []
    no_of_intervals = 10
    columns = [f"INTERVAL NO. {i}" for i in range(1, no_of_intervals + 1)]
