import requests
from bs4 import BeautifulSoup as BS
from pathlib import Path
from datetime import datetime, timedelta
import csv

# Number of historical days to retrieve
n_hist_days = 31

def fetch_exchange_rates(day_str):
    """
    Fetches exchange rates for a given day from x-rates.com.

    Args:
        day_str (str): The date string in the format 'YYYY-MM-DD'.

    Returns:
        list: A list of exchange rates for the given day.
    """
    url = f'https://www.x-rates.com/historical/?from=USD&amount=1&date={day_str}'
    response = requests.get(url)
    day_html = response.text.split("<span class='ratesTableAlpha'>Alphabetical order</span>")[1]
    soup = BS(day_html, "html.parser")
    tds = soup.find_all('td')
    rates = [td.text for td in tds[2::3]]
    return rates

def fetch_currency_names():
    """
    Fetches currency names from x-rates.com.

    Returns:
        list: A list of currency names.
    """
    url = 'https://www.x-rates.com/table/?from=USD&amount=1'
    response = requests.get(url)
    html = response.text.split("<span class='ratesTableAlpha'>Alphabetical order</span>")[1]
    soup = BS(html, "html.parser")
    tds = soup.find_all('td')
    names = [td.text for td in tds[::3]]
    return names

def generate_csv(n_hist_days):
    """
    Generates a CSV file with historical exchange rates for the past n_hist_days.

    Args:
        n_hist_days (int): Number of historical days to retrieve.
    """
    csv_data = []

    # Fetch and append currency names as the header row
    currency_names = fetch_currency_names()
    csv_data.append(currency_names)

    # Fetch and append exchange rates for each day
    for i in range(n_hist_days, 0, -1):
        day = datetime.today() - timedelta(days=i)
        day_str = day.strftime('%Y-%m-%d')
        day_rates = fetch_exchange_rates(day_str)
        csv_data.append(day_rates)

    # Write the collected data to a CSV file
    with open("data/hist_rates.csv", "w", newline='') as f:
        writer = csv.writer(f)
        writer.writerows(csv_data)

    print(f"Historical data obtained for the last {n_hist_days} days.")

def daily_update():
    """
    Updates the CSV file with today's exchange rates.
    """
    config = Path('data/hist_rates.csv')
    
    if not config.is_file():
        print("Historical data not found, obtaining from x-rates.com...")
        generate_csv(n_hist_days)
    
    today_str = datetime.today().strftime('%Y-%m-%d')
    today_rates = fetch_exchange_rates(today_str)
    
    # Append today's rates to the CSV file
    with open('data/hist_rates.csv', 'a', newline='') as f:
        writer = csv.writer(f)
        writer.writerow(today_rates)
    
    print(f"Today's currency rates ({today_str}) obtained...")

# Execute daily update
daily_update()
