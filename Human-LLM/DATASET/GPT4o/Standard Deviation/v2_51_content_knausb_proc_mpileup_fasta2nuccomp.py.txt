import argparse
from Bio import SeqIO

def parse_arguments():
    """
    Set up and parse command-line arguments.
    """
    parser = argparse.ArgumentParser(description='Determine nucleotide composition of FASTA files.')
    parser.add_argument("FASTA", help="FASTA file containing nucleotide sequences.")
    parser.add_argument("-v", "--verbose", help="increase output verbosity", action="store_true")
    return parser.parse_args()

def verbose_print(message, verbose):
    """
    Print a message if verbosity is enabled.
    """
    if verbose:
        print(message)

def calculate_nucleotide_composition(fasta_file, verbose):
    """
    Calculate nucleotide composition of each sequence in a FASTA file.
    """
    output_file = f"{fasta_file.split('.')[0]}_nuccomp.csv"
    verbose_print(f"Input FASTA file: {fasta_file}", verbose)
    verbose_print(f"Output file: {output_file}", verbose)
    
    # Open output file and write header
    with open(output_file, 'w') as f:
        f.write('Id,Length,a,A,c,C,g,G,t,T,w,W,s,S,m,M,k,K,r,R,y,Y,n,N\n')
        
        # Parse FASTA file and compute nucleotide composition
        for record in SeqIO.parse(fasta_file, "fasta"):
            counts = {
                'a': record.seq.count("a"), 'A': record.seq.count("A"),
                'c': record.seq.count("c"), 'C': record.seq.count("C"),
                'g': record.seq.count("g"), 'G': record.seq.count("G"),
                't': record.seq.count("t"), 'T': record.seq.count("T"),
                'w': record.seq.count("w"), 'W': record.seq.count("W"),
                's': record.seq.count("s"), 'S': record.seq.count("S"),
                'm': record.seq.count("m"), 'M': record.seq.count("M"),
                'k': record.seq.count("k"), 'K': record.seq.count("K"),
                'r': record.seq.count("r"), 'R': record.seq.count("R"),
                'y': record.seq.count("y"), 'Y': record.seq.count("Y"),
                'n': record.seq.count("n"), 'N': record.seq.count("N")
            }
            f.write(f"{record.id},{len(record)},")
            f.write(','.join(str(counts[nuc]) for nuc in counts))
            f.write('\n')

def main():
    args = parse_arguments()
    if args.verbose:
        print("Verbosity turned on")
    calculate_nucleotide_composition(args.FASTA, args.verbose)

if __name__ == "__main__":
    main()
