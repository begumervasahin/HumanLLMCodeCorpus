import numpy as np
import random
import matplotlib.pyplot as plt

# Set the random seed for reproducibility
random.seed(0)

# Define the number of runs to test
run_tests = [10, 100, 1000, 10000, 100000]

# Initialize lists to store means and standard deviations
means = []
stds = []

# Define the scale for the simulation
scale = 0.1

def main():
    # Count the number of rows in the 'abalone.data' file
    with open('abalone.data') as file:
        num_rows = sum(1 for _ in file)
    
    # Perform simulations for each number of runs
    for num_runs in run_tests:
        # Initialize data_in with zeros for the current number of rows
        data_in = [0] * num_rows
        
        # Run the simulation num_runs times
        for _ in range(num_runs):
            n_n = int(num_rows * scale)  # Number of positives to assign
            n_r = num_rows               # Total number of rows remaining
            
            # Assign positives to data_in
            for j in range(num_rows):
                if n_n / n_r > random.random():
                    data_in[j] += 1
                    n_n -= 1
                n_r -= 1
        
        # Calculate mean and standard deviation for the current run
        means.append(np.mean(data_in) / num_runs)
        stds.append(np.std(data_in) / num_runs)
    
    # Plot the results
    plt.figure(figsize=(10, 8))
    
    plt.subplot(2, 1, 1)
    plt.plot(run_tests, means, marker='o')
    plt.title("Mean vs. Runs")
    plt.ylabel("Mean")
    
    plt.subplot(2, 1, 2)
    plt.plot(run_tests, stds, marker='o')
    plt.title("Standard Deviation vs. Runs")
    plt.xlabel("Number of Runs")
    plt.ylabel("Standard Deviation")
    
    plt.tight_layout()
    plt.show()

# Run the main function
if __name__ == "__main__":
    main()
