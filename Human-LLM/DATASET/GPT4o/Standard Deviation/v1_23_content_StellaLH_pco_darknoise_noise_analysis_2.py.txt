import numpy as np
import h5py

frames = [100, 1000, 10000]
exp_time = [0.001]
pix_rate = ['slow', 'fast']

for i in frames:
    for j in exp_time:
        for k in pix_rate:
            print(i, j, k)
            mean = []
            std = []
            mnm = []
            mxm = []
            chunks = i // 100
            num = ["%06d" % a for a in range(1, chunks + 1)]
            
            for l in num:
                with h5py.File(f'{i}_frames_{k}_{l}.h5', 'r') as fid:
                    data = fid['/entry/data']
                    mean.append(np.mean(data, axis=0))
                    std.append(np.std(data, axis=0))
                    mnm.append(np.min(data, axis=0))
                    mxm.append(np.max(data, axis=0))
            
            grand_mean = np.mean(mean, axis=0)
            grand_std = np.sqrt(np.sum([s**2 for s in std]) / chunks)
            grand_mnm = np.min(mnm, axis=0)
            grand_mxm = np.max(mxm, axis=0)
            
            np.save(f'mean_{i}frames_{j}s_{k}.npy', grand_mean)
            np.save(f'std_{i}frames_{j}s_{k}.npy', grand_std)
            np.save(f'min_{i}frames_{j}s_{k}.npy', grand_mnm)
            np.save(f'max_{i}frames_{j}s_{k}.npy', grand_mxm)
