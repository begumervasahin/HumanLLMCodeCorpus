import csv

# Initialize an empty list to store data
data = []

# Open the output CSV file for writing
with open('rain2016.csv', 'w+', newline='') as file_out:
    writer = csv.writer(file_out)
    
    # Write the header row to the output file
    header = ["Station_ID", "Latitude", "Longitude", "Time", "Date", "Top Latitude", "Bottom Latitude",
              "Left Longitude", "Right Longitude", "TIR1_Temperature(K)", "WV_Radiance", "Elevation(m)",
              "Vegetation", "Mean3(K)", "Std3(K)", "Mean5(K)", "Std5(K)", "Air_Temp(°C)", "Wind_Speed(m/s)",
              "Wind_Direction(deg)", "Atmo_Pressure(hpa)", "Humidity(%)", "Rain_Fall(mm)"]
    writer.writerow(header)
    
    # Open the input CSV file for reading
    with open('final_rain2016_sorted.csv', 'r') as input_file:
        reader = csv.reader(input_file, delimiter=',')
        
        # Skip the header row in the input file
        next(reader)
        
        # Read data from the input file and store it in the data list
        for row in reader:
            data.append(row)
        
        # Process each row in the data list
        for row in data:
            row[13] = round(float(row[13]), 3)
            
            # Filter and write relevant rows to the output file
            if float(row[12]) != -9999.0 and row[13] != -49.995:
                filtered_row = [row[0], row[1], row[2], row[7], row[8], row[3], row[4], row[5],
                                row[6], row[10], row[11], row[12], row[13], row[14], row[15],
                                row[16], row[17], row[18], row[19], row[20], row[21], row[22],
                                row[23]]
                writer.writerow(filtered_row)
