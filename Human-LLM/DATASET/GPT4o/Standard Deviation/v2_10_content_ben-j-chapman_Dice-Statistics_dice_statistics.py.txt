import random
import math

def roll(sides):
    """Simulate rolling a die with a given number of sides."""
    return random.randint(1, sides)

def form_sample(times, sides, data_list):
    """Roll the die a specified number of times and store the results in data_list."""
    for _ in range(times):
        data_list.append(roll(sides))

def advantage(list_of_lists, list_length):
    """Determine the highest value roll from each position across multiple lists of rolls."""
    results_list = [1] * list_length
    for element in list_of_lists:
        for i in range(list_length):
            if element[i] > results_list[i]:
                results_list[i] = element[i]
    return results_list

def create_rolls(num_dice, sides, num_rolls, results_list):
    """Create rolls for multiple dice and store the results in results_list."""
    for _ in range(num_dice):
        temp_rolls = []
        form_sample(num_rolls, sides, temp_rolls)
        results_list.append(temp_rolls)

def average(data_list):
    """Calculate the average of a list of numbers."""
    return sum(data_list) / len(data_list)

def standard_deviation(data_list, mean):
    """Calculate the standard deviation of a list of numbers."""
    return sum(math.fabs(i - mean) for i in data_list) / len(data_list)

def main():
    """Main function to run the dice rolling simulation."""
    print('This program simulates the rolling of multiple dice, where the highest value is always taken.')
    print('Given a number of sides and number of rolls, it can return the average roll and standard deviation in rolls.\n')
    
    number_rolls = int(input('Number of rolls: '))
    dice_sides = int(input('Number of sides: '))
    sample_size = int(input('Sample size: '))
    print('\n')
    
    group_of_rolls = []
    create_rolls(number_rolls, dice_sides, sample_size, group_of_rolls)
    
    final_sample = advantage(group_of_rolls, sample_size)
    mu = average(final_sample)
    sigma = standard_deviation(final_sample, mu)
    
    print(f'Average roll: {mu}')
    print(f'Standard deviation: {sigma}\n')

if __name__ == "__main__":
    main()
