from obst_algorithms import *
from timeit import default_timer as timer
from platform_info import *

def print_main(output_file, all_trees, n_max, trees2, second_best, total_tree_count, comparisons, total_sum,
               file_name, p, avgs, std_devs, start):
    
    def print_optimal_trees(trees, probabilities, output):
        """Prints the first 5 optimal trees."""
        for idx, tree in enumerate(trees[:5]):
            print_tree(tree, probabilities, idx, output)

    def print_second_best_tree(trees2, probabilities, n_max, output):
        """Builds and prints the second-best tree."""
        build_trees(1, n_max, trees2, second_best, 1, n_max, total_tree_count, comparisons)
        all_trees2 = organize_trees(trees2, n_max)
        print("\n\nTree: Second Best (Sub-Optimal Tree)", file=output)
        for tree in all_trees2:
            print_tree(tree, probabilities, 0, output)
        return get_std_deviation(all_trees2, total_sum, n_max, probabilities)
    
    def print_tree_info(trees, avgs, std_devs, file_name, n_max, output):
        """Prints information about the optimal and second-best trees."""
        print("\n\nIMPORTANT TREE INFORMATION\n\nOptimal Tree(s): ", len(trees), file=output)
        for idx, tree in enumerate(trees[:5], start=1):
            print_table_info(file_name, n_max, avgs[idx-1], std_devs[idx-1], tree[-1][0], idx, output)

    def print_computation_time(start, output):
        """Prints the total computation time in seconds and minutes."""
        end = timer()
        total_time_seconds = end - start
        total_time_minutes = total_time_seconds / 60
        print(f"Total Computation Time: {total_time_seconds:4.2f} seconds", file=output)
        print(f"Total Computation Time: {total_time_minutes:4.2f} minutes", file=output)

    # Start by printing a header for the output
    print("\n\n", file=output_file)
    print("LEVEL-ORDER TRAVERSAL   (key)->{L: Level, R: Root, P: Probability/Frequency}", file=output_file)
    print("Tree(s): Optimal Tree(s)", file=output_file)
    
    # Print the optimal trees
    print_optimal_trees(all_trees, p, output_file)

    # If only one optimal tree, also print the second-best tree
    if len(all_trees) == 1:
        avgs2, std_devs2 = print_second_best_tree(trees2, p, n_max, output_file)
    else:
        avgs2, std_devs2 = None, None

    # Print information about the trees
    print_tree_info(all_trees, avgs, std_devs, file_name, n_max, output_file)
    if len(all_trees) == 1:
        print("\nSecond Best Tree (Sub-Optimal)", file=output_file)
        print_table_info(file_name, n_max, avgs2[0], std_devs2[0], all_trees2[0][-1][0], 1, output_file)

    # Print the total number of comparisons and computation time
    print("\nTotal Comparisons:", comparisons[0], file=output_file)
    print_computation_time(start, output_file)

    # Print platform information
    get_platform_info(output_file)
    print("\n\n\n", file=output_file)
