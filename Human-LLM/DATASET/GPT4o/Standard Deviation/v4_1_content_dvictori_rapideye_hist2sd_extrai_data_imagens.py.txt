Program to generate a shapefile of RapidEye tiles with the date of the images.
Note: This program does not handle scenes with more than one date.
Warning: The code contains fixed paths and positions.
Ensure to verify and adjust them before running.
@author: m330625
"""

import glob
from osgeo import ogr

# Define the directory containing the images and the path to the RapidEye tiles shapefile
image_directory = 'Z:\\filtro_radiometrico_2014_16bits\\*.tif'
tile_shapefile = 'C:\\geodb\\articulacao\\RapidEye_Tiles\\RapidEye_Tiles_Brasil.shp'

# Retrieve the list of image files
image_files = glob.glob(image_directory)

def extract_image_dates(image_files, tile_shapefile, output_shapefile):
    """
    Extracts the date from the image filenames and adds it to the RapidEye tile shapefile.
    
    Args:
        image_files (list): List of image file paths.
        tile_shapefile (str): Path to the RapidEye tile shapefile.
        output_shapefile (str): Path to the output shapefile with dates added.
        
    Warning:
        The position of the date in the filename is fixed.
    """
    
    # Open the tile shapefile
    input_data = ogr.Open(tile_shapefile)
    driver = ogr.GetDriverByName("ESRI Shapefile")
    
    # Copy the input shapefile to the output shapefile
    output_data = driver.CopyDataSource(input_data, output_shapefile)
    input_data = None
    
    # Get the layer from the output shapefile
    layer = output_data.GetLayer()
    
    # Add a new field to store the date
    layer.CreateField(ogr.FieldDefn('Date', ogr.OFTDate))
    
    # Define the fixed positions for extracting date and tile ID from the filename
    date_start_pos = 48
    tile_id_start_pos = 40
    
    # Loop through each image file
    for image_file in image_files:
        # Extract the date from the filename
        date_str = image_file[date_start_pos:date_start_pos + 10]
        date_parts = date_str.split('-')
        
        # Extract the tile ID from the filename
        tile_id = image_file[tile_id_start_pos:tile_id_start_pos + 7]
        
        # Set the attribute filter to select the correct tile
        layer.SetAttributeFilter(f'TILE_ID = {int(tile_id)}')
        
        # Loop through the selected features and update the date field
        for feature in layer:
            feature.SetField('Date', int(date_parts[0]), int(date_parts[1]), int(date_parts[2]), 0, 0, 0, 0)
            layer.SetFeature(feature)
    
    # Close the output shapefile
    output_data = None

# Run the function with the specified paths
extract_image_dates(image_files, tile_shapefile, 'I:\\articulacao_rapideye_2014.shp')
