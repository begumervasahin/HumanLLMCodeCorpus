import numpy as np
import math
import matplotlib.pyplot as plt

# Parameters
n = 150            # Number of data points
fvar = 0.1         # Fluctuation variance
m = 4              # Number of segments
r = 1              # Number of signals
pi = math.pi       # Value of pi
lim = 1.5          # Plot limit

# Initialize data
datos = np.round((m - 1) * np.random.rand(r, n))
a1bar = np.zeros(r)
a1var = np.zeros(r)
b1bar = np.zeros(r)
b1var = np.zeros(r)
datos_desv_estandar = np.zeros(r)

# Process each signal
for i in range(r):
    # Compute angles and magnitudes
    angles = datos[i, :] * 2 * pi / m
    mag = fvar * np.random.normal(0, 1, n) + 1
    a = mag * np.cos(angles + fvar * np.random.normal(0, 1, n))
    b = mag * np.sin(angles + fvar * np.random.normal(0, 1, n))
    
    # Plot the original points
    plt.plot(a, b, 'ro')
    plt.axis([-lim, lim, -lim, lim])
    plt.title('Original Points')
    plt.show()
    
    # Filter points within the specified range
    filter_mask = (0.5 < a) & (a < 1.5) & (b < 0.5) & (b > -0.5)
    a1 = a[filter_mask]
    b1 = b[filter_mask]
    
    # Plot the filtered points
    plt.plot(a1, b1, 'ro')
    plt.axis([-lim, lim, -lim, lim])
    plt.title('Filtered Points')
    plt.show()
    
    # Calculate statistics for the filtered points
    a1bar[i] = np.mean(a1)
    a1var[i] = np.var(a1, ddof=1)
    b1bar[i] = np.mean(b1)
    b1var[i] = np.var(b1, ddof=1)
    datos_desv_estandar[i] = np.sqrt(a1var[i] + b1var[i])
    
    # Print variance and standard deviation
    print(f"a1var[{i}]: {a1var[i]}")
    print(f"b1var[{i}]: {b1var[i]}")
    print(f"Standard deviation[{i}]: {datos_desv_estandar[i]}")

# Find the signal with the minimum standard deviation
resultado = np.argmin(datos_desv_estandar)
print('Best signal for the sample is signal:', resultado)
print('Signal data:', datos[resultado, :])
