import random
import math

def roll_die(sides):
    """Simulate rolling a die with a given number of sides."""
    return random.randint(1, sides)

def generate_rolls(times, sides):
    """Generate a list of rolls for a given number of times and sides."""
    return [roll_die(sides) for _ in range(times)]

def calculate_advantage(rolls_matrix):
    """Calculate the highest value for each roll position from multiple lists of rolls."""
    return [max(rolls) for rolls in zip(*rolls_matrix)]

def create_dice_rolls(num_dice, sides, num_rolls):
    """Create multiple sets of dice rolls."""
    return [generate_rolls(num_rolls, sides) for _ in range(num_dice)]

def calculate_average(data_list):
    """Calculate the average of a list of numbers."""
    return sum(data_list) / len(data_list)

def calculate_standard_deviation(data_list, mean):
    """Calculate the standard deviation of a list of numbers."""
    return sum(abs(x - mean) for x in data_list) / len(data_list)

def main():
    """Main function to run the dice rolling simulation."""
    print('This program simulates the rolling of multiple dice, where the highest value is always taken.')
    print('Given a number of sides and number of rolls, it can return the average roll and standard deviation in rolls.\n')
    
    number_rolls = int(input('Number of rolls: '))
    dice_sides = int(input('Number of sides: '))
    sample_size = int(input('Sample size: '))
    print('\n')
    
    rolls_matrix = create_dice_rolls(number_rolls, dice_sides, sample_size)
    
    final_rolls = calculate_advantage(rolls_matrix)
    average_roll = calculate_average(final_rolls)
    std_deviation = calculate_standard_deviation(final_rolls, average_roll)
    
    print(f'Average roll: {average_roll}')
    print(f'Standard deviation: {std_deviation}\n')

if __name__ == "__main__":
    main()
