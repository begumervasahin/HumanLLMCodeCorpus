import pandas as pd
from dateutil.parser import parse
import csv

# Load the data from the CSV file
df = pd.read_csv('final_rain_2014.csv')

# Print the first row of the dataframe for verification
print(df.loc[0])

# Initialize a list to keep track of missing data indices
missing_indices = []

# Loop through the dataframe to identify missing data points
for i in range(1, len(df)):
    current_date = parse(df.loc[i]['DATE(GMT)'])
    previous_date = parse(df.loc[i-1]['DATE(GMT)'])
    
    if df.loc[i]['@STATION_ID'] == df.loc[i-1]['@STATION_ID']:
        # Check if the dates are in the same month
        if current_date.month == previous_date.month:
            # Check if the days are the same
            if current_date.day == previous_date.day:
                # Check if the time difference is greater than 1 hour
                if int(df.loc[i]['TIME(GMT)']) - int(df.loc[i-1]['TIME(GMT)']) > 1:
                    missing_indices.append(i)
            else:
                # Check for time reset at the end of the day
                if not (int(df.loc[i]['TIME(GMT)']) == 0 and int(df.loc[i-1]['TIME(GMT)']) == 23):
                    missing_indices.append(i)
        else:
            # Handle month transitions
            if previous_date.month in [7, 8]:
                if current_date.day == 1 and previous_date.day == 31:
                    if not (int(df.loc[i]['TIME(GMT)']) == 0 and int(df.loc[i-1]['TIME(GMT)']) == 23):
                        missing_indices.append(i)
                else:
                    missing_indices.append(i)
            else:
                if current_date.day == 1 and previous_date.day == 30:
                    if not (int(df.loc[i]['TIME(GMT)']) == 0 and int(df.loc[i-1]['TIME(GMT)']) == 23):
                        missing_indices.append(i)
                else:
                    missing_indices.append(i)

# Write the non-missing data points to a new CSV file
with open('RAIN.csv', 'w', newline='') as file_out:
    writer = csv.writer(file_out)
    # Write the header row
    writer.writerow(['@STATION_ID', 'LATITUDE', 'LONGITUDE', 'TIME(GMT)', 'DATE(GMT)', 'RAIN_FALL(mm)'])
    
    # Write the non-missing rows
    for i in range(len(df)):
        if i not in missing_indices:
            row = df.loc[i][['@STATION_ID', 'LATITUDE', 'LONGITUDE', 'TIME(GMT)', 'DATE(GMT)', 'RAIN_FALL(mm)']].tolist()
            writer.writerow(row)
