import os

class Utilities:
    @staticmethod
    def clear():
        # Clear the console screen based on the operating system
        if os.name == 'nt':
            os.system('cls')
        else:
            os.system('clear')

    @staticmethod
    def create_ticker(stocks):
        while True:
            ticker = input('Enter ticker: ').strip()
            stock_filtered = [stock for stock in stocks if stock.ticker == ticker]
            if ticker and not stock_filtered:
                return ticker
            print('Empty or repeated value. Try again.')

    @staticmethod
    def create_percentage():
        while True:
            try:
                percentage = int(input('Enter percentage: '))
                if 0 <= percentage <= 100:
                    return percentage
                print('Invalid percentage. Try again.')
            except ValueError:
                print('Invalid input. Please enter a number.')

    @staticmethod
    def create_number_of_actions():
        while True:
            try:
                actions = int(input('Enter number of actions: '))
                return actions
            except ValueError:
                print('Invalid input. Please enter a number.')
