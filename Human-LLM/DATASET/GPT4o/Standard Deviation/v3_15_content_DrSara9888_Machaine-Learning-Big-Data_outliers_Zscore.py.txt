import numpy as np
import pandas as pd
from matplotlib import pyplot as plt
import seaborn as sns
from scipy.stats import norm

# Load the Bengaluru house price data
data_path = "/Users/khadeejah/Desktop/Bengaluru_House_Data.csv"
data = pd.read_csv(data_path)

# Display the first few rows of the dataset
print("Initial data preview:")
print(data.head())

# Function to plot histogram of house prices
def plot_histogram(data, title, xlabel='Price', ylabel='Count', bins=20, density=False, log_scale=False):
    plt.figure(figsize=(10, 5))
    plt.hist(data, bins=bins, rwidth=0.8, density=density)
    plt.xlabel(xlabel)
    plt.ylabel(ylabel)
    if log_scale:
        plt.yscale('log')
    plt.title(title)
    plt.show()

# Plot the initial distribution of house prices
plot_histogram(data.price, 'Initial Distribution of House Prices', log_scale=True)

# Define the quantile thresholds to identify outliers
max_limit = data["price"].quantile(0.999)
min_limit = data["price"].quantile(0.0001)

# Filter out the outliers based on quantile thresholds
data_filtered = data[(data["price"] > min_limit) & (data["price"] < max_limit)]

# Print the number of outliers removed
num_outliers_removed = data.shape[0] - data_filtered.shape[0]
print(f"Removed {num_outliers_removed} outliers based on quantiles")

# Plot the distribution of house prices after initial outlier removal
plt.figure(figsize=(20, 8))

plt.subplot(1, 2, 1)
plot_histogram(data_filtered.price, 'Distribution of House Prices After Quantile Filtering', log_scale=True)

plt.subplot(1, 2, 2)
sns.histplot(data_filtered.price, bins=20, kde=True)
plt.xlabel('Price')
plt.ylabel('Density')
plt.title('KDE Plot of House Prices After Quantile Filtering')

plt.tight_layout()
plt.show()

# Plot the histogram with normal distribution fit
plt.figure(figsize=(10, 5))
plt.hist(data_filtered.price, bins=20, rwidth=0.8, density=True)
plt.xlabel('Price')
plt.ylabel('Density')
plt.title('Histogram with Normal Distribution Fit')
rng = np.arange(-1000, data_filtered.price.max(), 100)
plt.plot(rng, norm.pdf(rng, data_filtered.price.mean(), data_filtered.price.std()))
plt.show()

# Calculate and print the mean and standard deviation of the filtered prices
mean_price = data_filtered.price.mean()
std_price = data_filtered.price.std()
print(f"Mean price: {mean_price}")
print(f"Standard deviation of price: {std_price}")

# Further remove extreme values using 4 standard deviations from the mean
upper_limit = mean_price + 4 * std_price
lower_limit = mean_price - 4 * std_price
data_filtered_2 = data_filtered[(data_filtered.price < upper_limit) & (data_filtered.price > lower_limit)]

# Print the number of outliers removed using the standard deviation method
num_outliers_removed_sd = data_filtered.shape[0] - data_filtered_2.shape[0]
print(f"Removed {num_outliers_removed_sd} outliers using standard deviation method")

# Plot the histogram of house prices after further outlier removal
plt.figure(figsize=(10, 5))
plt.hist(data_filtered_2.price, bins=20, rwidth=0.8, density=True)
plt.xlabel('Price')
plt.ylabel('Density')
plt.title('Histogram After Standard Deviation Filtering')
rng = np.arange(-600, data_filtered_2.price.max(), 100)
plt.plot(rng, norm.pdf(rng, data_filtered_2.price.mean(), data_filtered_2.price.std()))
plt.show()

# Calculate z-scores for further outlier detection
data_filtered["zscore"] = (data_filtered.price - mean_price) / std_price

# Filter data based on z-scores
data_filtered_z = data_filtered[(data_filtered.zscore > -4) & (data_filtered.zscore < 4)]

# Print the shape comparison of the datasets
print(f"Original data shape: {data.shape}")
print(f"Data shape after initial filtering: {data_filtered.shape}")
print(f"Data shape after standard deviation filtering: {data_filtered_2.shape}")
print(f"Data shape after z-score filtering: {data_filtered_z.shape}")

# Print the number of outliers removed using the z-score method
num_outliers_removed_zscore = data_filtered.shape[0] - data_filtered_z.shape[0]
print(f"Removed {num_outliers_removed_zscore} outliers using z-score method")
