import sys
import math

def update_aggregate(existing_aggregate, new_value):
    count, mean, M2 = existing_aggregate
    count += 1
    delta = new_value - mean
    mean += delta / count
    delta2 = new_value - mean
    M2 += delta * delta2
    return count, mean, M2

def finalize_aggregate(existing_aggregate):
    count, mean, M2 = existing_aggregate
    if count < 2:
        return mean, float('nan'), float('nan')
    variance = M2 / count
    sample_variance = M2 / (count - 1)
    return mean, variance, sample_variance

def print_results(key, aggregate):
    mean, P_variance, S_variance = finalize_aggregate(aggregate)
    S_deviation = math.sqrt(P_variance)
    print(f'{key}\t{S_deviation:.2f}')

# Initialize the aggregate tuple
aggregate = [0, 0, 0]
current_key = None

# Process each line from standard input
for line in sys.stdin:
    key, new_value = line.strip().split('\t')
    new_value = float(new_value)
    
    if current_key and key != current_key:
        # Finalize and print the results for the previous key
        print_results(current_key, aggregate)
        
        # Reset the aggregate for the new key
        current_key = key
        aggregate = [0, 0, 0]
    
    # Update the aggregate with the new value
    aggregate = update_aggregate(aggregate, new_value)
    current_key = key

# Finalize and print the results for the last key
if current_key:
    print_results(current_key, aggregate)
