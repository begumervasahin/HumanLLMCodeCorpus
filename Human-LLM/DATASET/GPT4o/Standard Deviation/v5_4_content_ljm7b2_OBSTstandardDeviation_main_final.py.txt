from output_methods import *
from memory_profiling import *
from timeit import default_timer as timer

file_name = "CS404FS15ProjectOBSTData6.dat"

def main():
    # Read the problem data from the file
    prob = read_file(file_name)
    
    for p in prob:
        n_max = len(p)
        total_sum = sum(p)

        # Initialize matrices and variables
        C, W, R, S = ([[0] * n_max for _ in range(n_max)] for _ in range(4))
        node_dict = {}
        second_best = {}
        comparisons = [0]
        trees = []
        total_tree_count = 0

        # Open output file in append mode
        with open("output.txt", "a") as output_file:
            start = timer()

            # Compute weights, costs, and roots
            compute_w_c_r(p, n_max, C, W, R, S, node_dict, second_best, comparisons)

            # Build trees up to 5 times
            for _ in range(5):
                build_trees(1, n_max, trees, node_dict, 1, n_max, total_tree_count, comparisons)

            # Organize and analyze the built trees
            all_trees = organize_trees(trees, n_max)
            avgs, std_devs = get_std_deviation(all_trees, total_sum, n_max, p)

            # Print results to the output file
            print_main(output_file, all_trees, n_max, [], second_best, total_tree_count, comparisons, total_sum,
                       file_name, p, avgs, std_devs, start)

    print("Complete, Your output is ready in file: output.txt")
    print("NOTE: This file is set to 'APPEND', multiple sets will be appended.")

if __name__ == "__main__":
    main()
