import os
import csv
import numpy as np
import statistics
from sklearn.linear_model import LinearRegression
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from joblib import Parallel, delayed
import multiprocessing

def train_and_evaluate_model(random_state, features, targets):
    """
    Train and evaluate a Linear Regression model.

    Parameters:
    - random_state: The random state for splitting the data.
    - features: The input features for the model.
    - targets: The target variable for the model.

    Returns:
    - accuracy: The accuracy of the model on the test set.
    """
    # Split the dataset into training and testing sets
    X_train, X_test, y_train, y_test = train_test_split(features, targets, test_size=0.2, random_state=random_state)
    
    # Convert data to the appropriate types
    X_train = np.array(X_train, dtype=float)
    X_test = np.array(X_test, dtype=float)
    y_train = np.array(y_train, dtype=int)
    y_test = np.array(y_test, dtype=int)
    
    # Initialize and train the Linear Regression model
    model = LinearRegression()
    model.fit(X_train, y_train)
    
    # Predict and display results
    predictions = model.predict(X_test)
    print("Mean of y_test:", statistics.mean(y_test))
    print("Predictions:", predictions)
    print("Coefficients:", model.coef_)
    print("Intercept:", model.intercept_)
    
    # Calculate and print the accuracy of the model
    accuracy = model.score(X_test, y_test)
    print("Model accuracy:", accuracy)
    
    return accuracy

def main():
    # Path to the data file
    data_path = "/home/geospacial2/PycharmProjects/BTP/rain2014_1.csv"
    
    # Read the data from CSV file
    with open(data_path, 'rt') as csvfile:
        reader = csv.reader(csvfile)
        next(reader, None)  # Skip the header row
        features = []
        targets = []
        for row in reader:
            features.append(row[9:17])
            targets.append(row[22])
    
    # Scale the features
    features = StandardScaler().fit_transform(features)
    
    # Number of iterations for parallel processing
    n_iterations = 1
    
    # Determine the number of CPU cores available
    num_cores = multiprocessing.cpu_count()
    
    # Run the train_and_evaluate_model function in parallel
    results = Parallel(n_jobs=num_cores)(delayed(train_and_evaluate_model)(random_state, features, targets) for random_state in range(n_iterations))
    
    # Print the results from all iterations
    for i, result in enumerate(results):
        print(f"Iteration {i+1} - Accuracy: {result}")

if __name__ == '__main__':
    main()
