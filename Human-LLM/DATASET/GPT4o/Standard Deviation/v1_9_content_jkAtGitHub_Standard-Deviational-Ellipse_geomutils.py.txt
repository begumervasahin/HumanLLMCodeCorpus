import numpy as np

# Constants
RADIUS_OF_EARTH = 6371.0088  # Radius of the Earth in kilometers
KMS_TO_METERS = 1000.0
METERS_TO_KILOMETERS = 0.001
MILES_TO_KILOMETERS = 1.60934
KILOMETERS_TO_MILES = 0.621371

def gcd(loc1, loc2, R=RADIUS_OF_EARTH * KMS_TO_METERS):
    """Great circle distance via Haversine formula

    Parameters
    ----------
    loc1: tuple
        (long, lat in decimal degrees)
    loc2: tuple
        (long, lat in decimal degrees)
    R: float
        Radius of the Earth (default is meters)

    Returns
    -------
    float
        Great circle distance between loc1 and loc2 in units of R
    """
    lon1, lat1 = loc1
    lon2, lat2 = loc2
    dLat = np.radians(lat2 - lat1)
    dLon = np.radians(lon2 - lon1)
    lat1 = np.radians(lat1)
    lat2 = np.radians(lat2)
    a = np.sin(dLat / 2)**2 + np.cos(lat1) * np.cos(lat2) * np.sin(dLon / 2)**2
    c = 2 * np.arcsin(np.sqrt(a))
    return R * c

def compareFloat(a, b, rTol=0.00001, aTol=0.00000001):
    """Uses the same formula as numpy's allclose function

    Parameters
    ----------
    a: float
        Float to be compared
    b: float
        Float to be compared
    rTol: float
        Relative tolerance
    aTol: float
        Absolute tolerance

    Returns
    -------
    bool
        True if |a - b| < aTol + (rTol * |b|)
    """
    try:
        return abs(a - b) < aTol + (rTol * abs(b))
    except:
        return False

def convert2Radians(degree):
    """Converts degrees to radians.

    Parameters
    ----------
    degree: float
        Degree of angle

    Returns
    -------
    float
        Radians of angle
    """
    return np.pi / 180.0 * degree

def convert2degrees(radians):
    """Converts radians to degrees.

    Parameters
    ----------
    radians: float
        Radians of angle

    Returns
    -------
    float
        Degree of angle
    """
    return 180.0 / np.pi * radians
