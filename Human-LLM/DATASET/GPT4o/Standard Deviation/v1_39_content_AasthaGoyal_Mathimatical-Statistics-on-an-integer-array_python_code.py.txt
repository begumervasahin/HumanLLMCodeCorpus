import math

def main():
    stat = []
    temp_list = []
    sort_list = []
    error_list = []
    condition = True
    
    while condition:
        phrase = input("Enter a list of numbers (or type 'Terminate' to end): ")
        temp_list = phrase.split(",")
        stat.extend(temp_list)
        
        for item in temp_list:
            if not item.isdigit():
                print(f"{item} is not an integer value")
                error_list.append(item)
                if item.strip().lower() == "terminate":
                    stat.remove(item)
                    print("You terminated the program")
                    condition = False
                else:
                    stat.remove(item)
    
    while stat:
        min_value = stat[0]
        for x in stat:
            if x < min_value:
                min_value = x
        sort_list.append(min_value)
        stat.remove(min_value)
    
    sort_list = [int(x) for x in sort_list]
    print("Sorted List: ", sort_list)
    
    maximum = sort_list[-1]
    minimum = sort_list[0]
    range_value = maximum - minimum
    total_count = len(sort_list)
    sum1 = sum(sort_list)
    mean = sum1 / total_count
    
    print("1) The number of each individual number:")
    for num in set(sort_list):
        count = sort_list.count(num)
        print(f"{num}: {count}")
    
    variance = sum((x - mean) ** 2 for x in sort_list) / total_count
    std = math.sqrt(variance)
    
    print("2) The Maximum value is:", maximum)
    print("3) The Minimum value is:", minimum)
    print("4) The Range of the numbers is:", range_value)
    print("5) Arithmetic Mean of the numbers is:", mean)
    print("6) Variance of the numbers is:", variance)
    print("7) The Standard Deviation of the numbers is:", std)
    print("8) Errors:", error_list)

if __name__ == "__main__":
    main()
