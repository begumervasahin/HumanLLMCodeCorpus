import numpy as np
import h5py

# Define parameters
frames = [100, 1000, 10000]
exp_time = [0.001, 0.01, 0.1, 1]
pix_rate = ['slow', 'fast']

# Loop over parameters and process data
for i in frames:
    for j in exp_time:
        for k in pix_rate:
            mean = []
            std = []
            mnm = []
            mxm = []
            chunks = i // 100
            num = ["%06d" % a for a in range(1, chunks + 1)]
            
            for l in num:
                filename = '%d_frames_%d_%s_%s' % (i, j, k, l)
                with h5py.File(filename, 'r') as fid:
                    data = fid['/entry/data']
                    mean.append(np.mean(data[0:99], axis=0))
                    std.append(np.std(data[0:99], axis=0))
                    mnm.append(np.min(data[0:99], axis=0))
                    mxm.append(np.max(data[0:99], axis=0))

            grand_mean = np.mean(mean)
            grand_std = np.sqrt(np.sum([s**2 / chunks for s in std]))
            grand_mnm = np.min(mnm)
            grand_mxm = np.max(mxm)

            np.save('mean_%dframes_%ds_%s.npy' % (i, j, k), grand_mean)
            np.save('std_%dframes_%ds_%s.npy' % (i, j, k), grand_std)
            np.save('min_%dframes_%ds_%s.npy' % (i, j, k), grand_mnm)
            np.save('max_%dframes_%ds_%s.npy' % (i, j, k), grand_mxm)

print("Data processing complete.")
