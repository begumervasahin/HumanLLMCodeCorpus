def convert_to_float(data):
    """
    Convert all elements of the data list to floats.
    """
    return [float(item) for item in data]

def get_frequency_dict(data):
    """
    Convert a list of data into a dictionary with data values as keys and their frequencies as values.
    """
    if not isinstance(data, list):
        return None
    
    data = convert_to_float(data)
    frequency_dict = {}
    for item in data:
        if item in frequency_dict:
            frequency_dict[item] += 1
        else:
            frequency_dict[item] = 1
    
    return frequency_dict

def print_frequency_dict(frequency_dict):
    """
    Print the frequency dictionary with asterisks representing the frequency of each value.
    """
    for key in sorted(frequency_dict.keys()):
        asterisks = '*' if key != 1.8 else ' *'
        print(f"{key} {asterisks * frequency_dict[key]}")

def calculate_average(data):
    """
    Calculate the average of a list of numbers.
    """
    return sum(data) / len(data)

def calculate_dispersion(data):
    """
    Calculate the dispersion (standard deviation) of a list of numbers.
    """
    avg = calculate_average(data)
    squared_diffs = [(item - avg) ** 2 for item in data]
    return (sum(squared_diffs) / len(data)) ** 0.5

def calculate_standard_deviation(data):
    """
    Calculate the standard deviation of a list of numbers.
    """
    avg = calculate_average(data)
    squared_diffs = [(item - avg) ** 2 for item in data]
    return (sum(squared_diffs) / (len(data) * (len(data) - 1))) ** 0.5

def calculate_random_error(data, k_student):
    """
    Calculate the random error given a list of data and the k-Student value.
    """
    return calculate_standard_deviation(data) * k_student

def calculate_absolute_error(instrument_error, computational_error, random_error):
    """
    Calculate the absolute error from instrument, computational, and random errors.
    """
    return (instrument_error ** 2 + computational_error ** 2 + random_error ** 2) ** 0.5
