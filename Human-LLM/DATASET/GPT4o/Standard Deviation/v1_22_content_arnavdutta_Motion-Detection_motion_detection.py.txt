import numpy as np
import cv2
from datetime import datetime

def distMap(frame1, frame2):
    """Outputs Pythagorean distance between two frames."""
    frame1_32 = np.float32(frame1)
    frame2_32 = np.float32(frame2)
    diff32 = frame1_32 - frame2_32
    num = np.sqrt(diff32[:, :, 0]**2 + diff32[:, :, 1]**2 + diff32[:, :, 2]**2)
    den = np.sqrt(255**2 + 255**2 + 255**2)
    norm32 = num / den
    dist = np.uint8(norm32 * 255)
    return dist

sdThresh = 5
font = cv2.FONT_HERSHEY_SIMPLEX
cv2.namedWindow('frame')
cv2.namedWindow('dist')

# Open the default webcam
cap = cv2.VideoCapture(0, cv2.CAP_DSHOW)

# Read the first two frames
_, frame1 = cap.read()
_, frame2 = cap.read()

while cap.isOpened():
    ret, frame3 = cap.read()
    if ret:
        # Calculate the Pythagorean distance between the frames
        dist = distMap(frame1, frame3)
        
        # Update the frames
        frame1 = frame2
        frame2 = frame3
        
        # Apply Gaussian blur to the distance map
        mod = cv2.GaussianBlur(dist, (7, 7), 0)
        
        # Apply threshold to the blurred distance map
        _, thresh = cv2.threshold(mod, 100, 255, 0)
        
        # Calculate the standard deviation of the thresholded image
        _, stDev = cv2.meanStdDev(thresh)
        
        # Display the distance map
        cv2.imshow('dist', mod)
        
        # Display the standard deviation on the frame
        cv2.putText(frame1, "Standard Deviation - {}".format(round(stDev[0][0], 0)), 
                    (70, 70), font, 1, (255, 0, 255), 1, cv2.LINE_AA)
        
        # Detect motion based on the standard deviation threshold
        if stDev > sdThresh:
            print('Motion Detected:', datetime.now().strftime("%B %d %Y %H:%M:%S"))
        
        # Display the frame
        cv2.imshow('frame', frame2)
    else:
        # Reset the frame position if the capture failed
        cap.set(cv2.CAP_PROP_POS_FRAMES, 0)
    
    # Break the loop on pressing 'Esc'
    if cv2.waitKey(1) & 0xFF == 27:
        break

# Release the capture and destroy all windows
cap.release()
cv2.destroyAllWindows()
