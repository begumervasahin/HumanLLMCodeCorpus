import numpy as np
import cv2
from datetime import datetime

def calculate_distance_map(frame1, frame2):
    """Calculate the Pythagorean distance between two frames."""
    frame1_float = np.float32(frame1)
    frame2_float = np.float32(frame2)
    diff = frame1_float - frame2_float
    distance = np.sqrt(np.sum(diff ** 2, axis=2))
    normalized_distance = (distance / np.sqrt(3 * 255 ** 2)) * 255
    return np.uint8(normalized_distance)

def main():
    sd_threshold = 5
    font = cv2.FONT_HERSHEY_SIMPLEX

    # Set up windows for display
    cv2.namedWindow('Frame')
    cv2.namedWindow('Distance')

    # Open the default webcam
    cap = cv2.VideoCapture(0, cv2.CAP_DSHOW)

    # Read the first two frames
    ret1, frame1 = cap.read()
    ret2, frame2 = cap.read()

    while cap.isOpened():
        ret, frame3 = cap.read()
        if not ret:
            cap.set(cv2.CAP_PROP_POS_FRAMES, 0)
            continue

        # Calculate the distance map
        distance_map = calculate_distance_map(frame1, frame3)

        # Update the frames
        frame1 = frame2
        frame2 = frame3

        # Apply Gaussian blur to the distance map
        blurred_distance = cv2.GaussianBlur(distance_map, (7, 7), 0)

        # Apply threshold to the blurred distance map
        _, thresholded = cv2.threshold(blurred_distance, 100, 255, cv2.THRESH_BINARY)

        # Calculate the standard deviation of the thresholded image
        _, st_dev = cv2.meanStdDev(thresholded)

        # Display the distance map
        cv2.imshow('Distance', blurred_distance)

        # Display the standard deviation on the frame
        cv2.putText(frame2, f"Standard Deviation: {round(st_dev[0][0], 0)}", 
                    (70, 70), font, 1, (255, 0, 255), 1, cv2.LINE_AA)

        # Detect motion based on the standard deviation threshold
        if st_dev > sd_threshold:
            print(f"Motion Detected: {datetime.now().strftime('%B %d %Y %H:%M:%S')}")

        # Display the frame
        cv2.imshow('Frame', frame2)

        # Break the loop on pressing 'Esc'
        if cv2.waitKey(1) & 0xFF == 27:
            break

    # Release the capture and destroy all windows
    cap.release()
    cv2.destroyAllWindows()

if __name__ == "__main__":
    main()
