import pandas as pd
from pandas.plotting import register_matplotlib_converters
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.dates as mdates
import os
import datetime

class PolarRRI:
    def __init__(self, file_path=None):
        if file_path is None:
            self.rri_df = pd.DataFrame()
            return
        
        # Load and process the RRI data from the provided file
        self.load_data(file_path)

    def load_data(self, file_path):
        file_name = os.path.basename(file_path)
        src_df = pd.read_csv(file_path, header=None, sep=' ')
        
        # Rename columns and convert RRI values from seconds to milliseconds
        src_df = src_df.rename(columns={0: 'time', 1: 'rri'})
        src_df['rri'] = src_df['rri'] * 1000
        
        # Extract datetime information from the file name
        file_date, file_time = file_name.split('.')[0].split('_')
        _, yy, MM, dd = file_date.split('-')
        hh, mm, ss = file_time.split('-')
        start_datetime = datetime.datetime(int(yy), int(MM), int(dd), int(hh), int(mm), int(ss))
        
        # Apply function to convert time to datetime
        src_df['datetime'] = src_df.apply(lambda x: start_datetime + datetime.timedelta(seconds=x[0]), axis=1)
        self.rri_df = src_df.set_index(['datetime'])

    def plot_rri(self):
        # Register matplotlib converters for datetime plotting
        register_matplotlib_converters()
        
        fig, ax = plt.subplots()
        ax.xaxis.set_major_formatter(mdates.DateFormatter('%H:%M:%S'))
        
        ax.plot(self.rri_df.index, self.rri_df['rri'], marker='.')
        plt.show()

    def del_outliers(self, thresh_min, thresh_max):
        # Filter out RRI values outside the specified threshold range
        self.rri_df['rri'] = self.rri_df['rri'].where((self.rri_df['rri'] > thresh_min) & (self.rri_df['rri'] < thresh_max))

    def get_rri_df(self):
        return self.rri_df

    def write(self, file_path):
        # Convert RRI values to integers and save to CSV
        self.rri_df['rri'] = self.rri_df['rri'].astype(int)
        self.rri_df.to_csv(file_path)

    def import_rri(self, file_path):
        # Import additional RRI data from a CSV file and append to the existing dataframe
        rri_df = pd.read_csv(file_path)
        rri_df['datetime'] = pd.to_datetime(rri_df['datetime'])
        rri_df = rri_df.set_index(['datetime'])
        self.rri_df = pd.concat([self.rri_df, rri_df])
