import csv
import numpy as np
import math
import pandas as pd
import xlsxwriter

def resistance(ticker):
    df = pd.read_csv(f"C:/Users/Hp/Desktop/New folder/Top500_stock/Original data/{ticker}.csv")
    if len(df.columns) >= 6:
        high_data = df['High'].tolist()
        maxi = math.ceil(max(high_data))
        mini = math.floor(min(high_data))
        interval = np.around(np.linspace(mini, maxi, no_of_intervals+1), 2).tolist()
        
        interval_data = []
        max_count = 0
        max_position = mini

        for i in interval[:-1]:
            count = sum(1 for j in high_data if i <= j < interval[interval.index(i) + 1])
            
            if count > max_count:
                max_count = count
                max_position = i
                
            interval_data.append((i, interval[interval.index(i) + 1], count))
        
        interval_data.append((max_position, interval[interval.index(max_position) + 1], max_count))
        intervals_data.append(interval_data)
        new_ticker_list.append(ticker)

def get_ticker_list():
    with open(r"C:\Users\Hp\Desktop\New folder\nifty500_list.csv", 'r') as f:
        reader = csv.reader(f)
        next(reader)  # Skip header
        for line in reader:
            ticker_list.append(line[2])

def main():
    get_ticker_list()
    
    for ticker in ticker_list:
        ticker = ticker.replace('&', '%26')
        print(ticker)
        resistance(ticker)

    sdf = pd.DataFrame(intervals_data, index=pd.Series(new_ticker_list, name="SYMBOL"), columns=columns)
    
    # Save to Excel
    writer = pd.ExcelWriter(r"C:/Users/Hp/Desktop/New folder/Top500_stock/resistance.xlsx", engine="xlsxwriter")
    sdf.to_excel(writer, sheet_name="Sheet1")
    writer.save()

if __name__ == "__main__":
    ticker_list = []
    intervals_data = []
    no_of_intervals = 10
    columns = [f"INTERVAL NO. {i}" for i in range(1, no_of_intervals+1)]
    columns.append("MAX FREQ.")
    new_ticker_list = []

    main()
