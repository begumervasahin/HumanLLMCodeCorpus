import pandas as pd
import numpy as np
import datetime
import calendar
from pandas_datareader import data as web

def main():
    # Define the time period for data extraction
    start = datetime.date(2016, 1, 1)
    end = datetime.date(2016, 12, 31)
    ticker = 'AAPL'
    
    # Fetch the stock price data from Yahoo Finance
    stock_prices_all = web.DataReader(ticker, 'yahoo', start, end)
    
    # Filter out rows where Volume is 0
    stock_prices = stock_prices_all[stock_prices_all['Volume'] != 0]
    
    # Reset the index to have a 'Date' column for easier processing
    stock_prices.reset_index(inplace=True)
    
    # Separate the stock prices data by month
    all_monthly_data = month_separate(stock_prices)
    
    # Calculate the mean closing price for each month
    monthly_avg_prices = mean_calculator(all_monthly_data)
    
    # Print the monthly average closing prices
    print(monthly_avg_prices)

def month_separate(stock_prices):
    # Initialize variables to keep track of current month and monthly data
    current_month = 0
    month_dict = {}
    month_array = []
    
    # Iterate through the stock prices data row by row
    for index, row in stock_prices.iterrows():
        day_month = row['Date'].month
        day_month_name = calendar.month_name[day_month]
        
        # Check if the row belongs to the current month
        if day_month == current_month:
            month_array.append(row)
        else:
            # If we are switching to a new month, save the current month's data
            if current_month != 0:
                monthly_data = pd.DataFrame(month_array)
                month_dict[last_month] = monthly_data
            
            # Start tracking the new month
            month_array = [row]
            current_month = day_month
            last_month = day_month_name
    
    # Add the last month's data to the dictionary
    if month_array:
        monthly_data = pd.DataFrame(month_array)
        month_dict[day_month_name] = monthly_data
    
    return month_dict

def mean_calculator(all_monthly_data):
    # Initialize lists to store the average prices and the names of the months
    monthly_average_price = []
    name_of_months = list(all_monthly_data.keys())
    
    # Iterate through each month's data to calculate the average closing price
    for data in all_monthly_data.values():
        average_price = data['Close'].mean()
        monthly_average_price.append(average_price)
    
    # Create a DataFrame to display the monthly average closing prices
    avg_price = pd.DataFrame(data=monthly_average_price, index=name_of_months, columns=["Mean Closing Price"])
    return avg_price

if __name__ == "__main__":
    main()
