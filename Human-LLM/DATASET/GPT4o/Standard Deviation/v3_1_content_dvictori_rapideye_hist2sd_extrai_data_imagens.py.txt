import glob
from osgeo import ogr

def extract_and_add_dates(image_files, articulation_shapefile, output_shapefile_path):
    """
    Extracts the date from the scene filenames and adds it to the articulation shapefile.
    Note: The position of the date in the filename is fixed!!!
    """
    # Open the input shapefile
    input_shapefile = ogr.Open(articulation_shapefile)
    if input_shapefile is None:
        raise Exception("Failed to open the input shapefile.")

    # Create a copy of the input shapefile to modify
    driver = ogr.GetDriverByName("ESRI Shapefile")
    output_shapefile = driver.CopyDataSource(input_shapefile, output_shapefile_path)
    input_shapefile = None  # Close the input shapefile
    layer = output_shapefile.GetLayer()

    # Add a new date field to the shapefile
    date_field = ogr.FieldDefn('Data', ogr.OFTDate)
    layer.CreateField(date_field)

    # Fixed positions for date and scene in the filename
    DATE_POSITION = 48
    SCENE_POSITION = 40

    # Iterate over the images to extract date and update shapefile
    for image_file in image_files:
        # Extract the date and scene ID from the filename
        date_str = image_file[DATE_POSITION:DATE_POSITION + 10].split('-')
        scene_id = image_file[SCENE_POSITION:SCENE_POSITION + 7]

        # Filter features by TILE_ID to find the corresponding scene
        layer.SetAttributeFilter(f'TILE_ID = {int(scene_id)}')

        # Update the date field for the filtered features
        for feature in layer:
            feature.SetField('Data', int(date_str[0]), int(date_str[1]), int(date_str[2]), 0, 0, 0, 0)
            layer.SetFeature(feature)

    # Close the output shapefile
    output_shapefile = None

def main():
    # Paths to the images and articulation shapefile
    image_folder = 'Z:\\filtro_radiometrico_2014_16bits\\*.tif'
    articulation_shapefile = 'C:\\geodb\\articulacao\\RapidEye_Tiles\\RapidEye_Tiles_Brasil.shp'
    output_shapefile_path = 'I:\\articulacao_rapideye_2014.shp'
    
    # Get the list of image files
    image_files = glob.glob(image_folder)

    # Process the scenes to update the articulation shapefile
    extract_and_add_dates(image_files, articulation_shapefile, output_shapefile_path)

if __name__ == "__main__":
    main()
