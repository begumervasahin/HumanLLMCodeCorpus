import numpy as np
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

# Load the data from the CSV file
data = pd.read_csv("/Users/khadeejah/Desktop/weight-height.csv")

# Display the first few rows of the data
print("First few rows of the dataset:")
print(data.head())

# Scatter plot of Weight vs Height
plt.figure(figsize=(20, 8))
plt.scatter(data['Weight'], data['Height'], alpha=0.5)
plt.title('Scatter Plot of Weight vs Height')
plt.xlabel('Weight')
plt.ylabel('Height')
plt.show()

# Display descriptive statistics
print("\nDescriptive statistics of the dataset:")
print(data.describe())

# Visualizing the distribution, box plot, and histogram of Height
plt.figure(figsize=(20, 8))

# Distribution plot of Height
plt.subplot(1, 3, 1)
sns.histplot(data['Height'], kde=True)
plt.title('Distribution of Height')
plt.ylabel('Distribution')

# Box plot of Height
plt.subplot(1, 3, 2)
sns.boxplot(y=data['Height'])
plt.title('Box Plot of Height')

# Histogram of Height
plt.subplot(1, 3, 3)
plt.hist(data['Height'], bins=20, rwidth=0.8)
plt.title('Histogram of Height')
plt.xlabel('Height')
plt.ylabel('Count')

plt.show()

# Calculate IQR for Height and identify outliers
Q1_height = data['Height'].quantile(0.25)
Q3_height = data['Height'].quantile(0.75)
IQR_height = Q3_height - Q1_height
lower_limit_height = Q1_height - 1.5 * IQR_height
upper_limit_height = Q3_height + 1.5 * IQR_height

print("\nHeight lower limit:", lower_limit_height)
print("Height upper limit:", upper_limit_height)

data_outlier_height = data[(data['Height'] < lower_limit_height) | (data['Height'] > upper_limit_height)]
print("\nOutliers in Height:")
print(data_outlier_height)

data_no_outlier_height = data[(data['Height'] > lower_limit_height) & (data['Height'] < upper_limit_height)]
print("\nData without outliers in Height (first 3 rows):")
print(data_no_outlier_height.head(3))
