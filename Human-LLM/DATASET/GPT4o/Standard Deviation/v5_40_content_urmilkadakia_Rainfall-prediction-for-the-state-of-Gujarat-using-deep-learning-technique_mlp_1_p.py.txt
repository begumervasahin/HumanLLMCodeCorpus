from sklearn.neural_network import MLPRegressor
import csv
import os
from sklearn.model_selection import train_test_split
import numpy as np
from sklearn.preprocessing import StandardScaler
from joblib import Parallel, delayed
import multiprocessing
import statistics

def train_and_evaluate_model(random_state, features, targets):
    """
    Train and evaluate an MLPRegressor model using the provided features and targets.
    
    Args:
    random_state (int): Random state for reproducibility.
    features (ndarray): Feature matrix.
    targets (ndarray): Target vector.
    
    Returns:
    None
    """
    # Split the data into training and testing sets
    X_train, X_test, y_train, y_test = train_test_split(features, targets, test_size=0.2, random_state=random_state)
    
    # Convert data to the appropriate format
    X_train = np.array(X_train, dtype=float)
    X_test = np.array(X_test, dtype=float)
    y_train = np.array(y_train, dtype=int).ravel()
    y_test = np.array(y_test, dtype=int).ravel()
    
    # Initialize the MLPRegressor
    mlp = MLPRegressor(max_iter=1000, learning_rate_init=0.05, solver='sgd', hidden_layer_sizes=(640,), alpha=0.1)
    
    # Print mean of the test targets
    print(f"Mean of y_test: {statistics.mean(y_test)}")
    
    # Train the model
    mlp.fit(X_train, y_train)
    
    # Predict on the test set
    predictions = mlp.predict(X_test)
    print(f"Predictions: {predictions}")
    
    # Calculate and print the accuracy score
    accuracy = mlp.score(X_test, y_test)
    print(f"Accuracy: {accuracy}")

def load_data(file_path):
    """
    Load data from a CSV file and return features and targets.
    
    Args:
    file_path (str): Path to the CSV file.
    
    Returns:
    tuple: Tuple containing the feature matrix and target vector.
    """
    features = []
    targets = []
    
    with open(file_path, 'rt') as csvfile:
        reader = csv.reader(csvfile)
        next(reader, None)  # Skip the header row
        
        for row in reader:
            features.append(row[9:17])
            targets.append(row[22])
    
    return np.array(features), np.array(targets)

def main():
    # Define the path to the data
    data_path = "/home/geospacial2/PycharmProjects/BTP/rain2014_1.csv"
    
    # Load the data
    features, targets = load_data(data_path)
    
    # Standardize the features
    features = StandardScaler().fit_transform(features)
    
    # Define the number of iterations and get the number of CPU cores
    num_iterations = 1
    num_cores = multiprocessing.cpu_count()
    
    # Run the train_and_evaluate_model function in parallel
    Parallel(n_jobs=num_cores)(
        delayed(train_and_evaluate_model)(i, features, targets) for i in range(num_iterations)
    )

if __name__ == '__main__':
    main()
