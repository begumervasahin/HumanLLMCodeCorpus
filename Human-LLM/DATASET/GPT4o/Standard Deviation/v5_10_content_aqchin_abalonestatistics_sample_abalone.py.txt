import numpy as np
import random
import matplotlib.pyplot as plt

# Set random seed for reproducibility
random.seed(0)

# List of different run counts to test
run_tests = [10, 100, 1000, 10000, 100000]

# Initialize lists to store results
means = []
stds = []
runs = []
scale = 0.1

def read_data(file_path):
    """Reads the file to determine the number of rows."""
    with open(file_path) as file:
        num_rows = sum(1 for _ in file)
    return num_rows

def perform_simulation(num_rows, run_count, scale):
    """Performs the simulation for the given number of runs and returns the data."""
    data_in = [0] * num_rows
    for _ in range(run_count):
        n_to_select = float(int(num_rows * scale))
        n_remaining = num_rows
        
        for j in range(num_rows):
            if n_to_select / n_remaining > random.random():
                data_in[j] += 1
                n_to_select -= 1
            n_remaining -= 1
    
    mean = np.mean(data_in) / run_count
    std = np.std(data_in) / run_count
    return mean, std

def main():
    # Read the abalone.data file to count the number of rows
    num_rows = read_data('abalone.data')

    # Run the simulation for each number of runs in run_tests
    for run_count in run_tests:
        mean, std = perform_simulation(num_rows, run_count, scale)
        means.append(mean)
        stds.append(std)
        runs.append(run_count)
    
    # Plotting the results
    plt.figure(figsize=(10, 8))
    
    plt.subplot(2, 1, 1)
    plt.plot(runs, means, marker='o')
    plt.title("Mean VS Runs")
    plt.ylabel("Mean")
    
    plt.subplot(2, 1, 2)
    plt.plot(runs, stds, marker='o')
    plt.title("Standard Deviation VS Runs")
    plt.xlabel("Runs")
    plt.ylabel("Standard Deviation")
    
    plt.tight_layout()
    plt.show()

# Run the main function
if __name__ == "__main__":
    main()
