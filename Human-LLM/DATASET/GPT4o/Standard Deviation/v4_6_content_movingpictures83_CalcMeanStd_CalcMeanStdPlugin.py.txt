import math

class CalcMeanStdPlugin:
    def input(self, inputfile):
        with open(inputfile, 'r') as noafile:
            self.scores = [float(line.split('\t')[1]) for line in noafile.readlines()[1:]]

    def run(self):
        n = len(self.scores)
        self.mean = sum(self.scores) / n
        variance = sum((score - self.mean) ** 2 for score in self.scores) / n
        self.std = math.sqrt(variance)

    def output(self, outputfile):
        stats = {
            "Two left": self.mean - 2 * self.std,
            "One left": self.mean - self.std,
            "Mean": self.mean,
            "One right": self.mean + self.std,
            "Two right": self.mean + 2 * self.std
        }
        for key, value in stats.items():
            print(f"[CalcMeanStdPlugin] {key}: {value}")
