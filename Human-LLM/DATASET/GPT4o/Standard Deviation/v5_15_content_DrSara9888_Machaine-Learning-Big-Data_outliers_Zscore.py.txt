import numpy as np
import pandas as pd
from matplotlib import pyplot as plt
import seaborn as sns
from scipy.stats import norm

# Load the data
data = pd.read_csv("/Users/khadeejah/Desktop/Bengaluru_House_Data.csv")
print(data.head())

# Plot the histogram of the house prices
plt.figure(figsize=(10, 5))
plt.hist(data.price, bins=20, rwidth=0.8)
plt.xlabel('Price')
plt.ylabel('Count')
plt.yscale('log')
plt.show()

# Determine the price limits for outliers
max_limit = data["price"].quantile(0.999)
min_limit = data["price"].quantile(0.0001)

# Display the outliers
print("Outliers above max limit:\n", data[data["price"] > max_limit])
print("Outliers below min limit:\n", data[data["price"] < min_limit])

# Remove the outliers from the data
data_filtered = data[(data["price"] > min_limit) & (data["price"] < max_limit)]
print(data_filtered.head())

# Number of rows removed
print(f"Number of outliers removed: {data.shape[0] - data_filtered.shape[0]}")

# Plot histograms of filtered data
plt.figure(figsize=(20, 8))

plt.subplot(1, 2, 1)
plt.hist(data_filtered.price, bins=20, rwidth=0.8)
plt.xlabel('Price')
plt.ylabel('Count')
plt.yscale('log')

plt.subplot(1, 2, 2)
sns.histplot(data_filtered.price, kde=True)
plt.xlabel('Price')
plt.ylabel('Density')

plt.show()

# Fit a normal distribution and plot it
plt.figure(figsize=(10, 5))
plt.hist(data_filtered.price, bins=20, rwidth=0.8, density=True)
plt.xlabel('Price')
plt.ylabel('Density')

rng = np.arange(-1000, data_filtered.price.max(), 100)
plt.plot(rng, norm.pdf(rng, data_filtered.price.mean(), data_filtered.price.std()))
plt.show()

# Statistical summary
print(data_filtered.describe())
print(f"Mean: {data_filtered.price.mean()}")
print(f"Standard Deviation: {data_filtered.price.std()}")

# Determine the 4-sigma limits
upper_limit = data_filtered.price.mean() + 4 * data_filtered.price.std()
lower_limit = data_filtered.price.mean() - 4 * data_filtered.price.std()

# Display the outliers based on 4-sigma rule
print("Outliers above upper limit:\n", data_filtered[data_filtered.price > upper_limit])
print("Outliers below lower limit:\n", data_filtered[data_filtered.price < lower_limit])

# Filter the data within 4-sigma limits
data_4sigma_filtered = data_filtered[(data_filtered.price < upper_limit) & (data_filtered.price > lower_limit)]
print(data_4sigma_filtered.head())

# Compare the shapes of the original and filtered datasets
print(f"Original data shape: {data.shape}")
print(f"Filtered data shape: {data_filtered.shape}")
print(f"4-sigma filtered data shape: {data_4sigma_filtered.shape}")

# Plot histogram with normal distribution fit for the 4-sigma filtered data
plt.figure(figsize=(10, 5))
plt.hist(data_4sigma_filtered.price, bins=20, rwidth=0.8, density=True)
plt.xlabel('Price')
plt.ylabel('Density')

rng = np.arange(-600, data_4sigma_filtered.price.max(), 100)
plt.plot(rng, norm.pdf(rng, data_4sigma_filtered.price.mean(), data_4sigma_filtered.price.std()))
plt.show()

# Calculate z-scores and filter further based on z-scores
data_filtered["zscore"] = (data_filtered.price - data_filtered.price.mean()) / data_filtered.price.std()
print(data_filtered.head())

# Display the z-score outliers
print("Z-score outliers above 4:\n", data_filtered[data_filtered["zscore"] > 4])
print("Z-score outliers below -4:\n", data_filtered[data_filtered['zscore'] < -4])
print("Z-score outliers overall:\n", data_filtered[(data_filtered.zscore < -4) | (data_filtered.zscore > 4)])

# Filter the data within z-score limits of -4 to 4
data_zscore_filtered = data_filtered[(data_filtered.zscore > -4) & (data_filtered.zscore < 4)]
print(data_zscore_filtered.head())

# Compare the shapes of the filtered datasets
print(f"Filtered data shape after z-score filtering: {data_zscore_filtered.shape}")
print(f"Rows removed by z-score filtering: {data_filtered.shape[0] - data_zscore_filtered.shape[0]}")
print(f"Rows removed by 4-sigma filtering: {data_filtered.shape[0] - data_4sigma_filtered.shape[0]}")
