from sklearn.neural_network import MLPRegressor
import csv
import os
from sklearn.model_selection import train_test_split
import numpy as np
from sklearn.preprocessing import StandardScaler
from joblib import Parallel, delayed
import multiprocessing
import matplotlib.pyplot as plt
import statistics

def machine(k, features, targets):
    # Split the data into training and testing sets
    X_train, X_test, y_train, y_test = train_test_split(features, targets, test_size=0.2, random_state=k)
    
    # Convert data to float for features and int for targets
    X_train = np.array(X_train, dtype=float)
    X_test = np.array(X_test, dtype=float)
    y_train = np.array(y_train, dtype=int).ravel()
    y_test = np.array(y_test, dtype=int).ravel()
    
    # Initialize the MLPRegressor
    clf = MLPRegressor(max_iter=1000, learning_rate_init=0.05, solver='sgd', hidden_layer_sizes=(640,), alpha=1e-1)
    
    # Print mean of the test targets
    print(f"Mean of y_test: {statistics.mean(y_test)}")
    
    # Train the model
    clf.fit(X_train, y_train)
    
    # Predict on the test set
    predictions = clf.predict(X_test)
    print(f"Predictions: {predictions}")
    
    # Calculate the accuracy score
    accuracy = clf.score(X_test, y_test)
    print(f"Accuracy: {accuracy}")

if __name__ == '__main__':
    # Define the path to the data
    dataPath = "/home/geospacial2/PycharmProjects/BTP/"
    
    # Load the data from CSV file
    with open(os.path.join(dataPath, "rain2014_1.csv"), 'rt') as csvfile:
        reader = csv.reader(csvfile)
        next(reader, None)  # Skip the header row
        
        features = []
        targets = []
        
        for row in reader:
            features.append(row[9:17])
            targets.append(row[22])
    
    # Standardize the features
    features = StandardScaler().fit_transform(features)
    
    # Define the number of iterations
    niter = 1
    
    # Get the number of CPU cores
    num_cores = multiprocessing.cpu_count()
    
    # Run the machine function in parallel
    Parallel(n_jobs=num_cores)(delayed(machine)(i, features, targets) for i in range(niter))
