Standard deviation and mean calculator for performance and pin outputs.

import re
import numpy as np

# Constants
PERF_OUTPUT_FILE = "measure.out"
SEARCH_TERMS = [
    "mem_inst_retired.all_loads:u",
    "instructions:u",
    "mem_inst_retired.all_stores:u",
    "br_inst_retired.all_branches:u"
]
DEBUG_MODE = False

def parse_perf_output(file_path, search_terms, debug_mode=False):
    """
    Parses the performance output file to extract numeric values for the given search terms.
    
    Args:
    - file_path (str): Path to the performance output file.
    - search_terms (list): List of search terms to look for in the file.
    - debug_mode (bool): If True, prints debug information.

    Returns:
    - dict: A dictionary with search terms as keys and lists of extracted numeric values as values.
    """
    runtimes = {term: [] for term in search_terms}

    with open(file_path, "r") as file:
        for line in file:
            for term in search_terms:
                if term in line:
                    numeric_value = int(re.findall(r"\d+", line)[0])
                    if debug_mode:
                        print(f"Found {numeric_value} for term {term}")
                    runtimes[term].append(numeric_value)

    return runtimes

def calculate_statistics(runtimes):
    """
    Calculates the standard deviation and mean for the extracted runtime values.

    Args:
    - runtimes (dict): Dictionary with search terms as keys and lists of numeric values as values.

    Returns:
    - dict: A dictionary with search terms as keys and tuples of (standard deviation, mean) as values.
    """
    statistics = {}
    for term, values in runtimes.items():
        std_dev = np.std(values)
        mean = np.mean(values)
        statistics[term] = (std_dev, mean)
    return statistics

def generate_report(statistics, report_file):
    """
    Generates a report file with the calculated statistics.

    Args:
    - statistics (dict): Dictionary with search terms as keys and tuples of (standard deviation, mean) as values.
    - report_file (str): Path to the report file to be created.
    """
    header = "Type of instruction:\tstd,\tmean\n"
    report_lines = [header]
    for term, (std_dev, mean) in statistics.items():
        report_lines.append(f"{term}:\t{std_dev:.2f},\t{mean:.2f}\n")

    with open(report_file, "w") as file:
        file.writelines(report_lines)
    
    print("Report generated successfully!")
    print("".join(report_lines))

def main():
    """
    Main function to parse the performance output, calculate statistics, and generate a report.
    """
    runtimes = parse_perf_output(PERF_OUTPUT_FILE, SEARCH_TERMS, DEBUG_MODE)
    
    if DEBUG_MODE:
        print(f"Runtimes collected: {runtimes}")
    
    statistics = calculate_statistics(runtimes)
    generate_report(statistics, "perf_report.txt")

if __name__ == '__main__':
    main()
