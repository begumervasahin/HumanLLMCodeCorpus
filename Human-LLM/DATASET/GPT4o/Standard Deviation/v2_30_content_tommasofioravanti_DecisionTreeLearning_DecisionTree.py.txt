class DecisionTree:
    def __init__(self, attribute, threshold, attrname=None, default_child=None, branches=None):
        """
        Initialize the DecisionTree node.
        
        Parameters:
        - attribute: The attribute this node tests.
        - threshold: The threshold value for decision making.
        - attrname: Optional attribute name (if different from the attribute).
        - default_child: The default child if the attribute value does not match any branch.
        - branches: A dictionary to hold the branches.
        """
        self.attr = attribute
        self.threshold = threshold
        self.attrname = attrname or attribute
        self.default_child = default_child
        self.branches = branches or {}

    def __call__(self, example):
        """
        Classify the given example using the attribute and branches of the decision tree.
        
        Parameters:
        - example: A dictionary representing the example to classify.
        
        Returns:
        - The result of classification based on the tree structure.
        """
        attrvalue = example[self.attr]
        if float(attrvalue) > self.threshold:
            return self.branches[(self.threshold, True)](example)
        else:
            return self.branches[(self.threshold, False)](example)

    def add(self, val, is_greater, subtree):
        """
        Add a branch to the decision tree for the given value and comparison result.
        
        Parameters:
        - val: The value to compare against the threshold.
        - is_greater: Boolean indicating if the branch is for values greater than the threshold.
        - subtree: The subtree to add for this branch.
        """
        self.branches[(val, is_greater)] = subtree

    def display(self, indent=0):
        """
        Display the structure of the decision tree.
        
        Parameters:
        - indent: The indentation level for pretty-printing the tree.
        """
        name = self.attrname
        print('Test', name)
        for (val, subtree) in self.branches.items():
            if val[1]:
                print(' ' * 4 * indent, name, '>', val[0], '==>', end=' ')
            else:
                print(' ' * 4 * indent, name, '<=', val[0], '==>', end=' ')
            subtree.display(indent + 1)


# Example usage:
if __name__ == "__main__":
    # Creating leaf nodes
    leaf1 = DecisionTree(None, None)
    leaf1.__call__ = lambda x: "Leaf 1"

    leaf2 = DecisionTree(None, None)
    leaf2.__call__ = lambda x: "Leaf 2"

    # Creating the root of the tree and adding branches
    tree = DecisionTree('attribute1', 10)
    tree.add(10, True, leaf1)
    tree.add(10, False, leaf2)

    # Displaying the tree structure
    tree.display()

    # Testing classification with example data
    example = {'attribute1': 12}
    print("Classification:", tree(example))  # Should classify as "Leaf 1"

    example = {'attribute1': 8}
    print("Classification:", tree(example))  # Should classify as "Leaf 2"
