import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import statsmodels.formula.api as smf
import csv

def read_csv_data(file_path, payroll_col, overtime_col):
    """
    Read data from CSV file and extract specific columns.
    
    Parameters:
        file_path (str): Path to the CSV file.
        payroll_col (int): Index of the payroll column.
        overtime_col (int): Index of the overtime pay column.
    
    Returns:
        tuple: Two lists containing payroll and overtime pay data.
    """
    payroll_list = []
    overtime_pay = []
    
    with open(file_path, 'r') as file:
        csv_file = csv.reader(file)
        next(csv_file)  # Skip header
        
        for row in csv_file:
            payroll_list.append(float(row[payroll_col]))
            overtime_pay.append(float(row[overtime_col]))
    
    return payroll_list, overtime_pay

def calculate_statistics(data):
    """
    Calculate and print mean and standard deviation of the data.
    
    Parameters:
        data (list): List of numerical data.
    
    Returns:
        tuple: Mean and standard deviation of the data.
    """
    mean = np.mean(data)
    std_dev = np.std(data)
    print(f"Mean: {mean}")
    print(f"Standard Deviation: {std_dev}")
    return mean, std_dev

def perform_linear_regression(data, formula):
    """
    Perform linear regression and print the model.
    
    Parameters:
        data (DataFrame): Pandas DataFrame containing the data.
        formula (str): Formula for the linear regression.
    
    Returns:
        RegressionResultsWrapper: Fitted linear model.
    """
    lm = smf.ols(formula=formula, data=data).fit()
    print(f"The linear regression model is: {lm.params[0]:.2f} + {lm.params[1]:.2f} * BP")
    return lm

def plot_histogram(data, bins, xlabel, ylabel):
    """
    Plot histogram of the data.
    
    Parameters:
        data (list): List of numerical data.
        bins (list): List of bin edges.
        xlabel (str): Label for the x-axis.
        ylabel (str): Label for the y-axis.
    """
    plt.hist(data, bins=bins)
    plt.xlabel(xlabel)
    plt.ylabel(ylabel)
    plt.show()

def main():
    # File path and column indices for payroll and overtime pay
    file_path = 'san-francisco-2014.csv'
    payroll_col = 7
    overtime_col = 3

    # Read data from CSV
    payroll_list, overtime_pay = read_csv_data(file_path, payroll_col, overtime_col)

    # Calculate and print statistics
    mean, std_dev = calculate_statistics(payroll_list)

    # Define bins for histogram
    bin_array = [i * 1000 for i in range(530)]

    # Load data into a pandas DataFrame
    data = pd.read_csv(file_path, index_col=0)

    # Perform and print linear regression model
    formula = 'TPB ~ BP'  # Replace 'TPB' and 'BP' with actual column names from your CSV
    lm = perform_linear_regression(data, formula)

    # Plot histogram
    plot_histogram(payroll_list, bin_array, 'Total Pay and Benefits for 2014', 'Number of people')

if __name__ == "__main__":
    main()
