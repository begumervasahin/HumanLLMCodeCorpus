import csv

# Function to process and write data to output CSV
def process_and_write_data(input_file, output_file):
    # Initialize an empty list to store the data
    data = []

    # Open the output file in write mode
    with open(output_file, 'w+', newline='') as file_out:
        writer = csv.writer(file_out)

        # Write the header row to the output file
        writer.writerow([
            "Station_ID", "Latitude", "Longitude", "Time", "Date", "Top Latitude",
            "Bottom Latitude", "Left Longitude", "Right Longitude",
            "TIR1_Temperature(K)", "WV_Radiance", "Elevation(m)", "Vegetation",
            "Mean3(K)", "Std3(K)", "Mean5(K)", "Std5(K)", "Air_Temp(°C)",
            "Wind_Speed(m/s)", "Wind_Direction(deg)", "Atmo_Pressure(hpa)",
            "Humidity(%)", "Rain_Fall(mm)"
        ])

        # Open the input file in read mode
        with open(input_file, 'r') as input1:
            reader = csv.reader(input1, delimiter=',')

            # Skip the header row in the input file
            next(reader)

            # Read the data from the input file and store it in the data list
            for row in reader:
                data.append(row)

            # Process each row in the data list
            for row in data:
                # Round the value at index 13 to 3 decimal places
                row[13] = round(float(row[13]), 3)

                # Filter the rows based on specific conditions
                if float(row[12]) != -9999.0 and row[13] != -49.995:
                    # Create a new list with selected columns
                    new_row = [
                        row[0], row[1], row[2], row[7], row[8],
                        row[3], row[4], row[5], row[6], row[10],
                        row[11], row[12], row[13], row[14],
                        row[15], row[16], row[17], row[18],
                        row[19], row[20], row[21], row[22],
                        row[23]
                    ]

                    # Write the processed row to the output file
                    writer.writerow(new_row)

# Specify the input and output file names
input_file_name = 'final_rain2016_sorted.csv'
output_file_name = 'rain2016.csv'

# Call the function to process and write data
process_and_write_data(input_file_name, output_file_name)
