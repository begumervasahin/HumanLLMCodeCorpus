import subprocess
import pytest

def run_command(args):
    """Helper function to run a command and return the decoded output."""
    return subprocess.check_output(args).decode('utf-8').lower()

def test_run():
    """Test that the program runs with the expected arguments."""
    result = subprocess.check_call(['python', 'classify.py'], stdout=subprocess.DEVNULL)
    assert result == 0

def test_valid_landcover():
    """Test that the program runs with the expected landcover optional argument."""
    result = subprocess.check_call(
        ['python', 'classify.py', '--landcover', 'water'], stdout=subprocess.DEVNULL
    )
    assert result == 0

def test_no_landcover():
    """Test that providing landcover without a value raises an error."""
    with pytest.raises(subprocess.CalledProcessError):
        subprocess.check_call(
            ['python', 'classify.py', '--landcover'],
            stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL
        )

def test_water():
    """Test that the water landcover returns the expected output."""
    output = run_command(['python', 'classify.py', '--landcover', 'water'])
    assert 'water' in output
    assert '0' in output

def test_woody_savannas():
    """Test that the woody savannas landcover returns the expected output."""
    output = run_command(['python', 'classify.py', '--landcover', 'woody savannas'])
    assert 'woody savannas' in output
    assert '130' in output
    assert '58.40' not in output

def test_woody_savannas_with_stddev():
    """Test that the woody savannas landcover with stddev optional argument returns the expected output."""
    output = run_command(
        ['python', 'classify.py', '--landcover', 'woody savannas', '--stddev']
    )
    assert 'woody savannas' in output
    assert '130' in output
    assert '58.40' in output
