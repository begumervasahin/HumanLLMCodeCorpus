import random
import math

def roll(sides):
    """Simulates rolling a single die with a given number of sides."""
    return random.randint(1, sides)

def form_sample(times, sides):
    """Generates a sample of rolls for a single die."""
    return [roll(sides) for _ in range(times)]

def advantage(rolls_list):
    """Computes the highest value for each roll across multiple dice."""
    highest_rolls = [max(rolls) for rolls in zip(*rolls_list)]
    return highest_rolls

def create_rolls(num_dice, sides, num_rolls):
    """Generates a list of roll samples for a given number of dice."""
    return [form_sample(num_rolls, sides) for _ in range(num_dice)]

def average(data_list):
    """Calculates the average of a list of numbers."""
    return sum(data_list) / len(data_list)

def standard_deviation(data_list, mean):
    """Calculates the mean absolute deviation of a list of numbers."""
    return sum(abs(x - mean) for x in data_list) / len(data_list)

def main():
    print('This program simulates the rolling of multiple dice, where the highest value is always taken.')
    print('Given a number of sides and number of rolls, it returns the average roll and standard deviation in rolls.')
    print('\n')
    
    number_rolls = int(input('Number of rolls: '))
    dice_sides = int(input('Number of sides: '))
    sample_size = int(input('Sample size: '))
    print('\n')
    
    # Generate roll samples and calculate results
    group_of_rolls = create_rolls(number_rolls, dice_sides, sample_size)
    final_sample = advantage(group_of_rolls)
    mu = average(final_sample)
    sigma = standard_deviation(final_sample, mu)
    
    # Display results
    print(f'Average roll: {mu}')
    print(f'Standard deviation: {sigma}')
    print('\n')

if __name__ == "__main__":
    main()
