import os
import csv
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.neural_network import MLPRegressor
from sklearn.preprocessing import StandardScaler
from joblib import Parallel, delayed
import multiprocessing
import statistics

def machine(k, X, y):
    """
    Train and evaluate the MLP regressor model.
    
    Args:
        k (int): Random state for train-test split.
        X (ndarray): Feature matrix.
        y (ndarray): Target vector.
    
    Returns:
        float: Model accuracy on the test set.
    """
    # Split the data into training and testing sets
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=k)
    
    # Initialize and train the MLP regressor
    clf = MLPRegressor(max_iter=1000, learning_rate_init=0.05, solver='sgd', hidden_layer_sizes=(640,), alpha=0.1)
    clf.fit(X_train, y_train)
    
    # Predict and evaluate the model
    predictions = clf.predict(X_test)
    accuracy = clf.score(X_test, y_test)
    
    # Print the results
    print(f"Mean of y_test: {statistics.mean(y_test):.2f}")
    print(f"Predictions: {predictions}")
    print(f"Accuracy: {accuracy:.2f}")
    
    return accuracy

def load_data(filepath):
    """
    Load data from a CSV file.
    
    Args:
        filepath (str): Path to the CSV file.
    
    Returns:
        tuple: Feature matrix and target vector.
    """
    temp1 = []
    temp2 = []
    
    # Read the CSV file and extract relevant data
    with open(filepath, 'rt') as csvfile:
        reader = csv.reader(csvfile)
        next(reader)  # Skip the header row
        for row in reader:
            temp1.append(row[9:17])
            temp2.append(row[22])
    
    return np.array(temp1, dtype=float), np.array(temp2, dtype=int)

def main():
    """
    Main function to execute the script.
    """
    # Define the data path and load the data
    data_path = "/home/geospacial2/PycharmProjects/BTP/"
    file_path = os.path.join(data_path, "rain2014_1.csv")
    
    X, y = load_data(file_path)
    
    # Standardize the features
    X = StandardScaler().fit_transform(X)
    
    # Define the number of iterations and the number of CPU cores
    n_iter = 1
    num_cores = multiprocessing.cpu_count()
    
    # Run the machine function in parallel
    results = Parallel(n_jobs=num_cores)(delayed(machine)(k, X, y) for k in range(n_iter))
    
    # Calculate and print the mean accuracy
    mean_accuracy = statistics.mean(results)
    print(f"Mean accuracy over all iterations: {mean_accuracy:.2f}")

if __name__ == '__main__':
    main()
