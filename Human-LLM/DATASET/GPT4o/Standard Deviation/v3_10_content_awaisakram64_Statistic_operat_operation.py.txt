class Statistics:
    def calculate_median(self, data):
        sorted_data = sorted(data)
        n = len(sorted_data)
        mid_index = n // 2

        if n % 2 == 0:
            median_value = (sorted_data[mid_index - 1] + sorted_data[mid_index]) / 2
        else:
            median_value = sorted_data[mid_index]
        
        return median_value
    
    def calculate_mode(self, data):
        frequency = {}
        for value in data:
            frequency[value] = frequency.get(value, 0) + 1
        
        max_frequency = max(frequency.values())
        
        if max_frequency <= 1:
            return [0]
        
        modes = [key for key, val in frequency.items() if val == max_frequency]
        return modes
    
    def calculate_mean(self, data):
        total_sum = sum(data)
        count = len(data)
        mean_value = total_sum / count
        return mean_value
    
    def calculate_variance(self, data):
        mean_value = self.calculate_mean(data)
        variance_value = sum((x - mean_value) ** 2 for x in data) / len(data)
        return variance_value
    
    def calculate_std_dev(self, data):
        variance_value = self.calculate_variance(data)
        std_dev_value = variance_value ** 0.5
        return std_dev_value

def main():
    statistics = Statistics()
    
    with open("data.txt", 'r') as file:
        data = [float(line.strip()) for line in file]
    
    median_value = statistics.calculate_median(data)
    mean_value = statistics.calculate_mean(data)
    mode_values = statistics.calculate_mode(data)
    variance_value = statistics.calculate_variance(data)
    std_dev_value = statistics.calculate_std_dev(data)
    
    mode_str = ', '.join(map(str, mode_values))
    
    print(f'Median is {median_value:.2f}')
    print(f'Mean is {mean_value:.2f}')
    print(f'Mode is {mode_str}')
    print(f'Variance is {variance_value:.2f}')
    print(f'Standard Deviation is {std_dev_value:.2f}')

if __name__ == "__main__":
    main()
