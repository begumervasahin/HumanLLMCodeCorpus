class Functions:
    def median(self, data):
        data = sorted(data)
        n = len(data)
        if n % 2 == 0:
            mid1, mid2 = data[n // 2 - 1], data[n // 2]
            return (mid1 + mid2) / 2
        else:
            return data[n // 2]
    
    def mode(self, data):
        frequency = {}
        for value in data:
            frequency[value] = frequency.get(value, 0) + 1
        
        max_frequency = max(frequency.values())
        if max_frequency <= 1:
            return [0]
        
        modes = [key for key, val in frequency.items() if val == max_frequency]
        return modes
    
    def mean(self, data):
        return sum(data) / len(data)
    
    def variance(self, data):
        mean_value = self.mean(data)
        return sum((x - mean_value) ** 2 for x in data) / len(data)
    
    def std_dev(self, data):
        return self.variance(data) ** 0.5

def main():
    func = Functions()
    with open("data.txt", 'r') as file:
        data = [float(line.strip()) for line in file]
    
    print(f'Median is {func.median(data):.2f}')
    print(f'Mean is {func.mean(data):.2f}')
    mode_result = func.mode(data)
    mode_str = ', '.join(map(str, mode_result))
    print(f'Mode is {mode_str}')
    print(f'Variance is {func.variance(data):.2f}')
    print(f'Standard Deviation is {func.std_dev(data):.2f}')

if __name__ == "__main__":
    main()
