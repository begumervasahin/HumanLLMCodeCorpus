import csv

# Initialize variables
station = 'ISRO0009_15F009(PRL..Ahmedabad)'
AWS_Data = []
rain = 0
prev_time = 23
err_neg = 0
err_ex = 0
err_mis = 0
time = 0
date = '05/31/2014'

# Open the output file for writing
with open('rain2016.csv', 'w+', newline='') as out_file:
    file_out = csv.writer(out_file)
    # Write the header row to the output file
    file_out.writerow(['@STATION_ID', 'LATITUDE', 'LONGITUDE', 'ALTITUDE(m)', 'TIME(GMT)', 'DATE(GMT)', 
                       'AIR_TEMP(°C)', 'WIND_SPEED(m/s)', 'WIND_DIRECTION(deg)', 'ATMO_PRESSURE(hpa)', 
                       'HUMIDITY(%)', 'RAIN_FALL(mm)', 'ERROR'])

    # Read the input file
    with open('output2016.csv', 'r') as in_file:
        reader = csv.reader(in_file, delimiter=',')
        next(reader)  # Skip the header row
        for row in reader:
            # Only process rows where the rainfall value is not the placeholder value 9999.9
            if float(row[11]) != 9999.9:
                AWS_Data.append(row)

    # Process the AWS data
    for i in range(len(AWS_Data)):
        # Skip initial condition setup for the specified date
        if AWS_Data[i][5] == '05/31/2016':
            prev_time = 23
            rain = float(AWS_Data[i][11])
            continue
        
        # Calculate the current rainfall value
        temp = float(AWS_Data[i][11])
        if rain - temp > 924:
            AWS_Data[i][11] = temp + (1023 - rain)
        else:
            AWS_Data[i][11] = temp - rain
        
        rain = temp
        
        # Check for negative or excessive rainfall values and append the appropriate error
        if float(AWS_Data[i][11]) < 0:
            AWS_Data[i].append('Neg')
            err_neg += 1
        elif float(AWS_Data[i][11]) > 100:
            AWS_Data[i].append('Excess')
            err_ex += 1
        
        # Write the processed row to the output file
        file_out.writerow(AWS_Data[i])

# Print the error counts
print(f"Negative Rainfall Errors: {err_neg}")
print(f"Excessive Rainfall Errors: {err_ex}")
