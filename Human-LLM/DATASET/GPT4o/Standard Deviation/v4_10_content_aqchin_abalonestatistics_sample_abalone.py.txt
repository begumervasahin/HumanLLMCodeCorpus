import numpy as np
import random
import matplotlib.pyplot as plt

# Set random seed for reproducibility
random.seed(0)

# List of different run counts to test
run_tests = [10, 100, 1000, 10000, 100000]

# Initialize lists to store results
means = []
stds = []
runs = []
scale = 0.1

def main():
    # Read the abalone.data file to count the number of rows
    with open('abalone.data') as file:
        num_rows = sum(1 for _ in file)
        data_in = [0] * num_rows

    # Run the simulation for each number of runs in run_tests
    for r in run_tests:
        data_in = [0] * len(data_in)
        
        # Perform the simulation for r runs
        for i in range(r):
            row_len = len(data_in)
            n_to_select = float(int(row_len * scale))
            n_remaining = row_len
            
            # Randomly increment elements in data_in
            for j in range(row_len):
                if n_to_select / n_remaining > random.random():
                    data_in[j] += 1
                    n_to_select -= 1
                n_remaining -= 1
        
        # Calculate mean and standard deviation
        means.append(np.mean(data_in) / r)
        stds.append(np.std(data_in) / r)
        runs.append(r)
    
    # Plotting the results
    plt.subplot(2, 1, 1)
    plt.plot(runs, means, marker='o')
    plt.title("Mean VS Runs")
    plt.ylabel("Mean")
    
    plt.subplot(2, 1, 2)
    plt.plot(runs, stds, marker='o')
    plt.title("Standard Deviation VS Runs")
    plt.xlabel("Runs")
    plt.ylabel("Standard Deviation")
    
    plt.tight_layout()
    plt.show()

# Run the main function
main()
