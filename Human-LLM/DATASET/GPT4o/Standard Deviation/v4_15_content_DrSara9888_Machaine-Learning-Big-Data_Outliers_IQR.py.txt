import numpy as np
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

# Load the dataset
data = pd.read_csv("/Users/khadeejah/Desktop/weight-height.csv")

# Display the first few rows of the dataset
print(data.head())

# Scatter plot of Weight vs Height
plt.figure(figsize=(20, 8))
plt.scatter(data['Weight'], data['Height'])
plt.xlabel('Weight')
plt.ylabel('Height')
plt.title('Scatter plot of Weight vs Height')
plt.show()

# Descriptive statistics of the dataset
print(data.describe())

# Distribution plots for Height
plt.figure(figsize=(20, 8))

plt.subplot(1, 3, 1)
sns.histplot(data['Height'], kde=True)
plt.title('Height Distribution')
plt.ylabel('Distribution')

plt.subplot(1, 3, 2)
sns.boxplot(y=data['Height'])
plt.title('Boxplot of Height')

plt.subplot(1, 3, 3)
plt.hist(data['Height'], bins=20, rwidth=0.8)
plt.title('Histogram of Height')
plt.xlabel('Height')
plt.ylabel('Count')

plt.tight_layout()
plt.show()

# Calculate the IQR and identify outliers for Height
Q1 = data['Height'].quantile(0.25)
Q3 = data['Height'].quantile(0.75)
IQR = Q3 - Q1
lower_limit = Q1 - 1.5 * IQR
upper_limit = Q3 + 1.5 * IQR

print(f"Lower limit for Height: {lower_limit}, Upper limit for Height: {upper_limit}")

# Filter out the outliers in Height
data_outlier_height = data[(data['Height'] < lower_limit) | (data['Height'] > upper_limit)]
data_no_outlier_height = data[(data['Height'] >= lower_limit) & (data['Height'] <= upper_limit)]

print(data_no_outlier_height.head(3))
