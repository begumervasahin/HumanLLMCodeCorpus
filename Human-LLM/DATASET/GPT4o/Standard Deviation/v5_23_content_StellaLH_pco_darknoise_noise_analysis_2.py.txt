import numpy as np
import h5py

# Define the parameters
frames_list = [100, 1000, 10000]
exposure_times = [0.001]
pixel_rates = ['slow', 'fast']

# Function to calculate statistics from data
def calculate_statistics(data):
    mean = np.mean(data, axis=0)
    std = np.std(data, axis=0)
    minimum = np.min(data, axis=0)
    maximum = np.max(data, axis=0)
    return mean, std, minimum, maximum

# Function to calculate grand statistics
def calculate_grand_statistics(stat_list, chunks):
    grand_mean = np.mean(stat_list['mean'], axis=0)
    grand_min = np.min(stat_list['min'], axis=0)
    grand_max = np.max(stat_list['max'], axis=0)
    total_variance = sum((std ** 2) for std in stat_list['std']) / chunks
    grand_std = np.sqrt(total_variance)
    return grand_mean, grand_std, grand_min, grand_max

# Function to process each file and gather statistics
def process_files(frame_count, rate):
    stat_list = {'mean': [], 'std': [], 'min': [], 'max': []}
    chunks = frame_count // 100
    chunk_numbers = ["%06d" % i for i in range(1, chunks + 1)]

    for chunk_num in chunk_numbers:
        file_name = f'{frame_count}_frames_{rate}_{chunk_num}.h5'
        with h5py.File(file_name, 'r') as fid:
            data = fid['/entry/data']
            mean, std, min_val, max_val = calculate_statistics(data)
            stat_list['mean'].append(mean)
            stat_list['std'].append(std)
            stat_list['min'].append(min_val)
            stat_list['max'].append(max_val)
    
    return calculate_grand_statistics(stat_list, chunks)

# Main loop through each combination of parameters
for frame_count in frames_list:
    for exposure in exposure_times:
        for rate in pixel_rates:
            print(f"Processing {frame_count} frames, {exposure}s exposure, {rate} rate")

            grand_mean, grand_std, grand_min, grand_max = process_files(frame_count, rate)

            # Save the final results
            np.save(f'mean_{frame_count}frames_{exposure}s_{rate}.npy', grand_mean)
            np.save(f'std_{frame_count}frames_{exposure}s_{rate}.npy', grand_std)
            np.save(f'min_{frame_count}frames_{exposure}s_{rate}.npy', grand_min)
            np.save(f'max_{frame_count}frames_{exposure}s_{rate}.npy', grand_max)
