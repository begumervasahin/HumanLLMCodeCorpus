import re
import numpy as np
import os

# Configuration
PIN_OUTPUT_DIR = "outputs/"
COMBINED_OUTPUT_FILE = "combined.out"
PIN_SPECIFIC_SEARCH_TERMS = [
    "Total number of instructions executed",
    "Number of load instructions",
    "Number of store instructions",
    "Number of branch instructions"
]
DEBUG_MODE = False

def combine_files(input_dir, output_file):
    """
    Combines all files in the specified directory into a single output file.

    Args:
        input_dir (str): Directory containing the input files.
        output_file (str): Path to the output file.
    """
    with open(output_file, 'w') as out_f:
        for root, _, files in os.walk(input_dir):
            for file in files:
                file_path = os.path.join(root, file)
                if DEBUG_MODE:
                    print(f"Processing file: {file_path}")
                with open(file_path, 'r') as in_f:
                    for line in in_f:
                        out_f.write(line)

def compute_statistics(stat_file, search_terms):
    """
    Computes the standard deviation and mean for specified search terms in the stat file.

    Args:
        stat_file (str): Path to the file containing the statistics.
        search_terms (list): List of terms to search for in the file.

    Returns:
        str: A formatted string containing the computed statistics.
    """
    runtimes = {term: [] for term in search_terms}

    with open(stat_file, "r") as f:
        for line in f:
            for term in search_terms:
                if term in line:
                    numeric_value = int(re.findall(r"\d+", line)[0])
                    if DEBUG_MODE:
                        print(f"Found value {numeric_value} for term: {term}")
                    runtimes[term].append(numeric_value)

    print("Parsing complete!")

    output_str = "Type of instruction: \t std, \t mean \n"
    for term, values in runtimes.items():
        std = np.std(values)
        mean = np.mean(values)
        output_str += f"{term} :\t{std},\t{mean}\n"

    print(output_str)
    return output_str

def main():
    """
    Main function to combine files and compute statistics.
    """
    combine_files(PIN_OUTPUT_DIR, COMBINED_OUTPUT_FILE)
    stats_report = compute_statistics(COMBINED_OUTPUT_FILE, PIN_SPECIFIC_SEARCH_TERMS)
    
    with open("pin_report.txt", "w") as report:
        report.write(stats_report)

if __name__ == '__main__':
    main()
