import numpy as np
import math

def get_scaling_factor(n):
    """
    Compute the scaling factor used for unbiased estimation of standard deviation.
    
    Parameters:
    n (int): The sample size.
    
    Returns:
    float: The scaling factor.
    """
    k = n // 2
    if n % 2 == 0:
        # For even sample sizes
        numerator = (2 ** (2 * k - 2)) * (math.factorial(k - 1) ** 2)
        denominator = math.factorial(2 * k - 2)
        scaling_factor = np.sqrt(2 / (np.pi * (2 * k - 1))) * (numerator / denominator)
    else:
        # For odd sample sizes
        numerator = math.factorial(2 * k - 1)
        denominator = (2 ** (2 * k - 1)) * (math.factorial(k - 1) ** 2)
        scaling_factor = np.sqrt(np.pi / k) * (numerator / denominator)
    
    return scaling_factor

def ustd(x):
    """
    Compute the unbiased estimation of standard deviation.
    
    Parameters:
    x (ndarray): The input data array.
    
    Returns:
    float or ndarray: The unbiased standard deviation.
    """
    if x.ndim == 1:
        n = x.size
        unbiased_variance = np.var(x, ddof=1)
    else:
        n = x.shape[1]
        unbiased_variance = np.var(x, ddof=1, axis=1)
    
    scaling_factor = get_scaling_factor(n)
    unbiased_std = np.sqrt(unbiased_variance) / scaling_factor
    
    return unbiased_std

def test_scaling_factor():
    """
    Test the get_scaling_factor function for known values.
    """
    assert abs(get_scaling_factor(2) - 0.7978845608) < 1e-6
    assert abs(get_scaling_factor(3) - 0.8862269255) < 1e-6
    assert abs(get_scaling_factor(4) - 0.9213177319) < 1e-6

def test_ustd():
    """
    Test the ustd function with a large sample of normally distributed data.
    """
    x = np.random.normal(size=(100000, 4))
    sigma = ustd(x)
    assert abs(np.mean(sigma) - 1) < 0.005

if __name__ == "__main__":
    # Run tests for the scaling factor
    test_scaling_factor()

    # Run tests for the unbiased standard deviation
    test_ustd()

    print("All tests passed successfully.")
