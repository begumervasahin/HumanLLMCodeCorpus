import pandas as pd
from pandas.plotting import register_matplotlib_converters
import matplotlib.pyplot as plt
import matplotlib.dates as mdates
import os
import datetime

class PolarRRI:
    def __init__(self, file_path=None):
        if file_path is None:
            self.rri_df = pd.DataFrame()
            return

        self.rri_df = self._load_rri_data(file_path)

    def _load_rri_data(self, file_path):
        file_name = os.path.basename(file_path)
        src_df = pd.read_csv(file_path, header=None, sep=' ')
        src_df = src_df.rename(columns={0: 'time', 1: 'rri'})

        # Convert RRI values to milliseconds
        src_df['rri'] = src_df['rri'] * 1000

        # Extract date and time from file name
        start_datetime = self._extract_datetime_from_filename(file_name)

        # Calculate datetime for each RRI entry
        src_df['datetime'] = src_df.apply(lambda x: start_datetime + datetime.timedelta(seconds=x['time']), axis=1)

        return src_df.set_index(['datetime'])

    def _extract_datetime_from_filename(self, file_name):
        file_date, file_time = file_name.split('.')[0].split('_')
        _, year, month, day = file_date.split('-')
        hour, minute, second = file_time.split('-')
        return datetime.datetime(int(year), int(month), int(day), int(hour), int(minute), int(second))

    def plot_rri(self):
        register_matplotlib_converters()
        fig, ax = plt.subplots()
        ax.xaxis.set_major_formatter(mdates.DateFormatter('%H:%M:%S'))
        ax.plot(self.rri_df.index, self.rri_df['rri'], marker='.')
        plt.show()

    def del_outliers(self, thresh_min, thresh_max):
        self.rri_df['rri'] = self.rri_df['rri'].where((self.rri_df['rri'] > thresh_min) & (self.rri_df['rri'] < thresh_max))

    def get_rri_df(self):
        return self.rri_df

    def write(self, file_path):
        self.rri_df['rri'] = self.rri_df['rri'].astype(int)
        self.rri_df.to_csv(file_path)

    def import_rri(self, file_path):
        new_rri_df = pd.read_csv(file_path)
        new_rri_df['datetime'] = pd.to_datetime(new_rri_df['datetime'])
        new_rri_df = new_rri_df.set_index(['datetime'])
        self.rri_df = pd.concat([self.rri_df, new_rri_df])
