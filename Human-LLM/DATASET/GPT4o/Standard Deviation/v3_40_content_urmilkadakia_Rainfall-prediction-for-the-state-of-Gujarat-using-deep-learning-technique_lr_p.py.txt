import csv
import os
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
import numpy as np
from sklearn.preprocessing import StandardScaler
from joblib import Parallel, delayed
import multiprocessing
import statistics

def machine(random_state, features, targets):
    """
    Train a Linear Regression model and evaluate its performance.

    Parameters:
    - random_state (int): Random state for reproducibility.
    - features (array-like): Feature matrix.
    - targets (array-like): Target vector.
    """
    # Split the data into training and testing sets
    X_train, X_test, y_train, y_test = train_test_split(features, targets, test_size=0.2, random_state=random_state)
    
    # Convert data to the appropriate types
    X_train = np.array(X_train, dtype=float)
    X_test = np.array(X_test, dtype=float)
    y_train = np.array(y_train, dtype=int)
    y_test = np.array(y_test, dtype=int)
    
    # Initialize and train the Linear Regression model
    model = LinearRegression()
    print(f"Mean of y_test: {statistics.mean(y_test)}")
    model.fit(X_train, y_train)
    
    # Make predictions and display results
    predictions = model.predict(X_test)
    print(f"Predictions: {predictions}")
    print(f"Coefficients: {model.coef_}")
    print(f"Intercept: {model.intercept_}")
    
    # Calculate and print the accuracy of the model
    accuracy = model.score(X_test, y_test)
    print(f"Accuracy: {accuracy}")

def read_csv_data(file_path):
    """
    Read data from a CSV file.

    Parameters:
    - file_path (str): Path to the CSV file.

    Returns:
    - features (list): List of features from the CSV file.
    - targets (list): List of target values from the CSV file.
    """
    features, targets = [], []
    with open(file_path, 'rt') as csvfile:
        reader = csv.reader(csvfile)
        next(reader)  # Skip the header
        for row in reader:
            features.append(row[9:17])
            targets.append(row[22])
    return features, targets

if __name__ == '__main__':
    # Path to the CSV file
    data_path = "/home/geospacial2/PycharmProjects/BTP/rain2014_1.csv"
    
    # Read the CSV file
    features, targets = read_csv_data(data_path)
    
    # Scale the features
    features = StandardScaler().fit_transform(features)
    
    # Set the number of iterations
    num_iterations = 1
    random_states = range(num_iterations)
    
    # Get the number of CPU cores
    num_cores = multiprocessing.cpu_count()
    
    # Run the machine function in parallel
    Parallel(n_jobs=num_cores)(delayed(machine)(state, features, targets) for state in random_states)
