import numpy as np

# Constants
RADIUS_OF_EARTH = 6371.0088  # Radius of the Earth in kilometers
KMS_TO_METERS = 1000.0
METERS_TO_KILOMETERS = 0.001
MILES_TO_KILOMETERS = 1.60934
KILOMETERS_TO_MILES = 0.621371

def great_circle_distance(loc1, loc2, radius=RADIUS_OF_EARTH * KMS_TO_METERS):
    """
    Calculates the great circle distance between two points on the Earth using the Haversine formula.

    Parameters:
    loc1 (tuple): (Longitude, Latitude) of the first location in decimal degrees
    loc2 (tuple): (Longitude, Latitude) of the second location in decimal degrees
    radius (float): Radius of the Earth in meters (default is Earth's radius in meters)

    Returns:
    float: Great circle distance between loc1 and loc2 in the units of the radius

    Notes:
    This calculation assumes a perfect sphere and does not take into account the Earth's ellipsoidal shape.
    
    Example:
    >>> san_diego = (-117.1611, 32.7157)
    >>> austin = (-97.7431, 30.2672)
    >>> great_circle_distance(san_diego, austin)
    1155.474644164695
    """
    lon1, lat1 = loc1
    lon2, lat2 = loc2
    
    # Convert degrees to radians
    d_lat = np.radians(lat2 - lat1)
    d_lon = np.radians(lon2 - lon1)
    lat1 = np.radians(lat1)
    lat2 = np.radians(lat2)
    
    # Haversine formula
    a = np.sin(d_lat / 2)**2 + np.cos(lat1) * np.cos(lat2) * np.sin(d_lon / 2)**2
    c = 2 * np.arcsin(np.sqrt(a))
    
    return radius * c

def compare_floats(a, b, relative_tolerance=0.00001, absolute_tolerance=0.00000001):
    """
    Compares two floating-point numbers within a given tolerance.

    Parameters:
    a (float): First number to compare
    b (float): Second number to compare
    relative_tolerance (float): Relative tolerance (default is 0.00001)
    absolute_tolerance (float): Absolute tolerance (default is 0.00000001)

    Returns:
    bool: True if the numbers are close within the specified tolerances, otherwise False
    """
    try:
        return abs(a - b) < absolute_tolerance + (relative_tolerance * abs(b))
    except:
        return False

def degrees_to_radians(degree):
    """
    Converts an angle from degrees to radians.

    Parameters:
    degree (float): Angle in degrees

    Returns:
    float: Angle in radians
    """
    return np.pi / 180.0 * degree

def radians_to_degrees(radians):
    """
    Converts an angle from radians to degrees.

    Parameters:
    radians (float): Angle in radians

    Returns:
    float: Angle in degrees
    """
    return 180.0 / np.pi * radians
