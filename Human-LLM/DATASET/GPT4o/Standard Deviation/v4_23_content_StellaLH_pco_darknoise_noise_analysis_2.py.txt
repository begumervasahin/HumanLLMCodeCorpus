import numpy as np
import h5py

# Define the parameters
frames = [100, 1000, 10000]
exp_time = [0.001]
pix_rate = ['slow', 'fast']

# Loop through each combination of parameters
for frame_count in frames:
    for exposure in exp_time:
        for rate in pix_rate:
            print(frame_count, exposure, rate)

            mean_values = []
            std_values = []
            min_values = []
            max_values = []

            chunks = frame_count // 100
            chunk_numbers = ["%06d" % i for i in range(1, chunks + 1)]

            for chunk_num in chunk_numbers:
                file_name = f'{frame_count}_frames_{rate}_{chunk_num}.h5'
                with h5py.File(file_name, 'r') as fid:
                    data = fid['/entry/data']
                    mean_values.append(np.mean(data, axis=0))
                    std_values.append(np.std(data, axis=0))
                    min_values.append(np.min(data, axis=0))
                    max_values.append(np.max(data, axis=0))

            # Calculate grand statistics
            grand_mean = np.mean(mean_values, axis=0)
            grand_min = np.min(min_values, axis=0)
            grand_max = np.max(max_values, axis=0)

            # Calculate grand standard deviation
            total_variance = sum((std ** 2) for std in std_values) / chunks
            grand_std = np.sqrt(total_variance)

            # Save the final results
            np.save(f'mean_{frame_count}frames_{exposure}s_{rate}.npy', grand_mean)
            np.save(f'std_{frame_count}frames_{exposure}s_{rate}.npy', grand_std)
            np.save(f'min_{frame_count}frames_{exposure}s_{rate}.npy', grand_min)
            np.save(f'max_{frame_count}frames_{exposure}s_{rate}.npy', grand_max)
