import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import statsmodels.formula.api as smf
import csv

# Function to read and process the CSV file
def read_csv(file_path):
    payroll_list = []
    overtime_pay = []
    
    with open(file_path, 'r') as file:
        csv_reader = csv.reader(file)
        next(csv_reader)  # Skip header row
        for row in csv_reader:
            payroll_list.append(float(row[7]))
            overtime_pay.append(float(row[3]))
    
    return payroll_list, overtime_pay

# Function to calculate and print statistics
def calculate_statistics(payroll_list):
    mean_payroll = np.mean(payroll_list)
    std_dev_payroll = np.std(payroll_list)
    
    print("Mean: ", mean_payroll)
    print("Standard Deviation: ", std_dev_payroll)

# Function to perform linear regression
def perform_regression(data):
    lm = smf.ols(formula='TPB ~ BP', data=data).fit()
    print("The linear regression model is: ", lm.params[0], "+ ", lm.params[1], "*BP")

# Function to plot histogram
def plot_histogram(payroll_list, bin_array):
    plt.hist(payroll_list, bins=bin_array)
    plt.xlabel('Total Pay and Benefits for 2014')
    plt.ylabel('Number of people')
    plt.title('Distribution of Total Pay and Benefits for 2014')
    plt.show()

# Main script
if __name__ == "__main__":
    file_path = 'san-francisco-2014.csv'
    
    # Read and process the CSV file
    payroll_list, overtime_pay = read_csv(file_path)
    
    # Calculate and print statistics
    calculate_statistics(payroll_list)
    
    # Create bins for histogram
    bin_array = np.arange(0, 530000, 1000)
    
    # Perform linear regression
    data = pd.read_csv(file_path, index_col=0)
    perform_regression(data)
    
    # Plot histogram
    plot_histogram(payroll_list, bin_array)
