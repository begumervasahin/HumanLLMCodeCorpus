import numpy as np
import cv2
from datetime import datetime

def distMap(frame1, frame2):
    """Calculate the Pythagorean distance between two frames."""
    frame1_32 = np.float32(frame1)
    frame2_32 = np.float32(frame2)
    diff32 = frame1_32 - frame2_32
    num = np.sqrt(diff32[:, :, 0]**2 + diff32[:, :, 1]**2 + diff32[:, :, 2]**2)
    den = np.sqrt(255**2 + 255**2 + 255**2)
    norm32 = num / den
    dist = np.uint8(norm32 * 255)
    return dist

# Set the threshold for motion detection
sdThresh = 5
font = cv2.FONT_HERSHEY_SIMPLEX

# Create windows to display the video feed and distance map
cv2.namedWindow('frame')
cv2.namedWindow('dist')

# Capture video from the webcam
cap = cv2.VideoCapture(0, cv2.CAP_DSHOW)

# Read the first two frames
_, frame1 = cap.read()
_, frame2 = cap.read()

while cap.isOpened():
    ret, frame3 = cap.read()
    if ret:
        # Calculate the distance map between the current frame and the reference frame
        dist = distMap(frame1, frame3)
        
        # Update the reference frames
        frame1 = frame2
        frame2 = frame3
        
        # Apply Gaussian blur to the distance map
        mod = cv2.GaussianBlur(dist, (7, 7), 0)
        
        # Threshold the blurred distance map
        _, thresh = cv2.threshold(mod, 100, 255, 0)
        
        # Calculate the standard deviation of the thresholded image
        _, stDev = cv2.meanStdDev(thresh)
        
        # Display the blurred distance map
        cv2.imshow('dist', mod)
        
        # Display the standard deviation on the video feed
        cv2.putText(frame1, "Standard Deviation - {}".format(round(stDev[0][0], 0)), (70, 70), font, 1, (255, 0, 255), 1, cv2.LINE_AA)
        
        # Check if the standard deviation exceeds the threshold
        if stDev > sdThresh:
            print('Motion Detected: ', datetime.now().strftime("%B %d %Y %H:%M:%S"))
        
        # Display the current frame
        cv2.imshow('frame', frame2)
    else:
        # Reset the video capture to the beginning if no frame is read
        cap.set(cv2.CAP_PROP_POS_FRAMES, 0)
    
    # Break the loop if the 'Esc' key is pressed
    if cv2.waitKey(1) & 0xFF == 27:
        break

# Release the video capture and destroy all windows
cap.release()
cv2.destroyAllWindows()
