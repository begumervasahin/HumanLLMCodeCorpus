import csv
import numpy as np
import math
import pandas as pd
import xlsxwriter

def resistance(ticker, no_of_intervals):
    """
    Calculate resistance levels for a given stock ticker.

    Parameters:
    ticker (str): Stock ticker symbol
    no_of_intervals (int): Number of intervals for resistance calculation

    Returns:
    tuple: (ticker, interval_data) where interval_data is a list of tuples representing intervals and their frequencies
    """
    df = pd.read_csv(f"C:/Users/Hp/Desktop/New folder/Top500_stock/Original data/{ticker}.csv")
    if len(df.columns) < 6:
        return None

    high_data = df['High'].tolist()
    maxi = math.ceil(max(high_data))
    mini = math.floor(min(high_data))
    interval = np.around(np.linspace(mini, maxi, no_of_intervals + 1), 2).tolist()

    interval_data = []
    max_count = 0
    max_position = mini

    for i in interval[:-1]:
        count = sum(1 for j in high_data if i <= j < interval[interval.index(i) + 1])
        if count > max_count:
            max_count = count
            max_position = i
        interval_data.append((i, interval[interval.index(i) + 1], count))

    interval_data.append((max_position, interval[interval.index(max_position) + 1], max_count))
    return ticker, interval_data

def get_ticker_list(file_path):
    """
    Retrieve the list of stock tickers from a CSV file.

    Parameters:
    file_path (str): Path to the CSV file containing stock tickers

    Returns:
    list: List of stock ticker symbols
    """
    ticker_list = []
    with open(file_path, 'r') as f:
        reader = csv.reader(f)
        next(reader)  # Skip header
        for line in reader:
            ticker_list.append(line[2])
    return ticker_list

def main():
    input_file_path = r"C:\Users\Hp\Desktop\New folder\nifty500_list.csv"
    output_file_path = r"C:/Users/Hp/Desktop/New folder/Top500_stock/resistance.xlsx"
    no_of_intervals = 10

    ticker_list = get_ticker_list(input_file_path)
    results = []

    for ticker in ticker_list:
        ticker = ticker.replace('&', '%26')
        print(f"Processing: {ticker}")
        result = resistance(ticker, no_of_intervals)
        if result:
            results.append(result)

    if results:
        tickers, intervals_data = zip(*results)
        columns = [f"INTERVAL NO. {i}" for i in range(1, no_of_intervals + 1)] + ["MAX FREQ."]
        sdf = pd.DataFrame(intervals_data, index=pd.Series(tickers, name="SYMBOL"), columns=columns)

        # Save to Excel
        with pd.ExcelWriter(output_file_path, engine="xlsxwriter") as writer:
            sdf.to_excel(writer, sheet_name="Sheet1")

if __name__ == "__main__":
    main()
