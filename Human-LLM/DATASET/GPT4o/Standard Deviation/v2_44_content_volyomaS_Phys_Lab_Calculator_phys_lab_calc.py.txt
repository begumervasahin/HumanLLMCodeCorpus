def convert_to_floats(data):
    """Convert all elements in the list to floats."""
    return [float(item) for item in data]

def count_occurrences(data):
    """Convert list elements to floats and count their occurrences in a dictionary."""
    if not isinstance(data, list):
        return None
    
    float_data = [float(item) for item in data]
    occurrences = {}
    for item in float_data:
        occurrences[item] = occurrences.get(item, 0) + 1
    return occurrences

def print_occurrences(occurrences):
    """Print the dictionary of occurrences in a sorted manner."""
    for key in sorted(occurrences.keys()):
        if key == 1.8:
            print(f"{key}: {' *' * occurrences[key]}")
        else:
            print(f"{key}: {'*' * occurrences[key]}")

def calculate_average(data):
    """Calculate the average of a list of numbers."""
    return sum(data) / len(data)

def calculate_dispersion(data):
    """Calculate the dispersion (standard deviation) of a list of numbers."""
    avg = calculate_average(data)
    variance = sum((x - avg) ** 2 for x in data) / len(data)
    return variance ** 0.5

def calculate_standard_deviation(data):
    """Calculate the standard deviation of a list of numbers."""
    avg = calculate_average(data)
    variance = sum((x - avg) ** 2 for x in data) / (len(data) - 1)
    return variance ** 0.5

def calculate_random_error(data, k_student):
    """Calculate the random error using the standard deviation and the k-Student value."""
    return calculate_standard_deviation(data) * k_student

def calculate_absolute_error(instrument_error, computational_error, random_error):
    """Calculate the absolute error given the instrument, computational, and random errors."""
    return (instrument_error ** 2 + computational_error ** 2 + random_error ** 2) ** 0.5

# Example usage
data = [1.0, 2.0, 1.8, 3.0, 1.8]
float_data = convert_to_floats(data)
occurrences = count_occurrences(float_data)
print_occurrences(occurrences)
print("Average:", calculate_average(float_data))
print("Dispersion:", calculate_dispersion(float_data))
print("Standard Deviation:", calculate_standard_deviation(float_data))
k_student = 2.262  # Example value for 95% confidence level and small sample size
random_err = calculate_random_error(float_data, k_student)
print("Random Error:", random_err)
instrument_error = 0.05
computational_error = 0.02
absolute_err = calculate_absolute_error(instrument_error, computational_error, random_err)
print("Absolute Error:", absolute_err)
