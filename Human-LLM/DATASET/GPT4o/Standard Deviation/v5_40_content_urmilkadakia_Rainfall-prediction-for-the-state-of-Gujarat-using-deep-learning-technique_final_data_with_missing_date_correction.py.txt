import pandas as pd
from dateutil.parser import parse
import csv

def is_consecutive_hour(current_time, previous_time):
    return int(current_time) - int(previous_time) == 1

def is_end_of_day_to_start_of_next(current_time, previous_time):
    return int(current_time) == 0 and int(previous_time) == 23

def is_same_station(current_row, previous_row):
    return current_row['@STATION_ID'] == previous_row['@STATION_ID']

def is_same_month(current_date, previous_date):
    return current_date.month == previous_date.month

def is_same_day(current_date, previous_date):
    return current_date.day == previous_date.day

def is_month_end_to_start(current_date, previous_date):
    if previous_date.month in [7, 8]:  # Handle 31-day months
        return current_date.day == 1 and previous_date.day == 31
    else:  # Handle 30-day months
        return current_date.day == 1 and previous_date.day == 30

# Read the CSV file into a DataFrame
df = pd.read_csv('final_rain_2014.csv')

# Print the first row of the DataFrame
print(df.loc[0])

# Initialize a list to store indices of missing data
missing = []

# Iterate through the DataFrame rows
for i in range(1, len(df)):
    current_row = df.loc[i]
    previous_row = df.loc[i - 1]
    current_date = parse(current_row['DATE(GMT)'])
    previous_date = parse(previous_row['DATE(GMT)'])
    current_time = current_row['TIME(GMT)']
    previous_time = previous_row['TIME(GMT)']
    
    if is_same_station(current_row, previous_row):
        if is_same_month(current_date, previous_date):
            if is_same_day(current_date, previous_date):
                if not is_consecutive_hour(current_time, previous_time):
                    missing.append(i)
            else:
                if not is_end_of_day_to_start_of_next(current_time, previous_time):
                    missing.append(i)
        else:
            if is_month_end_to_start(current_date, previous_date):
                if not is_end_of_day_to_start_of_next(current_time, previous_time):
                    missing.append(i)
            else:
                missing.append(i)

# Write the filtered data to a new CSV file
with open('RAIN.csv', 'w', newline='') as file_out:
    writer = csv.writer(file_out)
    writer.writerow(['@STATION_ID', 'LATITUDE', 'LONGITUDE', 'TIME(GMT)', 'DATE(GMT)', 'RAIN_FALL(mm)'])
    
    for i in range(len(df)):
        if i not in missing:
            row = [
                df.loc[i, '@STATION_ID'], 
                df.loc[i, 'LATITUDE'], 
                df.loc[i, 'LONGITUDE'], 
                df.loc[i, 'TIME(GMT)'], 
                df.loc[i, 'DATE(GMT)'], 
                df.loc[i, 'RAIN_FALL(mm)']
            ]
            writer.writerow(row)
