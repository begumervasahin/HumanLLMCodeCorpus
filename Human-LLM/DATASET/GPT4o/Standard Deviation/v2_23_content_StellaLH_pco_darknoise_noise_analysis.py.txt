import numpy as np
import h5py

# Define parameters
frames = [100, 1000, 10000]
exp_times = [0.001, 0.01, 0.1, 1]
pixel_rates = ['slow', 'fast']

# Process data for each combination of parameters
for frame_count in frames:
    for exp_time in exp_times:
        for pixel_rate in pixel_rates:
            mean_list = []
            std_list = []
            min_list = []
            max_list = []
            chunk_count = frame_count // 100
            file_numbers = ["%06d" % num for num in range(1, chunk_count + 1)]
            
            # Read data from each file and compute statistics
            for file_num in file_numbers:
                filename = f'{frame_count}_frames_{exp_time}_{pixel_rate}_{file_num}'
                with h5py.File(filename, 'r') as file:
                    data = file['/entry/data']
                    mean_list.append(np.mean(data[0:99], axis=0))
                    std_list.append(np.std(data[0:99], axis=0))
                    min_list.append(np.min(data[0:99], axis=0))
                    max_list.append(np.max(data[0:99], axis=0))

            # Calculate grand statistics across all files
            grand_mean = np.mean(mean_list)
            grand_std = np.sqrt(np.sum([std**2 / chunk_count for std in std_list]))
            grand_min = np.min(min_list)
            grand_max = np.max(max_list)

            # Save the grand statistics to .npy files
            np.save(f'mean_{frame_count}frames_{exp_time}s_{pixel_rate}.npy', grand_mean)
            np.save(f'std_{frame_count}frames_{exp_time}s_{pixel_rate}.npy', grand_std)
            np.save(f'min_{frame_count}frames_{exp_time}s_{pixel_rate}.npy', grand_min)
            np.save(f'max_{frame_count}frames_{exp_time}s_{pixel_rate}.npy', grand_max)

print("Data processing complete.")
