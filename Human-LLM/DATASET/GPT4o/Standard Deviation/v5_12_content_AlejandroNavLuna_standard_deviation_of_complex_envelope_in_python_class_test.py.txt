import numpy as np
import math

class DesvEstandar:
    """Descripción del bloque"""
    
    def __init__(self, name="e_desv_estandar_cf", in_sig=[np.float32], out_sig=[np.float32]):
        self.contador = 0
        self.acum = 0

    def work(self, input_items, output_items):
        in0 = input_items[0]
        in1 = input_items[1]
        out0 = output_items[0]
        
        # Filtering the input arrays
        mask = (0.5 < in0) & (in0 < 1.5) & (in1 < 0.5) & (in1 > -0.5)
        filtered_in0 = in0[mask]
        filtered_in1 = in1[mask]
        
        # Calculating mean and variance for filtered data
        mean_a1 = np.mean(filtered_in0)
        var_a1 = np.var(filtered_in0, ddof=1)
        
        mean_b1 = np.mean(filtered_in1)
        var_b1 = np.var(filtered_in1, ddof=1)
        
        # Calculating the combined standard deviation
        combined_std_dev = np.sqrt(var_a1 + var_b1)
        out0[:] = combined_std_dev
        return out0

def generate_random_data(m, r, n, fvar):
    """Generates random data for the algorithm"""
    pi = math.pi
    datos = np.round((m - 1) * np.random.rand(r, n))
    angles = datos * 2 * pi / m
    mag = fvar * np.random.normal(0, 1, n) + 1
    a = mag * np.cos(angles + fvar * np.random.normal(0, 1, n))
    b = mag * np.sin(angles + fvar * np.random.normal(0, 1, n))
    return a, b

# Create an instance of the DesvEstandar class
desv_estandar_instance = DesvEstandar()

# Parameters for data generation
m = 2
r = 1
n = 20
fvar = 0.1

# Generating random data
a, b = generate_random_data(m, r, n, fvar)

# Calculating the standard deviation
ro = desv_estandar_instance.work([a, b], [0.])
print(ro)
