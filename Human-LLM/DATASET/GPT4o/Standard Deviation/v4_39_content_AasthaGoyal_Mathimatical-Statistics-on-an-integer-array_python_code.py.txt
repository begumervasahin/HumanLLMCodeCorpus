import math

def get_user_input():
    """Get a list of numbers from the user."""
    return input("Enter a list of numbers separated by commas: ")

def validate_and_process_input(input_str):
    """Validate input and separate valid integers from errors."""
    stat = []
    error_list = []
    temp_list = input_str.split(",")
    
    for item in temp_list:
        item = item.strip()
        if item.isdigit():
            stat.append(int(item))
        elif item.lower() == "terminate":
            print("You terminated the program")
            return stat, error_list, True
        else:
            print(f"{item} is not an integer value")
            error_list.append(item)
    
    return stat, error_list, False

def sort_numbers(numbers):
    """Sort a list of numbers in ascending order."""
    return sorted(numbers)

def calculate_statistics(numbers):
    """Calculate and print various statistics for a list of numbers."""
    if not numbers:
        print("No valid numbers to calculate statistics.")
        return

    maximum = max(numbers)
    minimum = min(numbers)
    range_value = maximum - minimum
    total_count = len(numbers)
    mean = sum(numbers) / total_count
    variance = sum((x - mean) ** 2 for x in numbers) / total_count
    std_dev = math.sqrt(variance)

    print("1) The number of each individual number:")
    for num in set(numbers):
        print(f"{num} : {numbers.count(num)}")
    
    print("2) The Maximum value is:", maximum)
    print("3) The Minimum value is:", minimum)
    print("4) The Range of the numbers is:", range_value)
    print("5) Arithmetic Mean of the numbers is:", mean)
    print("6) Variance of the numbers is:", variance)
    print("7) The Standard Deviation of the numbers is:", std_dev)

def main():
    stat = []
    error_list = []
    condition = True

    while condition:
        user_input = get_user_input()
        numbers, errors, terminate = validate_and_process_input(user_input)
        stat.extend(numbers)
        error_list.extend(errors)
        condition = not terminate

    sorted_numbers = sort_numbers(stat)
    calculate_statistics(sorted_numbers)
    print("8) Error list:", error_list)

if __name__ == "__main__":
    main()
