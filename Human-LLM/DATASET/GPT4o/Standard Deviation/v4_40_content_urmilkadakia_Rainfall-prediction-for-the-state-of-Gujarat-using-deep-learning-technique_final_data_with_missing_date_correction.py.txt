import pandas as pd
from dateutil.parser import parse
import csv

# Read the CSV file into a DataFrame
df = pd.read_csv('final_rain_2014.csv')

# Print the first row of the DataFrame
print(df.loc[0])

# Initialize a list to store indices of missing data
missing = []

# Iterate through the DataFrame rows
for i in range(1, len(df)):
    today = parse(df.loc[i, 'DATE(GMT)'])
    prev = parse(df.loc[i - 1, 'DATE(GMT)'])
    
    # Check if the station ID is the same as the previous row
    if df.loc[i, '@STATION_ID'] == df.loc[i - 1, '@STATION_ID']:
        
        # Check if the month is the same
        if today.month == prev.month:
            
            # Check if the day is the same
            if today.day == prev.day:
                
                # Check if there is a missing hour
                if int(df.loc[i, 'TIME(GMT)']) - int(df.loc[i - 1, 'TIME(GMT)']) > 1:
                    missing.append(i)
            else:
                # Handle the case where the day changes but hours are consecutive
                if int(df.loc[i, 'TIME(GMT)']) == 0 and int(df.loc[i - 1, 'TIME(GMT)']) == 23:
                    continue
                else:
                    missing.append(i)
        else:
            # Handle month transition cases
            if prev.month in [7, 8]:
                if today.day == 1 and prev.day == 31:
                    if int(df.loc[i, 'TIME(GMT)']) == 0 and int(df.loc[i - 1, 'TIME(GMT)']) == 23:
                        continue
                    else:
                        missing.append(i)
                else:
                    missing.append(i)
            else:
                if today.day == 1 and prev.day == 30:
                    if int(df.loc[i, 'TIME(GMT)']) == 0 and int(df.loc[i - 1, 'TIME(GMT)']) == 23:
                        continue
                    else:
                        missing.append(i)
                else:
                    missing.append(i)

# Write the filtered data to a new CSV file
with open('RAIN.csv', 'w', newline='') as file_out:
    writer = csv.writer(file_out)
    writer.writerow(['@STATION_ID', 'LATITUDE', 'LONGITUDE', 'TIME(GMT)', 'DATE(GMT)', 'RAIN_FALL(mm)'])
    
    for i in range(len(df)):
        if i not in missing:
            row = [df.loc[i, '@STATION_ID'], df.loc[i, 'LATITUDE'], df.loc[i, 'LONGITUDE'], 
                   df.loc[i, 'TIME(GMT)'], df.loc[i, 'DATE(GMT)'], df.loc[i, 'RAIN_FALL(mm)']]
            writer.writerow(row)
