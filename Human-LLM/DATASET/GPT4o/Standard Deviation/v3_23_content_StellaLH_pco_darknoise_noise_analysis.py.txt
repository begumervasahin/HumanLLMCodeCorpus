import numpy as np
import h5py

# Define parameters
frames = [100, 1000, 10000]
exp_times = [0.001, 0.01, 0.1, 1]
pixel_rates = ['slow', 'fast']

def compute_statistics(data):
    """Compute mean, standard deviation, min, and max for the given data."""
    return {
        'mean': np.mean(data[0:99], axis=0),
        'std': np.std(data[0:99], axis=0),
        'min': np.min(data[0:99], axis=0),
        'max': np.max(data[0:99], axis=0)
    }

def aggregate_statistics(stats_list, chunk_count):
    """Aggregate statistics across multiple datasets."""
    grand_mean = np.mean([stats['mean'] for stats in stats_list])
    grand_std = np.sqrt(np.sum([stats['std']**2 / chunk_count for stats in stats_list]))
    grand_min = np.min([stats['min'] for stats in stats_list])
    grand_max = np.max([stats['max'] for stats in stats_list])
    
    return grand_mean, grand_std, grand_min, grand_max

def save_statistics(frame_count, exp_time, pixel_rate, grand_stats):
    """Save aggregated statistics to .npy files."""
    labels = ['mean', 'std', 'min', 'max']
    for label, stat in zip(labels, grand_stats):
        filename = f'{label}_{frame_count}frames_{exp_time}s_{pixel_rate}.npy'
        np.save(filename, stat)

# Process data for each combination of parameters
for frame_count in frames:
    for exp_time in exp_times:
        for pixel_rate in pixel_rates:
            stats_list = []
            chunk_count = frame_count // 100
            file_numbers = [f"{num:06d}" for num in range(1, chunk_count + 1)]
            
            # Read data from each file and compute statistics
            for file_num in file_numbers:
                filename = f'{frame_count}_frames_{exp_time}_{pixel_rate}_{file_num}'
                with h5py.File(filename, 'r') as file:
                    data = file['/entry/data']
                    stats = compute_statistics(data)
                    stats_list.append(stats)

            # Aggregate statistics across all files
            grand_stats = aggregate_statistics(stats_list, chunk_count)
            
            # Save the aggregated statistics
            save_statistics(frame_count, exp_time, pixel_rate, grand_stats)

print("Data processing complete.")
