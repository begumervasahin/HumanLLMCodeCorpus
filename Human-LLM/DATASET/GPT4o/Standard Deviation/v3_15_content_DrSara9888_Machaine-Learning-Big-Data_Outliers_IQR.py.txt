# Load the data from the CSV file
data = pd.read_csv("/Users/khadeejah/Desktop/weight-height.csv")

plt.figure(figsize=(20, 8))

# Function to plot distribution, box plot, and histogram for a given column
def plot_column_statistics(column, column_name):
    plt.figure(figsize=(20, 8))

    # Distribution plot
    plt.subplot(1, 3, 1)
    sns.histplot(column, kde=True)
    plt.title(f'Distribution of {column_name}')
    plt.ylabel('Distribution')

    # Box plot
    plt.subplot(1, 3, 2)
    sns.boxplot(y=column)
    plt.title(f'Box Plot of {column_name}')

    # Histogram
    plt.subplot(1, 3, 3)
    plt.hist(column, bins=20, rwidth=0.8)
    plt.title(f'Histogram of {column_name}')
    plt.xlabel(column_name)
    plt.ylabel('Count')

    plt.show()

# Plot statistics for Height
plot_column_statistics(data['Height'], 'Height')

# Calculate IQR for a given column and identify outliers
def calculate_iqr_outliers(column, column_name):
    Q1 = column.quantile(0.25)
    Q3 = column.quantile(0.75)
    IQR = Q3 - Q1
    lower_limit = Q1 - 1.5 * IQR
    upper_limit = Q3 + 1.5 * IQR

    print(f"\n{column_name} lower limit:", lower_limit)
    print(f"{column_name} upper limit:", upper_limit)

    outliers = data[(column < lower_limit) | (column > upper_limit)]
    no_outliers = data[(column > lower_limit) & (column < upper_limit)]

    print(f"\nOutliers in {column_name}:")
    print(outliers)

    print(f"\nData without outliers in {column_name} (first few rows):")
    print(no_outliers.head(3))

# Calculate and display outliers for Height
calculate_iqr_outliers(data['Height'], 'Height')

# Plot statistics for Weight
plot_column_statistics(data['Weight'], 'Weight')

# Calculate and display outliers for Weight
calculate_iqr_outliers(data['Weight'], 'Weight')
