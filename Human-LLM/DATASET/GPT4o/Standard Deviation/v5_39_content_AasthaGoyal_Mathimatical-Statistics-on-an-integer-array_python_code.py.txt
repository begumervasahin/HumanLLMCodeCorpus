import math

def get_user_input():
    """Prompt the user to enter a list of numbers separated by commas."""
    return input("Enter a list of numbers separated by commas: ")

def validate_and_process_input(input_str):
    """Validate the input string and separate valid integers from errors."""
    numbers = []
    errors = []
    items = input_str.split(",")
    
    for item in items:
        item = item.strip()
        if item.isdigit():
            numbers.append(int(item))
        elif item.lower() == "terminate":
            print("You terminated the program.")
            return numbers, errors, True
        else:
            print(f"{item} is not an integer value.")
            errors.append(item)
    
    return numbers, errors, False

def sort_numbers(numbers):
    """Return a sorted list of numbers in ascending order."""
    return sorted(numbers)

def calculate_statistics(numbers):
    """Calculate and print various statistics for a list of numbers."""
    if not numbers:
        print("No valid numbers to calculate statistics.")
        return

    maximum = max(numbers)
    minimum = min(numbers)
    range_value = maximum - minimum
    total_count = len(numbers)
    mean = sum(numbers) / total_count
    variance = sum((x - mean) ** 2 for x in numbers) / total_count
    std_dev = math.sqrt(variance)

    print("1) Frequency of each number:")
    for num in set(numbers):
        print(f"{num}: {numbers.count(num)}")
    
    print("2) Maximum value:", maximum)
    print("3) Minimum value:", minimum)
    print("4) Range of the numbers:", range_value)
    print("5) Arithmetic mean:", mean)
    print("6) Variance:", variance)
    print("7) Standard deviation:", std_dev)

def main():
    numbers = []
    errors = []
    condition = True

    while condition:
        user_input = get_user_input()
        new_numbers, new_errors, terminate = validate_and_process_input(user_input)
        numbers.extend(new_numbers)
        errors.extend(new_errors)
        condition = not terminate

    sorted_numbers = sort_numbers(numbers)
    calculate_statistics(sorted_numbers)
    print("8) Error list:", errors)

if __name__ == "__main__":
    main()
