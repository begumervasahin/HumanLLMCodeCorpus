import numpy as np
import math

class DesvEstandar:
    """Class to calculate the standard deviation of filtered data."""

    def __init__(self, name="e_desv_estandar_cf", in_sig=[np.float32], out_sig=[np.float32]):
        self.contador = 0
        self.acum = 0

    def work(self, input_items, output_items):
        in0 = input_items[0]
        in1 = input_items[1]
        out0 = output_items[0]
        
        # Filter the input data based on the given conditions
        a1 = in0[(0.5 < in0) & (in0 < 1.5) & (in1 < 0.5) & (in1 > -0.5)]
        b1 = in1[(0.5 < in0) & (in0 < 1.5) & (in1 < 0.5) & (in1 > -0.5)]
        
        # Calculate mean and variance for a1
        if a1.shape[0] > 0:
            a1bar = np.mean(a1)
            a1var = np.var(a1, ddof=1)
        else:
            a1var = 0
        
        # Calculate mean and variance for b1
        if b1.shape[0] > 0:
            b1bar = np.mean(b1)
            b1var = np.var(b1, ddof=1)
        else:
            b1var = 0
        
        # Calculate the combined standard deviation
        datos_desv_estandar = np.sqrt(a1var + b1var)
        out0 = datos_desv_estandar
        return out0

# Instantiate the class
obj1 = DesvEstandar()

# Set parameters
m = 2
r = 1
n = 20
fvar = 0.1
pi = math.pi

# Generate random data
datos = np.round((m - 1) * np.random.rand(r, n))
angles = datos * 2 * pi / m
mag = fvar * np.random.normal(0, 1, n) + 1

# Calculate a and b based on angles and magnitude
a = mag * np.cos(angles + fvar * np.random.normal(0, 1, n))
b = mag * np.sin(angles + fvar * np.random.normal(0, 1, n))

# Flatten the arrays to match the expected input format
a_flat = a.flatten()
b_flat = b.flatten()

# Call the work method and print the result
ro = obj1.work([a_flat, b_flat], [0.0])
print(ro)
