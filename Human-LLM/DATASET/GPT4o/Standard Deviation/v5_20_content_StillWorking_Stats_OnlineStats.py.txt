class OnlineStats:
    def __init__(self):
        """
        Initializes the OnlineStats object with mean, variance, and count.
        """
        self.mean = 0
        self.variance = 0
        self.count = 0

    def update_statistics(self, number):
        """
        Updates mean and variance with the new number.
        Parameters:
            number (float): The number to include in the statistics.
        Returns:
            tuple: The updated mean and variance.
        """
        if self.count == 0:
            self.mean = number
            self.variance = 0
        else:
            previous_mean = self.mean
            self.count += 1
            self.mean = previous_mean + (number - previous_mean) / self.count
            self.variance = (
                (self.count - 2) / (self.count - 1) * self.variance +
                (number - previous_mean) ** 2 / self.count
            )
        return self.mean, self.variance

    def process_input(self, input_func, output_func):
        """
        Processes user input until a negative number is entered.
        Parameters:
            input_func (function): Function to retrieve user input.
            output_func (function): Function to display results.
        """
        while True:
            try:
                number = input_func()
                if number < 0:
                    print("Negative number entered. Ending calculation.")
                    break
                mean, variance = self.update_statistics(number)
                output_func(mean, variance)
            except ValueError:
                print("Invalid input. Please enter a numeric value.")

def get_user_input():
    """
    Prompts the user to enter a number.
    Returns:
        float: The number entered by the user.
    """
    while True:
        try:
            return float(input("Enter a number: "))
        except ValueError:
            print("Invalid input. Please enter a numeric value.")

def display_statistics(mean, variance):
    """
    Displays the calculated mean and variance.
    Parameters:
        mean (float): The mean value.
        variance (float): The variance value.
    """
    print(f"Mean: {mean:.2f}, Variance: {variance:.2f}")
