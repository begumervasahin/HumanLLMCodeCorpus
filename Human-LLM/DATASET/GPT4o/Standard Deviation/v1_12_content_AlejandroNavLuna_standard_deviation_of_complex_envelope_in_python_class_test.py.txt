import numpy as np
import math

class DesvEstandar:
    """ Description of the block """
    def __init__(self, name="e_desv_estandar_cf", in_sig=[np.float32], out_sig=[np.float32]):
        self.contador = 0
        self.acum = 0

    def work(self, input_items, output_items):
        in0 = input_items[0]
        in1 = input_items[1]
        out0 = output_items[0]
        
        a1 = in0[(0.5 < in0) & (in0 < 1.5) & (in1 < 0.5) & (in1 > -0.5)]
        b1 = in1[(0.5 < in0) & (in0 < 1.5) & (in1 < 0.5) & (in1 > -0.5)]
        
        if a1.shape[0] > 0:
            a1bar = np.sum(a1) / a1.shape[0]
            a1var = np.sum((a1 - a1bar) ** 2) / (a1.shape[0] - 1)
        else:
            a1var = 0
            
        if b1.shape[0] > 0:
            b1bar = np.sum(b1) / b1.shape[0]
            b1var = np.sum((b1 - b1bar) ** 2) / (b1.shape[0] - 1)
        else:
            b1var = 0
        
        datos_desv_estandar = np.sqrt(a1var + b1var)
        out0 = datos_desv_estandar
        return out0

obj1 = DesvEstandar()
m = 2
r = 1
n = 20
fvar = 0.1
pi = math.pi

datos = np.round((m - 1) * np.random.rand(r, n))
angles = datos * 2 * pi / m
mag = fvar * np.random.normal(0, 1, n) + 1

a = mag * np.cos(angles + fvar * np.random.normal(0, 1, n))
b = mag * np.sin(angles + fvar * np.random.normal(0, 1, n))

ro = obj1.work([a.flatten(), b.flatten()], [0.])
