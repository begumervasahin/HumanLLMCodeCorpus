import requests
from bs4 import BeautifulSoup
import re
import statistics
import matplotlib.pyplot as plt

def get_player_links(url):
    """Fetch player profile links from the given URL."""
    response = requests.get(url)
    soup = BeautifulSoup(response.content, 'html.parser')
    table = soup.find('table', {'class': 'tablehead'})
    links = []

    if table:
        rows = table.findAll('tr')
        for tr in rows:
            cols = tr.findAll('td')
            if cols:
                atags = cols[0].findAll('a')
                for atag in atags:
                    links.append(atag.get('href'))
    return links

def get_player_height(player_url):
    """Extract player height in inches from the player profile URL."""
    response = requests.get(player_url)
    soup = BeautifulSoup(response.content, 'html.parser')
    general_info = soup.find('ul', {'class': 'general-info'})
    height_inches = None

    if general_info:
        info_pieces = general_info.findAll('li')
        if len(info_pieces) > 1:
            player_height = str(info_pieces[1])
            height_feet = re.search(r"<li>(\d)\W\s", player_height).group(1)
            height_inches = re.search(r"\s(\d+?)\W\W\s", player_height).group(1)
            total_height_inches = int(height_feet) * 12 + int(height_inches)
            return total_height_inches
    return None

def main():
    url = "http://www.espn.com/nfl/players/_/position/s"
    heights_list = []

    # Get player links
    links = get_player_links(url)

    # Get heights for each player
    for link in links:
        height = get_player_height(link)
        if height:
            heights_list.append(height)
            print(height)

    # Calculate and print statistics
    if heights_list:
        print("Standard Deviation:", statistics.pstdev(heights_list))
        print("Mean Height:", statistics.mean(heights_list))

        # Plot the histogram
        bins_intervals = list(range(65, 80))
        plt.hist(heights_list, bins=bins_intervals)
        plt.xlabel('Height (inches)')
        plt.ylabel('Frequency')
        plt.title('Distribution of NFL Player Heights')
        plt.show()

if __name__ == "__main__":
    main()
