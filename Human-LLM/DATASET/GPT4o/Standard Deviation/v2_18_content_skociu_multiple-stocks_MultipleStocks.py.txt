import datetime
from pandas_datareader import data
import pandas as pd
import numpy as np
import os

def test_run():
    # Define the list of tickers and the time period for data retrieval
    tickers = ['AMZN', 'SPY', 'M']
    start_date = datetime.datetime(2007, 5, 13)
    end_date = datetime.datetime(2017, 5, 13)
    
    # Initialize an empty list to store the data for each ticker
    listall = []
    
    # Retrieve the adjusted close price for each ticker and store in the list
    for ticker in tickers:
        mydata = data.get_data_yahoo(ticker, start=start_date, end=end_date, interval='m')['Adj Close']
        symbol = pd.Series(mydata, name=ticker)
        listall.append(symbol)
    
    # Create a DataFrame from the list of Series
    df = pd.DataFrame(data=listall).T
    print("Price data:\n", df, '\n')
    
    # Calculate the percentage change (returns) of the stock prices
    pctstocks = df.pct_change()
    print("Stock returns:\n", pctstocks, '\n')
    
    # Calculate the correlation matrix of the stock returns
    cormat = pctstocks.corr()
    print("Correlation matrix:\n\n", cormat, "\n")
    
    # Calculate the variance-covariance matrix of the stock returns
    covmat = pctstocks.cov()
    print("Variance-covariance matrix:\n\n", covmat, "\n")
    
    # Calculate the annualized mean returns
    means = (1 + pctstocks.mean())**12.0 - 1
    print("Annual means:\n\n", means, "\n")
    
    # Calculate the annualized standard deviation of returns
    stdev = pctstocks.std() * np.sqrt(12)
    print("Annual standard deviation:\n\n", stdev, "\n")
    
    # Assign equal weights to each stock in the portfolio
    weights = [1 / float(len(tickers))] * len(tickers)
    weights = pd.Series(data=weights, index=tickers)
    print("Weights:\n\n", weights, "\n")
    
    # Convert weights, mean returns, and covariance matrix to numpy arrays
    w1 = np.array(weights)
    m1 = np.array(means)
    cov1 = np.array(covmat)
    
    # Calculate the portfolio's expected annual return
    portret = np.dot(w1.T, m1) * 12.0
    print("Portfolio return:\n\n", portret, "\n")
    
    # Calculate the portfolio's annual standard deviation
    first = np.dot(w1.T, cov1)
    portstd = np.sqrt(np.dot(first, w1)) * np.sqrt(12)
    print("Portfolio standard deviation:\n\n", portstd, "\n")
    
    # Create the directory 'Excel' if it does not exist
    if not os.path.isdir('./Excel'):
        os.mkdir('./Excel')
    
    # Get the current directory path
    cur_dir = os.path.dirname(os.path.abspath(__file__))
    
    # Save the data to CSV files
    df.to_csv(os.path.join(cur_dir, 'Excel/mystocks_prices.csv'))
    pctstocks.to_csv(os.path.join(cur_dir, 'Excel/mystocks_pctchange.csv'))
    cormat.to_csv(os.path.join(cur_dir, 'Excel/mystocks_correlations.csv'))
    covmat.to_csv(os.path.join(cur_dir, 'Excel/mystocks_varcov.csv'))

if __name__ == "__main__":
    test_run()
