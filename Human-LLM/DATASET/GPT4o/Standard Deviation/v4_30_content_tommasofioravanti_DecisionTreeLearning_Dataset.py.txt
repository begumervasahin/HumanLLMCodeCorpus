class DataSet:
    """
    A data set for a machine learning problem.

    Attributes:
    - examples (list): A list of examples. Each example is a dictionary of attribute values.
    - attrs (list): A list of integers to index into an example, so example[attr] gives a value.
                    Normally the same as range(len(d.examples[0])).
    - attrnames (dict): A dictionary that identifies {attr: attrname, ...} for all d.attrs.
    - target (int): The attribute that a learning algorithm will try to predict. By default, the final attribute.
    - inputs (list): The list of attrs without the target.
    - values (dict): A dictionary that identifies every possible value for all d.attrs {attr: [value1, value2, ...]}.
    """

    def __init__(self, name='', examples=None, inputs=None, attributes=None, target=None, attrnames=None, values=None):
        self.name = name
        self.examples = examples
        self.target = target
        self.values = values
        self.inputs = inputs
        
        if attributes is None and examples is not None:
            attributes = list(range(len(examples[0])))
        self.attributes = attributes

        if isinstance(attrnames, str):
            self.attrnames = attrnames.split()
        else:
            self.attrnames = attrnames or attributes
