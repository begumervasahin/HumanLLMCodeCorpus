import csv
import pandas as pd
import statistics

def get_ticker_list(file_path):
    """Reads a CSV file to extract a list of ticker symbols."""
    ticker_list = []
    with open(file_path, 'r') as f:
        reader = csv.reader(f)
        for line in reader:
            if "Symbol" not in line:
                ticker_list.append(line[2])
    return ticker_list

def process_stock_data(ticker, input_dir, output_dir, ndays=9):
    """Processes stock data to calculate SMA and EMA, and returns the standard deviation of the SMA."""
    # Handle '&' character in ticker symbols
    ticker = ticker.replace('&', '%26')
    
    file_path = f"{input_dir}/{ticker}.csv"
    df = pd.read_csv(file_path)
    
    if len(df.columns) >= 6 and len(df.index) >= ndays:
        df['SMA'] = df['Close'].rolling(window=ndays).mean().round(2)
        df['EMA'] = df['Close'].ewm(span=15, min_periods=ndays).mean().round(2)
        
        sma_list = df['SMA'].tolist()
        stdev = statistics.stdev(sma_list[ndays-1:]).round(2)
        
        output_file_path = f"{output_dir}/{ticker}.csv"
        df.to_csv(output_file_path, index=False)
    else:
        stdev = 0

    return stdev

def main():
    """Main function to orchestrate the processing of stock data and save the standard deviations."""
    input_dir = r"C:/Users/Hp/Desktop/New folder/Top500_stock/Original data"
    output_dir = r"C:/Users/Hp/Desktop/New folder/Top500_stock/With SMA and EMA"
    ticker_file = r"C:/Users/Hp/Desktop/New folder/nifty500_list.csv"
    
    ticker_list = get_ticker_list(ticker_file)
    stdev_list = []

    for ticker in ticker_list:
        stdev = process_stock_data(ticker, input_dir, output_dir)
        stdev_list.append(stdev)
    
    stdev_df = pd.DataFrame(stdev_list, index=ticker_list, columns=["Standard Deviation"])
    stdev_df.to_csv(r"C:/Users/Hp/Desktop/New folder/st_dev.csv")

if __name__ == "__main__":
    main()
