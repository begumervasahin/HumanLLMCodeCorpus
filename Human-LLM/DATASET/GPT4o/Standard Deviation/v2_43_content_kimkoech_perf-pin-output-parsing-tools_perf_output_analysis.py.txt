import re
import numpy as np

def parse_perf_output(perf_output_file, search_terms, debug_mode=False):
    """
    Parses the performance output file to extract runtime values for the specified search terms.
    
    Args:
        perf_output_file (str): Path to the performance output file.
        search_terms (list of str): List of search terms to look for in the file.
        debug_mode (bool): If True, prints debug information.
    
    Returns:
        dict: A dictionary where keys are search terms and values are lists of extracted numeric values.
    """
    runtimes = {term: [] for term in search_terms}
    
    with open(perf_output_file, "r") as file:
        for line in file:
            for term in search_terms:
                if term in line:
                    numeric_value = int(re.findall(r"\d+", line)[0])
                    if debug_mode:
                        print(f"Found {numeric_value} for term {term}")
                    runtimes[term].append(numeric_value)
    
    print("Parsing complete!")
    return runtimes

def calculate_statistics(runtimes):
    """
    Calculates the standard deviation and mean for the runtime values of each search term.
    
    Args:
        runtimes (dict): Dictionary with search terms as keys and lists of numeric values as values.
    
    Returns:
        str: A formatted string with the calculated standard deviation and mean for each search term.
    """
    output_str = ""
    for term, values in runtimes.items():
        std_dev = np.std(values)
        mean_val = np.mean(values)
        output_str += f"{term} :\t{std_dev:.2f},\t{mean_val:.2f}\n"
    return output_str

def write_report(output_str, report_file="perf_report.txt"):
    """
    Writes the calculated statistics to a report file.
    
    Args:
        output_str (str): Formatted string with the statistics.
        report_file (str): Path to the report file.
    """
    header = "Type of instruction: \t std, \t mean\n"
    with open(report_file, "w") as report:
        report.write(header + output_str)

def main():
    """
    Main function to execute the performance parsing and reporting.
    """
    perf_output_file = "measure.out"
    search_terms = [
        "mem_inst_retired.all_loads:u",
        "instructions:u",
        "mem_inst_retired.all_stores:u",
        "br_inst_retired.all_branches:u"
    ]
    debug_mode = False
    
    runtimes = parse_perf_output(perf_output_file, search_terms, debug_mode)
    output_str = calculate_statistics(runtimes)
    
    print("Type of instruction: \t std, \t mean")
    print(output_str)
    
    write_report(output_str)

if __name__ == '__main__':
    main()
