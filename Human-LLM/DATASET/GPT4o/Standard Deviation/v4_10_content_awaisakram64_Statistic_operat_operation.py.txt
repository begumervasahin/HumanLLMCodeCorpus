class Functions:
    def Median(self, data):
        data = sorted(data)
        ln = len(data)
        if ln % 2 == 0:
            return (data[ln // 2 - 1] + data[ln // 2]) / 2
        else:
            return data[ln // 2]

    def Mode(self, data):
        frequency = {}
        for item in data:
            frequency[item] = frequency.get(item, 0) + 1
        max_freq = max(frequency.values())
        if max_freq == 1:
            return [0]  # No mode
        return [key for key, val in frequency.items() if val == max_freq]

    def Mean(self, data):
        return sum(data) / len(data)

    def Variance(self, data):
        mean = self.Mean(data)
        return sum((i - mean) ** 2 for i in data) / len(data)

    def StaDev(self, data):
        return self.Variance(data) ** 0.5

def main():
    func = Functions()
    with open("data.txt", 'r') as file:
        data = [float(i) for i in file]
    
    print('Median is %.2f' % func.Median(data))
    print('Mean is %.2f' % func.Mean(data))
    print('Mode is ' + ', '.join(map(str, func.Mode(data))))
    print('Variance is %.2f' % func.Variance(data))
    print('Standard Deviation is %.2f' % func.StaDev(data))

if __name__ == "__main__":
    main()
