class Statistics:
    @staticmethod
    def median(data):
        data = sorted(data)
        n = len(data)
        mid = n // 2
        if n % 2 == 0:
            return (data[mid - 1] + data[mid]) / 2
        else:
            return data[mid]

    @staticmethod
    def mode(data):
        frequency = {}
        for item in data:
            frequency[item] = frequency.get(item, 0) + 1
        max_freq = max(frequency.values())
        if max_freq == 1:
            return [0]  # No mode
        return [key for key, val in frequency.items() if val == max_freq]

    @staticmethod
    def mean(data):
        return sum(data) / len(data)

    @staticmethod
    def variance(data):
        mean = Statistics.mean(data)
        return sum((x - mean) ** 2 for x in data) / len(data)

    @staticmethod
    def standard_deviation(data):
        return Statistics.variance(data) ** 0.5

def main():
    with open("data.txt", 'r') as file:
        data = [float(line.strip()) for line in file]

    stats = Statistics()
    print('Median is %.2f' % stats.median(data))
    print('Mean is %.2f' % stats.mean(data))
    print('Mode is ' + ', '.join(map(str, stats.mode(data))))
    print('Variance is %.2f' % stats.variance(data))
    print('Standard Deviation is %.2f' % stats.standard_deviation(data))

if __name__ == "__main__":
    main()
