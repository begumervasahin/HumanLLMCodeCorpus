import math

def get_user_input():
    stat = []
    error_list = []
    while True:
        phrase = input("Enter a list of numbers (or type 'Terminate' to end): ")
        temp_list = phrase.split(",")
        stat.extend(temp_list)
        
        for item in temp_list:
            if not item.strip().isdigit():
                error_list.append(item)
                if item.strip().lower() == "terminate":
                    stat.remove(item)
                    print("You terminated the program")
                    return stat, error_list
                else:
                    stat.remove(item)
                    print(f"{item} is not an integer value")
    
def sort_numbers(numbers):
    return sorted(numbers)

def calculate_statistics(numbers):
    total_count = len(numbers)
    sum1 = sum(numbers)
    mean = sum1 / total_count
    variance = sum((x - mean) ** 2 for x in numbers) / total_count
    std = math.sqrt(variance)
    return {
        "maximum": numbers[-1],
        "minimum": numbers[0],
        "range": numbers[-1] - numbers[0],
        "mean": mean,
        "variance": variance,
        "std": std
    }

def count_occurrences(numbers):
    occurrences = {num: numbers.count(num) for num in set(numbers)}
    return occurrences

def display_results(sort_list, stats, occurrences, errors):
    print("Sorted List: ", sort_list)
    print("1) The number of each individual number:")
    for num, count in occurrences.items():
        print(f"{num}: {count}")
    print("2) The Maximum value is:", stats["maximum"])
    print("3) The Minimum value is:", stats["minimum"])
    print("4) The Range of the numbers is:", stats["range"])
    print("5) Arithmetic Mean of the numbers is:", stats["mean"])
    print("6) Variance of the numbers is:", stats["variance"])
    print("7) The Standard Deviation of the numbers is:", stats["std"])
    print("8) Errors:", errors)

def main():
    stat, error_list = get_user_input()
    stat = [int(x) for x in stat if x.isdigit()]
    
    if not stat:
        print("No valid numbers were entered.")
        return
    
    sort_list = sort_numbers(stat)
    stats = calculate_statistics(sort_list)
    occurrences = count_occurrences(sort_list)
    
    display_results(sort_list, stats, occurrences, error_list)

if __name__ == "__main__":
    main()
