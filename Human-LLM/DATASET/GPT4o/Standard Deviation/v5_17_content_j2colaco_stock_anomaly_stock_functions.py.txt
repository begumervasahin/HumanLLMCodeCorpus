import pandas as pd
from pandas_datareader import data
import datetime as dt

def extract_stock_data(stock_symbols, start_date, end_date=dt.datetime.today().date()):
    """
    Extract historical stock data for given symbols from Yahoo Finance.

    Parameters:
    stock_symbols (list): List of stock ticker symbols.
    start_date (datetime): Start date for data extraction.
    end_date (datetime): End date for data extraction (default is today's date).

    Returns:
    DataFrame: A DataFrame containing historical stock data for all provided symbols.
    """
    all_stock_data = pd.DataFrame()
    
    for stock in stock_symbols:
        print(f'Extracting data for: {stock}')
        try:
            stock_data = data.DataReader(stock, 'yahoo', start_date, end_date)
            stock_data['ticker'] = stock
            all_stock_data = pd.concat([all_stock_data, stock_data])
        except Exception as e:
            print(f"Failed to extract data for {stock}. Error: {e}")
    
    return all_stock_data

def calculate_moving_average(df, column_name, window_size):
    """
    Calculate moving average for a specified column in the DataFrame.

    Parameters:
    df (DataFrame): The input DataFrame.
    column_name (str): The column name for which to calculate the moving average.
    window_size (int): The window size for the moving average.

    Returns:
    DataFrame: The input DataFrame with an additional column for the moving average.
    """
    moving_avg_column = f"{column_name}_ma{window_size}"
    df[moving_avg_column] = df[column_name].rolling(window=window_size).mean()
    
    return df

def apply_moving_average_to_stocks(df, column_name, ticker_column, window_size):
    """
    Apply moving averages for each unique stock in the DataFrame.

    Parameters:
    df (DataFrame): The input DataFrame containing stock data.
    column_name (str): The column name for which to calculate the moving average.
    ticker_column (str): The column to split the DataFrame by (e.g., 'ticker').
    window_size (int): The window size for the moving average.

    Returns:
    DataFrame: The input DataFrame with additional columns for the moving averages.
    """
    all_stock_data = pd.DataFrame()
    
    for ticker in df[ticker_column].unique():
        stock_data = df[df[ticker_column] == ticker].sort_values(by='Date')
        stock_data = calculate_moving_average(stock_data, column_name, window_size)
        all_stock_data = pd.concat([all_stock_data, stock_data])
    
    return all_stock_data

# Example usage
if __name__ == "__main__":
    stock_symbols = ['AAPL', 'MSFT', 'GOOGL']
    start_date = dt.datetime(2020, 1, 1)
    end_date = dt.datetime(2023, 1, 1)

    # Extract stock data
    stock_data = extract_stock_data(stock_symbols, start_date, end_date)
    
    # Calculate moving averages
    window_size = 50
    stock_data_with_ma = apply_moving_average_to_stocks(stock_data, 'Close', 'ticker', window_size)
    
    print(stock_data_with_ma.head())
