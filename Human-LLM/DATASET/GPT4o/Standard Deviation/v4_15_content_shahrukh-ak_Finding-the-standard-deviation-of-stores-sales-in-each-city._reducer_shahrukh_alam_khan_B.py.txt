import sys
import math

def update(existingAggregate, newValue):
    """Update the running statistics for a new value."""
    count, mean, M2 = existingAggregate
    count += 1
    delta = newValue - mean
    mean += delta / count
    delta2 = newValue - mean
    M2 += delta * delta2
    return count, mean, M2

def finalize(existingAggregate):
    """Finalize the calculation of mean, variance, and sample variance."""
    count, mean, M2 = existingAggregate
    if count < 2:
        return float('nan'), float('nan'), float('nan')
    variance = M2 / count
    sampleVariance = M2 / (count - 1)
    return mean, variance, sampleVariance

# Initialize running statistics
S = [0, 0, 0]
last_key = None

# Process input lines
for line in sys.stdin:
    key, new_value = line.strip().split('\t')
    new_value = float(new_value)
    
    if last_key and key != last_key:
        # Finalize and print statistics for the last key
        average, P_Variance, S_Variance = finalize(S)
        S_Deviation = math.sqrt(P_Variance)
        print(f'{last_key}\t{S_Deviation:.2f}')
        
        # Reset for the new key
        S = [0, 0, 0]
        last_key = key
    
    if last_key is None:
        last_key = key
    
    # Update statistics for the current key
    S = update(S, new_value)

# Finalize and print statistics for the last key in the input
if last_key:
    average, P_Variance, S_Variance = finalize(S)
    S_Deviation = math.sqrt(P_Variance)
    print(f'{last_key}\t{S_Deviation:.2f}')
