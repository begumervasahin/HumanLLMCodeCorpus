import sys
import math

def update(existing_aggregate, new_value):
    count, mean, M2 = existing_aggregate
    count += 1
    delta = new_value - mean
    mean += delta / count
    delta2 = new_value - mean
    M2 += delta * delta2
    return count, mean, M2

def finalize(existing_aggregate):
    count, mean, M2 = existing_aggregate
    if count < 2:
        return mean, float('nan'), float('nan')
    else:
        variance = M2 / count
        sample_variance = M2 / (count - 1)
        return mean, variance, sample_variance

# Initialize the aggregate tuple
S = [0, 0, 0]
last_key = None

# Process each line from standard input
for line in sys.stdin:
    key, new_value = line.strip().split('\t')
    new_value = float(new_value)
    
    if last_key and key != last_key:
        # Finalize and print the results for the previous key
        average, P_variance, S_variance = finalize(S)
        S_deviation = math.sqrt(P_variance)
        print(f'{last_key}\t{S_deviation:.2f}')
        
        # Reset the aggregate for the new key
        last_key = key
        S = [0, 0, 0]
        S = update(S, new_value)
    else:
        last_key = key
        S = update(S, new_value)

# Finalize and print the results for the last key
if last_key:
    average, P_variance, S_variance = finalize(S)
    S_deviation = math.sqrt(P_variance)
    print(f'{last_key}\t{S_deviation:.2f}')
