import argparse
import sys

def classify_landcover(landcover, stddev):
    # Simulated classifications and standard deviations
    landcover_classification = {
        'water': 0,
        'woody savannas': 130
    }

    # Standard deviation values
    landcover_stddev = {
        'woody savannas': 58.40
    }

    if landcover not in landcover_classification:
        print(f"Unknown landcover type: {landcover}")
        sys.exit(1)

    classification = landcover_classification[landcover]
    print(f"Landcover: {landcover}, Classification: {classification}")

    if stddev and landcover in landcover_stddev:
        print(f"Standard Deviation: {landcover_stddev[landcover]}")

if __name__ == '__main__':
    parser = argparse.ArgumentParser(description="Classify landcover types")
    parser.add_argument('--landcover', type=str, required=True, help="Type of landcover to classify")
    parser.add_argument('--stddev', action='store_true', help="Include standard deviation in output")
    args = parser.parse_args()

    if not args.landcover:
        print("Error: --landcover requires a value")
        sys.exit(1)

    classify_landcover(args.landcover, args.stddev)
