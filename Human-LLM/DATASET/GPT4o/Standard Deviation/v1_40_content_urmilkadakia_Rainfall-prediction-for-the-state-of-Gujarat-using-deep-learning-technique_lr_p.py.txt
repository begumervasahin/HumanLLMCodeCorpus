import csv
import os
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
import numpy as np
from sklearn.preprocessing import StandardScaler
from joblib import Parallel, delayed
import multiprocessing
import statistics

def machine(k, temp1, temp2):
    # Splitting the data into training and testing sets
    X_train, X_test, y_train, y_test = train_test_split(temp1, temp2, test_size=0.2, random_state=k)
    
    # Converting data to the appropriate types
    X_train = [[float(j) for j in i] for i in X_train]
    X_test = [[float(j) for j in i] for i in X_test]
    y_train = [int(i) for i in y_train]
    y_test = [int(i) for i in y_test]
    
    # Flattening the target arrays
    y_train = np.ravel(y_train)
    y_test = np.ravel(y_test)
    
    # Initializing and training the Linear Regression model
    clf = LinearRegression()
    print("Mean of y_test:", statistics.mean(y_test))
    clf.fit(X_train, y_train)
    
    # Making predictions and displaying results
    predictions = clf.predict(X_test)
    print("Predictions:", predictions)
    print("Coefficients:", clf.coef_)
    print("Intercept:", clf.intercept_)
    
    # Calculating accuracy of the model
    accuracy = clf.score(X_test, y_test)
    print("Accuracy:", accuracy)

if __name__ == '__main__':
    # Path to the CSV file
    dataPath = "/home/geospacial2/PycharmProjects/BTP/rain2014_1.csv"
    
    # Reading the CSV file
    with open(dataPath, 'rt') as csvfile:
        reader = csv.reader(csvfile)
        temp1 = []
        temp2 = []
        
        # Skipping the header
        next(reader, None)
        
        # Reading data
        for row in reader:
            temp1.append(row[9:17])
            temp2.append(row[22])
    
    # Scaling the features
    temp1 = StandardScaler().fit_transform(temp1)
    
    # Setting the number of iterations
    niter = 1
    inputs = range(niter)
    
    # Getting the number of CPU cores
    num_cores = multiprocessing.cpu_count()
    
    # Running the machine function in parallel
    Parallel(n_jobs=num_cores)(delayed(machine)(l, temp1, temp2) for l in inputs)
