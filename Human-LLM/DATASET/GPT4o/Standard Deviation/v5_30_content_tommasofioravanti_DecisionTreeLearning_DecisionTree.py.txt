class DecisionTree:
    def __init__(self, attribute, threshold, attrname=None, default_child=None, branches=None):
        """
        Initialize the DecisionTree with a specified attribute and threshold.
        
        Parameters:
        - attribute: The attribute this node tests.
        - threshold: The threshold value for the attribute.
        - attrname: Optional; a name for the attribute.
        - default_child: Optional; a default child node to handle cases where the attribute is not found.
        - branches: Optional; a dictionary of branches.
        """
        self.attribute = attribute
        self.threshold = threshold
        self.attrname = attrname or attribute
        self.default_child = default_child
        self.branches = branches or {}
    
    def __call__(self, example):
        """
        Classify the given example using the attribute and branches.
        
        Parameters:
        - example: A dictionary representing an example to classify.
        
        Returns:
        The classification result.
        """
        attr_value = example.get(self.attribute, None)
        if attr_value is None:
            return self.default_child(example) if self.default_child else None
        
        if float(attr_value) > self.threshold:
            return self.branches.get((self.threshold, True), self.default_child)(example)
        else:
            return self.branches.get((self.threshold, False), self.default_child)(example)
    
    def add_branch(self, threshold, is_greater, subtree):
        """
        Add a branch to the decision tree.
        
        Parameters:
        - threshold: The threshold value for the branch.
        - is_greater: A boolean indicating if the branch is for values greater than the threshold.
        - subtree: The subtree to add.
        """
        self.branches[(threshold, is_greater)] = subtree
    
    def display(self, indent=0):
        """
        Print the decision tree structure.
        
        Parameters:
        - indent: The indentation level for printing the tree.
        """
        indent_str = ' ' * 4 * indent
        print(f"{indent_str}Test {self.attrname}")
        for (val, is_greater), subtree in self.branches.items():
            comparison = '>' if is_greater else '<='
            print(f"{indent_str}    {self.attrname} {comparison} {val} ==> ", end='')
            subtree.display(indent + 1)

# Example usage of DecisionTree class (for testing and illustration purposes)
if __name__ == "__main__":
    # Create a simple tree structure
    leaf1 = DecisionTree('leaf', 0)  # Placeholder leaf nodes
    leaf2 = DecisionTree('leaf', 0)
    root = DecisionTree('attribute', 5)
    root.add_branch(5, True, leaf1)
    root.add_branch(5, False, leaf2)

    # Example to classify
    example = {'attribute': 6}

    # Display the tree
    root.display()

    # Classify the example
    print("Classification:", root(example))
