import csv

def read_input_file(file_path):
    """Read the input CSV file and return the data as a list of rows."""
    data = []
    with open(file_path, 'r') as input_file:
        reader = csv.reader(input_file, delimiter=',')
        next(reader)  # Skip the header row
        data = [row for row in reader]
    return data

def write_output_file(file_path, data, header):
    """Write the filtered data to the output CSV file with the given header."""
    with open(file_path, 'w+', newline='') as file_out:
        writer = csv.writer(file_out)
        writer.writerow(header)
        writer.writerows(data)

def filter_and_format_data(data):
    """Filter and format the data according to specified conditions."""
    filtered_data = []
    for row in data:
        row[13] = round(float(row[13]), 3)
        if float(row[12]) != -9999.0 and row[13] != -49.995:
            filtered_row = [
                row[0], row[1], row[2], row[7], row[8], row[3], row[4], row[5],
                row[6], row[10], row[11], row[12], row[13], row[14], row[15],
                row[16], row[17], row[18], row[19], row[20], row[21], row[22],
                row[23]
            ]
            filtered_data.append(filtered_row)
    return filtered_data

def main():
    input_file_path = 'final_rain2016_sorted.csv'
    output_file_path = 'rain2016.csv'
    
    header = [
        "Station_ID", "Latitude", "Longitude", "Time", "Date", "Top Latitude", "Bottom Latitude",
        "Left Longitude", "Right Longitude", "TIR1_Temperature(K)", "WV_Radiance", "Elevation(m)",
        "Vegetation", "Mean3(K)", "Std3(K)", "Mean5(K)", "Std5(K)", "Air_Temp(°C)", "Wind_Speed(m/s)",
        "Wind_Direction(deg)", "Atmo_Pressure(hpa)", "Humidity(%)", "Rain_Fall(mm)"
    ]
    
    # Read data from the input file
    data = read_input_file(input_file_path)
    
    # Filter and format the data
    filtered_data = filter_and_format_data(data)
    
    # Write the filtered data to the output file
    write_output_file(output_file_path, filtered_data, header)

if __name__ == '__main__':
    main()
