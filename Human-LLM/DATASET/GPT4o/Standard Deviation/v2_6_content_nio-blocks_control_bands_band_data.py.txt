LIMIT_FACTOR = 1.128

class BandData:
    """A class to hold deviation summary data for a list of values."""
    
    def __init__(self, previous_value=None):
        self.count_items = 0
        self.sum_items = 0
        self.count_ranges = 0
        self.sum_ranges = 0
        self.last_val = previous_value
    
    def register_value(self, value):
        """Register a new value and update the summary data."""
        self.count_items += 1
        self.sum_items += value
        
        if self.last_val is not None:
            self.count_ranges += 1
            self.sum_ranges += abs(self.last_val - value)
        
        self.last_val = value
    
    def get_mean(self):
        """Return the mean of the registered items."""
        if self.count_items == 0:
            return 0
        return self.sum_items / self.count_items
    
    def get_range(self):
        """Return the average range size adjusted by the LIMIT_FACTOR."""
        if self.count_ranges == 0:
            return 0
        return self.sum_ranges / self.count_ranges / LIMIT_FACTOR
    
    def __add__(self, other):
        """Combine two BandData instances by summing their values."""
        result = BandData(self.last_val if self.count_items > 0 else other.last_val)
        result.count_items = self.count_items + other.count_items
        result.sum_items = self.sum_items + other.sum_items
        result.count_ranges = self.count_ranges + other.count_ranges
        result.sum_ranges = self.sum_ranges + other.sum_ranges
        return result
    
    def __str__(self):
        """Return a string representation of the BandData instance."""
        return str({
            "items": {
                "sum": self.sum_items,
                "count": self.count_items
            },
            "ranges": {
                "sum": self.sum_ranges,
                "count": self.count_ranges
            },
            "mean": self.get_mean(),
            "range": self.get_range()
        })

# Example usage
if __name__ == "__main__":
    data1 = BandData()
    data1.register_value(5)
    data1.register_value(10)
    data1.register_value(15)
    
    data2 = BandData()
    data2.register_value(20)
    data2.register_value(25)
    
    combined_data = data1 + data2
    
    print("Data 1:", data1)
    print("Data 2:", data2)
    print("Combined Data:", combined_data)
