import argparse
import sys
from math import sqrt

def get_args():
    """Get command-line arguments"""
    parser = argparse.ArgumentParser(
        description='Standard Deviation Calculator',
        formatter_class=argparse.ArgumentDefaultsHelpFormatter)

    parser.add_argument(
        'values', metavar='int', type=int, nargs='+', help='Enter numbers separated by spaces')
    
    return parser.parse_args()

def warn(msg):
    """Print a warning message to stderr"""
    print(msg, file=sys.stderr)

def die(msg='Something bad happened'):
    """Print an error message and exit the program"""
    warn(msg)
    sys.exit(1)

def main():
    """Main function to calculate standard deviation"""
    args = get_args()
    num_vals = args.values
    length = len(num_vals)

    if length < 2:
        die('You need to enter more than one number.')

    # Calculate the mean
    mean = sum(num_vals) / length

    # Calculate the variance
    sq_sum = sum((x - mean) ** 2 for x in num_vals)
    variance = sq_sum / (length - 1)

    # Calculate the standard deviation
    stdev = sqrt(variance)

    # Print results
    print(f'The Mean is: {mean}')
    print(f'The Variance is: {variance}')
    print(f'The Standard Deviation is: {stdev}')

if __name__ == '__main__':
    main()
