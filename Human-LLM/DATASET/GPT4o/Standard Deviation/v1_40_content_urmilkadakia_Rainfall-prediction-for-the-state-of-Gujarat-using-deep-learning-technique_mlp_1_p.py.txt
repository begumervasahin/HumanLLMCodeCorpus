import os
import csv
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.neural_network import MLPRegressor
from sklearn.preprocessing import StandardScaler
from joblib import Parallel, delayed
import multiprocessing
import statistics

def machine(k, temp1, temp2):
    X_train, X_test, y_train, y_test = train_test_split(temp1, temp2, test_size=0.2, random_state=k)
    X_train = [[float(j) for j in i] for i in X_train]
    X_test = [[float(j) for j in i] for i in X_test]
    y_train = [int(i) for i in y_train]
    y_test = [int(i) for i in y_test]
    y_train = np.ravel(y_train)
    y_test = np.ravel(y_test)
    
    clf = MLPRegressor(max_iter=1000, learning_rate_init=0.05, solver='sgd', hidden_layer_sizes=(640,), alpha=1e-1)
    print("Mean of y_test:", statistics.mean(y_test))
    clf.fit(X_train, y_train)
    print("Predictions:", clf.predict(X_test))
    accuracy = clf.score(X_test, y_test)
    print("Accuracy:", accuracy)
    return accuracy

if __name__ == '__main__':
    dataPath = "/home/geospacial2/PycharmProjects/BTP/"
    filePath = os.path.join(dataPath, "rain2014_1.csv")

    temp1 = []
    temp2 = []

    with open(filePath, 'rt') as csvfile:
        reader = csv.reader(csvfile)
        next(reader, None)  # Skip the header
        for row in reader:
            temp1.append(row[9:17])
            temp2.append(row[22])
    
    temp1 = StandardScaler().fit_transform(temp1)
    temp2 = np.array(temp2, dtype=int)

    niter = 1
    inputs = range(niter)
    num_cores = multiprocessing.cpu_count()
    
    results = Parallel(n_jobs=num_cores)(delayed(machine)(k, temp1, temp2) for k in inputs)
    mean_accuracy = statistics.mean(results)
    print("Mean accuracy over all iterations:", mean_accuracy)
