import argparse
import sys

def classify_landcover(landcover, include_stddev):
    """
    Classify the given landcover type and optionally include the standard deviation.

    Parameters:
    landcover (str): The type of landcover to classify.
    include_stddev (bool): Whether to include the standard deviation in the output.
    """
    # Predefined classifications for different landcover types
    landcover_classification = {
        'water': 0,
        'woody savannas': 130
    }

    # Predefined standard deviation values for specific landcover types
    landcover_stddev = {
        'woody savannas': 58.40
    }

    # Check if the provided landcover type is known
    if landcover not in landcover_classification:
        print(f"Unknown landcover type: {landcover}")
        sys.exit(1)

    # Retrieve the classification for the given landcover type
    classification = landcover_classification[landcover]
    print(f"Landcover: {landcover}, Classification: {classification}")

    # If requested, include the standard deviation in the output
    if include_stddev and landcover in landcover_stddev:
        stddev = landcover_stddev[landcover]
        print(f"Standard Deviation: {stddev}")

def parse_arguments():
    """
    Parse the command-line arguments.

    Returns:
    argparse.Namespace: Parsed command-line arguments.
    """
    parser = argparse.ArgumentParser(description="Classify landcover types")
    parser.add_argument('--landcover', type=str, required=True, help="Type of landcover to classify")
    parser.add_argument('--stddev', action='store_true', help="Include standard deviation in output")
    
    return parser.parse_args()

def main():
    """
    Main function to classify landcover based on command-line arguments.
    """
    args = parse_arguments()

    # Ensure that a value is provided for the --landcover argument
    if not args.landcover:
        print("Error: --landcover requires a value")
        sys.exit(1)

    # Call the classify_landcover function with the parsed arguments
    classify_landcover(args.landcover, args.stddev)

if __name__ == '__main__':
    main()
