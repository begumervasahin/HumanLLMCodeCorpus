import requests
from bs4 import BeautifulSoup as BS
from pathlib import Path
from datetime import datetime, timedelta
import csv

# Number of historical days to retrieve
n_hist_days = 31

def gen_csv(n_hist_days=31):
    """
    Generates a CSV file with historical exchange rates for the past n_hist_days.
    """
    csv_data = []
    
    for i in range(n_hist_days, 0, -1):
        day = datetime.today() - timedelta(days=i)
        day_str = day.strftime('%Y-%m-%d')
        day_url = f'https://www.x-rates.com/historical/?from=USD&amount=1&date={day_str}'
        day_html = requests.get(day_url).text.split("<span class='ratesTableAlpha'>Alphabetical order</span>")[1]
        day_soup = BS(day_html, "html.parser")
        tds = day_soup.find_all('td')
        
        if i == n_hist_days:
            # Extract and save currency names on the first iteration
            names = [td.text for td in tds[::3]]
            csv_data.append(names)
        
        # Extract and save exchange rates
        day_rates = [td.text for td in tds[2::3]]
        csv_data.append(day_rates)
    
    # Write the collected data to a CSV file
    with open("data/hist_rates.csv", "w") as f:
        writer = csv.writer(f)
        writer.writerows(csv_data)
    
    print(f"Historical data obtained for last {n_hist_days} days.")

def daily_update():
    """
    Updates the CSV file with today's exchange rates.
    """
    config = Path('data/hist_rates.csv')
    
    if not config.is_file():
        print("Historical data not found, obtaining from x-rates.com...")
        gen_csv(n_hist_days)
    
    today = datetime.today().strftime('%Y-%m-%d')
    url = 'https://www.x-rates.com/table/?from=USD&amount=1'
    html = requests.get(url).text.split("<span class='ratesTableAlpha'>Alphabetical order</span>")[1]
    soup = BS(html, "html.parser")
    tds = soup.find_all('td')
    
    # Extract today's exchange rates
    today_rates = [td.text for td in tds[2::3]]
    
    # Append today's rates to the CSV file
    with open('data/hist_rates.csv', 'a') as f:
        writer = csv.writer(f)
        writer.writerow(today_rates)
    
    print(f"Today's currency rates ({today}) obtained...")

# Execute daily update
daily_update()
