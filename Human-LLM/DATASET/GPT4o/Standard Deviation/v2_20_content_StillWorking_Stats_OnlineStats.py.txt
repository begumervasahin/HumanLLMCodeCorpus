class OnlineStats:
    def __init__(self):
        """
        Initializes the OnlineStats object with mean, variance, and the count of numbers.
        """
        self.mean = 0
        self.variance = 0
        self.count = 0

    def update_stats(self, number):
        """
        Updates the mean and variance based on the new number entered.
        
        :param number: The new number to include in the statistics
        :return: A tuple containing the updated mean and variance
        """
        previous_mean = self.mean
        self.count += 1
        self.mean = previous_mean + (number - previous_mean) / self.count
        
        if self.count > 1:
            previous_variance = self.variance
            self.variance = ((self.count - 2) / (self.count - 1)) * previous_variance + (number - previous_mean) ** 2 / self.count
        
        return self.mean, self.variance

    def compute_statistics(self, input_function, output_function):
        """
        Continuously reads numbers from the input function, updates statistics, and sends results to the output function.
        
        :param input_function: Function to get user input
        :param output_function: Function to display the results
        """
        number = input_function()
        while number >= 0:
            mean, variance = self.update_stats(number)
            output_function(mean, variance)
            number = input_function()


def get_user_input():
    """
    Prompts the user to enter a number. Exits if a negative number is entered,
    and handles non-numeric input errors.
    
    :return: The number entered by the user
    """
    while True:
        try:
            value = float(input("Enter a number: "))
            if value < 0:
                print("Negative number detected. Exiting the program.")
                raise SystemExit()
            return value
        except ValueError:
            print("Invalid input. Please enter a numeric value.")


def display_results(mean, variance):
    """
    Displays the calculated mean and variance.
    
    :param mean: The mean of the numbers
    :param variance: The variance of the numbers
    """
    print(f"Mean: {mean}, Variance: {variance}")
