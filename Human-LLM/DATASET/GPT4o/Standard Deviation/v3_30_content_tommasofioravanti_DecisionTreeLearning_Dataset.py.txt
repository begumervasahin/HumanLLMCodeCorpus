class DataSet:
    """
    A data set for a machine learning problem. It has the following fields:
    d.examples   A list of examples. Each one is a dictionary of attribute values.
    d.attrs      A list of integers to index into an example, so example[attr]
                 gives a value. Normally the same as range(len(d.examples[0])).
    d.attrnames  A dictionary that identifies {attr: attrname,..} for all d.attrs.
    d.target     The attribute that a learning algorithm will try to predict.
                 By default the final attribute.
    d.inputs     The list of attrs without the target.
    d.values     A dictionary that identifies every possible value for all d.attrs {attr: [value1, value2, ..]}.
    """
    
    def __init__(self, name='', examples=None, inputs=None, attributes=None, target=None, attrnames=None, values=None):
        self.name = name
        self.examples = examples if examples else []
        self.values = values if values else {}
        
        self.attributes = attributes if attributes else (list(range(len(self.examples[0]))) if self.examples else [])
        self.attrnames = self._process_attrnames(attrnames)
        self.target = target if target is not None else (self.attributes[-1] if self.attributes else None)
        self.inputs = inputs if inputs is not None else [attr for attr in self.attributes if attr != self.target]
        
    def _process_attrnames(self, attrnames):
        if isinstance(attrnames, str):
            return {i: name for i, name in enumerate(attrnames.split())}
        return attrnames if attrnames else {attr: attr for attr in self.attributes}

# Example usage
examples = [

    0: [25, 35, 45],

    2: ['no', 'yes'],
    3: ['fair', 'excellent'],
    4: ['no', 'yes']
}

dataset = DataSet(name='Computer Purchase', examples=examples, attributes=attributes, attrnames=attrnames, values=values)
