import numpy as np
import random
import matplotlib.pyplot as plt

random.seed(0)
run_tests = [10, 100, 1000, 10000, 100000]
means = []
stds = []
scale = 0.1

def main():
    # Count the number of rows in the file
    with open('abalone.data') as file:
        num_rows = sum(1 for _ in file)
    
    # Perform simulations for each number of runs
    for r in run_tests:
        data_in = [0] * num_rows
        for i in range(r):
            row_len = len(data_in)
            n_n = float(int(row_len * scale))
            n_r = row_len
            for j in range(row_len):
                if n_n / n_r > random.random():
                    data_in[j] += 1
                    n_n -= 1
                n_r -= 1
        means.append(np.mean(data_in) / r)
        stds.append(np.std(data_in) / r)
    
    # Plot the results
    plt.subplot(2, 1, 1)
    plt.plot(run_tests, means)
    plt.title("Mean VS Runs")
    plt.ylabel("Mean")
    
    plt.subplot(2, 1, 2)
    plt.plot(run_tests, stds)
    plt.title("Standard Deviation VS Runs")
    plt.xlabel("Runs")
    plt.ylabel("Standard Deviation")
    
    plt.tight_layout()
    plt.show()

main()
