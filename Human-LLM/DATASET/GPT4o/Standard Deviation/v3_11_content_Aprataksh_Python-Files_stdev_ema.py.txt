import csv
import pandas as pd
import statistics
import os

def get_ticker_list(filepath):
    """
    Reads the CSV file containing the list of tickers and returns a list of ticker symbols.
    """
    ticker_list = []
    with open(filepath, 'r') as file:
        reader = csv.reader(file)
        for row in reader:
            if "Symbol" not in row:
                ticker_list.append(row[2])
    return ticker_list

def calculate_moving_averages(df, ndays):
    """
    Calculates the Simple Moving Average (SMA) and Exponential Moving Average (EMA) for the given DataFrame.
    Returns the DataFrame with new columns for SMA and EMA, and the standard deviation of the SMA.
    """
    SMA = df['Close'].rolling(window=ndays).mean().round(2)
    df['SMA'] = SMA
    SMA_list = SMA.dropna().tolist()
    SD = round(statistics.stdev(SMA_list), 2) if len(SMA_list) > 1 else 0

    EMA = df['Close'].ewm(span=15, min_periods=ndays).mean().round(2)
    df['EMA'] = EMA

    return df, SD

def process_ticker_data(ticker, input_dir, output_dir, ndays):
    """
    Processes the stock data for a single ticker: calculates SMA and EMA, saves the data to a new CSV file,
    and returns the standard deviation of the SMA.
    """
    input_file = os.path.join(input_dir, ticker + ".csv")
    output_file = os.path.join(output_dir, ticker + ".csv")

    if not os.path.exists(input_file):
        return 0

    df = pd.read_csv(input_file)
    if len(df.columns) < 6 or len(df.index) < ndays:
        return 0

    df, SD = calculate_moving_averages(df, ndays)
    df.to_csv(output_file, index=False)

    return SD

def process_stock_data(ticker_list, input_dir, output_dir, ndays=9):
    """
    Processes stock data for each ticker in the ticker list.
    Returns a list of standard deviations for the SMA of each ticker.
    """
    stdev_list = []

    for ticker in ticker_list:
        cleaned_ticker = ticker.replace('&', '%26')
        stdev = process_ticker_data(cleaned_ticker, input_dir, output_dir, ndays)
        stdev_list.append(stdev)

    return stdev_list

def save_stdev_list(ticker_list, stdev_list, output_file):
    """
    Saves the list of standard deviations to a new CSV file.
    """
    sdf = pd.DataFrame({'Ticker': ticker_list, 'Standard Deviation': stdev_list})
    sdf.to_csv(output_file, index=False)

def main():
    """
    Main function to orchestrate the process: reads ticker symbols, processes stock data,
    and saves the results.
    """
    ticker_file = r"C:\Users\Hp\Desktop\New folder\nifty500_list.csv"
    input_dir = r"C:\Users\Hp\Desktop\New folder\Top500_stock\Original data"
    output_dir = r"C:\Users\Hp\Desktop\New folder\Top500_stock\With SMA and EMA"
    stdev_output_file = r"C:\Users\Hp\Desktop\New folder\st_dev.csv"

    ticker_list = get_ticker_list(ticker_file)
    stdev_list = process_stock_data(ticker_list, input_dir, output_dir)
    save_stdev_list(ticker_list, stdev_list, stdev_output_file)

if __name__ == "__main__":
    main()
