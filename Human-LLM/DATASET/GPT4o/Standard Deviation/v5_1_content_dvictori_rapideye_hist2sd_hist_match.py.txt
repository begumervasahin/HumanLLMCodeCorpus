Program to balance colors between two images using histogram matching.
Receives two images, the first being the reference.
Returns a third image after performing the histogram match using CDF.
Author: Daniel
"""

import sys
import argparse
from osgeo import gdal
import numpy as np
from scipy.interpolate import interp1d

def balance_colors(reference_image, adjust_image, output_image):
    """
    Balances colors between two images using CDF matching.
    
    Parameters:
    reference_image (str): Path to the reference image.
    adjust_image (str): Path to the image to be adjusted.
    output_image (str): Path to the output image.
    """
    
    ref = gdal.Open(reference_image)
    adj = gdal.Open(adjust_image)

    if ref.RasterCount != adj.RasterCount:
        print("Images have different number of bands")
        sys.exit(1)
    
    bands = ref.RasterCount

    # Compute histograms and CDFs for reference and adjust images
    ref_hist = [ref.GetRasterBand(b+1).GetHistogram() for b in range(bands)]
    ref_cdf = [np.cumsum(hist) / float(sum(hist)) for hist in ref_hist]
    
    adj_hist = [adj.GetRasterBand(b+1).GetHistogram() for b in range(bands)]
    adj_cdf = [np.cumsum(hist) / float(sum(hist)) for hist in adj_hist]

    # Create inverse CDF functions for reference image
    inv_cdf_func = [
        interp1d(cdf, range(len(cdf)), bounds_error=False, fill_value=(0, len(cdf)-1)) 
        for cdf in ref_cdf
    ]

    # Apply transformation to adjust image's CDF
    transf_b = [inv_cdf_func[b](adj_cdf[b]) for b in range(bands)]
    transf = [np.insert(l, 0, 0) for l in transf_b]
    transf_func = [lambda x, b=b: transf[b][x] for b in range(bands)]

    # Create output image
    driver = gdal.GetDriverByName('GTiff')
    outfile = driver.CreateCopy(output_image, adj, 0, ['COMPRESS=LZW'])
    
    for b in range(bands):
        band_data = adj.GetRasterBand(b+1).ReadAsArray()
        corrected_band = transf_func[b](band_data)
        outfile.GetRasterBand(b+1).WriteArray(corrected_band)

    # Close datasets
    ref = None
    adj = None
    outfile = None

def main():
    parser = argparse.ArgumentParser(
        description='Perform histogram matching between a reference image and an image to be adjusted.'
    )
    parser.add_argument("reference_image", help="Path to the reference image")
    parser.add_argument("adjust_image", help="Path to the image to be adjusted")
    parser.add_argument("output_image", help="Path to the output image")
    args = parser.parse_args()

    balance_colors(args.reference_image, args.adjust_image, args.output_image)

if __name__ == '__main__':
    main()
