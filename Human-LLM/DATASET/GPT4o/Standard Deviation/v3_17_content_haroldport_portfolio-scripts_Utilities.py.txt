import os

class Utilities:
    @staticmethod
    def clear():
        """
        Clears the console screen based on the operating system.
        """
        command = 'cls' if os.name == 'nt' else 'clear'
        os.system(command)

    @staticmethod
    def create_ticker(stocks):
        """
        Prompts the user to enter a stock ticker that is not empty or already in the list of stocks.
        
        Args:
            stocks (list): A list of stock objects with a 'ticker' attribute.
            
        Returns:
            str: A unique stock ticker entered by the user.
        """
        while True:
            ticker = input('Enter ticker: ').strip()
            if ticker and not any(stock.ticker == ticker for stock in stocks):
                return ticker
            print('Empty or repeated value. Try again.')

    @staticmethod
    def create_percentage():
        """
        Prompts the user to enter a percentage between 0 and 100.
        
        Returns:
            int: A valid percentage entered by the user.
        """
        while True:
            try:
                percentage = int(input('Enter percentage: '))
                if 0 <= percentage <= 100:
                    return percentage
                print('Invalid percentage. Try again.')
            except ValueError:
                print('Invalid input. Please enter a number.')

    @staticmethod
    def create_number_of_actions():
        """
        Prompts the user to enter the number of actions as an integer.
        
        Returns:
            int: The number of actions entered by the user.
        """
        while True:
            try:
                actions = int(input('Enter number of actions: '))
                return actions
            except ValueError:
                print('Invalid input. Please enter a number.')
