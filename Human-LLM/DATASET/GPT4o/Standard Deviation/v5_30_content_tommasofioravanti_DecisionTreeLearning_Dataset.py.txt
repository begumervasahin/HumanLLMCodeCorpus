class DataSet:
    """
    A data set for a machine learning problem.

    Attributes:
    - name (str): The name of the dataset.
    - examples (list): A list of examples. Each example is a dictionary of attribute values.
    - attributes (list): A list of integers to index into an example, so example[attr] gives a value.
                         Normally the same as range(len(examples[0])).
    - attrnames (list or dict): A list or dictionary that identifies {attr: attrname, ...} for all attributes.
    - target (int): The attribute that a learning algorithm will try to predict. By default, the final attribute.
    - inputs (list): The list of attributes without the target.
    - values (dict): A dictionary that identifies every possible value for all attributes {attr: [value1, value2, ...]}.
    """

    def __init__(self, name='', examples=None, inputs=None, attributes=None, target=None, attrnames=None, values=None):
        self.name = name
        self.examples = examples
        self.target = target if target is not None else (len(examples[0]) - 1 if examples else None)
        self.values = values
        self.inputs = inputs if inputs is not None else [i for i in range(len(examples[0])) if i != self.target] if examples else None
        
        # Initialize attributes
        if attributes is None and examples is not None:
            self.attributes = list(range(len(examples[0])))
        else:
            self.attributes = attributes

        # Initialize attrnames
        if isinstance(attrnames, str):
            self.attrnames = attrnames.split()
        else:
            self.attrnames = attrnames or self.attributes

    def __repr__(self):
        return f"DataSet(name={self.name}, num_examples={len(self.examples) if self.examples else 0})"
