import numpy as np

# Constants
RADIUS_OF_EARTH = 6371.0088  # Radius of the Earth in kilometers
KMS_TO_METERS = 1000.0
METERS_TO_KILOMETERS = 0.001
MILES_TO_KILOMETERS = 1.60934
KILOMETERS_TO_MILES = 0.621371

def gcd(loc1, loc2, R=RADIUS_OF_EARTH * KMS_TO_METERS):
    """
    Calculates the great circle distance between two points on the Earth
    using the Haversine formula.

    Parameters:
    loc1 (tuple): Longitude and latitude of the first location in decimal degrees
    loc2 (tuple): Longitude and latitude of the second location in decimal degrees
    R (float): Radius of the Earth in meters (default is Earth's radius in meters)

    Returns:
    float: Great circle distance between loc1 and loc2 in the units of R

    Notes:
    This calculation assumes a perfect sphere and does not take into account the Earth's ellipsoidal shape.
    
    Example:
    >>> san_diego = (-117.1611, 32.7157)
    >>> austin = (-97.7431, 30.2672)
    >>> gcd(san_diego, austin)
    1155.474644164695
    """
    lon1, lat1 = loc1
    lon2, lat2 = loc2
    
    # Convert degrees to radians
    dLat = np.radians(lat2 - lat1)
    dLon = np.radians(lon2 - lon1)
    lat1 = np.radians(lat1)
    lat2 = np.radians(lat2)
    
    # Haversine formula
    a = np.sin(dLat / 2)**2 + np.cos(lat1) * np.cos(lat2) * np.sin(dLon / 2)**2
    c = 2 * np.arcsin(np.sqrt(a))
    
    return R * c

def compareFloat(a, b, rTol=0.00001, aTol=0.00000001):
    """
    Compares two floating-point numbers within a given tolerance.

    Parameters:
    a (float): First number to compare
    b (float): Second number to compare
    rTol (float): Relative tolerance (default is 0.00001)
    aTol (float): Absolute tolerance (default is 0.00000001)

    Returns:
    bool: True if the numbers are close within the specified tolerances, otherwise False
    """
    try:
        return abs(a - b) < aTol + (rTol * abs(b))
    except:
        return False

def convert2Radians(degree):
    """
    Converts an angle from degrees to radians.

    Parameters:
    degree (float): Angle in degrees

    Returns:
    float: Angle in radians
    """
    return np.pi / 180.0 * degree

def convert2degrees(radians):
    """
    Converts an angle from radians to degrees.

    Parameters:
    radians (float): Angle in radians

    Returns:
    float: Angle in degrees
    """
    return 180.0 / np.pi * radians
