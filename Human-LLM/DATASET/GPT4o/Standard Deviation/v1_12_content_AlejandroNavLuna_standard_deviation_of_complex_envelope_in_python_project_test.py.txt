import numpy as np
import math
import matplotlib.pyplot as plt

# Initialize parameters
n = 150
fvar = 0.1
m = 4
r = 1
pi = math.pi
e = math.e
lim = 1.5

# Generate random data
datos = np.round((m-1)*np.random.rand(r,n))

# Initialize arrays for storing results
a1bar = np.zeros(r)
a1var = np.zeros(r)
b1bar = np.zeros(r)
b1var = np.zeros(r)
datos_desv_estandar = np.zeros(r)

# Loop through each row in 'datos'
for i in range(r):
    angles = datos[i, :] * 2 * pi / m
    mag = fvar * np.random.normal(0, 1, n) + 1
    a = mag * np.cos(angles + fvar * np.random.normal(0, 1, n))
    b = mag * np.sin(angles + fvar * np.random.normal(0, 1, n))
    
    # Plot all points
    plt.plot(a, b, 'ro')
    plt.axis([-lim, lim, -lim, lim])
    plt.title("All Points")
    plt.show()
    
    # Filter points within the specified range
    a1 = a[(0.5 < a) & (a < 1.5) & (b < 0.5) & (b > -0.5)]
    b1 = b[(0.5 < a) & (a < 1.5) & (b < 0.5) & (b > -0.5)]
    
    # Plot filtered points
    plt.plot(a1, b1, 'ro')
    plt.axis([-lim, lim, -lim, lim])
    plt.title("Filtered Points")
    plt.show()
    
    # Calculate mean and variance
    if a1.shape[0] > 0:  # Ensure there are points within the range
        a1bar[i] = np.sum(a1) / a1.shape[0]
        a1var[i] = np.sum((a1 - a1bar[i]) ** 2) / (a1.shape[0] - 1)
        b1bar[i] = np.sum(b1) / b1.shape[0]
        b1var[i] = np.sum((b1 - b1bar[i]) ** 2) / (b1.shape[0] - 1)
        datos_desv_estandar[i] = np.sqrt(a1var[i] + b1var[i])
    
    print("b1 variance:", b1var[i])
    print("a1 variance:", a1var[i])
    print("Standard deviation:", datos_desv_estandar[i])

# Find the signal with the minimum standard deviation
resultado = np.where(datos_desv_estandar == np.amin(datos_desv_estandar))
print('Best signal for the sample is signal:', resultado[0])
print('Data of the signal:', datos[resultado, :])
