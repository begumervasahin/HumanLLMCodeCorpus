import numpy as np
import h5py

# Define parameters
frames = [100, 1000, 10000]
exp_times = [0.001, 0.01, 0.1, 1]
pix_rates = ['slow', 'fast']

# Loop through each combination of frames, exposure time, and pixel rate
for frame_count in frames:
    for exp_time in exp_times:
        for pix_rate in pix_rates:
            mean = []
            std = []
            mnm = []
            mxm = []

            # Determine the number of chunks based on frame count
            chunks = frame_count // 100
            num_files = ["%06d" % a for a in range(1, chunks + 1)]

            # Loop through each chunk file
            for file_number in num_files:
                file_name = f'{frame_count}_frames_{exp_time}_{pix_rate}_{file_number}'
                with h5py.File(file_name, 'r') as fid:
                    data = fid['/entry/data']
                    mean.append(np.mean(data[0:99], axis=0))
                    std.append(np.std(data[0:99], axis=0))
                    mnm.append(np.min(data[0:99], axis=0))
                    mxm.append(np.max(data[0:99], axis=0))

            # Calculate grand statistics
            grand_mean = np.mean(mean)
            grand_std = np.sqrt(sum([m**2 for m in std]) / chunks)
            grand_mnm = np.min(mnm)
            grand_mxm = np.max(mxm)

            # Save the grand statistics to files
            np.save(f'mean_{frame_count}frames_{exp_time}s_{pix_rate}.npy', grand_mean)
            np.save(f'std_{frame_count}frames_{exp_time}s_{pix_rate}.npy', grand_std)
            np.save(f'min_{frame_count}frames_{exp_time}s_{pix_rate}.npy', grand_mnm)
            np.save(f'max_{frame_count}frames_{exp_time}s_{pix_rate}.npy', grand_mxm)
