import os

class Utilities:
    @staticmethod
    def clear():
        """Clears the console screen based on the operating system."""
        command = 'cls' if os.name == 'nt' else 'clear'
        os.system(command)

    @staticmethod
    def create_ticker(stocks):
        """Prompts the user to enter a unique stock ticker."""
        while True:
            ticker = input('Enter ticker: ').strip()
            if ticker and all(stock['ticker'] != ticker for stock in stocks):
                return ticker
            print('Empty or repeated value. Try again.')

    @staticmethod
    def create_percentage():
        """Prompts the user to enter a valid percentage between 0 and 100."""
        while True:
            try:
                percentage = int(input('Enter percentage: '))
                if 0 <= percentage <= 100:
                    return percentage
                else:
                    print('Percentage must be between 0 and 100. Try again.')
            except ValueError:
                print('Invalid input. Enter a valid percentage between 0 and 100.')

    @staticmethod
    def create_number_of_actions():
        """Prompts the user to enter a valid number of actions."""
        while True:
            try:
                actions = int(input('Enter number of actions: '))
                return actions
            except ValueError:
                print('Invalid input. Enter a valid number of actions.')

# Example usage
if __name__ == "__main__":
    Utilities.clear()

    # List of existing stocks
    stocks = [{'ticker': 'AAPL'}, {'ticker': 'GOOGL'}, {'ticker': 'MSFT'}]

    # Create a new ticker
    ticker = Utilities.create_ticker(stocks)
    print(f'Created ticker: {ticker}')

    # Create a valid percentage
    percentage = Utilities.create_percentage()
    print(f'Created percentage: {percentage}')

    # Create a valid number of actions
    actions = Utilities.create_number_of_actions()
    print(f'Created number of actions: {actions}')
