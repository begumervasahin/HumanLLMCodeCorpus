import argparse
import sys
from math import sqrt

def get_args():
    """Get command-line arguments."""
    parser = argparse.ArgumentParser(
        description='Standard Deviation Calculator',
        formatter_class=argparse.ArgumentDefaultsHelpFormatter)

    parser.add_argument(
        'values', metavar='int', type=int, nargs='+', help='Enter numbers separated by spaces')
    
    return parser.parse_args()

def warn(msg):
    """Print a warning message to stderr."""
    print(msg, file=sys.stderr)

def die(msg='Something bad happened'):
    """Print an error message and exit the program."""
    warn(msg)
    sys.exit(1)

def calculate_mean(numbers):
    """Calculate the mean of a list of numbers."""
    return sum(numbers) / len(numbers)

def calculate_variance(numbers, mean):
    """Calculate the variance of a list of numbers."""
    return sum((x - mean) ** 2 for x in numbers) / (len(numbers) - 1)

def main():
    """Main function to calculate standard deviation."""
    args = get_args()
    numbers = args.values

    if len(numbers) < 2:
        die('You need to enter more than one number.')

    mean = calculate_mean(numbers)
    variance = calculate_variance(numbers, mean)
    standard_deviation = sqrt(variance)

    print(f'The Mean is: {mean}')
    print(f'The Variance is: {variance}')
    print(f'The Standard Deviation is: {standard_deviation}')

if __name__ == '__main__':
    main()
