import csv

# Constants
STATION_ID = 'ISRO0009_15F009(PRL..Ahmedabad)'
HEADER = ['@STATION_ID', 'LATITUDE', 'LONGITUDE', 'ALTITUDE(m)', 'TIME(GMT)', 'DATE(GMT)',
          'AIR_TEMP(°C)', 'WIND_SPEED(m/s)', 'WIND_DIRECTION(deg)', 'ATMO_PRESSURE(hpa)',
          'HUMIDITY(%)', 'RAIN_FALL(mm)', 'ERROR']
INPUT_FILE = 'output2016.csv'
OUTPUT_FILE = 'rain2016.csv'
PLACEHOLDER_RAINFALL = 9999.9
MAX_RAINFALL_DIFFERENCE = 924
ADJUSTMENT_CONSTANT = 1023
NEGATIVE_THRESHOLD = 0
EXCESS_THRESHOLD = 100

# Variables
aws_data = []
rain = 0
prev_time = 23
err_neg = 0
err_ex = 0

def read_input_file(file_path):
    with open(file_path, 'r') as file:
        reader = csv.reader(file, delimiter=',')
        next(reader)  # Skip header
        return [row for row in reader if float(row[11]) != PLACEHOLDER_RAINFALL]

def process_data(data):
    global rain, err_neg, err_ex
    processed_data = []

    for i, row in enumerate(data):
        # Skip initial condition setup for the specified date
        if row[5] == '05/31/2016':
            rain = float(row[11])
            continue

        # Calculate the current rainfall value
        temp = float(row[11])
        if rain - temp > MAX_RAINFALL_DIFFERENCE:
            row[11] = temp + (ADJUSTMENT_CONSTANT - rain)
        else:
            row[11] = temp - rain
        
        rain = temp
        
        # Check for negative or excessive rainfall values and append the appropriate error
        if float(row[11]) < NEGATIVE_THRESHOLD:
            row.append('Neg')
            err_neg += 1
        elif float(row[11]) > EXCESS_THRESHOLD:
            row.append('Excess')
            err_ex += 1
        else:
            row.append('')

        processed_data.append(row)
    
    return processed_data

def write_output_file(file_path, data):
    with open(file_path, 'w+', newline='') as file:
        writer = csv.writer(file)
        writer.writerow(HEADER)
        writer.writerows(data)

def main():
    # Read input file
    aws_data = read_input_file(INPUT_FILE)

    # Process data
    processed_data = process_data(aws_data)

    # Write to output file
    write_output_file(OUTPUT_FILE, processed_data)

    # Print error counts
    print(f"Negative Rainfall Errors: {err_neg}")
    print(f"Excessive Rainfall Errors: {err_ex}")

if __name__ == '__main__':
    main()
