def print_results(output_file, optimal_trees, n_max, suboptimal_trees, second_best, total_tree_count, comparisons, total_sum, file_name, probabilities, avgs, std_devs, start_time):
    """
    Main function to print the results of the OBST algorithms, including optimal trees and sub-optimal trees.
    """
    print("\n\n", file=output_file)
    print("LEVEL-ORDER TRAVERSAL   (key)->{L: Level, R: Root, P: Probability/Frequency}", file=output_file)
    print("Tree(s): Optimal Tree(s)", file=output_file)

    # Print the first 5 optimal trees
    for idx, tree in enumerate(optimal_trees[:5]):
        print_tree(tree, probabilities, idx, output_file)
    
    # If there is only one optimal tree, build and print the second-best tree
    if len(optimal_trees) == 1:
        build_trees(1, n_max, suboptimal_trees, second_best, 1, n_max, total_tree_count, comparisons)
        organized_suboptimal_trees = organize_trees(suboptimal_trees, n_max)
        print("\n\nTree: Second Best (Sub-Optimal Tree)", file=output_file)
        for tree in organized_suboptimal_trees:
            print_tree(tree, probabilities, 0, output_file)
        suboptimal_avgs, suboptimal_std_devs = calculate_std_deviation(organized_suboptimal_trees, total_sum, n_max, probabilities)

    # Print important tree information
    print("\n\nIMPORTANT TREE INFORMATION\n\nOptimal Tree(s): ", len(optimal_trees), file=output_file)
    for idx, tree in enumerate(optimal_trees[:5]):
        print_tree_info(file_name, n_max, avgs[idx], std_devs[idx], tree[-1][0], idx + 1, output_file)
    
    # If there is only one optimal tree, print the second-best tree information
    if len(optimal_trees) == 1:
        print("\nSecond Best Tree (Sub-Optimal)", file=output_file)
        print_tree_info(file_name, n_max, suboptimal_avgs[0], suboptimal_std_devs[0], organized_suboptimal_trees[0][-1][0], 1, output_file)
    
    # Print total comparisons and computation time
    print("\nTotal Comparisons:", comparisons[0], file=output_file)
    end_time = timeit.default_timer()
    print("Total Computation Time: {:4.2f}".format(end_time - start_time), "seconds", file=output_file)
    print("Total Computation Time: {:4.2f}".format((end_time - start_time) / 60), "minutes", file=output_file)
    
    # Print platform information
    get_platform_info(output_file)
    print("\n\n\n", file=output_file)
