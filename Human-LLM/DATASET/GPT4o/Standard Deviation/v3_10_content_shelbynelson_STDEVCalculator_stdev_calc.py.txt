import argparse
import sys
from math import sqrt

def get_args():
    """Get command-line arguments"""
    parser = argparse.ArgumentParser(
        description='Standard Deviation Calculator',
        formatter_class=argparse.ArgumentDefaultsHelpFormatter)
    parser.add_argument(
        'values', metavar='int', type=int, nargs='+', help='Enter numbers separated by spaces')
    return parser.parse_args()

def warn(msg):
    """Print a warning message to stderr"""
    print(msg, file=sys.stderr)

def die(msg):
    """Print an error message to stderr and exit"""
    warn(msg)
    sys.exit(1)

def calculate_mean(numbers):
    """Calculate the mean of a list of numbers"""
    return sum(numbers) / len(numbers)

def calculate_variance(numbers, mean):
    """Calculate the variance of a list of numbers"""
    return sum((x - mean) ** 2 for x in numbers) / (len(numbers) - 1)

def calculate_standard_deviation(numbers):
    """Calculate the standard deviation of a list of numbers"""
    mean = calculate_mean(numbers)
    variance = calculate_variance(numbers, mean)
    return sqrt(variance), mean, variance

def main():
    """Main function to calculate the standard deviation"""
    args = get_args()
    num_vals = args.values
    
    if len(num_vals) < 2:
        die('You need to enter more than one number.')
    
    stdev, mean, variance = calculate_standard_deviation(num_vals)
    
    print(f'The Mean is: {mean:.2f}')
    print(f'The Variance is: {variance:.2f}')
    print(f'The Standard Deviation is: {stdev:.2f}')

if __name__ == '__main__':
    main()
