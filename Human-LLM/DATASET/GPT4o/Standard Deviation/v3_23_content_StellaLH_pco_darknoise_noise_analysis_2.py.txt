import numpy as np
import h5py

# Define the parameters for frames, exposure time, and pixel rate
frames = [100, 1000, 10000]
exposure_times = [0.001]
pixel_rates = ['slow', 'fast']

def calculate_statistics(data):
    """Calculate mean, std, min, and max along axis 0."""
    return {
        'mean': np.mean(data, axis=0),
        'std': np.std(data, axis=0),
        'min': np.min(data, axis=0),
        'max': np.max(data, axis=0),
    }

def aggregate_statistics(stats_list, chunks):
    """Aggregate mean, std, min, and max statistics."""
    means = [stat['mean'] for stat in stats_list]
    stds = [stat['std'] for stat in stats_list]
    mins = [stat['min'] for stat in stats_list]
    maxs = [stat['max'] for stat in stats_list]

    grand_mean = np.mean(means, axis=0)
    grand_std = np.sqrt(np.sum([std**2 for std in stds]) / chunks)
    grand_min = np.min(mins, axis=0)
    grand_max = np.max(maxs, axis=0)

    return grand_mean, grand_std, grand_min, grand_max

def save_statistics(frame_count, exposure, rate, stats):
    """Save the calculated statistics to numpy files."""
    np.save(f'mean_{frame_count}frames_{exposure}s_{rate}.npy', stats['mean'])
    np.save(f'std_{frame_count}frames_{exposure}s_{rate}.npy', stats['std'])
    np.save(f'min_{frame_count}frames_{exposure}s_{rate}.npy', stats['min'])
    np.save(f'max_{frame_count}frames_{exposure}s_{rate}.npy', stats['max'])

# Iterate through the combinations of frames, exposure times, and pixel rates
for frame_count in frames:
    for exposure in exposure_times:
        for rate in pixel_rates:
            print(f"Processing: frames={frame_count}, exposure={exposure}, rate={rate}")

            stats_list = []
            chunks = frame_count // 100
            file_indices = ["%06d" % i for i in range(1, chunks + 1)]

            # Process each chunk file
            for idx in file_indices:
                filename = f'{frame_count}_frames_{rate}_{idx}.h5'
                with h5py.File(filename, 'r') as file:
                    data = file['/entry/data']
                    stats_list.append(calculate_statistics(data))

            # Aggregate statistics
            grand_mean, grand_std, grand_min, grand_max = aggregate_statistics(stats_list, chunks)
            final_stats = {
                'mean': grand_mean,
                'std': grand_std,
                'min': grand_min,
                'max': grand_max,
            }

            # Save the final aggregated statistics
            save_statistics(frame_count, exposure, rate, final_stats)
