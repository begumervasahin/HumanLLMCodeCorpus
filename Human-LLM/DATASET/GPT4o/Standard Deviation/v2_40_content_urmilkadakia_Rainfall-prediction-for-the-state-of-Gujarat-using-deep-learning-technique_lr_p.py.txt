import os
import csv
import numpy as np
import statistics
from sklearn.linear_model import LinearRegression
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from joblib import Parallel, delayed
import multiprocessing

def machine(k, temp1, temp2):
    # Splitting the dataset into training and testing sets
    X_train, X_test, y_train, y_test = train_test_split(temp1, temp2, test_size=0.2, random_state=k)
    
    # Converting the data to float and int where necessary
    X_train = np.array(X_train, dtype=float)
    X_test = np.array(X_test, dtype=float)
    y_train = np.array(y_train, dtype=int)
    y_test = np.array(y_test, dtype=int)
    
    # Initializing and training the Linear Regression model
    clf = LinearRegression()
    clf.fit(X_train, y_train)
    
    # Predicting and displaying results
    predictions = clf.predict(X_test)
    print("Mean of y_test:", statistics.mean(y_test))
    print("Predictions:", predictions)
    print("Coefficients:", clf.coef_)
    print("Intercept:", clf.intercept_)
    
    # Calculating and printing the accuracy of the model
    accuracy = clf.score(X_test, y_test)
    print("Model accuracy:", accuracy)
    
    return accuracy

if __name__ == '__main__':
    # Path to the data file
    data_path = "/home/geospacial2/PycharmProjects/BTP/rain2014_1.csv"
    
    # Reading the data from CSV file
    with open(data_path, 'rt') as csvfile:
        reader = csv.reader(csvfile)
        next(reader, None)  # Skip the header row
        temp1 = []
        temp2 = []
        for row in reader:
            temp1.append(row[9:17])
            temp2.append(row[22])
    
    # Scaling the features
    temp1 = StandardScaler().fit_transform(temp1)
    
    # Number of iterations for parallel processing
    n_iter = 1
    inputs = range(n_iter)
    
    # Determine the number of CPU cores available
    num_cores = multiprocessing.cpu_count()
    
    # Run the machine function in parallel
    results = Parallel(n_jobs=num_cores)(delayed(machine)(k, temp1, temp2) for k in inputs)
    
    # Print the results from all iterations
    for i, result in enumerate(results):
        print(f"Iteration {i+1} - Accuracy: {result}")
