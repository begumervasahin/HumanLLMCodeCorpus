import os
import csv
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.neural_network import MLPRegressor
from sklearn.preprocessing import StandardScaler
from joblib import Parallel, delayed
import multiprocessing
import statistics

def machine(k, X, y):
    # Split the data into training and testing sets
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=k)
    
    # Convert the data to appropriate types
    X_train = np.array(X_train, dtype=float)
    X_test = np.array(X_test, dtype=float)
    y_train = np.array(y_train, dtype=int)
    y_test = np.array(y_test, dtype=int)
    
    # Initialize and train the MLP regressor
    clf = MLPRegressor(max_iter=1000, learning_rate_init=0.05, solver='sgd', hidden_layer_sizes=(640,), alpha=1e-1)
    clf.fit(X_train, y_train)
    
    # Predict and evaluate the model
    predictions = clf.predict(X_test)
    accuracy = clf.score(X_test, y_test)
    
    # Print the results
    print("Mean of y_test:", statistics.mean(y_test))
    print("Predictions:", predictions)
    print("Accuracy:", accuracy)
    
    return accuracy

def load_data(filepath):
    temp1 = []
    temp2 = []
    
    # Read the CSV file and extract relevant data
    with open(filepath, 'rt') as csvfile:
        reader = csv.reader(csvfile)
        next(reader)  # Skip the header row
        for row in reader:
            temp1.append(row[9:17])
            temp2.append(row[22])
    
    return np.array(temp1), np.array(temp2, dtype=int)

if __name__ == '__main__':
    # Define the data path and load the data
    dataPath = "/home/geospacial2/PycharmProjects/BTP/"
    filePath = os.path.join(dataPath, "rain2014_1.csv")
    
    X, y = load_data(filePath)
    
    # Standardize the features
    X = StandardScaler().fit_transform(X)
    
    # Define the number of iterations and the number of CPU cores
    n_iter = 1
    num_cores = multiprocessing.cpu_count()
    
    # Run the machine function in parallel
    results = Parallel(n_jobs=num_cores)(delayed(machine)(k, X, y) for k in range(n_iter))
    
    # Calculate and print the mean accuracy
    mean_accuracy = statistics.mean(results)
    print("Mean accuracy over all iterations:", mean_accuracy)
