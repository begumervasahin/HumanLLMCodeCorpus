import requests
from bs4 import BeautifulSoup

def generate_url(country, job, location, salary):
    """
    Generates the Indeed search URL based on user input.

    Args:
        country (str): Country code ('US' or 'CA').
        job (str): Job title or keywords.
        location (str): Desired job location.
        salary (str): Desired salary.

    Returns:
        str: Generated URL for the job search.
    """
    job_query = "+".join(job.split())
    location_query = "+".join(location.split())
    salary_query = "24" + str(int(salary) // 1000)
    
    if country == 'US':
        url = f"https://www.indeed.com/jobs?q={job_query}+{salary_query}%2C000&l={location_query}"
    elif country == 'CA':
        url = f"https://ca.indeed.com/jobs?q={job_query}+{salary_query}%2C000&l={location_query}"
    else:
        print("That's not a valid option.")
        url = ''
    
    return url

def extract_job_titles(soup):
    """
    Extracts job titles from the search result soup.

    Args:
        soup (BeautifulSoup): BeautifulSoup object of the search result page.

    Returns:
        list: List of job titles.
    """
    job_titles = [a["title"] for div in soup.find_all(name="div", attrs={"class":"jobsearch-SerpJobCard"})
                  for a in div.find_all(name="a", attrs={"data-tn-element":"jobTitle"})]
    return job_titles

def extract_locations(soup):
    """
    Extracts job locations from the search result soup.

    Args:
        soup (BeautifulSoup): BeautifulSoup object of the search result page.

    Returns:
        list: List of job locations.
    """
    locations = [div["data-rc-loc"] for div in soup.find_all(name="div", attrs={"class":"recJobLoc"})]
    return locations

def extract_salaries(soup):
    """
    Extracts job salaries from the search result soup.

    Args:
        soup (BeautifulSoup): BeautifulSoup object of the search result page.

    Returns:
        list: List of job salaries.
    """
    salaries = []
    for div in soup.find_all(name="div", attrs={"class":"jobsearch-SerpJobCard"}):
        try:
            salary = div.find("span", attrs={"class": "salaryText"}).text.strip()
        except AttributeError:
            salary = "Not specified"
        salaries.append(salary)
    return salaries

def extract_summaries(soup):
    """
    Extracts job summaries from the search result soup.

    Args:
        soup (BeautifulSoup): BeautifulSoup object of the search result page.

    Returns:
        list: List of job summaries.
    """
    summaries = [div.text.strip() for div in soup.find_all(name="div", attrs={"class":"summary"})]
    return summaries

def find_jobs(url):
    """
    Finds and prints job listings from a given URL.

    Args:
        url (str): URL of the job search page.
    """
    if not url:
        return
    
    page = requests.get(url)
    soup = BeautifulSoup(page.text, "html.parser")
    
    jobs = extract_job_titles(soup)
    locations = extract_locations(soup)
    salaries = extract_salaries(soup)
    
    fmt = '{:<10}{:<60}{:<70}{}'
    for i, (job, location, salary) in enumerate(zip(jobs, locations, salaries)):       
        print(fmt.format(i, job, location, salary))

def generate_page_urls(base_url):
    """
    Generates paginated URLs for further job listings.

    Args:
        base_url (str): Base URL for the job search.

    Returns:
        list: List of paginated URLs.
    """
    return [base_url + f"&start={x}" for x in range(10, 40, 10)]

def main():
    """
    Main function to prompt user input and display job listings.
    """
    country = input("Which country are you looking to work in (CA/US)?: ")
    job = input("Type of job you're looking for: ")
    location = input("Desired location: ")
    salary = input("Desired salary: ")
    
    base_url = generate_url(country, job, location, salary)
    page_urls = generate_page_urls(base_url)
    
    fmt = '{:<10}{:<60}{:<70}{}'
    print(fmt.format('', 'Job Title', 'Location', 'Salary'))
    
    for page_number, page_url in enumerate(page_urls, start=1):
        print(f"Page {page_number}")
        find_jobs(page_url)

if __name__ == "__main__":
    main()
