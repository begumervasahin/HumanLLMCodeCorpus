import time
from selenium import webdriver
from selenium.webdriver.support.ui import Select

# Setup Chromedriver path and URL
chromedriver_path = "C:\\Users\\raghu\\Desktop\\Chromedriver\\chromedriver.exe"
url = "http://court.mah.nic.in/courtweb/index_eng.php"

# Initialize the webdriver
driver = webdriver.Chrome(chromedriver_path)
driver.get(url)
driver.maximize_window()

# Function to select a district
def select_district(district_value):
    district_select = Select(driver.find_element_by_class_name("selectdistrict"))
    district_select.select_by_value(district_value)

# Function to navigate through the menus
def navigate_menus():
    driver.find_element_by_xpath('//*[@id="fourlinks"]/ul/li[2]/div[1]/a').click()
    time.sleep(3)
    driver.find_element_by_xpath('//*[@id="fourlinks"]/ul/li[2]/div[2]/ul/li[4]/a').click()
    time.sleep(5)

# Function to select the court complex
def select_court_complex():
    driver.switch_to.frame('data')
    driver.find_element_by_xpath('//*[@id="radCourtComplex"]').click()
    court_complex_select = Select(driver.find_element_by_xpath('//*[@id="court_complex_code"]'))
    court_complex_select.select_by_visible_text("District and Sessions Court, Shivajinagar, Pune - 411 005")

# Function to select the date range
def select_date_range(from_month_year, from_day, to_day):
    driver.find_element_by_xpath('//*[@id="from_date"]').click()
    from_date_select = Select(driver.find_element_by_class_name("datepick-month-year"))
    from_date_select.select_by_value(from_month_year)
    driver.find_element_by_xpath(f'/html/body/div[2]/div/div[2]/div/table/tbody/tr[1]/td[{from_day}]/a').click()
    time.sleep(2)

    driver.find_element_by_xpath('//*[@id="to_date"]').click()
    time.sleep(2)
    driver.find_element_by_xpath('/html/body/div[2]/div/div[2]/div/div/select[1]/option[1]').click()
    time.sleep(2)
    driver.find_element_by_xpath(f'/html/body/div[2]/div/div[2]/div/table/tbody/tr[1]/td[{to_day}]/a').click()

# Function to search for cases
def search_cases():
    print("ENTER CAPTCHA AND WAIT for 20 seconds")
    time.sleep(20)
    driver.find_element_by_xpath('//*[@id="caseNoDet"]/div[7]/span[3]/input[1]').click()
    time.sleep(10)

# Function to save case details to a file
def save_case_details(start_index):
    for i in range(3, 7):
        case_info_xpath = f'//*[@id="showList2"]/div[{i}]/a'
        case_info = driver.find_element_by_xpath(case_info_xpath).text
        filename, rows = case_info.split(':')[0], int(case_info.split(':')[1][1:])

        with open(f"{filename}.txt", "w") as file:
            headers = "(Sr No, Case Type/Case Number/Case Year, Order Date, Order No.)\n"
            file.write(headers)

            for row in range(start_index, start_index + rows):
                row_data = [driver.find_element_by_xpath(f'//*[@id="showList1"]/tr[{row}]/td[{col}]').text for col in range(1, 5)]
                file.write('\t'.join(row_data) + '\n')

            start_index += rows

# Execute the script steps
select_district("25")
navigate_menus()
select_court_complex()
select_date_range("1/2019", 3, 4)
search_cases()
save_case_details(2)

# Close the driver
driver.quit()
