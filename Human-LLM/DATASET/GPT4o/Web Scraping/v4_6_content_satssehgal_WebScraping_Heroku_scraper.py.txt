import os
import base64
import requests
import pandas as pd
from bs4 import BeautifulSoup as bs4
from sendgrid import SendGridAPIClient
from sendgrid.helpers.mail import (Mail, Attachment, FileContent, FileName, FileType, Disposition, ContentId)
import schedule
import time
from datetime import datetime
import pytz

# Constants for SendGrid
FROM_EMAIL = "your_from_email@example.com"
TO_EMAIL = "your_to_email@example.com"
SENDGRID_API_KEY = "your_sendgrid_api_key"

# Function to scrape data from the website
def job():
    prices = []
    stars = []
    titles = []
    urlss = []
    
    # Number of pages to scrape
    pages_to_scrape = 10
    pages = [f'http://books.toscrape.com/catalogue/page-{i}.html' for i in range(1, pages_to_scrape + 1)]
    
    for item in pages:
        page = requests.get(item)
        soup = bs4(page.text, 'html.parser')
        
        # Extract book titles
        for i in soup.findAll('h3'):
            titles.append(i.getText())
        
        # Extract book prices
        for j in soup.findAll('p', class_='price_color'):
            prices.append(j.getText())
        
        # Extract star ratings
        for s in soup.findAll('p', class_='star-rating'):
            for k, v in s.attrs.items():
                stars.append(v[1])
        
        # Extract book image URLs
        divs = soup.findAll('div', class_='image_container')
        for thumbs in divs:
            tgs = thumbs.find('img', class_='thumbnail')
            urls = 'http://books.toscrape.com/' + str(tgs['src'])
            newurls = urls.replace("../", "")
            urlss.append(newurls)
    
    # Create a DataFrame from the scraped data
    data = {'Title': titles, 'Prices': prices, 'Stars': stars, "URLs": urlss}
    df = pd.DataFrame(data)
    df.index += 1
    
    # Save DataFrame to CSV
    directory = os.path.dirname(os.path.realpath(__file__))
    filename = "scrapedfile.csv"
    file_path = os.path.join(directory, 'csvfiles/', filename)
    df.to_csv(file_path)
    
    # Read the CSV file and encode it in base64
    with open(file_path, 'rb') as f:
        data = f.read()
    encoded = base64.b64encode(data).decode()
    
    # Create the email with attachment
    message = Mail(
        from_email=FROM_EMAIL,
        to_emails=TO_EMAIL,
        subject='Your File is Ready',
        html_content='<strong>Attached is Your Scraped File</strong>'
    )
    attachment = Attachment(
        FileContent(encoded),
        FileName('scraped.csv'),
        FileType('text/csv'),
        Disposition('attachment'),
        ContentId('Example Content ID')
    )
    message.attachment = attachment
    
    # Send the email
    try:
        sg = SendGridAPIClient(SENDGRID_API_KEY)
        response = sg.send(message)
        print(response.status_code)
        print(response.body)
        print(response.headers)
    except Exception as e:
        print(e)

# Schedule the job to run every day at a specific time
schedule.every().day.at('13:58').do(job)

# Keep the script running
while True:
    schedule.run_pending()
    time.sleep(1)
