import re
from urllib.request import urlopen
from bs4 import BeautifulSoup

def fetch_tennis_scores():
    """Fetch and display tennis scores from ESPN."""
    url = 'http://m.espn.com/general/tennis/dailyresults?date=20200125&matchType=1&wjb='
    response = urlopen(url)
    soup = BeautifulSoup(response, 'html.parser')
    
    # Extract tournament information
    tournament_info = soup.find("div", class_="sec row")
    tournament_name = ''.join(re.findall(r'([A-Z]\w+-*\w*)', tournament_info.text))
    
    # Extract round information
    round_info = soup.findAll("div", class_="ind sub bold")
    round_name = next((line.text for line in round_info if line.text != "FULL TOURNAMENT RESULTS"), "")
    
    # Extract match scores
    matches_info = soup.findAll("div", class_="ind")
    match_scores = [line.text.split("Final")[0] for line in matches_info if "Final" in line.text]
    
    process_matches(match_scores, tournament_name, round_name)

def process_matches(match_scores, tournament_name, round_name):
    match_count = sum(1 for match in match_scores if match)
    display_final_scores(match_scores, match_count, tournament_name, round_name)

def display_final_scores(match_scores, match_count, tournament_name, round_name):
    for match in match_scores:
        if match and match not in ["FULL TOURNAMENT RESULTS", "TENNIS HOME PAGE"]:
            formatted_score = re.sub(r'([a-z])(\d{1})', r'\1 \2', match)
            players = re.findall(r'([A-Z]\w+-*\w*)', formatted_score)
            round_details = re.findall(r'^[^:]+', round_name)
            if len(players) > 1:
                print(f"{round_details[0]} of the {tournament_name}: {players[0]} vs {players[1]}")
                print(f"Score: {formatted_score}")

# Fetch and display tennis scores
fetch_tennis_scores()
