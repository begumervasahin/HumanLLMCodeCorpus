from flask import Flask, render_template, request
import socket
import time

app = Flask(__name__)
app.secret_key = "Non-seen"

COMMON_PORTS = [21, 22, 23, 25, 27, 50, 53, 69, 70, 80, 87, 88, 109, 110, 113, 143, 1080, 8080, 8088]

@app.route('/scan', methods=['GET', 'POST'])
def scan():
    if request.method == 'GET':
        return render_template("scanner.html")
    
    area = request.form['area'].lower()
    ip = request.form['url']
    
    addr_type = 'ip' if ip[0].isdigit() else 'url'
    
    if area == 'ip':
        return handle_ip_scan(addr_type, ip)
    elif area == 'name':
        return handle_name_resolution(ip)
    elif area == 'ports':
        return handle_port_scan(ip)
    elif area == 'services':
        return handle_service_scan(ip)
    else:
        return render_template('scanner.html', error=True)

def handle_ip_scan(addr_type, ip):
    try:
        resolved = socket.gethostbyname(ip) if addr_type == 'url' else socket.getfqdn(ip)
    except Exception:
        return render_template('scanner.html', error=True)
    return render_template('scanner.html', addr=addr_type, loc=ip, res=resolved)

def handle_name_resolution(ip):
    try:
        resolved_name = socket.getfqdn(ip)
    except Exception:
        return render_template('scanner.html', error=True)
    return render_template('scanner.html', addr='url', loc=ip, res=resolved_name)

def handle_port_scan(ip):
    open_ports = []
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as sock:
        for port in COMMON_PORTS:
            if sock.connect_ex((ip, port)) == 0:
                open_ports.append(port)
            time.sleep(0.001)
    
    if not open_ports:
        open_ports = [0]
    
    return render_template('scanner.html', hosts=open_ports)

def handle_service_scan(ip):
    services = {}
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as sock:
        for port in COMMON_PORTS:
            try:
                result = sock.connect_ex((ip, port))
                service_name = socket.getservbyport(port, 'tcp') if result == 0 else None
                status = "running" if result == 0 else "not running"
                services[port] = [service_name, status]
            except Exception as e:
                return f"An error Occurred:<hr>{e}"
    
    return render_template('scanner.html', services=services)

if __name__ == "__main__":
    app.run(debug=True)
