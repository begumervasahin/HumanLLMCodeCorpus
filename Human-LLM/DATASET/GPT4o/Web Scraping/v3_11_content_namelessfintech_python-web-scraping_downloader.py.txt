from random import choice
import requests
from throttle import Throttle

class Downloader:
    """
    A class to handle downloading pages with optional caching and throttling.

    Attributes:
        delay (int): Delay between requests to the same domain.
        user_agent (str): User agent string for HTTP requests.
        proxies (list): List of proxy dictionaries.
        cache (dict): Dictionary for caching downloaded pages.
        timeout (float): Request timeout duration.
    """

    def __init__(self, delay=5, user_agent='wswp', proxies=None, cache=None, timeout=60):
        self.throttle = Throttle(delay)
        self.user_agent = user_agent
        self.proxies = proxies if proxies else []
        self.cache = cache if cache is not None else {}
        self.timeout = timeout

    def __call__(self, url, num_retries=2):
        """
        Return the HTML content of a page, using cache if available.

        Args:
            url (str): The URL of the page to download.
            num_retries (int): Number of retries for failed downloads.

        Returns:
            str: The HTML content of the page.
        """
        self.num_retries = num_retries
        result = self.cache.get(url)

        if result:
            print(f'Loaded from cache: {url}')
            if 500 <= result['code'] < 600 and self.num_retries:
                result = None

        if not result:
            self.throttle.wait(url)
            headers = {'User-Agent': self.user_agent}
            proxies = choice(self.proxies) if self.proxies else None
            result = self._download(url, headers, proxies)
            self.cache[url] = result

        return result.get('html')
