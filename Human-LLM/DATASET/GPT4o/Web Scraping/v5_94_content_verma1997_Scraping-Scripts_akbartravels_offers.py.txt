def get_random_header():
    """Return a random user-agent header."""
    return {'User-Agent': choice(HEADERS)}

def fetch_offers(url):
    """Fetch the offers from the given URL and return parsed HTML."""
    response = requests.get(url, headers=get_random_header())
    response.raise_for_status()  # Ensure the request was successful
    return BeautifulSoup(response.text, 'html.parser')

def parse_offer(container):
    """Parse individual offer details from a container."""
    image = container.find("img")['src']
    title = container.find("div", {"class": "deals-cntent-name"}).text.strip()
    description = container.find("div", {"class": "deals-cntent-disc"}).text.strip()
    validity = container.find("div", {"class": "deals-box-ftr-left"}).text.strip()

    return {
        'name': title,
        'image': image,
        'description': description,
        'validity': validity
    }

def parse_offers(soup):
    """Parse the offer details from the BeautifulSoup object."""
    containers = soup.find_all("div", {"class": ["deals-cntent-left", "deals-cntent-center", "deals-cntent-right"]})
    offers = [parse_offer(container) for container in containers]
    return offers

def save_offers_to_csv(offers, filename):
    """Save the offer details to a CSV file."""
    with open(filename, 'w', newline='') as file:
        headers = "Offer Name,Image Link,Description,Validity\n"
        file.write(headers)
        for offer in offers:
            line = f"{offer['name']},{offer['image']},{offer['description']},{offer['validity']}\n"
            file.write(line)

def main():
    url = "https://www.akbartravels.com/offers"
    soup = fetch_offers(url)
    offers = parse_offers(soup)
    save_offers_to_csv(offers, "akbartravels_offers.csv")

if __name__ == "__main__":
    main()
