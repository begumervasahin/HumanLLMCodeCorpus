# Function to select a random user agent header
def get_random_header():
    return {'User-Agent': choice(HEADERS_LIST)}

# Function to fetch and parse HTML content from a given URL
def fetch_html(url):
    response = requests.get(url, headers=get_random_header())
    response.raise_for_status()  # Ensure we notice bad responses
    return BeautifulSoup(response.text, 'html.parser')

# Function to scrape offers from the Indigo campaigns page
def scrape_indigo_offers(soup):
    containers = soup.findAll("div", {"class": "commonBottomAdon"})
    
    offers = []
    for container in containers:
        offer = {
            "name": container.find("h6").text.replace(",", ""),
            "link": container.find("a")['href'],
            "description": container.find("p").text.replace(",", ""),
            "image_link": "https://www.goindigo.in" + container.find("img")['src']
        }
        offers.append(offer)
    
    return offers

# Function to save offers to a CSV file
def save_offers_to_csv(offers, filename):
    with open(filename, "a", encoding="utf-8") as file:
        file.write("Offer Name,Link,Description,Image Link\n")
        for offer in offers:
            file.write(f"{offer['name']},{offer['link']},{offer['description']},{offer['image_link']}\n")

# Main function to run the scraping process
def main():
    url = "https://www.goindigo.in/campaigns/indigo-offers.html"
    soup = fetch_html(url)
    offers = scrape_indigo_offers(soup)
    save_offers_to_csv(offers, "indigo_offers.csv")

# Run the main function
if __name__ == "__main__":
    main()
