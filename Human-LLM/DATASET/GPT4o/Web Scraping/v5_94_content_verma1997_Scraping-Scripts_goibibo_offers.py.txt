def get_random_header():
    """Return a random user-agent header."""
    return {'User-Agent': choice(USER_AGENTS)}

def fetch_page_content(url):
    """Fetch the HTML content of a web page."""
    response = requests.get(url, headers=get_random_header())
    return BeautifulSoup(response.text, "html.parser")

def extract_offers(page_soup):
    """Extract offer details from the BeautifulSoup object."""
    offer_containers = page_soup.findAll("div", {"class": "bot-section"})
    promo_containers = page_soup.findAll("div", {"class": "back-content"})

    offers = [offer.p.text.replace(",", "") for offer in offer_containers]
    links = ["https://www.goibibo.com" + content.a['href'] for content in promo_containers]
    promos, terms_conditions = [], []

    for content in promo_containers:
        promo_content = content.find_all("div", {"class": "back-content-top-inner"})
        conditions = content.find_all("div", {"class": "tnc-section"})
        
        for term in conditions:
            terms_conditions.append(term.li.text.replace(",", ""))
        
        for promo in promo_content:
            promo_code = promo.find_all("div", {"class": "promocode"})
            for promo_text in promo_code:
                promos.append(promo_text.text)
    
    return offers, links, promos, terms_conditions

def write_to_csv(filename, data):
    """Write the extracted data to a CSV file."""
    headers = ["Offer Name", "Link", "Promocode", "Description", "Booking Channel"]
    
    with open(filename, "a", newline='') as file:
        writer = csv.writer(file)
        writer.writerow(headers)
        for row in data:
            writer.writerow(row)

def scrape_goibibo_offers():
    """Main function to scrape offers from Goibibo and write to CSV."""
    url = "https://www.goibibo.com/offers/"
    page_soup = fetch_page_content(url)
    offers, links, promos, terms_conditions = extract_offers(page_soup)

    data = [
        [offers[i], links[0], promos[i], terms_conditions[i], "Booking Channel"]
        for i in range(len(offers))
    ]
    
    write_to_csv("goibibo_offers.csv", data)

def job():
    """Job to run the scraping process."""
    scrape_goibibo_offers()
