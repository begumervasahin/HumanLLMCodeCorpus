import time
import logging
from nltk.corpus import stopwords
from iterators.sqlite_iterator import SQLiteDataIterator
from tokenizers.ru_tokenizer import RussianTokenizer
from vectorizer import HashingTfIdfVectorizer

# Configure logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# Constants
DATA_PATH = '/media/olga/Data/projects/DeepPavlov/download/odqa/enwiki_test.db'
SAVE_PATH = '/media/olga/Data/projects/DeepPavlov/download/odqa/enwiki_test_1.npz'
BATCH_SIZE = 1000

# Load Russian stopwords
STOPWORDS = stopwords.words('russian')

# Initialize iterator and vectorizer
iterator = SQLiteDataIterator(DATA_PATH, batch_size=BATCH_SIZE)
tokenizer = RussianTokenizer(ngram_range=[1, 2], lemmas=True, stopwords=STOPWORDS)
vectorizer = HashingTfIdfVectorizer(iterator, tokenizer=tokenizer)

# Record start time
start_time = time.time()

# Fit and save the vectorizer
try:
    vectorizer.fit()
    vectorizer.save(SAVE_PATH)
    logger.info(f"Vectorizer fitted and saved successfully. Time taken: {time.time() - start_time:.2f} seconds")
except Exception as e:
    logger.exception("An error occurred during vectorization")
    raise
