import os
import matplotlib.pyplot as plt
from scipy.optimize import curve_fit

def func(x, a, b, c, d):
    return a * x**3 + b * x**2 + c * x + d

def get_data():
    x = []
    y = []
    cwd = os.getcwd()
    folder = input('Folder name:\n')
    pth = os.path.join(cwd, folder)
    
    for filename in os.listdir(pth):
        if filename == '.DS_Store':
            continue
        print(filename)
        with open(os.path.join(pth, filename), 'r') as file:
            for line in file:
                data = line.split()
                y.append(float(data[0]))
                x.append(float(data[1]))

    params, _ = curve_fit(func, x, y)
    a, b, c, d = params
    
    x_final = []
    y_final = []
    i = 0
    while i <= 1:
        x_final.append(i)
        y_final.append(func(i, a, b, c, d))
        i += 0.2
    
    return x_final, y_final

# Get data for both models
x_vsm, y_vsm = get_data()
x_fuzzy, y_fuzzy = get_data()

# Plotting
plt.xlabel('Recall')
plt.ylabel('Precision')
plt.title('Precision vs Recall')

plt.plot(x_vsm, y_vsm, label='Vector Space Model')
plt.plot(x_vsm, y_vsm, 'bs')

plt.plot(x_fuzzy, y_fuzzy, label='Fuzzy Retrieval Model')
plt.plot(x_fuzzy, y_fuzzy, 'r^')

plt.legend()
plt.axis([0.0, 1.0, 0.0, 1.0])
plt.show()
