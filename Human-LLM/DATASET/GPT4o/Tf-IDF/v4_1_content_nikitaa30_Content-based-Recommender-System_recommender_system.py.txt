import pandas as pd
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import linear_kernel

# Load dataset
ds = pd.read_csv("/home/nikita/Downloads/sample-data.csv")

# Create TF-IDF matrix
tf = TfidfVectorizer(analyzer='word', ngram_range=(1, 3), min_df=0, stop_words='english')
tfidf_matrix = tf.fit_transform(ds['description'])

# Compute cosine similarities
cosine_similarities = linear_kernel(tfidf_matrix, tfidf_matrix)

# Generate similarity results
results = {}
for idx, row in ds.iterrows():
    similar_indices = cosine_similarities[idx].argsort()[:-100:-1]
    similar_items = [(cosine_similarities[idx][i], ds['id'][i]) for i in similar_indices]
    results[row['id']] = similar_items[1:]

print('done!')

# Function to get item description by ID
def item(id):
    return ds.loc[ds['id'] == id]['description'].tolist()[0].split(' - ')[0]

# Function to recommend similar items
def recommend(item_id, num):
    print(f"Recommending {num} products similar to {item(item_id)}...")
    print("-------")
    recs = results[item_id][:num]
    for rec in recs:
        print(f"Recommended: {item(rec[1])} (score: {rec[0]})")

# Example recommendation
recommend(item_id=11, num=5)
