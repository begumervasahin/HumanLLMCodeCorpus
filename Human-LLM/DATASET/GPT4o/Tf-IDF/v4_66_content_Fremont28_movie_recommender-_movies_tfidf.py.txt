import pandas as pd
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import linear_kernel

# Load the dataset
cinema = pd.read_csv("tmdb_5000_movies.csv", encoding="latin-1")

# Display the first 3 rows of the dataset
print(cinema.head(3))

# Print the total number of rows in the dataset
print(len(cinema))

# Group by 'original_title' and count the number of occurrences
movie_grouped = cinema.groupby(['original_title']).agg({'popularity': 'count'}).reset_index()

# Get the number of unique movie titles
movie_count = cinema['original_title'].unique()
print(len(movie_count))

# Get the number of unique original languages
lengua_count = cinema['original_language'].unique()
print(lengua_count)

# Initialize the TF-IDF Vectorizer
tfidf = TfidfVectorizer(stop_words='english')

# Fill NaN values in 'overview' with empty strings
cinema['overview'] = cinema['overview'].fillna('')

# Fit and transform the 'overview' text data
tfidf_matrix = tfidf.fit_transform(cinema['overview'])

# Print the shape of the TF-IDF matrix
print(tfidf_matrix.shape)

# Compute the cosine similarity matrix
cosine_sim = linear_kernel(tfidf_matrix, tfidf_matrix)

# Create a Series with movie titles as index
indices = pd.Series(cinema.index, index=cinema['original_title']).drop_duplicates()

# Function to get movie recommendations based on title
def get_recs(title, cosine_sim=cosine_sim):
    # Get the index of the movie that matches the title
    idx = indices[title]
    
    # Get the pairwise similarity scores of all movies with that movie
    sim_scores = list(enumerate(cosine_sim[idx]))
    
    # Sort the movies based on similarity scores
    sim_scores = sorted(sim_scores, key=lambda x: x[1], reverse=True)
    
    # Get the scores of the 19 most similar movies
    sim_scores = sim_scores[1:20]
    
    # Get the movie indices
    movie_indices = [i[0] for i in sim_scores]
    
    # Return the top 19 most similar movies
    return cinema['original_title'].iloc[movie_indices]

# Get recommendations for the movie "No Reservations"
print(get_recs("No Reservations"))
