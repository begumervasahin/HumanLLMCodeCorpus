import os
import matplotlib.pyplot as plt
from scipy.optimize import curve_fit

def func(x, a, b, c, d):
    """Cubic function for curve fitting."""
    return a * x**3 + b * x**2 + c * x + d

def read_data_from_files(folder):
    """Reads x and y data from text files in the specified folder."""
    x, y = [], []
    folder_path = os.path.join(os.getcwd(), folder)
    
    for filename in os.listdir(folder_path):
        if filename == '.DS_Store':
            continue
        
        print(f"Processing file: {filename}")
        file_path = os.path.join(folder_path, filename)
        
        with open(file_path, 'r') as file:
            for line in file:
                data = line.split()
                y.append(float(data[0]))
                x.append(float(data[1]))

    return x, y

def fit_curve(x, y):
    """Fits a cubic curve to the given x and y data."""
    params, _ = curve_fit(func, x, y)
    return params

def generate_fitted_data(params):
    """Generates data points using the fitted curve parameters."""
    a, b, c, d = params
    x_final = [i for i in range(11)]
    y_final = [func(i / 10, a, b, c, d) for i in x_final]
    x_final = [i / 10 for i in x_final]
    
    return x_final, y_final

def plot_data(x_vsm, y_vsm, x_fuzzy, y_fuzzy):
    """Plots the precision vs recall graph for both models."""
    plt.xlabel('Recall')
    plt.ylabel('Precision')
    plt.title('Precision vs Recall')

    plt.plot(x_vsm, y_vsm, label='Vector Space Model')
    plt.plot(x_vsm, y_vsm, 'bs')

    plt.plot(x_fuzzy, y_fuzzy, label='Fuzzy Retrieval Model')
    plt.plot(x_fuzzy, y_fuzzy, 'r^')

    plt.legend()
    plt.axis([0.0, 1.0, 0.0, 1.0])
    plt.show()

def main():
    folder = input('Folder name:\n')
    
    x_vsm, y_vsm = read_data_from_files(folder)
    params_vsm = fit_curve(x_vsm, y_vsm)
    x_vsm, y_vsm = generate_fitted_data(params_vsm)

    x_fuzzy, y_fuzzy = read_data_from_files(folder)
    params_fuzzy = fit_curve(x_fuzzy, y_fuzzy)
    x_fuzzy, y_fuzzy = generate_fitted_data(params_fuzzy)

    plot_data(x_vsm, y_vsm, x_fuzzy, y_fuzzy)

if __name__ == "__main__":
    main()
