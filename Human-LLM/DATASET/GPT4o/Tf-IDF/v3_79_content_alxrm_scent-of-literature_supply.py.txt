import pandas as pd
from format import clean_str, shuffle_both

def read_train_data(file_name, x_col, y_col, separator='\t', with_cleaning=True, with_header=False, with_shuffle=True):
    """
    Reads and processes training data from a CSV file.
    
    Args:
        file_name (str): Path to the CSV file.
        x_col (int): Column index for the features.
        y_col (int): Column index for the labels.
        separator (str): Separator used in the CSV file.
        with_cleaning (bool): Whether to clean the text data.
        with_header (bool): Whether the CSV file has a header.
        with_shuffle (bool): Whether to shuffle the data.
        
    Returns:
        tuple: Processed features and labels, possibly shuffled.
    """
    header_option = 0 if with_header else None
    data = pd.read_csv(file_name, sep=separator, header=header_option)
    
    x_text = process_text_lines(data[x_col].tolist(), with_cleaning)
    y_text = data[y_col].tolist()
    
    if with_shuffle:
        x_text, y_text = shuffle_both(x_text, y_text)
    
    return x_text, y_text

def read_test_data(file_name, x_col, with_cleaning=True):
    """
    Reads and processes test data from a CSV file.
    
    Args:
        file_name (str): Path to the CSV file.
        x_col (int): Column index for the features.
        with_cleaning (bool): Whether to clean the text data.
        
    Returns:
        list: Processed features.
    """
    data = pd.read_csv(file_name, header=None, skiprows=[0])
    x_text = data[x_col].tolist()
    
    return process_text_lines(x_text, with_cleaning)

def process_text_lines(lines, with_cleaning=True):
    """
    Processes text lines by cleaning them if required.
    
    Args:
        lines (list): List of text lines.
        with_cleaning (bool): Whether to clean the text lines.
        
    Returns:
        list: Processed text lines.
    """
    if with_cleaning:
        return [clean_str(line) for line in lines]
    else:
        return lines
