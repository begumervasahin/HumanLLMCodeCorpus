import re

class Document:
    def __init__(self):
        self.docid = 0
        self.lines = []

    def print_doc(self, out=sys.stdout):
        print(f"\n[DOCID: {self.docid}]", file=out)
        for line in self.lines:
            print(line, file=out)


class ReadDocuments:
    def __init__(self, file):
        self.collection_file = file

    def __iter__(self):
        startdoc = re.compile(r'<document docid\s*=\s*"(\d+)"\s*>')
        enddoc = re.compile(r'</document\s*>')
        reading_doc = False
        with open(self.collection_file) as input_fs:
            for line in input_fs:
                if not reading_doc:
                    m = startdoc.search(line)
                    if m:
                        reading_doc = True
                        doc = Document()
                        doc.docid = int(m.group(1))
                else:
                    if enddoc.search(line):
                        reading_doc = False
                        yield doc
                    else:
                        doc.lines.append(line.strip())

if __name__ == "__main__":
    file_path = 'path_to_your_file.txt'
    reader = ReadDocuments(file_path)
    for document in reader:
        document.print_doc()
