import os

def create_domain_directory(directory):
    """
    Creates a directory if it doesn't already exist.
    
    Args:
        directory (str): The path of the directory to create.
    """
    if not os.path.exists(directory):
        print(f'Creating directory {directory}')
        os.makedirs(directory)

def create_data_files(project_name, base_url):
    """
    Creates 'queue.txt' and 'crawled.txt' files for a project.
    
    Args:
        project_name (str): The name of the project directory.
        base_url (str): The initial URL to write to 'queue.txt'.
    """
    queue_path = os.path.join(project_name, 'queue.txt')
    crawled_path = os.path.join(project_name, 'crawled.txt')
    
    if not os.path.isfile(queue_path):
        write_file(queue_path, base_url)
    
    if not os.path.isfile(crawled_path):
        write_file(crawled_path, '')

def write_file(path, data):
    """
    Writes data to a file.
    
    Args:
        path (str): The file path.
        data (str): The data to write.
    """
    with open(path, 'w') as file:
        file.write(data)

def append_to_file(path, data):
    """
    Appends data to a file.
    
    Args:
        path (str): The file path.
        data (str): The data to append.
    """
    with open(path, 'a') as file:
        file.write(data + '\n')

def delete_file_contents(path):
    """
    Deletes the contents of a file.
    
    Args:
        path (str): The file path.
    """
    with open(path, 'w'):
        pass

def get_set_from_file(file_name):
    """
    Reads a file and converts each line into a set item.
    
    Args:
        file_name (str): The file path.
    
    Returns:
        set: A set of lines from the file.
    """
    results = set()
    
    with open(file_name, 'rt') as file:
        for line in file:
            results.add(line.strip())
    
    return results

def write_set_to_file(links, file_name):
    """
    Writes a set of links to a file.
    
    Args:
        links (set): A set of links.
        file_name (str): The file path.
    """
    with open(file_name, 'w') as file:
        for link in sorted(links):
            file.write(link + '\n')

    create_domain_directory(project_name)
    create_data_files(project_name, base_url)
