def save_file(save_path, content):
    """
    Save content to a file.
    
    Args:
        save_path (str): The path to save the file.
        content (bytes): The content to write to the file.
    """
    with open(save_path, "wb") as fp:
        fp.write(content)

def read_file(path):
    """
    Read content from a file.
    
    Args:
        path (str): The path of the file to read.
        
    Returns:
        bytes: The content of the file.
    """
    with open(path, "rb") as fp:
        return fp.read()

def create_bunch(seg_path, wordbag_path):
    """
    Create a Bunch object from the segmented corpus and save it to a file.
    
    Args:
        seg_path (str): The path of the segmented corpus directory.
        wordbag_path (str): The path to save the Bunch object.
    """
    # Initialize Bunch object
    bunch = Bunch(target_name=[], label=[], filenames=[], contents=[])

    # Get list of categories (directories)
    category_list = os.listdir(seg_path)
    bunch.target_name.extend(category_list)

    # Process files for each category
    for category in category_list:
        class_path = os.path.join(seg_path, category)
        file_list = os.listdir(class_path)
        
        for file_name in file_list:
            full_path = os.path.join(class_path, file_name)
            bunch.label.append(category)
            bunch.filenames.append(full_path)
            bunch.contents.append(read_file(full_path).strip())

    # Save Bunch object to file
    with open(wordbag_path, "wb") as file_obj:
        pickle.dump(bunch, file_obj)

    print("Text object construction completed!!!")

if __name__ == "__main__":
