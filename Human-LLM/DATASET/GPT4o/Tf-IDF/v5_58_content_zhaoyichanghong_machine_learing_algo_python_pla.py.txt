import numpy as np
import metrics

class Perceptron:
    def __init__(self):
        self.weights = None
        self.bias = 0

    def fit(self, X, y):
        """
        Fit the Perceptron model to the training data.

        Parameters
        ----------
        X : ndarray of shape (n_samples, n_features)
            Training data.
        y : ndarray of shape (n_samples,)
            Target values, either 1 or -1.
        """
        n_samples, n_features = X.shape
        self.weights = np.zeros(n_features)
        self.bias = 0
        
        while True:
            errors = False
            for i in range(n_samples):
                prediction = self._predict_single_sample(X[i])
                if y[i] * prediction <= 0:
                    self.weights += y[i] * X[i]
                    self.bias += y[i]
                    errors = True
            
            if not errors:
                break

    def predict(self, X):
        """
        Predict class labels for samples in X.

        Parameters
        ----------
        X : ndarray of shape (n_samples, n_features)
            Data to predict.
        
        Returns
        -------
        y_pred : ndarray of shape (n_samples,)
            Predicted class labels for each sample, either 1 or -1.
        """
        return np.sign(X.dot(self.weights) + self.bias)
    
    def _predict_single_sample(self, x):
        """
        Predict the class label for a single sample.

        Parameters
        ----------
        x : ndarray of shape (n_features,)
            Single sample data.
        
        Returns
        -------
        y_pred : int
            Predicted class label for the sample, either 1 or -1.
        """
        return np.sign(np.dot(x, self.weights) + self.bias)
