from flask import Flask, render_template, request
from forms import SearchForm
import math
from tfidf import search as Searching

# Initialize the Flask application
app = Flask(__name__)
app.secret_key = 'development'  # Set the secret key for session management

# Define the index route to display the search form
@app.route("/")
def index():
    form = SearchForm()  # Create an instance of the search form
    return render_template("index.html", form=form)  # Render the index page with the form

# Define the search route to handle search requests and display results
@app.route("/search")
def search():
    form = SearchForm()  # Create an instance of the search form
    query = request.args.get('keyword')  # Retrieve the search query from request arguments
    page = int(request.args.get('page', 1))  # Get the current page number, defaulting to 1 if not provided

    # Perform the search using the TF-IDF module
    data = Searching.search_for(query)
    data_length = len(data)  # Get the total number of search results

    # Define pagination parameters
    pagination_size = 5
    start = (page - 1) * pagination_size  # Calculate the start index for the current page
    end = min(page * pagination_size, data_length)  # Calculate the end index for the current page
    total = math.ceil(data_length / pagination_size)  # Calculate the total number of pages

    # Render the search results page with the relevant data
    return render_template(
        "search.html", 
        form=form, 
        data=data[start:end], 
        keyword=query, 
        page=page, 
        total=total
    )

# Run the Flask application
if __name__ == '__main__':
    app.run(debug=True)  # Run the app in debug mode
