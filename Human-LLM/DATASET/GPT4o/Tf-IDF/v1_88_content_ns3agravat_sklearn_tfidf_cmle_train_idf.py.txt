# Import necessary libraries
from sklearn.datasets import fetch_20newsgroups
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.naive_bayes import MultinomialNB
import pickle

# Fetch the training data from the 20 newsgroups dataset
twenty_train = fetch_20newsgroups(subset='train', shuffle=True)

# Initialize the TfidfVectorizer with the desired parameters
word_vectorizer = TfidfVectorizer(
    stop_words='english',
    sublinear_tf=True,
    strip_accents='unicode',
    analyzer='word',
    token_pattern=r'\w{2,}',
    ngram_range=(1, 1),
    max_features=30000
)

# Fit the vectorizer on the training data and transform the training data into feature vectors
word_vectorizer.fit(twenty_train.data)
X_train_word_features = word_vectorizer.transform(twenty_train.data)

# Initialize and train the Multinomial Naive Bayes classifier
text_clf = MultinomialNB().fit(X_train_word_features, twenty_train.target)

# Save the trained model and the vectorizer to disk using pickle
with open('text_clf.pkl', 'wb') as model_file:
    pickle.dump(text_clf, model_file)

with open('vectorizer.pkl', 'wb') as vectorizer_file:
    pickle.dump(word_vectorizer, vectorizer_file)

print("Model and vectorizer have been saved to disk.")
