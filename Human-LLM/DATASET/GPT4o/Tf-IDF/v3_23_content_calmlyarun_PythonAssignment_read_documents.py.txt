import re
import sys

class Document:
    def __init__(self):
        self.docid = 0
        self.lines = []

    def print_doc(self, out=sys.stdout):
        """
        Print the document ID and its lines to the specified output.
        """
        print(f"\n[DOCID: {self.docid}]", file=out)
        for line in self.lines:
            print(line, file=out)

class ReadDocuments:
    def __init__(self, file_path):
        self.collection_file = file_path

    def __iter__(self):
        """
        Iterate over the lines in the collection file to extract and yield Document objects.
        """
        startdoc_pattern = re.compile(r'<document docid\s*=\s*"(\d+)"\s*>')
        enddoc_pattern = re.compile(r'</document\s*>')
        reading_doc = False

        with open(self.collection_file, 'r') as file:
            for line in file:
                if not reading_doc:
                    # Check for the start of a new document
                    start_match = startdoc_pattern.search(line)
                    if start_match:
                        reading_doc = True
                        doc = Document()
                        doc.docid = int(start_match.group(1))
                else:
                    # Check for the end of the current document
                    if enddoc_pattern.search(line):
                        reading_doc = False
                        yield doc
                    else:
                        # Add the line to the current document
                        doc.lines.append(line.strip())

# Example usage
if __name__ == "__main__":
    file_path = 'path_to_your_file.txt'
    reader = ReadDocuments(file_path)
    for document in reader:
        document.print_doc()
