import pandas as pd
import pickle
import numpy as np
from sklearn.feature_extraction.text import CountVectorizer, TfidfTransformer
from sklearn.pipeline import Pipeline
from sklearn import svm

# Set a random seed for reproducibility
np.random.seed(1000)

def load_classifier(filename):
    """
    Load a pre-trained classifier from a file.
    
    Parameters:
    filename (str): The name of the file containing the pre-trained classifier.
    
    Returns:
    classifier: The loaded classifier.
    """
    with open(filename, 'rb') as file:
        classifier = pickle.load(file)
    return classifier

def classify_text(classifier, input_text):
    """
    Classify the given text using the provided classifier.
    
    Parameters:
    classifier: The pre-trained classifier.
    input_text (str): The text to be classified.
    
    Returns:
    str: The predicted label for the input text.
    """
    predicted_label = classifier.predict([input_text])
    return predicted_label[0]

def main():
    """
    Main function to load the classifier and classify a sample text.
    """
    filename = 'Amz_5000_.sav'
    input_text = "Sample text for classification."
    
    classifier = load_classifier(filename)
    print("Classifier loaded. Classifying the input text...")
    
    predicted_label = classify_text(classifier, input_text)
    
    print(f"Input text: {input_text}")
    print(f"Predicted label: {predicted_label}")

if __name__ == "__main__":
    main()
