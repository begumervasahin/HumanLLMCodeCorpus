import pickle
from sklearn.datasets import fetch_20newsgroups
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.naive_bayes import MultinomialNB

# Fetch the training data
train_data = fetch_20newsgroups(subset='train', shuffle=True)

# Initialize the TF-IDF vectorizer with specific parameters
tfidf_vectorizer = TfidfVectorizer(
    stop_words='english',
    sublinear_tf=True,
    strip_accents='unicode',
    analyzer='word',
    token_pattern=r'\w{2,}',
    ngram_range=(1, 1),
    max_features=30000
)

# Fit the vectorizer on the training data and transform the data into TF-IDF features
X_train_tfidf = tfidf_vectorizer.fit_transform(train_data.data)

# Train a Multinomial Naive Bayes classifier on the TF-IDF features
text_classifier = MultinomialNB().fit(X_train_tfidf, train_data.target)

# Save the trained classifier and vectorizer to disk using pickle
with open('text_classifier.pkl', 'wb') as clf_file:
    pickle.dump(text_classifier, clf_file)

with open('tfidf_vectorizer.pkl', 'wb') as vec_file:
    pickle.dump(tfidf_vectorizer, vec_file)
