import numpy as np
from sklearn.cluster import KMeans

class BisectingKMeans:
    def fit(self, X, n_clusters):
        """
        Perform bisecting k-means clustering.

        Parameters
        ----------
        X : ndarray of shape (n_samples, n_features)
            The input data.
        n_clusters : int
            The number of clusters to form.

        Returns
        -------
        y : ndarray of shape (n_samples,)
            Predicted cluster label for each sample.
        """
        n_samples = X.shape[0]
        clusters = [np.arange(n_samples)]  # Start with all samples in one cluster

        while len(clusters) < n_clusters:
            # Compute the sum of squared errors for each cluster
            sse = [self._calculate_sse(X, cluster) for cluster in clusters]
            
            # Select the cluster with the maximum SSE to split
            index_to_split = np.argmax(sse)
            data_to_split = X[clusters[index_to_split]]

            # Apply KMeans to split the selected cluster into two
            new_clusters = self._split_cluster(data_to_split, clusters[index_to_split])

            # Replace the original cluster with the two new clusters
            clusters[index_to_split] = new_clusters[0]
            clusters.append(new_clusters[1])

        # Assign cluster labels to all samples
        y = self._assign_labels(n_samples, clusters)

        return y

    def _calculate_sse(self, X, cluster):
        """
        Calculate the sum of squared errors for a given cluster.

        Parameters
        ----------
        X : ndarray of shape (n_samples, n_features)
            The input data.
        cluster : ndarray of shape (n_cluster_samples,)
            Indices of samples in the cluster.

        Returns
        -------
        sse : float
            Sum of squared errors for the cluster.
        """
        return np.var(X[cluster])

    def _split_cluster(self, data_to_split, indices_to_split):
        """
        Split a cluster into two using KMeans.

        Parameters
        ----------
        data_to_split : ndarray of shape (n_cluster_samples, n_features)
            The data points to be split.
        indices_to_split : ndarray of shape (n_cluster_samples,)
            The indices of the data points to be split.

        Returns
        -------
        new_clusters : list of ndarrays
            Two new clusters created from splitting the original cluster.
        """
        kmeans = KMeans(n_clusters=2, random_state=0)
        labels = kmeans.fit_predict(data_to_split)

        new_clusters = [indices_to_split[labels == i] for i in range(2)]
        return new_clusters

    def _assign_labels(self, n_samples, clusters):
        """
        Assign cluster labels to all samples.

        Parameters
        ----------
        n_samples : int
            Total number of samples.
        clusters : list of ndarrays
            List of clusters with sample indices.

        Returns
        -------
        y : ndarray of shape (n_samples,)
            Cluster label for each sample.
        """
        y = np.zeros(n_samples, dtype=int)
        for i, cluster in enumerate(clusters):
            y[cluster] = i
        return y

# Example usage:
if __name__ == "__main__":
    # Generate random data for demonstration
    np.random.seed(42)
    X = np.random.rand(100, 2)

    bkm = BisectingKMeans()
    n_clusters = 3
    labels = bkm.fit(X, n_clusters)

    print("Cluster labels:", labels)
