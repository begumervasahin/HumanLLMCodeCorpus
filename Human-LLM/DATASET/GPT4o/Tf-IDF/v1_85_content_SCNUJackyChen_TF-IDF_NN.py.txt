import gensim
import numpy as np
import pickle
from sklearn.neighbors import BallTree
import TFIDF  # Assuming TFIDF is a custom module with the necessary dictionary
from pprint import pprint

# Load the word2vec model
model = gensim.models.KeyedVectors.load_word2vec_format('embedding.txt', binary=False)
vocab = model.wv.vocab

# Create the embeddings matrix and list of words
length = len(vocab)
embedding_dim = model.vector_size  # Typically 300, but better to use the model's attribute
X = np.ndarray(shape=(length, embedding_dim))
words = list(vocab.keys())

# Fill the embeddings matrix
for i, word in enumerate(words):
    X[i] = model[word]

# Build the BallTree
tree = BallTree(X)
serialized_tree = pickle.dumps(tree)

def word_rewrite(word: str) -> str:
    """
    Find the closest word in the TFIDF dictionary to the given word using the BallTree.

    Parameters:
    word (str): The input word to be rewritten.

    Returns:
    str: The closest word in the TFIDF dictionary.
    """
    if word in TFIDF.dictionary:
        return word

    tree_copy = pickle.loads(serialized_tree)
    q_embedding = model[word].reshape(1, -1)
    dist, ind = tree_copy.query(q_embedding, k=100)
    candidates = [words[idx] for idx in ind[0]]

    for candidate in candidates:
        if candidate in TFIDF.dictionary:
            return candidate
    return None

if __name__ == '__main__':
    rewritten_word = word_rewrite('fib')
    print(rewritten_word)
