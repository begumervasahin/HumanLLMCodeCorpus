import pandas as pd
import pickle
import numpy as np
from sklearn.feature_extraction.text import CountVectorizer
from sklearn.feature_extraction.text import TfidfTransformer
from sklearn.pipeline import Pipeline
from sklearn import svm

# Set a seed for reproducibility
np.random.seed(1000)

def classify_text(text):
    """
    Classify the input text using a pre-trained SVM classifier.
    
    Parameters:
    text (str): The text to classify.
    
    Returns:
    array: The predicted class for the input text.
    """
    # Prepare the test data
    test_data = [text]
    print("Test data:", test_data, '\n')
    
    # Load the pre-trained classifier
    model_filename = 'Amz_5000_.sav'
    with open(model_filename, 'rb') as model_file:
        classifier = pickle.load(model_file)
    
    print("Classifier loaded! \nClassifying test set...")
    
    # Predict the class of the test data
    prediction = classifier.predict(test_data)
    
    print("Prediction:", prediction, '\n')
    
    return prediction

if __name__ == "__main__":
    # Prompt the user to enter text for classification
    input_text = input("Enter the text to classify: ")
    
    # Classify the input text
    result = classify_text(input_text)
    
    # Display the classification result
    print("Result:", result)
