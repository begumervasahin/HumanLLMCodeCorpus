import sys
import os
import logging
from os.path import basename

# Configure logging
logging.basicConfig(format='%(levelname)s : %(asctime)s : %(message)s', level=logging.INFO)
x_separator = '\nXXXXXXX\n'

def usage():
    """
    Print the usage information for the script.
    """
    print('Usage:', sys.argv[0], '<corpus file> <number of parts> <output path>')

def main(argv):
    # Check for the correct number of arguments
    if len(argv) < 4:
        usage()
        sys.exit(2)

    # Import the text processing module dynamically
    import imp
    tp = imp.load_source('textpro', 'textpro.py')

    # Parse command line arguments
    corpus_file = argv[1]
    num_parts = int(argv[2])
    output_path = argv[3]

    # Ensure the output path ends with a '/'
    if not output_path.endswith('/'):
        output_path += '/'

    # Ensure the output directory exists
    tp.check_dir(output_path)

    # Load the corpus into an array of documents
    docs = tp.split_wikipedia_docs_into_array(corpus_file)
    logging.info('Corpus is loaded')

    # Split the documents into the specified number of parts
    parts = tp.split_list(docs, num_parts)
    f_name = basename(corpus_file).split()[0]

    # Write each part to a separate file
    for i, part in enumerate(parts):
        output_file = os.path.join(output_path, f'part-{i}-{f_name}')
        with open(output_file, 'w', encoding='utf-8') as out:
            for doc in part:
                out.write(doc + '\n')
        logging.info('Part %d is done', i)

if __name__ == "__main__":
    main(sys.argv)
