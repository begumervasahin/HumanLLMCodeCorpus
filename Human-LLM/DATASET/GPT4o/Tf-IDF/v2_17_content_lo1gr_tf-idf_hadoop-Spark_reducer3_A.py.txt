import sys
from math import log10

def process_input():
    """
    Process input from stdin and return a list of word information tuples and a dictionary
    with word counts across documents.
    """
    prev_word = None
    count = 1
    word_dict = {}
    temp_list = []
    D = 10.0  # Total number of documents

    for line in sys.stdin:
        line = line.strip()
        word, details = line.split('\t', 1)
        filename, n, N, c = details.split(' ', 3)
        n = float(n)
        N = float(N)
        c = int(c)

        if prev_word == word:
            count += c
        else:
            if prev_word is not None:
                word_dict[prev_word] = f'{n} {N} {count}'
                temp_list.append(f'{prev_word} {filename}')
            count = c
            prev_word = word

    # Handle the last word
    if prev_word is not None:
        word_dict[prev_word] = f'{n} {N} {count}'
        temp_list.append(f'{prev_word} {filename}')

    return temp_list, word_dict, D

def calculate_tfidf(temp_list, word_dict, D):
    """
    Calculate and print the TF-IDF for each word in the temporary list.
    """
    for item in temp_list:
        word, filename = item.split(' ', 1)
        if word in word_dict:
            n, N, m = word_dict[word].split(' ', 2)
            n = float(n)
            N = float(N)
            m = float(m)
            tfidf = (n / N) * log10(D / m)
            print(f'{item}\t{tfidf}')

def main():
    """
    Main function to orchestrate the processing and calculation of TF-IDF.
    """
    temp_list, word_dict, D = process_input()
    calculate_tfidf(temp_list, word_dict, D)

if __name__ == "__main__":
    main()
