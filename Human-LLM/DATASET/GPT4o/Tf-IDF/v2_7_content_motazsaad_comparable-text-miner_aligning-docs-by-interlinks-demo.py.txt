import sys
import importlib.util

def usage():
    """
    Prints the usage instructions for the script.
    """
    print(f'Usage: {sys.argv[0]} <source corpus file> <target corpus file> <source language> <target language> <output path>')

def load_textpro_module(module_path):
    """
    Dynamically loads the textpro module from the given path.
    
    Args:
    - module_path (str): The file path to the textpro module.
    
    Returns:
    - module: The loaded textpro module.
    """
    spec = importlib.util.spec_from_file_location("textpro", module_path)
    textpro = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(textpro)
    return textpro

def main(argv):
    """
    Main function to process the command-line arguments and run the alignment.
    
    Args:
    - argv (list): List of command-line arguments.
    """
    if len(argv) < 6:
        usage()
        sys.exit(2)

    source_corpus_file = argv[1]
    target_corpus_file = argv[2]
    source_language = argv[3]
    target_language = argv[4]
    output_path = argv[5]

    textpro = load_textpro_module('textpro.py')
    textpro.aligning_documents_by_interlanguage_links(
        source_corpus_file, 
        target_corpus_file, 
        source_language, 
        target_language, 
        output_path
    )

if __name__ == "__main__":
    """
    Entry point for the script. Executes the main function with command-line arguments.
    """
    main(sys.argv)
