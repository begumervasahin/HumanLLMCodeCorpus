import numpy as np
import k_means

class BisectingKMeans:
    def fit(self, X, n_clusters):
        """
        Fit the Bisecting K-Means model to the data.
        
        Parameters
        ----------
        X : array-like of shape (n_samples, n_features)
            Training data.
        
        n_clusters : int
            The number of clusters to form.
        
        Returns
        -------
        y : ndarray of shape (n_samples,)
            Predicted cluster label for each sample.
        """
        n_samples = X.shape[0]
        initial_cluster = np.arange(n_samples)
        clusters = [initial_cluster]
        
        while len(clusters) < n_clusters:
            # Select the cluster with the highest SSE for splitting
            sse = [self._calculate_sse(X[cluster]) for cluster in clusters]
            idx_to_split = np.argmax(sse)
            
            # Perform k-means on the selected cluster
            data_to_split = X[clusters[idx_to_split]]
            labels = self._bisect_cluster(data_to_split)
            
            # Create new clusters
            new_clusters = [clusters[idx_to_split][labels == i] for i in range(2)]
            clusters.pop(idx_to_split)
            clusters.extend(new_clusters)
        
        # Assign cluster labels to each sample
        y = self._assign_labels(clusters, n_samples)
        
        return y

    def _calculate_sse(self, cluster_data):
        """
        Calculate the Sum of Squared Errors (SSE) for a cluster.
        
        Parameters
        ----------
        cluster_data : array-like of shape (n_samples_in_cluster, n_features)
            Data points in the cluster.
        
        Returns
        -------
        sse : float
            Sum of Squared Errors for the cluster.
        """
        return np.var(cluster_data) * cluster_data.shape[0]

    def _bisect_cluster(self, data):
        """
        Perform k-means on a given cluster to bisect it into two clusters.
        
        Parameters
        ----------
        data : array-like of shape (n_samples_in_cluster, n_features)
            Data points in the cluster to be bisected.
        
        Returns
        -------
        labels : ndarray of shape (n_samples_in_cluster,)
            Labels indicating the cluster assignment (0 or 1).
        """
        model = k_means.KMeans()
        labels = model.fit(data, n_clusters=2, max_iter=100)
        return labels

    def _assign_labels(self, clusters, n_samples):
        """
        Assign cluster labels to each sample based on the clusters.
        
        Parameters
        ----------
        clusters : list of arrays
            List of arrays, each containing indices of samples in a cluster.
        
        n_samples : int
            Total number of samples.
        
        Returns
        -------
        y : ndarray of shape (n_samples,)
            Predicted cluster label for each sample.
        """
        y = np.zeros(n_samples, dtype=int)
        for cluster_idx, cluster in enumerate(clusters):
            y[cluster] = cluster_idx
        return y
