import numpy as np
import scipy.special
import pca

class ICA:
    def __init__(self):
        self.__W = None

    def fit_transform(self, X, epochs, optimizer):
        """
        Perform Independent Component Analysis (ICA) on the given data.

        Parameters
        ----------
        X : ndarray of shape (n_samples, n_features)
            Training data.
        epochs : int
            Number of epochs to run the optimization.
        optimizer : object
            Optimizer instance with a defined 'optimize' method.

        Returns
        -------
        S : ndarray of shape (n_samples, n_features)
            Predicted sources for each sample.
        """
        n_samples, n_features = X.shape

        # Step 1: Whiten the data using PCA
        pca_model = pca.PCA(n_features, whiten=True)
        X_whitened = pca_model.fit_transform(X)

        # Step 2: Initialize the unmixing matrix W
        self.__W = np.random.rand(n_features, n_features)

        # Step 3: Optimize W using the given optimizer and epochs
        for epoch in range(epochs):
            g_W = np.zeros_like(self.__W)
            for x in X_whitened:
                # Compute the gradient for the current sample
                term1 = 1 - 2 * scipy.special.expit(self.__W.dot(x.T))
                term2 = np.linalg.inv(self.__W.T)
                g_W += term1[:, None] * x + term2

            # Average the gradient over all samples
            g_W /= n_samples

            # Update W using the optimizer
            self.__W += optimizer.optimize([g_W])[0]

        # Step 4: Transform the whitened data using the learned unmixing matrix
        S = X_whitened.dot(self.__W)

        return S
