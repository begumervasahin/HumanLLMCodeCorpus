import math
import string

def preprocess_document(document):
    """
    Converts the document to lowercase and removes punctuation.
    """
    document = document.lower()
    exclude = set(string.punctuation)
    return ''.join(ch for ch in document if ch not in exclude)

def calculate_tf(document, term):
    """
    Calculates the term frequency (TF) of a term in a document.
    
    TF = (Number of times term appears in document) / (Total number of terms in document)
    """
    doc_terms = document.split()
    term_count = doc_terms.count(term)
    return term_count / len(doc_terms)

def calculate_idf(corpus, term):
    """
    Calculates the inverse document frequency (IDF) of a term in a corpus.
    
    IDF = log(Total number of documents / Number of documents containing the term)
    """
    num_docs_containing_term = sum(1 for doc in corpus if term in preprocess_document(doc).split())
    if num_docs_containing_term == 0:
        return 0
    return math.log(len(corpus) / num_docs_containing_term)

def tf_idf(corpus, document, term):
    """
    Calculates the TF-IDF score of a term in a document within a corpus.
    
    TF-IDF = TF * IDF
    """
    preprocessed_document = preprocess_document(document)
    preprocessed_term = term.lower()
    tf = calculate_tf(preprocessed_document, preprocessed_term)
    idf = calculate_idf(corpus, preprocessed_term)
    return tf * idf

if __name__ == "__main__":
    corpus = [
        "This is a sample document.",
        "This document is another example document.",
        "And this is a different document."
    ]
    document = "This document is a sample document."
    term = "document"

    tfidf_score = tf_idf(corpus, document, term)
    print(f"The TF-IDF score for the term '{term}' is: {tfidf_score}")
