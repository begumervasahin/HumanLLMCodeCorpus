from random import randint

class Node:
    def __init__(self, value):
        self.value = value
        self.left = None
        self.right = None

    def insert(self, value):
        if value < self.value:
            if self.left is None:
                self.left = Node(value)
            else:
                self.left.insert(value)
        elif value > self.value:
            if self.right is None:
                self.right = Node(value)
            else:
                self.right.insert(value)
        # If the value is equal, no insertion is needed as BST typically does not allow duplicates

    def convert_to_array(self, node, array):
        """Performs a Depth-First Search (DFS) to convert the tree into an array."""
        if node:
            array.append(node.value)
            self.convert_to_array(node.left, array)
            self.convert_to_array(node.right, array)
        return array

class BinaryTree:
    def __init__(self):
        self.root = None

    def insert(self, data):
        if self.root is None:
            self.root = Node(data)
        else:
            self.root.insert(data)

    def find(self, data):
        return self._find_node(self.root, data)

    def _find_node(self, node, data):
        if node is None:
            return False
        elif data == node.value:
            return True
        elif data < node.value:
            return self._find_node(node.left, data)
        else:
            return self._find_node(node.right, data)

    def convert_to_array(self):
        if self.root is None:
            return []
        return self.root.convert_to_array(self.root, [])

    def rearrange(self):
        """
        Re-arranges the binary tree so that the median of the values becomes the root.
        Returns the binary tree in DFS array form.
        """
        nodes = self.convert_to_array()
        nodes.sort()
        
        # Rebuild the tree with the median as the root
        middle = len(nodes) // 2
        self.root = Node(nodes.pop(middle))
        
        for value in nodes:
            self.insert(value)
        
        return self.convert_to_array()

# Example usage
if __name__ == "__main__":
    tree = BinaryTree()
    for _ in range(50):
        tree.insert(randint(0, 100))

    print("Original Tree (DFS array form):", tree.convert_to_array())
    print("Rearranged Tree (Median as root):", tree.rearrange())
    print("Finding 55 in the tree:", tree.find(55))
