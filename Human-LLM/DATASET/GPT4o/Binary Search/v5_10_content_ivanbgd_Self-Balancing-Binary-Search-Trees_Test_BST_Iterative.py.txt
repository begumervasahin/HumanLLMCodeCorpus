"""
Iterative Binary Search Tree implementation with testing functions.
This version assumes that keys of all nodes in the tree are unique.
"""

import BST_Iter as BST

def print_tree(bst, verbose=False):
    """Prints the tree in various orders and optionally prints all nodes in BFS order."""
    print("\nTree Structure:")
    print("In order:  ", bst.in_order())
    print("Pre order: ", bst.pre_order())
    print("BFS:       ", bst.bfs())

    if verbose:
        print("Nodes (in BFS order):")
        for node in bst.bfs():
            bst.find(node).print_node()
    print()

def create_tree():
    """Creates and returns a binary search tree with a predefined structure."""
    bst = BST.BinarySearchTree(BST.Node(7))
    bst.insert(4)
    bst.insert(1)
    bst.insert(6)
    bst.insert(13)
    bst.insert(15)
    bst.insert(10)
    return bst, bst.get_root()

def test1():
