class TreeNode:
    """Basic node structure for a binary tree."""
    
    def __init__(self, val):
        self.val = val
        self.left = None
        self.right = None
from TreeNode import TreeNode

class BST:
    """Basic Binary Search Tree implementation."""
    
    def __init__(self):
        self.root = None

    def insert(self, val):
        """Insert a new node with the given value."""
        if not self.root:
            self.root = TreeNode(val)
        else:
            self._insert(self.root, val)

    def _insert(self, node, val):
        """Helper method to insert a value into the BST."""
        if val < node.val:
            if node.left is None:
                node.left = TreeNode(val)
            else:
                self._insert(node.left, val)
        else:
            if node.right is None:
                node.right = TreeNode(val)
            else:
                self._insert(node.right, val)

    def delete(self, val):
        """Remove a node with the given value."""
        self.root = self._delete(self.root, val)

    def _delete(self, node, val):
        """Helper method to delete a node from the BST."""
        if not node:
            return None
        
        if val < node.val:
            node.left = self._delete(node.left, val)
        elif val > node.val:
            node.right = self._delete(node.right, val)
        else:
            if not node.left:
                return node.right
            if not node.right:
                return node.left
            
            min_larger_node = self._find_min(node.right)
            node.val = min_larger_node.val
            node.right = self._delete(node.right, min_larger_node.val)
        
        return node

    def _find_min(self, node):
        """Helper method to find the node with the minimum value."""
        while node.left:
            node = node.left
        return node

    def search(self, val):
        """Search for a node with the given value."""
        return self._search(self.root, val)

    def _search(self, node, val):
        """Helper method to search for a value in the BST."""
        if not node or node.val == val:
            return node
        if val < node.val:
            return self._search(node.left, val)
        return self._search(node.right, val)

    def inorder(self):
        """In-order traversal of the tree."""
        return self._inorder(self.root)

    def _inorder(self, node):
        """Helper method for in-order traversal."""
        result = []
        if node:
            result.extend(self._inorder(node.left))
            result.append(node.val)
            result.extend(self._inorder(node.right))
        return result

from TreeNode import TreeNode

class AVLTreeNode(TreeNode):
    """Node structure with additional height attribute for AVL trees."""
    
    def __init__(self, val):
        super().__init__(val)
        self.height = 1

class AVL(BST):
    """AVL Tree implementation with self-balancing functionality."""
    
    def insert(self, val):
        """Insert a new value and rebalance the tree if necessary."""
        self.root = self._insert(self.root, val)

    def _insert(self, node, val):
        """Helper method to insert a value into the AVL tree."""
        if not node:
            return AVLTreeNode(val)
        
        if val < node.val:
            node.left = self._insert(node.left, val)
        else:
            node.right = self._insert(node.right, val)
        
        self._update_height(node)
        return self._rebalance(node)

    def delete(self, val):
        """Remove a value and rebalance the tree if necessary."""
        self.root = self._delete(self.root, val)

    def _delete(self, node, val):
        """Helper method to delete a value from the AVL tree."""
        if not node:
            return None
        
        if val < node.val:
            node.left = self._delete(node.left, val)
        elif val > node.val:
            node.right = self._delete(node.right, val)
        else:
            if not node.left:
                return node.right
            if not node.right:
                return node.left
            
            min_larger_node = self._find_min(node.right)
            node.val = min_larger_node.val
            node.right = self._delete(node.right, min_larger_node.val)
        
        self._update_height(node)
        return self._rebalance(node)

    def _update_height(self, node):
        """Update the height of the node based on its children."""
        node.height = 1 + max(self._height(node.left), self._height(node.right))

    def _height(self, node):
        """Return the height of the node."""
        return node.height if node else 0

    def _rebalance(self, node):
        """Rebalance the tree if it becomes unbalanced."""
        balance = self._get_balance(node)
        
        if balance > 1:  # Left heavy
            if self._get_balance(node.left) < 0:
                node.left = self._rotate_left(node.left)
            return self._rotate_right(node)
        
        if balance < -1:  # Right heavy
            if self._get_balance(node.right) > 0:
                node.right = self._rotate_right(node.right)
            return self._rotate_left(node)
        
        return node

    def _get_balance(self, node):
        """Calculate the balance factor of the node."""
        return self._height(node.left) - self._height(node.right)

    def _rotate_left(self, node):
        """Perform a left rotation on the node."""
        new_root = node.right
        node.right = new_root.left
        new_root.left = node
        
        self._update_height(node)
        self._update_height(new_root)
        
        return new_root

    def _rotate_right(self, node):
        """Perform a right rotation on the node."""
        new_root = node.left
        node.left = new_root.right
        new_root.right = node
        
        self._update_height(node)
        self._update_height(new_root)
        
        return new_root
