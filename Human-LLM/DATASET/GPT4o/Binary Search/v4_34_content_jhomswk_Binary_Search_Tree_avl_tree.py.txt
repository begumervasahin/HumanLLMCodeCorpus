from binary_search_tree import Binary_Search_Tree

def height(node):
    """
    Returns the height of the AVL tree rooted at the given node.
    """
    return node.height if node else -1

def update_height(node):
    """
    Updates the height of the AVL tree rooted at the given node.
    """
    node.height = 1 + max(height(node.left), height(node.right))

class AVL_Tree(Binary_Search_Tree):
    """
    AVL Tree implementation, inheriting from the Binary_Search_Tree class.
    Supports self-balancing operations.
    """

    def rotate_left(self, node):
        """
        Left-rotates the subtree rooted at the given node.
        """
        child = node.right
        child.parent = node.parent
        
        # Update the root if necessary
        if node.parent is None:
            self.root = child
        else:
            if node is node.parent.left:
                node.parent.left = child
            else:
                node.parent.right = child

        # Perform the rotation
        node.right = child.left
        if node.right:
            node.right.parent = node
        child.left = node
        node.parent = child
        
        # Update heights after rotation
        update_height(node)
        update_height(child)

    def rotate_right(self, node):
        """
        Right-rotates the subtree rooted at the given node.
        """
        child = node.left
        child.parent = node.parent
        
        # Update the root if necessary
        if node.parent is None:
            self.root = child
        else:
            if node is node.parent.left:
                node.parent.left = child
            else:
                node.parent.right = child

        # Perform the rotation
        node.left = child.right
        if node.left:
            node.left.parent = node
        child.right = node
        node.parent = child
        
        # Update heights after rotation
        update_height(node)
        update_height(child)

    def balance(self, node):
        """
        Balances the AVL tree starting from the given node up to the root.
        """
        while node:
            update_height(node)
            balance_factor = height(node.left) - height(node.right)

            # Left heavy
            if balance_factor > 1:
                if height(node.left.left) >= height(node.left.right):
                    self.rotate_right(node)
                else:
                    self.rotate_left(node.left)
                    self.rotate_right(node)
            
            # Right heavy
            elif balance_factor < -1:
                if height(node.right.right) >= height(node.right.left):
                    self.rotate_left(node)
                else:
                    self.rotate_right(node.right)
                    self.rotate_left(node)
            
            node = node.parent

    def insert(self, key):
        """
        Inserts a node with the given key and rebalances the tree.
        """
        node = super(AVL_Tree, self).insert(key)
        self.balance(node)

    def delete(self, key):
        """
        Deletes the node with the given key and rebalances the tree.
        """
        node = super(AVL_Tree, self).delete(key)
        if node:
            self.balance(node.parent)
