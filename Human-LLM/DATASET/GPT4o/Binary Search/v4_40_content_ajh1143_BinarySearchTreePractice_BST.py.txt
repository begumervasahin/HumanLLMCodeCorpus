class Node:
    def __init__(self, value):
        """Initialize the node with a value and set left and right children to None."""
        self.value = value
        self.left_child = None
        self.right_child = None

    def insert(self, data):
        """
        Insert a value into the binary search tree.
        
        Args:
            data (int): The value to be inserted.
        
        Returns:
            bool: True if the value is inserted, False if the value already exists.
        """
        if self.value == data:
            return False
        elif data < self.value:
            if self.left_child:
                return self.left_child.insert(data)
            else:
                self.left_child = Node(data)
                return True
        else:
            if self.right_child:
                return self.right_child.insert(data)
            else:
                self.right_child = Node(data)
                return True

    def find(self, data):
        """
        Search for a value in the binary search tree.
        
        Args:
            data (int): The value to be found.
        
        Returns:
            bool: True if the value is found, False otherwise.
        """
        if self.value == data:
            return True
        elif data < self.value:
            if self.left_child:
                return self.left_child.find(data)
            else:
                return False
        else:
            if self.right_child:
                return self.right_child.find(data)
            else:
                return False

    def preorder(self):
        """Perform a preorder traversal of the tree, printing the values."""
        print(self.value)
        if self.left_child:
            self.left_child.preorder()
        if self.right_child:
            self.right_child.preorder()

    def inorder(self):
        """Perform an inorder traversal of the tree, printing the values."""
        if self.left_child:
            self.left_child.inorder()
        print(self.value)
        if self.right_child:
            self.right_child.inorder()

    def postorder(self):
        """Perform a postorder traversal of the tree, printing the values."""
        if self.left_child:
            self.left_child.postorder()
        if self.right_child:
            self.right_child.postorder()
        print(self.value)

class Tree:
    def __init__(self):
        """Initialize the tree with an empty root."""
        self.root = None

    def insert(self, data):
        """
        Insert a value into the tree.
        
        Args:
            data (int): The value to be inserted.
        
        Returns:
            bool: True if the value is inserted, False if the value already exists.
        """
        if self.root:
            return self.root.insert(data)
        else:
            self.root = Node(data)
            return True

    def find(self, data):
        """
        Search for a value in the tree.
        
        Args:
            data (int): The value to be found.
        
        Returns:
            bool: True if the value is found, False otherwise.
        """
        if self.root:
            return self.root.find(data)
        else:
            return False

    def preorder(self):
        """Perform a preorder traversal of the tree, printing the values."""
        print("Preorder traversal:")
        if self.root:
            self.root.preorder()
        else:
            print("Tree is empty.")

    def inorder(self):
        """Perform an inorder traversal of the tree, printing the values."""
        print("Inorder traversal:")
        if self.root:
            self.root.inorder()
        else:
            print("Tree is empty.")

    def postorder(self):
        """Perform a postorder traversal of the tree, printing the values."""
        print("Postorder traversal:")
        if self.root:
            self.root.postorder()
        else:
            print("Tree is empty.")

# Example usage of the Tree class
bst = Tree()
bst.insert(1)
