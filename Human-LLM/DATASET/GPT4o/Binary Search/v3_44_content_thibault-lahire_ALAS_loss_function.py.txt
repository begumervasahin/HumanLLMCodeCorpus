import numpy as np

def sigmoid(w):
    """Computes the sigmoid function."""
    return 1 / (1 + np.exp(-w))

def sigmoid_derivative(w):
    """Computes the first derivative of the sigmoid function."""
    exp_neg_w = np.exp(-w)
    
    if exp_neg_w < 1e16:
        if exp_neg_w > 1e-16:
            return exp_neg_w / (1 + exp_neg_w) ** 2
        else:
            return exp_neg_w  # When exp_neg_w is very small, return exp_neg_w directly
    else:
        return np.exp(w)  # When exp_neg_w is very large, return the positive exponent

def sigmoid_second_derivative(w):
    """Computes the second derivative of the sigmoid function."""
    exp_neg_w = np.exp(-w)
    
    if exp_neg_w < 1e16:
        if exp_neg_w > 1e-16:
            exp_w = np.exp(w)
            return -exp_w * (exp_w - 1) / (1 + exp_w) ** 3
        else:
            return -exp_neg_w  # For small values, approximate the second derivative
    else:
        return np.exp(w)  # For large values, return the positive exponent

def compute_loss(x, S, A, y, output_type=1, penalize=False):
    """
    Computes the loss function and optionally its gradient and Hessian.

    Parameters:
    x (array-like): The point at which to compute the loss.
    S (list): A subset of indices for the samples to be used.
    A (array-like): A matrix containing the datapoints.
    y (array-like): A vector of labels.
    output_type (int): Specifies the output (1 for loss, 2 for loss and gradient, 3 for loss, gradient, and Hessian).
    penalize (bool): If True, includes a penalty term in the loss function.

    Returns:
    - If output_type=1: The value of the loss function.
    - If output_type=2: The value of the loss function and its gradient.
    - If output_type=3: The value of the loss function, its gradient, and its Hessian.
    """
    d = A.shape[1]  # Dimensionality of the input data
    card_S = len(S)  # Cardinality of the subset S
    
    loss_value = 0  # Initialize loss value

    # Initialize gradient and Hessian if required
    gradient = np.zeros(d) if output_type >= 2 else None
    hessian = np.zeros((d, d)) if output_type == 3 else None
    
    for i in S:
        y_i = y[i]
        A_i = A[i]
        dot_product = np.dot(x, A_i)
        
        # Calculate sigmoid value and loss contribution
        sigmoid_value = sigmoid(dot_product)
        loss_value += (y_i - sigmoid_value) ** 2
        
        # Calculate gradient if required
        if output_type >= 2:
            sigmoid_deriv = sigmoid_derivative(dot_product)
            gradient += 2 * A_i * sigmoid_deriv * (sigmoid_value - y_i)
        
        # Calculate Hessian if required
        if output_type == 3:
            sigmoid_2nd_deriv = sigmoid_second_derivative(dot_product)
            hessian += 2 * (
                (sigmoid_value - y_i) * sigmoid_2nd_deriv + 
                sigmoid_deriv ** 2
            ) * np.outer(A_i, A_i)
    
    # Apply penalization if required
    if penalize:
        loss_value += 0.5 * np.linalg.norm(x) ** 2
    
    # Average the loss value by the size of the subset
    loss_value /= card_S
    
    # Return the appropriate output based on the requested output_type
    if output_type == 3:
        hessian = 0.5 * (hessian + hessian.T)  # Ensure symmetry
        if penalize:
            gradient += x
            hessian += np.eye(d)
        gradient /= card_S
        hessian /= card_S
        return loss_value, gradient, hessian
    elif output_type == 2:
        if penalize:
            gradient += x
        gradient /= card_S
        return loss_value, gradient
    else:
        return loss_value
