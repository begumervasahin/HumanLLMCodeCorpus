class Node:
    """
    Node class for a binary search tree (BST). 
    This class is designed to be hidden from the user.
    """

    def __init__(self, val):
        """
        Initialize the Node with a given value.
        """
        self.value = val
        self.left = None
        self.right = None

    def insert(self, val):
        """
        Insert a value into the BST. 
        Returns True if the insertion is successful, False if the value already exists.
        """
        if self.value == val:
            return False
        elif val < self.value:
            if self.left:
                return self.left.insert(val)
            else:
                self.left = Node(val)
                return True
        else:  # val > self.value
            if self.right:
                return self.right.insert(val)
            else:
                self.right = Node(val)
                return True

    def find(self, val):
        """
        Search for a value in the BST.
        Returns True if the value is found, False otherwise.
        """
        if self.value == val:
            return True
        elif val > self.value:
            return self.right.find(val) if self.right else False
        else:  # val < self.value
            return self.left.find(val) if self.left else False

    def preorder(self):
        """
        Perform a pre-order traversal of the BST.
        """
        print(self.value)
        if self.left:
            self.left.preorder()
        if self.right:
            self.right.preorder()

    def inorder(self):
        """
        Perform an in-order traversal of the BST.
        """
        if self.left:
            self.left.inorder()
        print(self.value)
        if self.right:
            self.right.inorder()

    def postorder(self):
        """
        Perform a post-order traversal of the BST.
        """
        if self.left:
            self.left.postorder()
        if self.right:
            self.right.postorder()
        print(self.value)

    def delete(self, val):
        """
        Delete a node with the specified value from the BST and rearrange the tree nodes accordingly.
        Returns the root of the modified subtree.
        """
        if self.value == val:
            # Node with two children
            if self.left and self.right:
                successor = self.right.find_successor()
                self.value = successor.value
                self.right = self.right.delete(successor.value)
            # Node with only the left child
            elif self.left:
                return self.left
            # Node with only the right child
            elif self.right:
                return self.right
            # Node with no children
            else:
                return None
        elif val > self.value:
            if self.right:
                self.right = self.right.delete(val)
        else:  # val < self.value
            if self.left:
                self.left = self.left.delete(val)
        
        return self

    def find_successor(self):
        """
        Find the successor (the next greater element) of the current node in the BST.
        The successor is the leftmost node in the right subtree.
        """
        if self.left:
            return self.left.find_successor()
