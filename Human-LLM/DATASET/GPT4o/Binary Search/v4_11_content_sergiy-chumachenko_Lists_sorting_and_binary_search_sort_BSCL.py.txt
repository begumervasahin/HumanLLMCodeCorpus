class MySorting:
    def __init__(self, nums):
        """Initialize with a list of numbers to sort."""
        self.nums = nums

    def __str__(self):
        """Return a string representation of the sorting method."""
        return "MySorting method can be used to sort your list:\n{}".format(self.nums)

    def asc(self):
        """Sort the list in ascending order using bubble sort."""
        n = len(self.nums)
        for i in range(n - 1):
            swapped = False
            for j in range(n - 1 - i):
                if self.nums[j] > self.nums[j + 1]:
                    self.nums[j], self.nums[j + 1] = self.nums[j + 1], self.nums[j]
                    swapped = True
            if not swapped:
                break
        return self.nums

    def desc(self):
        """Sort the list in descending order using bubble sort."""
        n = len(self.nums)
        for i in range(n - 1):
            swapped = False
            for j in range(n - 1 - i):
                if self.nums[j] < self.nums[j + 1]:
                    self.nums[j], self.nums[j + 1] = self.nums[j + 1], self.nums[j]
                    swapped = True
            if not swapped:
                break
        return self.nums

def binary_search(sorted_list, target):
    """Perform binary search on a sorted list and return the index of the target."""
    left, right = 0, len(sorted_list) - 1
    
    while left <= right:
        middle = (left + right) // 2
        if target == sorted_list[middle]:
            return middle
        elif target > sorted_list[middle]:
            left = middle + 1
        else:
            right = middle - 1

    return left

def insert_into_sorted_list(n, sorted_list):
    """Insert an element into its correct position in a sorted list."""
    if not sorted_list:
        sorted_list.append(n)
    else:
        index = binary_search(sorted_list, n)
        sorted_list.insert(index, n)
    return sorted_list

# Example Usage:
if __name__ == "__main__":
    sorter = MySorting([34, 10, -5, 72, 0, 8])
    print(sorter)  # Display the original list
    
    print("Sorted in ascending order:", sorter.asc())
    print("Sorted in descending order:", sorter.desc())
    
    sorted_list = [10, 20, 30, 40]
    print("Inserting 25 into sorted list:", insert_into_sorted_list(25, sorted_list))
