import random
import ABR
import RB
import sys
import pickle

# Adjust the recursion limit as necessary for deep recursion
sys.setrecursionlimit(4400)

def test_heights():
    """Load test parameters, run height tests on BST and RBT, and save results."""
    # Load the test parameters from a file
    test_heights_start = pickle.load(open("tests.p", "rb"))
    
    # Run the tests on BST and RBT
    result_ABR = test_height_ABR(test_heights_start)
    result_RB_random, result_RB_ordered = test_height_RB(test_heights_start)
    
    # Save the results to files
    pickle.dump(result_ABR, open("resultHeightABR.p", "wb"))
    pickle.dump(result_RB_random, open("resultHeightRB_random.p", "wb"))
    pickle.dump(result_RB_ordered, open("resultHeightRB_ordered.p", "wb"))

def test_height_ABR(test_heights_start):
    """Test the height of Binary Search Trees (BST) with random elements."""
    elements, range_elements = test_heights_start
    result_random_height_ABR = []

    print("\n" + "=" * 100)
    print("Average heights of Binary Search Trees for random elements. 3 executions.")
    print("Columns: ranges [20, 200, 1000, 10000000].")
    print("Lines: elements [10, 50, 500, 1000, 10000, 50000]")
    print("=" * 100 + "\n")

    for p in elements:
        average_array_height_ABR = []
        for i in range_elements:
            heights = []
            for _ in range(3):  # 3 executions for each combination
                ABR_tree = ABR.BinarySearchTree()
                for _ in range(p):
                    ABR_tree.insert(random.randint(0, i))
                height_ABR = ABR_tree.getHeight(ABR_tree.root)
                heights.append(height_ABR)
            average_array_height_ABR.append(sum(heights) / len(heights))
        result_random_height_ABR.append(average_array_height_ABR)
        print(average_array_height_ABR)
    
    return result_random_height_ABR

def test_height_RB(test_heights_start):
    """Test the height of Red-Black Trees (RBT) with random and ordered elements."""
    elements, range_elements = test_heights_start
    result_random_height_RB = []
    result_ordered_height_RB = []

    # Random elements test
    print("\n" + "=" * 100)
    print("Average heights of Red-Black Trees for random elements. 10 executions.")
    print("Columns: ranges [20, 200, 1000, 10000000].")
    print("Lines: elements [10, 50, 500, 1000, 10000, 50000]")
    print("=" * 100 + "\n")

    for p in elements:
        average_array_height_RB = []
        for i in range_elements:
            heights = []
            for _ in range(10):  # 10 executions for each combination
                RB_tree = RB.RedBlackTree()
                for _ in range(p):
                    RB_tree.insert(random.randint(0, i))
                height_RB = RB_tree.getHeight(RB_tree.root)
                heights.append(height_RB)
            average_array_height_RB.append(sum(heights) / len(heights))
        result_random_height_RB.append(average_array_height_RB)
        print(average_array_height_RB)

    # Ordered elements test
    print("\n" + "=" * 100)
    print("Average heights of Red-Black Trees for ordered elements. 5 executions.")
    print("Columns: ranges [20, 200, 1000, 10000000].")
    print("Lines: elements [10, 50, 500, 1000, 10000, 50000]")
    print("=" * 100 + "\n")

    for p in elements:
        average_array_height_RB = []
        for _ in range(len(range_elements)):
            heights = []
            for _ in range(5):  # 5 executions for each combination
                RB_tree = RB.RedBlackTree()
                for k in range(p):
                    RB_tree.insert(k)
                height_RB = RB_tree.getHeight(RB_tree.root)
                heights.append(height_RB)
            average_array_height_RB.append(sum(heights) / len(heights))
        result_ordered_height_RB.append(average_array_height_RB)
        print(average_array_height_RB)
    
    return result_random_height_RB, result_ordered_height_RB

if __name__ == "__main__":
    test_heights()
