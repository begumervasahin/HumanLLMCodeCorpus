import random

class Node:
    def __init__(self, data):
        """Initialize a Node with data."""
        self.data = data
        self.left = None
        self.right = None
        self.priority = random.random()

    def __str__(self):
        """Return string representation of data."""
        return str(self.data)

class TreapSet:
    def __init__(self):
        self.root = None
        self.count = 0

    def left_rotate(self, node):
        """Perform a left rotation on the subtree rooted at 'node'."""
        new_parent = node.right
        node.right = new_parent.left
        new_parent.left = node
        return new_parent

    def right_rotate(self, node):
        """Perform a right rotation on the subtree rooted at 'node'."""
        new_parent = node.left
        node.left = new_parent.right
        new_parent.right = node
        return new_parent

    def add(self, element):
        """Insert 'element' into the Treap."""
        self.count += 1
        if self.root is None:
            self.root = Node(element)
        else:
            self.root = self._add(element, self.root)

    def _add(self, element, node):
        if node is None:
            return Node(element)
        if element < node.data:
            node.left = self._add(element, node.left)
            if node.left.priority > node.priority:
                return self.right_rotate(node)
        else:
            node.right = self._add(element, node.right)
            if node.right.priority > node.priority:
                return self.left_rotate(node)
        return node

    def __contains__(self, element):
        """Check if 'element' is in the Treap."""
        return self._contains(self.root, element)

    def _contains(self, node, element):
        if node is None:
            return False
        if element == node.data:
            return True
        elif element < node.data:
            return self._contains(node.left, element)
        else:
            return self._contains(node.right, element)

    def height(self):
        """Return the height of the Treap."""
        return self._height(self.root)

    def _height(self, node):
        if node is None:
            return 0
        return 1 + max(self._height(node.left), self._height(node.right))

    def __len__(self):
        """Return the number of elements in the Treap."""
        return self.count
