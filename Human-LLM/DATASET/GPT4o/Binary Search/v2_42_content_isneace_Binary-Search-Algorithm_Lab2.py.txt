class Patient:
    def __init__(self, name, contact, age, days, balance):
        """Initialize a Patient object with name, contact, age, days, and balance."""
        self.name = name
        self.contact = contact
        self.age = age
        self.days = days
        self.balance = balance

    def get_name(self):
        """Return the patient's name."""
        return self.name

    def get_contact(self):
        """Return the patient's contact information."""
        return self.contact

    def get_age(self):
        """Return the patient's age."""
        return self.age

    def get_days(self):
        """Return the number of days the patient has stayed."""
        return self.days

    def get_balance(self):
        """Return the patient's balance."""
        return self.balance

    def update_name(self, name):
        """Update the patient's name."""
        self.name = name

    def update_contact(self, contact):
        """Update the patient's contact information."""
        self.contact = contact

    def update_age(self, age):
        """Update the patient's age."""
        self.age = age

    def update_days(self, days):
        """Update the number of days the patient has stayed."""
        self.days = days

    def update_balance(self, balance):
        """Update the patient's balance."""
        self.balance = balance


def load_patient_data(file_name):
    """
    Load patient data from a file and return a list of Patient objects.

    Args:
        file_name (str): The name of the file containing patient data.

    Returns:
        list: A list of Patient objects.
    """
    patient_list = []
    with open(file_name, "r") as fp:
        fp.readline()  # Skip header line
        for line in fp:
            line = line.strip()
            if line:
                fields = line.split()
                patient = Patient(
                    name=str(fields[0]),
                    contact=str(fields[1]),
                    age=int(fields[2]),
                    days=int(fields[3]),
                    balance=float(fields[4])
                )
                patient_list.append(patient)
    return patient_list


def binary_search_patient(patient_list, name):
    """
    Perform a binary search for a patient by name.

    Args:
        patient_list (list): A list of Patient objects, sorted by name.
        name (str): The name of the patient to search for.

    Returns:
        tuple: A tuple containing the index of the patient in the list and
               the number of iterations taken to find the patient.
               If the patient is not found, returns (-1, count).
    """
    first = 0
    last = len(patient_list) - 1
    iterations = 0

    while first <= last:
        iterations += 1
        mid = (first + last) // 2
        mid_name = patient_list[mid].get_name()

        if name > mid_name:
            first = mid + 1
        elif name < mid_name:
            last = mid - 1
        else:
            return mid, iterations

    return -1, iterations


def main():
    """Main function to load patient data, search for a patient, and display the results."""
    patient_list = load_patient_data("Patient_Info-sorted.dat")
    name = input("Enter a patient's name to search: ")
    index, iterations = binary_search_patient(patient_list, name)

    if index != -1:
        patient = patient_list[index]
        print(f"Patient found at index {index}.")
        print(f"Number of iterations: {iterations}")
        print(f"Patient Contact: {patient.get_contact()}")
        print(f"Patient Age: {patient.get_age()}")
        print(f"Patient Days: {patient.get_days()}")
        print(f"Patient Balance: {patient.get_balance()}")
    else:
        print("Patient not found.")
        print(f"Number of iterations: {iterations}")


if __name__ == "__main__":
    main()
