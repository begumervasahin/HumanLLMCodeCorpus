import sqlite3
import pandas as pd

# Connect to the SQLite database
def connect_to_db(db_name="courses.db"):
    return sqlite3.connect(db_name)

def cleanup_data(conn):
    """
    Clean the data by removing rows with invalid values in specific columns.
    """
    cleanup_query = """
    DELETE FROM coursedata 
    WHERE course_level = -1 
    OR course_category = -1 
    OR course_enrollment = -1
    """
    conn.execute(cleanup_query)

def load_data(conn):
    """
    Load the cleaned data from the database into a pandas DataFrame.
    
    Returns:
        pd.DataFrame: The cleaned data.
    """
    return pd.read_sql("SELECT * FROM coursedata", con=conn)

def calculate_normalized_range(column_data):
    """
    Calculate the normalized range of a given column.
    
    Args:
        column_data (pd.Series): The data of the column.
    
    Returns:
        float: The normalized range of the column.
    """
    return (column_data.max() - column_data.min()) / column_data.std()

def get_most_variable_column(data):
    """
    Determine the column with the highest variability based on the normalized range.
    
    Returns:
        str: The name of the column with the highest normalized range.
    """
    variability = []
    relevant_columns = ['course_level', 'course_category', 'course_enrollment']

    for column in relevant_columns:
        column_data = data[column]
        if column_data.max() != column_data.min():
            normalized_range = calculate_normalized_range(column_data)
            variability.append((normalized_range, column))

    # Return the column with the highest normalized range
    _, most_variable_column = max(variability)
    return most_variable_column

def display_course_options(low, mid, high):
    """
    Display three example courses representing low, medium, and high values.
    
    Args:
        low (pd.Series): The course with the lowest value.
        mid (pd.Series): The course with the median value.
        high (pd.Series): The course with the highest value.
    """
    print("Here are three example courses to choose from:")
    for i, course in enumerate([low, mid, high], start=1):
        print(f"\n{i}. {course['course_name']}")
        print(f"\tLevel:\t\t{course['course_level']}")
        print(f"\tCategory:\t{course['course_category']}")
        print(f"\tEnrollment:\t{course['course_enrollment']}")

def get_user_selection():
    """
    Prompt the user to select an option (1, 2, or 3) and ensure a valid input.
    
    Returns:
        int: The user's selection as an integer.
    """
    while True:
        selection = input("Please select a course (1, 2, or 3):\n")
        if selection in ["1", "2", "3"]:
            return int(selection)
        print("Invalid input. Please enter 1, 2, or 3.")

def refine_selection(data):
    """
    Refine the dataset based on user preferences, focusing on the most variable column.
    
    Args:
        data (pd.DataFrame): The current dataset.
    
    Returns:
        pd.DataFrame: A subset of the data narrowed down based on user selection.
    """
    # Determine the column with the highest variability and sort by it
    most_variable_column = get_most_variable_column(data)
    data_sorted = data.sort_values(by=most_variable_column)

    # Display three course options
    display_course_options(
        data_sorted.iloc[0],                    # Lowest value
        data_sorted.iloc[len(data_sorted) // 2],  # Median value
        data_sorted.iloc[-1]                    # Highest value
    )

    print(f"Based on the dimension of {most_variable_column.split('_')[1]},")
    selection = get_user_selection()

    # Narrow down the data based on the user's choice
    segment_start = round((selection - 1) * len(data_sorted) / 3)
    segment_end = round(selection * len(data_sorted) / 3)
    return data_sorted.iloc[segment_start:segment_end]

def main():
    conn = connect_to_db()
    
    try:
        cleanup_data(conn)
        data = load_data(conn)

        # Main loop: Continuously refine the selection until few courses remain
        while len(data) > 6:
            print("__________________________________________")
            data = refine_selection(data)

            print("Your top courses are:")
            print(data.head())
            print(f"There are {len(data)} courses remaining.")

            if len(data) < 3:
                print("Stopping execution because there are too few courses.")
                break
    finally:
        # Close the database connection
        conn.close()

if __name__ == "__main__":
    main()
