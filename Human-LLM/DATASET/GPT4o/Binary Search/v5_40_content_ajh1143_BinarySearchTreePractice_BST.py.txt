class Node:
    def __init__(self, value):
        """Initialize the node with a value and set left and right children to None."""
        self.value = value
        self.left = None
        self.right = None

    def insert(self, data):
        """
        Insert a value into the binary search tree.
        
        Args:
            data (int): The value to be inserted.
        
        Returns:
            bool: True if the value is inserted, False if the value already exists.
        """
        if data == self.value:
            return False
        elif data < self.value:
            if self.left:
                return self.left.insert(data)
            else:
                self.left = Node(data)
                return True
        else:
            if self.right:
                return self.right.insert(data)
            else:
                self.right = Node(data)
                return True

    def find(self, data):
        """
        Search for a value in the binary search tree.
        
        Args:
            data (int): The value to be found.
        
        Returns:
            bool: True if the value is found, False otherwise.
        """
        if data == self.value:
            return True
        elif data < self.value:
            return self.left.find(data) if self.left else False
        else:
            return self.right.find(data) if self.right else False

    def preorder(self):
        """Perform a preorder traversal of the tree, printing the values."""
        print(self.value)
        if self.left:
            self.left.preorder()
        if self.right:
            self.right.preorder()

    def inorder(self):
        """Perform an inorder traversal of the tree, printing the values."""
        if self.left:
            self.left.inorder()
        print(self.value)
        if self.right:
            self.right.inorder()

    def postorder(self):
        """Perform a postorder traversal of the tree, printing the values."""
        if self.left:
            self.left.postorder()
        if self.right:
            self.right.postorder()
        print(self.value)


class Tree:
    def __init__(self):
        """Initialize the tree with an empty root."""
        self.root = None

    def insert(self, data):
        """
        Insert a value into the tree.
        
        Args:
            data (int): The value to be inserted.
        
        Returns:
            bool: True if the value is inserted, False if the value already exists.
        """
        if self.root:
            return self.root.insert(data)
        else:
            self.root = Node(data)
            return True

    def find(self, data):
        """
        Search for a value in the tree.
        
        Args:
            data (int): The value to be found.
        
        Returns:
            bool: True if the value is found, False otherwise.
        """
        return self.root.find(data) if self.root else False

    def preorder(self):
        """Perform a preorder traversal of the tree, printing the values."""
        print("Preorder traversal:")
        if self.root:
            self.root.preorder()
        else:
            print("Tree is empty.")

    def inorder(self):
        """Perform an inorder traversal of the tree, printing the values."""
        print("Inorder traversal:")
        if self.root:
            self.root.inorder()
        else:
            print("Tree is empty.")

    def postorder(self):
        """Perform a postorder traversal of the tree, printing the values."""
        print("Postorder traversal:")
        if self.root:
            self.root.postorder()
        else:
            print("Tree is empty.")


# Example usage of the Tree class
bst = Tree()
bst.insert(1)
