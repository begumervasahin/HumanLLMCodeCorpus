import sqlite3
import numpy as np
import pandas as pd

# Establish connection to the database
conn = sqlite3.connect("courses.db")
c = conn.cursor()

# Delete rows with invalid course data
delete_query = """
    DELETE FROM coursedata 
    WHERE course_level = -1 
    OR course_category = -1 
    OR course_enrollment = -1
"""
conn.execute(delete_query)

# Fetch the remaining data
data = pd.read_sql("SELECT * FROM coursedata", con=conn)

def find_most_variable_column(data):
    """
    Determine which column has the highest relative variability.
    The function returns the name of the column with the highest
    (max - min) / std deviation.
    """
    results = []
    columns = ["course_level", "course_category", "course_enrollment"]
    
    for col in columns:
        column_data = data[col]
        if column_data.max() != column_data.min():
            variability = (column_data.max() - column_data.min()) / column_data.std()
            results.append((variability, col))
    
    max_result, max_col = max(results, key=lambda x: x[0])
    return max_col

def print_course_options(low, med, high):
    """Print details of three example courses."""
    print("Here are three example courses:")
    for course in [low, med, high]:
        print(course[1])
        print(f"\tLevel:\t\t{course[2]}")
        print(f"\tCategory:\t{course[3]}")
        print(f"\tEnrollment:\t{course[4]}")

def parse_selection(selection):
    """Parse the user's selection, ensuring it's a valid option."""
    if selection in ["1", "2", "3"]:
        return int(selection)
    else:
        return parse_selection(input("Please enter 1, 2, or 3.\n"))

def select_course_segment(data):
    """
    Guide the user to select a course segment based on the most variable column.
    The function returns the selected segment of the data.
    """
    most_variable_col = find_most_variable_column(data)
    data.sort_values(by=most_variable_col, inplace=True)
    
    print_course_options(
        data.iloc[0], 
        data.iloc[len(data) // 2], 
        data.iloc[-1]
    )
    
    print(f"Based on the dimension of {most_variable_col.split('_')[1]},")
    selection = parse_selection(input("Do you prefer course 1, 2, or 3?\n"))
    
    segment_start = (selection - 1) * len(data) // 3
    segment_end = selection * len(data) // 3
    return data.iloc[segment_start:segment_end]

# Main loop to guide the user through course selection
while len(data) > 6:
    print("__________________________________________")
    data = select_course_segment(data)
    
    print("Your top courses are: ")
    print(data.head())
    print(f"There are {len(data)} courses remaining.")
    
    if len(data) < 3:
        print("Stopping execution because there are too few courses.")
        break

# Close the database connection
conn.close()
