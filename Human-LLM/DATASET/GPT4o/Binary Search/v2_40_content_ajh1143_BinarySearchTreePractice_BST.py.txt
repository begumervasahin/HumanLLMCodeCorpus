class Node:
    def __init__(self, value):
        """Initialize a Node with a given value, and set left and right children to None."""
        self.value = value
        self.leftChild = None
        self.rightChild = None

    def insert(self, data):
        """
        Insert a new node with 'data' into the BST.
        
        Args:
            data: The value to insert into the BST.

        Returns:
            bool: True if the node was inserted, False if the value already exists.
        """
        if data == self.value:
            return False
        elif data < self.value:
            if self.leftChild:
                return self.leftChild.insert(data)
            else:
                self.leftChild = Node(data)
                return True
        else:
            if self.rightChild:
                return self.rightChild.insert(data)
            else:
                self.rightChild = Node(data)
                return True

    def find(self, data):
        """
        Search for a node with 'data' in the BST.

        Args:
            data: The value to search for in the BST.

        Returns:
            bool: True if the value is found, False otherwise.
        """
        if data == self.value:
            return True
        elif data < self.value:
            if self.leftChild:
                return self.leftChild.find(data)
            else:
                return False
        else:
            if self.rightChild:
                return self.rightChild.find(data)
            else:
                return False

    def preorder(self):
        """Perform a preorder traversal of the BST and print each node's value."""
        print(self.value)
        if self.leftChild:
            self.leftChild.preorder()
        if self.rightChild:
            self.rightChild.preorder()

    def inorder(self):
        """Perform an inorder traversal of the BST and print each node's value."""
        if self.leftChild:
            self.leftChild.inorder()
        print(self.value)
        if self.rightChild:
            self.rightChild.inorder()

    def postorder(self):
        """Perform a postorder traversal of the BST and print each node's value."""
        if self.leftChild:
            self.leftChild.postorder()
        if self.rightChild:
            self.rightChild.postorder()
        print(self.value)

class Tree:
    def __init__(self):
        """Initialize the BST with an empty root."""
        self.root = None

    def insert(self, data):
        """
        Insert 'data' into the BST.

        Args:
            data: The value to insert into the BST.

        Returns:
            bool: True if the node was inserted, False if the value already exists.
        """
        if self.root:
            return self.root.insert(data)
        else:
            self.root = Node(data)
            return True

    def find(self, data):
        """
        Search for 'data' in the BST.

        Args:
            data: The value to search for in the BST.

        Returns:
            bool: True if the value is found, False otherwise.
        """
        if self.root:
            return self.root.find(data)
        else:
            return False

    def preorder(self):
        """Print the preorder traversal of the BST."""
        if self.root:
            print("Preorder traversal:")
            self.root.preorder()

    def inorder(self):
        """Print the inorder traversal of the BST."""
        if self.root:
            print("Inorder traversal:")
            self.root.inorder()

    def postorder(self):
        """Print the postorder traversal of the BST."""
        if self.root:
            print("Postorder traversal:")
            self.root.postorder()
