def binary_search():
    while True:
        try:
            target = int(input("Enter an integer: "))  # Prompt the user for an integer input
            break
        except ValueError:
            print("You did not enter an integer. Please try again.")  # Handle invalid input

    # Open the file containing the number list and read the contents
    with open('numberlist.txt') as numberlist:
        split_list = numberlist.read().split(", ")
    
    # Convert the list of strings to a list of integers
    results = list(map(int, split_list))
    sorted_results = sorted(results)  # Sort the list for binary search

    low = 0
    high = len(sorted_results) - 1

    while high > low:
        middle = (high + low) // 2  # Calculate the middle index

        if target == sorted_results[middle]:
            print("Target found!")
            break
        elif middle == low:
            print("Target not found.")
            break
        elif target > sorted_results[middle]:
            low = middle  # Move the low boundary up
        elif target < sorted_results[middle]:
            high = middle  # Move the high boundary down

    # Ask the user if they want to search for another number
    while True:
        yn = input("Would you like to test another number? (y/n): ").lower()
        if yn == 'y':
            binary_search()  # Restart the search process
            break
        elif yn == 'n':
            print("Exiting. Goodbye!")
            break
        else:
            print("Invalid input. Please enter 'y' or 'n'.")

# Start the binary search function
binary_search()
