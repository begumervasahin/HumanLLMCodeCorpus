class MySorting:
    def __init__(self, nums):
        """Initialize the sorting class with a list of numbers."""
        self.nums = nums

    def __str__(self):
        """Return a string representation of the object."""
        return f"MySorting method can be used to sort your list:\n{self.nums}"

    def asc(self):
        """Sort the list in ascending order using bubble sort."""
        n = len(self.nums)
        for i in range(n):
            swapped = False
            for j in range(0, n-i-1):
                if self.nums[j] > self.nums[j+1]:
                    self.nums[j], self.nums[j+1] = self.nums[j+1], self.nums[j]
                    swapped = True
            if not swapped:
                break
        return self.nums

    def desc(self):
        """Sort the list in descending order using bubble sort."""
        n = len(self.nums)
        for i in range(n):
            swapped = False
            for j in range(0, n-i-1):
                if self.nums[j] < self.nums[j+1]:
                    self.nums[j], self.nums[j+1] = self.nums[j+1], self.nums[j]
                    swapped = True
            if not swapped:
                break
        return self.nums

def binary_search(lst, n):
    """Perform binary search to find the index where the element n should be inserted."""
    left, right = 0, len(lst) - 1
    while left <= right:
        middle = (left + right) // 2
        if lst[middle] < n:
            left = middle + 1
        elif lst[middle] > n:
            right = middle - 1
        else:
            return middle
    return left

def general_list(n, lst):
    """Insert a number n into the sorted list lst, maintaining the sorted order."""
    if not lst:
        lst.append(n)
    else:
        index = binary_search(lst, n)
        lst.insert(index, n)
    return lst

# Example usage:
if __name__ == "__main__":
    nums = [3, 1, 4, 1, 5, 9, 2, 6, 5, 3, 5]
    sorter = MySorting(nums)

    print("Original list:", nums)
    print("Sorted list (ascending):", sorter.asc())
    print("Sorted list (descending):", sorter.desc())

    # Inserting into a sorted list
    sorted_list = [1, 3, 5, 7, 9]
    new_number = 6
    print(f"Inserting {new_number} into {sorted_list}:")
    updated_list = general_list(new_number, sorted_list)
    print("Updated list:", updated_list)
