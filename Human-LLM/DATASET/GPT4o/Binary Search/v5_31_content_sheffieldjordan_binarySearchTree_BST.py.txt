class Node:
    def __init__(self, word):
        self.word = word
        self.count = 1
        self.left = None
        self.right = None


class BSTree:
    def __init__(self, root=None):
        self.root = root

    def find(self, word):
        """Returns the count of the word in the tree."""
        return self._find(self.root, word)

    def add(self, word):
        """Adds a word to the tree or increments the count if it already exists."""
        if self.root is None:
            self.root = Node(word)
        else:
            self._add(self.root, word)

    def in_order_print(self):
        """Prints the words in the tree in in-order traversal."""
        self._in_order_print(self.root)

    def size(self):
        """Returns the total number of nodes in the tree."""
        return self._size(self.root)

    def height(self):
        """Returns the height of the tree."""
        return self._height(self.root)

    def _add(self, node, word):
        """Helper function to recursively add a word to the tree."""
        if word == node.word:
            node.count += 1
        elif word < node.word:
            if node.left is None:
                node.left = Node(word)
            else:
                self._add(node.left, word)
        else:
            if node.right is None:
                node.right = Node(word)
            else:
                self._add(node.right, word)

    def _find(self, node, word):
        """Helper function to recursively find the count of a word in the tree."""
        if node is None:
            return 0  # Word not found
        if word == node.word:
            return node.count
        elif word < node.word:
            return self._find(node.left, word)
        else:
            return self._find(node.right, word)

    def _size(self, node):
        """Helper function to recursively calculate the size of the tree."""
        if node is None:
            return 0
        return 1 + self._size(node.left) + self._size(node.right)

    def _height(self, node):
        """Helper function to recursively calculate the height of the tree."""
        if node is None:
            return 0
        left_height = self._height(node.left)
        right_height = self._height(node.right)
        return max(left_height, right_height) + 1

    def _in_order_print(self, node):
        """Helper function to recursively print the tree in in-order traversal."""
        if node is None:
            return
        self._in_order_print(node.left)
        print(f"{node.word}: {node.count}")
        self._in_order_print(node.right)
