def binary_search():
    # Get user input and ensure it's an integer
    while True:
        try:
            target = int(input("Enter an integer to search: "))
            break
        except ValueError:
            print("Invalid input. Please enter an integer.")

    # Read and process the list of numbers from the file
    with open('numberlist.txt', 'r') as file:
        number_list = file.read().split(", ")
    
    # Convert the string list to a list of integers and sort it
    numbers = sorted(map(int, number_list))

    # Initialize binary search boundaries
    low, high = 0, len(numbers) - 1

    # Perform binary search
    while low <= high:
        mid = (low + high) // 2
        if numbers[mid] == target:
            print(f"Target {target} found in the list!")
            break
        elif numbers[mid] < target:
            low = mid + 1
        else:
            high = mid - 1
    else:
        print(f"Target {target} not found in the list.")

    # Ask the user if they want to search again
    while True:
        retry = input("Would you like to search for another number? (y/n): ").strip().lower()
        if retry == 'y':
            binary_search()  # Recursively call binary_search for a new search
            break
        elif retry == 'n':
            print("Exiting program. Goodbye!")
            break
        else:
            print("Invalid input. Please enter 'y' for yes or 'n' for no.")

# Start the binary search
binary_search()
