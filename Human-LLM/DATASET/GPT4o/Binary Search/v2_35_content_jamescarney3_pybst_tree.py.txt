class BinarySearchTreeNode:
    def __init__(self, value):
        self.value = value
        self.left = None  # Left child node
        self.right = None  # Right child node

    def insert(self, value):
        # Insert value into the appropriate position in the tree
        if value < self.value:
            # If value is less than current node's value, go left
            if self.left is None:
                self.left = BinarySearchTreeNode(value)  # Insert here if no left child
            else:
                self.left.insert(value)  # Recurse to left child
        elif value > self.value:
            # If value is greater than current node's value, go right
            if self.right is None:
                self.right = BinarySearchTreeNode(value)  # Insert here if no right child
            else:
                self.right.insert(value)  # Recurse to right child

    def lookup(self, value):
        # Search for a node with the given value
        if value == self.value:
            return self  # Node found
        elif value < self.value and self.left:
            return self.left.lookup(value)  # Recurse to left child
        elif value > self.value and self.right:
            return self.right.lookup(value)  # Recurse to right child
        return None  # Node not found


class BinarySearchTree:
    def __init__(self):
        self.root = None  # Start with an empty tree

    def insert(self, value):
        if self.root is None:
            self.root = BinarySearchTreeNode(value)  # The first node becomes the root
        else:
            self.root.insert(value)  # Insert into the existing tree

    def lookup(self, value):
        if self.root is None:
            return None  # Empty tree case
        return self.root.lookup(value)  # Start searching from the root


# Example usage
bst = BinarySearchTree()

# Insert values into the BST
bst.insert(10)
bst.insert(5)
bst.insert(15)
bst.insert(3)
bst.insert(7)
bst.insert(12)
bst.insert(18)

# Lookup nodes and display results
node = bst.lookup(7)
if node:
    print(f"Node with value {node.value} found in the tree.")
else:
    print("Value not found in the tree.")
