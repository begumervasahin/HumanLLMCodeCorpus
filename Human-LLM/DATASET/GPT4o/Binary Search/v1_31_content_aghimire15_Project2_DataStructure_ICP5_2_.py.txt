import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split
from sklearn import linear_model
from sklearn.metrics import mean_squared_error

# Load the dataset
train = pd.read_csv('winequality-red.csv')

# Select numerical features and handle missing data
data = train.select_dtypes(include=[np.number]).interpolate().dropna()

# Prepare target variable (log-transformed quality)
y = np.log(train.quality)

# Prepare feature matrix by dropping the target column
X = data.drop(['quality'], axis=1)

# Split the data into training and testing sets
X_train, X_test, y_train, y_test = train_test_split(X, y, random_state=42, test_size=0.33)

# Initialize and fit the linear regression model
lr = linear_model.LinearRegression()
model = lr.fit(X_train, y_train)

# Print the R^2 score of the model
print("R^2 is: \n", model.score(X_test, y_test))

# Generate predictions on the test set
predictions = model.predict(X_test)

# Calculate and print the Root Mean Squared Error (RMSE)
print('RMSE is: \n', mean_squared_error(y_test, predictions))

# Calculate and print the correlation matrix for numerical features
numeric_features = train.select_dtypes(include=[np.number])
corr = numeric_features.corr()
print(corr['quality'].sort_values(ascending=False)[:5], '\n')

# Display the number of nulls in each feature
nulls = pd.DataFrame(train.isnull().sum().sort_values(ascending=False)[:25])
nulls.columns = ['Null Count']
nulls.index.name = 'Feature'
print(nulls)
