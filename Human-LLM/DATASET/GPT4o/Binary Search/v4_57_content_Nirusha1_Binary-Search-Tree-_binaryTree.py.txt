class BinarySearchTree:
    def __init__(self):
        """Initialize an empty binary search tree."""
        self._size = 0
        self._root = None

    class _BSTNode:
        def __init__(self, key, value):
            """Initialize a node with a key-value pair."""
            self.key = key
            self.value = value
            self.left = None
            self.right = None

    def add(self, key, value):
        """Add a key-value pair to the binary search tree."""
        new_node = self._BSTNode(key, value)
        parent = None
        current = self._root
        
        while current is not None:
            parent = current
            if key < current.key:
                current = current.left
            else:
                current = current.right
        
        if parent is None:
            self._root = new_node
        elif key < parent.key:
            parent.left = new_node
        else:
            parent.right = new_node
        
        self._size += 1

    def smallest(self):
        """Return the smallest key in the binary search tree."""
        if self._root is not None:
            current = self._root
            while current.left is not None:
                current = current.left
            return current.key
        return None

    def largest(self):
        """Return the largest key in the binary search tree."""
        if self._root is not None:
            current = self._root
            while current.right is not None:
                current = current.right
            return current.key
        return None

    def search(self, key):
        """Search for a key in the binary search tree."""
        found = []
        self._search(self._root, key, found)
        return found

    def _search(self, subtree, key, found):
        """Recursive helper method to search for a key."""
        if subtree:
            if key == subtree.key:
                found.append(subtree.value)
            elif key < subtree.key:
                self._search(subtree.left, key, found)
            else:
                self._search(subtree.right, key, found)

    def is_empty(self):
        """Check if the binary search tree is empty."""
        return self._size == 0

    def size(self):
        """Return the number of nodes in the binary search tree."""
        return self._size

    def inorder_walk(self):
        """Return a list of keys in inorder traversal."""
        nodes = []
        self._inorder_walk(self._root, nodes)
        return nodes

    def _inorder_walk(self, subtree, nodes):
        """Recursive helper method for inorder traversal."""
        if subtree:
            self._inorder_walk(subtree.left, nodes)
            nodes.append(subtree.key)
            self._inorder_walk(subtree.right, nodes)

    def preorder_walk(self):
        """Return a list of keys in preorder traversal."""
        nodes = []
        self._preorder_walk(self._root, nodes)
        return nodes

    def _preorder_walk(self, subtree, nodes):
        """Recursive helper method for preorder traversal."""
        if subtree:
            nodes.append(subtree.key)
            self._preorder_walk(subtree.left, nodes)
            self._preorder_walk(subtree.right, nodes)

    def postorder_walk(self):
        """Return a list of keys in postorder traversal."""
        nodes = []
        self._postorder_walk(self._root, nodes)
        return nodes

    def _postorder_walk(self, subtree, nodes):
        """Recursive helper method for postorder traversal."""
        if subtree:
            self._postorder_walk(subtree.left, nodes)
            self._postorder_walk(subtree.right, nodes)
            nodes.append(subtree.key)
