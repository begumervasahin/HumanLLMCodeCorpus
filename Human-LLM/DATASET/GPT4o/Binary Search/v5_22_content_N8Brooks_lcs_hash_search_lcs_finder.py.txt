import sys
import serial_hash_search
import parallel_hash_search

# Toggle for choosing between parallel and serial LCS computation
USE_PARALLEL = False

def read_text(file_name):
    """
    Reads the content of a text file and returns it as a string.
    
    Args:
        file_name (str): The path to the text file.
        
    Returns:
        str: The content of the file as a single string.
    """
    try:
        with open(file_name, 'r', encoding='utf-8') as file:
            return file.read()
    except FileNotFoundError:
        print(f"Error: The file '{file_name}' was not found.")
        sys.exit(1)

def print_lcs_results(length, substrings):
    """
    Prints the results of the longest common substring (LCS) computation.
    
    Args:
        length (int): The length of the LCS.
        substrings (list): A list of all LCS found.
    """
    # Determine the appropriate grammar for output
    verb = 'is' if len(substrings) == 1 else 'are'
    plural_suffix = '' if len(substrings) == 1 else 's'
    
    print(f"The length of the longest common substring is: {length}")
    print(f"There {verb} {len(substrings)} longest common substring{plural_suffix}:")
    for substring in substrings:
        print(f'"{substring}"')

def main():
    """
    The main function for computing and printing the longest common substring (LCS).
    
    Command line arguments:
        str: The path to the first text file for LCS computation.
        str: The path to the second text file for LCS computation.
        
    Prints:
        int: The length of the longest common substring (LCS).
        list: A list of all longest common substrings.
    """
    if len(sys.argv) != 3:
        print("Usage: python script_name.py <file_path_1> <file_path_2>")
        sys.exit(1)

    file_a = read_text(sys.argv[1])
    file_b = read_text(sys.argv[2])

    # Choose the LCS function based on the USE_PARALLEL flag
    lcs_function = parallel_hash_search.lcs if USE_PARALLEL else serial_hash_search.lcs
    length, substrings = lcs_function(file_a, file_b)

    # Print the results
    print_lcs_results(length, substrings)

if __name__ == '__main__':
    main()
