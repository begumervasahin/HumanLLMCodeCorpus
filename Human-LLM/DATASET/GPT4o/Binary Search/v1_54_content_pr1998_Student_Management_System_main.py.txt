import time
from bst import BST  # Assuming bst.py contains the implementation of the BST class

class Student:
    def __init__(self, roll_num, name, address, mobile, course, marks):
        self.roll_num = roll_num
        self.name = name
        self.address = address
        self.mobile = mobile
        self.course = course
        self.marks = marks
        self.total = sum(marks)
        self.percent = self.total / len(marks)

    def __str__(self):
        return (f"Roll Number: {self.roll_num}, Name: {self.name}, Address: {self.address}, "
                f"Mobile: {self.mobile}, Course: {self.course}, Total Marks: {self.total}, "
                f"Percentage: {self.percent:.2f}%")

def main():
    n = int(input("ENTER THE NUMBER OF STUDENTS: \n"))
    b = BST()
    
    for i in range(n):
        print(f"\nENTER THE DETAILS OF STUDENT {i+1}:")
        roll_num = int(input("Enter the roll number: "))
        name = input("Enter name: ")
        address = input("Enter the address: ")
        mobile = int(input("Enter the mobile number: "))
        course = input("Enter the course: ")
        
        print('Enter the marks in three subjects:')
        marks = []
        marks.append(int(input("First subject: ")))
        marks.append(int(input("Second subject: ")))
        marks.append(int(input("Third subject: ")))
        
        student = Student(roll_num, name, address, mobile, course, marks)
        print("Total Marks: ", student.total)
        print("Average marks: ", student.percent)
        
        b.insert(roll_num, student)
    
    print("\nBST TRAVERSAL\n")
    b.inorder()
    
    print("DO YOU WANT TO DELETE ANY KEYS IN THE TREE?")
    ch = input("IF YES, ENTER y OR Y: ")
    
    if ch.lower() == 'y':
        while ch.lower() == 'y':
            num = int(input("Enter the number to be deleted: "))
            b.remove(num)
            ch = input("Enter y or Y to continue: ")
        
        print("AFTER THE DELETION:\n")
        b.inorder()
    
    print("\nDO YOU WANT TO SEARCH ANY KEYS IN THE TREE?")
    ch = input("IF YES, ENTER y OR Y: ")
    
    while ch.lower() == 'y':
        num = int(input("Enter the key to be searched: "))
        result = b.search(num)
        if result:
            print(f"Student found: {result}")
        else:
            print("Student not found.")
        ch = input("Enter y or Y to continue: ")
    
    print("\n*************************END OF BST OPERATIONS*******************************\n")

if __name__ == '__main__':
    main()
