import csv
from random import shuffle

class Tree:
    """Represents a node in a binary search tree."""
    def __init__(self, value=None):
        self.value = value
        self.left = None
        self.right = None

    def insert(self, x):
        """Inserts a new value into the binary search tree."""
        if self.value is None:
            self.value = x
        else:
            self._insert_recursive(self, x)

    @staticmethod
    def _insert_recursive(current, x):
        """Recursively inserts a value into the correct position in the BST."""
        if current is None:
            return Tree(x)
        elif x < current.value:
            current.left = Tree._insert_recursive(current.left, x)
        else:
            current.right = Tree._insert_recursive(current.right, x)
        return current

    def height(self):
        """Calculates the height of the binary search tree."""
        if self.value is None:
            return 0
        return self._calculate_height(self)

    @staticmethod
    def _calculate_height(node):
        """Helper method to calculate the height of the tree."""
        if node is None:
            return 0
        return max(Tree._calculate_height(node.left), Tree._calculate_height(node.right)) + 1

def generate_random_array(n):
    """Generates a shuffled array of numbers from 0 to n-1."""
    array = list(range(n))
    shuffle(array)
    return array

def build_tree(arr):
    """Builds a binary search tree from an array of values."""
    tree = Tree()
    for value in arr:
        tree.insert(value)
    return tree

def calculate_average_height(n, trials=500):
    """Calculates the average height of binary search trees built from random arrays."""
    total_height = 0
    for _ in range(trials):
        arr = generate_random_array(n)
        tree = build_tree(arr)
        total_height += tree.height()
    
    return total_height / trials

def save_results_to_csv(results, filename="results.csv"):
    """Saves the results to a CSV file."""
    with open(filename, "w+", newline='') as csv_file:
        writer = csv.writer(csv_file)
        writer.writerows(results)

def main():
    print("n is the number of nodes in the Binary Search Tree")
    print("h is the corresponding average height of 500 randomly generated samples\n")

    results = [["Number of Nodes", "Average Height"]]
    print("Generating results...")

    for n in range(100, 1001, 100):
        average_height = calculate_average_height(n)
        results.append([n, average_height])
        print(f"n = {n:4d}, h = {average_height:.3f}")

    save_results_to_csv(results)

if __name__ == "__main__":
    main()
