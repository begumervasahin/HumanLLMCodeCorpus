class Patient:
    def __init__(self, name, contact, age, days, balance):
        """Initialize a Patient object with name, contact, age, days, and balance."""
        self.name = name
        self.contact = contact
        self.age = age
        self.days = days
        self.balance = balance

    def get_name(self):
        """Return the name of the patient."""
        return self.name

    def get_contact(self):
        """Return the contact information of the patient."""
        return self.contact

    def get_age(self):
        """Return the age of the patient."""
        return self.age

    def get_days(self):
        """Return the number of days the patient has been admitted."""
        return self.days

    def get_balance(self):
        """Return the balance owed by the patient."""
        return self.balance

    def update_name(self, name):
        """Update the patient's name."""
        self.name = name

    def update_contact(self, contact):
        """Update the patient's contact information."""
        self.contact = contact

    def update_age(self, age):
        """Update the patient's age."""
        self.age = age

    def update_days(self, days):
        """Update the number of days the patient has been admitted."""
        self.days = days

    def update_balance(self, balance):
        """Update the patient's balance."""
        self.balance = balance

# Initialize an empty list to store Patient objects
patient_list = []

# Open the file containing patient information and read it
with open("Patient_Info-sorted.dat", "r") as fp:
    fp.readline()  # Skip the header line
    for line in fp:
        line = line.strip()
        if not line:
            break
        word = line.split()
        # Create a Patient object and add it to the patient list
        patient = Patient(
            name=str(word[0]),
            contact=str(word[1]),
            age=int(word[2]),
            days=int(word[3]),
            balance=float(word[4])
        )
        patient_list.append(patient)

# Perform a binary search on the patient list
def binary_search_patient(patient_list, target_name):
    """Perform a binary search for the target name in the patient list."""
    first = 0
    last = len(patient_list) - 1
    count = 0

    while first <= last:
        count += 1
        mid = (first + last) // 2
        mid_name = patient_list[mid].get_name()

        if target_name > mid_name:
            first = mid + 1
        elif target_name < mid_name:
            last = mid - 1
        else:
            # Patient found
            print("Patient found at index", mid)
            print("Number of iterations:", count)
            print("Patient contact:", patient_list[mid].get_contact())
            print("Patient age:", patient_list[mid].get_age())
            print("Patient days admitted:", patient_list[mid].get_days())
            print("Patient balance:", patient_list[mid].get_balance())
            return True

    # Patient not found
    print("Patient not found.")
    return False

# Prompt the user to enter a patient name for searching
target_name = input("Enter a patient name to search: ")
binary_search_patient(patient_list, target_name)
