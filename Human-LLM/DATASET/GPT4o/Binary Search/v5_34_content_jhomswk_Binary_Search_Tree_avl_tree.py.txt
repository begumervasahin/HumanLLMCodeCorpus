from binary_search_tree import Binary_Search_Tree

def height(node):
    """
    Returns the height of the tree rooted at the given node.
    """
    return node.height if node else -1

def update_height(node):
    """
    Updates the height of the tree rooted at the given node.
    """
    node.height = 1 + max(height(node.left), height(node.right))

class AVLTree(Binary_Search_Tree):
    """
    AVL Tree implementation, inheriting from the Binary_Search_Tree class.
    Supports self-balancing operations.
    """

    def rotate_left(self, node):
        """
        Left-rotates the subtree rooted at the given node.
        """
        right_child = node.right
        right_child.parent = node.parent

        # Update the root if the node is the root
        if node.parent is None:
            self.root = right_child
        else:
            if node is node.parent.left:
                node.parent.left = right_child
            else:
                node.parent.right = right_child

        # Perform the left rotation
        node.right = right_child.left
        if node.right:
            node.right.parent = node
        right_child.left = node
        node.parent = right_child

        # Update heights after rotation
        update_height(node)
        update_height(right_child)

    def rotate_right(self, node):
        """
        Right-rotates the subtree rooted at the given node.
        """
        left_child = node.left
        left_child.parent = node.parent

        # Update the root if the node is the root
        if node.parent is None:
            self.root = left_child
        else:
            if node is node.parent.left:
                node.parent.left = left_child
            else:
                node.parent.right = left_child

        # Perform the right rotation
        node.left = left_child.right
        if node.left:
            node.left.parent = node
        left_child.right = node
        node.parent = left_child

        # Update heights after rotation
        update_height(node)
        update_height(left_child)

    def balance(self, node):
        """
        Balances the AVL tree starting from the given node up to the root.
        """
        while node:
            update_height(node)
            balance_factor = height(node.left) - height(node.right)

            # Left heavy subtree
            if balance_factor > 1:
                if height(node.left.left) >= height(node.left.right):
                    self.rotate_right(node)
                else:
                    self.rotate_left(node.left)
                    self.rotate_right(node)

            # Right heavy subtree
            elif balance_factor < -1:
                if height(node.right.right) >= height(node.right.left):
                    self.rotate_left(node)
                else:
                    self.rotate_right(node.right)
                    self.rotate_left(node)

            node = node.parent

    def insert(self, key):
        """
        Inserts a node with the given key and rebalances the tree.
        """
        new_node = super(AVLTree, self).insert(key)
        self.balance(new_node)

    def delete(self, key):
        """
        Deletes the node with the given key and rebalances the tree.
        """
        node_to_delete = super(AVLTree, self).delete(key)
        if node_to_delete:
            self.balance(node_to_delete.parent)
