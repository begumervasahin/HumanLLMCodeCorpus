class Node:
    def __init__(self, val):
        self.val = val
        self.left = None
        self.right = None

def in_order_print(root):
    """Performs in-order traversal and prints the values."""
    if not root:
        return
    in_order_print(root.left)
    print(root.val)
    in_order_print(root.right)

def pre_order_print(root):
    """Performs pre-order traversal and prints the values."""
    if not root:
        return
    print(root.val)
    pre_order_print(root.left)
    pre_order_print(root.right)

def post_order_print(root):
    """Performs post-order traversal and prints the values."""
    if not root:
        return
    post_order_print(root.left)
    post_order_print(root.right)
    print(root.val)

def level_print(root):
    """Performs level-order traversal and prints the values by level."""
    if not root:
        return
    
    level = [root]
    counter = 1
    
    while level:
        next_level = []
        print(f'Level {counter}:')
        for node in level:
            print(node.val)
            if node.left:
                next_level.append(node.left)
            if node.right:
                next_level.append(node.right)
        level = next_level
        counter += 1

# Example usage
if __name__ == "__main__":
    # Construct the test binary tree
    test = Node(8)
    test.left = Node(3)
    test.right = Node(10)
    test.left.left = Node(1)
    test.left.right = Node(6)
    test.right.right = Node(14)
    test.right.right.left = Node(13)
    test.left.right.left = Node(4)
    test.left.right.right = Node(7)
    
    print('In-Order Traversal and print:')
    in_order_print(test)
    
    print('\nPre-Order Traversal and print:')
    pre_order_print(test)
    
    print('\nPost-Order Traversal and print:')
    post_order_print(test)
    
    print('\nLevel Traversal and print:')
    level_print(test)
