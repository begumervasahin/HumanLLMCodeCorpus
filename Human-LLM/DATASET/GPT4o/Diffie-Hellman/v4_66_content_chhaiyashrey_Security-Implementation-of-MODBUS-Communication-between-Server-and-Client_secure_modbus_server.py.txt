import logging
from socketserver import TCPServer
from collections import defaultdict
from umodbus import conf
from umodbus.server.tcp import RequestHandler, get_server
from umodbus.utils import log_to_stream

def setup_logging():
    """Set up logging for the Modbus server."""
    log_to_stream(level=logging.DEBUG)

def create_data_store():
    """Initialize a data store with default integer values."""
    return defaultdict(int)

def configure_umodbus():
    """Configure umodbus to use signed values."""
    conf.SIGNED_VALUES = True

def create_modbus_server():
    """Create and configure the Modbus TCP server."""
    TCPServer.allow_reuse_address = True
    return get_server(TCPServer, ('', 502), RequestHandler)

def define_routes(app, data_store):
    """Define routes for reading from and writing to the data store."""

    @app.route(slave_ids=[1], function_codes=[1, 2], addresses=list(range(0, 10)))
    def read_data_store(slave_id, function_code, address):
        """
        Return the value of the specified address.

        Parameters:
        slave_id (int): The Modbus slave ID.
        function_code (int): The Modbus function code.
        address (int): The address to read from.

        Returns:
        int: The value at the specified address.
        """
        return data_store[address]

    @app.route(slave_ids=[1], function_codes=[5, 15], addresses=list(range(0, 10)))
    def write_data_store(slave_id, function_code, address, value):
        """
        Set the value for the specified address.

        Parameters:
        slave_id (int): The Modbus slave ID.
        function_code (int): The Modbus function code.
        address (int): The address to write to.
        value (int): The value to write.

        Returns:
        None
        """
        data_store[address] = value

def run_server(app):
    """Run the Modbus TCP server."""
    try:
        app.serve_forever()
    finally:
        app.shutdown()
        app.server_close()

def main():
    """Main function to set up and run the Modbus TCP server."""
    setup_logging()
    data_store = create_data_store()
    configure_umodbus()
    app = create_modbus_server()
    define_routes(app, data_store)
    run_server(app)

if __name__ == '__main__':
    main()
