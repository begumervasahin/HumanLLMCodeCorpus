def mod_exp(x, y, N):
    """
    Perform modular exponentiation.
    
    Parameters:
    x (int): Base.
    y (int): Exponent.
    N (int): Modulus.
    
    Returns:
    int: Result of (x^y) % N.
    """
    if y == 0:
        return 1
    z = mod_exp(x, y // 2, N)
    if y % 2 == 0:
        return z ** 2 % N
    else:
        return (x * z ** 2) % N

def truncate_int(integer, digits_to_truncate):
    """
    Truncate an integer by removing a specified number of digits from the end.
    
    Parameters:
    integer (int): The integer to truncate.
    digits_to_truncate (int): The number of digits to remove.
    
    Returns:
    int: The truncated integer.
    """
    if digits_to_truncate > 0:
        truncating_divisor = 10 ** digits_to_truncate
        integer = integer // truncating_divisor
    return integer

def find_int_truncate_length(smaller_size_int, larger_size_int):
    """
    Find the number of digits to truncate to make the two integers of similar length.
    
    Parameters:
    smaller_size_int (int): The smaller integer.
    larger_size_int (int): The larger integer.
    
    Returns:
    int: The number of digits to truncate.
    """
    return len(str(larger_size_int)) - len(str(smaller_size_int))

def main():
    # Input values
    p = int(input("Enter the prime number for modulus (p): "))
    a = int(input("Enter the secret exponent (a): "))
    g = int(input("Enter the generator (g): "))
    
    # Generate g^a (mod p)
    g_pow_a_mod_p = mod_exp(g, a, p)
    print("Generating value for key exchange...")
    print(f"g^a (mod p) = {g_pow_a_mod_p}")
    
    # Input shared key from server
    g_pow_b_mod_p = int(input("Enter the shared key from the server: "))
    
    # Compute shared value
    shared_value = mod_exp(g_pow_b_mod_p, a, p)
    
    # Truncate shared value
    truncate_length = int(input("Enter the number of digits to truncate from the shared value: "))
    truncated_shared_value = truncate_int(shared_value, truncate_length)
    
    # Display the shared value
    print(f"Shared value is: {truncated_shared_value}")

if __name__ == '__main__':
    main()
