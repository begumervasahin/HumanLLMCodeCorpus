from math import gcd as bltin_gcd

def is_prime(n):
    """Check if a number is prime."""
    if n <= 1:
        return False
    if n <= 3:
        return True
    if n % 2 == 0 or n % 3 == 0:
        return False
    i = 5
    while i * i <= n:
        if n % i == 0 or n % (i + 2) == 0:
            return False
        i += 6
    return True

def primitive_roots(prime):
    """Find primitive roots of a prime number."""
    if not is_prime(prime):
        return []

    # Set of integers from 1 to prime-1
    required_set = {num for num in range(1, prime) if bltin_gcd(num, prime) == 1}
    
    # List of primitive roots
    roots = []
    for g in range(1, prime):
        # Check if g is a primitive root
        if required_set == {pow(g, powers, prime) for powers in range(1, prime)}:
            roots.append(g)
    
    return roots

def main():
    print('Enter a prime number: ')
    x = int(input())
    
    if is_prime(x):
        print('Checked: Number is a PRIME')
        print('Primitive root/roots:')
        roots = primitive_roots(x)
        if roots:
            print(roots)
        else:
            print('No primitive roots found')
    else:
        print('Warning: Number is not a PRIME')

if __name__ == '__main__':
    main()
