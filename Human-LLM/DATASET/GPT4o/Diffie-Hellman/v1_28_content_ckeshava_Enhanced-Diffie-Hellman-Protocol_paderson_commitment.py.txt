from Crypto.Util import number
import random
import constants

def generate_p_g_q(length_of_message=0):
    """
    Generates parameters for a cryptographic commitment scheme.

    Args:
        length_of_message (int): The length of the message to be committed.

    Returns:
        tuple: Contains the values of p, g, and q.
    """
    q = number.getPrime(max(constants.LENGTH_OF_PRIME, length_of_message + 1))
    p = 2 * q + 1
    g = 2
    return p, g, q

def generate_a():
    """
    Generates a random integer within the specified bounds.

    Returns:
        int: A random integer between RAND_NUMBER_LOWER_BOUND and RAND_NUMBER_UPPER_BOUND.
    """
    return random.randint(constants.RAND_NUMBER_LOWER_BOUND, constants.RAND_NUMBER_UPPER_BOUND)

def generate_h(g, p, a):
    """
    Computes a value h used in the commitment scheme.

    Args:
        g (int): The base value for the commitment.
        p (int): The prime modulus for the commitment.
        a (int): The exponent for the computation.

    Returns:
        int: The computed value h.
    """
    return pow(g, a, p)

def generate_r(q):
    """
    Generates a random integer for the commitment process.

    Args:
        q (int): The upper bound for the random integer.

    Returns:
        int: A random integer between 0 and q-1.
    """
    return random.randint(0, q - 1)

def commit(m, g, h, r, p):
    """
    Computes the commitment of a message using the given parameters.

    Args:
        m (int): The message to be committed.
        g (int): The base value for the commitment.
        h (int): The computed value used in the commitment.
        r (int): The random integer used in the commitment.
        p (int): The prime modulus for the commitment.

    Returns:
        int: The computed commitment.
    """
    t1 = pow(g, m, p)
    t2 = pow(h, r, p)
    return (t1 * t2) % p

def open(m, g, h, r, p, c):
    """
    Verifies the commitment and returns the message if valid.

    Args:
        m (int): The original message.
        g (int): The base value used in the commitment.
        h (int): The computed value used in the commitment.
        r (int): The random integer used in the commitment.
        p (int): The prime modulus for the commitment.
        c (int): The commitment to be verified.

    Returns:
        int or None: The original message if the commitment is valid; otherwise, None.
    """
    t1 = pow(g, m, p)
    t2 = pow(h, r, p)
    c_ = (t1 * t2) % p
    if c_ == c:
        return m
    else:
        return None
