def main():
    """
    Cryptanalysis of the Diffie-Hellman public key protocol.
    """
    print()
    print("---------------------------------------------------------------")
    print("    Cryptanalysis of the Diffie-Hellman public key protocol    ")
    print("---------------------------------------------------------------")
    print()

    # Confirm the prime and generator
    p = int(input("Confirm the prime: "))
    g = int(input("Confirm the generator: "))
    print()

    # Get the public keys sent by A and B
    A = int(input("A sent: "))
    B = int(input("B sent: "))
    print()

    # Find the private keys a and b
    a = b = None
    for x in range(1, p):
        if pow(g, x, p) == A:
            a = x
        if pow(g, x, p) == B:
            b = x

    if a is None or b is None:
        print("Failed to find private keys.")
        return

    print("Password for A:", a)
    print("Password for B:", b)
    print()

    # Calculate the common secret key
    k_a = pow(B, a, p)
    k_b = pow(A, b, p)
    k = k_a
    print("Their secret common key:", k)
    print()

if __name__ == "__main__":
    main()
