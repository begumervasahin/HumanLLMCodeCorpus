import sys
import re

def pow_x(g_base, exponent, modulus):
    """
    Compute g_base^exponent % modulus using an efficient method.
    
    Args:
        g_base (int): The base value.
        exponent (int): The exponent.
        modulus (int): The modulus value.
    
    Returns:
        int: Result of (g_base^exponent) % modulus.
    """
    result = 1
    bits = bin(exponent)[2:]  # Convert the exponent to binary
    for bit in bits:
        result = (result ** 2) % modulus
        if bit == '1':
            result = (result * g_base) % modulus
    return result

def attack(modulus, base, ga_value):
    """
    Attack the Diffie-Hellman exchange to find the secret exponent.
    
    Args:
        modulus (int): The prime modulus.
        base (int): The base.
        ga_value (int): The result of base^a % modulus.
    
    Returns:
        int: The secret exponent 'a' if found, otherwise 0.
    """
    for possible_a in range(modulus):
        if pow_x(base, possible_a, modulus) == ga_value:
            return possible_a
    return 0

def parse_input_file(file_path):
    """
    Parse the input file to extract modulus, base, and ga_value.
    
    Args:
        file_path (str): The path to the input file.
    
    Returns:
        tuple: A tuple containing the modulus (p), base (g), and ga_value (g^a % p).
    """
    with open(file_path, 'r') as file:
        content = file.readline().strip()
        p_str, g_str, ga_str = content.split(',')

        modulus = int(re.findall(r'\d+', p_str)[0])
        base = int(re.findall(r'\d+', g_str)[0])
        ga_value = int(re.findall(r'\d+', ga_str)[0])
    
    return modulus, base, ga_value

def main():
    """
    Main function to read input from a file and find the secret exponent.
    """
    if len(sys.argv) != 2:
        print("Usage: python script.py <input_file>")
        sys.exit(1)

    input_file = sys.argv[1]
    modulus, base, ga_value = parse_input_file(input_file)
    
    secret_exponent = attack(modulus, base, ga_value)
    
    if secret_exponent:
        print(f"The secret exponent is: {secret_exponent}")
    else:
        print("Exponent not found.")

if __name__ == '__main__':
    main()
