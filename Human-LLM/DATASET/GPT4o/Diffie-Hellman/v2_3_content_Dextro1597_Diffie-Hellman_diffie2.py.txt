def main():
    """
    Main function to perform Diffie-Hellman key exchange.
    """
    # Input values
    shared_prime = int(input("Enter the value of the shared prime (p): "))
    shared_base = int(input("Enter the value of the shared base (q): "))
    alice_secret = int(input("Enter Alice's secret key value (a): "))
    bob_secret = int(input("Enter Bob's secret key value (b): "))

    # Print shared variables
    print("\nShared Variables:")
    print(f"Shared Prime number: {shared_prime}")
    print(f"Shared Base number: {shared_base}")

    # Alice computes her public value to send over an insecure channel
    alice_public_value = pow(shared_base, alice_secret, shared_prime)
    print(f"\nAlice sends value over insecure channel: {alice_public_value}")

    # Bob computes his public value to send over an insecure channel
    bob_public_value = pow(shared_base, bob_secret, shared_prime)
    print(f"Bob sends value over insecure channel: {bob_public_value}")

    # Alice computes the shared secret using Bob's public value
    alice_shared_secret = pow(bob_public_value, alice_secret, shared_prime)
    print(f"\nValue computed by Alice: {alice_shared_secret}")

    # Bob computes the shared secret using Alice's public value
    bob_shared_secret = pow(alice_public_value, bob_secret, shared_prime)
    print(f"Value computed by Bob: {bob_shared_secret}")

if __name__ == "__main__":
    main()
