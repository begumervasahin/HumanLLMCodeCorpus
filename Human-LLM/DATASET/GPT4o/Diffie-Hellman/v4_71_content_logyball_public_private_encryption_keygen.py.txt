from random import randint, seed, getrandbits
from millerrabin import isPrime

GENERATOR = 2

def fake_32_bit_int(num):
    """
    Simulate a 32-bit integer by taking modulo 4294967295.

    Parameters:
    num (int): The number to be converted.

    Returns:
    int: The simulated 32-bit integer.
    """
    return num % 4294967295

def generate_prime(bit_length=31):
    """
    Generate a prime number of specified bit length.

    Parameters:
    bit_length (int): The bit length of the prime number to be generated.

    Returns:
    int: The generated prime number.
    """
    while True:
        random_num = getrandbits(bit_length)
        if random_num % 12 != 5:
            continue
        prime_candidate = (2 * random_num) + 1
        if isPrime(prime_candidate) and prime_candidate > 2147483648:
            return prime_candidate

def generate_private_key(user_seed, prime):
    """
    Generate a private key based on a user-provided seed and a prime number.

    Parameters:
    user_seed (any hashable type): The seed for the random number generator.
    prime (int): The prime number used to bound the private key.

    Returns:
    int: The generated private key.
    """
    seed(user_seed)
    return randint(1, prime - 2)

def compute_e2(private_key, prime, generator=GENERATOR):
    """
    Compute e2 using the private key, prime, and generator.

    Parameters:
    private_key (int): The private key.
    prime (int): The prime number.
    generator (int): The generator.

    Returns:
    int: The computed e2 value.
    """
    return pow(generator, private_key, prime)

def generate_keys(user_seed=None):
    """
    Generate a set of cryptographic keys.

    Parameters:
    user_seed (any hashable type, optional): The seed for the random number generator.

    Returns:
    dict: A dictionary containing the private key (d), prime (p), generator (g), and e2 value (e2).
    """
    prime = generate_prime()
    private_key = generate_private_key(user_seed, prime)
    e2 = compute_e2(private_key, prime)
    return {
        'd': private_key,
        'p': prime,
        'g': GENERATOR,
        'e2': e2
    }

if __name__ == '__main__':
    # Example usage
    keys = generate_keys(user_seed='user-defined-seed')
    print("Generated Keys:")
    print(f"Private Key (d): {keys['d']}")
    print(f"Prime (p): {keys['p']}")
    print(f"Generator (g): {keys['g']}")
    print(f"e2: {keys['e2']}")
