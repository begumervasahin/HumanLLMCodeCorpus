def get_encoding():
    """
    Create encoding and decoding dictionaries for the custom cipher.

    Returns:
    tuple: A tuple containing two dictionaries: 
           - encoding_dict (dict): Character to number encoding dictionary.
           - decoding_dict (dict): Number to character decoding dictionary.
    """
    encoding_dict = {}
    decoding_dict = {}

    # Space character
    encoding_dict[' '] = 0
    decoding_dict[0] = ' '

    # Uppercase letters
    for i in range(65, 65 + 26):
        encoding_dict[chr(i)] = i - 64
        decoding_dict[i - 64] = chr(i)

    # Special characters
    special_chars = {',': 27, '.': 28, '?': 29}
    for char, code in special_chars.items():
        encoding_dict[char] = code
        decoding_dict[code] = char

    # Numbers
    for i in range(48, 48 + 10):
        encoding_dict[chr(i)] = i - 18
        decoding_dict[i - 18] = chr(i)

    # Lowercase letters
    for i in range(97, 97 + 26):
        encoding_dict[chr(i)] = i - 57
        decoding_dict[i - 57] = chr(i)

    # Exclamation mark
    encoding_dict['!'] = 66
    decoding_dict[66] = '!'

    return encoding_dict, decoding_dict

def encrypt(text, key):
    """
    Encrypt a string using the custom cipher with the provided key.

    Parameters:
    text (str): The string to encrypt.
    key (int): The encryption key.

    Returns:
    str: The encrypted string, or -1 if an invalid character is found.
    """
    encoding_dict, decoding_dict = get_encoding()
    encrypted_output = ""

    for char in text:
        if char not in encoding_dict:
            return -1
        else:
            current_value = encoding_dict[char]
            encrypted_value = (current_value + key) % 67
            encrypted_output += decoding_dict[encrypted_value]

    return encrypted_output

def decrypt(text, key):
    """
    Decrypt a string using the custom cipher with the provided key.

    Parameters:
    text (str): The string to decrypt.
    key (int): The decryption key.

    Returns:
    str: The decrypted string, or -1 if an invalid character is found.
    """
    encoding_dict, decoding_dict = get_encoding()
    decrypted_output = ""

    for char in text:
        if char not in encoding_dict:
            return -1
        else:
            current_value = encoding_dict[char]
            decrypted_value = (current_value - key) % 67
            decrypted_output += decoding_dict[decrypted_value]

    return decrypted_output

if __name__ == '__main__':
    # Example usage
    original_text = "Hello, World!"
    key = 5

    encrypted_text = encrypt(original_text, key)
    if encrypted_text != -1:
        print(f"Encrypted: {encrypted_text}")
    else:
        print("Error: Invalid character in input string for encryption.")

    decrypted_text = decrypt(encrypted_text, key)
    if decrypted_text != -1:
        print(f"Decrypted: {decrypted_text}")
    else:
        print("Error: Invalid character in input string for decryption.")
