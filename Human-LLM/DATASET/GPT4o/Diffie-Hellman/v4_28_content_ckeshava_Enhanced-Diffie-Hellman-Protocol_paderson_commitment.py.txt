from Crypto.Util import number
import random
import constants

def generate_p_g_q(length_of_message=0):
    """
    Generates cryptographic parameters p, g, and q for a commitment scheme.

    Args:
        length_of_message (int): The length of the message to be committed.

    Returns:
        tuple: Contains the values of p, g, and q.
    """
    # Generate a prime number q that is large enough
    q = number.getPrime(max(constants.LENGTH_OF_PRIME, length_of_message + 1))
    
    # Compute p as 2*q + 1
    p = 2 * q + 1
    
    # Choose g as 2 (a common choice)
    g = 2
    
    return p, g, q

def generate_a():
    """
    Generates a random integer within specified bounds.

    Returns:
        int: A random integer between RAND_NUMBER_LOWER_BOUND and RAND_NUMBER_UPPER_BOUND.
    """
    return random.randint(constants.RAND_NUMBER_LOWER_BOUND, constants.RAND_NUMBER_UPPER_BOUND)

def generate_h(g, p, a):
    """
    Computes the value h used in the commitment scheme.

    Args:
        g (int): The base value for the commitment.
        p (int): The prime modulus for the commitment.
        a (int): The exponent used in the computation.

    Returns:
        int: The computed value h.
    """
    return pow(g, a, p)

def generate_r(q):
    """
    Generates a random integer for use in the commitment process.

    Args:
        q (int): The upper bound for the random integer.

    Returns:
        int: A random integer between 0 and q-1.
    """
    return random.randint(0, q - 1)

def commit(message, g, h, random_value, p):
    """
    Computes the commitment of a message using given parameters.

    Args:
        message (int): The message to be committed.
        g (int): The base value for the commitment.
        h (int): The computed value used in the commitment.
        random_value (int): The random integer used in the commitment.
        p (int): The prime modulus for the commitment.

    Returns:
        int: The computed commitment.
    """
    # Compute t1 as g^message mod p
    t1 = pow(g, message, p)
    
    # Compute t2 as h^random_value mod p
    t2 = pow(h, random_value, p)
    
    # Calculate and return the final commitment
    return (t1 * t2) % p

def open(message, g, h, random_value, p, commitment):
    """
    Verifies the commitment and returns the message if valid.

    Args:
        message (int): The original message.
        g (int): The base value used in the commitment.
        h (int): The computed value used in the commitment.
        random_value (int): The random integer used in the commitment.
        p (int): The prime modulus for the commitment.
        commitment (int): The commitment to be verified.

    Returns:
        int or None: The original message if the commitment matches; otherwise, None.
    """
    # Recompute the commitment with the given values
    t1 = pow(g, message, p)
    t2 = pow(h, random_value, p)
    computed_commitment = (t1 * t2) % p
    
    # Check if the computed commitment matches the provided commitment
    if computed_commitment == commitment:
        return message
    return None
