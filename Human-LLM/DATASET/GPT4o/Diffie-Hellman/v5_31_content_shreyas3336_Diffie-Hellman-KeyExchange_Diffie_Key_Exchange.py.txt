from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives.asymmetric import dh
import Seed_Generator

def generate_diffie_hellman_parameters():
    """
    Generates Diffie-Hellman parameters.

    Returns:
        dh.DHParameters: The generated Diffie-Hellman parameters.
    """
    return dh.generate_parameters(generator=5, key_size=1024, backend=default_backend())

def generate_key_pair(parameters):
    """
    Generates a private key and its corresponding public key using the given parameters.

    Args:
        parameters (dh.DHParameters): The Diffie-Hellman parameters.

    Returns:
        tuple: (private_key, public_key)
    """
    private_key = parameters.generate_private_key()
    public_key = private_key.public_key()
    return private_key, public_key

def compute_shared_key(private_key, peer_public_key):
    """
    Computes a shared key using a private key and a peer's public key.

    Args:
        private_key (dh.DHPrivateKey): The private key of the party.
        peer_public_key (dh.DHPublicKey): The public key of the peer.

    Returns:
        bytes: The computed shared key.
    """
    return private_key.exchange(peer_public_key)

def generate_access_key():
    """
    Generates an access key using the Diffie-Hellman key exchange protocol.
    
    Returns:
        bytes or bool: The shared key if successful, otherwise False.
    """
    # Generate Diffie-Hellman parameters
    parameters = generate_diffie_hellman_parameters()
    
    # Generate key pairs for both parties
    alice_private_key, alice_public_key = generate_key_pair(parameters)
    bob_private_key, bob_public_key = generate_key_pair(parameters)
    
    # Compute shared keys
    alice_shared_key = compute_shared_key(alice_private_key, bob_public_key)
    bob_shared_key = compute_shared_key(bob_private_key, alice_public_key)
    
    # Check if the shared keys match
    if alice_shared_key == bob_shared_key:
        # Generate a seed using the Seed_Generator module
        Seed_Generator.generate_seed(
            alice_private_key, bob_private_key,
            alice_public_key, bob_public_key
        )
        return alice_shared_key
    else:
        return False

def main():
    """
    Main function to generate and handle the access key.
    """
    shared_key = generate_access_key()
    if shared_key:
        print("Access key generated successfully.")
    else:
        print("Failed to generate access key.")

if __name__ == "__main__":
    main()
