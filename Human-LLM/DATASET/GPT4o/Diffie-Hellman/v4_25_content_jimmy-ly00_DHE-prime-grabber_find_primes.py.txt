import sys
import os
import subprocess
import re
import csv

def extract_dh_prime(cmd_output):
    """
    Extract the Diffie-Hellman prime number from the OpenSSL command output.

    Args:
        cmd_output (str): The output from the OpenSSL command.

    Returns:
        int: The Diffie-Hellman prime number if found, otherwise None.
    """
    for line in cmd_output.splitlines():
        if 'dh_p' in line:
            prime_hex = re.sub(".*: ", "", line)
            return int(prime_hex, 16)
    return None

def check_server_for_dh(server, servername, openssl_trace_path):
    """
    Check the server for Diffie-Hellman parameters using OpenSSL.

    Args:
        server (str): The server address.
        servername (str): The server name.
        openssl_trace_path (str): Path to the OpenSSL trace tool.

    Returns:
        tuple: A tuple containing the server, servername, and Diffie-Hellman prime if found, otherwise None.
    """
    try:
        cmd = subprocess.check_output(
            [openssl_trace_path, "s_client", "-trace", "-cipher", "DHE", "-connect", f"{server}:443"],
            stdin=subprocess.DEVNULL, stderr=subprocess.DEVNULL, timeout=1
        )
        prime = extract_dh_prime(cmd.decode("ISO-8859-1"))
        if prime:
            return (server, servername, prime)
    except (subprocess.CalledProcessError, subprocess.TimeoutExpired):
        pass
    return None

def process_servers(input_file, output_file, openssl_trace_path):
    """
    Process a list of servers to check for Diffie-Hellman prime numbers.

    Args:
        input_file (str): Path to the input CSV file containing server details.
        output_file (str): Path to the output CSV file to write results.
        openssl_trace_path (str): Path to the OpenSSL trace tool.
    """
    with open(output_file, 'w', newline='') as outfile:
        writer = csv.writer(outfile)
        
        with open(input_file, 'r') as csvfile:
            reader = csv.reader(csvfile, delimiter=',')
            for row in reader:
                server, servername = row[0], row[1]
                result = check_server_for_dh(server, servername, openssl_trace_path)
                if result:
                    writer.writerow(result)

def main():
    """
    Main function to process servers and check for Diffie-Hellman prime numbers.
    """
    input_file = 'alexa_top1mil'
    output_file = 'output.csv'
    openssl_trace_path = os.path.join(os.path.dirname(sys.argv[0]), "openssl-trace")

    process_servers(input_file, output_file, openssl_trace_path)

if __name__ == "__main__":
    main()
