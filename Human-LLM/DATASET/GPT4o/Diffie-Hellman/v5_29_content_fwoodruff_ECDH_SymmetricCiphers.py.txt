import TypeConversion as TC

def xor_encrypt(message, symmetric_key_b58):
    """
    Encrypts a plaintext message using XOR with a symmetric key.

    Args:
        message (str): The plaintext message to encrypt.
        symmetric_key_b58 (str): The base-58 encoded symmetric key.

    Returns:
        str: The base-58 encoded encrypted message.
    """
    # Convert the message and symmetric key from their respective formats to big integers
    message_int = TC.string_to_big_int(message)
    symmetric_key_int = TC.b58_to_int(symmetric_key_b58)
    
    # Encrypt the message by applying XOR with the symmetric key
    encrypted_message_int = message_int ^ symmetric_key_int
    
    # Convert the encrypted message back to a base-58 encoded string
    encrypted_message_b58 = TC.int_to_b58(encrypted_message_int)
    
    return encrypted_message_b58

def xor_decrypt(encrypted_message_b58, symmetric_key_b58):
    """
    Decrypts an encrypted message using XOR with a symmetric key.

    Args:
        encrypted_message_b58 (str): The base-58 encoded encrypted message.
        symmetric_key_b58 (str): The base-58 encoded symmetric key.

    Returns:
        str: The decrypted plaintext message.
    """
    # Convert the encrypted message and symmetric key from base-58 encoded strings to big integers
    encrypted_message_int = TC.b58_to_int(encrypted_message_b58)
    symmetric_key_int = TC.b58_to_int(symmetric_key_b58)
    
    # Decrypt the message by applying XOR with the symmetric key
    decrypted_message_int = encrypted_message_int ^ symmetric_key_int
    
    # Convert the decrypted big integer back to a string
    decrypted_message = TC.big_int_to_string(decrypted_message_int)
    
    return decrypted_message
