def calculate_shared_key(base, secret, prime):
    """
    Calculate the shared key using the formula (base^secret) % prime.

    Args:
        base (int): The base number (shared base).
        secret (int): The secret key.
        prime (int): The prime number (shared prime).

    Returns:
        int: The calculated result.
    """
    return pow(base, secret, prime)

def display_key_exchange_info(role, result, shared_key=None):
    """
    Display information about the key exchange process.

    Args:
        role (str): The role of the participant (Alice or Bob).
        result (int): The result of the key calculation.
        shared_key (int, optional): The shared key, if available.
    """
    print("--------------------------")
    if shared_key:
        print(f"{role} performs the operation using the received result and calculates the shared key: {shared_key}")
    else:
        print(f"{role} calculates the result and sends it: {result}")

def main():
    shared_base = 47
    shared_prime = 199

    print(f"g is equal to {shared_base} & p is equal to {shared_prime}")
    print("--------------------------")

    alice_secret = 6
    bob_secret = 2

    print("Alice performs the following operation: g^a mod p and sends result (A) to Bob")
    A = calculate_shared_key(shared_base, alice_secret, shared_prime)
    display_key_exchange_info("Alice", A)
    
    print("Bob performs the same operation and sends result (B) to Alice")
    B = calculate_shared_key(shared_base, bob_secret, shared_prime)
    display_key_exchange_info("Bob", B)
    
    print("Alice now performs the same operation using calculated result (B) from Bob")
    alice_shared_key = calculate_shared_key(B, alice_secret, shared_prime)
    display_key_exchange_info("Alice", B, alice_shared_key)
    
    print("Bob now performs the same operation using calculated result (A) from Alice")
    bob_shared_key = calculate_shared_key(A, bob_secret, shared_prime)
    display_key_exchange_info("Bob", A, bob_shared_key)
    
    print("--------------------------")
    assert alice_shared_key == bob_shared_key, "Shared keys do not match!"
    print(f"Shared Key is equal to {alice_shared_key}.")
    print("Now try this with large prime numbers!")

if __name__ == "__main__":
    main()
