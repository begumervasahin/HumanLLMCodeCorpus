import base58

# TypeConversion module
class TypeConversion:
    @staticmethod
    def int_to_b58(number):
        """
        Convert an integer to a base-58 string.
        
        Args:
            number (int): The integer to convert.
        
        Returns:
            str: The base-58 encoded string.
        """
        return base58.b58encode_int(number).decode()

    @staticmethod
    def b58_to_int(b58_string):
        """
        Convert a base-58 string to an integer.
        
        Args:
            b58_string (str): The base-58 string to convert.
        
        Returns:
            int: The decoded integer.
        """
        return base58.b58decode_int(b58_string)

# ModularOperations module
class ModularOperations:
    @staticmethod
    def modular_sqrt(a, p):
        """
        Compute the modular square root of a number modulo p.
        
        Args:
            a (int): The number to find the square root of.
            p (int): The modulus.
        
        Returns:
            int: The modular square root.
        """
        # Implementing the Tonelli-Shanks algorithm for finding modular square roots
        # Here assuming p is a prime and a is a quadratic residue modulo p
        if a % p == 0:
            return 0
        if p % 4 == 3:
            return pow(a, (p + 1) // 4, p)
        
        # Tonelli-Shanks algorithm
        s = 0
        q = p - 1
        while q % 2 == 0:
            s += 1
            q //= 2
        
        z = 2
        while pow(z, (p - 1) // 2, p) != p - 1:
            z += 1
        
        m = s
        c = pow(z, q, p)
        t = pow(a, q, p)
        r = pow(a, (q + 1) // 2, p)
        
        while t != 0 and t != 1:
            t2i = t
            i = 0
            for i in range(1, m):
                t2i = pow(t2i, 2, p)
                if t2i == 1:
                    break
            
            b = pow(c, 2 ** (m - i - 1), p)
            m = i
            c = pow(b, 2, p)
            t = (t * b) % p
            r = (r * b) % p
        
        return r

# ECRing module
class ECRing:
    class Point:
        def __init__(self, x, y):
            self.x = x
            self.y = y

def point_to_base58(P, curve):
    """
    Maps an elliptic curve point to a base-58 string.

    Args:
        P (ECRing.Point): The elliptic curve point to map.
        curve (object): The elliptic curve parameters.

    Returns:
        str: The base-58 encoded string representing the point.
    """
    key58 = TypeConversion.int_to_b58(P.x)
    if P.y > (curve.p / 2):
        return '03' + key58
    else:
        return '02' + key58

def base58_to_point(P58, curve):
    """
    Maps a base-58 string to an elliptic curve point.

    Args:
        P58 (str): The base-58 string representing the elliptic curve point.
        curve (object): The elliptic curve parameters.

    Returns:
        ECRing.Point: The elliptic curve point.
    """
    a = curve.a
    p = curve.p
    b = curve.b
    keyX = TypeConversion.b58_to_int(P58[2:])
    keyY_sq = (keyX ** 3 + a * keyX + b) % p
    keyY = ModularOperations.modular_sqrt(keyY_sq, p)
    
    if P58.startswith('03'):
        if keyY > (p / 2):
            return ECRing.Point(keyX, keyY)
        else:
            return ECRing.Point(keyX, p - keyY)
    else:
        if keyY > (p / 2):
            return ECRing.Point(keyX, p - keyY)
        else:
            return ECRing.Point(keyX, keyY)
