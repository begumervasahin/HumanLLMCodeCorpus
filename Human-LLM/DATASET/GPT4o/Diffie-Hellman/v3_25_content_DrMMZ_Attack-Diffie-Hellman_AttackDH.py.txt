def find_private_key(g, p, public_key):
    """
    Find the private key corresponding to the given public key.

    Args:
        g (int): The generator.
        p (int): The prime modulus.
        public_key (int): The public key to find the private key for.

    Returns:
        int: The private key if found, otherwise None.
    """
    for x in range(1, p):
        if pow(g, x, p) == public_key:
            return x
    return None

def cryptanalysis_diffie_hellman():
    """
    Perform cryptanalysis of the Diffie-Hellman public key protocol.
    """
    print("\n---------------------------------------------------------------")
    print("    Cryptanalysis of the Diffie-Hellman public key protocol    ")
    print("---------------------------------------------------------------\n")

    # Confirm the prime and generator
    p = int(input("Confirm the prime: "))
    g = int(input("Confirm the generator: "))
    print()

    # Get the public keys sent by A and B
    A = int(input("A sent: "))
    B = int(input("B sent: "))
    print()

    # Find the private keys a and b
    a = find_private_key(g, p, A)
    b = find_private_key(g, p, B)

    if a is None or b is None:
        print("Failed to find private keys.")
        return

    print(f"Password for A: {a}")
    print(f"Password for B: {b}\n")

    # Calculate the common secret key
    k_a = pow(B, a, p)
    k_b = pow(A, b, p)
    
    # Ensure the shared keys match
    assert k_a == k_b, "Shared keys do not match!"
    
    print(f"Their secret common key: {k_a}\n")

if __name__ == "__main__":
    cryptanalysis_diffie_hellman()
