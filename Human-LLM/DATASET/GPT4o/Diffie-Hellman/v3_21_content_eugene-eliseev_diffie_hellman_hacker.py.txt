from person import Person
from tools import AESCipher

class Hacker(Person):
    def read_data(self, sent):
        """
        Process the sent data and perform actions based on the context.
        
        Args:
            sent (tuple): A tuple containing sender, receiver, action, and data.
        
        Returns:
            tuple: Processed data in the form of a tuple.
        """
        send_from, send_to, action, data = sent
        
        # Handle initial parameter exchange
        if action == "my param":
            shared_key = (int(data) ** self.prime) % self.p
            self.keys[send_from] = AESCipher(str(shared_key))
            
            # Initiate communication
            _, _, action, data = self.start_communication_with("")
            return send_from, send_to, action, data
        
        # Handle message exchange if keys are established
        elif send_from in self.keys and send_to in self.keys:
            decoded_data = self.decode(self.keys[send_from], data)
            
            # Translate the message if it's in the known messages
            if decoded_data in self.messages:
                decoded_data = self.messages[decoded_data]
            
            encoded_data = self.encode(self.keys[send_to], decoded_data)
            return send_from, send_to, "message", encoded_data
        
        return sent

class Person:
    def __init__(self, name, prime, p):
        """
        Initialize a person with a name, prime number, and prime modulus.
        
        Args:
            name (str): The name of the person.
            prime (int): The prime number used in calculations.
            p (int): The prime modulus.
        """
        self.name = name
        self.prime = prime
        self.p = p
        self.keys = {}
        self.messages = {}

    def start_communication_with(self, person_name):
        """
        Dummy implementation for initiating communication.
        
        Args:
            person_name (str): The name of the person to start communication with.
        
        Returns:
            tuple: Dummy data to simulate communication initiation.
        """
        return self.name, person_name, "initiate", "some_data"

    def encode(self, cipher, data):
        """
        Dummy implementation for encoding data.
        
        Args:
            cipher (AESCipher): The cipher used for encryption.
            data (str): The data to be encoded.
        
        Returns:
            str: Encoded data.
        """
        return cipher.encrypt(data)

    def decode(self, cipher, data):
        """
        Dummy implementation for decoding data.
        
        Args:
            cipher (AESCipher): The cipher used for decryption.
            data (str): The data to be decoded.
        
        Returns:
            str: Decoded data.
        """
        return cipher.decrypt(data)
