import os
import socket

if os.name != "nt":
    import fcntl
    import struct

    def get_interface_ip(ifname):
        """Get the IP address of a given network interface."""
        with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as s:
            return socket.inet_ntoa(fcntl.ioctl(
                s.fileno(),
                0x8915,  # SIOCGIFADDR
                struct.pack('256s', ifname[:15].encode('utf-8'))
            )[20:24])

def get_lan_ip():
    """Get the LAN IP address of the current machine."""
    ip = socket.gethostbyname(socket.gethostname())

    # Check if the IP address starts with 127. (loopback address) and adjust for non-Windows systems
    if ip.startswith("127.") and os.name != "nt":
        interfaces = [
            "eth0", "eth1", "eth2",
            "wlan0", "wlan1",
            "wifi0",
            "ath0", "ath1",
            "ppp0"
        ]
        for ifname in interfaces:
            try:
                ip = get_interface_ip(ifname)
                if ip:
                    break
            except IOError:
                continue
    return ip

if __name__ == "__main__":
    lan_ip = get_lan_ip()
    print(f"LAN IP Address: {lan_ip}")
