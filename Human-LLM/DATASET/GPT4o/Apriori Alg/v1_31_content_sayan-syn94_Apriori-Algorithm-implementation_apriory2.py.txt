import sys
import csv
from collections import defaultdict
from itertools import chain, combinations
import time

# Start the timer
start_time = time.time()

# Global dictionary to store the frequency of itemsets
freqSet = defaultdict(int)

def data_gen(fname):
    """
    Function which reads from the file and yields a generator.

    Args:
        fname (str): The name of the input file.

    Yields:
        frozenset: A frozenset of items in each transaction.
    """
    with open(fname, 'r') as file:
        for line in file:
            line = line.strip().rstrip(',')  # Remove trailing comma
            record = frozenset(line.split(','))
            yield record

def transactions(data_iterator):
    """
    Generate the initial itemset and the list of transactions.

    Args:
        data_iterator (generator): Generator yielding transactions.

    Returns:
        tuple: A set of items and a list of transactions.
    """
    transactionList = list()
    items = set()
    for record in data_iterator:
        transactionList.append(record)
        for item in record:
            items.add(frozenset([item]))  # Generate 1-itemSets
    return items, transactionList

def joinSet(itemSet, length):
    """
    Join sets to return n-length itemset where n=length.

    Args:
        itemSet (set): A set of itemsets.
        length (int): The length of the itemsets to generate.

    Returns:
        set: A set of n-length itemsets.
    """
    return set([i.union(j) for i in itemSet for j in itemSet if len(i.union(j)) == length])

def min_support(itemlist, transaction, min_supp):
    """
    Calculate the support for itemsets and filter by minimum support.

    Args:
        itemlist (set): A set of itemsets.
        transaction (list): A list of transactions.
        min_supp (float): The minimum support threshold.

    Returns:
        set: A set of itemsets that meet the minimum support.
    """
    s = defaultdict(int)
    itemSet = set()
    for item in itemlist:
        for trans in transaction:
            if item.issubset(trans):
                s[item] += 1
    for item, count in s.items():
        support = float(count) / len(transaction)
        if support >= min_supp:
            itemSet.add(item)
            freqSet[item] = support
    return itemSet

def subsets(items):
    """
    Returns non-empty subsets of items.

    Args:
        items (set): A set of items.

    Returns:
        iterator: An iterator of subsets.
    """
    return chain(*[combinations(items, i + 1) for i, a in enumerate(items)])

def run_apriori(data_iter, minSupport, minConfidence):
    """
    Run the Apriori algorithm.

    Args:
        data_iter (generator): Generator yielding transactions.
        minSupport (float): The minimum support threshold.
        minConfidence (float): The minimum confidence threshold.
    """
    itemSet, transactionList = transactions(data_iter)
    largeSet = dict()
    oneCSet = min_support(itemSet, transactionList, minSupport)
    currentLSet = oneCSet
    k = 2
    while currentLSet:
        largeSet[k - 1] = currentLSet
        currentLSet = joinSet(currentLSet, k)
        currentCSet = min_support(currentLSet, transactionList, minSupport)
        print(k)
        print(currentCSet)
        currentLSet = currentCSet
        k += 1
    for key, value in list(largeSet.items())[1:]:
        for item in value:
            _subsets = map(frozenset, [x for x in subsets(item)])
            for element in _subsets:
                remain = item.difference(element)
                if len(remain) > 0:  # Ensure only subsets pass through except the original set
                    confidence = freqSet[item] / freqSet[element]
                    if confidence >= minConfidence:
                        print(f"{element} , {remain} => {confidence}")

if __name__ == "__main__":
    # Minimum support and confidence thresholds
    minSupport = 0.01
    minConfidence = 0.01

    # Input file containing transactions
    inFile = data_gen('T10I4D100k-copy.txt')

    # Run the Apriori algorithm
    run_apriori(inFile, minSupport, minConfidence)

    # Print the time taken to run the algorithm
    print("Time taken:", time.time() - start_time)
