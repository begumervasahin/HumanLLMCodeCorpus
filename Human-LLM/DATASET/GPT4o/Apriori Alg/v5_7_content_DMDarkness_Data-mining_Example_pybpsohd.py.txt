This script demonstrates how to use the BPSO-HD algorithm to mine frequent itemsets

import pybpsohd as pybp
import re

def read_dataset(file_name):
    """
    Read the transactional dataset from a file.

    Parameters:
    file_name (str): The name of the file containing the dataset.

    Returns:
    list: A list of transactions, each transaction is a list of items.
    """
    transactions = []
    with open(file_name, 'r') as file:
        content = file.readlines()
        for line in content:
            # Split the line by spaces or newline characters
            items = re.split(r' |\n', line)
            # Convert items to integers and filter out non-digit items
            transaction = [int(item) for item in items if item.isdigit()]
            if transaction:
                transactions.append(transaction)
    return transactions

def main():
    """
    Main function to read the dataset and mine frequent itemsets using the BPSO-HD algorithm.
    """
    # Read the transactional dataset
    # The dataset kosarak.dat can be downloaded from http://fimi.ua.ac.be/data/
    dataset = read_dataset("kosarak.dat")

    # Parameters for the BPSO-HD algorithm
    min_support = 0.00001
    population_size = 30
    num_generations = 30
    inertia_weight = 0.5
    accel_coeff1 = 1
    accel_coeff2 = 1
    min_pattern_length = 10

    # Mine long patterns from the dataset using the BPSO-HD algorithm
    frequent_itemsets = pybp.getFP(
        dataset, min_support, population_size, num_generations,
        inertia_weight, accel_coeff1, accel_coeff2, min_pattern_length
    )

    # Output the result
    print("Frequent itemsets:")
    for itemset in frequent_itemsets:
        print(itemset)

if __name__ == "__main__":
    main()
