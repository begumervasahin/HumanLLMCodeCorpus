Transforming frequent itemsets into association rules based
on specified minimum confidence and minimum lift.

def get_subsets(pat, sub1, sub2, L, temp, base):
    """
    Get all the subsets with length L and len(pat)-L of the object pattern pat.

    Args:
        pat (list): The pattern to find subsets of.
        sub1 (list): The list to store one set of subsets.
        sub2 (list): The list to store the complementary subsets.
        L (int): The length of the subsets to generate.
        temp (list): A temporary list to store intermediate results.
        base (int): The base index to start generating subsets.
    """
    if L < 1:
        if temp not in sub1:
            temp.sort()
            pat.sort()
            sub1.append(temp)
            sub2.append(pat)
    else:
        for i in range(base, len(pat)):
            get_subsets(pat[:i] + pat[i+1:], sub1, sub2, L-1, temp + [pat[i]], i)

def generate_association_rules(fi, N, min_conf, min_lift):
    """
    Generate association rules from frequent itemsets.

    Args:
        fi (list): The list of frequent itemsets.
        N (int): The total number of transactions.
        min_conf (float): The minimum confidence threshold.
        min_lift (float): The minimum lift threshold.

    Returns:
        list: A list of generated association rules.
    """
    rules = []
    fi_num = len(fi)
    fi_dict = {}

    # Transform frequent patterns to a dictionary with pattern as key and support as value
    for i in range(fi_num):
        pattern = tuple(sorted(fi[i][:-1]))
        fi_dict[pattern] = fi[i][-1]

    for i in range(fi_num):
        pattern = fi[i][:-1]
        support = fi[i][-1]
        subset_len = int((len(pattern) - 1) / 2)
        sub1, sub2 = [], []

        # Find all subsets of the pattern
        for j in range(1, subset_len + 1):
            get_subsets(pattern, sub1, sub2, j, [], 0)

        for j in range(len(sub1)):
            if sub1[j] in sub2:
                sub1[j] = [-1, -9]
                sub2[j] = [-2, -5]

            pat1, pat2 = sorted(sub1[j]), sorted(sub2[j])
            if tuple(pat1) in fi_dict and tuple(pat2) in fi_dict:
                sup1, sup2, sup3 = support, fi_dict[tuple(pat1)], fi_dict[tuple(pat2)]

                # Check lift and confidence
                if (sup1 * N) / (sup2 * sup3) > min_lift:
                    if sup1 / sup2 > min_conf:
                        rules.append([pat1, pat2, sup1, sup1 / sup2, (sup1 * N) / (sup2 * sup3)])
                    if sup1 / sup3 > min_conf:
                        rules.append([pat2, pat1, sup1, sup1 / sup3, (sup1 * N) / (sup2 * sup3)])

    return rules

if __name__ == "__main__":
    # Example frequent itemsets with their support counts
    frequent_itemsets = [
        [['A'], 0.5],
        [['B'], 0.6],
        [['A', 'B'], 0.3],
        [['A', 'C'], 0.2],
        [['B', 'C'], 0.25],
        [['A', 'B', 'C'], 0.1]
    ]

    # Total number of transactions
    total_transactions = 1000

    # Minimum confidence and lift thresholds
    min_confidence = 0.7
    min_lift = 1.2

    # Generate association rules
    rules = generate_association_rules(frequent_itemsets, total_transactions, min_confidence, min_lift)

    # Print the generated rules
    for rule in rules:
        print(f"Rule: {rule[0]} -> {rule[1]}, Support: {rule[2]}, Confidence: {rule[3]}, Lift: {rule[4]}")
