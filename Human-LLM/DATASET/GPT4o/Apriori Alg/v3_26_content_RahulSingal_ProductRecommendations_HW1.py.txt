def load_data(file_path):
    """
    Load dataset from the given file path.

    Args:
        file_path (str): Path to the dataset file.

    Returns:
        list: List of transactions.
    """
    with open(file_path, 'r') as file:
        dataset = file.read().splitlines()
    return dataset

def apriori_algorithm(dataset, min_support):
    """
    Apply the Apriori algorithm to find frequent itemsets.

    Args:
        dataset (list): List of transactions.
        min_support (int): Minimum support threshold.

    Returns:
        tuple: Frequent itemsets of size 1, 2, and 3.
    """
    # Initialize dictionaries for candidate itemsets
    C1 = {}  # Candidates of size 1
    C2 = {}  # Candidates of size 2
    C3 = {}  # Candidates of size 3

    # Initialize lists for frequent itemsets
    L1 = []  # Frequent items of size 1
    L2 = []  # Frequent items of size 2
    L3 = []  # Frequent items of size 3

    # Count occurrences of each item
    for transaction in dataset:
        items = transaction.split()
        for item in items:
            if item in C1:
                C1[item] += 1
            else:
                C1[item] = 1

    # Select frequent items of size 1
    L1 = [item for item, count in C1.items() if count >= min_support]

    # Generate candidate itemsets of size 2
    for i in range(len(L1)):
        for j in range(i + 1, len(L1)):
            C2[(L1[i], L1[j])] = 0

    # Count occurrences of each itemset of size 2
    for transaction in dataset:
        items = set(transaction.split())
        for itemset in C2:
            if set(itemset).issubset(items):
                C2[itemset] += 1

    # Select frequent itemsets of size 2
    L2 = [itemset for itemset, count in C2.items() if count >= min_support]

    # Generate candidate itemsets of size 3
    for (a, b) in L2:
        for c in L1:
            if c != a and c != b:
                C3[(a, b, c)] = 0

    # Count occurrences of each itemset of size 3
    for transaction in dataset:
        items = set(transaction.split())
        for itemset in C3:
            if set(itemset).issubset(items):
                C3[itemset] += 1

    # Select frequent itemsets of size 3
    L3 = [itemset for itemset, count in C3.items() if count >= min_support]

    return L1, L2, L3

def main():
    """
    Main function to execute the Apriori algorithm on a given dataset.
    """
    # Load dataset
    file_path = 'browsingdata.txt'
    dataset = load_data(file_path)

    # Set minimum support threshold
    min_support = 3  # Change to 100 for real data

    # Apply Apriori algorithm
    L1, L2, L3 = apriori_algorithm(dataset, min_support)

    # Print results
    print("Frequent itemsets of size 1:")
    for itemset in L1:
        print(itemset)

    print("\nFrequent itemsets of size 2:")
    for itemset in L2:
        print(itemset)

    print("\nFrequent itemsets of size 3:")
    for itemset in L3:
        print(itemset)

if __name__ == "__main__":
    main()
