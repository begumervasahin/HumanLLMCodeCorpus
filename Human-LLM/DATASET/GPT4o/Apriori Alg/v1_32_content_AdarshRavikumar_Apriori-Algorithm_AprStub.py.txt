from Apriori import Apriori
import re

def get_user_inputs():
    """
    Get user inputs for minimum support, confidence, and rule length.
    
    Returns:
    tuple: min_support (int), min_confidence (float), min_length (int)
    """
    min_support = int(input("Minimum support\n"))
    min_confidence = float(input("Minimum Confidence\n"))
    min_length = int(input("Minimum Length of rules \n"))
    return min_support, min_confidence, min_length

def load_dataset(file_path):
    """
    Load and preprocess the dataset from the given file path.

    Args:
    file_path (str): Path to the dataset file.

    Returns:
    list: List of transactions, where each transaction is a list of items.
    """
    alpha = [chr(i) for i in range(ord('a'), ord('z')+1)]
    transactions = []
    
    with open(file_path, 'r') as file:
        for line in file:
            line = line.strip()
            line = re.sub(r"[?\s]", 'a', line)
            items = line.split(',')
            transaction = [item for item in items if item in alpha]
            transactions.append(transaction)
    
    return transactions

def main():
    # Get user inputs
    min_support, min_confidence, min_length = get_user_inputs()
    
    # Load dataset
    file_path = 'datasetUCI.txt'
    transactions = load_dataset(file_path)
    
    # Apply Apriori algorithm
    rules = Apriori(transactions, min_support, min_confidence, min_length)
    
    # Print results (assuming the Apriori class has a method to print or return rules)
    for rule in rules:
        print(rule)

if __name__ == "__main__":
    main()
