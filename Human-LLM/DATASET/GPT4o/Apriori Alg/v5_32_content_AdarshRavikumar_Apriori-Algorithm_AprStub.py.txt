import re
from Apriori import Apriori

def load_dataset(file_path):
    """
    Load and preprocess the dataset from a text file.

    Parameters:
    file_path (str): Path to the text file containing the dataset.

    Returns:
    list: List of transactions where each transaction is a list of items.
    """
    # List of alphabet characters
    alphabet = [chr(i) for i in range(ord('a'), ord('z') + 1)]
    transactions = []

    with open(file_path, 'r') as file:
        for line in file:
            # Replace '?' and whitespace with 'a', then split by commas
            cleaned_line = re.sub(r"[?\s]", 'a', line.strip())
            # Filter out items not in the alphabet list
            items = [item for item in cleaned_line.split(',') if item in alphabet]
            transactions.append(items)

    return transactions

    """
    Main function to run the Apriori algorithm with user-specified parameters.

    # Get user input for Apriori parameters
    min_support = int(input("Minimum support: "))

    # Path to the dataset file
