import itertools

def getSubSet(S):
    """
    Return the list of subsets of S, each with length len(S) - 1.
    """
    subLens = len(S) - 1
    return [list(i) for i in itertools.combinations(S, subLens)]

def isSubSetExist(tar, L):
    """
    Check if each value in L is a subset of any set in tar.
    Returns 1 if exists, otherwise 0.
    """
    tar_sets = [set(i[0]) for i in tar]
    for i in L:
        for j in tar_sets:
            if j < set(i):
                return 0
    return 1

def Apriori(tar, thr, le):
    """
    Apriori algorithm for frequent item set mining.
    Args:
    tar (dict): Dictionary with transaction index as key and transaction details as value.
    thr (float): Support threshold.
    le (int): Maximum length of item sets to be considered.
    
    Returns:
    list: List of frequent item sets with their support values.
    """
    curFrequ = {}
    
    # Calculate the initial frequency of items
    for key in tar.keys():
        items = tar[key]
        for item in items:
            if item not in curFrequ:
                curFrequ[item] = 1
            else:
                curFrequ[item] += 1
    
    tarLen = len(tar)
    
    # Calculate support and prune infrequent items
    keys = list(curFrequ.keys())
    for key in keys:
        curFrequ[key] = curFrequ[key] / tarLen
        if curFrequ[key] < thr:
            del curFrequ[key]
    
    curFrequ = [[key, curFrequ[key]] for key in curFrequ]
    
    if le == 1:
        return curFrequ
    
    tempLe = 2
    while tempLe <= le:
        tempList = []
        
        # Generate candidate item sets
        for i in range(len(curFrequ)):
            for j in range(i + 1, len(curFrequ)):
                if tempLe == 2:
                    temp = [curFrequ[i][0], curFrequ[j][0]]
                else:
                    temp = list(set(curFrequ[i][0]) | set(curFrequ[j][0]))
                
                if len(temp) != tempLe:
                    continue
                
                temp.sort()
                if isSubSetExist(curFrequ, getSubSet(temp)) == 1:
                    if temp not in tempList:
                        tempList.append(temp)
        
        curFrequ = []
        
        # Calculate support for candidate item sets and prune infrequent sets
        for item in tempList:
            tempLen = sum(1 for key in tar if set(item) <= set(tar[key]))
            tempLen /= tarLen
            if tempLen >= thr:
                curFrequ.append([item, tempLen])
        
        tempLe += 1
    
    return curFrequ

if __name__ == "__main__":
    # Example transaction data
    transactions = {
        0: ['A', 'B', 'C'],
        1: ['A', 'C'],
        2: ['A', 'D'],
        3: ['B', 'C'],
        4: ['A', 'B', 'C', 'D']
    }
    
    # Minimum support threshold
    threshold = 0.5
    
    # Maximum length of item sets
    max_length = 3
    
    # Run Apriori algorithm
    frequent_itemsets = Apriori(transactions, threshold, max_length)
    
    # Print the result
    print("Frequent Itemsets:")
    for itemset in frequent_itemsets:
        print(f"Itemset: {itemset[0]}, Support: {itemset[1]:.2f}")
