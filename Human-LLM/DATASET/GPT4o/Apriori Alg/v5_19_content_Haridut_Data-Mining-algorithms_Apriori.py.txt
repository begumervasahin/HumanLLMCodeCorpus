from collections import defaultdict
from itertools import chain, combinations, groupby


def get_item_set_transaction_list(data):
    """
    Generate item set and transaction list from the data.
    """
    transaction_list = []
    item_set = set()
    
    for record in data:
        transaction = frozenset(record)
        transaction_list.append(transaction)
        for item in transaction:
            item_set.add(frozenset([item]))
    
    return item_set, transaction_list


def return_items_with_min_support(item_set, transaction_list, min_sup, freq_set):
    """
    Calculate item sets with minimum support.
    """
    valid_items = set()
    local_set = defaultdict(int)
    
    for item in item_set:
        for transaction in transaction_list:
            if item.issubset(transaction):
                freq_set[item] += 1
                local_set[item] += 1
    
    for item, count in local_set.items():
        if count >= min_sup:
            valid_items.add(item)
    
    return valid_items


def join_set(item_set, length):
    """
    Join a set with itself and return the n-element item sets.
    """
    return set([i.union(j) for i in item_set for j in item_set if len(i.union(j)) == length])


def apriori(data, min_sup):
    """
    Run the Apriori algorithm.
    """
    item_set, transaction_list = get_item_set_transaction_list(data)
    freq_set = defaultdict(int)
    large_set = {}
    
    one_c_set = return_items_with_min_support(item_set, transaction_list, min_sup, freq_set)
    current_l_set = one_c_set
    k = 2
    
    while current_l_set:
        large_set[k - 1] = current_l_set
        current_l_set = join_set(current_l_set, k)
        current_c_set = return_items_with_min_support(current_l_set, transaction_list, min_sup, freq_set)
        current_l_set = current_c_set
        k += 1
    
    return large_set, freq_set


def print_results(large_set, freq_set):
    """
    Print the results of the Apriori algorithm.
    """
    for key, value in large_set.items():
        print(f"Frequent {key}-itemsets:")
        for itemset in value:
            print(f"{list(itemset)}: {freq_set[itemset]}")


if __name__ == "__main__":
    data = [
        # Example data
        ['bread', 'milk'],
        ['bread', 'diaper', 'beer', 'egg'],
        ['milk', 'diaper', 'beer', 'coke'],
        ['bread', 'milk', 'diaper', 'beer'],
        ['bread', 'milk', 'diaper', 'coke'],
    ]
    
    min_sup = 2
    large_set, freq_set = apriori(data, min_sup)
    print_results(large_set, freq_set)
