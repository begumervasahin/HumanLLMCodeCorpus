def level2_candidate_gen(L, phi, MIS, support_count):
    """
    Returns all frequent 2-itemsets.
    
    Parameters:
    L (list): List of items.
    phi (float): Support difference constraint.
    MIS (dict): Minimum item support for each item.
    support_count (dict): Support count for each item.
    
    Returns:
    list: List of frequent 2-itemsets.
    """
    i = 0
    temp_list = []
    for l in L:
        i += 1
        if support_count[l] >= MIS[l]:
            for h in L[i:]:
                if support_count[h] >= MIS[l] and abs(support_count[h] - support_count[l]) <= phi:
                    tupple_to_be_added = (l, h)
                    temp_list.append(tupple_to_be_added)
    return temp_list

def MSCandidate_gen(F, phi, support_count, k, MIS):
    """
    Candidate Generation - Joining and Pruning of itemsets.
    
    Parameters:
    F (list): List of frequent k-1 itemsets.
    phi (float): Support difference constraint.
    support_count (dict): Support count for each item.
    k (int): Size of the itemsets to be generated.
    MIS (dict): Minimum item support for each item.
    
    Returns:
    list: List of candidate k-itemsets.
    """
    temp_list = []
    for f1 in F:
        for f2 in F:
            if f1 != f2:
                if (f1[len(f1)-1] < f2[len(f2)-1]) and (f1[:-1] == f2[:-1]) and abs(support_count[f1[len(f1)-1]] - support_count[f2[len(f2)-1]]) <= phi:
                    candidate = f1 + f2[-1:]
                    temp_list.append(candidate)

                    for j in range(len(candidate)):
                        s = candidate[:j] + candidate[j+1:]
                        if candidate[0] in s or MIS[candidate[1]] == MIS[candidate[0]]:
                            if s not in F:
                                temp_list.remove(candidate)
                                break
    return temp_list

    # Sample frequent itemsets
