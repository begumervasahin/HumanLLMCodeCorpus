## Functions
import re
from operator import itemgetter

def calculate_M(MIS):
    """
    Calculate and return a sorted list of MIS values.
    
    Args:
        MIS (dict): Dictionary of item MIS values.
        
    Returns:
        list: List of tuples sorted by MIS values.
    """
    return sorted(MIS.items(), key=itemgetter(1))

def create_list_of_items(I, MIS):
    """
    Create a list of items based on the MIS dictionary.
    
    Args:
        I (list): List to store item names.
        MIS (dict): Dictionary of item MIS values.
    """
    I.extend(MIS.keys())

def init_pass(M, L, MIS, support_count):
    """
    Initial pass over the sorted items list to populate the list L.
    
    Args:
        M (list): List of items sorted by MIS values.
        L (list): List to store items that meet the support criteria.
        MIS (dict): Dictionary of item MIS values.
        support_count (dict): Dictionary of item support counts.
    """
    conf_flag = True
    for item, mis_val in M:
        if support_count[item] >= MIS[item] and conf_flag:
            L.append(item)
            min_mis_required = MIS[item]
            conf_flag = False
        elif not conf_flag:
            if support_count[item] >= min_mis_required:
                L.append(item)

def check_cannot_be_together(F, cannot_be_together):
    """
    Filter out itemsets that contain items which cannot be together.
    
    Args:
        F (list): List of frequent itemsets.
        cannot_be_together (list): List of item pairs that cannot be together.
        
    Returns:
        list: Filtered list of frequent itemsets.
    """
    filtered_F = [[], [], [], [], []]
    level = 0

    for f in F:
        if len(f) > 0:
            for s in f:
                if level > 0:
                    for c in cannot_be_together:
                        if not set(c).issubset(s):
                            filtered_F[level].append(s)
                else:
                    filtered_F[level].append(s)
            level += 1
    return filtered_F

def check_must_have(F, must_have):
    """
    Ensure itemsets contain at least one of the must-have items.
    
    Args:
        F (list): List of frequent itemsets.
        must_have (list): List of must-have items.
        
    Returns:
        list: Filtered list of frequent itemsets.
    """
    final_F = [[], [], [], [], []]
    level = 0

    for f in F:
        if len(f) > 0:
            temp = set()
            for s in f:
                for must in must_have:
                    if level > 1:
                        if must in s:
                            temp.add(s)
                    else:
                        if s in must_have:
                            temp.add(s)
            final_F[level].extend(list(temp))
        level += 1
    return final_F
