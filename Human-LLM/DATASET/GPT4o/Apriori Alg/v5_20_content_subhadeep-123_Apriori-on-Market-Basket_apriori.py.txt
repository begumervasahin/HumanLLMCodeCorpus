# Apriori Algorithm Implementation

# Importing the necessary libraries
import pandas as pd
from apyori import apriori

def load_dataset(file_path):
    """
    Load the dataset and preprocess it into a list of transactions.
    
    Args:
        file_path (str): Path to the dataset file.
        
    Returns:
        list: List of transactions.
    """
    dataset = pd.read_csv(file_path, header=None)
    transactions = [
        [str(dataset.values[i, j]) for j in range(dataset.shape[1])]
        for i in range(len(dataset))
    ]
    return transactions

def apply_apriori(transactions, min_support, min_confidence, min_lift, min_length):
    """
    Apply the Apriori algorithm to find frequent itemsets and association rules.
    
    Args:
        transactions (list): List of transactions.
        min_support (float): Minimum support threshold.
        min_confidence (float): Minimum confidence threshold.
        min_lift (float): Minimum lift threshold.
        min_length (int): Minimum length of itemsets.
        
    Returns:
        list: List of association rules.
    """
    rules = apriori(
        transactions,
        min_support=min_support,
        min_confidence=min_confidence,
        min_lift=min_lift,
        min_length=min_length
    )
    return list(rules)

def display_rules(rules):
    """
    Display the association rules.
    
    Args:
        rules (list): List of association rules.
    """
    for rule in rules:
        items = [item for item in rule.items]
        print(f"Rule: {items[:-1]} -> {items[-1]}")
        print(f"Support: {rule.support}")
        for ordered_stat in rule.ordered_statistics:
            print(f"Confidence: {ordered_stat.confidence}")
            print(f"Lift: {ordered_stat.lift}")
        print("")

def main():
    # Define parameters
    file_path = 'Market_Basket_Optimisation.csv'
    min_support = 0.003
    min_confidence = 0.2
    min_lift = 3
    min_length = 2

    # Load dataset
    transactions = load_dataset(file_path)

    # Apply Apriori
    rules = apply_apriori(transactions, min_support, min_confidence, min_lift, min_length)

    # Display results
    display_rules(rules)

if __name__ == "__main__":
    main()
