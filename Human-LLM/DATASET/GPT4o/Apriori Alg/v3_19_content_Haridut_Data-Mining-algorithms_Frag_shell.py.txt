import sys
import itertools

def read_input():
    """Read input from stdin and process it into partitions and item_list."""
    raw_input = sys.stdin.readlines()
    partitions = int(raw_input[0].strip())
    item_list = [line.strip().split() for line in raw_input[1:]]
    return partitions, item_list

def create_dim_list(item_list):
    """Create a list of unique values for each dimension in the item_list."""
    transposed_list = list(zip(*item_list))
    dim_list = [sorted(set(dim)) for dim in transposed_list]
    return dim_list

def create_partitions(item_list, partitions):
    """Create partitions for the item list."""
    partitioned_indices = [[] for _ in range(partitions)]
    for i in range(len(item_list[0])):
        partitioned_indices[i % partitions].append(i)
    ind_count = [len(indices) for indices in partitioned_indices]
    return partitioned_indices, ind_count

def fragment_list(dim_list, partitions, ind_count):
    """Fragment the dim_list based on partitions."""
    fragmented_list = []
    index = 0
    for count in ind_count:
        fragmented_list.append(dim_list[index:index + count])
        index += count
    return fragmented_list

def count_combinations(subset, item_list):
    """Count how many item lists contain the subset."""
    count = sum(1 for item in item_list if set(subset).issubset(item))
    return count

def compute_fragments(fragmented_list, item_list):
    """Compute and print the fragments and their counts."""
    for fragment in fragmented_list:
        for length in range(1, len(fragment) + 1):
            for combination in itertools.combinations(fragment, length):
                flattened_combination = [item for sublist in combination for item in sublist]
                for subset in itertools.combinations(flattened_combination, length):
                    count = count_combinations(subset, item_list)
                    if count > 0:
                        print(" ".join(subset) + ": " + str(count))
        print()

def main():
    """Main function to orchestrate the process."""
    partitions, item_list = read_input()
    dim_list = create_dim_list(item_list)
    partitioned_indices, ind_count = create_partitions(item_list, partitions)
    fragmented_list = fragment_list(dim_list, partitions, ind_count)
    compute_fragments(fragmented_list, item_list)

if __name__ == "__main__":
    main()
