import sys
from collections import defaultdict
from optparse import OptionParser
import csv

def has_infreq_subset(candidate, previous_level_itemsets):
    """
    Check if a candidate itemset has any infrequent subset.

    Args:
        candidate (frozenset): The candidate itemset.
        previous_level_itemsets (list): List of frequent itemsets from the previous level.

    Returns:
        bool: True if the candidate has any infrequent subset, False otherwise.
    """
    for item in candidate:
        subset = candidate - frozenset([item])
        if subset not in previous_level_itemsets:
            return True
    return False

def apriori_gen(previous_level_itemsets, k):
    """
    Generate candidate itemsets of length k.

    Args:
        previous_level_itemsets (list): List of frequent itemsets from the previous level.
        k (int): The length of the candidate itemsets to generate.

    Returns:
        list: List of candidate itemsets of length k.
    """
    candidates = []
    for i in previous_level_itemsets:
        for j in previous_level_itemsets:
            candidate = i.union(j)
            if len(candidate) == k and not has_infreq_subset(candidate, previous_level_itemsets):
                candidates.append(candidate)
    return list(set(candidates))

def find_freq_1_itemsets(transactions, min_sup):
    """
    Find frequent 1-itemsets in the dataset.

    Args:
        transactions (list): List of transactions.
        min_sup (float): Minimum support threshold.

    Returns:
        list: List of frequent 1-itemsets.
    """
    item_count = defaultdict(int)
    for transaction in transactions:
        for item in transaction:
            item_count[item] += 1

    num_transactions = len(transactions)
    return [frozenset([item]) for item, count in item_count.items() if count / num_transactions >= min_sup]

def apriori(transactions, min_sup):
    """
    Apply the Apriori algorithm to find frequent itemsets in the dataset.

    Args:
        transactions (list): List of transactions.
        min_sup (float): Minimum support threshold.

    Returns:
        list: List of lists of frequent itemsets of different lengths.
    """
    transactions = [frozenset(transaction) for transaction in transactions]
    frequent_itemsets = [frozenset()]
    frequent_itemsets.append(find_freq_1_itemsets(transactions, min_sup))
    k = 2

    while frequent_itemsets[k-1]:
        candidates = apriori_gen(frequent_itemsets[k-1], k)
        candidate_count = defaultdict(int)

        for candidate in candidates:
            for transaction in transactions:
                if candidate.issubset(transaction):
                    candidate_count[candidate] += 1

        num_transactions = len(transactions)
        frequent_k_itemsets = [itemset for itemset, count in candidate_count.items() if count / num_transactions >= min_sup]
        frequent_itemsets.append(frozenset(frequent_k_itemsets))
        k += 1

    return frequent_itemsets

def main():
    """
    Main function to parse command line arguments and execute the Apriori algorithm.
    """
    opt_parser = OptionParser()
    opt_parser.add_option(
        '-f',
        '--inputFile',
        dest='input',
        help='Filename containing CSV data',
        default=None
    )
    opt_parser.add_option(
        '-s',
        '--minSupport',
        dest='min_sup',
        help='Minimum support value',
        default=0.15,
        type='float'
    )

    (options, args) = opt_parser.parse_args()

    if options.input is None:
        print("Please specify an input file using the -f option.")
        sys.exit(1)

    min_sup = options.min_sup

    with open(options.input, 'r') as infile:
        csv_reader = csv.reader(infile)
        transactions = [row for row in csv_reader]

    frequent_itemsets = apriori(transactions, min_sup)

    for i, level in enumerate(frequent_itemsets):
        print(f"Frequent itemsets of size {i}:")
        for itemset in level:
            print(itemset)

if __name__ == '__main__':
    main()
