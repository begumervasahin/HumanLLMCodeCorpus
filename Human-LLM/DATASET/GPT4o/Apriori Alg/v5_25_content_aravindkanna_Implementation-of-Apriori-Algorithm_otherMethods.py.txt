import itertools
from sets import Set  # Assuming Set is imported from a custom module

def generate(current_size_list):
    """
    Generate candidate itemsets of the next size from the current size itemsets.

    :param current_size_list: List of current size itemsets
    :return: List of next size itemsets
    """
    next_size_list = []
    size = len(current_size_list)
    
    for i in range(size):
        for j in range(i + 1, size):
            if current_size_list[i][:-1] == current_size_list[j][:-1]:
                candidate = current_size_list[i][:]
                candidate.append(current_size_list[j][-1])
                next_size_list.append(candidate)
            else:
                break

    return next_size_list

def one_less_subsets(itemset):
    """
    Generate all subsets of the itemset by removing one element at a time.

    :param itemset: List of items
    :return: List of subsets
    """
    return [itemset[:i] + itemset[i + 1:] for i in range(len(itemset))]

def prune(freqs_trie, curr_list):
    """
    Prune the candidate itemsets by checking if all their subsets are frequent.

    :param freqs_trie: Trie structure containing frequent itemsets
    :param curr_list: List of candidate itemsets
    :return: Pruned list of candidate itemsets
    """
    pruned_list = []
    for itemset in curr_list:
        subsets = one_less_subsets(itemset)
        if all(freqs_trie.hasNode(subset) for subset in subsets):
            pruned_list.append(itemset)
    return pruned_list

def item_sets_count(in_file, curr_list):
    """
    Count the support of each candidate itemset in the transactions.

    :param in_file: Input file containing transactions
    :param curr_list: List of candidate itemsets
    :return: List of support counts for each candidate itemset
    """
    counts = [0] * len(curr_list)
    
    with open(in_file) as infp:
        for line in infp:
            transaction = Set(line.strip().split(","))
            for idx, candidate in enumerate(curr_list):
                if transaction.issuperset(Set(candidate)):
                    counts[idx] += 1

    return counts

def print_associate_rules(freqs_trie, min_confidence):
    """
    Generate and print association rules from the frequent itemsets.

    :param freqs_trie: Trie structure containing frequent itemsets
    :param min_confidence: Minimum confidence threshold
    :return: Count of generated association rules
    """
    count = 0

    for itemset in freqs_trie.getItemSets([]):
        for j in range(1, len(itemset)):
            for subset in itertools.combinations(itemset, j):
                confidence = float(freqs_trie.getCount(itemset)) / float(freqs_trie.getCount(subset))
                if confidence >= min_confidence:
                    print(",".join(subset) + " => " + ",".join(set(itemset) - set(subset)))
                    count += 1

    return count
