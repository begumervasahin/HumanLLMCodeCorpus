import itertools

def get_subsets(S):
    """
    Return the list of subsets of S, each with length len(S) - 1.
    """
    subset_length = len(S) - 1
    return [list(subset) for subset in itertools.combinations(S, subset_length)]

def is_subset_exist(target_sets, subsets):
    """
    Check if each subset in subsets is a subset of any set in target_sets.
    Returns 1 if all subsets exist, otherwise 0.
    """
    target_sets = [set(item[0]) for item in target_sets]
    for subset in subsets:
        if not any(set(subset).issubset(target_set) for target_set in target_sets):
            return 0
    return 1

def calculate_initial_item_frequency(transactions):
    """
    Calculate the initial frequency of items in transactions.
    """
    item_frequency = {}
    for items in transactions.values():
        for item in items:
            item_frequency[item] = item_frequency.get(item, 0) + 1
    return item_frequency

def prune_infrequent_items(item_frequency, transaction_count, threshold):
    """
    Calculate support and prune infrequent items based on the threshold.
    """
    return {item: freq / transaction_count for item, freq in item_frequency.items() if freq / transaction_count >= threshold}

def generate_candidate_itemsets(frequent_itemsets, current_length):
    """
    Generate candidate item sets of a given length.
    """
    candidate_itemsets = []
    for i in range(len(frequent_itemsets)):
        for j in range(i + 1, len(frequent_itemsets)):
            if current_length == 2:
                candidate = [frequent_itemsets[i][0], frequent_itemsets[j][0]]
            else:
                candidate = list(set(frequent_itemsets[i][0]) | set(frequent_itemsets[j][0]))
            
            if len(candidate) != current_length:
                continue
            
            candidate.sort()
            if is_subset_exist(frequent_itemsets, get_subsets(candidate)) and candidate not in candidate_itemsets:
                candidate_itemsets.append(candidate)
    return candidate_itemsets

def calculate_support(candidate_itemsets, transactions, transaction_count, threshold):
    """
    Calculate support for candidate item sets and prune infrequent sets.
    """
    new_frequent_itemsets = []
    for candidate in candidate_itemsets:
        support = sum(1 for items in transactions.values() if set(candidate).issubset(set(items))) / transaction_count
        if support >= threshold:
            new_frequent_itemsets.append([candidate, support])
    return new_frequent_itemsets

def apriori(transactions, threshold, max_length):
    """
    Apriori algorithm for frequent item set mining.
    Args:
    transactions (dict): Dictionary with transaction index as key and transaction details as value.
    threshold (float): Support threshold.
    max_length (int): Maximum length of item sets to be considered.
    
    Returns:
    list: List of frequent item sets with their support values.
    """
    transaction_count = len(transactions)
    item_frequency = calculate_initial_item_frequency(transactions)
    item_frequency = prune_infrequent_items(item_frequency, transaction_count, threshold)
    
    frequent_itemsets = [[item, support] for item, support in item_frequency.items()]
    
    if max_length == 1:
        return frequent_itemsets
    
    current_length = 2
    while current_length <= max_length:
        candidate_itemsets = generate_candidate_itemsets(frequent_itemsets, current_length)
        frequent_itemsets = calculate_support(candidate_itemsets, transactions, transaction_count, threshold)
        current_length += 1
    
    return frequent_itemsets

if __name__ == "__main__":
    # Example transaction data
    transactions = {
        0: ['A', 'B', 'C'],
        1: ['A', 'C'],
        2: ['A', 'D'],
        3: ['B', 'C'],
        4: ['A', 'B', 'C', 'D']
    }
    
    # Minimum support threshold
    threshold = 0.5
    
    # Maximum length of item sets
    max_length = 3
    
    # Run Apriori algorithm
    frequent_itemsets = apriori(transactions, threshold, max_length)
    
    # Print the result
    print("Frequent Itemsets:")
    for itemset in frequent_itemsets:
        print(f"Itemset: {itemset[0]}, Support: {itemset[1]:.2f}")
