import sys
import csv
import time
from collections import defaultdict
from itertools import chain, combinations

def data_gen(fname):
    """
    Read from the file and yield a generator.
    
    Args:
        fname (str): Path to the input file.
        
    Yields:
        frozenset: A transaction as a frozenset of items.
    """
    with open(fname, 'r') as file_iter:
        for line in file_iter:
            line = line.strip().rstrip(',')  # Remove trailing comma
            record = frozenset(line.split(','))
            yield record

def transactions(data_iterator):
    """
    Generate transaction list and 1-item sets.
    
    Args:
        data_iterator (iterator): An iterator of transactions.
        
    Returns:
        tuple: A set of 1-item sets and a list of transactions.
    """
    transaction_list = []
    items = set()
    for record in data_iterator:
        transaction_list.append(record)
        for item in record:
            items.add(frozenset([item]))  # Generate 1-item sets
    return items, transaction_list

def join_set(item_set, length):
    """
    Join sets to return n-length item sets where n=length.
    
    Args:
        item_set (set): Set of item sets.
        length (int): Desired length of joined item sets.
        
    Returns:
        set: Joined item sets of specified length.
    """
    return set([i.union(j) for i in item_set for j in item_set if len(i.union(j)) == length])

def filter_by_support(item_list, transaction_list, min_support):
    """
    Filter item sets to ensure they meet the minimum support.
    
    Args:
        item_list (set): Set of item sets.
        transaction_list (list): List of transactions.
        min_support (float): Minimum support threshold.
        
    Returns:
        set: Item sets that meet the minimum support.
    """
    item_count = defaultdict(int)
    frequent_item_set = set()
    
    for item in item_list:
        for transaction in transaction_list:
            if item.issubset(transaction):
                item_count[item] += 1
                
    for item, count in item_count.items():
        support = float(count) / len(transaction_list)
        if support >= min_support:
            frequent_item_set.add(item)
            freq_set[item] = support
    
    return frequent_item_set

def get_subsets(items):
    """
    Return non-empty subsets of items.
    
    Args:
        items (iterable): An iterable of items.
        
    Returns:
        iterator: An iterator of non-empty subsets of items.
    """
    return chain(*[combinations(items, i + 1) for i in range(len(items))])

def run_apriori(data_iter, min_support, min_confidence):
    """
    Run the Apriori algorithm.
    
    Args:
        data_iter (iterator): An iterator of transactions.
        min_support (float): Minimum support threshold.
        min_confidence (float): Minimum confidence threshold.
    """
    item_set, transaction_list = transactions(data_iter)
    large_set = {}
    one_c_set = filter_by_support(item_set, transaction_list, min_support)
    current_l_set = one_c_set
    k = 2

    while current_l_set:
        large_set[k - 1] = current_l_set
        current_l_set = join_set(current_l_set, k)
        current_c_set = filter_by_support(current_l_set, transaction_list, min_support)
        current_l_set = current_c_set
        k += 1

    for key, value in list(large_set.items())[1:]:
        for item in value:
            subsets = map(frozenset, [x for x in get_subsets(item)])
            for element in subsets:
                remain = item.difference(element)
                if remain:  # Ensure subsets pass through except the original set
                    confidence = freq_set[item] / freq_set[element]
                    if confidence >= min_confidence:
                        print(f"{element} -> {remain} (Confidence: {confidence:.2f})")

    start_time = time.time()

    min_support = 0.01

    input_file = 'T10I4D100k-copy.txt'
    data_iterator = data_gen(input_file)

    freq_set = defaultdict(int)
    run_apriori(data_iterator, min_support, min_confidence)
