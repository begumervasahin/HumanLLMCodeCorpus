def read_from_file(fname):
    """
    Read dataset from a file and return as a list of transactions.

    Args:
        fname (str): Path to the input file.

    Returns:
        list: List of transactions, where each transaction is a list of items.
    """
    with open(fname, 'r') as file:
        dataset = file.read().split('\n')
    return [line.split(':')[1].split(',') for line in dataset if not (line.startswith('#') or line.strip() == "")]

def prune(item_set, threshold):
    """
    Remove items/sets whose frequency/count/support is less than the threshold.

    Args:
        item_set (dict): Dictionary of item sets and their counts.
        threshold (int): Minimum support threshold.

    Returns:
        dict: Pruned dictionary of item sets and their counts.
    """
    return {item: count for item, count in item_set.items() if count >= threshold}

def combine_sets(frequent_items):
    """
    Generate new item sets by combining frequent item sets.

    Args:
        frequent_items (list): List of frequent item sets.

    Returns:
        list: List of new item sets generated by combining frequent item sets.
    """
    new_item_set = []
    for i in range(len(frequent_items)):
        for j in range(i + 1, len(frequent_items)):
            new_set = frequent_items[i].union(frequent_items[j])
            new_item_set.append(new_set)
    return new_item_set

def get_count(new_item_set, database):
    """
    Calculate the frequency of new item sets in the database.

    Args:
        new_item_set (list): List of new item sets.
        database (list): List of transactions.

    Returns:
        dict: Dictionary of new item sets and their counts.
    """
    item_set = {}
    for item in new_item_set:
        for entry in database:
            if item.issubset(entry):
                if item in item_set:
                    item_set[item] += 1
                else:
                    item_set[item] = 1
    return item_set

def initial_frequency_count(database):
    """
    Get the initial frequency count for every item in the database.

    Args:
        database (list): List of transactions.

    Returns:
        dict: Dictionary of items and their counts.
    """
    item_set = {}
    for entry in database:
        for item in entry:
            item = frozenset([item])
            if item in item_set:
                item_set[item] += 1
            else:
                item_set[item] = 1
    return item_set

def display_frequent_items(frequent_items):
    """
    Display frequent items.

    Args:
        frequent_items (dict): Dictionary of frequent item sets.
    """
    print('\n**** Frequent Items ****')
    for item in frequent_items:
        print(item)

def display_frequency_count(item_set):
    """
    Display the frequency count of item sets.

    Args:
        item_set (dict): Dictionary of item sets and their counts.
    """
    print('\n*** Frequency Count ***')
    for item, count in item_set.items():
        print(item, ':', count)

def display_initial_database(database):
    """
    Display the initial database.

    Args:
        database (list): List of transactions.
    """
    print("Initial Database:")
    for entry in database:
        print(entry)

def main():
    input_file = 'input.dat'
    database = read_from_file(input_file)

    # Convert database entries to frozensets
    database = [frozenset(entry) for entry in database]

    display_initial_database(database)

    item_set = initial_frequency_count(database)
    
    print('\n*** Initial Frequency Count ***')
    display_frequency_count(item_set)

    threshold = int(input('Enter Support Threshold: '))
    print('Threshold =', threshold)

    previous_item_set = item_set

    while True:
        # Create a list of items/sets whose support is greater than the threshold
        frequent_items = prune(item_set, threshold)
        
        display_frequent_items(frequent_items)

        # Stop if no frequent items are left
        if len(frequent_items) < 1:
            frequent_items = previous_item_set
            break

        print("\n*** New Iteration ***")
        # Generate new item sets from the current frequent items
        new_item_set = combine_sets(list(frequent_items.keys()))

        previous_item_set = frequent_items

        # Get the frequency of new item sets in the database
        item_set = get_count(new_item_set, database)

        display_frequency_count(item_set)

    print('\n***************************************************************')
    print('The most frequently associated items are:')
    for item in frequent_items:
        print('{', ', '.join(item), '}')
    print('\n')

if __name__ == "__main__":
    main()
