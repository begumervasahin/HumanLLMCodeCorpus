def generate_level2_candidates(L, phi, MIS, support_count):
    """
    Generate all frequent 2-itemsets.

    Parameters:
    L (list): List of frequent 1-itemsets.
    phi (float): Support difference constraint.
    MIS (dict): Minimum item support for each item.
    support_count (dict): Support count for each item.

    Returns:
    list: List of frequent 2-itemsets.
    """
    candidates = []
    for i, l in enumerate(L):
        if support_count[l] >= MIS[l]:
            for h in L[i + 1:]:
                if support_count[h] >= MIS[l] and abs(support_count[h] - support_count[l]) <= phi:
                    candidates.append((l, h))
    return candidates

def generate_MS_candidates(F, phi, support_count, k, MIS):
    """
    Generate candidate itemsets by joining and pruning.

    Parameters:
    F (list): List of frequent k-itemsets.
    phi (float): Support difference constraint.
    support_count (dict): Support count for each item.
    k (int): Size of the itemsets.
    MIS (dict): Minimum item support for each item.

    Returns:
    list: List of candidate (k+1)-itemsets.
    """
    candidates = []
    for f1 in F:
        for f2 in F:
            # Join step: Check if the k-1 prefixes are the same and last items are ordered
            if f1 != f2 and f1[:-1] == f2[:-1] and f1[-1] < f2[-1] and abs(support_count[f1[-1]] - support_count[f2[-1]]) <= phi:
                candidate = f1 + (f2[-1],)
                candidates.append(candidate)

                # Prune step: Ensure all (k-1)-subsets of the candidate are frequent
                for i in range(len(candidate)):
                    subset = candidate[:i] + candidate[i + 1:]
                    if candidate[0] in subset or MIS[candidate[1]] == MIS[candidate[0]]:
                        if subset not in F:
                            candidates.remove(candidate)
                            break
    return candidates
