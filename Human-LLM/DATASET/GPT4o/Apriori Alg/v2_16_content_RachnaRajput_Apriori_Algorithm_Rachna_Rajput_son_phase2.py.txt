import json
import sys
import MapReduce

# Initialize the MapReduce object
mr = MapReduce.MapReduce()
intermediate_dict = {}

class Phase2:
    @staticmethod
    def mapper(record):
        """
        Mapper function to process each record and emit intermediate key-value pairs.
        
        Args:
            record (list): The record to process.
        """
        with open(sys.argv[1]) as f:
            for line in f:
                chunkdata = json.loads(line)
                count = 0
                length = len(chunkdata)
                
                # Count occurrences of the record in chunkdata
                for chunkline in chunkdata:
                    if set(chunkline) >= set(record):
                        count += 1
                
                # Create a list with the count and the length of chunkdata
                mylist = [count, length]
                
                # Convert the record to a tuple to use as a key
                recordtuple = tuple(record)
                
                # Store the intermediate result
                intermediate_dict[recordtuple] = mylist
                
                # Emit the intermediate key-value pair
                mr.emit_intermediate(recordtuple, mylist)

    @staticmethod
    def reducer(key, values):
        """
        Reducer function to process intermediate key-value pairs and emit final results.
        
        Args:
            key (tuple): The key for the intermediate values.
            values (list): The list of intermediate values.
        """
        # Sum the counts for the given key
        count = sum(value[0] for value in values)
        
        # Create the final result list
        result = [key, count]
        
        # Emit the final result
        mr.emit(result)

if __name__ == '__main__':
    # Open the input data file
    inputdata = open(sys.argv[2])
    
    # Execute the MapReduce process with the mapper and reducer
    output = mr.execute(inputdata, Phase2.mapper, Phase2.reducer)
