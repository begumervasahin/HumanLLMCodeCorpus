# Apriori Algorithm Implementation

# Importing the necessary libraries
import numpy as np
import pandas as pd
from apyori import apriori

# Load and preprocess the dataset
def load_dataset(file_path):
    """Load the dataset and preprocess it into a list of transactions."""
    dataset = pd.read_csv(file_path, header=None)
    transactions = [
        [str(dataset.values[i, j]) for j in range(dataset.shape[1])]
        for i in range(len(dataset))
    ]
    return transactions

# Apply Apriori algorithm
def apply_apriori(transactions, min_support, min_confidence, min_lift, min_length):
    """Apply the Apriori algorithm to find frequent itemsets and association rules."""
    rules = apriori(
        transactions,
        min_support=min_support,
        min_confidence=min_confidence,
        min_lift=min_lift,
        min_length=min_length
    )
    return list(rules)

# Main function to run the Apriori algorithm
def main():
    # Define parameters
    file_path = 'Market_Basket_Optimisation.csv'
    min_support = 0.003
    min_confidence = 0.2
    min_lift = 3
    min_length = 2

    # Load dataset
    transactions = load_dataset(file_path)

    # Apply Apriori
    rules = apply_apriori(transactions, min_support, min_confidence, min_lift, min_length)

    # Display results
    for rule in rules:
        print(rule)

if __name__ == "__main__":
    main()
