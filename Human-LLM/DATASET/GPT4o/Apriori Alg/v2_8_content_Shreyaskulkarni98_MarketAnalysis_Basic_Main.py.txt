import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from apyori import apriori

def load_dataset(file_path):
    """
    Load the dataset from a CSV file.
    
    Args:
    file_path (str): The path to the CSV file.
    
    Returns:
    DataFrame: The loaded dataset.
    """
    return pd.read_csv(file_path, header=None)

def preprocess_data(data):
    """
    Convert the pandas DataFrame into a list of lists for the apriori algorithm.
    
    Args:
    data (DataFrame): The dataset.
    
    Returns:
    list: A list of transactions.
    """
    records = []
    for i in range(len(data)):
        records.append([str(data.values[i, j]) for j in range(data.shape[1])])
    return records

def apply_apriori(transactions, min_support, min_confidence, min_lift, min_length):
    """
    Apply the apriori algorithm to find association rules.
    
    Args:
    transactions (list): The list of transactions.
    min_support (float): The minimum support.
    min_confidence (float): The minimum confidence.
    min_lift (float): The minimum lift.
    min_length (int): The minimum length of the itemsets.
    
    Returns:
    list: The list of association rules.
    """
    return list(apriori(transactions, min_support=min_support, 
                        min_confidence=min_confidence, min_lift=min_lift, 
                        min_length=min_length))

def print_association_rules(rules):
    """
    Print the association rules found by the apriori algorithm.
    
    Args:
    rules (list): The list of association rules.
    """
    print(f"Number of association rules: {len(rules)}")

    if rules:
        print(f"First association rule: {rules[0]}")

    for item in rules:
        # First index of the inner list contains the base item and the add item
        pair = item[0]
        items = [x for x in pair]
        print(f"Rule: {items[0]} -> {items[1]}")

        # Second index of the inner list contains the support
        print(f"Support: {item[1]}")

        # Third index of the list located at 0th of the third index of the inner list
        print(f"Confidence: {item[2][0][2]}")
        print(f"Lift: {item[2][0][3]}")
        print("=====================================")

def main():
    # Path to the dataset
    file_path = 'D:/Shreyas Kulkarni/Documents/PycharmProjects/MarketAnalysis_Apriori_Basic/store_data.csv'

    # Load the dataset
    store_data = load_dataset(file_path)

    # Display the first few rows of the dataset
    print(store_data.head())

    # Preprocess the data
    records = preprocess_data(store_data)

    # Apply the apriori algorithm
    association_rules = apply_apriori(records, min_support=0.0045, min_confidence=0.2, min_lift=3, min_length=2)

    # Print the association rules
    print_association_rules(association_rules)

if __name__ == "__main__":
    main()
