import nltk
from nltk.corpus import wordnet as wn
from nltk import pos_tag
from nltk.tokenize import word_tokenize

# Ensure you have the necessary NLTK data
nltk.download('wordnet')
nltk.download('averaged_perceptron_tagger')
nltk.download('punkt')

def get_hyponyms(noun):
    """
    Get hyponyms for a given noun.
    
    :param noun: The noun for which to find hyponyms.
    :return: A list of lists containing hyponyms.
    """
    # Retrieve synsets that contain the noun
    synsets_with_noun = wn.synsets(noun)
    
    # Filter synsets to those that explicitly contain the noun in their lemmas
    filtered_synsets = [synset for synset in synsets_with_noun if noun in synset.lemma_names()]
    
    # Function to get hyponyms of a synset
    def hyponyms_closure(synset):
        return synset.hyponyms()
    
    # Gather hyponyms for each filtered synset
    hyponyms_list = [list(synset.closure(hyponyms_closure, depth=1)) for synset in filtered_synsets]
    
    return hyponyms_list

def get_hypernyms(noun):
    """
    Get hypernyms for a given noun.
    
    :param noun: The noun for which to find hypernyms.
    :return: A tuple containing synset strings and a list of lists of hypernyms.
    """
    # Retrieve synsets that contain the noun
    synsets_with_noun = wn.synsets(noun)
    
    # Extract strings representing synsets that contain the noun in their lemmas
    synset_strings = [str(synset) for synset in synsets_with_noun if noun in synset.lemma_names()]
    
    # Function to get hypernyms of a synset
    def hypernyms_closure(synset):
        return synset.hypernyms()
    
    # Gather hypernyms for each synset string
    hypernyms_list = [list(wn.synset(synset_str).closure(hypernyms_closure, depth=8)) for synset_str in synset_strings]
    
    return synset_strings, hypernyms_list

def universal_pos_converter(tag):
    """
    Convert universal POS tags to WordNet POS tags.
    
    :param tag: Universal POS tag.
    :return: Corresponding WordNet POS tag, or None if no match.
    """
    if tag.startswith("NN"):
        return "n"
    elif tag.startswith("VB"):
        return "v"
    elif tag.startswith("JJ"):
        return "a"
    elif tag.startswith("RB"):
        return "r"
    else:
        return None

def lemmatize_word(word_tuple):
    """
    Lemmatize a word based on its POS tag.
    
    :param word_tuple: A tuple containing the word and its POS tag.
    :return: A tuple containing the lemmatized word and its POS tag.
    """
    word, tag = word_tuple
    lemma = wn.morphy(word, universal_pos_converter(tag))
    
    # Return the original word if no lemma is found
    if lemma is None:
        return word, tag
    else:
        return lemma, tag

def tag_text(corpus):
    """
    Lemmatize each word in the corpus.
    
    :param corpus: A list of sentences (strings).
    :return: A list of lemmatized sentences (lists of tuples).
    """
    lemmatized_corpus = []
    for sentence in corpus:
        # Tokenize the sentence and get POS tags
        tokens = word_tokenize(sentence)
        pos_tags = pos_tag(tokens)
        
        # Lemmatize each word in the sentence
        lemmatized_sentence = [lemmatize_word(word) for word in pos_tags]
        lemmatized_corpus.append(lemmatized_sentence)
    
    return lemmatized_corpus

# Example usage
if __name__ == "__main__":
    # Example noun
    noun = "dog"
    print(f"Hyponyms of '{noun}':")
    hyponyms = get_hyponyms(noun)
    for idx, hyps in enumerate(hyponyms):
        print(f"Synset {idx + 1}: {[hyp.name() for hyp in hyps]}")
    
    print(f"Hypernyms of '{noun}':")
    synset_strings, hypernyms_list = get_hypernyms(noun)
    for idx, (synset_str, hypernyms) in enumerate(zip(synset_strings, hypernyms_list)):
        print(f"Synset {idx + 1}: {synset_str}")
        print(f"Hypernyms: {[hyp.name() for hyp in hypernyms]}")
    
    # Example corpus
    corpus = ["The quick brown fox jumps over the lazy dog.", "The dog barked loudly."]
    lemmatized_text = tag_text(corpus)
    print("Lemmatized text:")
    for sentence in lemmatized_text:
        print(sentence)
