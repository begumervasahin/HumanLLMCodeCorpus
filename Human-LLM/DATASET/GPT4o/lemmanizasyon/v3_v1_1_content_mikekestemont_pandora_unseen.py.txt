from __future__ import print_function
import os
import codecs
from pandora.tagger import Tagger
import pandora.utils

def main():
    """
    Main function to load a trained Tagger model, annotate text files, and save the annotations.
    """
    print('::: started :::')

    # Load the trained Tagger model from the specified directory
    tagger = load_tagger_model('models/new_cg')
    print('Tagger loaded, now annotating...')

    # Set the paths for input and output directories
    input_directory = 'data/limborch/orig/'
    output_directory = 'data/limborch/tagged/'

    # Ensure the output directory exists
    os.makedirs(output_directory, exist_ok=True)

    # Process each file in the input directory
    process_files_in_directory(input_directory, output_directory, tagger)

    print('::: ended :::')

def load_tagger_model(model_directory):
    """
    Loads the trained Tagger model from the specified directory.

    Parameters:
    model_directory (str): The directory containing the trained model.

    Returns:
    Tagger: The loaded Tagger model.
    """
    return Tagger(load=True, model_dir=model_directory)

def process_files_in_directory(input_directory, output_directory, tagger):
    """
    Processes all text files in the input directory by loading, annotating, and saving the annotations.

    Parameters:
    input_directory (str): The directory containing the original files.
    output_directory (str): The directory to save the annotated files.
    tagger (Tagger): The loaded Tagger model.
    """
    for filename in os.listdir(input_directory):
        if filename.endswith('.txt'):
            process_file(input_directory, output_directory, filename, tagger)

def process_file(input_directory, output_directory, filename, tagger):
    """
    Processes a single file: loads, annotates, and writes the annotations.

    Parameters:
    input_directory (str): The directory containing the original files.
    output_directory (str): The directory to save the annotated files.
    filename (str): The name of the file to process.
    tagger (Tagger): The loaded Tagger model.
    """
    print(f'\t + {filename}')

    # Load unannotated data from the original file
    unannotated_data = load_unannotated_data(os.path.join(input_directory, filename))

    # Annotate the tokens using the loaded tagger
    annotations = tagger.annotate(unannotated_data)

    # Write annotated data to a new file
    output_file_path = os.path.join(output_directory, filename)
    write_annotations_to_file(output_file_path, annotations)

def load_unannotated_data(file_path):
    """
    Loads unannotated data from a file.

    Parameters:
    file_path (str): The path to the file to be loaded.

    Returns:
    list: The unannotated data.
    """
    return pandora.utils.load_unannotated_file(
        file_path,
        nb_instances=None,
        tokenized_input=False
    )

def write_annotations_to_file(output_file_path, annotations):
    """
    Writes the annotated data to a file.

    Parameters:
    output_file_path (str): The path to the output file.
    annotations (dict): The annotated data containing tokens, lemmas, and POS tags.
    """
    with codecs.open(output_file_path, 'w', 'utf8') as file:
        for token, lemma, pos in zip(annotations['tokens'], annotations['lemmas'], annotations['pos']):
            file.write('\t'.join((token.lower(), lemma.lower(), pos.lower())) + '\n')

if __name__ == '__main__':
    main()
