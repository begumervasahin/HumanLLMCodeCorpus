import os
import pandas as pd
import csv
import re

# Define constants
DATA_DIR = "data"
MBTI_CLEAN_CSV_PATH = os.path.join(DATA_DIR, "mbti_clean.csv")
DIMENSIONS = ["IE", "NS", "TF", "PJ"]

def filter_and_save_posts(df, dimension):
    """
    Filters posts based on the given MBTI dimension and saves them to a CSV file.

    Parameters:
    - df (DataFrame): The DataFrame containing MBTI types and posts.
    - dimension (str): The MBTI dimension to filter posts by (e.g., "IE", "NS").
    """
    # Extract individual letters for the given dimension
    letter_1, letter_2 = dimension
    filtered_posts = []

    # Iterate through each row in the DataFrame
    for _, row in df.iterrows():
        # Check if the row's MBTI type contains either of the specified letters
        if letter_1 in row["type"] or letter_2 in row["type"]:
            # Split the posts into a list
            posts = row["posts"].split("|||")

            # Filter the posts
            for post in posts:
                # Skip unwanted posts
                if "http" in post or not re.search("[a-zA-Z]", post):
                    continue

                # Append the filtered post to the list
                filtered_posts.append(post)

    # Save the filtered posts to a CSV file
    output_csv_path = os.path.join(DATA_DIR, f"filtered_posts_{dimension}.csv")
    with open(output_csv_path, "w", newline="", encoding="utf-8") as file:
        writer = csv.writer(file)
        # Write each post as a separate row in the CSV file
        writer.writerows([[post] for post in filtered_posts])

def main():
    """
    Main function to read the MBTI CSV file and filter/save posts for each dimension.
    """
    # Read the MBTI clean CSV file into a DataFrame
    df = pd.read_csv(MBTI_CLEAN_CSV_PATH)

    # Iterate through each dimension and filter/save posts
    for dimension in DIMENSIONS:
        filter_and_save_posts(df, dimension)

# Execute the main function if the script is run directly
if __name__ == "__main__":
    main()
