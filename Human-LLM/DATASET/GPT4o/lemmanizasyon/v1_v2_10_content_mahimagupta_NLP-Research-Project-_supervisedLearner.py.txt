# Define input and output file paths
hindi_dic_file = './datasets/hi_hdtb-ud-dev.conllu'
hindi_out_file = './outputs/train_data.txt'

# Function to check if a lemma is already present in the dataset
def is_lemma_in(lemmas, lemma):
    for l in lemmas:
        if l["root"] != lemma["root"] or l["tag1"] != lemma["tag1"] or l["tag2"] != lemma["tag2"]:
            continue
        return True
    return False

# Function to serialize a lemma into a string
def serialize_lemma(lemma):
    return '[{}, {}, {}]'.format(lemma["root"], lemma["tag1"], lemma["tag2"])

# Function to read training data from a file
def read_training_data(input_training_file, dest_file=None):
    dataset = {}
    with open(input_training_file, 'r', encoding='utf-8') as hindi_file:
        for line in hindi_file.readlines():
            if len(line) < 5 or line[0] == '#':
                continue
            word, root, tag1, tag2 = line.split("\t")[1:1+4]
            if word not in dataset:
                dataset[word] = []
            lemma = {
                "root": root, "tag1": tag1, "tag2": tag2,
            }
            if is_lemma_in(dataset[word], lemma):
                continue
            dataset[word].append(lemma)

    if dest_file is not None:
        with open(dest_file, 'w', encoding='utf-8') as output_file:
            lines = []
            for word, lemmas in dataset.items():
                lines.append(
                    '{}:{}\n'.format(word, ', '.join([serialize_lemma(lemma) for lemma in lemmas]))
                )
            output_file.writelines(lines)

    return dataset

# Function to perform supervised learning
def supervised_learn():
    training_data = read_training_data(hindi_dic_file, hindi_out_file)
    for word, lemmas in training_data.items():
        print('Processing word:', word, 'with', len(lemmas), 'lemmas')

# Execute the supervised learning function
supervised_learn()
