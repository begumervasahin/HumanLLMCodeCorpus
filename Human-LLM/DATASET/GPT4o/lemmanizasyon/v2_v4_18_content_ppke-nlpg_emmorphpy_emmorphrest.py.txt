from xtsv import pipeline_rest_api, singleton_store_factory

# Define the task configuration for stemming using EmMorphPy
em_morph_stem = (
    'emmorphpy.emmorphpy',  # Path to the module
    'EmMorphPy',            # Class name to use
    'stem',                 # Method to call on the class
    (),                     # Arguments to pass to the method (none in this case)
    {
        'task': 'stem',              # Task name
        'source_fields': {'form'},   # Input field(s)
        'target_fields': ['anas']    # Output field(s)
    }
)

# Define the task configuration for analyzing using EmMorphPy
em_morph_analyze = (
    'emmorphpy.emmorphpy',  # Path to the module
    'EmMorphPy',            # Class name to use
    'analyze',              # Method to call on the class
    (),                     # Arguments to pass to the method (none in this case)
    {
        'task': 'analyze',           # Task name
        'source_fields': {'form'},   # Input field(s)
        'target_fields': ['anas']    # Output field(s)
    }
)

# Define the task configuration for dstemming using EmMorphPy
em_morph_dstem = (
    'emmorphpy.emmorphpy',  # Path to the module
    'EmMorphPy',            # Class name to use
    'dstem',                # Method to call on the class
    (),                     # Arguments to pass to the method (none in this case)
    {
        'task': 'dstem',             # Task name
        'source_fields': {'form'},   # Input field(s)
        'target_fields': ['anas']    # Output field(s)
    }
)

# List of tools with their respective tasks
tools = [
    (em_morph_stem, ('stem',)),
    (em_morph_analyze, ('analyze',)),
    (em_morph_dstem, ('dstem',)),
]

# Configure the REST API pipeline
app = pipeline_rest_api(
    'emMorph',                           # API name
    tools,                               # List of tools and tasks
    {},                                  # Additional configurations (empty in this case)
    conll_comments=False,                # Disable CoNLL comments
    singleton_store=singleton_store_factory(),  # Singleton store for the pipeline
    form_title='emMorph demo',           # Title for the form
    form_type='radio',                   # Form type (radio buttons)
    doc_link='https://github.com/ppke-nlpg/emmorphpy'  # Documentation link
)

# Entry point to run the application
if __name__ == '__main__':
    app.run(debug=False)  # Run the app with debug mode disabled
