# File paths
hindi_dic_file = './datasets/hi_hdtb-ud-dev.conllu'
hindi_out_file = './outputs/train_data.txt'

def lemma_exists(lemmas, lemma):
    """
    Check if a given lemma already exists in the list of lemmas.

    Parameters:
    lemmas (list): List of existing lemmas.
    lemma (dict): Lemma to check.

    Returns:
    bool: True if lemma exists, False otherwise.
    """
    return any(
        existing_lemma["root"] == lemma["root"] and 
        existing_lemma["tag1"] == lemma["tag1"] and 
        existing_lemma["tag2"] == lemma["tag2"]
        for existing_lemma in lemmas
    )

def serialize_lemma(lemma):
    """
    Convert a lemma dictionary into a string representation.

    Parameters:
    lemma (dict): Lemma dictionary.

    Returns:
    str: String representation of the lemma.
    """
    return '[{}, {}, {}]'.format(lemma["root"], lemma["tag1"], lemma["tag2"])

def read_training_data(input_file, output_file=None):
    """
    Read the training data from a file and optionally write the processed data to an output file.

    Parameters:
    input_file (str): Path to the input file.
    output_file (str, optional): Path to the output file. Defaults to None.

    Returns:
    dict: Processed dataset.
    """
    dataset = {}

    with open(input_file, 'r', encoding='utf-8') as file:
        for line in file:
            # Skip short lines and comments
            if len(line) < 5 or line.startswith('#'):
                continue

            parts = line.split("\t")
            if len(parts) < 5:
                continue

            word, root, tag1, tag2 = parts[1:5]

            if word not in dataset:
                dataset[word] = []

            lemma = {"root": root, "tag1": tag1, "tag2": tag2}

            if not lemma_exists(dataset[word], lemma):
                dataset[word].append(lemma)

    if output_file:
        with open(output_file, 'w', encoding='utf-8') as out:
            for word, lemmas in dataset.items():
                serialized_lemmas = ', '.join(serialize_lemma(lemma) for lemma in lemmas)
                out.write('{}:{}\n'.format(word, serialized_lemmas))

    return dataset

def supervised_learn():
    """
    Main function to read and process the training data.
    """
    training_data = read_training_data(hindi_dic_file, hindi_out_file)
    for word, lemmas in training_data.items():
        print('Processing word:', word, 'with', len(lemmas), 'lemmas')

if __name__ == "__main__":
    supervised_learn()
