import os
import pandas as pd
import csv
import re

# Constants
DATA_DIR = "data"
MBTI_CLEAN_CSV_PATH = os.path.join(DATA_DIR, "mbti_clean.csv")
DIMENSIONS = ("IE", "NS", "TF", "PJ")

# Read the CSV file into a pandas DataFrame
df = pd.read_csv(MBTI_CLEAN_CSV_PATH)

def filter_posts(posts):
    """
    Filter out posts that contain URLs, are empty, or do not contain alphabetic characters.

    :param posts: List of posts to filter.
    :return: List of filtered posts.
    """
    return [post for post in posts if ("http" not in post) and post and re.search("[a-zA-Z]", post)]

def process_posts_for_letter(df, letter):
    """
    Extract and filter posts for rows where the specified letter is in the personality type.

    :param df: DataFrame containing the MBTI data.
    :param letter: Letter to filter posts by (e.g., 'I', 'E', 'N', 'S').
    :return: List of filtered posts for the specified letter.
    """
    posts = []
    for _, row in df.iterrows():
        if letter in row["type"]:
            hundred_posts = row["posts"].split("|||")
            posts.extend(filter_posts(hundred_posts))
    return posts

def write_posts_to_csv(posts, letter):
    """
    Write the filtered posts to a CSV file named according to the specified letter.

    :param posts: List of filtered posts to write to the CSV file.
    :param letter: Letter to use in the CSV file name.
    """
    csv_path = os.path.join(DATA_DIR, f"test_{letter}.csv")
    with open(csv_path, "w", newline='') as f:
        writer = csv.writer(f)
        for post in posts:
            writer.writerow([post])

# Main processing loop
for dimension in DIMENSIONS:
    for letter in dimension:
        posts = process_posts_for_letter(df, letter)
        write_posts_to_csv(posts, letter)
