import requests
from bs4 import BeautifulSoup

# Define the base URL of the news website
NEWS_URL = 'https://www.janmabhumidaily.com/news'

def fetch_news_content(url):
    """
    Fetches the content of the provided URL.

    Parameters:
    url (str): The URL of the news website.

    Returns:
    str: The content of the response if successful, None otherwise.
    """
    try:
        response = requests.get(url)
        response.raise_for_status()  # Raise an exception for HTTP errors
        return response.content
    except requests.RequestException as e:
        print(f"Error fetching content from {url}: {e}")
        return None

def extract_headlines(content):
    """
    Parses and extracts headlines from the news content.

    Parameters:
    content (str): The HTML content of the news website.

    Returns:
    list: A list of headlines found in the content.
    """
    soup = BeautifulSoup(content, 'html.parser')
    headlines = [headline.text.strip() for headline in soup.find_all('h2', class_='entry-title')]
    return headlines

def display_headlines(headlines):
    """
    Displays the extracted headlines.

    Parameters:
    headlines (list): A list of headlines to display.
    """
    if headlines:
        print("Headlines:")
        for headline in headlines:
            print(f"- {headline}")
    else:
        print("No headlines found.")

def main():
    """
    Main function to orchestrate fetching and parsing news content.
    """
    content = fetch_news_content(NEWS_URL)
    if content:
        headlines = extract_headlines(content)
        display_headlines(headlines)
    else:
        print("Failed to retrieve or parse news content.")

if __name__ == "__main__":
    main()
