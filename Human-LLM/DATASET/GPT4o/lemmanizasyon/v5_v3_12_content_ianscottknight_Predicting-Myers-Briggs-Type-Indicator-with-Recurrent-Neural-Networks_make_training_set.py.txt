import os
import collections
import pandas as pd
import csv
import re

# Constants
DATA_DIR = "data"
MBTI_UNCLEAN_CSV_PATH = os.path.join(DATA_DIR, "mbti_unclean.csv")
DIMENSIONS = ("IE", "NS", "TF", "PJ")

# Function to check if a post is valid
def is_valid_post(post):
    """Check if a post is valid by certain criteria."""
    return not any([
        "http" in post,
        post == "",
        post is None,
        not re.search("[a-zA-Z]", post)
    ])

# Function to count occurrences for each MBTI dimension
def count_mbti_occurrences(df, dimensions):
    """Count occurrences for each MBTI dimension in the dataframe."""
    counts = collections.defaultdict(int)

    for dimension in dimensions:
        letter_1, letter_2 = dimension

        for _, row in df.iterrows():
            mbti = row["type"]

            for post in row["posts"].split("|||"):
                if is_valid_post(post):
                    counts[letter_1] += int(letter_1 in mbti)
                    counts[letter_2] += int(letter_2 in mbti)
    return counts

# Function to extract posts for each MBTI dimension
def extract_posts(df, dimensions, counts, data_dir):
    """Extract posts for each MBTI dimension and save to CSV."""
    for dimension in dimensions:
        letter_1, letter_2 = dimension

        limit = min(counts[letter_1], counts[letter_2])
        posts = []
        i = 0

        for _, row in df.iterrows():
            if any([letter_1 in row["type"], letter_2 in row["type"]]):
                for post in row["posts"].split("|||"):
                    if is_valid_post(post):
                        posts.append(post)
                        i += 1

                        if i == limit:
                            break
            if i == limit:
                break

        train_csv_path = os.path.join(data_dir, f"train_{letter_1}.csv")
        with open(train_csv_path, "w", newline='') as f:
            writer = csv.writer(f)
            writer.writerows([[post] for post in posts])

# Main function to process MBTI data
def main():
    # Read CSV file into a pandas DataFrame
    df = pd.read_csv(MBTI_UNCLEAN_CSV_PATH)

    # Count occurrences for each MBTI dimension
    counts = count_mbti_occurrences(df, DIMENSIONS)

    # Extract posts for each MBTI dimension and save to CSV
    extract_posts(df, DIMENSIONS, counts, DATA_DIR)

# Run the main function
if __name__ == "__main__":
    main()
