from __future__ import print_function
import sys
import pandora.utils
from pandora.tagger import Tagger

def load_annotated_data(directory, params, nb_instances=None):
    """
    Load annotated data from a specified directory.

    Parameters:
    - directory: str, path to the directory containing the annotated data files.
    - params: dict, dictionary containing parameters for data loading.
    - nb_instances: int, number of instances to load (default is None, meaning load all).

    Returns:
    - list, loaded annotated data.
    """
    return pandora.utils.load_annotated_dir(
        directory,
        format='tab',
        extension='.tsv',
        include_pos=params.get('include_pos', False),
        include_lemma=params.get('include_lemma', False),
        include_morph=params.get('include_morph', False),
        nb_instances=nb_instances
    )

def train_tagger(config_path):
    """
    Train a tagger model using the specified configuration.

    Parameters:
    - config_path: str, path to the configuration file.
    """
    print('::: Training Started :::')

    # Load configuration parameters from the specified file
    params = pandora.utils.get_param_dict(config_path)
    params['config_path'] = config_path

    # Load training and development data
    train_data = load_annotated_data('data/medieval_latin/train', params)
    dev_data = load_annotated_data('data/medieval_latin/test', params)

    # Initialize and configure the tagger for training
    tagger = Tagger(**params)
    tagger.setup_to_train(train_data=train_data, dev_data=dev_data)

    # Train the tagger for the specified number of epochs
    nb_epochs = int(params['nb_epochs'])
    for epoch in range(nb_epochs):
        print(f'::: Epoch {epoch + 1}/{nb_epochs} :::')
        tagger.epoch()
        tagger.save()

    # Save the final trained model
    tagger.save()
    print('::: Training Ended :::')

if __name__ == '__main__':
    # Ensure the correct number of command-line arguments is provided
    if len(sys.argv) != 2:
        print('Usage: python script.py <config_file_path>')
        sys.exit(1)

    # Retrieve the configuration file path from the command-line argument
    config_path = sys.argv[1]

    # Start the training process
    train_tagger(config_path)
