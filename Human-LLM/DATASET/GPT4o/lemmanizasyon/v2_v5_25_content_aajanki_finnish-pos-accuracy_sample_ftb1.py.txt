import hashlib
import re

def main():
    input_file = 'data/test/ftb1/ftb1u.tsv'
    output_file = 'data/test/ftb1/ftb1u_sample.tsv'

    sentences = read_and_split_sentences(input_file)
    selected_sentences = filter_sentences_by_sha256(sentences)

    print(f'Sampled {len(selected_sentences)} out of {len(sentences)} sentences')

    write_sentences_to_file(output_file, selected_sentences)
    print(f'Wrote samples to {output_file}')

def read_and_split_sentences(input_file):
    """
    Reads an input file and splits its content into sentences.
    
    Args:
    - input_file (str): Path to the input file.
    
    Returns:
    - List of sentences (list of str).
    """
    with open(input_file) as file:
        lines = file.readlines()
        return list(split_sentences(lines))

def split_sentences(lines):
    """
    Splits lines into sentences based on a regular expression.
    
    Args:
    - lines (list of str): Lines from the input file.
    
    Yields:
    - Sentences (str).
    """
    sentence = []
    for line in lines:
        if re.match(r'^', line) and sentence:
            yield ''.join(sentence)
            sentence = []
        if line.strip():
            sentence.append(line)

    if sentence:
        yield ''.join(sentence)

def filter_sentences_by_sha256(sentences):
    """
    Filters sentences whose SHA256 hash starts with '0'.
    
    Args:
    - sentences (list of str): List of sentences.
    
    Returns:
    - Filtered sentences (list of str).
    """
    return [sentence for sentence in sentences if hashlib.sha256(sentence.encode('utf-8')).hexdigest().startswith('0')]

def write_sentences_to_file(output_file, sentences):
    """
    Writes sentences to an output file.
    
    Args:
    - output_file (str): Path to the output file.
    - sentences (list of str): List of sentences to write.
    """
    with open(output_file, 'w') as file:
        file.write('\n'.join(sentences))

if __name__ == '__main__':
    main()
