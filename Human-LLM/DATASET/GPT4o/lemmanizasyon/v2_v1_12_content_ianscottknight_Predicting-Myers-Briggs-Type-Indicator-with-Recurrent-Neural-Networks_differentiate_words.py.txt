import os
import collections
import re

# Constants
DATA_DIR = "data"
DIMENSIONS = ["IE", "NS", "FT", "PJ"]
WORDS_TO_REMOVE = [
    "intj", "intp", "infj", "infp", "istj", "istp", "isfj", "isfp",
    "entj", "entp", "enfj", "enfp", "estj", "estp", "esfj", "esfp",
    "si", "ni", "ti", "fi", "se", "ne", "te", "fe", "nt", "nf",
    "sxsp", "spsx", "spso", "sxso", "sosp", "sosx", "sp", "sx", "sj", "sf", "st",
    "le", "socionic", "socionics", "enneagram", "d", "w", "mbti",
]

# Regular expression to remove non-alphabetic characters
regex = re.compile("[^a-zA-Z]")

# Function to process each dimension pair
def process_dimension(dimension):
    # Initialize word count dictionaries
    wordcount_a = collections.Counter()
    wordcount_b = collections.Counter()

    # Read and count words for the first file
    with open(os.path.join(DATA_DIR, f"extreme_examples_{dimension[0]}.txt"), "r") as file_a:
        wordcount_a.update(file_a.read().split())

    # Read and count words for the second file
    with open(os.path.join(DATA_DIR, f"extreme_examples_{dimension[1]}.txt"), "r") as file_b:
        wordcount_b.update(file_b.read().split())

    # Combine keys from both word counts
    combined_keys = set(wordcount_a.keys()).union(set(wordcount_b.keys()))
    a_diff = {}
    b_diff = {}

    # Calculate differences
    for key in combined_keys:
        if key in wordcount_a and key not in wordcount_b:
            a_diff[key] = wordcount_a[key]
        elif key in wordcount_b and key not in wordcount_a:
            b_diff[key] = wordcount_b[key]
        elif key in wordcount_a and key in wordcount_b:
            diff = wordcount_a[key] - wordcount_b[key]
            if diff > 0:
                a_diff[key] = diff
            elif diff < 0:
                b_diff[key] = -diff

    # Write results to files
    with open(os.path.join(DATA_DIR, f"special_words_{dimension[0]}.txt"), "w") as file_a:
        for key, count in a_diff.items():
            cleaned_key = regex.sub("", key)
            if cleaned_key not in WORDS_TO_REMOVE and count > 2:
                file_a.write(cleaned_key + "\n")

    with open(os.path.join(DATA_DIR, f"special_words_{dimension[1]}.txt"), "w") as file_b:
        for key, count in b_diff.items():
            cleaned_key = regex.sub("", key)
            if cleaned_key not in WORDS_TO_REMOVE and count > 2:
                file_b.write(cleaned_key + "\n")

# Process each dimension pair
for dimension in DIMENSIONS:
    process_dimension(dimension)
