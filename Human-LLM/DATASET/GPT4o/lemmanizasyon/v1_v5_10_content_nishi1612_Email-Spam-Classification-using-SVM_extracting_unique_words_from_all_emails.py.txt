import os
import time
import string
import operator
import csv
from nltk.corpus import stopwords
from nltk.stem.wordnet import WordNetLemmatizer

def text_cleanup(text):
    """
    Clean up the input text by removing punctuation, stopwords, and converting to lowercase.

    Args:
    text (str): The text to be cleaned.

    Returns:
    list: A list of cleaned and lemmatized words.
    """
    # Remove punctuation
    text_without_punctuation = ''.join([char for char in text if char not in string.punctuation])
    
    # Remove stopwords
    stop_words = set(stopwords.words('english'))
    text_without_stopwords = ' '.join([word for word in text_without_punctuation.split() if word.lower() not in stop_words])
    
    # Convert to lowercase and lemmatize
    lemmatizer = WordNetLemmatizer()
    cleaned_text = [lemmatizer.lemmatize(word.lower()) for word in text_without_stopwords.split()]
    
    return cleaned_text

def process_files(directory_path):
    """
    Process all files in the specified directory, count word occurrences, and write the result to a CSV file.

    Args:
    directory_path (str): The path to the directory containing the email files.
    """
    start_time = time.time()
    file_count = 0
    word_count = {}

    # Get a list of files in the directory
    directory = os.fsencode(directory_path)
    
    for file in os.listdir(directory):
        file = file.decode("utf-8")
        file_path = os.path.join(os.getcwd(), 'emails', file)

        with open(file_path, "r", encoding='utf-8', errors='ignore') as file_reader:
            words = text_cleanup(file_reader.read())

            for word in words:
                # Check if the word is not a digit and has a length greater than 2
                if not word.isdigit() and len(word) > 2:
                    # Update word count
                    word_count[word] = word_count.get(word, 0) + 1

            file_count += 1
            if file_count % 100 == 0:
                print(f"Processed {file_count} files")

    # Sort the word count dictionary by frequency in descending order
    sorted_word_count = dict(sorted(word_count.items(), key=operator.itemgetter(1), reverse=True))

    # Write the sorted word count to a CSV file
    with open("wordslist.csv", "w+", newline='') as csv_file:
        writer = csv.writer(csv_file)
        writer.writerow(['word', 'count'])

        for word, count in sorted_word_count.items():
            if count < 100:
                break
            writer.writerow([word, count])

    print(f'Time taken to preprocess emails: {round(time.time() - start_time, 2)} seconds')

if __name__ == "__main__":
    process_files("emails/")
