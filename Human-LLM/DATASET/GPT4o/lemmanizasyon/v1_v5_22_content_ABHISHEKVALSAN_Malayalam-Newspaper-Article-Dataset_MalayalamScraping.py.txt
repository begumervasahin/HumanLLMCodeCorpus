import requests
from bs4 import BeautifulSoup
import os

# Define constants
URL_PREFIX = 'https://example.com/page/'  # Replace with the actual URL prefix

def month_to_digit(month):
    """Converts month name to digit."""
    month_dict = {
        'January': '01', 'February': '02', 'March': '03',
        'April': '04', 'May': '05', 'June': '06', 'July': '07',
        'August': '08', 'September': '09', 'October': '10',
        'November': '11', 'December': '12'
    }
    return month_dict.get(month, '00')

def convert_to_filename(date_str):
    """Converts a date string to a filename."""
    flist = date_str.split()
    month_digit = month_to_digit(flist[2])
    day = flist[1].zfill(2)  # Zero padding for day
    filename = "1" + flist[3][2:] + month_digit + day
    return filename, flist[3]

def fetch_data(index):
    """Fetches data from the website for a given index."""
    url = URL_PREFIX + str(index)
    response = requests.get(url)

    if response.status_code == 200:
        soup = BeautifulSoup(response.text, 'html.parser')
        links = soup.findAll("p", {"class": "text-center"})

        if len(links) != 1:
            title = soup.find("h1").text
            date = soup.findAll("div", {"class": "myd"})[1].text
            content = soup.find("p", {"class": ""}).text

            filename, year = convert_to_filename(date)
            path = "DataSet/"
            filename = f"{filename}{index}.utf8"
            filename_with_path = os.path.join(path, filename)

            with open(filename_with_path, 'w+', encoding='utf-8') as file:
                file.write(f"<DOC>\n<DOCNO>{filename}</DOCNO>\n<TEXT>\n{title}\n{date}\n{content}\n</TEXT>\n</DOC>")
            print(f"Writing file {filename} to {path}")
        else:
            print(f"File missing at {url}")
    else:
        print("Error fetching data from {url}")
