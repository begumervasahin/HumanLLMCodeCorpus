import requests
from bs4 import BeautifulSoup
import os
import constants as const

def convert_month_to_digit(month):
    """
    Convert a month name to its corresponding two-digit number as a string.
    
    Args:
    month (str): The name of the month.
    
    Returns:
    str: The two-digit representation of the month.
    """
    month_mapping = {
        'January': '01', 'February': '02', 'March': '03',
        'April': '04', 'May': '05', 'June': '06', 'July': '07',
        'August': '08', 'September': '09', 'October': '10',
        'November': '11', 'December': '12'
    }
    return month_mapping.get(month, '00')

def convert_string_to_filename(date_string):
    """
    Convert a date string to a formatted filename.
    
    Args:
    date_string (str): The date string in the format "Day Month Year".
    
    Returns:
    tuple: A tuple containing the filename and the year.
    """
    date_parts = date_string.split()
    month_digit = convert_month_to_digit(date_parts[2])
    day = f"{int(date_parts[1]):02d}"
    year_suffix = date_parts[3][2:]
    filename = f"1{year_suffix}{month_digit}{day}"
    return filename, date_parts[3]

def fetch_and_save_data(post_id):
    """
    Fetch data from a specified post and save it to a file.
    
    Args:
    post_id (int): The ID of the post to fetch data from.
    """
    url = f"{const.URL_PREFIX}{post_id}"
    response = requests.get(url)

    if response.status_code == 200:
        soup = BeautifulSoup(response.text, 'html.parser')
        text_elements = soup.find_all("p", {"class": "text-center"})

        if len(text_elements) != 1:
            title = soup.find("h1").text
            date_text = soup.find("div", {"class": "myd"}).text
            content = soup.find("p", {"class": ""}).text
            full_text = f"{title}\n{date_text}\n{content}"

            filename, year = convert_string_to_filename(date_text)
            directory = "DataSet/"
            full_filename = f"{filename}{post_id}.utf8"
            file_path = os.path.join(directory, full_filename)

            with open(file_path, 'w', encoding='utf-8') as file:
                file.write(f"<DOC>\n<DOCNO>{full_filename}</DOCNO>\n<TEXT>\n{full_text}\n</TEXT>\n</DOC>")
            
            print(f"Writing file {full_filename} to {directory}")
        else:
            print(f"File missing at {url}")
    else:
        print(f"Error fetching data from {url}")
