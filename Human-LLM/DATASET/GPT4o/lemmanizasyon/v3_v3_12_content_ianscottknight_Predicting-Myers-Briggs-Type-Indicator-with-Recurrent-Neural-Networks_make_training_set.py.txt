import os
import collections
import pandas as pd
import csv
import re

# Constants
DATA_DIR = "data"
MBTI_UNCLEAN_CSV_PATH = os.path.join(DATA_DIR, "mbti_unclean.csv")
DIMENSIONS = ("IE", "NS", "TF", "PJ")

# Read CSV file into a pandas DataFrame
df = pd.read_csv(MBTI_UNCLEAN_CSV_PATH)

# Initialize counts dictionary using defaultdict
counts = collections.defaultdict(int)

def is_valid_post(post):
    """
    Validate a post based on specific criteria:
    - Does not contain "http"
    - Is not an empty string
    - Is not None
    - Contains at least one alphabetical character
    """
    return not any([
        "http" in post,
        post == "",
        post is None,
        not re.search("[a-zA-Z]", post)
    ])

def count_mbti_dimension_occurrences(df, dimensions):
    """
    Count occurrences of each MBTI dimension in the dataset.
    
    Args:
        df (DataFrame): The dataset containing MBTI types and posts.
        dimensions (tuple): A tuple of MBTI dimensions to count.
    
    Returns:
        defaultdict: A dictionary with counts of each MBTI dimension.
    """
    counts = collections.defaultdict(int)
    for dimension in dimensions:
        letter_1, letter_2 = dimension
        for _, row in df.iterrows():
            mbti = row["type"]
            for post in row["posts"].split("|||"):
                if is_valid_post(post):
                    counts[letter_1] += int(letter_1 in mbti)
                    counts[letter_2] += int(letter_2 in mbti)
    return counts

def extract_and_save_posts(df, dimensions, counts, data_dir):
    """
    Extract posts for each MBTI dimension and save them to CSV files.
    
    Args:
        df (DataFrame): The dataset containing MBTI types and posts.
        dimensions (tuple): A tuple of MBTI dimensions to extract posts for.
        counts (defaultdict): A dictionary with counts of each MBTI dimension.
        data_dir (str): The directory to save the CSV files in.
    """
    for dimension in dimensions:
        letter_1, letter_2 = dimension
        limit = min(counts[letter_1], counts[letter_2])
        posts = []
        count = 0

        for _, row in df.iterrows():
            if any([letter_1 in row["type"], letter_2 in row["type"]]):
                for post in row["posts"].split("|||"):
                    if is_valid_post(post):
                        posts.append(post)
                        count += 1
                        if count == limit:
                            break
            if count == limit:
                break

        # Create a CSV file for the extracted posts
        train_csv_path = os.path.join(data_dir, f"train_{letter_1}.csv")
        with open(train_csv_path, "w", newline='', encoding='utf-8') as f:
            writer = csv.writer(f)
            writer.writerows([[post] for post in posts])

# Count occurrences for each MBTI dimension
counts = count_mbti_dimension_occurrences(df, DIMENSIONS)

# Extract posts for each MBTI dimension and save to CSV files
extract_and_save_posts(df, DIMENSIONS, counts, DATA_DIR)

print("CSV files created successfully.")
