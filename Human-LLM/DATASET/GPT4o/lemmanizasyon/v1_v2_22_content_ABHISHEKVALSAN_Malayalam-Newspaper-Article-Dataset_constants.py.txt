import requests
from bs4 import BeautifulSoup

# Define the URL of the news website
URL_PREFIX = 'https://www.janmabhumidaily.com/news'

# Function to fetch the content of the news website
def fetch_news_content(url):
    try:
        # Send a GET request to the provided URL
        response = requests.get(url)

        # Check if the request was successful (status code 200)
        if response.status_code == 200:
            # Return the content of the response
            return response.content
        else:
            # Print an error message if the request was not successful
            print("Failed to fetch content. Status code:", response.status_code)
            return None
    except Exception as e:
        # Print an error message if an exception occurs during the request
        print("An error occurred:", str(e))
        return None

# Function to parse and process the news content
def parse_news_content(content):
    if content:
        # Parse the content using BeautifulSoup
        soup = BeautifulSoup(content, 'html.parser')

        # Find all headlines on the webpage
        headlines = soup.find_all('h2', class_='entry-title')

        # Print each headline
        for headline in headlines:
            print(headline.text.strip())
    else:
        # Print a message if there is no content to parse
        print("No content to parse.")

# Main function
def main():
    # Construct the full URL using the URL_PREFIX
    full_url = URL_PREFIX

    # Fetch the content of the news website
    content = fetch_news_content(full_url)

    # Parse and process the news content
    parse_news_content(content)

# Check if the script is being run directly
if __name__ == "__main__":
    # Call the main function
    main()
