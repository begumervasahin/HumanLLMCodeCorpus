from __future__ import print_function
import os
import codecs
from pandora.tagger import Tagger
import pandora.utils

def load_tagger_model(model_directory='models/new_cg'):
    """
    Load the tagger model from the specified directory.
    """
    print('::: Loading Tagger Model :::')
    return Tagger(load=True, model_dir=model_directory)

def load_unannotated_data(file_path):
    """
    Load unannotated data from the specified file path.
    """
    print(f'::: Loading data from {file_path} :::')
    return pandora.utils.load_unannotated_file(file_path, nb_instances=None, tokenized_input=False)

def annotate_tokens(tagger, tokens):
    """
    Annotate the tokens using the specified tagger.
    """
    print('::: Annotating Tokens :::')
    return tagger.annotate(tokens)

def write_annotations_to_file(output_file_path, annotations):
    """
    Write the annotations to the specified output file.
    """
    print(f'::: Writing annotations to {output_file_path} :::')
    with codecs.open(output_file_path, 'w', 'utf8') as output_file:
        for token, lemma, pos in zip(annotations['tokens'], annotations['lemmas'], annotations['pos']):
            output_file.write('\t'.join((token.lower(), lemma.lower(), pos.lower())) + '\n')

def process_files(input_directory, output_directory, tagger):
    """
    Process all files in the input directory, annotate their content, and save the annotations to the output directory.
    """
    print('::: Annotation Process Started :::')

    for filename in os.listdir(input_directory):
        if not filename.endswith('.txt'):
            continue

        print(f'\t + Annotating {filename}')

        input_file_path = os.path.join(input_directory, filename)
        unseen_tokens = load_unannotated_data(input_file_path)

        annotations = annotate_tokens(tagger, unseen_tokens)

        output_file_path = os.path.join(output_directory, filename)
        write_annotations_to_file(output_file_path, annotations)

    print('::: Annotation Process Ended :::')

def main():
    """
    Main function to load the tagger model and process files for annotation.
    """
    print('::: Starting Main Process :::')
    tagger = load_tagger_model()

    input_directory = 'data/limborch/orig/'
    output_directory = 'data/limborch/tagged/'

    process_files(input_directory, output_directory, tagger)
    print('::: Main Process Completed :::')

if __name__ == '__main__':
    main()
