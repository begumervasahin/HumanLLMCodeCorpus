# Define file paths for the input and output files
HINDI_DIC_FILE = './datasets/hi_hdtb-ud-dev.conllu'
HINDI_OUT_FILE = './outputs/train_data.txt'

def lemma_exists(lemmas, lemma):
    """
    Check if a given lemma exists in the list of lemmas.

    :param lemmas: List of existing lemmas.
    :param lemma: Lemma to check.
    :return: True if lemma exists, False otherwise.
    """
    return any(
        existing_lemma["root"] == lemma["root"] and
        existing_lemma["tag1"] == lemma["tag1"] and
        existing_lemma["tag2"] == lemma["tag2"]
        for existing_lemma in lemmas
    )

def serialize_lemma(lemma):
    """
    Serialize a lemma dictionary into a string format.

    :param lemma: Lemma dictionary.
    :return: Serialized lemma string.
    """
    return '[{}, {}, {}]'.format(lemma["root"], lemma["tag1"], lemma["tag2"])

def read_training_data(input_file, output_file=None):
    """
    Read training data from the input file and process it into a dataset.
    Optionally write the dataset to an output file.

    :param input_file: Path to the input file.
    :param output_file: Path to the output file (optional).
    :return: Processed dataset as a dictionary.
    """
    dataset = {}

    with open(input_file, 'r', encoding='utf-8') as file:
        for line in file:
            if len(line) < 5 or line.startswith('#'):
                continue

            parts = line.strip().split("\t")
            if len(parts) < 5:
                continue

            word, root, tag1, tag2 = parts[1:5]

            if word not in dataset:
                dataset[word] = []

            lemma = {"root": root, "tag1": tag1, "tag2": tag2}

            if not lemma_exists(dataset[word], lemma):
                dataset[word].append(lemma)

    if output_file:
        with open(output_file, 'w', encoding='utf-8') as file:
            for word, lemmas in dataset.items():
                serialized_lemmas = ', '.join([serialize_lemma(lemma) for lemma in lemmas])
                file.write(f'{word}:{serialized_lemmas}\n')

    return dataset

def supervised_learn():
    """
    Main function to read the training data and process each word.
    """
    training_data = read_training_data(HINDI_DIC_FILE, HINDI_OUT_FILE)
    
    for word, lemmas in training_data.items():
        print(f'Processing word: {word} with {len(lemmas)} lemmas')

if __name__ == "__main__":
    supervised_learn()
