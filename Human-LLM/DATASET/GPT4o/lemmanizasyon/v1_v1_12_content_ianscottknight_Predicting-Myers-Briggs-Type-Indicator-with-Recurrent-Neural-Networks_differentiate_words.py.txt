import os
import collections
import re

# Constants
DATA_DIR = "data"
DIMENSIONS = ["IE", "NS", "FT", "PJ"]
WORDS_TO_REMOVE = [
    "intj", "intp", "infj", "infp", "istj", "istp", "isfj", "isfp",
    "entj", "entp", "enfj", "enfp", "estj", "estp", "esfj", "esfp",
    "si", "ni", "ti", "fi", "se", "ne", "te", "fe", "nt", "nf",
    "sxsp", "spsx", "spso", "sxso", "sosp", "sosx", "sp", "sx", "sj", "sf", "st",
    "le", "socionic", "socionics", "enneagram", "d", "w", "mbti",
]

# Regex to remove non-alphabetic characters
regex = re.compile("[^a-zA-Z]")

# Function to process each dimension pair
def process_dimension(dimension):
    wordcount_a = {}
    wordcount_b = {}

    # Read and count words for the first file
    with open(os.path.join(DATA_DIR, f"extreme_examples_{dimension[0]}.txt"), "r") as f:
        wordcount_a = collections.Counter(f.read().split())

    # Read and count words for the second file
    with open(os.path.join(DATA_DIR, f"extreme_examples_{dimension[1]}.txt"), "r") as f:
        wordcount_b = collections.Counter(f.read().split())

    # Combine keys from both word counts
    cache = list(set(wordcount_a.keys()) | set(wordcount_b.keys()))
    a = {}
    b = {}

    # Calculate differences
    for key in cache:
        if key in wordcount_a and key not in wordcount_b:
            a[key] = wordcount_a[key]
        elif key in wordcount_b and key not in wordcount_a:
            b[key] = wordcount_b[key]
        elif key in wordcount_a and key in wordcount_b:
            diff = wordcount_a[key] - wordcount_b[key]
            if diff > 0:
                a[key] = diff
            elif diff < 0:
                b[key] = -diff

    # Write results to files
    with open(os.path.join(DATA_DIR, f"special_words_{dimension[0]}.txt"), "w") as f:
        for key in a.keys():
            mod = regex.sub("", str(key))
            if mod not in WORDS_TO_REMOVE and a[key] > 2:
                f.write(mod + "\n")

    with open(os.path.join(DATA_DIR, f"special_words_{dimension[1]}.txt"), "w") as f:
        for key in b.keys():
            mod = regex.sub("", str(key))
            if mod not in WORDS_TO_REMOVE and b[key] > 2:
                f.write(mod + "\n")

# Process each dimension pair
for dimension in DIMENSIONS:
    process_dimension(dimension)
