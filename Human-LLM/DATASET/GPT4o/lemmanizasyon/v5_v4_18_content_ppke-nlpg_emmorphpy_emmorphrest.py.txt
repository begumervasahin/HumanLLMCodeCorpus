from xtsv import pipeline_rest_api, singleton_store_factory

# Define task configurations for stemming, analyzing, and dstemming using EmMorphPy
def create_task(module, class_name, method_name, task_type, source_fields, target_fields):
    return (
        module,            # Module path
        class_name,        # Class name
        method_name,       # Method name
        (),                # Arguments tuple (empty here)
        {
            'task': task_type,           # Task type
            'source_fields': source_fields,  # Source field(s)
            'target_fields': target_fields   # Target field(s)
        }
    )

# Stemming task
em_morph_stem = create_task(
    'emmorphpy.emmorphpy', 
    'EmMorphPy', 
    'stem', 
    'stem', 
    {'form'}, 
    ['anas']
)

# Analyzing task
em_morph_analyze = create_task(
    'emmorphpy.emmorphpy', 
    'EmMorphPy', 
    'analyze', 
    'analyze', 
    {'form'}, 
    ['anas']
)

# Dstemming task
em_morph_dstem = create_task(
    'emmorphpy.emmorphpy', 
    'EmMorphPy', 
    'dstem', 
    'dstem', 
    {'form'}, 
    ['anas']
)

# List of tools with their corresponding task names
tools = [
    (em_morph_stem, ('stem',)),
    (em_morph_analyze, ('analyze',)),
    (em_morph_dstem, ('dstem',)),
]

# Set up the REST API for the pipeline
app = pipeline_rest_api(
    'emMorph',                             # API name
    tools,                                 # List of tools
    {},                                    # Additional configurations (empty here)
    conll_comments=False,                  # Boolean flag for CoNLL comments
    singleton_store=singleton_store_factory(),  # Singleton store
    form_title='emMorph demo',             # Title for the form
    form_type='radio',                     # Form type (radio buttons)
    doc_link='https://github.com/ppke-nlpg/emmorphpy'  # Documentation link
)

# Run the application if this script is executed directly
if __name__ == '__main__':
    app.run(debug=False)  # Run the app (debug mode is set to False)
