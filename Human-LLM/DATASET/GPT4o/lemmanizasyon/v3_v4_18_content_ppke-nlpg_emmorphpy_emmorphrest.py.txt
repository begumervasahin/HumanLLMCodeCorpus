from xtsv import pipeline_rest_api, singleton_store_factory

def create_task(module_path, class_name, method_name, task_name, source_fields, target_fields):
    """Helper function to create a task configuration."""
    return (
        module_path,
        class_name,
        method_name,
        (),
        {
            'task': task_name,
            'source_fields': source_fields,
            'target_fields': target_fields
        }
    )

# Define the tasks for stemming, analyzing, and dstemming using EmMorphPy
em_morph_stem = create_task(
    'emmorphpy.emmorphpy',  # Module path
    'EmMorphPy',            # Class name
    'stem',                 # Method name
    'stem',                 # Task name
    {'form'},               # Source field(s)
    ['anas']                # Target field(s)
)

em_morph_analyze = create_task(
    'emmorphpy.emmorphpy',  # Module path
    'EmMorphPy',            # Class name
    'analyze',              # Method name
    'analyze',              # Task name
    {'form'},               # Source field(s)
    ['anas']                # Target field(s)
)

em_morph_dstem = create_task(
    'emmorphpy.emmorphpy',  # Module path
    'EmMorphPy',            # Class name
    'dstem',                # Method name
    'dstem',                # Task name
    {'form'},               # Source field(s)
    ['anas']                # Target field(s)
)

# List of tools with their respective tasks
tools = [
    (em_morph_stem, ('stem',)),
    (em_morph_analyze, ('analyze',)),
    (em_morph_dstem, ('dstem',)),
]

# Configure the REST API pipeline
app = pipeline_rest_api(
    'emMorph',                          # API name
    tools,                              # List of tools and tasks
    {},                                 # Additional configurations (empty in this case)
    conll_comments=False,               # Disable CoNLL comments
    singleton_store=singleton_store_factory(),  # Singleton store for the pipeline
    form_title='emMorph demo',          # Title for the form
    form_type='radio',                  # Form type (radio buttons)
    doc_link='https://github.com/ppke-nlpg/emmorphpy'  # Documentation link
)

# Entry point to run the application
if __name__ == '__main__':
    app.run(debug=False)  # Run the app with debug mode disabled
