from __future__ import print_function
import os
import codecs
from pandora.tagger import Tagger
import pandora.utils

def main():
    print('::: Annotation Process Started :::')

    # Load the pre-trained Tagger model
    tagger = load_tagger_model('models/new_cg')
    print('Tagger loaded. Now, annotating...')

    # Set the paths for input and output directories
    input_directory = 'data/limborch/orig/'
    output_directory = 'data/limborch/tagged/'

    # Create the output directory if it does not exist
    create_output_directory(output_directory)

    # Process each text file in the input directory
    process_files(input_directory, output_directory, tagger)

    print('::: Annotation Process Ended :::')

def load_tagger_model(model_directory):
    """
    Load the pre-trained Tagger model.

    Args:
        model_directory (str): The directory containing the model.

    Returns:
        Tagger: The loaded Tagger model.
    """
    return Tagger(load=True, model_dir=model_directory)

def create_output_directory(directory_path):
    """
    Create the output directory if it does not exist.

    Args:
        directory_path (str): The path to the output directory.
    """
    if not os.path.exists(directory_path):
        os.makedirs(directory_path)

def process_files(input_directory, output_directory, tagger):
    """
    Process each text file in the input directory.

    Args:
        input_directory (str): The directory containing input text files.
        output_directory (str): The directory to save annotated text files.
        tagger (Tagger): The loaded Tagger model.
    """
    for filename in os.listdir(input_directory):
        if filename.endswith('.txt'):
            print(f'\t + Annotating {filename}')
            annotate_file(input_directory, output_directory, filename, tagger)

def annotate_file(input_directory, output_directory, filename, tagger):
    """
    Annotate a single file and save the annotations.

    Args:
        input_directory (str): The directory containing input text files.
        output_directory (str): The directory to save annotated text files.
        filename (str): The name of the file to annotate.
        tagger (Tagger): The loaded Tagger model.
    """
    file_path = os.path.join(input_directory, filename)
    unseen_tokens = load_unannotated_data(file_path)
    annotations = tagger.annotate(unseen_tokens)
    output_file_path = os.path.join(output_directory, filename)
    write_annotations_to_file(output_file_path, annotations)

def load_unannotated_data(file_path):
    """
    Load unannotated data from a file.

    Args:
        file_path (str): The path to the file containing unannotated data.

    Returns:
        list: A list of unannotated tokens.
    """
    return pandora.utils.load_unannotated_file(
        file_path,
        nb_instances=None,
        tokenized_input=False
    )

def write_annotations_to_file(output_file_path, annotations):
    """
    Write the annotated data to a file.

    Args:
        output_file_path (str): The path to the output file.
        annotations (dict): The annotations to write.
    """
    with codecs.open(output_file_path, 'w', 'utf8') as output_file:
        for token, lemma, pos in zip(annotations['tokens'], annotations['lemmas'], annotations['pos']):
            output_file.write('\t'.join((token.lower(), lemma.lower(), pos.lower())) + '\n')

if __name__ == '__main__':
    main()
