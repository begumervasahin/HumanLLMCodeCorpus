import os
import collections
import re

# Define the directory where data files are stored
DATA_DIR = "data"

# Define the dimensions to process and the words to exclude
DIMENSIONS = ["IE", "NS", "FT", "PJ"]
WORDS_TO_REMOVE = [
    "intj", "intp", "infj", "infp", "istj", "istp", "isfj", "isfp",
    "entj", "entp", "enfj", "enfp", "estj", "estp", "esfj", "esfp",
    "si", "ni", "ti", "fi", "se", "ne", "te", "fe", "nt", "nf",
    "sxsp", "spsx", "spso", "sxso", "sosp", "sosx", "sp", "sx",
    "sj", "sf", "st", "le", "socionic", "socionics", "enneagram",
    "d", "w", "mbti"
]

# Compile regex pattern to remove non-alphabet characters
regex = re.compile("[^a-zA-Z]")

# Process each dimension pair
for dimension in DIMENSIONS:
    wordcount_a = collections.Counter()
    wordcount_b = collections.Counter()

    # Read and count words from the first file of the dimension pair
    with open(os.path.join(DATA_DIR, f"extreme_examples_{dimension[0]}.txt"), "r") as file_a:
        wordcount_a = collections.Counter(file_a.read().split())

    # Read and count words from the second file of the dimension pair
    with open(os.path.join(DATA_DIR, f"extreme_examples_{dimension[1]}.txt"), "r") as file_b:
        wordcount_b = collections.Counter(file_b.read().split())

    # Combine unique words from both files
    combined_keys = set(wordcount_a.keys()) | set(wordcount_b.keys())

    # Dictionaries to store word counts for the output
    filtered_wordcount_a = {}
    filtered_wordcount_b = {}

    # Calculate word count differences and filter based on the exclusion list
    for key in combined_keys:
        if key not in WORDS_TO_REMOVE:
            count_a = wordcount_a.get(key, 0)
            count_b = wordcount_b.get(key, 0)
            difference = count_a - count_b

            if difference > 0:
                filtered_wordcount_a[key] = difference
            elif difference < 0:
                filtered_wordcount_b[key] = -difference

    # Function to write filtered words to file if count > 2
    def write_filtered_words(filename, word_counts):
        with open(filename, "w") as output_file:
            for word, count in word_counts.items():
                cleaned_word = regex.sub("", word)
                if count > 2:
                    output_file.write((cleaned_word + "\n") * count)

    # Write the filtered words for both dimensions
    write_filtered_words(os.path.join(DATA_DIR, f"special_words_{dimension[0]}.txt"), filtered_wordcount_a)
    write_filtered_words(os.path.join(DATA_DIR, f"special_words_{dimension[1]}.txt"), filtered_wordcount_b)
