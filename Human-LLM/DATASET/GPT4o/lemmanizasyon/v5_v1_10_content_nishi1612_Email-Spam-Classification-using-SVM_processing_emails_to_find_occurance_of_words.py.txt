import os
import string
import numpy as np
import pandas as pd
from time import time
from nltk.corpus import stopwords
from nltk.stem.wordnet import WordNetLemmatizer

def read_words_list(filepath):
    """Reads the words list from a CSV file."""
    df = pd.read_csv(filepath, header=0)
    return df['word']

def initialize_output_file(filepath, words):
    """Initializes the output CSV file with headers."""
    with open(filepath, "w+") as f:
        f.write(','.join(words) + ',output\n')

def get_word_frequencies(file_path, words, lemmatizer):
    """Reads an email file and computes word frequencies."""
    word_frequencies = np.zeros(len(words))
    with open(file_path, "r", encoding='utf-8', errors='ignore') as email_file:
        for word in email_file.read().split():
            word = lemmatizer.lemmatize(word.lower())
            if word in stopwords.words('english') or word in string.punctuation or len(word) <= 2 or word.isdigit():
                continue
            if word in words.values:
                word_index = words[words == word].index[0]
                word_frequencies[word_index] += 1
    return word_frequencies

def determine_output_value(file_path):
    """Determines the output value based on the file path length."""
    if len(file_path) == 68:
        return "-1"
    elif len(file_path) == 71:
        return "1"
    return "0"

def append_frequencies_to_file(filepath, frequencies, output_value):
    """Appends word frequencies and output value to the CSV file."""
    with open(filepath, "a") as f:
        f.write(','.join(map(str, frequencies.astype(int))) + ',' + output_value + '\n')

def process_emails(directory_path, words, output_filepath, lemmatizer):
    """Processes each email file in the directory and writes frequencies to the output file."""
    file_count = 0
    for file in os.listdir(directory_path):
        file_name = os.fsdecode(file)
        full_file_path = os.path.join(os.getcwd(), directory_path, file_name)
        file_count += 1

        word_frequencies = get_word_frequencies(full_file_path, words, lemmatizer)
        output_value = determine_output_value(full_file_path)
        append_frequencies_to_file(output_filepath, word_frequencies, output_value)

        if file_count % 100 == 0:
            print(f"Processed {file_count} files")

def main():
    start_time = time()

    words_filepath = 'wordslist.csv'
    output_filepath = "frequency.csv"
    emails_directory = "emails/"

    words = read_words_list(words_filepath)
    lemmatizer = WordNetLemmatizer()

    initialize_output_file(output_filepath, words)
    process_emails(emails_directory, words, output_filepath, lemmatizer)

    elapsed_time = round(time() - start_time, 2)
    print(f"Time (in seconds) to process the dataset: {elapsed_time}")

if __name__ == "__main__":
    main()
