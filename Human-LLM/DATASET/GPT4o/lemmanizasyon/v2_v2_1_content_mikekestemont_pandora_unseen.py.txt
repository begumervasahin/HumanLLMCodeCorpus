from __future__ import print_function
import os
import codecs
from pandora.tagger import Tagger
import pandora.utils

def annotate_files(input_dir, output_dir, tagger):
    """
    Annotate text files in the input directory and save the results to the output directory.

    Parameters:
    input_dir (str): Path to the directory containing original text files.
    output_dir (str): Path to the directory where annotated files will be saved.
    tagger (Tagger): Pre-trained tagger model for annotation.
    """
    # Ensure the output directory exists
    os.makedirs(output_dir, exist_ok=True)

    # Process each file in the input directory
    for filename in os.listdir(input_dir):
        if not filename.endswith('.txt'):
            continue

        print(f'\t + Annotating {filename}')

        # Load unannotated data from the original file
        input_file_path = os.path.join(input_dir, filename)
        unseen_tokens = pandora.utils.load_unannotated_file(
            input_file_path,
            nb_instances=None,
            tokenized_input=False
        )

        # Annotate the tokens using the loaded tagger
        annotations = tagger.annotate(unseen_tokens)

        # Write annotated data to a new file
        output_file_path = os.path.join(output_dir, filename)
        write_annotations_to_file(output_file_path, annotations)

def write_annotations_to_file(output_file_path, annotations):
    """
    Write annotated data to a file.

    Parameters:
    output_file_path (str): Path to the output file.
    annotations (dict): Dictionary containing tokens, lemmas, and POS tags.
    """
    with codecs.open(output_file_path, 'w', 'utf8') as output_file:
        for token, lemma, pos in zip(annotations['tokens'], annotations['lemmas'], annotations['pos']):
            output_file.write('\t'.join((token.lower(), lemma.lower(), pos.lower())) + '\n')

def main():
    """
    Main function to load the tagger model and start the annotation process.
    """
    print('::: Annotation Started :::')

    # Load the pre-trained Tagger model
    tagger = Tagger(load=True, model_dir='models/new_cg')
    print('Tagger loaded, now annotating...')

    # Set the paths for input and output directories
    input_dir = 'data/limborch/orig/'
    output_dir = 'data/limborch/tagged/'

    # Annotate files in the input directory
    annotate_files(input_dir, output_dir, tagger)

    print('::: Annotation Ended :::')

if __name__ == '__main__':
    main()
