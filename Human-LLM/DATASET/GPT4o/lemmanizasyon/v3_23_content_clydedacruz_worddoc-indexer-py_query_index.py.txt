import json
from nltk.stem import WordNetLemmatizer

# Define a set of English stopwords
ENGLISH_STOPWORDS = set([
    "i", "me", "my", "myself", "we", "our", "ours", "ourselves", "you", "your", "yours", "yourself", "yourselves",
    "he", "him", "his", "himself", "she", "her", "hers", "herself", "it", "its", "itself", "they", "them", "their",
    "theirs", "themselves", "what", "which", "who", "whom", "this", "that", "these", "those", "am", "is", "are",
    "was", "were", "be", "been", "being", "have", "has", "had", "having", "do", "does", "did", "doing", "a", "an",
    "the", "and", "but", "if", "or", "because", "as", "until", "while", "of", "at", "by", "for", "with", "about",
    "against", "between", "into", "through", "during", "before", "after", "above", "below", "to", "from", "up",
    "down", "in", "out", "on", "off", "over", "under", "again", "further", "then", "once", "here", "there", "when",
    "where", "why", "how", "all", "any", "both", "each", "few", "more", "most", "other", "some", "such", "no", "nor",
    "not", "only", "own", "same", "so", "than", "too", "very", "s", "t", "can", "will", "just", "don", "should", "now"
])

def remove_stopwords(wordlist):
    """
    Remove stopwords from the input word list.
    """
    return [word for word in wordlist if word not in ENGLISH_STOPWORDS]

def set_list_intersection(set_list):
    """
    Find the intersection of all sets in the list.
    """
    if not set_list:
        return set()
    result = set_list[0]
    for s in set_list[1:]:
        result &= s
    return result

def load_index(index_file_path):
    """
    Load the index file and return the index dictionary.
    """
    with open(index_file_path, 'r') as index_file:
        return json.loads(index_file.read())

def process_query(query_terms, index_dictionary):
    """
    Process the query terms, lemmatize them, and return the result set intersection.
    """
    lemmatizer = WordNetLemmatizer()
    result_sets = []
    
    for term in query_terms:
        if term in ENGLISH_STOPWORDS:
            continue
        lem_term = lemmatizer.lemmatize(term)
        if lem_term in index_dictionary:
            result_sets.append(set(index_dictionary[lem_term]['postings']))
        else:
            print(f"No documents found matching query term: {term}")
            return set()
    
    return set_list_intersection(result_sets)

def main(query_terms):
    """
    Main function to process the query terms and print the query results.
    """
    INDEX_FILE = "INDEX_FILE.INDEX"
    index_dictionary = load_index(INDEX_FILE)
    
    print('_' * 50 + "\nQuery Result \n" + '_' * 50)
    
    query_result = process_query(query_terms, index_dictionary)
    if query_result:
        print('\n'.join(query_result))
    else:
        print(f"No documents found matching query: {' '.join(query_terms)}")

if __name__ == "__main__":
    query_terms = sys.argv[1:]
    main(query_terms)
