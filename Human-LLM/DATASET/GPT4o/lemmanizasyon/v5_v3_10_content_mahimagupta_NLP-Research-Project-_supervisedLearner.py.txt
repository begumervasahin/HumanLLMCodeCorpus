# Define input and output file paths
HINDI_DIC_FILE = './datasets/hi_hdtb-ud-dev.conllu'
HINDI_OUT_FILE = './outputs/train_data.txt'

def is_lemma_present(lemmas, lemma):
    """
    Check if a lemma is already present in the dataset.
    
    Args:
        lemmas (list): List of existing lemmas for a word.
        lemma (dict): Lemma to check for presence.
    
    Returns:
        bool: True if lemma is present, False otherwise.
    """
    return any(all(existing_lemma[key] == lemma[key] for key in ("root", "tag1", "tag2")) for existing_lemma in lemmas)

def format_lemma(lemma):
    """
    Format a lemma into a string representation.
    
    Args:
        lemma (dict): Lemma to format.
    
    Returns:
        str: Formatted string representation of the lemma.
    """
    return '[{}, {}, {}]'.format(lemma["root"], lemma["tag1"], lemma["tag2"])

def read_training_data(input_file, output_file=None):
    """
    Read training data from a file and return a dataset.
    
    Args:
        input_file (str): Path to the input file containing training data.
        output_file (str, optional): Path to the output file to save processed data.
    
    Returns:
        dict: Dictionary containing the processed dataset.
    """
    dataset = {}

    with open(input_file, 'r', encoding='utf-8') as file:
        for line in file:
            # Skip comments and empty lines
            if len(line) < 5 or line.startswith('#'):
                continue
            
            # Extract relevant parts from the line
            parts = line.split("\t")
            word, root, tag1, tag2 = parts[1], parts[2], parts[3], parts[4]
            
            # Initialize the word entry in the dataset if not already present
            if word not in dataset:
                dataset[word] = []
            
            # Create a lemma dictionary
            lemma = {"root": root, "tag1": tag1, "tag2": tag2}
            
            # Add the lemma to the dataset if it's not already present
            if not is_lemma_present(dataset[word], lemma):
                dataset[word].append(lemma)
    
    # Write the dataset to the output file if specified
    if output_file is not None:
        with open(output_file, 'w', encoding='utf-8') as file:
            for word, lemmas in dataset.items():
                formatted_lemmas = ', '.join([format_lemma(l) for l in lemmas])
                file.write(f'{word}: {formatted_lemmas}\n')
    
    return dataset

def supervised_learn():
    """
    Perform supervised learning and print processing details.
    """
    training_data = read_training_data(HINDI_DIC_FILE, HINDI_OUT_FILE)
    for word, lemmas in training_data.items():
        print(f'Processing word: {word} with {len(lemmas)} lemmas')

# Execute the supervised learning function
supervised_learn()
