import os
import sys
import argparse
from tqdm import tqdm
import numpy as np
from util import logging, logger, filechk, create_key, file_converter_enc, file_converter_dec
from xored import encXOR, decXOR, encXORrounds
from bitshift import SHIFTLEFT, SHIFTRIGHT
from sbox import sbox_enc, sbox_dec, sbox_dec_rounds

def encrypt(filename):
    """
    Encrypt the provided file using multiple rounds of encryption techniques.
    
    Args:
    filename (str): The name of the file to encrypt.
    """
    key = create_key()
    print(f'Share this Key {key} with the message receiver')
    key_bin = f'{key:016b}'
    logging.info("Key created")
    logging.info("Encryption has started...")
    
    # Initial encryption steps
    bit_xor = encXOR(filename, key_bin)
    bit_shift = SHIFTLEFT(bit_xor)
    bit_sbox = sbox_enc(bit_shift)
    bits = bit_sbox
    
    # 16 rounds of encryption
    for _ in tqdm(range(16)):
        bits = encXORrounds(bits, key_bin)
        bits = SHIFTLEFT(bits)
        bits = sbox_enc(bits)
    
    file_converter_enc(bits)

def decrypt(filename):
    """
    Decrypt the provided file using multiple rounds of decryption techniques.
    
    Args:
    filename (str): The name of the file to decrypt.
    """
    key = int(input("Input the key shared with you\n"))
    key_bin = f'{key:016b}'
    logging.info("Key received")
    logging.info("Decryption has started...")
    
    # Initial decryption steps
    inv_bits_box = sbox_dec(filename)
    bit_shift = SHIFTRIGHT(inv_bits_box)
    bit_xnor = decXOR(bit_shift, key_bin)
    bits = bit_xnor
    
    # 16 rounds of decryption
    for _ in tqdm(range(16)):
        bits = sbox_dec_rounds(bits)
        bits = SHIFTRIGHT(bits)
        bits = decXOR(bits, key_bin)
    
    file_converter_dec(bits)

def main():
    """
    Main function to handle user input and perform encryption or decryption.
    """
    logger()
    parser = argparse.ArgumentParser(description='Encrypt and Decrypt files')
    group = parser.add_mutually_exclusive_group()
    
    parser.add_argument("FileName", help='Input a filename')
    group.add_argument('-e', '--enc', help='Encrypt the file', action='store_true')
    group.add_argument('-d', '--dec', help='Decrypt the file', action='store_true')
    
    args = parser.parse_args()
    filechk(args.FileName)
    
    if args.enc:
        encrypt(args.FileName)
    elif args.dec:
        decrypt(args.FileName)

if __name__ == '__main__':
    main()
