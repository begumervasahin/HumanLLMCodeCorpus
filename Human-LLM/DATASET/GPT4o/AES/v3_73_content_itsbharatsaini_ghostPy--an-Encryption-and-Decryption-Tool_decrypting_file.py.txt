import base64
import os
from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC
from cryptography.fernet import Fernet

def derive_key(password: str, salt: bytes = b'ghostpy') -> bytes:
    """Derive a key from the provided password using PBKDF2HMAC."""
    kdf = PBKDF2HMAC(
        algorithm=hashes.SHA256(),
        length=32,
        salt=salt,
        iterations=100000,
        backend=default_backend()
    )
    return base64.urlsafe_b64encode(kdf.derive(password.encode()))

def decrypt_file(file_path: str, key: bytes):
    """Decrypt a single file using the provided key."""
    with open(file_path, 'rb') as file:
        data = file.read()
    fernet = Fernet(key)
    decrypted_data = fernet.decrypt(data)
    with open(file_path, 'wb') as file:
        file.write(decrypted_data)
    print(f"File '{file_path}' has been decrypted.")

def decrypt_folder(folder_path: str, key: bytes):
    """Decrypt all files in a folder recursively using the provided key."""
    for root, _, files in os.walk(folder_path):
        for file_name in files:
            file_path = os.path.join(root, file_name)
            try:
                decrypt_file(file_path, key)
            except Exception as e:
                print(f"File '{file_path}' could not be decrypted. Error: {e}")

def main():
    """Main function to handle decryption of files or folders."""
    print("\nWelcome to gostpy Decryption Tool...!!\n")

    password = input("Enter your encrypted file password: ").strip()
    if not password:
        print("Please provide a password.")
        return

    key = derive_key(password)

    files_path = input("Drag and drop your file or folder here: ").strip()
    if not files_path:
        print("Please provide a valid file or folder path.")
        return

    print("Files are decrypting...")

    if os.path.isfile(files_path):
        decrypt_file(files_path, key)
    elif os.path.isdir(files_path):
        decrypt_folder(files_path, key)
    else:
        print("The provided path is neither a file nor a folder.")

if __name__ == "__main__":
    main()
