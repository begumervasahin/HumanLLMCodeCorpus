import os
import sys
from cryptography.hazmat.primitives.ciphers import Cipher, algorithms, modes
from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives import padding

def generate_key():
    """
    Generates a 256-bit AES key and saves it to a file.
    """
    key = os.urandom(32)
    with open('AESkey.key', 'wb') as key_file:
        key_file.write(key)
    print("Key generated and saved to 'AESkey.key'")

def encrypt_file(filename):
    """
    Encrypts the file specified by filename using the generated AES key.
    """
    # Read plaintext from the file
    with open(filename, 'rb') as file:
        plaintext = file.read()

    # Read the AES key from the key file
    with open('AESkey.key', 'rb') as key_file:
        key = key_file.read()

    # Generate an initialization vector (IV) and save it
    iv = os.urandom(16)
    with open(f'{filename}.enc.iv', 'wb') as iv_file:
        iv_file.write(iv)

    # Pad the plaintext to be a multiple of the block size
    padder = padding.PKCS7(algorithms.AES.block_size).padder()
    padded_plaintext = padder.update(plaintext) + padder.finalize()

    # Encrypt the padded plaintext
    cipher = Cipher(algorithms.AES(key), modes.CBC(iv), backend=default_backend())
    encryptor = cipher.encryptor()
    ciphertext = encryptor.update(padded_plaintext) + encryptor.finalize()

    # Write the ciphertext to a file
    with open(f'{filename}.enc', 'wb') as encrypted_file:
        encrypted_file.write(ciphertext)

    print(f"File '{filename}' has been encrypted and saved as '{filename}.enc'")

def decrypt_file(filename):
    """
    Decrypts the file specified by filename using the generated AES key.
    """
    # Read the ciphertext from the file
    with open(filename, 'rb') as encrypted_file:
        padded_ciphertext = encrypted_file.read()

    # Read the AES key from the key file
    with open('AESkey.key', 'rb') as key_file:
        key = key_file.read()

    # Read the initialization vector (IV) from the file
    with open(f'{filename}.iv', 'rb') as iv_file:
        iv = iv_file.read()

    # Decrypt the ciphertext
    cipher = Cipher(algorithms.AES(key), modes.CBC(iv), backend=default_backend())
    decryptor = cipher.decryptor()
    decrypted_padded_msg = decryptor.update(padded_ciphertext) + decryptor.finalize()

    # Unpad the decrypted message
    unpadder = padding.PKCS7(algorithms.AES.block_size).unpadder()
    plaintext = unpadder.update(decrypted_padded_msg) + unpadder.finalize()

    # Write the plaintext to a file
    decrypted_filename = filename.replace('.enc', '.decrypted')
    with open(decrypted_filename, 'wb') as decrypted_file:
        decrypted_file.write(plaintext)

    print(f"File '{filename}' has been decrypted and saved as '{decrypted_filename}'")

def main():
    if len(sys.argv) < 2:
        print('Usage: python script.py [generate|encrypt|decrypt] [FILENAME]')
        sys.exit(1)

    command = sys.argv[1].lower()

    if command == 'generate':
        generate_key()
    elif command == 'encrypt':
        if len(sys.argv) != 3:
            raise Exception('Usage: python script.py encrypt FILENAME')
        encrypt_file(sys.argv[2])
    elif command == 'decrypt':
        if len(sys.argv) != 3:
            raise Exception('Usage: python script.py decrypt FILENAME')
        decrypt_file(sys.argv[2])
    else:
        print('Invalid command. Use "generate", "encrypt", or "decrypt".')

if __name__ == "__main__":
    main()
