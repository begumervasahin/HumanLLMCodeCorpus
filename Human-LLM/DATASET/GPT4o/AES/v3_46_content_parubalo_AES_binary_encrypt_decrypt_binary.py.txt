import base64
from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC
from cryptography.fernet import Fernet

def generate_key_from_password(password):
    """Generate a key from the provided password using PBKDF2HMAC."""
    password_bytes = password.encode()  # Convert the password to bytes
    salt = b'salt_'  # Define a fixed salt for key derivation
    kdf = PBKDF2HMAC(
        algorithm=hashes.SHA256(),  # Use SHA256 hash algorithm
        length=32,  # Generate a 32-byte key
        salt=salt,  # Use the defined salt
        iterations=100000,  # Perform 100,000 iterations
        backend=default_backend()  # Use the default cryptographic backend
    )
    key = base64.urlsafe_b64encode(kdf.derive(password_bytes))  # Derive the key and encode it
    return key

def encrypt_file(key, input_file):
    """Encrypt the file with the given key."""
    output_file = f'encrypted_{input_file}'  # Define the output file name
    with open(input_file, 'rb') as file:
        data = file.read()  # Read the input file
    fernet = Fernet(key)  # Create a Fernet instance with the provided key
    encrypted_data = fernet.encrypt(data)  # Encrypt the data
    with open(output_file, 'wb') as file:
        file.write(encrypted_data)  # Write the encrypted data to the output file

def decrypt_file(key, input_file):
    """Decrypt the file with the given key."""
    output_file = f'decrypted_{input_file}'  # Define the output file name
    with open(input_file, 'rb') as file:
        data = file.read()  # Read the input file
    fernet = Fernet(key)  # Create a Fernet instance with the provided key
    decrypted_data = fernet.decrypt(data)  # Decrypt the data
    with open(output_file, 'wb') as file:
        file.write(decrypted_data)  # Write the decrypted data to the output file

def convert_to_binary(input_file):
    """Convert the content of the file to a binary string representation."""
    output_file = f'binary_{input_file}'  # Define the output file name
    with open(input_file, 'r') as file:
        data = file.read()  # Read the input file
    binary_data = ''.join([bin(ord(c))[2:].zfill(8) for c in data])  # Convert each character to binary
    with open(output_file, 'w') as file:
        file.write(binary_data)  # Write the binary data to the output file

def convert_from_binary(input_file):
    """Convert the binary string representation in the file back to text."""
    output_file = f'debinary_{input_file}'  # Define the output file name
    with open(input_file, 'r') as file:
        data = file.read()  # Read the input file
    n = int(data, 2)  # Convert the binary data to an integer
    decoded_data = n.to_bytes((n.bit_length() + 7) // 8, 'big').decode()  # Convert the integer to bytes and decode
    with open(output_file, 'w') as file:
        file.write(decoded_data)  # Write the decoded data to the output file

def main():
    """Main function to handle user input and perform the selected operation."""
    menu = (
        "\nChoose an option:"
        "\n1. Encrypt a file"
        "\n2. Decrypt a file"
        "\n3. Convert file content to binary"
        "\n4. Convert binary content to text"
        "\n>> "
    )
    option = input(menu)
    
    if option == '1':
        password = input("Enter password >> ")
        file_name = input("Enter file name in current directory >> ")
        key = generate_key_from_password(password)
        encrypt_file(key, file_name)
        print(f"File '{file_name}' encrypted successfully.")
    elif option == '2':
        password = input("Enter password >> ")
        file_name = input("Enter file name in current directory >> ")
        key = generate_key_from_password(password)
        decrypt_file(key, file_name)
        print(f"File '{file_name}' decrypted successfully.")
    elif option == '3':
        file_name = input("Enter file name in current directory >> ")
        convert_to_binary(file_name)
        print(f"File '{file_name}' converted to binary successfully.")
    elif option == '4':
        file_name = input("Enter file name in current directory >> ")
        convert_from_binary(file_name)
        print(f"Binary file '{file_name}' converted to text successfully.")
    else:
        print("Invalid option. Please try again.")

if __name__ == "__main__":
    main()
