"""Server for multithreaded (asynchronous) chat application."""

from socket import AF_INET, socket, SOCK_STREAM
from threading import Thread
from Crypto.Cipher import AES
import base64

# Constants
BUFSIZ = 1024
HOST = ''
PORT = 33000
ADDR = (HOST, PORT)
KEY = b'master key'  # 16-byte AES key
BLOCK_SIZE = 16

# AES Encryption/Decryption functions
def pad(data):
    """Pads the data to be a multiple of BLOCK_SIZE."""
    padding_length = BLOCK_SIZE - (len(data) % BLOCK_SIZE)
    padding = chr(padding_length) * padding_length
    return data + padding.encode()

def unpad(data):
    """Removes padding from the data."""
    padding_length = ord(data[-1:])
    return data[:-padding_length]

def encrypt(key, data):
    """Encrypts the data using AES encryption."""
    data = pad(data)
    cipher = AES.new(key, AES.MODE_ECB)
    encrypted_data = cipher.encrypt(data)
    return base64.b64encode(encrypted_data)

def decrypt(key, enc_data):
    """Decrypts the data using AES encryption."""
    enc_data = base64.b64decode(enc_data)
    cipher = AES.new(key, AES.MODE_ECB)
    decrypted_data = cipher.decrypt(enc_data)
    return unpad(decrypted_data)

# Chat server functions
def accept_incoming_connections():
    """Sets up handling for incoming clients."""
    while True:
        client, client_address = SERVER.accept()
        print(f"{client_address} has connected.")
        welcome_message = "Greetings from the cave! Now type your name and press enter!"
        client.send(encrypt(KEY, welcome_message.encode('UTF-8')))
        addresses[client] = client_address
        Thread(target=handle_client, args=(client,)).start()

def handle_client(client):
    """Handles a single client connection."""
    name = decrypt(KEY, client.recv(BUFSIZ)).decode("utf-8")
    welcome_message = f'\nWelcome {name}! If you ever want to quit, type {{quit}} to exit.'
    client.send(encrypt(KEY, welcome_message.encode('UTF-8')))
    clients[client] = name

    while True:
        try:
            message = decrypt(KEY, client.recv(BUFSIZ)).decode("utf-8")
            if message != "{quit}":
                broadcast(message, name + ": ")
            else:
                client.close()
                del clients[client]
                if not clients:
                    SERVER.close()
                break
        except OSError:  # Handle client disconnection
            break

def broadcast(message, prefix=""):
    """Broadcasts a message to all the clients."""
    full_message = prefix + message
    encrypted_message = encrypt(KEY, full_message.encode('UTF-8'))
    for sock in clients:
        sock.send(encrypted_message)

# Global dictionaries to store client information
clients = {}
addresses = {}

# Setting up the server
SERVER = socket(AF_INET, SOCK_STREAM)
SERVER.bind(ADDR)

def main():
    """Main function to start the server."""
    SERVER.listen(5)
    print("Waiting for connection...")
    ACCEPT_THREAD = Thread(target=accept_incoming_connections)
    ACCEPT_THREAD.start()
    ACCEPT_THREAD.join()
    SERVER.close()

if __name__ == "__main__":
    main()
