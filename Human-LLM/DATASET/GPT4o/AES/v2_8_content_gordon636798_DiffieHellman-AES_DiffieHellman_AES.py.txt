import random
from gmpy2 import mpz, mpz_random, powmod, random_state
from Crypto.Cipher import AES
import hashlib

class User:
    def __init__(self, name, a, q):
        self.name = name
        self.a = a
        self.q = q
        temp = sum(ord(c) for c in name)
        self._private_key = mpz_random(random_state(temp), q)
        self.public_key = powmod(self.a, self._private_key, self.q)
        self._shared_keys = {}
        print(f"{self.name}'s master key: {self._private_key}")

    def generate_common_key(self, other_user, other_public_key):
        shared_key = powmod(other_public_key, self._private_key, self.q)
        shared_key_bytes = int(shared_key).to_bytes((shared_key.bit_length() + 7) // 8, byteorder='big')
        hashed_key = hashlib.sha256(shared_key_bytes).digest()
        print(f"\nCommon key for {self.name} and {other_user.name}: \n{hashed_key.hex()}")
        self._shared_keys[other_user.name] = hashed_key

    def aes_encrypt(self, other_user, plaintext):
        data = plaintext.encode('utf-8')
        padding_length = 16 - len(data) % 16
        data += bytes([padding_length] * padding_length)
        cipher = AES.new(self._shared_keys[other_user.name], AES.MODE_CBC, b'\x00' * 16)
        ciphertext = cipher.encrypt(data)
        print(f"\nCiphertext: \n{ciphertext.hex()}")
        return ciphertext, self.name

    def aes_decrypt(self, ciphertext):
        data, sender_name = ciphertext
        cipher = AES.new(self._shared_keys[sender_name], AES.MODE_CBC, b'\x00' * 16)
        plaintext = cipher.decrypt(data)
        padding_length = plaintext[-1]
        plaintext = plaintext[:-padding_length].decode('utf-8')
        print(f"\nPlaintext from {sender_name}: \n{plaintext}")
        return plaintext

def main():
    a = mpz(11)
    q = mpz(1040699159)
    print(f'a: {a}, q: {q}')
    
    plaintext = 'I love ncyucsie 噗嘻~~(≧Ꙫ≦)'
    print(f'Plaintext: {plaintext}\n')

    alice = User('Alice', a, q)
    bob = User('Bob', a, q)

    alice.generate_common_key(bob, bob.public_key)
    bob.generate_common_key(alice, alice.public_key)

    encrypted_data = alice.aes_encrypt(bob, plaintext)
    decrypted_data = bob.aes_decrypt(encrypted_data)

if __name__ == "__main__":
    main()
