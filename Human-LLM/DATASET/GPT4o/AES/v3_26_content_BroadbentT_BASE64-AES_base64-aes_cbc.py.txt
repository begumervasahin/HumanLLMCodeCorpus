def derive_key(company_key, salt='Pots de sel et de poivre'):
    """Derives a private key using PBKDF2 with the provided company key."""
    return PBKDF2(company_key, salt, dkLen=32, count=1000)

def pad(data, block_size):
    """Pads the data to be a multiple of the block size."""
    padding_length = block_size - len(data) % block_size
    padding = chr(padding_length) * padding_length
    return data + padding.encode()

def unpad(data):
    """Removes the padding from the data."""
    padding_length = data[-1]
    return data[:-padding_length]

def encrypt(plain_text, company_key):
    """Encrypts the plain text using AES encryption."""
    private_key = derive_key(company_key)
    padded_text = pad(plain_text.encode(), AES.block_size)
    iv = Random.new().read(AES.block_size)
    cipher = AES.new(private_key, AES.MODE_CBC, iv)
    encrypted_data = iv + cipher.encrypt(padded_text)
    return base64.b64encode(encrypted_data).decode()

def decrypt(encrypted_text, company_key):
    """Decrypts the encrypted text using AES encryption."""
    private_key = derive_key(company_key)
    encrypted_data = base64.b64decode(encrypted_text)
    iv = encrypted_data[:AES.block_size]
    cipher = AES.new(private_key, AES.MODE_CBC, iv)
    decrypted_data = unpad(cipher.decrypt(encrypted_data[AES.block_size:]))
    return decrypted_data.decode()
