import base64
import os
from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC
from cryptography.fernet import Fernet

def decryption():
    """Decrypt files or folders using a provided password."""
    print("\nWelcome to gostpy Decryption Tool...!!\n")

    def decrypt_file(file_path, key):
        """Decrypt a single file."""
        with open(file_path, 'rb') as file:
            data = file.read()
        fernet = Fernet(key)
        decrypted_data = fernet.decrypt(data)
        with open(file_path, 'wb') as file:
            file.write(decrypted_data)
        print(f"File '{file_path}' is decrypted.")

    def decrypt_folder(folder_path, key):
        """Decrypt all files in a folder recursively."""
        for root, _, files in os.walk(folder_path):
            for file_name in files:
                file_path = os.path.join(root, file_name)
                with open(file_path, 'rb') as file:
                    data = file.read()
                fernet = Fernet(key)
                try:
                    decrypted_data = fernet.decrypt(data)
                    with open(file_path, 'wb') as file:
                        file.write(decrypted_data)
                    print(f"File '{file_path}' is decrypted.")
                except Exception as e:
                    print(f"File '{file_path}' could not be decrypted. Error: {e}")

    # Get the password and derive the key
    password_provided = input("Enter your encrypted file password: ").strip()
    if not password_provided:
        print("Please provide a password.")
        return

    password = password_provided.encode()
    salt = b'ghostpy'
    kdf = PBKDF2HMAC(
        algorithm=hashes.SHA256(),
        length=32,
        salt=salt,
        iterations=100000,
        backend=default_backend()
    )
    key = base64.urlsafe_b64encode(kdf.derive(password))

    # Get the path of the file or folder to decrypt
    files_path = input("Drag and drop your file or folder here: ").strip()
    if not files_path:
        print("Please provide a valid file or folder path.")
        return

    print("Files are decrypting...")

    if os.path.isfile(files_path):
        decrypt_file(files_path, key)
    elif os.path.isdir(files_path):
        decrypt_folder(files_path, key)
    else:
        print("The provided path is neither a file nor a folder.")

if __name__ == "__main__":
    decryption()
