import numpy as np

def shift_left(bits):
    """
    Shift the rows of a 4x4 bit matrix to the left.

    Args:
    bits (list of str): List of bit strings representing 4x4 matrices.

    Returns:
    list of str: List of bit strings after left shifting the rows.
    """
    final_shift = []

    for bit_string in bits:
        # Convert bit string to 4x4 matrix
        matrix = np.array(list(bit_string)).reshape(4, 4)
        
        # Perform row-wise left shift
        shifted_matrix = [np.roll(row, -count) for count, row in enumerate(matrix)]
        
        # Append shifted matrix to final result
        final_shift.append(''.join(np.array(shifted_matrix).flatten()))
    
    return final_shift

def shift_right(bits):
    """
    Shift the rows of a 4x4 bit matrix to the right.

    Args:
    bits (list of str): List of bit strings representing 4x4 matrices.

    Returns:
    list of str: List of bit strings after right shifting the rows.
    """
    final_shift = []

    for bit_string in bits:
        # Convert bit string to 4x4 matrix
        matrix = np.array(list(bit_string)).reshape(4, 4)
        
        # Perform row-wise right shift
        shifted_matrix = [np.roll(row, count) for count, row in enumerate(matrix)]
        
        # Append shifted matrix to final result
        final_shift.append(''.join(np.array(shifted_matrix).flatten()))
    
    return final_shift

def display_bits(title, bits):
    """
    Display the bit strings with a title.

    Args:
    title (str): The title to display.
    bits (list of str): The bit strings to display.
    """
    print(f"\n{title}:")
    for bit_string in bits:
        print(bit_string)

def main():
    """
    Main function to demonstrate the shifting of bits.
    """
    bits = [
        "00010010001101000101011001111000",
        "11110000101010101111000010101010"
    ]
    
    left_shifted_bits = shift_left(bits)
    right_shifted_bits = shift_right(bits)
    
    display_bits("Original bits", bits)
    display_bits("Left shifted bits", left_shifted_bits)
    display_bits("Right shifted bits", right_shifted_bits)

if __name__ == "__main__":
    main()
