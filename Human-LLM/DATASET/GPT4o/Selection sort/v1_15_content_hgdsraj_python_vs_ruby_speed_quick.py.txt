# Initialize the array
arr = [1, 3, 5, 7, 9, 2, 4, 6, 8, 0]

# Implement the sorting algorithm
for i in range(0, 9):
    for j in range(0, 9):
        if arr[j] > arr[i]:
            # Swap the elements
            temp = arr[i]
            arr[i] = arr[j]
            arr[j] = temp

# Print the sorted array
print(arr)
print("\n")
