import time
import random

def selection_sort(array):
    """Sorts an array using the selection sort algorithm and measures the time taken."""
    start_time = time.time()
    
    for i in range(len(array) - 1):
        # Assume the current position holds the smallest element
        min_index = i
        # Find the smallest element in the remaining unsorted portion of the array
        for j in range(i + 1, len(array)):
            if array[min_index] > array[j]:
                min_index = j
        # Swap the found minimum element with the element at the current position
        array[i], array[min_index] = array[min_index], array[i]
    
    end_time = time.time()
    print(f"Time taken to sort the array: {end_time - start_time:.6f} seconds")
    
    return array

def generate_random_list(size, lower_bound, upper_bound):
    """Generates a list of random integers within specified bounds."""
    return [random.randint(lower_bound, upper_bound) for _ in range(size)]

def print_list(array):
    """Prints the elements of the list."""
    print(array)

# Example usage:
if __name__ == "__main__":
    # Generate a random list of 100 numbers for testing
    array = generate_random_list(100, 0, 1000)
    
    print("Original array:")
    print_list(array)
    
    # Sort the array using selection sort
    sorted_array = selection_sort(array)
    
    print("Sorted array:")
    print_list(sorted_array)
