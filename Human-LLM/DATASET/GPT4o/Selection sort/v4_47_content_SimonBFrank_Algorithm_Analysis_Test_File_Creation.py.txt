import random

# Define the number of elements and the filename
n = 100000
filename = "Descending_100000"

# Generate a descending list of numbers
number_list = list(range(n, 0, -1))

# Open the file for writing
with open(filename, "w") as f:
    # Write each number to the file, each on a new line
    for number in number_list:
        f.write(f"{number}\n")
