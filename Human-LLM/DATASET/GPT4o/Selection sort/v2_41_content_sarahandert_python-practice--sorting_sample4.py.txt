import random

def generate_nums(filename, n):
    """
    Generates a file with 'n' random numbers between 0 and 99.

    Args:
        filename (str): The name of the file where the numbers will be written.
        n (int): The number of random numbers to generate.
    """
    random.seed(0)  # Ensure reproducibility
    with open(filename, 'w') as file:
        for _ in range(n):
            file.write(f"{random.randrange(0, 100)}\n")

def main():
    """
    Main function to take user inputs and generate the file with random numbers.
    """
    filename = input('Enter the filename: ')
    n = int(input('Enter the length of the number list: '))
    generate_nums(filename, n)
    print(f"Generated {n} random numbers and saved them to {filename}")

if __name__ == "__main__":
    main()
