def selection_sort(num_list):
    """
    Apply selection sort algorithm on the input list and sort the elements.
    """
    sorted_list = []
    step = 1
    
    while len(num_list) > 0:
        min_value = num_list[0]
        
        # Find the minimum value in the current list
        for number in num_list:
            if number < min_value:
                min_value = number
        
        # Print the current step and the lists
        print(f'\nStep {step}')
        print('Remaining list:', num_list)
        print('Sorted list:', sorted_list)
        
        # Move the minimum value to the sorted list
        sorted_list.append(min_value)
        num_list.remove(min_value)
        
        step += 1
    
    print('\nFinish:')
    print('Final sorted list:', sorted_list)

if __name__ == "__main__":
    numbers = [2, 1, 5, 3, 4, 7, 6, 9, 8, 10]
    selection_sort(numbers)
