def selection_sort(numbers):
    """
    Sorts a list of numbers in ascending order using the selection sort algorithm.
    
    Args:
        numbers (list): The list of numbers to be sorted.
    """
    n = len(numbers)
    for i in range(n - 1):
        # Assume the current position holds the smallest number
        min_position = i
        # Check the rest of the array for a smaller number
        for j in range(i + 1, n):
            if numbers[j] < numbers[min_position]:
                min_position = j
        # Swap the found minimum element with the current element
        numbers[i], numbers[min_position] = numbers[min_position], numbers[i]

def test_selection_sort():
    """
    Test the selection_sort function to ensure it sorts the list correctly.
    """
    test_numbers = [5, 6, 4, 3, 7, 8]
    selection_sort(test_numbers)
    assert test_numbers == [3, 4, 5, 6, 7, 8], f"Test failed: {test_numbers}"
    print("Test passed!")

# Example usage
if __name__ == "__main__":
    numbers = [5, 6, 4, 3, 7, 8]
    print("Original list:", numbers)
    selection_sort(numbers)
    print("Sorted list:", numbers)  # Output: [3, 4, 5, 6, 7, 8]
    
    # Run the test
    test_selection_sort()
