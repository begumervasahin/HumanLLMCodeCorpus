def selection_sort(arr):
    """
    Perform selection sort on the given list.
    """
    n = len(arr)
    # Traverse through all array elements from the end to the start
    for fillslot in range(n - 1, 0, -1):
        # Assume the first element is the largest
        position_of_max = 0
        # Traverse the unsorted subarray to find the maximum element
        for location in range(1, fillslot + 1):
            if arr[location] > arr[position_of_max]:
                position_of_max = location
        # Swap the found maximum element with the element at the fillslot
        arr[fillslot], arr[position_of_max] = arr[position_of_max], arr[fillslot]

def main():
    """
    Main function to demonstrate selection sort.
    """
    # Example list to sort
    alist = [54, 45, 67, 12, 34, 98, 66]
    
    # Print the list before sorting
    print("List before sorting:", alist)
    
    # Sort the list using selection sort
    selection_sort(alist)
    
    # Print the list after sorting
    print("List after sorting:", alist)

if __name__ == "__main__":
    main()
