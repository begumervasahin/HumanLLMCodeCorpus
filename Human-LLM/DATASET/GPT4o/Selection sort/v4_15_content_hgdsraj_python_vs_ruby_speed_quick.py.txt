def sort_array(arr):
    # Get the length of the array
    n = len(arr)
    
    # Perform a simple sort by swapping elements
    for i in range(n - 1):
        for j in range(i + 1, n):
            if arr[j] < arr[i]:
                # Swap the elements if arr[j] is smaller than arr[i]
                arr[i], arr[j] = arr[j], arr[i]
    
    return arr

def main():
    # Initialize the array
    arr = [1, 3, 5, 7, 9, 2, 4, 6, 8, 0]
    
    # Sort the array
    sorted_arr = sort_array(arr)
    
    # Print the sorted array
    print(sorted_arr)
    print("\n")

if __name__ == "__main__":
    main()
