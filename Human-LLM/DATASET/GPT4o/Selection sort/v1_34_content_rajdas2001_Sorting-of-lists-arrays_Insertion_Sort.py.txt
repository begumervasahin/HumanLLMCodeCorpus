def insertion_sort(arr):
    """
    Sorts a list of numbers in ascending order using the insertion sort algorithm.

    Args:
        arr (list): The list of numbers to be sorted.
    """
    for i in range(1, len(arr)):
        item = arr[i]
        j = i - 1
        # Move elements of arr[0..i-1], that are greater than item, to one position ahead of their current position
        while j >= 0 and item < arr[j]:
            arr[j + 1] = arr[j]
            j -= 1
        arr[j + 1] = item

def main():
    # Example list of numbers to be sorted
    arr = [32, 13, 14, 15, 98]
    print("Original array:", arr)
    insertion_sort(arr)
    print("Sorted array is:")
    for i in range(len(arr)):
        print(arr[i], end=' ')

if __name__ == "__main__":
    main()
