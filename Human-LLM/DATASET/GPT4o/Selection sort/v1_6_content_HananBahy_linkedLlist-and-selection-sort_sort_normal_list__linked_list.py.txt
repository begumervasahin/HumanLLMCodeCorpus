from time import time
from random import randrange
from linkedlistuno import linkedlistuno
from selectionsort import selectionsort

# Create a list and a linked list
mylist = []
mylinkedlist = linkedlistuno()

# Populate the list and the linked list with random integers
for i in range(10000):
    j = randrange(-1000, 1001)
    mylist.insert(0, j)
    mylinkedlist.add(j)

# Measure the time it takes to sort the normal list using selection sort
start1 = time()
selectionsort(mylist)
end1 = time()

# Measure the time it takes to sort the linked list using selection sort
start2 = time()
mylinkedlist.selectionsort()
end2 = time()

# Print the time taken for both sorting operations
print("Time for normal list:", end1 - start1)
print("Time for linked list:", end2 - start2)
