def read_values_from_file(filename):
    """
    Read and extract values from a file containing lines with the word "real".
    
    Parameters:
    filename (str): The name of the file to read from.
    
    Returns:
    list: A list of extracted float values.
    """
    with open(filename) as file:
        lines = file.readlines()
    
    values = []
    for line in lines:
        if "real" in line:
            try:
                # Extract the value, convert to float, and scale it
                value = float(line[9:12]) / 1000
                values.append(value)
            except ValueError:
                print(f"Skipping line due to value extraction error: {line.strip()}")
    
    return values

def calculate_average(values):
    """
    Calculate the average of a list of values.
    
    Parameters:
    values (list): A list of float values.
    
    Returns:
    float: The average of the values.
    """
    if not values:
        return 0.0  # Handle the case where the list is empty
    
    total_sum = sum(values)
    return total_sum / len(values)

# Read values from the file
extracted_values = read_values_from_file('cpp_res.txt')

# Calculate the average of the values
average = calculate_average(extracted_values)

# Print the average
print("The average was:")
print(f"{average:.3f} seconds")
