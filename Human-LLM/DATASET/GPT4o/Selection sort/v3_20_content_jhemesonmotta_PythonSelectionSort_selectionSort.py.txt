def selection_sort(arr):
    """Sorts an array using the selection sort algorithm."""
    n = len(arr)
    for last_pos in range(n - 1, 0, -1):
        # Assume the first element is the maximum
        max_pos = 0
        # Find the maximum element in the unsorted portion
        for pos in range(1, last_pos + 1):
            if arr[pos] > arr[max_pos]:
                max_pos = pos
        # Swap the found maximum element with the last element in the unsorted portion
        arr[last_pos], arr[max_pos] = arr[max_pos], arr[last_pos]

def print_array(arr, message):
    """Prints the elements of the array with a given message."""
    print(message, arr)

# Example usage
if __name__ == "__main__":
    # Define the list to be sorted
    arr = [54, 26, 93, 17, 77, 31, 44, 55, 20]
    
    # Print the original list
    print_array(arr, "Original list:")
    
    # Sort the list using selection_sort
    selection_sort(arr)
    
    # Print the sorted list
    print_array(arr, "Sorted list:")
