def selection_sort(arr):
    """Sorts an array using the selection sort algorithm."""
    n = len(arr)
    for i in range(n):
        # Assume the current position holds the smallest element
        min_index = i
        # Find the smallest element in the remaining unsorted portion of the array
        for j in range(i + 1, n):
            if arr[j] < arr[min_index]:
                min_index = j
        # Swap the found minimum element with the element at the current position
        arr[i], arr[min_index] = arr[min_index], arr[i]

def print_array(arr, message):
    """Prints the elements of the array with a given message."""
    print(message, arr)

# Example usage
if __name__ == "__main__":
    # Define the list to be sorted
    arr = [6, 5, 8, 4, 3, 2, 8, 9, 10, 15, 0]
    
    # Print the original list
    print_array(arr, "Original list:")
    
    # Sort the list using selection_sort
    selection_sort(arr)
    
    # Print the sorted list
    print_array(arr, "Sorted list:")
