def selection_sort(arr):
    # Loop through each element in the array, starting from the end
    for last_pos in range(len(arr) - 1, 0, -1):
        # Assume the largest element is the first element
        largest_pos = 0
        
        # Find the largest element in the unsorted portion of the array
        for pos in range(1, last_pos + 1):
            if arr[pos] > arr[largest_pos]:
                largest_pos = pos
        
        # Swap the found largest element with the last unsorted element
        arr[last_pos], arr[largest_pos] = arr[largest_pos], arr[last_pos]

# Define the array to be sorted
arr = [54, 26, 93, 17, 77, 31, 44, 55, 20]

# Call the selection_sort function
selection_sort(arr)

# Print the sorted array
print("Sorted array:", arr)
