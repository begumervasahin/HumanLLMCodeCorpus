def read_and_extract_times(filename):
    # Open the file and read all lines
    with open(filename) as f:
        lines = f.readlines()

    # Initialize an empty list to store the extracted times
    times = []

    # Loop through each line and extract the time if the line contains "real"
    for line in lines:
        if "real" in line:
            # Extract the time value and convert it to float, dividing by 1000
            time_value = float(line.split()[1][:-1]) / 1000
            times.append(time_value)

    return times

def calculate_average(times):
    # Calculate the average of the extracted times
    return sum(times) / len(times) if times else 0

def main():
    # Read and process the file
    times = read_and_extract_times('ruby_res.txt')

    # Calculate the average time
    average_time = calculate_average(times)

    # Print the average time
    print("The average was")
    print(f"{average_time:.3f} seconds")

if __name__ == "__main__":
    main()
