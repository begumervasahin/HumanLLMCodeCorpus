def insertion_sort(arr):
    """Sorts an array using the insertion sort algorithm and returns the number of comparisons and swaps."""
    comparisons = 0
    swaps = 0
    
    for i in range(1, len(arr)):
        value = arr[i]
        j = i - 1
        comparisons += 1
        
        # Move elements of arr[0..i-1], that are greater than value, to one position ahead of their current position
        while j >= 0 and value < arr[j]:
            arr[j + 1] = arr[j]
            j -= 1
            comparisons += 1
            swaps += 1
        
        arr[j + 1] = value
    
    return f"INSERTION SORT:\nComparisons: {comparisons}\nSwaps: {swaps}"

def main():
    arr = [64, 34, 25, 12, 22, 11, 90]
    print("Unsorted array:")
    print(arr)
    
    result = insertion_sort(arr)
    
    print("Sorted array:")
    print(arr)
    print(result)

if __name__ == "__main__":
    main()
