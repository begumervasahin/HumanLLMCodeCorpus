def insertion_sort(array):
    """
    Perform insertion sort on the given array, counting the number of comparisons and swaps.
    """
    swaps_made = 0
    checks_made = 0

    for f in range(1, len(array)):
        value = array[f]
        value_index = f

        # Count the comparison made to enter the while loop
        checks_made += 1
        
        # Shift elements of array[0..f-1], that are greater than value, to one position ahead of their current position
        while value_index > 0 and value < array[value_index - 1]:
            array[value_index] = array[value_index - 1]
            value_index -= 1
            checks_made += 1
            swaps_made += 1
        
        # Place value at its correct position
        array[value_index] = value
    
    return f"INSERTION SORT:\nComparisons: {checks_made}\nSwaps: {swaps_made}"

if __name__ == "__main__":
    arr = [12, 11, 13, 5, 6]
    print("Array before sorting:")
    print(arr)
    
    result = insertion_sort(arr)
    
    print("Sorted array is:")
    print(arr)
    print(result)
