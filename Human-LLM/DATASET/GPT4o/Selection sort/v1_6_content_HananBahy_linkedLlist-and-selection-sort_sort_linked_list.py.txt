class Node:
    def __init__(self, data):
        self.data = data
        self.next = None


class LinkedListUno:
    def __init__(self):
        self.head = None

    def add(self, data):
        """Add a new node with the given data to the start of the linked list."""
        new_node = Node(data)
        new_node.next = self.head
        self.head = new_node

    def print_linked2(self):
        """Print all elements in the linked list."""
        current = self.head
        while current:
            print(current.data, end=" -> ")
            current = current.next
        print("None")

    def selectionsort(self):
        """Sort the linked list using selection sort."""
        current = self.head

        while current:
            smallest = current
            next_node = current.next

            while next_node:
                if next_node.data < smallest.data:
                    smallest = next_node
                next_node = next_node.next

            current.data, smallest.data = smallest.data, current.data
            current = current.next


if __name__ == "__main__":
    mylist = LinkedListUno()
