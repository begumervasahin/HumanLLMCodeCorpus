def insertion_sort(arr):
    """Sorts an array using the insertion sort algorithm."""
    for index in range(1, len(arr)):
        current_value = arr[index]
        position = index

        # Move elements of arr[0..index-1], that are greater than current_value, to one position ahead
        while position > 0 and arr[position - 1] > current_value:
            arr[position] = arr[position - 1]
            position -= 1

        # Place the current_value at its correct position
        arr[position] = current_value

def print_array(arr, message):
    """Prints the elements of the array with a given message."""
    print(message, arr)

# Example usage
if __name__ == "__main__":
    # Define the list to be sorted
    arr = [54, 26, 93, 17, 77, 31, 44, 55, 20]
    
    # Print the original list
    print_array(arr, "Original list:")
    
    # Sort the list using insertion_sort
    insertion_sort(arr)
    
    # Print the sorted list
    print_array(arr, "Sorted list:")
