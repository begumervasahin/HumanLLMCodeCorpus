import random

def partition(arr, low, high):
    """
    Partition the subarray arr[low:high+1] using Hoare's partition scheme.
    """
    pivot = arr[low]
    i = low
    j = high + 1

    while True:
        # Move i to the right until arr[i] >= pivot
        while i < high and arr[i] < pivot:
            i += 1
        
        # Move j to the left until arr[j] <= pivot
        while j > low and arr[j] > pivot:
            j -= 1
        
        # If two indices crossed, partitioning is complete
        if i >= j:
            break
        
        # Swap elements at i and j
        arr[i], arr[j] = arr[j], arr[i]
    
    # Swap pivot element with element at j
    arr[low], arr[j] = arr[j], arr[low]
    
    return j

def quicksort_helper(arr, low, high):
    """
    Recursively apply quicksort to the subarray arr[low:high+1].
    """
    if low < high:
        # Partition the array and get the partition index
        pivot_index = partition(arr, low, high)
        
        # Recursively apply quicksort to the left and right subarrays
        quicksort_helper(arr, low, pivot_index - 1)
        quicksort_helper(arr, pivot_index + 1, high)

def quicksort(arr):
    """
    Perform quicksort on the entire array.
    """
    quicksort_helper(arr, 0, len(arr) - 1)

def main():
    """
    Main function to demonstrate quicksort.
    """
    # Generate a random list of integers
    arr = [random.randint(1, 100) for _ in range(10)]
    
    # Print the array before sorting
    print("Array before sorting:", arr)
    
    # Sort the array using quicksort
    quicksort(arr)
    
    # Print the array after sorting
    print("Array after sorting:", arr)

if __name__ == "__main__":
    main()
