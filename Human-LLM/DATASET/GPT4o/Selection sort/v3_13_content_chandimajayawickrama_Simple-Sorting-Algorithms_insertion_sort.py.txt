import time
import random

def insertion_sort(array):
    """
    Perform an insertion sort on the given array.
    
    Parameters:
    array (list): The list of elements to be sorted.
    
    Returns:
    tuple: A tuple containing the sorted list and the time taken to sort the list.
    """
    start_time = time.time()  # Record the start time of the sorting process
    
    # Traverse through array elements starting from the second element
    for i in range(1, len(array)):
        key = array[i]  # Set the current element as the key
        j = i - 1
        
        # Move elements of array[0..i-1] that are greater than the key
        # to one position ahead of their current position
        while j >= 0 and key < array[j]:
            array[j + 1] = array[j]
            j -= 1
        
        array[j + 1] = key  # Place the key at the correct position
    
    end_time = time.time()  # Record the end time of the sorting process
    elapsed_time = end_time - start_time  # Calculate the time taken to sort the array
    
    return array, elapsed_time

def generate_random_list(size, lower_bound=0, upper_bound=100):
    """
    Generate a random list of integers within the specified range.
    
    Parameters:
    size (int): The number of elements in the list.
    lower_bound (int): The minimum possible value for the random integers. Default is 0.
    upper_bound (int): The maximum possible value for the random integers. Default is 100.
    
    Returns:
    list: A list of random integers.
    """
    return [random.randint(lower_bound, upper_bound) for _ in range(size)]

# Generate a random list of 10 integers
random_list = generate_random_list(10)

# Print the original list
print("Original list:", random_list)

# Sort the list using insertion sort
sorted_list, elapsed_time = insertion_sort(random_list)

# Print the sorted list and the time taken to sort
print("Sorted list:", sorted_list)
print("Time taken to sort:", elapsed_time, "seconds")
