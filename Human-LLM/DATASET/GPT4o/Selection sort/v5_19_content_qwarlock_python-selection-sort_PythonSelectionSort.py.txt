def selection_sort(arr):
    """Sort the array using the selection sort algorithm."""
    n = len(arr)
    
    for i in range(n):
        # Assume the smallest number is the first unsorted element
        smallest_index = i
        
        # Find the smallest element in the remaining unsorted portion
        for j in range(i + 1, n):
            if arr[j] < arr[smallest_index]:
                smallest_index = j
        
        # Swap the found smallest element with the first unsorted element
        arr[i], arr[smallest_index] = arr[smallest_index], arr[i]

def main():
    # Define the array to be sorted
    arr = [6, 5, 8, 4, 3, 2, 8, 9, 10, 15, 0]
    
    # Print the array before sorting
    print("Array before sorting:", arr)
    
    # Call the selection_sort function
    selection_sort(arr)
    
    # Print the array after sorting
    print("Array after sorting:", arr)

if __name__ == "__main__":
    main()
