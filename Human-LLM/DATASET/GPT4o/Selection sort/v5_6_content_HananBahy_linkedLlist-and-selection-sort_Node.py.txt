class Node:
    def __init__(self, data):
        """
        Initialize a new node with the given data.
        
        Args:
            data: The initial data for the node.
        """
        self.data = data
        self.next = None

    def get_data(self):
        """
        Get the data stored in the node.
        
        Returns:
            The data stored in the node.
        """
        return self.data

    def get_next(self):
        """
        Get the next node.
        
        Returns:
            The next node.
        """
        return self.next

    def set_data(self, data):
        """
        Set the data stored in the node.
        
        Args:
            data: The new data to store in the node.
        """
        self.data = data

    def set_next(self, next_node):
        """
        Set the next node.
        
        Args:
            next_node: The new next node.
        """
        self.next = next_node
