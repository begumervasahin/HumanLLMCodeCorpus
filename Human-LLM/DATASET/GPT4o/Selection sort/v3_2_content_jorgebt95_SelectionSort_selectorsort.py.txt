def selection_sort(numbers):
    """
    Sorts the list of numbers in ascending order using the selection sort algorithm.

    Args:
        numbers (list): The list of numbers to be sorted.
    """
    n = len(numbers)
    for i in range(n):
        min_position = i
        for j in range(i + 1, n):
            if numbers[j] < numbers[min_position]:
                min_position = j
        # Swap the found minimum element with the first element
        numbers[i], numbers[min_position] = numbers[min_position], numbers[i]

def test_selection_sort():
    """
    Test function for selection_sort to ensure it sorts the list correctly.
    """
    numbers = [5, 6, 4, 3, 7, 8]
    expected_result = [3, 4, 5, 6, 7, 8]
    
    selection_sort(numbers)
    
    assert numbers == expected_result, f"Expected {expected_result} but got {numbers}"
    print("Test passed!")

if __name__ == "__main__":
    test_selection_sort()
