def selection_sort(nums):
    """
    Sorts a list of numbers in ascending order using the selection sort algorithm.

    Args:
        nums (list): The list of numbers to be sorted.
    """
    n = len(nums)
    # Traverse through all list elements
    for i in range(n - 1):
        # Find the minimum element in the remaining unsorted part of the list
        min_pos = i
        for j in range(i + 1, n):
            if nums[j] < nums[min_pos]:
                min_pos = j
        # Swap the found minimum element with the first element
        nums[i], nums[min_pos] = nums[min_pos], nums[i]
        # Print the list after each swap (for debugging/understanding purposes)
        print(f"Step {i + 1}: {nums}")

def main():
    # Example list of numbers to be sorted
    nums = [5, 3, 7, 2, 4, 1, 11, 8, 10, 9, 6]
    print("Original list:", nums)
    selection_sort(nums)
    print("Sorted list:", nums)

if __name__ == "__main__":
    main()
