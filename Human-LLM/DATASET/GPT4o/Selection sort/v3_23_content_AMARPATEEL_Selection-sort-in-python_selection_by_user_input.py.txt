def selection_sort(arr):
    """
    Perform selection sort on the given list and print the process.
    """
    n = len(arr)
    for i in range(n):
        # Assume the minimum element is the first element of the unsorted part
        min_index = i
        print(f"Current minimum is: {arr[min_index]} at index {min_index}")
        
        # Find the minimum element in the remaining unsorted array
        for j in range(i + 1, n):
            if arr[j] < arr[min_index]:
                min_index = j
        
        # Swap the found minimum element with the first element of the unsorted part
        arr[i], arr[min_index] = arr[min_index], arr[i]
        print(f"Swapped {arr[min_index]} with {arr[i]}")
        print("Array state:", arr)
        print()

def get_user_input():
    """
    Get a list of integers from the user.
    """
    num_elements = int(input("Number of elements in the list: "))
    arr = []

    # Collect the elements from the user
    for _ in range(num_elements):
        element = int(input("Enter an element: "))
        arr.append(element)
    
    return arr

def main():
    """
    Main function to get user input, sort the list, and display the process.
    """
    # Get user input
    arr = get_user_input()
    
    # Print the list before sorting
    print("List before sorting:", arr)
    print()
    
    # Sort the list using selection sort
    selection_sort(arr)
    
    # Print the list after sorting
    print("List after sorting:", arr)

if __name__ == "__main__":
    main()
