import random

def random_number():
    """
    Generates a list of 100 unique random numbers in the range from 0 to 100000.

    Returns:
        list: A list of 100 unique random numbers.
    """
    return random.sample(range(100001), 100)

def selection_sort(li):
    """
    Sorts the input list using the selection sort algorithm.

    Args:
        li (list): The list of integers to be sorted.
    
    Returns:
        list: The sorted list.
    """
    print("Original list:")
    print(li)
    
    n = len(li)
    for i in range(n):
        min_index = i
        for j in range(i+1, n):
            if li[j] < li[min_index]:
                min_index = j
        # Swap the found minimum element with the first element
        li[i], li[min_index] = li[min_index], li[i]
    
    print("Sorted list:")
    print(li)
    return li

if __name__ == "__main__":
    random_list = random_number()
    selection_sort(random_list)
