def selection_sort(alist):
    # Traverse through all elements except the last one
    for j in range(len(alist) - 1):
        # Assume the smallest element is the first element of the unsorted portion
        smallest_index = j
        
        # Traverse the unsorted portion of the list
        for i in range(j + 1, len(alist)):
            # Update the smallest_index if a smaller element is found
            if alist[i] < alist[smallest_index]:
                smallest_index = i
        
        # Swap the found smallest element with the first element of the unsorted portion
        alist[j], alist[smallest_index] = alist[smallest_index], alist[j]
