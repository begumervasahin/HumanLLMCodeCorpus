import os
import sys

def main():
    """
    The main function to set up the Django settings module and execute the Django management commands.
    """
    # Set the default Django settings module environment variable
    os.environ.setdefault("DJANGO_SETTINGS_MODULE", "selection_sort.settings")

    try:
        # Import execute_from_command_line to execute Django management commands
        from django.core.management import execute_from_command_line
    except ImportError as import_error:
        try:
            # Attempt to import Django to provide a more specific error message
            import django
        except ImportError:
            raise ImportError(
                "Couldn't import Django. Are you sure it's installed and "
                "available on your PYTHONPATH environment variable? Did you "
                "forget to activate a virtual environment?"
            ) from import_error
        raise import_error

    # Execute the command line arguments passed to the script
    execute_from_command_line(sys.argv)

if __name__ == "__main__":
    main()
