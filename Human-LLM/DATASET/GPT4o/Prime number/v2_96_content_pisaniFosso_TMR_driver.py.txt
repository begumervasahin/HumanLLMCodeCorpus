from collections import defaultdict
import operator

class Driver:
    def __init__(self, modules):
        self._modules = modules
        self._intervals = []
        self._values = []
        self._print_module_headers()

    def _print_module_headers(self):
        """Print the headers for the module results."""
        print('Interval\t|\t', end='')
        for module in self._modules:
            print(f'{module}\t', end='')
        print('Winner')

    def set_module(self, index, value=None, interval=None):
        """Set the value or interval for a specific module."""
        if index < len(self._modules):
            if value is not None:
                self._modules[index]._lastValue = value
            if interval is not None:
                self._modules[index]._lastInterval = interval

    def test(self, intervals):
        """Test the modules over a list of intervals."""
        for interval in intervals:
            self._process_interval(interval)

    def run(self, interval):
        """Run a single interval test."""
        self._process_interval(interval)

    def _process_interval(self, interval):
        """Process a single interval and determine the winner."""
        results = [module.getResult(interval) for module in self._modules]
        votes = defaultdict(int)
        print(f'{interval}\t|\t', end='')
        for result in results:
            print(f'{result}\t', end='')
            votes[result] += 1
        winner = max(votes.items(), key=operator.itemgetter(1))[0]
        print(winner)
        self._values.append(winner)
        self._intervals.append(interval)
        self._provide_feedback(results, winner)

    def _provide_feedback(self, results, winner):
        """Provide feedback to the modules based on the results."""
        for i, module in enumerate(self._modules):
            feedback = {'status': 'ok'} if results[i] == winner else {'status': 'error', 'goodValue': winner}
            module.receiveFeedback(feedback)
