"""
Calculates the prime days of a specified day for a specified number of years.
"""
import os
import sys
import datetime
from prime import is_day_prime  # Assuming you have a function is_day_prime in prime module

def increment_day(date):
    """
    Returns the next valid date given a datetime.date object.
    
    Args:
        date (datetime.date): The current date.
    
    Returns:
        datetime.date: The next date.
    """
    year, month, day = date.year, date.month, date.day
    try:
        return datetime.date(year, month, day + 1)
    except ValueError:
        try:
            return datetime.date(year, month + 1, 1)
        except ValueError:
            return datetime.date(year + 1, 1, 1)

def num_prime_days(start_date, num_years):
    """
    Calculates the number of prime days on a specified day over a number of years.
    
    Args:
        start_date (datetime.date): The starting date.
        num_years (int): The number of years to check.
    """
    num_days = 0
    count = 0
    start_year = start_date.year
    
    while count < num_years:
        if is_day_prime(start_date):
            num_days += 1
        
        try:
            start_date = datetime.date(start_year + count + 1, start_date.month, start_date.day)
        except ValueError:
            if (start_year + count) % 4 == 0:
                if (start_year + count + 4) % 100 == 0 and (start_year + count + 4) % 400 != 0:
                    start_date = datetime.date(start_year + count + 8, start_date.month, start_date.day)
                    count += 7
                else:
                    start_date = datetime.date(start_year + count + 4, start_date.month, start_date.day)
                    count += 3
        
        count += 1
    
    write_file(start_date, num_days)
