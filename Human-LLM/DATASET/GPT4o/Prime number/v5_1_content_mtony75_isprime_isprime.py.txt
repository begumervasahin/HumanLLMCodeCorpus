def get_integer(prompt):
    """
    Prompt the user for input and ensure it is an integer.
    """
    while True:
        user_input = input(prompt)
        try:
            return int(user_input)
        except ValueError:
            print("The value entered is not an integer. Please enter a valid integer.")

def is_prime(number):
    """
    Check if a number is prime.
    """
    if number <= 1:
        return False
    if number == 2:
        return True
    for element in range(2, int(number ** 0.5) + 1):
        if number % element == 0:
            return False
    return True

def get_prime_range():
    """
    Get a range of numbers from the user and find prime numbers within that range.
    """
    print("Provide a range of numbers to find out which numbers in the range are prime numbers.")

def find_primes_in_range(start, end):
    """
    Find all prime numbers in a given range.
    """
    return [num for num in range(start, end + 1) if is_prime(num)]

def find_next_hundred_primes(start_number):
    """
    Find the next 100 prime numbers after the given start number.
    """
    primes = []
    candidate = start_number + 1
    while len(primes) < 100:
        if is_prime(candidate):
            primes.append(candidate)
        candidate += 1
    return primes
