def find_primes_between(bounds):
    """
    Find and format prime numbers between two bounds.

    Args:
        bounds (list): A list containing two integer bounds.

    Returns:
        str: A formatted string of prime numbers between the bounds.
    """
    lower_bound, upper_bound = sorted(bounds)
    primes = []

    for x in range(lower_bound + 1, upper_bound):
        if x > 1 and all(x % i != 0 for i in range(2, int(x ** 0.5) + 1)):
            primes.append(x)

    if not primes:
        return "No Primes"

    for i in range(0, len(primes), 3):
        primes[i] = str(primes[i]) + ":"
    for i in range(1, len(primes), 3):
        primes[i] = str(primes[i]) + "!"
    for i in range(2, len(primes), 3):
        primes[i] = str(primes[i]) + ","

    return "".join(primes)

def main():
    try:
        bound_1 = int(input("Enter first number: "))
        bound_2 = int(input("Enter second number: "))
        bounds = [bound_1, bound_2]
        result = find_primes_between(bounds)
        print(result)
    except ValueError:
        print("Please enter valid integers.")

if __name__ == "__main__":
    main()
