import math

def mersenne_number(p):
    """Returns the Mersenne number for a given p."""
    return (2 ** p) - 1

def is_prime(number):
    """Checks if a number is prime."""
    if number <= 1:
        return False
    for i in range(2, int(math.sqrt(number)) + 1):
        if number % i == 0:
            return False
    return True

def get_primes(a, b):
    """Returns a list of prime numbers in the range [a, b)."""
    return [j for j in range(a, b) if is_prime(j)]

def lucas_lehmer(p):
    """Generates the Lucas-Lehmer sequence for a given p."""
    lucas_lehmer_sequence = [4] * (p - 1)
    for i in range(1, p - 1):
        lucas_lehmer_sequence[i] = (((lucas_lehmer_sequence[i - 1]) ** 2) - 2) % ((2 ** p) - 1)
    return lucas_lehmer_sequence

def lucas_lehmer_prime(p):
    """Performs the Lucas-Lehmer test for primality of Mersenne numbers."""
    lucas_lehmer_sequence = lucas_lehmer(p)
    return lucas_lehmer_sequence[-1] == 0

def is_prime_fast(number):
    """Checks if a number is prime using a more optimized method."""
    if number <= 1:
        return False
    if number % 2 == 0:
        return number == 2
    for factor in range(3, int(math.sqrt(number)) + 1, 2):
        if number % factor == 0:
            return False
    return True

def get_primes_fast(n):
    """Returns a list of prime numbers up to n using the optimized prime check."""
    return [number for number in range(n) if is_prime_fast(number)]

def list_true(n):
    """Returns a list of boolean values where the index represents if it is initially considered prime."""
    return [False, False] + [True] * (n - 1)

def mark_false(bool_list, p):
    """Marks non-prime indices in the boolean list."""
    for i in range(2, len(bool_list)):
        if p * i < len(bool_list):
            bool_list[p * i] = False
    return bool_list

def find_next(bool_list, p):
    """Finds the next true value in the boolean list after index p."""
    for i in range(p + 1, len(bool_list)):
        if bool_list[i]:
            return i
    return None

def prime_from_list(bool_list):
    """Returns a list of prime numbers from the boolean list."""
    return [i for i, is_prime in enumerate(bool_list) if is_prime]

def sieve(n):
    """Performs the Sieve of Eratosthenes to find all primes up to n."""
    bool_list = list_true(n)
    p = 2
    while p is not None:
        bool_list = mark_false(bool_list, p)
        p = find_next(bool_list, p)
    return prime_from_list(bool_list)
