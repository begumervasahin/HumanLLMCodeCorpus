def is_valid(number):
    """
    Check if the input is a valid integer.
    
    Args:
    number (str): The input to check.

    Returns:
    bool: True if the input is a valid integer, False otherwise.
    """
    try:
        int(number)
        return True
    except ValueError:
        print("User input is not an integer.")
        return False

def is_prime(number):
    """
    Determine if a number is prime.
    
    Args:
    number (int): The number to check.

    Returns:
    bool: True if the number is prime, False otherwise.
    """
    if number < 2:
        print(f"{number} is not a prime number.")
        return False
    for i in range(2, int(number ** 0.5) + 1):
        if number % i == 0:
            print(f"{number} is not a prime number.")
            return False
    print(f"{number} is a prime number.")
    return True

def main():
    """
    Main function to get user input and check for prime numbers.
    """
    while True:
        user_input = input("Enter a number (or type 'stop' to end): ").strip()
        if user_input.lower() == 'stop':
            break
        if is_valid(user_input):
            number = int(user_input)
            is_prime(number)

if __name__ == "__main__":
    main()
