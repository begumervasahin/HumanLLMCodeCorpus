import time
import math

def get_elapsed_time(start_time):
    """Calculate and return the elapsed time since start_time."""
    return time.time() - start_time

def check_prime(number):
    """
    Check if a number is a prime number.
    """
    if number <= 1:
        return False
    if number == 2:
        return True  # 2 is the only even prime number
    if number % 2 == 0:
        return False  # Even numbers greater than 2 are not prime
    for num in range(3, int(math.sqrt(number)) + 1, 2):
        if number % num == 0:
            return False
    return True

def is_left_truncatable_prime(num):
    """
    Check if a number is a left-truncatable prime.
    """
    str_num = str(num)
    while str_num:
        if not check_prime(int(str_num)):
            return False
        str_num = str_num[1:]
    return True

def is_right_truncatable_prime(num):
    """
    Check if a number is a right-truncatable prime.
    """
    str_num = str(num)
    while str_num:
        if not check_prime(int(str_num)):
            return False
        str_num = str_num[:-1]
    return True

def main():
    number = int(input('Enter a number to check if it is truncatable: \n-> '))
    
    start_time = time.time()
    
    if number <= 1:
        print('The number must be greater than 1.')
        print(f'Elapsed time: {get_elapsed_time(start_time):.2f} seconds')
        return
    
    if not check_prime(number):
        print(f"{number} is not a prime number.")
        print(f'Elapsed time: {get_elapsed_time(start_time):.2f} seconds')
        return
    
    if '0' in str(number):
        print(f"{number} is not truncatable because it contains a '0'.")
        print(f'Elapsed time: {get_elapsed_time(start_time):.2f} seconds')
        return
    
    if is_left_truncatable_prime(number):
        print(f'{number} is a left-truncatable prime.')
    else:
        print(f'{number} is not a left-truncatable prime.')
    
    if is_right_truncatable_prime(number):
        print(f'{number} is a right-truncatable prime.')
    else:
        print(f'{number} is not a right-truncatable prime.')
    
    print(f"Total elapsed time: {get_elapsed_time(start_time):.2f} seconds")

if __name__ == "__main__":
    main()
