def find_special_numbers(n):
    """
    Find and print a sorted list of special numbers based on a specific condition.

    Args:
        n (int): The upper limit for the special number search.
    """
    special_numbers = []
    i = 1
    
    # Generate potential special numbers
    while True:
        b = (24 * i + 1) ** 0.5
        if b % 1 == 0 and int(b) < n:
            special_numbers.append(int(b))
        elif int(b) > n:
            break
        i += 1

    # Remove numbers divisible by 2 to 9
    for i in range(2, 10):
        special_numbers = [num for num in special_numbers if num % i != 0]

    # Remove squares of the numbers in the list
    squares = [y ** 2 for y in special_numbers]
    special_numbers = [num for num in special_numbers if num not in squares]

    # Add fixed prime numbers
    special_numbers += [2, 3, 5, 7]

    # Print the sorted list of special numbers
    print(sorted(special_numbers))

def main():
    try:
        n = int(input('Enter the end of the range: '))
        find_special_numbers(n)
    except ValueError:
        print("Please enter a valid integer.")

if __name__ == "__main__":
    main()
