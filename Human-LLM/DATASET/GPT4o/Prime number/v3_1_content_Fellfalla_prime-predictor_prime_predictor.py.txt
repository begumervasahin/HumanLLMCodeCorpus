import tensorflow as tf
import numpy as np

# Dictionary to store known prime states
prime_states = {
    2: True,
    3: True,
    4: False
}

def is_prime(given_number):
    """
    Check if a number is prime and store the result in prime_states.

    Args:
        given_number (int): The number to check for primality.

    Returns:
        bool: True if the number is prime, False otherwise.
    """
    if given_number not in prime_states:
        prime_states[given_number] = True
        for num in range(2, int(given_number ** 0.5) + 1):
            if given_number % num == 0:
                prime_states[given_number] = False
                break
    return prime_states[given_number]

def get_next_prime(x):
    """
    Get the next prime number greater than or equal to x.

    Args:
        x (int): The starting number to find the next prime.

    Returns:
        int: The next prime number greater than or equal to x.
    """
    while not is_prime(x):
        x += 1
    return x

def build_and_compile_model():
    """
    Build and compile a TensorFlow model for predicting prime numbers.

    Returns:
        tf.keras.Sequential: The compiled model.
    """
    model = tf.keras.Sequential([
        tf.keras.layers.Dense(24, activation='relu', input_dim=1),
        tf.keras.layers.Dense(24, activation='relu'),
        tf.keras.layers.Dense(24, activation='relu'),
        tf.keras.layers.Dense(24, activation='relu'),
        tf.keras.layers.Dense(24, activation='relu'),
        tf.keras.layers.Dense(1)
    ])
    model.compile(optimizer=tf.keras.optimizers.Adam(), loss=tf.keras.losses.mae)
    model.summary()
    return model

def train_model(model, batch_size, steps):
    """
    Train the model on batches of data.

    Args:
        model (tf.keras.Sequential): The model to train.
        batch_size (int): The batch size for training.
        steps (int): The number of steps to train for.
    """
    for step in range(steps):
        x = np.arange(step * batch_size, (step + 1) * batch_size).reshape(-1, 1)
        y = np.array([get_next_prime(n) for n in x.flatten()])
        loss = model.train_on_batch(x, y)
        print(f"Step {step + 1}/{steps}, Loss: {loss}")

def test_model(model, start, num_tests):
    """
    Test the model and print predictions compared to expected values.

    Args:
        model (tf.keras.Sequential): The trained model.
        start (int): The starting number for testing.
        num_tests (int): The number of test cases.
    """
    for i in range(start, start + num_tests):
        x = np.array([[i]])
        expected = get_next_prime(i)
        predicted = model.predict(x)
        print(f"input: {i}\texpected: {expected}\tpredicted: {int(predicted[0][0])}")

if __name__ == "__main__":
    # Find and print the next prime number greater than or equal to 500
    a = get_next_prime(500)
    print(a)

    # Build and compile the model
    prime_model = build_and_compile_model()

    # Training parameters
    batch_size = 64
    steps = 500

    # Train the model
    train_model(prime_model, batch_size, steps)

    # Test the model
    test_model(prime_model, steps * batch_size, 50)
