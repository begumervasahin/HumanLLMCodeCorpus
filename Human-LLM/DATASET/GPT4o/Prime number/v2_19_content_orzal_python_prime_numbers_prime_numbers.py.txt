def check_prime_numbers(limit):
    """
    Check and print prime numbers up to a given limit.
    
    :param limit: The upper limit for checking prime numbers.
    """
    for n in range(2, limit):
        for x in range(2, n):
            if n % x == 0:
                print(f"{n} equals {x} * {n // x}")
                break
        else:
            print(f"{n} is a prime number")

# Example usage
check_prime_numbers(100)
