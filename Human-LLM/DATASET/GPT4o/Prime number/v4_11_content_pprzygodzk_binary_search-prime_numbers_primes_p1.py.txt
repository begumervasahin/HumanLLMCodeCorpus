import math

def sieve_of_eratosthenes(n):
    """Determine prime numbers from 2 to n using the Sieve of Eratosthenes."""
    numbers = list(range(2, n+1))
    for i in numbers:
        for j in range(i*i, n+1, i):
            if j in numbers:
                numbers.remove(j)
    return numbers

def primes_generator(n):
    """Generate the first n prime numbers using trial division."""
    primes = []
    p = 2  # Starting prime number
    i = 1  # Prime counter
    k = 1  # Used to generate potential primes
    r = 1  # Switch between 6k ± 1
    while i <= n:
        if p == 2 or p == 3:
            primes.append(p)
            i += 1
            p = p+1 if p == 2 else p+2
            continue
        
        for d in range(2, int(math.sqrt(p))+1):
            if p % d == 0:
                break
            if p % d != 0 and d == int(math.sqrt(p)):
                primes.append(p)
                i += 1

        k = k+1 if p == 6*k+1 else k
        r = 1-r
        p = 6*k+1 if r == 0 else 6*k-1
    return primes

def primorial_generator(n):
    """Generate the first n primorial numbers (product of the first n primes)."""
    primes = primes_generator(n-1)
    primorials = [1]
    product = 1
    for prime in primes:
        product *= prime
        primorials.append(product)
    return primorials

def check_primality(n):
    """Check if a number n is prime using trial division."""
    assert n >= 2, "The smallest prime number is 2!"
    if n % 2 == 0 and n != 2:
        return "composite"
    for d in range(3, int(math.sqrt(n))+1, 2):
        if n % d == 0:
            return "composite"
    return "prime"

def decompose(n):
    """Decompose a number n into its prime factors."""
    if n == 1 or check_primality(n) == "prime":
        return [n]
    primes = sieve_of_eratosthenes(math.ceil(n**0.5))
    factors = []
    i = 0
    while primes[i] * primes[i] <= n:
        if n % primes[i] == 0:
            n //= primes[i]
            factors.append(primes[i])
        else:
            i += 1
    factors.append(n)
    return factors
