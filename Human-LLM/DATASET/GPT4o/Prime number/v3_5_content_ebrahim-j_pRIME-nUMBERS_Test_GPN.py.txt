import unittest

def is_prime(n):
    """
    Check if a number is prime.

    Args:
        n (int): The number to check.

    Returns:
        bool: True if the number is prime, False otherwise.
    """
    if n <= 1:
        return False
    for i in range(2, int(n ** 0.5) + 1):
        if n % i == 0:
            return False
    return True

def generate_prime(n):
    """
    Generate a prime number or handle invalid input.

    Args:
        n (int, float, str): The input to check.

    Returns:
        str: A message indicating whether the input is invalid, not a prime, or the prime number itself.
    """
    if isinstance(n, (str, float)):
        return 'Cannot input anything besides a whole number'
    if n <= 1:
        return "Retry with a positive integer greater than 1"
    return n if is_prime(n) else "Not a prime number"

class TestIsPrime(unittest.TestCase):
    def test_for_zero(self):
        """Test to check if 0 returns False"""
        self.assertFalse(is_prime(0))

    def test_negative_numbers(self):
        """Test to check if a negative number is NOT determined prime"""
        self.assertFalse(is_prime(-1))

    def test_one_is_prime(self):
        """Test to confirm if 1 is treated as non-prime"""
        self.assertFalse(is_prime(1))

    def test_prime_number(self):
        """Test if 13 is correctly considered as a prime number"""
        self.assertTrue(is_prime(13))

    def test_big_prime_number(self):
        """Test to check if a big prime number is successfully determined to be prime"""
        self.assertTrue(is_prime(373))

class TestGeneratePrime(unittest.TestCase):
    def test_string_type(self):
        """Test to check if user input was a string"""
        self.assertEqual(generate_prime('i'), 'Cannot input anything besides a whole number')

    def test_float_type(self):
        """Test to check if user input was a float"""
        self.assertEqual(generate_prime(2.58), 'Cannot input anything besides a whole number')

    def test_negative_input(self):
        """Test to check if a negative number was input"""
        self.assertEqual(generate_prime(-2), "Retry with a positive integer greater than 1")

    def test_zero_input(self):
        """Test to check if zero was input"""
        self.assertEqual(generate_prime(0), "Retry with a positive integer greater than 1")

    def test_input_one(self):
        """Test to check if int 1 was input"""
        self.assertEqual(generate_prime(1), "Retry with a positive integer greater than 1")

    def test_valid_prime(self):
        """Test to check if a valid prime number is returned correctly"""
        self.assertEqual(generate_prime(13), 13)

    def test_valid_non_prime(self):
        """Test to check if a non-prime number is correctly identified"""
        self.assertEqual(generate_prime(10), "Not a prime number")

if __name__ == '__main__':
    unittest.main()
